#include <iostream>
#include <algorithm>
#include <queue>
using namespace std;

int n;
vector <pair<int, int> > adj[10000];

//dijkstra
void dijkstra() {
  priority_queue<pair<int, int> > PQ;
  int d[10000];
  for ( int i = 0; i < n; i++ ) {
    d[i] = 10000000;
  }
  d[0] = 0;
  PQ.push(make_pair(0,0));

  while ( !PQ.empty() ) {
    pair<int, int> f = PQ.top(); PQ.pop();
    int u = f.second;
    if ( d[u] < f.first * -1 ) continue;

    for ( int j = 0; j < adj[u].size(); j++ ) {
      int v = adj[u][j].first;
      if ( d[v] > d[u] + adj[u][j].second ) {
	d[v] = d[u] + adj[u][j].second;
	PQ.push(make_pair(d[v] * -1,v));
      }
    }
  }

  //出力
  for ( int i = 0; i < n; i++ ) {
    cout << i << " " << d[i] << endl;
  }
  
}

int main() {
  int k,u,v,c;
  
  //入力
  cin >> n;
  for ( int i = 0; i < n; i++ ) {
    cin >> u >> k;
    for ( int j = 0; j < k; j++ ) {
      cin >> v >> c;
      adj[u].push_back(make_pair(v,c));
    }
  }

  dijkstra();

  return 0;
}
