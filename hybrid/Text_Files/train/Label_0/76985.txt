t = int(input())
for i in range(t):
	s = input()
	(k, n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	p = 0
	q = 0
	l = []
	f = 1
	while True:
		if p < n and a[p] <= k:
			if a[p] == 0:
				k = k + 1
			l.append(a[p])
			p = p + 1
		elif q < m and b[q] <= k:
			if b[q] == 0:
				k = k + 1
			l.append(b[q])
			q = q + 1
		else:
			break
	if p < n or q < m:
		print(-1)
	else:
		print(*l, end=' ')
	print()
