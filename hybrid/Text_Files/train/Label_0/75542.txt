n = int(input())
i = 0
sum = 0
if n >= 1 and n <= 200000:
	while i < n:
		a = str(input())
		if a == 'Icosahedron':
			sum += 20
		elif a == 'Dodecahedron':
			sum += 12
		elif a == 'Octahedron':
			sum += 8
		elif a == 'Cube':
			sum += 6
		elif a == 'Tetrahedron':
			sum += 4
		else:
			print('Вы допустили опечатку в написании...')
		i += 1
print(sum)
