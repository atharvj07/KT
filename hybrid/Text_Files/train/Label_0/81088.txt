def merge(x, y, c):
	N = len(x)
	n = N // 2
	ans = ''
	(i, j) = (0, 0)
	while i < N and j < N:
		if x[i] == y[j]:
			ans += x[i]
			i += 1
			j += 1
		elif x[i] != c:
			ans += x[i]
			i += 1
		else:
			ans += y[j]
			j += 1
	ans += x[i:] + y[j:]
	ans += '0' * (3 * n - len(ans))
	return ans
t = int(input())
for _ in range(t):
	n = int(input())
	S = [input(), input(), input()]
	C0 = [j for j in range(3) if S[j].count('0') >= n]
	if len(C0) >= 2:
		print(merge(S[C0[0]], S[C0[1]], '0'))
	else:
		C1 = [j for j in range(3) if j not in C0]
		print(merge(S[C1[0]], S[C1[1]], '1'))
