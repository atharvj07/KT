#include<stdio.h>
#include<algorithm>
#include<math.h>
using namespace std;
const double EPS = 1e-10;
const double INF = 1e+10;
const double PI = acos(-1);
int sig(double r) { return (r < -EPS) ? -1 : (r > +EPS) ? +1 : 0; }
inline double ABS(double a){return max(a,-a);}
struct Pt {
	double x, y;
	Pt() {}
	Pt(double x, double y) : x(x), y(y) {}
	Pt operator+(const Pt &a) const { return Pt(x + a.x, y + a.y); }
	Pt operator-(const Pt &a) const { return Pt(x - a.x, y - a.y); }
	Pt operator*(const Pt &a) const { return Pt(x * a.x - y * a.y, x * a.y + y * a.x); }
	Pt operator-() const { return Pt(-x, -y); }
	Pt operator*(const double &k) const { return Pt(x * k, y * k); }
	Pt operator/(const double &k) const { return Pt(x / k, y / k); }
	double ABS() const { return sqrt(x * x + y * y); }
	double abs2() const { return x * x + y * y; }
	double arg() const { return atan2(y, x); }
	double dot(const Pt &a) const { return x * a.x + y * a.y; }
	double det(const Pt &a) const { return x * a.y - y * a.x; }
};
double tri(const Pt &a, const Pt &b, const Pt &c) { return (b - a).det(c - a); }
int iSP(Pt a, Pt b, Pt c) {
	int s = sig((b - a).det(c - a));
	if (s) return s;
	if (sig((b - a).dot(c - a)) < 0) return -2; // c-a-b
	if (sig((a - b).dot(c - b)) < 0) return +2; // a-b-c
	return 0;
}
int iLL(Pt a, Pt b, Pt c, Pt d) {
	if (sig((b - a).det(d - c))) return 1; // intersect
	if (sig((b - a).det(c - a))) return 0; // parallel
	return -1; // correspond
}
Pt pLL(Pt a, Pt b, Pt c, Pt d) {
	b = b - a; d = d - c; return a + b * (c - a).det(d) / b.det(d);
}

pair<int,double> dp[1<<10][11][11];
Pt p[12];
int main(){
	int a;
	while(scanf("%d",&a),a){
		for(int i=0;i<a;i++){
			double x,y;
			scanf("%lf%lf",&x,&y);
			p[i]=Pt(x,y);
		}
		for(int i=0;i<(1<<a);i++)for(int j=0;j<a;j++)for(int k=0;k<a;k++)
			dp[i][j][k]=make_pair(999999999,999999999);
		for(int i=0;i<a;i++){
			for(int j=0;j<a;j++){
				if(i==j)continue;
				int to=(1<<i)+(1<<j);
				for(int k=0;k<a;k++){
					if(iSP(p[i],p[k],p[j])==2)to|=(1<<k);
				}
				dp[to][i][j]=make_pair(0,(p[i]-p[j]).ABS());
			}
		}
		for(int i=0;i<(1<<a);i++){
			for(int j=0;j<a;j++)for(int k=0;k<a;k++){
				if(dp[i][j][k].first>99999)continue;
				for(int l=0;l<a;l++){
					int to=i|(1<<l);
					for(int m=0;m<a;m++){
						if(iSP(p[k],p[m],p[l])==2)to|=(1<<m);
					}
					dp[to][k][l]=min(dp[to][k][l],make_pair(dp[i][j][k].first+1,dp[i][j][k].second+(p[k]-p[l]).ABS()));
				}
				for(int l=0;l<a;l++)for(int m=0;m<a;m++){
					if(l==m)continue;
					if(iLL(p[j],p[k],p[l],p[m])!=1)continue;
					Pt t=pLL(p[j],p[k],p[l],p[m]);
					if(iSP(p[j],p[k],t)==2&&iSP(t,p[l],p[m])==2){
						int to=i|(1<<l)|(1<<m);
						for(int n=0;n<a;n++){
							if(iSP(t,p[n],p[m])==2)to|=(1<<n);
						}
						dp[to][l][m]=min(dp[to][l][m],make_pair(dp[i][j][k].first+1,dp[i][j][k].second+(t-p[k]).ABS()+(p[m]-t).ABS()));
					}
				}
			}
		}
		pair<int,double>ret=make_pair(999999999,99999999);
		for(int i=0;i<a;i++)for(int j=0;j<a;j++)ret=min(ret,dp[(1<<a)-1][i][j]);
		printf("%d %.12f\n",ret.first,ret.second);
	}
}
