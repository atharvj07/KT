a = input()
b = input()
c = {'monday': 1, 'tuesday': 2, 'wednesday': 3, 'thursday': 4, 'friday': 5, 'saturday': 6, 'sunday': 7}
d = c[b] - c[a]
if a == b:
	print('YES')
elif d == 2 or d == 3:
	print('YES')
elif d < 0:
	d = c[b] + 7 - c[a]
	if d == 2 or d == 3:
		print('YES')
	else:
		print('NO')
else:
	print('NO')
