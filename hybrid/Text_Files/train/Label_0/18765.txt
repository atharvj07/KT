#include <bits/stdc++.h>
using namespace std;
int prime[100010], cnt1, cnt2, ans;
int main() {
  int n;
  cin >> n;
  if (n == 1) {
    cout << "1" << endl;
    cout << "1" << endl;
    return 0;
  }
  fill(prime, prime + n + 100, 1);
  for (int i = 2; i * i <= n + 10; i++) {
    if (prime[i]) {
      for (int j = 2 * i; j <= n + 10; j += i) {
        prime[j] = 0;
      }
    }
  }
  for (int i = 2; i <= n + 1; i++) {
    if (prime[i]) {
      cnt1++;
    } else {
      cnt2++;
    }
  }
  if (cnt1 > 0) ans++;
  if (cnt2 > 0) ans++;
  cout << ans << endl;
  for (int i = 2; i <= n + 1; i++) {
    if (prime[i]) {
      cout << "1 ";
    } else {
      cout << "2 ";
    }
  }
  return 0;
}
