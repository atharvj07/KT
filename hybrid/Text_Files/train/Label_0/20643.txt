#include <bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n, c;
    cin >> n >> c;
    ++n;
    int count = 0;
    for (int i = 0; i < c; ++i) {
        int k;
        cin >> k;
        count += k;
    }
    cout << (count + n - 1) / n << '\n';
}
