#include <bits/stdc++.h>
using namespace std;
int n, a[305];
string S[305];
bool vis[305];
vector<int> E[305], W;
void dfs(int x) {
  W.push_back(x);
  vis[x] = true;
  for (int i = 0; i < E[x].size(); i++) {
    if (!vis[E[x][i]]) dfs(E[x][i]);
  }
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> S[i];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i != j && S[i][j] == '1') E[i].push_back(j);
    }
  }
  for (int i = 0; i < n; i++) {
    if (!vis[i]) {
      W.clear();
      dfs(i);
      vector<int> AUX;
      sort(W.begin(), W.end());
      for (int i = 0; i < W.size(); i++) {
        AUX.push_back(a[W[i]]);
      }
      sort(AUX.begin(), AUX.end());
      for (int i = 0; i < W.size(); i++) {
        a[W[i]] = AUX[i];
      }
    }
  }
  for (int i = 0; i < n; i++) cout << a[i] << " ";
  cout << endl;
  return 0;
}
