import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class Main {
    public static void main(String[] args) {
//        Test.testing();
        ConsoleIO io = new ConsoleIO();
        new Main(io).solve();
        io.flush();
    }
    ConsoleIO io;
    Main(ConsoleIO io) {
        this.io = io;
    }

    ArrayList<ArrayList<Integer>> gr;
    boolean[] visit;
    //
    int[] tr;
    public void solve() {
        String[] s = io.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        H = Long.parseLong(s[1]);
        double p = Double.parseDouble(s[2]);
        leftP = p;
        rightP = 1.0-p;
        tr = io.readIntArray();
        Arrays.sort(tr);

        NN = n+1;
        memo = new double[NN*NN*2*2];
        for(int i = 0;i<memo.length;i++)
                memo[i] = -1.0;

        ltrJump = new int[n];
        rtlJump = new int[n];
        for(int i = 0;i<n;i++) {
            for (int j = i + 1; j <= n; j++)
                if (j == n || tr[j] - tr[j - 1] >= H) {
                    ltrJump[i] = j;
                    break;
                }
            for (int j = i - 1; j >= -1; j--)
                if (j == -1 || tr[j + 1] - tr[j] >= H) {
                    rtlJump[i] = j;
                    break;
                }
        }

        double res = calc(0, tr.length-1, 0, 0);

        io.writeLine(""+res);
    }

    int[] ltrJump;
    int[] rtlJump;
    double[]memo; // left down = 1, right down = 0
    double leftP, rightP;
    long H;
    int NN;
    double calc(int left, int right, int prevLeftDir, int prevRightDir) {
        int key  = left*NN*2*2 + right*2*2+prevLeftDir*2+prevRightDir;
        if (left > right) return 0.0;
        if (memo[key]>-1.0)
            return memo[key];
        double res = 0.0;

        long lx = left == 0 ? Integer.MIN_VALUE : (prevLeftDir == 1 ? tr[left - 1] : tr[left - 1] + H);
        res += leftP * (Math.min(tr[left] - lx, H) + calc(left + 1, right, 1, prevRightDir));

        long rx = right == tr.length - 1 ? Integer.MAX_VALUE : (prevRightDir == 1 ? tr[right + 1] - H : tr[right + 1]);
        res += rightP * (Math.min(rx - tr[right], H) + calc(left, right - 1, prevLeftDir, 0));

        if(left<right) {
            int next = ltrJump[left];
            res += rightP * (Math.min(tr[next - 1] + H, rx) - tr[left] + calc(next, right, 0, prevRightDir));

            next = rtlJump[right];
            res += leftP * (tr[right] - Math.max(tr[next + 1] - H, lx) + calc(left, next, prevLeftDir, 1));

            res /= 2;
        }

        memo[key] = res;
        return res;
    }

}

class ConsoleIO {
    BufferedReader br;
    PrintWriter out;
    public ConsoleIO(){br = new BufferedReader(new InputStreamReader(System.in));out = new PrintWriter(System.out);}
    public void flush(){this.out.close();}
    public void writeLine(String s) {this.out.println(s);}
    public void writeInt(int a) {this.out.print(a);this.out.print(' ');}
    public void writeWord(String s){
        this.out.print(s);
    }
    public String readLine() {try {return br.readLine();}catch (Exception ex){ return "";}}
    public long readLong() {
        return Long.parseLong(this.readLine());
    }
    public int readInt() {
        return Integer.parseInt(this.readLine().trim());
    }
    public long[] readLongArray() {
        String[]n=this.readLine().trim().split(" ");long[]r=new long[n.length];
        for(int i=0;i<n.length;i++)r[i]=Long.parseLong(n[i]);
        return r;
    }
    public int[] readIntArray() {
        String[]n=this.readLine().trim().split("\\s+");int[]r=new int[n.length];
        for(int i=0;i<n.length;i++)r[i]=Integer.parseInt(n[i]);
        return r;
    }
}
