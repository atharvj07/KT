#include <iostream>
#include <cstdio>

using namespace std;

#define loop(i,a,b) for(int i=(a);i<int(b);i++)
#define rep(i,b) loop(i,0,b)

int main(){
    string s;
    while(cin >> s){
        string a = "Z";
        for(char c:s){
            if(c != 'A' && c != 'Z') continue;
            if(a.back() != c) a += c;
        }
        a = a.substr(1,s.size());
        if(a.back() == 'A') a = a.substr(0,a.size()-1);
        if(a == "") a = "-1";
        cout << a << endl;
    }
}
