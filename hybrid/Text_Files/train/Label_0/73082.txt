n = int(input())
y = []
for i in range(n):
	a = input()
	y.append(a)
y1 = []
for j in range(n):
	b = input()
	y1.append(b)
y = sorted(y)
y1 = sorted(y1)
j = 0
while j < n:
	k = 0
	while k < n:
		if y[j] == y1[k]:
			y[j] = '0'
			y1[k] = '-1'
			break
		k += 1
	j += 1
y = sorted(y)
y1 = sorted(y1)
count = 0
j = n - 1
while j >= 0 and y[j] != '0' and (y1[j] != '-1'):
	i = 0
	while i < len(y[j]):
		if y[j][i] != y1[j][i]:
			count += 1
		i += 1
	j -= 1
print(count)
