(n, s, arr) = (int(input()), input(), [int(s) for s in input().split()])
(start, end) = (n, n)
for i in range(n):
	temp = int(s[i])
	if temp < arr[temp - 1]:
		start = min(start, i)
	if start == n or end != n:
		continue
	if temp > arr[temp - 1]:
		end = i
for i in range(n):
	if i < start or i >= end:
		print(s[i], end='')
	else:
		print(arr[int(s[i]) - 1], end='')
