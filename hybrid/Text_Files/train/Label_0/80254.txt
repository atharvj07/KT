from collections import *
(n, m, k) = map(int, input().split())
d = defaultdict(int)
for i in map(int, input().split()):
	d[i] += 1
l = [0] * (n + 1)
l[1] = 1
for i in range(k):
	(u, v) = map(int, input().split())
	if l[u] and u in d:
		print(u)
		exit(0)
	elif (l[u] or l[v]) and v in d:
		print(v)
		exit(0)
	else:
		(l[u], l[v]) = (l[v], l[u])
print(l.index(1))
