    class FastScanner{

        StringTokenizer st;
        BufferedReader br;

        FastScanner( BufferedReader _br ){
            br = _br;
        }

        String readLine(){
            try {
                return br.readLine();
            } catch (IOException e) {
                return null;
            }
        }

        String next(){
            while( st==null || !st.hasMoreElements() )
                st = new StringTokenizer(readLine());
            return st.nextToken();
        }

        int nextInt(){ return Integer.parseInt(next()); }
        long nextLong(){ return Long.parseLong(next()); }
        double nextDouble(){ return Double.parseDouble(next()); }
    }

    final int ZALUPA = (int)1e3;
    int n;
    int d;
    long x;
    int[] a, b;
    int[] pos, posOne;


    int getNextX() {
        x = (x * 37 + 10007) % 1000000007;
        return (int)x;
    }

    void initAB() {
        int i;
        for(i = 0; i < n; i = i + 1){
            a[i] = i + 1;
        }
        for(i = 0; i < n; i = i + 1){
            int j = getNextX() % (i + 1);
            int t = a[i];
            a[i] = a[j];
            a[j] = t;
        }
        for(i = 0; i < n; i = i + 1){
            if (i < d)
                b[i] = 1;
            else
                b[i] = 0;
        }
        for(i = 0; i < n; i = i + 1){
            int j = getNextX() % (i + 1);
            int t = b[i];
            b[i] = b[j];
            b[j] = t;
        }

        pos = new int[n+1];
        for ( i = 0; i < n; i++) {
            pos[a[i]] = i;
        }

        posOne = new int[d]; fill( posOne, -1 );
        int sz = 0;
        for (i= 0; i < n; ++i) {
            if( b[i] == 1 ){
                posOne[sz++] = i;
            }
        }
        if( posOne.length != sz ) MLE();
    }

    void solve(){
//        for( int i = 0; i < 1e4; ++i ) err.printf("%10d\n", getNextX());
        n = sc.nextInt();
        d = sc.nextInt();
        x = sc.nextInt();
        a = new int[n];
        b = new int[n];
        initAB();




        for (int i = 0; i < n; i++) {
            int ci = 0;
            for (int val = n, low = max(0, n - ZALUPA + 1); low <= val; val--) {
                if( pos[val] <= i && b[i-pos[val]] == 1 ){
                    ci = val;
                    break;
                }
            }
            if( ci == 0 ){
                for (int j : posOne) {
                    if( i < j ) break;
                    ci = max( ci, a[i-j] );
                }
            }
            out.println(ci);
        }
    }
