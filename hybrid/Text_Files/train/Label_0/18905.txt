#include<cstdio>
#include<algorithm>
using namespace std;
int n, K, w[16], D[17][65536], F[65537], InvF[65537], Mod = 1000000007, res;
int Pow(int a, int b) {
	int r = 1;
	while (b) {
		if (b & 1)r = 1ll * r*a%Mod;
		a = 1ll * a*a%Mod; b >>= 1;
	}
	return r;
}
int PP(int n, int k) {
	return 1ll*F[n] * InvF[n - k] % Mod;
}
int main() {
	int i, j, k;
	scanf("%d%d", &n, &K);
	F[0] = 1;
	for (i = 1; i <= (1 << n); i++)F[i] = 1ll * F[i - 1] * i%Mod;
	InvF[i - 1] = Pow(F[i - 1], Mod - 2);
	for (i = (1 << n); i >= 1; i--)InvF[i - 1] = 1ll * InvF[i] * i%Mod;
	for (i = 0; i < K; i++) {
		scanf("%d", &w[i]);
	}
	D[K][0] = 1;
	for (i = K-1; i >= 0; i--) {
		for (j = 0; j < (1 << n); j++) {
			D[i][j] = (D[i][j] + D[i + 1][j]) % Mod;
			for (k = 0; k < n; k++) {
				if ((j >> k) & 1)continue;
				int c = (1 << n) - w[i] - j;
				if (c < (1<<k)-1)continue;
				D[i][j | (1 << k)] = (D[i][j | (1 << k)] + 1ll * D[i + 1][j] * PP(c, (1 << k) - 1)%Mod * (1<<k)) % Mod;
			}
		}
	}
	for (i = 0; i < (1 << n); i++) {
		int c = 0;
		for (j = 0; j < n; j++) {
			if ((i >> j) & 1)c++;
		}
		int t = 1ll * D[0][i] * F[(1<<n) - i - 1] % Mod;
		if (c & 1) res = (res + Mod - t) % Mod;
		else res = (res + t) % Mod;
	}
	printf("%d\n", 1ll*res*(1<<n)%Mod);
}
