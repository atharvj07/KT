#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int m = 1, p = 1, s = 0;
    vector<int> ans;
    while (m < n) {
      if (n - m - p <= p) {
        s = n - m - p;
      } else {
        s = min(p, (n - m - 2 * p) / 2);
      }
      p += s;
      m += p;
      ans.push_back(s);
    }
    int d = ans.size();
    cout << d << "\n";
    for (int i = 0; i < d; i++) {
      cout << ans[i] << " ";
    }
    cout << "\n";
  }
}
