n = int(input())
a = list(map(int, input().split()))
g = 0
b = 0
s = 0
count = 0
for i in a:
	count += 1
	if count > 3:
		count = 1
	if count == 1:
		g = g + i
	if count == 2:
		b = b + i
	if count == 3:
		s = s + i
if g > b and g > s:
	print('chest')
elif b > s:
	print('biceps')
else:
	print('back')
