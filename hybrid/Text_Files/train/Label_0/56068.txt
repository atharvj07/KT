import java.io.*;
import java.util.*;

public class Main implements Runnable {

	public void _main() throws IOException {		
		int n = nextInt();
		String[] teams = new String[n];
		for (int i = 0; i < n; i++) {
			teams[i] = next();
		}
		int[] score = new int[n];
		int[] goaled = new int[n];
		int[] missed = new int[n];
		for (int i = 0; i < n * (n - 1) / 2; i++) {			
			String[] v = next().split("-");
			
			int a = id(teams, v[0]);
			int b = id(teams, v[1]);
			v = next().split(":");
			int num1 = Integer.parseInt(v[0]);
			int num2 = Integer.parseInt(v[1]);

			goaled[a] += num1;
			goaled[b] += num2;
			
			missed[a] += num2;
			missed[b] += num1;
			
			if (num1 > num2) {
				score[a] += 3;
			}
			else if (num2 > num1) {
				score[b] += 3;
			}
			else {
				score[a] += 1;
				score[b] += 1;
			}
		}
		Team[] p = new Team[n];
		for (int i = 0; i < n; i++)
			p[i] = new Team(i, score[i], goaled[i], missed[i]);
		Arrays.sort(p);
		String[] res = new String[n / 2];
		for (int i = 0; i < n / 2; i++)
			res[i] = teams[p[i].id];
		Arrays.sort(res);
		for (String s : res)
			out.println(s);
	}
	
	class Team implements Comparable<Team>  {
		int id, score, goaled, missed;
		Team(int id, int score, int goaled, int missed) {
			this.id = id;
			this.score = score;
			this.goaled = goaled;
			this.missed = missed;
		}
		public int compareTo(Team o) {
			if (score != o.score)
				return o.score - score;
			int d = goaled - missed;
			int od = o.goaled - o.missed;
			if (d != od)
				return od - d;
			return o.goaled - goaled;
		}
	}
	
	int id(String[] s, String str) {
		for (int i = 0; i < s.length; i++)
			if (s[i].equals(str))
				return i;
		return -1;
	}

	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer st;

	private String next() throws IOException {
		while (st == null || !st.hasMoreTokens()) {
			String rl = in.readLine();
			if (rl == null)
				return null;
			st = new StringTokenizer(rl);
		}
		return st.nextToken();
	}

	private int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	private long nextLong() throws IOException {
		return Long.parseLong(next());
	}

	private double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	public static void main(String[] args) {
		new Thread(new Main()).start();
	}

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);

			_main();

			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(202);
		}
	}

}
