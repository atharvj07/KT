import java.io.*;
import java.util.*;

public class CF1440D extends PrintWriter {
	CF1440D() { super(System.out); }
	static class Scanner {
		Scanner(InputStream in) { this.in = in; } InputStream in;
		byte[] bb = new byte[1 << 15]; int i, n;
		byte getc() {
			if (i == n) {
				i = n = 0;
				try { n = in.read(bb); } catch (IOException e) {}
			}
			return i < n ? bb[i++] : 0;
		}
		int nextInt() {
			byte c = 0; while (c <= ' ') c = getc();
			int a = 0; while (c > ' ') { a = a * 10 + c - '0'; c = getc(); }
			return a;
		}
	}
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1440D o = new CF1440D(); o.main(); o.flush();
	}

	int[] eo; int[][] ej;
	int[] dd; boolean[] removed;
	int[] qu;
	void init(int n) {
		eo = new int[n]; ej = new int[n][2];
		dd = new int[n]; removed = new boolean[n];
		qu = new int[n];
	}
	void append(int i, int j) {
		int o = eo[i]++; dd[i]++;
		if (o >= 2 && (o & o - 1) == 0)
			ej[i] = Arrays.copyOf(ej[i], o << 1);
		ej[i][o] = j;
	}
	void main() {
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			int m = sc.nextInt();
			int k = sc.nextInt();
			int k_ = (long) k * (k - 1) / 2 <= m ? k - 1 : k;
			init(n);
			HashSet<Long> hs = new HashSet<>();
			for (int h = 0; h < m; h++) {
				int i = sc.nextInt() - 1;
				int j = sc.nextInt() - 1;
				if (i > j) {
					int tmp = i; i = j; j = tmp;
				}
				append(i, j);
				append(j, i);
				hs.add((long) i << 20 | j);
			}
			TreeSet<Integer> pq = new TreeSet<>((i, j) -> dd[i] != dd[j] ? dd[i] - dd[j] : i - j);
			for (int i = 0; i < n; i++)
				pq.add(i);
			boolean found = false;
			while (!pq.isEmpty()) {
				int i = pq.first();
				if (dd[i] >= k) {
					println("1 " + pq.size());
					for (int j : pq)
						print(j + 1 + " ");
					println();
					found = true;
					break;
				}
				if (dd[i] >= k_) {
					boolean clique = true;
					int cnt = 0;
					qu[cnt++] = i;
					for (int o = eo[i]; o-- > 0; ) {
						int j = ej[i][o];
						if (!removed[j])
							qu[cnt++] = j;
					}
out:
					for (int h1 = 1; h1 < cnt; h1++) {
						int u = qu[h1];
						for (int h2 = h1 + 1; h2 < cnt; h2++) {
							int v = qu[h2];
							if (!hs.contains(u < v ? (long) u << 20 | v : (long) v << 20 | u)) {
								clique = false;
								break out;
							}
						}
					}
					if (clique) {
						println(2);
						while (cnt-- > 0)
							print(qu[cnt] + 1 + " ");
						println();
						found = true;
						break;
					}
				}
				pq.remove(i);
				removed[i] = true;
				for (int o = eo[i]; o-- > 0; ) {
					int j = ej[i][o];
					if (!removed[j]) {
						pq.remove(j);
						dd[j]--;
						pq.add(j);
					}
				}
			}
			if (!found)
				println(-1);
		}
	}
}
