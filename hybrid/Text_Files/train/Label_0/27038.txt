#include <iostream>
#include <cstdio>
#include <cstring>
#include <cstdlib>
#include <algorithm>
#define mod 998244353

using namespace std;
typedef long long ll;
const int MAXN = 405;
int n,a[MAXN],b[MAXN];
ll prb[MAXN],pw[MAXN][MAXN],f[MAXN][MAXN][MAXN],c[MAXN][MAXN],inv[MAXN];
int prea[MAXN],sum,preb[MAXN];
inline ll qpow(ll a, int b)
{
	ll res = 1;
	b = (b+mod-1)%(mod-1);
	for(; b; a = a*a%mod, b >>= 1)
	    if(b&1)
			res = res*a%mod;
	return res; 
}

int main()
{
	cin >> n;
	for(int i = 1; i<=n; i++)
	{
		cin >> a[i] >> b[i];
	    prea[i] = prea[i-1]+a[i], preb[i] = preb[i-1]+b[i];
	}
	sum = prea[n];
	for(int i = 1; i<=n; i++)
	{
		prb[i] = a[i]*qpow(sum,-1)%mod;
		pw[i][0] = 1;
		for(int j = 1; j<MAXN; j++)
			pw[i][j] = pw[i][j-1]*prb[i]%mod;
	}
	for(int i = 0; i<MAXN; i++)
		c[i][0] = c[i][i] = 1;
	for(int i = 2; i<MAXN; i++)
		for(int j = 1; j<i; j++)
			c[i][j] = (c[i-1][j-1]+c[i-1][j])%mod;
	inv[0] = 1;
	for(int i = 1; i<MAXN; i++)
		inv[i] = inv[i-1]*qpow(i,-1)%mod;
	f[0][0][0] = mod-1;
	for(int i = 1; i<=n; i++)
		for(int j = 0; j<=preb[i]; j++)
			for(int k = 0; k<=prea[i]; k++)
			{
				f[i][j][k] = (f[i][j][k]+f[i-1][j][k])%mod;
				if(k-a[i]>=0)
					for(int x = 0; x<b[i]&&x<=j; x++)
						f[i][j][k] = (f[i][j][k]-f[i-1][j-x][k-a[i]]*inv[x]%mod*pw[i][x]%mod+mod)%mod;
			}
	ll ans = 0;
	for(int j = 0; j<=preb[n]; j++)
		for(int k = 0; k<=sum; k++)
			ans = (ans+f[n][j][k]*qpow(inv[j],-1)%mod*qpow(sum*qpow(k,-1)%mod,j+1)%mod)%mod;
	cout << ans << endl;
	return 0;
}
