for _ in ' ' * int(input()):
	a = int(input())
	if a == 1:
		z = int(input())
		if z == 1:
			print('1 2')
		else:
			print(-1)
		continue
	b = list(map(int, input().split()))
	if 1 not in b or 2 * a in b:
		print(-1)
		continue
	c = [i for i in range(1, 2 * a + 1) if i not in b]
	r = []
	for i in range(a):
		for j in c:
			if j > b[i]:
				r += [j]
				c.remove(j)
				break
	if len(r) != a:
		print(-1)
	else:
		for i in range(a):
			print(b[i], r[i], end=' ')
		print()
