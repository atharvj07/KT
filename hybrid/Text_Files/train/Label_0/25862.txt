const int INF = 1 << 30;
const double EPS = 1e-9;

using namespace std;

void solve(VV<int> route_list, int H, int W, VV<pair<int, int>> f) {

    int ans = INF;
    LL mini_dist = INF;
    FOR(r, 0, route_list.size()) {

        VV<LL> dist(H, V<LL>(W, INF));
        dist[f[5][0].first][f[5][0].second] = 0;    // start

        FOR(i, 1, route_list[r].size()) {
            int now_c = route_list[r][i - 1];
            int next_c = route_list[r][i];

            FOE(p1, f[now_c]) {
                int y = p1.first;
                int x = p1.second;

                FOE(p2, f[next_c]) {
                    int ny = p2.first;
                    int nx = p2.second;
                    dist[ny][nx] = min(dist[ny][nx], dist[y][x] + abs(ny - y) + abs(nx - x));
                }
            }
        }

        LL d = dist[f[6][0].first][f[6][0].second];
        int first = route_list[r][1];
        if (first == 0) {
            first = 5;
        }
        if (d < mini_dist) {
            ans = first;
            mini_dist = d;
        }
        else if (d == mini_dist) {
            ans = min(ans, first);
        }
    }

    if (mini_dist == INF) {
        print("NA");
    }
    else {
        cout << ans << " " << mini_dist << endl;
    }
}

int main() {
    cin.tie(0);
    ios::sync_with_stdio(false);

    while (true) {
        int W, H;
        cin >> W >> H;
        if (W + H == 0) {
            break;
        }

        VV<pair<int, int>> f(7);
        FOR(y, 0, H) {
            string s;
            cin >> s;
            FOR(x, 0, W) {
                if (s[x] == 'S') {
                    f[5].emplace_back(make_pair(y, x));
                }
                else if (s[x] == 'G') {
                    f[6].emplace_back(make_pair(y, x));
                }
                else if (s[x] == '.') {
                }
                else {
                    f[(s[x] - '0') - 1].emplace_back(make_pair(y, x));
                }
            }
        }

        VV<int> route;
        FOR(i, 0, 5) {
            V<int> r = {5};
            FOR(j, 0, 4) {
                r.emplace_back((i + j) % 5);
            }
            r.emplace_back(6);
            route.emplace_back(r);
        }

        solve(route, H, W, f);
    }

}
