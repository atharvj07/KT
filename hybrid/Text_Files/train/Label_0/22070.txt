#include <bits/stdc++.h>
using namespace std;
int n, x;
int main() {
  cin >> n;
  cin >> x;
  int i;
  n = n % 6;
  for (i = n; i > 0; i--) {
    if (i % 2 == 0) {
      if (x == 1)
        x = 2;
      else if (x == 2)
        x = 1;
    } else if (i % 2 == 1) {
      if (x == 0)
        x = 1;
      else if (x == 1)
        x = 0;
    }
  }
  cout << x;
  return 0;
}
