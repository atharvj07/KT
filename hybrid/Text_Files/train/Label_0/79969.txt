def f(arr):
	arr = [None] + arr
	dp = [0] * len(arr)
	for i in range(1, len(arr)):
		j = int(arr[i])
		for k in range(int(j), 0, -1):
			dp[j] = max(dp[j], 1 + dp[k])
	return len(arr) - max(dp) - 1
