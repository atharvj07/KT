#include <bits/stdc++.h>
using namespace std;
template <typename Tp>
inline void outarr(Tp _begin, Tp _end, const char* _delim = " ") {
  for (Tp current = _begin; current != _end; ++current) {
    std::cout << *current << _delim;
  }
  std::cout << '\n';
}
using ll = long long;
using pii = std::pair<int, int>;
constexpr int INF = 0x3f3f3f3f;
constexpr int MOD = static_cast<const int>(1e9 + 7);
constexpr int MAXN = 200000;
int arr[MAXN];
int brr[MAXN];
int ans[MAXN];
ll cnt[30];
bool HasBit(int x, int b) { return (x >> b) & 1; }
bool Check(int n) {
  for (int i = 0; i < (n); ++i) {
    for (int b = 0; b < 30; ++b) {
      if (HasBit(ans[i], b)) {
        ++cnt[b];
      }
    }
  }
  for (int i = 0; i < (n); ++i) {
    ll res = 0;
    for (int b = 0; b < 30; ++b) {
      if (HasBit(ans[i], b)) {
        res += cnt[b] << b;
      }
    }
    if (arr[i] != res) {
      return false;
    }
  }
  return true;
}
int main() {
  std::ios_base::sync_with_stdio(false);
  std::cin.tie(nullptr);
  int n;
  cin >> n;
  for (int i = 0; i < (n); ++i) {
    cin >> arr[i];
  }
  for (int i = 0; i < (n); ++i) {
    cin >> brr[i];
  }
  for (int i = 0; i < (n); ++i) {
    if (arr[i] > brr[i]) {
      return cout << (-1) << '\n', 0;
      ;
    }
  }
  ll sum = accumulate(arr, arr + n, 0LL) + accumulate(brr, brr + n, 0LL);
  if (sum % (n + n) != 0) {
    return cout << (-1) << '\n', 0;
    ;
  }
  sum /= n + n;
  for (int i = 0; i < (n); ++i) {
    ans[i] = int(arr[i] + brr[i] - sum);
    if (ans[i] % n != 0) {
      return cout << (-1) << '\n', 0;
      ;
    }
    ans[i] /= n;
  }
  if (!Check(n)) {
    return cout << (-1) << '\n', 0;
    ;
  }
  outarr(ans, ans + n);
  return 0;
}
