#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define fir first
#define sec second
using namespace std;
typedef long long ll;
const int N = 3e5 + 5;
vector<pair<int, int>> G[N]; 
int n, m, x, fa[N], ans[N], hd, tl;
bool vis[N];
ll sum, a[N];
void Dfs(int u, int p) {
	vis[u] = true;
	for(pair<int, int> e : G[u]) {
		int v = e.fir, id = e.sec;
		if(vis[v]) {
			continue;
		}
		fa[v] = u;
		Dfs(v, id);
	}
	if(u == 1) {
		return;
	}
	if(a[u] >= x) {
		a[fa[u]] += a[u] - x;
		ans[++hd] = p;
	} else {
		ans[--tl] = p;
	}
}
int main() {
	ios::sync_with_stdio(false);
	cin >> n >> m >> x;
	for(int i = 1; i <= n; i++) {
		cin >> a[i];
		sum += a[i];
	}
	if(sum < (ll)x * (n - 1)) {
		cout << "NO\n";
		return 0;
	}
	for(int i = 1; i <= m; i++) {
		int u, v;
		cin >> u >> v;
		G[u].pb(mp(v, i)); 
		G[v].pb(mp(u, i));
	}
	hd = 0;
	tl = n;
	Dfs(1, 0);
	cout << "YES\n";
	for(int i = 1; i < n; i++) {
		cout << ans[i] << endl;
	}
	return 0;
}
