#include <bits/stdc++.h>
using namespace std;

const int64_t MOD = 998244353;
void add(int64_t& a, int64_t b){
    a = (a+b) % MOD;
}
void mul(int64_t& a, int64_t b){
    a = a*b % MOD;
}

int N, K;
vector<int> edges[100000];

int64_t dp[100000][301][2];
int64_t dp2[302][2];
int sz[100000];

void dfs(int i, int p){
    dp[i][1][1] = 1;
    dp[i][0][0] = 1;
    sz[i] = 1;
    for(int j : edges[i]) if(j != p){
        dfs(j, i);
        memset(dp2, 0, sizeof(dp2));
        for(int n1=0; n1<=min(K, sz[i]); n1++) for(int n2=0; n2<=min(K+1-n1, sz[j]); n2++){
            add(dp2[n1+n2][1], dp[i][n1][1] * dp[j][n2][0]);
            if(n1+n2) add(dp2[n1+n2-1][1], dp[i][n1][1] * dp[j][n2][1]);
            add(dp2[n1+n2][1], dp[i][n1][1] * dp[j][n2][1]);
            add(dp2[n1+n2][0], dp[i][n1][0] * dp[j][n2][1]);
            add(dp2[n1+n2][0], dp[i][n1][0] * dp[j][n2][0]);
        }
        sz[i] += sz[j];
        for(int n=0; n<=K; n++) for(int k=0; k<2; k++) dp[i][n][k] = dp2[n][k];
    }
}

int main(){
    cin >> N >> K;
    for(int i=0; i<N-1; i++){
        int a, b;
        cin >> a >> b;
        edges[a-1].push_back(b-1);
        edges[b-1].push_back(a-1);
    }

    dfs(0, -1);
    int64_t ans = 0;
    for(int k=0; k<2; k++) add(ans, dp[0][K][k]);
    cout << ans << endl;
    return 0;
}
