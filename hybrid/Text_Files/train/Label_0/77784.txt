n = int(input())
a = list(map(int, input().strip().split()))
stack = []
max_xor = 0
for k in range(n):
	if not stack:
		stack.append(a[k])
	elif a[k] < stack[-1]:
		max_xor = max(max_xor, a[k] ^ stack[-1])
		stack.append(a[k])
	elif a[k] > stack[-1]:
		while stack and a[k] > stack[-1]:
			popped = stack.pop()
			max_xor = max(max_xor, a[k] ^ popped)
		if stack:
			max_xor = max(max_xor, a[k] ^ stack[-1])
		stack.append(a[k])
print(max_xor)
