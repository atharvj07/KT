

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

public class E {
    static long mod = 1000000007;
    static long[][] ways = new long[1001][1001];
    static int k;

    public static long getTotal(String x) {
        long ans = 0;
        for (int i = 0; i < x.length(); i++)
            ans = (10 * ans + x.charAt(i) - '0') % mod;
        return ans;
    }

    public static long get(String x, int index, int cantPut47) {
        if (index == x.length())
            return 1;
        int maxDigit = x.charAt(index) - '0';
        long ans = 0;
        for (int i = 0; i <= maxDigit; i++) {
            if ((i == 4 || i == 7) && cantPut47 > 0)
                continue;
            if (i == 4 || i == 7)
                if (i < maxDigit)
                    ans = (ans + ways[x.length() - index - 1][k]) % mod;
                else
                    ans = (ans + get(x, index + 1, k)) % mod;
            else if (i < maxDigit)
                ans = (ans + ways[x.length() - index - 1][Math.max(
                        cantPut47 - 1, 0)]) % mod;
            else
                ans = (ans + get(x, index + 1, Math.max(cantPut47 - 1, 0)))
                        % mod;
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] S = in.readLine().split(" ");
        int t = Integer.parseInt(S[0]);
        k = Integer.parseInt(S[1]);
        for (int i = 0; i < 1001; i++)
            ways[0][i] = 1;
        for (int i = 1; i < 1001; i++)
            for (int j = 0; j < 1001; j++)
                if (j == 0)
                    ways[i][j] = (8 * ways[i - 1][0] + 2 * ways[i - 1][k])
                            % mod;
                else
                    ways[i][j] = (8 * ways[i - 1][j - 1]) % mod;
        for (int i = 0; i < t; i++) {
            S = in.readLine().split(" ");
            String left = "" + (new BigInteger(S[0]).subtract(BigInteger.ONE));
            String right = S[1];
            long tot = (getTotal(right) - getTotal(left) + mod) % mod;
            long bad = (get(right, 0, 0) - get(left, 0, 0) + mod) % mod;
            long ans = (tot - bad + mod) % mod;
            System.out.println(ans);
        }
    }
}
