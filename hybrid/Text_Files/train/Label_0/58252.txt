import java.io.*;
import java.util.*;
public class E {
	static ArrayList<Integer> [] adj;
	static ArrayList<Integer> path;
	static boolean [] black;
	static boolean [] ok;
	static int [] parent;
	static int cnt=0;
	static int n;
	static boolean first=true;
	public static void main(String[]args)throws Throwable{
		PrintWriter pw=new PrintWriter(System.out,true);
		MyScanner sc=new MyScanner();
		n=sc.nextInt();
		adj=new ArrayList[n];
		for(int i=0;i<n;i++)
			adj[i]=new ArrayList<Integer>();
		black=new boolean[n];
		ok=new boolean[n];
		Arrays.fill(ok, false);
		parent=new int[n];
		Arrays.fill(parent, -1);
		for(int i=0;i<n;i++){
			black[i]=(sc.nextInt()==1);
			if(black[i])
				cnt++;
		}
		for(int i=1;i<n;i++){
			int u=sc.nextInt()-1;
			int v=sc.nextInt()-1;
			adj[u].add(v);
			adj[v].add(u);
		}
		path=new ArrayList<Integer>();
		if(cnt==n)
			pw.println(1);
		else{
			dfs2(0);
			dfs(0);
			boolean f=true;
			for(int e : adj[0])
				f &=ok[e];
			if(ok[0]!=f && !black[0]){
				path.add(adj[0].get(0)+1);
				path.add(1);
				path.add(adj[0].get(0)+1);
			}
			for(int i=0;i<path.size();i++)
				if(i<path.size()-1)
					pw.print(path.get(i)+" ");
				else
					pw.println(path.get(i));
		}
		pw.flush();
		pw.close();
	}
	
	public static void dfs(int i){
		if(first){
			first=false;
		}else{
			black[i]=!black[i];
			if(black[i])
				cnt++;
			else
				cnt--;
		}
		path.add(i+1);
		if(cnt==n)
			return;
		for(int e : adj[i]){
			if(!ok[e]&& e!=parent[i]){
				if(cnt==n)
					return;
				dfs(e);
				ok[e]=black[e];
				for(int c : adj[e])
					if(c!=parent[e])
						ok[e] &=ok[c];
				if(cnt==n)
					return;
				dfs(i);
			}
		}
		if(cnt==n)
			return;
	}
	
	public static void dfs2(int i){
		ok[i]=black[i];
		for(int e : adj[i]){
			if(!ok[e] && e!=parent[i]){
				parent[e]=i;
				dfs2(e);
				if(!ok[e])
					ok[i]=false;
			}
		}
	}
	
	static class MyScanner {
	      BufferedReader br;
	      StringTokenizer st;
	 
	      public MyScanner() {
	         br = new BufferedReader(new InputStreamReader(System.in));
	      }
	 
	      String next() {
	          while (st == null || !st.hasMoreElements()) {
	              try {
	                  st = new StringTokenizer(br.readLine());
	              } catch (IOException e) {
	                  e.printStackTrace();
	              }
	          }
	          return st.nextToken();
	      }
	 
	      int nextInt() {
	          return Integer.parseInt(next());
	      }
	 
	      long nextLong() {
	          return Long.parseLong(next());
	      }
	 
	      double nextDouble() {
	          return Double.parseDouble(next());
	      }
	 
	      String nextLine(){
	          String str = "";
		  try {
		     str = br.readLine();
		  } catch (IOException e) {
		     e.printStackTrace();
		  }
		  return str;
	      }

	   }
}
