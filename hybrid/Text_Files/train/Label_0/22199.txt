#include <bits/stdc++.h>
using namespace std;
int n, m, s, t;
long long p[2200], dis[2200][2200];
vector<pair<int, long long> > g[2200];
priority_queue<pair<long long, int>, vector<pair<long long, int> >,
               greater<pair<long long, int> > >
    pq;
void dij(int s) {
  memset(dis[s], 0x3f, sizeof(dis[s]));
  dis[s][s] = 0;
  pq.push(make_pair(dis[s][s], s));
  while (!pq.empty()) {
    int x = pq.top().second;
    long long dd = pq.top().first;
    pq.pop();
    if (dis[s][x] < dd) continue;
    for (int i = 0; i < g[x].size(); i++) {
      int to = g[x][i].first;
      long long d = g[x][i].second;
      if ((dis[s][x] + d) < dis[s][to]) {
        dis[s][to] = dis[s][x] + d;
        pq.push(make_pair(dis[s][to], to));
      }
    }
  }
}
int perms[2200], permt[2200], idxs[2200], idxt[2200];
long long dps[2200][2200], dpt[2200][2200];
bool cans[2200][2200], cant[2200][2200];
bool lesss(int a, int b) { return dis[s][a] < dis[s][b]; }
bool lesst(int a, int b) { return dis[t][a] < dis[t][b]; }
int main() {
  scanf("%d%d%d%d", &n, &m, &s, &t);
  for (int i = 1; i <= n; i++) scanf("%I64d", &p[i]);
  memset(dis, 0x3f, sizeof(dis));
  for (int i = 1; i <= n; i++) dis[i][i] = 0;
  for (int i = 0; i < m; i++) {
    int u, v;
    long long w;
    scanf("%d%d%I64d", &u, &v, &w);
    if (w < dis[u][v]) dis[u][v] = dis[v][u] = w;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i != j && dis[i][j] < 0x3f3f3f3f3f3f3f3fll)
        g[i].push_back(make_pair(j, dis[i][j]));
  dij(s);
  dij(t);
  for (int i = 1; i <= n; i++) perms[i] = permt[i] = i;
  sort(perms + 1, perms + 1 + n, lesss);
  sort(permt + 1, permt + 1 + n, lesst);
  for (int i = 1; i <= n; i++) {
    idxs[perms[i]] = i;
    idxt[permt[i]] = i;
  }
  for (int i = 1; i <= n; i++)
    for (int lim = 1; lim <= n + 1; lim++) {
      cans[i][lim] = idxt[perms[i]] >= lim;
      cant[lim][i] = idxs[permt[i]] >= lim;
    }
  for (int ss = n; ss >= 1; ss--)
    for (int tt = n; tt >= 1; tt--) {
      if (!cans[ss][tt]) {
        dps[ss][tt] = dps[ss + 1][tt];
      } else {
        dps[ss][tt] = p[perms[ss]] + dps[ss + 1][tt];
        for (int nxt = ss + 1; nxt <= n; nxt++) {
          dps[ss][tt] = max(dps[ss][tt], p[perms[ss]] - dpt[nxt][tt]);
          if (cans[ss][tt]) break;
        }
      }
      if (!cant[ss][tt]) {
        dpt[ss][tt] = dpt[ss][tt + 1];
      } else {
        dpt[ss][tt] = p[permt[tt]] + dpt[ss][tt + 1];
        for (int nxt = tt + 1; nxt <= n; nxt++) {
          dpt[ss][tt] = max(dpt[ss][tt], p[permt[tt]] - dps[ss][nxt]);
          if (cant[ss][tt]) break;
        }
      }
    }
  printf(dps[1][1] > 0 ? "Break a heart" : dps[1][1] == 0 ? "Flowers" : "Cry");
  return 0;
}
