from math import sqrt, floor
from itertools import takewhile
inputs1st = list(map(int, input().split(' ')))
inputs2nd = list(map(int, input().split(' ')))

def solve(bourles, xs):
	s = set(xs)
	index1 = 0
	index2 = 0
	ans = set()
	for i in range(1, 150000):
		if i > bourles:
			break
		if len(s) > 0 and i in s:
			s.remove(i)
			continue
		ans.add(i)
		bourles -= i
	return list(ans)
ans = solve(inputs1st[1], inputs2nd)
print(len(ans))
print(' '.join(list(map(str, ans))))
