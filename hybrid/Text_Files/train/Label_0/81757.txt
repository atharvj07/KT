(n, m, d) = map(int, input().split())
arr = list(map(int, input().split()))
d -= 1
if sum(arr) + (len(arr) + 1) * d < n:
	print('NO')
	exit()
print('YES')
w = n - sum(arr)
res = [0] * n
ptr = 0
for i in range(m):
	ne = min(w, d)
	w -= ne
	for j in range(ptr + ne, ptr + ne + arr[i]):
		res[j] = i + 1
	ptr += ne + arr[i]
print(*res)
