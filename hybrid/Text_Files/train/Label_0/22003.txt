#include<iostream> 
#include<string> 
#include<vector> 
#include<algorithm> 
using namespace std; 
int main()
{ 
	long long int n, m; 
	cin >> n >> m;
	vector<long long int> a(n); 
	vector<long long int> b(m); 
	for (int i = 0; i < n; i++)
	{
		cin >> a.at(i);
	} 
	long long int sum = 0;
	for (int i = 0; i < m; i++)
	{
		cin >> b.at(i);
		sum += b.at(i);
	}
	long long int ans = 0; 
	for (int i = 0; i < n; i++)
	{
		ans += a.at(i)*sum;
	}
	cout << ans << endl;

}
