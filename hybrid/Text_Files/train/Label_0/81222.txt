n = int(input())
a = [int(a) for a in input().split()]
sereja = 0
dima = 0
flag = 0
j = n - 1
i = 0
s = []
d = []
while i <= j:
	if flag == 0:
		flag = 1
		if a[i] > a[j]:
			sereja += a[i]
			i += 1
			continue
		else:
			sereja += a[j]
			j -= 1
			continue
	elif flag == 1:
		flag = 0
		if a[i] > a[j]:
			dima += a[i]
			i += 1
		else:
			dima += a[j]
			j -= 1
print(str(sereja) + ' ' + str(dima))
