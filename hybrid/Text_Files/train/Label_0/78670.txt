import sys
n = int(input())
A = list(map(int, input().split()))
from collections import Counter
counter = Counter(A)
ANS = [[0] * n for i in range(n)]

def write(ANS):
	for i in range(n):
		print(' '.join([str(a) for a in ANS[i]]))
if n % 2 == 0:
	for v in counter.values():
		if v % 4 != 0:
			print('NO')
			sys.exit()
	i = j = 0
	for k in counter.keys():
		while counter[k] > 0:
			ANS[i][j] = ANS[n - i - 1][j] = ANS[i][n - j - 1] = ANS[n - i - 1][n - j - 1] = k
			counter[k] -= 4
			if j == n // 2 - 1:
				i += 1
				j = 0
			else:
				j += 1
	print('YES')
	write(ANS)
else:
	c1 = 0
	c2 = 0
	for v in counter.values():
		if v % 2 == 1:
			c1 += 1
			v -= 1
		if v % 4 == 2:
			c2 += 1
	if c1 > 1 or c2 > n - 1:
		print('NO')
		sys.exit()
	i2 = 0
	j2 = 0
	for k in counter.keys():
		if counter[k] > 0:
			if counter[k] % 2 == 1:
				ANS[n // 2][n // 2] = k
				counter[k] -= 1
			if counter[k] % 4 == 2 and i2 != n // 2:
				ANS[i2][n // 2] = ANS[n - i2 - 1][n // 2] = k
				i2 += 1
				counter[k] -= 2
			if counter[k] % 4 == 2 and i2 == n // 2:
				ANS[n // 2][j2] = ANS[n // 2][n - 1 - j2] = k
				j2 += 1
				counter[k] -= 2
	i = j = 0
	for k in counter.keys():
		if counter[k] > 0:
			while counter[k] > 0:
				if j == n // 2:
					if ANS[i][j] == 0:
						ANS[i][j] = ANS[n - i - 1][j] = ANS[j][i] = ANS[j][n - i - 1] = k
						counter[k] -= 4
					else:
						i += 1
						j = 0
				if i != n // 2 and j != n // 2:
					ANS[i][j] = ANS[n - i - 1][j] = ANS[i][n - j - 1] = ANS[n - i - 1][n - j - 1] = k
					counter[k] -= 4
				if i == n // 2 and ANS[i][j] == 0:
					ANS[i][j] = ANS[i][n - j - 1] = k
					counter[k] -= 2
				if j == n // 2:
					i += 1
					j = 0
				else:
					j += 1
	print('YES')
	write(ANS)
