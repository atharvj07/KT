import sys

def main():
	d_sum = 0
	max_sum = 0
	min_sum = 0
	(n, max_weight) = map(int, input().strip().split(' '))
	diffs = [int(diff) for diff in input().strip().split(' ')]
	for diff in diffs:
		d_sum += diff
		max_sum = max(max_sum, d_sum)
		min_sum = min(min_sum, d_sum)
	if abs(min_sum) > max_weight or max_sum > max_weight:
		return print(0)
	res = max_weight - (max_sum - min_sum) + 1
	if res < 0:
		return print(0)
	print(res)
sys.exit(main())
