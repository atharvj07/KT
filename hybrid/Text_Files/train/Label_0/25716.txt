#include <bits/stdc++.h>
using namespace std;
long long max_crossing_subarray(long long ar[], long long low, long long mid,
                                long long high) {
  long long left_sum = -1000000;
  long long sum = 0;
  long long i;
  for (i = mid; i >= low; i--) {
    sum = sum + ar[i];
    if (sum > left_sum) left_sum = sum;
  }
  long long right_sum = -1000000;
  sum = 0;
  for (i = mid + 1; i <= high; i++) {
    sum = sum + ar[i];
    if (sum > right_sum) right_sum = sum;
  }
  return (left_sum + right_sum);
}
long long max_sum_subarray(long long ar[], long long low, long long high) {
  if (high == low) {
    return ar[high];
  }
  long long mid = (high + low) / 2;
  long long maximum_sum_left_subarray = max_sum_subarray(ar, low, mid);
  long long maximum_sum_right_subarray = max_sum_subarray(ar, mid + 1, high);
  long long maximum_sum_crossing_subarray =
      max_crossing_subarray(ar, low, mid, high);
  return max(maximum_sum_left_subarray,
             max(maximum_sum_right_subarray, maximum_sum_crossing_subarray));
}
int main() {
  long long t;

    long long n;
    cin >> n;
    long long a[n];
    long long s = 0;

    }
    long long x = max_sum_subarray(a, 0, n - 2);

    x = max(x, y);
    if (s > x)

  }

}
