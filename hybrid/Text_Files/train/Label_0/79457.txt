import sys

def covered_segments(nsegments):
	covered_points = 0
	for i in range(nsegments):
		(x, y) = map(int, sys.stdin.readline().split())
		covered_points += y - x + 1
	return covered_points

def result(covered_points, k):
	if covered_points % k == 0:
		return 0
	elif k < covered_points:
		return k - covered_points % k
	else:
		return k - covered_points
(n, k) = map(int, sys.stdin.readline().split())
covered_points = covered_segments(n)
print(result(covered_points, k))
