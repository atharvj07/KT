#include <bits/stdc++.h>
using namespace std;
bool isPerfectSquare(long double x) {
  long double sr = sqrt(x);
  return ((sr - floor(sr)) == 0);
}
bool isPrime(int n) {
  if (n <= 1) return false;
  if (n <= 3) return true;
  if (n % 2 == 0 || n % 3 == 0) return false;
  for (int i = 5; i * i <= n; i = i + 6)
    if (n % i == 0 || n % (i + 2) == 0) return false;
  return true;
}
int main() {
  long long int n;
  cin >> n;
  while (n--) {
    long long int x;
    cin >> x;
    if (isPerfectSquare(x) && isPrime(sqrt(x)))
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
