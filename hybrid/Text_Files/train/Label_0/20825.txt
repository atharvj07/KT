#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int MAX_N = 1e5 + 9;
int f[MAX_N], n;
char pat[MAX_N];
vector<pair<int, int>> vc;
int getFailure(int len, char c) {
  while (len && pat[len] != c) len = f[len - 1];
  return len + (pat[len] == c);
}
void cmpF() {
  f[0] = 0;
  for (int i = 1; pat[i]; i++) f[i] = getFailure(f[i - 1], pat[i]);
}
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  cin >> pat;
  cmpF();
  n = strlen(pat);
  int len = f[n - 1];
  vector<int> ans(n + 1);
  for (int i = 0; i < n; i++) ans[f[i]]++;
  for (int i = n - 1; i > 0; i--) ans[f[i - 1]] += ans[i];
  for (int i = 0; i <= n; i++) ans[i]++;
  int idx = 0;
  while (len) {
    vc.push_back({len, ans[len]});
    len = f[len - 1];
  }
  vc.push_back({n, 1});
  sort(vc.begin(), vc.end());
  cout << vc.size() << "\n";
  for (auto i : vc) cout << i.first << " " << i.second << "\n";
}
