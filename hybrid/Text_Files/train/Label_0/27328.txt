#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, x;
  cin >> n >> x;
  char c;
  long long ans1 = 0;
  long long ans2 = 0;
  for (int i = 0; i < n; i++) {
    cin >> c;
    long long d;
    cin >> d;
    if (c == '+')
      x += d;
    else {
      if (x < d) {
        ans2++;
      } else {
        x -= d;
      }
    }
  }
  cout << x << " " << ans2 << endl;
  return 0;
}
