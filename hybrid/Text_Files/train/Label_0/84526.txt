rows = int(input())
possible = False
result = ''
for index in range(rows):
	row = input()
	(first, second) = row.split('|')
	if not possible:
		if first == 'OO':
			row = '++|' + second
			possible = True
		elif second == 'OO':
			row = first + '|++'
			possible = True
	result += row + '\n'
if possible:
	print('YES')
	print(result)
else:
	print('NO')
