n = int(input())
m = list(map(int, input().split()))
x = m.count(0)
y = n - x
a = 0
b = 0
l = max(x, y)
for i in range(n):
	if m[i] == 1:
		a += 1
		c = []
		d = 0
		e = 0
		for j in range(i + 1):
			if m[j] == 1:
				d += 1
			else:
				e += 1
			c.append(e + a - d)
		l = max(l, max(c) + y - a)
	else:
		b += 1
print(l)
