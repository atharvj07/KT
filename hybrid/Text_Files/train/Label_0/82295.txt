import sys, random
n = int(input())
l = list(map(int, input().split()))
l.sort()
d = {}
ans = []
for i in range(n):
	d[l[i]] = 1
f = 0
ans.append([l[n - 1]])
for i in range(n - 1):
	curr = l[i]
	for j in range(32):
		try:
			if d[curr + (1 << j)] == 1:
				try:
					if d[curr + (1 << j + 1)] == 1:
						print(3)
						print(curr, curr + (1 << j), curr + (1 << j + 1))
						sys.exit(0)
				except KeyError:
					ans.append([curr, curr + (1 << j)])
		except KeyError:
			continue
for i in ans:
	if len(i) == 2:
		print(2)
		print(i[0], i[1])
		sys.exit(0)
print(1)
print(ans[0][0])
