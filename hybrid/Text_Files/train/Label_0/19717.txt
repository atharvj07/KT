#include <set>
#include <string>
#include <iostream>
using namespace std;
int main() {
	int n; string s;
	cin >> n >> s;
	set<string> v;
	for (int i = 1; i < 1 << n; i *= 2) {
		for (int j = 0; j < 1 << n; j += 2 * i) {
			if (s.substr(j, i) != s.substr(j + i, i)) {
				v.insert(s.substr(j, 2 * i));
			}
		}
	}
	cout << v.size() << endl;
	return 0;
}
