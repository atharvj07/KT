from sys import stdin, stdout

def main():
	t = int(stdin.readline())
	for _ in range(t):
		(m, n) = map(int, stdin.readline().split())
		mat = []
		for _ in range(m):
			mat.append(stdin.readline())
		row = [0] * m
		col = [0] * n
		for i in range(m):
			for j in range(n):
				if mat[i][j] == '*':
					row[i] += 1
					col[j] += 1
		max_row = max(row)
		max_col = max(col)
		cr = []
		for i in range(m):
			if row[i] == max_row:
				cr.append(i)
		cc = []
		for i in range(n):
			if col[i] == max_col:
				cc.append(i)
		ans = m + n - 1
		for i in cr:
			for j in cc:
				if mat[i][j] == '.':
					ans = min(ans, m + n - (row[i] + col[j] + 1))
				else:
					ans = min(ans, m + n - (row[i] + col[j]))
		stdout.write(str(ans))
		stdout.write('\n')
main()
