(p, q, l, r) = map(int, input().split())
x = [[int(j) for j in input().split()] for i in range(p)]
z = [[int(j) for j in input().split()] for i in range(q)]
moment = [0] * 1001
ans = 0
for i in x:
	for j in z:
		if i[1] - j[0] <= r or i[0] - j[1] <= r:
			for e in range(max(l, i[0] - j[1]), min(i[1] - j[0], r) + 1):
				if moment[e] == 0:
					ans += 1
					moment[e] = 1
print(ans)
