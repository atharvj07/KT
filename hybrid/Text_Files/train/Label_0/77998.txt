from collections import defaultdict, deque, Counter
import sys
import bisect
import math
input = sys.stdin.readline
mod = 1000000007

def beauty(n, m, s, x, y):
	b = 0
	l = s
	l1 = x
	l2 = y
	f1 = []
	h1 = defaultdict(list)
	for i in range(n):
		f1.append(0)
	for i in range(m):
		(a, b) = (l1[i], l2[i])
		a -= 1
		b -= 1
		h1[a].append(b)
		f1[b] += 1
	q = deque()
	for i in range(n):
		if f1[i] == 0:
			q.append(i)
	count = 0
	ans = 0
	arr1 = [[0 for i in range(26)] for i in range(n)]
	while count < n and q:
		x = q.popleft()
		count += 1
		arr1[x][ord(l[x]) - 97] += 1
		for i in h1[x]:
			for j in range(26):
				arr1[i][j] = max(arr1[i][j], arr1[x][j])
			f1[i] -= 1
			if f1[i] == 0:
				q.append(i)
	for i in range(1):
		if b == 0:
			b = l1[0]
		else:
			b = l1[-1]
	if count != n:
		print(-1)
	else:
		ans = 0
		for i in range(n):
			ans = max(ans, max(arr1[i]))
		print(ans)
(n, m) = map(int, input().split())
s = input()
l1 = []
l2 = []
for i in range(m):
	(a, b) = map(int, input().split())
	l1.append(a)
	l2.append(b)
beauty(n, m, s, l1, l2)
