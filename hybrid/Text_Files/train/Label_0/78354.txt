n = int(input())
t = [input() for i in range(n)]
for i in range(n - 1):
	for j in range(i + 1, n):
		if len(t[j]) < len(t[i]):
			(t[i], t[j]) = (t[j], t[i])
h = True
for i in range(n - 1):
	if not t[i] in t[i + 1]:
		h = False
if h == True:
	print('YES')
	for i in range(n):
		print(t[i])
else:
	print('NO')
