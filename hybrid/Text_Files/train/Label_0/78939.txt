import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
a.sort()
L = a[:n]
R = a[n:][::-1]
s = sum((x - y for (x, y) in zip(R, L)))
MOD_NUM = 998244353
D = 1
for x in range(1, n + 1):
	D *= x
	D %= MOD_NUM
U = D
for x in range(n + 1, 2 * n + 1):
	U *= x
	U %= MOD_NUM
ans = U * pow(D, MOD_NUM - 2, MOD_NUM) * pow(D, MOD_NUM - 2, MOD_NUM)
print(ans * s % MOD_NUM)
