#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
long long cnt[30], c2[30], num[30][30];
int main() {
  string s;
  cin >> s;
  int len = s.size();
  long long ans = 0;
  for (int i = 0; i < len; ++i) ++cnt[s[i] - 'a'];
  for (int i = 0; i < 26; ++i) ans = max(ans, cnt[i]);
  for (int i = 0; i < len; ++i) {
    cnt[s[i] - 'a']--;
    for (int j = 0; j < 26; ++j) {
      num[s[i] - 'a'][j] += cnt[j];
    }
  }
  for (int i = 0; i < 26; ++i) {
    for (int j = 0; j < 26; ++j) ans = max(ans, num[i][j]);
  }
  cout << ans << "\n";
  return 0;
}
