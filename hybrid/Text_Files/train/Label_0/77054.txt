t = int(input())
for i in range(t):
	(x, y, a, b) = list(map(int, input().split()))
	if x > y:
		(x, y) = (y, x)
	if a < b:
		(a, b) = (b, a)
	ans = 0
	if a == b:
		ans = max(ans, x // a)
		print(ans)
	elif a * x - y * b <= 0:
		ans = max(ans, x // b)
		print(ans)
	elif a * y - x * b > 0:
		j = (a * x - y * b) // (a * a - b * b)
		start = max(0, j - 3)
		end = min(start + 6, x // a + 1)
		for j in range(start, end):
			b_poss_x = max(0, (x - j * a) // b)
			a_poss_y = max(0, (y - j * b) // a)
			ans = max(ans, j + min(b_poss_x, a_poss_y))
		print(ans)
	else:
		ans = max(ans, y // b)
		print(ans)
