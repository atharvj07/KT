#include <bits/stdc++.h>
using namespace std;
long long int mod = 1000000007;
vector<long long int> A;
vector<long long int> B;
set<long long int> ans;
int main() {
  long long int i, j, k, l, m, n, p, q, x, y, z, a, b, r;
  scanf("%lld", &k);
  scanf("%lld", &n);
  A.resize(k);
  for (i = 0; i < k; i++) scanf("%lld", &A[i]);
  B.resize(n);
  for (i = 0; i < n; i++) scanf("%lld", &B[i]);
  for (i = 1; i < k; i++) A[i] += A[i - 1];
  sort(A.begin(), A.end());
  sort(B.begin(), B.end());
  for (i = 0; i + n - 1 < k; i++) {
    x = A[i] - B[0];
    l = i;
    j = 0;
    while (j < n) {
      while (l < k && A[l] - B[j] < x) l++;
      if (l >= k || A[l] - B[j] != x) break;
      j++;
    }
    if (j >= n) {
      ans.insert(x);
    }
  }
  cout << (int)ans.size() << endl;
  return 0;
}
