template<class T,class U>
void chmin(T& t,U u){
	if(ll(t)>ll(u))
		t=u;
}

template<class T,class U>
void chmax(T& t,U u){
	if(ll(t)<ll(u))
		t=u;
}

int main(){
	ll a,b,n;
	cin>>a>>b>>n;
	ll ans=LLONG_MAX;
	for(ll q=1;q<=n;q++){
		if(b%q!=0)
			continue;
		ll k=b/q;
		ll x=a/k*k;
		ll y=x+k;
		ll z=k*n;
		if(0<x&&x<=z)
			chmin(ans,abs(a-x));
		if(0<y&&y<=z)
			chmin(ans,abs(a-y));
		if(0<z)
			chmin(ans,abs(a-z));
	}
	cout<<ans<<endl;
}
