#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int x, y, z, k;
  cin >> x >> y >> z >> k;
  while (x + y + z - 3 > k) {
    if (x > y && x > z) {
      x--;
    } else if (y > z) {
      y--;
    } else {
      z--;
    }
  }
  cout << x * y * z << endl;
}
