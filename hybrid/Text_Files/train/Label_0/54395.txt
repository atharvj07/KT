import java.io.*; import java.util.*;
public class VillageMax {

    /**
     * Centroid decomposition problem
     * Answer is twice the sum of the size of the smaller subtree for an edge over all edges
     */
    static ArrayList<Integer> arl[];
    static int N;
    static long ans;
    static int[] siz,par;//Size of subtree
    static int cent=-1;//Centroid
    public static class Pair implements Comparable<Pair>{
        int v; int siz;
        public Pair(int a, int b){
            this.v=a; this.siz=b;
        }
        public int compareTo(Pair other){
            if(this.siz<=other.siz)return -1;
            return 1;
            
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        N=Integer.parseInt(br.readLine());
        
         arl=new ArrayList[N];
         for (int i = 0; i < N; i++) {
            arl[i]=new ArrayList<>();
        }
         int a,b;
         for (int i = 0; i < N-1; i++) {
            StringTokenizer st=new StringTokenizer(br.readLine());
            a=Integer.parseInt(st.nextToken())-1;
            b=Integer.parseInt(st.nextToken())-1;
            arl[a].add(b); arl[b].add(a);
        }
         siz=new int[N];
         par=new int[N];
          for (int i = 0; i < N; i++) {
            par[i]=i;
        }
         Arrays.fill(siz,1);
         init(-1,0); 
        
         dfs(-1,0,true);
         
         //Now we use rerooting to find the centroid of the tree and the answer
         Arrays.fill(siz,1);//Redo
         
         init(-1,cent);//Root around center
         TreeSet<Pair> ts=new TreeSet<>();
         
        
         
       
         for (int i : arl[cent]) {
            
            ts.add(new Pair(i,siz[i]));
            
            
         }
         
         ArrayList<Integer> arl2[]=new ArrayList[N];
         for (int i = 0; i < N; i++) {
            arl2[i]=new ArrayList<>();
        }
        for (int i = 0; i < N; i++) {
            if(i!=cent){
                arl2[find(i,par)].add(i);
            }
        }
        
       
        int[] res=new int[N];
        int sum=N-1;
        while(sum>2){
            Pair p=ts.pollLast();
            Pair q=ts.pollLast();
            
            res[arl2[q.v].get(q.siz-1)]=arl2[p.v].get(p.siz-1); 
            res[arl2[p.v].get(p.siz-1)]=arl2[q.v].get(q.siz-1);
            arl2[q.v].remove(q.siz-1); arl2[p.v].remove(p.siz-1);
            if(q.siz>1){
                ts.add(new Pair(q.v,q.siz-1));
            }
            if(p.siz>1){
                ts.add(new Pair(p.v,p.siz-1));
            }
            sum-=2;
        }
        if(sum==1){
            Pair p=ts.pollLast();
            res[cent]=arl2[p.v].get(0);
            res[arl2[p.v].get(0)]=cent;
        }else{
            Pair p=ts.pollLast();
            Pair q=ts.pollLast();
            res[cent]=arl2[p.v].get(0);
            res[arl2[p.v].get(0)]=arl2[q.v].get(0);
            res[arl2[q.v].get(0)]=cent;
        }
        System.out.println(ans);
        StringBuilder sb=new StringBuilder();
        for (int i = 0; i <N; i++) {
            sb.append(res[i]+1).append(" ");
        }
        System.out.println(sb.toString());
    }
    public static void init(int p, int v){
        
       for (int i : arl[v]) {
            if(i==p)continue;
            init(v,i);
            siz[v]+=siz[i];
            if(cent>=0 && v!=cent){
                
                par[i]=v;
                
            }
        }
    }
    public static void dfs(int p, int v, boolean b){//Rerooting
        if(p>=0){
            
            ans+=2*Math.min(N-siz[p],siz[p]);
            
        }
        int max=0; int ind=-1;
        for (int i : arl[v]) {
            if(i==p)continue;
            if(siz[i]>max){
                max=siz[i]; 
                if(b){//Index is only used for centroid finding
                    ind=i;
                }
            }
            
        }
        
        if(b && 2*max>N){
            siz[v]-=siz[ind];
            siz[ind]+=siz[v];
            dfs(v,ind,true);
            siz[ind]-=siz[v];
            siz[v]+=siz[ind];
        }else if(b){
            
            ind=-1; cent=v;//Centroid found
        }
        for (int i : arl[v]) {
            if(i==p || i==ind)continue;
            siz[v]-=siz[i];
            siz[i]+=siz[v];
            dfs(v,i,false);
            siz[i]-=siz[v];
            siz[v]+=siz[i];
        }
    }
    public static int find(int x, int[] p){
        if(p[x]==x)return x;
        int ans=find(p[x],p);
        p[x]=ans; return ans;
    }
}
