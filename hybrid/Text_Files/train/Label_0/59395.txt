var n = parseInt(readline());

var l = readline().trim().split(' ').map((x)=>parseInt(x));

var ans = 0;

var isOne = false;
var nbrOnes=0;
for(var i=0;i<n;i++){
    if(l[i]==0){
        if(nbrOnes>0)
        ans+=nbrOnes+1;
        nbrOnes=0;
        isOne = false;
    }else{
        isOne=true;
        nbrOnes++;
    }
}
if(l[n-1]==0)
    ans=ans==0?0:(ans-1);
else
    ans+=nbrOnes;
print(ans)
