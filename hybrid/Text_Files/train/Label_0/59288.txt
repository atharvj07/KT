import java.util.Arrays;
import java.util.Scanner;

public class C {
    static char[] str;
    static long[][][][] dp;

    public static long solve(int st, int end, int invertedSt, int invertedEnd) {
//      System.out.println(st + " " + end + " " + invertedSt + " "
//              + invertedEnd);
        if (st == str.length || (st == end && st != 0))
            return 1;
        if (dp[st][end][invertedSt][invertedEnd] != -1)
            return dp[st][end][invertedSt][invertedEnd];

        long res = 0;
        if (st == 0) {
            if (str[st] == 'B') {
                int prev = (st - 1 + str.length) % str.length;
                if (str[prev] != 'A') {
                    res += solve(st + 1, 0, 0, 0);
                    int next = (st + 1) % str.length;
                    if (str[next] == 'A') {
                        res += solve(st + 2, 0, 1, 1);
                    }
                } else {
                    int beforePrev = prev - 1;
                    int next = (st + 1) % str.length;
                    //System.out.println(beforePrev + " " + prev);
                    if (str[beforePrev] == 'B') {
                        //System.out.println("dasj");
                        res += solve(st + 1, beforePrev, 0, 1);
                        //if (str[next] == 'A') {
                            //res += solve(st + 2, beforePrev, 1, 1);
                        //}
                    }
                    if (str[next] == 'A') {
                        res += solve(st + 2, 0, 1, 1);
                    }
                }
            } else {
                res += solve(st + 1, 0, 0, 0);
                int prev = (st - 1 + str.length) % str.length;
                if (str[prev] == 'B') {
                    res += solve(st + 1, str.length - 1, 1, 1);
                }
            }
        } else {
            if (str[st] == 'B') {
                int prev = st - 1;
                boolean mustChange = false;
                if (str[prev] == 'A' && invertedSt == 0) {
                    mustChange = true;
                }
                if (!mustChange)
                    res += solve(st + 1, end, 0, invertedEnd);
                int next = (st + 1) % str.length;
                if (next != end) {
                    // if (invertedEnd == 1 && str[next] == 'B'
                    // || invertedEnd == 0 && str[next] == 'A') {
                    // res += solve(st + 1, end, 0, invertedEnd);
                    // }
                    // } else {
                    if (str[next] == 'A') {
                        res += solve(st + 2, end, 1, invertedEnd);
                    }
                }
            } else {
                // int next = (st + 1) % str.length;
                // if (str[next] == 'B'&&next<end||(next==end&&(invertedEnd == 1
                // && str[next] == 'A'
                // || invertedEnd == 0 && str[next] == 'B'))) {
                // int prev = (st - 1 + str.length) % str.length;
                // if (str[prev] == 'A' && invertedSt == 0 || str[prev] == 'B'
                // && invertedSt == 1) {
                // res += solve(st + 1, end, 1, invertedEnd);
                // }
                // } else {
                res += solve(st + 1, end, 0, invertedEnd);
                int prev = (st - 1 + str.length) % str.length;
                if (str[prev] == 'B' && invertedSt == 1) {
                    res += solve(st + 1, end, 1, invertedEnd);
                }
                // }
            }
        }
        return dp[st][end][invertedSt][invertedEnd] = res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        str = sc.next().toCharArray();
        dp = new long[str.length][str.length + 1][2][2];
        for (int i = 0; i < dp.length; i++) {
            for (int j = 0; j <= str.length; j++) {
                for (int k = 0; k < 2; k++)
                    Arrays.fill(dp[i][j][k], -1);
            }
        }
        System.out.println(solve(0, 0, 0, 0));
        sc.close();
    }
}
