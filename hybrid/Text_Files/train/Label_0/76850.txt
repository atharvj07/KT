import sys
input = sys.stdin.readline
write = sys.stdout.write
n = int(input())
point = []
for i in range(n):
	point.append(list(map(int, input().split())) + [i + 1])
for _ in range(n // 2):
	(ax, ay, az, a) = point.pop()
	temp = 10 ** 18
	next = -1
	for i in range(len(point)):
		(bx, by, bz, _) = point[i]
		dist = (bx - ax) * (bx - ax) + (by - ay) * (by - ay) + (bz - az) * (bz - az)
		if temp >= dist:
			next = i
			temp = dist
	b = point.pop(next)
	print(a, b[3])
