#include <bits/stdc++.h>
using namespace std;
multiset<int,greater<int>> s[105];
int32_t main()
{
     int t = 1;
     cin >> t;
     while(t--)
     {
          int n,m;
          cin>>n>>m;
          vector<vector<int>> b(n+1,vector<int>(m+1));
          vector<vector<int>> ans(m+1,vector<int>(n+1));
          multiset<pair<int, int>> overall;
          for(int i=1;i<=n;i++)
          {
               for(int j=1;j<=m;j++)
               {
                    cin>>b[i][j];
                    overall.insert({b[i][j],i});
                    s[i].insert(b[i][j]);
               }      
          }
          int runner=1;
          while(!overall.empty())
          {
               pair<int, int> p=*overall.begin();
               overall.erase(overall.begin());
               ans[runner][p.second]=p.first;
               s[p.second].erase(s[p.second].find(p.first));
               for(int i=1;i<=n;i++)
               {
                    if(i==p.second)continue;
                    int max1=*s[i].begin();
                    s[i].erase(s[i].begin());
                    overall.erase(overall.find({max1,i}));
                    ans[runner][i]=max1;
               }
               runner++;
          }

          for(int i=1;i<=n;i++) s[i].clear();
          assert(overall.empty());
          for(int i=1;i<=n;i++)
          {
               for(int j=1;j<=m;j++)
               {
                    cout<<ans[j][i]<<" ";
               }
               cout<<"\n";
          }
     }
     return 0;
}
