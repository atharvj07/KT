import java.util.Scanner;

public class P821B
{
	public static void main(String[] args)
	{
		Scanner scan = new Scanner(System.in);
		int m = scan.nextInt();
		int b = scan.nextInt();
		long max = 0;
		int y = b;
		int rem = 0;
		int x = 0;
		while (y >= 0)
		{
			max = Math.max(max, sum(x)*(y+1) + sum(y)*(x+1));
			if (rem == 0)
			{
				y--;
				rem = m;
			}
			rem--;
			x++;
		}
		
		System.out.println(max);
	}

	private static long sum(long n)
	{
		return n*(n+1)/2;
	}
}
