def main(n, a):
	x = [i for i in a if i >= 0]
	y = [-i for i in a if i < 0]
	if sum(x) != sum(y):
		winner = sum(y) > sum(x)
	else:
		winner = [j > i for (i, j) in zip(x, y) if j != i]
		if winner != []:
			winner = winner[0]
		else:
			winner = a[-1] < 0
	print(['first', 'second'][winner])

def main_input():
	n = int(input())
	a = [int(input()) for i in range(n)]
	main(n, a)
main_input()
