#include <bits/stdc++.h>
using namespace std;
void solve() {
  int p, f;
  cin >> p >> f;
  int cntS, cntW;
  cin >> cntS >> cntW;
  int s, w;
  cin >> s >> w;
  int mx = -1;
  for (int a = 0; a <= cntS; a++) {
    if (a * s > p) break;
    int np = p - a * s;
    int q = a;
    int ncntW = cntW;
    int b = min(ncntW, np / w);
    q += b;
    ncntW -= b;
    int r = 0;
    int ncntS = max(cntS - a, 0);
    int nf = f;
    if (s < w) {
      int c = min(ncntS, nf / s);
      r += c;
      nf -= c * s;
      c = min(ncntW, nf / w);
      r += c;
    } else {
      int c = min(ncntW, nf / w);
      r += c;
      nf -= c * w;
      c = min(ncntS, nf / s);
      r += c;
    }
    mx = max(mx, q + r);
  }
  cout << mx << "\n";
}
signed main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) solve();
}
