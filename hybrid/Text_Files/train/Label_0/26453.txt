#include <bits/stdc++.h>
using namespace std;
int n, m, k, ans, x = 10010, cnt = 0, head[100010], dis[100010];
int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};
bool vis[100010];
map<pair<int, int>, int> mp;
pair<int, int> P[100010];
struct edge {
  int v, w, next;
} e[500010];
void add(int u, int v, int w) {
  e[++cnt] = (edge){v, w, head[u]};
  head[u] = cnt;
}
void dijkstra(int s) {
  memset(dis, 0x3f, sizeof(dis));
  priority_queue<pair<int, int>, vector<pair<int, int> >,
                 greater<pair<int, int> > >
      que;
  que.push(pair<int, int>(0, s));
  vis[s] = 1;
  dis[s] = 0;
  while (!que.empty()) {
    int u = que.top().second;
    que.pop();
    vis[u] = 1;
    for (int i = head[u]; i; i = e[i].next)
      if (!vis[e[i].v] && dis[e[i].v] > dis[u] + e[i].w)
        dis[e[i].v] = dis[u] + e[i].w,
        que.push(pair<int, int>(dis[e[i].v], e[i].v));
  }
}
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1, r, c; i <= k; i++) {
    scanf("%d%d", &r, &c);
    mp[pair<int, int>(r, c)] = i;
    P[i] = pair<int, int>(r, c);
    add(i, r + 1 + x, 1);
    add(i, r - 1 + x, 1);
    add(i, c + 1 + x * 2, 1);
    add(i, c - 1 + x * 2, 1);
    add(i, r + x, 1);
    add(i, c + x * 2, 1);
    add(r + 1 + x, i, 0);
    add(r - 1 + x, i, 0);
    add(c + 1 + x * 2, i, 0);
    add(c - 1 + x * 2, i, 0);
    add(r + x, i, 0);
    add(c + x * 2, i, 0);
  }
  for (int i = 1; i <= k; i++) {
    for (int j = 0; j < 4; j++) {
      int nx = P[i].first + dx[j], ny = P[i].second + dy[j];
      if (mp.find(pair<int, int>(nx, ny)) == mp.end()) continue;
      add(i, mp[pair<int, int>(nx, ny)], 0);
    }
  }
  if (mp.find(pair<int, int>(n, m)) == mp.end())
    add(n + x, ++k, 0), add(m + x * 2, k, 0);
  dijkstra(mp[pair<int, int>(1, 1)]);
  if (mp.find(pair<int, int>(n, m)) == mp.end())
    ans = dis[k];
  else
    ans = dis[mp[pair<int, int>(n, m)]];
  printf("%d", ans == 0x3f3f3f3f ? -1 : ans);
  return 0;
}
