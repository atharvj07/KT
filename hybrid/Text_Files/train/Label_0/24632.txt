#include <bits/stdc++.h>
using namespace std;
template <class T>
T min(T a, T b, T c) {
  return min(a, min(b, c));
}
template <class T>
T max(T a, T b, T c) {
  return max(a, max(b, c));
}
int n, a[5005], cnt[5005], ans[5005];
int main() {
  scanf("%d", &n);
  for (int i = (1); i <= (n); ++i) scanf("%d", &a[i]);
  for (int l = (1); l <= (n); ++l) {
    int curMax = 0, curCol = 0;
    memset(cnt, 0, sizeof cnt);
    for (int r = (l); r <= (n); ++r) {
      ++cnt[a[r]];
      if (cnt[a[r]] > curMax || (cnt[a[r]] == curMax && a[r] < curCol)) {
        curMax = cnt[a[r]];
        curCol = a[r];
      }
      ++ans[curCol];
    }
  }
  for (int i = (1); i <= (n); ++i) printf("%d ", ans[i]);
  return 0;
}
