def g(f):
	linein = f + ' '
	k = -1
	n = 0
	minus = False
	a = [0, 0, 0]
	for i in linein:
		k += 1
		if i != ' ':
			if linein[k + 1] == ' ':
				a[n] = a[n] * 10 + int(i)
				if minus:
					a[n] *= -1
					minus = False
				n += 1
			elif i == '-':
				minus = True
			else:
				a[n] = a[n] * 10 + int(i)
	n = 0
	if a[1] != 1:
		for i in range(a[2]):
			n += (a[0] - 4 * i) * (a[1] - 4 * i) - (a[0] - 2 - 4 * i) * (a[1] - 2 - 4 * i)
	else:
		n = a[0]
	return n
print(g(input()))
