#include <bits/stdc++.h>
using namespace std;
const int MAXN = 502;
int main() {
  int i;
  int N, M;
  double v[MAXN];
  cin >> N >> M;
  double ans = 0;
  for (i = 0; i < N; i++) cin >> v[i];
  for (i = 0; i < M; i++) {
    long long a, b, price;
    cin >> a >> b >> price;
    --a;
    --b;
    double now = (v[a] + v[b]) / price;
    if (now > ans) ans = now;
  }
  printf("%.9f\n", ans);
}
