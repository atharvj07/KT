#include <iostream>
#include <vector>
#include <string>
#include <set>
#include <algorithm>

using namespace std;

int n;
int num[10];
string edge[10][10];
string tmp;

int main(){
  vector<string> in;
  set<string> data;
  while(cin >> n && n){
    in.clear();
    data.clear();
    for(int i=0;i<n;i++){
      string str;
      cin >> str;
      in.push_back(str);
    }
    for(int i=0;i<n;i++){
      for(int j=0;j<n;j++){
        if(i == j) continue;
        if(in[i].size() >= in[j].size()){
          for(int k=0;k+in[j].size()<=in[i].size();k++){
            if(in[i].substr(k, in[j].size()) == in[j]) data.insert(in[j]);
          }
        }
      }
    }
    for(set<string>::iterator i=data.begin();i!=data.end();i++){
      for(vector<string>::iterator ite=in.begin();ite!=in.end();ite++){
        if(*ite == *i){
          in.erase(ite);
          n--;
          break;
        }
      }
    }
    for(int i=0;i<in.size();i++){
      for(int j=0;j<in.size();j++){
        if(i == j){
          continue;
        }
        int size = min(in[i].size(),in[j].size());
        for(int k=size-1;k>=0;k--){
          if(in[i].substr(in[i].size()-k, k) == in[j].substr(0, k)){
            edge[i][j] = in[j].substr(k);
            break;
          }
        }
      }
    }
    int valm = 100000;
    string ans = in[0];
    for(int i=0;i<10;i++) num[i] = i;
    do{
      tmp = in[num[0]];
      for(int i=1;i<n;i++){
        tmp += edge[num[i-1]][num[i]];
      }
      if(tmp.size() < valm){
        ans = tmp;
        valm = tmp.size();
      }else if(tmp.size() == valm){
        ans = min(ans, tmp);
      }
    }while(next_permutation(num, num+n));
    cout << ans << endl;
  }
  return 0;
}
