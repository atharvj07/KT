#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <cstring>

using namespace std;

int main(){

	int n;
	int card[3][10];

	cin>>n;

	while(n--){
			
		int a[10];
			
		memset(card, 0, sizeof(card));

		for(int i=0; i<9; ++i){
			cin>>a[i];
		}

		for(int i=0; i<9; ++i){

			char b;
			cin>>b;
			card[b=='R'?0:b=='G'?1:2][a[i]]++;
		}

		int cnt = 0;

		for(int c=0; c<3; ++c){
			for(int k=1; k<=9; ++k){
				
				cnt += card[c][k]/3;
				card[c][k] %= 3;
				
				if(k<=7){
					
					int sub = min(card[c][k], min(card[c][k+1], card[c][k+2]));

					for(int i=0; i<3; ++i){
						card[c][k+i] -= sub;
					}

					cnt += sub;
				}
			}
		}

		cout << (cnt>=3) << endl;
	}	
	
	return 0;
}
