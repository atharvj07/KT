def extgcd(a, b):
	if b == 0:
		x = 1
		y = 0
		return (a, x, y)
	(d, x, y) = extgcd(b, a % b)
	(x, y) = (y, x - y * (a // b))
	return (d, x, y)
(n, m, k) = map(int, inp().split())
A = [*map(int, inp().split())]
B = [*map(int, inp().split())]
(d, x, y) = extgcd(n, m)
l = n * m // d
ax = [-1] * 1000001
good_days = [l]
cnt = l
for i in range(n):
	ax[A[i]] = i
for i in range(m):
	if ax[B[i]] >= 0 and ax[B[i]] % d == i % d:
		c = i - ax[B[i]]
		y0 = (n * x * c // d + ax[B[i]]) % l
		good_days.append(y0)
		cnt -= 1
good_days.sort()
days = (k - 1) // cnt * l
rem = (k - 1) % cnt + 1
for (i, day) in enumerate(good_days):
	if day - i >= rem:
		days += rem + i
		break
print(days)
