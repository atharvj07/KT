import sys
inpu = sys.stdin.readline
prin = sys.stdout.write
(n, m) = map(int, inpu().split())
matrica = []
losekolone = [0 for i in range(m + 1)]
for i in range(0, n):
	matrica.append(inpu())
if n != 1 and m != 1:
	for j in range(0, m - 1):
		for i in range(1, n):
			if matrica[i][j] == 'X' and matrica[i - 1][j + 1] == 'X':
				losekolone[j + 1] += 1
				break
		losekolone[j + 1] += losekolone[j]
q = int(inpu())
for i in range(0, q):
	(xj, xd) = map(int, inpu().split())
	if losekolone[xj - 1] == losekolone[xd - 1]:
		prin('YES\n')
	else:
		prin('NO\n')
