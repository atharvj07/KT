//package d;
import java.util.*;
import java.io.*;

public class Main {

	public static void main(String[] args) throws Exception{
		Scanner in = new Scanner(new File("input.txt"));
		PrintWriter out = new PrintWriter(new File("output.txt"));
		
		int n = in.nextInt();
		int m = in.nextInt();
		int a[] = new int[n];
		for(int i=0;i<n;i++) a[i] = in.nextInt();
		
		int f[] = new int[n];
		for(int i=0;i<n;i++) f[i] = a[i]*(n-i);
		Arrays.sort(f);
		int ret = 0;
		int sum = 0;
		int cur = 0;
		while (cur < n && sum + f[cur] <= m){
			sum += f[cur];
			cur++;
			ret++;
		}
		out.println(ret);
		out.close();
	}
}
