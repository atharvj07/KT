from collections import Counter
(n, k) = map(int, input().split())
cards = [card for card in input().strip()]
options = sorted(Counter(cards).values(), reverse=True)
choices = []
for option in options:
	if k >= option:
		choices.append(option)
		k -= option
	else:
		choices.append(k)
		break
print(sum([choice ** 2 for choice in choices]))
