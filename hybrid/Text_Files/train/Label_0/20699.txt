#include <bits/stdc++.h>
using namespace std;
const long long maxn = 2e5 + 100;
const long long mod = 1e9 + 7;
const long long inf = 1e18;
long long n, ps[2][26][maxn];
__int128 ans;
string a, b;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  cout << setprecision(10) << fixed;
  cin >> n >> a >> b;
  for (long long i = n; i--;) {
    for (long long j = 0; j < 26; j++) {
      ps[0][j][i] = ps[0][j][i + 1];
      ps[1][j][i] = ps[1][j][i + 1];
    }
    ps[0][a[i] - 'A'][i] += n - i;
    ps[1][b[i] - 'A'][i] += n - i;
  }
  for (long long i = 0; i < n; i++) {
    ans += (i + 1) * ps[1][a[i] - 'A'][i];
    ans += (i + 1) * ps[0][b[i] - 'A'][i];
    if (a[i] == b[i]) ans -= (i + 1) * (n - i);
  }
  cout << ans * 6.0 / (n * (n + 1) * (2 * n + 1));
  return 0;
}
