n = int(input())
primes = [x for x in range(2, int(n / 2) + 1) if all((x % y != 0 for y in range(2, int(x ** 0.5) + 1)))]
primes = primes[::-1]
used = {}
for i in range(2, n + 1):
	used[i] = False

def solve(n, primes, used):
	outputs = []
	doubles = []
	counter = 0
	if n == 1:
		return (0, 0)
	for i in primes:
		multiples = []
		for x in range(1, n // i + 1):
			if used[i * x] == False:
				multiples.append(i * x)
				used[i * x] = True
		if len(multiples) % 2 == 0:
			for j in range(0, int(len(multiples)), 2):
				outputs.append((multiples[j], multiples[j + 1]))
				counter += 1
		else:
			multiples.pop(1)
			used[i * 2] = False
			for j in range(0, int(len(multiples)), 2):
				outputs.append((multiples[j], multiples[j + 1]))
				counter += 1
	return (outputs, counter)
a = solve(n, primes, used)
print(a[1])
if a[0] != 0:
	for i in a[0]:
		print(*i)
