from sys import stdin, stdout, exit

def price():
	(n, k) = map(int, stdin.readline().split())
	s = stdin.readline().strip()
	todo = [s]
	ans = 0
	found = {s: True}
	k -= 1
	while todo and k > 0:
		nexts = []
		for wd in todo:
			for i in range(len(wd)):
				candidate = wd[:i] + wd[i + 1:]
				if candidate not in found:
					nexts.append(candidate)
					found[candidate] = True
					ans += n - len(candidate)
					k -= 1
					if k == 0:
						return ans
		todo = nexts
	if k == 0:
		return ans
	return -1
print(price())
