#include <bits/stdc++.h>
using namespace std;
long long a[200005], sum[200005];
int n;
int ansl = 0, ansi = 1;
double ans = 0;
bool pd(int mid, int midmid, int x) {
  long long a, b;
  a = sum[x] - sum[x - mid - 1] + sum[n] - sum[n - mid];
  b = sum[x] - sum[x - midmid - 1] + sum[n] - sum[n - midmid];
  if (a * (2 * midmid + 1) > b * (2 * mid + 1))
    return true;
  else
    return false;
}
void solve(int x) {
  int l, r, mid, midmid;
  double val;
  l = 0;
  r = min(x - 1, n - x);
  while (l + 1 < r) {
    mid = (l + r) / 2;
    midmid = (mid + r) / 2;
    if (pd(mid, midmid, x))
      r = midmid;
    else
      l = mid;
  }
  if (pd(r, l, x)) l = r;
  val = (double)(sum[x] - sum[x - l - 1] + sum[n] - sum[n - l]) /
            (double)(2 * l + 1) -
        a[x] * 1.0;
  if (val > ans) {
    ans = val;
    ansl = l;
    ansi = x;
  }
}
int main() {
  int i;
  cin >> n;
  for (i = 1; i <= n; i++) cin >> a[i];
  sort(a + 1, a + n + 1);
  for (i = 1; i <= n; i++) sum[i] = sum[i - 1] + a[i];
  for (i = 1; i <= n; i++) solve(i);
  cout << 2 * ansl + 1 << endl;
  for (i = ansi - ansl; i <= ansi; i++) cout << a[i] << " ";
  for (i = n - ansl + 1; i <= n; i++) cout << a[i] << " ";
  puts("");
  return 0;
}
