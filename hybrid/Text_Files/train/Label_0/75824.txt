import collections

def same(a, b):
	a = list(a)
	b = list(b)
	return sum([1 * (a == b) for (a, b) in zip(a, b)])
f = int(input())
(h, m) = input().split(':')
if int(m) > 59:
	m = '0' + m[1]
if f == 24:
	l = ['0' + str(x) for x in range(0, 10, 1)]
	for x in range(10, 24, 1):
		l.append(str(x))
	e = {x: same(x, h) for x in l}
	stats = collections.Counter(e)
	print(stats.most_common(1)[0][0] + ':' + m)
else:
	l = ['0' + str(x) for x in range(1, 10, 1)]
	l.append('10')
	l.append('11')
	l.append('12')
	e = {x: same(x, h) for x in l}
	stats = collections.Counter(e)
	print(stats.most_common(1)[0][0] + ':' + m)
