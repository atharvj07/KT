#include <bits/stdc++.h>
using namespace std;
template <typename T1, typename T2>
ostream &operator<<(ostream &os, const pair<T1, T2> &p) {
  os << '(' << p.first << "; " << p.second << ')';
};
template <typename T>
ostream &operator<<(ostream &os, const vector<T> &vec) {
  os << '[';
  if (!vec.empty())
    for_each(vec.begin(), vec.end() - 1,
             [&os](const T &elem) { os << elem << ", "; }),
        os << vec.back();
  os << ']';
  return os;
}
template <typename T1, typename T2>
ostream &operator<<(ostream &os, const map<T1, T2> &mp) {
  os << '[';
  if (!mp.empty()) {
    auto end_it = --mp.end();
    for (auto it = mp.begin(); it != end_it; ++it) os << *it << ", ";
    os << *mp.rbegin();
  }
  os << ']';
  return os;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  string s;
  cin >> s;
  vector<int> psum;
  for (int i = 0; i != n - 1; ++i)
    psum.emplace_back((psum.empty() ? 0 : psum.back()) + s[i] - '0');
  function<bool(int, int)> swappable = [&](int pos1, int pos2) {
    if (pos1 == pos2) return true;
    if (pos1 > pos2) swap(pos1, pos2);
    int len = pos2 - pos1;
    return psum.at(pos2 - 1) - (pos1 ? psum.at(pos1 - 1) : 0) == len;
  };
  map<int, int> mp;
  for (unsigned i = 0; i != n; ++i) mp[a[i]] = i;
  for (unsigned i = 0; i != n; ++i)
    if (a[i] != i + 1)
      if (swappable(i, mp[i + 1])) {
        int cur = i, target = mp[i + 1];
        mp[a[cur]] = target, mp[a[target]] = cur;
        swap(a[cur], a[target]);
      } else {
        cout << "NO";
        return 0;
      }
  cout << "YES";
  return 0;
}
