#include <bits/stdc++.h>
using namespace std;
int main() {
  map<int, int> mp;
  int n, q, tmp, cnt2 = 0, cnt4 = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &tmp);
    mp[tmp]++;
    if (mp[tmp] % 2 == 0) cnt2++;
    if (mp[tmp] % 4 == 0) cnt4++;
  }
  cin >> q;
  char ch[2];
  while (q--) {
    scanf("%s %d", ch, &tmp);
    if (ch[0] == '+') {
      mp[tmp]++;
      if (mp[tmp] % 2 == 0) cnt2++;
      if (mp[tmp] % 4 == 0) cnt4++;
    } else {
      if (mp[tmp] % 2 == 0) cnt2--;
      if (mp[tmp] % 4 == 0) cnt4--;
      mp[tmp]--;
    }
    if (cnt4 >= 1 && cnt2 >= 4)
      puts("YES");
    else
      puts("NO");
  }
  return 0;
}
