#include<bits/stdc++.h>
using namespace std;

int main(){
	int n;
	cin>>n;
	for (int i=n/4+1;i<=3500;i++){
		for (int j=i;j<=3500;j++){
			long long k=(long long)i*j*n;
			long long p=(long long)4*i*j-n*i-n*j;
			if (p>0){
				if (k%p==0 && k/p<=3500){
					cout<<i<<" "<<j<<" "<<k/p<<endl;
					return 0;
				}
			}
		}
	}
}
