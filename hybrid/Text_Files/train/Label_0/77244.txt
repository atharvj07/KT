def solve():
	(n, m, x, y) = list(map(int, input().split()))
	visited = set()
	print('{} {}'.format(x, y))
	visited.add((x, y))
	y = 1
	print('{} {}'.format(x, y))
	visited.add((x, y))
	top = True
	for col in range(1, m + 1):
		if top == True:
			for row in range(1, n + 1):
				if (row, col) not in visited:
					print('{} {}'.format(row, col))
					visited.add((row, col))
			top = False
		else:
			for row in range(n, 0, -1):
				if (row, col) not in visited:
					print('{} {}'.format(row, col))
					visited.add((row, col))
			top = True
solve()
