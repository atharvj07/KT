import math
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	gcd = [a[0], a[1]]
	for i in range(2, n):
		gcd[i % 2] = math.gcd(gcd[i % 2], a[i])
	f1 = True
	f2 = True
	for i in range(0, n):
		if i % 2 == 0:
			if a[i] % gcd[1] == 0:
				f1 = False
		elif a[i] % gcd[0] == 0:
			f2 = False
	if f1:
		print(gcd[1])
	elif f2:
		print(gcd[0])
	else:
		print(0)
