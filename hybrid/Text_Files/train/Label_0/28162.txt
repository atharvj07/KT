#include <bits/stdc++.h>
const long double PI = 3.141592653589793236L;
using namespace std;
void solve() {
  long long int n;
  cin >> n;
  vector<long long int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) b[i] = a[i];
  sort(a.begin(), a.end());
  sort(b.begin(), b.end(), greater<int>());
  long long int ans = 0;
  for (int i = 0; i < n; i++) {
    ans = (ans % 10007 + (a[i] * b[i]) % 10007) % 10007;
  }
  cout << ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int T = 1;
  while (T--) {
    solve();
  }
  return 0;
}
