import sys
import io

def run(test, res):
	x = io.StringIO()
	with io.StringIO(test) as sys.stdin:
		with x as sys.stdout:
			work()
			z = x.getvalue().strip()
	sys.stdout = sys.__stdout__
	print(z == test)
	print(res)
	print(z)

def work():
	n = int(input())
	l = list(map(int, input().split()))
	a = [[], [], [], []]
	for (i, x) in enumerate(l):
		a[x].append(i + 1)
	z = min(len(a[1]), len(a[2]), len(a[3]))
	print(z)
	if z > 0:
		for i in range(0, z):
			print(a[1][i], a[2][i], a[3][i])

def test():
	run('7\n1 3 1 3 2 1 2\n', '-1')
	run('4\n2 1 1 2', '0')
if len(sys.argv) > 1:
	test()
else:
	work()
