(n, k) = map(int, input().split())
s = list(input())
ss = list(s)
kk = k
for i in range(n):
	z = ord('z') - ord(s[i])
	a = ord(s[i]) - ord('a')
	if k == 0:
		break
	if z > a:
		s[i] = chr(ord(s[i]) + min(k, z))
		k -= min(k, z)
	else:
		s[i] = chr(ord(s[i]) - min(k, a))
		k -= min(k, a)
sm = 0
for i in range(n):
	sm += abs(ord(s[i]) - ord(ss[i]))
if sm == kk:
	print(''.join(s))
else:
	print(-1)
