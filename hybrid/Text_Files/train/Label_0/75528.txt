(m, n) = map(int, input().split())

def x(l, y):
	t = int(l.replace(' ', ''), 2)
	return y ^ t

def bn(ls):
	s = ''
	for i in range(len(ls)):
		temp = ls[i].count('1') % 2
		s += str(temp)
	return s
a = []
b = []
for i in range(m):
	a.append(input())
for i in range(m):
	b.append(input())
g = h = 0
for i in range(m):
	g = x(a[i], g)
	h = x(b[i], h)
l = bn(a)
s = bn(b)
if l == s and g == h:
	print('YES')
else:
	print('NO')
