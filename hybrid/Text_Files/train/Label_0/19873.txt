#include <bits/stdc++.h>
using namespace std;
long long dp[1000001];
bool yes(char z) {
  if (z == 'a' || z == 'e' || z == 'i' || z == 'o' || z == 'u' || z == 'y')
    return true;
  return false;
}
int main() {
  long long n, m, k;
  string s;
  long long sum = 0;
  cin >> n >> s;
  for (int i = 1; i < n; i++) {
    if (yes(s[i])) {
      for (int j = i - 1; j >= 0; j--) {
        if (dp[j] == 1) continue;
        if (dp[j] == 0 && yes(s[j])) {
          dp[i] = 1;
          break;
        } else
          break;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (!dp[i]) {
      cout << s[i];
    }
  }
  cout << endl;
  return 0;
}
