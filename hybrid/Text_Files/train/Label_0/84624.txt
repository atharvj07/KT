n = int(input().strip())
arr = [int(i) for i in input().strip().split()]
for i in range(len(arr)):
	arr[i] = (arr[i], i + 1)
arr.sort(key=lambda x: x[0])
intro = 0
extro = []
inp = input().strip()
for i in inp:
	if i == '0':
		print(arr[intro][1], end=' ')
		extro.append(arr[intro][1])
		intro += 1
	else:
		print(extro.pop(), end=' ')
