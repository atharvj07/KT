for k in range(int(input())):
	(n, m) = map(int, input().split())
	arr = list(map(int, input().split()))
	if n > 2 and m == n:
		if m >= n:
			print(sum(arr) * 2)
			for i in range(n - 1):
				print(str(i + 1) + ' ' + str(i + 2))
			print(str(n) + ' ' + str(1))
		else:
			print(-1)
	else:
		print(-1)
