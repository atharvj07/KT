import java.util.*;
public class Main{
    public static void main(String [] args)
    {
        Scanner scan=new Scanner(System.in);
        int n=scan.nextInt();
        int m=scan.nextInt();
        ArrayList<Integer> arr=new ArrayList<Integer>();
        int ans=0;
        long sum=0;
        for(int i=0;i<n;i++)
        {
            int a=scan.nextInt();
            int b=scan.nextInt();
            sum +=a;
            arr.add(a-b);
        }
        Collections.sort(arr);
        for(int i=n-1 ; sum>m && i>=0 ;i--)
        {
            sum -=arr.get(i);
            ans++;
        }
        if(sum>m)
            System.out.println(-1);
        else
            System.out.println(ans);
        
    }
}
