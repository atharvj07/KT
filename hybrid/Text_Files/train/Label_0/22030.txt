#include <bits/stdc++.h>
using namespace std;
long long inf = 2147483647;
const long long ll = 9223372036854775807, ninf = 1000000000;
const double eps = 1e-6;
const long long nll = 1000000000000000000;
long long sum[200005] = {};
long long mod = 998244353;
long long qpow(long long now) {
  if (now == 0) return 1;
  long long ans = qpow(now / 2) * qpow(now / 2);
  ans %= mod;
  if (now % 2) {
    ans *= 2;
  }
  return ans;
}
int main() {
  int a, b;
  while (cin >> a >> b) {
    string c, d;
    cin >> c >> d;
    int now = 1;
    for (int i = c.size() - 1; i >= 0; i--) {
      sum[now] = sum[now - 1];
      if (c[i] == '1') {
        sum[now] += qpow(now - 1);
      }
      sum[now] %= mod;
      now++;
    }
    for (int i = now; i < 200005; i++) {
      sum[i] = sum[i - 1];
    }
    now = 1;
    long long ans = 0;
    for (int i = d.size() - 1; i >= 0; i--) {
      if (d[i] == '1') {
        ans += sum[now];
        ans %= mod;
      }
      now++;
    }
    cout << ans << endl;
  }
  return 0;
}
