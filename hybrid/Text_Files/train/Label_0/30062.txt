ll MOD=100000007;

ll dp1[100001][5];
ll dps1[100001];
ll dp2[100001][3];
ll dps2[100001];

int main(){
  string in;

  dp1[1][0]=1;
  dp2[1][0]=1;
  for(int i=2;i<=100000;++i){
    rep(j,5)
      dp1[i][0]+=dp1[i-1][j];

    rep(j,3)dp2[i][0]+=dp2[i-1][j];
    
    for(int j=1;j<5;++j){
      if(i-j>=0)dp1[i][j]+=dp1[i-j][0];
    }

    for(int j=1;j<3;++j){
      if(i-j>=0)dp2[i][j]+=dp2[i-j][0];
    }
    rep(j,5)dp1[i][j]%=MOD;
    rep(j,3)dp2[i][j]%=MOD;
  }



  rep(k,100001){
    int sum=0;    
    rep(i,5){
      //cout<<dp1[k][i]<<' ';
      sum+=dp1[k][i];
    }
    if(k>=5)sum+=dps1[k-5];
    //cout<<sum<<endl;
    dps1[k]=sum%MOD;

    sum=0;
    rep(i,3)sum+=dp2[k][i];
    if(k>=3)sum+=dps2[k-3];
    dps2[k]=sum%MOD;
  }
  
  while(cin>>in,in!="#"){
    int idx=0;
    ll ans=1;
    while(idx<SZ(in)){
      int num=0;
      while(idx+num<SZ(in) && in[idx]==in[idx+num])++num;
      if(in[idx]!='8' && in[idx]!='0')ans=(ans*dps1[num])%MOD;
      else ans=(ans*dps2[num])%MOD;
      idx+=num;
    }
    cout<<ans<<endl;
  }
}
