import collections
import itertools
n = int(input())
m = collections.defaultdict(lambda : 'x')
for y in range(n):
	for (x, ch) in enumerate(input()):
		m[x, y] = ch
deltas = ((-1, 0), (0, -1), (1, 0), (0, 1))
for p in itertools.product(range(n), repeat=2):
	cps = map(lambda dp: tuple(map(sum, zip(p, dp))), deltas)
	if len(tuple(filter(lambda cp: m[cp] == 'o', cps))) % 2:
		print('NO')
		exit()
print('YES')
