import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.Arrays;
import java.util.StringTokenizer;

public class CF292_B {

	/**
	 * @param args
	 */
	public static void main(String[] args) {

		MyScanner in = new MyScanner();
		String notUnique = "Not unique";

		int nRows = in.nextInt();
		int nCols = in.nextInt();

		char[][] board = new char[nRows + 2][nCols + 2];
		Arrays.fill(board[0], '*');
		Arrays.fill(board[board.length - 1], '*');
		for (int i = 1; i <= nRows; i++) {
			char[] row = in.next().toCharArray();
			board[i][0] = '*';
			board[i][board[i].length - 1] = '*';
			for (int j = 0; j < row.length; j++) {
				board[i][j + 1] = row[j];
			}
		}

		// Count empties
		int nEmpties = 0;
		for (int i = 0; i < board.length; i++) {
			for (int j = 0; j < board[i].length; j++) {
				if (board[i][j] == '.') {
					nEmpties++;
				}
			}
		}
		if (nEmpties % 2 == 1) {
			System.out.println(notUnique);
			return;
		}

		ArrayDeque<Integer> toProcI = new ArrayDeque<Integer>();
		ArrayDeque<Integer> toProcJ = new ArrayDeque<Integer>();
		ArrayDeque<Integer> toProcDir = new ArrayDeque<Integer>();
		int UP = 0;
		int DOWN = 1;
		int RIGHT = 2;
		int LEFT = 3;
		int[] delI = { 1, -1, 0, 0 };
		int[] delJ = { 0, 0, 1, -1 };

		// Find initial determined points
		for (int i = 0; i < board.length; i++) {
			for (int j = 0; j < board[i].length; j++) {
				if (board[i][j] == '.') {
					int count = 0;
					int dir = -1;
					if (board[i + 1][j] == '.') {
						count++;
						dir = UP;
					}
					if (board[i - 1][j] == '.') {
						count++;
						dir = DOWN;
					}
					if (board[i][j + 1] == '.') {
						count++;
						dir = RIGHT;
					}
					if (board[i][j - 1] == '.') {
						count++;
						dir = LEFT;
					}
					if (count == 1) {
						toProcI.add(i);
						toProcJ.add(j);
						toProcDir.add(dir);
					}
				}
			}
		}

		// Process points
		while (!toProcI.isEmpty()) {

			int i0 = toProcI.pop();
			int j0 = toProcJ.pop();
			int dir0 = toProcDir.pop();

			int toFillI = i0 + delI[dir0];
			int toFillJ = j0 + delJ[dir0];


			if (board[i0][j0] != '.' || board[toFillI][toFillJ] != '.') {
				continue;
			}

			nEmpties -= 2;

			if (dir0 == UP) {
				board[i0][j0] = '^';
				board[toFillI][toFillJ] = 'v';
			}
			if (dir0 == DOWN) {
				board[i0][j0] = 'v';
				board[toFillI][toFillJ] = '^';
			}
			if (dir0 == LEFT) {
				board[i0][j0] = '>';
				board[toFillI][toFillJ] = '<';
			}
			if (dir0 == RIGHT) {
				board[i0][j0] = '<';
				board[toFillI][toFillJ] = '>';
			}

			for (int iDirCheck = 0; iDirCheck < 4; iDirCheck++) {

				int i = toFillI + delI[iDirCheck];
				int j = toFillJ + delJ[iDirCheck];

				if (board[i][j] == '.') {
					int count = 0;
					int dir = -1;
					if (board[i + 1][j] == '.') {
						count++;
						dir = UP;
					}
					if (board[i - 1][j] == '.') {
						count++;
						dir = DOWN;
					}
					if (board[i][j + 1] == '.') {
						count++;
						dir = RIGHT;
					}
					if (board[i][j - 1] == '.') {
						count++;
						dir = LEFT;
					}
					if (count == 1) {
						toProcI.add(i);
						toProcJ.add(j);
						toProcDir.add(dir);
					}
				}

			}

		}

		if (nEmpties != 0) {
			System.out.println(notUnique);
			return;
		}

		StringBuilder sb = new StringBuilder();
		for (int i = 1; i < board.length - 1; i++) {
			for (int j = 1; j < board[i].length - 1; j++) {
				sb.append(board[i][j]);
			}
			sb.append('\n');
		}

		System.out.print(sb);

	}

	// -----------MyScanner class for faster input----------
	public static class MyScanner {
		BufferedReader br;
		StringTokenizer st;

		public MyScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}

		String nextLine() {
			String str = "";
			try {
				str = br.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}
			return str;
		}

	}
	// --------------------------------------------------------

}
