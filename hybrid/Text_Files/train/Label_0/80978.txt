t = int(input())
for i in range(t):
	(n, c0, c1, h) = map(int, input().split())
	s = input()
	total = 0
	count0 = s.count('0')
	count1 = s.count('1')
	if c1 > c0 + h:
		s = s.replace('1', '0')
		total = total + h * count1
	elif c0 > c1 + h:
		s = s.replace('0', '1')
		total = total + h * count0
	count0 = s.count('0')
	count1 = s.count('1')
	total = total + count0 * c0 + count1 * c1
	print(total)
