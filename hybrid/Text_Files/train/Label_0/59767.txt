//package Round_405;

import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Scanner;

public class E {

	public E() {
		Scanner sc = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = sc.nextInt();
		ArrayList<Integer> V = new ArrayList<Integer>(), K = new ArrayList<Integer>(), X = new ArrayList<Integer>();
		char[] s = sc.next().toCharArray();
		for (int i = 0; i < n; i++)
			if (s[i] == 'V')
				V.add(i);
			else if (s[i] == 'K')
				K.add(i);
			else
				X.add(i);
		long[][][][] dp = new long[n + 1][n + 1][n + 1][2];
		for (int i = 0; i <= n; i++)
			for (int j = 0; j <= n; j++)
				for (int k = 0; k <= n; k++)
					for (int d = 0; d < 2; d++)
						dp[i][j][k][d] = Integer.MAX_VALUE;
		dp[0][0][0][0] = 0;
		for (int v = 0; v <= V.size(); v++) {
			for (int k = 0; k <= K.size(); k++) {
				for (int x = 0; x <= X.size(); x++) {
					for (int t = 0; t < 2; t++) {
						long cur = dp[v][k][x][t];
						int m1 = 0, m2 = 0, m3 = 0;
						if (v < V.size()) {
							int vv = V.get(v);
							m1 = rem(V, v, vv) + rem(K, k, vv) + rem(X, x, vv);
							dp[v + 1][k][x][1] = Math.min(dp[v + 1][k][x][1],
									cur + m1);
						}
						if (k < K.size() && t == 0) {
							int kk = K.get(k);
							m2 = rem(V, v, kk) + rem(K, k, kk) + rem(X, x, kk);
							dp[v][k + 1][x][0] = Math.min(dp[v][k + 1][x][0],
									cur + m2);
						}
						if (x < X.size()) {
							int xx = X.get(x);
							m3 = rem(V, v, xx) + rem(K, k, xx) + rem(X, x, xx);
							dp[v][k][x + 1][0] = Math.min(dp[v][k][x + 1][0],
									cur + m3);
						}
					}
				}
			}
		}
		long ret = Long.MAX_VALUE;
		for (int i = 0; i < 2; i++)
			ret = Math.min(ret, dp[V.size()][K.size()][X.size()][i]);
		out.println(ret);
		sc.close();
		out.close();
	}

	private int rem(ArrayList<Integer> arr, int ind, int val) {
		int cnt = 0;
		for (int i = ind; i < arr.size() && arr.get(i) < val; i++)
			cnt++;
		return cnt;
	}

	public static void main(String[] args) {
		new E();
	}
}
