import java.util.Scanner;

public class divide {
	
	public static void main(String[] args) {
		Scanner ip=new Scanner(System.in);
		int t=ip.nextInt();
		for(int i=0;i<t;i++)
		{	int n=ip.nextInt();
			long a[]=new long[n];
			long x, ind0 = 0, ind1 = 0, ind2 = 0;
			for(int j=0;j<n;j++) 
			{
				
				a[j]=ip.nextLong();
				x=a[j];
				if(x % 3 == 0) 
					ind0++;
	            else if(x % 3 == 1) 
	            	ind1++;
	            else 
	            	ind2++;
			}
			long ans = ind0;

	        ans += Math.min(ind1, ind2);

	        if(ind1 > ind2){
	            ind1 -= ind2;
	            ans += ind1 / 3;
	        }
	        else{
	            ind2 -= ind1;
	            ans += ind2 / 3;
	        }

	        System.out.println(ans);
		}
		
		

	}

}
