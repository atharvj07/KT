(n, m) = map(int, input().split())
A = list(map(int, input().split()))
B = sorted(A)
dp = [0] * 200001
dp2 = [-1] * 200001
dp2[0] = 0
acc = 0
for i in range(len(B)):
	acc += B[i]
	dp[i + 1] = acc

def solve(n):
	if dp2[n] > 0:
		return dp2[n]
	if n < m:
		dp2[n] = dp[n]
		return dp2[n]
	dp2[n] = solve(n - m) + dp[n]
	return dp2[n]
answer = [solve(n) for n in range(1, n + 1)]
print(' '.join(map(str, answer)))
