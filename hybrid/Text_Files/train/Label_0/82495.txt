from math import sqrt
(n, s) = map(int, input().split())
dist = 0
d = dict()
for _ in range(n):
	(a, b, c) = map(int, input().split())
	square_dist = a * a + b * b
	if square_dist in d:
		d[square_dist] += c
	else:
		d[square_dist] = c
res = -1
sorted_d = sorted(d)
for it in sorted_d:
	s += d[it]
	if s >= 1000000:
		res = sqrt(int(it))
		break
print(res)
