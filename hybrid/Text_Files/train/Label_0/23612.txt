#include<bits/stdc++.h>
using namespace std;
const int N=4e3+10;
const int mod=998244353;

int gi() {
	int x=0,o=1;char ch=getchar();
	while(!isdigit(ch)&&ch!='-') ch=getchar();
	if(ch=='-') o=-1,ch=getchar();
	while(isdigit(ch)) x=x*10+ch-'0',ch=getchar();
	return x*o;
}

int n,p,bin[N],fl=1,ans=0;
bitset<N> a,b,c;

int main() {
	cin>>n>>a;
	for(int i=1;i<=n;i++) {
		cin>>c;
		while(c.any()) {
			for(int i=4000;;i--) if(c[i]) { p=i;break; }
			for(int i=4000;i>=p;i--) if(b[i]) b^=c<<(i-p);
			swap(b,c);
		}
	}
	bin[0]=1;
	for(int i=1;i<N;i++) bin[i]=2ll*bin[i-1]%mod;
	for(int i=4000;;i--) if(b[i]) { p=i;break; }
	c.reset();
	for(int i=4000;i>=p;i--) {
		if(a[i]) ans=(ans+bin[i-p])%mod;
		if(c[i]!=a[i]) c^=b<<(i-p);
	}
	for(int i=p-1;~i;i--)
		if(c[i]<a[i]) break;
		else if(c[i]>a[i]) { fl=0;break; }
	cout<<(ans+fl)%mod;
	return 0;
}
