#include <bits/stdc++.h>
using namespace std;
int n, c[100050], s[100050], pre[100050];
double dp[100050];
bool flag[100050];
vector<pair<int, int> > g[100050];
bool cmp(const pair<int, int> &a, const pair<int, int> &b) {
  return (s[b.first] + 1) * ((long long)c[a.first] + 2 * a.second) <
         (s[a.first] + 1) * ((long long)c[b.first] + 2 * b.second);
}
void dfs(int u) {
  flag[u] = true;
  s[u] = 0;
  c[u] = 0;
  for (int i = 0; i < g[u].size(); i++) {
    int v = g[u][i].first;
    int w = g[u][i].second;
    if (flag[v]) continue;
    pre[v] = u;
    dfs(v);
    c[u] += w * 2 + c[v];
    s[u] += s[v] + 1;
  }
  sort(g[u].begin(), g[u].end(), cmp);
  int lv = s[u];
  dp[u] = 0;
  for (int i = 0; i < g[u].size(); i++) {
    int v = g[u][i].first;
    int w = g[u][i].second;
    if (pre[v] != u) continue;
    lv -= s[v] + 1;
    dp[u] += s[v] / (double)(s[u]) * (dp[v] + w) + 1.0 / s[u] * w;
    dp[u] += lv / (double)(s[u]) * (w * 2 + c[v]);
  }
}
void solve() {
  memset(flag, false, sizeof(flag));
  pre[1] = 0;
  dfs(1);
  printf("%.10f\n", dp[1]);
}
int main() {
  int u, v, w;
  while (cin >> n) {
    for (int i = 1; i <= n; i++) g[i].clear();
    for (int i = 1; i < n; i++) {
      scanf("%d%d%d", &u, &v, &w);
      g[u].push_back(make_pair(v, w));
      g[v].push_back(make_pair(u, w));
    }
    solve();
  }
  return 0;
}
