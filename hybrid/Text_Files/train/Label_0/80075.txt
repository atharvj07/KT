import sys
input = lambda : sys.stdin.readline().rstrip()
from random import shuffle
T = int(input())
for _ in range(T):

	def calcI():
		for (i, a) in enumerate(A):
			I[a] = i

	def addans(i):
		ANS.append(i)
		A[:i] = A[:i][::-1]
		calcI()
	N = int(input())
	A = [int(a) - 1 for a in input().split()]
	I = [0] * N
	k = (N - 1) // 2
	ng = 0
	for (i, a) in enumerate(A):
		if (i ^ a) & 1:
			ng = 1
			break
	if ng:
		print(-1)
		continue
	calcI()
	ANS = []
	while k:
		r = 2 * k
		l = r - 1
		if I[r] == r and I[l] == l:
			k -= 1
			continue
		if I[r] == 0 and I[l] == 1:
			addans(r + 1)
			k -= 1
			continue
		if I[r] - I[l] == 1:
			addans(I[r] + 1)
			continue
		if I[l] - I[r] == 1:
			addans(I[l] + 2)
			continue
		if I[r] == 0:
			addans(I[l])
			continue
		if I[l] > I[r]:
			addans(I[r] + 1)
			continue
		addans(I[r] + 1)
	if not ANS:
		ANS = [3, 3]
	print(len(ANS))
	print(*ANS)
