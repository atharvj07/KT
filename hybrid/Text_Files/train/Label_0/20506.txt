#include <bits/stdc++.h>
using namespace std;

int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);  

	int a, b, c; cin >> a >> b >> c;
	int x = -1000;
	int s[3][3];
	while(true){
		s[0][0] = a, s[0][1] = b; s[0][2] = x;
		s[1][0] = 2 * x + b - 2 * c; s[1][1] = c; s[1][2] = a + c - x;
		s[2][0] = 2 * c - x; s[2][1] = x + a - c; s[2][2] = b - c + x;
		int total = a + b + x;
		bool flag = true;
		for(int i = 0; i < 3; i++){
			int tate = 0, yoko = 0;
			for(int j = 0; j < 3; j++){
				tate += s[j][i]; yoko += s[i][j];
			}
			if(tate != total || yoko != total) flag = false;
		}
		int naname1 = s[0][0] + s[1][1] + s[2][2], naname2 = s[2][0] + s[1][1] + s[0][2];
		if(naname1 != total || naname2 != total) flag = false;
		if(flag){
			for(int i = 0; i < 3; i++){
				for(int j = 0; j < 3; j++){
					cout << s[i][j];
					if(j < 2) cout << " ";
				}
				cout << endl;
			}
			return 0;
		}
			x++;
	}
	return 0;	
}
