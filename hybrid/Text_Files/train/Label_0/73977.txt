n = int(input())
a = list(map(int, input().split()))
pos = {0: 0}
to_skip = set()
for ai in a:
	if ai in to_skip:
		continue
	changed = False
	for (k, v) in list(pos.items()):
		if v < ai:
			nou = k ^ ai
			if nou not in pos or ai < pos[nou]:
				pos[nou] = ai
				changed = True
	if not changed:
		to_skip.add(ai)
	else:
		to_skip = set()
print(len(pos))
print(' '.join((str(v) for v in sorted(pos))))
