import math
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
rc = 0
bs = 0
for i in range(0, len(a)):
	if a[i] != b[i]:
		if a[i] == 1:
			rc += 1
		else:
			bs += 1
if rc == 0 and bs == 0 or (rc == 0 and bs > 0):
	print('-1')
else:
	print(str(math.ceil((bs + 1) / rc)))
