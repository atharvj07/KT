s = list(input())
(y, k) = map(lambda x: int(x), input().split())

def Kurisu(v, y_01, x_10):
	c_x = [0] * len(v)
	(one, two) = (0, 0)
	q = 0
	for i in range(len(v)):
		i = len(v) - i - 1
		c_x[i] = [one, two]
		if v[i] == '1' or (y_01 <= x_10 and v[i] == '?'):
			one += 1
			q += c_x[i][1] * x_10
		else:
			two += 1
			q += c_x[i][0] * y_01
	qm = q
	(one, two) = (0, 0)
	for i in range(len(v)):
		if v[i] == '1':
			one += 1
		elif v[i] == '0':
			two += 1
		else:
			if y_01 > x_10:
				q -= one * x_10 + c_x[i][0] * y_01 - two * y_01 - c_x[i][1] * x_10
				one += 1
			else:
				q -= -+one * x_10 - c_x[i][0] * y_01 + two * y_01 + c_x[i][1] * x_10
				two += 1
			if qm > q:
				qm = q
	print(qm)
Kurisu(s, y, k)
