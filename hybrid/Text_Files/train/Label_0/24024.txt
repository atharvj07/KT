#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
char s[2][maxn];
int n, k;
vector<int> g[maxn * 2];
int d[maxn * 2];
int main() {
  scanf("%d%d", &n, &k);
  scanf("%s%s", s[0], s[1]);
  for (int i = 0; i < n; i++) {
    if (s[0][i] == 'X') continue;
    if (i > 0 && s[0][i - 1] == '-') {
      g[i].push_back(i - 1);
    }
    if ((i < n - 1 && s[0][i + 1] == '-') || i == n - 1) {
      g[i].push_back(i + 1);
    }
    int j = min(n, i + k);
    if (j < n && s[1][j] == 'X') continue;
    g[i].push_back(j + n + 1);
  }
  for (int i = 0; i < n; i++) {
    if (s[1][i] == 'X') continue;
    if (i > 0 && s[1][i - 1] == '-') {
      g[i + n + 1].push_back(i + n);
    }
    if ((i < n - 1 && s[1][i + 1] == '-') || i == n - 1) {
      g[i + n + 1].push_back(i + n + 2);
    }
    int j = min(n, i + k);
    if (j < n && s[0][j] == 'X') continue;
    g[i + n + 1].push_back(j);
  }
  queue<int> q;
  q.push(0);
  fill(d, d + maxn * 2, 1e9 + 9);
  d[0] = 0;
  while (!q.empty()) {
    int t = q.front();
    q.pop();
    for (int i = 0; i < g[t].size(); i++) {
      int v = g[t][i];
      if (d[v] > d[t] + 1 && d[t] + 1 <= (v % (n + 1))) {
        d[v] = d[t] + 1;
        q.push(v);
      }
    }
  }
  if (min(d[n], d[n + n + 1]) < 1e9 + 9)
    puts("YES");
  else
    puts("NO");
  return 0;
}
