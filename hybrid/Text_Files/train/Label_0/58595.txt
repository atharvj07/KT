	static void fillmap() {
		map = new int[1 << 19];
		int cur = 1;
		loop: for (int i = 0; i < 1 << 19; i++) {
			for (int j = 0; j < 19; j++) {
				if ((i & (1 << j)) != 0 && (i & (1 << (j + 1))) != 0) {
					continue loop;
				}
			}
			map[i] = cur++;
		}
	}

	static int[] map;

	static boolean isValid(int[] arr, int lim) {
		for (int i = lim; i < arr.length; i++) {
			if (x % arr[i] == 0)
				return true;
		}
		if (arr.length - lim == 4) {
			if (x % (arr[lim] + arr[lim + 1]) == 0)
				return true;
			if (x % (arr[lim] + arr[lim + 2]) == 0)
				return true;
		}
		return false;
	}

	static int hash(int[] arr) {

		int mult = 1;
		int ans = 0;
		for (int i = 0; i < arr.length; i++) {
			ans += (arr[i]) * mult;
			mult *= 10;
		}

		return ans;
	}

	public static int dp(int idx, int[] prev, int hash) {

//		System.out.println(idx + " " + Arrays.toString(prev) + " " + hash);
		if (idx == arr.length)
			return 0;

		if (dp[idx][hash] != -1) {
			return dp[idx][hash];
		}
		if (arr[idx] - '0' == x) {
			return dp(idx + 1, new int[] {}, 0) + 1;
		}
		if (arr[idx] - '0' > x) {
			return dp(idx + 1, new int[] {}, 0);
		}
		int ans = dp(idx + 1, prev.clone(), hash) + 1;
		int sum = 0;
		for (int x : prev)
			sum += x;
		int[] temp = new int[prev.length + 1];
		for (int i = 0; i < temp.length - 1; i++) {
			temp[i] = prev[i];
		}
		sum += arr[idx] - '0';
		temp[prev.length] = arr[idx] - '0';
		int lim = 0;
		while (sum > x) {
			sum -= temp[lim++];
		}

		if (sum == x) {
			if (isValid(temp, lim)) {
				lim++;
			} else {
				return dp[idx][hash] = ans;
			}
		}

		while (temp.length - lim > 4)
			lim++;
		int[] nprev = new int[temp.length - lim];
		for (int i = 0; lim < temp.length; lim++, i++) {
			nprev[i] = temp[lim];
		}

		ans = Math.min(ans, dp(idx + 1, nprev, hash(nprev)));
		return dp[idx][hash] = ans;
	}

	static int[][] memo;

	static int dp(int idx, int mask, char[] arr) {

		if ((mask & (1 << x)) != 0) {
			return (int) 1e6;
		}
		if (idx == arr.length) {
			return 0;
		}
		int m = map[mask];

		if (memo[idx][m] != -1) {
			return memo[idx][m];
		}
		int cur = arr[idx] - '0';
		int nmask = (((mask | 1) << cur) & ((1 << (x + 1)) - 1));

		int ans = Math.min(dp(idx + 1, nmask, arr), dp(idx + 1, mask, arr) + 1);

		return memo[idx][m] = ans;
	}

	public static int solvePrime(char[] a) {

		memo = new int[a.length][11000];
		for (int[] x : memo)
			Arrays.fill(x, -1);
		return dp(0, 0, a);
	}

	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public Scanner(FileReader f) {
			br = new BufferedReader(f);
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public int[] nextIntArr(int n) throws IOException {
			int[] arr = new int[n];
			for (int i = 0; i < n; i++) {
				arr[i] = Integer.parseInt(next());
			}
			return arr;
		}
