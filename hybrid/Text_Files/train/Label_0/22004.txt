#include<bits/stdc++.h>

using namespace std;

int main()
{
  int N,M;
  cin>>N>>M;

  long long A = 0;
  long long B = 0;
  for(int i = 0;i < N;++i){
    int temp;
    cin>>temp;
    A += temp;
  }
  for(int i = 0;i < M;++i){
    int temp;
    cin>>temp;
    B += temp;
  }

  long long ans = A * B;
  cout<<ans<<endl;

  return 0;
}
