#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx,tune=native")
const int N = 1e5 + 1;
using namespace std;
int a[N];
int ans;
int n, x;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cin >> n;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) {
      cin >> x;
      if (i == n / 2 || j == n / 2 || i == j || n - i == j + 1) ans += x;
    }
  cout << ans;
  return 0;
}
