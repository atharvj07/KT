import java.io.BufferedInputStream;
import java.io.BufferedOutputStream;
import java.io.PrintStream;
import java.util.Scanner;

public class C1031 {
	static int A, B;

	public static void main(String args[]) throws Exception {
		Scanner in = new Scanner(new BufferedInputStream(System.in));
		PrintStream out = new PrintStream(new BufferedOutputStream(System.out));

		A = in.nextInt();
		B = in.nextInt();

		int l = 0, r = Integer.MAX_VALUE / 2;
		while (l < r) {
			int mid = (l + r) / 2;
			if (valid(mid)) l = mid + 1;
			else r = mid;
		}

		boolean[] fin = ans(l);
		int tot = 0;
		for (int i = 1; i < l; i++)
			tot += fin[i] ? 1 : 0;

		out.println(tot);
		for (int i = 1; i < l; i++) {
			if (fin[i]) out.print(i + " ");
		}
		out.println();
		out.println(l - 1 - tot);
		for (int i = 1; i < l; i++) {
			if (!fin[i]) out.print(i + " ");
		}
		out.println();

		in.close();
		out.close();
	}

	public static boolean valid(int n) {
		return (1L * n * (n + 1) / 2) <= A + B;
	}

	public static boolean[] ans(int n) {
		boolean[] ret = new boolean[n];
		for (int i = n - 1; i > 0; i--) {
			if (A > i) {
				A -= i;
				ret[i] = true;
			} else {
				ret[A] = true;
				A = 0;
			}
		}
		return ret;
	}
}
