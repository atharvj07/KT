l = list(map(int, input().split()))
(n, d) = (l[0], l[1])
s = input()
i = 0
ans = -1
c = 0
f = 0
while i < n:
	if s[i] != '1':
		f = 1
		break
	j = 1
	x = -1
	while i + j < n and j <= d:
		if s[i + j] == '1':
			x = i + j
		j += 1
	if x == -1:
		f = 1
		break
	i = x
	c += 1
	if i == n - 1:
		break
if f == 0:
	ans = c
print(ans)
