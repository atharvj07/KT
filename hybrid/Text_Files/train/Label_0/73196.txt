T = int(input())
letters = [chr(ord('a') + j) for j in range(26)]
for t in range(1, T + 1):
	N = int(input())
	A = [int(s) for s in input().split()]
	y = max(A) + 1
	ans = []
	str1 = ''
	for x in range(y):
		str1 += letters[x % 26]
	ans.append(str1)
	for n in range(N):
		z = A[n]
		new_str = str1[:z]
		if z < len(str1):
			ind = ord(str1[z]) - ord('a')
			new_ind = (ind + 1) % 26
			new_str += chr(new_ind + ord('a'))
			new_str += str1[z + 1:]
		ans.append(new_str)
		str1 = new_str
	for x in ans:
		print(x)
