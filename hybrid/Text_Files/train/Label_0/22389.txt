#include <bits/stdc++.h>
using namespace std;
int n;
int a[100010];
bool b[2];
int main() {
  scanf("%d", &n);
  for (int i = 2; i <= n; ++i) {
    if (!a[i]) {
      a[i] = ++a[1];
      for (int j = 2; i * j <= n; j++) a[i * j] = a[i];
    }
  }
  for (int i = 2; i <= n; ++i) printf("%d ", a[i]);
}
