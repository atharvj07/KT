#include<bits/stdc++.h>
#define int long long
#define fi first
#define se second
#define mp make_pair
#define pii pair<long long,long long>
#define pb push_back
using namespace std;
int n,k,a[100005]; 
bool check()
{
	if(a[n]==n-1) return 0;
	return 1;
}
void solve()
{
	cin>>n>>k;
	int maxx=0;
	for(int i=1;i<=n;i++) cin>>a[i],maxx=max(maxx,a[i]);
	sort(a+1,a+1+n);
	int mex;
	for(int i=1;i<=n;i++) if(a[i]-a[i-1]>1)
	{
		mex=a[i-1]+1;
		break; 
	}
	int now=(mex+maxx+1)/2;
	int flg=1;
	if(k==0) flg=0;
	for(int i=1;i<=n;i++) if(a[i]==now) flg=0;
	if(check()) cout<<n+flg<<endl;
	else cout<<n+k<<endl;
}
signed main()
{
	int _;
	cin>>_;
	while(_--) solve();
	return 0;
}
