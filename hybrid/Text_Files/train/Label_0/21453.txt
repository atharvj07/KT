#include <bits/stdc++.h>
using namespace std;
vector<long long> maq;
void make_maq(long long n) {
  maq.resize(0);
  long long i;
  for (i = 1; i * i < n; i++)
    if (n % i == 0) {
      maq.push_back(i);
      maq.push_back(n / i);
    }
  if (i * i == n) maq.push_back(i);
  return;
}
int main() {
  long long n;
  cin >> n;
  make_maq(n);
  long long Min = (-1), Max = (-1);
  for (int i = 0; i < maq.size(); i++)
    for (int j = 0; j < maq.size(); j++)
      if (n % (maq[i] * maq[j]) == 0) {
        long long A = maq[i] + 1;
        long long B = maq[j] + 2;
        long long C = (n / (maq[i] * maq[j])) + 2;
        if (Min == (-1)) Min = A * B * C;
        if (Max == (-1)) Max = A * B * C;
        if (Min > A * B * C) Min = A * B * C;
        if (Max < A * B * C) Max = A * B * C;
      }
  Min -= n;
  Max -= n;
  cout << Min << " " << Max << endl;
  return 0;
}
