n = int(input())
INF = -10 ** 9
for _ in ' ' * n:
	q = int(input())
	(*l1,) = map(int, input().split())
	(*l2,) = map(int, input().split())
	s = set()
	r = INF
	itr = -1
	f = 1
	for i in range(q):
		if l1[i] != l2[i]:
			if r == INF:
				r = l2[i] - l1[i]
				itr = i
			elif r != l2[i] - l1[i] or i - itr != 1:
				f = 0
				break
			else:
				itr = i
	if f and r > 0 or r == INF:
		print('YES')
	else:
		print('NO')
