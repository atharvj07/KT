#include <bits/stdc++.h>
using namespace std;
const int mn = 1e1;
const int inf = 1e9;
int main() {
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cout << setprecision(30);
  int n;
  cin >> n;
  set<string> ss;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    if (ss.find(s) == ss.end()) {
      cout << "NO"
           << "\n";
      ss.insert(s);
    } else {
      cout << "YES"
           << "\n";
    }
  }
  return 0;
}
