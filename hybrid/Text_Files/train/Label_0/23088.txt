#include <bits/stdc++.h>
using namespace std;
char s[112];
int rev(int q) {
  int ans = q & 1;
  q >>= 1;
  for (int i = 7; i > 0; --i) {
    ans = ans * 2 + (q & 1);
    q >>= 1;
  }
  return ans;
}
int hz(int p, int c) {
  int pp = rev(p);
  int cc = rev(c);
  return (pp - cc + 256) % 256;
}
int main() {
  int n = 0;
  while (scanf("%c", &s[n]) != -1) {
    if (s[n] == '\n') break;
    if (s[n] == 0) break;
    if (s[n] < 32 || s[n] > 126) break;
    ++n;
  }
  cout << hz(0, s[0]) << endl;
  for (int t = 1; t < n; ++t) cout << hz(s[t - 1], s[t]) << endl;
  return 0;
}
