#include <bits/stdc++.h>
using namespace std;

int main()
{
	int n;cin>>n;
	vector<int> p(n),l(n),r(n);
	for(int &e:p)	cin>>e;
	int cur = 0;
	for(int i=0;i<n;i++)
	{
		l[i] = cur;
		if(i==n-1)	break;
		if(p[i]<p[i+1])	cur++;
		else			cur=0;
	}
	cur=0;
	for(int i=n-1;i>=0;i--)
	{
		r[i] = cur;
		if(i==0)	break;
		if(p[i]<p[i-1])	cur++;
		else			cur=0;
	}

	int sm=0,dm=0,cnt=1,si=-1,dcnt=1,di=-1;
	for(int i=0;i<n;i++)
	{
		if(l[i]==r[i])
		{
			if(sm<l[i])
			{
				sm = l[i];
				cnt=1;
				si=i;
			}
			else if(sm==l[i])	cnt++;
		}
		else
		{
			int temp = max(l[i],r[i]);
			if(dm<temp)
			{
				dm = temp;
				dcnt=1;
				di=i;
			}
			else if(dm==temp)	dcnt++;
		}
	}
	bool pos = 0;
	if(dm<sm&&cnt==1&&sm%2==0)	pos = 1;
	
	cout<<pos;
	cin>>n;
}
