#include<stdio.h>
#include<algorithm>
#include<math.h>
using namespace std;
double dp[42][42][42][42];
int rev[42][42][42][42];
double x[42];
double y[42];
pair<pair<int,int>,int>in[42];
double d[42][42];
double th[42][42];
int out[42];
int conv[42];
int main(){
	int a;
	scanf("%d",&a);
	for(int i=0;i<a;i++){
		scanf("%d%d",&in[i].first.second,&in[i].first.first);
		in[i].second=i;
	}
	std::sort(in,in+a);
	for(int i=0;i<a;i++){
		x[i]=in[i].first.second;
		y[i]=in[i].first.first;
		conv[i]=in[i].second;
	}
	for(int i=0;i<a;i++)for(int j=0;j<a;j++){
		if(i==j)continue;
		th[i][j]=atan2(y[j]-y[i],x[j]-x[i]);
		if(th[i][j]<0)th[i][j]+=acos(-1.0)*2;
	//	printf("%d %d: %f\n",i,j,th[i][j]);
	}
	for(int i=0;i<a;i++)for(int j=0;j<a;j++){
		d[i][j]=x[i]*y[j]-y[i]*x[j];
	}
	for(int i=0;i<=a;i++)for(int j=0;j<=a;j++)for(int k=0;k<=a;k++)for(int l=0;l<=a;l++)
		dp[i][j][k][l]=99999999999.9;
	for(int i=0;i<a;i++){
		for(int j=i+1;j<a;j++){
			dp[i][1][j][i]=d[i][j];
		}
		for(int j=1;j<a;j++){
			for(int k=i+1;k<a;k++){
				for(int l=i;l<a;l++){
					if(dp[i][j][k][l]<9999999999.9){
						
						for(int m=i;m<a;m++){
							if(l==m||k==m)continue;
							if(th[l][k]>th[k][m])continue;
							if(dp[i][j+1][m][k]>dp[i][j][k][l]+d[k][m]){
//printf("%d %d %d %d: -> %d %f\n",i,j,k,l,m,dp[i][j][k][l]+d[k][m]);
								dp[i][j+1][m][k]=dp[i][j][k][l]+d[k][m];
								rev[i][j+1][m][k]=l;
							}
						}
					}
				}
			}
		}
	}
	int c;scanf("%d",&c);
	while(c--){
		int t;scanf("%d",&t);
		double res=99999999999.9;
		int I=0;
		int L=0;
		for(int i=0;i<a;i++){
			for(int l=0;l<a;l++){
				if(res>dp[i][t][i][l]){
					res=dp[i][t][i][l];
					I=i;L=l;
				}
			}
		}
		if(res>9999999999.9){
			printf("NA\n");continue;
		}
		int K=I;
		for(int i=0;i<t;i++){
			
			int toL=rev[I][t-i][K][L];
			K=L;L=toL;out[t-1-i]=conv[K]+1;
		}
		for(int i=0;i<t;i++){
			if(i)printf(" ");
			printf("%d",out[i]);
		}
		printf("\n");
	}
}
