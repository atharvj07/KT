#include <bits/stdc++.h>
using namespace std;
int n, m, odd, ans;
string str, str2;
int main() {
  cin >> str >> str2;
  n = str.size();
  m = str2.size();
  for (int i = 0; i < m; i++) {
    if (str[i] != str2[i]) odd = !odd;
  }
  if (!odd) ans++;
  for (int i = 0; i < n - m; i++) {
    if (str[i] != str[i + m]) odd = !odd;
    if (!odd) ans++;
  }
  cout << ans << '\n';
}
