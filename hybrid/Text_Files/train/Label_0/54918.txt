import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class CODEFORCES
{
	@SuppressWarnings("rawtypes")
	static InputReader in;
	static PrintWriter out;

	static boolean check(int h, int m)
	{
		while (h != 0)
		{
			if (h % 10 == 7)
				return true;
			h /= 10;
		}
		while (m != 0)
		{
			if (m % 10 == 7)
				return true;
			m /= 10;
		}
		return false;
	}

	static int cnt(ArrayList<Integer> a)
	{
		int cn = 0;
		int mx = a.get(a.size() - 1);
		for (int i : a)
			if (i == mx)
				cn++;
		return cn;
	}

	static void solve()
	{
		long n = in.nl();
		int k = in.ni();
		ArrayList<Integer> a = new ArrayList<Integer>();
		long tmp = n;
		long xx = ((long) 1) << 62;
		int cnt = 62;
		while (tmp != 0)
		{
			if ((tmp & xx) != 0)
			{
				a.add(cnt);
				tmp -= xx;
			}
			cnt--;
			xx >>= 1;
		}
		if (a.size() > k)
		{
			out.println("No");
			return;
		}
		while (true)
		{
			Collections.sort(a);
			int cn = cnt(a);
			if (a.size() + cn > k)
				break;
			int mx = a.get(a.size() - 1);
			ArrayList<Integer> b = new ArrayList<Integer>();
			for (int i : a)
			{
				if (i == mx)
					break;
				b.add(i);
			}
			for (int i = 0; i < 2 * cn; i++)
				b.add(mx - 1);
			a = b;
		}
		int mx = a.get(0);
		ArrayList<Integer> b = new ArrayList<Integer>();
		for (int i = a.size() - 1; i > 0; i--)
			b.add(a.get(i));
		a = b;
		while (a.size() < k - 1)
		{
			mx--;
			a.add(mx);
		}
		a.add(mx);
		Collections.sort(a);
		out.println("Yes");
		for (int i = k - 1; i >= 0; i--)
			out.print(a.get(i) + " ");
	}

	@SuppressWarnings("rawtypes")
	static void soln()
	{
		in = new InputReader(System.in);
		out = new PrintWriter(System.out);
		solve();
		out.flush();
	}

	static void debug(Object... o)
	{
		System.out.println(Arrays.deepToString(o));
	}

	public static void main(String[] args)
	{
		new Thread(null, new Runnable()
		{
			public void run()
			{
				try
				{
					soln();
				} catch (Exception e)
				{
					e.printStackTrace();
				}
			}
		}, "1", 1 << 26).start();
	}

	// To Get Input
	// Some Buffer Methods
	static class InputReader<SpaceCharFilter>
	{

		private final InputStream stream;
		private final byte[] buf = new byte[8192];
		private int curChar, snumChars;
		private SpaceCharFilter filter;

		public InputReader(InputStream stream)
		{
			this.stream = stream;
		}

		public int snext()
		{
			if (snumChars == -1)
				throw new InputMismatchException();
			if (curChar >= snumChars)
			{
				curChar = 0;
				try
				{
					snumChars = stream.read(buf);
				} catch (IOException e)
				{
					throw new InputMismatchException();
				}
				if (snumChars <= 0)
					return -1;
			}
			return buf[curChar++];
		}

		public int ni()
		{
			int c = snext();
			while (isSpaceChar(c))
			{
				c = snext();
			}
			int sgn = 1;
			if (c == '-')
			{
				sgn = -1;
				c = snext();
			}
			int res = 0;
			do
			{
				if (c < '0' || c > '9')
					throw new InputMismatchException();
				res *= 10;
				res += c - '0';
				c = snext();
			} while (!isSpaceChar(c));
			return res * sgn;
		}

		public long nl()
		{
			int c = snext();
			while (isSpaceChar(c))
			{
				c = snext();
			}
			int sgn = 1;
			if (c == '-')
			{
				sgn = -1;
				c = snext();
			}
			long res = 0;
			do
			{
				if (c < '0' || c > '9')
					throw new InputMismatchException();
				res *= 10;
				res += c - '0';
				c = snext();
			} while (!isSpaceChar(c));
			return res * sgn;
		}

		public int[] nextIntArray(int n)
		{
			int a[] = new int[n];
			for (int i = 0; i < n; i++)
			{
				a[i] = ni();
			}
			return a;
		}

		public long[] nextLongArray(int n)
		{
			long a[] = new long[n];
			for (int i = 0; i < n; i++)
			{
				a[i] = nl();
			}
			return a;
		}

		public String readString()
		{
			int c = snext();
			while (isSpaceChar(c))
			{
				c = snext();
			}
			StringBuilder res = new StringBuilder();
			do
			{
				res.appendCodePoint(c);
				c = snext();
			} while (!isSpaceChar(c));
			return res.toString();
		}

		public String nextLine()
		{
			int c = snext();
			while (isSpaceChar(c))
				c = snext();
			StringBuilder res = new StringBuilder();
			do
			{
				res.appendCodePoint(c);
				c = snext();
			} while (!isEndOfLine(c));
			return res.toString();
		}

		public boolean isSpaceChar(int c)
		{
			if (filter != null)
				return filter.isSpaceChar(c);
			return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
		}

		private boolean isEndOfLine(int c)
		{
			return c == '\n' || c == '\r' || c == -1;
		}

		public interface SpaceCharFilter
		{
			public boolean isSpaceChar(int ch);
		}

	}
}
