n = int(input())
a = list(map(int, list(input().split())))
a_sorted = sorted(a)
start = -1
end = -1
for i in range(n):
	if a[i] != a_sorted[i]:
		start = i
		break
	start = i
for i in range(n - 1, -1, -1):
	if a[i] != a_sorted[i]:
		end = i
		break
	end = i
if start > end:
	print('yes')
	print('1 1')
	exit(0)
if a[start:end + 1] == a_sorted[start:end + 1][::-1]:
	print('yes')
	print(start + 1, end + 1, sep=' ')
	exit(0)
print('no')
exit(0)
