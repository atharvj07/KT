#include <bits/stdc++.h>
using namespace std;
template <typename T1, typename T2>
bool Max(T1& a, T2 b) {
  return a < b ? a = b, 1 : 0;
}
template <typename T1, typename T2>
bool Min(T1& a, T2 b) {
  return a > b ? a = b, 1 : 0;
}
int arr[48];
void solve() {
  long long a, b, l, r;
  cin >> a >> b >> l >> r;
  for (int i = 0; i < (a); ++i) arr[i] = i;
  for (int i = a; i <= (a + b - 1); ++i) arr[i] = a - 1;
  long long t = min(b, a - 1);
  for (int i = 0; i < (t); ++i) arr[a + b + i] = i;
  for (int i = 0; i < (a - t); ++i) arr[a + b + t + i] = a + i;
  for (int i = 0; i < (b); ++i) arr[a + b + a + i] = a + a - t - 1;
  long long n = (a + b) * 2;
  l--;
  set<int> s;
  if (r - l >= n) {
    for (int i = 0; i < (n); ++i) s.insert(arr[i]);
  } else {
    l %= n;
    r %= n;
    if (b < a) {
      if (l >= a && l < a + b) {
        for (int i = a; i <= (a + b - 1); ++i) arr[i] = b;
      } else if (l >= a + b + a) {
        for (int i = 0; i < (b); ++i) arr[a + b + a + i] = a;
      }
    }
    for (; l != r; l = (l + 1) % n) s.insert(arr[l]);
  }
  cout << s.size() << '\n';
}
void init() {}
int main(void) {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.setf(ios::fixed);
  cout.precision(20);
  init();
  solve();
  return 0;
}
