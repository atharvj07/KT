#include <bits/stdc++.h>
using namespace std;
long long n, m, a[110], b[110], c[110][110], d, x, y, t, ans, p;
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    x = x xor a[i];
  }
  for (int i = 0; i < m; i++) {
    cin >> b[i];
    y = y xor b[i];
  }
  if (y != x) return cout << "NO", 0;
  for (int i = 0; i < n - 1; i++) {
    c[i][m - 1] = a[i];
  }
  for (int i = 0; i < m - 1; i++) {
    c[n - 1][i] = b[i];
  }
  x = x xor a[n - 1];
  y = y xor b[m - 1];
  x = x xor b[m - 1];
  y = y xor a[n - 1];
  if (x != y) return cout << "NO", 0;
  c[n - 1][m - 1] = x;
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cout << c[i][j] << ' ';
    }
    cout << endl;
  }
  return 0;
}
