#include <bits/stdc++.h>
using namespace std;
const int N = 105;
int n, m, k, a[N][N], ans;
bool check(int x, int y) {
  int c[2] = {0};
  for (int i = 0; i < 2; i++)
    for (int j = 0; j < 2; j++) c[a[x + i][y + j]]++;
  return c[0] == 3 || c[1] == 3;
}
void solve(int tot, int si) {
  if (tot >= ans) return;
  for (int i = si; i < n; i++)
    for (int j = 1; j < m; j++) {
      if (check(i, j)) {
        for (int d1 = 0; d1 < 2; d1++)
          for (int d2 = 0; d2 < 2; d2++) {
            a[i + d1][j + d2] ^= 1;
            solve(tot + 1, max(1, i - 1));
            a[i + d1][j + d2] ^= 1;
          }
        return;
      }
    }
  ans = min(ans, tot);
}
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) scanf("%d", a[i] + j);
  ans = k + 1;
  solve(0, 1);
  if (ans <= k)
    printf("%d\n", ans);
  else
    puts("-1");
}
