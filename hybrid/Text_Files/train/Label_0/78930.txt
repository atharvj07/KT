import sys
input = sys.stdin.readline

def inp():
	return int(input())

def st():
	return input().rstrip('\n')

def lis():
	return list(map(int, input().split()))

def ma():
	return map(int, input().split())
t = inp()
while t:
	t -= 1
	n = inp()
	a = lis()
	s = st()
	ra = []
	for i in range(n):
		if s[i] == 'B':
			if a[i] <= 0:
				continue
			r = min(n, a[i])
			ra.append([1, r])
		else:
			if a[i] > n:
				continue
			l = max(1, a[i])
			ra.append([l, n])
	if len(ra) != n:
		print('NO')
		continue
	ra.sort()
	po = 1
	fl = 0
	for i in range(n):
		if po >= ra[i][0] and po <= ra[i][1]:
			po += 1
		else:
			fl = 1
	if fl:
		print('NO')
	else:
		print('YES')
