from collections import deque
for _ in range(int(input())):
	n = int(input())
	dq = deque(list(map(int, input().split())))
	cnt = prev = end = a = b = 0
	while len(dq) > 0:
		cur = 0
		while len(dq) > 0 and cur <= prev:
			x = dq.pop() if end else dq.popleft()
			cur += x
		if end:
			b += cur
		else:
			a += cur
		end ^= 1
		cnt += 1
		prev = cur
	print(cnt, a, b)
