#include <bits/stdc++.h>
using namespace std;
bool final(vector<string> &v) {
  for (int i = 0; i < 8; i++) {
    for (int j = 0; j < 8; j++) {
      if (v[i][j] == 'M') return false;
    }
  }
  return true;
}
int movi[9] = {0, -1, -1, -1, 0, 0, 1, 1, 1};
int movj[9] = {0, -1, 1, 0, -1, 1, -1, 1, 0};
bool dentro(int i, int j) { return i >= 0 and i < 8 and j >= 0 and j < 8; }
int main() {
  vector<string> v(8);
  for (int i = 0; i < 8; i++) cin >> v[i];
  while (v[0][7] != 'M' and not final(v)) {
    vector<string> nextv(8, string(8, ' '));
    for (int i = 0; i < 8; i++) {
      for (int j = 0; j < 8; j++) {
        if (v[i][j] == 'M') {
          for (int k = 0; k < 9; k++) {
            int nexti = i + movi[k];
            int nextj = j + movj[k];
            if (dentro(nexti, nextj) and v[nexti][nextj] != 'S') {
              nextv[nexti][nextj] = 'M';
            }
          }
        }
      }
    }
    for (int i = 1; i < 8; i++)
      for (int j = 0; j < 8; j++)
        if (v[i - 1][j] == 'S') nextv[i][j] = 'S';
    v = nextv;
  }
  if (v[0][7] == 'M')
    cout << "WIN" << endl;
  else
    cout << "LOSE" << endl;
}
