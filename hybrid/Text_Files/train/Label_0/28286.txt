inline int getint(){
	int x=0,p=1;char c=getchar();
	while (c<=32)c=getchar();
	if(c==45)p=-p,c=getchar();
	while (c>32)x=x*10+c-48,c=getchar();
	return x*p;
}
using namespace std;
//ruogu_alter
const int N=45;
int n;
char s[N][N];
ll f[N][N][N][N],g[N][N][N][N];
//
ll get_f(int x,int y,int z,int w);
ll get_g(int x,int y,int z,int w){
	if(x>y||z>w)return 0ll;
	if(g[x][y][z][w]>=0)return g[x][y][z][w];
	ll res=0;
	for(int i=x;i<=y;i++){
		for(int j=z;j<=w;j++)if(s[i][j]=='1'){
			res+=get_f(x,i-1,i+1,y)*get_f(z,j-1,j+1,w);
		}	
	}
	return g[x][y][z][w]=res;
}
ll get_f(int x,int y,int z,int w){
	if(x>y&&z>w)return 1;
	if(f[x][y][z][w]>=0)return f[x][y][z][w];
	ll res=0;
	for(int i=x;i<=y;i++){
		for(int j=z;j<=w;j++){
			res+=get_g(x,i,j,w)*get_f(i+1,y,z,j-1);
		}
	}
	return f[x][y][z][w]=res;
}
int main(){
	//freopen(".in","r",stdin);
	//freopen(".out","w",stdout);
	n=2*getint();
	rep(i,n)scanf("%s",s[i]);
	memset(f,-1,sizeof(f));
	memset(g,-1,sizeof(g));
	ll res=0;
	for(int i=1;i<n;i++)if(s[0][i]=='1')res+=get_f(1,i-1,i+1,n-1);
	cout<<res<<endl;
	return 0;
}
