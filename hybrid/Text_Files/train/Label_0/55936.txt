
import java.util.*;

public class JavaApplication145 {
public static void main(String[] args){
    Scanner sc = new Scanner(System.in);
    int size = sc.nextInt();
    int[] data = new int[size];
    HashMap<Integer, Integer> map = new HashMap<>();
    for(int i = 0; i<size; i++)
    {
        data[i] = sc.nextInt();
        map.put(data[i], i);
    }
    int m = sc.nextInt();
    long ves = 0, pet = 0;
    for(int i = 0; i<m; i++)
    {
        int q = sc.nextInt();
        int index = map.get(q);
        ves += index+1;
        pet += size - index;
    }
    System.out.println(ves + " " + pet);
}
}
