n = int(input())
a = list(map(int, input().split()))
res = -1
xor_a = [0]
for i in range(n):
	xor_a.append(a[i] ^ xor_a[-1])
(step, i) = (1, 0)
while step < n - 1:
	i = 0
	while i < n:
		for x in range(step + 1):
			if step + 1 + i - x + 1 <= n and i - x >= 0:
				if xor_a[i - x] ^ xor_a[i + 1] > xor_a[1 + i] ^ xor_a[1 + i + step - x + 1]:
					res = step
					i = n
					step = n
		i += 1
	step += 1
print(res)
