#include <bits/stdc++.h>
using namespace std;
int t, n, x, c;
int a[1000000];
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    sort(a, a + n);
    c = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != a[i - 1]) c++;
    }
    cout << c << endl;
  }
}
