#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int n, m;
bool vs[maxn];
map<int, int> edges[maxn];
set<int> curver;
void Input() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    curver.insert(i);
  }
  for (int i = 1; i <= m; i++) {
    int u, v;
    cin >> u >> v;
    edges[u][v] = 1;
    edges[v][u] = 1;
  }
}
void bfs(int root) {
  queue<int> Q;
  Q.push(root);
  vs[root] = true;
  while (Q.size()) {
    int u = Q.front();
    vector<int> V;
    Q.pop();
    set<int>::iterator it;
    for (it = curver.begin(); it != curver.end(); it++) {
      int v = (*it);
      if (edges[u][v]) continue;
      if (vs[v]) continue;
      vs[v] = true;
      Q.push(v);
      V.push_back(v);
    }
    for (int u : V) curver.erase(u);
  }
}
void Output() {
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (!vs[i]) {
      ans++;
      bfs(i);
    }
  }
  cout << ans - 1 << "\n";
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  if (fopen("input.txt", "r")) {
    freopen("input.txt", "r", stdin);
  }
  Input();
  Output();
  return 0;
}
