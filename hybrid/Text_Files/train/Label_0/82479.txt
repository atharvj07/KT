alen = int(input())
a = input()
funcs = input().split()
start_index = None
end_index = None
ans = ''
for (i, c) in enumerate(a):
	ic = int(c)
	maps_to = int(funcs[ic - 1])
	if not end_index:
		if ic < maps_to:
			start_index = True
			ans += str(maps_to)
			continue
		elif ic > maps_to:
			if start_index is not None:
				end_index = True
	ans += c
print(ans)
