#include <bits/stdc++.h>
using namespace std;
long long pre[300005];
long long n, m, k;
long long fun(long long l, long long r) {
  if (l == r) return max(pre[l] - k, (long long)0);
  map<long long, long long> ma;
  long long sum = 0, mid = (l + r) >> 1;
  long long l1 = fun(l, mid), r1 = fun(mid + 1, r), ans = max(l1, r1);
  for (int i = mid; i >= l; i--) {
    sum += pre[i];
    ma[(mid - i + 1) % m] =
        max(ma[(mid - i + 1) % m], sum - k * ((mid - i + 1) / m));
  }
  sum = 0;
  for (int i = mid + 1; i <= r; i++) {
    sum += pre[i];
    for (int j = 0; j < m; j++) {
      ans = max(ans,
                (long long)(sum + ma[j] - k * ceil((1.0 * i - mid + j) / m)));
    }
  }
  return max(ans, (long long)0);
}
int main() {
  scanf("%lld%lld%lld", &n, &m, &k);
  for (int i = 1; i <= n; i++) scanf("%lld", &pre[i]);
  long long ans = fun(1, n);
  printf("%lld\n", ans);
  return 0;
}
