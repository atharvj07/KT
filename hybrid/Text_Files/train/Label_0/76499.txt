(n, m, k) = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
p = m - 1
i = p - 1
j = p + 1
x = 0

def inrange(b):
	return 0 <= b < n
while True:
	if inrange(i) and inrange(j) and (a[i] > 0) and (a[j] > 0):
		if a[i] < a[j]:
			x = i
		else:
			x = j
		if a[x] <= k:
			break
	elif inrange(i) and a[i] > 0:
		x = i
		if a[x] <= k:
			break
	elif inrange(j) and a[j] > 0:
		x = j
		if a[x] <= k:
			break
	i -= 1
	j += 1
print(abs(x - p) * 10)
