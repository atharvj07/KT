#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, t, pos, neg;
  bool flag;
  cin >> t;
  while (t--) {
    cin >> n;
    vector<long long> a(n), b(n);
    pos = n + 1;
    neg = n + 1;
    flag = true;
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
      if (a[i] == 1 && pos == n + 1) pos = i;
      if (a[i] == -1 && neg == n + 1) neg = i;
    }
    for (int i = 0; i < n; ++i) cin >> b[i];
    for (int i = 0; i < n; ++i) {
      if (neg >= i && b[i] < a[i]) {
        cout << "NO\n";
        flag = false;
        break;
      }
      if (pos >= i && b[i] > a[i]) {
        cout << "NO\n";
        flag = false;
        break;
      }
    }
    if (flag) cout << "YES\n";
  }
  return 0;
}
