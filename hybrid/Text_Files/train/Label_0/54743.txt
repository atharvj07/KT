import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int t = s.nextInt();
		for(int q=0;q<t;q++) {
			int n = s.nextInt();
			int m = s.nextInt();
			System.out.print(n-1);
			System.out.print(' ');
			System.out.println(m);
		}
	}

}
