#include <bits/stdc++.h>
using namespace std;
string s;
int i, j, r = 0, b = 0, y = 0, g = 0, ar = 0, ab = 0, ay = 0, ag = 0, cr = 0,
          cb = 0, cy = 0, cg = 0;
void checkred() {
  int a, b;
  for (a = r; a < s.length(); a += 4)
    if (s[a] == '!') s[a] = 'z', ar++;
  for (a = r; a >= 0; a -= 4)
    if (s[a] == '!') s[a] = 'z', ar++;
}
void checkblue() {
  int a;
  for (a = b; a < s.length(); a += 4)
    if (s[a] == '!') s[a] = 'z', ab++;
  for (a = b; a >= 0; a -= 4)
    if (s[a] == '!') s[a] = 'z', ab++;
}
void checkgreen() {
  int a;
  for (a = g; a < s.length(); a += 4)
    if (s[a] == '!') s[a] = 'z', ag++;
  for (a = g; a >= 0; a -= 4)
    if (s[a] == '!') s[a] = 'z', ag++;
}
void checkyellow() {
  int a;
  for (a = y; a < s.length(); a += 4)
    if (s[a] == '!') s[a] = 'z', ay++;
  for (a = y; a >= 0; a -= 4)
    if (s[a] == '!') s[a] = 'z', ay++;
}
int main() {
  cin >> s;
  for (i = 0; i < s.length(); i++)
    if (s[i] == 'R') cr = 1, r = i;
  for (i = 0; i < s.length(); i++)
    if (s[i] == 'B') cb = 1, b = i;
  for (i = 0; i < s.length(); i++)
    if (s[i] == 'G') cg = 1, g = i;
  for (i = 0; i < s.length(); i++)
    if (s[i] == 'Y') cy = 1, y = i;
  checkred();
  checkblue();
  checkyellow();
  checkgreen();
  if (cr == 0) {
    int ch = 0;
    ar = 0;
    for (i = 0; i < s.length();) {
      if (s[i] == '!') ch = 1, s[i] = 'z', ar++;
      if (ch == 0)
        i++;
      else
        i += 4;
    }
  }
  if (cb == 0) {
    int ch = 0;
    ab = 0;
    for (i = 0; i < s.length();) {
      if (s[i] == '!') ch = 1, s[i] = 'z', ab++;
      if (ch == 0)
        i++;
      else
        i += 4;
    }
  }
  if (cg == 0) {
    int ch = 0;
    ag = 0;
    for (i = 0; i < s.length();) {
      if (s[i] == '!') ch = 1, s[i] = 'z', ag++;
      if (ch == 0)
        i++;
      else
        i += 4;
    }
  }
  if (cy == 0) {
    int ch = 0;
    ay = 0;
    for (i = 0; i < s.length();) {
      if (s[i] == '!') ch = 1, s[i] = 'z', ay++;
      if (ch == 0)
        i++;
      else
        i += 4;
    }
  }
  cout << ar << " " << ab << " " << ay << " " << ag;
}
