n = int(input())
for _ in range(n):
	s = input()
	num_dict = {}
	for i in s:
		if i in num_dict.keys():
			num_dict[i] += 1
		else:
			num_dict[i] = 1
	final = ''
	key = max(num_dict, key=num_dict.get)
	value = num_dict[key]
	if key == 'P':
		final += 'S' * len(s)
	elif key == 'S':
		final += 'R' * len(s)
	else:
		final += 'P' * len(s)
	print(final)
