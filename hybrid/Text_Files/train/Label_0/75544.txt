import sys
(n, m) = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
assert len(a) == n
prev = [-1] * n
l = r = 0
while l < n:
	while r < n and a[l] == a[r]:
		r += 1
	for i in range(l, r):
		prev[i] = l - 1
	l = r
for i in range(m):
	(l, r, x) = map(int, sys.stdin.readline().split())
	l -= 1
	r -= 1
	if a[r] != x:
		print(r + 1)
	elif prev[r] == -1 or prev[r] < l:
		print(-1)
	else:
		print(prev[r] + 1)
