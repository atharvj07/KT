import collections
import time
import os
import sys
import bisect
import heapq
from typing import List
N = int(input())
maxdiv = [0 for _ in range(N + 1)]
maxdiv[1] = 1
for i in range(2, N + 1):
	if maxdiv[i]:
		continue
	v = i
	d = 1
	while v <= N:
		if maxdiv[v] == 0:
			maxdiv[v] = d
		v += i
		d += 1
maxdiv = maxdiv[1:]
maxdiv.sort()
ans = []
maxd = 0
for v in maxdiv:
	maxd = max(maxd, v)
	ans.append(maxd)
print(' '.join(map(str, ans[1:])))
