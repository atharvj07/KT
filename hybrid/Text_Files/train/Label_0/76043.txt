from bisect import *
from sys import stdin, stdout

def dfs(i, j):
	global visited
	visited[i] = j
	for i1 in range(n):
		if matrix[i][i1] == '1' and (not visited[i1]):
			dfs(i1, j)
n = int(input())
arr = list(map(int, input().split()))
matrix = []
for i in range(n):
	matrix.append(list(input()))
visited = [0] * n
count = 1
for i in range(n):
	if not visited[i]:
		dfs(i, count)
		count += 1
for i in range(n):
	for j in range(i + 1, n):
		if visited[i] == visited[j]:
			if arr[i] > arr[j]:
				(arr[i], arr[j]) = (arr[j], arr[i])
print(*arr)
