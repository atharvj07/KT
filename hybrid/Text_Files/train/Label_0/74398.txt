def number_of_lucky_digits(num):
	temp = num
	counter = 0
	while temp != 0:
		digit = temp % 10
		if digit == 4 or digit == 7:
			counter += 1
		temp = temp // 10
	return counter
vals = [int(n) for n in input().split()]
n = vals[0]
maxDigitis = vals[1]
nums = [int(x) for x in input().split()]
counter = 0
for num in nums:
	if number_of_lucky_digits(num) <= maxDigitis:
		counter += 1
print(counter)
