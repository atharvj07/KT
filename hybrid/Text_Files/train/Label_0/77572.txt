(n, k) = [int(s) for s in input().split()]
a = [int(s) for s in input().split()]
if sum(a) < k or 8 * n < k:
	print(-1)
else:
	s = 0
	i = 0
	while s < k:
		s += a[i] if a[i] < 8 else 8
		i += 1
		if i == n:
			break
		a[i] += 0 if a[i - 1] < 8 else a[i - 1] - 8
	if s < k:
		print(-1)
	else:
		print(i)
