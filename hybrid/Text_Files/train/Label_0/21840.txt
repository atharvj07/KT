int gcd(int x, int y){
    if (y == 0) return x;
    return gcd(y, x % y);
}

int main() {
    int p, q;
    cin >> p >> q;
    int d = gcd(p, q);
    q /= d;
    ll ans = 1;
    for (ll i = 2; i * i <= q; i++){
        if (q % i == 0){
            ans *= i;
            while (q % i == 0) q /= i;
        }
    }
    cout << ans*q << endl;

    return 0;
}
