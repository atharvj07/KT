from bisect import bisect_left
from sys import stdin

def input():
	return stdin.readline()[:-1]

def intput():
	return int(input())

def sinput():
	return input().split()

def intsput():
	return list(map(int, sinput()))
n = intput()
a = intsput()[::-1]
b = intsput()[::-1]
realloc = {}
for i in range(n):
	realloc[a[i]] = i
for i in range(n):
	b[i] = realloc[b[i]]
cnt = 0
cur = -1
for i in range(n):
	if cur > b[i]:
		cnt += 1
	else:
		cur = b[i]
print(cnt)
