#include <bits/stdc++.h>
int main() {
  int n, max, maxposit;
  scanf("%d", &n);
  int player[2 * n + 2];
  player[0] = 0;
  for (int i = 1; i <= 2 * n; i++) {
    scanf("%d", &player[i]);
  }
  for (int i = 1; i <= 2 * n; i++) {
    max = player[i];
    maxposit = i;
    for (int j = i; j <= 2 * n; j++) {
      if (player[j] >= max) {
        max = player[j];
        maxposit = j;
      }
    }
    player[maxposit] = player[i];
    player[i] = max;
  }
  if (player[n] > player[n + 1]) {
    printf("YES");
  } else {
    printf("NO");
  }
  return 0;
}
