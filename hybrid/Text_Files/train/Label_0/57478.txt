import java.io.*;
import java.util.*;
import java.math.*;


public class code{

	public static void main(String args[]){
		Scanner sc =new Scanner(System.in);
		//Writer wr=new Writer(System.out);
		int  lpf[]=new int[1000001];
		int N =1000001;
	
		lpf[0]=-1;
		lpf[1]=1;
		for(int i=2;i<N;i++){
			lpf[i]=i;
		}
		for(int i=2;i<N;i++){
			if(lpf[i]==i)
			{

				for(int j=i+i;j<N;j=j+i){
					if(lpf[j]==j){
						lpf[j]=i;
					}
				}
			}
		}
	
		 int ans=1;
		 int count=0;
		int cc=0;
		Vector<Integer>vec =new Vector<>();
		int n=sc.nextInt();
		while(n>1){
			cc=0;
			ans=ans*lpf[n];
			int t4=lpf[n];
			while(n!=1&&n%t4==0){
				n/=t4;
				cc++;
				
			}	
		vec.add(cc);
	}

		
		int is=1;
		for(int i=0;i<vec.size()-1;i++){
			if(vec.elementAt(i)!=vec.elementAt(i+1)){
				is=0;
			}
		}
		if(is==0){
			count++;
		}

		if(vec.size()>0){
			int max=vec.elementAt(0);
			for(int i=1;i<vec.size();i++){
				if(vec.elementAt(i)>max){
					max=vec.elementAt(i);
				}
			}
			if(is==1){
				int ttt=max&(max-1);
				if(ttt!=0)count++;
			}
			int t2=0;
			int t1=1;
			int flag=1;
			while(max!=1){
				if(max%2==1){
					flag=0;
				}
				max=max/2;
				t2++;
			}
			if(flag==0)t2++;
			count+=t2;

		}





		System.out.print(ans+" ");
		System.out.print(count);





	}


}
