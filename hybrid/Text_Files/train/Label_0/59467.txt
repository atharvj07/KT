import java.io.*;
import java.math.*;
import java.util.*;

public class Toposort {

    static BufferedReader cin = new BufferedReader(new InputStreamReader(System.in));
    static ArrayList<Integer>[] Graph;
    static ArrayList<Integer> Res;
    static boolean seen[];

    static void DFS(int src) {
        if (seen[src]) {
            return;
        }
        seen[src] = true;
        for (Integer v : Graph[src]) {
            if (!seen[v]) {
                DFS(v);
            }
        }
        Res.add(src);
        return;
    }

    public static void main(String[] args) throws Exception {
        int u, v, vertex, edge;
        String str[] = cin.readLine().split(" ");
        vertex = Integer.parseInt(str[0]);
        edge = Integer.parseInt(str[1]);
        
        seen = new boolean[vertex + 1];
        Arrays.fill(seen, false);
        Graph = new ArrayList[vertex + 1];
        Res = new ArrayList<>();
        for (int i = 0; i <= vertex; i++) {
            Graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < edge; i++) {
            str = cin.readLine().split(" ");
            u = Integer.parseInt(str[0]);
            v = Integer.parseInt(str[1]);
            Graph[u].add(v);
        }
        for (int i = 1; i <= vertex; i++) {
            if (!seen[i]) {
                DFS(i);
            }
        }
        //Collections.reverse(Res);
        for (Integer i : Res) {
            System.out.printf("%d ", i);
        }
        System.out.println("");
        cin.close();
    }
}
