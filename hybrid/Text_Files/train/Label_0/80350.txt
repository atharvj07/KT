import itertools
d = [(-1, 0), (1, 0), (0, -1), (0, 1)]
(n, m) = map(int, input().split())
s = [input() for i in range(n)]
a = input()
for i in range(n):
	for j in range(m):
		if s[i][j] == 'S':
			sx = i
			sy = j

def inside(x, y):
	return x >= 0 and x < n and (y >= 0) and (y < m)

def check(p):
	x = sx
	y = sy
	for i in a:
		id = p[int(i)]
		x += d[id][0]
		y += d[id][1]
		if not inside(x, y) or s[x][y] == '#':
			return 0
		if s[x][y] == 'E':
			return 1
	return 0
print(sum((check(i) for i in itertools.permutations([0, 1, 2, 3]))))
