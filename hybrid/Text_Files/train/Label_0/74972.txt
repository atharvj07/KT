t = int(input())
for _ in range(t):
	n = int(input())
	b = list(map(int, input().split()))
	if 2 * n in b:
		print(-1)
		continue
	arr = [0 for i in range(2 * n)]
	for i in b:
		arr[i - 1] = 1
	res = []
	for i in b:
		res.append(i)
		k = i
		while k < 2 * n:
			if arr[k] == 0:
				res.append(k + 1)
				arr[k] = 1
				break
			k += 1
	if len(res) < 2 * n:
		print(-1)
	else:
		for i in res:
			print(i, end=' ')
		print()
