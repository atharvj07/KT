#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string s;
  cin >> s;
  vector<pair<int, int> > v;
  int flg = -1;
  int pos;
  s += "a";
  for (int i = 0; i < s.size(); i++) {
    if (s[i] >= 'A' && s[i] <= 'Z' && flg == -1) {
      flg = i;
    } else if (s[i] >= 'a' && s[i] <= 'z' && flg != -1) {
      v.push_back(make_pair(flg, i - 1));
      flg = -1;
    }
  }
  if (v.size() == 0) {
    cout << "0\n";
  } else if (v[0].second + 1 == s.size() - 1 && v[0].first == 0) {
    cout << "0\n";
  } else {
    int n = s.size() - 1;
    int b[v.size()];
    int c[v.size()];
    for (int i = 0; i < v.size(); i++) {
      b[i] = v[i].second - v[i].first + 1;
    }
    c[0] = v[0].first;
    int pre[v.size()];
    pre[0] = c[0];
    for (int i = 1; i < v.size(); i++) {
      c[i] = v[i].first - v[i - 1].second - 1;
      pre[i] = pre[i - 1] + c[i];
    }
    int suff[v.size()];
    suff[v.size() - 1] = b[v.size() - 1];
    for (int i = v.size() - 2; i >= 0; i--) {
      suff[i] = suff[i + 1] + b[i];
    }
    int ans = 1000000;
    for (int i = 0; i < v.size() - 1; i++) {
      int cost = pre[i] + suff[i + 1];
      ans = min(cost, ans);
    }
    ans = min(ans, pre[v.size() - 1]);
    ans = min(ans, suff[0]);
    cout << ans << "\n";
  }
}
