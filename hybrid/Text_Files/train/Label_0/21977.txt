#include<bits/stdc++.h>
using namespace std;

int main(void){
   int N;
   cin>>N;
   vector<long long> A(N);
   for(int i=0;i<N;i++){
      cin>>A[i];
   }
   sort(A.begin(),A.end());
   long long result=0;
   for(int i=1;i<N;i++){
      result+=A[N-1-i/2];
   }
   cout<<result<<endl;
   return 0;
}
