str = input()
l = len(str)
a = []
for c in str:
	a.append(ord(c) - ord('a'))
a *= 2
tot = [[0] * l for i in range(26)]
for i in range(1, l):
	cnt = [[0] * 26 for i in range(26)]
	for j in range(l):
		cnt[a[j]][a[j + i]] += 1
	for j in range(l):
		if cnt[a[j]][a[j + i]] == 1:
			tot[a[j]][i] += 1
ans = 0
for sub in tot:
	ans += max(sub)
print(ans / l)
