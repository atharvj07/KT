#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, max = 0, c = 1;
  cin >> n;
  long long a[n];
  cin >> a[0];
  for (int i = 1; i < n; i++) {
    cin >> a[i];
    if (a[i] > a[i - 1])
      c++;
    else {
      if (max < c) max = c;
      c = 1;
    }
  }
  if (max < c) max = c;
  cout << max;
}
