#include <bits/stdc++.h>
using namespace std;
const int sz = 512;
const long long mod = 1000000007LL;
void add(int &a, int b) {
  a += b;
  a %= int(mod);
}
int times(long long a, long long b) { return int(a * b % mod); }
int e[sz][sz], dp[sz][sz];
int main() {
  for (int i = 0; i < sz; i++) e[i][0] = 1;
  e[0][1] = 1;
  for (int i = 1; i < sz; i++) e[i][1] = times(2, e[i - 1][1]);
  for (int i = 0; i < sz; i++)
    for (int j = 2; j < sz; j++) e[i][j] = times(e[i][j - 1], e[i][1]);
  for (int i = 0; i < sz; i++) dp[i][0] = 1;
  for (int k = 1; k < sz; k++)
    for (int n = 0; n < sz; n++)
      for (int l = 0; l < n; l++)
        add(dp[n][k], times(dp[l][k - 1], times(e[1][n - l] - 1, e[l][n - l])));
  int n, k;
  scanf("%d %d", &n, &k);
  printf("%d\n", dp[n][k]);
  return 0;
}
