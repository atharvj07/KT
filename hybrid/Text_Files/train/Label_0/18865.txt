#include<stdio.h>
#include<vector>
#include<algorithm>
using namespace std;
int map[500][500];
int ans1[500][500];
int main()
{
	int mx,my;
	scanf("%d%d",&mx,&my);
	for(int i=0;i<mx;i++)
	{
		for(int j=0;j<my;j++)
		{
			char z;
			scanf(" %c",&z);
			if(z=='#')map[i][j]=ans1[i][j]=1;
		}
	}
	for(int i=0;i<mx;i++)
	{
		for(int j=0;j<my;j++)
		{
			if(i==0||(0!=j&&j!=my-1&&0!=i&&i!=mx-1&&j%2==0)||map[i][j])printf("#");
			else printf(".");
		}
		printf("\n");
	}
	printf("\n");
	for(int i=0;i<mx;i++)
	{
		for(int j=0;j<my;j++)
		{
			if(i==mx-1||(0!=j&&j!=my-1&&0!=i&&i!=mx-1&&j%2==1)||map[i][j])printf("#");
			else printf(".");
		}
		printf("\n");
	}
}
