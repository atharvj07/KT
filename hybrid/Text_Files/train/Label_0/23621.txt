#include<bits/stdc++.h>
#define p pair<int,int>
#define P pair<int,p>
using namespace std;

vector<p>rinsetu[100];
vector<int>k[100];
int mincost[100][2];//0 ?????£????????? 1 ?????£??? 
void saiki(int a,int b,int c){//start now ?????°
    if(c==2){
	k[a].push_back(b);
	}
    else{
	for(p i:rinsetu[b]){
	    saiki(a,i.second,c+1);
	    }
	}
    }
int main(){
    int a,b;
    while(cin>>a>>b,a|b){
	for(int c=0;c<a;c++){
	    mincost[c][0]=mincost[c][1]=1<<29;
	    rinsetu[c].clear();k[c].clear();
	    }
	for(int c=0;c<b;c++){
	    int d,e,f;scanf("%d%d%d",&d,&e,&f);d--;e--;
	    rinsetu[d].push_back(p(f,e));
	    rinsetu[e].push_back(p(f,d));
	    }
      for(int c=0;c<a;c++)saiki(c,c,0);
      mincost[0][0]=0;
      priority_queue<P,vector<P>,greater<P>>Q;
      Q.push(P(0,p(0,0)));
      while(Q.size()){
	  P t=Q.top();Q.pop();
	  for(p i:rinsetu[t.second.first]){
	      if(mincost[i.second][t.second.second]>t.first+i.first){
		  mincost[i.second][t.second.second]=t.first+i.first;
		  Q.push(P(t.first+i.first,p(i.second,t.second.second)));
		  }
	      }
	      if(t.second.second==0){
		  for(int i:k[t.second.first]){
		      if(mincost[i][1]>t.first){
			  mincost[i][1]=t.first;
			  Q.push(P(t.first,p(i,1)));
			  }
		      }
		  }}
	  cout<<min(mincost[a-1][0],mincost[a-1][1])<<endl;
	  }
    }
