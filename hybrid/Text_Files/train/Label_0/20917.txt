#include <bits/stdc++.h>
using namespace std;
long long tests;
long long n, m;
map<string, bool> s;
string t[1000256];
void solve() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> t[i];
  for (int i = n - 1; i >= 0; i--) {
    if (s[t[i]] == 0) cout << t[i] << endl;
    s[t[i]] = 1;
  }
}
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  solve();
}
