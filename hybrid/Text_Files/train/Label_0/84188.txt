mod = 998244353

def egcd(a, b):
	if a == 0:
		return (b, 0, 1)
	else:
		(g, y, x) = egcd(b % a, a)
		return (g, x - b // a * y, y)

def modinv(a, m):
	(g, x, y) = egcd(a, m)
	if g != 1:
		raise Exception('modular inverse does not exist')
	else:
		return x % m

def choose(n, k, p):
	if k == 0:
		return 1
	x = n
	for y in range(n - 1, n - k, -1):
		x = x * y % p
	d = 1
	for i in range(2, k + 1):
		d = d * i % p
	return x * modinv(d, p) % p

def solve():
	n = int(input())
	wb_poss = n > 1
	bw_poss = n > 1
	w = 0
	b = 0
	q = 0
	free = 0
	any_bad_poss = n > 1
	for _ in range(n):
		s = input().strip()
		if s[0] == 'B':
			b += 1
			wb_poss = False
		elif s[0] == 'W':
			w += 1
			bw_poss = False
		elif s[0] == '?':
			q += 1
		if s[1] == 'B':
			b += 1
			bw_poss = False
		elif s[1] == 'W':
			w += 1
			wb_poss = False
		elif s[1] == '?':
			q += 1
		if s == '??':
			free += 1
		elif s[0] == s[1]:
			any_bad_poss = False
	w_needed = n - w
	if w_needed > q or w_needed < 0:
		print(0)
		return
	if any_bad_poss:
		num_bad_poss = pow(2, free, mod)
		num_bad_poss -= int(bw_poss) + int(wb_poss)
	else:
		num_bad_poss = 0
	print((choose(q, w_needed, mod) - num_bad_poss + mod) % mod)
solve()
