(n, m, b, mod) = map(int, input().split())
a = list(map(int, input().split()))
dp = [[0 for i in range(m + 1)] for j in range(b + 1)]
dp[0][0] = 1
for i in range(n):
	for j in range(a[i], b + 1):
		for k in range(m):
			dp[j][k + 1] = (dp[j][k + 1] + dp[j - a[i]][k]) % mod
res = 0
for i in range(b + 1):
	res = (res + dp[i][m]) % mod
print(res)
