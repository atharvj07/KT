#include<iostream>
#include<algorithm>
#include<string>
#include<vector>
#include<cmath>
#define for0(i, n) for(int i = 0; i < (n); i++)
#define for1(i, n) for(int i = 1; i <= (n);i++)
using namespace std;
int get() { int r; cin >> r; return r; }
int dist(int a, int b, int c, int d) { return abs(a - c) + abs(b - d); }
int n, a, b, c, d, xp, yp;
signed main() {
	while (cin >> n, n) {
		cin >> a >> b >> c >> d >> xp >> yp;
		int ans = 0;
		for1(i, n) {
			int x, y; cin >> x >> y;
			int xp1 = xp, yp1 = yp, x1 = x, y1 = y;
			if (xp1 < a)xp1 = a;
			if (x1 < a)x1 = a;
			if (xp1 > c)xp1 = c;
			if (x1 > c)x1 = c;
			if (yp1 < b)yp1 = b;
			if (y1 < b)y1 = b;
			if (yp1 > d)yp1 = d;
			if (y1 > d)y1 = d;
			ans += min(dist(xp, yp, x, y), max(dist(xp, yp, xp1, yp1) - 1, 0) + dist(x, y, x1, y1));
			xp = x; yp = y;
		}
		cout << ans << endl;
	}
}
