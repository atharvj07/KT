import java.util.*;

public class SuitAndTie {
	public static void main(String[] args) {
		Scanner scan=new Scanner(System.in);
		int n=scan.nextInt();
		int[] a=new int[2*n];
		Integer[][] ind=new Integer[n][2];
		int[] dist=new int[n];
		
		for(int i=0;i<2*n;i++) {
			int x=scan.nextInt()-1;
			a[i]=x;
			if(ind[x][0]==null) ind[x][0]=i;
			else {
				ind[x][1]=i;
				dist[x]=ind[x][1]-ind[x][0]-1;
			}
		}
		int res=0;
		HashSet<Integer> done=new HashSet<>();
		for(int i=0;i<2*n;i++) {
			if(done.contains(a[i])) continue;
//			System.out.println();
//			System.out.println("add "+dist[a[i]]);
			res+=dist[a[i]];
			for(int j=0;j<n;j++) {
				if(j==a[i]) continue;
//				System.out.println("is "+(a[i]+1)+" between "+(j+1));
//				System.out.println(ind[j][0]+" "+ind[a[i]][1]+" "+ind[j][1]);
				if(between(ind[j][0],ind[a[i]][1],ind[j][1])) {
					dist[j]--;
//					System.out.println("yea biatch");
				}
			}
//			System.out.println(Arrays.toString(dist));
			done.add(a[i]);
		}
		System.out.println(res);
	}
	public static boolean between(int a, int b, int c) {
		return b>=a&&b<=c;
	}
}
