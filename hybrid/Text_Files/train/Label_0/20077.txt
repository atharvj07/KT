#include <bits/stdc++.h>
using namespace std;
long long int a[501][501];
long long int sum1[501], sum2[501];
int f(int n) {
  int p = 0, q = 0;
  memset(sum1, 0, sizeof(sum1));
  memset(sum2, 0, sizeof(sum2));
  long long int r1 = 0, r2 = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      sum1[p] += a[i][j];
      if (i == j) {
        r1 += a[i][j];
      }
      if (i + j == n - 1) {
        r2 += a[i][j];
      }
    }
    p++;
  }
  for (int j = 0; j < n; j++) {
    for (int i = 0; i < n; i++) {
      sum2[q] += a[i][j];
    }
    q++;
  }
  for (int i = 0; i < p - 1; i++) {
    if (sum1[i] != sum1[i + 1]) return 0;
  }
  for (int i = 0; i < q - 1; i++) {
    if (sum2[i] != sum2[i + 1]) return 0;
  }
  if (sum1[0] != sum2[0]) return 0;
  if (r1 != r2) return 0;
  if (r1 != sum1[0])
    return 0;
  else
    return 1;
}
int main() {
  int n;
  int l = 505, k = 505;
  while (~scanf("%d", &n)) {
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        cin >> a[i][j];
        if (a[i][j] == 0) {
          l = i;
          k = j;
        }
      }
    }
    if (n == 1) {
      printf("1\n");
      continue;
    }
    if (f(n)) {
      printf("-1\n");
      continue;
    }
    long long int s1 = 0, s2 = 0;
    for (int i = 0; i < n; i++) {
      s1 += a[l][i];
    }
    for (int i = 0; i < n; i++) {
      if (i != l) {
        for (int j = 0; j < n; j++) {
          s2 += a[i][j];
        }
        break;
      }
    }
    long long int o = s2 - s1;
    a[l][k] = o;
    if (f(n)) {
      cout << (o < 0 ? -1 : o) << endl;
    } else {
      printf("-1\n");
    }
  }
  return 0;
}
