from math import ceil
for _ in range(int(input())):
	n = int(input())
	(a, b) = ([0] * (n + 1), [0] * (n + 1))
	for i in range(1, n + 1):
		(a[i], b[i]) = list(map(int, input().split()))
	tm = [0] + list(map(int, input().split()))
	travel_time = 0
	dep = 0
	for i in range(1, n):
		travel_time = dep + a[i] - b[i - 1] + tm[i]
		dep = max(b[i], travel_time + ceil((b[i] - a[i]) / 2))
	travel_time = dep + a[n] - b[n - 1] + tm[n]
	print(travel_time)
