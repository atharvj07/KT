import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
E = [0] * n
ans = n
for i in range(m):
	(u, v) = map(int, input().split())
	u -= 1
	v -= 1
	if u > v:
		(u, v) = (v, u)
	E[u] += 1
	if E[u] == 1:
		ans -= 1
q = int(input())
for i in range(q):
	tmp = input()
	if len(tmp) > 3:
		(d, u, v) = map(int, tmp.split())
		u -= 1
		v -= 1
		if u > v:
			(u, v) = (v, u)
		if d == 1:
			E[u] += 1
			if E[u] == 1:
				ans -= 1
		else:
			E[u] -= 1
			if E[u] == 0:
				ans += 1
	else:
		print(ans)
