n = int(input())
pine = dict()
mine = dict()
for _ in range(n):
	(a, b) = map(int, input().split())
	pine[a + b] = pine.setdefault(a + b, 0) + 1
	mine[a - b] = mine.setdefault(a - b, 0) + 1
rest = 0
for v in pine.values():
	if v > 1:
		rest += v * (v - 1) // 2
for v in mine.values():
	if v > 1:
		rest += v * (v - 1) // 2
print(rest)
