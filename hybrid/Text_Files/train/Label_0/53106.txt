import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Solution_505B {
    public static boolean[][][] colorMatrix;
    public static boolean[] used;

    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String strings[] = reader.readLine().split(" ");
        int nodes = Integer.parseInt(strings[0]);   //количество вершин графа
        int edges = Integer.parseInt(strings[1]);   // количество ребер графа

         colorMatrix = new boolean[edges + 1][nodes + 1][nodes + 1];
         used = new boolean[nodes + 1];

        for (int i = 1; i <= edges; i++) {
            String edge[] = reader.readLine().split(" ");
            int apex1 = Integer.parseInt(edge[0]);
            int apex2 = Integer.parseInt(edge[1]);
            int color = Integer.parseInt(edge[2]);

            colorMatrix[color][apex1][apex2] = true;
            colorMatrix[color][apex2][apex1] = true;
        }

        int requests = Integer.parseInt(reader.readLine());
        for (int i = 0; i < requests; i++) {
            String edge[] = reader.readLine().split(" ");
            int apex1 = Integer.parseInt(edge[0]);
            int apex2 = Integer.parseInt(edge[1]);
            int num = 0;

            for (int j = 1; j <= edges; j++) {
                used = new boolean[nodes + 1];
                DFS(j, apex1, nodes);
                if (used[apex2])
                    num++;
            }

            System.out.println(num);
        }
    }

    public static void DFS(int color, int v, int nodes) {
        used[v] = true; // помечаем вершину
        for (int nv = 1; nv <= nodes; nv++) // перебираем вершины
            if (!used[nv] && colorMatrix[color][v][nv]) // если вершина не помечена, и смежна с текущей
                DFS(color, nv, nodes); // рекурсивно запускаем от нее DFS
    }

}
