#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;
long long C(long long m, long long n) {
  if (n > m) return 0;
  if (n > m / 2) return C(m, m - n);
  long long ret = 1;
  for (int i = 1; i <= n; i++) {
    ret = ret * (m - i + 1) / i;
  }
  return ret;
}
int main() {
  long long n, m, t;
  cin >> n >> m >> t;
  long long ans = C(n + m, t);
  for (long long i = 0; i < 4; i++) {
    ans -= C(n, i) * C(m, t - i);
  }
  ans -= C(n, t);
  cout << ans << endl;
  return 0;
}
