#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;
vector<int>G[2000];
string s;
int N;
long ans=9e18;
long nowret;
bool vis[2000];
pair<int,pair<int,int> >dfs(int u,int p)
{
	int ct=s[u]=='1';
	if(G[u].size()==1&&G[u][0]==p)
	{
		return make_pair(ct,make_pair(0,0));
	}
	vector<pair<int,int> >A;
	for(int v:G[u])
	{
		if(v!=p)
		{
			pair<int,pair<int,int> >p=dfs(v,u);
			ct+=p.first;
			A.push_back(make_pair(p.first+p.second.first,p.first+p.second.second));
		}
	}
	sort(A.begin(),A.end());
	int sum=0,mv=0;
	for(int i=0;i+1<A.size();i++)sum+=A[i].second;
	mv=A.back().first;
	if(mv<=sum)
	{
		int now=sum+A.back().second;
		return make_pair(ct,make_pair(now%2,now));
	}
	else
	{
		return make_pair(ct,make_pair(mv-sum,sum+A.back().second));
	}
}
int main()
{
	cin>>N>>s;
	for(int i=1;i<N;i++)
	{
		int a,b;cin>>a>>b;
		a--,b--;
		G[a].push_back(b);
		G[b].push_back(a);
	}
	for(int i=0;i<N;i++)
	{
		nowret=0;
		pair<int,pair<int,int> >now=dfs(i,-1);
		if(now.second.first==0&&now.second.second%2==0)ans=min(ans,1L*now.second.second/2);
	}
	if(ans<9e18)cout<<ans<<endl;
	else cout<<-1<<endl;
}
