import java.util.*;
import java.io.*;

public class ArpaAmpitheater {
    public static void main(String[] args) {
        FastScanner scanner = new FastScanner();
        int N = scanner.nextInt();
        int M = scanner.nextInt();
        int W = scanner.nextInt();
        int[] beauty = new int[N];
        int[] weight = new int[N];
        DSU dsu = new DSU(N);
        for(int i = 0; i < N; i++) {
            weight[i] = scanner.nextInt();
            //dsu.total[i]+=weight[i];
            //dsu.max[i] = weight[i];
        }
        for(int i = 0; i < N; i++) {
            beauty[i] = scanner.nextInt();
        }
        for(int i = 0; i < M; i++) {
            int u = scanner.nextInt()-1;
            int v = scanner.nextInt()-1;
            dsu.union(u, v);
        }
        ArrayList<Integer>[] groups = new ArrayList[N];
        for(int i = 0; i < N; i++) groups[i] = new ArrayList<>();
        for(int i = 0; i < N; i++) {
            int pi = dsu.find(i);
            groups[pi].add(i);
        }
        int[] dp = new int[W+1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for(int i = 0; i < N; i++) {
            if (groups[i].size() == 0) continue;
            int curTotB = 0;
            int curTotW = 0;
            int[] dpNew = new int[W+1];
            for(int hos: groups[i]) {
                curTotB += beauty[hos];
                curTotW += weight[hos];
                for(int j = 0; j <= W-weight[hos]; j++) {
                    //not a valid party
                    if (dp[j] == -1) continue;
                    //valid party
                    dpNew[j+weight[hos]] = Math.max(dpNew[j+weight[hos]], dp[j] + beauty[hos]);
                }
            }
            for(int j = 0; j <= W-curTotW; j++) {
                if (dp[j] == -1) continue;
                dpNew[j+curTotW] = Math.max(dpNew[j+curTotW], dp[j]+curTotB);
            }
            for(int j = 0; j <= W; j++) dp[j] = Math.max(dp[j], dpNew[j]);
        }
        int max = 0;
        for(int i = 0; i <= W; i++) {
            max = Math.max(max, dp[i]);
        }
        System.out.println(max);
    }
    public static class DSU {
        int[] parent, rank, max, total;
        int nSets, sz;
        public DSU (int ss) {
            parent = new int[ss];
            rank = new int[ss];
            //max = new int[ss];
            //total = new int[ss];
            nSets = ss;
            sz = ss;
            for(int i = 0; i < ss; i++) {
                parent[i] = i;
                rank[i] = 1;
            }
        }
        public int find(int t) {
            int prev = t;
            while(parent[t] != t) t = parent[t];
            compress(prev, t);
            return t;
        }
        public void compress(int s, int t) {
            while(parent[s] != t) {
                int temp = parent[s];
                parent[s] = t;
                s = temp;
            }
        }
        public boolean union(int a, int b) {
            int pa = find(a);
            int pb = find(b);
            if (pa == pb) return false;
            nSets--;
            if (rank[pa] <= rank[pb]) {
                rank[pb]++;
                //max[pb] = Math.max(max[pa], max[pb]);
                //total[pb] += total[pa];
                parent[pa] = pb;
            }
            else {
                rank[pa]++;
                parent[pb] = pa;
                //max[pa] = Math.max(max[pa], max[pb]);
                //total[pa]+=total[pa];
            }
            return true;
        }
    }
    public static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        public FastScanner(Reader in) {
            br = new BufferedReader(in);
        }

        public FastScanner() {
            this(new InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String readNextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }

        int[] readIntArray(int n) {
            int[] a = new int[n];
            for (int idx = 0; idx < n; idx++) {
                a[idx] = nextInt();
            }
            return a;
        }
    }
}
