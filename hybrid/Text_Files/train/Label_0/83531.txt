import sys
input = lambda : sys.stdin.readline().rstrip()
q = int(input())
for _ in range(q):
	(n, r) = list(map(int, input().split()))
	arr = sorted(list(set(map(int, input().split()))), reverse=True)
	result = 0
	tt = 0
	for el in arr:
		if tt >= el:
			break
		result += 1
		tt += r
	print(result)
