import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Set;
import java.util.StringTokenizer;
import java.util.TreeSet;

public class d123 {

    public static void debug(Object... obs)
    {
        System.out.println(Arrays.deepToString(obs));
    }

    static long gcd(long a,long b)
    {
        return (b==0)?a:gcd(b,a%b);
    }
    
    public static void main(String[] args) throws Exception
    {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(br.readLine());
        
        Set<P> pts = new TreeSet<P>();
        
        for(int i=0;i<n;i++)
        {
            StringTokenizer st=new StringTokenizer(br.readLine());
            long ki = Long.parseLong(st.nextToken());
            long bi = Long.parseLong(st.nextToken());
            
            if(ki == 0) continue;
            
            
            long ak = Math.abs(ki);
            long ab = Math.abs(bi);
            
            long gd = gcd(ak,ab);
            
            ki = ki/gd;
            bi = bi/gd;
            
            P pair = new P(ki,bi);
            pts.add(pair);
        }
        
        System.out.println(pts.size());
    }
    
    private static class P implements Comparable<P>
    {
        long k, b;
        int sgn;

        public P(long k, long b)
        {
            super();
            if(b < 0 && k < 0)
                sgn = 1;
            else if(b > 0 && k > 0)
                sgn = 1;
            else if(b == 0)
                sgn = 1;
            else
                sgn = 2;
            
            this.k = Math.abs(k);
            this.b = Math.abs(b);
        }

        @Override
        public int compareTo(P arg0)
        {
            if(b == 0 && arg0.b ==0)
                return 0;
                
            if(k - arg0.k != 0)
                return sgn(k - arg0.k);
            
            if(b - arg0.b != 0)
                return sgn(b - arg0.b);
            
            return sgn - arg0.sgn;
        }

        private int sgn(long l)
        {
            if(l < 0) return -1;
            if (l > 0) return 1;
            return 0;
        }
    }
}
