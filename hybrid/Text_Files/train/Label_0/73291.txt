t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	for i in range(n):
		arr[i] = k - arr[i] % k
		if arr[i] == k:
			arr[i] = 0
	counts = {}
	for i in arr:
		if i not in counts.keys():
			counts[i] = 1
		else:
			counts[i] += 1
	if 0 in counts.keys():
		del counts[0]
	if counts == {}:
		x = 0
	else:
		x = max(counts.values())
	if x == 0 or counts == []:
		print(0)
	else:
		key = list(counts.keys())
		key.sort()
		for i in key[::-1]:
			if counts[i] == x:
				print((x - 1) * k + i + 1)
				break
