s = input()
ls = len(s)
s1 = s[::-1]
d = dict()
t = list()
u = list()
sets = sorted(set(s))
setsl = len(sets)
for i in range(setsl):
	try:
		idx = s1.index(sets[i])
		if i == 0:
			d[sets[i]] = (0, ls - idx - 1)
			prev = sets[i]
		elif ls - idx - 1 > d[prev][1]:
			d[sets[i]] = (d[prev][1] + 1, ls - idx - 1)
			prev = sets[i]
	except:
		pass
for i in d:
	while t and t[-1] <= i:
		u.append(t.pop())
	for j in range(d[i][0], d[i][1] + 1):
		if s[j] == i:
			u.append(s[j])
		else:
			t.append(s[j])
u = ''.join(u + t[::-1])
print(u)
