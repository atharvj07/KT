#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, j, cnt = 0, a[101], sum = 0, k = 0, c;
  cin >> n;
  for (i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (i = 0; i < n; i++) {
    int flag = 0;
    for (j = i + 1; j < n && a[i]; j++) {
      if (a[i] == a[j]) {
        cnt++;
        flag = 1;
        a[j] = NULL;
      }
    }
    if (flag == 1) {
      c = (1 + cnt) / 2;
      sum = sum + c;
      flag = 0;
      cnt = 0;
    }
  }
  cout << sum / 2 << endl;
  return 0;
}
