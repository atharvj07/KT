import atexit
import io
import sys
_I_B = sys.stdin.read().splitlines()
input = iter(_I_B).__next__
_O_B = io.StringIO()
sys.stdout = _O_B

@atexit.register
def write():
	sys.__stdout__.write(_O_B.getvalue())

def main():
	(u, v) = map(int, input().split())
	if u > v:
		print(-1)
	elif u == v:
		if u:
			print(1, u, sep='\n')
		else:
			print(u)
	elif v - u & 1:
		print(-1)
	elif u + (v - u) // 2 == u ^ (v - u) // 2:
		print(2)
		print((v - u) // 2, u + (v - u) // 2)
	else:
		print(3)
		print(u, (v - u) // 2, (v - u) // 2)
main()
