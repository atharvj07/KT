#include <bits/stdc++.h>
using namespace std;
const int N = 100000 + 10;
const int M = 1000000007;
const double PI = atan(1) * 4;
const int oo = 2100000000;
int n;
long long dp[N][8];
vector<vector<int> > v;
long long calc(int u, int st) {
  if (u == n) return (st != 0b001) * -1e17;
  long long &ret = dp[u][st];
  if (ret != -1) return ret;
  if (st == 0b100)
    ret = 0ll + v[0][u] +
          max(max(calc(u + 1, 0b100), v[1][u] + calc(u + 1, 0b010)),
              v[1][u] + v[2][u] + max(calc(u + 1, 0b111), calc(u + 1, 0b001)));
  if (st == 0b010)
    ret = 0ll + v[1][u] +
          max(calc(u + 1, 0b010),
              max(v[0][u] + calc(u + 1, 0b100), v[2][u] + calc(u + 1, 0b001)));
  if (st == 0b001)
    ret = 0ll + v[2][u] +
          max(max(calc(u + 1, 0b001), v[1][u] + calc(u + 1, 0b010)),
              v[0][u] + v[1][u] + max(calc(u + 1, 0b111), calc(u + 1, 0b100)));
  if (st == 0b111)
    ret = 0ll + v[0][u] + v[1][u] + v[2][u] +
          max(calc(u + 1, 0b111), max(calc(u + 1, 0b100), calc(u + 1, 0b001)));
  return ret;
}
int main() {
  cin >> n;
  v.resize(3);
  memset(dp, -1, sizeof(dp));
  for (int i = 0; i < 3; ++i) {
    v[i].resize(n);
    for (int j = 0; j < n; ++j) scanf("%d", &v[i][j]);
  }
  cout << calc(0, 0b100);
}
