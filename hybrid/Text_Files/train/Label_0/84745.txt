def main():
	a = b = res = 0
	for c in map({'0': 0, '1': 1, '2': 2, '3': 0, '4': 1, '5': 2, '6': 0, '7': 1, '8': 2, '9': 0}.get, input()):
		if not c:
			res += 1
			a = b = 0
		elif c == 1:
			if a == 2 or b == 1:
				res += 1
				a = b = 0
			elif a:
				b = 1
			else:
				a = 1
		elif a == 1 or b == 1:
			res += 1
			a = b = 0
		elif a:
			b = 1
		else:
			a = 2
	print(res)
main()
