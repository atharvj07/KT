import io, os, sys
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
output = sys.stdout.write

def solve(N, grid):
	sz = [0] * 5
	cols = [0] * 5
	for d in range(5):
		for r in range(N):
			cols[d] |= grid[r][d] << r
			sz[d] += grid[r][d]
	target = (1 << N) - 1
	for d1 in range(5):
		for d2 in range(d1 + 1, 5):
			if cols[d1] | cols[d2] == target and sz[d1] >= N // 2 and (sz[d2] >= N // 2):
				return 'YES\n'
	return 'NO\n'

def main():
	T = int(input())
	for _ in range(T):
		N = int(input())
		grid = [list(map(int, input().split())) for _ in range(N)]
		out = solve(N, grid)
		output(out)
main()
