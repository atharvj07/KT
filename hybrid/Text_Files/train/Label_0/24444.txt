#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, a[101], b[101];
  vector<int> v[101];
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &x);
    a[x] = i + 1;
  }
  for (int i = 0; i < n; i++) {
    scanf("%d", &b[i]);
    if (b[i] + i < n) {
      v[i + 1].push_back(b[i] + i + 1);
      v[b[i] + i + 1].push_back(i + 1);
    }
    if (i + 1 - b[i] > 0) {
      v[i + 1].push_back(i + 1 - b[i]);
      v[i + 1 - b[i]].push_back(i + 1);
    }
  }
  int fl, f = 1;
  for (int k = 1; k <= n; k++) {
    if (a[k] == k) continue;
    vector<int> v1, v2;
    v1.push_back(k);
    int mark[101];
    memset(mark, 0, sizeof(mark));
    mark[k] = 1;
    int ans;
    fl = 1;
    while (1) {
      for (int i = 0; i < v1.size(); i++) {
        for (int j = 0; j < v[v1[i]].size(); j++) {
          if (mark[v[v1[i]][j]] == 0) {
            if (a[k] == v[v1[i]][j]) {
              fl = 0;
              break;
            }
            v2.push_back(v[v1[i]][j]);
            mark[v[v1[i]][j]] = 1;
          }
        }
        if (fl == 0) break;
      }
      if (fl == 0) break;
      v1.clear();
      v1 = v2;
      v2.clear();
      if (v1.empty()) break;
    }
    if (fl) {
      f = 0;
      break;
    }
  }
  if (f == 0)
    printf("NO");
  else
    printf("YES");
}
