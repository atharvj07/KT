n = int(input())
A = [int(i) for i in input().split()]

def solution(n, A):
	right_doors = sum(A)
	if right_doors == 0 or right_doors == n:
		return n
	left_doors = n - right_doors
	count = 0
	i = 0
	while left_doors > 0 and right_doors > 0:
		if A[i] == 0:
			left_doors -= 1
		else:
			right_doors -= 1
		count += 1
		i += 1
	return count
print(solution(n, A))
