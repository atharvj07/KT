
import java.util.ArrayList;
import java.util.Scanner;

public class IndianSummer
{
	public static void main(String[] args) 
	{
		Scanner scan=new Scanner(System.in);
		int n=scan.nextInt();
		Mapp m=new Mapp(n);
		for(int i=0;i<n;i++) 
		{
			String key=scan.next();
			String value=scan.next();
			m.add(key, value);
		}
		System.out.println(m.filled);
		
	}

}

class Mapp<K,V>
{
	ArrayList<Hashnodee<K, V>> a;
	Hashnodee h;
	int size;
	int filled;
	public Mapp(int size )
	  {
		this.size=size;
		//System.out.println(size);
		a=new ArrayList<Hashnodee<K,V>>(size);
		for(int i=0;i<size;i++) 
		 {
		   a.add(i,null);
		 } 
	  }
	public ArrayList<Hashnodee<K, V>> get()
	  { 
		return a;
	  }
	public int getHashSize() 
	  {
		return a.size();
	  } 
	public int getKey(K key) //to get the index of an array where the value will be stored.
	  {
		if(size==0)size=+1;
		//System.out.println("hashcode:"+key.hashCode()+" key:"+key+" size "+size);
		int k=key.hashCode();
		if(k<0) 
		{
			 k=-k;;
		}
//		System.out.println(k%size+" "+k);
		
		
		return (k)%size;
	  }
	public void add(K key,V value)
	  {
	    int hashIndex=getKey(key);
		if(a.get(hashIndex)==null)
		 {
			Hashnodee<K,V> h=a.get(hashIndex);
			h=new Hashnodee<K,V>(key,value);
			h.next=null;
			a.set(hashIndex,h);
			filled++;
	 	 }
		else 
		 {
			Hashnodee<K,V> h=a.get(hashIndex);
			if(h.key.equals(key)&&h.value.equals(value)) 
			{
				return;
			}
			while(h.next!=null) 
			{
				h=h.next;
				if(h.key.equals(key)&&h.value.equals(value)) 
				{
					return;
				}
			}
			h.next=new Hashnodee<K,V>(key,value);
			h.next.next=null;
			filled++;
		 } 
	  }
	public void show(K key) 
	{
		int hashIndex=getKey(key);
		Hashnodee<K,V> itr=a.get(hashIndex);
	//	System.out.println(":"+itr);
		while(itr!=null) 
		{
			System.out.println(itr.key+" = "+itr.value);
			itr=itr.next;
		}
	

	}
	public void showAll() 
	{
		for(int i=0;i<a.size();i++) 
		{
			Hashnodee<K,V> h=a.get(i);
			if(h==null)
			{
				System.out.println(i+" "+null);
			}
			else 
			{
			  while(h!=null)
			   {
				System.out.println(i+" Key:"+h.key+" value: "+h.value);
				h=h.next;
			   }
			}
		}
	}
	public Hashnodee<K,V> getNode(K key)
	{
		return a.get(getKey(key));
	}
}
class Hashnodee<K,V>
{
	K key;
	V value;
	Hashnodee next=null;
	public Hashnodee(K Key,V Value)
	 {
		this.key=Key;
		this.value=Value;
	}
}
