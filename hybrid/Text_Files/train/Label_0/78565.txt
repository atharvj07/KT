for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	x = 2 ** n - 1
	d = []
	an = 'NO'
	for i in range(x + 1):
		c = sum([a[j] for j in range(n) if i & 1 << j])
		if c in d:
			an = 'YES'
			break
		else:
			d.append(c)
	print(an)
