for _ in range(int(input())):
	n = int(input())
	s = input()
	(lef, ri, sw) = (0, 0, 0)
	for i in range(n - 1):
		if s[i] > s[i + 1]:
			sw = 1
			break
	if sw == 0:
		print(s)
		continue
	for i in range(n):
		if s[i] == '1':
			lef = i
			break
	for i in range(n - 1, 0, -1):
		if s[i] == '0':
			ri = i
			break
	print(s[:lef] + '0' + s[ri + 1:])
