#include <bits/stdc++.h>
using namespace std;
int n, k, a[100], m, y, x;
string s;
char b[109];
int main() {
  cin >> n >> s;
  for (int i = 0; i < s.size(); i++) b[i] = s[i];
  for (int i = 2; i <= n; i++) {
    if (n % i == 0) {
      a[x] = i;
      x++;
    }
  }
  for (int i = 0; i < x; i++) reverse(b, b + a[i]);
  for (int i = 0; i < s.size(); i++) cout << b[i];
}
