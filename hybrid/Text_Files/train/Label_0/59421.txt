    static int fn()
    {
        int[][] dp = new int[n+1][n+1];
        for(int i = n-1; i >= 0; i--) {
        	for(int j = i; j < n; j++) {
        		if(i == j) {dp[i][j] = 1; continue;}
        		dp[i][j] = n+1;
        		dp[i][j] = min(dp[i][j], 1 + dp[i+1][j]);
        		if(arr[i] == arr[i+1]) dp[i][j] = min(dp[i][j], 1 + dp[i+2][j]);
        		for(int k = i+2; k <= j; k++) {
        			if(arr[i] == arr[k]) {
        				dp[i][j] = min(dp[i][j], dp[i+1][k-1] + dp[k+1][j]);
        			}
        		}
        	}
        }
        
        return dp[0][n-1];
    }
