def f(n, m, A, B):
	s_w_a = []
	s_w_b = []
	for i in range(n):
		s_w_a.append(0)
		s_w_b.append(0)
		for j in range(m):
			s_w_a[-1] += A[i][j]
			s_w_b[-1] += B[i][j]
	s_h_a = []
	s_h_b = []
	for i in range(m):
		s_h_a.append(0)
		s_h_b.append(0)
		for j in range(n):
			s_h_a[-1] += A[j][i]
			s_h_b[-1] += B[j][i]
	for i in range(len(s_w_a)):
		if (s_w_a[i] - s_w_b[i]) % 2 != 0:
			return 'No'
	for i in range(len(s_h_a)):
		if (s_h_a[i] - s_h_b[i]) % 2 != 0:
			return 'No'
	return 'Yes'
(n, m) = list(map(int, input().split()))
A = []
for i in range(n):
	A.append(list(map(int, input().split())))
B = []
for i in range(n):
	B.append(list(map(int, input().split())))
print(f(n, m, A, B))
