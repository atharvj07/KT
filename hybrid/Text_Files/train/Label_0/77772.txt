for _ in range(int(input())):
	a = input()
	b = input()
	n = len(a)
	m = len(b)
	(d1, d2) = (dict(), dict())
	for i in range(n):
		for j in range(i + 1, n + 1):
			d1[a[i:j]] = n - j + i
	for i in range(m):
		for j in range(i + 1, m + 1):
			d2[b[i:j]] = m - j + i
	d1[''] = n
	d2[''] = m
	ans = 10 ** 18
	for item in d1.keys():
		for item1 in d2.keys():
			if item1 == item:
				ans = min(ans, d1[item] + d2[item1])
	print(ans)
