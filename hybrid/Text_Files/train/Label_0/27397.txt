#include<bits/stdc++.h>
using namespace std;
#define FastIO ios_base::sync_with_stdio(false); cin.tie(0);
#define ll long long
#define ull unsigned long long
#define pb push_back
#define All(x) (x).begin(),(x).end()
#define mp make_pair
#define nl "\n"
typedef pair<int,int>ii;
typedef vector<int>vi;
typedef vector<ii>vii;
typedef vector<vi>vvi;
typedef vector<long long>vl;
typedef vector <vl>vvl;
template<class T>
void print(T& a)
{
    for(auto x:a)
        cout<<x<<" ";
    cout<<"\n";
}
int main()
{

    FastIO

    int tt;
    cin>>tt;

    while(tt--)
    {
        ll n,mx=0,sum=0;
        cin>>n;

        vl st(n);
        for(ll & x:st)
        {
            cin>>x;
            sum+=x;
            mx=max(mx,x);
        }

        n--;
        ll r;
        r=n*mx;

        ll ans=0;

        if(r>sum)
        {
            ans=r-sum;
        }

        else
        {
            if(sum%n)
            {
                ans=(((sum/n)+1)*n)-sum;
            }
        }
        cout<<ans<<nl;
    }

    return 0;
}
