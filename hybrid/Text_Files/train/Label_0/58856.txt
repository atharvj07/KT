import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;


public class cf167b {
  static int n, l;
  static int[] a;
  static double[] p;
  static double[][][] memo;
  public static void main(String[] args) {
    FastIO in = new FastIO(), out = in;
    n = in.nextInt();
    l = in.nextInt();
    int k = in.nextInt();
    p = new double[n];
    for(int i=0; i<n; i++) p[i] = in.nextInt()/100.;
    a = new int[n];
    for(int i=0; i<n; i++) a[i] = in.nextInt();
    memo = new double[n+1][n+1][2*n+1];
    for(double[][] x : memo)
      for(double[] y : x)
        Arrays.fill(y,-1.);
    out.printf("%.10f%n",go(0,0,k));
    out.close();
  }
  static double go(int pos, int won, int space) {
    if(space > n) space = n;
    if(pos == n) {
      if(won < l) return 0;
      if(space < 0) return 0;
      return 1;
    }
    if(memo[pos][won][space+n] != -1)
      return memo[pos][won][space+n];
    
    double ans = (1-p[pos])*go(pos+1,won,space);
    if(a[pos] < 0) {
      ans += p[pos]*go(pos+1,won+1,space-1);
    }
    else {
      ans += p[pos]*go(pos+1,won+1,space+a[pos]);
    }
    return memo[pos][won][space+n] = ans;
  }
  static class FastIO extends PrintWriter {
    BufferedReader br;
    StringTokenizer st;
    
    public FastIO() {
      this(System.in,System.out);
    }
    public FastIO(InputStream in, OutputStream out) {
      super(new BufferedWriter(new OutputStreamWriter(out)));
      br = new BufferedReader(new InputStreamReader(in));
      scanLine();
    }
    public void scanLine() {
      try {
        st = new StringTokenizer(br.readLine().trim());
      } catch(Exception e) {
        throw new RuntimeException(e.getMessage());
      }
    }
    public int numTokens() {
      if(!st.hasMoreTokens()) {
        scanLine();
        return numTokens();
      }
      return st.countTokens();
    }
    public String next() {
      if(!st.hasMoreTokens()) {
        scanLine();
        return next();
      }
      return st.nextToken();
    }
    public double nextDouble() {
      return Double.parseDouble(next());
    }
    public long nextLong() {
      return Long.parseLong(next());
    }
    public int nextInt() {
      return Integer.parseInt(next());
    }
  }
}
