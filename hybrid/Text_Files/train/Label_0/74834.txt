from math import nan, isnan

def readline():
	return map(int, input().split())

def is_good(x, y, z):
	return (y - x) * (y - z) > 0

def solve():
	n = int(input())
	ans = n + (n - 1)
	(p, q, r) = (nan, nan, nan)
	for k in readline():
		if is_good(q, r, k):
			ans += 1
			if is_good(p, q, r) and is_good(p, q, k) and is_good(p, r, k):
				ans += 1
		(p, q, r) = (q, r, k)
	print(ans)

def main():
	t = int(input())
	for __ in range(t):
		solve()
main()
