from collections import Counter
import operator
for t in range(int(input())):
	(n, k) = [*map(int, input().split())]
	cnt = Counter([e % k for e in [*map(int, input().split())]])
	(mod, times) = (0, 0)
	for (key, v) in cnt.items():
		if key != 0:
			if v > times:
				mod = key
				times = v
			elif v == times:
				mod = min(mod, key)
	if mod == 0:
		print(0)
	else:
		print((times - 1) * k + (k - mod + 1))
