from copy import deepcopy
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def main(t):
	(n, k) = map(int, input().split())
	group = [-1 for i in range(2 * n)]
	groups = [[] for i in range(n)]
	for i in range(k):
		(u, v) = map(int, input().split())
		group[u - 1] = i
		group[v - 1] = i
	ans = 0
	effect = 0
	for start in range(2 * n):
		if group[start] >= 0:
			continue
		if effect > 0:
			continue
		effect += 1
		copygroup = group[:]
		copygroups = deepcopy(groups)
		groupnum = k
		index = start
		while groupnum < n:
			if copygroup[index % (2 * n)] >= 0:
				index += 1
				continue
			copygroup[index % (2 * n)] = groupnum
			groupnum += 1
			index += 1
		groupnum = k
		while groupnum < n:
			if copygroup[index % (2 * n)] >= 0:
				index += 1
				continue
			copygroup[index % (2 * n)] = groupnum
			groupnum += 1
			index += 1
		temp = 0
		nums = []
		for i in range(2 * n):
			flag = True
			for j in range(len(nums)):
				if nums[j] == copygroup[i]:
					temp += len(nums) - j - 1
					nums.pop(j)
					flag = False
					break
			if flag:
				nums.append(copygroup[i])
		ans = max(ans, temp)
	if n == k:
		nums = []
		for i in range(2 * n):
			flag = True
			for j in range(len(nums)):
				if nums[j] == group[i]:
					ans += len(nums) - j - 1
					nums.pop(j)
					flag = False
					break
			if flag:
				nums.append(group[i])
	print(ans)
T = int(input())
t = 1
while t <= T:
	main(t)
	t += 1
