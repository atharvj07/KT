def pairProgramming(k, n, m, a, b):
	i = 0
	j = 0
	l = []
	while i < n or j < m:
		while i < n and (k >= a[i] or a[i] == 0):
			l.append(a[i])
			if a[i] == 0:
				k += 1
			i += 1
		if j < m and k < b[j]:
			print('-1')
			return
		while j < m and (k >= b[j] or b[j] == 0):
			l.append(b[j])
			if b[j] == 0:
				k += 1
			j += 1
		if i < n and k < a[i]:
			print('-1')
			return
	print(*l)
for _ in range(int(input())):
	input()
	(k, n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	pairProgramming(k, n, m, a, b)
