s = str(input())
alpha = [0] * 26
patt = [[0 for i in range(26)] for j in range(26)]
for i in range(len(s)):
	temp = ord(s[i]) - ord('a')
	for j in range(26):
		patt[j][temp] += alpha[j]
	alpha[temp] += 1
ans = 0
for i in alpha:
	ans = max(i, ans)
for i in range(26):
	for k in range(26):
		ans = max(ans, patt[i][k])
print(ans)
