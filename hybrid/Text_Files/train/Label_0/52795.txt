import java.util.Scanner;
import java.io.OutputStream;
import java.io.IOException;
import java.util.Arrays;
import java.io.PrintWriter;
import java.io.InputStream;

public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		Scanner in = new Scanner(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		TaskB solver = new TaskB();
		solver.solve(1, in, out);
		out.close();
	}
}

class TaskB
{
	final int oo = 2147483647;
	int n;
	String[] matrix = new String[25];
	int[][] mask = new int[45][30];
	int[][] f = new int[45][1100005];

	private int w(int c)
	{
		switch (c) {
			case 0:
				return 1;
			case 1:
				return -1;
			default:
				return 0;
		}
	}
	private int get(int x, int m, int c)
	{
		if (x == n + n - 1)
			return w(c);
		if (f[x][m] != -oo)
			return f[x][m] + w(c);
		if ((x & 1) == 1)
			f[x][m] = oo;
		for (int i = 0; i < 26; ++i) {
			int next = mask[x + 1][i] & (m | m << 1);
			if (next == 0)
				continue;
			if ((x & 1) == 0)
				f[x][m] = Math.max(f[x][m], get(x + 1, next, i));
			else
				f[x][m] = Math.min(f[x][m], get(x + 1, next, i));
		}
		return f[x][m] + w(c);
	}

	public void solve(int testNumber, Scanner in, PrintWriter out)
	{
		n = in.nextInt();
		for (int i = 0; i < n; ++i)
			matrix[i] = in.next();
		for (int i = 1; i < n + n; ++i) {
			Arrays.fill(mask[i], 0);
			Arrays.fill(f[i], -oo);
			for (int j = 0; j < 26; ++j)
				for (int x = 0; x < n; ++x) {
					int y = i - x - 1;
					if (y < 0 || y >= n)
						continue;
					if (matrix[x].charAt(y) == 'a' + j)
						mask[i][j] |= 1 << x;
				}
		}
		int res = get(1, 1, matrix[0].charAt(0) - 'a');
		out.println(res != 0 ? res > 0 ? "FIRST" : "SECOND" : "DRAW");
	}
}
