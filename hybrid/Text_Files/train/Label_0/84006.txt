(a, b) = map(int, input().split())
s = str(input())
sl = len(s)
x = [0 for i in range(101)]
y = [0 for i in range(101)]
for i in range(sl):
	x[i + 1] = x[i]
	y[i + 1] = y[i]
	if s[i] == 'U':
		y[i + 1] += 1
	elif s[i] == 'D':
		y[i + 1] -= 1
	elif s[i] == 'L':
		x[i + 1] -= 1
	else:
		x[i + 1] += 1
for i in range(0, sl):
	if x[sl] == 0:
		if a == x[i]:
			if y[sl] == 0 and b == y[i] or (y[sl] != 0 and (b - y[i]) % y[sl] == 0 and ((b - y[i]) // y[sl] >= 0)):
				print('Yes')
				exit()
	elif (a - x[i]) % x[sl] == 0 and (a - x[i]) // x[sl] >= 0 and (b == y[i] + (a - x[i]) // x[sl] * y[sl]):
		print('Yes')
		exit()
print('No')
