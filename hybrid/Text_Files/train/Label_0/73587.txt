board = []
for i in range(8):
	board.append(input())
minwdist = 999
minbdist = 999
for i in range(8):
	seenb = False
	currminwdist = None
	currminbdist = None
	for j in range(8):
		v = board[j][i]
		if v == 'B':
			seenb = True
		if seenb and v == 'W':
			currminbdist = None
		elif v == 'W':
			if not currminwdist:
				currminwdist = j
		if v == 'B':
			currminbdist = 8 - j - 1
	if currminwdist:
		minwdist = min(minwdist, currminwdist)
	if currminbdist:
		minbdist = min(minbdist, currminbdist)
if minwdist <= minbdist:
	print('A')
else:
	print('B')
