#include <bits/stdc++.h>
using namespace std;
long long max(long long a, long long b) { return (a > b) ? a : b; }
long long min(long long a, long long b) { return (a < b) ? a : b; }
long long mm(vector<long long> a, vector<long long> b) {
  if (a[1] < b[1]) return true;
  if (a[1] == b[1] && a[0] < b[0]) return true;
  return false;
}
long long gcd(long long a, long long b) {
  while (1) {
    if (a == 0) return b;
    b %= a;
    if (b == 0) return a;
    a %= b;
  }
}
vector<long long> p;
void kk() {
  int N = 1e7 + 5;
  p.resize(N, -1);
  for (int i = 2; i * i <= N; i++) {
    if (p[i] == -1) {
      for (int j = i * i; j <= N; j += i) {
        if (p[j] == -1) {
          p[j] = i;
        }
      }
    }
  }
}
int main() {
  long long n, m, i, j, t, k;
  t = 1;
  kk();
  cin >> t;
  vector<long long> a(t, -1), b(t, -1);
  for (long long ii = 0; ii < t; ii++) {
    cin >> n;
    m = n;
    k = p[n];
    if (p[n] == -1) continue;
    while (m % k == 0) m /= k;
    if (m > 1 && k > 1 && gcd(m + k, n) == 1) {
      a[ii] = m;
      b[ii] = k;
    }
  }
  for (i = 0; i < t; i++) cout << a[i] << " ";
  cout << endl;
  for (i = 0; i < t; i++) cout << b[i] << " ";
  return 0;
}
