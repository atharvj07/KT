#include <bits/stdc++.h>
using namespace std;
const int maxn = 100000 + 10;
int a[maxn], b[maxn];
int n;
inline long long dfs(long long t) {
  long long n = t;
  int sum = 0;
  while (n) {
    sum += n % 10;
    n /= 10;
  }
  if (sum == 10)
    return 1;
  else
    return 0;
}
int main() {
  scanf("%d", &n);
  int l = 0;
  for (int i = 1; i < 10800101; i++) {
    if (dfs(i)) b[++l] = i;
  }
  printf("%d\n", b[n]);
  return 0;
}
