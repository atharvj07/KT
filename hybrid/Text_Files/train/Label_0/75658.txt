f = [1] * 1001
fi = [1] * 1001
a = 1
m = 10 ** 9 + 7
for i in range(1, 1001):
	a *= i
	a %= m
	f[i] = a
b = pow(f[1000], m - 2, m)
fi[1000] = b
for i in range(999, 0, -1):
	fi[i] = b * (i + 1) % m
	b = fi[i]
for t in range(int(input())):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	l.sort(reverse=True)
	l1 = [0] * (n + 1)
	for i in l:
		l1[i] += 1
	a = l.index(l[k - 1])
	a = k - a
	print(f[l1[l[k - 1]]] * fi[l1[l[k - 1]] - a] * fi[a] % m)
