(n, k, M, D) = map(int, input().split())
ans = 0
for d in range(1, D + 1):
	l = 1
	r = k
	while l != r:
		mid = (l + r) // 2
		mxx = n // ((d - 1) * k + mid)
		mnx = n // ((d - 1) * k + mid + 1) + 1
		if mnx > M:
			l = mid + 1
		else:
			r = mid
	mxx = n // ((d - 1) * k + l)
	mnx = n // ((d - 1) * k + l + 1) + 1
	if mnx > M:
		continue
	mxx = min(mxx, M)
	ans = max(ans, mxx * d)
print(ans)
