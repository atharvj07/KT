import java.util.Scanner;

 * Created by Mohamed Essam on 4/9/2018.
 */
public class solve {
    public static void main(String[] args) throws Exception {
        Scanner in = new Scanner(System.in);
        int n=in.nextInt();
        int A=in.nextInt();
        int B=in.nextInt();
        int C=in.nextInt();
        int T=in.nextInt();
        int []ti=new int[n];
        for (int i = 0; i < n; i++) {
            ti[i]=in.nextInt();
            if (1 > ti[i] || ti[i] > T)
                throw new Exception("Wrong Input ");
        }
        int result=0;
       for(int t:ti){
            if (B<C)
                result+=A+(T-t)*(C-B);
            else
                result=A*n;
        }
        System.out.println(result);
    }
}
