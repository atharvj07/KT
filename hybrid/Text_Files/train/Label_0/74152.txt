n = int(input())
M = list(map(float, input().split()))
m = list(map(float, input().split()))
a = []
sa = [0]
b = []
sb = [0]
for i in range(0, n):
	p = sa[i] - sb[i] - M[i] - m[i]
	q = M[i] - sa[i] * (M[i] + m[i])
	d = p * p - 4 * q
	if d < 0:
		d = -d
	a.append((-p + d ** 0.5) / 2)
	b.append(M[i] + m[i] - a[i])
	sa.append(sa[i] + a[i])
	sb.append(sb[i] + b[i])
for i in range(0, n - 1):
	print(a[i], ' ', end='')
print(a[-1])
for i in range(0, n - 1):
	print(b[i], ' ', end='')
print(b[-1])
