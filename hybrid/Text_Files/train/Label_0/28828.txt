int main(){
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);

  int q; cin >> q;
  bitset<64> bit(0uLL);
  for (int i = 0; i < q; ++i) {
    int op; cin >> op;
    if (op == 0) {
      int pos; cin >> pos;
      cout << (int) bit.test(pos) << '\n';
    } else if (op == 1) {
      int pos; cin >> pos;
      bit.set(pos);
    } else if (op == 2) {
      int pos; cin >> pos;
      bit.reset(pos);
    } else if (op == 3) {
      int pos; cin >> pos;
      bit.flip(pos);
    } else if (op == 4) {
      cout << (int) bit.all() << '\n';
    } else if (op == 5) {
      cout << (int) bit.any() << '\n';
    } else if (op == 6) {
      cout << (int) bit.none() << '\n';
    } else if (op == 7) {
      ull x = bit.to_ullong();
      cout << __builtin_popcountll(x) << '\n';
      //cout << bit.count() << '\n';
    } else if (op == 8) {
      ull x = bit.to_ullong();
      cout << x << '\n';
    }
  }

  return 0;
}
