import java.util.Scanner;
/**
 *
 * @author Samyak Kothari
 */
public class Triangular {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n= sc.nextInt();
        int flag = 0;
        for(int i=0;i<=n;i++){
            if((i*(i+1))/2 == n){
                System.out.println("YES");
                flag=1;
                break;
                
            }
            
        }
        if(flag!=1){
            System.out.println("NO");
        }
    }
}
