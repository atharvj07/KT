#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

string s[1010];

int num(string s, int idx){
	int ret = 0;
	for(int i = idx; i < s.size(); i++){
		if(isdigit(s[i])){
			ret = ret * 10 + (s[i] - '0');
		}
		else break;
	}
	return ret;
}

bool f(string s, string t){
	for(int i = 0; i < min(s.size(), t.size()); i++){
		if(isdigit(s[i]) && isdigit(t[i])){
			int n1 = num(s, i), n2 = num(t, i);
			if(n1 != n2) return n1 <= n2;
		}
		else if(s[i] != t[i]){
			return s[i] <= t[i];
		}
	}
	return s.size() <= t.size();
}

int main(){
	cin.tie(0);
	ios::sync_with_stdio(false);
#ifdef LOCAL
	std::ifstream in("in");
	std::cin.rdbuf(in.rdbuf());
#endif

	int N;
	cin >> N;
	N++;
	for(int i = 0; i < N; i++){
		cin >> s[i];
	}

	for(int i = 1; i < N; i++){
		if(f(s[0], s[i])) cout << "+" << endl;
		else cout << "-" << endl;
	}
}
