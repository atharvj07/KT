def get_key(d, value):
	for (k, v) in d.items():
		if v == value:
			return k
n = int(input())
s = input()
d = {0: 'vaporeon', 1: 'jolteon', 2: 'flareon', 3: 'espeon', 4: 'umbreon', 5: 'leafeon', 6: 'glaceon', 7: 'sylveon'}
d1 = {}
for i in d:
	if len(d[i]) != len(s):
		continue
	else:
		q = d[i]
		k = 0
		for j in range(0, n):
			if q[j] == s[j]:
				k += 1
		d1[d[i]] = k
p = max(d1.values())
print(get_key(d1, p))
