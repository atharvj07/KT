for i in range(int(input())):
	n = int(input())
	a = [10000000000.0] + list(map(int, input().split()))
	sp = [1000000000.0] * (n + 1)
	for j in range(1, n + 1):
		if a[j] != a[j - 1]:
			if sp[a[j]] == 1000000000.0:
				sp[a[j]] = 1
			else:
				sp[a[j]] += 1
	c = min(sp)
	if c == 1000000000.0:
		print(0)
	else:
		for j in range(n):
			if abs(sp[j + 1] - c) == 1 and a[1] == j + 1 and (a[-1] == j + 1) or (abs(sp[j + 1] - c) == 0 and (a[1] == j + 1 or a[-1] == j + 1)):
				if abs(sp[j + 1] - c) == 0 and (a[1] == j + 1 and a[-1] == j + 1):
					c -= 2
				else:
					c -= 1
		print(c + 1)
