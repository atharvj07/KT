#include <bits/stdc++.h>
using namespace std;
int main() {
  string a;
  int x1, x2, y1, y2, n;
  cin >> x1 >> y1 >> x2 >> y2 >> n >> a;
  vector<int> x(n + 1), y(n + 1);
  for (int i = 1; i <= n; i++) {
    x[i] = x[i - 1];
    y[i] = y[i - 1];
    if (a[i - 1] == 'L') x[i]--;
    if (a[i - 1] == 'R') x[i]++;
    if (a[i - 1] == 'U') y[i]++;
    if (a[i - 1] == 'D') y[i]--;
  }
  long long l = 0, r = 1e18, ans = -1;
  while (l <= r) {
    long long mid = (l + r) >> 1;
    long long ny = y1 + mid / n * y[n] + y[mid % n];
    long long nx = x1 + mid / n * x[n] + x[mid % n];
    if (abs(ny - y2) + abs(nx - x2) <= mid) {
      ans = mid;
      r = mid - 1;
    } else {
      l = mid + 1;
    }
  }
  cout << ans << endl;
  return 0;
}
