#include <bits/stdc++.h>
using namespace std;


#define MOD 1000000007

int a[1010];
int b[1010];

int main(){
  int n, m;
  cin >> n >> m;

  for(int i = 0;i < n;i++) cin >> a[i+1];

  for(int j = 0;j < m;j++) cin >> b[j+1];

  a[0] = 0;
  b[0] = 0;
  sort(a, a+n+1);
  sort(b, b+m+1);


  for(int i = 0;i < n;i++){
    if(a[i] == a[i+1]){
      cout << 0 << endl;
      return 0;
    }
  }

  for(int i = 0;i < m;i++){
    if(b[i] == b[i+1]){
      cout << 0 << endl;
      return 0;
    }
  }

  long long ans = 1;
  long long aki = 0;
  for(int i = n*m;i >= 1;i--){
    long long np = lower_bound(a,a+n+1,i) - a;
    long long mp = lower_bound(b,b+m+1,i) - b;

    if(a[np] == i && b[mp] == i){
      aki += n-np + m-mp;
    }else if(a[np] == i){
      ans = (ans * (m-mp+1)) % MOD;
      aki += m-mp;
    }else if(b[mp] == i){
      ans = (ans * (n-np+1)) % MOD;
      aki += n-np;
    }else{
      ans = (ans * (aki)) % MOD;
      aki--;
    }
  }

  cout << ans << endl;

  return 0;
}
