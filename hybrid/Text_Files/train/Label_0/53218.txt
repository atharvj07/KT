import java.util.Arrays;
import java.util.Scanner;

// http://codeforces.com/contest/284/problem/E
public class CoinTroubles {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int q = s.nextInt();
        long t = s.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = s.nextInt();
        }
        int[] next = new int[n];
        int[] prev = new int[n];
        Arrays.fill(next, -1);
        Arrays.fill(prev, -1);
        for (int i = 0; i < q; i++) {
            int b = s.nextInt() - 1;
            int c = s.nextInt() - 1;
            next[c] = b;
            prev[b] = c;
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (next[i] == -1) {
                count++;
                int j = i;
                while (prev[j] != -1) {
                    t -= a[j];
                    count++;
                    a[prev[j]] += a[j];
                    j = prev[j];
                }
            }
        }
        if (count != n || t < 0) {
            System.out.println(0);
            return;
        }

        final int MD = 1000000007;
        int[] dp = new int[(int) t + 1];
        dp[0] = 1;
        for (int i = 0; i < n; i++) {
            for (int c = 0; c + a[i] <= t; c++) {
                dp[c + a[i]] = (dp[c + a[i]] + dp[c]) % MD;
            }
        }
        System.out.println(dp[(int) t]);
    }

}
