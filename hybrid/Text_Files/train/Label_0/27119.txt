#include <bits/stdc++.h>
using namespace std;
vector<int> v[1000001], vr[1000001], adj[1000001];
stack<int> st;
int vis[1000001], comp[1000001], id, size[1000001], u[1000001], bad[1000001];
vector<pair<int, int> > edges;
void dfs1(int cur) {
  vis[cur] = 1;
  for (int i = 0; i < v[cur].size(); i++) {
    int x = v[cur][i];
    if (vis[x]) continue;
    dfs1(x);
  }
  st.push(cur);
}
void dfs2(int cur) {
  vis[cur] = 1;
  comp[cur] = id;
  size[id]++;
  for (int i = 0; i < vr[cur].size(); i++) {
    int x = vr[cur][i];
    if (vis[x]) continue;
    dfs2(x);
  }
}
int main() {
  ios::sync_with_stdio(false);
  ;
  cin.tie(0);
  ;
  int i, j, n, m, h, x, y, mini, mini_id, cur;
  cin >> n >> m >> h;
  for (i = 1; i <= n; i++) cin >> u[i];
  for (i = 1; i <= m; i++) {
    cin >> x >> y;
    if ((u[x] + 1) % h == u[y]) {
      v[x].push_back(y);
      vr[y].push_back(x);
      edges.push_back(make_pair(x, y));
    }
    if ((u[y] + 1) % h == u[x]) {
      v[y].push_back(x);
      vr[x].push_back(y);
      edges.push_back(make_pair(y, x));
    }
  }
  for (i = 1; i <= n; i++) {
    if (vis[i]) continue;
    dfs1(i);
  }
  id = 0;
  memset(vis, 0, sizeof vis);
  while (st.size()) {
    cur = st.top();
    st.pop();
    if (vis[cur]) continue;
    id++;
    dfs2(cur);
  }
  for (auto it : edges) {
    x = it.first;
    y = it.second;
    x = comp[x];
    y = comp[y];
    if (x != y) bad[x] = 1;
  }
  mini = 10000000;
  mini_id = 0;
  for (i = 1; i <= id; i++) {
    if (bad[i]) continue;
    if (size[i] < mini) {
      mini = size[i];
      mini_id = i;
    }
  }
  cout << mini << "\n";
  for (i = 1; i <= n; i++) {
    if (comp[i] == mini_id) cout << i << " ";
  }
  return 0;
}
