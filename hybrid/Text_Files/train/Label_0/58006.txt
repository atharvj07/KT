import java.io.BufferedReader;
import java.io.InputStreamReader;

public class B {
	static class N {
		int max, min;

		N(int max, int min) {
			this.max = max;
			this.min = min;
		}
	}

	public static void main(String[] args) throws Exception {
		BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
		String l[] = bf.readLine().split(" ");

		int n = Integer.parseInt(l[0]);
		int m = Integer.parseInt(l[1]);
		StringBuffer sb = new StringBuffer();
		char[][] arr = new char[n][];
		for (int i = 0; i < arr.length; i++) {
			arr[i] = bf.readLine().toCharArray();
		}
		l = bf.readLine().split(" ");
		boolean isD[] = new boolean[n];
		int ind, len = -1;
		for (int i = 0; i < m; i++) {
			ind = Integer.parseInt(l[i]) - 1;
			isD[ind] = true;
			if (len == -1) {
				len = arr[ind].length;
			} else if (len != arr[ind].length) {
				System.out.println("No");
				return;
			}
		}
		for (int i = 0; i < len; i++) {
			char c = '0';
			for (int j = 0; j < n; j++) {
				if (isD[j]) {
					if (c == '0') {
						c = arr[j][i];
					} else if (c != arr[j][i]) {
						c = '?';
						break;
					}
				}
			}
			sb.append(c);
		}
		char[] patt = sb.toString().toCharArray();
		for (int i = 0; i < n; i++) {
			if (!isD[i] && arr[i].length == len && matches(arr[i], patt)) {
				System.out.println("No");
				return;
			}
		}
		System.out.println("Yes");
		System.out.println(sb);
	}

	private static boolean matches(char[] cs, char[] patt) {
		for (int i = 0; i < patt.length; i++) {
			if (patt[i] != '?' && patt[i] != cs[i]) {
				return false;
			}
		}
		return true;
	}

}
