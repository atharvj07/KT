const long double pi = acos(-1);
const int MOD = 1e9 + 7;
set<int> caras;
int n, k;
int gcd(int a, int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int main() {
  scanf("%d %d", &n, &k);
  int vai = k;
  for (int i = 0; i < n; i++) {
    int x;
    scanf("%d", &x);
    vai = gcd(vai, x);
  }
  printf("%d\n", k / vai);
  for (int i = 0; i < k; i += vai) printf("%d\n", i);
}
