line1 = input()
num_list1 = [int(l) for l in line1.split()]
n = num_list1[0]
k = num_list1[1]
line2 = input()
num_list2 = [int(l) for l in line2.split()]
dict = {}
res = [1] * n
count = []
num_ = []
for key in num_list2:
	dict[key] = dict.get(key, 0) + 1
for i in dict:
	if dict[i] == 1:
		num_.append(i)
	count.append(dict[i])
if max(count) > k:
	print('NO')
else:
	print('YES')
	temp = 1
	for i in dict:
		for j in range(dict[i]):
			res[num_list2.index(i)] = temp % k
			num_list2[num_list2.index(i)] = -1
			temp += 1
	for j in range(n):
		if num_list2[j] in num_:
			res[j] = temp % k
			temp += 1
	for r in res[:-1]:
		if r == 0:
			r = k
		print(r, end=' ')
	if res[-1] == 0:
		res[-1] = k
	print(res[-1])
