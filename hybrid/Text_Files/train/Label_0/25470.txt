#include <bits/stdc++.h>
using namespace std;
int n;
bool h[60] = {0};
bool w[60] = {0};
int main() {
  scanf("%d", &n);
  bool flag = 0;
  for (int i = 0; i < n * n; ++i) {
    int tmp1, tmp2;
    scanf("%d %d", &tmp1, &tmp2);
    if (!h[tmp1] && !w[tmp2]) {
      if (!flag)
        printf("%d", i + 1);
      else
        printf(" %d", i + 1);
      flag = 1;
      h[tmp1] = true;
      w[tmp2] = true;
    }
  }
  printf("\n");
  return 0;
}
