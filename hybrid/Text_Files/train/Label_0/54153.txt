import java.util.Arrays;
import java.util.HashSet;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;


public class CF_787C {

	int n;
	int k1, k2;
	HashSet<Integer> s1 = new HashSet<Integer>();
	HashSet<Integer> s2 = new HashSet<Integer>();
	int res[][];
	int cnt[][];
	
	Queue<State> queue = new LinkedList<>();

	private class State {
		public State(int player, int pos) {
			this.player = player;
			this.pos = pos;
		}
		public int player;
		public int pos;
	}
	
	public void doExtend(State state, int s) {
		int prePlayer = 1 - state.player;
		int prePos = state.pos - s;
		if (prePos < 0) {
			prePos += n;
		}
		if (res[prePlayer][prePos] != 0) {
			return;
		}
		if (res[state.player][state.pos] == -1) {
			res[prePlayer][prePos] = 1;
			queue.add(new State(prePlayer, prePos));
		} else if (res[state.player][state.pos] == 1 && --cnt[prePlayer][prePos] == 0) {
			res[prePlayer][prePos] = -1;
			queue.add(new State(prePlayer, prePos));
		}
	}
	
	public void solve() {
		Scanner in = new Scanner(System.in);
		n = in.nextInt();
		res = new int[2][n];
		cnt = new int[2][n];
		k1 = in.nextInt();
		for (int i = 0; i < k1; i++) {
			s1.add(in.nextInt());
		}
		k2 = in.nextInt();
		for (int i = 0; i < k2; i++) {
			s2.add(in.nextInt());
		}
		for (int i = 0; i < n; i++) {
			cnt[0][i] = k1;
			cnt[1][i] = k2;
		}
		
		Arrays.fill(res[0], 0);
		Arrays.fill(res[1], 0);
		res[0][0] = -1;
		res[1][0] = -1;
		queue.add(new State(0, 0));
		queue.add(new State(1, 0));
		while (!queue.isEmpty()) {
			State state = queue.poll();
			if (state.player == 0) {
				for (int s : s2) {
					doExtend(state, s);
				}
			} else {
				for (int s : s1) {
					doExtend(state, s);
				}
			}
		}
		for (int i = 0; i < 2; i++) {
			for (int j = 1; j < n; j++) {
				if (res[i][j] == -1) {
					System.out.print("Lose");
				} else if (res[i][j] == 1) {
					System.out.print("Win");
				} else {
					System.out.print("Loop");
				}
				System.out.print(" ");
			}
			System.out.println();
		}
	}
	public static void main(String[] args) {
		new CF_787C().solve();
	}
}
