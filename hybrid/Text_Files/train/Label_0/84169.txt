def solve(s, n):
	idx = -1
	for i in range(n):
		if s[i] == 'a':
			idx = i
			break
	if idx == -1:
		print('NO')
		return
	L = idx
	R = idx
	ch = 'b'
	while L != 0 or R != n - 1:
		if L > 0 and s[L - 1] == ch:
			L -= 1
		elif R < n - 1 and s[R + 1] == ch:
			R += 1
		else:
			break
		ch = chr(ord(ch) + 1)
	if L == 0 and R == n - 1:
		print('YES')
	else:
		print('NO')
for _ in range(int(input())):
	s = input()
	n = len(s)
	solve(s, n)
