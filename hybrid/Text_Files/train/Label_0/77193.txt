def makePair(z):

def memo(func):
	cache = {}

	def wrap(*args):
		if args not in cache:
			cache[args] = func(*args)
		return cache[args]
	return wrap

@memo
def comb(n, k):
	if k == 0:
		return 1
	if n == k:
		return 1
	return comb(n - 1, k - 1) + comb(n - 1, k)

def rotate():
	ex = 0
	ey = 0
	for x in range(2, 4):
		for y in range(1, N + 1):
			if zz[x][y] == 0:
				ex = x

			assert x == 3

		(tx1, ty1) = nextcircle(tx, ty)
