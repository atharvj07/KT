/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author Atharva Nagarkar
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        JoltyScanner in = new JoltyScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        ProblemCFoolsAndRoads solver = new ProblemCFoolsAndRoads();
        solver.solve(1, in, out);
        out.close();
    }

    static class ProblemCFoolsAndRoads {
        int n;
        int logn;
        int k;
        ArrayList<Edge>[] vs;
        int[][] p;
        int[] h;
        int[] delta;
        int[] edge;

        void bfs() {
            ArrayDeque<Integer> q = new ArrayDeque<>();
            boolean[] visited = new boolean[n];
            q.addLast(0);
            visited[0] = true;
            while (!q.isEmpty()) {
                int cur = q.pollFirst();
                for (Edge e : vs[cur]) {
                    int i = e.v;
                    if (visited[i]) continue;
                    p[0][i] = cur;
                    h[i] = h[cur] + 1;
                    visited[i] = true;
                    q.addLast(i);
                }
            }
        }

        void buildLift() {
            for (int log = 0; log < logn - 1; ++log) {
                for (int i = 0; i < n; ++i) {
                    if (p[log][i] != -1) {
                        p[log + 1][i] = p[log][p[log][i]];
                    }
                }
            }
        }

        int lca(int u, int v) {
            if (h[u] < h[v]) {
                int temp = u;
                u = v;
                v = temp;
            }
            for (int i = logn - 1; i >= 0; --i)
                if (p[i][u] != -1 && h[p[i][u]] >= h[v])
                    u = p[i][u];
            if (u == v) return u;
            for (int i = logn - 1; i >= 0; --i)
                if (p[i][u] != p[i][v]) {
                    u = p[i][u];
                    v = p[i][v];
                }
            return p[0][u];
        }

        int dfs(int node, int eid) {
            int ans = delta[node];
            for (Edge e : vs[node]) {
                int i = e.v;
                if (i == p[0][node]) continue;
                ans += dfs(i, e.id);
            }
            if (eid != -1) edge[eid] = ans;
            return ans;
        }

        public void solve(int testNumber, JoltyScanner in, PrintWriter out) {
            n = in.nextInt();
            logn = Integer.numberOfTrailingZeros(Integer.highestOneBit(n)) + 1;
            vs = new ArrayList[n];
            edge = new int[n - 1];
            p = new int[logn][n];
            h = new int[n];
            delta = new int[n];
            for (int[] a : p) Arrays.fill(a, -1);
            for (int i = 0; i < n; ++i) vs[i] = new ArrayList<>();
            for (int i = 0; i < n - 1; ++i) {
                int u = in.nextInt() - 1;
                int v = in.nextInt() - 1;
                vs[u].add(new Edge(i, v));
                vs[v].add(new Edge(i, u));
            }
            bfs();
            buildLift();
            k = in.nextInt();
            while (k-- > 0) {
                int u = in.nextInt() - 1;
                int v = in.nextInt() - 1;
                delta[u]++;
                delta[v]++;
                delta[lca(u, v)] -= 2;
            }
            dfs(0, -1);
            for (int i = 0; i < n - 1; ++i) out.print(edge[i] + " ");
            out.println();
        }

        class Edge {
            int v;
            int id;

            Edge(int i, int u) {
                id = i;
                v = u;
            }

        }

    }

    static class JoltyScanner {
        public int BS = 1 << 16;
        public char NC = (char) 0;
        public byte[] buf = new byte[BS];
        public int bId = 0;
        public int size = 0;
        public char c = NC;
        public double num = 1;
        public BufferedInputStream in;

        public JoltyScanner(InputStream is) {
            in = new BufferedInputStream(is, BS);
        }

        public JoltyScanner(String s) throws FileNotFoundException {
            in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
        }

        public char nextChar() {
            while (bId == size) {
                try {
                    size = in.read(buf);
                } catch (Exception e) {
                    return NC;
                }
                if (size == -1) return NC;
                bId = 0;
            }
            return (char) buf[bId++];
        }

        public int nextInt() {
            return (int) nextLong();
        }

        public long nextLong() {
            num = 1;
            boolean neg = false;
            if (c == NC) c = nextChar();
            for (; (c < '0' || c > '9'); c = nextChar()) {
                if (c == '-') neg = true;
            }
            long res = 0;
            for (; c >= '0' && c <= '9'; c = nextChar()) {
                res = (res << 3) + (res << 1) + c - '0';
                num *= 10;
            }
            return neg ? -res : res;
        }

    }
}
