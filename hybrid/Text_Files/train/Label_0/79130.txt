def main():
	import sys
	input = sys.stdin.readline
	(n, m) = map(int, input().split())
	grid = [list(map(int, input().split())) for _ in range(n)]
	grd = [[0] * m for _ in range(n)]
	moves = []
	for i in range(n - 1):
		for j in range(m - 1):
			if grid[i][j] and grid[i][j + 1] and grid[i + 1][j] and grid[i + 1][j + 1]:
				moves.append([i + 1, j + 1])
				for c in range(2):
					for d in range(2):
						grd[i + c][j + d] = 1
	ok = 1
	for i in range(n):
		for j in range(m):
			if grid[i][j] ^ grd[i][j]:
				ok = 0
	if not ok:
		print(-1)
	else:
		print(len(moves))
		for move in moves:
			print(*move)
	return 0
main()
