#include <bits/stdc++.h>
using namespace std;
long long int flip = 0;
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int t;
  cin >> t;
  while (t--) {
    long long int n;
    cin >> n;
    vector<long long int> p(2 * n);
    map<long long int, long long int> hash;
    long long int a = n, b = n;
    for (long long int i = 0; i < 2 * n; i++) {
      cin >> p[i];
      hash[p[i]] = i;
    }
    vector<long long int> arr;
    long long int last = 2 * n;
    for (long long int i = 2 * n; i >= 1; i--) {
      if (hash[i] < last) {
        arr.push_back(last - hash[i]);
        last = hash[i];
      }
    }
    long long int m = arr.size();
    vector<vector<long long int>> dp(m + 1, vector<long long int>(n + 1, 0));
    for (long long int i = 0; i < m + 1; i++) {
      dp[i][0] = 1;
    }
    for (long long int i = 1; i < n + 1; i++) {
      for (long long int j = 1; j < m + 1; j++) {
        if (arr[j - 1] <= i) {
          long long int check = max(dp[j - 1][i - arr[j - 1]], dp[j - 1][i]);
          dp[j][i] = check;
        } else {
          dp[j][i] = dp[j - 1][i];
        }
      }
    }
    bool ans = false;
    for (long long int i = 0; i < m + 1; i++) {
      if (dp[i][n] == 1) {
        cout << "YES" << endl;
        ans = true;
        break;
      }
    }
    if (!ans) {
      cout << "NO" << endl;
    }
  }
}
