#include <bits/stdc++.h>
using namespace std;
int main() {
  long long x0, y0, ax, ay, bx, by, t;
  cin >> x0 >> y0 >> ax >> ay >> bx >> by;
  pair<long long, long long> p;
  cin >> p.first >> p.second >> t;
  vector<pair<long long, long long> > v;
  while (x0 <= p.first + t && y0 <= p.second + t) {
    v.push_back({x0, y0});
    x0 = ax * x0 + bx;
    y0 = ay * y0 + by;
  }
  long long ans = 0;
  for (int i = 0; i < v.size(); i++) {
    for (int j = i; j < v.size(); j++) {
      long long dis =
          abs(v[i].first - v[j].first) + abs(v[i].second - v[j].second);
      long long B = abs(p.first - v[j].first) + abs(p.second - v[j].second);
      long long A = abs(p.first - v[i].first) + abs(p.second - v[i].second);
      if (dis + min(A, B) <= t) ans = max(ans, abs(i - j) + 1ll);
    }
  }
  cout << ans << endl;
}
