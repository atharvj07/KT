#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:100000000,100000000")
using namespace std;
const int N = 1000005;
int n, m, k;
int main() {
  cin >> n >> m >> k;
  if (k + n < m)
    puts("0");
  else if (k >= m)
    puts("1");
  else {
    double ans = 1.0;
    for (int i = 1; i <= n + k + 1 || i <= n || i <= m || i <= m - k - 1; i++) {
      if (i <= n) ans = ans * i;
      if (i <= m) ans = ans * i;
      if (i <= n + k + 1) ans = ans / i;
      if (i <= m - k - 1) ans = ans / i;
    }
    cout << 1 - ans << endl;
  }
  return 0;
}
