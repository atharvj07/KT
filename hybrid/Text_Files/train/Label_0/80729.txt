def longest_zipline(w, g, h, x):
	return sqrt(g ** 2 + x ** 2) + sqrt(h ** 2 + (w - x) ** 2)

def lz_derivative(w, g, h, r, x):
	return x / sqrt(g ** 2 + x ** 2) + (x - w) / sqrt(h ** 2 + (w - x) ** 2)

def is_ascend(s):
	low = 10 ** 6
	for i in range(len(s)):
		if s[i] < low:
			low = s[i]
		if s[i] > low:
			return True
	return False
low = []
high = []
count = 0
ascends = 0
for (i, line) in enumerate(sys.stdin):

		s = [int(x) for x in line.strip().split()]
		s = s[1:]
		if not is_ascend(s):
			low.append(s[-1])
			high.insert(bisect.bisect(high, s[0]), s[0])
		else:
			ascends += 1
for l in low:
	count += len(low) - bisect.bisect_right(high, l)
