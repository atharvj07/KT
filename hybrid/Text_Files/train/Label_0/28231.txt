#include <bits/stdc++.h>
using namespace std;
const int maxn = 5e3 + 5;
const int mod = 998244353;
const int inf = 0x3f3f3f3f;
long long dp[maxn][maxn];
long long pre[maxn];
int a[maxn], pos[maxn];
int main(void) {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; i++) {
    pos[i] = 0;
    for (int j = 1; j < i; j++) {
      if (2ll * a[j] > a[i])
        break;
      else
        pos[i] = j;
    }
  }
  for (int i = 1; i <= n; i++) dp[1][i] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) pre[j] = (pre[j - 1] + dp[i - 1][j]) % mod;
    for (int j = 1; j <= n; j++) {
      dp[i][j] += dp[i - 1][j] * max(0ll, 1ll * pos[j] - i + 2) % mod;
      dp[i][j] += pre[pos[j]];
      dp[i][j] %= mod;
    }
  }
  printf("%lld\n", dp[n][n]);
}
