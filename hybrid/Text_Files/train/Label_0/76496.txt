(n, m, k) = map(int, input().split())
houses = list(map(int, input().split()))
toright = houses[m:]
toleft = list(reversed(houses[:m - 1]))

def first_lower(lst, num):
	for i in range(len(lst)):
		if lst[i] <= num and lst[i] != 0:
			return i + 1
	return -1

def min(a, b):
	if a < 0 or b < 0:
		return a if b < 0 else b
	return a if a < b else b
print(min(first_lower(toright, k), first_lower(toleft, k)) * 10)
