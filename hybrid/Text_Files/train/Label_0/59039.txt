//package Technocup_2019_QualifyingRound3;

import java.util.Arrays;
import java.util.Scanner;

public class E {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int mas[] = new int[n + 1];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            mas[i] = in.nextInt();
            sum += mas[i];
        }
        mas[n] = 1000_000;
        Arrays.sort(mas);
        int h = 1;
        for (int i = 1; i <= n; i++) {
            if (mas[i - 1] != mas[i]) h++;
        }
        int p[][] = new int[h][2];
        p[0][0] = mas[0];
        p[0][1]++;
        int uk = 0;
        for (int i = 1; i <= n; i++) {
            if (mas[i - 1] != mas[i]) {
                uk++;
                p[uk][0] = mas[i];
                p[uk][1]++;
            } else {
                p[uk][1]++;
            }
        }
        h--;
        int dp[][][] = new int[sum + 1][h][n + 1];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < p[i][1]; j++) {
                dp[p[i][0] * (j + 1)][i][j + 1] = 1;
            }
            if (i != 0) {
                for (int j = 0; j <= sum; j++) {
                    for (int k = 0; k < n; k++) {
                        dp[j][i][k] += dp[j][i - 1][k];
                        for (int l = 1; l <= p[i][1]; l++) {
                            if (j + p[i][0] * l <= sum && k + l < n + 1) {
                                dp[j + p[i][0] * l][i][k + l] += dp[j][i - 1][k];
                            }
                        }
                    }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < h; i++) {
            for (int j = 1; j <= p[i][1]; j++) {
                if (j > ans) {
                    if (dp[j * p[i][0]][h - 1][j] == 1) {
                        ans = j;
                        if (h == 2 && j == p[i][1]) ans = n;
                    }
                }
                if (j > ans) {
                    if (dp[sum - j * p[i][0]][h - 1][n - j] == 1) {
                        ans = j;
                        if (h == 2 && j == p[i][1]) ans = n;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
