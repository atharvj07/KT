#include <bits/stdc++.h>
using namespace std;
double l[200005], r[200005], sum[200005], q, x, y, res, s, a, b, p1, p2;
int u, v, n;
int main() {
  scanf("%lf%lf%lf", &s, &a, &b);
  scanf("%d", &n);
  for (int i = (1); i <= (n); ++i) {
    scanf("%lf%lf", &l[i], &r[i]);
    sum[i] = sum[i - 1] + r[i] - l[i];
  }
  scanf("%lf", &q);
  while (q--) {
    scanf("%lf%lf", &x, &y);
    p1 = (x * s - a * y) / (s - y);
    p2 = (x * s - b * y) / (s - y);
    u = lower_bound(l + 1, l + n + 1, p1) - l;
    v = upper_bound(r + 1, r + n + 1, p2) - r;
    v--;
    res = sum[v] - sum[u - 1];
    if (u - 1 > 0 && p1 <= r[u - 1]) res += r[u - 1] - p1;
    if (v + 1 <= n && p2 >= l[v + 1]) res += p2 - l[v + 1];
    cout << fixed << setprecision(6);
    cout << res / (p2 - p1) * (b - a) << "\n";
  }
  return 0;
}
