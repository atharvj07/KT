from collections import deque
import sys

def input():
	return sys.stdin.readline().rstrip()

def slv():
	(n, q) = map(int, input().split())
	a = list(map(int, input().split()))
	query = list(map(int, input().split()))
	mindex = [1 << 128] * 100
	seen = [False] * 100
	stack = []
	for (i, v) in enumerate(a):
		if seen[v] == False:
			seen[v] = True
			stack.append((v, i))
	for q in query:
		for (i, elem) in enumerate(stack):
			if elem[0] == q:
				print(elem[1] + 1)
				for j in range(i - 1, -1, -1):
					(vp, ip) = stack[j]
					(vq, iq) = stack[j + 1]
					stack[j + 1] = (vp, ip + 1)
				stack[0] = (q, 0)
				break
	return

def main():
	t = 1
	for i in range(t):

main()
