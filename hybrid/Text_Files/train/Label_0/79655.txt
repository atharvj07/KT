def find_all_indexes(input_str, search_str):
	l1 = []
	length = len(input_str)
	index = 0
	while index < length:
		i = input_str.find(search_str, index)
		if i == -1:
			return l1
		l1.append(i)
		index = i + 1
	return l1
nos = int(input())

def some(has, original):
	for i in original:
		if i not in has or original.count(i) != has.count(i):
			return False
	return True
for i in range(nos):
	hh = list(input())
	hashh = input()
	point1 = 0
	point2 = len(hh) - 1
	flag = 0
	while point2 < len(hashh):
		if some(hashh[point1:point2 + 1], hh) != False:
			flag = 5
			print('YES')
			break
		point1 += 1
		point2 += 1
	if flag != 5:
		print('NO')
