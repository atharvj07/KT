def solve(test_case):
	d = int(input().split()[1])
	a = list(map(int, input().split()))
	print(min((sum((1 - (0 <= c - p <= d) for c in a)) for p in a)))
test_case = 1
while test_case:
	solve(test_case)
	test_case -= 1
