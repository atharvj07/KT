    private static void divideAndMerge(Point[] orig, int left, int right) {
        int mid = (orig[left].x + orig[right].x) / 2;
        for (int i = left; i <= right; i++) {
            superSet.add(new Point(mid, orig[i].y));
        }
        int index = Arrays.binarySearch(orig, left, right + 1, new Point(mid, 0));
        if (index < 0) {
            index = -1 - index;
            divideAndMerge(orig, left, index - 1);
            divideAndMerge(orig, index, right);
        } else {
            if (left < index - 1) {
                divideAndMerge(orig, left, index - 1);
            }
            if (right > index + 1) {
                divideAndMerge(orig, index + 1, right);
            }
        }
    }

    private static class Point implements Comparable {

        int x, y;

        private Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        private Point() {
        }

        @Override
        public int hashCode() {
            int hash = 5;
            hash = 79 * hash + this.x;
            return hash;
        }

        @Override
        public boolean equals(Object o) {
            return x == ((Point) o).x && y == ((Point) o).y;
        }

        @Override
        public int compareTo(Object o) {
            return x - ((Point) o).x;
        }
    }
    private static HashSet<Point> superSet;
}
