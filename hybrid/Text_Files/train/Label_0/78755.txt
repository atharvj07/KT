if True:
	n = int(input())
	s = input()
	t = input()
else:
	s = 'code??????'
	t = '??????test'
	n = len(s)
X = [[i + 1, s[i]] for i in range(n)]
Y = [[i + 1, t[i]] for i in range(n)]
X = sorted(X, key=lambda x: ord(x[1]) if x[1] != '?' else 200)
Y = sorted(Y, key=lambda x: ord(x[1]) if x[1] != '?' else 200)
i = 0
j = 0
ANS = []
XX = [0] * n
YY = [0] * n
while i < n and j < n:
	if X[i][1] == '?' or Y[j][1] == '?':
		break
	if X[i][1] == Y[j][1]:
		ANS.append([X[i][0], Y[j][0]])
		XX[i] = 1
		YY[j] = 1
		i += 1
		j += 1
	elif X[i][1] < Y[j][1]:
		i += 1
	else:
		j += 1
while i < n and X[i][1] != '?':
	i += 1
while j < n and Y[j][1] != '?':
	j += 1
jj = 0
ii = 0
while len(ANS) < n and i < n:
	while YY[jj] == 1 and jj < n:
		jj += 1
	if jj >= n:
		break
	ANS.append([X[i][0], Y[jj][0]])
	XX[i] = 1
	YY[jj] = 1
	i += 1
	jj += 1
while len(ANS) < n and j < n:
	while XX[ii] == 1 and ii < n:
		ii += 1
	while YY[j] == 1 and j < n:
		j += 1
	if ii >= n:
		break
	ANS.append([X[ii][0], Y[j][0]])
	XX[ii] = 1
	YY[j] = 1
	j += 1
	ii += 1
print(len(ANS))
for ans in ANS:
	print(*ans)
