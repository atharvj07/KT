#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:1024000000,1024000000")
using namespace std;
const int INF = 0x3f3f3f3f;
const double eps = 1e-8;
int main() {
  int n, s[1005], d[1005], ans = 0;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d%d", &s[i], &d[i]);
    for (int j = 0;; j++) {
      if (s[i] + d[i] * j > ans) {
        ans = s[i] + d[i] * j;
        break;
      }
    }
  }
  printf("%d\n", ans);
  return 0;
}
