n = int(input())
m = [list(map(int, input().split())) for i in range(n)]
if n == 1:
	print(m[0][0])
elif n == 2:
	print(1, m[0][1])
else:
	ansl = [None] * n
	for i in range(n - 2):
		ansl[i + 1] = int((m[i][i + 1] * m[i + 1][i + 2] // m[i][i + 2]) ** (1 / 2))
	ansl[0] = m[0][1] // ansl[1]
	ansl[-1] = m[-2][-1] // ansl[-2]
	print(' '.join(map(str, ansl)))
