import bisect
import io, os, sys
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
n = int(input())
l = list(map(int, input().decode().split()))
l.sort()
t = 0
for i in l:
	t += i
for i in range(int(input())):
	(a, b) = input().decode().split()
	a = int(a)
	b = int(b)
	x = bisect.bisect_left(l, a)
	if x < n and l[x] == a:
		if t - l[x] >= b:
			print(0)
		else:
			print(b - (t - l[x]))
	elif x == n:
		ans = abs(l[x - 1] - a)
		if t - (a - ans) >= b:
			print(ans)
		else:
			print(b - (t - a))
	elif x == 0:
		if t - l[x] >= b:
			print(0)
		else:
			print(b - (t - l[x]))
	else:
		ans = 0
		ans1 = 0
		if t - l[x] >= b:
			ans = 0
		else:
			ans = b - (t - l[x])
		ans1 = abs(l[x - 1] - a)
		if t - (a - ans1) >= b:
			pass
		else:
			ans1 = b - (t - a)
		print(min(ans, ans1))
