#include <bits/stdc++.h>
const double PI = 3.1415926535897932384626433;
const int KL = 1e6;
const long long MOD = 1e9 + 7;
using namespace std;
long long q, n, x[KL], y[KL], z[KL], timer;
map<pair<long long, long long>, long long> mp;
map<long long, long long> pm;
vector<pair<long long, long long>> vec;
vector<long long> u;
vector<pair<long long, long long>> adj[KL], adj2[KL];
int main() {
  scanf("%lld", &n);
  timer = 1;
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &x[i]);
    scanf("%lld", &y[i]);
    scanf("%lld", &z[i]);
    if (!mp[{x[i], y[i]}]) {
      mp[{x[i], y[i]}] = timer;
      timer++;
      vec.push_back({x[i], y[i]});
    }
    long long pos = mp[{x[i], y[i]}];
    adj[pos].push_back({z[i], i});
  }
  timer = 1;
  for (auto v : vec) {
    long long pos = mp[v];
    sort(adj[pos].begin(), adj[pos].end());
    while (adj[pos].size() >= 2) {
      printf("%lld ", adj[pos].back().second);
      adj[pos].pop_back();
      printf("%lld\n", adj[pos].back().second);
      adj[pos].pop_back();
    }
    if (adj[pos].size() == 1) {
      if (!pm[v.first]) {
        pm[v.first] = timer;
        timer++;
        u.push_back(v.first);
      }
      long long pos2 = pm[v.first];
      adj2[pos2].push_back({v.second, adj[pos].back().second});
    }
  }
  vec.clear();
  for (auto v : u) {
    long long pos = pm[v];
    sort(adj2[pos].begin(), adj2[pos].end());
    while (adj2[pos].size() >= 2) {
      printf("%lld ", adj2[pos].back().second);
      adj2[pos].pop_back();
      printf("%lld\n", adj2[pos].back().second);
      adj2[pos].pop_back();
    }
    if (adj2[pos].size() == 1) {
      vec.push_back({v, adj2[pos].back().second});
    }
  }
  sort(vec.begin(), vec.end());
  if (vec.size() % 2 == 1) assert(0);
  for (int i = 1; i < vec.size(); i += 2) {
    printf("%lld %lld\n", vec[i - 1].second, vec[i].second);
  }
  return 0;
}
