import math
for t in range(int(input())):
	(x, y, a, b) = tuple(map(int, input().split()))
	m1 = max(a, b)
	m2 = min(a, b)
	if m1 == m2:
		s = math.floor(min(x, y) / m1)
	else:
		s = 0
		if x > y:
			d = math.floor(min((x - y) / (m1 - m2), y / m2, x / m1))
			s += d
			x -= d * m1
			y -= d * m2
		elif y > x:
			d = math.floor(min((y - x) / (m1 - m2), y / m1, x / m2))
			s += d
			y -= d * m1
			x -= d * m2
		d = math.floor(min(x / (m1 + m2), y / (m1 + m2)))
		s += 2 * d
		y -= d * (m1 + m2)
		x -= d * (m1 + m2)
		if x >= a and y >= b or (x >= b and y >= a):
			s += 1
	print(s)
