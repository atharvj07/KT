#include <bits/stdc++.h>
using namespace std;
int N, M, T;
map<string, bool> mp;
char tb[10];
int main() {
  int i, j, k = 0;
  scanf("%d", &N);
  for (i = 0; i < N; i++) {
    scanf("%s%s%*s", tb, tb + 2);
    bool chk = false;
    for (j = 0; j < 4; j++) {
      chk |= (mp[tb]);
      swap(tb[0], tb[1]);
      swap(tb[0], tb[2]);
      swap(tb[2], tb[3]);
    }
    if (!chk) {
      mp[tb] = true;
      k++;
    }
  }
  printf("%d\n", k);
}
