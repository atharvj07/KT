(N, K) = map(int, input().split())
A = list(map(int, input().split()))
ok = 1
ng = N + 1
while ng - ok > 1:
	c = ok + ng >> 1
	B = [1 if a >= c else -1 for a in A]
	cumB = [0]
	for b in B:
		cumB.append(cumB[-1] + b)
	mi = 0
	for (l, r) in zip(cumB, cumB[K:]):
		if mi > l:
			mi = l
		if r - mi > 0:
			break
	else:
		ng = c
		continue
	ok = c
print(ok)
