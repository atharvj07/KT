#include <iostream>
#include <stdio.h>
#define MAX 501
using namespace std;

int main(int argc, char *argv[])
{
    int n;
    scanf("%d", &n);
    double p[MAX], q[MAX];
    p[0] = -1;
    for (int i = 1; i <= n; i++)
    {
        scanf("%lf", &p[i]);
    }
    for (int i = 0; i < n + 1; i++)
    {
        scanf("%lf", &q[i]);
    }

//    const int n = 5;
//      double p[n + 1] = {-1,0.15,0.1,0.05,0.1,0.2};
//     double q[n + 1] = {0.05,0.1,0.05,0.05,0.05,0.1};

    double e[MAX][MAX] = {0};
    double w[MAX][MAX] = {0};
    for (int i = 1; i <= n + 1; i++)
    {
        e[i][i - 1] = q[i - 1];
        w[i][i - 1] = q[i - 1];
    }

    int j = 0;
    for (int l = 1; l <= n; l++)
    {
        for (int i = 1; i <= n - l + 1; i++)
        {
            j = i + l - 1;
            e[i][j] = (1 << 21);
            w[i][j] = w[i][j - 1] + p[j] + q[j];
            for (int r = i; r <= j; r++)
            {
                double t = e[i][r - 1] + e[r + 1][j] + w[i][j];
                if (t < e[i][j])
                {
                    e[i][j] = t;
                }
            }
        }
    }
    printf("%lf", e[1][n]);

    return 0;
}
