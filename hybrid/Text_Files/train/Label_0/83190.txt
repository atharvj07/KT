from functools import reduce
from operator import *
from math import *
from sys import setrecursionlimit
setrecursionlimit(10 ** 7)
(n, m, q) = map(int, input().split())
grid = [0] * n
maxCnt = [0] * n

def cntmax1(seq):
	maxcnt = 0
	cnt = 0
	for i in seq:
		if i == 1:
			cnt += 1
		else:
			cnt = 0
		maxcnt = max(cnt, maxcnt)
	return maxcnt
for i in range(n):
	grid[i] = list(map(int, input().split()))
	maxCnt[i] = cntmax1(grid[i])
for i in range(q):
	(x, y) = map(int, input().split())
	x -= 1
	y -= 1
	grid[x][y] ^= 1
	maxCnt[x] = cntmax1(grid[x])
	print(max(maxCnt))
