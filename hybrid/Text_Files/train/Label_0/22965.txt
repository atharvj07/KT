#include<iostream>
#include<vector>
#include<string>
#include<algorithm>
using namespace std;

int parse(string a) {
	int l, r;
	l = r = 0;
	for (int b = 0; b < 4; b++) {
		l *= 10;
		l += a[b] - 48;
	}
	int b = 3;
	for (; b>-1; b--) {
		if (a[b] != '0')break;
	}
	for (int c = b; c!=-1; c--) {
		r *= 10;
		r += a[c] - 48;
	}
	return r - l;
}
int main() {
	while (1) {
		string a;
		cin >> a;
		if (a == "0000")break;
		if (a[0] == a[1] && a[1] == a[2] && a[2] == a[3]) { cout << "NA" << endl; continue; }
		for (int b = 0;; b++) {
			if (a == "6174") { cout << b << endl; break; }
			sort(a.begin(), a.end());
			a = to_string(parse(a));
		}
	}
}
