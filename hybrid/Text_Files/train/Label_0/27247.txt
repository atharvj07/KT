#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  long long u, v;
  cin >> u >> v;
  long long parityReq[62];
  long long init[62];
  for (long long i = 0; i < 62; ++i) {
    if (u & (1LL << i)) {
      parityReq[i] = 1;
    } else
      parityReq[i] = 0;
    if (v & (1LL << i)) {
      init[i] = 1;
    } else
      init[i] = 0;
  }
  bool fg = 0;
  for (long long i = 0; i < 61; ++i) {
    long long temp = init[i];
    init[i] = init[i] % 2;
    init[i + 1] += temp / 2;
    if (parityReq[i] == init[i]) {
      continue;
    }
    if (i == 0) {
      fg = 1;
      break;
    }
    if (init[i] == 1) {
      if (i == 0) fg = 1;
      init[i] = 0;
      init[i - 1] += 2;
    } else {
      bool fg1 = 0;
      for (long long j = i; j < 62; ++j) {
        if (init[j] != 0) {
          fg1 = 1;
          init[j] -= 1;
          break;
        }
        init[j] = 1;
      }
      if (fg1 == 0)
        fg = 1;
      else {
        init[i - 1] += 2;
      }
    }
  }
  if (fg) {
    cout << "-1";
    return 0;
  }
  long long maxm = 0;
  for (long long i = 0; i < 62; ++i) {
    maxm = max(init[i], maxm);
  }
  vector<long long> vec;
  for (long long i = 0; i < maxm; ++i) {
    long long val = 0;
    for (long long j = 0; j < 62; ++j) {
      init[j]--;
      if (init[j] >= 0) {
        val |= (1LL << j);
      }
    }
    vec.push_back(val);
  }
  cout << maxm << "\n";
  for (long long i = 0; i < vec.size(); ++i) {
    cout << vec[i] << " ";
  }
  return 0;
}
