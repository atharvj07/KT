#include <bits/stdc++.h>
using namespace std;
inline long long getnum() {
  char c = getchar();
  long long num, sign = 1;
  for (; c < '0' || c > '9'; c = getchar())
    if (c == '-') sign = -1;
  for (num = 0; c >= '0' && c <= '9';) {
    c -= '0';
    num = num * 10 + c;
    c = getchar();
  }
  return num * sign;
}
map<long long, int> Map;
int main() {
  int n = getnum();
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    long long x = getnum();
    for (long long y = 1LL << 40LL; y > 0; y /= 2) ans += Map[y - x];
    Map[x]++;
  }
  cout << ans;
}
