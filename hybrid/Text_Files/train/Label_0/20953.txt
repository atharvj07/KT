#include <bits/stdc++.h>
using namespace std;
int n, pos, l, r;
int main() {
  cin >> n >> pos >> l >> r;
  int ans = n * n;
  if (pos < l) {
    ans = l - pos + 1;
    if (r < n) ans += (r - l + 1);
  } else if (pos > r) {
    ans = pos - r + 1;
    if (l > 1) ans += (r - l + 1);
  } else {
    int tmp = n * n;
    if (l != 1) {
      tmp = pos - l + 1;
      if (r != n) tmp += r - l + 1;
      ans = min(ans, tmp);
      if (r != n) {
        tmp = 0;
        tmp = r - pos + 1;
        tmp += r - l + 1;
        ans = min(ans, tmp);
      }
    } else if (r != n) {
      tmp = r - pos + 1;
      ans = min(ans, tmp);
    } else
      ans = 0;
  }
  cout << ans;
}
