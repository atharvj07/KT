MAX = 1 + 2 * 10 ** 5
MIN = -1
ASC = 0
DSC = 1
n = int(input())
a = list(map(int, input().split()))
asc = MIN
dsc = MAX
res = []
for i in range(n):
	if a[i] > asc and a[i] < dsc:
		if i + 1 == n:
			res.append(DSC)
		elif a[i] > a[i + 1]:
			res.append(DSC)
			dsc = a[i]
		else:
			res.append(ASC)
			asc = a[i]
	elif a[i] > asc:
		res.append(ASC)
		asc = a[i]
	elif a[i] < dsc:
		res.append(DSC)
		dsc = a[i]
	else:
		print('NO')
		return
print('YES')
print(*res)
