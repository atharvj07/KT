#include <bits/stdc++.h>
using namespace std;
struct point {
  int x, y, z;
  void operator=(const point &rhs) {
    x = rhs.x;
    y = rhs.y;
    z = rhs.z;
  }
  point operator+(const point &rhs) const {
    point ret;
    ret.x = x + rhs.x;
    ret.y = y + rhs.y;
    ret.z = z + rhs.z;
    return ret;
  }
  point operator-(const point &rhs) const {
    point ret;
    ret.x = x - rhs.x;
    ret.y = y - rhs.y;
    ret.z = z - rhs.z;
    return ret;
  }
  bool operator==(const point &rhs) const {
    return x == rhs.x && y == rhs.y && z == rhs.z;
  }
  int operator*(const point &rhs) const {
    return x * rhs.x + y * rhs.y + z * rhs.z;
  }
  long long int len() {
    return ((long long int)x) * ((long long int)x) +
           ((long long int)y) * ((long long int)y) +
           ((long long int)z) * ((long long int)z);
  }
  bool hasNextPermutation() {
    int a[3];
    a[0] = x;
    a[1] = y;
    a[2] = z;
    return next_permutation(a, a + 3);
  }
  void permutateNext() {
    int a[3];
    a[0] = x;
    a[1] = y;
    a[2] = z;
    next_permutation(a, a + 3);
    x = a[0];
    y = a[1];
    z = a[2];
  }
  void print() { printf("%d %d %d\n", x, y, z); }
};
point makePoint(int xp, int yp, int zp) {
  point ret;
  ret.x = xp;
  ret.y = yp;
  ret.z = zp;
  return ret;
}
point points[8];
bool isCube() {
  long long int minlen = -1;
  point p = points[7];
  point adj[3];
  int curAdj = 0;
  bool marcado[8];
  for (int i = 0; i < 8; i++) {
    marcado[i] = false;
    for (int j = 0; j < 8; j++) {
      if (i != j && points[i] == points[j]) return false;
    }
  }
  marcado[7] = true;
  for (int i = 0; i < 7; i++) {
    if ((p - points[i]).len() < minlen || minlen == -1)
      minlen = (p - points[i]).len();
  }
  for (int i = 0; i < 7; i++) {
    if ((p - points[i]).len() == minlen) {
      adj[curAdj] = points[i] - p;
      for (int j = 0; j < curAdj; j++)
        if (adj[j] * adj[curAdj] != 0) return false;
      marcado[i] = true;
      curAdj++;
    }
  }
  if (curAdj != 3) {
    return false;
  }
  for (int i = 0; i < 7; i++) {
    if (!marcado[i]) {
      if (points[i] == p + adj[0] + adj[1]) {
        marcado[i] = true;
        continue;
      }
      if (points[i] == p + adj[0] + adj[2]) {
        marcado[i] = true;
        continue;
      }
      if (points[i] == p + adj[1] + adj[2]) {
        marcado[i] = true;
        continue;
      }
      if (points[i] == p + adj[0] + adj[1] + adj[2]) {
        marcado[i] = true;
        continue;
      }
    }
  }
  for (int i = 0; i < 7; i++) {
    if (!marcado[i]) return false;
  }
  return true;
}
