p = lambda : list(map(int, input().split()))
n = p()[0]
A = [0] + p()
B = [0] + p()
d = 0
v = [0] * (n + 1)
p = [-9 ** 42] + [0] * n
s = []
for i in range(1, n + 1):
	while s and A[s[-1]] > A[i]:
		d = max(d, v[len(s)])
		s.pop()
	s += [i]
	v[len(s)] = d
	d = p[len(s)] = max(p[len(s) - 1], d + B[i])
print(d)
