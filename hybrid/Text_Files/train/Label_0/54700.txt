import java.io.*;
import java.util.*;

public class Main {

	public static void main(String[] args) throws IOException, InterruptedException {

		Scanner sc = new Scanner(System.in);
		PrintWriter pw = new PrintWriter(System.out);

		int n = sc.nextInt();
		int m = sc.nextInt();

		int[] reg = new int[n];
		int[] auc = new int[m];

		for (int i = 0; i < n; i++)
			reg[i] = sc.nextInt();

		for (int i = 0; i < m; i++) {
			int ind = sc.nextInt() - 1;
			auc[i] = reg[ind];
			reg[ind] = -1;
		}
		
		Arrays.sort(reg);
		Arrays.sort(auc);
		
		long score = 0;
		
		for(int i : reg) {
			if(i != -1)
				score += i;
		}
		
		for(int i = m-1; i>=0; i--) {
			if(score > auc[i])
				score *= 2;
			else
				score += auc[i];
			
		}
		

		pw.println(score);
		
		pw.flush();
		pw.close();

	}

	public static int lower(int[] arr, int x) {
		int ans = -1;
		int lo = 0;
		int hi = arr.length - 1;

		while (lo <= hi) {
			int mid = (lo + hi) / 2;

			if (arr[mid] < x) {
				lo = mid + 1;
			} else {
				if (arr[mid] == x)
					ans = mid;
				hi = mid - 1;

			}

		}

		return ans;
	}

	public static int higher(int[] arr, int x) {
		int ans = -1;
		int lo = 0;
		int hi = arr.length - 1;

		while (lo <= hi) {
			int mid = (lo + hi) / 2;

			if (arr[mid] <= x) {

				lo = mid + 1;
			} else {
				ans = arr[mid];
				hi = mid - 1;
			}

		}

		return ans;
	}

	static String perms = "";

	public static void permutation(int n, String[] elements) {

		if (n == 1) {

			for (String s : elements) {
				perms += s + " ";
			}
		} else {
			for (int i = 0; i < n - 1; i++) {
				permutation(n - 1, elements);
				if (n % 2 == 0) {
					swap(elements, i, n - 1);
				} else {
					swap(elements, 0, n - 1);
				}
			}
			permutation(n - 1, elements);
		}
	}

	private static void swap(String[] input, int a, int b) {
		String tmp = input[a];
		input[a] = input[b];
		input[b] = tmp;
	}

	static boolean isUnique(int x, int y) {

		boolean[] arr = new boolean[10];

		for (int i = 0; i < 5; i++) {

			if (arr[x % 10])
				return false;

			arr[x % 10] = true;
			x = x / 10;
		}

		for (int i = 0; i < 5; i++) {

			if (arr[y % 10])
				return false;

			arr[y % 10] = true;
			y = y / 10;
		}

		return true;
	}

	static class Scanner {
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String nextLine() throws IOException {
			return br.readLine();
		}

		public double nextDouble() throws IOException {
			String x = next();
			StringBuilder sb = new StringBuilder("0");
			double res = 0, f = 1;
			boolean dec = false, neg = false;
			int start = 0;
			if (x.charAt(0) == '-') {
				neg = true;
				start++;
			}
			for (int i = start; i < x.length(); i++)
				if (x.charAt(i) == '.') {
					res = Long.parseLong(sb.toString());
					sb = new StringBuilder("0");
					dec = true;
				} else {
					sb.append(x.charAt(i));
					if (dec)
						f *= 10;
				}
			res += Long.parseLong(sb.toString()) / f;
			return res * (neg ? -1 : 1);
		}

		public boolean ready() throws IOException {
			return br.ready();
		}
	}
}

class Dragon implements Comparable<Dragon> {

	int x;
	int y;

	public Dragon(int x, int y) {
		this.x = x;
		this.y = y;
	}

	public String toString() {
		return x + "";
	}

	@Override
	public int compareTo(Dragon o) {
		if (this.x > o.x)
			return 1;
		else if (this.x == o.x)
			return 0;
		return -1;
	}

}
