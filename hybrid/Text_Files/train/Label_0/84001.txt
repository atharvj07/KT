def checcker(x, y):
	p = 0
	for mkT in range(6):
		if x[mkT] == y[mkT]:
			p = p + 1
		if p >= 4:
			break
	return p

def viewer_finish(x, k):
	if x != 1:
		if k >= 4:
			print('0')
		elif k == 3 or k == 2:
			print('1')
		elif k == 1 or k == 0:
			print('2')
	else:
		print(6)

def checcker_main(kList, x):
	k = 0
	for fmN in range(x - 1):
		for KKT in range(fmN + 1, x):
			p = checcker(kList[fmN], kList[KKT])
			if p > k:
				k = p
			if k >= 4:
				break
		if k >= 4:
			break
	return k

def main():
	kList = []
	x = int(input())
	for i in range(x):
		kList.append(input())
	k = checcker_main(kList, x)
	viewer_finish(x, k)
main()
