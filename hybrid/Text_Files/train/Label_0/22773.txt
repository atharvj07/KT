#include <bits/stdc++.h>
using namespace std;
int main() {
  int r, c;
  cin >> r >> c;
  char s[r + 1][c + 1];
  int flag = 0;
  for (int i = 1; i <= r; i++) {
    for (int j = 1; j <= c; j++) cin >> s[i][j];
  }
  for (int i = 1; i <= r; i++) {
    for (int j = 1; j <= c; j++) {
      if ((j != c) && s[i][j] == 'S' && s[i][j + 1] == 'W') {
        flag = 1;
      } else if ((i != 1) && s[i][j] == 'S' && s[i - 1][j] == 'W') {
        flag = 1;
      } else if ((j != 1) && s[i][j] == 'S' && s[i][j - 1] == 'W') {
        flag = 1;
      } else if ((i != r) && s[i][j] == 'S' && s[i + 1][j] == 'W') {
        flag = 1;
      } else {
        if (s[i][j] == '.') {
          s[i][j] = 'D';
        }
      }
    }
  }
  if (flag == 1) {
    cout << "NO\n";
    return 0;
  } else
    cout << "YES\n";
  for (int i = 1; i <= r; i++) {
    for (int j = 1; j <= c; j++) cout << s[i][j];
    cout << endl;
  }
}
