def good_string(n, s):
	k = 0
	res = [''] * n
	for i in range(n):
		if i == 0:
			res[i] = s[i]
			k += 1
		elif k % 2 == 0:
			res[k] = s[i]
			k += 1
		elif s[i] != res[k - 1]:
			res[k] = s[i]
			k += 1
	if k % 2 == 1:
		k -= 1
	return (n - k, ''.join(res)[:k])
n = int(input())
s = input()
(k, res) = good_string(n, s)
print(k)
print(res)
