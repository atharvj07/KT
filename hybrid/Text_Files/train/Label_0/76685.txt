(m, n) = map(int, input().split())
a = []
s = 1
for i in range(m):
	a.append(list(map(int, input().split())))
for i in range(m):
	for j in range(1, m):
		if a[j][0] < a[j - 1][0]:
			(a[j][0], a[j - 1][0]) = (a[j - 1][0], a[j][0])
			(a[j][1], a[j - 1][1]) = (a[j - 1][1], a[j][1])
if a[0][0] == 0:
	q = a[0][1]
	for i in range(1, m):
		if a[i][0] <= q:
			if a[i][1] > q:
				q = a[i][1]
		elif a[i][0] > q:
			break
else:
	q = n - 1
if n > q:
	print('NO')
else:
	print('YES')
