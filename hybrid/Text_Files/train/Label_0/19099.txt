#include <bits/stdc++.h>
using namespace std;
const int MAXN = (long long)2e3;
const int INF = (long long)1e9;
const int MOD = (long long)1e9 + 7;
const int CHR = 97;
char arr[MAXN / 2][MAXN / 2];
int mark[MAXN];
vector<int> g[MAXN];
void Add_Edge(int v, int u) {
  g[v].push_back(u);
  g[u].push_back(v);
}
void bfs(int v) {
  queue<int> Q;
  Q.push(v);
  while (Q.size()) {
    v = Q.front();
    Q.pop();
    for (int i = 0; i < g[v].size(); i++) {
      int u = g[v][i];
      if (!mark[u]) {
        mark[u] = mark[v] + 1;
        Q.push(u);
      }
    }
  }
}
void dfs(int v) {
  mark[v] = 1;
  for (int i = 0; i < g[v].size(); i++) {
    int u = g[v][i];
    if (!mark[u]) dfs(u);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> arr[i][j];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (arr[i][j] == '#') Add_Edge(i, n + j);
  mark[n - 1] = 1, bfs(n - 1);
  cout << mark[0] - 1 << "\n";
}
