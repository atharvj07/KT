n = int(input())
a = input()
tab = [int(x) for x in input().split()]
res = []
applied = False

def f(x):
	return int(tab[x - 1])
i = 0
while i < n:
	if f(int(a[i])) > int(a[i]) and (not applied):
		applied = True
		while i < n and f(int(a[i])) >= int(a[i]):
			res.append(f(int(a[i])))
			i += 1
	else:
		res.append(int(a[i]))
		i += 1
print(*res, sep='')
