import java.util.*;
import java.io.*;

import static java.lang.System.in;

public class Main {
    public static void main(String[] args)throws IOException{
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), a=0;
        int[] num = new int[m+1];
        for(int i=0;i<n;i++){
            a = sc.nextInt();
            num[a]++;
        }
        int[][][] dp = new int[2][3][3];
        for(int i=1;i<=m;i++){ // outest loop
            for(int k=0;k<3;k++) Arrays.fill(dp[(i+1)%2][k],0);
            for(int j=0;j<3;j++){ // scan through the possible former dp[i][j][k]
                for(int k=0;k<3;k++){
                    for(int L=0;L<3;L++){
                        if(num[i]<j+k+L) continue; // no need to do extra work for num[i]==0
                        int now = dp[i%2][j][k] + (num[i]-j-k-L)/3+L;
                        dp[(i+1)%2][k][L] = Math.max(now,dp[(i+1)%2][k][L]);
                    }
                }
            }
        }
        System.out.println(dp[(m+1)%2][0][0]);
    }
}
