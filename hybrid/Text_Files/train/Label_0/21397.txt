#include <bits/stdc++.h>
char s1[1000000 + 5], s2[1000000 + 5], s11[1000000 + 5], s22[1000000 + 5];
void fun(char s1[], char s11[]) {
  int i = 0, k = 0;
  if (s1[0] == '0') {
    while (s1[i] == '0') i++;
    k = 0;
    while (s1[i] != '\0') s11[k++] = s1[i++];
    s11[k] = '\0';
  } else
    strcpy(s11, s1);
}
int main() {
  gets(s1);
  gets(s2);
  int len1, len2;
  fun(s1, s11);
  fun(s2, s22);
  len1 = strlen(s11);
  len2 = strlen(s22);
  if (len1 > len2)
    printf(">\n");
  else if (len1 < len2)
    printf("<\n");
  else {
    if (strcmp(s11, s22) > 0)
      printf(">\n");
    else if (strcmp(s11, s22) < 0)
      printf("<\n");
    else
      printf("=\n");
  }
  return 0;
}
