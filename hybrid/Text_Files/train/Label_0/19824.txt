#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int M=1e5+7;
ll a, b, q, s[M], t[M], x;
int main(){
	cin>>a>>b>>q;
	for(int i=1;i<=a;i++) scanf("%lld", &s[i]);
	for(int i=1;i<=b;i++) scanf("%lld", &t[i]);
	s[a+1]=t[b+1]=1e18;
	s[0]=t[0]=-1e18;
	for(int i=1;i<=q;i++) {
		scanf("%lld", &x); 
		int m=lower_bound(s+1, s+a+2, x)-s;
		int n=lower_bound(t+1, t+b+2, x)-t;
		ll minn=1e18;
		minn=min(minn, abs(s[m]-t[n])+min(abs(s[m]-x), abs(t[n]-x)));
		minn=min(minn, abs(s[m-1]-t[n])+min(abs(s[m-1]-x), abs(t[n]-x)));
		minn=min(minn, abs(s[m]-t[n-1])+min(abs(s[m]-x), abs(t[n-1]-x)));
		minn=min(minn, abs(s[m-1]-t[n-1])+min(abs(s[m-1]-x), abs(t[n-1]-x)));
		printf("%lld\n", minn);
	}
}
