#include <bits/stdc++.h>
using namespace std;
long long k;
pair<long long, long long> dp[2][20][(1 << 11)];
long long pow1[30];
void pre() {
  pow1[0] = 1;
  for (long long i = 1; i < 30; i++) pow1[i] = (pow1[i - 1] * 10) % 998244353;
}
pair<long long, long long> solverem(long long zeroflag, long long rem,
                                    long long mask) {
  pair<long long, long long> &w = dp[zeroflag][rem][mask];
  if (__builtin_popcount(mask) > k) return {0, 0};
  if (w.first != -1) return w;
  if (rem == 0 && (__builtin_popcount(mask) <= k)) return {1, 0};
  w = {0, 0};
  if (zeroflag) {
    w = solverem(zeroflag, rem - 1, mask);
    for (long long j = 1; j < 10; j++) {
      pair<long long, long long> res = solverem(0, rem - 1, mask | (1 << j));
      w.second = ((w.second +
                   ((pow1[rem - 1] * j) % 998244353 * res.first) % 998244353) %
                      998244353 +
                  res.second) %
                 998244353;
      w.first = (w.first + res.first) % 998244353;
    }
  } else {
    for (long long j = 0; j < 10; j++) {
      pair<long long, long long> res = solverem(0, rem - 1, mask | (1 << j));
      w.second = ((w.second +
                   ((pow1[rem - 1] * j) % 998244353 * res.first) % 998244353) %
                      998244353 +
                  res.second) %
                 998244353;
      w.first = (w.first + res.first) % 998244353;
    }
  }
  w.first = w.first % 998244353;
  w.second = w.second % 998244353;
  return w;
}
string getString(long long x) {
  stack<long long> st;
  while (x >= 1) {
    st.push(x % 10);
    x /= 10;
  }
  string str = "";
  while (!st.empty()) {
    long long temp = st.top();
    str += to_string(temp);
    st.pop();
  }
  return str;
}
long long solve(long long x) {
  memset(dp, -1, sizeof(dp));
  string a = getString(x);
  long long len = a.length();
  pair<long long, long long> ans = {0, 0};
  long long mask = 0;
  long long prev1 = 0;
  long long flag2 = 0;
  for (long long i = 0; i < len; i++) {
    if (__builtin_popcount(mask) > k) break;
    for (long long j = 0; j < (a[i] - '0'); j++) {
      long long tempmask = (j == 0 && flag2 == 0) ? mask : (mask | (1 << (j)));
      pair<long long, long long> res =
          solverem((j == 0 && flag2 == 0), len - i - 1, tempmask);
      ans.second = ((ans.second + res.second) % 998244353 +
                    ((res.first * pow1[len - i - 1]) % 998244353 *
                     ((prev1 * 10LL) % 998244353 + j) % 998244353) %
                        998244353) %
                   998244353;
    }
    mask = mask | (1 << (a[i] - '0'));
    prev1 = (((prev1 * 10) % 998244353 + (a[i] - '0'))) % 998244353;
    flag2 = 1;
  }
  if (__builtin_popcount(mask) <= k) ans.second = (ans.second + x) % 998244353;
  return ans.second;
}
int main() {
  pre();
  pair<long long, long long> dp[2][20][(1 << 11)];
  for (long long i = 0; i < 2; i++) {
    for (long long j = 0; j < 20; j++) {
      for (long long k1 = 0; k1 < (1 << 11); k1++) dp[i][j][k] = {-1, -1};
    }
  }
  ios::sync_with_stdio(0);

}
