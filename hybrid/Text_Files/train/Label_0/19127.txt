template<typename T> ostream& operator<<(ostream& os, vector<T> &v){
  cout<<"{";
  rep(i,v.size())os<<v[i]<<","<<endl;cout<<"}"<<endl;
  return os;
}
template<typename T,typename S> ostream& operator<<(ostream& os, pair<T,S> &v){
  os<<"("<<v.X<<","<<v.Y<<")";
  return os;
}

int main(){
  int n;
  while(cin>>n){//cout<<"["<<n<<"]"<<endl;
    if(!n)break;
    map<int,pii> mp;
    rep(i,n){
      string str;
      cin>>str;
      char c=str[0];
      if(c=='W'){
	int m,l,pre=0;
	cin>>m>>l;
	//cout<<m<<","<<l;
	vector<pii> ad;
	for(auto it=mp.begin(); it!=mp.end(); ++it){
	  if(pre!=it->X){
	    int d=min(l,it->X-pre);
	    ad.eb(pre,pre+d);
	    l-=d;
	  }
	  pre=it->Y.X;
	}
	if(l)
	  ad.eb(pre,pre+l);
	//cout<<ad;
	rep(i,ad.size())
	  mp[ad[i].X]=pii(ad[i].Y,m);
      }else if(c=='D'){
	int m;
	cin>>m;
	//cout<<m<<endl;
	map<int,pii> tmp=mp;
	mp.clear();
	for(auto it=tmp.begin();it!=tmp.end();++it){
	  if(it->Y.Y!=m)
	    mp[it->X]=it->Y;
	}
      }else{
	int m;
	cin>>m;
	auto it=mp.upper_bound(m);
	if(it==mp.begin()){
	  cout<<-1<<endl;
	}else{
	  --it;
	  //cout<<*it<<endl;
	  if(it->X <= m && m < it->Y.X){
	    cout<<it->Y.Y<<endl;
	  }else{
	    cout<<-1<<endl;
	  }
	}
      }
      //for(auto it=mp.begin();it!=mp.end();++it)cout<<*it;    cout<<endl;
    }cout<<endl;
  }
  return 0;
}
