#include <bits/stdc++.h>
#pragma GCC optimize(3)
using namespace std;
const int inf = 0x3f3f3f3f;
const int maxn = 510;
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  int id, mn = inf, a, b;
  for (int i = 1; i <= n; i++) {
    scanf("%d%d", &a, &b);
    if (a >= m) {
      if (mn > a - m) id = i, mn = a - m;
    } else {
      int ar = a + ceil(1.0 * (m - a) / b) * b;
      if (mn > ar - m) id = i, mn = ar - m;
    }
  }
  printf("%d", id);
  return 0;
}
