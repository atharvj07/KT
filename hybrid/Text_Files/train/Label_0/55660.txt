import java.util.*;
import java.lang.*;
import java.io.*;
import java.math.*;

public class Ideone
{
    public static void main(String[] args)
    {
        Scanner in = new Scanner(System.in);
        BigInteger a = in.nextBigInteger();

        BigInteger have = BigInteger.TEN.pow(19);
        BigInteger have2 = BigInteger.TEN.pow(18);

        BigInteger rem = have2.multiply(BigInteger.valueOf(45).multiply(BigInteger.valueOf(19)));
        rem = rem.remainder(a);

        if (rem.compareTo(BigInteger.ZERO) == 0)
        {
            System.out.println("1 9999999999999999999");
            return;
        }

        BigInteger need = a.subtract(rem);
        System.out.println(need + " " + have.add(need.subtract(BigInteger.ONE)));
        // your code goes here
    }
}
