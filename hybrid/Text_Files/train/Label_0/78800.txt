import sys

def input():
	return sys.stdin.readline().strip()

def iinput():
	return int(input())

def rinput():
	return map(int, sys.stdin.readline().strip().split())

def get_list():
	return list(map(int, sys.stdin.readline().strip().split()))
a = []
for _ in range(4):
	a.append(input())
for i in range(3):
	for j in range(3):
		r = 0
		if a[i][j] == a[i][j + 1]:
			r += 1
		if a[i][j] == a[i + 1][j + 1]:
			r += 1
		if a[i][j] == a[i + 1][j]:
			r += 1
		if r > 1 or r == 0:
			print('YES')
			exit()
print('NO')
