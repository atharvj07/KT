import java.util.Scanner;

public class Main
{
    public static void main(String[] args)
    {
        Scanner in = new Scanner(System.in);
        String line1 = in.nextLine();
        String line2 = in.nextLine();
        String[] nkx = line1.split(" ");
        int n = Integer.parseInt(nkx[0]);
        int k = Integer.parseInt(nkx[1]);
        int x = Integer.parseInt(nkx[2]);
        String[] ai = line2.split(" ");
        long[] arr = new long[n];
        long max = -1;
        int index = 0;
        for (String s : ai)
        {
            long a = Long.parseLong(s);
            arr[index] = a;
            index++;
        }

        long[] leftarr = new long[n];
        leftarr[0] = 0;
        for (int i = 1; i < n; i++)
        {
            leftarr[i] = leftarr[i - 1] | arr[i - 1];
        }

        long[] rightarr = new long[n];
        rightarr[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--)
        {
            rightarr[i] = rightarr[i + 1] | arr[i + 1];
        }
        long result = -1;
        for (int i = 0; i < n; i++)
        {
            long temp = arr[i];
            for(int j = 1; j <= k ; j++)
            {
                temp*=x;
            }
            long t = leftarr[i]|rightarr[i]|temp;
            if(t > result)
            {
                result = t;
            }
        }
        
        System.out.println(result);
    }
}
