public class c {
	public static void main(String[] arg) throws IOException {
		new c();
	}
	ArrayList<Pair> even, odd;
	long[][][] evenMemo, oddMemo;
	static final long MOD = 998244353;
	public c() throws IOException {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int T = in.nextInt();
		while(T-- > 0) {
			int n = in.nextInt();
			even = new ArrayList<Pair>();
			odd = new ArrayList<Pair>();
			for(int i = 0; i < n; i++) {
				String s = in.next();
				if(s.length()%2 == 0) {
					even.add(makePair(s));
				} else {
					odd.add(makePair(s));
				}
			}
			evenMemo = new long[11][even.size()+1][even.size()+1];
			oddMemo = new long[11][(odd.size()+1)/2+1][odd.size()/2+1];
//			System.out.println(odd.get(0));
			for(long[][] a : evenMemo) for(long[] b : a) Arrays.fill(b, -1);
			for(long[][] a : oddMemo) for(long[] b : a) Arrays.fill(b, -1);
			for(int i = 0; i < 11; i++) {
//			for(int i = 1; i <= 1; i++) {
//				System.out.println(i + " " + solveOdd(i, (odd.size()+1)/2, odd.size()/2));
				solveOdd(i, (odd.size()+1)/2, odd.size()/2);
			}
//			System.out.println();
			out.println(solveEven(0, 0, 0));
		}
		in.close(); out.close();
	}
	long solveEven(int m, int e, int o) {
		int idx = e+o;
//		System.out.println(m + " " + e + " " + o);
		if(idx == even.size()) {
//			System.out.println(m + " " + oddMemo[(11-m)%11][(odd.size()+1)/2][odd.size()/2]);
			return oddMemo[(11-m)%11][(odd.size()+1)/2][odd.size()/2];
		}
		long ret = evenMemo[m][e][o];
		if(ret != -1) return ret;
		ret = 0;
		int nm = m+even.get(idx).a-even.get(idx).b;
		nm %= 11;
		nm += 11;
		nm %= 11;
		ret += (odd.size()/2+e+1)*solveEven(nm, e+1, o)%MOD;
		ret %= MOD;
		
		nm = m-even.get(idx).a+even.get(idx).b;
		nm %= 11;
		nm += 11;
		nm %= 11;
		ret += ((odd.size()+1)/2+o)*solveEven(nm, e, o+1)%MOD;
		ret %= MOD;
		
		return evenMemo[m][e][o] = ret;
	}
	long solveOdd(int m, int e, int o) {
//		System.out.println(m + " " + e + " " + o);
		int idx = odd.size()-e-o;
		if(e == 0 && o == 0) {
			if(m == 0) return oddMemo[m][e][o] = 1;
			return oddMemo[m][e][o] = 0;
		}
		long ret = oddMemo[m][e][o];
		if(ret != -1) return ret;
		ret = 0;
		if(e > 0) {
			int nm = m-odd.get(idx).a+odd.get(idx).b;
			nm %= 11;
			nm += 11;
			nm %= 11;
			ret += e*solveOdd(nm, e-1, o)%MOD;
			ret %= MOD;
		}
		if(o > 0) {
			int nm = m+odd.get(idx).a-odd.get(idx).b;
			nm %= 11;
			nm += 11;
			nm %= 11;
			ret += o*solveOdd(nm, e, o-1)%MOD;
			ret %= MOD;
		}
		return oddMemo[m][e][o] = ret;
	}
	Pair makePair(String s) {
		int a = 0, b = 0;
		for(int i = 0; i < s.length(); i++) {
			if(i%2 == 0) a += s.charAt(i)-'0';
			else b += s.charAt(i) -'0';
		}
		return new Pair(a, b);
	}
	class Pair {
		int a, b;
		public Pair(int a, int b) {
			this.a = a;
			this.b = b;
		}
		public String toString() {
			return a + " " + b;
		}
	}
	class FastScanner {
		BufferedReader br;
		StringTokenizer st;
		public FastScanner(InputStream in) {
			br = new BufferedReader(new InputStreamReader(in));
			st = new StringTokenizer("");
		}
		public String next() throws IOException {
			while(!st.hasMoreTokens()) st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}
		public void close() throws IOException {
			br.close();
		}
		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}
	}
}
