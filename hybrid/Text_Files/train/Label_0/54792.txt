import java.io.*;
import java.util.*;
import java.lang.*;

public class Rextester{
    public static void main(String[] args)throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = new Integer(br.readLine());
        br.close();
        int[][] mat = new int[n][n];
        int p=1;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                mat[i][j]=p++;
            }
        }
        StringBuffer sb = new StringBuffer();
        int done = 0,r=0,c=0;
        for(done=0;done<n;done++){
            if(done%2==0){
                for(int y=0;y<n;y++){
                    if(y<n/2){
                        sb.append(mat[r+y][c]).append(" ");
                    }
                    else if(y>n/2){
                        sb.append(mat[r+y][n-c-1]).append(" ");
                    }
                }
                if(n%2!=0){
                    sb.append(mat[r+(n/2)][c]);
                }
                else{
                    sb.append(mat[r+(n/2)][n-c-1]);
                }
                sb.append("\n");
            }
            else{
                for(int y=0;y<n;y++){
                    if(y<n/2){
                        sb.append(mat[r+y][n-c-1]).append(" ");
                    }
                    else if(y>n/2){
                        sb.append(mat[r+y][c]).append(" ");
                    }
                }
                if(n%2!=0){
                    sb.append(mat[r+(n/2)][n-c-1]).append("\n");
                }
                else{
                    sb.append(mat[r+(n/2)][c]).append("\n");
                }
                c++;
            }
        }
        System.out.println(sb);
    }
}
