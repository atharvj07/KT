#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
const int D = 2e4;
vector<int> v[maxn];
int pos[maxn], r[maxn];
int ans[maxn];
int main() {
  ios::sync_with_stdio(false);
  int n, m;
  cin >> n;
  int cnt = 0;
  int x, y;
  memset(ans, -1, sizeof(ans));
  for (int i = 1; i <= n; ++i) {
    cin >> pos[i] >> r[i];
    pos[i] += D;
    for (int j = pos[i] - r[i]; j <= pos[i] + r[i]; ++j) v[j].push_back(i);
  }
  cin >> m;
  for (int i = 1; i <= m; ++i) {
    cin >> x >> y;
    x += D;
    for (int j = 0; j < v[x].size(); ++j) {
      if (ans[v[x][j]] == -1) {
        if ((pos[v[x][j]] - x) * (pos[v[x][j]] - x) + y * y <=
            r[v[x][j]] * r[v[x][j]])
          ans[v[x][j]] = i, cnt++;
      }
    }
  }
  cout << cnt << endl;
  for (int i = 1; i <= n; ++i) cout << ans[i] << " ";
  return 0;
}
