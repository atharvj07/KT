import java.util.Scanner;


public class B {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char prezident = sc.next().charAt(0);
        char[][]a = new char[n+1][m+1];
        for (int i = 1; i <= n; i++) {
            String s = sc.next();
            for (int j = 1; j <= m; j++) {
                a[i][j] = s.charAt(j-1);
            }
        }
        boolean[]used = new boolean[27];
        int[]dx = {1, -1, 0, 0};
        int[]dy = {0, 0, 1, -1};
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (a[i][j] != '.' && a[i][j]!=prezident && !used[(int)a[i][j]-64]) {
                    for (int k = 0; k < 4; k++) {
                        int x = j+dx[k];
                        int y = i+dy[k];
                        if (x >= 1 && x <= m && y >= 1 && y <= n && a[y][x]==prezident) {
                            used[(int)a[i][j]-64] = true;
                            ans++;
                            break;
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }

}
