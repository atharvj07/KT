import heapq
(n, k) = map(int, input().split())
powers = list(map(int, input().split()))
coins = list(map(int, input().split()))
pc = [[powers[i], coins[i]] for i in range(n)]
pc.sort()
powerindex = sorted([i for i in range(n)], key=lambda i: powers[i])
currsum = 0
coinscollected = []
ans = [0] * n
curr = 0
if k > 0:
	for i in range(n):
		if curr < k:
			heapq.heappush(coinscollected, pc[i][1])
			curr += 1
			currsum += pc[i][1]
			ans[powerindex[i]] = currsum
		else:
			ans[powerindex[i]] = currsum + pc[i][1]
			least = heapq.heappop(coinscollected)
			if pc[i][1] > least:
				heapq.heappush(coinscollected, pc[i][1])
				currsum -= least
				currsum += pc[i][1]
			else:
				heapq.heappush(coinscollected, least)
	print(*ans)
else:
	print(*coins)
