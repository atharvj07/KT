import java.io.*; 
import java.util.*; 
import java.lang.*; 
  
public class Main { 
	public static void main(String[] args) {
	Scanner sc = new Scanner(System.in);
	int n = sc.nextInt(),s=sc.nextInt();
	int[] arr = new int[n];
	int[] brr = new int[n];
	for(int i=0;i<n;i++){arr[i]=sc.nextInt();}
	for(int i=0;i<n;i++){brr[i]=sc.nextInt();}
	if(arr[0] == 0){System.out.println("NO");return;}
	if(arr[s-1] == 0 && brr[s-1]==0){System.out.println("NO");return;}
	if(arr[s-1]==1){System.out.println("YES");return;}
	if(brr[s-1]==1){
		for(int i=n-1;i>=0;i--){
			if(arr[i]==1&&brr[i]==1){
				if(i>s-1) {System.out.println("YES");return;}
				if(i<s-1) {System.out.println("NO");return;}
			}
		}
	}System.out.println("NO");
	
	}
}
