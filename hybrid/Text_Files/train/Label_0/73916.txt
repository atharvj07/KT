def solve(arr):
	n = len(arr)
	dp = [[1, arr[0]]] + [None for i in range(n - 1)]
	for i in range(1, n):
		(ans, M) = dp[i - 1]
		if arr[i] > M:
			dp[i] = [ans + 1, arr[i]]
		elif arr[i] == M:
			dp[i] = [ans + 1, arr[i] + 1]
		else:
			dp[i] = [ans, M]
	return dp[-1][0]
t = int(input())
ans = []
for i in range(t):
	n = int(input())
	arr = [int(i) for i in input().split()]
	ans.append(solve(arr))
for i in ans:
	print(i)
