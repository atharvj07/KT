n = int(input())
for i in range(n):
	count = 0
	(b, p, f) = map(int, input().split())
	(h, c) = map(int, input().split())
	if h >= c:
		while b > 1 and p != 0:
			count = count + h
			b = b - 2
			p = p - 1
		while b > 1 and f != 0:
			count = count + c
			b = b - 2
			f = f - 1
	if c > h:
		while b > 1 and f >= 1:
			count = count + c
			b = b - 2
			f = f - 1
		while b > 1 and p >= 1:
			count = count + h
			b = b - 2
			p = p - 1
	print(count)
