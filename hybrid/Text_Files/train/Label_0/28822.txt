#include <bits/stdc++.h>
using namespace std;
bool compareWaste(const pair<int, int>& w1, const pair<int, int>& w2) {
  if (w1.first == w2.first) return w1.second < w2.second;
  return w1.first < w2.first;
}
int main() {
  int n, m, k, t;
  cin >> n >> m >> k >> t;
  vector<pair<int, int>> waste;
  while (k-- > 0) {
    int a, b;
    cin >> a >> b;
    --a;
    --b;
    waste.push_back({a, b});
  }
  sort(waste.begin(), waste.end(), compareWaste);
  while (t-- > 0) {
    int i, j;
    cin >> i >> j;
    --i;
    --j;
    int low = lower_bound(waste.begin(), waste.end(), make_pair(i, j)) -
              waste.begin();
    if (waste[low].first == i && waste[low].second == j) {
      cout << "Waste\n";
      continue;
    }
    int id = (i * m + j - low) % 3;
    switch (id) {
      case 0:
        cout << "Carrots\n";
        break;
      case 1:
        cout << "Kiwis\n";
        break;
      case 2:
        cout << "Grapes\n";
        break;
    }
  }
  return 0;
}
