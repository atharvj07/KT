#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[2005], am[2005], b[2005], bm[2005], ans, x, y, cap, n, c[2005], d[2005];
  vector<int> aa[2005], bb[2005];
  int k, h, m, i;
  int p = 0, q = 0;
  cin >> n >> cap;
  int tmp = cap;
  for (i = 1; i <= n; i++) {
    cin >> k >> h >> m;
    if (k) {
      a[p++] = h;
      aa[h].push_back(m);
    } else {
      b[q++] = h;
      bb[h].push_back(m);
    }
  }
  sort(a, a + p);
  sort(b, b + q);
  ans = 0;
  for (i = 0; i <= 2000; i++) {
    sort(aa[i].begin(), aa[i].end());
    sort(bb[i].begin(), bb[i].end());
  }
  int zz;
  zz = a[0];
  k = aa[zz].size();
  if (p != 0) am[0] = aa[zz][k - 1];
  k--;
  c[0] = a[0];
  for (i = 1; i < p; i++) {
    c[i] = a[i];
    if (a[i] == zz) {
      am[i] = aa[zz][k--];
    } else {
      k = aa[a[i]].size();
      am[i] = aa[a[i]][k - 1];
      k--;
    }
    zz = a[i];
  }
  zz = b[0];
  k = bb[zz].size();
  if (q != 0) bm[0] = bb[zz][k - 1];
  k--;
  d[0] = b[0];
  for (i = 1; i < q; i++) {
    d[i] = b[i];
    if (b[i] == zz) {
      bm[i] = bb[zz][k--];
    } else {
      k = bb[b[i]].size();
      bm[i] = bb[b[i]][k - 1];
      k--;
    }
    zz = b[i];
  }
  int ck = 1, z1 = 0, z2 = 0, br, mnz;
  cap = tmp;
  int mxz, zzz, save;
  while (ck) {
    br = 0, mxz = 0, save = -1, mnz = 2005;
    for (i = 0; i < p; i++) {
      if (a[i] != -1) {
        zzz = cap - a[i];
        if (zzz >= 0) {
          if (am[i] > mxz) {
            mxz = am[i];
            save = i;
          } else if (am[i] == mxz) {
            if (zzz < mnz) {
              mxz = am[i];
              save = i;
              mnz = zzz;
            }
          }
        }
      }
    }
    if (save != -1 && z1 == 0) {
      ans++;
      z1 = 1;
      cap += mxz;
      br = 1;
      a[save] = -1;
    }
    if (z1) z2 = 0;
    br = 0, mxz = 0, save = -1, mnz = 2005;
    for (i = 0; i < q; i++) {
      if (b[i] != -1) {
        zzz = cap - b[i];
        if (zzz >= 0) {
          if (bm[i] > mxz) {
            mxz = bm[i];
            save = i;
          } else if (bm[i] == mxz) {
            if (zzz < mnz) {
              mxz = bm[i];
              save = i;
              mnz = zzz;
            }
          }
        }
      }
    }
    if (save != -1 && z2 == 0) {
      ans++;
      z2 = 1;
      cap += mxz;
      br = 1;
      b[save] = -1;
    }
    if (p == 0 || q == 0) br = 0;
    if (z2) z1 = 0;
    if (br == 0) break;
  }
  int mx = ans;
  ck = 1, z1 = 0, z2 = 0, br;
  cap = tmp;
  ans = 0;
  while (ck) {
    br = 0, mxz = 0, save = -1, mnz = 2005;
    for (i = 0; i < q; i++) {
      if (d[i] != -1) {
        zzz = cap - d[i];
        if (zzz >= 0) {
          if (bm[i] > mxz) {
            mxz = bm[i];
            save = i;
          } else if (bm[i] == mxz) {
            if (zzz < mnz) {
              mxz = bm[i];
              save = i;
              mnz = zzz;
            }
          }
        }
      }
    }
    if (save != -1 && z1 == 0) {
      ans++;
      z1 = 1;
      cap += mxz;
      br = 1;
      d[save] = -1;
    }
    if (z1) z2 = 0;
    br = 0, mxz = 0, save = -1, mnz = 2005;
    for (i = 0; i < p; i++) {
      if (c[i] != -1) {
        zzz = cap - c[i];
        if (zzz >= 0) {
          if (am[i] > mxz) {
            mxz = am[i];
            save = i;
          } else if (am[i] == mxz) {
            if (zzz < mnz) {
              mxz = am[i];
              save = i;
              mnz = zzz;
            }
          }
        }
      }
    }
    if (save != -1 && z2 == 0) {
      ans++;
      z2 = 1;
      cap += mxz;
      br = 1;
      c[save] = -1;
    }
    if (p == 0 || q == 0) br = 0;
    if (z2) z1 = 0;
    if (br == 0) break;
  }
  mx = max(mx, ans);
  cout << mx;
}
