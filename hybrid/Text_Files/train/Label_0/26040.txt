#include <bits/stdc++.h>
#pragma GCC optimize(2)
#pragma G++ optimize(2)
using namespace std;
const long long inf = 0x3f3f3f3f;
const long long mod = 1e9 + 7;
const long long N = 1e6 + 10;
const double eps = 1e-6;
int n, m, t;
long long f[N] = {0, 1, 2, 3};
string s;
void init() {
  for (int i = 4; i <= 100000; i++) f[i] = (f[i - 1] + f[i - 2]) % mod;
}
int main() {
  ios::sync_with_stdio(false);
  long long ans = 1;
  init();
  cin >> s;
  s += '$';
  for (int i = 0; s[i]; i++) {
    if (s[i] == 'w' || s[i] == 'm') {
      puts("0");
      return 0;
    }
  }
  for (int i = 0; s[i]; i++) {
    if (s[i] == 'u') {
      int num = 0;
      for (int j = i; s[j]; j++) {
        if (s[j] == 'u')
          num++;
        else {
          i = j - 1;
          break;
        }
      }
      ans = (ans * f[num]) % mod;
    } else if (s[i] == 'n') {
      int num = 0;
      for (int j = i; s[j]; j++) {
        if (s[j] == 'n')
          num++;
        else {
          i = j - 1;
          break;
        }
      }
      ans = (ans * f[num]) % mod;
    }
  }
  cout << ans;
  return 0;
}
