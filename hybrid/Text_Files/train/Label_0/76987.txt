maxn = 200005
fac = [[] for i in range(maxn)]
cnt = [[] for i in range(maxn)]
n = int(input())
arr = list(map(int, input().split()))
if n == 1:
	print(arr[0])
else:
	for i in range(n):
		j = 2
		while j * j <= arr[i]:
			count = 0
			if arr[i] % j == 0:
				while arr[i] % j == 0:
					count += 1
					arr[i] = arr[i] // j
				fac[j].append(1)
				cnt[j].append(count)
			j += 1
		if arr[i] > 1:
			fac[arr[i]].append(arr[i])
			cnt[arr[i]].append(1)
	ans = 1
	for i in range(2, maxn):
		if len(fac[i]) >= n - 1:
			temp = cnt[i][:]
			temp.sort()
			if len(fac[i]) == n:
				ans = ans * pow(i, temp[1])
			else:
				ans = ans * pow(i, temp[0])
	print(ans)
