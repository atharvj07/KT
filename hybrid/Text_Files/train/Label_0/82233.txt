for i in range(int(input())):
	k = list(map(int, input().split()))
	if len(set(k)) == 2:
		if k[1] == k[2]:
			if k[0] < k[1]:
				print('YES')
				print(k[0], k[0], k[1])
			else:
				print('NO')
		if k[0] == k[1]:
			if k[2] < k[1]:
				print('YES')
				print(k[1], k[2], k[2])
			else:
				print('NO')
		if k[0] == k[2]:
			if k[1] < k[0]:
				print('YES')
				print(k[1], k[0], k[1])
			else:
				print('NO')
	elif len(set(k)) == 1:
		print('YES')
		print(k[0], k[0], k[0])
	else:
		print('NO')
