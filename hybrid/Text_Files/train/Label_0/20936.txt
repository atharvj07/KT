#include <bits/stdc++.h>
using namespace std;
int main() {
  int a;
  int b;
  cin >> a >> b;
  cout.precision(13);
  if (b > a) {
    cout << -1;
  } else if ((a % b == 0) && ((a / b) % 2 == 1)) {
    cout << b;
  } else {
    int k = a / b;
    if (k % 2 == 0) {
      k--;
    }
    int dist = a - k * b;
    double x1 = b;
    double x2 = (double)dist / (double)k + (double)b;
    double x = x1;
    double ang;
    double x_old;
    for (int i = 0; i < 50; i++) {
      ang = (x - b) / (k * x - a);
      x_old = x;
      if (ang > -1) {
        x = (x + x2) / 2.0;
        x1 = x_old;
      } else if (ang < -1) {
        x = (x + x1) / 2.0;
        x2 = x_old;
      } else {
        cout << x;
        return 0;
      }
    }
    cout << x;
  }
  return 0;
}
