from decimal import *
getcontext().prec = 300
split_strings = input().split()
[a, b, c] = map(lambda x: int(x), split_strings)
a = Decimal(a)
b = Decimal(b)
decimal = a / b * 10
for i in range(1, 300):
	ones = int(decimal)
	fractional = decimal % 1
	if ones == c:
		print(i)
		exit()
	decimal = fractional * 10
print(-1)
