	public int opt(int L, int R)
	{
		if(L == R)
			return dp[L][R] = 1;
		if(dp[L][R] != -1)
			return dp[L][R];
		if(tryCompress(L,R) != NO)
		{
			return dp[L][R] = 1;
		}
		int min = 10000;
		for(int i = L;i<R;i++)
		{
			int left = opt(L, i);
			int right = opt(i+1,R);
			min = Math.min(min, left + right);
		}
		return dp[L][R] = min;
	}
	
	public int tryCompress(int L, int R)//returns what range is compressed to, or NO otherwise
	{
		if(L == R)
			return compress[L][R] = ints[L];
		if(compress[L][R] != -1)
			return compress[L][R];
		for(int i = L;i<R;i++)
		{
			int left = tryCompress(L, i);
			int right = tryCompress(i+1, R);
			if(left != NO && right != NO && left == right)
				return compress[L][R] = left+1;
		}
		return compress[L][R] = NO;
	}
