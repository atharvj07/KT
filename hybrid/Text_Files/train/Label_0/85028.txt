import sys

def solve(ops, x, n):
	(mn, mx) = (x, x)
	for (l, r) in ops:
		if l <= mn and r >= mn:
			mn = l
		if l <= mx and r >= mx:
			mx = r
	return mx - mn + 1
IN = [x.strip() for x in sys.stdin.readlines()]
T = int(IN[0])
cur = 1
for ti in range(T):
	(n, x, m) = [int(x) for x in IN[cur].split(' ')]
	ops = []
	for i in range(m):
		(l, r) = [int(x) for x in IN[cur + 1 + i].split(' ')]
		ops.append((l, r))
	result = solve(ops, x, n)
	print(result)
	cur += 1 + m
