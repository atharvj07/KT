from sys import stdin, stdout
import math

def gcd(a, b):
	return math.gcd(a, b)
for _ in range(int(stdin.readline())):
	n = int(stdin.readline())
	a = list(map(int, stdin.readline().split()))
	x = 1
	xx = 0
	for i in range(n):
		g = gcd(x, i + 2)
		x *= i + 2
		x //= g
		if x > 1000000000:
			xx = 1
			break
		if a[i] % x == 0:
			xx = 2
			break
	if xx == 2:
		print('NO')
	else:
		print('YES')
