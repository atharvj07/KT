n = int(input())
s = input().strip()
a = [0] + list(map(int, input().split()))
MX = 105
dp = [0] * MX ** 3
idx = lambda i, j, k: i * MX * MX + j * MX + k
for i in range(n + 1):
	for k in range(1, n + 1):
		dp[idx(i, i, k)] = 0
for gap in range(1, n + 1):
	for i in range(n):
		j = i + gap
		if j > n:
			break
		for k in range(1, n + 1):
			cur = idx(i, j, k)
			dp[cur] = dp[idx(i + 1, j, 1)] + a[k]
			for m in range(i + 1, j):
				if s[i] == s[m]:
					dp[cur] = max(dp[cur], dp[idx(i + 1, m, 1)] + dp[idx(m, j, k + 1)])
print(dp[idx(0, n, 1)])
