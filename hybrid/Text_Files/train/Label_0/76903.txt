for _ in range(int(input())):
	(r, g, b, w) = map(int, input().split())
	m = min(r, g, b)
	if m == 0:
		s = 0
		if r % 2 == 1:
			s += 1
		if g % 2 == 1:
			s += 1
		if b % 2 == 1:
			s += 1
		if w % 2 == 1:
			s += 1
		if s > 1:
			print('NO')
		else:
			print('YES')
	else:
		s = 0
		if w % 2 == 0:
			if r % 2 == 1:
				s += 1
			if g % 2 == 1:
				s += 1
			if b % 2 == 1:
				s += 1
			if s == 2:
				print('NO')
			else:
				print('YES')
		else:
			if r % 2 == 1:
				s += 1
			if g % 2 == 1:
				s += 1
			if b % 2 == 1:
				s += 1
			if s == 1:
				print('NO')
			else:
				print('YES')
