#include <bits/stdc++.h>
using namespace std;

vector<pair<int, int> > graph[100010];

pair<int, int> p[100010];
pair<int, int> q[100010];

long long ans[100010] = {};

int main(){
  int n;
  cin >> n;

  int tmp;

  for(int i = 0;i < n;i++){
    cin >> tmp;
    p[i] = make_pair(tmp, i);
  }

  for(int i = 0;i < n;i++){
    cin >> tmp;
    q[i] = make_pair(tmp, i);
  }

  sort(p,p+n);
  sort(q,q+n);

  for(int i = 1;i < n;i++){
    graph[p[i].second].push_back(make_pair(p[i-1].second, p[i].first-p[i-1].first));
    graph[p[i-1].second].push_back(make_pair(p[i].second, p[i].first-p[i-1].first));
  }

  for(int i = 1;i < n;i++){
    graph[q[i].second].push_back(make_pair(q[i-1].second, q[i].first-q[i-1].first));
    graph[q[i-1].second].push_back(make_pair(q[i].second, q[i].first-q[i-1].first));
  }

  priority_queue<pair<long long, int>, vector<pair<long long, int> >, greater<pair<long long, int> > > que;

  ans[0] = 0;

  for(int i = 1;i < n;i++){
    ans[i] = LLONG_MAX;
  }

  que.push(make_pair(0, 0));

  while(!que.empty()){
    long long cost = que.top().first;
    int idx = que.top().second;
    que.pop();

    for(int i = 0;i < graph[idx].size();i++){
      int next = graph[idx][i].first;
      long long ncost = cost + graph[idx][i].second;
      if(ans[next] > ncost){
        ans[next] = ncost;
        que.push(make_pair(ncost, next));
      }
    }
  }

  for(int i = 0;i < n;i++){
    cout << ans[i] << endl;
  }

  return 0;
}
