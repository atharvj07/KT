(n, m) = list(map(int, input().split()))
matrix = [list(map(int, input().split())) for _ in range(n)]
ones_row = [sum(row) for row in matrix]
ones_col = []
for i in range(m):
	x = 0
	for j in range(n):
		if matrix[j][i]:
			x += 1
	ones_col.append(x)
answer = 0
for x in ones_row:
	ones = x
	zeroes = m - x
	answer += 2 ** ones - 1
	answer += 2 ** zeroes - 1
for x in ones_col:
	ones = x
	zeroes = n - x
	answer += 2 ** ones - 1
	answer += 2 ** zeroes - 1
answer -= m * n
print(answer)
