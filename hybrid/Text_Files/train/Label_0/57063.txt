import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

import static java.lang.Math.sqrt;
import static java.lang.Math.toIntExact;

public class SolutionF extends Thread {
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                                            InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }

    private static final FastReader scanner = new FastReader();
    private static final PrintWriter out = new PrintWriter(System.out);

    public static void main(String[] args) {
        solve();
        out.close();
    }

    //returns an unsorted list of all the divisors of a number
    public static List<Integer> getDivisorsOfN(int n) {
        List<Integer> result = new ArrayList<>();

        result.add(1);

        if (n != 1) {
            result.add(n);
        }

        for (int i = 2; i <= sqrt(n); i++) {
            if (n % i == 0) {
                result.add(i);

                if (n / i != i) {
                    result.add(n / i);
                }
            }
        }

        return result;
    }

    static int counter = 0;
    private static void solve() {
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        System.out.println("! " + (work(n, true, m) * work(m, false, n)));
        System.out.println(counter);
    }


    private static int work(int n, boolean isRows, int otherSize) {
        List<Integer> divisors = getDivisorsOfN(n);
        Collections.sort(divisors);

        int[] dp = new int[divisors.size()];
        dp[divisors.size()-1] = 1;

        div_loop:
        for (int i = divisors.size() - 2; i >= 0; i--) {
            if (dp[i] == 1) {
                continue;
            }

            for (int j = i+1; j < divisors.size(); j++) {
                for (int k = j+1; k < divisors.size(); k++) {
                    long gcd = gcd(divisors.get(j), divisors.get(k));
                    if (dp[j] == 1 && dp[k] == 1 && gcd == divisors.get(i)) {
                        dp[i] = 1;
                        continue div_loop;
                    }
                }
            }

            int x = 1_000_000_000;
            int r = divisors.get(i);
            boolean isEqual = true;
            for (int j = i+1; j < divisors.size(); j++) {
                if (divisors.get(j) % r == 0 && dp[j] == 1) {
                    x = Math.min(x, divisors.get(j));
                } else if (divisors.get(j) % r == 0) {
                    isEqual = false;
                    break;
                }
            }


            int height;
            int width;
            int rect2R;
            int rect2H;

            if (isRows) {
                height = r;
                width = otherSize;
                rect2R = r + 1;
                rect2H = 1;
            } else {
                height = otherSize;
                width = r;
                rect2R = 1;
                rect2H = r + 1;
            }

            while (r * 2 <= x && isEqual) {
                System.out.println("? " + height + " " + width + " 1 1 " + rect2R + " " + rect2H);
                counter++;
                int t = scanner.nextInt();
                if (t == 0) {
                    isEqual = false;
                    r*=2;
                }

                if (isRows) {
                    height *= 2;
                    rect2R = height + 1;
                } else {
                    width *= 2;
                    rect2H = width + 1;
                }
                r *= 2;
            }

            if (r != x && isEqual) {
                int leftSize = x - r;
                if (isRows) {
                    height = leftSize;
                    width = otherSize;
                    rect2R = x - leftSize + 1;
                    rect2H = 1;
                } else {
                    height = otherSize;
                    width = leftSize;
                    rect2R = 1;
                    rect2H = x - leftSize + 1;
                }
                System.out.println("? " + height + " " + width + " 1 1 " + rect2R + " " + rect2H);
                counter++;
                int t = scanner.nextInt();
                if (t == 0) {
                    isEqual = false;
                }
            }

            if (!isEqual) {
                dp[i] = 0;
            } else {
                dp[i] = 1;
            }
        }


        int sum = 0;
        for (int i = 0; i < divisors.size(); i++) {
            sum += dp[i];
        }

        return sum;
    }

    public static long gcd(long a, long b) {
        return (a == 0) ? b : gcd(b % a, a);
    }
