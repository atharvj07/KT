#include <bits/stdc++.h>
using namespace std;
int main() {
  int hash[26] = {0};
  string s;
  cin >> s;
  int n = s.length();
  if (n == 1) {
    cout << "Just a legend\n";
    return 0;
  }
  if (n >= 3) {
    for (int i = 0; i < n; i++) {
      hash[(int)s[i] - 'a']++;
    }
    int ct = 0;
    for (int i = 0; i < 26; i++) {
      if (hash[i] > 0) ct++;
    }
    if (ct == 1) {
      for (int i = 0; i < n - 2; i++) cout << s[i];
      cout << "\n";
      return 0;
    }
  }
  int z[1000006];
  int L = 0, R = 0;
  for (int i = 1; i < n; i++) {
    if (i > R) {
      L = R = i;
      while (R < n && s[R - L] == s[R]) R++;
      z[i] = R - L;
      R--;
    } else {
      int k = i - L;
      if (z[k] < R - i + 1)
        z[i] = z[k];
      else {
        L = i;
        while (R < n && s[R - L] == s[R]) R++;
        z[i] = R - L;
        R--;
      }
    }
  }
  set<pair<int, int> > st;
  int steps = 0;
  for (int i = n - 1; i >= 0; i--) {
    steps++;
    if (steps == z[i]) {
      st.insert(pair<int, int>(z[i], i));
    }
  }
  set<pair<int, int> >::reverse_iterator it;
  int start = -1, len = 0;
  for (it = st.rbegin(); it != st.rend(); it++) {
    if (len < it->first)
      for (int i = 0; i < it->second; i++) {
        if (z[i] >= it->first) {
          start = i;
          len = it->first;
        }
      }
  }
  if (start == -1) {
    cout << "Just a legend\n";
    return 0;
  }
  for (int i = start; i < start + len; i++) cout << s[i];
  cout << "\n";
  return 0;
}
