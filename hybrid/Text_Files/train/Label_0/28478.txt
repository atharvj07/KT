#include <bits/stdc++.h>
using namespace std;
long long comb[100][100];
long long nCr(long long n, long long r) {
  if (n < 0 || r < 0) return 0;
  long long &ans = comb[n][r];
  if (n < r) return 0;
  if (n == r || r == 0) return 1;
  if (ans != -1) return ans;
  ans = (nCr(n, r - 1) * (n - r + 1)) / (r);
  return ans;
}
int32_t main() {
  memset(comb, -1, sizeof comb);
  long long boy, girl, team;
  cin >> boy >> girl >> team;
  long long tot = nCr(boy + girl, team);
  long long unvalid = nCr(boy, team);
  for (long long i = 0; i < 4; i++) {
    unvalid += nCr(girl, team - i) * nCr(boy, i);
  }
  cout << tot - unvalid;
}
