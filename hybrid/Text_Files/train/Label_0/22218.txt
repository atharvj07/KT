#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, p[105][105], ans = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 2; j++) {
      cin >> p[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 2; j++) {
      if (p[i][j + 1] - p[i][j] >= 2) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
