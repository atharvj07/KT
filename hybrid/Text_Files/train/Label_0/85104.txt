for i in range(int(input())):
	n = int(input())
	ans = []
	(x1, y1, x2, y2) = (-1000000000, -1000000000, 1000000000, 1000000000)
	for i in range(n):
		ans.append(list(map(int, input().split())))
	for i in range(n):
		k = ans[i]
		if k[2] == 0:
			if k[0] > x1:
				x1 = k[0]
		if k[3] == 0:
			if k[1] < y2:
				y2 = k[1]
		if k[4] == 0:
			if k[0] < x2:
				x2 = k[0]
		if k[5] == 0:
			if k[1] > y1:
				y1 = k[1]
	ans1 = x1
	ans2 = y1
	if x1 == -1000000000 and x2 == 1000000000:
		ans1 = 0
	elif x1 == -1000000000:
		ans1 = x2
	elif x2 == 1000000000:
		ans1 = x1
	if y1 == -1000000000 and y2 == 1000000000:
		ans2 = 0
	elif y1 == -1000000000:
		ans2 = y2
	elif y2 == 1000000000:
		ans2 = y1
	if x1 > x2:
		print(0)
	elif y1 > y2:
		print(0)
	else:
		print(1, ans1, ans2)
