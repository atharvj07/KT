#include <bits/stdc++.h>
using namespace std;

#define INF 1<<60

typedef pair<long long int, long long int> PII;
typedef vector<PII> VPII;

int main() {
	long long int N;
	while( cin >> N, N ) {
		map<long long int, VPII> M;
		priority_queue<PII, VPII, greater<PII>>que;
		que.push( make_pair( 0, 1LL << INF ) );
		for( size_t i = 0; i < N; i++ ) {
			char c;
			cin >> c;
			if( c == 'W' ) {
				int a, b;
				cin >> a >> b;
				while( b ) {
					PII now = que.top(); que.pop();
					if( b < now.second - now.first ) {
						que.push( make_pair( now.first + b, now.second ) );
						M[a].push_back( make_pair( now.first, now.first + b - 1 ) );
						b = 0;
					} else if( b == now.second - now.first ) {
						M[a].push_back( make_pair( now.first, now.second ) );
						b = 0;
					} else {
						M[a].push_back( make_pair( now.first, now.second ) );
						b -= now.second - now.first + 1;
					}
				}
			} else if( c == 'D' ) {
				int a;
				cin >> a;
				for( auto x : M[a] ) {
					que.push( x );
				}
				M[a].clear();
			} else {
				int a;
				cin >> a;
				bool f = 0;
				for( auto x : M ) {
					for( auto y : x.second ) {
						if( y.first <= a &&a <= y.second ) {
							cout << x.first << endl;
							f = 1;
							break;
						}
					}
					if( f ) {
						break;
					}
				}
				if( !f ) {
					cout << -1 << endl;
				}
			}
		}
		cout << endl;
	}
}
