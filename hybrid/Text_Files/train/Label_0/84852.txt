tests = int(input())
res = []
for i in range(tests):
	(a, b, c) = map(int, input().split())
	arr = [a, b, c]
	if abs(a - b) == c or abs(a - c) == b or abs(b - c) == a:
		res.append('YES')
	elif arr.count(max(arr)) == 2 and min(arr) % 2 == 0:
		res.append('YES')
	elif arr.count(min(arr)) == 2 and max(arr) % 2 == 0:
		res.append('YES')
	elif arr.count(min(arr)) == 3 and max(arr) % 2 == 0:
		res.append('YES')
	else:
		res.append('NO')
for i in range(tests):
	print(res[i])
