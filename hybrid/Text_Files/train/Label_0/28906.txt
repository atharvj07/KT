#include <bits/stdc++.h>
using namespace std;
string x, y;
int n;
int move(int p, int en) {
  int c = 0;
  while (p != en) {
    if (p > en) {
      swap(y[p], y[p - 1]);
      c++;
      p--;
    } else {
      swap(y[p], y[p + 1]);
      c++;
      p++;
    }
  }
  return c;
}
int main() {
  cin >> x;
  n = x.size();
  int p5 = -1, p2 = -1, p7 = -1;
  vector<int> z;
  for (int i = 0; i < x.size(); i++) {
    if (x[i] == '5') p5 = i;
    if (x[i] == '2') p2 = i;
    if (x[i] == '7') p7 = i;
    if (x[i] == '0') z.push_back(i);
  }
  int ans = 1e9;
  if (z.size() > 1) {
    if (x[n - 1] == '0' && x[n - 2] == '0') return puts("0"), 0;
    y = x;
    int a = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (y[i] == '0') {
        a += move(i, n - 1);
        break;
      }
    }
    for (int i = n - 2; i >= 0; i--) {
      if (y[i] == '0') {
        a += move(i, n - 2);
        break;
      }
    }
    ans = min(ans, a);
  }
  if (p5 != -1 && z.size()) {
    if (x[n - 1] == '0' && x[n - 2] == '5') return puts("0"), 0;
    y = x;
    int a = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (y[i] == '0') {
        a += move(i, n - 1);
        break;
      }
    }
    char u = 0;
    for (int i = n - 2; i >= 0; i--) {
      if (y[i] == '5') {
        if (i == 0 && y[i + 1] == '0') {
          for (int j = 1; j < n; j++) {
            if (y[j] != '0') {
              a += move(j, 0);
              break;
            }
          }
          i++;
        }
        if (i == 0 && y[i + 1] == '0') break;
        u = '5';
        a += move(i, n - 2);
        break;
      }
    }
    if (u == '5') ans = min(ans, a);
  }
  if (p5 != -1 && p7 != -1) {
    if (x[n - 1] == '5' && x[n - 2] == '7') return puts("0"), 0;
    y = x;
    int a = 0;
    char u = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (y[i] == '5') {
        if (i == 0 && y[i + 1] == '0') {
          for (int j = 1; j < n; j++) {
            if (y[j] != '0') {
              a += move(j, 0);
              break;
            }
          }
          i++;
        }
        if (i == 0 && y[i + 1] == '0') break;
        u = '5';
        a += move(i, n - 1);
        break;
      }
    }
    for (int i = n - 2; i >= 0; i--) {
      if (y[i] == '7') {
        if (i == 0 && y[i + 1] == '0') {
          for (int j = 1; j < n; j++) {
            if (y[j] != '0') {
              a += move(j, 0);
              break;
            }
          }
          i++;
        }
        if (i == 0 && y[i + 1] == '0') break;
        if (u == '5') u = '7';
        a += move(i, n - 2);
        break;
      }
    }
    if (u == '7') ans = min(ans, a);
  }
  if (p5 != -1 && p2 != -1) {
    if (x[n - 1] == '5' && x[n - 2] == '2') return puts("0"), 0;
    y = x;
    int a = 0;
    char u = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (y[i] == '5') {
        if (i == 0 && y[i + 1] == '0') {
          for (int j = 1; j < n; j++) {
            if (y[j] != '0') {
              a += move(j, 0);
              break;
            }
          }
          i++;
        }
        if (i == 0 && y[i + 1] == '0') break;
        u = '5';
        a += move(i, n - 1);
        break;
      }
    }
    for (int i = n - 2; i >= 0; i--) {
      if (y[i] == '2') {
        bool can = true;
        if (i == 0 && y[i + 1] == '0') {
          for (int j = 1; j < n; j++) {
            if (y[j] != '0') {
              a += move(j, 0);
              break;
            }
          }
          i++;
        }
        if (i == 0 && y[i + 1] == '0') break;
        if (u == '5') u = '2';
        a += move(i, n - 2);
        break;
      }
    }
    if (u == '2') ans = min(ans, a);

    cout << "-1";
  else
    cout << ans << endl;

}
