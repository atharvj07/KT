list_ans = []
for _ in range(int(input())):
	n = int(input())
	list1 = list(map(int, input().split()))
	(a, b) = (0, 0)
	(a1, b1) = (0, 0)
	list1.sort()
	while list1:
		a1 = list1.pop()
		if a1 % 2 == 0:
			a += a1
		if list1:
			b1 = list1.pop()
			if b1 % 2 == 1:
				b += b1
	if a > b:
		list_ans.append('Alice')
	elif a < b:
		list_ans.append('Bob')
	else:
		list_ans.append('Tie')
for item in list_ans:
	print(item)
