import math
(n, k) = map(int, input().split())
cost = list(map(int, input().split()))
stock = list(map(int, input().split()))
lo = 0
hi = 2000000000
mid = 0
while lo < hi:
	oldmid = mid
	mid = math.ceil((lo + hi) // 2)
	need = 0
	for i in range(n):
		need += max(0, cost[i] * mid - stock[i])
	if need == k or oldmid == mid:
		break
	elif need < k:
		lo = mid
	else:
		hi = mid
print(mid)
