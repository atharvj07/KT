#include <bits/stdc++.h>
int t[300][300], cnt, ans[300 * 300];
void dfs(int x, int y) {
  if (t[x][y] == -1) return;
  cnt += t[x][y];
  t[x][y] = -1;
  dfs(x + 1, y);
  dfs(x - 1, y);
  dfs(x, y + 1);
  dfs(x, y - 1);
}
int main() {
  register int i, j;
  int w, h, n, k, x1, y1, x2, y2;
  scanf("%d %d %d", &w, &h, &n);
  for (i = 0; i < n; ++i) {
    scanf("%d %d %d %d", &x1, &y1, &x2, &y2);
    for (j = x1 * 2; j <= x2 * 2; ++j) {
      for (k = y1 * 2; k <= y2 * 2; ++k) t[k + 1][j + 1] = -1;
    }
  }
  for (i = 0; i <= h * 2 + 2; ++i) {
    t[i][0] = -1;
    t[i][w * 2 + 2] = -1;
  }
  for (j = 0; j <= w * 2 + 2; ++j) {
    t[0][j] = -1;
    t[h * 2 + 2][j] = -1;
  }
  for (i = 0; i < h; ++i)
    for (j = 0; j < w; ++j) t[i * 2 + 2][j * 2 + 2] = 1;
  for (i = 1; i < h * 2 + 1; ++i) {
    for (j = 1; j < w * 2 + 1; ++j) {
      if (t[i][j] == -1) continue;
      cnt = 0;
      dfs(i, j);
      ans[cnt]++;
    }
  }
  for (i = 1; i < 300 * 300; ++i) {
    for (j = 1; j <= ans[i]; ++j) printf("%d ", i);
  }
  return 0;
}
