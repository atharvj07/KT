for ___ in range(int(input())):
	(a, b, c) = map(int, input().split())
	v1 = 0
	v2 = 0
	v1 += a // 2 * 3
	v2 += a // 2 * 3
	if a % 2 == 1:
		v1 += 3
	if v1 > v2 and b != 0:
		b -= 1
		v2 += 2
	v1 += b // 2 * 2
	v2 += b // 2 * 2
	if b % 2 == 1:
		v2 += 2
	if v1 < v2:
		t = v1
		v1 = v2
		v2 = t
	while c != 0 and v1 > v2:
		v2 += 1
		c -= 1
	d = v1 - v2
	if c == 0:
		print(d)
	elif d % 2 == c % 2:
		print(0)
	else:
		print(1)
