#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int q, d, l, r, k, a[N], b[N], c[N];
long long ans;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> q;
  while (q--) {
    ans = 0;
    cin >> d >> l >> r;
    if (d == 1) {
      cin >> k;
      if (k > 0)
        for (int i = l; i < r; i++) {
          if (!a[i] || a[i] && a[i] > k) a[i] = k;
          if (b[i]) c[i] = a[i] + b[i];
        }
      else {
        k *= -1;
        for (int i = l; i < r; i++) {
          if (!b[i] || b[i] && b[i] > k) b[i] = k;
          if (a[i]) c[i] = a[i] + b[i];
        }
      }
    } else {
      for (int i = l; i < r; i++) ans += c[i];
      cout << ans << "\n";
    }
  }
  return 0;
}
