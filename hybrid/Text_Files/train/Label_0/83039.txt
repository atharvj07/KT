N = int(input())
for _ in range(N):
	t = input()
	s = ''
	(y, l) = (0, 0)
	for i in range(len(t) - 1):
		if t[i] == t[i + 1]:
			if t[i] == '0':
				s += '01'
			else:
				s += '10'
		else:
			s += t[i]
		if t[i] == '0':
			l += 1
		else:
			y += 1
	s += t[len(t) - 1]
	if t[len(t) - 1] == '0':
		l += 1
	else:
		y += 1
	if y != 0 and l != 0:
		print(s)
	else:
		print(t)
