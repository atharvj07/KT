#include <bits/stdc++.h>
using namespace std;
long long binpow(long long n, long long k) {
  long long ans = 1;
  for (long long i = 1; i <= k; i++) {
    ans *= n;
  }
  return ans;
}
long long sum(long long x) {
  long long ans = 0;
  while (x) {
    ans += (x % 10);
    x /= 10;
  }
  return ans;
}
signed main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  long long a, b, c;
  cin >> a >> b >> c;
  vector<long long> v;
  for (long long i = 1; i <= 81; i++) {
    long long x = b * binpow(i, a) + c;
    if (sum(x) == i && x < 1000000000) {
      v.push_back(x);
    }
  }
  cout << v.size() << '\n';
  for (long long x : v) {
    cout << x << ' ';
  }
  return 0;
}
