import sys
input = sys.stdin.readline
n = int(input())
s = list(map(int, input().split()))
s.sort()
q = int(input())
qrs = []
for i in range(q):
	(l, r) = map(int, input().split())
	qrs.append((r - l, i))
qrs.sort()
stoppers = sorted([s[i] - s[i - 1] for i in range(1, len(s))])
stopper_sum = 0
stopper_cnt = 0
si = 0
answers = []
for (x, i) in qrs:
	while si < len(stoppers) and stoppers[si] <= x:
		stopper_cnt += 1
		stopper_sum += stoppers[si]
		si += 1
	answers.append((i, (x + 1) * (len(s) - stopper_cnt) + stopper_sum))
answers.sort()
sys.stdout.write(' '.join((str(x[1]) for x in answers)))
