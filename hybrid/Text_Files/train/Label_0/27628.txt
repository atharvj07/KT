#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, ans = 0;
  cin >> n;
  int arr[n * n];
  for (int i = 0; i < n * n; i++) {
    cin >> arr[i];
  }
  for (i = ((n * n) / 2) - (n / 2); i <= ((n * n) / 2) + (n / 2); i++) {
    if (arr[i] != -1) {
      ans = ans + arr[i];
      arr[i] = -1;
    }
  }
  for (i = 0; i < n * n; i = i + (n + 1)) {
    if (arr[i] != -1) {
      ans = ans + arr[i];
      arr[i] = -1;
    }
  }
  for (i = n - 1; i < n * n - 1; i = i + (n - 1)) {
    if (arr[i] != -1) {
      ans = ans + arr[i];
      arr[i] = -1;
    }
  }
  for (i = n / 2; i < n * n - 1; i = i + n) {
    if (arr[i] != -1) {
      ans = ans + arr[i];
      arr[i] = -1;
    }
  }
  cout << ans << endl;
  return 0;
}
