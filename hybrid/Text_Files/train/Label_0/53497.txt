import java.util.*;
import java.io.*;

public class Main {

    static int mod = 998244353;
    static int add(int a, int b) {
        int c = a + b;
        if (c >= mod) return c - mod;
        if (c < 0) return c + mod;
        return c;
    }
    static int mult(int a, int b) {
        return (int) ((long) a * b % mod);
    }
    static int mult(int... as) {
        int r = 1;
        for (int a: as) r = mult(r, a);
        return r;
    }
    static int[] fact;
    static int[] invFact;
    static int pow(int a, int b) {
        int r = 1;
        while (b != 0) {
            if (b % 2 == 1) r = mult(r, a);
            a = mult(a, a);
            b >>= 1;
        }
        return r;
    }

    static void pre() {
        int N = 200001;
        fact = new int[N];
        invFact = new int[N];
        fact[0] = 1;
        for (int i = 1; i < N; i++) {
            fact[i] = mult(fact[i - 1], i);
        }
        for (int i = 0; i < N; i++) {
            invFact[i] = pow(fact[i], mod - 2);
        }
    }

    static int nck(int n, int k) {
        if (k > n) return 0;
        return mult(fact[n], invFact[n - k], invFact[k]);
    }

    static int halfBinomialSum(int n) { // sum_{2 * k <= n - 1} {n \choose k}
        int r = pow(2, n);
        int inv2 = pow(2, mod - 2);
        if (n % 2 == 0) {
            r = add(r, -nck(n, n / 2));
        }
        return mult(r, inv2);
    }

    public static void main(String[] args) throws IOException{
        pre();
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int k = sc.nextInt();
        int[] arr = new int[N];
        for (int i = 0; i < N; i++) {
            arr[i] = sc.nextInt();
        }
        int n = N;
        for (int i = 0; i < N - 1; i++) {
            if (arr[i] == arr[i + 1]) n--;
        }
        if (arr[0] == arr[N - 1]) n--;
        int r = 0;
        for (int i = 0; i <= n; i++) {
            r = add(r, mult(nck(n, i), pow(k - 2, i), halfBinomialSum(n - i)));
        }
        r = mult(r, pow(k, N - n));
        System.out.println(r);
    }
    public static class Scanner {
        StringTokenizer st;
        BufferedReader br;
        Scanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }
        String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }
        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }
}
