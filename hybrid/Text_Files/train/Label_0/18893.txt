// using meet in the meddle method for maximum independent set problem O(m*1.414^(m))


   #include <bits/stdc++.h>

  #define inf  1000000007
  #define  mp make_pair
  #define  pb push_back
  #define  pa pair<int,int> 
  using namespace std;



bool ok[1<<25];
bool miok[1<<25];
int an[1<<25];
int dp[1<<25];


struct MaxIndep{ 
	private:
	public:

	int hi,mi;
	vector<pa> e1,e2,e3;
 
	void makeok(){
		for(int i=0;i<(1<<hi);i++)ok[i]=true;
	
		for(auto t:e1){
			ok[(1<<t.first)+(1<<t.second)]=false;
		}
	
		for(int i=1;i<(1<<hi);i++){
			bool b=ok[i];
		
			for(int j=0;j<hi;j++){
				if(0==(i&(1<<j)))continue;
				if(ok[i^(1<<j)]==false){
					b=false;
					break;
				}
			}
			ok[i]=b;
		}
		return;
	
	}
 
	void makean(){
		for(int i=0;i<(1<<hi);i++)an[i]=(1<<(mi))-1;
	
		for(auto t:e3){
			an[(1<<t.first)]^= (1<<(t.second-hi));
		}
	
		for(int i=0;i<(1<<hi);i++){
		
			for(int j=0;j<hi;j++){
				if(0==(i&(1<<j)))continue;
				an[i]=an[1<<j]&an[i-(1<<j)];
				break;
			}
		}
	
	}
 
	void makedp(){
		for(int i=0;i<(1<<mi);i++)miok[i]=true;
	
		for(auto t:e2){
			miok[(1<<t.first)+(1<<t.second)]=false;
		}
	
		for(int i=1;i<(1<<mi);i++){
			bool b=miok[i];
		
			for(int j=0;j<mi;j++){
				if(0==(i&(1<<j)))continue;
				if(miok[i-(1<<j)]==false){
					b=false;
					break;
				}
			}
			miok[i]=b;
		}
	
		for(int i=0;i<(1<<mi);i++){
			if(miok[i]){
				int cnt=0;
				cnt=__builtin_popcount(i);
				dp[i]=cnt;
			}
			else dp[i]=0;
		}
	
		for(int i=0;i<(1<<mi);i++){
			for(int j=0;j<mi;j++){
				if(0==(i&(1<<j)))continue;
				dp[i]=max(dp[i],dp[i-(1<<j)]);
			}
		
		}
	}


	int solve(vector<vector<int>> ve){
		int n=ve.size();
		if(n==1) return 1;
		
		 hi=n/2;
   	 	 mi=n-hi;
		
		
		
		for(int i=0;i<n;i++)for(int j=i+1;j<n;j++)if(ve[i][j]){
    		if(j<hi)e1.pb(mp(i,j));
    		else if(i>=hi)e2.pb(mp(i-hi,j-hi));
    		else e3.pb(mp(i,j));
		}
	
	    makeok();
   		makean();
    	makedp();
    	
	    int ans=0;
	    	
	    for(int i=0;i<(1<<hi);i++)if(ok[i]){
	    	int cnt=0;
	    	for(int j=0;j<hi;j++)if(i&(1<<j))cnt++;
	    	
	    	ans=max(ans,cnt+dp[an[i]]);
	    }
		return ans;
	}


	
};


vector<int> a[60];

bool dame(int n,int x,int y){
	
	vector<pa> V;
	for(int i=0;i<n;i++)V.pb({a[x][i],a[y][i]});
	sort(V.begin(),V.end());
	for(int i=0;i<n-1;i++)if(V[i].second>V[i+1].second)return 0;
	return 1;
	
}

void SOL(){
 	int n,m;
 	cin>>n>>m;
	if(n==0 && m==0) exit(0);
	for(int i=0;i<m;i++)a[i].clear();
 	for(int i=0;i<n;i++)for(int j=0;j<m;j++){
 		int y;
 		cin>>y;
 		a[j].pb(y);
 	}
 	vector<vector<int>> ve(m,vector<int>(m,0));
 	
 	for(int i=0;i<m;i++)for(int j=i+1;j<m;j++){
 		if(!dame(n,i,j))ve[i][j]=1;
 	}
 	
 	 MaxIndep md;
 	cout<<md.solve(ve)<<endl;
}

 signed main(){
 	
 
    	       cin.tie(0);
   			ios::sync_with_stdio(false);

 	while(1)SOL();
 	return 0;
 
  }
