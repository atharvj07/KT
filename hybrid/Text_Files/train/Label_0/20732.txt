#include "bits/stdc++.h"
using namespace std;

int N, M;
vector<int>as;
vector<int>bs;
vector<int> getans(const int l, const int r,const int depth) {
	auto lit = upper_bound(as.begin(), as.end(),l);
	auto rit = lower_bound(as.begin(), as.end(), r);
	if (lit == rit) {
		int rank = bs[prev(lit) - as.begin()];
		vector<int>nums(N+1);
		int rest = (r - l) / 2;
		int ok = 0;
		for (int i = N; i >=0; --i) {
			nums[i] = rest;
			if (i == rank)ok = rest;
			rest /= 2;
		}
		vector<int>anss(N + 1);
		for (int i = 0; i <= depth; ++i) {
			anss[i] = ok+((i==rank)?1:0);
		}
		return anss;
	}
	else {
		auto lans(getans(l, (l + r) / 2, depth + 1));
		auto rans(getans((l + r) / 2, r, depth + 1));
		vector<int>anss(N + 1);
		for (int i = 0; i <=depth; ++i) {
			anss[i] = max(anss[i], lans[i] + rans[depth+1]);
		}
		for (int i = 0; i <=depth; ++i) {
			anss[i] = max(anss[i], rans[i] + lans[depth+1]);
		}
		return anss;
	}
}
int main() {cin >> N >> M;
	as.resize(M + 1);
	bs.resize(M);
	for (int i = 0; i <= M; ++i) {
		cin >> as[i];
	}
	for (int i = 0; i < M; ++i) {
		cin >> bs[i];
	}
	auto v= getans(0, (1 << N),0);
	cout << (1<<N)-v[0] << endl;
	return 0;
}
