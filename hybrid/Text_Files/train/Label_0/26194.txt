#include <bits/stdc++.h>
using namespace std;
const long long mod = 998244353;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<int> c(m);
  vector<int> w;
  for (int i = 0; i < m; i++) {
    cin >> c[i];
    c[i]--;
    if (!i || c[i] != c[i - 1]) w.push_back(i);
  }
  if (w.size() > n * 2) {
    cout << "0\n";
    return 0;
  }
  m = w.size();
  vector<vector<int> > o(n);
  for (int i = 0; i < m; i++) {
    o[c[w[i]]].push_back(i);
  }
  vector<vector<long long> > dp(m + 1, vector<long long>(m + 1, 1));
  for (int len = 1; len <= m; len++) {
    for (int l = 0; l + len <= m; l++) {
      int r = l + len;
      int mini = 1e9;
      for (int i = l; i < r; i++) mini = min(mini, c[w[i]]);
      int o1 = o[mini][0], o2 = o[mini].back();
      if (o1 < l || o2 >= r) {
        dp[l][r] = 0;
        continue;
      }
      long long left = 0, right = 0;
      for (int i = l; i <= o1; i++) {
        left += (dp[l][i] * dp[i][o1]) % mod;
        if (left >= mod) left -= mod;
      }
      for (int i = o2 + 1; i <= r; i++) {
        right += (dp[o2 + 1][i] * dp[i][r]) % mod;
        if (right >= mod) right -= mod;
      }
      dp[l][r] = (left * right) % mod;
      for (int i = 0; i < o[mini].size() - 1; i++) {
        dp[l][r] = (dp[l][r] * dp[o[mini][i] + 1][o[mini][i + 1]]) % mod;
      }
    }
  }
  cout << dp[0][m] << "\n";
  return 0;
}
