T = int(input())
for _ in range(T):
	(n, a, b) = map(int, input().split())
	if a + b > n - 2 or abs(a - b) > 1:
		print(-1)
		continue
	if a == b + 1:
		(lo, hi) = (1, n)
		for i in range(a):
			print(lo, end=' ')
			print(hi, end=' ')
			lo += 1
			hi -= 1
		while hi >= lo:
			print(hi, end=' ')
			hi -= 1
		print()
	elif b == a + 1:
		(lo, hi) = (1, n)
		for i in range(b):
			print(hi, end=' ')
			print(lo, end=' ')
			lo += 1
			hi -= 1
		while hi >= lo:
			print(lo, end=' ')
			lo += 1
		print()
	else:
		(lo, hi) = (1, n)
		for i in range(a):
			print(lo, end=' ')
			print(hi, end=' ')
			lo += 1
			hi -= 1
		print(lo, end=' ')
		lo += 1
		while hi >= lo:
			print(lo, end=' ')
			lo += 1
		print()
