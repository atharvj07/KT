//package CodeForces;
import java.io.*;
import java.util.Arrays;
import java.util.Comparator;
import java.util.HashMap;

import static java.lang.Math.*;
import static java.lang.Integer.*;
import static java.lang.Long.*;
@SuppressWarnings("unused")
public class round121D {
static class square{
    int idx;
    int need;
    public square(int i, int n){
        idx = i;
        need = n;
    }
}
    public static void main(String[] args)throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String [] use = null;
        use = br.readLine().split(" ");
        int n = parseInt(use[0]);
        int k = parseInt(use[1]);
        long b = parseLong(br.readLine());
        use = br.readLine().split(" ");
        square [] all = new square [n];
        for(int i = 0 ; i < n ; ++i)
            all[i] = new square(i, parseInt(use[i]));
        Arrays.sort(all, new Comparator<square>() {
            public int compare(square s, square t){
                if(s.need > t.need)
                    return -1;
                if(s.need < t.need)
                    return 1;
                return 0;
            }
        });
        int last_pos = 0;
        HashMap<Integer, Integer> mp = new HashMap<Integer, Integer>();
        for(int i = 0 ; i < n ; ++i){
            mp.put(all[i].idx, i);
            if(all[i].idx == n - 1)
                last_pos = i;
        }
//      for(int i = 0 ; i < all.length ; ++i)
//          System.out.println(all[i].idx + " "  + all[i].need);
        long [] cum_sum = new long [n];
        cum_sum[0] = all[0].idx == n - 1 ? 0 : all[0].need;
        for(int i = 1 ; i < n ; ++i){
            cum_sum[i] = cum_sum[i - 1] + ((all[i].idx == n - 1) ? 0 : all[i].need);
        }
        for(int i = 0 ; i < n - 1 ; ++i){// try greedily to get the best square
            int pos = mp.get(i);
            //System.out.println("pos: " + pos);
            long sum = 0;
            if((k - 1) <= pos){
                sum = k - 2 < 0 ? 0 : cum_sum[k - 2];
                if(last_pos <= k - 2){
                    if(k - 1 != pos)
                        sum += all[k - 1].need;
                    else
                        sum += all[k].need;
                }
            }
            else{
                sum = pos == 0 ? 0 : cum_sum[pos - 1];
                int taken = pos;
                int rem = (k - 1) - taken;
                sum += (cum_sum[pos + rem] - cum_sum[pos]);
                if(last_pos <= pos + rem){
                    if(pos + rem + 1 < n && pos + rem + 1 != pos)
                        sum += all[pos + rem + 1].need;
                    else
                        if(pos + rem + 2 < n)
                            sum += all[pos + rem + 1].need;
                }
                    
                    
            }
//          System.out.println("sum: " + sum);
//          System.out.println();
            if(sum > b || (b - sum) < all[pos].need){
                System.out.println(i + 1);
                return;
            }
        }
        System.out.println(n);
    }
}
