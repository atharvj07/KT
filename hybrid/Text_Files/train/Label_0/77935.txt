n = int(input())
pos = {}
for _ in range(n):
	(l, r) = [int(x) for x in input().split()]
	if l in pos:
		pos[l] += 1
	else:
		pos[l] = 1
	if r + 1 in pos:
		pos[r + 1] -= 1
	else:
		pos[r + 1] = -1
a = sorted(pos.items(), key=lambda x: x[0])
res = [0] * n
s = 0
for i in range(len(a)):
	if 1 <= s <= n:
		res[s - 1] += a[i][0] - a[i - 1][0]
	s += a[i][1]
print(*res)
