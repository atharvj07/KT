import sys

def main():
	n = int(input())
	currently_in = set()
	capacity = 0
	for _ in range(n):
		(event_type, id) = input().split()
		id = int(id)
		if event_type == '+':
			currently_in.add(id)
			if len(currently_in) > capacity:
				capacity = len(currently_in)
		elif id in currently_in:
			currently_in.remove(id)
		else:
			capacity += 1
	print(capacity)
main()
