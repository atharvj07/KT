import java.util.*;
import javafx.util.Pair;

public final class cont719A
{
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int T=sc.nextInt();
        while(T>0)
        {
            int n=sc.nextInt();
            //String st=Integer.toString();
            System.out.println(count_same_digit(n));
            T--;
        }
        
    }

    static int count_same_digit(int R)
{
    int tmp = 0, ans = 0;
  
    // Length of R
    int n = (int)Math.log10(R) + 1;
    String sr=Integer.toString(R);
    //System.out.println(sr);
    int x=sr.charAt(0)-'0';
    //System.out.println(x);

    ans+=(9*(n-1));
    for(int i=0;i<n;i++)
        tmp=(10*tmp)+x;
    //System.out.println(tmp);
    ans+=(R/(int)Math.pow(10,n-1));
    if(R<tmp)
    {

        ans=ans-1;
    }
    return ans;
}

    public void sortPairs(List<Pair<Integer,Integer>> lst)
    {
        Collections.sort(lst,(a,b)->a.getKey()-b.getKey());
    }

    public static int log2(int N)
    {
  
        // calculate log2 N indirectly
        // using log() method
        int result = (int)(Math.log(N) / Math.log(2));
  
        return result;
    }

    static long sum(long x)
    {
        long sum=0;
        while(x>0)
        {
            sum+=x%10;
            x=x/10;
        }
        return sum;

    }
    static long gcd(long a, long b)
    {
      if (b == 0)
        return a;
      return gcd(b, a % b);
    }

}
