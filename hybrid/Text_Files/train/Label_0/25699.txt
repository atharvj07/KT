#include <bits/stdc++.h>
using namespace std;
template <class T>
inline void read(T &k) {
  k = 0;
  int flag = 1;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') flag = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') {
    k = k * 10 + c - '0';
    c = getchar();
  }
  k *= flag;
}
const int N = 200005;
struct node {
  int x, y;
  bool operator<(const node &k) const { return y > k.y; }
} a[N];
int n, m, k, s, t, ans, head[N];
int b[N], tot, c[N], totc;
int main() {
  read(n);
  for (int i = 1; i < n; ++i) {
    read(a[i].x);
    read(a[i].y);
    if (a[i].y == n) {
      swap(a[i].x, a[i].y);
    }
    if (a[i].x != n) {
      puts("NO");
      return 0;
    }
  }
  sort(a + 1, a + n);
  int l = 1;
  for (int i = n - 1; i >= 1; --i) {
    k = 0;
    while (l <= n - 1 && a[l].y == i) {
      ++k;
      ++l;
    }
    if (k == 0)
      c[++totc] = i;
    else {
      b[i] = k;
    }
  }
  for (int i = 1; i < n; ++i) tot += max(0, b[i] - 1);
  if (tot != totc) {
    puts("NO");
    return 0;
  }
  tot = 0;
  for (int i = 1; i <= n - 1; ++i) {
    if (b[i] == 0)
      ++tot;
    else if (tot >= b[i] - 1) {
      tot -= b[i] - 1;
    } else {
      puts("NO");
      return 0;
    }
  }
  puts("YES");
  l = 1;
  tot = 0;
  sort(c + 1, c + 1 + totc);
  for (int i = 1; i < n; ++i) {
    if (b[i]) {
      --b[i];
      int now = n;
      for (int j = l; j <= l + b[i] - 1; ++j) {
        cout << now << ' ' << c[j] << endl;
        now = c[j];
      }
      cout << now << ' ' << i << endl;
      l += b[i];
    }
  }
}
