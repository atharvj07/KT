// practice with rainboy
import java.io.*;
import java.util.*;

public class CF584E extends PrintWriter {
	CF584E() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF584E o = new CF584E(); o.main(); o.flush();
	}

	int[] pp;
	int[] ii, jj; int m;
	void swap(int i, int j) {
		int tmp = pp[i]; pp[i] = pp[j]; pp[j] = tmp;
		ii[m] = i; jj[m] = j; m++;
	}
	void main() {
		int n = sc.nextInt();
		m = n * (n - 1) / 2;
		ii = new int[m]; jj = new int[m];
		m = 0;
		pp = new int[n];
		for (int i = 0; i < n; i++)
			pp[i] = sc.nextInt() - 1;
		int[] inv = new int[n];
		for (int i = 0; i < n; i++) {
			int s = sc.nextInt() - 1;
			inv[s] = i;
		}
		int c = 0;
		for (int i = 0; i < n; i++) {
			pp[i] = inv[pp[i]];
			c += Math.abs(pp[i] - i);
		}
		for (int h = 1; h < n; h++)
			for (int j = h, i = j - 1; i >= pp[j]; i--)
				if (pp[i] >= j) {
					swap(i, j);
					j = i;
				}
		println(c / 2);
		println(m);
		for (int h = 0; h < m; h++)
			println((ii[h] + 1) + " " + (jj[h] + 1));
	}
}
