(a1, a2, a3, a4) = list(map(int, input().split()))
if a1 + a2 == a3 + a4:
	print('YES')
elif a1 + a3 == a2 + a4:
	print('YES')
elif a1 + a4 == a2 + a3:
	print('YES')
elif a1 == a2 + a3 + a4:
	print('YES')
elif a2 == a1 + a3 + a4:
	print('YES')
elif a3 == a2 + a1 + a4:
	print('YES')
elif a4 == a2 + a3 + a1:
	print('YES')
else:
	print('NO')
