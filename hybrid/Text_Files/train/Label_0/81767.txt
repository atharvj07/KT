import math
import sys
(n, x1, y1, x2, y2) = [int(x) for x in input().split()]
flowers = []
r1 = 0
r2 = 0

def distanceQ(d1, d2):
	return d1 ** 2 + d2 ** 2

def ditance_sqr(p1, p2):
	return distanceQ(p1[0] - p2[0], p1[1] - p2[1])
for i in range(n):
	flowerpoint = tuple([int(x) for x in input().split()])
	flowers.append({'point': flowerpoint, 'd1': ditance_sqr((x1, y1), flowerpoint), 'd2': ditance_sqr((x2, y2), flowerpoint)})
flw = sorted(flowers, key=lambda t: t['d1'], reverse=True)
rt1 = 0
flw2 = max(flw, key=lambda t: t['d2'])
minf = rt1 + flw2['d2']
for (i, flower) in enumerate(flw):
	flowerpoint = flower['point']
	rt1 = flower['d1']
	rt2 = 0
	distance_flw = flw[:i]
	if len(distance_flw):
		flw2 = max(distance_flw, key=lambda t: t['d2'])
		rt2 = flw2['d2']
	rq = rt1 + rt2
	if rq < minf:
		minf = rq
print(round(minf))
