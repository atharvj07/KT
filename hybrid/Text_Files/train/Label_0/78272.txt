def readGraph(n, m):
	adj = [set() for _ in range(n)]
	for _ in range(m):
		(u, v) = map(int, input().split())
		adj[u - 1].add(v - 1)
		adj[v - 1].add(u - 1)
	return adj

def readTree(n):
	return readGraph(n, n - 1)

def solve():
	(n, k) = map(int, input().split())
	adj = readTree(n)
	if k == 1:
		return (n - 1, n, k)
	leaves = {}
	lc = [0] * n
	for i in range(n):
		if len(adj[i]) == 1:
			lc[adj[i].pop()] += 1
			lc[i] = -1
	st = [i for i in range(n) if lc[i] >= k]
	res = 0
	while st:
		i = st.pop()
		res += lc[i] // k
		if len(adj[i]) == lc[i] + 1 and lc[i] % k == 0:
			for a in adj[i]:
				if lc[a] != -1:
					lc[a] += 1
					if lc[a] == k:
						st.append(a)
			lc[i] = -1
		else:
			lc[i] %= k
	return (res, n, k)
