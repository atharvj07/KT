from sys import stdin, stdout
from math import *
import itertools
n = int(input())
a = [int(i) for i in input().split()]
a.sort()

def check(x):
	k = x
	for i in range(n):
		k -= max(x - a[i], 0)
		if k <= 0:
			break
	return k <= 0
lb = a[-1]
ub = a[0] + a[-1] + 1
while lb + 1 < ub:
	tx = (lb + ub) // 2
	dx = (ub - lb) // 2
	if check(tx):
		ub -= dx
	else:
		lb += dx
tx = (lb + ub) // 2
for i in range(2):
	if tx > a[-1] and check(tx - 1):
		tx -= 1
	elif not check(tx):
		tx += 1
print(tx)
