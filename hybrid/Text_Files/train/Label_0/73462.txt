import sys
(*data,) = sys.stdin.read().split('\n')[::-1]

def input():
	return data.pop()

def fprint(*args, **kwargs):
	print(*args, **kwargs, flush=True)

def eprint(*args, **kwargs):
	print(*args, **kwargs, file=sys.stderr)
for _ in range(int(input())):
	n = int(input())
	(*arr,) = map(int, input().split())
	color = 0
	mx = 1
	if len(set(arr)) == 1:
		print(1)
		print(*[1] * n)
	elif n % 2 == 0:
		print(2)
		print(*[1, 2] * (n // 2))
	else:
		colors = [0] * n
		i = 0
		is2 = False
		color = 0
		while i < n:
			if arr[i] != arr[(i + 1) % n] or is2:
				colors[i] = color + 1
				i += 1
			else:
				colors[(i + 1) % n] = colors[i] = color + 1
				is2 = True
				i += 2
			color = (color + 1) % 2
		if not is2:
			colors[-1] = 3
		print(max(colors))
		print(*colors)
