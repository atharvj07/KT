#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  long long a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  long long b[2 * n];
  for (int i = 0, k = 0; i < 2 * n; i++, k += 2) {
    b[i] = a[k % n];
  }
  long long sum = 0, maxsum, x = (n + 1) / 2;
  for (int i = 0; i < x; i++) sum += b[i];
  maxsum = sum;
  for (int i = 0; i < 2 * n - x; i++) {
    sum += b[x + i] - b[i];
    maxsum = max(sum, maxsum);
  }
  cout << maxsum;
  return 0;
}
