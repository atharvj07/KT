#include<bits/stdc++.h>
using namespace std;
int main(){
	ios::sync_with_stdio(false);
	int t;cin>>t;
	while(t--){
		int x,y,z;cin>>x>>y>>z;
		int a,b,c,d,e;cin>>a>>b>>c>>d>>e;
		if(a+b+c+d+e>x+y+z)cout<<"NO"<<endl;
		else if(a>x||b>y||c>z)cout<<"NO"<<endl;
		else{
			x-=a;
			y-=b;
			z-=c;
			if(x>d)d=0;
			else d-=x;
			if(y>e)e=0;
			else e-=y;
			if(d+e>z)cout<<"NO"<<endl;
			else cout<<"YES"<<endl;
		}
	}
}
