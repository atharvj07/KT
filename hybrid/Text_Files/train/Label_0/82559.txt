Q = int(input())
for something in range(Q):
	n = int(input())
	list1 = []
	count = 0
	even = 0
	flag = 0
	for _ in range(n):
		list1.append(input())
	for string in list1:
		ones = 0
		if len(string) % 2 == 0:
			for char in string:
				if char == '1':
					ones += 1
			if ones % 2 == 0:
				count += 1
			else:
				even += 1
		else:
			count += 1
			flag += 1
	if flag:
		print(n)
	else:
		print(count + even // 2 * 2)
