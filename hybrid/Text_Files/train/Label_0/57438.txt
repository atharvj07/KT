import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.OutputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;

 

public class Main {
	

	static String s=null;
	static int length=-1;
	static int INF= 10000000;
	static long MOD= (long) (1e9+7);
	
	static long[][]combination= new long[1005][1005];
	static long []dp= new long[1005];
	//static long [][][]dpMain= new long[1005][1005][2];


	static class InputReader {
		public BufferedReader reader;
		public StringTokenizer tokenizer;

		public InputReader(InputStream stream) {
			reader = new BufferedReader(new InputStreamReader(stream), 32768);
			tokenizer = null;
		}

		public String next() {
			while (tokenizer == null || !tokenizer.hasMoreTokens()) {
				try {
					tokenizer = new StringTokenizer(reader.readLine());
				} catch (IOException e) {
					throw new RuntimeException(e);
				}
			}
			return tokenizer.nextToken();
		}

		public int nextInt() {
			return Integer.parseInt(next());
		}

	}
	public static int countSetBits(long number){
		int count = 0; 
		while(number>0){
			++count; number &= number-1; 
		} return count; 
	}
	
	public static long rec(int n, int r){
		if(r>n||n<0||r<0)
			return 0;
		if(n==0||r==0){
			combination[n][r]=1;
			return 1;
		}
		if(combination[n][r]!=-1)
			return combination[n][r];
		
		long v= rec(n-1,r-1)+rec(n-1,r);
		combination[n][r]= v%MOD;
		return combination[n][r];
	}

	static long go(int index,int setBits,boolean before){
		if(index==length+1){
			if(setBits==0)
				return 1;
			else
				return 0;
		}
		if(index>length || setBits>(length-index+1))
			return 0;
		long ans=0;
		if(before)
			ans=rec(length-index+1,setBits );
		else{
			if(s.charAt(index)=='0'){
				ans= go(index+1, setBits, false);
			}
			else{
				long v1= go(index+1, setBits, true);
				long v2= go(index+1, setBits-1, false);
				
				ans=(v1+v2)%MOD;
			}
		}
		return ans%MOD;
	}
	static class TaskA {
		public void solve(int testNumber, InputReader in, PrintWriter out) {

			s= in.next();
			int k= in.nextInt();
			if(k==0){
				out.println(1);
				return;
			}
			length= s.length();
			s="#"+s;
			for(int i=0;i<=1000;i++){
				for(int j=0;j<=1000;j++)
					combination[i][j]=-1;
			}
			dp[1]=0;
			for(int i=2;i<=1002;i++){
				int x=countSetBits(i);
				dp[i]=dp[x]+1;
			}
			
			long result=0;
			for(int bit=1;bit<=1000;bit++){
				if(dp[bit]+1==k){
					long x= go(1,bit,false);
					if(k==1 && bit==1&& x>0)
						x--;
					//System.out.println("set bit:"+bit+" res:"+x);
					result=(result+x)%MOD;
				}
			}
			out.println(result);
		}



	}

	

	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		InputReader in = new InputReader(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		TaskA solver = new TaskA();
		solver.solve(1, in, out);
		out.close();
	}
	

}
