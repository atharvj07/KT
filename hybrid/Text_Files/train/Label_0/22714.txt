//拡張ユークリッド
//ax + by = 1　を満たす x, yを求めよ
//gcd(a, b) != 1　の時は解が存在しない(バグる)
#include<iostream>
using namespace std;

int gcd(int a, int b){

    if(b == 0) return a;
    else return gcd(b, (a % b));
}

int extgcd(int  a, int b, int& x, int& y){
    int d = a;
    if(b != 0){
        d = extgcd(b, a % b, y, x);
        y -= (a / b) * x;
    }else{
        x = 1; y = 0;
    }
    return d;
}

int main(){

    //ax + by = gcd(a, b)の解を求める
    int a, b; cin >> a >> b;
    int x, y;
    int d = gcd(a, b);
    a /= d;
    b /= d;
    d = extgcd(a, b, x, y);
    cout << x << " " << y <<  endl;
    return 0; 
}
