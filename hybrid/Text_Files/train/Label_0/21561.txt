#include <bits/stdc++.h>
using namespace std;
const int LEN = 2201000, LEN1 = 21, INF = 1e9;
char ch;
int n, m, ar[LEN], dp[LEN1][LEN], nxt, res, dp1[LEN], dp2[LEN];
int getmin(int mask) {
  int res = 0;
  for (int i = 0; i <= n; i++)
    res += min(i * dp[i][mask], (n - i) * dp[i][mask]);
  return res;
}
int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    scanf("%c", &ch);
    for (int j = 0; j < m; j++) {
      scanf("%c", &ch);
      ar[j] = ar[j] * 2 + int(ch - '0');
    }
  }
  for (int i = 0; i < m; i++) {
    dp[0][ar[i]]++;
    dp2[ar[i]]++;
  }
  for (int i = 1; i <= n; i++)
    for (int j = 0; j < (1 << n); j++) {
      for (int p = 0; p < n; p++) {
        nxt = (j ^ (1 << p));
        if ((i - 1) % 2 == 0)
          dp[i][j] += dp2[nxt] - dp1[j];
        else
          dp[i][j] += dp1[nxt] - dp2[j];
      }
      dp[i][j] /= i;
      if (i % 2)
        dp1[j] += dp[i][j];
      else
        dp2[j] += dp[i][j];
    }
  res = INF;
  for (int i = 0; i < (1 << n); i++) res = min(res, getmin(i));
  cout << res;
  return 0;
}
