from typing import List, Tuple

def parse_input() -> Tuple[int, List[Tuple[int, int]]]:
	n = int(input())
	arr = []
	for _ in range(n):
		(ai, bi) = [int(x) for x in input().split()]
		arr.append((ai, bi))
	return (n, arr)

def solve(n: int, ls: List[Tuple[int, int]]) -> str:
	for (ai, bi) in ls:
		if ai != bi:
			return 'Happy Alex'
	return 'Poor Alex'
(n, ls) = parse_input()
print(solve(n, ls))
