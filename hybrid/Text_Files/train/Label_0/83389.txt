n = int(input())
A = []
for i in range(n):
	B = list(input().split())
	if B[1] == 'North':
		A.append(-1 * int(B[0]))
	elif B[1] == 'South':
		A.append(int(B[0]))
	else:
		A.append('r')
del B
(flag, cord) = ('YES', 0)
for i in A:
	if i == 'r':
		if cord <= 0 or cord >= 20000:
			flag = 'NO'
			break
		else:
			continue
	if cord < 0:
		flag = 'NO'
		break
	elif cord > 20000:
		flag = 'NO'
		break
	cord += i
if cord == 0:
	print(flag)
else:
	print('NO')
