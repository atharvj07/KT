(n, k) = map(int, input().split())
p = [0 for i in range(n + 1)]
d = p.copy()
for i in range(k):
	a = list(map(int, input().split()))
	l = len(a)
	for j in range(1, l - 1):
		p[a[j]] = a[j + 1]
		d[a[j + 1]] = 1
c = 1
while p[c] == c + 1:
	c = p[c]
ans = 0
for i in range(c + 1, n + 1):
	ans += 1 + d[i]
print(ans)
