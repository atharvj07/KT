public class Solution {

    public static final int N = 20;

    boolean[] b = new boolean[1 << N];
    int[] subTreeMax = new int[1 << N];
    boolean[] visited = new boolean[1 << N];

    void solve() throws Throwable {
        char[] a = readString().toCharArray();
        for (int start = 0; start < a.length; start++) {
            int mask = 0;
            for (int i = 0; i < 20; i++) {
                int curInd = start + i;
                if (curInd >= a.length) {
                    break;
                }
                int ch = a[curInd] - 'a';
                if ((mask | (1 << ch)) == mask) {
                    break;
                }
                mask |= (1 << ch);
                b[mask] = true;
            }
        }
        rec((1 << 20) - 1);
        int ans = 0;
        for (int mask = 0; mask < (1 << 20); mask++) {
            ans = Math.max(ans, Integer.bitCount(subTreeMax[mask]) + Integer.bitCount(subTreeMax[((1 << 20) - 1) ^ mask]));
        }
        System.out.println(ans);
    }

    private void rec(int mask) {
        if (visited[mask]) {
            return;
        }
        visited[mask] = true;
        if (b[mask]) {
            subTreeMax[mask] = mask;
        }
        for (int i = 0; i < N; i++) {
            if ((mask & (1 << i)) != 0) {
                int to = mask & ~(1 << i);
                rec(to);
                if (Integer.bitCount(subTreeMax[to]) > Integer.bitCount(subTreeMax[mask])) {
                    subTreeMax[mask] = subTreeMax[to];
                }
            }
        }
    }

    //-------------------------------------------------

    final boolean ONLINE_JUDGE = !new File("input.txt").exists();

    BufferedReader in;
    PrintWriter out;
    StringTokenizer tok;

    public void run() {
        Runnable run = () -> {
            try {
                long startTime = System.currentTimeMillis();
                Locale.setDefault(Locale.US);
                if (ONLINE_JUDGE) {
                    in = new BufferedReader(new InputStreamReader(System.in));
                    out = new PrintWriter(System.out);
                } else {
                    in = new BufferedReader(new FileReader("input.txt"));
                    out = new PrintWriter("output.txt");
                }
                tok = new StringTokenizer("");
                solve();
                in.close();
                out.close();
                long endTime = System.currentTimeMillis();
                long totalMemory = Runtime.getRuntime().totalMemory();
                long freeMemory = Runtime.getRuntime().freeMemory();
                System.err.println();
                System.err.println("Time = " + (endTime - startTime) + " ms");
                //System.err.println("Memory = " + ((totalMemory - freeMemory) / 1024) + " KB");
            } catch (Throwable e) {
                e.printStackTrace(System.err);
                System.exit(-1);
            }
        };
        new Thread(null, run, "run", 256 * 1024 * 1024).start();
        min(0, 0);
    }

    String readString() {
        while (!tok.hasMoreTokens()) {
            String line;
            try {
                line = in.readLine();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
            if (line == null) return null;
            tok = new StringTokenizer(line);
        }
        return tok.nextToken();
    }

    int readInt() {
        return Integer.parseInt(readString());
    }

    long readLong() throws IOException {
        return Long.parseLong(readString());
    }

    double readDouble() throws IOException {
        return Double.parseDouble(readString());
    }

        if (!ONLINE_JUDGE) {
            System.err.println(Arrays.deepToString(o));
        }
    }

    static long gcd(long x, long y) {
        return y == 0 ? x : gcd(y, x % y);
    }

    private long[] readLongArray(int n) throws IOException {
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = readLong();
