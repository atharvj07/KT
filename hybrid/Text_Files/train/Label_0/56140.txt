import java.util.*;

public class Olympics{
	public static double dist(int a, int b){
		return Math.sqrt(a*a + b*b);
	}
	
	public static boolean intersect(int rd, int[] p, int [] q, int r, int R){
		if( dist(p[0]-q[0], p[1]-q[1]) >= rd + R ){
		//	System.out.println("A");
			return true;
		}
		if( dist(p[0]-q[0], p[1]-q[1]) <= r - rd ){
		//	System.out.println("B");
			return true;
			
		}
		if( dist(p[0]-q[0], p[1]-q[1]) <= rd - R ){
		//	System.out.println("C");
			return true;
			
		}
	//	System.out.println(false);
		return false;
	}
	
	public static void main(String[] args){
		Scanner stdIn = new Scanner(System.in);
		int[][] p = new int[2][2];
		int[] r = new int[2];
		int[] R = new int[2];
		int ans = 0;
		for(int i = 0; i < 2; i++){
			p[i][0] = stdIn.nextInt();
			p[i][1] = stdIn.nextInt();
			r[i] = stdIn.nextInt();
			R[i] = stdIn.nextInt();
		}
		if( intersect(r[0], p[0], p[1], r[1], R[1]) ){
			ans++;
		}
		if( intersect(R[0], p[0], p[1], r[1], R[1]) ){
			ans++;
		}
		if( intersect(r[1], p[1], p[0], r[0], R[0]) ){
			ans++;
		}
		if( intersect(R[1], p[1], p[0], r[0], R[0]) ){
			ans++;
		}
		System.out.println(ans);
	}
}
