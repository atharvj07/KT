#include <bits/stdc++.h>
#define INF 1000000000000000001LL
using namespace std;

typedef long long ll;
typedef pair<int,int> P;

ll n,k;

bool C(ll v){
	for(int i=0;i<n;i++){
		if(v<=0)return true;
		v-=((v+k-1LL)/k);
	}
	return false;
}

int main(void){
	scanf("%d%d",&n,&k);
	ll l=0,r=INF;
	for(int i=0;i<100;i++){
		ll mid=(l+r)/2LL;
		if(C(mid))l=mid;
		else r=mid;
	}
	printf("%lld\n",l);
	return 0;
}
