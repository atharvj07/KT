t = int(input())
for test in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	c = 0
	for i in range(1, n):
		tog = 2 * (i + 1)
		if tog / arr[i] <= 1:
			continue
		m = tog % arr[i]
		pos = i
		if m != 0:
			if i - m < 0:
				break
			n1 = (tog - m) // arr[i]
			n2 = arr[i - m]
			if n1 == n2:
				c += 1
			tog -= m
			pos = i - m
		while tog > arr[i]:
			tog -= arr[i]
			pos -= arr[i]
			if pos < 0:
				break
			if tog // arr[i] == arr[pos]:
				c += 1
	print(c)
