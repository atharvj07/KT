t = int(input())
for i in range(t):
	n = int(input())
	lst = [int(i) for i in input().split()]
	(z, s, t) = (0, 0, 0)
	for j in lst:
		if j == 0:
			z += 1
		else:
			s += j
	if z == 0:
		if s == 0:
			print(1)
		else:
			print(0)
	else:
		t = z
		s += z
		if s == 0:
			print(t + 1)
		else:
			print(t)
