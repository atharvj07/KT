import java.util.*;
import java.util.stream.*;

/*
 * Create shirts with ids
 * Create a map with colour keys and heap values (heaped by price)
 * Add shirts to heaps
 * For each customer
 *   If no shirts, return -1
 *   Otherwise cheapest or only shirt
 *   Remove shirt from all heaps
 */
public class TShirtBuying {
  private static class Shirt {
    public int id;
    public int price;
    public int front;
    public int back;
    public boolean purchased;
    
    public Shirt(int id, int p) {
      this.id = id;
      price = p;
    }
    
    public String toString() {
      return "{id: " + id + ", price: " + price + ", front: " + front + ", back: " + back + "}";
    }
    
    public boolean equals(Object other) {
      Shirt o = (Shirt) other;
      return id == o.id;
    }
  }
  
  public static class SortShirtsByPrice implements Comparator<Shirt> {
    public int compare(Shirt a, Shirt b) {
      return a.price - b.price;
    }
  }
  
  public static final int minColour = 1;
  public static final int maxColour = 3;
  
  public static void main (String args[]) {
    Scanner in = new Scanner(System.in);
    List<Shirt> shirts = constructShirts(in.nextInt(), in);
    List<Integer> customers = constructCusts(in.nextInt(), in);
    System.out.println(solve(shirts, customers));
  }
  
  public static String solve(List<Shirt> shirts, List<Integer> custs) {
    Map<Integer, Queue<Shirt>> map = constructMap(shirts);
    //printMap(map);
    List<Integer> purchases = getPurchases(map, shirts, custs);
    //System.out.println(purchases.toString());
    return getOutputStrFormat(purchases);
  }
  
  public static Map<Integer, Queue<Shirt>> constructMap(List<Shirt> shirts) {
    Map<Integer, Queue<Shirt>> toReturn = new HashMap<Integer, Queue<Shirt>>();
    for (int i = minColour; i <= maxColour; i++) {
      toReturn.put(i, new PriorityQueue<Shirt>(11, new SortShirtsByPrice()));
    }
    
    for (Shirt s : shirts) {
      toReturn.get(s.front).offer(s);
      if (s.front != s.back) {
        toReturn.get(s.back).offer(s);
      }
    }
    return toReturn;
  }
  
  private static List<Integer> getPurchases(Map<Integer, Queue<Shirt>> map,
                                            List<Shirt> shirts, List<Integer> custs) {
    List<Integer> res = new ArrayList<Integer>();
    
    for (int cust : custs) {
      Queue<Shirt> favs = map.get(cust);
      if (favs.isEmpty()) {
        res.add(-1);
        continue;
      }
      
      Shirt toBuy = favs.peek();
      res.add(toBuy.price);
      toBuy.purchased = true;
      cleanup(map);
    }
    
    return res;
  }
  
  private static void cleanup(Map<Integer, Queue<Shirt>> map) {
    for (int i = minColour; i <= maxColour; i++) {
      Queue<Shirt> q = map.get(i);
      while (!q.isEmpty() && q.peek().purchased) {
        q.poll();
      }
    }
  }
  
  private static String getOutputStrFormat(List<Integer> purchases) {
    StringBuilder sb = new StringBuilder();
    for (int p : purchases) {
      sb.append(p + " ");
    }
    return sb.toString();
  }
  
  private static List<Shirt> constructShirts(int n, Scanner in) {
    List<Shirt> toReturn = new ArrayList<Shirt>(n);
    for (int i = 0; i < n; i++) {
      toReturn.add(new Shirt(i, in.nextInt()));
    }
    
    for (int i = 0; i < n; i++) {
      toReturn.get(i).front = in.nextInt();
    }
    for (int i = 0; i < n; i++) {
      toReturn.get(i).back = in.nextInt();
    }
    return toReturn;
    
    /*int id = 0;

     id++;
     return t;
     })
     .collect(Collectors.toList());*/
  }
  
  private static List<Integer> constructCusts(int n, Scanner in) {
    List<Integer> toReturn = new ArrayList<Integer>(n);
    for (int i = 0; i < n; i++) {
      toReturn.add(in.nextInt());
    }
    return toReturn;
  }
  
  private static void printMap(Map<Integer, Queue<Shirt>> map) {
    for (int i = minColour; i <= maxColour; i++) {
      System.out.println("Colour: " + i + ", Shirts: " + map.get(i).toString());
    }
  }
}
