#include <bits/stdc++.h>
using namespace std;
const int maxn = 200000 + 10;
int n;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  priority_queue<pair<int, int>, vector<pair<int, int> >,
                 greater<pair<int, int> > >
      pq;
  priority_queue<pair<int, int> > pq2;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int tmp;
    cin >> tmp;
    pq.push(pair<int, int>(tmp, i));
  }
  for (int i = 0; i < 2 * n; i++) {
    char c;
    cin >> c;
    if (c == '0') {
      cout << pq.top().second + 1 << ' ';
      pq2.push(pair<int, int>(pq.top().first, pq.top().second));
      pq.pop();
    } else {
      cout << pq2.top().second + 1 << ' ';
      pq2.pop();
    }
  }
  cout << '\n';
  return 0;
}
