#include <cstdio>
#include <algorithm>
#include <cmath>
#define N 1000070

using namespace std;

inline int read(){
   int s=0,w=1;
   char ch=getchar();
   while(ch<'0'||ch>'9'){if(ch=='-')w=-1;ch=getchar();}
   while(ch>='0'&&ch<='9') s=s*10+ch-'0',ch=getchar();
   return s*w;
}

struct Node{
	int l,r,num;
}q[N];
int n,m,a[N],block,ans,res[N],cnt[N];

bool cmp(Node a,Node b){
	return (a.l/block)^(b.l/block)?a.l<b.l:(((a.l/block)&1)?a.r<b.r:a.r>b.r);
}

void add(int x){
	cnt[x]++;
	if(cnt[x]==1) ans++;
}

void del(int x){
	cnt[x]--;
	if(cnt[x]==0) ans--;
}

int main(void){
	n=read();m=read();
	for(int i=1;i<=n;++i)
		a[i]=read();
	block=n/sqrt(m*2/3);
	for(int i=1;i<=m;++i){
		q[i].l=read();
		q[i].r=read();
		q[i].num=i; 
	}
	std::sort(q+1,q+m+1,cmp);
	int l=0,r=0;
	for(int i=1;i<=m;++i){
		while(l>q[i].l) add(a[--l]);
		while(l<q[i].l) del(a[l++]);
		while(r>q[i].r) del(a[r--]);
		while(r<q[i].r) add(a[++r]);
		res[q[i].num]=ans;
	}
	for(int i=1;i<=m;++i)
		printf("%d\n",res[i]);
}
