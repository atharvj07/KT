(n, m) = input().split()
n = int(n)
m = int(m)
a = list(map(int, input().split()))[:n]
b = list(map(int, input().split()))[:m]
c = []
d = {}
for i in range(m):
	if a.__contains__(b[i]):
		c.append(a.index(b[i]))
		d.setdefault(a.index(b[i]), b[i])
c.sort()
for i in range(len(c)):
	print(d.get(c[i]), end=' ')
