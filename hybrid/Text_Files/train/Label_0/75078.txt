def main(a, b):
	minn = min(a, b)
	if minn == 0:
		return 0
	if minn == 1 and a == b:
		return 0
	if minn == 1:
		return 1
	diff = abs(a - b)
	if max(a, b) > 2 * minn:
		return minn
	a = minn - diff
	return diff + (2 * (a // 3) + a % 3 // 2)
for _ in range(int(input())):
	(a, b) = [int(x) for x in input().split()]
	print(main(a, b))
