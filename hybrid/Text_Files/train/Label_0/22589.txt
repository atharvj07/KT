#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  long long l, v1, v2;
  cin >> n >> l >> v1 >> v2 >> k;
  int p = (n % k == 0) ? (n / k) : (n / k + 1);
  if (p == 1) {
    printf("%.10f\n", (double)l / (double)v2);
    return 0;
  }
  double temp = (double)(v2 - v1) / (double)(v2 + v1);
  double co = (double)(p - 1) * (temp + 1.0);
  double tbus = (double)l / ((double)v1 * co + v2);
  double twalk = co * tbus;
  printf("%.10f\n", tbus + twalk);
}
