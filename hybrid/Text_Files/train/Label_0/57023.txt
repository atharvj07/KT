import java.util.Scanner;

public class Main {
    static long [] en=new long[] {1, 11, 111, 1111, 11111, 111111, 1111111, 11111111, 111111111, 1111111111,
            11111111111L, 111111111111L, 1111111111111L, 11111111111111L, 111111111111111L, 1111111111111111L,11111111111111111L};



    public static void main(String[] args) {
        System.out.println(solve(new Scanner(System.in).nextLong()));
    }
    static long solve(long s){
        if(s<=11){
            return Math.min(s, 2+11-s);
        }
        int dolzina = (int) (Math.log10(s)+1);
        if (Math.abs((en[dolzina])-s)<s ){
            return (Math.min((solve(Math.abs(s-en[dolzina-1]))+ dolzina),(solve((en[dolzina])-s))+(dolzina+1)));
        }

        return solve(Math.abs(s-en[dolzina-1]))+ dolzina;
    }

}
