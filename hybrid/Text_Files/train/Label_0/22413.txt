#include <bits/stdc++.h>
using namespace std;
const int inf = (int)1e9;
const long long linf = (long long)1e18;
const int mod = (int)1e9 + 7;
const long double eps = (long double)1e-8;
const int maxn = (int)2e4 + 5;
const long double pi = acos(-1);
int ans = 0;
int n, l, r;
pair<int, int> a[maxn];
vector<int> v0, v1;
int d[maxn];
void prodv(int& u, int& sumh, int h, int gr, int dop = 0) {
  if (h < 0 || u > gr + 1) return;
  while (sumh <= h && u <= gr) {
    sumh += a[v1[u]].first;
    ++u;
  }
  if (sumh <= h)
    ans = max(ans, u + dop);
  else
    ans = max(ans, u);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> l >> r;
  for (auto i = 0; i < n; ++i) {
    cin >> a[i].first;
  }
  for (auto i = 0; i < n; ++i) {
    cin >> a[i].second;
  }
  sort(a, a + n);
  for (auto i = 0; i < n; ++i) {
    if (a[i].second == 0)
      v0.push_back(i);
    else
      v1.push_back(i);
  }
  int u0 = 0;
  int sum0 = 0;
  int u1 = 0;
  int sum1 = 0;
  for (auto i = 0; i <= r; ++i) d[i] = inf;
  d[0] = 0;
  for (auto i : v0) {
    for (auto j = r; j >= 0; --j) {
      if (d[j] == inf) continue;
      d[j + a[i].first] = 0;
    }
  }
  int minl = r + 1;
  int minl2 = r + 1;
  for (auto i = l; i <= r; ++i)
    if (d[i] != inf) {
      minl = i;
      break;
    }
  for (auto i1 = ((int)(v1).size()) - 1; i1 >= 0; --i1) {
    int i = v1[i1];
    for (auto j = l; j <= r; ++j) {
      if (d[j] != inf) {
        minl = j;
        break;
      }
    }
    for (auto j = minl; j <= r; ++j) {
      if (d[j] != inf && d[j] < ((int)(v1).size()) - i1 - 1) {
        minl2 = j;
        break;
      }
    }
    prodv(u0, sum0, r - minl, i1);
    prodv(u1, sum1, r - minl2, i1, 1);
    for (auto j = r; j >= 0; --j) {
      if (d[j] == inf) continue;
      d[j + a[i].first] = min(d[j + a[i].first], d[j] + 1);
    }
  }
  for (auto j = l; j <= r; ++j) {
    if (d[j] != -1 && d[j] < ((int)(v1).size())) {
      minl2 = j;
      break;
    }
  }
  prodv(u1, sum1, r - minl2, -1, 1);
  cout << ans << endl;
  return 0;
}
