for _ in range(int(input())):
	(p, q) = map(int, input().split())
	d = {}
	ans = []
	if p % q != 0:
		print(p)
	else:
		for i in range(2, int(q ** 0.5) + 1):
			count = 0
			while q % i == 0:
				count += 1
				q //= i
			if count != 0:
				d[i] = count
		if q >= 2:
			d[q] = 1
		for i in d:
			tmp = p
			while tmp % i == 0:
				tmp //= i
			tmp *= i ** (d[i] - 1)
			ans.append(tmp)
		print(max(ans))
