#include <bits/stdc++.h>
using namespace std;
long long n, arr[300005], sum[300005], M = 1000000007ll;
int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> arr[i + 1];
  sort(arr + 1, arr + n + 1);
  for (int i = 1; i <= n; ++i) {
    sum[i] -= arr[i];
    sum[n - i + 1] += arr[i];
  }
  for (int i = 1; i <= n; ++i) {
    sum[i] += sum[i - 1];
    sum[i] %= M;
  }
  long long po = 1, ans = 0;
  for (int i = 1; i < n; ++i) {
    ans += (po * sum[i]) % M;
    ans %= M;
    po *= 2ll;
    po %= M;
  }
  cout << (ans + M) % M << endl;
  return 0;
}
