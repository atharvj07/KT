def dfs(ii, jj, nnum):
	global num, s, ans, d_map
	for v in d_map:
		if num.index(v) == 0:
			ii += 1
		elif num.index(v) == 1:
			jj -= 1
		elif num.index(v) == 2:
			ii -= 1
		else:
			jj += 1
		if ii < 0 or jj < 0 or ii == n or (jj == m):
			return
		if s[ii][jj] == '#':
			return
		if s[ii][jj] == 'E':
			ans += 1
			return
(n, m) = map(int, input().split())
s = []
for i in range(n):
	s.append(list(input()))
d_map = list(map(int, list(input())))
nnum = len(d_map)
(s_i, s_j) = (0, 0)
for i in range(n):
	for j in range(m):
		if s[i][j] == 'S':
			(s_i, s_j) = (i, j)
ans = 0
for x1 in range(4):
	for x2 in range(4):
		for x3 in range(4):
			for x4 in range(4):
				num = [x1, x2, x3, x4]
				if sorted(num) == [0, 1, 2, 3]:
					dfs(s_i, s_j, nnum)
print(ans)
