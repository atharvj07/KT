n = int(input())
t = [0] * n
for i in range(n):
	t[i] = list(map(int, input().split()))
value = [0] * 5
solve = [0] * 5
for i in range(5):
	solved = 0
	for j in range(n):
		if t[j][i] != -1:
			solved += 1
		else:
			t[j][i] = 250
	if solved * 2 > n:
		value[i] = 500
	elif solved * 4 > n:
		value[i] = 1000
	elif solved * 8 > n:
		value[i] = 1500
	elif solved * 16 > n:
		value[i] = 2000
	elif solved * 32 > n:
		value[i] = 2500
	else:
		value[i] = 3000
	solve[i] = solved
vasya = 0
for i in range(5):
	vasya += value[i] - value[i] // 250 * t[0][i]
petya = 0
for i in range(5):
	petya += value[i] - value[i] // 250 * t[1][i]

				pot[problem][cur_ac] = win
			else:

				pot[problem][cur_ac] = win
			cur_ac += 1
			if cur_ac == 20000:
				break
	elif t[0][problem] != 250:

				pot[problem][cur_ac] = win

	maxi = max(maxi, len(pot[i]))

		if len(pot[pr]) <= i:
			tmp += pot[pr][-1]
		else:
			tmp += pot[pr][i]

		res = i
