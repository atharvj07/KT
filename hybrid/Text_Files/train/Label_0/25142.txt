#include <bits/stdc++.h>
using namespace std;
int main(){
	int n; cin>>n;
	vector<int> e(1e5+1,0),o(1e5+1,0);
	for(int i=0;i<n;i++){
		int foo; cin>>foo;
		if(i%2==1) o[foo]++;
		else e[foo]++; 
	}
	//I love complement 
	int max_even=max_element(e.begin(),e.end())-e.begin();
	int max_odd=max_element(o.begin(),o.end())-o.begin();
	if(max_even!=max_odd) cout<<n-e[max_even]-o[max_odd];
	else{
		sort(e.rbegin(),e.rend());
		sort(o.rbegin(),o.rend());
		cout<<min(n-e[0]-o[1],n-e[1]-o[0])<<endl;
	}
	return 0;
}
