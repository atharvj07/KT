s = input()
t = input()
r = [0] * len(t)
for i in range(len(t) - 1, -1, -1):
	p = len(s) - 1
	if i + 1 < len(t):
		p = r[i + 1] - 1
	while t[i] != s[p]:
		p -= 1
	r[i] = p
ans = p = 0
for i in range(len(s)):
	rp = len(s) - 1
	if p < len(t):
		rp = r[p] - 1
	ans = max(rp - i + 1, ans)
	if p < len(t) and t[p] == s[i]:
		p += 1
print(ans)
