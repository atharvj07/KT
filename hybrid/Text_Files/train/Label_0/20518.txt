#include <bits/stdc++.h>
using namespace std;
struct node {
  int a, b;
} p[301800];
inline int read() {
  int x = 0, w = 0;
  char ch = getchar();
  while (!isdigit(ch)) w |= ch == '-', ch = getchar();
  while (isdigit(ch)) x = (x << 1) + (x << 3) + ch - '0', ch = getchar();
  return w ? -x : x;
}
inline bool cmp1(node a, node b) { return a.b > b.b; }
inline bool cmp2(node a, node b) { return a.a < b.a; }
inline bool cmp3(node a, node b) { return a.a + a.b > b.a + b.b; }
int n, m, loc, ans1 = -1, ans2, f[102][60300];
signed main() {
  n = read(), m = read();
  for (int i = 1; i <= n; ++i) p[i].a = read(), p[i].b = read();
  sort(p + 1, p + 1 + n, cmp1);
  for (int i = n; i; --i)
    if (p[i].b >= 0) {
      loc = i;
      break;
    }
  sort(p + 1, p + loc + 1, cmp2);
  for (int i = loc + 1; i <= n; ++i) p[i].a = max(p[i].a, -p[i].b);
  sort(p + loc + 1, p + 1 + n, cmp3);
  for (int i = 1; i <= loc; ++i) {
    if (m < p[i].a) {
      ans1 = i - 1;
      break;
    }
    m += p[i].b;
  }
  if (ans1 == -1) ans1 = loc;
  for (int i = 1 + loc; i <= n; ++i) {
    memcpy(f[i + 1], f[i], sizeof(f[i + 1]));
    for (int j = p[i].a; j <= m; ++j) {
      f[i + 1][j + p[i].b] = max(f[i + 1][j + p[i].b], f[i][j] + 1);
      ans2 = max(ans2, f[i + 1][j + p[i].b]);
    }
  }
  printf("%d\n", ans1 + ans2);
  return 0;
}
