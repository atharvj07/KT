import java.util.Scanner;

public class Code {
	public Code(){}
	public static void main(String args[]){
		long[] v=new long [100005];
		long[] dp=new long[100005];
		int[] c=new int[100005];
		Scanner input=new Scanner(System.in);
		int n=input.nextInt(),q=input.nextInt();
		for (int i=1;i<=n;i++) v[i]=input.nextInt();
		for (int i=1;i<=n;i++) c[i]=input.nextInt();
		for (int i=1;i<=q;i++){
			for (int j=1;j<=n;j++) dp[j]=-20000000000L;
			long a=input.nextInt(),b=input.nextInt();
			int col1=0,col2=0;
			long ans=0;
			for (int j=1;j<=n;j++){
				long res=dp[c[j]]+v[j]*a;
				if (col1!=c[j]) res=Math.max(res,dp[col1]+v[j]*b);
				if (col2!=c[j]) res=Math.max(res,dp[col2]+v[j]*b);
				dp[c[j]]=Math.max(res,dp[c[j]]);
				res=dp[c[j]];
				if (res>dp[col2]&&c[j]!=col1)
					col2=c[j];else
					if (res>dp[col1]&&c[j]!=col2)
						col1=c[j];
				if (dp[col2]>dp[col1]){
					int tmp=col1;col1=col2;col2=tmp;
				}
				ans=Math.max(ans,dp[c[j]]);
			}
			System.out.println(ans);
		}
	}
}
