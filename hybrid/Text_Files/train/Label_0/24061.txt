#include <bits/stdc++.h>
using namespace std;
template <typename T>
inline T abs(T t) {
  return t < 0 ? -t : t;
}
const long long modn = 1000000007;
inline long long mod(long long x) { return x % modn; }
int n;
map<long long, long long> cus;
long long h(long long x) {
  long long i = 0;
  while (x >= (1ll << i)) {
    i++;
  }
  return i;
}
int main() {
  scanf("%d", &n);
  for (int a = 0; a < n; a++) {
    int t;
    scanf("%d", &t);
    long long u, v, w;
    if (t == 1) {
      cin >> u >> v >> w;
      if (h(v) > h(u)) swap(u, v);
      while (h(u) > h(v)) {
        cus[u] += w;
        u /= 2ll;
      }
      while (u != v) {
        cus[u] += w;
        cus[v] += w;
        u /= 2;
        v /= 2;
      }
    } else {
      cin >> u >> v;
      long long ans = 0ll;
      if (h(v) > h(u)) swap(u, v);
      while (h(u) > h(v)) {
        ans += cus[u];
        u /= 2ll;
      }
      while (u != v) {
        ans += cus[u];
        ans += cus[v];
        u /= 2;
        v /= 2;
      }
      cout << ans << endl;
    }
  }
}
