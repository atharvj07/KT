long long mull(long long a, long long b, long long m) {
  return ((a % m) * (b % m)) % m;
}
long long gcd(long long a, long long b) {
  if (a < b) swap(a, b);
  while (a != 0 && b != 0) {
    long long r = a % b;
    a = b;
    b = r;
  }
  return a;
}
long long lcm(long long a, long long b) { return a / gcd(a, b) * b; }
