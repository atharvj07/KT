import bisect
(n, k) = map(int, input().split())
a = list(map(int, input().split()))

def work(number):
	a = []
	n = len(number)
	number = [number[i] - i for i in range(0, n)]
	for i in range(0, n):
		if number[i] < number[0] or number[i] > number[n - 1]:
			continue
		kk = bisect.bisect_right(a, number[i])
		if kk == len(a):
			a.append(number[i])
		else:
			a[kk] = number[i]
	nn = n - len(a)
	return nn
if k == 0:
	b = []
else:
	b = list(map(int, input().split()))
a = [-10 ** 10] + a + [10 ** 10]
b = [0] + b + [n + 1]
ans = 0
jud = True
(cut, number) = (b[0], a[0])
for c in b[1:]:
	if c - cut > a[c] - number:
		jud = False
		break
	ans += work(a[cut:c + 1])
	(cut, number) = (c, a[c])
if jud:
	print(ans)
else:
	print(-1)
