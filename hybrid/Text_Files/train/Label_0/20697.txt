#include <bits/stdc++.h>
using namespace std;
int n;
string s1, s2;
double res;
int main() {
  cin >> n >> s1 >> s2;
  for (char c = 'A'; c <= 'Z'; c++) {
    long long l = 0, r = 0;
    for (int j = 0; j < n; j++)
      if (s1[j] == c) r += n - j;
    for (int j = 0; j < n; j++) {
      if (s1[j] == c) r -= n - j;
      if (s2[j] == c) {
        if (s1[j] == c) res += (double)(j + 1) * (double)(n - j);
        res += l * (double)(n - j);
        res += (double)r * (j + 1);
      }
      if (s1[j] == c) {
        l += j + 1;
      }
    }
  }
  double tot = 0;
  for (int i = 1; i <= n; i++) tot += (double)i * i;
  double p = res / tot;
  printf("%.12f", p);
  return 0;
}
