def mex(A, B, m):
	find = [i for i in range(0, m + 2)]
	for item in find:
		if item not in A:
			mex_A = item
			break
	for item in find:
		if item not in B:
			mex_B = item
			break
	return mex_A + mex_B

def solve(a):
	a.sort()
	m = max(a)
	A = []
	B = []
	for item in a:
		if item not in A:
			A.append(item)
		else:
			B.append(item)
	return mex(list(set(A)), list(set(B)), m)
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	print(solve(a))
