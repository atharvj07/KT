IN = input().split('=')
(a, b) = IN[0].split('+')
c = IN[1]
if len(a) + len(b) + 1 == len(c) - 1:
	a = '|' + a
	c = c.replace('|', '', 1)
	print(f'{a}+{b}={c}')
elif len(a) + len(b) - 1 == len(c) + 1:
	if len(a) == 1:
		c = '|' + c
		b = b.replace('|', '', 1)
	elif len(a) == len(b) == 1:
		print('Impossible')
	else:
		c = '|' + c
		a = a.replace('|', '', 1)
	print(f'{a}+{b}={c}')
elif len(a) + len(b) == len(c):
	print(f'{a}+{b}={c}')
else:
	print('Impossible')
