board = list(map(int, input().split(' ')))
max_score = 0
for i in range(len(board)):
	if board[i] == 0:
		continue
	next_board = board[:]
	next_board[i] = 0
	increase_amount = int(board[i] / len(board))
	extra = board[i] % len(board)
	for ni in range(len(next_board)):
		next_board[ni] += increase_amount
	for e in range(1, extra + 1):
		next_board[(i + e) % len(next_board)] += 1
	cur_score = 0
	for c in next_board:
		if c % 2 == 0:
			cur_score += c
	max_score = max(max_score, cur_score)
print(max_score)
