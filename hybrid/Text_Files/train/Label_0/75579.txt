import math
t = int(input())
for _ in range(0, t):
	info = list(map(int, input().split()))
	l = info[0]
	m = info[1]
	numbers = list(input())
	result = ['1'] * l
	b = 0
	for d in range(l):
		if numbers[d] == '0':
			moves_to_go = min(m, d - b)
			m = m - moves_to_go
			result[d - moves_to_go] = '0'
			b += 1
	print(''.join(result))
