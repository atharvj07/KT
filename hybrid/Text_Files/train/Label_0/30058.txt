#include <bits/stdc++.h>
using namespace std;
int main() {
  int a;
  string b, c;
  cin >> a >> b >> c;
  if (c == "week") {
    if (a == 5 || a == 6)
      cout << "53\n";
    else
      cout << "52\n";
    return 0;
  }
  if (a == 31)
    cout << "7\n";
  else if (a == 30)
    cout << "11\n";
  else if (a <= 29)
    cout << "12\n";
  return 0;
}
