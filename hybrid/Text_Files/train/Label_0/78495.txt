from collections import defaultdict
t = int(input())
for _ in range(t):
	n = int(input())
	a = [int(x) for x in input().split()]
	cnt = 0
	m = defaultdict(list)
	for (i, x) in enumerate(a):
		y = 1
		while y * x < 2 * (i + 1):
			if y in m and y * x - i - 1 in m[y]:
				cnt += 1
			y += 1
		m[x].append(i + 1)
	print(cnt)
