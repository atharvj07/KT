#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  a = abs(a);
  b = abs(b);
  if (b == 0) {
    return a;
  } else {
    return gcd(b, a % b);
  }
}
void read(vector<int> &x) {
  for (int i = 0; i < x.size(); i++) {
    cin >> x[i];
  }
}
void write(vector<int> x) {
  for (int i = 0; i < x.size(); i++) {
    cout << x[i] << ' ';
  }
  cout << endl;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  long long n, k, a, b;
  cin >> n >> k >> a >> b;
  long long minn = 1e9;
  long long maxx = 0;
  for (int i = 0; i <= n; i++) {
    if (k - a - b + i * k > 0) {
      minn = min(minn, gcd(k - a - b + i * k, k * n));
      maxx = max(maxx, gcd(k - a - b + i * k, k * n));
    }
    if (a - b + i * k > 0) {
      minn = min(minn, gcd(a - b + i * k, k * n));
      maxx = max(maxx, gcd(a - b + i * k, k * n));
    }
    if (b - a + i * k > 0) {
      minn = min(minn, gcd(b - a + i * k, k * n));
      maxx = max(maxx, gcd(b - a + i * k, k * n));
    }
  }
  cout << n * k / maxx << ' ';
  cout << n * k / minn;
}
