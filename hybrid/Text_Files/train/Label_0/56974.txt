import java.util.*;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();

        long[] A = new long[n];
        long[] B = new long[n];

        String s = sc.next();

        for (int i = 0; i < n; i++) {
            A[i] = s.charAt(i) - '0';
        }

        for (int i = 0; i < k; i++) {
            B[i] = A[i];
        }

        for (int i = k; i < n; i++) {
            B[i] = B[i % k];
        }

        System.out.println(A.length);
        int kek = check(A, B);
        if (kek == -1) {
            for (int i = 0; i < B.length; i++) {
                System.out.print(B[i]);
            }
        } else {
            long BC[] = new long[n];
            int mem = 1;
            for (int i = k - 1; i >= 0; i--) {
                BC[i] = (B[i] + mem) % 10;
                if (BC[i] != 0) {
                    mem = 0;
                }
            }

            for (int i = k; i < n; i++) {
                BC[i] = BC[i % k];
            }
            for (int i = 0; i < BC.length; i++) {
                System.out.print(BC[i]);
            }

        }


    }

    static int check(long[] A, long[] B) {
        for (int i = 0; i < A.length; i++) {
            if (A[i] > B[i]) {
                return i;
            } else {
                if(A[i] < B[i]) {
                    return -1;
                }
            }
        }
        return -1;
    }

}
