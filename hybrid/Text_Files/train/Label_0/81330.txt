import math
from collections import Counter, defaultdict, deque

def f(n):
	k = 1
	m = [n]
	for i in range(2, math.floor(math.sqrt(n) + 1)):
		if n % i == 0:
			tempN = n
			tempans = 0
			while tempN % i == 0 and tempN / i % i == 0:
				tempN = tempN // i
				tempans += 1
			if k < tempans + 1:
				k = tempans + 1
				m = [i for j in range(tempans)]
				m.append(tempN)
	return m
t = int(input())
result = []
for i in range(t):
	n = int(input())
	result.append(f(n))
for i in range(t):
	print(len(result[i]))
	for j in range(len(result[i])):
		print(result[i][j], end=' ')
	print()
