#include <bits/stdc++.h>
using namespace std;
long long n, ans = 1;
long long C(long long m, long long n) {
  long long p = 1;
  for (long long i = 1; i <= n; i++) p *= i;
  for (long long i = 1; i <= m; i++) p /= i;
  for (long long i = 1; i <= n - m; i++) p /= i;
  return p;
}
signed main() {
  cin >> n;
  for (long long i = 1; i <= n / 2 - 1; i++) ans *= i;
  ans *= ans;
  ans *= C(n / 2, n);
  ans /= 2;
  cout << ans;
  return 0;
}
