#include <bits/stdc++.h>
using namespace std;
int read() {
  int f = 1, x = 0;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') f = -f;
    c = getchar();
  }
  while (c >= '0' && c <= '9') x = x * 10 + c - '0', c = getchar();
  return f * x;
}
int n, realn[66] = {0};
void init() {
  n = read();
  for (int i = 1; i <= 60; ++i) {
    int t = i;
    for (int j = 1; t >= j; ++j) realn[i]++, t -= j;
  }
}
void solve() {
  int res = 0;
  while (n--) res ^= realn[read()];
  printf("%s\n", res ? "NO" : "YES");
}
int main() {
  init();
  solve();
  return 0;
}
