MOD = 1000000007
fib = [1, 2]
for i in range(1 + int(100000.0)):
	fib.append((fib[-1] + fib[-2]) % MOD)
s = input()
n = len(s)
if 'm' in s or 'w' in s:
	print(0)
else:
	i = 0
	ans = 1
	while i < n:
		j = i + 1
		if s[i] == 'n':
			while j < n and s[j] == 'n':
				j += 1
			ans = ans * fib[j - i - 1] % MOD
		elif s[i] == 'u':
			while j < n and s[j] == 'u':
				j += 1
			ans = ans * fib[j - i - 1] % MOD
		i = j
	print(ans)
