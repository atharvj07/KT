t = int(input())
for _ in range(t):
	a = input()
	odd = []
	even = []
	for i in range(len(a)):
		if int(a[i]) % 2 == 0:
			even.append(int(a[i]))
		else:
			odd.append(int(a[i]))
	ans = []
	o_index = 0
	e_index = 0
	while o_index < len(odd) and e_index < len(even):
		if odd[o_index] < even[e_index]:
			ans.append(odd[o_index])
			o_index += 1
		else:
			ans.append(even[e_index])
			e_index += 1
	if o_index < len(odd):
		ans += odd[o_index:]
	else:
		ans += even[e_index:]
	print(''.join(map(str, ans)))
