t = int(input())
for _ in range(t):
	n = int(input())
	(a, b) = ([], [])
	c = dict()
	d = dict()
	for i in range(n):
		(x, y) = [int(i) for i in input().split()]
		a.append(x)
		b.append(y)
		c[x] = c.get(x, 0) + 1
		d[y] = d.get(y, 0) + 1
	ans = 0
	n1 = n
	for i in range(n):
		ans += (n1 - 2) * (n1 - 1) / 2 - (c[a[i]] - 1) * (d[b[i]] - 1)
		n1 -= 1
	print(int(ans))
