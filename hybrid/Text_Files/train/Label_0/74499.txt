for i in range(int(input())):
	x1 = x2 = x3 = out = x4 = int(0)
	ans = set()
	for x in list(input()):
		if x == 'N':
			x3 -= 1
		elif x == 'S':
			x3 += 1
		elif x == 'E':
			x4 += 1
		elif x == 'W':
			x4 -= 1
		if (x1, x2, x3, x4) not in ans and (x3, x4, x1, x2) not in ans:
			out += 5
			ans.add((x1, x2, x3, x4))
			ans.add((x3, x4, x1, x2))
		else:
			out += 1
		x1 = x3
		x2 = x4
	print(out)
