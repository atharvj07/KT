for case in range(int(input())):
	(n, m) = (int(i) for i in input().split())
	p = [sorted([int(i) for i in input().split()]) for j in range(n)]
	pp = [[] for i in p]
	for iii in range(m):
		mn = min(range(n), key=lambda x: p[x][0])
		for i in range(len(p)):
			pp[i].append(p[i].pop(0) if i == mn else p[i].pop())
	for t in pp:
		print(*t)
