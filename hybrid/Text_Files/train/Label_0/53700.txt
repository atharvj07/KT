
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class Fountains {
	static int INF = -10000000;

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int C = sc.nextInt();
		int D = sc.nextInt();
		TreeMap<Integer, pair> CMap = new TreeMap<>();
		TreeMap<Integer, pair> DMap = new TreeMap<>();

		ArrayList<pair> coins = new ArrayList<>();
		ArrayList<pair> diamonds = new ArrayList<>();

		ArrayList<Integer> coinsInteger = new ArrayList<>();
		ArrayList<Integer> diamondIntegers = new ArrayList<>();

		int maxCB = INF;
		int maxDB = INF;
		for (int i = 0; i < n; i++) {
			int b = sc.nextInt();
			int v = sc.nextInt();
			char c = sc.nextChar();
			if (c == 'C') {
				if (v <= C) {
					coins.add(new pair(v, b));
					if (b > maxCB)
						maxCB = b;

					coinsInteger.add(v);
				}

			} else if (v <= D) {
				diamonds.add(new pair(v, b));
				if (b > maxDB)
					maxDB = b;
				diamondIntegers.add(v);

			}
		}

		Collections.shuffle(coins);
		Collections.shuffle(diamonds);
		Collections.sort(coins);
		Collections.sort(diamonds);
		Collections.sort(diamondIntegers);
		Collections.sort(coinsInteger);

		for (int i = 0; i < coins.size(); i++) {
			pair p = coins.get(i);
			if (i != 0) {
				pair x = CMap.get(coins.get(i - 1).x);
				int a[] = { x.x, x.y, p.y };
				Arrays.sort(a);
				x = new pair(a[2], a[1]);
				CMap.put(p.x, x);
			} else {

				CMap.put(p.x, new pair(p.y, INF));
			}

		}
		for (int i = 0; i < diamonds.size(); i++) {
			pair p = diamonds.get(i);
			if (i != 0) {
				pair x = DMap.get(diamonds.get(i - 1).x);
				int a[] = { x.x, x.y, p.y };
				Arrays.sort(a);
				x = new pair(a[2], a[1]);
				DMap.put(p.x, x);
			} else {

				DMap.put(p.x, new pair(p.y, INF));
			}

		}

		int max = 0;

		for (int i = 0; i < coins.size(); i++) {
			pair p = coins.get(i);
			int x = Collections.binarySearch(coinsInteger, C - p.x);
			if (x < 0)
				x = (x + 1) * -1 - 1;

			if (x < 0)
				continue;

			pair p2 = CMap.get(coinsInteger.get(x));
			int v1 = p2.x;
			if (p2.x == p.y)
				v1 = p2.y;

			max = Math.max(max, Math.max(v1 + p.y, maxCB + maxDB));

		}

		for (int i = 0; i < diamonds.size(); i++) {
			pair p = diamonds.get(i);
			int x = Collections.binarySearch(diamondIntegers, D - p.x);
			if (x < 0)
				x = (x + 1) * -1 - 1;


			if (x < 0)
				continue;
			pair p2 = DMap.get(diamondIntegers.get(x));
			int v1 = p2.x;
			if (p2.x == p.y)
				v1 = p2.y;

			max = Math.max(max, Math.max(v1 + p.y, maxCB + maxDB));

		}
		max = Math.max(max, maxCB + maxDB);
		if (max < 0)
			System.out.println(0);
		else
			System.out.println(max);
	}

	static class pair implements Comparable<pair> {

		int x, y;

		public pair(int a, int b) {

			x = a;

			y = b;

		}

		@Override
		public int compareTo(pair o) {
			// TODO Auto-generated method stub
			if (x != o.x)
				return x > o.x ? 1 : -1;
			if (y != o.y)
				return y > o.y ? 1 : -1;

			return 0;
		}

	}

	static class Scanner {

		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream System) {
			br = new BufferedReader(new InputStreamReader(System));
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public String nextLine() throws IOException {
			return br.readLine();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public char nextChar() throws IOException {
			return next().charAt(0);
		}

		public Long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public boolean ready() throws IOException {
			return br.ready();
		}

		public void waitForInput() {
			for (long i = 0; i < 3e9; i++)
				;
		}
	}
}
