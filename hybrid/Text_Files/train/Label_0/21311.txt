#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  int q;
  cin >> q;
  vector<long long> C2(100000);
  for (long long i = 0; i < 100000; i++) C2[i] = i * (i - 1) / 2;
  while (q--) {
    long long x;
    cin >> x;
    long long i = upper_bound(C2.begin(), C2.end(), x) - C2.begin() - 2;
    long long a = x - i + i * (i - 1) / 2;
    long long b = 2 * (x - i - 1);
    cout << max(a, b) << endl;
    ;
  }
}
