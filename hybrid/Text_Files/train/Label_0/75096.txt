import sys
input = sys.stdin.readline

def gcd(a, b):
	if a == 0:
		return b
	return gcd(b % a, a)

def lcm(a, b):
	return a * b / gcd(a, b)

def main():
	s = input()
	st = s[0]
	en = s[len(s) - 2]
	preflen = 1
	suflen = 1
	lens = []
	c = 1
	for i in range(1, len(s) - 1):
		if s[i] != s[i - 1]:
			c += 1
		else:
			lens.append(c)
			c = 1
	if c != 1:
		lens.append(c)
	for i in range(1, len(s) - 1):
		if s[i] != s[i - 1]:
			preflen += 1
		else:
			break
	for i in range(len(s) - 2, 0, -1):
		if s[i] != s[i - 1]:
			suflen += 1
		else:
			break
	if preflen == len(s) - 1 or st == en:
		if len(s) - 1 > 1:
			print(max(lens))
		else:
			print(1)
	else:
		print(max(preflen + suflen, max(lens)))
	return
main()
