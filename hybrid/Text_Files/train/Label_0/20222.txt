#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  string a;
  cin >> a;
  string b;
  cin >> b;
  int pos = 0;
  for (int i = 0; i < b.size(); i++) {
    if (b[i] == a[pos] && pos < a.size() - 1) pos++;
  }
  cout << (pos + 1) << endl;
  return 0;
}
