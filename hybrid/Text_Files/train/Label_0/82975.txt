(n, k) = map(int, input().split())
a = list(map(int, input().split()))
a_sum = [0]
for i in range(n):
	a_sum.append(a_sum[-1] + a[i])
a = [0] + a

def get(i, j):
	assert 0 <= i <= j <= n
	if i > 0:
		return a_sum[j] - a_sum[i - 1]
	else:
		return a_sum[j]
ans = 0
nowmax = 0
for bit in range(60, -1, -1):
	tmpmax = nowmax + (1 << bit)
	dp = [[0] * (n + 10) for i in range(n + 10)]
	dp[0][0] = tmpmax
	for pos in range(1, n + 1):
		for div in range(1, k + 1):
			for l in range(1, pos + 1):
				dp[pos][div] = max(dp[pos][div], dp[pos - l][div - 1] & get(pos - l + 1, pos) & tmpmax)
	if dp[n][k] == tmpmax:
		nowmax += 1 << bit
print(nowmax)
