import sys
sys.setrecursionlimit(10 ** 5)
int1 = lambda x: int(x) - 1
p2D = lambda x: print(*x, sep='\n')

def II():
	return int(sys.stdin.readline())

def MI():
	return map(int, sys.stdin.readline().split())

def LI():
	return list(map(int, sys.stdin.readline().split()))

def LLI(rows_number):
	return [LI() for _ in range(rows_number)]

def SI():
	return sys.stdin.readline()[:-1]
n = II()
aa = LI()
dd = [0] * n
s = 0
for i in range(n - 1):
	dd[i + 1] = aa[i + 1] - aa[i]
for d in dd:
	if d > 0:
		s += d
b0 = (aa[0] - s) // 2
print(max(b0 + s, aa[0] - b0))

def add(i, a, s):
	if i == 0:
		aa[0] += a
		return s
	if dd[i] > 0:
		s -= dd[i]
	dd[i] += a
	if dd[i] > 0:
		s += dd[i]
	return s
for _ in range(II()):
	(l, r, x) = MI()
	l -= 1
	s = add(l, x, s)
	if r < n:
		s = add(r, -x, s)
	b0 = (aa[0] - s) // 2
	print(max(b0 + s, aa[0] - b0))
