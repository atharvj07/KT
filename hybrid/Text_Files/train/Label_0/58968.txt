import java.util.*;
import java.io.*;
import java.text.*;

public class F1326 {

	static int n;
	static int[][] adjMatrix;
	static long dp[][][];

	public static long[] dp(int mask, int r) {
		int bitc = Integer.bitCount(mask);
		if (bitc == 2) {
			return new long[] { 1 - adjMatrix[0][r], adjMatrix[0][r] };
		}
		if (dp[mask][r] != null)
			return dp[mask][r];

		int nmask = mask ^ (1 << r);
		long[] ans = new long[1 << (bitc - 1)];
		for (int i = 1; i < n; i++) {
			if (((nmask & (1 << i)) != 0)) {
				long[] u = (dp(nmask, i));
				for (int k = 0; k < u.length; k++) {
					int newk = k << 1;
					newk |= adjMatrix[r][i];
					ans[newk] += u[k];
				}

			}
		}
		return dp[mask][r] = ans;

	}

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);
		PrintWriter pw = new PrintWriter(System.out);
		n = sc.nextInt();
		adjMatrix = new int[n][n];
		for (int i = 0; i < n; i++) {
			String s = sc.next();
			for (int j = 0; j < n; j++) {
				adjMatrix[i][j] = s.charAt(j) - '0';
			}
		}

		dp = new long[1 << n][n][];
		long[] ans = new long[1 << (n - 1)];

		for (int j = 1; j < n; j++) {
			long[] temp = dp((1 << n) - 1, j);
			for (int k = 0; k < temp.length; k++) {
				int h = (k << 1) | adjMatrix[0][j];
				for (int i = 0; i < n; i++) {
					ans[h % (1 << n - 1)] += temp[k];
					int bit = h & 1;
					h >>= 1;
					h |= (bit << (n - 1));
				}

			}
		}

		for (int i = 0; i < ans.length; i++) {
			pw.print(ans[i] + " ");
		}
		pw.close();
	}

	static class Scanner {
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public Scanner(FileReader r) {
			br = new BufferedReader(r);
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public String nextLine() throws IOException {
			return br.readLine();
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public int[] nextIntArr(int n) throws IOException {
			int[] arr = new int[n];
			for (int i = 0; i < arr.length; i++)
				arr[i] = nextInt();
			return arr;
		}

		public boolean ready() throws IOException {
			return br.ready();
		}

	}
}
