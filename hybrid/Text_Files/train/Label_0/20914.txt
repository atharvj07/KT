#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  map<string, int> mp;
  deque<string>::iterator it;
  cin >> n;
  string s;
  deque<string> dq;
  for (int i = 0; i < n; i++) {
    cin >> s;
    dq.push_front(s);
    mp[s]++;
  }
  for (it = dq.begin(); it != dq.end(); it++) {
    if (mp[*it] != 0) {
      cout << *it << endl;
      mp[*it] = 0;
    }
  }
  return 0;
}
