n = int(input())
strArr = []
for x in range(n):
	strArr.append(input())

def sortStrByOrder(strArr):
	if strArr == None or len(strArr) == 0:
		return ''
	from functools import cmp_to_key
	strArr = sorted(strArr, key=cmp_to_key(lambda x, y: 1 if x + y > y + x else -1))
	str = ''
	for x in strArr:
		str += x
	return str
print(sortStrByOrder(strArr))
