#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 1;
int a[N];
int n, ans;
int mt[N];
void makemax();
int main() {
  int i;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", a + i);
  }
  makemax();
  ans = 1;
  for (i = 1; i <= n; i++) {
    if (mt[ans] < mt[i]) ans = i;
  }
  printf("%d\n", ans);
}
void makemax() {
  int m1 = 0, m2 = 0, i;
  for (i = 1; i <= n; i++) {
    if (m1 < a[i]) {
      m2 = m1;
      m1 = a[i];
    } else if (m2 < a[i]) {
      m2 = a[i];
    }
    if (m2 <= a[i] && m1 > a[i]) {
      mt[m1]++;
    }
    if (a[i] == m1) mt[m1]--;
  }
}
