from sys import stdin, stdout
from math import log10 as lg
(n, m) = map(int, stdin.readline().split())
Num = 7
hours = []
minutes = []
challengers = []
ans = 0

def pref(a, maxlen):
	if len(a) == maxlen:
		challengers.append(a[:])
	elif not len(a):
		for i in range(1, Num):
			if i not in a:
				a.append(i)
				pref(a, maxlen)
				a.pop()
	else:
		for i in range(Num):
			if i not in a:
				a.append(i)
				pref(a, maxlen)
				a.pop()
for i in range(1, Num):
	pref([], i)
challengers.append([0])
for i in challengers:
	hours.append((int(lg(max(1, n - 1)) / lg(7)) - len(i) + 1) * '0' + ''.join(list(map(str, i))))
	if hours[-1].count('0') > 1:
		hours.pop()
	minutes.append((int(lg(max(m - 1, 1)) / lg(7)) - len(i) + 1) * '0' + ''.join(list(map(str, i))))
	if minutes[-1].count('0') > 1:
		minutes.pop()
lenh = len(hours)
lenm = len(minutes)
for i in range(lenh):
	cnt = 0
	vl = str(hours[i])[::-1]
	for j in range(len(vl)):
		if int(vl[j]):
			cnt += int(vl[j]) * 7 ** j
	if cnt < n:
		hours.append(hours[i])
for i in range(lenm):
	cnt = 0
	vl = str(minutes[i])[::-1]
	for j in range(len(vl)):
		if int(vl[j]):
			cnt += int(vl[j]) * 7 ** j
	if cnt < m:
		minutes.append(minutes[i])
hours = hours[lenh:]
minutes = minutes[lenm:]
for f in hours:
	for s in minutes:
		if not set(list(str(f))) & set(list(str(s))):
			ans += 1
stdout.write(str(ans))
