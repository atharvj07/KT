#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;
using pll = pair<ll, ll>;
template <class T>
bool uin(T &a, T b) {
  return a > b ? (a = b, true) : false;
}
template <class T>
bool uax(T &a, T b) {
  return a < b ? (a = b, true) : false;
}
int main(void) {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  double s, a, b, c;
  cin >> s >> a >> b >> c;
  if (!a && !b && !c) {
    puts("0 0 0");
    return 0;
  }
  double l1 = a * s / (a + b + c);
  double l2 = b * s / (a + b + c);
  double l3 = c * s / (a + b + c);
  cout.precision(16);
  cout << fixed;
  cout << l1 << " " << l2 << " " << l3 << '\n';
  return 0;
}
