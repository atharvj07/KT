#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int t;
  cin >> t;
  while (t--) {
    long long int n;
    cin >> n;
    long long int to = 0, th = 0;
    while (n % 2 == 0) {
      to++;
      n /= 2;
    }
    while (n % 3 == 0) {
      th++;
      n /= 3;
    }
    if (to > th || n != 1) {
      cout << "-1"
           << "\n";
    } else {
      cout << th + (th - to) << "\n";
    }
  }
}
