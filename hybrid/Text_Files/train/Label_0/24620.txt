#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main()
{
	for(int t;cin>>t && t;){
		vector<double> es;
		for(int i=0;i<=t;i++){
			int n,m; cin>>n>>m;
			double e=0,psum=0;
			for(int j=0;j<m;j++){
				double v,p; cin>>v>>p;
				psum+=p;
				e+=v*p;
			}
			es.push_back(e/psum);
		}
		
		int i;
		for(i=0;i<t;i++)
			if(es[i]>es.back()+1e-7-1e-9)
				break;
		cout<<(i<t?"YES":"NO")<<endl;
	}
	
	return 0;
}
