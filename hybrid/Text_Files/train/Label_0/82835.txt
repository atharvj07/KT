n = int(input())
ans = []
for k in range(n):
	n2 = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	index = {}
	j = 1
	for i in range(n2):
		while j < b[i]:
			index[j] = i
			j += 2
	temp = 100000000
	for i in range(n2):
		temp = min(temp, i + index[a[i]])
	ans.append(temp)
for i in ans:
	print(i)
