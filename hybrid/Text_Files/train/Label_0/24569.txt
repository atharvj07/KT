#include<bits/stdc++.h>
using namespace std;

int main(){
    int h1,h2,k1,k2;
    cin>>h1>>h2;
    cin>>k1>>k2;
    int a,b,c,d;
    cin>>a>>b>>c>>d;
    int x=h1*a+h1/10*c+h2*b+h2/20*d;
    int y=k1*a+k1/10*c+k2*b+k2/20*d;
    if(x<y) cout<<"kenjiro"<<endl;
        else if(x>y) cout<<"hiroshi"<<endl;
        else cout<<"even"<<endl;
    return 0;
    
}
