import java.io.*;
import java.math.*;
import java.util.*;

public class chess
{
    public static void main(String []args)
    {
        Scanner sc=new Scanner(System.in);
        int r1=sc.nextInt();
         int c1=sc.nextInt();
          int r2=sc.nextInt();
           int c2=sc.nextInt();
        int c=Math.abs(c1-c2);
        int r=Math.abs(r1-r2);
        int d1=(r1+c1)%2;
        int d2=(r2+c2)%2;
        if(r1==r2||c1==c2)
        System.out.print("1 ");
        else
        System.out.print("2 ");
        
        if(d1==d2&&r==c)
        System.out.print("1 ");
        else if(d1==d2&&r!=c)
         System.out.print("2 ");
        else 
        System.out.print("0 ");
        
        
        System.out.print(Math.max(r,c));
    }
}
