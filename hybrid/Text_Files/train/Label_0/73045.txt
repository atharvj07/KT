t = int(input())
for _ in range(t):
	(n, k) = [int(x) for x in input().split()]
	az = [int(x) for x in input().split()]
	max_n = []
	for i in range(len(az) - 1):
		max_n.append(10 ** (az[i + 1] - az[i]) - 1)
	res = 0
	for i in range(len(az) - 1):
		l = min(k + 1, max_n[i])
		k -= l
		res += l * 10 ** az[i]
		if k == -1:
			break
	if k >= 0:
		res += (k + 1) * 10 ** az[-1]
	print(res)
