#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x, i, s = 0, sum = 0, day;
  vector<int> v;
  cin >> n;
  for (i = 0; i < 7; i++) {
    cin >> x;
    v.push_back(x);
    s = s + v[i];
    if (s >= n) {
      day = i + 1;
      break;
    }
  }
  if (s < n) {
    sum = n / s * s;
    if (sum == n) {
      for (i = 6; i >= 0; i--) {
        if (v[i] > 0) {
          day = i + 1;
          break;
        }
      }
    } else {
      for (i = 0; i < 7; i++) {
        sum = sum + v[i];
        if (sum >= n) {
          day = i + 1;
          break;
        }
      }
    }
  }
  cout << day;
}
