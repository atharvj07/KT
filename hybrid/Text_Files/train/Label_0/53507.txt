import java.util.*;
import java.io.*;
import java.math.*;
import java.awt.geom.*;
public class Blocks{
    public static void main(String[] args) {
        MyScanner sc=new MyScanner();
        int a=sc.ni();
        int b=sc.ni();
        int ar[]=new int[a];
        int br[]=new int[b];
        for(int i=0;i<a;i++) ar[i]=sc.ni();
        for(int i=0;i<b;i++) br[i]=sc.ni();
        long s1=0,s2=0;
        int pa=0,pb=0,ans=0;
        while(pa<a && pb<b){
          if(s1==0 && s2==0){
            s1+=ar[pa];
            pa++;
          }
          else if(s1>s2){
            s2+=br[pb];
            pb++;
          }
          else if(s1<s2){
            s1+=ar[pa];
            pa++;
          }
          else if(s1==s2 && s1!=0){
            ans++;
            s1=0;
            s2=0;
          }
        }//end of while
        ans++;//for last pair 
        System.out.println(ans);
    }


    private static class MyScanner {
      BufferedReader br;
      StringTokenizer st;

      public MyScanner() {
         br = new BufferedReader(new InputStreamReader(System.in));
      }

      String next() {
          while (st == null || !st.hasMoreElements()) {
              try {
                  st = new StringTokenizer(br.readLine());
              } catch (IOException e) {
                  e.printStackTrace();
              }
          }
          return st.nextToken();
      }

      int ni() {
          return Integer.parseInt(next());
      }
      float nf() {
        return Float.parseFloat(next());
      }
      long nl() {
          return Long.parseLong(next());
      }

      double nd() {
          return Double.parseDouble(next());
      }

      String nextLine(){
          String str = "";
	  try {
	     str = br.readLine();
	  } catch (IOException e) {
	     e.printStackTrace();
	  }
	  return str;
      }
   }
}
