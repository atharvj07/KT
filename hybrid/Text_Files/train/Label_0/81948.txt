from collections import defaultdict
n = int(input())
a = list(map(int, input().split()))
ans1 = [-10000000000000000, -1, -1]
if a[0] >= 0:
	prefix = [a[0]]
else:
	prefix = [0]
for i in range(1, n):
	if a[i] >= 0:
		prefix.append(a[i] + prefix[-1])
	else:
		prefix.append(prefix[-1])
pos = defaultdict(list)
ans = [-1000000000000000000, -1, -1]
for i in range(n):
	if len(pos[a[i]]) == 0:
		pos[a[i]].append(i)
	elif len(pos[a[i]]) == 1:
		pos[a[i]].append(i)
		if pos[a[i]][0] != 0:
			tot = prefix[i - 1] - prefix[pos[a[i]][0]] + a[pos[a[i]][0]] + a[pos[a[i]][1]]
		else:
			tot = prefix[i - 1] - prefix[0] + a[pos[a[i]][0]] + a[pos[a[i]][1]]
		if ans[0] <= tot:
			ans[0] = tot
			ans[1] = pos[a[i]][0]
			ans[2] = i
	else:
		pos[a[i]][-1] = i
		if pos[a[i]][0] != 0:
			tot = prefix[i - 1] - prefix[pos[a[i]][0]] + a[pos[a[i]][0]] + a[pos[a[i]][1]]
		else:
			tot = prefix[i - 1] - prefix[0] + a[pos[a[i]][0]] + a[pos[a[i]][1]]
		if ans[0] <= tot:
			ans[0] = tot
			ans[1] = pos[a[i]][0]
			ans[2] = i
tree = []
for i in range(ans[1]):
	tree.append(i + 1)
for i in range(ans[1] + 1, ans[2]):
	if a[i] < 0:
		tree.append(i + 1)
for i in range(ans[2] + 1, n):
	tree.append(i + 1)
print(ans[0], len(tree))
print(*tree)
