#include <bits/stdc++.h>
using namespace std;
int a[500101], n, m;
int b[500101];
pair<int, int> s[500101];
multiset<int> mp;
signed main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  sort(a + 1, a + 1 + n, greater<int>());
  int l = 0;
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    if (a[i] != a[i - 1]) l++;
    s[l].first = a[i];
    s[l].second++;
    ans += a[i];
  }
  n = l;
  int size = 0;
  for (int i = 1; i <= n; i++) {
    int x = s[i].first, y = s[i].second;
    int last = size;
    size = min(m, (m + y) >> 1);
    int cnt = max(0, size - y);
    for (int j = size - 1; j >= cnt; j--)
      if (j < mp.size())
        b[j] = *mp.begin(), mp.erase(mp.begin());
      else
        b[j] = 0;
    for (int j = cnt, k = m - j; j < k && j < size; j++)
      if (b[j] < x)
        b[j] = x;
      else if (--k < size)
        b[k] = max(0, 2 * x - b[j]);
    mp.insert(b + cnt, b + size);
    m += y;
  }
  for (set<int>::iterator it = mp.begin(); it != mp.end(); it++) {
    ans -= *it;
  }
  cout << ans;
  return 0;
}
