for i in range(int(input())):
	a = input()
	bi = input().split()
	l = []
	for k in bi:
		l.append(int(k))
	alice = l[0]
	bob = 0
	a = l[0]
	l.pop(0)
	b = 0
	j = 1
	while len(l) != 0:
		temp = 0
		n = 0
		m = len(l) - 1
		if j % 2 == 0:
			if sum(l) > b:
				while temp <= b:
					temp += l[n]
					l.pop(n)
				a = temp
				alice += temp
			else:
				a += sum(l)
				alice += sum(l)
				j += 1
				break
		elif sum(l) > a:
			while temp <= a:
				temp += l[m]
				l.pop(m)
				m -= 1
			b = temp
			bob += temp
		else:
			b += sum(l)
			bob += sum(l)
			j += 1
			break
		j += 1
	print(j, alice, bob)
