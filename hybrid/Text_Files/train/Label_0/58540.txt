import java.util.*;
import java.io.*;
import java.math.*;


public class scorify{
	public static void main(String[] args){
	Scanner in=new Scanner(System.in);
    int points = in.nextInt();
    int jumps = in.nextInt();
    int position =0,myjumps=jumps,count=0;
    in.nextLine();
    String string = in.nextLine();
    
    while(points - (position+1) > jumps) {
        if(string.charAt(myjumps+position)=='1') {
            position+=myjumps;
            myjumps=jumps;
            count++;
        }else{
            if(myjumps==1){
                count=-2;
                break;
            }
            myjumps--;
        }
    }
    System.out.println(count+1);
	
	
	}
}
