#include <bits/stdc++.h>
using namespace std;
int main() {
  double vp, vd, t, f, c, accum = 0, time;
  int res = 0;
  cin >> vp >> vd >> t >> f >> c;
  accum = t;
  if (vd <= vp) {
    cout << 0 << endl;
    return 0;
  }
  while (accum * vp < c) {
    time = vp * accum / (vd - vp);
    if (vp * (accum + time) >= c) break;
    res++;
    accum += time * 2 + f;
  }
  cout << res << endl;
  return 0;
}
