from math import sqrt

def findclosest(a):
	a1 = 1
	b1 = 1
	c1 = -1 * 2 * a
	d = b1 ** 2 - 4 * a1 * c1
	x = (-1 * b1 + sqrt(d)) / 2
	if x - int(x) == 0:
		return [x]
	return [int(x), int(x + 1)]

def findclosest2(a):
	a1 = 1
	b1 = 1
	c1 = -1 * 2 * (a + 2)
	d = b1 ** 2 - 4 * a1 * c1
	x = (-1 * b1 + sqrt(d)) / 2
	if x - int(x) == 0:
		return [x]
	return [int(x), int(x + 1)]
for i in range(int(input())):
	x = int(input())
	l = int(findclosest(x)[0])
	s = sum(list(range(l + 1)))
	if x == s:
		print(l)
	else:
		print(l + 1 + (x - (s + l + 1) == -1))
