#include <bits/stdc++.h>
using namespace std;
int nian;
bool check(int p) {
  if (p % 400 == 0) return 1;
  if (p % 4 == 0 && p % 100 != 0) return 1;
  return 0;
}
int main() {
  long long tian = 0;
  cin >> nian;
  int jud = check(nian);
  for (int i = nian + 1;; i++) {
    int jud2 = check(i);
    if (jud2) {
      tian += 366;
    } else
      tian += 365;
    if (tian % 7 == 0 && jud2 == jud) {
      printf("%d\n", i);
      return 0;
    }
  }
  return 0;
}
