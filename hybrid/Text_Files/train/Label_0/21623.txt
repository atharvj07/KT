#include <bits/stdc++.h>
using namespace std;
long long t, x, y, p, q;
bool check(long long t) {
  return y <= q * t && x <= p * t && q * t - y >= p * t - x;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> t;
  while (t--) {
    cin >> x >> y >> p >> q;
    long long l = 1, r = 1000 * 1000 * 1000;
    while (r - l > 1) {
      long long c = (r + l) / 2;
      if (check(c)) {
        r = c;
      } else {
        l = c;
      }
    }
    if (check(l)) {
      r = l;
    }
    if (check(r)) {
      long long res = max(q * r - y, p * r - x);
      cout << res << endl;
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
