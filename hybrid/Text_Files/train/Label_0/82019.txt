(x1, y1, x2, y2) = map(int, input().split())
x2 -= x1
y2 -= y1
x1 = 0
y1 = 0
odd_x = (x2 - x1) // 2 + 1 - (1 if x1 % 2 == 0 and x2 % 2 == 0 else 0)
odd_y = (y2 - y1) // 2 + 1 - (1 if y1 % 2 == 0 and y2 % 2 == 0 else 0)
even_x = (x2 - x1) // 2 + 1 - (1 if x1 % 2 == 1 and x2 % 2 == 1 else 0)
even_y = (y2 - y1) // 2 + 1 - (1 if y1 % 2 == 1 and y2 % 2 == 1 else 0)
print(odd_x * odd_y + even_x * even_y)
