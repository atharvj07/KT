def solve(platforms, n):
	h = platforms[0]
	count = 0
	i = 0
	while i < n:
		if h <= 2:
			break
		if platforms[i] == h:
			i += 1
		if i < n and i + 1 < n:
			if platforms[i] == h - 1 and platforms[i + 1] == h - 2:
				h -= 2
				i += 1
			elif platforms[i] == h - 1:
				count += 1
				h -= 1
			elif platforms[i] == h - 2:
				h -= 1
			else:
				h = platforms[i] + 1
		elif i < n:
			if platforms[i] == h - 1:
				count += 1
				h -= 1
			elif platforms[i] == h - 2:
				h -= 1
			else:
				h = platforms[i] + 1
	print(count)

def main():
	q = int(input())
	for i in range(q):
		(h, n) = map(int, input().split())
		platforms = list(map(int, input().split()))
		solve(platforms, n)
main()
