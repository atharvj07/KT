#include <bits/stdc++.h>
using namespace std;
int x[2], y[2], r[2];
double dist() {
  return sqrt(1. *
              ((x[0] - x[1]) * (x[0] - x[1]) + (y[0] - y[1]) * (y[0] - y[1])));
}
int main() {
  cin >> x[0] >> y[0] >> r[0];
  cin >> x[1] >> y[1] >> r[1];
  double rr = dist();
  if (rr >= r[0] + r[1]) {
    rr -= r[0] + r[1];
    if (rr < 0) rr = 0;
  } else if (rr + r[1] < r[0])
    rr = r[0] - rr - r[1];
  else if (rr + r[0] < r[1])
    rr = r[1] - rr - r[0];
  else
    rr = 0;
  printf("%.6lf\n", rr / 2.0);
}
