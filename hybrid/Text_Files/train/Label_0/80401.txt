res = 1000000000.0

def next(i):
	i += 1
	while i < N and S[i] == '1':
		i += 1
	return i
(N, K) = map(int, input().split())
S = input()
l = next(-1)
m = l
r = l
for j in range(K):
	r = next(r)
while r < N:
	while max(m - l, r - m) > max(next(m) - l, r - next(m)):
		m = next(m)
	res = min(res, max(m - l, r - m))
	l = next(l)
	r = next(r)
print(res)
