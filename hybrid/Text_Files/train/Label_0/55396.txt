import java.util.*;
import java.io.*;

public class Main{

    BufferedReader in;
    StringTokenizer str = null;
    PrintWriter out;

    private String next() throws Exception{
    	while (str == null || !str.hasMoreElements())
    	    str = new StringTokenizer(in.readLine());
    	return str.nextToken();
    }
    
    private int nextInt() throws Exception{
	   return Integer.parseInt(next());
    }
    
    final static long MOD = (long)1e9 + 7;;
    Map<Integer, Integer> map1;
    Map<Integer, Integer> map2;
    
    public void run() throws Exception{
    	in = new BufferedReader(new InputStreamReader(System.in));
    	out = new PrintWriter(System.out);
	int n = nextInt(), m = nextInt();
	P[] st = new P[m];
	int []x = new int[2*m];
	int at = 0;
	for(int i = 0; i < m; ++i){
	    st[i] = new P(nextInt(), nextInt());
	    x[at++] = st[i].l;
	    x[at++] = st[i].r;
	}
	Arrays.sort(st);
	Arrays.sort(x);
	int id = 0;
	
	/*
	for(int i = 0; i < m; ++i){
	    System.out.println(st[i]);
	}
	*/
	map2 = new HashMap<Integer, Integer>();
	map1 = new HashMap<Integer, Integer>();
	for(int i = 0; i < x.length; ++i){
	    if (!map1.containsKey(x[i])){
		map1.put(x[i], id);
		map2.put(id++, x[i]);
	    }
	}
	
	if (!map1.containsKey(0) || !map1.containsKey(n)){
	    out.println(0);
	    out.close();
	    return;
	}
	/*
	for(int i : map1.keySet()){
	    System.out.println(i + " " + map1.get(i));
	}
	*/
	long []dp = new long[id];
	long []sum = new long[id];
	dp[0] = 1;
	sum[0] = 1;
	at = 0;
	for(int i = 1; i < id; ++i){
	    int r = map2.get(i);
	    while(at < m && st[at].r == r){
		long diff = sum[i-1];
		int prev = map1.get(st[at].l);
		if (prev > 0) diff-=sum[prev-1];
	    
		dp[i] = (dp[i] + diff + MOD) % MOD;
		at++;
	    }
	    sum[i] = (sum[i - 1] + dp[i]) % MOD;
	}
	//System.out.println(Arrays.toString(sum));
	//System.out.println(Arrays.toString(dp));
	out.println(dp[id-1]);
        out.close();
    }
    
    class P implements Comparable<P>{
	int l, r;
	
	public P(int l, int r){
	    this.l = l;
	    this.r = r;
	}
	
	public int compareTo(P other){
	    if (this.r > other.r) return 1;
	    if (this.r < other.r) return -1;
	    if (this.l > other.l) return 1;
	    if (this.l < other.l) return -1;
	    return 0;
	}
	
	public String toString(){
	    return this.l + " " + this.r;
	}
    }

    public static void main(String args[]) throws Exception{
	   new Main().run();
    }
}
