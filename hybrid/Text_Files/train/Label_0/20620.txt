#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9 + 7;
const long long INF = 1e18;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, s, d;
  cin >> n >> s >> d;
  int n1 = (s * 2 + 2) / 3;
  int n2 = s - n1;
  mt19937_64 gen(12892374);
  vector<unsigned long long> vv(n);
  for (unsigned long long& t : vv) t = gen();
  vector<unsigned long long> prs;
  for (int i = 0; i < s; i++) {
    int c;
    cin >> c;
    unsigned long long cur = 0;
    for (int j = 0; j < c; j++) {
      int x;
      cin >> x;
      cur ^= vv[x - 1];
    }
    prs.emplace_back(cur);
  }
  vector<pair<unsigned long long, int> > pv1, pv2;
  pv1.reserve((1 << n1) + 1);
  pv1.emplace_back(0, 0);
  for (int j = 0; j < n1; j++)
    for (int sz = pv1.size(), i = 0; i < sz; i++)
      pv1.emplace_back(pv1[i].first ^ prs[j], pv1[i].second + 1);
  pv2.reserve((1 << n2) + 1);
  pv2.emplace_back(0, 0);
  for (int j = n1; j < s; j++)
    for (int sz = pv2.size(), i = 0; i < sz; i++)
      pv2.emplace_back(pv2[i].first ^ prs[j], pv2[i].second + 1);
  sort((pv1).begin(), (pv1).end());
  for (int i = 0; i < d; i++) {
    int c;
    cin >> c;
    unsigned long long cur = 0;
    for (int j = 0; j < c; j++) {
      int x;
      cin >> x;
      cur ^= vv[x - 1];
    }
    int ans = inf;
    for (auto& t : pv2) {
      auto it = lower_bound((pv1).begin(), (pv1).end(),
                            pair<unsigned long long, int>(cur ^ t.first, 0));
      if (it != pv1.end() && it->first == (cur ^ t.first))
        ans = min(ans, it->second + t.second);
    }
    if (ans == inf) ans = -1;
    cout << ans << '\n';
  }
  return 0;
}
