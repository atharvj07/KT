mod = 1000000000.0 + 7

def Phi(n):
	phi = n
	i = 2
	while i * i <= n:
		if n % i == 0:
			while n % i == 0:
				n //= i
			phi -= phi // i
		i += 1
	if n > 1:
		phi -= phi // n
	return phi

def f(n):
	return Phi(n)

def g(n):
	return n
(n, k) = map(int, input().split())
ans = n
k = (k + 1) // 2
while k != 0:
	ans = Phi(ans)
	if ans == 1:
		break
	k -= 1
ans = int(ans % mod)
print(ans)
