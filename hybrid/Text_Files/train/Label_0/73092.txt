[start, end] = [x for x in input().split()]
duration = int(input())
cw = ['v', '<', '^', '>']
ccw = ['v', '>', '^', '<']
cw_bool = end == cw[(cw.index(start) + duration) % 4]
ccw_bool = end == ccw[(ccw.index(start) + duration) % 4]
if cw_bool and ccw_bool:
	print('undefined')
elif cw_bool:
	print('cw')
else:
	print('ccw')
