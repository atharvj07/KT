def resolve(n, m, pairs):
	delta = []
	(l, r) = (0, 0)
	for (a, b) in pairs:
		delta.append([0, 0, l, r])
		(da, db) = (min(a, m), min(b, m))
		delta[-1][1] = b - (m - da) - (a - da)
		delta[-1][0] = b - db - (a - (m - db))
		l += delta[-1][0]
		r += delta[-1][1]
	if r <= 0 or l >= 0:
		x = 1 if r <= 0 else 0
		ret = [str(-r if r <= 0 else l)]
		for i in range(n):
			(ab, [a, b]) = (sum(pairs[i]) - m, pairs[i])
			ret.append(f'{a - (ab - delta[i][x] >> 1)} {b - (ab + delta[i][x] >> 1)}')
	else:
		ret = ['1' if r & 1 else '0'] + [None] * n
		tot = 0
		for i in range(n - 1, -1, -1):
			x = max(-(delta[i][3] + tot), delta[i][0])
			x += x & 1 ^ delta[i][0] & 1
			(ab, [a, b]) = (sum(pairs[i]) - m, pairs[i])
			ret[i + 1] = f'{a - (ab - x >> 1)} {b - (ab + x >> 1)}'
			tot += x
	return '\n'.join(ret)
t = int(input())
ans = []
for _ in range(t):
	input()
	(n, m) = [int(_) for _ in input().split()]
	ans.append(resolve(n, m, [[int(_) for _ in input().split()] for _ in range(n)]))
print('\n'.join(ans))
