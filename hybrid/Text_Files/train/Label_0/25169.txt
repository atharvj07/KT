#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c[105], a[105][105];
  cin >> n;
  int s = 0;
  for (int i = 1; i <= n; i++) cin >> c[i], s += c[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= c[i]; j++) {
      cin >> a[i][j];
    }
  }
  vector<int> q, w, e, r;
  int x = 1, y = 1, ii = 1;
  for (int k = 1; k <= s; k++) {
    int x1, y1, kk = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= c[i]; j++) {
        kk++;
        if (a[i][j] == k) {
          x1 = i;
          y1 = j;
          goto l;
        }
      }
    }
  l:;
    if (kk != k) {
      q.push_back(x), w.push_back(y), e.push_back(x1), r.push_back(y1);
      swap(a[x][y], a[x1][y1]);
    }
    y++;
    if (y > c[ii]) {
      ii++, y = 1, x++;
    }
  }
  cout << q.size() << endl;
  for (int i = 0; i < q.size(); i++) {
    cout << q[i] << " " << w[i] << " " << e[i] << " " << r[i] << endl;
  }
}
