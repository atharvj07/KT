import sys
from array import array
import typing as Tp

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def output(*args):
	sys.stdout.buffer.write(('\n'.join(map(str, args)) + '\n').encode('utf-8'))

def main():
	(n, k) = map(int, input().split())
	wave = [tuple(map(int, input().split())) for _ in range(n)]
	ans = inf = 10 ** 18
	dp = [0] + [inf] * (n - 1)
	for i in range(n):
		if dp[i] == inf:
			continue
		time = wave[i][0]
		bullet = k
		cost = dp[i]
		for j in range(i, n):
			time = max(time, wave[j][0])
			if bullet >= wave[j][2]:
				bullet -= wave[j][2]
			else:
				ub = wave[j][1] - time
				reload = (wave[j][2] - bullet - 1) // k + 1
				if ub < reload:
					break
				time += reload
				bullet = (-wave[j][2] + bullet) % k
			cost += wave[j][2]
			if j < n - 1 and time < wave[j + 1][0]:
				dp[j + 1] = min(dp[j + 1], cost + bullet)
		else:
			ans = min(ans, cost)
	print(ans if ans < inf else -1)
main()
