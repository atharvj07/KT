import sys, collections
sys.setrecursionlimit(10000)
d = collections.defaultdict(int)

def go(a, b):
	(res, c) = (0, a + b)
	if d.get(c) and d[c] > 0:
		d[c] -= 1
		res = go(b, c) + 1
		d[c] += 1
	return res
input()
for i in map(int, input().split()):
	d[i] += 1
ans = 2
for a in d:
	for b in d:
		if a != b or d[a] > 1:
			d[a] -= 1
			d[b] -= 1
			cnt = go(a, b) + 2
			d[a] += 1
			d[b] += 1
			ans = max(cnt, ans)
print(ans)
