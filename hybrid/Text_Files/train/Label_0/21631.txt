#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  int a[102];
  scanf("%d", &n);
  int mmax = 0;
  int mmin = 100000;
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
    if (a[i] > mmax) mmax = a[i];
    if (a[i] < mmin) mmin = a[i];
  }
  sort(a, a + n);
  for (int i = 0; i < n; i++) {
    if (i != 0) printf(" ");
    if (i == 0)
      printf("%d", a[n - 1]);
    else if (i == n - 1)
      printf("%d", a[0]);
    else
      printf("%d", a[i]);
  }
  return 0;
}
