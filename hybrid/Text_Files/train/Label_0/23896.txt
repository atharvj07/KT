#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int n, m;

int main()
{
	cin >> n >> m;

	vector<queue<int> > P(n);
	vector<int> Q;

	for (int i = 0; i < m; i++) {
		int t, u;

		cin >> t >> u;

		if (t == 0) {
			Q.push_back(P[u - 1].front());
			P[u - 1].pop();
		}
		else if (t == 1) {
			int w = P[0].size(), j = 0, jm = 0;

			while (j < n) {
				if (w > P[j].size()) {
					jm = j;
					w = P[j].size();
				}

				j++;
			}

			P[jm].push(u);
		}
	}

	for (int i = 0; i < Q.size(); i++)
		cout << Q[i] << endl;

	return 0;
}
