t1 = input()
h1 = int(t1[0] + t1[1])
m1 = int(t1[3] + t1[4])
t2 = input()
h2 = int(t2[0] + t2[1])
m2 = int(t2[3] + t2[4])
mid = (60 * h2 + m2 - (60 * h1 + m1)) // 2
t = 60 * h1 + m1 + mid
if len(str(t // 60)) == 1:
	print(0, end='')
	print(t // 60, end='')
else:
	print(t // 60, end='')
print(':', end='')
if len(str(t % 60)) == 1:
	print(0, end='')
	print(t % 60)
else:
	print(t % 60)
