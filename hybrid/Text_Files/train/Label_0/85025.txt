import sys

def maximum_goods(prices, n, p, k):
	ans = 0
	dp = [[0, 0] for _ in range(n + 1)]
	for i in range(1, n + 1):
		if i < k:
			dp[i][0] = dp[i - 1][0] + prices[i - 1]
			dp[i][1] = dp[i][0]
		else:
			dp[i][0] = prices[i - 1] + min(dp[i - 1][0], dp[i - 1][1])
			dp[i][1] = prices[i - 1] + min(dp[i - k][0], dp[i - k][1])
		if p >= min(dp[i][0], dp[i][1]):
			ans = max(ans, i)
	return ans

def main():
	t = int(sys.stdin.readline()[:-1])
	for _ in range(t):
		[n, p, k] = [int(s) for s in sys.stdin.readline()[:-1].split()]
		prices = [int(s) for s in sys.stdin.readline()[:-1].split()]
		print(maximum_goods(sorted(prices), n, p, k))
main()
