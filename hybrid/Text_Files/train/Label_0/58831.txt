

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;

public class ProblemB {

    public static void main(String[] args) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw=new BufferedWriter(new OutputStreamWriter(System.out));

        int r,c,n,m,q,arr[][],ans,rowMax,count,rowCount[];
        String line[]=br.readLine().split(" ");
        n=ip(line[0]);
        m=ip(line[1]);
        q=ip(line[2]);
        arr=new int[n][m];
        rowCount=new int[n];
        
        for(int i=0;i<n;i++){
            line=br.readLine().split(" ");
            for(int j=0;j<m;j++){
                arr[i][j]=ip(line[j]);
            }
        }
        
        ans=0;
                
        for(int a=0;a<n;a++){
            rowMax=0;
            count=0;
            
            for(int b=0;b<m;b++){
                                
                if(arr[a][b]==1)
                    {count++;
                    if(count>rowMax)
                        rowMax=count;
                    }
                else
                        count=0;
                
            }
                rowCount[a]=rowMax;
        }
        
        for(int i=0;i<q;i++){
            
            line=br.readLine().split(" ");
            r=ip(line[0]);
            c=ip(line[1]);
            
            if(arr[r-1][c-1]==0)
                arr[r-1][c-1]=1;
            else
                arr[r-1][c-1]=0;
            
            count=rowMax=0;
            
            for(int b=0;b<m;b++){
                
                if(arr[r-1][b]==1)
                    {count++;
                    if(count>rowMax)
                        rowMax=count;
                    }
                else
                        count=0;
            }
                rowCount[r-1]=rowMax;
            
            ans=0;  
            
            for(int p=0;p<n;p++){
                ans=Math.max(rowCount[p], ans);
            }
                        
            bw.write(Integer.toString(ans));
            bw.newLine();
        }
        
        br.close();
        bw.flush();
        bw.close();
    }

public static int ip(String s){
        
        return Integer.parseInt(s);
    }
}
