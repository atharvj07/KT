num_data = list(map(int, input().split(' ')))
square_data = list(map(int, input().split(' ')))
column_count = num_data[0]
number_count = num_data[1]
platform = [0] * column_count
score = 0
for i in square_data:
	col_idx = i - 1
	platform[col_idx] += 1
	all_fill = True
	for j in platform:
		all_fill = all_fill and j > 0
	if all_fill:
		score += 1
		for idx in range(0, column_count):
			platform[idx] -= 1
print(score)
