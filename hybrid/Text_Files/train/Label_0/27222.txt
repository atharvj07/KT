int N,M,T,P,Q;
char field[1010][1010];
char pattern[1010][1010];
ll hash_[1010][1010],tmp[1010][1010];

void compute_hash(char a[1010][1010],int n,int m){
    const ull B1=9973;
    const ull B2=100000007;
    ull t1=1;
    for(int j=0;j<Q;j++)t1*=B1;
    for(int i=0;i<n;i++){
        ull e=0;
        for(int j=0;j<Q;j++)e=e*B1+a[i][j];
        for(int j=0;j+Q<=m;j++){
            tmp[i][j]=e;
            if(j+Q<m)e=e*B1-t1*a[i][j]+a[i][j+Q];
        }
    }
    ull t2=1;
    for(int i=0;i<P;i++)t2*=B2;
    for(int j=0;j+Q<=m;j++){
        ull e=0;
        for(int i=0;i<P;i++)e=e*B2+tmp[i][j];
        for(int i=0;i+P<=n;i++){
            hash_[i][j]=e;
            if(i+P<n)e=e*B2-t2*tmp[i][j]+tmp[i+P][j];
        }
    }
}

    compute_hash(pattern,P,Q);
    ull ttmp=hash_[0][0];
    compute_hash(field,N,M);

            }
