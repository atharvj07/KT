t = int(input())
for _ in range(t):
	(n, m, k) = map(int, input().strip().split())
	if k % 2 == 0:
		if n % 2 == 0:
			if (n * m - n) / 2 < k and m & 1:
				print('NO')
			else:
				print('YES')
		elif k >= m / 2 and (k - m // 2) % 2 == 0:
			print('YES')
		else:
			print('NO')
	elif n % 2 == 0:
		print('NO')
	elif k >= m / 2 and (k - m // 2) % 2 == 0:
		print('YES')
	else:
		print('NO')
