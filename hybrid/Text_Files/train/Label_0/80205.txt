(n, k) = map(int, input().split())
l = list(map(int, input().split()))
l.sort()
s = 0
c = 1
v = []
for i in range(n // 2 + 1, n):
	v.append(c * (l[i] - l[i - 1]))
	c += 1
ans = 0
for i in range(len(v)):
	if k > 0 and v[i] > 0:
		m = min(v[i], k)
		ans += m // (i + 1)
		k -= m
if k > 0:
	ans += k // (len(v) + 1)
print(ans + l[n // 2])
