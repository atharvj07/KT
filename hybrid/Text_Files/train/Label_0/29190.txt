#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
long long isLcm(long long a, long long b) { return ((a * b) / gcd(a, b)); }
int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  long long n, count = 1, mini = 1e13;
  cin >> n;
  if (n == 1) {
    cout << 1 << " " << 1 << endl;
    return 0;
  }
  while (count * count <= n) {
    if (n % count == 0 && isLcm(count, n / count) == n) {
      long long maxi = max(count, n / count);
      mini = min(mini, maxi);
    }
    count++;
  }
  cout << mini << " " << n / mini << endl;
  return 0;
}
