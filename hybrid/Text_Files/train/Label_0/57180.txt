import java.util.*;
import java.io.*;
public class A
{
      public static void main(String ar[]) throws Exception
      {
            BufferedReader  br=new BufferedReader(new InputStreamReader(System.in));
            int n=Integer.parseInt(br.readLine());
            int a[]=new int[n];
            TreeMap<Integer,Integer> tm=new TreeMap<Integer,Integer>();
            String s1[]=br.readLine().split(" ");
            for(int i=0;i<n;i++)
            {
                  a[i]=Integer.parseInt(s1[i]);
                  if(tm.containsKey(a[i]))
                   tm.put(a[i],1+tm.get(a[i]));
                  else
                   tm.put(a[i],1);
            }
            int y=0,z=0;  int ynum=0,znum=0;
            int x=tm.firstKey();  int xnum=tm.get(x); tm.remove(x);
            if(!tm.isEmpty())
            {
            y=tm.firstKey();      ynum=tm.get(y); tm.remove(y);
            }
            if(!tm.isEmpty())
            {
            z=tm.firstKey();      znum=tm.get(z); tm.remove(z);
            }
            
            long m=0;
            if(xnum>=3)
            {
                 m=(long)(xnum);  m=(m*(m-1)*(m-2))/6;   
            }
            else if(xnum==2)
            {
                 m=(long)(ynum);
            }
            else if(xnum==1 && ynum>=2)
            {
                m=(long)(ynum);  m=(m*(m-1))/2;  
            }
            else if(xnum==1 && ynum==1)
            {
                m=(long)(znum);
            }
            
            System.out.println(m);
      }
}
