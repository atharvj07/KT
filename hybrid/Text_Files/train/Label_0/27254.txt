#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
int floyd[110][110];
bool linked[110][110];
int seq[1000010];
vector<int> ans;
int n, m;
void refresh(int k, int i, int j) {
  if (floyd[i][k] + floyd[k][j] < floyd[i][j]) {
    floyd[i][j] = floyd[i][k] + floyd[k][j];
  }
}
int main() {
  cin >> n;
  string inp;
  for (int i = 1; i <= n; i++) {
    cin >> inp;
    for (int j = 0; j < inp.size(); j++) {
      if (inp[j] == '1') {
        floyd[i][j + 1] = 1;
      } else {
        if (j + 1 != i) floyd[i][j + 1] = inf;
      }
    }
  }
  for (int k = 1; k <= n; k++) {
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        refresh(k, i, j);
      }
    }
  }
  cin >> m;
  for (int i = 1; i <= m; i++) {
    cin >> seq[i];
  }
  ans.push_back(seq[1]);
  for (int i = 1;;) {
    int j;
    for (j = i; j <= m + 1; j++) {
      if (floyd[seq[i]][seq[j]] != j - i) break;
    }
    j--;
    ans.push_back(seq[j]);
    i = j;
    if (i == m) {
      break;
    }
  }
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i] << " ";
  }
  return 0;
}
