
import java.io.*;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;





public class Main {

    static ArrayList<Integer> graph[];
    static ArrayList<Integer> res=new ArrayList<Integer>();
    public static void main(String[] args) throws IOException {
        // TODO code application logic here
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(br.readLine());
        Map<pair,Integer> hmap=new HashMap<pair,Integer>();
        graph=new ArrayList[256*256];
        for(int i=0;i<256*256;i++){
            graph[i]=new ArrayList<Integer>();
        }
        String[] s = new String[n];
        for (int i = 0; i < n; i++)
            s[i] = br.readLine();
        int [] in=new int[256*256];
        int [] out=new int[256*256];
        
        for(int i=0;i<n;i++){
            
            int prefix=256*s[i].charAt(0)+s[i].charAt(1);
            int suffix=256*s[i].charAt(1)+s[i].charAt(2);
            in[suffix]++;
            out[prefix]++;
            hmap.put(new pair(prefix,suffix), i);
            graph[prefix].add(suffix);
            
            
        }
        int dfsstart=0;
        for(int i=0;i<256*256;i++){
                if(out[i]>0)
                    dfsstart=i;}
        
        for(int i=0;i<256*256;i++){        
        if(out[i]>in[i])
                    dfsstart=i;
            }
        int cnt1=0,cnt2=0;
        boolean chk=true;
        for(int i=0;i<256*256;i++){
            if (out[i] - in[i] == 1)
                cnt1++;
            if (in[i] - out[i] == 1)
                cnt2++;
            if (Math.abs(in[i] - out[i]) >= 2)
                chk = false;
        }
        if(cnt1>1 || cnt2>1)
            chk=false;
        dfs(dfsstart);
        Collections.reverse(res);
        if (!chk || res.size() != n + 1) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 1; i <= n; i++) {
            int a = res.get(i - 1);
            int b = res.get(i);
            pair p=new pair(a,b);
            //System.out.println(i +" "+a+" "+b);
            String curr = s[hmap.get(p)];
            if (i == 1)
                System.out.print(curr);
            else
                System.out.print(curr.charAt(2));
       
                    }
        System.out.println();
    }
    
    public static void dfs(int v){
        while (graph[v].size() > 0) {
            int to = graph[v].get(graph[v].size() - 1);
            graph[v].remove(graph[v].size() - 1);
            dfs(to);
        }
        res.add(v);
    }
}
class pair implements Comparable<pair> {
 int pre;
 int suf;
 public pair(int pre,int suf){
     this.pre=pre;
     this.suf=suf;
 }

    @Override
    public int compareTo(pair o) {
      if (pre != o.pre)
            return Integer.compare(pre, o.pre);
        return Integer.compare(suf, o.suf);  
    }
    public int hashCode() {
        return pre * 1_000_003 + suf;
    }

    public boolean equals(Object o) {
        if (o == null || !(o instanceof pair))
            return false;
        pair p = (pair) o;
        return pre == p.pre && suf == p.suf;
    }
}
