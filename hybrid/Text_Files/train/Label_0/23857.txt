#include <bits/stdc++.h>
using namespace std;
int no_dec(int a[], int n) {
  int cnt = 1;
  for (int i = 0; i < n - 1; i++) {
    if (a[i] <= a[i + 1]) {
      cnt++;
    } else
      break;
  }
  return cnt;
}
int no_inc(int a[], int n) {
  int cnt = 1;
  for (int i = 0; i < n - 1; i++) {
    if (a[i] >= a[i + 1]) {
      cnt++;
    } else
      break;
  }
  return cnt;
}
int hill(int a[], int n) {
  int cnt = 1;
  int i = 0;
  for (; i < n - 1; i++) {
    if (a[i] <= a[i + 1]) {
      cnt++;
    } else
      break;
  }
  for (; i < n - 1; i++) {
    if (a[i] >= a[i + 1]) {
      cnt++;
    } else
      break;
  }
  return cnt;
}
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n;
    scanf("%d", &n);
    int a[n];
    for (int &i : a) scanf("%d", &i);
    reverse(a, a + n);
    int ans = max(no_dec(a, n), max(no_inc(a, n), hill(a, n)));
    printf("%d\n", n - ans);
  }
}
