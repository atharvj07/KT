template <class T>
istream &operator>>(istream &in, vector<T> &arr) {
  for (T &cnt : arr) {
    in >> cnt;
  }
  return in;
};
vector<pair<long long, long long> > ban;
bool in(pair<long long, long long> x) {
  for (auto i : ban) {
    if (i == x) return true;
  }
  return false;
}
void solve() {
  int k;
  cin >> k;
  if (k % 2 == 0) {
    cout << "NO";
    exit(0);
  }
  cout << "YES\n";
  if (k == 1) {
    cout << "2 1\n1 2\n";
    exit(0);
  }
  int n = k + 2;
  ban.emplace_back(0, n - 1);
  for (int i = 0; i < n - 1; i += 2) {
    ban.emplace_back(i, i + 1);
  }
  cout << 2 * n << " " << n * k << "\n";
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      if (!in(make_pair(i, j))) {
        cout << i + 1 << " " << j + 1 << "\n";
        cout << n + i + 1 << " " << n + j + 1 << "\n";
      }
    }
  }
  cout << 1 << " " << n + 1 << "\n";
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  solve();
  return 0;
}
