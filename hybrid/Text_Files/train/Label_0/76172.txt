def is_prefix(long, short):
	return short == long[:len(short)]

def is_suffix(long, short):
	return short == long[-len(short):]

def main():
	n = int(input())
	l = []
	shorts = []
	longs = []
	if n == 2:
		_ = input()
		_ = input()
		print('PS')
		return
	for i in range(2 * n - 2):
		inp = input()
		l.append(inp)
		if len(inp) == 1:
			shorts.append(inp)
		elif len(inp) == n - 1:
			longs.append(inp)
	(s1, s2) = shorts
	(l1, l2) = longs
	d = {}
	if s1 + l1 == l2 + s2:
		d[s1 + l1] = [[], {}]
	if s1 + l2 == l1 + s2:
		d[s1 + l2] = [[], {}]
	if s2 + l1 == l2 + s1:
		d[s2 + l1] = [[], {}]
	if s2 + l2 == l1 + s1:
		d[s2 + l2] = [[], {}]
	for wo in l:
		keys = list(d.keys())
		z = 0
		while True:
			if z >= len(keys):
				break
			k = keys[z]
			if is_prefix(k, wo):
				if wo in d[k][1] and d[k][1][wo] == 0:
					if is_suffix(k, wo):
						d[k][0].append('S')
					else:
						del d[k]
				else:
					d[k][1][wo] = 0
					d[k][0].append('P')
			elif is_suffix(k, wo):
				if wo in d[k][1] and d[k][1][wo] == 1:
					del d[k]
				else:
					d[k][1][wo] = 1
					d[k][0].append('S')
			else:
				del d[k]
			z += 1
	print(''.join(d[list(d.keys())[0]][0]))
main()
