#include <bits/stdc++.h>
using namespace std;
vector<int> adjList[100010];
int n;
namespace genLCA {
const int LOGN = 20;
int level[100010];
int DP[LOGN][100010];
void init_dfs(int u, int p) {
  for (auto it = adjList[u].begin(); it != adjList[u].end(); it++)
    if (*it != p) {
      DP[0][*it] = u;
      level[*it] = level[u] + 1;
      init_dfs(*it, u);
    }
}
void preprocess() {
  level[0] = DP[0][0] = 0;
  init_dfs(0, 0);
  for (int i = 1; i < LOGN; i++)
    for (int j = 0; j < n; j++) DP[i][j] = DP[i - 1][DP[i - 1][j]];
}
int lca(int a, int b) {
  if (level[a] > level[b]) swap(a, b);
  int diff = level[b] - level[a];
  for (int i = 0; i < LOGN; i++)
    if ((diff & (1 << i))) b = DP[i][b];
  if (a == b) return a;
  for (int i = LOGN - 1; i >= 0; i--)
    if (DP[i][a] != DP[i][b]) a = DP[i][a], b = DP[i][b];
  return DP[0][a];
}
int dist(int a, int b) { return level[a] + level[b] - 2 * level[lca(a, b)]; }
}  // namespace genLCA
using namespace genLCA;
int cnt[100010];
void solve_post(int u, int p) {
  for (auto it = adjList[u].begin(); it != adjList[u].end(); it++)
    if ((*it) != p) {
      solve_post(*it, u);
      cnt[u] += cnt[*it];
    }
}
int a[100010], b[100010];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int q, u, v;
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    cin >> u >> v;
    u -= 1, v -= 1;
    a[i] = u;
    b[i] = v;
    adjList[u].push_back(v);
    adjList[v].push_back(u);
  }
  preprocess();
  cin >> q;
  while (q--) {
    cin >> u >> v;
    u -= 1, v -= 1;
    cnt[u]++, cnt[v]++;
    int LCA = lca(u, v);
    cnt[LCA] -= 2;
  }
  solve_post(0, 0);
  for (int i = 0; i < n - 1; i++) {
    u = a[i], v = b[i];
    if (level[u] < level[v]) swap(u, v);
    cout << cnt[u] << " ";
  }
  return 0;
}
