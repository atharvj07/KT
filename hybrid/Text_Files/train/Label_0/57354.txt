import java.util.Scanner;

public class Cc404 {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		
		long n = scan.nextLong();
		long m = scan.nextLong();
		
		if (m >= n) {
			System.out.print(n);
			System.exit(0);
		}
		
		long ans = m;
		
		long s = 2 * (n - m);
		
		double kdouble = (Math.sqrt(1 + s * 4)) / 2;
		
		long k = (long) kdouble;
		for (long l = k - 1000; l < k + 1000; l++) {
			if (l <= 0) {
				continue;
			}
			long side = l * l + l;
			if (side >= s) {
				System.out.print(l + ans);
				System.exit(0);
			}
		}
	}
}
