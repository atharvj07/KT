from itertools import count

class Child:

	def __init__(self, id, v, d, p):
		self.id = id
		self.v = v
		self.d = d
		self.p = p
n = int(input())
children = [None] * n
for i in range(n):
	(v, d, p) = map(int, input().split())
	children[i] = Child(i, v, d, p)

def decrease(pos, v, regressive):
	for i in range(pos + 1, n):
		if i >= len(children) or v == 0:
			break
		children[i].p -= v
		if regressive:
			v -= 1
result = []
cur = 0
while cur < len(children):
	result.append(children[cur].id + 1)
	decrease(cur, children[cur].v, True)
	i = cur + 1
	while i < len(children):
		if children[i].p < 0:
			decrease(i, children[i].d, False)
			children.pop(i)
			i -= 1
		i += 1
	cur += 1
print(len(result))
print(' '.join(map(str, result)))
