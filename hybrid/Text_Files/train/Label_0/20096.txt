#include <stdio.h>
#include <string.h>
#include <algorithm>
#include <iostream>
#include <math.h>
#include <assert.h>
#include <vector>
#include <complex>

using namespace std;
typedef long long ll;
typedef unsigned int uint;
typedef unsigned long long ull;
static const double EPS = 1e-9;
static const double PI = acos(-1.0);

#define REP(i, n) for (int i = 0; i < (int)(n); i++)
#define FOR(i, s, n) for (int i = (s); i < (int)(n); i++)
#define FOREQ(i, s, n) for (int i = (s); i <= (int)(n); i++)
#define FORIT(it, c) for (__typeof((c).begin())it = (c).begin(); it != (c).end(); it++)
#define MEMSET(v, h) memset((v), h, sizeof(v))

typedef complex<double> Point;

enum {
  INSIDE,
  OUTSIDE,
  INTERSECT,
  OTHER
};

int n;
Point target[2];
int kind[1100];
int value[1100];
Point point[1100];
double radius[1100];
int edge[1001][1001];

inline double square(double x) { return x * x; }

int dfs(int index) {
  if (value[index] != -1) { return value[index]; }
  int ret = 1;
  REP(i, n) {
    if (edge[index][i] != INSIDE) { continue; }
    ret = max(ret, dfs(i) + 1);
  }
  return value[index] = ret;
}

int main() {
  while (scanf("%d", &n), n) {
    MEMSET(value, -1);
    MEMSET(edge, -1);
    REP(i, 2) {
      double x, y;
      scanf("%lf %lf", &x, &y);
      target[i] = Point(x, y);
    }
    int m = 0;
    REP(i, n) {
      double x, y, r;
      scanf("%lf %lf %lf", &x, &y, &r);
      Point p = Point(x, y);
      double r2 = r * r;
      Point vect = target[0] - p;
      double t12 = vect.real() * vect.real() + vect.imag() * vect.imag();
      vect = target[1] - p;
      double t22 = vect.real() * vect.real() + vect.imag() * vect.imag();
      if (t12 == r2 || t22 == r2) { continue; }
      if ((t12 >= r2 && t22 >= r2) ||
          (t12 <= r2 && t22 <= r2)) { continue; }
      kind[m] = abs(target[0] - p) < r ? 0 : 1;
      point[m] = p;
      radius[m] = r;
      m++;
    }
    n = m;
    REP(i, n) {
      REP(j, n) {
        if (i == j) {
          edge[i][j] = INTERSECT;
          continue;
        }
        Point vect = point[i] - point[j];
        double d2 = vect.real() * vect.real() + vect.imag() * vect.imag();
        if (radius[i] > radius[j] && square(radius[i]  - radius[j]) > d2) { edge[i][j] = INSIDE; }
        else if (radius[j] > radius[i] && square(radius[j] - radius[i]) > d2) { edge[i][j] = OUTSIDE; }
        else if (square(radius[i] + radius[j]) >= d2 || ((square(radius[i]) >= d2 || square(radius[j]) >= d2) && square(radius[i] - radius[j]) <= d2)) { edge[i][j] = INTERSECT; }
        else { edge[i][j] = OTHER; }
      }
    }
    REP(i, m) { dfs(i); }
    int ans = 0;
    REP(i, m) {
      ans = max(ans, value[i]);
      REP(j, m) {
        if (kind[i] == kind[j]) { continue; }
        if (edge[i][j] != OTHER) { continue; }
        ans = max(ans, value[i] + value[j]);
      }
    }
    printf("%d\n", ans);
  }
}
