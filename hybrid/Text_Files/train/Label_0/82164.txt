n = int(input())
count = 0
while n > 0:
	if n >= 100:
		count = count + int(n / 100)
		n = n % 100
	elif n >= 20:
		count = count + int(n / 20)
		n = n % 20
	elif n >= 10:
		count = count + int(n / 10)
		n = n - 10
	elif n >= 5:
		count = count + int(n / 5)
		n = n % 5
	elif n < 5:
		count = count + n
		n = 0
	else:
		break
print(count)
