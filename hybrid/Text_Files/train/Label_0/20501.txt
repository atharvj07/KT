#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0; i<(n); i++)


int main(){
	int N,M;
	cin >> N >> M;
	vector<int> H(N);
	rep(i,N)cin >> H[i];
	vector<int> C(N);
	rep(i,M){
		int a,b; cin >> a >> b; a--; b--;
		if(H[a]>H[b])C[b]=-1;
		if(H[b]>H[a])C[a]=-1;
		if(H[a] == H[b]){C[a]=-1; C[b]=-1; }
	}

	int ans = count(C.begin(),C.end(),0);
	cout << ans << endl;

	return 0;
}
