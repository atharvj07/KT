(n, m) = [int(j) for j in input().split()]
nums = [int(j) for j in input().split()]
(freq, ref) = (100000 * [0], n * [0])
count = 0
for j in range(n - 1, -1, -1):
	if freq[nums[j] - 1] == 0:
		count += 1
	freq[nums[j] - 1] += 1
	ref[j] = count
for j in range(m):
	l = int(input())
	print(ref[l - 1])
