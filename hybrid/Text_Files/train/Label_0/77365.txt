def lower_bound(a, x):
	l = -1
	r = len(a)
	while l + 1 != r:
		mid = l + r >> 1
		if a[mid] < x:
			l = mid
		else:
			r = mid
	return r

def upper_bound(a, x):
	l = -1
	r = len(a)
	while l + 1 != r:
		mid = l + r >> 1
		if a[mid] <= x:
			l = mid
		else:
			r = mid
	return r

def solve():
	n = get()
	s1 = input()
	s2 = input()
	ans = 1000000000.0
	c1 = s1.count('1')
	c2 = s2.count('1')
	if c1 == c2:
		cnt = 0
		for i in range(n):
			if s1[i] != s2[i]:
				cnt += 1
		ans = min(ans, cnt)
	if c1 + c2 == n + 1:
		cnt = 1
		flag = 0
		for i in range(n):
			if s1[i] == '1' and s2[i] == '1':
				flag = i
		for i in range(n):
			if i != flag:
				if s1[i] == '1':
					s1 = s1[:i] + '0' + s1[i + 1:]
				else:
					s1 = s1[:i] + '1' + s1[i + 1:]
		for i in range(n):
			if s1[i] != s2[i]:
				cnt += 1
		ans = min(ans, cnt)
	if ans == 1000000000.0:
		print(-1)
	else:
