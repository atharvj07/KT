import sys, heapq
for _ in range(int(sys.stdin.readline())):
	n = int(sys.stdin.readline())
	d = [[] for i in range(n)]
	h = []
	l = []
	for i in range(n):
		y = [int(i) for i in sys.stdin.readline().split()]
		l.append(y[0])
		if y[0] == 0:
			heapq.heappush(h, i)
		for j in range(1, y[0] + 1):
			d[y[j] - 1].append(i)
	c = t = 0
	while h:
		i = heapq.heappop(h)
		t += 1
		c = i // n
		i = i % n
		for j in d[i]:
			l[j] -= 1
			if l[j] == 0:
				heapq.heappush(h, j + c * n if j > i else j + n + c * n)
	sys.stdout.write('%d\n' % (c + 1 if t == n else -1))
