    static int N = 1005;
    int[] a = new int[N];
    double[][] inv = new double[N][N], inv2 = new double[N][N];

    void work() throws Exception {
        int n = nextInt(), m = nextInt();
        for(int i = 0; i < n; i ++) {
            a[i] = nextInt();
        }
        for(int i = 0; i < n; i ++) {
            for(int j = i+1; j < n; j ++) {
                if(a[i] > a[j]) inv[i][j] = 1;
                else inv[i][j] = 0;
            }
        }
        
        for(int i = 0; i < m; i ++) {
            int a = nextInt(), b = nextInt();
            a --; b --;
            if(a > b) {
                int tmp = a;
                a = b;
                b = tmp;
            }
            
            for(int j = 0; j < a; j ++) {
                inv2[j][a] = (inv[j][a] + inv[j][b]) / 2.0;
                inv2[j][b] = (inv[j][a] + inv[j][b]) / 2.0;
            }
            for(int j = a+1; j < b; j ++) {
                inv2[a][j] = (inv[a][j] + 1-inv[j][b]) / 2.0;
                inv2[j][b] = (inv[j][b] + 1-inv[a][j]) / 2.0;
            }
            for(int j = b+1; j < n; j ++) {
                inv2[a][j] = (inv[a][j] + inv[b][j]) / 2.0;
                inv2[b][j] = (inv[a][j] + inv[b][j]) / 2.0;
            }
            
            for(int j = 0; j < a; j ++) {
                inv[j][a] = inv2[j][a];
                inv[j][b] = inv2[j][b];
            }
            for(int j = a+1; j < b; j ++) {
                inv[a][j] = inv2[a][j];
                inv[j][b] = inv2[j][b];
            }
            for(int j = b+1; j < n; j ++) {
                inv[a][j] = inv2[a][j];
                inv[b][j] = inv2[b][j];
            }
            
            
            inv[a][b] = 0.5;
        }
        double ans = 0; 
        for(int i = 0; i < n; i ++) {
            for(int j = i+1; j < n; j ++) {
                ans += inv[i][j];
            }
        }
        
        out.println(df.format(ans));    
    }

    int upper_bound(int[] A, int l, int r, int val) {// upper_bound(A+l,A+r,val)-A;
        int pos = r;
        r--;
        while (l <= r) {
            int mid = (l + r) >> 1;
            if (A[mid] <= val) {
                l = mid + 1;
            } else {
                pos = mid;
                r = mid - 1;
            }
        }
        return pos;
    }
    int lower_bound(int[] A, int l, int r, int val) {// upper_bound(A+l,A+r,val)-A;
        int pos = r;
        r--;

                pos = mid;
                r = mid - 1;
            }
        }
