(n, m, k) = map(int, input().split())
sum = 0
a = list(map(int, input().split()))
for p in range(n):
	x = list(map(int, input().split()))
	for i in range(m):
		for j in range(k):
			if x[i] == a[j]:
				sum += j + 1
				for u in range(j, 0, -1):
					a[u] = a[u - 1]
				a[0] = x[i]
				break
print(sum)
