import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Solver {

    StringTokenizer st;
    BufferedReader in;
    PrintWriter out;

    public static void main(String[] args) throws NumberFormatException, IOException {
        Solver solver = new Solver();
        solver.open();
        long time = System.currentTimeMillis();
        solver.solve();
        if (!"true".equals(System.getProperty("ONLINE_JUDGE"))) {
            System.out.println("Spent time: " + (System.currentTimeMillis() - time));
        }
        solver.close();
    }

    public void open() throws IOException {
        in = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(System.out);
    }

    public String nextToken() throws IOException {
        while (st == null || !st.hasMoreTokens()) {
            st = new StringTokenizer(in.readLine());
        }
        return st.nextToken();
    }

    public int nextInt() throws NumberFormatException, IOException {
        return Integer.parseInt(nextToken());
    }

    public long nextLong() throws NumberFormatException, IOException {
        return Long.parseLong(nextToken());
    }

    public double nextDouble() throws NumberFormatException, IOException {
        return Double.parseDouble(nextToken());
    }

    enum VarType{
        INT,STRING,DOUBLE,T;
        public static VarType getValue(String s){
            s = s.trim();
            if (s.equals("int")){
                return INT;
            }else if (s.equals("T")){
                return T;
            }else if (s.equals("double")){
                return DOUBLE;
            }else{
                return STRING;
            }
        }
    }
    
    public class Procedure{
        String name;
        List<VarType> param = new ArrayList<VarType>(5);
        
        public Procedure(String signature){
            StringTokenizer strTok = new StringTokenizer(signature.trim().substring(4).trim());
            this.name = strTok.nextToken("(").trim();
            String nextParam;
            boolean end=false;
            do{
                nextParam = strTok.nextToken(",").trim();
                if(nextParam.startsWith("(")) nextParam = nextParam.substring(1).trim();
                if(nextParam.endsWith(")")) {nextParam = nextParam.substring(0,nextParam.length()-1).trim();end=true;}
                param.add(VarType.getValue(nextParam));
            }while(!end);
        }
        
        public Procedure(String signature, Map<String,VarType> vars){
            StringTokenizer strTok = new StringTokenizer(signature.trim());
            this.name = strTok.nextToken("(").trim();
            String nextParam;
            boolean end=false;
            do{
                nextParam = strTok.nextToken(",").trim();
                if(nextParam.startsWith("(")) nextParam = nextParam.substring(1).trim();
                if(nextParam.endsWith(")")) {nextParam = nextParam.substring(0,nextParam.length()-1).trim();end=true;}
                param.add(vars.get(nextParam));
            }while(!end);
        }
        
        public boolean equals(Procedure proc) {
            if (!name.equals(proc.name) || param.size()!=proc.param.size()) return false;
            
            for(int i=0;i<param.size();i++){
                VarType type1 = param.get(i);
                VarType type2 = proc.param.get(i);
                if (type1!=type2 && type1!=VarType.T && type2!=VarType.T) return false;
            }
            
            return true;
        }
        
        @Override
        public String toString() {
            return name+param.toString();
        }
    }
    
    public void solve() throws NumberFormatException, IOException {
        int n = nextInt();
        
        Procedure[] declaration = new Procedure[n];
        
        for(int i=0;i<n;i++){
            declaration[i] = new Procedure(in.readLine());
        }
        
        Map<String,VarType> vars = new HashMap<String, Solver.VarType>();
        int m = nextInt();
        
        for(int i=0;i<m;i++){
            String vtype = nextToken();
            String vname = nextToken();
            
            vars.put(vname, VarType.getValue(vtype));
        }
        
        
        int k=nextInt();
        for(int i=0;i<k;i++){
            int result = 0;
            
            Procedure next = new Procedure(in.readLine(),vars);
            
            for(int j=0;j<n;j++){
                if (next.equals(declaration[j])) result++;
            }
            
            out.println(result);            
        }
        
        
    }

    public void close() {
        out.flush();
        out.close();
    }

}
