#include<iostream>
#include<cstdio>
#include<cstring>
using namespace std;
const int mod=998244353;
#define N 20002000
inline int read(){
	int x=0,f=1;
	char c=getchar();
	while(c<'0'||c>'9'){
		if(c=='-')f=-1;
		c=getchar();
	}
	while(c>='0'&&c<='9'){
		x=(x<<1)+(x<<3)+c-'0';
		c=getchar();
	}
	return x*f;
}
int ans,n,A,B,fac[N],inv[N];
int qpow(int a,int b){
	int ans=1;
	while(b){
		if(b&1)ans=1LL*ans*a%mod;
		a=1LL*a*a%mod;
		b>>=1;
	}
	return ans;
}
void init(int n){
	fac[0]=1;
	for(int i=1;i<=n;++i){
		fac[i]=1LL*fac[i-1]*i%mod;
	}
	inv[n]=qpow(fac[n],mod-2);
	for(int i=n-1;i>=0;--i){
		inv[i]=1LL*inv[i+1]*(i+1)%mod;
	}
}
inline int C(int n,int m){
	return 1LL*fac[n]*inv[m]%mod*inv[n-m]%mod;
}
inline int calc(int x,int y){
	return (C(x+y,x)-C(x+y,x+1)+mod)%mod;
}
int main(){
	n=read(),A=read(),B=read();
	if(!A&&!B){
		return !printf("1\n");
	}
	init(n<<1);
	for(int i=0;i<=A&&i<=n-B-1;++i){
		ans=(ans+1LL*calc(B-1,i)*C(n-B-i-1+A-i,A-i))%mod;
	}
	if(n==A+B)ans=(ans+calc(B-1,A))%mod;
	printf("%d\n",ans);
	return 0;
}
