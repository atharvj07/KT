n = int(input())
s = list(map(int, input().split()))
m1 = 0
m2 = 0
i = -1
j = n
ns = []
if n == 1:
	print('0')
else:
	while i < j:
		if m1 == m2:
			ns += [m1]
			i += 1
			j -= 1
			m1 += s[i]
			m2 += s[j]
		elif m1 > m2:
			j -= 1
			m2 += s[j]
		elif m2 > m1:
			i += 1
			m1 += s[i]
	print(max(ns))
