import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;
import java.util.Stack;

public class ProblemD {
	
	static final String FIRST = "BitLGM";
	static final String SECOND = "BitAryo";
	
	static int[][] memo2;
	
	static int game2(int a, int b) {
		if (memo2[a][b] != 0) {
			return memo2[a][b];
		}
		boolean canbeat = false;
		for (int i = 1 ; i <= a ; i++) {
			if (game2(a-i, b) == -1) {
				canbeat = true;
			}
		}
		for (int i = 1 ; i <= b ; i++) {
			if (game2(a, b-i) == -1) {
				canbeat = true;
			}
		}
		for (int i = 1 ; i <= Math.min(a, b) ; i++) {
			if (game2(a-i, b-i) == -1) {
				canbeat = true;
			}
		}
		memo2[a][b] = canbeat ? 1 : -1;
		return memo2[a][b];
	}
	
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		int a = in.nextInt();
		if (n == 1) {
			out.println(a >= 1 ? FIRST : SECOND);
		} else if (n == 2) {
			int b = in.nextInt();
			memo2 = new int[300][300];
			memo2[0][0] = -1;
			out.println(game2(a, b) == 1 ? FIRST : SECOND);
		} else {
			int b = in.nextInt();
			int c = in.nextInt();
			int min = Math.min(a, Math.min(b, c));
			int max = Math.max(a, Math.max(b, c));
			int med = a + b + c - min - max;
			out.println((min^med) != max ? FIRST : SECOND);
		}
		
		out.flush();
	}
	

	public static void debug(Object... o) {
		System.err.println(Arrays.deepToString(o));
	}
}
