#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-8;
const double pi = acos(-1.0);
int sgn(double x) { return x < -eps ? -1 : x > eps; }
int main() {
  int w, h, a;
  while (~scanf("%d %d %d", &w, &h, &a)) {
    if (w < h) swap(w, h);
    if (a == 180) a -= 180;
    if (a == 90) {
      printf("%.10f\n", 1.0 * h * h);
      continue;
    }
    if (a > 90) a = 180 - a;
    if (a == 0) {
      printf("%.10f\n", 1.0 * w * h);
      continue;
    }
    double ans;
    if (sgn(tan(a * pi / 180 / 2) - (1.0 * h / w)) <= 0) {
      ans = 1.0 * w * h;
      double x1 = (w - h * tan(a * pi / 180 / 2)) / 2.0;
      double y1 = tan(a * pi / 180) * x1;
      ans -= x1 * y1;
      x1 = w - x1 * (1 + 1.0 / cos(a * pi / 180));
      y1 = x1 / tan(a * pi / 180);
      ans -= x1 * y1;
    } else {
      ans = h / sin(a * pi / 180) * h;
    }
    printf("%.10f\n", ans);
  }
}
