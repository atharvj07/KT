#include <bits/stdc++.h>
using namespace std;
struct edge {
  long long v, next;
} e[100000 + 5 << 1];
long long n, u, v, tot, a[100000 + 5], h[100000 + 5], dp[100000 + 5][2];
long long read() {
  char c;
  long long x = 0, f = 1;
  c = getchar();
  while (c > '9' || c < '0') {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') x = x * 10 + c - '0', c = getchar();
  return x * f;
}
void add(long long u, long long v) {
  e[++tot].v = v, e[tot].next = h[u], h[u] = tot;
}
void dfs(long long u, long long fa) {
  for (long long i = h[u]; i; i = e[i].next) {
    long long v = e[i].v;
    if (v == fa) continue;
    dfs(v, u);
    dp[u][0] = max(dp[u][0], dp[v][0]);
    dp[u][1] = max(dp[u][1], dp[v][1]);
  }
  if (dp[u][0] - dp[u][1] + a[u] >= 0)
    dp[u][1] += a[u] + dp[u][0] - dp[u][1];
  else
    dp[u][0] += -(a[u] + dp[u][0] - dp[u][1]);
}
signed main() {
  n = read();
  for (long long i = 1; i <= n - 1; ++i)
    u = read(), v = read(), add(u, v), add(v, u);
  for (long long i = 1; i <= n; ++i) a[i] = read();
  dfs(1, 0);
  printf("%lld", dp[1][0] + dp[1][1]);
  return 0;
}
