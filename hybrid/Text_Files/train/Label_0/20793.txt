#include <bits/stdc++.h>
int main() {
  int n, bi[1000], ai[1000], i, j, x, y, z, m;
  int test = 0;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", &bi[i]);
    ai[i] = bi[i];
  }
  m = bi[1];
  for (i = 1; i <= n; i++) {
    for (j = i + 1; j <= n; j++) {
      if (ai[i] >= ai[j]) {
        int t = ai[i];
        ai[i] = ai[j];
        ai[j] = t;
      }
    }
  }
  for (i = 1; i <= n; i++) {
    if (ai[1] == bi[i]) {
      test++;
    }
    if (ai[n] == bi[i]) {
      x = i - 1;
      for (i = i; i <= n; i++) {
        if (ai[1] == bi[i]) {
          test = 0;
        }
      }
    }
  }
  for (i = 1; i <= n; i++) {
    if (ai[1] == bi[i]) {
      y = i;
    }
  }
  if (test > 0) {
    printf("%d", x + n - y - 1);
  } else
    printf("%d", x + n - y);
}
