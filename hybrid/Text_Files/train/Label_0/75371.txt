from sys import stdin, stdout
input = stdin.readline
from itertools import tee
(n, m) = map(int, input().split())
(i1, i2) = tee((c == '.' for c in 'a' + input() + 'a'))
prev = False
res = 0
s = bytearray(i1)
for c in i2:
	res += c and prev
	prev = c

def calc(s, res):
	for i in range(m):
		(x, c) = input().split()
		x = int(x)
		nc = c == '.'
		if s[x] != nc:
			diff = s[x - 1] + s[x + 1]
			res += (-diff, diff)[nc]
			s[x] = nc
		yield str(res)
stdout.write('\n'.join(calc(s, res)))
