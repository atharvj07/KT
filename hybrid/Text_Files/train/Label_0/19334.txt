#include <bits/stdc++.h>
using namespace std;
const int oo = 0x3f3f3f3f;
const double eps = 1e-9;
struct P {
  double x[3];
  P() {}
  P(double a, double b, double c) {
    x[0] = a;
    x[1] = b;
    x[2] = c;
  }
};
P operator+(const P& a, const P& b) {
  P ret;
  for (int i = (0); i < (3); i++) ret.x[i] = a.x[i] + b.x[i];
  return ret;
}
P operator-(const P& a, const P& b) {
  P ret;
  for (int i = (0); i < (3); i++) ret.x[i] = a.x[i] - b.x[i];
  return ret;
}
double operator*(const P& a, const P& b) {
  double ret = 0.0;
  for (int i = (0); i < (3); i++) ret += a.x[i] * b.x[i];
  return ret;
}
double dist(const P& a, const P& b) {
  P dif = a - b;
  return sqrt(dif * dif);
}
P operator*(const P& a, double b) {
  P ret;
  for (int i = (0); i < (3); i++) ret.x[i] = a.x[i] * b;
  return ret;
}
double ti;
void printP(P a) {
  for (int i = (0); i < (3); i++) cout << a.x[i] << " ";
  cout << endl;
}
void calc(P p, P v, double r) {
  double lo = 0.0, hi = min(ti, 1e12);
  int ST = 200;
  while (ST--) {
    double m1 = lo + (hi - lo) * 0.4;
    double m2 = lo + (hi - lo) * 0.6;
    double f1 = dist(p, v * m1);
    double f2 = dist(p, v * m2);
    if (f1 < f2)
      hi = m2;
    else
      lo = m1;
  }
  double mini = lo + (hi - lo) * 0.5;
  if (dist(p, v * mini) - r >= eps) return;
  lo = 0.0, hi = mini;
  ST = 200;
  while (ST--) {
    double mid = lo + (hi - lo) * 0.5;
    if (dist(p, v * mid) - r < 0)
      hi = mid;
    else
      lo = mid;
  }
  ti = hi;
}
P readP() {
  P ret;
  for (int i = (0); i < (3); i++) scanf("%lf", &(ret.x[i]));
  return ret;
}
int main() {
  P d, v;
  double R;
  ti = 1e100;
  int N;
  d = readP();
  v = readP();
  scanf("%lf", &R);
  scanf("%d", &N);
  while (N--) {
    P c = readP();
    c = c - d;
    double r;
    int M;
    scanf("%lf%d", &r, &M);
    calc(c, v, R + r);
    while (M--) {
      P cc = readP();
      calc(c + cc, v, R);
    }
  }
  if (ti > 1e99) {
    printf("-1\n");
  } else {
    printf("%.9lf\n", ti);
  }
  return 0;
}
