def calc(numbers, newans, i):
	count = 0
	ansarr = ''.join([str(x) for x in newans])
	for num in numbers:
		if num[:i + 1] == ansarr:
			count += 1
	return count
t = int(input())
for _ in range(t):
	(n, m) = [int(c) for c in input().split()]
	numbers = []
	for i in range(n):
		numbers.append(input())
	ans = 0
	ansarr = []
	k = ((1 << m) - n - 1) // 2 + 1
	for i in range(m):
		nums = 1 << m - 1 - i
		newans = ans | 0
		remainums = nums - calc(numbers, ansarr + [0], i)
		if remainums < k:
			k -= remainums
			ansarr.append(1)
			newans = ans | 1 << m - i - 1
		else:
			ansarr.append(0)
			newans = ans | 0
		ans = newans
	print(''.join([str(c) for c in ansarr]))
