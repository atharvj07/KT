#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pii = pair<int, int>;

int N;
int x[100010], y[100010], w[100010];
map<pii, int> mp;

int calc(pii p) {
  int res1 = mp[p], res2 = mp[p];
  res1 += mp[pii(p.first, p.second + 1)];
  res1 += mp[pii(p.first + 1, p.second)];
  res1 += mp[pii(p.first + 1, p.second + 1)];
  res2 += mp[pii(p.first, p.second - 1)];
  res2 += mp[pii(p.first + 1, p.second)];
  res2 += mp[pii(p.first + 1, p.second - 1)];
  return max(res1, res2);
}

int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(false);
  cout << fixed << setprecision(10);
  
  cin >> N;
  for (int i = 0; i < N; i++) {
    cin >> x[i] >> y[i] >> w[i];
    mp[pii(x[i], y[i])] = w[i];
  }

  int ans = 0;
  for (int i = 0; i < N; i++) {
    ans = max(ans, calc(pii(x[i], y[i])));
  }
  cout << ans << " / " << 1 << endl;

  return 0;
}
