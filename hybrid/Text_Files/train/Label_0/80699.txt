t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	current = 0
	reps = n // 3
	if s.count('T') != 2 * s.count('M') or n % 3 != 0:
		print('NO')
		continue
	for c in s:
		if c == 'T':
			current += 1
		else:
			current -= 1
		if current < 0 or current > reps:
			print('NO')
			break
	else:
		print('YES')
