for _ in range(int(input())):
	n = int(input())
	a = input()
	b = input()
	x = y = x1 = y1 = 0
	for i in range(n):
		if a[i] == b[i]:
			x += 1
			if a[i] == '1':
				x1 += 1
		else:
			y += 1
			if a[i] == '1':
				y1 += 1
	ans = float('inf')
	if x % 2 and (x + 1) // 2 == x1:
		ans = x
	if y % 2 == 0 and y // 2 == y1:
		ans = min(ans, y)
	if ans == float('inf'):
		print(-1)
	else:
		print(ans)
