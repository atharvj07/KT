#include <bits/stdc++.h>

using namespace std;

vector<int> edge[3050];

int A[3050];
bool flg[3050];

void chk(int pos)
{
	flg[pos] = 0;
	for(auto v : edge[pos])
		if(A[v] < A[pos])
		{
			chk(v);
			if(!flg[v])
			{
				flg[pos] = 1;
				return ;
			}
		}
}

int main()
{
	int n; scanf("%d",&n);
	for(int i = 1;i <= n; ++ i) scanf("%d",&A[i]);
	for(int i = 1;i < n; ++ i)
	{
		int u,v; scanf("%d%d",&u,&v);
		edge[u].push_back(v);
		edge[v].push_back(u);
	}
	for(int i = 1;i <= n; ++ i)
	{
		chk(i);
		if(flg[i]) printf("%d ",i);
	}
	puts("");
	return 0;
}
