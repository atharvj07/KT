t = int(input())
ans = []
for i in range(t):
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	c = [0] * n
	for i in range(n):
		c[a[i] - 1] = i
	time = c[b[0] - 1] * 2 + 1
	s = c[b[0] - 1]
	for i in range(1, m):
		if s > c[b[i] - 1]:
			time += 1
		else:
			s = c[b[i] - 1]
			time = time + (c[b[i] - 1] - i) * 2 + 1
	ans.append(time)
for i in range(t):
	print(ans[i])
