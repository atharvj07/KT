def bl(arr, ele):
	n = len(arr)
	st = 0
	end = n - 1
	ans = -1
	while st <= end:
		mid = (st + end) // 2
		if arr[mid] >= ele:
			end = mid - 1
			ans = mid
		else:
			st = mid + 1
	return ans
for _ in range(it()):
	n = it()
	l = sorted(mp())
	c = C(l)
	ans = [0] * (n + 1)
	flag = 0
	for i in range(n):
		if l[i] > i:
			ans[i + 1] = -1
			flag = 1
		elif flag == 1:
			ans[i + 1] = -1
	count = 0
	ind = 0
	q = []
	heapify(q)
	for i in range(n + 1):
		if ans[i] == -1:
			True
		else:
			ans[i] = count + c[i]
		while ind < n and l[ind] <= i:
			heappush(q, -l[ind])
			ind += 1
		if len(q) == 0:
			continue
		else:
			x = -heappop(q)
			count += i - x
	print(*ans)
