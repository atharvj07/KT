#include <bits/stdc++.h>
using namespace std;
void solv() {
  long long n, x, k;
  cin >> n >> x >> k;
  if (n == 1) {
    cout << x << endl;
    return;
  }
  x -= n;
  long long ans = 1;
  if (x == 0) {
    cout << ans << endl;
    return;
  }
  long long a = min(n - k + 1, k);
  long long A = pow(a, 2);
  if (A >= x) {
    ans += sqrt(x);
    cout << ans << endl;
    return;
  }
  long long m = min(a * 2, n);
  ans += sqrt(A);
  x -= A;
  if (x <= n) {
    cout << ans + 1 << endl;
    return;
  }
  while (x >= m && m < n) {
    x -= m;
    m++;
    ans++;
  }
  if (m == n) ans += x / n;
  cout << ans << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int T = 1;
  while (T--) solv();
  return 0;
}
