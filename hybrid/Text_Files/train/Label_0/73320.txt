import sys
readline = sys.stdin.buffer.readline
ni = lambda : int(readline().rstrip())
nm = lambda : map(int, readline().split())

def solve():
	q = ni()
	n = 5 * 10 ** 5
	a = [0] * (n + 1)
	m = 710
	rem = [[0] * m for _ in range(m)]
	for _ in range(q):
		(t, x, y) = nm()
		if t == 1:
			a[x] += y
			for p in range(1, m):
				rem[p][x % p] += y
		elif x < m:
			print(rem[x][y])
		else:
			print(sum(a[y::x]))
solve()
