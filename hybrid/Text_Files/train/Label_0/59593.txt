    static FastReader in;

    static final int oo = (int) 2e9;
    static final long OO = (long) 4e18;
    static final int MOD = (int) 1e9 + 7;

    static class Pair {
        int x, y;

        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static void solve() {
        int n = in.nextInt();

        Pair[] pairs = new Pair[n];
        int left = oo, right = -oo, down = oo, top = -oo;
        for (int i = 0; i < n; i++) {
            String s = in.next();
            int cntB = 0, cntN = 0;
            for (int j = 0; j < s.length(); j++) {
                if (s.charAt(j) == 'B')
                    cntB++;
                else
                    cntN++;
            }
            pairs[i] = new Pair(cntB, cntN);
            left = Math.min(left, pairs[i].x);
            right = Math.max(right, pairs[i].x);
            down = Math.min(down, pairs[i].y);
            top = Math.max(top, pairs[i].y);
        }

        int low = 0, high = (int) 1e6, x0 = 0, y0 = 0;
        while (low < high) {
            int r = (low + high) / 2;
            if (right - left > r * 2 || top - down > r * 2) {
                low = r + 1;
                continue;
            }
            {
                int x = left + r, minY = -oo, maxY = oo;
                for (int i = 0; i < n; i++) {
                    if (pairs[i].x <= x) {
                        minY = Math.max(minY, pairs[i].y - (pairs[i].x - (x - r)));
                        maxY = Math.min(maxY, pairs[i].y + r);
                    } else {
                        minY = Math.max(minY, pairs[i].y - r);
                        maxY = Math.min(maxY, pairs[i].y + ((x + r) - pairs[i].x));
                    }
                }
                if (minY <= maxY) {
                    high = r;
                    x0 = x;
                    y0 = minY;
                    continue;
                }
            }
            {
                int x = right - r, minY = -oo, maxY = oo;
                for (int i = 0; i < n; i++) {
                    if (pairs[i].x <= x) {
                        minY = Math.max(minY, pairs[i].y - (pairs[i].x - (x - r)));
                        maxY = Math.min(maxY, pairs[i].y + r);
                    } else {
                        minY = Math.max(minY, pairs[i].y - r);
                        maxY = Math.min(maxY, pairs[i].y + ((x + r) - pairs[i].x));
                    }
                }
                if (minY <= maxY) {
                    high = r;
                    x0 = x;
                    y0 = minY;
                    continue;
                }
            }
            {
                int y = down + r, minX = -oo, maxX = oo;
                for (int i = 0; i < n; i++) {
                    if (pairs[i].y <= y) {
                        minX = Math.max(minX, pairs[i].x - (pairs[i].y - (y - r)));
                        maxX = Math.min(maxX, pairs[i].x + r);
                    } else {
                        minX = Math.max(minX, pairs[i].x - r);
                        maxX = Math.min(maxX, pairs[i].x + ((y + r) - pairs[i].y));
                    }
                }
                if (minX <= maxX) {
                    high = r;
                    x0 = minX;
                    y0 = y;
                    continue;
                }
            }
            {
                int y = top - r, minX = -oo, maxX = oo;
                for (int i = 0; i < n; i++) {
                    if (pairs[i].y <= y) {
                        minX = Math.max(minX, pairs[i].x - (pairs[i].y - (y - r)));
                        maxX = Math.min(maxX, pairs[i].x + r);
                    } else {
                        minX = Math.max(minX, pairs[i].x - r);
                        maxX = Math.min(maxX, pairs[i].x + ((y + r) - pairs[i].y));
                    }
                }
                if (minX <= maxX) {
                    high = r;
                    x0 = minX;
                    y0 = y;
                    continue;
                }
            }
            low = r + 1;
        }

        out.println(low);
        for (int i = 0; i < x0; i++)
            out.print('B');
        for (int i = 0; i < y0; i++)
            out.print('N');
        out.println();


    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        String nextLine() {
            String s = "";
            try {
                s = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return s;
        }
    }
