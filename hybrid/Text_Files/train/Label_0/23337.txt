long long power(long long base, int exponent) {
  long long ans = 1;
  while (exponent > 0) {
    if (exponent & 1) ans = (ans * base);
    exponent >>= 1;
    base = base * base;
  }
  return ans;
}
long long power(long long base, int exponent, int p) {
  long long ans = 1;
  base = base % p;
  while (exponent > 0) {
    if (exponent & 1) ans = (ans * base) % p;
    exponent >>= 1;
    base = (base * base) % p;
  }
  return ans;
}
int modInverse(int n, int p) { return power(n, p - 2, p); }
void factorialCalc(vector<int> &factorial, int n) {
  if (factorial.size() != n + 1) {
    factorial.resize(n + 1);
  }
  factorial[0] = 1;
  factorial[1] = 1;
  for (int i = 2; i <= n; i++) {
    factorial[i] = factorial[i - 1] * i;
  }
}
void factorialCalc(vector<int> &factorial, int n, int p) {
  if (factorial.size() != n + 1) {
    factorial.resize(n + 1);
  }
  factorial[0] = 1;
  factorial[1] = 1;
  for (int i = 2; i <= n; i++) {
    factorial[i] = (factorial[i - 1] * 1ll * i) % (p);
  }
}
unsigned long long nCr(unsigned long long n, int r, int p, vector<int> &fac) {
  if (r == 0) return 1;
  return (1ll * fac[n] * modInverse(fac[r], p) % p * modInverse(fac[n - r], p) %
          p) %
         p;
}
template <typename T>
void keySort(vector<T> &arr, vector<int> &key) {
  int n = arr.size();
  if (key.size() != n) {
    throw runtime_error("Size of data array not equals to size of key array");
  }
  vector<T> ans(n);
  int ar[n];
  for (int i = 0; i < n; i++) {
    ar[key[i]] = i;
  }
  for (int i = 0; i < n; i++) {
    ans[i] = arr[ar[i]];
  }
  arr = ans;
}

void solve() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int tmp = 0;
  int ans0 = 0;
  int ans1 = 0;
  for (auto i : s) {
    if (i == '1') {
      tmp++;
    } else {
      ans1 += max(0, tmp - 1);
      tmp = 0;
    }
  }
  ans1 += max(0, tmp - 1);
  tmp = 0;
  for (auto i : s) {
    if (i == '0') {
      tmp++;
    } else {
      ans0 += max(0, tmp - 1);
      tmp = 0;
    }
  }
  ans0 += max(0, tmp - 1);
  std::cout << max(ans0, ans1) << endl;
  return;
}
