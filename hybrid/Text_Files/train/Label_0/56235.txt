import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class CF284D {
    static int[] a;
    static int[] tree;

    static void build(int i, int l, int r) {
        if (l == r) {
            for (int k = 0; k < 60; k++) {
                if (k % a[l] == 0)
                    tree[i * 60 + k] = 2;
                else
                    tree[i * 60 + k] = 1;
            }
            return;
        }
        build(i * 2 + 1, l, (l + r) / 2);
        build(i * 2 + 2, (l + r) / 2 + 1, r);
        for (int k = 0; k < 60; k++) {
            int left = tree[(i * 2 + 1) * 60 + k];
            int right = tree[(i * 2 + 2) * 60 + (left + k) % 60];
            tree[i * 60 + k] = left + right;
        }

    }

    static int t = 0;

    static void query(int i, int a, int b, int l, int r) {
        if (b < l || a > r)
            return;
        if (a >= l && b <= r) {
            t += tree[i * 60 + t % 60];
            return;
        }
        query(i * 2 + 1, a, (a + b) / 2, l, r);
        query(i * 2 + 2, (a + b) / 2 + 1, b, l, r);
    }

    static void update(int i, int l, int r, int target) {
        if (l == r) {
            for (int k = 0; k < 60; k++) {
                if (k % a[target] == 0)
                    tree[i * 60 + k] = 2;
                else
                    tree[i * 60 + k] = 1;
            }
            return;
        }
        if (target <= (l + r >> 1))
            update(i * 2 + 1, l, l + r >> 1, target);
        else
            update(i * 2 + 2, (l + r >> 1) + 1, r, target);
        for (int k = 0; k < 60; k++) {
            int left = tree[(i * 2 + 1) * 60 + k];
            int right = tree[(i * 2 + 2) * 60 + (left + k) % 60];
            tree[i * 60 + k] = left + right;
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer strtok;
        strtok = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(strtok.nextToken());
        a = new int[n];
        tree = new int[n * 4 * 60];
        strtok = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(strtok.nextToken());
        }
        build(0, 0, n - 1);
        strtok = new StringTokenizer(in.readLine());
        int m = Integer.parseInt(strtok.nextToken());
        for (int q = 0; q < m; q++) {
            strtok = new StringTokenizer(in.readLine());
            String type = strtok.nextToken();
            if (type.equals("A")) {
                int x = Integer.parseInt(strtok.nextToken()) - 1;
                int y = Integer.parseInt(strtok.nextToken()) - 1;
                if (x == y) {
                    System.out.println(0);
                } else {
                    t = 0;
                    query(0, 0, n - 1, x, y - 1);
                    System.out.println(t);
                }
            } else {
                int x = Integer.parseInt(strtok.nextToken()) - 1;
                int y = Integer.parseInt(strtok.nextToken());
                a[x] = y;
                update(0, 0, n - 1, x);
            }
        }
    }
}
