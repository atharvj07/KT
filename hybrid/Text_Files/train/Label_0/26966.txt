#include <bits/stdc++.h>
const int MAXN = 1e5 + 5;
int a[MAXN], b[MAXN];
int n, m;
bool in[MAXN];
inline void Solve() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) in[i] = false;
  for (int i = 1; i <= n; ++i) scanf("%d", a + i);
  for (int i = 1; i <= m; ++i) scanf("%d", b + i);
  int tp = 1;
  long long ans = 0;
  int sz = 0;
  for (int i = 1; i <= m; ++i) {
    if (in[b[i]]) {
      sz--;
      ans++;
      continue;
    }
    int cnt = sz;
    while (tp <= n && b[i] != a[tp]) {
      in[a[tp]] = true;
      ++sz;
      ++tp;
      ++cnt;
    }
    ++tp;
    ans += cnt * 2 + 1;
  }
  printf("%lld\n", ans);
}
int main() {
  int T;
  scanf("%d", &T);
  while (T--) Solve();
  return 0;
}
