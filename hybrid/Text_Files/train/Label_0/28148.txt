int n, m, k;
void next(vector<int> &c) {
  c[k - 1]++;
  bool check = true;
  int cnt = 1;
  while (check) {
    if (c[k - cnt] > k - cnt + 1) {
      if (k - cnt - 1 >= 0) {
        c[k - cnt - 1]++;
        c[k - cnt] = 1;
      } else {
        check = false;
      }
      cnt++;
    } else {
      check = false;
    }
  }
}
int main() {
  cin >> n >> m >> k;
  vector<vector<pair<int, int>>> G(n);
  int u, v, w;
  for (int i = 0; i < m; i++) {
    cin >> u >> v >> w;
    u--;
    v--;
    G[u].push_back(make_pair(w, v));
  }
  int dim = k * k * k * k;
  vector<int> a(dim, true);
  vector<set<pair<int, int>>> H(n);
  int s, cnt;
  for (int i = 0; i < n; i++) {
    sort((G[i]).begin(), (G[i]).end());
    s = G[i].size();
    cnt = 1;
    for (pair<int, int> x : G[i]) {
      if (H[x.second].count(make_pair(s, cnt))) {
        a[k * k * k * (s - 1) + k * k * (cnt - 1) + k * (s - 1) + (cnt - 1)] =
            false;
      }
      H[x.second].insert(make_pair(s, cnt));
      cnt++;
    }
  }
  int cnt2;
  for (int i = 0; i < n; i++) {
    cnt = 0;
    for (pair<int, int> x : H[i]) {
      cnt2 = 0;
      for (pair<int, int> y : H[i]) {
        if (cnt == cnt2) {
          cnt2++;
          continue;
        }
        a[k * k * k * (x.first - 1) + k * k * (x.second - 1) +
          k * (y.first - 1) + (y.second - 1)] = false;
        cnt2++;
      }
      cnt++;
    }
  }
  vector<int> c(k, 1);
  vector<long long> fact(10);
  fact[0] = 1;
  fact[1] = 1;
  for (int i = 2; i < 10; i++) {
    fact[i] = fact[i - 1] * i % MOD;
  }
  long long ans = fact[k];
  bool check;
  for (int i = 0; i < fact[k]; i++) {
    check = true;
    for (int j = 0; j < k; j++) {
      for (int l = 0; l < k; l++) {
        if (a[k * k * k * j + k * k * (c[j] - 1) + k * l + (c[l] - 1)] ==
            false) {
          ans--;
          check = false;
          break;
        }
      }
      if (check == false) {
        break;
      }
    }
    next(c);
  }
  printf("%lld\n", ans);
}
