import sys
read = sys.stdin.readline
n = int(input())
a = [int(_) for _ in read().split()]
bound = 1000 * 2 + 1
inf = 10 ** 10

def solve(a):
	init = a[:]
	i = 0
	while True:
		if a == list(range(1, 2 * n + 1)):
			return i
		if i % 2 == 0:
			for j in range(0, 2 * n, 2):
				(a[j], a[j + 1]) = (a[j + 1], a[j])
		else:
			a = a[n:] + a[:n]
		if a == init:
			return inf
		i += 1
	return inf
ans = min(solve(a[:]), 1 + solve(a[n:] + a[:n]))
if ans >= inf:
	ans = -1
print(ans)
