#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  long long q;
  cin >> q;
  while (q--) {
    long long n, m;
    cin >> n >> m;
    vector<long long> cnt(64);
    long long s = 0;
    for (long long i = 0; i < m; i++) {
      long long x;
      cin >> x;
      s += x;
      for (long long j = 0; j < 60; j++)
        if (x & (1LL << j)) {
          cnt[j]++;
          break;
        }
    }
    if (s < n) {
      cout << -1 << '\n';
      continue;
    }
    long long i = 0, res = 0;
    while (i < 60) {
      if (n & (1LL << i)) {
        if (cnt[i])
          cnt[i]--;
        else {
          while (i < 60 && !cnt[i]) {
            i++;
            res++;
          }
          cnt[i]--;
          continue;
        }
      }
      cnt[i + 1] += cnt[i] / 2;
      i++;
    }
    cout << res << '\n';
  }
  return 0;
}
