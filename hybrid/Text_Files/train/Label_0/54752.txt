import java.io.*;
import java.util.*;
public class Main {

	public static final int mod=1000000007;	
	static long[][] dp;
	static Map<Integer,Integer> mp;
	static Set<Integer> set;
	public static void main(String[] args) throws IOException {
		
		//File filein = new File("input.txt");
		//BufferedReader sc=new BufferedReader(new FileReader(filein));
		//PrintWriter out=new PrintWriter(new FileWriter("output.txt"));		

		//Scanner sc=new Scanner(System.in);
		BufferedReader sc=new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out=new PrintWriter(System.out);
		int tt=Integer.parseInt(sc.readLine());
		int t=tt;
		mp=new HashMap<>();
		set=new HashSet<>();
		while(t-->0){
			out.flush();
			set.clear();
			String[] ip=sc.readLine().split(" ");
			int n=Integer.parseInt(ip[0]);
			int k=Integer.parseInt(ip[1]);
			int[][] arr=new int[n][2];
			for(int i=0;i<n;i++){
				ip=sc.readLine().split(" ");
				int[] temp=new int[2];
				temp[0]=Integer.parseInt(ip[0]);
				temp[1]=Integer.parseInt(ip[1]);
				arr[i]=temp;
			}
			int[][] d=new int[n][n];
			boolean pos=false;
			for(int i=0;i<n;i++){
				for(int j=0;j<n;j++){
					d[i][j]=Math.abs(arr[i][0]-arr[j][0])+Math.abs(arr[i][1]-arr[j][1]);					
				}
			}

			for(int i=0;i<n;i++){
				pos=true;
				for(int j=0;j<n;j++){
					if(i==j) continue;
					if(d[i][j]>k){ pos=false; break;}
				}
				if(pos) break;
			}

			if(pos) out.println(1);
			else out.println(-1);

		}
		out.flush();
		out.close();
		sc.close();
	}

	static boolean isSorted(int arr[],int i,int j){
		boolean sorted=true;
		while(i<j){
			if(arr[i]>arr[i+1]){
				sorted=false;
				break;
			}
			i++;
		}
		return sorted;
	}


}
