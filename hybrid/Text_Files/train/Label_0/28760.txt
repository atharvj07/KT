#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0, x; i < n; ++i) {
    cin >> x;
    a[i] -= x;
  }
  sort(a.begin(), a.end());
  long long ans = 0;
  for (int i = 0, j = n - 1; i < n; ++i) {
    for (j = min(j, n - 1); j > i && a[i] + a[j] > 0; --j)
      ;
    ++j;
    if (j == n) {
      continue;
    }
    if (a[i] + a[j] > 0) {
      ans += n - j;
    }
  }
  cout << ans << '\n';
  return 0;
}
