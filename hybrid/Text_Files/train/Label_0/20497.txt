#include <bits/stdc++.h>
using namespace std;
bool is(string s) {
  string k = s;
  reverse(k.begin(), k.end());
  return k == s;
}
string rev(string s) {
  reverse(s.begin(), s.end());
  return s;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n, mm;
  string s;
  cin >> n >> mm;
  vector<string> v;
  unordered_map<string, int> m, used;
  for (int i = 0; i < n; i++) {
    cin >> s;
    v.push_back(s);
    m[s]++;
  }
  string ans1, ans2, tmp, spe;
  int f = 0;
  for (int i = 0; i < n; i++) {
    tmp = rev(v[i]);
    if (m[tmp] && !used[tmp]) {
      if (v[i] == tmp) {
        int k = m[tmp];
        while (k--) spe += tmp;
        auto it = m.find(tmp);
        m.erase(it);
        tmp = ans1 + spe + ans2;
        if (is(tmp)) ans1 += spe;
      } else {
        ans1 = v[i] + ans1;
        ans2 = ans2 + tmp;
        used[tmp]++;
        used[v[i]]++;
        m[tmp]--;
      }
    }
  }
  tmp = ans1 + ans2;
  cout << tmp.length() << "\n" << tmp << "\n";
}
