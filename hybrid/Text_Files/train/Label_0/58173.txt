	static node idx[]=new node[100001];
	static node num[]=new node [100001];
	static void Init(int n){
	     for(int i=1;i<=n;i++){
	          idx[i]=num[i];
	          for(int j=1;j<Lowbit(i);j<<=1){
	        	  if(idx[i].v>idx[i-j].v){
	        		  idx[i]=idx[i-j];
	        	  }
	             //  idx[i]=Math.min(idx[i],idx[i-j]);
	          }
	     }
	}
	private static int Lowbit(int x) {
		// TODO Auto-generated method stub
		return   x & (-x);  
	}
	static node Query(int l,int r){
	     node  ans=num[r];
	     while(true){
	    	 if(num[r].v<ans.v)ans=num[r];
	          if(r==l) break;
	          for(r-=1;r-l>=Lowbit(r);r-=Lowbit(r)){
	               if(idx[r].v<ans.v)ans=idx[r];
	          }
	     }
	     return ans;
	}
	static int m;
	static int l;
	static Set<Integer>q=new HashSet<Integer>(); 
	static char s[];
	public static void main(String[] args) {
		//Scanner in=new Scanner(new BufferedInputStream(System.in));
		 m=in.nextInt();

		 for(int i=1;i<=l;i++){
			 num[i]=new node(i,s[i-1]);
		 }
		 Init(l);
		 int maz='a';
for(int bg=1;bg<=l-m+1;bg++){

	node k=Query(Math.max(1, bg),Math.min(l,bg+m-1));
	//System.out.println(bg+" "+k.id+" "+(char)k.v);
	q.add(k.id);
	maz=Math.max(k.v, maz);
	bg=k.id;
	
}
for(int i=1;i<=l;i++){
	if(q.contains(i))continue;
	if(s[i-1]<maz)q.add(i);
}
node[] ans=new node[q.size()];
int cc=0;
for(int p:q)ans[cc++]=num[p];
Arrays.sort(ans);
for(int i=0;i<cc;i++)out.print((char)ans[i].v);
out.println();
out.flush();
	}
	
		
		

	

}
class node implements Comparable<node>{
	int id;
	int v;
	node (int i,int k){
		id=i;
		v=k;
	}
	@Override
	public int compareTo(node o) {
		// TODO Auto-generated method stub
		return this.v-o.v;
	}
}

class InputReader {  //放到你的主程序的下方咯
    BufferedReader reader;  
    StringTokenizer tokenizer;  
  
    InputReader(InputStream stream) {  
        reader = new BufferedReader(new InputStreamReader(stream));  
        tokenizer = null;  
    }  
  
    String next() {  
        if (!hasNext())  
            throw new RuntimeException();  
        return tokenizer.nextToken();  
    }  
  
    boolean hasNext() {  
        while (tokenizer == null || !tokenizer.hasMoreTokens())  
            try {  
                tokenizer = new StringTokenizer(reader.readLine());  
            } catch (Exception e) {  
                return false;  
            }  
        return true;  
    }  
  
    long nextLong() {  
        return Long.parseLong(next());  
    }  
  
    int nextInt() {  
        return Integer.parseInt(next());  
    }  
	public String nextLine() {
		tokenizer = null;
		try {
			return reader.readLine();
		} catch (Exception e) {
			throw new RuntimeException();
		}
	}
}
