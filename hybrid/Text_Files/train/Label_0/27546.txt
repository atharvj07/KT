#include <bits/stdc++.h>
using namespace std;
template <typename TYPE>
inline TYPE sqr(const TYPE& a) {
  return (a) * (a);
}
const int INF = 1000 * 1000 * 1000;
const long double EPS = 1e-9;
const long double PI = 2 * acos(0.0);
const int N = 100100;
int n, m, x, y;
pair<int, int> a[N], b[N];
inline bool good(int u, int v) { return u - x <= v && v <= u + y; }
int main() {
  scanf("%d%d%d%d", &n, &m, &x, &y);
  for (int i = 0; i < int(n); ++i) {
    int val;
    scanf("%d", &val);
    a[i] = pair<int, int>(val, i);
  }
  for (int i = 0; i < int(m); ++i) {
    int val;
    scanf("%d", &val);
    b[i] = pair<int, int>(val, i);
  }
  sort(a, a + n);
  sort(b, b + m);
  int j = 0;
  vector<pair<int, int> > ans;
  for (int i = 0; i < int(n); ++i) {
    while (j < m && a[i].first - x > b[j].first) ++j;
    if (j < m && good(a[i].first, b[j].first)) {
      ans.push_back(pair<int, int>(a[i].second, b[j].second));
      ++j;
    }
    if (j >= m) break;
  }
  printf("%d\n", (int)ans.size());
  for (int i = 0; i < int(ans.size()); ++i)
    printf("%d %d\n", ans[i].first + 1, ans[i].second + 1);
  return 0;
}
