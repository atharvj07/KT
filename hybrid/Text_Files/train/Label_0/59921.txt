import java.util.*;
import java.io.*;
public class e {
    public static void main(String[] Args) throws Exception {
        FS sc =new FS(System.in);
        PrintWriter out =new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
        int n = sc.nextInt()-1;
        int[] mins = new int[n];
        int[] maxs = new int[n];
        for (int i = 0; i < n; i++) 
            mins[i] = sc.nextInt();
        EulerCircuit ec = new EulerCircuit(n * 2);
        for (int i = 0; i < n; i++) {
            maxs[i] = sc.nextInt();
            if (mins[i] > maxs[i]) {
                out.println(-1);
                out.close();
                return;
            }
            ec.addEdge(mins[i], maxs[i]);
        }
        int[] path = ec.getPath();
        if (path == null) {
            out.println(-1);
            out.close();
            return;
        }
        for (Integer x : path)
            out.print(ec.lookup.get(x) + " ");
        out.println();
        out.close();
    }
    
    public static class FS {
        StringTokenizer st;
        BufferedReader br;
        FS(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            st = new StringTokenizer("");
        }
        String next() throws Exception {
            if (st.hasMoreTokens())
                return st.nextToken();
            st = new StringTokenizer(br.readLine());
            return next();
        }
        int nextInt() throws Exception {
            return Integer.parseInt(next());
        }
    }
public static class EulerCircuit {
  TreeMap<Integer,Integer>[] g;
  int[] deg;
  int n,m=0;
  int x;
  TreeMap<Integer, Integer> map;
  ArrayList<Integer> lookup;

  EulerCircuit(int N) {
    g = new TreeMap[n=N];
    lookup = new ArrayList<Integer>();
    for(int i=0; i<N; i++) g[i] = new TreeMap<Integer,Integer>();
    deg = new int[n];
    map = new TreeMap<Integer, Integer>();
    x = 0;
  }
  void addEdge(int a, int b) {
    if (!map.containsKey(a)){
        lookup.add(a);
        map.put(a,x++);
    }
    a = map.get(a);
    if (!map.containsKey(b)){
        lookup.add(b);
        map.put(b,x++);
    }
    b = map.get(b);
    
      
    if (!g[a].containsKey(b)) {
        g[a].put(b,0);
        g[b].put(a,0);
    }
    g[a].put(b,g[a].get(b) + 1);
    g[b].put(a,g[b].get(a) + 1);
    deg[a]++;
    deg[b]++;
    m++;
  }
  int[] getPath() {
    int[] last = new int[3], ans = new int[m+1];
    int count = 0;
    for(int i=0; i<n; i++) {
      if((deg[i]&1)==1) 
          last[count++] = i;
      if(count == 3) return null;
    }
    int cur, left = m;
    cur = last[0];
    
    ArrayDeque<Integer> stack = new ArrayDeque<Integer>();
    while(stack.size() > 0 || deg[cur] > 0) {
        if(deg[cur] == 0) {
            ans[left--] = cur;
            cur = stack.pop();
        }
        else {
            int next = g[cur].firstKey();
            stack.push(cur);
            deg[cur]--;
            deg[next]--;
            int newVal = g[cur].get(next)-1;
            if (cur == next)
                newVal--;
            if (newVal == 0 ){
                g[cur].remove(next);
                g[next].remove(cur);
            } else {
                g[cur].put(next, newVal);
                g[next].put(cur, newVal);
            }
            cur = next;
        }
    }
    if(left > 0) return null;
    ans[left] = cur;
    return ans;
  }
}
}
