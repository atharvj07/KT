#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
long long a[MAXN];
long long pos[70];
int main() {
  int t;
  cin >> t;
  while (t--) {
    memset(pos, 0, sizeof(pos));
    long long n;
    long long m;
    cin >> n >> m;
    long long sum = 0;
    for (int y = 1; y <= m; y++) {
      long long dx;
      cin >> dx;
      for (long long y = 0; y <= 63; y++) {
        if ((dx >> y) & 1) pos[y]++;
      }
      sum += dx;
    }
    if (sum < n)
      cout << -1 << endl;
    else {
      long long ans = 0;
      for (long long y = 0; y <= 63; y++) {
        if ((n >> y) & 1) {
          if (pos[y]) {
            pos[y]--;
          } else {
            int j = y;
            int now;
            int i = 0;
            for (int u = j + 1; u <= 64; u++) {
              if (pos[u]) {
                i++;
                now = u;
                ;
                pos[u]--;
                break;
              }
            }
            while (1) {
              now--;
              pos[now]++;
              ans++;
              if (now == j) break;
            }
          }
        }
        pos[y + 1] += pos[y] / 2;
      }
      cout << ans << endl;
    }
  }
  return 0;
}
