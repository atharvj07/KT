from collections import deque
s = deque()
n = int(input())
s1 = deque()
l = str(input())
for x in range(n):
	if l[x] == 'D':
		s.append(x)
	else:
		s1.append(x)
for x in range(n):
	if len(s) != 0 and len(s1) != 0:
		if s[0] < s1[0]:
			s1.popleft()
			s.popleft()
			s.append(x + n)
		else:
			s.popleft()
			s1.popleft()
			s1.append(x + n)
if len(s) == 0:
	print('R')
else:
	print('D')
