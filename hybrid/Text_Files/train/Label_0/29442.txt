void solve()
{
    int n,m;
    cin>>n>>m;
    string s;
    cin>>s;
    vector<int> a(n,0);
    vector<int> dist(n,INF);
    int prev=-1;
    for(int i=0;i<n;i++){
        if(s[i]=='1'){
            prev=i;
            dist[i]=0;
        }
        if(prev>=0){
            dist[i]=min(dist[i],i-prev);
        }
    }
    prev=-1;
    for(int j=n-1;j>=0;j--){
        if(s[j]=='1'){
            prev=j;
        }
        if(prev>=0){
            if(dist[j]==prev-j){
                dist[j]=INF;
            }
            else{
                dist[j]=min(dist[j],prev-j);
            }            
        }
    }
    for(int i=0;i<n;i++){
        if(s[i]=='1'){
            dist[i]=0;
        }
        if(dist[i]<=m){
            a[i]=1;
        }
    }
    for(auto x : a){
        cout<<x;
    }
    cout<<endl;    
}
signed main() 
{ 
    FAST;
    int t;
    cin>>t;
    while(t--)
        solve();
    return 0; 
}

int power(int base, int exp) {
   int res=1;
   while(exp>0) {
      if(exp%2==1) res=(res*base)%MOD;
      base=(base*base)%MOD;
      exp/=2;
   }
   return res%MOD;
}
