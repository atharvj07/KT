n = int(input())
max_floors = list(map(int, input().split(' ')))
best_solution = [0] * n
best_total_floors = 0
for (i, mfi) in enumerate(max_floors):
	solution = max_floors.copy()
	for j in range(i - 1, -1, -1):
		solution[j] = min(max_floors[j], solution[j + 1])
	for j in range(i + 1, len(max_floors)):
		solution[j] = min(max_floors[j], solution[j - 1])
	solution_total_floors = sum(solution)
	if solution_total_floors > best_total_floors:
		best_solution = solution
		best_total_floors = solution_total_floors
print(' '.join(map(str, best_solution)))
