import sys
from math import *

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())
(n, k) = mints()
if k == 1:
	print(n)
	exit(0)
m = dict()
for i in mints():
	c = 0
	x = i
	while x % k == 0:
		x //= k
		c += 1
	if x not in m:
		m[x] = [0] * 34
	m[x][c] += 1
r = 0
for i in m:
	f = True
	for j in m[i]:
		if f and j != 0:
			r += 1
			f = False
		else:
			f = True
print(r)
