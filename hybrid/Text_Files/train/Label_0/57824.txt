

import java.awt.Point;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;
import java.util.PriorityQueue;
import java.util.Queue;
import java.util.Stack;

public class DDiv1 {
    public static void main(String[] args) throws NumberFormatException,
            IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        String[] S = in.readLine().split(" ");
        int[] A = new int[n];
        for (int i = 0; i < n; i++)
            A[i] = Integer.parseInt(S[i]);
        LinkedList<Integer>[] E = new LinkedList[n];
        for (int i = 0; i < n; i++)
            E[i] = new LinkedList<Integer>();
        for (int i = 1; i < n; i++) {
            S = in.readLine().split(" ");
            int x = Integer.parseInt(S[0]) - 1;
            int y = Integer.parseInt(S[1]) - 1;
            E[x].add(y);
            E[y].add(x);
        }
        int start = Integer.parseInt(in.readLine()) - 1;
        Queue<Integer> Q = new LinkedList<Integer>();
        Q.add(start);
        boolean[] visited = new boolean[n];
        int[] N = new int[n];
        int c = 0;
        while (!Q.isEmpty()) {
            int temp = Q.poll();
            visited[temp] = true;
            N[c++] = temp;
            int toRemove = -1;
            for (int y : E[temp])
                if (visited[y])
                    toRemove = y;
                else
                    Q.add(y);
            if (toRemove != -1)
                E[temp].remove((Integer) toRemove);
        }
        beaver[] DP = new beaver[n];
        A[N[0]]++;
        for (int i = n - 1; i >= 0; i--) {
            A[N[i]]--;
            long ans = 0;
            long children = 0;
            PriorityQueue<Long> P = new PriorityQueue<Long>();
            for (int y : E[N[i]]) {
                P.add(-DP[y].x);
                children += DP[y].y;
            }
            while (!P.isEmpty() && A[N[i]] > 0) {
                ans += (-P.poll()) + 2;
                A[N[i]]--;
            }
            long min = Math.min(A[N[i]], children);
            ans += 2 * min;
            A[N[i]] -= min;
            DP[N[i]] = new beaver(ans, A[N[i]]);
        }
        System.out.println(DP[N[0]].x);
    }
}

class beaver {
    long x;
    long y;

    public beaver(long i, long j) {
        x = i;
        y = j;
    }
}
