import sys

def solve():
	(n, m, k) = map(int, input().split())
	nums = list(map(int, input().split()))
	ops = [tuple(map(int, input().split())) for _ in range(m)]
	queries = [tuple(map(int, input().split())) for _ in range(k)]
	cnt_ops = [0 for i in range(m + 1)]
	for (l, r) in queries:
		l -= 1
		r -= 1
		cnt_ops[l] += 1
		cnt_ops[r + 1] -= 1
	for i in range(1, m + 1):
		cnt_ops[i] = cnt_ops[i - 1] + cnt_ops[i]
	deltas = [0 for i in range(n + 1)]
	for (ind, (l, r, d)) in enumerate(ops):
		l -= 1
		r -= 1
		deltas[l] += d * cnt_ops[ind]
		deltas[r + 1] -= d * cnt_ops[ind]
	sum_delta = [0 for i in range(n)]
	for i in range(n):
		if i == 0:
			sum_delta[i] = deltas[0]
		else:
			sum_delta[i] = sum_delta[i - 1] + deltas[i]
	for i in range(n):
		nums[i] += sum_delta[i]
		print(nums[i], end=' ')
input = sys.stdin.readline
solve()
