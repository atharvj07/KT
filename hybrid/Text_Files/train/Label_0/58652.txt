
import java.util.*;
import java.io.*;
public class Codeforces {
	static int MAX = (int)1e5+1;
	static boolean leaf[];
	static int count[],n;
	public static void main(String[] args) {
		Scanner input = new Scanner(System.in);
		int test = input.nextInt();
		while(test-->0){
			n = input.nextInt();
			count = new int[n];
			for(int i =0;i<n;i++)count[input.nextInt()-1]++;
			int l =0;
			int r = n-1;
			int res =0;
			while(l<=r) {
				int mid = (l+r)/2;
				if(check(mid)) {
					res = mid;
					l = mid+1;
				}else {
					r = mid-1;
				}
			}
			System.out.println(res-1);
		}
	}
	public static boolean check(int mid) {
		ArrayList<Integer>lists[] = new ArrayList[n];
		for(int i =0;i<n;i++)lists[i] = new ArrayList<>();
		PriorityQueue<Integer>q = new PriorityQueue<Integer>();
		for(int it:count) lists[0].add(-it);
		for(int i =0;i<n;i++) {
			for(int it:lists[i])q.add(it);
			if(q.isEmpty())return false;
			int val = q.remove();
			//System.out.println(val+"mid being "+mid);
			val++;
			if(val != 0) {
				if(i+mid>=n)return false;
				lists[i+mid].add(val);
			}
		}
		return true;
	}
	
}
