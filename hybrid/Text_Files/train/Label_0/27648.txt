#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int> > res;
vector<int> res1;
vector<int> pa;
int fi(int x) {
  if (pa[x] == x) return x;
  return pa[x] = fi(pa[x]);
}
void uni(int a, int b) {
  int x = fi(a);
  int y = fi(b);
  if (x == y) {
    res.push_back(make_pair(a, b));
  } else {
    pa[x] = y;
  }
}
int main() {
  int n, x, y;
  cin >> n;
  pa.push_back(0);
  for (int i = 1; i <= n; i++) {
    pa.push_back(i);
  }
  for (int i = 0; i < n - 1; i++) {
    cin >> x >> y;
    uni(x, y);
  }
  for (int i = 1; i <= n; i++) {
    if (fi(i) == i) {
      res1.push_back(i);
    }
  }
  cout << res.size() << endl;
  if (res1.size() <= 1) {
    return 0;
  }
  for (int i = 0; i < res1.size() - 1; i++) {
    cout << res[i].first << " " << res[i].second << " " << res1[i] << " "
         << res1[i + 1] << endl;
  }
  return 0;
}
