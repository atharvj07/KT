#include<bits/stdc++.h>

#define REP(i,s,n) for(int i=s;i<n;i++)
#define rep(i,n) REP(i,0,n)

using namespace std;

short dp[1<<12][1025],L[1000],bc[1<<12];
int N,M;

int main(){
  rep(i,(1<<12)) bc[i] = __builtin_popcount(i);
  while( cin >> N >> M, N|M ){
    rep(i,N) cin >> L[i];
    reverse(L,L+N);
    int limit = (1<<M);
    rep(i,limit) rep(j,N+M+5) dp[i][j] = -1;        // invalid
    rep(i,limit) if( bc[i] == L[0] ) dp[i][0] = -2; // terminator

    rep(i,N-1){
      rep(j,limit){
        if( dp[j][i] == -1 ) continue;
        int bitmask = (j<<1) & ((1<<M)-1);
        int target = ( (j>>(M-1)) & 1 );
        rep(k,2){
          int nbitmask = bitmask | k;
          if( bc[nbitmask] == L[i+1] ) dp[nbitmask][i+1] = target;
        }
      }
    }

    string ans = "";
    int bitmask = -1;
    rep(i,limit) if( dp[i][N-1] != -1 ){
      bitmask = i;
      rep(j,M) {
        if( (i>>j) & 1 ) ans += "1";
        else             ans += "0";
      }
      break;
    }
    if( bitmask == -1 ) {
      puts("Goofy");
      continue;
    }
    for(int i=N-1;i>=1;i--){
      int v = dp[bitmask][i];
      ans += string(1,(char)('0'+v));
      assert( v == 0 || v == 1);
      int tmp = bitmask;
      bitmask  >>= 1;
      bitmask |= (v<<(M-1));
      bitmask &= ((1<<M)-1);
      assert( bc[bitmask] == L[i-1] );
    }
    cout << ans << endl;
  }
  return 0;
}
