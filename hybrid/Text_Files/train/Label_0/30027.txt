#include <bits/stdc++.h>
using namespace std;
bool found[1 << 19];
bool win[1 << 19];
int now = 0, pos = 0;
int Link[3][19] = {1,  2,  -1, 4,  5,  6,  -1, 8,  9,  10, 11, -1, 13, 14, 15,
                   -1, 17, 18, -1, 4,  5,  6,  8,  9,  10, 11, 12, 13, 14, 15,
                   -1, 16, 17, 18, -1, -1, -1, -1, 3,  4,  5,  7,  8,  9,  10,
                   -1, 12, 13, 14, 15, -1, 16, 17, 18, -1, -1, -1};
bool Find(int sta) {
  if (found[sta] == true) return win[sta];
  found[sta] = true;
  if (sta == 0) {
    win[sta] = false;
    return false;
  }
  if ((sta & (sta - 1)) == 0) {
    win[sta] = true;
    return true;
  }
  win[sta] = false;
  for (int i = 0; i < 19; i++) {
    if ((sta & (1 << i)) > 0) {
      for (int j = 0; j < 3; j++) {
        int tmp = sta, pos = i;
        while (pos != -1 && (tmp & (1 << pos)) > 0) {
          tmp = (~(1 << pos)) & tmp;
          win[sta] |= !Find(tmp);
          if (win[sta] == true) {
            return true;
          }
          pos = Link[j][pos];
        }
      }
    }
  }
  return win[sta];
}
int main() {
  memset(found, 0, sizeof(found));
  while (pos < 19) {
    char c = getchar();
    if (c == '.')
      pos++;
    else if (c == 'O') {
      now |= 1 << pos;
      pos++;
    }
  }
  if (Find(now) == true) {
    puts("Karlsson");
  } else {
    puts("Lillebror");
  }
  return 0;
}
