#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k = 0, a[2010];
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int i = 0; i < n; i++) {
    k = 0;
    for (int j = 0; j < n; j++) {
      if (a[j] > a[i]) k++;
    }
    printf("%d ", k + 1);
  }
  return 0;
}
