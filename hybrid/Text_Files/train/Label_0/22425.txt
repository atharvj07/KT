#include <bits/stdc++.h>
using namespace std;
char s[200010];
int fv[200010], hv[200010];
int n;
int calc(int L, int R) {
  return ((hv[R] - 1LL * hv[L - 1] * fv[R - L + 1]) % 998244353 + 998244353) %
         998244353;
}
map<pair<int, int>, vector<int> > mp[200010];
int main() {
  scanf("%d", &n);
  scanf("%s", s + 1);
  fv[0] = 1;
  for (int i = 1; i <= n; i++) {
    fv[i] = 1LL * fv[i - 1] * 37 % 998244353;
    hv[i] = (1LL * hv[i - 1] * 37 + s[i] - 'a' + 1) % 998244353;
  }
  for (int i = 1; i <= n; i++)
    mp[1][pair<int, int>(calc(i, i), 1)].push_back(i);
  int ans;
  for (int i = 1; i <= n; i++) {
    if (mp[i].size()) {
      ans = i;
      for (auto R : mp[i]) {
        vector<int> &vt = R.second;
        int len = R.first.second;
        for (int j = 0; j < vt.size() - 1; j++) {
          int id = j;
          while (id < vt.size() - 1 && calc(vt[id] + len, vt[id + 1] - 1) ==
                                           calc(vt[j] + len, vt[j + 1] - 1))
            id++;
          id--;
          mp[i + id - j + 1][pair<int, int>(calc(vt[j], vt[id + 1] + len - 1),
                                            vt[id + 1] + len - vt[j])]
              .push_back(vt[j]);
          j = id;
        }
      }
    }
  }
  printf("%d\n", ans);
}
