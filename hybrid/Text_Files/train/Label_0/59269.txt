import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.HashSet;
import java.util.StringTokenizer;
import java.util.HashMap;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Scanner in = new Scanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        HBerOSFileSuggestion solver = new HBerOSFileSuggestion();
        solver.solve(1, in, out);
        out.close();
    }

    static class HBerOSFileSuggestion {
        int n;
        String[] arr;

        public void readInput(Scanner sc) {
            n = sc.nextInt();
            arr = new String[n];
            for (int i = 0; i < n; i++)
                arr[i] = sc.next();
        }

        public void solve(int testNumber, Scanner sc, PrintWriter pw) {
            int q = 1;
            while (q-- > 0) {
                readInput(sc);
                HashMap<String, Integer> map = new HashMap<>();
                HashMap<String, String> map2 = new HashMap<>();
                for (int i = 0; i < n; i++) {
                    HashSet<String> set = new HashSet<>();
                    for (int j = 0; j < arr[i].length(); j++)
                        for (int k = j; k < arr[i].length(); k++) {
                            String s = arr[i].substring(j, k + 1);
                            set.add(s);
                            map2.put(s, arr[i]);
                        }
                    for (String x : set)
                        map.put(x, map.getOrDefault(x, 0) + 1);
                }
                int tc = sc.nextInt();
                while (tc-- > 0) {
                    String s = sc.next();
                    pw.println(map.getOrDefault(s, 0) + " " + map2.getOrDefault(s, "-"));
                }
            }
        }

    }

    static class Scanner {
        StringTokenizer st;
        BufferedReader br;

        public Scanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }

        public String next() {
            try {
                while (st == null || !st.hasMoreTokens())
                    st = new StringTokenizer(br.readLine());
                return st.nextToken();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
