def main(s):
	isl = 0
	isu = 0
	for i in s:
		if i.isupper():
			isu += 1
		else:
			isl += 1
	retstr = int(abs(isl - isu) / 2)
	k = 0
	i = 0
	for j in s:
		if isl < isu and k < retstr and j.isupper():
			s[i] = j.lower()
			k += 1
		elif isl > isu and k < retstr and j.islower():
			s[i] = j.upper()
			k += 1
		i += 1
	s = ''.join(s)
	return str(retstr) + '\n' + s
n = input()
srting = list(input())
print(main(srting))
