#include <bits/stdc++.h>
using namespace std;



long long p = 5e9;

long long isprime(long long n){
	for( long long i = 2 ; i * i <= n ; i++ ){
		if( n % i == 0 ) return 0;
	}
	return 1;
}

inline long long mul(long long a,long long b){
	long long res = 0;
	while( b ){
		if(b&1){
			res += a;
			if( res >= p ) res -= p;
		}
		a *= 2;
		if( a >= p ) a -= p;
		b >>= 1;
	}
	return res;
}

inline long long bpow(long long a,long long b){
	long long res = 1;
	while( b ){
		if(b&1){
			res = mul(res,a);
			res %= p;
		}
		a = mul(a,a);
		b >>= 1;
	}
	return res;
}


int main(){
	while( !isprime(p) ) p++;

	int N;
	cin >> N;
	long long X = 0;
	for(int i = 0 ; i < N ; i++){
		long long op,Y;
		scanf("%lld %lld",&op,&Y);
		Y=(Y+p)%p;
		if( op == 1 ){ //+
			X = X + Y;
			X %= p;
		}else if( op == 2 ){ //-
			X = X - Y;
			X %= p;
		}else if( op == 3 ){ //*
			X = mul(X,Y);
		}else{ // /
			X = mul(X,bpow(Y,p-2));
		}
		X = (X+p) % p;
	}
	if( X < (1ll<<31) ){
		cout << X << endl;
	}else{
		cout << X-p << endl;
	}
}
