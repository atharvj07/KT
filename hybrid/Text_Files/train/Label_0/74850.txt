(a, b) = list(map(int, input().strip().split()))
stra = '{0:b}'.format(a)
strb = '{0:b}'.format(b)
stra1 = stra + '1'
strar = stra[::-1].strip('0')
strar1 = stra1[::-1].strip('0')
strsa = [stra.strip('0'), stra1, strar, strar1]
ans = stra == strb
for string in strsa:
	pos = strb.find(string)
	if pos == -1:
		continue
	b = strb[:pos] + strb[pos + len(string):]
	if b == '1' * len(b):
		ans = True
		break
print('YES' if ans else 'NO')
