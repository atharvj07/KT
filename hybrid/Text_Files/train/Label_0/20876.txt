#include <bits/stdc++.h>
using namespace std;
template <typename A>
ostream &operator<<(ostream &cout, vector<A> const &v);
template <typename A, typename B>
ostream &operator<<(ostream &cout, pair<A, B> const &p) {
  return cout << "(" << p.f << ", " << p.s << ")";
}
template <typename A>
ostream &operator<<(ostream &cout, vector<A> const &v) {
  cout << "[";
  for (int i = 0; i < v.size(); i++) {
    if (i) cout << ", ";
    cout << v[i];
  }
  return cout << "]";
}
template <typename A, typename B>
istream &operator>>(istream &cin, pair<A, B> &p) {
  cin >> p.first;
  return cin >> p.second;
}
vector<set<long long>> adj;
void solve() {
  long long v, e;
  cin >> v >> e;
  if (e <= 0) {
    cout << -1 << '\n';
    return;
  }
  long long n = v;
  adj = vector<set<long long>>(v + 5);
  set<long long> fs1, fs2, fs3;
  set<long long> s1, temps2;
  for (long long i = 0, u, v; i < e; i++) {
    cin >> u >> v;
    adj[u].insert(v);
    adj[v].insert(u);
    s1.insert(u);
    s1.insert(v);
  }
  for (long long i : adj[*s1.begin()]) {
    s1.erase(i);
    temps2.insert(i);
  }
  fs1 = s1;
  for (long long i : temps2) {
    for (long long x : adj[i]) {
      if (fs1.find(x) == fs1.end()) {
        fs3.insert(x);
        temps2.erase(x);
      }
    }
  }
  fs2 = temps2;
  bool ok = true;
  for (int i : fs1) {
    for (int j : fs2) {
      if (adj[i].find(j) == adj[i].end()) {
        ok = false;
        break;
      }
    }
  }
  for (int i : fs1) {
    for (int j : fs3) {
      if (adj[i].find(j) == adj[i].end()) {
        ok = false;
        break;
      }
    }
  }
  for (int i : fs2) {
    for (int j : fs3) {
      if (adj[i].find(j) == adj[i].end()) {
        ok = false;
        break;
      }
    }
  }
  for (long long i = 1; i <= n && ok; i++) {
    if (fs1.find(i) != fs1.end()) {
      for (long long x : adj[i]) {
        long long counter =
            (fs2.find(x) != fs2.end()) + (fs3.find(x) != fs3.end());
        if (counter ^ 1) {
          ok = false;
          break;
        }
      }
    } else if (fs2.find(i) != fs2.end()) {
      for (long long x : adj[i]) {
        long long counter =
            (fs1.find(x) != fs1.end()) + (fs3.find(x) != fs3.end());
        if (counter ^ 1) {
          ok = false;
          break;
        }
      }
    } else if (fs3.find(i) != fs3.end()) {
      for (long long x : adj[i]) {
        long long counter =
            (fs1.find(x) != fs1.end()) + (fs2.find(x) != fs2.end());
        if (counter ^ 1) {
          ok = false;
          break;
        }
      }
    } else {
      ok = false;
    }
  }
  if (fs1.size() == 0 || fs2.size() == 0 || fs3.size() == 0 ||
      (fs1.size() + fs2.size() + fs3.size()) != v) {
    ok = false;
  }
  if (ok) {
    for (long long i = 1; i <= n; i++) {
      if (fs1.find(i) != fs1.end()) {
        cout << 1 << " ";
      }
      if (fs2.find(i) != fs2.end()) {
        cout << 2 << " ";
      }
      if (fs3.find(i) != fs3.end()) {
        cout << 3 << " ";
      }
    }
    cout << '\n';
  } else {
    cout << -1 << '\n';
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int testcase = 1;
  for (int i = 0; i < testcase; i++) {
    solve();
  }
  return 0;
}
