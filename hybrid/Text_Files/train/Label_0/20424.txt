#include <cstdio>
int a[10], e[8], used[11];
bool ok() {
	for (int i = 0; i < 3; i++) e[i] = (a[i] + a[i + 3] + a[i + 6]) % a[9];
	for (int i = 0; i < 3; i++) e[i + 3] = (a[i * 3] + a[i * 3 + 1] + a[i * 3 + 2]) % a[9];
	e[6] = (a[2] + a[4] + a[6]) % a[9];
	e[7] = (a[0] + a[4] + a[8]) % a[9];
	for (int i = 0; i < 7; i++) if (e[i] != e[i + 1]) return false;
	return true;
}
int solve(int depth) {
	if (depth == 10) return ok() ? 1 : 0;
	if (a[depth]) return solve(depth + 1);
	int ret = 0;
	for (int i = 1; i <= 10; i++) {
		if (!used[i]) {
			used[i] = 1; a[depth] = i;
			ret += solve(depth + 1);
			used[i] = a[depth] = 0;
		}
	}
	return ret;
}
int main() {
	while (true) {
		for (int i = 1; i < 11; i++) used[i] = 0;
		for (int i = 0; i < 10; i++) {
			scanf("%d", a + i);
			if (a[i]) used[a[i]] = 1;
		}
		if (a[0] == -1) break;
		printf("%d\n", solve(0));
	}
}
