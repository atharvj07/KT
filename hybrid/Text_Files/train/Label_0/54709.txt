import java.io.*;
import java.util.*;

public class CF948B extends PrintWriter {
	CF948B() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF948B o = new CF948B(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int[] aa = new int[n + 1];
		for (int p = 2; p <= n; p++) {
			if (aa[p] != 0)
				continue;
			for (int q = p + p; q <= n; q += p)
				aa[q] = p;
		}
		for (int x = 2; x <= n; x++)
			if (aa[x] != 0)
				aa[x] = x - aa[x] + 1;
		int ans = n;
		for (int x = aa[n]; x <= n; x++)
			if (aa[x] >= 3)
				ans = Math.min(ans, aa[x]);
		println(ans);
	}
}
