
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {

	public static void quickSort(int array[], int start, int end)
	{
		int i = start;
		int k = end;
		if(end-start>0)
        {
			int pivot = array[start];
			while (k > i)
			{
				while(array[i]<=pivot && i<=end && k>i)
					i++;
				while(array[k]>pivot && k>=start && k>=i)
					k--;
				if (k > i) {
					int temp = array[i];
					array[i] = array[k];
					array[k] = temp;
				}
			}
        	int temp = array[start];
    		array[start] = array[k];
    		array[k] = temp;
			quickSort(array, start, k - 1);
			quickSort(array, k + 1, end);
        }
	}

	public static void main(String[] args) throws IOException {
		BufferedReader instream = new BufferedReader(new InputStreamReader(System.in));
		int total = Integer.parseInt(instream.readLine());
		int[] time = new int[total];
		String[] tokens = instream.readLine().split(" ");
		for(int i=0; i<total; i++)
			time[i] = Integer.parseInt(tokens[i]);
		
		quickSort(time, 0, time.length-1);
		
		int timeleft = 710;
		int penalty = 0;
		int solved = 0;
		int current = 0;

		while(current<total && timeleft>=time[current]){
			timeleft -= time[current];
			penalty += (timeleft >= 360) ? 0 : (360 - timeleft);
			solved++;
			current++;
		}
		
		System.out.print(solved);
		System.out.print(" ");
		System.out.println(penalty);	
	}
}
