t = int(input())
for _ in range(t):
	(n, x) = map(int, input().split())
	l = list(map(int, input().split()))
	if sum(l) // n >= x:
		print(n)
	else:
		c = 0
		l.sort(reverse=True)
		s = 0
		f = 0
		for i in range(n):
			s += l[i]
			if s // (i + 1) >= x:
				c += 1
			else:
				f == 1
				break
		print(c)
