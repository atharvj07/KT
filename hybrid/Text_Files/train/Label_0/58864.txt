import static java.util.Arrays.* ;
import static java.lang.Math.* ;

import java.util.*;
import java.io.*;

public class Main
{
    static int size [] ;
    static int depth [] ;
    static int [][] adjList;
    static int [] a ;
    static int K = 19 ;
    static long ans  = 0 ;
    static int getSize(int u , int p)
    {
        size[u] = 1 ;
        for(int v : adjList[u])
            if(v != p && depth[v] == -1)
                size[u] += getSize( v,  u) ;

        return size[u] ;
    }
    static int getCentroid(int u , int p  ,int n )
    {
        for(int v : adjList[u])
            if(v != p && depth[v] == -1 && size[v] * 2 > n)
                return getCentroid(v , u , n)  ;

        return u ;
    }
    static int computeOnes(int u , int p , int [] curr , int dist)
    {
        int size = 1 ;
        ans += dist ;

        for(int i = 0 ; i <= K ; i++)
            if(((dist >> i) & 1) == 1)
                curr[i] ++ ;

        for(int v : adjList[u])
            if(v != p && depth[v] == -1)
                size += computeOnes(v , u , curr , dist ^ a[v]) ;

        return size ;
    }

    static void decompose(int u , int lvl)
    {
        int centroid = getCentroid(u , -1 , getSize(u , -1)) ;

        depth[centroid] = lvl ;
        int [] totOnes = new int [K + 1] ;
        int [] curr ;
        int totSize = 0 ;
        ans += a[centroid] ;
        for(int v : adjList[centroid])
            if(depth[v] == -1)
            {
                int currSize = computeOnes(v , -1 , curr = new int [K + 1] , a[centroid] ^ a[v]) ;

                for(int i = 0 ; i <= K ; i++)
                {
                    if (((a[centroid] >> i) & 1) == 1)
                        ans += (1l << i) * (totOnes[i] * curr[i] +  (totSize - totOnes[i]) * (currSize - curr[i]));
                    else
                        ans += (1l << i) * (totOnes[i] * (currSize - curr[i]) + (totSize - totOnes[i]) * curr[i]);
                    totOnes[i] += curr[i] ;
                }
                totSize += currSize ;
            }
        for(int v : adjList[centroid])
            if(depth[v] == -1)
                decompose(v , lvl + 1) ;
    }

    public static void main (String [] args) throws Exception
    {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        int N = sc.nextInt() ;
        a = new int [N] ;
        for(int i  =0 ; i < N ;i++) a[i] = sc.nextInt() ;

        depth = new int [N] ;
        fill(depth , -1 );
        size = new int [N] ;
        int [] deg = new int [N] , x = new int [N - 1] , y = new int [N - 1] ;

        for(int i = 0 ; i < N - 1 ; i++)
        {
            deg[x[i] = sc.nextInt() - 1] ++ ;
            deg[y[i] = sc.nextInt() - 1] ++ ;
        }
        adjList = new int [N][] ;
        for(int i = 0 ; i < N ; i++)adjList[i] = new int [deg[i]] ;

        for(int i = 0 ; i < N - 1 ; i++)
        {
            adjList[x[i]][--deg[x[i]]] = y[i] ;
            adjList[y[i]][--deg[y[i]]] = x[i] ;
        }

        decompose(0 , 0 ) ;

        out.println(ans);

        out.flush();
        out.close();
    }

    static class Scanner
    {
        BufferedReader br;
        StringTokenizer st;

        Scanner(InputStream in)
        {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() throws Exception
        {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        int nextInt() throws Exception { return Integer.parseInt(next()); }

        long nextLong() throws Exception { return Long.parseLong(next()); }

        double nextDouble() throws Exception { return Double.parseDouble(next());}
    }

}
