#include <bits/stdc++.h>
using namespace std;
const int INFTY = 20000000;
const int MAX = 500100;
const int MOD = 10000000;
void coutTab(int* tab, int n) {
  for (int i = 0; i < n; i++) {
    cout << tab[i] << " ";
  }
  cout << "\n";
}
string create(int a, int b, int x, int start) {
  if (start == 0)
    a--;
  else
    b--;
  string s;
  s.push_back(start + '0');
  char l;
  for (int i = 0; i < x - 1; i++) {
    l = s[s.size() - 1];
    if (l == '1') {
      s.push_back('0');
      a--;
    } else {
      s.push_back('1');
      b--;
    }
  }
  l = s[s.size() - 1];
  if (l == '0') {
    while (a--) s.push_back('0');
    while (b--) s.push_back('1');
  } else {
    while (b--) s.push_back('1');
    while (a--) s.push_back('0');
  }
  return s;
}
bool check(string s, int x) {
  int j = 0;
  for (int i = 1; i < s.size(); i++) {
    if (s[i - 1] != s[i]) j++;
  }
  return x == j;
}
int main() {
  ios_base::sync_with_stdio(0);
  int a, b, x;
  cin >> a >> b >> x;
  string res = create(a, b, x, 0);
  if (!check(res, x)) res = create(a, b, x, 1);
  cout << res << "\n";
}
