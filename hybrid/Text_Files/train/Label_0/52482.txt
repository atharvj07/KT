
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringJoiner;
import java.util.TreeSet;

public class D {

    static class Interval implements Comparable<Interval> {
        int start;
        int end;

        public Interval(int start, int end) {
            this.start = start;
            this.end = end;
        }

        @Override
        public int compareTo(Interval o) {
            return this.start - o.start;
        }
    }

    private static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    public static void main(String [] args) throws IOException {

        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));

        String [] parts = bufferedReader.readLine().split(" ");
        int N = Integer.parseInt(parts[0]);
        int K = Integer.parseInt(parts[1]);

        int [] arr = new int [N];
        for(int i=1;i<=N;i++) {
            arr[i-1] = i;
        }

        if (K % 2 == 0 || 2 * N <= K) {
            System.out.println(-1);
            return;
        }

        TreeSet<Interval> intervalTreeSet = new TreeSet<>();
        intervalTreeSet.add(new Interval(0, N));

        while (K != 1) {

            TreeSet<Interval> newTreeSet = new TreeSet<>();

            while (!intervalTreeSet.isEmpty() && K > 1) {
                Interval top = intervalTreeSet.pollFirst();

                if (top.end - top.start > 1) {
                    int mid = (top.end + top.start) / 2;
                    swap(arr, mid - 1, mid);
                    newTreeSet.add(new Interval(top.start, mid));
                    newTreeSet.add(new Interval(mid, top.end));
                    K-=2;
                }
            }

            if (intervalTreeSet.size() == newTreeSet.size()) {
                break;
            }

            intervalTreeSet = newTreeSet;
        }

        if (K != 1) {
            System.out.println(-1);
        } else {
            StringJoiner joiner = new StringJoiner(" ");
            for(int element : arr) {
                joiner.add(element+" ");
            }

            System.out.println(joiner);
        }

    }
}
