#include <bits/stdc++.h>
using namespace std;
const int N = 1e2 + 5;
int n;
string s, ans;
int main() {
  while (scanf("%d", &n) != EOF) {
    cin >> s;
    ans = string(n, '9');
    for (int add = 0; add < 10; add++) {
      for (int shift = 0; shift < n; shift++) {
        ans = min(ans, s.substr(shift, n - shift) + s.substr(0, shift));
      }
      for (int i = 0; i < n; i++) {
        s[i]++;
        if (s[i] > '9') s[i] -= 10;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
