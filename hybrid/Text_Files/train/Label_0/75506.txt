(a, b, c) = map(int, input().split())
a1 = a // 3
b1 = b // 2
c1 = c // 2
ans = min(a1, b1, c1)
a -= ans * 3
b -= ans * 2
c -= ans * 2
if a == 0:
	if b >= 1:
		if c >= 1:
			print(ans * 7 + 2)
		else:
			print(ans * 7 + 1)
	elif c >= 1:
		print(ans * 7 + 1)
	else:
		print(ans * 7)
elif b == 0:
	if c >= 2:
		print(ans * 7 + 3)
	else:
		print(ans * 7 + 2)
elif c == 0:
	if a == 1:
		print(ans * 7 + 2)
	elif b == 1:
		print(ans * 7 + 3)
	else:
		print(ans * 7 + 4)
elif a == 1:
	if b == 1:
		if c == 1:
			print(ans * 7 + 3)
		else:
			print(ans * 7 + 4)
	elif c == 1:
		print(ans * 7 + 3)
	else:
		print(ans * 7 + 5)
elif a == 2:
	if b >= 2:
		if c >= 2:
			print(ans * 7 + 6)
		elif c == 1:
			print(ans * 7 + 5)
	elif b == 1:
		if c >= 1:
			print(ans * 7 + 4)
		else:
			print(ans * 7 + 3)
elif b == 1:
	if c >= 2:
		print(ans * 7 + 6)
	elif c == 1:
		print(ans * 7 + 6)
elif c == 1:
	print(ans * 7 + 6)
