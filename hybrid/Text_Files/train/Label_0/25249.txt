#include <bits/stdc++.h>
using namespace std;
void output(int k, int ct) {
  for (int i = 0; i <= k; i++) {
    if (ct && i == ct) printf("()");
    printf("(");
  }
  for (int i = 0; i <= k; i++) printf(")");
}
long long binary(long long l, long long r, long long sum) {
  if (l == r) return l;
  long long m = (l + r) >> 1;
  if ((m + 1) * (m + 2) / 2 <= sum) return binary(m + 1, r, sum);
  return binary(l, m, sum);
}
int main() {
  long long n, sum;
  while (cin >> n >> sum) {
    long long k = binary(0, 2e9, sum);
    long long s = k * (k + 1) / 2;
    long long ct = sum - s;
    long long MIN = k + 1;
    if (ct) MIN++;
    if (n < MIN)
      printf("Impossible\n");
    else {
      for (int i = 0; i < n - MIN; i++) printf("()");
      output(k, ct);
      printf("\n");
    }
  }
  return 0;
}
