def ToyCars(n, arr):
	badCar = [0] * n
	i = 0
	while i < n:
		j = i + 1
		seq = [int(x) for x in arr[i].split()]
		while j < n:
			if seq[j] == 1:
				badCar[i] = 1
			elif seq[j] == 2:
				badCar[j] = 1
			elif seq[j] == 3:
				badCar[j] = 1
				badCar[i] = 1
			j += 1
		i += 1
	i = 0
	totalCnt = 0
	result = ''
	while i < n:
		if badCar[i] == 0:
			totalCnt += 1
			result += str(i + 1)
			result += ' '
		i += 1
	return (totalCnt, result[:len(result) - 1])
n = int(input())
i = 0
arr = []
while i < n:
	arr.append(input())
	i += 1
(totalCnt, result) = ToyCars(n, arr)
print(totalCnt)
if result != '':
	print(result)
