#include <bits/stdc++.h>
using namespace std;
long long int dir[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
void solve() {
  long long int n, m, k, x, s, can_remove = 0;
  cin >> n >> m >> k >> x >> s;
  vector<pair<long long int, long long int>> v1(m), v2(k);
  for (int i = 0; i < m; i++) cin >> v1[i].first;
  for (int i = 0; i < m; i++) cin >> v1[i].second;
  for (int i = 0; i < k; i++) cin >> v2[i].first;
  for (int i = 0; i < k; i++) {
    cin >> v2[i].second;
    if (s >= v2[i].second) can_remove = v2[i].first;
  }
  long long int ans = n * x;
  ans = min(ans, (n - can_remove) * x);
  for (int i = 0; i < m; i++) {
    long long int pts = v1[i].second;
    if (pts > s) continue;
    int l = -1, r = k, mid;
    while (l + 1 < r) {
      mid = (l + r) / 2;
      if (v2[mid].second <= s - pts)
        l = mid;
      else
        r = mid;
    }
    if (l != -1) {
      long long int temp = (n - v2[l].first) * v1[i].first;
      ans = min(ans, temp);
    } else {
      long long int temp = n * v1[i].first;
      ans = min(ans, temp);
    }
  }
  cout << ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cout.tie(NULL);
  cin.tie(NULL);
  ;
  long long int t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
