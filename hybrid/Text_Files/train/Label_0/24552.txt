#include <bits/stdc++.h>
using namespace std;
const int oo = 1e9;
const long long mod = 100000000;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n, x, y;
  cin >> n >> x >> y;
  string s;
  cin >> s;
  int c = 0, i = 0;
  for (; i < y; ++i) {
    if (s[n - i - 1] == '0') continue;
    c++;
  }
  if (s[n - i - 1] == '0') c++;
  ++i;
  for (; i < x; ++i) {
    if (s[n - i - 1] == '0') continue;
    c++;
  }
  cout << c;
  return 0;
}
