from sys import stdin
(n, k) = map(int, stdin.readline().strip().split())
s = list(map(int, stdin.readline().strip().split()))
d = dict()
for i in range(k):
	d[i] = 0
for i in s:
	x = i % k
	d[x] += 1
ans = d[0] // 2 * 2
for i in range(1, k):
	if i > k - i:
		continue
	if i != k - i:
		y = min(d[i], d[k - i])
	else:
		y = d[i] // 2
	ans += y * 2
print(ans)
