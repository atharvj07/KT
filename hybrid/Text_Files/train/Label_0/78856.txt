a = []
answ = []

def solve(n, num):
	if n == 1:
		answ.append(str(num))
		return
	if n == 2:
		answ.append(str(num))
		answ.append(str(num * 2))
		return
	if n == 3:
		answ.append(str(num))
		answ.append(str(num))
		answ.append(str(num * 3))
		return
	for i in range(n):
		if a[i] % 2 == 1:
			answ.append(str(num))
	for i in range(n // 2):
		a[i] = a[i * 2 + 1] / 2
	solve(n // 2, num * 2)
n = int(input())
for i in range(n):
	a.append(i + 1)
solve(n, 1)
print(' '.join(answ))
