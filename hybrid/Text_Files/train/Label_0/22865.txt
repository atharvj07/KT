#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
long long powmod(long long a, long long b) {
  long long res = 1;
  a %= mod;
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
  }
  return res;
}
const int N = 1010;
int n, m, k, a[N][N], px, py;
long long ps[N][N][4], ps2[N][N][4], ret[N][N], ans;
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i < n + 1; i++)
    for (int j = 1; j < m + 1; j++) scanf("%d", &a[i][j]);
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 >= k && j + 1 <= m - k)
        ps[i][j][0] = ps[i - 1][j + 1][0] - a[i - k][j + 1] + a[i][j - k + 1];
      else
        for (int l = 0; l < k; l++) ps[i][j][0] += a[i - (k - 1) + l][j - l];
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 >= k && j - 1 >= k)
        ps[i][j][1] = ps[i - 1][j - 1][1] - a[i - k][j - 1] + a[i][j + k - 1];
      else
        for (int l = 0; l < k; l++) ps[i][j][1] += a[i - (k - 1) + l][j + l];
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 >= k && j + 1 <= m - k)
        ps[i][j][2] = ps[i - 1][j + 1][2] - a[i - 1][j + k] + a[i + k - 1][j];
      else
        for (int l = 0; l < k; l++) ps[i][j][2] += a[i + (k - 1) - l][j + l];
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 >= k && j - 1 >= k)
        ps[i][j][3] = ps[i - 1][j - 1][3] - a[i - 1][j - k] + a[i + k - 1][j];
      else
        for (int l = 0; l < k; l++) ps[i][j][3] += a[i + (k - 1) - l][j - l];
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 < k && j - 1 < k) {
        for (int dx = -k + 1; dx < 1; dx++)
          for (int dy = -k + 1; dy < k; dy++)
            if (abs(dx) + abs(dy) <= k - 1) ps2[i][j][0] += a[i + dx][j + dy];
      } else if (j - 1 >= k)
        ps2[i][j][0] = ps2[i][j - 1][0] - ps[i][j - 1][0] + ps[i][j][1];
      else {
        ps2[i][j][0] =
            ps2[i - 1][j][0] - ps[i - 1][j][0] - ps[i - 1][j][1] + a[i - k][j];
        for (int dy = -k + 1; dy < k; dy++) ps2[i][j][0] += a[i][j + dy];
      }
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = m + 2 - k - 1; j >= k; j--) {
      if (i - 1 < k && j + 1 >= m + 1 - k) {
        for (int dx = -k + 1; dx < k; dx++)
          for (int dy = 0; dy < k; dy++)
            if (abs(dx) + abs(dy) <= k - 1) ps2[i][j][1] += a[i + dx][j + dy];
      } else if (i - 1 >= k)
        ps2[i][j][1] = ps2[i - 1][j][1] - ps[i - 1][j][1] + ps[i][j][2];
      else {
        ps2[i][j][1] =
            ps2[i][j + 1][1] - ps[i][j + 1][1] - ps[i][j + 1][2] + a[i][j + k];
        for (int dx = -k + 1; dx < k; dx++) ps2[i][j][1] += a[i + dx][j];
      }
    }
  for (int i = n + 2 - k - 1; i >= k; i--)
    for (int j = k; j < m + 2 - k; j++) {
      if (i + 1 >= n + 1 - k && j - 1 < k) {
        for (int dx = 0; dx < k; dx++)
          for (int dy = -k + 1; dy < k; dy++)
            if (abs(dx) + abs(dy) <= k - 1) ps2[i][j][2] += a[i + dx][j + dy];
      } else if (j - 1 >= k)
        ps2[i][j][2] = ps2[i][j - 1][2] - ps[i][j - 1][3] + ps[i][j][2];
      else {
        ps2[i][j][2] =
            ps2[i + 1][j][2] - ps[i + 1][j][2] - ps[i + 1][j][3] + a[i + k][j];
        for (int dy = -k + 1; dy < k; dy++) ps2[i][j][2] += a[i][j + dy];
      }
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 < k && j - 1 < k) {
        for (int dx = -k + 1; dx < k; dx++)
          for (int dy = -k; dy < 1; dy++)
            if (abs(dx) + abs(dy) <= k - 1) ps2[i][j][3] += a[i + dx][j + dy];
      } else if (i - 1 >= k)
        ps2[i][j][3] = ps2[i - 1][j][3] - ps[i - 1][j][0] + ps[i][j][3];
      else {
        ps2[i][j][3] =
            ps2[i][j - 1][3] - ps[i][j - 1][0] - ps[i][j - 1][3] + a[i][j - k];
        for (int dx = -k + 1; dx < k; dx++) ps2[i][j][3] += a[i + dx][j];
      }
    }
  for (int i = k; i < n + 2 - k; i++)
    for (int j = k; j < m + 2 - k; j++) {
      if (i - 1 < k && j - 1 < k) {
        for (int dx = -k + 1; dx < k; dx++)
          for (int dy = -k; dy < k + 1; dy++)
            if (abs(dx) + abs(dy) <= k - 1)
              ret[i][j] += a[i + dx][j + dy] * (k - abs(dx) - abs(dy));
      } else if (i - 1 >= k)
        ret[i][j] = ret[i - 1][j] - ps2[i - 1][j][0] + ps2[i][j][2];
      else
        ret[i][j] = ret[i][j - 1] - ps2[i][j - 1][3] + ps2[i][j][1];
      if (ret[i][j] >= ans) ans = ret[i][j], px = i, py = j;
    }
  printf("%d %d\n", px, py);
}
