for t in range(int(input())):
	(n, k) = map(int, input().split())
	h = input().split()
	for i in range(len(h)):
		h[i] = int(h[i])
	final = 0

	def roll(h):
		for i in range(k):
			for j in range(n):
				if j == n - 1:
					return -1
				if h[j] < h[j + 1]:
					h[j] += 1
					break
			final = j
		return final + 1
	print(roll(h))
