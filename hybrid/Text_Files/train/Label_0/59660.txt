import java.util.*;
public class D {
	Scanner sc = new Scanner(System.in);

	private void mkanswer(long crt, long k, long[] a)
	{
			long cnt = 0;
			int n = a.length;
			for(int i = 0; i < n; i ++) {
				cnt += Math.min(a[i], crt);
				a[i]-=crt; if(a[i] < 0) a[i] = 0;
			}
			long rem = k - cnt;
			//System.out.println("Answer = "+ low + " " + high + " " + rem) ;
			int p = 0;
			while(rem > 0) {
				if(a[p] > 0) { rem--; a[p]--; } 
				p++;
			}
			for(int i = p; i < a.length; i++) 
				if(a[i] > 0) System.out.print((i+1) + " ");
			for(int i = 0; i < p; i++) 
				if(a[i] > 0) System.out.print((i+1) + " ");
			System.out.println();
			return;

	}
	private void doIt()
	{
		int n = sc.nextInt();
		long k = sc.nextLong();
		long [] a = new long [n];
		long max_v = 0;
		for(int i = 0; i < n; i++) {
			a[i] = sc.nextInt();
			max_v = Math.max(max_v, a[i]);
		}
		long low = 0, high = max_v + 1;
		while(low < high) {
			long mid = (low + high + 1) / 2;
			long cnt = 0;
			for(int i = 0; i < n; i ++) cnt += Math.min(a[i], mid);
			//System.out.println(low + " " + high + " " + cnt) ;
			if(cnt == k) {
				mkanswer(mid, k, a);
				return;
			}
			else if(cnt > k) {
				high = mid;
				if(low + 1 == high) {
					mkanswer(low, k, a);
					return;
				}
			}
			else low = mid;
		}
		System.out.println(-1);
	}
	public static void main(String[] args) {
		new D().doIt();
	}

}
