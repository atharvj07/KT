#include <bits/stdc++.h>
using namespace std;
long long int gcd(long long int m, long long int n) {
  if (n == 0) return m;
  return gcd(n, m % n);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  if (n == 1) {
    if (a[0] == 0)
      cout << "YES" << endl << "0\n";
    else
      cout << "NO\n";
    return 0;
  } else if (n == 2) {
    if (a[0] == 1 && a[1] == 0)
      cout << "YES\n"
           << "(" << a[0] << "->" << a[1] << ")" << endl;
    else
      cout << "NO" << endl;
    return 0;
  } else if (a[n - 1] == 1) {
    cout << "NO" << endl;
    return 0;
  } else if (a[n - 2] == 1) {
    cout << "YES" << endl;
    for (int i = 0; i < n - 1; i++) {
      cout << a[i] << "->";
    }
    cout << a[n - 1] << endl;
    return 0;
  } else if (a[n - 2] == 0) {
    int c = -1;
    for (int i = 0; i < n - 2; i++) {
      if (a[i] == 0) {
        c = i;
        break;
      }
    }
    if (c < 0) {
      cout << "NO" << endl;
      return 0;
    } else {
      cout << "YES" << endl;
      for (int i = 0; i < c; i++) {
        cout << a[i] << "->";
      }
      cout << "(0->(";
      for (int i = c + 1; i < n - 2; i++) {
        cout << a[i] << "->";
      }
      cout << "0))->0";
      cout << endl;
      return 0;
    }
  }
  return 0;
}
