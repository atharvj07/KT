def prob(n, k):
	pt = (k * (k - 2) * (k - 1) + 3 * (k - 1) * k * (n - k) + 3 / 2 * k * (n - k) * (n - k - 1)) / (n * (n - 2) * (n - 1))
	return pt
np = input()
np = np.split(' ')
n = int(np[0])
p = float(np[1])
left = 0
right = n
while left < right:
	k = (left + right) // 2
	if prob(n, k) < p:
		left = k + 1
	else:
		right = k
while prob(n, k) < p:
	k += 1
print(k)
