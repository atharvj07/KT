#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int k;
    string s;
    cin >> k >> s;
    s += 'A';
    vector<int> a;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == 'A') a.push_back(i);
    }
    int ans = 0;
    for (int i = 0; i + 1 < a.size(); i++) {
      ans = max(ans, a[i + 1] - a[i] - 1);
    }
    cout << ans << endl;
  }
  return 0;
}
