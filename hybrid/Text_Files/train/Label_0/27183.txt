#include <bits/stdc++.h>
using namespace std;
signed main() {
  string s;
  cin >> s;
  long long n = s.length();
  if (n == 1) {
    if (s[0] == 'a') {
      cout << 1;
      return 0;
    }
    cout << 0;
    return 0;
  }
  long long k = n / 2;
  k = k + 1;
  long long count = 0;
  for (long long i = 0; i < n; i++) {
    if (s[i] == 'a') {
      count++;
    }
  }
  if (count >= k) {
    cout << n;
    return 0;
  }
  long long l = 2 * count - 1;
  long long x = n - l;
  cout << n - x;
  return 0;
}
