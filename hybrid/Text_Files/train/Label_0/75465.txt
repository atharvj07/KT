from sys import stdin
(t, sx, sy, ex, ey) = [int(i) for i in stdin.readline().split(' ')]
wind = stdin.readline().strip()
direction = {'E': 0, 'W': 0, 'N': 0, 'S': 0}
if ex >= sx:
	direction['E'] = ex - sx
else:
	direction['W'] = sx - ex
if ey >= sy:
	direction['N'] = ey - sy
else:
	direction['S'] = sy - ey
count = 0
flag = 0
for i in wind:
	if direction[i] > 0:
		direction[i] -= 1
	count += 1
	temp = direction['S'] + direction['N'] + direction['W'] + direction['E']
	if temp == 0:
		flag = 1
		print(count)
		break
if flag == 0:
	print('-1')
