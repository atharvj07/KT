int ara[10000], ara2[10000];

int main(void) {
  int i, gcd, t = 0, n, count = 0;
  cin >> n;
  for (i = 0; i < n; i++) cin >> ara[i];
  ara2[0] = ara[0];
  int m = 1;
  for (i = 1; i < n; i++) {
    gcd = gcd_check(ara[i], ara[i - 1]);
    if (gcd == 1)
      ara2[m++] = ara[i];
    else
      ara2[m++] = 1, ara2[m++] = ara[i];
  }

  }

}
int gcd_check(int a, int b) {
  int n1, n2;
  if (a == b)
    return a;
  else if (a > b)
    n1 = a, n2 = b;
  else
    n1 = b, n2 = a;
  int mod = 1;
  while (mod) {
    mod = n1 % n2;
    if (mod) n1 = n2, n2 = mod;
  }
  return n2;
}
