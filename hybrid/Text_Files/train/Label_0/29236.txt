#include <bits/stdc++.h>
#define INF (1<<30)
#define MOD 1000000007
#define l_ength size
using ll = long long;
using namespace std;

int main(){
	ll n, p;
	cin >> n >> p;
	if( n == 1 ){
		cout << p << endl;
		return 0;
	}
	ll ans = 1;
	for( ll i = sqrt(p); i > 1; --i ){
		int cnt = 0;
		while( p % i == 0 ){
			p /= i;
			++cnt;
		}
		if( cnt >= n ){
			ans *= i;
		} else {
			for( int j = 0; j < cnt; ++j ){
				p *= i;
			}
		}
	}
	cout << ans << endl;
	return 0;
}
