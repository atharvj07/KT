#include <bits/stdc++.h>
using namespace std;
void _() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
}
int main() {
  _();
  int n;
  cin >> n;
  long long last = 10000000000, x;
  vector<pair<long long, int> > v;
  for (int i = 0; i < n; i++) {
    cin >> x;
    if (last == x)
      v[v.size() - 1].second++;
    else
      v.push_back({x, 1});
    last = x;
  }
  long long ans = 0;
  for (int i = 0; i < v.size(); i++) {
    for (int j = 1; j <= v[i].second; j++) {
      x = v[i].second - j;
      ans += x + 1;
    }
  }
  cout << ans;
}
