n = input()
(a, b, u, B, l, s, r) = (0, 0, 0, 0, 0, 0, 0)
for i in range(len(n)):
	if n[i] == 'B':
		B += 2
	if n[i] == 'a':
		a += 1
	if n[i] == 'u':
		u += 1
	if n[i] == 'b':
		b += 2
	if n[i] == 'l':
		l += 2
	if n[i] == 's':
		s += 2
	if n[i] == 'r':
		r += 2
print(min(a, b, u, B, l, s, r) // 2)
