def get_int():
	return int(stdin.readline().strip())

def get_ints():
	return map(int, stdin.readline().strip().split())

def get_array():
	return list(map(int, stdin.readline().strip().split()))

def get_string():
	return stdin.readline().strip()
INF = 10000000000.0
mod = 1000000000.0 + 7
for _ in range(int(stdin.readline())):
	n = get_int()
	a = get_array()
	vis = []
	max_d = -INF
	idx = -1
	f = 0
	for i in range(n - 1):
		f = 0
		for j in range(n - 1):
			if a[j] < a[j + 1]:
				continue
			if a[j] - a[j + 1] > max_d:
				if j not in vis:
					f = 1
					idx = j
		if f == 1:
			(a[idx], a[idx + 1]) = (a[idx + 1], a[idx])
			vis.append(idx)
	for i in a:
		print(i, end=' ')
