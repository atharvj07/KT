import math
import sys
from collections import Counter

def solve():
	(n, k) = [int(x) for x in input().split()]
	(a, b, c, d) = [int(x) for x in input().split()]
	if k <= n or n < 5:
		print(-1)
		return None
	S = list(range(1, n + 1))
	S.remove(a)
	S.remove(b)
	S.remove(c)
	S.remove(d)
	A = [a, c] + S + [d, b]
	B = [c, a] + S + [b, d]
	print(' '.join(map(str, A)))
	print(' '.join(map(str, B)))
solve()
