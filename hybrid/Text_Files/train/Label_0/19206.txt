#include <bits/stdc++.h>
using namespace std;
int n, m;
char F[1010][1010];
int ID[1010][1010], cv = 1;
int br[1000010];
bool is_in(int x, int y) { return (x >= 0 && x < n && y >= 0 && y < m); }
int used[1010][1010];
set<char> C;
void dfs(int x, int y) {
  used[x][y] = 1;
  if (is_in(x - 1, y) && F[x - 1][y] != '#') {
    C.erase(F[x - 1][y]);
    if (!used[x - 1][y] && ID[x][y] == ID[x - 1][y]) dfs(x - 1, y);
  }
  if (is_in(x + 1, y) && F[x + 1][y] != '#') {
    C.erase(F[x + 1][y]);
    if (!used[x + 1][y] && ID[x][y] == ID[x + 1][y]) dfs(x + 1, y);
  }
  if (is_in(x, y - 1) && F[x][y - 1] != '#') {
    C.erase(F[x][y - 1]);
    if (!used[x][y - 1] && ID[x][y] == ID[x][y - 1]) dfs(x, y - 1);
  }
  if (is_in(x, y + 1) && F[x][y + 1] != '#') {
    C.erase(F[x][y + 1]);
    if (!used[x][y + 1] && ID[x][y] == ID[x][y + 1]) dfs(x, y + 1);
  }
}
void dfs1(int x, int y, char col) {
  used[x][y] = 2;
  F[x][y] = col;
  if (is_in(x - 1, y) && F[x - 1][y] != '#' && ID[x - 1][y] == ID[x][y] &&
      used[x - 1][y] != 2)
    dfs1(x - 1, y, col);
  if (is_in(x + 1, y) && F[x + 1][y] != '#' && ID[x + 1][y] == ID[x][y] &&
      used[x + 1][y] != 2)
    dfs1(x + 1, y, col);
  if (is_in(x, y - 1) && F[x][y - 1] != '#' && ID[x][y] == ID[x][y - 1] &&
      used[x][y - 1] != 2)
    dfs1(x, y - 1, col);
  if (is_in(x, y + 1) && F[x][y + 1] != '#' && ID[x][y + 1] == ID[x][y] &&
      used[x][y + 1] != 2)
    dfs1(x, y + 1, col);
}
void color_fig(int x, int y) {
  for (int i = 0; i < 10; ++i) C.insert(i + '0');
  dfs(x, y);
  char col = *C.begin();
  dfs1(x, y, col);
}
int main() {
  int i, j, k, l;
  scanf("%d%d", &n, &m);
  for (i = 0; i < n; ++i) scanf("%s", &F[i]);
  for (i = 0; i < n; ++i)
    for (j = 0; j < (m - 1); ++j)
      if (F[i][j] == '.' && F[i][j + 1] == '.' && ID[i][j] == 0 &&
          ID[i][j + 1] == 0)
        ID[i][j] = ID[i][j + 1] = cv++;
  for (j = 0; j < m; ++j)
    for (i = 0; i < (n - 1); ++i)
      if (F[i][j] == '.' && F[i + 1][j] == '.' && ID[i][j] == 0 &&
          ID[i + 1][j] == 0)
        ID[i][j] = ID[i + 1][j] = cv++;
  for (i = 0; i < n; ++i)
    for (j = 0; j < m; ++j)
      if (ID[i][j] > 0) ++br[ID[i][j]];
  for (i = 0; i < n; ++i)
    for (j = 0; j < m; ++j) {
      if (F[i][j] == '#' || ID[i][j]) continue;
      int bv = 1000;
      if (is_in(i - 1, j) && ID[i - 1][j]) bv = min(bv, br[ID[i - 1][j]]);
      if (is_in(i + 1, j) && ID[i + 1][j]) bv = min(bv, br[ID[i + 1][j]]);
      if (is_in(i, j - 1) && ID[i][j - 1]) bv = min(bv, br[ID[i][j - 1]]);
      if (is_in(i, j + 1) && ID[i][j + 1]) bv = min(bv, br[ID[i][j + 1]]);
      if (is_in(i - 1, j) && ID[i - 1][j] && bv == br[ID[i - 1][j]])
        ID[i][j] = ID[i - 1][j];
      if (is_in(i + 1, j) && ID[i + 1][j] && bv == br[ID[i + 1][j]])
        ID[i][j] = ID[i + 1][j];
      if (is_in(i, j - 1) && ID[i][j - 1] && bv == br[ID[i][j - 1]])
        ID[i][j] = ID[i][j - 1];
      if (is_in(i, j + 1) && ID[i][j + 1] && bv == br[ID[i][j + 1]])
        ID[i][j] = ID[i][j + 1];
      ++br[ID[i][j]];
    }
  bool OK = true;
  for (i = 0; i < n; ++i)
    for (j = 0; j < m; ++j)
      if (F[i][j] == '.' && !ID[i][j]) OK = false;
  if (!OK)
    printf("-1\n");
  else {
    for (i = 0; i < n; ++i)
      for (j = 0; j < m; ++j)
        if (F[i][j] == '.') color_fig(i, j);
    for (i = 0; i < n; ++i) {
      for (j = 0; j < m; ++j) printf("%c", F[i][j]);
      printf("\n");
    }
  }
  return 0;
}
