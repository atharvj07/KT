t = int(input())
for test in range(t):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	dick = {}
	c = 0
	for ele in arr:
		dick[ele] = c
		c += 1
	count = 1
	sortarr = sorted(dick)
	for x in range(n - 1):
		if dick[sortarr[x]] + 1 != dick[sortarr[x + 1]]:
			count += 1
	if count > k:
		print('No')
	else:
		print('Yes')
