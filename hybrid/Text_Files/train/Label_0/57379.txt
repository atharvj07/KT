import java.io.*;
import java.util.*;

import static java.lang.Math.*;

public class Test {
    public static void main(String[] args) throws Exception {
        Scanner in = new Scanner(System.in);
        PrintWriter pw = new PrintWriter(System.out);
        
        String string = in.next();
        char[] str = string.toCharArray();
        int n = str.length;
        int[] z = computeZ(str);
        
        int[] sum = new int[n + 1];
        for (int i : z) {
            sum[1]++;
            sum[i + 1]--;
        }

        for (int i = 1; i <= n; i++) {
            sum[i] += sum[i - 1];
        }

        int res = -1;
        for (int i = 1; i < n; i++) {
            if (i + z[i] >= n && sum[z[i]] > 1) {
                res = z[i];
                break;
            }
        }

        if (res == -1) pw.println("Just a legend");
        else pw.println(string.substring(0, res));

        pw.close();
    }

    static int[] computeZ(char[] s) {
        int n = s.length;
        int[] z = new int[n];
        for (int i = 1, l = 0, r = 0; i < n; ++i) {
            if (i <= r)
                z[i] = min(r - i + 1, z[i - l]);
            while (i + z[i] < n && s[z[i]] == s[i + z[i]])
                ++z[i];
            if (i + z[i] - 1 > r) {
                l = i;
                r = i + z[i] - 1;
            }
        }
        return z;
    }

    static void debug(Object... obj) {
        System.err.println(Arrays.deepToString(obj));
    }
}
