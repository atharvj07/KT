for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	d = {i: 0 for i in a}
	for i in a:
		if i not in d:
			d[i] = 1
		d[i] += 1
	x = int(1000000000.0)
	for (i, j) in zip(d.keys(), d.values()):
		if j == 1:
			if i < x:
				x = i
	if x == int(1000000000.0):
		print(-1)
	else:
		print(a.index(x) + 1)
