def ii():
	return int(input())

def si():
	return input()

def mi():
	return map(int, input().split())

def li():
	return list(mi())
n = ii()
a = []
for i in range(n):
	b = si()
	a.append(list(b))
for i in range(1, n - 1):
	for j in range(1, n - 1):
		if a[i][j] == '.' and a[i][j + 1] == '.' and (a[i][j - 1] == '.'):
			if a[i - 1][j] == '.' and a[i + 1][j] == '.':
				a[i - 1][j] = '#'
				a[i + 1][j] = '#'
				a[i][j] = '#'
				a[i][j - 1] = '#'
				a[i][j + 1] = '#'
f = 0
for i in range(n):
	for j in range(n):
		if a[i][j] == '.':
			f = 1
			break
	if f == 1:
		break
if f == 1:
	print('NO')
else:
	print('YES')
