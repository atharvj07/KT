#include <bits/stdc++.h>
#define FOR(i,n) for(int i=0;i<(int)(n);i++)
#define FORR(i,m,n) for(int i=(int)(m);i<(int)(n);i++)
#define pb(a) push_back(a)
#define mp(x,y) make_pair(x,y)
#define ALL(a) a.begin(),a.end()
#define ZERO(a) memset(a,0,sizeof(a))
#define len(a) sizeof(a)
#define ll long long
#define pii pair<int,int>
#define INF 1<<29
#define MAX 
using namespace std;
 
int dp[2020][11];
 
void solve(){
 int n,K,c,g;
 cin>>n>>K;
 vector<int> book[10],pr[10];
 ZERO(dp);
 FOR(i,n){
  cin>>c>>g;
  g--;
  book[g].pb(c);
 }
 FOR(i,10) sort(ALL(book[i]),greater<int>());
 FOR(i,10){
  int tot=0;
  pr[i].pb(0);
  FOR(j,book[i].size()){
   tot+=book[i][j];
   pr[i].pb(tot+j*(j+1));
  }
 }
 for(int i=K;i>=0;i--){
  for(int j=9;j>=0;j--){
   FOR(k,pr[j].size()){
    if(i+k<=K) dp[i][j]=max(dp[i][j],dp[i+k][j+1]+pr[j][k]);
    else break;
   }
  }
 }
 cout<<dp[0][0]<<endl;
}
 
int main(){
 solve();
 return 0;
}
