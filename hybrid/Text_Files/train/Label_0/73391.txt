t = int(input())
for _ in range(t):
	n = int(input())
	q = list(map(int, input().split()))
	s = list((int(i) for i in list(input())))
	arr1 = []
	arr2 = []
	for i in range(n):
		if s[i] == 0:
			arr1.append(q[i])
		else:
			arr2.append(q[i])
	arr1.sort()
	arr2.sort()
	h = {}
	c = 1
	for i in arr1:
		h[i] = c
		c += 1
	for i in arr2:
		h[i] = c
		c += 1
	for i in q:
		print(h[i], end=' ')
	print()
