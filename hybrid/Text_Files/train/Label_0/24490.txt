using namespace std;
vector<pair<long long, long long> > dir = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};
bool valid(long long x, long long y, long long n, long long m) {
  return x >= 0 && y >= 0 && x < n && y < m;
}
mt19937 rng(1999999973);
const long long N = 100000 + 50;
long long a[N], pref[N], suf[N], n, k, ans, sum = 0;
inline long long getPosBiggerOrEqual(long long x) {
  if (x > a[n]) return n + 1;
  long long l = 1, r = n;
  while (l != r) {
    long long mid = (l + r) / 2;
    if (a[mid] >= x)
      r = mid;
    else
      l = mid + 1;
  }
  return l;
}
inline long long getPosLessOrEqual(long long x) {
  if (x < a[1]) return 0;
  long long l = 1, r = n;
  while (l != r) {
    long long mid = (l + r + 1) / 2;
    if (a[mid] <= x)
      l = mid;
    else
      r = mid - 1;
  }
  return l;
}
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> n >> k;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  if (k >= sum) {
    cout << 0;
    return 0;
  }
  sort(a + 1, a + n + 1);
  for (long long i = 1; i <= n; i++) {
    pref[i] = pref[i - 1] + a[i];
  }
  for (long long i = n; i >= 0; i--) {
    suf[i] = suf[i + 1] + a[i];
  }
  ans = a[n] - a[1];
  for (long long i = 1; i <= n; i++) {
    long long oper = a[i] * i - pref[i];
    if (oper > k) break;
    oper = k - oper;
    long long l = 0, r = 1e9 + 5;
    while (l != r) {
      long long mid = (l + r) / 2;
      long long pos = getPosBiggerOrEqual(a[i] + mid);
      if (suf[pos] - (a[i] + mid) * (n - pos + 1) <= oper)
        r = mid;
      else
        l = mid + 1;
    }
    ans = min(ans, l);
  }
  for (long long i = n; i >= 1; i--) {
    long long oper = suf[i] - a[i] * (n - i + 1);
    if (oper > k) break;
    oper = k - oper;
    long long l = 0, r = 1e9 + 5;
    while (l != r) {
      long long mid = (l + r) / 2;
      long long pos = getPosLessOrEqual(a[i] - mid);
      if ((a[i] - mid) * pos - pref[pos] <= oper)
        r = mid;
      else
        l = mid + 1;
    }
    ans = min(ans, l);
  }
  cout << ans;
  return 0;
}
