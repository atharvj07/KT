// upsolve with rainboy
import java.io.*;
import java.util.*;

public class CF1100F {
	static final int B = 20;
	static int[] aa, aa_ = new int[B];
	static boolean add(int i) {
		int a = aa[i];
		for (int b = B - 1; b >= 0; b--)
			if ((a & 1 << b) != 0)
				a ^= aa_[b];
		if (a != 0) {
			int b = B - 1;
			while ((a & 1 << b) == 0)
				b--;
			aa_[b] = a;
			return true;
		}
		return false;
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw = new PrintWriter(System.out);
		int n = Integer.parseInt(br.readLine());
		StringTokenizer st = new StringTokenizer(br.readLine());
		aa = new int[n];
		for (int i = 0; i < n; i++)
			aa[i] = Integer.parseInt(st.nextToken());
		int q = Integer.parseInt(br.readLine());
		int[] ll = new int[q];
		int[] rr = new int[q];
		Integer[] pp = new Integer[q];
		for (int p = 0; p < q; p++) {
			st = new StringTokenizer(br.readLine());
			ll[p] = Integer.parseInt(st.nextToken()) - 1;
			rr[p] = Integer.parseInt(st.nextToken()) - 1;
			pp[p] = p;
		}
		Arrays.sort(pp, (i, j) -> { return rr[i] - rr[j]; });
		int[] ans = new int[q];
		int[] ii = new int[B], jj = new int[B];
		int k = 0;
		for (int p = 0, i = 0; p < q; ) {
			int r = rr[pp[p]];
			Arrays.fill(aa_, 0);
			int k_ = 0;
			for (int j = r; j >= i; j--)
				if (add(j))
					jj[k_++] = j;
			while (k-- > 0) {
				int j = ii[k];
				if (add(j))
					jj[k_++] = j;
			}
			k = k_;
			for (int h = 0; h < k; h++)
				ii[h] = jj[k - 1 - h];
			i = r + 1;
			while (p < q && rr[pp[p]] == r) {
				int l = ll[pp[p]];
				Arrays.fill(aa_, 0);
				for (int h = 0; h < k; h++)
					if (l <= ii[h])
						add(ii[h]);
				int a = 0;
				for (int b = B - 1; b >= 0; b--)
					a = Math.max(a, a ^ aa_[b]);
				ans[pp[p++]] = a;
			}
		}
		for (int p = 0; p < q; p++)
			pw.println(ans[p]);
		pw.close();
	}
}
