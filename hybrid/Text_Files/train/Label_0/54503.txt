import java.io.*;
import java.util.*;

public class C {
	FastScanner in;
	PrintWriter out;

	class Point {
		long x, y;

		public Point(int x, int y) {
			super();
			this.x = x;
			this.y = y;
		}
	}

	long sqr(long x) {
		return x * x;
	}

	long[] minl, minr, maxl, maxr;
	Point[] pts;

	boolean fit(int l, int r, long mm) {
		long miny = 0;
		long maxy = 0;
		if (l != 0) {
			maxy = Math.max(maxy, maxl[l - 1]);
			miny = Math.min(miny, minl[l - 1]);
		}
		if (r != maxr.length - 1) {
			maxy = Math.max(maxy, maxr[r + 1]);
			miny = Math.min(miny, minr[r + 1]);
		}
		long mx = Math.max(Math.abs(pts[l].x), Math.abs(pts[r].x));
		if ((maxy - miny) * (maxy - miny) <= mm && Math.max(maxy * maxy, miny * miny) + mx * mx <= mm) {
			return true;
		}
		return false;
	}

	void solve() {
		int n = in.nextInt();
		pts = new Point[n];
		long minx = Integer.MAX_VALUE, miny = Integer.MAX_VALUE, maxx = Integer.MIN_VALUE, maxy = Integer.MIN_VALUE;
		for (int i = 0; i < n; i++) {
			pts[i] = new Point(in.nextInt(), in.nextInt());
			minx = Math.min(pts[i].x, minx);
			miny = Math.min(pts[i].y, miny);
			maxx = Math.max(pts[i].x, maxx);
			maxy = Math.max(pts[i].y, maxy);
		}
		long ans = Math.min(maxx - minx, maxy - miny);
		ans *= ans;
		minl = new long[n];
		minr = new long[n];
		maxl = new long[n];
		maxr = new long[n];
		Arrays.sort(pts, new Comparator<Point>() {
			@Override
			public int compare(Point o1, Point o2) {
				return Long.compare(o1.x, o2.x);
			}
		});
		for (int i = 0; i < n; i++) {
			minl[i] = maxl[i] = pts[i].y;
			if (i > 0) {
				minl[i] = Math.min(minl[i], minl[i - 1]);
				maxl[i] = Math.max(maxl[i], maxl[i - 1]);
			}
		}
		for (int i = n - 1; i >= 0; i--) {
			minr[i] = maxr[i] = pts[i].y;
			if (i < n - 1) {
				minr[i] = Math.min(minr[i], minr[i + 1]);
				maxr[i] = Math.max(maxr[i], maxr[i + 1]);
			}
		}
		long ml = -1, mr = ans;
		while (mr - ml > 1) {
			long mm = (ml + mr) / 2;
			boolean ok = false;
			int l = 0, r = 0;
			for (l = 0; l < n && pts[l].x <= 0; l++) {
				int tl = l, tr = n;
				while (tr - tl > 1) {
					int tm = (tl + tr) / 2;
					if (Math.abs(pts[tm].x) <= Math.abs(pts[l].x) && sqr(pts[tm].x - pts[l].x) <= mm) {
						tl = tm;
					} else {
						tr = tm;
					}
				}
				r = tl + 1;
				if (l != r && fit(l, r - 1, mm)) {
					ok = true;
					break;
				}
			}

			if (!ok) {
				l = n - 1;
				for (r = n - 1; r >= 0 && pts[r].x > 0; r--) {
					int tl = -1, tr = r;
					while (tr - tl > 1) {
						int tm = (tl + tr) / 2;
						if (Math.abs(pts[tm].x) <= Math.abs(pts[r].x) && sqr(pts[tm].x - pts[r].x) <= mm) {
							tr = tm;
						} else {
							tl = tm;
						}
					}
					l = tl;
					if (l != r && fit(l + 1, r, mm)) {
						ok = true;
						break;
					}
				}
			}
			if (ok) {
				mr = mm;
			} else {
				ml = mm;
			}
		}
		ans = mr;
		out.println(ans);
	}

	void run() {
		in = new FastScanner();
		out = new PrintWriter(System.out);

		solve();

		out.close();
	}

	class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreTokens()) {
				try {
					st = new StringTokenizer(in.br.readLine());
				} catch (IOException e) {
					// TODO Auto-generated catch block
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}
	}

	public static void main(String[] args) {
		new C().run();
	}
}
