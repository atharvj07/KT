def CF630O():
	(px, py, vx, vy, a, b, c, d) = list(map(int, input().split()))
	l = (vx ** 2 + vy ** 2) ** 0.5
	vx /= l
	vy /= l
	print(px + vx * b, py + vy * b)
	print(px - vy * a / 2, py + vx * a / 2)
	print(px - vy * c / 2, py + vx * c / 2)
	print(px - vy * (c / 2) - vx * d, py + vx * (c / 2) - vy * d)
	print(px + vy * (c / 2) - vx * d, py - vx * (c / 2) - vy * d)
	print(px + vy * c / 2, py - vx * c / 2)
	print(px + vy * a / 2, py - vx * a / 2)
CF630O()
