def dist(s1, s2):
	d = 0
	for i in range(len(s1)):
		if s1[i] != s2[i]:
			d += 1
	return d

def build_str(s1, s2, d):
	s = list(s1)
	for i in range(len(s2)):
		if s2[i] != s1[i] and d > 0:
			s[i] = s2[i]
			d -= 1
			if d == 0:
				break
	return ''.join(s)
s1 = input()
s2 = input()
d = dist(s1, s2)
if d % 2 != 0:
	print('impossible')
else:
	print(build_str(s1, s2, d / 2))
