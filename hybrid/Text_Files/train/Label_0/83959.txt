n = int(input())
(minx, miny, maxx, maxy) = (1000000000, 10000000000, -10000000000, -1000000000)
for i in range(n):
	(x, y) = map(int, input().split())
	minx = min(minx, x)
	miny = min(miny, y)
	maxx = max(maxx, x)
	maxy = max(maxy, y)
sidex = maxx - minx
sidey = maxy - miny
ans = max(sidex, sidey)
print(ans * ans)
