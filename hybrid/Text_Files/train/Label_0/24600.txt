#include <bits/stdc++.h>
using namespace std;
vector<int> adj[2000006];
int mark[2000006], self[2000006];
void dfs(int u) {
  mark[u] = 1;
  for (int i = 0; i < adj[u].size(); i++) {
    if (mark[adj[u][i]] == 0) dfs(adj[u][i]);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  long long n, m, len, ans = 0, u = 1, v = 1, flag = 0, cnt = 0;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    cin >> u >> v;
    if (u == v) {
      cnt++;
      self[u]++;
      continue;
    }
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(u);
  for (int i = 1; i <= n; i++) {
    if ((!mark[i]) && (adj[i].size() > 0 || self[i] > 0)) flag = 1;
  }
  for (int i = 1; i <= n; i++) {
    len = adj[i].size();
    ans += (len * (len - 1)) / 2;
  }
  ans += (m - cnt) * cnt + (cnt * (cnt - 1)) / 2;
  if (flag == 1)
    cout << "0\n";
  else
    cout << ans << endl;
}
