t = int(input())
results = [None] * t
for l in range(t):
	n = int(input())
	caves = [[int(x) for x in input().split()][1:] for _ in range(n)]
	min_levels = [(max((x - i + 1 for (i, x) in enumerate(cave))), len(cave)) for cave in caves]
	min_levels.sort(key=lambda x: x[0])
	sub = 0
	for (i, x) in enumerate(min_levels):
		min_levels[i] = x[0] - sub
		sub += x[1]
	results[l] = max((x for x in min_levels))
[print(x) for x in results]
