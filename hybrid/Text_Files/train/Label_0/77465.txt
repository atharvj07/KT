def gcd(a, b):
	while a and b:
		if a > b:
			a %= b
		else:
			b %= a
	return a + b

def check(n):
	s = str(n)
	nok = 1
	for elem in s:
		if elem != '0':
			nok = nok * int(elem) // gcd(nok, int(elem))
	return n % nok == 0
t = int(input())
for _ in range(t):
	n = int(input())
	for i in range(20):
		nn = n
		if i == 0:
			cur_s = str(n)
		elif i >= len(str(n)):
			cur_s = '0'
		else:
			cur_s = str(n)[:-i]
		st10 = 10 ** i
		nok = 1
		for elem in cur_s:
			if elem != '0':
				nok = nok * int(elem) // gcd(nok, int(elem))
		if i == 0 and n % nok == 0:
			print(n)
			break
		diff = nok - n % nok
		n += diff
		while str(n // st10) == cur_s and (not check(n)):
			n += nok
		if str(n // st10) == cur_s and check(n):
			print(n)
			break
		n = nn
