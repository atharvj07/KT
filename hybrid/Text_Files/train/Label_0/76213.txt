import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	w = list(map(int, input().split()))
	d = [0] * (n + 1)
	for (i, j) in enumerate(w):
		d[j] = i
	(a, x) = ([], n)
	for i in range(n, 0, -1):
		if d[i] < x:
			a.extend(w[d[i]:x])
			x = d[i]
	print(' '.join(map(str, a)))
