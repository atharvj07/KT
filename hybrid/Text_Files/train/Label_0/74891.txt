import sys
input = sys.stdin.readline

def main():
	T = int(input())
	for t in range(T):
		(n, m) = map(int, input().split())
		rows = []
		columns = []
		for j in range(n):
			rows.append(list(map(int, input().split())))
		for j in range(m):
			columns.append(list(map(int, input().split())))
		tops = []
		for j in range(m):
			tops.append(columns[j][0])
		tops_sort = sorted(tops)
		mat = []
		for row in rows:
			sortedrow = sorted(row)
			if sortedrow == tops_sort:
				mat.append(row)
				break
		for j in range(n - 1):
			mat.append([0] * m)
		for (i, top) in enumerate(mat[0]):
			idx = tops.index(top)
			for j in range(1, n):
				mat[j][i] = columns[idx][j]
		for i in range(n):
			print(*mat[i])
main()
