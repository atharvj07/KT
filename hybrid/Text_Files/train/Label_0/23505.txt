#include <bits/stdc++.h>
long double pi = 2 * acosl(0.0);
int test_case_number, test_case = 1;
void function_name() {
  int element_number, sum = 0;
  std::map<int, int> show_time;
  bool possible = true;
  scanf("%d", &element_number);
  for (int i = 0, start, ender; i < element_number; i++) {
    scanf("%d %d", &start, &ender);
    show_time[start]++;
    show_time[ender + 1]--;
  }
  for (auto it = show_time.begin(); it != show_time.end(); it++) {
    sum += it->second;
    if (sum > 2) {
      possible = false;
      break;
    }
  }
  if (possible) {
    printf("YES\n");
  } else {
    printf("NO\n");
  }
}
void function_name(int v1, int v2, int v3) {}
void function_name(std::string line) {}
class initial_works {
 public:
  void with_test_case_number() {
    scanf("%d", &test_case_number);
    while (test_case <= test_case_number) {
      function_name();
      test_case++;
    }
  }
  void single_iteration() { function_name(); }
  void without_test_case_number() {
    int v1, v2, v3;
    while (scanf("%d %d %d", &v1, &v2, &v3) == 3 && v1 != 0) {
      function_name(v1, v2, v3);
      test_case++;
    }
  }
  void eoof() {
    char dummy_char_array[100];
    while (scanf("%[^\n]", dummy_char_array) != EOF) {
      getchar();
      std::string line = dummy_char_array;
      function_name(line);
      test_case++;
    }
  }
};
int main() {
