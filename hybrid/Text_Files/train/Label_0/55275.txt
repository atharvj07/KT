import java.util.Scanner;

public class Main {
public static void main(String[] args) {
	Scanner sc = new Scanner(System.in);
	int q = sc.nextInt();
	while((q--)!=0) {
		int n = sc.nextInt();
		int u = sc.nextInt();
		int v = sc.nextInt();
		String s=sc.nextLine();
		String[] a = sc.nextLine().split(" ");
		int z=Integer.parseInt(a[0]);
		int w=0;
		for(int i=1;i<a.length;i++) {
			int d=Integer.parseInt(a[i]);
			w=Math.max(Math.abs(z-d), w);
			z=d;
		}
		if(w==0) {
			if(v<u)System.out.println(2*v);
			else System.out.println(u+v);
		}else if(w==1) {
			System.out.println(Math.min(v, u));
		}else {
			System.out.println("0");
		}
	}
}
}
