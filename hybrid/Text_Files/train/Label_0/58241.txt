    public static void main(String[] args) throws IOException{
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder sb = new StringBuilder();
        for(int q = pint(in); q > 0; q--) {
        	StringTokenizer st = new StringTokenizer(in.readLine());
        	long a = pint(st), b = pint(st), c = pint(st), d = pint(st);
        	if(a > b * c) {
        		sb.append(-1);
        	}else {
        		double bp = b * d;
        		double km = (a + .5 * bp) / bp;
        		long k1 =  (long) Math.floor(km), k2 = (long) Math.ceil(km);
        		long a1 = k1 * a - (k1 * (k1 - 1)) * b * d / 2;
        		long a2 = k2 * a - (k2 * (k2 - 1)) * b * d / 2;
        		sb.append(Math.max(a1, a2));
        	}
        	sb.append('\n');
        }
        System.out.println(sb);
        
    }
    static int[] resize(int[] a) {
    	int[] r = new int[a.length * 2];

    	}

    }
    static int gcd(int a, int b) {
    	if(a == 0) {return b;}
    	return gcd(b % a, a);
    }
