#include<bits/stdc++.h>
#define reg register
typedef long long ll;
using namespace std;
const int MN=5005;
const int mod=1e9+7;
int to[MN<<1],nxt[MN<<1],h[MN],cnt;
inline void ins(int s,int t){
	to[++cnt]=t;nxt[cnt]=h[s];h[s]=cnt;
	to[++cnt]=s;nxt[cnt]=h[t];h[t]=cnt;
}
int n,f[MN][MN],g[MN],F[MN],siz[MN];
void dfs(int st,int fa=0){
	siz[st]=1;f[st][1]=1;
	for(reg int i=h[st];i;i=nxt[i])
		if(to[i]!=fa){
			dfs(to[i],st);
			for(reg int j=1;j<=siz[st];j++)F[j]=f[st][j],f[st][j]=0;
			for(reg int j=1;j<=siz[st];j++)
				for(reg int k=0;k<=siz[to[i]];k++)
					f[st][j+k]=(f[st][j+k]+1ll*F[j]*f[to[i]][k])%mod;
			siz[st]+=siz[to[i]];
		}
	for(reg int i=1;i<=siz[st];i++)
		f[st][0]=(f[st][0]-1ll*f[st][i]*g[i])%mod;
	f[st][0]=(f[st][0]+mod)%mod;
}
int main(){
	scanf("%d",&n);g[0]=1;
	for(reg int i=1,s,t;i<n;i++)
		scanf("%d%d",&s,&t),ins(s,t);
	for(reg int i=2;i<=n;i+=2)g[i]=1ll*g[i-2]*(i-1)%mod;
	dfs(1);printf("%d\n",mod-f[1][0]);
	return 0;
}
