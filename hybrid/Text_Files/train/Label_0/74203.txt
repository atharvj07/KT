def Sort(sub_li):
	sub_li.sort(key=lambda x: x[0])
	return sub_li
o = input().rstrip().split(' ')
p = input().rstrip().split(' ')
l = []
q = []
w = []
S = []
D = []
for i in range(0, int(o[1])):
	g = input().rstrip().split(' ')
	D = []
	D.append(int(g[0]))
	D.append(int(g[1]))
	l.append(D)
l = Sort(l)
for i in range(0, len(l)):
	if i == 0:
		q.append(l[i][1])
	else:
		q.append(l[i][1] + q[len(q) - 1])
for i in range(0, len(p)):
	t = int(p[i])
	I = 0
	J = len(l) - 1
	V = -1
	while I <= J:
		m = (I + J) // 2
		if t >= l[m][0]:
			V = m
			I = m + 1
		else:
			J = m - 1
	if V == -1:
		print(0, end=' ')
	else:
		print(q[V], end=' ')
