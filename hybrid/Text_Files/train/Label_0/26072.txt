#include<iostream>
#include<vector>
#include<utility>
#include<algorithm>
using namespace std;


const int N=25*10000+5;
vector<int> rig[N],a[N],compressed;
int n,m;

void finalchk(vector<int> &comp)
{
  for(int i=1;i<=n;i++)
  {
    int cnt=0;
    for(int j=1;j<=m;j++)  if(a[i][j] != comp[j])  cnt++;
    if(cnt>2) return;  
  }

  cout<<"Yes\n";
  for(int j=1;j<=m;j++)  cout<<compressed[comp[j]-1]<<" ";
  cout<<"\n";
  exit(0);
}

int cnt[N],temp[N];
bool visited[N],ssh[N];


void chk(int idx1,int idx2)
{ 
   vector<int>v;
   vector< pair<int,int> >vec;
   for(int i=2;i<=n;i++)
   {
      v.clear();
      for(int j=1;j<=m;j++)  if(j!=idx1 && j!=idx2 && a[i][j]!=a[1][j] )  v.push_back(j);
      if( (int)v.size() > 2 )  return;
      else if( (int)v.size() == 2 )
      {
         vector<int> comp(m+1);
         for(int j=1;j<=m;j++) comp[j]=a[1][j];
         comp[idx1]=a[i][idx1]; 
         comp[idx2]=a[i][idx2]; 
         finalchk(comp);
         return;
      }
      else if( (int)v.size() == 1 )  
      {
        ssh[i]=1;
        vec.emplace_back(a[i][idx1],a[i][idx2]);
      }
   }

   if(vec.empty())
   {
      cout<<"Yes\n";
      for(int j=1;j<=m;j++)  cout<<compressed[a[1][j]-1]<<" ";
      cout<<"\n";
      exit(0);  
   }
   
   vector<int>who,v2;
   int totalDiff=0;
   for(int i=2;i<=n;i++) 
   {
      if(ssh[i])
      {
        rig[a[i][idx1]].push_back(a[i][idx2]); 
        cnt[a[i][idx2]]++;
        if(cnt[a[i][idx2]]==1) totalDiff++;
        who.push_back(a[i][idx1]);
        v2.push_back(a[i][idx2]);
      }
   }
   
   for(auto &i: who)
   {
      if(!visited[i])
      {
        visited[i]=1;

        int diff=0;
        for(auto &j: rig[i]) 
        {
          temp[j]++;
          if(temp[j]==1)  diff++;
        }

        if(diff+1>=totalDiff)
        {
           bool ok=1;
           for(auto &j : rig[i])  if(temp[j]!=cnt[j] && temp[j]+n-1-(int)rig[i].size()!=cnt[j] )  ok=0;
           if(ok)
           {
              int other=-1;
              for(auto &j: v2)  if(temp[j]!=cnt[j])  other=j;
              if(other==-1)  other=1;
              cout<<"Yes\n";
              for(int j=1;j<=m;j++)
              {
                if(j==idx1)  cout<<compressed[i-1]<<" ";
                else if(j==idx2)  cout<<compressed[other-1]<<" ";
                else cout<<compressed[a[1][j]-1]<<" ";
              }  
              cout<<"\n";
              exit(0);
           } 
        }
        
        for(auto &j: rig[i]) temp[j]--;
      }
   }


   for(int i=2;i<=n;i++) 
   {
      if(ssh[i])
      {
        rig[a[i][idx1]].pop_back(); 
        cnt[a[i][idx2]]--;
      }
   }

   for(auto &i: who)   
   {
      visited[i]=ssh[i]=0;
   }
}

int main()
{
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  
  cin>>n>>m;
  for(int i=1;i<=n;i++)  a[i].resize(m+2);

  for(int i=1;i<=n;i++)  
  {
    for(int j=1;j<=m;j++)
    {
      cin>>a[i][j];
      compressed.push_back(a[i][j]);
    }
  }  

  if(m<=2)
  {
    cout<<"Yes\n";
    for(int j=1;j<=m;j++)  cout<<a[1][j]<<" ";
    cout<<"\n";
    return 0;  
  }

  sort(compressed.begin(),compressed.end());
  compressed.resize(unique(compressed.begin(), compressed.end()) - compressed.begin());

  for(int i=1;i<=n;i++)
  {
    for(int j=1;j<=m;j++)
    {
      a[i][j]=lower_bound(compressed.begin(),compressed.end(),a[i][j])-compressed.begin();
      a[i][j]++;
    }
  }
  
  for(int i=2;i<=n;i++)
  {
    vector<int>v;
    for(int j=1;j<=m;j++)  if(a[i][j]!=a[1][j])  v.push_back(j);
    if((int)v.size()>4)
    {
      cout<<"No\n";
      return 0;
    }
    else if((int)v.size()==4)
    {
       for(int j=0;j<4;j++) for(int z=0;z<4;z++)  if(j!=z) chk(v[j],v[z]);
       cout<<"No\n";
       return 0;
    }
    else if((int)v.size()==3)
    {
      for(int j=0;j<3;j++) for(int z=0;z<3;z++) if(j!=z) chk(v[j],v[z]);
      cout<<"No\n";
      return 0;
    }
  }

  cout<<"Yes\n";
  for(int j=1;j<=m;j++)  cout<<compressed[a[1][j]-1]<<" ";
  cout<<"\n";  

}
