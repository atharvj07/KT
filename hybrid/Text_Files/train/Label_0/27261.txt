const double PI = acos(-1.0);

const int MOD = (int)1e9 + 7;
const int maxn = (int)2e5 + 5;
const int LOGN = 20;
int main() {
  int n, cnt, ans = maxn;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n - 3; i++) {
    int cnt = 0, f = 0;
    if (s[i] != 'A') {
      cnt = abs(('A' - s[i]));
      cnt = min(cnt, (26 - cnt));
    }
    if (s[i + 1] != 'C') {
      int cnt1 = abs(('C' - s[i + 1]));
      cnt1 = min(cnt1, (26 - cnt1));
      cnt += cnt1;
    }
    if (s[i + 2] != 'T') {
      int cnt1 = abs(('T' - s[i + 2]));
      cnt1 = min(cnt1, (26 - cnt1));
      cnt += cnt1;
    }
    if (s[i + 3] != 'G') {
      int cnt1 = abs(('G' - s[i + 3]));
      cnt1 = min(cnt1, (26 - cnt1));
      cnt += cnt1;
    }
    ans = min(ans, cnt);
  }
  cout << ans << endl;
  return 0;
}
