import java.io.*;
import java.util.*;

public class F implements Runnable {
	public static void main (String[] args) {new Thread(null, new F(), "_cf", 1 << 28).start();}
	
	int n, k, MAX = 502, curNode = 1,oo = (int)5e8;
	int[][] trie;
	int[] count, depth, firstKid;
	boolean[] leaf;
	int[][][][] dp;
	
	public void run() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("");

		trie = new int[10][MAX];
		count = new int[MAX];
		depth = new int[MAX];
		leaf = new boolean[MAX];
		n = fs.nextInt();
		k = fs.nextInt();
		for(int i = 0; i < n; i++) {
			char[] str = fs.next().toCharArray();
			int cur = 0;
			for(char c : str) {
				int id = c-'0';
				if(trie[id][cur] == 0) {
					depth[curNode] = 1+depth[cur];
					trie[id][cur] = curNode++;
				}
				cur = trie[id][cur];
			}
			count[cur] += fs.nextInt();
		}
		firstKid = new int[curNode];
		for(int i = 0; i < curNode; i++) {
			firstKid[i] = -1;
			for(int j = 0; j < 10; j++) {
				if(trie[j][i] != 0) {
					firstKid[i] = j;
					break;
				}
			}
			if(firstKid[i] == -1) {
				leaf[i] = true;
			}
		}
		
		
		dp = new int[k+1][10][MAX][MAX];
		for(int i = 0; i < dp.length; i++)
			for(int j = 0; j < dp[i].length; j++)
				for(int k = 0; k < dp[i][j].length; k++)
					Arrays.fill(dp[i][j][k], -1);
		
		out.println(solve(k, 0, 0, 0));
		
		out.close();
	}
	
	int solve(int left, int kid, int node, int anc) {
		if(kid == 10) return 0;
		int ret = dp[left][kid][node][anc];
		if(ret != -1) return ret;

		if(leaf[node]) {
			return dp[left][kid][node][anc] = count[node]*(depth[node]-depth[anc]);
		}
		
		if(trie[kid][node] == 0)
			return dp[left][kid][node][anc] = solve(left, kid+1, node, anc);
		
		int res = oo;

		
		for(int giveDial = 0; giveDial < 2; giveDial++) {
			for(int giveK = 0; giveK <= left; giveK++) {
				if(giveK+giveDial > left) continue;
				
				int who = trie[kid][node];
				
				ret = solve(giveK, 0, who, giveDial == 1 ? who : anc);
				ret += solve(left-giveK-giveDial, kid+1, node, anc);
				
				res = Math.min(res, ret);
			}
		}
		
		if(kid == firstKid[node]) {
			res += count[node] * (depth[node] - depth[anc]);
		}
		
		return dp[left][kid][node][anc] = res;
	}
	
	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) throws FileNotFoundException {
			in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}
		
		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}
		
	}

	
}
