#include <bits/stdc++.h>
using namespace std;
int sh[11];
int main() {
  int l, z = 2e9, h = 0;
  cin >> l;
  for (int i = 0; i < 9; i++) cin >> sh[i];
  for (int i = 0; i < 9; i++) {
    if (sh[i] <= z) {
      z = min(sh[i], z);
      h = i;
    }
  }
  string s;
  z = l / sh[h];
  while (z--) {
    for (int i = 8; i >= 0; i--) {
      if ((l - sh[i]) / sh[h] == z && l - sh[i] >= 0) {
        s += (i + 1) + '0';
        l -= sh[i];
        break;
      }
    }
  }
  sort(s.begin(), s.end());
  reverse(s.begin(), s.end());
  cout << (s.size() ? s : "-1");
}
