(n, r) = list(map(int, input().split()))
lines_used = set([])
columns_used = set([])
current = n * n
lines = n
columns = n
for i in range(r):
	(l, c) = list(map(int, input().split()))
	if l in lines_used and c in columns_used:
		print(current)
	elif l not in lines_used and c not in columns_used:
		lines -= 1
		columns -= 1
		lines_used.add(l)
		columns_used.add(c)
		current -= lines + columns + 1
		print(current)
	elif l in lines_used:
		current -= lines
		columns -= 1
		columns_used.add(c)
		print(current)
	elif c in columns_used:
		current -= columns
		lines -= 1
		lines_used.add(l)
		print(current)
