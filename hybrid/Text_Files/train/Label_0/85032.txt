import sys
from collections import defaultdict
from sys import maxsize

def ini():
	return int(sys.stdin.readline())

def inp():
	return map(int, sys.stdin.readline().strip().split())

def li():
	return list(map(int, sys.stdin.readline().strip().split()))
for _ in range(ini()):
	n = ini()
	s = sys.stdin.readline()
	t = 0
	if s[0] == '1':
		t = 1
	o = []
	c = 0
	for i in range(n):
		if s[i] == '1':
			c += 1
		else:
			if c != 0:
				o += [c]
			c = 0
	if c != 0:
		o += [c]
	t = 0
	x = sum(o) - len(o)
	if s[0] == '0':
		t = 1
	o = []
	c = 0
	for i in range(n):
		if s[i] == '0':
			c += 1
		else:
			if c != 0:
				o += [c]
			c = 0
	if c != 0:
		o += [c]
	y = sum(o) - len(o)
	print(max(x, y))
