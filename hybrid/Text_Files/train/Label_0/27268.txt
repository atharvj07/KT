#include <bits/stdc++.h>
using namespace std;
template <typename T>
void chkmax(T& x, T y) {
  if (x < y) x = y;
}
template <typename T>
void chkmin(T& x, T y) {
  if (x > y) x = y;
}
inline int read() {
  int x = 0;
  char c = getchar();
  bool f = 0;
  while (c < 48) f |= c == '-', c = getchar();
  while (c > 47) x = x * 10 + (c ^ 48), c = getchar();
  return f ? -x : x;
}
const int maxn = 1e5 + 10;
int n, q;
map<pair<int, int>, int> mp;
int par[maxn], dis[maxn], sz[maxn];
int top, st[maxn];
int as = 1;
int find(int x) {
  while (x != par[x]) x = par[x];
  return x;
}
int query(int x) {
  int res = 0;
  while (x != par[x]) res ^= dis[x], x = par[x];
  return res;
}
bool unite(int x, int y) {
  int c = query(x), d = query(y);
  if ((x = find(x)) != (y = find(y))) {
    if (sz[x] < sz[y]) swap(x, y);
    par[y] = x, sz[x] += sz[y], dis[y] = c ^ d ^ 1, st[++top] = y;
    return 1;
  } else {
    return c != d;
  }
}
void revert(int k) {
  for (int i = (top), iend = (k + 1); i >= iend; --i) {
    int u = st[i];
    sz[par[u]] -= sz[u], par[u] = u, dis[u] = 0;
  }
  top = k;
}
vector<pair<int, int> > vec[maxn << 2];
void add(int k, int l, int r, int ql, int qr, pair<int, int> e) {
  if (ql <= l && r <= qr) return vec[k].push_back(e), void();
  if (ql <= ((l + r) >> 1)) add(k << 1, l, ((l + r) >> 1), ql, qr, e);
  if (qr > ((l + r) >> 1)) add(k << 1 | 1, ((l + r) >> 1) + 1, r, ql, qr, e);
}
void divide(int k, int l, int r) {
  int tp = top, TAS = as;
  for (pair<int, int> p : vec[k]) {
    as &= unite(p.first, p.second);
  }
  if (l == r)
    puts(as ? "YES" : "NO");
  else
    divide(k << 1, l, ((l + r) >> 1)),
        divide(k << 1 | 1, ((l + r) >> 1) + 1, r);
  revert(tp), as = TAS;
}
void solve() {
  cin >> n >> q;
  for (int i = (1), iend = (n); i <= iend; ++i) par[i] = i, sz[i] = 1;
  for (int i = (1), iend = (q); i <= iend; ++i) {
    int u = read(), v = read(), x = mp[{u, v}];
    if (x) {
      add(1, 1, q, x, i - 1, {u, v}), mp.erase({u, v});
    } else {
      mp[{u, v}] = i;
    }
  }
  for (auto p : mp) {
    add(1, 1, q, p.second, q, p.first);
  }
  divide(1, 1, q);
}
signed main() {
  solve();
  return 0;
}
