n = int(input())
s = input()
c = 1
k = [0 for i in range(26)]
code = ord('a')
k[0] = 1
ans = []
for i in range(n):
	letter = ord(s[i]) - code
	if k[letter] != 0:
		ans.append(k[letter])
	else:
		for j in range(letter, -1, -1):
			if k[j] != 0:
				ans.append(k[j])
				k[letter] = k[j]
				k[j] = 0
				break
			if j == 0:
				c += 1
				k[letter] = c
				ans.append(c)
print(c)
print(*ans)
