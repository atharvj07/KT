#include "bits/stdc++.h"

using namespace std;

int main(void){
	int i,n,k,a,b,p,q,r,ans=0,dp[110][3][3],ls[110];
	cin >> n >> k;
	fill(ls,ls+110,-1);
	fill(dp[0][0],dp[110][0],0);
	for(i=0; i<k; i++){
		cin >> a >> b;
		a--;
		b--;
		ls[a] = b;
	}

	for(p=0; p<3; p++){
		if((ls[0]==-1) || (ls[0]==p)){

			for(q=0; q<3; q++){
				if((ls[1]==-1) || (ls[1]==q)){
					dp[1][p][q] = 1;
				}
			}

		}
	}

	for(i=2; i<n; i++){

		for(p=0; p<3; p++){
			if((ls[i-2]==-1) || (ls[i-2]==p)){

				for(q=0; q<3; q++){
					if((ls[i-1]==-1) || (ls[i-1]==q)){

						for(r=0; r<3; r++){
							if((ls[i]==-1) || (ls[i]==r)){
								dp[i][q][r] += (p==q && q == r)?0:dp[i-1][p][q];
								dp[i][q][r] %= 10000;
							}
						}

					}
				}

			}
		}

	}
	for(p=0; p<3; p++){
		for(q=0; q<3; q++){
			ans += dp[n-1][p][q];
			ans %= 10000;
		}
	}
	cout << ans << endl;
	return 0;
}
