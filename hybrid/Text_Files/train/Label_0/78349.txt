import math
(n, m) = map(int, input().split())
event = list(map(int, input().split()))
period = list(map(int, input().split()))
gcd1 = event[0]
period2 = []
for i in range(1, n):
	gcd1 = math.gcd(gcd1, event[i])
	period2.append(event[i] - event[i - 1])
idex = -1
gcd2 = period2[0]
for i in range(1, len(period2)):
	gcd2 = math.gcd(gcd2, period2[i])
for i in range(m):
	if gcd1 % period[i] == 0 or gcd2 % period[i] == 0:
		idex = i + 1
		break
if idex != -1:
	print('YES')
	print(event[0], idex)
else:
	print('NO')
