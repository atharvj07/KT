def f(a):
	for i in range(len(a)):
		if a[i] == 'R' and i % 2 == 0:
			return 'R0'
		elif a[i] == 'R' and i % 2 == 1:
			return 'R1'
		elif a[i] == 'B' and i % 2 == 0:
			return 'B0'
		elif a[i] == 'B' and i % 2 == 1:
			return 'B1'
	return '???'
for _ in range(int(input())):
	n = int(input())
	a = input()
	o = ''
	if n == 1:
		prev = ''
	else:
		check = f(a)
		if check == 'R0':
			prev = 'B'
		elif check == 'R1':
			prev = 'R'
		elif check == 'B0':
			prev = 'R'
		elif check == 'B1':
			prev = 'B'
		else:
			prev = ''
	for x in a:
		if x == '?':
			if prev == 'R':
				o += 'B'
				prev = 'B'
			else:
				o += 'R'
				prev = 'R'
		else:
			o += x
			prev = x
	print(o)
