import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Codewords3 {

	private final int N = 100001, BAR = 500, MOD = 1000000007;
	private int[] fac = new int[N], inv = new int[N];

	private int m;

	private char[] s = new char[N];

	private int[] ans = new int[N];

	public Codewords3() {
		main();
	}

	int powMod(int a, int n) {
		int ret = 1;
		while (n > 0) {
			if (n % 2 == 1) {
				ret = (int)((long)ret * a % MOD);
			}
			a = (int)((long)a * a % MOD);
			n >>= 1;
		}
		return ret;
	}

	int binom(int n, int m) {
		//	if (m > n) {
		//		return 0;
		//	}
		return (int)((long)fac[n] * inv[m] % MOD * inv[n - m] % MOD);
	}

	void prepare(int len) {
		int base = 1;
		for (int i = 0; i < N; ++i) {
			ans[i] = i < len ? 0 : (int)((long)binom(i - 1, len - 1) * base % MOD);
			if (i >= len) {
				base = (int)((long)base * 25 % MOD);
			}
		}
		for (int i = 0; i < N; ++i) {
			if (ans[i] >= MOD) {
				ans[i] -= MOD;
			}
			if (i + 1 < N) {
				ans[i + 1] += (int)((long)ans[i] * 26 % MOD);
			}
		}
	}


	private void main() {
		fac[0] = 1;
		for (int i = 1; i < N; ++i) {
			fac[i] = (int)((long)fac[i - 1] * i % MOD);
		}
		inv[N - 1] = powMod(fac[N - 1], MOD - 2);
		assert((long)inv[N - 1] * fac[N - 1] % MOD == 1);
		for (int i = N - 2; i >= 0; --i) {
			inv[i] = (int)((long)inv[i + 1] * (i + 1) % MOD);
		}

		try {
			BufferedReader rd = new BufferedReader(new InputStreamReader(System.in));

            //Read the first line
            String line = rd.readLine();

            StringTokenizer tokenizer = new StringTokenizer(line);

            /* The number of the events in the test case. */
            int m = Integer.parseInt(tokenizer.nextToken());
            String s = rd.readLine();

            int len = s.length();
    		if (len > BAR) {
    			prepare(len);
    		}
    		while (m-- > 0) {
    			line = rd.readLine();
    			tokenizer = new StringTokenizer(line);
    			int op = Integer.parseInt(tokenizer.nextToken());
//    			scanf("%d", &op);
    			if (op == 1) {
    				s = tokenizer.nextToken();
//    				scanf("%s", s);
    				len = s.length();
//    				len = strlen(s);
    				if (len > BAR) {
    					prepare(len);
    				}
    			} else {
    				int n = Integer.parseInt(tokenizer.nextToken());
//    				scanf("%d", &n);
    				if (len > BAR) {
    					System.out.println(ans[n]);
    				} else {
    					if (n < len) {
    						System.out.println("0");
//    						printf("0\n");
    					} else {
    						int ans = powMod(26, n);
    						int base = powMod(25, n - len + 1);
    						for (int i = len - 1; i >= 0; --i) {
    							ans -= (int)((long)binom(n, i) * base % MOD);
    							if (ans < 0) {
    								ans += MOD;
    							}
    							base = (int)((long)base * 25 % MOD);
    						}
    						System.out.println(ans);
//    						printf("%d\n", ans);
    					}
    				}
    			}
    		}

            rd.close();
		}

		catch (IOException ioe) {
			ioe.printStackTrace();
		}



	}

	public static void main(String[] args) {
		new Codewords3();
	}
}
