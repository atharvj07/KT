#include <bits/stdc++.h>
using namespace std;
int CountInversions(const vector<int>& perm) {
  int ans = 0;
  for (int i = 0; i < perm.size(); i++) {
    for (int j = i + 1; j < perm.size(); j++) {
      ans += (perm[i] > perm[j]);
    }
  }
  return ans;
}
double ExpectedNumberOfInversions(int m, int k, vector<int> perm) {
  if (m == k) {
    return CountInversions(perm);
  }
  double res = 0;
  int n = perm.size();
  int numVars = n * (n + 1) / 2;
  for (int left = 0; left < n; left++) {
    for (int right = left; right < n; right++) {
      vector<int> next = perm;
      reverse(next.begin() + left, next.begin() + right + 1);
      res += ExpectedNumberOfInversions(m + 1, k, next);
    }
  }
  res /= numVars;
  return res;
}
int main(int argc, char** argv) {
  int n, k;
  cin >> n >> k;
  vector<int> perm(n);
  for (int i = 0; i < n; i++) {
    cin >> perm[i];
  }
  cout.precision(15);
  cout << ExpectedNumberOfInversions(0, k, perm);
  return 0;
}
