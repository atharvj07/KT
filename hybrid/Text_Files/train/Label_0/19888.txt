#include <bits/stdc++.h>
using namespace std;
int vis[210] = {0};
int main() {
  int n, i, j, f, x, y, x1, y1, v = 0, a[210], m = 0, s = 0;
  scanf("%d", &n);
  for (i = 0; i < 2 * n - 1; i++) {
    scanf("%d", &a[i]);
    if (a[i] >= 0) {
      v++;
    } else
      a[i] = -a[i];
  }
  queue<int> q;
  q.push(v);
  vis[v] = 1;
  while (!q.empty()) {
    f = q.front();
    x = f;
    y = (2 * n - 1) - f;
    for (i = 0; i <= n; i++) {
      if (x - i >= 0) {
        x1 = x - i + (n - i);
        if (n - i <= y && x1 <= 2 * n - 1 && vis[x1] == 0) {
          q.push(x1);
          vis[x1] = 1;
        }
      }
    }
    q.pop();
  }
  for (i = 0; i <= 2 * n - 1; i++) {
    if (vis[i] == 1) m = i;
  }
  sort(a, a + 2 * n - 1);
  m = 2 * n - 1 - m;
  for (i = m - 1; i >= 0; i--) {
    a[i] = -a[i];
  }
  for (i = 0; i < 2 * n - 1; i++) {
    s += a[i];
  }
  printf("%d\n", s);
  return 0;
}
