namespace{
	long long fact[C_SIZE];
	long long finv[C_SIZE];
	long long inv[C_SIZE];
	long long Comb(int a,int b){
	 	if(a<b||b<0)return 0;
	 	return fact[a]*finv[b]%mod*finv[a-b]%mod;
	}
	void init_C(int n){
		fact[0]=finv[0]=inv[1]=1;
		for(int i=2;i<n;i++){
			inv[i]=(mod-(mod/i)*inv[mod%i]%mod)%mod;
		}
		for(int i=1;i<n;i++){
			fact[i]=fact[i-1]*i%mod;
			finv[i]=finv[i-1]*inv[i]%mod;
		}
	}
	long long pw(long long a,long long b){
		if(a<0LL)return 0;
		if(b<0LL)return 0;
		long long ret=1;
		while(b){
			if(b%2)ret=ret*a%mod;
			a=a*a%mod;
			b/=2;
		}
		return ret;
	}
	long long pw_mod(long long a,long long b,long long M){
		if(a<0LL)return 0;
		if(b<0LL)return 0;
		long long ret=1;
		while(b){
			if(b%2)ret=ret*a%M;
			a=a*a%M;
			b/=2;
		}
		return ret;
	}
	int pw_mod_int(int a,int b,int M){
		if(a<0)return 0;
		if(b<0)return 0;
		int ret=1;
		while(b){
			if(b%2)ret=(long long)ret*a%M;
			a=(long long)a*a%M;
			b/=2;
		}
		return ret;
	}

	int UF[UF_SIZE];
	void init_UF(int n){
		for(int i=0;i<n;i++)UF[i]=-1;
	}
	int FIND(int a){
		if(UF[a]<0)return a;
		return UF[a]=FIND(UF[a]);
	}
	void UNION(int a,int b){
		a=FIND(a);b=FIND(b);if(a==b)return;
		if(UF[a]>UF[b])swap(a,b);
		UF[a]+=UF[b];UF[b]=a;
	}

// ここから編集しろ
int dx[]={1,0,-1,0};
int dy[]={0,1,0,-1};
int v[510][510][6];
long long ijk[510][510][6];
int x[110000];
int y[110000];
int bfs[510][510];
int main(){
	int h,w,a,b,c,n;
	scanf("%d%d%d%d%d%d",&h,&w,&a,&b,&c,&n);
	for(int i=0;i<=h;i++)for(int j=0;j<=w;j++){
		bfs[i][j]=-1;
	}
	queue<pair<int,int> > q;
	for(int i=0;i<n;i++){
		scanf("%d%d",x+i,y+i);
		bfs[x[i]][y[i]]=0;
		q.push(make_pair(x[i],y[i]));
	}
	while(q.size()){
		int row=q.front().first;
		int col=q.front().second;
		q.pop();
		for(int i=0;i<4;i++){
			int tr=row+dx[i];
			int tc=col+dy[i];
			if(tr<0||tc<0||tr>h||tc>w)continue;
			if(bfs[tr][tc]!=-1)continue;
			bfs[tr][tc]=bfs[row][col]+1;
			q.push(make_pair(tr,tc));
		}
	}
	for(int i=0;i<=h;i++)for(int j=0;j<=w;j++)for(int k=0;k<6;k++)ijk[i][j][k]=inf;
	ijk[x[0]][y[0]][0]=0;
	priority_queue<pair<long long,pair<pair<int,int>,int> > > Q;
	Q.push(make_pair(0LL,make_pair(make_pair(x[0],y[0]),0)));
	while(Q.size()){
		long long cost=-Q.top().first;
		int row=Q.top().second.first.first;
		int col=Q.top().second.first.second;
		int ty=Q.top().second.second;
		Q.pop();
		if(v[row][col][ty])continue;
		v[row][col][ty]=1;
		if(ty==0){
			if(!v[row][col][1]&&ijk[row][col][1]>cost+(long long)bfs[row][col]*c){
				ijk[row][col][1]=cost+(long long)bfs[row][col]*c;
				Q.push(make_pair(-ijk[row][col][1],make_pair(make_pair(row,col),1)));
			}
		}else if(ty==1){
			for(int i=0;i<4;i++){
				int tr=row+dx[i];
				int tc=col+dy[i];
				if(tr<0||tc<0||tr>h||tc>w)continue;
				if(v[tr][tc][ty])continue;
				if(ijk[tr][tc][ty]>cost+c){
					ijk[tr][tc][ty]=cost+c;
					Q.push(make_pair(-cost-c,make_pair(make_pair(tr,tc),ty)));
				}
			}
			if(!v[row][col][0]&&ijk[row][col][0]>cost){
				ijk[row][col][0]=cost;
				Q.push(make_pair(-ijk[row][col][0],make_pair(make_pair(row,col),0)));
			}
			for(int i=0;i<4;i++){
				int tt=2+i;
				long long toc=cost+b;
				if(v[row][col][tt])continue;
				if(ijk[row][col][tt]>toc){
					ijk[row][col][tt]=toc;
					Q.push(make_pair(-toc,make_pair(make_pair(row,col),tt)));
				}
			}
		}else{
			if(!v[row][col][0]&&ijk[row][col][0]>cost){
				ijk[row][col][0]=cost;
				Q.push(make_pair(-ijk[row][col][0],make_pair(make_pair(row,col),0)));
			}
			int tr=row+dx[ty-2];
			int tc=col+dy[ty-2];
			long long toc=cost+a;
			if(0<=tr&&tr<=h&&0<=tc&&tc<=w&&!v[tr][tc][ty]&&ijk[tr][tc][ty]>toc){
				ijk[tr][tc][ty]=toc;
				Q.push(make_pair(-toc,make_pair(make_pair(tr,tc),ty)));
			}
		}
	}
	printf("%lld\n",ijk[x[n-1]][y[n-1]][0]);
}
