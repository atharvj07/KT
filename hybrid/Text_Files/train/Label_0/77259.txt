n = int(input())

def y(x, d):
	return -x * x + d * x - d

def dydx(x, d):
	return -2 * x + d
for i in range(0, n):
	d = int(input())
	if d * d / 4 - d < 0:
		print('N')
	else:
		xn = 0
		while abs(xn * (d - xn) - d) > 10 ** (-6) and xn <= d:
			xn = xn - y(xn, d) / dydx(xn, d)
		print('Y', xn, d - xn)
