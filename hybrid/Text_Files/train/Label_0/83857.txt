def main():
	(n, k) = [int(s) for s in input().split()]
	a = [int(s) for s in input().split()]
	low = min(a)
	high = max(a)
	while low < high:
		mid = (low + high) // 2
		if possible(a, k, mid):
			high = mid
		else:
			low = mid + 1
	print(low)

def possible(a, k, x):
	return greedy(a, k, x, 0) or greedy(a, k, x, 1)

def greedy(a, k, x, parity):
	i = 0
	s = []
	while i < len(a) and len(s) < k:
		if len(s) % 2 == parity:
			if a[i] <= x:
				s.append(a[i])
		else:
			s.append(a[i])
		i += 1
	return len(s) == k
main()
