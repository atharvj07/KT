#include<iostream>
using namespace std;

int p[] = {0,31,28,31,30,31,30,31,31,30,31,30,31};

bool leap(int y){

  if(y%400 == 0) return true;
  if(y%100 == 0) return false;
  if(y%4 == 0) return true;
  return false;
}

int solve(int y1, int m1, int d1, int y2, int m2, int d2){

  if(y1 == y2 && m1 == m2 && d1 == d2) return 0;

  if(leap(y1) && m1 == 2 && d1 == 28) return solve(y1,3,1,y2,m2,d2)+2;
  else if(m1 == 12 && d1 == 31) return solve(y1+1,1,1,y2,m2,d2)+1;
  else if(p[m1] == d1) return solve(y1,m1+1,1,y2,m2,d2)+1;
  else return solve(y1,m1,d1+1,y2,m2,d2)+1;

}

int main(){

  int y1,m1,d1,y2,m2,d2;

  while(true){
    cin >> y1 >> m1 >> d1 >> y2 >> m2 >> d2;
    if(y1 < 0 || m1 < 0 || d1 < 0) break;
    if(y2 < 0 || m2 < 0 || d2 < 0) break;

    cout << solve(y1, m1, d1, y2, m2, d2) << endl;
  }

}
