import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Scanner;
import java.util.StringTokenizer;

public class BBB {
	static class mushroom implements Comparable<mushroom> {
		int num;
		double score;

		public mushroom(int n, double s) {
			num = n;
			score = s;
		}

		public int compareTo(mushroom arg0) {
			if (Math.abs(score - arg0.score) < 1e-9)
				return num - arg0.num;
			else
				return (score - arg0.score) < 0 ? 1 : -1;
		}
	}

	public static void main(String[] args) throws IOException {
		BufferedReader buff = new BufferedReader(new InputStreamReader(
				System.in));

		String input = buff.readLine();
		StringTokenizer st = new StringTokenizer(input);
		int n = Integer.parseInt(st.nextToken());
		int t1 = Integer.parseInt(st.nextToken());
		int t2 = Integer.parseInt(st.nextToken());
		int k = Integer.parseInt(st.nextToken());

		mushroom[] array = new mushroom[n];

		for (int i = 0; i < n; i++) {
			st = new StringTokenizer(buff.readLine());
			int v = Integer.parseInt(st.nextToken());
			int u = Integer.parseInt(st.nextToken());

			double firstHeight = v * t1;
			double secondHeight = (firstHeight - (k / 100.0) * firstHeight) + u
					* t2;

			double firstHeight1 = u * t1;
			double secondHeight1 = (firstHeight1 - (k / 100.0) * firstHeight1)
					+ v * t2;

			array[i] = new mushroom(i + 1, Math
					.max(secondHeight, secondHeight1));
		}
		Arrays.sort(array);
		for (int i = 0; i < array.length; i++) {
			System.out.printf("%d %.2f\n", array[i].num, array[i].score);
		}
	}

}
