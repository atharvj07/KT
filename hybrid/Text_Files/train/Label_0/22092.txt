#include <bits/stdc++.h>
using namespace std;
const int INF = 2e9;
const long long ML = 4e18;
int n, m, n1, n2, n3;
vector<vector<vector<int>>> ctr;
vector<vector<int>> edges;
vector<int> color;
bool dfs(int i, int c, vector<vector<int>>& rs) {
  color[i] = c;
  rs[c].push_back(i);
  for (int v : edges[i]) {
    if (color[v] == -1) {
      dfs(v, c ^ 1, rs);
    } else {
      if (color[v] != c ^ 1) return false;
    }
  }
  return true;
}
void solve() {
  cin >> n >> m >> n1 >> n2 >> n3;
  edges = vector<vector<int>>(n + 1);
  for (int i = 0; i < m; ++i) {
    int a, b;
    cin >> a >> b;
    edges[a].push_back(b);
    edges[b].push_back(a);
  }
  color = vector<int>(n + 1, -1);
  for (int i = 1; i < n + 1; ++i) {
    if (color[i] == -1) {
      vector<vector<int>> rs(2);
      if (!dfs(i, 0, rs)) {
        cout << "NO\n";
        return;
      }
      ctr.push_back(rs);
    }
  }
  int sz = ctr.size();
  int dp[sz + 1][n2 + 1];
  memset(dp, -1, sizeof(dp));
  dp[0][0] = 0;
  for (int i = 0; i < sz; ++i)
    for (int j = 0; j < n2 + 1; ++j) {
      if (dp[i][j] != -1) {
        if (j + ctr[i][0].size() <= n2) dp[i + 1][j + ctr[i][0].size()] = 0;
        if (j + ctr[i][1].size() <= n2) dp[i + 1][j + ctr[i][1].size()] = 1;
      }
    }
  if (dp[sz][n2] == -1) {
    cout << "NO\n";
    return;
  }
  vector<int> ans(n + 1, -1);
  int cur = n2;
  for (int i = sz; i >= 1; --i) {
    for (int j : ctr[i - 1][dp[i][cur]]) ans[j] = 2;
    cur -= ctr[i - 1][dp[i][cur]].size();
  }
  int cnt = 0;
  for (int i = 1; i < n + 1; ++i) {
    if (ans[i] == -1) {
      if (cnt < n1)
        ans[i] = 1;
      else
        ans[i] = 3;
      cnt++;
    }
  }
  cout << "YES\n";
  for (int i = 1; i < n + 1; ++i) cout << ans[i];
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  solve();
  return 0;
}
