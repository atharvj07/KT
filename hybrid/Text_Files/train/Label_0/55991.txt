// practice with rainboy
import java.io.*;
import java.util.*;

public class CF932D {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw = new PrintWriter(System.out);
		int m = Integer.parseInt(br.readLine());
		int[][] pp = new int[m + 2][20];
		long[][] xx = new long[m + 2][20];
		int last = 0, cnt = 1;
		while (m-- > 0) {
			StringTokenizer st = new StringTokenizer(br.readLine());
			int t = Integer.parseInt(st.nextToken());
			long p = Long.parseLong(st.nextToken());
			long q = Long.parseLong(st.nextToken());
			int i = (int) (p ^ last);
			long x = q ^ last;
			if (t == 1) {
				while (i != 0 && xx[i][0] < x)
					i = pp[i][0];
				int j = ++cnt;
				pp[j][0] = i;
				xx[j][0] = x;
				if (i != 0)
					for (int k = 1; k < 20; k++) {
						i = pp[j][k - 1];
						pp[j][k] = pp[i][k - 1];
						xx[j][k] = xx[j][k - 1] + xx[i][k - 1];
						if (pp[j][k] == 0)
							break;
					}
			} else {
				int ans = 0;
				for (int k = 19; k >= 0; k--)
					if (pp[i][k] != 0 && xx[i][k] <= x) {
						ans += 1 << k;
						x -= xx[i][k];
						i = pp[i][k];
					}
				if (xx[i][0] <= x)
					ans++;
				pw.println(ans);
				last = ans;
			}
		}
		pw.close();
	}
}
