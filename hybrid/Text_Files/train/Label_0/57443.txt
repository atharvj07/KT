import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;
 
public class D {

	static Scanner sc = new Scanner(System.in);
	static PrintWriter out = new PrintWriter(System.out);

    static void exit() {
    	sc.close();
    	out.close();
    	System.exit(0);
    }
    static int gcd(int a, int b) {
    	return b == 0 ? a : gcd(b, a%b);
    }
    public static void main(String[] args) {
    	int n = sc.nextInt();
    	int arr [] = new int [n];
    	for(int i = 0; i < n; i++) arr[i] = sc.nextInt();
    	Arrays.sort(arr);
    	int min_bun = Integer.MAX_VALUE;
    	int gmin=0, gmax=0, gstep = 0;
    	for(int step = 0; step <= arr[n-1]-arr[0]; step++) {
    		int min = 0;
        	int max = 0;
    		for(int i = 1; i < n; i++) {
    			min = Math.min(min, arr[0]+i*step-arr[i]);
    			max = Math.max(max, arr[0]+i*step-arr[i]);
    		}
    		int bun = (int) Math.ceil((max-min)/2.0);
    		//System.err.println(step + " " + min+" "+max + " " + bun);
    		if(min_bun > bun) {
    			min_bun = bun;
    			gmin = min;
    			gmax = max;
    			gstep = step;
    		}
    	}
    	
    	out.println(min_bun);
    	out.println(arr[0] + (-min_bun-gmin) + " " + gstep);
        exit();
    }
    

}
