def main():
	for tc in range(int(input())):
		(n, q) = map(int, input().split())
		ar = list(map(int, input().split()))
		(ans, inc) = (0, True)
		for i in range(n - 1):
			if inc and ar[i + 1] < ar[i]:
				ans += ar[i]
				inc = False
			elif not inc and ar[i + 1] > ar[i]:
				ans -= ar[i]
				inc = True
		if inc:
			ans += ar[-1]
		print(ans)
main()
