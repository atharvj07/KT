#include <bits/stdc++.h>
using namespace std;
const int inf = 2000100100;
int main() {
  int n, m;
  cin >> n >> m;
  int al = inf, ah = -inf, bl = inf, bh = -inf;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    al = min(al, x + y);
    ah = max(ah, x + y);
    bl = min(bl, x - y);
    bh = max(bh, x - y);
  }
  int best = inf, besti = -1;
  cin >> m;
  for (int i = 0; i < m; i++) {
    int x, y;
    cin >> x >> y;
    int cur = max(max(abs(al - (x + y)), abs(ah - (x + y))),
                  max(abs(bl - (x - y)), abs(bh - (x - y))));
    if (best > cur) {
      best = cur;
      besti = i;
    }
  }
  cout << best << '\n' << besti + 1;
}
