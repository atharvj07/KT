struct node{
    string m;
    vi to;
};

vector<node> t;

void print(int i,int d){
    rep(_,d)cout<<'.';
    cout<<t[i].m<<endl;
    for(int j:t[i].to){
        print(j,d+1);
    }
}

void run(){
    int n;
    cin>>n;
    t.resize(n);
    rep(i,n){
        int k;
        cin>>k>>t[i].m;
        if(i){
            k--;
            t[k].to.pb(i);
        }
    }
    print(0,0);
}
