#include <bits/stdc++.h>
using namespace std;
string s1, s2;
int x, y;
int g(int a, int b) {
  if (a < b) return g(b, a);
  if (a % b == 0) {
    return b;
  } else
    return g(b, a % b);
}
void h(int m, int k, string a1, string a2) {
  for (int i = m; i < k; i++) {
    if (a1[i] == a2[i]) continue;
    if (a1[i] == 'R' && a2[i] == 'S') x++;
    if (a1[i] == 'R' && a2[i] == 'P') y++;
    if (a1[i] == 'S' && a2[i] == 'P') x++;
    if (a1[i] == 'S' && a2[i] == 'R') y++;
    if (a1[i] == 'P' && a2[i] == 'R') x++;
    if (a1[i] == 'P' && a2[i] == 'S') y++;
  }
}
int main() {
  int n;
  while (cin >> n) {
    string a1, a2;
    x = 0;
    y = 0;
    s1 = "";
    s2 = "";
    if (n < 1) break;
    cin >> s1;
    a1 = s1;
    cin >> s2;
    a2 = s2;
    int z1 = s1.length();
    int z2 = s2.length();
    int z = z1 * z2 / g(z1, z2);
    for (int i = 1; i < z / z1; i++) {
      s1 += a1;
    }
    for (int i = 1; i < z / z2; i++) {
      s2 += a2;
    }
    int left = n % z;
    n = n / z;
    h(0, z, s1, s2);
    x = x * n;
    y = y * n;
    h(0, left, s1, s2);
    cout << y << " " << x;
  }
  return 0;
}
