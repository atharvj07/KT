#include <bits/stdc++.h>
using namespace std;
const int N = 205;
int n, d, mo, ans, tr[N];
int C[N][N], f[N][N];
int main() {
  scanf("%d%d%d", &n, &d, &mo);
  for (int i = (int)(0); i <= (int)(n); i++) C[i][0] = 1 % mo;
  for (int i = (int)(1); i <= (int)(n); i++)
    for (int j = (int)(1); j <= (int)(i); j++)
      C[i][j] = (1ll * C[i - 1][j] + C[i - 1][j - 1]) % mo;
  f[1][0] = 1;
  for (int i = (int)(1); i <= (int)(n); i++) {
    int S = 0;
    tr[0] = 1;
    for (int j = (int)(0); j <= (int)(d - 1); j++) S = (1ll * S + f[i][j]) % mo;
    for (int j = (int)(1); j <= (int)(min(n / i, d)); j++)
      tr[j] = 1ll * tr[j - 1] * C[i * j - 1][i - 1] % mo * S % mo;
    for (int j = (int)(n); j >= (int)(0); j--)
      for (int k1 = (int)(1); k1 <= (int)(min(n / i, d)); k1++)
        for (int k2 = (int)(0); k2 <= (int)(d - k1); k2++)
          f[j][k1 + k2] = (f[j][k1 + k2] + 1ll * f[j - k1 * i][k2] * tr[k1] %
                                               mo * C[j - 1][k1 * i]) %
                          mo;
  }
  for (int i = (int)(0); i <= (int)(n - 1); i++)
    for (int j = (int)(0); j <= (int)(d); j++)
      for (int k = (int)(0); k <= (int)(d - j); k++)
        if (k != 1) ans = (ans + 1ll * f[i + 1][j] * f[n - i][k]) % mo;
  printf("%d\n", 2ll * n * (n - 1) * ans % mo);
}
