(N, K) = [int(x) for x in input().split(' ')]

def f(x):
	(ret, i) = (0, 0)
	while x << i <= N:
		ret += min(N, (x << i) + (1 << i) - 1) - (x << i) + 1
		i += 1
	return ret
(lo, hi) = (1, N // 2)
while lo < hi:
	mi = (lo + hi + 1) // 2
	if f(2 * mi) + f(2 * mi + 1) >= K:
		lo = mi
	else:
		hi = mi - 1
if f(2 * lo) + f(2 * lo + 1) < K:
	lo = 0
(lo2, hi2) = (0, (N - 1) // 2)
while lo2 != hi2:
	mi2 = (lo2 + hi2 + 1) // 2
	if f(2 * mi2 + 1) >= K:
		lo2 = mi2
	else:
		hi2 = mi2 - 1
print(max(2 * lo, 2 * lo2 + 1))
