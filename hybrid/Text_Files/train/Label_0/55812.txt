import java.util.*;
public class E {

	public static void main(String[] args) {
		Scanner seer = new Scanner(System.in);
		int n = seer.nextInt();
		Piecewise function = new Piecewise(seer.nextInt());
		for(int i = 1; i < n; i++){
			function.transform(seer.nextInt());
		}
		System.out.println(function.min());
	}
	static class Piecewise{
		TreeMap<Long, Long> map;
		long fac;
		public Piecewise(long a){
			map = new TreeMap<>();
			map.put(a,0L);
			fac = 1;
		}
		public void transform(long t){
			TreeMap<Long, Long> newmap = new TreeMap<>();
			long miny = Long.MAX_VALUE; 
			for(long l: map.keySet()){
				miny = Long.min(miny, map.get(l));
				map.put(l, miny);
			}
			long prevx = Long.MAX_VALUE;
			for(long l: map.keySet()){
				long x = l+1; long y = map.get(l);
				if(x <= t){
					newmap.put(x, y+(t - x));
				}
				else{
					newmap.put(x, y+(x - t));
				}
				if(prevx <= t && t <= x) {
					long a = map.get(prevx-1);
					long b = map.get(l);
					long d = (a - b)/(x - prevx);
					newmap.put(t, b + d*(x - t));
				}
				prevx = x;
			}
			
			if(t < map.firstKey() + 1){
				long a = map.firstKey();
				newmap.put(t, (a-t+1)*fac + map.get(a));
			}
			else if(t > map.lastKey() + 1){
				newmap.put(t, miny);
			}
			map = newmap;
			fac++;
		}
		public long min(){
			long res = Long.MAX_VALUE;
			for(long y: map.values()){
				res = Long.min(res, y);
			}
			return res;
		}
	}
}
