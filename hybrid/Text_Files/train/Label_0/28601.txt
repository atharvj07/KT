#include <bits/stdc++.h>
using namespace std;
vector<vector<int> > nbrs;
int cycleCount = 0;
vector<int> visited;
int cycleStart;
int cycleLength;
void dfs(int x, int parent, int dist) {
  visited[x] = -1;
  for (int i = 0; i < nbrs[x].size(); ++i) {
    int v = nbrs[x][i];
    if (v == parent)
      continue;
    else if (visited[v] == -1) {
      ++cycleCount;
      cycleStart = v;
      cycleLength = dist + 1;
    } else if (visited[v] == 0) {
      dfs(v, x, dist + 1);
    }
  }
  visited[x] = 1;
}
int main(void) {
  int n, m;
  cin >> n >> m;
  nbrs.resize(n + 1);
  visited.resize(n + 1, 0);
  int a, b;
  for (int i = 0; i < m; ++i) {
    scanf("%d%d", &a, &b);
    nbrs[a].push_back(b);
    nbrs[b].push_back(a);
  }
  dfs(1, 0, 0);
  for (int i = 1; i <= n; ++i) {
    if (!visited[i]) {
      cout << "NO";
      return 0;
    }
  }
  if (cycleCount == 1) {
    if (cycleLength >= 3)
      cout << "FHTAGN!";
    else
      cout << "NO";
    return 0;
  }
  cout << "NO";
}
