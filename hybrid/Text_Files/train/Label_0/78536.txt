import math
for _ in range(int(input())):
	s = input()
	t = input()
	sb = ''
	k = 0
	for i in range(1, min(len(s), len(t)) + 1):
		if len(s) % i == 0 and len(t) % i == 0:
			sub = s[0:i]
			if s == sub * (len(s) // i) and t == sub * (len(t) // i):
				sb = sub
				k = i
				break
	if sb == '':
		print(-1)
	else:
		a = len(s) // k
		b = len(t) // k
		print(sb * (a * b // math.gcd(a, b)))
