import java.util.PriorityQueue;
import java.util.Scanner;


public class SerejaAndCoatRack_368 {
  public static void main(String[] args) {
	  
	Scanner sc=new Scanner(System.in);
	int n =sc.nextInt();
	int d =sc.nextInt();
	PriorityQueue<Integer> pq=new PriorityQueue<>();
	for (int i = 0; i <n; i++) {
		pq.add(sc.nextInt());
	}
	int m =sc.nextInt();
	int profit =0;
	if(m>=n){
		for (int i = 0; i <n; i++) {
			profit+=pq.poll();
		}
		m=m-n;
		m=m*d;
		profit=profit-m;
		System.out.println(profit);
	}
	else {
		
		while(m-->0){
			profit+=pq.poll();
		}
		System.out.println(profit);

		
	}
}
}
