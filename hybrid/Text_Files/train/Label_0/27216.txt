#include <bits/stdc++.h>
using namespace std;
const int maxn = 5210;
int l, ans;
char s[maxn];
bool f[maxn], flag;
void close() { exit(0); }
void init() {
  scanf("%s", s + 1);
  l = strlen(s + 1);
  for (int i = 1; i <= l; i++) {
    if (s[i] == 'b' && s[i + 1] == 'e' && s[i + 2] == 'a' && s[i + 3] == 'r')
      f[i + 3] = true;
  }
  for (int i = 1; i <= l; i++) {
    flag = false;
    for (int j = i; j <= l; j++) {
      if (f[j] && (j - i + 1 >= 4)) flag = true;
      if (flag) ans++;
    }
  }
  printf("%d\n", ans);
}
int main() {
  init();
  close();
  return 0;
}
