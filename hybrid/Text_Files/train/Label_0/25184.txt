#include <bits/stdc++.h>
using namespace std;
int n, a[60], mi, ans;
int main() {
  scanf("%d", &n);
  mi = 1e9;
  for (int i = (1); i <= (n); ++i) scanf("%d", &a[i]), mi = min(mi, a[i]);
  for (int i = (1); i <= (n); ++i) ans += mi == a[i];
  if (ans > n / 2)
    puts("Bob");
  else
    puts("Alice");
  return 0;
}
