int n,m;
int skl[110][110];
int skr[110][110];
int in[110];
bool vis[110];
vector<int> e[110];
queue<int> q;

char s[10];

bool solve()
{
    int kk,num,p,v;
    rep(i,1,n)rep(j,1,m)skl[i][j]=0,skr[i][j]=100;
    bool flag=false;
    rep(i,1,n)
    {
        scanf("%d",&kk);
        rep(j,1,kk)
        {
            scanf("%d%s%d",&num,s,&p);
            if(s[0]=='>')
            {
                if(skl[i][num]<p)
                {
                    if(skr[i][num]<p) flag=true;
                    else skl[i][num]=p;
                }
            }
            else if(s[0]=='<')
            {
                if(skr[i][num]>p)
                {
                    if(skl[i][num]>p) flag=true;
                    else skr[i][num]=p;
                }
            }
        }
        e[i].clear();
    }

    if(flag)return 0;

    clr(in,0);
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            bool flag=false;
            rep(k,1,m)
            {
                if(skr[i][k]<skl[j][k]){flag=true;break;}
            }
            if(flag)
            {
                e[j].push_back(i);
                in[i]++;
            }
        }
    }

    clr(vis,0);
    int cnt=0;
    rep(i,1,n)if(in[i]==0)
    {
        q.push(i);
        vis[i]=1;
        cnt++;
    }
    while(!q.empty())
    {
        int p=q.front();q.pop();
        if(!e[p].empty())rep(i,0,e[p].size()-1)
        {
            v=e[p][i];
            if(vis[v])continue;
            if(--in[v]==0)
            {
                q.push(v);
                vis[v]=1;
                cnt++;
            }
        }
    }

    return cnt==n;
}

    {

}
