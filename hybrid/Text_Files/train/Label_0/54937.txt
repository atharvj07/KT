		String[] split = readLine().split(" ");
		
		Integer[] stock = new Integer[]{Integer.valueOf(split[0]),Integer.valueOf(split[1]),Integer.valueOf(split[2]),Integer.valueOf(split[3]),Integer.valueOf(split[4])};

		Integer folks = Integer.valueOf(readLine());

		for (int i = 0; i < requests.size(); i++) {

	HashMap<String, Integer> lookup = new HashMap<String, Integer>();
	HashMap<Integer, String> backLook = new HashMap<Integer, String>();
	
	private String getBestShirt(String string, Integer[] stock) {
		
		int target = lookup.get(string);
		int offset = 0;
		while(true){
			
			if(target+offset >= 0 && target+offset< stock.length && stock[target+offset] > 0){
				stock[target+offset] --;
				return backLook.get(target+offset);
			}
			
			if(offset < 0){
				offset *= (-1);
				offset ++;
			} else if(offset > 0){
				offset *= -1;
			}else {
				offset ++;
			}
		}
	}

		new Main();
	}
