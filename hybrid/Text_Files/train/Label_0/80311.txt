n = int(input())
a = list(map(int, input().split()))
b = [0] * n
c = [0] * n
for i in range(n - 1):
	x = abs(a[i] - a[i + 1])
	if i % 2 == 0:
		b[i] = x
		c[i] = -x
	else:
		b[i] = -x
		c[i] = x

def maxSubArray(arr):
	best = 0
	current = 0
	for x in arr:
		current = max(0, current + x)
		best = max(best, current)
	return best
print(max(maxSubArray(c), maxSubArray(b)))
