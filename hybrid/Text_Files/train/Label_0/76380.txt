(n, k) = map(int, input().split())
dg = [0, 0, 1, 2, 9]

def getCombinationValue(r):
	if r == 0:
		return 1
	elif r == 1:
		return n
	elif r == 2:
		return (n - 1) * n // 2
	elif r == 3:
		return (n - 1) * n * (n - 2) // 6
	elif r == 4:
		return (n - 1) * n * (n - 2) * (n - 3) // 24
ans = 1
for m in range(k + 1):
	ans += getCombinationValue(m) * dg[m]
print(ans)
