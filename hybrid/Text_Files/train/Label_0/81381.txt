for tests in range(int(input())):
	(N, D) = map(int, input().split())
	P = list(map(int, input().split()))
	for i in range(1, N):
		if i > D:
			break
		if P[i] > 0:
			dP = i
			if dP > D:
				break
			else:
				k = D // dP
				if k <= P[i]:
					D -= k * dP
					P[0] += k
					break
				else:
					D -= P[i] * dP
					P[0] += P[i]
	print(P[0])
