import java.util.*;

public class Solution
{
	public static void main(String[] args)
	{
		Scanner in=new Scanner(System.in);
		int n=in.nextInt();
		long s,min=999999999,sum=0,t=0;
		s=in.nextLong();
		long v[]=new long[n];
		for(int i=0;i<n;i++)
		{
			v[i]=in.nextLong();
			if(min>v[i])
				min=v[i];
			t+=v[i];
		}
		if(t<s)
			System.out.println(-1);
		else
		{
			for(int i=0;i<n;i++)
			{
				sum+=v[i]-min;
				v[i]=min;
			}
			if(sum>=s)
			{
				System.out.println(min);
			}
			else
			{
				s=s-sum;
				System.out.println((n*min-s)/n);
			}
		}
	}
}
