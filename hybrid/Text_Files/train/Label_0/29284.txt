#include<iostream>
#include<queue>
using namespace std;
int N_MAX = 100001;
typedef pair<int,int> P;
int main()
{
  int n,i,j=1,xn,xb=0,k,kk,k1=0,k2=0;
  int a[N_MAX],x[N_MAX];
  bool u[N_MAX];
  priority_queue<P> q;
  cin >> n;
  if(n==2){
    cout << "-1" << endl;
    return 0;
  }
  fill(x+1,x+n+1,0);
  fill(u+1,u+n+1,0);
  for(i=1;i<=n;i++){
    cin >> a[i];
    x[a[i]]++;
  }
  for(i=1;i<=n;i++)
    q.push(P(x[i],i));
  for(i=1;i<=n;i++){
    xn=q.top().first;
    k=q.top().second;
    while(u[k]==1||xn>=n-i&&x[k]<xn){
      q.pop();
      if(u[k]==0)
        q.push(P(x[k],k));
      xn=q.top().first;
      k=q.top().second;
    }
    if(xn<n-i){
      while(u[j]==1) j++;
      k=j;
      if(xb==k){
        k++;
        while(u[k]==1) k++;
      }
    }
    if(i==n-2){
      for(kk=1;kk<=n;kk++)
        if(k1==0&&u[kk]==0&&kk!=k) k1=kk;
        else if(u[kk]==0&&kk!=k) k2=kk;
      if(a[k1]==k2&&a[k2]==k1){
        k=k1;
        if(xb==k) k=k2;
      }
    }
    cout << k << " ";
    u[k]=1;
    x[a[k]]--;
    xb=a[k];
  }
  cout << endl;
  return 0;
}
