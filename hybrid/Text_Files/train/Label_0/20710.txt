#include <bits/stdc++.h>
using namespace std;
int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    int p[n + 1];
    for (int i = 1; i <= n; i++) {
      cin >> p[i];
    }
    int ans[n + 1];
    for (int i = 0; i <= n; i++) ans[i] = 0;
    for (int i = 1; i <= n; i++) {
      if (!ans[i]) {
        int temp = p[i];
        int cnt = 1;
        while ((temp != i)) {
          temp = p[temp];
          cnt++;
        }
        ans[i] = cnt;
        temp = p[i];
        while (temp != i) {
          ans[temp] = cnt;
          temp = p[temp];
        }
      }
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << endl;
  }
  return 0;
}
