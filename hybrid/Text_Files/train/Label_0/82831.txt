t = int(input())
while t > 0:
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	w = list(map(int, input().split()))
	co = w.count(1)
	a.sort(reverse=True)
	w.sort()
	ans = sum(a[:k]) + sum(a[:co])
	a = a[k:]
	c = -1
	for i in w:
		if i != 1:
			c = c + i - 1
			ans += a[c]
	print(ans)
	t -= 1
