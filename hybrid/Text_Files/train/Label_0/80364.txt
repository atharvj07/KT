from decimal import Decimal
n = int(input())
l = []
for i in range(n):
	(x, y) = map(int, input().split())
	l.append([x, y])
ans = 999999999999999
for i in range(n):
	x1 = l[i][0]
	y1 = l[i][1]
	x2 = l[(i + 1) % n][0]
	y2 = l[(i + 1) % n][1]
	x3 = l[(i + 2) % n][0]
	y3 = l[(i + 2) % n][1]
	a = Decimal((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1)).sqrt()
	b = Decimal((x2 - x3) * (x2 - x3) + (y2 - y3) * (y2 - y3)).sqrt()
	c = Decimal((x3 - x1) * (x3 - x1) + (y3 - y1) * (y3 - y1)).sqrt()
	s = Decimal(a + b + c) / 2
	area = (s * (s - a) * (s - b) * (s - c)).sqrt()
	ans = min(ans, area / a, area / b, area / c)
print(ans)
