def solve(n, m):
	res = 0
	temp = []
	x = m
	for i in range(1, 11):
		if x > n:
			return sum(temp)
		temp.append(x % 10)
		if n == i:
			return sum(temp)
		x += m
	p = n // m // 10
	res += p * sum(temp)
	rem = n // m % 10
	j = 0
	for i in temp:
		if j == rem:
			break
		res += i
		j += 1
	return int(res)
t = int(input())
for q in range(t):
	(n, m) = map(int, input().split())
	print(solve(n, m))
