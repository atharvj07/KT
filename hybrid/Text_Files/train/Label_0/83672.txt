(x1, y1, x2, y2) = map(int, input().split())
(x3, y3, x4, y4) = map(int, input().split())
(x5, y5, x6, y6) = map(int, input().split())
A = [(x1, y1), (x1, y2), (x2, y1), (x2, y2)]
risultato = False
Dentro = [-1, -1, -1, -1]
n = 0
for P in A:
	x = P[0]
	y = P[1]
	if not (x3 <= x <= x4 and y3 <= y <= y4 or (x5 <= x <= x6 and y5 <= y <= y6)):
		risultato = True
		break
	else:
		if x3 <= x <= x4 and y3 <= y <= y4:
			Dentro[n] = 1
		if x5 <= x <= x6 and y5 <= y <= y6:
			Dentro[n] = 2
	n = n + 1
if risultato:
	print('YES')
else:
	x = Dentro[0]
	tuttiinuno = True
	for n in range(4):
		if Dentro[n] != x:
			tuttiinuno = False
			break
		x = Dentro[n]
	if (x4 < x5 or x3 > x6 or y3 > y6 or (y4 < y5)) and (not tuttiinuno):
		print('YES')
	else:
		print('NO')
