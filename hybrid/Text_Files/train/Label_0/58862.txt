import java.io.*;
import java.util.*;

public class cf355b {
  static FastIO in = new FastIO(), out = in;

  public static void main(String[] args) {
    int c1 = in.nextInt(), c2 = in.nextInt(), c3 = in.nextInt(), c4 = in.nextInt();
    int n = in.nextInt(), m = in.nextInt();
    int ans = 0, tmp = 0;
    for(int i=0; i<n; i++)
      tmp += Math.min(c2, in.nextInt()*c1);
    ans += Math.min(tmp,c3);
    tmp = 0;
    for(int i=0; i<m; i++)
      tmp += Math.min(c2, in.nextInt()*c1);
    ans += Math.min(tmp,c3);
    out.println(Math.min(ans,c4));
    out.close();
  }

  static class FastIO extends PrintWriter {
    BufferedReader br;
    StringTokenizer st;

    public FastIO() {
      this(System.in, System.out);
    }

    public FastIO(InputStream in, OutputStream out) {
      super(new BufferedWriter(new OutputStreamWriter(out)));
      br = new BufferedReader(new InputStreamReader(in));
      scanLine();
    }

    public void scanLine() {
      try {
        st = new StringTokenizer(br.readLine().trim());
      } catch (Exception e) {
        throw new RuntimeException(e.getMessage());
      }
    }

    public int numTokens() {
      if (!st.hasMoreTokens()) {
        scanLine();
        return numTokens();
      }
      return st.countTokens();
    }

    public String next() {
      if (!st.hasMoreTokens()) {
        scanLine();
        return next();
      }
      return st.nextToken();
    }

    public double nextDouble() {
      return Double.parseDouble(next());
    }

    public long nextLong() {
      return Long.parseLong(next());
    }

    public int nextInt() {
      return Integer.parseInt(next());
    }
  }
}
