vector<int> g[1000];
int grundy(int v, int p);

int vvv;
void dfs(int v, int p, int gr, int vvv, vector<int> &gv) {
  FOR(it, g[v]) {
    if (*it == p) continue;
    gr ^= grundy(*it, v);
  }
  // if (gr == 0) {
  //   cout << p+1 << " " << vvv+1 << " " << v+1 << endl;
  // }
  gv.push_back(gr);
  FOR(it, g[v]) {
    if (*it == p) continue;
    dfs(*it, v, gr ^ grundy(*it, v), vvv, gv);
  }
}

int dp[1000];
int grundy(int v, int p) {
  if (dp[v] >= 0) return dp[v];
  vvv= v;
  vector<int> gv;
  dfs(v,p,0,v,gv);
  sort(ALL(gv));
  int res = 0;
  gv.erase(unique(ALL(gv)), gv.end());
  REP(i,gv.size()) {
    if (gv[i] != res) break;
    res++;
  }
  // cout << v+1 << " " << gv << " " << res << endl;
  return dp[v] = res;
}

      g[p-1].push_back(i);
      num[i]++;
    }
    memset(dp,-1,sizeof(dp));
    int gr = 0;
    REP(i,n) {
      if (num[i] == 0) {
        gr ^= grundy(i,-1);

    }

  }
}
