(n, p, k) = [int(i) for i in input().split()]
s = []
if p - k > 1:
	b = p - k
else:
	b = 1
if b != 1:
	s.append('<<')
if p + k < n:
	e = p + k
else:
	e = n
for i in range(b, e + 1):
	if i == p:
		s.append('(' + str(p) + ')')
	else:
		s.append(i)
if p + k < n:
	s.append('>>')
print(' '.join([str(i) for i in s]))
