import java.util.*;
public class p255a {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int N = in.nextInt();
		int[] arr = new int[3];
		for(int i = 0; i < N; i++)
			arr[i%3] += in.nextInt();
		
		if(arr[0] > arr[1] && arr[0] > arr[2])
			System.out.println("chest");
		else if(arr[1] > arr[2])
			System.out.println("biceps");
		else
			System.out.println("back");
		
	}
}
