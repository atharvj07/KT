#include<cstdlib>
#include<iostream>
#include<string>
using namespace std;

struct Node{
  int num;
  Node *right,*left,*parent;
};

Node *NIL,*root;

Node *treeMinimum(Node *T){
  while(T->left!=NIL)T=T->left;
  return T;
}

Node *find(Node *T,int k){
  while(T!=NIL&&k!=T->num){
    if(k<T->num)T=T->left;
    else T=T->right;
  }
  return T;
}

Node *treeSuccessor(Node *x){
  if(x->right!=NIL)return treeMinimum(x->right);
  Node *y=x->parent;
  while(y!=NIL&&x==y->right){
    x=y;
    y=y->parent;
  }
  return y;
}

void deletNode(Node *z){
  Node *y;
  Node *x;

  if(z->left==NIL||z->right==NIL) y=z;
  else y=treeSuccessor(z);

  if(y->left!=NIL) x=y->left;
  else x=y->right;

  if(x!=NIL) x->parent=y->parent;

  if(y->parent==NIL) root=x;
  else{
    if(y==y->parent->left){
      y->parent->left=x;
    }else y->parent->right=x;
  }

  if(y!=z) z->num=y->num;

  free(y);

}

void insert(int k){
  Node *y=NIL;
  Node *x=root;
  Node *T;
  
  T=(Node *)malloc(sizeof(Node));
  T->num=k;
  T->left=NIL;
  T->right=NIL;
  
  while(x!=NIL){
    y=x;
    if(T->num<x->num) x=x->left;
    else x=x->right;
    
  }
  
  T->parent=y;
    
  if(y==NIL) root=T;
  else{
    if(T->num<y->num) y->left=T;
    else y->right=T;
  }
}

void inorder(Node *T){
  if(T==NIL) return;
  inorder(T->left);
  cout<<" "<<T->num;
  inorder(T->right);
}

void preorder(Node *T){
  if(T==NIL) return;
  cout<<" "<<T->num;
  preorder(T->left);
  preorder(T->right);
}

int main(){
  int m,k;
  string S;

  cin>>m;
  for(int i=0;i<m;i++){
    cin>>S;
    if(S=="insert"){
      cin>>k;
      insert(k);
    }else if(S=="print"){
      inorder(root);
      cout<<endl;
      preorder(root);
      cout<<endl;
    }else if(S=="find"){
      cin>>k;
      Node *t=find(root,k);
      if(t!=NIL) cout<<"yes"<<endl;
      else cout<<"no"<<endl;
    }else if(S=="delete"){
      cin>>k;
      deletNode(find(root,k));
    }
  }
  return 0;
}
