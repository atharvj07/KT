import sys
input = sys.stdin.readline

def inp():
	return int(input())

def st():
	return input().rstrip('\n')

def lis():
	return list(map(int, input().split()))

def ma():
	return map(int, input().split())
t = inp()
while t:
	t -= 1
	(n, k) = ma()
	a = lis()
	if n == 1:
		print(k + 1)
		continue
	fl = 0
	k += 1
	res = ''
	for i in range(n - 1):
		ra = 10 ** a[i + 1] - 1
		curmo = 0
		num = ra
		for j in range(i, -1, -1):
			ra1 = ra // 10 ** a[j]
			curmo += ra1
			ra = ra % 10 ** a[j]
		if curmo < k:
			continue
		else:
			z1 = 10 ** a[i] - 1
			res = str(z1)
			moves = 0
			ra = z1
			for j in range(i - 1, -1, -1):
				ra1 = ra // 10 ** a[j]
				moves += ra1
				ra = ra % 10 ** a[j]
			rem = k - moves
			if int(res) != 0:
				res = str(rem) + res
			else:
				res = str(rem)
			fl = 1
			break
	if fl == 0:
		z1 = 10 ** a[-1] - 1
		res = str(z1)
		moves = 0
		ra = z1
		for j in range(n - 2, -1, -1):
			ra1 = ra // 10 ** a[j]
			moves += ra1
			ra = ra % 10 ** a[j]
		rem = k - moves
		if int(res):
			res = str(rem) + res
		else:
			res = str(rem)
		fl = 1
	print(int(res))
