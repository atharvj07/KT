#include <bits/stdc++.h>
using namespace std;
long long n, k, res = 0, temp = 1, arr[100100];
int main() {
  cin >> n >> k;
  cin >> arr[0];
  for (int i = 1; i < n; i++) {
    cin >> arr[i];
    if (arr[i] != arr[i - 1]) {
      temp++;
    } else {
      if (res < temp) {
        res = temp;
      }
      temp = 1;
    }
  }
  if (res < temp) {
    res = temp;
  }
  cout << res;
  return 0;
}
