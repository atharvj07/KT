from math import *
(s, a, b) = map(int, input().split())
n = 6 * s
if n <= a * b:
	print(a * b)
	print(a, b)
else:
	while True:
		x = min(a, b)
		val = sqrt(n)
		while n % x != 0 and x <= val:
			x += 1
		if n // x >= max(a, b) and n % x == 0:
			print(n)
			if a < b:
				print(x, n // x)
			else:
				print(n // x, x)
			break
		n += 1
