#include <bits/stdc++.h>
using namespace std;
const int INF = 1000000009;
const int MAXN = 1000 * 1000;
const double EPS = 1e-9;
const int RIGHT = -1;
const int LEFT = 1;
int q;
long long m;
set<pair<double, double> > hull;
int Det(pair<double, double> a, pair<double, double> b,
        pair<double, double> c) {
  b.first -= a.first;
  b.second -= a.second;
  c.first -= a.first;
  c.second -= a.second;
  double det = b.first * c.second - b.second * c.first;
  return det < 0 ? RIGHT : LEFT;
}
void Add(int first, int second) {
  if (first == 0) return;
  auto it = hull.lower_bound(make_pair(first, second));
  if (!(it == hull.end() ||
        Det(*(prev(it)), *it, make_pair(first, second)) == RIGHT))
    return;
  it = (hull.insert(make_pair(first, second))).first;
  while (prev(it) != hull.begin() &&
         Det(*(prev(prev(it))), *(prev(it)), make_pair(first, second)) == RIGHT)
    hull.erase(prev(it));
  while (it != hull.end() && next(it) != hull.end() &&
         next(next(it)) != hull.end() &&
         Det(*next(it), *next(next(it)), make_pair(first, second)) == RIGHT)
    hull.erase(next(it));
}
bool Query(int t, int h) {
  double first = ((double)h) / t;
  double second = ((double)m) / t;
  auto maks = hull.rbegin();
  if (maks->first < first) return false;
  auto it = hull.lower_bound(make_pair(first, second));
  if (it == hull.end()) it = prev(it);
  return Det(*prev(it), *it, make_pair(first, second)) == LEFT;
}
int main() {
  scanf("%d %lld", &q, &m);
  int j = 0;
  hull.insert(make_pair(0, 0));
  for (int i = 1; i <= q; i++) {
    int type, a, b;
    scanf("%d %d %d", &type, &a, &b);
    a = (a + j) % MAXN + 1;
    b = (b + j) % MAXN + 1;
    if (type == 1)
      Add(a, b);
    else {
      bool ans = Query(a, b);
      if (ans) {
        puts("YES");
        j = i;
      } else
        puts("NO");
    }
  }
  return 0;
}
