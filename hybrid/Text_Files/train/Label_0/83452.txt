def main():
	t = int(input())
	for _ in range(t):
		(a, b) = (int(x) for x in input().split())
		s = input()
		print(solver(s, a, b))

def solver(s, a, b):
	if a + b != len(s):
		return -1
	a -= s.count('0')
	b -= s.count('1')
	result = [c for c in s]
	for i in range(len(s) // 2):
		left = s[i]
		right = s[len(s) - i - 1]
		if left != right:
			if left == '?' and right == '0':
				result[i] = '0'
				a -= 1
			elif left == '0' and right == '?':
				result[len(s) - i - 1] = '0'
				a -= 1
			elif left == '?' and right == '1':
				result[i] = '1'
				b -= 1
			elif left == '1' and right == '?':
				result[len(s) - i - 1] = '1'
				b -= 1
			else:
				return -1
	if a < 0 or b < 0:
		return -1
	if len(s) % 2 == 1:
		if result[len(s) // 2] == '?':
			if a % 2 == 1:
				result[len(s) // 2] = '0'
				a -= 1
			elif b % 2 == 1:
				result[len(s) // 2] = '1'
				b -= 1
			else:
				return -1
	for i in range(len(s) // 2):
		left = result[i]
		if left == '?':
			if a >= 2:
				result[i] = '0'
				result[len(s) - i - 1] = '0'
				a -= 2
			elif b >= 2:
				result[i] = '1'
				result[len(s) - i - 1] = '1'
				b -= 2
			else:
				return -1
	if a == 0 and b == 0:
		return ''.join(result)
	else:
		return -1
