class TaskC {
	public void solve(int testNumber, StreamInputReader in, PrintWriter out) {
        String s = in.readString();

        boolean[] hasPair = new boolean[s.length()];
        Stack<Integer> br = new Stack<Integer>();

        for(int i = 0; i < s.length(); i++) {
            if(s.charAt(i) == '(' || s.charAt(i) == '[')
                br.add(i);
            if(s.charAt(i) == ')' || s.charAt(i) == ']') {
                boolean ok = true;
                if(s.charAt(i) == ')' && (br.isEmpty() || s.charAt(br.peek()) != '('))
                    ok = false;
                if(s.charAt(i) == ']' && (br.isEmpty() || s.charAt(br.peek()) != '['))
                    ok = false;

                if(ok) {
                    hasPair[i] = true;
                    hasPair[br.peek()] = true;
                    br.pop();
                }
                else {
                    br.clear();
                }
            }
        }

        int maxSq = 0;
        int maxleft = -1;
        int maxright = -1;
        for(int i = 0; i < s.length(); i++)
            if(hasPair[i] && (i == 0 || !hasPair[i - 1])) {
                int crt = 0;
                int j;
                for(j = i; j < s.length() && hasPair[j]; j++)
                    if(s.charAt(j) == '[')
                        crt++;

                if(crt > maxSq) {
                    maxSq = crt;
                    maxleft = i;
                    maxright = j - 1;
                }
            }

        out.println(maxSq);
        if(maxSq > 0)
            out.print(s.substring(maxleft, maxright + 1));
	}

}

class StreamInputReader extends InputReader {
    private InputStream stream;
    private byte[] buf = new byte[1024];
    private int curChar, numChars;

    public StreamInputReader(InputStream stream) {
        this.stream = stream;
    }

    public int read() {
        if (numChars == -1)
            throw new InputMismatchException();
        if (curChar >= numChars) {
            curChar = 0;
            try {
                numChars = stream.read(buf);
            } catch (IOException e) {
                throw new InputMismatchException();
            }
            if (numChars <= 0)
                return -1;
        }
        return buf[curChar++];
    }

    }

abstract class InputReader {

    public abstract int read();

    public String readString() {
        int c = read();
        while (isSpaceChar(c))
            c = read();
        StringBuffer res = new StringBuffer();
        do {
            res.appendCodePoint(c);
            c = read();
        } while (!isSpaceChar(c));
        return res.toString();
    }

    private boolean isSpaceChar(int c) {
        return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
    }

    }
