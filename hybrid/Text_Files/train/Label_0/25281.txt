#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN=1e5+1;
const ll mod=1e9+7;
ll power(ll a,ll b){
    ll res=1;
    while(b>0){
        if(b%2!=0)res=(res*a)%mod;
        b/=2;
        a=(a*a)%mod;
    }
    return res%mod;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL); 
    int t;
    cin>>t;
    while(t--){
       int n;
       int k;
       cin>>n>>k;
       cout<<power(n,k)<<endl;
    }

    return 0;
}
