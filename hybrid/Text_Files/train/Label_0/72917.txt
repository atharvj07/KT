from sys import stdin
input = stdin.readline
(n, m) = map(int, input().split())
l = list(map(int, input().split()))
leng = 1 << n
seg = [0] * leng + l
cnt = 0
(curr, brk) = (0, 1 << n - 1)
for i in range(leng - 1, 0, -1):
	curr += 1
	if cnt % 2:
		seg[i] = seg[i << 1] ^ seg[(i << 1) + 1]
	else:
		seg[i] = seg[i << 1] | seg[(i << 1) + 1]
	if curr == brk:
		curr = 0
		brk //= 2
		cnt += 1
for _ in range(m):
	(p, b) = map(int, input().split())
	idx = p + leng - 1
	seg[idx] = b
	cnt = 0
	while idx > 1:
		if cnt % 2:
			seg[idx >> 1] = seg[idx] ^ seg[idx ^ 1]
		else:
			seg[idx >> 1] = seg[idx] | seg[idx ^ 1]
		cnt += 1
		idx >>= 1
	print(seg[1])
