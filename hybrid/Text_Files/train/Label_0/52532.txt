	public static void main(String[] args)
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt(), m = sc.nextInt();
		int div1 = 1, div2 = n;
		boolean flag = true;
		for(int i = 0; i < m; i++)
		{
			int u = sc.nextInt(), v = sc.nextInt();
			if(u > v)
			{
				int t = u;
				u = v;
				v = t;
			}
			if(u > div1)
			{
				div1 = u;
			}
			if(v < div2)
			{
				div2 = v;
			}
			if(div1 > div2)
			{
				flag = false;
			}
		}
		if(flag)
		{
			out.println(div2 - div1);
		}
		else
		{
			out.println(0);
		}
	}
}
