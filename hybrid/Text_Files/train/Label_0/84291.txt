from itertools import groupby
n = int(input())
tab = sorted(list(map(int, input().split())))
while len(tab) > 1 and tab[-1] == tab[-2]:
	tab.pop()
S = []
result = []
for (num, iter_elems) in groupby(tab):
	amount = len(list(iter_elems))
	if amount > 1:
		result += [num]
		S += [num]
	else:
		result += [num]
for x in S[-1::-1]:
	result += [x]
print(len(result))
print(' '.join(map(str, result)))
