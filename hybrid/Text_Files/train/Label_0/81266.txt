from collections import deque
for loop in range(int(input())):
	n = int(input())
	lis = [[] for i in range(n)]
	link = [0] * n
	num = [1] * n
	for i in range(n):
		(u, v) = map(int, input().split())
		u -= 1
		v -= 1
		lis[u].append(v)
		lis[v].append(u)
		link[u] += 1
		link[v] += 1
	q = deque([])
	ans = n * (n - 1)
	for i in range(n):
		if link[i] == 1:
			q.append(i)
	while q:
		v = q.popleft()
		link[v] = 0
		for nex in lis[v]:
			if link[nex] != 0:
				num[nex] += num[v]
				link[nex] -= 1
				if link[nex] == 1:
					q.append(nex)
	for i in range(n):
		if link[i] != 0:
			ans -= num[i] * (num[i] - 1) // 2
	print(ans)
