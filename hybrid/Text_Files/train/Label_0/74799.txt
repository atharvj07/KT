from array import array
from sys import stdin
import bisect
from itertools import *

def scan_gen():
	for line in stdin:
		yield from iter(line.split())
scan = scan_gen()

def nint():
	return int(next(scan))

def nintk(k):
	return tuple((nint() for _ in range(k)))

def nfloat():
	return float(next(scan))

def intar_init(size):
	return array('i', [0]) * size

def intar(size=None):
	if size == None:
		size = nint()
	arr = intar_init(size)
	for x in range(size):
		arr[x] = nint()
	return arr

def count(s):
	r = 0
	for a in range(len(s)):
		for b in range(a + 1, len(s) + 1):
			if max(s[a:b]) - min(s[a:b]) == b - a - 1:
				r += 1
	return r
fact = [1, 1]
mod = 0

def doit(n):
	res = 0
	for ln in range(1, n + 1):
		counts = n - ln + 1
		starts = n - ln + 1
		res = (res + counts * starts * fact[ln] * fact[n - ln]) % mod
	return res

def solve():
	global mod
	(n, mod) = nintk(2)
	for i in range(2, 300000):
		fact.append(fact[-1] * i % mod)
	print(doit(n))
solve()
