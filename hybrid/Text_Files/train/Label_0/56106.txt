import java.util.*;
import java.io.*;

public class D_Powerful_Ksenia{
    public static void main(String[] args) {
        FastScanner s= new FastScanner();
        StringBuilder res = new StringBuilder();
        int n=s.nextInt();
        int array[]= new int[n];
        for(int i=0;i<n;i++){
            array[i]=s.nextInt();
        }
        if(n%2!=0){
            res.append("YES\n");
            res.append((n-2)+"\n");
            for(int i=0;i<n;i+=2){
                if((i+2)<n){
                    res.append((i+1)+" "+(i+2)+" "+(i+3)+"\n");
                }
               
            }
            for(int i=0;i<(n-3);i+=2){
                res.append((i+1)+" "+(i+2)+" "+(n-1)+"\n");
            }
        }
        else{
            int find=0;
            for(int i=0;i<n;i++){
                find=find^array[i];
            }
            if(find!=0){
                res.append("NO\n");
            }
            else{
                res.append("YES\n");
            res.append((n-3)+"\n");
            n--;
            for(int i=0;i<n;i+=2){
                if((i+2)<n){
                    res.append((i+1)+" "+(i+2)+" "+(i+3)+"\n");
                }
              
            }
            for(int i=0;i<(n-3);i+=2){
                res.append((i+1)+" "+(i+2)+" "+(n-1)+"\n");
            }

            }
        }
        System.out.println(res);

    }
    static class FastScanner {
		BufferedReader br;
		StringTokenizer st;
 
		public FastScanner(String s) {
			try {
				br = new BufferedReader(new FileReader(s));
			} catch (FileNotFoundException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}
 
		public FastScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}
 
		String nextToken() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					// TODO Auto-generated catch block
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}
 
		int nextInt() {
			return Integer.parseInt(nextToken());
		}
 
		long nextLong() {
			return Long.parseLong(nextToken());
		}
 
		double nextDouble() {
			return Double.parseDouble(nextToken());
		}
	}




}
