#include<iostream>
using namespace std;

int main(){
    ios_base::sync_with_stdio(false);
	cin.tie(NULL);
    int t;
    cin>>t;
    for(int i=0;i<t;i++){
        double r,b,d;
        cin>>r>>b>>d;
        if(r>=b){
            double x=(r-b)/b;
            if(x<=d){
                cout<<"YES"<<endl;
            }
            else{
                cout<<"NO"<<endl;
            }
        }
        else{
             double x=(b-r)/r;
            if(x<=d){
                cout<<"YES"<<endl;
            }
            else{
                cout<<"NO"<<endl;
            }
        }
    }
    return 0;
}
