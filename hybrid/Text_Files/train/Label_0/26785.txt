#define LL long long
int read(){   
    int f=1,x=0;char c=getchar();   
    while(c<'0'||'9'<c){if(c=='-')f=-1;c=getchar();}   
    while('0'<=c&&c<='9'){x=x*10+c-'0';c=getchar();}   
    return f*x;
}
#define MAXN int(5e6)
#define INF 0x3f3f3f3f
const LL P=924844033,g=5,rg=554906420;
LL Lim=1,lg2,a[MAXN+5],b[MAXN+5],rev[MAXN+5];
LL Pow(LL x,int y){
	LL ret=1;
	while(y){
		if(y&1) ret=(ret*x)%P;
		x=(x*x)%P;
		y>>=1;
	}
	return ret;
}
#define limit Lim
#define G g
#define Gi rg
void NTT(LL *A, int type) {
    for(int i=0;i<limit;i++) 
        if(i<rev[i]) swap(A[i],A[rev[i]]);
    for(int mid=1;mid<limit;mid <<=1) {  
        LL Wn=Pow(type==1?G:Gi,(P-1)/(mid<<1));
        for(int j=0;j<limit;j+=(mid<<1)) {
            LL w=1;
            for(int k=0;k<mid;k++,w=(w*Wn)%P) {
				int x=A[j+k],y=w*A[j+k+mid]%P;
				A[j+k]=(x+y)%P,A[j+k+mid]=(x-y+P)%P;
            }
        }
    }
}
vector<int> Gra[MAXN+5];
int N,siz[MAXN+5],cnt[MAXN+5];
void DFS(int u,int fa){
	siz[u]=1;
	for(int i=0;i<(int)Gra[u].size();i++){
		int v=Gra[u][i];
		if(v==fa) continue;
		DFS(v,u),siz[u]+=siz[v];
	}
	if(fa!=0)
		cnt[N]++,cnt[siz[u]]--,cnt[N-siz[u]]--;
	return ;
}
LL fac[MAXN+5],inv[MAXN+5];
int main(){
	N=read();
	fac[0]=1;
	for(int i=1;i<=N;i++)
		fac[i]=fac[i-1]*i%P;
	inv[N]=Pow(fac[N],P-2);
	for(int i=N-1;i>=0;i--)
		inv[i]=inv[i+1]*(i+1)%P;
	for(int i=1;i<N;i++){
		int u=read(),v=read();
		Gra[u].push_back(v);
		Gra[v].push_back(u);
	}
	DFS(1,0),cnt[N]=N;
	
	int n=N,m=N;
	for(int i=0;i<=n;i++)
		a[i]=cnt[i]*fac[i]%P;
	for(int i=0;i<=m;i++)
		b[m-i]=inv[i];
	while(Lim<=n+m) Lim<<=1,lg2++;
	for(int i=0;i<Lim;i++)
		rev[i]=(rev[i>>1]>>1)|((i&1)<<(lg2-1));
	NTT(a,1),NTT(b,1);
	for(int i=0;i<Lim;i++)
		a[i]=(a[i]*b[i])%P;
	NTT(a,-1);
	LL inv_Lim=Pow(Lim,P-2);
	for(int i=0;i<=n+m;i++)
		a[i]=a[i]*inv_Lim%P;
	for(int i=1;i<=n;i++)
		printf("%lld\n",a[n+i]*inv[i]%P);
	return 0;
}
