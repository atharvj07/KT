n = int(input())
q = [int(x) for x in input().split()]
min_offset = 10 ** 6
running = 0
r = []
r.append(running)
min_offset = min(min_offset, running)
for i in range(n - 1):
	running += q[i]
	r.append(running)
	min_offset = min(min_offset, running)
min_offset = 1 - min_offset
checker = [False for _ in range(n)]
p = []
for i in r:
	num = min_offset + i
	if num > n or num < 1:
		print(-1)
		exit()
	if checker[num - 1]:
		print(-1)
		exit()
	checker[num - 1] = True
	p.append(min_offset + i)
for i in p:
	print(i, end=' ')
