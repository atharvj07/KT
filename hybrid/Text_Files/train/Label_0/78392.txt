def calc(n, k):
	if n % 2 == 0 and k % 2 != 0:
		return 'NO'
	elif n % 2 != 0 and k % 2 == 0:
		return 'NO'
	elif n % 2 == 0:
		if k ** 2 <= n:
			return 'YES'
		else:
			return 'NO'
	elif n % 2 != 0:
		if k ** 2 <= n:
			return 'YES'
		else:
			return 'NO'
for _ in range(int(input())):
	(n, k) = list((int(i) for i in input().split()))
	print(calc(n, k))
