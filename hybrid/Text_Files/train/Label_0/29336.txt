#include <bits/stdc++.h>
using namespace std;
int arr[200009];
int main() {
  int a, b, c, d, e, f, g, h;
  scanf("%d", &a);
  for (d = 0; d < a; d++) scanf("%d", &arr[d]);
  sort(arr, arr + a);
  g = arr[a - 1] - arr[0];
  for (d = 0, e = a / 2; d < e; d++) {
    g = min(g, arr[e + d] - arr[d]);
  }
  printf("%d\n", g);
}
