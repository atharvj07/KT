import java.util.Scanner;
import java.io.PrintWriter;
import java.util.HashSet;
import java.util.Arrays;

public class A448 implements Runnable {
	private Scanner in = new Scanner(System.in);
	private PrintWriter out = new PrintWriter(System.out);
	private int cup [];
	private int medal [];
	private int shelve;

	int cupSum = 0;
	int medalSum = 0;
	private String ans = "NO";
	
	public static void main(String[] args) {
		new Thread(new A448()).start();
	}

	private void read() {

		cup = new int[3];
		medal = new int[3];

		for(int i = 0; i < 3; i++){
			cup[i] = in.nextInt();
			cupSum += cup[i];
		}
		
		for(int i = 0; i < 3; i++){
			medal[i] = in.nextInt();
			medalSum += medal[i];
		}

		shelve = in.nextInt();
	}

	private void solve() {
		
		if(cupSum > 5){
			shelve = shelve - (int) Math.ceil((double)cupSum / 5);
		} else if(cupSum > 0){
			shelve = shelve - 1;
		}

		if(medalSum > 10){
			shelve = shelve - (int) Math.ceil((double)medalSum / 10);
		} else if(medalSum > 0){
			shelve = shelve - 1;
		}


		if(shelve >= 0) {
			ans = "YES";
		}



	}

	private void write() {
		
			out.println(ans);
		
	}

	public void run() {
		read();
		solve();
		write();
		out.close();
	}
}
