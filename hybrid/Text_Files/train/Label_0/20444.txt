#include <bits/stdc++.h>
using namespace std;
const int MN = 200111;
int nVer, nHor;
struct Seg {
  int x, y, l;
  void read() { cin >> x >> y >> l; }
} vers[MN], hors[MN];
struct Event {
  int typ;
  int x, u, v;
  Event(int typ = 0, int x = 0, int u = 0, int v = 0)
      : typ(typ), x(x), u(u), v(v) {}
} events[MN];
int nEvent;
bool operator<(const Event &a, const Event &b) {
  if (a.x != b.x) return a.x < b.x;
  if (a.typ != b.typ) return a.typ < b.typ;
  if (a.u != b.u) return a.u < b.u;
  return a.v < b.v;
}
bool check(int val) {
  nEvent = 0;
  for (int i = (1), _b = (nHor); i <= _b; i++)
    if (hors[i].l >= val * 2) {
      events[++nEvent] = Event(0, hors[i].x + val, hors[i].y);
      events[++nEvent] = Event(2, hors[i].x + hors[i].l - val, hors[i].y);
    }
  for (int i = (1), _b = (nVer); i <= _b; i++)
    if (vers[i].l >= val * 2) {
      events[++nEvent] =
          Event(1, vers[i].x, vers[i].y + val, vers[i].y + vers[i].l - val);
    }
  sort(events + 1, events + nEvent + 1);
  multiset<int> s;
  for (int i = (1), _b = (nEvent); i <= _b; i++) {
    if (events[i].typ == 0) {
      s.insert(events[i].u);
    } else if (events[i].typ == 2) {
      s.erase(s.find(events[i].u));
    } else {
      auto it = s.lower_bound(events[i].u);
      if (it != s.end() && *it <= events[i].v) return true;
    }
  }
  return false;
}
int main() {
  ios ::sync_with_stdio(false);
  cin.tie(NULL);
  cout << (fixed) << setprecision(6);
  while (cin >> nVer >> nHor) {
    for (int i = (1), _b = (nVer); i <= _b; i++) vers[i].read();
    for (int i = (1), _b = (nHor); i <= _b; i++) hors[i].read();
    int l = 0, r = 100111000 / 2, res = 0;
    while (l <= r) {
      int mid = (l + r) >> 1;
      if (check(mid)) {
        l = mid + 1;
        res = mid;
      } else
        r = mid - 1;
    }
    cout << res << endl;
  }
  return 0;
}
