(n, m) = list(map(int, input().split()))
a = []
for _ in range(n):
	a.append(list(map(int, input().split())))

def isMirror(l):
	y = len(l)
	if y % 2 != 0:
		return False
	y = y // 2
	lb = l[:y]
	le = l[y:]
	if le[::-1] == lb:
		return True
	else:
		return False
while isMirror(a):
	a = a[:len(a) // 2]
print(len(a))
