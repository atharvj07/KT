import java.util.*;
import java.io.*;
public class cf1045c{
static ArrayList<Integer>[] adj;
static int[] d;
static int[][] p;
static int[] dfn;
static int[] low;
static int[] stk;
static ArrayList<Integer>[] g;
static int n;
	public static void main(String[] args)throws IOException {
		//BufferedReader bf=new BufferedReader(new FileReader("cf1045c.in"));
		//PrintWriter out = new PrintWriter(new BufferedWriter(new FileWriter("cf1045c.out")));
		BufferedReader bf=new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
		StringTokenizer st=new StringTokenizer(bf.readLine());
		n=Integer.parseInt(st.nextToken());
		ncom=n-1;
		int m=Integer.parseInt(st.nextToken());
		int q=Integer.parseInt(st.nextToken());
		adj=new ArrayList[n];
		g=new ArrayList[2*n];
		d=new int[2*n];
		p=new int[2*n][20];
		dfn=new int[n];
		low=new int[n];
		stk=new int[n+1];
		for(int i=0;i<n;i++) {
			adj[i]=new ArrayList<Integer>();
		}
		for(int i=0;i<2*n;i++) {
			g[i]=new ArrayList<Integer>();
		}
		for(int i=0;i<m;i++) {
			st=new StringTokenizer(bf.readLine());
			int a1=Integer.parseInt(st.nextToken())-1;
			int a2=Integer.parseInt(st.nextToken())-1;
			adj[a1].add(a2);
			adj[a2].add(a1);
		}
		scc(0,0);
		dfs(0,0,0);
		for(int i=1;i<20;i++) {
			for(int j=0;j<2*n;j++) {
				p[j][i]=p[p[j][i-1]][i-1];
			}
		}
		for(int i=0;i<q;i++) {
			st=new StringTokenizer(bf.readLine());
			int a1=Integer.parseInt(st.nextToken())-1;
			int a2=Integer.parseInt(st.nextToken())-1;
			out.println((d[a1]+d[a2]-d[lca(a1,a2)]*2)/2);
		}
		out.close();
	}
	static int dnm;
	static int top;
	static int ncom;
	static void scc(int x, int par) {
		dfn[x]=low[x]=++dnm;
		stk[++top]=x;
		for(int i:adj[x]) {
			if(i!=par) {
				if(dfn[i]==0) {
					scc(i,x);
					low[x]=Math.min(low[x],low[i]);
					if(low[i]>=dfn[x]) {
						ncom++;
						while(stk[top]!=i) {
							g[ncom].add(stk[top]);
							g[stk[top]].add(ncom);
							top--;
						}
						g[ncom].add(i);
						g[i].add(ncom);
						top--;
						g[x].add(ncom);
						g[ncom].add(x);
					}
				}
				else {
					low[x]=Math.min(low[x],dfn[i]);
				}
			}
		}
	}
	static void dfs(int x, int par, int dep) {
		d[x]=dep;
		p[x][0]=par;
		for(int i:g[x]) {
			if(i!=par) {
				dfs(i,x,dep+1);
			}
		}
	}
	static int lca(int a,int b){
		for(int j=19;j>=0;j--){
			if(d[p[a][j]] >= d[b]){
				a = p[a][j];
			}
		}
		for(int j=19;j>=0;j--){
			if(d[p[b][j]] >= d[a]){
				b = p[b][j];
			}
		}
		for(int j=19;j>=0;j--){
			if(p[a][j]!=p[b][j]){
				a = p[a][j]; b = p[b][j];
			}
		}
		if(a==b) return a;
		return p[a][0];
	}
}
