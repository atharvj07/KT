shows = int(input())
array = []
for _ in range(shows):
	(start, end) = input().split()
	(start, end) = (int(start), int(end))
	array.append((start, 1))
	array.append((end + 1, -1))
array.sort()
count = 0
poss = True
for item in array:
	count += item[1]
	if count >= 3:
		poss = False
		break
if poss:
	print('YES')
else:
	print('NO')
