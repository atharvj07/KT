from sys import stdin
input = stdin.readline
from itertools import groupby, accumulate

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	m = max(a)
	for (i, v) in enumerate(a):
		if v == m and (i > 0 and v > a[i - 1] or (i < n - 1 and v > a[i + 1])):
			print(i + 1)
			return
	print(-1)
t = int(input())
for _ in range(t):
	solve()
