#include<iostream>
#include<string>
using namespace std;
string S[15] = { "Rock", "Fire", "Scissors", "Snake", "Human", "Tree", "Wolf", "Sponge", "Paper", "Air", "Water", "Dragon", "Devil", "Lightning", "Gun" };
int N, a[10000], c[15]; string T;
bool solve(int a, int b) {
	if (a > b)b += 15;
	if (b - a <= 7)return true;
	return false;
}
int main() {
	while (true) {
		cin >> N;
		if (N == 0)break;
		for (int i = 0; i < N; i++) {
			cin >> T;
			for (int j = 0; j < 15; j++) {
				if (T == S[j])a[i] = j;
			}
		}
		for (int i = 0; i < 15; i++)c[i] = 0;
		for (int i = 0; i < N; i++) {
			for (int j = 0; j < N; j++) {
				if (!solve(a[i], a[j]))goto E;
			}
			c[a[i]] = 1;
		E:;
		}
		int G = 0;
		for (int i = 0; i < N - 1; i++) {
			if (a[i] != a[i + 1])G = 1;
		}
		int P = 0;
		if (G == 0)goto F;
		for (int i = 0; i < 15; i++) {
			if (c[i] == 1) {
				P = 1; cout << S[i] << endl;
			}
		}
	F:;
		if (P == 0)cout << "Draw" << endl;
	}
	return 0;
}
