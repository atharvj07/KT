(n, l, r) = map(int, input().split())
a = list(map(int, input().split()))
p = list(map(int, input().split()))
ans = [0] * n
order = [0] * n
for i in range(n):
	order[p[i] - 1] = i
cur = -10 ** 19
for o in order:
	bi = max(l, cur + 1 + a[o])
	if bi > r:
		print(-1)
		exit()
	ans[o] = bi
	cur = bi - a[o]
print(*ans)
