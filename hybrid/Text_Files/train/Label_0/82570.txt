def func(n, arr):
	sortedArr = sorted(arr)
	l = -1
	r = -1
	for i in range(n):
		if arr[i] != sortedArr[i]:
			l = i
			break
	if l == -1:
		print('yes')
		print(1, 1)
		return
	for j in range(n - 1, -1, -1):
		if arr[j] != sortedArr[j]:
			r = j
			break
	low = l
	high = r
	while r >= l:
		(arr[l], arr[r]) = (arr[r], arr[l])
		l += 1
		r -= 1
	for k in range(n):
		if arr[k] != sortedArr[k]:
			print('no')
			return
	print('yes')
	print(low + 1, high + 1)
n = int(input())
arr = list(map(int, input().split()))
func(n, arr)
