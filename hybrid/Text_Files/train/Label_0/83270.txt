a = input()
b = a.split()
x = int(b[0])
y = int(b[1])
import math
search = []
for i in range(max(2, y - 1000, x), y + 1):
	search.append(i)
search = search[::-1]
check = 0
searcher = 0
truefail = 0
primelist = []
if x > 31600:
	x = 31600
fail = 0
for i in range(2, x + 1):
	for j in primelist:
		fail = 0
		if i % j == 0:
			fail = 1
			break
	if fail == 0:
		primelist.append(i)
while check == 0:
	fail = 0
	z = min(x + 1, math.floor(math.sqrt(search[searcher]) + 10))
	for i in range(2, z):
		if search[searcher] % i == 0:
			fail = 1
			break
	if fail == 0:
		check = 1
	else:
		searcher += 1
	if searcher > 1000 or searcher == len(search):
		truefail = 1
		break
if truefail == 1:
	print(-1)
else:
	print(search[searcher])
