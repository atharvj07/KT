#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000100;
const int INF = 1e9 + 10;
int n;
int x[maxn];
int y[maxn], m;
int a, b;
int main() {
  while (cin >> n) {
    for (int i = 1; i <= n; i++) scanf("%d", &x[i]);
    sort(x + 1, x + n + 1);
    n = unique(x + 1, x + n + 1) - (x + 1);
    cin >> a >> b;
    if (a == b) {
      puts("0");
      continue;
    }
    int res = 0;
    while (1) {
      int na = a - 1;
      for (int i = 1; i <= n; i++) {
        int t = a - a % x[i];
        if (t >= b) na = min(na, t);
      }
      res++;
      if (na == b) break;
      a = na;
      m = 0;
      for (int i = 1; i <= n; i++)
        if (a - a % x[i] >= b) y[++m] = x[i];
      for (int i = 1; i <= m; i++) x[i] = y[i];
      n = m;
    }
    cout << res << endl;
  }
  return 0;
}
