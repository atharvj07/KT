#include <bits/stdc++.h>
using namespace std;
vector<int> GetPrefixVector(vector<int>& input, int val) {
  vector<int> pre(input.size());
  pre[0] = (input[0] == val) ? 1 : 0;
  for (int i = 1; i < input.size(); ++i) {
    pre[i] = pre[i - 1] + (input[i] == val ? 1 : 0);
  }
  return pre;
}
int GetIndex(vector<int>& p, int v) {
  int low = 0, high = p.size() - 1;
  while (low < high) {
    int mid = (low + high) / 2;
    if (p[mid] == v) {
      high = mid;
    } else if (p[mid] < v) {
      low = mid + 1;
    } else {
      high = mid - 1;
    }
  }
  return p[low] == v ? low : p.size();
}
int GetSForT(vector<int>& p1, vector<int>& p2, int t) {
  int n = p1.size();
  int awins = 0, bwins = 0;
  int asum = 0, bsum = 0;
  int index = 0;
  while (index < n) {
    index = min(GetIndex(p1, asum + t), GetIndex(p2, bsum + t));
    if (index >= n) {
      return -1;
    }
    if (p1[index] == asum + t) {
      awins++;
    }
    if (p2[index] == bsum + t) {
      bwins++;
    }
    if (index == n - 1) {
      if (p1[index] == asum + t && awins > bwins) return awins;
      if (p2[index] == bsum + t && bwins > awins) return bwins;
      return -1;
    }
    asum = p1[index];
    bsum = p2[index];
  }
}
int main() {
  int n, temp;
  cin >> n;
  vector<int> input;
  input.resize(n);
  for (int i = 0; i < n; ++i) {
    cin >> temp;
    input[i] = temp;
  }
  vector<int> prefix1 = GetPrefixVector(input, 1);
  vector<int> prefix2 = GetPrefixVector(input, 2);
  vector<pair<int, int> > ans;
  for (int t = 1; t <= n; ++t) {
    int s = GetSForT(prefix1, prefix2, t);
    if (s != -1) {
      ans.push_back(make_pair(s, t));
    }
  }
  sort(ans.begin(), ans.end());
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); ++i) {
    cout << ans[i].first << " " << ans[i].second << endl;
  }
  return 0;
}
