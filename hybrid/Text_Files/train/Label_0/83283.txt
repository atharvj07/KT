(a, b) = map(int, input().split())
t = a + b
cnt = 0
for i in range(10 ** 9):
	cnt += 1
	if cnt * (cnt + 1) // 2 > t:
		cnt -= 1
		break
firstday = []
secondday = []
for x in range(cnt, 0, -1):
	if a >= x:
		firstday.append(x)
		a -= x
	else:
		secondday.append(x)
		b -= x
print(len(firstday))
print(*firstday)
print(len(secondday))
print(*secondday)
