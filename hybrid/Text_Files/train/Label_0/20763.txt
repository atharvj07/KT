#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int x, y, a, b, i, j;
  cin >> x >> y >> a >> b;
  vector<pair<int, int>> v;
  for (i = a; i <= x; i++) {
    for (j = b; j <= y; j++) {
      if (j < i) {
        v.push_back(make_pair(i, j));
      }
    }
  }
  cout << v.size() << '\n';
  if (v.size() != 0) {
    sort(v.begin(), v.end());
    for (i = 0; i < v.size(); i++)
      cout << v[i].first << " " << v[i].second << '\n';
  }
}
