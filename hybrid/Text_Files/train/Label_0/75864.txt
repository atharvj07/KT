import sys
inp = sys.stdin.readlines()
(n, m) = [int(i) for i in inp[0].split()]
d = [[] for i in range(n + 1)]
for j in range(1, m + 1):
	(a, b) = [int(i) for i in inp[j].split()]
	d[a].append(b)
	d[b].append(a)
fnd = [0] * (n + 1)
comps = []
for i in range(1, n + 1):
	if not fnd[i]:
		comp = [i]
		fnd[i] = 1
		st = [i]
		while st:
			v = st.pop()
			for vv in d[v]:
				if not fnd[vv]:
					fnd[vv] = 1
					st.append(vv)
					comp.append(vv)
		comps.append((min(comp), max(comp)))
comps.sort()
cnt = 1
ub = comps[0][1]
for cp in comps:
	if cp[0] > ub:
		cnt += 1
	ub = max(ub, cp[1])
print(len(comps) - cnt)
