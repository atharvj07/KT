n = int(input())
arr = []
for _ in range(n):
	(a, b, c) = map(int, input().split())
	arr.append([a, b, c] + [_ + 1])
arr.sort()
ans = []
for i in range(2, -1, -1):
	st = []
	for x in arr:
		if st and all((st[-1][itr] <= x[itr] for itr in range(i + 1))):
			ans.append((st.pop()[-1], x[-1]))
		else:
			st.append(x)
	arr = st
for i in ans:
	print(i[0], i[1])
