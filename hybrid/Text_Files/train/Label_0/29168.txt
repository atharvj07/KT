#include <bits/stdc++.h>
using namespace std;
ifstream in("CGCDSSQ.in");
ofstream out("CGCDSSQ.out");
const int Nmax = 100005;
int n, q, last;
int logaritm[Nmax], dp[20][Nmax];
unordered_map<int, long long> nr;
int euclid(int a, int b) {
  int r;
  while (b) {
    r = a % b;
    a = b;
    b = r;
  }
  return a;
}
int main() {
  cin >> n;
  for (int i = 2; i <= n; i++) logaritm[i] = logaritm[i / 2] + 1;
  for (int i = 1; i <= n; i++) cin >> dp[0][i], nr[dp[0][i]]++;
  for (int i = 1; (1 << i) <= n; i++)
    for (int j = 1; j <= n - (1 << i) + 1; j++) {
      int k = 1 << (i - 1);
      dp[i][j] = euclid(dp[i - 1][j], dp[i - 1][j + k]);
    }
  for (int i = 1; i <= n; i++) {
    last = i;
    int val = dp[0][i];
    int st = i;
    int dr = n;
    int mij;
    while (st <= n) {
      while (st <= dr) {
        mij = (st + dr) / 2;
        int dist = logaritm[mij - i + 1];
        if (euclid(dp[dist][i], dp[dist][mij - (1 << dist) + 1]) == val)
          st = mij + 1;
        else
          dr = mij - 1;
      }
      int aux = logaritm[mij - i + 1];
      if (euclid(dp[aux][i], dp[aux][mij - (1 << aux) + 1]) == val) {
        nr[val] = nr[val] + mij - last;
        last = mij;
        st = mij + 1;
        aux = logaritm[mij - i + 2];
        val = euclid(dp[aux][i], dp[aux][mij - (1 << aux) + 2]);
      } else {
        nr[val] = nr[val] + mij - last - 1;
        last = mij - 1;
        st = mij;
        aux = logaritm[mij - i + 1];
        val = euclid(dp[aux][i], dp[aux][mij - (1 << aux) + 1]);
      }
      dr = n;
    }
  }
  cin >> q;
  for (int i = 1; i <= q; i++) {
    int x;
    cin >> x;
    cout << nr[x] << '\n';
  }
  return 0;
}
