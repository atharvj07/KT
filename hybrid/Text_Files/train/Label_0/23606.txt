#include <bits/stdc++.h>
using namespace std;
int N;
map<long long, int> mp;
int main() {
  ios::sync_with_stdio(0);
  cin >> N;
  long long prefix = 0;
  for (int i = 0, a; i < N; i++) {
    cin >> a;
    prefix += a;
    mp[prefix]++;
  }
  int best = 0;
  for (map<long long, int>::iterator it = mp.begin(); it != mp.end(); it++)
    if (it->second > best) best = it->second;
  cout << N - best << "\n";
  return 0;
}
