#include<bits/stdc++.h>
using namespace std;

const int N = 1e6+7, M = 1000000007;
int cnt[N];

int main() {
    cnt[0] = 3;

    long long ways = 1;
    int n;
    cin>>n;


    for (int i=1; i<=n; i++) {
        int x;
        cin>>x;

        x++;
        int rem = cnt[x-1] - cnt[x];
        if (rem <= 0) {
            ways = 0;
            break;
        }
        cnt[x]++;
        ways = (ways*rem)%M;
    }
    cout<<ways<<endl;
}
