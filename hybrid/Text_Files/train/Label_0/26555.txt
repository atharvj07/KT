#include <iostream>
#include <utility>
#include <stdio.h>
#include <math.h>
#include <algorithm>
#include <vector>
#include <map>
#include <set>
#include <stdlib.h>
#define llint long long
#define mod 998244353
#define eps 1e-8
#define inf 1e18

using namespace std;
typedef pair<llint, llint> P;

llint n, m;
vector<llint> G[200005], revG[200005];
vector<llint> g[200005];
int x[200005], y[200005], z[200005];
vector<llint> vec[200005];
P cand[200005][8];
llint ans[200005];

vector<int> topo;
bool used[200005];
int scc[200005];

void tpsort(int v)
{
	used[v] = true;
	for(int i = 0; i < G[v].size(); i++){
		if(!used[G[v][i]]) tpsort(G[v][i]);
	}
	topo.push_back(v);
}
void sccdfs(int v, int id)
{
	used[v] = true;
	scc[v] = id;
	for(int i = 0; i < revG[v].size(); i++){
		if(!used[revG[v][i]]) sccdfs(revG[v][i], id);
	}
}

llint dist(llint i, llint j)
{
	return abs(x[i]-x[j]) + abs(y[i]-y[j]) + abs(z[i]-z[j]);
}

int main(void)
{
	ios::sync_with_stdio(0);
	cin.tie(0);
	
	cin >> n >> m;
	for(int i = 1; i <= n; i++){
		cin >> x[i] >> y[i] >> z[i];
	}
	llint u, v;
	for(int i = 0; i < m; i++){
		cin >> u >> v;
		G[u].push_back(v);
		revG[v].push_back(u);
	}
	
	for(int i = 1; i <= n; i++) if(!used[i]) tpsort(i);
	reverse(topo.begin(), topo.end());
	
	int id = 0;
	for(int i = 1; i <= n; i++) used[i] = false;
	for(int i = 0; i < topo.size(); i++) if(!used[topo[i]]) sccdfs(topo[i], id++);
	for(int i = 1; i <= n; i++) vec[scc[i]].push_back(i);
	
	set<llint> S;
	for(int i = 0; i < id; i++){
		S.clear();
		for(int j = 0; j < vec[i].size(); j++){
			for(int k = 0; k < G[vec[i][j]].size(); k++){
				S.insert(scc[G[vec[i][j]][k]]);
			}
		}
		for(auto it = S.begin(); it != S.end(); it++){
			if(*it == i) continue;
			g[i].push_back(*it);
		}
	}
	
	/*for(int i = 0; i < id; i++){
		for(int j = 0; j < g[i].size(); j++){
			cout << g[i][j] << " ";
		}
		cout<< endl;
	}*/
	
	for(int k = id-1; k >= 0; k--){
		for(int i = 0; i < 8; i++) cand[k][i] = make_pair(-inf, -inf);
		for(int i = 0; i < 8; i++){
			for(llint j = 0; j < vec[k].size(); j++){
				llint v = vec[k][j];
				llint f = 0;
				if(i&1) f += x[v]; else f -= x[v];
				if(i&2) f += y[v]; else f -= y[v];
				if(i&4) f += z[v]; else f -= z[v];
				cand[k][i] = max(cand[k][i], make_pair(f, v));
			}
		}
	}
	
	for(int k = id-1; k >= 0; k--){
		for(int j = 0; j < g[k].size(); j++){
			int nk = g[k][j];
			for(int i = 0; i < 8; i++) cand[k][i] = max(cand[k][i], cand[nk][i]);
		}
		for(int i = 0; i < 8; i++){
			for(llint j = 0; j < vec[k].size(); j++){
				int v = vec[k][j];
				//cout << v << " " << cand[i].second << endl;
				ans[v] = max(ans[v], dist(v, cand[k][i].second));
			}
		}
	}
	
	for(int i = 1; i <= n; i++){
		cout << ans[i] << "\n";
	}
	flush(cout);
	
	return 0;
}
