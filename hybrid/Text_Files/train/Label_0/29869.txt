#include <bits/stdc++.h>
using namespace std;
int n, m, k;
int main() {
  while (~scanf("%d%d%d", &n, &m, &k)) {
    if (k >= m)
      printf("1\n");
    else if (n + k < m)
      printf("0\n");
    else {
      double ans = 1;
      for (int i = 1; i <= k + 1; i++) ans = ans * (m - i + 1) / (n + i);
      ans = 1.0 - ans;
      printf("%.6f\n", ans);
    }
  }
}
