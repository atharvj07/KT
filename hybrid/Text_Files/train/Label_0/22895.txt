#include <cstdio>
#include <cstring>
#define ll long long

ll a,b,c,d;
ll gcd(ll x,ll y){return y==0?x:gcd(y,x%y);}
ll solve(ll n)
{
	ll ans=n;
	ans-=n/c;
	ans-=n/d;
	ans+=n/(c*d/gcd(c,d));
	return ans;
}

int main()
{
	scanf("%lld %lld %lld %lld",&a,&b,&c,&d);
	printf("%lld",solve(b)-solve(a-1));
}
