import java.util.Arrays;
import java.util.Scanner;

public final class CodeForces {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while(t-->0) {
            int[] arr = new int[2];
            arr[0] = in.nextInt();
            arr[1] = in.nextInt();
            Arrays.sort(arr);
            arr[1] -= arr[0];
            arr[0] = 0;
            System.out.println(solve(arr));
        }
        in.close();
    }

    private static long solve(int[] arr) {
        if(arr[1] == 0)
            return 0;
        long num = 0, sum = 0;
        while(sum < arr[1]) {
            sum += num;
            num++;
        }
        if(sum%2 != arr[1]%2) {
            while(sum % 2 != arr[1] % 2) {
                sum += num;
                num++;
            }
        }
        return num - 1 > 0 ? num - 1 : 0;
    }
}
