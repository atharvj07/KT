from collections import defaultdict

def main(t1, t2, n, t, ha, m, yr):
	reds = {}
	yellows = {}
	teams = [t1, t2]
	for i in range(n):
		player = (ha[i], m[i])
		if player not in reds:
			if player not in yellows:
				yellows[player] = 0
			if yr[i] == 'y':
				yellows[player] += 1
			if yellows[player] >= 2 or yr[i] == 'r':
				reds[player] = 1
				print(teams[ha[i] == 'a'], m[i], t[i])

def main_input():
	t1 = input()
	t2 = input()
	n = int(input())
	t = list(range(n))
	ha = list(range(n))
	m = list(range(n))
	yr = list(range(n))
	for i in range(n):
		(t[i], ha[i], m[i], yr[i]) = input().split()
	t = [int(x) for x in t]
	m = [int(x) for x in m]
	main(t1, t2, n, t, ha, m, yr)
main_input()
