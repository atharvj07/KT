import sys
sys.setrecursionlimit(10 ** 9)
input = sys.stdin.readline

def solve():
	n = int(input())
	ans = set()
	ans.add(0)
	ans.add(1)
	ans.add(n)
	i = 2
	while i * i <= n:
		if n / i == n // i:
			ans.add(int(i))
			ans.add(int(n // i))
		else:
			ans.add(int(i))
			ans.add(n // i)
		i += 1
	l = sorted(list(ans))
	print(len(ans))
	print(' '.join(map(str, l)))
T = int(input())
for _ in range(T):
	solve()
