import sys
(n, a) = map(int, sys.stdin.readline().split())
t = list(map(int, sys.stdin.readline().split()))
ans = t[a - 1]
if a <= n // 2:
	for i in range(a - 1):
		if t[i] + t[2 * (a - 1) - i] == 2:
			ans += 2
	ans += sum(t[2 * a - 1:])
else:
	for i in range(a, n):
		if t[i] + t[2 * (a - 1) - i] == 2:
			ans += 2
	ans += sum(t[:2 * a - n - 1])
print(ans)
