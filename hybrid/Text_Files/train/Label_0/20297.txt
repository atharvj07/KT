#include <bits/stdc++.h>
using namespace std;
const long long modulo = 1000000007;
long long PowMod(long long a, long long b, long long p) {
  long long ret = 1;
  while (b) {
    if (b & 1) ret = (ret * a) % p;
    a = (a * a) % p;
    b >>= 1;
  }
  return ret;
}
long long Rev(long long a, long long p) { return PowMod(a, p - 2, p); }
int main() {
  long long n, C;
  cin >> n;
  C = 1;
  for (int i = 1; i < n; i++)
    C = C * (2 * n - i) % modulo * Rev(i, modulo) % modulo;
  C = ((C * 2 - n) % modulo + modulo) % modulo;
  cout << C << endl;
  return 0;
}
