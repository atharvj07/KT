#include <bits/stdc++.h>
using namespace std;
int n, m, k;
char A[106][106];
char c[63];
int re;
vector<char> vec;
int main() {
  int T;
  cin >> T;
  for (int i = 0; i < 26; ++i) c[i] = i + 'a';
  for (int i = 0; i < 26; ++i) c[i + 26] = i + 'A';
  for (int i = 0; i < 10; ++i) c[i + 52] = i + '0';
  while (T--) {
    scanf("%d%d%d", &n, &m, &k);
    re = 0;
    for (int i = 1; i <= n; ++i) {
      scanf("%s", A[i] + 1);
      for (int j = 1; j <= m; ++j) {
        if (A[i][j] == 'R') ++re;
      }
    }
    int ok = re / k, las = 0, cu = 0;
    for (int i = 1; i <= n; ++i) {
      vec.clear();
      if (i & 1) {
        for (int j = 1; j <= m; ++j) {
          if (A[i][j] == 'R') ++las;
          vec.push_back(c[cu]);
          if (las == ok + 1 && cu < re % k && cu < k - 1)
            ++cu, las = 0;
          else if (las == ok && cu >= re % k && cu < k - 1)
            ++cu, las = 0;
        }
      } else {
        for (int j = m; j >= 1; --j) {
          if (A[i][j] == 'R') ++las;
          vec.push_back(c[cu]);
          if (las == ok + 1 && cu < re % k && cu < k - 1)
            ++cu, las = 0;
          else if (las == ok && cu >= re % k && cu < k - 1)
            ++cu, las = 0;
        }
        reverse(vec.begin(), vec.end());
      }
      for (auto it : vec) putchar(it);
      putchar(10);
    }
  }
}
