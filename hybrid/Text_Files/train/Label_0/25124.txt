#include <bits/stdc++.h>
using namespace std;
long long a, b, w, x, c;
long long solve() {
  if (c <= a)
    return 0;
  else {
    long long det = c - a;
    double upp = det * x - b;
    double dow = w - x;
    long long lam = (long long)ceil(upp / dow);
    return lam + det;
  }
}
int main() {
  scanf("%I64d%I64d%I64d%I64d%I64d", &a, &b, &w, &x, &c);
  printf("%I64d\n", solve());
  return 0;
}
