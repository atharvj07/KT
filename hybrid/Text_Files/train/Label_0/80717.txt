for _ in range(int(input())):
	(x, y) = map(int, input().split())
	up = 0
	dow = 0
	rig = 0
	lef = 0
	posible = True
	for i in input():
		if i == 'U' or i == 'D':
			if i == 'U':
				up += 1
			else:
				dow += 1
		elif i == 'R':
			rig += 1
		else:
			lef += 1
	sumax = rig + lef * -1
	while sumax != x and posible == True:
		if x < sumax and rig != 0:
			sumax -= 1
			rig -= 1
		elif x > sumax and lef != 0:
			sumax += 1
			lef -= 1
		else:
			posible = False
	sumay = up + dow * -1
	while sumay != y and posible == True:
		if y < sumay and up != 0:
			sumay -= 1
			up -= 1
		elif y > sumay and dow != 0:
			sumay += 1
			dow -= 1
		else:
			posible = False
	if posible == True:
		print('YES')
	else:
		print('NO')
