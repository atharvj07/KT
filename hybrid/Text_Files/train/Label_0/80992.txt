from math import ceil
for ad in range(int(input())):
	n = int(input())
	print(n + 5)
	m = n + 5
	c = 0
	while True:
		k = ceil(n ** 0.5)
		if n == 2:
			break
		for i in range(n - 1, 0, -1):
			if i == k:
				print(n, i)
				print(n, i)
				n = k
				c += 2
				break
			else:
				print(i, n)
				c += 1
	for i in range(m - c):
		print(2, 1)
