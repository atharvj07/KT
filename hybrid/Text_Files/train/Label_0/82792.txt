(n, start) = map(int, input().split())
x = list(map(int, input().split()))
x = sorted(x)
if n == 1:
	print(0)
else:
	print(min(abs(start - x[0]) + abs(x[len(x) - 2] - x[0]), abs(start - x[1]) + abs(x[len(x) - 1] - x[1]), abs(x[len(x) - 1] - start) + abs(x[len(x) - 1] - x[1]), abs(x[len(x) - 2] - start) + abs(x[len(x) - 2] - x[0])))
