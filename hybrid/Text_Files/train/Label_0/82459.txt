n = int(input())
l = list(map(int, input().split()))
l1 = [[0, 0] for i in range(n + 1)]
p = -1
t = -1
v = -1
k = 0
y = -1
for i in range(n):
	if l[i] == 0 and t == -1 and (v == -1):
		v = i + 1
	elif l[i] == 0 and t == 1:
		if v != -1 and v < p:
			y = i + 1
		else:
			v = i + 1
	elif l[i] != 0 and t == -1:
		p = i + 1
		t = 1
	elif l[i] != 0 and t == 1:
		if v != -1 and y != -1:
			l1[i][0] = min(v, y)
			l1[i][1] = max(v, y)
		if v != -1 and y == -1:
			l1[i][0] = min(v, p)
			l1[i][1] = max(v, p)
		if v == -1 and y == -1:
			l1[i][0] = p
			l1[i][1] = p
		k = k + 1
		p = i + 1
		v = -1
		y = -1
		t = 1
if p != -1:
	if v != -1 and y != -1:
		l1[i + 1][0] = min(v, y)
		l1[i + 1][1] = max(v, y)
	if v != -1 and y == -1:
		l1[i + 1][0] = min(p, v)
		l1[i + 1][1] = max(p, v)
	if v == -1 and y == -1:
		l1[i + 1][0] = p
		l1[i + 1][1] = p
	k = k + 1
if k >= 1:
	print('YES')
	print(k)
	for i in range(n + 1):
		if sum(l1[i]) != 0:
			print(l1[i][0], l1[i][1])
else:
	print('NO')
