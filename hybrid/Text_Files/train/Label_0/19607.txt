#include <iostream>
#include <algorithm>
#include <cstdio>
#define INF 100000000
#define t first
#define x second
using namespace std;
typedef pair <double,int> P;
double b,v,e,f;
int r;
double mk(int x) {
  if(x >= r) return 1.0/(v-e*(x-r));
  return  1.0/(v-f*(r-x));
}

int main() {
  while(1) {
    int n,a[102]={};
    cin >> n;
    if(n==0) break;
    for(int i=1;i<=n;i++) cin >> a[i];
    cin >> b >> r >> v >> e >> f;
    P dp[10010]={};
    for(int i=1;i<=a[n];i++) dp[i] = make_pair(INF,0);
    
    for(int i=0;i<n;i++) {
      int nx=dp[a[i]].x,cx= 0;
      double nt=dp[a[i]].t,ct=dp[a[i]].t+b;

      for(int j=a[i]+1;j<=a[n];j++){
	nt += mk(nx),ct += mk(cx);
	dp[j] = min(dp[j],min((P){nt,++nx},(P){ct,++cx}));
      }
    }
    printf("%.4f\n",dp[a[n]].t);
  }
  return 0;
}
