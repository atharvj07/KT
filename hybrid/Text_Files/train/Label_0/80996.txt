o = input().rstrip().split(':')
p = input().rstrip().split(':')
f = 0
if int(o[0]) > int(p[0]):
	f = f + (13 - int(o[0])) * 60 - int(o[1])
	f = f + (int(p[0]) - 1) * 60 + int(p[1])
else:
	f = abs(f + abs(int(p[0]) - int(o[0])) * 60 + (int(p[1]) - int(o[1])))
g = f // 2
V = g // 60
h = g - V * 60
A = str(int(o[0]) + V)
B = str(int(o[1]) + h)
if int(B) >= 60:
	C = int(B) // 60
	A = str(int(A) + int(B) // 60)
	B = str(int(B) - int(C) * 60)
X = list(A)
Y = list(B)
if len(X) == 1:
	X.insert(0, '0')
if len(Y) == 1:
	Y.insert(0, '0')
C = ''.join(X)
D = ''.join(Y)
print(C, end='')
print(':', end='')
print(D)
