import java.util.*;
import java.io.*;

   public class E
   {
      public static void main(String omkar[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));  
         StringTokenizer st = new StringTokenizer(infile.readLine());
         int N = Integer.parseInt(st.nextToken());
         int[] arr = new int[N];
         st = new StringTokenizer(infile.readLine());
         for(int i=0; i < N; i++)
            arr[i] = Integer.parseInt(st.nextToken());
         //convex hoe
         ArrayList<Integer> lsX = new ArrayList<Integer>();
         ArrayList<Long> lsY = new ArrayList<Long>();
         lsX.add(0);
         lsY.add(0L);
         long sum = 0L;
         for(int i=1; i <= N; i++)
         {
            sum += arr[i-1];
            if(lsX.size() == 1)
            {
               lsX.add(i);
               lsY.add(sum);
               continue;
            }
            int M = lsX.size();
            long prev = (lsY.get(M-1)-lsY.get(M-2))*(i-lsX.get(M-1));
            long curr = (sum-lsY.get(M-1))*(lsX.get(M-1)-lsX.get(M-2));
            while(M >= 2 && prev > curr)
            {
               lsX.remove(M-1);
               lsY.remove(M-1);
               M--;
               if(M < 2)
                  break;
               prev = (lsY.get(M-1)-lsY.get(M-2))*(i-lsX.get(M-1));
               curr = (sum-lsY.get(M-1))*(lsX.get(M-1)-lsX.get(M-2));
            }
            lsX.add(i);
            lsY.add(sum);
         }
         int M = lsX.size();
         StringBuilder sb = new StringBuilder();
         for(int i=0; i < M-1; i++)
         {
            long len = lsX.get(i+1)-lsX.get(i);
            long sum1 = lsY.get(i+1)-lsY.get(i);
            double val = 1.0*sum1/len;
            for(int a=0; a < len; a++)
               sb.append(val).append(" ");
         }
         System.out.println(sb);
      }
   }
