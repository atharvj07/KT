#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

string make_num(int p, long long N, int l){
  string ret = to_string(p);
  int q = -1;
  for(int i = 0; i < p; ++i){
    if(N >= (1LL<<(l-2))) N -= (1LL<<(l-2));
    else{
      q = i;
      break;
    }
  }
  if(q < 0){
    if(N < 9*((1LL<<(l-2))-1)) return ret + make_num(p, N, l-1);
    N -= 9*((1LL<<(l-2))-1);
    for(int i = p+1; i < 10; ++i){
      if(N >= (1LL<<(l-2))) N -= (1LL<<(l-2));
      else{
        q = i;
        break;
      }
    }
  }
  ret += to_string(q);
  if(p < q) swap(p,q);
  string ret_(l-2,'0');
  for(int i = 0; i < l-2; ++i){
    if((N>>i)&1LL) ret_[i] += p;
    else ret_[i] += q;
  }
  reverse(ret_.begin(), ret_.end());
  ret += ret_;
  return ret;
}

int main(){
  long long N;
  while(cin >> N, N){
    int ng = 0, ok = 64;
    while(ok - ng > 1){
      int m = (ng + ok)/2;
      if(N <= 81*((1LL<<m) - 1 - m))
        ok = m;
      else ng = m;
    }
    //cout << ok << endl;
    N -= 81*((1LL<<(ok-1)) - ok) + 1;
    int p = N/(9*((1LL<<(ok-1))-1)) + 1;
    N %= (9*((1LL<<(ok-1))-1));
    cout << make_num(p,N,ok) << endl;
  }
  return 0;
}
