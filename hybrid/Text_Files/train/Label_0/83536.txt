(n, m) = input().split()
n = int(n)
m = int(m)
table = []
line = [0, 0]
column = [0, 0]
sets = 0
for i in range(n):
	table.append([int(i) for i in input().split()])
	for k in table[i]:
		if k == 0:
			line[0] += 1
		else:
			line[1] += 1
	sets += 2 ** line[0] - 1 + 2 ** line[1] - 1
	line = [0, 0]
for i in range(m):
	for k in range(n):
		if table[k][i] == 0:
			column[0] += 1
		else:
			column[1] += 1
	sets += 2 ** column[0] - 1 + 2 ** column[1] - 1
	column = [0, 0]
sets -= m * n
print(sets)
