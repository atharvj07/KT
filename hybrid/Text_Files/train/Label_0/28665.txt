#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
string s;
int n;

int main()
{
	cin>>s;
	n=s.size();
	if(s[n-1]=='1'||s[0]=='0')
	{
		cout<<-1<<endl;
		return 0;
	}
	for(int i=0;i<n-1;i++)
	{
		if(s[i]!=s[n-i-2])
		{
			cout<<-1<<endl;
			return 0;
		}
	}
	int p=1;
	for(int i=n-1;i--;)
	{
		cout<<p<<" "<<(n-i)<<endl;
		if(s[i]=='1')
		{
			p=n-i;
		}
	}
}
