import math
(hh, mm) = map(int, input().split())
(H, D, C, N) = map(int, input().split())
if hh < 20:
	i = math.ceil((((20 - hh - 1) * 60 + (60 - mm)) * D + H) / N) * C * 0.8
	j = math.ceil(H / N) * C
	if i > j:
		print('%.4f' % j)
	else:
		print('%.4f' % i)
else:
	k = math.ceil(H / N) * C * 0.8
	print('%.4f' % k)
