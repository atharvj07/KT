#include <bits/stdc++.h>
using namespace std;
int n, a, b;
struct tt {
  int x, y;
};
tt t[101];
int main() {
  while (cin >> n >> a >> b) {
    if (a > b) swap(a, b);
    for (int i = 0; i < n; i++) {
      cin >> t[i].x >> t[i].y;
      if (t[i].x > t[i].y) swap(t[i].x, t[i].y);
    }
    int res = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        if (i == j) continue;
        bool ok = false;
        int p1, p2;
        p1 = t[i].x + t[j].x;
        p2 = max(t[i].y, t[j].y);
        if (p1 <= a && p2 <= b) ok = true;
        if (p1 <= b && p2 <= a) ok = true;
        p1 = t[i].x + t[j].y;
        p2 = max(t[i].y, t[j].x);
        if (p1 <= a && p2 <= b) ok = true;
        if (p1 <= b && p2 <= a) ok = true;
        p1 = t[i].y + t[j].y;
        p2 = max(t[i].x, t[j].x);
        if (p1 <= a && p2 <= b) ok = true;
        if (p1 <= b && p2 <= a) ok = true;
        p1 = t[i].y + t[j].x;
        p2 = max(t[i].x, t[j].y);
        if (p1 <= a && p2 <= b) ok = true;
        if (p1 <= b && p2 <= a) ok = true;
        if (ok) res = max(res, t[i].x * t[i].y + t[j].x * t[j].y);
      }
    }
    cout << res << endl;
  }
  return 0;
}
