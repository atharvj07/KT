#include<bits/stdc++.h>
#define int long long
using namespace std;
int n,res=1;
signed main(){
	scanf("%lld",&n);
	while(n--){
		int a; scanf("%lld",&a);
		if(a==0) return puts("0"),0;
		else if(1000000000000000000/a<res) res=-1;
		if(res!=-1) res*=a;
		if(res>1000000000000000000) res=-1;
	}
	printf("%lld",res);
	return 0;
}
