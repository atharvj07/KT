    public static long[] generate(long mod){
        long[] arr={2, 9, 1, 4, 8, 10, 0, 2, 8, 7, 6};
        long[] ans=new long[11];
        for (int i = 0; i <11 ; i++) {
            for (int j = 0; j <11 ; j++) {
                ans[i]+=bin_pow(i,j,mod)*arr[j];
                ans[i]%=mod;
            }
        }
        System.out.println(Arrays.toString(ans));
        return ans;
    }
    public static long bin_pow(long a,long b, long mod){//a^b %mod
        long ans=1;
        while(b!=0){
            if((b&1)==1) ans*=a;
            a*=a;
            ans%=mod;
            a%=mod;
            b>>=1;
        }
        return ans;
    }
