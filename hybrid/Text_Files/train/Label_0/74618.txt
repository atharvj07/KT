from sys import stdin

def verificar(s, t):
	ls = len(s)
	if len(t) < ls:
		return False
	temp = ''
	j = 0
	for i in t:
		if j < ls and i == s[j]:
			temp = s[j]
			j += 1
		elif i != temp:
			return False
	if ls == j:
		return True
	else:
		return False

def main():
	n = int(stdin.readline().strip())
	for _ in range(n):
		s = stdin.readline().strip()
		t = stdin.readline().strip()
		if verificar(s, t):
			print('YES')
		else:
			print('NO')
main()
