import sys
n = ''
k = l = 0
d = [0] * 9

def dfs(cur, dset):
	p = len(cur)
	if p == l:
		return cur
	if cur > n:
		nd = 0
	else:
		nd = int(n[p])
	for i in range(nd, 10):
		if i in dset:
			r = dfs(cur + str(i), dset)
			if r != -1:
				return r
		elif len(dset) < k:
			r = dfs(cur + str(i), dset | set([i]))
			if r != -1:
				return r
	return -1
for _ in range(int(sys.stdin.readline())):
	(n, k) = sys.stdin.readline().split()
	k = int(k)
	l = len(n)
	sys.stdout.write(dfs('', set()) + '\n')
