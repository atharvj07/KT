import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;

/**
 *
 * @author Anextro
 */
public class Main {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new PrintWriter(System.out));
        
        String[] ln1 = br.readLine().split(" ");
        int x1 = Integer.parseInt(ln1[0]);
        int y1 = Integer.parseInt(ln1[1]);
        int x2 = Integer.parseInt(ln1[2]);
        int y2 = Integer.parseInt(ln1[3]);
        
        if(x1!=x2 && y1!=y2 && Math.abs(x1-x2)!=Math.abs(y1-y2)){
            bw.append("-1"+"\n");
        }else if(x1==x2){
            int a = Math.abs(y1-y2);
            bw.append((x1+a)+" "+y1+" "+(x2+a)+" "+y2+"\n");
        }else if(y1==y2){
            int a =  Math.abs(x1-x2);
            bw.append(x1+" "+(y1+a)+" "+x2+" "+(y2+a)+"\n");
        }else{
            bw.append(x1+" "+y2+" "+x2+" "+y1+"\n");
        }
        
      bw.close();  
    }
}
