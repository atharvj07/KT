from array import array

def solve(n, D):
	a = None
	b = None
	a_cnt = 0
	b_cnt = 0
	for (i, d) in enumerate(D):
		if a != d:
			if i < n - 1 and b == D[i + 1]:
				b_cnt += 1
				b = d
				continue
			a_cnt += 1
			a = d
			continue
		if b != d:
			b_cnt += 1
			b = d
	return a_cnt + b_cnt

def main():
	ans = []
	n = int(input())
	D = array('i', list(map(int, input().split(' '))))
	ans.append(solve(n, D))
	for a in ans:
		print(a)
main()
