#include <bits/stdc++.h>
int AN[366] = {0};
int AH[366] = {0};
int main() {
  int n, k, i, c, h, ans;
  while (scanf("%d%d", &n, &k) != EOF) {
    int lastRecv = 0;
    for (i = 0; i < 366; i++) AN[i] = 0, AH[i] = 0;
    ans = 0;
    scanf("%d", &c);
    for (i = 0; i < c; i++) {
      scanf("%d", &h);
      AH[h] = 1;
    }
    for (i = 1; i <= n; i++) {
      if (AH[i] == 1) {
        AN[i] = 1;
        lastRecv = i;
      }
      if (i - lastRecv >= k) {
        AN[i] = 1;
        lastRecv = i;
      }
    }
    for (i = 1; i <= n; i++)
      if (AN[i] == 1) ans++;
    printf("%d\n", ans);
  }
}
