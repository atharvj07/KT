#include <bits/stdc++.h>
using namespace std;
const int maxn = 110, maxd = 40;
char Sa[maxn], Sb[maxn];
int a, b, la, lb;
long long trans[maxn][maxd];
bool gettrans() {
  for (int i = 0; i < la; i++) {
    int now = i;
    for (int j = 0; j < lb; j++) {
      int cnt = 0;
      while (cnt <= la && Sa[now % la] != Sb[j]) now++, cnt++;
      if (Sa[now % la] != Sb[j]) return false;
      now++;
    }
    trans[i][0] = now - i;
  }
  for (int j = 1; j < maxd; j++)
    for (int i = 0; i < la; i++)
      trans[i][j] = trans[(trans[i][j - 1] + i) % la][j - 1] + trans[i][j - 1];
  return 1;
}
long long getmax() {
  long long now = 0, ans = 0;
  for (int i = maxd - 1; i >= 0; i--)
    if ((trans[now % la][i] + now) <= a * la) {
      now += trans[now % la][i];
      ans += 1LL << i;
    }
  return ans;
}
int main() {
  scanf("%d%d", &a, &b);
  scanf("%s", Sa);
  scanf("%s", Sb);
  la = strlen(Sa);
  lb = strlen(Sb);
  if (!gettrans())
    printf("0\n");
  else {
    long long ans = getmax();
    printf("%I64d\n", ans / b);
  }
  return 0;
}
