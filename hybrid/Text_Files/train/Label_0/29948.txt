#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  for (; b; a %= b, swap(a, b))
    ;
  return a;
}
int dx[4] = {1, -1, 0, 0};
int dy[4] = {0, 0, 1, -1};
int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
  long long n;
  cin >> n;
  vector<long long> a(n + 1);
  long long odd = (n + 1) / 2;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
    if (a[i] % 2) odd--;
  }
  long long dp[110][110][2];
  for (long long i = 0; i < 110; i++) {
    for (long long j = 0; j < 110; j++) {
      for (long long k = 0; k < 2; k++) dp[i][j][k] = 101112987654321;
    }
  }
  dp[0][0][0] = dp[0][0][1] = 0;
  for (long long i = 1; i <= n; i++) {
    for (long long use = 0; use <= odd; use++) {
      for (long long p = 0; p < 2; p++) {
        if (a[i] == 0) {
          if (use + 1 <= odd) {
            dp[i][use + 1][1] = min(dp[i - 1][use][0] + 1, dp[i - 1][use][1]);
          }
          dp[i][use][0] = min(dp[i - 1][use][0], dp[i - 1][use][1] + 1);
        } else if (a[i] % 2) {
          dp[i][use][1] = min(dp[i - 1][use][0] + 1, dp[i - 1][use][1]);
        } else {
          dp[i][use][0] = min(dp[i - 1][use][1] + 1, dp[i - 1][use][0]);
        }
      }
    }
  }
  cout << min(dp[n][odd][0], dp[n][odd][1]);
}
