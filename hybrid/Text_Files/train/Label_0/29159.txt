#include <bits/stdc++.h>
using namespace std;
string a;
int main() {
  long long n, x, y;
  scanf("%lld %lld %lld", &n, &x, &y);
  getchar();
  cin >> a;
  int i = 0;
  while (i < a.length() && a[i] == '1') i++;
  long long sum = 0;
  if (i != a.length()) sum++;
  char flag = a[i];
  for (; i <= a.length(); i++) {
    while (a[i] == flag) i++;
    flag = a[i];
    if (flag == '0') sum++;
  }
  if (!sum) {
    cout << "0";
    return 0;
  }
  long long ans = 9223372036854775807;
  ans = min(ans, sum * y);
  ans = min(ans, (sum - 1) * x + y);
  printf("%lld", ans);
  return 0;
}
