import math

def isprime(n):
	for i in range(2, int(n ** 0.5 + 1)):
		if n % i == 0:
			return False
	return True

def isPower(num, base):
	if base == 1 and num != 1:
		return False
	if base == 1 and num == 1:
		return True
	if base == 0 and num != 1:
		return False
	power = int(math.log(num, base) + 0.5)
	return base ** power == num
n = int(input())
ans = 1
if isprime(n):
	ans = n
for i in range(2, int(n ** 0.5 + 1)):
	if isPower(n, i):
		if isprime(i):
			ans = i
print(ans)
