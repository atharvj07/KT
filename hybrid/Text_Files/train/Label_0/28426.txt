#include<bits/stdc++.h>
#define int long long
using namespace std;
template<class T> inline bool chmax(T& a, T b){ if(a<b){ a=b; return 1; } return 0; }
template<class T> inline bool chmin(T& a, T b){ if(a>b){ a=b; return 1; } return 0; }

const int mod=1e9+7;

void add(int &a,int b)
{
    a=(a+b)%mod;
}

void mul(int &a,int b)
{
    a=(a*b)%mod;
}

signed main()
{
    int N,M; cin>>N>>M;
    vector<int> a(N),b(M);
    for(int i=0;i<N;i++){
	cin>>a[i]; 
    }
    for(int i=0;i<M;i++){
	cin>>b[i];
    }
    sort(a.rbegin(),a.rend());
    sort(b.rbegin(),b.rend());
    a.push_back(0);
    b.push_back(0);

    int row=0,col=0;
    int ans=1;
    for(int num=N*M;num>=1;num--){
	if(a[row]==num and b[col]==num){
	    row++; col++;
	}else if(a[row]==num){
	    mul(ans,col);
	    row++;
	}else if(b[col]==num){
	    mul(ans,row);
	    col++;
	}else{
	    int cnt=row*col-(N*M-num);
	    if(cnt==0){
		ans=0; break;
	    }
	    mul(ans,cnt);
	}
    }
    cout<<ans<<endl;
}
