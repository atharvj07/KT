#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int arr[n];
  vector<int> v[n + 1];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < 35; j++) v[i].push_back(0);
  for (int i = 0; i < n; i++) cin >> arr[i];
  sort(arr, arr + n);
  for (int i = 0; i < n; i++) {
    int a = arr[i], j = 0;
    while (a > 0) {
      v[i][j] = (a % 2);
      a = a / 2;
      j++;
    }
  }
  int val = 0;
  for (int i = 34; i >= 0; i--) {
    int count = 0, check = 0;
    for (int j = 0; j < n; j++) {
      if (v[j][i] == 1) count++;
      if (count > 1) break;
      if (count == 1 && check == 0) {
        val = j;
        check = 1;
      }
    }
    if (count == 1) break;
  }
  cout << arr[val] << " ";
  for (int i = 0; i < n; i++) {
    if (i != val) cout << arr[i] << " ";
  }
}
