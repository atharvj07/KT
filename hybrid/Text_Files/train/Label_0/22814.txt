const int INF = 1e9;

int N, M, hs, cs;

void dfs(int n, set<int> &S, vector< vector<int> > &es){
  S.insert(n);
  for(int next : es[n]) if(S.find(next) == S.end()) dfs(next, S, es);
}

void bfs(vector< vector<int> > &es, queue<int> &open, vector<int> &closed){
  while(!open.empty()){
    int now = open.front(), cost = closed[now]; open.pop();
    for(int next : es[now]){
      if(closed[next] > cost + 1){
        closed[next] = cost + 1;
        open.push(next);
      }
    }
  }
}

int main() {
  int T; cin >>T;
  while(T--){
    cin >>N >>M >>hs >>cs;
    N++;
    vector< vector<int> > hes(N), ces(N);
    REP(i, M){
      int a, b; char c; cin >>a >>b >>c;
      if(c == 'N') { hes[a].push_back(b); hes[b].push_back(a); }
      else { ces[a].push_back(b); ces[b].push_back(a); }
    }

    set<int> X, Y;
    dfs(cs, X, ces); dfs(0, Y, ces);
    if(X == Y) { cout <<0 <<endl; continue; }

    vector<int> A, B, C, ac(N, INF), bc(N, INF), cc(N, INF);
    queue<int> aq, bq, cq;
    aq.push(hs); ac[hs] = 0;
    for(int n : X) { bq.push(n); bc[n] = 0; }
    for(int n : Y) { cq.push(n); cc[n] = 0; }
    bfs(hes, aq, ac);
    bfs(hes, bq, bc);
    bfs(hes, cq, cc);

    int ans = INF;
    REP(i, N) ans = min(ans, ac[i] + bc[i] + cc[i]);

  }
  return 0;
}
