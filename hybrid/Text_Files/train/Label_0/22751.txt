#include <bits/stdc++.h>

const int INF = 0xfffffff;

int c[25], cube[25];
bool check(int num[25]) {
  for (int i = 1; i <= 21; i += 4)
    for (int j = i; j < i + 4; j++)
      if (num[j] != num[i]) return false;
  return true;
}

  memcpy(cube, c, sizeof c);
  swap(cube[2], cube[6]);

  swap(cube[10], cube[23]);
  swap(cube[12], cube[21]);
  return check(cube);
