#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
 string a,b;
 cin>>a>>b;
 vector<vector<ll>> dp(a.length()+1,vector<ll>(b.length()+1));
 for(ll i=1;i<=a.length();i++){
 	for(ll j=1;j<=b.length();j++){
 		if(a[i-1]==b[j-1]){
 			dp[i][j]=1+dp[i-1][j-1];	
		}	
		else{
			dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
		}
	}
 }
 string sol;
 ll I=a.length(),J=b.length();
 while(dp[I][J]){
 	if(a[I-1]==b[J-1]){
 		string chr="";
 		chr.push_back(a[I-1]);
 		sol=chr+sol;
 		I--,J--;
	}
	else{
		if(dp[I-1][J]==dp[I][J]){
			I--;
		}
		else{
			J--;
		}
	}
 }
 cout<<sol<<endl;
 return 0;
}
