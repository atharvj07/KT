t = int(input())
for i in range(t):
	(b, p, f) = map(int, input().split())
	(h, c) = map(int, input().split())
	ans = 0
	if h > c:
		ans += min(b // 2, p) * h
		b -= ans * 2 / h
		ans += min(b // 2, f) * c
	else:
		ans += min(b // 2, f) * c
		b -= ans * 2 / c
		ans += min(b // 2, p) * h
	print(int(ans))
