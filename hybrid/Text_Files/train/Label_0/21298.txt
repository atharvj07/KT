long long power(long long a, long long deg) {
  long long res = 1;
  for (; deg; a = mul(a, a), deg >>= 1) {
    if (deg & 1) {
      res = mul(res, a);
    }
  }
  return res;
}

long long magic_aux(long long sumL, long long sumR, long long sumStep,
                    long long cntL, long long cntStep) {
  if (sumR < sumL) {
    return 0;
  }
  long long stepCount = (sumR - sumL) / sumStep + 1;
  long long modSum = 0;
  add(modSum, mul(stepCount, cntL));
  add(modSum, mul(prog2(stepCount), cntStep));
  modSum = mul(modSum, sumL % 3);
  long long funcL = sumL / 3, funcStep = sumStep / 3;
  long long coeff0 = mul(cntL, funcL);
  long long coeff1 = sum(mul(cntL, funcStep), mul(funcL, cntStep));
  long long coeff2 = mul(cntStep, funcStep);
  long long divSum = 0;
  add(divSum, mul(coeff0, stepCount));
  add(divSum, mul(coeff1, prog2(stepCount)));
  add(divSum, mul(coeff2, prog3(stepCount)));
  long long res = sum(modSum, divSum);
  return res;
}
long long magic(long long sumL, long long sumR, long long sumStep,
                long long cntL, long long cntStep) {
  long long res = 0;
  for (long long it = 0; it < 3; ++it) {
    add(res, magic_aux(sumL + it * sumStep, sumR, 3 * sumStep,
                       cntL + cntStep * it, cntStep * 3));
  }
  return res;
}
long long seg2_aux(long long l, long long r) {
  if (l > r) {
    return 0;
  }
  long long r1 = l + 2 * (r - l) / 2;
  long long segCount = (r - l) / 2 + 1;
  long long res = magic(l, r1, 2, segCount, -1);
  return res;
}
long long seg2(long long l, long long r) {
  if (l > r) {
    return 0;
  }
  long long preRes = sum(seg2_aux(l, r), seg2_aux(l + 1, r));
  long long segLen = r - l + 1;
  long long segCount = (r - l) / 2 + 1;
  add(preRes, mul(segLen, prog2(segCount)));
  add(preRes, mul(MOD - 2, prog3(segCount)));
  return preRes;
}
long long calcSmallRect(long long first, long long second) {
  if (first < second) {
    swap(first, second);
  }
  long long rect1 = magic(0, second - 2, 1, 1, 1);
  long long rect2 = magic(second - 1, first - 1, 1, second, 0);
  long long rect3 = magic(first, first + second - 2, 1, second - 1, -1);
  long long right = seg2(1, second - 1);
  long long down = diff(seg2(1, first - 1),
                        seg2(2 * second + 1 + second, first - 1 + second));
  long long res = 0;
  add(res, rect1);
  add(res, rect2);
  add(res, rect3);
  subt(res, right);
  subt(res, down);
  return res;
}

template <long long MOD>
struct Ring {
  long long value = 0;
  Ring() {}
  Ring(long long _value) {
    value = _value;
    value = floor_mod(value, MOD);
  }
  Ring pow(long long p) const {
    Ring r = 1;
    Ring first;
    first.value = value;
    while (p) {
      if (p & 1) r *= first;
      first *= first;
      p /= 2;
    }
    return r;
  }
  Ring inv() const { return pow(MOD - 2); }
  void operator*=(const Ring<MOD> &b) {
    value *= b.value;
    value = floor_mod(value, MOD);
  }
  friend Ring operator*(Ring<MOD> a, const Ring<MOD> &b) {
    a *= b;
    return a;
  }
  void operator+=(const Ring<MOD> &b) {
    value += b.value;
    value -= (value >= MOD) * MOD;
  }
  friend Ring operator+(Ring a, const Ring &b) {
    a += b;
    return a;
  }
  void operator-=(const Ring<MOD> &b) {
    value -= b.value;
    value += (value < 0) ? MOD : 0;
  }
  friend Ring operator-(Ring a, const Ring &b) {
    a -= b;
    return a;
  }
  void operator/=(const Ring<MOD> &b) { (*this) *= b.inv(); }
  friend Ring operator/(Ring a, const Ring &b) {
    a /= b;
    return a;
  }
  bool operator==(const Ring<MOD> &b) { return value == b.value; }
  bool operator!=(const Ring<MOD> &b) { return value != b.value; }
  friend void fast_print(const Ring<MOD> &b) { fast_print(b.value); }
};
long long solve(long long n, long long a, long long b, long long m) {
  if (b == 0) return n * (a / m);
  if (a >= m) return n * (a / m) + solve(n, a % m, b, m);
  if (b >= m) return n * (n - 1) / 2 * (b / m) + solve(n, a, b % m, m);
  return solve((a + b * n) / m, (a + b * n) % m, m, b);
}

Ring<MOD> quadSum(long long n) {
  n--;
  Ring<MOD> m = n;
  return m * (m + 1) * (2 * m + 1) / Ring<MOD>(6);
}
Ring<MOD> solveBigFast(long long n, long long m) {
  Ring<MOD> rez = 0;
  long long ton = 2 * m;
  long long ton2 = ton;
  while (ton2 % 4 != n % 4) {
    ton2++;
  }
  if (ton2 < n) {
    Ring<MOD> D = solveBigFastDef(n, m, n - 1) + solveBigFastDef(n, m, n - 2) +
                  solveBigFastDef(n, m, n - 3) + solveBigFastDef(n, m, n - 4);
    rez += Ring<MOD>(D) * Ring<MOD>((n - ton2)) / 4;
    rez += Ring<MOD>(solve(n, 0, 1, 2)) * Ring<MOD>(m);
    rez -= Ring<MOD>(solve(ton2, 0, 1, 2)) * Ring<MOD>(m);
    n = ton2;
  }
  while (n % 4 != 0 || n > ton) {
    rez += solveBigFast(n, m, n - 1);
    n--;
  }
  rez += solve(n / 4, 0, 1, 1);
  rez += quadSum(n / 4) * 4;
  rez += quadSum(n / 4) * 4;
  rez += solve(n / 4, 0, 2, 1);
  rez += solve(n / 4, 1, 2, 1);
  rez += solve(n / 4, 1, 2, 2);
  rez += solve(n / 4, 2, 2, 2);
  rez += quadSum(n / 4) * 4;
  rez += solve(n / 4, 1, 4, 1);
  rez += solve(n / 4, 2, 2, 2);
  rez += solve(n / 4, 2, 2, 1);
  rez += quadSum(n / 4) * 4;
  rez += solve(n / 4, 2, 6, 1);
  return rez;
}
