n = int(input())
l = []
for i in range(n):
	l.append(input().strip())
s = input().strip()
length = len(s)
index1 = 0
a = 0
b = 0
c = 0
j = 0
for i in range(length):
	if a == 0:
		if s[i] == '<':
			b = 1
		if b == 1 and s[i] == '3':
			c = 1
		if b == 1 and c == 1:
			b = 0
			c = 0
			a = 1
	else:
		if index1 == n:
			break
		len1 = len(l[index1])
		if s[i] == l[index1][j]:
			j += 1
			if j == len1:
				index1 += 1
				a = 0
				j = 0
if index1 != n:
	print('no')
elif a == 1:
	print('yes')
else:
	print('no')
