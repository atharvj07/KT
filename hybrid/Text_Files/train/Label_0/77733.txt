import math
(x, y) = map(int, input().split())
t = int(input())
res = []
for _ in range(t):
	(a, b, v) = map(int, input().split())
	c = x - a
	if c < 0:
		c = 0 - c
	d = y - b
	if d < 0:
		d = 0 - d
	dis = c * c + d * d
	dis = math.sqrt(dis)
	l = dis / v
	res.append(l)
k = min(res)
print('{0:.20f}'.format(k))
