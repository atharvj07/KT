#include <bits/stdc++.h>
using namespace std;
const int N = 1e5;
const int D = 10;
int n;
int a[N], b[N];
struct DSU {
  vector<int> prev, sz, mne;
  int find(int x) { return prev[x] == x ? x : prev[x] = find(prev[x]); }
  void join(int x, int y) {
    x = find(x);
    y = find(y);
    if (x != y) {
      if (sz[x] > sz[y]) {
        swap(x, y);
      }
      prev[x] = y;
      sz[y] += sz[x];
      mne[y] = max(mne[y], mne[x]);
    }
  }
  int get_mne(int x) { return mne[find(x)]; }
  DSU() {}
  DSU(int n) : prev(n), sz(n, 1), mne(n) {
    for (int x = 0; x < n; ++x) {
      prev[x] = mne[x] = x;
    }
  }
};
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  cin >> n;
  for (int i = 0; i < n; ++i) {
    char c;
    cin >> c;
    a[i] = c - '0';
  }
  for (int i = 0; i < n; ++i) {
    char c;
    cin >> c;
    b[i] = c - '0';
  }
  DSU t(n + 1);
  vector<pair<int, int>> ans;
  long long cnt = 0;
  for (int i = 0; i < n - 1; ++i) {
    while (a[i] < b[i]) {
      if (a[i + 1] != 9) {
        ++a[i];
        ++a[i + 1];
        ++cnt;
        if (ans.size() < N) {
          ans.push_back({i, +1});
        }
      } else {
        int j = t.get_mne(i + 1) + 1;
        if (j == n) {
          cout << "-1\n";
          return 0;
        }
        if (a[j - 1] == 0) {
          if (a[j] == 9) {
            t.join(j - 1, j);
          } else {
            ++a[j];
            --a[i + 1];
            cnt += j - (i + 1);
            if (ans.size() < N) {
              for (int k = j - 1, flag = -1; k >= i + 1; --k) {
                ans.push_back({k, flag *= -1});
              }
            }
          }
        } else {
          if (a[j] == 0) {
            t.join(j - 1, j);
          } else {
            --a[j];
            --a[i + 1];
            cnt += j - (i + 1);
            if (ans.size() < N) {
              for (int k = j - 1, flag = 1; k >= i + 1; --k) {
                ans.push_back({k, flag *= -1});
              }
            }
          }
        }
      }
    }
    while (a[i] > b[i]) {
      if (a[i + 1] != 0) {
        --a[i];
        --a[i + 1];
        ++cnt;
        if (ans.size() < N) {
          ans.push_back({i, -1});
        }
      } else {
        int j = t.get_mne(i + 1) + 1;
        if (j == n) {
          cout << "-1\n";
          return 0;
        }
        if (a[j - 1] == 0) {
          if (a[j] == 9) {
            t.join(j - 1, j);
          } else {
            ++a[j];
            ++a[i + 1];
            cnt += j - (i + 1);
            if (ans.size() < N) {
              for (int k = j - 1, flag = -1; k >= i + 1; --k) {
                ans.push_back({k, flag *= -1});
              }
            }
          }
        } else {
          if (a[j] == 0) {
            t.join(j - 1, j);
          } else {
            --a[j];
            ++a[i + 1];
            cnt += j - (i + 1);
            if (ans.size() < N) {
              for (int k = j - 1, flag = 1; k >= i + 1; --k) {
                ans.push_back({k, flag *= -1});
              }
            }
          }
        }
      }
    }
  }
  if (a[n - 1] != b[n - 1]) {
    cout << "-1\n";
  } else {
    cout << cnt << '\n';
    for (int i = 0; i < min((int)ans.size(), N); ++i) {
      cout << ans[i].first + 1 << ' ' << ans[i].second << '\n';
    }
  }
  return 0;
}
