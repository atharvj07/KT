	public static int nextInt(BufferedInputStream in) throws IOException
	{
		int next = 0;
		boolean haveNext = false;
		int b;
		while ((b = in.read()) != -1)
		{
			if ('0' <= b && b <= '9')
			{
				next = 10*next + (b - '0');
				haveNext = true;
			}
			else if (haveNext)
			{
				break;
			}
		}
		return next;
	}
	
	public static boolean[] nonCentroidable(int[][] tree)
	{
		int[] weightUp = new int[tree.length + 1];
		int[] weightDown = new int[tree.length + 1];
		int[] bestSplitUp = new int[tree.length + 1];
		int[] bestSplitDown = new int[tree.length + 1];
		int[] bestBelow1 = new int[tree.length + 1];
		int[] bestBelow2 = new int[tree.length + 1];
		boolean[] visited = new boolean[tree.length + 1];
		int[] parent = new int[tree.length + 1];
		parent[1] = 0;
		boolean[] non = new boolean[tree.length + 1];
		int[] stack = new int[tree.length + 1];
		int sp = 0;
		// Initialize stack with root
		stack[sp] = 1;
		sp++;
		while (sp != 0)
		{
			int u = stack[sp - 1];
			// Have we processed the children yet?
			if (!visited[u])
			{
				visited[u] = true;
				// Push the children
				for (int v : tree[u])
				{
					if (v != parent[u])
					{
						parent[v] = u;
						stack[sp] = v;
						sp++;
					}
				}
			}
			else
			{
				sp--;
				stack[sp] = 0;
				// Sum the child weights
				int count = 1;
				int best = 0;
				for (int v : tree[u])
				{
					if (v != parent[u])
					{
						count += weightDown[v];
						int b = bestSplitDown[v];
						// Can't be a best edge of branch if w > n/2, no need to check
						if (b >= bestBelow1[u])
						{
							best = (b > best) ? b : best;
							bestBelow2[u] = bestBelow1[u];
							bestBelow1[u] = b;
						}
						else if (b > bestBelow2[u])
						{
							bestBelow2[u] = b;
						}
					}
				}
				weightDown[u] = count;
				weightUp[u] = tree.length - 1 - count;
				best = (count <= (tree.length - 1)/2) ? count : best;
				bestSplitDown[u] = best;
				if (count - best > (tree.length - 1)/2)
				{
					non[parent[u]] = true;
				}
			}
		}
		// In-tree edges
		// Initialize stack with root
		stack[sp] = 1;
		sp++;
		while (sp != 0)
		{
			sp--;
			int u = stack[sp];
			stack[sp] = 0;
			// note weird interaction with root, seems okay
			int x = weightUp[u];
			if (x <= (tree.length - 1)/2)
			{
				bestSplitUp[u] = x;
			}
			else
			{
				// Investigate the best edges of all out-branches, pick the one that maxes weight
				// i.e. in-edge towards the root, out-edges from v back down -- all are already evaluated
				int p = parent[u];
				int best = bestSplitUp[p];
				if (bestBelow1[p] == bestSplitDown[u])
				{
					best = (best > bestBelow2[p]) ? best : bestBelow2[p];
				}
				else
				{
					best = (best > bestBelow1[p]) ? best : bestBelow1[p];
				}
				bestSplitUp[u] = best;
				if (weightUp[u] - best > (tree.length - 1)/2)
				{
					non[u] = true;
				}
			}
			// Enqueue child in-tree edges
			for (int v : tree[u])
			{
				if (v != parent[u])
				{
					stack[sp] = v;
					sp++;
				}
			}
		}
		return non;
	}

	public static Map<Pair, Integer> countVertices(int[][] tree)
	{
		Map<Pair, Integer> weight = new HashMap<Pair, Integer>();
		Deque<Pair> stack = new ArrayDeque<>();
		// Initialize stack with root edges
		int root = 1;
		for (int u : tree[root])
		{
			Pair p = new Pair(root, u);
			stack.addFirst(p);
		}
		while (!stack.isEmpty())
		{
			Pair p = stack.peekFirst();
			// Have we processed the children yet?
			if (!weight.containsKey(p))
			{
				// Placeholder value as visited flag
				weight.put(p, 0);
				// Push the children
				int u = p.v, prev = p.u;
				for (int v : tree[u])
				{
					if (v != prev)
					{
						stack.addFirst(new Pair(u, v));
					}
				}
			}
			else
			{
				stack.removeFirst();
				// Sum the child weights
				int count = 1;
				int u = p.v, prev = p.u;
				for (int v : tree[u])
				{
					if (v != prev)
					{
						count += weight.get(new Pair(u, v));
					}
				}
				weight.put(p, count);
				weight.put(p.reverse(), tree.length - 1 - count);
			}
		}
		return weight;
	}
	
	public static Map<Pair, Integer> findBestSplits(int[][] tree,
			Map<Pair, Integer> weight)
	{
		Map<Pair, Integer> bestSplit = new HashMap<>();
		Deque<Pair> stack = new ArrayDeque<>();
		// Initialize stack with root edges
		int root = 1;
		for (int u : tree[root])
		{
			Pair p = new Pair(root, u);
			stack.addFirst(p);
		}
		// Find for out-tree pairs first
		while (!stack.isEmpty())
		{
			Pair p = stack.peekFirst();
			// Have we processed the children yet?
			if (!bestSplit.containsKey(p))
			{
				// Default value as visited flag
				bestSplit.put(p, weight.get(p));
				// Push the children
				int u = p.v, prev = p.u;
				for (int v : tree[u])
				{
					if (v != prev)
					{
						stack.addFirst(new Pair(u, v));
					}
				}
			}
			else
			{
				stack.removeFirst();
				// Only need to check children if p isn't good
				if (weight.get(p) > (tree.length - 1)/2)
				{
					int best = 0;
					// Investigate the best edges of all out-branches, pick the one that maxes weight
					int u = p.v, prev = p.u;
					for (int v : tree[u])
					{
						if (v != prev)
						{
							int w = bestSplit.get(new Pair(u, v));
							// Can't be a best edge of branch if w > n/2, no need to check
							if (w > best)
							{
								best = w;
							}
						}
					}
					bestSplit.put(p, best);
				}
			}
		}
		// In-tree edges
		Queue<Pair> queue = new ArrayDeque<>();
		for (int u : tree[root])
		{
			Pair p = new Pair(u, root);
			queue.add(p);
		}
		while (!queue.isEmpty())
		{
			Pair p = queue.remove();
			if (weight.get(p) <= (tree.length - 1)/2)
			{
				bestSplit.put(p, weight.get(p));
			}
			else
			{
				int best = 0;
				// Investigate the best edges of all out-branches, pick the one that maxes weight
				// i.e. in-edge towards the root, out-edges from v back down -- all are already evaluated
				int u = p.v, prev = p.u;
				for (int v : tree[u])
				{
					if (v != prev)
					{
						int w = bestSplit.get(new Pair(u, v));
						// Can't be a best edge of branch if w > n/2, no need to check
						if (w >= best)
						{
							best = w;
						}
					}
				}
				bestSplit.put(p, best);
			}
			// Enqueue child in-tree edges
			for (int u : tree[p.u])
			{
				if (u != p.v)
				{
					Pair q = new Pair(u, p.u);
					queue.add(q);
				}
			}
		}
		return bestSplit;
	}

class Pair
{
	public final int u;
	public final int v;
	
	public Pair(int u, int v)
	{
		this.u = u;
		this.v = v;
	}
	
	public Pair reverse()
	{
		return new Pair(v, u);
	}
	
	public boolean equals(Object o)
	{
		if (o instanceof Pair)
		{
			Pair p = (Pair)o;
			return u == p.u && v == p.v;
		}
		return false;
	}
	
	public int hashCode()
	{
		return 47*u + v;
	}
	
	public String toString()
	{
		return "(" + u + ", " + v + ")";
	}
	
	public static class LexComparator implements Comparator<Pair>
	{
		public int compare(Pair p1, Pair p2)
		{
			return (p1.u != p2.u) ? p1.u - p2.u : p1.v - p2.v;
		}
	}
}
