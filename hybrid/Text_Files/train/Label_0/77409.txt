n = int(input())
a = input().split(' ')
mp = {}
for i in range(len(a)):
	a[i] = int(a[i])
	mp[a[i]] = 0
for i in a:
	mp[i] += 1
sum = []
sum.append(a[0])
for i in range(1, len(a)):
	sum.append(sum[i - 1] + a[i])
ans = 0
for i in range(0, n - 1):
	cur = sum[n - 1] - sum[i]
	rang = n - 1 - i
	mp[a[i]] -= 1
	cur -= rang * a[i]
	cur -= mp.get(a[i] + 1, 0)
	cur += mp.get(a[i] - 1, 0)
	ans += cur
print(ans)
