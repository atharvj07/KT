T = 1
import random
import time
for _ in range(T):
	N = int(input())
	A = list(map(int, input().split()))
	M = 512
	inf = 1 << 60
	DP = [inf] * M
	DP[0] = -1
	for a in A:
		DP_new = [v for v in DP]
		for c in range(M):
			if DP[c] >= a:
				continue
			DP_new[c ^ a] = min(a, DP_new[c ^ a])
		DP = DP_new
	ans = []
	for c in range(M):
		if DP[c] < inf:
			ans.append(c)
	print(len(ans))
	print(*ans)
