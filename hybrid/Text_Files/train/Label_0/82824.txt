n = int(input())
A = list(map(int, input().split()))

def f(A, x, sign=1):
	cur = 0
	mn = 0
	ans = -1 << 32
	if sign == 1:
		for y in A:
			cur += y - x
			ans = max(ans, cur - mn)
			mn = min(cur, mn)
	else:
		for y in A:
			cur -= y - x
			ans = max(ans, cur - mn)
			mn = min(cur, mn)
	return ans
(lo, hi) = (-10000, 10000)
nt = 100
while nt:
	nt -= 1
	mid = (lo + hi) / 2
	X = f(A, mid, sign=1)
	Y = f(A, mid, sign=-1)
	if X > Y:
		lo = mid
	else:
		hi = mid
print(abs(f(A, lo)))
