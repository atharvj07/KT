(n, m, L, R) = [int(x) for x in input().split()]
if L == R:
	print(1)
	exit(0)
mod = 998244353
inv2 = mod + 1 >> 1
even = R // 2 - (L - 1) // 2
odd = (R + 1) // 2 - L // 2
cnt = n * m
if cnt % 2 == 1:
	print(pow(odd + even, cnt, mod) % mod)
else:
	print((pow(odd + even, cnt, mod) + pow(odd - even, cnt, mod)) * inv2 % mod)
