import math
import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	(a, b, n) = map(int, input().split())
	monster_power = list(map(int, input().split()))
	monster_health = list(map(int, input().split()))
	monster = []
	for i in range(n):
		monster.append([monster_power[i], monster_health[i]])
	monster.sort()
	for i in range(n - 1):
		b = b - math.ceil(monster[i][1] / a) * monster[i][0]
	if b < 0:
		print('NO')
		continue
	monster[-1][1] = monster[-1][1] - math.ceil(b / monster[-1][0]) * a
	b = b - math.ceil(b / monster[-1][0]) * monster[-1][0]
	if monster[-1][1] <= 0:
		print('YES')
	else:
		print('NO')
