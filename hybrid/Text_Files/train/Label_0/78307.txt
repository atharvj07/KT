import sys
(n, w, b) = map(int, sys.stdin.readline().split())
A = list(map(int, sys.stdin.readline().split()))
j = n - 1
test = True
s = 0
for x in range(n // 2 + n % 2):
	if A[x] == A[j - x]:
		if A[x] == 2:
			if x != j - x:
				s = s + 2 * min(w, b)
			else:
				s = s + min(w, b)
	elif A[x] == 2:
		if A[j - x] == 1:
			s = s + b
		else:
			s = s + w
	elif A[j - x] == 2:
		if A[x] == 1:
			s = s + b
		else:
			s = s + w
	else:
		test = False
		break
if test == False:
	print('-1')
else:
	print('%d' % s)
