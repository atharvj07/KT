from sys import stdin
input = stdin.readline

def put():
	return map(int, input().split())
(n, m, k) = put()
l = list(put())
if m == 1:
	l.sort(reverse=True)
	print(sum(l[:k]))
	exit()
p = []
summ = 0
for i in range(m):
	summ += l[i]
p.append(summ)
for i in range(m, n):
	summ += l[i] - l[i - m]
	p.append(summ)
r = p.copy()
c = len(p)
q = [0] * c
for j in range(k):
	for i in range(j, c):
		q[i] = p[i]
		if i - m >= 0 and j > 0:
			q[i] += r[i - m]
		if i - 1 >= 0:
			q[i] = max(q[i - 1], q[i])
	r = q.copy()
print(r[-1])
