#include <bits/stdc++.h>
using namespace std;
mt19937 rng(time(0));
uniform_int_distribution<int> uid(-1e9, 1e9);
long long mod = 998244353;
long long binpow(long long x, long long px) {
  if (px <= 0) return !px;
  if (px & 1) return x * binpow(x, px ^ 1) % mod;
  x = binpow(x, px >> 1);
  return x * x % mod;
}
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  for (int i = 1; i < n; ++i)
    cout << (180LL * binpow(10, n - i - 1) +
             810LL * (n - i - 1) * binpow(10, n - i - 2)) %
                mod
         << ' ';
  cout << "10\n";
}
