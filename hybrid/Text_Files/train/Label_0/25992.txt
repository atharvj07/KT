#include <bits/stdc++.h>
using namespace std;
const int MAXN = 500 + 10;
int x[MAXN];
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < n; i++) cin >> x[i];
  double ans = 0;
  for (int i = 0; i < m; i++) {
    double c;
    int a, b;
    cin >> a >> b >> c;
    ans = max(ans, (double)(x[a - 1] + x[b - 1]) / c);
  }
  cout << fixed << setprecision(9) << ans << endl;
}
