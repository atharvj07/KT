t = int(input())
for i in range(t):
	n = int(input())
	q = list(map(int, input().split()))
	f = list(sorted(q))
	f.reverse()
	sumi = sum(f)
	number = 1
	listo = []
	for j in range(len(f)):
		if f[j] >= sumi // 2 + 1:
			number = f[j]
			break
		sumi = sumi - f[j]
	k = 0
	for c in range(n):
		if q[c] >= number:
			listo.append(c + 1)
			k = k + 1
	print(k)
	print(*listo, sep=' ')
