#include <bits/stdc++.h>
using namespace std;
long long a[2 * (int)1e5 + 10], b[2 * (int)1e5 + 10];
map<int, int> A, B;
int main() {
  int n, m, p;
  cin >> n >> m >> p;
  for (int i = 1; i < n + 1; ++i) cin >> a[i];
  for (int i = 1; i < m + 1; ++i) cin >> b[i];
  if (n < m || m == n && p != 1) {
    cout << 0 << endl;
    return 0;
  }
  vector<int> ans;
  for (int i = 1; i < m + 1; ++i) B[b[i]]++;
  int i, j, c;
  for (i = 1; i <= p; ++i) {
    A.clear();
    for (j = i, c = 0; j <= n && c < m; j += p, c++) {
      A[a[j]]++;
    }
    for (c = i; j <= n; j += p, c += p) {
      if (A == B) ans.push_back(c);
      A[a[j]]++;
      A[a[c]]--;
      if (!A[a[c]]) A.erase(a[c]);
    }
    if (A == B) ans.push_back(c);
  }
  sort((ans).begin(), (ans).end());
  cout << (int)ans.size() << endl;
  for (int i = 0; i < (int)ans.size(); ++i) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
