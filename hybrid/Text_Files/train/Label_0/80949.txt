import math

def gcd(a, b):
	while 1:
		b = b % a
		if b == 0:
			return a
		a = a % b
		if a == 0:
			return b
t = int(input())
for i2 in range(t):
	(a, m) = map(int, input().split())
	k = gcd(a, m)
	ans = m // k
	x = m // k
	i = 1
	for i in range(2, math.ceil(x ** (1 / 2)) + 1):
		if x % i == 0:
			while x % i == 0:
				x //= i
			ans -= ans // i
	if x >= 2:
		ans -= ans // x
	print(ans)
