#include <bits/stdc++.h>
using namespace std;
long long a[100005];
int main() {
  long long n;
  scanf("%lld", &n);
  for (long long i = 0; i < n; i++) scanf("%lld", &a[i]);
  sort(a, a + n);
  long long sm = 0;
  for (long long i = 0; i < n / 2; i++) {
    sm = ((sm % 10007) + ((a[i] % 10007) * (a[n - i - 1] % 10007)) % 10007) %
         10007;
  }
  sm = (sm * 2) % 10007;
  if (n % 2 == 1)
    sm = ((sm % 10007) + ((a[n / 2] % 10007) * (a[n / 2] % 10007)) % 10007) %
         10007;
  cout << sm << endl;
  return 0;
}
