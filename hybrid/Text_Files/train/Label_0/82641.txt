INT_MAX = 10 ** 18 + 7
MOD = 10 ** 9 + 7

def INPUT():
	return list((int(i) for i in input().split()))

def LIST_1D_ARRAY(n):
	return [0 for _ in range(n)]

def LIST_2D_ARRAY(m, n):
	return [[0 for _ in range(n)] for _ in range(m)]

def clac(A, n):
	ans1 = 0
	sum1 = 0
	for i in range(1, n):
		if sum1 + A[i] > 0:
			sum1 += A[i]
		else:
			sum1 = 0
		ans1 = max(ans1, sum1)
	sum2 = 0
	ans2 = 0
	for i in range(n - 1):
		if sum2 + A[i] > 0:
			sum2 += A[i]
		else:
			sum2 = 0
		ans2 = max(ans2, sum2)
	return max(ans2, ans1)
for _ in range(int(input())):
	n = int(input())
	A = INPUT()
	s = sum(A)
	if s > clac(A, n):
		print('YES')
	else:
		print('NO')
