n = int(input())
a = [int(i) for i in input().split()]
c = [0]
for el in a:
	c.append(el)
for i in range(n):
	a[i] -= c[i]
r = []
for k in range(1, n + 1):
	q = a[:]
	if n % k != 0:
		x = k - n % k
		for i in range(x):
			q.append('@')
		h = n + x
	else:
		h = n
	v = h // k
	w = h // v
	f = True
	for i in range(w):
		p = q[i]
		for j in range(1, v):
			if q[j * w + i] != p and q[j * w + i] != '@':
				f = False
				break
	if f:
		r.append(k)
print(len(r))
print(*r)
