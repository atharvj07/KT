#include <bits/stdc++.h>
int dizi[200005][3], ans[200005];
int main() {
  int n, i, x, y;
  scanf("%d", &n);
  if (n == 3) {
    printf("2 1 3");
    return 0;
  }
  for (i = 1; i <= n; i++) {
    scanf("%d %d", &x, &y);
    dizi[i][1] = x;
    dizi[i][2] = y;
  }
  for (i = 1; i <= n; i++) {
    x = dizi[i][1];
    y = dizi[i][2];
    if (dizi[x][1] == y || dizi[x][2] == y) {
      dizi[i][1] = x;
      dizi[i][2] = y;
    } else {
      dizi[i][1] = y;
      dizi[i][2] = x;
    }
  }
  y = 1;
  for (i = 1; i <= n; i++) {
    x = dizi[y][1];
    printf("%d ", x);
    y = x;
  }
}
