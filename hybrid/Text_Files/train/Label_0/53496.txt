import java.util.*;

public class Main {
 
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int len = sc.nextInt();
        long modulo = 998244353;
        if(len == 1) {
            System.out.println(0);
            sc.close();
            return;
        }
        int[] a = new int[n];
        for(int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long[][] dpWithout = new long[n][k+1];
        long[][] dpWith    = new long[n][k+1];
        long[]   dpTotal   = new long[n];
        int[] sequenceLength = new int[k+1];
        if(a[0] != -1) sequenceLength[a[0]] = 1; 
        else Arrays.fill(sequenceLength, 1);
        if(a[0] == -1) {
            for(int j = 1; j <= k; j++) {
                dpWith[0][j] = 1;
                dpWithout[0][j] = k - 1;
            }
        } else {
            int firstNumber = a[0];
            dpWith[0][firstNumber] = 1;
            for(int j = 1; j <= k; j++) {
                if(j != firstNumber) {
                    dpWithout[0][j] = 1;
                }
            }
        }
        dpTotal[0] = a[0] == -1 ? k : 1;
        for(int i = 1; i < n; i++) {
            for(int j = 1; j <= k; j++) {
                if(a[i] == -1 || a[i] == j) {
                    sequenceLength[j]++;
                    dpWith[i][j] = dpTotal[i-1];
                    dpWith[i][j] %= modulo;
                    if(sequenceLength[j] >= len) {
                        if(i - len < 0) dpWith[i][j]--;
                        else dpWith[i][j] += modulo - dpWithout[i - len][j];
                    }
                } else {
                    sequenceLength[j] = 0;
                }
            }   
            long total = 0;
            for(int j = 1; j <= k; j++) {
                total += dpWith[i][j];
            }
            total %= modulo;
            dpTotal[i] = total;
            for(int j = 1; j <= k; j++) {
                dpWithout[i][j] = total + modulo - dpWith[i][j];
                dpWithout[i][j] %= modulo;
            }
        }
        long total = 0;
        for(int j = 1; j <= k; j++) {
            total += dpWith[n-1][j];
        }
        System.out.println(total % modulo);
        sc.close();
        
    }

    public static void print(int[][] a) {
        for(int i = 0; i < a.length; i++) {
            for(int j = 0; j < a[0].length; j++) {
            }
            System.out.println();
        }
    }
}
