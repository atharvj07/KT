import math
for _ in range(int(input())):
	(n, w) = map(int, input().split())
	l1 = list(map(int, input().split()))
	l = []
	for i in range(n):
		l.append([l1[i], i + 1])
	s = 0
	l.sort()
	ans = -1
	st = 0
	end = n - 1
	inf = -1
	for i in range(n):
		if l[i][0] >= math.ceil(w / 2) and l[i][0] <= w:
			inf = i
			ans = 1
			st = i
			end = i
			break
		elif l[i][0] < math.ceil(w / 2):
			s += l[i][0]
			if s >= math.ceil(w / 2):
				end = i
				ans = 1
				break
	k = []
	if ans != -1:
		for i in range(st, end + 1):
			k.append(l[i][1])
		print(len(k))
		print(*k)
	else:
		print(-1)
