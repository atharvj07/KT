#include <bits/stdc++.h>
using namespace std;

#define int long long

const int INI = 1e15;
const int INF = 1e9;

int dp[51][51][51][51][2][4];
int N, M;
vector<int> A, B;
vector<int> cuma, cumb;

int dfs(int a, int b, int c, int d, int t, int r) {
  // cout << a << " " << b << " " << c << " " << d << " " << t << endl;
  if ( dp[a][b][c][d][t][r] != INI ) return dp[a][b][c][d][t][r];

  if ( a == N && b == M && c == N && d == M ) return 0;
  if ( a == c && b == d && r >= 3 ) return 0;  

  int ret = 0;  
  if ( !t ) {
    ret = -INF;    
    if ( a < N ) {
      if ( A[a] == -1 ) ret = max(ret, dfs(a+1, b, c, b, !t, 0));
      else ret = max(ret, dfs(a+1, b, c, d, !t, 0));		     
    }
    int suma = cuma[a] - cuma[c], sumb = cumb[b] - cumb[d];    
    ret = max(ret, dfs(a, b, a, b, !t, r+1) + suma-sumb);    
  } else {
    ret = INF;
    if ( b < M ) {
      if ( B[b] == -1 ) ret = min(ret, dfs(a, b+1, a, d, !t, 0));
      else ret = min(ret, dfs(a, b+1, c, d, !t, 0));		     
    }
    int suma = cuma[a] - cuma[c], sumb = cumb[b] - cumb[d];    
    ret = min(ret, dfs(a, b, a, b, !t, r+1) + suma-sumb);    
  }

  return dp[a][b][c][d][t][r] = ret;  
}

signed main() {
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  cout << fixed << setprecision(12);

  fill_n(*****dp, 51*51*51*51*2*4, INI);
  cin >> N >> M;
  A = cuma = vector<int>(N+1, 0);  
  for ( int i = 0; i < N; i++ ) {
    cin >> A[i];
    if ( A[i] >= 0 ) cuma[i+1] = A[i];
    cuma[i+1] += cuma[i];    
  }
  B = cumb = vector<int> (M+1, 0);
  for ( int i = 0; i < M; i++ ) {
    cin >> B[i];
    if ( B[i] >= 0 ) cumb[i+1] = B[i];
    cumb[i+1] += cumb[i];    
  }

  cout << dfs(0, 0, 0, 0, 0, 0) << endl;  
  
  return 0;
}
