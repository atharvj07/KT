bool check(int t, string& week, string& time)
{
    static const string x[] = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"};

    if(week != "All" && x[t/(24*60)%7] != week)
        return false;
    if(time != "All"){
        t %= 24 * 60;
        if((time == "Day") ^ (6*60 <= t && t < 18*60))
            return false;
    }
    return true;
}

int main()
{
    for(;;){
        int s, n, cycle;
        string week, time;
        double prob;
        int m;
        cin >> s >> n >> cycle >> week >> time >> prob >> m;
        if(s == 0)
            return 0;
        prob = 1.0 - 1.0 / prob;

        double ret = 0.0;
        for(int i=0; i<60*24*7; ++i){
            double tmp = 1.0;
            for(int j=0; j<m; ++j){
                int t1 = i + j * cycle;
                int t2 = t1 + s;
                if(check(t1, week, time) && check(t2, week, time))
                    tmp *= pow(prob, n);
            }
            ret = max(ret, 1.0 - tmp);
        }

        printf("%.10f\n", ret);
    }
}
