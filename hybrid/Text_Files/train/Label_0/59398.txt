import java.io.IOException;
import java.util.Scanner;

public class B {
	public static final void main( String[] args ) throws IOException {
		Scanner s = new Scanner( System.in );
		int size = s.nextInt();
		int k = s.nextInt();
		
		boolean[][] map = new boolean[ size ][ size ];
		if ( fillMap( map , size , k ) ) {
			printMap( map , size );
			return;
		}
		System.out.println( "NO" );
	}
	
	public static final int[] dx = { 1 , 0 , -1 , 0 };
	public static final int[] dy = { 0 , 1 , 0 , -1 };
	public static final boolean hasNoAdj( boolean[][] map , int x , int y ) {
		boolean ok = true;
		for ( int i=0 ; i<4 ; ++ i ) {
			try {
				if ( map[ x+dx[ i ] ][ y+dy[ i ] ] == true ) {
					ok = false;
					break;
				}
			}
			catch ( IndexOutOfBoundsException e ) {
				//ignore
			}
		}
		return ok;
	}
	
	public static final boolean fillMap( boolean[][] map , int size , int k ) {
		int numRemaining = k;
		for ( int i=0 ; i<size ; ++i ) {
			for ( int j=0 ; j<size ; ++j ) {
				if ( hasNoAdj( map , i , j ) && numRemaining > 0  ) {
					map[ i ][ j ] = true;
					--numRemaining;
				}
			}
		}
		return numRemaining == 0;
	}
	
	public static final void printMap( boolean[][] map , int size ) {
		System.out.println( "YES" );
		for ( int i=0 ; i<size ; ++i ) {
			for ( int j=0 ; j<size ; ++j ) {
				if ( map[ i ][ j ] ) {
					System.out.print( "L" );
				}
				else {
					System.out.print( "S" );
				}
			}
			System.out.println();
		}
	}
}
