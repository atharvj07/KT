import java.util.*;
import java.io.*;
 
 
public class Solution{
 
	public static Integer INT(String s){
		return Integer.parseInt(s);
	}
 
	public static Long LONG(String s){
		return Long.parseLong(s);
	}
 
	//====================================================================================================================



 
	public static void main(String args[])throws IOException{
 
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));	Scanner in=new Scanner(System.in);	StringBuilder out=new StringBuilder();
 		
		int t=in.nextInt();

		while(t--!=0){

			int n=in.nextInt(),
				m=in.nextInt(),
				k=in.nextInt();

			int a[]=new int[n];
			for(int i=0; i<n; i++)
				a[i]=in.nextInt();

			boolean flag=true;

			for(int i=0; i<n-1; i++){
				
				int next=Math.max(0, a[i+1]-k);

				if(a[i]>=next)
					m+=(a[i]-next);
				else
					m-=(next-a[i]);

				if(m<0){
					flag=false;
					break;
				}
			}

			System.out.println(flag?"YES":"NO");
		}
 	}
}
