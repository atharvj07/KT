def solve():
	for _ in range(1, ii() + 1):
		n = ii()
		a = li()

		for i in range(2, n):

				x2 = gcd(x2, a[i])
			else:
				x1 = gcd(x1, a[i])
		ok = True
		for i in range(n):
			if i % 2 and a[i] % x1 == 0:
				ok = False
				break
		if ok:
			print(x1)
			continue
		ok = True
		for i in range(n):
			if i % 2 == 0 and a[i] % x2 == 0:
				ok = False
				break
		if ok:
			print(x2)
		else:
			print(0)
