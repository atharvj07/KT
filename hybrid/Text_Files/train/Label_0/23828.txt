#include <bits/stdc++.h>
using namespace std;
int main() {
  int N;
  cin >> N;
  int A[N];
  for (int i = 0; i < N; ++i) cin >> A[i];
  long long cur = 1;
  long long res = 0;
  for (int i = 1; i < N; ++i) {
    if (A[i] == A[i - 1])
      ++cur;
    else {
      res += (cur) * (cur + 1) / 2;
      cur = 1;
    }
  }
  res += (cur) * (cur + 1) / 2;
  cout << res;
  return 0;
}
