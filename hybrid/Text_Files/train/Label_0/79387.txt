(good, p, n) = (input(), input(), int(input()))
plen = len(p)
if '*' not in p:
	pflag = True
else:
	pflag = False
for c in range(n):
	s = input()
	(slen, i, j, flag) = (len(s), 0, 0, True)
	if slen < plen:
		if pflag:
			flag = False
		elif slen < plen - 1:
			flag = False
	while i < plen and flag:
		if p[i] == '*':
			break
		elif p[i] == '?':
			if s[j] not in good:
				flag = False
		elif p[i] != s[j]:
			flag = False
		i += 1
		j += 1
	if i < plen and flag:
		sp = j
		(i, j) = (plen - 1, slen - 1)
		while i >= 0 and flag:
			if p[i] == '*':
				break
			elif p[i] == '?':
				if s[j] not in good:
					flag = False
			elif p[i] != s[j]:
				flag = False
			i -= 1
			j -= 1
		while j >= sp and flag:
			if s[j] in good:
				flag = False
			j -= 1
	elif j < slen:
		flag = False
	if flag:
		print('YES')
	else:
		print('NO')
