#include<bits/stdc++.h>
using namespace std;

int main(){
	int n;
	cin>>n;
	string s;
	//getline(cin,s);
	for(int i=0;i<n;i++){
		//getline(cin,s);
		cin>>s;
		int len=s.size();
		int la=0,lb=0,res=0; 
		for(int j=0;j<len;j++){
			if(s[j]=='(') la++;
			else if(s[j]=='[') lb++;
			else if(s[j]==')' && la) la--,res++; 
			else if(s[j]==']' && lb) lb--,res++; 		
		}
		cout<<res<<endl;
	}
	return 0;
}
