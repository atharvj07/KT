#include <bits/stdc++.h>
#pragma warning(disable : 4996)
using namespace std;
int n, m, i, j, k, l, r, x;
int a[2009], b[2009];
int ones;
int gcd(int u, int v) {
  while (v != 0) {
    int r = u % v;
    u = v;
    v = r;
  }
  return u;
}
int solve(int start) {
  int res = 0;
  for (int i = start + 1; i < n && ones < n; i++) {
    b[i] = gcd(b[i], b[i - 1]);
    if (b[i] == 1) ones++;
    res++;
  }
  for (int i = n - 2; i >= 0 && ones < n; i--) {
    b[i] = gcd(b[i], b[i + 1]);
    if (b[i] == 1) ones++;
    res++;
  }
  for (int i = 0; i < n && ones < n; i++) {
    b[i] = gcd(b[i], b[i - 1]);
    if (b[i] == 1) ones++;
    res++;
  }
  if (ones < n) return INT_MAX;
  return res;
}
int main() {
  cin >> n;
  int minres = INT_MAX;
  for (i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (i = 0; i < n; i++) {
    ones = 0;
    for (j = 0; j < n; j++) {
      b[j] = a[j];
      if (b[j] == 1) ones++;
    }
    minres = min(solve(i), minres);
  }
  if (minres == INT_MAX)
    cout << -1;
  else
    cout << minres;
  return 0;
}
