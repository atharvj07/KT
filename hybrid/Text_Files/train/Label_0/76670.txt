for _ in range(int(input())):
	n = int(input())
	l = [*map(int, input().split())]
	lis = []
	for i in range(n):
		lis.append((l[i], i + 1))
	lis.sort()
	lk = []
	for i in range(n):
		lk.append(lis[i][1])
	li = sorted(l)
	s = 0
	b = -1
	for i in range(n):
		if i >= 1:
			if li[i] > s:
				b = i
		s += li[i]
	print(min(n - b, n))
	if b == -1:
		print(*sorted(lk))
	else:
		print(*sorted(lk[b:]))
