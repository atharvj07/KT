#include <bits/stdc++.h>
using namespace std;
const long long N = 1e9 + 7, MOD = 2e9 + 7;
int c = 31, d = 201;
pair<int, int> h(string str) {
  int a = 0, b = 0;
  for (auto x : str) {
    a = (a * 1LL * c + (x - '0' + 1)) % N;
    b = (b * 1LL * d + (x - '0' + 1)) % MOD;
  }
  return {a, b};
}
int main() {
  ios_base ::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  vector<map<pair<int, int>, int> > a(n);
  map<pair<int, int>, int> b;
  vector<string> arr(n);
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  for (int i = 0; i < n; i++) {
    for (int j = 1; j <= 9; j++) {
      for (int l = 0; l <= 9 - j; l++) {
        string c = arr[i].substr(l, j);
        pair<int, int> e = h(c);
        a[i][e]++;
        b[e]++;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    bool f = true;
    for (int j = 1; j <= 9 && f; j++) {
      for (int l = 0; l <= 9 - j && f; l++) {
        string c = arr[i].substr(l, j);
        pair<int, int> e = h(c);
        if (b[e] - a[i][e] <= 0) {
          cout << c << endl;
          f = false;
        }
      }
    }
  }
  return 0;
}
