#include <bits/stdc++.h>
using namespace std;
string s = "";
vector<long long> c;
char ns = 'a';
void create(long long l, long long cnt) {
  for (long long i = 0; i < cnt; i++) {
    for (long long j = 0; j < l; j++) s += ns;
    if (ns < 'z')
      ns++;
    else
      ns = 'a';
  }
}
void dele(long long l, long long cnt) {
  long long p = 1;
  for (long long i = l - 1; i > 0; i--) {
    cnt /= p;
    cnt *= ++p;
    c[i] -= cnt;
  }
}
int main() {
  long long n;
  cin >> n;
  c.resize(n + 1);
  for (long long i = 1; i <= n; i++) cin >> c[i];
  for (long long i = n; i > 0; i--)
    if (c[i]) {
      create(i, c[i]);
      dele(i, c[i]);
    }
  cout << s;
  return 0;
}
