
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

/*
 * To change this template, choose Tools | Templates
 * and open the template in the editor.
 */

/**
 *
 * @author utp
 */
public class Main {
    public static void main(String Args[]) throws IOException{
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer token;
        String in = input.readLine();
        token = new StringTokenizer(in);
        int n = Integer.valueOf(token.nextToken());
        int P1 = Integer.valueOf(token.nextToken());
        int P2 = Integer.valueOf(token.nextToken());
        int P3 = Integer.valueOf(token.nextToken());
        int T1 = Integer.valueOf(token.nextToken());
        int T2 = Integer.valueOf(token.nextToken());
        int total=0;
        int li=0,ri=0;
        for (int i = 0; i < n; i++) {
            token = new StringTokenizer(input.readLine());
            li = Integer.valueOf(token.nextToken());
            if(i!=0){
                int out_pc = li-ri;
                total += (out_pc>T1 )? T1*P1:out_pc*P1;
                out_pc-=T1;
                if(out_pc>0){
                    total += (out_pc>T2 )? T2*P2:out_pc*P2;
                    out_pc-=T2;                 
                    total += (out_pc>0)? out_pc*P3:0;
                }
            }            
            ri = Integer.valueOf(token.nextToken());
            total += (ri-li)*P1;            
        }
        System.out.println(total);
        
    }
}
