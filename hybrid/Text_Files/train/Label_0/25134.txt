#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9 + 7;
long long l[100009], r[100009], a[100009];
int main() {
  long long n;
  cin >> n;
  long long ans = 0;
  l[1] = 1;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 2; i <= n; i++) {
    l[i] = 1;
    if (a[i] > a[i - 1]) l[i] += l[i - 1];
    ans = max(l[i], ans);
  }
  r[n] = 1;
  for (int i = n - 1; i >= 1; i--) {
    r[i] = 1;
    if (a[i] < a[i + 1]) r[i] += r[i + 1];
  }
  ans = min(n, ans + 1);
  for (int i = 1; i <= n; i++) {
    if (a[i - 1] + 1 < a[i + 1]) {
      ans = max(ans, l[i - 1] + 1 + r[i + 1]);
    }
  }
  cout << ans;
  return 0;
}
