(n, m) = map(int, input().split())
num = []
used = []
usedstr = [0] * n
fl = True
usedstolb = [0] * m
for i in range(n):
	num.append(input())
for i in range(n):
	if usedstr[i] == 0:
		stroki = []
		stolbi = []
		for j in range(m):
			if num[i][j] == '#':
				stolbi.append(j)
				for x in range(n):
					if num[x][j] == '#' and x not in stroki:
						stroki.append(x)
		for x in range(len(stroki)):
			for y in range(m):
				if num[stroki[x]][y] == '#' and y not in stolbi:
					stolbi.append(y)
		for x in range(len(stolbi)):
			if usedstolb[stolbi[x]]:
				fl = False
				break
			else:
				usedstolb[stolbi[x]] = 1
		for x in range(len(stroki)):
			if usedstr[stroki[x]]:
				fl = False
				break
			else:
				usedstr[stroki[x]] = 1
		for x in range(len(stroki)):
			for y in range(len(stolbi)):
				if num[stroki[x]][stolbi[y]] == '.':
					fl = False
					break
		if not fl:
			break
if fl:
	print('Yes')
else:
	print('No')
