def MI():
	return map(int, input().split())

def I():
	return int(input())

def LI():
	return [int(i) for i in input().split()]
from collections import defaultdict
d = defaultdict(int)
n = I()
h = LI()
g = LI()
ans = 0
dp = [[0] * 3 for _ in range(n + 1)]
dp[1][0] = 0
dp[1][1] = h[0]
dp[1][2] = g[0]
for i in range(1, n):
	dp[i + 1][0] = max(dp[i][1], dp[i][2])
	dp[i + 1][1] = max(dp[i][0], dp[i][2]) + h[i]
	dp[i + 1][2] = max(dp[i][0], dp[i][1]) + g[i]
ans = max(dp[n])
print(ans)
