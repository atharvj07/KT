#include <bits/stdc++.h>
using namespace std;
map<string, int> mp;
vector<string> v[1000];
vector<string> ans[1000];
void verif(vector<string>& v, int x) {
  for (auto& i : v) reverse(i.begin(), i.end());
  for (int i(0); i < v.size(); i++) {
    bool ac(1);
    for (int j(0); j < v.size(); j++) {
      if (j == i || (j > i && v[j] == v[i])) continue;
      if (v[i].size() > v[j].size()) continue;
      bool ok(1);
      for (int k(0); k < v[i].size(); k++)
        if (v[i][k] != v[j][k]) ok = 0;
      if (ok) ac = 0;
    }
    if (ac) {
      ans[x].push_back(v[i]);
      reverse(ans[x].rbegin()->begin(), ans[x].rbegin()->end());
    }
  }
}
int main() {
  int n, cnt(0);
  cin >> n;
  while (n--) {
    string s, t;
    cin >> s;
    if (mp[s] == 0) mp[s] = ++cnt;
    int nr;
    cin >> nr;
    while (nr--) {
      cin >> t;
      v[mp[s]].push_back(t);
    }
  }
  cout << mp.size() << '\n';
  for (auto i : mp) {
    verif(v[i.second], i.second);
    cout << i.first << ' ' << ans[i.second].size() << ' ';
    for (auto j : ans[i.second]) cout << j << ' ';
    cout << '\n';
  }
  return 0;
}
