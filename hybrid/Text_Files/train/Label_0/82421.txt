import sys

def main():
	input = sys.stdin.readline
	q = int(input())
	for z in range(q):
		(n, k) = map(int, input().split())
		a = list(map(int, input().split()))
		s = [e % 2 for e in a]
		if sum(s) < k:
			print('NO')
		elif k == 1:
			if sum(a) % 2 == 0:
				print('NO')
			else:
				print('YES')
				print(n)
		elif sum(s) % 2 != k % 2:
			print('NO')
		else:
			tsum = 0
			kotae = []
			for y in range(n):
				tsum += s[y]
				if tsum % 2 == 1:
					kotae.append(y + 1)
					tsum = 0
					if len(kotae) == k - 1:
						kotae.append(n)
						print('YES')
						print(*kotae)
main()
