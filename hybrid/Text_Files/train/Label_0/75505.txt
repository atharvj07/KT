meal = [0, 1, 2, 0, 2, 1, 0]
a = [0] * 3
b = [0] * 3
(a[0], a[1], a[2]) = map(int, input().split())
ans = 0
for start in range(len(meal)):
	(l, r) = (0, int(1e+18))
	while r - l > 1:
		m = int((r + l) / 2)
		c = int(m / len(meal))
		(b[0], b[1], b[2]) = (meal.count(0) * c, meal.count(1) * c, meal.count(2) * c)
		for j in range(m % len(meal)):
			i = (start + j) % len(meal)
			b[meal[i]] += 1
		check = True
		for i in range(3):
			if b[i] > a[i]:
				check = False
		if check:
			l = m
		else:
			r = m
	ans = max(ans, l)
print(ans)
