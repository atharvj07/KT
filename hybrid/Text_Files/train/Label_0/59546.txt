    void solve(int TC)throws Exception{
        int n = ni(), m = ni();
        SegTree t = new SegTree(1+n);
        int[] a = new int[m];
        for(int i = 0; i< m; i++)a[i] = ni();
        for(int i = 1; i< m; i++){
            int x = Math.min(a[i-1], a[i]), y = Math.max(a[i-1], a[i]);
            if(x == y)continue;
            t.update(1, x-1, Math.abs(x-y));
            t.update(x+1, y-1, Math.abs(x+1-y));
            t.update(y+1, n, Math.abs(x-y));
            t.update(x, x, Math.abs(y-1));
            t.update(y, y, Math.abs(x));
        }
        for(int i = 1; i<= n; i++)p(t.query(i)+" ");pn("");
    }
    class SegTree{
        int m = 1;
        long[] sum, lazy;
        public SegTree(int n){
            while(m<=n)m<<=1;
            sum = new long[m<<1];
            lazy = new long[m<<1];
        }
        void push(int ll, int rr, int i){
            if(lazy[i] != 0){
                sum[i] += (rr-ll+1)*lazy[i];
                if(ll != rr){
                    lazy[i<<1]+=lazy[i];
                    lazy[i<<1|1] += lazy[i];
                }
                lazy[i] = 0;
            }
        }
        void update(int l, int r, long add){
            if(l > r)return;
            u(l, r, 0, m-1, 1, add);
        }
        void u(int l, int r, int ll, int rr, int i, long add){
            push(ll, rr, i);
            if(l == ll && r == rr){
                lazy[i] += add;
                push(ll, rr, i);
                return;
            }
            int mid = (ll+rr)/2;
            if(r <= mid){
                u(l, r, ll, mid, i<<1, add);
                push(mid+1, rr, i<<1|1);
            }else if(l > mid){
                push(ll, mid, i<<1);
                u(l, r, mid+1, rr, i<<1|1, add);
            }else{
                u(l, mid, ll, mid, i<<1, add);
                u(mid+1, r, mid+1, rr, i<<1|1, add);
            }
            sum[i] = sum[i<<1]+sum[i<<1|1];
            lazy[i] = 0;
        }
        long query(int p){
            return q(p, 0, m-1, 1);
        }
        long q(int p, int ll, int rr, int i){
            push(ll, rr, i);
            if(ll == rr)return sum[i];
            int mid = (ll+rr)/2;
            if(p <= mid)return q(p, ll, mid, i<<1);
            return q(p, mid+1, rr, i<<1|1);
        }
    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }
 
        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }
 
        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }
 
        String nextLine() throws Exception{
            String str = "";
            try{   
                str = br.readLine();
            }catch (IOException e){
                throw new Exception(e.toString());
            }  
            return str;
        }
    }
