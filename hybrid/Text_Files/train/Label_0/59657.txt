import java.util.*;

public class Main
{
    public static void main(String[] args)
    {
        Scanner sc=new Scanner(System.in);
        int n,m;;
        n=sc.nextInt();
        m=sc.nextInt();
        String[] arr1=new String[n];
        String[] arr2=new String[m];
        for(int i=0;i<n;i++)
        arr1[i]=sc.next();
        for(int i=0;i<m;i++)
        arr2[i]=sc.next();
        int q=0;
        q=sc.nextInt();
        String[] ss=new String[q];
        for(int j=0;j<q;j++)
        {
              int curr;
             curr=sc.nextInt();
            int curri=curr%n;
            if(curri==0)
            curri=n;
            int currj=curr%m;
            if(currj==0)
            currj=m;
            String s;s="";
            s=s+arr1[curri-1]+arr2[currj-1];
            ss[j]=s;
            
        }
         for(int j=0;j<q;j++)
         {
             System.out.println(ss[j]);
         }
    }
}
