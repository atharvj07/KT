import java.io.*;
import java.util.*;

/**
 * _223A
 * θ(|s|) time
 * θ(|s|) space
 *
 * @author artyom
 */
public class _223A implements Runnable {
    private BufferedReader in;
    private StringTokenizer tok;

    private Object solve() throws IOException {
        String s = nextToken();
        int n = s.length();
        int[] z = new int[n];
        Arrays.fill(z, -1);
        Deque<Character> stack = new LinkedList<>();
        Deque<Integer> pos = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            char ch = s.charAt(i);
            if (ch == '(' || ch == '[') {
                stack.push(ch);
                pos.push(i);
            } else if (!stack.isEmpty()) {
                int p = pos.pop();
                char x = stack.pop(), match = ch == ')' ? '(' : '[';
                if (x == match) {
                    z[i] = p > 0 && z[p - 1] >= 0 ? z[p - 1] : p;
                } else {
                    stack.clear();
                    pos.clear();
                }
            }
        }
        int max = 0, maxIndex = -1;
        for (int i = z.length - 1; i >= 0; i--) {
            if (z[i] >= 0) {
                int c = 0;
                for (int j = z[i]; j <= i; j++) {
                    if (s.charAt(j) == '[') {
                        c++;
                    }
                }
                if (c > max) {
                    max = c;
                    maxIndex = i;
                }
                i = z[i];
            }
        }
        return max + "\n" + (maxIndex >= 0 ? s.substring(z[maxIndex], maxIndex + 1) : "");
    }

    //--------------------------------------------------------------
    public static void main(String[] args) {
        new _223A().run();
    }

    @Override
    public void run() {
        try {
            in = new BufferedReader(new InputStreamReader(System.in));
            tok = null;
            PrintStream out = System.out;
            out.print(solve());
            in.close();
        } catch (IOException e) {
            System.exit(0);
        }
    }

    private String nextToken() throws IOException {
        while (tok == null || !tok.hasMoreTokens()) {
            tok = new StringTokenizer(in.readLine());
        }
        return tok.nextToken();
    }

    private int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    private long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }
}
