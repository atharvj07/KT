import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

/**
 *
 * @author quangpham
 */
public class D {

    private static final long MAXVAL = 1000000000000000000L;
    private List<Long> Fib;

    private long howManyWays(long n) {
        List<Integer> a = new ArrayList<Integer>();
        for (int i = Fib.size() - 1; i >= 0; --i) {
            if (n >= Fib.get(i)) {
                a.add(i);
                n -= Fib.get(i);
            }
        }

        int L = a.size();
        long[] dp1 = new long[L];
        long[] dp2 = new long[L];

        dp1[L - 1] = 1;
        dp2[L - 1] = a.get(L - 1) / 2;

        for (int i = L - 2; i >= 0; --i) {
            dp1[i] = dp1[i + 1] + dp2[i + 1];
            dp2[i] = (a.get(i) - a.get(i + 1) - 1) / 2 * dp1[i + 1]
                    + (a.get(i) - a.get(i + 1)) / 2 * dp2[i + 1];
        }

        return dp1[0] + dp2[0];
    }

    public void run() {
        Scanner scanner = new Scanner(System.in);

        Fib = new ArrayList<Long>();
        Fib.add(1L);
        Fib.add(2L);
        for (int i = 2; Fib.get(i - 2) + Fib.get(i - 1) <= MAXVAL; ++i) {
            Fib.add(Fib.get(i - 2) + Fib.get(i - 1));
        }

        int nTest = scanner.nextInt();
        for (int test = 0; test < nTest; ++test) {
            long n = scanner.nextLong();
            System.out.println(howManyWays(n));
        }

        scanner.close();
    }

    public static void main(String[] args) {
        new D().run();
    }
}
