#include <bits/stdc++.h>
using namespace std;
int a[200200];
int b[200200];
char enable[200200];
int main(void) {
  int n, i, j, l;
  while (scanf("%d", &n) != EOF) {
    for (i = 0; i < n; i++) {
      scanf("%d", &a[i]);
      b[i] = a[i];
    }
    scanf("%s", enable);
    sort(b, b + n);
    l = 0;
    int ans = 1;
    for (i = 0; i < n && ans; i++) {
      if (enable[i] == '0' || i == n - 1) {
        sort(a + l, a + i + 1);
        for (j = l; j <= i && ans; j++) {
          if (a[j] != b[j]) ans = 0;
        }
        l = i + 1;
      }
    }
    if (ans)
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
