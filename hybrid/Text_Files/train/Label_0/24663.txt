#include <bits/stdc++.h>
using namespace std;
int main() {
  char s[105];
  cin >> s;
  int i, j, n = strlen(s);
  for (i = 1; i < n - 1; i++) {
    int c[3] = {0, 0, 0};
    if (s[i - 1] != '.') c[s[i - 1] - 'A']++;
    if (s[i] != '.') c[s[i] - 'A']++;
    if (s[i + 1] != '.') c[s[i + 1] - 'A']++;
    if (c[0] && c[1] && c[2]) break;
  }
  if (i < n - 1)
    printf("Yes\n");
  else
    printf("No\n");
  return 0;
}
