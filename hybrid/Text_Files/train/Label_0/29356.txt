{
    long long n;
    cin>>n;
    vector<long long> a(n);
    for(long long i=0;i<n;i++)

        cin>>a[i];
    }
    if(!is_sorted(a.begin(),a.end()))
    {
        prev_permutation(a.begin(),a.end());
        for(long long i=0;i<n;i++)
        {
            if(i)
                cout<<" ";
            cout<<a[i];
        }
        cout<<endl;
        next_permutation(a.begin(),a.end());
    }
    for(long long i=0;i<n;i++)
    {
        if(i)
            cout<<" ";
        cout<<a[i];
    }
    cout<<endl;
    if(next_permutation(a.begin(),a.end()))
    {
        for(long long i=0;i<n;i++)
        {
            if(i)
                cout<<" ";
            cout<<a[i];
        }

        prev_permutation(a.begin(),a.end());
    }

}
