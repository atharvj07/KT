import java.util.*;

public class C {
	public static void main(String [] args){
		final Scanner s = new Scanner(System.in);
		final int n = Integer.valueOf(s.nextLine());
		final char [][] table = new char[n][];
		for (int i = 0; i < n; ++i){
			table[i] = s.nextLine().toCharArray(); 
		}
		final List<Integer> cols = new ArrayList<Integer>(n);
		for (int i = 0; i < n; ++i){
			for (int j = 0; j < n; ++j){
				if (table[i][j] != 'E'){
					cols.add(j);
					break;
				}
			}
		}
		if (cols.size() == n){
			for (int i = 0; i < n; ++i){
				System.out.println((i + 1) + " " + (cols.get(i) + 1));
			}
			
		}else{
			final List<Integer> rows = new ArrayList<Integer>(n);
			for (int i = 0; i < n; ++i){
				for (int j = 0; j < n; ++j){
					if (table[j][i] != 'E'){
						rows.add(j);
						break;
					}
				}
			}
			if (rows.size() == n){
				for (int i = 0; i < n; ++i){
					System.out.println((rows.get(i) + 1) + " " + (i + 1));
				}
			}else{
				System.out.println(-1);
			}
		}
		s.close();
	}
}
