#include <bits/stdc++.h>
using namespace std;
const int64_t INF = 1e9 + 7;
bool cmp(const pair<int64_t, int64_t>& a, const pair<int64_t, int64_t>& b) {
  if (a.first == b.first) return a.second > b.second;
  return a.first < b.first;
}
bool neg_cmp(const pair<int64_t, int64_t>& a, const pair<int64_t, int64_t>& b) {
  return a.first + a.second > b.first + b.second;
}
void ans(const bool& a) {
  cout << (a == 0 ? "NO\n" : "YES\n");
  exit(0);
}
signed main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int64_t n, r;
  cin >> n >> r;
  vector<pair<int64_t, int64_t> > pos;
  vector<pair<int64_t, int64_t> > neg;
  for (int64_t i = 0; i < n; ++i) {
    int64_t x, y;
    cin >> x >> y;
    y > 0 ? pos.push_back({x, y}) : neg.push_back({x, y});
  }
  sort((pos).begin(), (pos).end(), cmp);
  sort((neg).begin(), (neg).end(), neg_cmp);
  for (int64_t i = 0; i < pos.size(); ++i) {
    if (r >= pos[i].first)
      r += pos[i].second;
    else
      ans(0);
  }
  for (int64_t i = 0; i < neg.size(); ++i) {
    if (r >= neg[i].first)
      r += neg[i].second;
    else
      ans(0);
  }
  ans(r < 0 ? 0 : 1);
  return 0;
}
