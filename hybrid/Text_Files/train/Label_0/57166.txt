        public void addop(char o, long k)
        {
            if (o == 'P')
            {
                cp += k;
            }
            
            while ((c < M)&&(k>0))
            {
                R[(int)c] = (byte)o;
                c ++;
                k --;
            }
            
            c += k;
            
        }
        
        public void addop(char o)
        {
            addop(o, 1);
        }
        
        public void output_ops()
        {
            if (c>M)

                return;

            int cc = (int)c;
            
            out.printf("SMALL\n%d\n", c);
            byte[] b2 = new byte[cc];
            for (int i=0; i<cc; i++)
            {
                b2[cc-i-1] = R[i];
            }
            
            out.println(new String(b2));
        }

    public String readLn() {

            String s;

            return "";

    }

    public String readToken() {
        try {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                tokenizer = new StringTokenizer(in.readLine());
            }
            return tokenizer.nextToken();
        } catch (Exception e) {
            return "";
        }
    }

    public void printArray(long[] a, int n, char dl)
    {
        long x; 
        int i, l = 0;
        for (i=0; i<n; i++)
        {
            x = a[i];
            
            if (x<0)
            {
                x = -x;
                l++;
            }
            
            if (x==0)
            {
                l++;
            }
            else
            {
                while (x>0)
                {
                    x /= 10;
                    l++;
                }
            }
        }
        
        l += n-1;
        
        char[] s = new char[l];
        
        l--;
        boolean z;
        for (i=n-1; i>=0;  i--)
        {
            x = a[i];
            z = false;            
            if (x<0)
            {
                x = -x;
                z = true;
            }
            
            do{
                s[l--] = (char)('0' + (x % 10));
                x /= 10;
            } while (x>0);
            
            if (z)
            {
                s[l--] = '-';
            }
            
            if (i>0)
            {
                s[l--] = dl;
            }
        }
        
        out.println(new String(s));
    }
