import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        FastScanner in = new FastScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        int n;
        int[] answer;

        void makeRequest(ArrayList<Integer> req, FastScanner in, PrintWriter out) {
            out.println(req.size());
            for (int x : req) {
                out.print((x + 1) + " ");
            }
            out.println();
            out.flush();
            for (int i = 0; i < n; i++) {
                answer[i] = in.nextInt();
            }
        }

        public void solve(int testNumber, FastScanner in, PrintWriter out) {
            n = in.nextInt();
            answer = new int[n];
            int[] result = new int[n];
            Arrays.fill(result, Integer.MAX_VALUE);
            ArrayList<TaskD.Request> requests = new ArrayList<>();
            requests.add(new TaskD.Request(0, n));
            while (!requests.isEmpty()) {
                ArrayList<Integer> lReq = new ArrayList<>();
                for (TaskD.Request r : requests) {
                    int middle = (r.from + r.to) / 2;
                    for (int j = middle; j < r.to; j++) {
                        lReq.add(j);
                    }
                }
                makeRequest(lReq, in, out);
                for (TaskD.Request r : requests) {
                    int middle = (r.from + r.to) / 2;
                    for (int j = r.from; j < middle; j++) {
                        result[j] = Math.min(result[j], answer[j]);
                    }
                }
                ArrayList<Integer> rReq = new ArrayList<>();
                for (TaskD.Request r : requests) {
                    int middle = (r.from + r.to) / 2;
                    for (int j = r.from; j < middle; j++) {
                        rReq.add(j);
                    }
                }
                makeRequest(rReq, in, out);
                for (TaskD.Request r : requests) {
                    int middle = (r.from + r.to) / 2;
                    for (int j = middle; j < r.to; j++) {
                        result[j] = Math.min(result[j], answer[j]);
                    }
                }

                ArrayList<TaskD.Request> newRequests = new ArrayList<>();

                for (TaskD.Request r : requests) {
                    int middle = (r.from + r.to) / 2;
                    if (middle - r.from > 1) {
                        newRequests.add(new TaskD.Request(r.from, middle));
                    }
                    if (r.to - middle > 1) {
                        newRequests.add(new TaskD.Request(middle, r.to));
                    }
                }

                requests = newRequests;
            }
            out.println(-1);
            for (int i = 0; i < n; i++) {
                out.print(result[i] + " ");
            }
            out.println();
            out.flush();
        }

        static class Request {
            int from;
            int to;

            public Request(int from, int to) {
                this.from = from;
                this.to = to;
            }

        }

    }

    static class FastScanner {
        private BufferedReader br;
        private StringTokenizer st = null;

        public FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public boolean hasNext() {
            while (st == null || !st.hasMoreTokens()) {
                String line = null;
                try {
                    line = br.readLine();
                } catch (IOException e) {
                    e.printStackTrace();
                }
                if (line == null)
                    return false;
                st = new StringTokenizer(line);
            }
            return true;
        }

        public String next() {
            if (!hasNext()) {
                return null;
            }
            return st.nextToken();
        }

    }
}
