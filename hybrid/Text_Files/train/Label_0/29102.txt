#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<ll, ll> pi;
#define mp make_pair
vector<ll> d;
int main()
{
//freopen("input.txt", "r", stdin);
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);cout.tie(NULL);
    ll n, k, x; cin >> n >> k >> x;
	ll a[n+1]; for(int i=1;i<=n;i++) cin >> a[i];
	sort(a+1, a+n+1);
	for(int i=1;i<n;i++)
	{
		//cout << a[i] << " " << a[i+1] << "\n";
		if((a[i+1]-a[i])>x)
		d.push_back((a[i+1]-a[i]-1)/x);
	}
	sort(d.begin(), d.end());
	ll ans=1;

	//cout << d.size() << "\n";
	for(int i=0;i<d.size();i++)
	{
		//cout << i << " ";
		//cout << d[i] << " " << k << "\n";
		if(d[i]<=k) { k-=d[i]; }
		else { ans=d.size()-i+1; break; }

	}
	cout << ans;
}
