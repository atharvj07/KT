(n, k, x) = map(int, input().split())
array = sorted(list(map(int, input().split())))
array2 = []
for i in range(n - 1):
	val = (array[i + 1] - array[i] - 1) // x
	if val > 0:
		array2.append(val)
if len(array2) == 0:
	print(1)
else:
	array2.sort()
	cum = 0
	for i in range(len(array2)):
		cum += array2[i]
		if cum > k:
			print(len(array2) - i + 1)
			break
	else:
		print(1)
