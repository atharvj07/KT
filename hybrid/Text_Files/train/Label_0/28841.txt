#include<stdio.h>
#include<iostream>
#include<algorithm>
#include<memory.h>
#include<vector>
#include<functional>
using namespace std;

int main()
{
	int dp[2001];
	int n, k, a, b;
	vector<int> c[11], g[11];
	memset(dp, 0, sizeof(dp));
	cin >> n >> k;
	for (int i = 0; i<n; i++){
		cin >> a >> b; b--;
		g[b].push_back(a);
	}

	for (int i = 0; i<10; i++){
		sort(g[i].begin(), g[i].end(), greater<int>());
	}

	for (int i = 0; i<10; i++){
		if (!g[i].empty()){
			c[i].push_back(g[i].front());
		}
		else{
			c[i].push_back(0);
		}
		for (int i2 = 1; i2<g[i].size(); i2++){
			c[i].push_back(c[i][i2 - 1] + g[i][i2] + ((i2 + 1) - 1) * 2);
		}
	}
	for (int j = 0; j<10; j++){
		for (int i = k; i >= 0; i--){
			for (int i2 = 0; (i2<c[j].size()) && (i - i2 - 1 >= 0); i2++){
				dp[i] = max(dp[i], dp[i - i2 - 1] + c[j][i2]);
			}
		}
	}
	cout << dp[k] << endl;
}
