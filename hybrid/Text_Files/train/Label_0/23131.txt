#include <bits/stdc++.h>
using namespace std;

typedef map<int, set<int> > msi;

#define F first
#define S second

const int inf = 2e9;

int n, ans;
msi xup, xdown, yleft, yright;
msi upm, upp, downm, downp;
msi leftm, leftp, rightm, rightp;

void relax(msi &a, msi &b, int mult) {
	for (auto &t: a) {
		if (!b.count(t.F)) continue;
		for (int i: t.S) {
			auto it = b[t.F].lower_bound(i);
			if (it == b[t.F].end()) continue;
			ans = min(ans, (*it - i) * mult);
		}
	}
}

int main() {
	cin >> n;
	for (int i = 0; i < n; i++) {
		int x, y; char d;
		cin >> x >> y >> d;
		if (d == 'U') {
			xup[x].insert(y);
			upm[x - y].insert(y);
			upp[x + y].insert(y);
		} else if (d == 'D') {
			xdown[x].insert(y);
			downm[x - y].insert(y);
			downp[x + y].insert(y);
		} else if (d == 'L') {
			yleft[y].insert(x);
			leftm[x - y].insert(y);
			leftp[x + y].insert(y);
		} else {
			yright[y].insert(x);
			rightm[x - y].insert(y);
			rightp[x + y].insert(y);
		}
	}
	ans = inf;
	relax(xup, xdown, 5);      
	relax(yright, yleft, 5);  
	relax(upm, leftm, 10);    
	relax(upp, rightp, 10);    
	relax(rightm, downm, 10);  
	relax(leftp, downp, 10);  
	if (ans == inf) cout << "SAFE";
	else cout << ans;
}
