def main():
	a = input().split()
	for i in range(len(a)):
		a[i] = int(a[i])
	m = a[1]
	n = a[0]
	if n == 0:
		c1 = {(0, 0), (0, 1), (0, m - 1), (0, m)}
	elif m == 0:
		c1 = {(0, 0), (1, 0), (n - 1, 0), (n, 0)}
	else:
		c1 = {(0, 0), (0, 1), (1, 0), (1, 1), (0, m - 1), (0, m), (1, m - 1), (1, m), (n - 1, m - 1), (n - 1, m), (n, m - 1), (n, m), (n - 1, 0), (n, 0), (n - 1, 1), (n, 1)}
	maximum = 0
	for i in c1:
		for j in c1 - {i}:
			for k in c1 - {i, j}:
				for l in c1 - {i, j, k}:
					maximum = max(polyline(i, j, k, l), maximum)
	for i in c1:
		for j in c1 - {i}:
			for k in c1 - {i, j}:
				for l in c1 - {i, j, k}:
					if maximum == polyline(i, j, k, l):
						print(l[0], l[1])
						print(k[0], k[1])
						print(j[0], j[1])
						print(i[0], i[1])
						return

def dis(i, j):
	return ((i[0] - j[0]) ** 2 + (i[1] - j[1]) ** 2) ** 0.5

def polyline(i, j, k, l):
	return dis(i, j) + dis(j, k) + dis(k, l)
main()
