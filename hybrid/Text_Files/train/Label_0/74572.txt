import sys
input = sys.stdin.buffer.readline
for _ in range(int(input())):
	n = int(input())
	(a, b) = (int(x) for x in input().split())
	s = 0
	min_diff = float('inf')
	for i in range(n - 1):
		(an, bn) = (int(x) for x in input().split())
		dmg = max(0, an - b)
		s += dmg
		min_diff = min(min_diff, an - dmg)
		b = bn
	dmg = max(0, a - b)
	s += dmg
	min_diff = min(min_diff, a - dmg)
	sys.stdout.write(str(s + min_diff) + '\n')
