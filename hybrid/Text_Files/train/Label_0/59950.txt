import java.util.*;
import java.io.*;

public class C{

    public static void main(String[] args) throws Exception{
        new C().run();
    }

    private void fail() {
        throw new IllegalArgumentException();
    }

    private void print(String s){
        System.out.println(s);
    }

    private void run() throws Exception{
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String N = reader.readLine();
        int n = Integer.parseInt(N);
        Node[] g = new Node[n+1];
        for(int i= 0; i <= n; i++){
            g[i] = new Node(i);
        }
        for(int i = 0; i < n-1;i++){
            String[] ab = (reader.readLine()).split(" ");
            int a = Integer.parseInt(ab[0]);
            int b = Integer.parseInt(ab[1]);
            g[a].children.add(g[b]);
            g[b].children.add(g[a]);
        }
        Node startnode = g[1];
        for(int i = 0; i < n;i++){
            if(g[i].children.size() > 2){
                startnode = g[i];
                break;
            }
        }

        startnode.children.add(startnode);
        ArrayList<Path> paths = startnode.dfs(startnode, startnode);
        if(paths != null){
            print("Yes");
            System.out.println(paths.size());
            for(Path p:paths){
                print(p.toString());
            }
        }
        else{
            print("No");
        }

    }

    class Path{
        int start, end;
        public Path(int s, int e){
            start = s;
            end = e;
        }
        public String toString(){
            return ""+start + " " + end;
        }
    }

    class Node{
        public int id;
        public Node(int id){
            this.id =id;
        }

        public String toString(){
            return ""+id;
        }



        ArrayList<Node> children = new ArrayList<Node>();

        public ArrayList<Path> dfs(Node from, Node startpath){
            ArrayList<Path> paths = new ArrayList<Path>();

            if(children.size() == 1){
                // leaf
                paths.add(new Path(startpath.id, id));
                return paths;
            }
            if(children.size() > 2){
                for(Node c: children){
                    if(c.id !=from.id){
                        // not the parent
                        ArrayList<Path> cpaths = c.singleDfs(this, startpath);
                        if(cpaths == null) return null;
                        paths.addAll(cpaths);
                    }
                }
            }else{
                // exactly 2: one from parent and one down
            for(Node c: children){
                if(c.id !=from.id){
                    // not the parent
                    ArrayList<Path> cpaths = c.dfs(this, startpath);
                    if(cpaths == null) return null;
                    paths.addAll(cpaths);
                }
            }
        }
            return paths;
        }

       public ArrayList<Path> singleDfs(Node from, Node startpath){

           ArrayList<Path> paths = new ArrayList<Path>();
           if(children.size() == 1){
               // leaf
               paths.add(new Path(startpath.id, id));
        //       print("I leaf " + id);
               return paths;
           }
           if(children.size() > 2) {
        //       print("Returns null " + id + " size of paths " + children.size());

               return null;}

           for(Node c: children){
               if(c.id !=from.id){
                   // not the parent
                   ArrayList<Path> cpaths = c.singleDfs(this, startpath);
                //   print("In " + id + " size of paths " + cpaths.size() + "parent " + c.id + " from id " + from.id) ;
                   if(cpaths == null) return null;
                   paths.addAll(cpaths);
               }
           }

           return paths;
       }
    }
}
