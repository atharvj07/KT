    void run() throws IOException {
        int n = nextInt();
        int[] xor_for_p = new int[n];
        for (int i = 0; i < n; i++) {
            pw.println("? " + 0 + " " + i);
            xor_for_p[i] = nextInt();
        }
        int[] xor_for_b = new int[n];
        for (int i = 0; i < n; i++) {
            pw.println("? " + i + " " + 0);
            xor_for_b[i] = nextInt();
        }
        int cnt = 0;
        int[] check_b = new int[n];
        int[] check_p = new int[n];
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            check_b[0] = i;
            check_p[0] = xor_for_b[0] ^ i;
            for (int j = 1; j < n; j++) {
                check_p[j] = xor_for_b[j] ^ i;
            }
            for (int j = 1; j < n; j++) {
                check_b[j] = xor_for_p[j] ^ check_p[0];
            }
            boolean flag = true;
            int[] a = new int[n];
            for (int j = 0; j < check_b.length; j++) {
                int x = check_b[j];
                if (x > n - 1) continue;
                a[x]++;
            }
            int[] b = new int[n];
            for (int j = 0; j < check_p.length; j++) {
                int x = check_p[j];
                if (x > n - 1) continue;
                b[x]++;
            }
            for (int j = 0; j < a.length; j++) {
                if (a[j] != 1) flag = false;
            }
            for (int j = 0; j < b.length; j++) {
                if (b[j] != 1) flag = false;
            }
            if (!flag) continue;
            for (int j = 0; j < check_b.length; j++) {
                if (check_p[check_b[j]] != j) flag = false;
            }
            if (!flag) continue;
            for (int j = 0; j < ans.length; j++) {
                ans[j] = check_p[j];
            }
            cnt++;
        }
        pw.println("!");
        pw.println(cnt);
        for (int i = 0; i < ans.length; i++) {
            pw.print(ans[i] + " ");
        }
        pw.close();
    }

    class point implements Comparable<point> {
        long sum;
        int h;

        public point(long a, int b) {
            sum = a;
            h = b;
        }

        @Override
        public int compareTo(point o) {
            return Long.compare(this.sum, o.sum);
        }
    }
