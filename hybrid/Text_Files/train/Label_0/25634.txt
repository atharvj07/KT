#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n], b[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    int key = a[i];
    b[i] = 1;
    for (int j = 0; j < n; j++) {
      if (key < a[j]) b[i]++;
    }
  }
  for (int i = 0; i < n; i++) cout << b[i] << " ";
}
