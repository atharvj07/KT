#include <bits/stdc++.h>
using namespace std;
int min(int *arr, int n) {
  int min;
  int a = *arr;
  for (int i = 0; i < n; i++) {
    if (a >= *(arr + i)) {
      a = *(arr + i);
      min = i;
    }
  }
  return min;
}
int max(int *arr, int n) {
  int max;
  int a = *arr;
  for (int i = n - 1; i >= 0; i--) {
    if (a <= *(arr + i)) {
      a = *(arr + i);
      max = i;
    }
  }
  return max;
}
int main() {
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  int mi = min(arr, n);
  int mx = max(arr, n);
  if (mi < mx)
    cout << n - 2 - mi + mx;
  else
    cout << n - 1 - mi + mx;
  return 0;
}
