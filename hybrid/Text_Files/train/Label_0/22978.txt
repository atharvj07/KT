#include <bits/stdc++.h>
using namespace std;
long long n, m;
bool f(long long x) {
  long long j = 0, sum = 0, sum2 = 0, cnt = 0, cnt2 = 0, EN = 1, mt = m * x;
  while (mt >= j * EN && j <= x) {
    mt -= j * EN;
    sum += (EN);
    j++;
    if (sum > 1e9) break;
    EN = EN * (x - j + 1) / (j);
  }
  if (j != x + 1) sum += mt / j;
  return sum >= n;
}
int main() {
  int q = 0;
  cin >> q;
  while (q--) {
    cin >> n >> m;
    if (n == 1) {
      cout << 0 << "\n";
      continue;
    }
    long long d = -1, up = n;
    while (up - d > 1) {
      long long mid = (up + d) / 2;
      if (f(mid)) {
        up = mid;
      } else {
        d = mid;
      }
    }
    cout << up << "\n";
  }
  return 0;
}
