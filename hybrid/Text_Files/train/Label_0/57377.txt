import static java.lang.System.out;

import java.util.Arrays;
import java.util.Scanner;

public class Main {
	public static void main(String[] args){
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] s = new int[n];
		int m = 0;
		for (int i=0;i<n;i++)
		{
			s[i] = sc.nextInt();
			m = Math.max(m, s[i]);
		}
		Arrays.sort(s);
		int ans = 1;
		int len = 1;
		for (int i=1;i<n;i++)
		{
			if (s[i]==s[i-1] && s[i]!=1){
				len++;
			}else{
				ans = Math.max(ans, len);
				len=1;
			}
		}
		ans = Math.max(ans, len);
		for (int i=2;i*i<=m;i++)
		{
			int c1 = 0;
			int c2 = 0;
			int j = m/i;
			for (int x : s)
			{
				if (x%i==0) c1++;
				if (j>0 && x%j==0) c2++;
			}
			ans = Math.max(ans, Math.max(c1, c2));
		}
		out.println(ans);
		sc.close();
	}
}
