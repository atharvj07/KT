import java.util.Scanner;

public class Camp_Schedule {

	public static void main(String[] args) {
		Scanner scn = new Scanner(System.in);
		String s = scn.next();
		String t = scn.next();
		long p =1 ;
		long mod = 1000000007;
		long[] hash = new long[t.length()];
		long[] pp = new long[t.length()];
		int not1 = 0;
		int not0=0;
		
		for (int i = 0; i < t.length(); i++) {
			if (t.charAt(i) == '1') {
				not1++;
			}else{
				not0++;
			}
			hash[i] = ((i>0?hash[i - 1]:0) + (t.charAt(i)-'0'+1) * p) % mod;
			pp[i] = p;
			p = (p * 997) % mod;

		}
		if(s.length()<t.length()){
			System.out.println(s);
			return;
		}
		int ind = t.length();
		for (int i = 2; i < t.length(); i++) {
			long inver = expo(pp[i], mod - 2, mod);
			long com = (inver * ((hash[t.length() - 1] - hash[i - 1]+mod)%mod)) % mod;
			long pre = hash[t.length() - 1 - i];
			if (com == pre) {
				ind = i;
				break;
			}
		}
		int c1=0,c0 = 0;
		StringBuilder apd = new StringBuilder();

		for (int i = t.length() - ind; i < t.length(); i++) {
			char ch = t.charAt(i);
			if (ch == '1') {
				c1++;
			}else{
				c0++;
			}
			apd.append(ch);
		}

		int no1=0,no0 = 0;
		
		for (int i = 0; i < s.length(); i++) {
			char ch = s.charAt(i);
			if (ch == '1') {
				no1++;
			}else{
				no0++;
			}
		}
		if (s.length() < t.length()) {
			System.out.println(s);
		} else if (s.length() >= t.length() && no1 >= not1&&no0>=not0) {
			StringBuilder sb = new StringBuilder();
			sb.append(t);
			no1 -= not1;
            no0-=not0;
			while (no1 >= c1&&no0>=c0) {
				sb.append(apd);
				no1 -= c1;
				no0-=c0;
			}
			while (no1 > 0) {
				sb.append('1');
				no1--;
			}

			
				while (no0> 0) {
					sb.append('0');
					no0--;
				}
			
			System.out.println(sb);
		}else{
			System.out.println(s);
		}
	}

	public static long expo(long a, long b, long mod) {
		if (b == 0) {
			return 1;
		}
		long val = expo(a, b / 2, mod);
		val = (val * val) % mod;
		if (b % 2 != 0) {
			return (a * val) % mod;
		}
		return val;
	}

}
