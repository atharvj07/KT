#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

typedef pair<int, int> P; // <val, dist>
P dp[11][1 << 15];

struct S {
	int v, buy;
	P d;

	bool operator <(const S& x) const {
		return d > x.d;
	}
};

P rec(S s) {
	return dp[s.v][s.buy];
}

void save(S s) {
	dp[s.v][s.buy] = s.d;
}

struct edge {
	int to, d;
};

int main() {
	cin.tie(0);
	ios::sync_with_stdio(false);

	int N, Q, M;
	while(cin >> N, N) {
		map<string, int> store[10];
		for(int i = 0; i < N; i++) {
			int k;
			cin >> k;
			for(int j = 0; j < k; j++) {
				string name;
				int val;
				cin >> name >> val;
				store[i][name] = val;
			}
		}
		cin >> Q;
		vector<string> need(Q);
		for(int i = 0; i < Q; i++) {
			cin >> need[i];
		}
		cin >> M;
		vector<edge> G[11];
		for(int i = 0; i < M; i++) {
			int a, b, d;
			cin >> a >> b >> d;
			a--, b--;
			if(a == -1) a = N;
			if(b == -1) b = N;
			G[a].push_back({ b, d });
			G[b].push_back({ a, d });
		}

		priority_queue<S> q;
		S s = { N, 0, { 0, 0 } };
		q.push(s);
		const int INF = 1 << 25;
		fill((P*)begin(dp), (P*)end(dp), P{ INF, INF });
		save(s);

		while(q.size()) {
			s = q.top();
			q.pop();
			if(rec(s) < s.d) continue;
			if(s.v != N) {
				for(int i = 0; i < Q; i++) {
					if(s.buy >> i & 1) continue;
					if(store[s.v].count(need[i]) == 0) continue;
					S ns = s;
					ns.buy |= 1 << i;
					ns.d.first += store[s.v][need[i]];
					if(rec(ns) > ns.d) {
						save(ns);
						q.push(ns);
					}
				}
			}

			for(auto e : G[s.v]) {
				S ns = s;
				ns.v = e.to;
				ns.d.second += e.d;
				if(rec(ns) > ns.d) {
					save(ns);
					q.push(ns);
				}
			}
		}

		P ans = dp[N][(1 << Q) - 1];
		if(ans.first == INF) {
			cout << "impossible" << endl;
		}
		else {
			cout << ans.first << " " << ans.second << endl;
		}
	}
}
