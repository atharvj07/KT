#include <bits/stdc++.h>
using namespace std;
int main() {
  int c[6], t = 0;
  string s = "NO";
  for (int i = 0; i < 6; i++) {
    cin >> c[i];
    t += c[i];
  }
  if (t % 2 == 0) {
    t /= 2;
    for (int i = 1; i < 6; i++) {
      for (int j = 1; j < 6; j++) {
        if ((c[0] + c[i] + c[j] == t) && (i != j)) s = "YES";
      }
    }
  }
  cout << s;
  return 0;
}
