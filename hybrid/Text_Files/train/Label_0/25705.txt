#include <bits/stdc++.h>
using namespace std;
void prep(string &s) {
  for (int i = 0; i < s.size(); ++i) {
    if (s[i] == 'O' || s[i] == 'o' || s[i] == '0')
      s[i] = '0';
    else if (s[i] == 'I' || s[i] == 'i' || s[i] == 'L' || s[i] == 'l' ||
             s[i] == '1')
      s[i] = '1';
    else if (s[i] >= 'a' && s[i] <= 'z')
      s[i] -= 'a' - 'A';
  }
}
int main() {
  string s, p;
  int n;
  cin >> p >> n;
  set<string> dat;
  for (int i = 0; i < n; ++i) {
    cin >> s;
    prep(s);
    dat.insert(s);
  }
  prep(p);
  dat.insert(p);
  if (dat.size() == n)
    cout << "No";
  else
    cout << "Yes";
  return 0;
}
