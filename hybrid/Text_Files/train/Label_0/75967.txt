from math import ceil
(hh, mm) = map(int, input().split())
(hunger, hpm, cost, feed) = map(float, input().split())
price = ceil(hunger / feed) * cost * (1 if hh < 20 else 0.8)
if hh < 20:
	minutes = (20 - hh) * 60 - mm
	evening_hunger = hunger + minutes * hpm
	discount_price = ceil(evening_hunger / feed) * 0.8 * cost
	price = min(price, discount_price)
print(price)
