def get_ints():
	return map(int, input().split())

def palindrome(s):
	mid = len(s) // 2
	for i in range(mid):
		if s[i] != s[len(s) - i - 1]:
			return False
	return True

def solve(a, b, n):
	invflg = 0
	i = n - 1
	inv = [False for i in range(n)]
	count0 = 0
	count1 = 0
	for i in range(n):
		if a[i] == '0':
			count0 += 1
		else:
			count1 += 1
		if count0 == count1:
			inv[i] = True
		else:
			inv[i] = False
	while i >= 0:
		if invflg == 0:
			if a[i] == b[i]:
				i -= 1
			else:
				if inv[i] == False:
					print('NO')
					return
				invflg = 1
				i -= 1
		elif invflg == 1:
			if a[i] != b[i]:
				i -= 1
			else:
				if inv[i] == False:
					print('NO')
					return
				invflg = 0
				i -= 1
	print('YES')
t = int(input())
for tt in range(t):
	n = int(input())
	a = input()
	b = input()
	solve(a, b, n)
