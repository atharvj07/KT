for _ in range(int(input())):
	s = input()
	t = input()
	if t[0] not in s or t[1] not in s or t[2] not in s:
		print(''.join(sorted(s)))
	elif t != 'abc':
		print(''.join(sorted(s)))
	else:
		a1 = s.count('a')
		b1 = s.count('b')
		c1 = s.count('c')
		s1 = sorted(s)
		viv = ''.join(s1[a1 + c1 + b1:])
		print(a1 * 'a' + c1 * 'c' + b1 * 'b' + viv)
