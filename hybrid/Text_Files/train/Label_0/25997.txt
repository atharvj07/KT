#define int long long

#define rep(i,n) for(int i=0;i<(n);i++)
#define all(v) (v).begin(),(v).end()
#define pb push_back
#define fi first
#define se second

typedef vector<int>vint;

template<typename A,typename B>inline void chmax(A &a,B b){if(a<b)a=b;}

int N,M;
int H[111][111];

struct edge{
    int to,cap,cost,rev;
    edge(int a,int b,int c,int d):to(a),cap(b),cost(c),rev(d){}
};

const int INF=1001001001;

const int MV=300;
int src=MV-2,snk=MV-1;
vector<edge>G[MV];
int dist[MV];
int prevv[MV],preve[MV];

void init(){
    rep(i,MV)G[i].clear();
}

void add_edge(int from,int to,int cap,int cost){
    G[from].pb(edge(to,cap,cost,G[to].size()));
    G[to].pb(edge(from,0,-cost,G[from].size()-1));
}

int min_cost_flow(int s,int t,int f){
    int res=0;
    while(f>0){
        fill(dist,dist+MV,INF);
        dist[s]=0;
        bool update=true;
        while(update){
            update=false;
            for(int v=0;v<MV;v++){
                if(dist[v]==INF)continue;
                for(int i=0;i<G[v].size();i++){
                    edge &e=G[v][i];
                    if(e.cap>0&&dist[e.to]>dist[v]+e.cost){
                        dist[e.to]=dist[v]+e.cost;
                        prevv[e.to]=v;
                        preve[e.to]=i;
                        update=true;
                    }
                }
            }
        }
        if(dist[t]==INF)return -1;
        int d=f;
        for(int v=t;v!=s;v=prevv[v]){
            d=min(d,G[prevv[v]][preve[v]].cap);
        }

        f-=d;
        res+=d*dist[t];
        for(int v=t;v!=s;v=prevv[v]){
            edge &e=G[prevv[v]][preve[v]];
            e.cap-=d;
            G[v][e.rev].cap+=d;
        }
    }
    return res;
}

bool C(int x){
    int t=M/N;
    init();
    rep(i,N)add_edge(src,i,t,0);
    rep(i,N)add_edge(src,i,1,1);

    rep(i,N)rep(j,M)if(H[i][j]<=x)add_edge(i,j+N,1,0);
    rep(i,M)add_edge(i+N,snk,1,0);

    int tmp=min_cost_flow(src,snk,M);
    if(tmp==-1)return false;
    if(tmp>M%N)return false;
    return true;
}

signed main(){
    cin>>N>>M;
    rep(i,N)rep(j,M)cin>>H[i][j];

    int lb=0,ub=1001001001;
    while(ub-lb>1){
        int mid=(ub+lb)/2;
        if(C(mid))ub=mid;
        else lb=mid;
    }
    cout<<(M%N!=0)<<endl;
    cout<<ub<<endl;
    return 0;
}
