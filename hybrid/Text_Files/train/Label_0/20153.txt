#include <cstdio>
typedef long long ll;

int p[1024];
bool mrk[8096];

int main() {
	int n; scanf("%d",&n);
	int *q = p+1 + (n<<1);
	*p = *q = 1;
	for (int x = 2, *t = p+1; t < q; x++) {
		if (!mrk[x]) *t++ = x;
		for (int *i = p+1; i < t && x**i < 8096; i++) {
			mrk[x**i] = true;
			if (!(x % *i)) break;
		}
	}
	q = p+1 + n + (n>>1);
	for (int x = 0; x < n; x++, putchar(10))
		for (int y = 0; y < n; y++) {
			int s = x+y >> 1, d = x-y >> 1;
			if ((x ^ y) & 1) printf("%d ", p[s+1] * q[d]);
			else printf("%lld ", ll(p[s+1] * p[s]) * ll(q[d] * q[d-1]) + 1);
		}
	return 0;
}
