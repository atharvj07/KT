s = input().strip()
m = {}
for i in range(10):
	m[str(i)] = i
for i in range(ord('Z') - ord('A') + 1):
	m[chr(ord('A') + i)] = 10 + i
for i in range(ord('z') - ord('a') + 1):
	m[chr(ord('a') + i)] = 36 + i
m['-'] = 62
m['_'] = 63
zero_bits = 0
for c in s:
	n = m[c]
	for i in range(6):
		zero_bits += int(not n & 1)
		n >>= 1
res = 1
for z in range(zero_bits):
	res = res * 3 % 1000000007
print(res)
