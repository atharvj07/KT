import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        FastScanner scanner = new FastScanner();
        PrintWriter out = new PrintWriter(System.out);

        int t = scanner.nextInt();
        while (t-- > 0) {
            char[] arr = scanner.next().toCharArray();
            int n = arr.length;
            int[] freq = new int[26];
            for (char ch : arr)
                ++freq[ch - 'a'];
            HashSet<Character> hitlist = new HashSet<>();
            boolean[] remove = new boolean[n];
            int start = 0;
            int end;
            int[] seen = new int[26];
            char maxChar = arr[0];
            for (int i = 0; i < n; ++i) {
                if (freq[arr[i] - 'a'] == 1 || i == n - 1 || seen[arr[i] - 'a'] == freq[arr[i] - 'a'] - 1) {
                    if (hitlist.contains(arr[i])) {
                        remove[i] = true;
                    } else {
                        end = i;
                        if (arr[i] > maxChar)
                            maxChar = arr[i];
                        for (int j = start; j <= end; ++j) {
                            if (arr[j] == maxChar) {
                                hitlist.add(maxChar);
                                start = j + 1;
                                i = j;
                                maxChar = start < n ? arr[start] : 'a';
                                seen = new int[26];
                                break;
                            } else {
                                remove[j] = true;
                                --freq[arr[j] - 'a'];
                            }
                        }
                    }
                } else {
                    ++seen[arr[i] - 'a'];
                    if (hitlist.contains(arr[i]))
                        remove[i] = true;
                    else if (arr[i] > maxChar)
                        maxChar = arr[i];
                }
            }
            StringBuilder ans = new StringBuilder();
            for (int i = 0; i < n; ++i)
                if (!remove[i])
                    ans.append(arr[i]);
            out.println(ans.toString());
        }
        out.close();
    }

    // Shuffle and sort array
    static void shuffleAndSort(int[] arr) {
        for (int i = 0; i < arr.length; ++i) {
            int random_index = (int) (Math.random() * arr.length);
            int temp = arr[i];
            arr[i] = arr[random_index];
            arr[random_index] = temp;
        }
        Arrays.sort(arr);
    }

    // Fast IO
    static class FastScanner {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer("");

        String next() {
            while (!st.hasMoreTokens())
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        int[] readArray(int n) {
            int[] a = new int[n];
            for (int i = 0; i < n; ++i) a[i] = nextInt();
            return a;
        }

        long[] readLongArray(int n) {
            long[] a = new long[n];
            for (int i = 0; i < n; ++i) a[i] = nextLong();
            return a;
        }
    }
}
