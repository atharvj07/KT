from itertools import chain, combinations
T = int(input())
for _ in range(T):
	N = int(input())
	arr = [int(x) for x in input().split()]
	powerSet = chain.from_iterable((combinations(arr, r) for r in range(N + 1)))
	powerSetSums = set([sum(i) for i in powerSet])
	if len(powerSetSums) < 2 ** N:
		print('YES')
	else:
		print('NO')
