def readint():
	return int(input())

def readline():
	return [int(c) for c in input().split()]

def main():
	n = readint()
	string = input()
	if len(set(string)) == 1:
		print(string[:n - 1])
		return
	for i in range(1, n):
		if string[i] < string[i - 1]:
			print(string[:i - 1] + string[i:])
			return
	string = list(string)
	del string[string.index(max(string))]
	print(''.join(string))
main()
