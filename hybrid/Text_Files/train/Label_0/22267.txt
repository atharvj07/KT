#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int t;
  cin >> t;
  while (t != 0) {
    long long int n, x, i;
    cin >> n;
    n /= 2;
    if (n % 2 != 0) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
      for (i = 0; i < n / 2; i++) {
        cout << 2 + i * 2 << " ";
      }
      for (i = n / 2; i < n; i++) {
        cout << n + 4 + (i - n / 2) * 2 << " ";
      }
      for (i = 0; i < n; i++) {
        cout << 3 + i * 2 << " ";
      }
      cout << endl;
    }
    t--;
  }
}
