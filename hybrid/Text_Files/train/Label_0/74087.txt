n = int(input())
s = []
for i in range(n):
	s.append(list(input()))
for i in range(n - 2):
	for j in range(n - 1):
		if s[i][j] == '.':
			for (x, y) in ((0, 0), (-1, 1), (0, 1), (1, 1), (0, 2)):
				if s[i + y][j + x] == '.':
					s[i + y][j + x] = '#'
				else:
					print('NO')
					exit()
for i in range(n):
	if s[i].count('.'):
		print('NO')
		exit()
print('YES')
