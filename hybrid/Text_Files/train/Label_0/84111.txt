n = int(input())
s = input()
pos = set([i for i in s if i != '*'])
m = int(input())
first = False
ans = set()
for _ in range(0, m):
	nott = False
	mm = input()
	check = []
	for i in range(0, n):
		if s[i] != '*' and s[i] != mm[i]:
			nott = True
			break
	new = set([mm[i] for i in range(0, n) if s[i] == '*'])
	if len(new & pos) > 0 or nott:
		continue
	if not first:
		ans = new
		first = True
	else:
		ans = ans & new
print(len(ans))
