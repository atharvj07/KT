#include <bits/stdc++.h>
using namespace std;

#define repl(i,a,b) for(int i=(int)(a);i<(int)(b);i++)
#define rep(i,n) repl(i,0,n)
#define dbg(x) cout<<#x<<"="<<x<<endl
#define all(x) (x).begin(),(x).end()

#define INF INT_MAX/3

int gr(int v);

int N,M;
vector<int> g[1010];
int ind[1010];
int grd[1010];
vector<int> gs;

void dfs(int v,int sum){
  int res=sum;
  for(int nv : g[v]){
    res^=gr(nv);
  }
  gs.push_back(res);
  for(int nv : g[v]){
    res^=gr(nv);
    dfs(nv,res);
    res^=gr(nv);
  }
}

int gr(int v){
  if(grd[v]!=-1)return grd[v];
  for(int nv : g[v]){
    gr(nv);
  }

  gs.clear();
  dfs(v,0);
  sort(all(gs));
  gs.erase(unique(all(gs)),gs.end());
  int res=0;
  rep(j,gs.size()){
    if(res==gs[j])res++;
    else break;
  }
  return grd[v]=res;
}

int main(){
  cin>>N>>M;
  rep(i,M){
    int p;
    cin>>p;
    p--;
    g[p].push_back(i);
    ind[i]++;
  }
  //rep(i,N)dbg(ind[i]);

  memset(grd,-1,sizeof(grd));
  int res=0;
  rep(i,N){
    if(ind[i]==0){
      res^=gr(i);
      //dbg(gr(i));
    }
  }
  if(res==0)cout<<"Bob"<<endl;
  else cout<<"Alice"<<endl;
}
