#include <bits/stdc++.h>
long long int n;
int t;
int main() {
  scanf("%d", &t);
  for (int i = 0; i < t; i++) {
    scanf("%lld", &n);
    if (n == 1 || n == 2)
      printf("0\n");
    else {
      if (n % 2 == 0)
        printf("%lld\n", n / 2 - 1);
      else
        printf("%lld\n", n / 2);
    }
  }
  return 0;
}
