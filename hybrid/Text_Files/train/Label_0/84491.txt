from itertools import groupby
from collections import Counter
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	a = [x[0] for x in groupby(a)]
	c = Counter(a)
	d = {}
	for e in a:
		if e in d:
			continue
		num_seg = c[e]
		sub = 0
		if a[0] == e:
			sub += 1
		if a[-1] == e:
			sub += 1
		d[e] = num_seg + 1 - sub
	print(min(d.values()))
