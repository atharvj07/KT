x = int(input())
ls = [int(i) for i in input().split()]
v = sorted(ls)
arr = []
mx1 = v[len(v) - 1]
mx2 = v[len(v) - 2]
sm = sum(ls)
for i in range(len(ls)):
	if ls[i] == mx1:
		if sm - ls[i] - mx2 == mx2:
			arr.append(i + 1)
	elif sm - ls[i] - mx1 == mx1:
		arr.append(i + 1)
print(len(arr))
for i in range(len(arr)):
	print(arr[i], end=' ')
