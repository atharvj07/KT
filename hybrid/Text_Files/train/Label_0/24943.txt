using namespace std;

const long long int MOD = 1000000007;

long long int N, M, K, H, W, L, R, V, E;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);

	int dir[] = { 0,1,0,-1,0 };
	cin >> W >> H;
	vector<string>ans;
	while (H) {
		vector<string>a(H);
		vector<string>b(H);
		for (int i = 0; i < H; i++) {
			cin >> a[i];
			cin >> b[i];
		}
		vector<vector<bool>>flag(H*W, vector<bool>(H*W));
		int asx, asy, agx, agy, bsx, bsy, bgx, bgy;
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				if (a[i][j] == 'L') {
					asy = i;
					asx = j;
				}
				if (a[i][j] == '%') {
					agy = i;
					agx = j;
				}
			}
		}
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				if (b[i][j] == 'R') {
					bsy = i;
					bsx = j;
				}
				if (b[i][j] == '%') {
					bgy = i;
					bgx = j;
				}
			}
		}
		queue<pair<int, int>>Q;
		flag[asy*W + asx][bsy*W + bsx] = true;
		Q.push({ asy * W + asx,bsy * W + bsx });
		while (!Q.empty()) {
			auto c = Q.front();
			Q.pop();
			int ay = c.first / W;
			int ax = c.first % W;
			int by = c.second / W;
			int bx = c.second%W;
			for (int i = 0; i < 4; i++) {
				int nay = ay + dir[i];
				int nax = ax + dir[i + 1];
				int nby = by + dir[i];
				int nbx = bx - dir[i + 1];
				nay = max(0, min(nay, (int)H - 1));
				nby = max(0, min(nby, (int)H - 1));
				nax = max(0, min(nax, (int)W - 1));
				nbx = max(0, min(nbx, (int)W - 1));
				if (a[nay][nax] == '#') {
					nay = ay;
					nax = ax;
				}
				if (b[nby][nbx] == '#') {
					nby = by;
					nbx = bx;
				}
				if (a[nay][nax] == '%'^b[nby][nbx] == '%')continue;
				if (flag[nay*W + nax][nby*W + nbx])continue;
				flag[nay*W + nax][nby*W + nbx] = true;
				Q.push({ nay*W + nax,nby*W + nbx });
			}
		}
		if (flag[agy*W + agx][bgy*W + bgx])ans.push_back("Yes\n");
		else ans.push_back("No\n");
		cin >> W >> H;
	}
	for (auto i : ans) {
		cout << i;
	}
	return 0;
}
