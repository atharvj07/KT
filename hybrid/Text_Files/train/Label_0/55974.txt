	public void solve() throws IOException {
	//	Scanner scan = new Scanner(System.in);
		int n = in.nextInt();
		String []str = new String [n];
		HashMap<String, Integer> h = new HashMap<String, Integer>();
		for(int i = 0;i<n;i++){
			str[i] = in.next();
			HashSet<String>hNow = new HashSet();
			for(int j = 0;j<str[i].length();j++){
				String s = "";
				for(int k = j;k<str[i].length();k++){
					s += str[i].charAt(k);
					if((h.containsKey(s))){
						if(!hNow.contains(s)){
							h.put(s, h.get(s)+1);
							hNow.add(s);
						}
					} else {
						h.put(s, 1);
						hNow.add(s);
					}
				}
			}
		}
		for(int i = 0;i<n;i++){
			int min = Integer.MAX_VALUE;
			String minStr = " ";
			for(int j = 0;j<str[i].length();j++){
				String s = "";
				for(int k = j;k<str[i].length();k++){
					s += str[i].charAt(k);
					if((s.length() < min)&&(h.get(s) == 1)){
						min = s.length();
						minStr = s;
					}
				}
			}
			out.println(minStr);
		}
	}

	private static int gcd(int a, int b) {
		while ((a != 0) && (b != 0)) {
			if (a < b) {
				int tmp = a;
				a = b;
				b = tmp;
			}
			a %= b;
		}
		return (a + b);
	}

	class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		FastScanner(BufferedReader bufferedReader) {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreTokens()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

	}
