#include <bits/stdc++.h>
using namespace std;
long long ans = 0, st[500000 + 5];
int a[500000 + 5], num[500000 + 5], cnt = 0, c[500000 + 5], sum = 0, n;
priority_queue<long long> q;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  sort(a + 1, a + 1 + n);
  for (int i = 1; i <= n; i++) {
    if (i == 1 || a[i] != a[i - 1]) num[++cnt] = a[i];
    c[cnt]++;
    ans += a[i];
  }
  for (int i = cnt; i; i--) {
    int v = num[i],
        len = ((sum - (q.size() << 1)) < (c[i]) ? (sum - (q.size() << 1))
                                                : (c[i])),
        t = 0;
    for (int j = 1; j <= len; j++) st[++t] = v;
    for (int j = 1; j + len <= ((sum) < (c[i]) ? (sum) : (c[i])); j += 2) {
      long long x = -q.top();
      q.pop();
      if (x < v) {
        st[++t] = v;
        if (j + len != ((sum) < (c[i]) ? (sum) : (c[i]))) st[++t] = v;
      } else {
        st[++t] = x;
        if (j + len != ((sum) < (c[i]) ? (sum) : (c[i])) && (v << 1) - x > 0)
          st[++t] = (v << 1) - x;
      }
    }
    for (int j = 1; j <= t; j++) q.push(-st[j]);
    sum += c[i];
  }
  while (!q.empty()) ans += q.top(), q.pop();
  printf("%lld\n", ans);
  return 0;
}
