#include <bits/stdc++.h>
using namespace std;
long long ans = 0;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  int n, a;
  cin >> n >> a;
  double an = 180 * 1.0 - 360 * 1.0 / n;
  double an1 = an / (n - 2);
  int cnt = 1;
  double mi = 1000;
  for (double i = an1; i <= an; i += an1) {
    double z = abs(i - a);
    if (z < mi) {
      mi = z;
      ans = cnt;
    }
    cnt++;
  }
  cout << 2 + ans << " ";
  cout << "1 "
       << "2";
  return 0;
}
