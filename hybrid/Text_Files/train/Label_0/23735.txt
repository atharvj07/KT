#include <bits/stdc++.h>
using namespace std;
long long HP[222222];
long long DMG[222222];
long long Solve(int n, int a, int b) {
  long long sum_dmg = 0;
  for (int i = 1; i <= n; i++) sum_dmg += DMG[i];
  if (b == 0) {
    return sum_dmg;
  }
  set<pair<long long, int>> top_b;
  for (int i = 1; i <= n; i++) {
    long long x = max(0ll, HP[i] - DMG[i]);
    top_b.insert({x, i});
    if (top_b.size() > b) {
      top_b.erase(top_b.begin());
    }
  }
  long long sum_base_delta = 0;
  for (auto x : top_b) {
    sum_base_delta += x.first;
  }
  long long max_delta = sum_base_delta;
  for (int i = 1; i <= n; i++) {
    if ((1ll << a) * HP[i] <= DMG[i]) continue;
    pair<long long, int> prev = {max(0ll, HP[i] - DMG[i]), i};
    if (top_b.find(prev) != top_b.end()) {
      max_delta = max(
          max_delta, sum_base_delta + (1ll << a) * HP[i] - DMG[i] - prev.first);
    } else {
      max_delta = max(max_delta, sum_base_delta + (1ll << a) * HP[i] - DMG[i] -
                                     top_b.begin()->first);
    }
  }
  return max_delta + sum_dmg;
}
char buf[58000000];
int readint() {
  int m;
  static char *bp = buf;
  if (*bp < '0' || *bp > '9') *bp++;
  for (m = *bp++ - 48; *bp >= 48; bp++) m = m * 10 + *bp - 48;
  bp++;
  return m;
}
int main() {
  fread(buf, 1, 51000100, stdin);
  int n, a, b;
  n = readint();
  a = readint();
  b = readint();
  for (int i = 1; i <= n; i++) {
    HP[i] = readint();
    DMG[i] = readint();
  }
  cout << Solve(n, a, b) << endl;
}
