#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 7;
int m, n, a, b, num, yes[N], no[N], tmp[N], maybe[N];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> tmp[i];
    if (tmp[i] > 0)
      yes[tmp[i]]++;
    else
      no[-tmp[i]]++;
    a += tmp[i] > 0;
    b += tmp[i] < 0;
  }
  for (int i = 1; i <= n; i++) maybe[i] = (a - yes[i] + no[i] == n - m);
  int time = 0;
  for (int i = 1; i <= n; i++)
    if (maybe[i]) time++;
  for (int i = 1; i <= n; i++) {
    if (tmp[i] > 0) {
      if (maybe[tmp[i]]) {
        if (time > 1)
          cout << "Not defined" << endl;
        else
          cout << "Truth" << endl;
      } else
        cout << "Lie" << endl;
    } else {
      if (maybe[-tmp[i]]) {
        if (time > 1)
          cout << "Not defined" << endl;
        else
          cout << "Lie" << endl;
      } else
        cout << "Truth" << endl;
    }
  }
  return 0;
}
