#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
int n, fac = 1;
int pw(int a, int b) {
  int res = 1;
  while (b) {
    if (b & 1) res = (1LL * res * a) % MOD;
    a = (1LL * a * a) % MOD, b >>= 1;
  }
  return res;
}
int main(void) {
  cin >> n;
  for (int i = 1; i <= n; ++i) fac = (1LL * fac * i) % MOD;
  int inv = (1LL * pw(fac, MOD - 2)) % MOD;
  for (int i = n + 1; i <= n + n; ++i) fac = (1LL * fac * i) % MOD;
  inv = (1LL * inv * inv) % MOD;
  fac = (1LL * fac * inv) % MOD;
  fac = (fac - n + MOD) % MOD;
  cout << fac;
  return 0;
}
