def hard(a, n):
	di = {}
	if len(a) <= n:
		return n
	mak = -1
	for i in a:
		if i in di:
			di[i] += 1
		else:
			di[i] = 1
		mak = max(mak, di[i])
	if n == 1 and mak == len(a):
		l = len(a)
		return l - 1
	return min(mak + n, len(a))
n = int(input())
a = hard(input(), n)
b = hard(input(), n)
c = hard(input(), n)
if a > b and a > c:
	print('Kuro')
elif b > a and b > c:
	print('Shiro')
elif c > a and c > b:
	print('Katie')
else:
	print('Draw')
