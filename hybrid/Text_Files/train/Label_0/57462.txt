// practice with rainboy
import java.io.*;
import java.util.*;

public class CF875F extends PrintWriter {
	CF875F() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF875F o = new CF875F(); o.main(); o.flush();
	}

	int[] dsu; boolean[] full;
	int find(int i) {
		return dsu[i] < 0 ? i : (dsu[i] = find(dsu[i]));
	}
	boolean join(int i, int j) {
		i = find(i);
		j = find(j);
		if (i == j) {
			if (full[i])
				return false;
			full[i] = true;
			return true;
		}
		if (dsu[i] > dsu[j]) {
			int tmp = i; i = j; j = tmp;
		}
		if (dsu[i] == dsu[j])
			dsu[i]--;
		dsu[j] = i;
		if (full[i] && full[j])
			return false;
		full[i] = full[i] || full[j];
		return true;
	}
	static class E {
		int i, j, w;
		E(int i, int j, int w) {
			this.i = i; this.j = j; this.w = w;
		}
	}
	void main() {
		int n = sc.nextInt();
		int m = sc.nextInt();
		E[] ee = new E[m];
		for (int h = 0; h < m; h++) {
			int i = sc.nextInt() - 1;
			int j = sc.nextInt() - 1;
			int w = sc.nextInt();
			ee[h] = new E(i, j, w);
		}
		Arrays.sort(ee, (e, f) -> f.w - e.w);
		dsu = new int[n]; Arrays.fill(dsu, -1);
		full = new boolean[n];
		int ans = 0;
		for (int h = 0; h < m; h++) {
			E e = ee[h];
			if (join(e.i, e.j))
				ans += e.w;
		}
		println(ans);
	}
}
