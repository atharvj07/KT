#include <bits/stdc++.h>
using namespace std;
int main() {
  long long l, r, x, y, k;
  while (~scanf("%lld%lld%lld%lld%lld", &l, &r, &x, &y, &k)) {
    int flag = 0;
    for (int i = x; i <= y; i++) {
      if (i * k >= l && i * k <= r) {
        flag = 1;
        break;
      }
    }
    if (flag)
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
