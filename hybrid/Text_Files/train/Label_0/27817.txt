#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, b, c, t;
  cin >> n >> a >> b >> c >> t;
  int x[n];
  for (int i = 0; i < n; i++) {
    cin >> x[i];
  }
  sort(x, x + n);
  if (b >= c) {
    cout << n * a << endl;
  } else {
    long long int sum = 0, sum1 = 0;
    for (int i = 0; i < n; i++) {
      sum += (t - x[i]) * c;
    }
    for (int i = 0; i < n; i++) {
      if (x[i] != t)
        sum1 += (a - (t - x[i]) * b);
      else
        sum1 += a;
    }
    cout << sum + sum1 << endl;
  }
  return 0;
}
