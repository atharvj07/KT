G_EVEN = {0: 0, 1: 1, 2: 2}
G_ODD = {0: 0, 1: 1, 2: 0, 3: 1}

def grundy(k, ai):
	if k % 2:
		if ai <= 3:
			return G_ODD[ai]
		elif ai % 2:
			return 0
		else:
			p = 0
			j = ai
			while not j & 1:
				p += 1
				j >>= 1
			if j == 3:
				return 2 if p % 2 else 1
			else:
				return 1 if p % 2 else 2
			return 1 + p % 2
	elif ai <= 2:
		return G_EVEN[ai]
	else:
		return ~ai & 1

def nim_sum(ns):
	s = 0
	for ni in ns:
		s ^= ni
	return s

def winner(k, a):
	return bool(nim_sum((grundy(k, ai) for ai in a)))
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
print('Kevin' if winner(k, a) else 'Nicky')
