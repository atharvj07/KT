El_n = 0
El_k = 0
entrada = input().split(' ')
El_n = int(entrada[0])
El_k = int(entrada[1])
acumulado = 0
arreglo = []
for index in range(El_n):
	acumulado += (index + 1) * 5
	arreglo.append(acumulado)
max = El_n
min = 0
while max - min > 1:
	medio = int((max + min) / 2)
	if medio == min:
		break
	if arreglo[medio - 1] <= 240 - El_k:
		min = medio
	else:
		max = medio
if arreglo[min] <= 240 - El_k:
	print(min + 1)
else:
	print(min)
