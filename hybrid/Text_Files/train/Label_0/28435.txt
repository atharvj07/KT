#include <bits/stdc++.h>
using namespace std;
const int N = 3e3 + 10;
const long long INF = 1e9 * N;
long long n, dp[N], cost[N][N];
struct node {
  long long x, cost;
} a[N];
bool cmp(node a, node b) { return a.x < b.x; }
int main() {
  scanf("%lld", &n);
  for (int i = 1; i <= n; i++) scanf("%lld%lld", &a[i].x, &a[i].cost);
  sort(a + 1, a + 1 + n, cmp);
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      cost[i][j] = cost[i][j - 1] + a[j].x - a[i].x;
    }
  }
  dp[1] = a[1].cost;
  for (int i = 2; i <= n; i++) {
    dp[i] = INF;
    for (int j = i - 1; j >= 1; j--) {
      dp[i] = min(dp[i], dp[j] + cost[j][i - 1]);
    }
    dp[i] += a[i].cost;
  }
  long long ans = INF;
  for (int i = 1; i <= n; i++) {
    ans = min(ans, dp[i] + cost[i][n]);
  }
  printf("%lld\n", ans);
  return 0;
}
