#include <bits/stdc++.h>
using namespace std;
const int OnlineJudge = 0;
const int maxn = 500000 + 1912;
const int lim = 1e9;
const int block = 760;
int n, Q;
long long a[maxn];
long long added[block + 1];
void ReadData() {
  cin >> n >> Q;
  for (int _a = 1, _b = n, i = _a; i <= _b; i++) cin >> a[i];
}
map<long long, int> cnt[block + 1];
void add(const int pos, const int delta) {
  int cur = pos / block;
  map<long long, int>::iterator it = cnt[cur].find(a[pos]);
  it->second--;
  if (it->second == 0) cnt[cur].erase(it);
  a[pos] += delta;
  if (a[pos] > lim) return;
  cnt[cur][a[pos]]++;
}
void add(int low, int high, int val) {
  while (low <= high && low % block != 0) {
    add(low, val);
    low++;
  }
  while (high >= low && high % block != block - 1) {
    add(high, val);
    high--;
  }
  if (low > high) return;
  for (int _a = low / block, _b = high / block, cur = _a; cur <= _b; cur++) {
    added[cur] += val;
  }
}
int findMin(const long long val) {
  for (int _a = 0, _b = n / block, cur = _a; cur <= _b; cur++) {
    long long nexVal = val - added[cur];
    if (cnt[cur].find(nexVal) != cnt[cur].end() && cnt[cur][nexVal]) {
      for (int _a = max(1, cur * block), _b = n, pos = _a; pos <= _b; pos++)
        if (a[pos] == nexVal) {
          return pos;
        }
    }
  }
  return -1;
}
int findMax(const long long val) {
  for (int _a = n / block, _b = 0, cur = _a; cur >= _b; cur--) {
    long long nexVal = val - added[cur];
    if (cnt[cur].find(nexVal) != cnt[cur].end() && cnt[cur][nexVal]) {
      for (int _a = min(n, cur * block + block - 1), _b = 1, pos = _a;
           pos >= _b; pos--)
        if (a[pos] == nexVal) {
          return pos;
        }
    }
  }
  return -1;
}
int query(const int val) {
  int mn = findMin(val);
  if (mn == -1) return -1;
  int mx = findMax(val);
  return mx - mn;
}
void Init() {
  for (int _a = 1, _b = n, i = _a; i <= _b; i++) {
    int cur = i / block;
    cnt[cur][(long long)a[i]]++;
  }
}
void Process() {
  Init();
  while (Q--) {
    int type;
    cin >> type;
    if (type == 1) {
      int l, r, val;
      cin >> l >> r >> val;
      add(l, r, val);
    } else {
      int val;
      cin >> val;
      cout << query(val) << "\n";
    }
  }
}
int main() {
  std ::ios_base::sync_with_stdio(false);
  cin.tie(0);
  ReadData();
  Process();
  return 0;
}
