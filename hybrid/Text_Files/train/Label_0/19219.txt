#include <bits/stdc++.h>
using namespace std;
const int MAX_ = 99999;
int main() {
  ifstream in("input.txt");
  ofstream out("output.txt");
  int n;
  int ans = 0;
  in >> n;
  for (int i = 0; i < n; ++i) {
    int m;
    in >> m;
    vector<vector<int> > graph(m, vector<int>(m, MAX_));
    for (int i = 0; i < m; ++i) graph[i][i] = 0;
    for (int j = 0; j < m - 1; ++j) {
      int a, b;
      in >> a >> b;
      graph[a - 1][b - 1] = 1;
      graph[b - 1][a - 1] = 1;
    }
    for (int k = 0; k < m; ++k)
      for (int pp = 0; pp < m; ++pp)
        for (int qq = 0; qq < m; ++qq) {
          graph[pp][qq] = min(graph[pp][qq], graph[pp][k] + graph[k][qq]);
        }
    int max_ = 0;
    for (int pp = 0; pp < m; ++pp) {
      for (int qq = 0; qq < m; ++qq) {
        if (graph[pp][qq] != MAX_) max_ = max(max_, graph[pp][qq]);
      }
    }
    cout << max_ << endl;
    ans += max_;
  }
  cout << ans << endl;
  out << ans << endl;
  return 0;
}
