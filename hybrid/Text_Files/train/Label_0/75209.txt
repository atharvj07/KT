def getcount(n, k):
	res = n >> k + 1 << k
	if n >> k & 1:
		res += n & (1 << k) - 1
	return res
for _ in range(int(input())):
	(l, r) = list(map(int, input().split()))
	n = r - l + 1
	if l == 1:
		m = 0
		for i in range(0, 32):
			m = max(m, getcount(r + 1, i))
		print(n - m)
		continue
	bits = [0] * 32
	for i in range(0, 32):
		bits[i - 1] = getcount(r + 1, i) - getcount(l, i)
	print(n - max(bits))
