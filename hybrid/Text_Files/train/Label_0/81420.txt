(N,) = map(int, input().split())
X = list(map(int, input().split()))
if N > 1:
	print(1, N)
	Y = [0] * N
	for i in range(N):
		Y[i] = -(X[i] % (N - 1)) * N
		X[i] = -(X[i] + Y[i])
	print(*Y)
	print(1, 1)
	print(X[0])
	print(2, N)
	print(*X[1:])
else:
	print(1, 1)
	print(-X[0])
	print(1, 1)
	print(0)
	print(1, 1)
	print(0)
