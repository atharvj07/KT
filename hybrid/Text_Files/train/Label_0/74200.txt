from collections import defaultdict
from math import ceil

def solve():
	n = int(input())
	graph = defaultdict(list)
	largest_leaf = defaultdict(int)
	leaf_count = defaultdict(int)
	subtree_sum = defaultdict(int)
	ans = 0
	for (i, p) in enumerate(input().split(), 2):
		graph[int(p)] += [i]
	a = {i: int(x) for (i, x) in enumerate(input().split(), 1)}
	for v in reversed(range(1, n + 1)):
		subtree_sum[v] = a[v]
		if len(graph[v]) == 0:
			largest_leaf[v] = subtree_sum[v]
			leaf_count[v] = 1
		else:
			largest_child_leaf = 0
			for child in graph[v]:
				largest_child_leaf = max(largest_child_leaf, largest_leaf[child])
				subtree_sum[v] += subtree_sum[child]
				leaf_count[v] += leaf_count[child]
			result_sum = max(largest_child_leaf * leaf_count[v], subtree_sum[v])
			largest_leaf[v] = ceil(result_sum / leaf_count[v])
		ans = max(ans, largest_leaf[v])
	print(ans)
solve()
