#include <bits/stdc++.h>
using namespace std;
struct node {
  int a, b, c, d, e;
} p[100005];
int n, q, i, j, k, l, x, y, z, w, g, c[5005][10005], ans[100005];
void add(int x, int y, int w) {
  int z;
  while (x <= n) {
    z = y;
    while (z <= 2 * n) {
      c[x][z] += w;
      z += z & -z;
    }
    x += x & -x;
  }
}
void work(int lx, int rx, int ly, int ry, int c) {
  if (lx > rx || ly > ry) return;
  add(lx, ly, c);
  add(lx, ry + 1, -c);
  add(rx + 1, ly, -c);
  add(rx + 1, ry + 1, c);
}
int Ask(int x, int y) {
  int z, rtn = 0;
  while (x) {
    z = y;
    while (z >= 1) {
      rtn += c[x][z];
      z -= z & -z;
    }
    x -= x & -x;
  }
  return rtn;
}
int main() {
  scanf("%d%d", &n, &q);
  for (i = 1; i <= q; ++i) {
    scanf("%d", &p[i].a);
    if (p[i].a == 1)
      scanf("%d%d%d%d", &p[i].b, &p[i].c, &p[i].d, &p[i].e);
    else
      scanf("%d%d", &p[i].b, &p[i].c);
  }
  for (z = 1; z <= 4; ++z)
    for (g = 0; g <= 1; ++g) {
      for (w = 1; w <= q; ++w) {
        i = p[w].a;
        if (i == 1) {
          i = p[w].b, j = p[w].c, k = p[w].d, l = p[w].e;
          if (i != z) continue;
          if (i == 1) {
            if (g) work(j, n, j + k, j + k + l, 1);
            if (!g) work(1, k - 1, j + k, j + k + l, 1);
          } else if (i == 2) {
            k = n - k + 1;
            if (g) work(j, n, j + k, j + k + l, 1);
            if (!g) work(1, k - 1, j + k, j + k + l, 1);
          } else if (i == 3) {
            j = n - j + 1;
            if (g) work(j, n, j + k, j + k + l, 1);
            if (!g) work(1, k - 1, j + k, j + k + l, 1);
          } else {
            j = n - j + 1, k = n - k + 1;
            if (g) work(j, n, j + k, j + k + l, 1);
            if (!g) work(1, k - 1, j + k, j + k + l, 1);
          }
        } else {
          i = p[w].b, j = p[w].c;
          if (z == 1) {
            x = i, y = j;
            if (g)
              k = Ask(x, x + y);
            else
              k = -Ask(y, x + y);
          } else if (z == 2) {
            x = i, y = n - j + 1;
            if (g)
              k = Ask(x, x + y);
            else
              k = -Ask(y, x + y);
          } else if (z == 3) {
            x = n - i + 1, y = j;
            if (g)
              k = Ask(x, x + y);
            else
              k = -Ask(y, x + y);
          } else {
            x = n - i + 1, y = n - j + 1;
            if (g)
              k = Ask(x, x + y);
            else
              k = -Ask(y, x + y);
          }
          ans[w] += k;
        }
      }
      for (w = 1; w <= q; ++w) {
        i = p[w].a;
        if (i == 1) {
          i = p[w].b, j = p[w].c, k = p[w].d, l = p[w].e;
          if (i != z) continue;
          if (i == 1) {
            if (g) work(j, n, j + k, j + k + l, -1);
            if (!g) work(1, k - 1, j + k, j + k + l, -1);
          } else if (i == 2) {
            k = n - k + 1;
            if (g) work(j, n, j + k, j + k + l, -1);
            if (!g) work(1, k - 1, j + k, j + k + l, -1);
          } else if (i == 3) {
            j = n - j + 1;
            if (g) work(j, n, j + k, j + k + l, -1);
            if (!g) work(1, k - 1, j + k, j + k + l, -1);
          } else {
            j = n - j + 1, k = n - k + 1;
            if (g) work(j, n, j + k, j + k + l, -1);
            if (!g) work(1, k - 1, j + k, j + k + l, -1);
          }
        }
      }
    }
  for (i = 1; i <= q; ++i)
    if (p[i].a == 2) printf("%d\n", ans[i]);
  return 0;
}
