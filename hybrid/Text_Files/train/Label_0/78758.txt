import collections
n = int(input())
l = input()
r = input()
lcount = collections.defaultdict(int)
rcount = collections.defaultdict(int)
l_indexes = collections.defaultdict(list)
r_indexes = collections.defaultdict(list)
for i in range(n):
	lcount[l[i]] += 1
	rcount[r[i]] += 1
	l_indexes[l[i]].append(i + 1)
	r_indexes[r[i]].append(i + 1)
matches = []
for i in lcount.keys():
	if lcount[i] > 0 and rcount[i] > 0 and (i != '?'):
		while lcount[i] > 0 and rcount[i] > 0:
			lcount[i] -= 1
			rcount[i] -= 1
			matches.append((l_indexes[i].pop(), r_indexes[i].pop()))
if lcount['?'] > 0:
	for i in rcount.keys():
		while lcount['?'] > 0 and rcount[i] > 0 and (i != '?'):
			lcount['?'] -= 1
			rcount[i] -= 1
			matches.append((l_indexes['?'].pop(), r_indexes[i].pop()))
if rcount['?'] > 0:
	for i in lcount.keys():
		while rcount['?'] > 0 and lcount[i] > 0 and (i != '?'):
			rcount['?'] -= 1
			lcount[i] -= 1
			matches.append((l_indexes[i].pop(), r_indexes['?'].pop()))
if lcount['?'] > 0:
	for i in rcount.keys():
		while lcount['?'] > 0 and rcount[i] > 0:
			lcount['?'] -= 1
			rcount[i] -= 1
			matches.append((l_indexes['?'].pop(), r_indexes[i].pop()))
if rcount['?'] > 0:
	for i in lcount.keys():
		while rcount['?'] > 0 and lcount[i] > 0:
			rcount['?'] -= 1
			lcount[i] -= 1
			matches.append((l_indexes[i].pop(), r_indexes['?'].pop()))
print(len(matches))
for (lp, rp) in matches:
	print(str(lp) + ' ' + str(rp))
