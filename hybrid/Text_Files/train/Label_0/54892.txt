import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class ProblemC {
	public static void main(String[] args) throws IOException {
		BufferedReader s = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(System.out);

		int n = Integer.valueOf(s.readLine());
		Map<Integer, List<Integer>> graph = new HashMap<Integer, List<Integer>>();
		
		for (int i = 0 ; i < n ; i++) {
			String[] ln = s.readLine().split(" ");
			int a = Integer.valueOf(ln[0]);
			int b = Integer.valueOf(ln[1]);
			if (!graph.containsKey(a)) {
				graph.put(a, new ArrayList<Integer>());
			}
			if (!graph.containsKey(b)) {
				graph.put(b, new ArrayList<Integer>());
			}
			graph.get(a).add(b);
			graph.get(b).add(a);
		}
		
		int from = -1;
		for (int k : graph.keySet()) {
			if (graph.get(k).size() == 1) {
				from = k;
				break;
			}
		}
		
		int prev = -1;
		int now = from;
		StringBuffer b = new StringBuffer();
		for (int i = 0 ; i <= n ; i++) {
			b.append(" ").append(now);
			if (i == n) {
				break;
			}
			int next = -1;
			for (int l : graph.get(now)) {
				if (l != prev) {
					next = l;
					break;
				}
			}
			prev = now;
			now = next;
		}
		out.println(b.substring(1));
		out.flush();
	}

	public static void debug(Object... os){
		System.err.println(Arrays.deepToString(os));
	}
}
