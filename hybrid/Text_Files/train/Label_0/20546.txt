#include <iostream>
#include <string>
using namespace std;

string s;

int gcd(int a, int b) {
	if (b == 0) return a;
	return gcd(b, a % b);
}

int main() {
	while (cin >> s) {
		if (s == "#") break;
		
		string t;
		for (int i = s.length() - 1; i >= 0; i--) {
			if (s[i] == 'n' || s[i] == 'w') {
				t += s[i];
			}
		}
		
		int bo = (1 << 21);
		int si = 90 * bo;
		int sumSi = (t[0] == 'n') ? 0 : si;
		si /= 2;
		
		for (int i = 1; i < t.length(); i++) {
			if (t[i] == 'n') sumSi -= si;
			else sumSi += si;
			si /= 2;
		}
		
		int g = gcd(sumSi, bo);
		sumSi /= g;
		bo /= g;
		if (bo == 1) { cout << sumSi << endl; }
		else { cout << sumSi << "/" << bo << endl; }
	}
	return 0;
}
