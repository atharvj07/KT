import sys
(n, m, d) = map(int, sys.stdin.readline().split())
p = [0] + list(map(int, sys.stdin.readline().split()))
w = [0] * (n + 2)
idx = n
judge = False
if n < sum(p):
	print('NO')
else:
	for i in range(len(p) - 1, 0, -1):
		width = p[i]
		for j in range(idx, 0, -1):
			if width == 0:
				break
			w[j] = i
			idx -= 1
			width -= 1
	end_idx = idx + 1
	st_idx = 0
	for i in range(1, len(p)):
		count = p[i]
		if end_idx - st_idx <= d:
			print('YES')
			res = w[1:-1]
			res = list(map(str, res))
			res = ' '.join(res)
			print(res)
			judge = True
			break
		else:
			dist = abs(d - (end_idx - st_idx))
			for j in range(count):
				temp = w[end_idx]
				w[end_idx - dist] = temp
				w[end_idx] = 0
				st_idx = end_idx - dist
				end_idx += 1
	for k in range(n, 0, -1):
		if w[k] != 0:
			break
	if n - k >= d:
		print('NO')
	elif end_idx - st_idx <= d and judge == False:
		print('YES')
		res = w[1:-1]
		res = list(map(str, res))
		res = ' '.join(res)
		print(res)
