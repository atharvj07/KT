#include <bits/stdc++.h>
using namespace std;
int32_t const N = 1001;
int32_t X[N];
int32_t Y[N];
int32_t gcd(int32_t a, int32_t b) { return a == 0 ? b : gcd(b % a, a); }
struct point {
  int32_t x, y;
};
struct line {
  int32_t a, b;
  int64_t c;
  bool operator<(const line &l) const {
    if (a != l.a) {
      return a < l.a;
    } else {
      if (b != l.b) {
        return b < l.b;
      } else {
        return c < l.c;
      }
    }
  }
  line(point p1, point p2) {
    a = p1.y - p2.y;
    b = p1.x - p2.x;
    int32_t d = gcd(a, b);
    a /= d;
    b /= d;
    if (a < 0 || (a == 0 && b < 0)) {
      a = -a;
      b = -b;
    }
    c = 1LL * a * p1.x - 1LL * b * p1.y;
  }
};
void pr(line l) { cout << l.a << " " << l.b << " " << l.c << "\n"; }
int main() {
  int32_t n;
  vector<line> lines;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> X[i];
    cin >> Y[i];
  }
  int32_t total = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      point p1 = {X[i], Y[i]};
      point p2 = {X[j], Y[j]};
      lines.push_back(line(p1, p2));
      total++;
    }
  }
  sort(lines.begin(), lines.end());
  set<line> s(lines.begin(), lines.end());
  lines.assign(s.begin(), s.end());
  total = lines.size();
  int64_t count = 0;
  int32_t parallelLines = 1;
  int32_t curA = lines[0].a;
  int32_t curB = lines[0].b;
  set<int64_t> _set;
  _set.insert(lines[0].c);
  for (int i = 1; i < total; i++) {
    if (lines[i].a != curA || lines[i].b != curB) {
      count += parallelLines * (total - i);
      parallelLines = 1;
      _set = {lines[i].c};
      curA = lines[i].a;
      curB = lines[i].b;
    } else {
      if (_set.count(lines[i].c) == 0) {
        _set.insert(lines[i].c);
        parallelLines++;
      }
    }
  }
  cout << count << "\n";
}
