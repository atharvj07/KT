from sys import stdin, stdout
(N, M) = map(int, stdin.readline().strip().split())
req = [[[-1, -1] for __ in range(M)] for __ in range(N)]
input1 = stdin.readline().strip()
input2 = stdin.readline().strip()
for i in range(N):
	curx = 1 if input1[i] == '>' else -1
	for j in range(M):
		cury = -1 if input2[j] == '^' else 1
		req[i][j] = [cury, curx]
condn = True
for i in range(N):
	for j in range(M):
		notVisited = [[True for __ in range(M)] for __ in range(N)]
		count = N * M - 1
		stack = [(i, j)]
		notVisited[i][j] = False
		while stack:
			(curX, curY) = stack.pop()
			(modX, modY) = (curX + req[curX][curY][0], curY + req[curX][curY][1])
			if modX >= 0 and modX < N and notVisited[modX][curY]:
				stack.append((modX, curY))
				notVisited[modX][curY] = False
				count -= 1
			if modY >= 0 and modY < M and notVisited[curX][modY]:
				stack.append((curX, modY))
				notVisited[curX][modY] = False
				count -= 1
		if count:
			condn = False
			break
	if not condn:
		break
if condn:
	stdout.write('YES')
else:
	stdout.write('NO')
