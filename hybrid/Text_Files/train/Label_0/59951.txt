import java.util.*;

public class B{
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		
		int t = sc.nextInt();
		while(t-- != 0) {
			int d = sc.nextInt();
			int x = 1+d;
			int ans1 = 0;
			int ans2 = 0;
			while(true) {
				if(isPrime(x)) {
					ans1 = x;
					break;
				}
				x++;
			}
			x += d;
			while(true) {
				if(isPrime(x)) {
					ans2 = x;
					break;
				}
				x++;
			}
			System.out.println(ans1 * ans2);
	
		}
		
		sc.close();
	}
	
	public static boolean isPrime(int n) {
		for(int i=2; i*i<=n; i++) {
			if(n%i==0) return false;
		}
		return true;
	}
}
