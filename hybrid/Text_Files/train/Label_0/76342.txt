def dfs(index, arr, parent, visited):
	min_ele = arr[index]
	visited[index] = True
	q = parent[index]
	i = 0
	while i < len(q):
		if not visited[q[i]]:
			visited[q[i]] = True
			min_ele = min(min_ele, arr[q[i]])
			q.extend(parent[q[i]])
		i += 1
	return min_ele

def main():
	(N, K) = map(int, input().split())
	parent = [[] for _ in range(N + 4)]
	visited = [False for _ in range(N + 4)]
	arr = [-1]
	arr.extend(list(map(int, input().split())))
	for _ in range(K):
		(a, b) = map(int, input().split())
		parent[a].append(b)
		parent[b].append(a)
	add = 0
	for i in range(1, len(arr)):
		if not visited[i]:
			add += dfs(i, arr, parent, visited)
	print(add)
main()
