class Graph:

	def __init__(self, size=0):
		self.g = [[] for i in range(size)]
		self.size = size
		self.path = []

	def addEdge(self, u, v):
		try:
			self.g[u].append(v)
			self.g[v].append(u)
		except IndexError:
			pass

	def DFSAux(self, v, k, start, root):
		d = 0
		visit = [root]
		while len(visit) > 0:
			r = visit.pop()
			if v[r] == -1:
				v[r] = d
				d += 1
				self.path.append(r)
				for e in self.g[r]:
					if v[e] != -1:
						if v[r] - v[e] + 1 >= k:
							self.path = self.path[v[e]:v[r] + 1]
							return True
					else:
						visit.append(e)
		return False

	def findCycle(self, k):
		v = [-1 for i in range(self.size)]
		self.DFSAux(v, k, 0, 0)
values = list(map(int, input().split()))
g = Graph(values[0])
m = values[1]
while m > 0:
	m -= 1
	edges = list(map(int, input().split()))
	g.addEdge(edges[0] - 1, edges[1] - 1)
g.findCycle(values[2] + 1)
print(len(g.path))
for i in g.path:
	print(i + 1, end=' ')
print()
