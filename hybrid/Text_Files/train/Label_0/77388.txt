count = int(input())
result = []
for i in range(count):
	qnt = int(input())
	array = list(map(int, input().split()))
	cnt = 0
	for j in range(qnt):
		if array[j] < array[(j + 1) % qnt]:
			cnt += 1
	if cnt + 1 == qnt:
		result.append('YES')
	else:
		cnt = 0
		for j in range(qnt):
			if array[j] > array[(j + 1) % qnt]:
				cnt += 1
		if cnt + 1 == qnt:
			result.append('YES')
		else:
			result.append('NO')
print(*result, sep='\n')
