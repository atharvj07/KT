n = int(input())
visited = [False] * (n + 1)
l = [[] for i in range(n + 1)]
for i in range(n - 1):
	(a, b) = map(int, input().split())
	l[a].append(b)
	l[b].append(a)
b = list(map(int, input().split()))
if b[0] != 1:
	print('No')
	exit()
c = 1
aux = len(l[1])
for j in range(n):
	for i in l[b[j]]:
		visited[i] = True
	for i in range(c, c + aux):
		if not visited[b[i]]:
			print('No')
			exit()
	c += aux
	aux = len(l[b[(j + 1) % n]]) - 1
print('Yes')
