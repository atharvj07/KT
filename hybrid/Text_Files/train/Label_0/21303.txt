#include <bits/stdc++.h>
using namespace std;
char s[1009][512];
char p[1009][512];
int main() {
  int i, j, n, m;
  cin >> n >> m;
  cin.get();
  for (i = 1; i <= n; i++) cin >> s[i];
  for (j = 1; j <= m; j++) cin >> p[j];
  int ega = 0;
  for (i = 1; i <= n; i++)
    for (j = 1; j <= m; j++)
      if (strcmp(s[i], p[j]) == 0) {
        ega++;
        continue;
      }
  int pri, sec;
  if (ega % 2) {
    pri = ega / 2 + 1;
    sec = ega / 2;
  } else {
    pri = sec = ega / 2;
  }
  n -= ega;
  m -= ega;
  pri += n;
  sec += m;
  if (pri > sec)
    cout << "YES";
  else
    cout << "NO";
}
