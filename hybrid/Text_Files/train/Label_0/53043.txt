import java.io.*;
import java.util.*;

public class CF31D extends PrintWriter {
	CF31D() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF31D o = new CF31D(); o.main(); o.flush();
	}

	int[] ds;
	int find(int i) {
		return ds[i] < 0 ? i : (ds[i] = find(ds[i]));
	}
	void join(int i, int j) {
		i = find(i);
		j = find(j);
		if (i == j)
			return;
		if (ds[i] > ds[j]) {
			int tmp = i; i = j; j = tmp;
		}
		ds[i] += ds[j];
		ds[j] = i;
	}
	void main() {
		int n = sc.nextInt();
		int m = sc.nextInt();
		int k = sc.nextInt();
		boolean[][] hh = new boolean[n + 1][m];
		boolean[][] vv = new boolean[n][m + 1];
		for (int h = 0; h < k; h++) {
			int i1 = sc.nextInt();
			int j1 = sc.nextInt();
			int i2 = sc.nextInt();
			int j2 = sc.nextInt();
			if (i1 == i2)
				for (int i = i1, j = j1; j < j2; j++)
					hh[i][j] = true;
			else
				for (int j = j1, i = i1; i < i2; i++)
					vv[i][j] = true;
		}
		ds = new int[n * m]; Arrays.fill(ds, -1);
		for (int i = 1; i < n; i++)
			for (int j = 0; j < m; j++)
				if (!hh[i][j])
					join((i - 1) * m + j, i * m + j);
		for (int j = 1; j < m; j++)
			for (int i = 0; i < n; i++)
				if (!vv[i][j])
					join(i * m + j - 1, i * m + j);
		int[] aa = new int[k + 1];
		k = 0;
		for (int h = 0; h < n * m; h++)
			if (ds[h] < 0)
				aa[k++] = -ds[h];
		Arrays.sort(aa);
		for (int h = 0; h < k; h++)
			print(aa[h] + " ");
		println();
	}
}
