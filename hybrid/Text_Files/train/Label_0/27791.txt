#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 200010;
const int K = 5;
int n, k;
vector<int> adj[N];
ll ans = 0;
vector<pair<ll, int>> dfs(int u = 1, int p = -1) {
  vector<pair<ll, int>> ret(k);
  vector<vector<pair<ll, int>>> children;
  for (int v : adj[u]) {
    if (v == p) continue;
    children.emplace_back(dfs(v, u));
    auto& vec = children.back();
    for (int i = 0; i < k; i++) {
      ret[(i + 1) % k].first += vec[i].first + (i == 0) * vec[i].second;
      ret[(i + 1) % k].second += vec[i].second;
    }
  }
  ll tmp = 0;
  for (auto& vec : children) {
    for (int i = 0; i < k; i++) {
      ret[(i + 1) % k].first -= vec[i].first + (i == 0) * vec[i].second;
      ret[(i + 1) % k].second -= vec[i].second;
    }
    for (int i = 0; i < k; i++) {
      ll d = vec[i].first + (i == 0) * vec[i].second;
      ll q = vec[i].second;
      for (int j = 0; j < k; j++) {
        tmp += d * ret[j].second + ret[j].first * q;
        int fr = ((i + 1) % k > 0) + (j > 0);
        int to = (((i + 1) % k) + j + k - 1) / k;
        tmp += q * ret[j].second * (to - fr);
      }
    }
    for (int i = 0; i < k; i++) {
      ret[(i + 1) % k].first += vec[i].first + (i == 0) * vec[i].second;
      ret[(i + 1) % k].second += vec[i].second;
    }
  }
  tmp /= 2;
  for (int j = 0; j < k; j++) {
    tmp += ret[j].first;
  }
  ret[0].second++;
  ans += tmp;
  return ret;
}
int main() {
  scanf("%d %d", &n, &k);
  for (int i = 1; i < n; i++) {
    int u, v;
    scanf("%d %d", &u, &v);
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs();
  printf("%lld\n", ans);
  return 0;
}
