import java.io.*;
import java.util.*;

public class R1C
{
    public static StringTokenizer st;
    public static void nextLine(BufferedReader br) throws IOException
    {
        st = new StringTokenizer(br.readLine());
    }
    
    public static String next()
    {
        return st.nextToken();
    }
    
    public static int nextInt()
    {
        return Integer.parseInt(st.nextToken());
    }
    
    public static long nextLong()
    {
        return Long.parseLong(st.nextToken());
    }
    
    public static double nextDouble()
    {
        return Double.parseDouble(st.nextToken());
    }
    
    static String s;
    static int[] dp, next;
    
    public static void main(String[] args) throws IOException
    {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        nextLine(br);
        int n = nextInt();
        int k = nextInt();
        int[] p = new int[n];
        int[] e = new int[n];
        ArrayList<Integer> points = new ArrayList<Integer>();
        int maxp = 0;
        for (int i = 0; i < n; i++)
        {
            nextLine(br);
            p[i] = nextInt();
            e[i] = nextInt();
            points.add(p[i]);
            maxp = Math.max(maxp, p[i]);
        }
        
        if (k == n + 1)
        {
            System.out.println(0);
            return;
        }
        Collections.sort(points);
        int kpts = points.get(n - k);
        if (kpts > n)
        {
            System.out.println(-1);
            return;
        }
        ArrayList<Integer> pool1 = new ArrayList<Integer>();
        ArrayList<Integer> pool2 = new ArrayList<Integer>();
        ArrayList<Integer> pool3 = new ArrayList<Integer>();
        ArrayList<Integer> pool4 = new ArrayList<Integer>();
        int plow = 0, plt = 0, peq = 0, pgt = 0, phigh = 0;
        for (int i = 0; i < n; i++)
        {
            if (p[i] + 1 < kpts)
            {
                pool1.add(e[i]);
                plow++;
            }
            else if (p[i] > kpts + 1)
            {
                pool1.add(e[i]);
                phigh++;
            }
            else if (p[i] == kpts + 1)
            {
                pool4.add(e[i]);
                pgt++;
            }
            else if (p[i] + 1 == kpts)
            {
                pool2.add(e[i]);
                plt++;
            }
            else 
            {
                pool3.add(e[i]);
                peq++;
            }
        }
        
        ArrayList<Integer> cmlist1 = new ArrayList<Integer>();
        cmlist1.addAll(pool2);
        cmlist1.addAll(pool3);
        Collections.sort(cmlist1);
        ArrayList<Integer> cmlistx = new ArrayList<Integer>();
        cmlistx.addAll(pool1);
        cmlistx.addAll(pool4);
        Collections.sort(cmlistx);
        
        int mb1 = (n - k + 1) - plow;
        long score1 = 0;
        if (mb1 <= kpts)
        {
            for (int i = 0; i < mb1; i++)
            {
                score1 += cmlist1.get(i);
            }
            int tt = Math.min(cmlistx.size(), kpts - mb1);
            for (int i = 0; i < tt; i++)
            {
                score1 += cmlistx.get(i);
            }
            for (int i = 0; i < kpts - mb1 - tt; i++)
            {
                score1 += cmlist1.get(mb1 + i);
            }
            int pt = kpts - tt;
            long val1 = score1;
            while (tt > 0 && pt < cmlist1.size())
            {
                val1 -= cmlistx.get(tt - 1);
                val1 += cmlist1.get(pt);
                if (val1 < score1) score1 = val1;
                tt--;
                pt++;
            }
        }
        else score1 = Long.MAX_VALUE;
        
        int mb2 = (n - k + 1) - plow - plt;
        long score2 = 0;
        ArrayList<Integer> cmlist2 = new ArrayList<Integer>();
        cmlist2.addAll(pool1);
        cmlist2.addAll(pool2);
        Collections.sort(cmlist2);
        cmlistx.clear();
        cmlistx.addAll(pool3);
        cmlistx.addAll(pool4);
        Collections.sort(cmlistx);
        if (mb2 <= kpts + 1 && kpts + 1 <= n)
        {
            for (int i = 0; i < mb2; i++)
            {
                score2 += cmlistx.get(i);
            }
            int tt = Math.min(cmlist2.size(), kpts + 1 - mb2);
            for (int i = 0; i < tt; i++)
            {
                score2 += cmlist2.get(i);
            }
            for (int i = 0; i < kpts + 1 - mb2 - tt; i++)
            {
                score2 += cmlistx.get(mb2 + i);
            }
            int pt = kpts + 1 - tt;
            long val2 = score2;
            while (tt > 0 && pt < cmlistx.size())
            {
                val2 -= cmlist2.get(tt - 1);
                val2 += cmlistx.get(pt);
                if (val2 < score2) score2 = val2;
                tt--;
                pt++;
            }
        }
        else score2 = Long.MAX_VALUE;
        
        cmlistx.addAll(cmlist2);
        Collections.sort(cmlistx);
        long score3 = 0;
        if (kpts + 2 <= n)
        {
            for (int i = 0; i < kpts + 2; i++)
            {
                score3 += cmlistx.get(i);
            }
        }
        else score3 = Long.MAX_VALUE;
        
        long ans = Math.min(score1, Math.min(score2, score3));
        if (ans == Long.MAX_VALUE) System.out.println(-1);
        else System.out.println(ans);
    }
}
