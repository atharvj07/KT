from math import ceil

def main():
	t = int(input())
	for _ in range(t):
		(A, B, n) = map(int, input().split())
		a = [int(x) for x in input().split()]
		b = [int(x) for x in input().split()]
		max_a = max(a)
		truth = True
		for i in range(n):
			fights = ceil(b[i] / A)
			B -= fights * a[i]
			if B <= -max_a:
				print('NO')
				truth = False
				break
		if truth:
			print('YES')
main()
