template<class T> T &chmin(T &a,const T &b){ return a = min(a,b); }
template<class T> T &chmax(T &a,const T &b){ return a = max(a,b); }
 
const int INF=1e+18;
const double EPS=1e-9;
const int MOD=1000000007;

const int dx[]={1,0,-1,0},dy[]={0,-1,0,1};

signed main(){
	int dp[2][1 << 16],*now = dp[0],*nxt = dp[1];
	int h,w;
	string s[130];
	cin >> h >> w;
	for(int i = 0;i < h;i++) cin >> s[i];
	for(int i = 0;i < (1 << w);i++) now[i] = nxt[i] = -INF;
	now[0] = 0;
	int popcount[1 << 16];
	for(int i = 0;i < (1 << w);i++) popcount[i] = __builtin_popcountll(i);
	for(int i = 0;i < h;i++){
		for(int j = 0;j < w;j++){
			for(int k = 0;k < (1 << w);k++) nxt[k] = -INF;
			for(int k = 0;k < (1 << w);k++){
				int l = popcount[k % (1 << j)];
				if(now[k] < 0) continue;
				int up = s[i-1].length()-popcount[k >> j];
				int add = (((k>>j)&1) && i && s[i-1][up]==s[i][l])
				+ (l && ((k>>(j-1))&1) && s[i][l-1]==s[i][l]);
				chmax(nxt[k | (1 << j)],now[k] + 2 * add);
				chmax(nxt[k & ~(1 << j)],now[k]);
			}
			swap(now,nxt);
		}
		for(int k = 0;k < (1 << w);k++){
			if(popcount[k] != s[i].length()) now[k] = -INF;
		}
	}
	int ans = -INF;
	for(int i = 0;i < (1 << w);i++) chmax(ans,now[i]);
	cout << ans << endl;
}
