def solve():
	(n, m, no_pairs) = input().split()
	n = int(n)
	m = int(m)
	no_pairs = int(no_pairs)
	male = input().split()
	female = input().split()
	male_pair = [0] * n
	female_pair = [0] * m
	res = 0
	for i in male:
		male_pair[int(i) - 1] += 1
	for j in female:
		female_pair[int(j) - 1] += 1
	for i in range(len(male)):
		index_male = int(male[i])
		index_fmale = int(female[i])
		res += len(male) - male_pair[index_male - 1] - female_pair[index_fmale - 1] + 1
	return int(res / 2)
no_tests = int(input())
results = []
for i in range(no_tests):
	results.append(solve())
for result in results:
	print(result)
