def run(l):
	r = 0
	for i in range(len(l)):
		b = False
		for j in l[i]:
			if j == '1':
				b = True
			if j == '0' and b == True:
				r += 1
	return r
(a, b) = map(int, input().split())
r = [''.join(input().split()) for i in range(a)]
l = [i[::-1] for i in r]
u = [''.join([i[j] for i in r]) for j in range(b)]
d = [i[::-1] for i in u]
print(run(l) + run(r) + run(u) + run(d))
