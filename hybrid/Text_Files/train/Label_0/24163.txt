#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC target( \
    "avx,avx2,fma,sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx,avx,tune=native")
using namespace std;
long long MOD = 998244353;
long long INF = 1e18;
const int N = 2000;
long long dp[2005][4005];
long long st(long long t, long long id, long long mod) {
  long long res = 1, a = t, n = id;
  while (n) {
    if (n & 1) res = (res * a) % mod;
    a = (a * a) % mod;
    n >>= 1;
  }
  return res % mod;
}
long long fac[400000];
long long ifac[400000];
long long del(long long a, long long b) {
  return (a * st(b, MOD - 2, MOD)) % MOD;
}
long long cnk(long long n, long long k) {
  return fac[n] * ifac[n - k] % MOD * ifac[k] % MOD;
}
int main() {
  ios::sync_with_stdio(0), cin.tie(0), cout.tie(0),
      cout << fixed << setprecision(20);
  fac[0] = 1;
  for (int i = 1; i < 4e5; i++) {
    fac[i] = fac[i - 1] * i % MOD;
  }
  ifac[(int)(4e5 - 1)] = del(1, fac[(int)(4e5 - 1)]);
  for (int i = 4e5 - 2; i > -1; i--) {
    ifac[i] = ifac[i + 1] * (i + 1) % MOD;
  }
  long long n, k;
  cin >> n >> k;
  vector<long long> b(n);
  vector<long long> a;
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  long long ans = 1;
  for (int i = 0; i < n; i++) {
    if (b[i] != b[(i + 1) % n]) {
      a.push_back(b[i]);
    } else {
      ans *= k;
      ans %= MOD;
    }
  }
  dp[0][0 + N] = 1;
  n = a.size();
  long long ans0 = st(k, n, MOD);
  for (int i = 0; 2 * i <= n; i++) {
    ans0 -=
        st(k - 2, n - 2 * i, MOD) * cnk(n, 2 * i) % MOD * cnk(2 * i, i) % MOD;
    ans0 %= MOD;
  }
  ans0 = del(ans0, 2);
  ans *= ans0;
  ans %= MOD;
  ans += MOD;
  ans %= MOD;
  cout << ans;
}
