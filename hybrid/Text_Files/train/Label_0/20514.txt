#include <bits/stdc++.h>
using namespace std;
int dx[] = {0, 0, 1, -1, 1, 1, -1, -1};
int dy[] = {1, -1, 0, 0, -1, 1, 1, -1};
const int mod = 1e9 + 7;
int dcmp(long double x, long double y) {
  return fabs(x - y) <= 1e-12 ? 0 : x < y ? -1 : 1;
}
void fast() {
  std::ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  cout.tie(NULL);
}
int n, rate, N;
vector<pair<int, int> > pos, neg;
bool cmp(pair<int, int> a, pair<int, int> b) {
  return a.first + a.second > b.first + b.second;
}
int dp[101][30 * 1000 + 5];
int rec(int idx, int r) {
  if (idx == N) {
    return 0;
  }
  int &ret = dp[idx][r];
  if (~ret) return ret;
  if (neg[idx].first <= r)
    ret = max(ret, rec(idx + 1, r + neg[idx].second) + 1);
  ret = max(ret, rec(idx + 1, r));
  return ret;
}
int main() {
  fast();
  cin >> n >> rate;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    if (r >= 0)
      pos.push_back({l, r});
    else {
      l = max(l, -r);
      neg.push_back({l, r});
    }
  }
  sort(pos.begin(), pos.end());
  sort(neg.begin(), neg.end(), cmp);
  int ans = 0;
  for (int i = 0; i < pos.size(); i++) {
    if (pos[i].first <= rate) {
      ans++;
      rate += pos[i].second;
    } else
      break;
  }
  N = neg.size();
  memset(dp, -1, sizeof dp);
  ans += rec(0, rate);
  cout << ans << '\n';
  return 0;
}
