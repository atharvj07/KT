T = int(input())
for _ in range(T):
	(n, x, m) = input().split()
	n = int(n)
	x = int(x)
	m = int(m)
	count = 1
	x1 = x
	x2 = x
	for i in range(m):
		(l, r) = input().split()
		l = int(l)
		r = int(r)
		if x1 >= l and x1 <= r:
			count += x1 - l
			x1 = l
		if x2 >= l and x2 <= r:
			count += r - x2
			x2 = r
	print(count)
