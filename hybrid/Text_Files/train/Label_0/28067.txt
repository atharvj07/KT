#include<bits/stdc++.h>
#define REP(x,y,z) for(int x=y;x<=z;x++)
#define FORD(x,y,z) for(int x=y;x>=z;x--)
#define MSET(x,y) memset(x,y,sizeof(x))
#define FOR(x,y) for(__typeof(y.begin()) x=y.begin();x!=y.end();x++)
#define F first
#define S second
#define MP make_pair
#define PB push_back
#define SZ size()
#define M 100005
#define INF 2000000000
void RI(){}
template<typename... T>
void RI( int& head, T&... tail ) {
    scanf("%d",&head);
    RI(tail...);
}
using namespace std;
typedef long long LL;
typedef pair<int,int> Pi;
int n,fa[21][M],sum[M],dep[M];
vector<Pi> e[M];
namespace Init
{
	int m,t[M],dis[M],day[M],fr[M];
	vector<Pi> E[M];
	struct P
	{
		int d,day,fr,to;
		P(){}
		P(int a,int b,int c,int dd)
		{
			d = a;
			day = b;
			fr = c;
			to = dd;
		}
	};
	bool operator < (P a,P b)
	{
		if(a.d != b.d) return a.d > b.d;
		if(a.day != b.day) return a.day > b.day;
		return t[a.fr] > t[b.fr];
	}
	priority_queue<P> q;
	
	void init()
	{
		int cur;
		int x,y,z;
		P st;
		RI(n,m);
		REP(i,1,n) RI(t[i]);
		REP(i,1,m)
		{
			RI(x,y,z);
			E[x].PB( MP(y,z) );
			E[y].PB( MP(x,z) );
		}

		REP(i,1,n) dis[i] = INF;
		q.push( P(0,0,0,1) );
		while(!q.empty())
		{
			cur = q.top().to;
			st = q.top();
			q.pop();
			if(dis[cur] <= st.d) continue;
			
			dis[cur] = st.d;
			day[cur] = st.day;
			fr[cur] = st.fr;
			FOR(i,E[cur]) if(dis[i->F]==INF)
				q.push( P(dis[cur]+i->S, day[cur]+1, cur, i->F) );
		}
		///
		
		REP(i,1,n) if(fr[i])
		{
			z = dis[i] - dis[fr[i]];
			x = i;
			y = fr[i];
			e[x].PB( MP(y,z) );
			e[y].PB( MP(x,z) );
		}
	}
}
void dfs(int cur,int f,int s,int d)
{
	sum[cur] = s;
	fa[0][cur] = f;
	dep[cur] = d;

	FOR(i,e[cur]) if(i->F!=f)
		dfs(i->F, cur, s+i->S, d+1 );
}
int up(int x,int d)
{
	if(d<=0) return x;
	FORD(i,20,0) if(d & (1<<i)) x=fa[i][x];
	return x;
}
int main()
{
	Init::init();
	dfs(1,-1,0,0);
	REP(i,1,20) REP(j,1,n)
	{
		if(fa[i-1][j]==1) fa[i][j]=1;
		else fa[i][j] = fa[i-1][fa[i-1][j]];
	}

	int q;
	int st,d,p;
	int free;
	int Up, Down;
	RI(q);
	while(q--)
	{
		RI(st,d,p);
		free = up(st, dep[st]-d);

		Down = sum[st] - sum[free];
		Up = sum[free];
		Up = max(0, Up-p);
		printf("%d\n", Up+Down);
	}
	
	return 0;
}
