#include <bits/stdc++.h>
using namespace std;
int a[100010], b[100010];
int n, m, k;
int main() {
  scanf("%d %d %d", &n, &m, &k);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int i = 0; i < m; i++) scanf("%d", &b[i]);
  sort(a, a + n);
  sort(b, b + m);
  int con = 0;
  bool flag = false;
  int i, j;
  for (i = n - 1, j = m - 1; i >= 0 && j >= 0;) {
    if (a[i] == b[j]) {
      i--;
      j--;
    } else {
      if (a[i] < b[j]) {
        j--;
        con--;
      } else {
        i--;
        con++;
      }
    }
    if (con > 0) {
      flag = true;
      break;
    }
  }
  if (j < 0 && i >= 0 && con + i + 1 > 0) {
    printf("YES");
    return 0;
  }
  if (flag)
    printf("YES");
  else
    printf("NO");
}
