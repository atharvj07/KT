import java.util.Scanner;

public class A665 {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int a = scanner.nextInt();
    int ta = scanner.nextInt();
    int b = scanner.nextInt();
    int tb = scanner.nextInt();
    String[] la = scanner.next().split(":");
    int st = Integer.parseInt(la[0]) * 60 - 5 * 60 + Integer.parseInt(la[1]);
    int bd = 0;
    int ed = 1140;
    int et = Math.min(st + ta, ed);
    st = Math.max(st + 1 - tb, bd);
    boolean[] sb = new boolean[ed];
    for (int i = 0; i < ed; i += b) {
      sb[i] = true;
    }
    int res = 0;
    for (int i = st; i < et; i++) {
      if (sb[i])
        res++;
    }
    System.out.println(res);
  }
}
