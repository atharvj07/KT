#include <bits/stdc++.h>
using namespace std;
template <class t>
inline t read(t &x) {
  x = 0;
  char c = getchar();
  bool f = 0;
  while (!isdigit(c)) f |= c == '-', c = getchar();
  while (isdigit(c)) x = (x << 1) + (x << 3) + (c ^ 48), c = getchar();
  if (f) x = -x;
  return x;
}
template <class t>
inline void write(t x) {
  if (x < 0) {
    putchar('-'), write(-x);
  } else {
    if (x > 9) write(x / 10);
    putchar('0' + x % 10);
  }
}
const int N = 5e5 + 5;
map<int, int> v;
char s[N];
int n, ans;
int calc() {
  int n = strlen(s), x = 0, y = 0;
  bool l = 0, r = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(')
      x++;
    else
      x--;
    l |= x < 0;
  }
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == ')')
      y++;
    else
      y--;
    r |= y < 0;
  }
  if (l && r) return INT_MAX;
  return x;
}
signed main() {
  read(n);
  for (int i = 1; i <= n; i++) {
    scanf("%s", s);
    int x = calc();
    if (x == INT_MAX) continue;
    if (v[-x])
      v[-x]--, ans++;
    else
      v[x]++;
  }
  write(ans);
}
