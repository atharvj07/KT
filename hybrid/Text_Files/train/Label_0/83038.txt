for _ in range(int(input())):
	x = list(input())
	size = len(set(x))
	if size == 1:
		print(''.join(x), end='')
	else:
		i = 0
		prev = str((int(x[0]) + 1) % 2)
		while i < len(x):
			if prev != x[i]:
				print(x[i], end='')
				prev = x[i]
				i += 1
			else:
				print(str((int(x[i]) + 1) % 2), end='')
				prev = str((int(x[i]) + 1) % 2)
	print()
