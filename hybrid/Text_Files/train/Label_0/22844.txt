#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
const int logN = 18;
const int INF = 1e9 + 10;
int n, p, k, a[N], nxt[N][logN], cost[N][logN];
void Init() {
  for (int i = 1; i <= n - k + 1; i++) {
    nxt[i][0] = i + k;
    cost[i][0] = a[i + k - 1];
  }
  for (int i = n - k + 2; i <= n; i++) {
    nxt[i][0] = n + 1;
    cost[i][0] = INF;
  }
  nxt[n + 1][0] = n + 1;
  cost[n + 1][0] = INF;
  for (int j = 1; j <= logN - 1; j++)
    for (int i = 1; i <= n + 1; i++) {
      nxt[i][j] = nxt[nxt[i][j - 1]][j - 1];
      cost[i][j] = min(cost[i][j - 1] + cost[nxt[i][j - 1]][j - 1], INF);
    }
}
int main() {
  int tc = 0;
  scanf("%d", &tc);
  while (tc--) {
    scanf("%d %d %d", &n, &p, &k);
    for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
    sort(a + 1, a + 1 + n);
    Init();
    int curCost = 0;
    int ans = 0;
    for (int i = 1; i <= k; i++) {
      int tmpCost = curCost;
      int curPos = i;
      for (int j = logN - 1; j >= 0; j--) {
        if (tmpCost + cost[curPos][j] <= p) {
          tmpCost += cost[curPos][j];
          curPos = nxt[curPos][j];
        }
      }
      if (tmpCost <= p) ans = max(ans, curPos - 1);
      curCost += a[i];
    }
    printf("%d\n", ans);
  }
}
