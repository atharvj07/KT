t = int(input())
for _ in range(t):
	(n, m1) = [int(i) for i in input().split()]
	f = -1
	if n == m1:
		print(1)
		continue
	elif n < m1:
		print(1)
		continue
	else:
		m = n
		n1 = int(n ** 0.5)
		for i in range(1, n1 + 1):
			if n % i == 0:
				if i <= m1:
					m = min(m, int(n / i))
				if int(n / i) <= m1:
					m = min(m, i)
		print(m)
