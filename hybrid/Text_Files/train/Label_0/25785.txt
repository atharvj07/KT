#include <bits/stdc++.h>
using namespace std;
const int maxn = 5050;
const int num = 1e5;
int prime[num / 5], pri_cnt;
bool flag[num + 10];
set<int> bad;
int a[maxn], g[maxn];
int n, m;
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
int f(int x) {
  if (x == 1) return 0;
  int ret = 0;
  for (int i = 0; i < pri_cnt && prime[i] * prime[i] <= x; i++)
    if (x % prime[i] == 0) {
      if (bad.count(prime[i])) {
        while (x % prime[i] == 0) ret--, x /= prime[i];
      } else {
        while (x % prime[i] == 0) ret++, x /= prime[i];
      }
    }
  if (x > 1) {
    if (bad.count(x))
      ret--;
    else
      ret++;
  }
  return ret;
}
int main() {
  for (int i = 2; i * i <= num; i++)
    if (!flag[i]) {
      for (long long j = i * i; j <= num; j += i) flag[j] = true;
    }
  for (int i = 2; i <= num; i++)
    if (!flag[i]) prime[pri_cnt++] = i;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    g[i] = gcd(g[i - 1], a[i]);
  }
  for (int i = 1; i <= m; i++) {
    int x;
    scanf("%d", &x);
    bad.insert(x);
  }
  int ans = 0, s = 1;
  for (int i = n; i >= 1; i--) {
    g[i] /= s;
    if (f(g[i]) < 0) s *= g[i];
    a[i] /= s;
    ans += f(a[i]);
  }
  printf("%d\n", ans);
  return 0;
}
