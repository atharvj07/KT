#include <bits/stdc++.h>

using namespace std;

#define rep(i, n) for(int i = 0; i < (n); i++)
#define repl(i, l, r) for(int i = (l); i < (r); i++)
#define per(i, n) for(int i = ((n)-1); i >= 0; i--)
#define perl(i, l, r) for(int i = ((r)-1); i >= (l); i--)
#define all(x) (x).begin(),(x).end()
#define MOD9 998244353
#define MOD1 1000000007
#define IINF 1000000000
#define LINF 1000000000000000000
#define SP <<" "<<
#define CYES cout<<"Yes"<<endl
#define CNO cout<<"No"<<endl
#define CFS cin.tie(0);ios::sync_with_stdio(false)
#define CST(x) cout<<fixed<<setprecision(x)

typedef long long ll;
typedef long double ld;
typedef vector<int> vi;
typedef vector<vector<int>> mti;
typedef vector<ll> vl;
typedef vector<vector<ll>> mtl;

int main(){
    int n, k;
    cin >> n >> k;
    vi a(n);
    int s = 0;
    rep(i, n) cin >> a[i], s += a[i];
    sort(all(a), greater<int>());
    vi r(k), b(k);
    int rs = 0, bs = 0;
    rep(i, k) cin >> r[i], rs += r[i];
    sort(all(r), greater<int>());
    rep(i, k) cin >> b[i], bs += b[i];
    sort(all(b), greater<int>());
    vector<mti> dp(k + 1, mti(rs + 1, vi(2, 10000)));
    dp[0][0][0] = 0;
    rep(i, n){
        perl(j, 1, k + 1){
            per(x, rs + 1){
                int tmp = dp[j][x][0];
                if(x >= a[i]){
                    dp[j][x][0] = min(dp[j - 1][x - a[i]][0], dp[j - 1][x - a[i]][1]) + max(0, a[i] - r[j - 1]);
                    dp[j][x][0] = min(10000, dp[j][x][0]);
                }else{
                    dp[j][x][0] = 10000;
                }
                if(0 <= i-j && i-j < k){
                    dp[j][x][1] = min(tmp, dp[j][x][1]) + max(0, a[i] - b[i - j]);
                    dp[j][x][1] = min(10000, dp[j][x][1]);
                }else{
                    dp[j][x][1] = 10000;
                }
            }
        }
        if(i < k){
            per(x, rs + 1){
                dp[0][x][1] = min(dp[0][x][0], dp[0][x][1]) + max(0, a[i] - b[i]);
                dp[0][x][1] = min(10000, dp[0][x][1]);
            }
        }else{
            per(x, rs + 1){
                dp[0][x][1] = 10000;
            }
        }
        dp[0][0][0] = 10000;
    }
    int ans = 10000;
    rep(j, k + 1){
        rep(x, rs + 1){
            if(x <= rs && s - x <= bs){
                ans = min(ans, min(dp[j][x][0], dp[j][x][1]));
            }
        }
    }
    if(ans == 10000) cout << -1 << endl;
    else cout << ans << endl;
    return 0;
}
