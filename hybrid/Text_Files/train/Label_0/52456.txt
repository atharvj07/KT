import java.io.*;
import java.util.*;
public class Main
{
	
	private void solve()throws Exception
	{
		int n=nextInt();
		int m=nextInt();
		Edge[] edges=new Edge[m+1];
		for(int i=1;i<=m;i++)
			edges[i]=new Edge(nextInt(),nextInt(),nextInt());
		Arrays.sort(edges,1,m+1,(p,q)->p.w-q.w);
		int low=n,high=m,ans=-1;
		while(low<=high)
		{
			int mid=(low+high)/2;
			HopcroftKarp hk=new HopcroftKarp(n,n);
			for(int i=1;i<=mid;i++)
				hk.addEdge(edges[i].u,edges[i].v);
			if(hk.maxFlow()==n)
			{
				ans=edges[mid].w;
				high=mid-1;
			}
			else
				low=mid+1;
		}
		out.println(ans);
	}
	class Edge
	{
		int u,v,w;
		Edge(int u,int v,int w)
		{
			this.u=u;
			this.v=v;
			this.w=w;
		}
	}
	class HopcroftKarp
	{
		int m,n;
		ArrayList<Integer> list[];
		int pair_u[],pair_v[],dist[];
		HopcroftKarp(int m,int n)
		{
			this.m=m;
			this.n=n;
			list=new ArrayList[m+1];
			for(int i=1;i<=m;i++)
				list[i]=new ArrayList<>();
			pair_u=new int[m+1];
			pair_v=new int[n+1];
			dist=new int[m+1];
		}
		void addEdge(int u,int v)
		{
			list[u].add(v);
		}
		int maxFlow()
		{
			int res=0;
			while(bfs())
			{
				for(int u=1;u<=m;u++)
					if(pair_u[u]==0 && dfs(u))
						res++;
			}
			return res;
		}
		boolean bfs()
		{
			Queue<Integer> queue=new LinkedList<>();
			for(int u=1;u<=m;u++)
				if(pair_u[u]==0)
				{
					dist[u]=0;
					queue.add(u);
				}
				else
					dist[u]=1<<30;
			boolean hasPath=false;
			while(!queue.isEmpty())
			{
				int u=queue.remove();
				for(int v:list[u])
					if(pair_v[v]==0)
						hasPath=true;
					else if(dist[pair_v[v]]>dist[u]+1)
					{
						dist[pair_v[v]]=dist[u]+1;
						queue.add(pair_v[v]);
					}
			}
			return hasPath;
		}
		boolean dfs(int u)
		{
			for(int v:list[u])
				if(pair_v[v]==0 || dist[pair_v[v]]==dist[u]+1 && dfs(pair_v[v]))
				{
					pair_v[v]=u;
					pair_u[u]=v;
					return true;
				}
			return false;
		}
	}

	 
	///////////////////////////////////////////////////////////

	public void run()throws Exception
	{
		br=new BufferedReader(new InputStreamReader(System.in));
		st=null;
		out=new PrintWriter(System.out);

		try{
			solve();

		}catch(Exception e){e.printStackTrace();}
		finally{
			br.close();
			out.close();
		}
		
	}
	public static void main(String args[])throws Exception{
		new Main().run();
	}
	BufferedReader br;
	StringTokenizer st;
	PrintWriter out;
	String nextToken()throws Exception{
		while(st==null || !st.hasMoreTokens())
		st=new StringTokenizer(br.readLine());
		return st.nextToken();
	}
	String nextLine()throws Exception{
		return br.readLine();
	}
	int nextInt()throws Exception{
		return Integer.parseInt(nextToken());
	}
	long nextLong()throws Exception{
		return Long.parseLong(nextToken());
	}
	double nextDouble()throws Exception{
		return Double.parseDouble(nextToken());
	}
}
