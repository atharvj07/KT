def lower_bound(arr, left, right, target):
	if arr[right] < target:
		return right + 1
	while left < right:
		mid = left + right >> 1
		if arr[mid] >= target:
			right = mid
		else:
			left = mid + 1
	return left
T = int(input())
arr = [0] * T
dp = [0] * (T + 1)
for i in range(T):
	arr[i] = int(input())
for i in range(T):
	dp[i + 1] = dp[i] + 20
	dp[i + 1] = min(dp[i + 1], dp[lower_bound(arr, 0, T - 1, arr[i] - 89)] + 50)
	dp[i + 1] = min(dp[i + 1], dp[lower_bound(arr, 0, T - 1, arr[i] - 1439)] + 120)
	print(dp[i + 1] - dp[i])
