import java.util.*;

public class p001129c {
	static int[][] dp;
	static int MOD = (int)1e9+7;
	static boolean[][] done;
	static int[] string;
	
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		dp = new int[n][n];
		string = new int[n];
		long tot = 0;
		int nextInd = 1;
		int size = (int)5e6;
		boolean[] isWord = new boolean[size];
		int[][] child = new int[2][size];
		for(int i = 0; i < n; i++) string[i] = in.nextInt();
		for(int i = 0; i < n; i++) {
			int curr = 0;
			if(child[string[i]][curr] == 0) {
				child[string[i]][curr] = nextInd++;
			}
			curr = child[string[i]][curr];
			for(int j = i; j >= 0; j--) {
				if(!isWord[curr]) {
					tot += dp(j, i);
					isWord[curr] = true;
				}
				if(j != 0) {
					if(child[string[j-1]][curr] == 0) {
						child[string[j-1]][curr] = nextInd++;
					}
					curr = child[string[j-1]][curr];
				}
			}
			tot %= MOD;
			System.out.println(tot);
		}
	}

	static int dp(int s, int n) {
		if(n < s) return 1;
		if(n == s) return 1;
		if(dp[s][n] != 0) return dp[s][n];
		int res = 0;
		long num = 0;
		for(int i = n; i >= Math.max(n-3, s); i--) {
			res = (res << 1) | (string[i]);
			if(i == n-3 &&(res == 0b1100 || res == 0b1010 || res == 0b0111 || res == 0b1111)) continue;
			num += dp(s, i-1);
		}
		num %= MOD;
		return dp[s][n] = (int)num;
	}
}
