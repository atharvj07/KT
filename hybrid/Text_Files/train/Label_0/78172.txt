from typing import Counter
n = int(input())
l = list(map(int, input().split()))

def solve(l):
	c = Counter(l)
	points = [0] * (max(l) + 1)
	for i in sorted(list(set(l))):
		points[i] = i * c[i]

	def robber(points):
		if not points:
			return 0
		if len(points) == 1:
			return points[0]
		if len(points) == 2:
			return max(points[0], points[1])
		n = len(points)
		dp = [0] * n
		dp[0] = points[0]
		dp[1] = max(points[0], points[1])
		for i in range(2, len(points)):
			dp[i] = max(dp[i - 1], points[i] + dp[i - 2])
		return dp[-1]
	return robber(points)
print(solve(l))
