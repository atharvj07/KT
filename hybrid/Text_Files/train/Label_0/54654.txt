import java.util.Scanner;

public class Codeforces {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String a = in.nextLine();
        int result=0;
        int n = (a.charAt(0) - '0');
        if(a.charAt(1)>='0'&&a.charAt(1)<='9'){
        n = 10*n + a.charAt(1) - '0';
        }
        if (a.substring(1, a.length()).equals(" of month")||a.substring(2, a.length()).equals(" of month")) {
            if(n==31) result = 7;
            if(n==30) result = 11;
            if(n<30) result = 12;
        }
        if(a.substring(1, a.length()).equals(" of week")){
            if(n==5||n==6){
                result=53;
            }else result=52;
        }
        System.out.println(result);

    }
}
