def find_ans(arr):
	if len(arr) <= 2:
		return 0
	diff_arr = []
	for i in range(0, len(arr) - 1):
		diff_arr.append(arr[i + 1] - arr[i])
	diff_arr = list(set(diff_arr))
	diff_arr.sort()
	if len(diff_arr) > 2:
		return -1
	elif len(diff_arr) == 2:
		ans = find_m_c(diff_arr)
