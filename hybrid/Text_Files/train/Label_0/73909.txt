from sys import stdin

def solve():
	for _ in range(int(stdin.readline())):
		(a, b) = map(int, stdin.readline().split())
		a1 = min(a, b)
		b1 = max(a, b)
		k = 2 * b1 - a1
		if k % 3 == 0:
			if a1 - k // 3 >= 0 and b1 - 2 * (k // 3) >= 0:
				print('YES')
			else:
				print('NO')
		else:
			print('NO')
solve()
