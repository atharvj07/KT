#include <iostream>
#include <vector>
#include <algorithm>
#include <map>

using namespace std;
typedef pair<int,int> pii;
typedef pair<int,pii> pp;

int main(void){
    int n;
    cin >> n;
    int ans = 0;
    vector<pp> a;
    vector<pii> b;
    vector<int> c;
    for (int i = 0; i < n; i++) {
        int x,y,z;
        cin>>x>>y>>z;
        a.push_back(make_pair(x, make_pair(y,z)));
    }
    sort(a.begin(), a.end());
    
    for (int i = 0; i < n; i++) {
        int l = i;
        map<int,int> t, u;
        int y = a[i].second.first;
        int z = a[i].second.second;
        t[y] = z;
        while (i+1 < n && a[i].first == a[i+1].first) {
            i++;
            y = a[i].second.first;
            z = a[i].second.second;
            t[y] = z;
        }
        int j = i;
        if (a[j].first+1 == a[j+1].first) {
            j++;
            y = a[j].second.first;
            z = a[j].second.second;
            u[y] = z;
            while (j+1 < n && a[j].first == a[j+1].first) {
                j++;
                y = a[j].second.first;
                z = a[j].second.second;
                u[y] = z;
            }
        }
        for (int j = l; j < i+1; j++) {
            y = a[j].second.first;
            ans = max(ans, t[y]+t[y+1]+u[y]+u[y+1]);
            ans = max(ans, t[y]+t[y-1]+u[y]+u[y-1]);
        }
    }
    cout << ans << " / 1" << endl;
}
