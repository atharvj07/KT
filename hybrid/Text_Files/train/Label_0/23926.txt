#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:100000000,100000000")
using namespace std;
const long long inf = 1e18 + 7;
const long long mod = 1e9 + 7;
const double eps = 1e-10;
const double PI = 2 * acos(0.0);
const double E = 2.71828;
unsigned long long dp[25][25][2][2], a[21], bit[21];
unsigned long long t[25], sum[11 * 6], se[25];
void cnt(unsigned long long num) {
  string s = "";
  while (num) {
    s += num % 10 + '0';
    num /= 10;
  }
  reverse(s.begin(), s.end());
  while (s.size() < 19) s = '0' + s;
  unsigned long long len = 19;
  for (unsigned long long i = 1; i <= len; i++) a[i] = s[i - 1] - '0';
  for (unsigned long long i = 0; i <= a[1]; i++) {
    dp[1][bit[i] * (i != 0)][i < a[1]][i != 0]++;
  }
  for (unsigned long long i = 2; i <= len; i++)
    for (unsigned long long j = 0; j <= 19; j++)
      for (unsigned long long x = 0; x <= 1; x++)
        for (unsigned long long y = 0; y <= 1; y++)
          if (dp[i - 1][j][x][y])
            for (unsigned long long p = 0; p <= 9; p++)
              if (!x && p > a[i])
                break;
              else
                dp[i][j + (bit[p] * (y || p))][x | (p < a[i])][y || p] +=
                    dp[i - 1][j][x][y];
}
void dfs(unsigned long long pos = 0, unsigned long long many = 0,
         unsigned long long now = 1) {
  if (pos == 6)
    sum[many] += now, sum[many] %= mod;
  else {
    for (unsigned long long i = 0; i < 11; i++) {
      unsigned long long x = now;
      x *= t[i];
      x %= mod;
      t[i]--;
      dfs(pos + 1, many + i, x);
      t[i]++;
    }
  }
}
int main(void) {
  unsigned long long n;
  cin >> n;
  bit[4] = 1;
  bit[7] = 1;
  cnt(n + 1);
  unsigned long long qt = 0;
  for (unsigned long long i = 0; i < 19; i++) {
    t[qt] = dp[19][i][1][1];
    qt++;
  }
  dfs();
  unsigned long long rez = 0;
  for (unsigned long long i = 1; i < 19; i++) {
    for (unsigned long long j = 0; j < i; j++) {
      unsigned long long now = (t[i] * sum[j]) % mod;
      rez += now;
      rez %= mod;
    }
  }
  cout << rez << endl;
  return 0;
}
