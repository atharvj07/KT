#include <cstdio>
#include <vector>

inline int read(){
	char c=getchar(); int x=0,ft=1;
	for(;c<'0'||c>'9';c=getchar()) if(c=='-') ft=-1;
	for(;c>='0'&&c<='9';c=getchar()) x=x*10+c-'0';
	return x*ft;
}

int n,val[3005],f[3005]; std::vector<int> g[3005];

inline void add_edge(int u,int v){
	g[u].push_back(v); g[v].push_back(u);
}

void dfs(int u,int pre){
	f[u]=false;
	for(auto v: g[u]) if(v!=pre&&val[v]<val[u]){
		dfs(v,u); if(!f[v]){ f[u]=true; return; }
	}
}

int main(){
	n=read(); for(int i=1;i<=n;++i) val[i]=read();
	for(int i=1;i<=n-1;++i) add_edge(read(),read());
	for(int i=1;i<=n;++i){ dfs(i,0); if(f[i]) printf("%d ",i); }
	return 0;
}
