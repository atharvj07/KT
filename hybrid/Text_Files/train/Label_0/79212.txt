[n, a, b] = [int(x) for x in input().split()]
[a, b] = sorted([a, b])
matrix = [[[0, 1], [0, 1]]]
sizes = []
for i in range(n):
	sizes.append([int(x) for x in input().split()])
max_area = 0
for x in range(1, n):
	for y in range(x):
		s0 = sizes[x]
		s1 = sizes[y]
		total_area = s1[0] * s1[1] + s0[0] * s0[1]
		combined_size = sorted([max(s0[0], s1[0]), s0[1] + s1[1]])
		if combined_size[0] <= a and combined_size[1] <= b:
			max_area = max(max_area, total_area)
		combined_size = sorted([max(s0[0], s1[1]), s0[1] + s1[0]])
		if combined_size[0] <= a and combined_size[1] <= b:
			max_area = max(max_area, total_area)
		combined_size = sorted([max(s0[1], s1[0]), s0[0] + s1[1]])
		if combined_size[0] <= a and combined_size[1] <= b:
			max_area = max(max_area, total_area)
		combined_size = sorted([max(s0[1], s1[1]), s0[0] + s1[0]])
		if combined_size[0] <= a and combined_size[1] <= b:
			max_area = max(max_area, total_area)
print(max_area)
