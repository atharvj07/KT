n = int(input())
l = list(map(int, input().split()))
m = min(l)
if n < m:
	print('-1')
elif n % m == 0:
	s = -1
	for i in range(8, -1, -1):
		if l[i] == m:
			s = i + 1
			break
	print(str(s) * int(n / m))
else:
	s = -1
	for i in range(8, -1, -1):
		if l[i] == m:
			s = i + 1
			break
	st = str(s) * int(n / m)
	st = list(st)
	n -= len(st) * m
	j = 8
	index = 0
	while n != 0 and j >= 0 and (index < len(st)):
		if l[j] <= m + n:
			st[index] = str(j + 1)
			index += 1
			n = n + m - l[j]
		else:
			j -= 1
	print(''.join(st))
