import math
(l, r, k) = map(int, input().split())
base_l = math.floor(math.log(l, k))
base_r = math.ceil(math.log(r, k))
pwrs = []
base_k = k ** base_l
for i in range(base_l, base_r + 1):
	if r >= base_k >= l:
		pwrs.append(base_k)
	base_k *= k
if len(pwrs) == 0:
	print(-1)
else:
	for p in pwrs:
		print(p, end=' ')
