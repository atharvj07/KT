#include <bits/stdc++.h>
using namespace std;
int n, arr[105], ans;
string s;
char k;
int main() {
  int n;
  cin >> n;
  cin >> s;
  for (long long i = 1; i <= n; i++) {
    if (n % i == 0) {
      for (int j = 0; j < i / 2; j++) {
        k = s[i - 1 - j];
        s[i - 1 - j] = s[j];
        s[j] = k;
      }
    }
  }
  cout << s << endl;
}
