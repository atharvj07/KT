import math
from collections import defaultdict
T = int(input())

def dfs(edges, node):
	nodes = {node}
	res = {node: 0}
	visited = {node}
	d = 0
	while nodes:
		d += 1
		next_nodes = set()
		for node in nodes:
			for edge in edges[node]:
				if edge in visited:
					continue
				visited.add(edge)
				next_nodes.add(edge)
				res[edge] = d
		nodes = next_nodes
	return res
for t in range(T):
	(N, M, A, B, C) = [int(i) for i in input().split()]
	P = sorted([int(i) for i in input().split()])
	pp = [0]
	for p in P:
		pp.append(pp[-1] + p)
	edges = defaultdict(list)
	for m in range(M):
		(u, v) = [int(i) for i in input().split()]
		edges[u].append(v)
		edges[v].append(u)
	dfsA = dfs(edges, A)
	dfsB = dfs(edges, B)
	dfsC = dfs(edges, C)
	answer = math.inf
	for m in range(1, N + 1):
		if dfsA[m] + dfsB[m] + dfsC[m] < len(pp):
			answer = min(answer, pp[dfsA[m] + dfsB[m] + dfsC[m]] + pp[dfsB[m]])
	print(answer)
