#include <bits/stdc++.h>
using namespace std;
const long long maxn = 27;
int main() {
  int t;
  long long i, j, k, l, m, n, a, b;
  cin >> n;
  vector<pair<int, int>> pr;
  int y = 500000;
  int x = 0;
  pr.push_back({x, y});
  pr.push_back({x + 1, y});
  pr.push_back({x, y - 1});
  pr.push_back({x + 1, y - 1});
  x++, y--;
  while (n--) {
    pr.push_back({x + 1, y});
    pr.push_back({x, y - 1});
    pr.push_back({x + 1, y - 1});
    x++, y--;
  }
  cout << pr.size() << "\n";
  for (auto it : pr) {
    cout << it.first << " " << it.second << "\n";
  }
}
