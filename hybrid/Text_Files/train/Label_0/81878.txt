t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	for i in range(k - 3):
		print(1, end=' ')
	n = n - (k - 3)
	if n == 0:
		print()
		continue
	if n % 2:
		print(n // 2, n // 2, 1)
	elif n % 4:
		print(n // 2 - 1, n // 2 - 1, 2)
	else:
		print(n // 2, n // 4, n // 4)
	print()
