import javax.management.Query;
import java.util.*;
import java.io.*;

public class A100187 {
    static int n, m;
    static int[] left, right;

    static boolean goleft(int s, int[][] board) {
        int l = s;
        boolean pre = false;
        for (int i = s; i > 0; i--) {
            if (board[i][0] == 0) {
                continue;
            } else {
                if (right[i] >= l) {
                    pre = false;
                    l = i - 1;
                } else pre = true;
            }
        }
//        System.err.println(s+" "+!pre);
        return !pre;
    }

    static boolean goright(int s, int[][] board) {
        boolean pre = false;
        int l = s;
        for (int i = s; i < board.length; i++) {
            if (board[i][0] == 0) {
                continue;
            } else {
                if (left[i] <= l) {
                    l = i + 1;
                    pre = false;
                } else pre = true;
            }
        }
        return !pre;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        PrintWriter pw = new PrintWriter(System.out);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] board = new int[n + 1][2];
        int f = n;
        int l = 0;
        for (int i = 1; i <= m; i++) {
            int id = sc.nextInt();
            board[id][0] = i;
            f = Math.min(f, id);
            l = Math.max(l, id);
            board[id][1] = sc.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            int v = sc.nextInt();
            if (v != 0) {
                board[i][1] = v;
            }
        }
        left = new int[n + 1];
        right = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (board[i][0] == 0)
                continue;
            for (int j = i; j <= n; j++) {
                long v = board[i][1];
                boolean valid = true;
                for (int k = i; k <= j; k++) {
                    if (board[k][0] == 0)
                        v += board[k][1];
                    if (v < 0)
                        valid = false;
                }
                if (valid)
                    right[i] = j;
            }
        }
        for (int i = 1; i <= n; i++) {
            if (board[i][0] == 0)
                continue;
            for (int j = i; j > 0; j--) {
                long v = board[i][1];
                boolean valid = true;
                for (int k = i; k >= j; k--) {
                    if (board[k][0] == 0)
                        v += board[k][1];
                    if (v < 0)
                        valid = false;
                }
                if (valid)
                    left[i] = j;
            }
        }
//        System.err.println(Arrays.toString(left));
//        System.err.println(Arrays.toString(right));

//        System.err.println(Arrays.deepToString(board));

        int sol = -1;
        boolean leftt = true;
        for (int i = 1; i <= n; i++) {

            if (i >= f && goleft(i, board) && goright(i + 1, board)) {
                sol = i;
                leftt = true;
                break;
            }
            if (i <= l && goleft(i - 1, board) && goright(i, board)) {
                sol = i;
                leftt = false;
                break;
            }

        }
        if (sol != -1) {
            pw.println(sol);
            if (leftt) {
                Queue<Integer> q = new LinkedList<>();
                long v = 0;
                l = sol;
                for (int i = sol; i > 0; i--) {
                    if (board[i][0] == 0) {
                        continue;
                    } else {
                        if (right[i] >= l) {
                            l = i - 1;
                            pw.print(board[i][0] + " ");
                            while (!q.isEmpty())
                                pw.print(q.poll() + " ");
                            v = 0;
                        } else {
                            ((LinkedList<Integer>) q).add(board[i][0]);
                        }
                    }
                }
                q = new LinkedList<>();
                v = 0;
                l = sol + 1;
                for (int i = sol + 1; i <= n; i++) {
                    if (board[i][0] == 0) {
                        continue;
                    } else {
                        if (left[i] <= l) {
                            l = i + 1;
                            pw.print(board[i][0] + " ");
                            while (!q.isEmpty())
                                pw.print(q.poll() + " ");
                            v = 0;
                        } else {
                            ((LinkedList<Integer>) q).add(board[i][0]);
                        }
                    }
                }
            } else {
                Queue<Integer> q = new LinkedList<>();
                long v = 0;
                l = sol;
                for (int i = sol; i <= n; i++) {
                    if (board[i][0] == 0) {
                        continue;
                    } else {
                        if (left[i] <= l) {
                            l = i + 1;
                            pw.print(board[i][0] + " ");
                            while (!q.isEmpty())
                                pw.print(q.poll() + " ");
                            v = 0;
                        } else {
                            ((LinkedList<Integer>) q).add(board[i][0]);
                        }
                    }
                }
                q = new LinkedList<>();
                v = 0;
                l = sol - 1;
                for (int i = sol - 1; i > 0; i--) {
                    if (board[i][0] == 0) {
                        continue;
                    } else {
                        if (right[i] >= l) {
                            l = i - 1;
                            pw.print(board[i][0] + " ");
                            while (!q.isEmpty())
                                pw.print(q.poll() + " ");
                            v = 0;
                        } else {
                            ((LinkedList<Integer>) q).add(board[i][0]);
                        }
                    }
                }
            }
        } else {
            pw.println(-1);
        }
        pw.flush();
    }

    static class Scanner {
        BufferedReader br;
        StringTokenizer st;

        public Scanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }

        public Scanner(FileReader f) {
            br = new BufferedReader(f);
        }

        public String next() throws IOException {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        public double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }

        public int[] nextIntArr(int n) throws IOException {
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = Integer.parseInt(next());
            }
            return arr;
        }

    }
}
