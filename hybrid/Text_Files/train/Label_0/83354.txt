import bisect
(l, c) = ([], '0123456789')
for i in c:
	for j in c:
		z = [i, j]
		for k in z:
			z += [k + i, k + j]
			if len(k) > 10:
				break
		l += z
(l, x, y) = ({*l}, [], [])
for i in l:
	if len({*i}) == 1:
		x += [i]
	else:
		y += [i]
(x, y) = (sorted(map(int, x)), sorted(map(int, y)))
for _ in [*open(0)][1:]:
	(n, k) = map(int, _.split())
	print(n if len({*str(n)}) <= k else x[bisect.bisect(x, n)] if k == 1 else y[bisect.bisect(y, n)])
