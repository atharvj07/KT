Mod = 10 ** 9 + 7
n = int(input())
s = input()
(cnt, a, ab, abc) = (1, 0, 0, 0)
for i in range(len(s)):
	if s[i] == 'a':
		a += cnt
		a %= Mod
	elif s[i] == 'b':
		ab += a
		ab %= Mod
	elif s[i] == 'c':
		abc += ab
		abc %= Mod
	else:
		abc *= 3
		abc += ab
		abc %= Mod
		ab *= 3
		ab += a
		ab %= Mod
		a *= 3
		a += cnt
		a %= Mod
		cnt *= 3
		cnt %= Mod
print(abc)
