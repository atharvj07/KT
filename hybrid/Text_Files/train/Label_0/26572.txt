#include <bits/stdc++.h>
using namespace std;
int a[100005];
int aa[100005];
int num[105];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> num[i];
  for (int i = n; i > 1; i--)
    for (int j = 1; j < i; j++) {
      if (num[j] > num[j + 1]) {
        swap(num[j], num[j + 1]);
        cout << j << " " << j + 1 << endl;
      }
    }
  return 0;
}
