def solve(n, a, b):
	list_of_indexes = [0] * (2 * n + 1)
	for (i, el) in enumerate(a):
		list_of_indexes[el] = i
	for (i, el) in enumerate(b):
		list_of_indexes[el] = i
	best_choice_from_b = 2 * n
	answer = 2 * n
	for i in range(2 * n - 1, 0, -2):
		best_choice_from_b = min(best_choice_from_b, list_of_indexes[i + 1])
		answer = min(answer, list_of_indexes[i] + best_choice_from_b)
	return answer
test_count = int(input())
output = list()
for i in range(test_count):
	n = int(input())
	a = input().split(' ')
	a = list(map(int, a))
	b = input().split(' ')
	b = list(map(int, b))
	output.append(solve(n, a, b))
for i in output:
	print(i)
