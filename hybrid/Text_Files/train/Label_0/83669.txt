(a1, b1, a2, b2) = [int(j) for j in input().split()]
(x1, y1, x2, y2) = [int(j) for j in input().split()]
(l1, m1, l2, m2) = [int(j) for j in input().split()]
tot = (a2 - a1) * (b2 - b1)

def overlap(x1, y1, x2, y2, a1, b1, a2, b2):
	return max(min(y2, b2) - max(y1, b1), 0) * max(min(x2, a2) - max(x1, a1), 0)

def overlap_sp(x1, y1, x2, y2, l1, m1, l2, m2, a1, b1, a2, b2):
	return max(min(y2, b2, m2) - max(y1, b1, m1), 0) * max(min(x2, a2, l2) - max(x1, a1, l1), 0)
ov1 = overlap(x1, y1, x2, y2, a1, b1, a2, b2)
ov2 = overlap(l1, m1, l2, m2, a1, b1, a2, b2)
ov3 = overlap_sp(x1, y1, x2, y2, l1, m1, l2, m2, a1, b1, a2, b2)
ov4 = overlap_sp(l1, m1, l2, m2, x1, y1, x2, y2, a1, b1, a2, b2)
tot -= ov1 + ov2
tot += max(ov3, ov4)
if tot == 0:
	print('NO')
else:
	print('YES')
