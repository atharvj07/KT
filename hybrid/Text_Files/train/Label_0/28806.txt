struct UnionFind {
	vector<int> data;
	UnionFind(int size) : data(size, -1) { }
	bool unionSet(int x, int y) {
		x = root(x); y = root(y);
		if (x != y) {
			if (data[y] < data[x]) swap(x, y);
			data[x] += data[y]; data[y] = x;
		}
		return x != y;
	}
	int root(int x) {
		return data[x] < 0 ? x : data[x] = root(data[x]);
	}
	int size(int x) {
		return -data[root(x)];
	}
};
void add(int a, int b, vector<vector<int>>&field) {
	field[a][b]=true;
	field[b][a]=true;
}
void mk_perfect(const vector<int>&v, vector<vector<int>>&field) {
	for (int i = 0; i < v.size(); ++i) {
		for (int j = 0; j < v.size(); ++j) {
			if (i != j) {
				add(v[i],v[j],field);
			}
		}
	}
}
int dfs(const int now, const vector<vector<int>>&field, vector<vector<int>>&add_field, vector<int>&oks,vector<int>&comes) {
	if (oks[now]) {
		oks[now]=false;
		return true;
	}
	else {
		comes[now]=true;
		for (int i = 0; i < field[now].size(); ++i) {
			if (field[now][i] == 0) {
				if (!comes[i]) {
					bool ok = dfs(i, field, add_field, oks, comes);
					if (ok) {
						add_field[now][i] ^=1;
						add_field[i][now]^=1;
						return true;
					}
				}
			}
		}
	}
	return false;
}

bool check(const vector<vector<int>>field) {
	UnionFind uf(field.size());
	int N=field.size();
	vector<int>jisuu(N);
	for (int i = 0; i < N; ++i) {
		for (int j = 0; j < N; ++j) {
			if (i != j) {
				if (field[i][j]) {
					jisuu[i]++;
					uf.unionSet(i,j);
				}
			}
		}
	}
	bool ok1=uf.size(0)==N;
	bool ok2 = all_of(jisuu.begin(), jisuu.end(), [](const int a) {
		return a%2==0;
	});
	assert(ok1&&ok2);
	return true;
}
