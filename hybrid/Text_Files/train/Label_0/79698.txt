def bs(l, r, tar):
	if l > r:
		return -1
	mid = l + (r - l) // 2
	if b[mid] >= tar and hi[mid - 1] == 1:
		return mid
	if b[mid + 1] >= tar and b[mid] < tar:
		return mid + 1
	if b[mid] >= tar and b[mid - 1] < tar:
		return mid
	if b[mid] > tar:
		return bs(l, mid - 1, tar)
	return bs(mid + 1, r, tar)
