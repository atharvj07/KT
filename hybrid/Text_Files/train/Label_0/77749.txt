def readline():
	return map(int, input().split())

def solve():
	n = int(input())
	a = sorted(readline())
	b = sorted(readline())
	q = n // 4
	sa = sum(a[q:])
	sb = sum(b[q:])
	k = 0
	while sa < sb:
		k += 1
		sa += 100
		if (n + k) % 4 == 0:
			sa -= a[q] if q < n else 100
			q += 1
		else:
			sb += b[q - k] if q >= k else 0
	print(k)

def main():
	t = int(input())
	for __ in range(t):
		solve()
main()
