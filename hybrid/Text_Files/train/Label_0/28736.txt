#include <bits/stdc++.h>
using namespace std;
const int N = 100009;
const long long oo = 1000;
bool notPrime[N];
vector<int> Prime;
int n;
int mark[N];
vector<int> cnt[N];
void Init() {
  notPrime[1] = 1;
  for (int i = 2; i <= sqrt(n); i++)
    if (!notPrime[i]) {
      for (int j = 2; j * i <= n; j++) notPrime[i * j] = 1;
    }
  for (int i = 1; i <= n; i++)
    if (!notPrime[i]) Prime.push_back(i);
  for (int i = 1; i <= n; i++) {
    int j = 0;
    int x = i;
    while (x != 1) {
      if (!notPrime[x]) {
        mark[i] = x;
        x = 1;
      }
      if (x % Prime[j] == 0) {
        mark[i] = Prime[j];
        while (x % Prime[j] == 0) x /= Prime[j];
      }
      j++;
    }
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  Init();
  int ans = 0, even = 0;
  for (int i = 1; i <= n; i++) cnt[mark[i]].push_back(i);
  vector<pair<int, int> > result;
  for (int i = n; i >= 2; i--) {
    if (cnt[i].size() <= 1) continue;
    if (cnt[i].size() & 1) {
      result.push_back(make_pair(cnt[i][0], cnt[i][2]));
      for (int j = 3; j < cnt[i].size(); j += 2) {
        result.push_back(make_pair(cnt[i][j], cnt[i][j + 1]));
      }
      cnt[2].push_back(cnt[i][1]);
    } else {
      for (int j = 0; j < cnt[i].size(); j += 2) {
        result.push_back(make_pair(cnt[i][j], cnt[i][j + 1]));
      }
    }
  }
  cout << result.size() << endl;
  for (int i = 0; i < result.size(); i++)
    cout << result[i].first << ' ' << result[i].second << endl;
}
