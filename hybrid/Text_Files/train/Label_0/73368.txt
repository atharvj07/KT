n = int(input())
a = list(map(int, input().split()))
ans = []
s = set([a[0]])
was = set()
lst = 0
for q in range(1, n):
	x = a[q]
	if x < 0:
		d = x * -1
		if d not in s:
			print(-1)
			exit()
		else:
			s -= set([d])
		if len(s) == 0:
			ans.append(q + 1 - lst)
			was = set()
			lst = q + 1
	elif x in s or x in was:
		print(-1)
		exit()
	else:
		was.add(x)
		s.add(x)
if len(s) != 0:
	print(-1)
	exit()
print(len(ans))
print(*ans)
