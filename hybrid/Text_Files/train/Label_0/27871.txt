#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  int k1, k2, a;
  cin >> k1;
  deque<int> q1, q2;
  for (int i = 0; i < k1; i++) {
    cin >> a;
    q1.push_back(a);
  }
  cin >> k2;
  for (int i = 0; i < k2; i++) {
    cin >> a;
    q2.push_back(a);
  }
  int ans = 0, i = 1000;
  deque<int>::iterator it1 = q1.begin(), it2 = q2.begin();
  while (i--) {
    if (*it1 > *it2) {
      q1.push_back(*it2);
      q1.push_back(*it1);
      q1.pop_front();
      q2.pop_front();
      it1 = q1.begin();
      it2 = q2.begin();
      ;
    } else {
      q2.push_back(*it1);
      q2.push_back(*it2);
      q2.pop_front();
      q1.pop_front();
      it1 = q1.begin();
      it2 = q2.begin();
      ;
    }
    ans++;
    if (q1.empty() || q2.empty()) {
      break;
    }
  }
  if (q1.empty()) {
    cout << ans << " 2" << endl;
  } else {
    if (q2.empty()) {
      cout << ans << " 1" << endl;
    } else {
      cout << "-1" << endl;
    }
  }
  return 0;
}
