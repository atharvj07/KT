#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  int i;
  for (i = 1; i <= n; i++) {
    cin >> a[i];
  }
  int sum = abs(a[2] - a[1]), c = 0, temp;
  int x = a[2] - a[1];
  for (i = 1; i <= n; i++) {
    if ((abs(a[i] - a[i + 1])) == sum) {
      c++;
    }
  }
  if (c == n - 1) {
    if (0 <= a[n]) {
      temp = a[n] + x;
      cout << temp;
    } else {
      temp = a[n] + x;
      cout << temp;
    }
  } else {
    cout << a[n];
  }
}
