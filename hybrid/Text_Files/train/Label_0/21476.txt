#include <bits/stdc++.h>
const int N = 100100;
std::vector<int> G[N], comm;
int n, m, h, t;
int app[N], atop;
bool com[N];
bool ok(int u, int v) {
  atop++;
  for (int j = G[u].size() - 1; j >= 0; j--)
    if (G[u][j] != v) app[G[u][j]] = atop;
  int cnt = 0;
  for (int j = G[v].size() - 1; j >= 0; j--)
    if (G[v][j] != u && app[G[v][j]] == atop) cnt++;
  if (G[u].size() + G[v].size() - cnt < h + t + 2) return false;
  return true;
}
void print(int u, int v) {
  atop++;
  printf("YES\n%d %d\n", u, v);
  for (int j = G[v].size() - 1; j >= 0; j--)
    if (G[v][j] != u) app[G[v][j]] = atop;
  int cnt = 0;
  for (int j = G[u].size() - 1; j >= 0; j--)
    if (G[u][j] != v && app[G[u][j]] == atop)
      comm.push_back(G[u][j]), com[G[u][j]] = true;
  for (int j = G[u].size() - 1; j >= 0; j--)
    if (G[u][j] != v && !com[G[u][j]]) {
      cnt++;
      printf("%d", G[u][j]);
      if (cnt == h) {
        puts("");
        break;
      } else
        putchar(' ');
    }
  int now = 0;
  while (cnt < h) {
    cnt++;
    printf("%d", comm[now++]);
    if (cnt == h) {
      puts("");
      break;
    } else
      putchar(' ');
  }
  cnt = 0;
  for (int j = G[v].size() - 1; j >= 0; j--)
    if (G[v][j] != u && !com[G[v][j]]) {
      cnt++;
      printf("%d", G[v][j]);
      if (cnt == t) {
        puts("");
        break;
      } else
        putchar(' ');
    }
  while (cnt < t) {
    cnt++;
    printf("%d", comm[now++]);
    if (cnt == t) {
      puts("");
      break;
    } else
      putchar(' ');
  }
  scanf(" ");
}
bool chk(int u, int v) {
  if (G[u].size() <= h) return false;
  if (G[v].size() <= t) return false;
  if (ok(u, v)) {
    print(u, v);
    return true;
  }
  return false;
}
int main() {
  scanf("%d%d%d%d", &n, &m, &h, &t);
  for (int i = 0; i < m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    G[x].push_back(y);
    G[y].push_back(x);
  }
  for (int i = 1; i <= n; i++)
    for (int j = G[i].size() - 1; j >= 0; j--)
      if (chk(i, G[i][j])) return 0;
  puts("NO");
}
