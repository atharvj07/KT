for _ in range(int(input())):
	N = int(input())
	lis = list(map(int, input().split()))
	dic = {}
	output = 0
	for i in range(N):
		dic[lis[i]] = dic.get(lis[i], 0) + 1
		if dic[lis[i]] == 2:
			if lis[i] != lis[i - 1]:
				output = 1
				break
		if dic[lis[i]] == 3:
			output = 1
			break
	if output == 1:
		print('YES')
	else:
		print('NO')
