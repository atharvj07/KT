#include <bits/stdc++.h>
using namespace std;
const int N = 100050;
char str[N];
int n, dp[2][26];
int fbd[26][26];
inline void update(int &x, int y) {
  if (y > x) x = y;
}
int main() {
  scanf("%s", str);
  n = strlen(str);
  for (int i = 0; i < n; ++i) str[i] -= 'a';
  str[n] = -1;
  int k, res(0);
  for (scanf("%d", &k); k--;) {
    char st[4];
    scanf("%s", st);
    st[0] -= 'a';
    st[1] -= 'a';
    int c[2] = {0, 0};
    for (int i = 0; i <= n; ++i)
      if (str[i] != st[0] && str[i] != st[1]) {
        res += min(c[0], c[1]);
        c[0] = c[1] = 0;
      } else
        ++c[str[i] == st[1]];
  }
  printf("%d\n", res);
  return 0;
}
