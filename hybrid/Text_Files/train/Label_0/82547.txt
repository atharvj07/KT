def solve(answers):
	l = 0
	r = int(input()) - 1
	numbers = [int(x) for x in input().split()]
	marginals = []
	while l < r:
		if numbers[l] == numbers[r]:
			l += 1
			r -= 1
		else:
			marginals += [numbers[l], numbers[r]]
			break
	marginals.append(-1)
	for x in marginals:
		tmp = [y for y in numbers if y != x]
		if tmp == list(reversed(tmp)):
			answers.append('YES')
			return
	answers.append('NO')
sets_num = int(input())
answers = []
for i in range(sets_num):
	solve(answers)
for i in answers:
	print(i)
