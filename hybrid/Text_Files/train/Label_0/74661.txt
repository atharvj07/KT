from bisect import bisect_left

def get_len(a):
	i = 0
	ret = []
	while i < len(a):
		if a[i] == 0:
			i += 1
		else:
			j = i
			while j < len(a) and a[j] == 1:
				j += 1
			ret.append(j - i)
			i = j
	return sorted(ret)

def get_sum(a):
	ret = [0]
	s = 0
	for v in a:
		s += v
		ret.append(s)
	return ret

def count_len(i, a, sa):
	p = bisect_left(a, i)
	n = len(a)
	if p == n:
		return 0
	return sa[n] - sa[p] - (i - 1) * (n - p)

def solve(n, m, k, a, b):
	(a, b) = (get_len(a), get_len(b))
	(sa, sb) = (get_sum(a), get_sum(b))
	ret = 0
	size = []
	for i in range(1, int(k ** 0.5) + 1):
		if k % i == 0:
			size += [i] if k // i == i else [i, k // i]
	for i in size:
		j = k // i
		ret += count_len(i, a, sa) * count_len(j, b, sb)
	return ret

def main():
	inp = lambda : [int(x) for x in input().split()]
	(n, m, k) = inp()
	(a, b) = (inp(), inp())
	print(solve(n, m, k, a, b))
main()
