import sys
input = sys.stdin.readline
n = int(input())
seq = input()
cat = ['.', '.']
count = 1
for i in range(n):
	letter = seq[i]
	if letter == 'R' and cat[0] == 'L':
		count += 1
		cat = ['R', '.']
	elif letter == 'L' and cat[0] == 'R':
		count += 1
		cat = ['L', '.']
	elif letter == 'U' and cat[1] == 'D':
		count += 1
		cat = ['.', 'U']
	elif letter == 'D' and cat[1] == 'U':
		count += 1
		cat = ['.', 'D']
	elif letter == 'R' or letter == 'L':
		cat[0] = letter
	elif letter == 'U' or letter == 'D':
		cat[1] = letter
print(count)
