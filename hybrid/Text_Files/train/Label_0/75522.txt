from math import ceil
s = input()
n = int(input())
count = {}
for i in s:
	if i in count:
		count[i] += 1
	else:
		count[i] = 1
if len(count) > n:
	print(-1)
else:
	if len(s) < n:
		print(1)
		ret = s
	else:
		(l, h) = (len(s) // n, len(s))
		while l < h:
			m = (l + h) // 2
			tot = 0
			for i in count:
				tot += ceil(count[i] / m)
			if tot > n:
				l = m + 1
			else:
				h = m
		print(l)
		ret = ''
		for i in count:
			for j in range(ceil(count[i] / l)):
				ret += i
	for i in range(n - len(ret)):
		ret += s[0]
	print(ret)
