#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
const int N = 3e5 + 5;
const int mx = 1.5e7;
long long gcd(long long a, long long b) { return b ? gcd(b, a % b) : a; }
long long pwr(long long x, long long y);
bool isprime(long long x);
int factor[mx], numdiv[mx], a[N];
void s() {
  for (long long i = 2; i * i <= mx; ++i) {
    if (!factor[i]) {
      for (long long j = 2 * i; j <= mx; j += i) {
        factor[j] = i;
      }
    }
  }
  for (int i = 1; i < mx; ++i) {
    if (!factor[i]) {
      factor[i] = i;
    }
  }
  return;
}
int lgcdseq(int n) {
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    int element = a[i];
    while (element > 1) {
      int d = factor[element];
      ++numdiv[d];
      ans = max(ans, numdiv[d]);
      while (element % d == 0) {
        element /= d;
      }
    }
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  s();
  int n, p = 0;
  cin >> n;
  for (int i = 1; i < n + 1; i++) {
    cin >> a[i];
    p = gcd(p, a[i]);
  }
  for (int i = 1; i < n + 1; i++) {
    a[i] /= p;
  }
  int t = lgcdseq(n);
  if (t == 0) {
    cout << -1 << endl;
    return 0;
  }
  cout << (n - t) << endl;
  return 0;
}
long long pwr(long long x, long long y) {
  long long ans = 1;
  x = x % mod;
  while (y > 0) {
    if (y & 1) ans = (x * ans) % mod;
    x = (x * x) % mod;
    y = y / 2;
  }
  return ans;
}
bool isprime(long long x) {
  if (x < 2) return false;
  if (x < 4) return true;
  if (x % 2 == 0 || x % 3 == 0) return false;
  for (long long i = 5; i * i <= x; i += 6) {
    if (x % i == 0 || x % (i + 2) == 0) return false;
  }
  return true;
}
