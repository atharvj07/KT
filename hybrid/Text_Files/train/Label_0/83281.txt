from sys import setcheckinterval, stdin, setrecursionlimit
setcheckinterval(1000)
setrecursionlimit(10 ** 7)

def iin():
	return int(stdin.readline())

def lin():
	return list(map(int, stdin.readline().split()))
(a, b) = lin()
A = a
ch = 1
ch1 = 1
while ch1 <= a + b:
	ch += 1
	ch1 += ch
a1 = []
a2 = []
for i in range(ch - 1, 0, -1):
	if a - i >= 0:
		a1.append(i)
		a -= i
	else:
		a2.append(i)
print(len(a1))
print(*a1)
print(len(a2))
print(*a2)
