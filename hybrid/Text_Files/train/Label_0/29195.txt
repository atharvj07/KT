#include <bits/stdc++.h>
using namespace std;
string sss = "0123456789ABCDEF";
long long grid[200][200], grid1[200][200];
vector<long long> rwvec, clmvec, rw1, col1;
void solve(long long CS) {
  long long n, m, i, j, mn = 1 << 30;
  cin >> n >> m;
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      cin >> grid[i][j];
      grid1[i][j] = grid[i][j];
    }
  }
  for (i = 1; i <= n; i++) {
    mn = 1 << 30;
    for (j = 1; j <= m; j++) {
      mn = min(mn, grid[i][j]);
    }
    for (j = 1; j <= m; j++) grid[i][j] -= mn;
    while (mn--) {
      clmvec.push_back(i);
    }
  }
  for (j = 1; j <= m; j++) {
    mn = 1 << 30;
    for (i = 1; i <= n; i++) {
      mn = min(mn, grid[i][j]);
    }
    for (i = 1; i <= n; i++) grid[i][j] -= mn;
    while (mn--) rwvec.push_back(j);
  }
  for (i = 1; i <= n; i++) {
    for (j = 1; j <= m; j++) {
      if (grid[i][j]) {
        cout << -1;
        return;
      }
    }
  }
  for (j = 1; j <= m; j++) {
    mn = 1 << 30;
    for (i = 1; i <= n; i++) {
      mn = min(mn, grid1[i][j]);
    }
    for (i = 1; i <= n; i++) grid1[i][j] -= mn;
    while (mn--) rw1.push_back(j);
  }
  for (i = 1; i <= n; i++) {
    mn = 1 << 30;
    for (j = 1; j <= m; j++) {
      mn = min(mn, grid1[i][j]);
    }
    for (j = 1; j <= m; j++) grid1[i][j] -= mn;
    while (mn--) col1.push_back(i);
  }
  if (clmvec.size() + rwvec.size() < col1.size() + rw1.size()) {
    cout << clmvec.size() + rwvec.size() << endl;
    for (i = 0; i < clmvec.size(); i++) cout << "row " << clmvec[i] << endl;
    for (i = 0; i < rwvec.size(); i++) cout << "col " << rwvec[i] << endl;
  } else {
    cout << col1.size() + rw1.size() << endl;
    for (i = 0; i < col1.size(); i++) cout << "row " << col1[i] << endl;
    for (i = 0; i < rw1.size(); i++) cout << "col " << rw1[i] << endl;
  }
}
int main() {
  long long t, no = 0;
  t = 1;
  while (t--) solve(++no);
}
