n = int(input())
coordinates = [tuple(map(int, input().split())) for i in range(n)]
if n == 4:
	coordinates.sort()
	s = int(((coordinates[0][1] - coordinates[1][1]) ** 2 + (coordinates[0][0] - coordinates[1][0]) ** 2) ** 0.5 * ((coordinates[1][1] - coordinates[3][1]) ** 2 + (coordinates[1][0] - coordinates[3][0]) ** 2) ** 0.5)
elif n == 3:
	if coordinates[0][0] == coordinates[1][0] and coordinates[0][1] == coordinates[2][1] or (coordinates[0][1] == coordinates[1][1] and coordinates[0][0] == coordinates[2][0]):
		coordinates = [coordinates[1], coordinates[0], coordinates[2]]
	elif coordinates[2][0] == coordinates[1][0] and coordinates[2][1] == coordinates[0][1] or (coordinates[2][1] == coordinates[1][1] and coordinates[2][0] == coordinates[0][0]):
		coordinates = [coordinates[0], coordinates[2], coordinates[1]]
	s = int(((coordinates[0][1] - coordinates[1][1]) ** 2 + (coordinates[0][0] - coordinates[1][0]) ** 2) ** 0.5 * ((coordinates[1][1] - coordinates[2][1]) ** 2 + (coordinates[1][0] - coordinates[2][0]) ** 2) ** 0.5)
elif n == 2 and coordinates[0][0] != coordinates[1][0] and (coordinates[0][1] != coordinates[1][1]):
	s = abs((coordinates[0][0] - coordinates[1][0]) * (coordinates[0][1] - coordinates[1][1]))
else:
	s = -1
print(s)
