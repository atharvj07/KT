using namespace std;

#define MAX_SIZE 400000

long long segMax[MAX_SIZE - 1], segAdd[MAX_SIZE - 1];


int n;

//区間[a, b)に値xを加算する.
void add(int a, int b, long long x, int k = 0, int l = 0, int r = n)
{
	if (r <= a || b <= l) return;
	
	if (a <= l && r <= b){
		segAdd[k] += x;
		while (k){
			k = (k - 1) / 2;
			segMax[k] = max(segMax[k * 2 + 1] + segAdd[k * 2 + 1], segMax[k * 2 + 2] + segAdd[k * 2 + 2]);
		}
		return;
	}
	
	add(a, b, x, k * 2 + 1, l, (l + r) / 2);
	add(a, b, x, k * 2 + 2, (l + r) / 2, r);
}

long long getMax(int a, int b, int k = 0, int l = 0, int r = n)
{
	if (r <= a || b <= l) return (LLONG_MIN);
	
	if (a <= l && r <= b) return (segMax[k] + segAdd[k]);
	
	long long left = getMax(a, b, k * 2 + 1, l, (l + r) / 2);
	long long right = getMax(a, b, k * 2 + 2, (l + r) / 2, r);
	
	return (max(left, right) + segAdd[k]);
	
}


int a[100000];
void init(){
	memset(segMax,0,sizeof(segMax));
	memset(segAdd,0,sizeof(segAdd));
	rep(i,n)add(i,i+1,a[i]);
}


long long m;
int c;

long long calc(int r){
	long long s=0;
	for(int i=1;i<n;i++){
		add(i,min(i+r+1,n),-c);
	}
	//cout<<'r'<<r<<endl;
	rep(i,n){
		//rep(i,n)cout<<getMax(i,i+1)<<' ';cout<<endl;
		s+=getMax(max(0,i-r),min(n,i+r+1));
		add(0,i+1,-c);
		add(i+1,min(n,i+r+2),c);
	}
	return s;
}


int main(){
	cin>>n>>m>>c;
	rep(i,n)cin>>a[i];
	int l=-1,r=n;
	while(r-l>1){
		int mid=(l+r)/2;
		init();
		if(calc(mid)<m)l=mid;
		else r=mid;
	}
	if(r==n)r=-1;
	cout<<r<<endl;
	return 0;
}
