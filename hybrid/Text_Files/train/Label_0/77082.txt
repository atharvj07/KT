(n, k) = list(map(int, input().split()))
d = list(map(int, input().split()))
ans = 0
l = []
for i in range(n):
	if d[i] % k == 0:
		ans = ans + 1
	else:
		l.append(d[i] % k)
ans = ans // 2
l.sort()
i = 0
j = len(l) - 1
while i < j:
	if k - l[i] == l[j]:
		i = i + 1
		j = j - 1
		ans = ans + 1
	elif k - l[i] < l[j]:
		j = j - 1
	elif k - l[i] > l[j]:
		i = i + 1
print(ans * 2)
