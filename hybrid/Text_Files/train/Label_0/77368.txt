import sys, random
raw_input = sys.stdin.readline

def slove(a, b):
	n = len(a)

	def reverse_pair(a, b):
		ret = 0
		for i in range(n):
			if a[i] != b[i]:
				ret += 1
		return ret

	def countOdd(a, b):

		def lit(s, i):
			ret = list(map(int, list(s)))
			for j in range(n):
				if j != i:
					ret[j] ^= 1
			return ''.join(map(str, ret))
		ret = 1000000
		cases = [0, 0]
		for i in range(n):
			if sum(cases) == 2:
				break
			if a[i] == '1':
				if b[i] == '0':
					cases[0] = 1
					ret = min(ret, 1 + reverse_pair(lit(a, i), b))
				else:
					cases[1] = 1
					ret = min(ret, 1 + reverse_pair(lit(a, i), b))
		return ret
	res = 10000000
	count_a = a.count('1')
	count_b = b.count('1')
	if count_a == count_b:
		res = reverse_pair(a, b)
	if n - count_a + 1 == count_b:
		res = min(res, countOdd(a, b))
	return res if res != 10000000 else -1
for _ in range(int(raw_input())):
	raw_input()
	a = raw_input().strip()
	b = raw_input().strip()
	print(slove(a, b))
