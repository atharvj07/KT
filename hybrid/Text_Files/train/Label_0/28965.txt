#include <bits/stdc++.h>
using namespace std;
const int inf = 2147483647, dx[] = {-1, 0, 1, 0}, dy[] = {0, -1, 0, 1};
const int N = 100005, M = 1000005, K = 200005, mod = 1000000007;
const long long llinf = 9223372036854775807ll;
long long n, m, s, ans, ar;
long long f(long long a, long long b) { return (n - a + 1) * (m - b + 1); }
int main() {
  cin >> n >> m >> s;
  for (int i = 1; i <= n; i += 2)
    for (int j = 1; j <= m; j += 2) {
      if (i * j == s)
        ans += f(i, j) * (((i + 1) * (j + 1) / 2) - 1);
      else if (i * j > s) {
        ar = (i * j - s);
        if (ar % 4 != 0) continue;
        ar /= 4;
        for (int l = 1; l <= (i - 1) / 2; l++)
          if ((ar % l) == 0 && (ar / l) <= (j - 1) / 2) ans += f(i, j) * 2;
      }
    }
  cout << ans;
  return 0;
}
