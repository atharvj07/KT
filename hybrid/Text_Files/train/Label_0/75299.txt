a = int(input())
n = a
z = list(map(int, input().split()))
for i in range(1, len(z)):
	z[i] = z[i - 1] + z[i]
a = a * (a + 1) // 2
valu = a - sum(z)
p1 = valu // n
ans = []
if p1 <= 0 or p1 > n:
	print(-1)
	exit()
from collections import defaultdict
al = defaultdict(int)
for i in range(-1, len(z)):
	if i == -1:
		ans.append(p1)
		al[p1] = 1
	else:
		if al[z[i] + p1] or z[i] + p1 <= 0:
			print(-1)
			exit()
		ans.append(z[i] + p1)
		al[ans[-1]] = 1
print(*ans)
