(a, b) = map(int, input().split())
slogan = list(input())
if b - 1 <= a - b:
	print('LEFT\n' * (b - 1), end='')
	for i in range(a - 1):
		print(f'PRINT {slogan[i]}\nRIGHT')
	print(f'PRINT {slogan[-1]}')
else:
	print('RIGHT\n' * (a - b), end='')
	for i in range(a - 1):
		print(f'PRINT {slogan[-(i + 1)]}\nLEFT')
	print(f'PRINT {slogan[0]}')
