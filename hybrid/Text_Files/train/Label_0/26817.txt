#include<bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
  ios::sync_with_stdio(false), cin.tie(0);
  int n; cin >> n;
  ll s = 0;
  ll t = 1e18;
  for(int i = 0; i < n; i++) {
    ll a, b;
    cin >> a >> b;
    if(a > b) t = min(t, b);
    s += a;
  }
  cout << max(0ll, s - t) << endl;
}
