import java.util.*;
import java.io.*;
public class a {
	static final FS sc = new FS();
	static final PrintWriter pw = new PrintWriter(System.out);

	static int n, m;
	static ArrayDeque<Integer>[] edges;
	static int[] top;
	static blog[] blogs;
	public static void main(String[] args) {
		n = sc.nextInt();
		m = sc.nextInt();
		edges = new ArrayDeque[n];
		for(int i = 0; i < n; ++i) edges[i] = new ArrayDeque<>();

		for(int i = 0; i < m; ++i) {
			int a = sc.nextInt() - 1, b = sc.nextInt() - 1;
			edges[a].add(b);
			edges[b].add(a);
		}

		top = new int[n];
		blogs = new blog[n];
		for(int i = 0; i < n; ++i) {
			top[i] = sc.nextInt() - 1;
			blogs[i] = new blog(i, top[i]);
		}
		Arrays.sort(blogs);
		
		int[] out = new int[n];
		int idx = 0;
		for(blog curr : blogs) {
			boolean[] got = new boolean[curr.topic + 1];

			for(int next : edges[curr.idx]) {
				int get = top[next];
				if(get < got.length) got[get] = true;
			}
			
			boolean weGood = true;
			// check to make sure we got everything before us
			for(int i = 0; i < curr.topic; ++i) {
				if(!got[i]) weGood = false;
			}
			// check to make sure we haven't already been got
			if(got[curr.topic]) weGood = false;
			
			if(!weGood) {
				System.out.println(-1);
				return;
			}

			// otherwise assume weGood
//			pw.print((curr.idx + 1) + " ");
			out[idx++] = curr.idx + 1;
		}
		
		for(int i = 0; i < n; ++i) pw.print(out[i] + " ");
		
		pw.println();
		pw.flush();
	}
	static class blog implements Comparable<blog>{
		int idx, topic;
		blog(int ii, int tt){
			idx = ii;
			topic = tt;
		}
		public int compareTo(blog in) {
			return Integer.compare(topic, in.topic);
		}
	}
	static class FS {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");
		String next() {
			while(!st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch(Exception e) {}
			}
			return st.nextToken();
		}
		int nextInt() {
			return Integer.parseInt(next());
		}
		long nextLong() {
			return Long.parseLong(next());
		}
	}
}
