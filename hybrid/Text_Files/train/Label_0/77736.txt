import math
(a, b) = map(float, input().split())
n = int(input())
x = []
y = []
v = []
xx = 0.0
yy = 0.0
vv = 0.0
mini = 1000000.0
for i in range(n):
	(xx, yy, vv) = map(float, input().split())
	x.append(xx)
	y.append(yy)
	v.append(vv)
for i in range(n):
	mini = min(mini, math.sqrt((x[i] - a) ** 2 + (y[i] - b) ** 2) / v[i])
print(mini)
