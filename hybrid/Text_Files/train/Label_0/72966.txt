n = int(int(input()))
Coin = list(map(int, input().split()))
Coin[0] = Coin[0] - 1
Coin[1] = Coin[1] - 1
WhiteKing = [0, 0]
BlackKing = [n - 1, n - 1]
WTCX = WhiteKing[0] - Coin[0]
WTCY = WhiteKing[1] - Coin[1]
BTCX = BlackKing[0] - Coin[0]
BTCY = BlackKing[1] - Coin[1]
Wans = max(abs(WTCX), abs(WTCY))
Bans = max(abs(BTCX), abs(BTCY))
if Wans <= Bans:
	print('White')
else:
	print('Black')
