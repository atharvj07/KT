def sub(a, b):
	if a - b < 0:
		return a - b + MOD
	else:
		return a - b
MOD = 10 ** 9 + 7
(n, m) = [int(x) for x in input().split()]
ans = n * m % MOD
k = min(n, m)
d1 = 1
while d1 * d1 <= n:
	if d1 <= k:
		ans = sub(ans, n // d1 * d1 % MOD)
	d1 += 1
d = 1
while d * d <= n:
	r = min(k, n // d)
	l = max(d1, n // (d + 1) + 1)
	if l <= r and l <= k:
		s = (l + r) * (r - l + 1) // 2
		ans = sub(ans, d * s % MOD)
	d += 1
print(ans)
