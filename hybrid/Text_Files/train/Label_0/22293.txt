#include <bits/stdc++.h>
using namespace std;
struct Res {
  int64_t sumDist = 0, sumReducedDepth = 0, sumReducibleDepth = 0,
          countReducedDepth = 1, countReducibleDepth = 0;
};
Res solve(const vector<vector<int>>& edges, int i, int parent) {
  Res res;
  for (int j : edges[i]) {
    if (j != parent) {
      Res res2 = solve(edges, j, i);
      res2.sumReducedDepth += res2.countReducedDepth;
      swap(res2.sumReducedDepth, res2.sumReducibleDepth);
      swap(res2.countReducedDepth, res2.countReducibleDepth);
      res.sumDist += res2.sumDist +
                     (res.sumReducedDepth + res.sumReducibleDepth) *
                         (res2.countReducedDepth + res2.countReducibleDepth) +
                     (res2.sumReducedDepth + res2.sumReducibleDepth) *
                         (res.countReducedDepth + res.countReducibleDepth) -
                     res.countReducibleDepth * res2.countReducibleDepth;
      res.sumReducedDepth += res2.sumReducedDepth;
      res.sumReducibleDepth += res2.sumReducibleDepth;
      res.countReducedDepth += res2.countReducedDepth;
      res.countReducibleDepth += res2.countReducibleDepth;
    }
  }
  return res;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n;
  cin >> n;
  cin.ignore();
  vector<vector<int>> edges(n);
  for (int i = 1; i < n; ++i) {
    int u, v;
    cin >> u >> v;
    cin.ignore();
    --u;
    --v;
    edges[u].push_back(v);
    edges[v].push_back(u);
  }
  cout << solve(edges, 0, -1).sumDist << endl;
  return 0;
}
