import java.util.Arrays;
import java.util.ArrayList;
import java.util.ArrayDeque;
import java.util.Collections;
import java.util.Scanner;

public class a {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int h = in.nextInt(), w = in.nextInt();
        String[] lines = new String[h];
        in.nextLine();
        for (int i = 0; i < h; i++) lines[i] = in.nextLine();
        int[] gp = new int[h], sp = new int[h];
        boolean[] oc = new boolean[w];
        for (int i = 0; i < w; i++) oc[i] = false;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if (lines[i].charAt(j) == 'G') gp[i] = j;
                if (lines[i].charAt(j) == 'S') sp[i] = j;
            }
            if (gp[i] > sp[i]) {
                s("-1");
                System.exit(0); 
            }
            oc[sp[i] - gp[i]] = true;
        }
        int res = 0;
        for (int i = 0; i < w; i++)
            if (oc[i]) res++;
        s(res + "");
    }
    
    public static void sl(String str) {
       System.out.println(str);
    }
    
    public static void s(String str) {
        System.out.print(str);
    }
}
