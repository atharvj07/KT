#include <bits/stdc++.h>
using namespace std;
#define LL long long
#define N 310
LL read(){
	LL x=0,f=1;
	char ch=getchar();
	while (ch<'0' || ch>'9'){
		if (ch=='-') f=-1;
		ch=getchar();
	}
	while (ch>='0' && ch<='9'){
		x=x*10+ch-'0';ch=getchar();
	}
	return x*f;
}
int n,m,a[N],b[N];
LL sum,cnt,f[N][35];
LL solve(){
	for (int i=1;i<=n*m;++i){
		for (int j=0;j<1<<n-1;++j) f[i][j]=0;
	}
	f[n*a[n-1]][0]=1;
	for (int i=0;i<n*m;++i){
		for (int j=i;j<=n*m;++j){
			for (int k=0;k<1<<n-1;++k){
				if (i%n && !(k>>b[i%n-1]&1)){
					f[min(n*m,max(j,i+n*a[b[i%n-1]]))][k|1<<b[i%n-1]]+=f[j][k];
				}
			}
		}
	}
	return f[n*m][(1<<n-1)-1];
}
signed main(){
	n=read(),m=read();
	for (int i=0;i<n;++i) a[i]=read();
	sort(a,a+n);
	for (int i=0;i<n-1;++i) b[i]=i;
	do{
		sum+=solve();cnt++;
	}while (next_permutation(b,b+n-1));
	for (int i=0;i<n-1;++i) cnt=cnt*m;
	printf("%.12lf\n",(double)sum/cnt);
	return 0;
}
