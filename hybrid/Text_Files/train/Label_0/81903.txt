from sys import stdout, stdin
import math
r = lambda : int(input())
ra = lambda : [*map(int, input().split())]
raw = lambda : [*stdin.readline()]
out = lambda a: stdout.write(''.join(a))
a = r()
b = r()
n = a + b + 1
p = [str(i + 1) + ' ' for i in range(1, n)]
res = []
if (a or b) == 0:
	out('1')
else:
	if a == 0:
		res = p[0:b][::-1] + ['1']
	elif b == 0:
		res = ['1 '] + p[0:n]
	else:
		res = p[n - a - 2:n] + p[0:n - a - 2][::-1] + ['1']
	out(res)
