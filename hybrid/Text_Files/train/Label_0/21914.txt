#include<bits/stdc++.h>
using namespace std;

int dat[5000*5001/2];
int conv(int y,int x){
    return y*(y+1)/2+x;
}

int main(){
    int N,M;
    cin>>N>>M;

    for(int i=0;i<M;i++){
        int y,x,d;
        cin>>y>>x>>d;
        y--;x--;
        int idx=conv(y,x);
        dat[idx]=max(dat[idx],d+1);
    }

    for(int i=0;i<N-1;i++){
        for(int j=0;j<=i;j++){
            int idx=conv(i,j);
            int l=conv(i+1,j);
            int r=conv(i+1,j+1);
            dat[l]=max(dat[l],dat[idx]-1);
            dat[r]=max(dat[r],dat[idx]-1);
        }
    }

    int ans=0;
    for(int i=0;i<N*(N+1)/2;i++)if(dat[i])ans++;

    cout<<ans<<endl;
    return 0;
}
