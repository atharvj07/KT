#include <bits/stdc++.h>
using namespace std;
int p, xx;
int num[1000005];
bool judge = false;
int main() {
  scanf("%d%d", &p, &xx);
  for (int i = 1; i < 10; ++i) {
    int c = 0, k = p;
    num[p] = i;
    while (k--) {
      num[k] = (num[k + 1] * xx + c) % 10;
      c = (num[k + 1] * xx + c) / 10;
    }
    if (num[0] == num[p] && num[1] > 0 && c == 0) {
      judge = true;
      break;
    }
  }
  if (judge) {
    for (int i = 1; i <= p; ++i) printf("%d", num[i]);
  } else
    printf("Impossible\n");
  return 0;
}
