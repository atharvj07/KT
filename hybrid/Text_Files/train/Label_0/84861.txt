def findten(x):
	ans = 1
	while ans <= x:
		ans *= 10
	return ans

def s(x):
	ans = 0
	while x:
		ans += 1
		x //= 10
	return ans
(w, m, k) = (int(i) for i in input().split())
ans = 0
p = findten(m)
while w > 0:
	temp = s(m) * k * (p - m)
	if w - temp >= 0:
		ans += p - m
		m = p
		p *= 10
		w -= temp
	else:
		t = w // (s(m) * k)
		ans += t
		break
print(ans)
