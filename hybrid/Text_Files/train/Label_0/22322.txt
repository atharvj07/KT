#include <bits/stdc++.h>
using namespace std;
int n, m, c[200005];
bool vis[200005];
struct ss {
  int a, b, p, id;
  bool operator<(const ss &zech) const {
    return p > zech.p || (p == zech.p && id > zech.id);
  }
} E[200005];
priority_queue<ss> q[4];
int ip() {
  int x = 0, w = 0;
  char ch = 0;
  while (!isdigit(ch)) w |= ch == '-', ch = getchar();
  while (isdigit(ch)) x = (x << 1) + (x << 3) + (ch ^ 48), ch = getchar();
  return w ? -x : x;
}
int main() {
  int i;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> E[i].p;
    E[i].id = i;
  }
  for (i = 1; i <= n; i++) cin >> E[i].a;
  for (i = 1; i <= n; i++) cin >> E[i].b;
  for (i = 1; i <= n; i++) q[E[i].a].push(E[i]), q[E[i].b].push(E[i]);
  cin >> m;
  for (i = 1; i <= m; i++) {
    c[i] = ip();
    if (q[c[i]].empty()) {
      cout << -1 << endl;
      continue;
    }
    int u = q[c[i]].top().id;
    while (vis[u] && !q[c[i]].empty()) {
      q[c[i]].pop();
      u = q[c[i]].top().id;
    }
    if (q[c[i]].empty()) {
      cout << -1 << endl;
      continue;
    }
    printf("%d ", q[c[i]].top().p);
    vis[q[c[i]].top().id] = 1;
  }
  return 0;
}
