from math import sqrt

def f(x):
	for i in range(3, int(sqrt(x + 2)) + 2, 2):
		if x % i == 0:
			return i
	return x
t = int(input())
for _ in range(t):
	n = int(input())
	if n % 2 == 0:
		print(n // 2, n // 2)
	else:
		q = f(n)
		if q != n:
			p = n // q
			print(p, n - p)
		else:
			print(1, n - 1)
