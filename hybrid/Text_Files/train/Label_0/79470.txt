(n, k) = map(int, input().split())
(a, b, c, d) = map(int, input().split())
if k - n > 0 and n != 4:
	print(a, c, end=' ')
	for i in range(1, n + 1):
		if i != a and i != b and (i != d) and (i != c):
			print(i, end=' ')
	print(d, b)
	print(c, a, end=' ')
	for i in range(1, n + 1):
		if i != a and i != b and (i != d) and (i != c):
			print(i, end=' ')
	print(b, d)
else:
	print(-1)
