#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 1e5 + 1;
const int MOD = 998244353;
const int INF = 1e9;
const long long LINF = 1e18;
map<int, pair<int, int> > mp;
int get_highest_bit(int x, int i) {
  int ret = 0;
  int k = 0;
  while (x != 0) {
    if (x % 2 == 1) {
      mp[k].first++;
      mp[k].second = i;
    }
    k++;
    x /= 2;
  }
  return ret;
}
void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  int ind = 0;
  bool all_zero = true;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    if (a[i] != 0) all_zero = false;
    get_highest_bit(a[i], i);
  }
  if (all_zero) {
    for (int i = 0; i < n; ++i) cout << a[i] << " ";
    cout << endl;
    return;
  }
  auto iter = mp.end();
  iter--;
  for (auto it = iter;; --it) {
    int z = (*it).second.first;
    if (z == 1) {
      ind = (*it).second.second;
      break;
    }
    if (it == mp.begin()) break;
  }
  int temp = a[0];
  a[0] = a[ind];
  a[ind] = temp;
  for (int i = 0; i < n; ++i) cout << a[i] << " ";
  cout << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  while (t--) {
    solve();
  }
}
