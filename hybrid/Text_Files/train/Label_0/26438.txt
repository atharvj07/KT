#include <bits/stdc++.h>
using namespace std;
long long n, k, x, a[200005], q, w;
set<pair<long long, long long> > s;
int main() {
  scanf("%I64d%I64d%I64d", &n, &k, &x);
  for (long long i = 1; i <= n; i++) {
    scanf("%I64d", a + i);
    if (a[i] == 0) w++;
    if (a[i] < 0) q++;
  }
  if (w > k) {
    for (long long i = 1; i <= n; i++)
      if (k && a[i] == 0) a[i] += x, k--;
    for (long long i = 1; i < n; i++) printf("%I64d ", a[i]);
    printf("%I64d\n", a[n]);
    return 0;
  }
  if (w) {
    k -= w;
    if (q & 1) {
      for (long long i = 1; i <= n; i++)
        if (!a[i]) a[i] += x;
    } else {
      bool bbb = 1;
      q++;
      for (long long i = 1; i <= n; i++) {
        if (!a[i]) {
          if (bbb)
            bbb = 0, a[i] = -x;
          else
            a[i] = x;
        }
      }
    }
  }
  for (long long i = 1; i <= n; i++) s.insert(make_pair(abs(a[i]), i));
  for (long long i = 1; i <= k; i++) {
    long long o = s.begin()->second;
    s.erase(s.begin());
    if (q & 1) {
      if (a[o] < 0)
        a[o] -= x;
      else
        a[o] += x;
    } else {
      long long t = a[o];
      if (a[o] < 0)
        a[o] += x;
      else
        a[o] -= x;
      if (t < 0 && a[o] >= 0) q--;
      if (t >= 0 && a[o] < 0) q++;
    }
    s.insert(make_pair(abs(a[o]), o));
  }
  for (long long i = 1; i < n; i++) printf("%I64d ", a[i]);
  printf("%I64d\n", a[n]);
  return 0;
}
