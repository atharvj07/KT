def solve(arr, n):
	mx = 10 ** 6
	pos = [0] * mx
	neg = [0] * mx
	for br in arr:
		sm = 0
		mn = 0
		for it in br:
			if it == '(':
				sm += 1
			else:
				sm -= 1
				if sm < mn:
					mn = sm
		if sm >= 0 and mn == 0:
			pos[sm] += 1
		elif sm < 0 and sm == mn:
			neg[-sm] += 1
	res = pos[0] // 2
	for i in range(1, mx):
		res += min(pos[i], neg[i])
	print(res)
n = int(input())
arr = []
for i in range(n):
	t = input()
	arr.append(t)
solve(arr, n)
