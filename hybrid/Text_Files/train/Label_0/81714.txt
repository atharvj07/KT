import sys
input = sys.stdin.readline
(n, m, k) = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
b = [0]
for i in range(k - 1, -1, -1):
	b.append(a[i])
c = [0] * (k + 1)
for i in range(1, k + 1):
	c[i] += b[i] + c[i - 1]
s = [0] * (k + 1)
for _ in range(m):
	(x, y) = map(int, input().split())
	if x <= k:
		s[x] = max(s[x], y)
inf = 1145141919810
dp = [0] * (k + 1)
for i in range(1, k + 1):
	dpi = inf
	for j in range(1, i + 1):
		d = i - j + 1
		dpi = min(dpi, dp[j - 1] - c[j - 1] + c[i - s[d]])
	dp[i] = dpi
ans = dp[k]
print(ans)
