def main():
	t = int(input())
	for i in range(t):
		(n, s) = list(map(int, input().split()))
		a = list(map(int, input().split()))
		(l1, l2, r_l1, r_l2) = (0, 0, 0, 0)
		summa = 0
		while l1 < n:
			if l2 < l1:
				l2 = l1
				summa = 0
			while l2 < n and summa + a[l2] + s >= 0:
				summa += a[l2]
				l2 += 1
			if r_l2 - r_l1 < l2 - l1:
				(r_l2, r_l1) = (l2, l1)
			summa -= a[l1]
			l1 += 1
		print(r_l1 + 1, r_l2) if r_l1 < r_l2 else print(-1)
main()
