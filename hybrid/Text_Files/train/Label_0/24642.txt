#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  unsigned long long cnte[2] = {0};
  unsigned long long cnto[2] = {0};
  for (int i = 0; i < (int)s.size(); i++) {
    int id = s[i] - 'a';
    if ((i) % 2 == 0) {
      cnte[id]++;
    } else {
      cnto[id]++;
    }
  }
  unsigned long long anseven = 0;
  unsigned long long ansodd = 0;
  for (int i = 0; i < 2; i++)
    ansodd += (unsigned long long)cnte[i] * (cnte[i] - 1) / 2;
  for (int i = 0; i < 2; i++)
    ansodd += (unsigned long long)cnto[i] * (cnto[i] - 1) / 2;
  for (int i = 0; i < 2; i++) anseven += (unsigned long long)cnte[i] * cnto[i];
  cout << anseven << " " << ansodd + s.size() << endl;
  return 0;
}
