import sys
t = int(input())
for test in range(t):
	n = int(input())
	a = list(map(int, input().split(' ')))
	s = [[] for i in range(n + 1)]
	res = ['1' for i in range(n)]
	for i in range(n):
		if s[a[i]] != []:
			res[0] = '0'
		s[a[i]].append(i)
	k = 0
	for i in range(1, n + 1):
		if len(s[i]) == 0:
			for j in range(0, n - i + 1):
				res[j] = '0'
			break
		elif len(s[i]) > 1:
			for j in range(0, n - i):
				res[j] = '0'
			break
		elif s[i][0] == k:
			k += 1
		elif s[i][0] != n - i + k:
			for j in range(1, n - i):
				res[j] = '0'
			break
	print(''.join(res))
