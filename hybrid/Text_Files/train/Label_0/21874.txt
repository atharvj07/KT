#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int a[N], b[N], mp[N], l[N], ner[N], fa[N][25], n, m, q;
int solve(int u) {
  int dif = n - 1;
  for (int i = 20; i >= 0; --i) {
    if ((dif >> i) & 1) {
      u = fa[u][i];
    }
  }
  return u;
}
int main() {
  cin >> n >> m >> q;
  for (int i = (1); i < (n + 1); i++) {
    cin >> a[i];
    mp[a[i]] = i;
  }
  int x;
  for (int i = (1); i < (m + 1); i++) {
    cin >> x;
    b[i] = mp[x];
  }
  for (int i = 1; i <= m; ++i) {
    int fx = b[i] - 1;
    if (fx == 0) fx = n;
    if (ner[fx]) {
      fa[i][0] = ner[fx];
    }
    ner[b[i]] = i;
  }
  for (int i = 1; i <= m; ++i) {
    for (int j = 1; j <= 20; ++j) {
      fa[i][j] = fa[fa[i][j - 1]][j - 1];
    }
  }
  for (int i = 1; i <= m; ++i) {
    l[i] = solve(i);
    l[i] = max(l[i], l[i - 1]);
  }
  int tl, tr;
  while (q--) {
    cin >> tl >> tr;
    if (l[tr] >= tl) {
      cout << 1;
    } else
      cout << 0;
  }
  return 0;
}
