#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
long long qpow(long long a, long long b, long long m) {
  long long r = 1;
  a %= m;
  for (; b; b >>= 1) {
    if (b & 1) r = r * a % m;
    a = a * a % m;
  }
  return r;
}
const int inf = 0x7fffffff;
const int maxn = 2e5 + 10;
map<pair<int, int>, bool> mp;
set<int> s;
int cnt[maxn << 1];
int n, k, m;
struct node {
  int mx, tag;
  int l, r;
} t[maxn << 3];
void pushup(int rt) { t[rt].mx = max(t[rt << 1].mx, t[rt << 1 | 1].mx); }
void pushdown(int rt) {
  if (t[rt].tag) {
    t[rt << 1].tag += t[rt].tag;
    t[rt << 1 | 1].tag += t[rt].tag;
    t[rt << 1].mx += t[rt].tag;
    t[rt << 1 | 1].mx += t[rt].tag;
    t[rt].tag = 0;
  }
}
void build(int l, int r, int rt) {
  t[rt].mx = t[rt].tag = 0;
  t[rt].l = l;
  t[rt].r = r;
  if (l == r) {
    t[rt].mx = l - 1 - n;
    return;
  }
  int mid = (l + r) >> 1;
  build(l, mid, rt << 1);
  build(mid + 1, r, rt << 1 | 1);
  pushup(rt);
}
void update(int l, int r, int rt, int x) {
  if (t[rt].l > r || t[rt].r < l) return;
  if (t[rt].l >= l && t[rt].r <= r) {
    t[rt].tag += x;
    t[rt].mx += x;
    return;
  }
  pushdown(rt);
  update(l, r, rt << 1, x);
  update(l, r, rt << 1 | 1, x);
  pushup(rt);
}
int qmax(int l, int r, int rt) {
  if (t[rt].l >= l && t[rt].r <= r) return t[rt].mx;
  int mid = (t[rt].l + t[rt].r) >> 1;
  pushdown(rt);
  if (r <= mid)
    return qmax(l, r, rt << 1);
  else if (l > mid)
    return qmax(l, r, rt << 1 | 1);
  else
    return max(qmax(l, r, rt << 1), qmax(l, r, rt << 1 | 1));
}
int lst() {
  set<int>::iterator it = s.end();
  it--;
  return *it;
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> k >> m;
  build(1, 2 * n, 1);
  while (m--) {
    int x, y, xx = 1;
    cin >> x >> y;
    int yy = abs(x - k) + y;
    if (mp[pair<int, int>(x, y)]) {
      xx = -1;
      cnt[yy]--;
      if (cnt[yy] == 0) s.erase(yy);
    } else {
      if (cnt[yy] == 0) s.insert(yy);
      cnt[yy]++;
    }
    mp[pair<int, int>(x, y)] = !mp[pair<int, int>(x, y)];
    update(1, yy, 1, xx);
    if (s.empty())
      cout << 0 << endl;
    else
      cout << max(qmax(1, lst(), 1), 0) << endl;
  }
  return 0;
}
