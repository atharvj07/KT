#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using P = pair<ll,ll>;
#define pb push_back
#define fi first
#define sec second
#define INF 1000000000
#define DINF 1000000000.0
#define eps 1e-18
int N,M;
int A[100100],B[100100];
ll x[100100];
struct edge{
	int to;
	vector<ll> costs;
	edge(){}
	edge(int to,vector<ll> costs):to(to),costs(costs){}
};
vector<edge> g[100100];
bool used[100100];
vector<ll> pr = {1000000007ll,1000000009ll,998244353ll};
vector<ll> dp[100100];
vector<ll> add(vector<ll>& a,vector<ll>& b){
	vector<ll> res;
	for(int i=0;i<a.size();i++){
		res.pb((a[i]*b[i])%pr[i]);
	}
	return res;
}
ll mypow(ll a,ll x,ll mod){
	ll res = 1ll;
	while(x){
		if(x&1ll)res = (res*a)%mod;
		a = (a*a)%mod;
		x >>= 1;
	}
	return res;
}
int main(){
	cin >> N >> M;
	cout << fixed << setprecision(20);
	for(int i=0;i<M;i++){
		cin >> A[i] >> B[i] >> x[i];
		A[i]--;B[i]--;
		vector<ll> c,d;
		for(int j=0;j<pr.size();j++){
			c.pb(x[i]%pr[j]);
			d.pb(mypow(x[i],pr[j]-2,pr[j]));
		}
		g[A[i]].pb(edge(B[i],c));
		g[B[i]].pb(edge(A[i],d));
		// cout << log(x[i]) << endl;
	}
	// toposort();
	/*for(int i=0;i<M;i++){
		if(rev[A[i]]>rev[B[i]]){
			cout << "No" << endl;
			return 0;
		}
	}*/	
	for(int i=0;i<N;i++){	
		if(dp[i].size()>0)continue;
		for(int j=0;j<pr.size();j++)dp[i].pb(1ll);
		queue<int> q;
		q.push(i);
		while(!q.empty()){
			int a = q.front();
			q.pop();
			for(int i=0;i<g[a].size();i++){
				edge& e = g[a][i];
				vector<ll> c = add(dp[a],e.costs);
				if(dp[e.to].size()==0){
					dp[e.to] = c;
					q.push(e.to);
				}else{
					if(dp[e.to]!=c){
						cout << "No" << endl;
						return 0;
					}
				}
			}
		}
	}
	cout << "Yes" << endl;
	return 0;
}
