#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int arr[n][2];
  for (int i = 0; i < n; ++i) {
    cin >> arr[i][0] >> arr[i][1];
  }
  int b[m];
  for (int i = 0; i < m; ++i) {
    cin >> b[i];
  }
  int sum = arr[0][0] * arr[0][1];
  int ans = 1;
  for (int i = 0; i < m; ++i) {
    if (b[i] <= sum)
      cout << ans << endl;
    else if (b[i] > sum) {
      while (sum < b[i]) {
        sum += (arr[ans][0] * arr[ans][1]);
        ans++;
      }
      cout << ans << endl;
    }
  }
  return 0;
}
