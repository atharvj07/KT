(n, m) = map(int, input().split())
graph = []
from sys import *
for _ in range(0, n + 2):
	graph.append([])
for _ in range(0, m):
	(a, b) = map(int, stdin.readline().split())
	graph[b].append(a)
vis = [0] * (n + 2)
f = []

def topo(start):
	global f
	vis[start] = 1
	s = [start]
	ans = []
	while s:
		temp = s[-1]
		vis[temp] = 1
		for i in graph[temp]:
			if vis[i] == 0:
				s.append(i)
				break
		else:
			f.append(s.pop())
for i in range(1, n + 1):
	if vis[i] == 0:
		topo(i)
f.reverse()
for i in f:
	print(i, end=' ')
