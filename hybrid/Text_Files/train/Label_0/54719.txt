import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class D {
    static int MOD = 1000000007;

    static class BIT {
        int[] tree;

        public BIT(int sz) {
            tree = new int[sz];
        }

        public void add(int i, int v) {
            for (; i < tree.length; i += i & -i)
                tree[i] = (tree[i] + v);
        }

        public int get(int i) {
            int r = 0;
            for (; i > 0; i &= (i - 1))
                r = (r + tree[i]);
            return r;
        }
    }

    static long[] f;
    static long[] g;
    static int MAXN = (int) 1e6 + 1;

    public static void main(String[] args) throws Exception {
        f = new long[MAXN];
        g = new long[MAXN];
        f[0] = f[1] = 1;
        g[0] = g[1] = 0;
        for (int i = 2; i < MAXN; i++) {
            f[i] = (i * f[i - 1]) % MOD;
            long s = (i * (i - 1L) / 2) % MOD;
            g[i] = ((g[i - 1] * i) % MOD + (s * f[i - 1]) % MOD) % MOD;
        }
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer strtok = new StringTokenizer(in.readLine());
//      int n = 1000000;
//      int[] p = new int[n];
//      for (int i = 0; i < n; i++)
//          p[i] = n - i;
        int n = Integer.parseInt(strtok.nextToken());
        int[] p = new int[n];
        strtok = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++)
            p[i] = Integer.parseInt(strtok.nextToken());
        BIT count = new BIT(n + 5);
        long[] h = new long[n + 1];
        h[n] = 1;
        for (int i = n - 1; i >= 0; i--) {
            h[i] = (count.get(p[i]) * f[n - 1 - i] + h[i + 1]) % MOD;
            count.add(p[i], 1);
        }
        BIT used = new BIT(n + 5);
        long result = 0;
        for (int i = 0; i < n; i++) {
            long prev = p[i] - 1 - used.get(p[i]);
            long s = (prev * (prev - 1) / 2) % MOD;
            result = (result + (s * f[n - i - 1]) % MOD + (prev * g[n - i - 1])
                    % MOD)
                    % MOD;
            result = (result + prev * h[i + 1]) % MOD;
            used.add(p[i], 1);
        }
        System.out.println(result);
