(x, y, l, r) = map(int, input().split())
s = set()
px = 1
py = 1
while px <= r:
	while py <= r:
		if l <= px + py and px + py <= r:
			s.add(px + py)
		py = py * y
	py = 1
	px = px * x
s.add(l - 1)
s.add(r + 1)
t = []
for x in s:
	t.append(x)
t.sort()
ans = 0
for i in range(1, len(t)):
	ans = max(ans, t[i] - t[i - 1] - 1)
print(ans)
