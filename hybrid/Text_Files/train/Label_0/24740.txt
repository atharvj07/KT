#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const int MOD = 1e9 + 7;
int main() {
  double w, h, a;
  cin >> w >> h >> a;
  if (a > 90) a = 180 - a;
  a = a * acos(-1.0) / 180;
  if (w < h) swap(w, h);
  double ang = atan2(h * 1.0, w * 1.0) * 2;
  if (a == 0) {
    printf("%.10lf\n", w * h);
    return 0;
  }
  if (a > ang) {
    double x = w - (h / sin(a));
    printf("%.10lf\n", w * h - x * h);
  } else {
    double ans = (1.0 / sin(a) + 1.0 / tan(a));
    double x = (h - ans * w) / (1 - ans * ans);
    double y = (w - x / tan(a) - x / sin(a));
    double sum = x * (x / tan(a)) + y * y / tan(a);
    printf("%.10lf\n", w * h - sum);
  }
  return 0;
}
