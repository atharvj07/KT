import math
Q = int(input())
for _ in range(Q):
	n = int(input())
	arr = list(map(int, input().split()))
	summa = sum(arr)
	xor = 0
	for i in arr:
		xor ^= i
	ans = [0]
	if summa % 2 == 1:
		summa += 1
		xor ^= 1
		ans[0] = 1
	if xor * 2 < summa:
		t = math.log2(summa - xor)
		t = int(t + 1)
		xor ^= 2 ** t
		summa += 2 ** t
		ans[0] += 2 ** t
	ans.append((xor * 2 - summa) // 2)
	ans.append((xor * 2 - summa) // 2)
	print(len(ans))
	print(*ans)
