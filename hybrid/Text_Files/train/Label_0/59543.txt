import java.util.*;
import java.util.Map.*;

public class MageAndMonster {
	static TreeMap<Double, Double> map;
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int q = in.nextInt();
		double mana = in.nextDouble();
		int j = 0;
		map = new TreeMap<Double, Double>();
		for (int i = 1; i <= q; ++i) {
			int type = in.nextInt();
			double a = helper(j, in.nextInt());
			double b = helper(j, in.nextInt());
			
			if (type == 1) {
				double val = mana / b * a;
				if (!check(a, val)) {
					map.put(a, val);
					Double left = map.ceilingKey(a + 1);
					Double right = map.floorKey(a - 1);
					
					adjustLeft(a, left);
					adjustRight(a, right);
					
				}
			} else {
				//System.out.print("Current J: " + j + ", ");
				if (check(b / a, b)) {
					System.out.println("YES");
					j = i;
				} else {
					System.out.println("NO");
					
				}
			}
		}
	}
	
	public static void adjustLeft(Double a, Double d) {
		double ax = map.get(a) / a, ay = map.get(a);
		while (d != null) {
			Double next = map.ceilingKey(d + 1);
			if (next != null) {
				double dx = map.get(d) / d, dy = map.get(d),
						nx = map.get(next) / next, ny = map.get(next);
				if ((ay - dy) * (dx - nx) - (ax - dx) * (dy - ny) >= 0) {
					map.remove(d);
				} else {
					break;
				}
			}
			d = next;
		}
	}
	
	public static void adjustRight(Double a, Double d) {
		double ax = map.get(a) / a, ay = map.get(a);
		while (d != null) {
			Double next = map.floorKey(d - 1);
			if (ay >= map.get(d)) {
				map.remove(d);
			} else if (next != null) {
				double dx = map.get(d) / d, dy = map.get(d),
						nx = map.get(next) / next, ny = map.get(next);
				if ((ay - dy) * (dx - nx) - (ax - dx) * (dy - ny) <= 0) {
					map.remove(d);
				} else {
					break;
				}
			}
			d = next;
		}
	}
	
	//returns true if [x, y] can be reached
	public static boolean check(Double x, Double y) {
		Entry<Double, Double> left = map.ceilingEntry(x);
		Entry<Double, Double> right = map.floorEntry(x);
		if (left == null) {
			return false;
		} else if (right == null){
			return left.getValue() >= y;
		} else {
			double ly = left.getValue(), lx = left.getValue() / left.getKey(),
					ry = right.getValue(), rx = right.getValue() / right.getKey();
			x = y / x;
			return (y - ly) * (rx - lx) - (x - lx) * (ry - ly) <= 0;
		}
	}
	
	public static int helper(int j, int k) {
		return (j + k) % 1000000 + 1;
	}

}
