for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	r = []
	mov = 0
	for i in a:
		m = i % 3
		r.append(m)
	c0 = r.count(0)
	c1 = r.count(1)
	c2 = r.count(2)
	per = n // 3
	if c0 < per:
		while c0 < per:
			if c2 > per:
				mov = mov + 1
				c2 = c2 - 1
			else:
				mov = mov + 2
				c1 = c1 - 1
			c0 = c0 + 1
	if c1 < per:
		while c1 < per:
			if c0 > per:
				mov = mov + 1
				c0 = c0 - 1
			else:
				mov = mov + 2
				c2 = c2 - 1
			c1 = c1 + 1
	if c2 < per:
		while c2 < per:
			if c1 > per:
				mov = mov + 1
				c1 = c1 - 1
			else:
				mov = mov + 2
				c0 = c0 - 1
			c2 = c2 + 1
	print(mov)
