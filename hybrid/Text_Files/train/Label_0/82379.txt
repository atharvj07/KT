def calc(a, v):
	n = len(a)
	ans = 0
	for i in range(n):
		if a[i] > v + 1e-10:
			b = a.copy()
			b[i] -= v
			if n == 2:
				b[1 - i] += v / n
			ans += (calc(b, v) + 1) * a[i]
		elif n == 2:
			ans += (calc([a[1 - i] + a[i] / 2], v) + 1) * a[i]
		else:
			ans += (1 + 1) * a[i]
	ans += 1 * (1 - sum(a))
	return ans

def solve_case():
	a = list(map(float, input().split()))
	print(f'{calc(a[:2], a[-1]):.12f}')
n = int(input())
for i in range(n):
	solve_case()
