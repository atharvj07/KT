#include <bits/stdc++.h>
using namespace std;
long long n, t, data[55];
long long f[55][55], g[55][55];
int mem[55], top, last, D;
long long Ans;
int main() {
  scanf("%I64d%I64d", &n, &t);
  n++;
  data[0] = 1LL;
  for (int i = 1; i <= 40; i++) data[i] = data[i - 1] << 1LL;
  D = lower_bound(data, data + 41, t) - data;
  if (data[D] != t) {
    puts("0");
    return 0;
  }
  D++;
  f[1][1] = 1;
  g[1][1] = 1;
  g[0][0] = g[1][0] = 1;
  for (int i = 2; i <= 40; i++) {
    for (int j = 1; j <= i; j++) {
      f[i][j] = f[i - 1][j] + f[i - 1][j - 1];
      g[i][j] = g[i - 1][j] + f[i][j];
    }
    g[i][0] = 1;
  }
  for (top = 0; n; n >>= 1) mem[++top] = n & 1;
  int ck = 0;
  for (int i = 1; i <= top; i++)
    if (mem[i]) ck++;
  Ans = g[top - 1][D] + (ck == D);
  if (D == 1) Ans--;
  for (int i = top; i; i--) {
    int st = (i == top) ? (1) : (0);
    for (int j = st; j < mem[i]; j++) Ans += g[i - 1][D - last];
    if (mem[i]) last++;
    if (last > D) break;
  }
  printf("%I64d\n", Ans);
  return 0;
}
