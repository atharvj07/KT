import java.io.BufferedReader;
// import java.io.FileInputStream;
// import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.*;

import static java.lang.Math.*;
import static java.util.Arrays.copyOf;
import static java.util.Arrays.fill;
import static java.util.Arrays.sort;
import static java.util.Collections.reverseOrder;
import static java.util.Collections.sort;

public class Main {
    FastScanner in;
    PrintWriter out;

    private void solve() throws IOException {
        int k = in.nextInt();
        char[] s = in.nextLine().toCharArray(),
                a = in.nextLine().toCharArray(),
                b = in.nextLine().toCharArray();

        int n = s.length;
        for (int i = 0; i < n; i++) {
            s[i] -= 'a';
            a[i] -= 'a';
            b[i] -= 'a';
        }

        for (int cntTry = 0; cntTry < 2; cntTry++) {
            int[] pattern = new int[k];
            fill(pattern, -1);
            boolean[] used = new boolean[k];
            boolean flag = false;

            main:
            for (int i = 0; i < n; i++) {
                if (pattern[s[i]] != -1) {
                    flag |= cntTry == 0 ? a[i] < pattern[s[i]] && pattern[s[i]] <= b[i] :
                            a[i] <= pattern[s[i]] && pattern[s[i]] < b[i];
                    continue;
                }

                if (!flag) {
                    for (int j = a[i] + (cntTry + 1) % 2; j < b[i] + (cntTry + 1) % 2; j++) {
                        if (!used[j]) {
                            pattern[s[i]] = j;
                            used[j] = true;
                            flag = true;
                            continue main;
                        }
                    }
                    if (!used[cntTry == 0 ? a[i] : b[i]]) {
                        pattern[s[i]] = cntTry == 0 ? a[i] : b[i];
                        used[cntTry == 0 ? a[i] : b[i]] = true;
                        continue;
                    }
                }

                for (int j = cntTry == 0 ? 0 : k - 1; 0 <= j && j < k; j = cntTry == 0 ? j + 1 : j - 1) {
                    if (!used[j]) {
                        pattern[s[i]] = j;
                        used[j] = true;
                        continue main;
                    }
                }
            }

            for (int i = 0, j = 0; i < k; i++) {
                if (pattern[i] == -1) {
                    while (used[j])
                        j++;
                    pattern[i] = j++;
                }
            }

            char[] cur = new char[n];
            for (int i = 0; i < n; i++)
                cur[i] = (char) pattern[s[i]];

            if (new String(a).compareTo(new String(cur)) <= 0 && new String(cur).compareTo(new String(b)) <= 0) {
                out.println("YES");
                for (int i = 0; i < k; i++)
                    out.print((char) (pattern[i] + 'a'));
                out.println();
                return;
            }
        }

        out.println("NO");
    }

    class FastScanner {
        StringTokenizer st;
        BufferedReader br;

        FastScanner(InputStream s) {
            br = new BufferedReader(new InputStreamReader(s));
        }

        String next() throws IOException {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        boolean hasNext() throws IOException {
            return br.ready() || (st != null && st.hasMoreTokens());
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }

        String nextLine() throws IOException {
            return br.readLine();
        }

        boolean hasNextLine() throws IOException {
            return br.ready();
        }

    }

    private void run() throws IOException {
        in = new FastScanner(System.in); // new FastScanner(new FileInputStream(".in"));
        out = new PrintWriter(System.out); // new PrintWriter(new FileOutputStream(".out"));

        for (int t = in.nextInt(); t-- > 0; ) {
            solve();
            out.flush();
        }

        out.flush();
        out.close();
    }

    public static void main(String[] args) throws IOException {
        new Main().run();
    }
}
