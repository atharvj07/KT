import sys
(n, m) = map(int, input().split())
cost = [int(x) for x in input().split()]
adj = [[] for _ in range(n)]
for i in range(m):
	(u, v) = map(int, input().split())
	adj[u - 1].append(v - 1)
	adj[v - 1].append(u - 1)
visited = [0 for x in range(n)]
minCost = -1 - sys.maxsize

def dfs(v):
	l = [v]
	minCost = cost[v]
	while l:
		top = l.pop()
		visited[top] = 1
		minCost = min(minCost, cost[top])
		for u in adj[top]:
			if visited[u] == 0:
				l.append(u)
	return minCost
res = 0
for vertex in range(n):
	if visited[vertex] == 0:
		res += dfs(vertex)
print(res)
