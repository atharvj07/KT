import sys, math
input = sys.stdin.readline
for _ in range(int(input())):
	s = input()
	z = None
	f = None
	for i in range(len(s) - 2, -1, -1):
		if s[i] == '5' and f == None:
			f = i
		if z == None and s[i] == '0':
			z = i
		elif z and (s[i] == '5' or s[i] == '0'):
			print(len(s) - 3 - i)
			break
		if f and (s[i] == '2' or s[i] == '7'):
			print(len(s) - 3 - i)
			break
