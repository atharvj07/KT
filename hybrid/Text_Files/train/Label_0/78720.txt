for _ in range(int(input())):
	np = int(input())
	even_q = 0
	odd_q = 0
	p = [1 for i in input().split() if int(i) % 2 == 0]
	nq = int(input())
	q = [1 for i in input().split() if int(i) % 2 == 0]
	even_p = sum(p)
	odd_p = np - even_p
	even_q = sum(q)
	odd_q = nq - even_q
	print(even_p * even_q + odd_p * odd_q)
