n = int(input())
num = list(map(int, input().split()))
left = -1
right = -1
for i in range(n):
	if num[i] - 1 < 10 ** 6 - num[i]:
		left = i
for i in range(n - 1, -1, -1):
	if num[i] - 1 > 10 ** 6 - num[i]:
		right = i
if right == -1:
	print(num[-1] - 1)
elif left == -1:
	print(10 ** 6 - num[0])
else:
	print(max(num[left] - 1, 10 ** 6 - num[right]))
