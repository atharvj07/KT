import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inara():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())
MOD = 998244353
n = inp()
dp = [0 for i in range(n + 1)]
fact = [1 for i in range(n + 1)]
for i in range(2, n + 1):
	dp[i] = i * (dp[i - 1] + 1)
	dp[i] %= MOD
	fact[i] = fact[i - 1] * i
	fact[i] %= MOD
ans = n * fact[n] - dp[n]
ans %= MOD
if ans < 0:
	ans += MOD
print(ans)
