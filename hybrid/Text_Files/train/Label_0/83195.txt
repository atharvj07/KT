def main():
	s = input()
	cnt = 0
	min_symbol = 96
	ord_list = [ord(el) for el in s]
	for (index, symbol) in enumerate(ord_list):
		if cnt == 26:
			break
		if symbol <= min_symbol + 1:
			ord_list[index] = min_symbol + 1
			min_symbol += 1
			cnt += 1
	if cnt != 26:
		print(-1)
		return
	print(''.join([chr(el) for el in ord_list]))
main()
