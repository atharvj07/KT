struct edge {
  int from, to, w;
  edge(int from, int to, int w) : from(from), to(to), w(w) {}
  bool operator<(const edge& e) const { return w > e.w; }
};

struct UnionFind {
  vector<int> rank, parent;
  int trees;
  UnionFind(int n) {
    rank = vector<int>(n), parent = vector<int>(n);
    trees = n;
    for (int i = 0; i < n; i++) parent[i] = i, rank[i] = 1;
  }
  int find_set(int x) {
    if (x == parent[x]) return x;
    return parent[x] = find_set(parent[x]);
  }
  void link(int x, int y) {
    if (rank[x] > rank[y]) swap(x, y);
    parent[x] = y;
    if (rank[x] == rank[y]) ++rank[y];
  }
  bool union_sets(int x, int y) {
    x = find_set(x), y = find_set(y);
    if (x != y) {
      link(x, y);
      --trees;
    }
    return x != y;
  }
};

int Kruskal(vector<edge> edgeList, int n) {
  UnionFind uf(n);
  edges = vector<vector<edge> >(n);
  int cost = 0;
  priority_queue<edge> q;
  for (int i = 0; i < ((int)((edgeList).size())); i++) q.push(edgeList[i]);
  while (!q.empty()) {
    edge e = q.top();
    q.pop();
    if (uf.union_sets(e.from, e.to)) {
      cost += e.w;
      edges[e.from].push_back(e);
      edges[e.to].push_back(edge(e.to, e.from, e.w));
    }
  }
  return cost;
}
int dist(int i, int j) {
  i = i * n;
  j = j * n;
  int sum = 0;
  for (int e = i; e < i + n; ++e, ++j) {
    for (int f = 0; f < m; ++f) {
      if (grid[e][f] != grid[j][f]) sum += w;
    }
  }
  return sum;
}
void dfs(int node) {
  visited[node] = true;
  for (int i = 0; i < ((int)((edges[node]).size())); ++i) {
    if (!visited[edges[node][i].to]) {
      printf("%d %d\n", edges[node][i].to, edges[node][i].from);
      dfs(edges[node][i].to);
    }
  }
}
