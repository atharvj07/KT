import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.io.FilterInputStream;
import java.io.BufferedInputStream;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author Jenish
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        ScanReader in = new ScanReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        BWorkingOut solver = new BWorkingOut();
        solver.solve(1, in, out);
        out.close();
    }

    static class BWorkingOut {
        int n;
        int m;
        long[][] mat;
        long[][] dp1;
        long[][] dp2;
        long[][] dp3;
        long[][] dp4;

        public void solve(int testNumber, ScanReader in, PrintWriter out) {
            n = in.scanInt();
            m = in.scanInt();
            mat = new long[n + 5][m + 5];
            dp1 = new long[n + 5][m + 5];
            dp2 = new long[n + 5][m + 5];
            dp3 = new long[n + 5][m + 5];
            dp4 = new long[n + 5][m + 5];


            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    mat[i][j] = in.scanLong();
                }
            }


            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    dp1[i][j] = mat[i][j] + Math.max(dp1[i][j - 1], dp1[i - 1][j]);
                }
            }


            for (int j = m; j >= 1; j--) {
                for (int i = 1; i <= n; i++) {
                    dp2[i][j] = mat[i][j] + Math.max(dp2[i - 1][j], dp2[i][j + 1]);
                }
            }


            for (int i = n; i >= 1; i--) {
                for (int j = 1; j <= m; j++) {
                    dp3[i][j] = mat[i][j] + Math.max(dp3[i][j - 1], dp3[i + 1][j]);
                }
            }

            for (int i = n; i >= 1; i--) {
                for (int j = m; j >= 1; j--) {
                    dp4[i][j] = mat[i][j] + Math.max(dp4[i + 1][j], dp4[i][j + 1]);
                }
            }


            long ans = 0;
            for (int i = 2; i < n; i++) {
                for (int j = 2; j < m; j++) {
                    ans = Math.max(dp1[i - 1][j] + dp2[i][j + 1] + dp4[i + 1][j] + dp3[i][j - 1], ans);
                    ans = Math.max(dp1[i][j - 1] + dp4[i][j + 1] + dp3[i + 1][j] + dp2[i - 1][j], ans);
                }

            }


            out.println(ans);
        }

    }

    static class ScanReader {
        private byte[] buf = new byte[4 * 1024];
        private int INDEX;
        private BufferedInputStream in;
        private int TOTAL;

        public ScanReader(InputStream inputStream) {
            in = new BufferedInputStream(inputStream);
        }

        private int scan() {
            if (INDEX >= TOTAL) {
                INDEX = 0;
                try {
                    TOTAL = in.read(buf);
                } catch (Exception e) {
                    e.printStackTrace();
                }
                if (TOTAL <= 0) return -1;
            }
            return buf[INDEX++];
        }

        public int scanInt() {
            int I = 0;
            int n = scan();
            while (isWhiteSpace(n)) n = scan();
            int neg = 1;
            if (n == '-') {
                neg = -1;
                n = scan();
            }
            while (!isWhiteSpace(n)) {
                if (n >= '0' && n <= '9') {
                    I *= 10;
                    I += n - '0';
                    n = scan();
                }
            }
            return neg * I;
        }

        private boolean isWhiteSpace(int n) {
            if (n == ' ' || n == '\n' || n == '\r' || n == '\t' || n == -1) return true;
            else return false;
        }

        public long scanLong() {
            long I = 0;
            int n = scan();
            while (isWhiteSpace(n)) n = scan();
            int neg = 1;
            if (n == '-') {
                neg = -1;
                n = scan();
            }
            while (!isWhiteSpace(n)) {
                if (n >= '0' && n <= '9') {
                    I *= 10;
                    I += n - '0';
                    n = scan();
                }
            }
            return neg * I;
        }

    }
}
