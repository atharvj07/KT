#include <bits/stdc++.h>
using namespace std;

int N, M, Q, ans;
vector<vector<int>> ABCD;

void dfs(vector<int> A, int pre) {
  if (A.size() == N) {
    int sum = 0;
    for (auto a : ABCD) if (A[a[1]] - A[a[0]] == a[2]) sum += a[3];
    ans = max(ans, sum);
    return;
  }
  for (int i = pre; i <= M; i++) {
    vector<int> newA = A;
    newA.push_back(i);
    dfs(newA, i);
  }
}

int main() {
  cin >> N >> M >> Q;
  for (int a, b, c, d; cin >> a >> b >> c >> d;) ABCD.push_back({--a, --b, c, d});
  vector<int> A;
  dfs(A, 1);
  cout << ans << "\n";
}
