#include <bits/stdc++.h>
using namespace std;
const long long M = 1e9 + 7;
long long inf = 2e18;
string to_string(long long x) {
  stringstream second;
  second << x;
  return second.str();
}
long long to_integer(string s) {
  stringstream second(s);
  long long x;
  second >> x;
  return x;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n;
  cin >> n;
  long long a[n + 1];
  a[0] = 0;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i];
  }
  long long ans = 0;
  vector<long long> v;
  for (long long k = 1; k <= n; k++) {
    long long x[k];
    long long f = 1;
    for (long long j = 0; j < k; j++) x[j] = inf;
    x[0] = a[1];
    for (long long i = 2; i <= n; i++) {
      long long y = a[i] - a[i - 1];
      long long ind = (i - 1) % k;
      if (x[ind] != inf && x[ind] != y) {
        f = 0;
        break;
      } else
        x[ind] = y;
    }
    if (f == 1) {
      ans++;
      v.push_back(k);
    }
  }
  sort(v.begin(), v.end());
  cout << ans << "\n";
  for (long long i = 0; i < v.size(); i++) cout << v[i] << " ";
  return 0;
}
