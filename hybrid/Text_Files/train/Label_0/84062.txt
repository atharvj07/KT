from collections import Counter, OrderedDict
from itertools import permutations as perm
from fractions import Fraction
from collections import deque
from sys import stdin
from bisect import *
from heapq import *
g = lambda : stdin.readline().strip()
gl = lambda : g().split()
gil = lambda : [int(var) for var in gl()]
gfl = lambda : [float(var) for var in gl()]
gcl = lambda : list(g())
gbs = lambda : [int(var) for var in g()]
mod = int(1000000000.0) + 7
inf = float('inf')
(n,) = gil()
if n & 1:
	print(-1)
	exit()
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
	(x, y) = gil()
	adj[x].append(y)
	adj[y].append(x)
vis = [0] * (n + 1)
ans = [0] * (n + 1)
stack = [1]
final = -1
while stack:
	p = stack[-1]
	if vis[p]:
		stack.pop()
		for c in adj[p]:
			ans[p] += ans[c]
		ans[p] += 1
		if ans[p] & 1 == 0:
			final += 1
	else:
		vis[p] = 1
		for c in adj[p]:
			if not vis[c]:
				stack.append(c)
print(final)
