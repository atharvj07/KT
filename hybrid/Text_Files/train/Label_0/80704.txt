xy = list(map(int, input().strip().split()))
x = xy[0]
y = xy[1]
n = int(input().strip())
n = n % 6
if n == 1:
	print(x % (10 ** 9 + 7))
elif n == 2:
	print(y % (10 ** 9 + 7))
elif n == 3:
	print((y - x) % (10 ** 9 + 7))
elif n == 4:
	print(-x % (10 ** 9 + 7))
elif n == 5:
	print(-y % (10 ** 9 + 7))
else:
	print((x - y) % (10 ** 9 + 7))
