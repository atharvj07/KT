kase = int(input().strip())
while kase > 0:
	kase = kase - 1
	n = int(input().strip())
	s = [str(si) for si in input().strip()]
	slen = len(s)
	ans = 0
	if len(list({}.fromkeys(s).keys())) == 1:
		print((slen - 1) // 3 + 1)
		continue
	for (i, si) in enumerate(s):
		u = i
		v = (i + 1) % slen
		if s[u] is 'R' and s[v] is 'L':
			cnt_r = 0
			while s[u] == 'R':
				cnt_r = cnt_r + 1
				u = (u + slen - 1) % slen
			cnt_l = 0
			while s[v] == 'L':
				cnt_l = cnt_l + 1
				v = (v + 1) % slen
			ans = ans + cnt_l // 3 + cnt_r // 3
	print(ans)
