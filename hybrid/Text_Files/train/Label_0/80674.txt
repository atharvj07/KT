import string
l = list(str(input()))
k = int(input())
a = list(string.ascii_lowercase)
b = [int(n1) for n1 in input().split()]
(b, a) = zip(*sorted(zip(b, a), reverse=True))
l = l + k * list(a[0])
sum1 = 0
for i in range(len(l)):
	for j in range(len(a)):
		if a[j] == l[i]:
			sum1 = sum1 + (i + 1) * b[j]
print(sum1)
