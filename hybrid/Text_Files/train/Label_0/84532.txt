import os
import sys
if os.path.exists('C:\\Users\\User\\codeforces'):
	f = iter(open('A.txt').readlines())

	def input():
		return next(f).strip()
else:
	input = lambda : sys.stdin.readline().strip()
fprint = lambda *args: print(*args, flush=True)
(n, k, d) = map(int, input().split())
M = 10 ** 9 + 7
N = 101
lessD = [0] * N
lessD[0] = 1
for i in range(N):
	for add in range(1, d):
		if i + add < N:
			lessD[i + add] += lessD[i]
anyD = [0] * N
anyD[0] = 1
for i in range(N):
	for add in range(1, k + 1):
		if i + add < N:
			anyD[i + add] += anyD[i]
res = 0
for i in range(d, k + 1):
	for j in range(0, n - i + 1):
		k = n - i - j
		res = (res + lessD[j] * anyD[k]) % M
print(res)
