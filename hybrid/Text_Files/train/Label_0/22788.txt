#include <bits/stdc++.h>
const int N = 50005, mo = 1e8 + 7;
int n;
char s[N];
int power[N], sum[N];
bool del[N];
inline int calc(int l, int r) {
  return ((sum[l] - (long long)power[r - l] * sum[r]) % mo + mo) % mo;
}
int main() {
  scanf("%s", s);
  n = strlen(s);
  for (int i = 0; i < n; i++) s[i] -= 'a';
  sum[n] = 0;
  for (int i = n - 1; i >= 0; i--) sum[i] = (s[i] + 26LL * sum[i + 1]) % mo;
  power[0] = 1;
  for (int i = 1; i <= n; i++) power[i] = power[i - 1] * 26LL % mo;
  for (int len = 1; len <= n; len++) {
    bool flag = 0;
    for (int i = len; i + len <= n; i += len) {
      int l = 0, r = len + 1;
      while (r - l > 1) {
        int mid = l + (r - l >> 1);
        if (!del[i - mid] && calc(i - mid, i) == calc(i + len - mid, i + len))
          l = mid;
        else
          r = mid;
      }
      if (i + len + len - l > n ||
          calc(i, i + len - l) != calc(i + len, i + len + len - l))
        continue;
      flag = 1;
      for (int j = i - l; j < i + len - l; j++) del[j] = 1;
    }
    if (!flag) continue;
    for (int i = 0, j = 0;; i++) {
      for (; j < n && del[j]; j++) del[j] = 0;
      if (j == n) {
        n = i;
        break;
      }
      s[i] = s[j++];
    }
    sum[n] = 0;
    for (int i = n - 1; i >= 0; i--) sum[i] = (s[i] + 26LL * sum[i + 1]) % mo;
  }
  for (int i = 0; i < n; i++) putchar('a' + s[i]);
  puts("");
}
