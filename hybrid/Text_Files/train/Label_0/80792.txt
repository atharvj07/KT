n = int(input())
p = n // 3
a = b = p
c = n - 2 * p
if not p % 3:
	a -= 1
	b += 1
	if not c % 3:
		c -= 1
		b += 1
	print(' '.join(map(str, [a, b, c])))
elif not c % 3:
	c -= 1
	b += 1
	if not b % 3:
		b += 1
		a -= 1
	print(' '.join(map(str, [a, b, c])))
else:
	print(' '.join(map(str, [a, b, c])))
