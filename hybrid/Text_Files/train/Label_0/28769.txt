template<typename T>T read()
{
	T ans=0,f=1;
	char ch=getchar();
	while(!isdigit(ch)&&ch!='-')ch=getchar();
	if(ch=='-')f=-1,ch=getchar();
	while(isdigit(ch))ans=(ans<<3)+(ans<<1)+(ch-'0'),ch=getchar();
	return ans*f;
}
template<typename T>void write(T x,char y)
{
	if(x==0)
	{
		putchar('0'),putchar(y);
		return;
	}
	if(x<0)
	{
		putchar('-');
		x=-x;
	}
	static char wr[20];
	int top=0;
	for(;x;x/=10)wr[++top]=x%10+'0';
	while(top)putchar(wr[top--]);
	putchar(y);
}

char s[N];
int n;
const int mo=998244353;
int dp[N][3][3][2];
void add(int &x,int y){x+=y;x-=(x>=mo?mo:0);}
void input()
{
	scanf("%s",s+1);
	n=strlen(s+1);
}
void check()
{
	int flag=1;
	For(i,2,n)if(s[i-1]!=s[i])flag=0;
	if(flag){puts("1");exit(0);}
	if(n<=3)
	{
		if(n==2)puts("2");
		else if(s[1]!=s[2]&&s[2]!=s[3]&&s[1]!=s[3])puts("3");
		else if(s[1]==s[3])puts("7");
		else puts("6");
		exit(0);
	}
}
void work()
{
	check();
	int d=0;
	For(i,1,n)d=(d+s[i]-'a')%3;
	For(i,0,2)dp[1][i][i][0]=1;
	int ans=1;
	For(i,2,n)if(s[i-1]==s[i])ans=0;
	For(i,2,n)For(j,0,2)For(a,0,2)For(l,0,1)if(dp[i-1][j][a][l])
	{
		For(b,0,2)add(dp[i][(j+b)%3][b][l|(a==b)],dp[i-1][j][a][l]);
	}
	add(ans,dp[n][d][0][1]);
	add(ans,dp[n][d][1][1]);
	add(ans,dp[n][d][2][1]);
	write(ans,'\n');
}
int main()
{

	input();
	work();
	return 0;
