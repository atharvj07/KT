from bisect import bisect_right
(n, k, l) = list(map(int, input().split()))
a = list(map(int, input().split()))
a.sort()
if k == 1:
	if a[-1] - a[0] <= l:
		print(sum(a))
	else:
		print(0)
elif a[n - 1] - a[0] > l:
	print(0)
elif a[0] + l > a[-1]:
	print(sum(a[::k]))
else:
	ans = 0
	i = bisect_right(a, a[0] + l)
	b = a[:i]
	c = len(a[i:])
	while b and c:
		if c >= k - 1:
			c -= k - 1
			ans += b[-1]
			b = b[:-1]
		else:
			ans += b[-(k - c)]
			b = b[:-(k - c)]
			c = 0
	ans += sum(b[::k])
	print(ans)
