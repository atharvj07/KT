ans = []

for _ in range(gil()[0]):
	res = gl()
	if len(res) == 1:
		if h == []:
			ans.append('insert 1')
		ans.append(res[0])
		if h:
			heappop(h)
	elif res[0] == 'insert':
		ans.append(' '.join(res))
		heappush(h, int(res[1]))
	else:

		while h and h[0] < val:
			ans.append('removeMin')
			heappop(h)

			ans.append('insert ' + res[1])
