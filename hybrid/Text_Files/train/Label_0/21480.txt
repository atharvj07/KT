#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long LL;

#define MOD 1000000007

struct BIT{
	vector<int> b;

	BIT(int n): b(n + 1) {}

	int sum(int i){
		int s = 0;
		while(i > 0){
			s += b[i];
			i -= i & -i;
		}
		return s;
	}

	void add(int i, int x){
		while(i < b.size()){
			b[i] += x;
			i += i & -i;
		}
	}
};

int fact[100001];

int main(){
	fact[0] = 1;
	for(LL i = 1; i <= 100000; ++i){
		fact[i] = fact[i-1] * i % MOD;
	}

	int n, r, s, t;
	while(scanf("%d%d", &n, &r) == 2){
		vector<int> p(n);
		for(int i = 0; i < n; ++i){
			p[i] = i + 1;
		}
		for(int i = 0; i < r; ++i){
			scanf("%d%d", &s, &t);
			swap(p[s-1], p[t-1]);
		}

		BIT bit(n);
		int ans = 0;
		for(int i = 0; i < n; ++i){
			LL s = bit.sum(p[n - 1 - i]);
			ans = (ans + s * fact[i]) % MOD;
			bit.add(p[n - 1 - i], 1);
		}
		printf("%d\n", ans);
	}
}
