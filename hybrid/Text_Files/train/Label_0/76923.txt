a = int(input())
z = list(map(int, input().split()))
ans = [-1 for i in range(len(z))]
from collections import *
al = defaultdict(int)
for i in range(1, len(z)):
	if z[i] != z[i - 1]:
		ans[i] = z[i - 1]
for i in range(len(z)):
	al[z[i]] = 1
temp = []
for i in range(len(z) + 1):
	if al[i] == 0:
		temp.append(i)
c1 = 0
for i in range(len(ans)):
	if ans[i] == -1:
		ans[i] = temp[c1]
		c1 += 1
print(*ans)
