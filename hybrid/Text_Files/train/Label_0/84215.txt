import math
t = int(input())
for testcase in range(t):
	A = input()
	if A == '01' or A == '10':
		print('DA')
	else:
		ones = 0
		zeros = 0
		for i in range(len(A)):
			if A[i] == '0':
				zeros = zeros + 1
			else:
				ones = ones + 1
		p = min(zeros, ones)
		if p == 0:
			print('NET')
		elif p % 2 == 0:
			print('NET')
		else:
			print('DA')
