from copy import copy
(k, n) = list(map(int, input().split()))
temp = list(map(int, input().split()))
l = list(map(int, input().split()))
t = [temp[0]]
s = temp[0]
for i in range(1, k):
	s += temp[i]
	t.append(s)
temp = t
temp.sort()
l.sort()
i = 0
while i < k - 1:
	if temp[i] == temp[i + 1]:
		del temp[i]
		k -= 1
		i -= 1
	i += 1

def f(index):
	global add
	add = l[0] - temp[index]
	cop = copy(temp)
	for i in range(k):
		cop[i] += add
	found = 0
	for i in range(index, k):
		if found + 1 == n:
			return True
		if cop[i] == l[found + 1]:
			found += 1
	if found + 1 == n:
		return True
	return False
out = 0
for i in range(0, k - n + 1):
	if f(i):
		out += 1
print(out)
