vector<int32_t> readI(int size) {
        vector<int32_t> result;
        for (int i = 0; i < size; ++i) {
                int32_t x;
                assert(1 == fscanf(inp, "%i", &x));
                result.push_back(x);
        }
        return result;
}

int readI() {

        assert(1 == fscanf(inp, "%i", &x));

}


template<typename T>
T divup(T a, T b) {
        if (b < 0) { a = -a; b = -b; } 
        if (a < 0) return -((-a) / b);
        if ((a % b) == 0) return a / b; return 1 + a / b;
}
