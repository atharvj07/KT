def binarySearch(s, n, ind):
	v = 0
	x = (s >> 1) + 1
	if x < ind:
		ind -= x
		s -= x
		n -= 1
		v = binarySearch(s, n, ind)
	elif x > ind:
		s -= x
		n -= 1
		v = binarySearch(s, n, ind)
	else:
		v = n
	return v
inp = input().split()
n = int(inp[0])
ind = int(inp[1])
s = 0
for i in range(n):
	s = 2 * s + 1
x = binarySearch(s, n, ind)
print(x)
