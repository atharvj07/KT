import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map.Entry;
import java.util.StringTokenizer;

public class c {
	

	public static void main(String args[]) throws IOException {
		FastScanner in = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		//BEGIN HERE
		int n = in.nextInt();
		HashMap<String, Integer> names = new HashMap<>();
		ArrayList<HashSet<String>> numbers = new ArrayList<>();
		int idx =0;
		for(int i = 0; i < n; i++) {
			String s = in.next();
			Integer id = names.get(s);
			if(id == null) {
				names.put(s, idx++);
				id = idx-1;
				numbers.add(new HashSet<String>());
			}
			
			int m = in.nextInt();
			for(int j = 0; j < m; j++) {
				String num = in.next();
				String rem = null;
				boolean add = true;
				for(String pn: numbers.get(id)) {
					if(pn.compareTo(num) == 0) {
						add = false;
						break;
					}
					if(pn.endsWith(num)) {
						add = false;
						break;
					} else if(num.endsWith(pn)) {
						rem = pn;
						break;
					}
				}
				if(add) {
					numbers.get(id).add(num);
				} if (rem != null) {
					numbers.get(id).remove(rem);
				}
			}
			
			
		}
		System.out.println(idx);
		for(Entry<String, Integer> e : names.entrySet()) {
			System.out.print(e.getKey() + " ");
			System.out.print(numbers.get(e.getValue()).size() + " ");
			for(String ss : numbers.get(e.getValue())) {
				System.out.print(ss + " ");
			}
			System.out.println();
		}
		out.close();
	}

static class FastScanner {
	BufferedReader br;
	StringTokenizer st;

	public FastScanner(InputStream i) {
		br = new BufferedReader(new InputStreamReader(i));
		st = null;
	}

	public String next() throws IOException {
		while (st == null || !st.hasMoreTokens())
			st = new StringTokenizer(br.readLine());
		return st.nextToken();
	}

	public String nextLine() throws IOException {
		if (st == null) {
			st = new StringTokenizer(br.readLine());
		}
		String line = st.nextToken("\n");
		st = null;
		return line;
	}

	public int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	public long nextLong() throws IOException {
		return Long.parseLong(next());
	}

	public double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}
}}
