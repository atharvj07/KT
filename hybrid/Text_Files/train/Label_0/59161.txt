
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.HashMap;
import java.util.PriorityQueue;
import java.util.StringTokenizer;
import java.util.TreeSet;

/*

 */

/**
 *
 * @author Andy Phan
 */
public class c {
    public static void main(String[] args) {
        FS in = new FS(System.in);
        PrintWriter out = new PrintWriter(System.out);
        
        int n = in.nextInt();
        TreeSet<Integer> compX = new TreeSet<>();
        TreeSet<Integer> compY = new TreeSet<>();
        
        compX.add(1987654321);
        compY.add(1987654321);
        
        Point[] arr = new Point[n];
        for(int i = 0; i < n; i++) {
            arr[i] = new Point(in.nextInt(), in.nextInt());
            compX.add(arr[i].x);
            compY.add(arr[i].y);
        }
        
        HashMap<Integer, Integer> xs = new HashMap<>();
        int curr = 0;
        int prev = -1;
        while(compX.ceiling(prev+1) != null) {
            xs.put(prev = compX.ceiling(prev+1), curr++);
        }
        
        HashMap<Integer, Integer> ys = new HashMap<>();
        curr = 0;
        prev = -1;
        while(compY.ceiling(prev+1) != null) {
            ys.put(prev = compY.ceiling(prev+1), curr++);
        }
        
        TreeSet<Integer>[] nextX = new TreeSet[ys.size()];
        
        for(int i = 0; i < ys.size(); i++) {
            nextX[i] = new TreeSet();
            nextX[i].add(1987654321);
        }
        for(int i = 0; i < n; i++) {
            arr[i] = new Point(xs.get(arr[i].x), ys.get(arr[i].y));
            nextX[arr[i].y].add(arr[i].x);
        }
        
        Node nums = new Node(0, xs.size());
        
        PriorityQueue<Point> q = new PriorityQueue<Point>((Point a, Point b) -> a.y == b.y ? a.x-b.x : b.y-a.y);
        
        for(int i = 0; i < n; i++) q.add(arr[i]);
        
        long tot = 0;
        while(!q.isEmpty()) {
            Point cur = q.poll();
            nums.upd(cur.x, 1);
            tot += (long)nums.num(0, cur.x+1)*nums.num(cur.x, nextX[cur.y].ceiling(cur.x+1));
        }
        
        System.out.println(tot);
        
        out.close();
    }
    
    static class Node {
        int l, r, num;
        Node left, right;
        
        public Node(int le, int ri) {
            l = le;
            r = ri;
            if(l == r-1) return;
            int mid = (l+r)>>1;
            left = new Node(l, mid);
            right = new Node(mid, r);
        }
        
        public void upd(int ind, int val) {
            if(ind < l || ind >= r) return;
            if(left == null) {
                num = Math.max(num, val);
                return;
            }
            left.upd(ind, val);
            right.upd(ind, val);
            num = left.num+right.num;
        }
        
        public int num(int le, int ri) {
            if(le <= l && ri >= r) return num;
            if(le >= r || ri <= l) return 0;
            return left.num(le, ri)+right.num(le, ri);
        }
    }
    
    static class Point {
        int x, y;
        public Point(int X, int Y) {
            x = X;
            y = Y;
        }
    }
    
    static class FS {

        BufferedReader in;
        StringTokenizer token;
        
        public FS(InputStream str) {
            in = new BufferedReader(new InputStreamReader(str));
        }
        
        public String next() {
            if (token == null || !token.hasMoreElements()) {
                try {
                    token = new StringTokenizer(in.readLine());
                } catch (IOException ex) {
                }
                return next();
            }
            return token.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(next());
        }
    }
}
