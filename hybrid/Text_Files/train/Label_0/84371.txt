(n, m) = map(int, input().split())
t = [None] * (n + 1)
for i in range(1, n + 1):
	t[i] = [0] + [int(i) for i in input().split()]
d = [[0] * (m + 1) for i in range(n + 1)]
for i in range(1, n + 1):
	for j in range(1, m + 1):
		d[i][j] = max(d[i - 1][j], d[i][j - 1]) + t[i][j]
print(*(d[i][m] for i in range(1, n + 1)))
