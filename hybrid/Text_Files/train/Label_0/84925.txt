(p, q, l, r) = map(int, input().strip().split())
lz = [0 for __ in range(1001)]
lx = lz.copy()
res = 0
for i in range(p):
	(a, b) = map(int, input().split())
	for j in range(a, b + 1):
		lz[j] = 1
for i in range(q):
	(a, b) = map(int, input().split())
	for j in range(a, b + 1):
		lx[j] = 1
for i in range(l, r + 1):
	for ii in range(len(lz) + 1):
		if ii + i < 1001:
			if lx[ii] + lz[ii + i] == 2:
				res += 1
				break
print(res)
