import java.util.*;
import java.io.*;
import java.lang.*;
import java.math.*;
public class B {
    public static void main(String[] args) throws Exception {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        // int n = Integer.parseInt(bf.readLine());
        StringTokenizer st = new StringTokenizer(bf.readLine());
        // int[] a = new int[n]; for(int i=0; i<n; i++) a[i] = Integer.parseInt(st.nextToken());
        int h = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int s = Integer.parseInt(st.nextToken());
        int t1 = Integer.parseInt(st.nextToken());
        int t2 = Integer.parseInt(st.nextToken());
        int betweencount = 0;
        if(((t1 <= h+m/60.0+s/3600.0) && (h+m/60.0+s/3600.0<= t2)) || ((t2 <=h+m/60.0+s/3600.0) && (h+m/60.0+s/3600.0 <= t1))) betweencount++;
        if(((t1 <= m/5.0+s/300.0) && (m/5.0+s/300.0 <= t2)) || ((t2 <= m/5.0+s/300.0) && (m/5.0+s/300.0 <= t1))) betweencount++;
        if(((t1 <= s/5.0) && (s/5.0 <= t2)) || ((t2 <= s/5.0) && (s/5.0 <= t1))) betweencount++;
        if(betweencount % 3 == 0) out.println("YES");
        else out.println("NO");
        out.close(); System.exit(0);
    }
}
