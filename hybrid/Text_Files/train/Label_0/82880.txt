from collections import Counter
a = set('aeiouy')

def cnt(s):
	return sum((x[1] for x in filter(lambda y: y[0] in a, Counter(s).items())))
n = int(input())
b = [int(x) for x in input().split()]
c = [sum((cnt(s) for s in input().split())) for i in range(n)]
if b == c:
	print('YES')
else:
	print('NO')
