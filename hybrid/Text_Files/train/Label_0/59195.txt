import java.util.*;
import java.io.*;
public class NewYearAndFindingRoots
{
	public static Scanner in;
	public static PrintWriter out;
	public static int[][] adj;
	public static int height;
	
	public static void main(String[] args) throws Exception
	{
		in = new Scanner(System.in);
		out = new PrintWriter(System.out);
		
		int t = in.nextInt();
		outer:
		for(int x = 0; x < t; x++)
		{
			int h = in.nextInt();
			
			adj = new int[1 << h][];
			
			if(query(1) == 2)
			{
				guess(1);
			}
			else
			{
				int parent = getHeight(1);
				int current = 1;
				
				query(parent);
				
				while(height < 4 && height < h - 1)
				{
					int prev = current;
					int node = parent;
					int first = -1;
					for(int y = 0; y < height; y++)
					{
						int next = traverse(node, prev);
						if(y == 0)
						{
							first = next;
						}
						
						prev = node;
						node = next;
					}
					
					if(adj[node].length == 1)
					{
						int nextParent = other(parent, current, first);
						current = parent;
						parent = nextParent;
					}
					else
					{
						current = parent;
						parent = first;
					}
					
					height++;
					query(parent);
				}
				
				if(height == h - 1)
				{
					guess(parent);
				}
				else if(height == 5)
				{
					for(int next : adj[parent])
					{
						if(query(next) == 2)
						{
							guess(next);
							break;
						}
					}
				}
				else
				{
					int node = parent;
					
					int next1 = other(node, current);
					int next2 = other(node, current, next1);
					
					if(query(next1) == 2)
					{
						guess(next1);
						continue;
					}
					
					for(int next : adj[next1])
					{
						if(next != node && query(next) == 2)
						{
							guess(next);
							continue outer;
						}
					}
					
					if(query(next2) == 2)
					{
						guess(next2);
						continue;
					}
					
					int other = other(next2, node);
					if(query(other) == 2)
					{
						guess(other);
					}
					else
					{
						guess(other(next2, node, other));
					}
				}
			}
		}
	}
	
	public static int query(int index)
	{
		out.println("? " + index);
		out.flush();
		
		int k = in.nextInt();
		if(k == 0)
		{
			System.exit(0);
		}
		
		adj[index] = new int[k];
		for(int i = 0; i < k; i++)
		{
			adj[index][i] = in.nextInt();
		}
		
		return k;
	}
	
	public static void guess(int index)
	{
		out.println("! " + index);
		out.flush();
	}
	
	public static int other(int node, int i1)
	{
		for(int next : adj[node])
		{
			if(next != i1)
			{
				return next;
			}
		}
		
		return -1;
	}
	
	public static int other(int node, int i1, int i2)
	{
		for(int next : adj[node])
		{
			if(next != i1 && next != i2)
			{
				return next;
			}
		}
		
		return -1;
	}
	
	public static int getHeight(int node)
	{
		if(adj[node].length == 1)
		{
			height = 1;
			return adj[node][0];
		}
		else
		{
			height = 1;
			int parent1 = -1;
			int parent2 = adj[node][0];
			int node1 = node;
			int node2 = node;
			while(true)
			{
				int next1 = traverse(node1, parent1);
				int next2 = traverse(node2, parent2);
				height++;
				
				int k1 = adj[next1].length;
				int k2 = adj[next2].length;
				
				if(k1 == 1 && k2 == 1)
				{
					return adj[node][2];
				}
				else if(k1 == 1)
				{
					return adj[node][1];
				}
				else if(k2 == 1)
				{
					return adj[node][0];
				}
				
				parent1 = node1;
				node1 = next1;
				parent2 = node2;
				node2 = next2;
			}
		}
	}
	
	public static int traverse(int node, int prev)
	{
		for(int i = 0; i < adj[node].length; i++)
		{
			if(adj[node][i] != prev)
			{
				query(adj[node][i]);
				return adj[node][i];
			}
		}
		
		return -1;
	}
}
