n = int(input())
x = list(map(int, input().split()))
sum1 = 0
sum2 = 0
equal = True
i = 0
while i < 2 * n - 1:
	if x[i] != x[i + 1] and equal:
		equal = False
	if i < n:
		sum1 += x[i]
	else:
		sum2 += x[i]
	i += 1
sum2 += x[i]
if equal:
	print(-1)
else:
	i = n
	j = 0
	if sum1 == sum2:
		while j < n:
			if x[j] != x[i]:
				tmp = x[j]
				x[j] = x[i]
				x[i] = tmp
				break
			elif i == 2 * n - 1:
				i = n
				j += 1
			else:
				i += 1
	for i in x:
		print(i, end=' ')
