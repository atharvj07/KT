#include <bits/stdc++.h>
using namespace std;
int main(){
  int N,M,c=0,x=0 ;cin>>N>>M;
  vector<pair<int,int>>V(M);
  for(int i=0;i<M;i++)
    cin>>V.at(i).first>>V.at(i).second;
  sort(V.begin(),V.end());
  while(x<M){
    int y=x,MIN=V.at(x).second;
    while(y<M&&MIN>V.at(y).first){
      MIN=min(MIN,V.at(y++).second);
    }
    x=y;c++;
  }
  cout<<c<<endl;
}
