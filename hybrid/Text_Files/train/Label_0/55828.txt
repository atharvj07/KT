import java.util.Scanner;

public class Main {
   public static void main(String[] args) {
	  Scanner sc=new Scanner(System.in);
	  int n=sc.nextInt();
	  int[][] a=new int[n][n];
	  for(int i=0;i<n;i++) {
		  for(int j=0;j<n;j++) {
			  a[i][j]=sc.nextInt();
		  }
	  }
	  //主对角线
	  int k=0,kk=0;
	 int i=0,j=0,ii=n-1;
	  while(i!=n) {
		k+=a[i][j];
		kk+=a[ii][j];
		j++;
		i++;
		ii--;
	  }
	  k+=kk;
	  kk=0;
	  i=n/2;
	  for(int w=0;w<n;w++) {
		  k+=a[i][w];
		  k+=a[w][i];
	  }
	 System.out.println(k-(3*a[i][i]));
   }
}
