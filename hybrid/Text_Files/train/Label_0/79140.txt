import sys
fin = sys.stdin
fout = sys.stdout
n = int(fin.readline())
array = list(map(int, fin.readline().split()))
unimodal = True
state = 0
for i in range(1, n):
	if array[i - 1] < array[i]:
		if state == 1 or state == 2:
			unimodal = False
			break
	elif array[i - 1] == array[i]:
		if state == 0 or state == 1:
			state = 1
		else:
			unimodal = False
			break
	else:
		state = 2
if unimodal:
	fout.write('YES')
else:
	fout.write('NO')
fout.close()
