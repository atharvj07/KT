import java.io.*;
import java.util.*;

public class CF1243E extends PrintWriter {
	CF1243E() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1243E o = new CF1243E(); o.main(); o.flush();
	}

	void main() {
		int k = sc.nextInt();
		HashMap<Integer, Integer> mp = new HashMap<>();
		long[] ss = new long[k]; long sum = 0;
		int[][] aa = new int[k][];
		for (int h = 0; h < k; h++) {
			int n = sc.nextInt();
			aa[h] = new int[n];
			for (int i = 0; i < n; i++) {
				int a = sc.nextInt();
				mp.put(a, h | i << 4);
				ss[h] += aa[h][i] = a;
			}
			sum += ss[h];
		}
		if (sum % k != 0) {
			println("No");
			return;
		}
		sum /= k;
		int[][] hh = new int[k][];
		int[][] ii = new int[k][];
		int[][] tt = new int[k][];
		int mask = 0;
		for (int h = 0; h < k; h++) {
			int n = aa[h].length;
			hh[h] = new int[n]; Arrays.fill(hh[h], -1);
			ii[h] = new int[n];
			tt[h] = new int[n];
			long d = sum - ss[h];
			if (d == 0)
				continue;
			mask |= 1 << h;
			for (int i = 0; i < n; i++) {
				int a = aa[h][i]; long b = a + d;
				int hi = Math.abs(b) > 1000000000 ? -1 : mp.getOrDefault((int) b, -1);
				if (hi != -1) {
					int h_ = hi & 15, i_ = hi >> 4;
					if (h_ != h) {
						hh[h][i] = h_;
						ii[h][i] = i_;
					}
				}
			}
		}
		int[] hh_ = new int[1 << k]; Arrays.fill(hh_, -1);
		int[] ii_ = new int[1 << k];
		int time = 1;
		for (int h = 0; h < k; h++) {
			if ((mask & 1 << h) == 0)
				continue;
			int n = aa[h].length;
			for (int i = 0; i < n; i++, time++) {
				int h1 = h, i1 = i;
				while (true) {
					int h2 = hh[h1][i1], i2 = ii[h1][i1];
					if (h2 == -1)
						break;
					int t = tt[h1][i1];
					if (t == 0) {
						tt[h1][i1] = time;
						h1 = h2; i1 = i2;
						continue;
					}
					if (t == time) {
						int h_ = h1, i_ = i1, m = 0;
						while ((m & 1 << h1) == 0) {
							m |= 1 << h1;
							h2 = hh[h1][i1]; i2 = ii[h1][i1];
							h1 = h2; i1 = i2;
						}
						if (h_ == h1 && i_ == i1) {
							hh_[m] = h1;
							ii_[m] = i1;
						}
					}
					break;
				}
			}
		}
		boolean[] dp = new boolean[1 << k]; dp[0] = true;
		int[] mm = new int[1 << k];
		for (int m = 0; m < 1 << k; m++)
			if (dp[m]) {
				int m_ = (1 << k) - 1 ^ m;
				int m1 = 0;
				do
					if (hh_[m1] != -1) {
						dp[m | m1] = true;
						mm[m | m1] = m1;
					}
				while ((m1 = m1 - m_ & m_) != 0);
			}
		if (!dp[mask]) {
			println("No");
			return;
		}
		println("Yes");
		int[] cc = new int[k];
		int[] pp = new int[k];
		for (int m = mask; m != 0; m ^= mm[m]) {
			int m1 = mm[m], h1 = hh_[m1], i1 = ii_[m1];
			while ((m1 & 1 << h1) != 0) {
				m1 ^= 1 << h1;
				int h2 = hh[h1][i1], i2 = ii[h1][i1];
				cc[h2] = aa[h2][i2];
				pp[h2] = h1;
				h1 = h2; i1 = i2;
			}
		}
		for (int h = 0; h < k; h++) {
			int c, p;
			if ((mask & 1 << h) != 0) {
				c = cc[h]; p = pp[h];
			} else {
				c = aa[h][0]; p = h;
			}
			println(c + " " + (p + 1));
		}
	}
}
