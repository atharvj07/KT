#include <bits/stdc++.h>
using namespace std;
struct ab {
  int tag, sum, minn, maxx, minl, minr, maxl, maxr, minnl, minnr, maxxl, maxxr,
      minlr, minrl, maxlr, maxrl;
  ab() {}
  ab(int x, int y) {
    sum = minn = maxx = minl = minr = maxl = maxr = x;
    minnl = minnr = maxxl = maxxr = minlr = minrl = maxlr = maxrl = y;
    tag = 0;
  }
} t[500005];
int l[505], r[505];
ab operator+(ab x, ab y) {
  ab z;
  z.sum = x.sum + y.sum;
  (x.maxl > x.sum + y.maxl) ? (z.maxl = x.maxl, z.maxlr = x.maxlr)
                            : (z.maxl = x.sum + y.maxl, z.maxlr = y.maxlr);
  (y.maxr > y.sum + x.maxr) ? (z.maxr = y.maxr, z.maxrl = y.maxrl)
                            : (z.maxr = y.sum + x.maxr, z.maxrl = x.maxrl);
  (x.minl < x.sum + y.minl) ? (z.minl = x.minl, z.minlr = x.minlr)
                            : (z.minl = x.sum + y.minl, z.minlr = y.minlr);
  (y.minr < y.sum + x.minr) ? (z.minr = y.minr, z.minrl = y.minrl)
                            : (z.minr = y.sum + x.minr, z.minrl = x.minrl);
  (x.maxx > y.maxx) ? (z.maxx = x.maxx, z.maxxl = x.maxxl, z.maxxr = x.maxxr)
                    : (z.maxx = y.maxx, z.maxxl = y.maxxl, z.maxxr = y.maxxr);
  (x.minn < y.minn) ? (z.minn = x.minn, z.minnl = x.minnl, z.minnr = x.minnr)
                    : (z.minn = y.minn, z.minnl = y.minnl, z.minnr = y.minnr);
  if (x.maxr + y.maxl > z.maxx)
    z.maxx = x.maxr + y.maxl, z.maxxl = x.maxrl, z.maxxr = y.maxlr;
  if (x.minr + y.minl < z.minn)
    z.minn = x.minr + y.minl, z.minnl = x.minrl, z.minnr = y.minlr;
  z.tag = 0;
  return z;
}
void work(int x) {
  swap(t[x].maxx, t[x].minn);
  swap(t[x].maxl, t[x].minl);
  swap(t[x].maxr, t[x].minr);
  swap(t[x].maxlr, t[x].minlr);
  swap(t[x].maxrl, t[x].minrl);
  swap(t[x].maxxl, t[x].minnl);
  swap(t[x].maxxr, t[x].minnr);
  t[x].tag ^= 1;
  t[x].sum *= -1;
  t[x].minn *= -1;
  t[x].maxx *= -1;
  t[x].minl *= -1;
  t[x].minr *= -1;
  t[x].maxl *= -1;
  t[x].maxr *= -1;
}
void pushup(int x) { t[x] = t[x << 1] + t[x << 1 | 1]; }
void pushdown(int x) {
  if (t[x].tag) {
    work(x << 1);
    work(x << 1 | 1);
    t[x].tag = 0;
  }
}
void build(int l, int r, int x) {
  if (l == r) {
    int y;
    scanf("%d", &y);
    t[x] = ab(y, l);
  } else {
    int mid = (l + r) >> 1;
    build(l, mid, x << 1);
    build(mid + 1, r, x << 1 | 1);
    pushup(x);
  }
}
void add(int y, int z, int l, int r, int x) {
  if (l == r) {
    t[x] = ab(z, l);
    return;
  }
  pushdown(x);
  int mid = (l + r) >> 1;
  if (mid >= y)
    add(y, z, l, mid, x << 1);
  else
    add(y, z, mid + 1, r, x << 1 | 1);
  pushup(x);
}
ab que(int L, int R, int l, int r, int x) {
  if (l >= L && r <= R) {
    return t[x];
  }
  pushdown(x);
  int mid = (l + r) >> 1;
  if (mid >= R) return que(L, R, l, mid, x << 1);
  if (mid < L) return que(L, R, mid + 1, r, x << 1 | 1);
  return que(L, R, l, mid, x << 1) + que(L, R, mid + 1, r, x << 1 | 1);
}
void flp(int L, int R, int l, int r, int x) {
  if (l >= L && r <= R) return work(x);
  int mid = (l + r) >> 1;
  pushdown(x);
  if (mid >= L) flp(L, R, l, mid, x << 1);
  if (mid < R) flp(L, R, mid + 1, r, x << 1 | 1);
  pushup(x);
}
int main() {
  int n, m;
  scanf("%d", &n);
  build(1, n, 1);
  scanf("%d", &m);
  while (m--) {
    int k, x, y, z;
    scanf("%d%d%d", &k, &x, &y);
    if (k == 0)
      add(x, y, 1, n, 1);
    else {
      scanf("%d", &z);
      int ans = 0, tot = 0;
      while (z--) {
        ab h = que(x, y, 1, n, 1);
        if (h.maxx <= 0) break;
        ans += h.maxx;
        l[++tot] = h.maxxl, r[tot] = h.maxxr;
        flp(l[tot], r[tot], 1, n, 1);
      }
      while (tot) {
        flp(l[tot], r[tot], 1, n, 1);
        tot--;
      }
      printf("%d\n", ans);
    }
  }
  return 0;
}
