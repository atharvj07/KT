t = int(input())
for i in range(0, t):
	n = int(input())
	arr = list(map(lambda x: int(x), input().split()))
	dict_num = {}
	maxValue = 0
	for ele in arr:
		if ele not in dict_num:
			dict_num[ele] = 1
		else:
			dict_num[ele] += 1
		maxValue = max(maxValue, dict_num[ele])
	dict_len = len(dict_num)
	if dict_len > maxValue:
		print(min(dict_len - 1, maxValue))
	else:
		print(min(dict_len, maxValue - 1))
