import bisect
(n, m) = map(int, input().split())
l = list(map(int, input().split()))
l = l + l
pre = [0, l[0]]
su = [0, l[0] * (l[0] + 1) // 2]
for i in range(2, 2 * n):
	pre.append(pre[i - 1] + l[i - 1])
	su.append(su[i - 1] + l[i - 1] * (l[i - 1] + 1) // 2)
ans = 0
for i in range(1, n + 1):
	le = m + pre[i - 1]
	ind = bisect.bisect_left(pre, le)
	di = pre[ind] - pre[i - 1] - m
	cans = su[ind] - su[i - 1] - di * (di + 1) // 2
	ans = max(ans, cans)
print(ans)
