template<class T> inline void amin(T &x, const T &y) { if (y<x) x=y; }
template<class T> inline void amax(T &x, const T &y) { if (x<y) x=y; }
template<class Iter> void rprintf(const char *fmt, Iter begin, Iter end) {
    for (bool sp=0; begin!=end; ++begin) { if (sp) putchar(' '); else sp = true; printf(fmt, *begin); }
    putchar('\n');
}

int N, K;
string S[2011];
char buf[100111];

using Bits = bitset<10011>;
Bits vis;
Bits dp[2011];

const int INF = 1<<29;
int mi[10011];
string ans;

void MAIN() {
    scanf("%d%d", &N, &K);
    REP (i, N) {
	scanf("%s", buf);
	S[i] = buf;
    }

    vis.set(K);
    for (int i=N; i--;) {
	REP (j, K-(int)S[i].size()+1) if (vis.test(j+S[i].size())) {
	    vis.set(j);
	    dp[i].set(j);
	}
    }

    REP (k, K+1) mi[k] = INF;
    mi[0] = 0;

    ans = string(K, '~');

    REP (k, K) {
	for (int i=mi[k]; i<N; i++) if (dp[i].test(k)) {
	    int p = -1;
	    REP (j, S[i].size()) {
		if (S[i][j] != ans[k+j]) {
		    p = j;
		    break;
		}
	    }
	    if (p != -1) {
		if (S[i][p] < ans[k+p]) {
		    for (int j=k+p; j<K; j++) {

			if (j-k < (int)S[i].size()) {
			    ans[j] = S[i][j-k];
			} else {
			    if (ans[j] == '~') break;
			    ans[j] = '~';
			}
			mi[j+1] = INF;
		    }
		} else {
		    continue;
		}
	    }

	    amin(mi[k+(int)S[i].size()], i+1);
	}
    }

    puts(ans.c_str());
}

int main() {
    int TC = 1;
//    scanf("%d", &TC);
    REP (tc, TC) MAIN();
    return 0;
}
