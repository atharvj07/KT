namespace templates {
ll modpow(ll x, ll b) {
  ll res = 1;
  while (b) {
    if (b & 1) res = res * x % MOD;
    x = x * x % MOD;
    b >>= 1;
  }
  return res;
}
ll modinv(ll x) { return modpow(x, MOD - 2); }

  vvector<int> line(q, vector<int>(3));
  for (auto &i : (line))
    for (auto &j : (i)) j = in();
  for (auto &i : (line)) --i[1];
  for (auto &i : (line)) --i[2];
  reverse((line).begin(), (line).end());
  function<int(int p)> calc = (function<int(int p)>)[&](int p)->int {
    for (auto &i : (line)) {
      int kind = i[0];
      int l = i[1];
      int r = i[2];
      if (l <= p and p <= r) {
        if (kind == 1) {
          if (l == p) {
            p = r;
          } else {
            p = p - 1;
          }
        } else {
          p = r - (p - l);
        }
      }
    }
    return nums[p];
  };
  vector<int> res;
  for (int _ = (m) * (strlen("") != 0); _ < (int)(strlen("") ?: (m)); ++_) {

    res.emplace_back(calc(p));
