def is_square_string(string: str):
	result = False
	length = len(string)
	if length % 2 == 0:
		length_half = int(length / 2)
		string1 = string[:length_half]
		string2 = string[length_half:]
		if string1 == string2:
			result = True
	return result
cases = int(input())
for i in range(0, cases):
	string = input()
	if is_square_string(string):
		print('YES')
	else:
		print('NO')
