import java.io.IOException;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class ProblemC {
	static final long MOD = 1000000007;
	
	public static void main(String[] args) throws IOException {
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = in.nextInt();
		long[] a = new long[n];
		for (int i = 0 ; i < n ; i++) {
			a[i] = in.nextLong();
		}
		Arrays.sort(a);
		
		long best = Long.MAX_VALUE;
		
		boolean[] isp = new boolean[1000010];
		Arrays.fill(isp, true);
		isp[1] = false;
		for (int i = 2 ; i < 1000010 ; i++) {
			if (isp[i]) {
				for (long ii = 1L * i * i ; ii < 1000010 ; ii += i) {
					isp[(int)ii] = false;
				}
			}
		}
		List<Integer> primes = new ArrayList<Integer>();
		for (int i = 1 ; i < 1000010 ; i++) {
			if (isp[i]) {
				primes.add(i);
			}
		}
		
		cnt = new long[n];
		for (int i = 0 ; i < n ; i++) {
			long nw = a[i];
			for (int p : primes) {
				while (nw % p == 0) {
					nw /= p;
					cnt[i]++;
				}
			}
			if (nw >= 2) {
				cnt[i]++;
			}
		}
		N = n;
		
		int[] parent = new int[n];
		Arrays.fill(parent, -1);
		out.println(doit(0, a, parent));
		out.flush();
	}
	
	static int N;
	static long[] cnt;

	public static long doit(int done, long[] a, int[] parent) {
		if (done == (1<<N)-1) {
			long best = 0;
			for (int i = 0 ; i < N ; i++) {
				long deg = cnt[i];
				boolean hasChild = false;
				for (int j = 0 ; j < N ; j++) {
					if (parent[j] == i) {
						deg -= cnt[j];
						hasChild = true;
					}
				}
				best += deg;
				if (hasChild || cnt[i] >= 2) {
					best++;
				}
			}
			int pcnt = 0;
			for (int j = 0 ; j < N ; j++) {
				if (parent[j] == -1) {
					pcnt++;
				}
			}
			best += (pcnt == 1) ? 0 : 1;
			//debug(pcnt,best);
			return best;
		}
		long best = Long.MAX_VALUE;
		for (int pl = N-1 ; pl >= 0 ; pl--) {
			if ((done & (1<<pl)) != 0) {
				continue;
			}
			for (int p = 0 ; p < N ; p++) {
				if ((done & (1<<p)) == 0) {
					continue;
				}
				if (a[p] % a[pl] == 0) {
					parent[pl] = p;
					long rev = a[p];
					a[p] /= a[pl];
					best = Math.min(best, doit(done | (1<<pl), a, parent));
					a[p] = rev;
					parent[pl] = -1;
				}
			}
			parent[pl] = -1;
			best = Math.min(best, doit(done | (1<<pl), a, parent));
			break;
		}
		return best;
	}

	
	public static void debug(Object... o) {
		System.err.println(Arrays.deepToString(o));
	}
}
