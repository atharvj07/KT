#include <bits/stdc++.h>

#define int long long
#define inf  1000000007
#define pb push_back
  using namespace std;


int po(int r,int k){
	if(k==1) return r;
	if(k%2) return po(r,k-1)*r%inf;
	int v=po(r,k/2);
	return v*v%inf;
}

int inv(int r){
	return po(r,inf-2);
}


int dp[2005][6005];
vector<pair<int,int>> G[2005];


void solve(){
	int n,m;
	if(!(cin>>n>>m))exit(0);
	if(n==0) exit(0);
	for(int i=0;i<n;i++)G[i].clear();
	for(int i=0;i<n;i++)for(int j=0;j<=3*n+4;j++)dp[i][j]=0;
	dp[0][0]=1;
	vector<int> nyu(n,0);
	
	for(int i=0;i<m;i++){
		int y,yy,yyy;
		cin>>y>>yy>>yyy;
		y--,yy--;
		G[y].pb({yy,yyy});
		nyu[yy]++;
	}
	
	// topological sort
	vector<int> topo(n);
	for(int i=0;i<n;i++)topo[i]=i;
	assert((int)topo.size()==n);
	
	// DP t= 0 ~ 3*n
	for(auto v:topo){
		for(int i=1;i<=3*n;i++){
				dp[v][i]+=dp[v][i-1];
				dp[v][i]%=inf;
		}
		for(auto w:G[v]){
		
			for(int i=0;i+w.second<=3*n;i++){
				dp[w.first][i+w.second]+=dp[v][i];
				dp[w.first][i+w.second]%=inf;
			}
		}
	}
	
	// 
	vector<int> frac(n+1,1);
	for(int i=1;i<=n;i++)frac[i]=frac[i-1]*i%inf;
	vector<int> frac_inv(n+1,1);
	frac_inv[n]=inv(frac[n]);
	for(int i=n;i>0;i--)frac_inv[i-1]=frac_inv[i]*i%inf;
	
	int Q;
	cin>>Q;

	while(Q--){
		
		int e,t;
		cin>>e>>t;
		e--;
		if(t<=3*n){
			cout<<dp[e][t]<<endl;
		}
		else{
			// i = 2*n ~ 3*n を使ってn次式を復元
			vector<int>bunsi_mae(n+1,0);
			bunsi_mae[0]=1;
			for(int i=1;i<=n;i++){
				bunsi_mae[i]=bunsi_mae[i-1]*(t-2*n-i+1)%inf;
			}
			vector<int>bunsi_ato(n+1,0);
			bunsi_ato[n]=1;
			for(int i=n-1;i>=0;i--){
				bunsi_ato[i]=bunsi_ato[i+1]*(t-2*n-i-1)%inf;
			}
			int bunsi_all=1;
			for(int i=2*n;i<=3*n;i++){
				bunsi_all*=t-i;
				bunsi_all%=inf;
			}
			
			int ans=0;
			for(int i=2*n;i<=3*n;i++){
				int bunsi=bunsi_mae[i-2*n]*bunsi_ato[i-2*n]%inf;
				int bunbo=frac_inv[i-2*n]*((3*n-i)%2?(inf-1)*frac_inv[3*n-i]%inf:frac_inv[3*n-i])%inf;
				ans+=dp[e][i]*bunsi%inf*(bunbo)%inf;
				ans%=inf;
			}
			cout<<ans<<endl;
		}
	}
}

 signed main(){
 	while(1) solve();
 	return 0;

  }
