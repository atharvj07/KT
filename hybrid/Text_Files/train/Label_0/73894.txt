mat = []
for i in range(3):
	mat.append([int(i) for i in input().split()])
a = mat[0][1] + mat[0][2]
b = mat[1][0] + mat[1][2]
c = mat[2][0] + mat[2][1]

def check(x):
	z = x - a
	y = x - b
	k = x - c
	return y + z + k == x
for i in range(0, 3 * 10 ** 5 + 1, 3):
	if check(i):
		ans = i
		break
mat[0][0] = ans - a
mat[1][1] = ans - b
mat[2][2] = ans - c
for i in range(3):
	print(*mat[i])
