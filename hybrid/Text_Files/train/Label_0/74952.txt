from sys import stdin, stdout
t = int(stdin.readline().strip())
for _ in range(t):
	n = int(stdin.readline().strip())
	s = stdin.readline().strip()
	ss = []
	ans = 10000000000000000
	for i in range(n):
		if s[i] not in ss:
			ss.append(s[i])
			l = 0
			r = n - 1
			num = 0
			while l < r:
				if s[l] == s[r]:
					l += 1
					r -= 1
					continue
				if s[l] == s[i] or s[r] == s[i]:
					num += 1
					if s[l] == s[i]:
						l += 1
					else:
						r -= 1
				else:
					num = 10000000000000000
					break
			ans = min(ans, num)
	if ans == 10000000000000000:
		stdout.write(str(-1) + '\n')
	else:
		stdout.write(str(ans) + '\n')
