n = int(input())
l = list(map(int, input().split()))

def fun(a):
	if len(a) <= 3:
		return len(a)
	maxE = max(a)
	if maxE == 0:
		return len(a)
	msb = 1
	while 2 * msb <= maxE:
		msb *= 2
	l1 = []
	l2 = []
	for x in a:
		if x >= msb:
			l1.append(x - msb)
		else:
			l2.append(x)
	max1 = fun(l1)
	max2 = fun(l2)
	if max1 == 0:
		return max2
	if max2 == 0:
		return max1
	return max(1 + max1, 1 + max2)
print(n - fun(l))
