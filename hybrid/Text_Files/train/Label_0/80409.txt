from math import sqrt

def prime(x):
	flag = True
	for i in range(2, int(sqrt(x)) + 1):
		if x % i == 0:
			flag = False
			break
	return flag
arr = []
for i in range(2, 25001):
	if prime(i):
		arr.append(i)
ans = []
pos1 = 0
pos2 = 0
d = 1
while d <= 10000:
	while arr[pos1] - 1 < d:
		pos1 += 1
	while arr[pos2] - arr[pos1] < d:
		pos2 += 1
	ans.append(arr[pos1] * arr[pos2])
	d += 1
t = int(input())
for i in range(t):
	n = int(input())
	print(ans[n - 1])
