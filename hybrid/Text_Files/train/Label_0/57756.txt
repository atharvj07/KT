import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.util.*;

public class Q449A {	
	BufferedReader reader;
    StringTokenizer tokenizer;
    PrintWriter out;

  String f0 = "What are you doing at the end of the world? Are you busy? Will you save us?";
  String f11 = "What are you doing while sending \"";
  String f12 = "\"? Are you busy? Will you send \"";
  String f13 = "\"?";
  long[] lengths;
    
	public void solve() throws IOException {				
    lengths = new long[100001];
    lengths[0] = f0.length();
    for (int i = 1; i <= 100000; i++) {
      if (i <= 55) {
        lengths[i] = lengths[i-1] * 2 + f11.length() + f12.length() + f13.length();
      } else {
        lengths[i] = lengths[i-1]; // not relevant
      }
    }

		int Q = nextInt();
    StringBuilder sb = new StringBuilder();
    for (int q = 0; q < Q; q++) {
      int N = nextInt();
      long K = nextLong() - 1;

      sb.append(compute(N, K));
    }
    out.println(sb.toString());
	}

  public char compute(int N, long K) {
    if (K >= lengths[N]) return '.';

    if (N == 0) {
      return f0.charAt((int)K);
    }
    else if (K < f11.length()) {
      return f11.charAt((int)K);
    } 
    else if (K < f11.length() + lengths[N-1]) {
      return compute(N-1, K-f11.length());
    }
    else if (K < f11.length() + lengths[N-1] + f12.length()) {
      return f12.charAt((int)K - f11.length() - (int)lengths[N-1]);
    }
    else if (K < f11.length() + lengths[N-1] + f12.length() + lengths[N-1]) {
      return compute(N-1, K-f11.length() - lengths[N-1] - f12.length());
    }
    else {
      return f13.charAt((int)K - f11.length() - (int)lengths[N-1] - f12.length() - (int)lengths[N-1]);
    }
  }
	
	/**
	 * @param args
	 */
	public static void main(String[] args) {
		new Q449A().run();
	}
	
	public void run() {
        try {
            reader = new BufferedReader(new InputStreamReader(System.in));
            tokenizer = null;
            out = new PrintWriter(System.out);
            solve();
            reader.close();
            out.close();
        } catch (Exception e) {
            e.printStackTrace();
            System.exit(1);
        }
    }

    int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    String nextToken() throws IOException {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            tokenizer = new StringTokenizer(reader.readLine());
        }
        return tokenizer.nextToken();
    }

}
