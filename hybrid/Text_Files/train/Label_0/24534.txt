const long long mod = 1e9 + 7;
const long long N = 300050;
long long ask(string s) {
  cout << s << endl;
  long long ans;
  cin >> ans;
  if (ans == 0) exit(0);
  return ans;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  ;
  long long test = 1;
  while (test--) {
    string A(300, 'a'), B(300, 'b');
    long long cnta = ask(A);
    cnta = 300 - cnta;
    long long cntb = ask(B);
    cntb = 300 - cntb;
    long long answer = cntb, n = cnta + cntb;
    string final(n, 'a');
    if (cntb == 0) ask(final);
    for (long long i = 0; i < n; i++) {
      final[i] = 'b';
      long long temp = ask(final);
      if (temp < answer) {
        answer = temp;
        continue;
      } else
        final[i] = 'a';
    }
  }
  return 0;
}
