import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Iterator;
import java.util.Map;


public class CF714C {
	public static void printMap(Map mp) {
	    Iterator it = mp.entrySet().iterator();
	    while (it.hasNext()) {
	        Map.Entry pair = (Map.Entry)it.next();
	        System.out.println(pair.getKey() + " = " + pair.getValue());
	        //it.remove(); // avoids a ConcurrentModificationException
	    }
	}
	public static void main(String[] args) throws IOException
	{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int t = Integer.parseInt(br.readLine());
		//System.out.println("bla");
		HashMap<String,Integer> hm = new HashMap<String,Integer>();
		for(int i=0;i<t;i++)
		{
			String[] givStr = br.readLine().split(" ");
			//System.out.println(givStr[0].charAt(0));
			switch(givStr[0].charAt(0))
			{
			case '+':
				String numa = givStr[1];
				//System.out.println(numa);
				int len = numa.length();
				StringBuilder sb = new StringBuilder();
				for(int j=0;j<31-len;j++)
					sb.append("0");
				//System.out.println(sb.toString());
				for(int j=0;j<len;j++)
				{
					if((int)numa.charAt(j)%2==0)
					{
						sb.append("0");
					}
					else
					{
						sb.append("1");
					}
				}
				//System.out.println(sb.toString());
				if(hm.containsKey(sb.toString()))
				{
					int oldvalue=hm.get(sb.toString());
					hm.replace(sb.toString(),oldvalue,oldvalue+1);
				}
				else
					hm.put(sb.toString(), 1);
				sb = null;
				break;
			case '-':
				String numr = givStr[1];
				int lenr = numr.length();
				sb = new StringBuilder();
				for(int j=0;j<31-lenr;j++)
					sb.append("0");
				for(int j=0;j<lenr;j++)
				{
					if((int)numr.charAt(j)%2==0)
					{
						sb.append("0");
					}
					else
					{
						sb.append("1");
					}
				}
				if(hm.containsKey(sb.toString()))
				{
					int oldvalue=hm.get(sb.toString());
					if(oldvalue==1)
						hm.remove(sb.toString());
					else
						hm.replace(sb.toString(),oldvalue,oldvalue-1);
				}
				sb = null;
				break;
			case '?':
				String pat = givStr[1];
				int lenp = pat.length();
				StringBuilder sbp= new StringBuilder("");
				for(int j=0;j<31-lenp;j++)
					sbp.append("0");
				sbp.append(pat);
				//System.out.println(sbp.toString());
				if(hm.get(sbp.toString())!=null)
					System.out.println(hm.get(sbp.toString()));
				else
					System.out.println("0");
				sbp = null;
				break;
			}
			//System.out.println("MAP");
			//printMap(hm);
		}
	}
}
