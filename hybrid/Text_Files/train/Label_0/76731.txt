n = int(input())
s = list(input())
t = list(input())
if (s.count('a') + t.count('a')) % 2 != 0:
	print(-1)
else:
	a = []
	b = []
	for i in range(n):
		if s[i] == 'a' and t[i] == 'b':
			a.append(i)
		elif s[i] == 'b' and t[i] == 'a':
			b.append(i)
	a = a[::-1]
	b = b[::-1]
	ans = []
	while len(a) >= 2:
		ans.append([a.pop() + 1, a.pop() + 1])
	while len(b) >= 2:
		ans.append([b.pop() + 1, b.pop() + 1])
	if a:
		ans.append([a[0] + 1, a[0] + 1])
		ans.append([a[0] + 1, b[0] + 1])
	print(len(ans))
	for x in ans:
		print(*x)
