#include <cstdio>

const int MAXN = 3e2+3;

int n, x;
double mem[MAXN][MAXN][MAXN];
bool solved[MAXN][MAXN][MAXN];
int cnt[5];

double dp(int x, int y, int z) {
  if (x == 0 && y == 0 && z == 0) return 0;
  
  double &ret = mem[x][y][z];
  if (solved[x][y][z]) return ret;
  solved[x][y][z] = 1;

  double t = x+y+z;
  ret += n / t;
  if (x != 0) ret += x / t * dp(x-1, y, z);
  if (y != 0) ret += y / t * dp(x+1, y-1, z);
  if (z != 0) ret += z / t * dp(x, y+1, z-1);

  return ret;
}

int main() {
  
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &x);
    cnt[x]++;
  }
  

  printf("%.15f\n", dp(cnt[1], cnt[2], cnt[3]));
  return 0;
}
