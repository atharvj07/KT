        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);

        solver.solve(1, in, out);

class TaskD {
    public void solve(int testNumber, InputReader in, PrintWriter out) {
        int n = in.nextInt();
        int m = in.nextInt();
        Island[] islands = new Island[n];
        for (int i = 0; i < n; i++) {
            islands[i] = new Island(in.nextLong(), in.nextLong());
        }
        Bridge[] bridges = new Bridge[m];
        for (int i = 0; i < m; i++) {
            bridges[i] = new Bridge(in.nextLong(), i + 1);
        }
        Arrays.sort(bridges);
        Arrays.sort(islands);
        Require[] requires = new Require[n - 1];
        for (int i = 1; i < n; i++) {
            long min = islands[i].left - islands[i - 1].right;
            long max = islands[i].right - islands[i - 1].left;
            requires[i - 1] = new Require(i - 1, min, max);
        }
        Arrays.sort(requires, new Comparator<Require>() {
            public int compare(Require o1, Require o2) {
                return Long.compare(o1.min, o2.min);
            }
        });
        PriorityQueue<Require> queue = new PriorityQueue<Require>(1, new Comparator<Require>() {
            public int compare(Require o1, Require o2) {
                return Long.compare(o1.max, o2.max);
            }
        });
        int[] result = new int[requires.length];
        int tot = 0;
        for (int i = 0; i < bridges.length; i++) {
            while (tot < requires.length && requires[tot].min <= bridges[i].value) queue.add(requires[tot++]);
            while (!queue.isEmpty() && queue.peek().max < bridges[i].value) queue.poll();
            if (queue.isEmpty()) continue;
            result[queue.poll().index] = bridges[i].index;
        }
        for (int i = 0; i < result.length; i++) {
            if (result[i] == 0) {
                out.println("No");
                return;
            }
        }
        out.println("Yes");
        for (int i = 0; i < result.length; i++) {
            out.printf("%d ", result[i]);
        }
    }
}

class Bridge implements Comparable<Bridge> {
    long value;
    int index;

    public Bridge(long value, int index) {
        this.value = value;
        this.index = index;
    }

    public int compareTo(Bridge o) {
        return Long.compare(this.value, o.value);
    }
}

class Require {
    int index;
    long min;
    long max;

    public Require(int index, long min, long max) {
        this.index = index;
        this.min = min;
        this.max = max;
    }
}

class Island implements Comparable<Island> {
    long left;
    long right;

    public Island(long left, long right) {
        this.left = left;
        this.right = right;
    }

    public int compareTo(Island o) {
        return Long.compare(this.left, o.left);
    }
}

class InputReader {
    public BufferedReader reader;
    public StringTokenizer tokenizer;

    public InputReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream), 32768);
        tokenizer = null;
    }

    public String next() {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            try {
                tokenizer = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }
        return tokenizer.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(next());
    }

    public long nextLong() {
        return Long.parseLong(next());
    }

}
