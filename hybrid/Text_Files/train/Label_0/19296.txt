#include <bits/stdc++.h>
using namespace std;
const int N = 200010;
const long long mod = 1e9 + 7;
int n;
int a[N];
int roadto2k4[N];
int beforeMarch[N];
int whatever[N];
long long sum[N];
struct SegmentTree {
  int T[N << 2];
  int lazy[N << 3];
  void push(int x, int l, int r) {
    if (lazy[x]) {
      T[x] = 1LL * lazy[x] * (r - l + 1) % mod;
      lazy[x + x] = lazy[x];
      lazy[x + x + 1] = lazy[x];
    }
    lazy[x] = 0;
  }
  void modify(int x, int l, int r, int u, int v, int val) {
    push(x, l, r);
    if (l > v || r < u) return;
    if (l >= u && r <= v) {
      lazy[x] = val;
      push(x, l, r);
      return;
    }
    modify(x + x, l, ((l + r) >> 1), u, v, val);
    modify(x + x + 1, ((l + r) >> 1) + 1, r, u, v, val);
    T[x] = T[x + x] + T[x + x + 1];
    T[x] %= mod;
  }
  long long query(int x, int l, int r, int u, int v) {
    push(x, l, r);
    if (l > v || r < u) return 0;
    if (l >= u && r <= v) return T[x];
    long long ret = query(x + x, l, ((l + r) >> 1), u, v) +
                    query(x + x + 1, ((l + r) >> 1) + 1, r, u, v);
    T[x] = T[x + x] + T[x + x + 1];
    T[x] %= mod;
    return ret % mod;
  }
} Tmax, Tmin;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", a + i);
    sum[i] = sum[i - 1] + i;
    sum[i] %= mod;
  }
  map<int, int> f;
  for (int i = 1; i <= n; ++i) {
    roadto2k4[i] = f[a[i]] + 1;
    f[a[i]] = i;
  }
  f.clear();
  for (int i = n; i >= 1; --i) {
    beforeMarch[i] = (f[a[i]] ? f[a[i]] - 1 : n);
    whatever[i] = (f[a[i]] ? f[a[i]] : n + 1);
    f[a[i]] = i;
  }
  int to = n;
  long long maxmin = 0, iright = 0, jleft = 0, ij = 0, answer = 0;
  deque<int> dqmax, dqmin;
  for (int i = n; i >= 1; --i) {
    iright = ij = 0;
    while (!dqmax.empty() && roadto2k4[i] >= roadto2k4[dqmax.back()]) {
      int j = dqmax.back();
      dqmax.pop_back();
      int last = (dqmax.empty() ? to : dqmax.back() - 1);
      jleft +=
          1LL * (roadto2k4[i] - roadto2k4[j]) * (sum[last] - sum[j - 1] + mod);
      jleft %= mod;
      maxmin +=
          1LL * Tmin.query(1, 1, n, j, last) * (roadto2k4[i] - roadto2k4[j]);
      maxmin %= mod;
      Tmax.modify(1, 1, n, j, last, roadto2k4[i]);
    }
    while (!dqmin.empty() && beforeMarch[i] <= beforeMarch[dqmin.back()]) {
      int j = dqmin.back();
      dqmin.pop_back();
      int last = (dqmin.empty() ? to : dqmin.back() - 1);
      maxmin += 1LL * Tmax.query(1, 1, n, j, last) *
                (beforeMarch[i] - beforeMarch[j]);
      maxmin = (maxmin % mod + mod) % mod;
      Tmin.modify(1, 1, n, j, last, beforeMarch[i]);
    }
    maxmin += 1LL * roadto2k4[i] * beforeMarch[i];
    maxmin %= mod;
    jleft += 1LL * i * roadto2k4[i];
    jleft %= mod;
    dqmax.push_back(i);
    dqmin.push_back(i);
    Tmax.modify(1, 1, n, i, i, roadto2k4[i]);
    Tmin.modify(1, 1, n, i, i, beforeMarch[i]);
    while (to >= whatever[i]) {
      maxmin -= 1LL * Tmin.query(1, 1, n, to, to) * Tmax.query(1, 1, n, to, to);
      maxmin = (maxmin % mod + mod) % mod;
      jleft -= 1LL * to * Tmax.query(1, 1, n, to, to);
      jleft = (jleft % mod + mod) % mod;
      if (dqmax.front() == to) dqmax.pop_front();
      if (dqmin.front() == to) dqmin.pop_front();
      --to;
    }
    ij += 1LL * i * (sum[to] - sum[i - 1] + mod);
    ij %= mod;
    iright += 1LL * i * (Tmin.query(1, 1, n, i, to));
    iright %= mod;
    answer += (iright + jleft - ij - maxmin + mod * mod) % mod;
    answer %= mod;
  }
  cout << answer % mod << endl;
  return 0;
}
