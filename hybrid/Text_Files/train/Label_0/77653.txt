n = int(input())
(l, r) = ([], [])
Lmin = 1000000000.0
Rmax = 0
for i in range(0, n):
	(a, b) = map(int, input().split())
	l.append(a)
	r.append(b)
	Lmin = min(Lmin, l[i])
	Rmax = max(Rmax, r[i])
flag = 0
for i in range(0, n):
	if l[i] == Lmin and r[i] == Rmax:
		flag = 1
		print(i + 1)
if flag == 0:
	print(-1)
