#include <bits/stdc++.h>
using namespace std;
string s[1001];
int n, m;
int dist[3][1001][1001];
int dr[] = {0, 1, 0, -1};
int dc[] = {1, 0, -1, 0};
void compute(int dist[1001][1001], char c) {
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j) dist[i][j] = 1e8 + 9;
  queue<pair<int, int> > Q;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j)
      if (s[i][j] == c) dist[i][j] = 0, Q.push(make_pair(i, j));
  while (!Q.empty()) {
    int r = Q.front().first;
    int c = Q.front().second;
    Q.pop();
    for (int k = 0; k < 4; ++k) {
      int nr = r + dr[k];
      int nc = c + dc[k];
      if (nr >= 0 && nr < n && nc >= 0 && nc < m && s[nr][nc] != '#') {
        if (dist[nr][nc] > dist[r][c] + (s[nr][nc] == '.'))
          dist[nr][nc] = dist[r][c] + (s[nr][nc] == '.'),
          Q.push(make_pair(nr, nc));
      }
    }
  }
}
int main() {
  std::ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; ++i) cin >> s[i];
  for (int id = 0; id < 3; ++id) compute(dist[id], '1' + id);
  int ret = 1e8;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j)
      if (s[i][j] != '#') {
        if (s[i][j] == '.')
          ret = min(ret, dist[0][i][j] + dist[1][i][j] + dist[2][i][j] - 2);
        else
          ret = min(ret, dist[0][i][j] + dist[1][i][j] + dist[2][i][j]);
      }
  if (ret >= 1e8) ret = -1;
  cout << ret;
}
