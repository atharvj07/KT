#include <iostream>
#include <vector>
#include <sstream>
#include <algorithm>
#include <map>
using namespace std;
#define rep(i,n) for(int i=0;i<(n);i++)
#define all(e) (e).begin(), (e).end()
typedef long long ll;
ll n;
vector<ll> tubo28;

map<ll,ll> dp;

void make(ll m,int digit) {
    if(m) tubo28.push_back(m);
    if(digit==18) {
        return;
    }
    make(m*10+2,digit+1);
    make(m*10+8,digit+1);
}

ll dfs(ll n) {
    if(n==1) {
        return 0;
    }
    if(dp.count(n)) return dp[n];
    ll ret=-1;

    rep(i,tubo28.size()) {
        if(tubo28[i]>n) break;
        if(n%tubo28[i]==0) {
            ll t=dfs(n/tubo28[i]);
            if(t>=0) ret=max(ret,t+1);
        }
    }
    
    return dp[n]=ret;
}

int main() {
    make(0,0);
    sort(all(tubo28));
    //reverse(all(tubo28));
    //comp();
    cin>>n;
    ll ans=dfs(n);
    if(ans<=0) ans=-1;
    cout<<ans<<endl;
    return 0;
}
