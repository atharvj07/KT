#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 4 != 0) {
      cout << "NO" << endl;
    } else {
      int a[n];
      a[0] = 2;
      int sume = 2;
      for (int i = 1; i < n / 2; i++) {
        a[i] = a[i - 1] + 2;
        sume += a[i];
      }
      a[n / 2] = 1;
      int sumo = 1;
      for (int i = (n / 2) + 1; i < n - 1; i++) {
        a[i] = a[i - 1] + 2;
        sumo += a[i];
      }
      a[n - 1] = sume - sumo;
      cout << "YES" << endl;
      for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
      }
      cout << endl;
    }
  }
  return 0;
}
