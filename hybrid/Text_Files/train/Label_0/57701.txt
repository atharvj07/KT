import java.util.Scanner;
public class B
{
public static void main(String args[])
{
Scanner sc=new Scanner(System.in);
long N=sc.nextLong();
int K=sc.nextInt();
int i=0;
long ar[]=new long[K];
for(;i<K;i++)
{
ar[i]=sc.nextLong();
if(N%ar[i]==0)
{
System.out.println((i+1)+" "+(N/ar[i]));
return;
}
}

long min=(N%ar[0]),tmp,rem;
int pos=0;
for(i=0;i<K;i++)
{
rem=N%ar[i];
if(rem<min)
{
pos=i;
min=rem;
}
}
System.out.println((pos+1)+" "+(N/ar[pos]));
}
}
