#include <iostream>

using namespace std;

int main() {
    int case_num, gx, gy, p, x1, y1, x2, y2;
    int count[16][16];
    int banh[16][16], banv[16][16];
    cin >> case_num;
    for (int case_i = 0; case_i < case_num; ++case_i) {
        cin >> gx >> gy >> p;
        for (int y = 0; y <= gy; ++y) {
            for (int x = 0; x <= gx; ++x) {
                banh[x][y] = banv[x][y] = false;
            }
        }
        for (int i = 0; i < p; ++i) {
            cin >> x1 >> y1 >> x2 >> y2;
            if (x1 == x2) {
                banv[x1][max(y1, y2)] = true;
            } else {
                banh[max(x1, x2)][y1] = true;
            }
        }
        for (int y = 0; y <= gy; ++y) {
            for (int x = 0; x <= gx; ++x) {
                if (x == 0 && y == 0) {
                    count[x][y] = 1;
                } else {
                    count[x][y] = 0;
                    if (x > 0 && !banh[x][y]) {
                        count[x][y] += count[x - 1][y];
                    }
                    if (y > 0 && !banv[x][y]) {
                        count[x][y] += count[x][y - 1];
                    }
                }
            }
        }
        if (count[gx][gy] > 0) {
            cout << count[gx][gy] << endl;
        } else {
            cout << "Miserable Hokusai!" << endl;
        }
    }
    return 0;
}
