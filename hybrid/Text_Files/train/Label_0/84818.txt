(cs, ln) = map(int, input().split())
ar = list(map(int, input().split()))
z = ar.count(0)
(o, sol) = (cs - z, [])
while ln:
	ln -= 1
	(qst, pl) = map(int, input().split())
	if qst == 2:
		if cs - pl >= z:
			sol.append(1)
		else:
			sol.append(0)
	elif ar[pl - 1] == 0:
		ar[pl - 1] = 1
		o += 1
		z -= 1
	else:
		ar[pl - 1] = 0
		o -= 1
		z += 1
print(*sol, sep='\n')
