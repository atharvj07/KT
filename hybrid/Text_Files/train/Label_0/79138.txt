length = int(input())
nums = list(map(int, input().split(' ')))
asc = True
equal = False
descending = False
correct = True
for (i, n) in enumerate(nums):
	if i == 0:
		continue
	elif asc and nums[i - 1] < n:
		continue
	elif asc and nums[i - 1] == n:
		asc = False
		equal = True
	elif asc and nums[i - 1] > n:
		asc = False
		descending = True
	elif equal and nums[i - 1] == n:
		continue
	elif equal and nums[i - 1] > n:
		equal = False
		descending = True
	elif descending and nums[i - 1] > n:
		continue
	else:
		print('NO')
		correct = False
		break
if correct:
	print('YES')
