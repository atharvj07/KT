import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;


public class Pipeline {
	public static void main(String[] args) throws IOException{
		BufferedReader lector = new BufferedReader(new InputStreamReader(System.in)); //lectura de la entrada
		String linea = lector.readLine(); //string de la entrada
		String[] datos = linea.split(" ");
		
		long n = Long.parseLong(datos[0]);
		Long k = Long.parseLong(datos[1]);
		
		if(n == 0 || n == 1){
			System.out.println(0);
		}else if(n> k*(k-1)/2 + 1){
			System.out.println(-1);
		}else if(k>=n){
			System.out.println(1);
		}else{
			//Hacemos busqueda binaria
			long min = 2;
			long max = k-1;
			long mid = 0;
			long total = 0;
			while(true){
				mid = (max + min)/2;
				total = maxTub(k,mid);
				//System.out.println("Max, min y mid son : " + max + " " + min + " " + mid);
				if((total == n) || (total >= n && maxTub(k, mid-1)<n)){
					System.out.println(mid);
					break;
				}else if(total > n){
					max = mid - 1;
				}else{
					min = mid +1;
				}
				//System.out.println("Estoy en el while");
			}
			
			
		}
		
	}
	
	//Maximo que podemos lograr con m tuberias
	public static long maxTub(long k, long m){
		return m*k - ((m-1)*(m+2))/2;
	}
}
