import sys
input = sys.stdin.readline
n = int(input())
for _ in range(n):
	(x, y) = map(int, input().split())
	x = max(x, y // 2 + 1)
	(lt, rt) = (x, y)
	answer = 0
	while lt <= rt:
		mid = (lt + rt) // 2
		if y % mid > answer:
			answer = y % mid
			rt = mid - 1
		else:
			lt = mid + 1
	print(answer)
