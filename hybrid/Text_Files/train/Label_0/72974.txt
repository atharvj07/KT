import math
for t in range(int(input())):
	(a, b) = map(int, input().split())
	(c, d) = map(int, input().split())
	(e, f) = map(int, input().split())
	ans = 0
	if e > f:
		t = e
		e = f
		f = t
		t1 = c
		c = d
		d = t1
	for i in range(min(c, a // e) + 1):
		w1 = min(d, (a - i * e) // f)
		s2 = min(c - i, b // e)
		w2 = min(d - w1, (b - s2 * e) // f)
		ans = max(ans, i + s2 + w1 + w2)
	print(ans)
