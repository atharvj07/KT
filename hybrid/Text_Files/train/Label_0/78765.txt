mod = 998244353
maxn = 1005

def solve(n, m):
	if n > m:
		(n, m) = (m, n)
	(ans, s) = (0, 1)
	for i in range(0, n + 1):
		ans = ans + s
		s = s * (n - i) // (i + 1) * (m - i)
	return ans
while True:
	try:
		(a, b, c) = map(int, input().split())
		print(solve(a, b) * solve(a, c) * solve(b, c) % mod)
	except:
		break
