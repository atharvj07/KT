#include <bits/stdc++.h>
using namespace std;
int const INF = (1 << 30);
int const MAXN = (int)1e5 + 100;
double a, b, m, vx, vy, vz;
double x, y, z;
int main() {
  cin >> a >> b >> m;
  cin >> vx >> vy >> vz;
  double t = (-1 * m) / vy;
  x = a / 2;
  x += vx * t;
  z += vz * t;
  while (x < 0 || x > a) {
    if (x < 0) x = (-1) * x;
    if (x > a) x = a - (x - a);
  }
  while (z < 0 || z > b) {
    if (z < 0) z = (-1) * z;
    if (z > b) z = b - (z - b);
  }
  printf("%.10f %.10f", x, z);
  return 0;
}
