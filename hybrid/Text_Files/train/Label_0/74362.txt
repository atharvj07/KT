_ = int(input())
for __ in range(_):
	(n, a, b) = map(int, input().split())
	out = [x for x in range(n, 0, -1)]
	pos_a = out.index(a)
	nn = n // 2
	if pos_a > nn - 1:
		(out[nn - 1], out[pos_a]) = (out[pos_a], out[nn - 1])
	pos_b = out.index(b)
	if pos_b < nn:
		(out[nn], out[pos_b]) = (out[pos_b], out[nn])
	if min(out[0:nn]) == a and max(out[nn:n]) == b:
		print(' '.join([str(x) for x in out]))
	else:
		print(-1)
