import java.util.*;
import java.io.*;
import java.math.*;

   public class x958B2
   {
      static int N;
      static ArrayDeque<Integer>[] edges;
      public static void main(String hi[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
         StringTokenizer st = new StringTokenizer(infile.readLine());
         N = Integer.parseInt(st.nextToken());
         if(N < 3)
         {
            int[] lol = {1, 2};
            for(int i=0; i < N; i++)
               System.out.print(lol[i]+" ");
            return;
         }
         edges = new ArrayDeque[N+1];
         for(int i=1; i <= N; i++)
            edges[i] = new ArrayDeque<Integer>();
         for(int i=0; i < N-1; i++)
         {
            st = new StringTokenizer(infile.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            edges[a].add(b);  edges[b].add(a);
         }
         depth = new int[N+1];
         path = new int[N+1];
         Arrays.fill(depth, -1);
         int first = farthest(1);
         int second = farthest(first);
         int root = second;
         dfs(root, 0);
         ArrayList<Integer> ls = getPath(first, second);
         PriorityQueue<Integer> pq = new PriorityQueue<Integer>( (x,y) -> {
            return depth[y]-depth[x];
         });
         int[] res = new int[N+1];
         res[1] = 1;   res[2] = ls.size();
         for(int v=1; v <= N; v++)
            pq.add(v);
         boolean[] taken = new boolean[N+1];
         for(int v: ls)
            taken[v] = true;
         int cnt = ls.size();
         int dex = 3;
         while(pq.size() > 0)
         {
            int curr = pq.poll();
            if(taken[curr])
               continue;
            int head = curr;
            while(head != 0)
            {
               if(!taken[head])
                  cnt++;
               taken[head] = true;
               head = path[head];
            }
            res[dex++] = cnt;
         }
         for(; dex <= N; dex++)
            res[dex] = cnt;
         StringBuilder sb = new StringBuilder();
         for(int i=1; i <= N; i++)
            sb.append(res[i]+" ");
         System.out.println(sb);
      }
      public static ArrayList<Integer> getPath(int a, int b)
      {
         int[] par = new int[N+1];
         Arrays.fill(par, -1);
         par[b] = 0;
         ArrayDeque<Integer> q = new ArrayDeque<Integer>();
         q.add(b);
         while(q.size() > 0)
         {
            int curr = q.poll();
            for(int next: edges[curr])
               if(par[next] == -1)
               {
                  par[next] = curr;
                  q.add(next);
               }
         }
         ArrayList<Integer> ls = new ArrayList<Integer>();
         int head = a;
         while(head != b)
         {
            ls.add(head);
            head = par[head];
         }
         ls.add(b);
         return ls;
      }
      public static int farthest(int root)
      {
         int[] dist = new int[N+1];
         Arrays.fill(dist, Integer.MAX_VALUE/2);
         dist[root] = 0;
         ArrayDeque<Integer> q = new ArrayDeque<Integer>();
         q.add(root);
         int farthest = root;
         while(q.size() > 0)
         {
            int curr = q.poll();
            for(int next: edges[curr])
               if(dist[next] == Integer.MAX_VALUE/2)
               {
                  dist[next] = dist[curr]+1;
                  farthest = next;
                  q.add(next);
               }
         }
         return farthest;
      }
      static int[] depth;
      static int[] path;
      public static void dfs(int curr, int par)
      {
         for(int next: edges[curr])
            if(next != par)
            {
               dfs(next, curr);
               if(depth[curr] == -1 || depth[curr] < depth[next]+1)
               {
                  depth[curr] = depth[next]+1;
                  path[curr] = next;
               }
            }
         if(depth[curr] == -1)
            depth[curr] = 0;
      }
   }
