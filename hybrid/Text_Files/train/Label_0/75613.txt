for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	i = 1
	l = 0
	g = a[0]
	while i < n and a[i] <= a[i - 1]:
		g = min(g, a[i])
		i += 1
	l = i - 1
	i = n - 2
	h = a[-1]
	while i >= 0 and a[i + 1] >= a[i]:
		h = min(h, a[i])
		i -= 1
	r = i + 1
	if r <= l:
		print('YES')
	else:
		p = 0
		q = 0
		for i in range(l + 1, r + 1):
			if a[i] <= a[i - 1]:
				p += abs(a[i - 1] - a[i])
		for i in range(r - 1, l - 1, -1):
			if a[i] <= a[i + 1]:
				q += abs(a[i + 1] - a[i])
		if p <= g or q <= h:
			print('YES')
		else:
			print('NO')
