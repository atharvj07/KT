MOD = 998244353
(N, M) = map(int, input().split())
S = [input() for _ in range(N)]
L = max(N, M)
imos = [0] * (L + 2)
for i in range(N):
	cnt = 0
	for j in range(M):
		if S[i][j] == '*':
			cnt = 0
		else:
			cnt += 1
			imos[1] += 1
			imos[cnt + 1] -= 1
for j in range(M):
	cnt = 0
	for i in range(N):
		if S[i][j] == '*':
			cnt = 0
		else:
			cnt += 1
			imos[1] += 1
			imos[cnt + 1] -= 1
ans = 0
for i in range(2, min(L, imos[1] // 2) + 1):
	imos[i] += imos[i - 1]
	ans += imos[i] * pow(2, imos[1] // 2 - i, MOD) * (-1) ** (i % 2) % MOD
	ans %= MOD
print(ans)
