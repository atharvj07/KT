#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, mod = 1e9 + 7;
long long a, b, k, r;
int main() {
  cin.tie(0);
  cout.tie(0);
  ios_base ::sync_with_stdio();
  cin >> a >> b;
  k = a * (a + 1);
  k /= 2;
  k %= mod;
  for (long long i = 1; i < b; i++) {
    long long z, y;
    z = b * i;
    z %= mod;
    z *= k;
    z %= mod;
    z += i * a;
    z %= mod;
    r += z;
    r %= mod;
  }
  cout << r;
  return 0;
}
