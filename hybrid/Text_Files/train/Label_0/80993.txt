from math import sqrt
from math import ceil
t = int(input())
for _ in range(t):
	n = int(input())
	arr = [0] * (n + 1)
	arr[-1] = n
	x = []
	y = []
	i = n
	while i > 2:
		j = i
		if i == n:
			j = n - 1
		while j > sqrt(arr[-1]) // 1 and j > 2:
			x.append(j)
			y.append(n)
			j -= 1
		i = j
		x.append(n)
		y.append(i)
		arr[-1] = ceil(arr[-1] / i)
	while arr[-1] != 1:
		arr[-1] = ceil(arr[-1] / 2)
		x.append(n)
		y.append(2)
	print(len(x))
	for i in range(len(x)):
		print(x[i], y[i])
