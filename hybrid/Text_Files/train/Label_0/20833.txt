#include <bits/stdc++.h>
using namespace std;
int Scan() {
  int res = 0, ch, flag = 0;
  if ((ch = getchar()) == '-')
    flag = 1;
  else if (ch >= '0' && ch <= '9')
    res = ch - '0';
  while ((ch = getchar()) >= '0' && ch <= '9') res = res * 10 + ch - '0';
  return flag ? -res : res;
}
void Out(int a) {
  if (a > 9) Out(a / 10);
  putchar(a % 10 + '0');
}
int main() {
  int ans, n, v1, v2, t;
  while (scanf("%d %d", &v1, &v2) != EOF) {
    int temp = v1;
    scanf("%d %d", &t, &n);
    if (n == 0) {
      printf("%d\n", t * v1);
    } else {
      ans = 0;
      if (v1 > v2) {
        int temp = v1;
        v1 = v2;
        v2 = temp;
      }
      int temp1 = (v2 - v1 + n + n * t) * 1.0 / (2 * n);
      int temp2 = 0;
      for (int i = 1; i <= temp1; i++) {
        ans += v1;
        ans += (temp2 * n);
        temp2++;
      }
      temp2 = 0;
      for (int i = t; i >= temp1 + 1; i--) {
        ans += v2;
        ans += (temp2 * n);
        temp2++;
      }
      printf("%d\n", ans);
    }
  }
  return 0;
}
