def readline():
	return map(int, input().split())

def solve():
	n = int(input())
	seq = list(readline())
	seq.extend([0, 0])
	ans = 0
	(a, b) = (0, 0)
	for c in seq:
		if a < b > c:
			diff = b - max(a, c)
			ans += diff
			b -= diff
		ans += abs(b - a)
		(a, b) = (b, c)
	print(ans)

def main():
	t = int(input())
	for __ in range(t):
		solve()
main()
