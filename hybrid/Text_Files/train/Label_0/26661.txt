#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, x;
  cin >> n >> x;
  int total = 2 * n - 1;

  if (x == 1 || x == total) {
    cout << "No" << endl;
    return 0;
  }

  int cur = n;
  int start = 1;
  while (cur != x) {
    ++start;
    ++cur;
    if (cur == 2 * n)
      cur = 1;
  }

  cout << "Yes" << endl;
  for (int i = start; i <= total; ++i)
    cout << i << ' ';
  for (int i = 1; i < start; ++i)
    cout << i << ' ';
  cout << endl;

  return 0;
}
