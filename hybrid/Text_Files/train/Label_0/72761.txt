from sys import stdin, stdout

def getmaxval(n, k):
	if n == k:
		return 1
	ol = 1
	oh = n // 2
	while ol < oh:
		om = (ol + oh + 1) // 2
		cnt = counteven(om * 2, n)
		if cnt >= k:
			ol = om
		else:
			oh = om - 1
	r1 = ol * 2
	if r1 < n:
		cnt2 = countodd(r1 + 1, n)
		if cnt2 >= k:
			return r1 + 1
	return r1

def counteven(b, n):
	res = 2
	inc = 2
	mv = b + 1
	while mv <= n:
		inc *= 2
		res += inc
		mv = b * inc // 2 + inc - 1
	if mv - inc <= n < mv:
		res -= mv - n
	elif n < mv - inc:
		res -= inc
	return res

def countodd(b, n):
	res = 1
	inc = 1
	mv = b
	while mv <= n:
		inc *= 2
		mv = b * inc + inc - 1
		res += inc
	if mv - inc <= n < mv:
		res -= mv - n
	elif n < mv - inc:
		res -= inc
	return res
nk = list(map(int, stdin.readline().split()))
n = nk[0]
k = nk[1]
res = getmaxval(n, k)
stdout.write(str(res))
