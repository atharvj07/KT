import java.io.*;
import java.util.*;

public class E {
    public static void main(String[] args) throws IOException {
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));
        InputReader in=new InputReader(System.in);
        int n=in.nextInt();
        int m=in.nextInt();
        int t=in.nextInt();
        char [][]chess=new char[n][m];
        int [][]position=new int[n][m];
        for(int i=0;i<n;i++)
             for(int j=0;j<m;j++)
                 position[i][j]= Integer.MAX_VALUE;
        for(int i=0;i<n;i++) chess[i]=in.next().toCharArray();
        solve(chess,position);
        while (t-->0){
            int i=in.nextInt();
            int j=in.nextInt();
            i--;j--;
            long p=in.nextLong();
            if(p<=position[i][j]||position[i][j]== Integer.MAX_VALUE) out.println(chess[i][j]);
            else{
                p-=position[i][j];
                if(p%2==0) out.println(chess[i][j]);
                else out.println(chess[i][j]=='1'?'0':'1');
            }
        }
        out.flush();
    }
    public static void solve(char [][]chess,int [][]position){
        List<Integer[]> list=new ArrayList<>();
        for(int i=0;i<chess.length;i++)
            for(int j=0;j<chess[0].length;j++){
                if(isNeighbor(i,j,chess)) list.add(new Integer[]{i,j,0});
            }
        ArrayDeque<Integer[]> queue=new ArrayDeque<>(list);
        while (queue.size()>0){
            Integer[] poll = queue.poll();
            int i=poll[0];int j=poll[1];
            if(i<0||i>chess.length-1||j<0||j>chess[0].length-1)
                continue;
            int val=poll[2];
            if(position[i][j]>val) {
                position[i][j]=val;
                if(i+1<chess.length&&position[i+1][j]==Integer.MAX_VALUE) queue.addLast(new Integer[]{i+1,j,val+1});
                if(j+1<chess[0].length&&position[i][j+1]==Integer.MAX_VALUE) queue.addLast(new Integer[]{i,j+1,val+1});
                if(j>0&&position[i][j-1]==Integer.MAX_VALUE)  queue.addLast(new Integer[]{i,j-1,val+1});
                if(i>0&&position[i-1][j]==Integer.MAX_VALUE) queue.addLast(new Integer[]{i-1,j,val+1});
            }
        }
    }
    public static boolean isNeighbor(int i,int j,char [][]chess){
        if(i>0&&chess[i][j]==chess[i-1][j]) return true;
        if(j>0&&chess[i][j]==chess[i][j-1]) return true;
        if(i<chess.length-1&&chess[i+1][j]==chess[i][j]) return true;
        if(j<chess[0].length-1&&chess[i][j+1]==chess[i][j]) return true;
        return false;
    }
    static class InputReader{
        private BufferedReader in;
        private StringTokenizer tokenizer;
        public InputReader(InputStream stream){
            in = new BufferedReader(new InputStreamReader(stream),32768);
            tokenizer = null;
        }
        public String next() throws IOException{
            while(tokenizer==null || !tokenizer.hasMoreTokens()){
                tokenizer = new StringTokenizer(in.readLine());
            }
            return tokenizer.nextToken();
        }
        public Long nextLong() throws IOException {
            String next = next();
            return Long.valueOf(next);
        }
        public int nextInt() throws IOException{
            return Integer.valueOf(next());
        }
    }
}
