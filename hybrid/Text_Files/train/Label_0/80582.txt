def mod(x, m):
	return x % m

def mul(x, y, m):
	return x * y % m

def add(x, y, m):
	return (x + y) % m

def gcdext(a, b):
	if b == 0:
		return (a, 1, 0)
	(g, y, x) = gcdext(b, a % b)
	return (g, x, y - x * (a // b))

def CRT(r1, m1, r2, m2):
	(g, x, y) = gcdext(m1, m2)
	if (r1 - r2) % g:
		return (-1, -1)
	z = m2 // g
	lcm = m1 * z
	sol = add(mod(r1, lcm), m1 * mul(mod(x, z), mod((r2 - r1) // g, z), z), lcm)
	return (sol, lcm)
(n, m, x, y, vx, vy) = [int(x) for x in input().split()]
if not vx and x and (x != n) or (not vy and y and (y != m)):
	print(-1)
	exit()
if not x and vx:
	vx = 1
if x == n and vx:
	vx = -1
if not y and vy:
	vy = 1
if y == m and vy:
	vy = -1
dx = x
if vx > 0:
	dx = n - x
dy = y
if vy > 0:
	dy = m - y
if not x or x == n:
	dx = 0
if not y or y == m:
	dy = 0
if not vx:
	if vy > 0:
		print(x, m)
	else:
		print(x, 0)
	exit()
if not vy:
	if vx > 0:
		print(n, y)
	else:
		print(0, y)
	exit()
res = CRT(dx, n, dy, m)
if res[0] == -1:
	print(-1)
	exit()
t = res[0] + res[1]
choques_y = (res[0] - dy) // m
choques_x = (res[0] - dx) // n
if not x or x == n:
	choques_x += 1
if not y or y == m:
	choques_y += 1
ansx = 0
if vx > 0 and choques_x % 2 == 0:
	ansx = n
elif vx < 0 and choques_x % 2:
	ansx = n
ansy = 0
if vy > 0 and choques_y % 2 == 0:
	ansy = m
elif vy < 0 and choques_y % 2:
	ansy = m
print(ansx, ansy)
