def addOne(s):
	s = list(s)
	n = len(s)
	for i in range(n):
		if s[i] == '9':
			s[i] = '0'
		else:
			s[i] = int(s[i])
			s[i] += 1
			s[i] = str(s[i])
	return ''.join(s)
n = int(input())
s = input()
a = []
b = []
c = []
x = s
s = addOne(s)
a.append(s)
while x != s:
	s = addOne(s)
	a.append(s)
for i in a:
	if '0' in i:
		b.append(i)
for y in b:
	m = len(y)
	for j in range(0, m):
		if y[j] == '0':
			q = y[j:m] + y[0:j]
			c.append(q)
c += b
print(min(c))
