#include <bits/stdc++.h>
using namespace std;
int rank1[1000005] = {0};
int parent[1000005];
int mark[1000005] = {0};
long long int pow(long long int x, long long int y, long long int M) {
  long long int m = 1;
  while (y != 0) {
    if (y & 1) {
      m *= x;
      m %= M;
    }
    y /= 2;
    x = x * x;
    x %= M;
  }
  m %= M;
  return m;
}
int find(int x) {
  if (parent[x] != x) return find(parent[x]);
  return x;
}

    rank1[roota]++;
    parent[rootb] = roota;
  }
}
int main() {
  long long int p, k;
  for (int i = 0; i <= 1000000; i++) parent[i] = i;
  cin >> p >> k;
  int t = 0;
  if (k > 1) {
    for (long long int i = 1; i < p; i++) {
      union1(i, (i * k) % p);
    }
    for (int i = 1; i < p; i++) {
      mark[find(i)]++;
    }
    for (int i = 1; i < p; i++)
      if (mark[i] > 0) t++;
  }
  if (k == 0)
    cout << pow(p, p - 1, 1000000007);
  else if (k == 1)
    cout << pow(p, p, 1000000007);
  else
    cout << pow(p, t, 1000000007);
  return 0;
}
