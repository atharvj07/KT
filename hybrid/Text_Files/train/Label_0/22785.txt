#include <bits/stdc++.h>
using namespace std;
int read() {
  char ch = getchar();
  int x = 0, fl = 1;
  for (; !isdigit(ch); ch = getchar())
    if (ch == '-') fl = -1;
  for (; isdigit(ch); ch = getchar()) x = (x << 3) + (x << 1) + (ch - '0');
  return x * fl;
}
const int NN = 50000 + 17;
void open() {
  freopen("a.in", "r", stdin);
  freopen("a.out", "w", stdout);
}
int n;
int stk[NN];
char str[NN];
int main() {
  scanf("%s", str + 1);
  n = strlen(str + 1);
  for (int l = 1; l * 2 <= n; l++) {
    int now = 0, tp = 0;
    for (int i = 1; i <= n; i++) {
      stk[++tp] = str[i];
      if (tp > l && stk[tp] == stk[tp - l]) {
        now++;
        if (now == l) tp -= l, now = 0;
      } else {
        now = 0;
      }
    }
    for (int i = 1; i <= tp; i++) str[i] = stk[i];
    n = tp;
  }
  for (int i = 1; i <= n; i++) {
    printf("%c", str[i]);
  }
  return 0;
}
