const int MX = 2e5 ;
pair<ll,ll>arr[MX];
ll n , m ;
bool can (ll mid) {
    ll req = 0 ;
    for (int i = 0 ; i < n ; i++){
        if (arr[i].first>=mid)continue;
        req+=((mid-arr[i].first)*arr[i].second);
        if (req>m)return false ;
    }
    return true ;
}
ll bs () {
    ll l = 0 , r = 2e13 , mid , ans = 0;
    while(l <= r){
        mid = (l+r)/2;
        if (can(mid)){
            l = mid +1;
            ans = mid ;
        }else {
            r = mid - 1;
        }
    }
    return ans ;
}
