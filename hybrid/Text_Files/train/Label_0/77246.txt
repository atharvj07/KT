(n, m, x, y) = map(int, input().split())
xx = int(x)
print(x, y)
i = 0
rev = False
flag = True
while i != n:
	for j in range(1, m + 1):
		if i == 0:
			if j != y:
				print(x, j)
		elif rev:
			print(x, m - (j - 1))
		else:
			print(x, j)
	if x > 1 and flag:
		x -= 1
	else:
		x = xx + 1
		xx += 1
		flag = False
	rev = not rev
	i += 1
