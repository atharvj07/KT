t = int(input())
a = list(map(int, input().split()))
v = True
if sum(a) != 0:
	print('YES')
	print('1')
	print('1 ' + str(len(a)))
	v = False
else:
	sum = 0
	for i in range(0, t):
		sum += a[i]
		if sum != 0:
			print('YES')
			print('2')
			print('1 ' + str(i + 1))
			print(str(i + 2) + ' ' + str(t))
			v = False
			break
if v:
	print('NO')
