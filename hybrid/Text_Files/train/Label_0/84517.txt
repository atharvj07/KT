b = input()
b = str(b)
a = input()
a = int(a)
if len(b) % a != 0:
	print('NO')
else:
	kol = len(b) / a
	kol = int(kol)
	b = b + ' '
	flag = 0
	for i in range(a):
		if b[i * kol:i * kol + kol] == b[i * kol + kol - 1:i * kol:-1] + b[i * kol]:
			flag += 1
	if flag == a:
		print('YES')
	else:
		print('NO')
