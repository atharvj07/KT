def getarray():
	return list(map(int, input().split()))

def getints():
	return map(int, input().split())

def getint():
	return int(input())

def solve():
	n = getint()
	s = input()
	ops = 0
	for k in range(n - 1):
		i = s[k]
		if int(i) > 0:
			ops += int(i) + 1
	if int(s[-1]) > 0:
		ops += int(s[-1])
	print(ops)
for t in range(getint()):
	solve()
