#include <bits/stdc++.h>
using namespace std;
pair<int, pair<int, int> > Edges[200005];
multiset<long long> SP;
multiset<long long>::iterator Sit;
long long Graph[805][805];
map<int, int> M;
map<int, int>::iterator it;
long long MAX = 1000000000000000000;
long long FindAns(int k, int N) {
  int i, j;
  for (i = 1; i <= N; ++i) {
    for (j = i + 1; j <= N; ++j) {
      if (Graph[i][j] != MAX) SP.insert(Graph[i][j]);
    }
  }
  i = 1;
  Sit = SP.begin();
  while (i < k) {
    ++Sit;
    ++i;
  }
  return *Sit;
}
void FloydWarshall(int N) {
  int i, j, k;
  for (i = 1; i <= N; ++i) {
    for (j = i; j <= N; ++j)
      if (Graph[i][j] == 0LL) Graph[i][j] = Graph[j][i] = MAX;
  }
  for (k = 1; k <= N; ++k) {
    for (i = 1; i <= N; ++i) {
      for (j = 1; j <= N; ++j)
        Graph[j][i] = Graph[i][j] = min(Graph[i][j], Graph[i][k] + Graph[k][j]);
    }
  }
}
int BuildGraph(int N, int m, int k) {
  int u, v, w, i, j, limit = min(m, k);
  for (i = 1, j = 1; i <= limit; ++i) {
    u = Edges[i].second.first;
    v = Edges[i].second.second;
    w = Edges[i].first;
    it = M.find(u);
    if (it == M.end()) M[u] = j++;
    u = M[u];
    it = M.find(v);
    if (it == M.end()) M[v] = j++;
    v = M[v];
    Graph[u][v] = Graph[v][u] = w;
  }
  return int(M.size());
}
int main() {
  int N, m, k, i;
  scanf("%d%d%d", &N, &m, &k);
  for (i = 1; i <= m; ++i)
    scanf("%d%d%d", &Edges[i].second.first, &Edges[i].second.second,
          &Edges[i].first);
  sort(Edges + 1, Edges + m + 1);
  int vertices = BuildGraph(N, m, k);
  FloydWarshall(vertices);
  printf("%lld\n", FindAns(k, vertices));
  return 0;
}
