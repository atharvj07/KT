from math import gcd
for _ in range(int(input())):
	res = 0
	n = int(input())
	nums = []
	one = 0
	even = 0
	for i in input().split():
		i = int(i)
		if i == 0:
			continue
		if i == 1:
			one += 1
		elif i % 2 == 0:
			even += 1
		else:
			nums.append(i)
	n = len(nums)
	res += one * even
	del one
	res += even * n
	res += sum(range(even))
	if len(nums) > 1:
		while nums:
			num = nums.pop()
			res += len([1 for i in nums if gcd(num, i) > 1])
	print(res)
