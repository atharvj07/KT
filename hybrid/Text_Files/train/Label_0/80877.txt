d = {'a': 1, 'b': 2, 'c': 3, 'd': 4, 'e': 5, 'f': 6, 'g': 7, 'h': 8, 'i': 9, 'j': 10, 'k': 11, 'l': 12, 'm': 13, 'n': 14, 'o': 15, 'p': 16, 'q': 17, 'r': 18, 's': 19, 't': 20, 'u': 21, 'v': 22, 'w': 23, 'x': 24, 'y': 25, 'z': 26}
(n, q) = map(int, input().split())
s = str(input())
arr = []
arr.append(d[s[0]])
for i in range(1, n):
	x = d[s[i]]
	y = arr[-1]
	arr.append(x + y)
for i in range(q):
	(a, b) = map(int, input().split())
	if a == 1:
		if b == n:
			print(arr[-1])
		else:
			print(arr[b - 1])
	else:
		print(arr[b - 1] - arr[a - 2])
