(n, x) = map(int, input().split())
l = list(map(int, input().split()))
s = 0
l1 = []
for i in l:
	s = s + i
	if s == 0:
		l1 = []
	else:
		l1.append(i)
s1 = 0
c = 0
for i in l1:
	s1 = s1 + i
if s1 == 0:
	print(0)
elif abs(s1) <= x:
	print(1)
else:
	s1 = abs(s1)
	z = int(s1 // x)
	c = c + z
	y = s1 % x
	if y > 0:
		c = c + 1
	print(c)
