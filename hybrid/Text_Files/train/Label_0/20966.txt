#include <bits/stdc++.h>
using namespace std;
string s[8][8];
bool mark[8][8][8];
bool f(int t, int i, int j) {
  if (i < 0 || j < 0 || i == 8 || j == 8) return false;
  return mark[t - 1][i][j];
}
bool ch(int t, int i, int j) {
  bool k = false;
  if (s[t][i][j] != 'S' && s[t - 1][i][j] != 'S')
    for (int x = -1; x < 2; x++)
      for (int y = -1; y < 2; y++) k = (k || f(t, i + x, j + y));
  return k;
}
int main() {
  for (int i = 0; i < 8; i++) cin >> s[0][i];
  mark[0][7][0] = true;
  for (int i = 0; i < 8; i++)
    for (int j = 0; j < 8; j++) s[i][j] += "........";
  for (int i = 0; i < 8; i++)
    for (int j = 0; j < 8; j++)
      if (s[0][i][j] == 'S')
        for (int t = 1; t < 8; t++) {
          if (i + t == 8) break;
          s[t][i + t][j] = 'S';
        }
  for (int t = 1; t < 8; t++)
    for (int i = 0; i < 8; i++)
      for (int j = 0; j < 8; j++) mark[t][i][j] = ch(t, i, j);
  bool k = false;
  for (int i = 0; i < 8; i++)
    for (int j = 0; j < 8; j++)
      if (mark[7][i][j]) k = true;
  cout << ((k) ? "WIN" : "LOSE") << endl;
  return 0;
  return 0;
}
