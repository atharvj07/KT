#include <bits/stdc++.h>
using namespace std;
int main() {
  int m, n;
  cin >> m;
  cin >> n;
  int flag = 0;
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
      char var;
      cin >> var;
      if ((flag == 0) && ((var == 'C') || (var == 'M') || (var == 'Y'))) {
        flag = 1;
      }
    }
  }
  if (flag == 1) {
    cout << "#Color" << endl;
  } else {
    cout << "#Black&White" << endl;
  }
}
