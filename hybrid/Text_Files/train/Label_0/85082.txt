n = int(input())
d = {}
d1 = {}
l = set()
ans = [0] * n
for i in range(n):
	(a, b) = map(int, input().split())
	d[a] = b
	d1[b] = a
	l.add(a)
	l.add(b)
i = 1
b = 0
l.remove(0)
while d.get(b, 0) != 0:
	b = d[b]
	l.remove(b)
	ans[i] = b
	i += 2
if n % 2 == 0:
	i = n - 2
	b = 0
	while d1.get(b, 0) != 0:
		b = d1[b]
		ans[i] = b
		i -= 2
else:
	for x in l:
		if d.get(x, -1) == -1:
			i = n - 1
			b = x
			while b != -1:
				ans[i] = b
				b = d1.get(b, -1)
				i -= 2
			break
print(*ans)
