#include <bits/stdc++.h>
using namespace std;
long long RP(long long a, long long b, long long M) {
  long long r = 0;
  while (b > 0) {
    if (b & 1) r = (r + a) % M;
    a = a << 1;
    a = a % M;
    b = b >> 1;
  }
  return r % M;
}
long long bMwithRP(long long a, long long b, long long M) {
  long long r = 1LL;
  while (b) {
    if (b & 1) r = RP(r, a, M);
    b >>= 1, a = RP(a, a, M);
  }
  return r % M;
}
long long add(long long a, long long b) {
  if (a < 0) a += 1000000007;
  if (b < 0) b += 1000000007;
  long long res = a % 1000000007 + b % 1000000007;
  while (res < 0) res += 1000000007;
  while (res > 1000000007) res %= 1000000007;
  return (res + 1000000007) % 1000000007;
}
int main() {
  long long n, k;
  cin >> n >> k;
  if (!n) {
    puts("0");
    return 0;
  }
  long long res = bMwithRP(2, k + 1, 1000000007);
  res = RP(res, n, 1000000007);
  long long res2 = bMwithRP(2, k, 1000000007);
  res = add(res - res2, 1);
  cout << res << endl;
  return 0;
}
