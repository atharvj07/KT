#include <bits/stdc++.h>
using namespace std;
const bool testing = true;
void program() {
  int n, s, t;
  cin >> n >> s >> t;
  int x = s;
  int y = t;
  vector<int> v(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> v[i];
  }
  int cn = 0;
  if (s == t) {
    cout << "0" << endl;
    return;
  }
  while (v[x] != t) {
    if (v[x] == s) {
      cout << "-1" << endl;
      return;
    } else {
      cn++;
      x = v[x];
    }
  }
  cout << cn + 1 << endl;
}
