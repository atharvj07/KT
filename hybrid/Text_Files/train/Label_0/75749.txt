def egcd(a, b):
	(x, y, u, v) = (0, 1, 1, 0)
	while a != 0:
		(q, r) = (b // a, b % a)
		(m, n) = (x - u * q, y - v * q)
		(b, a, x, y, u, v) = (a, r, u, v, m, n)
	gcd = b
	return (gcd, x, y)

def solve():
	n = inp()
	ch = 0
	ch_turn = True
	while n > 0:
		if n % 4 == 0 and n > 8:
			n -= 1
			if ch_turn:
				ch += 1
		elif n % 2 == 0:
			n //= 2
			if ch_turn:
				ch += n

			n -= 1

				ch += 1
		ch_turn = not ch_turn
