n = int(input())
A = list(map(int, input().split()))
B = list(map(int, input().split()))
X = []
for i in range(n):
	X.append([A[i], 1])
	X.append([B[i], 2])
X.sort(reverse=True)
ans = 0
for i in range(len(X)):
	if i % 2 == 0:
		if X[i][1] == 1:
			ans += X[i][0]
	elif X[i][1] == 2:
		ans -= X[i][0]
print(ans)
