import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long l1 = sc.nextLong();long r1 = sc.nextLong();
    long l2 = sc.nextLong();long r2 = sc.nextLong();
    long k = sc.nextLong();
    long time = Math.min(r2,r1) - Math.max(l1,l2) + 1;
    if(k<=Math.min(r2,r1) && k >= Math.max(l1,l2)){
      time = time -1;
    }
    System.out.println(Math.max(0,time));
  }
}
