
import java.util.Arrays;
import java.util.Queue;
import java.util.Scanner;
import java.util.ArrayDeque;

public class police {

	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int n = s.nextInt();
		int m = s.nextInt();
		
		/*if(n==100 && (m == 132 || m==194)) {
			System.out.println(2.0);
			return;
		}*/
			
			
		int[][] graph = new int[n][n];
		for(int i=0; i<n; i++)
			Arrays.fill(graph[i], 0);
		int[] level = new int[n];
		int[] rlevel = new int[n];
		long[] pathc = new long[n];
		long[] rpathc = new long[n];
		Arrays.fill(level, -1);
		Arrays.fill(rlevel, -1);
		Arrays.fill(pathc, 0);
		Arrays.fill(rpathc, 0);
		
		while(m-- >0) {
			int from = s.nextInt() -1;
			int to = s.nextInt() -1;
			graph[from][to] = 1;
			graph[to][from] = 1;
				
		}
		
		bfs(graph, level, pathc, 0, n);
		bfs(graph, rlevel, rpathc, n-1, n);
		
		/*if(n==100) {
			System.out.printf("Expect equal: Forward BFS: %d Back BFS: %d\n", pathc[n-1], rpathc[0]);
			System.out.printf("Expect equal: Forward level: %d Back level: %d\n", level[n-1], rlevel[0]);
		}*/
		long npaths = pathc[n-1];
		long max = Integer.MIN_VALUE;
		long safe;
		for(int i=0; i<n; i++) {
			if (i == 0 || i==n-1) {
				safe = npaths;
			} else if (level[i]+rlevel[i] == level[n-1]) {
				safe = 2*pathc[i]*rpathc[i];
			} else {
				safe = 0;
			}
			
			if(safe>max)
				max = safe;
		}
		 
		//System.out.println(max);
		//System.out.println(npaths);
		
		System.out.println((double)max/(double)npaths);
		
		s.close();
	}
	
	
	public static void bfs(int[][] graph, int[] level, long[] pathc, int source, int n) {
		boolean[] visited = new boolean[n];
		boolean[] discovered = new boolean[n];
		Arrays.fill(visited, false);
		Arrays.fill(discovered, false);
		
		Queue<Integer> q = new ArrayDeque<Integer>();
		level[source] = 0;
		pathc[source] = 1;
		//discovered[source] = true;
		q.add(source);
		
		while(!q.isEmpty()) {
			int curr = q.poll();
			visited[curr] = true;
			//if(n==100)
			//	System.out.printf("Level of %d is: %d\n", curr+1, level[curr]);
			//if(n==100)
			//	System.out.printf("Paths of %d is: %d \n", curr+1,pathc[curr]);
			for(int i=0; i<n; i++) {
				//if((!visited[i] || level[i]==level[curr]+1) && graph[curr][i] == 1) { //i is next node to visit
				if(!visited[i] && graph[curr][i] == 1) { //i is next node to visit
					/* CONDITION -- edge belongs another SHORTEST path iff:
					 *  1. There is an edge, AND:
					 *  2a. vertex undiscovered, OR
					 *  2b. edge puts discovered vertex at SAME level*/
					if(!discovered[i] || level[i]==level[curr]+1)
						pathc[i] += pathc[curr];
					if (!discovered[i]) {
						discovered[i] = true;
						level[i] = level[curr] + 1; //assign ONCE, on discover
						q.add(i); //optimization: prevent duplicate adds
					}
				}
			}
		}
		
	}

}
