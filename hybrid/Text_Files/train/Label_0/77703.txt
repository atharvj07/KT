def Superman():
	ScarletWitch = list(map(int, input().split()))
	(Deadpool, Death, Falcon, Cap) = ScarletWitch
	if Deadpool > Death:
		Wolverine = Deadpool
	else:
		Wolverine = Death
	if Falcon > Cap:
		DoctorStrange = Falcon
	else:
		DoctorStrange = Cap
	ScarletWitch.sort()
	Jormungandr = [Wolverine, DoctorStrange]
	Jormungandr.sort()
	if ScarletWitch[2] == Jormungandr[0] and ScarletWitch[3] == Jormungandr[1]:
		print('YES')
	else:
		print('NO')
