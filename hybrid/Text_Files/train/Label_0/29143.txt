#include <bits/stdc++.h>
using namespace std;
char buf[110];
string input(void) {
  gets(buf);
  string s = buf;
  return s;
}
int input_int(void) {
  string s = input();
  return atoi(s.c_str());
}
vector<string> parsed_input(void) {
  string s = input();
  int i;
  for ((i) = 0; (i) < (int)(s.length()); (i)++)
    if (s[i] == '(' || s[i] == ')' || s[i] == ',') s[i] = ' ';
  istringstream ss(s);
  vector<string> ans;
  while (ss >> s) ans.push_back(s);
  return ans;
}
string name[1010];
vector<string> types[1010];
map<string, string> mp;
bool check(int id, vector<string> &v) {
  if (v.size() != types[id].size() + 1) return false;
  if (name[id] != v[0]) return false;
  int i;
  for ((i) = 0; (i) < (int)(types[id].size()); (i)++)
    if (types[id][i] != "T" && mp[v[i + 1]] != types[id][i]) return false;
  return true;
}
int main(void) {
  int N, V, Q, i, j;
  N = input_int();
  for ((i) = 0; (i) < (int)(N); (i)++) {
    vector<string> v = parsed_input();
    name[i] = v[1];
    for (j = 2; j < v.size(); j++) types[i].push_back(v[j]);
  }
  V = input_int();
  for ((i) = 0; (i) < (int)(V); (i)++) {
    vector<string> v = parsed_input();
    mp[v[1]] = v[0];
  }
  Q = input_int();
  for ((i) = 0; (i) < (int)(Q); (i)++) {
    vector<string> v = parsed_input();
    int ans = 0;
    for ((j) = 0; (j) < (int)(N); (j)++)
      if (check(j, v)) ans++;
    cout << ans << endl;
  }
  return 0;
}
