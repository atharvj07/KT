#include <bits/stdc++.h>
using namespace std;
const int N = 5e3 + 5, M = 26 + 5, mod = 1e9 + 7;
long long dp[N][M], res[N], fact[N], rev[N];
long long sum(long long a, long long b) {
  a += b;
  if (a < 0) a += mod;
  if (a >= mod) a -= mod;
  return a;
}
long long pw(long long a, long long b) {
  if (!b) return 1;
  long long ans = pw(a, b / 2);
  ans *= ans;
  ans %= mod;
  if (b & 1) ans *= a;
  return ans % mod;
}
void prep() {
  fact[0] = 1;
  for (int i = 1; i < N; i++) fact[i] = (fact[i - 1] * i) % mod;
  rev[N - 1] = pw(fact[N - 1], mod - 2);
  for (int i = N - 2; ~i; i--) rev[i] = (rev[i + 1] * (i + 1)) % mod;
}
long long c(int n, int k) {
  long long ans = fact[n] * rev[k];
  ans %= mod;
  ans *= rev[n - k];
  return ans % mod;
}
int main() {
  int n;
  string s;
  cin >> n >> s;
  prep();
  for (int j = 0; j < n; j++) {
    const int x = s[j] - 'a';
    if (!dp[1][x]) {
      dp[1][x] = 1;
      res[1]++;
    }
    for (int i = 1; i < n; i++) {
      res[i + 1] = res[i + 1] - dp[i + 1][x];
      dp[i + 1][x] = res[i] - dp[i][x] + mod;
      dp[i + 1][x] %= mod;
      res[i + 1] += dp[i + 1][x];
      res[i + 1] %= mod;
      res[i + 1] += mod;
      res[i + 1] %= mod;
    }
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    ans += 1LL * res[i + 1] * c(n - 1, i) % mod;
    ans %= mod;
  }
  cout << ans << "\n";
}
