#include<iostream>
#include<vector>
#include<complex>
#include<algorithm>
using namespace std;

#define REP(i, n) for(int i = 0; i < (int)(n); i++)

typedef complex<double> P;
const double EPS = 1e-8;

double dot(P a, P b) {return real( conj(a)*b);}
double cross(P a, P b) {return imag( conj(a)*b);}

namespace std {
  bool operator < (const P& a, const P& b) {
    return real(a) != real(b) ? real(a) < real(b) : imag(a) < imag(b);
  }
}

int ccw(P a, P b, P c) {
  b -= a; c -= a;
  if (cross(b, c) > EPS)   return +1;       // counter clockwise
  if (cross(b, c) < -EPS)   return -1;       // clockwise
  if (dot(b, c) < -EPS)     return +2;       // c--a--b on line
  if (norm(b) < norm(c) - EPS) return -2; // a--b--c on line
  return 0;                               // a--c--b on line (or b == c)
}

int WindingNumber (const vector<P>& g, P p) {
  int wn = 0;
  int n = g.size();
  REP(i, n) {
    int j = (i + 1) % n;
    if( g[i].imag() <= p.imag() + EPS ) {
      if( g[j].imag() > p.imag() + EPS && ccw(g[i], g[j], p) > 0 )
        wn++;
    }
    else {
      if( g[j].imag() <= p.imag() + EPS && ccw(g[i], g[j], p) < 0) {
        wn--;
      }
    }
  }
  return wn;
}

bool IsIntersectSP(P a, P b, P c) {
  return (abs(a-c) + abs(c-b) <= abs(a-b) + EPS);
}

enum {OUT, CROSS, IN};
int ContainGS (const vector<P>& g, P p) {
  REP(i, g.size()) {
    if( IsIntersectSP( g[i], g[(i+1) % g.size()], p) )
      return CROSS;
  }
  int wn = WindingNumber(g, p);
  return wn > 0;
}

vector<P> ConvexHull(vector<P> g) {
  int n = g.size();
  if( n <= 1 ) return g;
  sort(g.begin(), g.end());
  vector<P> res(2 * n);
  int k = 0;
  for(int i = 0; i < n; res[k++] = g[i++] )
    while( k >= 2 && ccw( res[k-2], res[k-1], g[i]) <= 0 ) k--;
  for(int i = n - 2, t = k + 1; i >= 0; res[k++] = g[i--] )
    while( k >= t && ccw( res[k-2], res[k-1], g[i]) <= 0 ) k--;
  res.resize(k-1);
  return res;
}

bool IsIntersectSS(P a1, P a2, P b1, P b2) {
  return ccw(a1,a2,b1) * ccw(a1,a2,b2) <= 0 &&
    ccw(b1,b2,a1) * ccw(b1,b2,a2) <= 0;
}

int IsIntersectGG(const vector<P> &g1, const vector<P>& g2) {
  if( g1.size() <= 1 ) {
    if( ContainGS( g2, g1[0] ) ) return IN;
    return OUT;
  }

  if( g2.size() <= 1 ) {
    if( ContainGS( g1, g2[0] ) ) return IN;
    return OUT;
  }

  REP(i, g1.size()) {
    REP(j, g2.size()) {
      if( IsIntersectSS( g1[i], g1[(i+1) % g1.size()],
                         g2[j], g2[(j+1) % g2.size()]) )
        return CROSS;
    }
  }
  if( ContainGS( g1, g2[0] ) || ContainGS( g2, g1[0] )) {
    return IN;
  }
  return OUT;
}

int main() {
  int A, B;
  while( cin >> A >> B ) {
    if( !(A || B) ) break;
    vector<P> a, b;
    REP(i, A) {
      int x, y; cin >> x >> y;
      a.push_back( P(x,y) );
    }
    REP(i, B) {
      int x, y; cin >> x >> y;
      b.push_back( P(x,y) );      
    }
    a = ConvexHull(a);
    b = ConvexHull(b);

    int t = IsIntersectGG(a,b);
    if( t == OUT ) puts("YES");
    else puts("NO");
  }
}
