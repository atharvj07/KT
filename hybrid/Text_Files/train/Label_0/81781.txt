from sys import stdin, stdout
from math import ceil
for _ in range(int(stdin.readline())):
	(n, k) = [int(x) for x in stdin.readline().split()]
	p = [int(x) for x in stdin.readline().split()]
	a = p[0]
	s = 0
	for i in range(1, n):
		if p[i] / a > k / 100:
			s = max(s, ceil(p[i] * 100 / k) - a)
		a += p[i]
	print(s)
