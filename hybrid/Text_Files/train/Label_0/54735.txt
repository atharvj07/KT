import java.io.*;
import java.util.*;

public class cf283E {
    public static void main(String[] args) throws Exception {
    	// BufferedReader in = new BufferedReader(new FileReader("cf283E.in"));
    	BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    	
		int n = Integer.parseInt(in.readLine());
		PriorityQueue<Seg> pqseg = new PriorityQueue<Seg>();
		for(int i = 0; i < n; i++) {
			String str = in.readLine();
			int a = Integer.parseInt(str.substring(0, str.indexOf(" ")));
			int b = Integer.parseInt(str.substring(str.indexOf(" ") + 1));
			pqseg.add(new Seg(false, a, b, 0, i));
		}
		int m = Integer.parseInt(in.readLine());
		for(int i = 0; i < m; i++) {
			String str = in.readLine();
			int a = Integer.parseInt(str.substring(0, str.indexOf(" ")));
			int b = Integer.parseInt(str.substring(str.indexOf(" ") + 1, str.lastIndexOf(" ")));
			int c = Integer.parseInt(str.substring(str.lastIndexOf(" ") + 1));
			pqseg.add(new Seg(true, a, b, c, i));
		}
		int[] assign = new int[n];
		boolean works = true;
		TreeSet<Actor> seta = new TreeSet<Actor>();
		while(!pqseg.isEmpty() && works) {
			Seg s = pqseg.remove();
			if(s.isActor) {
				seta.add(new Actor(s, s.end, s.k));
			} else {
				Actor temp = seta.ceiling(new Actor(null, s.end, 0));
				if(temp == null) {
					works = false;
				} else {
					temp.k--;
					if(temp.k == 0) {
						seta.remove(temp);
					}
					assign[s.id] = temp.s.id;
				}
			}
		}
		System.out.println(works ? "YES" : "NO");
		if(works) {
			for(int i = 0; i < n; i++) {
				System.out.print(assign[i] + 1 + (i == n - 1 ? "\n" : " "));
			}
		}
	}

	static class Actor implements Comparable<Actor> {
		Seg s;
		int end;
		int k;

		public Actor(Seg temps, int e, int tempk) {
			s = temps;
			end = e;
			k = tempk;
		}

		public int compareTo(Actor other) {
			if(this.end != other.end) {
				return this.end - other.end;
			}
			if(this.s == null || other.s == null) {
				return 0;
			}
			return this.s.id - other.s.id;
		}
	}

	static class Seg implements Comparable<Seg> {
		boolean isActor;
		int start;
		int end;
		int k;
		int id;

		public Seg(boolean iA, int s, int e, int tempk, int i) {
			isActor = iA;
			start = s;
			end = e;
			k = tempk;
			id = i;
		}

		public int compareTo(Seg other) {
			if(this.start != other.start) {
				return this.start - other.start;
			}
			if(this.isActor && !other.isActor) {
				return -1;
			}
			if(!this.isActor && other.isActor) {
				return 1;
			}
			if(this.end != other.end) {
				return this.end - other.end;
			}
			return this.id - other.id;
		}
	}
}
