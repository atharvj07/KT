#include <bits/stdc++.h>
using namespace std;
long long int dp[255][105][105];
long long int c[255][255];
long long int c1[1005];
long long int N;
string s;
long long int solve(char j, long long int i, long long int k) {
  if (i == N) return 0;
  if (dp[j][i][k] != -1) return dp[j][i][k];
  long long int res = solve(s[i], i + 1, k) + c[j][s[i]];
  if (k) {
    for (char x = 'a'; x <= 'z'; x++)
      res = max(res, solve(x, i + 1, k - 1) + c[j][x]);
  }
  return dp[j][i][k] = res;
}
int main() {
  char x, y;
  long long int z, k, m;
  cin >> s >> k >> m;
  N = s.size();
  for (long long int i = 0; i < m; i++) {
    cin >> x >> y >> z;
    c[x][y] = z;
  }
  memset(dp, -1, sizeof(dp));
  long long int ans = solve('*', 0, k);
  printf("%d\n", ans);
  return 0;
}
