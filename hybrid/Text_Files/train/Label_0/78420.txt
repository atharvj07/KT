n = int(input())
a = list(map(int, input().split()))
a = sorted(a, reverse=True)
last_w = float('INF')
k = 0
for i in range(n):
	if a[i] + 1 < last_w:
		last_w = a[i] + 1
		k += 1
		continue
	elif a[i] < last_w:
		last_w = a[i]
		k += 1
		continue
	elif a[i] - 1 < last_w:
		if a[i] - 1 > 0:
			last_w = a[i] - 1
			k += 1
			continue
print(k)
