from collections import Counter
inp = lambda : map(int, input().rstrip().split())
(n, m) = inp()
a = list(inp())
dp = [0] * (n + 1)
c = Counter()
for i in range(n - 1, -1, -1):
	if c[a[i]] == 0:
		dp[i] = dp[i + 1] + 1
	else:
		dp[i] = dp[i + 1]
	c[a[i]] += 1
for _ in range(m):
	l = int(input())
	print(dp[l - 1])
