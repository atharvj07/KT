#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int x = 0, f = 1;
  char c = getchar();
  for (; !isdigit(c); c = getchar())
    if (c == '-') f = -1;
  for (; isdigit(c); c = getchar()) x = x * 10 + c - '0';
  return x * f;
}
const int MAXN = 3010;
int main() {
  int Cas = read();
  while (Cas--) {
    int N = read(), K = read();
    if (((N - (K - 1)) & 1) && (N - (K - 1) > 0)) {
      printf("YES\n");
      for (int i = 1; i < K; i++) printf("1 ");
      printf("%d\n", N - (K - 1));
      continue;
    }
    if (((N - (K - 1) * 2) % 2 == 0) && (N - (K - 1) * 2 > 0)) {
      printf("YES\n");
      for (int i = 1; i < K; i++) printf("2 ");
      printf("%d\n", N - (K - 1) * 2);
      continue;
    }
    printf("NO\n");
  }
  return 0;
}
