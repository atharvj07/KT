#include <bits/stdc++.h>
using namespace std;
int mod = 1e9 + 7;
void readFile() {}
long long int power(long long int a, long long int b) {
  long long int ans = 1;
  while (b > 0) {
    if (b & 1) ans = ans * a;
    a *= a;
    b = b >> 1;
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  readFile();
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n % 4 == 0)
      cout << "YES";
    else
      cout << "NO";
    cout << '\n';
  }
}
