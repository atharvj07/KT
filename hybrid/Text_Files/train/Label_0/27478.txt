#include <bits/stdc++.h>
using namespace std;
string s[30];
int r, c, k;
bool all;
int main() {
  for (int i = 0; i < 27; i++) {
    cin >> s[i];
  }
  cin >> r >> c;
  all = true;
  r = (r - 1) % 3;
  c = (c - 1) % 3;
  for (int i = r * 9 + c, k = 0; k < 3; i += 3, k++) {
    for (int j = 0; j < 3; j++) {
      if (s[i][j] == '.') {
        all = false;
      }
    }
  }
  if (all) {
    for (int i = 0; i < 27; i++) {
      for (int j = 0; j < 3; j++) {
        if (s[i][j] == '.') s[i][j] = '!';
      }
    }
  } else {
    for (int i = r * 9 + c, k = 0; k < 3; i += 3, k++) {
      for (int j = 0; j < 3; j++) {
        if (s[i][j] == '.') s[i][j] = '!';
      }
    }
  }
  for (int i = 0; i < 27; i += 3) {
    if (i > 0 && i % 9 == 0) {
      cout << "\n";
    }
    for (int j = i; j < i + 3; j++) {
      cout << s[j] << " ";
    }
    cout << "\n";
  }
  return 0;
}
