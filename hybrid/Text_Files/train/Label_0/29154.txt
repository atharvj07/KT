#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
map<string, int> mp;
int cost[N], mcost[N];
struct Union_Find {
  int pre[N], ranks[N];
  void Init(int n) {
    for (int i = 0; i <= n; i++) pre[i] = i, ranks[i] = 0;
  }
  int Find(int x) { return x == pre[x] ? x : Find(pre[x]); }
  void Union(int a, int b) {
    int x = Find(a);
    int y = Find(b);
    if (x == y) return;
    ranks[x] < ranks[y] ? (pre[x] = y) : (pre[y] = x);
    if (ranks[x] == ranks[y]) ranks[x]++;
  }
} un;
int main() {
  int n, m, k, x, first, next;
  while (cin >> n >> k >> m) {
    un.Init(n);
    mp.clear();
    string s;
    for (int i = 1; i <= n; i++) {
      cin >> s;
      mp[s] = i;
    }
    for (int i = 1; i <= n; i++) cin >> cost[i];
    while (k--) {
      cin >> x >> first;
      for (int i = 2; i <= x; i++) {
        cin >> next;
        un.Union(first, next);
      }
    }
    memset(mcost, 0x3f3f3f3f, sizeof(mcost));
    for (int i = 1; i <= n; i++)
      mcost[un.Find(i)] = min(mcost[un.Find(i)], cost[i]);
    long long ans = 0;
    for (int i = 1; i <= m; i++) {
      cin >> s;
      ans += mcost[un.Find(mp[s])];
    }
    cout << ans << endl;
  }
  return 0;
}
