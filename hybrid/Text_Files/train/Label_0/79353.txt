a = input().split(' ')
b = input().split(' ')
n = int(a[0])
k = int(a[1])

def gen_name(i):
	name = ''
	offset = ord('A')
	while i > 0:
		name += chr(i % 26 + offset)
		offset = ord('a')
		i //= 26
	if name == '':
		name = 'A'
	return name
ans = [gen_name(i) for i in range(n)]
for i in range(n - k + 1):
	if b[i] == 'NO':
		ans[i + k - 1] = ans[i]
print(' '.join(ans))
