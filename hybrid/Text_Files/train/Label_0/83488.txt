import sys
input = lambda : sys.stdin.readline().rstrip()
s = input()
mas = []
k = 1
for i in s:
	p = [i, k]
	mas.append(p)
	k += 1
(l, r) = (0, len(s) - 1)
ans = []
while True:
	if len(mas) == 0:
		break
	if mas[l][0] != '(':
		l += 1
	if mas[r][0] != ')':
		r -= 1
	if r == 0 or l == len(mas) or r <= l:
		break
	if mas[l][0] == '(' and mas[r][0] == ')':
		ans.append(mas[l][1])
		ans.append(mas[r][1])
		del mas[l]
		del mas[r - 1]
		r = len(mas) - 1
		l = 0
if len(ans) == 0:
	print(0)
else:
	print(1)
	print(len(ans))
	print(*sorted(ans))
