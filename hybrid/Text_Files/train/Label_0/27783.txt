#include <bits/stdc++.h>
using namespace std;
const long long Maxn = 1e5 + 10;
long long sum[110][110][12], star[110][110][12];
long long n, q, c;
int main() {
  std::ios::sync_with_stdio(0);
  std::cin.tie(0);
  cin >> n >> q >> c;
  for (long long i = 1; i <= n; i++) {
    long long x, y, s;
    cin >> x >> y >> s;
    star[x][y][s]++;
  }
  for (long long i = 1; i <= 100; i++) {
    for (long long j = 1; j <= 100; j++) {
      for (long long k = 0; k <= c; k++) {
        sum[i][j][k] = sum[i - 1][j][k] + sum[i][j - 1][k] -
                       sum[i - 1][j - 1][k] + star[i][j][k];
      }
    }
  }
  for (long long i = 1; i <= q; i++) {
    long long xi1, yi1, xi2, yi2, t;
    cin >> t >> xi1 >> yi1 >> xi2 >> yi2;
    long long ans = 0;
    for (long long j = 0; j <= c; j++) {
      ans += ((j + t) % (c + 1)) *
             (sum[xi2][yi2][j] - sum[xi2][yi1 - 1][j] - sum[xi1 - 1][yi2][j] +
              sum[xi1 - 1][yi1 - 1][j]);
    }
    cout << ans << endl;
  }
  return 0;
}
