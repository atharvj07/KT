a = int(input())
for i in range(a):
	b = int(input())
	c = input()
	if c[0:4] == '2020':
		print('YES')
	elif c[0:3] == '202' and c[b - 1] == '0':
		print('YES')
	elif c[0:2] == '20' and c[b - 2:b] == '20':
		print('YES')
	elif c[0] == '2' and c[b - 3:b] == '020':
		print('YES')
	elif c[b - 4:b] == '2020':
		print('YES')
	else:
		print('NO')
