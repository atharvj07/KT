import java.io.PrintWriter;
import java.util.Scanner;
  
public class Main {
    Scanner in;
    PrintWriter out;
      
    void solve() {
    	long n = in.nextLong();
    	long ans = n + (n-1)*n/2;
    	
    	out.println(ans*6 + 1);
    }

      
    void run() {

            in = new Scanner(System.in);
            out = new PrintWriter(System.out);      

            solve();

            out.close();
    }
  
    public static void main(String[] args) {
        new Main().run();
          
    }
}
