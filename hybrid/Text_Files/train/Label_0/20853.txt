#include <bits/stdc++.h>
using namespace std;
int main() {
  int m;
  string s;
  cin >> m;
  cin >> s;
  set<int> added;
  set<int> addedc;
  set<int>::iterator it;
  int n = s.size();
  string f = "";
  char max = 'a';
  for (int i = 0; i < n - m + 1;) {
    char c = s[i];
    int pos = i;
    for (int k = i + 1; k < i + m; ++k) {
      if (c >= s[k]) {
        c = s[k];
        pos = k;
      }
    }
    added.insert(pos);
    if (c >= max) max = c;
    i = pos + 1;
  }
  for (int i = 0; i < n; i++) {
    if (added.find(i) == added.end() && s[i] < max) f = f + s[i];
  }
  while (!added.empty()) {
    it = added.begin();
    f = f + s[*it];
    added.erase(it);
  }
  sort(f.begin(), f.end());
  cout << f;
}
