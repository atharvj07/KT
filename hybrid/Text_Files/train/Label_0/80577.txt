n = int(input())
m = list(map(int, input().split()))
lst = []
count = 0
s = 0
for i in range(n):
	s += 1
	for q in range(s, n):
		if q == i:
			continue
		else:
			lst.append(m[q] + m[i])
lst.sort()
recordcount = 0
for i in range(len(lst) - 1):
	if lst[i] == lst[i + 1]:
		count += 1
		if count > recordcount:
			recordcount = count
	else:
		count = 0
print(recordcount + 1)
