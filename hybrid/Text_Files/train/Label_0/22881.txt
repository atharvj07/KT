#include <bits/stdc++.h>
using namespace std;

int a[2222];
int main() {
	long n, k, t = 0, b = 0, p = 1e9 + 7;
	cin >> n >> k;
	set<int> s;
	for (int i = 0; i < n; i++) {
		cin >> a[i];
		s.insert(a[i]);
		for (int j = 0; j < i; j++)
			if (a[i] < a[j]) t++;
	}
	for (int i = 0; i < n; i++)
		for (int j = 0; j < n; j++)
			if (a[i] < a[j]) b++;
	cout << (t * k + (k - 1)*k / 2 % p*b) % p << endl;
	return 0;
}
