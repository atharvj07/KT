#include <iostream>
using namespace std;
int  s[100000];
bool p(int X, int N, int M){
  bool clear = false;
  int i = 0;
  int L = 1;
  int cnt = 1;
  // 初手で最初の敵が倒せなきゃクリアできない
  // 逆に，初手で最初の敵が倒せさえすれば地道にレベルアップして倒せる
  if(s[0] >= L + X){
    return false;
  }else{
    while(!clear && cnt < M){
      // まず，ラスボスが倒せるんだったらつべこべ言わず倒してしまえ
      if(s[N-1] < L+X){
	clear=true;
	cnt++;
      }else{
	// まだラスボスが倒せない状態なのでレベル上げ．
	// abs(L-s[i]) が最小となるiをimaxとし，このimaxを求める．要するにs[i]が最もLに近く位置を求める．
	//   abs(L-s[0]) > abs(L-s[1]) > ... > abs(L-s[imax]) < abs(L-s[imax+1]) < ... < abs(L-s[n-1])
	// となるので，減少してかつ倒せる間 i を増やし続けていけば良い．
	int imax = -1;
	while(s[i] < L+X && i < M-1 && imax < 0){
	  if(abs(L-s[i]) < abs(L-s[i+1])){
	    //増加に転ずる地点があればそのiが最大
	    imax = i;
	  }else{
	    i++;
	  }
	}
	//倒せる敵がいるのでレベル上げ
	// cout << "| L : " << L << ", imax : " << imax << endl;
	L += max(1, X - abs(L - s[imax]));
	cnt++;
      }
    }
  }
  // M 回目の時点でクリアしてないなら Xmax 更新
  // cout << "X : " << X << ", cnt : " << cnt << endl;
  if(!clear){
    return true;
  }else{
    return false;
  }
}

int main(void){
  for(;;){
    int N, M;
    cin >> N >> M;
    if(N == 0 && M == 0){return 0;}
    for(int i = 0 ; i < N ; i++){
      cin >> s[i];
    }
    // 実際に敵を倒してみる
    // 今倒せる敵の中で最もレベルの上がるもの(一番Lに近いもの)を倒していくのが最短である
    // cout << "p(3): " << p(3,N,M) << endl
    // X は探索範囲が広いので二分探索する
    int left  = s[0]-1;
    int right = s[N-1];
    while(left < right){
      int mid = (left + right)/2;
      if(p(mid, N, M) == true){
        left  = mid + 1;
      }else{
	right = mid;
      }
    }
    left--;
    cout << ((left <= s[0]-1)?(-1):(left)) << endl;
  }
}
