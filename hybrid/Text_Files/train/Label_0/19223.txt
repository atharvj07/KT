#include <bits/stdc++.h>
int a[901], b[901] = {0};
int main() {
  int n, k, i, j, x, y, total;
  scanf("%d%d", &n, &k);
  for (i = 1; i <= k; i++) {
    scanf("%d", &a[i]);
    b[a[i]] = 1;
  }
  x = n;
  y = 1;
  while (k--) {
    total = 0;
    for (j = 1; j <= 901; j++) {
      if (total < x - 1) {
        if (b[j] == 0) {
          printf("%d ", j);
          b[j] = 1;
          total++;
        }
      } else
        break;
    }
    printf("%d\n", a[y++]);
  }
  return 0;
}
