from sys import stdin
item = lambda : stdin.readline().split()
it = lambda : stdin.readline()[:-1]
for _ in range(int(stdin.readline())):
	(n, a, b) = map(int, item())
	s = it()
	x = s.find('1')
	if x == -1:
		print(n * (a + b) + b)
		continue
	res = (n + 1) * (a + b) + b
	for y in range(x + 1, n):
		if s[y] == '1':
			if y - x - 1 > 1:
				if 2 * a > (y - x - 2) * b:
					res += (y - x - 2) * b
				else:
					res += 2 * a
			res += b
			x = y
		if s[y] == '0' and s[y - 1] == '1':
			res += b
	res += a
	print(res)
