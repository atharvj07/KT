n = input()
s = str(input())
ans = 0
tmp = 0
foundedRight = False
for i in range(len(s)):
	if not foundedRight:
		if s[i] == '.':
			tmp = tmp + 1
		elif s[i] == 'L':
			tmp = 0
		elif s[i] == 'R':
			ans = ans + tmp
			tmp = 0
			foundedRight = True
	elif s[i] == '.':
		tmp = tmp + 1
	elif s[i] == 'R':
		tmp = 0
	elif s[i] == 'L':
		foundedRight = False
		if tmp % 2 == 1:
			ans = ans + 1
		tmp = 0
if not foundedRight:
	ans = ans + tmp
print(ans)
