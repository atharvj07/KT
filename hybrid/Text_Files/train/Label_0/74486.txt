q = int(input())
for _ in range(q):
	(a, b, m) = map(int, input().split())
	mn = [a]
	mx = [a]
	sum_mn = a
	sum_mx = a
	ok = False
	for n in range(51):
		if mn[-1] <= b <= mx[-1]:
			ok = True
			break
		mn.append(sum_mn + 1)
		mx.append(sum_mx + m)
		sum_mn += mn[-1]
		sum_mx += mx[-1]
		if mn[-1] > b:
			break
	if not ok:
		print(-1)
		continue
	s = b - mn[-1]
	r = [0] * n
	for i in range(n):
		x = s // max(1, 2 ** (n - i - 2))
		if x > m - 1:
			x = m - 1
		r[i] = x + 1
		s -= x * max(1, 2 ** (n - i - 2))
	res = [a]
	s = a
	for i in range(n):
		res.append(s + r[i])
		s += res[-1]
	print(len(res), *res)
