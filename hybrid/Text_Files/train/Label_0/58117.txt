import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class CandiesForChildren2 {
    
    int LIMIT = (int) 1e6;
    long INF = (long) 1e18;
    
    void solve() {
        long n = in.nextLong(), l = in.nextLong() - 1, r = in.nextLong() - 1, k = in.nextLong();
        
        if (r < l) r += n;
        long a = r - l + 1, b = n - a;
        
        long ans = -1;
        if (n <= LIMIT) {
            for (long c = n; c >= 0; c--) {
                long rem = k % (n + c);
                if (rem == 0) rem = n + c;
                long x = rem - a, y = c - x;
                if (x >= 0 && x <= a && y >= 0 && y <= b) {
                    ans = Math.max(ans, c);
                    break;
                }
            }
            k++;
            for (long c = n; c > 0; c--) {
                long rem = k % (n + c);
                if (rem == 0) rem = n + c;
                long x = rem - a, y = c - x;
                if (x > 0 && x <= a && y >= 0 && y <= b) {
                    ans = Math.max(ans, c);
                    break;
                }
            }
        } else {
            for (long t = 0; t <= k / n; t++) {
                long c = k - n * t - a;
                if (c >= 0) {
                    if (t == 0) {
                        if (c <= a) ans = Math.max(ans, c + b);
                    } else {
                        long x = c, y = -c;
                        
                        long kl = -INF, kr = INF;
                        if (a - x >= 0) kr = Math.min(kr, (a - x) / t);
                        else kr = Math.min(kr, -((x - a + t - 1) / t));
                        if (x >= 0) kl = Math.max(kl, -x / t);
                        else kl = Math.max(kl, (-x + t - 1) / t);
                        if (y - b >= 0) kl = Math.max(kl, (y - b + t) / (t + 1));
                        else kl = Math.max(kl, (y - b) / (t + 1));
                        if (y >= 0) kr = Math.min(kr, y / (t + 1));
                        else kr = Math.min(kr, -((-y + t) / (t + 1)));
                        
                        if (kl <= kr) ans = Math.max(ans, x + y - kl);
                    }
                }
                c++;
                if (c >= 0) {
                    if (t == 0) {
                        if (c <= a) ans = Math.max(ans, c + b);
                    } else {
                        long x = c, y = -c;
                        long kl = -INF, kr = INF;
                        
                        if (a - x >= 0) kr = Math.min(kr, (a - x) / t);
                        else kr = Math.min(kr, -((x - a + t - 1) / t));
                        if (x - 1 >= 0) kl = Math.max(kl, (1 - x) / t);
                        else kl = Math.max(kl, (1 - x + t - 1) / t);
                        
                        if (y - b >= 0) kl = Math.max(kl, (y - b + t) / (t + 1));
                        else kl = Math.max(kl, (y - b) / (t + 1));
                        if (y >= 0) kr = Math.min(kr, y / (t + 1));
                        else kr = Math.min(kr, -((-y + t) / (t + 1)));
                        
                        if (kl <= kr) ans = Math.max(ans, x + y - kl);
                    }
                }
            }
        }
        out.println(ans);
        
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new CandiesForChildren2().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
