#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 5;
int a[N], b[N], c[N], d[N], e[N], len;
int m, k, n, s;
void out(int s, int x) {
  vector<int> v;
  for (int i = x; i < x + len && v.size() != len - k; ++i) {
    e[a[i]]++;
    if (e[a[i]] > d[a[i]]) v.push_back(i);
    if (v.size() == len - k) break;
  }
  cout << v.size() << endl;
  for (int i = 0; i < v.size(); ++i) cout << v[i] + 1 << " ";
  exit(0);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> m >> k >> n >> s;
  for (int i = 0; i < m; ++i) cin >> a[i];
  int ss = 0;
  for (int i = 0; i < s; ++i) {
    int x;
    cin >> x;
    d[x]++;
    if (d[x] == 1) ++ss;
  }
  len = m - k * n + k;
  int tek = 0;
  for (int i = 0; i < len; ++i) {
    c[a[i]]++;
    if (c[a[i]] == d[a[i]]) ++tek;
  }
  if (tek == ss) {
    out(s, 0);
  }
  for (int i = len; i < m; ++i) {
    int z = a[i - len];
    if (c[z] == d[z]) --tek;
    c[z]--;
    c[a[i]]++;
    if (c[a[i]] == d[a[i]]) ++tek;
    if (tek == ss && (i - len + 1) % k == 0) out(s, i - len + 1);
  }
  cout << -1;
}
