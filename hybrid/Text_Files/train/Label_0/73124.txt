from sys import stdin, stdout
from collections import defaultdict as dd
import math
input = stdin.readline
flush = stdout.flush

def o2(l):
	b = l[len(l) // 2:].copy() + l[:len(l) // 2].copy()
	return b

def o1(l):
	b = []
	for i in range(1, len(l) // 2 + 1):
		b.append(l[2 * i - 1])
		b.append(l[2 * i - 2])
	return b
tc = 1
for _ in range(tc):
	n = int(input())
	p = list(map(int, input().split()))
	ans = float('inf')
	x = p.copy()
	x.sort()
	if x == p:
		ans = -1
	q = p.copy()
	for i in range(n):
		if i % 2 == 0:
			q = o2(q)
		else:
			q = o1(q)
		if q == x:
			ans = min(ans, i)
			break
	for i in range(n):
		if i % 2 == 0:
			p = o1(p)
		else:
			p = o2(p)
		if p == x:
			ans = min(ans, i)
			break
	if ans == float('inf'):
		ans = -2
	print(ans + 1)
