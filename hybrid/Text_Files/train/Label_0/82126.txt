import sys
data = sys.stdin.read().splitlines()
(n, k, m) = map(int, data[0].split())
nums = list(map(int, data[1].split()))
nums.sort()
s = sum(nums)
mx = (s + min(m, k * n)) / n
for (idx, num) in enumerate(nums):
	if idx == n - 1:
		break
	if idx < m:
		s -= num
		add = min((n - idx - 1) * k, m - idx - 1)
		mx = max(mx, (s + add) / (n - idx - 1))
	else:
		break
print(mx)
