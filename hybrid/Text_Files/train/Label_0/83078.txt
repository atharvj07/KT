import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(a, b) = map(int, input().split())
	s = input().strip()
	s = s.strip('0')
	s = s.rstrip('0')
	x = a // b
	(ans, f) = (0, 0)
	l = s.split('1')
	for i in range(len(l)):
		t = len(l[i])
		if t > 0:
			if t * b < a:
				ans += t * b
			else:
				ans += a
	if '1' in s:
		ans += a
