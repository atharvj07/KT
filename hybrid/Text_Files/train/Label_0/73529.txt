for i in range(int(input())):
	n = int(input())
	mat = []
	for i in range(n):
		mat.append(list(map(str, input())))
	l = []
	for i in range(n):
		for j in range(n):
			if mat[i][j] == '*':
				l.append(i)
				l.append(j)
	(x, y, a, b) = l
	if x == a and x != n - 1:
		mat[x + 1][y] = '*'
		mat[a + 1][b] = '*'
	if x == a and x == n - 1:
		mat[x - 1][y] = '*'
		mat[a - 1][b] = '*'
	if y == b and y != n - 1:
		mat[x][y + 1] = '*'
		mat[a][b + 1] = '*'
	if y == b and y == n - 1:
		mat[x][y - 1] = '*'
		mat[a][b - 1] = '*'
	mat[a][y] = '*'
	mat[x][b] = '*'
	s = ''
	for i in range(n):
		for j in range(n):
			s += mat[i][j]
		print(s)
		s = ''
