#include <bits/stdc++.h>
#define rep(i, n) for(int i = 0; i < int(n); i++)
#define all(in) in.begin(),in.end()
using namespace std;

const int INF = 1<<30;
vector<int> num, dp, ans;
vector<vector<int>> G;

void dfs(int u){
    int i = lower_bound(all(dp), num[u]) - dp.begin();
    int save = dp[i]; dp[i] = num[u];
    ans[u] = lower_bound(all(dp), INF) - dp.begin();
    for(auto next : G[u]) if(ans[next] == 0) dfs(next);
    dp[i] = save;
}

int main(){
    int n; cin >> n;
    num.assign(n, 0); rep(i,n) cin >> num[i];
    dp.assign(n, INF);
    ans.assign(n, 0);
    G.assign(n, vector<int>());
    rep(i,n-1){
        int a, b; cin >> a >> b;
        a--; b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }

    dfs(0);

    rep(i,n) cout << ans[i] << endl;
}
