import sys
input = sys.stdin.buffer.readline
from math import gcd, log2, ceil
MAX_N = int(1e+16)
MAX_F = ceil(log2(MAX_N)) + 1
MOD = int(1000000000.0) + 7

def lcm(a, b):
	return a // gcd(a, b) * b

def solve():
	last_lcm = 1
	ans = 0
	for fi in range(2, MAX_F):
		cur_lcm = lcm(last_lcm, fi)
		include = N // last_lcm
		exclude = N // cur_lcm
		ans += fi * (include - exclude) % MOD
		if cur_lcm > N:
			break
		last_lcm = cur_lcm
	return ans % MOD
test_cases = int(input())
for test_case in range(test_cases):
	N = int(input())
	print(solve())
