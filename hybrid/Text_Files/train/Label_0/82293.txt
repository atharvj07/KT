n = int(input())
data = list(map(int, input().split(' ')))
data.sort()
flag3 = False
result3 = []
flag2 = False

mydict = {}
for num in data:
	mydict[num] = '*'
ref_list = [2 ** i for i in range(31)]
for idx in range(31):
	ref = ref_list[idx]
	if idx % 2 == 0:
		data_tmp = []
		for num in data:
			if mydict.get(num + 2 * ref) != None:
				if flag2 == False:
					flag2 = True
					result2 = [num, num + 2 * ref]
				data_tmp.append(num)
				if mydict.get(num + ref) != None:
					flag3 = True
					result3 = [num, num + ref, num + 2 * ref]
					break
	else:
		for num in data_tmp:
			if mydict.get(num + 2 * ref) != None:
				flag3 = True
				result3 = [num, num + ref, num + 2 * ref]
				break
		if flag3:
			break
if flag3:
	print(3)
	print(result3[0], result3[1], result3[2])
elif flag2:
	print(2)
	print(result2[0], result2[1])
else:
	idx = 0
	while idx < 31:
		ref = ref_list[idx]
		for num in data:
			if mydict.get(num + ref) != None:
				flag2 = True
				break
		if flag2:
			break
		idx += 2
	if flag2:
