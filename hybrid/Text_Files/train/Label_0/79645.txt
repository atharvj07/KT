def sieve(n):
	prime = [True for i in range(n + 1)]
	prime[0] = False
	prime[1] = False
	p = 2
	while p * p <= n:
		if prime[p] == True:
			for i in range(p * p, n + 1, p):
				prime[i] = False
		p += 1
	return prime

def factor(n, p):
	i = 2
	f = {}
	while i < len(p) and n > 1:
		while n > 1 and n % i == 0 and (p[i] == True):
			if i in f:
				f[i] += 1
			else:
				f[i] = 1
			n //= i
		i += 1
	return f
n = int(input())
s = sieve(n)
visited = {}
q = []
c = 0
for i in range(2, n + 1):
	f = factor(i, s)
	question = True
	for k in f:
		if visited.get(k ** f[k], False):
			question = False
			break
	if question:
		c += 1
		q.append(str(i))
	visited[i] = True
print(c)
print(' '.join(q))
