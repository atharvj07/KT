import math
(n, r) = map(int, input().split())
alpha = 2 * math.pi / n
beta = (math.pi - alpha) / 2
target = math.sin(alpha) / (2 * math.sin(beta))
(left, right) = (0, 1000000000)
while right - left > 10 ** (-6):
	R = (left + right) / 2
	val = R / (R + r)
	if val < target:
		left = R
	else:
		right = R
print((left + right) / 2)
