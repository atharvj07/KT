#include <bits/stdc++.h>
using namespace std;
using lint = long long int;
template<class T = int> using V = vector<T>;
template<class T = int> using VV = V< V<T> >;
template<class T> void assign(V<T>& v, int n, const T& a = T()) { v.assign(n, a); }
template<class T, class... Args> void assign(V<T>& v, int n, const Args&... args) { v.resize(n); for (auto&& e : v) assign(e, args...); }


int main() {
  cin.tie(nullptr); ios::sync_with_stdio(false);
  int n, m; cin >> n >> m;
  struct Edge { int to, d; };
  VV<Edge> g(n);
  for (int i = 0; i < m; ++i) {
    int s, t, d; cin >> s >> t >> d;
    g[s].push_back({t, d});
    g[t].push_back({s, d});
  }
  VV<> d; assign(d, n, n, (int) 1e9);
  for (int i = 0; i < n; ++i) d[i][i] = 0;
  for (int v = 0; v < n; ++v) for (const auto& e : g[v]) d[v][e.to] = min(d[v][e.to], e.d);
  for (int k = 0; k < n; ++k) for (int i = 0; i < n; ++i) for (int j = 0; j < n; ++j) d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
  V<> vs;
  for (int v = 0; v < n; ++v) if (g[v].size() & 1) vs.push_back(v);
  int N = vs.size();
  VV<> w; assign(w, 1 << N, N);
  for (int u = 0; u < N; ++u) for (int v = 0; v < N; ++v) w[u][v] = d[vs[u]][vs[v]];
  V<> dp; assign(dp, 1 << N, (int) 1e9);
  if (N) dp[0] = 0;
  for (int bit = 1; bit < 1 << N; ++bit) for (int i = 0; i < N; ++i) if (bit & 1 << i) for (int j = 0; j < N; ++j) if (bit & 1 << j) {
    dp[bit] = min(dp[bit], dp[bit ^ 1 << i ^ 1 << j] + w[j][i]);
  }
  int res = 0;
  for (int v = 0; v < n; ++v) for (const auto& e : g[v]) if (v < e.to) res += e.d;
  if (N) res += dp.back();
  cout << res << '\n';
}
