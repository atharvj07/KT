from sys import stdin, stdout
ip = lambda : int(stdin.readline())
inp = lambda : map(int, stdin.readline().split())
ips = lambda : stdin.readline().rstrip()
out = lambda x: stdout.write(str(x) + '\n')

def qpow(a, b, m):
	ans = 1
	while b:
		if b & 1:
			ans *= a
			ans %= m
		a *= a
		a %= m
		b >>= 1
	return ans
mod = 998244353
(n, k) = inp()
C = [0] * (n + 1)
C[k] = 1
for i in range(k + 1, n + 1):
	C[i] = C[i - 1] * (i - 1) % mod * pow(i - k, mod - 2, mod) % mod
op = []
for _ in range(n):
	(l, r) = inp()
	op.append(2 * l)
	op.append(2 * r + 1)
op.sort()
now = 0
ans = 0
for opt in op:
	if opt & 1 == 0:
		now += 1
		if now >= k:
			ans += C[now]
			ans %= mod
	else:
		now -= 1
print(ans)
