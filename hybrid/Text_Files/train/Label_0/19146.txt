#include<bits/stdc++.h>
using namespace std;
using Int = long long;
Int w,h;
Int cnt,sum,ans,tmp;
Int a[20][20];
Int us[20][20];
Int dp[20][20];
Int pn[20][20];
void inc(Int y,Int x){
  cnt+=!us[y][x];
  us[y][x]++;
}
void dec(Int y,Int x){
  us[y][x]--;
  cnt-=!us[y][x];
}
void show(){
  cout<<cnt<<" "<<sum<<":"<<tmp<<endl;
  for(Int i=0;i<h;i++)
    for(Int j=0;j<w;j++)
      cout<<us[i][j]<<" \n"[j==w-1];
  cout<<endl;
}

void dfs(Int y,Int x){
  Int ny=y+(x+1)/w,nx=(x+1)%w;
  if(ny==h||tmp+1>=ans) return;
  
  if(us[y][x]) dfs(ny,nx);
  if(!dp[y][x]) return;
  
  Int bc=cnt,k=dp[y][x];
  for(Int i=0;i<k;i++)
    for(Int j=0;j<k;j++)
      inc(y+i,x+j);

  if(bc<cnt){
    tmp++;
    if(cnt==sum) ans=min(ans,tmp);
    else if(tmp+1<ans)  dfs(ny,nx);
    tmp--;
  }
  
  for(Int i=0;i<k;i++)
    for(Int j=0;j<k;j++)
      dec(y+i,x+j);
}

signed main(){
  while(cin>>w>>h,w){
    
    cnt=sum=0;
    memset(a,0,sizeof(a));
    memset(us,0,sizeof(us));
    for(Int i=0;i<h;i++){
      for(Int j=0;j<w;j++){
	cin>>a[i][j];
	sum+=a[i][j];
	us[i][j]=!a[i][j];
      }
    }
    
    memset(dp,0,sizeof(dp));
    for(Int i=h-1;i>=0;i--)
      for(Int j=w-1;j>=0;j--)
	dp[i][j]=a[i][j]*(min({dp[i][j+1],dp[i+1][j],dp[i+1][j+1]})+1);
    
    for(Int i=h-1;i>=0;i--)
      for(Int j=w-1;j>=0;j--)
	if(dp[i][j]==dp[i+1][j+1]+1)
	  dp[i+1][j+1]=0;
    
    memset(pn,0,sizeof(pn));
    for(Int i=0;i<h;i++)
      for(Int j=0;j<w;j++)
	for(Int k=0;k<dp[i][j];k++)
	  for(Int l=0;l<dp[i][j];l++)
	    pn[i+k][j+l]++;

    Int pre=0;
    for(Int i=0;i<h;i++){
      for(Int j=0;j<w;j++){
	if(dp[i][j]&&pn[i][j]==1){
	  pre++;
	  for(Int k=0;k<dp[i][j];k++)
	    for(Int l=0;l<dp[i][j];l++)
	      inc(i+k,j+l);
	}
      }
    }

    if(0){
      cout<<pre<<" "<<sum<<endl;
      for(Int i=0;i<h;i++)
	for(Int j=0;j<w;j++)
	  cout<<dp[i][j]<<" \n"[j==w-1];
      cout<<flush;

      show();
    }

    ans=w*h*(cnt!=sum);tmp=0;
    dfs(0,0);
    
    cout<<pre+ans<<endl;
  }
  return 0;
}
