#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, h[2500], v[2500], p = 0;
  cin >> n;
  for (int i = 0; i < n * n; i++) {
    cin >> h[i] >> v[i];
  }
  for (int z = 0; z < n * n; z++) {
    for (int y = 0; y < z; y++) {
      if (h[z] == h[y] || v[z] == v[y]) {
        p++;
        break;
      }
    }
    if (p == 0) {
      cout << z + 1 << " ";
    } else {
      p = 0;
      h[z] = 0;
      v[z] = 0;
    }
  }
  cout << endl;
  return 0;
}
