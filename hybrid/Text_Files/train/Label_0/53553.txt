import java.io.*;
import java.util.HashMap;
import java.util.StringTokenizer;

/**
 * Created by sbabkin on 9/14/2015.
 */
public class SolverD {

    public static void main(String[] args) throws IOException {
        new SolverD().Run();
    }

    BufferedReader br;
    PrintWriter pw;
    StringTokenizer stok;

    private String nextToken() throws IOException {
        while (stok==null || !stok.hasMoreTokens()){
            stok = new StringTokenizer(br.readLine());
        }
        return stok.nextToken();
    }

    private int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    private long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    private double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    private void Run() throws IOException {
//        br=new BufferedReader(new FileReader("input.txt"));
//        pw=new PrintWriter("output.txt");
        br=new BufferedReader(new InputStreamReader(System.in));
        pw=new PrintWriter(new OutputStreamWriter(System.out));

        solve();
        pw.flush();
        pw.close();
    }

    final long mn= (long) 1e9;

    public long getHash(long a, long b, long c) {
        long n1=a-b;
        long n2=b-c;
        long result = n1*mn + n2;
        return result;
    }

    class State {

        int a;
        int b;
        int c;

        public State(int a, int b, int c, int len) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    HashMap<Long, State> start = new HashMap<Long, State>();
    HashMap<Long, State> fin = new HashMap<Long, State>();
    int[] mas1;
    int[] mas2;
    int[] mas3;
    State resStart=null;
    State resFin=null;

    private void solve() throws IOException {
        int n = nextInt();
        mas1=new int[n];
         mas2=new int[n];
         mas3=new int[n];
        for (int i=0; i<n; i++) {
            mas1[i]=nextInt();
            mas2[i]=nextInt();
            mas3[i]=nextInt();
        }
        int l1=n/2;
        pos=new byte[n];
        find(0, 0, 0, 0, 0, l1, start);
        find(0, 0, 0, 0, l1, n-l1, fin);
        for (State st : start.values()) {
            long v1=-st.a+st.b;
            long v2=-st.b+st.c;
            long hash = v1*mn + v2;
            State finState = fin.get(hash);
            if (finState!=null) {
                if (resStart==null || resStart.a+resFin.a<st.a+finState.a) {
                    resStart=st;
                    resFin=finState;
                }
            }
        }
        if (resStart==null) {
            pw.println("Impossible");
        } else {
            found=false;
            findPos(0,0,0,0,0,l1, resStart);
            for (int i=0; i<l1; i++) {
                if (pos[i]==0) {
                    pw.println("MW");
                } else if (pos[i]==1) {
                    pw.println("LW");
                } else if (pos[i]==2) {
                    pw.println("LM");
                }

            }
            found=false;
            findPos(0,0,0,0,l1,n-l1, resFin);
            for (int i=0; i<n-l1; i++) {
                if (pos[i]==0) {
                    pw.println("MW");
                } else if (pos[i]==1) {
                    pw.println("LW");
                } else if (pos[i]==2) {
                    pw.println("LM");
                }

            }
        }
    }

    boolean found = false;

    private void findPos(int a, int b, int c, int dep, int nach, int len, State resFin) {
        if (len<=dep ) {
            if (a==resFin.a && b==resFin.b && c==resFin.c){
                found = true;
            }
//            State st = new State(a,b,c,len);
//            long hash =getHash(a,b,c);
//            State curState = map.get(hash);
//            if (curState==null || curState.a<st.a) {
//                map.put(hash, st);
//            }
            return;
        }
        a+=mas1[nach+dep];
        b+=mas2[nach+dep];
        c+=mas3[nach+dep];
        pos[dep]=0;
        findPos(a - mas1[nach + dep], b, c, dep + 1, nach, len, resFin);
        if (found){
            return;
        }

        pos[dep]=1;
        findPos(a, b - mas2[nach + dep], c, dep + 1, nach, len, resFin);
        if (found){
            return;
        }
        pos[dep] = 2;
        findPos(a, b, c - mas3[nach + dep], dep+1, nach, len, resFin);
    }

    byte[] pos;



    private void find(int a, int b, int c, int dep, int nach, int len, HashMap<Long, State> map) {
        if (len<=dep) {
            State st = new State(a,b,c,len);
            long hash =getHash(a,b,c);
            State curState = map.get(hash);
            if (curState==null || curState.a<st.a) {
                map.put(hash, st);
            }
            return;
        }
        a+=mas1[nach+dep];
        b+=mas2[nach+dep];
        c+=mas3[nach+dep];
//        pos[dep]=0;
        find(a-mas1[nach+dep], b, c, dep+1, nach, len, map);
//        pos[dep]=1;
        find(a, b-mas2[nach+dep], c, dep+1, nach, len, map);
//        pos[dep]=2;
        find(a, b, c-mas3[nach+dep], dep+1, nach, len, map);

    }

}
