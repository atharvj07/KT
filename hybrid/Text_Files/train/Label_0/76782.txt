(n, p) = [float(x) for x in input().split()]
n = int(n)

def nck(n, k):
	if k > n:
		return 0
	tmp = 1
	for i in range(2, k):
		tmp *= k
	ans = 1
	for i in range(k):
		ans *= n - i
	return float(ans) / float(tmp)
for i in range(n + 1):
	known = i
	unknown = n - i
	ans = nck(unknown, 2) * nck(known, 1) * 0.5 + nck(unknown, 1) * nck(known, 2) * 1.0 + nck(known, 3) * 1.0
	ans /= max(nck(n, 3), 1)
	if ans >= p:
		print(i)
		break
