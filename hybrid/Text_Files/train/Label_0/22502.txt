#include <bits/stdc++.h>
using namespace std;
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }
const int maxn = 1010000;
int main() {
  int n, num = 1, a[110], maxx = 0, gcdd;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  maxx = a[1];
  gcdd = a[1];
  for (int i = 2; i <= n; i++) {
    gcdd = gcd(gcdd, a[i]);
    maxx = max(maxx, a[i]);
  }
  int ans = maxx / gcdd - n;
  if (ans & 1)
    cout << "Alice" << endl;
  else
    cout << "Bob" << endl;
  return 0;
}
