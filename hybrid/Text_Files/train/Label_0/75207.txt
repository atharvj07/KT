from math import gcd
for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	print(n - 1)
	(val, idx) = min(((val, idx) for (idx, val) in enumerate(a)))
	counter = 1
	for i in range(idx + 1, n):
		a[i] = val + counter
		counter += 1
		print(idx + 1, i + 1, val, a[i])
	counter = 1
	for i in range(idx - 1, -1, -1):
		a[i] = val + counter
		counter += 1
		print(idx + 1, i + 1, val, a[i])
