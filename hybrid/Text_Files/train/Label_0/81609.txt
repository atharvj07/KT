import io, os, sys
from atexit import register
from functools import cmp_to_key
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
sys.stdout = io.StringIO()
register(lambda : os.write(1, sys.stdout.getvalue().encode()))
tokens = []
tokens_next = 0

def next_str():
	global tokens, tokens_next
	while tokens_next >= len(tokens):
		tokens = input().split()
		tokens_next = 0
	tokens_next += 1
	return tokens[tokens_next - 1]

def nextInt():
	return int(next_str())
n = nextInt()
s = next_str()
s = list(map(int, list(s)))
res = s
for i in range(-10, 10):
	s_new = map(lambda x: (x + i + 10) % 10, s)
	s_new = list(s_new)
	best = sorted(list(range(n)), key=lambda x: s_new[x:] + s_new[0:x])[0]
	res = min(res, s_new[best:] + s_new[0:best])
print(''.join(map(str, res)))
