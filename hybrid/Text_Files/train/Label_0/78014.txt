n = int(input())
rat = []
flag = 0
for i in range(n):
	(a, b) = map(int, input().split())
	if a != b:
		flag = 1
	rat.append(a)

def issorted(rat):
	return all((rat[i] <= rat[i + 1] for i in range(0, n - 1)))

def issorted1(rat):
	return all((rat[i] >= rat[i + 1] for i in range(0, n - 1)))
if flag:
	print('rated')
elif issorted1(rat):
	print('maybe')
else:
	print('unrated')
