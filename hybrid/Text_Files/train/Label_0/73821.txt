for _ in range(int(input())):
	(n, x, y) = map(int, input().split())
	a = [x, y]
	if n > 2:
		n -= 2
		d = y - x
		i = 1
		dif = i
		l = []
		for i in range(1, d + 1):
			if d % i == 0:
				l.append(i)
		for i in l:
			if n >= d // i - 1:
				dif = i
				break
		start = x + dif
		while start < y and n > 0:
			a.append(start)
			n -= 1
			start += dif
		x -= dif
		while x > 0 and n > 0:
			a.append(x)
			x -= dif
			n -= 1
		y += dif
		while n > 0:
			n -= 1
			a.append(y)
			y += dif
	print(*a)
