#include <iostream>
#include <stdio.h>
using namespace std;
int s[26];
int b[3];
int main(void){
     int h,w;
     cin>>h>>w;
     for(int i=0;i<h*w;i++){
         char x;
         scanf(" %c",&x);
         s[x-'a']++;
     }
     b[0]=h*w%2;
     b[1]=(h%2)*(w/2)+(w%2)*(h/2);
     b[2]=(h/2)*(w/2);
     
     int l=4;
     for(int j=2;j>=0;j--,l/=2){
     for(int i=0;i<='z'-'a';i++){
         while(s[i]>=l&&b[j]){
             s[i]-=l;
             b[j]--;
         }
     }
     if(b[j]){
         cout<<"No"<<endl;
         return 0;
     }
     }
     cout<<"Yes"<<endl;
}
