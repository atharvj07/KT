using namespace std;

vector<int> a(12, 4);
vector<int> washizu(12, 0);
vector<bool> ret(12, false);

void solve(int i, int n)
{
    if(n == 0){
        vector<int> select(12, 0);
        for(int j=0; j<12; ++j)
            select[j] = 4 - a[j];

        for(int j=0; j<12; ++j){ // 対子
            if(a[j] < 2)
                continue;

            select[j] += 2;

            bool ok = true;
            for(int k=0; k<12; ++k){
                if(select[k] < washizu[k])
                    ok = false;
            }

            if(ok){
                for(int k=0; k<12; ++k){
                    if(select[k] > washizu[k])
                        ret[k] = true;
                }
            }

            select[j] -= 2;
        }
        return;
    }

    if(i < 12){ // 刻子
        if(a[i] >= 3){
            a[i] -= 3;
            solve(i+1, n-1);
            a[i] += 3;
        }
        solve(i+1, n);
    }else if(i < 22){ // 順子
        int j = i - 12;
        if(a[j] > 0 && a[j+1] > 0 && a[j+2] > 0){
            -- a[j];
            -- a[j+1];
            -- a[j+2];
            solve(i, n-1);
            ++ a[j+2];
            ++ a[j+1];
            ++ a[j];
        }
        solve(i+1, n);
    }
}

int main()
{
    int n;
    cin >> n;

    for(int i=0; i<3*n+1; ++i){
        string s;
        cin >> s;
        if(s != "*"){
            int x = atoi(s.c_str());
            -- x;
            ++ washizu[x];
        }
    }

    solve(0, n);

    bool ng = true;
    for(int i=0; i<12; ++i){
        if(ret[i]){
            cout << (i + 1) << endl;
            ng = false;
        }
    }
    if(ng)
        cout << -1 << endl;

    return 0;
}
