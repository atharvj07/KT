for i in range(int(input())):
	(a, b, c, d) = map(int, input().split())
	(x, y, x1, y1, x2, y2) = map(int, input().split())
	if x1 == x2 and (a != 0 or b != 0):
		print('No')
	elif y1 == y2 and (c != 0 or d != 0):
		print('No')
	elif x1 <= x + b - a <= x2 and y1 <= y + d - c <= y2:
		print('Yes')
	else:
		print('No')
