def ipl():
	return list(map(int, ip().split()))

def ipn():
	return int(ip())

def ipf():
	return float(ip())

def solve():
	n = ipn()
	a = ipl()
	b = sorted(a)
	h = defaultdict(int)
	bi = defaultdict(list)
	ai = defaultdict(list)
	for i in range(n):
		if a[i] not in ai:
			ai[a[i]] = [0, 0]
		if b[i] not in bi:
			bi[b[i]] = [0, 0]
		if i % 2:
			ai[a[i]][1] += 1
		else:
			ai[a[i]][0] += 1
		if i % 2:
			bi[b[i]][1] += 1
		else:
			bi[b[i]][0] += 1
	for (k, v) in ai.items():
		if v[0] != bi[k][0] or v[1] != bi[k][1]:
			print('NO')
			return
	print('YES')
t = ipn()
for _ in range(t):
	solve()
