from bisect import bisect_right as br
from sys import stdin
(n, k) = map(int, stdin.readline().strip().split())
s = list(map(int, stdin.readline().strip().split()))
s.sort()
dp = [-1 for i in range(s[-1])]
for i in range(s[-1]):
	dp[i] = n - br(s, i)
ans = 0
acum = 0
while len(dp) > 0 and dp[-1] != n:
	if acum + dp[-1] > k:
		ans += 1
		acum = 0
	acum += dp[-1]
	dp.pop()
if acum != 0:
	ans += 1
print(ans)
