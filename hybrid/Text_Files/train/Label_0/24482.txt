#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pll = pair<ll, ll>;
ll n;
bool good(vector<pll>& a, vector<pll>& b) {
  vector<vector<ll>> v;
  for (int i = 0; i < n; i++) {
    v.push_back({a[i].first, a[i].second, b[i].first, b[i].second});
  }
  sort(v.begin(), v.end());
  multiset<ll> s, e;
  multiset<pair<ll, pll>> u;
  for (int i = 0; i < n; i++) {
    s.insert(v[i][2]);
    e.insert(v[i][3]);
    u.insert({v[i][1], {v[i][2], v[i][3]}});
    while (u.begin()->first < v[i][0]) {
      s.erase(s.find(u.begin()->second.first));
      e.erase(e.find(u.begin()->second.second));
      u.erase(u.begin());
    }
    if (i + 1 < n && v[i][0] == v[i + 1][0]) continue;
    if (*s.rbegin() > *e.begin()) return false;
  }
  return true;
}
int main() {
  cin >> n;
  vector<pll> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i].first >> a[i].second >> b[i].first >> b[i].second;
  }
  if (good(a, b) && good(b, a))
    cout << "YES\n";
  else
    cout << "NO\n";
}
