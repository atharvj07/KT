#include <bits/stdc++.h>
using namespace std;
int o[11];
int check(int arr[], int x, int n) {
  int m = 0, one = 0, f = -1, val = -1, ind;
  if (x < n - 1) {
    o[arr[x + 1]]--;
  }
  for (int i = 0; i < 11; i++) {
    if (o[i] > m) {
      m = o[i];
      ind = i;
    }
    if (o[i] == 1) one++;
  }
  int flag = 1, flag2 = 1;
  for (int i = 0; i < 11; i++) {
    if (o[i] > 0 && o[i] != m - 1 && i != ind) flag = 0;
  }
  if (flag) return flag;
  if (one == 0)
    return 0;
  else {
    if (one == 1) {
      for (int i = 0; i < 11; i++) {
        if (f == -1 && o[i] != 1 && o[i] > 0) {
          val = o[i];
          f = 0;
        }
        if (o[i] != val && o[i] > 0 && o[i] != 1) flag2 = 0;
      }
    }
    if (one > 1) {
      for (int i = 0; i < 11; i++) {
        if (o[i] > 0 && o[i] != 1) flag2 = 0;
      }
    }
    if (flag2 == 1) return 1;
    return 0;
  }
}
int main() {
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    o[arr[i]]++;
  }
  for (int x = n - 1; x >= 2; x--) {
    if (check(arr, x, n)) {
      cout << x + 1;
      return 0;
    }
  }
  cout << "1";
}
