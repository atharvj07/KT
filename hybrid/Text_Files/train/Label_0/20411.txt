#include <bits/stdc++.h>
using namespace std;
const int inf = 0x3f3f3f3f;
const int N = 1e6 + 100;
const int mod = 1e9 + 7;
long long q_pow(long long a, long long b) {
  long long ans = 1;
  while (b) {
    if (b & 1) ans = ans * a % mod;
    a = a * a % mod;
    b >>= 1;
  }
  return ans;
}
int main() {
  long long n, m, k;
  scanf("%lld%lld%lld", &n, &m, &k);
  if (((n + m) & 1) && k == -1)
    puts("0");
  else
    printf("%lld\n", q_pow(q_pow(2, m - 1), n - 1));
  return 0;
}
