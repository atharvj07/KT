import logging
import copy
import sys
import math
logging.basicConfig(stream=sys.stderr, level=logging.DEBUG)

def solve(total, limit):
	last_bit = 0
	bits = []
	while limit > 0:
		bit = limit & 1
		last_bit += 1
		limit >>= 1
		n = limit if bit == 0 else limit + 1
		bits.append(n)
	bits[-1] = 1
	bit_list = []
	for i in range(len(bits) - 1, -1, -1):
		base = int(math.pow(2, i))
		n = bits[i]
		used = 0
		while total > 0 and total >= base and (used < n):
			total -= base
			used += 1
		bit_list.append(used)
	bit_list.reverse()
	result = []
	for (i, n) in enumerate(bit_list):
		base = int(math.pow(2, i))
		for j in range(0, n):
			mask = j << i + 1
			result.append(base | mask)
	if total > 0:
		print(-1)
		return
	print(len(result))
	print(' '.join(list(map(lambda x: str(x), result))))

def main():
	firstLine = input().split()
	solve(int(firstLine[0]), int(firstLine[1]))

def log(*message):
	logging.debug(message)
main()
