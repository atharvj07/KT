#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  if (n == 1) {
    cout << "YES\n";
    return 0;
  }
  int eq = 0;
  for (int i = 1; i < n; i++) {
    if (!eq) {
      if (a[i] == a[i - 1])
        eq = 1;
      else if (a[i] < a[i - 1])
        eq = 2;
      else if (a[i] < a[i - 1]) {
        cout << "NO\n";
        return 0;
      }
    } else if (eq == 1) {
      if (a[i] < a[i - 1])
        eq = 2;
      else if (a[i] > a[i - 1]) {
        cout << "NO\n";
        return 0;
      }
    } else {
      if (a[i] >= a[i - 1]) {
        cout << "NO\n";
        return 0;
      }
    }
  }
  cout << "YES\n";
  return 0;
}
