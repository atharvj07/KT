#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    while (true) {
        int n;
        cin >> n;
        if (!n) return 0;

        int maxi = 0, mini = 10000000;
        while (n--) {
            int a, b, c, d, e;
            cin >> a >> b >> c >> d >>e;
            int sum = a + b + c + d + e;
            maxi = max(maxi, sum);
            mini = min(mini, sum);
        }
        cout << maxi << " " << mini << endl;
    }
}
