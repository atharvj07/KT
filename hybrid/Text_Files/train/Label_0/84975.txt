a1 = int(input())
a2 = int(input())
k1 = int(input())
k2 = int(input())
k = int(input())
maax = 0
c = k - a1 * (k1 - 1) - a2 * (k2 - 1)
if c < 0:
	man = 0
else:
	man = c
if k1 < k2:
	for j in range(a1):
		if k < k1:
			break
		k = k - k1
		maax = maax + 1
	for j in range(a2):
		if k < k2:
			break
		k = k - k2
		maax = maax + 1
else:
	for j in range(a2):
		if k < k2:
			break
		k = k - k2
		maax = maax + 1
	for j in range(a1):
		if k < k1:
			break
		k = k - k1
		maax = maax + 1
print(man, maax)
