def main():
	n = int(input())
	xy = []
	for _ in range(n):
		xy.append(readIntArr())
	c = readIntArr()
	k = readIntArr()

	def getCost(i, j):
		if i == n:
			return c[j]
		if j == n:
			return c[i]
		(xi, yi) = xy[i]
		(xj, yj) = xy[j]
		return (k[i] + k[j]) * (abs(xi - xj) + abs(yi - yj))
	total = 0
	poweredCities = []
	connectedCities = []
	visited = [False] * (n + 1)
	costCheapest = [inf] * (n + 1)
	fromCheapest = [-1] * (n + 1)
	currCity = 0
	visited[currCity] = True
	for _ in range(n):
		for destination in range(n + 1):
			if currCity == destination:
				continue
			if visited[destination]:
				continue
			if getCost(currCity, destination) < costCheapest[destination]:
				costCheapest[destination] = getCost(currCity, destination)
				fromCheapest[destination] = currCity
		minCost = inf
		minCityFrom = -1
		minCityTo = -1
		for destination in range(n + 1):
			if currCity == destination:
				continue
			if visited[destination]:
				continue
			if costCheapest[destination] < minCost:
				minCost = costCheapest[destination]
				minCityFrom = fromCheapest[destination]
				minCityTo = destination
		assert minCityFrom != -1
		total += minCost
		visited[minCityTo] = True
		if minCityFrom == n:
			poweredCities.append(minCityTo + 1)
		elif minCityTo == n:
			poweredCities.append(minCityFrom + 1)
		else:
			connectedCities.append([minCityFrom + 1, minCityTo + 1])
		currCity = minCityTo
	print(total)
	print(len(poweredCities))
	oneLineArrayPrint(poweredCities)
	print(len(connectedCities))
	multiLineArrayOfArraysPrint(connectedCities)
	return
import sys
input = sys.stdin.buffer.readline

def oneLineArrayPrint(arr):
	print(' '.join([str(x) for x in arr]))

def multiLineArrayPrint(arr):
	print('\n'.join([str(x) for x in arr]))

def multiLineArrayOfArraysPrint(arr):
	print('\n'.join([' '.join([str(x) for x in y]) for y in arr]))

def readIntArr():
	return [int(x) for x in input().split()]

def makeArr(defaultVal, dimensionArr):
	dv = defaultVal
	da = dimensionArr
	if len(da) == 1:
		return [dv for _ in range(da[0])]
	else:
		return [makeArr(dv, da[1:]) for _ in range(da[0])]
