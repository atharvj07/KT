from collections import defaultdict
n = int(input())
d = defaultdict(list)
a = []
mp = [i for i in range(0, 500001)]
for i in range(n):
	A = list(map(int, input().split()))
	if A[0] == 1:
		a.append((A[0], A[1]))
	else:
		a.append((A[0], A[1], A[2]))
ans = []
for i in range(n - 1, -1, -1):
	if len(a[i]) == 2:
		ans.append(mp[a[i][1]])
	else:
		mp[a[i][1]] = mp[a[i][2]]
ans = ans[::-1]
print(*ans)
