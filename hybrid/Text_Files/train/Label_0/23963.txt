#include <bits/stdc++.h>
using namespace std;
map<int, string> ma;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n;
  string s;
  cin >> n >> s;
  ma[4] = "223";
  ma[6] = "35";
  ma[8] = "2227";
  ma[9] = "3327";
  string ans;
  for (int i = 0; i < n; i++) {
    int d = s[i] - '0';
    if (d <= 1) continue;
    if (ma.find(d) == ma.end())
      ans += char(d + '0');
    else
      ans += ma[d];
  }
  sort(ans.begin(), ans.end());
  reverse(ans.begin(), ans.end());
  cout << ans;
  return 0;
}
