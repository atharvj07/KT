def func(dp, a, ind):
	if ind > n:
		return 0
	if dp[ind] != -1:
		return dp[ind]
	count = 0
	for i in range(ind, n + 1, ind):
		if a[i] > a[ind]:
			count = max(count, 1 + func(dp, a, i))
	dp[ind] = count
	return dp[ind]
t = int(input())
while t > 0:
	t -= 1
	n = int(input())
	arr = [0]
	arr = arr + list(map(int, input().split(' ')))
	dp = [-1] * (len(arr) + 1)
	maxi = 1
	for i in range(1, n + 1):
		maxi = max(maxi, 1 + func(dp, arr, i))
	print(maxi)
