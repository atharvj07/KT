def main():
	t = int(input())
	for _ in range(t):
		(b, p, q) = [int(s) for s in input().split()]
		(h, c) = [int(s) for s in input().split()]
		sol = 0
		if h > c:
			if p <= b // 2:
				b -= 2 * p
				sol += p * h
				sol += min(b // 2, q) * c
			else:
				sol += b // 2 * h
		elif q <= b // 2:
			b -= 2 * q
			sol += q * c
			sol += min(b // 2, p) * h
		else:
			sol += b // 2 * c
		print(sol)
main()
