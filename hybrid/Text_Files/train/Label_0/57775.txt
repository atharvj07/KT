import java.util.Scanner;
import java.lang.*;

public class B {
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        long k  = in.nextLong();
        String s = "s";
        String o = "o";
        String c = "c";
        String d = "d";
        String f = "f";
        String e = "e";
        String r = "r";
        int[] freq = new int[10];
        for(int i=0; i<10; i++){
            freq[i] = 1;
        }
        long mul = 1;
        int idx = 0;
        while(mul < k){
            freq[idx]++;
            if(idx == 9){
                idx = 0;
            }
            else{
                idx ++;
            }
            mul = getmul(freq);
        }
        System.out.println(c.repeat(freq[0]) + o.repeat(freq[1]) + d.repeat(freq[2]) + e.repeat(freq[3]) + f.repeat(freq[4]) + o.repeat(freq[5]) + r.repeat(freq[6]) + c.repeat(freq[7]) + e.repeat(freq[8]) + s.repeat(freq[9]));
    }
    static long getmul(int[] freq){
        long num = 1;
        for(int i=0; i<freq.length; i++){
            num *= freq[i];
        }
        return num;
    }
}
