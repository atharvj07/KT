#include <bits/stdc++.h>
#define pb push_back
#define mp make_pair
#define fi first
#define se second
#define sz(x) (int) x.size()
#define cat(x) cerr << #x << " = " << x << endl
#define rep(i, j, n) for (int i = j; i <= n; ++i)
#define per(i, j, n) for (int i = n; j <= i; --i)
 
using ll = long long;
using namespace std;

const int N = 2e5 + 5;
const int MOD = 1e9 + 7;

int n, a;
pair<int,int> t[N];

int main() {
	scanf ("%d", &n);
	for (int j = 0; j < 2; ++j)
		for (int i = 0; i < n; ++i) {
			scanf ("%d", &a);
			t[j * n + i] = {a, j};
		}
	sort(t, t + n + n);
	int res = 1, bilans = 0;
	for (int i = 0; i < n + n; ++i) {
		int k = t[i].se;
		if (!k) k = -1;
		if (bilans * k >= 0)
			bilans += k;
		else {
			res = (ll) res * abs(bilans) % MOD;
			bilans += k;
		}
	}
	printf ("%d\n", res);
	return 0;
}
