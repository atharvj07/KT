l = [i for i in range(101)]
(c, x, y) = (0, 0, 0)
for i in range(int(input())):
	(n, m) = map(int, input().split())
	if i == 0:
		(x, y) = (n, m)
	else:
		for j in range(n, m):
			l[j] = -1
for i in range(x, y):
	if l[i] != -1 and l[i + 1] == -1 or (l[i] != -1 and l[i + 1] != -1):
		c += 1
print(c)
