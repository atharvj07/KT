def bSearch(l, r, func):
	r += 1
	while l < r:
		m = (l + r) // 2
		if func(m) == False:
			l = m + 1
		else:
			r = m
	return l

def checkPossible(med, a, n, k):
	diff = 0
	minDiff = [inf for _ in range(n)]
	minDiff[0] = 0
	res = -1
	for (i, x) in enumerate(a):
		if x >= med:
			diff += 1
		else:
			diff -= 1
		minDiff[i] = min(minDiff[i], diff)
		if i > 0:
			minDiff[i] = min(minDiff[i], minDiff[i - 1])
		l = i + 1 - k
		if l == 0 and diff >= 1:
			res = 1
			break
		if l - 1 >= 0 and diff - minDiff[l - 1] >= 1:
			res = 1
			break
	return res

def main():
	(n, k) = readIntArr()
	a = readIntArr()

	def func(med):
		if checkPossible(med, a, n, k) == -1:
			return True
		else:
			return False
	ans = bSearch(1, n, func) - 1
	print(ans)
	return
