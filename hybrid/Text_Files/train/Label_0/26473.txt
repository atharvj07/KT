#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[2 * n], b[n], c[n - 1];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int min = 5000, x, y;
  for (int i = 1; i < n; i++) {
    if (abs(a[i] - a[i - 1]) < min) {
      min = abs(a[i] - a[i - 1]);
      x = i;
      y = i - 1;
    }
  }
  if (min > abs(a[n - 1] - a[0])) {
    cout << n << " "
         << "1";
  } else {
    cout << x + 1 << " " << y + 1;
  }
}
