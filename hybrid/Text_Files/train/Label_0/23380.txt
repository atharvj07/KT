#include <bits/stdc++.h>
using namespace std;
const int maxn = 10010;
vector<int> adj[maxn];
map<string, int> mp;
bool vs[maxn];
int eval(vector<int> adj[], int u) {
  map<int, int> counter;
  bitset<maxn> bs;
  for (int i = 0; i < adj[u].size(); i++) {
    int v = adj[u][i];
    bs.set(v);
    for (int j = 0; j < adj[v].size(); j++) {
      int w = adj[v][j];
      if (w == u) continue;
      if (counter.find(w) != counter.end())
        counter[w]++;
      else
        counter[w] = 1;
    }
  }
  int mx = -1;
  for (map<int, int>::iterator it = counter.begin(); it != counter.end();
       it++) {
    if (!bs.test(it->first)) mx = max(mx, it->second);
  }
  int ret = 0;
  for (map<int, int>::iterator it = counter.begin(); it != counter.end();
       it++) {
    if (!bs.test(it->first) && it->second == mx) ret++;
  }
  return ret;
}
int eval_1(vector<int> adj[], int u) {
  int cnt = 0;
  int mx = 0;
  memset(vs, 0, sizeof(vs));
  for (int i = 0; i < adj[u].size(); i++) vs[adj[u][i]] = true;
  for (int i = 1; i <= mp.size(); i++) {
    if (i != u && !vs[i]) {
      int tmp = 0;
      for (int k = 0; k < adj[i].size(); k++) tmp += vs[adj[i][k]];
      if (tmp > mx) {
        mx = tmp;
        cnt = 1;
      } else if (tmp == mx) {
        cnt++;
      }
    }
  }
  return cnt;
}
int main() {
  int n;
  while (cin >> n) {
    int m = 0;
    for (int i = 0; i < n; i++) {
      string a, b;
      cin >> a >> b;
      if (mp.find(a) == mp.end()) {
        mp[a] = mp.size();
      }
      if (mp.find(b) == mp.end()) {
        mp[b] = mp.size();
      }
      adj[mp[a]].push_back(mp[b]);
      adj[mp[b]].push_back(mp[a]);
    }
    cout << mp.size() << endl;
    for (map<string, int>::iterator it = mp.begin(); it != mp.end(); it++) {
      cout << it->first << " " << eval_1(adj, it->second) << endl;
    }
  }
  return 0;
}
