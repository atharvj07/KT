t = int(input())
for i in range(t):
	n = int(input())
	if n in [1, 2, 4]:
		print(-1)
	elif n == 3:
		print('1 0 0')
	else:
		k = n // 5 - 1
		n = n - 5 * k
		if n == 5:
			print(0, k + 1, 0)
		if n == 6:
			print(2, k, 0)
		if n == 7:
			print(0, k, 1)
		if n == 8:
			print(1, k + 1, 0)
		if n == 9:
			print(3, k, 0)
