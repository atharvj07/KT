#include <bits/stdc++.h>
using namespace std;
bool isPrime(int x) {
  for (long long int i = 2; i * i <= x; i++) {
    if (x % i == 0) return false;
  }
  return true;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int aux;
  cin >> aux;
  while (aux--) {
    int n;
    cin >> n;
    int num = n;
    vector<int> fat;
    for (int d : {2, 3, 5}) {
      while (n % d == 0) {
        fat.push_back(d);
        n /= d;
      }
    }
    int incremento[] = {4, 2, 4, 2, 4, 6, 2, 6};
    int aux = 0;
    for (int i = 7; i * i <= n; i += incremento[(aux++) % 8]) {
      while (n % i == 0) {
        fat.push_back(i);
        n /= i;
      }
    }
    if (n > 1) fat.push_back(n);
    if (fat.size() < 3) {
      cout << "NO\n";
      continue;
    }
    int p = fat[0];
    int s = fat[1];
    if (s == p) {
      s *= fat[2];
    }
    int t = num / (p * s);
    if (p != s and p != t and s != t and s >= 2 and p >= 2 and t >= 2) {
      cout << "YES\n";
      cout << p << ' ' << s << ' ' << t << '\n';
      continue;
    }
    cout << "NO\n";
  }
  return 0;
}
