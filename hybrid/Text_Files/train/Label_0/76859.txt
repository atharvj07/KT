(i, j, mn) = (1, 2, [])

def ms(l, l1, r, x):
	global i, j
	if l1 <= r:
		m = (l1 + r) // 2
		mn.append((l1, r))
		l[m] = i
		i += 1
		ms(l, l1, m - 1, 1)
		ms(l, m + 1, r, 0)
	return mn
for _ in range(int(input())):
	(i, mn, n) = (1, [], int(input()))
	l = [0] * n
	a = ms(l, 0, n - 1, 1)
	a.sort(key=lambda x: -(x[1] - x[0]))
	for i in range(n):
		l[(a[i][0] + a[i][1]) // 2] = i + 1
	print(' '.join(list(map(str, l))))
