    int n;

    void set( int pos, int val,   int v, int tl, int tr ){
        if( tl == tr && pos == tl )
            tree[v] = val;
        else{
            int tm = (tl + tr)/2;
            if( pos <= tm )
                set( pos, val,    2*v  , tl  , tm );
            else
                set( pos, val,    2*v+1, tm+1, tr );
            tree[v] = tree[2*v] + tree[2*v+1];
        }
    }

    int sum( int l, int r,   int v, int tl, int tr ){
        if( l > r ) return 0;
        if( l==tl && r==tr ) return tree[v];
        int tm = (tl + tr)/2;
        return sum( l          , min(r,tm), 2*v  , tl  , tm ) +
               sum( max(l,tm+1), r        , 2*v+1, tm+1, tr );
    }

    int sum( int l, int r ){
        return sum( l, r,  1,1,n );
    }

    void set( int pos, int val ){

        upd( a[l]     );
        if( a[l] + 1 <= n ) upd( a[l] + 1 );
    }

    void upd( int a ){
        set( a, (a==n||ind[a+1]<ind[a])? 1 : 0 );
    }
