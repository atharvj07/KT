#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 3;
const int K = 2e3 + 3;
const int inf = 1e9;
int n, m, k, a[N], b[K], s[K];
long long dp[K];
int main() {
  scanf("%d%d%d", &n, &m, &k);
  for (int i = 1; i <= n; i++) scanf("%d", a + i);
  sort(a + 1, a + n + 1);
  s[0] = 0;
  for (int i = 1; i <= k; i++) s[i] = s[i - 1] + a[i];
  memset(b, 0, sizeof b);
  int t = 0;
  for (int i = 1; i <= m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    if (x <= k) b[x] = max(b[x], y);
  }
  dp[0] = 0;
  for (int i = 1; i <= k; i++) dp[i] = inf;
  for (int i = 1; i <= k; i++) {
    dp[i] = min(dp[i], dp[i - 1] + a[i]);
    for (int j = 0; j < i; j++)
      dp[i] = min(dp[i], dp[j] + s[i] - s[j + b[i - j]]);
  }
  cout << dp[k];
  return 0;
}
