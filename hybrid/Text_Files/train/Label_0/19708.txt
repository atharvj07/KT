#include <bits/stdc++.h>
using namespace std;
string query(long long int i) {
  string x;
  cout << i << endl;
  getline(cin, x);
  return x;
}
void Solve() {
  for (long long int i = 0; i <= 9; i++) {
    string ans = query(i);
    if (ans == "no") continue;
    if (ans == "go die in a hole" || ans == "terrible" || ans == "worse" ||
        ans == "are you serious" || ans == "don't even" || ans == "no way") {
      cout << "grumpy" << endl;
      return;
    } else if (ans == "great" || ans == "don't think so" ||
               ans == "don't touch me" || ans == "not bad" || ans == "cool") {
      cout << "normal" << endl;
      return;
    }
  }
}
int main(int argc, char const *argv[]) {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(0);
  ;
  int T = 1;
  while (T--) Solve();
  return 0;
}
