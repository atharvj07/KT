n = int(input())
a = []
for i in range(n):
	b = input()
	c = {}
	count = 0
	for j in b:
		if j in c:
			c[j] += 1
		else:
			c[j] = 1
			count += 1
		if count > 2:
			break
	if 1 <= count <= 2:
		a.append(c)
eng = 'abcdefghijklmnopqrstuvwxyz'
maxs = 0
for i in range(26):
	for j in range(i + 1, 26):
		sum = 0
		for k in a:
			if eng[i] in k and eng[j] in k:
				sum += k[eng[i]] + k[eng[j]]
			elif eng[i] in k and len(list(k.keys())) == 1:
				sum += k[eng[i]]
			elif eng[j] in k and len(list(k.keys())) == 1:
				sum += k[eng[j]]
		maxs = max(sum, maxs)
print(maxs)
