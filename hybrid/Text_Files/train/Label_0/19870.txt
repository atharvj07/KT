#include <bits/stdc++.h>
using namespace std;
string s;
long long n;
bool len(char x) {
  return x == 'a' || x == 'e' || x == 'i' || x == 'o' || x == 'u' || x == 'y';
}
int main() {
  cin >> n;
  cin >> s;
  cout << s[0];
  for (int i = 1; i < n; i++) {
    if (len(s[i]) && len(s[i - 1])) continue;
    cout << s[i];
  }
  cout << endl;
  return 0;
}
