import java.util.*;
public class Main{
  public static void main(String[] args){
    Scanner scan = new Scanner(System.in);
    int n = scan.nextInt();
    int[] lengths = new int[101];
    for (int i = 0; i < n; i++)
      lengths[scan.nextInt()]++;
    int pairCount = 0;
    for (int i = 0; i < 101; i++)
      pairCount += lengths[i] / 2;
    System.out.println(pairCount / 2);
  }
}
