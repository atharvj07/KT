(n, k) = map(int, input().split())
l = list(map(int, input().split()))
ans = []
d = dict()
limit = 0
for i in range(n):
	if l[i] in d.keys():
		if d[l[i]]:
			continue
		else:
			d[l[i]] = 1
			ans.insert(0, l[i])
			limit += 1
			if limit > k:
				x = ans.pop()
				d[x] = 0
				limit -= 1
	else:
		d[l[i]] = 1
		ans.insert(0, l[i])
		limit += 1
		if limit > k:
			x = ans.pop()
			d[x] = 0
			limit -= 1
print(limit)
print(*ans)
