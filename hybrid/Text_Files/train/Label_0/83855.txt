def valid(m, even=True):
	l = 0
	for i in range(n):
		if not even:
			l += 1
			even = not even
		elif a[i] <= m:
			l += 1
			even = not even
	return l >= k
(n, k) = map(int, input().strip().split())
a = list(map(int, input().strip().split()))
l = 1
h = 1000000000.0
while l < h:
	m = l + (h - l) // 2
	if valid(m, even=True) or valid(m, even=False):
		h = m
	else:
		l = m + 1
print(int(h))
