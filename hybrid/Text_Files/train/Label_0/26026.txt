#include <iostream>
#include <algorithm>
#include <vector>
#include <string>
#define int long long
using namespace std;
signed main() {
	int n; cin >> n;
	int p = n; if (n % 2 == 0)p++;
	cout << n * (n - 1) / 2 - (n / 2) << endl;
	for (int i = 1; i <= n; i++) {
		for (int j = i + 1; j <= n; j++) {
			if (i + j != p)cout << i << ' ' << j << endl;
		}
	}
}
