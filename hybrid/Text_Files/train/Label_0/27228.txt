#include <bits/stdc++.h>
using namespace std;
const int mod = 1000000007;
long long d[16] = {1,         18,        1800,      670320,
                   734832000, 890786230, 695720788, 150347555};
int n;
bool v[16];
bool u[16];
long long f[20];
long long r;
void dfs(int cur) {
  if (cur == n) {
    r++;
    return;
  }
  for (int i = 0; i < n; i++)
    if (!v[i] && !u[(i + cur) % n]) {
      v[i] = 1;
      u[(i + cur) % n] = 1;
      dfs(cur + 1);
      u[(i + cur) % n] = 0;
      v[i] = 0;
    }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin >> n;
  if (n % 2)
    cout << d[n / 2] << endl;
  else
    cout << 0 << endl;
  return 0;
}
