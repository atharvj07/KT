import java.util.Scanner;
public class tree {

	public static void main(String[] args) {
	    Scanner in=new Scanner(System.in);
	    int n=in.nextInt();
        int x[]=new int[n];
        int y[]=new int[n];
        for(int i=0;i<n;i++){
        	 x[i]=in.nextInt();
        	 y[i]=in.nextInt();
       }int count=0;
        int c1=0,c2=0,c3=0,c4=0;
        for(int i=0;i<n;i++){
        	c1=0;
        	c2=0;
        	c3=0;
        	c4=0;
        	for(int j=0;j<n;j++){
        		if(i!=j && y[i]==y[j]){
        			if(x[j]>x[i]) c1++;
        			if(x[j]<x[i]) c2++;
        		}
        		if(i!=j && x[i]==x[j]){
        			if(y[j]>y[i]) c3++;
        			if(y[j]<y[i]) c4++;
        		}
        	}
           if(c1>0 && c2>0 && c3>0 && c4>0) count++;	
        }
	
	System.out.println(count);
	
	}
}
