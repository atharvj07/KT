#include <bits/stdc++.h>
using namespace std;
const long long int N = 1e5;
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long int n;
  cin >> n;
  vector<pair<long long int, long long int> > v(n);
  for (long long int i = 0; i < n; i++) cin >> v[i].second >> v[i].first;
  sort(v.begin(), v.end());
  long long int ans = 1;
  long long int r = v[0].first;
  for (long long int i = 1; i < n; i++) {
    if (v[i].second > r) {
      r = v[i].first;
      ans++;
    }
  }
  cout << ans;
}
