def isPalindrome(s):
	for i in range(len(s)):
		if s[i] != s[len(s) - i - 1]:
			return 0
	return 1

def all(a, z):
	for i in a:
		if a[i] != z:
			return 0
	return 1
for s in [*open(0)][2::2]:
	s = s[0:len(s)]
	s = s.rstrip('\n')
	n = len(s)
	a = {}
	for i in s:
		a[i] = 0
	for i in a:
		l = 0
		r = n - 1
		while l < r:
			if s[l] != s[r] and s[r] == i:
				r -= 1
				a[i] += 1
			elif s[l] != s[r] and s[l] == i:
				l += 1
				a[i] += 1
			elif s[l] == s[r]:
				l += 1
				r -= 1
			else:
				a[i] = -1
				break
	if all(a, -1):
		print(-1)
	else:
		min = 1000000000
		for i in a:
			if a[i] < min and a[i] >= 0:
				min = a[i]
		print(min)
