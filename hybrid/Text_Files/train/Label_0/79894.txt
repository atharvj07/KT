(n, b) = map(int, input().split())
A = list(map(int, input().split()))
B = []
(o, e) = (0, 0)
if A[0] % 2 == 0:
	o += 1
else:
	e += 1
for i in range(1, n - 1):
	if A[i] % 2 == 0:
		o += 1
	else:
		e += 1
	if o == e:
		B.append(abs(A[i + 1] - A[i]))
B.sort()
i = 0
while i < len(B) and B[i] <= b:
	b -= B[i]
	i += 1
print(i)
