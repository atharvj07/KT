#include <bits/stdc++.h>
using namespace ::std;
int main() {
  int N;
  int i;
  int tmp;
  int value;
  int times = 1;
  long long totalTime;
  vector<long long> sameLen;
  cin >> N;
  totalTime = N;
  cin >> tmp;
  for (i = 1; i < N; i++) {
    cin >> value;
    if (value == tmp) {
      times++;
    } else if (times > 1) {
      sameLen.push_back(times);
      times = 1;
      tmp = value;
    } else {
      times = 1;
      tmp = value;
    }
  }
  sameLen.push_back(times);
  for (i = 0; i < sameLen.size(); i++) {
    totalTime += (sameLen[i] * (sameLen[i] - 1)) / 2;
  }
  cout << totalTime;
  return 0;
}
