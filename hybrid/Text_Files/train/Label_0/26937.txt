#include <bits/stdc++.h>
using namespace std;
int main(int agrv, char** argc) {
  long long l, r, x, y, k;
  scanf("%I64d%I64d%I64d%I64d%I64d", &l, &r, &x, &y, &k);
  bool ans = false;
  for (int j = x; j <= y; ++j) {
    long long res = k * j;
    if (res >= l && res <= r) {
      ans = true;
      break;
    }
    if (res > r) {
      break;
    }
  }
  if (ans) {
    printf("YES");
  } else {
    printf("NO");
  }
  return 0;
}
