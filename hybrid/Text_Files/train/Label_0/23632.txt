#include <bits/stdc++.h>
using namespace std;
using ll = long long;
template <class T>
using V = vector<T>;
template <class T, class U>
using P = pair<T, U>;
using vll = V<ll>;
using vvll = V<vll>;
template <class T>
inline bool chmax(T& a, T b) {
  if (a < b) {
    a = b;
    return true;
  }
  return false;
}
template <class T>
inline bool chmin(T& a, T b) {
  if (a > b) {
    a = b;
    return true;
  }
  return false;
}
constexpr ll MOD = 1000000007;
constexpr ll HIGHINF = (ll)1e18;
constexpr int INF = 1e9;
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  int t;
  cin >> t;
  while (t-- > 0) {
    ll n;
    cin >> n;
    ll ans = HIGHINF;
    for (ll i = 1; i * i <= n; i++) {
      ll div = n / i;
      if (n % i == 0) {
        chmin(ans, (i - 1) + (div - 1));
      } else {
        chmin(ans, i + (div - 1));
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
