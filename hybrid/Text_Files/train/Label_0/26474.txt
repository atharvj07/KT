#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, j, k, l, min = 1000, m;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    if (i == 0) {
      cin >> a[i];
    } else {
      cin >> a[i];
      m = a[i] - a[i - 1];
      if (m < 0) {
        m = -m;
      }
      if (m < min) {
        min = m;
        j = i;
        k = i - 1;
      }
    }
  }
  m = a[0] - a[n - 1];
  if (m < 0) {
    m = -m;
  }
  if (m < min) {
    min = m;
    j = 0;
    k = n - 1;
  }
  if (a[j] > a[k]) {
    cout << ++k << " " << ++j;
  } else
    cout << ++j << " " << ++k;
}
