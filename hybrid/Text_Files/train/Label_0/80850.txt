(n, k) = map(int, input().split())
x = list(map(int, input().split()))
s = [sum(x[0:k])]
m = [0]
sm = [s[-1]]
for i in range(1, n - k + 1):
	s.append(s[-1] + x[i + k - 1] - x[i - 1])
	if s[-1] > sm[-1]:
		m.append(i)
		sm.append(s[-1])
	else:
		m.append(m[-1])
		sm.append(sm[-1])
(a, b) = (0, k)
sms = sm[0] + s[k]
for i in range(k + 1, n - k + 1):
	if sm[i - k] + s[i] > sms:
		sms = sm[i - k] + s[i]
		a = m[i - k]
		b = i
print(a + 1, b + 1)
