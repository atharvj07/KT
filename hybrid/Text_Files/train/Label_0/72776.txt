import sys
import math
inf = float('inf')

def lcm(a, b):
	return int(a / math.gcd(a, b) * b)
for _ in range(int(input())):
	n = int(input())
	a = input()
	b = input()
	changed = False
	count_1 = 0
	count_0 = 0
	for i in range(n):
		if a[i] == '1':
			count_1 += 1
		else:
			count_0 += 1
	flag = True
	for i in range(n - 1, -1, -1):
		if changed:
			if a[i] != b[i]:
				pass
			elif count_0 == count_1:
				changed = False
			else:
				flag = False
				break
		elif not changed:
			if a[i] == b[i]:
				pass
			elif count_0 == count_1:
				changed = True
			else:
				flag = False
				break
		if a[i] == '1':
			count_1 -= 1
		else:
			count_0 -= 1
	if flag:
		print('YES')
	else:
		print('NO')
