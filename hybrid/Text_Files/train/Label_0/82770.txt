def check(mid, p, f):
	ptr = 0
	for x in p:
		From = x
		To = x
		while ptr < len(f):
			From = min(From, f[ptr])
			To = max(To, f[ptr])
			need = To - From + min(To - x, x - From)
			if need > mid:
				break
			ptr += 1
		if ptr == len(f):
			return 1
	return 0

def f(s, n):
	p = []
	f = []
	for i in range(n):
		if s[i] == '*':
			f.append(i)
		if s[i] == 'P':
			p.append(i)
	lo = 0
	hi = 10 ** 9
	while lo <= hi:
		mid = (lo + hi) // 2
		if check(mid, p, f):
			hi = mid - 1
		else:
			lo = mid + 1
	return lo
n = int(input())
s = list(input().strip())
print(f(s, n))
