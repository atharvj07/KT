#include<bits/stdc++.h>
using namespace std;
template<class T = int> using V = vector<T>; 
template<class T = int> using VV = V< V<T> >;

template<class T> struct Dinic {
    struct Edge {
        int to, rev;
        T cap;
    };
    const int n;
    const T inf = numeric_limits<T>::max();
    VV<Edge> g;
    V<> dist, i;
    Dinic(int n) : n(n), g(n), dist(n), i(n) {}
    void add_edge(int from, int to, T cap) {
        g[from].emplace_back(Edge{to, (int) g[to].size(), cap});
        g[to].emplace_back(Edge{from, (int) g[from].size() - 1, 0});
    }
    void bfs(int s) {
        fill(begin(dist), end(dist), -1);
        queue<int> que;
        dist[s] = 0;
        que.push(s);
        while (!que.empty()) {
            int v = que.front(); que.pop();
            for (const auto& e : g[v]) {
                if (dist[e.to] != -1 or e.cap <= 0) continue;
                dist[e.to] = dist[v] + 1;
                que.push(e.to);
            }
        }
    }
    T dfs(int v, int s, T f) {
        if (v == s) return f;
        for (; i[v] < g[v].size(); ++i[v]) {
            Edge& e = g[v][i[v]];
            if (dist[e.to] >= dist[v] or g[e.to][e.rev].cap <= 0) continue;
            T d = dfs(e.to, s, min(f, g[e.to][e.rev].cap));
            if (d > 0) {
                g[e.to][e.rev].cap -= d;
                e.cap += d;
                return d;
            }
        }
        return 0;
    }
    T max_flow(int s, int t) {
        T res = 0;
        while (true) {
            bfs(s);
            if (dist[t] == -1) return res;
            fill(begin(i), end(i), 0);
            while (true) {
                T f = dfs(t, s, inf);
                if (f == 0) break;
                res += f;
            }
        }
    }
};
int main() {
    cin.tie(nullptr); ios::sync_with_stdio(false);
    int n, m; cin >> n >> m;
    string U; cin >> U;
    V<> A(n); for (auto&& e : A) cin >> e;
    Dinic<int> g(n + 2);
    int S = n, T = n + 1;
    for (int i = 0; i < n; ++i) {
        if (U[i] == 'L') {
            g.add_edge(S, i, A[i]);
            g.add_edge(i, T, 0);
        } else {
            g.add_edge(S, i, 0);
            g.add_edge(i, T, A[i]);
        }
    }
    while (m--) {
        int i, j, B; cin >> i >> j >> B, --i, --j;
        if (i > j) swap(i, j);
        g.add_edge(j, i, B);
    }
    cout << g.max_flow(S, T) << '\n';
    return 0;
}
