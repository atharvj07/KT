from collections import defaultdict
N = int(input())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
Total = 0
Dict = defaultdict(list)
for i in range(N):
	Dict[a[i]].append(b[i])
Total = 0
Visited = defaultdict(bool)
Selected = []
for i in Dict.keys():
	if len(Dict[i]) > 1:
		Total += sum(Dict[i])
		Selected.append(i)
		Visited[i] = True
for i in range(N):
	if Visited[a[i]] == False:
		for j in range(len(Selected)):
			if Selected[j] | a[i] == Selected[j]:
				Total += b[i]
				Visited[a[i]] = True
				break
print(Total)
