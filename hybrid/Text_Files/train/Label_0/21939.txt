#include <bits/stdc++.h>
using namespace std;
int n;
int arr[105];
int dx[105];
vector<vector<int> > action;
bool check(int cast) {
  set<pair<int, int> > cur;
  action.clear();
  int A = 0;
  for (int i = 0; i < (int)(n); i++) dx[i] = 0, A += arr[i] - cast;
  for (int i = 0; i < (int)(n); i++)
    if (arr[i] != cast) cur.insert(make_pair(arr[i] - cast, i));
  if (A % 2 != 0) {
    if ((int)cur.size() < 3) return false;
    set<pair<int, int> >::iterator it = cur.end();
    vector<int> mind;
    for (int index = 0; index < (int)(3); index++) {
      it--;
      mind.push_back(it->second);
    }
    for (int index = 0; index < (int)(3); index++) {
      cur.erase(
          make_pair(arr[mind[index]] - dx[mind[index]] - cast, mind[index]));
      dx[mind[index]]++;
    }
    for (int index = 0; index < (int)(3); index++) {
      int i = mind[index];
      if (arr[i] - dx[i] != cast)
        cur.insert(make_pair(arr[i] - dx[i] - cast, i));
    }
    action.push_back(mind);
  }
  while ((int)cur.size() > 1) {
    set<pair<int, int> >::iterator it = cur.end();
    vector<int> mind;
    for (int index = 0; index < (int)(2); index++) {
      it--;
      mind.push_back(it->second);
    }
    for (int index = 0; index < (int)(2); index++) {
      cur.erase(
          make_pair(arr[mind[index]] - dx[mind[index]] - cast, mind[index]));
      dx[mind[index]]++;
    }
    for (int index = 0; index < (int)(2); index++) {
      int i = mind[index];
      if (arr[i] - dx[i] != cast)
        cur.insert(make_pair(arr[i] - dx[i] - cast, i));
    }
    action.push_back(mind);
  }
  if ((int)cur.size() != 0) return false;
  return true;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n;
  for (int i = 0; i < (int)(n); i++) cin >> arr[i];
  int cast = arr[0];
  for (int i = 0; i < (int)(n); i++) cast = min(cast, arr[i]);
  while (cast && !check(cast)) cast--;
  cout << cast << '\n';
  if (cast == 0) {
    int cnt = 0;
    for (int i = 1; i < n; i++) cnt += max(arr[0], arr[i]);
    cout << cnt << '\n';
    for (int i = 1; i < n; i++) {
      string s = "";
      for (int j = 0; j < (int)(n); j++) s += '0';
      s[0] = s[i] = '1';
      for (int k = 0; k < (int)(max(arr[0], arr[i])); k++) cout << s << '\n';
    }
  } else {
    cout << (int)action.size() << '\n';
    for (int i = 0; i < (int)((int)action.size()); i++) {
      string s = "";
      for (int j = 0; j < (int)(n); j++) s += '0';
      for (int j = 0; j < (int)((int)action[i].size()); j++)
        s[action[i][j]] = '1';
      cout << s << '\n';
    }
  }
  return 0;
}
