from math import ceil
for _ in range(int(input())):
	(n, w) = map(int, input().split())
	l = list(map(int, input().split()))
	x = ceil(w / 2)
	su = 0
	for i in l:
		if i <= w:
			su += i
	if min(l) > w or su < x:
		print(-1)
	else:
		(ans, f) = (0, 0)
		num = []
		for i in range(n):
			if x <= l[i] <= w:
				f = 1
				ans = i + 1
				break
			if ans + l[i] < w:
				ans += l[i]
				num.append(i + 1)
			if x <= ans <= w:
				break
		if f:
			print(1)
			print(ans)
		else:
			print(len(num))
			print(*num)
