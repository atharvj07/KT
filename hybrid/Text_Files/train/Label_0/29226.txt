#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  long long ai[100000], bi[100000];
  map<long long, int> topAchievers;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> ai[i] >> bi[i];
    if (i < (n / 2)) {
      topAchievers.insert(pair<long long, int>(ai[i], 1));
      topAchievers.insert(pair<long long, int>(bi[i], 1));
    } else {
      topAchievers.insert(pair<long long, int>(ai[i], 0));
      topAchievers.insert(pair<long long, int>(bi[i], 0));
    }
  }
  int j = 0;
  for (map<long long, int>::iterator it = topAchievers.begin();
       it != topAchievers.end(); ++it, ++j) {
    if (j < n) {
      it->second = 2;
    }
  }
  for (int i = 0; i < n; ++i) {
    if (topAchievers[ai[i]] == 1 || topAchievers[ai[i]] == 2) {
      cout << "1";
    } else
      cout << "0";
  }
  cout << "\n";
  for (int i = 0; i < n; ++i) {
    if (topAchievers[bi[i]] == 1 || topAchievers[bi[i]] == 2) {
      cout << "1";
    } else
      cout << "0";
  }
}
