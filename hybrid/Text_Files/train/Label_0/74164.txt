t = int(input())
for _ in range(t):
	(n, k) = tuple(map(int, input().split()))
	string = input().strip('.')
	n = len(string)
	if n == 1:
		print('1')
	else:
		precomp = [-1] * n
		lastOcc = -1
		for i in range(n):
			precomp[i] = i if string[i] == '*' else precomp[i - 1]
		i = 0
		count = 0
		prev = 0
		while i < n:
			if string[i] == '*':
				count += 1
				i = i + k
			else:
				i = precomp[i]
		temp = 0 if i - k == n - 1 else 1
		print(count + temp)
