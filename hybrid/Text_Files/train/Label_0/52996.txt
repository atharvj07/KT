import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.util.TreeMap;

public class Task722D {

    public static void main(String[] args) throws IOException {
        InputScanner is = new InputScanner();
        int n = is.nextInt();
        TreeMap<Integer, Integer> tree = new TreeMap<>();
        int tmp;
        for (int i = 0; i < n; i++) {
            tmp = is.nextInt();
            tree.put(tmp, tmp);
        }
        boolean allIn = false;
        while (!allIn) {
            int e = tree.lastKey();
            int maxElem = Math.max(tree.lastKey()/2, 1);
            while(tree.containsKey(maxElem) && maxElem > 1) maxElem/=2;
            if (tree.containsKey(maxElem)) allIn = true;
            else {
                tree.remove(e);
                tree.put(maxElem, maxElem);
            }
        }
        StringBuilder ans = new StringBuilder();
        for (Integer i : tree.keySet()) {
            ans.append(i);
            ans.append(" ");
        }
        System.out.println(ans);
    }

    static class InputScanner {
        BufferedReader br;
        StringTokenizer st;

        public InputScanner() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public String next() throws IOException {
            if (st == null || !st.hasMoreTokens()) {
                String line = br.readLine();
                st = new StringTokenizer(line);
            }
            return st.nextToken();
        }

        public int nextInt() throws IOException {
            String next = next();
            return Integer.parseInt(next);
        }

        public long nextLong() throws IOException {
            String next = next();
            return Long.parseLong(next);
        }
    }
}
