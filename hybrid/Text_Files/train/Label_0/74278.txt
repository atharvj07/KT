import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
import sys
from collections import deque

def main():
	n = int(input())
	(Cost, Fr, To) = ([], [], [])
	for _ in range(n):
		(cost, fr, to) = map(int, input().split())
		Cost.append(cost)
		Fr.append(fr)
		To.append(to)
	if sum(Fr) != sum(To):
		return -1
	if n == 1:
		return 0
	Edges = [[] for _ in range(n)]
	for _ in range(n - 1):
		(u, v) = map(lambda x: int(x) - 1, input().split())
		Edges[u].append(v)
		Edges[v].append(u)
	Chi = [[] for _ in range(n)]
	Par = [-1] * n
	Depth = [0] * n
	q = deque()
	q.append(0)
	while q:
		fr = q.popleft()
		for to in Edges[fr]:
			if to == Par[fr]:
				continue
			Chi[fr].append(to)
			Par[to] = fr
			Cost[to] = min(Cost[to], Cost[fr])
			Depth[to] = Depth[fr] + 1
			q.append(to)
	d = max(Depth)
	D = [[] for _ in range(d + 1)]
	for i in range(n):
		depth = Depth[i]
		D[depth].append(i)
	(ToOne, ToZero) = ([0] * n, [0] * n)
	ans = 0
	for i in range(d + 1)[::-1]:
		for par in D[i]:
			(a, b, cost) = (Fr[par], To[par], Cost[par])
			if a == 0 and b == 1:
				ToOne[par] += 1
			elif a == 1 and b == 0:
				ToZero[par] += 1
			for chi in Edges[par]:
				if chi == Par[par]:
					continue
				ToOne[par] += ToOne[chi]
				ToZero[par] += ToZero[chi]
			change = min(ToOne[par], ToZero[par])
			ans += change * cost * 2
			ToOne[par] -= change
			ToZero[par] -= change
			v = Par[par]
	return ans
print(main())
