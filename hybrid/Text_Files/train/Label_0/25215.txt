#include<stdio.h>
#include<string.h>
int main(void)
{
	int cnt1,cnt2,i;
	char s[10001];
	while(scanf("%s",s)!=EOF){
		cnt1=0; cnt2=0;
		for(i=0;i<strlen(s);i++){
			if(s[i]=='J'){
				if(s[i+1]=='O'&&s[i+2]=='I') cnt1++;
			}
			if(s[i]=='I'){
				if(s[i+1]=='O'&&s[i+2]=='I') cnt2++;
			}
		}
		printf("%d\n%d\n",cnt1,cnt2);
	}
	return 0;
}
