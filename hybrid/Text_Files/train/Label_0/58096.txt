import java.util.*;
import java.math.*;
import java.io.*;

// Don't forget to change the class name!

public class Main{
    
    public static char[][] func(int N, int K, char[][] s){
        int M=1,i,j,k;
        
        for(i=0;i<K;i++) M *= N;
        char ans[][] = new char[M][M];
        
        for(i=0;i<M;i++) for(j=0;j<M;j++){
            boolean white = true;
            int x = i, y = j;
            for(k=0;k<K;k++){
                if(s[x%N][y%N] == '*') white = false;
                x /= N; y /= N;
            }
            ans[i][j] = (white ? '.' : '*');
        }
        
        return ans;
    }

    public static void main(String args[]) throws FileNotFoundException{
        Scanner scan = new Scanner(new File("input.txt"));
        PrintWriter out = new PrintWriter("output.txt");
        
        // -----------------file i/o-------------------------------------
        
        int N,K,i,j;
        N = scan.nextInt(); K = scan.nextInt();
        
        char s[][] = new char[N][N];
        for(i=0;i<N;i++){
            String tmp = scan.next();
            for(j=0;j<N;j++) s[i][j] = tmp.charAt(j);
        }
        
        char ans[][] = func(N,K,s);
        int M = ans.length;
        for(i=0;i<M;i++){
            for(j=0;j<M;j++) out.print(ans[i][j]);
            out.println();
        }
        
        // -----------------file i/o-------------------------------------
        
        out.close();
    }

}
