void shortestPath(const vector<vector<int> >& graph, int start, vector<int>& dist)
{
    int n = graph.size();
    dist.assign(n, INT_MAX);
    dist[start] = 0;
    priority_queue<pair<int,int> ,vector<pair<int,int> >, greater<pair<int,int> > > q;
    q.push(make_pair(0, start));

    while(!q.empty()){
        pair<int, int> p = q.top();
        q.pop();
        int v = p.second;
        if(dist[v] < p.first)
            continue;
        for(int i=0; i<n; ++i){
            if(graph[v][i] >= 0 && dist[v] + graph[v][i] < dist[i]){
                dist[i] = dist[v] + graph[v][i];
                q.push(make_pair(dist[i], i));
            }
        }
    }

    replace(dist.begin(), dist.end(), INT_MAX, -1);
}

double dfs1(vector<vector<int> >& graph, vector<int>& dist, vector<double>& num, int pos){
     if(num[pos] != -1)
         return num[pos];

     int n = graph.size();
     if(pos == n-1)
         return num[pos] = 1;

     double tmp = 0;
     for(int i=0; i<n; ++i){
         if(graph[pos][i] != -1 && dist[pos] + graph[pos][i] == dist[i])
             tmp += dfs1(graph, dist, num, i);
     }
     return num[pos] = tmp;
}

double dfs2(vector<vector<int> >& graph, vector<int>& dist, vector<double>& num, int pos){
     if(num[pos] != -1)
         return num[pos];

     int n = graph.size();
     if(pos == 0)
         return num[pos] = 1;

     double tmp = 0;
     for(int i=0; i<n; ++i){
         if(graph[i][pos] != -1 && dist[pos] == dist[i] + graph[i][pos])
             tmp += dfs2(graph, dist, num, i);
     }
     return num[pos] = tmp;
}

void solve(vector<vector<int> >& graph, vector<double>& p)
{
    int n = graph.size();
    vector<int> dist;
    shortestPath(graph, 0, dist);

    vector<double> num1(n, -1.0), num2(n, -1.0);
    dfs1(graph, dist, num1, 0);
    dfs2(graph, dist, num2, n-1);

    p.resize(n, 0.0);
    for(int i=0; i<n; ++i){
        if(num2[i] > -0.5)
            p[i] = num1[i] * num2[i] / num1[0];
    }
}

int main()
{
    for(;;){
        int n, m, p;
        cin >> n >> m >> p;
        if(n == 0)
            return 0;

        vector<vector<int> > graph(n, vector<int>(n, -1));
        for(int i=0; i<m; ++i){
            int a, b, c;
            cin >> a >> b >> c;
            graph[a][b] = graph[b][a] = c;
        }

        vector<double> prob;
        solve(graph, prob);
        for(int i=0; i<p; ++i){
            int a;
            cin >> a;
            printf("%.10f\n", prob[a]);
        }
        cout << endl;
    }
}
