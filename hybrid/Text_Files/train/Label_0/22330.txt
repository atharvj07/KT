#include <bits/stdc++.h>
using namespace std;
const int N = 200010;
int n;
long long ans;
int a[N], l[N], r[N];
int main() {
  scanf("%d", &n);
  ans = (long long)n * (long long)(n + 1) / 2ll;
  for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
  a[0] = a[n + 1] = -1;
  for (int i = 1; i <= n; ++i) {
    l[i] = i - 1;
    while ((a[i] | a[l[i]]) == a[i]) l[i] = l[l[i]];
  }
  for (int i = n; i; --i) {
    r[i] = i + 1;
    while ((a[i] | a[r[i]]) == a[i] && a[i] > a[r[i]]) r[i] = r[r[i]];
  }
  for (int i = 1; i <= n; ++i)
    ans -= (long long)(i - l[i]) * (long long)(r[i] - i);
  printf("%lld\n", ans);
  return 0;
}
