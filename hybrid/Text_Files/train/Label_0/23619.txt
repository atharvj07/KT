#include<iostream>
#include<vector>
#include<queue>

using namespace std;

#define rep(i, n) for (int i = 0; i < int(n); ++i)

typedef pair<int, pair<int, bool> > data;

int main() {
  while (true) {
    int n, m;
    cin >> n >> m;
    if (n == 0 && m == 0) break;
    vector<vector<pair<int, int> > > edge(n);
    rep (i, m) {
      int a, b, c;
      cin >> a >> b >> c;
      --a, --b;
      edge[a].push_back(make_pair(c, b));
      edge[b].push_back(make_pair(c, a));
    }
    vector<vector<int> > free(n);
    rep (i, n) rep (j, edge[i].size()) rep (k, edge[edge[i][j].second].size()) {
      free[i].push_back(edge[edge[i][j].second][k].second);
    }
    vector<int> cost[2];
    rep (j, 2) rep (i, n) cost[j].push_back(1000000000);
    priority_queue<data, vector<data>, greater<data> > que;
    que.push(make_pair(0, make_pair(0, false)));
    while (!que.empty()) {
      data now = que.top();
      que.pop();
      if (cost[now.second.second][now.second.first] <= now.first) continue;
      cost[now.second.second][now.second.first] = now.first;
      //cout << now.second.first << " " << now.first << " " << now.second.second << endl;
      rep (i, edge[now.second.first].size()) {
	que.push(make_pair(now.first + edge[now.second.first][i].first, make_pair(edge[now.second.first][i].second, now.second.second)));
      }
      if (now.second.second) continue;
      rep (i, free[now.second.first].size()) {
	que.push(make_pair(now.first, make_pair(free[now.second.first][i], true)));
      }
    }
    cout << min(cost[0][n - 1], cost[1][n - 1]) << endl;
  }
  return 0;
}
