a = eval(input())
for _ in range(a):
	b = eval(input())
	c = input().split()
	c = [int(x) for x in c]
	i = len(c) - 1
	j = len(c)
	ls = [0] * b
	while i >= 0:
		t = i - c[i]
		if t < j:
			j = t
		if j < i:
			ls[i] = 1
		elif j == len(c) and c[i] != 0:
			ls[i] = 1
		i -= 1
	ls = [str(x) for x in ls]
	print(' '.join(ls))
