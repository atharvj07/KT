def ceil(x, y):
	if x % y == 0:
		return x // y
	return x // y + 1
from math import gcd
(n, m, q) = map(int, input().split(' '))
inn = n // gcd(n, m)
inm = m // gcd(n, m)
for _ in range(q):
	(sx, sy, ex, ey) = map(int, input().split(' '))
	if n == 1 or m == 1:
		print('YES')
	elif sx != ex:
		if sx == 1:
			if ceil(sy, inn) == ceil(ey, inm):
				print('YES')
			else:
				print('NO')
		elif ceil(ey, inn) == ceil(sy, inm):
			print('YES')
		else:
			print('NO')
	elif sx == 1:
		if ceil(sy, inn) == ceil(ey, inn):
			print('YES')
		else:
			print('NO')
	elif ceil(ey, inm) == ceil(sy, inm):
		print('YES')
	else:
		print('NO')
