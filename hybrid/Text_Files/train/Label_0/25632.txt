#include <bits/stdc++.h>
void quicksort(int a[], int b[], int l, int r);
void quicksort(int a[], int b[], int l, int r) {
  int i, j;
  int x, y;
  i = l;
  j = r;
  x = a[i];
  y = b[i];
  while (i < j) {
    while ((i < j) && (a[j] <= x)) {
      j--;
    }
    if (i < j) {
      a[i] = a[j];
      b[i] = b[j];
      i++;
    }
    while ((i < j) && (a[i] >= x)) {
      i++;
    }
    if (i < j) {
      a[j] = a[i];
      b[j] = b[i];
      j--;
    }
  }
  a[i] = x;
  b[i] = y;
  if (l < j - 1) quicksort(a, b, l, i - 1);
  if (i + 1 < r) quicksort(a, b, i + 1, r);
}
int main() {
  int i, n, re;
  int a[2010], b[2010], c[2010];
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    b[i] = i;
  }
  quicksort(a, b, 1, n);
  re = 1;
  c[b[1]] = 1;
  for (i = 2; i <= n; i++) {
    if (a[i] == a[i - 1]) {
      c[b[i]] = re;
    } else {
      re = i;
      c[b[i]] = re;
    }
  }
  for (i = 1; i <= n; i++) printf("%d ", c[i]);
  printf("\n");
}
