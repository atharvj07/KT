public class C00441D2PB {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        int[] count = new int[m];
        
        int maxR = -1;
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            int r = a[i] % m;
            count[r]++;
            if (count[r] == k) {
                maxR = r;
                break;
            }
        }
        
        if (maxR == -1) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            int countSameR = 0;
            for (int i = 0; i < n; i++) {
                int r = a[i] % m;
                if (r == maxR) {
                    System.out.print(a[i] + " ");
                    countSameR++;
                    if (countSameR == k) {
                        break;
                    }
                }
                
            }
        }
    }
}
