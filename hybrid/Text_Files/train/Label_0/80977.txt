import math

def buy_the_string():
	t = int(input())
	for _ in range(t):
		(n, c0, c1, h) = [int(x) for x in input().split(' ')]
		binary_str = input()
		sum_str = 0
		for x in binary_str:
			if x == '1':
				sum_str += min(c1, c0 + h)
			else:
				sum_str += min(c0, c1 + h)
		print(sum_str)
buy_the_string()
