#include <bits/stdc++.h>
using namespace std;
int main() {
  char s[100005], t[100005], p[100005];
  int i, n, c = 0;
  scanf("%s", s);
  scanf("%s", t);
  for (i = 0; s[i] != '\0'; i++) {
    if (s[i] == t[i])
      p[i] = s[i];
    else {
      if (c % 2 == 0)
        p[i] = (s[i] == '0' ? '1' : '0');
      else
        p[i] = s[i];
      c++;
    }
  }
  p[i] = '\0';
  if (c % 2 == 0)
    puts(p);
  else
    cout << "impossible";
}
