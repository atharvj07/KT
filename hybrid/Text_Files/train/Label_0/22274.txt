#include <bits/stdc++.h>
using namespace std;
template <typename T, typename U>
bool ckmin(T &a, U b) {
  return b < a ? a = b, true : false;
}
template <typename T, typename U>
bool ckmax(T &a, U b) {
  return b > a ? a = b, true : false;
}
int n, MOD;
vector<int> v, second, res, st;
void calc(const vector<int> &second) {
  int n = second.size();
  res.assign(n, 0);
  int l = 0, r = 0;
  for (int i = 1; i < n; ++i) {
    res[i] = max(min(res[i - l], r - i), 0);
    while (i + res[i] < n && second[i + res[i]] == second[res[i]]) ++res[i];
    if (i + res[i] > r) {
      l = i;
      r = i + res[i];
    }
  }
  if (n) res[0] = n;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> MOD;
  v.assign(n, 0);
  for (int i = 0; i < n; ++i) {
    cin >> v[i];
  }
  sort(v.begin(), v.end());
  second.push_back((v[0] - v[n - 1] + MOD) % MOD);
  for (int i = 1; i < n; ++i) {
    second.push_back(v[i] - v[i - 1]);
  }
  reverse(second.begin(), second.end());
  second.push_back(-1);
  for (int j = 0; j < 2; ++j)
    for (int i = n - 1; i >= 0; --i) {
      second.push_back(second[i]);
    }
  calc(second);
  for (int i = 1; i < (int)(second).size(); ++i) {
    if (res[i] == n) st.push_back((v[(i - 2) % n] + v[n - 1]) % MOD);
  }
  sort(st.begin(), st.end());
  st.erase(unique(st.begin(), st.end()), st.end());
  cout << (int)(st).size() << "\n";
  for (auto i : st) cout << i << " ";
  return 0;
}
