from sys import stdin
n = int(input())
if n == 3:
	print('3 1 2')
else:
	pairs = [(0, 0)]
	for i in range(n):
		(a, b) = [int(x) for x in stdin.readline().split()]
		pairs.append((a, b))
	ans = []
	j = 1
	while len(ans) < n:
		p = pairs[j]
		if set([p[0]]) & set(pairs[p[1]]) != set():
			ans.append(p[1])
			ans.append(p[0])
			j = p[0]
		if set([p[1]]) & set(pairs[p[0]]) != set():
			ans.append(p[0])
			ans.append(p[1])
			j = p[1]
	print(*ans[0:n])
