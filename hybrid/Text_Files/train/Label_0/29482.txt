#include <bits/stdc++.h>
using namespace std;
const int maxN = 405;
int N, M, dis[maxN];
int mp[maxN][maxN];
struct node {
  int id, dis;
  node(int a = 0, int b = 0) : id(a), dis(b) {}
  friend bool operator<(node e1, node e2) { return e1.dis > e2.dis; }
};
priority_queue<node> Q;
inline int dij(int op) {
  memset(dis, 0x3f3f3f3f, sizeof(dis));
  dis[1] = 0;
  while (!Q.empty()) Q.pop();
  Q.push(node(1, 0));
  while (!Q.empty()) {
    node now = Q.top();
    Q.pop();
    int u = now.id;
    for (int i = 2; i <= N; i++) {
      if (mp[u][i] == op) {
        if (dis[i] > now.dis + 1) {
          dis[i] = now.dis + 1;
          Q.push(node(i, dis[i]));
        }
      }
    }
  }
  return dis[N];
}
int main() {
  scanf("%d%d", &N, &M);
  memset(mp, 0, sizeof(mp));
  for (int i = 1, u, v; i <= M; i++) {
    scanf("%d%d", &u, &v);
    mp[u][v] = mp[v][u] = 1;
  }
  int d1 = dij(0), d2 = dij(1);
  int ans = max(d1, d2);
  if (ans >= 0x3f3f3f3f)
    printf("-1\n");
  else
    printf("%d\n", ans);
  return 0;
}
