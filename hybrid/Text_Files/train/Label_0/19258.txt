#include <cstdio>
#define rng(x, y) (0 <= x && x < W && 0 <= y && y < H)
#pragma warning(disable : 4996)
using namespace std;
int H, W, d[105][105][4]; long long L; char c[105][105], w[5] = "NESW";
int dx[4] = { 0, 1, 0, -1 }, dy[4] = { -1, 0, 1, 0 };
int main() {
	while (scanf("%d%d%lld", &H, &W, &L), L) {
		for (int i = 0; i < H; i++) scanf("%s", c[i]);
		int sx = -1, sy = -1, sd = -1;
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				for (int k = 0; k < 4; k++) {
					if (c[i][j] == w[k]) sx = j, sy = i, sd = k;
					d[i][j][k] = -1;
				}
			}
		}
		d[sy][sx][sd] = 0;
		int x = sx, y = sy, r = sd;
		while (true) {
			int tx = x, ty = y, tr = r;
			while (!rng(x + dx[r], y + dy[r]) || c[y + dy[r]][x + dx[r]] == '#') r = (r + 1) % 4;
			x += dx[r], y += dy[r];
			if (d[y][x][r] == -1) d[y][x][r] = d[ty][tx][tr] + 1;
			else {
				int t = d[ty][tx][tr] + 1, sa = t - d[y][x][r];
				if (L >= t) L = t + (L - t) % sa - sa;
				break;
			}
		}
		for (int i = 0; i < H; i++) {
			for (int j = 0; j < W; j++) {
				for (int k = 0; k < 4; k++) {
					if (d[i][j][k] == L) printf("%d %d %c\n", i + 1, j + 1, w[k]);
				}
			}
		}
	}
	return 0;
}
