def decode(s):
	res = '' + s[0]
	string = []
	for letter in s:
		string.append(letter)
	string = string[1:]
	while len(string) != 0:
		if len(string) % 2 == 0:
			res = str(string[0]) + res
		else:
			res += str(string[0])
		string = string[1:]
	return res

arg = input()
