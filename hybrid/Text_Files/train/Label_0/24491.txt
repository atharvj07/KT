#include <bits/stdc++.h>

using namespace std;

typedef long double ld;
typedef long long ll;

const int M = 10;

int n = 6;
ld p[M], q[M];
vector<ld> a;

void read() {
  for (int i = 0; i < n; ++i) {
    cin >> p[i];
    p[i] /= 100;
  }

  for (int i = 0; i < n; ++i) {
    cin >> q[i];
    q[i] /= 100;
  }
}

ld fun(ld x) {
  ld ans = 0;
  for (int i = 0; i < n; ++i)
    ans += max<ld>(0, x * (p[i] + q[i]) - q[i]);
  ans += 1 - x;
  return ans;
}

void kill() {
  a.push_back(0);
  a.push_back(1);
  for (int i = 0; i < n; ++i)
    if (p[i] + q[i] > 0)
      a.push_back(q[i] / (p[i] + q[i]));

  ld ans = 1;
  for (ld x : a) {
    //cerr << x << " " << fun(x) << endl;
    ans = min(ans, fun(x));
  }

  cout << ans << endl;
}

int main() {
#ifdef LOCAL
  assert(freopen("d.in", "r", stdin));
#endif

  ios_base::sync_with_stdio(false);
  cout.precision(13);
  cout << fixed;

  read();
  kill();

}
