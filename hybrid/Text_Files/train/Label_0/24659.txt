#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.0);
const int INF = 1000900000;
const long long mod = 1000000007;
vector<vector<int> > g;
vector<bool> used;
vector<bool> changed;
vector<int> root, p;
void dfs(int v, int from) {
  used[v] = true;
  for (int i = 0; i < (int)g[v].size(); i++) {
    int to = g[v][i];
    if (to == v) {
      root.push_back(v);
      continue;
    }
    if (to == from) continue;
    if (!used[to]) {
      dfs(to, v);
    } else {
      if (p[to] == v) {
        p[to] = to;
        changed[to] = 1;
        root.push_back(to);
      } else if (p[v] == to) {
        p[v] = v;
        changed[v] = 1;
        root.push_back(v);
      }
    }
  }
}
int main() {
  int n;
  cin >> n;
  p.resize(n);
  g.resize(n);
  used.resize(n, 0);
  changed.resize(n, 0);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    p[i]--;
    g[i].push_back(p[i]);
    if (i != p[i]) g[p[i]].push_back(i);
  }
  for (int i = 0; i < n; i++) {
    if (!used[i]) {
      dfs(i, i);
    }
  }
  int to = 0;
  for (int i = 0; i < (int)root.size(); i++) {
    if (!changed[root[i]]) {
      to = i;
      break;
    }
  }
  int ans = 0;
  for (int i = 0; i < (int)root.size(); i++) {
    if (i != to) {
      p[root[i]] = root[to];
      changed[root[i]] = 1;
    }
    ans += changed[root[i]];
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) {
    cout << p[i] + 1 << ' ';
  }
  return 0;
}
