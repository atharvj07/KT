def lower_bound(v, x):
	low = 0
	high = len(v) - 1
	ans = len(v)
	while low <= high:
		mid = (low + high) // 2
		if v[mid] < x:
			low = mid + 1
		else:
			ans = mid
			high = mid - 1
	return ans
