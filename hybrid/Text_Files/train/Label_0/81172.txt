import sys
ans = []
t = int(sys.stdin.readline())
for _ in range(t):
	n = int(sys.stdin.readline())
	s = sys.stdin.readline().strip()
	if s.find('aa') > -1:
		ans.append('2')
	elif s.find('aca') > -1 or s.find('aba') > -1:
		ans.append('3')
	elif s.find('abca') > -1 or s.find('acba') > -1:
		ans.append('4')
	elif s.find('abbacca') > -1 or s.find('accabba') > -1:
		ans.append('7')
	else:
		ans.append('-1')
print('\n'.join(ans))
