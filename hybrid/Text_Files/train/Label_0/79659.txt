elem = int(input())
for x in range(elem):
	temp = input()
	temp2 = input()
	ar = list(temp)
	ar2 = list(temp2)
	for q in range(len(ar2)):
		temp = ar.copy()
		ch = 0
		for x in range(q, len(ar2)):
			if ar2[x] in temp:
				temp.remove(ar2[x])
			else:
				break
			if len(temp) == 0:
				print('YES')
				ch = 1
				break
		if ch == 1:
			break
	if ch == 0:
		print('NO')
