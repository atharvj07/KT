#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  ;
  long long n, nn = 0;
  cin >> n;
  long long a[n];
  for (long long i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  for (long long i = 0; i < n - 2; i++) {
    nn = 0;
    if (a[i] + a[i + 1] > a[i + 2] && a[i + 1] + a[i + 2] > a[i] &&
        a[i] + a[i + 2] > a[i + 1]) {
      nn = 1;
      cout << "YES";
      break;
    }
  }
  if (nn == 0) cout << "NO";
  return 0;
}
