#include <bits/stdc++.h>
using namespace std;
vector<int> e[100010];
int a[210], b[210], an, bn;
int d[210], dn;
struct abc {
  int x, y;
} c[200010];
int u[100010];
int p[100010];
int main() {
  int n, m, h, t, i, j, x, y, s, rrr = 1;
  scanf("%d%d%d%d", &n, &m, &h, &t);
  m <<= 1;
  for (i = 0; i < m; i += 2) {
    scanf("%d%d", &x, &y);
    c[i].x = x, c[i].y = y;
    c[i + 1].x = y, c[i + 1].y = x;
    e[x].push_back(y);
    e[y].push_back(x);
  }
  for (i = 0; i < m; i++) {
    x = c[i].x;
    y = c[i].y;
    an = bn = dn = 0;
    s = h + t;
    for (j = 0; j < (int)e[x].size(); j++) {
      int z = e[x][j];
      if (z == y) continue;
      u[z] = rrr;
      p[z] = an;
      a[an++] = z;
      s--;
      if (s == 0) break;
    }
    s = h + t;
    for (j = 0; j < (int)e[y].size(); j++) {
      int z = e[y][j];
      if (z == x) continue;
      s--;
      if (u[z] == rrr) {
        swap(a[p[z]], a[an - 1]);
        an--;
        d[dn++] = z;
      } else {
        b[bn++] = z;
      }
      if (s == 0) break;
    }
    rrr++;
    if (an + dn >= h && bn + dn - max(h - an, 0) >= t) {
      printf("YES\n");
      printf("%d %d\n", x, y);
      for (i = 0; i < an && i < h; i++) {
        printf("%d%c", a[i], i == (h - 1) ? '\n' : ' ');
      }
      for (i = an; i < h; i++) {
        printf("%d%c", d[--dn], i == (h - 1) ? '\n' : ' ');
      }
      for (i = 0; i < bn && i < t; i++) {
        printf("%d%c", b[i], i == (t - 1) ? '\n' : ' ');
      }
      for (i = bn; i < t; i++) {
        printf("%d%c", d[--dn], i == (t - 1) ? '\n' : ' ');
      }
      return 0;
    }
  }
  printf("NO\n");
  return 0;
}
