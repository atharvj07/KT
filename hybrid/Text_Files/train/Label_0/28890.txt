#include <cstdio>
#include <cctype>
#include <cassert>
#include <string>
#include <map>
#include <algorithm>

char parse(const std::string& s, size_t& i, std::map<char, int>& vic) {
  if (islower(s[i])) return s[i++];

  assert(s[i] == '[');
  char c1=parse(s, ++i, vic);
  assert(s[i] == '-');
  char c2=parse(s, ++i, vic);
  assert(s[i] == ']');
  ++i;

  if (vic[c1] == vic[c2]) throw 0;
  if (vic[c1] > vic[c2]) std::swap(c1, c2);
  if (vic[c1] != 0) throw 0;
  --vic[c2];
  return c2;
}

int main() {
  char buf[1024];
  scanf("%s", buf);

  std::map<char, int> vic;
  while (true) {
    char a;
    int v;
    if (scanf(" %c %d", &a, &v) == EOF) break;
    vic[a] = v;
  }

  std::string s=buf;
  size_t i=0;
  try {
    if (vic[parse(s, i, vic)] != 0) throw 0;
    puts("Yes");
  } catch (int) {
    puts("No");
  }
}
