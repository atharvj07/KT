(n, m) = [int(x) for x in input().split()]
ms = []
con = [set() for i in range(n)]
loc = [0] * n
for i in range(m):
	(a, b) = [int(x) for x in input().split()]
	a -= 1
	b -= 1
	con[a].add(b)
	con[b].add(a)
	ms.append((a, b))
ans = [[] for i in range(n)]
if len(con[0]) == 0:
	con[0].add(n + 10)
l = len(con[0])
ans[0] = [i for i in range(l)]
for i in range(1, n):
	loc[i] = l
	if len(con[i]) == 0:
		con[i].add(n + 10)
	for s in con[i]:
		if s < i:
			ans[i].append(loc[s])
			loc[s] += 1
		else:
			ans[i].append(l)
			l += 1
for i in range(n):
	print(len(ans[i]))
	for s in ans[i]:
		print(i + 1, s + 1)
