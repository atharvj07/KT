def give(ind, n):
	if n % 2 == 1 or ind <= n * n:
		return ind % (n * n) + int(ind == n * n) * n * n
	else:
		return ind + 1 - n * n
t = int(input())
for _ in range(t):
	n = int(input())
	if n == 2:
		print(-1)
	elif n == 1:
		print(1)
	else:
		grid = [[0] * n for i in range(n)]
		val = -1
		for i in range(n):
			for j in range(n):
				val += 2
				grid[i][j] = give(val, n)
			print(*grid[i])
