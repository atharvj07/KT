#include <bits/stdc++.h>
using namespace std;
struct p {
  int id;
  long long x, y;
};
bool myfunc(p A, p B) {
  if (A.x < B.x) return true;
  if (A.x == B.x && A.y < B.y) return true;
  return false;
}
bool func(long long x0, long long y0, long long x1, long long y1, long long x2,
          long long y2) {
  if ((y1 - y0) * (x2 - x1) != (x1 - x0) * (y2 - y1)) return true;
  return false;
}
vector<p> G;
int main() {
  int n;
  long long a, b;
  int aux, aux2, aux3;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a >> b;
    p z;
    z.id = i + 1;
    z.x = a;
    z.y = b;
    G.push_back(z);
  }
  sort(G.begin(), G.end(), myfunc);
  aux = G[0].id;
  aux2 = G[1].id;
  for (int i = 2; i < n; i++) {
    if (func(G[0].x, G[0].y, G[1].x, G[1].y, G[i].x, G[i].y)) {
      aux3 = G[i].id;
      break;
    }
  }
  printf("%d %d %d\n", aux, aux2, aux3);
  return 0;
}
