#include <bits/stdc++.h>
using namespace std;
string s;
vector<vector<string>> ans(1000002);
int mx = 0;
bool vis[1000005] = {0};
void dfs(string u, int depth, int index, int cnt) {
  mx = max(mx, depth);
  ans[depth].push_back(u);
  for (int i = index; i < s.size(); i++) {
    if (cnt == 0) return;
    if (vis[i]) continue;
    cnt--;
    string f = "";
    int j = i, c = 0;
    while (s[j] != ',') {
      vis[j] = 1;
      f += s[j];
      j++;
    }
    vis[j] = 1;
    j++;
    while (s[j] != ',' && j < s.size()) {
      vis[j] = 1;
      c = c * 10 + (s[j] - '0');
      j++;
    }
    vis[j] = 1;
    j++;
    dfs(f, depth + 1, j, c);
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> s;
  for (int i = 0; i < s.size(); i++) {
    if (vis[i]) continue;
    string f = "";
    int j = i, c = 0;
    while (s[j] != ',') {
      vis[j] = 1;
      f += s[j];
      j++;
    }
    vis[j] = 1;
    j++;
    while (s[j] != ',' && j < s.size()) {
      vis[j] = 1;
      c = c * 10 + (s[j] - '0');
      j++;
    }
    vis[j] = 1;
    j++;
    dfs(f, 1, j, c);
  }
  cout << mx << endl;
  for (int i = 1; i <= mx; i++) {
    for (int j = 0; j < ans[i].size(); j++) cout << ans[i][j] << " ";
    cout << endl;
  }
  return 0;
}
