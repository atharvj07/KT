#include <iostream>
#include <vector>
using namespace std;
const int MAX_NK = 410;
const int INF = 1 << 20;
struct edge{
  int to;
  int cap;
  int cost;
  int rev;
  edge(){};
  edge(int to, int cap, int cost, int rev):to(to),cap(cap),cost(cost),rev(rev){
  }
};

int V;
vector<edge> G[MAX_NK];
int dist[MAX_NK];
int prevv[MAX_NK];
int preve[MAX_NK];

void add_edge(int from, int to, int cap, int cost){
  G[from].push_back(edge(to, cap, cost, G[to].size()));
  G[to].push_back(edge(from, 0, -cost, G[from].size() -1));
}

int min_cost_flow(int s, int t, int f, int K){
  int res = 0;
  int use_cost = 0;
  
  while (f > 0){
    fill(dist, dist + V, INF);
    dist[s] = 0;
    bool update = true;
    while (update){
      update = false;
      for (int v = 0; v < V; v++){
	if (dist[v] == INF)continue;
	for (int i = 0; i < G[v].size(); i++){
	  edge &e = G[v][i];
	  if (e.cap > 0 and dist[e.to] > dist[v] + e.cost){
	    dist[e.to] = dist[v] + e.cost;
	    prevv[e.to] = v;
	    preve[e.to] = i;
	    update = true;
	  }
	}
      }
    }

    //cout << dist[t] << " " << (K - use_cost) << endl;
    if (dist[t] >= INF || (K - use_cost) < dist[t]){
      return f;
    }
    
    int d = f;
    for (int v = t; v != s; v = prevv[v]){
      d = min(d, G[prevv[v]][preve[v]].cap);
    }
    f -= d;
    res += d * dist[t];
    
    for (int v = t; v != s; v = prevv[v]){
      edge &e = G[prevv[v]][preve[v]];
      e.cap -= d;
      G[v][e.rev].cap += d;
      use_cost += e.cost;
    }
  }
  
  return f;
}

int A,B,K;
vector<string> RA,RB;
int main(){
  cin >> A >> B >> K;
  V = A + B + 2;
  string tmp;

  for (int i = 0; i < A; i++){
    cin >> tmp;
    RA.push_back(tmp);
  }
  for (int i = 0; i < B; i++){
    cin >> tmp;
    RB.push_back(tmp);
  }

  for (int i = 0; i < A; i++){
    for (int j = 0; j < B; j++){
      int AB = RA[i][j] - '0';
      int BA = RB[j][i] - '0';

      if ((AB & BA) == 1){//1 1
	add_edge(i + 1, A + 1 + j, 1, 2);
	//cout << i << " " << j << " " << 2 << endl;
      }else if ((AB ^ BA) == 1){//0 1 1 0
	add_edge(i + 1, A + 1 + j, 1, 1);
	//cout << i << " " << j << " " << 1 << endl;	
      }else{//0 0
	add_edge(i + 1, A + 1 + j, 1, 0);
	//cout << i << " " << j << " " << 0 << endl;		
      }
    }
  }
  for (int i = 0; i < A; i++){//src A
    add_edge(0, i + 1, 1, 0);
  }
  for (int j = 0; j < B; j++){//B tar
    add_edge(A + 1 + j, A + B + 1, 1, 0);
  }
 
  //  cout << "hoge" << endl;

  int res = min_cost_flow(0, A + B + 1, (A + B), K);
  
  cout << (A + B - res) << endl;  
}
