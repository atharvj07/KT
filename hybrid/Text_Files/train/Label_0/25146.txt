#include <bits/stdc++.h>

#define REP(i, n) for (int i = 0; i < n; i++)
#define FOR(i, n, m) for (int i = n; i < m; i++)
#define ll long long int

using namespace std;

int main()
{
  int n;
  pair<int, int> v1[100005] = {}, v2[100005] = {};
  cin >> n;
  REP(i,n/2) {
    int a,b;
    cin >> a >> b;
    v1[a].first++;
    v1[a].second = a;
    v2[b].first++;
    v2[b].second = b;
  }
  sort(v1, v1 + 100005, greater<pair<int, int> >());
  sort(v2, v2 + 100005, greater<pair<int, int> >());

  int m = 0;
  if(v1[0].second != v2[0].second) {
    m = v1[0].first + v2[0].first;
  } else {
    m = max(v1[1].first + v2[0].first, v1[0].first + v2[1].first);
  }

  cout << n - m << endl;
}
