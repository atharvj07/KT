import math as mt
for _ in range(int(input())):
	(A, B, n) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	for i in range(n):
		b[i] = mt.ceil(b[i] / A)
	s = 0
	ratio = []
	for i in range(n):
		ratio.append([a[i], b[i]])
	ratio = sorted(ratio, key=lambda x: x[0])
	flag = True
	for i in range(n):
		if B - ratio[i][0] * (ratio[i][1] - 1) > 0:
			B -= ratio[i][0] * ratio[i][1]
		else:
			flag = False
	if flag:
		print('YES')
	else:
		print('NO')
