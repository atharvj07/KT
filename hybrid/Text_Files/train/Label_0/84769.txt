(n, k) = map(int, input().strip().split())
a = list(map(int, input().strip().split()))
l = 0
ans = 0
resL = -1
resR = -1
cntZero = 0
for r in range(n):
	if a[r] == 0:
		cntZero += 1
	while cntZero > k:
		if a[l] == 0:
			cntZero -= 1
		l += 1
	if ans < r - l + 1:
		ans = r - l + 1
		resL = l
		resR = r
print(ans)
if resL != -1:
	for i in range(resL, resR + 1):
		a[i] = 1
for i in range(n):
	print(a[i], end=' ')
