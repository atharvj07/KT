#include <bits/stdc++.h>
using namespace std;
char s[300005] = {}, tmp[300005] = {};
int pos[30][100000], top[30];
int ABS(int a) {
  if (a < 0)
    return -1 * a;
  else
    return a;
}
int main() {
  int n, k, i, j, lens, lent, left, right, flag;
  long long ans, minlen;
  scanf("%d%d%*c", &n, &k);
  gets(s);
  lens = strlen(s);
  for (i = 0; i < lens; i++) {
    pos[s[i] - 'a'][top[s[i] - 'a']] = i;
    top[s[i] - 'a']++;
  }
  while (n--) {
    gets(tmp);
    lent = strlen(tmp);
    ans = 0;
    for (i = 0; i < lent; i++) {
      left = i;
      right = i;
      if (left >= lens) left = lens - 1;
      minlen = 2140000000ll;
      if (top[tmp[i] - 'a'] > 0)
        flag = 0;
      else
        flag = 1;
      for (j = 0; j < top[tmp[i] - 'a']; j++) {
        if (ABS(pos[tmp[i] - 'a'][j] - i) < minlen)
          minlen = ABS(pos[tmp[i] - 'a'][j] - i);
      }
      if (flag == 0)
        ans += minlen;
      else
        ans += lent;
    }
    cout << ans << endl;
  }
  return 0;
}
