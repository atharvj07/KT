from collections import deque
n = int(input())
a = [int(x) for x in input().split()]
rs = deque()
num_nonzero = 0
for (i, x) in enumerate(a):
	if x != 0:
		num_nonzero += 1
		if num_nonzero == 2:
			rs.append(i)
			num_nonzero = 1
else:
	if num_nonzero == 0:
		print('NO')
	else:
		print('YES')
		print(len(rs) + 1)
		l = 1
		while rs:
			r = rs.popleft()
			print(l, r)
			l = r + 1
		print(l, n)
