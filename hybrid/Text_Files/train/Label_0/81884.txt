(n, m) = map(int, input().split(' '))
pk = [i for i in range(1, n + 1)]
dele = [False for i in range(n)]
for i in range(m):
	(u, v) = map(int, input().split(' '))
	if dele[u - 1] == False:
		pk.remove(u)
		dele[u - 1] = True
	if dele[v - 1] == False:
		pk.remove(v)
		dele[v - 1] = True
cen = pk.pop(0)
print(n - 1)
for i in range(1, n + 1):
	if i == cen:
		continue
	else:
		print(cen, end=' ')
		print(i)
