#include <iostream>
using namespace std;

int main() {
    int t; 
    cin>>t;
    while(t--){
        int n,m;
        cin>>n>>m;
        int x[n];
        int y[m];
        for(int i=0; i<n; i++)
            cin>>x[i];
        for(int i=0; i<m; i++)
            cin>>y[i];
            
        int count = 0;
        
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(x[j]==y[i])
                    count++;
            }
        }
        cout<<count<<endl;
    }
    return 0;
}
