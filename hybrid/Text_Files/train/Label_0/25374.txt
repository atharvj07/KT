#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

map<int,int> mp;

int main(){
  int n;
  cin>>n;
  vector<int> t(n);
  for(int i=0;i<n;i++){
    cin>>t[i];
    mp[ t[i] ]++;
  }
  int ans=0;
  for(auto p : mp ){
    if( p.second >= p.first ){
      ans+= min( p.second ,  p.second-p.first );
    }else{
      ans+=p.second;
    }
  }
  cout<<ans<<endl;
  return 0;
}
