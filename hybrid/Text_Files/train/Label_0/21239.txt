#include <bits/stdc++.h>
using namespace std;
const int maxn = 5e5 + 100;
vector<int> G[maxn];
int a[maxn];
int mx[maxn];
int main() {
  std::ios::sync_with_stdio(false);
  int n, m;
  cin >> n >> m;
  for (int i = 0; i < m; ++i) {
    int u, v;
    cin >> u >> v;
    --u;
    --v;
    G[u].push_back(v);
    G[v].push_back(u);
  }
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  vector<int> order(n);
  iota(order.begin(), order.end(), 0);
  sort(order.begin(), order.end(), [&](int i, int j) { return a[i] < a[j]; });
  bool flag = true;
  vector<int> res;
  for (int u : order) {
    if (mx[u] != a[u] - 1) {
      cout << -1 << endl;
      return 0;
    }
    res.push_back(u + 1);
    for (int v : G[u]) {
      if (a[v] == a[u]) {
        cout << -1 << endl;
        return 0;
      }
      if (mx[v] == a[u] - 1) {
        mx[v] = a[u];
      }
    }
  }
  for (int i = 0; i < n; i++) {
    cout << res[i] << " ";
  }
  cout << endl;
}
