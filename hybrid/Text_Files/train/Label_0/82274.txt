count = int(input())
count_variable = 1
colours = input()
answer = 'Yes'
colour = colours[0]
if colour == '?':
	count_variable += 1
for i in range(1, count):
	if colours[i] == colour and colour != '?':
		answer = 'No'
	if colours[i] == '?':
		if i == count - 1:
			count_variable += 1
		elif colours[i - 1] == colours[i + 1] or colours[i - 1] == '?' or colours[i + 1] == '?':
			count_variable += 1
	colour = colours[i]
if count_variable == 1:
	answer = 'No'
print(answer)
