(n, m) = map(int, input().split())
a = []
b = []
for i in range(n):
	a.append(list(input()))
	b.append(list('.' * m))
for i in range(1, n - 1):
	for j in range(1, m - 1):
		if a[i - 1][j - 1] != '.' and a[i - 1][j] != '.' and (a[i - 1][j + 1] != '.') and (a[i][j - 1] != '.') and (a[i][j + 1] != '.') and (a[i + 1][j - 1] != '.') and (a[i + 1][j] != '.') and (a[i + 1][j + 1] != '.'):
			b[i - 1][j - 1] = '#'
			b[i - 1][j] = '#'
			b[i - 1][j + 1] = '#'
			b[i][j - 1] = '#'
			b[i][j + 1] = '#'
			b[i + 1][j - 1] = '#'
			b[i + 1][j] = '#'
			b[i + 1][j + 1] = '#'
for i in range(n):
	for j in range(m):
		if a[i][j] != b[i][j]:
			print('NO')
			exit()
print('YES')
