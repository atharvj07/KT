int Sum(int y,int x){
  int re=sum[y][x];
  //cout<<pii(x,y)<<endl;
  if(x-t-2>=0)
    re+=sumd[y+1][x-t-2];
  //cout<<re<<",";
  if(y-t>=0){
    if(y-t)re-=sum[y-t-1][x];
    if(x)re-=sumd[y-t][x-1];
  }else{
    if(x-t+y)re-=sumd[0][x-t+y-1];
  }
  //cout<<re<<endl;
  return re;
}
