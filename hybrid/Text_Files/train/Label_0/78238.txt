(n, k) = list(map(int, input().split()))
a = list(map(int, input().split()))
t = max(a)
c = 0
ct = 0
s = []
if k == 1:
	print('Yes')
	print(len(a))
elif n == k:
	if a.count(a[0]) == len(a):
		print('Yes')
		for i in range(k):
			print(1, end=' ')
	else:
		print('No')
else:
	ic = 0
	for i in a:
		ct += i
		if ct < t:
			if ic == n - 1:
				print('No')
				exit()
			c += 1
		elif ct == t:
			c += 1
			ct = 0
			s.append(c)
			c = 0
		else:
			print('No')
			exit()
		ic += 1
	if len(s) == k:
		print('Yes')
		for i in s:
			print(i, end=' ')
	else:
		print('No')
