(OPEN, CLOSE) = ('(', ')')

def validate(brackets: str) -> bool:

	def check(brackets: str) -> bool:
		stack = []
		for item in brackets:
			if item == CLOSE:
				if stack:
					stack.pop()
					continue
				return False
			stack.append(item)
		return not stack
	superset = {'A', 'B', 'C'}
	(start, end) = (brackets[0], brackets[-1])
	middle = (superset - {start} - {end}).pop()
	brackets = brackets.replace(start, OPEN).replace(end, CLOSE)
	return check(brackets.replace(middle, OPEN)) or check(brackets.replace(middle, CLOSE))

def read_input(filename: str=None) -> None:
	from sys import stdin
	stdin = open(filename) if filename else stdin
	cases = int(stdin.readline())
	while cases:
		cases = cases - 1
		brackets = stdin.readline().strip()
		print('YES') if validate(brackets) else print('NO')

def main():
	read_input()
main()
