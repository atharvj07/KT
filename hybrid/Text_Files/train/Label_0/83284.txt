(a, b) = map(int, input().split(' '))
s = 0
k = 0
c = '1'
q = 0
while s < a + b:
	k = k + 1
	s += k
	c = c + '0'
n = k
if s > a + b:
	c = c[:s - a - b] + '1' + c[s - a - b + 1:]
	n = k - 1
l = k
m = 0
while m < a:
	if c[l] == '0':
		c = c[:l] + '2' + c[l + 1:]
		m = m + l
		q += 1
	l = l - 1
c = c[:l + 1] + '0' + c[l + 2:]
c = c[:a - m + l + 1] + '2' + c[a - m + l + 2:]
print(q)
for i in range(1, k + 1):
	if c[i] == '2':
		print(i, end=' ')
print()
if n - q > 0:
	print(n - q)
	for i in range(1, k + 1):
		if c[i] == '0':
			print(i, end=' ')
else:
	print(0)
