from collections import defaultdict
from sys import stdin
input = stdin.readline
import math

def put():
	return map(int, input().split())
n = int(input())
l = list(put())
seek = defaultdict()
seek[l[0]] = 1
sum = l[0]
ans = 0
for i in range(1, n):
	sum += l[i]
	if sum in seek or sum == 0:
		ans += 1
		sum = l[i]
		seek = defaultdict()
		seek[l[i]] = 1
	else:
		seek[sum] = 1
print(ans)
