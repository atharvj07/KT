#include <bits/stdc++.h>
using namespace std;
int n, k, i, j, Max, Max1, Max2, l, ans, a[101], b[101], f1[100001], f2[100001];
int main() {
  scanf("%d%d\n", &n, &k);
  for (i = 1; i <= n; i++) scanf("%d", &a[i]);
  scanf("\n");
  for (i = 1; i <= n; i++) {
    scanf("%d", &b[i]);
    b[i] = a[i] - k * b[i];
    l += abs(b[i]);
  }
  for (i = 1; i <= l; i++) f1[i] = f2[i] = -1e9;
  for (i = 1; i <= n; i++)
    if (!b[i]) ans += a[i];
  for (i = 1; i <= n; i++)
    if (b[i] > 0) {
      Max1 += b[i];
      for (j = Max1 + b[i]; j >= b[i]; j--)
        f1[j] = max(f1[j], f1[j - b[i]] + a[i]);
    }
  for (i = 1; i <= n; i++)
    if (b[i] < 0) {
      Max2 -= b[i];
      for (j = Max2 - b[i]; j >= -b[i]; j--)
        f2[j] = max(f2[j], f2[j + b[i]] + a[i]);
    }
  for (Max = 0, i = min(Max1, Max2); i; i--) Max = max(Max, f1[i] + f2[i]);
  if (!Max && !ans) {
    printf("-1");
    return 0;
  }
  printf("%d", ans + Max);
}
