inp = lambda : int(input())
inpm = lambda : map(int, input().split())
inpl = lambda : list(inpm())
l2mn = lambda _n, _m: [[0 for __ in range(_m)] for _ in range(_n)]
l2n = lambda _n: l2mn(_n, _n)
INF = int(1e+18)
MOD = int(1000000000.0) + 7
N = 2000009

def solve():
	(n, w) = inpm()
	left = (w + 1) // 2
	a = inpl()
	res = []
	s = 0
	ok = False
	for i in range(n):
		x = a[i]
		if x < left:
			s += x
			res.append(i + 1)
			if s >= left:
				ok = True
				break
		elif x > w:
			continue
		else:
			res = [i + 1]
			ok = True
			break
	if ok:
		print(len(res))
		print(*res)
	else:
		print(-1)

def main():
	t = inp()
	for i in range(t):
		solve()
main()
