#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    vector<int> v(64);
    long long n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
      int x;
      cin >> x;
      ++v[__builtin_ctz(x)];
    }
    int ans = 0;
    for (int i = 0; i <= 62; i++) {
      if (n & (1LL << i)) {
        if (!v[i]) {
          int p = i + 1;
          while (p < 63 && !v[p]) ++p;
          if (p == 63) {
            ans = -1;
            break;
          }
          while (p > i) {
            --v[p];
            --p;
            v[p] += 2;
            ++ans;
          }
        }
        --v[i];
      }
      v[i + 1] += (v[i] >> 1);
    }
    cout << ans << '\n';
  }
  return 0;
}
