(n, m) = map(int, input().split())
dl = [0] * (n + 1)
cl = []
g = [list() for i in range(n + 1)]
for i in range(m):
	(u, v) = map(int, input().split())
	g[u].append(v)
	g[v].append(u)
	dl[u] += 1
	dl[v] += 1
	cl.append((u, v))
pos = dl.index(max(dl))
dl = [0] * (n + 1)
queue = []
dl[pos] = 1
for u in g[pos]:
	print(pos, u)
	queue.append(u)
	dl[u] = 1
for u in queue:
	for v in g[u]:
		if dl[v] != 1:
			print(u, v)
			dl[v] = 1
			queue.append(v)
