(n, c) = map(int, input().split())
problems = list(map(int, input().split()))
time = list(map(int, input().split()))
(lmk, rdw) = (0, 0)
total_time_1 = 0
total_time_2 = sum(time)
for i in range(n):
	total_time_1 += time[i]
	lmk += max(0, problems[i] - c * total_time_1)
	rdw += max(0, problems[i] - c * total_time_2)
	total_time_2 -= time[i]
if lmk > rdw:
	print('Limak')
elif rdw > lmk:
	print('Radewoosh')
else:
	print('Tie')
