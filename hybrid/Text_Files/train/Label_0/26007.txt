#include <bits/stdc++.h>
using namespace std;
int n, i, j, m;
char sol[100005];
string cad;
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  cin >> sol;
  m = strlen(sol);
  for (i = 1; i < n; i++) {
    cin >> cad;
    for (j = 0; j < m; j++) {
      if (sol[j] == '!' || cad[j] == '?' || (sol[j] == '?' && cad[j] == '?'))
        continue;
      if (sol[j] == '?')
        sol[j] = cad[j];
      else if (sol[j] != cad[j])
        sol[j] = '!';
    }
  }
  for (j = 0; j < m; j++) {
    if (sol[j] == '!')
      cout << '?';
    else if (sol[j] == '?')
      cout << 'a';
    else
      cout << sol[j];
  }
  cout << endl;
  return 0;
}
