#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int kval = 3;
    while (n % kval != 0) {
      kval = (kval << 1) + 1;
    }
    cout << n / kval << "\n";
  }
  return 0;
}
