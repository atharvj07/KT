#include <bits/stdc++.h>
using namespace std;
void Karl_Heinrich_Marx() {}
int n, m;
char c[605][605];
void Vladimir_Ilyich_Lenin() {
  int d = 0, cc = 0;
  cin >> n >> m;
  for (int i = 0; i <= n + 10; ++i)
    for (int j = 0; j <= m + 10; ++j) c[i][j] = '.';
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      cin >> c[i][j];
    }
  }
  int i = d, j = cc;
  for (int i = 1; i <= n; ++i)
    for (int j = 1; j <= m; ++j)
      if (i != 0 && j != 0) {
        if (c[i][j] == 'W')
          if (c[i + 1][j] == 'S' || c[i][j + 1] == 'S' || c[i - 1][j] == 'S' ||
              c[i][j - 1] == 'S') {
            cout << "No";
            return;
          }
      }
  cout << "Yes" << endl;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) {
      if (c[i][j] == '.')
        cout << "D";
      else
        cout << c[i][j];
    }
    cout << endl;
  }
}
int main() {
  Karl_Heinrich_Marx();
  Vladimir_Ilyich_Lenin();
  return 0;
}
