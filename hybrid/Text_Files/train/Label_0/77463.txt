for i in range(int(input())):
	(n, m) = map(int, input().split())
	l = []
	for i in range(n):
		l.append(list(map(int, input().split())))
	n1 = n // 2
	m2 = m // 2
	s = 0
	t = [0 for i in range(4)]
	for i in range(n1):
		for j in range(m2):
			t[0] = l[i][j]
			t[1] = l[n - i - 1][j]
			t[2] = l[i][m - j - 1]
			t[3] = l[n - i - 1][m - j - 1]
			t.sort()
			a = t[1]
			s += abs(abs(l[i][j] - a) + abs(l[n - i - 1][j] - a) + abs(l[i][m - j - 1] - a) + abs(l[n - i - 1][m - j - 1] - a))
	if n & 1:
		for i in range(m2):
			s += abs(l[n1][i] - l[n1][m - i - 1])
			l[n1][m - i - 1] = l[n1][i]
	if m & 1:
		for i in range(n1):
			s += abs(l[i][m2] - l[n - i - 1][m2])
			l[i][m2] = l[n - i - 1][m2]
	print(s)
