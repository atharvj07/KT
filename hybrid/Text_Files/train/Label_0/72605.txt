s = list(input())

def is_pretty(s):
	d = {chr(ord('A') + i): 0 for i in range(26)}
	q = 0
	qs = []
	f = 26
	for l in range(len(s)):
		if s[l] == '?':
			q += 1
			qs.append(l)
		elif d[s[l]] == 0:
			f -= 1
			d[s[l]] += 1
		else:
			return ''
	if f == 0:
		return ''.join(s)
	if f - q > 0:
		return ''
	qx = 0
	for l in d.keys():
		if d[l] == 0:
			s[qs[qx]] = l
			qx += 1
	return ''.join(s)
if len(s) < 26:
	print(-1)
elif len(s) == 26:
	if is_pretty(s) != '':
		print(''.join(is_pretty(s)))
	else:
		print(-1)
else:
	ok = False
	for i in range(26, len(s) + 1):
		pretty = is_pretty(s[i - 26:i])
		if len(pretty) != 0:
			s = s[:i - 26] + list(pretty) + s[i:]
			for sl in range(len(s)):
				if s[sl] == '?':
					s[sl] = 'X'
			print(''.join(s))
			ok = True
			break
	if not ok:
		print(-1)
