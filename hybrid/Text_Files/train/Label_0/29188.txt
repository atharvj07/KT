#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  string s;
  long long ans = 0;
  scanf("%d %d", &n, &k);
  long long a[n];
  for (int i = 0; i < (int)n; i++) scanf("%lld", &a[i]);
  cin >> s;
  int j = 0;
  for (int i = 0; i < n; i = j) {
    vector<long long> store;
    while (j < n && s[i] == s[j]) {
      store.push_back(a[j]);
      j++;
    }
    i = j;
    sort(store.begin(), store.end());
    for (int x = store.size() - 1, y = 0; x >= 0 && y < k; x--, y++)
      ans += store[x];
  }
  printf("%lld\n", ans);
}
