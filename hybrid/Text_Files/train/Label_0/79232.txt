import sys

def is_ok(a1, b1, a2, b2, a3, b3, step):
	if step == 1:
		return is_ok(a1, b1, a2, b2, a3, b3, step + 1) or is_ok(a1, b1, a2, b2, b3, a3, step + 1)
	elif step == 2:
		return is_ok(a1, b1, a2, b2, a3, b3, step + 1) or is_ok(a1, b1, b2, a2, a3, b3, step + 1)
	if a1 >= a2 + a3 and b1 >= max(b2, b3):
		return True
	if b1 >= b2 + b3 and a1 >= max(a2, a3):
		return True
	return False
(a1, b1) = map(int, input().split())
(a2, b2) = map(int, input().split())
(a3, b3) = map(int, input().split())
print('YES' if is_ok(a1, b1, a2, b2, a3, b3, 1) else 'NO')
