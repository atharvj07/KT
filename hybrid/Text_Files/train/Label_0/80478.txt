(x, y) = map(int, input().split())
grid = []
for i in range(x):
	line = list(input())
	grid.append(line)
C = 0
for i in range(x):
	if 'S' not in grid[i]:
		C += y
		grid[i] = [1] * y
for i in range(y):
	temp = x
	for j in range(x):
		if grid[j][i] == 'S':
			break
		elif grid[j][i] == 1:
			temp -= 1
	else:
		C += temp
print(C)
