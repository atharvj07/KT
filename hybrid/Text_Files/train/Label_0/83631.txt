t = int(input())
for _ in range(t):
	n = int(input())
	a = [int(x) for x in input().split()]
	adj = []
	for i in range(n):
		if a[i] == -1:
			if i - 1 >= 0 and a[i - 1] != -1:
				adj.append(a[i - 1])
			if i + 1 < n and a[i + 1] != -1:
				adj.append(a[i + 1])
	if len(adj) == 0:
		k = 1
		m = 0
	else:
		small = min(adj)
		large = max(adj)
		k = (small + large) // 2
		m = max(large - k, k - small)
	for i in range(1, n):
		if a[i] != -1 and a[i - 1] != -1:
			m = max(m, abs(a[i] - a[i - 1]))
	print('{} {}'.format(m, k))
