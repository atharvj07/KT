#include <bits/stdc++.h>
using namespace std;
int n, m;
char c[16][16];
int a[16][16], ans[16][16];
vector<pair<int, int> > v;
int vis[16][16];
int dir[4][2] = {{0, 1}, {1, 0}, {-1, 0}, {0, -1}};
bool cmp(pair<int, int> ai, pair<int, int> bi) {
  return a[ai.first][ai.second] < a[bi.first][bi.second];
}
vector<pair<int, int> > becoming(vector<pair<int, int> > now, int x, int y) {
  vector<pair<int, int> > does;
  does.push_back(pair<int, int>(x, y));
  for (int i = 0; i < now.size() - 1; i++) does.push_back(now[i]);
  return does;
}
int main() {
  cin >> n >> m;
  int len = 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      cin >> c[i][j];
      if (isdigit(c[i][j])) {
        a[i][j] = c[i][j] - '0';
        v.push_back(pair<int, int>(i, j));
        len = max(len, a[i][j]);
      }
      if (c[i][j] == '#') a[i][j] = -1;
      if (c[i][j] == '@') a[i][j] = 10;
    }
  sort(v.begin(), v.end(), cmp);
  queue<vector<pair<int, int> > > q;
  q.push(v);
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) ans[i][j] = 0;
  while (!q.empty()) {
    vector<pair<int, int> > now = q.front();
    q.pop();
    int x = now[0].first, y = now[0].second;
    if (a[x][y] == 10) {
      cout << ans[x][y];
      return 0;
    }
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        if (a[i][j] > 0 && a[i][j] <= 9) a[i][j] = 0;
    for (int i = 0; i < now.size(); i++) a[now[i].first][now[i].second] = i + 1;
    for (int i = 0; i < 4; i++) {
      int ux = x + dir[i][0], vy = y + dir[i][1];
      if (ux < 0 || vy < 0 || ux >= n || vy >= m) continue;
      if (a[ux][vy] == -1 || (a[ux][vy] > 0 && a[ux][vy] < len)) continue;
      if (vis[ux][vy] >= 250) continue;
      q.push(becoming(now, ux, vy));
      vis[ux][vy]++;
      ans[ux][vy] = ans[x][y] + 1;
    }
  }
  cout << "-1";
  return 0;
}
