import java.io.*;
import java.util.*;

public class P398B
{
    public static StringTokenizer st;
    public static void nextLine(BufferedReader br) throws IOException
    {
        st = new StringTokenizer(br.readLine());
    }
    
    public static String next()
    {
        return st.nextToken();
    }
    
    public static int nextInt()
    {
        return Integer.parseInt(st.nextToken());
    }
    
    public static long nextLong()
    {
        return Long.parseLong(st.nextToken());
    }
    
    public static double nextDouble()
    {
        return Double.parseDouble(st.nextToken());
    }
    
    static double[][] dp;
    static boolean[][] done;
    
    public static void main(String[] args) throws IOException
    {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        nextLine(br);
        int n = nextInt();
        int m = nextInt();
        int[] rc = new int[n];
        int[] cc = new int[n];
        int[] r = new int[m];
        int[] c = new int[m];
        for (int i = 0; i < m; i++)
        {
            nextLine(br);
            r[i] = nextInt() - 1;
            c[i] = nextInt() - 1;
            rc[r[i]]++;
            cc[c[i]]++;
        }
        int rowp = 0, colp = 0;
        for (int i = 0; i < n; i++)
        {
            if (rc[i] > 0) rowp++;
            if (cc[i] > 0) colp++;
        }
        dp = new double[n+1][n+1];
        done = new boolean[n+1][n+1];
        for (int i = 0; i < n+1; i++)
        {
            Arrays.fill(done[i], false);
        }
        double ans = solve(rowp, colp, n);
        System.out.println(ans);
    }
    
    private static double solve(int r, int c, int n)
    {
        if (done[r][c])
        {
            return dp[r][c];
        }
        if (r == n && c == n)
        {
            dp[r][c] = 0;
            done[r][c] = true;
            return 0;
        }
        if (r == n)
        {
            dp[r][c] = solve(r, c+1, n) + (double)n/(double)(n-c);
            done[r][c] = true;
            return dp[r][c];
        }
        if (c == n)
        {
            dp[r][c] = solve(r+1, c, n) + (double)n/(double)(n-r);
            done[r][c] = true;
            return dp[r][c];
        }
        double n2 = n * n;
        double val = 1 + (r * (n-c) * solve(r, c+1, n)/ n2) +
                         ((n-r) * c * solve(r+1, c, n)/ n2) +
                         ((n-r)*(n-c)*solve(r+1,c+1,n)/ n2);
        dp[r][c] = n2 * val / (n2 - r*c);
        done[r][c] = true;
        return dp[r][c];
    }
}
