#include<iostream>
#include<map>

using namespace std;
typedef pair<int,pair<int,pair<int,int> > > PP;

void solve()
{
	int n;
	while(cin >> n, n)
	{
		int id, Univ, AC, WA;
		
		map<PP,int> m;
		for(int i = 0; i < n; ++i)
		{
			cin >> id >> Univ >> AC >> WA;
			m[make_pair(-AC, make_pair(WA, make_pair(id, Univ)))] = id;
		}
		int count = 0, u[1001] = {};
		for(map<PP,int>::iterator it = m.begin(); it != m.end(); ++it)
		{
			int j = it->first.second.second.second;
			bool flag = false;
			if(count < 10 && u[j] < 3 || count < 20 && u[j] < 2 || count < 26 && !u[j])
			{
				flag = true;
			}
			if(flag)
			{
				++count, ++u[j];
				cout << it->second << endl;
			}
		}
	}
}

int main()
{
	solve();
	return(0);
	
}
