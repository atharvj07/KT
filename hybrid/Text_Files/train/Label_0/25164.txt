#include <bits/stdc++.h>
using namespace std;
const int N = (int)2e5 + 9;
const int LOG = 18;
vector<int> T[N];
int p[N];
int par[LOG][N];
int tin[N];
int tout[N];
int dist[N];
int tim = 0;
void dfs(int u, int pr) {
  tin[u] = ++tim;
  par[0][u] = pr;
  for (int i = 1; i < LOG; i++) {
    par[i][u] = par[i - 1][par[i - 1][u]];
  }
  for (auto x : T[u]) {
    dist[x] = dist[u] + 1;
    dfs(x, u);
  }
  tout[u] = tim;
}
bool is_anc(int a, int b) { return tin[a] <= tin[b] && tout[a] >= tout[b]; }
int lca(int a, int b) {
  if (is_anc(a, b)) return a;
  if (is_anc(b, a)) return b;
  for (int i = LOG - 1; i >= 0; i--) {
    if (!is_anc(par[i][a], b)) a = par[i][a];
  }
  return par[0][a];
}
pair<int, int> unite(pair<int, int> a, pair<int, int> b) {
  if (a.first == -2) return b;
  if (b.first == -2) return a;
  if (a.first == -1 || a.second == -1 || b.first == -1 || b.second == -1)
    return make_pair(-1, -1);
  vector<int> lis = {a.first, a.second, b.first, b.second};
  int up = lis[0];
  for (int i = 1; i < 4; i++) {
    if (dist[lis[i]] < dist[up]) {
      up = lis[i];
    }
  }
  for (int i = 0; i < 4; i++) {
    for (int j = 0; j < 4; j++) {
      if (lis[i] == -1 || lis[j] == -1 || i == j) continue;
      if (is_anc(lis[i], lis[j])) {
        lis[i] = -1;
        break;
      }
    }
  }
  int c = 0;
  for (int i = 0; i < 4; i++) {
    if (lis[i] != -1) {
      lis[c++] = lis[i];
    }
  }
  if (c == 1)
    return make_pair(up, lis[0]);
  else if (c == 2 && dist[lca(lis[0], lis[1])] <= dist[up])
    return make_pair(lis[0], lis[1]);
  else
    return make_pair(-1, -1);
}
pair<int, int> Tree[N * 4 + 512];
void setid(int node, int cl, int cr, int p, pair<int, int> vl) {
  if (cl == cr) {
    Tree[node] = vl;
    return;
  }
  int mid = (cl + cr) / 2;
  if (mid >= p)
    setid(node * 2, cl, mid, p, vl);
  else
    setid(node * 2 + 1, mid + 1, cr, p, vl);
  Tree[node] = unite(Tree[node * 2], Tree[node * 2 + 1]);
}
int query(int node, int cl, int cr, pair<int, int> cur) {
  if (unite(cur, Tree[node]) != make_pair(-1, -1)) return cr;
  if (cl == cr) return cl - 1;
  pair<int, int> un = unite(cur, Tree[node * 2]);
  int mid = (cl + cr) / 2;
  if (un == make_pair(-1, -1)) {
    return query(node * 2, cl, mid, cur);
  } else {
    return query(node * 2 + 1, mid + 1, cr, un);
  }
}
void print(int node, int cl, int cr) {
  cout << cl << " " << cr << "| " << Tree[node].first << " "
       << Tree[node].second << "\n";
  if (cl == cr) {
    return;
  }
  int mid = (cl + cr) / 2;
  print(node * 2, cl, mid);
  print(node * 2 + 1, mid + 1, cr);
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  int n;
  cin >> n;
  int a;
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
  }
  for (int i = 2; i <= n; i++) {
    cin >> a;
    T[a].push_back(i);
  }
  dfs(1, 1);
  for (int i = 0; i < n * 4 + 512; i++) {
    Tree[i] = make_pair(-2, -2);
  }
  for (int i = 1; i <= n; i++) {
    setid(1, 0, n - 1, p[i], make_pair(i, i));
  }
  int q;
  cin >> q;
  int typ;
  int pi, pj;
  for (int i = 0; i < q; i++) {
    cin >> typ;
    if (typ == 1) {
      cin >> pi >> pj;
      setid(1, 0, n - 1, p[pj], make_pair(pi, pi));
      setid(1, 0, n - 1, p[pi], make_pair(pj, pj));
      swap(p[pi], p[pj]);
    } else {
      cout << query(1, 0, n - 1, make_pair(-2, -2)) + 1 << "\n";
    }
  }
  return 0;
}
