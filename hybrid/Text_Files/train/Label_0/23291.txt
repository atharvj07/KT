#include <bits/stdc++.h>
using namespace std;
vector<vector<long long>> g;
vector<long long> colors;
long long cur_clr = 1;
vector<long long> used;
long long num = 0;
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  long long n;
  cin >> n;
  g.resize(n);
  colors.resize(n);
  used.resize(n);
  for (long long i = 0; i < n - 1; i++) {
    long long u, v;
    cin >> u >> v;
    u--;
    v--;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  long long col = 0;
  for (long long i = 0; i < n; i++) {
    if (g[i].size() > 2) {
      col++;
      num = i;
    }
  }
  if (col > 1) {
    cout << "No" << endl;
    return 0;
  } else {
    cout << "Yes" << endl;
  }
  vector<long long> ans;
  for (long long i = 0; i < n; i++) {
    if (num != i && g[i].size() == 1) {
      ans.push_back(i);
    }
  }
  cout << ans.size() << endl;
  for (auto t : ans) {
    cout << num + 1 << " " << t + 1 << endl;
  }
}
