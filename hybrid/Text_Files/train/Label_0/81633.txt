def solve(n, a):
	inc = [a[0]]
	dec = []
	for i in range(1, n):
		if a[i] == a[i - 1]:
			dec.append(a[i])
			if len(dec) >= 2 and dec[-1] == dec[-2]:
				return 'NO'
		else:
			inc.append(a[i])
			if len(inc) >= 2 and inc[-1] == inc[-2]:
				return 'NO'
	return (inc, dec)
n = int(input())
a = sorted(list(map(int, input().split())))
p = solve(n, a)
if p == 'NO':
	print(p)
else:
	print('YES')
	print(len(p[0]))
	print(*p[0])
	print(len(p[1]))
	print(*p[1][::-1])
