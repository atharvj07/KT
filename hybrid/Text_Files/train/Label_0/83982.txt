(y, b, r) = input().split()
y = int(y)
b = int(b)
r = int(r)
m = min(y, b, r)
if m == y:
	if m + 1 <= b and m + 2 <= r:
		print(3 * m + 3)
	else:
		while m + 1 > b or m + 2 > r:
			m -= 1
		print(3 * m + 3)
elif m == b:
	if m - 1 <= y and m + 1 <= r:
		print(3 * m)
	else:
		while m - 1 > y or m + 1 > r:
			m -= 1
		print(3 * m)
elif m - 1 <= b and m - 2 <= y:
	print(3 * m - 3)
else:
	while m - 1 > b or m - 2 > y:
		m -= 1
	print(3 * m - 3)
