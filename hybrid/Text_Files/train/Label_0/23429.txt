#include <bits/stdc++.h>
using namespace std;
int main() {
  double a[10005], k, max1, min1, sum, x;
  int n, i;
  cin >> n >> k;
  for (i = 0; i < n; i++) cin >> a[i];
  min1 = 0;
  max1 = 1000;
  while (max1 - min1 > 0.00000001) {
    sum = 0;
    x = (max1 + min1) / 2;
    for (i = 0; i < n; i++)
      if (a[i] > x)
        sum += (a[i] - x) * (100 - k) / 100;
      else
        sum += (a[i] - x);
    if (sum > 0)
      min1 = x;
    else
      max1 = x;
  }
  printf("%0.9f", x);
  return 0;
}
