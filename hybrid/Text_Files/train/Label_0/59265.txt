import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class PaintTree197E {

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		Vertex[] vertices = new Vertex[n];
		for (int i = 0; i < n; i++) {
			vertices[i] = new Vertex();
			vertices[i].id = i;
		}
		for (int i = 0; i < n - 1; i++) {
			int from = in.nextInt();
			int to = in.nextInt();
			Vertex v = vertices[from - 1];
			Vertex t = vertices[to - 1];
			v.children.add(t);
			t.children.add(v);
		}
		Point[] points = new Point[n];
		long min_x, min_y;
		min_x = min_y = Long.MAX_VALUE;
		for (int i = 0; i < n; i++) {
			long x = in.nextInt();
			long y = in.nextInt();
			Point p = new Point(i, x, y);
			points[i] = p;
			if (y < min_y) {
				min_y = y;
				min_x = x;
			} else if (y == min_y && x < min_x) {
				min_x = x;
			}
		}
		boolean[] visited = new boolean[n];
		int[] sizes = new int[n];
		countSize(0, vertices, visited, sizes);
		int[] pos = new int[n];
		Arrays.fill(visited, false);
		countPos(vertices[0], 0, pos, points, sizes, visited);
		for (int i = 0; i < n; i++)
			System.out.print(pos[i] + 1 + (i == n - 1 ? "\n" : " "));
	}

	private static void countPos(Vertex v, int i, int[] pos, Point[] points,
			int[] sizes, boolean[] visited) {
		visited[v.id] = true;
		Point bp = basePoint(points, i, i + sizes[v.id]);
		Arrays.sort(points, i, i + sizes[v.id], new PointSorter(bp));
		pos[points[i].id] = v.id;

		int next_i = i + 1;
		for (Vertex child : v.children) {
			if (visited[child.id])
				continue;
			countPos(child, next_i, pos, points, sizes, visited);
			next_i = next_i + sizes[child.id];
		}
	}

	private static Point basePoint(Point[] points, int i, int j) {
		long min_x, min_y;
		Point bp = points[i];
		min_x = bp.x;
		min_y = bp.y;
		for (int index = i; index < j; index++) {
			Point p = points[index];
			if (p.y < min_y) {
				min_x = p.x;
				min_y = p.y;
				bp = p;
			} else if (p.y == min_y && p.x < min_x) {
				min_x = p.x;
				bp = p;
			}
		}
		return bp;
	}

	private static void countSize(int i, Vertex[] vertices, boolean[] visited,
			int[] sizes) {
		if (visited[i])
			return;
		visited[i] = true;
		int size = 1;
		Vertex v = vertices[i];
		for (Vertex child : v.children) {
			countSize(child.id, vertices, visited, sizes);
			size += sizes[child.id];
		}
		sizes[v.id] = size;
	}

	static class Point {
		long x, y;
		int id;

		Point(int _id, long _x, long _y) {
			id = _id;
			x = _x;
			y = _y;
		}

		public boolean equals(Object o) {
			if (o instanceof Point) {
				Point other = (Point) o;
				return other.x == this.x && other.y == this.y;
			}
			return false;
		}

		public String toString() {
			return String.format("%d,%d", x, y);
		}
	}

	static class PointSorter implements Comparator<Point> {
		private Point basePoint;

		PointSorter(Point _basePoint) {
			basePoint = _basePoint;
		}

		@Override
		public int compare(Point p1, Point p2) {
			if (p1.equals(basePoint))
				return -1;
			if (p2.equals(basePoint))
				return 1;
			double v1, v2;
			v1 = arccotangent(p1);
			v2 = arccotangent(p2);
			if (v1 == v2)
				return 0;
			else
				return v1 > v2 ? 1 : -1;
		}

		private double arccotangent(Point p) {
			if (p.y == basePoint.y)
				return p.x > basePoint.x ? Double.MAX_VALUE : Double.MIN_VALUE;
			else
				return 1.0 * (p.x - basePoint.x) / (p.y - basePoint.y);
		}
	}

	static class Vertex {
		int id;
		ArrayList<Vertex> children = new ArrayList<Vertex>();

		public String toString() {
			return String.valueOf(id);
		}
	}
}
