#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e5 + 10;
const int MOD = 1e9 + 7;
int N, M, K, Q;
vector<pair<int, int>> vec[110];
int main() {
  int i, j, k, a, b;
  scanf("%d%d", &N, &M);
  int sx = 101;
  for (i = 0; i < M; ++i) {
    scanf("%d%d", &a, &b);
    int sy = (a - 1) * 100 + 1;
    sy += vec[a].size();
    vec[a].push_back({a, sy});
    vec[b].push_back({b, sy});
  }
  for (i = 1; i <= N; ++i) {
    if (vec[i].size() != 0) {
      printf("%d\n", vec[i].size());
      for (j = 0; j < vec[i].size(); ++j) {
        printf("%d %d\n", vec[i][j].first, vec[i][j].second);
      }
    } else {
      printf("1\n");
      printf("%d %d\n", i, (i - 1) * 100 + 1);
    }
  }
}
