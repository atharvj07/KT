#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
long long powmod(long long a, long long b) {
  long long res = 1;
  a %= mod;
  assert(b >= 0);
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
  }
  return res;
}
const int N = 101000;
int n, m, p[N], px[N], py[N], q[N];
bool adj(int u, int v) {
  if (u > v) swap(u, v);
  if (px[u] == px[v] && py[u] + 1 == py[v]) return 1;
  if (py[u] == py[v] && px[u] + 1 == px[v]) return 1;
  return 0;
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n * m; i++) px[i] = i / m, py[i] = i % m;
  while (clock() <= 1.9 * CLOCKS_PER_SEC) {
    for (int i = 0; i < n * m; i++) p[i] = i;
    random_shuffle(p, p + n * m);
    set<pair<int, int> > hs;
    for (int i = 0; i < n * m; i++) hs.insert(make_pair(p[i], i));
    bool val = 1;
    for (int i = 0; i < n * m; i++) {
      int x = i / m, y = i % m;
      auto it = hs.begin();
      bool suc = 0;
      while (1) {
        suc = 1;
        q[i] = it->second;
        if (y && adj(q[i], q[i - 1])) {
          suc = 0;
        }
        if (x && adj(q[i], q[i - m])) {
          suc = 0;
        }
        if (suc) {
          hs.erase(it);
          break;
        }
        it++;
        if (it == hs.end()) break;
      }
      if (suc == 0) {
        val = 0;
        break;
      }
    }
    if (val) {
      puts("YES");
      for (int i = 0; i < n * m; i++) {
        printf("%d ", q[i] + 1);
        if (i % m == m - 1) puts("");
      }
      return 0;
    }
  }
  puts("NO");
}
