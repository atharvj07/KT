t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	s = input()
	L = [-1]
	for i in range(n):
		if s[i] == '1':
			L.append(i)
	L.append(n)
	cnt = 0
	m = len(L)
	S = [L[i + 1] - L[i] - 1 for i in range(m - 1)]
	m = len(S)
	for i in range(1, m - 1):
		cnt += (S[i] - k) // (k + 1)
	if m == 1:
		cnt += S[0] // (k + 1) + (S[0] % (k + 1) != 0)
	else:
		cnt += (S[0] - k) // (k + 1) + ((S[0] - k) % (k + 1) != 0)
	if m >= 2:
		cnt += (S[-1] - k) // (k + 1) + ((S[-1] - k) % (k + 1) != 0)
	print(cnt)
