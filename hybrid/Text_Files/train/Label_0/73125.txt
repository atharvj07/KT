def swap1(arr1):
	for i in range(0, len(arr1) - 1, 2):
		(arr1[i], arr1[i + 1]) = (arr1[i + 1], arr1[i])
	return arr1

def swap2(arr):
	for i in range(0, len(arr) // 2):
		(arr[i], arr[i + len(arr) // 2]) = (arr[i + len(arr) // 2], arr[i])
	return arr
n = int(input())
per = list(map(int, input().split()))
per1 = [] + per
mas = list(range(1, 2 * n + 1, 1))
i = 0
while per != mas and per1 != mas:
	if i > 2 * n:
		i = -1
		break
	if i % 2 == 0:
		per = swap1(per)
		per1 = swap2(per1)
	else:
		per = swap2(per)
		per1 = swap1(per1)
	i += 1
print(i)
