n = int(input())
l = []
x = []
y = []
c1 = c2 = 0
for i in range(0, n):
	(p, q) = map(int, input().split())
	temp = []
	temp.append(p)
	temp.append(q)
	x.append(p)
	y.append(q)
	l.append(temp)
x.sort()
y.sort()
for i in range(len(x) - 1):
	if x[i] == x[i + 1] and i == len(x) - 2:
		c1 += 1
	elif x[i] == x[i + 1]:
		continue
	elif i == len(x) - 2 and x[i] != x[i + 1]:
		c1 += 2
	else:
		c1 += 1
for i in range(len(y) - 1):
	if y[i] == y[i + 1] and i == len(y) - 2:
		c2 += 1
	elif y[i] == y[i + 1]:
		continue
	elif i == len(y) - 2 and y[i] != y[i + 1]:
		c2 += 2
	else:
		c2 += 1
if len(x) == 1:
	c1 = c2 = 1
print(min(c1, c2))
