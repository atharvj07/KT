import math

def solve(n, k, l1, r1, l2, r2):
	if l1 > l2:
		(l1, l2) = (l2, l1)
		(r1, r2) = (r2, r1)
	k -= n * max(min(r1, r2) - l2, 0)
	if k <= 0:
		return 0
	le_dif = l2 - l1
	ri_dif = max(r1, r2) - min(r1, r2)
	to_join = 0 if r1 >= l2 else l2 - r1
	eq_len_intersection = le_dif + ri_dif - to_join
	if eq_len_intersection >= k:
		return to_join + k
	elif eq_len_intersection + to_join < 2 * eq_len_intersection:
		req = math.ceil(k / eq_len_intersection)
		if req <= n:
			lo_lim = k // eq_len_intersection
			res = (eq_len_intersection + to_join) * lo_lim
			rem = k - eq_len_intersection * lo_lim
			if rem > 0:
				res += min(to_join + rem, 2 * rem)
			return res
		else:
			return (eq_len_intersection + to_join) * n + 2 * (k - eq_len_intersection * n)
	else:
		return to_join + eq_len_intersection + 2 * (k - eq_len_intersection)
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	(l1, r1) = map(int, input().split())
	(l2, r2) = map(int, input().split())
	print(solve(n, k, l1, r1, l2, r2))
