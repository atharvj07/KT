(x, y) = ([], [])
for i in range(3):
	(x_t, y_t) = map(int, input().split())
	x.append(x_t)
	y.append(y_t)
r = set()
r.add((x[2] + x[0] - x[1], y[2] + y[0] - y[1]))
r.add((x[2] + x[1] - x[0], y[2] + y[1] - y[0]))
r.add((x[0] + x[1] - x[2], y[0] + y[1] - y[2]))
r.add((x[0] + x[2] - x[1], y[0] + y[2] - y[1]))
r.add((x[1] + x[2] - x[0], y[1] + y[2] - y[0]))
r.add((x[1] + x[0] - x[2], y[1] + y[0] - y[2]))
print(len(r))
for i in r:
	print(i[0], i[1])
