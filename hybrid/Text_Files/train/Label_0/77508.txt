(n, k) = map(int, input().split())
s = input()
st = ''
for i in s:
	if i <= 'm':
		t = ord('z') - ord(i)
		if k >= t:
			k -= t
			st += 'z'
		elif k < t and k != 0:
			st += chr(ord(i) + k)
			k = 0
		elif k == 0:
			st += i
	else:
		t = abs(ord(i) - ord('a'))
		if k >= t:
			k -= t
			st += 'a'
		elif k < t and k != 0:
			st += chr(ord(i) - k)
			k = 0
		elif k == 0:
			st += i
if k != 0:
	print('-1')
else:
	print(st)
