#include <bits/stdc++.h>
using namespace std;
int mat[87][87];
int n, k;
int dp[83][83][83];
int rc(int tk, int u, int v) {
  int rt = 10000000, i;
  if (tk == k) return 0;
  if (abs(v - u) < k - tk) return rt;
  if (dp[tk][u][v] != -1) return dp[tk][u][v];
  if (u < v) {
    for (i = u + 1; i <= v; i++) {
      if (!mat[u][i]) continue;
      rt = min(mat[u][i] + rc(tk + 1, i, v), rt);
      rt = min(mat[u][i] + rc(tk + 1, i, u + 1), rt);
    }
  } else {
    for (i = u - 1; i >= v; i--) {
      if (!mat[u][i]) continue;
      rt = min(mat[u][i] + rc(tk + 1, i, v), rt);
      rt = min(mat[u][i] + rc(tk + 1, i, u - 1), rt);
    }
  }
  return dp[tk][u][v] = rt;
}
int main(void) {
  int i, m, u, v, c, ans = 10000000;
  std::ios::sync_with_stdio(false);
  cin >> n >> k >> m;
  while (m--) {
    cin >> u >> v >> c;
    if (!mat[u][v]) mat[u][v] = c;
    mat[u][v] = min(mat[u][v], c);
  }
  memset(dp, -1, sizeof dp);
  for (i = 1; i <= n; i++) {
    ans = min(ans, rc(1, i, 1));
    ans = min(ans, rc(1, i, n));
  }
  if (ans - 10000000)
    cout << ans << endl;
  else
    cout << -1 << endl;
  return 0;
}
