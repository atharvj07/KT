list_ans = []
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	list1 = list(map(int, input().split()))
	dict1 = {}
	for x in list1:
		s = ((x - 1) // k + 1) * k - x
		if s != 0:
			if s not in dict1:
				dict1[s] = 1
			else:
				dict1[s] += 1
	a = 0
	b = 0
	for x in sorted(list(dict1.keys())):
		if dict1[x] >= a:
			a = dict1[x]
			b = x
	list_ans.append(max((a - 1) * k + b + 1, 0))
print(*list_ans, sep='\n')
