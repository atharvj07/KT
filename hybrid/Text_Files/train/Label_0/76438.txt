import sys
input = sys.stdin.readline
n = int(input())
A = [0] + list(map(int, input().split()))
q = int(input())
SUMA = sum(A)
SUM = 0
B = [0] * (n + 1)
DICT = dict()
for i in range(q):
	(x, y, z) = map(int, input().split())
	if DICT.get((x, y)) == None:
		DICT[x, y] = z
		B[z] += 1
		if B[z] <= A[z]:
			SUM += 1
	else:
		bef = DICT[x, y]
		DICT[x, y] = z
		if bef != z:
			B[bef] -= 1
			B[z] += 1
			if B[bef] < A[bef]:
				SUM -= 1
			if B[z] <= A[z]:
				SUM += 1
	print(SUMA - SUM)
