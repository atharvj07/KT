#include <bits/stdc++.h>
#pragma GCC optimize(3)
using namespace std;
int n, m, p;
bool vis[110][110][6];
bool has[110][110];
int dy[] = {0, 0, -1, 1};
int dx[] = {-1, 1, 0, 0};
int cnt;
void dfs(int x, int y) {
  has[x][y] = 1;
  cnt++;
  for (int i = 0; i < 4; i++) {
    int xx = x + dx[i], yy = y + dy[i];
    if (vis[x][y][i] && xx >= 0 && xx < n && yy >= 0 && yy < m &&
        has[xx][yy] == 0) {
      dfs(xx, yy);
    }
  }
}
int main() {
  ios::sync_with_stdio(0);
  cin >> m >> n >> p;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      for (int k = 0; k < 4; k++) {
        vis[i][j][k] = 1;
      }
  for (int i = 0; i < n; i++) {
    vis[i][0][2] = 0;
    vis[i][m - 1][3] = 0;
  }
  for (int j = 0; j < m; j++) {
    vis[0][j][0] = 0;
    vis[n - 1][j][1] = 0;
  }
  for (int i = 0; i < p; i++) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a == c) {
      for (int j = b; j < d; j++) {
        vis[j][a - 1][3] = 0;
        vis[j][a][2] = 0;
      }
    } else {
      for (int j = a; j < c; j++) {
        vis[b - 1][j][1] = 0;
        vis[b][j][0] = 0;
      }
    }
  }
  vector<int> v;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (!has[i][j]) {
        cnt = 0;
        dfs(i, j);
        v.push_back(cnt);
      }
    }
  }
  sort(v.begin(), v.end());
  for (int i = 0; i < v.size(); i++) {
    cout << v[i] << " ";
  }
  return 0;
}
