#include <bits/stdc++.h>
using namespace std;
int mark[100000 + 5], ara[100000 + 5];
int main() {
  int i, j, k, n, l, gcd, mx = 0;
  cin >> n;
  for (i = 1; i <= n; i++) {
    scanf("%d", &l);
    ara[l]++;
  }
  if (ara[1] > 0) mx = 1;
  for (i = 2; i <= 100000; i++) {
    if (!mark[i]) {
      gcd = ara[i];
      for (j = i + i; j <= 100000; j += i) {
        gcd += ara[j];
      }
      mx = max(gcd, mx);
    }
  }
  cout << mx << endl;
  return 0;
}
