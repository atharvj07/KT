def solve():
	n = int(input())
	arr = sorted(list(map(int, input().split())))
	d = dict()
	temp = 0
	for x in arr:
		d[x] = d[x] + 1 if x in d else 1
		temp = max(temp, d[x])
	if temp >= n // 2:
		print(-1)
		return
	ans = 0
	prev = None
	for i in range(n // 2 + 1):
		if arr[i] == prev:
			continue
		prev = arr[i]
		diff = [abs(arr[i] - arr[j]) for j in range(i + 1, n)]
		factors = dict()
		factors[1] = 0
		add = 0
		for x in diff:
			if x == 0:
				add += 1
				continue
			y = 1
			while y * y <= x:
				if x % y == 0:
					if y in factors:
						factors[y] += 1
					else:
						factors[y] = 1
					if y != x // y:
						if x // y in factors:
							factors[x // y] += 1
						else:
							factors[x // y] = 1
				y += 1
		for x in factors:
			if factors[x] + add + 1 >= n // 2:
				ans = max(ans, x)
	print(ans)
for _ in range(int(input())):
	solve()
