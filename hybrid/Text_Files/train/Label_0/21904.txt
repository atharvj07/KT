#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int x = 0, f = 1;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') f = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9')
    x = (x << 3) + (x << 1) + (c ^ '0'), c = getchar();
  return x * f;
}
const int inf = 2e9 + 7;
const int N = 3e5 + 5;
int n, w, ans[N];
int a[N], b[N];
long long sum;
priority_queue<pair<int, int> > q1, q2, q3, q4, q5;
inline void inc0(int i) {
  ans[i] = 0, q1.push(make_pair(-a[i], i)), q3.push(make_pair(-b[i], i));
}
inline void inc1(int i) {
  ans[i] = 1, q2.push(make_pair(a[i] - b[i], i)), q4.push(make_pair(a[i], i));
}
inline void inc2(int i) { ans[i] = 2, q5.push(make_pair(b[i] - a[i], i)); }
int main() {
  n = read(), w = read();
  q1.push(make_pair(-inf, n + 1));
  q2.push(make_pair(-inf, n + 1));
  q3.push(make_pair(-inf, n + 1));
  q4.push(make_pair(-inf, n + 1));
  q5.push(make_pair(-inf, n + 1));
  for (int i = 1; i <= n; i++) {
    a[i] = read(), b[i] = read();
    inc0(i);
  }
  for (int k = 1; k <= w; k++) {
    int p1 = q1.top().second, p2 = q2.top().second, p3 = q3.top().second,
        p4 = q4.top().second, p5 = q5.top().second;
    while (q1.size() > 1 && ans[p1] != 0) q1.pop(), p1 = q1.top().second;
    while (q2.size() > 1 && ans[p2] != 1) q2.pop(), p2 = q2.top().second;
    while (q3.size() > 1 && ans[p3] != 0) q3.pop(), p3 = q3.top().second;
    while (q4.size() > 1 && ans[p4] != 1) q4.pop(), p4 = q4.top().second;
    while (q5.size() > 1 && ans[p5] != 2) q5.pop(), p5 = q5.top().second;
    long long v1 = q1.top().first, v2 = q2.top().first, v3 = q3.top().first,
              v4 = q4.top().first, v5 = q5.top().first;
    long long w1 = -v1, w2 = -v2, w3 = -v3 - v4, w4 = -v3 - v5;
    long long mn = min(min(w1, w2), min(w3, w4));
    sum += mn;
    if (w1 == mn)
      q1.pop(), inc1(p1);
    else if (w2 == mn)
      q2.pop(), inc2(p2);
    else if (w3 == mn)
      q3.pop(), q4.pop(), inc0(p4), inc2(p3);
    else if (w4 == mn)
      q3.pop(), q5.pop(), inc1(p5), inc2(p3);
  }
  printf("%lld\n", sum);
  for (int i = 1; i <= n; i++) printf("%d", ans[i]);
  return 0;
}
