typedef long long lli;

const lli MAXN = 500001;
const lli INF = 1LL<<60;

lli N, V;
Graph G;

int num[MAXN], depth[MAXN];
lli cost[MAXN], mem[MAXN];

int main() {
  while(cin >> N) {
    V = N - 1;
    G = Graph(V);
    for(lli i = 0; i < N - 1; ++i) {
      lli a, b; cin >> a >> b;
      --a; --b;
      G[i].push_back(a);
      G[i].push_back(b);
    }

    {
      queue<lli> que;
      que.push(0);
      depth[0] = 0;
      while(que.size()) {
        lli v = que.front();
        que.pop();
        for(lli i = 0; i < G[v].size(); ++i) {
          lli nv = G[v][i];
          if(nv == N-1) continue;
          que.push(nv);
          depth[nv] = depth[v] + 1;
        }
      }
    }

    vector<pair<lli, lli> > w;
    for(lli i = 0; i < V; ++i) {
      w.push_back(make_pair(-depth[i], i));
    }
    sort(w.begin(), w.end());

    for(lli i = 0; i < V; ++i) {
      lli v = w[i].second;
      if(G[v].empty()) {
        assert(false);
        cost[v] = depth[v];
        num[v] = 1;
      } else {
        lli a = G[v][0];
        lli b = G[v][1];
        cost[v] = num[v] = 0;
        if(a == N-1) {
          cost[v] += depth[v] + 1;
          num[v] += 1;
        } else {
          cost[v] += cost[a];
          num[v] += num[a];
        }
        if(b == N-1) {
          cost[v] += depth[v] + 1;
          num[v] += 1;
        } else {
          cost[v] += cost[b];
          num[v] += num[b];
        }
      }
    }

    mem[N-1] = 0;
    for(lli i = 0; i < V; ++i) {
      lli v = w[i].second;
      lli &res = mem[v];
      res = INF;
      lli a = G[v][0];
      lli b = G[v][1];
      lli smallA = 0, smallB = 0;
      if(a == N-1) {
        smallA = depth[v] + 1 - 1 * depth[v];
      } else {
        smallA = cost[a] - num[a] * depth[v];
      }

      if(b == N-1) {
        smallB = depth[v] + 1 - 1 * depth[v];
      } else {
        smallB = cost[b] - num[b] * depth[v];
      }
      res = min(res, smallA + mem[b] + 1);
      res = min(res, mem[a] + smallB + 1);
      res = min(res, mem[a] + mem[b] + depth[v] + 2);
    }

    cout << mem[0] << endl;
  }
  return 0;
}
