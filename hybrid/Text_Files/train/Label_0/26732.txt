const int N = 5;
double dp[N][N][N][N][N][N][N][N];
double prob[8][8];

double solve(int n,int now,const vector<int> &ord,const vector<int> &ori,vector<int> &st){
  if (ord.size() == now){
    return 1;
  }
  double &ret=dp[st[0]][st[1]][st[2]][st[3]][st[4]][st[5]][st[6]][st[7]];
  //cout <<"ret " << ret << endl;
  if (ret > -0.5)return ret;
  ret=0;
  rep(i,n){
    if (st[i]+1 > ori[i])continue;
    st[i]++;
    ret+=prob[ord[now]][i]*solve(n,now+1,ord,ori,st);
    st[i]--;
  }
  return ret;
}
