long long n, k;
long long isi[1005];
map<long long, long long> cnt;
pair<long long, long long> ask(vector<long long>& all) {
  cout << "? ";
  for (int i = 0; i < all.size(); i++)
    cout << all[i] << " \n"[i == all.size() - 1];
  long long pos, tmp;
  cin >> pos >> tmp;
  isi[pos] = tmp;
  return {pos, tmp};
}
long long answer(long long ans) {
  cout << "! " << ans << endl;
  exit(0);
}
int main() {
  cin >> n >> k;
  n = k + 1;
  vector<long long> all;
  for (int i = 1; i <= n; i++) {
    vector<long long> q;
    for (int j = 1; j <= n; j++) {
      if (i == j) continue;
      q.push_back(j);
    }
    all.push_back(ask(q).second);
  }
  sort(all.begin(), all.end());
  for (int i = 0; i < all.size(); i++) {
    cnt[all[i]]++;
  }
  auto it = cnt.end();
  it--;
  answer((*it).second);
  return 0;
}
