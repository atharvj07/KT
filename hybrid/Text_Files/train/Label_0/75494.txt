def solve(xs):
	if len(xs) == 1:
		return 0
	heights = sorted(xs)[::-1]
	used = 0
	needed = heights[0]
	for i in range(len(heights)):
		current = heights[i]
		if needed <= current:
			used += 1
			needed -= 1
		else:
			used += needed - current
			used += 1
			needed = current - 1
	if needed > 0:
		used += needed
	return sum(xs) - used
(n, m) = map(int, input().split())
xs = list(map(int, input().split()))[:n]
print(solve(xs))
