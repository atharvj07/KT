import collections
s = input()
t = input()
letter_s = collections.Counter(s)
letter_t = collections.Counter(t)
yay = 0
whoops = 0
for i in s:
	if letter_t[i] > 0:
		letter_s[i] -= 1
		letter_t[i] -= 1
		yay += 1
for i in s:
	if letter_s[i] > 0:
		if i.islower() and letter_t[i.upper()] > 0:
			letter_s[i] -= 1
			letter_t[i.upper()] -= 1
			whoops += 1
		elif i.isupper() and letter_t[i.lower()] > 0:
			letter_s[i] -= 1
			letter_t[i.lower()] -= 1
			whoops += 1
print(yay, whoops)
