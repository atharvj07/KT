import java.io.*;
import java.util.*;

import static java.lang.Math.*;

public class problem {

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int t = scan.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scan.nextInt();
            int p = scan.nextInt();
            int k = scan.nextInt();
            int a[] = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scan.nextInt();
            }
            Arrays.sort(a);
            int pref = 0;
            int ans = 0;
            for (int j = 0; j <= k; j++) {
                int sum = pref;
                if (sum > p) break;;
                int cnt = j;
                for (int l = j + k - 1; l < n; l += k) {
                    if (sum + a[l] <= p) {
                        sum += a[l];
                        cnt += k;
                    } else {
                        break;
                    }
                }
                ans = max(ans, cnt);
                if (j < n) pref += a[j];
            }

            System.out.println(ans);
        }
    }
}
