from collections import defaultdict
from heapq import heappop, heappush, heapify
n = int(input())
a = list(map(int, input().split()))
t = list(map(int, input().split()))
d = defaultdict(list)
for i in range(n):
	d[a[i]].append(t[i])
heap2 = list(set(a))
heapify(heap2)
ans = 0
used = set()
heap = []
s = 0
while heap2:
	cur = heappop(heap2)
	if cur in used:
		continue
	used.add(cur)
	for ti in d[cur]:
		heappush(heap, -ti)
		s += ti
	ti = -heappop(heap)
	s -= ti
	ans += s
	if heap:
		heappush(heap2, cur + 1)
print(ans)
