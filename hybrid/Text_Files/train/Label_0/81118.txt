for i in range(int(input())):
	a = int(input())
	if a in [1, 2, 4]:
		print(-1)
	elif str(a)[-1] == '1':
		print(a // 10 + 1, 1, a // 10 - 1)
	elif str(a)[-1] == '2':
		print(a // 10 - 1, 1, a // 10)
	elif str(a)[-1] == '3':
		print(a // 10 + 1, 0, a // 10)
	elif str(a)[-1] == '4':
		print(a // 10 - 1, 0, a // 10 + 1)
	elif str(a)[-1] == '5':
		print(a // 10, 1, a // 10)
	elif str(a)[-1] == '6':
		print(a // 10 + 2, 0, a // 10)
	elif str(a)[-1] == '7':
		print(a // 10, 0, a // 10 + 1)
	elif str(a)[-1] == '8':
		print(a // 10 + 1, 1, a // 10)
	elif str(a)[-1] == '9':
		print(a // 10 + 3, 0, a // 10)
	else:
		print(a // 10, 0, a // 10)
