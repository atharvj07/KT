var inp = readline().split(' ').map(function(cur,i,ar){return parseInt(cur.trim());});
var n = inp[0], k = inp[1];
var a = {}, len = new Array(n);
inp = readline().split(' ').map(function(cur,i,ar){return parseInt(cur.trim());});
inp.sort(function(a,b){return a - b;});

for(var i = 0; i < n; i++) {
    a[inp[i]] = i;
    if(inp[i] % k == 0 && a[inp[i] / k] != undefined){
        var t = a[inp[i] / k];
        len[i] = len[t] + 1;
        len[t] = 0;
    } 
    else len[i] = 1; 
}
var ans = 0;
for(var i = 0; i < n; i++)
    ans += Math.floor((len[i] + 1) / 2);
print((k == 1)? n : ans);
