  cin.tie(nullptr), cout.tie(nullptr);
  long long l, r, x, y, k, temp;
  cin >> l >> r >> x >> y >> k;
  for (long long i = x; i <= y && i * k <= r; i++) {
    temp = i * k;
    if (temp >= l && temp <= r) {

    }
  }
  cout << "No\n";
