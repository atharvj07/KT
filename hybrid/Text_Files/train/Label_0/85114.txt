import sys
z = int(input())
for _ in range(z):
	ans = 99999999
	(n, m) = [int(i) for i in sys.stdin.readline().split()]
	df = []
	for _ in range(n):
		df.append(sys.stdin.readline())
	l = [df[i].count('.') for i in range(n)]
	df2 = list(map(list, zip(*df)))
	w = [df2[i].count('.') for i in range(m)]
	for i in range(n):
		ans1 = l[i]
		x = w.copy()
		for j in range(m):
			if df[i][j] == '.':
				x[j] -= 1
		ans2 = min(x)
		ans = min(ans, ans1 + ans2)
	print(ans)
