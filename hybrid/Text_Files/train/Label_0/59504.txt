
import java.util.Scanner;
import java.util.*;
public class Main {//前缀和
    public static void main(String[] args) {
        int slength, n;
        Scanner inp = new Scanner(System.in);
        slength = inp.nextInt();
        n = inp.nextInt();
        inp.nextLine();//nextline()
        String song = inp.nextLine();
        char[] arr = song.toCharArray();
        int[] sum = new int[100050];//C++只能用定值
        int[] out = new int[n];
        sum[0] = 0;
        for (int i = 1; i < slength+1; i++) {
            sum[i] = sum[i - 1] + arr[i-1] - 'a' + 1;

        }
        for (int i = 0; i < n; i++) {
            int l = inp.nextInt();
            int r = inp.nextInt();
            out[i] = sum[r] - sum[l - 1];

        }
        for (int i = 0; i < n; i++) {
            System.out.println(out[i]);


        }


    }
}
