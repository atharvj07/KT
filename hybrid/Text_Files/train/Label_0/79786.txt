t = int(input())
for i in range(t):
	p = 0
	r = 0
	c = 0
	(n, k) = map(int, input().split())
	s = input()
	for i in range(n):
		if s[i] == '1':
			if p:
				r = r + max(0, (c + 1) // (k + 1) - 1)
			else:
				r = r + c // (k + 1)
			p = 1
			c = 0
		else:
			c = c + 1
	if p:
		r = r + c // (k + 1)
	else:
		r = 1 + (n - 1) // (k + 1)
	print(r)
