#include<iostream>
#include<cstdio>
#include<cmath>
#include<algorithm>
#include<string>
#include<vector>
#include<map>

using namespace std;

int main(){
	int N,M;
	int dt,vt;
	while(scanf("%d %d",&N,&M)){
		if(N==0&&M==0)
		break;
		map<int, int> mp;
		for(int i=0;i<N;i++){
			scanf("%d %d",&dt,&vt);
			auto itr=mp.find(dt);
			if(itr!=mp.end()){
				if(itr->second < vt)
				mp[dt]=vt;
			}
			else{
				mp[dt]=vt;
			}
		}
		int sum=0;
		for(auto itr=mp.begin();itr!=mp.end();itr++){
			sum+=itr->second;
		}
		printf("%d\n",sum);
	}
}
