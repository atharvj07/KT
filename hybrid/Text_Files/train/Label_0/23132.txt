#define int long long

const int INF = 1e12 + 239;

int solve(vector<pair<int, char>> a, char F, char S) {
	int mx = -2 * INF;
	int ans = INF;
	for (auto [coord, c] : a) {
		if (c == S) {
			ans = min(ans, coord - mx);
		}
		if (c == F) {
			mx = coord;
		}
	}
	return ans;
}

signed main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);

	int n;
	cin >> n;
	map<int, vector<pair<int, char>>> xs;
	map<int, vector<pair<int, char>>> ys;
	map<int, vector<pair<int, char>>> dl;
	map<int, vector<pair<int, char>>> sm;
	for (int i = 0; i < n; i++) {
		int x, y;
		char c;
		cin >> x >> y >> c;
		x *= 2;
		y *= 2;
		xs[x].push_back({y, c});
		ys[y].push_back({x, c});
		dl[x - y].push_back({x, c});
		sm[x + y].push_back({x, c});
	}
	int ans = INF;
	for (auto [x, v] : xs) {
		sort(v.begin(), v.end());
		int C = solve(v, 'U', 'D');
		if (C == INF) continue;
		ans = min(ans, C / 2);
	}
	for (auto [y, v] : ys) {
		sort(v.begin(), v.end());
		int C = solve(v, 'R', 'L');
		if (C == INF) continue;
		ans = min(ans, C / 2);
	}
	for (auto [D, v] : dl) {
		sort(v.begin(), v.end());
		int C = solve(v, 'R', 'D');
		if (C != INF) ans = min(ans, C);
		C = solve(v, 'U', 'L');
		if (C != INF) ans = min(ans, C);
	}
	for (auto [S, v] : sm) {
		sort(v.begin(), v.end());
		int C = solve(v, 'D', 'L');
		if (C != INF) ans = min(ans, C);
		C = solve(v, 'R', 'U');
		if (C != INF) ans  = min(ans, C);
	}
	if (ans == INF) {
		cout << "SAFE\n";
	} else {
		cout << ans * 5 << endl;
	}
}
