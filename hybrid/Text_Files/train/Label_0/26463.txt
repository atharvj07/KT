#include <bits/stdc++.h>
using namespace std;
struct Line {
  long long x, y, r;
};
const int MAX_N = 1e5 + 100;
int BIT[MAX_N], n, m;
vector<pair<int, int> > value_X;
vector<int> value_Y;
Line horLines[MAX_N], verLines[MAX_N];
void update(int x, int val) {
  for (; x < MAX_N; x += x & (-x)) BIT[x] += val;
}
int get(int x) {
  int res = 0;
  for (; x > 0; x -= x & (-x)) res += BIT[x];
  return res;
}
void init(int L) {
  memset(BIT, 0, sizeof BIT);
  value_X.clear();
  value_Y.clear();
  for (int i = 1; i <= m; i++) {
    int first = horLines[i].x + L, last = horLines[i].x + horLines[i].r + 1 - L;
    if (first > last) continue;
    value_X.push_back(make_pair(first, -i));
    value_X.push_back(make_pair(last, -i - MAX_N));
    value_Y.push_back(horLines[i].y);
  }
  for (int i = 1; i <= n; i++) {
    value_X.push_back(make_pair(verLines[i].x, i));
    int first = verLines[i].y + L, last = verLines[i].y - L + verLines[i].r;
    if (first > last) continue;
    value_Y.push_back(first);
    value_Y.push_back(last);
  }
  sort(value_X.begin(), value_X.end());
  sort(value_Y.begin(), value_Y.end());
  value_Y.resize(unique(value_Y.begin(), value_Y.end()) - value_Y.begin());
}
bool check(int L) {
  init(L);
  for (int i = 0; i < value_X.size(); i++) {
    if (value_X[i].second < 0) {
      int pos;
      if (value_X[i].second <= -MAX_N) {
        pos = lower_bound(value_Y.begin(), value_Y.end(),
                          horLines[-(value_X[i].second + MAX_N)].y) -
              value_Y.begin() + 1;
        update(pos, -1);
      } else {
        pos = lower_bound(value_Y.begin(), value_Y.end(),
                          horLines[-value_X[i].second].y) -
              value_Y.begin() + 1;
        update(pos, 1);
      }
    } else {
      long long x = verLines[value_X[i].second].y + L,
                y = verLines[value_X[i].second].y +
                    verLines[value_X[i].second].r - L;
      if (x > y) continue;
      x = lower_bound(value_Y.begin(), value_Y.end(), x) - value_Y.begin() + 1;
      y = lower_bound(value_Y.begin(), value_Y.end(), y) - value_Y.begin() + 1;
      int res = get(y) - get(x - 1);
      if (res > 0) return true;
    }
  }
  return false;
}
void solve() {
  int kq = 0;
  int first = 0, last = 1e9;
  while (first <= last) {
    int mid = (first + last) / 2;
    if (check(mid)) {
      first = mid + 1;
      kq = max(kq, mid);
    } else
      last = mid - 1;
  }
  cout << kq;
}
void enter() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    cin >> verLines[i].x >> verLines[i].y >> verLines[i].r;
  for (int i = 1; i <= m; i++)
    cin >> horLines[i].x >> horLines[i].y >> horLines[i].r;
  solve();
}
int main() { enter(); }
