import sys
from math import ceil
readline = sys.stdin.readline
t = int(readline())
for i in range(t):
	(n, peak) = map(int, readline().split())
	arr = [x for x in range(1, n + 1)]
	if ceil(n / 2) > peak:
		if n % 2 == 1:
			for j in range(2, 2 * (peak + 1), 2):
				(arr[j - 1], arr[j]) = (arr[j], arr[j - 1])
			print(*arr)
		else:
			for j in range(2, 2 * (peak + 1), 2):
				(arr[j], arr[j + 1]) = (arr[j + 1], arr[j])
			print(*arr)
	else:
		print(-1)
