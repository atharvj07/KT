#include <bits/stdc++.h>
using namespace std;
const int nmax = 1000000 + 10;
vector<int> mars, q, on, help;
int x[nmax], y[nmax];
int n, k, i, j, aux;
vector<pair<int, int> > sol;
int main() {
  scanf("%d", &n);
  scanf("%d", &k);
  for (i = 1; i <= n; ++i) {
    scanf("%d", &x[i]);
    scanf("%d", &y[i]);
    q.push_back(x[i]);
    q.push_back(y[i]);
  }
  sort(q.begin(), q.end());
  q.resize(distance(q.begin(), unique(q.begin(), q.end())));
  for (i = 0; i < q.size(); ++i) help.push_back(q[i]), help.push_back(q[i]);
  q = help;
  mars.resize(q.size());
  on.resize(q.size());
  for (i = 1; i <= n; ++i) {
    x[i] = lower_bound(q.begin(), q.end(), x[i]) - q.begin();
    y[i] = lower_bound(q.begin(), q.end(), y[i]) - q.begin();
    y[i] += 1;
    mars[x[i]] += 1;
    mars[y[i]] -= 1;
  }
  for (i = 1; i < mars.size(); ++i) mars[i] += mars[i - 1];
  for (i = 0; i < mars.size(); ++i)
    if (mars[i] >= k) on[i] = 1;
  i = 0;
  while (true) {
    while (i < on.size() && !on[i]) i++;
    if (i == on.size()) break;
    j = i;
    while (j < on.size() && on[j]) j++;
    sol.push_back({q[i], q[j - 1]});
    i = j;
    if (i == on.size()) break;
  }
  printf("%d\n", sol.size());
  for (i = 0; i < sol.size(); ++i)
    printf("%d %d\n", sol[i].first, sol[i].second);
  return 0;
}
