int gcd(int x, int y) {
    int r;
    if(x < y) swap(x, y);

    while(y > 0){
        r = x % y;
        x = y;
        y = r;
    }
    return x;
}

int main(){
    int p, q;
    cin >> p >> q;

    q/= gcd(p, q);

    int ans = 1;
    for(int i = 2; q >= i * i; i++){
        if(q % i == 0){
            ans*=i;
            while(q % i == 0){
                q/=i;
            }
        }
    }
    cout << ans * q << endl;
}
