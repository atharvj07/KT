from heapq import heappush, heappop
import os
import io
from math import inf
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

def li():
	return [int(i) for i in input().rstrip().split()]
(n, m) = li()
(sx, sy, fx, fy) = li()
s = (sx, sy)
instant = [s]
for i in range(m):
	(x, y) = li()
	instant.append((x, y))
g = {(x, y): set() for (x, y) in instant}
w = {}

def build_axis(instant_slots):
	for i in range(1, len(instant_slots)):
		(x1, y1) = instant_slots[i - 1]
		(x2, y2) = instant_slots[i]
		v1 = (x1, y1)
		v2 = (x2, y2)
		c = min(abs(x1 - x2), abs(y1 - y2))
		if v2 != s:
			g[v1].add(v2)
			w[v1, v2] = c
		if v1 != s:
			g[v2].add(v1)
			w[v2, v1] = c
instant_x = sorted(instant)
build_axis(instant_x)
instant_y = sorted(instant, key=lambda pos: pos[1])
build_axis(instant_y)

def dijsktra():
	d = {(x, y): inf for (x, y) in instant}
	d[s] = 0
	q = [(0, s)]
	while q:
		(_, u) = heappop(q)
		for v in g[u]:
			x = d[u] + w[u, v]
			if x < d[v]:
				d[v] = x
				heappush(q, (d[v], v))
	return d
d = dijsktra()
print(min([d[x, y] + abs(x - fx) + abs(y - fy) for (x, y) in instant]))
