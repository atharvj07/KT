(n, k) = map(int, input().strip().split())
arr = list(map(int, input().strip().split()))
z = 0
zeros = []
for i in range(n):
	if arr[i] == 1:
		z += 1
	else:
		zeros.append([i, z])
		if len(zeros) > 1:
			zeros[-2].append(z)
		z = 0
try:
	zeros[-1].append(z)
except:
	print(n)
	print(*arr)
	exit()
if k == 0:
	print(max(max(zeros, key=lambda x: max(x[1], x[2]))[1:]))
	print(*arr)
elif len(zeros) <= k:
	print(n)
	print(*[1 for i in range(n)])
else:
	maxi = [-1, -1]
	for i in range(len(zeros) - k + 1):
		diff = zeros[i][1] + zeros[i + k - 1][2] + zeros[i + k - 1][0] - zeros[i][0] + 1
		if diff > maxi[0]:
			maxi = [diff, i]
	for i in range(zeros[maxi[1]][0], zeros[maxi[1] + k - 1][0] + 1):
		arr[i] = 1
	print(maxi[0])
	print(*arr)
