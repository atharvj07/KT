n = int(input())
(k, h, a) = (0, 0, list())
for i in range(n):
	(x, y) = map(str, input().split())
	if x == '+' and k:
		k -= 1
		a.append(y)
	elif x == '+' and (not k):
		h += 1
		a.append(y)
	elif x == '-' and y in a:
		k += 1
		a.remove(y)
	elif x == '-' and y not in a:
		h += 1
		k += 1
print(h)
