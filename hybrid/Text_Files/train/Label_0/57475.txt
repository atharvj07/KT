import javax.sound.midi.SysexMessage;
import java.io.IOException;
import java.io.InputStream;
import java.util.ArrayList;
import java.util.Arrays;

public class Solution {
    static FastReader in;
    static ArrayList<Integer> tree[];
    static ArrayList<Integer> level[];
    static int maxdepth,count=0;
    static int st[][];
    static int[] ask(ArrayList<Integer> list)throws IOException{
        count++;
        System.out.print("? "+list.size()+" ");
        for(int x:list)
            System.out.print(x+" ");
        System.out.println();
        int ans[]={in.nextInt(),in.nextInt()};
        if(ans[0]==ans[1] && ans[0]==-1)
            System.exit(0);
        if(count>=13)
            System.out.println(1/0);
        return ans;
    }

    static void dfs(int u,int p,int lev){
        level[lev].add(u);
        maxdepth=Math.max(maxdepth,lev);
        for(int v:tree[u]){
            if(v!=p){
                dfs(v,u,lev+1);
            }
        }
    }
    public static void main(String[] args) throws IOException {
        int i, j;
        in = new FastReader(System.in);
        boolean testcases = true;
        int t = testcases ? in.nextInt() : 1;
        while (t-- > 0) {
            count=0;
            maxdepth=0;
            st=new int[1009][2];
            for(i=0;i<=1000;i++)
                Arrays.fill(st[i],-1);
            int n = in.nextInt();
            tree=new ArrayList[n+1];
            for(i=1;i<=n;i++){
                tree[i]=new ArrayList<>();
            }

            for(i=0;i<n-1;i++){
                int x=in.nextInt();
                int y=in.nextInt();
                tree[x].add(y);
                tree[y].add(x);
            }
            ArrayList<Integer> list=new ArrayList<>();
            for(i=1;i<=n;i++)
                list.add(i);
            int ans[]=ask(list);
            int root=ans[0];
            int totaldist=ans[1];
            level=new ArrayList[n+1];
            for(i=0;i<=n;i++)
                level[i]=new ArrayList<>();
            dfs(root,-1,0);
            int l=totaldist/2;
            int r=Math.min(totaldist,maxdepth);
            int firstnode=-1;
            int lastans[]=new int[2];
            while(l<=r){
                if(l==r){
                    if(st[l][0]==-1) {
                        ans = ask(level[l]);
                    }
                    else
                        ans=st[l];
                    if(totaldist!=ans[1])
                        ans=lastans;
                    firstnode=ans[0];
                    break;
                }
                if(l+1==r){
                    if(st[r][0]==-1) {
                        ans = ask(level[r]);
                        st[r]=ans;
                    }
                    else
                        ans=st[r];
                    if(totaldist==ans[1])
                        firstnode=ans[0];
                    else{
                        if(st[l][0]==-1) {
                            ans = ask(level[l]);
                            st[l]=ans;
                        }
                        else
                            ans=st[l];
                        if(totaldist!=ans[1])
                            ans=lastans;
                        firstnode=ans[0];
                    }
                    break;
                }
                int m=(l+r)/2;
                if(st[m][0]==-1) {
                    ans = ask(level[m]);
                    st[m]=ans;
                }
                else
                    ans=st[m];
                if(ans[1]!=totaldist)
                    r=m-1;
                else {
                    l = m + 1;
                    lastans=ans;
                }
            }
            level=new ArrayList[n+1];
            for(i=0;i<=n;i++)
                level[i]=new ArrayList<>();
            dfs(firstnode,-1,0);
            ans=ask(level[totaldist]);
            int secondnode=ans[0];
            System.out.println("! "+firstnode+" "+secondnode);
            String s=in.next();
            if(!s.equals("Correct"))
                break;


        }
    }
}

class FastReader {

    byte[] buf = new byte[2048];
    int index, total;
    InputStream in;

    FastReader(InputStream is) {
        in = is;
    }

    int scan() throws IOException {
        if (index >= total) {
            index = 0;
            total = in.read(buf);
            if (total <= 0) {
                return -1;
            }
        }
        return buf[index++];
    }

    String next() throws IOException {
        int c;
        for (c = scan(); c <= 32; c = scan()) ;
        StringBuilder sb = new StringBuilder();
        for (; c > 32; c = scan()) {
            sb.append((char) c);
        }
        return sb.toString();
    }

    int nextInt() throws IOException {
        int c, val = 0;
        for (c = scan(); c <= 32; c = scan()) ;
        boolean neg = c == '-';
        if (c == '-' || c == '+') {
            c = scan();
        }
        for (; c >= '0' && c <= '9'; c = scan()) {
            val = (val << 3) + (val << 1) + (c & 15);
        }
        return neg ? -val : val;
    }

    long nextLong() throws IOException {
        int c;
        long val = 0;
        for (c = scan(); c <= 32; c = scan()) ;
        boolean neg = c == '-';
        if (c == '-' || c == '+') {
            c = scan();
        }
        for (; c >= '0' && c <= '9'; c = scan()) {
            val = (val << 3) + (val << 1) + (c & 15);
        }
        return neg ? -val : val;
    }
}
