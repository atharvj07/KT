ans = []
for _ in range(int(input())):
	(n, m, a, b) = map(int, input().split())
	if a == b == 1 or (a == n and b == m):
		(w, x, y, z) = (n, 1, 1, m)
	elif a == 1 and b == m or (a == n and b == 1):
		(w, x, y, z) = (1, 1, n, m)
	else:
		(w, x, y, z) = (1, 1, n, m)
	ans.append(str(w) + ' ' + str(x) + ' ' + str(y) + ' ' + str(z))
print('\n'.join(ans))
