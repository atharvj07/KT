#include <iostream>
using namespace std;

int n;
const int Max_n = 200 + 10;
int a[Max_n];

void solve(){
	scanf("%d",&n);
	int cnt = 0;
	for (int i=1;i<=n+n;++i){
		scanf("%d",&a[i]);
		if (a[i] & 1){
			cnt++;
		}
	}
	if (cnt==n){
		cout << "Yes" << endl;
		return;
	}
	cout << "No" << endl;
}

int main(void){
	int t;
	scanf("%d",&t);
	for (int i=1;i<=t;++i){
		solve();
	}
	return 0;
}
