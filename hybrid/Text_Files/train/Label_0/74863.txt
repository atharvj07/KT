q = int(input())
for i in range(0, q):
	n = int(input())
	p = [int(x) for x in input().split()]
	if n == 1 and p[0] == p[1] and (p[2] == p[3]):
		print('YES')
	else:
		p = sorted(p)
		s = p[0] * p[4 * n - 1]
		e = 0
		for j in range(0, n):
			if p[2 * j] == p[2 * j + 1] and p[4 * n - 2 * j - 2] == p[4 * n - 2 * j - 1] and (p[2 * j] * p[4 * n - 2 * j - 1] == s):
				e += 1
		if e == n:
			print('YES')
		else:
			print('NO')
