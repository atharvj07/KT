#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC optimize("unroll-loops")
using namespace std;
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    long long x, y;
    cin >> x >> y;
    long long r = x - y;
    if (r != 1) {
      cout << "YES";
    } else
      cout << "NO";
    cout << "\n";
  }
  return 0;
}
