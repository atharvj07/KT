for _ in range(int(input())):
	(n, h) = map(int, input().split())
	a = list(map(int, input().split()))
	mi = a[0]
	ma = a[0]
	a = a[1:]
	s = 1
	for x in a:
		mi = max(x, mi - h + 1)
		ma = min(x + h - 1, ma + h - 1)
		if ma < mi:
			s = 0
	if mi != a[-1]:
		s = 0
	print('YES') if s == 1 else print('NO')
