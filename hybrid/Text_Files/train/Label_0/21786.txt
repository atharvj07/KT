#include<iostream>
#include<vector>
#include<string>
#include<map>
using namespace std;

#define rep(i,j) for(int i = 0;i<(j);i++)

int num_len(int n){
  int res = 0;
  while(n){
    n /= 10;
    res++;
  }
  return res;
}

int count(vector<char> &cs, int cnt){
  if(cs.size() == 0 or cnt == 0) return 0;
  
  if(cs.size() == 1){
    if(cnt > 1) return 3;
    else return 1;
  } else if(cnt > 1){
    return cs.size() * 2 - 1 + 2 + num_len(cnt) + 1;
  } else {
    return cs.size() * 2 - 1;
  }
}

int main(){
  int N; cin >> N;
  string s; cin >> s;
  vector<int> c_cnt('z'-'a' + 1, 0);
  for(char c : s){
    if('a' <= c and c <= 'z') c_cnt[c-'a']++;
  }
  map<int, vector<char>> i_cnt;
  rep(i, c_cnt.size()){
    if(c_cnt[i] > 0){
      i_cnt[c_cnt[i]].push_back(i + 'a');
    }
  }
  bool flg = false;
  int ans  = 0;
  for(auto& p : i_cnt){
    int cnt = p.first;
    vector<char> cs = p.second;
    // cerr << cnt << " " ;
    // rep(j, cs.size()) cerr << cs[j] << " ";
    // cerr << endl;
    if(flg){
      ans++;
    }
    flg = true;
    ans += count(cs, cnt);
  }
  cout << ans << endl;
  return 0;
}
