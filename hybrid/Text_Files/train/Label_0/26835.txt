#include <bits/stdc++.h>
using namespace std;
int main(void) {
  long long int n;
  scanf("%lld\n", &n);
  long long int k = 3 * (n + 1) + 1;
  printf("%lld\n", k);
  printf("%d %d\n", 0, 0);
  for (long long int i = 1; i <= n + 1; i++) {
    printf("%lld %lld\n", i, i);
    printf("%lld %lld\n", i, i - 1);
    printf("%lld %lld\n", i - 1, i);
  }
  return 0;
}
