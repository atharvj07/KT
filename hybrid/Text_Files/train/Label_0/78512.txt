from collections import Counter
import math
n = int(input())
ls = {}
arr = {}
for i in range(n):
	(n, m) = map(int, input().split())
	if n + m not in ls:
		ls[n + m] = 1
	else:
		ls[n + m] += 1
	if n - m not in arr:
		arr[n - m] = 1
	else:
		arr[n - m] += 1
ans = 0
for i in ls:
	if ls[i] > 1:
		ans += ls[i] * (ls[i] - 1) // 2
for i in arr:
	if arr[i] > 1:
		ans += arr[i] * (arr[i] - 1) // 2
print(ans)
