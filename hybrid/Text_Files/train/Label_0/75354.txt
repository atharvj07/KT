from bisect import bisect
n = int(input())
S = list(map(int, input().split()))
C = list(map(int, input().split()))

def insert_sign(S, C, s, c):
	pos_s = bisect(S, s)
	pos_c = bisect(C, c)
	if pos_s >= pos_c:
		S[pos_c:pos_s] = (s,)
		C[pos_c:pos_s] = (c,)

def minimal_rest_cost(S, C, s):
	pos = bisect(S, s)
	return C[pos]
MAX = 10 ** 10
(S1, C1) = [[MAX], [MAX]]
(S2, C2) = [[MAX], [MAX]]
(S3, C3) = [[MAX], [MAX]]
for (s, c) in zip(S[::-1], C[::-1]):
	insert_sign(S3, C3, s, c + minimal_rest_cost(S2, C2, s))
	insert_sign(S2, C2, s, c + minimal_rest_cost(S1, C1, s))
	insert_sign(S1, C1, s, c)
if C3[0] < MAX:
	print(C3[0])
else:
	print(-1)
