import sys
input = sys.stdin.readline
from collections import defaultdict

def gcd(a, b):
	while b:
		(a, b) = (b, a % b)
	return a
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
zero = 0
bonus = 0
d = defaultdict(dict)
for (ai, bi) in zip(a, b):
	if ai == 0:
		if bi == 0:
			bonus += 1
		continue
	if bi == 0:
		zero += 1
		continue
	g = gcd(ai, bi)
	(ai2, bi2) = (ai // g, bi // g)
	if ai2 < 0:
		ai2 *= -1
		bi2 *= -1
	if bi2 not in d[ai2]:
		d[ai2][bi2] = 1
	else:
		d[ai2][bi2] += 1
ans = 0
for k in d.keys():
	for k2 in d[k].keys():
		ans = max(ans, d[k][k2])
ans = max(ans, zero)
ans += bonus
print(ans)
