void solve()
{
    ll i,n;

    cin>>n>>k;
    vector<long double> p(n);
    fr(i,n)cin>>p[i];
    long double sum=p[0],ans=0;
    for(ll i=1;i<n;i++){
        if(p[i]*100.0/sum>k){
            ans+=ceil((p[i]*100.0/k)-sum);
            sum=ceil(p[i]*100.0/k);
        }
        sum+=p[i];
    }
    cout<<(ll)ans<<endl;
}
