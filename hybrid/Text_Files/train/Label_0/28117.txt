#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <utility>
#include <iomanip>
using namespace std;
int t, a[200];
vector<string> ans;

void solve() {
    int n, odd = 0, even = 0;
    cin >> n;
    for (int i = 0; i < 2 * n; i++) {
        cin >> a[i];
        if (a[i] % 2 == 1)
            odd++;
        else
            even++;
    }
    if (odd == even)
        ans.push_back("Yes");
    else
        ans.push_back("No");
}
int main(){
    cin >> t;
    while (t--) {
        solve();
    }
    for (auto x : ans)
        cout << x << endl;
    return 0;
}
