import sys
s = input()
j = len(s)
n = len(s)
ans = []
for i in range(n):
	if i >= j:
		break
	if s[i] == '(':
		while True:
			j -= 1
			if i >= j:
				break
			if s[j] == ')':
				ans.append(i)
				ans.append(j)
				break
if len(ans) == 0:
	print(0)
	sys.exit()
else:
	print(1)
print(len(ans))
ans = [ans[i] + 1 for i in range(len(ans))]
ans.sort()
print(*ans)
