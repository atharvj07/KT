#include <bits/stdc++.h>
using namespace std;
struct data {
  int cs, x, y;
} mem[100000];
int a[10][100], n, k, cnt, res;
void lui(int h, int c) {
  if (n == 1) {
    if (h == 2)
      mem[++cnt] = {a[3][c], 2, c};
    else
      mem[++cnt] = {a[2][c], 3, c};
    swap(a[2][c], a[3][c]);
    return;
  }
  if (h == 2) {
    for (int j = c - 1; j >= 1; --j)
      if (a[2][j])
        mem[++cnt] = {a[2][j], 2, j + 1}, a[2][j + 1] = a[2][j], a[2][j] = 0;
    if (a[3][1]) mem[++cnt] = {a[3][1], 2, 1}, a[2][1] = a[3][1], a[3][1] = 0;
    for (int j = 2; j <= n; ++j)
      if (a[3][j])
        mem[++cnt] = {a[3][j], 3, j - 1}, a[3][j - 1] = a[3][j], a[3][j] = 0;
    if (a[2][n]) mem[++cnt] = {a[2][n], 3, n}, a[3][n] = a[2][n], a[2][n] = 0;
    for (int j = n - 1; j > c; --j)
      if (a[2][j])
        mem[++cnt] = {a[2][j], 2, j + 1}, a[2][j + 1] = a[2][j], a[2][j] = 0;
    return;
  }
  if (h == 3) {
    for (int j = c + 1; j <= n; ++j)
      if (a[3][j])
        mem[++cnt] = {a[3][j], 3, j - 1}, a[3][j - 1] = a[3][j], a[3][j] = 0;
    if (a[2][n]) mem[++cnt] = {a[2][n], 3, n}, a[3][n] = a[2][n], a[2][n] = 0;
    for (int j = n - 1; j > 0; --j)
      if (a[2][j])
        mem[++cnt] = {a[2][j], 2, j + 1}, a[2][j + 1] = a[2][j], a[2][j] = 0;
    if (a[3][1]) mem[++cnt] = {a[3][1], 2, 1}, a[2][1] = a[3][1], a[3][1] = 0;
    for (int j = 2; j < c; ++j)
      if (a[3][j]) {
        mem[++cnt] = {a[3][j], 3, j - 1}, a[3][j - 1] = a[3][j], a[3][j] = 0;
      }
  }
}
int main() {
  cin >> n >> k;
  for (int i = 1; i <= 4; ++i)
    for (int j = 1; j <= n; ++j) cin >> a[i][j];
  cnt = 0;
  res = 0;
  if (k == 2 * n) {
    int tmp = 0;
    for (int i = 1; i <= n; ++i) {
      if (a[1][i] && a[1][i] == a[2][i]) {
        tmp = a[1][i];
        mem[++cnt] = {tmp, 1, i};
        a[2][i] = 0;
        ++res;
      }
      if (a[3][i] && a[3][i] == a[4][i]) {
        tmp = a[3][i];
        mem[++cnt] = {tmp, 4, i};
        ++res;
        a[3][i] = 0;
      }
    }
    if (!tmp) {
      cout << -1;
      return 0;
    }
  }
  while (res < k) {
    int tmp;
    for (int i = 1; i <= n; ++i) {
      if (a[2][i] && a[1][i] == a[2][i]) {
        tmp = a[2][i];
        mem[++cnt] = {tmp, 1, i};
        a[2][i] = 0;
        ++res;
      }
      if (a[3][i] && a[3][i] == a[4][i]) {
        tmp = a[3][i];
        mem[++cnt] = {tmp, 4, i};
        ++res;
        a[3][i] = 0;
      }
    }
    if (res == k) break;
    for (int i = 1; i <= n; ++i) {
      if (!a[2][i]) {
        lui(2, i);
        break;
      }
      if (!a[3][i]) {
        lui(3, i);
        break;
      }
    }
  }
  cout << cnt << endl;
  for (int i = 1; i <= cnt; ++i)
    cout << mem[i].cs << " " << mem[i].x << " " << mem[i].y << endl;
  return 0;
}
