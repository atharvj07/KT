from math import ceil
for test in range(int(input())):
	(n, k) = map(int, input().split())
	a = [int(i) for i in input().split()]
	s = sum(a)
	m = 0
	a.sort()
	for i in range(1, n):
		m += a[i] - a[0]
	res = max(ceil((s - m - k) / n), 0) + n - 1
	for i in range(1, n):
		m -= a[i] - a[0]
		c = max(ceil((s - m - k) / (n - i)), 0) + n - i - 1
		res = min(res, c)
	print(res)
