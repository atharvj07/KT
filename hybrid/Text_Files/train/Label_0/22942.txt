#include <bits/stdc++.h>
using namespace std;
void print(string str) {
  int a[11];
  memset(a, 0, sizeof(a));
  int len = str.length();
  int dir = 1;
  int cp = 0;
  int pre = -1;
  while (cp < len && cp >= 0) {
    if (str[cp] == '>' && pre != -1) {
      dir = 1;
      str[pre] = '0' - 1;
      pre = cp;
      cp++;
    } else if (str[cp] == '<' && pre != -1) {
      dir = 0;
      str[pre] = '0' - 1;
      pre = cp;
      cp--;
    } else if (str[cp] == '>') {
      dir = 1;
      pre = cp;
      cp++;
    } else if (str[cp] == '<') {
      dir = 0;
      pre = cp;
      cp--;
    } else if (str[cp] >= '0' && str[cp] <= '9') {
      pre = -1;
      a[str[cp] - '0']++;
      str[cp]--;
      if (dir == 1)
        cp++;
      else
        cp--;
    } else {
      if (dir == 1)
        cp++;
      else
        cp--;
    }
  }
  for (int i = int(0); i <= int(9); i++) {
    printf("%d ", a[i]);
  }
  printf("\n");
}
int main() {
  int n;
  scanf("%d", &n);
  int q;
  scanf("%d", &q);
  string str;
  cin >> str;
  for (int i = int(0); i <= int(q - 1); i++) {
    int a, b;
    scanf("%d", &a);
    scanf("%d", &b);
    string temp = "";
    a--;
    b--;
    for (int i = int(a); i <= int(b); i++) {
      temp += str[i];
    }
    print(temp);
  }
  return 0;
}
