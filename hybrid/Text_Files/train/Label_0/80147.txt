from collections import Counter
t = int(input())

def main():
	(n, m) = [int(s) for s in input().split()]
	c = Counter()
	res = [0] * m
	arr = []
	for i in range(m):
		l = [int(s) for (i, s) in enumerate(input().split()) if i > 0]
		arr.append(l)
		if len(l) == 1:
			c[l[0]] += 1
			res[i] = l[0]
	for value in c.values():
		if value > (m + 1) // 2:
			print('NO')
			return
	for i in range(m):
		if res[i] == 0:
			if c[arr[i][0]] < c[arr[i][1]]:
				res[i] = arr[i][0]
				c[arr[i][0]] += 1
			else:
				res[i] = arr[i][1]
				c[arr[i][1]] += 1
	print('YES')
	print(' '.join([str(e) for e in res]))
for _ in range(t):
	main()
