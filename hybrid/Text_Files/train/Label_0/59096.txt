import java.util.*;
import java.io.*;

public class Main {
    public static StreamTokenizer st;
    public static PrintWriter cout;
    public static int nextInt() throws IOException {
        st.nextToken();
        return (int)(st.nval);
    }
    public static double nextDouble() throws IOException {
        st.nextToken();
        return (double)(st.nval);
    }    
    public static long nextlong() throws IOException {
        st.nextToken();
        return (long)(st.nval);
    }
    public static void init() throws IOException {
        st = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
        cout = new PrintWriter(new OutputStreamWriter(System.out));
    }

    public static int[] head = new int[1000000];
    public static int[] link = new int[2000000];
    public static int[] dest = new int[2000000];
    public static int[] tmp = new int[1000000];
    public static int g_sz = 0;
    public static int tmp_sz = 0;
    public static long[] hash1 = new long[1000000];
    public static long[] hash2 = new long[1000000];
    public static void add_arc(int u, int v) {
        dest[g_sz] = v;
        link[g_sz] = head[u];
        head[u] = g_sz++;
        
        dest[g_sz] = u;
        link[g_sz] = head[v];
        head[v] = g_sz++;
    }
    public static void get_hashes(int id) {
        long p = 1000000007;
        long sum1 = 0, sum2 = 0;
        long pow1 = 1, pow2 = 1;
        for (int i = 0; i < tmp_sz; ++i) {
            if (tmp[i] > id && (i == 0 || tmp[i - 1] < id)) {
                sum1 += pow1 * (id + 1);
                pow1 *= p;
            }
            sum1 += pow1 * (tmp[i] + 1);
            sum2 += pow2 * (tmp[i] + 1);
            pow1 *= p;
            pow2 *= p;
        }
        if (tmp_sz == 0 || id > tmp[tmp_sz - 1]) {
            sum1 += pow1 * (id + 1);
        }
        hash1[id] = sum1;
        hash2[id] = sum2;
        //cout.println(sum1 + " " + sum2);
    }
    public static void main(String[] argc) throws IOException {
        init();
        int n = nextInt();
        int m = nextInt();
        for (int i = 0; i < n; ++i) {
            head[i] = -1;
        }
        for (int i = 0; i < m; ++i) {
            int u = nextInt();
            int v = nextInt();
            add_arc(u - 1, v - 1);
        }
        for (int i = 0; i < n; ++i) {
            tmp_sz = 0;
            for (int arc = head[i]; arc != -1; arc = link[arc]) {
                tmp[tmp_sz++] = dest[arc];
            }
           // cout.println("SZ = " + tmp_sz);
            Arrays.sort(tmp, 0, tmp_sz);
            get_hashes(i);
        }
        Arrays.sort(hash1, 0, n);
        Arrays.sort(hash2, 0, n);
        //cout.flush();
        
        int p1 = 0, p2 = 0;
        long ans = 0;
        for (int i = 1; i <= n; ++i) {
            if (i == n || hash1[i] != hash1[p1]) {
                long cur = i - p1;
                ans += cur * (cur - 1) / 2;
                p1 = i;
            }
            if (i == n || hash2[i] != hash2[p2]) {
                long cur = i - p2;
                ans += cur * (cur - 1) / 2;
                p2 = i;
            }
        }
        cout.println(ans);
        cout.flush();
    } 
}
