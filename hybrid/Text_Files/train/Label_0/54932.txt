import java.util.Scanner;
public class Test {

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		int l=sc.nextInt();
		double[]arr=new double[n+2];
		arr[0]=0;
		arr[n+1]=l;
		for(int i=1;i<n+1;i++){
			arr[i]=sc.nextInt();
		}
		for(int i=0;i<n+2;i++){
			for(int j=0;j<n+1-i;j++){
				if(arr[j]>arr[j+1]){
					double temp=arr[j];
					arr[j]=arr[j+1];
					arr[j+1]=temp;
				}
			}
		}
		double maxdiff=0;
		for(int i=1;i<n;i++){
			if((arr[i+1]-arr[i])>maxdiff) maxdiff=arr[i+1]-arr[i];
		}
		double a=maxdiff/2;
		if(arr[1]==0&&arr[n]==l) a=maxdiff/2;
		else if(arr[1]==0&&arr[n+1]-arr[n]>a) a=arr[n+1]-arr[n];
		else if(arr[n]==l&&arr[1]-arr[0]>a) a=arr[1];
		else if(arr[1]!=0&&arr[n]!=l&&arr[1]-arr[0]>a&&arr[n+1]-arr[n]<arr[1]-arr[0]) a=arr[1];
		else if(arr[1]!=0&&arr[n]!=l&&arr[n+1]-arr[n]>a) a=arr[n+1]-arr[n];
		System.out.println(a);
	}

}
