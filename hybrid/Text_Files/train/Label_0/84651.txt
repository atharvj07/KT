(n, m) = map(int, input().split())
d = [[] for i in range(n)]
f = [0] * n
for i in range(m):
	a = list(map(int, input().split()))
	f[a[0] - 1] += 1
	f[a[1] - 1] += 1
	a.sort()
	d[a[1] - 1] += [a[0]]
ans = []
for i in range(n):
	d[i].sort()
	d[i].reverse()
	for j in d[i]:
		for k in d[i]:
			if k in d[j - 1]:
				res = f[k - 1] - 2 + f[j - 1] - 2 + f[i] - 2
				ans += [res]
if ans:
	print(min(ans))
else:
	print('-1')
