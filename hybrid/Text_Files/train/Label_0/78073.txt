n = int(input())
a = []
for z in range(n):
	a.append(list(input()))
x = a[0][0]
y = a[0][1]
if x == y:
	print('NO')
	quit()
for i in range(n):
	for j in range(n):
		if (i == j or i + j == n - 1) and a[i][j] != x:
			print('NO')
			quit()
		elif i != j and i + j != n - 1 and (a[i][j] != y):
			print('NO')
			quit()
print('YES')
