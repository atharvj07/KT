(a, b, c) = input().split()
(a, b, c) = (int(a), int(b), int(c))
usb = 'USB'
ps2 = 'PS/2'
u = []
ps = []
n = int(input())
while n > 0:
	(cost, s) = input().split()
	if s == usb:
		u.append(int(cost))
	elif s == ps2:
		ps.append(int(cost))
	n -= 1
u.sort()
ps.sort()
(cost, count) = (0, 0)
(a_curr, ps_curr) = (0, 0)
(a_count, ps_count) = (len(u), len(ps))
cost += sum(u[0:min(a_count, a)])
a_curr = min(a_count, a)
count = a_curr
cost += sum(ps[0:min(ps_count, b)])
ps_curr = min(ps_count, b)
count += ps_curr
new_arr = u[a_curr:] + ps[ps_curr:]
count += min(c, len(new_arr))
new_arr.sort()
cost += sum(new_arr[:c])
print(count, cost, sep=' ')
