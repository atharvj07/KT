from functools import reduce
n = int(input())
a = [tuple(map(int, input().split())) for i in range(n)]
a = list(map(lambda x: tuple(map(lambda t: t % 2, x)), a))
(sx, sy) = reduce(lambda x, y: (x[0] + y[0], x[1] + y[1]), a, (0, 0))
sx %= 2
sy %= 2
a = [x for x in a if x in ((0, 1), (1, 0))]
if sx + sy == 0:
	print(0)
elif sx + sy == 1:
	print(-1)
elif len(a) and n > 1:
	print(1)
else:
	print(-1)
