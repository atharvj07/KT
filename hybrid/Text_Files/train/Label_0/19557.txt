//ALDS1_10B Matrix Chain Multiplication
#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

static const int N=100;
vector<int> p(N+1);
vector< vector<int> > m(N+1,vector<int>(N+1));

int main(){
 int n;
 cin >> n;
 for(int i=1;i<=n;i++){
 cin>>p[i-1]>>p[i];
 }
 for(int l=2;l<=n;l++){
  for(int i=1;i<=n-l+1;i++){
   int j=i+l-1;
   m[i][j]=(1<<21);
   for(int k=i;k<=j-1;k++){
    m[i][j]=min(m[i][j],m[i][k]+m[k+1][j]+p[i-1]*p[k]*p[j]);
   }
  }
 }

 cout<<m[1][n]<< endl;

 return 0;
}
