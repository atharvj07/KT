#include <bits/stdc++.h>
using namespace std;
class InputReader {
 public:
  InputReader() {
    input_file = stdin;
    cursor = 0;
    fread(buffer, SIZE, 1, input_file);
  }
  inline InputReader &operator>>(int &n) {
    while ((buffer[cursor] < '0' || buffer[cursor] > '9') &&
           buffer[cursor] != '-') {
      advance();
    }
    int sign = 1;
    if (buffer[cursor] == '-') {
      sign = -1;
      advance();
    }
    n = 0;
    while ('0' <= buffer[cursor] && buffer[cursor] <= '9') {
      n = n * 10 + buffer[cursor] - '0';
      advance();
    }
    n *= sign;
    return *this;
  }

 private:
  FILE *input_file;
  static const int SIZE = 1 << 17;
  int cursor;
  char buffer[SIZE];
  inline void advance() {
    ++cursor;
    if (cursor == SIZE) {
      cursor = 0;
      fread(buffer, SIZE, 1, input_file);
    }
  }
};
ifstream fin("");
ofstream fout("");
vector<int> v[200010];
int b[200010];
int c[200010];
int zeros[200010];
int ones[200010];
int used[200010];
int cost[200010];
int t[200010];
void df(int x) {
  used[x] = 1;
  if (b[x] != c[x]) {
    if (b[x] == 0)
      zeros[x]++;
    else
      ones[x]++;
  }
  for (auto it : v[x]) {
    if (used[it]) continue;
    t[it] = x;
    df(it);
    zeros[x] += zeros[it];
    ones[x] += ones[it];
  }
}
void df2(int x) {
  zeros[x] = 0;
  ones[x] = 0;
  used[x] = 1;
  if (b[x] != c[x]) {
    if (b[x] == 0)
      zeros[x]++;
    else
      ones[x]++;
  }
  for (auto it : v[x]) {
    if (it == t[x]) continue;
    if (used[it]) {
      zeros[x] += zeros[it];
      ones[x] += ones[it];
      continue;
    }
    df2(it);
    zeros[x] += zeros[it];
    ones[x] += ones[it];
  }
}
void check(int x) {
  used[x] = 1;
  for (auto it : v[x]) {
    if (it != t[x] && !used[it]) check(it);
  }
}
int main() {
  InputReader cin;
  cout.sync_with_stdio(false);
  int n;
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> cost[i] >> b[i] >> c[i];
  }
  for (int i = 1; i < n; ++i) {
    int x, y;
    cin >> x >> y;
    v[x].push_back(y);
    v[y].push_back(x);
  }
  int x = 0;
  int y = 0;
  for (int i = 1; i <= n; ++i)
    if (b[i] != c[i]) {
      if (b[i] == 0)
        ++x;
      else
        ++y;
    }
  if (x != y) {
    cout << -1;
    return 0;
  }
  vector<pair<int, int> > w;
  for (int i = 1; i <= n; ++i) w.push_back(make_pair(cost[i], i));
  sort(w.begin(), w.end());
  df(1);
  long long ans = 0;
  memset(used, 0, sizeof(used));
  for (int i = 0; i < w.size(); ++i) {
    if (used[w[i].second] || min(zeros[w[i].second], ones[w[i].second]) == 0)
      continue;
    df2(w[i].second);
    ans +=
        1LL * min(zeros[w[i].second], ones[w[i].second]) * 2 * 1LL * w[i].first;
    int x = min(zeros[w[i].second], ones[w[i].second]);
    zeros[w[i].second] -= x;
    ones[w[i].second] -= x;
  }
  cout << ans;
}
