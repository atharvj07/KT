import math
t1 = list(map(int, input().split(':')))
t2 = list(map(int, input().split(':')))
hour = (t1[0] + t2[0]) / 2 if t1 <= t2 else (t1[0] + t2[0]) / 2 - 12
minute = 0
if math.floor(hour) != hour:
	minute = 30
	hour = math.trunc(hour)
minute += (t1[1] + t2[1]) / 2
if minute >= 60:
	minute -= 60
	hour = (hour + 1) % 24
hour = str(math.trunc(hour)) if hour >= 10 else '0' + str(math.trunc(hour))
minute = str(math.trunc(minute)) if minute >= 10 else '0' + str(math.trunc(minute))
print(hour + ':' + minute)
