inf = -10 ** 18
(n, m) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
a.sort(reverse=True)
if len(a) > 3:
	Max = [(a[0] * max(b), a[0] * min(b), 0 if 0 in b else inf), (a[1] * max(b), a[1] * min(b), 0 if 0 in b else inf), (0 if 0 in a else inf, 0 if 0 in a else inf), (a[len(a) - 2] * max(b), a[len(a) - 2] * min(b), 0 if 0 in b else inf), (a[len(a) - 1] * max(b), a[len(a) - 1] * min(b), 0 if 0 in b else inf)]
	MAX = [max(Max[0]), max(Max[1]), max(Max[2]), max(Max[3]), max(Max[4])]
else:
	Max = [list((j * i for i in b)) for j in a]
	MAX = [max(i) for i in Max]
MAX.sort(reverse=True)
print(MAX[1])
