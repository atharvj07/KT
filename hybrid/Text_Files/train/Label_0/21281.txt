#include <bits/stdc++.h>
using namespace std;
bool dp[5004][5004] = {0};
int val[5004][5004] = {0};
int ans[5005];
int main() {
  string str;
  cin >> str;
  str = "!" + str;
  for (int i = 1; i <= (int)str.length() - 1; i++)
    dp[i][i] = 1, val[i][i] = 1, ans[1]++;
  for (int l = 2; l <= (int)str.length() - 1; l++)
    for (int i = 1; i <= (int)str.length() - 1; i++) {
      int j = i + l - 1;
      if (j > str.length() - 1) break;
      if (l == 2) {
        if (str[i] == str[j]) dp[i][j] = 1;
        if (dp[i][j]) val[i][j] = val[i][i + l / 2 - 1] + 1, ans[val[i][j]]++;
        continue;
      }
      if (str[i] == str[j] && dp[i + 1][j - 1] == 1)
        dp[i][j] = 1;
      else
        dp[i][j] = 0;
      if (dp[i][j]) val[i][j] = val[i][i + l / 2 - 1] + 1, ans[val[i][j]]++;
    }
  for (int i = str.length() - 1; i >= (int)1; i--) ans[i] += ans[i + 1];
  for (int i = 1; i <= (int)str.length() - 1; i++) cout << ans[i] << " ";
  return 0;
}
