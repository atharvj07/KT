#include <bits/stdc++.h>
using namespace std;
long long mod = 1e9 + 7;
long long min(long long a, long long b) { return (a < b) ? a : b; }
long long max(long long a, long long b) { return (a > b) ? a : b; }
long long fp(long long a, long long b) {
  if (b == 0) return 1;
  long long x = fp(a, b / 2);
  x = (x * x) % mod;
  if (b & 1) x = (x * a) % mod;
  return x;
}
auto random_add = [] {
  char *p = new char;
  delete p;
  return uint64_t(p);
};
const uint64_t SEED =
    chrono::steady_clock::now().time_since_epoch().count() * (random_add() | 1);
mt19937_64 rng(SEED);
const long long N = 2e5 + 5;
set<long long> v[N], cur;
vector<long long> vis(N);
void dfs(long long node) {
  vis[node] = 1;
  vector<long long> tmp;
  for (auto i : cur) {
    if (v[node].find(i) == v[node].end()) {
      tmp.push_back(i);
    }
  }
  for (auto i : tmp) cur.erase(i);
  for (auto i : tmp) {
    dfs(i);
  }
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long n, m, x, y;
  cin >> n >> m;
  for (long long i = 1; i <= n; i++) {
    cur.insert(i);
  }
  for (long long i = 1; i <= m; i++) {
    cin >> x >> y;
    v[x].insert(y);
    v[y].insert(x);
  }
  long long ans = 0;
  for (long long i = 1; i <= n; i++) {
    if (vis[i] == 0) {
      cur.erase(i);
      ans++;
      dfs(i);
    }
  }
  cout << ans - 1;
}
