#include <bits/stdc++.h>
int main() {
  int n, a, now, t, ans;
  scanf("%d", &n);
  t = 0;
  now = 0;
  ans = 0;
  for (int i = 0; i < n; i++) {
    scanf("%d", &a);
    if (t == 0 && a == 1)
      t++;
    else if (t == 1 && a == 0)
      t++;
    else if (t == 2 && a == 1) {
      t = 1;
      now++;
    } else {
      ans += (now + 1) / 2;
      t = 0;
      now = 0;
      if (a == 1) t++;
    }
  }
  ans += (now + 1) / 2;
  printf("%d\n", ans);
  return 0;
}
