def main():
	n = int(input())
	(ww, hh) = ([0] * n, [0] * n)
	for i in range(n):
		(w, h) = map(int, input().split())
		(ww[i], hh[i]) = (w, h)
	idx = max(range(n), key=hh.__getitem__)
	mx0 = hh[idx]
	wmx = ww[idx]
	hh[idx] = 1
	mx1 = max(hh)
	width = sum(ww)
	for (i, w) in enumerate(ww):
		ww[i] = (width - ww[i]) * mx0
	ww[idx] = (width - wmx) * mx1
	print(' '.join(map(str, ww)))
main()
