#include <bits/stdc++.h>
long long n, K, ans, F1[1000005], F2[1000005], sum1[1000005], sum2[1000005],
    S1[1000005], S2[1000005], T1[1000005], T2[1000005];
char S[1000005];
long long read() {
  char c = getchar();
  long long ans = 0;
  while (c < '0' || c > '9') c = getchar();
  while (c >= '0' && c <= '9') ans = ans * 10 + c - '0', c = getchar();
  return ans;
}
void Write(long long x) {
  if (x < 10)
    putchar(x ^ 48);
  else
    Write(x / 10), putchar((x % 10) ^ 48);
  return;
}
int main() {
  n = read(), K = read(), scanf("%s", S + 1);
  S1[0] = S2[n + 1] = 1;
  for (int i = 1, s = 0; i <= n; i++) {
    if (S[i] == 'W')
      s = 0;
    else
      ++s;
    if (s >= K)
      F1[i] =
          ((S1[i - K] - T1[i - K] - (sum1[i - 1] - sum1[i - K])) % 1000000007 +
           1000000007) %
          1000000007;
    if (S[i] == 'X')
      S1[i] = S1[i - 1] * 2 % 1000000007;
    else
      S1[i] = S1[i - 1];
    if (S[i] == 'X')
      T1[i] = (T1[i - 1] * 2 + F1[i]) % 1000000007;
    else
      T1[i] = (T1[i - 1] + F1[i]) % 1000000007;
    sum1[i] = (sum1[i - 1] + F1[i]) % 1000000007;
  }
  for (int i = n, s = 0; i; i--) {
    if (S[i] == 'B')
      s = 0;
    else
      ++s;
    if (s >= K)
      F2[i] =
          ((S2[i + K] - T2[i + K] - (sum2[i + 1] - sum2[i + K])) % 1000000007 +
           1000000007) %
          1000000007;
    if (S[i] == 'X')
      S2[i] = S2[i + 1] * 2 % 1000000007;
    else
      S2[i] = S2[i + 1];
    if (S[i] == 'X')
      T2[i] = (T2[i + 1] * 2 + F2[i]) % 1000000007;
    else
      T2[i] = (T2[i + 1] + F2[i]) % 1000000007;
    sum2[i] = (sum2[i + 1] + F2[i]) % 1000000007;
  }
  for (int i = 1; i < n; i++) ans = (ans + F1[i] * T2[i + 1]) % 1000000007;
  return Write(ans), 0;
}
