n = int(input())
a = [int(i) for i in input().split(' ')]
c = [int(i) for i in input().split(' ')]
b = [int(i) for i in input().split(' ')]
d = []
s1 = 0
s2 = sum(c)
z = b[0] + s2
d.append(z)
for i in range(1, n):
	z = b[i]
	s1 += a[i - 1]
	s2 -= c[i - 1]
	z += s1 + s2
	d.append(z)
d.sort()
print(d[0] + d[1])
