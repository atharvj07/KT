q = int(input())
for i in range(q):
	a = input()
	L = 0
	R = 0
	U = 0
	D = 0
	for j in range(len(a)):
		if a[j] == 'L':
			L += 1
		elif a[j] == 'R':
			R += 1
		elif a[j] == 'U':
			U += 1
		else:
			D += 1
	if L == 0 or R == 0:
		if U == 0 or D == 0:
			print(0)
		else:
			print(2)
			print('UD')
	elif U == 0 or D == 0:
		print(2)
		print('LR')
	else:
		x = min(L, R)
		y = min(U, D)
		print((x + y) * 2)
		print(x * 'L' + y * 'U' + x * 'R' + y * 'D')
