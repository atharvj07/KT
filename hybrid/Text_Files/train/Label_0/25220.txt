#include <bits/stdc++.h>
int main(void) {
  int t, i;
  int a, b, c, d, k;
  int p, q, x, y;
  scanf("%d", &t);
  for (i = 0; i < t; i++) {
    scanf(
        "%d"
        "%d"
        "%d"
        "%d"
        "%d",
        &a, &b, &c, &d, &k);
    for (p = 1; p <= a; p++) {
      if ((p * c) >= a) {
        x = p;
        break;
      }
    }
    for (q = 1; q <= b; q++) {
      if ((q * d) >= b) {
        y = q;
        break;
      }
    }
    if ((x + y) <= k) {
      printf("%d %d\n", x, y);
    } else {
      printf("-1\n");
    }
  }
  return 0;
}
