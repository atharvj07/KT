def answer(n, k):
	if n % 4 == 0:
		return ' '.join(map(str, [n // 2, n // 4, n // 4]))
	elif n % 2 == 0:
		return ' '.join(map(str, [n // 2 - 1, n // 2 - 1, 2]))
	else:
		return ' '.join(map(str, [n // 2, n // 2, 1]))

def main():
	t = int(input())
	n = [0] * t
	k = [0] * t
	for j in range(t):
		(n[j], k[j]) = [int(i) for i in input().split()]
	for j in range(t):
		print(answer(n[j], k[j]))
main()
