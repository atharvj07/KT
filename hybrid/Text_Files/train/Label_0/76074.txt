import math

def main():
	n = int(input())
	a = int(input())
	b = int(input())
	if n % math.gcd(a, b):
		print('NO')
	else:
		flag = 0
		if a < b:
			(a, b) = (b, a)
			flag = 1
		for i in range(1 + n // a):
			if (n - i * a) % b == 0:
				print('YES')
				if flag:
					print((n - i * a) // b, i)
				else:
					print(i, (n - i * a) // b)
				return
		print('NO')
main()
