f = 200001
v = [0] * f
g = [[] for _ in [0] * f]
s = [1]
(n, m, k) = map(int, input().split())
for _ in range(m):
	(a, b) = map(int, input().split())
	g[b] += (a,)
	g[a] += (b,)
while 1:
	v[s[-1]] = 1
	for i in g[s[-1]]:
		if v[i] < 1:
			s += (i,)
			break
	else:
		t = set(g[s[-1]])
		for i in range(len(s)):
			if s[i] in t:
				exit(print(len(s) - i, *s[i:]))
