#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 111;
const int P = 1111;
int n, k, a;
bool B[2 * P], ok = false;
set<int> St;
set<int>::iterator it;
void BFS() {
  queue<pair<int, int> > Qu;
  memset(B, 1, sizeof(B));
  Qu.push(make_pair(0, 0));
  while (!Qu.empty()) {
    int u = Qu.front().first;
    int d = Qu.front().second;
    Qu.pop();
    if (u == 0 && !B[P + u]) {
      cout << d;
      ok = true;
      return;
    }
    for (it = St.begin(); it != St.end(); it++) {
      int v = u + *it;
      if (-1000 <= v && v <= 1000 && B[P + v]) {
        Qu.push(make_pair(v, d + 1));
        B[P + v] = false;
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= k; i++) {
    cin >> a;
    St.insert(a - n);
  }
  BFS();
  if (!ok) cout << -1;
  return 0;
}
