#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

const double g = 9.8;
const double EPS = 1.0e-6;

void kouhoth(int x, int y, int v, vector<double> &allTH)
{
	double K = 1.0*v*v/g;
	double d = K*K*x*x - x*x*(x*x+2*K*y);
	if(d<0) return;
	double kai1 = (K*x + sqrt(d)) / x / x;
	double kai2 = (K*x - sqrt(d)) / x / x;
	if(kai1+EPS>0) allTH.push_back(atan(kai1+EPS));
	if(kai1-EPS>0) allTH.push_back(atan(kai1-EPS));
	if(kai2+EPS>0) allTH.push_back(atan(kai2+EPS));
	if(kai2-EPS>0) allTH.push_back(atan(kai2-EPS));
}

double f(double th, int v, int x)
{
	double K = 1.0*v*v/g;
	double si = sin(th);
	double co = cos(th);
	double ta = tan(th);
	double alpha = -0.5*(1.0+ta*ta)/K;
	double beta = 0.5*K*si*si;
	double naka = 1.0*x - K*si*co;
	return alpha * naka*naka + beta;
}

int main(){
	int n;
	int v, x, y;
	pair<double, double> buf;
	while(cin>>n>>v>>x>>y){
		vector<int> allL, allB, allR, allT;
		vector<double> allTH;
		for(int i=0; i<n; i++){
			int l, b, r, t;
			cin>>l>>b>>r>>t;
			// ³Ê
			if(x<l) continue;
			// ³ÊÈª»
			if(x<r) r = x;
			allL.push_back(l);
			allB.push_back(b);
			allR.push_back(r);
			allT.push_back(t);
			kouhoth(l, b, v, allTH);
			kouhoth(r, b, v, allTH);
			kouhoth(l, t, v, allTH);
			kouhoth(r, t, v, allTH);
		}
		kouhoth(x, y, v, allTH);

		bool okflg = false;
		for(int i=0; i<allTH.size(); i++){
			double th = allTH[i];
			double mauey = f(th,v,x);
			if(mauey < y) continue;
			bool dameflg = false;
			for(int j=0; j<allL.size(); j++){
				if(
					(f(th,v,allL[j])<allB[j]&&f(th,v,allR[j])<allB[j]) ||
					(f(th,v,allL[j])>allT[j]&&f(th,v,allR[j])>allT[j]) ){
						continue;
				} else {
					dameflg = true;
					break;
				}
			}

			if(!dameflg){
				bool butukaruflg = false;
				for(int j=0; j<allL.size(); j++){
					if(allR[j]==x){
						if(y<allT[j] && allT[j]<mauey){
							butukaruflg = true;
							break;
						}
					}
				}
				if(!butukaruflg){
					okflg = true;
					break;
				}
			}
		}
		if(okflg) cout<<"Yes"<<endl;
		else cout<<"No"<<endl;
	}
}
