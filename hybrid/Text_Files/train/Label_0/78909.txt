n = int(input())
d = list(map(int, input().split()))
uk1 = -1
uk2 = n
ans = []
suma = 0
sumb = 0
while uk2 - uk1 > 1:
	if suma == sumb:
		ans.append(suma)
		uk1 += 1
		suma += d[uk1]
	elif suma < sumb:
		uk1 += 1
		suma += d[uk1]
	else:
		uk2 -= 1
		sumb += d[uk2]
if suma == sumb:
	ans.append(suma)
print(ans[-1])
