#include<stdio.h>

using namespace std;

struct Node{
  int im;
  Node *ri,*le,*pa;
};

Node *ne,*Ni;

Node * tmi(Node *A){
  while(A->le != Ni){
    A = A->le;
  }
  return A;
}

Node * mik(Node *x,int z){
  while( x != Ni && z != x->im){
    if(z<x->im){
      x=x->le;
    }
    else x = x->ri;
  }
  return x;
}

Node * ts(Node *A){
  if(A->ri != Ni){
    return tmi(A->ri);
  }
    Node *B = A->pa;
    while(B !=Ni && A == B->ri){
      A= B;
      B=B->pa;
    }
    return B;
  }

  void td(Node *K){
    Node *B;
    Node *A;

    if(K->le == Ni || K->ri ==Ni){
      B=K;
    }
    else B = ts(K);

    if(B->le != Ni){
      A = B->le;
    }
    else{
      A = B->ri;
    }

    if(A !=Ni){
      A ->pa = B->pa;
    }
    if(B->pa == Ni){
      ne = A;
    }
    else{
      if(B==B->pa->le){
	B->pa->le=A;
      }else{
	B->pa->ri=A;
      }
    }
    if(B!=K){
      K->im = B->im;
    }
    free(B);
  }

void na(int O){

  Node *B = Ni;
  Node *A = ne;
  Node *C;

  C = (Node *)malloc(sizeof(Node));
  C->im = O;
  C->le = Ni;
  C->ri = Ni;

  while(A!=Ni){
    B =A;
    if(C->im < A->im){
      A = A->le;
    }
    else{
      A= A->ri;
    }
  }
  C->pa = B;
  if(B==Ni){
    ne =C;
  }
  else{
    if(C->im < B->im){
      B->le =C;
    }
    else{
      B->ri = C;
    }
  }

}

void io(Node *A){
  if(A == Ni){
    return;
  }
  io(A->le);
  printf(" %d",A->im);
  io(A->ri);
}

void po(Node *A){
  if(A == Ni){
    return;
  }
  printf(" %d",A->im);
  po(A->le);
  po(A->ri);
}
