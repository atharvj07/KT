#include <bits/stdc++.h>
using namespace std;
long long k, a, b;
long long al[4][4], bo[4][4], pts1[100005], pts2[100005];
map<pair<long long, long long>, long long> ind;
long long fp, sp, pt1[100005], pt2[100005];
int main() {
  ios_base::sync_with_stdio(false);
  long long precision = numeric_limits<double>::digits10;
  cin >> k >> a >> b;
  for (long long i = 1; i <= 3; i++) {
    for (long long j = 1; j <= 3; j++) cin >> al[i][j];
  }
  for (long long i = 1; i <= 3; i++) {
    for (long long j = 1; j <= 3; j++) cin >> bo[i][j];
  }
  long long staInd = -1, per = -1, f = a, s = b;
  long long in = 0;
  for (long long i = 1; i <= k; i++) {
    if (ind.count(make_pair(f, s)) == 0) {
      ind[make_pair(f, s)] = i;
      if (f != s) {
        long long MAX = max(f, s);
        if (abs(f - s) == 1) {
          if (MAX == f)
            pts1[i] += 1;
          else
            pts2[i] += 1;
        } else {
          if (MAX == f)
            pts2[i] += 1;
          else
            pts1[i] += 1;
        }
      }
      pts1[i] += pts1[i - 1];
      pts2[i] += pts2[i - 1];
      long long u, v;
      u = al[f][s];
      v = bo[f][s];
      f = u;
      s = v;
    } else {
      staInd = ind[make_pair(f, s)];
      per = i - staInd;
      break;
    }
    in++;
  }
  if (in == k) {
    cout << pts1[k] << " " << pts2[k] << endl;
    return 0;
  }
  fp += pts1[min(staInd - 1, k)];
  sp += pts2[min(staInd - 1, k)];
  if (k >= staInd) {
    for (long long i = 1; i <= per; i++)
      pt1[i] = pts1[staInd + i - 1] - pts1[staInd - 1];
    for (long long i = 1; i <= per; i++)
      pt2[i] = pts2[staInd + i - 1] - pts2[staInd - 1];
    long long q = (k - staInd + 1) / per, r = (k - staInd + 1) % per;
    fp += q * pt1[per];
    sp += q * pt2[per];
    fp += pt1[r];
    sp += pt2[r];
  }
  cout << fp << " " << sp << endl;
}
