for _ in range(int(input())):
	N = int(input())
	A = list(map(int, input().split()))
	if N == 1:
		print(0)
	else:
		dic = {}
		for i in A:
			if i in dic:
				dic[i] += 1
			else:
				dic[i] = 1
		l = max(dic.values())
		r = len(dic.keys())
		if l == r:
			print(r - 1)
		if r > l:
			print(l)
		if r < l:
			print(r)
