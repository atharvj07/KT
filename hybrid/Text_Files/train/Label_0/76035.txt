import math
t = int(input())
while t:
	(n, k) = input().split()
	n = int(n)
	k = int(k)
	data = [int(x) for x in input().split()]
	sum = 0
	if n == 1:
		for ele in data:
			sum += ele
		print(sum)
	elif n == 2:
		i = 0
		while i < k * n:
			sum += data[i]
			i += 2
		print(sum)
	else:
		i = n * k - (n // 2 + 1)
		while i >= k * (n - (n // 2 + 1)):
			sum += data[i]
			i = i - (n // 2 + 1)
		print(sum)
	t -= 1
