#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC optimize("no-stack-protector")
using namespace std;
const long long INF = 1e18;
const long long MAX_N = 200000 + 228;
long long si = 16;
pair<long long, long long> elem[MAX_N];
vector<long long> vec[MAX_N], vec2[MAX_N];
deque<long long> deq;
set<long long> num;
multiset<long long> now;
map<long long, vector<long long>> map1;
long long ans, sum;
long long a, b, c, d, e, f, p = 1000000007;
long long x, y;
signed main() {
  cin >> a;
  for (long long i = 0; i < a; i++) {
    cin >> c;
    elem[i].first = c;
    num.insert(c);
  }
  for (long long i = 0; i < a; i++) {
    cin >> d;
    elem[i].second = d;
    map1[elem[i].first].push_back(d);
  }
  while (num.size() > 0) {
    auto it = num.begin();
    for (auto i : map1[(*it)]) {
      now.insert(i);
      sum += i;
    }
    if (now.size() == 1) {
      now.clear();
      sum = 0;
    } else {
      auto it2 = now.end();
      it2--;
      sum -= (*it2);
      ans += sum;
      now.erase(it2);
      num.insert((*it) + 1);
    }
    num.erase((*it));
  }
  cout << ans;
  cin >> b;
  return 0;
}
