const int MX = 1;
const int MM = 1000000007;

vector<pii> L;
int main()
{
	ll A, B, C, D;
	scanf("%lld%lld%lld%lld", &A, &B, &C, &D);
	for(int i = 1; i <= D+5; i++){
		if(1 <= i && i < A) L.push_back(pii(i, A));
		if(1 <= i && i < B) L.push_back(pii(i, B));
		if(1 <= i && i < C) L.push_back(pii(i, C));
		
		if(1 <= i && i < A) L.push_back(pii(A-i, A));
		if(1 <= i && i < B) L.push_back(pii(B-i, B));
		if(1 <= i && i < C) L.push_back(pii(C-i, C));
	}
	int sz = L.size();
	for(int i = 0; i < sz; i++){
		pii c = L[i];
		ll v = c.first * A / c.second;
		if(1 <= v && v < A ) L.push_back(pii(v, A));
		if(1 <= v+1 && v+1 < A ) L.push_back(pii(v+1, A));

		v = c.first * B / c.second;
		if(1 <= v && v < B ) L.push_back(pii(v, B));
		if(1 <= v+1 && v+1 < B ) L.push_back(pii(v+1, B));

		v = c.first * C / c.second;
		if(1 <= v && v < C ) L.push_back(pii(v, C));
		if(1 <= v+1 && v+1 < C ) L.push_back(pii(v+1, C));
	}

	sort(L.begin(), L.end(), [](pii l, pii r){
		return (ll)l.first * r.second < (ll)r.first * l.second;
	});

	L.resize(unique(L.begin(), L.end()) - L.begin());

	ll x = 0, y = 0, z = 0;
	ll ans = (ll)min(D+1, A) * min(D+1, B) * min(D+1, C) % MM;
	for(pii c : L){
		ll dx = min(A-1, x+D) - max(0ll, x-D) + 1;
		ll dy = min(B-1, y+D) - max(0ll, y-D) + 1;
		ll dz = min(C-1, z+D) - max(0ll, z-D) + 1;
		if(c.second == A){
			ans += dy*dz%MM * max(0ll, min((ll)c.first, A-D-1) - x);
			x = c.first;
		}
		if(c.second == B){
			ans += dx*dz%MM * max(0ll, min((ll)c.first, B-D-1) - y);
			y = c.first;
		}
		if(c.second == C){
			ans += dx*dy%MM * max(0ll, min((ll)c.first, C-D-1) - z);
			z = c.first;
		}
		ans %= MM;
	}
	printf("%lld\n", ans);
}
