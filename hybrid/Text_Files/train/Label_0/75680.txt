from sys import stdin

def main():
	n = int(stdin.readline().strip())
	x = stdin_get_ints_list_from_line()
	q = int(stdin.readline().strip())
	x.sort()
	for _ in range(q):
		m = int(stdin.readline().strip())
		print(find_nearest_left(x, m))

def find_nearest_left(array, value):
	left = 0
	right = len(array) - 1
	if value < array[0]:
		return 0
	elif value >= array[len(array) - 1]:
		return len(array)
	middle = int((left + right) / 2)
	i = 0
	while True:
		if array[middle] <= value < array[middle + 1]:
			break
		if array[middle] > value:
			right = middle
		elif array[middle] <= value:
			left = middle
		middle = int((left + right) / 2)
		i += 1
	return middle + 1

def stdin_get_ints_from_line():
	return (int(x) for x in stdin.readline().strip().split(' '))

def stdin_get_ints_list_from_line():
	return list((int(x) for x in stdin.readline().strip().split(' ')))

def stdin_get_string_from_line():
	return stdin.readline().strip()
main()
