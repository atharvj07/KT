#include <bits/stdc++.h>
using namespace std;
long long dp[1005][2005][5];
long long n, k;
void rset() {
  for (long long i = 0; i < 1005; i++) {
    for (long long j = 0; j < 2005; j++) {
      for (long long k = 0; k < 5; k++) dp[i][j][k] = -1;
    }
  }
}
long long func(long long pos, long long taken, long long pat) {
  if (pos > n) {
    if (taken == k) return 1;
    return 0;
  }
  if (dp[pos][taken][pat] != -1) return dp[pos][taken][pat];
  long long ret = 0;
  if (pat == 0) {
    ret += func(pos + 1, taken, 0);
    ret %= 998244353;
    ret += func(pos + 1, taken + 1, 1);
    ret %= 998244353;
    ret += func(pos + 1, taken + 1, 2);
    ret %= 998244353;
    ret += func(pos + 1, taken + 1, 3);
    ret %= 998244353;
  }
  if (pat == 1) {
    ret += func(pos + 1, taken, 0);
    ret %= 998244353;
    ret += func(pos + 1, taken, 1);
    ret %= 998244353;
    ret += func(pos + 1, taken + 2, 2);
    ret %= 998244353;
    ret += func(pos + 1, taken, 3);
    ret %= 998244353;
  }
  if (pat == 2) {
    ret += func(pos + 1, taken, 0);
    ret %= 998244353;
    ret += func(pos + 1, taken + 2, 1);
    ret %= 998244353;
    ret += func(pos + 1, taken, 2);
    ret %= 998244353;
    ret += func(pos + 1, taken, 3);
    ret %= 998244353;
  }
  if (pat == 3) {
    ret += func(pos + 1, taken + 1, 0);
    ret %= 998244353;
    ret += func(pos + 1, taken + 1, 1);
    ret %= 998244353;
    ret += func(pos + 1, taken + 1, 2);
    ret %= 998244353;
    ret += func(pos + 1, taken, 3);
    ret %= 998244353;
  }
  return dp[pos][taken][pat] = ret;
}
int main() {
  rset();
  cin >> n >> k;
  long long ans = 0;
  ans += func(2, 1, 0);
  ans += func(2, 2, 1);
  ans += func(2, 2, 2);
  ans += func(2, 1, 3);
  ans %= 998244353;
  cout << ans << endl;
  return 0;
}
