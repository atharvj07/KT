n = int(input())
animals = list(map(int, input().split()))
copy = list(animals)
copy.sort()
moves = []
while True:
	for i in range(n - 1):
		if animals[i] > animals[i + 1]:
			temp = animals[i]
			animals[i] = animals[i + 1]
			animals[i + 1] = temp
			moves.append(str(i + 1) + ' ' + str(i + 2))
	if animals == copy:
		break
for i in moves:
	print(i)
