#include <cstdio>
#include <cstring>
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 200000;

int n, co = 0;
int fir[N + 5], nex[N * 2 + 5], vv[N * 2 + 5];
int sz[N + 5];
bool fl[N * 2 + 5] = {0};//�߱�� 
long long ff[N + 5];
int t1, t2;

void dfs1(int o, int fa)
{
	sz[o] = 1;
	for (int i = fir[o]; i; i = nex[i])
	{
		if (vv[i] == fa || fl[i])
		{
			continue;
		}
		dfs1(vv[i], o);
		sz[o] += sz[vv[i]];
	}
}

bool dfs(int o, int fa, int c)
{
	if (sz[o] == ff[c - 1] || sz[o] == ff[c - 2])
	{
		t1 = o;
		t2 = fa;
		for (int i = fir[o]; i; i = nex[i])
		{
			if (vv[i] == fa)
			{
				fl[i % n] = true;
				fl[i % n + n] = true;
				break;
			}
		}
		return true;
	}
	for (int i = fir[o]; i; i = nex[i])
	{
		if (vv[i] == fa || fl[i])
		{
			continue;
		}
		if (dfs(vv[i], o, c))
		{
			return true;
		}
	}
	return false;
}

bool search1(int t, int o)
{
	if (ff[t] <= 3)
	{
		return true;
	}
	dfs1(o, 0);
	if (dfs(o, 0, t))
	{
		int tmp1 = t1, tmp2 = t2;
		if (sz[tmp1] == ff[t - 1])
		{
			return search1(t - 1, tmp1) && search1(t - 2, tmp2);
		}
		else
		{
			return search1(t - 2, tmp1) && search1(t - 1, tmp2);
		}
	}
	else
	{
		return false;
	}
}

int main()
{
	ff[1] = ff[0] = 1;
	scanf("%d", &n);
	for (int i = 2; ; ++i)
	{
		ff[i] = ff[i - 1] + ff[i - 2];
		if (ff[i] >= n)
		{
			co = i;
			break;
		}
	}
	for (int i = 1; i < n; ++i)
	{
		scanf("%d%d", &vv[i + n], &vv[i]);
		nex[i] = fir[vv[i + n]];
		fir[vv[i + n]] = i;
		nex[i + n] = fir[vv[i]];
		fir[vv[i]] = i + n;
	}
	if (n <= 3)
	{
		printf("YES\n");
		return 0;
	}
	if (n != ff[co])
	{
		printf("NO\n");
		return 0;
	}
	bool cc = search1(co, 1);
	if (cc)
	{
		printf("YES\n");
	}
	else
	{
		printf("NO\n");
	}
	return 0;
}
