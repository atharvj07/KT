#include <bits/stdc++.h>
const double pi = cos(-1.);
const double eps = 10e-6;
const double eps1 = 10e-10;
const int inf = 0x7fffffff;
const long long infl = 1ll << 62;
const double infd = 20000000000.0;
using namespace std;
int main() {
  double a, b, m;
  double vx, vy, vz;
  while (cin >> a >> b >> m >> vx >> vy >> vz) {
    double px = a / 2.0, py = m, pz = 0.0;
    double tl = -m / vy;
    double t = 0.0, tt;
    if (vx < eps1 && vx > -eps1 && vz < eps1 && vz > -eps1) {
      cout << px << " " << pz << endl;
      continue;
    }
    while (true) {
      double t1 = infd, t2 = infd;
      if (vx > eps1) t1 = (a - px) / vx;
      if (vx < -eps1) t1 = -px / vx;
      if (vz > eps1) t2 = (b - pz) / vz;
      if (vz < -eps1) t2 = -pz / vz;
      tt = min(t1, t2);
      t += tt;
      if (t < tl - eps) {
        px += tt * vx;
        pz += tt * vz;
        if (t1 < t2 + eps1) vx = -vx;
        if (t2 < t1 + eps1) vz = -vz;
      } else {
        tt = tl - t + tt;
        px += tt * vx;
        pz += tt * vz;
        break;
      }
    }
    printf("%.10lf %.10lf\n", px, pz);
  }
  return 0;
}
