from sys import stdin

def input():
	return stdin.readline()[:-1]
for _ in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	idx = [0] * (n + 1)
	for i in range(n):
		idx[l[i]] = i
	maxs = [idx[1]] * (n + 1)
	mins = [idx[1]] * (n + 1)
	for i in range(2, n + 1):
		maxs[i] = max(maxs[i - 1], idx[i])
		mins[i] = min(mins[i - 1], idx[i])
	out = []
	for i in range(1, n + 1):
		if maxs[i] - mins[i] == i - 1:
			out.append(1)
		else:
			out.append(0)
	print(''.join(list(map(str, out))))
