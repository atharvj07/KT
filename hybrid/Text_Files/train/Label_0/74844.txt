q = int(input())
for w in range(q):
	n = int(input())
	arr = list(map(int, input().strip().split()))
	flag = [False for i in range(n - 1)]
	k = -1
	while k < n - 1:
		arr1 = arr[k + 1:]
		m = min(arr1)
		k1 = arr.index(m)
		if k1 != k + 1:
			for j in range(k1, k + 1, -1):
				(arr[j], arr[j - 1]) = (arr[j - 1], arr[j])
				flag[j - 1] = True
			k = k1
		else:
			k += 1
	for i in range(n - 1):
		if not flag[i]:
			if arr[i] > arr[i + 1]:
				flag[i] = True
				(arr[i], arr[i + 1]) = (arr[i + 1], arr[i])
	print(*arr)
