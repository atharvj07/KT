public class A {
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        //k is even sg[0] = 0 sg[1]= 1 sg[2] = 2 sg[i]= 1(i is even) 0(i is odd)
        //k is odd  sg[0] = 0 sg[1] = 1 sg[2] = 0 sg[3] = 1  sg[6]=2   odd: 0  2^2k: 2  2^2k+1:1
        int n = in.nextInt();
        int k = in.nextInt();
        int sg = 0;
        for (int i = 0; i < n; i++) {
            int x = in.nextInt();
            sg = sg^getSG(x,k);
        }
        if (sg != 0) System.out.println("Kevin");
        else System.out.println("Nicky");
    }
    private static int getSG(int x, int k) {
        if (x == 0) return 0;
        if (k%2 == 0) {
            if (x <= 2) return x;
            else if (x % 2 == 1) return 0;
            else return 1;
        } else {
            if (x == 1 || x == 3) return 1;
            else if (x == 2) return 0;
            int cnt=0;
            while ((x&1) == 0) {
                x = x >> 1;
                cnt++;
            }
            if (cnt == 0) return 0;
            else if ((cnt&1) == 0) {
                if (x == 3) return 1;
                else return 2;
            }
            else{
                if (x == 3) return 2;
                else return 1;
            }
        }
    }
}
