abc = ['a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
num = {}
for i in abc:
	num[i] = 0
(n, k) = map(int, input().split())
s = input()
for i in s:
	if i in abc:
		num[i] += 1
mx = max(num.values())
if mx > k:
	print('NO')
else:
	print('YES')
