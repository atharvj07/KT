fib = []
summ = []
mod = 1000000007

def pre_calc(k):
	for i in range(100001):
		summ.append(0)
		fib.append(0)
	fib[0] = 1
	for i in range(1, 100001):
		fib[i] = fib[i - 1]
		if i >= k:
			fib[i] = (fib[i] % mod + fib[i - k] % mod) % mod
	for i in range(1, 100001):
		summ[i] = (summ[i - 1] % mod + fib[i] % mod) % mod

def solve():
	(n, k) = input().split()
	n = int(n)
	k = int(k)
	pre_calc(k)
	for i in range(n):
		(l, r) = input().split()
		l = int(l)
		r = int(r)
		ans = summ[r] - summ[l - 1]
		while ans < 0:
			ans += mod
		print(ans)
solve()
