#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int& x : a) cin >> x;
    for (int& x : b) cin >> x;
    long long ans = 0;
    int end_a = a[0], end_b = b[0];
    for (int i = 0; i < n; ++i) {
      end_a = min(end_a, a[i]);
      end_b = min(end_b, b[i]);
    }
    for (int i = 0; i < n; ++i) {
      int change_a = a[i] - end_a, change_b = b[i] - end_b;
      ans += max(change_a, change_b);
    }
    cout << ans << endl;
  }
  return 0;
}
