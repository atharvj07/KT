import sys
input = sys.stdin.readline

def foo(x, y, k, a, b):
	if k < 0:
		return -1
	if k * b > x or k * a > y:
		return -1
	if (x - k * b) // a >= (y - k * a) // b:
		return 1
	else:
		return -1

def bsearch(l, r, x, y, a, b):
	while l < r:
		m = (l + r) // 2
		if foo(x, y, m, a, b) == 1:
			l = m + 1
		else:
			r = m
	return l - 1

def f(x, y, a, b, k):
	if k < 0 or x < k * b or y < k * a:
		return -1
	return k + min((x - k * b) // a, (y - k * a) // b)
t = int(input())
for you in range(t):
	l = input().split()
	x = int(l[0])
	y = int(l[1])
	a = int(l[2])
	b = int(l[3])
	if a > b:
		(a, b) = (b, a)
	pp = bsearch(0, 10 ** 10, x, y, a, b)
	ans = max(f(x, y, a, b, pp), f(x, y, a, b, pp + 1))
	pp = bsearch(0, 10 ** 10, y, x, a, b)
	ans = max(f(y, x, a, b, pp), f(y, x, a, b, pp + 1), ans)
	print(ans)
