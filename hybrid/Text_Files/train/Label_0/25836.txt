#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cout << fixed << setprecision(9);
  vector<double> powers(1, 1.0);
  for (int i = 0; i < 10; ++i) powers.push_back(powers[i] * 2.0);
  int N, teams, rounds, c;
  cin >> N;
  teams = powers[N];
  rounds = N + 1;
  double p[rounds][teams], S[rounds][teams], w[teams][teams];
  memset(p, 0.0, rounds * teams * sizeof(double));
  memset(S, 0.0, rounds * teams * sizeof(double));
  for (int i = 0; i < teams; ++i) {
    for (int j = 0; j < teams; ++j) {
      cin >> c;
      w[i][j] = c / 100.0;
    }
  }
  for (int i = 0; i < teams; ++i) {
    p[N][i] = 1.0;
    S[N][i] = 0.0;
  }
  for (int r = N - 1; r >= 0; --r) {
    for (int m = 0; m < powers[r]; ++m) {
      for (int a = m * powers[N - r]; a < (2 * m + 1) * powers[N - r - 1];
           ++a) {
        for (int b = (2 * m + 1) * powers[N - r - 1];
             b < (m + 1) * powers[N - r]; ++b) {
          p[r][a] += p[r + 1][b] * w[a][b];
          p[r][b] += p[r + 1][a] * w[b][a];
        }
      }
      for (int t = m * powers[N - r]; t < (m + 1) * powers[N - r]; ++t) {
        p[r][t] *= p[r + 1][t];
      }
      for (int a = m * powers[N - r]; a < (2 * m + 1) * powers[N - r - 1];
           ++a) {
        for (int b = (2 * m + 1) * powers[N - r - 1];
             b < (m + 1) * powers[N - r]; ++b) {
          S[r][a] = max(
              S[r][a], S[r + 1][a] + S[r + 1][b] + p[r][a] * powers[N - r - 1]);
          S[r][b] = max(
              S[r][b], S[r + 1][b] + S[r + 1][a] + p[r][b] * powers[N - r - 1]);
        }
      }
    }
  }
  double res = 0;
  for (int i = 0; i < teams; ++i) {
    res = max(res, S[0][i]);
  }
  cout << res << endl;
  return 0;
}
