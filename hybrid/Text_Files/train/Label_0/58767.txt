import java.util.*;
public class Main
{
	public static void main(String[] args)
	{
		Scanner sc = new Scanner(System.in);
		int s = sc.nextInt();
		int n = sc.nextInt();
		
		int[] reqS = new int[n];
		int[] gainS = new int[n];
		int ctr = 1;

		
		for(int i = 0; i < n; i++)
		{
			reqS[i] = sc.nextInt();
			gainS[i] = sc.nextInt();
			
		}
		
		while(ctr != 0)
		{
			ctr = 0;
			for(int i = 0; i < n; i++)
			{
				if(reqS[i] < s)
				{
					s += gainS[i];
					reqS[i] = 10000000;
					ctr++;
				}
			
			}
		}
		
		if(win(reqS))
		{
			System.out.println("YES");
		}
		else
		{
			System.out.println("NO");
		}
	}
	static boolean win(int[] x)
	{
		for(int i = 0; i < x.length; i++)
		{
			if(x[i] != 10000000)
			{
				return false;
			}
		}
		
		return true;
	}
}
