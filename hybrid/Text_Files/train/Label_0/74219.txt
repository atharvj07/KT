n = int(input())
s = input()
m = input()
scount = [0] * 10
mcount = [0] * 10

def get(i):
	return int(i) - 0
for i in range(0, n):
	scount[get(s[i])] += 1
	mcount[get(m[i])] += 1
cmcount = [i for i in mcount]
for i in range(0, 9):
	if scount[i] > mcount[i]:
		mcount[i + 1] -= scount[i] - mcount[i]
print(scount[9] - mcount[9])
a = cmcount[0]
max = 0
for i in range(9, 0, -1):
	if cmcount[i] > scount[i - 1]:
		cmcount[i - 1] += cmcount[i] - scount[i - 1]
		max += scount[i - 1]
	else:
		max += cmcount[i]
print(max)
