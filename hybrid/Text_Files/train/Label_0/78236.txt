n = int(input())
s = input()
l = list(s)
i = 0
while i < len(l) - 1:
	if l[i] == 'a' or l[i] == 'e' or l[i] == 'i' or (l[i] == 'o') or (l[i] == 'u') or (l[i] == 'y'):
		if l[i + 1] == 'a' or l[i + 1] == 'e' or l[i + 1] == 'i' or (l[i + 1] == 'o') or (l[i + 1] == 'u') or (l[i + 1] == 'y'):
			l.pop(i + 1)
			i -= 1
	i += 1
s = ''.join(l)
print(s)
