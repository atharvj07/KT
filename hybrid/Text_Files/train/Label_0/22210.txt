#include <bits/stdc++.h>
using namespace std;
int num[200002];
int main() {
  int n, i, j, k, a = 0, b = 0;
  long long ans = 0;
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d", num + i);
  }
  num[0] = 2;
  num[n + 1] = 2;
  i = 1;
  j = n;
  while (i != n + 1 && num[i] == 0) {
    num[i++] = 2;
  }
  while (j != 0 && num[j] == 1) {
    num[j--] = 2;
  }
  while (i != n + 1 && j != 0) {
    while (j != 0) {
      if (num[j] == 0)
        b++, j--;
      else if (num[j] == 2)
        j--;
      else
        break;
    }
    while (i != n + 1) {
      if (num[i] == 1)
        a++, i++;
      else if (num[i] == 2)
        i++;
      else
        break;
    }
    if (i == n + 1) {
      break;
    } else if (j == 0) {
      break;
    } else if (i < j) {
      if (a < b) {
        num[i] = 2;
        ans += a;
        i++;
      } else {
        num[j] = 2;
        ans += b;
        j--;
      }
    } else {
      if (a < b) {
        num[i] = 2;
        ans += a;
        b--;
        i++;
      } else {
        num[j] = 2;
        ans += b;
        a--;
        j--;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
