#include <bits/stdc++.h>
using namespace std;
pair<int, int> a[111111];
int n;
long long k, sum;
inline bool cmp(const pair<int, int>& a, const pair<int, int>& b) {
  return a.second < b.second;
}
int main() {
  scanf("%d%I64d", &n, &k);
  for (int(i) = 0; (i) < (n); ++(i)) {
    scanf("%d", &a[i].first);
    sum += (long long)a[i].first;
    a[i].second = i;
  }
  sort(a, a + n);
  if (sum < k) {
    cout << "-1\n";
    return 0;
  }
  int pos = 0;
  int all = 0;
  while (pos < n && (long long)(a[pos].first - all) * (n - pos) <= k) {
    k -= (long long)(a[pos].first - all) * (n - pos);
    all = a[pos].first;
    a[pos].first = 0;
    ++pos;
  }
  if (pos == n) {
    cout << "\n";
    return 0;
  }
  int times = (int)(k / (n - pos));
  int also = (int)(k % (n - pos));
  sort(a, a + n, cmp);
  int who = -1;
  for (int(j) = 0; (j) < (n); ++(j))
    if (a[j].first) {
      a[j].first -= all;
      a[j].first -= times;
      if (also) {
        a[j].first--;
        also--;
      } else if (who == -1)
        who = j;
    }
  rotate(a, a + who, a + n);
  bool fst = 1;
  for (int(i) = 0; (i) < (n); ++(i))
    if (a[i].first) {
      if (fst)
        fst = 0;
      else
        printf(" ");
      printf("%d", a[i].second + 1);
    }
  printf("\n");
}
