#include <cmath>
#include <iostream>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n;
	cin>>n;
	long long ans=(long long)n; 
	int m=(int)floor(sqrt(n)+0.5);
	for(int i=2; i<=m; i++) {
		if(n%i==0) {
			ans = (long long)ans * (i - 1) / i;
			while(n%i==0) n/=i;
		}
	}
	if(n>1) ans=ans/n*(n-1);
	cout<<ans<<'\n';
	return 0;
}
