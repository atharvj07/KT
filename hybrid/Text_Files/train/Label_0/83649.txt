import sys

def main():
	(n, m) = map(int, sys.stdin.readline().split())
	grid = [[int(x) for x in sys.stdin.readline().strip()] for _ in range(n)]
	res = [0 for _ in range(m)]
	for j in range(m):
		res[j] = sum([grid[i][j] for i in range(n)])
	for i in range(n):
		ok = True
		for j in range(m):
			if res[j] == 1 and grid[i][j]:
				ok = False
		if ok:
			print('YES')
			return
	print('NO')
main()
