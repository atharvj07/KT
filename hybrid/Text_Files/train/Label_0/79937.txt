from collections import deque
t = int(input())
while t:
	l = list(map(int, input().strip()))
	e = []
	o = []
	for i in l:
		if i % 2:
			o.append(i)
		else:
			e.append(i)
	i = 0
	j = 0
	res = []
	while i < len(e) and j < len(o):
		if e[i] < o[j]:
			res.append(e[i])
			i += 1
		else:
			res.append(o[j])
			j += 1
	while i < len(e):
		res.append(e[i])
		i += 1
	while j < len(o):
		res.append(o[j])
		j += 1
	for i in res:
		print(i, end='')
	print()
	t -= 1
