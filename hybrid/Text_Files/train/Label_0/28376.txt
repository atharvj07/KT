#include <bits/stdc++.h>
inline int readin() {
  int x = 0, c;
  for (; (c = getchar()) < '0' || c > '9';)
    ;
  for (; c >= '0' && c <= '9'; c = getchar()) x = (x << 1) + (x << 3) + c - 48;
  return x;
}
long long gcd(long long a, long long b) { return b ? gcd(b, a % b) : a; }
int main() {
  int n = readin();
  int p = readin(), q = readin();
  long long x = p * 1ll * q;
  for (int i = 1; i < n; i++) x = gcd(x, readin() * 1ll * readin());
  if (x == 1) {
    puts("-1");
    return 0;
  }
  int tmp = (int)gcd(x, 1ll * p);
  if (tmp == 1)
    x = gcd(x, q);
  else
    x = tmp;
  tmp = sqrt(x);
  for (int i = 2; i <= tmp; ++i)
    if (x % i == 0) {
      printf("%d\n", i);
      return 0;
    }
  printf("%d\n", x);
}
