#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long int _;
  cin >> _;
  while (_--) {
    long long int n, m, i, j = 0, fl = 0;
    cin >> n >> m;
    long long int gr[n][m];
    for (i = 0; i < n; i++) {
      for (j = 0; j < m; j++) {
        cin >> gr[i][j];
        if (i == 0 || i == n - 1) {
          if (j == 0 || j == m - 1) {
            if (gr[i][j] > 2)
              fl = 1;
            else
              gr[i][j] = 2;
          } else {
            if (gr[i][j] > 3)
              fl = 1;
            else
              gr[i][j] = 3;
          }
        } else {
          if (j == 0 || j == m - 1) {
            if (gr[i][j] > 3)
              fl = 1;
            else
              gr[i][j] = 3;
          } else {
            if (gr[i][j] > 4)
              fl = 1;
            else
              gr[i][j] = 4;
          }
        }
      }
    }
    if (fl == 1) {
      cout << "NO\n";
    } else {
      cout << "YES\n";
      for (i = 0; i < n; i++) {
        for (j = 0; j < m; j++) {
          cout << gr[i][j] << " ";
        }
        cout << "\n";
      }
    }
  }
  return 0;
}
