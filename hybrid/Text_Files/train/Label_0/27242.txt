#define MAX     1000
int k, failCost[MAX+10];

double dp[MAX+10], minCost[MAX+10][40][MAX+10], avg[40];

double call(int have, int needToUse, int curLen)
{
    assert(have >= needToUse);
    if(needToUse == 0)
    {
//        DBG;
        return solve(have) + failCost[0];
    }
    int total = have + curLen;
    double p = ((double)(curLen+1))/(total+1); /// probability of this failing
    if(have == needToUse)
    {
//        DBG;
//        D(p*failCost[needToUse]);
        double ret = p * (solve(curLen) + failCost[needToUse]);
        ret += (1-p) * (avg[needToUse]);
        return ret;
    }
//    assert(false);
    double &ret = minCost[have][needToUse][curLen];
    if(ret == ret)
        return ret;



    /// pick this

    double tmp = p * (solve(curLen) + failCost[needToUse])
                +(1-p) * call(have-1, needToUse-1, 0);

    ret = tmp;

    /// dont pick this
    tmp = call(have-1, needToUse, curLen+1);

    ret = min(ret, tmp);
    return ret;
}

double solve(int n)
{
//    D(n);
    if(n == 0)
        return 0;
    double &ret = dp[n];
    if(ret == ret)
        return ret;
    ret = 1e100;
    for(int i = 1; i<=min(n,k); i++)
    {
        ret = min(ret, call(n, i, 0));
    }
    return ret;
}
