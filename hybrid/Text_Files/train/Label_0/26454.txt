#include <bits/stdc++.h>
using namespace std;
long long take2(long long maxsum) {
  if (maxsum < 0) return 0;
  return (maxsum + 2) * (maxsum + 1) / 2;
}
long long notmore(long long maxval) {
  if (maxval < 0) return 0;
  return (maxval + 2) * (maxval + 1) / 2;
}
long long take2eq(long long val) {
  if (val < 0) return 0;
  return val + 1;
}
long long notdiff(long long maxval) {
  if (maxval < 0) return 0;
  return maxval + 1;
}
void run() {
  int N, x, M, y;
  scanf("%d%d%d%d", &N, &x, &M, &y);
  int d = abs(x - y);
  long long nzero = 0, none = 0, ntwo = 0, ntotal = (long long)N * M;
  nzero +=
      take2(d - 2) - take2(d - 2 - M) - take2(d - 2 - N) + take2(d - 2 - N - M);
  nzero += notmore(N - d - 2) - notmore(N - M - d - 2);
  nzero += notmore(M - d - 2) - notmore(M - N - d - 2);
  none += take2eq(d - 1) - take2eq(d - M - 1) - take2eq(d - N - 1) +
          take2eq(d - N - M - 1);
  none += notdiff(N - d - 1) - notdiff(N - M - d - 1);
  none += notdiff(M - d - 1) - notdiff(M - N - d - 1);
  ntwo = ntotal - nzero - none;
  long long ret = 1;
  ret += 1 * none + 2 * ntwo;
  ret += N;
  if (M >= d + 1) ret -= min(N, M - d);
  ret += M;
  if (N >= d + 1) ret -= min(M, N - d);
  cout << ret << endl;
}
int main() {
  run();
  return 0;
}
