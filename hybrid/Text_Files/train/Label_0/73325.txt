(n, m) = map(int, input().split())
s = []
for i in range(n):
	s.append(input())
fir = []
yc = -1
for i in range(n):
	for j in range(m):
		if s[i][j] == 'B':
			fir = [i, j]
			break
	if len(fir):
		break
cnt = 0
for i in range(fir[1], m):
	if s[fir[0]][i] == 'B':
		cnt += 1
	else:
		break
finy = fir[1] + cnt // 2
cnt = 0
for i in range(fir[0], n):
	if s[i][fir[1]] == 'B':
		cnt += 1
	else:
		break
finx = fir[0] + cnt // 2
print(finx + 1, finy + 1)
