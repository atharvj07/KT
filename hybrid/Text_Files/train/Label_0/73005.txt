from math import sqrt
for _ in range(int(input())):
	(k, x) = map(int, input().split())
	if 2 * x <= k * (k + 1):
		xd = int(sqrt(2 * x))
		while xd * (xd + 1) < 2 * x:
			xd += 1
		print(xd)
	elif x >= k ** 2:
		print(2 * k - 1)
	else:
		ans = k
		temp = k
		x -= k * (k + 1) // 2
		xd = k - int(sqrt(k ** 2 - k - 2 * x)) - 1
		while xd * (2 * k - xd - 1) // 2 < x:
			xd += 1
		ans += xd
		print(ans)
