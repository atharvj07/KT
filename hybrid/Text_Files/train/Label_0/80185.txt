import sys
input = sys.stdin.readline
(n, q, k) = map(int, input().split())
a = list(map(int, input().split()))
front_shift = [0]
back_shift = [0]
for i in range(n):
	front_shift.append(k - a[i])
	back_shift.append(a[i] - 1)
for _ in range(q):
	(l, r) = map(int, input().split())
	print(front_shift[r] + 2 * back_shift[r] - back_shift[l] - 2 * (r - l))
