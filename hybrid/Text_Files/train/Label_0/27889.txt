#include<stdio.h>
int main(){
    int t,i,j,k,n,a[100],temp,win;
    scanf("%d",&t);
    for(i=0;i<t;i++){
        scanf("%d",&n);
        win=0;
        for(j=0;j<n;j++){
            scanf("%d",&a[j]);
        }
        for(j=0;j<n;j++){
            for(k=j+1;k<n;k++){
                if(a[k]>a[j])
                {
                    temp=a[j];
                    a[j]=a[k];
                    a[k]=temp;
                }
            }
        }
        for(j=0;j<n;j++){
            if(a[j]>a[n-1])
                win++;
        }
        printf("%d\n",win);
    }
    return 0;
}
