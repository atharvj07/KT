#include <bits/stdc++.h>
using namespace std;
inline void fg(vector<int> G[], const int& A, const int& B) {
  G[A].push_back(B);
  G[B].push_back(A);
}
vector<int> G[(1000005)];
vector<int> SEX;
int ANG[(1000005)];
bool chk[(1000005)];
long long Ans, SEBINSEX;
int N, M;
void dfs(const int& idx) {
  chk[idx] = true;
  for (const int& v : G[idx])
    if (!chk[v]) dfs(v);
}
int main() {
  scanf("%d%d", &N, &M);
  if (M < 2) {
    puts("0");
    return 0;
  }
  for (int i = 0, a, b; i < M; i++) {
    scanf("%d%d", &a, &b);
    if (a == b) {
      SEX.push_back(a);
      G[a].push_back(a);
    } else {
      fg(G, a, b);
      ANG[a]++;
      ANG[b]++;
      SEBINSEX++;
    }
  }
  for (int i = 1; i <= N; i++) {
    if (G[i].empty()) continue;
    dfs(i);
    break;
  }
  for (int i = 1; i <= N; i++) {
    if (G[i].empty()) continue;
    if (chk[i]) continue;
    puts("0");
    return 0;
  }
  Ans += (long long)((int)(SEX).size()) * (((int)(SEX).size()) - 1) / 2;
  for (int i = 1; i <= N; i++) {
    Ans += (long long)ANG[i] * (ANG[i] - 1) / 2;
  }
  Ans += (long long)((int)(SEX).size()) * SEBINSEX;
  cout << Ans << "\n";
  return 0;
}
