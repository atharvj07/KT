for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	b = []
	cr = a[0]
	cnt = 1
	for i in range(1, n):
		if a[i] == cr:
			cnt += 1
		else:
			b.append(cnt)
			cnt = 1
			cr = a[i]
	b.append(cnt)
	b.sort()
	c = [b[0]]
	d = []
	cr = b[0]
	cnt = 1
	for i in range(1, len(b)):
		if b[i] == cr:
			cnt += 1
		else:
			d.append(cnt)
			cnt = 1
			cr = b[i]
			c.append(cr)
	d.append(cnt)
	ans = []
	for i in range(len(c)):
		sm = 0
		for j in range(i):
			sm += c[j] * d[j]
		for j in range(i + 1, len(c)):
			sm += (c[j] - c[i]) * d[j]
		ans.append(sm)
	print(min(ans))
