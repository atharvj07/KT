#include <bits/stdc++.h>
using namespace std;
using namespace std;
int main() {
  string s, s2, s3;
  cin >> s >> s2;
  s3 = s;
  for (int i = s.length() - 1; i >= 0; i--) {
    if (s3[i] < 'z') {
      s3[i]++;
      if (s3 < s2) {
        cout << s3 << endl;
        return 0;
      } else {
        s3[i] = 'a';
      }
    }
    if (s3[i] == 'z') s3[i] = 'a';
  }
  cout << "No such string" << endl;
  return 0;
}
