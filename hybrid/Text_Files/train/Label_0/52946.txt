import java.util.Scanner;
import java.util.TreeSet;
 
public class fD {
 
	public static void main(String[] args) {
		Scanner stdin = new Scanner(System.in);
		int n = stdin.nextInt();
		int m = stdin.nextInt();
		TreeSet<Integer>[] list = new TreeSet[n];
		for(int i = 0; i < n; i++)
		{
			list[i] = new TreeSet<>(); 
		}
		for(int i = 0; i < m; i++)
		{
			int a = stdin.nextInt();
			int b = stdin.nextInt();
			int temp = Math.max(a, b) - Math.min(a, b);
			int min = Math.min(temp, n-temp);
			list[a-1].add(min);
			list[b-1].add(min);
		}
		int[] hashes = new int[n];
		for(int i = 0; i < n; i++)
		{
			hashes[i] = list[i].hashCode();
		}
		for(int i = 1; i < n; i++)
		{
			boolean rotatable = true;
			for(int u = 0; u < n; u++)
			{
				if(hashes[u] != hashes[(u+i)%n])
				{
					rotatable = false;
					break;
				}
			}
			if(rotatable)
			{
				System.out.println("Yes");
				return;
			}
		}
		System.out.println("No");
		stdin.close();
	}
}
