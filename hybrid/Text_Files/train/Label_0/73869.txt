M = list(map(int, input().split()))
M.sort()
if M[0] == 1:
	(x, y) = divmod(M[1], 6)
	ANS = x * 6
	if y <= 3:
		print(ANS)
	elif y == 4:
		print(ANS + 2)
	elif y == 5:
		print(ANS + 4)
elif M[0] == 2:
	if M[1] == 2:
		print(0)
	elif M[1] == 3:
		print(4)
	elif M[1] == 7:
		print(12)
	else:
		print(M[1] * 2)
elif M[0] == 3:
	if M[1] == 3:
		print(8)
	elif M[1] >= 4:
		print(M[0] * M[1] // 2 * 2)
else:
	print(M[0] * M[1] // 2 * 2)
