#include <iostream>
using namespace std;

const int MAX_N = 10000001;
bool is_p[MAX_N+1] = {false,false};

void sieve(){
	for(int i = 2 ; i <= MAX_N ; i++ ) is_p[i] = true;
	for(int i = 2 ; i*i <= MAX_N ; i++ ){
		if( is_p[i] ){
			for(int j = i+i ; j <= MAX_N ; j += i ){
				is_p[j] = false;
			}
		}
	}
}


int main(){
	sieve();
	int n;
	while( cin >> n , n ){
		int ans = -1;
		for(int i=n ; true ; i-- ){
			if( is_p[i-8] && is_p[i-6] && is_p[i-2] && is_p[i] ){
				ans = i;
				break;
			}
		}
		cout << ans << endl;
	}
}
