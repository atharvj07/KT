s = input()
n = int(input())

def is_tandem(string, half):
	if len(string) <= half:
		return True
	second = string[half:]
	first = string[0:len(second)]
	return first == second
min_len = 0
if n >= len(s):
	min_len = (len(s) + n) // 2 * 2
for i in range(n, len(s) + 1, 2):
	total_len = i + n
	half = total_len // 2
	sub_str = s[len(s) - i:]
	if is_tandem(sub_str, half):
		min_len = i + n
if min_len < len(s):
	for i in range(len(s)):
		for j in range(i + 2, len(s) + 1, 2):
			sub_str = s[i:j]
			half = (j - i) // 2
			if is_tandem(sub_str, half) and len(sub_str) > min_len:
				min_len = len(sub_str)
print(min_len)
