from math import gcd

def mcm(a, b):
	return int(a * b / gcd(int(a), int(b)))
(n, m, q) = [int(i) for i in input().split(' ')]
M = mcm(n, m)
(s1, s2) = (M // n, M // m)
M = mcm(s1, s2)
for i in range(q):
	(sx, sy, ex, ey) = [int(i) for i in input().split(' ')]
	(Ns, Ne) = (-1, -1)
	if sx == 1:
		Ns = sy * s1 // M - (sy * s1 % M == 0)
	else:
		Ns = sy * s2 // M - (sy * s2 % M == 0)
	if ex == 1:
		Ne = ey * s1 // M - (ey * s1 % M == 0)
	else:
		Ne = ey * s2 // M - (ey * s2 % M == 0)
	if Ns == Ne:
		print('YES')
	else:
		print('NO')
