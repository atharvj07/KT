#include<iostream>

using namespace std;

int a[10];

bool check(){
  static int c[][3]={
    {0,1,2},
    {3,4,5},
    {6,7,8},
    {0,3,6},
    {1,4,7},
    {2,5,8},
    {0,4,8},
    {2,4,6},
  };
  int r=-1;
  bool f=false;
  for(int j=0;j<8;j++){
    if(a[c[j][0]]==0||a[c[j][1]]==0||a[c[j][2]]==0)continue;
    int s=(a[c[j][0]]+a[c[j][1]]+a[c[j][2]])%a[9];
    if(r==-1){
      r=s;
    }else if(r!=s){
      f=true;
    }
  }
  return !f;
}

int dfs(int x,int u){
  if(x<0)return check();
  if(a[x])return dfs(x-1,u);
  int s=0;
  for(int i=1;i<=10;i++){
    if(u>>i&1)continue;
    a[x]=i;
    if(check()){
      s+=dfs(x-1,u|1<<i);
    }
    a[x]=0;
  }
  return s;
}

int main(){
  for(;;){
    int u=0;
    for(auto &e:a){
      cin>>e;
      u|=1<<e;
    }
    if(a[0]==-1)break;
    cout<<dfs(9,u)<<endl;
  }
}
