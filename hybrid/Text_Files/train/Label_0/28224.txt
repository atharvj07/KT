#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-10;
vector<long long> primes;
void GeneratingPrimes(int top) {
  vector<bool> p(top + 1, 1);
  p[0] = p[1] = 0;
  for (int i = 2; i <= sqrt(top); i++) {
    if (p[i]) {
      for (int j = i * i; j <= top; j += i) {
        p[j] = 0;
      }
    }
  }
  for (int i = 0; i < top + 1; i++)
    if (p[i]) primes.push_back(i);
}
int gcd(int a, int b) { return (b == 0) ? a : gcd(b, a % b); }
long long POW(long long x, long long p) {
  long long ret = 1;
  while (p) {
    if (p & 1) ret *= x;
    x *= x;
    p /= 2;
  }
  return ret;
}
double dist(int a, int b, int x, int y) {
  return sqrt(pow(a - x, 2) + pow(b - y, 2));
}
pair<int, int> X[10009];
int R[10009];
int ans[10009];
int main() {
  int n, m;
  cin >> n;
  memset(ans, -1, sizeof ans);
  for (int i = 0; i < n; i++) {
    cin >> X[i].first;
    cin >> R[i];
    X[i].second = i;
  }
  sort(X, X + n);
  cin >> m;
  int cnt = 0;
  for (int i = 0; i < m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    auto I = lower_bound(X, X + n, make_pair(x, 0));
    int j = (I - X), k = j - 1, h = j + 1;
    if (j < n && ans[X[j].second] == -1 &&
        dist(x, y, X[j].first, 0) - EPS <= R[X[j].second]) {
      ans[X[j].second] = i + 1;
      cnt++;
    }
    if (k >= 0 && ans[X[k].second] == -1 &&
        dist(x, y, X[k].first, 0) - EPS <= R[X[k].second]) {
      ans[X[k].second] = i + 1;
      cnt++;
    }
    if (h < n && ans[X[h].second] == -1 &&
        dist(x, y, X[h].first, 0) - EPS <= R[X[h].second]) {
      ans[X[h].second] = i + 1;
      cnt++;
    }
  }
  cout << cnt << endl;
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
  return 0;
}
