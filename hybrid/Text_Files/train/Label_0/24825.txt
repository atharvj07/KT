#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 5;
int vis[N + 5], prime[N / 5], tot;
void sieve() {
  for (int i = 2; i <= N; i++) {
    if (!vis[i]) prime[tot++] = i;
    for (int j = 0; j < tot; j++) {
      if (prime[j] * i >= N) break;
      vis[i * prime[j]] = 1;
      if (i % prime[j] == 0) break;
    }
  }
}
long long a[N], n;
long long solve(long long x) {
  long long pre = 0, res = 0;
  for (int i = 1; i <= n; i++) {
    pre = (pre + a[i]) % x;
    res += min(pre, x - pre);
  }
  return res;
}
int main() {
  sieve();
  scanf("%lld", &n);
  long long sum = 0;
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    sum += a[i];
  }
  if (sum == 1) return 0 * printf("-1\n");
  long long min1 = 1e18;
  for (int i = 0; prime[i] * prime[i] <= sum && i < tot; i++) {
    if (sum % prime[i] == 0) {
      while (sum % prime[i] == 0) sum /= prime[i];
      min1 = min(min1, solve(prime[i]));
    }
  }
  if (sum > 1) min1 = min(min1, solve(sum));
  printf("%lld\n", min1);
}
