def main():
	input()
	claws = list(map(int, input().split()))
	claws.reverse()
	alive = 1
	kill_counter = claws[0]
	index = 1
	while index < len(claws):
		if kill_counter == 0:
			alive += 1
		else:
			kill_counter -= 1
		if claws[index] > kill_counter:
			kill_counter = claws[index]
		index += 1
	print(alive)
main()
