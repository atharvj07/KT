for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	t = []
	for i in range(n):
		t.append(arr[i] - i)
	dic = {}
	for i in range(n):
		if t[i] not in dic:
			dic[t[i]] = 1
		else:
			dic[t[i]] += 1
	count = 0
	for i in dic:
		if dic[i] > 1:
			count += abs(dic[i]) * abs(dic[i] - 1) // (1 + 1)
	print(count)
