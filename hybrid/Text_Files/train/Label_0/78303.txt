s = input().split()
(n, a, b) = (int(s[0]), int(s[1]), int(s[2]))
s = input().split()
l = [int(s[i]) for i in range(n)]
res = 0
flag = False
if n % 2 == 0:
	l1 = l[:n // 2]
	l2 = l[n // 2:]
else:
	if l[n // 2] == 2:
		res += min(a, b)
	l1 = l[:n // 2]
	l2 = l[n // 2 + 1:]
l2 = l2[-1::-1]
for i in range(len(l1)):
	if l1[i] == 0:
		if l2[i] == 0:
			pass
		elif l2[i] == 1:
			flag = True
		else:
			res += a
	elif l1[i] == 1:
		if l2[i] == 0:
			flag = True
		elif l2[i] == 1:
			pass
		else:
			res += b
	elif l2[i] == 0:
		res += a
	elif l2[i] == 1:
		res += b
	else:
		res += min(a, b) * 2
if flag:
	print('-1')
else:
	print(res)
