import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        long cnt = 0;
        long sum = 0;
        long a = 0;
        for (int i = 0; i < n; ++i) {
            a = cin.nextLong();
            long min = Math.min(a / 2, cnt);
            sum += min;
            cnt -= min;
            a -= 2 * min;
            sum += a / 3;
            cnt += a % 3;
        }
        System.out.println(sum);
    }
}
