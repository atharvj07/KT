(n, m) = map(int, input().split())
t = []
for k in range(n):
	t.append(list(map(int, input().split())))
T = []
TB = []
for i in range(n):
	u = []
	for j in range(m):
		if j == 0:
			u.append(t[i][j])
		else:
			u.append(u[-1] + t[i][j])
	T.append(u)
	v = []
	for s in range(m - 1, -1, -1):
		if s == m - 1:
			v.append(t[i][s])
		else:
			v.append(v[-1] + t[i][s])
	TB.append(v[::-1])
f = []
for j in range(m):
	u = []
	for i in range(n):
		u.append(t[i][j])
	f.append(u)
UT = []
UB = []
for i in range(m):
	u = []
	for j in range(n):
		if j == 0:
			u.append(f[i][j])
		else:
			u.append(u[-1] + f[i][j])
	UT.append(u)
	v = []
	for k in range(n - 1, -1, -1):
		if k == n - 1:
			v.append(f[i][k])
		else:
			v.append(v[-1] + f[i][k])
	UB.append(v[::-1])
p = 0
for i in range(n):
	for j in range(m):
		if t[i][j] == 0:
			if T[i][j] > 0:
				p += 1
			if TB[i][j] > 0:
				p += 1
			if UT[j][i] > 0:
				p += 1
			if UB[j][i] > 0:
				p += 1
print(p)
