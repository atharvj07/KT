inline long long gcd(long long a, long long b) {
  long long r;
  while (b) {
    r = a % b;
    a = b;
    b = r;
  }
  return a;
}
inline long long lcm(long long a, long long b) { return a / gcd(a, b) * b; }
inline long long fpow(long long n, long long k, int p = MOD) {
  long long r = 1;
  for (; k; k >>= 1) {
    if (k & 1) r = r * n % p;
    n = n * n % p;
  }
  return r;
}

inline unsigned long long isqrt(unsigned long long k) {
  unsigned long long r = sqrt(k) + 1;
  while (r * r > k) r--;
  return r;
}
inline unsigned long long icbrt(unsigned long long k) {
  unsigned long long r = cbrt(k) + 1;
  while (r * r * r > k) r--;
  return r;
}

mt19937 mt(chrono::high_resolution_clock::now().time_since_epoch().count());
inline int myrand() { return abs((int)mt()); }
const int maxn = 600 + 5;
int n;
int f[maxn];
int g[maxn];
vector<int> adj[maxn];
int vis[maxn];
int ask(vector<int> ver) {
  static int num = 0;
  sort((ver).begin(), (ver).end()),
      (ver).erase(unique((ver).begin(), (ver).end()), (ver).end());
  if (int((ver).size()) <= 1) {
    return 0;
  }
  assert(num++ < 20000);
  cout << "? " << int((ver).size()) << "\n";
  for (int i = (0); i < (int((ver).size())); ++i)
    cout << ver[i] + 1 << " \n"[i == int((ver).size()) - 1];
  cout.flush();
  int res;
  cin >> res;
  if (res == -1) {
    exit(0);
  }
  return res;
}
int from[maxn];
void dfs(int u, int p) {
  vis[u] = 1;
  for (int v : adj[u]) {
    if (v != p && !vis[v]) {
      from[v] = u;
      dfs(v, u);
    }
  }
}
void chemthan() {
  srand(time(NULL));
  cin >> n;
  queue<int> que;
  f[0] = 1, que.push(0);
  while (int((que).size())) {
    int u = que.front();
    que.pop();
    while (1) {
      vector<int> ver;
      for (int i = (0); i < (n); ++i)
        if (!f[i]) ver.push_back(i);
      int x = ask(ver);
      ver.push_back(u);
      int y = ask(ver);
      ver.pop_back();
      if (x == y) break;
      while (int((ver).size()) > 1) {
        vector<int> ver1, ver2;
        for (int i = (0); i < (int((ver).size())); ++i) {
          if (i < int((ver).size()) / 2) {
            ver1.push_back(ver[i]);
          } else {
            ver2.push_back(ver[i]);
          }
        }
        if (myrand() & 1) {
          swap(ver1, ver2);
        }
        int x = ask(ver1);
        ver1.push_back(u);
        int y = ask(ver1);
        ver1.pop_back();
        if (x < y) {
          ver = ver1;
        } else {
          ver = ver2;
        }
      }
      assert(int((ver).size()) == 1);
      int v = ver[0];
      if (!f[v]) {
        g[v] = g[u] ^ 1;
        adj[u].push_back(v), adj[v].push_back(u);
        f[v] = 1, que.push(v);
      }
    }
  }
  vector<int> ver1, ver2;
  for (int i = (0); i < (n); ++i) {
    if (!g[i]) {
      ver1.push_back(i);
    } else {
      ver2.push_back(i);
    }
  }
  for (vector<int> v : {ver1, ver2}) {
    if (ask(v)) {
      while (int((v).size()) > 2) {
        while (1) {
          random_shuffle((v).begin(), (v).end());
          vector<int> ver1, ver2;
          for (int i = (0); i < (int((v).size())); ++i) {
            if (i < int((v).size()) / 2) {
              ver1.push_back(v[i]);
            } else {
              ver2.push_back(v[i]);
            }
          }
          if (myrand() & 1) {
            swap(ver1, ver2);
          }
          if (ask(ver1)) {
            v = ver1;
            break;
          }
          if (ask(ver2)) {
            v = ver2;
            break;
          }
        }
      }
      assert(int((v).size()) == 2);
      int st = v[0];
      int ed = v[1];
      dfs(st, -1);
      assert(vis[ed]);
      vector<int> cycle;
      while (ed != st) {
        cycle.push_back(ed);
        ed = from[ed];
      }
      cycle.push_back(ed);
      cout << "N " << int((cycle).size()) << "\n";
      for (int i = (0); i < (int((cycle).size())); ++i)
        cout << cycle[i] + 1 << " \n"[i == int((cycle).size()) - 1];
      cout.flush();
      return;
    }
  }
  cout << "Y " << int((ver1).size()) << "\n";
  for (int i = (0); i < (int((ver1).size())); ++i)
    cout << ver1[i] + 1 << " \n"[i == int((ver1).size()) - 1];
  cout.flush();
}
