def f():
	(n, t) = map(int, input().split())
	l = list(map(int, input().split()))
	d = 0
	(x, y) = (0, 0)
	(p, q) = (-1, -1)
	for i in range(n):
		if t + l[i] >= 0:
			y += 1
			t += l[i]
		else:
			if d < y - x:
				d = y - x
				(p, q) = (x, y)
			while t + l[i] < 0 and x != y:
				t -= l[x]
				x += 1
			y += 1
			if t + l[i] < 0:
				x += 1
			else:
				t += l[i]
		if d < y - x:
			d = y - x
			(p, q) = (x, y)
	if d:
		return [p + 1, q]
	else:
		return [-1]
for _ in range(int(input())):
	print(*f())
