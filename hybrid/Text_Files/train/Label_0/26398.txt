#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int mod=1e9+7;

ll a[100500][100];

ll comb(int x,int y){
	if(a[x][y]) 
		return a[x][y];
	if(y==0||y==x ) 
		return a[x][y]=1;
	return a[x][y]=(comb(x-1,y)+comb(x-1,y-1))%mod;
}

int main(){
	ll m,n;
	cin>>n>>m;
	ll x=m,p=2,ans=1;
	while(p*p<=m){
		int cnt=0;
		while(x%p==0){
			x/=p;
			cnt++;
		}
		if(cnt){
			ans=(ans*comb(n+cnt-1,cnt)) % mod;
		}
		p++;
	}
	if(x!=1)
		ans=(ans*n)%mod;
	cout<<ans<<endl;
}
