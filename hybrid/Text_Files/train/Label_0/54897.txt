import java.io.*;
import java.util.*;
public class cf3021C{
    public static void main(String[] args) throws IOException {
    	Scanner f = new Scanner(System.in);
        StringTokenizer st = new StringTokenizer(f.nextLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        char[][] str = new char[n][m];
        for(int i = 0; i < n; i++){
        	st = new StringTokenizer(f.nextLine());
        	String s = st.nextToken();
        	for(int j = 0; j < m; j++){
        		str[i][j] = s.charAt(j);
        	}
        }
        int[][] cost = new int[n][m];
        for(int i = 0; i < n; i++){
        	st = new StringTokenizer(f.nextLine());
        	for(int j = 0; j < m; j++){
        		cost[i][j] = Integer.parseInt(st.nextToken());
        	}
        }
        f.close();
        
        //precomputation of intersections and costs
        int[][] strn = new int[n][m];
        int[][] costn = new int[n][m];
        for(int i = 0; i < n; i++){
        	for(int j = 0; j < m; j++){
        		int maxcost = 0;
        		for(int k = 0; k < n; k++){
        			if(str[i][j] == str[k][j]){
        				strn[i][j] |= (1 << k);//makes sure kth bit is on
        				costn[i][j] += cost[k][j];
        				maxcost = Math.max(maxcost, cost[k][j]);
        				
        			}
        		}
        		costn[i][j] -= maxcost;
        	}
        }
        
        //dp
        int[] dp = new int[(1 << n)];//holds mincost for making a subset of strings easy
        Arrays.fill(dp, 20000000);
        dp[0] = 0;
        for(int i = 1; i < (1 << n); i++){
        	int minstr = n;
        	for(int j = 0; j < n; j++){
        		//System.out.println((i & (1 << j)) + " " +  (1 << j));
        		if((i & (1 << j)) == (1 << j)){
        			minstr = j;
        			break;
        		}
        	}
        	//System.out.println(i + " " + minstr);
        	
        	for(int col = 0; col < m; col++){
        		dp[i] = Math.min(dp[i], dp[i ^ (1 << minstr)] + cost[minstr][col]);
        		dp[i] = Math.min(dp[i], dp[(i ^ strn[minstr][col]) & i] + costn[minstr][col]);
        	}
        }
        
        /*for(int i = 0; i < (1<<n); i++){
        	System.out.println(i + " " + dp[i]);
        }*/
        
        System.out.println(dp[(1 << n) - 1]);
    }
}
