n = input()
n = int(n)
s = input()
s = str(s)
state = 0
c = 0
l = 0
w = 0
for i in s:
	if state == 0:
		if i == '_':
			if c > l:
				l = c
			c = 0
		elif i == '(':
			if c > l:
				l = c
			c = 0
			state = 1
		else:
			c += 1
	elif i == '_':
		if c != 0:
			w += 1
		c = 0
	elif i == ')':
		if c != 0:
			w += 1
		c = 0
		state = 0
	else:
		c += 1
if c > l:
	l = c
print('%d %d' % (l, w))
