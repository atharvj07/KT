

import java.util.Scanner;

public class JavaApplication8 {

    public static void main(String[] args) {

        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a[] = new int[n];
        int p[] = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            p[i] = in.nextInt();
        }

        int res = 0;
        int min = p[0];

        for (int i = 0; i < n; i++) {
            if (min > p[i]) {
                min = p[i];
            }
            res += min * a[i];
        }

        System.out.println(res);

    }
}
