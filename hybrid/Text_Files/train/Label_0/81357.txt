n = int(input())
money = list(map(int, input().split()))
q = int(input())
tranzactions = list((list(map(int, input().split())) for _ in range(q)))
result = [-1] * n
temp = -1
for i in range(q - 1, -1, -1):
	if tranzactions[i][0] == 1:
		if result[tranzactions[i][1] - 1] == -1:
			result[tranzactions[i][1] - 1] = max(tranzactions[i][2], temp)
	else:
		temp = max(tranzactions[i][1], temp)
for i in range(n):
	if result[i] == -1:
		result[i] = max(temp, money[i])
print(' '.join(map(str, result)))
