ticket = [int(x) for x in input()]
fst = ticket[:3]
sec = ticket[3:]
diff = abs(sum(fst) - sum(sec))
if sum(fst) > sum(sec):
	(fst, sec) = (sec, fst)
if diff == 0:
	print(0)
elif 9 - min(fst) >= diff or max(sec) >= diff:
	print(1)
elif 18 - (sum(fst) - max(fst)) >= diff or sum(sec) - min(sec) >= diff or 9 - min(fst) + max(sec) >= diff:
	print(2)
else:
	print(3)
