#include <bits/stdc++.h>
using namespace std;
long long posx[] = {1, -1, 0, 0};
long long posy[] = {0, 0, 1, -1};
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    vector<pair<long long, long long> > v;
    vector<long long> vec;
    for (long long i = 0, x; i < n; i++) {
      cin >> x;
      v.push_back({x, i});
    }
    sort(v.begin(), v.end());
    long long ans = (v[n - 1].first * v[0].first);
    for (long long i = 2; i * i <= ans; i++) {
      if (ans % i == 0) {
        vec.push_back(i);
        if (ans / i != i) {
          vec.push_back(ans / i);
        }
      }
    }
    sort(vec.begin(), vec.end());
    bool ok = true;
    long long cnt = 0;
    for (long long i = 0; i < n && ok; i++) {
      if (v[i].first != vec[i]) {
        ok = false;
      } else {
        cnt++;
      }
    }
    cout << (ok && cnt == n ? ans : -1) << endl;
  }
  return 0;
}
