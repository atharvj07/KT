from sys import stdin, stdout

def solution(a, b):
	if max(a, b) < min(a, b) * 2 or not a or (not b):
		stdout.write(str(a) + ' ' + str(b))
		return
	if a > b:
		m = a // b
		if m & 1:
			solution(a - (m - 1) * b, b)
		else:
			solution(a - m * b, b)
	else:
		m = b // a
		if m & 1:
			solution(a, b - (m - 1) * a)
		else:
			solution(a, b - a * m)
(a, b) = map(int, stdin.readline().split())
solution(a, b)
