#include <bits/stdc++.h>
using namespace std;
void play(int a, string &s) {
  bitset<26> cnt;
  cnt.reset();
  for (int i = 1; i <= a; ++i) {
    cnt[s[s.length() - i] - 'a'] = 1;
  }
  for (int i = 0; i < a; ++i) {
    for (int j = 0; j < 26; ++j) {
      if (!cnt[j]) {
        s += j + 'a';
        cnt[j] = 1;
        break;
      }
    }
  }
}
int main() {
  int a, b, l, r;
  scanf("%d%d%d%d", &a, &b, &l, &r);
  l--, r--;
  if (r % (a + b) >= a) {
    r = r / (a + b) * (a + b) + a - 1;
  }
  if (l > r) return 0 * puts("1");
  int fff = l / (a + b);
  l -= fff * (a + b);
  r -= fff * (a + b);
  if (r >= 4 * (a + b)) {
    if (b >= a)
      printf("%d\n", a + 1);
    else
      printf("%d\n", 2 * a - min(a, b));
  } else {
    int ans = 26;
    for (char c1 = 'a'; c1 <= 'z'; ++c1) {
      for (char c2 = 'a'; c2 <= 'z'; ++c2) {
        for (char c3 = 'a'; c3 <= 'z'; ++c3) {
          for (char c4 = 'a'; c4 <= 'z'; ++c4) {
            string s = "";
            for (int i = 0; i < a; ++i) s += 'a' + i;
            for (int i = 0; i < b; ++i) s += c1;
            play(a, s);
            for (int i = 0; i < b; ++i) s += c2;
            play(a, s);
            for (int i = 0; i < b; ++i) s += c3;
            play(a, s);
            for (int i = 0; i < b; ++i) s += c4;
            bitset<26> res;
            res.reset();
            for (int i = l; i <= r; ++i) res[s[i] - 'a'] = 1;
            ans = min(ans, (int)res.count());
          }
        }
      }
    }
    printf("%d\n", ans);
  }
}
