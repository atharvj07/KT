n = int(input())
nBinary = str(bin(n))[2:]
count = 0
lst = []
count1 = 0
while '0' in nBinary and count < 40:
	if count % 2 == 0:
		lst.append(len(nBinary))
		count += 1
		nList = list(nBinary)
		for i in range(len(nList)):
			if nList[i] == '1':
				nList[i] = '0'
			else:
				nList[i] = '1'
		nBinary = ''.join(nList)
		tmp = int(nBinary, 2)
		nBinary = str(bin(tmp))[2:]
	else:
		count += 1
		tmp = int(nBinary, 2) + 1
		nBinary = str(bin(tmp))[2:]
print(count)
print(' '.join(map(str, lst)))
