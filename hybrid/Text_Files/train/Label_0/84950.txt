n = int(input())
c = list(map(int, input().strip().split()))
subStrings = []
alt = 0
for i in range(n - 1, -1, -1):
	while c[i] > 0:
		alt = (alt + 1) % 2
		if alt == 1:
			substr = 'a' * (i + 1)
			subStrings.append(substr)
		else:
			substr = 'b' * (i + 1)
			subStrings.append(substr)
		for j in range(i, -1, -1):
			c[j] -= i - j + 1
print(''.join(subStrings))
