import sys
input = sys.stdin.readline
t = int(input())
ans = []
for _ in range(t):
	(n, m, k) = map(int, input().split())
	if n % m == 0:
		c = n // m
		ans0 = []
		for _ in range(k):
			for i in range(m):
				ans0.append(' '.join(map(str, [c] + [j + i * c + 1 for j in range(c)])))
		ans.append('\n'.join(ans0))
		continue
	x = [n // m + min(n % m, 1)] * m
	s = sum(x)
	u = 0
	for i in range(s - n):
		x[i] -= 1
		u += x[i]
	c = u * k
	y = [set() for _ in range(k)]
	z = [c // n + min(c % n, 1)] * n
	s = sum(z)
	for i in range(s - c):
		z[i] -= 1
	j = 0
	for i in range(n):
		for _ in range(z[i]):
			y[j].add(i + 1)
			j += 1
			j %= k
	ans0 = []
	for i in range(k):
		a = y[i]
		b = set()
		for j in range(1, n + 1):
			if not j in a:
				b.add(j)
		a = list(a)
		b = list(b)
		for j in x:
			ans00 = [j]
			if j == x[0]:
				for _ in range(j):
					ans00.append(a.pop())
			else:
				for _ in range(j):
					ans00.append(b.pop())
			ans0.append(' '.join(map(str, ans00)))
	ans.append('\n'.join(ans0))
sys.stdout.write('\n\n'.join(ans))
