for _ in range(int(input())):
	n = int(input())
	a = [int(s) for s in input().split()]
	l = [bool(int(s)) for s in input().split()]
	aa = []
	for i in range(n):
		if not l[i]:
			aa.append(a[i])
	aa.sort(reverse=True)
	k = 0
	for i in range(n):
		if l[i]:
			print(a[i], end=' ')
		else:
			print(aa[k], end=' ')
			k += 1
	print()
