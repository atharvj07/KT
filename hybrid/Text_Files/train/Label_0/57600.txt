import java.io.*;
import java.util.*;
public class r245a {
    private ArrayList<Integer> adj[];
    
    public r245a(int n)
    {
        adj=new ArrayList[n+1];
        for(int i=0;i<=n;i++)
            adj[i]=new ArrayList<Integer>();
    }
    void setEdge(int u,int v)
    {
        adj[u].add(v);
    }
    void dfs(int src,int p,int changes[],int level,int y,int flag)
    {
        //System.out.println(src+" "+level);
        if(level==2)
        {
            if(p!=-1)
                changes[src]+=y;
            if(flag==1)
                changes[src]++;
            return;
        }
        
        Iterator<Integer> it=adj[src].iterator();
        while(it.hasNext())
        {
            int x=it.next();
            if(x!=p)
                dfs(x,src,changes,level+1,y,flag);
        }
    }
    void bfs(int goal[],int val[],int changes[],ArrayList<Integer> p)
    {
        ArrayList<Integer> al=new ArrayList<Integer>();
        int parent[]=new int[adj.length];
        parent[1]=-1;
        al.add(1);
        
        while(!al.isEmpty())
        {
            int x=al.get(0);
            al.remove(0);
            //System.out.println(x+" "+goal[x]+" "+val[x]+" "+changes[x]);
            if(goal[x]!=val[x]&&changes[x]%2==0)
            {
                dfs(x,parent[x],changes,0,changes[x],1);
                p.add(x);
            }
            else if(goal[x]==val[x]&&changes[x]%2!=0)
            {
                dfs(x,parent[x],changes,0,changes[x],1);
                p.add(x);
            }
            else
                dfs(x,parent[x],changes,0,changes[x],0);
            Iterator<Integer> it=adj[x].iterator();
            while(it.hasNext())
            {
                int y=it.next();
                if(parent[x]!=y)
                {
                    al.add(y);
                    parent[y]=x;
                }
            }
        }
    }
    public static void main(String args[])
    {
        fastio in=new fastio(System.in);
        PrintWriter pw=new PrintWriter(System.out);

        int n=in.nextInt();
        r245a ob=new r245a(n);
        for(int i=0;i<n-1;i++)
        {
            int u=in.nextInt();
            int v=in.nextInt();
            ob.setEdge(u, v);
            ob.setEdge(v, u);
        }
        
        int val[]=new int[n+1];
        for(int i=1;i<=n;i++)
            val[i]=in.nextInt();
        
        int goal[]=new int[n+1];
        for(int i=1;i<=n;i++)
            goal[i]=in.nextInt();
        
        int changes[]=new int[n+1];
        ArrayList<Integer> p=new ArrayList<Integer>();
        ob.bfs(goal, val, changes, p);
        
        pw.println(p.size());
        for(int i:p)
            pw.println(i);
        pw.close();
    }

    static class fastio {
     
            private final InputStream stream;
            private final byte[] buf = new byte[8192];
            private int cchar, snchar;
            private SpaceCharFilter filter;
     
            public fastio(InputStream stream) {
                this.stream = stream;
            }
     
            public int nxt() {
                if (snchar == -1)
                    throw new InputMismatchException();
                if (cchar >= snchar) {
                    cchar = 0;
                    try {
                        snchar = stream.read(buf);
                    } catch (IOException e) {
                        throw new InputMismatchException();
                    }
                    if (snchar <= 0)
                        return -1;
                }
                return buf[cchar++];
            }
     
            public int nextInt() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                int sgn = 1;
                if (c == '-') {
                    sgn = -1;
                    c = nxt();
                }
                int res = 0;
                do {
                    if (c < '0' || c > '9')
                        throw new InputMismatchException();
                    res *= 10;
                    res += c - '0';
                    c = nxt();
                } while (!isSpaceChar(c));
                return res * sgn;
            }
     
            public long nextLong() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                int sgn = 1;
                if (c == '-') {
                    sgn = -1;
                    c = nxt();
                }
                long res = 0;
                do {
                    if (c < '0' || c > '9')
                        throw new InputMismatchException();
                    res *= 10;
                    res += c - '0';
                    c = nxt();
                } while (!isSpaceChar(c));
                return res * sgn;
            }
     
            public int[] nextIntArray(int n) {
                int a[] = new int[n];
                for (int i = 0; i < n; i++) {
                    a[i] = nextInt();
                }
                return a;
            }
     
            public String readString() {
                int c = nxt();
                while (isSpaceChar(c)) {
                    c = nxt();
                }
                StringBuilder res = new StringBuilder();
                do {
                    res.appendCodePoint(c);
                    c = nxt();
                } while (!isSpaceChar(c));
                return res.toString();
            }
     
            public String nextLine() {
                int c = nxt();
                while (isSpaceChar(c))
                    c = nxt();
                StringBuilder res = new StringBuilder();
                do {
                    res.appendCodePoint(c);
                    c = nxt();
                } while (!isEndOfLine(c));
                return res.toString();
            }
     
            public boolean isSpaceChar(int c) {
                if (filter != null)
                    return filter.isSpaceChar(c);
                return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
            }
     
            private boolean isEndOfLine(int c) {
                return c == '\n' || c == '\r' || c == -1;
            }
     
            public interface SpaceCharFilter {
                public boolean isSpaceChar(int ch);
            }
        }
    

}
