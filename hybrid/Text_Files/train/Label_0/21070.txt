#include <bits/stdc++.h>
using namespace std;
long long int gcd(long long int a, long long int b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}
int main() {
  long long int a, b;
  cin >> a >> b;
  long long int l = (a * b) / gcd(a, b);
  long long int aa = l / a;
  long long int bb = l / b;
  if (aa > bb) {
    bb++;
    if (aa == bb)
      cout << "Equal";
    else
      (aa > bb) ? cout << "Dasha" : cout << "Masha";
  } else {
    aa++;
    if (aa == bb)
      cout << "Equal";
    else
      (aa > bb) ? cout << "Dasha" : cout << "Masha";
  }
}
