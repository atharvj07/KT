n = int(input())
team1 = [0] * (n + 1)
team2 = [0] * (n + 1)
for i in range(1, n + 1):
	(team1[i], team2[i]) = map(int, input().split())
for i in range(1, n + 1):
	if i <= n // 2 or team1[i] < team2[n - i + 1]:
		print(1, end='')
	else:
		print(0, end='')
print()

		print(0, end='')
