t = int(input())
for you in range(t):
	l = input().split()
	a = int(l[0])
	b = int(l[1])
	c = int(l[2])
	poss = [1, 2, 3, 4, 5, 6, 7]
	count = [0]
	for i in range(2 ** 7):
		l = []
		for j in range(7):
			if i & 1 << j:
				l.append(poss[j])
		x = 0
		y = 0
		z = 0
		for i in l:
			for k in range(3):
				if 1 << k & i:
					if k == 0:
						x += 1
					if k == 1:
						y += 1
					if k == 2:
						z += 1
		if x <= a and y <= b and (z <= c):
			count.append(len(l))
	print(max(count))
