import java.util.*;
import java.util.Vector;

public class main
{
  public static void main(String ar[])
  {
    Scanner s=new Scanner(System.in);
    int n=s.nextInt();
    int m=s.nextInt();
    int[] a=new int[n];
    int[] b=new int[m];
    List<Integer> v=new ArrayList<Integer>();
    for(int i=0;i<n;i++)
    {
      a[i]=s.nextInt();
    }
    for(int i=0;i<m;i++)
    {
      b[i]=s.nextInt();
    }
    for(int i=0;i<n;i++)
    {
      for(int j=0;j<m;j++)
      {
        if(a[i]==b[j])
        {
          v.add(b[j]);
          //System.out.print(a[i]+" ");
         }
     }
    }
    Set<Integer> hs = new LinkedHashSet<>();
hs.addAll(v);
v.clear();
v.addAll(hs);
    Iterator<Integer> it=v.iterator();
    while(it.hasNext())
    {
        System.out.print(it.next()+" ");
    }
  }
}
