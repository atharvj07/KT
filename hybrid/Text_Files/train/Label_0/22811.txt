#include<stdio.h>
#include<algorithm>
#include<queue>
#include<vector>
using namespace std;
vector<int> g[110000];
vector<int> g2[110000];
char in[4];
int v1[110000];
int v2[110000];
int A[110000];
int B[110000];
int C[110000];
int main(){
	int T;scanf("%d",&T);
	while(T--){
		int a,b;scanf("%d%d",&a,&b);
		int c,d;scanf("%d%d",&c,&d);
		for(int i=0;i<a+1;i++)g[i].clear();
		for(int i=0;i<a+1;i++)g2[i].clear();
		for(int i=0;i<b;i++){
			int p,q;scanf("%d%d%s",&p,&q,in);
			if(in[0]=='N'){
				g[p].push_back(q);
				g[q].push_back(p);
			}else{
				g2[p].push_back(q);
				g2[q].push_back(p);
			}
		}
		for(int i=0;i<a+1;i++)v1[i]=v2[i]=0;
		queue<int>Q;
		Q.push(0);
		v1[0]=1;
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int i=0;i<g2[at].size();i++)if(!v1[g2[at][i]]){
				v1[g2[at][i]]=1;
				Q.push(g2[at][i]);
			}
		}
		v2[d]=1;
		Q.push(d);
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int i=0;i<g2[at].size();i++)if(!v2[g2[at][i]]){
				v2[g2[at][i]]=1;
				Q.push(g2[at][i]);
			}
		}
		if(v1[d]){
			printf("0\n");continue;
		}
		for(int i=0;i<a+1;i++)A[i]=B[i]=C[i]=99999999;
		A[c]=0;
		Q.push(c);
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int i=0;i<g[at].size();i++){
				if(A[g[at][i]]>9999999){
					A[g[at][i]]=A[at]+1;
					Q.push(g[at][i]);
				}
			}
		}
		for(int i=0;i<a+1;i++)if(v1[i]){
			Q.push(i);
			B[i]=0;
		}
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int i=0;i<g[at].size();i++){
				if(B[g[at][i]]>9999999){
					B[g[at][i]]=B[at]+1;
					Q.push(g[at][i]);
				}
			}
		}
		for(int i=0;i<a+1;i++)if(v2[i]){
			Q.push(i);
			C[i]=0;
		}
		while(Q.size()){
			int at=Q.front();Q.pop();
			for(int i=0;i<g[at].size();i++){
				if(C[g[at][i]]>9999999){
					C[g[at][i]]=C[at]+1;
					Q.push(g[at][i]);
				}
			}
		}
		int ret=999999999;
		for(int i=0;i<a+1;i++)ret=min(ret,A[i]+B[i]+C[i]);
		printf("%d\n",ret);
	}
}
