#include <iostream>
using namespace std;

int main()
{
int a, b, c;
cin >> a >> b >> c;
for (int iter = 0; iter <= 100000; ++iter)
{
if (a % 2 == 1 || b % 2 == 1 || c % 2 == 1)
{
cout << iter << endl;
return 0;
}
int d = (a + b)/2, e = (a + c)/2, f = (b + c)/2;
a = d, b = e, c = f;
}
cout << -1 << endl;
return 0;
}
