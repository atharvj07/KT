public class Main
{
   
    public static void main(String args[]) throws IOException
    {
        final DataInputStream br = new DataInputStream(System.in);
        final PrintWriter bw = new PrintWriter(new BufferedOutputStream(System.out));
        Scanner scan = new Scanner(System.in);
        StringBuffer str = new StringBuffer();
        int t = ni(br);
        char ch[][] = new char[t][t];
        for (int i=0;i<t;i++)   
        {
            
            for (int j=0;j<t;j++)
            {
                char cha = (char)br.read();
                if (cha == '\n' || cha == '\r')
                {
                    j--;
                    continue;
                }
                ch[i][j] = cha;
            }
        }
 
        int flag = 0;
        for (int i=0;i<t;i++)
        {
            int count = 0;
            for (int j=0;j<t;j++)
            {
                count += check(i+1,j,t,ch);
                count += check(i-1,j,t,ch);
                count += check(i,j+1,t,ch);
                count += check(i,j-1,t,ch);
            
                if (count % 2 != 0)
                {
                    flag = -1;
                    break;
                }
                count = 0;
            }
        }
        if (flag == -1)
            str.append("NO");
        else
            str.append("YES");
        
        System.out.println(str);
            
}
    static int check(int i,int j,int t,char[][] ch)
    {
        if (i >= t || i < 0 || j >= t || j < 0)
            return 0;
        else
        {
            if (ch[i][j] == 'o')
                return 1;
            else
                return 0;
        }
    }
    static int ni(DataInputStream br) throws IOException
    {
 
        int ret = 0;
        int x = br.read();
        while (x != -1 && (x < '0' || x > '9') && x != '-')
            x = br.read();
        ret = x - '0';
        x = br.read();
        while (x >= '0' && x <= '9' && x != -1 && x != ' ')
        {
            ret = (ret << 3) + (ret << 1) + x - '0';
            x = br.read();
        }
        return ret;
    }
}
