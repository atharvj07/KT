import org.omg.PortableServer.POA;

import java.awt.*;
import java.io.*;
import java.util.*;

public class Abc {
    public static void main(String[] args) throws Exception {
        FastReader sc = new FastReader();
        int n=sc.nextInt();
        if (n%2==1){
            System.out.println("no");
            return;
        }
        Point p[]=new Point[n];
        for (int i=0;i<n;i++)p[i]=new Point(sc.nextInt(),sc.nextInt());
        int m=n/2;
        for (int i=0;i<m;i++){
            if (!isParallel(p[i],p[i+1],p[i+m],p[(i+1+m)%n])){
                System.out.println("no");
                return;
            }
        }
        System.out.println("yes");
    }
    static boolean isParallel(Point a1,Point a2,Point b1,Point b2){
        return (long)(a1.y-a2.y)*(b1.x-b2.x)==(long) (b1.y-b2.y)*(a1.x-a2.x) &&
                (long)(a1.x-a2.x)*(a1.x-a2.x)+(long) (a1.y-a2.y)*(a1.y-a2.y)==(long) (b1.x-b2.x)*(b1.x-b2.x)+
                        (long) (b1.y-b2.y)*(b1.y-b2.y);
    }
    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new
                    InputStreamReader(System.in));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }
}
