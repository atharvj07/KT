#include <bits/stdc++.h>
using namespace std;
long long max(long long a, long long b) { return a > b ? a : b; }
double nCr[108][108];
double ncr(int n, int k) {
  if (k < 0 || n < k) return 0.0;
  if (k == 0 || n == k) return 1.0;
  if (nCr[n][k] != -1) return nCr[n][k];
  return nCr[n][k] = (ncr(n - 1, k) + ncr(n - 1, k - 1));
}
int main() {
  int t, m, i, j, k, l, n;
  t = 1;
  double fact[108];
  for (i = 0; i <= 100; i++) {
    for (j = 0; j <= 100; j++) nCr[i][j] = -1.0;
  }
  fact[0] = 1.0;
  for (i = 1; i <= 100; i++) fact[i] = fact[i - 1] * 1.0 * i;
  while (t--) {
    scanf("%d", &n);
    int a[n + 9], p;
    int sum = 0;
    for (i = 0; i < n; i++) {
      scanf("%d", &a[i]);
      sum += a[i];
    }
    scanf("%d", &p);
    if (sum <= p) {
      double ans = n;
      printf("%.7lf\n", ans);
      continue;
    }
    double dp[2504][54], tmp[2504][54];
    for (i = 0; i <= 2500; i++) {
      for (j = 0; j <= 50; j++) dp[i][j] = tmp[i][j] = 0.0;
    }
    for (i = 0; i < n; i++) {
      dp[0][0] = tmp[0][0] = 1.0;
      for (k = 1; k <= n; k++) {
        for (j = a[i]; j <= 2500; j++) dp[j][k] += tmp[j - a[i]][k - 1];
      }
      for (j = 0; j <= 2500; j++) {
        for (k = 0; k <= n; k++) {
          tmp[j][k] = dp[j][k];
        }
      }
    }
    double v[54];
    for (i = 0; i < 54; i++) v[i] = 0.0;
    for (j = 1; j <= n; j++) {
      for (k = p + 1; k <= 2500; k++) v[j] += dp[k][j];
    }
    double ans[n + 9];
    for (i = 0; i <= n; i++) ans[i] = 0.0;
    for (i = 1; i <= n; i++) {
      ans[i] += fact[i] * 1.0 * v[i];
      int rem = n - i;
      for (j = i + 1; j <= n; j++) {
        ans[j] -= ans[i] * 1.0 * ncr(rem, j - i) * 1.0 * fact[j - i];
      }
      ans[i] = ans[i] * 1.0 * fact[n - i] * (i - 1);
    }
    double anss = 0.0;
    for (i = 1; i <= n; i++) anss += ans[i];
    anss = anss / fact[n];
    printf("%.7lf\n", anss);
  }
  return 0;
}
