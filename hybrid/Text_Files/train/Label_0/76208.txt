def get_num():
	return int(input())

def print_arr(arr: list):
	for val in arr:
		print(val)

def read_num_list(k):
	num_str = input()
	return [int(v) for v in num_str.split(' ')]
n = get_num()
problems = read_num_list(n)
problems = sorted(problems)
i = 1
j = 0
while True:
	if j >= len(problems):
		break
	if i <= problems[j]:
		j += 1
		i += 1
	else:
		j += 1
print(i - 1)
