#include <bits/stdc++.h>
using namespace std;
long long n, i, j, d1[2000], d2[2000], sum[5000], ans;
string s;
int main() {
  cin >> s;
  n = s.length();
  for (i = 0; i < n; i++) {
    j = 0;
    while (i - j >= 0 && i + j < n && s[i - j] == s[i + j]) {
      d1[i - j]++;
      d2[i + j]++;
      j++;
    }
    j = 0;
    while (i - j - 1 >= 0 && i + j < n && s[i - j - 1] == s[i + j]) {
      d1[i - j - 1]++;
      d2[i + j]++;
      j++;
    }
  }
  for (i = n - 1; i >= 0; i--) sum[i] = sum[i + 1] + d1[i];
  for (i = 0; i < n; i++) ans += d2[i] * sum[i + 1];
  cout << ans;
  return 0;
}
