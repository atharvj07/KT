from sys import stdin, stdout
from collections import defaultdict
import math

def main():
	t = int(stdin.readline())
	for _ in range(t):
		(n, m) = list(map(int, stdin.readline().split()))
		a = list(map(int, stdin.readline().split()))
		b = list(map(int, stdin.readline().split()))
		one = []
		two = []
		for i in range(n):
			if b[i] == 1:
				one.append(a[i])
			else:
				two.append(a[i])
		one.sort(reverse=True)
		two.sort(reverse=True)
		count_one = -1
		sm_i = 0
		min_con = 100000000
		for (i, o) in enumerate(one):
			sm_i += o
			if sm_i >= m:
				count_one = i + 1
				min_con = i + 1
				break
		if count_one == -1:
			count_one = len(one)
		sm_t = 0
		for (i, t) in enumerate(two):
			sm_t += t
			if sm_t + sm_i < m:
				continue
			while sm_t + sm_i >= m and count_one > 0:
				count_one -= 1
				sm_i -= one[count_one]
			if sm_t + sm_i < m:
				sm_i += one[count_one]
				count_one += 1
			min_con = min(min_con, (i + 1) * 2 + count_one)
		if min_con == 100000000:
			min_con = -1
		stdout.write(str(min_con) + '\n')
main()
