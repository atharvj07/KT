#include <bits/stdc++.h>
using namespace std;
long long int n, m, k, x, y;
long long int max(long long int a, long long int b) { return a > b ? a : b; }
long long int f(long long int a, long long int b) {
  long long int prd = 2 * n * m - 2 * m;
  long long int d = (n == 1 ? k / m : k / prd);
  long long int r = (n == 1 ? k % m : k % prd);
  if (a == 1 || a == n) {
    return d + (r >= (a - 1) * m + b);
  } else {
    if (r >= m * n) {
      return 2 * d + 1 + (r - m * n >= (n - 1 - a) * m + b);
    } else {
      return 2 * d + (r >= (a - 1) * m + b);
    }
  }
}
int main() {
  cin >> n >> m >> k >> x >> y;
  if (n == 1 || n == 2)
    cout << f(1, 1) << " " << f(n, m) << " " << f(x, y) << endl;
  else
    cout << max(max(f(1, 1), f(2, 1)), f(n - 1, 1)) << " " << f(n, m) << " "
         << f(x, y) << endl;
}
