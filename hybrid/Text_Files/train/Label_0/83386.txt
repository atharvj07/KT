I = lambda : int(input())
IL = lambda : list(map(int, input().split()))
(n, k) = IL()
s = input()
l1 = list(map(len, s.split('a')))
l2 = list(map(len, s.split('b')))
ms = cs = sum(l1[:k + 1]) + len(l1[:k + 1]) - 1
for i in range(len(l1) - k - 1):
	cs += l1[i + k + 1] - l1[i]
	ms = max(ms, cs)
cs = sum(l2[:k + 1]) + len(l2[:k + 1]) - 1
ms = max(ms, cs)
for i in range(len(l2) - k - 1):
	cs += l2[i + k + 1] - l2[i]
	ms = max(ms, cs)
print(ms)
