#include <bits/stdc++.h>
using namespace std;
int len, Len[200000 * 2 + 10], n, m;
int a[200000 * 2 + 10], b[200000 + 10];
void read() {
  scanf("%d%d", &n, &m);
  int i;
  for (i = 1; i <= n; i++) scanf("%d", &b[i]);
  for (i = 1; i < n; i++) a[i] = b[i + 1] - b[i];
  for (i = n - 1; i; i--) a[i * 2] = a[i], a[i * 2 - 1] = 2e9 + 1;
  a[n * 2] = 2e9 + 2;
  a[0] = 2e9 + 3;
  len = n * 2;
}
void manacher() {
  int po = 0, p = 0, i, j;
  for (i = 1; i < len; i++) {
    if (p >= i) {
      j = po * 2 - i;
      if (Len[j] <= p - i) {
        Len[i] = Len[j];
        continue;
      } else
        Len[i] = p - i + 1;
    } else
      Len[i] = 1;
    while (a[i - Len[i]] == a[i + Len[i]]) Len[i]++;
    p = i + Len[i] - 1, po = i;
  }
}
vector<int> ans;
bool check(int l, int r) {
  if (l > r) return 1;
  int tl = l * 2 - 1, tr = r * 2 + 1;
  int mid = (tl + tr) / 2;
  return Len[mid] >= r - l + 1;
}
int main() {
  read();
  manacher();
  for (int i = 1; i < n; i++)
    if ((b[1] + b[i]) % m == (b[n] + b[i + 1]) % m && check(1, i - 1) &&
        check(i + 1, n - 1))
      ans.push_back((b[1] + b[i]) % m);
  if (check(1, n - 1)) ans.push_back((b[1] + b[n]) % m);
  sort(ans.begin(), ans.end());
  printf("%d\n", ans.size());
  for (auto x : ans) printf("%d ", x);
}
