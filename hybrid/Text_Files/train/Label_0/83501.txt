n = int(input())
a = [0] + list(map(int, input().split()))
cnt = 0
res = []

def swap(i: int, j: int):
	(a[i], a[j]) = (a[j], a[i])
pos = [0] * (n + 1)
for i in range(1, n + 1):
	pos[a[i]] = i
for i in range(1, n + 1):
	p = pos[i]
	if p == i:
		continue
	elif abs(p - i) * 2 >= n:
		swap(i, p)
		cnt += 1
		res.append((i, p))
	elif (i - 1) * 2 >= n:
		swap(i, p)
		cnt += 3
		res.extend([(1, i), (1, p), (1, i)])
	elif (p - 1) * 2 < n:
		swap(i, p)
		cnt += 3
		res.extend([(p, n), (i, n), (p, n)])
	else:
		swap(i, p)
		cnt += 5
		res.extend([(i, n), (1, p), (1, n), (i, n), (1, p)])
	pos[a[p]] = p
print(cnt)
print('\n'.join((str(i) + ' ' + str(j) for (i, j) in res)))
