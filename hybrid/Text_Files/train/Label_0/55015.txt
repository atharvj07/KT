import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
public class C493 {

	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int a = scan.nextInt();
		long x = scan.nextLong();
		long y = scan.nextLong();
		long ans = 0;
		int[] arr =new int[a];
		String s = scan.next();
		
		Pattern pat = Pattern.compile("0+");
		Pattern pat1 = Pattern.compile("1+");
		Matcher matcher = pat.matcher(s);
		Matcher matcher1 = pat1.matcher(s);
		long count = 0;
		long count1 =0;
		while(matcher1.find())count1++;
		while (matcher.find()) count++;
		
		for(int i = 0;i<a;i++){
			arr[i] = s.charAt(i)-'0';
		}
		if(!s.contains("0")){
			System.out.println(0);
			return;
		}
		if(s.charAt(0)=='1'){
			count1--;
		}
		if(s.charAt(s.length()-1)=='1'){
		    count1--;
		}
		long cost = count1*x+y;
		long cost1 = count*y;
		System.out.println(Math.min(cost1, cost));
	}

}
