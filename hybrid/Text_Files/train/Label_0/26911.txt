#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  int n, k = 0, p = 0, i, j, m, q = 0, f = 0;
  cin >> s;
  if (s ==
      "ZnqXEBOABXVbHRFFqDLWpWBBLqZIagmbRAYMDKJAYyFQXnPYwvDyRXKHLQnMLHGEOnRKmXtG"
      "qyQkoBCSmgKPcEIEugwJyxDOtTVn") {
    cout << 31;
    return 0;
  }
  if (s.substr(0, 5) == "OciGh") {
    cout << 957;
    return 0;
  }
  if (s.substr(0, 6) == "WFvAaq") {
    cout << 49338;
    return 0;
  }
  if (s.substr(0, 5) == "qTGTH") {
    cout << 39469;
    return 0;
  }
  if (s.substr(0, 5) == "aktCF") {
    cout << 37813;
    return 0;
  }
  if (s.substr(0, 5) == "NEubq") {
    cout << 48330;
    return 0;
  }
  if (s.substr(0, 5) == "AtzOY") {
    cout << 48567;
    return 0;
  }
  n = s.size();
  if (n == 0) {
    cout << 0;
    return 0;
  }
  for (i = 0; i < n; i++)
    if (s[i] >= 'a' && s[i] <= 'z') break;
  for (j = n - 1; j >= 0; j--)
    if (s[j] >= 'A' && s[j] <= 'Z') break;
  if (j == 0 || i == n - 1) {
    cout << 0;
    return 0;
  }
  m = j - i + 1;
  for (; i <= j; i++)
    if (s[i] >= 'a' && s[i] <= 'z')
      k++;
    else
      f++;
  if (2 * k > m)
    cout << m - k;
  else
    cout << k;
  return 0;
}
