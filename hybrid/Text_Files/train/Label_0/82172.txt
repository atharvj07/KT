n = int(input())
lst = list(map(int, input().split()))
c = int(input())
clst = list(map(lambda x: int(x) - 1, input().split()))
lst = sorted(lst, key=lambda x: x, reverse=True)
sleft = [0 for _ in range(n)]
sright = [0 for _ in range(n)]
cnt = 0
for i in range(n):
	cnt += lst[i]
	sright[i] = cnt
for i in range(n):
	cnt -= lst[i]
	sleft[i] = cnt
for coupon in clst:
	print(sright[coupon] - lst[coupon] + sleft[coupon])
