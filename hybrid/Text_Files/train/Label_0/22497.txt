#include <bits/stdc++.h>
using namespace std;
int n, m, k;
bool debug = false;
int c[4];
int main() {
  scanf("%d", &n);
  string a, b, t;
  map<int, int> ss;
  bool bl;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (i) cin >> t;
    cin >> a >> b;
    c[0] = a[0] - '0';
    c[1] = a[1] - '0';
    c[2] = b[1] - '0';
    c[3] = b[0] - '0';
    bl = true;
    for (int j = 0; j < 4; j++) {
      m = c[j % 4] * 1000 + c[(j + 1) % 4] * 100 + c[(j + 2) % 4] * 10 +
          c[(j + 3) % 4];
      if (ss.find(m) != ss.end()) {
        ss[m]++;
        bl = false;
        break;
      }
    }
    if (bl) ss[m] = 1;
  }
  printf("%d\n", ss.size());
  return 0;
}
