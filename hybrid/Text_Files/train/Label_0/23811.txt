#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 1000;
long long v[N];
vector<int> in, out, ans1, ans2;
vector<long long> ans3;
int n, m;
int main() {
  scanf("%d %d", &n, &m);
  while (m--) {
    int x, y;
    long long z;
    cin >> x >> y >> z;
    v[x] += z;
    v[y] -= z;
  }
  for (int i = 1; i <= n; i++) {
    if (v[i] > 0) out.push_back(i);
    if (v[i] < 0) in.push_back(i);
  }
  int i = 0, j = 0;
  while (1) {
    if (i == out.size() && j == in.size()) break;
    long long value = min(v[out[i]], -v[in[j]]);
    ans1.push_back(out[i]);
    ans2.push_back(in[j]);
    ans3.push_back(value);
    v[out[i]] -= value;
    v[in[j]] += value;
    if (v[out[i]] == 0) i++;
    if (v[in[j]] == 0) j++;
  }
  cout << ans1.size() << endl;
  for (int i = 0; i < ans1.size(); i++) {
    cout << ans1[i] << " " << ans2[i] << " " << ans3[i] << endl;
  }
  return 0;
}
