import math

def powerMod(x, y):
	res = 1
	while y > 0:
		if y & 1:
			res = res * x
		y = y >> 1
		x = x * x
	return res
n = int(input())
a = list(map(int, input().split()))
(c1, c2) = (0, 0)
a.sort()
x = a[n - 1] ** (1 / (n - 1))
x1 = math.ceil(x)
x2 = math.floor(x)
for i in range(n):
	d1 = abs(a[i] - pow(x1, i))
	c1 += d1
	d2 = abs(a[i] - pow(x2, i))
	c2 += d2
print(min(c1, c2))
