#include <bits/stdc++.h>
using namespace std;
struct item {
  long long ans1, ans2;
  bool w;
  item(){};
};
bool operator<(item a, item b) {
  return ((a.w == false && b.w == true) || (a.w == b.w && a.ans1 < b.ans1) ||
          (a.w == b.w && a.ans1 == b.ans1 && a.ans2 > b.ans2));
}
map<string, long long> ss;
set<string> use;
map<string, item> res;
vector<string> w;
item calc(string s) {
  use.insert(s);
  item ans;
  ans.w = false;
  ans.ans1 = 0;
  ans.ans2 = 0;
  for (int i = 0; i < 52; i++) {
    string t;
    if (i < 26)
      t = s + char('a' + i);
    else
      t = char('a' + i - 26) + s;
    if (ss.find(t) == ss.end()) continue;
    long long maxx = 0, cost = 0;
    for (int j = 0; j < t.size(); j++) {
      cost += int(t[j]) - 'a' + 1;
      maxx = max(maxx, 1ll * int(t[j]) - 'a' + 1);
    }
    cost = cost * maxx;
    item k;
    if (use.find(t) == use.end())
      k = calc(t);
    else
      k = res[t];
    k.w = 1 - k.w;
    swap(k.ans1, k.ans2);
    k.ans1 = k.ans1 + cost + ss[t];
    if (ans < k) ans = k;
  }
  res[s] = ans;
  return ans;
}
int main() {
  int n;
  scanf("%d\n", &n);
  w.resize(n);
  for (int i = 0; i < n; i++) {
    getline(cin, w[i]);
    for (int j = 0; j < w[i].size(); j++) {
      string s1 = "";
      for (int k = j; k < w[i].size(); k++) {
        s1 += w[i][k];
        if (use.find(s1) != use.end())
          continue;
        else
          ss[s1]++;
        use.insert(s1);
      }
      s1.clear();
    }
    use.clear();
  }
  item ans = calc("");
  if (ans.w == 0)
    cout << "Second" << endl;
  else
    cout << "First" << endl;
  cout << ans.ans1 << ' ' << ans.ans2;
  return 0;
}
