#include <bits/stdc++.h>
using namespace std;
int n, start, x, nxt, value;
vector<char> used;
int cur = 1999;
vector<unsigned int> tt;
int main(int argc, char *argv[]) {
  for (int i = 0; i <= 5000; ++i) {
    tt.push_back(i * (unsigned int)(int(1e9) + 7));
  }
  scanf("%d%d%d", &n, &start, &x);
  used.resize(n, false);
  --start;
  used[start] = true;
  printf("? %d\n", start + 1);
  fflush(stdout);
  scanf("%d%d", &value, &nxt);
  if (nxt != -1) {
    --nxt;
  }
  --cur;
  int cur_n = n - 1;
  for (int i = 1; i * i <= n && cur_n > 0; ++i) {
    int a = tt.back() % n;
    tt.pop_back();
    while (used[a]) {
      a = tt.back() % n;
      tt.pop_back();
    }
    --cur_n;
    used[a] = true;
    printf("? %d\n", a + 1);
    fflush(stdout);
    int cur_value, cur_nxt;
    scanf("%d%d", &cur_value, &cur_nxt);
    if (cur_value <= x && cur_value > value) {
      value = cur_value;
      nxt = cur_nxt;
      if (nxt != -1) {
        --nxt;
      }
      start = a;
    }
    --cur;
  }
  if (n == 50000) {
  }
  while (cur > 0 && value < x) {
    int u = nxt;
    if (u != -1) {
      printf("? %d\n", u + 1);
      fflush(stdout);
      scanf("%d%d", &value, &nxt);
      if (nxt != -1) {
        --nxt;
      }
    } else {
      break;
    }
    --cur;
  }
  if (value < x) {
    printf("! -1");
  } else {
    printf("! %d", value);
  }
  return 0;
}
