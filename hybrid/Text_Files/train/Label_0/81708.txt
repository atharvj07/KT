(n, x, a) = map(int, input().split())
ans = 1
cnt = 0
cnt1 = 0
left = 0
right = n
mod = 1000000007
while left < right:
	mid = (left + right) // 2
	if mid < a:
		ans *= x - 1 - cnt1
		ans %= mod
		cnt1 += 1
		left = mid + 1
	elif mid > a:
		ans *= n - x - cnt
		ans %= mod
		cnt += 1
		right = mid
	else:
		left = mid + 1
cr = n - cnt - cnt1 - 1
for i in range(1, cr + 1):
	ans *= i
	ans %= mod
print(ans)
