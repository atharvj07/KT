int n;
vi G[maxn],sk;
int isroot[maxn];
int sz[maxn];

int dfs(int u,int fa)
{
    sk.pb(u);
    isroot[u] = 0;
    for(auto v:G[u])
    {
        if(v==fa)continue;
        if(isroot[v]==-1)
        {
            if(dfs(v,u))return 1;
        }
        else
        {
            for(;sk.back()!=v;sk.pop_back())
            {
                isroot[sk.back()] = 1;
            }
            isroot[v] = 1;
            return 1;
        }
    }
    sk.pop_back();
    return 0;
}

int Dp(int u,int fa)
{
    int sz = 1;
    for(auto v:G[u])
    {
        if(v==fa||isroot[v]==1)continue;
        sz+=Dp(v,u);
    }
    return sz;
}

inline LL solve()
{
    sk.clear();
    fill(isroot,isroot+n+1,-1);
    assert(dfs(1,1)==1);
    LL ans = 0;
    _rep(i,1,n)
    {
        if(isroot[i]==1)
        {
            LL tmp = Dp(i,i);
            ans+=tmp*(tmp-1)/2+tmp*(n-tmp);
        }
    }
    return ans;
}
