#include <bits/stdc++.h>
using namespace std;
struct bl {
  int a, b;
};
bool cmp(bl a, bl b) {
  if (a.a < b.a) return true;
  return false;
}
int main() {
  int n;
  scanf("%d", &n);
  vector<bl> v(2 * n - 2);
  for (int i = 0; i < n - 1; ++i) {
    scanf("%d%d", &v[2 * i].a, &v[2 * i].b);
    v[2 * i + 1].a = v[2 * i].b;
    v[2 * i + 1].b = v[2 * i].a;
  }
  sort(v.begin(), v.end(), cmp);
  vector<int> w(2 * n - 2);
  for (int i = 0; i < 2 * n - 2; ++i) w[i] = v[i].a;
  vector<bl> z(n + 1);
  for (int i = 0; i <= n; ++i) {
    z[i].a = -1;
    z[i].b = 0;
  }
  queue<int> q;
  q.push(1);
  z[1].a = 0;
  while (!q.empty()) {
    int a = q.front();
    q.pop();
    int b = lower_bound(w.begin(), w.end(), a) - w.begin();
    while (w[b] == a) {
      int c = v[b].b;
      if (z[c].a == -1) {
        z[c].a = a;
        ++z[a].b;
        q.push(c);
      }
      ++b;
    }
  }
  long long k = 0;
  for (int i = n; i > 0; --i) {
    int a = z[i].a;
    --z[a].b;
    if (z[a].b > 0) k += z[a].b;
    if (a > 1) ++k;
  }
  printf("%d\n", k);
  return 0;
}
