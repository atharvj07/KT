import collections, heapq, sys
(n, m) = map(int, sys.stdin.readline().rstrip().split(' '))
a = []
for _ in range(m):
	(x, y) = map(int, sys.stdin.readline().rstrip().split(' '))
	a.append([x, y])
(x, y) = a[0]

def test(x):
	for i in range(1, m):
		(q, w) = a[i]
		if q != x and w != x:
			fst = True
			snd = True
			for j in range(i + 1, m):
				if x in a[j]:
					continue
				if q not in a[j]:
					fst = False
				if w not in a[j]:
					snd = False
				if not fst and (not snd):
					return False
			return True
	return True
if test(x) or test(y):
	print('YES')
else:
	print('NO')
