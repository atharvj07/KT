import java.io.*;
import java.util.regex.*;
import java.util.*;
public class C
{
	public static void main(String[] args)
	{
		Pattern p=Pattern.compile(" ");
		String fir="(\\w*(lios|liala)\\b)*\\w*(etr|etra)\\b";
		String adj="(lios|liala)\\b";
		String nou="(etr|etra)\\b";
		String ver="(initis|inites)\\b";
		Pattern pm=Pattern.compile("(lios|etr|initis)\\b");
		Pattern pf=Pattern.compile("(liala|etra|inites)\\b");
		String str;
		Scanner sc=new Scanner(System.in);
		str=sc.nextLine();
		Matcher mm=pm.matcher(str);
		Matcher mf=pf.matcher(str);
		Matcher ma=Pattern.compile(adj).matcher(str);
		Matcher mn=Pattern.compile(nou).matcher(str);
		Matcher mv=Pattern.compile(ver).matcher(str);
		Matcher mfir=Pattern.compile(fir).matcher(str);
		if(!p.matcher(str).find())
		{
			if(ma.find()||mn.find()||mv.find())
				System.out.println("YES");
			else System.out.println("NO");
			return ;
		}
		int sp=0;
		Matcher msp=p.matcher(str);
		while(msp.find())++sp;
		int m=0,f=0;
		while(mm.find())++m;
		while(mf.find())++f;
	//	System.out.println(m+"\t"+f+"\t"+sp);
		if(m*f!=0||sp!=m+f-1)
		{
			System.out.println("NO");return ;
		}
		if(!mfir.find()){System.out.println("NO");return;}
		int s=mfir.end();
//		System.out.println(s);
//		System.out.println(mn.find(s));
		if(mn.find(s)||ma.find(s)){System.out.println("NO");return;}
		else 
		{
			if(mv.find()&&mv.end()<s){System.out.println("NO");return;}
		}
		System.out.println("YES");
	}
}
