#include <bits/stdc++.h>
using namespace std;
const int N = 0;
string s;
int main() {
  cin >> s;
  int ans = 0;
  for (int i = s.length() - 1; i >= 0; i--) {
    int j = i;
    ans++;
    while (s[j] == '0') j--;
    if (i - j + 1 > j || i - j + 1 == j && s[0] < s[j]) break;
    i = j;
  }
  cout << ans;
}
