#include <bits/stdc++.h>
using namespace std;
long long int res[1005][1005];
void solve() {
  long long int n, a, b;
  cin >> n >> a >> b;
  if ((a > 1 && b > 1) || (n > 1 && n <= 3 && a == 1 && b == 1)) {
    cout << "NO";
    exit(0);
  }
  long long int t = max(a, b);
  for (long long int i = 0; i < n - t; i++) res[i][i + 1] = res[i + 1][i] = 1;
  if (a == 1) {
    for (long long int i = 0; i < n; i++)
      for (long long int j = 0; j < n; j++)
        if (i != j) res[i][j] ^= 1;
  }
  cout << "YES\n";
  for (long long int i = 0; i < n; i++) {
    for (long long int j = 0; j < n; j++) cout << res[i][j];
    cout << "\n";
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  long long int qc = 1;
  for (long long int i = 1; i <= qc; i++) solve();
}
