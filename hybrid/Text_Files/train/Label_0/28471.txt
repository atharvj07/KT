#include <cstdio>
#include <cstring>
#include <algorithm>
#include <string>
#include <cmath>
#include <stack>
#include <vector>
#include <map>
#include <set>
#include <queue>
#include <list>
#include <iostream>
#include <climits>
#include <cfloat>

using namespace std;


int main()
{
	while (true)
	{
		int n;
		scanf("%d", &n);
		if (n == 0)
			break;

		int tall[128][128];
		int up[128][128];
		memset(tall, 0, sizeof(tall));
		memset(up, 0, sizeof(up));

		for (int i = 0; i < n; ++i)
			for (int j = 0; j < n; ++j)
				scanf("%d", &tall[i][j]);

		for (int i = 0; i < n; ++i)
		{
			int minTall = 1000000;
			int maxTall = -1;
			for (int j = 0; j < n; ++j)
			{
				minTall = min(minTall, tall[i][j]);
				maxTall = max(maxTall, tall[j][i]);
			}
			for (int j = 0; j < n; ++j)
			{
				if (tall[i][j] == minTall)
					up[i][j] |= 1;
				if (tall[j][i] == maxTall)
					up[j][i] |= 2;
			}
		}
		bool any = false;
		for (int i = 0; i < n; ++i)
		{
			for (int j = 0; j < n; ++j)
			{
				if (up[i][j] == 3)
				{
					printf("%d\n", tall[i][j]);
					any = true;
					break;
				}
			}
			if (any)
				break;
		}
		if (!any)
			printf("0\n");
	}

	return 0;
}
