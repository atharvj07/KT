for _ in range(int(input())):
	(n, p, k) = map(int, input().split())
	a = list(map(int, input().split()))
	a = sorted(a)
	ans = 0
	sm = 0
	ct = 0
	for i in range(k + 1):
		if i >= len(a):
			break
		if sm > p:
			break
		ct = i
		pseudocost = sm
		j = i + k - 1
		while j < n:
			if p >= pseudocost + a[j]:
				pseudocost += a[j]
				ct += k
			else:
				break
			j += k
		sm += a[i]
		ans = max(ans, ct)
	print(ans)
