def main():
	t = int(input())
	for i in range(t):
		solve()

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	s = sum(a)
	counter = [[_, 0] for _ in a]
	for i in range(n - 1):
		(u, v) = map(int, input().split())
		counter[u - 1][1] += 1
		counter[v - 1][1] += 1
	counter = sorted(counter, key=keyf, reverse=True)
	for i in range(len(counter) - 1, -1, -1):
		if counter[i][1] <= 1:
			counter.pop(i)
	print(s, end=' ')
	for i in range(n - 2):
		s += counter[0][0]
		counter[0][1] -= 1
		if counter[0][1] <= 1:
			counter.pop(0)
		print(s, end=' ')
	print()

def keyf(e):
	return e[0]
main()
