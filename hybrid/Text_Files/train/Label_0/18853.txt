#include <bits/stdc++.h>
#define N 20005
using namespace std;
typedef long long ll;
ll ri[N],n;
int main()
{
    scanf("%lld",&n);
    ll i,x;
    for(i=1;i<=n;i++){
        scanf("%lld",&x);
        ri[x]=i;
    }
    for(i=1;i<=n;i++){
        printf("%lld ",30000*i);
    }
    printf("\n");
    for(i=1;i<=n;i++){
        printf("%lld ",30000*(n-i)+ri[i]);
    }
}
