t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	l = 0
	r = n - 1
	left = []
	right = []
	while l < r:
		while l < n and s[l] != '1':
			l += 1
		while r >= 0 and s[r] != '0':
			r -= 1
		if l >= r:
			break
		left.append(l + 1)
		right.append(r + 1)
		l += 1
		r -= 1
	l = len(left)
	r = len(right)
	right = right[::-1]
	if l + r == 0:
		print('0')
	else:
		print('1')
		print(l + r, end=' ')
		for e in left:
			print(e, end=' ')
		for e in right:
			print(e, end=' ')
		print()
