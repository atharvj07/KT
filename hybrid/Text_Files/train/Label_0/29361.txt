#include <bits/stdc++.h>
using namespace std;
const int INF = int(1e9);
const int N = 5005;
int a[N], temp[N];
struct {
  int t;
  int l;
  int r;
  int d;
} query[N];
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    a[i] = INF;
  }
  for (int i = 1; i <= m; i++) {
    int t, l, r, d;
    cin >> t >> l >> r >> d;
    query[i].t = t;
    query[i].l = l;
    query[i].r = r;
    query[i].d = d;
  }
  for (int i = m; i >= 1; i--) {
    int t = query[i].t;
    int l = query[i].l;
    int r = query[i].r;
    int d = query[i].d;
    if (t == 2) {
      for (int i = l; i <= r; i++) {
        a[i] = min(a[i], d);
      }
    } else {
      for (int i = l; i <= r; i++) {
        a[i] = a[i] - d;
        a[i] = min(a[i], INF);
        a[i] = max(a[i], -INF);
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    temp[i] = a[i];
  }
  for (int i = 1; i <= m; i++) {
    int t = query[i].t;
    int l = query[i].l;
    int r = query[i].r;
    int d = query[i].d;
    if (t == 1) {
      for (int i = l; i <= r; i++) {
        temp[i] += d;
        temp[i] = min(temp[i], INF);
        temp[i] = max(temp[i], -INF);
      }
    } else {
      int mx = -INF - 1;
      for (int i = l; i <= r; i++) {
        mx = max(mx, temp[i]);
      }
      if (mx != d) {
        cout << "NO";
        return 0;
      }
    }
  }
  cout << "YES" << endl;
  for (int i = 1; i <= n; i++) {
    cout << a[i] << " ";
  }
  return 0;
}
