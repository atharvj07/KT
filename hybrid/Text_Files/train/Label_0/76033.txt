from math import ceil

def calc(n, k, a):
	a.sort()
	curr = 0
	if n == 1:
		return sum(a)
	if n == 2:
		for i in range(0, len(a), 2):
			curr += a[i]
	else:
		j = len(a) - 1 - n // 2
		c = 0
		while c < k:
			curr += a[j]
			j -= n // 2 + 1
			c += 1
	return curr
t = int(input())
for i in range(t):
	(n, k) = list(map(int, input().split()))
	a = list(map(int, input().split()))
	print(calc(n, k, a))
