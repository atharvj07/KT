#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  long long A, total = 0;
  scanf("%d%lld", &n, &A);
  vector<int> d(n);
  for (int &i : d) {
    scanf("%d", &i);
    total += i;
  }
  for (int i : d) {
    total -= i;
    int l = 1, r = min((long long)i, A - (n - 1));
    if (total < A) {
      l = A - total;
    }
    printf("%d ", i - (r - l + 1));
    total += i;
  }
  return 0;
}
