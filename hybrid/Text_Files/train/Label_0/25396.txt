#include <bits/stdc++.h>
using namespace std;
int const N = 100100;
vector<pair<pair<int, int>, int> > p;
bool f(pair<int, int> p1, pair<int, int> p2, pair<int, int> p3) {
  return (
      (long long)(p1.second - p2.second) * (long long)(p1.first - p3.first) !=
      (long long)(p1.second - p3.second) * (long long)(p1.first - p2.first));
}
int main() {
  ios_base::sync_with_stdio(false);
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    p.push_back({{x, y}, i + 1});
  }
  sort((p).begin(), (p).end());
  pair<pair<int, int>, int> a = p[0], b = p[1], c;
  for (int i = 2; i < n; i++) {
    c = p[i];
    if (f(a.first, b.first, c.first)) {
      cout << a.second << " " << b.second << " " << c.second << "\n";
      return 0;
    }
  }
  return 0;
}
