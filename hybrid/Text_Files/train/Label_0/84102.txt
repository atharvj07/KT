def solve(n, s):
	cnt = 0
	for i in s:
		if i == '1':
			cnt += 1
	if cnt % 2 == 1 or s[0] == '0' or s[-1] == '0':
		print('NO')
		return
	k = 0
	flip = 0
	a = ''
	b = ''
	for i in range(n):
		if s[i] == '1':
			if 2 * k < cnt:
				a += '('
			else:
				a += ')'
			b += a[-1]
			k += 1
		else:
			if flip == 1:
				a += '('
				b += ')'
			else:
				a += ')'
				b += '('
			flip = not flip
	print('YES')
	print(a)
	print(b)
	return
for t in range(int(input())):
	n = int(input())
	s = input()
	solve(n, s)
