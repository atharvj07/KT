def hv_test(a, b, c):
	if a < b > c or a > b < c:
		return True
	else:
		return False
t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(lambda x: int(x), input().split()))
	is_hv = [0] * n
	ans = 0
	if n > 2:
		i = 1
		while i < n - 1:
			if hv_test(arr[i - 1], arr[i], arr[i + 1]):
				is_hv[i] = 1
				ans += 1
			i += 1
		total = ans
		i = 1
		while i < n - 1:
			ans1 = total
			ans2 = total
			if is_hv[i]:
				tmp = 1
				x = arr[i - 1]
				i -= 1
				if i - 1 >= 0:
					t1 = is_hv[i]
					t2 = hv_test(arr[i - 1], arr[i], x)
					if t1 and (not t2):
						tmp += 1
					elif not t1 and t2:
						tmp -= 1
				i += 2
				if i + 1 < n:
					t1 = is_hv[i]
					t2 = hv_test(x, arr[i], arr[i + 1])
					if t1 and (not t2):
						tmp += 1
					elif not t1 and t2:
						tmp -= 1
				i -= 1
				ans1 = min(ans1, total - tmp)
				tmp = 1
				x = arr[i + 1]
				i -= 1
				if i - 1 >= 0:
					t1 = is_hv[i]
					t2 = hv_test(arr[i - 1], arr[i], x)
					if t1 and (not t2):
						tmp += 1
					elif not t1 and t2:
						tmp -= 1
				i += 2
				if i + 1 < n:
					t1 = is_hv[i]
					t2 = hv_test(x, arr[i], arr[i + 1])
					if t1 and (not t2):
						tmp += 1
					elif not t1 and t2:
						tmp -= 1
				i -= 1
				ans2 = min(ans2, total - tmp)
				ans = min(ans, ans1, ans2)
			i += 1
	print(ans)
