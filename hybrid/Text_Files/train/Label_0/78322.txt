def g_eq(a1, a2):
	for i in range(len(a1)):
		if a1[i] < a2[i]:
			return False
	return True

def sharpen(arr, arr_len):
	k = arr_len // 2
	to_compare = list(range(k)) + [k] + list(range(k)[::-1])
	if arr_len % 2 == 0:
		to_compare = [to_compare[1:], to_compare[:-1], to_compare[:k - 1] + to_compare[k:], to_compare[:k + 1] + to_compare[k + 2:]]
	else:
		to_compare = [to_compare]
	for i in to_compare:
		if g_eq(arr, i):
			return 'Yes'
	return 'No'
t = int(input())
for _ in range(t):
	l = int(input())
	a = [int(i) for i in input().split()]
	print(sharpen(a, l))
