import java.util.Scanner;

public class A318 {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        long len = sc.nextLong();
        long pos = sc.nextLong();
        long odd = (len / 2) + (len % 2 == 0 ? 0 : 1);
        long ans = 0;
        if (pos <= odd){
            long x = (pos - 1);
            ans = 2 * x + 1;
        }else {
            ans = 2 * (pos - odd);
        }
        System.out.println(ans);
        sc.close();
    }
}
