def isVowel(ch):
	if ch == 'A' or ch == 'E' or ch == 'I' or (ch == 'O') or (ch == 'U') or (ch == 'Y'):
		return True
	return False
s = input()
last = -1
ans = 1
for i in range(0, len(s)):
	if isVowel(s[i]):
		ans = max(ans, i - last)
		last = i
ans = max(ans, len(s) - last)
print(len(s) + 1 if last == -1 else ans)
