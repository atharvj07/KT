#include <bits/stdc++.h>
using namespace std;
const long double eps = 1e-6;
long long Round(long double x) { return (long long)round(x); }
long double Lound(long double x) { return round(x - 0.5) + 0.5; }
struct Q {
  long double s, x, y, z;
  Q() {
    s = 0;
    cin >> x >> y >> z;
  }
  Q(long double _s, long double _x = 0, long double _y = 0, long double _z = 0)
      : s(_s), x(_x), y(_y), z(_z) {}
  Q operator/(long double v) const { return Q(s / v, x / v, y / v, z / v); }
  long double len() { return s * s + x * x + y * y + z * z; }
  Q conj() { return Q(s, -x, -y, -z); }
  Q inv() { return conj() / len(); }
  Q round() {
    Q q1 = Q(Round(s), Round(x), Round(y), Round(z));
    Q q2 = Q(Lound(s), Lound(x), Lound(y), Lound(z));
    return (q1 - *(this)).len() < (q2 - (*this)).len() ? q1 : q2;
  }
  Q operator+(Q F) const { return Q(s + F.s, x + F.x, y + F.y, z + F.z); }
  Q operator-(Q F) const { return Q(s - F.s, x - F.x, y - F.y, z - F.z); }
  Q operator*(Q F) const {
    return Q(s * F.s - x * F.x - y * F.y - z * F.z,
             s * F.x + x * F.s + y * F.z - z * F.y,
             s * F.y - x * F.z + y * F.s + z * F.x,
             s * F.z + x * F.y - y * F.x + z * F.s);
  }
  Q operator/(Q F) const { return F.inv() * (*this); }
  Q operator%(Q F) const { return (*this) - F * ((*this) / F).round(); }
};
bool isInt(Q F) {
  if (abs(F.s - Round(F.s)) > eps) return false;
  if (abs(F.x - Round(F.x)) > eps) return false;
  if (abs(F.y - Round(F.y)) > eps) return false;
  if (abs(F.z - Round(F.z)) > eps) return false;
  return true;
}
void debug(Q F) {
  cout << F.s << " " << F.x << " " << F.y << " " << F.z << "\n";
}
void print(Q F) {
  cout << Round(F.x) << " " << Round(F.y) << " " << Round(F.z) << "\n";
}
Q gcd(Q A, Q B) { return B.len() <= eps ? A : gcd(B, A % B); }
long long gcd(long long A, long long B) { return B ? gcd(B, A % B) : A; }
vector<Q> V;
vector<long long> T;
int main() {
  int n;
  cin >> n;
  Q g = Q(0);
  long long G = 0;
  while (n--) {
    V.push_back(Q());
    g = gcd(g, V.back());
    G = gcd(G, V.back().len());
  }
  for (long long i = 1; i * i <= G; i++) {
    if (G % (i * i) == 0) T.push_back(i);
  }
  reverse(T.begin(), T.end());
  for (long long c : T) {
    Q q = gcd(g, Q(c));
    if ((long long)q.len() != c) continue;
    Q q1 = q / c, q2 = q.inv();
    bool f = 0;
    for (auto F : V)
      if (!isInt(q2 * F * q1)) {
        f = 1;
        break;
      }
    if (f) continue;
    cout << c * c << "\n";
    print(q * Q(0, 1, 0, 0) * q.conj());
    print(q * Q(0, 0, 1, 0) * q.conj());
    print(q * Q(0, 0, 0, 1) * q.conj());
    break;
  }
  return 0;
}
