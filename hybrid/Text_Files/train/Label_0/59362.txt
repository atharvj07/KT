const golorp = readline().match(/^\?\((.+)\):-(.+)\.$/);
const expression = golorp[1];
const conditions = golorp[2].split(",").map(condition => {
  const match = condition.match(/[<>]/);
  const index = match.index;
  const left = condition.substring(0, index);
  const right = condition.substring(index + 1);
  return match[0] === ">" ? [right, left] : [left, right];
});
const sequence = expression.match(/_+/g);
const variables = Array.from(new Set(sequence));
const nodes = new Map;
const values = new Map;
for (var variable of variables) {
  nodes.set(variable, [variable, new Set, new Set]);
  values.set(variable, -1);
}
for (var condition of conditions) {
  var less = condition[0];
  var greater = condition[1];
  nodes.get(less)[1].add(greater);
  nodes.get(greater)[2].add(less);
}
const seen = new Set;
for (var i = 10; i; i--)
  for (var from of variables) {
    if (seen.has(from)) continue;
    var fromNode = nodes.get(from);
    if (!fromNode[2].size) {
      seen.add(from);
      var newValue = values.get(from) + 1;
      values.set(from, newValue);
      for (var to of fromNode[1]) {
        if (newValue > values.get(to)) values.set(to, newValue);
        nodes.get(to)[2].delete(from);
      }
    }
  }
var found = true;
for (var variable of variables)
  if (!seen.has(variable) || values.get(variable) >= 10) {
    found = false;
    break;
  }
print(found ? sequence.map(variable => values.get(variable)).join("") : "false");
