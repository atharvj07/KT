(s, t) = (input(), input())
d = set()
u = set()
for i in range(len(s)):
	(l1, l2) = (min(s[i], t[i]), max(s[i], t[i]))
	if (s[i] in u or t[i] in u) and (l1, l2) not in d:
		print(-1)
		exit()
	else:
		u.add(s[i])
		u.add(t[i])
		d.add((l1, l2))
ans = [x for x in d if x[0] != x[1]]
print(len(ans))
for a in ans:
	print(*a)
