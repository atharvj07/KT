#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  vector<int> lista[200];
  sort(arr, arr + n);
  int exit = 0;
  for (int i = 0; i < n; i++) {
    for (int c = 0; c < 200; c++) {
      if (lista[c].size() == 0 || arr[i] >= lista[c].size()) {
        exit = max(c + 1, exit);
        lista[c].push_back(1);
        break;
      }
    }
  }
  cout << exit << endl;
  return 0;
}
