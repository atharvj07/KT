import sys
cases = sys.stdin.readline()
numbers = [int(x) for x in sys.stdin.readline().strip().split(' ')]
my_sum = sum(numbers)
answers = []
ones_so_far = 0
zeroes_so_far = 0
index = 0
for x in numbers:
	if x == 1:
		ones_so_far += 1
	else:
		zeroes_so_far += 1
	remaining_ones = my_sum - ones_so_far
	remaining_zeroes = len(numbers) - remaining_ones - index - 1
	a = max(ones_so_far, zeroes_so_far) + remaining_ones
	b = zeroes_so_far + remaining_zeroes
	index += 1
	answers.append(max(a, b))
print(max(answers))
