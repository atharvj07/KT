n = int(input())
maxs = list(map(float, input().split()))
mins = list(map(float, input().split()))
(d1s, d2s) = ([], [])
(cd1sum, cd2sum) = (0, 0)
for i in range(n):
	z = mins[i] + maxs[i]
	c = cd2sum
	d = cd1sum
	s = mins[i]
	b = d - z - c
	c = z - d * z - s
	disc = max(0, b ** 2 - 4 * c) ** 0.5
	ans = min((-b + disc) / 2.0, (-b - disc) / 2.0)
	d1s.append(ans)
	d2s.append(z - ans)
	cd1sum += d1s[-1]
	cd2sum += d2s[-1]
print(' '.join(map(str, d1s)))
print(' '.join(map(str, d2s)))
