for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	a.sort()
	m = a[-1]
	if a[0] != 0:
		if (m + 1) // 2 in a:
			print(n)
		elif k == 0:
			print(n)
		else:
			print(n + 1)
	else:
		mex = 0
		for i in range(n):
			if a[i] != i:
				mex = i
				break
		if mex == 0 or mex == m - 1:
			print(n + k)
		elif (m + mex + 1) // 2 in a:
			print(n)
		elif k == 0:
			print(n)
		else:
			print(n + 1)
