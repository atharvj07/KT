(a, b) = map(int, input().split())
r = lambda x: 2 ** x - 1
k = [[0, 0] for _ in ' ' * b]
ans = 0
for i in range(a):
	z = list(map(int, input().split()))
	w = 0
	for j in range(b):
		k[j][z[j]] += 1
		if z[j] == 0:
			w += 1
	ans += r(w) + r(b - w)
for i in range(b):
	if k[i][0] > 1:
		ans += r(k[i][0]) - k[i][0]
	if k[i][1] > 1:
		ans += r(k[i][1]) - k[i][1]
print(ans)
