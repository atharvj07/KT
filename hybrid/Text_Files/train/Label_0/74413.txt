for _ in range(int(input())):
	input()
	li = list(map(int, input().split()))
	days_not_watered = int(not li[0])
	x = li[0] + 1
	for i in range(1, len(li)):
		if days_not_watered == 2:
			x = -1
			break
		if li[i - 1] == 1 and li[i] == 1:
			x += 5
			days_not_watered = 0
		elif li[i] == 1:
			x += 1
			days_not_watered = 0
		else:
			days_not_watered += 1
			if days_not_watered == 2:
				x = -1
				break
	print(x)
