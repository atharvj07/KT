#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int a = n, b = m, c = 0, d = 0;
  char arr[n][m];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> arr[i][j];
      if (arr[i][j] == '*') {
        if (i < a) {
          a = i;
        }
        if (j < b) {
          b = j;
        }
        if (i > c) {
          c = i;
        }
        if (j > d) {
          d = j;
        }
      }
    }
  }
  for (int i = a; i <= c; i++) {
    for (int j = b; j <= d; j++) {
      cout << arr[i][j];
    }
    cout << endl;
  }
  return 0;
}
