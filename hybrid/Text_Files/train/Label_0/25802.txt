#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 2e5 + 1;
int solve();
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  solve();
  return 0;
}
int n;
vector<int> g[MAX_N];
set<int> s[MAX_N];
bool used[MAX_N];
vector<int> ans;
void dfs(int v) {
  used[v] = true;
  ans.push_back(v);
  for (int to : g[v]) {
    if (!used[to] && s[v].find(to) != s[v].end()) {
      dfs(to);
      return;
    }
  }
}
int solve() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    int a;
    int b;
    cin >> a >> b;
    g[a].push_back(b);
    g[b].push_back(a);
    s[i].insert(a);
    s[i].insert(b);
  }
  memset(used, false, sizeof(used));
  dfs(1);
  for (int it : ans) {
    cout << it << ' ';
  }
  return 0;
}
