#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 7;
int a[maxn];
int main() {
  int n;
  long long x;
  scanf("%d%lld", &n, &x);
  long long total = 0;
  for (int i = 1; i <= n; i++) {
    scanf("%lld", &a[i]);
    total += a[i];
  }
  if (x - total == n - 1) {
    cout << "YES\n";
  } else {
    cout << "NO\n";
  }
}
