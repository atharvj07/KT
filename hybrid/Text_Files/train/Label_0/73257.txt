def wis(l1, l2, n, m):
	l2.sort()
	a = 0
	while a != m:
		a = 0
		for i in range(len(l2)):
			if l1[l2[i] - 1] > l1[l2[i]]:
				c = l1[l2[i] - 1]
				l1[l2[i] - 1] = l1[l2[i]]
				l1[l2[i]] = c
			else:
				a = a + 1
	if l1 == sorted(l1):
		print('yes')
	else:
		print('no')
	return
n = int(input())
for i in range(n):
	m = input()
	t = input()
	p = input()
	l1 = list(map(int, t.split(' ')))
	l2 = list(map(int, p.split(' ')))
	wis(l1, l2, len(l1), len(l2))
