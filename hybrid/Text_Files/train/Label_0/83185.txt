def gcd(a, b):
	if a < b:
		(a, b) = (b, a)
	if b == 0:
		return a
	while a % b != 0:
		(a, b) = (b, a % b)
	return b
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	if len(list(set(a))) == 1:
		print(-1)
		continue
	d = []
	for i in range(n):
		for j in range(i + 1, n):
			d.append(abs(a[i] - a[j]))
	e = gcd(d[0], d[1])
	for i in range(2, len(d)):
		e = gcd(e, d[i])
	print(e)
