def solve():
	(n, x) = list(map(int, input().split()))
	a = list(map(int, input().split()))
	b = []
	SU = 0
	for i in a:
		now = 1
		IT = i
		while IT % x == 0:
			IT //= x
			now += 1
		b.append(now)
	mi = min(b)
	index = b.index(mi)
	for i in range(0, index + 1):
		SU += min(b[i], mi + 1) * a[i]
	for i in range(index + 1, n):
		SU += min(b[i], mi) * a[i]
	print(SU)
TTTT = int(input())
while TTTT:
	solve()
	TTTT -= 1
