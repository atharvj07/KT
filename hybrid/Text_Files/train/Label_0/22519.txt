#include <bits/stdc++.h>
using namespace std;
struct segtree {
  vector<pair<int, int>> val;
  int n;
  segtree(int n) : val(4 * (n + 1)), n(n) {}
  void update(int id, int l, int r, int a, pair<int, int> x) {
    if (l == r) {
      val[id].first += x.first;
      val[id].second += x.second;
    } else {
      int mid = (l + r) / 2;
      if (a <= mid)
        update(((id) << 1), l, mid, a, x);
      else
        update((((id) << 1) + 1), mid + 1, r, a, x);
      val[id] = min(val[((id) << 1)], val[(((id) << 1) + 1)]);
    }
  }
  pair<int, int> get(int id, int l, int r, int a, int b) {
    if (l == a && r == b)
      return val[id];
    else {
      int mid = (l + r) / 2;
      if (b <= mid)
        return get(((id) << 1), l, mid, a, b);
      else if (a > mid)
        return get((((id) << 1) + 1), mid + 1, r, a, b);
      else
        return min(get(((id) << 1), l, mid, a, mid),
                   get((((id) << 1) + 1), mid + 1, r, mid + 1, b));
    }
  }
  pair<int, int> get(int a, int b) { return get(1, 0, n - 1, a, b); }
  void update(int a, pair<int, int> x) { update(1, 0, n - 1, a, x); }
};
vector<vector<int>> adj;
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  adj.resize(n);
  for (int i = 0; i < m; i++) {
    int a, b;
    scanf("%d %d", &a, &b);
    a--;
    b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  segtree st = segtree(n);
  for (int i = 0; i < n; i++) st.update(i, {0, i});
  st.update(0, {112345, 0});
  for (int i = 0; i < adj[0].size(); i++) st.update(adj[0][i], {1, 0});
  int res = 0;
  for (int i = 1; i < n; i++) {
    pair<int, int> mn = st.get(0, n - 1);
    if (mn.first == i) res++;
    int nv = mn.second;
    st.update(nv, {112345, 0});
    for (int j = 0; j < adj[nv].size(); j++) {
      st.update(adj[nv][j], {1, 0});
    }
  }
  printf("%d\n", res);
  return 0;
}
