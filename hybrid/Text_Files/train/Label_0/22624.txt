#include<bits/stdc++.h>
using namespace std;

#define int long long
#define pb push_back 
#define pf push_front 
#define mp make_pair
#define fr first
#define sc second
#define Rep(i,n) for(int i=0;i<(n);i++)
#define All(v) v.begin(),v.end()

typedef pair<int, int> Pii; typedef pair<int, Pii> Pip;
const int INF = 1107110711071107;

int N, M, K;
vector<Pii> ab;
int a[16], b[16];
int dp[16][10001][16];

int dfs(int n, int m, int k) {
  if ( n == N || k == K || m == 0 ) return m;
  if ( dp[n][m][k] >= 0 ) return dp[n][m][k];

  int a = ab[n].fr, b = ab[n].sc;
  return dp[n][m][k] = max(dfs(n+1, max(0LL, m-b), k+1) + min(m, b)*a, dfs(n+1, m, k));
}

signed main() {
  memset(dp, -1, sizeof(dp));
  cin >> N >> M >> K;
  Rep(i, N) cin >> a[i];
  Rep(i, N) cin >> b[i];
 
  Rep(i, N) {
    ab.pb(Pii(a[i], b[i]));
  }
  
  sort(All(ab), greater<Pii>());

  cout << dfs(0, M, 0) << endl;
  
  return 0;
}
