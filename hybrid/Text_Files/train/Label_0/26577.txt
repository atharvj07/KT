#include <bits/stdc++.h>
using namespace std;
long long a, b, c;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> a >> b >> c;
  if (c == 0) {
    if (a == b)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
    return 0;
  }
  if (c > 0) {
    if (a == b || (a < b && (b - a) % c == 0))
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  } else {
    if (a == b || (a > b && (a - b) % c == 0))
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
