(n, m) = map(int, input().split())
cn = []
un = 0
d = {}
for i in range(n):
	(p, c) = map(int, input().split())
	if p == 1:
		un += 1
	else:
		cn.append((p, c))
cn.sort(key=lambda x: x[1])
for i in range(len(cn)):
	(p, c) = cn[i]
	if p not in d:
		d[p] = [(c, i)]
	else:
		d[p].append((c, i))
min_val = 99999999999999999999999
for i in range(1, n // 2 + 2):
	total = 0
	v = set()
	for x in d:
		if len(d[x]) >= i:
			r = len(d[x]) - (i - 1)
			for j in range(0, r):
				total += d[x][j][0]
				v.add(d[x][j][1])
	for j in range(n):
		if len(v) + un >= i:
			break
		if j not in v:
			total += cn[j][1]
			v.add(j)
	min_val = min(total, min_val)
print(min_val)
