

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;

public class C {
    static boolean[][] L = new boolean[7][7];
    static int[] E = new int[3];
    static int diff = Integer.MAX_VALUE;
    static int like = 0;
    static int[] As = new int[7];

    public static void rec(int x) {
        if (x == 7) {
            int[] W = new int[3];
            for (int i = 0; i < 7; i++)
                W[As[i]]++;
            if (W[0] * W[1] * W[2] == 0)
                return;
            int max = 0, min = Integer.MAX_VALUE;
            for (int i = 0; i < 3; i++) {
                max = Math.max(E[i] / W[i], max);
                min = Math.min(E[i] / W[i], min);
            }
            int d = max - min;
            int l = 0;
            for (int i = 0; i < 7; i++)
                for (int j = 0; j < 7; j++)
                    if (As[i] == As[j] && L[i][j])
                        l++;
            if (d < diff | (d <= diff && l > like)) {
                diff = d;
                like = l;
            }
            return;
        }
        for (int i = 0; i < 3; i++) {
            As[x] = i;
            rec(x + 1);
        }
    }

    public static void main(String[] args) throws NumberFormatException,
            IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] A = { "Anka", "Chapay", "Cleo", "Troll", "Dracul", "Snowy",
                "Hexadecimal" };
        HashMap<String, Integer> H = new HashMap<String, Integer>();
        for (int i = 0; i < 7; i++)
            H.put(A[i], i);
        int n = Integer.parseInt(in.readLine());
        for (int i = 0; i < n; i++) {
            String[] S = in.readLine().split(" ");
            L[H.get(S[0])][H.get(S[2])] = true;
        }
        String[] S = in.readLine().split(" ");
        for (int i = 0; i < 3; i++)
            E[i] = Integer.parseInt(S[i]);
        rec(0);
        System.out.println(diff + " " + like);
    }
}
