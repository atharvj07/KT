cases = int(input())
for i in range(0, cases):
	(n, x) = [int(i) for i in input().split()]
	numbers = [int(i) for i in input().split()]
	running = 0
	found = False
	for i in range(0, len(numbers)):
		running += numbers[i]
	if found:
		continue
	if running == x:
		print('NO')
	else:
		print('YES')
		running = 0
		for i in range(0, len(numbers)):
			if running + numbers[i] == x:
				temp = numbers[i + 1]
				numbers[i + 1] = numbers[i]
				numbers[i] = temp
			running += numbers[i]
		print(' '.join((str(p) for p in numbers)))
