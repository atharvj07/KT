#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int n;
string s;
int totl, totr;
int cntl, cntr;
void ReadInput() {
  cin >> n;
  cin >> s;
  totl = totr = 0;
  cntl = cntr = 0;
  int mid = n / 2;
  for (int i = 0; i < mid; ++i) {
    if (s[i] != '?')
      totl += s[i] - '0';
    else
      ++cntl;
  }
  for (int i = mid; i < n; ++i) {
    if (s[i] != '?')
      totr += s[i] - '0';
    else
      ++cntr;
  }
}
bool Left() {
  int mxl = 0, mxr = 0;
  int mnl = 0, mnr = 0;
  if (cntl % 2 == 1) {
    int t = cntl / 2 + 1;
    mxl += 9 * t;
    mnl += 9 * (cntl / 2);
    t = cntr / 2 + 1;
    mxr += 9 * t;
    mnr += 9 * (cntr / 2);
  } else {
    int t = cntl / 2;
    mxl += 9 * t;
    mnl += 9 * t;
    t = cntr / 2;
    mxr += 9 * t;
    mnr += 9 * t;
  }
  int diff = mxl - mxr;
  if (diff + totl > totr) return true;
  if (totl + mnl < totr + mnr) return true;
  return false;
}
bool Right() {
  int mxl = 0, mxr = 0;
  int mnl = 0, mnr = 0;
  if (cntl % 2 == 1) {
    int t = cntl / 2 + 1;
    mxl += 9 * t;
    mnl += 9 * (cntl / 2);
    t = cntr / 2 + 1;
    mxr += 9 * t;
    mnr += 9 * (cntr / 2);
  } else {
    int t = cntl / 2;
    mxl += 9 * t;
    mnl += 9 * t;
    t = cntr / 2;
    mxr += 9 * t;
    mnr += 9 * t;
  }
  int diff = mxr - mxl;
  if (diff + totr > totl || totr + mnr < totl + mnl) return true;
  return false;
}
void Solve() {
  if (Left()) {
    cout << "Monocarp\n";
    return;
  }
  if (Right()) {
    cout << "Monocarp\n";
    return;
  }
  cout << "Bicarp\n";
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  ReadInput();
  Solve();
}
