#include <bits/stdc++.h>
using namespace std;
int main() {
  int tst;
  long long n, a, b, c;
  vector<long long> v;
  bool flag;
  cin >> tst;
  while (tst--) {
    flag = false;
    cin >> n;
    for (int i = 2; i <= sqrt(n); i++) {
      if (n % i == 0) {
        v.push_back(i);
        n /= i;
      }
      if (v.size() == 2) {
        v.push_back(n);
        if (v[0] != v[1] && v[1] != v[2]) flag = true;
        break;
      }
    }
    if (flag) {
      cout << "YES" << endl;
      sort(v.begin(), v.end());
      cout << v[0] << " " << v[1] << " " << v[2] << endl;
    } else
      cout << "NO" << endl;
    v.clear();
  }
  return 0;
}
