#include <bits/stdc++.h>
int min(int a, int b) { return a < b ? a : b; }
int max(int a, int b) { return a > b ? a : b; }
int main() {
  static char cc[50][50 + 1];
  static int pp[50][50], qq[50][50], dp[50][50][50][50];
  int n, i, i0, i1, j, j0, j1;
  scanf("%d", &n);
  for (i = 0; i < n; i++) scanf("%s", cc[i]);
  for (i = 0; i < n; i++)
    for (j = 0; j < n; j++)
      if (cc[i][j] == '#') pp[i][j] = qq[i][j] = 1;
  for (i = 0; i < n; i++)
    for (j = 1; j < n; j++) pp[i][j] += pp[i][j - 1];
  for (j = 0; j < n; j++)
    for (i = 1; i < n; i++) qq[i][j] += qq[i - 1][j];
  for (i0 = n - 1; i0 >= 0; i0--)
    for (j0 = n - 1; j0 >= 0; j0--)
      for (i1 = i0; i1 < n; i1++)
        for (j1 = j0; j1 < n; j1++) {
          int x = max(i1 - i0, j1 - j0) + 1, u, v;
          if (i1 - i0 >= j1 - j0) {
            for (i = i0; i <= i1; i++)
              if (pp[i][j1] - (j0 == 0 ? 0 : pp[i][j0 - 1]) == 0) {
                u = (i0 == i ? 0 : dp[i0][j0][i - 1][j1]);
                v = (i1 == i ? 0 : dp[i + 1][j0][i1][j1]);
                x = min(x, u + v);
              }
          } else {
            for (j = j0; j <= j1; j++)
              if (qq[i1][j] - (i0 == 0 ? 0 : qq[i0 - 1][j]) == 0) {
                u = (j0 == j ? 0 : dp[i0][j0][i1][j - 1]);
                v = (j1 == j ? 0 : dp[i0][j + 1][i1][j1]);
                x = min(x, u + v);
              }
          }
          dp[i0][j0][i1][j1] = x;
        }
  printf("%d\n", dp[0][0][n - 1][n - 1]);
  return 0;
}
