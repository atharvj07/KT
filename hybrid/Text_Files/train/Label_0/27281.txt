#include <bits/stdc++.h>
using namespace std;
int r[100010];
long long N, k;
int main() {
  while (cin >> N >> k) {
    memset(r, 0, sizeof(r));
    for (int i = 0; i < N; i++) {
      cin >> r[i];
    }
    long long ans = 0, t;
    t = (sqrt(1 + 8 * k) - 1) / 2;
    ans = k - (t * (t + 1)) / 2;
    if (ans == 0) {
      ans = t;
    }
    cout << r[ans - 1] << endl;
  }
}
