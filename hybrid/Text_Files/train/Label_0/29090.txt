#include <bits/stdc++.h>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int T;
    cin >> T;
    while (T--) {
        int n, m;
        cin >> n >> m;
        vector<int> a(n + 1);
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
        }
        int total = 0;
        vector<int> b(n + 1);
        for (int i = 1; i <= n; i++) {
            cin >> b[i];
            total += b[i];
        }
        vector<int> reg;
        vector<int> imp;
        for (int i = 1; i <= n; i++) {
            if (b[i] == 1) {
                reg.push_back(a[i]);
            } else {
                imp.push_back(a[i]);
            }
        }
        sort(reg.begin(), reg.end(), [&](int x, int y) {
            return x > y;
        });
        sort(imp.begin(), imp.end(), [&](int x, int y) {
            return x > y;
        });
        if (reg.empty()) {
            int current = 0;
            int ptr = 0;
            int ans = -1;
            for (int cost = 2; cost <= total; cost += 2) {
                current += imp[ptr];
                ptr += 1;
                if (current >= m) {
                    ans = cost;
                    break;
                }
            }
            cout << ans << "\n";
            continue;
        }
        int ptr1 = 0;
        int ptr2 = 0;
        int current = 0;
        int ans = -1;
        for (int cost = 1; cost <= total; cost++) {
            int gain1 = INT_MIN;
            int gain2 = INT_MIN;
            if (ptr1 < reg.size()) {
                gain1 = reg[ptr1];
            }
            if (ptr1 > 0 && ptr2 < imp.size()) {
                gain2 = imp[ptr2] - reg[ptr1 - 1];
            }
            if (gain1 == INT_MIN && gain2 == INT_MIN) {
                assert(false);
            }
            if (gain1 > gain2) {
                ptr1 += 1;
                current += gain1;
            } else {
                ptr1 -= 1;
                ptr2 += 1;
                current += gain2;
            }
            if (current >= m) {
                ans = cost;
                break;
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
