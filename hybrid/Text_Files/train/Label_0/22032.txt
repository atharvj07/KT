#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  map<int, int> m;
  set<pair<int, int>> s;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    m[x]++;
  }
  for (auto it = m.begin(); it != m.end(); it++) {
    s.insert({it->second, it->first});
  }
  s.insert({0, -1});
  s.insert({0, -2});
  s.insert({0, -3});
  int q;
  cin >> q;
  while (q--) {
    char c;
    int temp;
    cin >> c >> temp;
    if (c == '-') {
      s.erase({m[temp], temp});
      m[temp]--;
      s.insert({m[temp], temp});
    } else {
      s.erase({m[temp], temp});
      m[temp]++;
      s.insert({m[temp], temp});
    }
    vector<int> v(3);
    auto it = s.rbegin();
    v[0] = (*it).first;
    it++;
    v[1] = (*it).first;
    it++;
    v[2] = (*it).first;
    it++;
    bool res = 0;
    if (v[0] >= 8 || (v[0] >= 6 && v[1] >= 2) || (v[0] >= 4 && v[1] >= 4) ||
        (v[0] >= 4 && v[1] >= 2 && v[2] >= 2))
      res = 1;
    if (res)
      cout << "YES";
    else
      cout << "NO";
    cout << endl;
  }
}
