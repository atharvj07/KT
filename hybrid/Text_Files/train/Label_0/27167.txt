#include<iostream>
#include<map>
#include<string>
using namespace std;

int main()
{
    map<string,int> m;                  //通常のmapはkeyに対して1つの値を持つ。multimapになると、keyにいくつもの値を持たせることが出来る。
    int n, q, x;
    string k;

    cin >> n;

    for( int i = 0; i < n; i++ )
    {
        cin >> q >> k;

        if( q == 0 )
        {
            cin >> x;
            m[k] = x;
        }
        else if( q == 1 )
            cout << m[k] << endl;
        else
            m[k] = 0;
    }
    return 0;
}
