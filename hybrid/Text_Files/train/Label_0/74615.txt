import sys
from collections import Counter
from itertools import zip_longest

def countergenerator(word):
	count = 1
	n = len(word)
	for i in range(1, n):
		if word[i] != word[i - 1]:
			yield (word[i - 1], count)
			count = 1
		else:
			count += 1
	yield (word[n - 1], count)
n = int(sys.stdin.readline().strip())
for pairnumber in range(0, n):
	first = sys.stdin.readline().strip()
	second = sys.stdin.readline().strip()
	res = True
	for (tup1, tup2) in zip_longest(countergenerator(first), countergenerator(second), fillvalue=('\n', 0)):
		if tup1[0] != tup2[0] or tup1[1] > tup2[1]:
			res = False
			break
	if res:
		print('YES')
	else:
		print('NO')
