from collections import defaultdict

def il():
	return list(map(int, input().split()))

def ii():
	return int(input())

def imi():
	return map(int, input().split())

def ist():
	return input()
n = ii()
ar = il()
li = defaultdict(lambda : 0)
li[0] = 1
g = [[0] for i in range(501)]
pg = [defaultdict(lambda : 0) for j in range(501)]
pr = defaultdict(lambda : 0)
for i in ar:
	pr[i] = 1
	pg[i][0] = 1
for a in ar:
	for f in g[a]:
		x = a ^ f
		li[x] = 1
		for b in range(a + 1, 501):
			if pr[b]:
				if pg[b][x] == 0:
					pg[b][x] = 1
					g[b].append(x)
				else:
					break
	g[a] = []
la = []
l = 0
for i in range(250007):
	if li[i]:
		la.append(i)
		l += 1
print(l)
print(*la)
