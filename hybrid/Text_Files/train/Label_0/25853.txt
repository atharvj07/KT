#include <bits/stdc++.h>
using namespace std;
#define FOR(i,k,n) for(int i = (int)(k); i < (int)(n); i++)
#define REP(i,n) FOR(i,0,n)
#define ALL(a) a.begin(), a.end()
#define MS(m,v) memset(m,v,sizeof(m))
typedef long long ll;
typedef long double ld;
typedef vector<int> vi;
typedef vector<string> vs;
typedef pair<int, int> pii;
const int MOD = 1e9 + 7;
template<class T> T &chmin(T &a, const T &b) { return a = min(a, b); }
template<class T> T &chmax(T &a, const T &b) { return a = max(a, b); }
template<class T>
istream& operator >> (istream& is, vector<T>& v)
{
	for (auto &i : v) is >> i;
	return is;
}
template<class T>
ostream& operator<<(ostream& os, vector<T>& v)
{
	const string delimiter = "\n";
	REP(i, v.size())
	{
		os << v[i];
		if (i != v.size() - 1) os << delimiter;
	}
	return os;
}
/*--------------------template--------------------*/

const int INF = 1e9;

int n, m;
vi a, b;
int dp[2][2][51][51][51][51];
bool memo[2][2][51][51][51][51];

int solve(int turn, int pass, int a_stack_btm, int a_deck_top, int b_stack_btm, int b_deck_top)
{
	if (memo[turn][pass][a_stack_btm][a_deck_top][b_stack_btm][b_deck_top]) return dp[turn][pass][a_stack_btm][a_deck_top][b_stack_btm][b_deck_top];
	int res;
	if (turn == 0)
	{
		res = -INF;
		// put on stack
		if (a_deck_top < n)
		{
			chmax(res, solve(1, 0, a_stack_btm, a_deck_top + 1, b_stack_btm, b_deck_top));
		}
		// pass
		if (pass)
		{
			chmax(res, 0);
		}
		else
		{
			int score_a = 0, score_b = 0;
			int ta = a_stack_btm, tb = b_stack_btm;
			int empty = (a_deck_top == a_stack_btm && b_deck_top == b_stack_btm);
			if (a_deck_top - a_stack_btm < b_deck_top - b_stack_btm)
			{
				if (b[tb] != -1) score_b += b[tb];
				tb++;
			}
			REP(i, a_deck_top - a_stack_btm)
			{
				if (a[ta + i] == -1) score_b = 0;
				else score_a += a[ta + i];
				if (b[tb + i] == -1) score_a = 0;
				else score_b += b[tb + i];
			}
			chmax(res, solve(1, empty, a_deck_top, a_deck_top, b_deck_top, b_deck_top) + score_a - score_b);
		}
	}
	else
	{
		res = INF;
		// put on stack
		if (b_deck_top < m)
		{
			chmin(res, solve(0, 0, a_stack_btm, a_deck_top, b_stack_btm, b_deck_top + 1));
		}
		// pass
		if (pass)
		{
			chmin(res, 0);
		}
		else
		{
			int score_a = 0, score_b = 0;
			int ta = a_stack_btm, tb = b_stack_btm;
			int empty = (a_deck_top == a_stack_btm && b_deck_top == b_stack_btm);
			if (a_deck_top - a_stack_btm > b_deck_top - b_stack_btm)
			{
				if (a[ta] != -1) score_a += a[ta];
				ta++;
			}
			REP(i, b_deck_top - b_stack_btm)
			{
				if (b[tb + i] == -1) score_a = 0;
				else score_b += b[tb + i];
				if (a[ta + i] == -1) score_b = 0;
				else score_a += a[ta + i];
			}
			chmin(res, solve(0, empty, a_deck_top, a_deck_top, b_deck_top, b_deck_top) + score_a - score_b);
		}
	}
	memo[turn][pass][a_stack_btm][a_deck_top][b_stack_btm][b_deck_top] = true;
	return dp[turn][pass][a_stack_btm][a_deck_top][b_stack_btm][b_deck_top] = res;
}

int main()
{
	cin.sync_with_stdio(false); cout << fixed << setprecision(10);
	MS(memo, false);
	cin >> n >> m;
	a.resize(n);
	b.resize(m);
	cin >> a >> b;
	cout << solve(0, 0, 0, 0, 0, 0) << endl;
	return 0;
}
