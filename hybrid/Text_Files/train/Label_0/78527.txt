input = __import__('sys').stdin.readline
n = int(input())
lis = []
fre = [0] * 10005
c1 = 0
ans = [[0, 0]]
has = [0] * 10005
for i in range(n - 1):
	(a, b) = map(int, input().split())
	if b < n:
		print('NO')
		exit()
	lis.append(a)
	fre[a] += 1
lis.sort()
for i in range(n - 1):
	if lis[i] <= i:
		print('NO')
		exit()
else:
	zer = 1
	for i in range(1, n):
		if fre[i] == 1:
			ans.append([i, n])
		elif fre[i] > 1:
			co = 0
			while co != fre[i] - 1:
				if fre[zer] == 0 and co == 0:
					co += 1
					ans.append([zer, n])
				elif fre[zer] == 0:
					co += 1
					ans.append([zer, ans[-1][0]])
				zer += 1
			ans.append([i, ans[-1][0]])
	print('YES')
	for i in ans[1:]:
		print(*i)
