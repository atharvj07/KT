#include <bits/stdc++.h>
using namespace std;
long long n, m, fa[1000005], cnt, rt[500005], x[500005], y[500005];
long long zhi[1000005], tot, add[1000005], sum[1000005], ans[500005];
char s[500005][5];
struct node {
  long long rt, id, bh;
} l[500005], r[500005];
long long getfaj(long long rt) {
  if (!fa[rt]) {
    return rt;
  }
  long long tt = getfaj(fa[rt]);
  zhi[rt] = max(zhi[rt], zhi[fa[rt]]);
  return fa[rt] = tt;
}
long long getfax(long long rt) {
  if (!fa[rt]) {
    return rt;
  }
  long long tt = getfax(fa[rt]);
  zhi[rt] += zhi[fa[rt]];
  return fa[rt] = tt;
}
bool cmp(node aa, node bb) { return aa.bh < bb.bh; }
int main() {
  scanf("%I64d%I64d", &n, &m);
  cnt = n;
  for (long long i = 1; i <= m; ++i) {
    scanf("%s", s[i]);
    if (s[i][0] == 'M') {
      scanf("%I64d%I64d", &x[i], &y[i]);
      cnt++;
      if (!rt[x[i]]) rt[x[i]] = x[i];
      if (!rt[y[i]]) rt[y[i]] = y[i];
      fa[rt[x[i]]] = cnt;
      fa[rt[y[i]]] = cnt;
      rt[x[i]] = cnt;
    } else if (s[i][0] == 'Z') {
      scanf("%I64d", &x[i]);
      if (!rt[x[i]]) rt[x[i]] = x[i];
      zhi[rt[x[i]]] = i;
    } else if (s[i][0] == 'Q') {
      scanf("%I64d", &x[i]);
      getfaj(x[i]);
      l[++tot].bh = zhi[x[i]];
      l[tot].rt = x[i];
      l[tot].id = tot;
      r[tot].bh = i;
      r[tot].id = tot;
      r[tot].rt = x[i];
    } else if (s[i][0] == 'U') {
      scanf("%I64d%I64d", &x[i], &y[i]);
    } else if (s[i][0] == 'A') {
      scanf("%I64d", &x[i]);
    }
  }
  cnt = n;
  memset(fa, 0, sizeof(fa));
  memset(zhi, 0, sizeof(zhi));
  memset(rt, 0, sizeof(rt));
  long long ll = 1, rr = 1;
  sort(l + 1, l + tot + 1, cmp);
  sort(r + 1, r + tot + 1, cmp);
  for (long long i = 1; i <= n; ++i) sum[i] = 1;
  for (long long i = 0; i <= m; ++i) {
    if (s[i][0] == 'U') {
      ++cnt;
      if (!rt[x[i]]) rt[x[i]] = x[i];
      if (!rt[y[i]]) rt[y[i]] = y[i];
      zhi[rt[x[i]]] = add[rt[x[i]]];
      zhi[rt[y[i]]] = add[rt[y[i]]];
      add[rt[x[i]]] = 0;
      add[rt[y[i]]] = 0;
      sum[cnt] = sum[rt[x[i]]] + sum[rt[y[i]]];
      fa[rt[x[i]]] = cnt;
      fa[rt[y[i]]] = cnt;
      rt[x[i]] = cnt;
    } else if (s[i][0] == 'A') {
      if (!rt[x[i]]) rt[x[i]] = x[i];
      add[rt[x[i]]] += sum[rt[x[i]]];
    }
    while (i == l[ll].bh && ll <= tot) {
      long long kk = getfax(l[ll].rt);
      ans[l[ll].id] -= (zhi[l[ll].rt] + add[kk]);
      ll++;
    }
    while (i == r[rr].bh && rr <= tot) {
      long long kk = getfax(r[rr].rt);
      ans[r[rr].id] += (zhi[r[rr].rt] + add[kk]);
      rr++;
    }
  }
  for (long long i = 1; i <= tot; ++i) printf("%I64d\n", ans[i]);
  return 0;
}
