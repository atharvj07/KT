#include<bits/stdc++.h>
using namespace std ;
typedef long long ll ;
#define F first 
#define S second
const ll N = 200222 ; 
int n, ans, M, A, d[N];
vector<int> g[N] ; 
void dfs(int v, int par){
	if(d[v] >= M){M = d[v], A = v;}
	for(int i = 0 ; i < g[v].size() ; i ++){
		int u = g[v][i] ; if(u == par) continue ;
		d[u] = d[v] + 1 ; dfs(u, v) ;
	}
}
int main()
{
	scanf("%d", &n) ;
	for(int i = 1 ; i < n ; i ++){
		int u, v ; scanf("%d%d", &u, &v) ;
		g[v].push_back(u), g[u].push_back(v) ;
	}
	dfs(1, 0) ; M = 0 ; d[A] = 0 ; 
	dfs(A, 0) ;
	if(M % 3 == 1) return cout << "Second", 0;
	cout << "First" ;
}
