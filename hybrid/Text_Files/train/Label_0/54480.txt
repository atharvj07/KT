
import java.util.Scanner;

public class TaskD {

    
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        
        int n = s.nextInt();
        int m = s.nextInt();
        boolean right = true;       
        int res = 0;
        
        int rightPos = 0;
        int leftPos = 0;
        int curPos = 0;
        
        int lastLine = 0;
        
        char[][] field = new char[150][150];
        String tmp = s.nextLine();
        
        for (int i = 0; i < n; i++) {
            String str = s.nextLine();
            for (int j = 0; j < m; j++) {
                field[i][j] = str.charAt(j);
                if (field[i][j] == 'W')
                    lastLine = i;
            }
        }
                
        for (int i = 0; i < lastLine; i++) {
            
            if (right) {
                rightPos = curPos;
                for (int j = curPos + 1; j < m; j++) {
                    if (field[i+1][j] == 'W' || field[i][j] == 'W')
                        rightPos = j;
                }
                //System.out.println(curPos + "  " + rightPos);
                res += Math.abs(rightPos - curPos);
                curPos = rightPos;

            }
            else {
                leftPos = curPos;
                for (int j = curPos - 1; j >= 0; j--) {
                    if (field[i+1][j] == 'W' || field[i][j] == 'W')
                        leftPos = j;
                }
                //System.out.println(curPos + "  " + leftPos);
                res += Math.abs(curPos - leftPos);
                curPos = leftPos;
            }           
            right = !right;
            res += 1;
        }
        
        
        for (int i = lastLine; i <= lastLine; i++) {
            
            if (right) {
                rightPos = curPos;
                for (int j = curPos + 1; j < m; j++) {
                    if (field[i][j] == 'W')
                        rightPos = j;
                }
                //System.out.println(curPos + "  " + rightPos);
                res += Math.abs(rightPos - curPos);
                curPos = rightPos;

            }
            else {
                leftPos = curPos;
                for (int j = curPos - 1; j >= 0; j--) {
                    if (field[i][j] == 'W')
                        leftPos = j;
                }
                //System.out.println(curPos + "  " + leftPos);
                res += Math.abs(curPos - leftPos);
                curPos = leftPos;
            }           
        }
        System.out.print(res);
        s.close();
    }
}
