import java.io.*;
import java.util.*;

public class CF1539E extends PrintWriter {
	CF1539E() { super(System.out, true); }
	static class Scanner {
		Scanner(InputStream in) { this.in = in; } InputStream in;
		byte[] bb = new byte[1 << 15]; int i, n;
		byte getc() {
			if (i == n) {
				i = n = 0;
				try { n = in.read(bb); } catch (IOException e) {}
			}
			return i < n ? bb[i++] : 0;
		}
		int nextInt() {
			byte c = 0; while (c <= ' ') c = getc();
			int a = 0; while (c > ' ') { a = a * 10 + c - '0'; c = getc(); }
			return a;
		}
	}
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1539E o = new CF1539E(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int m = sc.nextInt();
		int[] xx = new int[n + 1];
		int[] aa = new int[n + 1];
		int[] bb = new int[n + 1];
		int[] cc = new int[n + 1];
		int[] dd = new int[n + 1];
		for (int i = 1; i <= n; i++) {
			xx[i] = sc.nextInt();
			aa[i] = sc.nextInt();
			bb[i] = sc.nextInt();
			cc[i] = sc.nextInt();
			dd[i] = sc.nextInt();
		}
		TreeSet<Integer> ll = new TreeSet<>();
		TreeSet<Integer> rr = new TreeSet<>();
		ll.add(0); rr.add(0);
		boolean[] flipl = new boolean[n + 1];
		boolean[] flipr = new boolean[n + 1];
		for (int i = 1; i <= n; i++) {
			int x = xx[i], x_ = xx[i - 1], a = aa[i], b = bb[i], c = cc[i], d = dd[i];
			boolean l_ = !ll.isEmpty();
			boolean r_ = !rr.isEmpty();
			if (a <= x && x <= b) {
				while (!ll.isEmpty() && ll.first() < c)
					ll.pollFirst();
				while (!ll.isEmpty() && ll.last() > d)
					ll.pollLast();
				if (r_ && c <= x_ && x_ <= d) {
					flipl[i] = true;
					ll.add(x_);
				}
			} else
				ll.clear();
			if (c <= x && x <= d) {
				while (!rr.isEmpty() && rr.first() < a)
					rr.pollFirst();
				while (!rr.isEmpty() && rr.last() > b)
					rr.pollLast();
				if (l_ && a <= x_ && x_ <= b) {
					flipr[i] = true;
					rr.add(x_);
				}
			} else
				rr.clear();
			if (ll.isEmpty() && rr.isEmpty()) {
				println("No");
				return;
			}
		}
		println("Yes");
		int[] ans = new int[n + 1];
		boolean left = !ll.isEmpty();
		int a = 0, b = m, c = 0, d = m;
		for (int i = n; i >= 1; i--) {
			ans[i] = left ? 0 : 1;
			int x_ = xx[i - 1];
			if (left) {
				c = Math.max(c, cc[i]); d = Math.min(d, dd[i]);
				if (flipl[i] && c <= x_ && x_ <= d) {
					left = false;
					a = 0; b = m;
				}
			} else {
				a = Math.max(a, aa[i]); b = Math.min(b, bb[i]);
				if (flipr[i] && a <= x_ && x_ <= b) {
					left = true;
					c = 0; d = m;
				}
			}
		}
		for (int i = 1; i <= n; i++)
			print(ans[i] + " ");
		println();
	}
}
