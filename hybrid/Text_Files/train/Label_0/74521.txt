(n, m) = map(int, input().split())
arr = list(map(int, input().split()))
arr1 = list(map(int, input().split()))
a = []
ans = 0
for i in range(n):
	if i + 1 in arr1:
		a.append(arr[i])
	else:
		ans += arr[i]
a = sorted(a)[::-1]
for i in range(m):
	if ans > a[i]:
		ans += ans
	else:
		ans += a[i]
print(ans)
