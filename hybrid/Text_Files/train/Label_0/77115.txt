import sys
from array import array

def readline():
	return sys.stdin.buffer.readline().decode('utf-8')
q = int(readline())
ans = [0] * q

def solve(x):
	if x == 0:
		return 0
	s = str(x)
	dp = [[0] * 5 for _ in range(len(s))]
	for i in range(int(s[0])):
		dp[0][1 if i > 0 else 0] += 1
	bc = 1
	for (i, d) in enumerate(map(int, s[1:]), start=1):
		for j in range(4):
			for k in range(10):
				dp[i][j + (1 if k > 0 else 0)] += dp[i - 1][j]
		if bc < 4:
			for k in range(d):
				dp[i][bc + (1 if k > 0 else 0)] += 1
		bc += 1 if d > 0 else 0
	return sum(dp[-1][:4]) + (1 if bc < 4 else 0) - 1
for i in range(q):
	(l, r) = map(int, readline().split())
	ans[i] = solve(r) - solve(l - 1)
print(*ans, sep='\n')
