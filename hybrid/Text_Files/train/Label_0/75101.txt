from heapq import *

class priority_queue(object):

	def __init__(self):
		self.que = []
		heapify(self.que)

	def push(self, obj):
		heappush(self.que, obj)

	def pop(self):
		return heappop(self.que)

	def top(self):
		return self.que[0]

	def __bool__(self):
		return bool(self.que)

	def __repr__(self):
		return repr(self.que)

	def __len__(self):
		return len(self.que)
m = {}
n = int(input())
a = list(map(int, input().split()))
for (i, _) in enumerate(a):
	m.setdefault(_, priority_queue()).push(i)
que = priority_queue()
for (k, v) in m.items():
	que.push(k)
while que:
	cur = que.pop()
	if len(m[cur]) <= 1:
		continue
	while len(m[cur]) > 1:
		m[cur].pop()
		m.setdefault(cur << 1, priority_queue()).push(m[cur].pop())
		que.push(cur << 1)
l = [(k, v.top()) for (k, v) in m.items() if v]
l.sort(key=lambda x: x[1])
print(len(l))
for (k, v) in l:
	print(k, end=' ')
