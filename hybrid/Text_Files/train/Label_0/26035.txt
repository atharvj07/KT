#include <bits/stdc++.h>
using namespace std;
struct st {
  long double x, y, z, res;
};
long long n;
long double g = (2 / (1 + sqrt(5)));
long long X[109], Y[109], Z[109];
long double dist(long double x, long double y, long double z, long long i) {
  return (x - X[i]) * (x - X[i]) + (y - Y[i]) * (y - Y[i]) +
         (z - Z[i]) * (z - Z[i]);
}
st F(long double x, long double y, long double z) {
  long double res = dist(x, y, z, 1);
  for (int i = 2; i <= n; i++) {
    res = max(res, dist(x, y, z, i));
  }
  return {x, y, z, res};
}
st F2(long double x, long double y) {
  long double L = -10000.0, R = 10000.0;
  st f1, f2;
  long double m1 = R - (R - L) * g;
  long double m2 = L + (R - L) * g;
  f1 = F(x, y, m1);
  f2 = F(x, y, m2);
  while (R - L > 0.0000000001) {
    if (f1.res > f2.res) {
      L = m1;
      m1 = m2;
      f1 = f2;
      m2 = L + (R - L) * g;
      f2 = F(x, y, m2);
    } else {
      R = m2;
      m2 = m1;
      f2 = f1;
      m1 = R - (R - L) * g;
      f1 = F(x, y, m1);
    }
  }
  return f1;
}
st F1(long double x) {
  long double L = -10000.0, R = 10000.0;
  st f1, f2;
  long double m1 = R - (R - L) * g;
  long double m2 = L + (R - L) * g;
  f1 = F2(x, m1);
  f2 = F2(x, m2);
  while (R - L > 0.0000000001) {
    if (f1.res > f2.res) {
      L = m1;
      m1 = m2;
      f1 = f2;
      m2 = L + (R - L) * g;
      f2 = F2(x, m2);
    } else {
      R = m2;
      m2 = m1;
      f2 = f1;
      m1 = R - (R - L) * g;
      f1 = F2(x, m1);
    }
  }
  return f1;
}
void func() {
  long double L = -10000.0, R = 10000.0;
  st f1, f2;
  long double m1 = R - (R - L) * g;
  long double m2 = L + (R - L) * g;
  f1 = F1(m1);
  f2 = F1(m2);
  while (R - L > 0.0000000001) {
    if (f1.res > f2.res) {
      L = m1;
      m1 = m2;
      f1 = f2;
      m2 = L + (R - L) * g;
      f2 = F1(m2);
    } else {
      R = m2;
      m2 = m1;
      f2 = f1;
      m1 = R - (R - L) * g;
      f1 = F1(m1);
    }
  }
  cout << fixed << setprecision(12) << f1.x << " " << f1.y << " " << f2.z
       << endl;
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> X[i] >> Y[i] >> Z[i];
  func();
  return 0;
}
