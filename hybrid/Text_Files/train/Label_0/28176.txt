#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 10;
const int inf = INT_MAX;
template <typename T>
void read(T &x) {
  x = 0;
  int f = 1;
  char c = getchar();
  for (; !isdigit(c); c = getchar())
    if (c == '-') f = -1;
  for (; isdigit(c); c = getchar()) x = x * 10 + c - '0';
  x *= f;
}
int n, m, maxr;
int f[N][19];
int main() {
  read(n), read(m);
  for (int i = 1; i <= n; i++) {
    int l, r;
    read(l), read(r);
    f[l][0] = max(f[l][0], r);
    maxr = max(maxr, r);
  }
  for (int i = 1; i <= maxr; i++) f[i][0] = max(f[i][0], f[i - 1][0]);
  for (int j = 1; j <= 18; j++)
    for (int i = 0; i <= maxr; i++) f[i][j] = f[f[i][j - 1]][j - 1];
  int ans = 0;
  for (int i = 1; i <= m; i++) {
    ans = 0;
    int l, r;
    read(l), read(r);
    for (int i = 18; i >= 0; i--)
      if (f[l][i] < r) ans += 1 << i, l = f[l][i];
    if (f[l][0] >= r)
      printf("%d\n", ans + 1);
    else
      puts("-1");
  }
  return 0;
}
