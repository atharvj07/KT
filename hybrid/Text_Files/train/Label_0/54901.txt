import java.io.*; import java.util.*;
public class CF1425D {

    /**
     * Classical swapping order of summation problem
     * See analysis in doc
     */
    public static class Point{
        int x; int y; int v;
        public Point(int a, int b, int c){
            this.x=a; this.y=b; this.v=c;
        }
    }
    static int MAXL=1001;
    static int MOD=1000000007;
    public static void main(String[] args) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st=new StringTokenizer(br.readLine());
        int N=Integer.parseInt(st.nextToken());
        int M=Integer.parseInt(st.nextToken());
        int R=Integer.parseInt(st.nextToken());
        Point[] a=new Point[N];
        int x,y,v;
        //Binomial coefficients
        long[] C=new long[N+1]; C[M]=1;
        for (int i = M+1; i <=N; i++) {
            C[i]=(C[i-1]*i)%MOD;
            C[i]*=inv(i-M,MOD); C[i]%=MOD;
        }
        
        //Prefix Sum + Input
        int[][] d=new int[MAXL][MAXL];
        for (int i = 0; i < N; i++) {
            st=new StringTokenizer(br.readLine());
            x=Integer.parseInt(st.nextToken());
            y=Integer.parseInt(st.nextToken());
            v=Integer.parseInt(st.nextToken());
            d[x][y]++;
            a[i]=new Point(x,y,v);
        }
        int[][] psa=new int[MAXL][MAXL];
        for (int i = 1; i < MAXL; i++) {
            for (int j = 1; j < MAXL; j++) {
                psa[i][j]=psa[i-1][j]+psa[i][j-1]-psa[i-1][j-1]+d[i][j];
            }
        }
        
        //Swapping the order of Summation!
        long ans=0;
        int x1,x2,y1,y2,vv;long CC;
        int[] adj=new int[N];
        for (int i = 0; i < N; i++) {//Count b_i^2 contribution
            x1=Math.max(1,a[i].x-R);
            x2=Math.min(1000,a[i].x+R);
            y1=Math.max(1,a[i].y-R);
            y2=Math.min(1000,a[i].y+R);
            vv=psa[x2][y2]-psa[x1-1][y2]-psa[x2][y1-1]+psa[x1-1][y1-1];//Number of snakes in that range
            adj[i]=vv; CC=C[N]-C[N-vv]; CC%=MOD;
            if(CC<0){
                 CC+=MOD;
            }
            CC*=a[i].v; CC%=MOD;
            CC*=a[i].v; CC%=MOD;
            ans+=CC; ans%=MOD;
        }
        //System.out.println(Arrays.toString(adj));
        
        for (int i = 0; i < N; i++) {
            for (int j = i+1; j < N; j++) {
                x1=Math.max(1,Math.max(a[i].x,a[j].x)-R);
                x2=Math.min(1000,Math.min(a[i].x,a[j].x)+R);
                y1=Math.max(1,Math.max(a[i].y,a[j].y)-R);
                y2=Math.min(1000,Math.min(a[i].y,a[j].y)+R);
                if(x1<=x2 && y1<=y2){
                    vv=psa[x2][y2]-psa[x1-1][y2]-psa[x2][y1-1]+psa[x1-1][y1-1];
                }else{
                    vv=0;
                }
                
                
                CC=C[N]-C[N-adj[i]]-C[N-adj[j]]+C[N-adj[i]-adj[j]+vv];
                CC%=MOD;
                if(CC<0){
                    CC+=MOD;
                }
                CC*=a[i].v; CC%=MOD;
                CC*=a[j].v; CC%=MOD;
                CC*=2; CC%=MOD;
                ans+=CC; ans%=MOD;
            }
        }
        System.out.println(ans);
    }
    public static long inv(long a, long b){//Computes modular inverse of a mod b
        //Assume b>a and gcd(a,b)=1, 
        if(a==1)return 1;
        long q=b/a;
        long r=b-q*a;
        long y=-inv(r,a);
        return ((b*y+1)/a)+b;
        
    }
}
