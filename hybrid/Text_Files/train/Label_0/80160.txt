q = int(input())
for p in range(q):
	n = int(input())
	line = list(map(int, input().split()))
	one = line.index(1)
	lens = [() for i in range(n + 1)]
	for i in range(n):
		if i < one:
			lens[line[i]] = (one - i, 0)
		else:
			lens[line[i]] = (0, i - one)
	max_len_l = 0
	max_len_r = 0
	for i in range(1, n + 1):
		max_len_l = max(max_len_l, lens[i][0])
		max_len_r = max(max_len_r, lens[i][1])
		if max_len_l + max_len_r < i:
			print(1, end='')
		else:
			print(0, end='')
	print()
