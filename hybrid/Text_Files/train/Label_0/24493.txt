using namespace std;
typedef long long in;
typedef vector<in> VI;
typedef vector<VI> VVI;
in p2(in a){
  return 1LL<<a;
}
const double EPS=1e-10;
vector<vector<double> > s(2,vector<double>(6));
VVI f(2,VI(6));
double clc(double pred, in msk){
  /*
  cout<<pred<<" ";
  forn(i,6)
    cout<<(!!(msk&p2(i)));
  cout<<endl;*/
  double sm=0;
  forn(i,6){
    if(msk&p2(i))
      sm+=pred*s[0][i];
    else
      sm+=(1-pred)*s[1][i];
  }
  return sm;
}
int main(){
  ios::sync_with_stdio(0);
  cin.tie(0);
  forn(z,2){
    forn(i,6){
      cin>>f[z][i];
      s[z][i]=f[z][i]/100.0;
    }
  }
  double bst=0;
  forn(msk,p2(6)){
    double mn=0;
    double mx=1;
    //  choose blue if p*s[0][i]<(1-p)*s[1][i]
    //  p*(s[0][i]+s[1][i])<s[1][i]
    // p<s[1][i]/(s[1][i]+s[0][i]);
    forn(i,6){
      if(f[1][i]==0){
	if(msk&p2(i))
	  mx=-1;
	continue;
      }
      else{
	double pcrit=s[1][i]/(s[1][i]+s[0][i]);
	if(msk&p2(i))
	  mx=min(mx,pcrit);
	else
	  mn=max(mn,pcrit);
      }
    }
    if(mn-EPS<mx){
      bst=max(bst,clc(mn,msk));
      bst=max(bst,clc(mx,msk));
    }
  }
  cout<<setprecision(15);
  cout<<(1-bst)<<endl;
  return 0;
}
