#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-6;
class Point {
 public:
  double x, y;
  Point() {}
  Point(double x, double y) : x(x), y(y) {}
  void read() { scanf("%lf%lf", &x, &y); }
  void print() const { printf("%lf %lf\n", x, y); }
  Point operator+(const Point &b) const { return Point(x + b.x, y + b.y); }
  Point operator-(const Point &b) const { return Point(x - b.x, y - b.y); }
  Point operator*(const double &b) const { return Point(x * b, y * b); }
  Point operator/(const double &b) const { return Point(x / b, y / b); }
  Point rot() const { return Point(y, -x); }
  double len2() const { return x * x + y * y; }
  double len() const { return sqrt(len2()); }
} a[111];
double det(const Point &a, const Point &b) { return a.x * b.y - a.y * b.x; }
double dot(const Point &a, const Point &b) { return a.x * b.x + a.y * b.y; }
bool intersection(const Point &a, const Point &b, const Point &c,
                  const Point &d, Point &e) {
  if (!det(a - b, c - d)) return 0;
  double s1 = det(a - c, a - d);
  double s2 = det(b - d, b - c);
  e = a + (b - a) / (s1 + s2) * s1;
  return 1;
}
bool wx(const Point &a, const Point &b, const Point &c, Point &d) {
  Point ab1 = (a + b) / 2;
  Point ac1 = (a + c) / 2;
  return intersection(ab1, ab1 + (a - b).rot(), ac1, ac1 + (a - c).rot(), d);
}
int n;
int cmp(double a, double b) {
  if (a + EPS < b) return -1;
  if (b + EPS < a) return 1;
  return 0;
}
bool insj(const Point &key, const Point &a, const Point &c, const Point &d) {
  if (!dot(c - a, d - a) || !dot(a - c, d - c) || !dot(a - d, c - d)) return 0;
  double s1 = det(a - key, c - key);
  double s2 = det(c - key, d - key);
  double s3 = det(d - key, a - key);
  double s4 = det(a - d, a - c);
  s1 = fabs(s1);
  s2 = fabs(s2);
  s3 = fabs(s3);
  s4 = fabs(s4);
  return (!cmp(s4, s1 + s2 + s3));
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) a[i].read();
  double ans = -1;
  Point cur;
  for (int i = 1; i <= n; ++i) {
    for (int j = i + 1; j <= n; ++j) {
      for (int p = j + 1; p <= n; ++p) {
        if (wx(a[i], a[j], a[p], cur))
          if (insj(cur, a[i], a[j], a[p])) {
            bool flag1 = 0;
            double key = (a[i] - cur).len2();
            for (int o = 1; o <= n; ++o) {
              if (cmp((a[o] - cur).len2(), key) == -1) {
                flag1 = 1;
                break;
              }
            }
            if (!flag1) ans = max(ans, (a[i] - cur).len());
          }
      }
    }
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n; ++j) {
      if (i != j)
        for (int p = 1; p <= n; ++p) {
          if (p != j && p != i && !dot(a[i] - a[j], a[j] - a[p]))
            for (int q = 1; q <= n; ++q) {
              if (q != i && q != j && q != p)
                if (!dot(a[i] - a[q], a[i] - a[j]) &&
                    !dot(a[j] - a[p], a[p] - a[q])) {
                  if (wx(a[i], a[j], a[p], cur)) {
                    bool flag1 = 0;
                    double key = (a[i] - cur).len2();
                    for (int o = 1; o <= n; ++o) {
                      if ((a[o] - cur).len2() < key) {
                        flag1 = 1;
                        break;
                      }
                    }
                    if (!flag1) ans = max(ans, (a[i] - cur).len());
                  }
                }
            }
        }
    }
  }
  if (ans == -1)
    printf("-1\n");
  else
    printf("%.6lf\n", ans);
  return 0;
}
