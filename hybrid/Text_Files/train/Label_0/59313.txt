import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class ProblemC {
	
	static int cnt;
	
	static long[] tbl = {2, 3, 5, 7, 11};
	
	static int[] whore;
	
	static List<Candidate> val;
	
	static class Candidate implements Comparable<Candidate> {
		long value;
		int k;
		Candidate(long _, int __) {
			value = _;
			k = __;
		}
		@Override
		public int compareTo(Candidate arg0) {
			if (Integer.bitCount(k) == Integer.bitCount(arg0.k)) {
				return k - arg0.k;
			}
			return Integer.bitCount(k) - Integer.bitCount(arg0.k);
		}
	}
	
	public static void dfs(long x, int i, long limit, int head) {
		if (x > limit) {
			return;
		}
		if (x >= 2) {
			cnt++;
			int type = 0;
			for (int a = 0 ; a < head ; a++) {
				if (x % tbl[a] == 0) {
					type |= 1<<a;
					whore[a]++;
				}
			}
			val.add(new Candidate(x, type));
		}
		
		for (int a = i ; a < head ; a++) {
			dfs(x*tbl[a], a, limit, head);
		}
	}
	

	public static void main(String[] args) throws NumberFormatException, IOException {
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter out = new PrintWriter(System.out);
		
		int n = Integer.valueOf(in.readLine());
		List<Long> ans = new ArrayList<Long>();
		for (int u = 2 ; u <= tbl.length ; u++) {
			ans = make(u, n);
			if (ans != null) {
				break;
			}
			
		}
		
		boolean first = true;
		for (long l : ans) {
			if (first) {
				out.print(l);
				first = false;
			} else {
				out.print(" ");
				out.print(l);
			}
		}
		out.println();
		out.flush();
	}
	
	
	public static List<Long> make(int head, int n) {
		val = new ArrayList<Candidate>();
		whore = new int[tbl.length];
		dfs(1, 0, 2L*n*n, head);
		if (val.size() < n) {
			return null;
		}
		
		List<Long> left = new ArrayList<Long>();
		int needRemove = val.size() - n;
		for (Candidate cd : val) {
			long l = cd.value;
			if (needRemove == 0) {
				left.add(l);
			} else {
				boolean can = true;
				for (int i = 0 ; i < head ; i++) {
					if (l % tbl[i] == 0 && whore[i] == (n+1)/2) {
						can = false;
					}
				}
				if (can) {
					for (int i = 0 ; i < head ; i++) {
						if (l % tbl[i] == 0) {
							whore[i]--;
						}
					}
					needRemove--;
				} else {
					left.add(l);
				}
			}
			
			
		}
		return left;
		
	}

	public static void debug(Object... o) {
		System.err.println(Arrays.deepToString(o));
	}
}
