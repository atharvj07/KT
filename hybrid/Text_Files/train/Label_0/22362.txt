#include <bits/stdc++.h>
using namespace std;
const int INF = 0x3f3f3f3f;
const double eps = 1e-8;
const int MAXN = 1e5 + 5;
int sgn(double x) {
  if (fabs(x) < eps) return 0;
  if (x < 0)
    return -1;
  else
    return 1;
}
long long na, nb, a, b;
int n, ans, A[10], B[10], C[10], L[20], R[20];
bool up(int t) {
  for (int i = 0; i <= 9; i++) C[i] = B[i];
  if (C[R[t]] > 0) {
    C[R[t]]--;
    t--;
    bool flag = true;
    for (int i = t; i >= 1; i--) {
      for (int j = R[i] - 1; j >= 0; j--) {
        if (C[j] > 0) return true;
      }
      if (C[R[i]] <= 0) {
        flag = false;
        break;
      }
      C[R[i]]--;
    }
    return flag;
  }
  return false;
}
bool down(int t) {
  for (int i = 0; i <= 9; i++) C[i] = B[i];
  if (C[L[t]] > 0) {
    C[L[t]]--;
    t--;
    bool flag = true;
    for (int i = t; i >= 1; i--) {
      for (int j = L[i] + 1; j <= 9; j++) {
        if (C[j] > 0) return true;
      }
      if (C[L[i]] <= 0) {
        flag = false;
        break;
      }
      C[L[i]]--;
    }
    return flag;
  }
  return false;
}
bool ok() {
  for (int i = 0; i <= 9; i++) B[i] = A[i];
  if (na == nb) {
    for (int i = 1; i <= n; i++) {
      if (B[R[i]] <= 0) return false;
      B[R[i]]--;
    }
    return true;
  }
  int t = n;
  while (L[t] == R[t] && t > 1) {
    if (B[R[t]] <= 0) return false;
    B[R[t]]--;
    t--;
  }
  for (int i = L[t] + 1; i < R[t]; i++) {
    if (B[i]) return true;
  }
  if (up(t) || down(t))
    return true;
  else
    return false;
}
void dfs(int x, int sum) {
  if (x == 10) {
    A[0] = sum;
    if (ok()) {
      ans++;
    }
    return;
  }
  for (int i = 0; i <= sum; i++) {
    A[x] = i;
    dfs(x + 1, sum - i);
  }
}
int main() {
  cin >> na >> nb;
  a = na, b = nb;
  n = 0;
  while (b) {
    R[++n] = b % 10;
    b /= 10;
  }
  for (int i = 1; i <= n; i++) {
    L[i] = a % 10;
    a /= 10;
  }
  ans = 0;
  dfs(1, n);
  printf("%d\n", ans);
}
