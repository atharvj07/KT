import sys
from string import ascii_lowercase

class AlphabetGraph:

	def __init__(self):
		self.letters = {letter: index for (index, letter) in enumerate(ascii_lowercase)}
		self.nodes = [self.Node(x) for x in range(len(ascii_lowercase))]

	def add_edge(self, u, v):
		u_index = self.letters[u]
		v_index = self.letters[v]
		n1 = self.nodes[u_index]
		n2 = self.nodes[v_index]
		if n2 not in n1.adjacent:
			n1.add_edge(n2)
			n2.in_degree += 1

	def init_set(self):
		s = set()
		for node in self.nodes:
			if node.in_degree == 0:
				s.add(node)
		return s

	def kahn(self):
		topo = []
		s = self.init_set()
		while s:
			n = s.pop()
			topo.append(n)
			for m in n.adjacent:
				m.in_degree -= 1
				if m.in_degree == 0:
					s.add(m)
			n.adjacent.clear()
		if len(topo) < len(ascii_lowercase):
			return -1
		return topo

	class Node:

		def __init__(self, num):
			self.num = num
			self.in_degree = 0
			self.adjacent = set()

		def add_edge(self, v):
			self.adjacent.add(v)
g = AlphabetGraph()
n = int(sys.stdin.readline())
a = sys.stdin.readline().strip()
result = 0
for x in range(n - 1):
	b = sys.stdin.readline().strip()
	min = len(a) if len(a) < len(b) else len(b)
	for y in range(min):
		if a[y] != b[y]:
			g.add_edge(a[y], b[y])
			break
		if y == len(b) - 1 and len(a) > len(b):
			result = -1
			break
	if result == -1:
		break
	a = b
if result == -1:
	print('Impossible')
else:
	result = g.kahn()
	if result == -1:
		print('Impossible')
	else:
		for node in result:
			print(ascii_lowercase[node.num], end='')
