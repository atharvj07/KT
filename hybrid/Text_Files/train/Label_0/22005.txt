#include<bits/stdc++.h>
using namespace std;

int main(){
    int n,m;    cin>>n>>m;
    long long a=0;
    
    long long b=0;

    long long ans=0;

    for(int i=0;i<n;i++){
        int x;  cin>>x;
        a+=x;
    }
    for(int i=0;i<m;i++){
        int x;  cin>>x;
        b+=x;
    }

    ans=a*b;

    cout<<ans<<endl;

    return 0;
}
