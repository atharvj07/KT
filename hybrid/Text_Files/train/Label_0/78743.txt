def main():
	import sys
	input = sys.stdin.readline
	query = int(input())
	for q in range(query):
		n = int(input())
		f = []
		for _ in range(n):
			f.append(list(map(int, input().split())))
		dp = [[0, f[0][1], f[0][1] * 2]]
		for i in range(1, n):
			if f[i][0] == f[i - 1][0]:
				dp1 = min(dp[i - 1][1], dp[i - 1][2])
				dp2 = min(dp[i - 1][0], dp[i - 1][2]) + f[i][1]
				dp3 = min(dp[i - 1][0], dp[i - 1][1]) + f[i][1] * 2
			elif f[i][0] == f[i - 1][0] - 1:
				dp1 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2])
				dp2 = min(dp[i - 1][1], dp[i - 1][2]) + f[i][1]
				dp3 = min(dp[i - 1][0], dp[i - 1][2]) + f[i][1] * 2
			elif f[i][0] == f[i - 1][0] - 2:
				dp1 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2])
				dp2 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2]) + f[i][1]
				dp3 = min(dp[i - 1][1], dp[i - 1][2]) + f[i][1] * 2
			elif f[i][0] == f[i - 1][0] + 1:
				dp1 = min(dp[i - 1][0], dp[i - 1][2])
				dp2 = min(dp[i - 1][0], dp[i - 1][1]) + f[i][1]
				dp3 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2]) + f[i][1] * 2
			elif f[i][0] == f[i - 1][0] + 2:
				dp1 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2])
				dp2 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2]) + f[i][1]
				dp3 = min(dp[i - 1][1], dp[i - 1][2]) + f[i][1] * 2
			else:
				dp1 = min(dp[i - 1][0], dp[i - 1][1])
				dp2 = min(dp[i - 1][0], dp[i - 1][1], dp[i - 1][2]) + f[i][1]
				dp3 = min(dp[i - 1][1], dp[i - 1][1], dp[i - 1][2]) + f[i][1] * 2
			dp.append([dp1, dp2, dp3])
		print(min(dp[-1]))
main()
