
import java.util.*;
import java.lang.*;
import java.math.*;

public class A {
    
    Scanner sc = new Scanner(System.in);
    int n;
    int[][] road;
    int c = 0;
    
    void run(){
        
        int nowP = 0;
        int nextP ;
        int[] point =new int [2];
        int[][] vRoad;
        
        n = sc.nextInt();
        
        road = new int [n][3];
        vRoad = new int [n][3];
        
        for (int i=0;i<n;i++) {
            road[i][0] = sc.nextInt();
            road[i][1] = sc.nextInt();
            road[i][2] = sc.nextInt();
            vRoad[i][0] = road[i][0];
            vRoad[i][1] = road[i][1];
            vRoad[i][2] = road[i][2];
        }
        
        
        int cost1 = 0;
        do {
            point = retNum(nowP);
            nextP = point[0];
            
            if ( point[1] == 1 ) {
                revRoad( nextP );
                cost1 += road[nextP][2];
//              System.out.println(cost1);
            }
            nowP = nextP;
            
//          System.out.println(road[nowP][0] + " " + road[nowP][1]);
            
        }while ( nextP != 0 );
        
        road = vRoad;
        
//      System.out.println();
        
        int cost2 = 0;
        
        nowP = 0;
        
        revRoad( nowP );
        
        
        cost2 += road[nowP][2];
//      System.out.println(cost2);
        
        do {
            point = retNum(nowP);
            nextP = point[0];
            
            if (point[1] == 1) {
                revRoad( nextP );
                cost2 += road[nextP][2];
//              System.out.println(cost1);
//              System.out.println("a");
            }
            
//          System.out.println(road[nowP][0] + " " + road[nowP][1]);
            
            nowP = nextP;
            
        }while ( nextP != 0 );
        
//      System.out.println(cost2);
        
        int c = Math.min(cost1, cost2);
        
        System.out.println(c);
        
    }
    
    int[] retNum(int num) {
        
        int[] re = new int[2];
        
        for (int i=0;i<n;i++) {
            for (int j=0;j<2;j++) {
                if ( i != num ) {
                    if ( road[num][1] == road[i][j] ) {
                        
                        re[0] = i;
                        re[1] = j;
                        return re;
                    }
                }
            }
        }
        
        return re;
        
    }
    
    void revRoad(int num) {
        
        int tmp = road[num][0];
        road[num][0] = road[num][1];
        road[num][1] = tmp;
        
        
    }
    
    public static void main(String[] args){
        
        new A().run();
        
    }

}
