t = int(input())
for _ in range(t):
	i = input()
	(xa, ya) = map(int, input().split())
	(xb, yb) = map(int, input().split())
	(xf, yf) = map(int, input().split())
	if xa == xb == xf and min(ya, yb) < yf < max(ya, yb):
		ans = abs(ya - yb) + 2
	elif ya == yb == yf and min(xa, xb) < xf < max(xa, xb):
		ans = abs(xa - xb) + 2
	else:
		ans = abs(xa - xb) + abs(ya - yb)
	print(ans)
