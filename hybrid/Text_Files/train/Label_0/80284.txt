def main():
	(n, d, m) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(filter(lambda i: i > m, a))
	c = list(filter(lambda i: i <= m, a))
	del a
	b.sort(reverse=True)
	c.sort(reverse=True)
	for (idx, i) in enumerate(b):
		if idx:
			b[idx] += b[idx - 1]
	for (idx, i) in enumerate(c):
		if idx:
			c[idx] += c[idx - 1]
	mx = 0
	k = (n + d) // (d + 1)
	if len(b) >= k:
		mx = b[k - 1]
	for (idx, i) in enumerate(c):
		bcnt = (n - idx - 1 + d) // (d + 1)
		if len(b) >= bcnt:
			if bcnt:
				mx = max(mx, b[bcnt - 1] + c[idx])
			elif len(c) == n:
				mx = max(mx, c[idx])
	print(mx)
	return
main()
