#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1);
const long long INF = 1LL << 62;
const long long MINF = -(1LL << 62);
template <typename T>
T getint() {
  T val = 0;
  char c;
  bool neg = false;
  while ((c = getchar()) && !(c >= '0' && c <= '9')) {
    neg |= c == '-';
  }
  do {
    val = (val * 10) + c - '0';
  } while ((c = getchar()) && (c >= '0' && c <= '9'));
  return val * (neg ? -1 : 1);
}
const int MAXN = 100001;
int n, m;
vector<long double> sol;
vector<pair<int, int>> adj[MAXN];
using linear = pair<long double, long double>;
using lineq = pair<linear, linear>;
vector<lineq> eqs;
linear lst[MAXN];
bool no = false;
void bfs() {
  queue<int> q;
  vector<int> par(n + 1);
  for (int x = 1; x <= n; ++x) {
    if (par[x] == 0) {
      q.push(x);
      lst[x] = {1, 0};
      par[x] = -1;
      eqs.clear();
      vector<int> vs;
      while (!q.empty()) {
        int fr = q.front();
        q.pop();
        vs.push_back(fr);
        for (auto j : adj[fr]) {
          if (j.first != par[fr]) {
            if (par[j.first] == 0) {
              par[j.first] = fr;
              lst[j.first] = {-lst[fr].first, j.second - lst[fr].second};
              q.push(j.first);
            } else {
              linear lehet = {lst[fr].first + lst[j.first].first,
                              lst[fr].second + lst[j.first].second};
              linear szum = {0, j.second};
              eqs.push_back({lehet, szum});
            }
          }
        }
      }
      long double eps = 1e-7;
      set<long double> sols;
      bool csak0 = false;
      for (auto i : eqs) {
        if ((i.first.first - i.second.first) != 0)
          sols.insert((i.second.second - i.first.second) /
                      (i.first.first - i.second.first));
        else {
          if (i.first.second != i.second.second) {
            csak0 = true;
          } else {
          }
        }
      }
      ((void)0);
      if (csak0) {
        bool ok = true;
        for (auto i : vs) {
          for (auto j : adj[i]) {
            ok &= j.second == lst[i].second + lst[j.first].second;
          }
        }
        if (!ok)
          no = true;
        else {
          for (int i : vs) {
            sol[i] = lst[i].second;
          }
        }
      } else if (sols.empty()) {
        vector<long double> vals;
        for (int i : vs) {
          vals.push_back(-lst[i].second);
        }
        sort(vals.begin(), vals.end());
        long double L = -1000, R = 1000;
        auto sum = [&](long double mid) -> long double {
          long double res = 0.0;
          for (int i : vs) {
            res += abs(mid * lst[i].first + lst[i].second);
          }
          return res;
        };
        while (fabs(R - L) > 1e-12) {
          long double mid1 = (2 * L + R) / 3.0, mid2 = (L + 2 * R) / 3.0;
          if (sum(mid1) > sum(mid2)) {
            L = mid1;
          } else {
            R = mid2;
          }
        }
        long double L2 = L;
        for (int i : vs) {
          sol[i] = lst[i].first * L2 + lst[i].second;
        }
      } else {
        if (sols.size() == 1 || *prev(sols.end()) - *sols.begin() < eps) {
          for (int i : vs) {
            sol[i] = lst[i].first * (*sols.begin()) + lst[i].second;
          }
        } else {
          no = true;
        }
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  sol.resize(n + 1);
  map<pair<int, int>, int> volt;
  for (int i = 0; i < m; ++i) {
    int a, b, c;
    cin >> a >> b >> c;
    adj[a].push_back({b, c});
    adj[b].push_back({a, c});
    if (volt.count(make_pair(min(a, b), max(a, b)))) {
      no |= volt[make_pair(min(a, b), max(a, b))] != c;
    }
    volt[make_pair(min(a, b), max(a, b))] = c;
  }
  bfs();
  cout << fixed << setprecision(9);
  if (no) {
    cout << "NO\n";
  } else {
    cout << "YES\n";
    for (int i = 1; i <= n; ++i) cout << sol[i] << " \n"[i == n];
    bool ok = true;
    for (auto i : volt) {
      ok &= fabs(sol[i.first.first] + sol[i.first.second] -
                 (long double)i.second) < 1e-6;
    }
    assert(ok);
  }
  return 0;
}
