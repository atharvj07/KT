#include <bits/stdc++.h>
using namespace std;
double PI = acos(-1);
const int inf = 0x3f3f3f3f;
int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; ++i) cin >> v[i];
  vector<int> ct(n, 0);
  vector<int> co(n, 0);
  co[0] = (v[0] == 1) ? 1 : 0;
  ct[0] = (v[0] == 2) ? 1 : 0;
  for (int i = 1; i <= n - 1; ++i) {
    co[i] = (v[i] == 1) ? co[i - 1] + 1 : 0;
    ct[i] = (v[i] == 2) ? ct[i - 1] + 1 : 0;
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (ct[i] != 0) {
      if (i + ct[i] >= n)
        ;
      else if (co[i + ct[i]] == ct[i])
        ans = max(ct[i], ans);
      if (i - ct[i] < 0)
        ;
      else if (co[i - ct[i]] >= ct[i]) {
        ans = max(ct[i], ans);
      }
    }
    if (co[i] != 0) {
      if (i + co[i] >= n)
        ;
      else if (ct[i + co[i]] == co[i])
        ans = max(co[i], ans);
      if (i - co[i] < 0)
        ;
      else if (ct[i - co[i]] >= co[i]) {
        ans = max(co[i], ans);
      }
    }
  }
  cout << ans * 2;
  return 0;
}
