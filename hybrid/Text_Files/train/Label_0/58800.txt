import java.util.ArrayList;
import java.util.Scanner;
import java.math.*;

public class cf {
	
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int[] arr = new int[n];
		for(int i=0; i < n; i++){
			arr[i] = in.nextInt();
		
		}
		
		int[] ans = new int[n];
		for(int i=0; i < n; i++){
			int count = 0;
			for(int j=0; j < n; j++){
				if(i==j) continue;
				if(arr[j]>arr[i]) count++;
			}
			ans[i] = count+1;
			System.out.print(ans[i] + " " );
		}
		
		
		
	}

}
