import java.util.*;

public class Main
{
    public static void main(String[] args)
    {
        Scanner scan = new Scanner(System.in);
        int d, n, m;
        int[][] t = new int[200005][2];
        long cur, next, ans = 0;
        PriorityQueue<ABC> q = new PriorityQueue<>();
        d = scan.nextInt();
        n = scan.nextInt();
        m = scan.nextInt();
        for (int i = 0; i < m; i++)
        {
            t[i][0] = scan.nextInt();
            t[i][1] = scan.nextInt();
        }
        t[m][0] = d;
        Arrays.sort(t, 0, m + 1, Comparator.comparing((int[] arr) -> arr[0]));
        for (int i = 0; i < m; i++)
        {
            q.add(new ABC(t[i][0], t[i][1]));
            if (t[i + 1][0] <= n)
                continue;
            cur = Math.max(t[i][0], n);
            next = t[i + 1][0];
            while (cur < next)
            {
                while (!q.isEmpty() && q.peek().pos + n <= cur)
                    q.poll();
                if (q.isEmpty())
                {
                    System.out.println(-1);
                    System.exit(0);
                }
                ans += q.peek().price * Math.min(q.peek().pos + n - cur, next - cur);
                cur = Math.min(q.peek().pos + n, next);
            }
        }
        System.out.println(ans);
    }
}

class ABC implements Comparable<ABC>
{
    int pos, price;
    public ABC(int a, int b)
    {
        this.pos = a;
        this.price = b;
    }
    @Override
    public int compareTo(ABC x)
    {
        return this.price - x.price;
    }
}
