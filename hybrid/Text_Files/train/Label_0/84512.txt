t = int(input())
move = {'L': -1, 'R': 1, 'U': -1, 'D': 1}
for i in range(t):
	n = int(input())
	s = list(input())
	ans = [-1, n]
	sw = False
	tem = (0, 0)
	ha = {}
	ha[tem] = 0
	for i in range(n):
		if s[i] == 'L' or s[i] == 'R':
			tem = (tem[0] + move[s[i]], tem[1])
		else:
			tem = (tem[0], tem[1] + move[s[i]])
		if tem in ha.keys():
			if ans[1] - ans[0] > i - ha[tem]:
				ans = [ha[tem] + 1, i + 1]
				sw = True
		ha[tem] = i + 1
	if sw:
		print(ans[0], ans[1])
	else:
		print(-1)
