def gcd(a, b):
	while a and b:
		if a > b:
			a %= b
		else:
			b %= a
	return a + b

def f(x, y):
	if x == 1:
		return y // (n // d)
	else:
		return y // (m // d)
(n, m, q) = map(int, input().split())
