n = lst[0]
t = lst[1]
k = lst[2]
arr = []
lst = [int(x) for x in input().split()]
while t:
	input_arr = input()
	arr.append(input_arr)
	t -= 1
sol = []
for a in arr:
	(l, r) = a.split()
	l = int(l)
	r = int(r)
	sol.append(k + lst[r - 1] - lst[l - 1] - 2 * (r - l) - 1)
for i in sol:
