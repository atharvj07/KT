from sys import stdin

def get_freq(string):
	opening = [0, -1]
	closing = [0, -1]
	colon = 0
	for i in range(len(string)):
		if string[i] == '[':
			opening[0] += 1
			if opening[1] == -1:
				opening[1] = i
				continue
		elif string[i] == ']':
			closing[0] += 1
			closing[1] = max(closing[1], i)
		elif string[i] == ':':
			colon += 1
	return (opening, closing, colon)
string = list(stdin.readline().strip())
(opening, closing, colon) = get_freq(string)
if opening[0] > 0 and closing[0] > 0 and (colon >= 2):
	if opening[1] < closing[1]:
		colon_count = [0, -1, -1]
		for i in range(opening[1] + 1, closing[1]):
			if string[i] == ':':
				colon_count[0] += 1
				if colon_count[1] == -1:
					colon_count[1] = i
				colon_count[2] = max(colon_count[2], i)
		if colon_count[0] >= 2:
			chars = 0
			for i in range(colon_count[1], colon_count[2] + 1):
				if string[i] == '|':
					chars += 1
			print(4 + chars)
			exit()
		else:
			print(-1)
			exit()
	else:
		print(-1)
		exit()
else:
	print(-1)
	exit()
