from collections import defaultdict
n = int(input())
S = list(input())
C = defaultdict(list)
I = defaultdict(int)
for (i, c) in enumerate(S):
	C[c].append(i)
up = 0
SR = S[::-1]

def brute():
	swa = 0
	for i in range(n):
		if S[i] == SR[i]:
			continue
		c = SR[i]
		j = S.index(c, i)
		swa += j - i
		S[i:i] = S.pop(j)
	return swa
s = [0] * n

def update(pos, dif):
	while pos < n:
		s[pos] += dif
		pos |= pos + 1

def query(pos):
	res = 0
	while pos > 0:
		res += s[pos - 1]
		pos &= pos - 1
	return res
p = 0
swa = 0
for (i, c) in enumerate(SR):
	L = C[c]
	j = I[c]
	at = up + L[j] + query(L[j])
	assert at >= i
	if up + L[j] > i:
		swa += at - i
	update(L[j], -1)
	up += 1
	I[c] += 1
print(swa)
