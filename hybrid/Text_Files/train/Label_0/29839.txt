#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int INF = 0x3f3f3f3f;
const int N = 2000 + 100;
const ll mod = 1e9 + 7;
int n,m;
char s[N];
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);
    int t;
    cin>>t;
    while(t--)
    {
        cin>>n>>m;
        ll ans=1;
        int tot=0;
        for(int i=1;i<=n;i++)
        {
            cin>>s;
            for(int i=0;i<m;i++)
            {
                if(s[i]=='#')
                {
                    tot++;ans=ans*2%mod;
                }
            }

        }
        if(tot==n*m)
            ans--;
        cout<<ans<<endl;
    }


    return 0;
}
