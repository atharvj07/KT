(n, m) = map(int, input().split())
gol = [[0] * n for i in range(m)]
ans = [0] * n
for i in range(m):
	gol[i] = list(map(int, input().split()))
for i in range(m):
	maxim = [-1, -1]
	for j in range(n):
		if gol[i][j] > maxim[0]:
			maxim = [gol[i][j], j]
	ans[maxim[1]] += 1
maxim = (-1, -1)
for i in range(n):
	if ans[i] > maxim[0]:
		maxim = [ans[i], i]
print(maxim[1] + 1)
