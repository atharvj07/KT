n = int(input())
lst = list(map(int, input().split()))
lst1 = []
lst2 = []
for i in range(0, n, 2):
	lst1.append(lst[i])
for j in range(1, n, 2):
	lst2.append(lst[j])
lst = lst1 + lst2
k = (n + 1) // 2
sum_of_k_element = [0] * n
sum_of_k_element[0] = sum(lst[:k])
for i in range(1, n):
	sum_of_k_element[i] = sum_of_k_element[i - 1] - lst[i - 1] + lst[(i + k - 1) % n]
print(max(sum_of_k_element))
