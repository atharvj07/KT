def chk(a, b):
	i = 0
	j = 0
	cmn = 0
	for i in a:
		if i == b[j]:
			cmn += 1
			j += 1
			if j >= len(b):
				break
	return len(a) + len(b) - 2 * cmn
i = 1
lst = []
while i <= 1e+19:
	lst.append(str(i))
	i *= 2
for _ in range(int(input())):
	s = input()
	ans = 1e+19
	for i in lst:
		ans = min(ans, chk(s, i))
	print(ans)
