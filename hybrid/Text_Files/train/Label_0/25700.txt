#include <bits/stdc++.h>
long long mod = 1e9 + 7;
const int MAX = 1e5 + 5;
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  vector<pair<long long, long long> > v(n);
  int mark[1001] = {0};
  long long occur[1001] = {0};
  int flg = 0;
  for (int i = 0; i < n - 1; i++) {
    cin >> v[i].first >> v[i].second;
    if (v[i].first < v[i].second) {
      swap(v[i].first, v[i].second);
    }
    occur[v[i].second]++;
    mark[v[i].second] = 1;
    if (v[i].first != n && v[i].second != n) {
      flg = 1;
    }
  }
  if (flg == 1) {
    cout << "NO";
    return 0;
  }
  set<long long> s;
  for (int i = 1; i < n; i++) {
    if (mark[i] == 0) s.insert(i);
  }
  vector<long long> adj[n + 1];
  int flag = 0;
  for (int i = 1; i < n; i++) {
    if (mark[i] == 1) {
      long long curr = n;
      for (int j = 0; j < occur[i] - 1; j++) {
        if (s.empty() == true) {
          flag = 1;
          break;
        }
        long long top = *(s.begin());
        if (top < i) {
          s.erase(*(s.begin()));
          adj[curr].push_back(top);
          curr = top;
        } else {
          flag = 1;
          break;
        }
      }
      adj[curr].push_back(i);
    }
  }
  if (flag == 1) {
    cout << "NO";
    return 0;
  }
  cout << "YES";
  cout << '\n';
  for (int i = 1; i <= n; i++) {
    if (adj[i].size() > 0) {
      for (int j = 0; j < adj[i].size(); j++) {
        long long pados = adj[i][j];
        cout << i << " " << pados << endl;
      }
    }
  }
  return 0;
}
