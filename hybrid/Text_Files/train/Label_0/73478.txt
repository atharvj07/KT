n = int(input())
A = list(map(int, input().split()))
l = [-1 for i in range(n + 1)]
r = [-1 for i in range(n + 1)]
for i in range(n):
	if l[A[i]] == -1:
		l[A[i]] = i
	r[A[i]] = i
DP = [0 for i in range(n + 1)]
cnt = [0 for i in range(n + 1)]
for i in range(n - 1, -1, -1):
	cnt[A[i]] = cnt[A[i]] + 1
	if l[A[i]] == i:
		DP[i] = cnt[A[i]] + DP[r[A[i]] + 1]
		DP[i] = max(DP[i], DP[i + 1])
	else:
		DP[i] = cnt[A[i]]
		DP[i] = max(DP[i], DP[i + 1])
print(n - DP[0])
