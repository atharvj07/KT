#include <bits/stdc++.h>
using namespace std;
const int INFTY = 1 << 29;
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
void dfs(vector<vector<int> >& g1, vector<vector<int> >& g2, int v, int c,
         vector<int>& cs, int& res) {
  if (cs[v] != INFTY) {
    if (cs[v] == c) return;
    if (res == INFTY)
      res = abs(cs[v] - c);
    else
      res = gcd(res, abs(cs[v] - c));
  } else {
    cs[v] = c;
    for (int i = int(0); i < int(g1[v].size()); i++)
      dfs(g1, g2, g1[v][i], c + 1, cs, res);
    for (int i = int(0); i < int(g2[v].size()); i++)
      dfs(g1, g2, g2[v][i], c - 1, cs, res);
  }
}
int main() {
  for (int n, m; cin >> n >> m && n | m;) {
    vector<vector<int> > g1(n), g2(n);
    for (int i = int(0); i < int(m); i++) {
      int s, d;
      cin >> s >> d;
      s--, d--;
      g1[s].push_back(d);
      g2[d].push_back(s);
    }
    vector<int> cs(n, INFTY);
    int res = INFTY;
    for (int i = int(0); i < int(n); i++)
      if (cs[i] == INFTY) dfs(g1, g2, i, 0, cs, res);
    cout << (res == INFTY ? n : res) << endl;
  }
  return 0;
}
