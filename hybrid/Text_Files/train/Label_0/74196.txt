for _ in range(int(input())):
	n = int(input())
	dp = [0] * (n + 1)
	l = list(map(int, input().split()))
	for i in range(0, n):
		if l[i] > 1:
			r = min(2 + i, n)
			dp[r] += 1
			r = min(l[i] + 1 + i, n)
			dp[r] -= 1
	for i in range(0, n):
		dp[i + 1] = dp[i + 1] + dp[i]
	for i in range(0, n):
		if dp[i] - (l[i] - 1) >= 1:
			dp[i + 1] += dp[i] - (l[i] - 1)
	ans = 0
	for i in range(0, n):
		ans += max(0, l[i] - dp[i] - 1)
	print(ans)
