from itertools import accumulate
t = int(input())
for _ in range(t):
	(n, k) = list(map(int, input().split(' ')))
	nums = list(map(int, input().split(' ')))
	ans = 0
	nums.sort(reverse=True)
	acc = list(accumulate(nums))
	s = 0
	OK = -1
	NG = n
	m = (OK + NG) // 2

	def judge(m):
		return acc[m] >= k * (m + 1)
	while NG - OK > 1:
		if judge(m):
			OK = m
		else:
			NG = m
		m = (OK + NG) // 2
	print(OK + 1)
