import java.util.*;
import java.io.*;

// Solution

public class Main 
{    
  public static void main (String[] argv) 
  {
	  new Main();
  } 
     
     
  
  boolean test = false;
  
 
  public Main() {
	  FastReader in = new FastReader(new BufferedReader(new InputStreamReader(System.in)));
	  //FastReader in = new FastReader(new BufferedReader(new FileReader("Main.in")));
      int n = in.nextInt();
      int l = in.nextInt();
      int w = in.nextInt();
      
      List<Integer> pos = new ArrayList<>(n);
      List<Integer> fpos = new ArrayList<>(n);
      List<Integer> neg = new ArrayList<>(n);
      List<Integer> fneg = new ArrayList<>(n);
      
      long nl = 0, nr = 0;
      for (int i = 0; i < n; i++) {
          int x = in.nextInt();
          int v = in.nextInt();
          
          if (x < 0) {
              if (l + x > 0) {
                 if (v > 0) ++nl;
                 else       ++nr;                 
              }else if (l + x ==0) {
                 if (v > 0) nl++;
                 else if (w > 1) ++nr;
              }else {
                if (v > 0) {
                    ++nl;
                    fneg.add(-x-l);
                }else {
                    fpos.add(-x-l);
                }
              }
          }else {
              if (v > 0) {
                 pos.add(x);   
              }else {
                  ++nr;
                  neg.add(x);
              }
          }
      }
      //System.out.println(nl + " " + nr);
      long ans = nl * nr;
      if (w > 1) {
        ans += process(pos, neg, l, w);
        ans += process(fpos, fneg, l, w);
      }
      
      System.out.println(ans);
     
  }
  
  private long process(List<Integer> pos, List<Integer> neg, int l, int w) {
      if (pos.isEmpty() || neg.isEmpty()) return 0;
      
      Collections.sort(neg);
      long cnt = 0;
      for (int x : pos) {
          long ymin = 1L * x * (w + 1) / (w - 1) - l + 1;
          if (ymin < Integer.MAX_VALUE)
             cnt += binSearch(neg, (int)ymin);
      }
      return cnt;
  }
  
  private int binSearch(List<Integer> a, int v) {
      int n = a.size();
      if (a.get(n-1) < v) return 0;
      
      if (a.get(0) >= v) return n;
      
      int lo = 0, hi = n - 1;
      while (lo <= hi) {
          int mid = (lo + hi) / 2;
          if (a.get(mid) >= v) hi = mid - 1;
          else lo = mid + 1;
          
      }
      return n - lo;
  }
  
  private int max(int a, int b) {
      return a >  b ? a : b;
  }
  
  private int min(int a, int b) {
      return a >  b ? b : a;
  }
  
  
  static class FastReader
    {
        BufferedReader br;
        StringTokenizer st;
 
        public FastReader(BufferedReader in)
        {            
            br = in;
        }
 
        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    String line = br.readLine();
                    if (line == null || line.length() == 0) return "";
                    st = new StringTokenizer(line);
                }
                catch (IOException  e)
                {
                    return "";
                    //e.printStackTrace();
                }
            }
            return st.nextToken();
        }
 
        int nextInt()
        {
            return Integer.parseInt(next());
        }
 
        long nextLong()
        {
            return Long.parseLong(next());
        }
 
        double nextDouble()
        {
            return Double.parseDouble(next());
        }
 
        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                return "";
                //e.printStackTrace();
            }
            return str;
        }
    }
}
