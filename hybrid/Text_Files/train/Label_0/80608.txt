n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
p1 = 0
p2 = 0
k = 0
f = 0
if a == b:
	print(-1)
else:
	for i in range(n):
		if a[i] == 0 and b[i] == 0:
			pass
		elif a[i] == 0 and b[i] == 1:
			p2 = p2 + 1
		elif a[i] == 1 and b[i] == 1:
			p1 = p1 + 1
			p2 = p2 + 1
		else:
			k = k + 1
	for j in range(1, 1000):
		if k * j + p1 > p2:
			print(j)
			f = 1
			break
	if f == 0:
		print(-1)
