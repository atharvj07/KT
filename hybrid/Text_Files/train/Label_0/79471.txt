(n, k) = map(int, input().split())
(a, b, c, d) = map(int, input().split())
if n == 4:
	print(-1)
	exit()
if k < n + 1:
	print(-1)
	exit()
vertices = set([i for i in range(1, n + 1)])
vertices.remove(a)
vertices.remove(b)
vertices.remove(c)
vertices.remove(d)
lst = []
lst.append(a)
lst.append(c)
for x in vertices:
	lst.append(x)
lst.append(d)
lst.append(b)
print(*lst)
print(lst[1], lst[0], end=' ')
print(*lst[2:-2], end=' ')
print(lst[-1], lst[-2])
