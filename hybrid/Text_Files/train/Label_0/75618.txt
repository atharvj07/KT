INF = float('inf')

def gcd(a, b):
	while b:
		(a, b) = (b, a % b)
	return a

def isPrime(n):
	if n <= 1:
		return False
	i = 2
	while i ** 2 <= n:
		if n % i == 0:
			return False
		i += 1
	return True

def array():

def main():
	n = int(input())
	pressed = 0
	ans = 0
	while n > 0:
		ans += n * (pressed + 1) - pressed
		pressed += 1
		n -= 1
	print(ans)
t = 1
for _ in range(t):
	main()
