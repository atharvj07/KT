import sys
from collections import defaultdict
n = int(sys.stdin.readline())
(ver, hor) = (defaultdict(list), defaultdict(list))
points = []
for i in range(4 * n + 1):
	(x, y) = map(int, sys.stdin.readline().split())
	ver[y].append(x)
	hor[x].append(y)
	points.append([x, y])
(lh, lv) = ([], [])
for i in sorted(hor):
	if len(hor[i]) >= n:
		left = i
		break
for i in sorted(hor, reverse=True):
	if len(hor[i]) >= n:
		right = i
		break
for i in sorted(ver):
	if len(ver[i]) >= n:
		down = i
		break
for i in sorted(ver, reverse=True):
	if len(ver[i]) >= n:
		up = i
		break
for i in range(4 * n + 1):
	(x, y) = points[i]
	if x == left and down <= y <= up:
		continue
	if x == right and down <= y <= up:
		continue
	if y == down and left <= x <= right:
		continue
	if y == up and left <= x <= right:
		continue
	print(x, y)
	break
