(n, q) = list(map(int, input().strip().split()))
a = list(map(int, input().strip().split()))
res = a + [0] * n
lf = 0
rg = n - 1
results = []
for j in range(n):
	results.append([res[lf], res[lf + 1]])
	if res[lf] > res[lf + 1]:
		res[rg + 1] = res[lf + 1]
		res[lf + 1] = res[lf]
	else:
		res[rg + 1] = res[lf]
	lf += 1
	rg += 1
for qu in range(q):
	i = int(input())
	if i <= n:
		print(str(results[i - 1][0]) + ' ' + str(results[i - 1][1]))
	else:
		i = i - n
		i = i % (n - 1)
		if i == 0:
			i = n - 1
		print(str(res[lf]) + ' ' + str(res[lf + i]))
