#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
  }
  if (n % 2 == 0) {
    int x = 0;
    for (int i = 0; i < n; i++) {
      x = x ^ v[i];
    }
    if (x == 0) {
      cout << "YES\n" << n - 2 << "\n";
      for (int i = 1; i < n / 2; i++) {
        cout << 2 * i - 1 << " " << 2 * i << " " << n << "\n";
      }
      for (int i = 1; i < n / 2; i++) {
        cout << 2 * i - 1 << " " << 2 * i << " " << n << "\n";
      }
    } else {
      cout << "NO\n";
    }
  } else {
    cout << "YES\n" << n - 2 << "\n";
    for (int i = 1; i < (n + 1) / 2; i++) {
      cout << 2 * i - 1 << " " << 2 * i << " " << n << "\n";
    }
    for (int i = 1; i < (n - 1) / 2; i++) {
      cout << 2 * i - 1 << " " << 2 * i << " " << n << "\n";
    }
  }
}
