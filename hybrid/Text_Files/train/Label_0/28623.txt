void func(){

	int NUMBER,length;
	char buf[201];

	scanf("%d %s",&NUMBER,buf);
	for(length = 0; buf[length] != '\0'; length++);

	int maximum = 0,base_loc,del_loc;
	int B_NUM,G_NUM,loc;

	for(int count = 0; count < length; count++){
		list<char> LIST;
		for(int i = 0; i < length; i++)LIST.push_back(buf[i]);
		base_loc = 0;

		for(int loop = 0; ; loop++){
			if(loop == count){
				del_loc = (base_loc+NUMBER)%LIST.size();
			}else{
				del_loc = (base_loc+NUMBER-1)%LIST.size();
			}
			if(del_loc == (int)(LIST.size()-1)){
				base_loc = 0;
			}else{
				base_loc = del_loc;
			}


			loc = 0;
			for(list<char>::iterator it = LIST.begin(); it != LIST.end(); it++){
				if(loc == del_loc){
					LIST.erase(it);
					break;
				}
				loc++;
			}

			B_NUM = 0,G_NUM = 0;

			for(list<char>::iterator it = LIST.begin(); it != LIST.end(); it++){
				if(*it == 'B'){
					B_NUM++;
				}else{
					G_NUM++;
				}
			}
			if(B_NUM == 0 || G_NUM == 0){
				if(G_NUM == 0){
					maximum = max(maximum,B_NUM);
				}
				break;
			}
		}
	}

	printf("%d\n",maximum);
}


int main(){

	int N;
	scanf("%d",&N);

	for(int i = 0; i < N; i++)func();

	return 0;
}
