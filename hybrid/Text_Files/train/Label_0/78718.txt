def main():

	def solve():
		int(input())
		pp = [int(p) for p in input().split()]
		int(input())
		qq = [int(q) for q in input().split()]
		ppe = [p for p in pp if p % 2 == 0]
		qqe = [q for q in qq if q % 2 == 0]
		result = len(ppe) * len(qqe) + (len(pp) - len(ppe)) * (len(qq) - len(qqe))
		print(result)
	q = int(input())
	for _ in range(q):
		solve()
main()
