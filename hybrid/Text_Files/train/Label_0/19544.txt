#include <bits/stdc++.h>
using namespace std;
const long long mod = 1e9 + 7;
const int maxn = 70005;
string a[maxn];
string ans[maxn];
map<string, int> mymap, vis;
int main() {
  std::ios::sync_with_stdio(false);
  ;
  int n;
  cin >> n;
  mymap.clear();
  for (int i = 0; i < n; i++) {
    vis.clear();
    cin >> a[i];
    for (int k = 1; k <= 9; k++)
      for (int j = 0; j + k - 1 < 9; j++)
        if (vis[a[i].substr(j, k)] == 0) {
          vis[a[i].substr(j, k)] = 1, mymap[a[i].substr(j, k)]++;
        }
  }
  bool flag = 0;
  for (int i = 0; i < n; i++) {
    flag = 0;
    for (int k = 1; k <= 9; k++)
      if (!flag) {
        for (int j = 0; j + k - 1 < 9; j++)
          if (!flag) {
            if (mymap[a[i].substr(j, k)] == 1) {
              flag = 1;
              ans[i] = a[i].substr(j, k);
              break;
            }
          }
      }
  }
  for (int i = 0; i < n; i++) cout << ans[i] << '\n';
  return 0;
}
