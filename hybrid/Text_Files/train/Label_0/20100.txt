#include <bits/stdc++.h>
using namespace std;
int n;
string s;
int ltl[500000];
int ltr[500000];
long long getlap() {
  long long tot = 0;
  int mx = 0;
  for (int i = 0; i < n; ++i) {
    mx = max(mx, ltl[i]);
    tot += mx;
  }
  return tot;
}
int rmq[20][500001];
int pot[500001];
void build() {
  for (size_t s = 1, j = 1; s * 2 <= n + 1; s *= 2, ++j) {
    for (size_t i = 0; i + 2 * s <= n + 1; i++) {
      rmq[j][i] = max(rmq[j - 1][i], rmq[j - 1][i + s]);
    }
  }
}
int query(int l, int r) {
  int s = r - l;
  int lp = pot[s];
  return max(rmq[lp][l], rmq[lp][r - (1 << lp)]);
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n >> s;
  int p = 0;
  for (size_t i = 1; i < 500001; i++) {
    if (1 << (p + 1) < i) {
      ++p;
    }
    pot[i] = p;
  }
  if (s[0] == '1') ltl[0] = 1;
  for (int i = 1; i < n; ++i) {
    if (s[i] == '1')
      ltl[i] = ltl[i - 1] + 1;
    else
      ltl[i] = 0;
  }
  if (s[n - 1] == '1') ltr[n - 1] = 1;
  for (int i = n - 2; i >= 0; --i) {
    if (s[i] == '1')
      ltr[i] = ltr[i + 1] + 1;
    else
      ltr[i] = 0;
  }
  for (int i = 0; i < n; ++i) rmq[0][i] = ltl[i];
  rmq[0][n] = 100000000;
  build();
  long long firstlap = getlap();
  long long ans = firstlap;
  for (int i = 0; i < n - 1; ++i) {
    if (s[i] == '1') {
      int lessUntil = ltr[i];
      int searchStart = i + ltr[i];
      int lo = searchStart;
      int hi = n;
      while (lo < hi) {
        int mid = (lo + hi) / 2;
        int t = query(searchStart, mid + 1);
        if (t >= lessUntil) {
          hi = mid;
        } else {
          lo = mid + 1;
        }
      }
      int diff = lo - i;
      firstlap -= diff;
    }
    ans += firstlap;
  }
  cout << ans;
}
