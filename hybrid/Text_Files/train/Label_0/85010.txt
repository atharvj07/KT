z = int(input())
for i in range(z):
	ans = []
	(n, d, x, y) = list(map(int, input().split()))
	if abs(d - x) % y == 0:
		print(abs(d - x) // y)
		continue
	if (x - 1) % y == 0:
		ans.append((d - 1) // y + (x - 1) // y)
		if (d - 1) % y != 0:
			ans[-1] += 1
	if (n - x) % y == 0:
		ans.append((n - d) // y + (n - x) // y)
		if (n - d) % y != 0:
			ans[-1] += 1
	if len(ans) == 0:
		print(-1)
	else:
		print(min(ans))
