#include <bits/stdc++.h>
using namespace std;
vector<pair<long long int, pair<long long int, long long int> > > g[500011];
map<long long int, long long int> m, mr;
long long int from[500011];
long long int cnt = 0;
long long int d[500011];
long long int id[500011];
bool vis[500011];
long long int L;
long long int n;
vector<long long int> v;
void dj() {
  for (int i = 0; i < cnt; i++) from[i] = -1, d[i] = INFINITY;
  set<pair<long long int, long long int> > pq;
  pq.insert({0, 0});
  pair<long long int, long long int> t;
  from[0] = 0;
  d[0] = 0;
  while (pq.size()) {
    t = *pq.begin();
    t = {t.second, t.first};
    pq.erase(pq.begin());
    if (vis[t.first]) continue;
    vis[t.first] = 1;
    if (t.first == cnt - 1) {
      break;
    }
    for (auto x : g[t.first]) {
      if (!vis[x.first]) {
        if (d[x.first] > d[t.first] + x.second.first) {
          d[x.first] = d[t.first] + x.second.first;
          pq.insert({d[x.first], x.first});
          from[x.first] = t.first;
          id[x.first] = x.second.second;
        }
      }
    }
  }
  cout << d[cnt - 1] << "\n";
  long long int cur = cnt - 1;
  while (cur != 0) {
    if (id[cur]) v.emplace_back(id[cur]);
    cur = from[cur];
  }
  reverse(v.begin(), v.end());
  cout << v.size() << "\n";
  for (auto x : v) cout << x << " ";
}
int main() {
  cin >> n >> L;
  long long int x[n], d[n], t[n], p[n];
  set<long long int> second;
  for (int i = 0; i < n; i++) {
    cin >> x[i] >> d[i] >> t[i] >> p[i];
    if (x[i] - p[i] >= 0) second.insert(x[i] - p[i]);
    second.insert(x[i] + d[i]);
  }
  second.insert(0);
  second.insert(L);
  for (auto x : second) {
    m[cnt] = x;
    mr[x] = cnt;
    if (cnt != 0) {
      g[cnt - 1].emplace_back(cnt, make_pair(x - m[cnt - 1], 0));
      g[cnt].emplace_back(cnt - 1, make_pair(x - m[cnt - 1], 0));
    }
    cnt++;
  }
  for (int i = 0; i < n; i++) {
    if (x[i] - p[i] >= 0 and t[i] <= d[i]) {
      g[mr[x[i] - p[i]]].emplace_back(mr[x[i] + d[i]],
                                      make_pair(p[i] + t[i], i + 1));
    }
  }
  dj();
}
