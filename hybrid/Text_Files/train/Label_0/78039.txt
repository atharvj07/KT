t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	(xor, acc_a, pos0) = (0, [], [])
	for i in range(n):
		xor ^= a[i]
		acc_a.append(xor)
		if xor == 0:
			pos0.append(i)
	if xor == 0:
		print('YES')
	else:
		T = len(pos0) > 0 and xor in acc_a[:pos0[-1]]
		print('YES' if T else 'NO')
