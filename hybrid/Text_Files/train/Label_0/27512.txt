#include <bits/stdc++.h>
using namespace std;
int w, h;
char s[105][105];
int main() {
  scanf("%d %d", &w, &h);
  for (int i = (0); i <= (h - 1); i++) scanf("\n%s", s[i]);
  for (int i = (0); i <= (w + w - 1); i++) {
    for (int j = (0); j <= (h + h - 1); j++) printf("%c", s[j / 2][i / 2]);
    printf("\n");
  }
  return 0;
}
