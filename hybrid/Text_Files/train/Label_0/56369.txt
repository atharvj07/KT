
import java.util.Arrays;
import java.util.Scanner;


public class Main {
    public static void main(String[] args) {
        Scanner r = new Scanner(System.in);
        
        N = r.nextInt();
        u = r.nextInt();
        R = r.nextInt();
        
        int[] a = new int[N];
        b = new int[N];
        k = new long[N];
        t = new int[N];
        p = new int[N];
        
        for(int i = 0; i < N; i++)
            a[i] = r.nextInt();
        for(int i = 0; i < N; i++)
            b[i] = r.nextInt();
        for(int i = 0; i < N; i++)
            k[i] = r.nextInt();
        for(int i = 0; i < N; i++)
            p[i] = r.nextInt() - 1;
        
        long res = rec(0, a, true);
        
        System.out.println(res);
    }
    private static long rec(int i, int[] a, boolean can) {
        long max = Long.MIN_VALUE;
        if((u - i) % 2 == 0){
            long s = 0;
            for(int j = 0; j < N; j++)
                s += a[j] * k[j];
            max = s;
        }
        
        if(i == u){
            long s = 0;
            for(int j = 0; j < N; j++)
                s += a[j] * k[j];
            return s;
        }else{
            if(can){    
                for(int j = 0; j < N; j++)
                    a[j] ^= b[j];
                max = Math.max(max, rec(i + 1, a, false));
                for(int j = 0; j < N; j++)
                    a[j] ^= b[j];
            }
            
            for(int j = 0; j < N; j++)
                t[j] = a[p[j]] + R;
            for(int j = 0; j < N; j++)
                a[j] = t[j];
            
            max = Math.max(max, rec(i + 1, a, true));
            
            for(int j = 0; j < N; j++)
                t[p[j]] = a[j] - R;
            for(int j = 0; j < N; j++)
                a[j] = t[j];
            
            return max;
        }
    }
    static int N, u, R;
    static long[] k;
    static int[] b, t, p;
    
}
