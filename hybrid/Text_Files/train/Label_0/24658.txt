#include <bits/stdc++.h>
using namespace std;
const int maxn = 222222;
int a[maxn];
int f[maxn];
int book[maxn];
int find(int x) {
  if (f[x] == x) return x;
  return f[x] = find(f[x]);
}
int merge(int x, int y) {
  int u = find(x);
  int v = find(y);
  if (u != v) {
    f[v] = u;
    return 1;
  }
  return 0;
}
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    f[i] = i;
  }
  int cnt = 0;
  int pre = -1;
  for (int i = 1; i <= n; i++) {
    if (!merge(a[i], i) && a[i] != i) {
      cnt++;
      if (pre != -1) {
        merge(i, pre);
      }
      pre = i;
      book[i] = 1;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (i == f[i]) {
      cnt++;
      book[i] = 1;
      if (pre != -1) {
        merge(i, pre);
      }
      pre = i;
    }
  }
  printf("%d\n", cnt - 1);
  for (int i = 1; i <= n; i++) {
    if (book[i])
      printf("%d ", f[i]);
    else
      printf("%d ", a[i]);
  }
  return 0;
}
