import math

def euler_totient(m):
	ans = m
	i = 2
	while i * i <= m:
		if m % i == 0:
			ans -= ans // i
			while m % i == 0:
				m //= i
		i += 1
	if m > 1:
		ans -= ans // m
	return ans
t = int(input())
while t > 0:
	(a, m) = map(int, input().split())
	print(euler_totient(m // math.gcd(a, m)))
	t -= 1
