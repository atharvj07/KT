from __future__ import nested_scopes, generators, division, absolute_import, with_statement, print_function, unicode_literals
import sys, os, time

def rv():
	return map(int, input().split())

def rl(n):
	return [list(map(int, input().split())) for _ in range(n)]
if os.path.exists('test.txt'):
	sys.stdin = open('test.txt')

def get(k):
	if k == 0:
		return [[1]]
	smaller = get(k - 1)
	newdim = len(smaller) * 2
	res = [[0] * newdim for _ in range(newdim)]
	for row in range(newdim):
		for col in range(newdim):
			res[row][col] = smaller[row % len(smaller)][col % len(smaller)]
			if row >= len(smaller) and col >= len(smaller):
				res[row][col] *= -1
	return res

def solve():
	(k,) = rv()
	res = get(k)
	for row in res:
		for char in row:
			print('+' if char == 1 else '*', end='')
		print()
solve()
