a = int(input())
for i in range(a):
	(x, y) = list(map(int, input().split()))
	if x % 2 == 0:
		if y - x == 0:
			print(x)
		else:
			ans = y - x + 1
			if ans % 2 == 0:
				print(int(-1 * ans / 2))
			else:
				print(int(y + -1 * (ans - 1) / 2))
	elif y - x == 0:
		print(-x)
	else:
		ans = y - x + 1
		if ans % 2 == 0:
			print(int(ans / 2))
		else:
			print(int(-y + (ans - 1) / 2))
