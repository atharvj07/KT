#include <bits/stdc++.h>
using namespace std;
signed main() {
  long long n;
  cin >> n;
  long long A[n], B[n], C[n];
  for (long long i = 0; i < n; i++) {
    cin >> A[i];
    C[A[i]] = i;
  }
  multiset<long long> X, Y;
  for (long long i = 0; i < n; i++) {
    cin >> B[i];
    long long x = i - C[B[i]];
    if (x <= 0)
      X.insert(x);
    else
      Y.insert(x);
  }
  long long ans[n];
  long long pp = 0;
  for (long long i = 0; i < n; i++) {
    auto it = X.end();
    it--;
    long long y = 1000000000;
    if (X.size() > 0) y = min(y, abs(*it + pp));
    auto itr = Y.begin();
    if (Y.size() > 0) y = min(y, abs(*itr + pp));
    ans[i] = y;
    pp--;
    long long q = i - C[B[i]];
    auto itrr = X.lower_bound(q);
    X.erase(itrr);
    Y.insert(q + n);
    long long cnt = 0;
    for (auto itt : Y) {
      if (itt == -pp)
        cnt++;
      else
        break;
    }
    Y.erase(-pp);
    for (long long j = 0; j < cnt; j++) {
      X.insert(-pp);
    }
  }
  for (long long i = 0; i < n; i++) {
    cout << ans[i] << endl;
  }
}
