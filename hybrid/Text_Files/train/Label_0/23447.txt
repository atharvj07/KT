#include <bits/stdc++.h>
using namespace std;
int n, m;
vector<int> adj[100001];
int color[100001];
bool valid = true;
void dfs(int u) {
  for (int v : adj[u]) {
    if (color[v] == 0) {
      color[v] = -color[u];
      dfs(v);
    } else if (color[v] == color[u]) {
      valid = false;
    }
  }
}
int main() {
  cin >> n >> m;
  int u, v;
  for (int i = 0; i < m; i++) {
    cin >> u >> v;
    u--;
    v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  for (int i = 0; i < n; i++) {
    if (color[i] == 0) {
      color[i] = 1;
      dfs(i);
    }
  }
  if (valid) {
    vector<int> a, b;
    for (int i = 0; i < n; i++) {
      if (color[i] == -1)
        a.push_back(i + 1);
      else
        b.push_back(i + 1);
    }
    cout << a.size() << endl;
    for (int c : a) {
      cout << c << " ";
    }
    cout << endl;
    cout << b.size() << endl;
    for (int c : b) {
      cout << c << " ";
    }
    cout << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
