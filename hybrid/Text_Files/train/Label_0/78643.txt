n = int(input())
x = list(map(int, input().split()))
x.sort()
flag = [False] * (n + 2)
for xi in x:
	if not flag[xi - 1]:
		flag[xi - 1] = True
	elif not flag[xi]:
		flag[xi] = True
	else:
		flag[xi + 1] = True
ans1 = flag.count(True)
x = list(set(x))
flag = [False] * (n + 2)
for xi in x:
	if flag[xi - 1] or flag[xi]:
		continue
	else:
		flag[xi + 1] = True
ans0 = flag.count(True)
print(ans0, ans1)
