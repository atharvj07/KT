def main():
	(n, aa, s) = (int(input()), list(map(float, input().split())), input())
	(l, r, abuf, bbuf) = (-1000000001.0, 1000000001.0, aa[:5], [False] * 4)
	for i in range(4, n):
		(abuf[i % 5], b) = (aa[i], s[i] == '1')
		if b:
			if not any(bbuf) and l < max(abuf):
				l = max(abuf)
		elif all(bbuf) and r > min(abuf):
			r = min(abuf)
		bbuf[i % 4] = b
	print(int(l) + 1, int(r) - 1)
main()
