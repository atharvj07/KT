def gcd(a, b):
	if b == 0:
		return a
	else:
		return gcd(b, a % b)
ls = list(map(int, input().split()))
t = ls[0]
w = ls[1]
b = ls[2]
tmp = 0
if w < b:
	tmp = w
	w = b
	b = tmp
x = int(w * b // gcd(w, b))
cnt = int(t // x)
r = int(t - cnt * x)
res = int(cnt * b + min(r, b - 1))
print('{:d}/{:d}'.format(int(res // gcd(res, t)), int(t // gcd(t, res))))
