#include <bits/stdc++.h>
const long long mxn = 1e6 + 110;
const long long mnn = 1e3 + 2;
const long long mod = 1e9 + 7;
const long long inf = 1e18;
const long long OO = 1e9;
using namespace std;
long long n, x, y, p, ini[mxn];
vector<long long> v(5, 0);
vector<vector<long long> > mul(vector<vector<long long> > a,
                               vector<vector<long long> > b) {
  vector<vector<long long> > ans(5, vector<long long>(5, 0));
  for (int i = 0; i < 5; i++) {
    for (int j = 0; j < 5; j++) {
      for (int k = 0; k < 5; k++) {
        ans[i][j] = (ans[i][j] + a[i][k] * b[k][j]) % p;
      }
    }
  }
  return ans;
}
vector<vector<long long> > sym() {
  vector<vector<long long> > lol(5, vector<long long>(5, 0));
  for (int i = 0; i < 5; i++) lol[i][i] = 1;
  return lol;
}
vector<vector<long long> > binpow(vector<vector<long long> > a, long long k) {
  if (k == 0) {
    return sym();
  } else if (k % 2 == 1) {
    return mul(binpow(a, k - 1), a);
  } else {
    vector<vector<long long> > x = binpow(a, k / 2);
    return mul(x, x);
  }
}
vector<long long> going_to_havchik(vector<long long> v, long long k) {
  vector<vector<long long> > mat(5, vector<long long>(5, 0));
  mat[0][0] = 1;
  mat[1][0] = mat[1][4] = -1;
  mat[1][1] = 3;
  mat[2][3] = mat[3][2] = mat[3][3] = 1;
  mat[4][4] = 1;
  mat = binpow(mat, k);
  vector<long long> nw(5, 0);
  for (int i = 0; i < 5; i++) {
    for (int k = 0; k < 5; k++) {
      nw[i] = (nw[i] + mat[i][k] * v[k]) % p;
      nw[i] = (nw[i] + p) % p;
    }
  }
  return nw;
}
int main() {
  ios_base::sync_with_stdio(0), cout.tie(0), cin.tie(0);
  ;
  cin >> n >> x >> y >> p;
  for (int i = 1; i <= n; i++) {
    cin >> ini[i];
    ini[i] %= p;
    v[1] = (v[1] + ini[i]) % p;
  }
  if (n == 1) {
    cout << ini[1];
    return 0;
  }
  v[0] = ini[1], v[4] = ini[n];
  v[2] = ini[n - 1];
  v[3] = ini[n];
  v = going_to_havchik(v, x);
  v[4] = v[3];
  v = going_to_havchik(v, y);
  cout << v[1];
}
