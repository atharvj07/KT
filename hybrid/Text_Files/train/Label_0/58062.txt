import java.util.*;
public class MoleculesCodeforces
{
	public static void main(String args[])
	{
		Scanner sc =new Scanner(System.in);
		int a=sc.nextInt();
		int b=sc.nextInt();
		int c=sc.nextInt();
		int max=Math.max(a,Math.max(b,c));
		int count=0; 
		if(max==a)
		{
			while((b+c)>a)
			{
				count++;
				b--;
				c--;
			}
			if(b+c==a)
			{
			  System.out.print(b+" ");
			  System.out.print(count+" ");
			  System.out.println(c+" ");
			}
			else
			System.out.println("Impossible");
		}
		if(max==b)
		{
			while((a+c)>b)
			{
				count++;
				a--;
				c--;
			}
			if(a+c==b)
			{
			  System.out.print(a+" ");
			  System.out.print(c+" ");
			  System.out.print(count+" ");
			}
			else
			System.out.println("Impossible");
		}
		if(max==c)
		{
			while((a+b)>c)
			{
				count++;
				b--;
				a--;
			}
			if(a+b==c)
			{
			  System.out.print(count+" ");
			  System.out.print(b+" ");
			  System.out.print(a+" ");
			}
			else
			System.out.println("Impossible");
		}
	}
}
