import math
t = int(input())
answers = []
for i in range(t):
	n = int(input())
	if n % 2 == 0:
		answers.append([n / 2, n / 2])
	else:
		factor = 1
		for j in range(2, int(math.sqrt(n)) + 1):
			if n % j == 0:
				factor = n / j
				break
		answers.append([factor, n - factor])
for i in range(t):
	for j in range(2):
		print(int(answers[i][j]), end=' ')
	print()
