#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, j, k, m, n;
  while (cin >> m) {
    if (m == 1) {
      cout << 1 << ' ' << 2 << endl;
      cout << 1 << ' ' << 2 << endl;
      continue;
    }
    if (m == 2) {
      cout << 2 << ' ' << 2 << endl;
      cout << 1 << ' ' << 2 << endl;
      continue;
    }
    m--;
    cout << m * 2 << ' ' << 2 << endl;
    cout << 1 << ' ' << 2 << endl;
  }
  return 0;
}
