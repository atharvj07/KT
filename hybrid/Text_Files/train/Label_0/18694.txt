#include <bits/stdc++.h>
using namespace std;
char str[100005];
int pre[100005], prek[100005];
int main() {
  int n, k, w;
  scanf("%d%d%d%s", &n, &k, &w, str + 1);
  for (int i = (1); i < (n + 1); i++) pre[i] = pre[i - 1] + str[i] - '0';
  for (int i = (k); i < (n + 1); i++) prek[i] = prek[i - k] + str[i] - '0';
  while (w--) {
    int l, r;
    scanf("%d%d", &l, &r);
    int ans1 = pre[r] - pre[l - 1];
    int num = (r - l) / k + 1;
    int ans2 = prek[r] - prek[l - 1];
    printf("%d\n", ans1 - ans2 + num - ans2);
  }
  return 0;
}
