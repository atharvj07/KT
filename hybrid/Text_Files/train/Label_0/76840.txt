import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, q) = map(int, input().split())
	rod = input()
	dp = [0]
	for (i, j) in enumerate(rod):
		dp.append(dp[-1] + 1) if (j == '+') ^ (i % 2 == 0) else dp.append(dp[-1] - 1)
	for i in range(q):
		(l, r) = map(int, input().split())
		if dp[r] - dp[l - 1] == 0:
			print(0)
		elif (dp[r] - dp[l - 1]) % 2 == 1:
			print(1)
		else:
			print(2)
