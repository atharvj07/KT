#include <bits/stdc++.h>
using namespace std;
int a[1001];
int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  for (int i = 1; i <= n; ++i) scanf("%d", a + i);
  sort(a + 1, a + n + 1);
  int ans = 0;
  k *= 2;
  for (int i = 1; i <= n; ++i) {
    while (k < a[i]) {
      ++ans;
      k *= 2;
    }
    k = max(k, a[i] * 2);
  }
  printf("%d", ans);
  return 0;
}
