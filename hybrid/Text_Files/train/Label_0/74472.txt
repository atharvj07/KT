t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	start = -1
	end = -1
	i = 0
	for c in s:
		if c == '1' and start == -1:
			start = i
		if c == '1':
			end = i + 1
		i += 1
	if start == -1:
		print(n)
	else:
		start = n - start
		if start > end:
			print(start * 2)
		else:
			print(end * 2)
