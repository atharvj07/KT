for _ in range(int(input())):
	s = input()
	t = input()
	x = [0] * 3
	xd = []
	for i in range(len(s)):
		if s[i] == 'a':
			x[0] += 1
		elif s[i] == 'b':
			x[1] += 1
		elif s[i] == 'c':
			x[2] += 1
		else:
			xd += [s[i]]
	xd = sorted(xd)
	if x[0] > 0 and x[1] > 0 and (x[2] > 0) and (t == 'abc'):
		for _ in range(x[0]):
			print('a', end='')
		for _ in range(x[2]):
			print('c', end='')
		for _ in range(x[1]):
			print('b', end='')
		for i in range(len(xd)):
			print(xd[i], end='')
		print()
	else:
		for _ in range(x[0]):
			print('a', end='')
		for _ in range(x[1]):
			print('b', end='')
		for _ in range(x[2]):
			print('c', end='')
		for i in range(len(xd)):
			print(xd[i], end='')
		print()
