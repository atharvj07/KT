// package Quarantine;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class WorldIsJustAProgRammingTask {
    public static void main(String[] args)throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        int n=Integer.parseInt(br.readLine());
        char c[]=br.readLine().toCharArray();
        int max=0;
        int l=1,r=1;
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                char d[]=new char[n];
                for(int k=0;k<n;k++){
                    d[k]=c[k];
                }
                char temp=d[j];
                d[j]=d[i];
                d[i]=temp;
//                System.out.println(i+" "+j);
                int small=solve(d,n);
                if(small>max){
                    max=small;
                    l=i+1;
                    r=j+1;
                }
            }
        }
        System.out.println(max);
        System.out.println(l+" "+r);
    }
    public static int solve(char c[],int n){
        int dp[]=new int[n];
        dp[0]=c[0]=='('?1:-1;
//        System.out.print(dp[0]+" ");
        for(int i=1;i<n;i++){
            if(c[i]=='('){
                dp[i]=dp[i-1]+1;
            }
            else{
                dp[i]=dp[i-1]-1;
            }
//            System.out.print(dp[i]+" ");
        }
//        System.out.println();
        int min[]=new int[n];
        int min2[]=new int[n];
        min2[0]=dp[0];
        for(int i=1;i<n;i++){
            min2[i]=Math.min(min2[i-1],dp[i]);
        }
        min[n-1]=dp[n-1];
        for(int i=n-2;i>=0;i--){
            min[i]=Math.min(dp[i],min[i+1]);
        }
        int ans=0;
        if(dp[0]==1&&dp[n-1]==0&&min[0]==0){
            ans++;
        }
        for(int i=0;i<n-1;i++){
            if(dp[i]+1==dp[i+1]&&min[i+1]>=dp[i]&&min2[i]>=dp[i]&&dp[n-1]==0){
                ans++;
            }
        }
//        System.out.println(ans);
        return ans;
    }
}
