def list_from_input():
	return list(map(int, input().split()))

class Snag:

	def __init__(self, x, y, r):
		self.x = x
		self.y = y
		self.r = r

	@classmethod
	def from_input(cls):
		return Snag(*list_from_input())

	def dist_to_pizza_center(self):
		return (self.x ** 2 + self.y ** 2) ** 0.5

	def is_on_board(self, r, d):
		l = self.dist_to_pizza_center()
		return l - self.r >= r - d and l + self.r <= r

def main():
	(r, d) = list_from_input()
	n = int(input())
	count = 0
	for i in range(n):
		next_snack = Snag.from_input()
		if next_snack.is_on_board(r, d):
			count += 1
	print(count)
main()
