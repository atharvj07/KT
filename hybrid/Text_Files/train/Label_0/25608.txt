#include <bits/stdc++.h>
using namespace std;
int a[100010], res = 0, n;
void solve(void) {
  vector<int> st;
  for (int i = 0; i < n; i++) {
    while (!st.empty() && st.back() <= a[i]) st.pop_back();
    st.push_back(a[i]);
    if (st.size() >= 2) res = max(res, st[st.size() - 1] ^ st[st.size() - 2]);
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  solve();
  reverse(a, a + n);
  solve();
  cout << res << endl;
  return 0;
}
