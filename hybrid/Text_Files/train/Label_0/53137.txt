    private static final long INF=(long)(1e18);
    private static ArrayDeque<Pair>[] edge;
    private static long dist[][], dist2[];
    private static int[] parent;
    private static ArrayList<Pair> pairs;

    private static int tin[], tout[], timer=0,N, logN, anc[][];

    static class Pair implements Comparable<Pair>
    {
        int f; long s;
        Pair(int f, long s){this.f = f;this.s = s;}
        public int compareTo(Pair b){return Long.compare(this.s,b.s);}
    }
    private static void Dijkstra(int id, int s)
    {
        Arrays.fill(dist[id],INF);
        dist[id][s]=0;

        PriorityQueue<Pair> PQ=new PriorityQueue<>();
        PQ.add(new Pair(s,0));

        while (!PQ.isEmpty())
        {
            Pair tmp=PQ.poll();
            int u=tmp.f;
            long cur=tmp.s;
            if(cur!=dist[id][u]) continue;

            for(Pair p: edge[u])
            {
                int v=p.f;
                long d=p.s;

                if(dist[id][u]+d<dist[id][v])
                {
                    dist[id][v]=dist[id][u]+d;
                    PQ.add(new Pair(v,dist[id][v]));
                }
            }
        }
    }
    private static void DFS(int u, int p)
    {
        tin[u]=timer++;
        anc[u][0]=p;

        for(int i=1;i<=logN;i++)
            anc[u][i]=anc[anc[u][i-1]][i-1];

        for(Pair tmp:edge[u])
        {
            int v=tmp.f;
            long d=tmp.s;

            if(v==p) continue;
            else if(parent[v]==-1)
            {
                parent[v]=u; dist2[v]=dist2[u]+d;
                DFS(v,u);
            }
            else if(u<v) pairs.add(new Pair(u,v));
        }

        tout[u]=timer++;
    }
    private static boolean isAncestor(int u, int v)
    {
        return tin[u]<=tin[v]&&tout[v]<=tout[u];
    }
    private static int LCA(int u, int v)
    {
        if(isAncestor(u,v)) return u;
        if(isAncestor(v,u)) return v;

        for(int i=logN;i>=0;i--)
        {
            if (!isAncestor(anc[u][i], v))
                u = anc[u][i];
        }
        return anc[u][0];
    }
    private static void init()
    {
        logN=Integer.numberOfTrailingZeros(Integer.highestOneBit(N));
        tin=new int[N];
        tout=new int[N];
        anc=new int[N][logN+1];
    }
    public static void main(String[] args) throws IOException
    {
        Reader r=new Reader();

        int i;

        N=r.nextInt();
        int M=r.nextInt();

        edge=new ArrayDeque[N];
        for(i=0;i<N;i++) edge[i]=new ArrayDeque<>();

        for(i=0;i<M;i++)
        {
            int u=r.nextInt()-1;
            int v=r.nextInt()-1;
            int d=r.nextInt();

            edge[u].add(new Pair(v,d));
            edge[v].add(new Pair(u,d));
        }

        parent=new int[N]; dist2=new long[N];
        pairs=new ArrayList<>();

        init();

        Arrays.fill(parent,-1);
        parent[0]=0;
        DFS(0,0);

        int lim=2*pairs.size();
        dist=new long[lim][N];

        int k=0;
        for(Pair tmp:pairs)
        {
            int u=tmp.f;
            int v=(int)tmp.s;

            Dijkstra(k++,u);
            Dijkstra(k++,v);
        }

        int Q=r.nextInt();
        StringBuilder sb=new StringBuilder();

        while (Q-->0)
        {
            int u=r.nextInt()-1;
            int v=r.nextInt()-1;

            long ans=dist2[u]+dist2[v]-2*dist2[LCA(u,v)];
            for(i=0;i<lim;i++) ans=Math.min(ans,dist[i][u]+dist[i][v]);

            sb.append(ans).append("\n");
        }
        System.out.println(sb);
    }
