#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int l1, r1, l2, r2;
    cin >> l1 >> r1 >> l2 >> r2;
    int on = 0;
    for (int i = l1; i <= r1; i++) {
      for (int j = l2; j <= r2; j++) {
        if (i != j) {
          cout << i << " " << j << endl;
          on = 1;
          break;
        }
      }
      if (on == 1) {
        break;
      }
    }
  }
  return 0;
}
