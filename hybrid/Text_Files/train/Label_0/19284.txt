#include<iostream>
#include<vector>
#include<algorithm>

using namespace std;

const int MAX = 1e6+1;
int score[MAX];
int N,Q;
vector<int> leader;
vector<int> sc;

void input(){
  cin >> N >> Q;
  sc.resize(N);
  for(int i = 1; i <= N; i++) {
    cin >> score[i];
    sc[i-1] = score[i];
  }
  sort(sc.begin(),sc.end());
}

void add(int n){
  leader.push_back(n);
}

void remove(int n){
  vector<int>::iterator ite = leader.begin();
  while(ite < leader.end()){
    if(*ite == n) ite = leader.erase(ite);
    else ite++;
  }
}

bool isOK(int r, int n){
  int out = 0;
  int pos = 0; // top position that is not cavered yet

  vector<int> v(leader.size());
  for(int i = 0; i < (int)leader.size(); i++) v[i] = score[leader[i]];
  sort(v.begin(),v.end());
  
  for(int i = 0; i < (int)v.size(); i++){
    int llim = lower_bound(sc.begin(),sc.end(),v[i]-r) - sc.begin();
    int rlim = upper_bound(sc.begin(),sc.end(),v[i]) - sc.begin();
    
    // can cover [llim, rlim)
    if(llim > pos) out += llim-pos;
    pos = rlim;
  }

  out += sc.size()-pos;
  return out <= n;
}

void check(int n){
  const int LIM = 1e9+1;
  int l = 0, r = LIM;

  while(l < r){
    int c = (l+r)/2;
    if(isOK(c,n)) r = c;
    else l = c+1;
  }

  if(r == LIM) cout << "NA" << endl;
  else cout << l << endl;
}

void solve(){

  for(int i = 0; i < Q; i++){
    string q;
    int n;
    cin >> q >> n;
    if(q == "ADD") add(n);
    if(q == "REMOVE") remove(n);
    if(q == "CHECK") check(n);
  }
}

int main(){
  input();
  solve();
  return 0;
}
