import java.io.PrintWriter;
import java.util.*;

public class Main {

   Scanner in;
   PrintWriter out;

   void asserT(boolean e) {
      if (!e) {
         throw new Error();
      }
   }

   class Man {
      int id;
      HashSet<Man> friends = new HashSet<Main.Man>();
      HashSet<Man> realFriends = new HashSet<Main.Man>();
      HashSet<Man> enimes = new HashSet<Main.Man>();
      boolean wasUsed = false;
      int nUsedFriends = 0;

      public Man(int id) {
         this.id = id;
      }

      @Override
      public int hashCode() {
         final int prime = 31;
         int result = 1;
         result = prime * result + getOuterType().hashCode();
         result = prime * result + id;
         return result;
      }

      @Override
      public boolean equals(Object obj) {
         if (this == obj)
            return true;
         if (obj == null)
            return false;
         if (getClass() != obj.getClass())
            return false;
         Man other = (Man) obj;
         if (!getOuterType().equals(other.getOuterType()))
            return false;
         if (id != other.id)
            return false;
         return true;
      }

      private Main getOuterType() {
         return Main.this;
      }

      void dfs(HashSet<Man> set) {
         this.wasUsed = true;
         set.add(this);
         for (Man next : friends) {
            if (next.id != this.id && !next.wasUsed && this.friends.contains(next)) {
               next.dfs(set);
            }

         }
      }

   }

   void solve() {
      int nMans = in.nextInt();
      Man mans[] = new Man[1 + nMans];
      int nPairs = in.nextInt();
      for (int i = 0; i < mans.length; i++) {
         mans[i] = new Man(i);
      }
      for (int i = 0; i < nPairs; i++) {
         int first = in.nextInt();
         int second = in.nextInt();
         mans[first].friends.add(mans[second]);
         mans[second].friends.add(mans[first]);

      }
      nPairs = in.nextInt();
      for (int i = 0; i < nPairs; i++) {
         int first = in.nextInt();
         int second = in.nextInt();
         mans[first].enimes.add(mans[second]);
         mans[second].enimes.add(mans[first]);
      }
      int best = 0;
      for (int i = 1; i <= nMans; i++) {
         Man man = mans[i];
         if (!man.wasUsed) {
            HashSet<Man> com = new HashSet<Main.Man>();
            man.dfs(com);
            if (isOk(com)) {
               best = Math.max(best, com.size());
            }
         }

      }
      out.println(best);
   }

   void reset(HashSet<Man> set) {
      for (Man man : set) {
         man.nUsedFriends = 0;
      }
   }

   boolean isOk(HashSet<Man> set) {
      reset(set);
      for (Man first : set) {
         for (Man second : set) {
            if (first.id != second.id) {
               if (first.friends.contains(second)) {
                  first.nUsedFriends++;
               } else if (first.enimes.contains(second)) {
                  return false;
               }
            }
         }
      }
      for (Man man : set) {
         if (man.nUsedFriends != man.friends.size()) {
            return false;
         }
      }
      return true;
   }

   void run() {

      in = new Scanner(System.in);
      out = new PrintWriter(System.out);

      try {
         solve();
      } finally {
         out.close();
      }
   }

   public static void main(String[] args) {
      new Main().run();
   }
}
