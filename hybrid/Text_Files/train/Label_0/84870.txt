def solve(A, n, k):
	prev = -1
	dist = []
	for it in A:
		if it != prev:
			dist.append(it)
			prev = it
	td = len(dist)
	if k == 1:
		if td == 1:
			return 1
		else:
			return -1
	if td <= k:
		return 1
	ans = (td - 2) // (k - 1) + 1
	return ans
for case in range(int(input())):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	ans = solve(a, n, k)
	print(ans)
