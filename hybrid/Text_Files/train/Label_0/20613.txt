#include <bits/stdc++.h>
using namespace std;
string E[256], F[256], T[256];
int cmp(string &a, string &b) {
  return ((int)(a).size()) != ((int)(b).size())
             ? ((int)(a).size()) < ((int)(b).size())
             : a < b;
}
int upd(string &a, string b) {
  if (a == "" || cmp(b, a)) {
    a = b;
    return 1;
  }
  return 0;
}
int main() {
  F[0b11110000] = "x";
  F[0b11001100] = "y";
  F[0b10101010] = "z";
  while (1) {
    int ch = 0;
    for (int i = 0; i < 256; i++)
      if (E[i] != "") ch |= upd(F[i], "(" + E[i] + ")");
    for (int i = 0; i < 256; i++)
      if (F[i] != "") ch |= upd(F[i ^ 255], "!" + F[i]);
    for (int i = 0; i < 256; i++)
      if (F[i] != "") ch |= upd(T[i], F[i]);
    for (int i = 0; i < 256; i++)
      if (T[i] != "")
        for (int j = 0; j < 256; j++)
          if (F[j] != "") ch |= upd(T[i & j], T[i] + "&" + F[j]);
    for (int i = 0; i < 256; i++)
      if (T[i] != "") ch |= upd(E[i], T[i]);
    for (int i = 0; i < 256; i++)
      if (E[i] != "")
        for (int j = 0; j < 256; j++)
          if (T[j] != "") ch |= upd(E[i | j], E[i] + "|" + T[j]);
    if (ch == 0) break;
  }
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    char ch[10];
    scanf("%s", ch);
    int c = 0;
    for (int i = 0; i < 8; i++) c |= (ch[i] - '0') << i;
    printf("%s\n", E[c].c_str());
  }
}
