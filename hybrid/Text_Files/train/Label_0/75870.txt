import sys
n = sys.stdin.buffer.readline()
for i in range(int(n)):
	(a, b, h) = map(int, input().split())
	f = 0
	s = input()
	s = list(s)
	n = len(s)
	j = n - 1
	sum_ = 0
	while j >= 0 and sum_ <= h:
		if s[j - 1] != s[j] or j == n - 1:
			if s[j - 1] == 'A':
				sum_ += a
			else:
				sum_ += b
		j -= 1
	print(j + 1 + 1)
