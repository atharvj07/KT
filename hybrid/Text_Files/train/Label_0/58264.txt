import java.util.Scanner;

//*1300
public class Multiplication_Table
{
    static Scanner sc = new Scanner(System.in);
    public static void main(String[] args)
    {


            solve();


    }
    protected  static  void solve()
    {
        int n = sc.nextInt();
        long a[][] = new long[n][n];
       for(int i = 0; i < n; i++)
        {
            for(int j = 0;j<n;j++)
            {
                a[i][j] = sc.nextLong();
            }
        }
        long k= (long)Math.sqrt((a[0][1]*a[0][2])/a[1][2]);


        System.out.print(k+" ");
        for(int i = 1;i<n;i++)
            System.out.print(a[0][i]/k+" ");
        System.out.println();


    }
}
