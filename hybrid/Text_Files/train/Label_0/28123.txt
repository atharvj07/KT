#include <bits/stdc++.h>
using namespace std;
mt19937 rng((int)std::chrono::steady_clock::now().time_since_epoch().count());
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, b, a;
  cin >> n >> b >> a;
  vector<int> c(n);
  for (auto &x : c) cin >> x;
  int maxa = a;
  int ans = 0;
  for (auto &x : c) {
    if (x == 0) {
      if (a)
        a--;
      else if (b)
        b--;
      else
        break;
    } else {
      if (a == maxa)
        a--;
      else if (b)
        b--, a = min(a + 1, maxa);
      else if (a)
        a--;
      else
        break;
    }
    ans++;
  }
  cout << ans << endl;
}
