#include <bits/stdc++.h>
using namespace std;
vector<vector<long long> > g;
vector<long long> a;
vector<bool> vis;
vector<long long> amx, appnd;
long long ans;
long long dfs(long long s) {
  vis[s] = true;
  long long cur = 0;
  for (int i = 0; i < g[s].size(); i++) {
    if (!vis[g[s][i]]) {
      cur += dfs(g[s][i]);
    }
  }
  ans += (a[s] + cur);
  if (a[s] + cur < 0) {
    appnd.push_back(s);
    return 0;
  } else {
    amx.push_back(s);
    return a[s] + cur;
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long T = 1;
  while (T--) {
    long long n;
    cin >> n;
    a.clear();
    a.resize(n + 1);
    vis.clear();
    vis.resize(n + 1);
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
    }
    g.clear();
    g.resize(n + 1);
    vector<long long> in(n + 1);
    for (int i = 1; i <= n; i++) {
      long long b;
      cin >> b;
      if (b != -1) {
        g[b].push_back(i);
        in[i]++;
      }
    }
    ans = 0;
    amx.clear();
    for (int i = 1; i <= n; i++) {
      if (in[i] == 0) {
        dfs(i);
      }
    }
    long long m = appnd.size();
    for (int i = m - 1; i >= 0; i--) {
      amx.push_back(appnd[i]);
    }
    cout << ans << "\n";
    for (int i = 0; i < n; i++) {
      cout << amx[i] << " ";
    }
    cout << "\n";
  }
}
