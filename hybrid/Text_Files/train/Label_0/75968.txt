def main():
	(num, len_r) = map(int, input().split())
	lan = list(map(int, input().split()))
	lan.sort()
	max_d = 0
	for i in range(1, len(lan)):
		max_d = max(max_d, (lan[i] - lan[i - 1]) / 2)
	if lan[0] != 0:
		max_d = max(max_d, lan[0] - 0)
	if lan[-1] != len_r:
		max_d = max(max_d, len_r - lan[-1])
	print(max_d)
main()
