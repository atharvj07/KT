import bisect
from itertools import permutations
for _ in range(int(input())):
	(nr, ng, nb) = map(int, input().split())
	r = sorted(list(map(int, input().split())))
	g = sorted(list(map(int, input().split())))
	b = sorted(list(map(int, input().split())))
	all = [r, g, b]
	ans = float('inf')
	for a in permutations(all):
		for i in a[0]:
			(j, k) = (bisect.bisect(a[1], i) - 1, min(len(a[2]) - 1, bisect.bisect_left(a[2], i)))
			ans = min(ans, (i - a[1][j]) ** 2 + (i - a[2][k]) ** 2 + (a[1][j] - a[2][k]) ** 2)
	print(ans)
