#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int> > v;
  int cur = 0, max = 0, index = 0, s1 = 0, s2 = 0;
  for (int i = 0; i < n; i++) {
    int b, c;
    cin >> b >> c;
    v.push_back(make_pair(b, c));
    s1 += b;
    s2 += c;
  }
  max = abs(s1 - s2);
  for (int i = 0; i < n; i++) {
    cur = s1 - s2 - 2 * v[i].first + 2 * v[i].second;
    if (abs(cur) > max) {
      max = abs(cur);
      index = i + 1;
    }
  }
  cout << index << endl;
  return 0;
}
