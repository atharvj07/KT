import math
inp = int(input())
for _ in range(inp):
	n = input()
	n = int(n)
	lcm = 1
	mod = 10 ** 9 + 7
	count = n
	i = 1
	res = 0
	for i in range(1, 50):
		lcm = lcm * i // math.gcd(lcm, i)
		res = res % mod + (count - n // lcm) * i
		res %= mod
		count = n // lcm
		if count == 0:
			break
	print(res)
