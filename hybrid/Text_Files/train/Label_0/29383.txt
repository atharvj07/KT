#include <bits/stdc++.h>
using namespace std;

#define rep(i, n) for(int i = 0; i < int(n) ;++i)

int main(int argc, char *argv[])
{
  int n;
  while(cin >> n && n){
    vector<int> k(n);
    rep(i, n) cin >> k[i];
    string s;
    cin >> s;
    int idx = 0;
    for(auto c : s){
      int p = islower(c) ? c - 'a' : c - 'A' + 26;
      p -= k[idx];
      p %= 52;
      p += 52;
      p %= 52;
      cout << char(p < 26 ? p + 'a' : p-26 + 'A');
      idx = (idx + 1) % n;
    }
    cout << endl;
  }
  
  return 0;
}
