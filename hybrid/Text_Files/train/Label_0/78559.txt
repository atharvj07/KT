from collections import deque
import sys
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
sys.setrecursionlimit(200010)

def judgebuds(children, buds, seq):
	for index in seq:
		if not children[index]:
			continue
		for nextindex in children[index]:
			if not buds[nextindex]:
				buds[index] = True
				break

def main(t):
	n = int(input())
	neigh = [[] for i in range(n)]
	for i in range(n - 1):
		(u, v) = map(int, input().split())
		neigh[u - 1].append(v - 1)
		neigh[v - 1].append(u - 1)
	children = [[] for i in range(n)]
	queue = deque()
	queue.append(0)
	visited = [False] * n
	visited[0] = True
	seq = []
	while queue:
		index = queue.popleft()
		seq.append(index)
		for nextindex in neigh[index]:
			if visited[nextindex]:
				continue
			visited[nextindex] = True
			children[index].append(nextindex)
			queue.append(nextindex)
	seq = seq[::-1]
	buds = [False] * n
	judgebuds(children, buds, seq)
	ans = n
	for i in range(1, n):
		if buds[i]:
			ans -= 2
	if buds[0]:
		ans -= 1
	print(ans)
T = int(input())
t = 1
while t <= T:
	main(t)
	t += 1
