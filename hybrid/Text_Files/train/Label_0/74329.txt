string = input()
l = string.split('+')
(s1, s2) = (l[0], l[1])
l = s2.split('=')
(s3, s4) = (l[0], l[1])
k1 = len(s1) + len(s3)
k2 = len(s4)
if k1 < k2:
	if k2 - k1 == 2:
		s1 += '|'
		s4 = s4[:-1]
		print(s1 + '+' + s3 + '=' + s4)
	else:
		print('Impossible')
elif k2 < k1:
	if k1 - k2 == 2:
		s4 += '|'
		if len(s3) == 1:
			s1 = s1[:-1]
		else:
			s3 = s3[:-1]
		print(s1 + '+' + s3 + '=' + s4)
	else:
		print('Impossible')
else:
	print(s1 + '+' + s3 + '=' + s4)
