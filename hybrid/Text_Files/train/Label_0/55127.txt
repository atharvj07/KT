import java.util.Scanner;

public class C {
    static class Node {
        long v, p, d;
        boolean alive;
    }

    static int n, num;
    static Node[] a;
    static int[] ans;

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);

        n = cin.nextInt();
        a = new Node[n];
        for (int i = 0; i < n; i++) {
            a[i] = new Node();
            a[i].v = cin.nextInt();
            a[i].d = cin.nextInt();
            a[i].p = cin.nextInt();
            a[i].alive = true;
        }

        long nd, nv;
        num = 0;
        ans = new int[n];
        for (int now = 0; now < n; now++) {
            if (!a[now].alive) continue;

            ans[num++] = now + 1;
            nd = 0;
            nv = a[now].v;
            for (int i = now + 1; i < n; i++) {
                if (!a[i].alive) continue;
                a[i].p -= nd + nv;
                if (a[i].p < 0) {
                    a[i].alive = false;
                    nd += a[i].d;
                }
                if (nv != 0) nv--;
            }
        }

        System.out.println(num);
        for (int i = 0; i < num; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();

        cin.close();
    }
}
