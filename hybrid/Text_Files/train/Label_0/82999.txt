n = input()
s = input()

def jap_cross(s):
	res = []
	while True:
		start = s.find('B')
		if start == -1:
			return res
		else:
			s = s[start:]
			length = s.find('W')
			if length == -1:
				res.append(len(s))
				return res
			else:
				res.append(length)
				s = s[length:]
res = jap_cross(s)
print(len(res))
for x in res:
	print(x, end=' ')
