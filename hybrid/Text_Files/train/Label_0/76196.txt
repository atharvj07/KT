from bisect import bisect_right as br
for t in range(int(input())):
	s = input()
	t = input()
	alp = 'abcdefghijklmnopqrstuvwxyz'
	dic = {}
	for i in alp:
		dic[i] = []
	for i in range(len(s)):
		dic[s[i]].append(i)
	prev = -1
	ans = 1
	for i in t:
		idx = br(dic[i], prev)
		if len(dic[i]) == 0:
			ans = -1
			break
		if idx == len(dic[i]):
			ans += 1
			prev = dic[i][0]
		else:
			prev = dic[i][idx]
	print(ans)
