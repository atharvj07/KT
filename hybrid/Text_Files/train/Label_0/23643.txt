#include <bits/stdc++.h>
using namespace std;
int n;
char a[200010];
int main(void) {
  int i, j, ans = 0;
  scanf("%s", a + 1);
  n = strlen(a + 1);
  for (i = 1; i <= n; i++) a[i + n] = a[i];
  for (i = 1; i <= 2 * n; i = j) {
    for (j = i + 1; j <= 2 * n; j++) {
      if (a[j] == a[j - 1]) break;
      ans = max(ans, j - i);
    }
  }
  printf("%d\n", min(ans + 1, n));
}
