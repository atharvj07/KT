from collections import defaultdict as dd
n = int(input())
a = [int(i) for i in input().split()]
a.sort()
dic = dd(int)
ans = 0
for i in range(len(a)):
	if a[i] - 1 > 0 and a[i] - 1 not in dic:
		dic[a[i] - 1] += 1
		ans += 1
	elif a[i] not in dic:
		dic[a[i]] += 1
		ans += 1
	elif a[i] + 1 not in dic:
		dic[a[i] + 1] += 1
		ans += 1
print(ans)
