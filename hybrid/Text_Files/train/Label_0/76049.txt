class UnionFind:

	def __init__(self, n):
		self.n = n
		self.parents = [-1] * n

	def find(self, x):
		if self.parents[x] < 0:
			return x
		else:
			self.parents[x] = self.find(self.parents[x])
			return self.parents[x]

	def union(self, x, y):
		x = self.find(x)
		y = self.find(y)
		if x == y:
			return
		if self.parents[x] > self.parents[y]:
			(x, y) = (y, x)
		self.parents[x] += self.parents[y]
		self.parents[y] = x

	def size(self, x):
		return -self.parents[self.find(x)]

	def roots(self):
		return [i for (i, x) in enumerate(self.parents) if x < 0]
n = int(input())
p = list(map(int, input().split()))
uf = UnionFind(n)
for i in range(n):
	uf.union(i, p[i] - 1)
a = []
for i in uf.roots():
	a.append([uf.size(i), i])
a = sorted(a, reverse=True)
if len(a) >= 2:
	uf.union(a[0][1], a[1][1])
ans = 0
for i in uf.roots():
	ans += uf.size(i) ** 2
print(ans)
