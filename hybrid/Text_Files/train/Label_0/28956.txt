#include<iostream>
using namespace std;

int main()
{
	int W, H, C;
	int i;

	cin >> W >> H >> C;

	if (W < H)
	{
		for (i = H; i > 0; i--)
		{
			if (H % i == 0)
			{
				if(W % i == 0)
				{
					break;
				}
			}
		}
		cout << (H / i) * (W / i) * C << endl;
	}
	else if (W > H)
	{
		for (i = W; i > 0; i--)
		{
			if (W % i == 0)
			{
				if (H % i == 0)
				{
					break;
				}
			}
		}
		cout << (H / i) * (W / i) * C << endl;
	}
	else
	{
		cout << C << endl;
	}

	return 0;
}
