(a, b) = map(int, input().split())
x = []
y = []
z = []
for i in range(a):
	c = str(input())
	x.append(c)
	if 'S' in c:
		y.append(i)
		y.append(c.index('S'))
	if 'E' in c:
		z.append(i)
		z.append(c.index('E'))
d = str(input())
e = ['0123', '0132', '0213', '0231', '0312', '0321', '1023', '1032', '1203', '1230', '1302', '1320', '2013', '2031', '2103', '2130', '2301', '2310', '3012', '3021', '3102', '3120', '3201', '3210']
sum = 0
for i in range(len(e)):
	j = y[0]
	k = y[1]
	l = 0
	while (j != z[0] or k != z[1]) and l < len(d):
		if d[l] == e[i][0]:
			j += 1
		elif d[l] == e[i][1]:
			k -= 1
		elif d[l] == e[i][2]:
			j -= 1
		elif d[l] == e[i][3]:
			k += 1
		if j < a and k < b and (j >= 0) and (k >= 0):
			if x[j][k] == 'E':
				sum += 1
				break
			elif x[j][k] == '.':
				pass
			elif x[j][k] == '#':
				break
		else:
			break
		l += 1
print(sum)
