(n, m, k) = map(int, input().split())
m -= n
k -= 1
frodo = 1
left = right = k
while True:
	needed = right - left + 1
	if m >= needed:
		frodo += 1
		m -= needed
		left = max(0, left - 1)
		right = min(n - 1, right + 1)
		if left == 0 and right == n - 1:
			frodo += m // n
			break
	else:
		break
print(frodo)
