import sys
import itertools
input = sys.stdin.readline

def solve(N, K):
	if K > 1 << N - 1:
		return [-1]
	left = 1
	right = N + 1
	ret = []
	while right > left:
		r = right - left
		block = 1
		while 1 << max(0, r - 1 - block) < K:
			K -= 1 << max(0, r - 1 - block)
			block += 1
		ret.extend(range(left, left + block)[::-1])
		left += block
	return ret

def main():
	(T,) = map(int, input().split())
	for _ in range(T):
		(N, K) = map(int, input().split())
		print(' '.join(map(str, solve(N, K))))
main()
