import sys
mod = 10 ** 9 + 7
n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
i = 0
cnt = {}
for x in arr:
	lb = x & -x
	cnt[lb] = cnt.get(lb, 0) + 1
c2 = [cnt.get(1 << i, 0) for i in range(32)]
while c2[-1] == 0:
	c2.pop()
x = 1
y = n - c2[0]
for i in c2[1:]:
	y -= i
	if i > 0:
		x = (x + pow(2, i - 1, mod) * pow(2, y, mod)) % mod
print((pow(2, n, mod) - x) % mod)
