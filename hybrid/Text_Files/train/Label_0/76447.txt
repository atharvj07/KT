import sys
string = input()
dot = string.index('.')
a = string[0:dot]
d = string[dot + 1:string.index('e')]
b = int(string[string.index('e') + 1:len(string)])
if a == '0' and int(d) == 0 and (b == 0):
	res = 0
elif a != '0' and int(d) == 0 and (b == 0):
	res = a
elif len(d) > b and a != '0':
	res = a + d[0:b] + '.' + d[b:]
elif len(d) == b and a != '0':
	res = a + d
elif b > len(d) and a != '0':
	res = a + d + '0' * (b - len(d))
elif b == 0 and a == '0':
	res = string[0:string.index('e')]
elif b > 0 and a == '0':
	if b > len(d):
		res = d + '0' * abs(b - len(d))
	elif b < len(d):
		res = d[0:b] + '.' + d[b:]
	elif b == len(d):
		res = d
print(res)
