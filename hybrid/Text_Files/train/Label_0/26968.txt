#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, m, tc = -1, ans = 0;
    cin >> n >> m;
    map<long long, long long> mymap;
    vector<long long> o(m);
    for (int i = 0; i < n; i++) {
      long long no;
      cin >> no;
      mymap[no] = i + 1;
    }
    for (int i = 0; i < m; i++) cin >> o[i];
    for (int i = 0; i < m; i++) {
      if (mymap[o[i]] < tc)
        ans = ans + 1;
      else {
        ans = ans + 2 * (mymap[o[i]] - (i + 1)) + 1;
        tc = mymap[o[i]];
      }
    }
    cout << ans << endl;
  }
}
