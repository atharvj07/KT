#include <bits/stdc++.h>
using namespace std;
int check(long long int n, long long int k, long long int d1,
          long long int d2) {
  long long int A[4] = {1, -1, -1, 1}, B[4] = {1, -1, 1, -1}, x, a, b, c, D1,
                D2;
  x = n / 3;
  if (n % 3 != 0) return 0;
  for (int i = (0); i < (4); ++i) {
    D1 = d1 * A[i];
    D2 = d2 * B[i];
    if ((k + 2 * D1 + D2) % 3 == 0) {
      a = (k + 2 * D1 + D2) / 3;
      b = a - D1;
      c = b - D2;
      if (a <= x && b <= x && c <= x && a >= 0 && b >= 0 && c >= 0) return 1;
    }
  }
  return 0;
}
int main() {
  long long int t, n, k, d1, d2;
  scanf("%llu", &t);
  while (t--) {
    cin >> n >> k >> d1 >> d2;
    if (check(n, k, d1, d2))
      cout << "yes" << endl;
    else
      cout << "no" << endl;
  }
  return 0;
}
