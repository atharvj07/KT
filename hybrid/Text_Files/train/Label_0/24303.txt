#include <bits/stdc++.h>
using namespace std;
int main() {
  int hh, mm;
  cin >> hh >> mm;
  int h, d, c, n;
  cin >> h >> d >> c >> n;
  double ans;
  int k = h / n;
  if (h % n != 0) k++;
  if (hh >= 20) {
    ans = k * c * 0.8;
    printf("%.4lf\n", ans);
    return 0;
  }
  ans = k * c * 1.0;
  int km = (19 - hh) * 60 + 60 - mm;
  int k2 = (h + km * d) / n;
  if ((h + km * d) % n != 0) k2++;
  double ans2 = k2 * c * 0.8;
  ans = min(ans, ans2);
  printf("%.4lf\n", ans);
  return 0;
}
