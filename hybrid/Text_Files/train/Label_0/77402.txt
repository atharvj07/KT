(a, b) = map(int, input().split())
l = list(map(int, input().split()))
l = [0] + l + [b]
lo = [0] * len(l)
li = [0] * len(l)
for i in range(1, len(l)):
	if i % 2 == 0:
		li[i] = li[i - 1] + l[i] - l[i - 1]
		lo[i] = lo[i - 1]
	else:
		lo[i] = lo[i - 1] + l[i] - l[i - 1]
		li[i] = li[i - 1]
num = lo[len(l) - 1]
for i in range(1, len(l) - 1):
	x = lo[i]
	y = li[len(l) - 1]
	z = li[i + 1]
	num = max(num, x + (y - z) + (l[i + 1] - l[i] - 1))
print(num)
