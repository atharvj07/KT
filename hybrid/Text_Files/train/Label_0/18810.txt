#include <bits/stdc++.h>
using namespace std;
template <typename T = int>
T input() {
  T res;
  cin >> res;
  return res;
}
template <typename It>
void input_seq(It begin, It end, istream &in = cin) {
  generate(begin, end,
           input<typename remove_reference<decltype(*begin)>::type>);
}
template <class It>
void output(It begin, It end, ostream &out = cerr) {
  while (begin != end) {
    out << *(begin++) << ' ';
  }
  out << endl;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n = input();
  vector<int> a(n), b(n);
  input_seq(a.begin(), a.end()), input_seq(b.begin(), b.end());
  vector<pair<int, int> > to(n);
  for (int i = 0; i < n; ++i) {
    to[i] = {a[i] - 1, i};
  }
  sort(to.begin(), to.end());
  vector<int> nb(n);
  for (int i = 0; i < n; ++i) {
    nb[i] = to[b[i] - 1].second;
  }
  vector<int> stack;
  for (int i = 0; i < n; ++i) {
    while ((int)stack.size() && stack.back() > nb[i]) {
      stack.pop_back();
    }
    stack.emplace_back(nb[i]);
  }
  cout << n - (int)stack.size() << endl;
}
