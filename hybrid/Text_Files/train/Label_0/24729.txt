#include <bits/stdc++.h>
using namespace std;
int main() {
  int x, y;
  double way;
  cin >> x >> y;
  way = sqrt(x * x + y * y * 1.0);
  if (way - int(way) == 0 || x * y == 0)
    cout << "black";
  else {
    for (int i = 1; i <= 5000; i++)
      if (x * x + y * y <= i * i) {
        if (x * x + y * y == i * i)
          cout << "black";
        else {
          if (i % 2 == 1 && x * y > 0 || i % 2 == 0 && x * y < 0)
            cout << "black";
          else
            cout << "white";
        }
        break;
      }
  }
  return 0;
}
