from sys import stdin, stdout
MOD = 10 ** 9 + 7
p = 137

def hashh(a):
	cnt = [0]
	for i in range(1, len(a) + 1):
		cnt.append((cnt[-1] * p + a[i - 1]) % MOD)
	return cnt[len(a)]
(n, k, x) = map(int, stdin.readline().split())
values = sorted(list(map(int, stdin.readline().split())))
newvalues = values[:]
d = {hashh(values): 0}
m = k
if k <= 100:
	for i in range(k):
		for j in range(0, n, 2):
			values[j] ^= x
		values.sort()
	stdout.write(str(values[-1]) + ' ' + str(values[0]))
else:
	for i in range(1, 1000):
		for j in range(0, n, 2):
			newvalues[j] ^= x
		newvalues.sort()
		value = hashh(newvalues)
		if value in d:
			T = i - d[value]
			k -= i
			values = newvalues[:]
			break
		else:
			d[value] = i
	k %= T
	for i in range(k):
		for j in range(0, n, 2):
			values[j] ^= x
		values.sort()
	stdout.write(str(values[-1]) + ' ' + str(values[0]))
