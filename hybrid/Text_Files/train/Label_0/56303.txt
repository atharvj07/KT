
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigDecimal;
import java.math.BigInteger;
import java.text.DecimalFormat;
import java.util.*;
import java.text.NumberFormat;

import java.util.regex.Pattern;

public class Main {

    public static void main(String[] args) throws IOException {
        InputStreamReader r = new InputStreamReader(System.in);
        BufferedReader input = new BufferedReader(r);
        int t = Integer.parseInt(input.readLine());
        for (int k = 0; k < t; k++) {
            int n = Integer.parseInt(input.readLine());
            long[] a = new long[n];
            String[] num = input.readLine().split(" ");
            
            for (int i = 0; i < n; i++) {
                a[i] = Long.parseLong(num[i]);
            }
            long temp1 = +1;
            if (Math.abs(a[n - 1] - a[n - 2]) != 0) {
                temp1 = (a[n - 1] - a[n - 2]) / Math.abs(a[n - 1] - a[n - 2]);
            }
            for (int i = n - 2; i > 0; i--) {

                if ((temp1 >= 0 && a[i] - a[i - 1] <= 0) || (temp1 <= 0 && a[i] - a[i - 1] >= 0)) {
                    if (Math.abs(a[i] - a[i - 1]) != 0) {
                        temp1 = (a[i] - a[i - 1]) / Math.abs((a[i] - a[i - 1]));
                    } else {
                        temp1 = temp1 * (-1);
                    }
                    
                    continue;
                }

                long temp = Math.max(Math.abs(a[i]), Math.abs(a[i - 1]));
                
                if (Math.abs(a[i]) == Math.abs(a[i - 1])) {
                    a[i - 1] *= (-1);
                } else if (temp == Math.abs(a[i])) {
                   
                    for (int j = i+1; j < n; j++) {
                        a[j] = a[j] * (-1);
                    }
                } else if (temp == Math.abs(a[i - 1])) {
                    a[i - 1] = a[i - 1] * -1;

                }
                if (Math.abs(a[i] - a[i - 1]) != 0) {
                    temp1 = (a[i] - a[i - 1]) / Math.abs((a[i] - a[i - 1]));
                } else {
                    temp1 = temp1 * (-1);
                }
            }
            for (int i = 0; i < n; i++) {
                System.out.print(a[i]+" ");
            }
            System.out.println();
        }

    }
}
