#include <bits/stdc++.h>
#include <algorithm>
using namespace std;

int main(){
 long long  int n,a[200010];
  cin >> n ;
  
  for(int i=0;i<n;i++)
    cin >> a[i];
  
 map<int, int> b;
	for (int i = 0; i < n; i++)
		b[a[i]]++;
 
	int ans = 0;
	for (auto p: b) {
		int x = p.first;
		int n = p.second;
		if (n < x) ans += n;
		else ans += n - x;
	}
  
  cout << ans << endl;
}
