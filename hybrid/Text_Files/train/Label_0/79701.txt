T = int(input())
for i in range(T):
	arr = list(map(int, input().split()))
	s = arr[0]
	i = arr[1]
	e = arr[2]
	temp = e
	maxstr = s + e
	count = 0
	if maxstr > i:
		rem = maxstr - i
		if rem % 2 == 0:
			count += rem // 2
		else:
			count += (rem + 1) // 2
	if count > e + 1:
		count = e + 1
	print(count)
