import java.util.*;
import java.io.*;

public class Solution{
	long rem = 998244353L;
	
	public void solve(int test, Scanner sc){
		int n = sc.nextInt();
		
		long blockWays = 10;
		long[] res = new long[n+1];
		
		for(int i=1; i<=n; i++){
			long numPos = n-i+1L;
		
			long midPos = Math.max(numPos-2L, 0L);
			long cornerPos = Math.min(2L, numPos);
			
			long mid = 81L*power(10L, Math.max(n-i-2L, 0L));
			mid %= rem;
			long corner = 9L*power(10L, Math.max(n-i-1L, 0L));
			if((n-i) == 0){
				corner = 1L;
			}
			corner %= rem;
			
			res[i] = ((((midPos*mid)%rem)*blockWays)%rem);
			res[i] += ((((cornerPos*corner)%rem)*blockWays)%rem);
			res[i] %= rem;
		}
		
		for(int i=1; i<n; i++){
			System.out.print(res[i] + " ");
		}
		System.out.println(res[n]);
	}
	
	public long power(long a, long b){
		if(b == 0L){
			return 1L;
		}
		if(b%2L == 0L){
			return (power(((a*a)%rem), b/2L));
		}
		else{
			return ((a*power(((a*a)%rem), b/2L))%rem);
		}
	}
	
	public Solution(){
		Scanner sc = new Scanner(System.in);
		int tests = 1;
        
        for(int t=1; t<=tests; t++){
			solve(t, sc);
        }
	}
	
	public static void main(String[] args){
		new Solution();
	}
}
