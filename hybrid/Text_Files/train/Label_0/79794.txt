n = int(input())
a = [[0, 0]] * n
b = [[0, 0]] * n
for i in range(n):
	(x, y) = input().split()
	a[i] = [int(x), 0]
	b[i] = [int(y), 0]
j = 0
k = 0
for i in range(n):
	if a[j][0] < b[k][0]:
		a[j][1] = 1
		j += 1
	else:
		b[k][1] = 1
		k += 1
k = n // 2
for i in range(k):
	a[i][1] = 1
	b[i][1] = 1
s1 = ''
s2 = ''
for i in range(n):
	s1 += str(a[i][1])
	s2 += str(b[i][1])
print(s1)
print(s2)
