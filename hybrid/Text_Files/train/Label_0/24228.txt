#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.0);
const double EPS = 1e-6;
void task();
int main() {
  cin.tie(0);
  ios::sync_with_stdio(false);
  srand(time(0));
  task();
  return 0;
}
struct Team {
  int index;
  int points;
  int delta;
  Team(int ind, int p, int d) : index(ind), points(p), delta(d) {}
  friend bool operator<(const Team& t1, const Team& t2) {
    return t1.points < t2.points ||
           (t1.points == t2.points && t1.index > t2.index);
  }
  friend bool operator>(const Team& t1, const Team& t2) { return !(t1 < t2); }
};
void task() {
  int n;
  cin >> n;
  set<Team> teams;
  for (int i = 0; i < n; ++i) {
    int a, d;
    cin >> a >> d;
    teams.emplace(i, a, d);
  }
  int ans = 0;
  while (true) {
    bool ok = false;
    for (auto it = teams.begin(); it != teams.end(); ++it) {
      if (it->delta > 0) {
        auto new_team = Team(it->index, it->points + it->delta, 0);
        for (auto it_in = it; it_in != teams.end(); ++it_in, ++ans) {
          if (*it_in > new_team) break;
        }
        --ans;
        teams.emplace(new_team);
        teams.erase(Team(it->index, it->points, it->delta));
        ok = true;
        break;
      }
    }
    if (!ok) break;
  }
  while (true) {
    bool ok = false;
    for (auto it = teams.rbegin(); it != teams.rend(); ++it) {
      auto new_team = Team(it->index, it->points + it->delta, 0);
      if (it->delta < 0) {
        for (auto it_in = it; it_in != teams.rend(); ++it_in, ++ans) {
          if (*it_in < new_team) break;
        }
        --ans;
        teams.emplace(new_team);
        teams.erase(Team(it->index, it->points, it->delta));
        ok = true;
        break;
      }
    }
    if (!ok) break;
  }
  cout << ans;
}
