#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<int> a;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a.push_back(x);
  }
  for (int i = 0; i < a.size() - 2; i++) {
    for (int j = i + 1; j < a.size() - 1; j++) {
      for (int k = j + 1; k < a.size(); k++) {
        if (a[i] + a[j] == a[k]) {
          cout << k + 1 << " " << j + 1 << " " << i + 1;
          return 0;
        } else if (a[i] - a[j] == a[k]) {
          cout << i + 1 << " " << j + 1 << " " << k + 1;
          return 0;
        }
        if (a[j] - a[i] == a[k]) {
          cout << j + 1 << " " << i + 1 << " " << k + 1;
          return 0;
        }
      }
    }
  }
  cout << -1;
  return 0;
}
