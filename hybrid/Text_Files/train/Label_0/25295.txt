#include <string>
#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
	int n, t;
	cin >> n >> t;
	vector<int> a(n);
	for(int i = 0; i < n; i++){
		cin >> a[i];
	}
	
	int minV = 1000000000;
	int maxSa = -1;
	int ans = 0;
	for(int i = 0; i < n; i++){
		if(minV > a[i]){
			minV = a[i];
		}else{
			if(a[i] - minV > maxSa){
				maxSa = a[i] - minV;
				ans = 1;
			}else if(a[i] - minV == maxSa){
				ans++;
			}
		}
	}
	
	cout << ans << endl;
	
	return 0;
}
