#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <set>
#include <cstring>

using namespace std;

bool used[2000];
bool check(string s){
    for(int i = 0; i < s.size()/2; i++){
        if(s[i]==s[s.size()-1-i])
            continue;
        else
            return false;
    }
    return true;
}

int main(){

    string str;
    while(getline(cin,str)){

        memset(used,0,sizeof(used));
        string s;
        for(int i = 0; i < str.size(); i++){
            if((str[i]>='a'&&str[i]<='z')||(str[i]>='A'&&str[i]<='Z')){
                s+=toupper(str[i]);
            }
        }
        set<string> ss;
        for(int i = s.size(); i >= 3; i--){
            for(int j = 0; j + i -1 < s.size(); j++){
                if(check(s.substr(j,i))){
                    ss.insert(s.substr(j,i));
                }
            }
        }
        vector<string> tmps=vector<string>(ss.begin(),ss.end());
        for(int i = 0; i < tmps.size(); i++){
            for(int j = i+1; j < tmps.size(); j++){
                if(tmps[i].size()<tmps[j].size()){
                    int diff=tmps[j].size()-tmps[i].size();
                    diff/=2;
                    if(tmps[j].substr(diff,tmps[i].size())==tmps[i]){
                        used[i]=true;
                    }
                }
                else{
                    int diff=tmps[i].size()-tmps[j].size();
                    diff/=2;
                    if(tmps[i].substr(diff,tmps[j].size())==tmps[j]){
                        used[j]=true;
                    }
                }
            }
        }
        ss.clear();
        for(int i = 0; i < tmps.size(); i++){
            if(!used[i])
                ss.insert(tmps[i]);
        }
        for(set<string>::iterator it=ss.begin();it!=ss.end();it++){
            cout<<*it;
            set<string>::iterator iit=it;
            iit++;
            if(iit==ss.end()){
                cout<<endl;
            }
            else
                cout<<" ";
        }
        if(ss.size()==0)cout<<endl;
    }

    return 0;
}
