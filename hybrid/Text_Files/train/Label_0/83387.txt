(n, k) = map(int, input().split())
s = input()

def solve(x, y):
	(res, j, t) = (0, 0, k)
	for i in range(n):
		if j < i:
			j = i
			t = k
		while j < n and (s[j] == x or (s[j] == y and t > 0)):
			t -= int(s[j] == y)
			j += 1
		res = max(res, j - i)
		t += int(s[i] == y)
	return res
print(max(solve('a', 'b'), solve('b', 'a')))
