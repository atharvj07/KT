import java.util.*;
import java.io.*;
public class q2
{
  static int mat[][]=new int[27][27];
  static int visited[]=new int[27];
  static int st=0;
  static int end=0;
  static int fff=0;
  public static void main(String args[])
  {
    Scanner in=new Scanner(System.in);
    PrintWriter pw=new PrintWriter(System.out, true);
    int n=in.nextInt();
    String s1=in.next();
    String s2=in.next();
    int c=0;
    String ans[]=new String[1000001];
    for(int i=0;i<s1.length();i++)
    {
      char c1=s1.charAt(i);
      char c2=s2.charAt(i);
      st=(int)c1-96;
      end=(int)c2-96;
      Arrays.fill(visited,0);
      fff=0;
      if(!dfs(st,end))
      {
        mat[st][end]=1;
        mat[end][st]=1;
        ans[c]=""+c1+c2;
        c++;
      }
    }
    pw.println(c);
    for (int i=0;i<c;i++)
    {
      pw.println(ans[i].charAt(0)+" "+ans[i].charAt(1));
   }
    pw.close();
  }
  public static boolean dfs(int st, int end)
	{
    visited[st]=1;
    if(st==end)
    fff=1;
    for(int j=0;j<27;j++)
    {
      if(visited[j]==0 && mat[st][j]==1)
      dfs(j,end);
    }
    if(fff==0)
    return false;
    return true;
  }
}
