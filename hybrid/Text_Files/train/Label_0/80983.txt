import sys

def main():
	n = int(sys.stdin.readline())
	a = list(map(int, sys.stdin.readline().split()))
	last_pair = -1
	res = 0
	lhs = (0, a[0])
	for (key, value) in enumerate(a):
		if value != lhs[1]:
			if lhs[1] != last_pair and value != last_pair:
				last_pair = -1
			if key - lhs[0] > 1:
				res += 1
				if lhs[1] == last_pair:
					res -= 1
				last_pair = lhs[1]
			res += 1
			lhs = (key, value)
	res += 1
	if n - lhs[0] > 1:
		res += 1
		if lhs[1] == last_pair:
			res -= 1
	print(res)
	return
main()
