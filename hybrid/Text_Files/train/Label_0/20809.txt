#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll n,a;
ll sum=0;
vector<ll> t;

ll calc(ll a){
  return a*(a-1LL)/2LL;
}

ll solve5(){
  if(t[0]>=4LL){
    return t[4]+t[3]+t[2]+t[1]+t[0]-10LL;
  }
  if(t[1]>=4LL){
    return t[4]+t[3]+t[2]+t[1]+t[0]-9LL;
  }
  if(t[2]>=4LL){
    return t[4]+t[3]+t[2]+t[1]+t[0]-9LL;
  }
  if(t[3]>=4LL){
    return t[4]+t[3]+t[2]+t[1]+t[0]-9LL;
  }
  if(t[4]>=4LL){
    return t[4]+t[3]+t[2]+t[1]+t[0]-8LL;
  }
  return 7LL;
}

ll solve(){
  if(t.size()==0)return 1LL;
  if(t.size()==1)return t[0];
  if(t.size()==2)return t[0]+t[1]-1LL;
  if(t.size()==3)return t[0]+t[1]+t[2]-3LL;
  if(t.size()==4)return t[0]+t[1]+t[2]+t[3]-6LL;
  return solve5();
}

int main(){
  cin>>n;
  for(int i=0;i<n;i++){
    cin>>a;
    if(a>2LL)t.push_back(a);
    sum+=calc(a);
  }
  sort(t.begin(),t.end());
  ll ans=solve();
  while(calc(ans)<sum)ans++;
  cout<<ans<<endl;
  return 0;
}
