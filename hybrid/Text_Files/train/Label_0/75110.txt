from sys import stdin, stdout
from math import sqrt
import math
import heapq
from itertools import accumulate

def near_sq(num):
	return round(sqrt(num)) ** 2
N = int(input())
arr = [int(x) for x in stdin.readline().split()]
sqr = 0
non = 0
sqr_mini = []
non_mini = []
for num in arr:
	p = near_sq(num)
	if p - num == 0:
		sqr += 1
		if num == 0:
			sqr_mini.append(2)
		else:
			sqr_mini.append(1)
	else:
		non += 1
		non_mini.append(abs(p - num))
if sqr == non:
	print(0)
	quit()
res = 0
half = N // 2
if sqr > non:
	required = sqr - half
	sqr_mini.sort()
	res = sum(sqr_mini[:required])
elif non > sqr:
	required = non - half
	non_mini.sort()
	res = sum(non_mini[:required])
print(res)
