#include <bits/stdc++.h>
using namespace std;
pair<pair<int, int>, pair<int, int> > ed[100010];
int n, m, ct = 0;
int p[100010];
int vis[100010];
int ans[100010];
int low[100010], tim[100010];
vector<int> vec;
vector<pair<int, int> > lis[100010];
int findset(int a) { return (a == p[a]) ? a : (p[a] = findset(p[a])); }
void merge(int a, int b) { p[findset(a)] = findset(b); }
void dfs(int cur, int le) {
  low[cur] = ct;
  tim[cur] = ct++;
  int i;
  for (i = 0; i < (int)lis[cur].size(); i++) {
    if (lis[cur][i].second == le) continue;
    if (tim[lis[cur][i].first] != -1)
      low[cur] = min(low[cur], tim[lis[cur][i].first]);
    else {
      dfs(lis[cur][i].first, lis[cur][i].second);
      low[cur] = min(low[cur], low[lis[cur][i].first]);
    }
  }
}
int main() {
  scanf("%d%d", &n, &m);
  int i;
  for (i = 1; i <= n; i++) p[i] = i;
  for (i = 0; i < m; i++) {
    int x, y, z;
    scanf("%d%d%d", &x, &y, &z);
    ed[i].first.first = z;
    ed[i].first.second = i;
    ed[i].second.first = x;
    ed[i].second.second = y;
  }
  sort(ed, ed + m);
  int cur = 0;
  for (int ii = 0; ii <= m; ii++) {
    if (ii == m || ed[ii].first.first != ed[cur].first.first) {
      for (i = cur; i < ii; i++) {
        int u = ed[i].second.first;
        int v = ed[i].second.second;
        int pl = ed[i].first.second;
        if (findset(u) != findset(v)) {
          int nu = findset(u);
          int nv = findset(v);
          if (vis[nu] == 0) {
            vis[nu] = 1;
            lis[nu].clear();
            low[nu] = tim[nu] = -1;
            vec.push_back(nu);
          }
          if (vis[nv] == 0) {
            vis[nv] = 1;
            lis[nv].clear();
            low[nv] = tim[nv] = -1;
            vec.push_back(nv);
          }
          lis[nu].push_back(make_pair(nv, i));
          lis[nv].push_back(make_pair(nu, i));
        } else
          ans[pl] = 3;
      }
      ct = 0;
      for (i = 0; i < (int)vec.size(); i++)
        if (tim[vec[i]] == -1) dfs(vec[i], -1);
      for (i = cur; i < ii; i++) {
        int u = ed[i].second.first;
        int v = ed[i].second.second;
        int pl = ed[i].first.second;
        if (findset(u) != findset(v)) {
          int nv = findset(v), nu = findset(u);
          if (tim[nv] > tim[nu]) {
            if (low[nv] > tim[nu])
              ans[pl] = 1;
            else
              ans[pl] = 2;
          } else {
            if (low[nu] > tim[nv])
              ans[pl] = 1;
            else
              ans[pl] = 2;
          }
        }
        vis[findset(u)] = 0;
        vis[findset(v)] = 0;
      }
      for (i = cur; i < ii; i++) {
        int u = ed[i].second.first;
        int v = ed[i].second.second;
        if (findset(u) != findset(v)) merge(u, v);
      }
      vec.clear();
      cur = ii;
    }
  }
  for (i = 0; i < m; i++) {
    if (ans[i] == 1)
      printf("any\n");
    else if (ans[i] == 2)
      printf("at least one\n");
    else if (ans[i] == 3)
      printf("none\n");
  }
  return 0;
}
