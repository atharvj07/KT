#include <bits/stdc++.h>
double dis(double a, double b, double c) { return sqrt(a * a + b * b + c * c); }
double x[105], y[105], z[105];
int main() {
  int n;
  scanf("%d", &n);
  double tx = 0, ty = 0, tz = 0, eps = 1.0;
  for (int i = 0; i < n; i++) {
    scanf("%lf %lf %lf", &x[i], &y[i], &z[i]);
    tx += x[i];
    ty += y[i];
    tz += z[i];
  }
  tx /= n;
  ty /= n;
  tz /= n;
  while (eps > 1e-8) {
    double d = dis(tx - x[0], ty - y[0], tz - z[0]);
    int p = 0;
    for (int i = 1; i < n; i++) {
      double td = dis(tx - x[i], ty - y[i], tz - z[i]);
      if (td > d) d = td, p = i;
    }
    tx += (x[p] - tx) * eps;
    ty += (y[p] - ty) * eps;
    tz += (z[p] - tz) * eps;
    eps *= 0.99993;
  }
  printf("%.9f %.9f %.9f\n", tx, ty, tz);
}
