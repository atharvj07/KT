for t in range(int(input())):
	a = []
	n = int(input())
	for i in range(n):
		(x, y) = map(int, input().split())
		a.append([x, y])
	w = ''
	x = 0
	y = 0
	a.sort()
	i = 0
	while i < n:
		if x < a[i][0]:
			w = w + 'R'
			x += 1
		elif y < a[i][1]:
			w = w + 'U'
			y += 1
		elif x == a[i][0] and y == a[i][1]:
			i += 1
		else:
			break
	if i == n:
		print('YES')
		print(w)
	else:
		print('NO')
