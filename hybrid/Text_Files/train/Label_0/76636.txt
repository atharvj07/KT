from sys import stdin
input = stdin.readline
(n, m) = map(int, input().split())
s = [input() for i in range(n)]
q = int(input())
pre = [0] * m
for i in range(1, n):
	for j in range(1, m):
		if s[i - 1][j] == 'X' and s[i][j - 1] == 'X':
			pre[j] = 1
for i in range(1, m):
	pre[i] += pre[i - 1]
for i in range(q):
	(l, r) = map(int, input().split())
	if pre[r - 1] > pre[l - 1]:
		print('NO')
	else:
		print('YES')
