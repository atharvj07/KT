t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	a.sort()
	alice = 0
	bob = 0
	turn = True
	while len(a) > 0:
		if turn:
			num = a.pop()
			if num % 2 == 0:
				alice += num
			turn = False
		else:
			num = a.pop()
			if num % 2 == 1:
				bob += num
			turn = True
	if alice > bob:
		print('Alice')
	elif bob > alice:
		print('Bob')
	else:
		print('Tie')
