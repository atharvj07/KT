import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	N = n
	a = list(map(int, input().split()))
	a = [a[i] - 1 for i in range(n)]
	E = []
	cnt = [0 for i in range(n)]
	L = 0
	for i in range(1, n):
		if a[i] == a[i - 1]:
			cnt[a[i - 1]] += 1
			cnt[a[L]] += 1
			E.append((L, i - 1))
			L = i
	cnt[a[n - 1]] += 1
	cnt[a[L]] += 1
	E.append((L, n - 1))
	idx = -1
	M = max(cnt)
	for i in range(n):
		if cnt[i] == M:
			idx = i
			break
	k = len(E)
	cut = 0
	for (L, R) in E:
		for i in range(L, R):
			if a[i] != idx and a[i + 1] != idx:
				cut += 1
	ans = -1
	if k + 1 >= M:
		ans = k - 1
	else:
		need = M - (k + 1)
		if need <= cut:
			ans = M - 2
	print(ans)
