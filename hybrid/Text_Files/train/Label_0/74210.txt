import math

def factorize(n):
	count = 0
	while (n % 2 > 0) == False:
		n >>= 1
		count += 1
	ans = []
	if count > 0:
		ans.append(2)
	for i in range(3, int(math.sqrt(n)) + 1):
		count = 0
		while n % i == 0:
			count += 1
			n = int(n / i)
		if count > 0:
			ans.append(i)
		i += 2
	if n > 2:
		ans.append(n)
	return ans
n = int(input())
divs = factorize(n)
if len(divs) == 0:
	print(n)
elif len(divs) == 1:
	print(divs[0])
else:
	print('1')
