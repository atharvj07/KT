#include <bits/stdc++.h>
using namespace std;
string s1, s2;
vector<int> v1, v2;
bool ans = false;
int main() {
  cin >> s1 >> s2;
  if (s1[0] == '1')
    v1.push_back(1);
  else
    v1.push_back(0);
  for (int(i) = int(1); i < int(s1.size()); i++)
    if (s1[i] == '1')
      v1.push_back(v1[v1.size() - 1] + 1);
    else
      v1.push_back(v1[v1.size() - 1]);
  if (s2[0] == '1')
    v2.push_back(1);
  else
    v2.push_back(0);
  for (int(i) = int(1); i < int(s2.size() + 1); i++)
    if (s2[i] == '1')
      v2.push_back(v2[v2.size() - 1] + 1);
    else
      v2.push_back(v2[v2.size() - 1]);
  for (int(i) = int(1); i < int(s1.size()); i++) {
    bool q = true;
    for (int(j) = int(i); j < int(s1.size()); j++)
      if (s1[j] != s2[j - i]) q = false;
    if (q) {
      if (v1[v1.size() - 1] % 2 == 1 &&
          v1[i - 1] + 1 >= v2[v2.size() - 1] - v2[s1.size() - i - 1])
        ans = true;
      if (v1[v1.size() - 1] % 2 == 0 &&
          v1[i - 1] >= v2[v2.size() - 1] - v2[s1.size() - i - 1])
        ans = true;
    }
  }
  if (v1[v1.size() - 1] % 2 == 1 && v1[v1.size() - 1] + 1 >= v2[v2.size() - 1])
    ans = true;
  if (v1[v1.size() - 1] % 2 == 0 && v1[v1.size() - 1] >= v2[v2.size() - 1])
    ans = true;
  if (ans || s1 == s2)
    cout << "YES" << '\n';
  else
    cout << "NO" << '\n';
  return 0;
}
