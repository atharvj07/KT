#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  vector<string> v;
  int n = s.length();
  if (s[n - 1] == '.') {
    cout << "NO";
    return 0;
  }
  for (int i = 0; i < n; i++) {
    string t = "";
    int x = 0;
    while (i < n) {
      if (x == 12) {
        cout << "NO";
        return 0;
      }
      if (s[i] == '.') break;
      t = t + s[i];
      i++;
      x++;
    }
    v.push_back(t);
  }
  if (v.size() == 1) {
    cout << "NO";
    return 0;
  }
  vector<string> ans;
  for (int i = 1; i < v.size(); i++) {
    string t = "";
    if (i == 1) {
      if (v[0].length() > 8 || v[0].length() == 0) {
        cout << "NO";
        return 0;
      } else {
        t = v[0] + ".";
      }
    } else if (v[i - 1].length() < 2) {
      cout << "NO";
      return 0;
    } else if (v[i - 1].length() < 10) {
      int j = 1;
      while (j < v[i - 1].length()) {
        t = t + v[i - 1][j];
        j++;
      }
      t = t + ".";
    } else if (v[i - 1].length() == 10) {
      int j = 2;
      while (j < v[i - 1].length()) {
        t = t + v[i - 1][j];
        j++;
      }
      t = t + ".";
    } else if (v[i - 1].length() == 11) {
      int j = 3;
      while (j < v[i - 1].length()) {
        t = t + v[i - 1][j];
        j++;
      }
      t = t + ".";
    } else {
      cout << "NO";
      return 0;
    }
    if (i == v.size() - 1) {
      if (v[v.size() - 1].length() > 3 || v[v.size() - 1].length() == 0) {
        cout << "NO";
        return 0;
      } else {
        t = t + v[v.size() - 1];
      }
    } else if (v[i].length() < 2) {
      cout << "NO";
      return 0;
    } else if (v[i].length() < 10) {
      t = t + v[i][0];
    } else if (v[i].length() == 10) {
      t = t + v[i][0] + v[i][1];
    } else if (v[i].length() == 11) {
      t = t + v[i][0] + v[i][1] + v[i][2];
    } else {
      cout << "NO";
      return 0;
    }
    ans.push_back(t);
  }
  cout << "YES" << endl;
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i] << endl;
  }
  return 0;
}
