#include "bits/stdc++.h"
using namespace std;
typedef long long ll;

ll goukei[100];//n桁までの57577の個数
ll kakuketa[100];//最上位を固定したときの個数

string fixit(int keta,int n1,ll n)
{
	string ret="";
	int n2=-1;
	//cerr<<kakuketa[keta]<<endl;
	//今見ている最上位から2番目の数を確定する
	for(int i=0;i<10;i++)
	{
		//cerr<<i<<" "<<n<<endl;
		if(i==n1)
		{
			if(n<kakuketa[keta])
			{
				n2=n1;
				break;
			}else{
				n-=kakuketa[keta];
			}
		}else{
			if(n<(1LL<<(keta-1LL)))
			{
				n2=i;
				break;
			}else
				n-=(1LL<<(keta-1LL));
		}
	}
	//cerr<<n2<<endl;
	keta--;
	//3桁目以降
	if(n1!=n2)
	{
		ret+=(char)(n2+'0');
		//cerr<<keta<<endl;
		for(ll i=keta;i>0;i--)
		{
			ll tmp=(n&(1LL<<(i-1)));
			if(tmp>0LL)
				ret+=(char)(max(n1,n2)+'0');
			else
				ret+=(char)(min(n1,n2)+'0');
		}
		return ret;
	}else{
		ret+=(char)(n1+'0');
		return ret+fixit(keta,n2,n);
	}
}

int main()
{
	goukei[0]=0;
	goukei[1]=0;
	kakuketa[0]=0;
	kakuketa[1]=0;
	for(int i=2;i<100;i++)
	{
		kakuketa[i]=9LL*((1LL<<((ll)i-1LL))-1LL);
		goukei[i]=goukei[i-1]+kakuketa[i]*9LL;
	}

	while(1)
	{
		long long nin;
		cin>>nin;
		ll n=nin;
		if(n==0)
			break;
		//nが何桁の数か
		n--;
		int nketa=0;
		for(int i=0;i<100;i++)
		{
			if(n<goukei[i])
			{
				nketa=i;
				n-=goukei[i-1];//余計な桁の数を引く
				break;
			}
		}
		int n1=n/kakuketa[nketa]+1;//nの最上位の数を確定する
		n%=kakuketa[nketa];
		if(n<0)
			n+=kakuketa[nketa];
		//cout<<nketa<<" "<<n1<<" "<<n<<endl;//for debug
		string ans="";
		ans+=(char)(n1+'0');
		ans+=fixit(nketa-1,n1,n);
		cout<<ans<<endl;
	}
	return 0;
}
