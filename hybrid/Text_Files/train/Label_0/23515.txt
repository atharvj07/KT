#include <bits/stdc++.h>
using namespace std;
int sc1() {
  int x;
  scanf("%d", &x);
  return x;
}
long long sc2() {
  long long x;
  scanf("%lld", &x);
  return x;
}
int main() {
  int n = sc1();
  int neg = 0, pos = 0, ze = 0;
  for (int i = 0; i < n; i++) {
    int x = sc1();
    if (x == 0)
      ze++;
    else if (x > 0)
      pos++;
    else
      neg++;
  }
  int op = ceil((double)n / 2.0);
  if (pos >= op)
    cout << 4 << endl;
  else if (neg >= op)
    cout << -4 << endl;
  else
    cout << 0 << endl;
  return 0;
}
