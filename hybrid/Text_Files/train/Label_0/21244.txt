#include <bits/stdc++.h>
using namespace std;
int main(int argc, char** argv) {
  int i, vp, vd, t, f, c;
  double sum, s, t1;
  int ans;
  while (scanf("%d%d%d%d%d", &vp, &vd, &t, &f, &c) != EOF) {
    if (vp >= vd) {
      printf("0\n");
      continue;
    }
    ans = 0;
    sum = 0;
    sum = vp * t;
    while (1) {
      t1 = (sum) / (vd - vp);
      s = vp * t1;
      if (sum + s < c) {
        ans++;
        sum += s;
        sum += (vp * (f + t1));
        if (sum >= c) {
          printf("%d\n", ans);
          break;
        }
      } else {
        printf("%d\n", ans);
        break;
      }
    }
  }
  return 0;
}
