import java.util.Arrays;
import java.util.Scanner;


public class d2_121_d {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int k = sc.nextInt();
		long b = sc.nextLong();
		int[] a = new int[n-1];
		int[] sa = new int[n-1];
		for (int i=0; i<n-1; i++) {
			a[i] = sc.nextInt();
			sa[i] = a[i];
		}
		Arrays.sort(sa);
		long cnt = 0;
		for (int i=0; i<k; i++) {
			cnt += sa[n-2-i];
		}
		if (cnt <= b) {
			System.out.println(n);
		} else {
			cnt -= sa[n-k-1];
			long left = b - cnt;
			for (int i=0; i<n-1; i++) {
				if (a[i] > left) {
					System.out.println(i+1);
					break;
				}
			}
		}
	}

}
