#include <iostream>
#include <vector>
using namespace std;
const int INF = 1001001001;
const int MAX = 200005;

int a[MAX], ans[MAX];
vector<vector<int>> g;

void dfs(int v, int parent, vector<int> &dp) {
	auto itr = lower_bound(dp.begin(), dp.end(), a[v]);
	int tmp = *itr;
	*itr = a[v];
	ans[v] = lower_bound(dp.begin(), dp.end(), INF) - dp.begin();
	for (int nv : g[v]) {
		if (nv == parent) continue;
		dfs(nv, v, dp);
	}
	*itr = tmp;
}

int main() {
	int n;
	cin >> n;
	g.resize(n);
	for (int i = 0; i < n; i++) {
		scanf("%d", a+i);
	}
	for (int i = 0; i < n-1; i++) {
		int u, v;
		scanf("%d%d", &u, &v);
		u--, v--;
		g[u].push_back(v);
		g[v].push_back(u);
	}
	vector<int> dp(n, INF);
	dfs(0, -1, dp);
	for (int i = 0; i < n; i++) {
		cout << ans[i] << endl;
	}
	return 0;
}
