#include <bits/stdc++.h>
using namespace std;
const long long int mod = 1e9 + 7;
const long long int MAXN = 100200;
long long int i, j, mask;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long int t;
  cin >> t;
  while (t--) {
    long long int p, f, a, b, s, w;
    cin >> p >> f >> a >> b >> s >> w;
    if (s > w) {
      swap(s, w);
      swap(a, b);
    }
    long long int ans = 0;
    for (i = 0; i < min(p / s, a) + 1; i++) {
      long long int x = p, y = f, z = a, g = b, cnt = i;
      x -= i * s;
      z -= i;
      cnt += min(x / w, b);
      g -= min(x / w, b);
      long long int k = min(f / s, z);
      cnt += k;
      z -= k;
      y -= k * s;
      cnt += min(y / w, g);
      ans = max(ans, cnt);
    }
    cout << ans;
    cout << "\n";
  }
  return 0;
}
