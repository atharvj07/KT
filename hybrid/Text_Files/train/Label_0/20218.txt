#include <bits/stdc++.h>
using namespace std;
long long int m = 998244353;
long long int fact(long long int n) {
  long long int fac = 1;
  for (long long int i = 1; i <= n; i++) {
    fac = ((fac % m) * (i % m)) % m;
  }
  return fac;
}
bool compare(pair<long long int, long long int> a,
             pair<long long int, long long int> b) {
  if (a.first != b.first)
    return a.first < b.first;
  else
    return a.second < b.second;
}
int main() {
  long long int n;
  cin >> n;
  long long int a[n + 1];
  long long int b[n + 1];
  pair<long long int, long long int> pairs[n + 1];
  map<pair<long long int, long long int>, long long int> m3;
  map<long long int, long long int> m1;
  map<long long int, long long int> m2;
  for (long long int i = 1; i <= n; i++) {
    cin >> a[i] >> b[i];
    m1[a[i]]++;
    m2[b[i]]++;
    m3[(make_pair(a[i], b[i]))]++;
    pairs[i] = make_pair(a[i], b[i]);
  }
  long long int first = 1;
  long long int second = 1;
  vector<long long int> cs;
  sort(pairs + 1, pairs + n + 1, compare);
  long long int f = 1;
  for (long long int i = 1; i <= n - 1; i++) {
    if (pairs[i + 1].second < pairs[i].second) {
      f = 0;
      break;
    }
  }
  long long int both = 1;
  if (!f) both = 0;
  for (auto x : m1) {
    first = (first * fact(x.second)) % m;
  }
  for (auto x : m2) {
    second = (second * fact(x.second)) % m;
  }
  for (auto x : m3) {
    both = (both * fact(x.second)) % m;
  }
  long long int sub = (first % m + second % m - both % m + m) % m;
  cout << (fact(n) % m - sub % m + m) % m;
}
