from math import sqrt
(x, n) = (int(x) for x in input().split(' '))

def factorize(n):
	factors = []
	if n % 2 == 0:
		factors.append(2)
		while n % 2 == 0:
			n //= 2
	factor = 3
	sqrtt = sqrt(n) + 10
	while n >= factor and factor <= sqrtt:
		if n % factor == 0:
			factors.append(factor)
			while n % factor == 0:
				n //= factor
		factor += 2
	if n != 1:
		factors.append(n)
	return factors
M = 1000000007

def mod_mul(a, b):
	return a % M * (b % M) % M

def fast_pow(x, p):
	result = 1
	current_power = 1
	current_factor = x
	while p >= current_power:
		if p & current_power:
			result = mod_mul(result, current_factor)
		current_factor = mod_mul(current_factor, current_factor)
		current_power *= 2
	return result
factors_of_x = factorize(x)
result = 1
for factor in factors_of_x:
	power = 0
	current = factor
	while current <= n:
		power += n // current
		current *= factor
	result = mod_mul(result, fast_pow(factor, power))
print(result)
