def li():

def num():

def nu():

def find_gcd(x, y):
	while y:
		(x, y) = (y, x % y)
	return x

def lcm(x, y):
	gg = find_gcd(x, y)
	return x * y // gg

yp = 0

def solve():
	t = 1
	for _ in range(t):
		(n, k) = num()

		xp = [99999999] * (n + 1)
		op = [-1] * (n + 1)
		for i in range(k):

			op[a[i]] = max(op[a[i]], i)
		cc = 0
		for i in range(1, n + 1):
			if op[i] == -1:
				cc += 1
			if i + 1 <= n:
				if op[i + 1] == -1:
					cc += 1
				elif op[i + 1] < xp[i]:
					cc += 1
			if i - 1 > 0:
				if op[i - 1] == -1:
					cc += 1
				elif op[i - 1] < xp[i]:
					cc += 1
		print(cc)
solve()
