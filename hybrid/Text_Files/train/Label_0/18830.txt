#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000007;
long long powmod(long long a, long long b) {
  long long res = 1;
  a %= mod;
  assert(b >= 0);
  for (; b; b >>= 1) {
    if (b & 1) res = res * a % mod;
    a = a * a % mod;
  }
  return res;
}
long long gcd(long long a, long long b) { return b ? gcd(b, a % b) : a; }
const int N = 301000;
map<int, long long> dp[N];
map<int, pair<int, int> > pre[N], way[N];
int n, a[N];
pair<int, int> cur;
vector<int> ret, ret2;
void upd(int p, int v, long long cost, pair<int, int> ww) {
  if (!dp[p].count(v) || cost < dp[p][v]) {
    dp[p][v] = cost;
    way[p][v] = ww;
    pre[p][v] = cur;
  }
}
void dfs(int n, int v) {
  if (n == 1) return;
  if (way[n][v].first != -1) ret.push_back(way[n][v].first);
  if (way[n][v].second != -1) ret.push_back(way[n][v].second);
  dfs(pre[n][v].first, pre[n][v].second);
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i < n + 1; i++) scanf("%d", a + i);
  for (int i = 1; i < n + 1; i++) dp[i].clear(), pre[i].clear(), way[i].clear();
  dp[1][a[1]] = 0;
  for (int i = 1; i < n; i++) {
    long long mv = 1ll << 60;
    for (auto p : dp[i]) {
      int v = p.first;
      long long w = p.second;
      if (w >= mv) continue;
      mv = w;
      cur = make_pair(i, v);
      if (v == 0) upd(i + 1, a[i + 1], w, make_pair(-1, -1));
      upd(i + 1, max(a[i + 1] - v, 0), w + min(v, a[i + 1]), make_pair(i, -1));
      if (i + 2 <= n && a[i + 1] <= a[i + 2])
        upd(i + 2, a[i + 2] - a[i + 1], w + a[i + 1], make_pair(i + 1, -1));
      if (i + 2 <= n && a[i + 1] >= a[i + 2])
        upd(i + 2, 0, w + a[i + 2] + min(v, (a[i + 1] - a[i + 2])),
            make_pair(i, i + 1));
    }
  }
  int mv = -1;
  long long w = 1ll << 60;
  for (auto p : dp[n])
    if (p.second < w) w = p.second, mv = p.first;
  dfs(n, mv);
  reverse((ret).begin(), (ret).end());
  for (auto p : ret) {
    if (a[p] > 0 && a[p + 1] > 0) {
      int z = min(a[p], a[p + 1]);
      a[p] -= z;
      a[p + 1] -= z;
      ret2.push_back(p);
    }
  }
  printf("%d\n", ((int)(ret2).size()));
  for (auto p : ret2) printf("%d ", p);
  puts("");
}
