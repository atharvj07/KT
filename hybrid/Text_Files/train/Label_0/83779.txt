import sys
input = sys.stdin.readline
t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	if n < 70 and pow(2, n - 1) < k:
		print(-1)
		continue
	k -= 1
	k = bin(k)[2:].zfill(n - 1)
	l = []
	cnt = 1
	for i in range(n - 1):
		if k[i] == '1':
			cnt += 1
		else:
			l.append(cnt)
			cnt = 1
	l.append(cnt)
	idx = 0
	res = [0] * n
	for x in l:
		for i in range(idx, idx + x):
			res[i] = idx * 2 + x - i
		idx += x
	print(*res)
