#include <bits/stdc++.h>

using namespace std;
typedef long long int lli;
typedef pair<int,int> pii;

const int MAXN = 300010;

int n, m;

int last[MAXN];

bool mark[MAXN];

vector<int> adj[MAXN];

vector<int> answer;

queue<int> q;

void addAnswer(int cur)
{
	mark[cur] = true;
	answer.push_back( cur );

	for(int i = 0 ; i < (int)adj[cur].size() ; i++)
	{
		int viz = adj[cur][i];

		if( mark[viz] ) continue;

		q.push( viz );
		mark[viz] = true;
	}
}

void BFS()
{
	addAnswer( 1 );

	while( !q.empty() )
	{
		int nxtAnswer = -1;

		while( !q.empty() && nxtAnswer == -1 )
		{
			int cur = q.front();

			while( last[cur] < (int)adj[cur].size() )
			{
				int viz = adj[cur][ last[cur]++ ];

				if( !mark[viz] )
				{
					nxtAnswer = viz;
					break;
				}
			}

			if( last[cur] == adj[cur].size() )
				q.pop();
		}

		if( nxtAnswer == -1 ) break;

		addAnswer( nxtAnswer );
	}
}

int main()
{
	int t = 1;
	scanf("%d",&t);

	while( t-- )
	{
		scanf("%d %d",&n,&m);

		for(int i = 1 ; i <= n ; i++)
			adj[i].clear(), mark[i] = false, last[i] = 0;

		answer.clear();

		for(int i = 1 ; i <= m ; i++)
		{
			int U, V;
			scanf("%d %d",&U,&V);

			adj[U].push_back( V );
			adj[V].push_back( U );
		}

		BFS();

		bool connected = true;

		for(int i = 1 ; i <= n ; i++)
			connected = ( connected && mark[i] );

		if( !connected ) printf("NO");
		else 
		{
			printf("YES\n");

			printf("%d\n",(int)answer.size());

			for(int i = 0 ; i < (int)answer.size() ; i++)
				printf("%d ",answer[i]);
		}

		printf("\n");
	}
}
