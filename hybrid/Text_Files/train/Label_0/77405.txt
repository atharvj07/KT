n_elements = int(input())
elements = [int(x) for x in input().split()]
res = 0
for (i, x) in enumerate(elements):
	res -= (n_elements - 1 - i) * x
for (i, y) in enumerate(elements):
	res += i * y
m = dict()
m[elements[0]] = 1
for num in elements[1:]:
	if num + 1 in m:
		res += m[num + 1]
	if num - 1 in m:
		res -= m[num - 1]
	if num in m:
		m[num] += 1
	else:
		m[num] = 1
print(res)
