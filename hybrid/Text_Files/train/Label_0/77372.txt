n = int(input())
array = str(input()).split(' ')
n0 = n
strt_of_grp = 0
out = 1
while n > 1:
	for i in range(1, int(n0 / n) + 1):
		for j in range(strt_of_grp, n * i - 1):
			if int(array[j + 1]) >= int(array[j]):
				out += 1
				if out == n:
					print(out)
					exit()
					out = 1
			else:
				out = 1
				break
		strt_of_grp += n
	strt_of_grp = 0
	n = int(n / 2)
print(out)
