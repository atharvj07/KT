inputs = input().split(' ')
(ran, base, limit) = (int(inputs[0]), int(inputs[1]), int(inputs[2]))
if base - limit > 0:
	start = base - limit
else:
	start = 1
if base + limit > ran:
	finish = ran + 1
else:
	finish = base + limit + 1
output = ''
if start is not 1:
	output += '<< '
for i in range(start, finish):
	if i == base:
		output += '(' + str(base) + ') '
	else:
		output += str(i) + ' '
if ran > finish - 1:
	output += '>>'
print(output)
