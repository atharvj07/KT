#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main(){

    int n, m;
    while(true){
        cin>>n>>m;
        if(n==0)break;
        vector<int>vec(m+1, 0);
        for(int i=0;i<n;i++){
            int d, v;
            cin>>d>>v;
            vec[d] = max(vec[d], v);
        }
        int ans=0;
        for(int i=1;i<=m;i++){
            ans+=vec[i];
        }
        cout<<ans<<endl;
    }
    return 0;
}
