t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	v = 0
	op = []
	c = s.count('1')
	for i in range(n):
		if s[i] == '0' and i > n - 1 - c:
			op.append(i + 1)
			v += 1
		elif s[i] == '1' and i <= n - 1 - c:
			op.append(i + 1)
			v += 1
	if v == 0:
		print(0)
	else:
		print(1)
		print(v, end=' ')
		for i in op:
			print(i, end=' ')
		print()
