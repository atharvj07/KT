import sys
input = lambda : sys.stdin.readline().strip()

def get(arr, p):
	locs = [i for (i, a) in enumerate(arr) if a == p]
	goal = list(range(0, len(arr), 2))
	return sum((abs(i - j) for (i, j) in zip(locs, goal)))

def solve():
	n = int(input())
	arr = list(map(int, input().split()))
	arr = [a & 1 for a in arr]
	x1 = sum(arr)
	x0 = n - x1
	if abs(x1 - x0) > 1:
		return -1
	if n & 1:
		if x0 > x1:
			return get(arr, 0)
		else:
			return get(arr, 1)
	else:
		return min(get(arr, 0), get(arr, 1))
for _ in range(int(input())):
	print(solve())
