import java.io.*;
import java.util.*;

public class john{
    static ArrayList<Integer>[] edge, back;
    static int[] depth;
    static int[] father;
    static boolean[] leaf;
    
    public static void dfs(int u) {
	for (int v : edge[u]) {
	    if (father[v] == -1) {
		depth[v] = depth[u] + 1;
		father[v] = u;
		dfs(v);
		leaf[u] = false;
	    } else if (v != father[u]) {
		back[u].add(v);
	    }
	}
    }

    public static void printPath(int length, int start, int end, PrintWriter out) {
	int u = start;
	out.println(length);
	while (u != end) {
	    out.print(u + 1 + " ");
	    u = father[u];
	}
	out.println();
    }
    
    public static void main(String[] args) throws IOException {
	BufferedReader buf = new BufferedReader(new InputStreamReader(System.in));
	StringTokenizer scan = new StringTokenizer(buf.readLine());
	PrintWriter out = new PrintWriter(System.out);
	
	int n, m, k;
	n = Integer.parseInt(scan.nextToken());
	m = Integer.parseInt(scan.nextToken());
	k = Integer.parseInt(scan.nextToken());

	edge = new ArrayList [n];
	back = new  ArrayList [n];
	depth = new int [n];
	father = new int [n];
	leaf = new boolean [n];
	
	for (int i = 0; i < n; ++i) {
	    edge[i] = new ArrayList<> ();
	    back[i] = new ArrayList<> ();
	    father[i] = -1;
	    leaf[i] = true;
	}
	    
	for (int i = 1; i <= m; ++i) {
	    scan = new StringTokenizer(buf.readLine());
	    int u = Integer.parseInt(scan.nextToken()) - 1;
	    int v = Integer.parseInt(scan.nextToken()) - 1;
	    edge[u].add(v);
	    edge[v].add(u);
	}
	
	depth[0] = 1;
	father[0] = n;
	dfs(0);

	for (int i = 0; i < n; ++i) {
	    if (depth[i] >= (n + k - 1) / k) {
		out.println("PATH");
		printPath(depth[i], i, n, out);
		out.close();
		return;
	    }
	}

	out.println("CYCLES");
	for (int i = 0; i < n; ++i) {
	    if (leaf[i]) {
		int x = back[i].get(0);
		int y = back[i].get(1);
		
		if ((depth[i] - depth[x] + 1) % 3 != 0 && (depth[i] - depth[x] + 1) > 2)
		    printPath(depth[i] - depth[x] + 1, i, father[x], out);
		else if ((depth[i] - depth[y] + 1) % 3 != 0 && (depth[i] - depth[y] + 1) > 2)
		    printPath(depth[i] - depth[y] + 1, i, father[y], out);
		else {
		    if (depth[x] < depth[y]) {
			int tmp = y;
			y = x;
			x = tmp;
		    }
		    
		    int u = x;
		    out.println(depth[x] - depth[y] + 2);
		    while (u != father[y]) {
			out.print(u + 1 + " ");
			u = father[u];
		    }
		    out.print(i + 1 + " ");
		    out.println();
		}

		if (--k == 0) break;
	    }
	}
	out.close();
    }
}
