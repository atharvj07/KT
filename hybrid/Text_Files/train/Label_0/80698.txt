def res(s):
	t = 0
	m = 0
	for i in s:
		if i == 'T':
			t += 1
		elif t == 0:
			return False
		else:
			t -= 1
			m += 1
	if t != m:
		return False
	return True

def solve():
	n = int(input())
	first = input()
	if not res(first):
		return 'NO'
	if not res(first[::-1]):
		return 'NO'
	else:
		return 'YES'
for _ in range(int(input())):
	print(solve())
