t = int(input())
for _ in range(t):
	n = int(input())
	l = list(map(int, input().split()))
	index_list = [1] * n
	for i in range(n):
		index_list[l[i] - 1] = i
	lp = [1]
	(mi, ma) = (index_list[0], index_list[0])
	for j in range(1, n):
		if j == n - 1:
			lp.append(1)
		else:
			mi = min(mi, index_list[j])
			ma = max(ma, index_list[j])
			if ma - mi == j:
				lp.append(1)
			else:
				lp.append(0)
	print(*lp, sep='')
