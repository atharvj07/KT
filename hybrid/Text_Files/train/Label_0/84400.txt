completo = False

def lol(l):
	como_estas = len(l)
	for amigo in range(como_estas):
		if amigo % 2 == 0:
			l[amigo] = (l[amigo] + 1) % como_estas
		else:
			l[amigo] = (l[amigo] - 1) % como_estas
	return l
en = int(input())
numeros = input().split(' ')
integ = []
for numero in numeros:
	integ.append(int(numero))
cero = []
for numero in range(en):
	cero.append(numero)
for i in range(en + 1337):
	integ = lol(integ)
	if integ == cero and completo == False:
		print('Yes')
		completo = True
if completo == False:
	print('No')
