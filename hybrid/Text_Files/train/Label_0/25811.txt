#include <bits/stdc++.h>
using namespace std;
char _buf[100000], *_op(_buf), *_ed(_buf);
inline int getint() {
  int _s = 0;
  char _ch = (_op == _ed && (_ed = (_op = _buf) + fread(_buf, 1, 100000, stdin),
                             _op == _ed)
                  ? EOF
                  : *_op++);
  while (!('0' <= _ch && _ch <= '9'))
    _ch = (_op == _ed && (_ed = (_op = _buf) + fread(_buf, 1, 100000, stdin),
                          _op == _ed)
               ? EOF
               : *_op++);
  while (('0' <= _ch && _ch <= '9'))
    _s = _s * 10 + _ch - 48,
    _ch = (_op == _ed && (_ed = (_op = _buf) + fread(_buf, 1, 100000, stdin),
                          _op == _ed)
               ? EOF
               : *_op++);
  return _s;
}
const int N = 2e5 + 1, INF = 0x3f3f3f3f;
inline void cmax(int& x, int y) {
  if (x - y >> 31) x = y;
}
long long int k, lim;
set<pair<int, int> > mp;
vector<int> e[N];
int T, n, rt, now, tot, cnt, minn, flag, id[N];
int dep[N], val[N], siz[N], vis[N], tmp[N];
struct node {
  int x, y, w;
  inline int operator<(const node& u) const { return w < u.w; }
};
vector<node> t[N];
inline void add_edge(int x, int y) { e[x].push_back(y), e[y].push_back(x); }
inline void input() {
  cin >> n >> k, cnt = n >> 1;
  for (int i = 2; i <= n; i++) add_edge(getint(), getint());
}
inline void getrt(int u, int fa = 0) {
  int maxn(siz[u] = 1);
  for (int v : e[u])
    if (v != fa) getrt(v, u), cmax(maxn, siz[v]), siz[u] += siz[v];
  if (cmax(maxn, n - siz[u]), maxn < minn) minn = maxn, rt = u;
}
inline void init(int u, int fa) {
  siz[u] = 1, val[u] = u, lim += dep[u] = dep[fa] + 1, minn++, id[u] = now;
  for (int v : e[u])
    if (v != fa && (init(v, u), siz[u] += siz[v], val[v])) {
      if (!val[u])
        val[u] = val[v], minn++;
      else
        t[now].push_back((node){val[u], val[v], dep[u] << 1}), val[u] = 0,
                                                               minn--;
    }
}
inline void init() {
  minn = INF, getrt(1), minn = 0;
  for (int v : e[rt])
    init(now = v, rt), sort(t[v].begin(), t[v].end()),
        mp.insert(pair<int, int>(siz[v], v));
}
inline void solve(int x, int y) {
  k -= dep[x] + dep[y], lim -= dep[x] + dep[y], vis[x] = vis[y] = 1, cnt--,
                                                cout << x << ' ' << y << '\n';
}
inline void solve(const node& u) { k += u.w, solve(u.x, u.y); }
inline void find(int u, int w, int fa) {
  val[u] = !vis[u] * u;
  for (int v : e[u])
    if (v != fa && (find(v, w, u), val[v])) {
      if (!val[u])
        val[u] = val[v];
      else if (flag && dep[u] == w)
        solve(val[u], val[v]), k += w << 1, flag = 0;
    }
}
inline void dfs(int u, int fa = 0) {
  if (!vis[u]) tmp[++tot] = u;
  for (int v : e[u])
    if (v != fa) dfs(v, u);
}
inline void work() {
  if (lim < k || k < minn || (k - lim & 1)) return void(cout << "NO\n");
  cout << "YES\n", flag = 1;
  for (int u; lim != k;) {
    u = (*--mp.end()).second, mp.erase(pair<int, int>(siz[u], u)),
    mp.insert(pair<int, int>(siz[u] -= 2, u));
    if (lim - t[u].back().w >= k)
      solve(t[u].back()), t[u].pop_back();
    else
      find(u, lim - k >> 1, rt);
  }
  dfs(rt), tot >>= 1;
  for (int i = 1; i <= tot; i++) solve(tmp[i], tmp[i + tot]);
}
int main() {
  input();
  init();
  work();
  return 0;
}
