import java.util.*;

public class codechef {
    private static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int t = 0;
        if (sc.hasNext())
            t = sc.nextInt();
        for (int i = 1; i <= t; i++) {
            solve();
        }
    }

    public static void solve() {
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[][] matrix = new char[n][m];
        int[][] heights = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                heights[i][j] = 0;
            }
        }
        for (int i = 0; i < n; i++) {
            matrix[i] = sc.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            if (matrix[i][0] == '*') {
                heights[i][0] = 1;
            }
            if (matrix[i][m - 1] == '*') {
                heights[i][m - 1] = 1;
            }
        }
        for (int i = 0; i < m; i++) {
            if (matrix[n - 1][i] == '*') {
                heights[n - 1][i] = 1;
            }
        }
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 1; j < m - 1; j++) {
                if (matrix[i][j] == '*') {
                    int min = Integer.MAX_VALUE;
                    for (int x = j - 1; x <= j + 1; x++) {
                        min = Math.min(heights[i + 1][x], min);
                    }
                    if (min != 0)
                        heights[i][j] = min + 1;
                    else {
                        heights[i][j] = 1;
                    }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                ans += heights[i][j];
            }
        }
        System.out.println(ans);
    }


}
