n = int(input())
alph = ['a', 'b', 'c', 'd', 'e', 'f', 'g', 'h', 'i', 'j', 'k', 'l', 'm', 'n', 'o', 'p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
for i in range(n):
	(a, b, c) = map(int, input().split())
	ans = ''
	while a != len(ans):
		for q in range(c):
			if len(ans) + 1 <= a:
				ans += alph[q]
			else:
				break
	print(ans)
