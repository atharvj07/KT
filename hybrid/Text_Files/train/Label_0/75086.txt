import sys
import math
import bisect
from collections import deque
sys.setrecursionlimit(1000000000)

def input():
	return sys.stdin.readline().strip()

def iinput():
	return int(input())

def finput():
	return float(input())

def tinput():
	return input().split()

def rinput():
	return map(int, tinput())

def rlinput():
	return list(rinput())

def modst(a, s):
	res = 1
	while s:
		if s % 2:
			res *= a
		a *= a
		s //= 2
	return res

def main():
	(n, k) = rinput()
	s = input()
	q = -1
	w = -1
	e = -1
	r = -1
	for i in range(n):
		if s[i] == '0':
			if q == -1:
				q = i
			w = i
		else:
			if e == -1:
				e = i
			r = i
	z = w - q
	x = r - e
	if min(z, x) < k:
		return 'tokitsukaze'
	if z > k or x > k:
		return 'once again'
	c = 0
	v = 0
	for i in range(n):
		if s[i] == '0':
			v += 1
			c = 0
		else:
			c += 1
			v = 0
		if c >= k or v >= k:
			return 'once again'
	return 'quailty'
for i in range(1):
	print(main())
