t = int(input())
for _ in range(t):
	s = input()
	x = 0
	y = 0
	D = {}
	ans = 0
	for i in range(len(s)):
		c_x = x
		c_y = y
		if s[i] == 'N':
			c_y = y + 1
		if s[i] == 'S':
			c_y = y - 1
		if s[i] == 'W':
			c_x = x - 1
		if s[i] == 'E':
			c_x = x + 1
		if (x, y, c_x, c_y) in D:
			ans += 1
		else:
			D[x, y, c_x, c_y] = 1
			D[c_x, c_y, x, y] = 1
			ans += 5
		x = c_x
		y = c_y
	print(ans)
