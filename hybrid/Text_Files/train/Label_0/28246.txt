#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n, m, k, i, a[100005], b[100005], A = 0, B = 0, j;
  cin >> n >> m >> k;
  for (i = 1; i <= n; i++) cin >> a[i];
  for (i = 1; i <= m; i++) cin >> b[i];
  sort(a, a + n + 1);
  sort(b, b + m + 1);
  i = n;
  j = m;
  while (i > 0 && j >= 0) {
    if (a[i] == b[j]) {
      i--;
      j--;
    } else if (a[i] > b[j]) {
      A++;
      if (A > B) {
        cout << "YES";
        return 0;
      }
      i--;
    } else {
      B++;
      j--;
    }
  }
  cout << "NO";
  return 0;
}
