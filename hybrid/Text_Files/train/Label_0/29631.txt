#include <iostream>
#include <stack>
#include <algorithm>
#include <cstdio>
#include <cstring>
#include <vector>
#include <set>
using namespace std;
 
pair<int,int> rot(pair<int,int> x){
	return make_pair(x.second,x.first);
}

int main(){
	int N;
	cin >> N;
	vector< pair<int,int> > w;
	for(int i = 0 ; i < N ; i++){
		int a,b;
		cin >> a >> b;
		if( b < a ) swap(a,b);
		w.push_back(make_pair(a,b));
	}
	sort(w.begin(),w.end());
	
	multiset<int> current;
	current.insert(-1e9);
	for(int i = 0 ; i < w.size() ; ){
		int j = i;
		while( j < w.size() && w[j].first == w[i].first ) j++;
		
		for(int k = i ; k < j ; k++){
			multiset<int>::iterator it = --current.lower_bound(w[k].second);
			if( (*it) > 0 ){
				current.erase(it);
			}
		}
		for(int k = i ; k < j ; k++)
			current.insert(w[k].second);
		
		i = j;
	}
	cout << current.size() - 1 << endl;
	
	
}
