def plus(a, b):
	c = ''
	i = len(a) - 1
	arg = 0
	while i >= 0:
		k = arg + ord(a[i]) + ord(b[i]) - 194
		arg = 1
		if k < 26:
			arg = 0
		i -= 1
		c = chr(k % 26) + c
	c = chr(1) * arg + c
	return c

def minus(a):
	c = ''
	i = 0
	arg = 0
	while i < len(a):
		c += chr((arg * 26 + ord(a[i])) // 2)
		arg = ord(a[i]) % 2
		i += 1
	if arg:
		c = c[:len(c) - 1] + chr(ord(c[len(c) - 1]) + 1)
	return c
n = int(input())
a = input()
b = input()
a1 = ''
b1 = ''
i = 0
while i < n:
	a1 += chr(ord(a[i]))
	b1 += chr(ord(b[i]))
	i += 1
c = plus(a1, b1)
c = minus(c)
c1 = ''
for q in c:
	c1 += chr(ord(q) + ord('a'))
print(c1[len(c1) - n:])
