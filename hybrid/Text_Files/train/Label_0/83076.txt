for z in range(int(input())):
	(a, b) = map(int, input().split())
	s = input()
	st = s.strip('0')
	if len(st) != 0:
		count = 0
		total = a
		for i in st:
			if i == '0':
				count += 1
			elif count != 0:
				if count * b <= a:
					total += count * b
				else:
					total += a
				count = 0
		print(total)
	else:
		print(0)
