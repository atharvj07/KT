#include <bits/stdc++.h>
int i;
int max(int a, int b) { return (a < b) ? b : a; }
int main() {
  int n;
  scanf("%d", &n);
  int even;
  int odd;
  odd = even = 0;
  for (i = 0; i < n; ++i) {
    int x;
    scanf("%d", &x);
    if (x % 2)
      odd++;
    else
      even++;
  }
  if (even > odd) return 0 * printf("%d", odd);
  int ans = even;
  odd = odd - even;
  if (odd == 1) {
    ans--;
  }
  printf("%d", max(0, ans + odd / 3));
  return 0;
}
