#include <bits/stdc++.h>

using namespace std;

#define MAX 1010

struct Point{
  double x,y;
  Point(){}
  Point(double x,double y) : x(x),y(y) {}
};

int N,X,Y;
Point sp;
vector<Point> p;
vector<int> G[MAX];

double getDist(Point &a,Point &b){
  return sqrt(pow(a.x-b.x,2)+pow(a.y-b.y,2));
}

void init(){
  for(int i = 0 ; i < MAX ; i++){
    G[i].clear();
  }  
}

void solve(){
  int res = 0;
  queue<int> que;
  que.push(0);
  int d[N+1];
  memset(d,0,sizeof(d));
  bool ok=1;
  while(!que.empty()) {
    int x=que.front();que.pop();
    for(int i=0; i<G[x].size(); i++) {
      int y=G[x][i];
      if(d[y]) continue;
      if(x && X>Y) ok=0;
      d[y]=d[x]+1;
      que.push(y);
    }
  }
  for(int i=1; i<=N; i++) res=max(res,d[i]*X);
  if(ok) cout << res << endl;
  else cout << "You're always welcome!" << endl;
}

int main(){
  int Tc;
  cin >> Tc;
  while(Tc--){
    init();
    cin >> N >> X >> Y;
    cin >> sp.x >> sp.y;
    p.resize(N);
    for(int i = 0 ; i < N ; i++){
      cin >> p[i].x >> p[i].y;
      if(getDist(sp,p[i]) <= 10){
	G[0].push_back(i+1);
      }
    }
    for(int i = 0 ; i < N ; i++){
      for(int j = 0 ; j < N ; j++){
	if(i == j){ continue; }
	if(getDist(p[i],p[j]) <= 50){
	  G[i+1].push_back(j+1);
	}
      }
    }
    solve();
  }
  return 0;
}
