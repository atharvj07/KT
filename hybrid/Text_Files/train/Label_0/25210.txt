#include <bits/stdc++.h>
using namespace std;
int main() {
  string s1, s2;
  cin >> s1 >> s2;
  for (int i = 0; i < s1.length(); i++) {
    char ch1 = tolower(s1[i]);
    char ch2 = tolower(s2[i]);
    if (ch1 < ch2) {
      cout << "-1";
      return 0;
    } else if (ch1 > ch2) {
      cout << "1";
      return 0;
    }
  }
  cout << "0";
  return 0;
}
