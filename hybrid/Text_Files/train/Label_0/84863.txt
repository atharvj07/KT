import math
z = input().split()
w = int(z[0])
m = int(z[1])
k = int(z[2])
digs = int(math.log10(m)) + 1
upper = int('9' * digs)
ans = 0
while w > 0:
	if (upper - m + 1) * digs * k <= w:
		w = w - (upper - m + 1) * digs * k
		ans += upper - m + 1
		digs += 1
		m = int('1' + '0' * (digs - 1))
		upper = int('9' * digs)
	else:
		ans += int(w / (k * digs))
		break
print(ans)
