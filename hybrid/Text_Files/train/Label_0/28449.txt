#include <bits/stdc++.h>
using namespace std;
int tc;
bool cmp(string str1, string str2, long long int arr[], long long int K) {
  int j = 0;
  for (long long int i = 0; i < K; i++) {
    str1[arr[i] - 1] = '-';
  }
  for (int i = 0; i < str1.length() && j < str2.length(); i++) {
    if (str1[i] == '-') continue;
    if (str1[i] == str2[j]) j++;
  }
  return j == str2.length();
}
int main(void) {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  ;
  string s1, s2;
  cin >> s1 >> s2;
  long long int n = s1.size();
  long long int a[n];
  for (long long int i = 0; i < n; i++) cin >> a[i];
  long long int l = 0;
  long long int r = n;
  long long int cnt = 0;
  while (l <= r) {
    long long int mid = l + (r - l) / 2;
    bool ans = cmp(s1, s2, a, mid);
    if (ans) {
      l = mid + 1;
      cnt = mid;
    } else {
      r = mid - 1;
    }
  }
  cout << cnt;
}
