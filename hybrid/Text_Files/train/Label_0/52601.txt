import java.util.Scanner;

public class MargariteAndTheBestPresent {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int q = s.nextInt();
        int l = 0, r = 0;
        for(int i = 0; i < q; i++) {
            l = s.nextInt();
            r = s.nextInt();
            int res = 0;
            int len = r - l + 1;
            if(len % 2 == 0) {
                if(l % 2 == 1) res = len / 2;
                else res = -(len / 2);
            }
            else {
                if(len > 1) {
                    if(l % 2 == 1) res = (len - 1) / 2;
                    else res = -((len - 1) / 2);

                }
                if(l % 2 == 1) res -= r;
                else res += r;
            }
            System.out.println(res);
        }
    }
}
