#include <bits/stdc++.h>
using namespace std;
void Preset() {
  cin.sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
}
typedef struct Point2D {
  int x;
  int y;
} Point2D;
vector<Point2D> coordinate;
int main() {
  Preset();
  int N, K;
  long double res = 0;
  cin >> N >> K;
  for (int i = 0; i < N; i++) {
    int x, y;
    cin >> x >> y;
    coordinate.push_back({x, y});
  }
  for (int i = 0; i < N - 1; i++) {
    long double tmp = pow(coordinate[i].x - coordinate[i + 1].x, 2) +
                      pow(coordinate[i].y - coordinate[i + 1].y, 2);
    res += sqrt(tmp);
  }
  cout.precision(100);
  cout << res * K / 50 << endl;
  return 0;
}
