import java.util.*;
import java.io.*;
public class Main{
    static int n,k,a[];
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        n=sc.nextInt();
        k=sc.nextInt();
        a=new int[n];
        for(int i=0;i<n;i++)a[i]=sc.nextInt();
        int ans=bs(1,(int)1e9);
        System.out.println(ans);
    }
    public static int bs(int lo,int hi){
        int ans=-1;
        while(lo<hi){
            int mid=(lo+hi)/2;
            if(check(mid,0)||check(mid,1)){
                ans=mid;
                hi=mid;
            }else lo=mid+1;
        }
        return ans;
    }
    public static boolean check(int val,int curr){
        int ans=0;
        for(int i=0;i<n;i++){
            if(curr==0){
                ans++;
                curr^=1;
            }else{
                if(a[i]<=val){
                    ans++;
                    curr^=1;
                }
            }
        }
        return ans>=k;
    }
}
