#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<int> hozon[26];
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  for (int a = 0; a < k; a++) {
    hozon[s[a] - 'a'].push_back(a);
  }
  for (int a = 0; a < 26; a++) hozon[a].push_back(1000000);
  for (int a = 0; a < n; a++) {
    string t;
    cin >> t;
    long long sum = 0;
    int m = t.size();
    for (int b = 0; b < m; b++) {
      int p = t[b] - 'a';
      if (hozon[p].size() == 1)
        sum += (long long)m;
      else {
        int q = lower_bound(hozon[p].begin(), hozon[p].end(), b) -
                hozon[p].begin() - 1;
        sum += (long long)min(abs(hozon[p][q] - b), abs(hozon[p][q + 1] - b));
      }
    }
    cout << sum << endl;
  }
  return 0;
}
