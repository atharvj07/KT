import java.util.*;
import java.math.*;
public class Hie
{
public static void main(String[] args) {
Scanner sc=new Scanner(System.in);
int n=sc.nextInt();
long a[]=new long[n];
for(int i=0;i<n;i++)
a[i]=sc.nextLong();
Arrays.sort(a);
long s=0;
for(int i=0;i<n-1;i++)
{
if(a[i+1]-a[i]>1)
s=s+(a[i+1]-a[i]-1);
}
System.out.print(s);
		

}
}
