#include <bits/stdc++.h>
using namespace std;
const long long size = 1e7 + 10;
long long n, ans, top, q[size];
long long read() {
  long long x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = (x << 1) + (x << 3) + ch - '0';
    ch = getchar();
  }
  return x * f;
}
int main() {
  n = read();
  if (n % 3 == 0) {
    n /= 3;
    for (long long i = 1; i <= sqrt(n); i++) {
      if (n % i == 0) {
        q[++top] = i;
      }
    }
    for (long long i = 1; i <= top; i++) {
      for (long long j = i; j <= top; j++) {
        long long x = q[i], y = q[j], z = n / x / y;
        if (n % (x * y) == 0 && (x + y + z) % 2 == 0 && x + y > z &&
            x + z > y && y + z > x) {
          ans += (i != j) ? 2 : 1;
        }
      }
    }
    printf("%lld\n", ans);
  } else
    printf("0");
  return 0;
}
