#include <bits/stdc++.h>
signed main() {
  long long sum = 0, n, k, iden = 0;
  scanf("%lld %lld", &n, &k);
  for (long long i = 0; i < k; i++) {
    long long x, y;
    scanf("%lld %lld", &x, &y);
    sum += y;
    iden = (iden + x * y) % n;
  }
  if (sum > n) {
    printf("-1");
    return 0;
  }
  iden = iden % n;
  if (sum == n && iden != (((n * n - n)) / 2) % n) {
    printf("-1");
    return 0;
  }
  printf("1");
}
