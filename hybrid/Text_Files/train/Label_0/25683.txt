#include <bits/stdc++.h>
using namespace std;
int h[11][11];
double turns[110][2];
pair<int, int> get(int pos) {
  int row = pos / 10;
  int col = (row % 2 == 0) ? (pos % 10) : (9 - (pos % 10));
  return make_pair(row, col);
}
int inv(int i, int j) {
  int col = (i % 2 == 0) ? j : 9 - j;
  return 10 * i + col;
}
int main() {
  for (int i = 0; i < 10; i++) {
    for (int j = 0; j < 10; j++) {
      scanf("%d", &h[i][j]);
    }
  }
  turns[0][0] = turns[0][1] = 0;
  for (int pos = 1; pos < 100; pos++) {
    int prob = 0;
    double t = 1;
    for (int j = 1; j <= 6; j++) {
      if (j > pos) {
        prob++;
      } else {
        t += turns[pos - j][1] / 6;
      }
    }
    t = t / (1.0 - prob / 6.0);
    turns[pos][0] = turns[pos][1] = t;
    pair<int, int> i = get(pos);
    if (h[i.first][i.second] > 0) {
      pair<int, int> j = get(inv(i.first, i.second - h[i.first][i.second]));
      turns[pos][1] =
          min(turns[pos][1],
              turns[inv(i.first - h[i.first][i.second], i.second)][0]);
    }
  }
  printf("%.10lf\n", turns[99][0]);
  return 0;
}
