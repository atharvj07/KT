struct UnionFind{
	vector<int> d;
	UnionFind(int n):d(n, -1){}
	bool unite(int x, int y){
		if((x = root(x)) != (y = root(y))){
			if(d[y] < d[x]) swap(x, y);
			d[x] += d[y]; d[y] = x;
		}
		return x != y;
	}
	int root(int x){return d[x] < 0 ? x : d[x] = root(d[x]);}
};

vi bfs(const vector<vi> &g, vi s){
	vi dp(n+1, INF);
	queue<int> q;
	for(int u : s){
		q.push(u);
		dp[u] = 0;
	}
	while(!q.empty()){
		int u = q.front(); q.pop();
		for(int v : g[u])if(dp[u]+1 < dp[v]){
			q.push(v);
			dp[v] = dp[u] + 1;
		}
	}
	return dp;
}
