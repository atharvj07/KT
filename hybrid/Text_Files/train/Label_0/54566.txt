import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Airport {

	/**
	 * @param args
	 * @throws IOException
	 */
	public static void main(String[] args) throws IOException {
		int mxo = 0, mno = 0;
		BufferedReader rd = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer t = new StringTokenizer(rd.readLine(), " ");
		int n = Integer.parseInt(t.nextToken()), m = Integer.parseInt(t
				.nextToken());
		int[] ms = new int[m];
		int[] ms1 = new int[m];
		t = new StringTokenizer(rd.readLine(), " ");
		int max = -1;
		int min = -1;
		for (int i = 0; i < m; i++) {
			ms[i] = Integer.parseInt(t.nextToken());
			ms1[i] = ms[i];
			if (ms1[i] != 0)
				if (min == -1 || ms1[i] <= ms1[min])
					min = i;
			if (ms[i] != 0)
				if (max == -1 || ms[i] >= ms[max])
					max = i;

		}
		mxo += ms[max];
		mno += ms1[min];
		ms[max]--;
		ms1[min]--;
		for (int i = 1; i < n; i++) {
			max = -1;
			min = -1;
			for (int j = 0; j < m; j++) {
				if (ms1[j] != 0)
					if (min == -1 || ms1[j] <= ms1[min])
						min = j;
				if (ms[j] != 0)
					if (max == -1 || ms[j] >= ms[max])
						max = j;
			}
			mxo += ms[max];
			mno += ms1[min];
			ms[max]--;
			ms1[min]--;
		}
		System.out.println(mxo + " " + mno);
	}
}
