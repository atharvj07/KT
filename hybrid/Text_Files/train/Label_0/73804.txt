N = int(input())
for _ in range(N):
	n = int(input())
	nums = [0] + list(map(int, input().strip().split())) + [1]
	for (i, (cur_r, next_r)) in enumerate(zip(nums[:], nums[1:])):
		if cur_r == 0 and next_r == 1:
			break
	out_nums = list(range(1, i + 1)) + [n + 1] + list(range(i + 1, n + 1))
	print(' '.join(map(str, out_nums)))
