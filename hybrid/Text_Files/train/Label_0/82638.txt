import sys
import math
from functools import reduce
import bisect

def getN():
	return int(input())

def getNM():
	return map(int, input().split())

def getList():
	return list(map(int, input().split()))

def input():
	return sys.stdin.readline().rstrip()

def index(a, x):
	i = bisect.bisect_left(a, x)
	if i != len(a) and a[i] == x:
		return i
	return False

def find_dividers(x):
	dividers = []
	for i in range(2, int(math.sqrt(x)) + 1):
		if x % i == 0:
			dividers.append(i)
			if i * i != x:
				dividers.append(x // i)
	return dividers
for _ in range(int(input())):
	s = input()
	amount = dict()
	for x in s:
		if x == 'R':
			amount.setdefault('R', 0)
			amount['R'] += 1
		elif x == 'S':
			amount.setdefault('S', 0)
			amount['S'] += 1
		else:
			amount.setdefault('P', 0)
			amount['P'] += 1
	letter = max(amount, key=amount.get)
	if letter == 'R':
		print('P' * len(s))
	elif letter == 'S':
		print('R' * len(s))
	else:
		print('S' * len(s))
