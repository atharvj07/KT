import java.util.*;
public class Main {
	static int mod=1000000007;
	static int n,K;
	static int[] p2=new int[5123];
	static int[][] dp=new int[5123][5123];
	static int ksm(int x,int n) {
		int ret=1;
		while (n>0) {
			if ((n&1)==1) ret=(int)(1L*ret*x%mod);
			n>>=1;
			x=(int)(1L*x*x%mod);
		}
		return ret;
	}
	static int DP(int a,int b) {
		if (a==0) return p2[b];
		if (dp[a][b]>0) return dp[a][b];
		int ret=0,c=n-b;
		if (b>0) ret=(ret+(int)(1L*b*DP(a-1,b)%mod))%mod;
		if (c>0) ret=(ret+(int)(1L*c*DP(a-1,b+1)%mod))%mod;
		return dp[a][b]=ret;
	}
	public static void main(String[] args) {
		Scanner in=new Scanner(System.in);
		n=in.nextInt();
		K=in.nextInt();
		p2[0]=ksm(2,n);
		for (int i=1;i<=K;++i) p2[i]=(int)(500000004L*p2[i-1]%mod);
		System.out.println(DP(K,0));
	}
}
