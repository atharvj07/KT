import heapq
import sys
input = sys.stdin.readline
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
ans = [0] * n
h = []
for i in range(n):
	heapq.heappush(h, (-a[i], i))
(l, r) = ([i - 1 for i in range(n)], [i + 1 for i in range(n)])
(l[0], r[-1]) = (-1, -1)
c = 1
while h:
	(ai, i) = heapq.heappop(h)
	if ans[i]:
		continue
	ans[i] = c
	j = i
	for _ in range(k):
		if l[j] == -1:
			break
		if ans[l[j]]:
			break
		j = l[j]
		ans[j] = c
	l0 = l[j]
	j = i
	for _ in range(k):
		if r[j] == -1:
			break
		if ans[r[j]]:
			break
		j = r[j]
		ans[j] = c
	r0 = r[j]
	if not r0 == -1:
		l[r0] = l0
	if not l0 == -1:
		r[l0] = r0
	c ^= 3
print(''.join(map(str, ans)))
