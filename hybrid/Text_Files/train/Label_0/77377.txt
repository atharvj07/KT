(a, b) = map(int, input().split())
(x, y, z, X, Y, Z) = (0, 0, 0, 0, 0, 0)
while a != 0 and a % 2 == 0:
	a = a / 2
	x += 1
while a != 0 and a % 3 == 0:
	a = a / 3
	y += 1
while a != 0 and a % 5 == 0:
	a = a / 5
	z += 1
while b != 0 and b % 2 == 0:
	b = b / 2
	X += 1
while b != 0 and b % 3 == 0:
	b = b / 3
	Y += 1
while b != 0 and b % 5 == 0:
	b = b / 5
	Z += 1
if a != b:
	print(-1)
else:
	print(max(X, x) - min(X, x) + max(y, Y) - min(y, Y) + max(z, Z) - min(z, Z))
