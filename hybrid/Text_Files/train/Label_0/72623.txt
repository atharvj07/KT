import math
import itertools
from collections import defaultdict
getInputList = lambda : list(input().split())
getInputIntList = lambda : list(map(int, input().split()))
(n, m) = getInputIntList()
s = 0
arr = []
for i in range(n):
	(a, b) = getInputIntList()
	s += a
	arr.append(a - b)
step = 0
arr.sort()
while s > m and len(arr) > 0:
	s -= arr.pop()
	step += 1
if s <= m:
	print(step)
else:
	print(-1)
