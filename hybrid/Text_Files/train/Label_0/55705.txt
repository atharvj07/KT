
import java.io.File;
import java.io.PrintStream;
import java.util.Map.Entry;
import java.util.Scanner;
import java.util.TreeMap;

public class A {
    
    PrintStream cout = new PrintStream(System.out);
    Scanner cin;
    final int MAX_LIMIT = 100 * 1000 + 100;
    
    int[] arr = new int[100];
    int[] div = new int[1111];

    void solve() {
        int cnt = cin.nextInt();
        int n = 0;
        
        for(int i = 0; i < cnt; i++){
            arr[i] = cin.nextInt();
            n += arr[i];
        }
        
        if(cnt == 1){
            cout.println(n);
            for(int i = 0; i < n; i++) cout.print('a');
            return;
        }
        
        
        int divnum = 0;
        //get even divisors of n
        for(int i = 1; i < n; i++){
            if(n%i == 0){
                div[divnum++] = i;
            }
        }
        
        int ans = 0;
        for(int i = divnum - 1; i >= 0; i--){
            int j;
            int nonzero = 0;
            if(div[i]%2==0){
                for(j = 0; j < cnt; j++){
                    if(arr[j]%div[i] != 0) break;
                }
            }
            else{
                for(j = 0; j < cnt; j++){
                    if(arr[j]%(2*div[i]) != 0){
                        nonzero++;
                        if(arr[j]%(2*div[i]) != div[i]){
                            break;
                        }
                    }
                }
            }
            if(j == cnt){//this is our answer
                if(div[i]%2 == 1 && nonzero != 1) continue;
                ans = div[i];
                break;
            }
        }
        
        cout.println(ans);
        if(ans == 0){
            for(int i = 0; i < cnt; i++){
                for(int j = 0; j < arr[i]; j++){
                    cout.print((char)('a' + i));
                }
            }
            return;
        }
        
        StringBuilder str = new StringBuilder("");
        char extra = ' ';
        for(int i = 0; i < cnt; i++){
            int limit = arr[i]/ans;
            if(ans%2 == 1){
                limit = arr[i]/(2*ans);
                if(arr[i]%(2*ans) == ans){
                    extra = (char)('a' + i);
                }
            }
            for(int j = 0; j < limit; j++){
                str.append((char)('a' + i));
            }
        }
        
        String s1 = str.toString();
        String s2 = str.reverse().toString();
        int sw = 1;
        int bound = ans;
        if(ans%2 == 1) bound *= 2;
        for(int i = 0; i < bound; i++){
            if(sw == 1) {
                cout.print(s1);
                if(ans%2 == 1) cout.print(extra);
            }
            else{
                cout.print(s2);
            }
            sw = 1 - sw;
        }
    }

    void run(String[] args) throws Exception {

        boolean is_local = args.length > 0 && args[0].equals("MASHREGHI");
        if (is_local) {//this is local host
            File inputFile = new File(System.getProperty("user.dir")
                    + "\\src\\in.in");
            cin = new Scanner(inputFile);
            cout.print(System.getProperty("user.dir") + "\\src\\"
                    + this.getClass().getName() + ".java\n"); //this is to be used when submitting code
        } else {
            cin = new Scanner(System.in);
        }

        solve();

        if (is_local) {
            cout.println();
        }
    }

    public static void main(String[] args) throws Exception {
        new A().run(args);
    }

}
