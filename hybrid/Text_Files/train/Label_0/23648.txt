using namespace std;
typedef long long ll;
typedef pair<ll,ll> pll;
const ll inf=1e12;
const double eps=1e-9;

int main(void)
{
    map<pll,ll> Foxmap;
    int N;
    cin >> N;
    ll ans=0;
    vector<pll> xy(N);
    for(int i=0;i<N;i++)
    {
        ll x,y,w;
        cin >> x >> y>> w;
        Foxmap[pll(x,y)]=w;
        xy[i]=pll(x,y);
    }
    for(int i=0;i<N;i++)
    {
        ll s1=0,s2=0,s3=0,s4=0;
        //各点を中心として,4つの1*1の正方形にして考えると...
        //左上
        s1=Foxmap[pll(xy[i].first,xy[i].second)]+Foxmap[pll(xy[i].first-1LL,xy[i].second)]+Foxmap[pll(xy[i].first-1LL,xy[i].second-1LL)]+Foxmap[pll(xy[i].first,xy[i].second-1LL)];
        //左下
        s2=Foxmap[pll(xy[i].first,xy[i].second)]+Foxmap[pll(xy[i].first-1LL,xy[i].second)]+Foxmap[pll(xy[i].first-1LL,xy[i].second+1LL)]+Foxmap[pll(xy[i].first,xy[i].second+1LL)];
        //右下
        s3=Foxmap[pll(xy[i].first,xy[i].second)]+Foxmap[pll(xy[i].first+1LL,xy[i].second)]+Foxmap[pll(xy[i].first+1LL,xy[i].second+1LL)]+Foxmap[pll(xy[i].first,xy[i].second+1LL)];
        //右上
        s4=Foxmap[pll(xy[i].first,xy[i].second)]+Foxmap[pll(xy[i].first+1LL,xy[i].second)]+Foxmap[pll(xy[i].first+1LL,xy[i].second-1LL)]+Foxmap[pll(xy[i].first,xy[i].second-1LL)];
        //cout << s1 << ' ' << s2 << ' ' << s3 << ' ' << s4 << endl;
        ans=max(ans,s1);
        ans=max(ans,s2);
        ans=max(ans,s3);
        ans=max(ans,s4);
    }
    cout << ans << ' ' << '/' << ' ' << 1 << endl;
}
