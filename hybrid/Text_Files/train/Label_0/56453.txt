import java.util.Scanner;

public class Main {

    private static Scanner in = new Scanner(System.in);

    public static void main(String[] args) {
        int n = in.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
        }
        long ans = 0;
        int m = in.nextInt();
        int f = in.nextInt();
        for (int i = 0; i < n; i++) {
            ans += (x[i] - 1L + f) / (m + f) * f;
        }
        System.out.println(ans);
    }
}
