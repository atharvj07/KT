from collections import defaultdict
import sys
import threading
ans = 100000.0

def dfs1(i, p):
	c = 1
	for j in tree[i]:
		if j != p:
			c += dfs1(j, i)
	dic[p, i] = c
	dic[i, p] = n - c
	return c

def dfs2(i, p):
	global ans
	tmp = dic[i, p]
	for j in tree[i]:
		if j != p:
			dfs2(j, i)
			tmp = max(tmp, dic[i, j])
	ans = min(ans, tmp)

def dfs3(i, p):
	tmp = dic[i, p]
	for j in tree[i]:
		if j != p:
			dfs3(j, i)
			tmp = max(tmp, dic[i, j])
	if ans == tmp:
		res.append(i)

def solve():
	global ans, tree, dic, n, res
	t = int(input())
	for _ in range(t):
		n = int(input())
		ans = 100000.0
		res = []
		tree = [[] for _ in range(n + 1)]
		dic = defaultdict()
		for _ in range(n - 1):
			(x, y) = map(int, input().split())
			tree[x].append(y)
			tree[y].append(x)
		dfs1(1, 0)
		dfs2(1, 0)
		dfs3(1, 0)
		if len(res) == 1:
			a = res[0]
			b = tree[a][0]
			print(a, b)
			print(a, b)
		else:
			(a, b) = (res[0], res[1])
			c = 0
			for j in tree[a]:
				if j != b:
					c = j
					break
			print(a, c)
			print(b, c)
res = []
tree = []
dic = []
n = 0
max_recur_size = 10 ** 5 * 2 + 1000
max_stack_size = max_recur_size * 500
sys.setrecursionlimit(max_recur_size)
threading.stack_size(max_stack_size)
thread = threading.Thread(target=solve)
thread.start()
