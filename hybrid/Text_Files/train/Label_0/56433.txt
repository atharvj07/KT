

import java.util.*;
import java.lang.*;
import java.io.*;
public class Ideone1
{
	public static void main (String[] args) throws java.lang.Exception
	{
	 Scanner in = new Scanner(System.in);
	 String st = in.next();
	 char s[] = st.toCharArray();
	 int len = st.length();
	 long dcn=0;
	 long ans=0;
	 long mod = 1000000000+7;
	 for(int i=len-1;i>=0;i--)
	 {
	 	if(s[i]=='b')
	 	  {
	 	  	 dcn++;
	 	  }
	 	  else
	 	  {
	 	  	ans+=dcn;
	 	  	ans%=mod;
	 	  	dcn*=2;
	 	  	dcn%=mod;
	 	  }
	 	  
	 }
	 System.out.println(ans);
	}
}
