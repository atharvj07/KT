#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        string s;
        cin>>s;
        int c=0,flag=0;
        for(int i=0;i<s.size()-1;i++)
        {
            if(s[i]=='1' && s[i+1]=='1')c=1;
            if(c==1 && s[i]=='0' && s[i+1]=='0' )flag=1;
        }
        if(c && flag)cout<<"NO"<<endl;
        else cout<<"YES"<<endl;




    }
}
