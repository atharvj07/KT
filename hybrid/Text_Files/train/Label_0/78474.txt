from math import gcd

def b(n, g):
	s = bin(n).replace('0b', '')
	while len(s) < g:
		s = '0' + s
	return s

def main():
	for _ in range(int(input())):
		n = int(input())
		a = list(map(int, input().split()))
		jjk = 0
		if max(a) == min(a) and min(a) == 0:
			for i in range(n):
				print(i + 1, end=' ')
			print()
			continue
		for i in range(n):
			jjk = max(len(bin(a[i])) - 2, jjk)
		j = []
		for i in range(n):
			j += [b(a[i], jjk)]
		k = {}
		for i in range(n):
			for jk in range(jjk):
				if j[i][jk] == '1':
					if jk in k:
						k[jk] += 1
					else:
						k[jk] = 1
		jh = 0
		for i in k:
			jh = gcd(jh, k[i])
		l = []
		for i in range(1, jh + 1):
			if jh % i == 0:
				print(i, end=' ')
		print()
main()
