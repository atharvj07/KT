n = int(input())
s = list(input())
i = 0
s.append('_')
lengm = 0
c = 0
while i < n + 1:
	leng = 0
	if 'A' <= s[i] <= 'Z' or 'a' <= s[i] <= 'z':
		while 'A' <= s[i] <= 'Z' or 'a' <= s[i] <= 'z':
			leng += 1
			i += 1
		lengm = max(lengm, leng)
	if s[i] == '(':
		while s[i] != ')':
			if 'A' <= s[i] <= 'Z' or 'a' <= s[i] <= 'z':
				c += 1
				while 'A' <= s[i] <= 'Z' or 'a' <= s[i] <= 'z':
					i += 1
			else:
				i += 1
	i += 1
print(lengm, c)
