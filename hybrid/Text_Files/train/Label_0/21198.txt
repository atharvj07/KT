#include<cstdio>

#define N 100001

using namespace std;

char s[4][N<<1],ans[N<<2];
char bl[4];

int main()
{
	int T,n,m,a,b,ss,l1,l2,t;
	scanf("%d",&T);
	while(T--)
	{
		scanf("%d",&n);
		for(int i=1;i<=3;++i) scanf("%s",s[i]+1);
		m=n<<1;
		for(int i=1;i<=3;++i)
		{
			ss=0;
			for(int j=1;j<=m;++j) 
				if(s[i][j]=='1') ++ss;
			if(ss>=n)  bl[i]='1';
			else bl[i]='0';
		}
		if(bl[1]==bl[2]) a=1,b=2;
		else if(bl[1]==bl[3]) a=1,b=3;
		else a=2,b=3;
		l1=l2=1;
		t=1;
		while(l1<=m && l2<=m)
		{
			if(s[a][l1]==s[b][l2])
			{
				ans[t++]=s[a][l1];
				l1++;
				l2++;
			}
			else if(s[a][l1]!=bl[a]) ans[t++]=s[a][l1++];
			else ans[t++]=s[b][l2++];
		}
		while(l1<=m) ans[t++]=s[a][l1++];
		while(l2<=m) ans[t++]=s[b][l2++];
		ans[t]='\0';
		printf("%s\n",ans+1); 
	}
}
