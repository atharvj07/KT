#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<long long> v(n);
  long long a1 = 0, a2 = 0, a3 = 0;
  for (long long i = 0; i < n; i++) {
    cin >> v[i];
    if (v[i] == 1)
      a1++;
    else if (v[i] == 2)
      a2++;
    else
      a3++;
  }
  long long a = max(a1, max(a2, a3));
  cout << (a1 + a2 + a3) - a;
  return 0;
}
