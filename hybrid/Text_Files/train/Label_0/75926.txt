(n, k) = map(int, input().split())
l = [[1, 2 * n]]
a = 2
for i in range(n - 1):
	l.append([a, a + 1])
	a += 2
x = 2 * n - 1 + n - 1
y = x - 2 * k
c = 0
ans = []
for i in range(n):
	if c <= y:
		ans.extend(l[i][::-1])
		c += l[i][1] - l[i][0]
	else:
		ans.extend(l[i])
		c -= l[i][1] - l[i][0]
print(*ans)
