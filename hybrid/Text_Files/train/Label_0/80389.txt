(n, k) = map(int, input().split())
a = list(map(int, input().split()))
c = [a.index(max(a))]
d = [a[c[-1]]]
for i in range(1, k):
	a[c[-1]] = -1
	c.append(a.index(max(a)))
	d.append(a[c[-1]])
c.sort()
print(sum(d))
if k > 1:
	print(c[1], end=' ')
	for i in range(1, len(c) - 1):
		print(c[i + 1] - c[i], end=' ')
	print(n - c[-1])
else:
	print(n)
