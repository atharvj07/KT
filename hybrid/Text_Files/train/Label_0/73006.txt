import itertools
from math import floor, log, log2, ceil, sqrt
from itertools import groupby
from decimal import Decimal, getcontext
import sys
t = int(input())
res = []
for i in range(t):
	[k, x] = list(map(int, input().split()))
	ans = 0
	half = k * (k + 1) // 2
	if x >= k ** 2:
		ans = 2 * k - 1
	elif x == half:
		ans = k
	elif x < half:
		ans = ceil((Decimal(1 + 8 * x).sqrt() - 1) / 2)
	else:
		ans = 2 * k - floor((Decimal(1 + 8 * (k ** 2 - x)).sqrt() + 1) / 2)
	res.append(ans)
for i in range(t):
	print(res[i])
