from collections import defaultdict, Counter
T = int(input())
for _ in range(T):
	global lst
	(N, M) = map(int, input().split())
	visit = set()
	oddNodes = []
	directed = []
	G = defaultdict(list)
	oriG = defaultdict(list)
	Gra = [[0] * (N + 1) for _ in range(N + 1)]
	deg = [0] * (N + 1)
	for k in range(M):
		(u, v) = map(int, input().split())
		Gra[u][v] += 1
		Gra[v][u] += 1
		deg[u] += 1
		deg[v] += 1
	ans = 0
	for i in range(1, N + 1):
		if deg[i] % 2 == 0:
			ans += 1
		else:
			oddNodes.append(i)
	for i in range(0, len(oddNodes), 2):
		Gra[oddNodes[i]][oddNodes[i + 1]] += 1
		Gra[oddNodes[i + 1]][oddNodes[i]] += 1
		deg[oddNodes[i]] += 1
		deg[oddNodes[i + 1]] += 1

	def eulerPath(u):
		stk = [u]
		while stk:
			u = stk.pop()
			for i in range(1, N + 1):
				if Gra[u][i]:
					Gra[u][i] -= 1
					Gra[i][u] -= 1
					directed.append((u, i))
					stk.append(i)
					break
	for i in range(1, N + 1):
		eulerPath(i)
	for i in range(0, len(oddNodes), 2):
		Gra[oddNodes[i]][oddNodes[i + 1]] += 1
		Gra[oddNodes[i + 1]][oddNodes[i]] += 1
	print(ans)
	for (u, v) in directed:
		if Gra[u][v] != 0:
			Gra[u][v] -= 1
			Gra[v][u] -= 1
		else:
			print(str(u) + ' ' + str(v))
