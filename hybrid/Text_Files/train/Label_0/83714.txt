(x, y, z) = map(int, input().split())
(a, b, c) = map(int, input().split())
if x + y + z > a + b + c:
	print('NO')
	exit()
if x <= a:
	a -= x
	x = 0
else:
	print('NO')
	exit()
if z >= c:
	z -= c
	c = 0
else:
	c -= z
	z = 0
if z > 0:
	if a + b >= z + y:
		print('YES')
	else:
		print('NO')
	exit()
elif y <= a + b:
	print('YES')
else:
	print('NO')
