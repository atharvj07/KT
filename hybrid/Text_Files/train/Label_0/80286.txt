import sys
from array import array
import typing as Tp

def input():
	return sys.stdin.buffer.readline().decode('utf-8')

def main():
	(n, d, m) = map(int, input().split())
	(small, big) = ([], [])
	for x in map(int, input().split()):
		if x > m:
			big.append(x)
		else:
			small.append(x)
	small.sort(reverse=True)
	big.sort()
	if not big:
		print(sum(small))
		exit()
	n -= 1
	ans = big.pop()
	small.extend([0] * len(big))
	while len(small) >= d + 1 and big:
		x = sum((small.pop() for i in range(d + 1)))
		ans += max(x, big.pop())
	print(ans + sum(small))
main()
