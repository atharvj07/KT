#include <bits/stdc++.h>
using namespace std;
vector<int> adj[100000];
pair<long long, long long> modified_DFS(long long *v, long long start,
                                        long long parent) {
  pair<long long, long long> ans = make_pair(0, 0);
  for (auto f : adj[start]) {
    if (f != parent) {
      pair<long long, long long> res = modified_DFS(v, f, start);
      if (res.first > ans.first) ans.first = res.first;
      if (res.second > ans.second) ans.second = res.second;
    }
  }
  v[start - 1] += ans.first;
  v[start - 1] -= ans.second;
  if (v[start - 1] > 0) ans.second += v[start - 1];
  if (v[start - 1] < 0) ans.first += abs(v[start - 1]);
  return ans;
}
int main(int argc, char **argv) {
  long long n;
  cin >> n;
  for (int i = 0; i < n - 1; ++i) {
    long long a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  long long v[n];
  for (int i = 0; i < n; ++i) {
    cin >> v[i];
  }
  pair<long long, long long> ans = modified_DFS(v, 1, -1);
  cout << ans.first + ans.second << endl;
  return 0;
}
