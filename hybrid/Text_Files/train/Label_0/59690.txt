import java.util.*;

public class S{
    static int[][][] res = new int[21][21][21];
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
		int q = in.nextInt();
		int a[] = new int[105];
		int b[] = new int[105];
		int c[] = new int[105];
		int r[] = new int[105];
		for(int t = 0; t < q; t++){
			int n = in.nextInt();
			for(int i = 0; i < n; i++){
				a[i] = in.nextInt();
			}
			for(int i = 0; i < n; i++){
				b[i] = in.nextInt();
			}
			for(int i = 0; i < n; i++){
				c[i] = in.nextInt();
			}
			r[0] = a[0];
			for(int i = 1; i < n - 1; i++){
				if(a[i] != r[i - 1]){
					r[i] = a[i];
				}else if(b[i] != r[i - 1]){
					r[i] = b[i];
				}else{
					r[i] = c[i];
				}
			}
			if(a[n - 1] != r[0] && a[n - 1] != r[n - 2]){
				r[n - 1] = a[n - 1];
				// System.out.println("1");
			}else if(b[n - 1] != r[0] && b[n - 1] != r[n - 2]){
				r[n - 1] = b[n - 1];
				// System.out.println("2");
			}else{
				r[n - 1] = c[n - 1];
				// System.out.println("3");
			}

			for(int i = 0; i < n; i++){
				System.out.print(r[i] + " ");
			}
			System.out.println();
		}

    }


}
