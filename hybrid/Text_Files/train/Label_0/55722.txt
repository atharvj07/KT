import java.util.Scanner;

public class test{
	int numcom;
	int Anumnotreceive;
	int Anumreceive;
	int Bnumnotreceive;
	int Bnumreceive;
	String Alive;
	String Blive;
	class comand{
		comand(){};
		comand(int x, int y, int z){
			if(x == 1) {
				Anumreceive += y; 
				Anumnotreceive += z;
			}
			if(x==2) {
				Bnumreceive += y; 
				Bnumnotreceive += z;
			}
		}
	}
	void proverka(){
		if(Anumreceive >= Anumnotreceive) Alive = "LIVE"; 
		else Alive = "DEAD";
		if(Bnumreceive >= Bnumnotreceive) Blive = "LIVE"; 
		else Blive = "DEAD";
	}
	@SuppressWarnings("null")
	public static void main(String[] Args){
		Scanner sc = new Scanner(System.in);
		test t = new test();
		t.numcom = sc.nextInt();
		test.comand tc[] = new test.comand[t.numcom];
		for(int i = 0;i < t.numcom;i++){
			tc[i] = t.new comand(sc.nextInt(),sc.nextInt(),sc.nextInt());
		}
		t.proverka();
		System.out.println(t.Alive);
		System.out.println(t.Blive);
	}
}
