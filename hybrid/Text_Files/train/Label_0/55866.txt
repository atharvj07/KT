	public static long lcm (long num1,long num2) {
		long l=0;
		long max =Math.max(num1, num2);
		long min=Math.min(num1, num2);
		long range=num1*num2;
		for (long i=max;i<=range;i+=max) {
			if (i%min==0) {
				l= i;
				break;
			}
		}
		return l;
		
	}
