public class Ps {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(),m=in.nextInt(),h=in.nextInt();
        int[] a=new int[m],b=new int[n];
        for(int i =0;i<m;i++)
        {
            a[i]=in.nextInt();
        }
        for(int i =0;i<n;i++)
        {
            b[i]=in.nextInt();
        }
        int y;
        int[][] x=new int[n][m];
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                y=in.nextInt();
                if(y==1)
                {
                    x[i][j]=Math.min(a[j], b[i]);
                }
                else
                {x[i][j]=0;}
            }
        }
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                System.out.print(x[i][j]+" ");
            }
            System.out.println("");
        }
    }}
