n = int(input())
o = [i for i in range(1, n ** 2 + 1, 2)]
e = [i for i in range(2, n ** 2, 2)]
x = [[0 for i in range(n)] for j in range(n)]
for i in range(n):
	y = n // 2
	if y - i >= 0:
		for j in range(y - i, y + i + 1):
			x[i][j] = o.pop()
	else:
		for j in range(y + i - n + 1, y - i + n):
			x[i][j] = o.pop()
	for j in range(n):
		if x[i][j] == 0:
			x[i][j] = e.pop()
for i in range(n):
	print(*x[i])
