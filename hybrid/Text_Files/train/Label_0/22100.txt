#include <bits/stdc++.h>
using namespace std;
int n, m, k;
int x, y;
int vis[1003][1003];
int vis1[1003][1003];
int grid[1003][1003];
int res[1003][1003];
int solve(int x, int y) {
  if (vis[x][y]) return 0;
  vis[x][y]++;
  int ans = 0;
  (!grid[x - 1][y]) ? ans += solve(x - 1, y) : ans++;
  (!grid[x][y - 1]) ? ans += solve(x, y - 1) : ans++;
  (!grid[x + 1][y]) ? ans += solve(x + 1, y) : ans++;
  (!grid[x][y + 1]) ? ans += solve(x, y + 1) : ans++;
  return ans;
}
void solve1(int x, int y, int ress) {
  if (vis1[x][y] != 0) return;
  vis1[x][y]++;
  res[x][y] = ress;
  if (!grid[x - 1][y]) solve1(x - 1, y, ress);
  if (!grid[x][y - 1]) solve1(x, y - 1, ress);
  if (!grid[x + 1][y]) solve1(x + 1, y, ress);
  if (!grid[x][y + 1]) solve1(x, y + 1, ress);
  return;
}
int main() {
  memset(vis, 0, sizeof vis);
  memset(vis1, 0, sizeof vis1);
  memset(res, 0, sizeof res);
  cin >> n >> m >> k;
  for (int i = 1; i <= n; ++i) {
    string s;
    cin >> s;
    for (int j = 1; j <= m; ++j) {
      if (s[j - 1] == '*')
        grid[i][j] = 1;
      else
        grid[i][j] = 0;
    }
  }
  for (int i = 2; i < n; ++i) {
    for (int j = 2; j < m; ++j) {
      if (grid[i][j] || vis[i][j]) continue;
      res[i][j] = solve(i, j);
      solve1(i, j, res[i][j]);
    }
  }
  while (k--) {
    int x, y;
    scanf("%d%d", &x, &y);
    printf("%d\n", res[x][y]);
  }
}
