#include <bits/stdc++.h>
using namespace std;
int n, t[300];
int main() {
  scanf("%d", &n);
  for (int i = 0; i < (int)(2 * n - 1); i++) scanf("%d", &t[i]);
  vector<int> pos, neg;
  for (int i = 0; i < (int)(2 * n - 1); i++)
    if (t[i] < 0)
      neg.push_back(t[i]);
    else
      pos.push_back(t[i]);
  int ans = 0;
  if (neg.size() % 2 && n % 2 == 0) {
    for (int i = 0; i < (int)(neg.size()); i++) pos.push_back(-neg[i]);
    sort(pos.begin(), pos.end());
    for (int i = 0; i < (int)(pos.size()); i++) ans += i ? pos[i] : -pos[i];
  } else {
    for (int i = 0; i < (int)(neg.size()); i++) neg[i] *= -1;
    for (int i = 0; i < (int)(pos.size()); i++) ans += pos[i];
    for (int i = 0; i < (int)(neg.size()); i++) ans += neg[i];
  }
  printf("%d\n", ans);
}
