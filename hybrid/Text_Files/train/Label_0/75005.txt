def cint():
	return list(map(int, input().split()))

def cstr():
	return list(map(int, input().split(' ')))

def main():
	(n, k) = cint()
	lst = [0 for i in range(n)]
	for i in range(0, k):
		lst[i] = 'a'
		occ = 0
	for i in range(k, n):
		lst[i] = ['c', 'b', 'a'][occ]
		occ = (occ + 1) % 3
	print(''.join(lst))
t = int(input())
while t != 0:
	main()
	t -= 1
