import sys
sys.setrecursionlimit(10000)
n = int(input())
a = list(map(int, input().split()))

def solve(i, j, x):
	if i > j:
		return 0
	idx = a.index(min(a[i:j + 1]), i, j + 1)
	mn = a[idx]
	case = mn - x + solve(i, idx - 1, mn) + solve(idx + 1, j, mn)
	return min(case, j - i + 1)
print(solve(0, n - 1, 0))
