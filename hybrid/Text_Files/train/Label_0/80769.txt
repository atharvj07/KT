def task5(n, a, b):
	F = [None] * (n + 1)
	paths = [[] for i in range(n + 1)]
	F[n] = 0
	visited = set()
	current = [n]
	high = n
	while True:
		new_vertexes = []
		for i in current:
			for j in range(i - a[i], high + 1):
				k = j + b[j]
				if k not in visited:
					new_vertexes.append(k)
					visited.add(k)
				if F[k] is None:
					paths[k] = (i, j)
					F[k] = F[i] + 1
				elif F[i] + 1 < F[k]:
					paths[k] = (i, j)
					F[k] = F[i] + 1
			high = min(high, i - a[i])
		current = new_vertexes
		if not current:
			break
	path = []
	if F[0] is not None:
		current = paths[0]
		while current[0] != n:
			path.append(current[1])
			current = paths[current[0]]
		path.append(current[1])
	return (-1 if F[0] is None else F[0], path[::-1])

def main():
	n = int(input().strip())
	arr_a = [0] + list(map(int, input().strip().split()))
	arr_b = [0] + list(map(int, input().strip().split()))
	result = task5(n, arr_a, arr_b)
	print(result[0])
	if result[1]:
		print(*result[1], sep=' ')
main()
