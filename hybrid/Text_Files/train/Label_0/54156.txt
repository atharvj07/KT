

import java.util.*;

public class Main {

    public static void allSubstrings(String x) {

        Queue<String> q = new LinkedList<>();
        q.add(x);

        while (!q.isEmpty()) {

            String temp = q.remove();
            System.out.println(temp);

            if (temp.length() > 1) {
                for (int i = 0; i < temp.length(); i++) {
                    q.add(temp.substring(0, i) + temp.substring(i+1));
                }
            }
        }

    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();

        if (k == 1) {
            System.out.println(0);
            return;
        }

        int cost = 0;
        int num = 1;

        HashSet<String> set = new HashSet<>();

        Queue<String> q = new LinkedList<>();
        String initial = sc.next();
        q.add(initial);
        set.add(initial);

        while (!q.isEmpty()) {

            String temp = q.remove();

            for (int i = 0; i < temp.length(); i++) {

                String t = temp.substring(0, i) + temp.substring(i+1);
                if (!set.contains(t)) {
                    q.add(t);
                    set.add(t);
                    num += 1;
                    cost += n - t.length();
                }

                if (num == k) {
                    System.out.println(cost);
                    return;
                }
            }
        }

        System.out.println(-1);
    }
}
