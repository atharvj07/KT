inp = input()
(n, m) = [int(x) for x in inp.split(' ')]
inp = input()
a = [int(x) for x in inp.split(' ')]
ok = 1
perm = [0 for i in range(n + 1)]
used = [0 for i in range(n + 1)]
for i in range(m - 1):
	ai = a[i + 1] - a[i]
	if ai <= 0:
		ai += n
	if used[ai] and used[ai] != a[i]:
		ok = 0
	used[ai] = a[i]
	if perm[a[i]] and perm[a[i]] != ai:
		ok = 0
	perm[a[i]] = ai
j = 1
for i in range(1, n + 1):
	while j < n and used[j]:
		j += 1
	if perm[i] == 0:
		perm[i] = j
		used[j] = i
if ok == 0:
	print(-1)
else:
	print(' '.join([str(x) for x in perm[1:]]))
