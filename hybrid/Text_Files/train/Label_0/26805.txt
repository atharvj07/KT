#include <bits/stdc++.h>
using namespace std;
const int N = 501;
const int M = 12;
const int oo = 2e9;
int dp[N][M][N], n, k, m[N], cnt, dp1[N][M][N][M];
string s[N];
struct node {
  int nxt[10], cnt, h;
  node() {
    memset(nxt, -1, sizeof(nxt));
    cnt = h = 0;
  }
} nil;
vector<node> nd{nil};
struct trie_tree {
  void add(string s, int m) {
    int t1 = 0, t2;
    for (int i = 0; i < s.size(); i++) {
      t2 = nd[t1].nxt[s[i] - '0'];
      if (t2 == -1) {
        nd[t1].nxt[s[i] - '0'] = t2 = nd.size();
        nd.push_back(nil);
      }
      nd[t2].h = nd[t1].h + 1;
      t1 = t2;
    }
    nd[t1].cnt += m;
  }
} T;
int Dp(int x, int rem, int y) {
  int& res = dp[x][rem][y];
  if (res != -1) return res;
  res = (rem > 0) ? Dp(x, rem - 1, x) : oo;
  vector<int> ch;
  for (int i = 0; i <= 9; i++)
    if (nd[x].nxt[i] != -1) ch.push_back(nd[x].nxt[i]);
  dp1[x][rem][y][ch.size()] = 0;
  for (int i = (int)ch.size() - 1; i >= 0; i--)
    for (int j = 0; j <= rem; j++)
      dp1[x][rem][y][i] =
          min(dp1[x][rem][y][i], Dp(ch[i], j, y) + dp1[x][rem - j][y][i + 1]);
  res = min(res, dp1[x][rem][y][0] + nd[x].cnt * (nd[x].h - nd[y].h));
  return res;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> s[i] >> m[i];
  for (int i = 1; i <= n; i++) T.add(s[i], m[i]);
  memset(dp, -1, sizeof(dp));
  for (int i = 0; i < N; i++)
    for (int j = 0; j < M; j++)
      for (int i1 = 0; i1 < N; i1++)
        for (int j1 = 0; j1 < M; j1++) dp1[i][j][i1][j1] = oo;
  int ans = Dp(0, k, 0);
  cout << ans << '\n';
}
