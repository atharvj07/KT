chars = {'A': {'A'}, 'b': {'d'}, 'd': {'b'}, 'H': {'H'}, 'M': {'M'}, 'O': {'O'}, 'o': {'o'}, 'T': {'T'}, 'U': {'U'}, 'V': {'V'}, 'v': {'v'}, 'W': {'W'}, 'w': {'w'}, 'X': {'X'}, 'x': {'x'}, 'Y': {'Y'}, 'q': {'p'}, 'p': {'q'}, 'I': {'I'}}
not_clean = {'p', 'q', 'b', 'd'}
s = input()
ans = True
for i in range(len(s) // 2):
	if s[i] not in chars.keys() or s[len(s) - i - 1] not in chars[s[i]]:
		ans = False
		break
if len(s) % 2 != 0 and (s[len(s) // 2] not in chars.keys() or s[len(s) // 2] in not_clean):
	ans = False
if ans:
	print('TAK')
else:
	print('NIE')
