n = int(input())
a = [int(t) for t in input().split(' ')]
prefix_odd = [0]
prefix_even = [0]
for i in range(len(a)):
	if i % 2 == 0:
		prefix = prefix_even
		antiprefix = prefix_odd
	else:
		prefix = prefix_odd
		antiprefix = prefix_even
	prefix.append(prefix[-1] + a[i])
	antiprefix.append(antiprefix[-1])
count = 0
for i in range(len(a)):
	sum_even = prefix_even[i] + prefix_odd[-1] - prefix_odd[i + 1]
	sum_odd = prefix_odd[i] + prefix_even[-1] - prefix_even[i + 1]
	if sum_even == sum_odd:
		count += 1
print(count)
