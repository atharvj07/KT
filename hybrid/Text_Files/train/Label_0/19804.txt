#include <bits/stdc++.h>
using namespace std;
const int nmax = 100;
int aib[nmax + 1][nmax + 1];
inline int lsb(int x) { return x & -x; }
void update(int x, int y) {
  for (int i = x; i <= nmax; i += lsb(i)) {
    for (int j = y; j <= nmax; j += lsb(j)) {
      aib[i][j] = max(aib[i][j], x * y);
    }
  }
}
int query(int x, int y) {
  int ans = 0;
  for (int i = x; i > 0; i -= lsb(i)) {
    for (int j = y; j > 0; j -= lsb(j)) {
      ans = max(ans, aib[i][j]);
    }
  }
  return ans;
}
int main() {
  int n, a, b;
  cin >> n >> a >> b;
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    int x, y;
    cin >> x >> y;
    if (x <= a && y <= b) {
      int k = query(a - x, b);
      if (k > 0) {
        ans = max(ans, k + x * y);
      }
      k = query(a, b - y);
      if (k > 0) {
        ans = max(ans, k + x * y);
      }
    }
    if (y <= a && x <= b) {
      int k = query(a - y, b);
      if (k > 0) {
        ans = max(ans, k + x * y);
      }
      k = query(a, b - x);
      if (k > 0) {
        ans = max(ans, k + x * y);
      }
    }
    if (x <= a && y <= b) update(x, y);
    if (y <= a && x <= b) update(y, x);
  }
  cout << ans;
  return 0;
}
