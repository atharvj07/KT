#include <bits/stdc++.h>
const int N = 1000001;
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n;
  cin >> n;
  long long int a[n], x = 0;
  int fr[60];
  vector<int> ind[60];
  memset(fr, 0, sizeof fr);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    int msb = 0;
    for (int j = 0; j < 60; j++) {
      if (a[i] & (1ll << j)) {
        msb = j;
      }
    }
    ind[msb].push_back(i);
  }
  int ans[n];
  int vis[n];
  memset(vis, 0, sizeof vis);
  for (int i = 0; i < n; i++) {
    int cs = 0;
    for (int j = 0; j < 60; j++) {
      if (fr[j] == 0 && !ind[j].empty()) {
        ans[i] = ind[j].back();
        ind[j].pop_back();
        for (int k = 0; k < 60; k++) {
          if (a[ans[i]] & (1ll << k)) {
            fr[k] = 1 - fr[k];
          }
        }
        cs = 1;
        break;
      }
    }
    if (cs == 0) {
      cout << "No";
      return 0;
    }
  }
  cout << "Yes\n";
  for (int e : ans) cout << a[e] << ' ';
}
