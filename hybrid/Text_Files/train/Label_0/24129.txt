#include <bits/stdc++.h>
using namespace std;
void fun(vector<long long int> v, long long int n, long long int k,
         set<long long int> s) {
  long long int i, p, d = 1;
  long long int flag = 1;
  long long int sz = s.size();
  if (sz > k) {
    cout << "-1\n";
    return;
  }
  for (i = 0; i < n - k; i++) {
    if (v[i] != v[i + k]) {
      flag = 0;
      break;
    }
  }
  if (flag == 1) {
    cout << n << "\n";
    for (i = 0; i < n; i++) {
      cout << v[i] << " ";
    }
    cout << "\n";
    return;
  }
  set<long long int>::iterator it;
  long long int len = n * k;
  cout << len << "\n";
  while (s.size() < k) s.insert(d++);
  for (int i = 0; i < n; i++) {
    for (it = s.begin(); it != s.end(); it++) cout << (*it) << " ";
  }
  cout << "\n";
  return;
}
int main() {
  long long int t, n, k, i, j, p, d;
  cin >> t;
  while (t--) {
    cin >> n >> k;
    ;
    vector<long long int> v(n);
    set<long long int> s;
    for (i = 0; i < n; i++) {
      cin >> v[i];
      s.insert(v[i]);
    }
    fun(v, n, k, s);
    v.clear();
    s.clear();
  }
  return 0;
}
