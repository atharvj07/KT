#include <bits/stdc++.h>
using namespace std;
long long t[150500];
int arr[10];
bool comp(int i, int j) { return t[i] > t[j]; }
int main() {
  int n, k, q;
  cin >> n >> k >> q;
  map<int, int> mp;
  for (int i = 1; i <= n; i++) scanf("%lld", &t[i]);
  int sum = 0;
  for (int i = 1; i <= q; i++) {
    int type, id;
    cin >> type >> id;
    if (type == 1) {
      sort(arr, arr + sum, comp);
      if (sum < k) {
        mp[id] = 1;
        arr[sum++] = id;
      } else {
        if (t[id] < t[arr[sum - 1]]) continue;
        mp[arr[sum - 1]] = 0;
        arr[sum - 1] = id;
        mp[id] = 1;
      }
    } else {
      if (mp[id])
        cout << "YES" << endl;
      else
        cout << "NO" << endl;
    }
  }
  return 0;
}
