for _ in range(int(input())):
	N = int(input())
	L = list(map(int, input().split()))
	r = ['2' if i % 2 else '1' for i in range(N)]
	if len(set(L)) == 1:
		ans = [1] * N
		print(1)
		print(*ans)
	else:
		if N % 2 == 0:
			ans = [1, 2] * (N // 2)
			print(2)
			print(*ans)
		if N % 2 != 0:
			if L[N - 1] == L[0]:
				ans = [1, 2] * (N // 2)
				ans.append(1)
				print(2)
				print(*ans)
			else:
				FLAG = 0
				for i in range(N):
					if L[i] == L[i - 1]:
						FLAG = 1
						break
				if FLAG:
					r[i:] = r[i - 1:-1]
					print(2)
					print(*r)
				else:
					print(3)
					r[N - 1] = 3
					print(*r)
