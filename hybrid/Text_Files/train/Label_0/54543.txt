import java.io.*;
import java.util.*;

public class C {
    public static void main(String[] args) throws Exception {
        new C().solve();
    }

    void solve() throws IOException {
        BufferedReader in = new BufferedReader(new
                InputStreamReader(System.in));
        String[] sp;
        // Scanner sc = new Scanner(System.in);

        sp = in.readLine().split(" ");
        int n = Integer.parseInt(sp[0]);
        int m = Integer.parseInt(sp[1]);
        int[] a = new int[m];
        int[] b = new int[m];
        LinkedList<Long>[] lists = new LinkedList[n];
        HashMap<Integer, Boolean>[] dir = new HashMap[n];
        for (int i = 0; i < n; i++) {
            lists[i] = new LinkedList<Long>();
            dir[i] = new HashMap<Integer, Boolean>();
        }

        for (int i = 0; i < m; i++) {
            sp = in.readLine().split(" ");
            a[i] = Integer.parseInt(sp[0]) - 1;
            b[i] = Integer.parseInt(sp[1]) - 1;
            long v = Integer.parseInt(sp[2]);
            lists[a[i]].add((long) b[i] * Integer.MAX_VALUE + v);
            lists[b[i]].add((long) a[i] * Integer.MAX_VALUE + v);
        }

        long[] input = new long[n];
        for (long p : lists[0]) {
            input[0] += p % Integer.MAX_VALUE;
        }
        long[] all = new long[n];
        for (int i = 1; i < n - 1; i++) {
            for (long p : lists[i]) {
                all[i] += p % Integer.MAX_VALUE;
            }
        }

        LinkedList<Integer> Q = new LinkedList<Integer>();
        boolean[] done = new boolean[n];
        Q.add(0);

        while (Q.size() > 0) {
            int now = Q.poll();
            done[now] = true;
            for (long p : lists[now]) {
                int next = (int) (p / Integer.MAX_VALUE);
                long v = p % Integer.MAX_VALUE;
                if (!done[next]) {
                    input[next] += v;
                    if (input[next] == all[next] / 2) {
                        Q.add(next);
                    }
                    dir[now].put(next, true);
                    dir[next].put(now, false);
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            sb.append((dir[a[i]].get(b[i]) ? 0 : 1) + "\n");
        }
        System.out.print(sb);
    }
}
