#include <iostream>
#include <string>
#include <map>
#include <vector>
using namespace std;


int main()
{
    int N;
    while(cin>>N)
    {
        if(N==0) break;
        vector<int> minis(N,1000000);
        vector<int> maxis(N,0);
        for(int c=0;c<N;c++)
        {
            for(int c2=0;c2<N;c2++)
            {
                int z;
                cin>>z;
                minis[c]=min(minis[c],z);
                maxis[c2]=max(maxis[c2],z);
            }
        }
        for(int c=0;c<N;c++)
            for(int c2=0;c2<N;c2++)
                if(minis[c]==maxis[c2])
                {
                    cout<<minis[c]<<endl;
                    goto end;
                }
        cout<<0<<endl;
        end:;

    }
}
