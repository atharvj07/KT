#include <bits/stdc++.h>
using namespace std;
map<pair<long long, int>, pair<long long, int> > mp;
pair<long long, int> calc(long long n, int m) {
  if (n <= 0) return pair<long long, int>(0, n);
  if (n < 10) return pair<long long, int>(1, n - max((int)n, m));
  if (mp.find(pair<long long, int>(n, m)) != mp.end())
    return mp[pair<long long, int>(n, m)];
  long long temp = n, M = 1;
  while (temp) {
    temp /= 10;
    M *= 10;
  }
  M /= 10;
  long long a = n / M * M;
  pair<long long, int> ret = calc(n % M, max(m, (int)(n / M)));
  a += ret.second;
  temp = a;
  int tm = m;
  while (temp) {
    tm = max(tm, (int)(temp % 10));
    temp /= 10;
  }
  pair<long long, int> tret = calc(a - tm, m);
  ret = pair<long long, int>(ret.first + tret.first + 1, tret.second);
  return mp[pair<long long, int>(n, m)] = ret;
}
int main() {
  long long n;
  scanf("%I64d", &n);
  printf("%I64d\n", calc(n, 0).first);
  fclose(stdin);
  fclose(stdout);
  return 0;
}
