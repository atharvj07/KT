def profit(l1, r1, l2, r2):
	if r1 > r2:
		(r1, r2) = (r2, r1)
		(l1, l2) = (l2, l1)
	if l2 > r1:
		return r2 - l1
	elif l2 > l1:
		return r2 - l1 - r1 + l2
	else:
		return r2 - r1 + l1 - l2

def tax(l1, r1, l2, r2):
	if r1 > r2:
		(r1, r2) = (r2, r1)
		(l1, l2) = (l2, l1)
	if l2 > r1:
		return l2 - r1
	else:
		return 0

def overlap(l1, r1, l2, r2):
	if r1 > r2:
		(r1, r2) = (r2, r1)
		(l1, l2) = (l2, l1)
	if l2 > r1:
		return 0
	elif l2 > l1:
		return r1 - l2
	else:
		return r1 - l1
tt = int(input())
while tt:
	tt -= 1
	(n, k) = map(int, input().split())
	(l1, r1) = map(int, input().split())
	(l2, r2) = map(int, input().split())
	s = 0
	p = profit(l1, r1, l2, r2)
	t = tax(l1, r1, l2, r2)
	o = overlap(l1, r1, l2, r2)
	k -= n * o
	if k <= 0:
		print(s)
		continue
	s += p + t
	k -= p
	n -= 1
	if k <= 0:
		print(s + k)
		continue
	if k - p * n > 0:
		s += (p + t) * n
		s += 2 * (k - p * n)
		print(s)
		continue
	n = k // p
	if k - n * p > t:
		s += (p + t) * n
		s += k - n * p + t
		print(s)
		continue
	s += (p + t) * n
	s += 2 * (k - n * p)
	print(s)
