n = int(input())
l = list(map(int, input().split()))
y = 0
(c, b, ba) = (0, 0, 0)
if n % 3 == 0:
	k = n // 3
else:
	k = n // 3
	u = n % 3
	if u == 1:
		c = l[-1] + c
	elif u == 2:
		c = l[-2] + c
		b = l[-1] + b
for i in range(k):
	c = c + l[y]
	b = b + l[y + 1]
	ba = ba + l[y + 2]
	y = y + 3
k = max(c, b, ba)
if k == c:
	print('chest')
elif k == b:
	print('biceps')
else:
	print('back')
