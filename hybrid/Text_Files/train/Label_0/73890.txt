sq = []
i = 0
while i < 3:
	row = [int(x) for x in input().split()]
	sq.append(row)
	i += 1
eq1 = sum(sq[1]) - sum(sq[0])
eq2 = sum(sq[2])
sq[0][0] = (eq1 + eq2) // 2
sq[1][1] = sum(sq[0]) - sum(sq[1])
sq[2][2] = sum(sq[0]) - sum(sq[2])
for i in range(3):
	for j in range(3):
		if j == 2:
			print(sq[i][j])
		else:
			print(sq[i][j], end=' ')
