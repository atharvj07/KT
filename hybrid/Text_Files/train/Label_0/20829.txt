#include <bits/stdc++.h>
using namespace std;
map<string, int> ma;
int main() {
  ma["monday"] = 1 - 1;
  ma["tuesday"] = 2 - 1;
  ma["wednesday"] = 3 - 1;
  ma["thursday"] = 4 - 1;
  ma["friday"] = 5 - 1;
  ma["saturday"] = 6 - 1;
  ma["sunday"] = 7 - 1;
  string str[] = {
      "monday", "tuesday",  "wednesday", "thursday",
      "friday", "saturday", "sunday",
  };
  string a, b;
  cin >> a >> b;
  int aa = ma[a];
  if (a == b || str[(aa + 2) % 7] == b || str[(aa + 3) % 7] == b)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
