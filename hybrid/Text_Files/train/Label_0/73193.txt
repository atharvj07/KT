for ii in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	s = 'a' * a[0]
	c = ord('a')
	if a[0] == 0:
		s = 'a'
	print(s)
	for jj in range(n - 1):
		if a[jj + 1] <= a[jj]:
			s = s[:a[jj]]
			if a[jj] == 0:
				c += 1
				if c == 122:
					c = 97
				s = chr(c)
			print(s)
		else:
			c += 1
			if c == 122:
				c = 97
			s = s[:a[jj]] + chr(c) * (a[jj + 1] - a[jj])
			print(s)
	if a[-1] == 0:
		if c == 122:
			c = 97
		print(chr(c + 1))
	else:
		print(s[:a[-1]])
