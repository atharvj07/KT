def N():
	return int(input())

def NM():
	return map(int, input().split())

def L():
	return list(NM())

def LN(n):
	return [N() for i in range(n)]

def LL(n):
	return [L() for i in range(n)]
t = N()

def f():
	(n, k) = NM()
	l = []
	for i in range(1, int(n ** 0.5) + 2):
		if n % i == 0:
			l.append(i)
			l.append(n // i)
	ans = n
	for i in l:
		if i <= k:
			ans = min(ans, n // i)
	print(ans)
for i in range(t):
	f()
