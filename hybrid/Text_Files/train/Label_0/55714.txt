import java.io.*;
import java.util.*;

public class Main implements Runnable{
    public static void main(String[] args){
        //new Thread(null,new Main(),"Thread-1",1024*1024*10).start();
        new Main().run();
    }
    SegTree tree;
    private void solve() throws Exception {
        InputReader in = new InputReader(System.in);
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
        int n=in.nextInt(),q=in.nextInt();
        char[]s=in.next().toCharArray();
        int[]sum=new int[2*n-1];
        for(int i=1;i<2*n-1;i++){
            sum[i]=sum[i-1]+(s[i-1]=='('?1:-1);
        }
        tree=new SegTree(0,2*n-2);
        tree.build(sum);
        out.println(tree.query());
        for(int i=0;i<q;i++){
            int l=in.nextInt(),r=in.nextInt();
            if(l>r){
                int tmp=l;
                l=r;r=tmp;
            }
            if(s[l-1]!=s[r-1]){
                if(s[l-1]==')') {
                    tree.update(1,l,r-1,0,2*(n-1),2);
                    s[l-1]='(';s[r-1]=')';
                }else {
                    tree.update(1,l,r-1,0,2*(n-1),-2);
                    s[l-1]=')';s[r-1]='(';
                }
            }
            out.println(tree.query());
        }
        out.flush();
    }
    class SegTree{
        int[]max,min,ans,pre,suf,lazy;
        int start,end;
        SegTree(int s,int t){
            start=s;end=t;
            int len=t-s+1;
            max=new int[len<<2];
            min=new int[len<<2];
            lazy=new int[len<<2];
            ans=new int[len<<2];
            pre=new int[len<<2];
            suf=new int[len<<2];
            Arrays.fill(min,0x3f3f3f3f);
        }
        void update(int p,int l,int r,int s,int t,int v){
            if(l<=s&&r>=t){
                modify(p,v);
                return;
            }
            pushdown(p);
            int mid=(s+t)>>>1;
            if(l<=mid)update(p<<1,l,r,s,mid,v);
            if(r>mid)update(p<<1|1,l,r,mid+1,t,v);
            pushup(p);
        }
        int query(){
            return ans[1];
        }
        void pushdown(int p){
            if(lazy[p]!=0) {
                int l=p<<1,r=p<<1|1;
                modify(l,lazy[p]);
                modify(r,lazy[p]);
                lazy[p] = 0;
            }
        }
        void modify(int p,int v){
            max[p]+=v;
            min[p]+=v;
            suf[p]-=v;
            pre[p]-=v;
            lazy[p]+=v;
        }
        void pushup(int p){
            int ls=p<<1,rs=p<<1|1;
            ans[p]=Math.max(Math.max(pre[ls]+max[rs],suf[rs]+max[ls]),Math.max(ans[rs],ans[ls]));
            max[p]=Math.max(max[ls],max[rs]);
            min[p]=Math.min(min[ls],min[rs]);
            suf[p]=Math.max(Math.max(suf[ls],suf[rs]),max[rs]-2*min[ls]);
            pre[p]=Math.max(Math.max(pre[ls],pre[rs]),max[ls]-2*min[rs]);
        }
        int[]a;
        int id;
        void build(int[]arr){
            a=arr;
            id=0;
            dfs(1,start,end);
        }
        void dfs(int pos,int l,int r){
            if(l==r){
                max[pos]=a[id];
                min[pos]=a[id];
                suf[pos]=pre[pos]=-a[id];
                id++;
                return;
            }
            int mid=(l+r)>>>1;
            dfs(pos<<1,l,mid);
            dfs(pos<<1|1,mid+1,r);
            pushup(pos);
        }
    }
    @Override
    public void run() {
        try {
            solve();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
    void shuffle(int[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i),tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    void shuffle(Object[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i);
            Object tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    void shuffle(long[]a,int l,int r){
        Random rnd=new Random();
        for (int i=r; i>l+1; i--) {
            int j=rnd.nextInt(i);
            long tmp=a[j];
            a[j]=a[i-1];
            a[i-1]=tmp;
        }
    }
    class InputReader{
        StreamTokenizer tokenizer;
        public InputReader(InputStream stream){
            tokenizer=new StreamTokenizer(new BufferedReader(new InputStreamReader(stream)));
            tokenizer.ordinaryChars(33,126);
            tokenizer.wordChars(33,126);
        }
        public String next() throws IOException {
            tokenizer.nextToken();
            return tokenizer.sval;
        }
        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }
        public boolean hasNext() throws IOException {
            int res=tokenizer.nextToken();
            tokenizer.pushBack();
            return res!=tokenizer.TT_EOF;
        }
    }
}
