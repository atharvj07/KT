#include <bits/stdc++.h>
using namespace std;
int main() {
  int timer;
  cin >> timer;
  while (timer--) {
    long long n, i, j;
    cin >> n;
    vector<long long> vt;
    vt.push_back(1);
    i = 2;
    long long sum = 1;
    while (true) {
      sum += i;
      if (sum >= n) {
        vt.push_back(n - sum + i);
        break;
      } else {
        vt.push_back(i);
        i *= 2;
      }
    }
    sort(vt.begin(), vt.end());
    cout << vt.size() - 1 << "\n";
    for (i = 0; i < vt.size() - 1; ++i) cout << (vt[i + 1] - vt[i]) << " ";
    cout << "\n";
  }
}
