(n, q) = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))
for i in range(1, n):
	a[i] += a[i - 1]

def binary_search(curr):
	(l, r) = (0, n - 1)
	while l <= r:
		mid = l + r >> 1
		if a[mid] >= curr:
			r = mid - 1
		else:
			l = mid + 1
	return l
res = ''
curr = 0
for i in k:
	curr += i
	if curr >= a[n - 1]:
		res += str(n) + '\n'
		curr = 0
	else:
		it = binary_search(curr)
		res += str(n - it - (a[it] == curr)) + '\n'
print(res)
