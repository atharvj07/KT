for _ in range(int(input())):
	n = int(input())
	s = input()
	stack = [s[n - 1]]
	i = n - 2
	zero = None
	one = None
	for i in range(n):
		if s[i] == '1':
			one = i
			break
	for i in range(n - 1, -1, -1):
		if s[i] == '0':
			zero = i
			break
	if zero == None or one == None:
		print(s)
	elif zero < one:
		print(s)
	else:
		ans = ''
		for i in range(n):
			if i >= one and i < zero:
				continue
			else:
				ans += s[i]
		print(ans)
