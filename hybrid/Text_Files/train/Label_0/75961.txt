wwght = {'Q': 9, 'R': 5, 'B': 3, 'N': 3, 'P': 1}
bwght = {'q': 9, 'r': 5, 'b': 3, 'n': 3, 'p': 1}
white = black = 0
for i in range(8):
	s = input()
	for i in s:
		if i in wwght:
			white += wwght[i]
		elif i in bwght:
			black += bwght[i]
if white == black:
	print('Draw')
elif white > black:
	print('White')
else:
	print('Black')
