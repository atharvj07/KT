from collections import defaultdict
import bisect
for _ in range(int(input())):
	(d, total, k, current) = (defaultdict(list), 1, 0, -1)
	s = input()
	t = input()
	for i in range(len(s)):
		d[s[i]].append(i)
	while k != len(t):
		if len(d[t[k]]) == 0:
			total = -1
			break
		current = bisect.bisect_right(d[t[k]], current)
		if current >= len(d[t[k]]):
			current = -1
			total += 1
		else:
			current = d[t[k]][current]
			k += 1
	print(total)
