#include"bits/stdc++.h"
using namespace std;

#define ll long long
#define endl "\n"
#define fastio ios_base::sync_with_stdio(false)

int main(){
	fastio;

	int t;
	cin>>t;
	while(t--){
		ll n,m;
		cin>>n>>m;
		vector<pair<int,pair<char,int>>>odd,even;

		ll x[n+1],ans[n+1];
		for(int i=1;i<=n;i++) cin>>x[i];
		char c[n+1];
		for(int i=1;i<=n;i++) cin>>c[i];

		for(int i=1;i<=n;i++){
			ans[i]=-1;
			if(x[i]%2){
				odd.push_back({x[i],{c[i],i}});
			}else{
				even.push_back({x[i],{c[i],i}});
			}
		}
		sort(odd.begin(),odd.end());
		sort(even.begin(),even.end());


		stack<pair<int,pair<char,int>>>s;
		for(auto u:odd){
			if(!s.empty() && s.top().second.first=='R' && u.second.first=='L'){
				ll need=(u.first-s.top().first)/2;
				ans[s.top().second.second]=need,ans[u.second.second]=need;
				s.pop();
				continue;
			}
			s.push(u);
		}
		deque<pair<int,pair<char,int>>>aseL,aseR;
		while(!s.empty()){
			if(s.top().second.first=='L') aseL.push_front(s.top());
			else aseR.push_front(s.top());
			s.pop();
		}

		sort(aseL.begin(),aseL.end());
		sort(aseR.begin(),aseR.end());

		while(aseL.size()>1){
			ll need=aseL[0].first+(aseL[1].first-aseL[0].first)/2;
			ans[aseL[0].second.second]=need;
			ans[aseL[1].second.second]=need;
			aseL.pop_front();
			aseL.pop_front();
		}

		while(aseR.size()>1){
			ll sz=aseR.size();
			ll need=m-aseR[sz-1].first+(m-(aseR[sz-2].first+m-aseR[sz-1].first))/2;
			ans[aseR[sz-1].second.second]=need;
			ans[aseR[sz-2].second.second]=need;
			aseR.pop_back();
			aseR.pop_back();
		}

		if(!aseR.empty() && !aseL.empty()){
			ll need=aseL[0].first+m-aseR[0].first+m;
			need=need/2;
			ans[aseL[0].second.second]=need;
			ans[aseR[0].second.second]=need;
		}
		aseR.clear();
		aseL.clear();


		for(auto u:even){
			if(!s.empty() && s.top().second.first=='R' && u.second.first=='L'){
				ll need=(u.first-s.top().first)/2;
				ans[s.top().second.second]=need,ans[u.second.second]=need;
				s.pop();
				continue;
			}
			s.push(u);
		}
		//deque<pair<int,pair<char,int>>>aseL,aseR;
		while(!s.empty()){
			if(s.top().second.first=='L') aseL.push_front(s.top());
			else aseR.push_front(s.top());
			s.pop();
		}

		sort(aseL.begin(),aseL.end());
		sort(aseR.begin(),aseR.end());

		while(aseL.size()>1){
			ll need=aseL[0].first+(aseL[1].first-aseL[0].first)/2;
			ans[aseL[0].second.second]=need;
			ans[aseL[1].second.second]=need;
			aseL.pop_front();
			aseL.pop_front();
		}

		while(aseR.size()>1){
			ll sz=aseR.size();
			ll need=m-aseR[sz-1].first+(m-(aseR[sz-2].first+m-aseR[sz-1].first))/2;
			ans[aseR[sz-1].second.second]=need;
			ans[aseR[sz-2].second.second]=need;
			aseR.pop_back();
			aseR.pop_back();
		}

		if(!aseR.empty() && !aseL.empty()){
			ll need=aseL[0].first+m-aseR[0].first+m;
			need=need/2;
			ans[aseL[0].second.second]=need;
			ans[aseR[0].second.second]=need;
		}
		
		for(int i=1;i<=n;i++) cout<<ans[i]<<" ";
		cout<<endl;
	}
}
