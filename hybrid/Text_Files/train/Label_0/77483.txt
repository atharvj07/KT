n = int(input())
list = [int(i) for i in input().split()]
if n == 1 and list[0] == 0:
	print('UP')
	exit()
elif n == 1 and list[0] == 15:
	print('DOWN')
	exit()
elif n == 1:
	print('-1')
	exit()
(a, b) = (list[n - 2], list[n - 1])
if b == a + 1 and a >= 0 and (a <= 13) or (a == 1 and b == 0):
	print('UP')
elif a >= 2 and a <= 15 and (a == b + 1) or (a == 14 and b == 15):
	print('DOWN')
else:
	print('-1')
