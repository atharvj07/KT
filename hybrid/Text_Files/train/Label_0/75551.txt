day_in_month = [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

def month(x):
	cnt = 0
	for t in day_in_month:
		if t >= x:
			cnt += 1
	return cnt

def week(x):
	k = 51
	if x < 7:
		k += 1
	if x > 4:
		k += 1
	return k
a = input().strip()
spl = a.split()
if spl[2] == 'week':
	print(week(int(spl[0])))
if spl[2] == 'month':
	print(month(int(spl[0])))
