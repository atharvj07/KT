
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.ArrayList;
import java.util.Collections;

public class TyndexDotBrome {

    void run() {
        try {
            BufferedReader bfd = new BufferedReader(new InputStreamReader(
                    System.in));
            BufferedWriter out = new BufferedWriter(new OutputStreamWriter(
                    System.out));
            String sp[] = bfd.readLine().split(" +");
            int n = Integer.parseInt(sp[0]);
            int i , j; 
            long res;
            char[] uArr = bfd.readLine().toCharArray(), arr;

            @SuppressWarnings("unchecked")
            ArrayList<Integer>[] pos = new ArrayList[26];
            for(i=0; i<pos.length; ++i)
                pos[i] = new ArrayList<Integer>();
            for(i=0; i<uArr.length; ++i)
                pos[uArr[i]-'a'].add(i);
            for(i=0; i<n; ++i){
                arr = bfd.readLine().trim().toCharArray();
                res = 0;
                for(j=0; j<arr.length; ++j){
                    if(j<uArr.length && arr[j]==uArr[j]);
                    else {
                        if(pos[arr[j]-'a'].size()==0)
                            res += arr.length;
                        else {
                            int x = Collections.binarySearch(pos[arr[j]-'a'], j);
                            int insertPt = - x - 1;
                            if(insertPt!=0){
                                int plus = Integer.MAX_VALUE;
                                if(insertPt>0)
                                    plus = Math.min(plus, Math.abs(pos[arr[j]-'a'].get(insertPt-1)-j));
                                if(insertPt>=0 && insertPt<pos[arr[j]-'a'].size())
                                    plus = Math.min(plus, Math.abs(pos[arr[j]-'a'].get(insertPt)-j));
                                if(insertPt+1<pos[arr[j]-'a'].size())
                                    plus = Math.min(plus, Math.abs(pos[arr[j]-'a'].get(insertPt+1)-j));
                                res += plus;
                            } else {
                                res += Math.abs(pos[arr[j]-'a'].get(insertPt)-j);
                            }
                        }
                    }
                }
                out.write(res + "\n");
            }
            out.flush();
            out.close();
        } catch (Exception e) {
            
        }
    }
    
    public static void main(String[] args) {
        new TyndexDotBrome().run();
    }

}
