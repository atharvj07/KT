from operator import itemgetter
import sys
import bisect
input = sys.stdin.readline

class BIT:

	def __init__(self, n):
		self.n = n
		self.bit0 = [0] * (n + 1)
		self.bit1 = [0] * (n + 1)

	def _add(self, bit, i, val):
		i = i + 1
		while i <= self.n:
			bit[i] += val
			i += i & -i

	def _get(self, bit, i):
		s = 0
		while i > 0:
			s += bit[i]
			i -= i & -i
		return s

	def add(self, l, r, val):
		self._add(self.bit0, l, -val * l)
		self._add(self.bit0, r, val * r)
		self._add(self.bit1, l, val)
		self._add(self.bit1, r, -val)

	def get_sum(self, l, r):
		return self._get(self.bit0, r) + r * self._get(self.bit1, r) - self._get(self.bit0, l) - l * self._get(self.bit1, l)

def compress(array):
	n = len(array)
	m = len(array[0])
	set_ = set()
	for i in array:
		for j in i:
			set_.add(j)
	array2 = sorted(set_)
	memo = {value: index for (index, value) in enumerate(array2)}
	for i in range(n):
		for j in range(m):
			array[i][j] = memo[array[i][j]]
	return (array, len(memo))

bii = [v for (i, v) in bi]
bit = BIT(m)
tmp = 0

	ind = bisect.bisect_left(bii, l)

		ii = bi[tmp][0]
		bit.add(info[ii][2], info[ii][3] + 1, 1)
		tmp += 1
	if bit.get_sum(l2, r2 + 1) >= 1:
