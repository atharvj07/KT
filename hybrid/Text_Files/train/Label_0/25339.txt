#include<bits/stdc++.h>
using namespace std;
char s[333][333];int n;
int main(){
	scanf("%d",&n);
	for(int i=0; i<n; i++)
		scanf("%s",s[i]);
	int res = 0;
	for(int k=0; k<n; k++){
		bool flag = true;
		for(int i=0; i<n && flag; i++)
		for(int j=0; j<n && flag; j++){
			if(s[i][(j+k)%n] != s[j][(i+k)%n]){
				flag = false;
				break;
			}
		}
		if(flag)res += n;
	}
	printf("%d\n",res);
	return 0;
}
