import math
n = int(input())

def leap(n):
	if n % 4 == 0 and n % 100 != 0 or n % 400 == 0:
		return 2
	else:
		return 1

def years(n):
	extra = leap(n)
	total = 0
	for i in range(50):
		n += 1
		total += leap(n)
		total %= 7
		if total % 7 == 0 and leap(n) == extra:
			print(n)
			exit(0)
years(n)
