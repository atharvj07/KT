import java.io.*;
import java.util.*;

public class Main {

    public static ArrayList<Integer>[] tree = null;
    public static boolean[] visited = null;

    public static void main(String... args) throws IOException {
        MyScanner in = new MyScanner(System.in);

        int leafCount = in.nextInt();
        tree = new ArrayList[leafCount + 1];
        visited = new boolean[leafCount + 1];

        for (int i = 0; i < leafCount + 1; i++) {
            tree[i] = new ArrayList<>();
        }

        for (int i = 0; i < leafCount - 1; i++) {
            int from = in.nextInt();
            int to = in.nextInt();

            tree[from].add(to);
            tree[to].add(from);
        }

        int result = 0;
        visited[1] = true;

        for (int child : tree[1]) {
            result = Math.max(result, visit(child));
        }

        System.out.println(result);
    }

    public static int visit(int node) {
        LinkedList<Pair> queue = new LinkedList<>();
        ArrayList<Integer> depths = new ArrayList<>();

        queue.addFirst(new Pair(node, 1));
        visited[node] = true;

        while (!queue.isEmpty()) {
            Pair next = queue.removeLast();

            if (tree[next.node].size() == 1) {
                depths.add(next.depth);
            } else {
                for (int child : tree[next.node]) {
                    if (!visited[child]) {
                        queue.addFirst(new Pair(child, next.depth + 1));
                        visited[child] = true;
                    }
                }
            }
        }

        int result = 0;

        for (int entry : depths) {
            result = Math.max(result + 1, entry);
        }

        return result;
    }

    public static class Pair {
        public int node, depth;

        public Pair(int node, int depth) {
            this.node = node;
            this.depth = depth;
        }
    }

    public static class MyScanner {
        private BufferedReader reader;
        private StringTokenizer tokenizer;

        public MyScanner(InputStream in) {
            reader = new BufferedReader(new InputStreamReader(in), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreElements()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }

            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }
    }
}
