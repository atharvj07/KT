#include <bits/stdc++.h>
using namespace std;
int pos;
void simulate(int n, int x) {
  if (n == 0) {
    pos = x;
    return;
  }
  if (n % 2) {
    if (x == 1) {
      x = 0;
    } else if (x == 0) {
      x = 1;
    }
  } else {
    if (x == 1) {
      x = 2;
    } else if (x == 2) {
      x = 1;
    }
  }
  simulate(n - 1, x);
}
int main() {
  long long n, x;
  while (cin >> n >> x) {
    if (x == 2) {
      if (n % 2) {
        --n;
      }
    }
    if (x == 0) {
      if (n % 2 == 0) {
        --n;
      }
    }
    if (x == 1) {
      n %= 3;
    } else if (x == 0) {
      n %= 6;
    } else if (x == 2) {
      n %= 6;
    }
    simulate(n, x);
    cout << pos << endl;
  }
  return 0;
}
