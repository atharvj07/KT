#include <bits/stdc++.h>
using namespace std;
int n, x;
long long ans = 0;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> x;
  for (int i = 1; i * i <= x; i++) {
    if (x % i == 0 && x / i <= n && i <= n) {
      if (x / i == i)
        ans++;
      else
        ans += 2;
    }
  }
  cout << ans;
  return 0;
}
