q = list(map(int, input().strip().split()))
b = []
c = []
s = 0
x = 0
j = 0
for i in range(q[0]):
	b.append(list(map(int, input().strip().split())))
	c.append(b[i][0] - b[i][1])
	s = s + b[i][0]
c.sort(reverse=True)
while s > q[1] and x < q[0]:
	s = s - c[j]
	j = j + 1
	x = x + 1
if x == q[0] and s > q[1]:
	print(-1)
else:
	print(x)
