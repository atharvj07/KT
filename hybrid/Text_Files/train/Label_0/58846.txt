import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        FastScanner in = new FastScanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        new Main().run(in, out);
        out.close();
    }

    public static long mod = 17352642619633L;

    void run(FastScanner in, PrintWriter out) {

        int N = in.nextInt();
        long K = in.nextLong(); // number of visits

        int[][] a = new int[N][2];
        long sum = 0;
        for (int i = 0; i < N; i++) {
            a[i][0] = in.nextInt();
            a[i][1] = i+1;
            sum += a[i][0];
        }
        Arrays.sort(a, (p, q) -> {
            if (p[0] != q[0]) return p[0]-q[0];
            return p[1] - q[1];
        });

        if (sum < K) {
            out.println(-1);
            return;
        } else if (sum == K) {
            return;
        }

        long doneRotations = 0;
        for (int i = 0; i < N ; ) {

            // always reset to head
            // if can't reset, then we have our answer in next rotation

            long numElements = N-i;
            // can we make full rotations back to head to remove all elements of size a[i][0]?
            // val 1 = 1 rotation
            long diffRotations = a[i][0]-doneRotations;
            long numMoves = diffRotations * numElements;

            if (K-numMoves > 0) {
                doneRotations = a[i][0];

                int rot = a[i][0];
                while (i < N && a[i][0] == rot) i++;

                K -= numMoves;
            } else {
                // K <= numMoves
                // somewhere along the rotations we can't go any further

                // how many full rotations can we do?

                // diffRotations*numElements > K
                // n*numElements <= K
                // n <= K/numElements

                long numRot = K/numElements;
                numMoves = numRot * numElements;
                long carry = K - numMoves;

                // after numMoves, we are back to head
                TreeSet<int[]> ts = new TreeSet<>((p, q) -> p[1]-q[1]);

                while (i < N) {
                    if (a[i][0] > numRot + doneRotations) {
                        ts.add(a[i]);
                    }
                    i++;
                }

                LinkedList<int[]> ll = new LinkedList<>(ts);

                for (int j = 0; j < carry; j++) {
                    int[] first = ll.pollFirst();
                    if (first[0]-doneRotations != numRot+1) {
                        ll.offerLast(first);
                    }
                }

                for (int[] x : ll) out.print(x[1]+ " ");
                out.println();
            }
        }
    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        public FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            st = null;
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }
}
