import java.util.Scanner;

public class NewYearAndHurry {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt(), k = sc.nextInt();
		
		int m = 240 - k;
		int s = 0,i = 1;
		if(m == 0) {
			System.out.println(0);
			System.exit(0);
		}
		while(s + (5*i) <= m && i<=n) {
			s += 5*i;
			i++;
		}
		i--;
		if(i == n+1) i--;
		
		
		System.out.println(i);
	}

}
