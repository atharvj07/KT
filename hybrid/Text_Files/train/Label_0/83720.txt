(n, d) = (int(x) for x in input().split())
ms = []
for _ in range(n):
	ms.append(tuple((int(x) for x in input().split())))
ms.sort()
ans = 0
(start, end, ssum) = (0, 0, 0)
while end < n:
	end += 1
	ssum += ms[end - 1][1]
	while start < end and ms[end - 1][0] >= ms[start][0] + d:
		ssum -= ms[start][1]
		start += 1
	ans = max(ans, ssum)
print(ans)
