#include<bits/stdc++.h>
#define ll long long
#define pb push_back
#define lowbit(x) ((x)&(-(x)))
#define mid ((l+r)>>1)
#define lson rt<<1, l, mid
#define rson rt<<1|1, mid+1, r
#define fors(i, a, b) for(int i = (a); i < (b); ++i)
#define all(vec) vec.begin(),vec.end()
using namespace std;
const int maxn = 505;
char mp[maxn][maxn];
int n, m;
int main()
{
    int T;cin>>T;
    while(T--){
        scanf("%d%d", &n, &m);
        fors(i,1,n+1) scanf("%s",mp[i]+1);
        for(int i=1; i <= n; i += 3){
            fors(j,1,m+1) mp[i][j]='X';
        }
        for(int i = 3; i < n; i += 3){
            int x = -1, y = -1;
            fors(j,1,m+1){
                if(mp[i][j]=='X'){x = i; y = j; break;}
            }
            if(x!=-1){
                mp[x-1][y] = 'X'; continue;
            }
            fors(j,1,m+1){
                if(mp[i-1][j]== 'X'){x = i-1; y = j; break;}
            }
            if(x!=-1){
                mp[x+1][y] = 'X'; continue;
            }
            mp[i][1]=mp[i-1][1] = 'X';
        }
        if(n%3 == 0){
            fors(i,1,m+1) if(mp[n][i]=='X') mp[n-1][i] = 'X';
        }
        fors(i,1,n+1) printf("%s\n", mp[i]+1);
    }
	return 0;
}
