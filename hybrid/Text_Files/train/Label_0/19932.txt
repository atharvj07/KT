#include<stdio.h>
#include<algorithm>
using namespace std;
int dat[5000][5];
int m[5];
int main(){
	int a;
	while(scanf("%d",&a),a){
		for(int i=0;i<5000;i++)
			for(int j=0;j<5;j++)dat[i][j]=0;
		for(int i=0;i<5;i++)m[i]=0;
		int count=0;
		for(int p=0;p<a;p++){
			int b,c,d;
			scanf("%d%d%d",&b,&c,&d);d--;
			count+=c;
			if(b==2){
				for(int i=0;i<c;i++){
					dat[m[d]][d]=1;
					bool ok=true;
					for(int j=0;j<5;j++)if(!dat[m[d]][j])ok=false;
					if(ok){
						for(int j=0;j<5;j++){
							dat[m[d]][j]=0;
							for(int k=m[d];k<4999;k++)dat[k][j]=dat[k+1][j];
							for(int k=4999;k>=0;k--){
								if(dat[k][j]){
									m[j]=k+1;break;}
								if(k==0)m[j]=0;
							}
						}
						count-=5;
					}else m[d]++;
				//	m[d]++;
				}
				//m[d]+=c;
			}else{
				int h=0;
				for(int i=0;i<c;i++)h=max(h,m[i+d]);
				for(int i=0;i<c;i++)dat[h][i+d]=1;
				for(int i=0;i<c;i++)m[i+d]=h+1;
				bool ok=true;
				for(int j=0;j<5;j++)if(!dat[h][j])ok=false;
				if(ok){
					for(int j=0;j<5;j++){
						dat[h][j]=0;
						for(int k=h;k<4999;k++)dat[k][j]=dat[k+1][j];
						for(int k=4999;k>=0;k--){
							if(dat[k][j]){m[j]=k+1;break;}
							if(k==0)m[j]=0;
						}
					}
					count-=5;
				}
			}
	//		for(int i=6;i>=0;i--,printf("\n"))for(int j=0;j<5;j++)printf("%d",dat[i][j]);
	//		printf("\n");
		}
		printf("%d\n",count);
	}
}
