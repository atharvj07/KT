#include<stdio.h>
#define MOD 1000000007

int main(){
	int N;
	scanf("%d",&N);
	char S1[N+1],S2[N+1];
	scanf("%s%s",S1,S2);
	long long ans=1;
	int flag;
	
	int i;
	if(S1[0]==S2[0]){
		ans=3;
		i=1;
		flag=0;
	}
	else{
		ans=6;
		i=2;
		flag=1;
	}
	
	for(i;i<N;i++){
		if(S1[i]==S2[i]){
			if(flag==0)ans*=2;
			flag=0;
		}
		else{
			if(flag==0)ans*=2;
			else ans*=3;
			i++;
			flag=1;
		}
		ans%=MOD;
	}
	
	printf("%lld\n",ans);
	return 0;
}
