t = int(input())
for _ in range(t):
	(n, s) = map(str, input().split())
	small = [0 for i in range(int(n))]
	big = [0 for i in range(int(n))]
	n = int(n)
	ma = n
	p = s
	s = ' ' + s
	for i in range(n - 1, -1, -1):
		if s[i] == '<' or i == 0:
			big[i] = ma
			ma -= 1
			j = i + 1
			while j < n:
				if not big[j]:
					big[j] = ma
					ma -= 1
					j += 1
				else:
					break
	s1 = ''
	for i in range(n - 1 - 1, -1, -1):
		if s[i] == '<':
			s1 = s1 + '>'
		else:
			s1 += '<'
	small = big
	big = [0 for i in range(n)]
	ma = n
	big[-1] = 1
	s = p + ' '
	for i in range(0, n):
		if s[i] == '>' or i == n - 1:
			j = i - 1
			big[i] = ma
			ma -= 1
			while j >= 0:
				if not big[j]:
					big[j] = ma
					ma -= 1
					j -= 1
				else:
					break
	c = 0
	s = p
	for i in s:
		if i == '>':
			c += 1
	if c == n - 1:
		for i in range(1, n + 1):
			big[i - 1] = n - i + 1
		small = big
	c = 0
	for i in s:
		if i == '<':
			c += 1
	if c == n - 1:
		for i in range(n - 1, -1, -1):
			big[i] = i + 1
		small = big
	print(*big)
	print(*small)
