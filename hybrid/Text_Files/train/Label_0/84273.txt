for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	p = [1]
	curr = 0
	for i in range(1, n):
		if a[i] > a[i - 1]:
			curr += 1
		else:
			p.append(curr)
			curr = 1
	if curr > 0:
		p.append(curr)
	d = 1
	curr = 0
	prev = 1
	i = 1
	while i < len(p):
		if prev == 0:
			prev = curr
			curr = 0
			d += 1
		prev -= 1
		curr += p[i]
		i += 1
	print(d)
