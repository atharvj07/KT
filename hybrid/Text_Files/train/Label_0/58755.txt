public class Main
{
	static class dian
	{
		int x;
		int y;
		public dian(int x,int y)
		{
			this.x=x;
			this.y=y;
		}
	}
	public static void main(String[] args) throws IOException
	{
		Scanner sc=new Scanner(System.in);
		int n=ini();
		int m=ini();
		int dx=ini();
		int dy=ini();
		dian g[]=new dian[m+1];
		for (int i=1;i<=m;i++)
		{
			g[i]=new dian(ini(),ini());
		}
		int x[]=new int [n];
		int a=0,b=0;
		for (int i=1;i<n;i++)
		{
			a=(a+dx)%n;
			b=(b+dy)%n;
			x[a]=b;
		}
		int sz[]=new int [n+1];
		int max=0;
		int w=0;
		for (int i=1;i<=m;i++)
		{
			int q=(g[i].y-x[g[i].x]+n)%n;
			sz[q]++;
			if (max<sz[q])
			{
				max=sz[q];
				w=q;
			}
		}
		out.print(0+" "+w);
		
		
		out.flush();
	}

	static StreamTokenizer in=new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
	static PrintWriter out=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
	static int ini() throws IOException
	{
		in.nextToken();
		return (int)in.nval;
	}
	static String ins() throws IOException
	{
		in.nextToken();
		return in.sval;
	}
