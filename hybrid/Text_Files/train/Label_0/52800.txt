class Flag {
    public void solve(int testNumber, PandaScanner in, PrintWriter out) {
        int n = in.nextInt();
        int m = in.nextInt();
        int board[][] = new int[n][m];
        int freq[][][] = new int[2][26][n];
        for (int i = 0; i < n; i++) {
            char[] c = in.next().toCharArray();
            for (int j = 0; j < m; j++) {
                board[i][j] = c[j] - 'a';
                freq[j & 1][board[i][j]][i]++;
            }
        }
        int cost[][][] = new int[26][26][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 26; j++) {
                for (int k = 0; k < 26; k++) {
                    cost[j][k][i] = m - freq[0][j][i] - freq[1][k][i];
                }
            }
        }

//        ArrayDeque<int[]> good[][] = new ArrayDeque[26][26];
//        for (int i = 0; i < 26; i++) {

//                int[] arr = new int[] {i, j};
//                for (int k = 0; k < 26; k++) {

//                        if (k != l && i != k && j != l) {
//                            good[k][l].add(arr);
//                        }
//                    }
//                }
//            }
//        }
        int dp[][][] = new int[26][26][n + 1];
        int to[][][] = new int[26][26][n];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                Arrays.fill(dp[i][j], 100000000);
                dp[i][j][n] = 0;
            }
        }
        int hash[][] = new int[26][26];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                hash[i][j] = i * 26 + j;
            }
        }
        for (int idx = n - 1, idx2 = idx + 1; idx >= 0; idx--, idx2--) {
            for (int i = 0; i < 26; i++) {
                for (int j = 0; j < 26; j++) {
                    dp[i][j][idx2] += cost[i][j][idx];
                }
            }
            for (int i = 0; i < 26; i++) {
                for (int j = 0; j < i; j++) {
                    for (int k = 0; k < 26; k++) {
                        for (int l = 0; l < k; l++) {
                            if (i != k && j != l) {
                                if (dp[i][j][idx] > dp[k][l][idx2]) {
                                    dp[i][j][idx] = dp[k][l][idx2];
                                    to[i][j][idx] = hash[k][l];
                                }
                                if (dp[j][i][idx] > dp[l][k][idx2]) {
                                    dp[j][i][idx] = dp[l][k][idx2];
                                    to[j][i][idx] = hash[l][k];
                                }
                            }
                            if (i != l && j != k) {
                                if (dp[i][j][idx] > dp[l][k][idx2]) {
                                    dp[i][j][idx] = dp[l][k][idx2];
                                    to[i][j][idx] = hash[l][k];
                                }
                                if (dp[j][i][idx] > dp[k][l][idx2]) {
                                    dp[j][i][idx] = dp[k][l][idx2];
                                    to[j][i][idx] = hash[k][l];
                                }
                            }
                        }
                    }
                }
            }
        }
        int min = Integer.MAX_VALUE;
        int curr[] = new int[2];
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                if (min > dp[i][j][0]) {
                    min = dp[i][j][0];
                    curr[0] = i;
                    curr[1] = j;
                }
            }
        }
        out.println(min);
        StringBuilder res = new StringBuilder();
        for (int i = 0; i < n; i++) {
            int h = to[curr[0]][curr[1]][i];
            int[] a = new int[] { h / 26, h % 26 };
            for (int j = 0; j < m; j++) {
                res.append((char) (a[j & 1] + 'a'));
            }
            res.append("\n");
            curr = a;
        }
        out.print(res);
    }
}

class PandaScanner {
    public BufferedReader br;
    public StringTokenizer st;
    public InputStream in;

    public PandaScanner(InputStream in) {
        br = new BufferedReader(new InputStreamReader(this.in = in));
    }

    public String nextLine() {
        try {
            return br.readLine();
        }
        catch (Exception e) {
            return null;
        }
    }

    public String next() {
        if (st == null || !st.hasMoreTokens()) {
            st = new StringTokenizer(nextLine().trim());
            return next();
        }
        return st.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(next());
    }

    }
