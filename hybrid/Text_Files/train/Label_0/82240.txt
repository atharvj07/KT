from bisect import bisect_left as bl
(n, m) = map(int, input().split())
(pn, l) = ([], [])
q = 10 ** 5 + 4
k = [True for i in range(q + 2)]
for p in range(2, int(q ** 0.5) + 2):
	if k[p] == True:
		for i in range(p ** 2, q + 2, p):
			k[i] = False
for p in range(2, q + 1):
	if k[p]:
		pn.append(p)
for i in range(n):
	l.append(list(map(int, input().split())))

def f(l, q):
	for i in l:
		x = 0
		for j in i:
			x += pn[bl(pn, j)] - j
		q = min(q, x)
	return q
print(f(zip(*l), f(l, q)))
