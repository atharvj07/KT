def main():
	n = int(*input().split())
	L = [x for x in range(n + 1)]
	ans = []

	def swp(x, y):
		(L[x], L[y]) = (L[y], L[x])
	swp(1, n)
	ans.append((1, n))
	for i in range(2, (n + 1) // 2 + 1):
		swp(i, n)
		ans.append((i, n))
		if not n - i + 1 == i:
			swp(n - i + 1, 1)
			ans.append((n - i + 1, 1))
	val = sum(((i - j) * (i - j) for (i, j) in ans))
	print(val)
	for x in range(1, n + 1):
		print(L[x], end=' ')
	print('\n%d' % (n - 1,))
	for x in range(n - 2, -1, -1):
		print(ans[x][0], ans[x][1])
T = int(*input().split())
for ii in range(T):
	main()
