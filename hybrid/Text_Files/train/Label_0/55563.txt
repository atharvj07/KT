import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Scanner;
import java.util.StringTokenizer;

public class C {

	/**
	 * @param args
	 * @throws IOException
	 */
	public static void main(String[] args) throws IOException {
		// Scanner sc = new Scanner(System.in);
		// int n = sc.nextInt();
		BufferedReader rd = new BufferedReader(new InputStreamReader(System.in));
		// StringTokenizer t = new StringTokenizer(rd.readLine(), " ");
		int n = Integer.parseInt(rd.readLine());
		StringTokenizer t = new StringTokenizer(rd.readLine(), " ");
		BigInteger h[] = new BigInteger[n + 1];
		h[1] = new BigInteger(t.nextToken());
		for (int i = 2; i <= n; i++) {
			h[i] = new BigInteger(t.nextToken());
			h[i] = h[i].compareTo(h[i - 1]) == 1 ? h[i] : h[i - 1];
		}
		StringBuilder out = new StringBuilder();
		int m = Integer.parseInt(rd.readLine());
		int f = 1;
		for (int i = 0; i < m; i++) {
			t = new StringTokenizer(rd.readLine(), " ");
			int w_i = Integer.parseInt(t.nextToken());
			int h_i = Integer.parseInt(t.nextToken());
			int cu = h[w_i].compareTo(h[f]) == 1 ? w_i : f;
			out.append(h[cu] + "\n");
			// if(w_i>f){
			// out.append(b)
			// }
			h[w_i] = h[cu].add(BigInteger.valueOf(h_i));
			f = w_i;
			// h[f] = h[f].add();
		}
		System.out.print(out);
	}
}
