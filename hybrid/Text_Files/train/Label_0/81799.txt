T = int(input())
for t in range(T):
	(n, m, x, y) = map(int, input().split())
	lst = []
	for i in range(n):
		lst.append(list(input()))
	tot = 0
	if 2 * x < y:
		price = 2 * x
	else:
		price = y
	for i in range(n):
		for j in range(m - 1):
			if lst[i][j] == '.':
				if lst[i][j + 1] == '.':
					tot += price
					lst[i][j + 1] = '*'
				else:
					tot += x
		if lst[i][m - 1] == '.':
			tot += x
	print(tot)
