t = int(input())
while t > 0:
	t = t - 1
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	qwer = 0
	zxc = 0
	if n == 2 or m < n:
		print(-1)
	else:
		for i in range(n - 1):
			qwer += a[i] + a[i + 1]
			zxc += 1
		qwer += a[0] + a[n - 1]
		zxc += 1
		temp = [i for i in range(n)]
		xyz = [i[0] for i in sorted(enumerate(a), key=lambda x: x[1])]
		zxczxc = zxc
		while zxc < m:
			qwer += a[0] + a[1]
		print(qwer)
		for i in range(n - 1):
			print(i + 1, i + 2)
		print(n, 1)
		while zxczxc < m:
			print(temp[0] + 1, temp[1] + 1)
			zxczxc += 1
