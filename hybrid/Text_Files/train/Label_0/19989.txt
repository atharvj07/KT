#include <bits/stdc++.h>
using namespace std;
const int N = (int)1e5 + 7;
const int inf = (int)1e9 + 7;
struct fenwick {
  map<int, int> tree;
  void upd(int pos, int val) {
    for (; pos < N; pos += pos & -pos) {
      tree[pos] = max(tree[pos], val);
    }
  }
  int get(int r) {
    int res = 0;
    for (; r > 0; r -= r & -r) {
      res = max(res, tree[r]);
    }
    return res;
  }
};
int x[N], y[N], w[N];
fenwick tree[N];
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= m; i++) {
    scanf("%d %d %d", &x[i], &y[i], &w[i]);
  }
  int ans = 0;
  for (int i = m; i >= 1; i--) {
    int cur = tree[y[i]].get(N - w[i] - 1) + 1;
    tree[x[i]].upd(N - w[i], cur);
    ans = max(ans, cur);
  }
  printf("%d", ans);
}
