import java.util.Scanner;


//http://blog.csdn.net/glqac/article/details/44959819
public class TVGame {
	
	static int n;
    static String A;
    static String[][] dp;
	
    static long calc(String S, int start){
        long h=0, m=0;
        for(int i=0; i<S.length();i++){
            if(S.charAt(i) == 'H'){
                h = h*10 + (A.charAt(start + i)-'0');
            }else{
                m = m*10 + (A.charAt(start + i)-'0');
            }
        }
        return h+m;
    }
    
	static String solve(int ind, int hCnt){
		
		if(dp[ind][hCnt]!=null)  {
			return dp[ind][hCnt];
		}
		
		int rem = 2*n-ind;
		String ret = "";
		if(hCnt==n){
			for(int i=0; i<rem; i++) ret += 'M';
			return dp[ind][hCnt] = ret;
		} else if(ind-hCnt==n){
			for(int i=0; i<rem; i++) ret+= 'H';
			return dp[ind][hCnt] = ret;
		}
		
		String a = "H" + solve(ind+1, hCnt+1);
		String b = "M" + solve(ind+1, hCnt);
		long aa = calc(a, ind);
		long bb = calc(b, ind);
		
		return dp[ind][hCnt] = aa>bb? a : b;
	}
    
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		n = s.nextInt();
		A = s.next();
		dp = new String[2*n][n+1];
		System.out.println(solve(0, 0));
		s.close();

	}

}
