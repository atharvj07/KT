#include <bits/stdc++.h>
using namespace std;
const int MAXN = 300010;
const int MAXM = 300010;
int n, m;
vector<int> c[MAXN];
namespace Tree {
int head[MAXN], nxt[MAXN << 1], to[MAXN << 1], eidx = 0;
void init() { memset(head, -1, sizeof(head)); }
void adde(int u, int v) {
  to[eidx] = v, nxt[eidx] = head[u], head[u] = eidx++;
  to[eidx] = u, nxt[eidx] = head[v], head[v] = eidx++;
}
}  // namespace Tree
int ans[MAXM] = {0};
bool vis[MAXM] = {0};
int nxtc;
void init_nxtc() {
  nxtc = 1;
  while (vis[nxtc]) ++nxtc;
}
int get_nxtc() {
  int rtn = nxtc++;
  while (vis[nxtc]) ++nxtc;
  return rtn;
}
void dfs(int u, int fa) {
  using namespace Tree;
  for (int i = 0; i < (int)c[u].size(); ++i) vis[ans[c[u][i]]] = true;
  init_nxtc();
  for (int i = 0; i < (int)c[u].size(); ++i)
    if (!ans[c[u][i]]) ans[c[u][i]] = get_nxtc();
  for (int i = 0; i < (int)c[u].size(); ++i) vis[ans[c[u][i]]] = false;
  for (int i = head[u]; ~i; i = nxt[i]) {
    int v = to[i];
    if (v == fa) continue;
    dfs(v, u);
  }
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) {
    int s;
    scanf("%d", &s);
    while (s--) {
      int color;
      scanf("%d", &color);
      c[i].push_back(color);
    }
  }
  Tree::init();
  for (int i = 0; i < n - 1; ++i) {
    int u, v;
    scanf("%d%d", &u, &v);
    Tree::adde(u, v);
  }
  dfs(1, 0);
  for (int i = 1; i <= m; ++i)
    if (!ans[i]) ans[i] = 1;
  printf("%d\n", *max_element(ans + 1, ans + m + 1));
  for (int i = 1; i <= m; ++i) printf("%d ", ans[i]);
  return 0;
}
