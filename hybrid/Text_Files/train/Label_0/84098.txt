n = int(input())
a = map(int, input().split(' '))
least = float('inf')
most = float('-inf')
cache = {}
for ai in a:
	cache[ai] = cache[ai] + 1 if ai in cache else 1
	least = min(least, ai)
	most = max(most, ai)
del cache[least]
if least != most:
	del cache[most]
res = sum(cache.values())
print(res)
