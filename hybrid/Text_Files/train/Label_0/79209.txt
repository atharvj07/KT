inpdata = input().split()
n = int(inpdata[0])
a = int(inpdata[1])
b = int(inpdata[2])
c = int(inpdata[3])
d = int(inpdata[4])
ans = 0
for x in range(1, n + 1):
	y = x + b - c
	z = x + a - d
	w = a + y - d
	if 1 <= y and y <= n and (1 <= z) and (z <= n) and (1 <= w) and (w <= n):
		ans += 1
ans = ans * n
print(ans)
