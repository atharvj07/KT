n = int(input())
x = []
for i in range(n):
	(a, b) = [int(h) for h in input().split()]
	x.append([a, i + 1])
	x.append([b, i + 1])
x.sort()
start_wale = set()
end_wale = set()
if n == 1:
	print(1)
elif n >= 2:
	start = 0
	while True:
		if start >= n:
			break
		elif x[start][0] != x[start + 1][0]:
			break
		else:
			start_wale.add(x[start][1])
			start += 1
	start_wale.add(x[start][1])
	end = -1
	while True:
		if end <= -(n + 1):
			break
		elif x[end][0] != x[end - 1][0]:
			break
		else:
			end_wale.add(x[end][1])
			end -= 1
	end_wale.add(x[end][1])
	xx = start_wale.intersection(end_wale)
	if len(xx) == 0:
		print(-1)
	else:
		print(xx.pop())
