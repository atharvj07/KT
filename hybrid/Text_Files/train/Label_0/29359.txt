using namespace std;
#define rep(i,n) for (int i = 0; i < (n); i++)
#define repd(i,a,b) for (int i = (a); i < (b); i++)

void print(const vector<int> v)
{
	rep(i, v.size())
	{
		if (i) cout << " ";
		cout << v[i];
	}
	cout << endl;
}

int main(void)
{
	int n;
	cin >> n;
	vector<int> a;
	vector<int> b(n);
	vector<int> c(n);
	rep(i, n)
	{
		int in;
		cin >> in;
		a.push_back(in);
	}
	
	copy(a.begin(), a.end(), b.begin());
	copy(a.begin(), a.end(), c.begin());
	
	if (prev_permutation(a.begin(), a.end()))
	{
	    print(a);
	}
	
	print(b);
	
	if (next_permutation(c.begin(), c.end()))
	{
	    print(c);
	}
}
