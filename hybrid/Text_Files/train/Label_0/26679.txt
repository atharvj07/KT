#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int constexpr N = 1111;
void split(string& s, vector<string>& vc) {
  string tmp_str = "";
  for (auto& ch : s) {
    if (ch == '.') {
      vc.push_back(tmp_str);
      tmp_str = "";
    } else {
      tmp_str += ch;
    }
  }
  vc.push_back(tmp_str);
}
bool is_valid(string& x) { return (x.size() >= 2 && x.size() <= 11); }
int main() {
  string s;
  cin >> s;
  vector<string> vc;
  split(s, vc);
  auto sz = vc.size();
  if (vc[0].size() < 1 || vc[0].size() > 8) {
    cout << "NO\n";
    return 0;
  }
  if (vc[sz - 1].size() < 1 || vc[sz - 1].size() > 3 || vc.size() < 2) {
    cout << "NO\n";
    return 0;
  }
  for (int i = 1; i < vc.size() - 1; ++i) {
    if (!is_valid(vc[i])) {
      cout << "NO\n";
      return 0;
    }
  }
  cout << "YES\n";
  int ext_sz = 0;
  for (int i = 1; i < vc.size(); ++i) {
    cout << vc[i - 1].substr(ext_sz) << ".";
    ext_sz = vc[i].size();
    while (i + 1 < vc.size() && (ext_sz > 3 || vc[i].size() - ext_sz == 0)) {
      --ext_sz;
    }
    cout << vc[i].substr(0, ext_sz) << endl;
  }
}
