import operator
from operator import itemgetter
for _ in range(int(input())):
	n = int(input())
	a = []
	for i in range(n):
		(l, r) = map(int, input().split())
		a.append([l, r, i])
	a.sort(key=itemgetter(1))
	p = -1
	mn = a[n - 1][1]
	for i in range(n - 1, -1, -1):
		if mn > a[i][1]:
			p = i
		mn = min(mn, a[i][0])
	if p == -1:
		print(-1)
	else:
		ans = [-1] * n
		for i in range(n):
			ans[a[i][2]] = 1 + (i <= p)
		print(*ans)
