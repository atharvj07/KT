#include <bits/stdc++.h>
using namespace std;
const long long N = 2e5, sz = 666, inf = 1e9;
long long a[N], p[N], b[N];
vector<long long> pos;
set<long long> s;
signed main() {
  long long k, n, cnt = 0;
  cin >> k >> n;
  for (long long i = 1; i <= k; i++) {
    cin >> a[i];
    p[i] = p[i - 1] + a[i];
  }
  for (long long i = 0; i < n; i++) {
    cin >> b[i];
  }
  for (long long i = 1; i <= k; i++) {
    s.insert(b[0] - p[i]);
  }
  for (auto u : s) {
    pos.push_back(u);
  }
  for (auto u : pos) {
    long long cur = u, f = 1;
    s.clear();
    for (long long i = 1; i <= k; i++) {
      cur += a[i];
      s.insert(cur);
    }
    for (long long i = 0; i < n; i++) {
      if (s.end() == s.find(b[i])) f = 0;
    }
    cnt += f;
  }
  cout << cnt;
}
