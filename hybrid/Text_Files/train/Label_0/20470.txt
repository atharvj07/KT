#include <bits/stdc++.h>
using namespace std;
long long int nmax = 5 * 1e5;
vector<long long int> minprime(nmax + 1);
void sieve() {
  vector<bool> prime(nmax + 1, false);
  for (long long int i = 2; i <= nmax; i += 2) {
    minprime[i] = 2;
  }
  for (long long int i = 3; i <= nmax; i += 2) {
    if (!prime[i]) {
      minprime[i] = i;
      for (int j = i; j * i <= nmax; j += 2) {
        if (prime[i * j] == false) {
          prime[i * j] = true;
          minprime[i * j] = i;
        }
      }
    }
  }
}
struct cmp {
  bool operator()(pair<long long int, long long int> &p1,
                  pair<long long int, long long int> &p2) {
    if (p1.first != p2.first) return p1.first > p2.first;
    return p1.second > p2.second;
  }
};
void test_case() {
  int n;
  cin >> n;
  vector<int> ans;
  sieve();
  priority_queue<pair<long long int, long long int>,
                 vector<pair<long long int, long long int>>, cmp>
      pq;
  for (int i = 2; i <= n; i++) {
    pq.push(make_pair(i / minprime[i], i));
  }
  for (int i = 2; i <= n; i++) {
    ans.push_back(pq.top().first);
    pq.pop();
  }
  for (int x : ans) {
    cout << x << ' ';
  }
  cout << endl;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  while (t--) {
    test_case();
  }
  return 0;
}
