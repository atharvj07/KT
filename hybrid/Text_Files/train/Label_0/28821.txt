#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n, m, k, t, x, y;
  cin >> n >> m >> k >> t;
  vector<pair<int, int>> waste_coords;
  while (k--) {
    cin >> x >> y;
    waste_coords.push_back(make_pair(x, y));
  }
  sort(waste_coords.begin(), waste_coords.end());
  vector<string> crops = {"Carrots", "Kiwis", "Grapes"};
  while (t--) {
    cin >> x >> y;
    int total = (x - 1) * m + y;
    int wastes = 0;
    bool is_waste = false;
    for (int i = 0; i < waste_coords.size(); i++) {
      if (waste_coords[i].first == x && waste_coords[i].second == y) {
        is_waste = true;
        break;
      } else if (waste_coords[i].first < x ||
                 (waste_coords[i].first == x && waste_coords[i].second < y)) {
        wastes++;
      } else {
        break;
      }
    }
    if (is_waste) {
      cout << "Waste\n";
    } else
      cout << crops[(total - wastes - 1) % 3] << "\n";
  }
  return 0;
}
