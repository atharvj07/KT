#include <bits/stdc++.h>
using namespace std;
const int N = 120010;
int n, m, a[N];
long long ans[N];
vector<pair<int, int> > q[N];
int mn[N * 4], t[N * 4], lz[N * 4];
long long s[N * 4], tag[N * 4];
int s1[N], t1, s2[N], t2;
void pushup(int u) {
  t[u] = 0, mn[u] = min(mn[u * 2], mn[u * 2 + 1]);
  if (mn[u * 2] == mn[u]) t[u] += t[u * 2];
  if (mn[u * 2 + 1] == mn[u]) t[u] += t[u * 2 + 1];
  s[u] = s[u * 2] + s[u * 2 + 1];
}
void pushdown(int u) {
  if (lz[u]) {
    mn[u * 2] += lz[u], mn[u * 2 + 1] += lz[u];
    lz[u * 2] += lz[u], lz[u * 2 + 1] += lz[u];
    lz[u] = 0;
  }
  if (tag[u]) {
    if (mn[u * 2] == mn[u]) s[u * 2] += tag[u] * t[u * 2], tag[u * 2] += tag[u];
    if (mn[u * 2 + 1] == mn[u])
      s[u * 2 + 1] += tag[u] * t[u * 2 + 1], tag[u * 2 + 1] += tag[u];
    tag[u] = 0;
  }
}
void build(int u, int l, int r) {
  mn[u] = l, t[u] = 1;
  if (l >= r) return;
  int mid = (l + r) >> 1;
  build(u * 2, l, mid), build(u * 2 + 1, mid + 1, r);
}
void mdy(int u, int l, int r, int L, int R, int w) {
  if (L <= l && r <= R) {
    mn[u] += w, lz[u] += w;
    return;
  }
  int mid = (l + r) >> 1;
  pushdown(u);
  if (L <= mid) mdy(u * 2, l, mid, L, R, w);
  if (R > mid) mdy(u * 2 + 1, mid + 1, r, L, R, w);
  pushup(u);
}
long long ask(int u, int l, int r, int L, int R) {
  if (L <= l && r <= R) return s[u];
  int mid = (l + r) >> 1;
  pushdown(u);
  long long res = 0;
  if (L <= mid) res += ask(u * 2, l, mid, L, R);
  if (R > mid) res += ask(u * 2 + 1, mid + 1, r, L, R);
  return res;
}
int main() {
  scanf("%d", &n);
  for (int i = (1), ied = (n); i <= ied; i++) scanf("%d", &a[i]);
  scanf("%d", &m);
  for (int i = (1), ied = (m); i <= ied; i++) {
    int x, y;
    scanf("%d%d", &x, &y), q[y].push_back(make_pair(x, i));
  }
  build(1, 1, n);
  for (int i = (1), ied = (n); i <= ied; i++) {
    --mn[1], --lz[1];
    for (; t1 && a[s1[t1]] < a[i]; --t1)
      mdy(1, 1, n, s1[t1 - 1] + 1, s1[t1], a[i] - a[s1[t1]]);
    s1[++t1] = i;
    for (; t2 && a[s2[t2]] > a[i]; --t2)
      mdy(1, 1, n, s2[t2 - 1] + 1, s2[t2], a[s2[t2]] - a[i]);
    s2[++t2] = i;
    s[1] += t[1], ++tag[1];
    for (auto v : q[i]) ans[v.second] = ask(1, 1, n, v.first, i);
  }
  for (int i = (1), ied = (m); i <= ied; i++) printf("%I64d\n", ans[i]);
  return 0;
}
