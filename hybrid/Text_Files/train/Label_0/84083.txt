import math
a_sum = 0
n = int(input())
a = [0] + [int(i) for i in input().split()]
if n % 2 == 0 or n == 1:
	print(-1)
	exit()
else:
	i = n
	while i > 3:
		if a[i] >= a[i - 1]:
			if a[i] > 0:
				a_sum += a[i]
				a[(i - 1) // 2] = a[(i - 1) // 2] - a[i]
		elif a[i - 1] > 0:
			a_sum += a[i - 1]
			a[(i - 1) // 2] = a[(i - 1) // 2] - a[i - 1]
		a[n] = 0
		a[n - 1] = 0
		i -= 1
		i -= 1
	a_max = a[3]
	if a[2] > a[3]:
		a_max = a[2]
	if a[1] > a_max:
		a_max = a[1]
	if a_max > 0:
		a_sum += a_max
	print(a_sum)
	exit()
