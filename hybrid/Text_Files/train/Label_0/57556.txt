import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;

public class Main {

    void solve(Scanner in, PrintWriter out) {
        int t = in.nextInt();
        for (int i = 0; i < t; ++i) {
            int n = in.nextInt();
            long k = in.nextLong();
            Long xx[] = new Long[n];

            for (int j = 0; j < n; ++j) {
                xx[j] = in.nextLong();
            }

            for (int j = 0; j < n; ++j) {
                in.nextLong();
            }
            Arrays.sort(xx);

            long[] ll = new long[n];
            long[] rr = new long[n];
            int k1 = 0;
            long res = 1;

            for (int k2 = 0; k2 < n; ++k2) {
                while (xx[k2] - xx[k1] > k)
                    ++k1;
                ll[k2] = k2 - k1 + 1;
                if (k2 > 0)
                    ll[k2] = Math.max(ll[k2], ll[k2 - 1]);
            }

            k1 = n - 1;
            for (int k2 = n - 1; k2 >= 0; --k2) {
                while (xx[k1] - xx[k2] > k)
                    --k1;
                rr[k2] = k1 - k2 + 1;
                if (k2 < n - 1)
                    rr[k2] = Math.max(rr[k2], rr[k2 + 1]);
            }

            for (int m = 0; m < n - 1; ++m) {
                res = Math.max(res, rr[m + 1] + ll[m]);
            }
            out.println(res);
        }
    }

    public static void main(String[] args) {
        try(Scanner in = new Scanner(System.in);
            PrintWriter out = new PrintWriter(System.out)) {
            new Main().solve(in, out);
        }
    }
}
