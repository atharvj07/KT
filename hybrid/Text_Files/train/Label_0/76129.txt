import sys
from collections import deque
t = int(sys.stdin.readline())
for _ in range(t):
	empty = sys.stdin.readline()
	(n, k) = map(int, sys.stdin.readline().split())
	x = list(map(int, sys.stdin.readline().split()))
	conn = [[] for i in range(n + 1)]
	for i in range(n - 1):
		(v, u) = map(int, sys.stdin.readline().split())
		conn[v].append(u)
		conn[u].append(v)
	check = [n for i in range(n + 1)]
	q = deque()
	for friend in x:
		q.append(friend)
		check[friend] = 0
	while q:
		pos = q.popleft()
		for nextPos in conn[pos]:
			if check[nextPos] < n:
				continue
			q.append(nextPos)
			check[nextPos] = check[pos] + 1
	q = deque()
	q.append(1)
	visit = [n for i in range(n + 1)]
	visit[1] = 0
	answer = 'NO'
	while q:
		pos = q.popleft()
		if pos != 1 and len(conn[pos]) == 1:
			answer = 'YES'
			break
		for nextPos in conn[pos]:
			if visit[nextPos] < n:
				continue
			elif visit[pos] + 1 >= check[nextPos]:
				continue
			q.append(nextPos)
			visit[nextPos] = visit[pos] + 1
	print(answer)
