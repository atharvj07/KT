from math import *
sInt = lambda : int(input())
mInt = lambda : map(int, input().split())
lInt = lambda : list(map(int, input().split()))
t = sInt()
for _ in range(t):
	n = sInt()
	a = lInt()
	x = n - 1
	while a[x - 1] >= a[x] and x > 0:
		x -= 1
	while a[x - 1] <= a[x] and x > 0:
		x -= 1
	print(x)
