cases = int(input())
for t in range(cases):
	(n, k, z) = list(map(int, input().split()))
	a = list(map(int, input().split()))
	b = [0] * (n + 1)
	for i in range(n):
		b[i + 1] = b[i] + a[i]
	s = 0
	for i in range(1, k + 1):
		s1 = b[i + 1]
		left = k - i
		if left / 2 > z:
			s1 += z * a[i - 1] + z * a[i] + b[i + 1 + left - 2 * z] - b[i + 1]
		else:
			s1 += left // 2 * a[i] + (left - left // 2) * a[i - 1]
		s = max(s1, s)
	print(s)
