import math

def main():
	sums = dict()
	sums[0] = -1
	pos = dict()
	pos[-1] = 0
	n = int(input())
	a = list(map(int, input().split()))
	cur = 0
	st = -1
	sm = 0
	for i in range(n):
		cur += a[i]
		if not cur in sums:
			sums[cur] = i
			pos[i] = cur
			sm += i - st
		elif cur in sums:
			for k in range(st, sums[cur]):
				del sums[pos[k]]
				del pos[k]
			st = sums[cur] + 1
			sums[cur] = i
			pos[i] = cur
			sm += i - st
	print(sm)
t = 1
for i in range(t):
	main()
