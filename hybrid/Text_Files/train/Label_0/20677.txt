#include <bits/stdc++.h>
using namespace std;
int ansmin(vector<int> x) {
  vector<int> xn;
  if (x.size() == 1) {
    return x[0];
  } else {
    for (int i = 1; i < x.size(); ++ ++i) {
      cout << "? " << x[i - 1] << ' ' << x[i] << '\n';
      fflush(stdout);
      char c;
      cin >> c;
      if (c == '=')
        xn.push_back(x[i - 1]);
      else if (c == '<')
        xn.push_back(x[i - 1]);
      else
        xn.push_back(x[i]);
    }
    if (x.size() % 2 != 0) {
      cout << "? " << x.back() << ' ' << xn[0] << '\n';
      char c;
      cin >> c;
      if (c == '<') xn[0] = x.back();
    }
    return ansmin(xn);
  }
}
int ansmax(vector<int> x) {
  vector<int> xn;
  if (x.size() == 1) {
    return x[0];
  } else {
    for (int i = 1; i < x.size(); ++ ++i) {
      cout << "? " << x[i - 1] << ' ' << x[i] << '\n';
      fflush(stdout);
      char c;
      cin >> c;
      if (c == '=')
        xn.push_back(x[i - 1]);
      else if (c == '<')
        xn.push_back(x[i]);
      else
        xn.push_back(x[i - 1]);
    }
    if (x.size() % 2 != 0) {
      cout << "? " << x.back() << ' ' << xn[0] << '\n';
      char c;
      cin >> c;
      if (c == '>') xn[0] = x.back();
    }
    return ansmax(xn);
  }
}
void ans(vector<int> x) {
  vector<int> xl, xr, xt;
  for (int i = 1; i < x.size(); ++ ++i) {
    cout << "? " << x[i - 1] << ' ' << x[i] << '\n';
    fflush(stdout);
    char c;
    cin >> c;
    if (c == '=') {
      xl.push_back(x[i - 1]);
      xr.push_back(x[i]);
    } else if (c == '<') {
      xl.push_back(x[i - 1]);
      xr.push_back(x[i]);
    } else {
      xr.push_back(x[i - 1]);
      xl.push_back(x[i]);
    }
  }
  if (x.size() % 2 != 0) {
    cout << "? " << x.back() << ' ' << x[0] << '\n';
    char c;
    cin >> c;
    if (c == '<' || c == '==')
      xl.push_back(x.back());
    else
      xr.push_back(x.back());
  }
  cout << "! " << ansmin(xl) << ' ' << ansmax(xr) << '\n';
  fflush(stdout);
  return;
}
int main() {
  int t, n;
  cin >> t;
  for (int _ = 0; _ < t; _++) {
    cin >> n;
    if (n == 1) {
      cout << "! 1 1\n";
      fflush(stdout);
      continue;
    }
    vector<int> s(n);
    for (int i = 0; i < n; i++) {
      s[i] = i + 1;
    }
    ans(s);
  }
  return 0;
}
