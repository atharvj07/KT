void __print(unsigned x) { cout << x; }
void __print(unsigned long x) { cout << x; }
void __print(unsigned long long x) { cout << x; }
void __print(float x) { cout << x; }

void __print(char x) { cout << '\'' << x << '\''; }

void __print(const string &x) { cout << '\"' << x << '\"'; }
void __print(bool x) { cout << (x ? "true" : "false"); }
template <typename T, typename V>
void __print(const pair<T, V> &x) {
  cout << '{';
  __print(x.first);
  cout << ',';
  __print(x.second);
  cout << '}';
}
template <typename T>
void __print(const T &x) {
  int f = 0;
  cout << '{';
  for (auto &i : x) cout << (f++ ? "," : ""), __print(i);
  cout << "}";
}
void _print() { cout << "]\n"; }
template <typename T, typename... V>
void _print(T t, V... v) {
  __print(t);
  if (sizeof...(v)) cout << ", ";
  _print(v...);
}

const int d8i[8] = {-1, -1, 0, 1, 1, 1, 0, -1},
          d8j[8] = {0, 1, 1, 1, 0, -1, -1, -1};
void solve() {
  int n, q;
  cin >> n >> q;
  set<int> points;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    points.insert(x);
  }
  map<int, int> values;
  int prevv = *points.begin();
  auto it = next(points.begin());
  while (it != points.end()) {
    values[*it - prevv]++;
    prevv = *it;
    it++;
  }
  if (points.size() < 2)
    cout << "0\n";
  else
    cout << (*points.rbegin() - *points.begin() - values.rbegin()->first)
         << endl;
  auto add_hole = [&](int value) { values[value]++; };
  auto remove_hole = [&](int value) {
    values[value]--;
    if (!values[value]) values.erase(value);
  };
  while (q--) {
    int t, x;
    cin >> t >> x;
    if (t) {
      if (!points.empty()) {
        if (x < *points.begin()) {
          add_hole(*points.begin() - x);
        } else if (x > *points.rbegin()) {
          add_hole(x - *points.rbegin());
        } else {
          auto r = points.lower_bound(x);
          auto l = prev(r);
          remove_hole(*r - *l);
          add_hole(*r - x);
          add_hole(x - *l);
        }
      }
      points.insert(x);
    } else {
      if (points.size() == 1) {
        points.clear();
        values.clear();
      } else {
        if (x == *points.begin()) {
          auto it = next(points.begin());
          remove_hole(*it - x);
        } else if (x == *points.rbegin()) {
          auto it = prev(points.end());
          remove_hole(x - *prev(it));
        } else {
          auto it = points.find(x);
          auto l = prev(it);
          auto r = next(it);
          add_hole(*r - *l);
          remove_hole(*r - x);
          remove_hole(x - *l);
        }
        points.erase(x);
      }
    }
    if (points.size() < 2)
      cout << "0\n";
    else
      cout << (*points.rbegin() - *points.begin() - values.rbegin()->first)
           << endl;
  }
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  solve();
  return 0;
}
