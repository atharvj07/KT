#include <bits/stdc++.h>
using namespace std;
const int maxn = 4e6;
long long b1[maxn], b2[maxn], b3[maxn], b4[maxn], tot;
int n, m, q;
inline int Lowbit(int i) { return i & -i; }
void Add(long long *b1, int n, int i, long long x) {
  for (; i < n; i += Lowbit(i + 1)) {
    b1[i] += x;
  }
}
long long Sum(const long long *b1, int i) {
  long long s = 0;
  for (--i; i >= 0; i -= Lowbit(i + 1)) {
    s += b1[i];
  }
  return s;
}
void Add(long long *b1, long long *b2, int n, int l, int r, long long x) {
  Add(b2, n, l, x);
  Add(b2, n, r, -x);
  Add(b1, n, l, -l * x);
  Add(b1, n, r, r * x);
}
long long Sum(const long long *b1, const long long *b2, int l, int r) {
  long long s = Sum(b1, r) - Sum(b1, l);
  s += Sum(b2, r) * r;
  s -= Sum(b2, l) * l;
  return s;
}
int main(void) {
  scanf("%d%d%d", &n, &m, &q);
  while (q--) {
    int t, x1, y1, x2, y2;
    scanf("%d%d%d%d%d", &t, &x1, &y1, &x2, &y2);
    --x1;
    --y1;
    if (t == 0) {
      int w;
      scanf("%d", &w);
      tot += (long long)(x2 - x1) * (y2 - y1) * w;
      Add(b1, b2, n, x1, x2, (long long)(y2 - y1) * w);
      Add(b3, b4, m, y1, y2, (long long)(x2 - x1) * w);
    } else {
      long long ans = -tot;
      ans += Sum(b1, b2, x1, x2);
      ans += Sum(b3, b4, y1, y2);
      printf("%lld\n", ans);
    }
  }
  return 0;
}
