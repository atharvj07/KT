#include<bits/stdc++.h>
#define ll long long
#define mod 998244353
#define db double
using namespace std;
inline ll read(){
	ll res=0,f=1;char ch=getchar();
	while(!isdigit(ch)) {if(ch=='-') f=-f;ch=getchar();}
	while(isdigit(ch)) {res=(res<<1)+(res<<3)+(ch^48);ch=getchar();}
	return res*f;
}
const int N=1e6+5;
struct Vector{ll v;db t;}q[N];
ll t[N],v[N];
ll n,x,y,L;
ll l,r;
inline void merge(Vector &a,Vector b){
	a.t=(a.t*a.v+b.t*b.v)/(a.v+b.v);
	a.v+=b.v;
}
int main(){
	n=read();L=read();
	l=1,r=0;
	for(ll i=1;i<=n;i++){
		t[i]=read();v[i]=read();
		while(x+v[i]>L){
			ll del=min(q[l].v,x+v[i]-L);
			q[l].v-=del;
			x-=del,y-=del*q[l].t;
			if(!q[l].v) ++l;
		}
		q[++r].v=v[i];q[r].t=t[i];
		y+=v[i]*t[i],x+=v[i];
		while(l<r && q[r].t<q[r-1].t) merge(q[r-1],q[r]),--r;
        printf("%lf\n",(db)y/L);
	}
	return 0;
}
