#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, q, x, y, tot = 0;
  cin >> n >> q;
  vector<int> v[n + 1];
  for (int i = int(1); i <= int(q); i++) {
    cin >> x >> y;
    for (int j = int(x); j <= int(y); j++) v[j].push_back(i);
  }
  int a[q + 1][q + 1];
  for (int i = int(0); i <= int(q); i++)
    for (int j = int(0); j <= int(q); j++) a[i][j] = 0;
  for (int i = int(1); i <= int(n); i++) {
    if (v[i].size() == 0)
      tot++;
    else if (v[i].size() == 2) {
      a[v[i][0]][v[i][1]]++;
      a[v[i][1]][v[i][0]]++;
    } else if (v[i].size() == 1) {
      for (int j = int(1); j <= int(q); j++) {
        a[v[i][0]][j]++;
        a[j][v[i][0]]++;
      }
    }
  }
  int c = 1000000009;
  for (int i = int(1); i <= int(q); i++) {
    for (int j = int(1); j <= int(q); j++) {
      if (i != j) c = min(c, a[i][j]);
    }
  }
  cout << n - tot - c << endl;
}
