#include <bits/stdc++.h>
using namespace std;
int x[10], y[10];
int sp(int i1, int i2, int i3) {
  return (x[i2] - x[i1]) * (x[i3] - x[i2]) + (y[i2] - y[i1]) * (y[i3] - y[i2]);
}
int l(int i1, int i2) {
  return (x[i2] - x[i1]) * (x[i2] - x[i1]) + (y[i2] - y[i1]) * (y[i2] - y[i1]);
}
int main() {
  int a, b;
  bool sq[10];
  int cur[5];
  bool flag = false;
  bool issq;
  int i, j, i1, i2, i3, i4;
  for (i = 0; i < 8; i++) scanf("%d%d", &x[i], &y[i]);
  memset(sq, false, sizeof(sq));
  for (i1 = 0; i1 < 5; i1++) {
    sq[i1] = true;
    for (i2 = i1 + 1; i2 < 6; i2++) {
      sq[i2] = true;
      for (i3 = i2 + 1; i3 < 7; i3++) {
        sq[i3] = true;
        for (i4 = i3 + 1; i4 < 8; i4++) {
          sq[i4] = true;
          cur[1] = i1;
          cur[2] = i2;
          cur[3] = i3;
          cur[4] = i4;
          if (sp(i1, i2, i3)) {
            swap(cur[2], cur[3]);
            a = l(cur[1], cur[2]);
            if (l(cur[2], cur[3]) == a && l(cur[3], cur[4]) == a &&
                l(cur[4], cur[1]) == a && !sp(cur[1], cur[2], cur[3]) &&
                !sp(cur[2], cur[3], cur[4]) && !sp(cur[3], cur[4], cur[1]) &&
                !sp(cur[4], cur[1], cur[2]))
              issq = true;
            else {
              swap(cur[3], cur[4]);
              if (l(cur[2], cur[3]) == a && l(cur[3], cur[4]) == a &&
                  l(cur[4], cur[1]) == a && !sp(cur[1], cur[2], cur[3]) &&
                  !sp(cur[2], cur[3], cur[4]) && !sp(cur[3], cur[4], cur[1]) &&
                  !sp(cur[4], cur[1], cur[2]))
                issq = true;
              else {
                issq = false;
              }
            }
          } else {
            a = l(i1, i2);
            if (l(i2, i3) == a && l(i3, i4) == a && l(i4, i1) == a &&
                !sp(i2, i3, i4) && !sp(i3, i4, i1) && !sp(i4, i1, i2))
              issq = true;
            else
              issq = false;
          }
          if (issq) {
            j = 1;
            for (i = 0; i < 8; i++)
              if (!sq[i]) {
                cur[j] = i;
                j++;
              }
            if (sp(cur[1], cur[2], cur[3])) {
              swap(cur[2], cur[3]);
              a = l(cur[1], cur[2]);
              b = l(cur[2], cur[3]);
              if (l(cur[3], cur[4]) == a && l(cur[4], cur[1]) == b &&
                  !sp(cur[1], cur[2], cur[3]) && !sp(cur[2], cur[3], cur[4]) &&
                  !sp(cur[3], cur[4], cur[1]) && !sp(cur[4], cur[1], cur[2])) {
                flag = true;
                break;
              } else {
                swap(cur[3], cur[4]);
                a = l(cur[1], cur[2]);
                b = l(cur[2], cur[3]);
                if (l(cur[3], cur[4]) == a && l(cur[4], cur[1]) == b &&
                    !sp(cur[1], cur[2], cur[3]) &&
                    !sp(cur[2], cur[3], cur[4]) &&
                    !sp(cur[3], cur[4], cur[1]) &&
                    !sp(cur[4], cur[1], cur[2])) {
                  flag = true;
                  break;
                }
              }
            } else {
              a = l(cur[1], cur[2]);
              b = l(cur[2], cur[3]);
              if (l(cur[3], cur[4]) == a && l(cur[4], cur[1]) == b &&
                  !sp(cur[2], cur[3], cur[4]) && !sp(cur[3], cur[4], cur[1]) &&
                  !sp(cur[4], cur[1], cur[2])) {
                flag = true;
                break;
              }
            }
          }
          if (flag) break;
          sq[i4] = false;
        }
        if (flag) break;
        sq[i3] = false;
      }
      if (flag) break;
      sq[i2] = false;
    }
    if (flag) break;
    sq[i1] = false;
  }
  if (flag) {
    printf("YES\n");
    for (i = 0; i < 8; i++)
      if (sq[i]) printf("%d ", i + 1);
    printf("\n");
    for (i = 0; i < 8; i++)
      if (!sq[i]) printf("%d ", i + 1);
  } else
    printf("NO\n");
  return 0;
}
