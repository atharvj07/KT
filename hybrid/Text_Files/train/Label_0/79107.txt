from sys import stdin
input = stdin.readline
from collections import deque

def bfs(p):
	global notpossible
	q = deque([p])
	who[p] = 0
	(ans, total) = (0, 0)
	while len(q):
		p = q.popleft()
		ans += who[p]
		total += 1
		for x in adj[p]:
			(node, t) = x
			if who[node] != -1:
				if t == 1 and who[p] == who[node]:
					notpossible = True
				if t == 0 and who[p] != who[node]:
					notpossible = True
				continue
			if t == 1:
				who[node] = who[p] ^ 1
			else:
				who[node] = who[p]
			q.append(node)
	return max(ans, total - ans)

		if who[i] != -1:

		ans += bfs(i)

for T in range(int(input())):
	(n, m) = map(int, input().split())
	adj = [[] for i in range(n + 1)]
	for i in range(m):
		s = input().split()
		(u, v, x) = (int(s[0]), int(s[1]), s[2])
		if x == 'imposter':
			adj[u].append([v, 1])
			adj[v].append([u, 1])
		else:
			adj[u].append([v, 0])
			adj[v].append([u, 0])
	notpossible = False
	who = [-1] * (n + 1)
	print(answer())
