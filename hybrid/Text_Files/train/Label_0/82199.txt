arr = input().split(' ')
num = [int(x) for x in arr]
num_unique = list(set(num))
if len(num_unique) >= k:
	print('YES')
	ans = num_unique[:k]
	result = []
	for _ in ans:
		result.append(num.index(_) + 1)
	result.sort()
	for _ in result:
		print(_, end=' ')
else:
	print('NO')
