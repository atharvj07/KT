import java.util.*;
import java.lang.*;

public class remove_boxes{
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		long[] array = new long[n];
		scan.nextLine();
		String str = scan.nextLine();
		for(int i=0;i<n;i++){
			array[i] = scan.nextLong();
		}

		// System.out.println(str);
		// for(long v: array){
		// 	System.out.print(v+" ");
		// }
		// System.out.println();


		char[] strarray = str.toCharArray();
		Long[][][] dp = new Long[n][n][n];
		System.out.println(maximumFunc(dp, array, str, 0, n-1, 0));
	}


	public static long maximumFunc(Long[][][] dp, long[] values, String str, int i, int j, int k){
		if(i>j){
			return 0;
		}
		if(dp[i][j][k]!=null){
 			return dp[i][j][k];
		}

		int count = k+1;
		long res = (long)values[count-1] + maximumFunc(dp, values, str, i+1, j, 0);
	
		for(int m=i+1;m<=j;m++){
			if(str.charAt(i) == str.charAt(m)){
			   long temp = maximumFunc(dp, values, str, i+1, m-1, 0) + maximumFunc(dp, values, str, m, j, k+1);
			   res = Math.max(res, temp);
		    }
		}

		dp[i][j][k] = res;
		return res;

	}
}
