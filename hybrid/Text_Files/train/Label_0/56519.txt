//package TIM;


import java.io.*;
import java.lang.*;
import java.math.*;
import java.util.*;


public class Main {
    static Scanner input = new Scanner(System.in);
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static PrintWriter pw = new PrintWriter(System.out);

    public static void main(String[] args) throws IOException {
        int n = input.nextInt();
        int[] a = new int[n];
        int[] ans = new int[n];
        B[] b = new B[n];
        for (int i = 0; i < n; i++)
            a[i] = input.nextInt();
        for (int i = 0; i < n; i++) {
            b[i] = new B();
            b[i].val = input.nextInt();
            b[i].index = i;
        }
        Arrays.sort(a);
        Arrays.sort(b, 0, n, new cmp());
        for (int i = 0; i < n; i++)
            ans[b[i].index] = a[i];
        for (int i = 0; i < n; i++)
            System.out.print(ans[i] + " ");
    }

}

class B {
    int val, index;
}

class cmp implements Comparator<B> {
    public int compare(B x, B y) {
        return y.val - x.val;
    }
}
