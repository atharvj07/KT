#include <bits/stdc++.h>
using namespace std;
long long n, x, y, p;
void add(long long &x, long long y) { x = (x + y) % p; }
vector<vector<long long> > mul(const vector<vector<long long> > &a,
                               const vector<vector<long long> > &b) {
  int n = a.size();
  int m = a[0].size();
  int l = b[0].size();
  vector<vector<long long> > v(n, vector<long long>(l, 0));
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < l; ++j)
      for (int k = 0; k < m; ++k) add(v[i][j], 1ll * a[i][k] * b[k][j]);
  return v;
}
vector<vector<long long> > pw(const vector<vector<long long> > &a,
                              long long n) {
  if (n == 0) {
    vector<vector<long long> > m(a.size(), vector<long long>(a.size(), 0));
    for (int i = 0; i < a.size(); ++i) m[i][i] = 1;
    return m;
  }
  if (n == 1) return a;
  if (n & 1) return mul(pw(a, n - 1), a);
  vector<vector<long long> > b = pw(a, n >> 1);
  return mul(b, b);
}
long long pw(long long a, long long n) {
  if (n == 0) return 1;
  if (n == 1) return a % p;
  if (n & 1) return pw(a, n - 1) * a % p;
  long long b = pw(a, n >> 1);
  return b * b % p;
}
long long get_ans(long long sum, long long edges, long long minutes) {
  if (minutes == 0) return sum % p;
  long long curans = pw(3, minutes) * sum % p;
  vector<vector<long long> > matr(2, vector<long long>(2, 0));
  vector<vector<long long> > s(1, vector<long long>(2, 1));
  matr[0][0] = 1;
  matr[1][0] = matr[1][1] = 3;
  auto m = mul(s, pw(matr, minutes - 1));
  curans -= m[0][0] * edges % p;
  curans += 1ll * 1000000000 * 1000000000 / p * p;
  return curans % p;
}
long long fib(long long a, long long b, long long x) {
  vector<vector<long long> > matr(2, vector<long long>(2, 0));
  matr[1][0] = matr[1][1] = matr[0][1] = 1;
  vector<vector<long long> > s(1, vector<long long>(2, 1));
  s[0][0] = a;
  s[0][1] = b;
  return mul(s, pw(matr, x))[0][0] % p;
}
int main() {
  cin >> n >> x >> y >> p;
  vector<int> a(n);
  long long sum = 0;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    sum += a[i];
  }
  if (n == 1) return cout << a[0] % p, 0;
  long long edges1 = a[0] + a[n - 1];
  long long edges2 =
      (a[0] + fib(0, 1, x) * a[n - 2] + fib(0, 1, x + 1) * a[n - 1]) % p;
  long long ans1 = get_ans(sum % p, edges1 % p, x) % p;
  long long ans2 = get_ans(ans1 % p, edges2 % p, y) % p;
  cout << ans2 << endl;
}
