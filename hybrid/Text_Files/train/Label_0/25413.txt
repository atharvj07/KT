#include <bits/stdc++.h>
using namespace std;
int t;
long long l, r;
void solve() {
  long long a = 2 * l;
  if (a <= 2 * (r - l)) {
    cout << "NO" << '\n';
    return;
  }
  cout << "YES" << '\n';
}
int main() {
  cin >> t;
  for (int i = 1; i <= t; i++) {
    cin >> l >> r;
    solve();
  }
  return 0;
}
