#include <bits/stdc++.h>
using namespace std;
int h[300][300], i, j, n, m, k, H, a[300], b[300];
int main() {
  cin >> n >> m >> H;
  for (i = 0; i < m; i++) cin >> a[i];
  for (i = 0; i < n; i++) cin >> b[i];
  for (i = 0; i < n; i++)
    for (j = 0; j < m; j++) {
      cin >> k;
      if (!k) continue;
      h[i][j] = min(a[j], b[i]);
    }
  for (i = 0; i < n; i++, cout << endl)
    for (j = 0; j < m; j++) cout << h[i][j] << ' ';
}
