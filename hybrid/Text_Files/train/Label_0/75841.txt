import math

def change_char(s, p, r):
	return s[:p] + r + s[p + 1:]
(n, size) = map(int, input().split())
arr = list(map(int, input().split()))
freq_sum = []
i = 0
for element in arr:
	freq_sum.append(freq_sum[i - 1] + element if i > 0 else 0 + element)
	i += 1
ans = 200000 * 100
index = -1
for i in range(n - size + 1):
	ans2 = ans
	right = freq_sum[i + size - 1]
	left = freq_sum[i - 1] if i > 0 else 0
	ans = min(ans, right - left)
	index = index if ans == ans2 else i
print(index + 1)
