import math

def sieve_of_eratosthenes(n=100000):
	prime = [True for i in range(n + 1)]
	prime[0] = False
	prime[1] = False
	p = 2
	while p * p <= n:
		if prime[p] == True:
			for i in range(p * 2, n + 1, p):
				prime[i] = False
		p += 1
	return prime

def prime_factorization(n, k, prime):
	a = list()
	b = list()
	i = 2
	while n > 1:
		if prime[i]:
			count = 0
			while n % i == 0:
				count += 1
				n = n // i
			if count % k != 0:
				a.append((i, count % k))
				b.append((i, k - count % k))
		i += 1
	return [tuple(a), tuple(b)]

def prime_factorization2(n, k):
	a = list()
	b = list()
	count = 0
	while n % 2 == 0:
		count += 1
		n = n / 2
	if count > 0 and count % k != 0:
		a.append((2, count % k))
		b.append((2, k - count % k))
	for i in range(3, int(math.sqrt(n)) + 1, 2):
		count = 0
		while n % i == 0:
			count += 1
			n = n / i
		if count > 0 and count % k != 0:
			a.append((i, count % k))
			b.append((i, k - count % k))
	if n > 2:
		a.append((n, 1))
		b.append((n, k - 1))
	return [tuple(a), tuple(b)]

def solve(a, k):
	result = 0
	hashmap = {}
	for i in a:
		(x, y) = prime_factorization2(i, k)
		if y in hashmap:
			result += hashmap[y]
		if x in hashmap:
			hashmap[x] += 1
		else:
			hashmap[x] = 1
	return result
values = input().split()
(_, k) = (int(values[0]), int(values[1]))
a = list(map(int, input().split()))
print(solve(a, k))
