#include <bits/stdc++.h>
using namespace std;
const int N = 4e6;
struct edge {
  int next, to;
  long long w;
} a[N];
int head[N], cnt = 1, n, m, cur[N], dep[N], S, T;
int L[N], R[N], nod;
queue<int> Q;
map<int, int> Map[51];
long long Ans, inf = 1e14;
struct function {
  long long a, b, c;
  void rev() {
    a = -a;
    b = -b;
    c = -c;
  }
  long long f(long long x) { return (1ll << 25) + (a * x * x + b * x + c); }
} F[N];
void link(int x, int y, long long w) {
  a[++cnt] = (edge){head[x], y, w};
  head[x] = cnt;
  a[++cnt] = (edge){head[y], x, 0};
  head[y] = cnt;
}
int BFS() {
  memset(dep, 0, sizeof(dep));
  Q.push(S);
  dep[S] = 1;
  while (!Q.empty()) {
    int x = Q.front();
    Q.pop();
    for (int i = head[x]; i; i = a[i].next)
      if (!dep[a[i].to] && a[i].w) dep[a[i].to] = dep[x] + 1, Q.push(a[i].to);
  }
  return dep[T];
}
int DFS(int x, long long flow) {
  if (x == T) return flow;
  for (int &i = cur[x]; i; i = a[i].next) {
    int R = a[i].to;
    if (!a[i].w || dep[R] != dep[x] + 1) continue;
    long long k = DFS(R, min(flow, a[i].w));
    if (k) {
      a[i].w -= k, a[i ^ 1].w += k;
      return k;
    }
  }
  return 0;
}
int id(int p, int c) {
  int &v = Map[p][c];
  if (!v) v = ++nod;
  return v;
}
int main() {
  cin >> n >> m;
  S = ++nod;
  T = ++nod;
  for (int i = 1; i <= n; i++) cin >> F[i].a >> F[i].b >> F[i].c, F[i].rev();
  for (int i = 1; i <= n; i++) {
    cin >> L[i] >> R[i];
    link(S, id(i, L[i]), inf);
    for (int c = L[i]; c <= R[i]; c++) link(id(i, c), id(i, c + 1), F[i].f(c));
    link(id(i, R[i] + 1), T, inf);
  }
  for (int i = 1, u, v, d; i <= m; i++) {
    cin >> u >> v >> d;
    for (int c = L[v]; c <= R[v] + 1; c++)
      if (c + d >= L[u] && c + d <= R[u] + 1) link(id(u, c + d), id(v, c), inf);
  }
  Ans = n * (1ll << 25);
  while (BFS()) {
    for (int i = 1; i <= nod; i++) cur[i] = head[i];
    while (long long tmp = DFS(S, inf)) Ans -= tmp;
  }
  cout << Ans << endl;
}
