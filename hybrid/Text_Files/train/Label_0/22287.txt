#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-9;
const int MAXN = 5000005;
const int MOD = 1e9 + 7;
const int INF = (1 << 30);
const long long INFl = 1e18;
int n, m, s;
pair<int, int> bugs[MAXN];
pair<int, pair<int, int> > st[MAXN];
vector<int> ind;
int ans[MAXN];
int bSearch(int val, int r) {
  int l = 0;
  while (l < r) {
    int mid = (l + r) >> 1;
    if (val <= st[mid].first) r = mid;
    if (val > st[mid].first) l = mid + 1;
  }
  return l;
}
bool canFind(int t) {
  ind.clear();
  int right = n;
  long long sum = 0;
  priority_queue<pair<int, int>, vector<pair<int, int> >,
                 greater<pair<int, int> > >
      q;
  for (int i = m - 1; i >= 0; --i) {
    int in = bSearch(bugs[i].first, right);
    for (int j = in; j < right; ++j) {
      q.push(st[j].second);
    }
    right = in;
    if (q.empty()) return false;
    pair<int, int> z = q.top();
    q.pop();
    sum += z.first;
    ind.push_back(z.second);
    i -= t - 1;
  }
  if (sum > s) return false;
  return true;
}
int main() {
  scanf("%d%d%d", &n, &m, &s);
  for (size_t i = 0; i < m; ++i) {
    scanf("%d", &bugs[i].first);
    bugs[i].second = i + 1;
  }
  for (size_t i = 0; i < n; ++i) {
    scanf("%d", &st[i].first);
  }
  for (size_t i = 0; i < n; ++i) {
    scanf("%d", &st[i].second.first);
    st[i].second.second = i + 1;
  }
  sort(bugs, bugs + m);
  sort(st, st + n);
  int l = 1, r = m + 1;
  if (!canFind(r)) {
    puts("NO");
    return 0;
  }
  puts("YES");
  while (l < r) {
    int mid = (l + r) >> 1;
    if (canFind(mid))
      r = mid;
    else
      l = mid + 1;
  }
  canFind(l);
  int c = l, j = 0;
  for (size_t i = 0; i < m; ++i) {
    ans[bugs[m - i - 1].second - 1] = ind[j];
    --c;
    if (!c) {
      j++;
      c = l;
    }
  }
  for (int i = 0; i < m; ++i) {
    printf("%d ", ans[i]);
  }
  return 0;
}
