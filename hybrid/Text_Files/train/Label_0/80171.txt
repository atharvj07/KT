t = int(input())
for _ in range(t):
	n = int(input())
	q = list(map(int, input().split()))
	ans = [0 for _ in range(n)]
	ans[0] = q[0]
	s = set(range(1, max(q) + 1))
	s -= {q[0]}
	tr = False
	for i in range(1, n):
		if q[i - 1] == q[i]:
			for j in s:
				if j < q[i]:
					ans[i] = j
					s -= {j}
					break
			else:
				tr = True
				break
		elif q[i - 1] < q[i]:
			ans[i] = q[i]
			if q[i] in s:
				s -= {q[i]}
			else:
				tr = True
				break
	if tr:
		print('-1')
	else:
		print(*ans)
