t = int(input())
for _ in range(t):
	(n, m) = map(int, input().split())
	friends = sorted(list(map(int, input().split())))
	gifts = list(map(int, input().split()))
	initial = n
	ans = 0
	for i in friends:
		if i > initial:
			break
		else:
			ans += gifts[i - 1]
			initial -= 1
	ans += sum(gifts[:initial])
	print(ans)
