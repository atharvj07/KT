#include <bits/stdc++.h>
using namespace std;
long long n, d, b, a[100050], cntl, cntr;
long long sum(long long l, long long r) {
  if (l < 1) l = 1;
  if (r > n) r = n;
  return a[r] - a[l - 1];
}
signed main() {
  scanf("%I64d%I64d%I64d", &n, &d, &b);
  for (long long i = 1; i <= n; i++) scanf("%I64d", &a[i]), a[i] += a[i - 1];
  for (long long i = 1; i <= n / 2; i++) {
    if (sum(1, i * (d + 1)) >= (cntl + 1) * b) cntl++;
    if (sum(n - i + 1 - i * d, n) >= (cntr + 1) * b) cntr++;
  }
  printf("%I64d\n", max(n / 2 - cntl, n / 2 - cntr));
}
