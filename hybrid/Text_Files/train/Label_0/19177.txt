#include<vector>

#include<math.h>
using namespace std;
#define N (1000000+3)
#define INF 1e16
typedef long long ll;
typedef pair<ll,ll> P;

ll kaijo[1000010];
ll kaijosize=1000010;

void init() {
	kaijo[0] = 1;
	for (ll i = 1;i < kaijosize;i++)kaijo[i] = (kaijo[i - 1] * i) % N;
}

ll inv(ll x,ll power) {
	ll res = 1;
	ll k = power;
	ll y = x;
	while (k) {
		if (k & 1)res = (res*y) % N;
		y = (y%N*y%N) % N;
		k /= 2;
	}
	return res;
}

int main(void){
    init();
    int q;
    cin>>q;
    for(int i=0;i<q;i++){
        ll x,d,n;
        cin>>x>>d>>n;
        if(x==0){
            cout<<0<<endl;
            continue;
        }
        if(d==0){
            cout<<inv(x,n)<<endl;
            continue;
        }
        ll d1=inv(d,n);
        ll f=x*inv(d,N-2)%N;
        if(f+n-1>=N)cout<<0<<endl;
        else{
            cout<<(((d1*kaijo[f+n-1])%N)*inv(kaijo[f-1],N-2))%N<<endl;
        }
    }
    return 0;
}
