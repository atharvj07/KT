import time

def main():
	n = i_input()
	a = li_input()
	p = [0] * n
	s = 0
	for (i, aa) in enumerate(a):
		s += aa
		c = 0
		while not aa % 2:
			aa //= 2
			c += 1
		p[i] = c
	if s % 2:
		print(0)
		return
	a.sort()
	limit = s // 2
	l = [0]
	for aa in a:
		b = []
		p1 = p2 = 0
		while p1 < len(l):
			while p2 < len(l) and l[p1] + aa >= l[p2] and (l[p2] <= limit):
				b.append(l[p2])
				p2 += 1
			if l[p1] + aa != b[-1] and l[p1] + aa <= limit:
				b.append(l[p1] + aa)
			p1 += 1
		l = b
	if l[-1] != limit:
		print(0)
		return
	minp = (100, -1)
	for i in range(n):
		minp = min(minp, (p[i], i + 1))
	print(1)
	print(minp[1])

def i_input():
	return int(input())

def l_input():
	return input().split()

def li_input():
	return list(map(int, l_input()))
TT = time.time()
main()
