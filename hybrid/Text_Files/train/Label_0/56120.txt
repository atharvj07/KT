    public static void solve(){
        int n = in.nextInt();
        int m = in.nextInt();
        int arr[] = new int[n];
        for(int i=0; i<n; i++)
            arr[i] = in.nextInt()-1;
        int days[] = new int[m];
        for(int i=0; i<m; i++)
            days[i] = in.nextInt();
        int start = 0, end = n-1;
        
        int ans = -1;
        while(start <= end)
        {
            int mid = (start + end)/2;
            if(check(mid,arr,days,n,m))
            {
                ans = mid+1;
                end = mid-1;
            }
            else
                start = mid+1;
        }
        out.println(ans);
    }

    private static boolean check(int mid, int[] arr, int[] days, int n, int m) {
       
        int map[] = new int[m];
        Arrays.fill(map,-1);
        for(int i=0; i<=mid; i++)
        {
            if(arr[i] != -1)
            {
                map[arr[i]] = i;
            }
        }
        int count = 0;
        for(int i=0; i<m; i++)
            if(map[i] == -1)
                return false;
        for(int i=0; i<=mid; i++)
        {
            if(arr[i] == -1 || map[arr[i]] != i)
                count ++;
            else
            {
                if(count < days[arr[i]])
                    return false;
                count -= days[arr[i]];
                
            }
        }
        return true;
    }

    public static void init()

        for(int i=0; i<union.length; i++)

    public static int find(int n)
    {
        return (union[n]==n)?n:(union[n]=find(union[n]));
    }
    public static void unionSet(int i ,int j)
    {
        union[find(i)]=find(j);
    }
    public static boolean connected(int i,int j)
    {   
        return union[i]==union[j];
    }

    public static long gcd(long a, long b) {
        long x = Math.min(a,b);
        long y = Math.max(a,b);
        while(x!=0)
        {
            long temp = x;
            x = y%x;
            y = temp;
        }
        return y;       
    }

    public static long modPow(long base, long exp, long mod) {
        base = base % mod;
        long result =1;
        while(exp > 0)
        {
            if(exp % 2== 1)
            {
                result = (result * base) % mod;
                exp --;
            }
            else
            {
                base = (base * base) % mod;
                exp = exp >> 1;
            }
            
        }
        return result;
    }
    
    public static void cal()
    {
        fac = new long[1000005];
        inv = new long[1000005];
        fac[0]=1;
        inv[0]=1;
        for(int i=1; i<=1000000; i++)
        {
            fac[i]=(fac[i-1]*i)%mod;
            inv[i]=(inv[i-1]*modPow(i,mod-2,mod))%mod;
        }
    }
    public static long ncr(int n, int r)
    {
        return (((fac[n]*inv[r])%mod)*inv[n-r])%mod;
    }

    public static class Reader {
        public BufferedReader reader;
        public StringTokenizer st;

        public Reader(InputStreamReader stream) {
            reader = new BufferedReader(stream);
            st = null;
        }

        public String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }
        public String nextLine() throws IOException{
            return reader.readLine();
        }
        public long nextLong(){
            return Long.parseLong(next());
        }
        public double nextDouble(){
            return Double.parseDouble(next());
        }

    }
