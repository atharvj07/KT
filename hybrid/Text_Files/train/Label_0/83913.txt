import sys

def input():
	return sys.stdin.readline().rstrip()

def slv():
	n = int(input())
	a = list(map(int, input().split()))
	if n == 1:
		print(a[0])
		return
	elif all((a[i] <= 0 for i in range(n))):
		minabs = 1 << 128
		s = 0
		for v in a:
			minabs = min(minabs, abs(v))
			s += abs(v)
		print(s - 2 * minabs)
	elif all((a[i] >= 0 for i in range(n))):
		minabs = 1 << 128
		s = 0
		for v in a:
			minabs = min(minabs, abs(v))
			s += abs(v)
		print(s - 2 * minabs)
	else:
		ans = sum((abs(v) for v in a))
		print(ans)
	return

def main():
	t = 1
	for i in range(t):
		slv()
	return
main()
