(n, x, k) = list(map(int, input().split()))
sorted_array = sorted(list(map(int, input().split())))
summ = i = j = 0
for val in sorted_array:
	dis = (val // x - k) * x
	if i < n or j < n:
		while i < n and sorted_array[i] <= min(dis, val):
			i += 1
		while j < n and sorted_array[j] <= min(dis + x, val):
			j += 1
	summ += j - i
print(summ)
