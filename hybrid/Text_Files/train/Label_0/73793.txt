def read():
	n = int(input())
	s = input()
	return (n, s)

def solve(n, s):
	st = [s[0]]
	for i in range(1, n):
		if s[i] == '(':
			st.append(s[i])
		elif len(st) == 0:
			st.append(s[i])
		elif st[-1] == '(':
			st.pop()
		else:
			st.append(s[i])
	if len(st) == 0:
		return True
	elif len(st) == 2:
		return st[0] != st[1]
result = solve(*read())
print('Yes' if result else 'No')
