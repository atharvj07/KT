#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 34;
const int maxlog = 20;
const int mod = 1e9 + 7;
const int sq = 720;
const int inf = 1e9;
vector<int> adj[maxn];
long double prob[maxn];
int cnt[maxn];
void dfs(int v) {
  cnt[v] = 1;
  for (auto u : adj[v]) {
    dfs(u);
    cnt[v] += cnt[u];
  }
}
void dfs2(int v) {
  for (auto u : adj[v])
    prob[u] = prob[v] + (long double)(cnt[v] - 1 - cnt[u]) / 2 + 1;
  for (auto u : adj[v]) dfs2(u);
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  cout.precision(20);
  int n;
  cin >> n;
  for (int i = 2; i <= n; i++) {
    int p;
    cin >> p;
    adj[p].push_back(i);
  }
  prob[1] = 1;
  dfs(1);
  dfs2(1);
  for (int i = 1; i <= n; i++) cout << prob[i] << " ";
  return 0;
}
