#include<cstdio>
#include<vector>
#include<map>
#include<algorithm>
const int MaxN=100010,MaxM=100010;
int N,M,A[MaxM],B[MaxM],C[MaxM],ind[MaxN],outd[MaxN],fa[MaxN],max_outd[MaxN];
int L1[MaxM],L2[MaxM];
int find(int i){return fa[i]==i?i:fa[i]=find(fa[i]);}
int calc(){
	for(int i=1;i<=N;i++)fa[i]=i,ind[i]=outd[i]=max_outd[i]=0;
	for(int i=0;i<M;i++){
		outd[A[i]]++;ind[B[i]]++;
		fa[find(A[i])]=find(B[i]);
	}
	for(int i=1;i<=N;i++){
		if(outd[i]&&!ind[i])return-1;
		if(outd[i]>max_outd[find(i)])max_outd[find(i)]=outd[i];
	}
	int ans=0,cc=0,cnt1=0,cnt2=0;
	long long sc1=0,sc2=0;
	for(int i=1;i<=N;i++)if(fa[i]==i&&max_outd[i]==1)cc++;
	for(int i=0;i<M;i++){
		if(A[i]!=B[i]){
			ans++;
			if(max_outd[find(A[i])]==1)sc2+=C[i]-1;
			else sc1+=C[i]-1;
		}
		else{
			if(max_outd[find(A[i])]==1){
				cc--;
				if(C[i]>2)L2[cnt2++]=2-C[i];
			}
			else if(C[i]>1)L1[cnt1++]=1-C[i];
		}
	}
	ans+=cc;
	cc=cc?sc2<cc?cc-sc2:1:0;
	cc=sc1<cc?cc-sc1:0;
	if(!cc)return ans;
	std::sort(L1,L1+cnt1);
	std::sort(L2,L2+cnt2);
	long long rest=cc;
	int res=-1;
	for(int i=sc1?-1:0,j=0;i<cnt1;i++){
		if(i>=0)rest+=L1[i];
		while(j<cnt2&&rest>0)rest+=L2[j++];
		while(j&&rest-L2[j-1]<=0)rest-=L2[--j];
		if(rest<=0&&(res==-1||ans+i+1+j*2<res))res=ans+i+1+j*2;
	}
	return res;
}
int main(){
	scanf("%d%d",&N,&M);
	for(int i=0;i<M;i++)scanf("%d%d%d",A+i,B+i,C+i);
	printf("%d\n",calc());
}
