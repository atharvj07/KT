#include <bits/stdc++.h>
using namespace std;
long long find(vector<long long> &v, long long x);
int main() {
  long long n;
  cin >> n;
  string str;
  cin >> str;
  vector<long long> v1, v2;
  for (long long i = 0; i < n; i++) {
    if (str[i] == 'A') {
      v1.push_back(i);
    } else {
      v2.push_back(i);
    }
  }
  long long ans = 0;
  for (long long i = 1; i < n; i++) {
    if (str[i] == 'A') {
      if (str[i - 1] == 'A') {
        long long a = find(v2, i);
        ans += i;
        if (a != -1) {
          ans = ans - 1;
        }
      } else {
        long long a = find(v1, i);
        ans += i;
        if (a == -1) {
          ans = ans - i;
        } else {
          ans = ans - (i - a - 1);
        }
      }
    } else {
      if (str[i - 1] == 'B') {
        long long a = find(v1, i);
        ans += i;
        if (a != -1) {
          ans = ans - 1;
        }
      } else {
        long long a = find(v2, i);
        ans += i;
        if (a == -1) {
          ans = ans - i;
        } else {
          ans = ans - (i - a - 1);
        }
      }
    }
  }
  cout << ans;
}
long long find(vector<long long> &v, long long x) {
  if (v.size() == 0) {
    return -1;
  }
  long long n = v.size();
  long long a = 0;
  long long b = n - 1;
  long long mid;
  long long ans = -1;
  while (a <= b) {
    mid = (a + b) / 2;
    long long k = v[mid];
    if (k > x) {
      b = mid - 1;
    } else if (k < x) {
      ans = k;
      a = mid + 1;
    } else {
      b = mid - 1;
    }
  }
  return ans;
}
