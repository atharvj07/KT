#include <bits/stdc++.h>
using namespace std;
const int MAX = 100 + 10;
const long long delta = 1000 * 1000 * 1000 + 7;
long long dp[10][MAX][MAX];
long long ans[10][MAX];
long long en[MAX][MAX];
long long answer;
int number[10];
int n;
void input() {
  cin >> n;
  for (int i = 0; i < 10; ++i) cin >> number[i];
}
void build_en() {
  for (int i = 0; i < MAX; ++i) en[i][0] = 1;
  for (int i = 1; i < MAX; ++i)
    for (int j = 1; j <= i; ++j)
      en[i][j] = (en[i - 1][j - 1] + en[i - 1][j]) % delta;
}
void build_ans(int ind) {
  for (int i = number[ind]; i <= n; ++i)
    for (int j = number[ind]; j <= i; ++j)
      ans[ind][i] = (ans[ind][i] + dp[ind][i][j]) % delta;
}
void build_dp(int ind) {
  for (int i = number[ind]; i <= n; ++i)
    for (int j = number[ind]; j <= i; ++j)
      dp[ind][i][j] = (en[i][j] * ans[ind - 1][i - j]) % delta;
}
void solve() {
  build_en();
  for (int i = number[1]; i <= n; ++i) ans[1][i] = 1;
  for (int i = 2; i < 10; ++i) {
    build_dp(i);
    build_ans(i);
  }
  for (int i = number[0]; i <= n; ++i)
    for (int j = number[0]; j <= i; ++j)
      dp[0][i][j] = (en[i - 1][j] * ans[9][i - j]) % delta;
  build_ans(0);
}
void output() {
  for (int i = number[0]; i <= n; ++i) answer = (answer + ans[0][i]) % delta;
  cout << answer << endl;
}
int main() {
  input();
  solve();
  output();
  return 0;
}
