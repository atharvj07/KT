while True:
	try:
		(n, m) = map(int, input().strip().split())
		list = [i for i in map(int, input().strip().split())]
		ai = []
		for i in range(m):
			ai.append(int(input()))
		dict = {}
		res = []
		j = len(list) - 1
		while j >= 0:
			if dict.get(list[j], 0) == 0:
				dict[list[j]] = 1
			res.append(len(dict))
			j -= 1
		for k in ai:
			print(res[-k])
	except:
		break
