#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  string s;
  cin >> s;
  int n = s.size(), consec = 1;
  vector<int> freqs;
  string pal;
  pal.push_back(s[0]);
  for (int i = 1; i < n; i++) {
    if (s[i] == s[i - 1]) {
      consec++;
    } else {
      freqs.push_back(consec);
      consec = 1;
      pal.push_back(s[i]);
    }
  }
  freqs.push_back(consec);
  if ((int)freqs.size() % 2 == 0) {
    cout << 0 << '\n';
    return 0;
  }
  assert(freqs.size() == pal.size());
  int lo = 0, hi = pal.size() - 1;
  while (lo < hi) {
    if (pal[lo] != pal[hi]) {
      cout << 0 << '\n';
      return 0;
    }
    if (freqs[lo] + freqs[hi] < 3) {
      cout << 0 << '\n';
      return 0;
    }
    ++lo;
    --hi;
  }
  int mid = freqs.size() / 2;
  if (freqs[mid] < 2) {
    cout << 0 << '\n';
    return 0;
  }
  cout << freqs[mid] + 1 << '\n';
  return 0;
}
