n = int(input())
a = list(map(int, input().split()))
last_updated = [0 for _ in range(n)]
q = int(input())
pay_days = [0 for _ in range(q)]
for i in range(q):
	qu = list(map(int, input().split()))
	if qu[0] == 1:
		(_, p, x) = qu
		a[p - 1] = x
		last_updated[p - 1] = i
	else:
		(_, x) = qu
		pay_days[i] = x
for i in range(q - 1):
	pay_days[q - i - 2] = max(pay_days[q - i - 1], pay_days[q - i - 2])
for i in range(n):
	print(max(a[i], pay_days[last_updated[i]]), end=' ')
print()
