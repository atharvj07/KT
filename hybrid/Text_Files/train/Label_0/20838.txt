#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, j, m, s;
  long long sum, a[305];
  scanf("%d", &n);
  while (n--) {
    scanf("%d", &m);
    for (i = 0; i < m; i++) scanf("%lld", &a[i]);
    sort(a, a + m);
    sum = a[0] * a[m - 1];
    for (i = 1, j = m - 2; i <= j; i++, j--) {
      if (sum != a[i] * a[j]) break;
    }
    if (i <= j)
      printf("-1\n");
    else {
      for (i = a[++j], s = 0; i >= 2; i--) {
        if (sum % i == 0) {
          if (a[j--] != i) {
            s = 1;
            break;
          }
        }
      }
      if (s)
        printf("-1\n");
      else
        printf("%lld\n", sum);
    }
  }
  return 0;
}
