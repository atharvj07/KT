bool is_prime(int p){
	if(p<=1)return false;
	if(p==2)return true;
	if(!(p&1))return false;
	for(int x=3;x<=sqrt(p);++x) if(!(p%x))return false;
	return true;
}

int main(){
	bool primes[10002];
	memset(primes,0,sizeof(primes));
	for(int x=2;x<=10000;++x)if(is_prime(x))primes[x]=true;
	int n;
	while(scanf("%d",&n)==1){
		int c =0 ;
		int l = n/2;
		if(n%2==1){
			++l;
			if(primes[l])--c;
		}
		for(int x=1;x<=l;++x){
			if(primes[x] && primes[n-x+1])c +=2;
		}
		cout<<c<<endl;
	}
	return 0;	
}
