#include <bits/stdc++.h>
using namespace std;
vector<int> options;
int n, v[2 * 2020], dp[2 * 2020][2020], pos[2 * 2020];
int solve(int position, int val) {
  if (val > n) return 0;
  if (val == n) return 1;
  if (position >= (int)options.size()) return 0;
  if (dp[position][val] != -1) return dp[position][val];
  return dp[position][val] = max(solve(position + 1, val),
                                 solve(position + 1, val + options[position]));
}
int main() {
  int noCases;
  scanf("%d", &noCases);
  ;
  while (noCases--) {
    options.clear();
    scanf("%d", &n);
    ;
    for (int i = 1; i <= 2 * n; i++) {
      scanf("%d", &v[i]);
      ;
      pos[v[i]] = i;
    }
    int mini = 2 * n + 1, last = 2 * n + 1;
    pos[last] = last;
    for (int i = 2 * n; i >= 1; i--) {
      if (pos[i] <= mini) {
        options.emplace_back(pos[last] - pos[i]);
        last = i;
      }
      mini = min(mini, pos[i]);
    }
    for (int i = 0; i <= (int)options.size(); i++) {
      for (int j = 0; j <= n; j++) dp[i][j] = -1;
    }
    if (solve(0, 0) == 1)
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
