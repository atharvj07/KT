#include <bits/stdc++.h>
using namespace std;
int GCD(int x, int y) { return !y ? x : GCD(y, x % y); }
int LCM(int x, int y) { return x * y / GCD(x, y); }
long long int ceilDivision(long long int dividend, long long int divisor) {
  return (dividend + divisor - 1) / divisor;
}
