#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10, mod = 1e9 + 7;
struct edge {
  int to, pre, c;
} e[N << 1];
int u[N], l = 0;
void ins(int a, int b, int c) { e[++l] = (edge){b, u[a], c}, u[a] = l; }
int pw(int a, int b) {
  int w = 1;
  for (; b; b >>= 1, a = 1LL * a * a % mod)
    if (b & 1) w = 1LL * w * a % mod;
  return w;
}
int n, d[N], r[N];
pair<int, int> f[N];
void dfs(int first, int fa) {
  for (int i = u[first]; i; i = e[i].pre)
    if (e[i].to != fa) r[e[i].to] = e[i].c, dfs(e[i].to, first);
  if (d[first] == 1 && first > 1) return;
  f[first] = make_pair(r[first], 0);
  int s = d[first];
  for (int i = u[first]; i; i = e[i].pre)
    if (e[i].to != fa)
      f[first].first =
          ((f[first].first + f[e[i].to].first) % mod + e[i].c) % mod,
      s = (s - f[e[i].to].second) % mod;
  f[first].second = pw(s, mod - 2),
  f[first].first = 1LL * f[first].first * f[first].second % mod;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n - 1; i++) {
    int a, b, c;
    scanf("%d%d%d", &a, &b, &c);
    a++, b++;
    ins(a, b, c), ins(b, a, c);
    d[a]++, d[b]++;
  }
  dfs(1, 0);
  if (f[1].first < 0) f[1].first += mod;
  printf("%d\n", f[1].first);
  return 0;
}
