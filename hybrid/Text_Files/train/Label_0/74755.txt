import math
import queue
import sys

def Run(L, R, H):
	if L >= R:
		return 0
	Flag = True
	for i in range(L, R):
		if Fences[i] - H > 0:
			Flag = False
	if Flag == True:
		return 0
	minHigh = min(Fences[L:R]) - H
	if minHigh >= R - L:
		return R - L
	CountStroke = 0
	tempL = L
	tempR = R
	while True:
		if tempL >= tempR:
			return CountStroke
		HighL = Fences[tempL] - H
		HighR = Fences[tempR - 1] - H
		Maxhigh = max(HighL, HighR)
		if Maxhigh < tempR - tempL:
			break
		CountStroke += 1
		if Maxhigh == HighL:
			tempL += 1
			continue
		elif Maxhigh == HighR:
			tempR -= 1
	Min = tempL
	for i in range(tempL, tempR):
		if Fences[Min] > Fences[i]:
			Min = i
	tempArry = []
	for i in range(tempL, tempR):
		if Fences[i] == Fences[Min]:
			tempArry.append(i)
	Min = tempArry[len(tempArry) // 2]
	return min(R - L, CountStroke + Fences[Min] - H + Run(tempL, Min, Fences[Min]) + Run(Min + 1, tempR, Fences[Min]))
fileInput = sys.stdin
N = int(fileInput.readline())
Fences = list(map(int, fileInput.readline().split()))
Output = Run(0, N, 0)
print(Output)
