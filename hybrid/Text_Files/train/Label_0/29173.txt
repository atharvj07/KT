#include <bits/stdc++.h>
using namespace std;
int main() {
  int t = 1;
  while (t--) {
    stack<char> st;
    string str;
    cin >> str;
    for (int i = 0; i < str.length(); i++) {
      if (!st.empty() && st.top() == str[i])
        st.pop();
      else
        st.push(str[i]);
    }
    cout << (st.empty() ? "Yes" : "No") << endl;
  }
  return 0;
}
