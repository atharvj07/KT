(n, a, b, c) = map(int, input().split())
x = 10000000000
n = n % 4
n = 4 - n
if n == 4:
	print(0)
elif n == 1:
	x = min(x, a)
	x = min(x, b + c)
	x = min(x, 3 * c)
	print(x)
elif n == 2:
	x = min(x, 2 * a)
	x = min(x, b)
	x = min(x, c * 2)
	print(x)
elif n == 3:
	x = min(x, 3 * a)
	x = min(x, a + b)
	x = min(x, c)
	print(x)
