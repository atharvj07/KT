#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000000 + 6;
const long long inf = 0x7fffffffffffffff;
int a[maxn];
long long res;
int n;
long long sum;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    sum += 1ll * a[i];
  }
  res = inf;
  for (long long p = 2; p * p <= sum; p++) {
    if (sum % p == 0) {
      long long tmp = 0;
      long long pre = 0;
      for (int i = 1; i <= n; i++) {
        pre = (pre + a[i]) % p;
        tmp += min(pre, p - pre);
      }
      res = min(res, tmp);
      if (sum / p > 1) {
        tmp = 0;
        pre = 0;
        long long tp = sum / p;
        for (int i = 1; i <= n; i++) {
          pre = (pre + a[i]) % tp;
          tmp += min(pre, tp - pre);
        }
        res = min(res, tmp);
      }
    }
    while (sum % p == 0) sum /= p;
  }
  if (sum > 1) {
    long long tmp = 0;
    long long pre = 0;
    for (int i = 1; i <= n; i++) {
      pre = (pre + a[i]) % sum;
      tmp += min(pre, sum - pre);
    }
    res = min(res, tmp);
  }
  if (res >= inf)
    printf("-1\n");
  else
    printf("%I64d\n", res);
  return 0;
}
