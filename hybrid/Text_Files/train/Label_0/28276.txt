#include<bits/stdc++.h>
using namespace std;	
int n,ans;
pair<int,int> a[55];
map<pair<int,int>,int> m;
int main(){
	scanf("%d",&n);
	for(int i=0;i<n;i++){
		scanf("%d %d",&a[i].first,&a[i].second);
	}
	for(int i=0;i<n;i++){
		for(int j=0;j<n;j++){
			if(i==j) continue;
			m[make_pair(a[j].first-a[i].first,a[j].second-a[i].second)]++;
			if(ans<m[make_pair(a[j].first-a[i].first,a[j].second-a[i].second)]) ans=m[make_pair(a[j].first-a[i].first,a[j].second-a[i].second)];
		}
	}
	printf("%d",n-ans);
}
