class Solution:

	def stableGroups(self, N, K, X, stuLevel):
		groups = 1
		gapSize = []
		prev = stuLevel[0]
		for level in stuLevel:
			fill = 0
			if prev + X < level:
				fill = (level - prev - 1) // X
				gapSize.append(fill)
				groups += 1
			prev = level
		(N, i) = (len(gapSize), 0)
		gapSize.sort()
		while i < N and K >= gapSize[i]:
			K -= gapSize[i]
			groups -= 1
			i += 1
		return groups
s = Solution()
t = 1
while t:
	(N, K, X) = map(int, input().split(' '))
	arr = list(map(int, input().split(' ')))
	arr.sort()
	print(s.stableGroups(N, K, X, arr))
	t -= 1
