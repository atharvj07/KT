#include <bits/stdc++.h>
using namespace std;
int n, m;
unsigned long long oddCount, evenCount, oddCount2, evenCount2;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    oddCount = 0;
    evenCount = 0;
    oddCount2 = 0;
    evenCount2 = 0;
    register int temp;
    while (n--) {
      cin >> temp;
      if (temp & 1) {
        oddCount++;
      } else {
        evenCount++;
      }
    }
    cin >> m;
    while (m--) {
      cin >> temp;
      if (temp & 1) {
        oddCount2++;
      } else {
        evenCount2++;
      }
    }
    cout << oddCount * oddCount2 + evenCount * evenCount2 << "\n";
  }
}
