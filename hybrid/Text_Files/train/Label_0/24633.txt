#include <bits/stdc++.h>
using namespace std;
const int N = 6000;
int n, a[N], ans[N];
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &a[i]);
  }
  for (int i = 0; i < n; ++i) {
    int cnt = 0;
    int id = -1;
    vector<int> c(N, 0);
    for (int j = i; j < n; ++j) {
      c[a[j]]++;
      if (c[a[j]] > cnt || (c[a[j]] == cnt && a[j] < id)) {
        cnt = c[a[j]];
        id = a[j];
      }
      ans[id]++;
    }
  }
  for (int i = 1; i <= n; ++i) {
    printf("%d ", ans[i]);
  }
  return 0;
}
