import math
l = input().split()
for i in range(6):
	l[i] = int(l[i])
s = 3 ** 0.5 / 4
b = (l[0] ** 2 + l[1] ** 2 + l[0] * l[1]) ** 0.5
c = (l[2] ** 2 + l[3] ** 2 + l[2] * l[3]) ** 0.5
d = (l[4] ** 2 + l[5] ** 2 + l[4] * l[5]) ** 0.5

def ar(b, c, d):
	s = b + c + d
	s /= 2
	return (s * (s - b) * (s - c) * (s - d)) ** 0.5
A = (ar(l[0], l[1], b) + ar(l[2], l[3], c) + ar(l[4], l[5], d) + ar(c, d, b)) / s
if int(A) == int(A + 0.5):
	print(int(A))
else:
	print(int(A + 0.5))
