#include <bits/stdc++.h>
using namespace std;
int main() {
  int c, i;
  char a[20];
  scanf("%s", &a);
  c = 0;
  int l = strlen(a);
  for (i = 0; i < l / 2; i++) {
    if (a[l - 1 - i] - '0' == a[i] - '0')
      continue;
    else
      c++;
  }
  if (strlen(a) % 2 != 0 && c == 0)
    printf("YES\n");
  else if (c == 1)
    printf("YES\n");
  else
    printf("NO\n");
  return 0;
}
