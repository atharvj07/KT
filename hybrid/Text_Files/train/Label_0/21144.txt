#include <bits/stdc++.h>
using namespace std;
int main() {
  int x;
  cin >> x;
  int a[x];
  int b[x];
  int i;
  for (i = 0; i < x; i++) {
    cin >> a[i];
    b[i] = i + 1;
  }
  i = 0;
  for (i = 0; i < x; i++) {
    int j = 0;
    while (b[i] != a[j]) {
      j++;
    }
    cout << j + 1 << " ";
  }
}
