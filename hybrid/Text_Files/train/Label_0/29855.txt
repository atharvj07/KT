#include<iostream>
using namespace std;
#define ll long long 
#define w 1000000007
int n;
char s[100],t[100];
ll a=1;
bool b,c;

int main(){
  cin>>n>>s>>t;

  for(int i=0;i<n;i++){
    if(s[i]==t[i]){
      if(!b&&!c)
        a*=3;
      if(b)
        a*=2;
      if(c)
        a*=1;
      a%=w;
      b=true;
      c=false;
    }
    if(s[i]==s[i+1]&&t[i]==t[i+1]){
      if(!b&&!c)
        a*=6;
      if(b)
        a*=2;
      if(c)
        a*=3;
      a%=w;
      b=false;
      c=true;
    }
  }

  cout<<a<<endl;

  return 0;
}
