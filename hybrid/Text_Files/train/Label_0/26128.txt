#include <bits/stdc++.h>
using namespace std;
int c[200005], nxt[200005], a[200005];
set<int> mySet;
set<int>::iterator it;
int main() {
  int i, j, n, m, q, t, p, val, curr;
  scanf("%d", &n);
  for (i = 1; i <= n + 1; i++) mySet.insert(i);
  for (i = 1; i <= n; i++) {
    scanf("%d", &c[i]);
    a[i] = 0;
    nxt[i] = i + 1;
  }
  scanf("%d", &q);
  while (q--) {
    scanf("%d", &t);
    if (t == 1) {
      scanf("%d %d", &p, &val);
      if (val <= c[p] - a[p])
        a[p] += val;
      else {
        while (val > 0 && p <= n) {
          if (val <= c[p] - a[p]) {
            a[p] += val;
            if (a[p] == c[p]) mySet.erase(p);
            break;
          } else {
            val = val - c[p] + a[p];
            a[p] = c[p];
            mySet.erase(p);
            it = mySet.upper_bound(p);
            p = (*it);
          }
        }
      }
    } else {
      scanf("%d", &p);
      printf("%d\n", a[p]);
    }
  }
  return 0;
}
