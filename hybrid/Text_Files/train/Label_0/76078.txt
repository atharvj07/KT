n = int(input())
a = int(input())
b = int(input())
f = 0
if a % 2 == 0 and b % 2 == 0 and (n % 2 != 0):
	print('NO')
elif a == b:
	if n % a != 0:
		print('NO')
	else:
		print('YES')
		print(int(n / a), 0)
elif a % 3 == 0 and b % 3 == 0 and (n % 3 != 0):
	print('NO')
else:
	if n % a == 0:
		print('YES')
		f = 1
		print(int(n / a), 0)
	elif n % b == 0:
		print('YES')
		f = 1
		print(0, int(n / b))
	else:
		t = max(a, b)
		s = min(a, b)
		while t < n:
			if (n - t) % s == 0:
				print('YES')
				f = 1
				if a < b:
					print(int((n - t) / a), int(t / b))
				else:
					print(int(t / a), int((n - t) / b))
				break
			t = t + max(a, b)
	if f == 0:
		print('NO')
