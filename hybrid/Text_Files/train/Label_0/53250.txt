import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class SerajaAndSubsequences {
    static long MOD = 1000000007;

    static class BIT {
        long[] arr;

        public BIT(int n) {
            arr = new long[n + 10];
        }

        public void add(int i, long val) {
            while (i < arr.length) {
                arr[i] = (arr[i] + val) % MOD;
                i += i & -i;
            }
        }

        public long query(int i) {
            long ret = 0;
            while (i > 0) {
                ret = (ret + arr[i]) % MOD;
                i -= i & -i;
            }
            return ret;
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer strtok;
        BIT bit = new BIT(1000006);
        strtok = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(strtok.nextToken());
        int[] arr = new int[n];
        strtok = new StringTokenizer(in.readLine());
        for (int i = 0; i < arr.length; i++) {
            arr[i] = Integer.parseInt(strtok.nextToken());
        }
        for (int i = arr.length - 1; i >= 0; i--) {
            long sum = (bit.query(1000005) - bit.query(arr[i] - 1) + MOD) % MOD;
            sum = (sum + 1) * arr[i];
            long single = (bit.query(arr[i]) - bit.query(arr[i] - 1) + MOD)
                    % MOD;
            single = (single * -1 + MOD) % MOD;
            bit.add(arr[i], single);
            bit.add(arr[i], sum);
        }
        System.out.println(bit.query(1000005));
    }
}
