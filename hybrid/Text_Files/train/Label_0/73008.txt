for _ in range(int(input())):
	(a, b) = map(int, input().split())
	k = a * (a + 1) // 2
	if b >= 2 * k - a:
		print(2 * a - 1)
	elif b <= k:
		c = int((-1 + (1 + 8 * b) ** (1 / 2)) // 2)
		if c * (c + 1) // 2 >= b:
			print(c)
		else:
			print(c + 1)
	else:
		b = b - a * (a + 1) // 2
		b = a * (a - 1) // 2 - b
		c = int((-1 + (1 + 8 * b) ** (1 / 2)) // 2)
		if c * (c + 1) // 2 <= b:
			print(2 * a - c - 1)
		else:
			print(2 * a - c)
