public class Main
{
	private final long INF = Long.MAX_VALUE / 2;
	
	public void foo()
	{
		MyScanner scan = new MyScanner();
		int n = scan.nextInt();
		int m = scan.nextInt();
		int k = scan.nextInt();
		long[][] cost = new long[n + 1][n + 1];
		for(int i = 0;i <= n;++i)
		{
			for(int j = 0;j <= n;++j)
			{
				cost[i][j] = INF;
			}
		}
		for(int i = 0;i < m;++i)
		{
			int left = scan.nextInt();
			int right = scan.nextInt();
			int c = scan.nextInt();
			cost[left][right] = Math.min(cost[left][right], (long)c);
		}
		for(int i = 1;i <= n;++i)
		{
			for(int j = n - 1;j >= i;--j)
			{
				cost[i][j] = Math.min(cost[i][j], cost[i - 1][j]);
				cost[i][j] = Math.min(cost[i][j], cost[i][j + 1]);
			}
		}
		long[][] dp = new long[n + 1][n + 1];
		for(int i = 0;i <= n;++i)
		{
			dp[i][0] = 0;
			for(int j = 1;j <= n;++j)
			{
				dp[i][j] = INF;
			}
		}
		for(int i = 1;i <= n;++i)
		{
			for(int j = 1;j <= i;++j)
			{
				dp[i][j] = dp[i - 1][j];
				for(int p = i - j;p < i;++p)
				{
					dp[i][j] = Math.min(dp[i][j], dp[p][j - (i - p)] + cost[p + 1][i]);
				}
			}
		}
		long minCost = INF;
		for(int i = k;i <= n;++i)
		{
			minCost = Math.min(minCost, dp[n][i]);
		}
		System.out.println(minCost != INF ? minCost : -1);
	}

	public static void main(String[] args)
	{
		new Main().foo();
	}
	
	class MyScanner
	{
	    private byte[] buf = new byte[1024];
	    private int curChar;
	    private int numChars;
	    BufferedInputStream bis = new BufferedInputStream(System.in);

	    public int read() 
	    {
	        if (-1 == numChars)
	        {
	        	throw new InputMismatchException();
	        }
	        if (curChar >= numChars)
	        {
	            curChar = 0;
	            try 
	            {
	                numChars = bis.read(buf);
	            }
	            catch (IOException e)
	            {
	                throw new InputMismatchException();
	            }
	            if (numChars <= 0)
	            {
	            	return -1;
	            }
	        }
	        return buf[curChar++];
	    }

	    public int nextInt() 
	    {
	        int c = read();
	        while (isSpaceChar(c))
	        {
	        	c = read();
	        }
	        int sgn = 1;
	        if (c == '-') 
	        {
	            sgn = -1;
	            c = read();
	        }
	        int res = 0;
	        do 
	        {
	            if (c < '0' || c > '9')
	            {
	            	throw new InputMismatchException();
	            }
	            res *= 10;
	            res += c - '0';
	            c = read();
	        } while (!isSpaceChar(c));
	        return res * sgn;
	    }
	    
	    public long nextLong() 
	    {
	    	int c = read();
	    	while (isSpaceChar(c))
	    	{
	    		c = read();
	    	}
	    	int sgn = 1;
	    	if (c == '-') 
	    	{
	    		sgn = -1;
	    		c = read();
	    	}
	    	long res = 0;
	    	do 
	    	{
	    		if (c < '0' || c > '9')
	    		{
	    			throw new InputMismatchException();
	    		}
	    		res *= 10;
	    		res += c - '0';
	    		c = read();
	    	} while (!isSpaceChar(c));
	    	return res * sgn;
	    }
	    
	    public String next()
	    {
	        int c = read();
	        while (isSpaceChar(c))
	        {
	        	c = read();
	        }
	        StringBuilder res = new StringBuilder();
	        do 
	        {
	            res.appendCodePoint(c);
	            c = read();
	        } while (!isSpaceChar(c));
	        return res.toString();
	    }

	    private boolean isSpaceChar(int c) 
	    {
	        return ' ' == c || '\n' == c || '\r' == c || '\t' == c || -1 == c;
	    }
	}
}
