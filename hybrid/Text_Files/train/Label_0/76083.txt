s = list(input())
from math import *
dp = [0 for i in range(s.count('o') + 2)]
c = 0
j = 1
for i in range(len(s)):
	if s[i] == 'o':
		dp[j] = dp[j - 1] + max(0, c - 1)
		j += 1
		c = 0
	else:
		c += 1
s = 0
dp[len(dp) - 1] = dp[len(dp) - 2] + max(0, c - 1)
for i in range(1, len(dp) - 1):
	z = dp[len(dp) - 1] - dp[i]
	s += dp[i] * z
print(s)
