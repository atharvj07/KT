import re
(a, b) = [int(i) for i in input().split()]

def binSlice(x):
	return bin(x)[2:]

def removeLeadingZeros(s):
	for k in range(len(s)):
		if s[k] != '0':
			break
	return s[k:]

def reversed(s):
	return s[::-1]
temp = reversed(binSlice(a))
temp2 = removeLeadingZeros(temp)
check = re.fullmatch(f'1*(1{temp}|{temp2}|{reversed(temp)}1|{reversed(temp2)})1*', binSlice(b)) is not None
print('YES' if check or a == b else 'NO')
