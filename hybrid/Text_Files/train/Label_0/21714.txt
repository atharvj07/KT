#include <bits/stdc++.h>
using namespace std;
vector<int> water;
vector<vector<int> > maps;
unordered_map<int, pair<int, int> > nodes;
set<pair<int, int> > lakes;
class UnionFind {
 private:
  vector<int> p, rank, t;

 public:
  UnionFind(int N) {
    rank.assign(N, 0);
    t.assign(N, 1);
    p.assign(N, 0);
    for (int i = 0; i < N; i++) p[i] = i;
  }
  int findSet(int i) { return (p[i] == i) ? i : (p[i] = findSet(p[i])); }
  int countSet(int i) { return t[findSet(i)]; }
  bool isSameSet(int i, int j) { return findSet(i) == findSet(j); }
  void unionSet(int i, int j) {
    if (!isSameSet(i, j)) {
      int x = findSet(i), y = findSet(j);
      if (rank[x] > rank[y]) {
        p[y] = x;
        t[x] += t[y];
      } else {
        p[x] = y;
        t[y] += t[x];
        if (rank[x] == rank[y]) rank[y]++;
      }
    }
  }
};
int main() {
  int n, m, k, i, j, node = 0, cant, pat;
  char row[51];
  scanf("%d %d %d", &n, &m, &k);
  water = vector<int>(n * m);
  maps = vector<vector<int> >(n, vector<int>(m));
  UnionFind UF(n * m);
  for (i = 0; i < n; i++) {
    scanf("%s", row);
    for (j = 0; j < m; j++) {
      if (row[j] == '.') {
        maps[i][j] = node;
        if (j > 0) {
          if (maps[i][j - 1] != -1) {
            if (!UF.isSameSet(maps[i][j], maps[i][j - 1]))
              UF.unionSet(maps[i][j], maps[i][j - 1]);
          }
        }
        if (i > 0) {
          if (maps[i - 1][j] != -1) {
            if (!UF.isSameSet(maps[i][j], maps[i - 1][j]))
              UF.unionSet(maps[i][j], maps[i - 1][j]);
          }
        }
        if (i == 0 or i == n - 1 or j == 0 or j == m - 1)
          water[node] = 2;
        else
          water[node] = 1;
        nodes[node] = pair<int, int>(i, j);
        node++;
      } else
        maps[i][j] = -1;
    }
  }
  for (i = 0; i < node; i++) {
    pat = UF.findSet(i);
    cant = UF.countSet(i);
    if (water[i] == 1 and water[pat] == 1)
      lakes.insert(pair<int, int>(cant, pat));
    else if (water[i] == 2) {
      water[pat] = 2;
      lakes.erase(pair<int, int>(cant, pat));
    }
  }
  i = 0;
  cant = 0;
  for (auto it = lakes.begin(); i < lakes.size() - k; it++) {
    water[(*it).second] = -1;
    cant += (*it).first;
    i++;
  }
  printf("%d\n", cant);
  for (i = 0; i < n; i++) {
    for (j = 0; j < m; j++) {
      if (maps[i][j] == -1)
        printf("*");
      else {
        if (water[UF.findSet(maps[i][j])] == -1)
          printf("*");
        else
          printf(".");
      }
    }
    printf("\n");
  }
  return 0;
}
