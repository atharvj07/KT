import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

public class Try2 {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();
		int k = in.nextInt();

		in.nextLine();

		int r = 0;

		char[][] s = new char[n][m];

		for (int i = 0; i < n; i++) {
			s[i] = in.nextLine().toCharArray();
		}

		for (int i = 0; i < n; i++) {
			int f = 0;
			for (int j = 0; j < m; j++) {
				if (s[i][j] == '.')
					f++;
				else {
					if (f >= k)
						r += f - k + 1;
					f = 0;
				}
			}
			if (f >= k)
				r += f - k + 1;
		}

		if (k > 1) {
			for (int j = 0; j < m; j++) {
				int f = 0;
				for (int i = 0; i < n; i++) {
					if (s[i][j] == '.')
						f++;
					else {
						if (f >= k)
							r += f - k + 1;
						f = 0;
					}
				}
				if (f >= k)
					r += f - k + 1;

			}
		}

		System.out.println(r);

		in.close();
	}
}
