import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Map;
import java.util.Random;
import java.util.Scanner;
import java.util.StringTokenizer;
import java.util.TreeMap;
import java.util.TreeSet;

public class D {
	public static void main(String[] args) {
		MyScanner in = new MyScanner();
		PrintWriter out = new PrintWriter(System.out);
		int n = in.nextInt();
		int[] as = new int[n];
		for (int i = 0; i < n; i++)
			as[i] = in.nextInt();
		int[] par = new int[n];
		long[] wpar = new long[n];
		par[0] = 0;
		ArrayList<Integer>[] cs = new ArrayList[n];
		for (int i = 0; i < n; i++)
			cs[i] = new ArrayList<Integer>();
		int[] indeg = new int[n];
		for (int i = 1; i < n; i++) {
			int p = in.nextInt() - 1;
			int w = in.nextInt();
			par[i] = p;
			wpar[i] = w;
			indeg[p]++;
			cs[p].add(i);
		}
		Tree t = new Tree(par);
		long[] distToRoot = new long[n];
		ArrayDeque<Integer> q = new ArrayDeque<Integer>();
		q.offer(0);
		while(!q.isEmpty()) {
			int node = q.poll();
			for(int child : cs[node]) {
				distToRoot[child] = distToRoot[node] + wpar[child];
				q.offer(child);
			}
		}
		for (int i = 0; i < n; i++) {
			if (indeg[i] == 0)
				q.offer(i);
		}
		long[] ans = new long[n];
		while (!q.isEmpty()) {
			int node = q.poll();
			for (int child : cs[node]) {
				ans[node] += ans[child];
			}
			long a = as[node];
			if (par[node] != node) {
				int d = t.lifting[node].length-1;
				int c = node;
				while(d >= 0) {
					int lift = t.lifting[c][d];
					long dist = distToRoot[node] - distToRoot[lift];
					if(dist <= a) {
						c = lift;
					}
					d--;
				}
				ans[par[node]]++;
				if(par[c] != c) {
					ans[par[c]]--;
				}
				
				indeg[par[node]]--;
				if (indeg[par[node]] == 0) {
					q.offer(par[node]);
				}
			}
		}
		for(int i=0;i<n;i++){
			out.print(ans[i]);
			out.print(' ');
		}
		out.println();
		out.close();

	}
	static class Tree {
		int[] par;
		int[][] lifting;
		Tree(int[] par) {
			this.par = par;
			int n = par.length;
			int lg = (int)Math.ceil(Math.log(n) / Math.log(2))+1;
			lifting = new int[n][lg];
			for(int i=0;i<n;i++) {
				lifting[i][0] = par[i];
			}
			for(int d=1;d<lg;d++) {
				for(int i=0;i<n;i++) {
					lifting[i][d] = lifting[lifting[i][d-1]][d-1];
				}
			}
		}
	}
	private static class MyScanner {
		BufferedReader br;
		StringTokenizer st;

		public MyScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}
	}
}
