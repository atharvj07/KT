(n, m) = map(int, input().split())
L = [input() for i in range(n)]
Visited = [[0] * m for i in range(n)]
start = list(map(int, input().split()))
end = list(map(int, input().split()))
if abs(end[0] - start[0]) + abs(end[1] - start[1]) == 1:
	if L[end[0] - 1][end[1] - 1] == 'X':
		print('YES')
	else:
		k = 0
		if end[0] != 1:
			if L[end[0] - 2][end[1] - 1] == '.':
				k += 1
		if end[1] != 1:
			if L[end[0] - 1][end[1] - 2] == '.':
				k += 1
		if end[0] != n:
			if L[end[0]][end[1] - 1] == '.':
				k += 1
		if end[1] != m:
			if L[end[0] - 1][end[1]] == '.':
				k += 1
		if k >= 1:
			print('YES')
		else:
			print('NO')
	exit()

def Do(i, j):
	q = [[i, j]]
	f = False
	while q:
		d = q[0]
		if d[0] != 0:
			if Visited[d[0] - 1][d[1]] == 0 and L[d[0] - 1][d[1]] == '.':
				Visited[d[0] - 1][d[1]] = 1
				q.append([d[0] - 1, d[1]])
		if d[0] != n - 1:
			if Visited[d[0] + 1][d[1]] == 0 and L[d[0] + 1][d[1]] == '.':
				Visited[d[0] + 1][d[1]] = 1
				q.append([d[0] + 1, d[1]])
		if d[1] != 0:
			if Visited[d[0]][d[1] - 1] == 0 and L[d[0]][d[1] - 1] == '.':
				Visited[d[0]][d[1] - 1] = 1
				q.append([d[0], d[1] - 1])
		if d[1] != m - 1:
			if Visited[d[0]][d[1] + 1] == 0 and L[d[0]][d[1] + 1] == '.':
				Visited[d[0]][d[1] + 1] = 1
				q.append([d[0], d[1] + 1])
		del q[0]
Do(start[0] - 1, start[1] - 1)
if end[0] == start[0] and start[1] == end[1]:
	Visited[end[0] - 1][end[1] - 1] = 0
if Visited[end[0] - 1][end[1] - 1] == 1:
	k = 0
	if end[0] != 1:
		if L[end[0] - 2][end[1] - 1] == '.':
			k += 1
	if end[1] != 1:
		if L[end[0] - 1][end[1] - 2] == '.':
			k += 1
	if end[0] != n:
		if L[end[0]][end[1] - 1] == '.':
			k += 1
	if end[1] != m:
		if L[end[0] - 1][end[1]] == '.':
			k += 1
	if k >= 2:
		print('YES')
	else:
		print('NO')
elif L[end[0] - 1][end[1] - 1] == 'X':
	k = 0
	f = False
	if end[0] != 1:
		if L[end[0] - 2][end[1] - 1] == '.':
			k += 1
		if Visited[end[0] - 2][end[1] - 1] == 1:
			f = True
	if end[1] != 1:
		if L[end[0] - 1][end[1] - 2] == '.':
			k += 1
		if Visited[end[0] - 1][end[1] - 2] == 1:
			f = True
	if end[0] != n:
		if L[end[0]][end[1] - 1] == '.':
			k += 1
		if Visited[end[0]][end[1] - 1] == 1:
			f = True
	if end[1] != m:
		if L[end[0] - 1][end[1]] == '.':
			k += 1
		if Visited[end[0] - 1][end[1]] == 1:
			f = True
	if k >= 1 and f:
		print('YES')
	else:
		print('NO')
else:
	print('NO')
