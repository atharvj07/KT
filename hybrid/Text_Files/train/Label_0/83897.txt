total = int(input())
nums = [[int(n) for n in input().split()] for i in range(total)]
for n in nums:
	bin1 = bin(n[0])[2:]
	bin2 = bin(n[1])[2:]
	maxlen = max(len(bin1), len(bin2))
	bin1 = bin1.zfill(maxlen)
	bin2 = bin2.zfill(maxlen)
	print(int(''.join([str(int(bin1[i]) ^ int(bin2[i])) for i in range(maxlen)]), 2))
