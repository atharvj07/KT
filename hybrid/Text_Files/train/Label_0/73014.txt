x = int(input())
output = []
for i in range(x):
	(first, second, third) = input().split()
	first = int(first)
	second = int(second)
	third = int(third)
	if first >= second and first >= third:
		if first == second == third:
			output.append('1 1 1')
		elif first == second:
			output.append('1 1 ' + str(first - third + 1))
		elif first == third:
			output.append('1 ' + str(first - second + 1) + ' 1')
		else:
			output.append('0 ' + str(first - second + 1) + ' ' + str(first - third + 1))
	elif second >= first and second >= third:
		if first == second:
			output.append('1 1 ' + str(second - third + 1))
		elif second == third:
			output.append(str(second - first + 1) + ' 1 1')
		else:
			output.append(str(second - first + 1) + ' 0 ' + str(second - third + 1))
	elif third >= first and second <= third:
		if third == first:
			output.append('1 ' + str(third - second + 1) + ' 1')
		elif second == third:
			output.append(str(third - first + 1) + ' 1 1')
		else:
			output.append(str(third - first + 1) + ' ' + str(third - second + 1) + ' 0')
for i in range(len(output)):
	print(output[i])
