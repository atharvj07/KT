(n, x) = map(int, input().split())
arrx = list(map(int, input().split()))
(m, y) = map(int, input().split())
arry = list(map(int, input().split()))
sumx = sumy = 0
for i in range(n):
	sumx += pow(x, n - 1 - i) * arrx[i]
for j in range(m):
	sumy += pow(y, m - 1 - j) * arry[j]
if sumx > sumy:
	print('>')
elif sumx < sumy:
	print('<')
else:
	print('=')
