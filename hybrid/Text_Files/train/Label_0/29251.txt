#include <bits/stdc++.h>
using namespace std;
#pragma GCC optimize("O1")
#pragma GCC optimize("O2")
#pragma GCC optimize("Ofast")
#pragma GCC target("avx,avx2,fma")
void input() { return; }
template <typename T1, typename... T2>
void input(T1 &x, T2 &...args) {
  ((cin >> x), input(args...));
}
void wrt() {
  cout << "\n";
  return;
}
template <typename T1, typename... T2>
void wrt(T1 x, T2... args) {
  ((cout << x << ' '), wrt(args...));
}
template <typename T>
void inputlst(T &lst, int x, int y) {
  for (int i = x; i < y; i++) cin >> lst[i];
}
template <typename T>
void printlst(T &lst, int x, int y) {
  for (int i = x; i < y; i++) cout << lst[i] << ' ';
  wrt();
}
const long long inf = INT64_MAX, MOD = 1e9 + 7, N = 1e5;
int main(int argc, char const *argv[]) {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  long long n, A, temp;
  input(n, A);
  vector<long long> lst(n), ans(n, 0);
  inputlst(lst, 0, n);
  temp = accumulate(lst.begin(), lst.end(), 0ll);
  for (long long i = 0; i < n; ++i) {
    if (A - (temp - lst[i]) > 1) ans[i] += A - (temp - lst[i]) - 1;
    if (lst[i] > A - (n - 1)) ans[i] += (lst[i] - (A - (n - 1)));
  }
  printlst(ans, 0, n);
  return 0;
}
