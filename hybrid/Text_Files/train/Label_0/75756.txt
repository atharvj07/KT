n = int(input())
in_ = []
for i in range(n):
	(ai1, ai2) = map(int, input().split())
	in_.append((ai1, ai2))
if in_[0][0] in in_[in_[0][1] - 1]:
	(l, r) = (in_[0][1], in_[0][0])
else:
	(l, r) = (in_[0][0], in_[0][1])
for i in range(n):
	print(r, end=' ')
	if in_[l - 1][0] != r:
		(l, r) = (r, in_[l - 1][0])
	else:
		(l, r) = (r, in_[l - 1][1])
print()
