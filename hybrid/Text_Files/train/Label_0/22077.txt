int gcd(int a, int b){
  return b == 0 ? a : gcd(b, a % b);
}

int main() {
  int H, h, m, s, a;
  while(cin >> H >> h >> m >> s, H | h | m | s) {
    a = -1;
    while(true) {
      int d = 119 * H - 1;
      int n = 60 * m * (H + 1) + 60 * 60 * (h + a * H);
      // LOG("n :%d d:%d\n", n, d);
      if(s * d <= n && n < 60 * d && n != m * d) {
        int g = gcd(n, d);
        // LOG("g %d\n", g);
        n /= g;
        d /= g;
        printf("%d %d %d %d\n", h, m, n, d);
        break;
      }

      a++;
      if(a > 1) {
        a = -1;
        s = 0;
        m++;
        if(m >= 60) {
          m = 0;
          h++;
          if(h >= H) {
            h = 0;
          }
        }
      }
    }
  }
}
