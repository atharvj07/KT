import sys
input = sys.stdin.readline

def f(x):
	ret = int(((1 + 8 * x) ** 0.5 - 1) // 2)
	if ret * (ret + 1) // 2 <= x:
		return ret + 1
	else:
		return ret
t = int(input())
for _ in range(t):
	n = int(input())
	A = []
	while n > 1:
		A.append(f(n))
		n -= f(n) * (f(n) - 1) // 2
	if n > 0:
		A.append(2)
	if len(A) == 1:
		print('1' + '3' * A[0] + '7')
	else:
		ans = ''
		for i in range(len(A) - 1):
			a = A[i] - A[i + 1]
			ans += '7' + '3' * a
		ans += '7' + '3' * A[-1] + '1'
		print(ans[::-1])
