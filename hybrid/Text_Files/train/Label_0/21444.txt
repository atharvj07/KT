#include <bits/stdc++.h>
using namespace std;
int n, q;
set<int> v;
vector<int> u[307200];
int main() {
  scanf("%d%d", &n, &q);
  int c = 0;
  for (int i = 0, t, x; i < q; ++i) {
    scanf("%d%d", &t, &x);
    if (t == 1) {
      ++c;
      u[x].push_back(c);
      v.insert(c);
    } else if (t == 2) {
      for (int y : u[x]) v.erase(y);
      u[x].clear();
    } else {
      for (; v.size() && *v.begin() <= x;) v.erase(v.begin());
    }
    printf("%u\n", v.size());
  }
}
