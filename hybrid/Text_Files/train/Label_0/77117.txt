import math

def power(x, y, m):
	if y == 0:
		return 1
	ans = power(x, y // 2, m)
	ans = ans * ans % m
	if y % 2 == 1:
		ans = ans * x % m
	return ans
arr = [int(i) for i in input().split()]
(x, n, v, ans) = (arr[0], arr[1], [], 1)
for i in range(2, int(math.sqrt(x) + 1)):
	if x % i == 0:
		v.append(i)
		while x % i == 0:
			x //= i
if x != 1:
	v.append(x)
for i in v:
	(p, count) = (i, 0)
	while p <= n:
		count += n // p
		p *= i
	ans = ans * power(i, count, int(1000000007)) % 1000000007
print(ans)
