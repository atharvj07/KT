#include <bits/stdc++.h>
using namespace std;
typedef struct {
  string table[200001];
  long long int tam;
  long long int maiorCaminho;
} Hash;
long long int primos[10] = {38611, 54583, 24077, 15527, 4507,
                            6869,  14149, 33289, 50459, 69371};
long long int h(string key) {
  long long int s = key.length();
  long long int x = 0;
  for (long long int i = 0; i < s; i++) {
    x += ((primos[i % 10]) * ((long long int)key[i] * (i + 1))) % 200001;
  }
  return x % 200001;
}
Hash iniHash() {
  Hash aux;
  aux.tam = 0;
  aux.maiorCaminho = 0;
  for (long long int i = 0; i < 200001; i++) aux.table[i] = "";
  return aux;
}
void addHash(Hash *map, string str) {
  long long int pos = h(str);
  if (map->table[pos] != str) {
    if (map->table[pos] == "") {
      map->table[pos] = str;
      map->tam++;
    } else {
      long long int pos2 = pos;
      long long int qt = 0;
      while (1) {
        qt++;
        pos2 = ((pos2 * primos[qt % 10] * 43) % 200001 +
                (qt * primos[(qt * 67) % 10]) % 200001) %
               200001;
        if (map->table[pos2] == str) {
          break;
        }
        if (map->table[pos2] == "") {
          map->table[pos2] = str;
          map->tam++;
          break;
        }
      }
      if (qt > map->maiorCaminho) map->maiorCaminho = qt;
    }
  }
}
bool findHash(Hash *map, string str) {
  long long int pos = h(str);
  if (map->table[pos] == "")
    return false;
  else if (map->table[pos] == str)
    return true;
  else {
    long long int pos2 = pos;
    for (long long int i = 1; i <= map->maiorCaminho; i++) {
      pos2 = ((pos2 * primos[i % 10] * 43) % 200001 +
              (i * primos[(i * 67) % 10]) % 200001) %
             200001;
      if (map->table[pos2] == str)
        return true;
      else if (map->table[pos2] == "")
        return false;
    }
    return false;
  }
}
int main() {
  long long int n;
  scanf("%lld", &n);
  string friends[n];
  for (long long int i = 0; i < n; i++) {
    cin >> friends[i];
  }
  Hash map = iniHash();
  for (long long int i = n - 1; i >= 0; i--) {
    if (!findHash(&map, friends[i])) {
      addHash(&map, friends[i]);
      cout << friends[i] << '\n';
    }
  }
  return 0;
}
