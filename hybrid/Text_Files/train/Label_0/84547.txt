from bisect import bisect_left
(n, k) = map(int, input().split())
c = list(map(int, input().split()))
a = [[] for _ in range(n)]
for i in range(n * k):
	a[c[i] - 1].append(i)
t = -1
d = [0] * n
cnt = 0
resa = [0] * n
resb = [0] * n
while cnt < n:
	mi = 10000
	mj = -1
	midx = 0
	for j in range(n):
		if d[j] == 0:
			idx = bisect_left(a[j], t)
			if idx < k - 1:
				if mi > a[j][idx + 1]:
					mi = a[j][idx + 1]
					mj = j
					midx = idx
	if mj == -1:
		t = -1
	else:
		resa[mj] = a[mj][midx] + 1
		resb[mj] = a[mj][midx + 1] + 1
		cnt += 1
		d[mj] = 1
		t = mi
for i in range(n):
	print(resa[i], resb[i])
