def divisbleBy25(n):
	count = {}
	ans = 0
	for i in range(len(n) - 1, -1, -1):
		if n[i] not in ('7', '2', '0', '5'):
			continue
		if n[i] == '0' or n[i] == '5':
			if '0' in count:
				ans += count['0'] - i - 1
				ans += len(n) - 1 - count['0']
				break
			count[n[i]] = i
		elif n[i] == '7' or n[i] == '2':
			if '5' in count:
				ans += count['5'] - i - 1
				ans += len(n) - 1 - count['5']
				break
	return ans
output = []
for i in range(int(input())):
	output.append(divisbleBy25(input()))
print(*output, sep='\n')
