class MaleOrFemale:

	def __init__(self):
		self.count = 0
		self.distChars = []

	def countOfChars(self, chars):
		for i in chars:
			if i in self.distChars:
				continue
			else:
				self.count += 1
				self.distChars.append(i)
		return self.count

	def chatOrNot(self, chars):
		count = self.countOfChars(chars)
		if count % 2 == 0:
			print('CHAT WITH HER!')
		else:
			print('IGNORE HIM!')
username = input()
chars = [char for char in username]
inst = MaleOrFemale()
inst.chatOrNot(chars)
