import java.io.* ;
import java.util.*;
import static java.lang.Math.* ;
import static java.util.Arrays.* ;

public class Main {
    
    public static void main(String[] args) {
        
    
            new Main().solveProblem();
        
        out.close();
    }

    static Scanner in = new Scanner(new InputStreamReader(System.in));
    static PrintStream out = new PrintStream(new BufferedOutputStream(System.out));
    
    int n, m, k ;
    Vak[] vakken ;
    public void solveProblem() {        

        n = in.nextInt() ;
        m = in.nextInt() ;
        k = in.nextInt() ;
        
        vakken = new Vak[m] ;
        for( int i = 0 ; i < m ; i++)
            vakken[i] = new Vak(in.nextLong(),in.nextLong(),in.nextInt(),i+1) ;
        
        sort(vakken) ;
        Max max = null ;
        for( int i = 0 ; i < m - n + 1 ; i++ ){
            
            for( long j = vakken[i].a ; j <= vakken[i].b  ; j++ ){
                Max nu =geef( i  , j, 1) ;
                
                if( nu == null )
                    continue ;
                if( max == null  || nu.v + j >= max.v ){
                    max = new Max(vakken[i].ind+ " " + j + '\n' + nu.s,nu.v+j) ;                    
                }
            }
        }
        
        if( max == null ){
            out.println("NO") ;
        }else{
            out.println("YES") ;
            out.print(max.s) ;
        }
            
        
    }
       
    Max[][][] map = new Max[51][101][51] ;
    boolean[][][] done = new boolean[51][101] [51];

    
    private Max geef(int i, long aantal, int dag) {
        
        if( dag == n )
            return new Max("",0) ;
        
        if( done[i][(int) (aantal-vakken[i].a)][dag])
            return map[i][(int) (aantal-vakken[i].a)][dag] ;
        done[i][(int) (aantal-vakken[i].a)][dag] = true ;
        
        
        long nk1 = aantal + k;
        long nk2 = aantal*k ;
        
        Max max = null;
        
        for( int j = i + 1 ; j < m ; j++ ){         
            
            if( vakken[j].c == vakken[i].c)
                continue ;
            if( vakken[j].a <= nk1 && vakken[j].b >= nk1){
                
                Max mnu = geef(j, nk1, dag+1) ;
                
                if( mnu != null ){
                    long nu = nk1 + mnu.v ;
                    
                    if( max == null  || nu >= max.v ){
                        max = new Max(vakken[j].ind+ " " + nk1 + '\n' + mnu.s,nu) ;                 
                    }
                }
            }
            if( vakken[j].a <= nk2 && vakken[j].b >= nk2){
                
                
                Max mnu = geef(j, nk2, dag+1) ;
                if( mnu != null ){
                    long nu = nk2 + mnu.v ;
                    
                    if( max == null  || nu >= max.v ){
                        max = new Max(vakken[j].ind + " " + nk2 + '\n' + mnu.s,nu) ;                    
                    }
                }
            }
    
        }

        return map[i][(int) (aantal-vakken[i].a)][dag]  = max ;
        
    }

    class Max{
        
        String s ;
        long v ;
    
        
        public Max(String s, long v) {
            super();
            this.s = s;
            this.v = v;
        }
        
        
    }

    class Vak implements Comparable<Vak>{
        
        long a ;
        long b ;
        int c ;
        int ind ;
        public Vak(long a, long b, int c, int ind) {
            super();
            this.a = a;
            this.b = b;
            this.c = c;
            this.ind = ind;
        }
        @Override
        public int compareTo(Vak an) {
            return c-an.c;
        }
        
        
        
        
    }
}
