def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())
n = inp()
for _ in range(n):
	m = inp()
	l = inlt()
	l.sort()
	res = 0
	for t in range(m * 2 + 2):
		cnt = 0
		lp = 0
		rp = m - 1
		while lp < rp:
			if l[lp] + l[rp] < t:
				lp += 1
			elif l[lp] + l[rp] > t:
				rp -= 1
			else:
				cnt += 1
				lp += 1
				rp -= 1
		res = max(cnt, res)
	print(res)
