#include <bits/stdc++.h>
using namespace std;
int n, one[100009], whe[100009], hh[100009];
int main() {
  int i, j, k, x;
  while (scanf("%d", &n) + 1) {
    for (i = 1; i <= n; i++) scanf("%d", &one[i]);
    for (i = 1; i <= n; i++) {
      scanf("%d", &x);
      whe[x] = i;
    }
    for (i = 1; i <= n; i++) hh[whe[one[i]]] = i;
    for (i = 1; i < n; i++) printf("%d ", hh[i]);
    printf("%d\n", hh[i]);
  }
  return 0;
}
