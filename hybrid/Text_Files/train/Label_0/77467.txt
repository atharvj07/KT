import math

def lcm(lista):
	k = 1
	for i in lista:
		if i != 0:
			k = k * i // math.gcd(k, i)
	return k

def lyans():
	a = input()
	listn = []
	for i in a:
		listn.append(int(i))
	n = len(listn)
	for i in range(len(a)):
		l = lcm(listn[:n - i - 1])
		nb = int(a[n - i - 1:])
		if i == len(a) - 1:
			star = int(a)
		elif int(a) % l == 0:
			star = nb
		else:
			temp = l - int(a) % l
			star = temp + nb
		for m in range(star, 10 ** (i + 1), l):
			lsm = [int(i) for i in str(m)]
			lsm.append(l)
			lm = lcm(lsm)
			last = int(a[:n - i - 1] + '0' * (i + 1 - len(str(m))) + str(m))
			if last % lm == 0:
				return last
t = int(input())
for ti in range(t):
	print(lyans())
