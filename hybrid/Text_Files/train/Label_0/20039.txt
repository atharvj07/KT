#include <bits/stdc++.h>
#define ll long long

using namespace std;

ll Dp[3030][3030];

int main()
{
    string s,t; cin>>s>>t;

    ll n = s.size();
    ll m = t.size();

    for(ll i=n-1;i>=0;i--)
        for(ll j=m-1;j>=0;j--)
            Dp[i][j] = max( max ( Dp[i][j+1] , Dp[i+1][j] ) , (ll)(s[i]==t[j]) + Dp[i+1][j+1] );

    for(ll i=0,j=0;i<n&&j<m;)
        if(Dp[i][j]==Dp[i][j+1])
            j++;
        else if(Dp[i][j]==Dp[i+1][j])
            i++;
        else
        {
            if(s[i]==t[j])  cout<<s[i];
            i++;j++;
        }

}
