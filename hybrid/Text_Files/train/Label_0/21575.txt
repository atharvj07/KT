#include<bits/stdc++.h>
using namespace std;
int main(){
  int h,w,m;
  cin>>h>>w>>m;
  set<vector<int>> yx;
  vector<vector<int>> sy(h);
  for(int i=0;i<h;i++)
    sy.at(i)={0,i};
  vector<vector<int>> sx(w);
  for(int i=0;i<w;i++)
    sx.at(i)={0,i};
  for(int _=0;_<m;_++){
    int hi,wi;
    cin>>hi>>wi;
    hi--;
    wi--;
    yx.insert({hi,wi});
    sy.at(hi).at(0)++;
    sx.at(wi).at(0)++;
  }
  sort(sy.begin(),sy.end(),greater<vector<int>>());
  sort(sx.begin(),sx.end(),greater<vector<int>>());
  int my=sy.at(0).at(0);
  int mx=sx.at(0).at(0);
  bool f=0;
  for(int i=0;i<h&&sy.at(i).at(0)==my;i++){
    for(int j=0;j<w&&sx.at(j).at(0)==mx;j++){
      if(!yx.count({sy.at(i).at(1),sx.at(j).at(1)})){
        f=1;
        i=h;
        j=w;
      }
    }
  }
  cout<<mx+my-(f?0:1)<<endl;
}
