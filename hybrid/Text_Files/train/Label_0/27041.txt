#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, k;
  scanf("%d %d %d", &n, &m, &k);
  int rem = (m - n);
  int left = (k - 1);
  int right = (n - k);
  int temp = 0, cleft = 0, cright = 0;
  int cnt = 1;
  int prev = -1;
  while (temp <= rem) {
    int ctemp = 1 + cleft + cright;
    if (temp + ctemp <= rem) {
      if (prev == ctemp) {
        int ndelta = (rem - temp);
        cnt += (ndelta / prev);
        temp = rem + 1;
      } else {
        ++cnt;
        temp = temp + ctemp;
        if (cleft != left) cleft += 1;
        if (cright != right) cright += 1;
      }
    } else {
      temp = rem + 1;
    }
    prev = ctemp;
  }
  printf("%d\n", cnt);
  return 0;
}
