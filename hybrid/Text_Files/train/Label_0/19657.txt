#include <bits/stdc++.h>
using namespace std;

struct data{
    int a, b, c;
};


int n, m;
vector<int> v;
vector<data> dv;

bool solve(){
    vector<int> va[11][11];
    bool dp[110][110][11];


    memset(dp,0,sizeof(dp));
    // for(int i=0;i<n;i++){
    //     for(int j=0;j<n;j++){
    //         for(int k=0;k<m;k++){
    //             dp[i][j][k] = false;
    //         }
    //     }
    // }

    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            if(dv[j].a == v[i]) {
                dp[i][i][j] = true;
            }
        }
    }

    for(int i=0;i<m;i++){
        va[dv[i].b][dv[i].c].push_back(i);
    }

    for(int kk=1;kk<n;kk++){
        for(int i=0;i<n-1;i++){
            int k = i + kk;
            if(i + kk >= n) break;
            for(int j=i;j<i+kk;j++){
                for(int g=0;g<m;g++){
                    for(int h=0;h<m;h++){
                        if(dp[i][j][g] && dp[j+1][k][h]){
                            for(int ii=0;ii<va[g][h].size();ii++){
                                dp[i][k][va[g][h][ii]] = true;
                            }
                        }
                    }
                }
            }
        }
    }

    return dp[0][n-1][0];
}

int main(){
    int a, b, c;
    while(cin >> n){
        v.clear();
        dv.clear();
        for(int i=0;i<n;i++){
            cin >> a;
            v.push_back(a);
        }
        cin >> m;
        for(int i=0;i<m;i++){
            cin >> a >> b >> c;
            b--; c--;
            dv.push_back((data){a, b, c});
        }
        cout << (solve() ? "Yes" : "No") << endl;
    }
}
