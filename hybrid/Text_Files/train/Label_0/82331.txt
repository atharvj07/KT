def gcd(a, b):
	if b == 0:
		return a
	return gcd(b, a % b)

def lcm(a, b):
	return a * b // gcd(a, b)

def solve(case_no):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	ans = 1
	pr = 0
	for i in range(n):
		if a[i] - pr > k:
			ans = 1
		else:
			ans += 1
		pr = a[i]
	print(min(ans, n))
t = 1
for i in range(1, t + 1):
	solve(i)
