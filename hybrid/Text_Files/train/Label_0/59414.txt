import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Scanner;
import java.util.StringTokenizer;

public class E {
	private static int OFFSET=2;
	private static int tree[];
	public static void main(String[] args) throws Exception{
		Scanner sc = new Scanner(System.in);
		String s = readString();
		int[] A = new int[s.length()+1];
		int offset = 1;
		while(OFFSET < s.length()+2){
			OFFSET*=2;
		}
		tree = new int[2*OFFSET+1];
		int[] l = new int[s.length()+1];
		int[] r = new int[s.length()+1];
		for(int i = 1; i <= s.length(); i++){
			if(s.charAt(i-1) == '('){
				A[i] = A[i-1] + 1;
				l[i] = l[i-1] + 1;
				r[i] = r[i-1];
			}
			else{
				A[i] = A[i-1] - 1;
				l[i] = l[i-1];
				r[i] = r[i-1] + 1;
			}
			update(i, A[i]);
		}
		int n = readInt();
		for(int i = 0; i < n; i++){
			int a = readInt();
			int b = readInt();
			int off = query(a, b);
			off = off - A[a-1];
			int best = 0;
			if(off < 0){
				best = Math.min(r[b]-r[a-1] + off, l[b]-l[a-1]);
			}
			else{
				best = r[b]-r[a-1];
			}
			System.out.println(2*best);
		}
		
			
	}
	
	private static void update(int i, int v){
		i += OFFSET;
		tree[i] = v;
		i /= 2;
		while(i > 0){
			tree[i] = Math.min(tree[2*i], tree[2*i+1]);
			i/=2;
		}
	}
	public static int query(int l, int r){
		int result = Integer.MAX_VALUE;
		l+= OFFSET -1;
		r+= OFFSET +1;
		while(l/2 != r/2){
			if(l%2 == 0){
				result = Math.min(result, tree[l^1]);
			}
			if(r%2 == 1){
				result = Math.min(result, tree[r^1]);
			}
			l/=2;
			r/=2;
		}
		return result;
	}
	
	static BufferedReader stdin = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer st = new StringTokenizer(" ");
	static String readString() throws Exception{
		while(!st.hasMoreTokens()){
			st = new StringTokenizer(stdin.readLine());
		}
		return st.nextToken();
	}
	static int readInt() throws Exception {
		return Integer.parseInt(readString());
	}
}
