#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long t;
  cin >> t;
  while (t--) {
    bool flag = 1;
    long long n;
    cin >> n;
    vector<long long> a(n);
    vector<long long> obr(n);
    set<pair<long long, long long>> chec;
    set<long long> have;
    vector<long long> num(n + 1, 0);
    for (long long i = 0; i < n; i++) {
      cin >> a[i];
      a[i]--;
      obr[a[i]] = i;
    }
    for (long long i = n - 1; i >= 0; i--) {
      long long pos = obr[i];
      long long nxt = -1, pr = -1;
      auto uk = have.lower_bound(pos);
      if (uk != have.end()) {
        nxt = *uk;
      }
      if (uk != have.begin()) {
        uk--;
        pr = *uk + 1;
      } else
        pr = 0;
      if (nxt == -1) {
        chec.erase({-num[pos], pos});
        num[pos] += (pos - pr + 1);
        chec.insert({-num[pos], pos});
      } else {
        chec.erase({-num[nxt], nxt});
        num[nxt] -= (pos - pr + 1);
        chec.insert({-num[nxt], nxt});
        chec.erase({-num[pos], pos});
        num[pos] += (pos - pr + 1);
        chec.insert({-num[pos], pos});
      }
      have.insert(pos);
      if ((*chec.begin()).first != -num[pos]) {
        flag = 0;
        break;
      }
    }
    if (flag) {
      cout << "Yes" << '\n';
    } else {
      cout << "No" << '\n';
    }
  }
  return 0;
}
