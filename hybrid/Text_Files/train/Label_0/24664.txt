#include <bits/stdc++.h>
using namespace std;
string s;
signed main() {
  cin >> s;
  for (int i = 1; i < s.size() - 1; ++i) {
    if (isupper(s[i]) && isupper(s[i - 1]) && isupper(s[i + 1]) &&
        s[i] != s[i - 1] && s[i - 1] != s[i + 1] && s[i] != s[i + 1]) {
      cout << "Yes\n";
      return 0;
    }
  }
  cout << "No\n";
  return 0;
}
