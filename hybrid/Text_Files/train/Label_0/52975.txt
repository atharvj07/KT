	private static BigInteger lcm(BigInteger p, BigInteger q){
		return p.multiply(q).divide(ggT(p, q));
	}
	
	private static BigInteger ggT(BigInteger a, BigInteger b){
		BigInteger i = a;
		BigInteger j = b;
		BigInteger save = j;
		BigInteger r;
		r = i.mod(j);
		while(r.compareTo(BigInteger.ZERO) != 0){
			save = r;
			i = j;
			j = r;
			r = i.mod(j);
		}
		return save;
	}
	
	public static boolean isPrime(long n) {
		boolean prime = true;
		for (long i = 3; i <= Math.sqrt(n); i += 2)
			if (n % i == 0) {
				prime = false;
				break;
			}
		if (( n%2 !=0 && prime && n > 2) || n == 2) {
			return true;
		} else {
			return false;
		}
	}

	/**
	 * Read one line and split contents to an int array
	 */
	private int[] ints() throws Exception {
		String[] split = readLine().split(" ");
		int[] list = new int[split.length];
		for (int i = 0; i < list.length; i++) {
			list[i] = Integer.parseInt(split[i]);
		}
		return list;
	}
