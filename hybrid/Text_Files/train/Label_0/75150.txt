def mp():
	return map(int, input().split())

def f(s):
	r = 0
	r += min(abs(ord(s[0]) - ord('A')), 26 - abs(ord(s[0]) - ord('A')))
	r += min(abs(ord(s[1]) - ord('C')), 26 - abs(ord(s[1]) - ord('C')))
	r += min(abs(ord(s[2]) - ord('T')), 26 - abs(ord(s[2]) - ord('T')))
	r += min(abs(ord(s[3]) - ord('G')), 26 - abs(ord(s[3]) - ord('G')))
	return r
n = int(input())
s = input()
m = 10 ** 9
for i in range(n - 3):
	m = min(m, f(s[i:i + 4]))
print(m)
