t = int(input())
while t:
	t = t - 1
	(n, m, x, y) = map(int, input().split())
	l = []
	cnt = 0
	hmm = 0
	for i in range(n):
		z = input()
		hmm += z.count('.')
		l.append(list(z))
	for i in range(n):
		for j in range(m - 1):
			z = l[i][j]
			if z == '-':
				continue
			if z == '.' and l[i][j + 1] == '.':
				cnt += y
				l[i][j + 1] = '-'
			elif z == '.':
				cnt += x
		if l[i][m - 1] == '.':
			cnt += x
	print(min(cnt, hmm * x))
