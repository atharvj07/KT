import java.util.Arrays;
import java.util.Calendar;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        try {

            Scanner input = new Scanner(System.in);

            long size = input.nextLong();
            input.nextLine();

            while (--size >= 0 && input.hasNext()) {
                String code = input.nextLine();
                if (code == null || code.isEmpty()) {
                    break;
                }
                System.out.println(inverse(code));
            }
        } catch (Exception ex) {
            System.out.print("Error: " + Arrays.toString(ex.getStackTrace()));
        }
    }

    private static String inverse(final String code) {
        if (code.charAt(0) == 'R' && Character.isDigit(code.charAt(1)) && code.contains("C")) {
            String[] rc = code.split("C");
            String row = rc[0].substring(1);
            long col_n = Long.valueOf(rc[1]);
            StringBuilder col = new StringBuilder();
            while (col_n > 0) {
                long r = col_n % 26;
                if (r == 0) {
                    col.append('Z');
                    col_n = (col_n / 26) - 1;
                } else {
                    col.append((char) ((r - 1) + 'A'));
                    col_n = col_n / 26;
                }
            }
            return col.reverse().toString() + row;
        } else {
            Double col = 0D;
            StringBuilder row = new StringBuilder();

            String newCode = new StringBuilder(code).reverse().toString();

            int countNumber = 0;
            for (Character ch : newCode.toCharArray()) {
                if (countNumber == 0 && Character.isDigit(ch)) {
                    row.append(ch);
                } else {
                    col = col + Math.pow(26, countNumber) * (ch - 'A' + 1);
                    countNumber++;
                }
            }


            return "R" + Long.valueOf(row.reverse().toString()) + "C" + col.longValue();
        }
    }
}
