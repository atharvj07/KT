import java.io.BufferedInputStream;
import java.io.BufferedOutputStream;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Scanner;


public class R296C {
	public static void main(String[] args) {
		Scanner scan = new Scanner(new BufferedInputStream(System.in));
		PrintWriter out = new PrintWriter(new BufferedOutputStream(System.out));
		int n = scan.nextInt();
		int m = scan.nextInt();
		Graph g = new Graph(n, m);
		for (int i = 0; i < m; i ++){
			int a = scan.nextInt() - 1;
			int b = scan.nextInt() - 1;
			g.add(a, b);
		}
		
		g.preEuler();
		g.euler();
		if (g.q.size() % 2 == 0)
			g.q.add(g.q.get(0));
		out.println(g.q.size() - 1);
		boolean rev = false;
		for (int i = 0; i < g.q.size() - 1; i ++){
			if (rev)
				out.println(g.q.get(i + 1) + " " + g.q.get(i));
			else
				out.println(g.q.get(i) + " " + g.q.get(i + 1));
			rev = !rev;
		}
		out.flush();
		out.close();
	}
}


class Graph{
	int v[];
	ArrayList<Integer> q;
	ArrayList<Edge> node[];
	int[] lastIdx;
	boolean[] visited;
	int n, m;
	public Graph(int n, int m) {
		node = new ArrayList[n];
		for (int i = 0; i < n; i ++)
			node[i] = new ArrayList<Edge>();
		q = new ArrayList<Integer>();
		v = new int[n];
		visited = new boolean[n];
		lastIdx = new int[n];
		this.n = n;
		this.m = m;
	}
	
	public void add(int a, int b){
		Edge e1 = new Edge();
		e1.p = b;
		Edge e2 = new Edge();
		e2.p = a;
		e1.rev = e2;
		e2.rev = e1;
		v[a] ++;
		v[b] ++;
		e1.v = e2.v = false;
		node[a].add(e1);
		node[b].add(e2);
	}
	
	public boolean dfs(int p){
		boolean flag = false;
		for (; lastIdx[p] < node[p].size(); lastIdx[p] ++){
			int i = lastIdx[p];
			if (!node[p].get(i).v){
				Edge e = node[p].get(i);
				e.v = e.rev.v = true;
				lastIdx[p] = i + 1;
				dfs(e.p);
				flag = true;
			}
		}
		q.add(p + 1);
		return flag;
	}
	
	public void euler(){
		dfs(0);
	}
	
	public void preEuler(){
		ArrayList<Integer> bad = new ArrayList<Integer>();
		for (int i = 0; i < n; i ++){
			if (v[i] % 2 == 1){
				bad.add(i);
			}
		}
		for (int i = 0; i < bad.size(); i += 2){
			add(bad.get(i), bad.get(i + 1));
			m ++;
		}
	}
}

class Edge{
	int p;
	Edge rev;
	boolean v;
}
