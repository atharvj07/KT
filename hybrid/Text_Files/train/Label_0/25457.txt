#include <bits/stdc++.h>
using namespace std;
bool isprime(int n) {
  if (n < 2) return 0;
  if (n < 4) return 1;
  int l = (int)sqrt(n) + 2;
  for (int i = 2; i <= l && i < n; i++) {
    if (n % i == 0) return 0;
  }
  return 1;
}
int main() {
  int n, m, mst;
  scanf("%d %d", &n, &m);
  for (int i = n;; i++) {
    if (isprime(i)) {
      printf("2 %d\n", i);
      mst = i;
      break;
    }
  }
  m -= (n - 1);
  printf("1 %d 2\n", n);
  for (int i = 2; i < n; i = i + 1) {
    printf("%d %d ", i, i - 1);
    if (i == 2)
      printf("%d\n", mst - n + 1);
    else
      printf("1\n");
  }
  while (m > 0) {
    for (int i = 1; m > 0 && i < n; i++) {
      for (int j = min(n, i + 2); m > 0 && j <= n; j++) {
        if (i == 1 && j == n) continue;
        printf("%d %d 1000000000\n", i, j);
        m--;
      }
    }
  }
  return 0;
}
