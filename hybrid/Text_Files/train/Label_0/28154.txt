#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
int sum[3], a[MAXN];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  long long cnt = 0, wyn = 0;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    if (a[i] != 0)
      ++sum[0];
    else
      ++cnt;
    if (a[i] < 0) swap(sum[0], sum[1]);
    if (a[i] != 0) wyn += sum[1];
  }
  cout << wyn << " " << (n - cnt) * (n - cnt + 1) / 2 - wyn << "\n";
}
