import java.util.*;
public class p2{
  static boolean[] pr=new boolean[100005];
  public static void main(String[] args) {
    Scanner in=new Scanner(System.in);

    int n=in.nextInt();
    pre();
    //for(int i=0;i<100;i++){System.out.println(i+" "+pr[i]);}
    for(int i=1;i<1000;i++){
      int p=n*i+1;
      if(!pr[p]){System.out.println(i);break;}
    }

  }
  public static void pre()    //to find all prime numbers...
  {
  	for(int i=2; i<100005; i++)
  		pr[i] = true;
  	pr[0] = pr[1] = false;

  	for(int i=2; i<100005; i++)
  			for(int j=2*i; j<100005; j+=i)
  				pr[j] = false;
  }
}
