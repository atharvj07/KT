#include <bits/stdc++.h>
using namespace std;
int main() {
  map<int, int> Mappy;
  map<int, int>::iterator it;
  int res = 0;
  long long inp;
  for (int i = 0; i < 4; i++) {
    cin >> inp;
    it = Mappy.find(inp);
    if (it != Mappy.end()) {
      it->second++;
      if (it->second >= 2) {
        res++;
      }
    } else {
      Mappy.insert(pair<int, int>(inp, 1));
    }
  }
  cout << res << endl;
  return 0;
}
