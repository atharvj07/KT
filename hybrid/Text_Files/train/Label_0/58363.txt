import java.util.*;

public class maze2 {
    static int goal;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        String[] first = scan.nextLine().split(" ");
        int height = Integer.parseInt(first[0]); 
        int width = Integer.parseInt(first[1]);
        int k = Integer.parseInt(first[2]);
        int[][] matrix = new int[height][width];
        boolean[][] visited = new boolean[height][width];
        int startr = 0;
        int startc = 0;
        int numdots = 0;

        for (int i = 0; i < height; i++) {
            String line = scan.nextLine();
            for (int j = 0; j < line.length(); j++) {
                if (line.charAt(j) == '#') {
                    matrix[i][j] = 0;
                }
                else {
                    matrix[i][j] = 2;
                    startr = i;
                    startc = j;
                    numdots++; 
                }
            }
        }
        //System.out.println("numdots = " + numdots);
        //System.out.println("walls to add = " + k);
        goal = numdots - k + 1;
        DFS(matrix, startr, startc, visited);
        printMatrix(matrix, height, width); 
    }

    public static void DFS(int[][] matrix, int startr, int startc, boolean[][] visited) {
        int[] dx = {0, -1, 0, 1};
        int[] dy = {-1, 0, 1, 0};
        goal--;
        if (goal <= 0) return;
        matrix[startr][startc] = 1;
        visited[startr][startc] = true;
        //printMatrix(matrix, 3, 4);
        for (int i = 0; i < 4; i++) {
            try {
                if (matrix[startr + dx[i]][startc + dy[i]] == 2 && !visited[startr + dx[i]][startc + dy[i]]) {
                    visited[startr + dx[i]][startc + dy[i]] = true;
                    DFS(matrix, startr + dx[i], startc + dy[i], visited);
                }
            } catch (ArrayIndexOutOfBoundsException e) {
            }
        }
    }

    public static void printMatrix(int[][] matrix, int height, int width) {
        //System.out.println();
        for (int i = 0; i < height; i++) {
            for (int j = 0; j < width; j++) {
                if (matrix[i][j] == 0) {
                    System.out.print("#");
                }
                else if (matrix[i][j] == 2) {
                    System.out.print("X");
                }
                else System.out.print(".");
            }
            System.out.println();
        }
    }
}
