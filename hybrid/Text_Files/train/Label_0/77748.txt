import sys
import math
input = sys.stdin.readline
imp = 'IMPOSSIBLE'
t = int(input())
for test in range(t):
	n = int(input())
	a = list(map(int, input().split(' ')))
	a.sort(reverse=True)
	b = list(map(int, input().split(' ')))
	b.sort(reverse=True)
	bs = sum(b)
	kmax = n
	kmin = 0
	k = n // 2
	while kmin < kmax:
		top = n + k - math.floor((n + k) / 4)
		if top >= n:
			btop = bs
		else:
			btop = sum(b[:top])
		atop = k * 100 + sum(a[:top - k])
		if atop >= btop:
			kmax = k
			k = kmin + (kmax - kmin) // 2
		else:
			kmin = k + 1
			k = kmin + (kmax - kmin) // 2
	print(k)
