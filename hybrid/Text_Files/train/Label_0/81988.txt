def InfinitePrefix(n, x, s):
	rec = [0] * n
	for (i, c) in enumerate(s):
		if c == '0':
			rec[i] = rec[i - 1] + 1
		else:
			rec[i] = rec[i - 1] - 1
	loop = rec[-1]
	res = 0
	if x == 0:
		res = 1
	if loop == 0:
		for r in rec:
			if x == r:
				return -1
		return res
	for r in rec:
		if (x - r) % loop == 0 and (x - r) // loop >= 0:
			res += 1
	return res
t = int(input())
for _ in range(t):
	(n, x) = [int(s) for s in input().split(' ')]
	s = input()
	print(InfinitePrefix(n, x, s))
