#include <bits/stdc++.h>
using namespace std;
vector<int> sizes[2], par[2];
int find_par(int x, int c) {
  if (par[c][x] == x) return x;
  return par[c][x] = find_par(par[c][x], c);
}
void Union_(int x, int y, int c) {
  x = find_par(x, c);
  y = find_par(y, c);
  if (sizes[c][x] < sizes[c][y]) swap(x, y);
  par[c][y] = x;
  sizes[c][x] += sizes[c][y];
}
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < 2; ++i) {
    sizes[i].resize(n, 1);
    for (int j = 0; j < n; ++j) par[i].push_back(j);
  }
  for (int i = 0; i + 1 < n; ++i) {
    int x, y, c;
    scanf("%d%d%d", &x, &y, &c);
    --x, --y;
    Union_(x, y, c);
  }
  auto res = 0ll;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < 2; ++j)
      if (par[j][i] == i) {
        res += 1ll * (sizes[j][i] - 1) * sizes[j][i];
      }
    res +=
        1ll * (sizes[0][find_par(i, 0)] - 1) * (sizes[1][find_par(i, 1)] - 1);
  }
  printf("%lld", res);
  return 0;
}
