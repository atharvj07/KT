#include <bits/stdc++.h>
using namespace std;
template <class T>
T abs(T x) {
  if (x < 0) return -x;
  return x;
}
template <class T>
T sqr(T a) {
  return a * a;
}
const double pi = acos(-1.0);
const double eps = 1e-8;
int ar[1110], n, r, x, i, j;
int main() {
  while (1 == scanf("%d", &n)) {
    for (i = 0; i < n; i++) {
      cin >> r;
      for (j = 0; j < r; j++) {
        cin >> x;
        ar[x]++;
      }
    }
    for (i = 0; i <= 110; i++)
      if (ar[i] == n) printf("%d ", i);
  }
  return 0;
}
