#include <iostream>
#include <math.h>
using namespace std;
int main(){
	int h,w,ok=1;
	char s[50][50];
	cin>>h>>w;
	for(int i=0;i<h;i++){
		for(int j=0;j<w;j++){
		    cin>>s[i][j];
		}
	}
	for(int i=0;i<h;i++){
		for(int j=0;j<w;j++){
			if(s[i][j]=='#'&&(i==0||s[i-1][j]=='.')&&(i==h-1||s[i+1][j]=='.')&&(j==0||s[i][j-1]=='.')&&(j==w-1||s[i][j+1]=='.'))ok=0;
		}
	}
	
	if(ok==1){
		cout<<"Yes"<<endl;
	}else{
		cout<<"No"<<endl;
	}
}
