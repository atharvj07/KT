#include <bits/stdc++.h>
using namespace std;
int main(int argc, char const **argv) {
  int n, q;
  scanf("%d%d", &n, &q);
  vector<int> L(q), R(q);
  vector<int> cum(n + 2, 0);
  for (int i = 0; i < q; ++i) {
    scanf("%d%d", &L[i], &R[i]);
    ++cum[L[i]];
    --cum[R[i] + 1];
  }
  int sections = 0;
  vector<int> cum1(n + 2, 0), cum2(n + 2, 0);
  for (int i = 1; i <= n; ++i) {
    cum[i] += cum[i - 1];
    if (cum[i] == 1)
      cum1[i] = 1;
    else if (cum[i] == 2)
      cum2[i] = 1;
    cum1[i] += cum1[i - 1];
    cum2[i] += cum2[i - 1];
    if (cum[i] > 0) ++sections;
  }
  int max_sections = 0;
  for (int i = 0; i < q; ++i)
    for (int j = i + 1; j < q; ++j) {
      int inL = max(L[i], L[j]);
      int inR = min(R[i], R[j]);
      int can = sections;
      if (inL > inR) {
        can -= cum1[R[i]] - cum1[L[i] - 1];
        can -= cum1[R[j]] - cum1[L[j] - 1];
      } else {
        can -= cum2[inR] - cum2[inL - 1];
        if (L[i] < inL) can -= cum1[inL - 1] - cum1[L[i] - 1];
        if (L[j] < inL) can -= cum1[inL - 1] - cum1[L[j] - 1];
        if (R[i] > inR) can -= cum1[R[i]] - cum1[inR];
        if (R[j] > inR) can -= cum1[R[j]] - cum1[inR];
      }
      max_sections = max(max_sections, can);
    }
  printf("%d\n", max_sections);
  return 0;
}
