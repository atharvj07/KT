#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
const int MAXN = 1e5 + 5;
const long long int MOD = 1e9 + 7;
void Input() {
  string s;
  cin >> s;
  stack<int> st;
  for (int i = 0; i < s.length(); i++) {
    if (!st.size())
      st.push(s[i] - 'A');
    else if (s[i] == 'B')
      st.pop();
    else
      st.push(0);
  }
  cout << st.size() << '\n';
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    Input();
  }
}
