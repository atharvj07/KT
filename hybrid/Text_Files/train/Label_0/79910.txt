(guests, guards) = map(int, input().split())
arrival = input()
inactive = [0] * 26
active = [0] * 26
flag = 0
for i in range(guests):
	each = arrival[i]
	inactive[ord(each) - 65] = i
for i in range(guests):
	each = arrival[i]
	active[ord(each) - 65] = 1
	if sum(active) > guards:
		flag = 1
		break
	if inactive[ord(each) - 65] == i:
		active[ord(each) - 65] = 0
if flag == 1:
	print('YES')
else:
	print('NO')
