N = int(input())
a = [0 for i in range(2 * N)]
if N % 2 == 0:
	print('NO')
else:
	print('YES')
	a[0] = 1
	for i in range(1, N):
		if i % 2 == 1:
			a[i] = a[i - 1] + 3
		else:
			a[i] = a[i - 1] + 1
	a[N] = 2
	for i in range(N + 1, 2 * N):
		if (i - N) % 2 == 1:
			a[i] = a[i - 1] + 1
		else:
			a[i] = a[i - 1] + 3
	for i in range(0, 2 * N):
		print(a[i], end=' ')
