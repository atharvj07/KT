#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, cnt = 1;
  char c1 = '#', c2 = '#';
  string s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'L' || s[i] == 'R') {
      if (c1 == '#')
        c1 = s[i];
      else if (c1 != s[i]) {
        c1 = s[i], cnt++;
        c2 = '#';
      }
    } else if (s[i] == 'D' || s[i] == 'U') {
      if (c2 == '#')
        c2 = s[i];
      else if (c2 != s[i]) {
        c2 = s[i], cnt++;
        c1 = '#';
      }
    }
  }
  cout << cnt << endl;
  return 0;
}
