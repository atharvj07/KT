#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int a[200], b[200];
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  int k = 0, t = 0;
  for (int i = 0; i < n; i++)
    if (a[i] != b[i] && a[i] == 1)
      k++;
    else if (a[i] != b[i] && a[i] == 0)
      t++;
  if (k == 0)
    cout << -1;
  else {
    t++;
    if (t % k == 0)
      cout << t / k;
    else
      cout << t / k + 1;
  }
  return 0;
}
