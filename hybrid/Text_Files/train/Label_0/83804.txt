from operator import attrgetter
n = int(input().strip())

class Kg:
	lists = 0
	order = []

	def __init__(self, i, mm):
		self.index = i
		self.minmax = mm
		self.next = []
		self.todo = []
		self.vol = 0

	def add(self, item):
		self.next.append(item)

	def count(self):
		if self.next:
			if self.minmax:
				self.vol = min((item.vol for item in self.next))
			else:
				self.vol = sum((item.vol for item in self.next))
		else:
			self.vol = 1
			type(self).lists += 1

	def hang(self):
		queue = [self]
		while queue:
			cur = queue.pop()
			type(self).order.append(cur)
			queue.extend(cur.next)
total = [Kg(i, mm) for (i, mm) in enumerate(map(int, input().split()))]
for (i, v) in enumerate(map(int, input().split())):
	total[v - 1].add(total[i + 1])
root = total[0]
root.hang()
for item in reversed(Kg.order):
	item.count()
print(Kg.lists - root.vol + 1)
