#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  for (int i = 0; i < t; ++i) {
    int s;
    string a;
    cin >> s;
    cin >> a;
    if (s < 11)
      cout << "NO" << '\n';
    else {
      bool flag = false;
      int keep;
      for (int j = 0; j < a.size(); ++j) {
        if (a[j] == '8') {
          keep = j;
          flag = true;
          break;
        }
      }
      if (flag == true) {
        if (a.size() - keep >= 11)
          cout << "YES" << '\n';
        else
          cout << "NO" << '\n';
      } else
        cout << "NO" << '\n';
    }
  }
}
