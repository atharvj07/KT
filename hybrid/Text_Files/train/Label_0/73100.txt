from collections import deque
(n, k) = list(map(int, input().split()))
Gg = [set() for _ in range(n + 1)]
(q, nq) = (deque(), deque())
for _ in range(n - 1):
	(u, v) = list(map(int, input().split()))
	Gg[u].add(v)
	Gg[v].add(u)
for u in range(1, n + 1):
	if len(Gg[u]) == 1:
		q.append(u)
step = 0
removed = 0
ok = True
while removed < n - 1:
	each = {}
	for u in q:
		nxt = Gg[u].pop()
		Gg[nxt].remove(u)
		each[nxt] = each.get(nxt, 0) + 1
		removed += 1
		if len(Gg[nxt]) == 0:
			break
		if len(Gg[nxt]) == 1:
			nq.append(nxt)
	if any((v < 3 for (k, v) in each.items())):
		ok = False
		break
	(q, nq) = (nq, deque())
	step += 1
if ok and step == k and (removed == n - 1):
	print('Yes')
else:
	print('No')
