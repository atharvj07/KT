T = int(input())
for _ in range(T):
	(N, K) = map(int, input().split())
	A = list(map(int, input().split()))
	if K == 0:
		print(N)
		continue
	ma = max(A)
	me = 0
	for a in sorted(A):
		if a == me:
			me += 1
	if ma < me:
		print(N + K)
		continue
	st = set(A)
	for i in range(K):
		add = (ma + me + 1) // 2
		if add in st:
			break
		st.add(add)
	print(len(st))
