#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  string s;
  string t;
  cin >> n;
  int lev = 0;
  int th_lev = -1;
  getline(cin, s);
  for (int i = 0; i < n; ++i) {
    getline(cin, s);
    int k = s.find_first_not_of(' ');
    if (k == string::npos) k = s.size();
    s.erase(0, k);
    if (s.empty()) continue;
    int a, b;
    switch (s[1]) {
      case 'r':
        ++lev;
        break;
      case 'h':
        th_lev = lev;
        a = s.find_first_not_of(' ', s.find_first_of('(') + 1);
        b = s.find_first_of(" )", a);
        t = s.substr(a, b - a);
        break;
      case 'a':
        if (lev == th_lev) {
          int a = s.find_first_not_of(' ', s.find_first_of('(') + 1);
          int b = s.find_first_of(" ,", a);
          if (t == s.substr(a, b - a)) {
            a = s.find_first_of('"');
            b = s.find_first_of('"', a + 1);
            cout << s.substr(a + 1, b - a - 1);
            return 0;
          }
          th_lev--;
        }
        lev--;
        break;
    }
  }
  cout << "Unhandled Exception";
  return 0;
}
