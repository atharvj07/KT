import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	D = {}
	for i in range(n):
		line = list(map(int, input().split()))
		D[line[0]] = line
	res = []
	for i in range(m):
		colum = list(map(int, input().split()))
		if colum[0] in D:
			res = colum
	for i in res:
		print(*D[i])
