#include <bits/stdc++.h>
using namespace std;
int main() {
  long n;
  cin >> n;
  multimap<long, long> cows;
  for (long i = 0; i < n; i++) {
    long x, y;
    cin >> x >> y;
    cows.insert(map<long, long>::value_type(x, y));
  }
  long l, r, b, t;
  long lb, lt, rb, rt;
  map<long, long>::iterator temp = cows.begin();
  l = r = temp->first;
  b = t = temp->second;
  lb = rt = temp->second + temp->first;
  lt = rb = temp->second - temp->first;
  for (map<long, long>::iterator cow = cows.begin(); cow != cows.end(); cow++) {
    if (l > cow->first) {
      l = cow->first;
    }
    if (r < cow->first) {
      r = cow->first;
    }
    if (b > cow->second) {
      b = cow->second;
    }
    if (t < cow->second) {
      t = cow->second;
    }
    if (cow->second < -cow->first + lb) {
      lb = cow->second + cow->first;
    }
    if (cow->second > cow->first + lt) {
      lt = cow->second - cow->first;
    }
    if (cow->second < cow->first + rb) {
      rb = cow->second - cow->first;
    }
    if (cow->second > -cow->first + rt) {
      rt = cow->second + cow->first;
    }
  }
  l--;
  r++;
  b--;
  t++;
  lb--;
  lt++;
  rb--;
  rt++;
  cout << ((r - l) + (t - b)) * 2 -
              ((-l + lb - b) + (t - l - lt) + (r + rb - b) + (t + r - rt))
       << endl;
  return 0;
}
