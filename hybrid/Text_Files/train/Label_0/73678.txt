(n, m, b, mod) = map(int, input().split())
a = list(map(int, input().split()))
dp = [[0 for col in range(b + 1)] for row in range(m + 1)]
dp[0][0] = 1
for i in range(n):
	for j in range(1, m + 1):
		for k in range(a[i], b + 1):
			dp[j][k] = (dp[j][k] + dp[j - 1][k - a[i]]) % mod
ans = 0
for i in range(b + 1):
	ans = (ans + dp[m][i]) % mod
print(ans)
