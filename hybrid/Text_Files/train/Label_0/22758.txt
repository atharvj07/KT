#include<bits/stdc++.h>
using namespace std;

int n, inf=1e9, ans=inf;
vector<int>G[2020];
char s[2020];

int mx[2020], mn[2020], sz[2020];

void dfs(int x,int fff){
	sz[x]=s[x]=='1';
	int smx=0, smn=0, bs=0;
	for (auto y:G[x])
		if (y!=fff){
			dfs(y,x);
			smx+=mx[y];
			smn+=mn[y];
			sz[x]+=sz[y];
			bs=max(bs,mn[y]);
		}
	mx[x]=smx;
	mn[x]=0;
	for (auto y:G[x])
		if (y!=fff){
			if (mn[y]==bs){
				if (mn[y]<=smx-mx[y]){
					mn[x]=smx&1;
				}else{
					mn[x]=mn[y]-(smx-mx[y]);
				}
			}
		}
	if (fff){
		mx[x]+=sz[x];
		mn[x]+=sz[x];
	}
}

int fuck(int rt){
	dfs(rt,0);
	if (mn[rt]==0) return mx[rt]/2;
	return inf;
}

int main(){
	cin>>n>>s+1;
	for (int i=1;i<n;++i){
		int x, y; cin>>x>>y;
		G[x].push_back(y);
		G[y].push_back(x);
	}
	for (int i=1;i<=n;++i)
		ans=min(ans,fuck(i));
	if (ans==inf) cout<<-1<<endl;
	else cout<<ans<<endl;
}
