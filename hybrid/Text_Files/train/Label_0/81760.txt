(n, m, d) = map(int, input().split())
arr = list(map(int, input().split()))
tmp = [0] * (d - 1)
ans = list()
for idx in range(m):
	tmp.extend([idx + 1] * arr[idx])
	tmp.extend([0] * (d - 1))
if len(tmp) < n:
	print('NO')
	exit()
clear_zeros = len(tmp) - n
for v in tmp:
	if v == 0 and clear_zeros > 0:
		clear_zeros -= 1
	else:
		ans.append(v)
print('YES')
print(*ans)
