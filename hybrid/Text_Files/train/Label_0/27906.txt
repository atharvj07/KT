#include <bits/stdc++.h>
using namespace std;
int n, x, y, a[100001];
int main() {
  scanf("%d%d%d", &n, &x, &y);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", &a[i]);
  }
  for (int i = 1; i <= n; ++i) {
    for (int j = max(1, i - x); j < i; ++j) {
      if (a[i] >= a[j]) {
        goto next;
      }
    }
    for (int j = i + 1; j <= min(i + y, n); ++j) {
      if (a[i] >= a[j]) {
        goto next;
      }
    }
    printf("%d\n", i);
    break;
  next:;
  }
  return 0;
}
