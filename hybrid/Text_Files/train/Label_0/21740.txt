#include <iostream>
#include <string.h>
using namespace std;

int main()
{
	int n, k, t = 0, c = 0, b = 0, tt = 0, cc = 0, bb = 0;
	int p[100];

	memset(p, 0, sizeof(p));

	cin >> n >> k;

	for (int i = 0; i < k; i++) {
		int a, b;

		cin >> a >> b;
		p[a - 1] = b;
	}

	if (p[0] == 1)
		t = 1;
	else if (p[0] == 2)
		c = 1;
	else if (p[0] == 3)
		b = 1;
	else {
		t = 1;
		c = 1;
		b = 1;
	}

	for (int i = 1; i < n; i++) {
		int ty = t, cy = c, by = b, tp = tt, cp = cc, bp = bb;
		t = 0;
		c = 0;
		b = 0;
		tt = 0;
		cc = 0;
		bb = 0;

		if (p[i] == 1 || p[i] == 0) {
			t = (cy + by + cp + bp) % 10000;
			tt = ty;
		}

		if (p[i] == 2 || p[i] == 0) {
			c = (by + ty + bp + tp) % 10000;
			cc = cy;
		}

		if (p[i] == 3 || p[i] == 0) {
			b = (ty + cy + tp + cp) % 10000;
			bb = by;
		}
	}

	cout << (t + c + b + tt + cc + bb) % 10000 << endl;

	return 0;
}
