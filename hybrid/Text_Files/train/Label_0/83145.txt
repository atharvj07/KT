from sys import stdin
from collections import deque
import math

def mxb(n):
	k = int(math.log(n, 2))
	return 2 ** k
n = int(stdin.readline().strip())
s = list(map(int, stdin.readline().strip().split()))
mx = [mxb(i) for i in s]
for i in range(2, n):
	if mx[i] == mx[i - 1] and mx[i - 1] == mx[i - 2]:
		print(1)
		exit(0)
last = -100
ans = 100000000000
for i in range(n):
	if i > 0:
		last = s[i - 1]
	xor = s[i]
	for j in range(i + 1, n):
		xor = xor ^ s[j]
		if xor < last:
			ans = min(ans, j - i)
		if j != n - 1 and xor > s[j + 1]:
			ans = min(ans, j - i)
		if j == n - 1:
			continue
		c = 1
		xor1 = s[j + 1]
		for k in range(j + 2, n):
			xor1 = xor1 ^ s[k]
			if xor > xor1:
				ans = min(ans, j - i + c)
			c += 1
if ans != 100000000000:
	print(ans)
else:
	print(-1)
