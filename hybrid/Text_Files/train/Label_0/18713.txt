#include <bits/stdc++.h>
using namespace std;
long long vis[100004];
int main() {
  long long t;
  cin >> t;
  {
    while (t--) {
      int n, m;
      cin >> n >> m;
      vector<int> op;
      int i;
      for (i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (!vis[x]) op.push_back(x);
        vis[x] = 1;
      }
      sort(op.begin(), op.end());
      long long ans = 0;
      long long how = 0;
      for (i = op.size() - 1; i >= 0; i--) {
        vis[op[i]] = 0;
        if (op[i] - how <= 0) continue;
        ans++;
        how += m;
      }
      cout << ans << endl;
    }
  }
  return 0;
}
