def solve(arr):
	arr.sort(key=lambda q: q[1] - q[0])
	m = sum((si for (a, b, si) in arr))
	k = s * (m // s)
	n = m - k
	(x, y, z) = (0, 0, 0)
	for (a, b, si) in arr:
		if k >= si:
			k -= si
			z += si * a
		elif k > 0:
			z += k * a
			x = (si - k) * a
			y = (si - k) * b
			k = 0
		else:
			x += si * a
			y += si * b
	return (x, y, z, n)
(n, s) = map(int, input().split())
(arr1, arr2) = ([], [])
for i in range(n):
	(si, ai, bi) = map(int, input().split())
	if ai > bi:
		arr1.append((ai, bi, si))
	else:
		arr2.append((bi, ai, si))
(x1, y1, z1, n1) = solve(arr1)
(x2, y2, z2, n2) = solve(arr2)
d = x1 + x2 if n1 + n2 > s else max(x1 + y2, x2 + y1)
print(z1 + z2 + d)
