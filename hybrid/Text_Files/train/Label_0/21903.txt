#include <bits/stdc++.h>
using namespace std;
template <class T>
inline T abs(T &a) {
  return a > 0 ? a : -a;
}
template <class T>
inline T min(T &a, T &b) {
  return a < b ? a : b;
}
template <class T>
inline T max(T &a, T &b) {
  return a > b ? a : b;
}
inline void read(int &x) {
  char ch;
  while ((ch = getchar()) && !(ch >= '0' && ch <= '9'))
    ;
  x = ch - '0';
  while ((ch = getchar()) && (ch >= '0' && ch <= '9')) x = x * 10 + ch - '0';
}
struct point {
  int x, y;
  point() {}
  point(int _x, int _y) : x(_x), y(_y) {}
};
const int N = 400000;
set<pair<int, int> > S;
struct lj {
  int w, c;
} down[N], up[N];
int a[N], b[N], n, m, T[N], ul, dl;
long long ans;
int cmp(lj a, lj b) { return a.w > b.w; }
int cmp2(lj a, lj b) { return a.w < b.w; }
void change() {
  for (int i = 1; i <= n; i++)
    if (T[i] == 1)
      up[++ul].w = b[i] - a[i], up[ul].c = i;
    else if (T[i] == 2)
      down[++dl].w = b[i] - a[i], down[dl].c = i;
  sort(up + 1, up + ul + 1, cmp2);
  sort(down + 1, down + dl + 1, cmp);
  int i = 1, j = 1;
  while (i <= ul && j <= dl) {
    if (down[j].w > up[i].w) {
      ans -= down[j].w - up[i].w;
      T[up[i].c]++;
      T[down[j].c]--;
      j++, i++;
    } else
      break;
  }
  ul = dl = 0;
  for (int i = 1; i <= n; i++)
    if (T[i] == 0)
      up[++ul].w = b[i], up[ul].c = i;
    else if (T[i] == 1)
      down[++dl].w = a[i], down[dl].c = i;
    else if (T[i] == 2)
      down[++dl].w = b[i] - a[i], down[dl].c = i;
  sort(up + 1, up + ul + 1, cmp2);
  sort(down + 1, down + dl + 1, cmp);
  while (i <= ul && j <= dl - 1) {
    if (down[j].w + down[j + 1].w > up[i].w) {
      ans -= down[j].w + down[j + 1].w - up[i].w;
      T[up[i].c] += 2;
      T[down[j].c]--;
      T[down[j + 1].c]--;
      j += 2, i++;
    } else
      break;
  }
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%d%d", &a[i], &b[i]);
  for (int i = 1; i <= n; i++)
    S.insert(make_pair(a[i] * 2, i * 2)), S.insert(make_pair(b[i], i * 2 + 1));
  while (m) {
    int p = S.begin()->first, q = S.begin()->second;
    if ((q & 1) && m < 2) {
      S.erase(*S.begin());
      continue;
    }
    if (q & 1) {
      if (T[q / 2] != 0) {
        S.erase(*S.begin());
        continue;
      }
      ans += (long long)p;
      T[q / 2] = 2;
      S.erase(*S.begin());
      m -= 2;
      continue;
    }
    if (T[q / 2] == 0) {
      m--;
      ans += (long long)p / 2;
      T[q / 2] = 1;
      S.erase(*S.begin());
      S.insert(make_pair((b[q / 2] - a[q / 2]) * 2, q));
      continue;
    }
    if (T[q / 2] == 1) {
      m--;
      ans += (long long)p / 2;
      T[q / 2] = 2;
      S.erase(*S.begin());
      continue;
    }
    S.erase(*S.begin());
  }
  change();
  printf("%lld\n", ans);
  for (int i = 1; i <= n; i++) printf("%d", T[i]);
}
