import java.util.*;
import java.io.*;


public class contest182C {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[2 * n - 1];
        
        int minus = 0;
        int sum  = 0;
        for (int i = 0; i < arr.length; i++) {
            arr[i] = sc.nextInt();
            if(arr[i] < 0) {
                minus ++;
                sum += (arr[i]*(-1));
            }else sum += arr[i];
        }
        
        if(minus % 2 == 0) {
            System.out.println(sum);
            return ;
        }
         if(n % 2 == 1) {
             System.out.println(sum);
             return;
         }else{
            int min = Integer.MAX_VALUE;
            for (int i = 0; i < arr.length; i++) {
                if(arr[i] < 0) min = Math.min(min, arr[i] * (-1));
                else min = Math.min(min, arr[i]);
            }
            System.out.println(sum - 2 * min);
        }
        
    }
}
