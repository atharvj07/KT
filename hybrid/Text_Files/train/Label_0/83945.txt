num = int(input())
area = []
for i in range(num):
	(number, letter) = (int(input()), input())
	area.append(letter)
new_area = []
for line in area:
	if 'U' in line:
		new_area.append(line.replace('U', 'D'))
	elif 'D' in line:
		new_area.append(line.replace('D', 'U'))
	else:
		new_area.append(line)
print(*new_area, sep='\n')
