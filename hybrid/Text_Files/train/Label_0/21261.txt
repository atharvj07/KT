#include <bits/stdc++.h>
using namespace std;
const long long N = 1e6 + 5;
long long n, m;
long long a[N], b[N], x[N], y[N];
map<long long, long long> ans;
int32_t main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  cin >> n;
  for (long long i = 1; i <= n; i++) {
    cin >> a[i] >> x[i];
    ans[a[i]] = x[i];
  }
  cin >> m;
  for (long long i = 1; i <= m; i++) {
    cin >> b[i] >> y[i];
    ans[b[i]] = max(ans[b[i]], y[i]);
  }
  long long answer = 0;
  for (auto it : ans) answer += it.second;
  cout << answer;
  return 0;
}
