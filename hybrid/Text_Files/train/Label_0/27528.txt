#include <bits/stdc++.h>
using namespace std;
int n, m, a, b;
double price = 2000000000;
int read() {
  int ret = 0, neg = 1;
  char c = getchar();
  while (c < '0' || c > '9') {
    if (c == '-') neg = -1;
    c = getchar();
  }
  while (c >= '0' && c <= '9') {
    ret = ret * 10 + c - '0';
    c = getchar();
  }
  return ret * neg;
}
int main() {
  n = read();
  m = read();
  for (int i = 1; i <= n; i++)
    price = min(price, (double)((double)read() / ((double)read())));
  printf("%.8lf", price * m);
}
