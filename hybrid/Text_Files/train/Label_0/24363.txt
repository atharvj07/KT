#include<iostream>
#include<algorithm>
#include<string>
#include<vector>
using namespace std;
typedef pair<pair<int, int>, pair<int, int>> Pr;
signed main() {
	int n;
	while (cin >> n, n) {
		Pr p; vector<Pr>v;
		for (int i = 0; i < n; i++) {
			cin >> p.second.first >> p.second.second >> p.first.first >> p.first.second;
			p.first.first = 11 - p.first.first;
			v.push_back(p);
		}
		sort(v.begin(), v.end());
		int f[1200]; for (int i = 1; i <= 1000; i++)f[i] = 0;
		vector<int>ans;
		for (Pr i : v) {
			if (ans.size() < 10) {
				if (f[i.second.second] < 3) {
					ans.push_back(i.second.first);
					f[i.second.second]++;
				}
			}
			else if (ans.size() < 20) {
				if (f[i.second.second] < 2) {
					ans.push_back(i.second.first);
					f[i.second.second]++;
				}
			}
			else if (ans.size() < 26) {
				if (f[i.second.second] < 1) {
					ans.push_back(i.second.first);
					f[i.second.second]++;
				}
			}
		}
		for (int i : ans)cout << i << endl;
	}
}
