const int maxn = 1000007;

int n;
int a[maxn];
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", a + i);
  for (int i = 0; i < n; i++) a[i]--;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    while (a[i] != i) {
      std::swap(a[i], a[a[i]]);
      cnt++;
    }
  }
  if ((3 * n - cnt) % 2 == 0)
    puts("Petr");
  else
    puts("Um_nik");
  return 0;
}
