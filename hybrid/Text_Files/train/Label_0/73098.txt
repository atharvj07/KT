(n, m) = input().split()
resultado = int(n) * int(m)
boleano = True
for x in range(int(n)):
	lista = list(map(str, input().split()))
	for y in range(int(m)):
		if lista[y] == 'W' or lista[y] == 'B' or lista[y] == 'G':
			boleano = True
		else:
			boleano = False
			break
	if boleano == False:
		break
if boleano:
	print('#Black&White')
else:
	print('#Color')
