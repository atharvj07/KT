xx = []
yy = []
for _ in range(3):
	(x, y) = map(int, input().split())
	xx.append(x)
	yy.append(y)
print(max(xx) - min(xx) + max(yy) - min(yy) + 1)
w = sorted(zip(xx, yy))
for x in range(w[0][0], w[1][0]):
	print(x, w[0][1])
for x in range(w[2][0], w[1][0], -1):
	print(x, w[2][1])
for y in range(min(yy), max(yy) + 1):
	print(w[1][0], y)
