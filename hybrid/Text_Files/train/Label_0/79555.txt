t = int(input())
results = []
for i in range(t):
	(n, a, b) = map(int, input().split())
	digits = input()
	if b > 0:
		results.append(n * (a + b))
		continue
	current = None
	count = 0
	for j in range(n):
		if current != digits[j]:
			count += 1
			current = digits[j]
	results.append(n * a + b * (count // 2 + 1))
for i in range(t):
	print(results[i])
