#include <bits/stdc++.h>
using namespace std;
long long int mpow(long long int a, long long int n,
                   long long int m = 1000000007) {
  a %= m;
  long long int r = 1;
  while (n) {
    if (n & 1) r = (r * a) % m;
    a = (a * a) % m;
    n >>= 1;
  }
  return r;
}
int main(void) {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long int x, k;
  cin >> x >> k;
  if (x == 0) {
    cout << 0 << '\n';
    return 0;
  }
  long long int p = mpow(2, k);
  long long int sub = ((p) * (p - 1 + 1000000007) % 1000000007) % 1000000007;
  long long int y = (p * 2) % 1000000007;
  long long int add = (y * (x % 1000000007)) % 1000000007;
  add = (add * p) % 1000000007;
  long long int ans = (add - sub + 1000000007) % 1000000007;
  ans = (ans * mpow(p, 1000000007 - 2)) % 1000000007;
  cout << ans << '\n';
}
