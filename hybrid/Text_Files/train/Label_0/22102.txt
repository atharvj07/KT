#include <bits/stdc++.h>
using namespace std;
int fr[1007][1007], a[1007][1007], n, m, k;
int res[1000111], Count = 0;
int xx[] = {0, 0, 1, -1}, yy[] = {1, -1, 0, 0};
void bfs(int i, int j) {
  queue<pair<int, int> > p;
  p.push({i, j});
  fr[i][j] = ++Count;
  int sum = 0;
  while (p.size()) {
    int x = p.front().first;
    int y = p.front().second;
    p.pop();
    for (int i = 0; i < 4; i++) {
      int u = x + xx[i];
      int v = y + yy[i];
      if (a[u][v] && !fr[u][v]) {
        fr[u][v] = Count;
        p.push({u, v});
      }
      if (!a[u][v]) sum++;
    }
  }
  res[Count] = sum;
}
int main() {
  if (fopen("text.cpp", "r")) freopen("text.cpp", "r", stdin);
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      char x;
      cin >> x;
      if (x == '*')
        a[i][j] = 0;
      else
        a[i][j] = 1;
    }
  }
  for (int i = 2; i < n; i++) {
    for (int j = 2; j < m; j++) {
      if (fr[i][j] || !a[i][j]) continue;
      bfs(i, j);
    }
  }
  for (; cin >> n >> m; cout << res[fr[n][m]] << '\n')
    ;
}
