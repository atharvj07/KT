n = int(input())
d = 40000
y = 0
cor = True
for i in range(n):
	(t, s) = input().split()
	t = int(t)
	if y == 0 and s[0] != 'S':
		cor = False
		break
	if y == 20000 and s[0] != 'N':
		cor = False
		break
	if s[0] == 'N':
		y -= t
		if y < 0:
			cor = False
			break
	elif s[0] == 'S':
		y += t
		if y > 20000:
			cor = False
			break
if y != 0:
	cor = False
if cor:
	print('YES')
else:
	print('NO')
