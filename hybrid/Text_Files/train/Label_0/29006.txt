#include <bits/stdc++.h>
using namespace std;
int n, x, y, xcnt, ycnt;
bool xvis[105], yvis[105];
int main() {
  cin >> n;
  while (n--) {
    cin >> x >> y;
    if (!xvis[x]) {
      xvis[x] = 1;
      xcnt++;
    }
    if (!yvis[y]) {
      yvis[y] = 1;
      ycnt++;
    }
  }
  printf("%d", min(xcnt, ycnt));
  return 0;
}
