n = int(input())
arr = list(map(int, input().split()))
odd = []
odd_copy = []
even = []
even_copy = []
for elem in arr:
	if elem % 2 == 0:
		even.append(elem)
		even_copy.append(elem)
	else:
		odd.append(elem)
		odd_copy.append(elem)
even.sort()
odd.sort()
even_copy.sort()
odd_copy.sort()
if len(even) > 0:
	el = even.pop()
	while True:
		if el % 2 == 0:
			if len(odd) > 0:
				el = odd.pop()
			else:
				break
		elif len(even) > 0:
			el = even.pop()
		else:
			break
even_sum = sum(even + odd)
if len(odd_copy) > 0:
	el = odd_copy.pop()
	while True:
		if el % 2 == 0:
			if len(odd_copy) > 0:
				el = odd_copy.pop()
			else:
				break
		elif len(even_copy) > 0:
			el = even_copy.pop()
		else:
			break
odd_sum = sum(even_copy + odd_copy)
print(min(even_sum, odd_sum))
