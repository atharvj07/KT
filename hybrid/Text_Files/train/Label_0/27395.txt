#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main()
{
    ios::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    ll t,n,i,sum,now;
    cin>>t;
    while(t--)
    {
        cin>>n;sum=0;
        ll mx=INT_MIN;
        for(i=0;i<n;i++)
        {
            cin>>now;
            sum+=now;
            mx=max(mx,now);
        }
        mx=max(((sum+n-2)/(n-1)),mx);
        cout<<(n-1)*mx-sum<<"\n";
    }
}
