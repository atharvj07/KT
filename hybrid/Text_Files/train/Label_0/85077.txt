def res(l, n):
	if n == 1:
		return 1
	flag = 0
	for i in range(n - 1):
		if abs(l[i] - l[i + 1]) <= 1:
			flag = 1
		else:
			return 0
	if flag == 1:
		return 1
t = int(input())
for i in range(t):
	n = int(input())
	l = list(map(int, input().split()))
	l.sort()
	ans = res(l, n)
	if ans == 1:
		print('YES')
	else:
		print('NO')
