import math as mt
import sys
I = lambda : list(map(int, input().split()))
(n,) = I()
maxint = mt.inf
s = list(input())
t = list(input())
if sorted(s) != sorted(t):
	print(-1)
	sys.exit()

def ma(a, size):
	max_so_far = -maxint - 1
	max_ending_here = 0
	for i in range(0, size):
		max_ending_here = max_ending_here + a[i]
		if max_so_far < max_ending_here:
			max_so_far = max_ending_here
		if max_ending_here < 0:
			max_ending_here = 0
	return max_so_far
ar = [0] * n

def mi(a, size):
	max_so_far = maxint - 1
	max_ending_here = 0
	for i in range(0, size):
		max_ending_here = max_ending_here + a[i]
		if max_so_far > max_ending_here:
			max_so_far = max_ending_here
		if max_ending_here > 0:
			max_ending_here = 0
	return abs(max_so_far)
for i in range(n):
	if s[i] == '1' and t[i] == '0':
		ar[i] = 1
	elif s[i] == '0' and t[i] == '1':
		ar[i] = -1
print(max(ma(ar, n), mi(ar, n)))
