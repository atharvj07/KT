n = int(input())
a = []
b = []
check = 0
for i in range(n):
	k = int(input())
	if k > 0:
		check = 1
		a.append(k)
		continue
	check = 0
	b.append(abs(k))
arr = sum(a)
brr = sum(b)
flag = 1
if arr > brr:
	print('first')
elif brr > arr:
	print('second')
else:
	for i in range(min(len(a), len(b))):
		if a[i] > b[i]:
			print('first')
			flag = 0
			break
		elif b[i] > a[i]:
			print('second')
			flag = 0
			break
	if flag:
		if len(a) > len(b):
			print('first')
		elif len(b) > len(a):
			print('second')
		elif check:
			print('first')
		else:
			print('second')
