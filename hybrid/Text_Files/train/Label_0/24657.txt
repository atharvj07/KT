#include <bits/stdc++.h>
using namespace std;
int n, x, ans = 0, a[200100], cycleAt, root = -1, p[200100];
bool vis[200100] = {0}, isCycle = false, onStack[200100] = {0};
void dfs(int u) {
  vis[u] = 1;
  int v = p[u];
  if (v == u) return;
  onStack[u] = 1;
  if (vis[v]) {
    if (onStack[v]) {
      isCycle = true;
      cycleAt = v;
    }
  } else {
    vis[v] = 1;
    dfs(v);
  }
  onStack[u] = 0;
}
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &x);
    if (x == i) root = x;
    p[i] = x;
  }
  set<int> st;
  for (int i = 1; i <= n; i++) {
    isCycle = false;
    dfs(i);
    if (isCycle) {
      if (root == -1) root = cycleAt;
      p[cycleAt] = root;
      ans++;
    } else if (i != root && p[i] == i) {
      if (root == -1) root = i;
      p[i] = root;
      ans++;
    }
  }
  cout << ans << endl;
  for (int i = 1; i <= n; i++) printf("%d ", p[i]);
  cout << endl;
  return 0;
}
