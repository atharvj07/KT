#include <bits/stdc++.h>
using namespace std;
int tree[60][400005] = {0};
struct STree {
  int n;
  void init(int _n) { n = _n; }
  void add(int idx, int x, int k, int l, int r) {
    if (idx < l || r <= idx) return;
    if (idx == l && l + 1 == r) {
      for (int i = 0; i < 60; ++i) tree[i][k] = (i % x == 0 ? 2 : 1);
    } else {
      add(idx, x, k * 2 + 1, l, (l + r) / 2);
      add(idx, x, k * 2 + 2, (l + r) / 2, r);
      for (int i = 0; i < 60; ++i) {
        tree[i][k] =
            tree[i][k * 2 + 1] + tree[(i + tree[i][k * 2 + 1]) % 60][k * 2 + 2];
      }
    }
  }
  void add(int idx, int x) { add(idx, x, 0, 0, n); }
  int find(int t, int a, int b, int k, int l, int r) {
    if (b <= l || r <= a)
      return 0;
    else if (a <= l && r <= b)
      return tree[t][k];
    else {
      int ret = 0, tmp;
      tmp = find(t, a, b, k * 2 + 1, l, (l + r) / 2), ret += tmp;
      t = (t + tmp) % 60;
      tmp = find(t, a, b, k * 2 + 2, (l + r) / 2, r), ret += tmp;
      return ret;
    }
  }
  int find(int a, int b) { return find(0, a, b, 0, 0, n); }
} T;
int n, a[100005], q;
int main() {
  scanf("%d", &n), T.init(n + 5);
  for (int i = 0; i < n; ++i) scanf("%d", a + i), T.add(i, a[i]);
  scanf("%d", &q);
  while (q--) {
    char tmp[3];
    int x, y;
    scanf("%s%d%d", tmp, &x, &y);
    if (tmp[0] == 'A') {
      x--, y--;
      printf("%d\n", T.find(x, y));
    } else {
      x--, T.add(x, y);
    }
  }
}
