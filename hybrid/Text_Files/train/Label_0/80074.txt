t = int(input())
for tc in range(t):
	n = int(input())
	row = list(map(int, input().split()))
	rowsort = sorted(row)[::-1]
	permutable = True
	for i in range(n):
		if (i - row[i]) % 2 == 0:
			permutable = False
			break
	if not permutable:
		print(-1)
	else:
		ret = []
		for i in range(int(n / 2)):
			rowi = row.index(n - 2 * i)
			if rowi != 0:
				ret.append(rowi + 1)
				row = row[rowi::-1] + row[rowi + 1:]
			rowi = row.index(n - 2 * i - 1)
			if rowi != 1:
				ret.append(rowi)
				ret.append(rowi + 2)
				ret.append(3)
				row = [row[0], row[rowi], row[rowi + 1]] + row[1:rowi] + row[rowi + 2:]
			ret.append(n - 2 * i)
			row = row[n - 2 * i - 1::-1] + row[n - 2 * i:]
		print(len(ret))
		print(' '.join(list(map(str, ret))))
