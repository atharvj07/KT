input = sys.stdin.readline
for ii in range(int(input())):
	(cap1, cap2) = map(int, input().split())
	(ns, na) = map(int, input().split())
	(sweight, aweight) = map(int, input().split())
	if aweight < sweight:
		(aweight, sweight) = (sweight, aweight)
		(ns, na) = (na, ns)
	ans = 0
	for keep in range(ns + 1):
		if keep * sweight > cap1:
			continue
		hiskeep = min(ns - keep, cap2 // sweight)
		myleft = cap1 - keep * sweight
		hisleft = cap2 - hiskeep * sweight
		myaxe = min(myleft // aweight, na)
		hisaxe = min(hisleft // aweight, na - myaxe)
		ans = max(ans, hiskeep + keep + myaxe + hisaxe)
	print(ans)
