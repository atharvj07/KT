#include <bits/stdc++.h>
using namespace std;
long long n, i, a[105], dp[2][105];
int main() {
  std::ios_base::sync_with_stdio(false);
  cin >> n;
  for (i = 0; i < n; i++) cin >> a[i];
  dp[0][0] = 0;
  dp[1][0] = a[0];
  for (i = 1; i < n; i++) {
    if (a[i] == 0) {
      dp[0][i] = dp[1][i] = dp[0][i - 1];
      continue;
    }
    if (a[i - 1] == 1) {
      dp[0][i] = dp[0][i - 1];
      dp[1][i] = 1 + dp[0][i - 1];
      continue;
    }
    if (i > 1 && a[i - 2] == 1) {
      dp[0][i] = dp[1][i - 2];
      dp[1][i] = 1 + dp[0][i - 1];
    } else {
      dp[0][i] = dp[0][i - 1];
      dp[1][i] = 1 + dp[0][i - 1];
    }
  }
  cout << min(dp[0][n - 1], dp[1][n - 1]) << endl;
  return 0;
}
