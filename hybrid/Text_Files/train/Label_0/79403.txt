(s, t) = (input(), input())
(n, m) = (len(s), len(t))
t = '$'.join((t, s))
p = [0]
k = 0
for i in range(1, n + m + 1):
	while k and t[k] != t[i]:
		k = p[k - 1]
	if t[k] == t[i]:
		k += 1
	p.append(k)
ans = [0] * n
sums = [0] * (n + 1)
curs = 0
was = False
j = 0
mod = 1000000007
for i in range(n):
	if p[i + m + 1] == m:
		if not was:
			was = True
			curs = 1
		while j <= i - m:
			curs = (curs + sums[j] + 1) % mod
			j += 1
	ans[i] = curs
	sums[i] = (sums[i - 1] + ans[i]) % mod
print(sum(ans) % mod)
