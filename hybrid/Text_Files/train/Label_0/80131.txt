t = int(input())
for _ in range(t):
	(n, x) = map(int, input().split())
	l = list(map(int, input().split()))
	s = sum(l)
	if s == x:
		print('NO')
	else:
		print('YES')
		ll = []
		c = 0
		for i in l:
			c += i
			if c == x:
				c -= i
				ll.append(i)
			else:
				print(i, end=' ')
		for i in ll:
			print(i, end=' ')
		print()
