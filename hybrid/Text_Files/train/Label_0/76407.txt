def process_sequence(numbers):
	s = numbers
	result = ''
	last = -1
	l = 0
	r = len(s) - 1
	while l <= r:
		leftmost = s[l]
		rightmost = s[r]
		if leftmost < last and rightmost < last:
			break
		if leftmost <= rightmost:
			if leftmost > last:
				result += 'L'
				last = leftmost
				l += 1
			elif rightmost > last:
				result += 'R'
				last = rightmost
				r -= 1
		elif leftmost > rightmost:
			if rightmost > last:
				result += 'R'
				last = rightmost
				r -= 1
			elif leftmost > last:
				result += 'L'
				last = leftmost
				l += 1
	return result

def main():
	n = int(input())
	sequence = list(map(int, input().split()))
	result = process_sequence(sequence)
	print(len(result))
	print(result)
main()
