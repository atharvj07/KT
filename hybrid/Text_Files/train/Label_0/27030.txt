#include <bits/stdc++.h>
using namespace std;
const double pi = acos(-1.0);
const int mod = 100000;
int main() {
  int i, n;
  unsigned long long ans;
  int a[11];
  while (~scanf("%d", &n)) {
    for (i = 5; i >= 1; --i) {
      a[i] = n % 10;
      n /= 10;
    }
    swap(a[2], a[5]);
    swap(a[2], a[3]);
    for (i = 1, n = 0; i <= 5; ++i) {
      n = n * 10 + a[i];
    }
    int t = 5;
    ans = 1;
    while (t--) {
      ans = ans * n;
      ans %= mod;
    }
    printf("%05d", (int)ans);
  }
  return 0;
}
