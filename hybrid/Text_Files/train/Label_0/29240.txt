#include <bits/stdc++.h>
using namespace std;
vector<vector<long long> > v;
long long n, ans = 0;
long long dfs(long long s, long long parent = -1) {
  long long count = 1;
  for (auto u : v[s]) {
    if (u == parent) continue;
    count += dfs(u, s);
  }
  if (count % 2 == 0) ans += 1;
  return count;
}
signed main() {
  cin >> n;
  v.resize(n + 1);
  for (long long i = 0; i < n - 1; i++) {
    long long x, y;
    cin >> x >> y;
    v[x].push_back(y), v[y].push_back(x);
  }
  if (n % 2 == 1) {
    cout << "-1" << endl;
    return 0;
  }
  dfs(1);
  cout << (ans - 1) << endl;
}
