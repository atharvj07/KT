#include<bits/stdc++.h>
using namespace std;

const auto CHARS = 64;
using Bit = bitset<800>;

int to_i(char c) {
  if(isdigit(c)) return c - '0';
  if(islower(c)) return c - 'a' + 10;
  if(isupper(c)) return c - 'A' + 36;
  return 63;
}

int main() {
  int AW, AH, BW, BH;
  cin >> AW >> AH >> BW >> BH;
  vector<string> a(AH), b(BH);
  for(auto v: {&a, &b}) for(auto& i: *v) cin >> i;

  vector<vector<int>> A(AH), B(BH);
  for(auto i = 0; i < AH; ++i) for(auto c: a[i]) A[i].emplace_back(to_i(c));
  for(auto i = 0; i < BH; ++i) for(auto c: b[i]) B[i].emplace_back(to_i(c));

  vector<vector<Bit>> T(CHARS, vector<Bit>(AH));
  for(auto i = 0; i < AH; ++i) for(auto j = 0; j < AW; ++j) {
    if(A[i][j] == to_i('?')) for(auto& k: T) k[i].set(j);
    else                     T[A[i][j]][i].set(j);
  }

  vector<Bit> ok(AH);
  for(auto& i: ok) i.set();
  for(auto i = 0; i < AH - BH + 1; ++i) {
    for(auto y = 0; y < BH; ++y) for(auto x = 0; x < BW; ++x) {
      auto c = B[y][x];
      if(c == to_i('?')) continue;
      ok[i] &= T[c][i+y] >> x;
    }
  }

  auto res = 0;
  for(auto i = 0; i < AH - BH + 1; ++i) for(auto j = 0; j < AW - BW + 1; ++j) if(ok[i][j]) ++res;
  cout << res << endl;
}
