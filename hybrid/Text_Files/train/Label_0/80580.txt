import math

def inv(a, m):
	m0 = m
	x0 = 0
	x1 = 1
	if m == 1:
		return 0
	while a > 1:
		if m == 0:
			return -1
		q = a // m
		t = m
		m = a % m
		a = t
		t = x0
		x0 = x1 - q * x0
		x1 = t
	if x1 < 0:
		x1 = x1 + m0
	return x1

def CRT(num, rem):
	k = 2
	prod = 1
	for i in range(0, k):
		prod = prod * num[i]
	result = 0
	for i in range(0, k):
		pp = prod // num[i]
		is_zero = inv(pp, num[i])
		if is_zero == -1:
			return -1
		result = result + rem[i] * inv(pp, num[i]) * pp
	return result % prod

def gcdExtended(a, b):
	if a == 0:
		return (b, 0, 1)
	(gcd, x1, y1) = gcdExtended(b % a, a)
	x = y1 - b // a * x1
	y = x1
	return (gcd, x, y)

def main():
	(n, m, x, y, vx, vy) = [int(x) for x in input().split(' ')]
	if vx == 0 and x != 0 and (x != n):
		return -1
	elif vx == 0:
		if vy == 0:
			return -1
		elif vy == 1:
			return str(x) + ' ' + str(m)
		elif vy == -1:
			return str(x) + ' ' + str(0)
	if vy == 0 and y != 0 and (y != m):
		return -1
	elif vy == 0:
		if vx == 0:
			return -1
		elif vx == 1:
			return str(n) + ' ' + str(y)
		elif vx == -1:
			return str(0) + ' ' + str(y)
	if vx == 1:
		rest1 = n - x
	else:
		rest1 = x
	if vy == 1:
		rest2 = m - y
	else:
		rest2 = y
	nk = [n, m]
	rest = [rest1, rest2]
	g = math.gcd(n, m)
	if g == 1:
		t = CRT(nk, rest)
	elif rest1 % g != rest2 % g:
		return -1
	else:
		(g, p, q) = gcdExtended(n, m)
		mod = m // g
		lcm = n * mod
		t = rest1 % lcm + n * (p % mod * ((rest2 - rest1) // g % mod) % mod) % lcm
	if t == -1:
		return -1
	else:
		kx = (t - rest1) / n
		ky = (t - rest2) / m
		izq = False
		up = False
		if vx == 1:
			if kx % 2 == 1:
				izq = True
			else:
				izq = False
		elif vx == -1:
			if kx % 2 == 1:
				izq = False
			else:
				izq = True
		if vy == 1:
			if ky % 2 == 1:
				up = False
			else:
				up = True
		elif vy == -1:
			if ky % 2 == 1:
				up = True
			else:
				up = False
		if up and izq:
			return str(0) + ' ' + str(m)
		elif not up and (not izq):
			return str(n) + ' 0'
		elif up and (not izq):
			return str(n) + ' ' + str(m)
		else:
			return '0 0'
print(main())
