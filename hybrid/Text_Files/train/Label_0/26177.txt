#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
#pragma GCC target("avx2", "sse4")
using namespace std;
int main() {
  const long long inf = 1LL << 60;
  const vector<int> p10 = {1, 10, 100, 1000, 10000, 100000, 1000000};
  auto upd = [&](long long &x, long long y) -> void {
    if (y > x) x = y;
  };
  auto solve = [&](int k, const vector<int> &cost) -> vector<long long> {
    vector<long long> f(1e6, -inf);
    f[0] = 0;
    for (int i = 0; i < (int)(cost.size()); ++i) {
      vector<array<long long, 2> > nw(1e6, {-inf, -inf}),
          cur(1e6, {-inf, -inf});
      for (int j = 0; j < (int)(1e6); ++j) nw[j][0] = f[j];
      unsigned lg = 0;
      while ((1 << lg) <= k) ++lg;
      while (lg--) {
        for (int j = 0; j < (int)(1e6); ++j)
          for (int flag = 0; flag < (int)(2); ++flag) cur[j][flag] = -inf;
        for (int j = 0; j < (int)(1e6); ++j)
          for (int flag = 0; flag < (int)(2); ++flag)
            if (nw[j][flag] >= 0) {
              int kc = k >> lg & 1;
              for (int c = 0; c < (int)(2); ++c) {
                if (!flag && c > kc) continue;
                if (j + 1LL * (c << lg) * p10[i] < int(1e6)) {
                  upd(cur[j + 1LL * (c << lg) * p10[i]][flag || c < kc],
                      nw[j][flag] + 1LL * (c << lg) * cost[i]);
                }
              }
            }
        nw.swap(cur);
      }
      for (int j = 0; j < (int)(1e6); ++j) {
        f[j] = max(nw[j][0], nw[j][1]);
      }
    }
    return f;
  };
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int k;
  cin >> k;
  vector<int> cost(6);
  for (int i = 0; i < (int)(6); ++i) cin >> cost[i];
  vector<long long> f(solve(3 * (k - 1), cost));
  vector<long long> ans(1e6, -inf);
  for (int i = 0; i < (int)(1e6 / 3); ++i) ans[i * 3] = f[i];
  for (int i = 0; i < (int)(cost.size()); ++i) {
    vector<long long> nw(1e6, -inf);
    for (int j = 0; j < (int)(1e6); ++j) {
      for (int c = 0; c < (int)(10); ++c) {
        long long w = c % 3 == 0 ? 1LL * (c / 3) * cost[i] : 0;
        if (j + c * p10[i] < 1e6) {
          upd(nw[j + c * p10[i]], ans[j] + w);
        }
      }
    }
    ans.swap(nw);
  }
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    cout << ans[n] << "\n";
  }
  return 0;
}
