q = int(input())
for _ in range(q):
	n = int(input())
	a_list = list(map(int, input().split()))
	a_list.sort()
	area_list = []
	flag = 0
	for i in range(n):
		if a_list[2 * i] == a_list[2 * i + 1] and a_list[-2 * i - 1] == a_list[-2 * (i + 1)]:
			area_list.append(a_list[2 * i] * a_list[-2 * i - 1])
		else:
			flag = 1
			break
	if len(set(area_list)) == 1 and flag == 0:
		print('YES')
	else:
		print('NO')
