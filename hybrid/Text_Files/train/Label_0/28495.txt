#include <bits/stdc++.h>
#pragma GCC optimize("O3")
using namespace std;
using LL = long long;
constexpr int MAXN = 1001;
int n, m, dis[MAXN];
vector<int> G[MAXN];
bool vis[MAXN];
int f(int a) {
  int ret = 0;
  queue<int> q;
  q.push(a), dis[a] = 0;
  while (!q.empty()) {
    a = q.front(), q.pop(), ret = max(ret, dis[a]);
    for (int i : G[a])
      if (dis[i] == 1e9) q.push(i), dis[i] = dis[a] + 1;
  }
  return ret;
}
int solve(int a) {
  vector<int> ver;
  queue<int> q;
  q.push(a), dis[a] = 0;
  while (!q.empty()) {
    ver.push_back(a = q.front()), q.pop();
    vis[a] = true;
    for (int i : G[a]) {
      if (dis[i] == 1e9)
        q.push(i), dis[i] = dis[a] + 1;
      else if ((dis[i] - dis[a]) % 2 == 0)
        return -1;
    }
  }
  int ret = 0;
  for (int i : ver) {
    for (int j : ver) dis[j] = 1e9;
    ret = max(ret, f(i));
  }
  return ret;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> m;
  fill_n(dis + 1, n, 1e9);
  while (m--) {
    int a, b;
    cin >> a >> b;
    G[a].push_back(b), G[b].push_back(a);
  }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (!vis[i]) {
      int tmp = solve(i);
      if (tmp == -1) return cout << "-1\n", 0;
      ans += tmp;
    }
  }
  cout << ans << '\n';
  return 0;
}
