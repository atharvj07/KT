t = int(input())
for _ in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	for i in range(n):
		arr[i] = [arr[i], i]
	arr.sort()
	if n == 1:
		print(0)
		continue
	cur = arr[-1][0]
	ind = n - arr[-1][1]
	su = 0
	for i in range(n - 2, -1, -1):
		if arr[i][0] != cur:
			cur = arr[i][0]
			ind = n - arr[i][1]
			continue
		su += (arr[i][1] + 1) * ind
		ind += n - arr[i][1]
	print(su)
