def two(s, x):
	aa = (s - x) // 2
	a = 0
	b = 0
	for i in range(0, 100):
		nowi = x & 1 << i
		nowa = aa & 1 << i
		if nowi == 0 and nowa == 0:
			continue
		if nowi == 0 and nowa > 0:
			a = 1 << i | a
			b = 1 << i | b
		elif nowi > 0 and nowa == 0:
			a = 1 << i | a
		else:
			return (0, 0)
	return (a, b)

def main():
	(u, v) = [int(i) for i in input().split()]
	if (v - u) % 2 != 0:
		print(-1)
		return 0
	if u == v:
		if u == 0:
			print(0)
			return 0
		else:
			print(1, v)
			return 0
	if u > v:
		print(-1)
		return 0
	if two(v, u) != (0, 0):
		print(2)
		print(two(v, u)[0], two(v, u)[1])
		return 0
	if (v - u) % 2 != 0:
		print(-1)
		return 0
	print(3)
	print((v - u) // 2, (v - u) // 2, u)
main()
