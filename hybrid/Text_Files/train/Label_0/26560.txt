#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int T;
  T = 1;
  while (T--) {
    long long int n;
    cin >> n;
    if (n != 1)
      cout << n << " " << n + 1 << " " << n * (n + 1) << "\n";
    else
      cout << "-1\n";
  }
}
