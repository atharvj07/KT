import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Function;

import static java.util.Arrays.*;

public class ProblemD2 {
    BufferedReader rd;
    private Map<String, Integer> cache = new HashMap<>();

    ProblemD2() throws IOException {
        rd = new BufferedReader(new InputStreamReader(System.in));
        compute();
    }

    private void compute() throws IOException {
        int n = pint();
        int[] pos = solve(n, this::queryExternal);
        out("! "+(pos[0] + 1)+" "+(pos[1] + 1));
    }

    private int[] solve(int n, Function<String, Integer> query) {
        int[] pos = new int[2];
        char[] u = new char[n];
        fill(u, '0');
        int d1 = query.apply(new String(u));
        u[0] = '1';
        int d2 = query.apply(new String(u));
        int k = d1 < d2 ? 0 : 1;
        pos[k] = 0;
        char base = k==0?'0':'1';
        char e = k==0?'1':'0';
        if(k == 1) {
            fill(u, base);
            d1 = query.apply(new String(u));
        }
        int lo = 1;
        int hi = n-1;
        while(hi - lo > 1) {
            int md = (lo + hi) / 2;
            fill(u, base);
            fill(u, 1, md, e);
            int d = query.apply(new String(u));
            int left = (md - 1 + d1 - d) / 2;
            if(left > 0) {
                hi = md - 1;
            } else {
                lo = md;
            }
        }
        for(int i=0;i<3;i++) {
            fill(u, base);
            fill(u, 1, lo, e);
            int d = query.apply(new String(u));
            int left = (lo - 1 + d1 - d) / 2;
            if (left > 0) {
                pos[1-k] = lo - 1;
                break;
            }
            lo++;
        }
        return pos;
    }

    private Integer queryExternal(String s) {
        Integer res = cache.get(s);
        if(res == null) {
            try {
                out("? "+s);
                res = pint();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
            cache.put(s, res);
        }
        return res;
    }

    private int pint() throws IOException {
        return pint(rd.readLine());
    }

    private int pint(String s) {
        return Integer.parseInt(s);
    }

    private static void out(Object x) {
        System.out.println(x);
        System.out.flush();
    }

    public static void main(String[] args) throws IOException {
        new ProblemD2();
    }
}
