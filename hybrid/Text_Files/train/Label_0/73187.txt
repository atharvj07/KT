n = int(input())
s = input()
s1 = s
t = input()
c = 0
R = []
l = list(s)
cond = 1
for ch in set(t):
	if t.count(ch) != s.count(ch):
		cond = 0
if cond:
	if s == t:
		print(0)
	else:
		for i in range(1, n + 1):
			if t[i - 1] != s[i - 1]:
				for j in range(i + 1, n + 1):
					if t[i - 1] == s[j - 1]:
						break
				for k in range(j, i, -1):
					(l[k - 2], l[k - 1]) = (l[k - 1], l[k - 2])
					R.append(k - 1)
					c = c + 1
				s = ''.join(l)
		print(c)
		for i in R:
			print(i, end=' ')
else:
	print(-1)
