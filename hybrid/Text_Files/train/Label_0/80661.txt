import math

def half_the_same(n, a):
	best_divisor = float('-inf')
	a.sort()
	if a[0] == a[-1]:
		return -1
	for i in range(n // 2 + 1):
		if i >= 1 and a[i] == a[i - 1]:
			for divisor in counts_by_divisor:
				counts_by_divisor[divisor] += 1
			counts_by_divisor[float('inf')] -= 1
		else:
			counts_by_divisor = {}
			for j in range(i + 1, n):
				diff = a[j] - a[i]
				for divisor in get_divisors(diff):
					if divisor in counts_by_divisor:
						counts_by_divisor[divisor] += 1
					else:
						counts_by_divisor[divisor] = 1
		for (divisor, count) in counts_by_divisor.items():
			if count >= n // 2 - 1:
				if divisor > best_divisor:
					best_divisor = divisor
		if best_divisor == float('inf'):
			return -1
	return best_divisor

def get_divisors(value):
	if value == 0:
		yield float('inf')
		return
	m = int(math.ceil(math.sqrt(value)))
	for d in range(1, m + 1):
		if value % d == 0:
			e = value // d
			if d < e:
				yield d
				yield e
			elif d == e:
				yield d
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split(' ')))
	print(half_the_same(n, a))
