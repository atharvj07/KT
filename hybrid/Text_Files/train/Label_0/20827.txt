#include <bits/stdc++.h>
using namespace std;
const int Len = 100000;
string S;
int n;
int pre[Len + 5], dp[Len + 5];
int ans[Len + 5][2];
void getPre() {
  pre[0] = -1;
  int j = -1;
  for (int i = 1; i < n; ++i) {
    while (j >= 0 && S[j + 1] != S[i]) j = pre[j];
    if (S[j + 1] == S[i]) ++j;
    pre[i] = j;
  }
}
void solve() {
  memset(dp, 0, sizeof(dp));
  for (int i = n - 1; i >= 0; --i) {
    ++dp[i];
    if (pre[i] != -1) dp[pre[i]] += dp[i];
  }
  int len = 0;
  for (int i = n - 1; i >= 0; i = pre[i])
    ans[len][0] = i + 1, ans[len++][1] = dp[i];
  printf("%d\n", len);
  for (int i = len - 1; i >= 0; --i) printf("%d %d\n", ans[i][0], ans[i][1]);
}
int main() {
  while (cin >> S) {
    n = S.size();
    getPre();
    solve();
  }
  return 0;
}
