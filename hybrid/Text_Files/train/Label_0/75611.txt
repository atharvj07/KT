for _ in range(int(input())):
	n = int(input())
	arr = list(map(int, input().split()))
	ans = True
	i = n - 2
	total = 0
	while i > 0:
		if arr[i] <= arr[i + 1]:
			i -= 1
		else:
			total += arr[i] - arr[i + 1]
			i -= 1
		if arr[i] - total < 0:
			ans = False
			break
	if ans:
		print('YES')
	else:
		print('NO')
