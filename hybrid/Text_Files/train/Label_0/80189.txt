n = int(input())
a = list(map(int, input().split()))
d = {}
for i in range(1, n + 1):
	d[i] = 0
p = []
for i in range(len(a)):
	if a[i] > n:
		p.append(i)
	elif d[a[i]] >= 1:
		p.append(i)
	else:
		d[a[i]] += 1
c = {}
for i in range(1, n + 1):
	c[i] = 0
for i in range(len(a)):
	if a[i] <= n:
		c[a[i]] += 1
o = []
for key in c:
	if c[key] == 0:
		o.append(key)
index = 0
for i in range(len(p)):
	a[p[i]] = o[index]
	index += 1
print(*a)
