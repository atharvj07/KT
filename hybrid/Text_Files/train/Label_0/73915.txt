from sys import stdin
input = stdin.readline

def lastmin(arr):
	curr = None
	cnt = [0] * (max(arr) + 3)
	for i in arr:
		cnt[i] += 1
	for i in range(len(arr)):
		if arr[i] != curr:
			curr = arr[i]
			if cnt[arr[i]] > 1:
				arr[i] += 1
				cnt[arr[i]] += 1
	return len(set(arr))
for i in range(int(input())):
	a = input()
	lst = list(map(int, input().strip().split()))
	print(lastmin(lst))
