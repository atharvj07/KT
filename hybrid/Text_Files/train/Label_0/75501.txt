n = input()
arr = [int(j) for j in input().split()]
count = []
if arr[0] == 1:
	temp = 1
else:
	temp = 0
for i in range(1, len(arr)):
	if arr[i] == 0 and arr[i - 1] == 1:
		count.append(temp)
		temp = 0
	if arr[i] == 1:
		temp = temp + 1
count.append(temp)
max = 0
for i in count:
	if i > max:
		max = i
if len(count) > 1 and arr[0] == 1 and (arr[len(arr) - 1] == 1):
	if max < count[0] + count[len(count) - 1]:
		max = count[0] + count[len(count) - 1]
print(max)
