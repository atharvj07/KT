n = int(input())
l = [int(x) for x in input().split()]
x = 0
count = 0
p = 0
for i in range(n):
	if l[i] == -1:
		x += 1
	elif l[i] < 0:
		count += -1 - l[i]
		x += 1
	elif l[i] == 0:
		p += 1
	else:
		count += l[i] - 1
if p != 0:
	print(count + p)
elif x & 1 == 1:
	print(count + 2)
else:
	print(count)
