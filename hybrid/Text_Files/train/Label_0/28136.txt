#include<bits/stdc++.h>
using namespace std;

using vi=vector<int>;
using vvi=vector<vi>;
using vvvi=vector<vvi>;

const int INF=1e9;
const int SIZE=6002;
const int OFFSET=3001;

void cmin(int &lhs,int rhs){
    lhs=min(lhs,rhs);
}
int main(){
    int N,K;
    cin>>N>>K;
    vector<int> a(N),r(K),b(K);
    for(int i=0;i<N;i++) cin>>a[i];
    for(int i=0;i<K;i++) cin>>r[i];
    for(int i=0;i<K;i++) cin>>b[i];

    sort(a.begin(),a.end());
    sort(r.begin(),r.end());
    sort(b.begin(),b.end());
    vi rsum(K+1),bsum(K+1);
    for(int i=0;i<K;i++) rsum[i+1]=rsum[i]+r[i];
    for(int i=0;i<K;i++) bsum[i+1]=bsum[i]+b[i];
    vi asum(N+1);
    for(int i=0;i<=N;i++) asum[i+1]=asum[i]+a[i];
    auto sumR=[&](int l,int r){
        return rsum[r]-rsum[l];
    };
    auto sumB=[&](int l,int r){
        return bsum[r]-bsum[l];
    };
    auto sumA=[&](int l,int r){
        return asum[r]-asum[l];
    };

    vvvi dp(K+1,vvi(K+1,vi(SIZE+1,INF)));
    dp[0][0][OFFSET]=0;
    for(int i=0;i<=K;i++){
        for(int j=0;j<=K;j++){
            for(int X=0;X<=SIZE;X++){
                int k=N-(i+j)-1;
                if(k==-1) continue;
                int Y=sumR(K-i,K)+sumB(K-j,K)-sumA(N-(i+j),N)-(X-OFFSET);
                if(i==1 && j==1 && X==OFFSET+3){
                    
                }
                if(i+1<=K){
                    int toX=X-a[k]+r[K-i-1];
                    if(0<=toX && toX<=SIZE){
                        cmin(dp[i+1][j][toX],dp[i][j][X]+max(0,X-toX));
                    }
                }
                if(j+1<=K){
                    int toY=Y-a[k]+b[K-j-1];
                    cmin(dp[i][j+1][X]  ,dp[i][j][X]+max(0,Y-toY));
                }
            }
        }
    }
    int res=INF;
    for(int i=0;i<=K;i++){
        for(int j=0;j<=K;j++){
            for(int X=0;X<=SIZE;X++){
                int Y=sumR(K-i,K)+sumB(K-j,K)-sumA(N-(i+j),N)-(X-OFFSET);
                if(sumR(0,K-i)+X-OFFSET>=0 && sumB(0,K-j)+Y>=0 && i+j==N){
                    cmin(res,dp[i][j][X]);
                }
            }
        }
    }
    // cerr<<dp[0][1][OFFSET]<<endl;
    // cerr<<dp[1][1][OFFSET+3]<<endl;
    // cout<<dp[2][1][OFFSET]<<endl;
    cout<<(res==INF ? -1 : res)<<endl;
    
    return 0;
}
