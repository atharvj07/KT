(queue, ice_packs) = map(int, input().split(' '))
empty_hand = 0
for i in range(queue):
	person = input().split(' ')
	if person[0] == '+':
		ice_packs += int(person[1])
	if person[0] == '-':
		if ice_packs >= int(person[1]):
			ice_packs -= int(person[1])
		else:
			empty_hand += 1
print(ice_packs, empty_hand)
