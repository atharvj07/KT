#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:64000000")
using namespace std;
const int INF = ((1 << 31) - 1);
const long long LLINF = (((1LL << 63) - 1LL));
const double eps = 1e-9;
const double PI = 3.14159265358979323846;
vector<string> board;
inline bool good(int i, int j) {
  return 0 <= i && i < board.size() && 0 <= j && j < board[0].size();
}
inline int neighbours(int i, int j, bool left, bool right, bool up, bool down) {
  int res = 0;
  if (up && good(i - 1, j)) res += '.' == board[i - 1][j];
  if (down && good(i + 1, j)) res += '.' == board[i + 1][j];
  if (left && good(i, j - 1)) res += '.' == board[i][j - 1];
  if (right && good(i, j + 1)) res += '.' == board[i][j + 1];
  return res;
}
int main() {
  int lines, cols;
  cin >> lines >> cols;
  for (int i = 0; i < lines; ++i) {
    char buf[1 << 15];
    scanf("%s", buf);
    board.push_back(string(buf));
  }
  bool fail = false;
  for (int i = 0; i < lines; ++i) {
    for (int j = 0; j < cols; ++j) {
      if (board[i][j] != '.') continue;
      vector<pair<int, int> > to_paint;
      if (j + 1 < cols && board[i][j + 1] == '.') {
        to_paint.push_back(make_pair(i, j));
        to_paint.push_back(make_pair(i, j + 1));
        if (j + 2 < cols && board[i][j + 2] == '.' &&
            neighbours(i, j + 2, 0, 1, 0, 1) == 0) {
          to_paint.push_back(make_pair(i, j + 2));
        }
        if (i + 1 < lines && board[i + 1][j] == '.' &&
            neighbours(i + 1, j, 1, 1, 0, 1) == 0)
          to_paint.push_back(make_pair(i + 1, j));
        if (i + 1 < lines && j + 1 < cols && board[i + 1][j + 1] == '.' &&
            neighbours(i + 1, j + 1, 1, 1, 0, 1) == 0)
          to_paint.push_back(make_pair(i + 1, j + 1));
      } else {
        if (i + 1 < lines && board[i + 1][j] == '.') {
          to_paint.push_back(make_pair(i, j));
          to_paint.push_back(make_pair(i + 1, j));
        } else {
          fail = true;
        }
        if (i + 1 < lines && j - 1 >= 0 && board[i + 1][j - 1] == '.' &&
            neighbours(i + 1, j - 1, 1, 0, 0, 1) == 0)
          to_paint.push_back(make_pair(i + 1, j - 1));
        if (i + 1 < lines && j + 1 < cols && board[i + 1][j + 1] == '.' &&
            neighbours(i + 1, j + 1, 0, 1, 0, 1) == 0)
          to_paint.push_back(make_pair(i + 1, j + 1));
        if (i + 2 < lines && board[i + 2][j] == '.' &&
            neighbours(i + 2, j, 1, 1, 0, 1) == 0)
          to_paint.push_back(make_pair(i + 2, j));
      }
      if (fail) break;
      vector<bool> was(10, false);
      for (int k = 0; k < to_paint.size(); ++k) {
        for (int dx = -1; dx <= +1; ++dx)
          for (int dy = -1; dy <= +1; ++dy) {
            if (dx == 0 ^ dy == 0) {
              int ni = to_paint[k].first + dx;
              int nj = to_paint[k].second + dy;
              if (good(ni, nj) && isdigit(board[ni][nj]))
                was[board[ni][nj] - '0'] = true;
            }
          }
      }
      for (int k = 0; k < 10; ++k) {
        if (!was[k]) {
          for (int p = 0; p < to_paint.size(); ++p)
            board[to_paint[p].first][to_paint[p].second] = char('0' + k);
        }
      }
    }
  }
  if (fail)
    cout << -1;
  else {
    for (int i = 0; i < lines; ++i) printf("%s\n", board[i].c_str());
  }
  return 0;
}
