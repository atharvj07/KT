#include<bits/stdc++.h>
using namespace std;

const int maxn=2e4+10;
const int maxnd=maxn<<2;
const int maxm=maxn<<7;

int n,totnd,x[maxn],y[maxn],pos[maxn];
int tote,FIR[maxnd],TO[maxm],NEXT[maxm];
int color,col[maxnd],times,dfn[maxnd],low[maxnd],st[maxnd];
bool vis[maxnd];

void addedge(int u,int v)
{
	TO[++tote]=v;
	NEXT[tote]=FIR[u];
	FIR[u]=tote;
}

struct Data{
	int x,id;
	Data() {}
	Data(int x,int id):x(x),id(id) {}
	bool operator < (const Data &d) const {return x<d.x;}
}D[maxn];

namespace SegTree{
	int id[maxnd];
#define lc (nd<<1)
#define rc (nd<<1|1)
#define mid ((s+t)>>1)

	void build(int nd,int s,int t)
	{
		if (s==t)
		{
			id[nd]=D[s].id^1;
			return;
		}
		id[nd]=++totnd;
		build(lc,s,mid);
		build(rc,mid+1,t);
	}

	void addedge0(int nd,int s,int t)
	{
		if (s==t) return;
		addedge(id[nd],id[lc]);
		addedge(id[nd],id[rc]);
		addedge0(lc,s,mid);
		addedge0(rc,mid+1,t);
	}

	void addedge1(int nd,int s,int t,int l,int r,int u)
	{
		if (l<=D[s].x&&D[t].x<=r)
		{
			bool flag=0;
			flag|=s<=pos[u]&&pos[u]<=t;
//			flag|=s<=pos[u^1]&&pos[u^1]<=t;
			if (!flag) addedge(u,id[nd]);
			else if (s!=t)
			{
				addedge1(lc,s,mid,l,r,u);
				addedge1(rc,mid+1,t,l,r,u);
			}
			return;
		}
		if (s==t) return;
		if (l<=D[mid].x) addedge1(lc,s,mid,l,r,u);
		if (r>=D[mid+1].x) addedge1(rc,mid+1,t,l,r,u);
	}
#undef lc
#undef rc
#undef mid
};

void init()
{
	int i;
	for (i=1;i<=n;i++)
	{
		D[i*2-1]=Data(x[i],i<<1);
		D[i*2]=Data(y[i],i<<1|1);
	}
	totnd=n<<1|1;
	sort(D+1,D+n+n+1);
	for (i=1;i<=n+n;i++)
		pos[D[i].id]=i;
	SegTree::build(1,1,n+n);
}

void tarjan(int u)
{
	vis[u]=1;
	st[++st[0]]=u;
	low[u]=dfn[u]=++times;
	for (int p=FIR[u];p;p=NEXT[p])
	{
		int v=TO[p];
		if (!dfn[v])
		{
			tarjan(v);
			low[u]=min(low[u],low[v]);
		}
		else if (vis[v]) low[u]=min(low[u],dfn[v]);
	}
	if (low[u]==dfn[u])
	{
		color++;
		while (st[st[0]]!=u)
		{
			vis[st[st[0]]]=0;
			col[st[st[0]]]=color;
			st[0]--;
		}
		vis[u]=0;
		col[u]=color;
		st[0]--;
	}
}

bool solve(int lim)
{
	int i;
	tote=0;
	memset(FIR,0,sizeof(FIR));
	SegTree::addedge0(1,1,n+n);
	for (i=1;i<=n+n;i++)
		SegTree::addedge1(1,1,n+n,D[i].x-lim+1,D[i].x+lim-1,D[i].id);
	memset(vis,0,sizeof(vis));
	memset(col,0,sizeof(col));
	memset(dfn,0,sizeof(dfn));
	memset(low,0,sizeof(low));
	times=0; color=0; st[0]=0;
	for (i=2;i<=totnd;i++)
		if (!dfn[i]) tarjan(i);
	for (i=1;i<=n;i++)
		if (col[i<<1]==col[i<<1|1]) return 0;
	return 1;
}

int main()
{
#ifdef h10
	freopen("F.in","r",stdin);
	freopen("F.out","w",stdout);
#endif
	int i;
	scanf("%d",&n);
	for (i=1;i<=n;i++)
	{
		scanf("%d%d",&x[i],&y[i]);
		if (x[i]>y[i]) swap(x[i],y[i]);
	}
	init();
	int L=1,R=1000000000/(n-1),mid;
	while (L<=R)
	{
		mid=(L+R)>>1;
		if (solve(mid))
			L=mid+1;
		else
			R=mid-1;
	}
	printf("%d\n",R);
}
