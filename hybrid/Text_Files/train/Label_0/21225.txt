using ll = long long;

ll fizzbuzz(ll X) { return X / 3 + X / 5 - X / 15; }

ll f(ll X) {
  ll ans = 0;
  ll d = 1, p10 = 1;
  while (10 * p10 <= X) {
    ll c = p10 * 9;
    ll fb = fizzbuzz(p10 * 10 - 1) - fizzbuzz(p10 - 1);
    ans += (c - fb) * d;
    ++d;
    p10 *= 10;
  }
  ll c = X - p10 + 1;
  ll fb = fizzbuzz(X) - fizzbuzz(p10 - 1);
  ans += (c - fb) * d;
  ans += 4 * (X / 3) + 4 * (X / 5);
  return ans;
}

int main() {
  ll s;
  while (cin >> s) {
    ll fail = 0, pass = 1ll << 60;
    while (pass - fail > 1) {
      ll X = (fail + pass) / 2;
      if (f(X) >= s) {
        pass = X;
      } else {
        fail = X;
      }
    }
    if (pass > 1) {
      s -= f(pass - 1);
    }
    string ans = "";
    for (ll X = pass; ans.size() < s + 19; X++) {
      if (X % 15 == 0) {
        ans += "FizzBuzz";
      } else if (X % 3 == 0) {
        ans += "Fizz";
      } else if (X % 5 == 0) {
        ans += "Buzz";
      } else {
        ans += to_string(X);
      }
    }
    ans = ans.substr(s - 1, 20);
    cout << ans << endl;
  }
  return 0;
}
