from sys import stdin
for _ in range(int(input())):
	(n, k) = list(map(int, input().split()))
	lst = list(map(int, input().split()))
	(bot, top) = (lst[0], lst[0])
	for i in range(1, n):
		bot = max(0, max(lst[i], bot - k + 1))
		top = min(lst[i] + k - 1, top + k - 1)
		if bot > top:
			print('NO')
			break
	else:
		if bot <= lst[-1] <= top:
			print('YES')
		else:
			print('NO')
