#include <bits/stdc++.h>
int main() {
  int N, M, i, j, CAM, MAX;
  scanf("%d%d", &N, &M);
  int HELP[N];
  int V[N];
  int AUX[N];
  for (i = 0; i < N; i++) {
    V[i] = 0;
    AUX[i] = 0;
    HELP[i] = 0;
  }
  int MAT[M][4];
  for (i = 0; i < M; i++) {
    for (j = 0; j < 4; j++) {
      scanf("%d", &MAT[i][j]);
    }
    if (MAT[i][0] == 1) {
      for (j = MAT[i][1] - 1; j <= MAT[i][2] - 1; j++)
        AUX[j] = AUX[j] + MAT[i][3];
    } else {
      for (j = MAT[i][1] - 1; j <= MAT[i][2] - 1; j++) {
        if (V[j] != 0) {
          if (MAT[i][3] - AUX[j] < V[j]) V[j] = MAT[i][3] - AUX[j];
        } else
          V[j] = MAT[i][3] - AUX[j];
      }
    }
  }
  CAM = 0;
  for (i = 0; i < M; i++) {
    MAX = V[MAT[i][1] - 1] + HELP[MAT[i][1] - 1];
    if (MAT[i][0] == 1) {
      for (j = MAT[i][1] - 1; j <= MAT[i][2] - 1; j++)
        HELP[j] = HELP[j] + MAT[i][3];
    } else {
      for (j = MAT[i][1] - 1; j <= MAT[i][2] - 1; j++) {
        if (V[j] + HELP[j] > MAX) MAX = V[j] + HELP[j];
      }
      if (MAX != MAT[i][3]) {
        CAM = 1;
        break;
      }
    }
  }
  if (CAM == 0) {
    printf("YES");
    printf("\n");
    for (i = 0; i < N; i++) printf("%d ", V[i]);
  } else {
    printf("NO");
    printf("\n");
  }
  return 0;
}
