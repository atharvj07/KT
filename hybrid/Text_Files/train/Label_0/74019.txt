import sys
r = sys.stdin.readlines()
M = 10 ** 9 + 7
(n, q) = r[0].strip().split()
n = int(n)
q = int(q)
s = r[1]
p = [0]
k = 0
for v in range(n):
	d = int(s[v])
	k += 1 - d
	p.append(k)
for k in range(q):
	(a, b) = r[k + 2].strip().split()
	a = int(a)
	b = int(b)
	l = b - a + 1
	zero = p[b] - p[a - 1]
	sys.stdout.write(str((pow(2, l, M) - pow(2, zero, M) + M) % M))
	sys.stdout.write('\n')
