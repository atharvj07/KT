import sys
import math
import collections
import heapq
import decimal
input = sys.stdin.readline
(n, k) = map(int, input().split())
l = set()
for i in range(k):
	(a, b) = map(int, input().split())
	l.add((a, b))
	l.add((b, a))
p = set([i for i in range(1, n + 1)])
ans = 0
for i in range(1, n + 1):
	if i in p:
		p.remove(i)
		if i != 1:
			ans += 1
		q = collections.deque([i])
		while len(q) > 0:
			t = q.popleft()
			q1 = set()
			for j in p:
				if (t, j) not in l and (j, t) not in l:
					q.append(j)
					q1.add(j)
			for j in q1:
				p.remove(j)
print(ans)
