from math import sqrt
from typing import List, Tuple

def find_distance(x1: int, y1: int, x2: int, y2: int) -> float:
	return sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)

def solve(R: int, x1: int, y1: int, x2: int, y2: int) -> Tuple[float, float, float]:
	ab = find_distance(x1, y1, x2, y2)
	if ab >= R:
		return (x1, y1, R)
	elif ab == 0:
		return (x1 + R / 2, y1, R / 2)
	s = (ab + R) / 2
	Xap = s * (x1 - x2) / ab + x2
	Yap = s * (y1 - y2) / ab + y2
	return (Xap, Yap, s)

def parse_input() -> List[int]:
	return [int(x) for x in input().split()]
(R, x1, y1, x2, y2) = parse_input()
print(' '.join((str(x) for x in solve(R, x1, y1, x2, y2))))
