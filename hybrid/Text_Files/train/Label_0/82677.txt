def nok(first, second):
	mx = max(first, second)
	final = mx
	while final % first != 0 or final % second != 0:
		final += mx
	return final

def nod(first, second):
	(_mx, _mn) = (max(first, second), min(first, second))
	while _mx % _mn != 0:
		(_mx, _mn) = (_mn, _mx % _mn)
	return _mn
ratios = [int(i) for i in input().split()]
if ratios[0] == ratios[1] and ratios[2] == ratios[3]:
	print('0/1')
	quit()
nk = nok(ratios[0], ratios[2])
ratios[1] *= nk // ratios[0]
ratios[0] *= nk // ratios[0]
ratios[3] *= nk // ratios[2]
ratios[2] *= nk // ratios[2]
(left, right) = (max(ratios[1], ratios[3]) - min(ratios[1], ratios[3]), max(ratios[1], ratios[3]))
print(str(left // nod(left, right)) + '/' + str(right // nod(left, right)))
