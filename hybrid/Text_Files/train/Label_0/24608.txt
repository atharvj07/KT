#include <bits/stdc++.h>
using namespace std;
const int maxn = 200009;
const int maxl = 1000000;
int n, sol;
int A[maxn];
int B[2 * maxl + 9];
int D[maxn], d;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &A[i]);
  sort(A, A + n);
  for (int i = 0; i < n; i++) {
    if (A[i] == A[i + 1]) continue;
    D[d++] = A[i];
  }
  for (int i = 1; i < d; i++)
    for (int h = D[i - 1] + 1; h <= D[i]; h++) B[h] = D[i - 1];
  for (int h = D[d - 1] + 1; h <= 2 * maxl; h++) B[h] = D[d - 1];
  for (int i = 0; i < d; i++)
    for (int h = 2; D[i] * h <= 2 * maxl; h++)
      sol = max(sol, B[D[i] * h] % D[i]);
  printf("%d", sol);
}
