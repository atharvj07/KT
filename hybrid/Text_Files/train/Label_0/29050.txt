struct student{
    int club;
    bool seen;
    vector<int> withHim;
};

void tell(student* p,int t){
    p[t].seen = true;
    rep(i, p[t].withHim.size()){
        if(!p[p[t].withHim[i]].seen){
            p[p[t].withHim[i]].club = p[t].club;
            tell(p, p[t].withHim[i]);
        }
    }
}

void initialize(student* p,int t){
    p[t].seen = false;
    rep(i, p[t].withHim.size()){
        if(p[p[t].withHim[i]].seen){
            initialize(p, p[t].withHim[i]);
        }
    }
}

int main() {
    int n, m ,k;
    cin >> n >> m >> k;
    student* students;
    students = new student[n];
    rep(i, n){
        students[i].club = 0;
    }

    rep(i, k){
        int flag, a, b;
        cin >> flag >> a >> b;
        if(flag == 1){
            if(students[a-1].club == 0){
                if(students[b-1].club != 0){
                    students[a-1].club = students[b-1].club;
                    tell(students, a-1);
                    initialize(students,a-1);
                }
                students[a-1].withHim.push_back(b-1);
                students[b-1].withHim.push_back(a-1);
            }else{
                if(students[b-1].club == 0){
                    students[b-1].club = students[a-1].club;
                    tell(students, b-1);
                    initialize(students, b-1);
                    students[a-1].withHim.push_back(b-1);
                    students[b-1].withHim.push_back(a-1);
                }else{
                    if(students[a-1].club != students[b-1].club){
                        cout << i+1 << "\n";
                        return 0;
                    }
                }
            }
        }else{
            if(students[a-1].club == 0){
                students[a-1].club = b;
                tell(students, a-1);
                initialize(students, a-1);
            }else{
                if(students[a-1].club != b){
                    cout << i+1 << "\n";
                    return 0;
                }
            }
        }
    }

    cout << 0 << "\n";

    return 0;
}
