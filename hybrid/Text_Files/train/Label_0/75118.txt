n = int(input())
memory = input()
room_state = ['0'] * 10
empty = [x for x in range(10)]
for state in memory:
	if state == 'L':
		room = min(empty)
		room_state[room] = '1'
		empty.remove(room)
	elif state == 'R':
		room = max(empty)
		room_state[room] = '1'
		empty.remove(room)
	else:
		empty.append(int(state))
		room_state[int(state)] = '0'
print(''.join(room_state))
