import re

def main():
	n = eval(input())
	a = []
	s = []
	s.append(0)
	s.append(0)
	while n:
		n -= 1
		temp = re.split(' ', input())
		k = eval(temp[0])
		for i in range(k >> 1):
			s[0] += eval(temp[i + 1])
		if k & 1:
			a.append(eval(temp[k + 1 >> 1]))
		for i in range(k + 1 >> 1, k):
			s[1] += eval(temp[i + 1])
	a.sort(reverse=True)
	k = 0
	for i in a:
		s[k] += i
		k ^= 1
	print(s[0], s[1])
main()
