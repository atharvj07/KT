n = int(input())
e = []
for i in range(n):
	c = []
	(a, b) = input().split()
	c.append(int(a))
	c.append(int(b))
	d = input().split()
	for j in d:
		c.append(int(j))
	e.append(c)
for i in e:
	k = i[1]
	p = i[2:]
	p.sort()
	if abs(p[0] + k - p[-1]) <= k and abs(p[0] + k - p[0]) <= k:
		print(p[0] + k)
	else:
		print(-1)
