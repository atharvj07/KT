#include <bits/stdc++.h>
using namespace std;
int n, m, l, r, k, R, L, ans;
set<pair<int, int> > S[2];
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  cin >> n >> m;
  S[0].insert({1, n});
  ans = n;
  S[1].insert({-1, n});
  while (m--) {
    cin >> l >> r >> k;
    R = r;
    L = l;
    if (k == 2) {
      while (S[0].upper_bound({l, 0}) != S[0].end()) {
        pair<int, int> aft = *S[0].upper_bound({l, 0});
        if (aft.first > r) break;
        S[0].erase(aft);
        ans -= aft.second - aft.first + 1;
        S[1].erase({-aft.first, aft.second});
        R = max(R, aft.second);
      }
      if (S[1].upper_bound({-l, 0}) != S[1].end()) {
        pair<int, int> bef = *S[1].upper_bound({-l, 0});
        if (bef.second >= l) {
          S[1].erase(bef);
          S[0].erase({-bef.first, bef.second});
          ans -= bef.second + bef.first + 1;
          R = max(R, bef.second);
          L = min(L, -bef.first);
        }
      }
      S[0].insert({L, R});
      S[1].insert({-L, R});
      ans += R - L + 1;
    } else {
      while (S[0].upper_bound({l, 0}) != S[0].end()) {
        pair<int, int> aft = *S[0].upper_bound({l, 0});
        if (aft.first > r) break;
        S[0].erase(aft);
        S[1].erase({-aft.first, aft.second});
        if (aft.second > r) {
          ans -= r - aft.first + 1;
          S[0].insert({r + 1, aft.second});
          S[1].insert({-(r + 1), aft.second});
        } else
          ans -= aft.second - aft.first + 1;
      }
      if (S[1].upper_bound({-l, 0}) != S[1].end()) {
        pair<int, int> bef = *S[1].upper_bound({-l, 0});
        if (bef.second >= l) {
          S[1].erase(bef);
          S[0].erase({-bef.first, bef.second});
          if (-bef.first != l)
            S[1].insert({bef.first, l - 1}), S[0].insert({-bef.first, l - 1});
          if (bef.second > r) {
            S[1].insert({-r - 1, (bef.second)});
            S[0].insert({r + 1, (bef.second)});
            ans -= r - l + 1;
          } else {
            ans -= bef.second - l + 1;
          }
        }
      }
    }
    cout << ans << " ";
  }
}
