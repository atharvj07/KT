import sys
I = sys.stdin.readline

def rails(x):
	(z, ret) = (1, set())
	while x:
		if x & 1:
			ret.add(z)
		z += 1
		x //= 2
	return ret
n = int(I())
a = map(int, I().split())
a = [i for i in a if i > 0]
n = len(a)
(b, e) = ([[] for i in range(n)], {})
for (i, x) in enumerate(a):
	rs = rails(x)
	for r in rs:
		if not r in e:
			e[r] = []
		e[r].append(i)
		if len(e[r]) > 2:
			print(3)
			sys.exit(0)

def bfs(i, j, k):
	dist = 1
	v = [0] * n
	v[i] = 1
	v[j] = 1
	(q, nq) = ([], [])
	q.append(j)
	while len(q):
		while len(q):
			x = q.pop()
			for y in b[x]:
				if v[y] == 0:
					v[y] = 1
					nq.append(y)
					if y == k:
						return dist
		(q, nq) = (nq, q)
		dist += 1
	return n

def find(i):
	ret = n
	for (j, jj) in enumerate(b[i]):
		for (k, kk) in enumerate(b[i][j + 1:]):
			ret = min(ret, bfs(i, jj, kk))
	return 2 + ret
for v in e.values():
	if len(v) > 1:
		b[v[0]].append(v[1])
		b[v[1]].append(v[0])
ans = min((find(i) for i in range(n)))
print(ans if ans <= n else -1)
