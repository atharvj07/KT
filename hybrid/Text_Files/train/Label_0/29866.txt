#include <bits/stdc++.h>
const int N = 1e5 + 5;
int n, m, k;
int main() {
  scanf("%d%d%d", &n, &m, &k);
  if (k >= m) return printf("1.000000\n"), 0;
  if (m - n - k > 0) return printf("0.000000\n"), 0;
  double ans = -1;
  for (int i = 0; i <= k; i++) ans *= m - k + i, ans /= n + 1 + i;
  ans++;
  printf("%.6lf\n", ans);
  return 0;
}
