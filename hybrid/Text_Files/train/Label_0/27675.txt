#include <iostream>
#include <vector>
using namespace std;
int main() {
    int N, K;
    cin >> N >> K;
    vector<int> x(N);
    for(int i=0; i<N; i++) cin >> x[i];

    int ans = 1e9;
    for(int l=0; l<N-K+1; l++) {
        int r=l+K-1;
        ans = min(ans, min(abs(x[r]), abs(x[l])) + x[r]-x[l]);
    }
    cout << ans << endl;

    return 0;
}
