	public static boolean check(Box boxes[], int mid)
	{
		boolean good = checkers(boxes, mid);
		swap(boxes);
		good &= checkers(boxes, mid);
		swap(boxes);

		return good;
	}

	public static boolean checkers(Box boxes[], int mid)

			int x1 = Math.max(boxes[i].x-mid, 0);
			int x2 = Math.min(boxes[i].x+mid, n-1);
			int y1 = Math.max(boxes[i].y-mid, 0);
			int y2 = Math.min(boxes[i].y+mid, m-1);

//			System.out.println(x1 + " " + x2 + " " + y1 + " " + y2);

		int bot = 0;
		int prevX = e[0].x;

		HashMap<Integer, Integer> enders = new HashMap<>();
		
		if(e[0].x != 0 || e[e.length-1].x != n)
		{
//			System.out.println("NO START START");
			top = 0;
			bot = m-1;

				int pos = 0;

							if(cnt == 0 && yes[j].y != m)

	public static void swap(Box boxes[])
	{
		int t = n;
		n = m;
		m = t;

		for(int i = 0; i < boxes.length; i++)
			boxes[i].swap();
	}

	public static class RowEvent implements Comparable<RowEvent>
	{
		int y, c;
		boolean f;

		public RowEvent(int yy, int cnt, boolean ff)
		{
			y = yy;
			c = cnt;
			f = ff;
		}

		//Start to come first - false
		@Override
		public int compareTo(RowEvent o)
		{
			if(y != o.y)
				return Integer.compare(y, o.y);
			return Boolean.compare(f, o.f);
		}
	}

	public static class Event implements Comparable<Event>
	{
		int x, y1, y2;
		boolean f;

		public Event(int xx, int yy1, int yy2, boolean ff)
		{
			x = xx;
			y1 = yy1;
			y2 = yy2;
			f = ff;
		}

		//Start to come first - false
		@Override
		public int compareTo(Event o)
		{
			if(x != o.x)
				return Integer.compare(x, o.x);
			return Boolean.compare(f, o.f);
		}
	}

	public static class Box
	{
		int x, y;

		public Box(int xx, int yy)
		{
			this.x = xx;
			this.y = yy;
		}

		public void swap()
		{
			int t = y;
			y = x;
			x = t;
		}
	}

	public static class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(String s)
		{
			try {
				br = new BufferedReader(new FileReader(s));
			} catch (FileNotFoundException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}

		public FastScanner()
		{
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String nextToken()
		{
			while (st == null || !st.hasMoreElements())
			{
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					// TODO Auto-generated catch block
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt()
		{
			return Integer.parseInt(nextToken());
		}

		long nextLong()
		{
			return Long.parseLong(nextToken());
		}

		double nextDouble()
		{
			return Double.parseDouble(nextToken());
		}

		String next()
		{
			return nextToken();
		}
	}
