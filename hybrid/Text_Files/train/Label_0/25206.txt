#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
const int mod = 1e9 + 7;
void add(int &a, int b) {
  a += b;
  if (a >= mod) a -= mod;
}
int sum(int a, int b) {
  add(a, b);
  return a;
}
int mult(int a, int b) { return (1ll * a * b) % mod; }
int n, k, first[N], inv[N], ans;
map<int, int> x;
int bp(int a, int n) {
  int ans = 1;
  while (n) {
    if (n & 1) ans = mult(ans, a);
    a = mult(a, a);
    n >>= 1;
  }
  return ans;
}
int cnk(int n, int k) {
  if (k > n) return 0;
  int ans = mult(first[n], inv[k]);
  ans = mult(ans, inv[n - k]);
  return ans;
}
int main() {
  ios_base::sync_with_stdio(0);
  clock_t start = clock();
  cin >> n >> k;
  first[0] = inv[0] = 1;
  for (int i = 1; i <= n; i++) {
    first[i] = mult(first[i - 1], i);
    inv[i] = bp(first[i], mod - 2);
  }
  for (int i = 1; i <= n; i++) {
    int l, r;
    cin >> l >> r;
    x[l]++, x[r + 1]--;
  }
  vector<pair<int, int> > all;
  for (auto i : x) all.push_back(make_pair(i.first, i.second));
  for (int i = 0, cur = 0; i + 1 < all.size(); i++) {
    int len = all[i + 1].first - all[i].first;
    cur += all[i].second;
    add(ans, mult(len, cnk(cur, k)));
  }
  cout << ans << endl;
  double duration = ((double)clock() - start) / (double)CLOCKS_PER_SEC;
  return 0;
}
