#include <bits/stdc++.h>
int arr[200];
int main() {
  int n, d, l;
  int ni, np, dmin, dmax;
  scanf("%d %d %d", &n, &d, &l);
  np = n / 2;
  ni = n - np;
  dmin = ni - l * np;
  dmax = l * ni - np;
  bool so = true;
  if (d > dmax || d < dmin)
    printf("-1\n");
  else {
    bool par = false;
    for (int i = 0; i < n; i++) {
      if (!so) break;
      int dp, a = 1;
      if (par)
        np--;
      else
        ni--;
      dmin = ni - l * np;
      dmax = l * ni - np;
      while (1) {
        if (par)
          dp = d + a;
        else
          dp = d - a;
        if (dmin <= dp && dmax >= dp) {
          arr[i] = a;
          d = dp;
          break;
        }
        a++;
        if (a > l) {
          so = false;
          break;
        }
      }
      par = !par;
    }
    if (so)
      for (int i = 0; i < n; i++) {
        printf("%d", arr[i]);
        if (i < n - 1)
          printf(" ");
        else
          printf("\n");
      }
    else
      printf("-1\n");
  }
  return 0;
}
