#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      char t;
      cin >> t;
      if (t != 'B' && t != 'W' && t != 'G') {
        cout << "#Color";
        return 0;
      }
    }
  cout << "#Black&White";
  return 0;
}
