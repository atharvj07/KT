#define ID(x, y) ((y) * w + x)

struct Edge {
    ll to, cost;
};
vector<ll> dijkstra(const vector<vector<Edge>>& G, const ll s) {
    const ll n = G.size();
    vector<ll> dist(n, linf);
    dist[s] = 0;
    priority_queue<P, vector<P>, greater<P>> Q;
    Q.push({0, s});
    while ( !Q.empty() ) {
        P p = Q.top(); Q.pop();
        ll v = p.second;
        ll cost = p.first;
        if (cost > dist[v]) continue;
        each(e, G[v]) {
            if (dist[v] + e.cost < dist[e.to]) {
                dist[e.to] = dist[v] + e.cost;
                Q.push({dist[e.to], e.to});
            }
        }
    }
    return dist;
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    ll h, w; cin >> h >> w;
    char sc, tc; cin >> sc >> tc;
    vector<string> m(h); cin >> m;
    auto inRange = [&](ll x, ll y) {
        return 0 <= x && x < w && 0 <= y && y < h;
    };
    ll s, t;
    vector<vector<Edge>> G(w*h);
    auto add_edge = [&](ll a, ll b, ll c) {
        G[a].pb({b, c});
        G[b].pb({a, c});
    };
    rep(y, h) rep(x, w) {
        if (m[y][x] == sc) s = ID(x, y);
        if (m[y][x] == tc) t = ID(x, y);
        if (m[y][x] == '-') {
            add_edge(ID(x-1, y), ID(x, y), 0);
            add_edge(ID(x+1, y), ID(x, y), 0);
        }
        else if (m[y][x] == '|') {
            add_edge(ID(x, y-1), ID(x, y), 0);
            add_edge(ID(x, y+1), ID(x, y), 0);
        }
        else if ('A' <= m[y][x] && m[y][x] <= 'Z') {
            rep(dx, -1, 2) rep(dy, -1, 2) {
                if (dx != 0 || dy != 0) {
                    add_edge(ID(x+dx, y+dy), ID(x, y), 1);
                }
            }
        }
    }
    ll ans = dijkstra(G, s)[t];
    assert(ans % 2 == 0);
    cout << ans/2 << endl;
}
