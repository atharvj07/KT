#include <bits/stdc++.h>
using namespace std;
void deb(istream_iterator<string> it) { cerr << endl; }
template <typename T, typename... Args>
void deb(istream_iterator<string> it, T a, Args... args) {
  cerr << *it << " = " << a << ",   ";
  deb(++it, args...);
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  int mn = 1000000007;
  for (auto &it : a) {
    cin >> it;
    mn = min(mn, it);
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == mn) cnt++;
  }
  if (cnt > n / 2)
    cout << "Bob" << '\n';
  else
    cout << "Alice\n";
  return 0;
}
