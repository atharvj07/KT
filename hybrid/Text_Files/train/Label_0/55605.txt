import java.util.*;
public class Main {
	static long a[][]=new long [100][100];
	static long b[][]=new long [100][100];
	static void swap(int i,int j)
	{
		long temp=a[i][j];
		a[i][j]=b[i][j];
		b[i][j]=temp;
	}
 public static void main(String args[])
 {
	 //System.out.println(1);
	 Scanner input=new Scanner(System.in);
	int n,m;
	n=input.nextInt();m=input.nextInt();
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=m;j++)
			a[i][j]=input.nextLong();
	}
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=m;j++)
			b[i][j]=input.nextLong();
	}
	boolean f=true;
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=m;j++)
		{
			if(j+1<=m)
			{
				if(a[i][j]>=a[i][j+1])
				{
					if(b[i][j+1]<=a[i][j])f=false;
					else
						swap(i,j+1);
				}
				else if(a[i][j]<a[i][j+1])
				{
					if(b[i][j+1]>a[i][j]&&b[i][j+1]<a[i][j+1])
						swap(i,j+1);
				}
			}
			if(i+1<=n)
			{
				if(a[i][j]>=a[i+1][j])
				{
					if(b[i+1][j]<=a[i][j])f=false;
					else
						swap(i+1,j);
				}
				else if(a[i][j]<a[i+1][j])
				{
					if(b[i+1][j]>a[i][j]&&b[i+1][j]<a[i+1][j])
						swap(i+1,j);
				}
			}
		}
	}
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=m;j++)
		{
			if(j+1<=m)if(b[i][j]>=b[i][j+1]||a[i][j]>=a[i][j+1])f=false;
			if(i+1<=n)if(b[i][j]>=b[i+1][j]||a[i][j]>=a[i+1][j])f=false;
		}
	}
	if(!f)System.out.println("Impossible");
	else
		System.out.println("Possible");
 }
}
