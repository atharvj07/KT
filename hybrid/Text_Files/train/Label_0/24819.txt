#include <bits/stdc++.h>
int n, i, j, t;
int a[10010], ans;
int main() {
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%d", &a[i]);
  }
  for (i = 0; i < n; i++)
    for (j = i + 1; j < n; j++)
      if (a[i] > a[j]) {
        t = a[i];
        a[i] = a[j];
        a[j] = t;
      }
  ans = 0;
  for (i = 0; i < n / 2; i++) ans += a[2 * i + 1] - a[2 * i];
  printf("%d", ans);
}
