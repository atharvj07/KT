(n, m) = map(int, input().split())
flag = False
daf = list(map(int, input().split()))
if sum(daf) > 0:
	flag = True
if n >= 2:
	for i in range(n - 2):
		daf = input()
		if daf[0] == '1' or daf[-1] == '1':
			flag = True
	daf = list(map(int, input().split()))
	if sum(daf) > 0:
		flag = True
if flag:
	print(2)
else:
	print(4)
