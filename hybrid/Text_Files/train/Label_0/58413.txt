import java.util.*;
public class Jeff {
	public static void main (String [] arg) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		double [] s = new double [(int)2*n];
		boolean [] isInt = new boolean[(int)2*n];
		int n0 = 0;
		long sumMin = 0;
		long sumMax = 0;
		double sumT = 0;
		for (int i = 0; i<2*n; ++i) {
			String ss = sc.next();
			String [] b = ss.split("\\.");
			s[i] = Double.parseDouble(ss);
			isInt[i] = b[1].equals("000");
			if (isInt[i]) n0++;
			sumT += s[i];
			sumMin += (isInt[i]) ? Integer.parseInt(b[0]) : Math.floor(s[i]+1e-9);
			sumMax += (isInt[i]) ? Integer.parseInt(b[0]) : Math.ceil(s[i]-1e-9);
		}
		double ans = 0.0;
		if (n0 >= n) {
			ans = Math.abs(Math.round(sumT) - sumT);
			if (sumT <= sumMin) ans = Math.abs(sumT - sumMin);
			if (sumT >= sumMax) ans = Math.abs(sumT - sumMax);
		} else {
			int floors = n-n0;
			int ceils = n-n0;
			sumMin = 0;
			sumMax = 0;
			for (int i = 0; i<2*n; ++i) {
				if (isInt[i]) { sumMin += s[i]; sumMax += s[i]; continue;}
				if (floors > 0) {
					sumMin += Math.floor(s[i]+1e-9);
					floors--;
				} else {
					sumMin += Math.ceil(s[i]-1e-9);
				}
				if (ceils > 0) {
					sumMax += Math.ceil(s[i] - 1e-9);
					ceils--;
				} else {
					sumMax += Math.floor(s[i] + 1e-9);
				}
			}
			long tmp1 = sumMin;
			long tmp2 = sumMax;
			sumMin = Math.min(tmp1,tmp2);
			sumMax = Math.max(tmp1,tmp2);
			
			
			ans = Math.abs(Math.round(sumT) - sumT);
			if (sumT <= sumMin) ans = Math.abs(sumT - sumMin);
			if (sumT >= sumMax) ans = Math.abs(sumT - sumMax);
			
		}
		
		System.out.printf("%.3f\n", ans);
		
	}
}
