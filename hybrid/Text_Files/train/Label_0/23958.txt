#include<bits/stdc++.h>
using namespace std;

using ll = long long;
using ld = long double;
map<pair<int,int>,bool> used;
vector<int> ask(ll r,int n)
{
    cout<<"? "<<r<<endl;
    vector<int> vec(n+1);
    for(int i=1;i<=n;++i) 
    {
        cin>>vec[i];
    }
    return vec;
}


void solve()
{
    int n;
    cin>>n;
    vector<int> vec=ask(1,n);
    int c=0;
    vector<pair<int,int>> ans;
    for(int i=1;i<=n;i++)
    {
        if(vec[i]%2==0) c+=1;
        if(vec[i]==1) {
        ans.push_back({1,i});
        used[{1,i}]=used[{i,1}]=1;
    }
    }
    if(c<=(n+1)/2) c=0;
    else c=1;
    for(int i=2;i<=n;++i)
    {
        if((vec[i]%2)==c)
        {
            vector<int> rp=ask(i,n);
            for(int j=1;j<=n;++j)
            {
                if((rp[j]==1)&&!used[{i,j}])
                {
                    used[{i,j}]=used[{j,i}]=1;
                    ans.push_back({i,j});
                }
            }
        }
    }
    cout<<"!"<<endl;
    for(int i=0;i<ans.size();i++) 
    {
        cout<<ans[i].first<<" "<<ans[i].second<<endl;
    }
}
 
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);cout.tie(NULL);
    ll t=1;
    //cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;
}
