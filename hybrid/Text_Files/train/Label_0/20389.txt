#include <bits/stdc++.h>
char str[10000005];
int main() {
  int len;
  long long int b = 0, sum = 0;
  scanf("%s", str);
  len = strlen(str);
  for (int i = len - 1; i >= 0; i--) {
    if (str[i] == 'b')
      b++;
    else {
      sum = (sum + b) % (1000000000 + 7);
      b = (2 * b) % (1000000000 + 7);
    }
  }
  printf("%d", sum);
}
