import java.io.*; 
import java.util.*;
import java.math.*;
import java.awt.geom.*;

import static java.lang.Math.*;

public class Solution implements Runnable {
	
	class SegmentTree {
		
		static final int INF = 4000000;
		
		int n;
		int t [];
		int add [];
		
		public SegmentTree(int n) {
			this.n = n;
			t = new int [4 * n];
			add = new int [4 * n];
			build (1, 0, n - 1);
		}
		
		private void build (int v, int tl, int tr) {
			if (tl == tr) {
				t [v] = INF;
			} else {
				int tm = (tl + tr) >> 1;
				build (v * 2, tl, tm);
				build (v * 2 + 1, tm + 1, tr);
				t [v] = min (t[v * 2], t[v * 2 + 1]);
			}
		}
		
		private void push (int v) {
			t [v] += add [v];
			add [v * 2] += add [v];
			add [v * 2 + 1] += add [v];
			add [v] = 0;
		}
		
		private void pop (int v) {
			t [v] = min (t[v * 2] + add [v * 2], t[v * 2 + 1] + add[v * 2 + 1]);
		}
		
		private void update (int v, int tl, int tr, int pos, int value) {
			if (tl == tr) {
				t [v] = value;
				add [v] = 0;
			} else {
				push (v);
				int tm = (tl + tr) >> 1;
				if (pos <= tm) {
					update (v * 2, tl, tm, pos, value);
				} else {
					update (v * 2 + 1, tm + 1, tr, pos, value);
				}
				pop(v);
			}
		}
		
		public void update (int pos, int value) {
			update (1, 0, n - 1, pos, value);
		}
		
		public void increase () {
			add [1]++;
		}
		
		public void decrease () {
			add [1]--;
		}
		
		public int getAnswer () {
			return t[1] + add[1];
		}
		
	}
	
	public void solve () throws Exception {
		int n = nextInt();
		
		if (n == 1) {
			out.println(0);
			return;
		}

		int a [] = new int [n];
		int b [] = new int [2 * n];
		int apos [] = new int [n];
		
		for (int i = 0; i < n; i++) {
			a [i] = nextInt() - 1;
			apos[a[i]] = i;
		}
		
		int pos [][] = new int [n][2];
		
		for (int i = 0; i < n; i++) {
			b [i] = b[i + n] = nextInt() - 1;
			pos[b[i]][0] = i;
			pos[b[i]][1] = i + n;
		}
		
		SegmentTree treeMinus = new SegmentTree(n);
		SegmentTree treePlus = new SegmentTree(n);
		
		ArrayList<Integer> toPlus [] = new ArrayList [n];
		for (int i = 0; i < n; i++) {
			toPlus [i] = new ArrayList<Integer>();
		}
		
		for (int i = 0; i < n; i++) {
			if (i < pos[a[i]][0]) {
				treeMinus.update(a[i], pos[a[i]][0] - i);
			} else {
				treePlus.update(a[i], i - pos[a[i]][0]);
			}
			if (pos[a[i]][1] - i < n) {
				toPlus[pos[a[i]][1] - i].add(a[i]);
			}
			if (pos[a[i]][0] - i < n && pos[a[i]][0] - i >= 0) {
				toPlus[pos[a[i]][0] - i].add(a[i]);
			}
		}
		
		for (int i = 0; i < n; i++) {
			out.println(min (treeMinus.getAnswer(), treePlus.getAnswer()));
			
			for (int index : toPlus[i]) {
				treeMinus.update(index, SegmentTree.INF);
				treePlus.update(index, 0);
			}
			
			treePlus.update(b[i], SegmentTree.INF);
			treeMinus.update(b[i], pos[b[i]][1] - apos[b[i]] - i);

			treePlus.increase();
			treeMinus.decrease();
		}
		
	}
	
	private static final String fname = "";
	private static long stime = 0;
	
	private BufferedReader in;
	private PrintWriter out;
	private StringTokenizer st;
	
	private String nextToken () throws Exception {
		while (st == null || !st.hasMoreTokens()) {
			st = new StringTokenizer(in.readLine());
		}
		return st.nextToken();
	}
	
	private int nextInt () throws Exception {
		return Integer.parseInt(nextToken());
	}
	
	private long nextLong () throws Exception {
		return Long.parseLong(nextToken());
	}
	
	private double nextDouble () throws Exception {
		return Double.parseDouble(nextToken());
	}
	
	@Override
	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
			solve ();
		} catch (Exception e) {
			throw new RuntimeException(e);
		} finally {
			out.close();
		}
	}
	
	public static void main(String[] args) {
		new Thread(null, new Solution(), "", 1<<26).start();
	}

}
