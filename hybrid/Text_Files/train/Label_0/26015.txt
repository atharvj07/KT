const double PI = acos(-1.0);
const int INF = 1000 * 1000 * 1000 + 7;

struct rectangle {
  int x1, y1, x2, y2;
};
vector<rectangle> aa;
vector<rectangle> A;
int cnt;
int get(int x1, int y1, int x2, int y2) {
  cnt++;
  if (x1 > x2) swap(x1, x2);
  if (y1 > y2) swap(y1, y2);
  cout << "? " << x1 << ' ' << y1 << ' ' << x2 << ' ' << y2 << endl;
  int n = 0;
  cin >> n;
  for (int i = (0); i < ((int)A.size()); ++i) {
    if (x1 <= A[i].x1 && A[i].x2 <= x2 && y1 <= A[i].y1 && A[i].y2 <= y2) n--;
  }
  return n;
}
rectangle get(int n) {
  int L = 1, R = n + 1;
  while (R - L > 1) {
    int m = (R + L) / 2;
    int v = get(m, 1, n, n);
    if (v == 0)
      R = m;
    else
      L = m;
  }
  int x1 = L;
  L = x1 - 1, R = n;
  while (R - L > 1) {
    int m = (R + L) / 2;
    int v = get(x1, 1, m, n);
    if (v == 0)
      L = m;
    else
      R = m;
  }
  int x2 = R;
  L = 1, R = n + 1;
  while (R - L > 1) {
    int m = (R + L) / 2;
    int v = get(x1, m, x2, n);
    if (v == 0)
      R = m;
    else
      L = m;
  }
  int y1 = L;
  L = y1 - 1, R = n;
  while (R - L > 1) {
    int m = (R + L) / 2;
    int v = get(x1, y1, x2, m);
    if (v == 0)
      L = m;
    else
      R = m;
  }
  int y2 = R;
  return rectangle{x1, y1, x2, y2};
}
