#include <bits/stdc++.h>
using namespace std;
long long n, f[3001][3001];
struct M {
  long long x, c;
} a[3001];
bool cmp(M x, M y) { return x.x < y.x; }
signed main() {
  scanf("%lld", &n);
  for (long long i = 1; i <= n; i++) scanf("%lld%lld", &a[i].x, &a[i].c);
  sort(a + 1, a + 1 + n, cmp);
  long long minn;
  f[1][1] = a[1].c;
  for (long long i = 2; i <= n; i++) {
    minn = LONG_LONG_MAX;
    for (long long j = 1; j < i; j++) {
      f[i][j] = f[i - 1][j] + a[i].x - a[j].x;
      minn = min(minn, f[i - 1][j]);
    }
    f[i][i] = minn + a[i].c;
  }
  long long ans = LONG_LONG_MAX;
  for (long long i = 1; i <= n; i++) ans = min(ans, f[n][i]);
  printf("%lld", ans);
}
