import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Locale;

public class Main {
	static int n;
	static long[] x, velocity;
	
	public static void main(String[] args) throws IOException{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		n = Integer.parseInt(br.readLine());
		x=new long[n];
		velocity=new long[n];
		for(int i = 0; i < n; i++){
			String[] line = br.readLine().split(" ");
			x[i] = Long.parseLong(line[0]);
			velocity[i] = Long.parseLong(line[1]);
		}
		double t = bs(0, 2L*1000000000L, 100);
		if (check(t)) System.out.printf(Locale.US,"%.20f\n",t);
		else System.out.println(-1);
	}

	private static double bs(double l, double r,int iterate) {
		double middle = (l+r)/2;
		if (iterate == 0) return r;
		boolean colapsed = check(middle);
		if(colapsed) return bs(l,middle,iterate-1);
		else return bs(middle,r,iterate-1);
	}

	private static boolean check(double time) {
		double vision=0;
		boolean first=true;
		for(int i = 0; i < n; i++){
			if(velocity[i] > 0){
				double pos =  x[i] + velocity[i]*time;
				if (first) {vision=pos; first=false;}
				else vision = Math.max(vision, pos);
			}
			else {
				if (first) continue;
				double pos =  x[i] + velocity[i]*time;
				if(pos <= vision) return true;
			}
		}
		return false;
	}
}
