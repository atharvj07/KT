n = int(input())
a = list(map(int, input().split()))
c = 0
s = d = 0
for i in range(1, n):
	if i % 2 == 1:
		s += a[i]
	else:
		d += a[i]
if s == d:
	c += 1
for i in range(1, n):
	if i % 2 == 1:
		s = s + a[i - 1] - a[i]
		if s == d:
			c += 1
	else:
		d = d + a[i - 1] - a[i]
		if s == d:
			c += 1
print(c)
