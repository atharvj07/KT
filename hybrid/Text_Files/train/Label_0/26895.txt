#include <algorithm>
#include <cstdio>
#include <cstring>
const int N = 8050;
int xx[N], yy[N], nx, ny;
int f[N];
bool able(int *aa, int a, int n) {
  a = std::abs(a);
  int sum = 0;
  for (int i = 0; i < n; ++i)
    sum += aa[i];
  if (sum < a) return false;
  if ((sum ^ a) & 1) return false;
  a = (sum + a) >> 1;
  for (int j = 0; j <= a; ++j)
    f[j] = 0;
  f[0] = true;
  for (int i = 0; i < n; ++i)
    for (int j = a; j >= aa[i]; --j)
      f[j] |= f[j - aa[i]];
  return f[a];
}
char s[N];
int main() {
  int x, y;
  scanf("%s%d%d", s, &x, &y);
  int n = (int)strlen(s);
  int p = 1;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 'F') {
      if (p) ++xx[nx];
      else ++yy[ny];
    } else {
      if (p) yy[++ny] = 0;
      else xx[++nx] = 0;
      p ^= 1;
    }
  }
  puts(able(xx + 1, x - xx[0], nx) && able(yy + 1, y, ny) ?  "Yes" : "No");
  return 0;
}
