def competition_winner(points, times, c):
	limak_points = 0
	rage_points = 0
	for i in range(len(points)):
		limak_points += max(0, points[i] - c * sum(times[0:i + 1]))
		rage_points += max(0, points[len(points) - i - 1] - c * sum(times[len(points) - i - 1:len(points)]))
	if limak_points > rage_points:
		print('Limak')
	elif rage_points > limak_points:
		print('Radewoosh')
	else:
		print('Tie')
(n, c) = map(int, input().split())
points = list(map(int, input().split()))
times = list(map(int, input().split()))
competition_winner(points, times, c)
