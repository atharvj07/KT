n = int(input())
t = list(map(int, input().split()))
flag = True
while len(t) > 1 and flag:
	nn = max(t)
	for i in range(0, len(t)):
		if t[i] == nn:
			if i + 1 < len(t) and t[i] - t[i + 1] >= 2 or (i - 1 >= 0 and t[i] - t[i - 1] >= 2):
				flag = False
	t.remove(nn)
if flag:
	print('yes')
else:
	print('no')
