import io
import os
from collections import Counter, defaultdict, deque

def solve(N, M, A, P):
	P = sorted(set(P))
	runs = [[P[0], P[0] + 1]]
	for p in P[1:]:
		if runs[-1][1] == p:
			runs[-1][1] = p + 1
		else:
			runs.append([p, p + 1])
	Asorted = list(A)
	for (start, end) in runs:
		Asorted[start:end + 1] = sorted(Asorted[start:end + 1])
	if Asorted == sorted(A):
		return 'YES'
	return 'NO'
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
T = int(input())
for t in range(T):
	(N, M) = [int(x) for x in input().split()]
	A = [int(x) for x in input().split()]
	P = [int(x) - 1 for x in input().split()]
	ans = solve(N, M, A, P)
	print(ans)
