t = int(input())
for i in range(0, t):
	l = list(map(int, input().split()))
	a = l[0]
	s = l[1]
	s1 = []
	a1 = []
	while s > 0:
		s1.append(s % 10)
		s = s // 10
	while a > 0:
		a1.append(a % 10)
		a = a // 10
	n = 0
	i = 0
	j = 0
	k = 0
	p = 0
	c1 = len(s1)
	c2 = len(a1)
	while j < len(s1):
		if i == len(a1):
			d = s1[j]
			n += d * 10 ** p
			j += 1
			p += 1
		elif i <= len(a1) - 1:
			d = s1[j] - a1[i]
			if d >= 0:
				n += d * 10 ** i
				i += 1
				j += 1
				p = i
				c1 -= 1
				c2 -= 1
			elif d < 0:
				if j < len(s1) - 1 and s1[j + 1] == 1:
					d = 10 + s1[j] - a1[i]
					n += d * 10 ** i
					i += 1
					j += 2
					p = i
					c1 -= 2
					c2 -= 1
				else:
					k = 1
					i += 1
					p = i
					j += 1
	if c2 > c1:
		k = 1
	if k == 1:
		print(-1)
	else:
		print(n)
