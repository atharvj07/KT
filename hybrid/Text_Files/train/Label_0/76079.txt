def getdict(n):
	d = {}
	if type(n) is list:
		for i in n:
			if i in d:
				d[i] += 1
			else:
				d[i] = 1
	else:
		for i in range(n):
			t = ii()
			if t in d:
				d[t] += 1
			else:
				d[t] = 1
	return d

def cdiv(n, k):
	return n // k + (n % k != 0)

def ii():
	return int(input())

def mi():
	return map(int, input().split())

def li():

def geto(s, i, j):
	res = 0
	for k in range(i + 1, j):
		if s[k] == 'o':
			res += 1
	return res

def count(a, b):
	m = len(a)
	n = len(b)
	lookup = [[0] * (n + 1) for i in range(m + 1)]
	for i in range(n + 1):
		lookup[0][i] = 0
	for i in range(m + 1):
		lookup[i][0] = 1
	for i in range(1, m + 1):
		for j in range(1, n + 1):
			if a[i - 1] == b[j - 1]:
				lookup[i][j] = lookup[i - 1][j - 1] + lookup[i - 1][j]
			else:
				lookup[i][j] = lookup[i - 1][j]
	return lookup[m][n]
s = input()

n = len(s)
vpos = []
d = []
i = 0
mode = 'v'
while i < n - 1:
	if s[i] == s[i + 1] == 'v':
		vpos.append(i)
		t = 0
		while i < n and s[i] == mode:
			t += 1
			i += 1
		d.append(t - 1)
	i += 1
r = []
for i in range(len(d) - 1):
	r.append(d[i])
	r.append(geto(s, vpos[i], vpos[i + 1]))
r.append(d[-1])
news = []
mode = 'w'
for i in r:
	news.append(mode * i)
	mode = 'w' if mode == 'o' else 'o'
