import java.io.*;
import java.util.*;
import java.math.*;

public class p458d {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double[] f = new double[100010];
        double[][] arr = new double[310][310];
        f[0] = 1.0;
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        double max = 1e99;
        double ans = 0.0;
        for (int i = 0; i < k; i++) {
            f[i + 1] = f[i] * (k - i) / (m - i);
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                if (j == 0 || j == i) {
                    arr[i][j] = 1;
                } else {
                    arr[i][j] = arr[i - 1][j - 1] + arr[i - 1][j];
                }
            }
        }
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                ans = Math.min(max, ans + f[i * n + j * n - i * j] * arr[n][i] * arr[n][j]);
            }
        }
        System.out.println(ans);
    }
}
