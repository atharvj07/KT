from itertools import groupby

def solve(ar, n):
	pos = [ar[i] == i + 1 for i in range(n)]
	cnt = [key for (key, groups) in groupby(pos)]
	cnt = cnt.count(False)
	if cnt == 0:
		print(0)
	elif cnt == 1:
		print(1)
	else:
		print(2)

def read():
	t = int(input())
	for i in range(t):
		n = int(input())
		ar = list(map(int, input().strip().split()))
		solve(ar, n)
read()
