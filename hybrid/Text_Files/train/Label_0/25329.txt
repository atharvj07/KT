#include <bits/stdc++.h>
using namespace std;
struct SegTree{
	vector<int> val;
	void build(){
		val.resize((1<<18)*2);
		for(int i=0;i<(1<<18)*2;++i){
			val[i]=INT_MIN;
		}
	}
	void modify(int x,int v){
		x+=(1<<18)-1;
		val[x]=v;
		while(x>0){
			x=(x-1)/2;
			val[x]=max(val[x*2+1],val[x*2+2]);
		}
	}
	int query(int lc,int rc,int x=0,int l=0,int r=(1<<18)){
		if(rc<=l||r<=lc){
			return INT_MIN;
		}
		if(lc<=l&&r<=rc){
			return val[x];
		}
		int mid=(l+r)/2;
		return max(query(lc,rc,x*2+1,l,mid),query(lc,rc,x*2+2,mid,r));
	}
}seg,seg2;
int n;
int b[210000];
int sum[210000];
int ans;
int main(){
	seg.build(),seg2.build();
	cin>>n;
	for(int i=0;i<n;++i){
		cin>>b[i];
		sum[i+1]=sum[i]+!b[i];
	}
	int q;
	cin>>q;
	vector<pair<int,int> > qs(q);
	for(int i=0;i<q;++i){
		cin>>qs[i].first>>qs[i].second;
	}
	sort(qs.begin(),qs.end());
	for(auto op:qs){
		int l=op.first,r=op.second;
		int v=r-l+1-2*(sum[r]-sum[l-1]);
		int res=max(0,seg.query(0,l));
		res+=v;
		int res2=seg2.query(l,r+1);
		if(res2!=INT_MIN){
			res2+=r-2*sum[r];
		}
		int ret=max(res,res2);
		ans=max(ret,ans);
		seg.modify(r,ret);
		seg2.modify(r,ret-r+2*sum[r]);
	}
	cout<<n-sum[n]-ans<<'\n';
	return 0;
}
