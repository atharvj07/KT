#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int main() {
    ll X, Y, Z, A, B, C, N;
    cin >> X >> Y >> Z >> A >> B >> C >> N;
    vector<ll> res(N);
    const int cnt = (Z-C) / N + (C+1) / N;
    for(int i=0; i<N; ++i) {
        res[i] = X*Y*cnt;
    }
    vector<ll> rest(N);
    {
        const int cnt2 = (Y-B) / N + (B+1) / N;
        for(int i=0; i<N; ++i) {
            rest[i] = X*cnt2;
        }
        vector<ll> rest2(N);
        for(int i=0; i<X; ++i) {
            rest2[abs(i-A) % N]++;
        }
        for(int i=0; i<N; ++i) {
            rest[i] -= rest2[i];
        }
        for(int i=0; i<max((Y-B)%N, (B+1)%N); ++i) {
            for(int j=0; j<N; ++j) {
                if(i < (Y-B) % N) {
                    rest[j] += rest2[j];
                }
                if(i < (B+1) % N) {
                    rest[j] += rest2[j];
                }
            }
            auto tmp = rest2;
            for(int j=0; j<N; ++j) {
                tmp[(j+1)%N] = rest2[j];
            }
            rest2.swap(tmp);
        }
    }
    for(int i=0; i<N; ++i) {
        res[i] -= rest[i];
    }
    for(int i=0; i<max((Z-C)%N, (C+1)%N); ++i) {
        for(int j=0; j<N; ++j) {
            if(i < (C+1)%N) {
                res[j] += rest[j];
            }
            if(i < (Z-C)%N) {
                res[j] += rest[j];
            }
        }
        auto tmp = rest;
        for(int j=0; j<N; ++j) {
            tmp[(j+1)%N] = rest[j];
        }
        rest.swap(tmp);
    }
    for(int i=0; i<N; ++i) {
        cout << res[i] << " \n"[i == N-1] << flush;
    }
}
