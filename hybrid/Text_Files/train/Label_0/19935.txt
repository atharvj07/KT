#include <bits/stdc++.h>
using namespace std;
const long long INF = 1e18 + 1;
inline long long input(void) {
  char t;
  long long x = 0;
  int neg = 0;
  t = getchar();
  while ((t < 48 || t > 57) && t != '-') t = getchar();
  if (t == '-') {
    neg = 1;
    t = getchar();
  }
  while (t >= 48 && t <= 57) {
    x = (x << 3) + (x << 1) + t - 48;
    t = getchar();
  }
  if (neg) x = -x;
  return x;
}
inline void output(long long x) {
  char a[20];
  int i = 0, j;
  a[0] = '0';
  if (x < 0) {
    putchar('-');
    x = -x;
  }
  if (x == 0) putchar('0');
  while (x) {
    a[i++] = x % 10 + 48;
    x /= 10;
  }
  for (j = i - 1; j >= 0; j--) {
    putchar(a[j]);
  }
  putchar(' ');
}
int main() {
  long long n, b, p, x = 0, y = 0, z;
  n = input();
  b = input();
  p = input();
  y = n * p;
  while (n > 1) {
    z = 1;
    while (2 * z <= n) z *= 2;
    x += (z / 2) * (2 * b + 1);
    n -= (z / 2);
  }
  output(x);
  output(y);
  putchar('\n');
  return 0;
}
