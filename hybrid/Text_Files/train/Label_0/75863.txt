import sys
global par, sz, mx

def find(u):
	global par, sz, mx
	if par[u] == u:
		return u
	return find(par[u])

def unite(u, v):
	global par, sz, mx
	p1 = find(u)
	p2 = find(v)
	if p1 == p2:
		return
	if sz[p1] < sz[p2]:
		(p1, p2) = (p2, p1)
	sz[p1] += sz[p2]
	mx[p1] = max(mx[p1], mx[p2])
	par[p2] = p1
(n, m) = map(int, sys.stdin.readline().split())
par = [i for i in range(n)]
mx = [i for i in range(n)]
sz = [1] * n
for _ in range(m):
	(u, v) = map(int, sys.stdin.readline().split())
	u -= 1
	v -= 1
	unite(u, v)
comp = 0
mxb = 0
for i in range(n):
	if find(i) == i:
		comp += 1
for i in range(n):
	mxb = max(mxb, mx[find(i)])
	if mxb == i:
		comp -= 1
print(comp)
