from collections import Counter
(n, k) = map(int, input().split())
S = list(map(int, input().split()))
c = Counter(S)
l = [(i, c[i]) for i in list(c)]
l_ = []
for (l, m) in l:
	j = 1
	while m // j != 0:
		l_.append((l, m // j))
		j += 1
l_.sort(key=lambda x: x[1], reverse=True)
for i in range(k):
	print(l_[i][0], end=' ')
