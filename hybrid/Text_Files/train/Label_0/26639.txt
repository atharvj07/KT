#include <bits/stdc++.h>
using namespace std;
const int M = 1e9 + 7;
int n, k;
vector<int> s;
vector<vector<int>> dp;

int solve(int x, int rem) {
  if (x == n - 1) return rem % s[x];
  int &ret = dp[x][rem];

  if (ret != -1)  return ret;
  ret = 0;

  ret = solve(x + 1, rem % s[x]);
  ret = (ret + 1ll * (n-1-x) * solve(x + 1, rem) % M) % M;
  return ret;
}

int main() {

  ios::sync_with_stdio(false);
  cout.tie(nullptr);
  cin.tie(nullptr);

  cin >> n >> k;
  s.resize(n);
  dp.assign(n, vector<int> (k + 1, -1));
  for (int i = 0; i < n; ++i) {
    cin >> s[i];
  }

  sort(s.begin(), s.end(), greater<int>());
  cout << solve(0, k) << '\n';
}
