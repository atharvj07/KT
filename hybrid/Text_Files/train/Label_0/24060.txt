#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  int q;
  cin >> q;
  map<long long, long long> m;
  while (q--) {
    int t;
    long long u, v;
    cin >> t >> u >> v;
    if (t == 1) {
      long long w;
      cin >> w;
      while (u != v) {
        if (u < v) swap(u, v);
        m[u] += w;
        u /= 2;
      }
    } else {
      long long ans = 0;
      while (u != v) {
        if (u < v) swap(u, v);
        ans += m[u];
        u /= 2;
      }
      cout << ans << "\n";
    }
  }
  return 0;
}
