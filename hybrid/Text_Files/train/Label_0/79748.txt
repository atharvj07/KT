from bisect import bisect

def solve():
	MOD = 998244353
	n = int(input())
	x = [0] * n
	y = [0] * n
	s = [0] * n
	z = [0] * n
	p = [0] * (n + 1)
	r = 0
	for i in range(n):
		(x[i], y[i], s[i]) = map(int, input().split())
	for i in range(n):
		nn = bisect(x, y[i])
		z[i] = (p[i] - p[nn] + x[i] - y[i]) % MOD
		p[i + 1] = (p[i] + z[i]) % MOD
		if s[i] == 1:
			r = (r + z[i]) % MOD
	print((x[n - 1] + 1 + r) % MOD)
solve()
