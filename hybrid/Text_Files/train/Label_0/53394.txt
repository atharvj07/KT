import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(new BufferedInputStream(System.in));
        int n = scan.nextInt();
        int q = scan.nextInt();
        int root = 0;
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            int s = i + 2;
            int f = scan.nextInt();
            if (!map.containsKey(f)) {
                map.put(f, new ArrayList<Integer>());
            }
            map.get(f).add(s);
        }

        int[] array = new int[n];
        int[] limit = new int[n];
        dfs(1, map, array, limit);
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            pos[array[i] - 1] = i;
        }
        StringBuffer sb = new StringBuffer();
        for (int i = 0; i < q; i++) {
            int e = scan.nextInt() - 1, len = scan.nextInt();
            int realPos = pos[e] + len - 1;
            if (realPos >= limit[e]) {
                sb.append("-1\n");
            } else {
                sb.append(array[realPos]);
                sb.append("\n");
            }
        }
        System.out.println(sb.toString());
    }

    static int pos = 0;
    static final List<Integer> EMPTY = new LinkedList<>();

    public static void dfs(int f, Map<Integer, List<Integer>> map, int[] array, int[] limit) {
        int curr = pos;
        array[pos++] = f;
        for (Integer i : map.getOrDefault(f, EMPTY)) {
            dfs(i, map, array, limit);
        }
        limit[f - 1] = pos;
    }
}
