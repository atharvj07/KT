import java.io.*;
import java.util.*;

public class CF1478D extends PrintWriter {
	CF1478D() { super(System.out); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1478D o = new CF1478D(); o.main(); o.flush();
	}

	static final long INF = 0x3fffffffffffffffL;
	long gcd(long a, long b) {
		return b == 0 ? a : gcd(b, a % b);
	}
	void main() {
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			long k = sc.nextLong();
			long[] aa = new long[n]; long amin = INF, amax = -INF;
			for (int i = 0; i < n; i++) {
				long a = sc.nextLong();
				aa[i] = a;
				amin = Math.min(amin, a);
				amax = Math.max(amax, a);
			}
			boolean yes;
			if (k == amin || k == amax) {
				println("YES");
				continue;
			}
			if (amin == amax) {
				println("NO");
				continue;
			}
			if (k > amin) {
				k -= amin;
				for (int i = 0; i < n; i++)
					aa[i] -= amin;
			} else {
				k = amax - k;
				for (int i = 0; i < n; i++)
					aa[i] = amax - aa[i];
			}
			long d = 0;
			for (int i = 0; i < n; i++)
				d = gcd(d, aa[i]);
			println(k % d == 0 ? "YES" : "NO");
		}
	}
}
