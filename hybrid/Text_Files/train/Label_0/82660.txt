from collections import Counter
n = int(input())
m = [int(j) for j in input().split()]
m = m[:4000]
new = {}
res = []
for i in range(len(m)):
	for j in range(i + 1, len(m)):
		if m[i] + m[j] in new:
			if i not in new[m[i] + m[j]] and j not in new[m[i] + m[j]]:
				z = new[m[i] + m[j]][0]
				x = new[m[i] + m[j]][1]
				res.extend([i, j, z, x])
				break
			else:
				continue
		else:
			new[m[i] + m[j]] = [i, j]
	else:
		continue
	break
if res:
	print('YES')
	print(' '.join([str(e + 1) for e in res]))
else:
	print('NO')
