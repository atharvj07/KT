#include <iostream>
#include <iomanip>
#include <complex>
#include <vector>
#include <algorithm>
#include <cmath>
#include <set>
using namespace std;
const double EPS = 1e-8;
const double INF = 1e12;
const double PI = acos(-1);
#define EQ(n,m) (abs((n)-(m)) < EPS)
#define X real()
#define Y imag()

typedef complex<double> P;
typedef vector<P> VP;
namespace std{
    bool operator < (const P& a, const P& b){
        return !EQ(a.X,b.X) ? a.X<b.X : a.Y+EPS<b.Y;
    }
    bool operator == (const P& a, const P& b){
        return abs(a-b) < EPS;
    }
}

double dot(P a, P b){
    return (conj(a)*b).X;
}
double cross(P a, P b){
    return (conj(a)*b).Y;
}
int ccw(P a, P b, P c){
    b -= a;
    c -= a;
    if(cross(b,c) > EPS) return +1; //ccw
    if(cross(b,c) < -EPS) return -1; //cw
    if(dot(b,c) < -EPS) return +2; //c-a-b
    if(abs(c)-abs(b) > EPS) return -2; //a-b-c
    return 0; //a-c-b
}
P rotate(const P &p, double rad){
    return p *P(cos(rad), sin(rad));
}

int main(){
    while(1){
        int n;
        cin >> n;
        if(n==0) break;

        if(n==1){
            double d;
            cin >> d >> d >> d >> d;
            cout << 0 << endl;
            continue;
        }

        //重心を原点に平行移動
        vector<VP> p(2, VP(n));
        for(int d=0; d<2; d++){
            P centroid(0, 0);
            for(int i=0; i<n; i++){
                double x,y;
                cin >> x >> y;
                p[d][i] = P(x, y);
                centroid += p[d][i];
            }
            centroid /= n;
            //p0が重心に重なると角度が取れなくなるので例外処理
            if(p[d][0] == centroid){
                swap(p[d][0], p[d][1]);
            }
            for(int i=0; i<n; i++){
                p[d][i] -= centroid;
            }
        }

        //同じ点が2点以上ないという仮定で、全部このsetに入って入ればokとする
        set<P> pset;
        for(int i=0; i<n; i++){
            pset.insert(p[0][i]);
        }
        double ans = INF;
        for(int i=0; i<n; i++){
            //p[1][i]を回転させてp[0][0]に合わせる
            if(!EQ(abs(p[0][0]), abs(p[1][i]))) continue;
            double angle = arg(p[1][i]/p[0][0]);
            bool ok = true;
            for(int j=0; j<n; j++){
                P rot = rotate(p[1][j], -angle);
                if(pset.find(rot) == pset.end()){
                    ok = false;
                    break;
                }
            }
            if(ok){
                ans = min(ans, abs(angle));
            }
        }
        cout << fixed << setprecision(10);
        cout << ans << endl;
    }
    return 0;
}
