#include <bits/stdc++.h>
using namespace std;
const int Z = (int)3e3 + 111;
const int N = (int)1e7 + 111;
const int INF = (int)1e9 + 111;
const int MOD = (int)1e9 + 7;
int used[1005][1005], n, m, k(1);
char a[1005][1005];
bool f(1), f1(1), ff(0);
vector<pair<pair<int, int>, int> > ans;
int proh(int i, int j) {
  k = 1;
  ff = 0;
  while (i - k >= 0 && j - k >= 0 && i + k < n && j + k < m &&
         a[i - k][j] == '*' && a[i + k][j] == '*' && a[i][j - k] == '*' &&
         a[i][j + k] == '*') {
    used[i][j] = 1;
    if (!used[i - k][j] || !used[i + k][j] || !used[i][j + k] ||
        !used[i][j - k])
      ff = 1;
    used[i - k][j] = 1;
    used[i + k][j] = 1;
    used[i][j - k] = 1;
    used[i][j + k] = 1;
    k++;
  }
  if (ff == 0) return 1;
  return k;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) cin >> a[i][j];
  while (f1 && f) {
    f = 0;
    f1 = 0;
    ff = 0;
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        if (a[i][j] == '*') {
          int pr = proh(i, j);
          if (pr != 1)
            ans.push_back(make_pair(make_pair(i + 1, j + 1), pr - 1));
          if (k > 1) f1 = 1;
        }
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        if (a[i][j] == '*' && !(used[i][j])) f = 1;
    if (ff == 0) break;
  }
  if (f == 1) {
    cout << -1;
    return 0;
  }
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); i++)
    cout << ans[i].first.first << ' ' << ans[i].first.second << ' '
         << ans[i].second << endl;
  return 0;
}
