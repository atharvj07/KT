	public static class LongSet {

		int size = 0;

		public long[][] table;

		public LongSet(int size) {
			table = new long[1 << (32 - Integer.numberOfLeadingZeros(size - 1))][1];
			for (int i = 0; i < table.length; i++) {
				table[i][0] = 1;
			}
		}

		final void add(long key) {
			int idx = (int) (key & (table.length - 1));
			long[] vec = table[idx];
			/*
			 * resize the bucket if needed. vec[0] stores the number of keys in
			 * the bucket
			 */
			if (vec[0] >= vec.length) {
				vec = Arrays.copyOf(vec, 2 * vec.length);
				table[idx] = vec;
			}
			/*
			 * if key is already in bucket don't do anything
			 */
			for (int i = 1; i < vec[0]; i++) {
				if (vec[i] == key) {
					return;
				}
			}
			/*
			 * add the key snd increment bucket and table size
			 */
			vec[(int) vec[0]] = key;
			vec[0]++;
			size++;
		}

		void remove(long key) {
			int idx = (int) (key & (table.length - 1));
			long[] vec = table[idx];

			for (int i = 1; i < vec[0]; i++) {
				if (vec[i] == key) {
					vec[i] = vec[(int) (vec[0] - 1)];
					vec[0]--;
					size--;
				}
			}
			if (vec[0] >= 1 && vec[0] <= vec.length / 4) {
				vec = Arrays.copyOf(vec, vec.length / 2);
				table[idx] = vec;
			}
		}

		boolean contains(long value) {
			int idx = (int) (value & (table.length - 1));
			long[] vec = table[idx];
			for (int i = 1; i < vec[0]; i++) {
				if (vec[i] == value) {
					return true;
				}
			}
			return false;
		}
	}

	static MyScanner in = new MyScanner();
	static PrintWriter out = new PrintWriter(new BufferedOutputStream(System.out));

	public static void main(String[] args) {

		LongSet set1 = new LongSet(1 << 20);
		LongSet set2 = new LongSet(1 << 20);
		int n = in.nextInt();
		String[] strings = new String[n];
		for (int i = 0; i < n; i++) {
			Set<Long> used = new HashSet<>();
			String s = in.next();
			strings[i] = s;
			char[] arr = s.toCharArray();
			for (int l = 0; l < arr.length; l++) {
				long h = 1;
				for (int j = l; j < arr.length; j++) {
					h = h * 5381 + arr[j] - '0' + 1;
					if (!used.contains(h) && set1.contains(h)) {
						set2.add(h);
					} else {
						set1.add(h);
					}
					used.add(h);
				}
			}
		}
		for (int i = 0; i < n; i++) {
			String s = strings[i];
			int min = 100;
			int left = 0, right = 0;
			char[] arr = s.toCharArray();
			for (int l = 0; l < arr.length; l++) {
				long h = 1;
				for (int j = l; j < arr.length; j++) {
					h = h * 5381 + arr[j] - '0' + 1;
					if (set1.contains(h) && !set2.contains(h)) {
						if (j - l + 1 <= min) {
							left = l;
							right = j;
							min = j - l + 1;
						}
					}
				}
			}
			out.println(s.substring(left, right + 1));
		}

		/*
		 * Dont delete
		 */
		out.close();
		/*
		 * 
		 */
	}

	public static long GCD(long a, long b) {
		return b == 0 ? a : GCD(b, a % b);
	}

	public static long LCM(long a, long b) {
		return a * b / GCD(a, b);
	}

	public static class MyScanner {
		BufferedReader br;
		StringTokenizer st;

		public MyScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		String next() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				} catch (IOException e) {
					e.printStackTrace();
				}
			}
			return st.nextToken();
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}

		String nextLine() {
			String str = "";
			try {
				str = br.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}
			return str;
		}
	}
