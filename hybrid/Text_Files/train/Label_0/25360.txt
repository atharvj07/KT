#include <bits/stdc++.h>
using namespace std;
long long P = (long long)12e10, MOD = (long long)1e10;
long long mul(long long a, long long b, long long p) {
  a %= p, b %= p;
  return (a * b - (long long)(a / (long double)p * b + 1e-10) * p + p) % p;
}
long long S(long long t) { return (mul(t, t + 1, P) / 2) % MOD; }
long long S1(long long t) {
  return mul(mul(t, t + 1, P), 2 * t + 1, P) / 6 % MOD;
}
long long A(long long first, long long second) {
  return (mul((S1(first) - S1(second) + MOD), second, MOD) -
          mul(S(second - 1), (first - second), MOD) + MOD) %
         MOD;
}
long long B(long long first, long long second) {
  return (mul((S1(second - 1) - S1(first - 1) + MOD), first, MOD) +
          mul(S(first), (second - first), MOD) + MOD) %
         MOD;
}
long long calc(long long first, long long second) {
  if (!first || !second) return 0;
  long long Tohka = S(((min(first, second)) * (min(first, second))));
  if (first > second)
    Tohka = (Tohka + A(first, second)) % MOD;
  else
    Tohka = (Tohka + B(first, second)) % MOD;
  return Tohka;
}
void solve() {
  long long xl, yl, xr, yr;
  cin >> xl >> yl >> xr >> yr;
  P = (long long)12e10, MOD = (long long)1e10;
  long long a = (calc(xr, yr) - calc(xl - 1, yr) - calc(xr, yl - 1) +
                 calc(xl - 1, yl - 1) + MOD * 2) %
                MOD;
  P = (long long)12e10 + 12, MOD = (long long)1e10 + 1;
  long long b = (calc(xr, yr) - calc(xl - 1, yr) - calc(xr, yl - 1) +
                 calc(xl - 1, yl - 1) + MOD * 2) %
                MOD;
  if (a == b)
    cout << a << endl;
  else
    printf("...%010I64d\n", a);
}
int main() {
  int test;
  scanf("%d", &test);
  for (; test; --test) solve();
  fclose(stdin);
  fclose(stdout);
  return 0;
}
