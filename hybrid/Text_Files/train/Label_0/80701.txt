from collections import Counter

def solve(letters):
	counts = Counter(letters)
	if counts['M'] * 2 != counts['T']:
		print('NO')
		return
	n = counts['M']
	(pre, post, m) = ([], [], [])
	for (pos, l) in enumerate(letters):
		if l == 'T':
			if len(pre) < n:
				pre.append(pos)
			else:
				post.append(pos)
		else:
			m.append(pos)
	if len(pre) != len(post) != len(m) != n:
		print('NO')
		return
	for x in range(n):
		if not pre[x] < m[x] < post[x]:
			print('NO')
			return
	print('YES')
for _ in range(int(input())):
	_ = input()
	solve(input())
