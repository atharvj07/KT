	public static int[][] toadjacency(int[][] list, int v) {
		int[] counter = new int[v];
		int n = list.length;
		for (int i = 0; i < n; i++) {
			counter[list[i][0]]++;
			counter[list[i][1]]++;
		}
		int[][] adjacency = new int[v][];
		for (int i = 0; i < v; i++) {
			adjacency[i] = new int[counter[i]];
		}
		int[] counter2 = new int[v];
		for (int i = 0; i < n; i++) {
			adjacency[list[i][0]][counter2[list[i][0]]] = list[i][1];
			counter2[list[i][0]]++;
			adjacency[list[i][1]][counter2[list[i][1]]] = list[i][0];
			counter2[list[i][1]]++;
		}
		
		return adjacency;
	}
	
	public static int[][] BFSsearch (int[][] adjacency, int v, int index) {
		boolean[] found = new boolean[v];
		found[index] = true;
		int[][] indexList = new int[v][2];
		int start = 0;
		int end = 1;
		indexList[0][0] = index;
		indexList[0][1] = index;
		while ((end < v) && (start < end)) {
			for (int i = 0; i < adjacency[indexList[start][0]].length; i++) {
				if (found[adjacency[indexList[start][0]][i]] == false) {
					found[adjacency[indexList[start][0]][i]] = true;
					indexList[end][0] = adjacency[indexList[start][0]][i];
					indexList[end][1] = indexList[start][0];
					end++;
				}
			}
			start++;
		}
		return indexList;
	}
	
	public static int findIndex(int[] n, int[][] list) {
		int r = list.length;
		if ((n[0] > list[r-1][0]) || ((n[0] == list[r-1][0]) && (n[1] > list[r-1][1])))
			return r;
		else {
			int low = 0;
			int high = r-1;
			int val = 0;
			while (low <= high) {
				val = (low+high)/2;
				if ((list[val][0] == n[0]) && (list[val][1] == n[1]))
					return val;
				else if ((list[val][0] > n[0]) || ((list[val][0] == n[0]) && (list[val][1] > n[1])))
					high = val-1;
				else
					low = val+1;
			}
			return low;
		}
	}
	
	public static int[][] keysort(int[][] lst, int key) {
		int n = lst.length;
		if (n == 1) {
			return lst;
		}
		int m = n/2;
		int[][] lst1 = keysort(Arrays.copyOfRange(lst, 0, m), key);
		int[][] lst2 = keysort(Arrays.copyOfRange(lst, m, n), key);
		int[][] lstNew = new int[n][2];
		int counter1 = 0;
		int counter2 = 0;
		while ((counter1 < m) && (counter2 < n-m)) {
			if (lst1[counter1][key] <= lst2[counter2][key]) {
				lstNew[counter1+counter2] = lst1[counter1];
				counter1++;
			}
			else {
				lstNew[counter1+counter2] = lst2[counter2];
				counter2++;
			}
		}
		
		while (counter1 < m) {
			lstNew[counter1+counter2] = lst1[counter1];
			counter1++;
		}
		
		while (counter2 < n-m) {
			lstNew[counter1+counter2] = lst2[counter2];
			counter2++;
		}
		
		return lstNew;
	}

}
