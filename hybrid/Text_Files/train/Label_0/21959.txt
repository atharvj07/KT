#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 7;
int a[200001], pos[200001];
void solve() {
  int i, j, k, n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    pos[x] = i;
  }
  long long low = 1, high = n;
  while (low < high) {
    long long mid = (low + high + 1) >> 1;
    for (i = 2; i <= mid; i++) {
      if (pos[a[i]] < pos[a[i - 1]]) break;
    }
    if (i > mid)
      low = mid;
    else
      high = mid - 1;
  }
  cout << n - low << endl;
  return;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  solve();
  return 0;
}
