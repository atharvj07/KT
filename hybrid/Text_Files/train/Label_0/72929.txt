for T in range(int(input())):
	(a, b, k) = map(int, input().split())
	l = list(map(int, input().split()))
	m = list(map(int, input().split()))
	p = [0] * (a + 1)
	q = [0] * (b + 1)
	for i in l:
		p[i] += 1
	for i in m:
		q[i] += 1
	s = 0
	for i in range(k):
		s += k - p[l[i]] - q[m[i]] + 1
	print(s // 2)
