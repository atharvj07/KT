n = input()
m = input()
a = ['monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday', 'sunday']
k = a.index(n)
for i in range(k, 7):
	s = a[i]
	a.remove(s)
	a.insert(i - k, s)
l = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
for i in l:
	p = i % 7
	if m == a[p]:
		print('YES')
		break
else:
	print('NO')
