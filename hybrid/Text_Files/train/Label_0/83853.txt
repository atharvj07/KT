def check(x, n):
	global k
	count = 0
	for e in arr:
		if e <= x and count % 2 == n:
			count += 1
		elif count % 2 != n:
			count += 1
		if count == k:
			return True
	return False
(n, k) = map(int, input().split())
arr = list(map(int, input().split()))
l = 1
r = 10 ** 9
while l < r:
	mid = (l + r) // 2
	if check(mid, 0) or check(mid, 1):
		r = mid
	else:
		l = mid + 1
print(l)
