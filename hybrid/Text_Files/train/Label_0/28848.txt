#include<bits/stdc++.h>
#define ll long long
#define fo(i,x,y) for(int i=x;i<=y;i++)
#define fd(i,x,y) for(int i=x;i>=y;i--)
using namespace std;

const int maxn=500007;

int n,L;
int SumV=0;

int hd,tl;
struct Water{
	int V;
	double T;
	Water(){}
	Water(int _V,double _T):V(_V),T(_T){}
	void operator +=(const Water &b){
		T=(T*V+b.T*b.V)/(V+b.V);
		V+=b.V;
	}
	void operator -=(const Water &b){
		if (V-b.V==0){
			T=0;
			V=0;
		}else{
			T=(T*V-b.T*b.V)/(V-b.V);
			V-=b.V;
		}
	}
	void operator -=(const int &b){
		SumV-=b;
	}
	double operator()(){
		return T/V;
	}
}b[maxn],SumT;

void Init(){
	scanf("%d%d",&n,&L);
}

void Solve(){
	hd=1,tl=0;
	fo(i,1,n){
		int T,V;
		scanf("%d%d",&T,&V);
		b[++tl]=Water(V,T);
		SumV+=V;
		SumT+=b[tl];
		while (SumV>L){
			int Temp=min(SumV-L,b[hd].V);
			SumT-=b[hd];
			b[hd].V-=Temp;
			SumV-=Temp;
			if (b[hd].V==0) hd++;
			else SumT+=b[hd];
		}
		while (hd<tl && b[tl].T<=b[tl-1].T){
			SumT-=b[tl];
			SumT-=b[tl-1];
			b[tl-1]+=b[tl];
			SumT+=b[tl-1];
			tl--;
		}
		printf("%.7lf\n",SumT.T);
	}
}

int main(){
	Init();
	Solve();
	return 0;
}
