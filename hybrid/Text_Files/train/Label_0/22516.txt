#include <bits/stdc++.h>
using namespace std;
const int N = 2010;
int n;
struct node {
  int x, y, z, id;
  bool operator<(const node &tmp) const {
    if (x != tmp.x) return x < tmp.x;
    return y != tmp.y ? y < tmp.y : z < tmp.z;
  }
} A[N];
bool vis[N];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> A[i].x >> A[i].y >> A[i].z, A[i].id = i;
  }
  sort(A + 1, A + n + 1);
  int l = 1;
  for (int i = 2; i <= n; i++) {
    if (l && A[i].x == A[l].x && A[i].y == A[l].y) {
      vis[i] = vis[l] = 1;
      cout << A[i].id << " " << A[l].id << endl;
      l = 0;
    } else
      l = i;
  }
  l = 1;
  while (vis[l]) l++;
  for (int i = l + 1; i <= n; i++) {
    if (vis[i]) continue;
    if (l && A[i].x == A[l].x) {
      vis[i] = vis[l] = 1;
      cout << A[i].id << " " << A[l].id << endl;
      l = 0;
    } else
      l = i;
  }
  l = 1;
  while (vis[l]) l++;
  for (int i = l + 1; i <= n; i++) {
    if (vis[i]) continue;
    if (l)
      cout << A[i].id << " " << A[l].id << endl, l = 0;
    else
      l = i;
  }
  return 0;
}
