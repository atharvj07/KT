from sys import stdin
from heapq import heappush, heappop

def main():
	(n, r, m) = (int(input()), 0, 1000000007)
	(bb, ss, bs) = ([0], [m], [])
	for s in stdin.read().splitlines():
		if s[1] == 'D':
			p = int(s[4:])
			if ss and ss[0] < p:
				heappush(ss, p)
			elif bb and -bb[0] > p:
				heappush(bb, -p)
			else:
				bs.append(p)
		else:
			p = int(s[7:])
			if -bb[0] < p < ss[0]:
				r += 1
			elif p == ss[0]:
				heappop(ss)
			elif p == -bb[0]:
				heappop(bb)
			else:
				print(0)
				return
			for x in bs:
				if x < p:
					heappush(bb, -x)
				elif x > p:
					heappush(ss, x)
			bs = []
	print((len(bs) + 1) * pow(2, r, m) % m)
main()
