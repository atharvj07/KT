#include<bits/stdc++.h>
using namespace std;
int n,m,s,t;
int op[200][200];
int fa[200][200];
int fb[200][200];
int fc[200][200];
int fd[200][200];
int num[200][200];
int Anum[200][200];
int Cnum[200][200];
int Onum[200][200];
int numO[200][200];

struct data{
  int from,to,type;
  data(int a,int b,int c):from(a),to(b),type(c){}
};

void init(){
  for(int i=0;i<n;i++){
    for(int j=0;j<n;j++){
      op[i][j]=0;
      fa[i][j]=0;
      fb[i][j]=0;
      fc[i][j]=0;
      fd[i][j]=0;
      num[i][j]=0;
      Anum[i][j]=0;
      Cnum[i][j]=0;
      Onum[i][j]=0;
    }
  }
}

bool solve(){
  queue< data > Q;
  for(int i=0;i<n;i++)
    for(int j=0;j<n;j++)
      if(num[i][j]==1)
        Q.push(data(i,j,0));
  
  while(!Q.empty()){
    data d=Q.front();Q.pop();
    if(d.from==s&&d.to==t&&d.type==0)return true;
    if(d.type==0){
      for(int nfrom=0;nfrom<n;nfrom++){
        if(op[nfrom][d.from]==1){
          if(Onum[nfrom][d.to]==0){
            Onum[nfrom][d.to]=1;
            Q.push(data(nfrom,d.to,3));
          }
        }
        if(fa[nfrom][d.from]==1){
          if(Anum[nfrom][d.to]==0){
            Anum[nfrom][d.to]=1;
            Q.push(data(nfrom,d.to,1));
          }
        }
        if(fc[nfrom][d.from]==1){
          if(Cnum[nfrom][d.to]==0){
            Cnum[nfrom][d.to]=1;
            Q.push(data(nfrom,d.to,2));
          }
        }
      }
      for(int nto=0;nto<n;nto++){
        if(Onum[d.to][nto]==1){
          if(num[d.from][nto]==0){
            num[d.from][nto]=1;
            Q.push(data(d.from,nto,0));
          }
        }
      }
    }
    if(d.type==1){
      for(int nto=0;nto<n;nto++){
        if(fb[d.to][nto]==1){
          if(num[d.from][nto]==0){
            num[d.from][nto]=1;
            Q.push(data(d.from,nto,0));
          }
        }
      }
    }
    if(d.type==2){
      for(int nto=0;nto<n;nto++){
        if(fd[d.to][nto]==1){
          if(num[d.from][nto]==0){
            num[d.from][nto]=1;
            Q.push(data(d.from,nto,0));
          }
        }
      }
    }
    if(d.type==3){
      for(int nfrom=0;nfrom<n;nfrom++){
        if(num[nfrom][d.from]==1){
          if(num[nfrom][d.to]==0){
            num[nfrom][d.to]=1;
            Q.push(data(nfrom,d.to,0));
          }
        }
      }
    }
  }
  return false;
}

int main(){
  while(1){
    cin>>n>>m>>s>>t;
    if(n==0&&m==0&&s==0&&t==0)break;
    s--,t--;
    init();
    for(int i=0;i<m;i++){
      int from,to;
      string str;
      cin>>from>>to>>str;
      from--,to--;
      if(str=="*"||str=="+"){
        op[from][to]=1;
      }else if(str=="("){
        fa[from][to]=1;
      }else if(str==")"){
        fb[from][to]=1;
      }else if(str=="["){
        fc[from][to]=1;
      }else if(str=="]"){
        fd[from][to]=1;
      }else if(str=="a"){
        num[from][to]=1;
      }
    }
    cout<< (solve()?"Yes":"No") <<endl;
  }
  return 0;
}
