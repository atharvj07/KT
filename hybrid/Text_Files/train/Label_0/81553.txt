import math
num = []
s = input()
for i in s:
	num.append(int(i))
number = 0
n = len(num)
num2 = []
num2.append(num[0])
num2.append(num[2])
num2.append(num[4])
num2.append(num[3])
num2.append(num[1])
for i in range(n):
	number = number * 10 + num2[i]
number = number ** 5
answer = number % 100000
l = [int(x) for x in str(answer)]
l.reverse()
while len(l) < 5:
	l.append(0)
l.reverse()
strans = ''.join((str(e) for e in l))
print(strans)
