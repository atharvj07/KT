a = list(map(int, input().split()))
(pages, current, wide) = (a[0], a[1], a[2])
mid = '(' + str(current) + ')'
(start, end) = (current - wide, current + wide)
(prefix, suffix) = ('', '')
if start < 2:
	prefix = ' '.join([str(i) for i in range(1, current)]).rstrip()
else:
	prefix = ('<< ' + ' '.join([str(i) for i in range(start, current)])).rstrip()
if end >= pages:
	suffix = ' '.join([str(i) for i in range(current + 1, pages + 1)]).rstrip()
else:
	suffix = (' '.join([str(i) for i in range(current + 1, end + 1)]) + ' >>').rstrip()
if prefix == '' and suffix == '':
	print(mid)
elif prefix == '':
	print(mid, suffix)
elif suffix == '':
	print(prefix, mid)
else:
	print(prefix, mid, suffix)
