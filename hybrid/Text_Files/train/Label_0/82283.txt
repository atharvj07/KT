for test in range(int(input())):
	n = int(input())
	b = list(map(int, input().split()))
	has = {}
	s = 0
	for val in b:
		s += val
		if val in has:
			has[val] += 1
		else:
			has[val] = 1
	ok = False
	for val in b:
		x = s - 2 * val
		has[val] -= 1
		if x in has and has[x] > 0:
			has[x] -= 1
			ok = True
			break
		has[val] += 1
	if ok == False:
		print('-1')
	else:
		for (key, values) in has.items():
			for j in range(values):
				print(key, end=' ')
		print()
