from math import log2
for _ in range(int(input())):
	(least, large) = map(int, input().strip().split())
	leastbin = bin(least)[2:]
	best = 1
	b = int(log2(large))
	for i in range(b, -1, -1):
		power = 2 ** i
		firstone = int(leastbin[:len(leastbin) - i - 1] + '1' + '0' * i, 2)
		if least < power:
			firstone = power
		dif = large - firstone + 1
		if firstone > large:
			continue
		ans = dif // (power * 2) * power + min(power, dif % (power * 2))
		if firstone <= least:
			ans -= least - firstone
		best = max(best, ans)
	print(large - least - best + 1)
