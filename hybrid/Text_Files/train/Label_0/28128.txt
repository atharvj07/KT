#include <bits/stdc++.h>
using namespace std;
const int Mod = 1000000007;
void add(int &x, int y) {
  x += y;
  if (x >= Mod) x -= Mod;
}
int mul(long long x, long long y) {
  x *= y;
  if (x >= Mod) x %= Mod;
  return x;
}
struct mat {
  int g[16][16], n, m;
  mat operator*(const mat &o) {
    mat c;
    c.n = n;
    c.m = o.m;
    for (int i = 0; i < n; ++i)
      for (int j = 0; j < o.m; ++j) {
        c.g[i][j] = 0;
        for (int k = 0; k < m; ++k) add(c.g[i][j], mul(g[i][k], o.g[k][j]));
      }
    return c;
  }
  mat operator+(const mat &o) {
    mat c;
    c.n = n;
    c.m = m;
    for (int i = 0; i < n; ++i)
      for (int j = 0; j < m; ++j) {
        c.g[i][j] = g[i][j];
        add(c.g[i][j], o.g[i][j]);
      }
    return c;
  }
} a, s, f;
int l, r, ans;
int calc(int x) {
  if (x <= 1) return 0;
  a.n = 1;
  a.m = 16;
  for (int i = 0; i < 16; ++i) a.g[0][i] = 0;
  f.n = f.m = 16;
  for (int i = 0; i < 16; ++i)
    for (int j = 0; j < 16; ++j) f.g[i][j] = 0;
  for (int i = 0; i < 4; ++i)
    for (int j = 0; j < 4; ++j) {
      if (i == j) continue;
      if (((i) < (j) ? (i) : (j)) == 0 && ((i) > (j) ? (i) : (j)) == 3)
        continue;
      if (((i) < (j) ? (i) : (j)) == 1 && ((i) > (j) ? (i) : (j)) == 2)
        continue;
      a.g[0][i * 4 + j]++;
      for (int k = 0; k < 4; ++k) {
        if (j == k) continue;
        if (((j) < (k) ? (j) : (k)) == 0 && ((j) > (k) ? (j) : (k)) == 3)
          continue;
        if (((j) < (k) ? (j) : (k)) == 1 && ((j) > (k) ? (j) : (k)) == 2)
          continue;
        if (i == 1 && j == 0 && k == 2) continue;
        if (i == 2 && j == 0 && k == 1) continue;
        f.g[i * 4 + j][j * 4 + k]++;
      }
    }
  s = a;
  x -= 2;
  while (x) {
    if (x & 1) s = s * f + a;
    x >>= 1;
    a = a * f + a;
    f = f * f;
  }
  int ans = 0;
  for (int i = 0; i < 16; ++i) add(ans, s.g[0][i]);
  return ans;
}
int main() {
  scanf("%d%d", &l, &r);
  if (l == 1) ans = 8, ++l;
  add(ans, calc(r));
  add(ans, calc((r + 1) >> 1));
  ans -= calc(l - 1);
  if (ans < 0) ans += Mod;
  ans -= calc(l >> 1);
  if (ans < 0) ans += Mod;
  ans = mul(ans, 500000004);
  printf("%d\n", ans);
  return 0;
}
