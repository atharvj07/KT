import java.util.*;
import java.io.*;

public class Main implements Runnable {
	private void solution() throws IOException {
		int n = in.nextInt();
		int m = in.nextInt();
		int[] a = new int[m];
		for (int i = 0; i < m; ++i) {
			a[i] = in.nextInt();
		}
		double res = 0;
		for (int i = 1; i <= n; ++i) {
			res += i * f(i, n, m, a);
		}
		out.printf(Locale.US, "%.12f\n", res);
	}

	private double f(int exact, int n, int m, int[] a) {
		Double[][][] am = new Double[n + 1][m + 1][2];
		double div = Math.pow(m, n);
		double res = solve(n, 0, a, exact, 0, am);
		return res / div;
	}

	private double solve(int n, int at, int[] a, int exact, int have, Double[][][] am) {
		if (at == a.length) {
			return (n == 0 && have == 1) ? 1 : 0;
		}
		if (am[n][at][have] != null) {
			return am[n][at][have];
		}
		double res = 0;
		for (int i = 0; i <= exact * a[at] && i <= n; ++i) {
			double cnt = CNK(n, i);
			int nhave = Math.max(have, (i + a[at] - 1) / a[at] == exact ? 1 : 0);
			res += cnt * solve(n - i, at + 1, a, exact, nhave, am);
		}
		return am[n][at][have] = res;
	}
	double[][] C = new double[100][100];
	{
		C[0][0] = 1;
		for (int i = 1; i < C.length; ++i) {
			C[i][0] = C[i][i] = 1;
			for (int j = 1; j < i; ++j) {
				C[i][j] = C[i - 1][j] + C[i - 1][j - 1];
			}
		}
	}

	private double CNK(int n, int k) {
		return C[n][k];
	}

	public void run() {
		try {
			solution();
			in.reader.close();
			out.close();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	private class Scanner {
		BufferedReader reader;
		StringTokenizer tokenizer;

		public Scanner(Reader reader) {
			this.reader = new BufferedReader(reader);
			this.tokenizer = new StringTokenizer("");
		}

		public boolean hasNext() throws IOException {
			while (!tokenizer.hasMoreTokens()) {
				String next = reader.readLine();
				if (next == null) {
					return false;
				}
				tokenizer = new StringTokenizer(next);
			}
			return true;
		}

		public String next() throws IOException {
			hasNext();
			return tokenizer.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(in.next());
		}

		public String nextLine() throws IOException {
			tokenizer = new StringTokenizer("");
			return reader.readLine();
		}
	}

	public static void main(String[] args) throws IOException {
		new Thread(null, new Main(), "", 1 << 28).start();
	}
	PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
	Scanner in = new Scanner(new InputStreamReader(System.in));
}
