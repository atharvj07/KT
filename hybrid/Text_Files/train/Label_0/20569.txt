#include <bits/stdc++.h>
using namespace std;
const int Nmax = 5 * 1e5 + 10;
int n, a, b, T, i, l, ans;
int rcost, lcost;
bool v[Nmax];
int cost(int l, int r) {
  if (l == n + 1) return rcost;
  int cost1 = rcost + (r - 1) * a + lcost;
  int cost2 = lcost + (n - l + 1) * a + rcost;
  return min(cost1, cost2);
}
int main() {
  scanf("%d", &n);
  scanf("%d %d", &a, &b);
  scanf("%d\n", &T);
  for (i = 1; i <= n; ++i) {
    char ch;
    scanf("%c", &ch);
    if (ch == 'w') v[i] = 1;
  }
  rcost = 1 + v[1] * b;
  for (i = n; i > 1; --i) {
    lcost += a + 1 + v[i] * b;
    if (cost(i, 1) > T) break;
  }
  if (cost(i, 1) > T) lcost -= a + 1 + v[i] * b, i++;
  if (i == 1) i++;
  l = i;
  if (cost(l, 1) <= T) ans = max(ans, n - l + 1 + 1);
  for (i = 2; i <= n; ++i) {
    rcost += a + 1 + v[i] * b;
    while (l != n + 1 && (cost(l, i) > T || l == i))
      lcost -= a + 1 + v[l] * b, l++;
    if (cost(l, i) > T) break;
    ans = max(ans, n - l + 1 + i);
  }
  printf("%d\n", ans);
  return 0;
}
