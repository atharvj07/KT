import atexit
import io
import sys
_I_B = sys.stdin.read().splitlines()
input = iter(_I_B).__next__
_O_B = io.StringIO()
sys.stdout = _O_B

@atexit.register
def write():
	sys.__stdout__.write(_O_B.getvalue())

def main():
	for _ in range(int(input())):
		n = int(input())
		a = list(map(int, input().split()))
		d = dict()
		for x in a:
			try:
				d[x] += 1
			except KeyError:
				d[x] = 1
		ld = len(d)
		ans = -1
		for i in d:
			if d[i] == 1:
				ans = max(ans, min(d[i], ld - 1))
			else:
				ans = max(ans, max(min(ld - 1, d[i]), min(ld, d[i] - 1)))
		print(ans)
main()
