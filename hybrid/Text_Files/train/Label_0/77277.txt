import math

def root(n):
	(l, r) = (int(0), int(10 ** 20))
	while l <= r:
		m = (l + r) // 2
		if m * m == n:
			return m
		elif m * m > n:
			r = m - 1
		else:
			l = m + 1
	return int(l)
n = int(input())
ans = []
a = 0
for i in range(63):
	disc = int((2 * a - 1) * (2 * a - 1) + 8 * n)
	r = root(disc)
	if r * r == disc and r % 2 == 1:
		x = (1 - 2 * a + r) // 2
		y = (1 - 2 * a - r) // 2
		if x % 2 == 1 and x > 0:
			ans.append(int(x * (a + 1)))
		if y % 2 == 1 and y > 0:
			ans.append(int(y * (a + 1)))
	a = 2 * a + 1
if len(ans) == 0:
	print(-1)
else:
	ans = list(set(ans))
	for x in ans:
		print(int(x))
