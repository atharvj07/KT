import sys
input = sys.stdin.buffer.readline
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
(mi, ma) = (min(a), max(a))
l = [0] * (ma + 1)
l[0] = n
for i in a:
	l[i] -= 1
for i in range(1, ma + 1):
	l[i] += l[i - 1]
sum = 0
ans = 0
for i in range(ma + 1):
	if l[i] < n:
		sum += l[i]
		if sum > k:
			sum = l[i]
			ans += 1
ans += 1 if sum else 0
print(ans)
