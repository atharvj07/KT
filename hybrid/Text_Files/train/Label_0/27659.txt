#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  vector<int> sta;
  cin >> T;
  for (int i = 0; i < T; i++) {
    int temp;
    cin >> temp;
    sta.push_back(temp);
  }
  int s, t;
  cin >> s >> t;
  if (s > t) {
    int temp = s;
    s = t;
    t = temp;
  }
  int result1 = 0;
  int result2 = 0;
  for (int i = s; i != t; i++) {
    result1 += sta[i - 1];
  }
  for (int i = t; i != s; i++) {
    result2 += sta[i - 1];
    if (i == sta.size()) i = 0;
  }
  if (result1 > result2)
    cout << result2 << endl;
  else
    cout << result1 << endl;
  return 0;
}
