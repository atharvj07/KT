import java.io.*;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Random;
import java.util.StringTokenizer;

public class C {
 
    public static void main(String[] args) {
        FastReader scan = new FastReader();
        PrintWriter out = new PrintWriter(System.out);
        Task solver = new Task();
        int t = scan.nextInt();
        for(int tt = 1; tt <= t; tt++) solver.solve(tt, scan, out);
        out.close();
    }

    static class Task {

        public void solve(int testNumber, FastReader scan, PrintWriter out) {
            int n = scan.nextInt(), k = scan.nextInt();
            char[] s = scan.next().toCharArray();
            shuffle(s);
            Arrays.sort(s);
            if(k == 1) out.println(new String(s));
            else {
                HashSet<Character> h = new HashSet<>();
                for(char c : s) h.add(c);
                if(h.size() == 1) {
                    int need = n / k + Math.min(1, n % k);
                    for(int i = 0; i < need; i++) out.print(s[0]);
                    out.println();
                }
                else if(s[k - 1] != s[0]){
                    out.println(s[k - 1]);
                }
                else {
                    if(h.size() == 2 && s[k] != s[0]) {
                        int[] count = new int[26];
                        for(int i = k; i < n; i++) count[s[i] - 'a']++;
                        int last = s[n - 1] - 'a';
                        int need = count[last] / k + Math.min(1, count[last] % k);
                        out.print(s[k - 1]);
                        for(int i = 0; i < need; i++) out.print(s[n - 1]);
                        out.println();
                    }
                    else {
                        for(int i = k - 1; i < n; i++) out.print(s[i]);
                        out.println();
                    }
                }
            }
        }
    }

    static void shuffle(int[] a) {
        Random get = new Random();
        for (int i = 0; i < a.length; i++) {
            int r = get.nextInt(a.length);
            int temp = a[i];
            a[i] = a[r];
            a[r] = temp;
        }
    }

    static void shuffle(char[] a) {
        Random get = new Random();
        for (int i = 0; i < a.length; i++) {
            int r = get.nextInt(a.length);
            char temp = a[i];
            a[i] = a[r];
            a[r] = temp;
        }
    }

    static void shuffle(long[] a) {
        Random get = new Random();
        for (int i = 0; i < a.length; i++) {
            int r = get.nextInt(a.length);
            long temp = a[i];
            a[i] = a[r];
            a[r] = temp;
        }
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        public FastReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public FastReader(String s) throws FileNotFoundException {
            br = new BufferedReader(new FileReader(new File(s)));
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine() {
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }
    }

}
