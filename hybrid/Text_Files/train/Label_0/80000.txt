for _ in range(int(input())):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	x = sorted(list(set(a)))
	s = len(x)
	if k < s:
		print(-1)
	else:
		print(k * n)
		diff = k - s
		x.extend([i for i in range(1, min(1 + diff, x[0]))])
		diff -= x[0] - 1
		for i in range(s - 1):
			if diff <= 0:
				break
			if x[i + 1] - x[i] - 1 > 0:
				x.extend([i for i in range(x[i] + 1, x[i] + 1 + min(x[i + 1] - x[i] - 1, diff))])
				diff -= x[i + 1] - x[i] - 1
		if diff > 0:
			x.extend([i + x[s - 1] for i in range(1, diff + 1)])
		x.sort()
		temp = [str(i) for i in x]
		st = ' '.join(temp) + ' '
		print(st * n)
