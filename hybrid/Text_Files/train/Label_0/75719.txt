def ok(x, y, z):
	if abs(x - y) == 1 and abs(y - z) == 1:
		return False
	else:
		return True
(n, m) = map(int, input().split())
if m == 0:
	(print('YES'), exit())
a = list(map(int, input().split()))
a.sort()
if n == m:
	print('NO')
elif a[0] == 1 or a[m - 1] == n:
	print('NO')
elif m < 3:
	print('YES')
else:
	print('NO') if False in list(map(ok, a[0:m - 2], a[1:m - 1], a[2:m])) else print('YES')
