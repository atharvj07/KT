import java.io.*;
import java.util.*;

public class Solution {
    StreamTokenizer ik;
    BufferedReader in;
    PrintWriter out;

    public static void main(String[] args) throws Exception {
        new Solution().run();
    }

    int nextInt() throws Exception {
        ik.nextToken();
        return (int) ik.nval;
    }

    String next() throws Exception {
        ik.nextToken();
        return ik.sval;
    }


    public void run() throws Exception {
        ik = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
     //   in = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(new OutputStreamWriter(System.out));
        char [][] a = new char [1050][1050];
        char [] ch = new char [1000];
        int [][] fin = new int [1050][1050];
        ik.ordinaryChars('.','.');
        ik.wordChars('.','.');
        int n = nextInt();
        int m = nextInt();
        for (int i=0; i<n; i++)
        {
           ik.nextToken();
           String st =ik.sval;
           ch = st.toCharArray();
            for (int j=0; j<m; j++)
                a[i][j] = ch[j];
        }
        int por=0;
        for (int i=0; i<n; i++)
            for (int j=0; j<m; j++)
            {
               if (a[i][j]=='b') {out.println("NO"); out.flush(); return;} else
                   if (a[i][j]=='w')
                   {
                       if ((a[i][j+1]=='b') && (a[i][j+2]=='w'))
                       {
                           por++;
                           fin[i][j]=por;
                           fin[i][j+1]=por;
                           fin[i][j+2]=por;
                           a[i][j]='X';
                           a[i][j+1]='X';
                           a[i][j+2]='X';
                       } else
                       if ((a[i+1][j]=='b') && (a[i+2][j]=='w'))
                       {
                           por++;
                           fin[i][j]=por;
                           fin[i+1][j]=por;
                           fin[i+2][j]=por;
                           a[i][j]='X';
                           a[i+1][j]='X';
                           a[i+2][j]='X';
                       }
                       else { out.println("NO"); out.flush(); return;}

                   }
            }
        out.println("YES");
        for (int i=0; i<n; i++)
            for (int j=0; j<m;j++)
               if (fin[i][j]>0) //if not '.' go to next
                   if (fin[i+1][j]==fin[i][j])
                   {
                       if ((j&1)==1) fin[i][j]=-1; //mask..
                       else fin[i][j]=-2;
                       fin[i+1][j]=fin[i][j];
                       fin[i+2][j]=fin[i][j];
                   } else
                   {
                       if ((i&1)==1) fin[i][j]=-3;
                       else fin[i][j]=-4;
                       fin[i][j+1]=fin[i][j];
                       fin[i][j+2]=fin[i][j];
                   }
        for (int i=0; i<n; i++)
        {
            for (int j=0; j<m; j++)
            {
                if (fin[i][j]==0) out.print('.'); else
                    out.print((char)(96-fin[i][j])); //'a,b,c,d'
            }
            out.println();
        }
        out.flush();
    }

}
