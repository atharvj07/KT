from sys import stdin
from collections import defaultdict
input = stdin.readline

def put():
	return map(int, input().split())

def dfs(i):
	s = [i]
	points.remove(i)
	while s:
		i = s.pop()
		for j in tuple(points):
			if (min(i, j), max(i, j)) not in mp:
				s.append(j)
				points.remove(j)
(n, m) = put()
mp = defaultdict()
points = set(range(n))
for _ in range(m):
	(x, y) = put()
	(x, y) = (x - 1, y - 1)
	key = (min(x, y), max(x, y))
	mp[key] = 1
ans = -1
for i in range(n):
	if i in points:
		dfs(i)
		ans += 1
print(ans)
