#include <bits/stdc++.h>
using namespace std;
long long c[500010], S[500010];
int main() {
  int N, K, fir, se, st, en, x;
  long long val;
  scanf("%d%d", &N, &K);
  for (int i = (1); i < (N + 1); i++) scanf("%I64d", c + i);
  c[0] = 0;
  sort(c, c + N + 1);
  S[0] = 0;
  for (int i = (1); i < (N + 1); i++) S[i] = S[i - 1] + c[i];
  fir = N + 1;
  se = -1;
  for (int i = (1); i < (N + 1); i++) {
    val = c[i] * i - S[i];
    if (val > K) break;
    x = c[i] + (K - val + i - 1) / i;
    fir = i;
    if ((K - val) % i == 0)
      st = x;
    else
      st = x - 1;
  }
  for (int i = N; i > 0; i--) {
    val = S[N] - S[i - 1] - c[i] * (N - i + 1);
    if (val > K) break;
    x = c[i] - (K - val + N - i) / (N - i + 1);
    se = i;
    if ((K - val) % (N - i + 1) == 0)
      en = x;
    else
      en = x + 1;
  }
  if (fir >= se || (fir == se - 1 && en <= st)) {
    if (S[N] % N == 0)
      puts("0");
    else
      puts("1");
  } else
    printf("%d\n", en - st);
  return 0;
}
