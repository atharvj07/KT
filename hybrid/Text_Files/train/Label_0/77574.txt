(n, k) = map(int, input().strip().split())
a = []
a = list(map(int, input().strip().split()))
candies = 0
candinstr = 0
flag = 0
for i in range(n):
	if a[i] > 8:
		candies += 8
		candinstr = candinstr + a[i] - 8
	elif candinstr + a[i] > 8:
		candies += 8
		candinstr = candinstr + a[i] - 8
	else:
		candies += candinstr + a[i]
		candinstr = 0
	if candies >= k:
		flag = 1
		print(i + 1)
		break
if flag == 0:
	print('-1')
