#include <bits/stdc++.h>
using namespace std;
const double PI = acos(-1.0);
template <class T>
inline int size(const T& c) {
  return c.size();
}
int main() {
  int n, m, s;
  cin >> n >> m >> s;
  long long ret = 0;
  for (int a = 0; a * 2 + 1 <= n; a++)
    for (int b = 0; b * 2 + 1 <= m; b++) {
      if ((a * 2 + 1) * (b * 2 + 1) == s)
        ret += (a + 1) * (b + 1) * (n - a * 2) * (m - b * 2);
      else
        for (int c = 0; c * 2 + 1 <= n; c++) {
          int d =
              s - (a * 2 + 1) * (b * 2 + 1) + (min(a, c) * 2 + 1) * (b * 2 + 1);
          if (d > 0 && d % (c * 2 + 1) == 0) {
            d /= c * 2 + 1;
            if ((d - 1) % 2 == 0) {
              d = (d - 1) / 2;
              if (d >= b && d * 2 + 1 <= m)
                ret += (n - max(a, c) * 2) * (m - d * 2);
            }
          }
          d = s - (a * 2 + 1) * (b * 2 + 1);
          if (d > 0 && (c * 2 + 1 - min(a, c) * 2 - 1) > 0 &&
              d % (c * 2 + 1 - min(a, c) * 2 - 1) == 0) {
            d /= c * 2 + 1 - min(a, c) * 2 - 1;
            if ((d - 1) % 2 == 0) {
              d = (d - 1) / 2;
              if (d >= 0 && d < b) ret += (n - max(a, c) * 2) * (m - b * 2);
            }
          }
        }
    }
  cout << ret;
  return 0;
}
