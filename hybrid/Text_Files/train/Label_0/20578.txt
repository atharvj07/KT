#define For(i,l,r) for (int i = l; i <= r; ++i)
#define Cor(i,l,r) for (int i = l; i >= r; --i)
#define Fill(a,b) memset(a,b,sizeof(a))
#define FI first
#define SE second
#define MP make_pair
#define PII pair<int,int>
#define flt double
#define INF (0x3f3f3f3f)
#define MaxN 1020304
#define MaxNode 1020304
#define MD 1000000007
#define ele(x) (1 << ((x) - 1))
#define PFF pair<flt,flt>
#define UPD(a,b) { a = min(a,b); }

const flt eps = 1e-7;
int dcmp(flt x) {
	return (x < -eps) ? -1 : ((x > eps) ? 1 : 0);
}

PFF operator - (PFF a,PFF b) { return MP(a.FI - b.FI,a.SE - b.SE); }
PFF operator + (PFF a,PFF b) { return MP(a.FI + b.FI,a.SE + b.SE); }
PFF operator * (PFF a,flt b) { return MP(a.FI * b,a.SE * b); }
flt operator * (PFF a,PFF b) { return a.FI * b.SE - a.SE * b.FI; }
flt operator / (PFF a,PFF b) { 
	if (dcmp(b.FI)) return a.FI / b.FI;
	return a.SE / b.SE;
}

flt dot(PFF a,PFF b) { return a.FI * b.FI + a.SE * b.SE; }
flt mod(PFF a) {
	return sqrt(a.FI * a.FI + a.SE * a.SE); 
}

bool Longer(PFF a,PFF b,PFF c) {
	return dcmp((c - a) / (b - a) - 1.) >= 0;
}

flt xmul(PFF a,PFF b,PFF c) {
	return (c - a) * (c - b);
}

PFF GetInter(PFF a,PFF b,PFF c,PFF d) {
	PFF as = a, ae = a + b;
	PFF bs = c,be = c + d;
	flt u = xmul(as,ae,bs);
	flt v = xmul(ae,as,be);
	if (dcmp(u + v) == 0) return MP(-INF,-INF);
	return MP(bs.FI * v + be.FI * u,bs.SE * v + be.SE * u) * (1. / (u + v));
}

int n;
PFF a[MaxN]; pair<int,flt> f[1111][11][11];

int main() {
	//freopen("input.txt","r",stdin); freopen("output.txt","w",stdout);
	while (scanf("%d",&n) != EOF && n) {
		For(i,1,n) scanf("%lf%lf",&a[i].FI,&a[i].SE);
		int All = (1 << n) - 1;
		For(i,1,n) For(j,1,n) For(k,0,All) f[k][i][j] = MP(INF,INF);
		For(i,1,n) For(j,1,n) if (i != j) {
			f[ele(i) | ele(j)][i][j] = MP(0,mod(a[i] - a[j]));
		}
		For(S,0,All) For(i,1,n) For(j,1,n) if (f[S][i][j].FI < n) {
			For(k,1,n) if (!(S & ele(k))) {
				int tInc = 1;
				if (dcmp(fabs((a[k] - a[i]) * (a[i] - a[j]))) == 0 && Longer(a[j],a[i],a[k])) tInc = 0; 
				UPD(f[S | ele(k)][k][i],MP(f[S][i][j].FI + tInc,f[S][i][j].SE + mod(a[k] - a[i])));
			}
			For(k,1,n) if (!(S & ele(k))) {
				For(l,1,n) if (!(S & ele(l)) && k != l) {
					if (dcmp((a[i] - a[j]) * (a[l] - a[k])) != 0) {
						if (k == 4 && l == 3) {
							int z = k;
						}
						PFF tInter = GetInter(a[j],a[i] - a[j],a[k],a[l] - a[k]);
						if (Longer(a[j],a[i],tInter) && Longer(a[l],a[k],tInter)) {
							UPD(f[S | ele(k) | ele(l)][l][k],MP(f[S][i][j].FI + 1,f[S][i][j].SE + mod(tInter - a[i]) + mod(tInter - a[l])));
						}
					}
				}
			}
		}
		pair<int,flt> ans = MP(INF,INF);
		For(i,1,n) For(j,1,n) ans = min(ans,f[All][i][j]);
		printf("%d %.12lf\n",ans.FI,ans.SE);
	}
	return 0;
}
