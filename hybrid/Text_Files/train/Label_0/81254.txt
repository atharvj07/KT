from collections import deque, defaultdict
import time
from math import sqrt, ceil, factorial, floor, inf, log2, sqrt, gcd
import bisect

import heapq

def get_array():
	return list(map(int, sys.stdin.readline().strip().split()))

def get_ints():
	return map(int, sys.stdin.readline().strip().split())

def input():
	return sys.stdin.readline().strip()

def tot(a, n, d):
	return n * (2 * a + (n - 1) * d) // 2
for _ in range(int(input())):
	(n, m) = get_ints()
	s = input()
	x = [0 for i in range(len(s))]
	y = [0 for i in range(len(s))]
	for i in range(len(s)):
		if s[i] == 'R':
			x[i] += 1
		if s[i] == 'L':
			x[i] -= 1
		if s[i] == 'U':
			y[i] -= 1
		if s[i] == 'D':
			y[i] += 1
		if i != 0:
			x[i] += x[i - 1]
			y[i] += y[i - 1]
	x_max = [x[0]]
	x_min = [x[0]]
	y_max = [y[0]]
	y_min = [y[0]]
	ans_x = -inf
	ans_y = -inf
	for i in range(1, len(s)):
		x_max.append(max(x_max[i - 1], x[i]))
		x_min.append(min(x_min[i - 1], x[i]))
		y_max.append(max(y_max[i - 1], y[i]))
		y_min.append(min(y_min[i - 1], y[i]))
	for i in range(len(s)):
		x_cor = -inf
		y_cor = -inf
		if x_max[i] >= 0 and x_min[i] >= 0:
			x_cor = m - x_max[i]
		if x_max[i] <= 0 and x_min[i] <= 0:
			if -x_min[i] + 1 <= m:
				x_cor = -x_min[i] + 1
		if x_max[i] >= 0 and x_min[i] <= 0:
			if m - x_max[i] >= -x_min[i] + 1:
				x_cor = m - x_max[i]
		if x_cor != -inf:
			if y_max[i] >= 0 and y_min[i] >= 0:
				y_cor = n - y_max[i]
			if y_max[i] <= 0 and y_min[i] <= 0:
				if -y_min[i] + 1 <= n:
					y_cor = -y_min[i] + 1
			if y_max[i] >= 0 and y_min[i] <= 0:
				if n - y_max[i] >= -y_min[i] + 1:
					y_cor = n - y_max[i]
		if x_cor != -inf and y_cor != -inf and (x_cor >= 1) and (x_cor <= m) and (y_cor >= 1) and (y_cor <= n):
			ans_x = x_cor
			ans_y = y_cor
		else:
			break
	if ans_x != -inf and ans_y != -inf:
		print(ans_y, ans_x)
	else:
		print(1, 1)
