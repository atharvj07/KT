#include <bits/stdc++.h>
using namespace std;
int main() {
  int a[26], n, k;
  char b[110];
  memset(a, 0, sizeof(a));
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> b[i];
    a[b[i] - 'a']++;
  }
  for (int i = 0; i < 26; i++) {
    if (a[i] > k) {
      cout << "No" << endl;
      return 0;
    }
  }
  cout << "Yes" << endl;
  return 0;
}
