import sys, math
(a00, a01, a10, a11) = map(int, input().split())
s = a00 + a01 + a10 + a11
if not s:
	print('1')
	sys.exit()
s = a01 + a10
if a00:
	z = a00 << 1
	x = math.ceil(math.sqrt(z))
	if (x - 1) * x != z:
		print('Impossible')
		sys.exit()
else:
	x = s and 1
if a11:
	z = a11 << 1
	y = math.ceil(math.sqrt(z))
	if (y - 1) * y != z:
		print('Impossible')
		sys.exit()
else:
	y = s and 1
if x * y != s:
	print('Impossible')
	sys.exit()
if s:
	(i, j) = divmod(a10, x)
	print(i * '1', (x - j) * '0', sep='', end='')
	if y > i:
		print('1', j * '0', (y - i - 1) * '1', sep='')
	else:
		print(j * '0')
else:
	print(x * '0', y * '1', sep='')
