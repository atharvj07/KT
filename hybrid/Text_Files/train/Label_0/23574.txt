#include <bits/stdc++.h>
using namespace std;
int arr[100005];
struct state {
  int a, b, c;
} op1[8], op2[8][2];
state tran(int l, state now) { return state{l + now.a, l + now.b, l + now.c}; }
int get(int l, int len = 3) {
  int ret = 0;
  for (int i = 0; i < len; i++, ret <<= 1) ret |= arr[l + i];
  return (ret >> 1);
}
const int limit = 17;
bool vis[(1 << limit)];
int pre[(1 << limit)];
state pre_op[(1 << limit)];
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  op1[(0 << 2) | (0 << 1) | 1] = state{2, 3, 4};
  op1[(0 << 2) | (1 << 1) | 0] = state{1, 3, 5};
  op1[(0 << 2) | (1 << 1) | 1] = state{1, 2, 3};
  op1[(1 << 2) | (0 << 1) | 0] = state{0, 3, 6};
  op1[(1 << 2) | (0 << 1) | 1] = state{0, 2, 4};
  op1[(1 << 2) | (1 << 1) | 1] = state{0, 1, 2};
  op2[(0 << 2) | (0 << 1) | 0][0] = state{0, 4, 8},
                            op2[(0 << 2) | (0 << 1) | 0][1] = state{1, 4, 7};
  op2[(0 << 2) | (0 << 1) | 1][0] = state{0, 3, 6},
                            op2[(0 << 2) | (0 << 1) | 1][1] = state{1, 3, 5};
  op2[(0 << 2) | (1 << 1) | 0][0] = state{0, 1, 2},
                            op2[(0 << 2) | (1 << 1) | 0][1] = state{2, 4, 6};
  op2[(0 << 2) | (1 << 1) | 1][0] = state{0, 4, 8},
                            op2[(0 << 2) | (1 << 1) | 1][1] = state{1, 5, 9};
  op2[(1 << 2) | (0 << 1) | 0][0] = state{0, 5, 10},
                            op2[(1 << 2) | (0 << 1) | 0][1] = state{1, 3, 5};
  op2[(1 << 2) | (0 << 1) | 1][0] = state{0, 3, 6},
                            op2[(1 << 2) | (0 << 1) | 1][1] = state{1, 5, 9};
  op2[(1 << 2) | (1 << 1) | 0][0] = state{0, 3, 6},
                            op2[(1 << 2) | (1 << 1) | 0][1] = state{1, 4, 7};
  op2[(1 << 2) | (1 << 1) | 1][0] = state{0, 4, 8},
                            op2[(1 << 2) | (1 << 1) | 1][1] = state{1, 3, 5};
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> arr[i];
  vector<state> ans;
  int st = 1;
  for (; st + limit - 1 <= n; st += 3) {
    int now = get(st);
    if (now != 0) {
      if (now != 6) {
        state s = tran(st, op1[now]);
        ans.push_back(s);
        arr[s.a] ^= 1;
        arr[s.b] ^= 1;
        arr[s.c] ^= 1;
      } else {
        now = get(st + 3);
        for (int j = 0; j < 2; j++) {
          state s = tran(st, op2[now][j]);
          ans.push_back(s);
          arr[s.a] ^= 1;
          arr[s.b] ^= 1;
          arr[s.c] ^= 1;
        }
        st += 3;
      }
    }
  }
  int init = get(st, n - st + 1);
  queue<int> q;
  q.push(init);
  vis[init] = true;
  while (!q.empty()) {
    int now = q.front();
    q.pop();
    for (int i = 0; i < limit; i++)
      for (int j = 1; j < limit / 2; j++)
        if (st + i + j + j <= n) {
          int tem = ((1 << (j + j)) | (1 << j) | 1) << (n - (st + i + j + j));
          int nxt = now ^ tem;
          if (vis[nxt]) continue;
          vis[nxt] = true;
          pre[nxt] = now;
          pre_op[nxt] = state{st + i, st + i + j, st + i + j + j};
          q.push(nxt);
        }
  }
  if (!vis[0])
    cout << "NO" << '\n';
  else {
    int tem = 0;
    while (tem != init) {
      ans.push_back(pre_op[tem]);
      tem = pre[tem];
    }
    cout << "YES" << '\n';
    cout << (int)ans.size() << '\n';
    for (state now : ans) cout << now.a << " " << now.b << " " << now.c << '\n';
  }
  return 0;
}
