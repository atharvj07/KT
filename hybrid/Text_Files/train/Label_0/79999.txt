t = int(input())
for w in range(t):
	(n, k) = map(int, input().split())
	l = list(map(int, input().split()))
	s = list(set(l))
	if len(s) > k:
		print(-1)
	else:
		u = []
		for i in range(len(l)):
			for j in range(len(s)):
				u.append(str(s[j]))
			for j in range(k - len(s)):
				u.append(str(1))
		lg = len(u)
		f = ' '.join(u)
		print(lg)
		print(f)
