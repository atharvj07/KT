#include<stdio.h>
#include<map>
#include<queue>
#include<algorithm>
#include<string>
using namespace std;
#define maxn 10
struct my
{
    int tu[10];
    int space;
    int  ans;
    bool operator <(const my &p)const{
    for (int i=0;i<9;i++)
        {
            if(tu[i]==p.tu[i]) continue;
        return tu[i]>p.tu[i];
        }
        return false ;
    }
};
bool is(my s)
{
    for (int i=0;i<9;i++)
        if(s.tu[i]!=(i+1)) return false;
    return true;
}
int dx[]={-1,0,1,0};
int dy[]={0,-1,0,1};
//char dir[]={'u','l','d','r'};
int bfs(my s)
{
    queue<my>Q;
    map<my,bool>V;
    s.ans=0;
    Q.push(s);
    V[s]=true;
    my u,v;

    while(!Q.empty())
    {
        u=Q.front();
        Q.pop();
        if(is(u)) return u.ans;
        int sx=u.space/3;
        int sy=u.space%3;
        for (int i=0;i<4;i++)
        {
            int tx=sx+dx[i];
            int ty=sy+dy[i];
            if(tx<0||ty<0||tx>=3||ty>=3) continue;
            v=u;
            swap(v.tu[u.space],v.tu[tx*3+ty]);
            v.space=tx*3+ty;
            if(!V[v])
            {
                V[v]=true;
                v.ans++;
                Q.push(v);
            }
        }
    }
}
int main()
{
    my in;
    for (int i=0;i<9;i++)
    {
        scanf("%d",&in.tu[i]);
        if(in.tu[i]==0){
            in.space=i;
            in.tu[i]=9;
        }
    }
    int sum=bfs(in);
    printf("%d\n",sum);
    return 0;
}
