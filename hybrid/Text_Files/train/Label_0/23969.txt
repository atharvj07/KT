#include <bits/stdc++.h>
using namespace std;
string d[10][550];
int k;
string mokh(string a) {
  for (int i = 0; i < a.size(); ++i) {
    if (a[i] == '*') {
      a[i] = '+';
      continue;
    }
    if (a[i] == '+') a[i] = '*';
  }
  return a;
}
int power(int a) {
  int ans = 1;
  while (a) {
    ans *= 2;
    a--;
  }
  return ans;
}
int main() {
  d[0][0] = "*";
  cin >> k;
  string help, help1;
  for (int i = 1; i < k + 1; ++i) {
    for (int j = 0; j < power(i - 1); ++j) {
      d[i][j] = d[i - 1][j];
      help = d[i][j] + d[i - 1][j];
      d[i][j] = help;
    }
    for (int j = power(i - 1); j < power(i); ++j) {
      d[i][j] = d[i - 1][j - power(i - 1)];
      help1 = mokh(d[i - 1][j - power(i - 1)]);
      help = d[i][j] + help1;
      d[i][j] = help;
    }
  }
  for (int i = 0; i < power(k); ++i) cout << d[k][i] << endl;
}
