from sys import stdin
input = stdin.buffer.readline
mod = 998244353
n = int(input())
arr = [int(x) for x in input().split()]
d = {}
for i in range(n):
	if arr[i] in d:
		d[arr[i]].append(i)
	else:
		d[arr[i]] = [i]
i = 0
des = d[arr[0]][-1]
while i <= des:
	des = max(des, d[arr[i]][-1])
	i = i + 1
pos = i
ans = 1
start = pos
while start < n:
	i = start
	des = d[arr[start]][-1]
	while i <= des:
		des = max(des, d[arr[i]][-1])
		i = i + 1
	ans = ans * 2 % mod
	start = i
print(ans)
