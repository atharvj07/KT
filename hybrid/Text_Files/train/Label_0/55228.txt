import java.util.Arrays;
import java.util.Collections;
import java.util.Scanner;

public class CF1241C {
	static Scanner input = new Scanner(System.in);
	static int n, x, a, y, b;
	static long k;
	static Integer[] p = new Integer[200005];

	public static void main(String[] args) {
		int q = input.nextInt();
		for (int ii = 0; ii < q; ++ii) {
			n = input.nextInt();
			for (int i = 0; i < n; i++)
				p[i] = input.nextInt();
			Arrays.sort(p, 0, n, Collections.reverseOrder());
			x = input.nextInt();
			a = input.nextInt();
			y = input.nextInt();
			b = input.nextInt();
			k = input.nextLong();
			if (x < y) {
				int t = x;
				x = y;
				y = t;
				t = a;
				a = b;
				b = t;
			}
			int l = 0;
			int r = n + 1;
			int mid;
			while (l < r - 1) {
				mid = (l + r) / 2;
				if (f(mid) < k)
					l = mid;
				else
					r = mid;
			}
			if (r == n + 1)
				System.out.println("-1");
			else
				System.out.println(r);
		}
	}

	public static long f(int n) {
		int AB = 0, A = 0, B = 0;
		long sum = 0;
		for (int i = 1; i <= n; i++) {
			if (i % a == 0 && i % b == 0)
				AB++;
			else if (i % a == 0)
				A++;
			else if (i % b == 0)
				B++;
		}
		for (int i = 0; i < n; i++) {
			if (AB > 0) {
				sum += p[i] / 100 * (x + y);
				AB--;
			} else if (A > 0) {
				sum += p[i] / 100 * x;
				A--;
			} else if (B > 0) {
				sum += p[i] / 100 * y;
				B--;
			}
		}
		return sum;
	}
}
