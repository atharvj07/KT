m10 = 10000000000
mp = 21
p2 = 2 ** mp

def qmd(log, md):
	ans = 1
	a = 2
	while log > 0:
		if log % 2:
			ans = ans * a % md
		log = log // 2
		a = a * a % md
	return ans
n = int(input())
for nn in range(n):
	a = int(input())
	x = a * m10 // p2 * p2 + p2
	while x % 5 == 0:
		x += p2
	log = 0
	while 2 ** log % 5 != x % 5:
		log += 1
	md = 5
	phi = 4
	for i in range(1, mp):
		md *= 5
		while qmd(log, md) != x % md:
			log += phi
		phi *= 5
	if log < mp:
		log += phi
	print(log)
