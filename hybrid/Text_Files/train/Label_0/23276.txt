#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<int> v(1300000, 0);
  int n, cont = 0;
  unsigned long long int k = 0;
  v[0] = 1;
  v[1] = 1;
  for (int i = 2; i * i <= v.size(); ++i) {
    if (v[i] == 0) {
      for (int j = i * i; j <= v.size(); j += i) v[j] = 1;
    }
  }
  cin >> n;
  k = 0;
  while (cont < n) {
    if (v[k] == 0) {
      cout << k << " ";
      ++cont;
    }
    ++k;
  }
  return 0;
}
