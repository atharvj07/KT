for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	st = []
	mx = 0
	for i in range(n):
		if a[i] == -1:
			if i + 1 < n and a[i + 1] != -1:
				st.append(a[i + 1])
			if i - 1 >= 0 and a[i - 1] != -1:
				st.append(a[i - 1])
		if i > 0 and a[i] != -1 and (a[i - 1] != -1) and (abs(a[i] - a[i - 1]) > mx):
			mx = abs(a[i] - a[i - 1])
	st.sort()
	if len(st):
		mx1 = (st[0] + st[-1]) // 2
	else:
		mx1 = 0
	if len(st):
		mx = max(mx, mx1 - st[0], st[-1] - mx1)
	print(mx, mx1)
