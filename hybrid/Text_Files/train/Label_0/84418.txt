from collections import Counter
for _ in range(int(input())):
	n = int(input())
	arr = list(map(float, input().split()))
	if n == 1:
		print(0)
	else:
		ans = float('inf')
		for i in range(n):
			ht = {}
			for j in range(i + 1, n):
				d = (arr[i] - arr[j]) / (i - j)
				if d in ht:
					ht[d] += 1
				else:
					ht[d] = 1
			if ht:
				ans = min(ans, n - 1 - max(ht.values()))
		print(ans)
