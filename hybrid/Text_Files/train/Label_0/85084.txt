import sys
dic = {}
n = int(input())
for i in range(n):
	(a, b) = map(int, sys.stdin.readline().split())
	dic[a] = b
inv_dic = {v: k for (k, v) in dic.items()}
temp = dic.keys() - inv_dic.keys()
a = list(temp)[0]
b = 0
sys.stdout.write(str(a))
c = 0
for i in range(n - 1):
	if i % 2 == 0:
		c = dic[b]
		b = c
	else:
		c = dic[a]
		a = c
	sys.stdout.write(' ' + str(c))
