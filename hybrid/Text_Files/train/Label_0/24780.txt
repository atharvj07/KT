#include <bits/stdc++.h>
using namespace std;
template <class T>
T cdiv(T num, T den) {
  return (num / den) + (num % den > 0);
}
template <typename TH>
void _dbg(const char* sdbg, TH h) {
  cerr << sdbg << "=" << h << "\n";
}
template <typename TH, typename... TA>
void _dbg(const char* sdbg, TH h, TA... t) {
  while (*sdbg != ',') cerr << *sdbg++;
  cerr << "=" << h << ",";
  _dbg(sdbg + 1, t...);
}
template <class t>
t mag(t a) {
  return a > 0 ? a : -a;
}
std::ifstream term("/dev/tty");
string tempstring;
int tc = 1;
long long k, b, n, t;
long long fun(long long val) { return val + b * (val - 1) / (k - 1); }
int escape() {
  cin >> k >> b >> n >> t;
  if (t == 1) {
    cout << n << '\n';
    return 0;
  }
  if (k == 1) {
    long long ans = n - (t - 1) / b;
    cout << max(0ll, ans);
    return 0;
  }
  long long initpower = 0, initval = 1;
  while (fun(initval * k) <= t) {
    initpower++;
    initval *= k;
  }
  cout << max(0ll, n - initpower);
  return 0;
}
int main() {
  ios_base::sync_with_stdio(false);
  while (tc--) escape();
}
