#include <bits/stdc++.h>
using namespace std;
inline bool EQ(double a, double b) { return fabs(a - b) < 1e-9; }
long long a[100007], ans, k;
int n;
void cal(long long d) {
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    sum += d * ((a[i] + d - 1) / d);
  }
  if (sum <= k) ans = max(ans, d);
}
int main() {
  int i;
  long long j;
  scanf("%d %lld", &n, &k);
  for (i = 0; i < n; i++) {
    scanf("%lld", &a[i]);
    k += a[i];
  }
  for (j = 1; j * j <= k; j++) {
    cal(j);
    cal(k / j);
  }
  printf("%lld\n", ans);
  return 0;
}
