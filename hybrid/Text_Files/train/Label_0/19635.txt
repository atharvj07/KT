#include <cstdio>
#include <cstring>
#include <algorithm>
#pragma warning(disable : 4996)
using namespace std;
int dir[] = { 0, 1, 0, -1 };
int W, H, N, xs, ys, xa[1009], xb[1009], ya[1009], yb[1009], xc[2009], yc[2009], d[1009][1009];
void rec(int x, int y) {
	d[x][y] = 1;
	for (int i = 0; i < 4; i++) {
		int tx = x + dir[i], ty = y + dir[i ^ 1];
		if (0 <= tx && tx < xs - 1 && 0 <= ty && ty < ys - 1 && d[tx][ty] == 0) rec(tx, ty);
	}
}
int main() {
	while (scanf("%d%d", &W, &H), W) {
		scanf("%d", &N);
		for (int i = 0; i < N; i++) {
			scanf("%d%d%d%d", &xa[i], &ya[i], &xb[i], &yb[i]);
			xc[i] = xa[i], yc[i] = ya[i];
			xc[i + N] = xb[i], yc[i + N] = yb[i];
		}
		xc[2 * N] = 0, xc[2 * N + 1] = W; sort(xc, xc + 2 * N + 2);
		xs = unique(xc, xc + 2 * N + 2) - xc;
		yc[2 * N] = 0, yc[2 * N + 1] = H; sort(yc, yc + 2 * N + 2);
		ys = unique(yc, yc + 2 * N + 2) - yc;
		memset(d, 0, sizeof(d));
		for (int i = 0; i < N; i++) {
			xa[i] = lower_bound(xc, xc + xs, xa[i]) - xc;
			ya[i] = lower_bound(yc, yc + ys, ya[i]) - yc;
			xb[i] = lower_bound(xc, xc + xs, xb[i]) - xc;
			yb[i] = lower_bound(yc, yc + ys, yb[i]) - yc;
			d[xa[i]][ya[i]]++; d[xb[i]][yb[i]]++;
			d[xa[i]][yb[i]]--; d[xb[i]][ya[i]]--;
		}
		for (int i = 0; i < xs; i++) {
			for (int j = 0; j < ys; j++) {
				d[i][j + 1] += d[i][j];
			}
		}
		for (int i = 0; i < xs; i++) {
			for (int j = 0; j < ys; j++) {
				d[i + 1][j] += d[i][j];
			}
		}
		int ret = 0;
		for (int i = 0; i < xs - 1; i++) {
			for (int j = 0; j < ys - 1; j++) {
				if (d[i][j] == 0) rec(i, j), ret++;
			}
		}
		printf("%d\n", ret);
	}
	return 0;
}
