for _ in range(int(input())):
	s = input()
	(x, y) = (0, 0)
	b = {}
	k = 0
	for i in s:
		(x1, y1) = (x, y)
		if i == 'N':
			y = y + 1
		elif i == 'S':
			y = y - 1
		elif i == 'E':
			x = x + 1
		else:
			x = x - 1
		if (x, y, x1, y1) in b or (x1, y1, x, y) in b:
			k = k + 1
		else:
			k = k + 5
			b[x1, y1, x, y] = 1
	print(k)
