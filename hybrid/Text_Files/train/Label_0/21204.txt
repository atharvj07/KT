#include<bits/stdc++.h>
#define Rint register int
using namespace std;
const int N = 200003, M = N << 2;
template<typename T>
inline void read(T &x){
    int ch = getchar(); x = 0;
    for(;ch < '0' || ch > '9';ch = getchar());
    for(;ch >= '0' && ch <= '9';ch = getchar()) x = x * 10 + ch - '0';
}
int n, S, T, p[N], q[N], head[N], to[M], nxt[M], cap[M], cp[N], cq[N], cc, ans;
inline void add(int a, int b, int c){
	static int cnt = 1;
	to[++ cnt] = b; nxt[cnt] = head[a]; head[a] = cnt; cap[cnt] = c;
	to[++ cnt] = a; nxt[cnt] = head[b]; head[b] = cnt; cap[cnt] = 0;
}
int dep[N], cur[N], qu[N], f, r;
inline bool bfs(){
	memset(dep, -1, sizeof dep);
	memcpy(cur, head, sizeof cur);
	f = r = 0; qu[r ++] = S; dep[S] = 0;
	while(f < r){
		int x = qu[f ++];
		for(Rint i = head[x];i;i = nxt[i])
			if(cap[i] && dep[to[i]] == -1){
				dep[to[i]] = dep[x] + 1;
				qu[r ++] = to[i];
			}
	}
	return ~dep[T];
}
inline int dfs(int x, int lim){
	if(x == T || !lim) return lim;
	int flow = 0, f;
	for(Rint i = cur[x];i;i = nxt[i]){
		cur[x] = i;
		if(dep[to[i]] == dep[x] + 1 && (f = dfs(to[i], min(lim, cap[i])))){
			flow += f; lim -= f; cap[i] -= f; cap[i ^ 1] += f;
			if(!lim) break;
		}
	}
	return flow;
}
int main(){
	read(n); ans = n;
	for(Rint i = 1;i <= n;++ i) read(p[i]), ++ p[i];
	for(Rint i = 1;i <= n;++ i) read(q[i]), ++ q[i];
	for(Rint i = 1;i <= n;++ i) if(!cp[i]){
		cp[i] = ++ cc;
		for(Rint x = p[i];x != i;x = p[x]) cp[x] = cc;
	}
	for(Rint i = 1;i <= n;++ i) if(!cq[i]){
		cq[i] = ++ cc;
		for(Rint x = q[i];x != i;x = q[x]) cq[x] = cc;
	}
	S = ++ cc; T = ++ cc;
	for(Rint i = 1;i <= n;++ i)
		if(p[i] == i && q[i] == i) -- ans;
		else if(p[i] == i) add(cq[i], T, 1);
		else if(q[i] == i) add(S, cp[i], 1);
		else {
			add(cq[i], cp[i], 1);
			if(p[i] == q[i]) add(cp[i], cq[i], 1);
		}
	while(bfs()) ans -= dfs(S, 1e9);
	printf("%d\n", ans);
}
