#include <bits/stdc++.h>
using namespace std;
const int mxn = 2e5 + 1;
vector<int> arr[mxn];
int main() {
  int n;
  cin >> n;
  int ind = 0;
  vector<int> A(n);
  for (int i = 0; i < n; i++) cin >> A[i];
  for (int i = 0; i < n; i++) {
    if (ind == 0 || A[i] < arr[ind - 1].back()) {
      arr[ind++].push_back(A[i]);
    } else {
      int ans = -1, low = 0, high = ind;
      while (low <= high) {
        int md = low + (high - low) / 2;
        if (A[i] > arr[md].back()) {
          ans = md;
          high = md - 1;
        } else {
          low = md + 1;
        }
      }
      arr[ans].push_back(A[i]);
    }
  }
  for (int i = 0; i < ind; i++) {
    for (int it : arr[i]) cout << it << ' ';
    cout << endl;
  }
  return 0;
}
