#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  long long int n, x1, y1, x2, y2;
  cin >> n >> x1 >> y1 >> x2 >> y2;
  long long int r1, r2;
  pair<long long int, pair<int, int> > arr[n + 1];
  for (int i = 0; i < n; i++) {
    cin >> arr[i].second.first >> arr[i].second.second;
    arr[i].first = (arr[i].second.first - x1) * (arr[i].second.first - x1) +
                   (y1 - arr[i].second.second) * (y1 - arr[i].second.second);
  }
  sort(arr, arr + n);
  long long int an = 999999999999999;
  for (int i = 0; i < n; i++) {
    r2 = 0;
    r1 = arr[i].first;
    for (int j = i + 1; j < n; j++) {
      r2 = max(r2,
               (arr[j].second.first - x2) * (arr[j].second.first - x2) +
                   (y2 - arr[j].second.second) * (y2 - arr[j].second.second));
    }
    an = min(r2 + r1, an);
  }
  for (int i = 0; i < n; i++) {
    arr[i].first = (arr[i].second.first - x2) * (arr[i].second.first - x2) +
                   (y2 - arr[i].second.second) * (y2 - arr[i].second.second);
  }
  sort(arr, arr + n);
  for (int i = 0; i < n; i++) {
    r1 = 0;
    r2 = arr[i].first;
    for (int j = i + 1; j < n; j++) {
      r1 = max(r2,
               (arr[j].second.first - x2) * (arr[j].second.first - x2) +
                   (y2 - arr[j].second.second) * (y2 - arr[j].second.second));
    }
    an = min(r2 + r1, an);
  }
  cout << an;
  return 0;
}
