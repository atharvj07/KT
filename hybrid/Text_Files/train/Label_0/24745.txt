#include <bits/stdc++.h>
using namespace std;
int vis[97];
int judge(char a) {
  if (a == 'R' && vis['L'] == 1) return 1;
  if (a == 'L' && vis['R'] == 1) return 1;
  if (a == 'D' && vis['U'] == 1) return 1;
  if (a == 'U' && vis['D'] == 1) return 1;
  return 0;
}
int main() {
  long n;
  string s;
  int sum = 0;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    if (judge(s[i])) {
      sum++;
      memset(vis, 0, sizeof(vis));
    }
    vis[s[i]] = 1;
  }
  cout << sum + 1 << endl;
}
