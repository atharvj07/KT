from math import *
import os, sys
from io import BytesIO
input = BytesIO(os.read(0, os.fstat(0).st_size)).readline
sys.setrecursionlimit(10 ** 9)

def koll(ca, cb, t, ga, gb, a, b):
	t -= ca * a + cb * b
	cx = min(ga - ca, t // a)
	t -= cx * a
	cx += min(gb - cb, t // b)
	return max(cx, 0)
for i in range(min(10 ** 18, int(input()))):
	(n, ct, a, b) = map(int, input().split())
	m = list(map(int, input().split()))
	t = list(map(int, input().split()))
	cur = {}
	s = set()
	ga = 0
	gb = 0
	for i in range(n):
		if m[i] == 0:
			ga += 1
		else:
			gb += 1
		if t[i] not in cur:
			cur[t[i]] = [0, 0]
		cur[t[i]][m[i]] += 1
		s.add(t[i])
	s = sorted(s)
	end = [0] * (len(s) + 1)
	lol = set()
	ca = 0
	cb = 0
	for i in range(len(s) - 1):
		ca += cur[s[i]][0]
		cb += cur[s[i]][1]
		if ca * a + cb * b < s[i + 1]:
			lol.add(s[i])
		end[i + 1] = ca * a + cb * b
	i = len(s) - 1
	ca += cur[s[i]][0]
	cb += cur[s[i]][1]
	end[i + 1] = ca * a + cb * b
	lol.add(s[-1])
	s.append(ct + 1)
	ca = 0
	cb = 0
	mx = koll(0, 0, min(ct, max(s[0] - 1, 0)), ga, gb, a, b)
	for i in range(len(s) - 1):
		ca += cur[s[i]][0]
		cb += cur[s[i]][1]
		if s[i] in lol and end[i + 1] <= ct:
			mx = max(mx, ca + cb + koll(ca, cb, s[i + 1] - 1, ga, gb, a, b))
	print(mx)
