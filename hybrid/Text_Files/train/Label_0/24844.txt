using namespace std;
typedef long long ll;
const int N=505;
char buf[1<<23],*p1=buf,*p2=buf;
int n,m,ans,bk[N][N];
char s[N][N];
int read()
{
	int res=0,fl=0; char a=getchar();
	while(a<'0'||a>'9') fl|=a=='-',a=getchar();
	while(a>='0'&&a<='9') res=res*10+a-'0',a=getchar();
	return fl? -res:res; 
}
int main()
{
//	freopen("1.in","r",stdin);
//	freopen("1.out","w",stdout);
	int i,j,t=read();
	while(t--)
	{
		n=read(),m=read();
		for(i=1;i<=n;i++) scanf("%s",s[i]+1);
		for(i=1;i<=n;i++)
			for(j=1;j<=m;j++) bk[i][j]=(i%3!=1+(n%3==0))*(s[i][j]=='.');
		for(i=1+(n%3==0);i+3<=n;i+=3)
		{
			for(j=1;j<=m;j++)
				if(!bk[i+1][j]||!bk[i+2][j]) {bk[i+1][j]=bk[i+2][j]=0;break;}
			if(j>m) bk[i+1][1]=bk[i+2][1]=0;
		}
		for(i=1;i<=n;i++,cout<<'\n')
			for(j=1;j<=m;j++) cout<<(bk[i][j]? '.':'X');
	}
	return 0;
}
