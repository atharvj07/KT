from sys import stdin, stdout
input = stdin.readline
t = int(input())
for _ in range(t):
	n = int(input())
	arr = [int(x) for x in input().split()]
	vals = [0 for x in range(0, n + 1)]
	for i in arr:
		vals[i] += 1
	ans = 0
	for i in range(1, n + 1):
		a = vals[i]
		p = 0
		if i + 1 <= n:
			p += vals[i + 1]
		if i + 2 <= n:
			p += vals[i + 2]
		ans += a * (p * (p - 1) // 2) + a * (a - 1) // 2 * p + a * (a - 1) * (a - 2) // 6
	print(ans)
