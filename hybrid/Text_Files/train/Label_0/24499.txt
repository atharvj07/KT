void solve() {
  string s;
  cin >> s;
  long long n = ((long long)s.size());
  stack<char> st;
  long long i = 0;
  while (i < n) {
    if (st.empty()) {
      st.push(s[i]);
      i++;
    } else {
      if (s[i] == 'B') {
        st.pop();
        i++;
      } else {
        st.push(s[i]);
        i++;
      }
    }
  }
  cout << ((long long)st.size()) << '\n';
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout << setprecision(6);
  long long t = 1;
  cin >> t;
  while (t--) solve();
  return 0;
}
