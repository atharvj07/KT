import sys
input = sys.stdin.readline

def solve():
	(n, m) = map(int, input().split())
	a = [None] * (n * m)
	b = [None] * n
	w = [None] * n
	for i in range(n):
		j = 0
		c = [None] * m
		w[i] = [None] * m
		for v in map(int, input().split()):
			a[i * m + j] = (v, i, j)
			c[j] = v
			j += 1
		b[i] = c
	a.sort()
	k = 0
	for (v, i, j) in a[:m]:
		w[i][k] = j
		k += 1
	for i in range(n):
		h = [False] * m
		for j in w[i]:
			if j is not None:
				h[j] = True
		k = 0
		for j in range(m):
			if w[i][j] is None:
				while k < m and h[k]:
					k += 1
				w[i][j] = k
				k += 1
		print(' '.join((str(b[i][j]) for j in w[i])))
for i in range(int(input())):
	solve()
