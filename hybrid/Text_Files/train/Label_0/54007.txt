import java.util.Scanner;


public class A {
    
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int array[]= new int[n];
        for(int i=0;i<n;i++) array[i]=in.nextInt();
        int max = 0;
        for(int i=0;i<n;i++){           
            for(int j=i;j<n;j++){
                int current = 0;
                for(int k=i;k<=j;k++){
                    current^=array[k];
                }
                max = Math.max(current, max);
            }           
        }
        System.out.println(max);
    }
}
