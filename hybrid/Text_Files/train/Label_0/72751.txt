x = int(input())
y = []
z = []
for c in range(x):
	y.append(list(map(int, input().split())))
	z.append(list(map(int, input().split())))
for c in range(x):
	pat = y[c][0] // 2
	beef = y[c][1]
	chick = y[c][2]
	pricebeef = z[c][0]
	pricechick = z[c][1]
	win = 0
	if pricechick >= pricebeef:
		while chick > 0 and pat > 0:
			win = win + pricechick
			chick = chick - 1
			pat = pat - 1
		while beef > 0 and pat > 0:
			win = win + pricebeef
			beef = beef - 1
			pat = pat - 1
	if pricechick < pricebeef:
		while beef > 0 and pat > 0:
			win = win + pricebeef
			beef = beef - 1
			pat = pat - 1
		while chick > 0 and pat > 0:
			win = win + pricechick
			chick = chick - 1
			pat = pat - 1
	print(win)
