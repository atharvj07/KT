from collections import defaultdict, deque
from heapq import heappush, heappop
from math import inf
ri = lambda : map(int, input().split())

def solve():
	A = []
	for _ in range(10):
		A.append(list(input()))
	valid = False
	for r in range(10):
		for c in range(10):
			if A[r][c] != '.':
				continue
			left = 0
			pos = c - 1
			while pos >= 0 and A[r][pos] == 'X':
				left += 1
				pos -= 1
			pos = c + 1
			right = 0
			while pos < 10 and A[r][pos] == 'X':
				right += 1
				pos += 1
			if left + right + 1 >= 5:
				valid = True
				break
			diag1 = 0
			pr = r - 1
			cr = c + 1
			while pr >= 0 and cr < 10 and (A[pr][cr] == 'X'):
				diag1 += 1
				pr -= 1
				cr += 1
			diag2 = 0
			pr = r + 1
			cr = c - 1
			while pr < 10 and cr >= 0 and (A[pr][cr] == 'X'):
				diag2 += 1
				pr += 1
				cr -= 1
			if diag1 + diag2 + 1 >= 5:
				valid = True
			diag1 = 0
			pr = r - 1
			cr = c - 1
			while pr >= 0 and cr >= 0 and (A[pr][cr] == 'X'):
				diag1 += 1
				pr -= 1
				cr -= 1
			diag2 = 0
			pr = r + 1
			cr = c + 1
			while pr < 10 and cr < 10 and (A[pr][cr] == 'X'):
				diag2 += 1
				pr += 1
				cr += 1
			if diag1 + diag2 + 1 >= 5:
				valid = True
			up = 0
			pos = r - 1
			while pos >= 0 and A[pos][c] == 'X':
				up += 1
				pos -= 1
			down = 0
			pos = r + 1
			while pos < 10 and A[pos][c] == 'X':
				down += 1
				pos += 1
			if up + down + 1 >= 5:
				valid = True
		if valid:
			break
	if valid:
		print('YES')
	else:
		print('NO')
t = 1
while t:
	t -= 1
	solve()
