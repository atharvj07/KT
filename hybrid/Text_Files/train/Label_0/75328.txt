n = int(input())
q = []
for i in range(0, n):
	s = 0
	(a, b, c) = map(int, input().split())
	if b < a:
		(a, b) = (b, a)
	if b < c:
		(c, b) = (b, c)
	if a > 0:
		s += 1
		a -= 1
	if b > 0:
		s += 1
		b -= 1
	if c > 0:
		s += 1
		c -= 1
	if b > 0 and a > 0:
		s += 1
		b -= 1
		a -= 1
	if b > 0 and c > 0:
		s += 1
		c -= 1
		b -= 1
	if a > 0 and c > 0:
		s += 1
		a -= 1
		c -= 1
	if a > 0 and c > 0 and (b > 0):
		s += 1
		a -= 1
		b -= 1
		c -= 1
	q.append(s)
for i in range(0, len(q)):
	print(q[i])
