#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
#ifdef APURBA
#include "DEBUG_TEMPLATE.h"
#else
#define HERE
#define debug(args...)
#endif
const int N = 3e5 +5;
const ll mod = 1e9+7;
typedef pair<int,int> pii;

struct Node
{
    int rnk,parent;
    ll sum;
    Node()
    {
        sum = 0;
        rnk=-1;
        parent=-1;
    }
};


Node node[N];
set<pii>adj[N];


void make_set(int x, ll y)
{
    node[x].rnk=0;
    node[x].parent=x;
    node[x].sum = y;
}
int find_set(int x)
{
    if(node[x].parent!=x)
    {
        node[x].parent = find_set(node[x].parent);
    }
    return node[x].parent;
}
inline ll getsum(int x)
{
    return node[find_set(x)].sum;
}

void Merge(set<pii>&s1 , set<pii>&s2 ,int id)
{
    if(s2.size()> s1.size())
    {
        swap(s1,s2);
    }
    for(auto it:s2)
    {
        if(find_set(it.first)==id)
            continue;
        s1.insert(it);
    }
}

void Union(int p,int q, ll x)
{
    p = find_set(p);
    q = find_set(q);
    if(p==q)
        return;
    if( node[p].rnk > node[q].rnk)
    {
        node[q].parent= p;
        node[p].sum += node[q].sum-x;
        Merge(adj[p] , adj[q],p);
    }
    else
    {
        node[p].parent= q;
        node[q].sum += node[p].sum-x;
        Merge(adj[q] , adj[p],q);
        if(node[p].rnk==node[q].rnk)
        {
            node[q].rnk +=1;
        }
    }
}
int n,m;
ll x;
ll a[N];
vector<pii>edges;
vector<int>ans;


int32_t main()
{
#ifndef APURBA
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
#endif
    cin>>n>>m>>x;
    ll sum = 0;
    priority_queue<pair<ll,ll> >pq;
    for(int i=0; i<n; i++)
    {
        cin>>a[i];
        sum += a[i];
        make_set(i,a[i]);
        pq.push({a[i],i});
    }

    for(int i=0; i<m; i++)
    {
        int u,v;
        cin>>u>>v;
        u--;
        v--;
        edges.push_back({u,v});
        adj[u].insert({v,i});
        adj[v].insert({u,i});
    }
    if(sum < (n-1)*1ll*x)
    {
        cout<<"NO\n";
        return 0;
    }
    while(!pq.empty())
    {
        auto it = pq.top();
        pq.pop();
        int s = it.second;
        if(it.first != getsum(s))
        {
            continue;
        }
        if(s != find_set(s))
            continue;
        vector<pii>tmp;
        for(auto i:adj[s])
        {
            if(find_set(i.first)==s){
                tmp.push_back(i);
                continue;
            }
            ll eta = it.first;
            ll ota = getsum(i.first);
            if(eta + ota >=x)
            {
                Union(s,i.first,x);
                ans.push_back(i.second);
                pq.push({getsum(s) , find_set(s)});
                break;
            }
        }
        for(auto i:tmp)
            adj[s].erase(i);
    }
    cout<<"YES\n";
    for(auto i:ans)
    {
        cout<<i+1<<"\n";
    }
    return 0;
}
