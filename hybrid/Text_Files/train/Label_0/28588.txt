#include <bits/stdc++.h>
using namespace std;
namespace FGF {
int n, m, op, x;
const int N = 1e5 + 5;
const int mo = 1e9 + 7;
char s[N];
vector<int> f[N];
bool vis[N];
int fac[N], inv[N], pw[N];
void init() {
  fac[0] = inv[0] = inv[1] = pw[0] = 1;
  for (int i = 1; i <= 1e5; i++) fac[i] = 1LL * fac[i - 1] * i % mo;
  for (int i = 2; i <= 1e5; i++)
    inv[i] = 1LL * inv[mo % i] * (mo - mo / i) % mo;
  for (int i = 2; i <= 1e5; i++) inv[i] = 1LL * inv[i] * inv[i - 1] % mo;
  for (int i = 1; i <= 1e5; i++) pw[i] = 1LL * pw[i - 1] * 25 % mo;
}
long long C(int x, int y) {
  return 1LL * fac[x] * inv[y] % mo * inv[x - y] % mo;
}
void solve() {
  if (!vis[n]) {
    vis[n] = 1;
    for (int i = 0; i < n; i++) f[n].push_back(0);
    for (int i = n; i <= 1e5; i++)
      f[n].push_back((1LL * C(i - 1, n - 1) % mo * pw[i - n] % mo +
                      1LL * f[n][i - 1] * 26 % mo) %
                     mo);
  }
  printf("%d\n", f[n][x]);
}
void work() {
  scanf("%d", &m);
  scanf("%s", s + 1);
  n = strlen(s + 1);
  init();
  while (m--) {
    scanf("%d", &op);
    if (op == 1)
      scanf("%s", s + 1), n = strlen(s + 1);
    else
      scanf("%d", &x), solve();
  }
}
}  // namespace FGF
int main() {
  FGF::work();
  return 0;
}
