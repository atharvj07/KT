#include <bits/stdc++.h>
using namespace std;
int main() {
  long long i, j, k, n, p = 0, neg = 0, pos = 0;
  cin >> n;
  if (n & 1)
    k = (n + 1) / 2;
  else
    k = n / 2;
  for (i = 0; i < n; i++) {
    cin >> j;
    if (j < 0)
      neg++;
    else if (j > 0)
      pos++;
  }
  if (pos > neg)
    i = pos;
  else
    i = -neg;
  j = max(neg, pos);
  if (j >= k)
    cout << i;
  else
    cout << p;
  return 0;
}
