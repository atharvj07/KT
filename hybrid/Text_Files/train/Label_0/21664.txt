const long double EPS = 1e-9;
const long double INF = 1LL<<60;
//const frac EPS = 0;

template<class T> struct Polynomial : vector<T> {
    Polynomial() : vector<T>(1, 0) { }
    Polynomial(T num) : vector<T>(1, num) { }
    Polynomial(int size, T num) : vector<T>(size, num) { }
    Polynomial(vector<T> vec) { *this = vec; }
    
    Polynomial<T>& normalize() {
        while ( (*this).size() > 1 && abs((*this).back()) <= EPS ) (*this).pop_back();
        return (*this);
    }
    T sub(T x) {
        T res = 0;
        for (int i =  (*this).size()-1; i > 0; --i) res = res + (*this)[i], res = res * x;
        res += (*this)[0];
        return res;
    }
    inline Polynomial<T> operator - () { 
        for (int i = 0; i < (*this).size(); ++i) (*this)[i] = (*this)[i] * (-1); 
        return (*this);
    }
    inline const Polynomial<T>& operator += (const Polynomial<T> &x);
    inline const Polynomial<T>& operator -= (const Polynomial<T> &x);
    inline const Polynomial<T>& operator *= (T x);
    inline const Polynomial<T>& operator *= (const Polynomial<T> &x);
    inline const Polynomial<T>& operator /= (T x);
    inline const Polynomial<T>& operator /= (const Polynomial<T> &x);
    inline const Polynomial<T>& operator %= (T x);
    inline const Polynomial<T>& operator %= (const Polynomial<T> &x);
};

template<class T> inline bool operator > (Polynomial<T> x, Polynomial<T> y) { return x.size() > y.size(); }
template<class T> inline bool operator < (Polynomial<T> x, Polynomial<T> y) { return y > x; }
template<class T> inline bool operator <= (Polynomial<T> x, Polynomial<T> y) { return !(y < x); }
template<class T> inline bool operator >= (Polynomial<T> x, Polynomial<T> y) { return !(x < y); }
template<class T> inline bool operator != (Polynomial<T> x, Polynomial<T> y) { return x < y || y < x; }
template<class T> inline bool operator == (Polynomial<T> x, Polynomial<T> y) { return !(x < y) && !(y < x); }

template<class T> ostream &operator << (ostream &os, Polynomial<T> x) {
    for (int i = x.size()-1; i >= 0; --i) { 
        if (i != x.size()-1) os << abs(x[i]); else os << x[i];
        if (i != 0) {
            os << "x"; if (i != 1) os << "^" << i; os << " ";
            if (x[i-1] < 0) os << "- "; else os << "+ ";
        }
    }
    return os;
}

template<class T> inline Polynomial<T> operator + (Polynomial<T> x, Polynomial<T> y) {
    while (x.size() > y.size()) y.push_back(0);
    while (y.size() > x.size()) x.push_back(0);
    Polynomial<T> z((int)x.size(), 0);
    for (int i = 0; i < x.size(); ++i) z[i] = x[i] + y[i];
    return z.normalize();
} 
template<class T> inline Polynomial<T> operator - (Polynomial<T> x, Polynomial<T> y) {
    y = -y;
    return x + y;
}
template<class T> inline Polynomial<T> operator * (Polynomial<T> x, T a) {
    Polynomial<T> z((int)x.size(), 0);
    for (int i = 0; i < x.size(); ++i) z[i] = x[i] * a;
    return z.normalize();
}
template<class T> inline Polynomial<T> operator * (Polynomial<T> x, Polynomial<T> y) {
    Polynomial<T> z((int)x.size() + (int)y.size() - 1, 0);
    for (int i = 0; i < x.size(); ++i) for (int j = 0; j < y.size(); ++j) z[i+j] = z[i+j] + x[i] * y[j];
    return z.normalize();
}
template<class T> pair<Polynomial<T>,T> divmod(Polynomial<T> x, T a) {
    for (int i = x.size()-1; i >= 0; --i) x[i] = x[i] / a;
    x.normalize();
    return pair<Polynomial<T>,T>(x, 0);
}
template<class T> Polynomial<T> operator / (Polynomial<T> x, T a) {
    return divmod(x, a).first;
}
template<class T> T operator % (Polynomial<T> x, T a) {
    return divmod(x, a).second;
}
template<class T> pair<Polynomial<T>, Polynomial<T> > divmod(Polynomial<T> x, Polynomial<T> y) {
    if (x.size() < y.size()) return make_pair(0, x);
    Polynomial<T> q((int)x.size() - (int)y.size() + 1, 0);
    for (int i = x.size()-1; i >= y.size()-1; --i) {
        T div = x[i] / y.back();
        q[i-(int)y.size()+1] = div;
        for (int j = 0; j < y.size(); ++j) x[i+j-(int)y.size()+1] = x[i+j-(int)y.size()+1] - y[j] * div;
    }
    return make_pair(q, x.normalize());
}
template<class T> Polynomial<T> operator / (Polynomial<T> x, Polynomial<T> y) {
    return divmod(x, y).first;
}
template<class T> Polynomial<T> operator % (Polynomial<T> x, Polynomial<T> y) {
    return divmod(x, y).second;
}
template<class T> inline Polynomial<T> pow(Polynomial<T> a, T n) {
    Polynomial<T> res = 1;
    while (n > 0) { if (n & 1) { res = res * a; } a = a * a; n >>= 1; }
    return res;
}

vector<long double> quad_eq(long double a, long double b, long double c) {
    vector<long double> res;
    if (abs(a) < EPS) {
        if (abs(b) < EPS) {
            if (abs(c) < EPS) return res;               // all real numbers
            else return res;                            // no solutions
        }
        else return vector<long double>(1, -c/b);
    }
    long double D = b*b - a*c*4;
    if (D > EPS) {
        res.push_back( (-b+sqrt(D))/(a*2) );
        res.push_back( (-b-sqrt(D))/(a*2) );
        return res;
    }
    else if (abs(D) <= EPS) {
        return vector<long double>(2, -b/(a*2));
        //return vector<long double>(1, -b/(a*2));
    }
    else return res;
}
