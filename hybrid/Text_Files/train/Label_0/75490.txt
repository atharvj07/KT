s = input()
cnt = [0 for i in range(26)]
for el in s:
	cnt[ord(el) - ord('a')] += 1
tps = 0
nos = 0
for i in range(26):
	if cnt[i] > 0:
		tps += 1
	if cnt[i] > 1:
		nos += 1

def f():
	if tps > 4:
		return False
	if tps < 2:
		return False
	if tps == 2:
		if nos < 2:
			return False
		return True
	if tps == 3:
		if len(s) > 3:
			return True
		return False
	return True
if f():
	print('Yes')
else:
	print('No')
