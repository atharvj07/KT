#include <bits/stdc++.h>
using namespace std;
const int MAX = 1e6 + 10;
int vis[MAX + MAX];
int n;
string s;
bool check(int mid) {
  memset(vis, 0, sizeof(vis));
  int now = 0;
  vis[now + n]++;
  for (auto &c : s) {
    int cur = 1;
    if (c == 'L') cur = -1;
    now += cur;
    vis[now + n]++;
    if (now == mid) now -= cur;
  }
  return (vis[now + n] == 1);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cin >> s;
  n = s.size();
  int now = 0;
  vis[now + n]++;
  for (auto &c : s) {
    int cur = 1;
    if (c == 'L') cur = -1;
    now += cur;
    vis[now + n]++;
  }
  if (vis[now + n] == 1) return cout << 1 << "\n", 0;
  int l = 1, r = n;
  int idx = 0, ans = 0;
  while (l <= r) {
    int mid = (l + r) >> 1;
    if (check(mid))
      idx = mid, l = mid + 1;
    else
      r = mid - 1;
  }
  ans += idx;
  l = -n, r = -1;
  idx = 0;
  while (l <= r) {
    int mid = (l + r) >> 1;
    if (check(mid))
      idx = mid, r = mid - 1;
    else
      l = mid + 1;
  }
  ans += abs(idx);
  return cout << ans << "\n", 0;
}
