(_, B) = map(int, input().split())
arr = list(map(int, input().split()))
even = 0
odd = 0
costs = []
for (i, n) in enumerate(arr[:-1]):
	if n % 2 == 0:
		even += 1
	else:
		odd += 1
	if even == odd:
		costs.append(abs(arr[i] - arr[i + 1]))
costs.sort()
price = 0
ans = 0
for c in costs:
	if price + c <= B:
		price += c
		ans += 1
	else:
		break
print(ans)
