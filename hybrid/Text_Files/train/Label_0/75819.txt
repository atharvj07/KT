s = input()
if len(s) % 2 == 1:
	print(-1)
else:
	r = 0
	l = 0
	u = 0
	d = 0
	res1 = 0
	res2 = 0
	for i in range(len(s)):
		if s[i] == 'L':
			l += 1
		elif s[i] == 'R':
			r += 1
		elif s[i] == 'D':
			d += 1
		else:
			u += 1
	res1 = abs(d - u)
	res2 = abs(r - l)
	c = 0
	if res1 % 2 == 0:
		c += res1 // 2
		c += res2 // 2
	else:
		c += res1 // 2
		c += res2 // 2
		c += 1
	print(c)
