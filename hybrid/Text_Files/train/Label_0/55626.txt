	void solve(){
		int M=Integer.highestOneBit(m)<<2;
		double[] xr=new double[M];
		double[] xi=new double[M];
		boolean[] exists=new boolean[m+1];
		for(int i : a){
			xr[i]=1;
			exists[i]=true;
		}
		// debug(xr);

		for(int i=0; i<M; i++){
			double re=xr[i]*xr[i]-xi[i]*xi[i];
			double im=2*xr[i]*xi[i];
			xr[i]=re;
			xi[i]=im;
		}
		fft(xr, xi, -1);
		// debug(xr);

		for(int i=0; i<=m; i++){
			boolean rep=round(xr[i])>0;

				}

			}
		}

	}

	void fft(double[] xr, double[] xi, int sign){
		int n=xr.length, nlz=Integer.numberOfLeadingZeros(n);
		for(int i=0; i<n; i++){
			int j=Integer.reverse(i)>>>nlz+1;
			if(i<j){
				double tr=xr[i];
				xr[i]=xr[j];
				xr[j]=tr;
				double ti=xi[i];
				xi[i]=xi[j];
				xi[j]=ti;
			}
		}
		for(int m=2; m<=n; m*=2){
			double pr=cos(sign*2*PI/m), pi=sin(sign*2*PI/m);
			int hm=m/2;
			for(int j=0; j<n; j+=m){
				double qr=1, qi=0;
				for(int i=j; i<j+hm; i++){
					double tr=qr*xr[i+hm]-qi*xi[i+hm];
					double ti=qr*xi[i+hm]+qi*xr[i+hm];
					xr[i+hm]=xr[i]-tr;
					xi[i+hm]=xi[i]-ti;
					xr[i]+=tr;
					xi[i]+=ti;
					double _tr=qr*pr-qi*pi;
					qi=qr*pi+qi*pr;
					qr=_tr;
				}
			}
		}
		if(sign==-1)
			for(int i=0; i<n; i++){
				xr[i]/=n;
				xi[i]/=n;
			}
	}

	void fftold(double[] xr, double[] xi, int sign){
		int n=xr.length, nlz=Integer.numberOfLeadingZeros(n);
		for(int i=0; i<n; i++){
			int j=Integer.reverse(i)>>>nlz+1;
			if(i<j){
				double tr=xr[i];
				xr[i]=xr[j];
				xr[j]=tr;
				double ti=xi[i];
				xi[i]=xi[j];
				xi[j]=ti;
			}
		}
		for(int m=2; m<=n; m*=2){
			double pr=cos(sign*2*PI/m), pi=sin(sign*2*PI/m);
			double qr=1, qi=0;
			for(int k=0; k<m/2; k++){
				// double qr=cos(sign*2*PI/m*k), qi=sin(sign*2*PI/m*k);
				for(int i=k; i<n; i+=m){
					double tr=qr*xr[i+m/2]-qi*xi[i+m/2];
					double ti=qr*xi[i+m/2]+qi*xr[i+m/2];
					xr[i+m/2]=xr[i]-tr;
					xi[i+m/2]=xi[i]-ti;
					xr[i]+=tr;
					xi[i]+=ti;
				}
				double tr=qr*pr-qi*pi;
				qi=qr*pi+qi*pr;
				qr=tr;
			}
		}
		if(sign==-1)
			for(int i=0; i<n; i++){
				xr[i]/=n;
				xi[i]/=n;
			}
	}

	class Scanner{
		BufferedReader br;
		StringTokenizer st;

		Scanner(InputStream in){
			br=new BufferedReader(new InputStreamReader(in));
			eat("");
		}

		void eat(String s){
			st=new StringTokenizer(s);
		}

		String nextLine(){
			try{
				return br.readLine();
			}catch(IOException e){
				throw new IOError(e);
			}
		}

		boolean hasNext(){
			while(!st.hasMoreTokens()){
				String s=nextLine();
				if(s==null)
					return false;
				eat(s);
			}
			return true;
		}

		String next(){
			hasNext();
			return st.nextToken();
		}

		int nextInt(){
			return Integer.parseInt(next());
		}

		long nextLong(){
			return Long.parseLong(next());
		}

		double nextDouble(){
			return Double.parseDouble(next());
		}
	}
