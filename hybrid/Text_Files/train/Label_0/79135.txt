(b, k) = [int(x) for x in input().split()]
a = list((int(x) for x in input().split()))
last_num = 0
for i in range(k):
	if k - 1 - i == 0:
		if a[i] % 2 == 0:
			last_num += 2
		else:
			last_num += 1
		continue
	if a[i] * b % 2 == 0:
		last_num += 2
	else:
		last_num += 1
if last_num % 2 == 0:
	print('even')
else:
	print('odd')
