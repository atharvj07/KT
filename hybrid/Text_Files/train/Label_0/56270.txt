import java.util.Scanner;

public class D {

    private static Scanner in;

    private static int N, K, first = -1, second = -1;

    private static void solve() {
        N = in.nextInt();
        K = in.nextInt();
        in.nextLine();
        first = find(0, N - 1);
        second = find(0, first - 1);
        if (!ask(second, first))
            second = find(first + 1, N - 1);
        System.out.println(2 + " " + (first + 1) + " " + (second + 1));
    }

    private static int find(int lo, int hi) {
        if (lo < 0 || hi >= N || lo >= N || hi < 0)
            return -1;
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (ask(mid, mid + 1))
                hi = mid;
            else
                lo = mid + 1;
        }
        return lo;
    }

    private static boolean ask(int a, int b) {
        if (a < 0 || a >= N || b < 0 || b >= N || a == b)
            return false;
        System.out.println(1 + " " + (a + 1) + " " + (b + 1));
        System.out.flush();
        return in.nextLine().equals("TAK");
    }

    public static void main(String[] args) {
        in = new Scanner(System.in);
        solve();
        System.exit(0);
    }

}
