n = int(input())
m = list(map(int, input().split()))
mold = list(m)
mres = [0]
for ind in range(n):
	m = list(mold)
	for i in range(ind + 1, n):
		m[i] = min(m[i - 1], m[i])
	for i in range(ind - 1, -1, -1):
		m[i] = min(m[i + 1], m[i])
	if sum(m) >= sum(mres):
		mres = list(m)
print(' '.join(list(map(str, mres))))
