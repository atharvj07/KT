#include <bits/stdc++.h>
using namespace std;
int n;
long long a[77780], ans;
int spg(long long x) {
  if (x <= 3) return 0;
  if (x <= 15) return 1;
  if (x <= 81) return 2;
  if (x <= 6723) return 0;
  if (x <= 50625) return 3;
  if (x <= 2562991875ll)
    return 1;
  else
    return 2;
}
int main() {
  int i;
  std::ios::sync_with_stdio(false);
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> a[i];
    ans ^= spg(a[i]);
  }
  if (ans)
    cout << "Furlo";
  else
    cout << "Rublo";
  return 0;
}
