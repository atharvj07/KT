import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Stack;


public class topsort {

    static LinkedList<Integer>[] arr;
    static Stack<Integer>s;
    static boolean cycle;
    static int tries;
    static String[] names;
    static HashMap<Character,Integer> map;
    static HashMap<Integer,Character> fh;
    static char[] seen;
    
    public static void main(String[]args) throws Throwable{
        
        arr = new LinkedList[26];
         map = new HashMap();
         s = new Stack();
         cycle = false;
        int count = 0;
        fh = new HashMap();
        for(int i=97;i<123;i++){
            char x = (char)i;
            map.put(x,count);
            fh.put(count,x);
            count++;
        }
        fill();
        
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        tries = Integer.parseInt(br.readLine());
        names = new String[tries];
        for(int i=0;i<tries;i++){
            names[i] = br.readLine();
        } 
        boolean invalid = false;
        for(int j=1;j<names.length;j++){
            String prev = names[j-1];
            String curr = names[j];
            int start = 0;
           
            boolean flag = false;
            while(prev.length()>start&&curr.length()>start){
                if(prev.charAt(start)!=curr.charAt(start)){
                    arr[map.get(prev.charAt(start))].add(map.get(curr.charAt(start)));
                    flag = true;
                    break;
                }
                start++;
            }
            if(!flag) if(prev.length()>curr.length()){
                invalid = true;
                break;
            }
        }
        if(invalid){
            System.out.print("Impossible");
            return;
            
        }
        for(int i=0;i<26;i++){
            
            dfs(i);
            if(cycle) break;
        }
        if(cycle) System.out.print("Impossible");
        else {
            String res = "";
            while(!s.isEmpty()){
                int index = s.pop();
                res = res+fh.get(index);
            }
            System.out.print(res);
        }
        }
        
        
        
    
    
    
    public static void dfs(int i){
        if(cycle) return;
        if(seen[i]=='b') return;
        if(seen[i]=='g') {cycle = true;
        return;}
        seen[i] ='g';
        for(Integer x: arr[i]){
            dfs(x);
        }
        s.push(i);
        seen[i] ='b';
    }

    public static void fill(){
        seen = new char[26];
        for(int i=0;i<26;i++){
            arr[i] = new LinkedList();
            seen[i] = 'w';
        }
    }
}
