k = int(input())
seat_map = ['+------------------------+', '|#.#.#.#.#.#.#.#.#.#.#.|D|)', '|#.#.#.#.#.#.#.#.#.#.#.|.|', '|#.......................|', '|#.#.#.#.#.#.#.#.#.#.#.|.|)', '+------------------------+']
seat = list(map(list, seat_map))
for L in range(len(seat[0])):
	for W in range(len(seat)):
		if seat[W][L] == '#' and k > 0:
			seat[W][L] = 'O'
			k -= 1
for line in seat:
	print(''.join(line))
