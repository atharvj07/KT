#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 5, mod = 1e9 + 7;
int n, ans, o;
struct point {
  int x, y;
  point(int x = 0, int y = 0) : x(x), y(y) {}
  point operator-(const point &tmp) const {
    return point(x - tmp.x, y - tmp.y);
  }
  void operator+=(const point &tmp) {
    x = (x + tmp.x % mod) % mod, y = (y + tmp.y % mod) % mod;
    x += mod;
    x %= mod;
    y += mod;
    y %= mod;
  }
  void operator-=(const point &tmp) {
    x = (x - tmp.x % mod) % mod, y = (y - tmp.y % mod) % mod;
    x += mod;
    x %= mod;
    y += mod;
    y %= mod;
  }
  point operator*(const int &k) const {
    return point((1ll * x * k % mod + mod) % mod,
                 (1ll * y * k % mod + mod) % mod);
  }
} a[N], p;
long long cross(point a, point b) {
  return (1ll * a.x * b.y - 1ll * a.y * b.x);
}
int mcross(point a, point b) {
  return (1ll * a.x * (b.y % mod + mod) % mod + mod -
          1ll * a.y * (b.x % mod + mod) % mod) %
         mod;
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i].x >> a[i].y;
  unsigned long long s = 0, m = 0, cnt = 0;
  for (int i = 2; i < n; ++i) s += cross(a[i] - a[0], a[i - 1] - a[0]);
  for (int i = 0, j = 0; i < n; ++i) {
    while (1) {
      int k = (j + 1) % n;
      unsigned long long w = cross(a[k] - a[i], a[j] - a[i]);
      if (2 * (m + w) >= s) {
        if (2 * (m + w) == s) ++cnt;
        break;
      }
      (o += (m += w) % mod) %= mod;
      p += a[k] - a[i];
      j = k;
    }
    (ans -= o * 2 % mod) %= mod;
    int k = (i + 1) % n;
    m -= cross(a[j] - a[i], a[k] - a[i]);
    o = (o + mod - mcross(p, a[k] - a[i])) % mod;
    p -= (a[k] - a[i]) * ((j - i + n) % n);
  }
  (ans += (1ll * n * (n - 3) - cnt) / 2 % mod * (s % mod) % mod) %= mod;
  ans += mod;
  ans %= mod;
  cout << ans << endl;
  return 0;
}
