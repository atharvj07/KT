#include<stdio.h>
#include<vector>
#include<algorithm>
using namespace std;
int c[5][5];
int dx[]={1,0,-1,0};
int dy[]={0,1,0,-1};
struct wolf{
	int m[4][4];
};
int solve(int a,wolf b){
	if(a==0){
		for(int i=0;i<4;i++)for(int j=0;j<4;j++)if(b.m[i][j])return 0;
		return 1;
	}
	
	for(int i=0;i<4;i++)for(int j=0;j<4;j++){
		wolf to=b;
		to.m[i][j]++;
		if(to.m[i][j]>=5){
			to.m[i][j]=0;
			vector<pair<pair<int,int>,int> >p;
			for(int k=0;k<4;k++)p.push_back(make_pair(make_pair(i,j),k));
			while(p.size()){
				vector<pair<pair<int,int>,int> >tmp;
				for(int k=0;k<p.size();k++){
					int nr=p[k].first.first;
					int nc=p[k].first.second;
					int dir=p[k].second;
					nr+=dx[dir];
					nc+=dy[dir];
					if(nr<0||nc<0||nr>=4||nc>=4)continue;
					if(to.m[nr][nc]){to.m[nr][nc]++;continue;}
					tmp.push_back(make_pair(make_pair(nr,nc),dir));
				}
				for(int k=0;k<4;k++)for(int l=0;l<4;l++)
					if(to.m[k][l]>=5){
						to.m[k][l]=0;
						for(int m=0;m<4;m++)tmp.push_back(make_pair(make_pair(k,l),m));
					}
				p=tmp;
			}
		}
		if(solve(a-1,to))return 1;
	}
	return 0;
}
int main(){
	for(int i=0;i<4;i++)for(int j=0;j<4;j++)scanf("%d",&c[i][j]);
	for(int i=0;i<=5;i++){
		wolf st;
		for(int j=0;j<4;j++)for(int k=0;k<4;k++)st.m[j][k]=c[j][k];
		if(solve(i,st)){
			printf("%d\n",i);return 0;
		}
	}
	printf("-1\n");
}
