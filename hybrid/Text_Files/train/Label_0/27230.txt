#include <bits/stdc++.h>
using namespace std;
const int N = 100;
const long long mod = (long long)1e9 + 7;
long long a[N];
int main() {
  int n;
  a[1] = 1;
  a[3] = 3;
  a[5] = 15;
  a[7] = 133;
  a[9] = 2025;
  a[11] = 37851;
  a[13] = 1030367;
  a[15] = 36362925;
  while (cin >> n) {
    if (n % 2 == 0) {
      cout << 0 << endl;
      continue;
    }
    long long power = 1;
    for (int i = 1; i <= n; i++) {
      power *= i;
      power %= mod;
    }
    cout << a[n] * power % mod << endl;
  }
  return 0;
}
