n = int(input())
s = input()
A = s.count('A')
C = s.count('C')
G = s.count('G')
T = s.count('T')
Max = n / 4
if n % 4 == 0 and max(A, C, G, T) <= Max:
	if '?' not in s:
		print(s)
	else:
		s = list(s)
		for i in range(n):
			if s[i] == '?':
				if A < Max:
					s[i] = 'A'
					A += 1
				elif C < Max:
					s[i] = 'C'
					C += 1
				elif G < Max:
					s[i] = 'G'
					G += 1
				elif T < Max:
					s[i] = 'T'
					T += 1
				else:
					break
		print(''.join(s))
else:
	print('===')
