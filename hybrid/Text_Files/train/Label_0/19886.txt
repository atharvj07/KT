#include <bits/stdc++.h>
using namespace std;
const int big = 1000000000;
const double INF = 1e9;
const double EPS = 1e-9;
int main() {
  int n;
  scanf("%d", &n);
  int vals[200 + 5];
  int tmp;
  int plus = 0, mins = 0, nuls = 0;
  for (int i = 0, _c = 2 * n - 1; i < _c; ++i) {
    scanf("%d", &vals[i]);
    if (vals[i] == 0) nuls++;
    if (vals[i] < 0) mins++;
    if (vals[i] > 0) plus++;
    vals[i] = abs(vals[i]);
  }
  sort(vals, vals + 2 * n - 1);
  reverse(vals, vals + 2 * n - 1);
  int sum = 0;
  for (int i = 0, _c = 2 * n - 1; i < _c; ++i) sum += vals[i];
  if (nuls > 0)
    printf("%d\n", sum);
  else {
    if (n % 2)
      printf("%d\n", sum);
    else {
      if (plus % 2 == 1)
        printf("%d\n", sum);
      else
        printf("%d\n", sum - 2 * vals[2 * n - 2]);
    }
  }
  return 0;
}
