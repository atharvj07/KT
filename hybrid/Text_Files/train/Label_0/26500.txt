#include<bits/stdc++.h>
using namespace std;
using Int = long long;
template<typename T1,typename T2> inline void chmin(T1 &a,T2 b){if(a>b) a=b;}
template<typename T1,typename T2> inline void chmax(T1 &a,T2 b){if(a<b) a=b;}

//INSERT ABOVE HERE
signed main(){
  Int n,m,k;
  cin>>n>>m>>k;
  Int a,b,c;
  cin>>a>>b>>c;
  Int t;
  cin>>t;
  vector<Int> s(m);
  for(Int i=0;i<m;i++) cin>>s[i],s[i]--;

  k-=m;
  using P = pair<Int, Int>;
  priority_queue<P> pq;
  Int ans=0;
  if((n-1)*b<=t) ans++;
  
  vector<Int> lst(s);
  for(Int i=0;i+1<m;i++){    
    Int x=s[i]*b;
    if(x>t) continue;

    Int d=(t-x)/a+1;
    if(s[i]+d>=s[i+1]){
      ans+=s[i+1]-s[i];
      continue;
    }
    ans+=d;
    
    Int z=x+d*c;
    if(z>t) continue;
    Int e=d+(t-z)/a+1;
    
    //cout<<i<<" "<<d<<" "<<e<<endl;
    if(s[i]+e>=s[i+1]){
      pq.emplace(s[i+1]-(s[i]+d),-1);
      continue;
    }
    lst[i]=e;
    pq.emplace(e-d,i);
  }

  while(!pq.empty()&&k){
    --k;
    P p=pq.top();pq.pop();
    ans+=p.first;
    Int i=p.second;
    if(i<0) continue;
    Int x=s[i]*b;
    Int d=lst[i];
    Int z=x+d*c;
    if(z>t) continue;
    Int e=d+(t-z)/a+1;
    if(s[i]+e>=s[i+1]){
      pq.emplace(s[i+1]-(s[i]+d),-1);
      continue;
    }
    lst[i]=e;
    pq.emplace(e-d,i);
  }
  
  cout<<ans-1<<endl;
  return 0;
}
