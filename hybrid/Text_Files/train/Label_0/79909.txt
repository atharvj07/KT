import os
from io import BytesIO
input = BytesIO(os.read(0, os.fstat(0).st_size)).readline
(n, k) = map(int, input().split())
g = [x - 65 for x in input()]
e = [-1] * 26
for i in range(n):
	e[g[i]] = i
curS = 0
met = [0] * 26
for i in range(n):
	if not met[g[i]]:
		curS += 1
		met[g[i]] = 1
	if curS > k:
		print('YES')
		exit()
	if i == e[g[i]]:
		curS -= 1
print('NO')
