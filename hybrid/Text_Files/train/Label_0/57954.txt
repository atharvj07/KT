static void testcase() {
	n = s.nextInt();
	int[] a = s.nextIntArray(n);
	
	
	// now each element is number according to its position in the sorted array 
	// now each element in a is b/w [0, n-1]
	cordinateCompress(a); 
	
	int ans = 0;
	ArrayList<Integer>[] indices = new ArrayList[n];
	for(int i = 0; i < n; i++) indices[i] = new ArrayList<Integer>();
	for(int i = 0; i < n; i++) indices[a[i]].add(i);
	for(int i = 0; i < n; i++) ans = Math.max(ans, indices[i].size());
	
	int[] dp = new int[n];
	dp[n - 1] = indices[n - 1].size();
	for(int i = n - 2; i >= 0; i--) {
		if(indices[i].size() == 0)continue;
		dp[i] = indices[i].size();
		if(indices[i + 1].size() == 0)continue;
		if(indices[i + 1].get(0) > indices[i].get(indices[i].size() - 1))
			dp[i] += dp[i + 1];
		else
			dp[i] += howManyInRange(indices[i + 1], indices[i].get(indices[i].size() - 1), n - 1);
		
	}
	
	for(int i = 0; i < n; i++) {
		int prefix = i == 0 ? 0 : howManyInRange(indices[i - 1], 0, indices[i].isEmpty() ? -1 : indices[i].get(0) - 1);
		ans = Math.max(ans, dp[i] + prefix);
		if(a[i] < n - 1) {
			ans = Math.max(ans, howManyInRange(indices[a[i]], 0, i) + howManyInRange(indices[a[i] + 1], i + 1, n - 1));
		}
	}
	
	out.println(n - ans);
	
}


private static int howManyInRange(ArrayList<Integer> arr, int l, int r) {
	int len = arr.size();
	int lo = 0, hi = len - 1;
	int left = len;
	while(lo <= hi) {
		int mid = (lo + hi) >> 1;
		if(arr.get(mid) >= l) {
			left = mid;
			hi = mid - 1;
		}else lo = mid + 1;
	}
	int right = -1;
	lo = 0; hi = len - 1;
	while(lo <= hi) {
		int mid = (lo + hi) >> 1;
		if(arr.get(mid) <= r) {
			right = mid;
			lo = mid + 1;
		}else hi = mid - 1;
	}
	return Math.max(0, right - left + 1);
//	if(left == len || right == -1)return 0;
//	out.println("in range =>  [" + l + " , " + r + " ] => " + (right - left + 1));
//	return right - left + 1;
}


static int nInRange(ArrayList<Integer> list, int l, int r) {
//	System.out.println("Querying nInRange: "+l+" "+r+" for array: "+list);
	int n=list.size();
	if (list.isEmpty() || list.get(0)>r || list.get(n-1)<l || r<l) return 0;
	
	//first index >= l
	int min=0, max=n-1;
	while (min!=max) {
		int mid=(min+max)/2;
		if (list.get(mid) >= l) {
			max=mid;
		}
		else {
			min=mid+1;
		}
	}
	int firstLegal=min;
	
	//last index <=r
	min=0; max=n-1;
	while (min!=max) {
		int mid=(min+max+1)/2;
		if (list.get(mid)<=r) {
			min=mid;
		}
		else {
			max=mid-1;
		}
	}
	int lastLegal=min;
	return lastLegal-firstLegal+1;
}

	for(int x : values)mp.put(x, index++);

}

static void shuffleArray(int[] a) {
	Random random = new Random();
	for (int i = a.length-1; i > 0; i--) {
		int index = random.nextInt(i + 1);
		int tmp = a[index];
		a[index] = a[i];
		a[i] = tmp;
	}
}
static void shuffleSort(int[] a) {

}
