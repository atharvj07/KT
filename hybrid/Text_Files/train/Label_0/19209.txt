#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  scanf("%d %d", &n, &m);
  int a[100005];
  a[0] = 0;
  for (int i = 1; i <= n; i++) {
    int c, t;
    scanf("%d %d", &c, &t);
    a[i] = c * t + a[i - 1];
  }
  int j = 1;
  for (int i = 1; i <= m; i++) {
    int x;
    scanf("%d", &x);
  m:
    if (x <= a[j]) {
      printf("%d %c", j, '\n');
    } else {
      j++;
      goto m;
    }
  }
  return 0;
}
