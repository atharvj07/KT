import java.util.*;

public class Packets{
    public static void main(String[] args){
        int n;
        int k=1;
        int r=1;
        int ans=1;
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        while(n>k){
            r*=2;
            k+=r;
            ans++;
        }
        System.out.println(ans);
    }
}
