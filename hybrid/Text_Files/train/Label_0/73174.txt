s = input()
t = input()
a = [0] * 26
for x in s:
	a[ord(x) - 97] += 1
flag = False
for x in t:
	if a[ord(x) - 97] > 0:
		a[ord(x) - 97] -= 1
	else:
		flag = True
		break
if flag:
	print('need tree')
else:
	i = 0
	j = 0
	while i < len(s) and j < len(t):
		if s[i] == t[j]:
			j += 1
		i += 1
	if len(s) != len(t):
		if j == len(t):
			print('automaton')
		else:
			print('both')
	else:
		print('array')
