def xor(a, b):
	a = bin(a)[2:]
	b = bin(b)[2:]
	while len(a) > len(b):
		b = '0' + b
	while len(b) > len(a):
		a = '0' + a
	c = ['0'] * len(a)
	for i in range(len(a)):
		if a[i] != b[i]:
			c[i] = '1'
	a = ''
	for i in c:
		a += i
	return int(a, 2)
