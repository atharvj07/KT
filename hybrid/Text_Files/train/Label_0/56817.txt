import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Scanner;
import java.util.TreeSet;

public class P813B
{
	public static void main(String[] args)
	{
		Scanner scan = new Scanner(System.in);
		long x = scan.nextLong();
		long y = scan.nextLong();
		ArrayList<Long> powX = getPow(x);
		ArrayList<Long> powY = getPow(y);
		TreeSet<Long> ts = new TreeSet<>();
		for (int i = 0; i < powX.size(); i++)
		{
			for (int j = 0; j < powY.size(); j++)
			{
				long a = powX.get(i);
				long b = powY.get(j);
				if (BigInteger.valueOf(a).add(BigInteger.valueOf(b)).compareTo(BigInteger.valueOf(1000000000000000000L)) > 0)
					continue;
				ts.add(a+b);
			}
		}
		long max = 0;
		long left = scan.nextLong();
		long right = scan.nextLong();
		long curr = left;
		while (curr <= right)
		{
			Long hi = ts.higher(curr);
			int sub = ts.contains(curr) ? 1 : 0;
			if (hi == null || hi > right)
			{
				max = Math.max(max, right+1-curr-sub);
				break;
			}
			max = Math.max(max, hi-curr-sub);
			curr = hi;
		}
		System.out.println(max);
	}

	private static ArrayList<Long> getPow(long n)
	{
		ArrayList<Long> list = new ArrayList<>();
		long c = 1;
		while (true)
		{
			list.add(c);
			if (BigInteger.valueOf(c).multiply(BigInteger.valueOf(n)).compareTo(BigInteger.valueOf(1000000000000000000L)) > 0)
				break;
			c *= n;
		}
		return list;
	}
}
