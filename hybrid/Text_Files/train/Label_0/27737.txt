#include <bits/stdc++.h>
using namespace std;
void read(int &k) {
  k = 0;
  char x = getchar();
  while (x < '0' || x > '9') x = getchar();
  while (x >= '0' && x <= '9') {
    k = k * 10 - 48 + x;
    x = getchar();
  }
}
int a, n, ans[10000010];
long long ret;
int main() {
  cin >> a >> n;
  for (int i = 1; i * i <= 10000000; i++)
    for (int j = 1; j * i * i <= 10000000; j++) ans[j * i * i] = j;
  for (int i = a; i < a + n; i++) ret += ans[i];
  cout << ret << endl;
  return 0;
}
