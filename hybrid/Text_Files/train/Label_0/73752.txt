n = int(input())
x = [int(r) for r in input().split()]
m = 1
l = list()
for i in range(0, len(x) - 1):
	if x[i] == x[i + 1]:
		m = m + 1
	else:
		l.append(m)
		m = 1
		continue
l.append(m)
maxe = 1
for i in range(len(l) - 1):
	if l[i] <= l[i + 1] or l[i] >= l[i + 1]:
		maxe = max(min(l[i + 1], l[i]), maxe)
print(maxe * 2)
