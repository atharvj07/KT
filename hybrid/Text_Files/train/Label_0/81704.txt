N = int(input())
candies = [x + 1 for x in range(N ** 2)]
sumof_candies_per_person = sum(candies) / N
candies = list(reversed(sorted(candies)))
bags_per_person = []
for i in range(N):
	bags_per_person.append(candies[0:N // 2] + candies[-N // 2:])
	del candies[0:N // 2]
	del candies[-N // 2:]
for bags in bags_per_person:
	for bag in bags:
		print(bag, end=' ')
	print('')
