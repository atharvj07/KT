line = input()
q = int(line)
for _ in range(q):
	line = input()
	(n, s) = [x for x in line.split()]
	n = int(n)
	minseq = [n]
	potential = 0
	maxseq = [1]
	minel = 1
	maxel = 1
	for x in s:
		if x == '<':
			maxseq.append(maxel + 1)
			maxel += 1
			potential += 1
		elif x == '>':
			if potential == 0:
				minseq.append(minseq[-1] - 1)
			else:
				minseq[-1] -= potential
				tmp = minseq[-1]
				for _ in range(potential):
					minseq.append(minseq[-1] + 1)
				potential = 0
				minseq.append(tmp - 1)
			maxseq.append(minel - 1)
			minel -= 1
	if potential != 0:
		minseq[-1] -= potential
		for _ in range(potential):
			minseq.append(minseq[-1] + 1)
	for x in minseq:
		print(x, end=' ')
	print('')
	for x in maxseq:
		print(x - minel + 1, end=' ')
	print('')
