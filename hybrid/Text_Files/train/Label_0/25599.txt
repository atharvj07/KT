#include <bits/stdc++.h>
using namespace std;
const int inf = 1000111222;
const int N = 100007;
int n, m, row[N], rooks[N];
vector<pair<int, int> > adj[5007];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    ++rooks[i];
    row[i] = 1;
    adj[i].push_back({i, i});
  }
  int k = n + 1;
  while (m--) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back({k, a});
    adj[b].push_back({k, b});
    ++rooks[a];
    ++rooks[b];
    row[k++] = 1;
  }
  for (int i = 1; i <= n; i++) {
    cout << rooks[i] << "\n";
    for (pair<int, int> x : adj[i]) {
      cout << x.first << " " << x.second << "\n";
    }
  }
  return 0;
}
