import java.util.*;
public class _557A_Ilya_and_Diplomas {
    public static void main(String[] args) {
        Scanner input=new Scanner (System.in);
        int n=input.nextInt();
        int min1=input.nextInt(),max1=input.nextInt();
        int min2=input.nextInt(),max2=input.nextInt();
        int min3=input.nextInt(),max3=input.nextInt();
        while (min1+min2+min3!=n)
        {
            if (min1<max1)
                min1++;
            else if (min2<max2)
                min2++;
            else if (min3<max3)
                min3++;
        }
        System.out.println(min1+" "+min2+" "+min3);
        input.close();
    }
}
