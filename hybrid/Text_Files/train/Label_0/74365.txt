import sys
for _ in range(int(sys.stdin.readline())):
	(n, a, b) = [int(x) for x in sys.stdin.readline().split()]
	left = [a]
	right = [b]
	arr = list(range(1, n + 1))
	arr.remove(a)
	arr.remove(b)
	left = left + arr[len(arr):len(arr) // 2 - 1:-1]
	right = right + arr[:len(arr) // 2]
	if min(left) != a or max(right) != b:
		print(-1)
	else:
		print(' '.join(map(str, left + right)))
