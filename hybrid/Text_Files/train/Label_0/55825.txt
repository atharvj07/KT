    private static class Trie {
        int[] count;
        int[] left;
        int[] right;
        int index = 1;

        public Trie(String source) {
            int len = source.length() * source.length() + 2;
            count = new int[len];
            left = new int[len];
            right = new int[len];
        }

        public void add(String source, int start, int[] polyCount) {
            int len = source.length();
            int current = 1;
            count[current] += polyCount[start];
            for (int i = start; i < len; ++i) {
                if (source.charAt(i) == 'a') {
                    if (left[current] == 0) {
                        left[current] = ++index;
                    }
                    current = left[current];
                } else {
                    if (right[current] == 0) {
                        right[current] = ++index;
                    }
                    current = right[current];
                }
                count[current] += polyCount[i];
            }
        }

        public String search(int order) {
            int current = 1;
            StringBuilder builder = new StringBuilder();
            while (count[current] >= order) {
                order -= count[current] - count[left[current]] - count[right[current]];
                if (order <= 0) {
                    break;
                }
                if (order <= count[left[current]]) {
                    current = left[current];
                    builder.append('a');
                } else {
                    order -= count[left[current]];
                    current = right[current];
                    builder.append('b');
                }
            }
            return builder.toString();
        }

        @Override
        public String toString() {
            return "r " + Arrays.toString(right) +
                    "\nl " + Arrays.toString(left) +
                    "\nc " + Arrays.toString(count);
        }
    }

    private static String getHalfPolyndrome(String input, int order) {
        Trie trie = new Trie(input);
        int len = input.length();
        int polyCount[][] = new int[len][len];
        for (int i = 0; i < len; i++) {
            fillPolyndromes(input, i, i, polyCount);
            fillPolyndromes(input, i, i + 1, polyCount);
            fillPolyndromes(input, i, i + 2, polyCount);
            fillPolyndromes(input, i, i + 3, polyCount);
        }
        for (int i = 0; i < len; ++i) {
            for (int j = len - 2; j >= i; --j) {
                polyCount[i][j] += polyCount[i][j + 1];
            }
        }
        for (int i = 0; i < len; ++i) {
            trie.add(input, i, polyCount[i]);
        }
        return trie.search(order);
    }

    private static void fillPolyndromes(String input, int left, int right, int[][] polyCount) {
        while (left >= 0 && right < input.length() && input.charAt(left) == input.charAt(right)) {
            polyCount[left][right] = 1;
            left -= 2;
            right += 2;
        }
    }

    private static List<String> getHalfPolyndromesBrute(String input) {
        List<String> halfPolyndromes = new ArrayList<>();
        for (int i = 0; i < input.length(); ++i) {
            fillPolyndromesBrute(input, i, i, halfPolyndromes);
            fillPolyndromesBrute(input, i, i + 1, halfPolyndromes);
            fillPolyndromesBrute(input, i, i + 2, halfPolyndromes);
            fillPolyndromesBrute(input, i, i + 3, halfPolyndromes);
        }
        Collections.sort(halfPolyndromes);
        return halfPolyndromes;
    }

    private static void fillPolyndromesBrute(String input, int left, int right, List<String> collector) {
        while (left >= 0 && right < input.length() && input.charAt(left) == input.charAt(right)) {
            collector.add(input.substring(left, right + 1));
            left -= 2;
            right += 2;
        }
    }

    private static String generateInput(Random r, int length, double skew) {
        StringBuilder builder = new StringBuilder(length);
        for (int i = 0; i < length; i++) {
            builder.append(r.nextDouble() > skew ? 'a' : 'b');
        }
        return builder.toString();
    }

    private static void solve() {
        Scanner s = new Scanner(System.in);
        String input = s.next();
        int k = s.nextInt();
        String result = getHalfPolyndrome(input, k);
        System.out.println(result);
    }

    private static void test() {
        Random r = new Random(137);
        for (int i = 0; i < 100; ++i) {
            int length = r.nextInt(100);
            String input = generateInput(r, length, 0.5);
            List<String> result = getHalfPolyndromesBrute(input);
