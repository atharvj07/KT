def main():
	(n, x1, y1, x2, y2) = map(int, input().split())
	(rr1, rr2) = ([], [])
	fontains = ((x1, y1, rr1.append), (x2, y2, rr2.append))
	for _ in range(int(n)):
		(x, y) = map(int, input().split())
		for (u, v, f) in fontains:
			u -= x
			v -= y
			f(u * u + v * v)
	idx1 = sorted(range(n), key=lambda i: (rr1[i], -rr2[i]))
	idx2 = sorted(range(n), key=rr2.__getitem__)
	if idx1[-1] == idx2[-1]:
		print(min(rr1[idx1[-1]], rr2[idx2[-1]]))
		return
	res = [rr1[idx1.pop()], rr2[idx2[-1]]]
	for i in idx1:
		idx2.remove(i)
		res.append(rr1[i] + rr2[idx2[-1]])
	print(min(res))
main()
