#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  int k[n], x[100] = {0};
  for (int i = 0; i < n; i++) {
    cin >> k[i];
  }
  int m[100][100];
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < k[i]; j++) {
      cin >> m[i][j];
      x[i] = x[i] + m[i][j];
    }
  }
  for (int i = 0; i < n; i++) {
    x[i] = x[i] * 5 + k[i] * 15;
  }
  cout << *min_element(x, x + n) << endl;
  return 0;
}
