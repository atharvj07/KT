#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  string s, t;
  int pos = 0;
  cin >> s >> t;
  for (int i = 0; i < t.size(); i++) {
    if (s[pos] == t[i]) {
      pos++;
    }
    if (pos == s.size()) {
      cout << pos << endl;
      return 0;
    }
  }
  cout << pos + 1 << endl;
  return 0;
}
