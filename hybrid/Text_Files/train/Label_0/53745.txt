    static void solve() throws IOException
    {
        //initIo(true);
        initIo(false);
        StringBuilder sb = new StringBuilder();
        int min_steps[][][][] = new int[10][10][10][10];
        for(int i=0;i<10;++i) {
            for(int j=0;j<10;++j) {
                for(int k=0;k<10;++k) {
                    for(int m=0;m<10;++m) {
                        min_steps[i][j][k][m] = -1;
                    }
                }
            }
        }
        for(int i=0;i<10;++i) {
            for(int j=0;j<10;++j) {
                for(int start=0;start<10;++start) {
                    int min[] = new int[10];
                    Arrays.fill(min, -1);
                    LinkedList<Integer> list = new LinkedList<>();
                    list.add((start+i)%10);
                    min[(start+i)%10] = 1;
                    if((start+j)%10 != (start+i)%10) {
                        list.add((start+j)%10);
                        min[(start+j)%10] = 1;
                    }
                    while(!list.isEmpty()) {
                        int curr = list.pollFirst();
                        if(min[(curr+i)%10]==-1) {
                            min[(curr+i)%10] = 1+min[curr];
                            list.add((curr+i)%10);
                        }
                        if(min[(curr+j)%10]==-1) {
                            min[(curr+j)%10] = 1+min[curr];
                            list.add((curr+j)%10);
                        }
                    }
                    for(int des=0;des<10;++des) {
                        min_steps[i][j][start][des] = min[des];
                    }
                }
            }
        }
        //pl("Pre calc over");
        char c[] = ne().toCharArray();
        int n = c.length;
        for(int dig1=0;dig1<10;++dig1) {
            for(int dig2=0;dig2<10;++dig2) {
                //pl("Dig1: "+dig1+" Dig2: "+dig2);
                int ans = 0, curr = 0, ptr = 1;
                while(ptr<n) {
                    int req = c[ptr] - '0';
                   // pl("curr: "+curr+" req: "+req+" will have to perform: "+ min_steps[dig1][dig2][curr][req]);
                    if(min_steps[dig1][dig2][curr][req] == -1) {
                        ans = -1;
                        break;
                    }
                    else {
                        ans+=min_steps[dig1][dig2][curr][req]-1;
                        curr = req;
                        ptr++;
                    }
                }
                sb.append(ans);
                sb.append(" ");
            }
            sb.append("\n");
        }
        p(sb);
        pw.flush();
        pw.close();
    }
