#include <bits/stdc++.h>
using namespace std;
const int maxn = 100001;
int a[maxn];
long long least(long long n) {
  long long res;
  if (n & 1) {
    res = ((n * (n - 1)) >> 1) + 1;
  } else {
    res = (n * n) >> 1;
  }
  return res;
}
int main() {
  int n, m, x;
  scanf("%d %d", &n, &m);
  for (int i = 0; i < m; i++) {
    scanf("%d %d", &x, &a[i]);
  }
  sort(a, a + m, greater<int>());
  long long res = 0;
  for (int i = 0; i < m; i++) {
    long long h = i + 1;
    long long ans = least(h);
    if (ans <= n) {
      res += a[i];
    } else {
      break;
    }
  }
  cout << res;
  return 0;
}
