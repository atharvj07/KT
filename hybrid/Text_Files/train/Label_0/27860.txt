#include <bits/stdc++.h>
using namespace std;
struct Isap {
  static const int MXN = 54;
  struct Edge {
    int v, f, re;
  };
  int n, s, t, h[MXN], gap[MXN];
  vector<Edge> E[MXN];
  void init(int _n, int _s, int _t) {
    n = _n;
    s = _s;
    t = _t;
    for (int i = 0; i < n; i++) E[i].clear();
  }
  void add_edge(int u, int v, int f) {
    E[u].push_back({v, f, ((int)((E[v]).size()))});
    E[v].push_back({u, 0, ((int)((E[u]).size())) - 1});
  }
  int DFS(int u, int nf, int res = 0) {
    if (u == t) return nf;
    for (auto &it : E[u]) {
      if (h[u] == h[it.v] + 1 && it.f > 0) {
        int tf = DFS(it.v, min(nf, it.f));
        res += tf;
        nf -= tf;
        it.f -= tf;
        E[it.v][it.re].f += tf;
        if (nf == 0) return res;
      }
    }
    if (nf) {
      if (--gap[h[u]] == 0) h[s] = n;
      gap[++h[u]]++;
    }
    return res;
  }
  int flow(int res = 0) {
    memset(h, 0, sizeof(h));
    memset(gap, 0, sizeof(gap));
    gap[0] = n;
    while (h[s] < n) res += DFS(s, 2147483647);
    return res;
  }
} flow;
struct Edge {
  int u, v, w;
} edge[514];
int N, M, X;
inline int check_ok(long double m) {
  flow.init(N, 0, N - 1);
  for (int i = 0; i < (M); i++) {
    int w = min(edge[i].w / m, (long double)X);
    w = max(w, 0);
    if (w <= 0) continue;
    flow.add_edge(edge[i].u, edge[i].v, w);
  }
  int f = flow.flow();
  return f >= X;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  ;
  cin >> N >> M >> X;
  for (int i = 0; i < (M); i++) {
    cin >> edge[i].u >> edge[i].v >> edge[i].w;
    edge[i].u--;
    edge[i].v--;
  }
  long double l = 0, r = 1e6 + 5;
  for (int _ = 0; _ < (300); _++) {
    long double m = (l + r) / 2;
    if (check_ok(m))
      l = m;
    else
      r = m;
  }
  l *= X;
  cout << fixed << setprecision(10) << l << endl;
  return 0;
}
