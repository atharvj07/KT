def sol():
	am = int(input())
	darr = list(map(int, input().split()))
	if am == 1 or am == 2:
		print(0)
		return 0
	sl = darr[2:]
	f = darr[0]
	s = darr[1]

	def c(arr):
		dif = arr[1] - arr[0]
		counter = 0
		last = arr[0]
		for i in range(1, len(arr)):
			if arr[i] - last == dif:
				last = arr[i]
				continue
			if arr[i] - last == dif - 1:
				last = arr[i] + 1
				counter += 1
				continue
			if arr[i] - last == dif + 1:
				last = arr[i] - 1
				counter += 1
				continue
			break
		else:
			return counter
		return 999999
	d = [2 + c([f + -1] + [s + -1] + sl), 1 + c([f + -1] + [s + 0] + sl), 1 + c([f + 0] + [s + -1] + sl), 0 + c([f + 0] + [s + 0] + sl), 2 + c([f + 1] + [s + 1] + sl), 1 + c([f + 1] + [s + 0] + sl), 1 + c([f + 0] + [s + 1] + sl), 2 + c([f + 1] + [s + 1] + sl), 2 + c([f + -1] + [s + 1] + sl), 2 + c([f + 1] + [s + -1] + sl)]
	m = 9999999
	for i in range(10):
		if d[i] <= 100000:
			m = min(m, d[i])
	print(m if m != 9999999 else -1)
sol()
