import math
(n, m, k) = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
arr = [0] * (len(a) + 1)
brr = [0] * (len(b) + 1)
for i in range(1, len(a) + 1):
	arr[i] = a[i - 1] + arr[i - 1]
for i in range(1, len(b) + 1):
	brr[i] = b[i - 1] + brr[i - 1]
ans = 0
for f in range(1, int(k ** 0.5) + 1):
	if k % f == 0:
		x = f
		y = k // f
		if y <= m:
			cc = 0
			for i in range(len(arr)):
				if i + x >= len(arr):
					break
				if arr[i + x] - arr[i] == x:
					cc += 1
			bb = 0
			for i in range(len(brr)):
				if i + y >= len(brr):
					break
				if brr[i + y] - brr[i] == y:
					bb += 1
			ans += cc * bb
		if x != y:
			y = f
			x = k // f
			cc = 0
			for i in range(len(arr)):
				if i + x >= len(arr):
					break
				if arr[i + x] - arr[i] == x:
					cc += 1
			bb = 0
			for i in range(len(brr)):
				if i + y >= len(brr):
					break
				if brr[i + y] - brr[i] == y:
					bb += 1
			ans += cc * bb
print(ans)
