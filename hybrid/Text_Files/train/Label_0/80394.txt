from sys import stdin
input = stdin.readline
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	x = bin(n)[2:].zfill(32)
	y = bin(m)[2:].zfill(32)
	ans = []
	mini = ['1'] * 32
	for i in range(32):
		if y[i] == '1':
			if x[i] == '1':
				ans.append('0')
			else:
				ans.append('1')
		elif x[i] == '1':
			mini = min(mini, ans + ['0'] * (32 - len(ans)))
			ans.append('1')
		else:
			mini = min(mini, ans + ['1'] + ['0'] * (31 - len(ans)))
			ans.append('0')
	print(int(''.join(mini), 2))
