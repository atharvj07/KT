#include <bits/stdc++.h>
#include <iostream>
using namespace std;
 
const int n = 1e5+10;
int a[n];




void solve()
{
   int n;
 cin>>n;

 vector<int>v1;
 vector<int>v2;

 for(int i=0; i<n; i++)
 {
    cin>>a[i];
 }

 for(int i=0; i<n; i++)
 {
    if(a[i]%2==0)
      v1.push_back(a[i]);
    else
      v2.push_back(a[i]);
 }

 for(auto i : v2)
 {
  cout<<i<<" ";
 }

 for(auto i : v1)
 {
  cout<<i<<" ";
 }

 cout<<endl;

}
 
int main()
{

// #endif
 
    int t;
    cin >> t;
 
    while (t--)
    {
        solve();
    }
 
 
    return 0;
}
