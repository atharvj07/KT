#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define mk make_pair
typedef long long ll;
ll min(ll a, ll b){
                if(a<b)
                return a;
                else
                return b;
}
ll dp[5001][5001];
//dp[i][j] denotes person at index i is seated at chair at j
//and all 0 to i-1 persons are already seated optimally
vector<int> one,zero;
ll calc(ll i, ll j){
  //j = position of zero
  //i = position of one;
  if(i == one.size())
  return 0;
  if(j == zero.size()) //not a valid answer bcz persons are  still there
    return 2 * 1e18 + 1;       //standing and chairs are exhausted

  if(dp[i][j] != -1)
    return dp[i][j];

  int possibility1 = calc(i+1,j+1) + abs(one[i] - zero[j]);
  //means one[i]th person chooses chair at index zero[j] and 
  //our search space now reduces to find from i+1th person and j+1th chair

  int possibility2 = calc(i,j+1);
  //person i refuses to sit at chair j so  answer now is answer for ith person till end from 
  //j+1th chair till end
  //start placing ith person from j+1th chair
  //actually dp[i][j] is optimal answer for ith to last  person with available chairs from jth to zero.size()
  //then our final answer will be at dp[0][0]
  //bcz we want from starting person with all the chairs available to sit

  dp[i][j] = min(possibility1,possibility2);
  //It's like ith person went in the future and saw what will cost him the minimum 
  //sum if he will sit on the jth chair then he calculated future ans from i+1th and j+1th 
  //and if we will not sit on jth chair instead choose from j+1th chair
  //now if ith person rejects jth chair it won't we there  to sit 
  //for any of i+1th to nth person so he calculated (i,j+1);
  //we are calculating ans backward
  return dp[i][j];
}
void solve(){
int n;
cin>>n;
int x;
memset(dp,-1,sizeof(dp));
for(int i = 0; i<n;i++){
  cin>>x;
  if(x)
    one.pb(i);
  else
    zero.pb(i);
  // Eg. 12
  // 0 0 1 1 0 1 1 0 1 1 0 0
  // we need to place all the ones at some positions 
  // where zeros are kept such that sum of abs difference is minimum
  // observations 1 :
  // indexes at which one occur are = 2, 3, 5, 6, 8, 9
  // indexes at which zero occur are= 0, 1, 4, 7, 10, 11
  //now since we are humans we know first one must come at index 0 or 1 
  //but WTLOG let we choose to place first one at index 4 
  //then we know for sure that second one which is at index 3 must be placed
  //after index 4 because if the person at index 3 finds a vacant seat that is
  //to the left of person at index 2 then why would not person at 2 go and 
  //sit there because it is nearer to person at 2 than person at 3.
  //so if we are considering that 1st person is seated at position x then 
  //to find optimal answer we must consider that 2nd person will be seated 
  //at any index of zero that is after the chosen index of zero.
 //we are at first index of 0(0) and first index of 1(2)
}
if(one.size() == 0)

}
