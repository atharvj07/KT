n = int(input())
arr = [[] for _ in range(n)]
t = []
for _ in range(2 * n - 2):
	x = input()
	t.append(x)
	arr[len(x)].append(x)
words = {arr[1][0] + arr[n - 1][0], arr[1][1] + arr[n - 1][0], arr[1][0] + arr[n - 1][1], arr[1][1] + arr[n - 1][1]}
wrong_words = set()
for x in words:
	for i in range(1, n):
		if not (arr[i][0] == x[:i] and arr[i][1] == x[-i:] or (arr[i][1] == x[:i] and arr[i][0] == x[-i:])):
			wrong_words.add(x)
			break
words = words - wrong_words
s = words.pop()
ans = []
check = [[True, True] for _ in range(n)]
for x in t:
	if s[:len(x)] == x and check[len(x)][0]:
		check[len(x)][0] = False
		ans.append('P')
	else:
		ans.append('S')
		check[len(x)][1] = False
print(*ans, sep='')
