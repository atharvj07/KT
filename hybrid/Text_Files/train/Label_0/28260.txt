#include <bits/stdc++.h>
#pragma warning(disable : 4786)
using namespace std;
int a[200000];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  a[n + 1] = a[1];
  int y = 0;
  for (int i = 2; i <= n + 1; i++) {
    if (a[i] < a[i - 1]) {
      if (y != 0) {
        printf("-1\n");
        return 0;
      } else {
        y = i;
      }
    }
  }
  if (y == 0) y = n + 1;
  printf("%d", n - y + 1);
  return 0;
}
