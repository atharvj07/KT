	{

		int n = in.nextInt(); in.nextLine();
		String[] fullNames = new String[n];
		String[] names1 = new String[n];
		String[] names2 = new String[n];
		int[] namesAns = new int[n];
		
		HashMap<String,ArrayList<Integer>> nameInds = new HashMap<String,ArrayList<Integer>>();
		HashSet<String> badNames = new HashSet<String>();
		
		//HashSet<String> firstNames = new HashSet<String>();
		//HashSet<String> secondNames = new HashSet<String>();
		//LinkedList<String> toRemove = new LinkedList<String>();
		//HashSet<String> currentNames = new HashSet<String>();
		
		for(int i = 0; i < n; ++i) {
			String newName = in.nextLine();
			fullNames[i] = newName;
			int spaceIndex = newName.indexOf(' ');
			names1[i] = newName.substring(0, 3).toUpperCase();
			names2[i] = (newName.substring(0, 2)+newName.charAt(spaceIndex+1)).toUpperCase();
			namesAns[i] = 1;
			
			if(nameInds.containsKey(names1[i])) {
				nameInds.get(names1[i]).add(i);
				badNames.add(names1[i]);
			} else {
				ArrayList<Integer> inds = new ArrayList<Integer>();
				inds.add(i);
				nameInds.put(names1[i], inds);
			}

		}
//		System.out.println(Arrays.toString(names1));
//		System.out.println(Arrays.toString(names2));
//		System.out.println(Arrays.toString(namesAns));

		while(badNames.size() > 0) {
			String dupl = badNames.iterator().next();
			ArrayList<Integer> badInds = nameInds.get(dupl);
			ArrayList<Integer> newInds = new ArrayList<Integer>();
			
			boolean backupName = false;
			
			for(int i : badInds) {
				if(namesAns[i] == 1) {
					String newName = names2[i];
					namesAns[i] = 2;
					if(names1[i].equals(names2[i])) {
						if(backupName) {
							System.out.println("NO");
							return;
						} else {
							backupName = true;
							newInds.add(i);
						}
					} else if(nameInds.containsKey(newName)) {
						nameInds.get(newName).add(i);
						badNames.add(newName);
					} else {
						ArrayList<Integer> inds = new ArrayList<Integer>();
						inds.add(i);
						nameInds.put(newName, inds);
					}
				} else if(namesAns[i] == 2){
					if(backupName) {
						System.out.println("NO");
						return;
					} else {
						backupName = true;
						newInds.add(i);
					}
				} else {
					System.out.println("ERROR");
				}
			}
			
			badNames.remove(dupl);
			if(newInds.size() > 0) {
				nameInds.put(dupl, newInds);
			} else {
				nameInds.remove(dupl);
			}
		}
		System.out.println("YES");
		for(int i = 0; i < namesAns.length; ++i) {
			if(namesAns[i] == 1) {
				System.out.println(names1[i]);
			} else {
				System.out.println(names2[i]);
			}
		}
