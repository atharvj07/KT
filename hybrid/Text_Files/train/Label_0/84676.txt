from sys import stdin
input = stdin.readline

def canine(s):
	lst = [i for i in s]
	cnt = 0
	for i in range(len(lst) - 1):
		if lst[i] == lst[i + 1]:
			lst[i + 1] = None
			cnt += 1
		if i > 0:
			if lst[i - 1] == lst[i + 1] and lst[i + 1] != None:
				lst[i + 1] = None
				cnt += 1
	return cnt
for i in range(int(input())):
	print(canine(input().strip()))
