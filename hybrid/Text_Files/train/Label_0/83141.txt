n = int(input())
s = input()
b = []
k = 1
a0 = s[0]
ma = 0
for i in range(1, n):
	if a0 == s[i] and i == n - 1:
		k += 1
		ma = max(ma, k)
		b.append(k)
	elif a0 == s[i]:
		k += 1
	elif a0 != s[i] and i == n - 1:
		ma = max(ma, k)
		b.append(k)
		b.append(1)
	else:
		ma = max(ma, k)
		b.append(k)
		a0 = s[i]
		k = 1
key = 0
if n == 1:
	key = 1
	print(1)
if ma > 2:
	key = 1
	print(len(b) + 2)
elif key == 0:
	i = 0
	k = 0
	while i < len(b):
		if b[i] > 1:
			k += 1
		if k > 1:
			print(len(b) + 2)
			key = 1
			break
		i += 1
if key == 0:
	if k > 0:
		print(len(b) + 1)
	else:
		print(len(b))
