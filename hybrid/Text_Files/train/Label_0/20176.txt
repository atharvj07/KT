#include <bits/stdc++.h>
using namespace std;
int main() {
  int N, i, x, y;
  cin >> N;
  vector<pair<int, int>> v1, v2;
  for (i = 0; i < N; i++) {
    cin >> x >> y;
    if (x > y)
      v1.push_back({y, i + 1});
    else
      v2.push_back({-y, i + 1});
  }
  sort(v1.begin(), v1.end());
  sort(v2.begin(), v2.end());
  if (v1.size() > v2.size()) swap(v1, v2);
  cout << v2.size() << endl;
  for (i = 0; i < v2.size(); i++) cout << v2[i].second << " ";
  return 0;
}
