import java.util.*;
public class Modification {
public static void main(String [] args){
Scanner in=new Scanner(System.in);
int n=in.nextInt();
int m=in.nextInt();
int k=in.nextInt();
int p=in.nextInt();
Comparator<Long>comp=new Comparator<Long>(){
public int compare(Long a,Long b){
return b.compareTo(a);
}
};
PriorityQueue<Long>rows=new PriorityQueue<Long>(n,comp);
PriorityQueue<Long>columns=new PriorityQueue<Long>(m,comp);
long rows1[]=new long[n];
long cols1[]=new long[m];
for(int i=0;i<n;i++){
for(int j=0;j<m;j++){
int a =in.nextInt();
rows1[i]+=a;
cols1[j]+=a;
}
}
for(int i=0;i<n;i++){
rows.add(rows1[i]);
}
for(int i=0;i<m;i++){
columns.add(cols1[i]);
}
long finrow[]=new long[k+1];
long fincol[]=new long[k+1];
//System.out.println(rows);
for(int i=1;i<=k;i++){
long a=rows.poll();
long b=columns.poll();
finrow[i]=finrow[i-1]+a;
fincol[i]=fincol[i-1]+b;
rows.add(a-m*p);
columns.add(b-n*p);
//System.out.println(a+" "+b+" "+finrow[i]+" "+fincol[i]+" ");
}
//System.out.println(rows);
long sum=Long.MIN_VALUE;
for(int i=0;i<=k;i++){
sum=Math.max(finrow[i]+fincol[k-i]-(long)i*(k-i)*p,sum);
//System.out.println(sum+" "+temp);
}
System.out.println(sum);
}
}
