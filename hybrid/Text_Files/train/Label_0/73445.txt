from collections import defaultdict as dd
import sys
input = sys.stdin.readline
t = int(input())
while t:
	n = int(input())
	d = dd(int)
	l = list(map(int, input().split()))
	ll = list(map(int, input().split()))
	lol = 0
	for i in range(n // 2):
		(a, b) = (l[i], l[n - i - 1])
		if a > b:
			(a, b) = (b, a)
		d[a, b] += 1
	for i in range(n // 2):
		(a, b) = (ll[i], ll[n - i - 1])
		if a > b:
			(a, b) = (b, a)
		if d[a, b] == 0:
			lol = 1
			break
		d[a, b] -= 1
	if n % 2:
		if l[n // 2] != ll[n // 2]:
			lol = 1
	if lol:
		print('no')
	else:
		print('yes')
	t -= 1
