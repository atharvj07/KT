n = int(input())
y = list(map(int, input().split()))
z = list(map(int, input().split()))
a = sorted([[y[i], i] for i in range(n)] + [[z[i], i] for i in range(n)])
f = 0
for i in range(n):
	if y[i] == z[i]:
		f += 1
m = int(input())
d = 0
e = 1
for i in range(1, 2 * n):
	if a[i][0] != a[i - 1][0]:
		for j in range(1, i - d + 1):
			while f > 0 and j % 2 == 0:
				j //= 2
				f -= 1
			e *= j
			e %= m
		d = i
for j in range(1, 2 * n - d + 1):
	while f > 0 and j % 2 == 0:
		j //= 2
		f -= 1
	e *= j
	e %= m
print(e % m)
