n = int(input())
a = list(map(int, input().split()))
a1 = sorted(a)
if a1 != a:
	print(-1)
else:
	m = max(a) - 1
	mm = m + 10
	ans = [-1] * n
	has = [0] * (n + 10)
	for i in range(n - 1, -1, -1):
		has[a[i]] = 1
		if a[i] > i + 1:
			print(-1)
			exit()
		elif i > 0:
			if a[i - 1] != a[i]:
				ans[i] = a[i - 1]
	c = 0
	for i in range(n):
		if ans[i] != -1:
			continue
		while has[c]:
			c += 1
		ans[i] = c
		has[c] = 1
	print(*ans)
