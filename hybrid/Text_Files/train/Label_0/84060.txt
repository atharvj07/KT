import sys, collections
input = sys.stdin.readline

def bfs(adj_list, n):
	c = [1] * n
	v = [-1] * n
	v[0] = 0
	q = [0]
	while q:
		x = q[-1]
		flag = False
		for to in adj_list[x]:
			if v[to] == -1:
				v[to] = x
				q.append(to)
				flag = True
		if not flag:
			q.pop()
			c[v[x]] += c[x]
	ans = 0
	for i in c[1:]:
		if i & 1 == 0:
			ans += 1
	print(ans)
n = int(input())
if n % 2 == 1:
	print(-1)
	exit()
adj_list = [[] for _ in range(n + 10)]
for _ in range(n - 1):
	(a, b) = map(int, input().split())
	a -= 1
	b -= 1
	adj_list[a].append(b)
	adj_list[b].append(a)
bfs(adj_list, n)
