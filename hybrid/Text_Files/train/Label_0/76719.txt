read = lambda : map(int, input().split())
(n, m) = read()
l = list(read())
a = [0] * (n + 1)

def fail():
	print(-1)
	exit()
for i in range(m - 1):
	pr = -1
	if a[l[i]]:
		pr = a[l[i]]
	a[l[i]] = (l[i + 1] - l[i]) % n
	if a[l[i]] == 0:
		a[l[i]] = n
	if pr != -1 and pr != a[l[i]]:
		fail()
s = set(a)
mex = 0
for i in range(1, n + 1):
	if not a[i]:
		while mex in s:
			mex += 1
		a[i] = mex
		s.add(mex)
p = [i for i in range(1, n + 1)]
if sorted(a[1:]) != p:
	fail()
print(*a[1:])
