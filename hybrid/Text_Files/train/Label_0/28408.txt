#include <bits/stdc++.h>
using namespace std;
using LL = long long;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  int w, h;
  vector<pair<int, int>> vp;
  vp.reserve(n);
  for (int i = 0; i < n; i++) {
    cin >> w >> h;
    vp.push_back(make_pair(w, h));
  }
  int d = 0;
  int f = 0;
  d = max(vp[0].first, vp[0].second);
  for (int i = 1; i < n; i++) {
    if (d >= max(vp[i].first, vp[i].second)) {
      d = max(vp[i].first, vp[i].second);
    } else if (d >= min(vp[i].first, vp[i].second)) {
      d = min(vp[i].first, vp[i].second);
    } else {
      f = 1;
      break;
    }
  }
  if (f == 1)
    cout << "NO" << endl;
  else
    cout << "YES" << endl;
  return 0;
}
