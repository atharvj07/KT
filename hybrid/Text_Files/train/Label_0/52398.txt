import java.util.*;
public class Bears{
    public static void main(String args[])
    {
        Scanner s=new Scanner(System.in);
        int w1=s.nextInt();
        int w2=s.nextInt();
        int years=0;
        do
        {
          if(w1>w2)
          break;
          else
          {
              years++;
              w1=w1*3;
              w2=w2*2;
          }
        }while(true);
        System.out.println(years);
        
    }
}
