import java.util.*;
import java.io.*;

public class D1495 {
	static int[][] adjList;
	static int[][] adjMatrix;
	static int n;

	public static void floyd(int V) {
		for (int k = 0; k < V; k++)
			bfs(k);
	}

	public static void bfs(int u) {
		Queue<Integer> q = new LinkedList<Integer>();
		q.add(u);
		int[] dist = new int[n];
		Arrays.fill(dist, -1);
		dist[u] = 0;
		while (!q.isEmpty()) {
			int cur = q.poll();
			for (int x : adjList[cur]) {
				if (dist[x] == -1) {
					dist[x] = dist[cur] + 1;
					q.add(x);
				}
			}
		}
		adjMatrix[u] = dist;
	}

	static int mod = 998244353;

	static int getGroup(int x, int y, int z) {
		return (adjMatrix[y][x] + adjMatrix[x][z] - adjMatrix[y][z]) / 2;
	}

	static int getDepth(int x, int y, int z) {
		return (adjMatrix[x][z] + adjMatrix[y][z] - adjMatrix[x][y]) / 2;
	}

	public static long solve(int x, int y, int n) {
		int len = adjMatrix[x][y] + 1;
		int[] leader = new int[len];
		for (int z = 0; z < n; z++) {
			int grp = (adjMatrix[y][x] + adjMatrix[x][z] - adjMatrix[y][z]) / 2;
			if (adjMatrix[x][z] + adjMatrix[z][y] == adjMatrix[x][y]) {
				if (leader[grp] > 0)
					return 0;
				leader[grp]++;
			}
			if (((adjMatrix[x][z] + adjMatrix[y][z] - adjMatrix[x][y]) & 1) == 1)
				return 0;

		}

		long ans = 1;
		for (int i = 0; i < n; i++) {
			int curDepth = getDepth(x, y, i);
			int curGroup = getGroup(x, y, i);
			if (curDepth == 0)
				continue;

			int w = 0;
			for (int u : adjList[i]) {
				if (getGroup(x, y, u) == curGroup && getDepth(x, y, u) + 1 == curDepth) {
					w++;
				}
			}
			ans = ans * w % mod;
		}

		return ans;
	}

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);
		PrintWriter pw = new PrintWriter(System.out);
		n = sc.nextInt();
		int m = sc.nextInt();
		adjList = new int[n][];
		adjMatrix = new int[n][];
		int[] deg = new int[n];
		int[][] edge = new int[m][];
		for (int i = 0; i < m; i++) {
			int u = sc.nextInt() - 1;
			int v = sc.nextInt() - 1;
			deg[u]++;
			deg[v]++;
			edge[i] = new int[] { u, v };
		}
		for (int i = 0; i < n; i++) {
			adjList[i] = new int[deg[i]];
		}
		for (int[] a : edge) {
			int u = a[0];
			int v = a[1];
			adjList[u][--deg[u]] = v;
			adjList[v][--deg[v]] = u;
		}
		floyd(n);
		long[][] ans = new long[n][n];
		for (int x = 0; x < n; x++) {
			for (int y = 0; y < n; y++) {
				if (x <= y) {
					long a = solve(x, y, n);
					ans[x][y] = a;
					pw.print(a + " ");
				} else {
					pw.print(ans[y][x] + " ");
				}
			}
			pw.println();
		}
		pw.close();
	}

	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		public Scanner(InputStream s) {
			br = new BufferedReader(new InputStreamReader(s));
		}

		public Scanner(FileReader f) {
			br = new BufferedReader(f);
		}

		public String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		public long nextLong() throws IOException {
			return Long.parseLong(next());
		}

		public double nextDouble() throws IOException {
			return Double.parseDouble(next());
		}

		public int[] nextIntArr(int n) throws IOException {
			int[] arr = new int[n];
			for (int i = 0; i < n; i++) {
				arr[i] = Integer.parseInt(next());
			}
			return arr;
		}

	}

}
