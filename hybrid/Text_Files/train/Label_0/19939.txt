#include <bits/stdc++.h>
using namespace std;
int n;
int a[1001], pos[1001];
int main() {
  ios_base::sync_with_stdio(0);
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
    ;
    if (a[i] == i)
      pos[i] = 0;
    else
      pos[i] = 1;
  }
  int first = 0, last = 0;
  for (int i = 1; i <= n; ++i) {
    if (pos[i] == 0) continue;
    if (first == 0)
      first = i;
    else
      last = i;
  }
  if (first == 0 && last == 0) {
    cout << "0 0";
    return 0;
  }
  int k = last;
  for (int i = first; i <= last; ++i) {
    if (a[i] == k) {
      --k;
      continue;
    }
    cout << "0 0";
    return 0;
  }
  cout << first << " " << last;
  return 0;
}
