def main():
	input()
	arr = list(map(int, input().split()))
	dict = {}
	for i in range(len(arr)):
		item = arr[i]
		while item in dict:
			del dict[item]
			item *= 2
		dict[item] = i
	numbers = [str(x[0]) for x in sorted(dict.items(), key=lambda x: x[1])]
	print(len(numbers))
	print(' '.join(numbers))
main()
