#include <bits/stdc++.h>
using namespace std;
string s, t;
map<char, bool> good;
int main() {
  cin >> s;
  for (char x : s) good[x] = true;
  cin >> s;
  int n, m = s.length(), pos = -29102001;
  for (int i = 0; i < m; ++i)
    if (s[i] == '*') {
      pos = i;
      break;
    }
  cin >> n;
  while (n--) {
    bool ok = true;
    cin >> t;
    int k = t.length();
    if (pos < 0 && k != m) {
      puts("NO");
      continue;
    }
    if (k < m - 1) {
      puts("NO");
      continue;
    }
    for (int i = 0; i < min(abs(pos), m) && ok; ++i) {
      if (s[i] == '?') {
        if (!good[t[i]]) ok = false;
      } else {
        if (s[i] != t[i]) ok = false;
      }
    }
    int rr = k;
    for (int i = m - 1, j = k - 1; i > max(-1, pos) && ok; --i, --j) {
      if (s[i] == '?') {
        if (!good[t[j]]) ok = false;
      } else {
        if (s[i] != t[j]) ok = false;
      }
      rr = j;
    }
    if (ok && pos >= 0)
      for (int i = pos; i < rr && ok; ++i)
        if (good[t[i]]) ok = false;
    puts(ok ? "YES" : "NO");
  }
  return 0;
}
