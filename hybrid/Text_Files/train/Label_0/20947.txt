#include <bits/stdc++.h>
using namespace std;
long long int gcd(long long int a, long long int b) {
  return b ? gcd(b, a % b) : a;
}
long long int lcm(long long int a, long long int b) {
  return a * b / gcd(a, b);
}
const int N = 500;
vector<int> adj[N];
bool vis[N];
int n, m;
void dfs(int v) {
  vis[v] = 1;
  for (auto u : adj[v])
    if (!vis[u]) dfs(u);
}
int cnc() {
  int cntr = 0;
  for (int i = 100; i < n + 100; i++)
    if (!vis[i]) {
      if ((int)adj[i].size() == 0) continue;
      dfs(i);
      cntr++;
    }
  return cntr;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
  cin >> n >> m;
  int cntr = 0;
  for (int i = 0; i < n; i++) {
    int g;
    cin >> g;
    if (g == 0) {
      cntr++;
      continue;
    }
    int v = i + 100;
    for (int j = 0; j < g; j++) {
      int x;
      cin >> x;
      x += 300;
      adj[v].push_back(x);
      adj[x].push_back(v);
    }
  }
  int c = cnc();
  if (c) c--;
  cntr += c;
  cout << cntr << endl;
  return 0;
}
