#include <bits/stdc++.h>
using namespace std;
const long long int N = 200010;
const long long int mod = 1e9 + 7;
long long int power(long long int x, long long int y, long long int p) {
  long long int res = 1;
  x = x % p;
  while (y > 0) {
    if (y & 1) res = (res * x) % p;
    y = y >> 1;
    x = (x * x) % p;
  }
  return res;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int n, p, selected, ind, temp, minq, st;
  long long int ans[N];
  set<long long int> s;
  queue<long long int> q;
  vector<pair<long long int, long long int>> arr;
  cin >> n >> p;
  for (int i = 0; i < n; i++) {
    cin >> temp;
    arr.push_back(make_pair(temp, i));
  }
  sort(arr.begin(), arr.end());
  ind = 0;
  st = 0;
  selected = 0;
  while (selected < n) {
    if (s.empty() && q.empty()) {
      st = arr[ind].first + p;
      ans[arr[ind].second] = st;
      minq = arr[ind].second;
      ind++;
    } else if (!q.empty()) {
      st += p;
      ans[q.front()] = st;
      q.pop();
    } else {
      st += p;
      ans[*s.begin()] = st;
      minq = *s.begin();
      s.erase(s.begin());
    }
    while (ind < n) {
      if (arr[ind].first > st) break;
      if (arr[ind].second > minq)
        s.insert(arr[ind].second);
      else {
        q.push(arr[ind].second);
        minq = arr[ind].second;
      }
      ind++;
    }
    selected++;
  }
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
