#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a[100005], dp[100005], ans = 1;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    dp[i] = 1;
  }
  for (int i = 2; i <= n; i++) {
    if (a[i] > a[i - 1]) dp[i] += dp[i - 1];
    ans = max(ans, dp[i]);
  }
  printf("%d\n", ans);
  return 0;
}
