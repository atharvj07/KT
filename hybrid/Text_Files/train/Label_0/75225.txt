def oneLineArrayPrint(arr):
	print(' '.join([str(x) for x in arr]))

def multiLineArrayPrint(arr):
	print('\n'.join([str(x) for x in arr]))

def multiLineArrayOfArraysPrint(arr):
	print('\n'.join([' '.join([str(x) for x in y]) for y in arr]))

def getCnts(upperBound):
	arr = [int(x) for x in str(upperBound)]
	n = len(arr)
	dp = [[[0 for _ in range(2)] for __ in range(11)] for _a in range(n + 1)]
	dp[n][0][0] = 1
	dp[n][0][1] = 1
	for idx in range(n - 1, -1, -1):
		for digit in range(10):
			for summ in range(11):
				sum2 = digit + summ
				if sum2 <= 10:
					dp[idx][sum2][0] += dp[idx + 1][summ][0]
					if digit < arr[idx]:
						dp[idx][sum2][1] += dp[idx + 1][summ][0]
					elif digit == arr[idx]:
						dp[idx][sum2][1] += dp[idx + 1][summ][1]
	return dp[0][10][1]
main()
