#include <bits/stdc++.h>
#pragma GCC optimize("O3")
using namespace std;
;
const long long maxn = 1e3;
bool M[maxn][maxn];
long long D[maxn];
vector<pair<long long, long long> > V[maxn];
int main() {
  long long q, w, e, r, n, m, k, l, a, b;
  cin >> n >> m;
  for (q = 0; q < m; q++) {
    cin >> a >> b;
    a--;
    b--;
    M[a][b] = M[b][a] = 1;
  }
  for (q = 0; q < n; q++) {
    D[q] = q;
  }
  long long maxf = n;
  for (q = 0; q < n; q++) {
    V[q].push_back(make_pair(q, q));
    for (w = 0; w < q; w++) {
      if (M[q][w]) {
        V[q].push_back(make_pair(q, D[w]));
        V[w].push_back(make_pair(w, maxf));
        D[w] = maxf;
        maxf++;
      }
    }
  }
  for (q = 0; q < n; q++) {
    cout << V[q].size() << endl;
    for (w = 0; w < V[q].size(); w++) {
      cout << V[q][w].first + 1 << " " << V[q][w].second + 1 << endl;
    }
  }
  return 0;
}
