import java.util.*;

public class A {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        StringBuilder sb = new StringBuilder();
        int testCases = sc.nextInt();

        for (int t = 0; t < testCases; t++) {
            long n = sc.nextLong();

            long count = 0;
            int flag = 1;

            while (n != 0) {

                if (flag == 1) {
                    if (n % 2 != 0) {
                        count = count + 1;
                        n = n - 1;
                    } else if (n % 2 == 0 && ((n / 2) % 2 != 0 || n == 4)) {
                        count = count + n / 2;
                        n = n / 2;
                    } else {
                        count = count + 1;
                        n = n - 1;
                    }
                } else {

                    if (n % 2 != 0) {
                        n = n - 1;
                    } else if (n % 2 == 0 && ((n / 2) % 2 != 0 || n == 4)) {
                        n = n / 2;
                    } else {
                        n = n - 1;
                    }

                }

                flag = flag ^ 1;

            }

            sb.append(count).append("\n");

        }

        System.out.print(sb);
    }
}
