#include <bits/stdc++.h>
using namespace std;
#define REP(asd,fgh) for(int asd=0; asd<fgh; asd++)

int main(void){
  vector< vector<string> > defeat;
  string handlist[] = {"Rock", "Fire", "Scissors", "Snake", "Human", "Tree", "Wolf", "Sponge", "Paper", "Air", "Water", "Dragon", "Devil", "Lightning", "Gun"};
  map<string, int> hand;
  REP(i, 15){
    hand[handlist[i]] = i;
  }
  
  int N;
  while(cin >> N, N){
    vector<string> players(N);
    string ans = "Draw";
    REP(i, N){
      cin >> players[i];
    }
    string same = players[0];
    bool draw = true;
    REP(i, N){
      if(players[i] != same) draw = false;
    }
    if(draw){
      cout << ans << endl;
      continue;
    }
    REP(i, N){
      bool flg = true;
      REP(j, N){
	if(i == j) continue;
	if(!(((hand[players[j]] - hand[players[i]])+15)%15 <= 7)){
	  flg = false;
	}
      }
      if(flg)
	ans = players[i];
    }
    cout << ans << endl;
  }

  return 0;
}
