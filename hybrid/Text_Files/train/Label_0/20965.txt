#include <bits/stdc++.h>
using namespace std;
char S[10];
int M[10][10][10];
int main(int argc, char **argv) {
  ios::sync_with_stdio(false);
  for (int i = 0; i < 8; i++) {
    scanf("%s", S);
    for (int j = 0; j < 8; j++) {
      if (S[j] == 'S') {
        M[0][i][j] = 1;
        M[1][i][j] = 1;
        for (int k = 1; i + k < 8; k++) {
          M[k][i + k][j] = 1;
          M[k + 1][i + k][j] = 1;
        }
      }
    }
  }
  queue<pair<int, pair<int, int> > > q;
  q.push(make_pair(0, make_pair(7, 0)));
  int ok = 0;
  while (q.size()) {
    auto p = q.front();
    q.pop();
    int k = p.first;
    int x = p.second.first;
    int y = p.second.second;
    if (k == 8 || x == 0 && y == 7) {
      ok = 1;
      break;
    }
    for (int i = max(0, x - 1); i <= min(7, x + 1); i++) {
      for (int j = max(0, y - 1); j <= min(7, y + 1); j++) {
        if (M[k + 1][i][j] == 0) q.push(make_pair(k + 1, make_pair(i, j)));
      }
    }
  }
  if (ok)
    printf("WIN\n");
  else
    printf("LOSE\n");
  return 0;
}
