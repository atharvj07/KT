#include <bits/stdc++.h>
using namespace std;
const int N = 100010;
long long n, d, b, a[N], Ans, sum[N], cnt1, cnt2;
int main() {
  cin >> n >> d >> b;
  for (int i = 1; i <= n; i++) {
    scanf("%I64d", &a[i]);
    sum[i] = sum[i - 1] + a[i];
  }
  for (int i = 1; i <= n / 2; i++) {
    long long x = sum[min(n, i + i * d)] - cnt1 * b;
    if (x >= b) cnt1++;
    x = sum[n] - sum[max(0LL, n - i - i * d)] - cnt2 * b;
    if (x >= b) cnt2++;
  }
  Ans = n / 2 - min(cnt1, cnt2);
  cout << Ans << endl;
  return 0;
}
