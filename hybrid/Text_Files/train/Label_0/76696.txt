import math

def get_n():
	return int(input())

def get_int_vector():
	return [int(x) for x in input().split()]

def list2string(list):
	result = []
	for i in list:
		result.append(str(i))
	return ':'.join(result)

def string2vector(string):
	return [int(x) for x in string.split(':')]
(b_x, b_y) = get_int_vector()
(e_x, e_y) = get_int_vector()
n = get_n()
result = 0
for _ in range(n):
	(a_i, b_i, c_i) = get_int_vector()
	if (a_i * b_x + b_i * b_y + c_i) * (a_i * e_x + b_i * e_y + c_i) < 0:
		result += 1
print(result)
