def interval(d, h):
	if d == h:
		vysledek = d
	elif not intervaly[d]:
		vysledek = d
		intervaly[d + 1].remove(h - d - 1)
		zasobnik.append((d + 1, h))
	elif h - d - 1 in intervaly[d]:
		vysledek = intervaly[d].pop(0) + d + 1
		zasobnik.append((d, h - 1))
	else:
		vysledek = intervaly[d].pop(0) + d + 1
		intervaly[vysledek + 1].remove(h - vysledek - 1)
		zasobnik.append((d, vysledek - 1))
		zasobnik.append((vysledek + 1, h))
	return vysledek
poctest = int(input())
for i in range(poctest):
	intervaly = dict()
	n = int(input())
	for j in range(n):
		(l, r) = map(int, input().split())
		if l in intervaly:
			intervaly[l].append(r - l)
		else:
			intervaly[l] = [r - l]
	for k in intervaly.keys():
		intervaly[k].sort(reverse=True)
	zasobnik = [(1, n)]
	intervaly[1].remove(n - 1)
	while zasobnik:
		(left, right) = zasobnik.pop()
		result = interval(left, right)
		print(left, right, result)
	print()
