#include <bits/stdc++.h>
using namespace std;
const long long int MOD = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  long long int test = 1;
  while (test--) {
    long long int n, i, j;
    cin >> n;
    long long int a[n];
    for (i = 0; i < n; i++) cin >> a[i];
    stack<long long int> q;
    long long int ans = 0;
    q.push(a[0]);
    for (i = 1; i < n; i++) {
      while (q.size()) {
        if (q.top() < a[i]) {
          ans = max(ans, q.top() ^ a[i]);
          q.pop();
        } else
          break;
      }
      q.push(a[i]);
    }
    while (q.size()) q.pop();
    reverse(a, a + n);
    q.push(a[0]);
    for (i = 1; i < n; i++) {
      while (q.size()) {
        if (q.top() < a[i]) {
          ans = max(ans, q.top() ^ a[i]);
          q.pop();
        } else
          break;
      }
      q.push(a[i]);
    }
    cout << ans;
  }
  cerr << "Time Taken:" << (float)clock() / CLOCKS_PER_SEC * 1000 << "ms"
       << "\n";
  return 0;
}
