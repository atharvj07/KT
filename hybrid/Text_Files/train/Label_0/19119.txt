#include <iostream>
#include <cmath>
#include <queue>
#include <algorithm>
#include <vector>

using namespace std;

struct vec{
    int x,y;
    vec(int x,int y){
        this->x=x;this->y=y;
    }
    long long int cross(vec b){
        return x*b.y-y*b.x;
    }
    vec operator-(vec b){
        return vec(x-b.x,y-b.y);
    }
    vec operator+(vec b){
        return vec(x+b.x,y+b.y);
    }
    double norm(){
        return hypot(x,y);
    }
};

bool isCrossing(vec& v1, vec& v2, vec& v3, vec& v4){
    return (v2-v1).cross(v3-v1)*(v2-v1).cross(v4-v1)<0&&(v4-v3).cross(v1-v3)*(v4-v3).cross(v2-v3)<0;
}

int main(){
    int NA,NB,x,y;
    cin>>NA>>NB;
    vector<vec> A,B;
    for(int i=0;i<NA;i++){
        cin>>x>>y;
        A.emplace_back(x,y);
    }
    for(int i=0;i<NB;i++){
        cin>>x>>y;
        B.emplace_back(x,y);
    }
    double EA[NA][NA];
    double EB[NB][NB];
    fill(EA[0],EA[NA],0);
    fill(EB[0],EB[NB],0);
    for(int i=0;i<NA;i++){
        for(int j=i+1;j<NA;j++){
            if(!isCrossing(A[i],A[j],B[0],B[1])) EA[i][j]=EA[j][i]=(A[i]-A[j]).norm();
        }
    }
    for(int i=0;i<NB;i++){
        for(int j=i+1;j<NB;j++){
            if(!isCrossing(B[i],B[j],A[0],A[1])) EB[i][j]=EB[j][i]=(B[i]-B[j]).norm();
        }
    }
    double Amin[NA],Bmin[NB];
    fill(Amin,Amin+NA,1e10);
    fill(Bmin,Bmin+NB,1e10);
    double Acost,Bcost;
    int s=0,end=1;
    typedef pair<double, int> P;
    priority_queue<P,vector<P>,greater<P>> que;
    Amin[s]=0;
    que.push(P(0,s));
    while(!que.empty()){
        P p=que.top(); que.pop();
        int v=p.second;
        double cost=p.first;
        if(cost>Amin[v])continue;
        for(int i=0;i<NA;i++){
            if(EA[i][v]!=0&&cost+EA[i][v]<Amin[i]){
                Amin[i]=cost+EA[i][v];
                que.emplace(Amin[i],i);
            }
        }
    }
    if(Amin[end]!=1e10){
        Acost=Amin[end]+(B[s]-B[end]).norm();
    }else{
        Acost=1e10;
    }
    Bmin[s]=0;
    que.push(P(0,s));
    while(!que.empty()){
        P p=que.top(); que.pop();
        int v=p.second;
        double cost=p.first;
        if(cost>Bmin[v])continue;
        for(int i=0;i<NB;i++){
            if(EB[i][v]!=0&&cost+EB[i][v]<Bmin[i]){
                Bmin[i]=cost+EB[i][v];
                que.emplace(Bmin[i],i);
            }
        }
    }
    if(Bmin[end]!=1e10){
        Bcost=Bmin[end]+(A[s]-A[end]).norm();
    }else{
        Bcost=1e10;
    }
    if(min(Acost,Bcost)==1e10){
        cout<<-1<<endl;
    }else{
        cout.precision(12);
        cout << fixed;
        cout<<min(Acost,Bcost)<<endl;
    }
    return 0;
}
