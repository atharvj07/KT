#include <bits/stdc++.h>
using namespace std;
unsigned short n, m, k;
vector<pair<unsigned short, unsigned short> > v[10001];
bool flag;
unsigned short dijkstra(unsigned short s) {
  unsigned short dist[10001];
  set<pair<unsigned short, unsigned short> > pq;
  pq.insert({0, s});
  pair<unsigned short, unsigned short> node;
  unsigned short i, u;
  for (i = 0; i < 10001; i++) dist[i] = 50000;
  dist[s] = 0;
  if (flag) k--;
  while (pq.size()) {
    node = *pq.begin();
    pq.erase(pq.begin());
    u = node.second;
    if (u == k) break;
    for (i = 0; i < v[u].size(); i++) {
      if (dist[v[u][i].first] > v[u][i].second + dist[u]) {
        pq.erase({dist[v[u][i].first], v[u][i].first});
        dist[v[u][i].first] = v[u][i].second + dist[u];
        pq.insert({dist[v[u][i].first], v[u][i].first});
      }
    }
  }
  if (dist[k] == 50000) return 50000;
  return dist[k];
}
void buildGraph(vector<pair<unsigned short, unsigned short> >& points) {
  unsigned short i, j, d, x1, y1, x2, y2, x, y;
  for (i = 0; i < k; i++) {
    x1 = points[i].first;
    y1 = points[i].second;
    for (j = 0; j < k; j++) {
      if (i == j) continue;
      x2 = points[j].first;
      y2 = points[j].second;
      if (abs(x1 - x2) > 2 && abs(y1 - y2) > 2) continue;
      if ((x1 == x2 && abs(y1 - y2) == 1) || (y1 == y2 && abs(x1 - x2) == 1))
        d = 0;
      else
        d = 1;
      v[i].push_back({j, d});
    }
  }
  if (!flag) {
    for (i = 0; i < k; i++) {
      x1 = points[i].first;
      y1 = points[i].second;
      if (abs(y1 - m) < 2 || abs(x1 - n) < 2) {
        v[i].push_back({k, 1});
      }
    }
  }
  points.clear();
}
int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  unsigned short src, i, j, x, y, res;
  vector<pair<unsigned short, unsigned short> > points;
  cin >> n >> m >> k;
  flag = false;
  for (i = 0; i < k; i++) {
    cin >> x >> y;
    if (x == 1 && y == 1) src = i;
    if (x == n && y == m) flag = true;
    points.push_back({x, y});
  }
  buildGraph(points);
  points.resize(0);
  res = dijkstra(src);
  if (res == 50000)
    cout << -1 << endl;
  else
    cout << res << endl;
  return 0;
}
