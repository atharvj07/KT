def color(rbs, bin):
	opened = 0
	for i in range(len(rbs)):
		if rbs[i] == ')' and opened % 2 == 1:
			bin[i] = '1'
			opened -= 1
		elif rbs[i] == '(' and opened % 2 == 0:
			bin[i] = '1'
			opened += 1
		elif rbs[i] == '(':
			opened += 1
		elif rbs[i] == ')':
			opened -= 1
n = int(input())
bin = ['0'] * n
rbs = input()
color(rbs, bin)
print(''.join(bin))
