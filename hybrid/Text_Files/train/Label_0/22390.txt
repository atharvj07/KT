#include <bits/stdc++.h>
using namespace std;
bool prime(int x) {
  if (x <= 1) return false;
  for (int i = 2; i * i <= x; i++) {
    if (x % i == 0) return false;
  }
  return true;
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  cin >> n;
  int arr[100005] = {0};
  int k = 1;
  for (int i = 2; i <= n; i++) {
    if (!arr[i]) {
      for (int j = 1; j * i <= n; j++) arr[i * j] = k;
      k++;
    }
    cout << arr[i] << " ";
  }
  return 0;
}
