import java.util.Scanner;

public class A680 {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] stat = new int[101];
        int sum = 0;
        for (int i=0; i<5; i++) {
            int value = in.nextInt();
            sum += value;
            stat[value]++;
        }
        int minSum = sum;
        for (int i=1; i<=100; i++) {
            if (stat[i] > 2) {
                minSum = Math.min(minSum, sum - 3*i);
            } else if (stat[i] > 1) {
                minSum = Math.min(minSum, sum - 2*i);
            }
        }
        System.out.println(minSum);
    }

}
