n = int(input())
p = input().rstrip().split(' ')
i = 0
j = len(p) - 1
l = []
w = []
T = min(int(p[0]), int(p[len(p) - 1]))
if T == int(p[0]):
	l.append(int(p[0]))
	w.append('L')
	i += 1
elif int(p[len(p) - 1]):
	l.append(int(p[len(p) - 1]))
	w.append('R')
	j -= 1
while i < j:
	if int(p[i]) > int(l[len(l) - 1]) and int(p[i]) < int(p[j]):
		l.append(int(p[i]))
		w.append('L')
		i += 1
	elif int(p[j]) > int(l[len(l) - 1]) and int(p[j]) < int(p[i]):
		l.append(int(p[j]))
		w.append('R')
		j -= 1
	elif int(p[i]) > int(l[len(l) - 1]):
		l.append(int(p[i]))
		w.append('L')
		i += 1
	elif int(p[j]) > int(l[len(l) - 1]):
		l.append(int(p[j]))
		w.append('R')
		j -= 1
	else:
		break
if len(w) == n - 1:
	if int(p[i]) > int(l[len(l) - 1]):
		w.append('R')
print(len(w))
print(''.join(w))
