from collections import OrderedDict
ordered = {}
sms_numbers = int(input())
for i in range(0, sms_numbers):
	ordered[input()] = sms_numbers - i
OrderedDict(sorted(ordered.items(), key=lambda t: t[1]))
print('\n'.join((item for item in OrderedDict(sorted(ordered.items(), key=lambda t: t[1])).keys())))
