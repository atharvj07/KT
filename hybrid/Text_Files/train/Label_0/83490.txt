s = input()
n = len(s)
(a, b) = ([], [])
for i in range(n):
	if s[i] == '(':
		a.append(i)
	if s[i] == ')':
		b.append(i)
ans = []
(cnt_a, cnt_b) = (0, len(b) - 1)
while cnt_a < len(a) and cnt_b >= 0:
	if a[cnt_a] > b[cnt_b]:
		break
	ans.append(a[cnt_a])
	ans.append(b[cnt_b])
	cnt_a += 1
	cnt_b -= 1
ans.sort()
if len(ans) == 0:
	print(0)
else:
	print(1)
	print(len(ans))
	print(' '.join([str(i + 1) for i in ans]))
