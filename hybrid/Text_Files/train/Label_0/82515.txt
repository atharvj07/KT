def do():
	q = int(input())
	for _ in range(q):
		n = int(input())
		s = input()
		if s.find('<') == -1:
			print(n)
			continue
		if s.find('>') == -1:
			print(n)
			continue
		s = s[-1] + s + s[0]
		res = 0
		for i in range(1, n + 1):
			if s[i - 1] == '-' or s[i] == '-':
				res += 1
		print(res)
do()
