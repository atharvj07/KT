Q = int(input())
from copy import copy, deepcopy

def solve():
	[N, M] = list(map(int, input().split()))
	grid = []
	for _ in range(N):
		grid.append(list(input()))
	row_count = []
	column_count = []
	for row in grid:
		x = 0
		for g in row:
			if g == '*':
				x += 1
		row_count.append(x)
	for m in range(M):
		x = 0
		for n in range(N):
			if grid[n][m] == '*':
				x += 1
		column_count.append(x)
	ans = N * M
	for n in range(N):
		for m in range(M):
			ans_ = N + M - (row_count[n] + column_count[m])
			if grid[n][m] == '.':
				ans_ -= 1
			ans = min(ans, ans_)
	return ans
for _ in range(Q):
	print(solve())
