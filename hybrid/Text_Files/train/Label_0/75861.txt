import sys
import collections
(n, m) = [int(x) for x in sys.stdin.readline().split(' ')]
adj = collections.defaultdict(list)
vis = [0] * (n + 1)
for i in range(m):
	(u, v) = [int(x) for x in sys.stdin.readline().split(' ')]
	adj[u].append(v)
	adj[v].append(u)
maxu = -1
ans = 0
for i in range(1, n + 1):
	if not vis[i]:
		if i < maxu:
			ans += 1
		s = [i]
		while len(s) > 0:
			u = s.pop()
			maxu = max(u, maxu)
			vis[u] = True
			for v in adj[u]:
				if not vis[v]:
					s.append(v)
					vis[v] = True
print(ans)
