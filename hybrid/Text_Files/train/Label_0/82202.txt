n = int(input())
a = list(map(int, input().split()))
a.sort()
res = int(1000000000.0 * n)
lim = res + int(1000000000.0 + 5)
for c in range(1, int(1000000000.0)):
	r = 0
	z = 1
	ok = True
	for i in range(n):
		if z > lim:
			ok = False
			break
		r += abs(a[i] - z)
		z *= c
	if not ok:
		break
	res = min(res, r)
print(res)
