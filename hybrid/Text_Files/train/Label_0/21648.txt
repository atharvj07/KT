#include <bits/stdc++.h>
using namespace std;
int n;
string s[30];
map<string, pair<int, pair<int, int> > > mp;
pair<int, pair<int, int> > solve(string &str) {
  pair<int, pair<int, int> > res = make_pair(-1, make_pair(0, 0));
  if (mp.find(str) != mp.end()) return mp[str];
  for (char c = 'a'; c <= 'z'; c++)
    for (int i = 0; i <= 1; i++) {
      int score = 0;
      int cnt = 0, maxn = 0, sum = 0;
      string tmp = (i == 0) ? (str + c) : (c + str);
      for (int j = 0; j < n; j++)
        if (s[j].find(tmp) != string::npos) cnt++;
      if (cnt == 0) continue;
      for (int j = 0; j < (int)tmp.size(); j++) {
        sum += tmp[j] - 'a' + 1;
        maxn = max(maxn, tmp[j] - 'a' + 1);
      }
      score = sum * maxn + cnt;
      pair<int, pair<int, int> > next = solve(tmp);
      res = max(res, make_pair(next.first * (-1),
                               make_pair(score - next.second.second,
                                         -next.second.first)));
    }
  mp[str] = res;
  return res;
}
int main() {
  pair<int, pair<int, int> > res;
  string ori = "";
  cin >> n;
  for (int i = 0; i < n; i++) cin >> s[i];
  res = solve(ori);
  if (res.first == 1)
    cout << "First\n";
  else
    cout << "Second\n";
  cout << res.second.first << " " << -res.second.second << endl;
  return 0;
}
