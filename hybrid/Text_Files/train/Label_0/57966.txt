import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Comparator;
import java.util.TreeMap;
import java.util.TreeSet;


public class CF209C {
    static int[] uf;
    public static void mix(int a, int b) {
        uf[find(a)] = find(b);
    }
    public static int find(int x) {
        if(uf[x] == x) return x;
        return uf[x] = find(uf[x]);
    }
    public static int i(String s) { return Integer.parseInt(s); }
    public static void add(TreeMap<Integer, Integer> X, int x) {
        X.put(x, X.containsKey(x) ? X.get(x)+1 : 1);
    }
    public static int last(TreeMap<Integer, Integer> X) {
        int k = X.lastKey();
        int v = X.get(k);
        if(v == 1) {
            X.remove(k);
        } else {
            X.put(k, v-1);
        }
        return k;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] arr = in.readLine().split(" ");
        int n = i(arr[0]);
        int m = i(arr[1]);
        uf = new int[n];
        int[] d = new int[n];
        d[0] = 2;
        for(int i=0; i<n; i++)
            uf[i] = i;
        boolean[] S = new boolean[n];
        int[] O = new int[n];

        for(int i=0; i<m; i++) {
            arr = in.readLine().split(" ");
            int x = i(arr[0])-1;
            int y = i(arr[1])-1;
            d[x]++;
            d[y]++;
            mix(x,y);
        }

        for(int i=0; i<n; i++) {
            if(d[i] == 0) continue;
           // System.out.println(i+" "+find(i));
            O[find(i)] += (d[i]%2==1)?1:0;
            S[find(i)] = true;
        }
        int cnt = 0;
        TreeMap<Integer, Integer> X = new TreeMap<Integer, Integer>();
        for(int i=0; i<n; i++) {
            if(S[i]) {
                add(X, O[i]);
                cnt++;
            }
        }
        int ans = 0;
        while(cnt > 1) {
            cnt--;
            ans++;
            int a = last(X);
            int b = last(X);
            if(a>0 && b>0) add(X, a+b-2);
            else if(a>0 || b>0) add(X, a+b);
            else add(X, 2);
        }
        ans += X.keySet().iterator().next()/2;
        System.out.println(ans);
    }
}
