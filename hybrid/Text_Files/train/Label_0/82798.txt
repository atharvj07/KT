params = []
params.append(int(input()))
for i in range(1, params[0] + 1):
	params.append(input().split(' '))
	for j in range(0, 2):
		params[i][j] = int(params[i][j])
params.append(int(input()))
mark = params[params[0] + 1]
for i in range(1, params[0] + 1):
	if mark in range(params[i][0], params[i][1] + 1):
		print(params[0] - i + 1)
		break
