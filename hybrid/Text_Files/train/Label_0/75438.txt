def srk(s, n):
	z = n
	if s == s[::-1]:
		return 1
	for i in range(z // 2):
		if s[i] == s[n - i - 1]:
			continue
		elif s[i] != s[n - i - 1]:
			if abs(ord(s[i]) - ord(s[n - i - 1])) == 2:
				continue
			else:
				return 0
	if n == 2:
		if abs(ord(s[0]) - ord(s[1])) == 2:
			return 1
		else:
			return 0
	return 1
t = int(input())
for _ in range(t):
	n = int(input())
	s = list(input())
	p = srk(s, n)
	if p == 1:
		print('YES')
	else:
		print('NO')
