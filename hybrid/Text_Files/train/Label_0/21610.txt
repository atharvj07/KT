#include<iostream>
#include<string>
#include<algorithm>
using namespace std;
typedef struct {
	int m[6];
} Dice;

Dice d;
string s;

void roll(char c, Dice *d) {
	switch (c) {
	case 'N':
		swap(d->m[0], d->m[1]);
		swap(d->m[1], d->m[5]);
		swap(d->m[5], d->m[4]);
		break;
	case 'S':
		swap(d->m[0], d->m[4]);
		swap(d->m[4], d->m[5]);
		swap(d->m[5], d->m[1]);
		break;
	case 'E':
		swap(d->m[0], d->m[3]);
		swap(d->m[3], d->m[5]);
		swap(d->m[5], d->m[2]);
		break;
	case 'W':
		swap(d->m[0], d->m[2]);
		swap(d->m[2], d->m[5]);
		swap(d->m[5], d->m[3]);
		break;
	}
}
int main() {
	for (int i = 0; i < 6; i++) cin >> d.m[i];
	cin >> s;
	for (int i = 0; i < s.size(); i++) roll(s[i], &d);
	cout << d.m[0] << endl;
}
