import sys
import math

def fn(n, a):
	count = 0
	cnt = [0] * 3
	for i in range(n):
		cnt[a[i] % 3] = cnt[a[i] % 3] + 1
	if cnt[0] == cnt[1] == cnt[2]:
		return 0
	for i in range(3):
		m1 = max(cnt)
		j = cnt.index(m1)
		if m1 == n / 3:
			break
		if j == 2:
			cnt[0] = cnt[0] + m1 - n // 3
			cnt[2] = n // 3
			count = count + m1 - n // 3
		else:
			cnt[j] = n // 3
			cnt[j + 1] = cnt[j + 1] + m1 - n // 3
			count = count + m1 - n // 3
	return count
input = sys.stdin.read()
data = list(map(int, input.split()))
t = data[0]
l = 0
for i in range(t):
	n = data[1 + l]
	a = data[l + 2:l + 2 + n]
	l = l + n + 1
	print(fn(n, a))
