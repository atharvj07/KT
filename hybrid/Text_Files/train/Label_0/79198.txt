import heapq
towerindex = 1

def main():
	global towerindex
	t = int(input())
	for _ in range(1, t + 1):
		towerindex = 1
		(n, m, x) = map(int, input().split())
		hs = list(map(int, input().split()))
		blocks = []
		for (index, h) in enumerate(hs):
			blocks.append(Block(h, index))
		blocks = sorted(blocks, reverse=True)
		towers = codeforces(m, x, blocks)
		if towers:
			print('YES')
			blocks = []
			for tower in towers:
				for bl in tower.blocks:
					bl.towerindex = tower.index
					blocks.append(bl)
			blocks = sorted(blocks, key=lambda x: x.index)
			print(' '.join(list(map(lambda x: str(x.towerindex), blocks))))
		else:
			print('NO')

def codeforces(m, x, blocks):
	towers = []
	for i in range(m):
		heapq.heappush(towers, tower(blocks.pop(0)))
	while blocks:
		tow = heapq.heappop(towers)
		tow.add(blocks.pop(0))
		heapq.heappush(towers, tow)
	return towers

class tower:

	def __init__(self, block):
		global towerindex
		self.height = block.height
		self.blocks = [block]
		self.index = towerindex
		towerindex += 1

	def __lt__(self, other):
		return self.height < other.height

	def __repr__(self):
		return str(self.index)

	def add(self, block):
		self.height += block.height
		self.blocks.append(block)

class Block:

	def __init__(self, h, index):
		self.height = h
		self.index = index

	def __repr__(self):
		return str(self.index)

	def __lt__(self, other):
		return self.height < other.height
main()
