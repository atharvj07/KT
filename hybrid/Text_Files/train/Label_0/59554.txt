

import java.util.Scanner;

public class C {
    static int[]s;
    static int[][]a;
    static boolean[][][]calculated;
    static int L;
    static int[][][]d;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String ss = sc.next();
        L = ss.length();
        s = new int[L+1];
        for (int i = 1; i <= L; i++) {
            s[i] = ss.codePointAt(i-1)-96;
        }
        int k = sc.nextInt();
        int n = sc.nextInt();
        a = new int[30][30];
        d = new int[101][30][101];
        calculated = new boolean[101][30][101];
        for (int i = 1; i <= n; i++) {
            int k1 = sc.next().codePointAt(0)-96;
            int k2 = sc.next().codePointAt(0)-96;
            a[k1][k2] = sc.nextInt();
        }
        System.out.println(calc(1, 27, k));
    }

    private static int calc(int ind, int p, int k) {
        if (ind > L || k < 0)
            return 0;
        if (calculated[ind][p][k])
            return d[ind][p][k];
        int max = calc(ind+1, s[ind], k)+a[p][s[ind]];
        for (int i = 1; i <= 26; i++) {
            if (i != s[ind] && k > 0) {
                max = Math.max(max, calc(ind+1, i, k-1)+a[p][i]);
            }
        }
        calculated[ind][p][k] = true;
        d[ind][p][k] = max;
        return max;
    }
}
