for _ in range(int(input())):
	n = int(input())
	s = [int(a) for a in input()]
	cnt = sum(s)
	if s[0] and s[-1] and n & 1 ^ 1 and cnt & 1 ^ 1:
		(a, b, p) = ([0] * n, [0] * n, 1)
		for (i, x) in enumerate(s):
			if x:
				if cnt > 0:
					a[i] = b[i] = 1
					cnt -= 2
			else:
				a[i] = p
				p ^= 1
				b[i] = p
		print('YES', ''.join(['(' if x else ')' for x in a]), ''.join(['(' if x else ')' for x in b]), sep='\n')
	else:
		print('NO')
