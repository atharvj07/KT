#include <bits/stdc++.h>
using namespace std;

#define for_(i,a,b) for(int i=(a);i<(b);++i)

vector< string > reply;
vector< vector< int > > tree;

void rec(int v, int d) {
	for_(i,0,d) cout << ".";
	cout << reply[v] << endl;
	
	int m = tree[v].size();
	for_(i,0,m) rec(tree[v][i], d + 1);
}

int main() {
	int n;
	cin >> n;
	
	reply.assign(n, string());
	tree.assign(n, vector< int >());
	
	for_(i,0,n) {
		int k;
		cin >> k >> reply[i];
		
		--k;
		if (k >= 0) tree[k].push_back(i);
	}
	
	rec(0, 0);
}
