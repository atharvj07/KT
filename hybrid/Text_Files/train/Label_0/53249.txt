import java.io.IOException;
import java.util.ArrayList;
import java.util.Scanner;

public class ProblemC {
	static Scanner sc = new Scanner(System.in);
	static int n, ans = -1, color[];
	static ArrayList<Integer> adjList[];
	static int parent[];

	static boolean check(int idx) {
		parent = new int[n];
		parent[idx] = idx;
		for (int i = 0; i < adjList[idx].size(); i++) {
			int u = adjList[idx].get(i);
			parent[u] = idx;
			if (!dfs(u, color[u]))
				return false;
		}
		return true;
	}

	static boolean dfs(int idx, int c) {
		if (color[idx] != c)
			return false;
		for (int i = 0; i < adjList[idx].size(); i++) {
			int u = adjList[idx].get(i);
			if (u != parent[idx]) {
				parent[u] = idx;
				if (!dfs(u, c))
					return false;
			}
		}
		return true;
	}

	public static void main(String[] args) throws IOException {
		n = sc.nextInt();
		if (n == 2) {
			System.out.println("YES\n1");
			return;
		}
		adjList = new ArrayList[n];
		for (int i = 0; i < n; i++)
			adjList[i] = new ArrayList<>();
		for (int i = 0; i < n - 1; i++) {
			int x = sc.nextInt() - 1, y = sc.nextInt() - 1;
			adjList[x].add(y);
			adjList[y].add(x);
		}
		color = new int[n];
		boolean atleastOne = false;
		for (int i = 0; i < n; i++) {
			color[i] = sc.nextInt();
			if (i != 0 && color[i] != color[i - 1])
				atleastOne = true;
		}
		if (!atleastOne) {
			System.out.println("YES\n1");
			return ;
		}

		boolean done[] = new boolean[n];

		for (int i = 0; i < n; i++)
			if (!done[i]) {
				int diff = 0;
				for (int j = 0; j < adjList[i].size(); j++)
					if (color[i] != color[adjList[i].get(j)])
						diff++;

				if (diff > 1) {
					if (ans == -1 && check(i))
						ans = i + 1;
					else {
						ans = -1;
						break;
					}
				} else if (diff == 1) {
					int diffOne = 0;
					for (int j = 0; j < adjList[i].size(); j++)
						if (color[i] != color[adjList[i].get(j)]) {
							diffOne = adjList[i].get(j);
							break;
						}
					if (!done[diffOne]) {
						boolean x1 = check(i), x2 = check(diffOne);
						if (!(x1 || x2)) {
							ans = -1;
							break;
						} else if (x1) {
							done[i] = true;
							ans = i + 1;
						} else {
							done[diffOne] = true;
							ans = diffOne + 1;
						}
					}
				}
			}
		if (ans == -1)
			System.out.println("NO");
		else
			System.out.println("YES\n" + ans);
	}
}
