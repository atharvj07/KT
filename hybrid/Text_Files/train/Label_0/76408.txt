n = input()
n = int(n)
array = input()
array = [int(x) for x in array.split(' ')]
maximum = 0
output = ''
left = 0
right = len(array) - 1
while left <= right:
	if array[left] > maximum and array[right] > maximum:
		if array[left] <= array[right]:
			maximum = array[left]
			output += 'L'
			left += 1
		else:
			maximum = array[right]
			output += 'R'
			right -= 1
	elif array[left] > maximum:
		maximum = array[left]
		output += 'L'
		left += 1
	elif array[right] > maximum:
		maximum = array[right]
		output += 'R'
		right -= 1
	else:
		break
print(len(output))
print(output)
