#include <bits/stdc++.h>
using namespace std;
int main() {
  vector<string> names;
  for (char i = 'A'; i <= 'Z'; i++) {
    for (char j = 'a'; j <= 'z'; j++) {
      string p = "";
      p += i;
      p += j;
      names.push_back(p);
    }
  }
  int n, k;
  cin >> n >> k;
  vector<string> ans;
  for (int i = 0; i < k - 1; i++) ans.push_back(names[i]);
  for (int i = 0; i < n - k + 1; i++) {
    string res;
    cin >> res;
    set<string> prevs;
    int cnt = k - 1;
    for (int i = ans.size() - 1; cnt > 0; i--, cnt--) {
      prevs.insert(ans[i]);
    }
    if (res == "YES") {
      for (auto x : names) {
        if (prevs.find(x) == prevs.end()) {
          ans.push_back(x);
          break;
        }
      }
    } else {
      ans.push_back(ans[ans.size() - k + 1]);
    }
  }
  for (auto x : ans) cout << x << " ";
  cout << "\n";
  return 0;
}
