#include <bits/stdc++.h>
using namespace std;
const long long INF = 1LL << 61;
const double EPS = 1e-8;
const double PI = acos(-1);
pair<int, int> p[200000];
int e[200000];
int n, k;
long long func(int score) {
  if (score > n) return INF;
  vector<int> v, u;
  int ls = 0;
  for (int i = 0; i < (int)n; ++i) {
    if (p[i].first == score || p[i].first == score - 1) {
      v.push_back(p[i].second);
    } else {
      u.push_back(p[i].second);
    }
    if (p[i].first < score - 1) ls++;
  }
  long long cost = 0;
  if (ls >= n - k) {
    sort(e, e + n);
    for (int i = 0; i < (int)score; ++i) {
      cost += e[i];
    }
    return cost;
  }
  int need = n - k - ls;
  if (need > v.size()) return INF;
  if (need > score) return INF;
  sort((v).begin(), (v).end());
  for (int i = 0; i < (int)need; ++i) {
    cost += v[i];
  }
  for (int i = need; i < v.size(); ++i) u.push_back(v[i]);
  sort((u).begin(), (u).end());
  for (int i = 0; i < (int)score - need; ++i) {
    cost += u[i];
  }
  return cost;
}
int main() {
  while (cin >> n >> k) {
    for (int i = 0; i < (int)n; ++i) {
      cin >> p[i].first >> p[i].second;
      e[i] = p[i].second;
    }
    k--;
    if (k == n) {
      cout << 0 << endl;
      continue;
    }
    sort(p, p + n, greater<pair<int, int> >());
    int score = p[k].first;
    long long res = INF;
    for (int i = 0; i < (int)3; ++i)
      (res > (func(score + i)) ? (res = func(score + i), 1) : 0);
    if (res == INF)
      cout << -1 << endl;
    else
      cout << res << endl;
  }
}
