import sys
n = int(sys.stdin.readline())
stack = tuple((int(i) for i in sys.stdin.readline().split()))
order = tuple((int(i) for i in sys.stdin.readline().split()))
start = 0
backpack = set()
results = []
for book in order:
	if book in backpack:
		results.append(0)
	else:
		for i in range(start, len(stack)):
			backpack.add(stack[i])
			if stack[i] == book:
				results.append(i - start + 1)
				start = i + 1
				break
print(' '.join((str(s) for s in results)))
