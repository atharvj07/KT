import math

def main():
	a = input()
	a = a.split()
	n = int(a[0])
	w = int(a[1])
	b = input()
	b = b.split()
	arr2 = []
	arr = []
	add = 0
	for i in range(0, n):
		b[i] = int(b[i])
		arr2.append(b[i])
		if b[i] % 2 == 0:
			add += b[i] / 2
			arr.append(b[i] / 2)
		else:
			add += math.floor(b[i] / 2) + 1
			arr.append(b[i] / float(2) + 1)
	arr2.sort()
	left = w - add
	i = n - 1
	if left < 0:
		print(-1)
	else:
		while left > 0:
			if arr2[i] % 2 == 0:
				if left >= arr2[i] / 2:
					left = left - arr2[i] / 2
					for k in range(0, n):
						if arr[k] == arr2[i] / 2:
							arr[k] = arr2[i]
							break
				else:
					for k in range(0, n):
						if arr[k] == arr2[i] / 2:
							arr[k] = arr[k] + left
							break
					left = 0
			elif left > math.floor(arr2[i] / 2):
				left = left - math.floor(arr2[i] / 2)
				for k in range(0, n):
					if arr[k] == arr2[i] / float(2) + 1:
						arr[k] = arr2[i]
						break
			else:
				for k in range(0, n):
					if arr[k] == arr2[i] / float(2) + 1:
						arr[k] = arr[k] + left
						break
				left = 0
			i = i - 1
		for z in range(0, n):
			print(math.floor(arr[z]), end=' ')
main()
