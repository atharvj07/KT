#include <bits/stdc++.h>
using namespace std;
bool visited[1000010];
long long arr[1000010];
int main() {
  long long n, i, j, k, m, t, second;
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  long long K;
  cin >> n;
  for (i = 1; i <= n; i++) cin >> arr[i];
  long long cnt = 0;
  for (i = 1; i <= n; i++) {
    if (visited[i]) continue;
    long long op = i;
    visited[op] = 1;
    op = arr[op];
    while (op != i) visited[op] = 1, op = arr[op];
    cnt++;
  }
  cnt = n - cnt;
  cnt = 3 * n - cnt;
  if (cnt % 2 == 0)
    cout << "Petr\n";
  else
    cout << "Um_nik\n";
  return 0;
}
