a = tuple(map(int, input().split()))
b = tuple(map(int, input().split()))
c = tuple(map(int, input().split()))
print(3)

def middle(x, y):
	return ((x[0] + y[0]) / 2, (x[1] + y[1]) / 2)

def sym(a, o):
	return (int(2 * o[0] - a[0]), int(2 * o[1] - a[1]))
print(sym(c, middle(a, b))[0], sym(c, middle(a, b))[1])
print(sym(b, middle(a, c))[0], sym(b, middle(a, c))[1])
print(sym(a, middle(b, c))[0], sym(a, middle(b, c))[1])
