def main():
	string = input('')
	list = string.split('e')
	b = int(list[1])
	L = len(list[0]) - 2
	list1 = str(list[0]).split('.')
	final = ''.join(list1)
	if L > b:
		position = 2 + b
		ans = ''
		ans1 = final[0:position - 1]
		ans2 = final[position - 1:]
		ans = ans1 + '.' + ans2
	elif L == b:
		ans = ''
		ans = final
	elif L < b:
		ans1 = ''
		ans = ''
		for i in range(b - L):
			ans1 = ans1 + '0'
		ans = final + ans1
	if ans[1] == '.' and ans[2] == '0' and (len(ans) == 3):
		print(int(eval(ans)))
	elif eval(ans) == int(eval(ans)):
		print(ans)
	else:
		listans = ans.split('.')
		finalans1 = ''
		finalans2 = ''
		finalans1 = str(int(listans[0]))
		finalans2 = str(listans[1])
		finalans = finalans1 + '.' + finalans2
		print(finalans)
main()
