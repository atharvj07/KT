vova = list(map(int, input().split()))
crab = list(map(int, input().split()))
result = []
while crab[0] > 0:
	if vova[0] - crab[1] <= 0 and crab[0] - vova[1] > 0:
		vova[0] += vova[2]
		result.append('HEAL')
	else:
		crab[0] -= vova[1]
		result.append('STRIKE')
	vova[0] -= crab[1]
print(len(result))
for i in result:
	print(i)
