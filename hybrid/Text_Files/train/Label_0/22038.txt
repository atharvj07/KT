#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) { return b == 0 ? a : gcd(b, a % b); }
const int MAXN = 2000;
const int MAXM = 10000;
const int MAXROWS = MAXN;
const int MAXCOLS = MAXN + 1;
const int MAXWORDS = (MAXCOLS + 63) / 64;
struct BinaryGaussJord {
  int R, C, W;
  unsigned long long A[MAXROWS][MAXWORDS];
  void init(int _R, int _C) {
    R = _R, C = _C, W = (C + 63) / 64;
    for (int r = (0); r < (R); ++r)
      for (int w = (0); w < (W); ++w) A[r][w] = 0;
  }
  void toggle(int r, int c) { A[r][c / 64] ^= 1LL << (c % 64); }
  bool isset(int r, int c) { return ((A[r][c / 64] >> (c % 64)) & 1) != 0; }
  void solve() {
    int r = 0, c = 0;
    while (r < R && c < C) {
      {
        int rr = r;
        while (rr < R && !isset(rr, c)) ++rr;
        if (rr >= R) {
          ++c;
          continue;
        }
        if (rr != r)
          for (int w = (0); w < (W); ++w) swap(A[r][w], A[rr][w]);
      }
      for (int rr = (r + 1); rr < (R); ++rr)
        if (isset(rr, c))
          for (int w = (0); w < (W); ++w) A[rr][w] ^= A[r][w];
      ++r, ++c;
    }
    while (--r >= 0) {
      c = 0;
      while (c < C && !isset(r, c)) ++c;
      assert(c < C);
      for (int rr = (0); rr < (r); ++rr)
        if (isset(rr, c))
          for (int w = (0); w < (W); ++w) A[rr][w] ^= A[r][w];
    }
  }
};
int n, m;
vector<int> adj[MAXN];
int ans[MAXN], nans;
BinaryGaussJord mat;
void solve() {
  bool alleven = true;
  for (int i = (0); i < (n); ++i)
    if (((int)(adj[i]).size()) % 2 != 0) alleven = false;
  if (alleven) {
    nans = 1;
    for (int i = (0); i < (n); ++i) ans[i] = 0;
    return;
  }
  mat.init(n, n + 1);
  for (int i = (0); i < (n); ++i)
    for (int j = (0); j < (((int)(adj[i]).size())); ++j) {
      int k = adj[i][j];
      mat.toggle(i, k);
    }
  for (int i = (0); i < (n); ++i)
    if (((int)(adj[i]).size()) % 2 != 0) mat.toggle(i, i), mat.toggle(i, n);
  mat.solve();
  nans = 2;
  for (int i = (0); i < (n); ++i) ans[i] = 0;
  for (int i = (0); i < (n); ++i) {
    int idx = -1;
    for (int j = (0); j < (n); ++j)
      if (mat.isset(i, j) && idx == -1) idx = j;
    if (mat.isset(i, n)) {
      assert(idx != -1 && ans[idx] == 0);
      ans[idx] = 1;
    }
  }
}
void run() {
  scanf("%d%d", &n, &m);
  for (int i = (0); i < (m); ++i) {
    int a, b;
    scanf("%d%d", &a, &b);
    --a, --b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  solve();
  printf("%d\n", nans);
  for (int i = (0); i < (n); ++i) {
    if (i != 0) printf(" ");
    printf("%d", ans[i] + 1);
  }
  puts("");
  for (int i = (0); i < (n); ++i) adj[i].clear();
}
int main() {
  int n;
  scanf("%d", &n);
  for (int i = (1); i <= (n); ++i) run();
  return 0;
}
