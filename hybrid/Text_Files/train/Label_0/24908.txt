#include <bits/stdc++.h>
using namespace std;
int main() {
  set<long long> sx, sy;
  set<pair<long long, long long> > sxy;
  long long tx, ty, bx, by;
  long long x[10], y[10];
  map<pair<long long, long long>, bool> done;
  for (long long i = 1; i <= 8; i++) {
    cin >> x[i] >> y[i];
    done[make_pair(x[i], y[i])] = true;
    sx.insert(x[i]);
    sy.insert(y[i]);
    sxy.insert(make_pair(x[i], y[i]));
  }
  if (sxy.size() < 8 || sx.size() < 3 || sy.size() < 3) {
    cout << "ugly";
    return 0;
  }
  bx = *sx.begin();
  by = *sy.begin();
  vector<long long> vx, vy;
  for (set<long long>::iterator it = sx.begin(); it != sx.end(); it++) {
    vx.push_back(*it);
  }
  for (set<long long>::iterator it = sy.begin(); it != sy.end(); it++) {
    vy.push_back(*it);
  }
  sort(vx.begin(), vx.end());
  sort(vy.begin(), vy.end());
  for (long long i = 0; i < 3; i++) {
    for (long long j = 0; j < 3; j++) {
      if (i == 1 && j == 1) {
        continue;
      }
      if (sxy.count(make_pair(vx[i], vy[j])) != 1) {
        cout << "ugly";
        return 0;
      }
    }
  }
  cout << "respectable";
  return 0;
}
