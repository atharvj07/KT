#include <cstdio>
#include <algorithm>
#include <cmath>
#include <vector>
#define S second
using namespace std;
double ans, pi=acos(-1.0);
pair<int,int> b[200005];
vector<int> v;

int main() {
	int x1, y1, x2, y2, n, M=1e9, bc=0;
	int i, t, tx, ty, fl=0;
	scanf("%d%d%d%d%d", &x1, &y1, &x2, &y2, &n);
	if(x1>x2) swap(x1, x2), swap(y1, y2);
	if(y1>y2) y1 = M-y1, y2 = M-y2, fl=1;
	for(i=0; i<n; i++) {
		scanf("%d%d", &tx, &ty);
		if(fl) ty = M-ty;
		if(x1<=tx && tx<=x2 && y1<=ty && ty<=y2) b[bc++] = {tx, ty};
	}
	sort(b, b+bc);
	for(i=0; i<bc; i++) {
		t = lower_bound(v.begin(), v.end(), b[i].S) - v.begin();
		if(t==v.size()) v.push_back(b[i].S);
		else v[t] = b[i].S;
	}
	ans = (x2-x1 + y2-y1) * 100.0 + v.size() * (5*pi - 20);
	if(v.size() == min(x2-x1, y2-y1)+1) ans += 5*pi;
	printf("%.15f", ans);
	return 0;
}
