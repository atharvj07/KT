#include <bits/stdc++.h>
using namespace std;
const long long inf = 0x3f3f3f3f3f3f3f3f;
const long long mod = 1e9 + 7;
void redirect_input() { freopen("./input.txt", "r", stdin); }
void redirect_output() { freopen("./output.txt", "w", stdout); }
const int maxn = 2e5 + 1000;
struct seg {
  int l, r, index;
  bool operator<(const seg& s1) const {
    if (l == s1.l) return r < s1.r;
    return l > s1.l;
  }
};
int val[maxn * 4], lazy[maxn * 4];
void build(int l, int r, int rt) {
  val[rt] = lazy[rt] = 0;
  if (l == r) return;
  int mid = (l + r) >> 1;
  build(l, mid, rt << 1);
  build(mid + 1, r, rt << 1 | 1);
}
void push_down(int rt) {
  if (lazy[rt] == 0) return;
  lazy[rt << 1] += lazy[rt];
  lazy[rt << 1 | 1] += lazy[rt];
  val[rt << 1] += lazy[rt];
  val[rt << 1 | 1] += lazy[rt];
  lazy[rt] = 0;
}
void update(int ul, int ur, int l, int r, int rt) {
  if (ul <= l && ur >= r) {
    val[rt] += 1, lazy[rt] += 1;
    return;
  }
  push_down(rt);
  int mid = (l + r) >> 1;
  if (ul <= mid) update(ul, ur, l, mid, rt << 1);
  if (ur > mid) update(ul, ur, mid + 1, r, rt << 1 | 1);
  val[rt] = max(val[rt << 1], val[rt << 1 | 1]);
}
int query(int ql, int qr, int l, int r, int rt) {
  if (ql <= l && qr >= r) return val[rt];
  push_down(rt);
  int mid = (l + r) >> 1;
  int ret = -1;
  if (ql <= mid) ret = query(ql, qr, l, mid, rt << 1);
  if (qr > mid) ret = max(ret, query(ql, qr, mid + 1, r, rt << 1 | 1));
  return ret;
}
void solve() {
  int n, k;
  cin >> n >> k;
  build(1, maxn, 1);
  vector<seg> segs;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    segs.push_back({l, r, i + 1});
  }
  sort(segs.begin(), segs.end());
  int ans = 0;
  vector<int> idx;
  for (seg s : segs) {
    if (query(s.l, s.r, 1, maxn, 1) < k) {
      update(s.l, s.r, 1, maxn, 1);
    } else {
      ans++;
      idx.push_back(s.index);
    }
  }
  cout << ans << '\n';
  for (int i : idx) cout << i << ' ';
  cout << endl;
}
signed main() {
  ios::sync_with_stdio(false), cin.tie(NULL);
  solve();
}
