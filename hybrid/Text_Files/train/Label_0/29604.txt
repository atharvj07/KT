#include <bits/stdc++.h>
using namespace std;
const int oo = 0x7f7f7f7f;
const int Max = 1000010;
char S[Max];
char F[2][Max];
int L, R[2][Max];
int T[2][2][2];
void O(char b, int x) { printf("%c2^%d\n", b, L - x); }
void out(int x, int f) {
  char c = S[x - 1] - '0';
  int p = F[f][x];
  if (p == 0 && f == 0 && c == 1) {
    O('+', x);
  } else if (p == 0 && f == 1 && c == 1) {
    O('+', x - 1);
  } else if (p == 0 && f == 1 && c == 0) {
    O('+', x - 1);
    O('-', x);
  } else if (p == 1 && f == 0) {
    O('-', x - !c);
  } else if (p == 1 && f == 1 && c == 0) {
    O('-', x);
  }
  if (x == 0) return;
  out(x - 1, p);
}
int main() {
  T[0][0][1] = 1;
  T[0][1][0] = 2;
  T[0][1][1] = 1;
  T[1][0][0] = 1;
  T[1][0][1] = 1;
  T[1][1][0] = 1;
  while (scanf("%s", S) != EOF) {
    L = strlen(S);
    R[0][0] = 0;
    R[1][0] = oo;
    memset(R[0], 0x7f, sizeof(R[0][0]) * (L + 1));
    memset(R[1], 0x7f, sizeof(R[1][0]) * (L + 1));
    R[0][0] = 0;
    for (int i = 0; i < L; i++) {
      int c = S[i] - '0';
      for (int j = 0; j < 2; j++)
        if (R[j][i] < oo)
          for (int k = 0; k < 2; k++) {
            int cd = R[j][i] + T[j][k][c];
            if (cd < R[k][i + 1]) {
              R[k][i + 1] = cd;
              F[k][i + 1] = j;
            }
          }
    }
    printf("%d\n", R[0][L]);
    out(L, 0);
  }
  return 0;
}
