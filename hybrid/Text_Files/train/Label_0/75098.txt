import sys
import math
from collections import defaultdict, deque
input = sys.stdin.readline

def inar():
	return [int(el) for el in input().split()]

def toogle(char):
	if char == 'b':
		return 'w'
	else:
		return 'b'

def main():
	string = input().strip()
	n = len(string)
	new = string + string
	ans = 1
	temp = 1
	for i in range(2 * n):
		if new[i] != new[i - 1]:
			temp += 1
		else:
			ans = max(ans, temp)
			temp = 1
	ans = max(ans, temp)
	print(min(ans, n))
main()
