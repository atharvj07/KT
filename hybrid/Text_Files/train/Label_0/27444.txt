#include <bits/stdc++.h>

using namespace std;
using ll = long long;

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(0), cout.tie(0);

	int n, k;
	cin >> n >> k;
	vector<int> a(n);
	for (auto &x : a) {
		cin >> x;
	}
	int L = 0, R = *max_element(a.begin(), a.end()) + 1;
	while (R - L > 1) {
		int mid = (L + R) / 2;
		ll cnt = 0;
		for (int i = 0; i < n; i++) {
			cnt += (a[i] - 1) / mid;
		}
		(cnt > k ? L : R) = mid;
	}
	cout << R << '\n';
}
