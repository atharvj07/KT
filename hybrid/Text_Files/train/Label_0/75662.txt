s = input()
cur = 0
tmp = []
a = []
for i in s:
	if i == '+' or i == '-':
		a.append(cur)
		tmp.append(i)
		cur = 0
	else:
		cur = cur * 10 + (ord(i) - ord('0'))
a.append(cur)
print('+' * a[0] + '>')
for i in range(len(tmp)):
	print('+' * a[i + 1] + '>')
	if tmp[i] == '+':
		print('<[<+>-]<')
	else:
		print('<[<->-]<')
	if i < len(tmp) - 1:
		print('>')
print('[>>+>+<<<-]>>>[<<<+>>>-]<<+>[<->[>++++++++++<[->-[>+>>]>[+[-<+>]>+>>]<<<<<]>[-]++++++++[<++++++>-]>[<<+>>-]>[<<+>>-]<<]>]<[->>++++++++[<++++++>-]]<[.[-]<]<')
