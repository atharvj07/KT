long long gcd(long long a, long long b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
long long fact(long long n) {
  long long res = 1;
  for (long long z = 2; z <= n; z++) res = res * z;
  return res;
}
long long Pow(long long a, long long b) {
  long long res = 1;
  while (b) {
    if (b & 1) res = res * a % 1000000007;
    a = a * a % 1000000007;
    b >>= 1;
  }
  return res;
}
long long amax(long long a[], long long n) {
  long long max = a[0];
  for (long long i = 0; i < n; i++) {
    if (a[i] > max) max = a[i];
  }
  return max;
}
long long amin(long long a[], long long n) {
  long long min = 1000000000000000001;
  for (long long i = 0; i < n; i++) {
    if (a[i] < min) min = a[i];
  }
  return min;
}

  }
}
bool comp(const pair<int, int> &a, const pair<int, int> &b) {
  return (a.first > b.first);
}
long long divisor(long long x) {
  long long cnt = 0;
  for (int i = 1; i <= sqrt(x); i++) {
    if (x % i == 0) {
      if (i != x / i)
        cnt += 2;
      else
        cnt += 1;
    }
  }
  return cnt;
}
string d2b(int n) {
  string t = "";
  long long key = 0;
  for (int i = 31; i >= 0; i--) {
    int k = n >> i;
    if (k & 1) {
      t += '1';
      key = 1;
    } else {
      if (key) t += '0';
    }
  }
  return t;
}
long long b2d(string n) {
  string num = n;
  long long dec_value = 0;
  long long base = 1;
  long long len = num.length();
  for (long long i = len - 1; i >= 0; i--) {
    if (num[i] == '1') dec_value += base;
    base = base * 2;
  }
  return dec_value;
}
bool palindrome(long long le, long long ri, string s) {
  if (ri - le <= 0) return true;
  if (s[le] != s[ri]) return false;
  return palindrome(le + 1, ri - 1, s);
}
long long isprm(long long n) {
  if (n <= 1) return 0;
  if (n <= 3) return 1;
  if (n % 2 == 0 || n % 3 == 0) return 0;
  for (int i = 5; i * i <= n; i = i + 6)
    if (n % i == 0 || n % (i + 2) == 0) return 0;
  return 1;
}
long long lcm(long long a, long long b) { return a / gcd(a, b) * b; }
bool psqr(long double x) {
  long double sr = sqrt(x);
  return ((sr - floor(sr)) == 0);
}
bool isPowerOfTwo(long long n) {
  if (n == 0) return false;
  return (ceil(log2(n)) == floor(log2(n)));
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  {
    long long n;
    cin >> n;
    long long a, b, l, r;
    vector<pair<long long, long long>> v;
    map<long long, long long> m, m1;
    for (long long i = 0; i < n; i++) {
      cin >> a >> b;
      m[a]++;
      m1[b]++;
      v.push_back({a, b});
    }
    l = v[0].first, r = v[0].second;
    for (long long i = 1; i < n; i++) {
      l = max(l, v[i].first);
      r = min(r, v[i].second);
    }
    long long key = 1;
    {
      long long ans = (r - l) + 1;
      long long l1, r1;
      for (long long i = 0; i < n; i++) {
        l1 = l, r1 = r;
        if (v[i].first == l1) {
          if (m[v[i].first] == 1) {
            auto it = m.lower_bound(v[i].first);
            it--;
            l1 = (*it).first;
          }
        }
        if (v[i].second == r1) {
          if (m1[v[i].second] == 1) {
            auto it = m1.upper_bound(v[i].second);
            r1 = (*it).first;
          }
        }
        if (l1 <= r1) key = 0;
        ans = max(ans, r1 - l1 + 1);
      }
      if (key)
        cout << 0 << endl;
      else
        cout << ans - 1 << endl;
    }
  }
  return (0);
}
