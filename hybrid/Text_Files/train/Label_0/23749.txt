#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int f = 1, x = 0;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = x * 10 + ch - '0';
    ch = getchar();
  }
  return x * f;
}
inline void write(int x) {
  if (x < 0) putchar('-'), x = -x;
  if (x > 9) write(x / 10);
  putchar(x % 10 + '0');
}
inline void print(int x) {
  write(x);
  printf(" ");
}
int n, a[310000], b[310000], blen;
int gcd(int a, int b) { return a == 0 ? b : gcd(b % a, a); }
bool v[15000005];
int p[15000005], fac[15000005], plen;
void get_pri() {
  memset(v, true, sizeof(v));
  fac[1] = 1;
  for (int i = 2; i <= 15000000; i++) {
    if (v[i]) p[++plen] = i, fac[i] = i;
    for (int j = 1; j <= plen && i * p[j] <= 15000000; j++) {
      v[i * p[j]] = false;
      fac[i * p[j]] = fac[p[j]];
      if (!(i % p[j])) break;
    }
  }
}
int S[15000005];
int main() {
  n = read();
  get_pri();
  bool bk = false;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    a[i] = read();
    if (a[i] != a[i - 1] && i != 1) bk = true;
  }
  if (!bk) {
    puts("-1");
    return 0;
  }
  int gd = a[1];
  for (int i = 2; i <= n; i++) gd = gcd(gd, a[i]);
  for (int i = 1; i <= n; i++) a[i] /= gd;
  for (int i = 1; i <= n; i++) {
    while (a[i] != 1) {
      S[fac[a[i]]]++;
      int u = fac[a[i]];
      while (a[i] % u == 0) a[i] /= u;
    }
  }
  int mx = 0;
  for (int i = 1; i <= 15000000; i++)
    if (S[i] < n) mx = max(mx, S[i]);
  printf("%d\n", n - mx);
  return 0;
}
