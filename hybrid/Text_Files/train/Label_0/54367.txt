import java.io.*;
import java.util.*;

public class D {
    public static void main(String[] args) throws IOException {
        new D().solve();
    }

    void solve() throws IOException {
        // Scanner sc = new Scanner(System.in);
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] sp;

        sp = in.readLine().split(" ");
        int n = Integer.parseInt(sp[0]);
        int h = Integer.parseInt(sp[1]);

        long[] a = new long[n];

        long IM = Integer.MAX_VALUE;

        sp = in.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(sp[i]) * IM + i;
        }

        Arrays.sort(a);
        int[] idx = new int[n];
        for (int i = 0; i < n; i++) {
            idx[i] = (int) (a[i] % IM);
            a[i] /= IM;
        }

        int max;
        int min;
        int g1, g2;
        // 1222...
        max = (int) Math.max(a[0] + a[n - 1] + h, a[n - 2] + a[n - 1]);
        if (n > 2) {
            min = (int) Math.min(a[0] + a[1] + h, a[1] + a[2]);
        } else {
            min = (int) (a[0] + a[1] + h);
        }
        g1 = max - min;
        // 2222...
        max = (int) (a[n - 2] + a[n - 1]);
        min = (int) (a[0] + a[1]);
        g2 = max - min;

        if (g1 < g2) {
            System.out.println(g1);
            StringBuilder str = new StringBuilder();
            for (int i = 0; i < n; i++) {
                if (i == idx[0]) {
                    str.append("1 ");
                } else {
                    str.append("2 ");
                }
            }
            System.out.println(str);
        } else {
            System.out.println(g2);
            StringBuilder str = new StringBuilder();
            str.append(2);
            for (int i = 0; i < n - 1; i++) {
                str.append(" 2");
            }
            System.out.println(str);
        }
    }
}
