
import java.io.*;
import java.util.*;

public class T5 extends PrintWriter {
    public T5() {
        super(System.out);
    }

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        T5 main = new T5();
        int n = 1;
        while (n-- > 0) {
            main.mainPrint(s);
        }
        main.flush();
    }

    private void mainPrint(Scanner s) {
        int n = s.nextInt(), k = s.nextInt();
        int[] A = new int[n];
        for (int i = 0; i < n; i++) A[i] = s.nextInt();
        int[] D = new int[n];
        for (int i = 0; i < n; i++) D[i] = s.nextInt();
        long ans = 0L;
        long[] sum = new long[n];
        sum[n - 1] = A[n - 1];
        int min = Math.min(D[n - 2], D[n - 3]), min2 = D[n - 3] + D[n - 2] - min;
        int minA = Math.min(A[n - 2], A[n - 3]);
        ans = Math.max(ans,A[n - 1] - D[n - 1]);

        int temp;
        for (int i = n - 2; i >= 0; i--) {
            sum[i] = sum[i + 1] + A[i];
            if(k == 0 || i != 0) ans = Math.max(ans, sum[i] - D[i]);
        }

        for (int i = n - 4; i >= 0; i--) {
            if(i!=0) minA = Math.min(minA,A[i]);
            if (D[i] < min2) {
                min2 = D[i];
                if (min2 < min) {
                    temp = min;
                    min = min2;
                    min2 = temp;
                }
            }
        }

        if(k > 0) {
            ans = Math.max(sum[0] - minA - D[0],ans);
            ans = Math.max(sum[0] - sum[n - 1] - min + Math.max(A[n - 1] - D[n - 1], 0), ans);
            ans = Math.max(ans,sum[0] - min - min2);
        }
        if (k > 1) {
            ans = Math.max(sum[0] - min,ans);
        }
        println(ans);
    }

}
