(n, m) = map(int, input().strip().split(' '))
c = []
s = 0
s1 = 0
for i in range(n):
	(a, b) = map(int, input().strip().split(' '))
	c.append(a - b)
	s += b
	s1 += a
if s > m:
	print('-1')
elif s1 <= m:
	print(0)
else:
	c.sort()
	count = 0
	for i in range(n):
		if s + c[i] <= m:
			s += c[i]
		else:
			break
	print(n - i)
