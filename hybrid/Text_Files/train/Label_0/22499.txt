#include <bits/stdc++.h>
using namespace std;
int a[105];
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
int main() {
  int n, ma = -1;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%d", &a[i]);
    ma = max(ma, a[i]);
  }
  int g = gcd(a[0], a[1]);
  for (int i = 2; i < n; i++) g = gcd(g, a[i]);
  int t = ma / g - n;
  if (t & 1)
    printf("Alice\n");
  else
    printf("Bob\n");
}
