import collections
import itertools
t = int(input())
for _ in range(t):
	n = int(input())
	students = []
	days = [0 for _ in range(5)]
	days_to_students = collections.defaultdict(set)
	for i in range(n):
		student = list(map(int, input().split()))
		for (day, amount) in enumerate(student):
			if amount:
				days[day] += 1
				days_to_students[day].add(i)
	good_days = []
	for i in range(5):
		if days[i] >= n // 2:
			good_days.append(i)
	for (a, b) in itertools.combinations(good_days, 2):
		if len(days_to_students[a] | days_to_students[b]) == n:
			print('YES')
			break
	else:
		print('NO')
