#include <bits/stdc++.h>
using namespace std;
vector<vector<int> > g, gr;
vector<bool> used;
vector<int> order, component;
void dfs1(int v) {
  used[v] = true;
  for (size_t i = 0; i < g[v].size(); ++i)
    if (!used[g[v][i]]) dfs1(g[v][i]);
  order.push_back(v);
}
void dfs2(int v) {
  used[v] = true;
  component.push_back(v);
  for (size_t i = 0; i < gr[v].size(); ++i)
    if (!used[gr[v][i]]) dfs2(gr[v][i]);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t, n;
  string s;
  vector<int> v;
  cin >> t;
  while (t--) {
    cin >> n >> s;
    g.assign(n, v);
    gr.assign(n, v);
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == '>') {
        g[i].push_back((i + 1) % n);
        gr[(i + 1) % n].push_back(i);
      } else if (s[i] == '<') {
        g[(i + 1) % n].push_back(i);
        gr[i].push_back((i + 1) % n);
      } else {
        g[i].push_back((i + 1) % n);
        g[(i + 1) % n].push_back(i);
        gr[(i + 1) % n].push_back(i);
        gr[i].push_back((i + 1) % n);
      }
    }
    used.assign(n, false);
    for (int i = 0; i < n; ++i)
      if (!used[i]) dfs1(i);
    used.assign(n, false);
    int ans = 0;
    for (int i = 0; i < n; ++i) {
      int v = order[n - 1 - i];
      if (!used[v]) {
        dfs2(v);
        if (component.size() >= 2) ans += component.size();
        component.clear();
      }
    }
    cout << ans << "\n";
    component.clear();
    order.clear();
  }
  return 0;
}
