#include <bits/stdc++.h>
using namespace std;
const double PI = 2 * acos(0);
double resuelve(double w, double h, double a) {
  a = (a * PI) / 180;
  double total = w * h;
  double parte1 = (h / 2 - w * tan(a / 2) / 2);
  parte1 *= parte1;
  parte1 *= tan(a);
  double parte2 = (w / 2 - h * tan(a / 2) / 2);
  parte2 *= parte2;
  parte2 *= tan(a);
  double respuesta = h * h / sin(a);
  if (a <= 2 * atan(h / w)) {
    respuesta = total - parte1 - parte2;
  }
  return respuesta;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  double w, h, a;
  cin >> w >> h >> a;
  if (w < h) swap(w, h);
  if (a > 90) {
    a = 180 - a;
  }
  double resp = resuelve(w, h, a);
  cout << setprecision(10) << fixed;
  cout << resp << "\n";
  return 0;
}
