import java.io.PrintWriter;
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class PINCodesB extends PrintWriter {
    //this trick improves performances
    PINCodesB() {
        super(System.out);
    }

    public static void main(String[] $) {
        PINCodesB o = new PINCodesB();
        o.main();
        o.flush();
    }

    void main() {
        Scanner sc = new Scanner(System.in);
        int count = sc.nextInt();
        //use this if just a single test
        //int count = 1;
        while (count-- > 0) {
            final int n = sc.nextInt();
            final char[][] cards = new char[n][4];
            final Set<Integer> setFirstDigit = new HashSet<>();
            final Set<String> words = new HashSet<>();
            String tmp;

            for (int i = 0; i < n; i++) {
                tmp=sc.next();
                cards[i]=tmp.toCharArray();
                setFirstDigit.add(Integer.valueOf(tmp.substring(0,1)));
            }

            for (int i = 0; i < n; i++) {
                if(words.add(new String(cards[i]))){
                    continue;
                }
                for (int j = 0; j < n; j++) {
                    if(setFirstDigit.contains(j)){
                        continue;
                    }
                    cards[i][0]=String.valueOf(j).toCharArray()[0];
                    setFirstDigit.add(j);
                    break;
                }
            }
            int result=n-words.size();
            println(result);
            for (int i = 0; i < n; i++) {
               println(cards[i]);
            }
        }
        sc.close();
    }
}
