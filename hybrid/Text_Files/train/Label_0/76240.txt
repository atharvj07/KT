for _ in range(int(input())):
	(a, s) = input().split()
	b = ''
	q = 0
	(i, j) = (-1, -1)
	if len(s) < len(a):
		print(-1)
		break
	while i != -len(a) - 1 and j != -len(s) - 1:
		if not q:
			if int(a[i]) < int(s[j]):
				b += str(int(s[j]) - int(a[i]))
				i -= 1
				j -= 1
			elif int(a[i]) == int(s[j]):
				b += '0'
				i -= 1
				j -= 1
			else:
				b += str(10 + int(s[j]) - int(a[i]))
				q = 1
				j -= 1
		else:
			if s[j] == '1':
				q = 0
			else:
				print(-1)
				break
			i -= 1
			j -= 1
		if i != -len(a) - 1 and j == -len(s) - 1:
			print(-1)
			break
	else:
		if j != -len(s) - 1:
			b += s[:j + 1][::-1]
		print(int(b[::-1]))
