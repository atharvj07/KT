import math
import sys
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input().strip()
	return list(s[:len(s)])

def invr():
	return map(int, input().split())
for _ in range(inp()):
	n = inp()
	l = inlt()
	cnt = 0
	mn = l[-1]
	for i in range(n - 2, -1, -1):
		if l[i] > mn:
			cnt += 1
		else:
			mn = l[i]
	print(cnt)
