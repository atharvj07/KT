n = int(input())
a = list(map(int, input().split()))
for i in range(n):
	a[i] -= i
sorted_a = sorted(a)
dp = [0.0] * n
dp2 = [0.0] * n
for i in range(n):
	mn_prev_state = 10000000000000.0
	for j in range(n):
		mn_prev_state = min(mn_prev_state, dp[j])
		dp2[j] = mn_prev_state + abs(a[i] - sorted_a[j])
	for j in range(n):
		dp[j] = dp2[j]
print(int(min(dp)))
