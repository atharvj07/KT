from collections import Counter

def serials_to_buy(n, k, d, shedule):
	current_sequence = shedule[:d]
	counter = Counter(current_sequence)
	ans = len(counter)
	for i in range(d, n):
		if shedule[i] in counter:
			counter[shedule[i]] += 1
		else:
			counter[shedule[i]] = 1
		if counter[shedule[i - d]] != 1:
			counter[shedule[i - d]] -= 1
		else:
			del counter[shedule[i - d]]
		ans = min(ans, len(counter))
	return ans
t = int(input())
for _ in range(t):
	(n, k, d) = map(int, input().split())
	shedule = input().split()
	if d == 1:
		print(d)
	else:
		print(serials_to_buy(n, k, d, shedule))
