#include <bits/stdc++.h>
using namespace std;
struct Edge {
  int x;
  int y;
  int cap;
  int flow;
  int cost;
  Edge() { x = y = cap = flow = cost = 0; }
  Edge(int x_, int y_, int cap_, int cost_)
      : x(x_), y(y_), cap(cap_), cost(cost_) {
    flow = 0;
  }
};
vector<Edge> edgeList;
vector<vector<int> > adj;
int par[150 + 5];
int dist[150 + 5];
int N, M, K;
int ax, ay, acost, acap;
char buf[1000];
map<int, int> charMap;
int maxchars;
map<int, int> charsSink;
int main() {
  for (int i = 0; i <= 150 + 10; i++) {
    vector<int> vec;
    adj.push_back(vec);
  }
  int src, sink, flow;
  int len, curCost, edgeNum;
  int mf, mfCost, v, d, nv, nd;
  int other, maxcap;
  pair<int, int> ip;
  scanf("%s", buf);
  src = 0;
  sink = 150;
  len = strlen(buf);
  for (int i = 0; i < len; i++) {
    if (charMap.find(buf[i]) == charMap.end()) {
      charMap[buf[i]] = 1;
    } else {
      charMap[buf[i]]++;
    }
  }
  int charsStart = sink;
  for (map<int, int>::iterator it = charMap.begin(); it != charMap.end();
       it++) {
    charsStart--;
    do {
      adj[charsStart].push_back(edgeList.size());
      edgeList.push_back(Edge(charsStart, sink, it->second, 0));
      adj[sink].push_back(edgeList.size());
      edgeList.push_back(Edge(sink, charsStart, 0, -0));
    } while (0);
    charsSink[it->first] = charsStart;
  }
  scanf("%d", &N);
  for (int i = 1; i <= N; i++) {
    scanf("%s %d", buf, &maxchars);
    do {
      adj[src].push_back(edgeList.size());
      edgeList.push_back(Edge(src, i, maxchars, 0));
      adj[i].push_back(edgeList.size());
      edgeList.push_back(Edge(i, src, 0, -0));
    } while (0);
    charMap.clear();
    len = strlen(buf);
    for (int k = 0; k < len; k++) {
      if (charMap.find(buf[k]) == charMap.end()) {
        charMap[buf[k]] = 1;
      } else {
        charMap[buf[k]]++;
      }
    }
    for (map<int, int>::iterator it = charMap.begin(); it != charMap.end();
         it++) {
      if (charsSink.find(it->first) != charsSink.end()) {
        other = charsSink[it->first];
        maxcap = min(maxchars, it->second);
        do {
          adj[i].push_back(edgeList.size());
          edgeList.push_back(Edge(i, other, maxcap, i));
          adj[other].push_back(edgeList.size());
          edgeList.push_back(Edge(other, i, 0, -i));
        } while (0);
      }
    }
  }
  mf = 0;
  mfCost = 0;
  while (1) {
    memset(par, -1, sizeof par);
    memset(dist, 0x1f, sizeof dist);
    priority_queue<pair<int, int> > pq;
    pq.push(pair<int, int>(0, src));
    dist[src] = 0;
    while (!pq.empty()) {
      ip = pq.top();
      pq.pop();
      d = -ip.first;
      v = ip.second;
      if (dist[v] != d) {
        continue;
      }
      if (v == sink) {
        break;
      }
      len = adj[v].size();
      for (int i = 0; i < len; i++) {
        edgeNum = adj[v][i];
        const Edge &edge = edgeList[edgeNum];
        nv = edge.y;
        nd = d + edge.cost;
        if (edge.cap - edge.flow > 0 && nd < dist[nv]) {
          par[nv] = edgeNum;
          dist[nv] = nd;
          pq.push(pair<int, int>(-nd, nv));
        }
      }
    }
    if (par[sink] == -1) {
      break;
    }
    flow = 0x1f1f1f1f;
    for (int v = sink; v != src; v = edgeList[par[v]].x) {
      flow = min(flow, edgeList[par[v]].cap - edgeList[par[v]].flow);
    }
    mf += flow;
    for (int v = sink; v != src; v = edgeList[par[v]].x) {
      edgeNum = par[v];
      Edge &edge = edgeList[edgeNum];
      mfCost += (edge.cost * flow);
      edge.flow += flow;
      edgeList[edgeNum ^ 1].flow -= flow;
    }
  }
  int can = 1;
  len = adj[sink].size();
  for (int i = 0; i < len; i++) {
    const Edge &e = edgeList[adj[sink][i] ^ 1];
    if (e.flow < e.cap) {
      can = 0;
      break;
    }
  }
  if (can) {
    printf("%d\n", mfCost);
  } else {
    printf("-1\n");
  }
  return 0;
}
