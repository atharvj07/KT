import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.InputMismatchException;
import java.io.IOException;
import java.util.Collections;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskE solver = new TaskE();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskE {
        static final long MODULO = (long) (1e9 + 7);

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.readInt();
            long k = in.readLong();
            ArrayList<Line> lineSegments = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                lineSegments.add(new Line(in.readLong(), in.readLong(), in.readInt()));
            }
            Collections.sort(lineSegments);
            lineSegments.get(lineSegments.size() - 1).b = k;

            Matrix ans = new Matrix(1, lineSegments.get(0).c + 1);
            ans.array[0][0] = 1L;


            for (int i = 0; i < n; i++) {
                int m = lineSegments.get(i).c + 1;
                Matrix transformer = new Matrix(m, m);
                for (int b = 0; b < m; b++) {
                    if (b - 1 >= 0) transformer.array[b - 1][b] = 1;
                    transformer.array[b][b] = 1;
                    if (b + 1 < m) transformer.array[b + 1][b] = 1;
                }

                long levels = lineSegments.get(i).b - lineSegments.get(i).a;
                transformer = transformer.power(levels);

                ans = ans.multiply(transformer);

                if (i != n - 1) {
                    Matrix anscopy = new Matrix(1, lineSegments.get(i + 1).c + 1);
                    for (int j = 0; j < Math.min(ans.array[0].length, anscopy.array[0].length); j++) {
                        anscopy.array[0][j] = ans.array[0][j];
                    }
                    ans = anscopy;
                }

            }

            out.println(ans.array[0][0] % MODULO);

        }

        class Matrix {
            int rows;
            int cols;
            long[][] array;

            public Matrix(int rows, int cols) {
                this.rows = rows;
                this.cols = cols;
                array = new long[rows][cols];
            }

            public Matrix multiply(Matrix dest) {
                Matrix ans = new Matrix(rows, dest.cols);

                for (int i = 0; i < rows; i++) {
                    for (int j = 0; j < dest.cols; j++) {
                        for (int k = 0; k < cols; k++) {
                            ans.array[i][j] += (array[i][k] * dest.array[k][j]) % MODULO;
                            ans.array[i][j] %= MODULO;
                        }
                    }
                }

                return ans;
            }

            public Matrix power(long p) {
                int n = rows;
                Matrix base = new Matrix(n, n);
                for (int i = 0; i < n; i++) {
                    for (int j = 0; j < n; j++) {
                        base.array[i][j] = array[i][j];
                    }
                }

                Matrix ans = new Matrix(n, n);
                for (int i = 0; i < n; i++) {
                    ans.array[i][i] = 1L;
                }

                while (p > 0) {
                    if ((p & 1) > 0) {
                        ans = ans.multiply(base);
                    }
                    base = base.multiply(base);
                    p >>= 1;
                }

                return ans;
            }

        }

        class Line implements Comparable<Line> {
            long a;
            long b;
            int c;

            public Line(long a, long b, int c) {
                this.a = a;
                this.b = b;
                this.c = c;
            }


            public int compareTo(Line o) {
                return a < o.a ? -1 : 1;
            }

        }

    }

    static class InputReader {
        private InputStream stream;
        private byte[] buf = new byte[1024];
        private int curChar;
        private int numChars;

        public InputReader(InputStream stream) {
            this.stream = stream;
        }

        public int read() {
            if (numChars == -1)
                throw new InputMismatchException();
            if (curChar >= numChars) {
                curChar = 0;
                try {
                    numChars = stream.read(buf);
                } catch (IOException e) {
                    throw new InputMismatchException();
                }
                if (numChars <= 0)
                    return -1;
            }
            return buf[curChar++];
        }

        public int readInt() {
            int c = read();
            while (isSpaceChar(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            int res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public long readLong() {
            int c = read();
            while (isSpaceChar(c))
                c = read();
            int sgn = 1;
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            long res = 0;
            do {
                if (c < '0' || c > '9')
                    throw new InputMismatchException();
                res *= 10;
                res += c - '0';
                c = read();
            } while (!isSpaceChar(c));
            return res * sgn;
        }

        public static boolean isSpaceChar(int c) {
            return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
        }

    }
}
