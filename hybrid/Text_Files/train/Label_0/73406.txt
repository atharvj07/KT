t = int(input())
while t:
	t -= 1
	n = input()
	s1 = ''
	s2 = ''
	for i in range(len(n)):
		if i % 2 == 0:
			s1 += str(n[i])
		else:
			s2 += str(n[i])
	i = j = 0
	while i < len(s1) and s1[i] == '0':
		i += 1
	while j < len(s2) and s2[j] == '0':
		j += 1
	s1 = s1[i:]
	s2 = s2[j:]
	if len(s1) == 0:
		s1 = 0
	else:
		s1 = int(s1)
	if len(s2) == 0:
		s2 = 0
	else:
		s2 = int(s2)
	print((int(s1) + 1) * (int(s2) + 1) - 2)
