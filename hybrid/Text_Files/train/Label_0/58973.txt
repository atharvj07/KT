import	java.util.*;
public	class	Main{
	static	int	MCntNb(String	x,int	p,String	y){
		int	s=0,a=x.length(),b=y.length(),c,n=p*a,z,i=-1;
		for(int[]E=new	int[a],S=new	int[a];;E[c]=i,S[c]=s){
			for(z=0;b>z;++z){
				for(c=y.codePointAt(z);n>++i&&x.codePointAt(i%a)!=c;);
				if(n==i)	return	s;
			}
			++s;
			if(0<S[c=i%a]){
				z=(n-i-1)/(i-E[c]);s+=z*(s-S[c]);i+=z*(i-E[c]);
				break;
			}
		}
		for(;;++s)	for(z=0;b>z;++z){
			for(c=y.codePointAt(z);n>++i&&x.codePointAt(i%a)!=c;);
			if(n<=i)	return	s;
		}
	}
	static	int	CntNb(String	x,int	p,String	y){
		int	a=x.length(),b=y.length(),c,n=p*a,z=0,i,j=-1;
		for(;b>z;++z){
			for(c=y.codePointAt(z),i=Math.min(n,a+1+j);i>++j&&x.codePointAt(j%a)!=c;);
			if(i==j)	return	0;
		}
		if(a>j)	return	MCntNb(x,p,y);
		for(n=j/a,i=1+j;0<z;)
			for(c=y.codePointAt(--z);x.codePointAt(--i%a)!=c;);
		return	j%a<i?(p-1)/n:p/(1+n);
	}
	public	static	void	main(String[]args){
		Scanner	cin=new	Scanner(System.in);
		int	a=cin.nextInt(),b=cin.nextInt();
		String	z=cin.next();
		System.out.print(CntNb(z,a,cin.next())/b);
	}
}
