#include <bits/stdc++.h>
const int maxn = 2100;
const int d = 1000;
long long low[maxn], upp[maxn];
int n, x, y;
int main() {
  scanf("%d", &n);
  while (n--) {
    scanf("%d%d", &x, &y);
    low[x + y]++;
    upp[x - y + d]++;
  }
  long long ans = 0;
  for (int i = 0; i < maxn; i++) {
    ans += ((low[i] * (low[i] - 1LL)) >> 1) + ((upp[i] * (upp[i] - 1LL)) >> 1);
  }
  printf("%I64d\n", ans);
  return 0;
}
