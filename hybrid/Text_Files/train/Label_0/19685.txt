using namespace std;


int main()
{
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  string s;
  cin >> s;
  char now = 'A';
  string ans;
  rep(i,sz(s)){
    if (s[i] == now) {
      ans += now;
      if (now == 'A') {
        now = 'Z';
      }else{
        now = 'A';
      }
    }
  }
  bool flag = ans[sz(ans) - 1] == 'Z';
  if (!flag) {
    string new_ans;
    rep(i,sz(ans)-1){
      new_ans += ans[i];
    }
    ans = new_ans;
  }
  
  if (sz(ans) > 0) {
    cout << ans << endl;
  }else{
    cout << -1 << endl;
  }
  return 0;
}
