	static void solve() {
		int n=read();
		int m=read();
		long []a=readLongArray(n);
		long []b=readLongArray(m);
		long g=a[1]-a[0];
		for(int i=2;i<n;i++) {
			g=gcd(g,a[i]-a[i-1]);
		}
		for(int i=0;i<m;i++) {
			if(g%b[i]==0) {
				println("YES");
				print(a[0]);
				println(i+1);
				return;
			}
		}
		println("NO");
	}

		int b[]=new int[a.length];;
