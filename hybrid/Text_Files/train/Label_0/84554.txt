import string
import sys
input = sys.stdin.readline

def getInts():
	return [int(s) for s in input().split()]

def getInt():
	return int(input())

def getStrs():
	return [s for s in input().split()]

def getStr():
	return input().strip()

def listStr():
	return list(input().strip())
import collections as col
import math, itertools

def solve():
	N = getInt()
	F = [0] + getInts()
	F_set = set(F)
	F_set.remove(0)
	friends = set([j + 1 for j in range(N)]) - F_set
	indices = set([j for j in range(1, N + 1) if F[j] == 0])
	overlap = friends.intersection(indices)
	L = len(overlap)
	if L > 1:
		overlap = list(overlap)
		for j in range(len(overlap)):
			F[overlap[(j + 1) % L]] = overlap[j]
			friends.remove(overlap[j])
			indices.remove(overlap[j])
	if L == 1:
		tmp = friends - overlap
		friend = next(iter(tmp))
		index = next(iter(overlap))
		F[index] = friend
		friends.remove(friend)
		indices.remove(index)
	while friends:
		friend = next(iter(friends))
		index = next(iter(indices))
		F[index] = friend
		friends.remove(friend)
		indices.remove(index)
	print(*F[1:])
	return
solve()
