prime = [1 for i in range(1000002)]
p = 2
(prime[0], prime[1]) = (0, 0)
while p * p <= 1000001:
	if prime[p]:
		for i in range(p * p, 1000002, p):
			prime[i] = 0
	p += 1
for _ in range(int(input())):
	(n, e) = map(int, input().split())
	arr = list(map(int, input().split()))
	ans = 0
	for i in range(n):
		if prime[arr[i]]:
			j = i + e
			c = 0
			while j < n and arr[j] == 1:
				c += 1
				j += e
			j = i - e
			d = 0
			while j >= 0 and arr[j] == 1:
				d += 1
				j -= e
			ans = ans + c + (c + 1) * d
	print(ans)
