import sys

def inp():
	return sys.stdin.readline().rstrip('\r\n')

def N():
	return int(inp())

def testcase(t):
	for p in range(t):
		solve()
bad = set()
for i in range(1, 40, 2):
	bad.add(2 ** i)

def solve():
	n = N()
	if n % 2:
		print('Bob')
		return
	if n in bad:
		print('Bob')
		return
	print('Alice')
testcase(N())
