#include <bits/stdc++.h>
using namespace std;
set<int> s;
int fa[10005];
int get(int x) {
  if (fa[x] == x) return x;
  return fa[x] = get(fa[x]);
}
void unite(int x, int y) {
  x = get(x);
  y = get(y);
  if (x == y) return;
  fa[x] = y;
}
int main() {
  int n, i, num;
  while (scanf("%d", &n) != EOF) {
    for (i = 1; i <= n; i++) fa[i] = i;
    for (i = 1; i <= n; i++) {
      scanf("%d", &num);
      unite(i, num);
    }
    s.clear();
    for (i = 1; i <= n; i++) s.insert(get(i));
    printf("%d\n", s.size());
    fflush(stdout);
  }
  return 0;
}
