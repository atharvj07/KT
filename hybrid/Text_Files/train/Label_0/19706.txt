#include <bits/stdc++.h>
using namespace std;
int main(int argc, char const *argv[]) {
  string str = "";
  string ncat[6] = {"cool", "not bad", "don't touch me!", "don't think so",
                    "great"};
  string gcat[6] = {"terrible",         "worse",      "go die in a hole",
                    "are you serious?", "don't even", "no way"};
  for (int i = 0; i <= 9; i++) {
    cout << i << endl;
    str = "";
    getline(cin, str);
    if (str == "no") {
      continue;
    }
    for (int j = 0; j < 5; ++j) {
      if (str == ncat[j]) {
        cout << "normal" << endl;
        return 0;
      }
    }
    for (int j = 0; j < 6; ++j) {
      if (str == gcat[j]) {
        cout << "grumpy" << endl;
        return 0;
      }
    }
  }
  return 0;
}
