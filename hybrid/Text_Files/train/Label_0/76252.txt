N = 200005
n = int(input())
A = input().split()
A = list(map(lambda x: int(x), A))
pref = [0] * N
for a in A:
	pref[a] += 1
for i in range(1, N):
	pref[i] += pref[i - 1]
ans = 0
for i in range(1, N):
	if pref[i] != pref[i - 1]:
		suma = 0
		for j in range(i, N, i):
			suma += j * (pref[min(N - 1, j + i - 1)] - pref[j - 1])
		ans = max(ans, suma)
print(ans)
