#include<bits/stdc++.h>
#define fori(x) for(int i=0;i<x;i++)
#define for1(i,x) for(int i=0;i<x;i++)
using namespace std;
int main(){
	int t0;cin>>t0;
	while(t0--){
		int n;cin>>n;
		int mn=1e9+5,num=0;
		fori(n){
			int x;cin>>x;
			if(x<mn){
				mn=x;
				num=1;
			}
			else if(x==mn){
				num++;
			}
		}
		cout<<(n-num)<<"\n";
	}
	return 0;
}
