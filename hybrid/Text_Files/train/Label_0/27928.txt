#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 7;
char s[N];
int d[N];
bool mak[30];
long long cnt[30];
int main() {
  scanf("%s", s);
  int len = strlen(s);
  int ct = 0;
  for (int i = 0; i < len; i++) {
    if (!mak[s[i] - 'a']) {
      mak[s[i] - 'a'] = 1;
      ct++;
    }
    d[i] = ct;
  }
  int ct1;
  for (int i = 0; i < len; i++) {
    int ct = 0;
    if (i != 0) {
      if (s[i] == s[i - 1]) {
        ct1++;
      } else {
        ct1 = 1;
      }
    } else {
      ct1 = 1;
    }
    cnt[1] += (ct1 - 1);
    memset(mak, 0, sizeof(mak));
    for (int j = i - ct1 + 1; j >= 0; j--) {
      if (!mak[s[j] - 'a']) {
        ct++;
        mak[s[j] - 'a'] = 1;
      }
      cnt[ct]++;
      if (d[i] == ct) {
        cnt[ct] += j;
        break;
      }
    }
  }
  printf("%d\n", ct);
  for (int i = 1; i <= ct; i++) {
    printf("%I64d\n", cnt[i]);
  }
}
