#include <bits/stdc++.h>
using namespace std;
const int maxn = (int)(1e6 + 6);
const long long mod = (long long)(998244353);
pair<int, int> a[maxn];
int fact[maxn], inv_fact[maxn];
inline int mul(long long a, long long b) { return (a * b) % mod; }
inline int add(int a, int b) {
  int ret = a + b;
  if (ret >= mod) {
    ret -= mod;
  }
  return ret;
}
inline int sub(int a, int b) {
  int ret = a - b;
  if (ret < 0) {
    ret += mod;
  }
  return ret;
}
inline int poww(long long a, long long b) {
  int x = 1, y = a;
  while (b > 0) {
    if (b % 2) {
      x = mul(x, y);
    }
    y = mul(y, y);
    b /= 2;
  }
  return (int)(x % mod);
}
void build() {
  fact[0] = 1;
  for (int i = 1; i < maxn; i++) {
    fact[i] = mul(fact[i - 1], i);
  }
  inv_fact[maxn - 1] = poww(fact[maxn - 1], mod - 2);
  for (int i = maxn - 2; i >= 0; i--) {
    inv_fact[i] = mul(inv_fact[i + 1], i + 1);
  }
}
int C(int n, int r) {
  if (n < r || min(n, r) < 0) {
    return 0;
  }
  int val1 = fact[n], val2 = mul(inv_fact[r], inv_fact[n - r]);
  return mul(val1, val2);
}
int main() {
  ios_base::sync_with_stdio(0);
  build();
  build();
  int n, res = 0;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    int curr = mul(C(n, i), fact[i]);
    curr = mul(curr, sub(fact[n - i], 1));
    res = add(res, curr);
  }
  res = add(res, fact[n]);
  cout << res << endl;
  return 0;
}
