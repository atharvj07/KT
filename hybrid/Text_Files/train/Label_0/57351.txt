    class Line{ //y = ax + b
        long a;
        long b;
 
        public Line(long a, long b) {
            this.a = a;
            this.b = b;
        }
        
        long f(long x) {
            return a * x + b;
        }
    }
    
    int MX = 100000;
    
    void solve()  {

        
        int n = sc.nextInt();
        long[] a = new long[n];
        for(int i = 0; i < n; i++) a[i] = sc.nextLong();
        
        int m = sc.nextInt();
        
        int[][] queries = new int[m][];
        
        for(int i = 0; i < m; i++) queries[i] = new int[] {sc.nextInt(), sc.nextInt(), i};
        
        Arrays.sort(queries, Comparator.comparing((int[]arr) -> arr[1]));
                
        long[] ans = new long[m];
        
        t = new Line[4*n];
        Arrays.fill(t, new Line(0, 1L<<60));
        
        long sum = 0L;
        ArrayDeque<Long> coefs = new ArrayDeque<>();
        int pos = 0;
        for(int j = 0; j < n; j++) {
            while(!coefs.isEmpty() && coefs.getLast() >= a[j]) {
                coefs.removeLast();
                undo();
            }
            sum += a[j];
            changes.add(new State(-1, null));
            coefs.addLast(a[j]);
            insert(2-n, 1, 0, new Line(a[j], j*a[j] - sum));
            while(pos < m && queries[pos][1] == j+1) {
                ans[queries[pos][2]] = query(2-n, 1, 0, queries[pos][0]-j) + sum;
                pos++;
            }
        }
        for(long x : ans) println(x);
    }
    
    Line[] t;
    
    ArrayDeque<State> changes = new ArrayDeque<>();
    
    void undo() {
        while(changes.getLast().line != null) {
            State change = changes.removeLast();
            t[change.i] = change.line;
        }
        changes.removeLast();
    }
    
    void insert(int l, int r, int i, Line line) {
        if(l + 1 == r) {
            if(line.f(l) < t[i].f(l)) {
                changes.addLast(new State(i, t[i]));
                t[i] = line;
            }
            return;
        }
        
        int mid = (l + r) / 2;
        
        if(t[i].a < line.a) {
            changes.addLast(new State(i, t[i]));
            Line temp = line; line = t[i]; t[i] = temp;

        }
        if(t[i].f(mid) > line.f(mid)) {
            changes.addLast(new State(i, t[i]));
            Line temp = line; line = t[i]; t[i] = temp;
            insert(l, mid, 2*i+1, line);
        } else {
            insert(mid, r, 2*i+2, line);
        }
    }
    
    long query(int l, int r, int i, int x) {
        if(l + 1 == r) {
            return t[i].f(x);
        }
        
        int mid = (l + r) / 2;
        if(x < mid) return Math.min(t[i].f(x), query(l, mid, 2*i+1, x));
        else return Math.min(t[i].f(x), query(mid, r, 2*i+2, x));
    }
    
    class State{
        int i;
        Line line;
        
        public State(int i_, Line line_) {
            i = i_;
            line = line_;
        }
    }

  static class InputReader {
      InputReader(InputStream in) { this.in = in; } InputStream in;
      
      private byte[] buf = new byte[16384];
      private int    curChar;
      private int    numChars;
      
 
      public int read() {
          if (numChars == -1)
              throw new InputMismatchException();
          if (curChar >= numChars) {
              curChar = 0;
              try {
                  numChars = in.read(buf);
              } catch (IOException e) {
                  throw new InputMismatchException();
              }
              if (numChars <= 0)
                  return -1;
          }
          return buf[curChar++];
      }
 
      public String nextLine() {
          int c = read();
          while (isSpaceChar(c))
              c = read();
          StringBuilder res = new StringBuilder();
          do {
              res.appendCodePoint(c);
              c = read();
          } while (!isEndOfLine(c));
          return res.toString();
      }
 
      public String nextString() {
          int c = read();
          while (isSpaceChar(c))
              c = read();
          StringBuilder res = new StringBuilder();
          do {
              res.appendCodePoint(c);
              c = read();
          } while (!isSpaceChar(c));
          return res.toString();
      }
 
      public long nextLong() {
          int c = read();
          while (isSpaceChar(c))
              c = read();
          int sgn = 1;
          if (c == '-') {
              sgn = -1;
              c = read();
          }
          long res = 0;
          do {
              if (c < '0' || c > '9')
                  throw new InputMismatchException();
              res *= 10;
              res += c - '0';
              c = read();
          } while (!isSpaceChar(c));
          return res * sgn;
      }
 
      public int nextInt() {
          int c = read();
          while (isSpaceChar(c))
              c = read();
          int sgn = 1;
          if (c == '-') {
              sgn = -1;
              c = read();
          }
          int res = 0;
          do {
              if (c < '0' || c > '9')
                  throw new InputMismatchException();
              res *= 10;
              res += c - '0';
              c = read();
          } while (!isSpaceChar(c));
          return res * sgn;
      }
 
      private boolean isSpaceChar(int c) {
          return c == ' ' || c == '\n' || c == '\r' || c == '\t' || c == -1;
      }
 
      private boolean isEndOfLine(int c) {
          return c == '\n' || c == '\r' || c == -1;
      }
  }
