import java.util.*;

public class Main
{
    public static void main(String[] args)
    {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.next();
        
        int n = s.length();
        char cur = 0;
        for (int i = 0; i < n; i++) 
        {
            int diff = (int)cur - (int)s.charAt(i);
            diff += 256;
            diff %= 256;
            if (diff == 0)
		diff = 256;
		
            System.out.printf("...\n");
		
            for (int j = 1; j < diff; j++) 
            {
		System.out.printf("...\n");
		System.out.printf(".X.\n");
		System.out.printf("...\n");
            }
		
            System.out.printf("...\n");
            System.out.printf(".X.\n");
            System.out.printf(".X.\n");
            System.out.printf("...\n");
		
            cur = s.charAt(i);
	}
    }
}
