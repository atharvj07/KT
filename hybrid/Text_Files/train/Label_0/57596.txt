import java.io.*;
import java.util.*;

public class a1 {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = in.nextInt();
		String s = in.next();
		
		long ans = 0;
		int curx = 0;
		int cury = 0;
		
		for (int i = 0; i < n; i++) {
			curx = 0;
			cury = 0;
			
			for (int j = i; j < n; j++) {
				if (s.charAt(j) == 'U'){
					curx++;
				}
				if (s.charAt(j) == 'D'){
					curx--;
				}
				if (s.charAt(j) == 'L'){
					cury--;
				}
				if (s.charAt(j) == 'R'){
					cury++;
				}
				
				if (Math.abs(curx) + Math.abs(cury) == 0){
					ans++;
				}
			}
		}
		
		out.println(ans);
		
		
		out.close();
	}
}
