t = int(input())
for i in range(t):
	n = int(input())
	l = input().split()
	total = 0
	pair = 0
	odd = 0
	for i in l:
		total += int(i)
	if total % 2 != 0:
		print('YES')
	else:
		for i in l:
			if int(i) % 2 == 0:
				pair = 1
			else:
				odd = 1
			if pair != 0 and odd != 0:
				print('YES')
				break
		if odd == 0 or pair == 0:
			print('NO')
