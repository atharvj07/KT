#include<bits/stdc++.h>
using namespace std;
bool used[2000001];
int n,h,w,x;
int main(){
  cin>>n>>h>>w;
  for(int i=0;i<n;i++){
    cin>>x;
    if(i%2==0)for(int j=(w)*(i)+x;j<(w)*(i)+x+w;j++){
	if(j>=0&&j<=w*n)used[j]=1;
    }
    else for(int j=w*(i)-x;j<w*(i)-x+w;j++){
      if(j>=0&&j<=w*n)used[j]=1;
    }
  }
  int sum=0;
  for(int i=0;i<w*n;i++){
    if(!used[i])sum+=h;
  }
  cout<<sum<<endl;
}
