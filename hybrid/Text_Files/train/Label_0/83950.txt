def solve(s):
	alice = 0
	bob = 0
	s = list(s)
	print(s)
	can_reverse = False
	last_index = None
	index_with_0 = []
	is_alice_turn = True
	for i in range(len(s) // 2 + 1):
		if s[i] == '0':
			index_with_0.append(i)
	if len(s) == 1:
		if s[0] == '1':
			print('DRAW')
		else:
			print('BOB')
		return
	while len(index_with_0) > 0:
		if can_reverse:
			s = s[::-1]
			can_reverse = False
		else:
			if is_alice_turn:
				alice += 1
			else:
				bob += 1
			if last_index == None:
				if len(s) % 2 != 0:
					if s[len(s) // 2] == '0':
						s[len(s) // 2] = '1'
						last_index = None
						can_reverse = True
				else:
					i = index_with_0.pop()
					last_index = i
					s[i] = '1'
					can_reverse = False
			else:
				s[last_index] = '1'
				last_index = None
				can_reverse = True
	if alice == bob:
		print('DRAW')
	elif alice > bob:
		print('BOB')
	else:
		print('ALICE')

def solve2(s):
	n = s.count('0')
	if n == 0:
		print('DRAW')
	elif n == 1:
		print('BOB')
	elif n % 2 == 0:
		print('BOB')
	else:
		print('ALICE')
t = int(input())
for _ in range(t):
	n = int(input())
	s = input()
	solve2(s)
