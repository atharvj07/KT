globalArr = []

def Get_binary_chain(arr):
	binaryChain = [0 for j in range(len(arr))]
	for i in range(len(arr)):
		value = arr[i]
		if value != 0:
			binaryChain[value - 1] = 1
	return binaryChain

def Get_chain(arr, index):
	chain = []
	chain.append(index + 1)
	while arr[index] != 0:
		index = arr[index] - 1
		chain.append(index + 1)
	return chain

def Main():
	num = int(input())
	arr = list(map(int, input().split()))
	binaryChain = Get_binary_chain(arr)
	for i in range(len(arr)):
		if binaryChain[i] == 0:
			globalArr.append(Get_chain(arr, i))
	for i in range(len(globalArr) - 1):
		arr[globalArr[i][-1] - 1] = globalArr[i + 1][0]
	arr[globalArr[-1][-1] - 1] = globalArr[0][0]
	for i in arr:
		print(f'{i} ', end='')
Main()
