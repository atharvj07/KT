#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b, c, sum = 0, a1, b1, c1;
  scanf("%d", &a);
  scanf("%d", &b);
  scanf("%d", &c);
  sum += a / 3 + b / 3 + c / 3;
  a1 = a % 3;
  b1 = b % 3;
  c1 = c % 3;
  sum += min(min(a1, b1), c1);
  if ((a1 == 2 && b1 == 2 && c1 == 0 && c > 0) ||
      (a1 == 2 && b1 == 0 && c1 == 2 && b > 0) ||
      (a1 == 0 && b1 == 2 && c1 == 2 && a > 0))
    sum++;
  printf("%d\n", sum);
}
