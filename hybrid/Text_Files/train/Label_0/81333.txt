from collections import Counter
from itertools import accumulate, product, groupby
import operator
import math
from functools import reduce

def solve():
	n = int(input())
	arr = [int(char) for char in input().split()]
	(even, odd) = ([], [])
	for a in arr:
		if a & 1:
			odd.append(a)
		else:
			even.append(a)
	(E, O) = (len(even), len(odd))
	ans = 0
	for j in range(1, E + 1):
		ans += n - j
	for (i, num) in enumerate(odd):
		for j in range(i + 1, O):
			if math.gcd(num, odd[j]) > 1:
				ans += 1
	return ans
for _ in range(int(input())):
	print(solve())
