#include<bits/stdc++.h>
#define ll long long
using namespace std;

int main()
{
	ll n,k,r,s,p;
	cin>>n>>k>>r>>s>>p;
	string t;
	cin>>t;

	
	ll sum = 0;
	for(ll i=0;i<n;i++)
	{
		if(i>=k && t[i]==t[i-k])
		{
			t[i] = ' ';
			continue;
		}
		else if(t[i]=='p')
		{
			sum += s;
		}
		else if(t[i]=='s')
		{
			sum += r;
		}
		else if(t[i]=='r')
		{
			sum += p;
		}
	}
	
	
	cout<<sum;
}
