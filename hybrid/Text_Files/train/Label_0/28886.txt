#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1000005;
int N, C, A, B;
int L[MAXN], R[MAXN], Rmin[MAXN];
string S;
vector<int> res;
int dfs(int v, int r) {
  if (L[v] >= 0) {
    L[v] = max(dfs(v + 1, L[v]), L[v]);
  }
  L[v] = max(L[v], v);
  if (L[v] >= Rmin[v]) {
    cout << "IMPOSSIBLE\n";
    exit(0);
  }
  res.push_back(v);
  R[v] = max(R[v], r);
  if (R[v] > L[v]) {
    return dfs(L[v] + 1, R[v]);
  }
  return L[v];
}
int main() {
  ios::sync_with_stdio(0);
  cin >> N >> C;
  memset(L, -1, sizeof(L));
  memset(R, -1, sizeof(R));
  memset(Rmin, 63, sizeof(Rmin));
  for (int i = 0; i < C; i++) {
    cin >> A >> B >> S;
    A -= 1, B -= 1;
    if (A >= B) {
      cout << "IMPOSSIBLE\n";
      return 0;
    }
    if (S == "LEFT") {
      L[A] = max(L[A], B);
    } else {
      R[A] = max(R[A], B);
      Rmin[A] = min(Rmin[A], B);
    }
  }
  dfs(0, N - 1);
  for (int v : res) {
    cout << v + 1 << ' ';
  }
  cout << '\n';
}
