#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 100, M = 1e6 + 100, LG = 30, base = 2;
const int MOD = 1e9 + 7, mod = 1e9 + 9;
const long long INF = 1e9, inf = 1e18;
pair<long long, pair<long long, pair<long long, long long> > > edge[N];
vector<pair<long long, long long> > adj[N];
short ans[N];
long long st[N], num[N], par[N], siz[N], k, stt;
bool mark[N], mark3[N];
void Cutedge(int v, int parr, int ind) {
  mark3[v] = 1, st[v] = stt++;
  bool flag = 0;
  int kk = 0;
  for (int i = 0; i < adj[v].size(); i++) {
    int u = adj[v][i].first, indexx = adj[v][i].second;
    if (!mark3[u])
      kk += k, k = 0, Cutedge(u, v, indexx);
    else {
      if ((u == parr && !flag) || st[u] > st[v])
        int temp = 0;
      else
        num[v]++, num[u]--;
      if (u == parr) flag = 1;
    }
  }
  k += num[v] + kk;
  if (k == 0 && ans[ind] != 1 && v != parr) ans[ind] = 3;
  return;
}
int getpar(int v) {
  if (par[v] == -1) return v;
  return par[v] = getpar(par[v]);
}
void mergee(int a, int b) {
  a = getpar(a), b = getpar(b);
  if (a == b) return;
  if (siz[a] > siz[b]) swap(a, b);
  par[a] = b, siz[b] += siz[a];
  return;
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
  ;
  int n, m, x, y, w;
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    cin >> x >> y >> w;
    edge[i] = make_pair(w, make_pair(i, make_pair(--x, --y)));
  }
  sort(edge, edge + m);
  fill(par, par + n, -1);
  fill(siz, siz + n, 1);
  for (int i = 0; i <= m; i++) {
    int u = edge[i].second.second.first, v = edge[i].second.second.second,
        ind = edge[i].second.first;
    if ((i && edge[i].first != edge[i - 1].first) || i == m) {
      for (int j = i - 1; j >= 0 && !mark[j]; j--) {
        int x = getpar(edge[j].second.second.first),
            y = getpar(edge[j].second.second.second);
        adj[x].push_back(make_pair(y, edge[j].second.first));
        adj[y].push_back(make_pair(x, edge[j].second.first));
      }
      for (int j = i - 1; j >= 0 && !mark[j]; j--) {
        if (!mark3[getpar(edge[j].second.second.second)]) {
          int t = getpar(edge[j].second.second.second);
          Cutedge(t, t, 0);
          k = 0;
        }
      }
      for (int j = i - 1; j >= 0 && mark[j] == 0; j--) {
        int uu = getpar(edge[j].second.second.second);
        int vv = getpar(edge[j].second.second.first);
        adj[uu].clear(), adj[vv].clear();
        mark3[uu] = 0, mark3[vv] = 0;
        st[uu] = 0, st[vv] = 0;
        num[uu] = 0, num[vv] = 0;
      }
      for (int j = i - 1; j >= 0 && !mark[j]; j--)
        mark[j] = 1,
        mergee(edge[j].second.second.second, edge[j].second.second.first);
    }
    if (i < m)
      if (getpar(u) == getpar(v))
        ans[ind] = 1;
      else
        ans[ind] = 2;
  }
  for (int i = 0; i < m; i++) {
    if (ans[i] == 1)
      cout << "none\n";
    else if (ans[i] == 2)
      cout << "at least one\n";
    else
      cout << "any\n";
  }
  return 0;
}
