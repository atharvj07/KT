#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
const int MOD = 1e9 + 7;
const int INF = 0x3f3f3f3f;
long long t;
struct Point {
  long long x, y;
} P0, a, b, St, Pre;
vector<Point> vp;
void init() {
  vp.clear();
  vp.clear();
}
long long dist(Point &p1, Point &p2) {
  long long dx = p1.x - p2.x;
  long long dy = p1.y - p2.y;
  return abs(dx) + abs(dy);
}
int main() {
  scanf("%lld%lld%lld%lld%lld%lld", &P0.x, &P0.y, &a.x, &a.y, &b.x, &b.y);
  scanf("%lld%lld%lld", &St.x, &St.y, &t);
  Pre = P0;
  if (dist(P0, St) <= t) vp.emplace_back(P0);
  while (true) {
    Point newp;
    newp.x = a.x * Pre.x + b.x;
    newp.y = a.y * Pre.y + b.y;
    if (dist(newp, St) <= t) vp.emplace_back(newp);
    if (newp.x > 3e16 || newp.y > 3e16) break;
    Pre = newp;
  }
  int sz = vp.size();
  long long tmp_d;
  int ans = 0;
  for (int l = 0; l < sz; ++l) {
    for (int r = l; r < sz; ++r) {
      tmp_d = dist(vp[l], vp[r]) + min(dist(St, vp[l]), dist(St, vp[r]));
      if (tmp_d <= t) {
        ans = max(ans, r - l + 1);
      }
    }
  }
  printf("%d\n", ans);
  return 0;
}
