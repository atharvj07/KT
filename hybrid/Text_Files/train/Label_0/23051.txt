#include <bits/stdc++.h>
using namespace std;
template <typename T>
void get_int(T &x) {
  char t = getchar();
  bool neg = false;
  x = 0;
  for (; (t > '9' || t < '0') && t != '-'; t = getchar())
    ;
  if (t == '-') neg = true, t = getchar();
  for (; t <= '9' && t >= '0'; t = getchar()) x = x * 10 + t - '0';
  if (neg) x = -x;
}
template <typename T>
void print_int(T x) {
  if (x < 0) putchar('-'), x = -x;
  short a[20] = {}, sz = 0;
  while (x > 0) a[sz++] = x % 10, x /= 10;
  if (sz == 0) putchar('0');
  for (int i = sz - 1; i >= 0; i--) putchar('0' + a[i]);
}
const int inf = 0x3f3f3f3f;
const long long Linf = 1ll << 61;
const double pi = acos(-1.0);
struct matrix {
  long long a[201][201];
  void clear() {
    for (int i = 0; i < 201; i++)
      for (int j = 0; j < 201; j++) a[i][j] = -Linf;
  }
  matrix() { clear(); }
} trans, cur;
matrix operator*(const matrix a, const matrix b) {
  matrix ret;
  for (int i = 0; i < 201; i++)
    for (int j = 0; j < 201; j++)
      for (int k = 0; k < 201; k++)
        ret.a[i][j] = max(ret.a[i][j], a.a[i][k] + b.a[k][j]);
  return ret;
}
matrix power(matrix x, long long y) {
  matrix ret = x;
  y--;
  while (y) {
    if (y & 1) ret = ret * x;
    x = x * x;
    y >>= 1;
  }
  return ret;
}
struct automata {
  int ch[233][26], f[233], word[233], cnt, root;
  int newnode() {
    word[cnt] = 0;
    f[cnt] = 0;
    memset(ch[cnt], 0, sizeof(ch[cnt]));
    return cnt++;
  }
  automata() {
    cnt = 0;
    root = newnode();
  }
  void addstring(char *s, int val) {
    int cur = root, n = strlen(s + 1);
    for (int i = 1; i <= n; i++) {
      if (!ch[cur][s[i] - 'a']) ch[cur][s[i] - 'a'] = newnode();
      cur = ch[cur][s[i] - 'a'];
    }
    word[cur] += val;
  }
  void construct() {
    queue<int> q;
    f[root] = 0;
    for (int c = 0; c < 26; c++) {
      int u = ch[0][c];
      if (u) {
        f[u] = 0;
        q.push(u);
      }
    }
    while (!q.empty()) {
      int x = q.front();
      q.pop();
      for (int c = 0; c < 26; c++) {
        int u = ch[x][c], v;
        if (!u) continue;
        q.push(u);
        v = f[x];
        while (v && !ch[v][c]) v = f[v];
        f[u] = ch[v][c];
        word[u] += word[f[u]];
      }
    }
  }
  int getnext(int cur, int c) {
    int j = cur;
    while (j && !ch[j][c]) j = f[j];
    j = ch[j][c];
    return j;
  }
  void buildmat(matrix &t, matrix &cur) {
    t.clear();
    cur.clear();
    for (int i = 0; i < cnt; i++)
      for (int j = 0; j < cnt; j++) t.a[i][j] = cur.a[i][j] = -Linf;
    for (int i = 0; i < cnt; i++)
      for (int j = 0; j < 26; j++) {
        int nxt = getnext(i, j);
        t.a[nxt][i] = max(t.a[nxt][i], word[nxt] * 1ll);
      }
    cur.a[root][0] = 0;
  }
} ac;
char s[233];
int n, a[233];
long long l;
int main() {
  get_int(n), get_int(l);
  for (int i = 1; i <= n; i++) get_int(a[i]);
  for (int i = 1; i <= n; i++) {
    scanf("%s", s + 1);
    ac.addstring(s, a[i]);
  }
  ac.construct();
  ac.buildmat(trans, cur);
  cur = power(trans, l) * cur;
  long long ans = 0;
  for (int i = 0; i < ac.cnt; i++) ans = max(ans, cur.a[i][0]);
  print_int(ans), puts("");
  return 0;
}
