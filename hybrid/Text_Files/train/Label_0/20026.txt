#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int a = 0, g = 0, t = 0, c = 0;
  if (n % 4 != 0)
    cout << "===";
  else {
    for (int i = 0; i < n; i++) {
      if (s[i] == 'A') a++;
      if (s[i] == 'G') g++;
      if (s[i] == 'T') t++;
      if (s[i] == 'C') c++;
    }
    if (a > n / 4 || g > n / 4 || t > n / 4 || c > n / 4)
      cout << "===";
    else {
      for (int i = 0; i < n; i++) {
        if (s[i] == '?') {
          if (a < n / 4) {
            s[i] = 'A';
            a++;
          } else if (g < n / 4) {
            s[i] = 'G';
            g++;
          } else if (t < n / 4) {
            s[i] = 'T';
            t++;
          } else if (c < n / 4) {
            s[i] = 'C';
            c++;
          }
        }
      }
      cout << s;
    }
  }
  return 0;
}
