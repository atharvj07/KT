#include<iostream>
#include<vector>

using namespace std;

int main()
{
	int n;
	cin >> n;
	vector<int> p(n),a(n),b(n);
	for(int i=0;i<n;i++)
		cin >> p[i];
	
	for(int i=0;i<n;i++)
	{
		a[i]=i+1;
		b[i]=n-i;
	}

	for(int i=1;i<=n;i++)
	{
		for(int j=p[i-1];j<=n;j++)
			a[j-1]+=i;
		for(int j=1;j<=p[i-1];j++)
			b[j-1]+=i;
	}
	for(int i=0;i<n;i++)
		cout << a[i] << " ";
	cout << endl;
	for(int i=0;i<n;i++)
		cout << b[i] << " ";
	cout << endl;
}
