l = len(s)
p = x2 - x1
if p > 0:
	r = 'E'
else:
	r = 'W'
q = y2 - y1
if q > 0:
	t = 'N'
else:
	t = 'S'
f = False

d = 0
for i in range(l):
	if s[i] == r:
		c += 1
	if s[i] == t:
		d += 1
	if c >= abs(p) and d >= abs(q):
		f = True
		break
