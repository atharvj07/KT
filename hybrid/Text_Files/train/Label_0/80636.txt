[h, m, s, t1, t2] = list(map(int, input().split()))
if h == 12:
	h = 0
if t1 == 12:
	t1 = 0
if t2 == 12:
	t2 = 0
if t1 > t2:
	(t1, t2) = (t2, t1)
sa = s * 360 * 12 * 60
ma = m * 360 * 12 * 60 + s * 360 * 12
ha = h * 360 * 60 * 60 + m * 360 * 60 + s * 360

def check(x):
	if t1 * 360 / 12 * 12 * 60 * 60 < x and x < t2 * 360 / 12 * 12 * 60 * 60:
		return 1
	return -1
if check(sa) == check(ma) and check(sa) == check(ha):
	print('YES')
	exit(0)
print('NO')
