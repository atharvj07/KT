import math
r = 0
d = 0

def is_sausage_in_crust(x, y, ra):
	x = math.fabs(x)
	y = math.fabs(y)
	ra = math.fabs(ra)
	t = math.sqrt(x ** 2 + y ** 2)
	return float(r - d) <= t - ra and float(r) >= t + ra

def main():
	global r, d
	l = input().split(' ')
	r = math.fabs(int(l[0]))
	d = math.fabs(int(l[1]))
	count = 0
	n = int(input())
	for i in range(0, n):
		l = input().split(' ')
		if is_sausage_in_crust(int(l[0]), int(l[1]), int(l[2])):
			count += 1
	print(count)
main()
