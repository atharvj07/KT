def maxvlozh(stroka):
	stack = 0
	maxvl = 0
	for i in range(len(stroka)):
		if stroka[i] == '[':
			stack += 1
		else:
			stack -= 1
		if maxvl < stack:
			maxvl = stack
	return maxvl
