(n, k) = map(int, input().split())
ans = 0
if n == k:
	ans = 1
elif k == 1:
	ans = n
else:
	p = 2
	s = 2
	while s < k:
		p *= 2
		s += p
	n1 = n
	n -= k
	ans = (n + p - 1) // p
	ans *= 2
	flag = False
	for i in range(1, 64):
		if 2 ** i - 1 == k:
			flag = True
	flag1 = False
	if flag and n1 % p == p - 1:
		ans += 1
print(ans)
