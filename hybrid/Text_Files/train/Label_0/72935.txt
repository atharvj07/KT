import math
x4 = float('inf')
(n, k) = map(int, input().split())
for d1 in range(1, math.trunc(n ** 0.5) + 1):
	if n % d1 == 0:
		d2 = n / d1
		if (d1 * k + d2) // k * ((d1 * k + d2) % k) == n:
			x1 = d1 * k + d2
		else:
			x1 = float('inf')
		if (d2 * k + d1) // k * ((d2 * k + d1) % k) == n:
			x2 = d2 * k + d1
		else:
			x2 = float('inf')
		x3 = min(x1, x2)
		x4 = min(x3, x4)
print(math.trunc(x4))
