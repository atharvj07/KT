(n, b) = map(int, input().split())
num1 = 0
num2 = 0
n -= 1
for c in list(map(int, input().split())):
	num1 += b ** n * c
	n -= 1
(n, a) = map(int, input().split())
n -= 1
for c in list(map(int, input().split())):
	num2 += a ** n * c
	n -= 1
if num1 > num2:
	print('>')
elif num1 == num2:
	print('=')
else:
	print('<')
