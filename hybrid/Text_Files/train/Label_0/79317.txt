def main():
	n = int(input())
	w = [[0 for j in range(n)] for i in range(n)]
	d = [int(v) for v in input().split()]
	for i in range(n):
		w[i][i] = d[i]
	for s in range(n):
		meet = False
		for j in range(n - s):
			i = s + j
			if w[i][j] == s + 1:
				meet = True
			elif meet:
				w[i][j - 1] = w[i][j]
			else:
				w[i + 1][j] = w[i][j]
	for i in range(n):
		for j in range(i + 1):
			print(w[i][j], end=' ')
		print()
main()
