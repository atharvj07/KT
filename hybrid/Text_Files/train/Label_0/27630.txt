#include <bits/stdc++.h>
using namespace std;
int n;
int mat[101][101];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      cin >> mat[i][j];
    }
  }
  long long res = 0;
  for (int i = 0; i < n; ++i) {
    res += mat[i][i];
    res += mat[i][n - i - 1];
    res += mat[i][(n - 1) / 2] + mat[(n - 1) / 2][i];
  }
  if (n == 1)
    res = mat[0][0];
  else
    res -= 3 * mat[(n - 1) / 2][(n - 1) / 2];
  cout << res;
  return 0;
}
