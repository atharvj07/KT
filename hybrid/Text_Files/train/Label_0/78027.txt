R = lambda : map(int, input().split())
(n, k) = R()
p = sorted(zip(R(), R(), range(n)))
c = []
r = [0] * n
s = 0
for i in range(k):
	(_, b, j) = p[i]
	s += b
	r[j] = s
	c.append(b)
(_, b, j) = p[k]
r[j] = s + b
for i in range(k + 1, n):
	(_, b, j) = p[i - 1]
	c.sort()
	if c and b > c[0]:
		s += b - c[0]
		c[0] = b
	(_, b, j) = p[i]
	r[j] = b + s
print(*r)
