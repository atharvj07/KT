#define llint long long

using namespace std;

llint mod;

llint modpow(llint a, llint n)
{
	if(n == 0) return 1;
	if(n % 2){
		return ((a%mod) * (modpow(a, n-1)%mod)) % mod;
	}
	else{
		return modpow((a*a)%mod, n/2) % mod;
	}
}

struct UnionFind{
	int size;
	vector<int> parent;
	vector<llint> dif;
	
	UnionFind(){}
	UnionFind(int size){
		this->size = size;
		parent.resize(size+1);
		dif.resize(size+1);
		init();
	}
	void init(){
		for(int i = 0; i <= size; i++) parent[i] = i, dif[i] = 1;
	}
	int root(int i){
		if(parent[i] == i) return i;
		int res = root(parent[i]);
		dif[i] *= dif[parent[i]], dif[i] %= mod;
		return parent[i] = res;
	}
	bool same(int i, int j){
		return root(i) == root(j);
	}
	bool unite(int i, int j, llint d){
		int root_i = root(i), root_j = root(j);
		
		llint dif_i = dif[i], dif_j = dif[j];
		if(root_i == root_j){
			if(dif_i * d % mod != dif_j) return false;
			return true;
		}
		
		parent[root_i] = root_j;
		dif[root_i] = dif_j * modpow(dif_i * d % mod, mod-2) % mod;
		return true;
	}
};

struct edge{
	int to, cost;
	edge(int a, int b){
		to = a, cost = b;
	}
};

llint n, m;
llint u[100005], v[100005], w[100005];
UnionFind uf(100005);

int main(void)
{
	cin >> n >> m;
	for(int i = 1; i <= m; i++) cin >> u[i] >> v[i] >> w[i];
	
	mod = 1000000021;
	uf.init();
	for(int i = 1; i <= m; i++){
		if(!uf.unite(u[i], v[i], w[i])){
			cout << "No" << endl;
			return 0;
		}
	}
	
	mod = 1000000033;
	uf.init();
	for(int i = 1; i <= m; i++){
		if(!uf.unite(u[i], v[i], w[i])){
			cout << "No" << endl;
			return 0;
		}
	}
	
	cout << "Yes" << endl;
	
	return 0;
}
