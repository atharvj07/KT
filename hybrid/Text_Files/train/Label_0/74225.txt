import math
l = list(map(int, input().split()))
count = 0
s = l[0] + l[0] - 1
if l[1] > s:
	print('0')
else:
	k = l[1] / 2
	if k % 1 == 0:
		if k - 1 < l[0] - k:
			count += k - 1
		else:
			count += l[0] - k
		print(int(count))
	else:
		z1 = math.ceil(k)
		z2 = math.floor(k)
		if z2 - 1 < l[0] - z1:
			count += z2 - 1
		else:
			count += l[0] - z1
		print(count + 1)
