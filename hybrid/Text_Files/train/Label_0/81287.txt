import sys
(n, k) = map(int, input().split())
a = list(map(int, input().split()))
l = 0
r = n
while l < r:
	if l + 1 == r:
		try1 = r
		flag = 0
		tmp = 0
		grid = [0]
		for i in range(n):
			if a[i] >= try1:
				tmp += 1
			else:
				tmp -= 1
			grid.append(tmp)
		lowestprev = 0
		for i in range(k, n + 1):
			lowestprev = min(lowestprev, grid[i - k])
			if grid[i] - lowestprev > 0:
				flag = 1
				break
		if flag:
			l = r
		break
	try1 = (l + r) // 2
	flag = 0
	tmp = 0
	grid = [0]
	for i in range(n):
		if a[i] >= try1:
			tmp += 1
		else:
			tmp -= 1
		grid.append(tmp)
	lowestprev = 0
	for i in range(k, n + 1):
		lowestprev = min(lowestprev, grid[i - k])
		if grid[i] - lowestprev > 0:
			flag = 1
			break
	if flag:
		l = try1
	else:
		r = try1
print(l)
