def gcd(a, b):
	while b > 0:
		(a, b) = (b, a % b)
	return a
(n, m, q) = map(int, input().split())
(sx, sy, ex, ey) = (0, 0, 0, 0)
t = gcd(n, m)
ri = n // t
ro = m // t
(s, e) = (0, 0)
for i in range(q):
	(sx, sy, ex, ey) = map(int, input().split())
	if sx == 1:
		s = (sy - 1) // ri
	else:
		s = (sy - 1) // ro
	if ex == 1:
		e = (ey - 1) // ri
	else:
		e = (ey - 1) // ro
	if s == e:
		print('YES')
	else:
		print('NO')
