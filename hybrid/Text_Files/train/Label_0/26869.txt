#include <bits/stdc++.h>
using namespace std;
int a[100200], b[100200], c[100200];
int main() {
  int n, m, C;
  cin >> n >> m >> C;
  for (register int i = (0); i < (int)(n); i++) {
    cin >> a[i];
  }
  for (register int i = (0); i < (int)(m); i++) {
    cin >> b[i];
    if (i)
      c[i] = c[i - 1] + b[i];
    else
      c[i] = b[i];
  }
  int kh = 0, gv = 0;
  for (register int i = (0); i < (int)(n); i++) {
    a[i] += c[kh];
    if (m > n - i) gv++;
    if (gv) a[i] -= c[gv - 1];
    if (kh != m - 1) kh++;
  }
  for (register int i = (0); i < (int)(n); i++) cout << a[i] % C << " ";
  cout << endl;
  cin >> n;
}
