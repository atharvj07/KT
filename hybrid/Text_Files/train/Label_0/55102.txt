import java.util.*;
import java.io.*;
import java.math.*;

   public class x887F
   {
      public static void main(String hi[]) throws Exception
      {
         BufferedReader infile = new BufferedReader(new InputStreamReader(System.in));
         StringTokenizer st = new StringTokenizer(infile.readLine());
         int N = Integer.parseInt(st.nextToken());
         int K = Integer.parseInt(st.nextToken());
         int[] arr = new int[N];
         st = new StringTokenizer(infile.readLine());
         for(int i=0; i < N; i++)
            arr[i] = Integer.parseInt(st.nextToken());
         int[] right = new int[N];
         Arrays.fill(right, N);
         Stack<Integer> stack = new Stack<Integer>();
         for(int i=0; i < N; i++)
         {
            while(stack.size() > 0 && arr[stack.peek()] > arr[i])
               right[stack.pop()] = i;
            stack.push(i);
         }
         boolean good = true;
         for(int i=0; i < N; i++)
            if(right[i]-i > K)
               good = false;
         if(good)
            System.out.println("YES");
         else
         {
            Thing[] loc = new Thing[N];
            for(int i=0; i < N; i++)
               loc[i] = new Thing(i, arr[i]);
            Arrays.sort(loc);
            int closest = N;
            int last = N;
            for(Thing t: loc)
            {
               int dex = t.dex;
               if(closest-K > dex)
               {
                  System.out.println("NO");
                  return;
               }
               if(last-dex > K)
               {
                  boolean works = true;
                  for(int i=0; i < dex; i++)
                     if(right[i]-i > K)
                        works = false;
                  System.out.println(works ? "YES" : "NO");
                  return;
               }
               closest = Math.min(closest, last-K);
               last = Math.min(last, dex);
            }
            System.out.println("YES");
         }
      }
   }
   class Thing implements Comparable<Thing>
   {
      public int dex;
      public int val;
      
      public Thing(int a, int b)
      {
         dex = a;
         val = b;
      }
      public int compareTo(Thing oth)
      {
         return val-oth.val;
      }
   }
