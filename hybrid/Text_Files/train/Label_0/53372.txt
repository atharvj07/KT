import java.util.*;
import java.io.*;

public class File {
	public static class FastScanner {
		BufferedReader br;
		StringTokenizer st;
		
		public FastScanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}
		
		String next() {
			while (st == null || !st.hasMoreElements()) {
				try {
					st = new StringTokenizer(br.readLine());
				}
				catch (IOException e) {
					e.printStackTrace();
				}
			}
			
			return st.nextToken();
		}
		
		int nextInt() {
			return Integer.parseInt(next());
		}
		
		long nextLong() {
			return Long.parseLong(next());
		}
	}
	
	public static void main(String[] args) {
		FastScanner sc = new FastScanner();
		PrintWriter out = new PrintWriter(new BufferedOutputStream(System.out));
		
		
		
		char trumpSuit = sc.next().charAt(0);		
		
		Map<Character, Integer> ranks = new HashMap<>();
		ranks.put('6', 1);
		ranks.put('7', 2);
		ranks.put('8', 3);
		ranks.put('9', 4);
		ranks.put('T', 5);
		ranks.put('J', 6);
		ranks.put('Q', 7);
		ranks.put('K', 8);
		ranks.put('A', 9);
		
		String card1 = sc.next();
		String card2 = sc.next();
		
		
		int val1 = ranks.get(card1.charAt(0));
		int val2 = ranks.get(card2.charAt(0));
		
		char suit1 = card1.charAt(1);
		char suit2 = card2.charAt(1);
		
		
		boolean oneWins = true;
		
		if (suit1 == suit2) {
			oneWins = val1 > val2;
		}
		else if (suit1 == trumpSuit) {
			oneWins = true;
		}
		else {
			oneWins = false;
		}
		
		
		if (oneWins) {
			out.println("YES");
		}
		else {
			out.println("NO");
		}
		
		
		out.close();
	}
}
