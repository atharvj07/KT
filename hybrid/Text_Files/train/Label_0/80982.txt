import sys
input = sys.stdin.readline
flush = sys.stdout.flush

def stupid(pos, p1, p2, c1, c2, a):
	if pos == len(a):
		return p1 + p2
	return max(stupid(pos + 1, a[pos], p2, c1 + int(p1 != a[pos]), c2, a), stupid(pos + 1, p1, a[pos], c1, c2 + int(p2 != a[pos]), a))

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	p = -1
	p1 = -1
	pos1 = -1
	c1 = 0
	c2 = 0
	j = 0
	for i in a:
		if i != p:
			p = i
			c1 += 1
		else:
			if i != p1:
				c2 += 1
			else:
				for k in range(pos1 + 1, j):
					if a[k] != p1 and a[k - 1] != a[k + 1]:
						c2 += 1
						break
			p1 = i
			pos1 = j
		j += 1
	print(c1 + c2)
solve()
