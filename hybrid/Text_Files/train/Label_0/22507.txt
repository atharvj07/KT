#include <bits/stdc++.h>
using namespace std;
bool isPrime(int x) {
  if (x != 2 && x % 2 == 0) return false;
  for (int i = 3; i < x; i += 2) {
    if (x % i == 0) return false;
  }
  return true;
}
vector<int> getAllPrimes(int x) {
  vector<int> v;
  v.push_back(2);
  ;
  for (int i = 3; i <= x; i += 2) {
    if (isPrime(i)) v.push_back(i);
    ;
  }
  return v;
}
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v = getAllPrimes(n);
  int c = 0;
  for (int p : v) {
    for (int i = v.size() - 1; i > 0; i--) {
      if (v[i] + v[i - 1] == p - 1) {
        c++;
        break;
      }
    }
  }
  if (c >= k)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
