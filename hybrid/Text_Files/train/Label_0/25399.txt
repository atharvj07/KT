#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int h = s.size() - 1;
  for (int i = 0; i < h; i++) {
    for (int j = i + 1; j < s.size(); j++) {
      if (s[i] == s[j]) {
        s[j] = '?';
      }
    }
  }
  int counter = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] != '?') {
      counter++;
    }
  }
  if (counter % 2 == 0) {
    cout << "CHAT WITH HER!";
  } else {
    cout << "IGNORE HIM!";
  }
}
