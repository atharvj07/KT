import java.util.*;
import java.io.*;
public class MrKitayutaVsBamboos
{
	public static void main(String[] args) throws Exception
	{
		FastScanner in = new FastScanner(System.in);
		
		int n = in.nextInt();
		long m = in.nextInt();
		int k = in.nextInt();
		long p = in.nextInt();
		
		long[] height = new long[n];
		long[] growth = new long[n];
		for(int x = 0; x < n; x++)
		{
			height[x] = in.nextInt();
			growth[x] = in.nextInt();
		}
		
		long low = 1;
		long high = 10000000000000L;
		while(low < high)
		{
			long mid = (low + high) / 2;
			
			TreeMap<Integer, Integer> tm = new TreeMap<Integer, Integer>();
			for(int y = 0; y < m; y++)
			{
				tm.put(y, k);
			}
			
			boolean valid = true;
			outer:
			for(int z = 0; z < height.length; z++)
			{
				long total = height[z] + m * growth[z];
				long need = (total - mid + p - 1) / p;
				if(need > 1000000)
				{
					valid = false;
					break;
				}
				
				long slack = mid - (total - p * need);
				
				for(int a = 0; a < need; a++)
				{
					long time = (p - slack + a * p - height[z] + growth[z] - 1) / growth[z];
					
					if(time > m)
					{
						valid = false;
						break outer;
					}
					
					if(tm.ceilingKey((int)time) == null)
					{
						valid = false;
						break outer;
					}
					else
					{
						int key = tm.ceilingKey((int)time);
						tm.put(key, tm.get(key) - 1);
						if(tm.get(key) == 0)
						{
							tm.remove(key);
						}
					}
				}
			}
			
			if(!valid)
			{
				low = mid + 1;
			}
			else
			{
				high = mid;
			}
		}
		
		System.out.println(low);
	}

	static class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;
		
		public FastScanner(InputStream input)
		{
			br = new BufferedReader(new InputStreamReader(input));
			st = new StringTokenizer("");
		}
		
		public String next() throws IOException
		{
			if(st.hasMoreTokens())
			{
				return st.nextToken();
			}
			else
			{
				st = new StringTokenizer(br.readLine());
				return next();
			}
		}
		
		public int nextInt() throws IOException
		{
			return Integer.parseInt(next());
		}
	}
}
