def main():
	n = int(input())
	a = map(int, input().split())
	cnt = [{} for _ in range(2)]
	xor = 0
	result = 0
	cnt[1][0] = 1
	for (i, x) in enumerate(a):
		xor ^= x
		index = i & 1
		result += int(cnt[index].get(xor) or 0)
		cnt[index][xor] = int(cnt[index].get(xor) or 0) + 1
	print(result)
main()
