#include <iostream>
#include <algorithm>
#include <vector>
#include <string>
#define int long long
using namespace std;
int a[2345];
signed main() {
	int n, z, w; cin >> n >> z >> w;
	for (int i = 1; i <= n; i++)cin >> a[i];
	if (n == 1)cout << abs(a[n] - w) << endl;
	else cout << max(abs(a[n] - a[n - 1]), abs(a[n] - w)) << endl;
}
