import java.util.*;
public class ProE {
	static int n,k,ans,ct,kk,a,res;
	static String ss;
	static int[] mm=new int[250005];
	static int[] pp=new int[250005];
	static int[] qq=new int[250005];
	static char[][] map=new char[505][];
	static int[][] vis=new int[505][505];
	static int[] vx={0,0,1,-1},vy={1,-1,0,0};
	static void dfs(int x,int y)
	{
		vis[x][y]=kk;ct++;
		for(int i=0;i<4;i++)
		{
			int xx=x+vx[i],yy=y+vy[i];
			if(0<xx&&xx<=n&&0<yy&&yy<=n&&vis[xx][yy]==0&&map[xx][yy]!='X')
				dfs(xx,yy);
		}
	}

	public static void main(String[] args) {
		Scanner in=new Scanner(System.in);
		n=in.nextInt();k=in.nextInt();
		for(int i=1;i<=n;i++)
		{
			ss=" "+in.next();
			map[i]=ss.toCharArray();
		}
		for(int i=1;i<=n;i++)
		for(int j=1;j<=n;j++)
		{
			if(vis[i][j]==0&&map[i][j]!='X')
			{
				ct=0;kk++;
				dfs(i,j);mm[kk]=ct;
			}
		}
		ans=0;
		for(int i=1;i<=n-k+1;i++)
		{
			Arrays.fill(pp,0);
			for(int j=0;j<k;j++)
			for(int p=1;p<=k;p++)
			{
				if(map[i+j][p]=='X') continue;
				pp[vis[i+j][p]]++;
			}
			for(int j=1;j<=n-k+1;j++)
			{
				res=0;
				for(int p=0;p<k;p++)
				{
					a=vis[i-1][j+p];
					if(qq[a]==0) { qq[a]=1;res+=mm[a];res-=pp[a];}
					a=vis[i+k][j+p];
					if(qq[a]==0) { qq[a]=1;res+=mm[a];res-=pp[a];}
					a=vis[i+p][j-1];
					if(qq[a]==0) { qq[a]=1;res+=mm[a];res-=pp[a];}
					a=vis[i+p][j+k];
					if(qq[a]==0) { qq[a]=1;res+=mm[a];res-=pp[a];}
				}
				for(int p=0;p<k;p++)
				{
					qq[vis[i-1][j+p]]=0;
					qq[vis[i+k][j+p]]=0;
					qq[vis[i+p][j-1]]=0;
					qq[vis[i+p][j+k]]=0;	
				}
				res+=k*k;
				ans=Math.max(ans,res);
				if(j==n-k+1) continue;
				for(int p=0;p<k;p++)
				{
					if(map[i+p][j]!='X')
						pp[vis[i+p][j]]--;
					if(map[i+p][j+k]!='X')
						pp[vis[i+p][j+k]]++;
				}
			}
		}
		System.out.println(ans);
	}
}
