import math, cmath
parse_int = lambda : list(map(int, input().split()))
n = int(input())
if n < 3:
	print(-1)
	exit()
if n % 4 == 0:
	n //= 4
	print(n * 3, n * 5)
	exit()
a = n
mno = 1
for i in range(2, min(n, round(math.sqrt(n)) + 3)):
	if n % i == 0:
		(a, mno) = (i, n // i)
		if a == 2:
			(a, mno) = (mno, a)
		break
b = a ** 2 // 2
c = b + 1
print(b * mno, c * mno)
