vector<char*> tokenizer(const char* args) {
  char *token = new char[111]; strcpy(token, args); token = strtok(token, ", ");
  vector<char*> v({token});
  while(token = strtok(NULL,", ")) v.push_back(token);
  return reverse(v.begin(), v.end()), v;
}

int main(){
	IOS; 
	int t;
	cin>>t;
	while(t--){
		string s;
		cin>>s;
		int i,siz=0,j,temp,flag,last=-1;
		vi cnt(26);
		vector<vi> pre(sz(s),vi(26));
		queue<int> q[26];
		REP(i,0,sz(s)-1){
			cnt[s[i]-'a']++;
			q[s[i]-'a'].push(i);

			if(i) REP(j,0,25) pre[i][j]=pre[i-1][j];
			pre[i][s[i]-'a']++;
		}
		REP(i,0,25) {
			if(!cnt[i]) cnt[i]=-1;
			else siz++;
		}
		string ans;
		while(siz){
			RAP(i,25,0){
				if(cnt[i]==-1) continue;
				while(q[i].front()<last) q[i].pop();
				temp=q[i].front();
				flag=1;
				REP(j,0,25){
					if(cnt[j]==-1 or i==j) continue;
					if(cnt[j]-pre[temp][j]<1){
						flag=0;
						break;
					}
				}
				if(flag){
					ans+=i+'a';
					while(!q[i].empty()) q[i].pop();
					cnt[i]=-1;
					siz--;
					last=temp;
					break;
				}
			}
		}
		cout<<ans<<"\n";
	}
	return 0;
}
