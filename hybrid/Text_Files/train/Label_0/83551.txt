from collections import Counter
(l, r) = map(int, input().split())
flag = 0
for i in range(l, r + 1):
	count = 0
	s = str(i)
	l = [char for char in s]
	vals = list(Counter(l).values())
	for i in vals:
		if i != 1:
			count += 1
			break
	if count == 0:
		flag = 1
		print(*l, sep='')
		break
if flag == 0:
	print(-1)
