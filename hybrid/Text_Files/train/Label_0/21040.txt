#include <bits/stdc++.h>
using namespace std;
const int MAXN = 200;
int t, n, k1, k2, a[MAXN], b[MAXN];
int main() {
  cin >> t;
  while (t--) {
    cin >> n >> k1 >> k2;
    a[0] = b[0] = 0;
    for (int i = 1; i <= k1; i++) {
      cin >> a[i];
      a[0] = max(a[0], a[i]);
    }
    for (int i = 1; i <= k2; i++) {
      cin >> b[i];
      b[0] = max(b[0], b[i]);
    }
    if (a[0] > b[0])
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
  return 0;
}
