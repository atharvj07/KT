void err(istream_iterator<string> it) {}
template <typename T, typename... Args>
void err(istream_iterator<string> it, T a, Args... args) {
  cerr << *it << " = " << a << "\n";
  err(++it, args...);
}
const long long int INF = 1e18;
int val[600005];
vector<int> compress(vector<int> &temp, int n) {
  vector<pair<int, int> > tp(n);
  int i, j, k;
  for (i = 0; i < n; i++) tp[i].first = temp[i], tp[i].second = i;
  sort(tp.begin(), tp.end());
  vector<int> v(n);
  i = 0, k = 1, j = 0;
  while (i < n) {
    val[k] = tp[i].first;
    while (j < n && tp[i].first == tp[j].first) v[tp[j].second] = k, j++;
    i = j, k++;
  }
  return v;
}
int work[4 * 600005], flag[4 * 600005];
void pull(int p) {
  if (flag[2 * p] == flag[2 * p + 1])
    flag[p] = flag[2 * p];
  else
    flag[p] = 2;
  work[p] = work[2 * p] + work[2 * p + 1];
}
void build(int l, int h, int p) {
  if (l == h) {
    work[p] = val[l] - val[l - 1], flag[p] = 1;
    return;
  }
  int m = ((l + h) >> 1);
  build(l, m, 2 * p);
  build(m + 1, h, 2 * p + 1);
  pull(p);
  return;
}
void update(int a, int b, int l, int h, int p, int f) {
  if (b < l || a > h) return;
  if (flag[p] == f) return;
  if (a <= l && b >= h) {
    flag[p] = f;
    if (f)
      work[p] = val[h] - val[l - 1];
    else
      work[p] = 0;
    return;
  }
  int m = ((l + h) >> 1);
  if (flag[p] < 2) {
    flag[2 * p] = flag[p];
    flag[2 * p + 1] = flag[p];
    if (flag[p])
      work[2 * p] = val[m] - val[l - 1], work[2 * p + 1] = val[h] - val[m];
    else
      work[2 * p] = 0, work[2 * p + 1] = 0;
  }
  update(a, b, l, m, 2 * p, f);
  update(a, b, m + 1, h, 2 * p + 1, f);
  pull(p);
  return;
}
