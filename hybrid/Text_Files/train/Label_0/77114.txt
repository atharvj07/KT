from sys import stdin
C = [[0] * 20 for _ in range(20)]
for i in range(20):
	C[i][0] = C[i][i] = 1
	for j in range(1, i):
		C[i][j] = C[i - 1][j] + C[i - 1][j - 1]
pw = [1, 9, 81, 729]

def get(n, lft):
	return sum((C[n][i] * pw[i] for i in range(lft + 1) if 0 <= i <= n))

def calc(x):
	s = str(x)
	res = 0
	cur = 3
	n = len(s)
	for i in range(n):
		if s[i] == '0':
			continue
		res += get(n - i - 1, cur)
		cur -= 1
		if cur == -1:
			break
		res += get(n - i - 1, cur) * (int(s[i]) - 1)
	return res
stdin.readline()
test = stdin.readlines()
output = []
for line in test:
	(L, R) = map(int, line.split())
	ans = calc(R + 1) - calc(L)
	output.append(ans)
print('\n'.join(map(str, output)))
