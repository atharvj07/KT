(pages, current, k) = [int(x) for x in input().split(' ')]
string = ''
if current - k > 1:
	string += '<< '
for extra in range(current - k, current):
	if extra >= 1:
		string += str(extra) + ' '
string += '(' + str(current) + ') '
for extra in range(current + 1, current + (k + 1)):
	if extra <= pages:
		string += str(extra) + ' '
if current + k < pages:
	string += '>>'
print(string)
