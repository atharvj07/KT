#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
typedef pair<int,int> P;

#define int long long

int N;
int A[100005];
int low, high;

int bit[300005];
int dat[300005];

int sum(int i){
	int ret = 0;
	while(i > 0){
		ret += bit[i];
		i -= i & -i ;
	}
	return ret;
}

void add(int i, int x){
	while(i <= N*2+100){
		bit[i] += x;
		i += i & -i;
	}
	return;
}

bool is_bigger(int val){
	memset(bit,0,sizeof(bit));
	memset(dat,0,sizeof(dat));

	dat[0] = N+10;
	for (int i = 0;i < N;i++){
		if(A[i] < val) dat[i+1] = dat[i]-1;
		else dat[i+1] = dat[i]+1;
	}

	ll s = 0;
	for (int i = 0;i <= N;i++){
		s += sum(dat[i]);
		add(dat[i],1);
	}
	
	ll all = (N*(N+1))/2;

	//cout << val << " " << s << endl;

	if(s*2 >= all) return true;
	else return false;

}

signed main() {
	cin.tie(0);
	ios::sync_with_stdio(false);

	cin >> N;

	low  = INT_MAX;
	high = INT_MIN;

	for (int i = 0;i < N;i++){
		cin >> A[i];
		low = min(low, A[i]);
		high = max(high, A[i]);
	}

	high++;

	while(high-low > 1){
		int mid = (low+high)/2;
		if(is_bigger(mid)) low = mid;
		else high = mid;
	}

	cout << low << endl;

	return 0;

}
