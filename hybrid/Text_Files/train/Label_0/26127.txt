#include <bits/stdc++.h>
using namespace std;
int v[200011], f[200011];
int n, m, a[200001];
int gf(int i) {
  if (i == f[i]) return (i);
  f[i] = gf(f[i]);
  return (f[i]);
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  memset(v, 0, sizeof(v));
  for (int i = 1; i <= n; i++) f[i] = i;
  scanf("%d", &m);
  for (int i = 1; i <= m; i++) {
    int type;
    scanf("%d", &type);
    if (type == 1) {
      int x, y;
      scanf("%d%d", &x, &y);
      for (int now = f[x]; now;) {
        if (y < a[now] - v[now]) {
          v[now] += y;
          break;
        } else {
          y -= a[now] - v[now];
          v[now] = a[now];
          if (now != n)
            f[now] = gf(now + 1), now = gf(now);
          else
            break;
        }
      }
    } else {
      int x;
      scanf("%d", &x);
      printf("%d\n", v[x]);
    }
  }
}
