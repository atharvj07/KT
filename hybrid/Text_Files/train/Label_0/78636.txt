import io
import os
from collections import Counter, defaultdict, deque
from bisect import bisect_right
inf = float('inf')

def lengthOfLIS(nums, mn, mx):
	mnVal = mn - -1
	mxVal = mx - len(nums)
	chain = [mnVal]
	for (i, x) in enumerate(nums):
		v = x - i
		if mnVal > v or v > mxVal:
			continue
		newLen = bisect_right(chain, v)
		assert chain[newLen - 1] <= v
		if newLen == len(chain):
			chain.append(v)
		else:
			chain[newLen] = min(chain[newLen], v)
	return len(chain) - 1

def solve(N, K, A, B):
	ans = 0
	sB = [-1] + sorted(B) + [N]
	for (b1, b2) in zip(sB, sB[1:]):
		mn = -inf if b1 == -1 else A[b1]
		mx = inf if b2 == N else A[b2]
		nums = A[b1 + 1:b2]
		if mn >= mx:
			return -1
		if mx - mn - 1 < len(nums):
			return -1
		ans += len(nums) - lengthOfLIS(nums, mn, mx)
	if ans == inf:
		return -1
	return ans
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
(N, K) = [int(x) for x in input().split()]
A = [int(x) for x in input().split()]
B = []
if K != 0:
	B = set((int(x) - 1 for x in input().split()))
print(solve(N, K, A, B))
