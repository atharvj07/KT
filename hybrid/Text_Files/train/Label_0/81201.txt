def process(S, k):
	n = len(S)
	if k > n:
		return 'NO'
	current = 0
	for i in range(n):
		if S[i] == 'N':
			current += 1
			if current > k:
				return 'NO'
		else:
			current = 0
	start_work = True
	for i in range(k):
		if S[i] == 'Y':
			start_work = False
	if start_work and (k == n or (k < n and S[k] != 'N')):
		return 'YES'
	if n == 1:
		if k == 1 and S[0] in ['?N']:
			return 'YES'
		if k == 0 and S[0] in ['?Y']:
			return 'YES'
		return 'NO'
	if k == n:
		if 'Y' in S:
			return 'NO'
		return 'YES'
	Y_count = 0
	p1 = 0
	for i in range(1, k + 1):
		if S[i] == 'Y':
			Y_count += 1
	p2 = i
	for i in range(k + 1, n):
		if Y_count == 0 and S[p1] != 'N' and (p2 == n - 1 or S[p2 + 1] != 'N'):
			return 'YES'
		p1 += 1
		p2 += 1
		if p2 < n and S[p2] == 'Y':
			Y_count += 1
		if p1 < n and S[p1] == 'Y':
			Y_count -= 1
	if Y_count == 0 and p1 < n and (S[p1] != 'N'):
		return 'YES'
	return 'NO'
(n, k) = [int(x) for x in input().split()]
S = input()
print(process(S, k))
