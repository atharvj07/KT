b = input()
j = len(b)
c = list(b)
j -= 1
d = 1
while j >= 1:
	d = d * 9
	j -= 1
e = int(b[0]) - 1
if e != 0:
	d = d * e
f = 1
j = len(b) - 1
B = ''
while j >= 1:
	e = int(c[j])
	g = int(c[j - 1])
	h = e * g
	i = (g - 1) * 9
	if i > h:
		l = len(B) - 1
		B = ''
		while l >= 0:
			B = B + '9'
			l -= 1
		B = '9' + B
		g -= 1
		c[j - 1] = str(g)
	else:
		B = c[j] + B
	j -= 1
B = c[0] + B
j = len(B) - 1
while j >= 0:
	f = f * int(B[j])
	j -= 1
f = max(f, d)
print(f)
