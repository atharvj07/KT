t = int(input())
for q in range(t):
	s = set()
	(n, x, m) = input().split()
	(n, x, m) = (int(n), int(x), int(m))
	(l1, r1) = (x, x)
	for w in range(m):
		(l, r) = input().split()
		(l, r) = (int(l), int(r))
		if x >= l and x <= r or (l1 <= l and r1 >= l) or (l1 <= r and r1 >= r):
			if l < l1:
				l1 = l
			if r > r1:
				r1 = r
	print(r1 - l1 + 1)
