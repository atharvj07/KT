def solve(n):
	min_number_change = 1 + len(n) - 2 * longest_subseq_of_m_in_n('1', n)
	length = 0
	k = 0
	while length < 2 * len(n) + 1:
		candidate = str(2 ** k)
		length = len(candidate)
		k += 1
		if len(candidate) + len(n) - 2 * longest_subseq_of_m_in_n(candidate, n) < min_number_change:
			min_number_change = len(candidate) + len(n) - 2 * longest_subseq_of_m_in_n(candidate, n)
	return min_number_change

def longest_subseq_of_m_in_n(m, n):
	position = -1
	embedded_letters = 0
	for letter in m:
		nextpos = n[position + 1:].find(letter)
		if nextpos > -1:
			position += nextpos + 1
			embedded_letters += 1
		else:
			return embedded_letters
	return embedded_letters
t = int(input())
mylist = []
for i in range(t):
	mylist.append(str(input()))
for n in mylist:
	print(solve(n))
