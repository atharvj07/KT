#include <bits/stdc++.h>
using namespace std;
long long n, m, k, j, o, p, g, l, r, ans = 0, dp[2][200002];
string s, d;
int main() {
  cin >> r >> g;
  if (r > g) swap(r, g);
  dp[0][r] = 1;
  long long k = 0;
  for (int i = 0; i < 1000; i++) {
    k = -1;
    for (int j = 0; j <= r; j++)
      if (dp[0][j]) {
        if (j >= i + 1)
          dp[1][j - i - 1] += dp[0][j], dp[1][j - i - 1] %= (long long)1e9 + 7,
              k = 1;
        if (g - (i * (i + 1) / 2 - r + j) >= i + 1)
          dp[1][j] += dp[0][j], k = 1, dp[1][j] %= (long long)1e9 + 7;
      }
    if (k == -1) {
      p = i;
      break;
    }
    for (int j = 0; j <= r; j++) dp[0][j] = dp[1][j], dp[1][j] = 0;
  }
  for (int i = 0; i <= r; i++) ans += dp[0][i], ans %= (long long)1e9 + 7;
  cout << ans;
  return 0;
}
