#include <bits/stdc++.h>
using namespace std;
long long n, m, p, a[200020], b[200020], balance, bal0;
vector<long long> ans;
map<int, int> mb, cur;
int main() {
  cin >> n >> m >> p;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  mb.clear();
  bal0 = 0;
  for (int i = 1; i <= m; i++) {
    bal0 -= mb[b[i]];
    mb[b[i]]++;
    bal0 += mb[b[i]];
  }
  for (int i = 1; i <= p; i++) {
    cur.clear();
    balance = bal0;
    if (i + (m - 1) * p > n) continue;
    for (int j = i; j <= i + (m - 1) * p; j = j + p) {
      balance -= abs(mb[a[j]] - cur[a[j]]);
      cur[a[j]]++;
      balance += abs(mb[a[j]] - cur[a[j]]);
    }
    if (balance == 0) ans.push_back(i);
    int k = i;
    for (int j = i + m * p; j <= n; j = j + p) {
      balance -= abs(mb[a[k]] - cur[a[k]]);
      cur[a[k]]--;
      balance += abs(mb[a[k]] - cur[a[k]]);
      k = k + p;
      balance -= abs(mb[a[j]] - cur[a[j]]);
      cur[a[j]]++;
      balance += abs(mb[a[j]] - cur[a[j]]);
      if (balance == 0) ans.push_back(k);
    }
  }
  sort(ans.begin(), ans.end());
  cout << ans.size() << endl;
  for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
  return 0;
}
