#include <bits/stdc++.h>
using namespace std;
const int N = 10000010;
const int mod = 1e9 + 7;
int prim[N], pn;
bool is_prim[N];
int first[N];
void make_prime() {
  memset(is_prim, 1, sizeof(is_prim));
  pn = 0;
  is_prim[0] = is_prim[1] = 0;
  for (int i = 2; i < N; i++) {
    if (is_prim[i]) prim[pn++] = i, first[i] = i;
    for (int j = 0; j < pn && prim[j] * i < N; j++) {
      is_prim[i * prim[j]] = 0;
      first[i * prim[j]] = prim[j];
      if (i % prim[j] == 0) break;
    }
  }
}
int n;
int main() {
  make_prime();
  scanf("%d", &n);
  for (int i = 0; i < n; i++) printf("%d ", prim[i]);
  puts("");
  return 0;
}
