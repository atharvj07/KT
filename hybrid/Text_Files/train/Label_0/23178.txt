#include <bits/stdc++.h>
using namespace std;
const int N = 150100;
const int P = 998244353;
inline int Add(int a, int b) {
  a += b;
  if (a >= P) a -= P;
  return a;
}
inline int Dec(int a, int b) {
  a -= b;
  if (a < 0) a += P;
  return a;
}
inline int mul(int a, int b) { return (long long)a * b % P; }
inline long long qpow(long long a, long long b) {
  long long ans = 1;
  for (; b; b >>= 1, a = a * a % P)
    if (b & 1) ans = ans * a % P;
  return ans;
}
int n, q, u, v, opt, L[N], R[N], dfn, sz[N], inv_n, fa[N], top[N], son[N];
vector<int> G[N];
void dfs(int u, int pa) {
  sz[u] = 1;
  fa[u] = pa;
  for (int v : G[u])
    if (v != pa) {
      dfs(v, u), sz[u] += sz[v];
      if (!son[u] || sz[son[u]] < sz[v]) son[u] = v;
    }
}
void dfs2(int u, int pa, int tp) {
  top[u] = tp;
  L[u] = ++dfn;
  if (son[u]) dfs2(son[u], u, tp);
  for (int v : G[u])
    if (v != pa && v != son[u]) dfs2(v, u, v);
  R[u] = dfn;
}
int S[N], B[N];
void add(int p, int v) {
  for (int i = p; i <= n; i += i & -i) B[i] = Add(B[i], v);
}
int ask(int p) {
  int ans = 0;
  for (int i = p; i; i -= i & -i) ans = Add(ans, B[i]);
  ;
  return ans;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> q;
  inv_n = qpow(n, P - 2);
  for (int i = (1); i <= (n - 1); ++i)
    cin >> u >> v, G[u].push_back(v), G[v].push_back(u);
  dfs(1, 0);
  dfs2(1, 0, 1);
  while (q--) {
    cin >> opt >> v;
    if (opt == 1) {
      int d;
      cin >> d;
      S[v] = Add(S[v], d);
      int tmp = mul(d, sz[v]);
      add(1, tmp);
      add(L[v], P - tmp);
      add(R[v] + 1, tmp);
      if (son[v]) {
        tmp = mul(d, n - sz[son[v]]);
        add(L[son[v]], tmp);
        add(R[son[v]] + 1, P - tmp);
      }
    } else {
      int ans = Add(mul(S[v], n), ask(L[v]));
      for (v = top[v]; v; v = top[fa[v]])
        ans = Add(ans, mul(n - sz[v], S[fa[v]]));
      cout << mul(ans, inv_n) << '\n';
    }
  }
}
