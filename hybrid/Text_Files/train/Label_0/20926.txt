#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, q, counter = 0;
  double x;
  cin >> n >> m >> x;
  int arr[26];
  vector<vector<pair<double, double>>> data(26);
  vector<pair<double, double>> shi;
  memset(arr, 0, sizeof(arr));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      char c;
      cin >> c;
      if (c == 'S')
        shi.push_back(make_pair(i + 0.5, j + 0.5));
      else
        data[c - 'a'].push_back(make_pair(i + 0.5, j + 0.5));
    }
  }
  cin >> q;
  char s[q];
  for (int mm = 0; mm < q; mm++) {
    char a;
    cin >> a;
    if (a >= 97 && data[a - 'a'].size() < 1) {
      cout << -1;
      return 0;
    } else if (a < 97) {
      if (data[a - 'A'].size() < 1) {
        cout << -1;
        return 0;
      }
      arr[a - 'A']++;
    }
  }
  for (int k = 0; k < 26; k++) {
    char c = 'A' + k;
    if (arr[c - 'A'] < 1) continue;
    double distance;
    double min = 100000.0;
    for (int v = 0; v < data[c - 'A'].size(); v++)
      for (int i = 0; i < shi.size(); i++) {
        double p1 = shi[i].first;
        double p2 = shi[i].second;
        distance = sqrt(pow(p1 - data[c - 'A'][v].first, 2) +
                        pow(p2 - data[c - 'A'][v].second, 2));
        if (min > distance) min = distance;
      }
    if (min > 99999.0) {
      cout << -1;
      return 0;
    }
    if (min > x) counter += arr[c - 'A'];
  }
  cout << counter;
  return 0;
}
