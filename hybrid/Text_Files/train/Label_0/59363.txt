public class Main extends Thread {
    public Main() {
        this.setPriority(Thread.MAX_PRIORITY);
    }

    public void run() {
        try {
            BufferedReader in   = new BufferedReader(new InputStreamReader(System.in));
            int tests = Integer.parseInt(in.readLine());
            int a, b;
            String []data;
            for (int i = 0; i < tests; ++i){
                data = in.readLine().split("[ ]+");
                a = Integer.parseInt(data[0]);
                b = Integer.parseInt(data[1]);
                if (b == 0){
                    System.out.println("1.000000000");
                    continue;
                }
                if (a == 0){
                    System.out.println("0.5000000000");
                    continue;
                }

                double limit = Math.min(b * 4, a);
                double nom = a;
                nom *= b;
                nom += 0.125 * limit * limit;
                nom += (double) b * ( - limit + a);
                double den = b * 2;
                den *= a;
                double ans = nom / den;
                Formatter ft = new Formatter(Locale.US);
                System.out.println(ft.format("%.12f",ans));
            }




        } catch (Throwable e) {
            System.out.println(e.getMessage() + "\n" + e.getStackTrace());
        }

    }

    public static void main(String[] args) {
        try {
            new Main().start();
        } catch (Throwable e) {
            System.out.println(e.getMessage() + "\n" + e.getStackTrace());
        }
    }
}
