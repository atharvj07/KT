a = int(input())
b = [int(i) for i in input()]
n = len(b)
c = [0] * (sum(b) + 1)
d = [0]
e = 0
for i in range(n):
	e += b[i]
	d.append(e)
for i in range(n):
	for j in range(i, n):
		c[d[j + 1] - d[i]] += 1
if a == 0:
	print(2 * c[0] * sum(c) - c[0] ** 2)
else:
	f = 0
	c1 = len(c)
	for i in range(1, int(a ** 0.5) + 1):
		if a % i == 0 and i < c1 and (a // i < c1):
			f += c[i] * c[a // i]
			if i != a // i:
				f += c[a // i] * c[i]
	print(f)
