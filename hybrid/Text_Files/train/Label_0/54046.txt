import java.io.*;
import java.util.*;
import java.math.*;

import static java.lang.Math.*;

public class Solution implements Runnable {
	
	class State implements Comparable<State> {
		int i, j, time;
		
		public State(int i, int j, int time) {
			this.i = i;
			this.j = j;
			this.time = time;
		}
		
		@Override
		public int compareTo(State arg0) {
			if (this.time < arg0.time) return -1;
			if (this.time > arg0.time) return 1;
			if (this.i < arg0.i) return -1;
			if (this.i > arg0.i) return 1;
			return this.j - arg0.j;
		}
		
	}
	
	int di [] = {0, 0, 1, -1};
	int dj [] = {1, -1, 0, 0};
	
	int n, m;
	char s [][];
	
	int path [][];
	
	int ci [];
	int cj [];
	
	int findPath (int si, int sj, int fi, int fj, boolean flag) {
		if (flag && s[si][sj] >= 'a' && s[si][sj] <= 'z' && s[fi][fj] >= 'a' && s[fi][fj] <= 'z') {
			return path[s[si][sj] - 'a'][s[fi][fj] - 'a'];
		}
		
		int d [][] = new int [n][m];
		for (int i = 0; i < n; i++) {
			Arrays.fill(d[i], Integer.MAX_VALUE / 2);
		}
		
		d[si][sj] = 0;
		PriorityQueue<State> q = new PriorityQueue<Solution.State>();
		q.add(new State (si, sj, 0));
		
		while (!q.isEmpty()) {
			State cur = q.poll();
			int curi = cur.i;
			int curj = cur.j;
			
			int travelTime = 1;
			if (s[curi][curj] >= '0' && s[curi][curj] <= '9') {
				travelTime = s[curi][curj] - '0'; 
			}
			
			for (int k = 0; k < 4; k++) {
				int ti = di[k] + curi;
				int tj = dj[k] + curj;
				if (ti >= 0 && tj >= 0 && ti < n && tj < m && s[ti][tj] != '#') {
					boolean ok = true;
					if (s[si][sj] >= 'a' && s[si][sj] <= 'z' && s[fi][fj] >= 'a' && s[fi][fj] <= 'z') {
						if (s[ti][tj] >= 'a' && s[ti][tj] <= 'z' && s[ti][tj] != s[si][sj] && s[ti][tj] != s[fi][fj]) {
							ok = false;
						}
					}
					if (ok && d[ti][tj] > d[curi][curj] + travelTime) {
						d[ti][tj] = d[curi][curj] + travelTime;
						q.add(new State(ti, tj, d[ti][tj]));
					}
				}
			}
			
		}
		
		return d [fi][fj];
	}
	
	State recoverPath (int si, int sj, int fi, int fj, int time) {
		int d [][] = new int [n][m];
		int pi [][] = new int [n][m];
		int pj [][] = new int [n][m];
		
		for (int i = 0; i < n; i++) {
			Arrays.fill(d[i], Integer.MAX_VALUE / 2);
		}
		
		d[si][sj] = 0;
		PriorityQueue<State> q = new PriorityQueue<Solution.State>();
		q.add(new State (si, sj, 0));
		
		while (!q.isEmpty()) {
			State cur = q.poll();
			int curi = cur.i;
			int curj = cur.j;
			
			int travelTime = 1;
			if (s[curi][curj] >= '0' && s[curi][curj] <= '9') {
				travelTime = s[curi][curj] - '0'; 
			}
			
			for (int k = 0; k < 4; k++) {
				int ti = di[k] + curi;
				int tj = dj[k] + curj;
				if (ti >= 0 && tj >= 0 && ti < n && tj < m && s[ti][tj] != '#') {
					boolean ok = true;
					if (s[si][sj] >= 'a' && s[si][sj] <= 'z' && s[fi][fj] >= 'a' && s[fi][fj] <= 'z') {
						if (s[ti][tj] >= 'a' && s[ti][tj] <= 'z' && s[ti][tj] != s[si][sj] && s[ti][tj] != s[fi][fj]) {
							ok = false;
						}
					}
					if (ok && d[ti][tj] > d[curi][curj] + travelTime) {
						d[ti][tj] = d[curi][curj] + travelTime;
						pi[ti][tj] = curi;
						pj[ti][tj] = curj;
						q.add(new State(ti, tj, d[ti][tj]));
					}
				}
			}
		}
		
		ArrayList<Integer> posi = new ArrayList<Integer>();
		ArrayList<Integer> posj = new ArrayList<Integer>();
		ArrayList<Integer> travelTime = new ArrayList<Integer>();
		
		int curi = fi;
		int curj = fj;
		
		posi.add(curi);
		posj.add(curj);
		
		while (curi != si || curj != sj) {
			int prei = pi[curi][curj];
			int prej = pj[curi][curj]; 
			int ttime = 1;
			if (s[prei][prej] >= '0' && s[prei][prej] <= '9') {
				ttime = s[prei][prej] - '0';
			}
			posi.add(prei);
			posj.add(prej);
			travelTime.add(ttime);
			curi = prei;
			curj = prej;
		}
		
		for (int i = travelTime.size() - 1; i >= 0; i--) {
			if (travelTime.get(i) > time) {
				return new State (posi.get(i + 1), posj.get(i + 1), -1);
			}
			time -= travelTime.get(i);
		}
		
		return new State (fi, fj, -1);
	}
	
	void solve() throws Exception {
		n = sc.nextInt();
		m = sc.nextInt();
		int k = sc.nextInt();
		
		s = new char [n][m];
		for (int i = 0; i < n; i++) {
			s [i] = in.readLine().toCharArray();
		}
		
		ci = new int [26];
		cj = new int [26];
		
		Arrays.fill(ci, -1);
		Arrays.fill(cj, -1);
		
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				if (s[i][j] >= 'a' && s[i][j] <= 'z') {
					ci[s[i][j] - 'a'] = i;
					cj[s[i][j] - 'a'] = j;
				}
			}
		}
		
		path = new int [26][26];
		for (int i = 0; i < 26; i++) {
			for (int j = 0; j < 26; j++) {
				if (ci[i] != -1 && ci[j] != -1) {
					if (i == j) {
						path[i][j] = 0;
					} else {
						path[i][j] = findPath(ci[i], cj[i], ci[j], cj[j], false);
					}
				}
			}
		}
		
		int si = sc.nextInt() - 1;
		int sj = sc.nextInt() - 1;
		char t [] = sc.nextToken().toCharArray();
		
		int fi = sc.nextInt() - 1;
		int fj = sc.nextInt() - 1;
		
		ArrayList<Integer> pathi = new ArrayList<Integer>();
		ArrayList<Integer> pathj = new ArrayList<Integer>();
		
		pathi.add(si);
		pathj.add(sj);
		
		for (int i = 0; i < t.length; i++) {
			int cur = t[i] - 'a';
			pathi.add(ci[cur]);
			pathj.add(cj[cur]);
		}
		
		pathi.add(fi);
		pathj.add(fj);
		
		for (int i = 0; i < pathi.size() - 1; i++) {
			int elapsedTime = findPath(pathi.get(i), pathj.get(i), pathi.get(i + 1), pathj.get(i + 1), true);
			if (elapsedTime >= k) {
				State answer = recoverPath(pathi.get(i), pathj.get(i), pathi.get(i + 1), pathj.get(i + 1), k);
				out.println((answer.i + 1)+" "+(answer.j + 1));
				return;
			} else {
				k -= elapsedTime;
			}
		}
		
		out.println((fi + 1)+" "+(fj + 1));
		
	}

	BufferedReader in;
	PrintWriter out;
	FastScanner sc;

	static Throwable uncaught;

	@Override
	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new PrintWriter(System.out);
			sc = new FastScanner(in);
			solve();
		} catch (Throwable t) {
			Solution.uncaught = t;
		} finally {
			out.close();
		}
	}

	public static void main(String[] args) throws Throwable {
		Thread t = new Thread(null, new Solution(), "", 128 * 1024 * 1024);
		t.start();
		t.join();
		if (uncaught != null) {
			throw uncaught;
		}
	}

}

class FastScanner {

	BufferedReader reader;
	StringTokenizer strTok;

	public FastScanner(BufferedReader reader) {
		this.reader = reader;
	}

	public String nextToken() throws IOException {
		while (strTok == null || !strTok.hasMoreTokens()) {
			strTok = new StringTokenizer(reader.readLine());
		}
		return strTok.nextToken();
	}

	public int nextInt() throws IOException {
		return Integer.parseInt(nextToken());
	}

	public long nextLong() throws IOException {
		return Long.parseLong(nextToken());
	}

	public double nextDouble() throws IOException {
		return Double.parseDouble(nextToken());
	}

}
