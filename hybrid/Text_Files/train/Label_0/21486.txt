#include <bits/stdc++.h>
using namespace std;

int main(){
  int w,h,n;
  cin >> w >> h >> n;
  int xmin,xmax,ymin,ymax;
  xmin = ymin = 0;
  xmax = w;
  ymax = h;
  for(int i=0; i<n; i++){
    int x, y, a;
    cin >> x >> y >> a;
    if(a==1) xmin = max(xmin,x);
    else if(a==2) xmax = min(xmax, x);
    else if(a==3) ymin = max(ymin,y);
    else ymax = min(ymax,y);
  }
  cout << max(0,xmax-xmin)*max(0,ymax-ymin) << endl;
}
