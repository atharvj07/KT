import bisect
import time
import heapq
import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
T = int(input())
t = 1
while t <= T:
	n = int(input())
	seg = []
	for i in range(n):
		(l, r) = map(int, input().split())
		seg.append([l, r])
	ans = [0] * n
	temp = []
	seg = sorted(seg)
	diff = 0
	for i in range(n):
		ele = seg[i]
		while temp:
			ele2 = heapq.heappop(temp)
			if ele2[0] >= ele[0]:
				heapq.heappush(temp, ele2)
				break
			loc = ele2[1]
			ans[loc] += i - loc - 1
		heapq.heappush(temp, [ele[1], i])
		ans[i] += len(temp)
	for ele in temp:
		ans[ele[1]] += n - ele[1] - 1
	print(n - max(ans))
	t += 1
newtime = time.time()
