#include <bits/stdc++.h>
using namespace std;
const int N = 2222222;
int n, m, p[N], mst[N], rnk[N], ans, used[N], uused[N];
pair<int, pair<int, pair<int, int> > > edges[N];
map<int, int> weight, mst_weight, to_cnt;
vector<pair<pair<int, int>, int> > g[N];
queue<pair<int, int> > q;
int dsu_get(int a) {
  if (p[a] == a) {
    return a;
  }
  return p[a] = dsu_get(p[a]);
}
void dsu_unite(int a, int b) {
  a = dsu_get(a);
  b = dsu_get(b);
  if (a != b) {
    if (rnk[a] < rnk[b]) {
      swap(a, b);
    }
    p[b] = a;
    rnk[a] += rnk[b];
  }
}
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    rnk[i] = 1;
    p[i] = i;
  }
  for (int i = 1; i <= m; i++) {
    int v, u, w;
    cin >> v >> u >> w;
    g[v].push_back({{u, w}, i});
    g[u].push_back({{v, w}, i});
    weight[w]++;
    edges[i] = {w, {u, {v, i}}};
  }
  sort(edges + 1, edges + m + 1);
  int k = 0;
  int oldw = 0;
  for (int i = 1; i <= m; i++) {
    int v = edges[i].second.first, u = edges[i].second.second.first,
        w = edges[i].first, id = edges[i].second.second.second;
    if (oldw != w) {
      int allof = q.size();
      int good = 0;
      while (q.size()) {
        int a = q.front().first, b = q.front().second;
        a = dsu_get(a);
        b = dsu_get(b);
        if (a != b) {
          good++;
          dsu_unite(a, b);
        }
        q.pop();
      }
      ans += allof - good;
    }
    v = dsu_get(v);
    u = dsu_get(u);
    if (v != u) {
      q.push({u, v});
      used[id] = 1;
    }
    oldw = w;
  }
  int allof = q.size();
  int good = 0;
  while (q.size()) {
    int a = q.front().first, b = q.front().second;
    a = dsu_get(a);
    b = dsu_get(b);
    if (a != b) {
      good++;
      dsu_unite(a, b);
    }
    q.pop();
  }
  ans += allof - good;
  cout << ans;
}
