(n, m) = map(int, input().split())
a = []
b = []
for i in range(n):
	a.append(list(map(int, input().split())))
	b.append(list(map(int, ('0 ' * m).split())))
t = 0
x = []
y = []
u = 0
e = 0
for i in range(n - 1):
	for j in range(m - 1):
		if a[i][j] + a[i + 1][j] + a[i][j + 1] + a[i + 1][j + 1] == 4:
			x.append(i)
			y.append(j)
			b[i][j] = 1
			b[i + 1][j] = 1
			b[i][j + 1] = 1
			b[i + 1][j + 1] = 1
			u += 1
for i in range(n):
	for j in range(m):
		if a[i][j] != b[i][j]:
			print('-1')
			exit()
print(u)
for i in range(u):
	print(x[i] + 1, y[i] + 1)
