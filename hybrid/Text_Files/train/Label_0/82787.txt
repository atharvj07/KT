def answer(n):
	a = []
	for i in range(2, int(n ** 0.5) + 1):
		if n % i == 0:
			a.append(i)
			a.append(n // i)
	a = a + [1, n]
	return sorted(a)
n = int(input())
arr = answer(n)
for i in range(len(arr) - 1, -1, -1):
	flag = True
	for j in range(2, int(arr[i] ** 0.5) + 1):
		if arr[i] % j ** 2 == 0:
			flag = False
			break
	if flag:
		print(arr[i])
		break
	else:
		flag = True
