#include <bits/stdc++.h>
using namespace std;
int N, M, K;
int S[1005][1005];
struct QUE {
  struct O {
    int first, second;
  } up, down;
  long long val;
} Q[5000];
int tail, head;
pair<long long, pair<int, int> > ans;
inline void init(int x) {
  head = tail = 0;
  long long mx = 0;
  int rr = 0, ll = 1;
  for (int i = x - K + 1, j = x + K - 1; ll <= K; i++, j--) {
    long long sum = 0;
    Q[tail].up.first = i, Q[tail].up.second = K;
    Q[tail].down.first = j, Q[tail].down.second = K;
    for (int uu = i, dd = j, k = K; k > rr + 1; uu++, dd--, k--)
      sum += S[uu][k] + S[dd][k];
    sum += S[x][++rr];
    Q[tail++].val = sum, mx += sum * ll, ll++;
  }
  ll = K - 1;
  for (int i = x, j = x; ll > 0; i--, j++) {
    long long sum = 0;
    Q[tail].up.first = i, Q[tail].up.second = K + 1;
    Q[tail].down.first = j, Q[tail].down.second = K + 1;
    for (int uu = i, dd = j, k = K + 1; k < rr + 1; uu++, dd--, k++)
      sum += S[uu][k] + S[dd][k];
    sum += S[x][++rr];
    Q[tail++].val = sum, mx += sum * ll, ll--;
  }
  if (mx > ans.first) ans = make_pair(mx, make_pair(x, K));
}
inline void solve(int x, int y) {
  head++;
  int hh = head;
  long long mx = 0;
  for (int j = 1; j < K; hh++, j++) {
    Q[hh].up.first--, Q[hh].up.second++;
    Q[hh].down.first++, Q[hh].down.second++;
    Q[hh].val += S[Q[hh].up.first][Q[hh].up.second] +
                 S[Q[hh].down.first][Q[hh].down.second];
    mx += Q[hh].val * j;
  }
  for (int j = K; j > 1; hh++, j--) {
    if (Q[hh].up.first != Q[hh].down.first) {
      Q[hh].val -= S[Q[hh].up.first][Q[hh].up.second] +
                   S[Q[hh].down.first][Q[hh].down.second];
      Q[hh].up.first++, Q[hh].up.second++;
      Q[hh].down.first--, Q[hh].down.second++;
    }
    mx += Q[hh].val * j;
  }
  long long sum = 0;
  if (K > 1) {
    Q[tail].up.first = x - K + 2, Q[tail].up.second = y + 1;
    Q[tail].down.first = x + K - 2, Q[tail].down.second = y + 1;
  } else {
    Q[tail].up.first = Q[tail].down.first = x;
    Q[tail].up.second = Q[tail].down.second = y;
  }
  for (int uu = x - K + 2, dd = x + K - 2, ll = y + 1; ll < y + K - 1;
       ll++, uu++, dd--)
    sum += S[uu][ll] + S[dd][ll];
  sum += S[x][y + K - 1], mx += sum;
  Q[tail++].val = sum;
  if (mx > ans.first) ans = make_pair(mx, make_pair(x, y));
}
int main() {
  scanf("%d%d%d", &N, &M, &K);
  ans = make_pair(-1, make_pair(0, 0));
  for (int i = 1; i <= N; i++)
    for (int j = 1; j <= M; j++) scanf("%d", &S[i][j]);
  for (int i = K; i <= N - K + 1; i++) {
    init(i);
    for (int j = K + 1; j <= M - K + 1; j++) solve(i, j);
  }
  printf("%d %d\n", ans.second.first, ans.second.second);
  return 0;
}
