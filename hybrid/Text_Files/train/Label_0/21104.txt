#include <string>
#include <iostream>
#include <algorithm>
using namespace std;
int N, T, a[107], c[107], used[4][4]; string s[107], m[4];
int rec(int x, int y, int d, int v) {
	if (d == s[v].size()) return 1;
	int ret = 0;
	for (int i = -1; i <= 1; i++) {
		for (int j = -1; j <= 1; j++) {
			int tx = x + j, ty = y + i;
			if (0 <= tx && tx <= 3 && 0 <= ty && ty <= 3 && !used[ty][tx] && m[ty][tx] == s[v][d]) {
				used[ty][tx] = 1;
				ret += rec(tx, ty, d + 1, v);
				used[ty][tx] = 0;
			}
		}
	}
	return ret;
}
int main() {
	cin >> N;
	for (int i = 0; i < N; i++) cin >> s[i] >> a[i];
	for (int i = 0; i < 4; i++) cin >> m[i];
	cin >> T;
	for (int i = 0; i < N; i++) {
		for (int j = 0; j < 4; j++) {
			for (int k = 0; k < 4; k++) {
				if (m[j][k] == s[i][0]) {
					used[j][k] = 1;
					c[i] += rec(k, j, 1, i);
					used[j][k] = 0;
				}
			}
		}
	}
	int dp[10009] = { 0 };
	for (int i = 0; i < N; i++) {
		int x = c[i];
		for (int j = 1; x > 0; j <<= 1) {
			int mul = min(j, x);
			for (int k = T; k >= s[i].size() * mul; k--) dp[k] = max(dp[k], dp[k - s[i].size() * mul] + a[i] * mul);
			x -= mul;
		}
	}
	cout << dp[T] << endl;
	return 0;
}
