#include<iostream>
#include<vector>
#include<algorithm>
#include<queue>
#include<tuple>

using namespace std;
const long long MOD = (long long)1e9 + 7;

int main(){
    int n;
    cin >> n;
    vector<pair<int, int>> c(2 * n);
    for(int i = 0; i < n; i++){
        int x;
        cin >> x;
        c[i] = {x, 0};
    }
    for(int i = 0; i < n; i++){
        int x;
        cin >> x;
        c[i + n] = {x, 1};
    }
    sort(begin(c), end(c));
    long long ans = 1;
    queue<int> a, b;
    for(int i = 0; i < 2 * n; i++){
        int x, y;
        tie(x, y) = c[i];
        if(y == 0){
            if(b.empty()){
                a.push(x);
            }else{
                ans *= b.size();
                ans %= MOD;
                b.pop();
            }
        }else{
           if(a.empty()){
                b.push(x);
            }else{
                ans *= a.size();
                ans %= MOD;
                a.pop();
            }     
        }
    }
    cout << ans << endl;
    return 0;
}
