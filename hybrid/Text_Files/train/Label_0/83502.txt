n = int(input())
perm = [int(t) for t in input().split(' ')]
idx = {n: i for (i, n) in enumerate(perm)}
ridx = {i: n for (i, n) in enumerate(perm)}
swaps = []
for i in range(n):
	desired = i + 1
	didx = idx[desired]
	if didx == i:
		ridx_ = dict()
	elif abs(i - didx) * 2 >= n:
		swaps.append((i, didx))
		ridx_ = {i: desired, didx: ridx[i]}
	else:
		if didx < n // 2:
			inter = n - 1
		else:
			inter = 0
		if abs(i - inter) * 2 >= n:
			swaps.append((didx, inter))
			swaps.append((i, inter))
		else:
			inter2 = 0 if inter == n - 1 else n - 1
			swaps.append((didx, inter))
			swaps.append((inter, inter2))
			swaps.append((i, inter2))
			swaps.append((inter2, inter))
		ridx_ = {i: desired, inter: ridx[i], didx: ridx[inter]}
	for (k, v) in ridx_.items():
		ridx[k] = v
		idx[v] = k
print(len(swaps))
print('\n'.join(['{} {}'.format(swap[0] + 1, swap[1] + 1) for swap in swaps]))
