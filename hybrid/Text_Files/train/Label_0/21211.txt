#include <bits/stdc++.h>
using namespace std;
typedef complex <double> C;

class Point{
public:
  double x;
  double y;
  Point operator + (Point p)const{
    return (Point){x + p.x, y + p.y};
  }
  bool operator < (Point p)const{
    return x*x + y*y < p.x*p.x + p.y*p.y;
  }
};


bool throughO(Point p, Point v){
  //  if(abs(v.x) < 1e-7 ){
  //    return abs(p.y) < 1e-7;
  //  }

  return abs(p.x*v.y-p.y*v.x) < 1e-8;

  //  double a = v.y / v.x;
  //  return abs(p.y - a*p.x) < 0.0000001;
}


int main(void){
  double D;
  while(cin >> D){
    if(D==0)break;

    Point p, v;
    double ans = 0;

    cin >> p.x >> p.y >> v.x >> v.y;
    double len = sqrt(p.x * p.x + p.y * p.y);


    if(!throughO(p, v) ){
      cout << "impossible" << endl;
      //impossible
      continue;
    }

    double naiseki = -p.x * v.x + -p.y * v.y;
    if(abs(naiseki - len*sqrt(v.x*v.x + v.y*v.y)) > 0.000001){
      ans = 1 + (1 - len);
    }else{
      ans = len;
    }
    if(ans < D + 0.000001)
      printf("%.10f\n", ans);
    else
      cout << "impossible" << endl;
    //impossible
  }
}
