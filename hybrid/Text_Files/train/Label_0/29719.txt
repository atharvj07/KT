#include<bits/stdc++.h>
#define MAX 100001
#define EXP 1000000007
using namespace std;
int dp[27];
char s[MAX],t[MAX];
int main(void)
{
	int n;
	cin>>n;
	cin>>s>>t;
	fill(dp,dp+26,0);
	dp[s[0]-'a']++;
	for(int i=1;i<n-1;i++){
		dp[s[i]-'a']+=dp[t[i]-'a'];
		dp[s[i]-'a']%=EXP;
	}
	cout<<dp[t[n-1]-'a']<<endl;
	return 0;
}
