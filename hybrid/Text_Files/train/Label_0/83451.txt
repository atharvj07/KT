import sys
import math
input = sys.stdin.readline
mod = 10 ** 9 + 7

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())
for _ in range(inp()):
	(a, b) = invr()
	l = insr()
	a1 = 0
	b1 = 0
	c = 0
	n = a + b
	for i in range(n):
		if l[i] == '0':
			a1 += 1
		if l[i] == '1':
			b1 += 1
		if l[i] == '?':
			c += 1
	if a - a1 + b - b1 != c:
		print(-1)
		continue
	a = a - a1
	b = b - b1
	if a < 0 or b < 0:
		print(-1)
		continue
	s = 0
	for i in range(n):
		if l[i] != '?':
			if l[i] == '0':
				if l[n - i - 1] == '1':
					s = 1
				elif l[n - i - 1] == '?':
					l[n - i - 1] = '0'
					a = a - 1
					c = c - 1
			elif l[n - i - 1] == '0':
				s = 1
			elif l[n - i - 1] == '?':
				l[n - i - 1] = '1'
				b = b - 1
				c = c - 1
	if a < 0 or b < 0:
		print(-1)
		continue
	if s == 1:
		print(-1)
		continue
	if c == 0:
		print(''.join((str(i) for i in l)))
		continue
	if n % 2 == 1:
		if l[n // 2] == '?':
			if a % 2 == 1:
				l[n // 2] = '0'
				a -= 1
			else:
				l[n // 2] = '1'
				b -= 1
	if a % 2 == 1 or b % 2 == 1:
		print(-1)
		continue
	for i in range(n):
		if l[i] == '?':
			if a > 0:
				l[i] = '0'
				l[n - i - 1] = '0'
				a = a - 2
				continue
			if b > 0:
				l[i] = '1'
				l[n - i - 1] = '1'
				b = b - 2
	for i in range(n):
		if l[i] != l[n - i - 1]:
			s = 1
			break
	if s == 1:
		print(-1)
		continue
	print(''.join((str(i) for i in l)))
