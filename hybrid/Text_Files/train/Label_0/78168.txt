n = int(input())
a = list(map(int, input().split()))
a.sort()
b = list()
cur = a[0]
k = 1
for i in range(1, n):
	if a[i - 1] == a[i]:
		k += 1
	else:
		b.append((cur, k))
		cur = a[i]
		k = 1
b.append((cur, k))
n1 = len(b)
dp = [0] * n1
dp[0] = b[0][0] * b[0][1]
maxx = b[0][0] * b[0][1]
for i in range(1, n1):
	if b[i - 1][0] + 1 == b[i][0]:
		if i != 1:
			dp[i] = max(b[i][0] * b[i][1] + dp[i - 2], dp[i - 1])
		else:
			dp[i] = max(b[i][0] * b[i][1], dp[i - 1])
	else:
		dp[i] = dp[i - 1] + b[i][0] * b[i][1]
print(max(dp))
