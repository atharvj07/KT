n = int(input())
stro = input()
ans = 0
x = 0
while stro[x] == '<' and x <= n - 1:
	x += 1
	ans += 1
	if x == n - 1 and stro[x] == '<':
		ans += 1
		break
	if x >= n:
		break
x = 1
while stro[-x] == '>' and x <= n:
	x += 1
	ans += 1
	if x == n and stro[-x] == '>':
		ans += 1
		break
	if x > n:
		break
print(ans)
