import java.util.*;
import java.io.*;
import java.math.BigInteger;

public class Solution {
	
	public static void main(String[] args) {
		new Solution().run();
	}
	
	public void run()
	{
		Scanner file = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
		int N = file.nextInt();
		int alice = file.nextInt();
		ArrayList<Integer> ans = new ArrayList<>();
		ans.add(1);
		int sum = alice;
		int totalSum = alice;
		for(int i = 1;i<N;i++)
		{
			int next = file.nextInt();
			totalSum += next;
			if(next*2<=alice){
				ans.add(i+1);
				sum+=next;
			}
		}
		if(sum*2>totalSum)
		{
			System.out.println(ans.size());
			for(int i:ans)
				System.out.print(i+" ");
		}else{
			System.out.println(0);
		}
	}
}
