#include <bits/stdc++.h>
using namespace std;
const int maxn = 11005;
struct edge {
  int to, nxt, l, r;
} e[maxn];
struct node {
  long long val;
  int id;
  node(long long v_ = 0, int i_ = 0) : val(v_), id(i_) {}
  inline bool operator<(const node &rhs) const { return val > rhs.val; }
};
int n, m, k;
int s1, s2, t, mem[maxn], vis[maxn];
long long dis1[maxn], dis2[maxn];
int chs[maxn];
int lnk[maxn], ptr;
inline void add(int bgn, int end, int l, int r) {
  e[++ptr] = (edge){end, lnk[bgn], l, r};
  lnk[bgn] = ptr;
}
inline int rd() {
  register int x = 0, c = getchar();
  while (!isdigit(c)) c = getchar();
  while (isdigit(c)) x = x * 10 + (c ^ 48), c = getchar();
  return x;
}
inline void spfa1() {
  priority_queue<node> q;
  memset(dis1, 0x3f, sizeof dis1);
  memset(dis2, 0x3f, sizeof dis2);
  dis1[s1] = 0;
  dis2[s2] = 0;
  memset(chs, -1, sizeof chs);
  q.push(node(0, s1));
  q.push(node(0, s2));
  while (!q.empty()) {
    bool tag;
    node u = q.top();
    q.pop();
    if (u.val > dis1[u.id] && u.val > dis2[u.id]) continue;
    if (dis1[u.id] < dis2[u.id])
      tag = true;
    else
      tag = false;
    for (int p = lnk[u.id]; p; p = e[p].nxt) {
      int y = e[p].to, del;
      if (tag)
        chs[p] = del = e[p].l;
      else
        chs[p] = del = e[p].r;
      if (dis1[y] > dis1[u.id] + del) {
        dis1[y] = dis1[u.id] + del;
        q.push(node(dis1[y], y));
      }
      if (dis2[y] > dis2[u.id] + del) {
        dis2[y] = dis2[u.id] + del;
        q.push(node(dis2[y], y));
      }
    }
  }
}

int main() {
  n = rd();
  m = rd();
  k = rd();
  s1 = rd();
  s2 = rd();
  t = rd();
  int u, v, w, l, r;
  for (int i = 1; i <= m; ++i) {
    u = rd();
    v = rd();
    w = rd();
    add(u, v, w, w);
  }
  for (int i = 1; i <= k; ++i) {
    u = rd();
    v = rd();
    l = rd();
    r = rd();
    add(u, v, l, r);
    mem[i] = ptr;
  }
  spfa1();
  if (dis1[t] < dis2[t]) {
    puts("WIN");
    for (int i = 1; i <= k; ++i) {
      if (chs[mem[i]] == -1) chs[mem[i]] = e[mem[i]].l;
      printf("%d ", chs[mem[i]]);
    }
    return 0;
  }
  spfa2();
  if (dis1[t] == dis2[t]) {
    puts("DRAW");
    for (int i = 1; i <= k; ++i) {
      if (chs[mem[i]] == -1) chs[mem[i]] = e[mem[i]].l;
      printf("%d ", chs[mem[i]]);
