import java.util.Scanner;


public class D78 {

    static double R;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        R = sc.nextInt();
        long ans = 0;
        for (int i = 1; 1+1.5*i <= R; i++) {
            double t = 0;
            if (i % 2==0)
                t = 0.5;
            ans += Math.min(BinarySearch(0.5+1.5*i, t), BinarySearch(1+1.5*i, t-0.5));
        }
        ans *= 4;
        int t = 1;
        while ((-0.5+t)*(-0.5+t)*3 <= R*R-0.25)
            t++;
        t--;
        ans += 2*t-1;
        t = 1;
        while (3*t+1 <= R)
            t++;
        t--;
        ans += 2*t;
        System.out.println(ans);
    }
    private static int BinarySearch(double x, double delta) {
        if ((delta+1)*(delta+1)*3 > R*R-x*x)
            return 0;
        int left = 0, right = (int) 1e9;
        while (true) {
            int midle = (left+right) >> 1;
            if ((delta+midle)*(delta+midle)*3<=R*R-x*x && (delta+midle+1)*(delta+midle+1)*3 > R*R-x*x) {
                left = midle;
                break;
            }
            if ((delta+midle)*(delta+midle)*3 > R*R-x*x)
                right = midle-1;
            else
                left = midle+1;
        }
        return left;
    }
}
