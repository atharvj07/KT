n = int(input())
s = input()
(a, b) = (0, 0)
E = [[0, 0]]
times = [1]
for i in range(n):
	if s[i] == 'U':
		b += 1
	elif s[i] == 'R':
		a += 1
	elif s[i] == 'D':
		b -= 1
	else:
		a -= 1
	if [a, b] not in E:
		E.append([a, b])
		times.append(1)
	else:
		times[E.index([a, b])] += 1
c = 0
for i in times:
	c += i * (i - 1) // 2
print(c)
