int eulers_phi(int N){
  int ret = N;
  for(int i = 2; i * i <= N; ++i){
    if(!(N % i)){
      ret -= ret / i;
      while(!(N % i)) N /= i;
    }
  }
  if(N > 1) ret -= ret / N;
  return ret;
}
