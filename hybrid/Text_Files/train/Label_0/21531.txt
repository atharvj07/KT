#include <bits/stdc++.h>
using namespace std;
const int N = 500;
int n;
char s[N + 1];
void inv(int i) { s[i] = s[i] == '(' ? ')' : '('; }
int get() {
  int val = 0;
  int mn = ~(1 << 31), amt;
  for (int i = 0; i < n; ++i) {
    val += (-2 * (s[i] == ')')) + 1;
    if (val < mn) {
      mn = val;
      amt = 1;
    } else if (val == mn)
      ++amt;
  }
  if (val) return 0;
  return amt;
}
int main() {
  scanf("%d", &n);
  scanf("%s", s);
  int ans = 0;
  pair<int, int> sw(1, 1);
  for (int i = 0; i < n; ++i) {
    for (int j = i; j < n; ++j) {
      inv(i);
      inv(j);
      int t = get();
      if (ans < t) {
        ans = t;
        sw.first = i + 1;
        sw.second = j + 1;
      }
      inv(i);
      inv(j);
    }
  }
  printf("%d\n%d %d\n", ans, sw.first, sw.second);
  return 0;
}
