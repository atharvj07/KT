#include<iostream>
#include<cstdio>
#include<cstring>
#define FIRST 735224;
using namespace std;
bool URUDOSHI[10000001];
void YEAR(){
  for(int i=0;i < 10000001;i++){
    if(i % 400 == 0 || (i % 100 != 0 && i % 4 == 0)){
      URUDOSHI[i] = true;
    } else {
      URUDOSHI[i] = false;
    }
  }
}
int main(){
  YEAR();
  long long a,b,c,d,e,cnt,day;
  char str[111];
  while(1){
    cnt = 0;
    cin >> str;
    if(strcmp(str,"#")==0) break;
    for(int i=0,l=strlen(str);i<l;i++){
      if(str[i] == '.')cnt++;
    }
    if(cnt == 2){
      sscanf(str,"%lld.%lld.%lld",&a,&b,&c);
      day = -FIRST;
      for(int i=0;i<a;i++)
	day += (URUDOSHI[i]==true ? 366 : 365);
      day %= 1872000;
      for(int i=1;i<b;i++){
	if(i==4||i==6||i==9||i==11)day+=30;
	else if(i==2){
	  if(URUDOSHI[a]==true) day+=29;
	  else day+=28;
	}else day+=31;
      }
      day+=c;
      day %= 1872000;
      cout << day/144000 << "." << day%144000/7200 << "." <<
	day%7200/360 << "." << day%360/20 << "." << day%20 << endl; 
    }else{
      sscanf(str,"%lld.%lld.%lld.%lld.%lld",&a,&b,&c,&d,&e);
      day = a * 144000 + b * 7200 + c * 360 + d * 20 + e;
      day += FIRST;
      long long y=0,m=1,dday;
      for(dday=0;dday<day;y++){
	dday += (URUDOSHI[y]==true ? 366 : 365);
      }
      y--;
      day = (URUDOSHI[y]==true ? 366:365) + (day - dday);
      for(dday=0;dday<day;m++){
	if(m==4||m==6||m==9||m==11)dday+=30;
	else if(m==2){
	  if(URUDOSHI[y]==true) dday+=29;
	  else dday+=28;
	}else dday+=31;
      }
      m--;
      day = ((m==4||m==6||m==9||m==11)?30:
	     (m==2?(URUDOSHI[y]==true?29:28):31))+(day-dday);
      cout << y << "." << m << "." << day << endl;
    }
  }
}
