import sys
input = sys.stdin.readline
I = lambda : list(map(int, input().split()))
(n,) = I()
l = []
an = [0] * n
st = [-1] * (10 ** 6 + 1)
en = [-1] * (10 ** 6 + 1)
for i in range(n):
	(a, b) = I()
	st[a] = b
	en[b] = a
	if a == 0:
		an[1] = b
	if b == 0:
		an[n - 2] = a
	l.append([a, b])
for i in range(1, n - 2, 2):
	an[i + 2] = st[an[i]]
for i in range(n - 2, 1, -2):
	an[i - 2] = en[an[i]]
if n % 2:
	x = 0
	for i in range(1, len(st)):
		if st[i] != -1 and en[i] == -1:
			x = i
			break
	an[0] = x
	for i in range(0, n - 2, 2):
		an[i + 2] = st[an[i]]
print(*an)
