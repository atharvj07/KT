import math

def posibility(x, n, g, b):
	half = math.ceil(n / 2)
	combine = g + b
	cycles = x // combine
	extra = x % combine
	extraGood = 0
	extraBad = 0
	if extra > 0:
		extraGood = min(g, extra)
		extra -= extraGood
	if extra > 0:
		extraBad = min(b, extra)
		extra -= extraBad
	totalGood = cycles * g + extraGood
	totalBad = cycles * b + extraBad
	if totalGood < half or totalGood + totalBad < n:
		return False
	return True
testCases = int(input())
for _ in range(testCases):
	(n, g, b) = [int(x) for x in input().split()]
	start = 0
	end = 10 ** 18
	while end - start > 1:
		mid = (start + end) // 2
		if posibility(mid, n, g, b):
			end = mid
		else:
			start = mid
	print(end)
