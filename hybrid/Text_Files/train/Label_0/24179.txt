#include <bits/stdc++.h>
using namespace std;
int n, m, i, j, k, po, len, now, g[2][5001], f[5001];
char s[5001], mn;
int main() {
  scanf("%s", s);
  n = m = strlen(s);
  f[0] = 1;
  while ((1 << len) <= n) len++;
  len--;
  m -= (1 << len) - 1;
  for (i = 0; i < m; i++) {
    memcpy(g[now], f, sizeof(f));
    for (j = 0; j < len; j++, now ^= 1) {
      memcpy(g[now ^ 1], g[now], sizeof(f));
      for (k = 0; k < (1 << len); k++) g[now ^ 1][k | (1 << j)] |= g[now][k];
    }
    memcpy(f, g[now], sizeof(f));
    mn = 'z';
    for (j = 0; j < (1 << len); j++) {
      po = i + j;
      if (!f[j] || po >= n) continue;
      mn = min(mn, s[po]);
    }
    printf("%c", mn);
    for (j = 0; j < (1 << len); j++) {
      po = i + j;
      if (!f[j] || po >= n || s[po] != mn) f[j] = 0;
    }
  }
}
