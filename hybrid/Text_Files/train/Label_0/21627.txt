#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, s;
ll f(ll b, ll n)
{
	return n < b ? n : f( b , n / b ) + n % b;
}
template <class T>
inline void _read(T &x)
{
	x = 0;
	char t = getchar();
	while (!isdigit(t)) t = getchar();
	while (isdigit(t))
	{
		x = x * 10 + t - '0';
		t = getchar();
	}
}
int main()
{
	_read(n), _read(s);
	if (s > n)
    {
        return puts("-1"), 0;
    }
	if (s == n)
	{
		printf("%lld", n + 1);
		return 0;
	}
	ll bs = sqrt(n) + 1;
	for (ll i = 2; i <= bs; ++i)
	{
		if (f(i, n) == s)
		{
			printf("%lld", i);
			return 0;
		}
	}
	ll ans = 1e18;
	n -= s;
	for (ll i = 1; i * i <= n; ++i)
	{
		if (n % i == 0)
		{
			ll tmp = n / i + 1;
			if (f(tmp, n + s) == s) ans = min(ans, tmp);
		}
	}
	printf("%lld\n" , ans == 1e18 ? -1 : ans);
	return 0 ;
}
