import math
a1 = input()
a1 = a1.split(' ')
a1 = [int(x) for x in a1]
R = a1[0]
C = a1[1]
t = int(input())
lst = []
for i in range(t):
	d = input()
	d = d.split(' ')
	d = [int(x) for x in d]
	lst.append(d)
c = 0
for i in lst:
	ox = i[0]
	oy = i[1]
	r = i[2]
	d = math.sqrt(math.pow(ox, 2) + math.pow(oy, 2))
	if d + r <= R and d - r >= R - C and (r <= C / 2):
		c += 1
print(c)
