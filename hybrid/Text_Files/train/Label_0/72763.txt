s = int(input())
A = ['zero', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'nine']
B = ['ten', 'eleven', 'twelve', 'thirteen', 'fourteen', 'fifteen', 'sixteen', 'seventeen', 'eighteen', 'nineteen']
C = ['', '', 'twenty', 'thirty', 'forty', 'fifty', 'sixty', 'seventy', 'eighty', 'ninety']
if s < 10:
	print(A[s])
elif s < 20:
	print(B[s - 10])
elif s % 10 == 0:
	print(C[s // 10])
else:
	print(C[s // 10] + '-' + A[s % 10])
