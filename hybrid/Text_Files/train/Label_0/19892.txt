#include <bits/stdc++.h>
using namespace std;
vector<int> gr[100100];
int mas[100100], ans;
void go(int u, int x) {
  mas[x]++;
  for (int i = 0; i < gr[u].size(); i++) go(gr[u][i], x + 1);
}
int main() {
  int n;
  cin >> n;
  for (int i = 2; i <= n; i++) {
    int x;
    cin >> x;
    gr[x].push_back(i);
  }
  go(1, 1);
  for (int i = 1; i <= n; i++) {
    ans += mas[i] % 2;
  }
  cout << ans << endl;
}
