#include <bits/stdc++.h>
using namespace std;
int a, b, x;
int main() {
  cin >> a >> b >> x;
  int last = 0;
  if (a > b) {
    for (int i = 0; i < x; i++) {
      if (i % 2 == 0) {
        cout << 0;
        a--;
        last = 0;
      } else {
        cout << 1;
        b--;
        last = 1;
      }
    }
    if (last == 0) {
      while (a) {
        cout << 0;
        a--;
      }
      while (b) {
        cout << 1;
        b--;
      }
    } else {
      while (b) {
        cout << 1;
        b--;
      }
      while (a) {
        cout << 0;
        a--;
      }
    }
    return 0;
  } else
    for (int i = 0; i < x - 1; i++) {
      if (i % 2 == 0) {
        cout << 1;
        b--;
        last = 1;
      } else {
        cout << 0;
        a--;
        last = 0;
      }
    }
  if (last == 0) {
    while (b) {
      cout << 1;
      b--;
    }
    while (a) {
      cout << 0;
      a--;
    }
  } else {
    while (a) {
      cout << 0;
      a--;
    }
    while (b) {
      cout << 1;
      b--;
    }
  }
  return 0;
}
