#include <bits/stdc++.h>
using namespace std;
int main() {
  long long a, b, c, d, e, i;
  cin >> a >> b;
  if (a % 2 == 0) {
    c = a / 2;
    if (b <= c)
      d = 2 * b - 1;
    else {
      e = b - c;
      d = 2 * e;
    }
  } else {
    c = a / 2;
    i = c + 1;
    if (b <= i)
      d = 2 * b - 1;
    else {
      e = b - i;
      d = 2 * e;
    }
  }
  cout << d << endl;
}
