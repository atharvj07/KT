import sys
from math import gcd

def rint():
	return map(int, sys.stdin.readline().split())
n = int(input())
ans = []
cur_num = n
for i in range(n):
	if cur_num <= 3:
		ans += [2 ** i] * (cur_num - 1) + [2 ** i * cur_num]
		break
	del_num = cur_num - cur_num // 2
	ans += [2 ** i] * del_num
	cur_num = cur_num - del_num
print(*ans)
