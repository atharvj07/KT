import os, sys, bisect, copy
from collections import defaultdict, Counter, deque
from functools import lru_cache
if os.path.exists('in.txt'):
	sys.stdin = open('in.txt', 'r')
if os.path.exists('out.txt'):
	sys.stdout = open('out.txt', 'w')

def input():
	return sys.stdin.readline()

def mapi(arg=0):
	return map(int if arg == 0 else str, input().split())
mod = 998244353

def modPow(x, y, p):
	(res, x) = (1, x % p)
	while y > 0:
		if y & 1 == 1:
			res = res * x % p
		(y, x) = (y >> 1, x * x % p)
	return res

def modInv(s):
	return modPow(s, mod - 2, mod)
(n, m) = mapi()
ans = 0
cur = 1
for i in range(1, n - 2):
	cur = cur * 2 % mod
cur = cur * (n - 2) % mod
ans = (ans + cur) % mod
x = 1
y = n - 2
cumx = n
while cumx <= m:
	cur = cur * (y + x) % mod
	cur = cur * modInv(x) % mod
	ans = (ans + cur) % mod
	cumx += 1
	x += 1
ans %= mod
print(int(ans))
