import java.math.BigInteger;
import java.util.Scanner;

public class editable {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		int n2 = 0;
		while(n % 2 == 0) {
			n >>= 1;
			++n2;
		}
		if(n == 1) {
			System.out.println(n2 + "/1");
			return;
		}
		Type A = new Type(BigInteger.ZERO, BigInteger.ONE), B = new Type(BigInteger.ONE, BigInteger.ONE);
		for(int i = 1, k = 0, cnt = 0;; ++cnt) {
			for(k = 0; (i << k) < n; ++k);
			int rem = (i << k) - n;
			A = A.add(B.multiply(k));
			B = B.multiply(rem);
			B.b = B.b.shiftLeft(k).multiply(BigInteger.valueOf(i));
			if(rem == 1) {
				B.a = B.b.subtract(B.a);
				A = A.divide(B);
				break;
			}
			i = rem;
			if(cnt > 20) {
				A.reduce();
				B.reduce();
				cnt = 0;
			}
		}
		A.a = A.a.add(A.b.multiply(BigInteger.valueOf(n2)));
		A.reduce();
		System.out.println(A.a + "/" + A.b);
	}
}

class Type {
	public BigInteger a, b;
	public Type(BigInteger a, BigInteger b) {
		this.a = a;
		this.b = b;
	}
	
	public Type add(Type curr) {
		BigInteger d = this.b.gcd(curr.b);
		BigInteger y = this.b.divide(d).multiply(curr.b);
		BigInteger x = this.a.multiply(y.divide(this.b)).add(curr.a.multiply(y.divide(curr.b)));
		return new Type(x, y);
	}
	
	public Type multiply(Type curr) {
		BigInteger arb = this.a.gcd(curr.b);
		BigInteger bra = this.b.gcd(curr.a);
		BigInteger x = this.a.divide(arb).multiply(curr.a.divide(bra));
		BigInteger y = this.b.divide(bra).multiply(curr.b.divide(arb));
		return new Type(x, y);
	}
	
	public Type multiply(long curr) {
		BigInteger r = BigInteger.valueOf(curr);
		BigInteger d = r.gcd(this.b);
		r = r.divide(d);
		return new Type(this.a.multiply(r), this.b.divide(d));
	}
	
	public Type divide(Type curr) {
		Type r = new Type(curr.b, curr.a);
		return this.multiply(r);
	}
	
	public void reduce() {
		BigInteger d = this.a.gcd(b);
		this.a = this.a.divide(d);
		this.b = this.b.divide(d);
	}
}
