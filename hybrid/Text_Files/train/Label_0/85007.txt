t = int(input())
for i in range(t):
	(n, x, y, d) = map(int, input().split())
	if abs(x - y) % d == 0:
		print(abs(x - y) // d)
	else:
		num = 10 ** 10
		num2 = 10 ** 10
		if (y - 1) % d == 0:
			num = (x - 1) // d + (y - 1) // d
			if (x - 1) % d != 0:
				num += 1
		if (n - y) % d == 0:
			num2 = (n - x) // d + (n - y) // d
			if (n - x) % d != 0:
				num2 += 1
		if min(num, num2) == 10 ** 10:
			print(-1)
		else:
			print(min(num, num2))
