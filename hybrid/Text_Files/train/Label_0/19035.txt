double cross(const xy_t &a, const xy_t &b){
	return imag(conj(a) * b);
}

double dot(const xy_t &a, const xy_t &b){
	return real(conj(a) * b);
}

int ccw(xy_t a, xy_t b, xy_t c){
	b-=a, c-=a;
	if(cross(b, c) > 0) return 1;
	if(cross(b, c) < 0) return -1;
	if(dot(b, c) < 0) return 2;
	if(norm(b) < norm(c)) return -2;
	return 0;
}

bool intersectSS(const line &l, const line &m){
	return ccw(l.first, l.second, m.first) * ccw(l.first, l.second, m.second) <= 0 &&
		ccw(m.first, m.second, l.first) * ccw(m.first, m.second, l.second) <= 0;
}

bool intersectPOP(const polygon &pol, const xy_t &p){
	bool in = false;
	int n = pol.size();
	rep(i, n){
		xy_t a = pol[i] - p, b = pol[(i+1)%n] - p;
		if(imag(a) > imag(b)) swap(a, b);
		if(imag(a) < EPS && 0 < imag(b) && cross(a, b) < 0) in = !in;
		if(abs(cross(a, b)) < EPS && dot(a, b) < EPS) return true;
	}
	return in;
}

int intersectPOPO(const polygon &p1, const polygon &p2){
	int n = p1.size();
	int m = p2.size();
	rep(i, n) rep(j, m) if(intersectSS(line(p1[i], p1[(i+1)%n]), line(p2[j], p2[(j+1)%m]))) return 1;
	rep(i, n) if(intersectPOP(p2, p1[i])) return 2;
	rep(i, m) if(intersectPOP(p1, p2[i])) return 3;
	return 0;
}
