#include <bits/stdc++.h>
using namespace std;
long long a[12][104][104];
int n, q, c;
long long sm(int t, int x, int y) {
  long long p = 0;
  for (int i1 = x; i1 > 0; i1 -= (i1 & (-i1)))
    for (int j1 = y; j1 > 0; j1 -= (j1 & (-j1))) p += a[t][i1][j1];
  return p;
}
void up(int k, int x, int y, int s) {
  for (int i1 = x; i1 <= 100; i1 += (i1 & (-i1)))
    for (int j1 = y; j1 <= 100; j1 += (j1 & (-j1)))
      a[k][i1][j1] += (s + k) % (c + 1);
}
int main() {
  cin >> n >> q >> c;
  for (int i = 1; i <= n; i++) {
    int x, y, s;
    cin >> x >> y >> s;
    for (int k = 0; k <= c; k++) {
      up(k, x, y, s);
    }
  }
  for (int i = 1; i <= q; i++) {
    int t, x, y, x2, y2;
    cin >> t >> x >> y >> x2 >> y2;
    x--;
    y--;
    t = t % (c + 1);
    long long s = 0;
    cout << sm(t, x2, y2) - sm(t, x, y2) - sm(t, x2, y) + sm(t, x, y) << endl;
  }
}
