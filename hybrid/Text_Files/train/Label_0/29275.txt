#include <bits/stdc++.h>
using namespace std;
const int inf = (1 << 30) - 1;
const long long linf = (1ll << 62) - 1;
const int N = 1e6 + 100;
const int K = 61;
int n;
long long x[N];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    scanf("%lld", &x[i]);
  }
  vector<char> used(n, false);
  vector<long long> a;
  for (int i = K - 1; i >= 0; i--) {
    vector<long long> b;
    int cur = -1;
    for (int j = 0; j < n; j++) {
      if (!used[j] && (x[j] & (1ll << i))) {
        while (cur < ((int)(a).size()) && cur != -1 && !(a[cur] & (1ll << i))) {
          b.push_back(a[cur++]);
        }
        if (cur < ((int)(a).size())) {
          if (cur != -1) {
            b.push_back(a[cur]);
          }
          cur++;
          b.push_back(x[j]);
          used[j] = true;
        } else {
          puts("No");
          return 0;
        }
      }
    }
    while (cur < ((int)(a).size())) {
      if (cur != -1) {
        b.push_back(a[cur]);
      }
      cur++;
    }
    a = move(b);
  }
  puts("Yes");
  for (auto i : a) {
    printf("%lld ", i);
  }
  puts("");
  return 0;
}
