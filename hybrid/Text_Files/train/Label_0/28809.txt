#include <bits/stdc++.h>
using namespace std;
const int N = 1e5;
vector<int> e[N], ans;
int n, a[N], goal[N];
void rec(int v, int p = -1, int k_now = 0, int k_next = 0) {
  a[v] = (a[v] + k_now) % 2;
  if (a[v] != goal[v]) {
    ans.push_back(v + 1);
    k_now++;
  }
  for (int i = 0; i < e[v].size(); i++) {
    int to = e[v][i];
    if (to == p) continue;
    rec(to, v, k_next, k_now);
  }
}
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n - 1; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    x--;
    y--;
    e[x].push_back(y);
    e[y].push_back(x);
  }
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int i = 0; i < n; i++) scanf("%d", &goal[i]);
  rec(0);
  printf("%d\n", ans.size());
  for (int i = 0; i < ans.size(); i++) printf("%d\n", ans[i]);
  return 0;
}
