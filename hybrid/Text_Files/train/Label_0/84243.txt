from collections import defaultdict as dd
n = int(input())
l = list(map(int, input().strip().split()))
nmfr = [0] * 100001
ocfr = [0] * 100001
(nmfr[0], ocfr[0]) = (0, 0)
setb = 0
(mnb, mxb) = (1, 1)
strk = 0
for (ind, i) in enumerate(l):
	nmfr[i] += 1
	ocfr[nmfr[i]] += 1
	ocfr[nmfr[i] - 1] -= 1
	if ocfr[nmfr[i]] == 1:
		setb += 1
	if ocfr[nmfr[i] - 1] == 0:
		setb -= 1
		if mnb == nmfr[i] - 1:
			mnb += 1
	mnb = min(mnb, nmfr[i])
	mxb = max(mxb, nmfr[i])
	if mxb == mnb and ocfr[mxb] == 1 or mxb == mnb == 1 or (mxb - mnb == 1 and ocfr[mxb] == 1) or (setb == 2 and ocfr[mnb] == mnb == 1):
		strk = ind + 1
print(strk)
