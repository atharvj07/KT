#include <bits/stdc++.h>
using namespace std;
string s;
int n, gg;
int check(int x) {
  int i = 1;
  int mm = gg;
  while (i <= s.size()) {
    if (s[i] == x) {
      if (abs(s[i] - s[i - 1]) == 1 or abs(s[i] - s[i + 1] == 1)) {
        s.erase(i, 1);
        gg++;
      }
    }
    i++;
  }
  if (mm == gg)
    return 0;
  else
    return 1;
}
int main() {
  cin >> n;
  cin >> s;
  int len = s.size();
  s = '0' + s + '0';
  gg = 0;
  int k = 'z';
  while (k >= 'a') {
    while (check(k)) {
    }
    k--;
  }
  cout << gg << endl;
  return 0;
}
