import java.util.Scanner;
import java.util.HashSet;
public class A
{
       public static void main(String args[])
       {
              Scanner sc=new Scanner(System.in);
              int n=sc.nextInt();
              int a[]=new int[n];
              for(int i=0;i<n;i++)
                     a[i]=sc.nextInt();
              int max=1;
              int t=1;
              for(int i=0;i<n-1;i++)
              {
                     if(a[i+1]<=2*a[i])
                     {
                            ++t;
                            //System.out.println(a[i]);
                     }
                     else
                            t=1;
                     
                     if(t>max)
                            max=t;
              }
              System.out.println(max);
              
       }
}
