def gg(k):
	if k * n > s:
		return True
	else:
		return False
q = int(input())
for i in range(q):
	(a, b, n, s) = map(int, input().split())
	l = -1
	r = a + 1
	while l + 1 < r:
		m = (l + r) // 2
		if gg(m):
			r = m
		else:
			l = m
	if l <= a and l * n <= s and (l * n + b >= s):
		print('YES')
	else:
		print('NO')
