import copy

def f(n, w, L):
	D = dict()
	count = 0
	for i in range(n):
		D[i] = -1 * (-1 * L[i] // 2)
		count += D[i]
	if count > w:
		return -1
	E = dict()
	M = copy.deepcopy(L)
	M.sort()
	N = copy.deepcopy(L)
	j = n - 1
	while j > -1:
		m = max(N)
		i = N.index(m)
		E[j] = i
		N[i] = -1
		j -= 1
	remaining = w - count
	k = n - 1
	while remaining > 0:
		i = E[k]
		diff = L[i] - D[i]
		if remaining >= diff:
			D[i] = L[i]
		else:
			D[i] += remaining
		remaining -= diff
		k -= 1
	s = ''
	for i in range(n):
		s += str(D[i]) + ' '
	return s

def run():
	i = 0
	for line in iter(input, ''):
		if i == 0:
			a = line.split(' ')
			(n, w) = (int(a[0]), int(a[1]))
		else:
			line = line.split(' ')
			L = list(map(int, line))
			break
		i += 1
	print(f(n, w, L))
run()
