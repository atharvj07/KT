from math import gcd

def check(l2, p):
	sum1 = 0
	for j in l2:
		if gcd(p, 2 * j) > 1:
			sum1 += 1
	return sum1
for _ in range(int(input())):
	n = int(input())
	l1 = list(map(int, input().split()))
	l2 = []
	l3 = []
	for i in l1:
		if i % 2 == 0:
			l2.append(i)
		else:
			l3.append(i)
	l1 = l2 + l3
	count = 0
	for i in range(n):
		if l1[i] % 2 == 0:
			count += len(l1[i + 1:])
		else:
			p = check(l1[i + 1:], l1[i])
			count += p
	print(count)
