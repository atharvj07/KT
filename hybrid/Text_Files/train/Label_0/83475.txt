from collections import Counter

def main():
	n = int(input())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	rv = 0
	nums = {}
	for i in range(len(a)):
		if a[i] not in nums:
			nums[a[i]] = []
		nums[a[i]].append(i)
	visited = set()
	for num in nums:
		if len(nums[num]) > 1:
			i = nums[num][0]
			j = nums[num][1]
			if i not in visited:
				visited.add(i)
				rv += b[i]
			if j not in visited:
				visited.add(j)
				rv += b[j]
			for k in range(len(a)):
				if k in visited:
					continue
				if a[i] | a[k] == a[i]:
					visited.add(k)
					rv += b[k]
	return rv
print(main())
