#include <bits/stdc++.h>
using namespace std;
int w[1111], h[1111];
int main() {
  int n;
  while (cin >> n) {
    for (int i = 0; i < n; i++) cin >> w[i] >> h[i];
    long long result = LLONG_MAX;
    for (int curh = 1; curh <= 1000; curh++) {
      int curw = 0, swaps = 0;
      vector<int> diff;
      for (int i = 0; i < n; i++)
        if (w[i] <= curh && h[i] <= curh) {
          if (w[i] > h[i]) diff.push_back(h[i] - w[i]);
          curw += w[i];
        } else if (w[i] <= curh && h[i] > curh) {
          curw += h[i];
          ++swaps;
        } else if (w[i] > curh && h[i] <= curh) {
          curw += w[i];
        } else {
          swaps = n;
          break;
        }
      sort(diff.begin(), diff.end());
      for (int i = 0; i < diff.size() && swaps < n / 2; i++) {
        curw += diff[i];
        ++swaps;
      }
      if (swaps <= n / 2) result = min(result, curh * 1LL * curw);
    }
    cout << result << "\n";
  }
  return 0;
}
