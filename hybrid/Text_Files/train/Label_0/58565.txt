import java.util.Arrays;
import java.util.Scanner;

public class Main {


    static int counter;

    public static void main(String[] args) {

        Scanner in = new Scanner(System.in);

        int n = in.nextInt();

        long [] array = new long[n];

        long type1 = -1,type2 = -1,type3=-1;
        boolean can = true;
        for(int i =0;i<n;++i){
            long input = in.nextLong();
            if(!can)continue;
            if(type1 == -1 || type1 == input){
                type1 = input;
                continue;
            }
            if(type2 == -1|| type2 == input){
                type2 = input;
                continue;
            }
            if(type3 == -1|| type3 == input){
                type3 = input;
                continue;
            }
            can = false;
        }
        
        if(can){
            if(type1 == -1 || type2 == -1||type3 == -1){
                System.out.println("YES");
                return;
            }
            if(max(type1,type2,type3) - midle(type1,type2,type3) == midle(type1,type2,type3) - min(type1,type2,type3)){
                System.out.println("YES");
                return;
            }
        }
        System.out.println("NO");
    }

    static long midle(long l1,long l2,long l3){
       if(l1>min(l1,l2,l3)&&l1<max(l1,l2,l3))return l1;
        if(l2>min(l1,l2,l3)&&l2<max(l1,l2,l3))return l2;
       return l3;
    }
    static long max(long l1,long l2,long l3){
        if(l1<l2&&l2>l3) return l2;
        if(l2<l1&&l1>l3)return l1;
        return l3;
    }
    static long min(long l1,long l2,long l3){
        if(l1>l2&&l2<l3) return l2;
        if (l2>l1&&l1<l3)return l1;
        return l3;
    }

}
