import java.util.*;
public class vv {

 public static void main(String[] args) {
  Scanner no=new Scanner(System.in);
  int n=no.nextInt();
  System.out.print(n+" ");
  while(n!=1){
    if(n%(n/2)==0){
      System.out.print(n/2+" ");
      n=n/2;
    }
    else{
      int i=n/2;
      while(n%i!=0){
        i--;
      }
      System.out.println(i+" ");
      n=i;
    }
  }
 }

}
