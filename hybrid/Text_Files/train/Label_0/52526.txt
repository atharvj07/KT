import java.util.*;
public class JavaApplication139 {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
      
        Scanner sc = new Scanner(System.in);
        int t=sc.nextInt();
        while(t--!=0)
        {
            int n=sc.nextInt();
            int m=sc.nextInt();
            for(int i=0;i<n;i++)
            {
                for(int j=0;j<m;j++)
                {
                    if(i==n-1 && j==m-1)
                        System.out.print("W");
                    else
                        System.out.print("B");
                }
                System.out.println();
            }
  
              
                   
        }
        
    }
    
}
