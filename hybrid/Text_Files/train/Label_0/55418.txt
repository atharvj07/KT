import java.util.*;
import java.math.*;

public class A
{
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		char[] S = in.next().toCharArray();

		boolean[] alph = new boolean[26];
		int wilds = 0;

		int N = S.length;
		for (int i = 0; i < N; i++)
			if (S[i] == '?')
				wilds++;
			else if (S[i] >= 'A' && S[i] <= 'J')
				alph[S[i] - 'A'] = true;

		int alphs = 0;
		for (int i = 0; i < 26; i++)
			if (alph[i])
				alphs++;

		BigInteger res = BigInteger.ONE;

		if (S[0] == '?')
		{
			res = BigInteger.valueOf(9);
			for (int i = 0; i < alphs; i++)
				res = res.multiply(BigInteger.valueOf(10-i));

			System.out.print(res);
			for (int i = 0; i < wilds-1; i++)
				System.out.print("0");
			System.out.println();
		}
		else if (S[0] >= 'A' && S[0] <= 'J')
		{
			res = BigInteger.valueOf(9);
			for (int i = 0; i < alphs-1; i++)
				res = res.multiply(BigInteger.valueOf(9-i));

			System.out.print(res);
			for (int i = 0; i < wilds; i++)
				System.out.print("0");
			System.out.println();
		}
		else
		{
			for (int i = 0; i < alphs; i++)
				res = res.multiply(BigInteger.valueOf(10-i));

			System.out.print(res);
			for (int i = 0; i < wilds; i++)
				System.out.print("0");
			System.out.println();
		}

	}
}
