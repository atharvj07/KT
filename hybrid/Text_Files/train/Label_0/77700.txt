t = int(input())
l = []
for i in range(t):
	s = input().split()
	s = [int(i) for i in s]
	(a, b, c, d) = (s[0], s[1], s[2], s[3])
	one = max(a, b)
	two = max(c, d)
	s.sort(reverse=True)
	if max(one, two) == s[0] and min(one, two) == s[1]:
		l.append('YES')
	else:
		l.append('NO')
for i in l:
	print(i)
