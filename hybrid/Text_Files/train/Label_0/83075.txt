n = int(input())
ls = [int(x) for x in input().split()]
max_dif = 10 ** 9
temp_dif = 0
elem_ind = 0
for ind in range(1, len(ls) - 1):
	temp_dif = abs(ls[ind - 1] - ls[ind]) + abs(ls[ind + 1] - ls[ind])
	if temp_dif < max_dif:
		max_dif = min(temp_dif, max_dif)
		elem_ind = ind
del ls[elem_ind]
global_res = -1
for ind in range(1, len(ls)):
	res = abs(ls[ind] - ls[ind - 1])
	global_res = max(global_res, res)
print(global_res)
