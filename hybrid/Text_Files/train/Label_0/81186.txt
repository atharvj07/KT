for _ in range(int(input())):
	n = int(input())
	miners = []
	diamond = []
	for _ in range(2 * n):
		(x, y) = map(int, input().split())
		if x == 0:
			miners.append(abs(y))
		else:
			diamond.append(abs(x))
	miners.sort()
	diamond.sort()
	ans = 0
	for i in range(len(miners)):
		(x, y) = (miners[i], diamond[i])
		ans += (x ** 2 + y ** 2) ** 0.5
	print(ans)
