#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, min = 1000;
  string s;
  cin >> n;
  cin >> s;
  for (int i = 0; i < n - 3; i++) {
    int pa1, pa2, pa3, pa4, temp = 0;
    pa1 = fabs(s[i] - 'A');
    if (pa1 <= 13)
      temp += pa1;
    else
      temp += 26 - pa1;
    pa2 = fabs(s[i + 1] - 'C');
    if (pa2 <= 13)
      temp += pa2;
    else
      temp += 26 - pa2;
    pa3 = fabs(s[i + 2] - 'T');
    if (pa3 <= 13)
      temp += pa3;
    else
      temp += 26 - pa3;
    pa4 = fabs(s[i + 3] - 'G');
    if (pa4 <= 13)
      temp += pa4;
    else
      temp += 26 - pa4;
    if (temp < min) min = temp;
  }
  cout << min;
  return 0;
}
