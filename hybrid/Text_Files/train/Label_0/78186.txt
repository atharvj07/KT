from sys import stdin
inp = stdin.readline
(n, p) = map(int, inp().split())

def cnt(l, r):
	return r // p - l // p if l % p else r // p - l // p + 1

def pb(l, r):
	return cnt(l, r) / (r - l + 1)
w = [pb(*map(int, inp().split())) for _ in range(n)]
ans = 0
for i in range(n):
	ans += 2 * w[i] - w[i] * w[(i + 1) % n]
print(ans * 2000)
