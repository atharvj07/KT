#include <bits/stdc++.h>
using namespace std;
inline int read() {
  int ans = 0;
  char c = getchar();
  while (!isdigit(c)) c = getchar();
  for (; isdigit(c); c = getchar()) ans = ans * 10 + c - '0';
  return ans;
}
template <typename _Tp>
inline int chkmin(_Tp &x, _Tp y) {
  return x > y ? x = y, 1 : 0;
}
template <typename _Tp>
inline int chkmax(_Tp &x, _Tp y) {
  return x < y ? x = y, 1 : 0;
}
map<pair<int, int>, int> mem;
inline int ask(int x, int y) {
  if (mem.count(make_pair(x, y))) return mem[make_pair(x, y)];
  printf("? %d %d\n", x, y), fflush(stdout);
  return mem[make_pair(x, y)] = read();
}
inline vector<int> Reduce(vector<int> rows, vector<int> cols) {
  vector<int> ans;
  const int n = ((int)(rows).size());
  int i = 0, r;
  for (int c : cols) {
    if (ans.empty()) {
      ans.push_back(c);
      continue;
    }
    r = rows[i];
    int y = ans.back();
    while (ask(r, y) > ask(r, c)) {
      ans.pop_back();
      --i;
      if (ans.empty()) break;
      r = rows[i];
      y = ans.back();
    }
    if (((int)(ans).size()) == n)
      continue;
    else {
      ans.push_back(c);
      ++i;
    }
  }
  return ans;
}
vector<int> Solve(vector<int> rows, vector<int> cols) {
  if (((int)(rows).size()) < ((int)(cols).size())) {
    cols = Reduce(rows, cols);
  }
  const int n = ((int)(rows).size()), m = ((int)(cols).size());
  vector<int> nrows;
  for (int i = 1; i < n; i += 2) {
    nrows.push_back(rows[i]);
  }
  vector<int> nans;
  if (!nrows.empty()) {
    nans = Solve(nrows, cols);
  }
  vector<int> ans;
  int lc = 0, rc;
  int x, y;
  for (int i = (0), ed_i = (n - 1); i <= ed_i; ++i) {
    if (i & 1) {
      ans.push_back(nans[i >> 1]);
      continue;
    }
    rc = m - 1;
    if (i + 1 < n) {
      rc = lc;
      while (cols[rc] < nans[i >> 1]) {
        ++rc;
      }
    } else {
      rc = m - 1;
    }
    x = rows[i];
    y = cols[lc];
    int mn = ask(x, y);
    for (int j = (lc), ed_j = (rc); j <= ed_j; ++j) {
      if (chkmin(mn, ask(x, cols[j]))) {
        y = cols[j];
      }
    }
    ans.push_back(y);
    lc = rc;
  }
  return ans;
}
int main() {
  int n = read(), m = read();
  vector<int> rows, cols;
  for (int i = (1), ed_i = (n); i <= ed_i; ++i) rows.push_back(i);
  for (int i = (1), ed_i = (m); i <= ed_i; ++i) cols.push_back(i);
  vector<int> ans = Solve(rows, cols);
  int mn = 0x3f3f3f3f;
  for (int i = (0), ed_i = (n - 1); i <= ed_i; ++i) {
    chkmin(mn, ask(rows[i], ans[i]));
  }
  printf("! %d\n", mn), fflush(stdout);
  return 0;
}
