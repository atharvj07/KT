(h1, a1, c1) = (int(i) for i in input().split())
(h2, a2) = map(int, input().split())
n = abs(-h2 // a1)
a = list()
while n > 0:
	if h2 - a1 > 0 and h1 - a2 <= 0:
		a.append('HEAL')
		h1 += c1
		h1 -= a2
	else:
		a.append('STRIKE')
		h2 -= a1
		h1 -= a2
		n -= 1
print(len(a))
for i in range(len(a)):
	print(a[i])
