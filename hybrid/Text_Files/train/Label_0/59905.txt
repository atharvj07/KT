import java.util.*;
import java.lang.*;
import java.math.*;
import java.io.*;
import static java.lang.Math.*;
import static java.util.Arrays.*;
import static java.util.Collections.*;

public class P053E{
	Scanner sc=new Scanner(System.in);

	int INF=1<<28;
	double EPS=1e-9;

	int n, m, k;
	int[] u, v;

	void run(){
		n=sc.nextInt();
		m=sc.nextInt();
		k=sc.nextInt();
		u=new int[m];
		v=new int[m];
		for(int i=0; i<m; i++){
			u[i]=sc.nextInt()-1;
			v[i]=sc.nextInt()-1;
		}
		solve();
	}

	void solve(){
		long[][] dp=new long[1<<n][1<<n];
		// dp[s][t]:=「sが連結かつtが葉」の個数

		for(int i=0; i<m; i++){
			dp[(1<<u[i])|(1<<v[i])][(1<<u[i])|(1<<v[i])]=2;
		}
		for(int s=0; s<1<<n; s++){
			int t=s;
			do{
				if(Integer.bitCount(t)>0){
					dp[s][t]/=Integer.bitCount(t);
				}
				for(int i=0; i<m; i++){
					boolean ub=(s>>u[i]&1)==1;
					boolean vb=(s>>v[i]&1)==1;
					if(ub&!vb){
						dp[s|(1<<v[i])][t&~(1<<u[i])|(1<<v[i])]+=dp[s][t];
					}
					if(!ub&vb){
						dp[s|(1<<u[i])][t&~(1<<v[i])|(1<<u[i])]+=dp[s][t];
					}
				}
				t=(t-1)&s;
			}while(t!=s);
		}

		long ans=0;
		int comb=(1<<k)-1;
		for(; comb<1<<n;){
			ans+=dp[(1<<n)-1][comb];
			int x=comb&-comb, y=comb+x;
			comb=((comb&~y)/x>>>1)|y;
		}
		println(ans+"");
	}

	void println(String s){
		System.out.println(s);
	}

	void print(String s){
		System.out.print(s);
	}

	public static void main(String[] args){
		Locale.setDefault(Locale.US);
		new P053E().run();
	}
}
