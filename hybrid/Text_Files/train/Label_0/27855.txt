#include <bits/stdc++.h>
using namespace std;
int n, m;
int bo[25], f1[25];
vector<int> ans;
int answer;
void dfs(int k, int cnt) {
  if (cnt == answer) return;
  if (k == n) {
    for (int i = 0; i < n; ++i)
      if (f1[i] != (1 << n) - 1) return;
    ans.clear();
    for (int i = 0; i < n; ++i)
      if (bo[i]) ans.push_back(i);
    answer = ans.size();
    return;
  }
  bo[k] = 1;
  int f2[25];
  for (int i = 0; i < n; ++i) f2[i] = f1[i];
  for (int i = 0; i < n; ++i)
    if (f2[k] & (1 << i)) f1[i] |= f2[k];
  dfs(k + 1, cnt + 1);
  bo[k] = 0;
  for (int i = 0; i < n; ++i) f1[i] = f2[i];
  dfs(k + 1, cnt);
}
int main() {
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; ++i) f1[i] |= (1 << i);
  for (int i = 1; i <= m; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    --x, --y;
    f1[x] |= (1 << y);
    f1[y] |= (1 << x);
  }
  answer = n;
  dfs(0, 0);
  printf("%d\n", answer);
  for (int i = 0; i < answer; ++i) printf("%d ", ans[i] + 1);
}
