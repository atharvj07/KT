(n, k) = [int(i) for i in input().split()]
a = [int(i) for i in input().split()]
b = [int(i) for i in input().split()]
cook = 0
f = True
while f:
	for i in range(n):
		if b[i] >= a[i]:
			b[i] -= a[i]
		elif k >= a[i] - b[i]:
			k -= a[i] - b[i]
			b[i] = 0
		else:
			f = False
	if f:
		cook += 1
print(cook)
