import sys
import math

def II():
	return int(sys.stdin.readline())

def LI():
	return list(map(int, sys.stdin.readline().split()))

def MI():
	return map(int, sys.stdin.readline().split())

def SI():
	return sys.stdin.readline().strip()
t = II()
a = LI()
ans = []
for j in range(30, 0, -1):
	s = 0
	answer = 0
	m = 0
	for i in range(t):
		if a[i] > j:
			s = 0
			m = 0
			continue
		s += a[i]
		if s - m > answer:
			answer = s - m
		if m > s:
			m = s
	ans.append(answer)
m = 0
for i in range(30):
	m = max(m, ans[i] - (30 - i))
print(m)
