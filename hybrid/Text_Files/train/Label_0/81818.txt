for tc in range(int(input())):
	(n, s) = input().split(' ')
	n = int(n)
	inc = [-1] * n
	dec = [-1] * n
	cnti = 1
	for i in range(len(s)):
		if s[i] == '<':
			inc[i] = cnti
			cnti += 1
	lcnt = n
	for i in range(n):
		if inc[i] == -1:
			inc[i] = lcnt
			lcnt -= 1
	cntl = 1
	i = n - 2
	while i >= 0:
		if s[i] == '<':
			si = i
			ei = i
			while si >= 0 and s[si] != '>':
				si -= 1
			for j in range(si + 1, ei + 1):
				dec[j] = cntl
				cntl += 1
			i = si
		else:
			i -= 1
	lcnt = n
	for i in range(n):
		if dec[i] == -1:
			dec[i] = lcnt
			lcnt -= 1
	print(*dec)
	print(*inc)
