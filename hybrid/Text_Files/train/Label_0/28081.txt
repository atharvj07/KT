#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e6 + 10;
int n, m;
int a[maxn];
int d[maxn];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  int A = a[1];
  int LA = 1;
  int B = 0;
  int LB = 0;
  d[1] = -1;
  for (int i = 2; i <= n; i++) {
    if (a[i] == A) {
      d[i] = LB;
      LA = i;
    } else if (a[i] == B || B == 0) {
      B = a[i];
      LB = i;
      d[i] = LA;
    } else {
      if (LA < LB) {
        d[i] = i - 1;
        A = a[i];
        LA = i;
      } else {
        d[i] = i - 1;
        B = a[i];
        LB = i;
      }
    }
  }
  int l, r, x;
  for (int i = 1; i <= m; i++) {
    scanf("%d%d%d", &l, &r, &x);
    if (l == r) {
      if (a[l] == x)
        printf("-1\n");
      else
        printf("%d\n", l);
    } else if (d[r] < l && a[r] == x)
      printf("-1\n");
    else {
      if (a[r] != x)
        printf("%d\n", r);
      else {
        printf("%d\n", d[r]);
      }
    }
  }
  return 0;
}
