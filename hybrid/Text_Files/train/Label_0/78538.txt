def ceil(c):
	if int(c) == c:
		return int(c)
	return int(c) + 1

def compute_lcm(x, y):
	if x > y:
		greater = x
	else:
		greater = y
	while True:
		if greater % x == 0 and greater % y == 0:
			lcm = greater
			break
		greater += 1
	return lcm

def solve():
	s = input()
	t = input()
	ls = len(s)
	lt = len(t)
	if s == t:
		return s
	mi = min(ls, lt)
	(ts, tt) = ('', '')
	for i in range(ls):
		ts += s[i]
		if s.count(ts) == ceil(ls / len(ts)):
			break
	for i in range(lt):
		tt += t[i]
		if t.count(tt) == ceil(lt / len(tt)):
			break
	if ts != tt:
		return -1
	return ts * compute_lcm(t.count(ts), s.count(ts))
for _ in range(int(input())):
	print(solve())
