#include <bits/stdc++.h>
using namespace std;
const long long inf = 1000000007LL;
const double EPS = 1e-14;
const long long inf_ll = (long long)1e18;
const long long maxn = 200005LL;
const long long mod = 810872LL;
struct Graph {
  struct edge {
    int u, v, nxt;
    edge(int u = 0, int v = 0, int nxt = 0) : u(u), v(v), nxt(nxt) {}
  } E[maxn * 2];
  int cnt, head[maxn];
  inline void link(int u, int v) {
    E[++cnt] = edge(u, v, head[u]);
    head[u] = cnt;
  }
  inline void link2(int u, int v) {
    link(u, v);
    link(v, u);
  }
} G;
int n;
int vis[maxn], D[maxn];
int deg[maxn];
int order[maxn], order_clock;
inline void del(int u) {
  order[++order_clock] = u;
  D[u] = 1;
}
void dfs2(int u, int p) {
  if (!D[u]) del(u);
  for (register int(i) = G.head[u], (v) = G.E[i].v; (i);
       (i) = G.E[i].nxt, (v) = G.E[i].v)
    if (v != p && !D[v]) dfs2(v, u);
}
void dfs1(int u, int p) {
  vis[u] = 1;
  int cnt1 = !!p, cnt2 = 0;
  for (register int(i) = G.head[u], (v) = G.E[i].v; (i);
       (i) = G.E[i].nxt, (v) = G.E[i].v)
    if (v ^ p) {
      ++cnt1;
      dfs1(v, u);
      if (D[v]) ++cnt2;
    }
  if ((cnt1 - cnt2) % 2 == 0) {
    del(u);
    dfs2(u, p);
  }
}
int main() {
  scanf("%d", &n);
  for (register int(i) = (1); (i) <= (n); ++i) {
    int u;
    scanf("%d", &u);
    if (!u) continue;
    G.link2(i, u);
    deg[i] ^= 1;
    deg[u] ^= 1;
  }
  if (n % 2 == 0) {
    puts("NO");
    return 0;
  }
  dfs1(1, 0);
  puts("YES");
  for (register int(i) = (1); (i) <= (n); ++i) printf("%d\n", order[i]);
  return 0;
}
