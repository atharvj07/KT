	return sys.stdin.readline()[:-1]
import string

MOD = int(1000000000.0) + 7

def solve():
	(n, M) = [int(x) for x in input().split()]
	a = [int(x) for x in input().split()]
	total = sum(a)
	if total < M:
		print(-1)
		return
	l = 0
	r = n + 1
	a.sort(reverse=True)

	def f(x):
		cnt = 0
		cycle = 0
		i = 0
		for e in a:
			if e > cycle:
				i = (i + 1) % x
				cnt += e - cycle
			if i == 0:
				cycle += 1
		return cnt >= M
	while l + 1 < r:
		m = (l + r) // 2
		if f(m):
			r = m
		else:
			l = m
	print(r)
t = 1
for case in range(1, t + 1):
	ans = solve()
