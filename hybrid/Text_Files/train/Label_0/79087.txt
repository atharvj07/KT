def func(arr, l, r):
	if len(arr) == 0:
		return A
	elif l == r:
		return B * len(arr)
	else:
		z = (l + r) // 2
		back = []
		fow = []
		for i in arr:
			if i <= z:
				back.append(i)
			else:
				fow.append(i)
		return min(B * len(arr) * (r - l + 1), func(back, l, z) + func(fow, z + 1, r))
(n, k, A, B) = map(int, input().split())
lst = list(map(int, input().split()))
lst.sort()
print(func(lst, 1, 2 ** n))
