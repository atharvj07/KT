import java.io.*;
import java.util.StringTokenizer;

public class B {

	public static void main(String[] args) throws Exception {

		Scanner sc = new Scanner(System.in);

		int n = sc.nextInt(), m = sc.nextInt(), c = sc.nextInt();
		int[] arr = new int[n];

		int leftEnd = -1, rightEnd = n;
		while(leftEnd + 1 < rightEnd && m-->0)
		{
			int p = sc.nextInt();
			if(p * 2 <= c)
			{
				// left pointer
				int i = 0;
				while(arr[i] != 0 && arr[i] <= p)
					++i;
				arr[i] = p;
				leftEnd = Math.max(leftEnd, i);
				System.out.println(i + 1);
				System.out.flush();
			}
			else
			{
				// right pointer
				int i = n - 1;
				while(arr[i] != 0 && arr[i] >= p)
					--i;
				arr[i] = p;
				rightEnd = Math.min(rightEnd, i);
				System.out.println(i + 1);
				System.out.flush();
			}
		}

	}

	static class Scanner
	{
		StringTokenizer st;
		BufferedReader br;

		public Scanner(InputStream s){	br = new BufferedReader(new InputStreamReader(s));}

		public Scanner(FileReader s) throws FileNotFoundException {	br = new BufferedReader(s);}

		public String next() throws IOException
		{
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		public int nextInt() throws IOException {return Integer.parseInt(next());}

		public long nextLong() throws IOException {return Long.parseLong(next());}

		public String nextLine() throws IOException {return br.readLine();}

		public double nextDouble() throws IOException { return Double.parseDouble(next()); }

		public boolean ready() throws IOException {return br.ready();}
	}
}
