import math
T = int(input())
for _ in range(T):
	(N, K) = map(int, input().split(' '))
	if N <= K:
		res = 1
	else:
		num = min(int(math.sqrt(N)), K)
		res = N
		while num >= 1:
			if N % num == 0:
				t = N // num
				if t <= K:
					res = min(res, num)
				else:
					res = min(res, t)
			num -= 1
	print(res)
