import java.util.Scanner;

//https://codeforces.com/contest/1178/problem/F1
public class ShortColorfulStrip {
    private static long[][] dp;
    private static int[][] minInd;
    private static long MOD = 998244353L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] min = new int[n][n];
        minInd = new int[n][n];
        for (int i = 0; i < n; i++) {
            min[i][i] = a[i];
            minInd[i][i] = i;
            for (int j = i + 1; j < n; j++) {
                if (a[j] < min[i][j - 1]) {
                    min[i][j] = a[j];
                    minInd[i][j] = j;
                } else {
                    min[i][j] = min[i][j - 1];
                    minInd[i][j] = minInd[i][j - 1];
                }
            }
        }
        dp = new long[n][n];
        System.out.println(solve(0, n - 1));
    }

    private static long solve(int stInd, int endInd) {
        if (endInd < stInd) {
            return 1;
        }
        if (dp[stInd][endInd] != 0) {
            return dp[stInd][endInd];
        }
        int mid = minInd[stInd][endInd];
        long lowWays = 0;
        for (int j = stInd - 1; j < mid; j++) {
            lowWays = (lowWays + solve(stInd, j) * solve(j + 1, mid - 1)) % MOD;
        }
        long highWays = 0;
        for (int j = mid; j <= endInd; j++) {
            highWays = (highWays + solve(mid + 1, j) * solve(j + 1, endInd)) % MOD;
        }
        long ways = (lowWays * highWays) % MOD;
        dp[stInd][endInd] = ways;
        return dp[stInd][endInd];
    }
}
