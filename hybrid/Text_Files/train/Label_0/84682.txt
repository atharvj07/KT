for _ in range(int(input())):
	(n, m) = map(int, input().split())
	a = []
	b = []
	for i in range(n):
		a.append(input())
		b.append([0] * m)
	s = 0
	for i in range(n - 1, -1, -1):
		for j in range(m):
			if a[i][j] == '*':
				if i == n - 1:
					b[i][j] = 1
					s += 1
				elif j == 0 or j == m - 1:
					b[i][j] = 1
					s += 1
				else:
					x = b[i + 1][j - 1:j + 2]
					b[i][j] = min(x) + 1
					s += b[i][j]
	print(s)
