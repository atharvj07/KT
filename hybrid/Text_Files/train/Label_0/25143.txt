#include <bits/stdc++.h>
using namespace std;
using p=pair<int,int>;
const int maxn=100001;

signed main(){
	int n;cin>>n;
  	vector<int> v(n);
  	for(auto&& w:v)cin>>w;
	vector<p> even(maxn),odd(maxn);
	for(int i=0;i<maxn;i++)even[i]=odd[i]=make_pair(i,0);	
	for(int i=0;i<n;i++){
	if(i%2==0){
      (even[v[i]].second)++;
    }else{
      (odd[v[i]].second)++;
	}
    }  
	sort(even.begin(),even.end(),[&](p x,p y){return x.second>y.second;});
	sort(odd.begin(),odd.end(),[&](p x,p y){return x.second>y.second;});
	int ans;
	if(even[0].first==odd[0].first){
      ans=min(n-odd[0].second-even[1].second,n-odd[1].second-even[0].second);
    }else{
      ans=n-odd[0].second-even[0].second;
    }
    cout<<ans<<endl;
	
}
