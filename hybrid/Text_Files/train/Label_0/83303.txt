(n, m, k) = map(int, input().split())
(i, j, res) = (1, 1, [])
while i <= n:
	if j < 1 or j > m:
		i += 1
		j += 1 if i % 2 == 1 else -1
	if i > n:
		break
	res.append(str(i) + ' ' + str(j))
	j += 1 if i % 2 == 1 else -1
d = len(res) // k
for i in range(k - 1):
	print(d, ' '.join(res[i * d:i * d + d]))
if k == 1:
	i = -1
res = res[i * d + d:]
print(len(res), ' '.join(res))
