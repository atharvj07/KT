#include <bits/stdc++.h>
using namespace std;
const int N = 5e5 + 7, NN = 1e7 + 3;
int n, v, f, first, second, factor[NN];
pair<int, int> ans[N];
void sieve() {
  iota(factor, factor + NN, 0);
  factor[1] = -1;
  for (int i = 2; i < NN / i; i += 1 + (i & 1)) {
    if (factor[i] == i)
      for (int j = i * i; j < NN; j += i)
        if (factor[j] == j) factor[j] = i;
  }
}
int main() {
  sieve();
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> v;
    second = v;
    f = factor[v];
    first = 1;
    while (factor[v] == f) {
      first *= f;
      v /= factor[v];
    }
    second /= first;
    if (second == 1)
      ans[i] = {-1, -1};
    else
      ans[i] = {first, second};
  }
  for (int i = 0; i < n - 1; i++) cout << ans[i].first << " ";
  cout << ans[n - 1].first << endl;
  for (int i = 0; i < n - 1; i++) cout << ans[i].second << " ";
  cout << ans[n - 1].second << endl;
  return 0;
}
