/*
 * solution mostly inspired by https://codeforces.com/contest/892/submission/32569876
 * if some edge with weight w is in the MST then all edges of weight less than w must be included in the MST
 * as well (provided that they do not form a cycle)  
 * to implement this , i used the following, let us sort all the edges in non-decreasing order
 * and let us union all the edges as we process them in this order, while doing this we remember the set that each edge
 * belonged to at this point , the idea is that this way we can skip simulating the union of edges less than w per query. 
 * per query, we can just union the edges by their parents we remembered earlier and make sure there is no cycle,
 * and then let us undo what we have done after each query,
 * I am not sure though what is the complexity of the Union Find Operations 
 */
import java.io.*;
import java.util.*;

public class B {

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner();
		PrintWriter out = new PrintWriter(System.out);
		int n = sc.nextInt(), m = sc.nextInt();
		int[] from = new int[m], to = new int[m], cost = new int[m];
		Integer[] indices = new Integer[m];
		for (int i = 0; i < m; i++) {
			from[i] = sc.nextInt() - 1;
			to[i] = sc.nextInt() - 1;
			cost[i] = sc.nextInt();
			indices[i] = i;
		}
		int[] upTo = new int[m], upFrom = new int[m];

		Arrays.sort(indices, Comparator.comparingInt(i -> cost[i]));
		DSU dsu = new DSU(n);
		for (int i = 0; i < m;) {
			int j = i;
			while (j < m && cost[indices[j]] == cost[indices[i]]) {
				int u = from[indices[j]], v = to[indices[j]];
				int x = dsu.findSet(u), y = dsu.findSet(v);

				upFrom[indices[j]] = x;
				upTo[indices[j]] = y;
				j++;
			}
			while (i < j) {
				dsu.union(from[indices[i]], to[indices[i]]);
				i++;
			}

		}
		dsu = new DSU(n);
		int q = sc.nextInt();
		while (q-- > 0) {
			boolean ok = true;
			int k = sc.nextInt();
			Integer[] queries = new Integer[k];
			for (int i = 0; i < k; i++) {
				queries[i] = sc.nextInt() - 1;

			}
			Arrays.sort(queries, Comparator.comparingInt(i -> cost[i]));
			for (int idx : queries) {
				int u = upTo[idx], v = upFrom[idx];
				if (!dsu.union(u, v)) {
					ok = false;
					break;
				}
			}
			for (int idx : queries) {

				dsu.p[upFrom[idx]] = -1;
				dsu.p[upTo[idx]] = -1;
			}
			out.println(ok ? "YES" : "NO");
		}
		out.close();

	}

	static class DSU {
		int[] p, r;

		public DSU(int n) {
			p = new int[n];
			r = new int[n];
			for (int i = 0; i < n; i++)
				p[i] = -1;
		}

		public boolean isSameSet(int i, int j) {
			return findSet(i) == findSet(j);
		}

		public int findSet(int i) {
			if (p[i] < 0)
				return i;
			else
				return p[i] = findSet(p[i]);
		}

		public boolean union(int i, int j) {
			int x = findSet(i);
			int y = findSet(j);
			if (x == y)
				return false;
			if (r[x] > r[y])
				p[y] = x;
			else {
				p[x] = y;
				if (r[x] == r[y])
					r[y]++;
			}
			return true;
		}
	}

	static class Scanner {
		BufferedReader br;
		StringTokenizer st;

		Scanner() {
			br = new BufferedReader(new InputStreamReader(System.in));
		}

		Scanner(String fileName) throws FileNotFoundException {
			br = new BufferedReader(new FileReader(fileName));
		}

		String next() throws IOException {
			while (st == null || !st.hasMoreTokens())
				st = new StringTokenizer(br.readLine());
			return st.nextToken();
		}

		String nextLine() throws IOException {
			return br.readLine();
		}

		int nextInt() throws IOException {
			return Integer.parseInt(next());
		}

		long nextLong() throws NumberFormatException, IOException {
			return Long.parseLong(next());
		}

		double nextDouble() throws NumberFormatException, IOException {
			return Double.parseDouble(next());
		}
	}
}
