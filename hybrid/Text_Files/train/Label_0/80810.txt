def solve(n, l, r, s):
	sol = list(range(1, r - l + 2))
	sol.reverse()
	somme = sum(sol)
	i = 0
	while somme < s:
		if i >= len(sol):
			print(-1)
			return
		if sol[i] < n and (i == 0 or sol[i] < sol[i - 1] - 1):
			sol[i] += 1
			somme += 1
		else:
			i += 1
	if somme < s or somme > s:
		print(-1)
		return
	reste = [x for x in range(1, n + 1) if x not in sol]
	result = reste[:l - 1] + sol + reste[l - 1:]
	print(*result)
t = int(input())
for i in range(t):
	(n, l, r, s) = list(map(int, input().split()))
	solve(n, l, r, s)
