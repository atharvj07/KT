#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n];
    int temp[105] = {0};
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      temp[a[i]]++;
    }
    sort(a, a + n);
    int c = 0, max = 0;
    int i;
    for (i = 0; i < n; i++) {
      if (c == 2) break;
      if (temp[i] == 2)
        continue;
      else if (temp[i] == 1 && c == 0) {
        max += i;
        c++;
      } else if (temp[i] == 1 && c == 1) {
        continue;
      } else if (temp[i] == 0 && c == 0) {
        max += 2 * i;
        break;
      } else if (temp[i] == 0 && c == 1) {
        max += i;
        break;
      }
    }
    if (i == n)
      cout << max + i << "\n";
    else
      cout << max << "\n";
  }
  return 0;
}
