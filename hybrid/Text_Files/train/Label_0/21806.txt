void solve()
{
  int n,k;
  cin>>n>>k;
  vector<int>v(n);
  set<int>s;
  for(int i=0;i<n;i++)
  {
    cin>>v[i];
    s.insert(v[i]);
  }
  if(k==0)
  {
    cout<<s.size()<<endl;
    return;
  }
  sort(v.begin(),v.end());
  int t=n;
  for(int i=0;i<n;i++)
  {
    if(i!=v[i])
    {
      t=i;
      break;
    }
  }
  if(t==n)
  {
    cout<<n+k<<endl;
    return;
  }

  int x=ceil((t+v[n-1])/(2*1.0));
  //cout<<x<<endl;
  s.insert(x);
  cout<<s.size()<<endl;
}
