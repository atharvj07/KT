 */
public class B318 {
    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String s = in.readLine();
        long res = 0;
        long heavy = 0;
        if (s.length() > 4) {
            for (int i = 4 ; i < s.length() ; i++) {
                if (s.charAt(i - 4) == 'h') {
                    if (s.charAt(i - 3) == 'e') {
                        if (s.charAt(i - 2) == 'a') {
                            if (s.charAt(i - 1) == 'v') {
                                if (s.charAt(i) == 'y') {
                                    heavy++;
                                }
                            }
                        }
                    }
                } else if (s.charAt(i - 4) == 'm') {
                    if (s.charAt(i - 3) == 'e') {
                        if (s.charAt(i - 2) == 't') {
                            if (s.charAt(i - 1) == 'a') {
                                if (s.charAt(i) == 'l') {
                                    res += heavy;
                                }
                            }
                        }
                    }
                }
            }
        }
        System.out.println(res);
    }
}
