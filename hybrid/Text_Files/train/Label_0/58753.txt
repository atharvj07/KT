import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;
import java.util.StringTokenizer;

public class ThreeSwaps2{
	
	public static void main(String[] args)
	{
		new ThreeSwaps2().run();
	}
	
	public void run()
	{
		Scanner file = new Scanner(System.in);
		int N = file.nextInt();
		int[] ints = new int[N];
		for(int i = 0;i<N;i++)
			ints[i] = file.nextInt();
		ArrayList<Integer> list = new ArrayList<Integer>();
		int start = ints[0];
		for(int i = 1;i<ints.length;i++)
		{
			if(Math.abs(ints[i-1] - ints[i]) != 1)
			{
				list.add(start);
				if(start!=ints[i-1])
					list.add(ints[i-1]);
				start = ints[i];
			}
		}
		list.add(start);
		if(start != ints[ints.length - 1])
			list.add(ints[ints.length - 1]);
		int[] comp = new int[list.size()];
		for(int i = 0;i<comp.length;i++)
			comp[i] = list.get(i);
		int[] clone = comp.clone();
		ArrayList<String> answers = new ArrayList<>();
		bruteForce(comp, 0,"", answers);
		for(String s:answers)
		{
			if(test(comp.clone(), ints.clone(), s))
				return;
		}
	}
	
	public boolean test(int[] compClone, int[] intsClone, String ans)
	{
		Scanner sc = new Scanner(ans);
		String out = "";
		int x = 0;
		while(sc.hasNext())
		{
			int a = sc.nextInt();
			int b = sc.nextInt();
			int finda = find(intsClone, compClone[a]);
			int findb = find(intsClone, compClone[b]);
			flip(compClone,a,b);
			flip(intsClone,Math.min(finda, findb), Math.max(finda, findb));
			x++;
			out = (Math.min(finda, findb)+1)+" "+(Math.max(finda, findb)+1)+ "\n"+out;
		}
		if(done(intsClone))
		{
			System.out.println(x);
			System.out.println(out);
			return true;
		}
		return false;
	}

	public int find(int[] ints, int x)
	{
		for(int i = 0;i<ints.length;i++)
			if(ints[i] == x)
				return i;
		return -1;
	}
	
	public void flip(int[] ints, int start, int end)
	{
		for(int i = 0;i<=(end-start)/2;i++)
		{
			int save = ints[start + i];
			ints[start + i] = ints[end - i];
			ints[end - i] = save;
		}
	}
	
	public void bruteForce(int[] ints, int turn,String current, ArrayList<String> answers)
	{
		if(done(ints))
			answers.add(current);
		if(turn == 3)
			return;	
		for(int i = 0;i<ints.length;i++)
			for(int j = i+1;j<ints.length;j++)
			{
				flip(ints,i,j);
				bruteForce(ints, turn + 1,current+" "+i+" "+j, answers);
				flip(ints,i,j);
			}
	}
	
	public boolean done(int[] ints)
	{
		for(int i = 0;i<ints.length-1;i++)
			if(ints[i] > ints[i+1])
				return false;
		return true;
	}
}
