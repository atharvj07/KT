a = [''] * 10
for i in range(10):
	a[i] = input()
AliceWins = False
for i in range(10 - 4):
	for j in range(10):
		count = 0
		dot = 0
		for k in range(5):
			if a[i + k][j] == 'X':
				count += 1
			elif a[i + k][j] == '.':
				dot += 1
		if count == 4 and dot:
			AliceWins = True
