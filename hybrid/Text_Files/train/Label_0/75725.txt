(n, m) = map(int, input().split())
N = 10 ** 5 + 1
sbj_count = [0] * N
sbj_skl = [0] * N
sbj_ans = [0] * N
store = [[] for i in range(N)]
for i in range(n):
	(l, r) = map(int, input().split())
	store[l].append(r)
for i in range(N):
	store[i].sort(reverse=True)
for i in range(1, N):
	for j in range(len(store[i])):
		sbj_skl[i] += store[i][j]
		if sbj_skl[i] > 0:
			sbj_ans[j + 1] += sbj_skl[i]
print(max(sbj_ans))
