#include <bits/stdc++.h>
using namespace std;
void printArray(int _a[], int _n) {
  if (_n == 0) {
    puts("");
    return;
  }
  printf("%d", _a[0]);
  for (int i = 1; i < _n; ++i) {
    printf(" %d", _a[i]);
  }
  puts("");
}
char s[5];
void f(vector<int> &v, vector<int> &r, int a, int b, int c, int d) {
  r[0] = v[a];
  r[1] = v[b];
  r[2] = v[c];
  r[3] = v[d];
}
int main() {
  set<vector<int> > S;
  gets(s);
  int n;
  sscanf(s, "%d", &n);
  for (int i = 0; i < n; ++i) {
    if (i) gets(s);
    vector<int> v;
    gets(s);
    v.push_back(s[0] - '0');
    v.push_back(s[1] - '0');
    gets(s);
    v.push_back(s[0] - '0');
    v.push_back(s[1] - '0');
    vector<int> mini = v;
    vector<int> r(4);
    f(v, r, 2, 0, 3, 1);
    if (r < mini) mini = r;
    f(v, r, 3, 2, 1, 0);
    if (r < mini) mini = r;
    f(v, r, 1, 3, 0, 2);
    if (r < mini) mini = r;
    S.insert(mini);
  }
  printf("%d\n", S.size());
}
