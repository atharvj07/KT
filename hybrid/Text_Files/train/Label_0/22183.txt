#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m;
  cin >> n >> m;
  vector<string> mas(n);
  for (int i = 0; i < n; ++i) {
    cin >> mas[i];
  }
  for (int i = 0; i < m; ++i) {
    string last = "";
    for (int j = 0; j < n; ++j) {
      if (mas[j][i] == '#') {
        if (last == "") {
          last = mas[j];
        } else {
          if (last != mas[j]) {
            cout << "No" << endl;
            return 0;
          }
        }
      }
    }
  }
  cout << "Yes" << endl;
  return 0;
}
