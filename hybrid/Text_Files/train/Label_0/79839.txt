(n, m) = map(int, input().split())
q = list(map(int, input().split()))
k = list(map(int, input().split()))
if m == 1:
	print(n)
else:
	x = []
	for i in range(n + m):
		if k[i] == 1:
			x.append(i)
	l = [0] * m
	i = 0
	while i < m:
		if i == 0:
			l[i] += x[i]
		if i == m - 1:
			l[i] += m + n - 1 - x[i]
			i += 1
			continue
		a = q[x[i]]
		b = q[x[i + 1]]
		x1 = x[i] + 1
		y1 = x[i + 1]
		for j in range(x1, y1):
			d = q[j]
			if d - a <= b - d:
				l[i] += 1
			else:
				l[i + 1] += 1
		i += 1
	for i in l:
		print(i, end=' ')
