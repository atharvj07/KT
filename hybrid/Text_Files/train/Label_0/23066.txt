#include <bits/stdc++.h>
using namespace std;
int p, i, number[4000];
char s[4000], symbol[4000];
int POW(int a, int b) {
  int j, t = 1;
  for (j = 1; j <= b; j++) t *= a;
  return t;
}
bool can() {
  if ((s[i] == '+' || s[i] == '-') && symbol[p] != '(') return 1;
  if ((s[i] == '*' || s[i] == '/') &&
      (symbol[p] == '*' || symbol[p] == '/' || symbol[p] == '^'))
    return 1;
  if (s[i] == '^' && symbol[p] == '^') return 1;
  return 0;
}
void POP() {
  switch (symbol[p--]) {
    case '+':
      number[p] += number[p + 1];
      break;
    case '-':
      number[p] -= number[p + 1];
      break;
    case '*':
      number[p] *= number[p + 1];
      break;
    case '/':
      number[p] /= number[p + 1];
      break;
    case '^':
      number[p] = POW(number[p], number[p + 1]);
      break;
  }
}
int main() {
  int t;
  t = 1;
  while (t--) {
    int x, len;
    p = 1;
    i = 0;
    gets(s);
    len = strlen(s);
    s[len] = ')';
    symbol[p] = '(';
    while (i < len) {
      while (s[i] == '(') {
        symbol[++p] = s[i];
        i++;
      }
      x = 0;
      while (s[i] >= '0' && s[i] <= '9') {
        x = x * 10 + s[i++] - '0';
      }
      number[p] = x;
      do {
        if (s[i] == ')') {
          while (symbol[p] != '(') POP();
          number[--p] = number[p + 1];
        } else {
          while (can()) POP();
          symbol[++p] = s[i];
        }
        i++;
      } while (i <= len && s[i - 1] == ')');
    }
    if (number[0] == 0) {
      for (int i = 1; i <= 48 + number[0]; i++) {
        printf("+");
      }
      puts(".");
      return 0;
    }
    if (number[0] < 10) {
      for (int i = 1; i <= 48 + number[0]; i++) {
        printf("+");
      }
      puts(".");
      return 0;
    }
    if (number[0] < 100) {
      for (int i = 1; i <= 48 + (number[0] / 10); i++) {
        printf("+");
      }
      puts(".");
      puts(">");
      for (int i = 1; i <= 48 + (number[0] % 10); i++) {
        printf("+");
      }
      puts(".");
      return 0;
    }
    for (int i = 1; i <= 48 + (number[0] / 100); i++) {
      printf("+");
    }
    puts(".");
    puts(">");
    for (int i = 1; i <= 48 + (number[0] / 10) % 10; i++) {
      printf("+");
    }
    puts(".");
    puts(">");
    for (int i = 1; i <= 48 + (number[0] % 10); i++) {
      printf("+");
    }
    puts(".");
  }
  return 0;
}
