vector<vl> mul(vector<vl> a, vector<vl> b){
  int n = a.size();
  vector<vl> ret(n,vl(n,0));
  REP(k,n)REP(i,n)REP(j,n){
    ret[i][j] += a[i][k]*b[k][j]%mod;
    ret[i][j] %= mod;
    // ret[i][j] = get_mod(ret[i][j] + get_mod(a[i][k])*get_mod(b[k][j]));
  }
  return ret;
}
