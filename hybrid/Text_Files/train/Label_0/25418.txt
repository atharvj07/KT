#include <bits/stdc++.h>
using namespace std;
const int M = 100 + 10;
const int M1 = 200000;
long long int dp[101][M1 + 1];
int a[M];
long long int DFS(int pos, long long int num, int k) {
  if (pos == k - 1) {
    long long int ans =
        num - num / a[pos] - num / a[pos + 1] + num / a[pos] / a[pos + 1];
    return ans;
  } else {
    if (num <= M1) {
      return dp[pos][num];
    } else {
      long long int ans = DFS(pos + 1, num, k) - DFS(pos + 1, num / a[pos], k);
      return ans;
    }
  }
}
bool cmp(int a, int b) { return a > b; }
int main() {
  long long int n;
  int k;
  cin >> n >> k;
  for (int i = 1; i <= k; i++) {
    cin >> a[i];
  }
  sort(a + 1, a + k + 1, cmp);
  for (long long int i = 1; i <= M1; i++) {
    dp[k][i] = i - i / a[k];
  }
  for (int i = k - 1; i >= 1; i--) {
    for (int j = 1; j <= M1; j++) {
      dp[i][j] = dp[i + 1][j] - dp[i + 1][j / a[i]];
    }
  }
  long long int ans = DFS(1, n, k);
  cout << ans << endl;
  return 0;
}
