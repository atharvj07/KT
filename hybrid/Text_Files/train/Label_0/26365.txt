#include<iostream>
#include<vector>
using namespace std;

int main(){
    int n;
    const int N = 1e6;

    vector<bool> is_prime(N,true);
    is_prime[0] = false;
    is_prime[1] = false;

    for(int i=2;i*i<N;i++){
        if(is_prime[i]){
            for(int j=i*2;j<N;j+=i){
                is_prime[j] = false;
            }
        }
    }

    while(cin >> n){
        int small = n-1;
        int big = n + 1;
        while(!is_prime[small]) small--;
        while(!is_prime[big]) big++;

        cout << small << " " << big << endl;
    }


}
