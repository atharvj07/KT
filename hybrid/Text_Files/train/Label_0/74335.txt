(a, b, p, x) = map(int, input().split())
x += 1
res = 0
res += x // (p * (p - 1)) * (p - 1)
x -= x // (p * (p - 1)) * (p * (p - 1))
for i in range(p - 1):
	ap = pow(a, i * (p - 2) % (p - 1), p) * b % p
	l = x // (p - 1) + (1 if i < x % (p - 1) else 0)
	if i >= ap and i - ap < l:
		res += 1
	elif i < ap and i + p - ap < l:
		res += 1
print(res)
