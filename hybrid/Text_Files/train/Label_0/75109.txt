n = int(input())
a = map(int, input().split())
x = []
for i in a:
	r = int(i ** 0.5)
	x.append((min(i - r * r, (r + 1) ** 2 - i), i))
x.sort()
if x[n // 2][0] != 0:
	ans = 0
	for i in range(0, n // 2):
		ans += x[i][0]
	print(ans)
else:
	(ans, i) = (0, n // 2)
	while i < n and x[i][0] == 0:
		if x[i][1] == 0:
			ans += 2
		else:
			ans += 1
		i += 1
		i
	print(ans)
