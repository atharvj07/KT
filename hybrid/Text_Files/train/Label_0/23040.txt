#include<iostream>
#include<algorithm>
#include<vector>
#include<string>
using namespace std;
struct box{
	int win,lose;
	string s;
};
int cmp(const box &a,const box &b){
	if(a.win==b.win){
		return a.lose<b.lose;
	}else{
		return a.win>b.win;
	}
}
int main(){
	int n,w,l,p;
	string t;
	vector<box> vec;
	box pre;
	while(cin>>n,n){
		for(int i=0;i<n;i++){
			cin>>t;
			w=0;
			l=0;
			for(int j=0;j<n-1;j++){
				cin>>p;
				if(p==0)
				w++;
				else if(p==1)
				l++;
			}
			pre.win=w;
		pre.lose=l;
		pre.s=t;
		vec.push_back(pre);
		}
		stable_sort(vec.begin(),vec.end(),cmp);
		for(int i=0;i<n;i++)
		cout<<vec[i].s<<endl;
		vec.clear();
	}
	return 0;
}
