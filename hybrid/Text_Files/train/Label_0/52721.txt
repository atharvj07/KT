public class liegesOfLegendre {

    public static Integer preDetGrundy[] = {0,1,0,1,2};
    public static Integer nPilesOfCows, kPiles, result = 0;


    public static void main(String[] args){

        Scanner sc = new Scanner(System.in);
        String line;

        line = sc.nextLine();
        String[] tmp = line.split(" ");
        nPilesOfCows = Integer.parseInt(tmp[0]);
        kPiles = Integer.parseInt(tmp[1]);

        int[] initialState = new int[nPilesOfCows];

        for(int i = 0; i < nPilesOfCows; i++){
            initialState[i] = sc.nextInt();
            result ^= grundyNumber(initialState[i]);
        }

        System.out.println( result > 0 ? "Kevin":"Nicky");

    }

    public static int grundyNumber(int value){
        if(kPiles % 2 == 0){
            if(value == 1) return 1;
            if(value == 2) return 2;
            return (value % 2) ^ 1;
        }else{
            if(value < 5) return preDetGrundy[value];
            if(value % 2 == 1) return 0;
            return (grundyNumber(value/2)) == 1 ? 2 : 1;
        }
    }
}
