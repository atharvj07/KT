from collections import Counter
from fractions import Fraction
mod = 998244353

def calc_ans(x, y):
	return x * pow(y, mod - 2, mod) % mod
n = int(input())
item_list = [list(map(int, input().split()))[1:] for i in range(n)]
counter = Counter([i for items in item_list for i in items])
ans = Fraction(0)
numer = 0
denom = 1
for kid in range(n):
	l = len(item_list[kid])
	s = sum((counter[i] for i in item_list[kid]))
	numer = (numer * l + denom * s) % mod
	denom = denom * l % mod
denom = denom * n * n % mod
print(calc_ans(numer, denom))
