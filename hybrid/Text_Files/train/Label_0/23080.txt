typedef unsigned long long ull;

struct RollingHash
{
	vector<ull> hash, pow;
	RollingHash(string s, ull _b = 1000000007)
	{
		_b = _b;
		int sz = s.length();
		hash.assign(sz + 1, 0);
		pow.assign(sz + 1, 0);

		pow[0] = 1;
		for (int i = 0; i<sz; i++) {
			pow[i + 1] = pow[i] * _b;
		}
		for (int i = 0; i<sz; i++) {
			hash[i + 1] = (hash[i] + s[i])*_b;
		}
	}
	// s[l]s[l+1]....s[r-1]のハッシュ値を取得する
	// 半開区間
	ull get(int l, int r) {
		return (hash[r] - hash[l] * pow[r - l]);
	}
};
signed main() {
	my_io();
	vector<int>ans;
	string t, s;
	cin >> t >> s;
	RollingHash t_Roll(t);
	RollingHash s_Roll(s);
	int s_hash=s_Roll.get(0, (int)s.size());
	for (int i = 0; i <=(int)t.size()-(int)s.size(); i++) {
		if (s_hash == t_Roll.get(i, i + s.size())) {
			ans.push_back(i);
		}
	}
	for (int i = 0; i < (int)ans.size(); i++) {
		cout << ans[i] << endl;
	}
	return 0;
}
