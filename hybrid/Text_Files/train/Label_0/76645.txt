t = int(input())
for _ in range(t):
	n = int(input())
	A = list(map(int, input().split(' ')))
	x = A[0]
	y = -1
	for i in range(1, n):
		if A[i] != A[0]:
			y = i
	if y == -1:
		print('NO')
	else:
		print('YES')
		for i in range(1, n):
			if A[i] == x:
				print(i + 1, y + 1)
			else:
				print(i + 1, 1)
