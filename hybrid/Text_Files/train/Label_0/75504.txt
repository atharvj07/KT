(a, b, c) = map(int, input().split())
Q = [a // 3, b // 2, c // 2]
q = min(Q)
R = [a - 3 * q, b - 2 * q, c - 2 * q]
s = [0, 1, 2, 0, 2, 1, 0]
s *= 2
import copy
ans = -1
for i in range(7):
	temp = [a - 3 * q, b - 2 * q, c - 2 * q]
	for j in range(i, i + 7):
		if temp[s[j]] == 0:
			break
		else:
			temp[s[j]] -= 1
	ans = max(7 * q + j - i, ans)
print(ans)
