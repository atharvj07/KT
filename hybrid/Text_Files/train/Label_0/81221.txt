def main():
	n = int(input())
	s = input().split(' ')
	r1 = 0
	r2 = 0
	t = 0
	while len(s) != 0:
		temp = max((int(s[0]), int(s[len(s) - 1])))
		i = 0
		if temp == int(s[0]):
			i = 0
		else:
			i = len(s) - 1
		if t == 0:
			r1 += temp
			t = 1
		else:
			r2 += temp
			t = 0
		s.pop(i)
	print(r1, r2)
main()
