class TaskF {
    public void solve(int testNumber, InputReader in, PrintWriter out) {
        int n = in.nextInt();
        int[] x = Utils.readIntArray(in, n);
        int[] y = Utils.readIntArray(in, n);

        List<Operation> operX = getOperations(x);
        List<Operation> operY = getOperations(y);
        Collections.reverse(operY);

        List<Operation> middle = doMagic(x, y);
        if (middle == null) {
            out.println(-1);
            return;
        }
        operX.addAll(middle);
        operX.addAll(operY);
        out.println(operX.size());
        for (Operation o : operX) {
            out.println((o.what + 1) + " " + (o.withWhat + 1));
        }
    }

    private List<Operation> doMagic(int[] x, int[] y) {
        int ptrX = 0;
        int ptrY = 0;
        List<Operation> operations = new ArrayList<Operation>();

        while (ptrY < y.length && y[ptrY] != 0) {
            int smallestBit = Integer.numberOfTrailingZeros(y[ptrY]);
            while (ptrX < x.length && Integer.numberOfTrailingZeros(x[ptrX]) != smallestBit) {
                ++ptrX;
            }
            if (ptrX == x.length) return null;
            if (ptrX > ptrY) {
                operations.add(new Operation(ptrX, ptrY));
                operations.add(new Operation(ptrY, ptrX));
                operations.add(new Operation(ptrX, ptrY));

                int temp = x[ptrX];
                x[ptrX] = x[ptrY];
                x[ptrY] = temp;
            }

            for (int i = ptrX + 1; i < x.length && x[i] != 0; ++i) {
                if (BitUtils.checkBit(y[ptrY], Integer.numberOfTrailingZeros(x[i]))) {
                    operations.add(new Operation(ptrY, i));
                    x[ptrY] ^= x[i];
                }
            }
            if (x[ptrY] != y[ptrY]) return null;
            ++ptrY;
        }

        for (int i = 0; i < x.length; ++i) {
            if (x[i] != 0 && y[i] == 0) {
                operations.add(new Operation(i, i));
                x[i] = 0;
            }
        }

        if (!Arrays.equals(x, y)) {
            return null;
            //throw new AssertionError("!");
        }
        return operations;
    }

    private List<Operation> getOperations(int[] x) {
        List<Operation> operations = new ArrayList<Operation>();

        int lines = 0;
        int n = x.length;
        for (int bit = 0; bit < 31; ++bit) {
            int j = lines;
            while (j < n && !BitUtils.checkBit(x[j], bit)) ++j;

            if (j == n) {
                continue;
            }

            if (lines != j) {
                operations.add(new Operation(lines, j));
                operations.add(new Operation(j, lines));
                operations.add(new Operation(lines, j));

                int temp = x[lines];
                x[lines] = x[j];
                x[j] = temp;
            }
            for (int k = 0; k < n; ++k) {
                if (k == lines) continue;
                if (BitUtils.checkBit(x[k], bit)) {
                    operations.add(new Operation(k, lines));
                    x[k] ^= x[lines];
                }
            }
            ++lines;
        }
        return operations;
    }

    class Operation {
        int what;
        int withWhat;

        Operation(int what, int withWhat) {
            this.what = what;
            this.withWhat = withWhat;
        }
    }
}

class InputReader {
    private BufferedReader reader;
    private StringTokenizer stt;

    public InputReader(InputStream stream) {
        reader = new BufferedReader(new InputStreamReader(stream));
    }

    public String nextLine() {
        try {
            return reader.readLine().trim();
        } catch (IOException e) {
            return null;
        }
    }

    public String nextString() {
        while (stt == null || !stt.hasMoreTokens()) {
            stt = new StringTokenizer(nextLine());
        }
        return stt.nextToken();
    }

    public int nextInt() {
        return Integer.parseInt(nextString());
    }

}

class Utils {
    public static int[] readIntArray(InputReader in, int n) {
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = in.nextInt();
        }
        return a;
    }

}

class BitUtils {
    public static boolean checkBit(int mask, int bit) {
        return (mask & (1 << bit)) > 0;
    }

}
