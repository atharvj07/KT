from collections import defaultdict
t = int(input())
while t != 0:
	d = defaultdict(int)

	list1.sort()
	temp = 0
	for i in range(len(list1)):
		temp += list1[i][0]
	ans = []
	ans.append(temp)
	anothre = []
	i = 0
	j = len(list1) - 1
	while i < n and j >= 0:
		ver = list1[j][1]
		val = list1[j][0]
		while d[ver] > 1:
			d[ver] -= 1
			ans.append(ans[-1] + val)
			i += 1
