for _ in range(int(input())):
	(n, k) = map(int, input().split())
	arr = list(map(int, input().split()))
	index = {}
	for i in range(n):
		if arr[i] in index:
			index[arr[i]].append(i)
		else:
			index[arr[i]] = [i]
	brr = list(index.values())
	(group, temp) = ([], [])
	for i in range(len(brr)):
		if len(brr[i]) < k:
			temp += brr[i]
			if len(temp) >= k:
				group.append(temp[:k])
				temp = temp[k:]
		else:
			group.append(brr[i])
	ans = [0] * n
	for num in group:
		for i in range(k):
			ans[num[i]] = i + 1
	for num in ans:
		print(num, end=' ')
	print()
