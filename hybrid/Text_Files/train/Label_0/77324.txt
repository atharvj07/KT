n = int(input())
lx = []
lh = []
for i in range(n):
	(x, h) = [int(x) for x in input().split()]
	lx.append(x)
	lh.append(h)
if n == 1:
	print('1')
else:
	num = 2
	for i in range(1, n - 1):
		if lh[i] < lx[i] - lx[i - 1]:
			num += 1
		elif lh[i] < lx[i + 1] - lx[i]:
			num += 1
			lx[i] += lh[i]
	print(num)
