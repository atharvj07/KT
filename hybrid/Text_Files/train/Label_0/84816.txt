import sys
input = sys.stdin.readline

def read_number():
	return int(input())

def read_list():
	return list(map(int, input().split()))

def read_string():
	s = input()
	return list(s[:len(s) - 1])

def read_separated_numbers():
	return map(int, input().split())
first_row = read_list()
(n, q) = (first_row[0], first_row[1])
values = read_list()
flag = False
total = sum(values)
for i in range(q):
	input_line = read_list()
	t = input_line[0]
	if t == 1:
		x = input_line[1]
		values[x - 1] = 1 - values[x - 1]
		if values[x - 1] == 1:
			total += 1
		else:
			total -= 1
	else:
		k = input_line[1]
		if k <= total:
			print(str(1))
		else:
			print(str(0))
