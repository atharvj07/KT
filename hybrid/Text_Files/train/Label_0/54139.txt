import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.List;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Collections;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        FastScanner in = new FastScanner(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskD solver = new TaskD();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskD {
        List<Integer>[] children;
        int[] depth;
        int[] bamboo;
        int[] bambooNext;
        int[] freeVertex;
        List<Integer> ops = new ArrayList<>();

        public void solve(int testNumber, FastScanner in, PrintWriter out) {
            int n = in.nextInt();
            int[] p = new int[n];
            children = new List[n];
            for (int i = 0; i < n; i++) {
                children[i] = new ArrayList<>();
            }
            p[0] = -1;
            for (int i = 1; i < n; i++) {
                p[i] = in.nextInt();
                children[p[i]].add(i);
            }

            depth = new int[n];
            for (int i = 1; i < n; i++) {
                depth[i] = 1 + depth[p[i]];
            }

            bambooNext = new int[n];
            freeVertex = new int[n];
            build(0);
            {
                int j = 0;
                bamboo = new int[n];
                for (int i = 0; i < n; i++) {
                    bamboo[i] = j;
                    j = bambooNext[j];
                }
            }

            for (int i = 0; i < bamboo.length; i++) {
                if (i > 0) {
                    out.print(" ");
                }
                out.print(bamboo[i]);
            }
            out.println();


            out.println(ops.size());
            for (int i = 0; i < ops.size(); i++) {
                if (i > 0) {
                    out.print(" ");
                }
                out.print(ops.get(i));
            }
            out.println();
        }

        private void build(int root) {
            for (int c : children[root]) {
                build(c);
            }
            Collections.sort(children[root], (u, v) -> (depth[freeVertex[u]] - depth[freeVertex[v]]));
            int prev = root;
            for (int c : children[root]) {
                bambooNext[prev] = c;
                int times = depth[prev] - depth[root];
                for (int i = 0; i < times; i++) {
                    ops.add(c);
                }
                freeVertex[root] = freeVertex[c];
                prev = freeVertex[c];
            }
            freeVertex[root] = prev;
        }

    }

    static class FastScanner {
        private BufferedReader in;
        private StringTokenizer st;

        public FastScanner(InputStream stream) {
            in = new BufferedReader(new InputStreamReader(stream));
        }

        public String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return st.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
