for _ in range(int(input())):
	(a, b, c) = input().split()
	a = int(a)
	b = int(b)
	c = int(c)
	count = 0
	if a > 0:
		a -= 1
		count += 1
	if b > 0:
		b -= 1
		count += 1
	if c > 0:
		c -= 1
		count += 1
	l = [a, b, c]
	l.sort()
	if l[2] > 0 and l[1] > 0:
		l[2] -= 1
		l[1] -= 1
		count += 1
	if l[2] > 0 and l[0] > 0:
		l[2] -= 1
		l[0] -= 1
		count += 1
	if l[1] > 0 and l[0] > 0:
		l[1] -= 1
		l[0] -= 1
		count += 1
	if l[0] > 0 and l[1] > 0 and (l[2] > 0):
		l[2] -= 1
		l[1] -= 2
		l[0] -= 1
		count += 1
	print(count)
