#include <bits/stdc++.h>
using namespace std;
const int M = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k, x;
  cin >> n >> k >> x;
  vector<long long> A(n);
  long long mn = 2e9;
  long long mx = -1;
  int ncnt = 0;
  auto comp = [&](const pair<long long, long long>& a,
                  const pair<long long, long long>& b) {
    return llabs(a.first) > llabs(b.first);
  };
  priority_queue<pair<long long, long long>, vector<pair<long long, long long>>,
                 decltype(comp)>
      pq(comp);
  for (int i = 0; i < n; ++i) {
    cin >> A[i];
    mn = min(mn, A[i]);
    mx = max(mx, A[i]);
    if (A[i] < 0) {
      ++ncnt;
    }
    pq.push({A[i], i});
  }
  auto add1 = [&](long long v) {
    if (v < 0) {
      return v + x;
    } else {
      return v - x;
    }
  };
  auto add2 = [&](long long v) {
    if (v < 0) {
      return v - x;
    } else {
      return v + x;
    }
  };
  auto getsign = [&](long long v) {
    if (v < 0) {
      return -1;
    } else {
      return 1;
    }
  };
  for (int i = 0; i < k; ++i) {
    if ((ncnt % 2) == 0) {
      auto top = pq.top();
      pq.pop();
      int sign = getsign(top.first);
      top.first = add1(top.first);
      int nsign = getsign(top.first);
      if (sign != nsign) {
        ++ncnt;
      }
      pq.push(top);
    } else {
      auto top = pq.top();
      pq.pop();
      top.first = add2(top.first);
      pq.push(top);
    }
  }
  while (!pq.empty()) {
    auto top = pq.top();
    pq.pop();
    A[top.second] = top.first;
  }
  for (long long i : A) {
    cout << i << " ";
  }
  cout << "\n";
  return 0;
}
