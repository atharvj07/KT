import math
for i1 in range(int(input())):
	(n, k) = [int(x) for x in input().split()]
	lis = sorted([int(x) for x in input().split()])
	value = lis[-k]
	index = n - k
	last_index = n - lis[::-1].index(value) - 1
	first_index = lis.index(value)
	print(math.comb(last_index - first_index + 1, last_index - index + 1) % (10 ** 9 + 7))
