#include <bits/stdc++.h>
using namespace std;
const int N = 1e3 + 1, M = 1e9 + 7;
mt19937 rng(std::chrono::system_clock::now().time_since_epoch().count());
int n, dp[N];
int main() {
  scanf("%d", &n);
  for (int i = 0, a; i < n; i++) {
    scanf("%d", &a);
    dp[i] = (2 * dp[i - 1] + 2) % M;
    if (a > 1) dp[i] = (dp[i] - dp[a - 2] + M) % M;
  }
  cout << dp[n - 1] << endl;
}
