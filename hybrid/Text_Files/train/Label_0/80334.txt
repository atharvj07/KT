n = int(input())
a = list(map(int, input().split()))
s = list(input())
l = -1000000000
r = 1000000000
for i in range(4, len(s)):
	if s[i] == '1' and s[i - 1] == '0':
		l = max(l, max(a[i - 1], a[i - 2], a[i - 3], a[i - 4], a[i]) + 1)
	elif s[i] == '0' and s[i - 1] == '1':
		r = min(r, min(a[i - 1], a[i - 2], a[i - 3], a[i - 4], a[i]) - 1)
print(l, r)
