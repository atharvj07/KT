int GetDays(int y, int m, int d)
{
if (m <= 2)y-- , m += 12;
int dy = 365 * (y - 1);
int c = y / 100;
int dl = (y >> 2) - c + (c >> 2);
int dm = (m * 979 - 1033) >> 5;
return dy + dl + dm + d - 1;
}

int main(){
	int y1,y2,m1,m2,d1,d2;
	while(cin>>y1>>m1>>d1>>y2>>m2>>d2)
	{
		if(y1==-1)break;
		cout<< GetDays(y2,m2,d2) -  GetDays(y1,m1,d1)<<endl;
	}
	return 0;
}
