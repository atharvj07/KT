from bisect import bisect_left
(n, k) = map(int, input().split())
s = input()
l = len(s)
flag = -1
pos0 = [-1]
pos1 = [-1]
for i in range(l):
	if s[i] == '0':
		pos0.append(i)
	else:
		pos1.append(i)
pos0.append(10 ** 18)
pos1.append(10 ** 18)
d0 = 1 + (pos0[-2] - pos0[1])
d1 = 1 + (pos1[-2] - pos1[1])
if min(d0, d1) <= k:
	flag = 1
else:
	for i in range(n - k + 1):
		l = i
		r = i + k - 1
		tposl = bisect_left(pos1, l)
		tposr = bisect_left(pos1, r)
		if pos1[tposl - 1] == -1:
			tl = -1
		else:
			tl = pos1[tposl - 1]
		if pos1[tposr] == r:
			tr = pos1[tposr + 1]
			if tr == 10 ** 18:
				tr = -1
		else:
			tr = pos1[tposr]
			if tr == 10 ** 18:
				tr = -1
		td1 = 0
		if tl == -1:
			td1 = 1 + (pos1[-2] - tr)
		elif tr == -1:
			td1 = 1 + (tl - pos1[1])
		else:
			td1 = 1 + (pos1[-2] - pos1[1])
		tposl = bisect_left(pos0, l)
		tposr = bisect_left(pos0, r)
		if pos0[tposl - 1] == -1:
			tl = -1
		else:
			tl = pos0[tposl - 1]
		if pos0[tposr] == r:
			tr = pos0[tposr + 1]
			if tr == 10 ** 18:
				tr = -1
		else:
			tr = pos0[tposr]
			if tr == 10 ** 18:
				tr = -1
		if tl == -1:
			td2 = 1 + (pos0[-2] - tr)
		elif tr == -1:
			td2 = 1 + (tl - pos0[1])
		else:
			td2 = 1 + (pos0[-2] - pos0[1])
		td = max(td1, td2)
		if td > k:
			break
	else:
		flag = 0
if flag == 1:
	print('tokitsukaze')
elif flag == 0:
	print('quailty')
elif flag == -1:
	print('once again')
