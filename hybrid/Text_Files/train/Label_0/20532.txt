using namespace std;
int main() {
  int n1, num, count;
  cin >> n1;
  int n = n1 * 2;
  int inum = sqrt(n);
  bool is = false;
  for (int i = 1; i <= inum; i++) {
    num = n - (i * i + i);
    int l = 0, r = inum, m;
    if (i == 30) {
      int b;
      int c;
      int s;
    }
    while (l <= r) {
      m = (l + r) / 2;
      if (m * m + m > num) {
        r = m - 1;
      } else if (m * m + m < num) {
        l = m + 1;
      } else {
        is = true;
        count = m;
        goto loop;
      }
    }
  }
loop:
  if (is && count != 0) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
