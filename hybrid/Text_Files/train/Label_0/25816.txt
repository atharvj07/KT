#include <bits/stdc++.h>
using namespace std;
int s[100005];
bool good[100005];
bool visited[100006];
vector<int> list1[100005], list2[100005];
int main() {
  int n, m, a, b;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; i++) {
    scanf("%d", &s[i]);
  }
  for (int i = 0; i < m; i++) {
    scanf("%d%d", &a, &b);
    list1[a - 1].push_back(b - 1);
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == 1) {
      queue<int> q;
      q.push(i);
      visited[i] = true;
      while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int j = 0; j < list1[v].size(); j++) {
          int h = list1[v][j];
          if ((!visited[h]) && (s[h] != 1)) {
            list2[h].push_back(v);
            q.push(h);
            visited[h] = true;
          } else {
            if ((visited[h]) && (s[h] != 1)) {
              list2[h].push_back(v);
            }
          }
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == 2) {
      queue<int> q;
      q.push(i);
      while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int j = 0; j < list2[v].size(); j++) {
          good[v] = true;
          int h = list2[v][j];
          if (!good[h]) {
            if (s[h] == 1) {
              good[h] = 1;
            } else {
              good[h] = 1;
              q.push(h);
            }
          }
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (good[i]) {
      printf("1\n");
    } else {
      printf("0\n");
    }
  }
  return 0;
}
