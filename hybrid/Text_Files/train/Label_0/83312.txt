length = int(input())
sequence = list(map(int, input().split()))
if length <= 3:
	print('no')
else:
	for i in range(length - 1):
		(w, x) = sorted([sequence[i], sequence[i + 1]])
		for j in range(length - 1):
			(y, z) = sorted([sequence[j], sequence[j + 1]])
			if w < y < x < z or y < w < z < x:
				print('yes')
				exit()
	print('no')
