from sys import stdin, stdout
size = 10 ** 5
n = int(stdin.readline())
value = list(map(int, stdin.readline().split()))
counting = [0 for i in range(size + 1)]
ans = 1
for i in range(n):
	counting[value[i]] += 1
value = sorted(list(set(value)))
for i in range(2, size + 1):
	cnt = 0
	j = 1
	while i * j <= size:
		cnt += counting[i * j]
		j += 1
	ans = max(ans, cnt)
stdout.write(str(ans))
