import java.util.*;
import java.math.*;
import java.io.*;

public class CF1042E {
	class Pos implements Comparable <Pos> {
		int row, col; long val;
		public Pos(int rr, int cc, long vv) { row = rr;  col = cc;  val = vv; }
		public boolean equals(Pos p) { return row == p.row && col == p.col; }
		public String toString() { return String.format("matrix[%d][%d] = %d", row, col, val); }
		public int compareTo(Pos p) { return Long.compare(val, p.val); }
	}
	long exp(long b, long e) {
		if(e == 0)
			return 1L;
		if(e % 2 == 0) {
			long tmp = exp(b, e / 2);
			return (tmp * tmp) % MOD;
		} else {
			return (b * exp(b, e - 1)) % MOD;
		}
	}
	long inv(long x) { return exp(x, MOD - 2); }
	final int MOD = 998_244_353;
	public CF1042E() {
		FS scan = new FS();
		int r = scan.nextInt(), c = scan.nextInt();
		int n = r * c;
		long[][] matrix = new long[r][c];
		Pos[] p = new Pos[n];
		int pointer = 0;
		for(int i = 0 ; i < r ; i++) {
			for(int j = 0 ; j < c ; j++) {
				matrix[i][j] = scan.nextLong();
				p[pointer++] = new Pos(i, j, matrix[i][j]);
			}
		}
		Pos start = new Pos(scan.nextInt() - 1, scan.nextInt() - 1, 0);
		start.val = matrix[start.row][start.col];
		int[] index = new int[n];
		Arrays.sort(p);
		for(int i = 1 ; i < n ; i++) {
			if(p[i].val == p[i - 1].val) {
				index[i] = index[i - 1];
			} else {
				index[i] = i;
			}
		}
		long[] rSqSum = new long[n];
		long[] cSqSum = new long[n];
		long[] rSum = new long[n];
		long[] cSum = new long[n];
		long[] expSum = new long[n];
		rSqSum[1] = sq(p[0].row) % MOD;
		cSqSum[1] = sq(p[0].col) % MOD;
		rSum[1] = p[0].row;
		cSum[1] = p[0].col;
		long[] inv = new long[n];
		if(p[0].equals(start)) {
			System.out.println(0);
		} else {
			for(int i = 1 ; i < n ; i++) {
				
				// (i^2 + j^2) * options
				long currSum = (sq(p[i].row) + sq(p[i].col));
				currSum = (currSum * index[i]) % MOD;
				
				// x^2 + y^2 for all x and y such that matrix[i][j] > matrix[x][y]
				long res = (currSum + rSqSum[index[i]] + cSqSum[index[i]]) % MOD;

				// 2i * (x1 + x2 + x3 + ...)
				long xTmp = (p[i].row << 1);
				xTmp = (xTmp * rSum[index[i]]) % MOD;

				// 2j * (y1 + y2 + y3 + ...)
				long yTmp = (p[i].col << 1);
				yTmp = (yTmp * cSum[index[i]]) % MOD;

				// subtracting away 2*i*sum(x) and 2*j*sum(y)
				res = (res - xTmp + MOD) % MOD;
				res = (res - yTmp + MOD) % MOD;

				// sum(E(x, y))
				res = (res + expSum[index[i]]) % MOD;

				// check if I have this inverse already
				if(inv[index[i]] == 0) inv[index[i]] = inv(index[i]);
				
				// multiply by inv(options)
				res = (res * inv[index[i]]) % MOD;

				if(p[i].equals(start)) {
					System.out.println(res);
					break;
				}
				rSqSum[i + 1] = (rSqSum[i] + sq(p[i].row)) % MOD;
				cSqSum[i + 1] = (cSqSum[i] + sq(p[i].col)) % MOD;
				rSum[i + 1] = (rSum[i] + p[i].row) % MOD;
				cSum[i + 1] = (cSum[i] + p[i].col) % MOD;
				expSum[i + 1] = (expSum[i] + res) % MOD;
			}
		}
	}
	long dist(int i, int j, int ii, int jj) { return sq(i - ii) + sq(j - jj); }
	long sq(long x) { return x * x; }
	class FS {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");
		public String next() {
			while(!st.hasMoreTokens()) {
				try { st = new StringTokenizer(br.readLine()); }
				catch(Exception e) { e.printStackTrace(); }
			}
			return st.nextToken();
		}
		public int nextInt() { return Integer.parseInt(next()); }
		public long nextLong() { return Long.parseLong(next()); }
	}
	public static void main(String[] args) { new CF1042E(); }
}
