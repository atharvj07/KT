#include <bits/stdc++.h>
using namespace std;
pair<pair<int, int>, int> p[100010];
int vst[100010];
set<int> s;
set<int>::iterator it;
int main() {
  int N, M, i;
  scanf("%d %d", &N, &M);
  for (i = 1; i <= N; i++) s.insert(i);
  for (i = 0; i < M; i++) {
    scanf("%d %d", &p[i].first.first, &p[i].first.second);
    p[i].second = i + 1;
  }
  sort(p, p + M);
  int ans = 0;
  for (i = 0; i < M; i++) {
    if (s.empty()) break;
    it = s.lower_bound(N - p[i].first.second + 1);
    if (it == s.end()) continue;
    if ((*it) <= p[i].first.first) {
      s.erase(it);
      ans++;
      vst[p[i].second] = 1;
    }
  }
  printf("%d\n", ans);
  for (i = 1; i <= M; i++) {
    if (vst[i]) printf("%d ", i);
  }
  return 0;
}
