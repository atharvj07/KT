(n, m, k) = map(int, input().split())
div = [(i, k // i) for i in range(1, int(k ** 0.5) + 1) if k % i == 0]
div += [(j, i) for (i, j) in div if i != j]
a = [len(i) for i in input().replace(' ', '').split('0') if len(i) != 0]
b = [len(i) for i in input().replace(' ', '').split('0') if len(i) != 0]
print(sum((sum((x - div1 + 1 for x in a if div1 <= x)) * sum((y - div2 + 1 for y in b if div2 <= y)) for (div1, div2) in div)))
