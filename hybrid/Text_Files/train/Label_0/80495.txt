n = int(input())
s = input()
if n % 4 != 0:
	print('===')
else:
	bkv = []
	bkv.append(s.count('A'))
	bkv.append(s.count('C'))
	bkv.append(s.count('G'))
	bkv.append(s.count('T'))
	mx = max(bkv)
	if mx > n / 4:
		print('===')
	else:
		ma = n // 4
		A = bkv[0]
		C = bkv[1]
		G = bkv[2]
		T = bkv[3]
		nom = 0
		st = ''
		for i in s:
			if i == '?':
				if A < ma:
					A += 1
					st += 'A'
				elif C < ma:
					C += 1
					st += 'C'
				elif G < ma:
					G += 1
					st += 'G'
				elif T < ma:
					T += 1
					st += 'T'
			else:
				st += i
		print(st)
