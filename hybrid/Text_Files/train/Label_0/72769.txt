for _ in range(int(input())):
	n = int(input())
	lst = list(input())
	(a, b) = (['0'] * n, ['0'] * n)
	for i in range(n):
		if lst[i] == '1':
			a[i] = '1'
			b[i] = '0'
			for j in range(i + 1, n):
				b[j] = lst[j]
			break
		elif lst[i] == '0':
			a[i] = b[i] = '0'
		elif lst[i] == '2':
			a[i] = b[i] = '1'
	print(''.join(a))
	print(''.join(b))
