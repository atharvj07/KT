import java.util.*;
import java.io.*;

public class Main{
    
    BufferedReader in;
    StringTokenizer str = null;
    PrintWriter out;
    
    private String next() throws Exception{
    	if (str == null || !str.hasMoreElements())
    	    str = new StringTokenizer(in.readLine());
    	return str.nextToken();
    }
    
    private int nextInt() throws Exception{
	return Integer.parseInt(next());
    }
    
    private long nextLong() throws Exception{
	return Long.parseLong(next());
    }


    int n, e;
    long V;

    int [][]g;
    long []a, b, amount;
    boolean used[];
    
    long total = 0;
    int size = 0, yk = 0;
    int []comp, prev;

    int []from, to;
    public void run() throws Exception{
    	in = new BufferedReader(new InputStreamReader(System.in));
    	out = new PrintWriter(System.out);

	n = nextInt();
	V = nextLong();
	e = nextInt();
	
	a = new long[n];
	for(int i=0;i<n;i++){
	    a[i] = nextLong();
	}
	b = new long[n];
	for(int i=0;i<n;i++){
	    b[i] = nextLong();
	}
	
	from = new int[2*n*n];
	to = new int[2*n*n];
	amount = new long[2*n*n];

	g = new int[n][n];
	for(int i=0;i<e;i++){
	    int x = nextInt()-1, y = nextInt()-1;
	    g[x][y] = g[y][x] = 1;
	}
	
	comp = new int[n];
	prev = new int[n];
	Arrays.fill(prev, -1);
	used = new boolean[n];
	boolean ok  =true;
	
	for(int i=0;i<n;i++){
	    if (!used[i]){
		size = 0;
		total = 0;
		dfs(i);
		if (total != 0){
		    ok = false;
		    break;
		}
		boolean process = true;
		while(process){

		    process = false;

		    for(int j=0;j<size;j++){
			int v = comp[j];
			if (a[v] <= b[v]) continue;
			process = true;
			for(int k=0;k<size;k++){
			    prev[comp[k]] = -1;
			}
			parents(v);
			int u = -1;
			for(int k=0;k<size;k++){
			    int t = comp[k];
			    if (a[t] < b[t]){
				u = t;
				break;    
			    }
			}
			long d = Math.min(a[v] - b[v], b[u] - a[u]);
			pour(v, u, d);
		    }
		}
	    }
	}

	if (!ok){
	    out.println("NO");
	}else{
	    out.println(yk);
	    for(int i=0;i<yk;i++){
		out.println(from[i] + " " + to[i] + " " + amount[i]);
	    }
	}
	out.close();
    }

    private void pour(int s, int t, long d){
	if (s == t) return;
	int parent = prev[t];
	long x = Math.min(d, a[parent]);
	x = Math.min(x, V - a[t]);
	add(parent, t, x);
	pour(s, parent, d);
	add(parent, t, d-x);
    }

    private void add(int s, int t, long d){
	if (d == 0) return;
	from[yk] = s+1;
	to[yk] = t+1;
	amount[yk++] = d;
	a[s]-=d;
	a[t]+=d;
    }

    private void parents(int x){
	for(int i=0;i<n;i++){
	    if (prev[i] < 0 && g[x][i] == 1) {
		prev[i] = x;
		parents(i);
	    }
	}
    }

    private void dfs(int x){
	used[x] = true;
	total+=a[x] - b[x];
	for(int i=0;i<n;i++){
	    if (!used[i] && g[x][i] == 1){
		dfs(i);
	    }
	}
	comp[size++] = x;
    }

    public static void main(String args[]) throws Exception{
	   new Main().run();
    }
}
