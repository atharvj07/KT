from sys import stdin
s = input()
(k,) = map(int, stdin.readline().split())
n = len(s)
MOD = 10 ** 9 + 7
numbers = 0
for i in range(n):
	if s[i] == '0' or s[i] == '5':
		numbers += pow(2, i, MOD)
a = 1 - pow(2, n * k, MOD) % MOD
b = 1 - pow(2, n, MOD)
inverse_b = pow(b, MOD - 2, MOD)
sumx = a * inverse_b % MOD
resp = sumx * numbers % MOD
print(int(resp))
