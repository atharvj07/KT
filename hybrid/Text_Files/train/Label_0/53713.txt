public class d implements Runnable {
    FS in;
    PrintWriter out;
    ArrayList<Integer> v, t;
    boolean[] vis;
    
    public d(FS i, PrintWriter o) {
        in = i;
        out = o;
    }
    
    ArrayList<Integer>[] adj;
    
    public static void main(String[] args) throws Throwable {
        FS in = new FS(System.in);
        PrintWriter out = new PrintWriter(System.out);
        
        Thread s = new Thread(null, new d(in, out), "haha gottem", 1l<<28);
        
        s.start();
        
        s.join();
        
        out.close();
    }
    
    public void run() {
        int n = in.nextInt();
        adj = new ArrayList[n];
        for(int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for(int i = 1; i < n; i++) {
            int u = in.nextInt()-1;
            int v = in.nextInt()-1;
            adj[u].add(v);
            adj[v].add(u);
        }
        vis = new boolean[n];
        v = new ArrayList<>();
        t = new ArrayList<>();
        dfs(0, 0);
        out.println(v.size());
        for(int i = 0; i < v.size(); i++) {
            out.println(v.get(i)+1 + " " + t.get(i));
        }
    }
    
    void dfs(int ind, int val) {
        vis[ind] = true;
        v.add(ind);
        t.add(val);
        int curr = val;
        int max = Math.max(adj[ind].size(), val);
        int min = max-adj[ind].size();
        if(val != 0 && val >= adj[ind].size()) {
            v.add(ind);
            t.add(min);
            curr = min;
        }
        for(int i : adj[ind]) {
            if(vis[i]) continue;
            dfs(i, curr+1);
            v.add(ind);
            t.add(curr+1);
            curr++;
            if(curr == max && val != 0) {
                v.add(ind);
                t.add(min);
                curr = min;
            }
        }
        if(curr == max && val != 0) {
            v.add(ind);
            t.add(min);
            curr = min;
        }
    }
        
    static class FS {

        BufferedReader in;
        StringTokenizer token;
        
        public FS(InputStream str) {
            in = new BufferedReader(new InputStreamReader(str));
        }
        
        public String next() {
            if (token == null || !token.hasMoreElements()) {
                try {
                    token = new StringTokenizer(in.readLine());
                } catch (IOException ex) {
                }
                return next();
            }
            return token.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(next());
        }
    }
}
