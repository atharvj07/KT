#include <bits/stdc++.h>
#pragma warning(disable : 4996)
using namespace std;
long long _max = (1LL << 62) - 1;
int main() {
  long long a, b, ans = 0;
  cin >> a >> b;
  while (_max) {
    for (int i = 0; (1LL << i) <= _max / 2; i++) {
      long long res = _max - (1LL << i);
      if (res >= a && res <= b) ans++;
    }
    _max >>= 1;
  }
  cout << ans;
  return 0;
}
