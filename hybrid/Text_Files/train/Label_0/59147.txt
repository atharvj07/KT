import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in); 
        int n = input.nextInt();
        String[] s = new String[51];
        s[0] = input.nextLine();
        for (int i = 0; i < n; ++i) {
            s[i] = input.nextLine();
        }
        String t = input.nextLine();
        int st = 0, res = 0;
        while (st != t.length()) {
            int add = 0;
            for (int i = 0; i < n; ++i) {
                int cur = 0;
                for (int j = 0; j < s[i].length(); ++j) {
                    if (s[i].charAt(j) == t.charAt(st+cur)) ++cur;
                    if (st+cur == t.length()) break;
                }
                if (add < cur) add = cur;
            }
            if (add == 0) {
                System.out.print(-1);
                System.exit(0);
            }
            st += add;
            ++res;   
        }
        System.out.print(res);
    }
}
