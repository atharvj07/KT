(n, k) = map(int, input().split())
l = []
for i in range(n):
	l.append(input())
password = input()
maximum = 0
for i in l:
	if len(i) > maximum:
		maximum = len(i)
danhdau = [0] * (maximum + 1)
best = 0
worst = 0
for i in l:
	danhdau[len(i)] += 1
for i in range(len(danhdau)):
	if i <= len(password):
		best += danhdau[i]
		worst += danhdau[i]
best = best - danhdau[len(password)] + 1
if best % k == 0:
	best = best + (best // k - 1) * 5
else:
	best = best // k * 5 + best
if worst % k == 0:
	worst = worst + (worst // k - 1) * 5
else:
	worst = worst // k * 5 + worst
print('{0} {1}'.format(best, worst))
