#include <bits/stdc++.h>
using namespace std;
const int N6 = 1e6 + 6, N3 = 1e3 + 6, oo = 1e9 + 9;
const long long ool = 1e18 + 9;
int n, p, q;
string s;
vector<string> ans;
int main() {
  ios_base ::sync_with_stdio(0);
  cin >> n >> p >> q;
  cin >> s;
  int x = 0;
  int cnt = 0;
  while ((n - x) % q != 0 && x <= n) {
    ++cnt;
    x += p;
  }
  if (x <= n && (n - x) % q == 0) {
    int cur = 0;
    for (int i = 1; i <= cnt; ++i) {
      string t = "";
      for (int j = cur; j <= cur + p - 1; ++j) {
        t += s[j];
      }
      cur += p;
      ans.push_back(t);
    }
    for (int i = 1; i <= (n - x) / q; ++i) {
      string t = "";
      for (int j = cur; j <= cur + q - 1; ++j) t += s[j];
      cur += q;
      ans.push_back(t);
    }
    cout << ans.size() << "\n";
    for (auto it : ans) cout << it << "\n";
    return 0;
  }
  cout << -1;
  return 0;
}
