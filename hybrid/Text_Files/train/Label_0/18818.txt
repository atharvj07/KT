#include <bits/stdc++.h>

using namespace std;
#define lint long long

lint gcd(lint a, lint b){
    if(b == 0){
        return a;
    }
    return gcd(b, a % b);
}

int main() {
    lint A, B, C, D;
    cin >> A >> B >> C >> D;
    
    lint X = abs(C - A);
    lint Y = abs(D - B);
    
    lint g = gcd(X, Y);
    
    lint x = X / g;
    lint y = Y / g;
    
    cout << g * (x + y - 1) << endl;
}
