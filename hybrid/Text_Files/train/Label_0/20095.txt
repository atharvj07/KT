class Point
{
public:
    int y, x;
    Point(){
        y = x = 0;
    }
    Point(int y0, int x0){
        y = y0;
        x = x0;
    }
    Point operator+(const Point& p) const{
        return Point(y + p.y, x + p.x);
    }
    Point operator-(const Point& p) const{
        return Point(y - p.y, x - p.x);
    }
    Point operator*(int a) const{
        return Point(y * a, x * a);
    }
    int length2() const{
        return y * y + x * x;
    }
    int dist2(const Point& p) const{
        return (y - p.y) * (y - p.y) + (x - p.x) * (x - p.x);
    }
    int dot(const Point& p) const{
        return y * p.y + x * p.x; // |a|*|b|*cos慮
    }
    int cross(const Point& p) const{
        return x * p.y - y * p.x; // |a|*|b|*sin慮
    }
};

class Circle
{
public:
    Point p;
    int r;
    bool isCollide(const Circle& c) const{
        return p.dist2(c.p) <= (r + c.r) * (r + c.r);
    }
    bool isInside(const Circle& c) const{
        return p.dist2(c.p) < (r - c.r) * (r - c.r);
    }
    bool operator<(const Circle& c) const{
        return r < c.r;
    }
};

int main()
{
    for(;;){
        int n;
        cin >> n;
        if(n == 0)
            return 0;

        Point s, t;
        cin >> s.x >> s.y >> t.x >> t.y;

        vector<Circle> c0(n);
        for(int i=0; i<n; ++i)
            cin >> c0[i].p.x >> c0[i].p.y >> c0[i].r;
        sort(c0.begin(), c0.end());

        vector<vector<Circle> > c(2);
        for(int i=0; i<n; ++i){
            bool a = (c0[i].p.dist2(s) < c0[i].r * c0[i].r);
            bool b = (c0[i].p.dist2(t) < c0[i].r * c0[i].r);
            if(a && !b)
                c[0].push_back(c0[i]);
            if(!a && b)
                c[1].push_back(c0[i]);
        }

        vector<vector<int> > num(2);
        for(int i=0; i<2; ++i){
            num[i].assign(c[i].size(), 1);
            for(unsigned j=0; j<c[i].size(); ++j){
                for(unsigned k=0; k<j; ++k){
                    if(c[i][k].isInside(c[i][j]))
                        num[i][j] = max(num[i][j], num[i][k] + 1);
                }
            }
        }

        int ret = 0;
        for(int i=0; i<2; ++i){
            for(unsigned j=0; j<num[i].size(); ++j){
                ret = max(ret, num[i][j]);
            }
        }
        for(unsigned i=0; i<num[0].size(); ++i){
            for(unsigned j=0; j<num[1].size(); ++j){
                if(!c[0][i].isCollide(c[1][j]))
                    ret = max(ret, num[0][i] + num[1][j]);
            }
        }
        cout << ret << endl;
    }
}
