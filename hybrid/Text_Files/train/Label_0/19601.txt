#include <bits/stdc++.h>
using namespace std;
namespace MCF {
const int wEPS = 0;
const int wINF = 1001001001;
const int cEPS = 0;
const int cINF = 1001001001;
int n, m, ptr[100010], next[100010], zu[100010];
int capa[100010], tof;
int cost[100010], toc, d[100010], pot[100010];
int vis[100010], pree[100010];
void init(int _n) {
  n = _n;
  m = 0;
  memset(ptr, ~0, n << 2);
}
void ae(int u, int v, int w, int c) {
  next[m] = ptr[u];
  ptr[u] = m;
  zu[m] = v;
  capa[m] = w;
  cost[m] = +c;
  ++m;
  next[m] = ptr[v];
  ptr[v] = m;
  zu[m] = u;
  capa[m] = 0;
  cost[m] = -c;
  ++m;
}
bool spRep(int src, int ink, int flo = wINF) {
  int f;
  int c, cc;
  int i, u, v;
  memset(pot, 0, n * sizeof(int));
  for (bool cont = 1; cont;) {
    cont = 0;
    for (u = 0; u < n; ++u)
      for (i = ptr[u]; ~i; i = next[i])
        if (capa[i] > wEPS) {
          if (pot[zu[i]] > pot[u] + cost[i] + cEPS) {
            pot[zu[i]] = pot[u] + cost[i];
            cont = 1;
          }
        }
  }
  for (toc = 0, tof = 0; tof + wEPS < flo;) {
    priority_queue<pair<int, int>, vector<pair<int, int> >,
                   greater<pair<int, int> > >
        q;
    for (u = 0; u < n; ++u) {
      d[u] = cINF;
      vis[u] = 0;
    }
    for (q.push(make_pair(d[src] = 0, src)); !q.empty();) {
      c = q.top().first;
      u = q.top().second;
      q.pop();
      if (vis[u]++) continue;
      for (i = ptr[u]; ~i; i = next[i])
        if (capa[i] > wEPS) {
          cc = c + cost[i] + pot[u] - pot[v = zu[i]];
          if (d[v] > cc) {
            q.push(make_pair(d[v] = cc, v));
            pree[v] = i;
          }
        }
    }
    if (!vis[ink]) return 0;
    f = flo - tof;
    for (v = ink; v != src; v = zu[i ^ 1]) {
      i = pree[v];
      f = min(f, capa[i]);
    }
    for (v = ink; v != src; v = zu[i ^ 1]) {
      i = pree[v];
      capa[i] -= f;
      capa[i ^ 1] += f;
    }
    tof += f;
    toc += f * (d[ink] - pot[src] + pot[ink]);
    for (u = 0; u < n; ++u) pot[u] += d[u];
  }
  return 1;
}
}  // namespace MCF
int get(string &inp, char my) {
  int ret = 0;
  for (__typeof(((long long)inp.size())) i = 0; i < (((long long)inp.size()));
       i++)
    ret += (inp[i] == my);
  return ret;
}
string T, take[110];
int maxi[110];
int main() {
  int tot;
  cin >> T;
  tot = ((long long)T.size());
  map<char, int> ase;
  for (__typeof(((long long)T.size())) i = 0; i < (((long long)T.size())); i++)
    ase[T[i]]++;
  string temp;
  for (__typeof((ase).begin()) it = (ase).begin(); it != (ase).end(); it++)
    temp += it->first;
  T = temp;
  int n = ((long long)T.size()), m;
  cin >> m;
  MCF::init(n + m + 2);
  int src = n + m, snk = src + 1;
  for (__typeof(m) i = 0; i < (m); i++) cin >> take[i] >> maxi[i];
  for (__typeof(n) i = 0; i < (n); i++) {
    for (__typeof(m) j = 0; j < (m); j++) {
      int cnt = get(take[j], T[i]);
      MCF::ae(i, j + n, cnt, j + 1);
    }
  }
  for (__typeof(n) i = 0; i < (n); i++) MCF::ae(src, i, ase[T[i]], 0);
  for (__typeof(m) i = 0; i < (m); i++) MCF::ae(i + n, snk, maxi[i], 0);
  MCF::spRep(src, snk);
  if (MCF::tof != tot) MCF::toc = -1;
  cout << MCF::toc << endl;
  return 0;
}
