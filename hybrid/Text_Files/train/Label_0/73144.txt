def read_one_line_input():
	s = input()
	return int(s)

def read_lines_input(n):
	lines = []
	for i in range(n):
		s = input()
		lines.append(s)
	return lines

def print_list(out_list):
	[print(s) for s in out_list]

def main(line):
	out = ''
	length = len(line)
	if length == 1:
		return str(line)
	yi_list = [[0 for col in range(2)] for row in range(2 * length + 1)]
	yi_cur = length
	for i in range(length):
		if line[i] == '0':
			yi_list[yi_cur][0] += 1
			yi_cur -= 1
		else:
			yi_list[yi_cur][1] += 1
			yi_cur += 1
	yi_cur = length
	for i in range(length):
		if yi_list[yi_cur][0] > 0 and yi_list[yi_cur - 1][1] > 0 or yi_list[yi_cur][1] == 0:
			yi_list[yi_cur][0] -= 1
			yi_cur -= 1
			out += '0'
		else:
			yi_list[yi_cur][1] -= 1
			yi_cur += 1
			out += '1'
	return out
n = read_one_line_input()
lines = read_lines_input(n)
out_list = []
for line in lines:
	out = main(line)
	out_list.append(out)
print_list(out_list)
