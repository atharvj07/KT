#include <bits/stdc++.h>
using namespace std;
map<char, char> q;
int main() {
  char a[30], b[30];
  scanf("%s", a);
  scanf("%s", b);
  q.clear();
  for (int i = 0; i < 26; i++) {
    q[a[i]] = b[i];
    q[a[i] - 32] = b[i] - 32;
  }
  char aim[2000];
  scanf("%s", aim);
  for (int i = 0; aim[i]; i++) {
    if (q[aim[i]])
      printf("%c", q[aim[i]]);
    else
      printf("%c", aim[i]);
  }
  printf("\n");
}
