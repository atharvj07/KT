def get_list(func=None):
	if not func:
		return input().split()
	return list(map(func, input().split()))
(n, a, b) = get_list(int)
aplusb = a + b
s = input()
edge = True
last = 'c'
for ch in s:
	if ch == '*':
		edge = True
	if ch == '.':
		if edge:
			if a < b:
				b -= 1
				last = 'b'
			else:
				a -= 1
				last = 'a'
			edge = False
		elif last == 'a':
			b -= 1
			last = 'b'
		else:
			a -= 1
			last = 'a'
print(aplusb - max(a, 0) - max(b, 0))
