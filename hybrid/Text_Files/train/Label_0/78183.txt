(n, p) = map(int, input().split())
a = [tuple(map(int, input().split())) for _ in range(n)]

def divCnt(left, right):
	return right // p - (left - 1) // p
prob = [0] * n
for (i, (left, right)) in enumerate(a):
	(nextLeft, nextRight) = a[(i + 1) % n]
	cnt1 = divCnt(left, right)
	cnt2 = divCnt(nextLeft, nextRight)
	sz1 = right - left + 1
	sz2 = nextRight - nextLeft + 1
	good = cnt1 * sz2 + sz1 * cnt2 - cnt1 * cnt2
	prob[i] += good / (sz1 * sz2)
ret = 0
for pr in prob:
	ret += 2000 * pr
print(ret)
