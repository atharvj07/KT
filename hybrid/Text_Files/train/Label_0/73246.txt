for _ in range(int(input())):
	(n, m) = map(int, input().split())
	k = sorted(list(map(int, input().split())), reverse=True)
	c = list(map(int, input().split()))
	x = 0
	y = 0
	i = 0
	while i < n and y + 1 < k[i] and (y < m):
		x += c[y]
		y += 1
		i += 1
	while i < n:
		x += c[k[i] - 1]
		i += 1
	print(x)
