(n,) = map(int, input().split())
arr = list(map(int, input().split()))
ans = 0
maxi = 0
for i in range(n):
	if arr[i] == 1:
		ans += 1
	elif arr[i] == 0:
		maxi = max(ans, maxi)
		ans = 0
pre = 0
suf = 0
for i in range(n):
	if arr[i] == 1:
		pre += 1
	else:
		break
for i in reversed(range(n)):
	if arr[i] == 1:
		suf += 1
	else:
		break
print(max(pre + suf, maxi))
