import sys
input = sys.stdin.buffer.readline
modulo = 998244353

def invMod(n):
	return pow(n, modulo - 2, modulo)

def fracMod(a, b):
	return a * invMod(b) % modulo
n = int(input())
wishList = [list(map(int, input().split())) for _ in range(n)]
wanted = [0] * (10 ** 6 + 2)
for wish in wishList:
	for present in wish[1:]:
		wanted[present] += 1
denom = pow(n, 2, modulo)
nom = 0
for wish in wishList:
	k = wish[0]
	invK = invMod(k)
	add = 0
	for present in wish[1:]:
		add += wanted[present]
	nom = (nom + add * invK) % modulo
ans = fracMod(nom, denom)
print(ans)
