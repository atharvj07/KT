import math
t = int(input())
a = list(map(int, input().split()))
d = math.gcd(a[0], a[1])
for i in range(2, len(a)):
	d = math.gcd(d, a[i])
for i in range(len(a)):
	a[i] = a[i] // d
total = sum(a)
if total % 2 == 1:
	print(0)
else:
	target = total // 2
	dp = [False] * (target + 1)
	dp[0] = True
	for v in a:
		for i in range(target, v - 1, -1):
			dp[i] = dp[i] or dp[i - v]
	if dp[target]:
		j = 0
		while a[j] % 2 == 0:
			j += 1
		print(1)
		print(j + 1)
	else:
		print(0)
