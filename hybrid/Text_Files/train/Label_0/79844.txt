import sys
n = int(sys.stdin.readline())
while n > 0:
	n -= 1
	ab = sys.stdin.readline().split()
	a = int(ab[0])
	b = int(ab[1])
	ch = (a - 1) % 4
	xr = 0
	if ch == 0:
		xr = a - 1
	elif ch == 1:
		xr = 1
	elif ch == 2:
		xr = a
	elif ch == 3:
		xr = 0
	if b == xr:
		print(a)
	elif xr ^ a == b:
		print(a + 2)
	else:
		print(a + 1)
