import bisect

def solve(nums):
	n = len(nums)
	a = []
	nums = [nums[i] - i for i in range(n)]
	for i in range(n):
		if nums[i] < nums[0] or nums[i] > nums[n - 1]:
			continue
		k = bisect.bisect_right(a, nums[i])
		if k == len(a):
			a.append(nums[i])
		else:
			a[k] = nums[i]
	res = n - len(a)
	return res
(n, k) = input().split()
n = int(n)
k = int(k)
a = input().split()
for i in range(len(a)):
	a[i] = int(a[i])
if k == 0:
	b = []
else:
	b = list(map(int, input().split()))
a = [-10 ** 10] + a + [10 ** 10]
b = [0] + b + [n + 1]
res = 0
flag = True
(t, temp) = (b[0], a[0])
for i in b[1:]:
	if i - t > a[i] - temp:
		flag = False
		break
	res += solve(a[t:i + 1])
	(t, temp) = (i, a[i])
if flag:
	print(res)
else:
	print(-1)
