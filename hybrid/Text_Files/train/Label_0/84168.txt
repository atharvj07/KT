for _ in range(int(input())):
	s = input()
	if not 'a' in s:
		print('NO')
		continue
	l = s.index('a')
	r = s.index('a')
	i = 97
	ans = 'YES'
	while l > 0 or r < len(s) - 1:
		i += 1
		if l > 0 and s[l - 1] == chr(i):
			l -= 1
		elif r < len(s) - 1 and s[r + 1] == chr(i):
			r += 1
		else:
			ans = 'NO'
			break
	print(ans)
