(n, k, m, d) = map(int, input().split())

def binary_search(curr):
	(l, r) = (1, m)
	while l <= r:
		mid = (l + r) // 2
		if n // mid >= curr:
			l = mid + 1
		else:
			r = mid - 1
	return l - 1
res = 0
for i in range(1, d + 1):
	x = binary_search((i - 1) * k + 1)
	if x <= 0 or n // x > i * k:
		continue
	res = max(res, x * i)
print(res)
