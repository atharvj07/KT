import java.io.*;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {

    private static final int left = 1 << 3;
    private static final int up = 1 << 2;
    private static final int right = 1 << 1;
    private static final int down = 1;
    private static final Direct[] dir = new Direct[]{
            new Direct(1, 0, 0, 15, 15),
            new Direct(0, 1, 0, down, up),
            new Direct(0, -1, 0, up, down),
            new Direct(0, 0, 1, right, left),
            new Direct(0, 0, -1, left, right)};

    private static Queue<Point> queue;
    private static int x, y;
    private static int[][][] map;

    public static void main(String[] args) throws IOException {

        Reader in = new Reader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(new OutputStreamWriter(System.out));

        int n = in.nextInt();
        int m = in.nextInt();

        map = new int[4][n][m];
        for (int i = 0; i < n; i++) {
            char[] now = in.next().toCharArray();
            for (int j = 0; j < m; j++) {
                if (now[j] == '+') {
                    map[0][i][j] = left | up | right | down;
                } else if (now[j] == '-') {
                    map[0][i][j] = left | right;
                } else if (now[j] == '|') {
                    map[0][i][j] = up | down;
                } else if (now[j] == '^') {
                    map[0][i][j] = up;
                } else if (now[j] == '>') {
                    map[0][i][j] = right;
                } else if (now[j] == '<') {
                    map[0][i][j] = left;
                } else if (now[j] == 'v') {
                    map[0][i][j] = down;
                } else if (now[j] == 'L') {
                    map[0][i][j] = up | right | down;
                } else if (now[j] == 'R') {
                    map[0][i][j] = left | up | down;
                } else if (now[j] == 'U') {
                    map[0][i][j] = left | right | down;
                } else if (now[j] == 'D') {
                    map[0][i][j] = left | up | right;
                } else {
                    map[0][i][j] = 0;
                }
            }
        }
        for (int level = 1; level < 4; level++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    map[level][i][j] = map[level - 1][i][j] >> 1 | ((map[level - 1][i][j] & 1) << 3);
                }
            }
        }

        Point start = new Point(0, in.nextInt() - 1, in.nextInt() - 1);

        x = in.nextInt() - 1;
        y = in.nextInt() - 1;

        int[][][] dis = new int[4][n][m];
        boolean[][][] vis = new boolean[4][n][m];
        queue = new LinkedList<>();
        queue.add(start);
        vis[start.z][start.x][start.y] = true;
        while (!queue.isEmpty()) {
            Point now = queue.poll();
            for (Direct d : dir) {
                Point next = new Point(now.z + d.z, now.x + d.x, now.y + d.y);
                if (next.x < 0 || next.x >= n || next.y < 0 || next.y >= m) {
                    continue;
                }
                if (vis[next.z][next.x][next.y]) {
                    continue;
                }
                if ((map[now.z][now.x][now.y] & d.from) == 0 || (map[next.z][next.x][next.y] & d.to) == 0) {
                    continue;
                }
                vis[next.z][next.x][next.y] = true;
                dis[next.z][next.x][next.y] = dis[now.z][now.x][now.y] + 1;
                queue.add(next);
            }
        }

        int ans = -1;
        for (int i = 0; i < 4; i++) {
            if (!vis[i][x][y]) continue;

            if (ans == -1) {
                ans = dis[i][x][y];
            } else {
                ans = Math.min(ans, dis[i][x][y]);
            }
        }

        out.println(ans);

        out.flush();
        in.close();
        out.close();
    }

    static class Point {
        int z, x, y;

        public Point(int z, int x, int y) {
            this.z = z % 4;
            this.x = x;
            this.y = y;
        }
    }

    static class Direct extends Point {
        int from, to;

        public Direct(int z, int x, int y, int from, int to) {
            super(z, x, y);
            this.from = from;
            this.to = to;
        }
    }

    static class Reader {
        BufferedReader reader;
        StringTokenizer st;

        Reader(InputStreamReader stream) {
            reader = new BufferedReader(stream);
            st = null;
        }

        void close() throws IOException {
            reader.close();
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        String nextLine() throws IOException {
            return reader.readLine();
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

    }
}
