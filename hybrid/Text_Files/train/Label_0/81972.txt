n = int(input())
l = list(input())
l1 = [0] * n
(j, k) = (0, 0)
for i in range(len(l) - 2):
	if len(l) % 2 == 0:
		if i == 0:
			m = n // 2 - 1
			l1[m] = l[0]
		elif i % 2 != 0:
			j += 1
			l1[m + j] = l[i]
		else:
			k -= 1
			l1[m + k] = l[i]
	elif i == 0:
		m = n // 2
		l1[m] = l[0]
	elif i % 2 != 0:
		j += 1
		l1[m - j] = l[i]
	else:
		k += 1
		l1[m + k] = l[i]
l1[0] = l[n - 2]
l1[-1] = l[n - 1]
s = ''.join(l1)
print(s)
