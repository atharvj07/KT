(n, k) = map(int, input().split())
a = list(map(int, input().split()))
lo = 0
hi = n + 1

def works(v):
	c = [1 if av >= v else -1 for av in a]
	s = [0]
	for i in range(len(c)):
		s.append(s[-1] + c[i])
	mins = 0
	for i in range(k, len(c) + 1):
		mins = min(mins, s[i - k])
		if s[i] - mins > 0:
			return True
	return False
while lo + 1 < hi:
	m = (lo + hi) // 2
	if works(m):
		lo = m
	else:
		hi = m
print(lo)
