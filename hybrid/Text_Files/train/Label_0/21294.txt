#include <bits/stdc++.h>
using namespace std;
struct floodlight {
  int X;
  int Y;
  int angle;
};
double dp[1 << 22];
floodlight f[22];
int mask, i, j, N, L, R;
double MAX, current;
int main() {
  scanf("%d%d%d", &N, &L, &R);
  for (i = 1; i <= N; ++i) scanf("%d%d%d", &f[i].X, &f[i].Y, &f[i].angle);
  fill(dp, dp + (1 << (N + 1)) + 2, (double)L);
  for (mask = 1; mask < (1 << (N + 1)); ++mask)
    for (j = 1; j <= N; ++j) {
      if (mask & (1 << j)) continue;
      current = f[j].angle * acos(-1) / 180.0;
      current += atan((dp[mask] - f[j].X) / f[j].Y);
      current = f[j].X + tan(current) * f[j].Y;
      current = min(current, 1.0 * R);
      if (current < dp[mask]) current = R;
      dp[mask + (1 << j)] = max(dp[mask + (1 << j)], current);
    }
  printf("%.9lf\n",
         max(1.0 * L, *max_element(dp + 1, dp + (1 << (N + 1)))) - 1.0 * L);
  return 0;
}
