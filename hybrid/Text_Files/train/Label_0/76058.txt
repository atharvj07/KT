import sys
for m in range(int(sys.stdin.readline())):
	n_k = list(map(int, sys.stdin.readline().strip().split(' ')))
	a = list(map(int, sys.stdin.readline().strip().split(' ')))
	n = n_k[0]
	k = n_k[1]
	b = a.copy()
	b.sort()
	hold = 0
	maximum = b[n - 1]
	Dict = {}
	counter = 0
	for elem in b:
		Dict[elem] = counter
		counter += 1
	for i in range(n - 1):
		number = a[i]
		nextnumber = a[i + 1]
		if nextnumber > number:
			if number != maximum:
				othernumber = b[Dict[number] + 1]
				if nextnumber == othernumber:
					hold += 1
	if n - hold <= k:
		print('YES')
	else:
		print('NO')
