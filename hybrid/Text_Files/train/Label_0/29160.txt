#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, x, y, i;
  cin >> n >> x >> y;
  vector<bool> v(n);
  string s;
  cin >> s;
  for (i = 0; i < n; i++) v[i] = s[i] == '0' ? false : true;
  long long turnsCount = 0;
  bool itIsTheEnd = false;
  long long start, end, count0, count1;
  i = 0;
  while (i < n && v[i]) i++;
  end = i;
  while (end <= n - 1) {
    start = i;
    while (i < n && !v[i]) {
      i++;
      count0++;
    }
    while (i < n && v[i]) {
      i++;
      count1++;
    }
    end = i;
    turnsCount++;
  }
  if (turnsCount == 0) {
    cout << 0;
  } else {
    long long withTurns = (turnsCount - 1) * min(x, y) + y;
    cout << withTurns;
  }
  return 0;
}
