#include <bits/stdc++.h>
#pragma GCC optimize("O3")
using namespace std;
const int mod = 1e9 + 7, nmax = 100001;
long long a[nmax], b[nmax];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    int h, m;
    cin >> h >> m;
    a[i] = m + h * 60;
  }
  a[n] = -s - 1;
  sort(a, a + n + 1);
  for (int i = 1; i < n + 1; i++)
    if (a[i] - a[i - 1] >= 2 * s + 2) {
      cout << (a[i - 1] + s + 1) / 60 << ' ' << (a[i - 1] + s + 1) % 60;
      return 0;
    }
  cout << (a[n] + s + 1) / 60 << ' ' << (a[n] + s + 1) % 60;
}
