t = int(input())
for _ in range(0, t):
	n = int(input())
	list1 = [int(num) for num in input().split()]
	list1.sort()
	pro1 = 1
	for i in range(n - 1, n - 5 - 1, -1):
		pro1 = pro1 * list1[i]
	pro2 = list1[0] * list1[1] * list1[n - 1] * list1[n - 2] * list1[n - 3]
	pro3 = list1[0] * list1[1] * list1[2] * list1[3] * list1[n - 1]
	print(max([pro1, pro2, pro3]))
