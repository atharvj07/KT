#include <bits/stdc++.h>
using namespace std;
const int N = 1005;
int a[N], ans[N], sum;
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
    sum += a[i];
  }
  for (int i = 1; i <= n; i++) {
    scanf("%d", &ans[i]);
    sum -= ans[i];
  }
  if (sum < 0)
    printf("NO");
  else
    printf("YES");
  return 0;
}
