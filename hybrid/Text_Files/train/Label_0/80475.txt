(a, b) = list(map(int, input().split()))
u = 0
t = []
for i in range(a):
	t.append(list(input()))
for j in t:
	if 'S' in j:
		pass
	else:
		for k in range(len(j)):
			j[k] = 'a'
			u += 1
for m in range(b):
	y = []
	for l in range(a):
		y.append(t[l][m])
	if 'S' not in y:
		for e in range(len(y)):
			if y[e] == '.':
				u += 1
				y[e] = 'a'
print(u)
