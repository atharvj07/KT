(n, k) = list(map(int, input().split()))
s = input()
count = [0] * 26
for c in s:
	count[ord(c) - 65] += 1
count = sorted(count, reverse=True)
res = 0
i = 0
while i < len(count) and k != 0:
	if k >= count[i]:
		res += count[i] * count[i]
		k -= count[i]
	else:
		res += k * k
		k = 0
	i += 1
print(res)
