from sys import stdin
n = int(input())
(x, ans, sum) = ([[0] * 3 for i in range(n)], 0, 0)
d = []
for i in range(n):
	(x[i][1], x[i][0], x[i][2]) = map(int, stdin.readline().split())
x.sort()
for i in range(n - 1, -1, -1):
	while len(d) > 0 and d[len(d) - 1][1] >= x[i][0]:
		sum -= d[len(d) - 1][2]
		d.pop()
	d.append(x[i])
	sum += x[i][2]
	ans = max(ans, sum)
print(ans)
