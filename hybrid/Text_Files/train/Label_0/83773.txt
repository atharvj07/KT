import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(s, n) = input().split()
	n = int(n)
	left = int(s)
	s = list(map(int, s))
	nums = [0] * n
	curr = 0
	for i in range(len(s)):
		exp = len(s) - 1 - i
		d = s[i]
		for use in range(d, -1, -1):
			if left - use * 10 ** exp >= n - 1 - (curr + use - 1):
				left -= use * 10 ** exp
				for j in range(use):
					nums[curr] += 10 ** exp
					if curr != n - 1:
						curr += 1
				if i != len(s) - 1:
					s[i + 1] += 10 * (d - use)
				break
	print(*nums)
