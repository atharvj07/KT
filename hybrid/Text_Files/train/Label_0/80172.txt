for i in range(int(input())):
	n = int(input())
	b = list(map(int, input().split()))
	ss = [0] * n
	ans = b[:]
	maxx = b[0]
	for i in range(n):
		ss[b[i] - 1] += 1
	tt = []
	y = 0
	for i in range(n):
		if ss[i] == 0:
			tt.append(i + 1)
	mm = len(tt)
	x = True
	for i in range(1, n):
		if ans[i] > maxx:
			maxx = ans[i]
		elif ans[i] == maxx:
			if y == mm:
				print(-1)
				x = False
				break
			elif tt[y] > ans[i]:
				print(-1)
				x = False
				break
			ans[i] = tt[y]
			y += 1
		else:
			print(-1)
			x = False
			break
	if x:
		print(*ans)
