public class D implements Runnable
{
    String file = "input";
    
    boolean TEST = false;
    
    void solve() throws IOException
    {
        int n = nextInt();
        int[] a = new int[n];
        boolean sorted = true;
        int luckyIndex = -1;
        Item[] is = new Item[n];
        for(int i = 0; i < n; i++)
        {
            a[i] = nextInt();
            if(i - 1 >= 0) sorted &= a[i] >= a[i - 1];
            if(isLucky(a[i])) luckyIndex = i;
            is[i] = new Item(a[i], i);
        }
        if(sorted)
        {
            out.println(0);
            return;
        }
        if(luckyIndex == -1) 
        {
            out.println(-1);
            return;
        }
        sort(is);
        int[] src = new int[n];
        int[] dest = new int[n];
        for(int i = 0; i < n; i++)
        {
            src[i] = is[i].index;
            dest[src[i]] = i;
        }
        List<String> as = new ArrayList<String>();
        for(int i = 0; i < n; i++)
            if(src[i] != i && src[i] != luckyIndex)
            {
                if(luckyIndex != i)
                {
                    as.add((luckyIndex + 1) + " " + (i + 1));
                    swap(luckyIndex, i, src, dest);
                    luckyIndex = i;
                }       
                
                as.add((luckyIndex + 1) + " " + (src[i] + 1));
                int tmp = src[i];
                swap(luckyIndex, src[i], src, dest);
                luckyIndex = tmp;
            }
        out.println(as.size());
        for(String s : as) out.println(s);
    }
    void swap(int i, int j, int[] src, int[] dest)
    {
        int di = dest[i];
        int dj = dest[j];
        int tmp = dest[i]; dest[i] = dest[j]; dest[j] = tmp;
        tmp = src[di]; src[di] = src[dj]; src[dj] = tmp;
    }
    
    boolean isLucky(long w)
    {
        while(w > 0)
        {
            long q = w % 10;
            if(q != 4 && q != 7) return false;
            w /= 10;
        }
        return true;
    }
    class Item implements Comparable<Item>
    {
        int value, index;
        Item(int value, int index)
        {
            this.value = value;
            this.index = index;
        }
        public int compareTo(Item it)
        {
            if(value != it.value) return value - it.value;
            return index - it.index;
        }
    }
    
    String next() throws IOException
    {
        while(st == null || !st.hasMoreTokens()) st = new StringTokenizer(input.readLine());
        return st.nextToken();
    }
    
    int nextInt() throws IOException
    {
        return Integer.parseInt(next());
    }
    
    long nextLong() throws IOException
    {
        return Long.parseLong(next());
    }
    
    double nextDouble() throws IOException
    {
        return Double.parseDouble(next());
    }
    
    void print(Object... o)
    {
        System.out.println(deepToString(o));

    void gcj(Object o)
    {
        String s = String.valueOf(o);
        out.println("Case #" + test + ": " + s);
        System.out.println("Case #" + test + ": " + s);
    }
    
    BufferedReader input;
    PrintWriter out;
    StringTokenizer st;
    int test;
    
    void init() throws IOException
    {
        if(TEST) input = new BufferedReader(new FileReader(file + ".in")); 
        else input = new BufferedReader(new InputStreamReader(System.in));
        out = new PrintWriter(new BufferedOutputStream(System.out));
    }
    
    public static void main(String[] args) throws IOException
    {
        new Thread(null, new D(), "", 1 << 20).start();
    }
    
    public void run()
    {
        try
        {
            init();
            if(TEST) 
            {
                int runs = nextInt();
                for(int i = 0; i < runs; i++) solve();
            }
            else solve();
            out.close();        
        }
        catch(Exception e)
        {
            e.printStackTrace();
            System.exit(1);
        }
    }
