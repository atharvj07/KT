k = int(input())
while k:
	n = int(input())
	s = list(input())
	t = list(input())
	c = 0
	l = [0] * 27
	d = {}
	for i in range(n):
		l[ord(s[i]) - 97] += 1
		l[ord(t[i]) - 97] += 1
	for i in l:
		if i % 2 != 0:
			c = 1
			break
	count = 0
	if c == 1:
		print('No')
	else:
		ans = []
		for i in range(n):
			if s[i] != t[i]:
				for j in range(i + 1, n):
					if t[j] == t[i]:
						t[j] = s[i]
						s[i] = t[i]
						d[i + 1] = 9
						ans.append(i + 1)
						ans.append(j + 1)
						break
				if d.get(i + 1, 0) == 0:
					for j in range(i + 1, n):
						if s[j] == t[i]:
							s[j] = t[i + 1]
							t[i + 1] = t[i]
							ans.append(j + 1)
							ans.append(i + 2)
							t[i + 1] = s[i]
							s[i] = t[i]
							ans.append(i + 1)
							ans.append(i + 2)
							break
		ll = len(ans) // 2
		if ll == 0:
			print('No')
		else:
			print('Yes')
			print(ll)
			m = 0
			for i in range(ll):
				print(ans[m], ans[m + 1], end=' ')
				m += 2
				print()
	k -= 1
