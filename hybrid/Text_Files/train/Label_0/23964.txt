#include <bits/stdc++.h>
using namespace std;
int a[10];
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < s.length(); i++) s[i] -= '0';
  for (int i = 0; i < s.length(); i++) {
    while (s[i] > 1) {
      if (s[i] == 2) a[2]++;
      if (s[i] == 3) a[3]++;
      if (s[i] == 4) a[2] += 2;
      if (s[i] == 5) a[5]++;
      if (s[i] == 6) a[3]++, a[2]++;
      if (s[i] == 7) a[7]++;
      if (s[i] == 8) a[2] += 3;
      if (s[i] == 9) a[3] += 2;
      s[i]--;
    }
  }
  string ans = "";
  for (int i = 7; i >= 0; i--) {
    while (a[i]) {
      if (i == 7) {
        ans += "7";
        a[7]--;
      }
      if (i >= 6) {
        a[3]--;
        a[2]--;
      }
      if (i == 5) ans += "5";
      if (i >= 5) a[5]--;
      if (i >= 4) a[2] -= 2;
      if (i == 3) ans += "3";
      if (i >= 3) a[3]--;
      if (i == 2) ans += "2";
      if (i >= 2) a[2]--;
    }
  }
  cout << ans;
  return 0;
}
