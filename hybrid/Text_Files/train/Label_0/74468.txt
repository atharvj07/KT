t = int(input())
for i in range(0, t):
	n = int(input())
	s = input()
	a = 0
	aa = 0
	b = 0
	c = 0
	for j in s:
		a = a + 1
		if j == '1' and b == 0:
			b = a
		if j == '1':
			c = a
	if b != 0:
		b = n - b + 1
	for j in s:
		aa = aa + 1
		if j == 1:
			aa = aa + 1
	if b * 2 > c * 2:
		if b * 2 > aa:
			print(b * 2)
		else:
			print(aa)
	elif c * 2 > aa:
		print(c * 2)
	else:
		print(aa)
