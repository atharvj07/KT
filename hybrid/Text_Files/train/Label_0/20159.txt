#include <iostream>
using namespace std;

const int MAX = 1<<17;
bool ifnprime[MAX+1];
long long int p[MAX];
long long int ind = 0;

void set(){
    ifnprime[0] = ifnprime[1] = true;
    long long int n,m;
    for(n=2;n<=MAX;n++){
        if(!ifnprime[n]){
            p[ind] = n;
            ind++;
            for(m=n*n;m<=MAX;m+=n) ifnprime[m] = true;
        }
    }
}

int main(){
    set();
    int n,fault=0;
    for(cin>>n;n--;){
        long long int a,x;
        bool ok = false;
        cin >> a;
        a = 2*a+1;
        for(x=0;p[x]*p[x]<=a;x++){
            if(a%p[x]==0){
                ok = true;
                break;
            }
        }
        if( !ok ) fault++;
    }
    cout << fault << endl;
    return 0;
}
