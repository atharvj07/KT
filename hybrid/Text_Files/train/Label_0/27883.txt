#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<vector<vector<int>>> adj(2, vector<vector<int>>(n));
  for (int u, v, c, i = 0; i < m; ++i) {
    cin >> u >> v >> c;
    --u, --v;
    adj[c][v].push_back(u);
  }
  vector<int> d(n, -1), col(n, -1);
  d[n - 1] = col[n - 1] = 0;
  queue<int> q;
  q.push(n - 1);
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    for (int i = 0; i < 2; ++i)
      for (auto v : adj[i][u]) {
        if (col[v] == -1)
          col[v] = i ^ 1;
        else if (col[v] == i && d[v] == -1) {
          d[v] = d[u] + 1;
          q.push(v);
        }
      }
  }
  cout << d[0] << "\n";
  for (auto &i : col) cout << max(i, 0);
  cout << "\n";
  return 0;
}
