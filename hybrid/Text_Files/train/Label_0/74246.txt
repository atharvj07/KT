n = int(input())
l = lambda i: int(i) % 2
h = [[*map(l, input().split())] for _ in ' ' * n]
s1 = sum((x[0] for x in h))
s2 = sum((x[1] for x in h))
if s1 % 2 == 0 and s2 % 2 == 0:
	print(0)
	exit(0)
else:
	c1 = h.count([1, 0])
	c2 = h.count([0, 1])
	if s1 % 2 != 0 and s2 % 2 != 0 and (c1 or c2):
		print(1)
		exit(0)
print(-1)
