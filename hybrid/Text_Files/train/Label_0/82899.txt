T = int(input())
while T > 0:
	T -= 1
	n = int(input())
	a = list(map(int, input('').split()))
	Sum = sum(a)
	tot = 0
	i = 0
	while i < n:
		tot += 2 * (a[i] - 1)
		i += 2
	if tot <= Sum:
		i = 0
		ans = []
		while i < n:
			if i % 2 == 1:
				ans.append(a[i])
			else:
				ans.append(1)
			i += 1
		print(*ans)
	else:
		i = 0
		ans = []
		while i < n:
			if i % 2 == 0:
				ans.append(a[i])
			else:
				ans.append(1)
			i += 1
		print(*ans)
