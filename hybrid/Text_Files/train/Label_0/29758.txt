const int MOD = 1e9 + 7;
const int maxn = 1 << 4;
vector<pair<int, int>> parent(maxn, {-1, -1});
int operate(int mask, int type) { return mask ^ ((maxn - 1) ^ (1 << type)); }
void pre() {
  queue<int> q;
  parent[0] = {0, -1};
  q.push(0);
  while (!q.empty()) {
    int curr = q.front();
    q.pop();
    for (auto type : {0, 1, 2, 3}) {
      int child = operate(curr, type);
      if (parent[child].first == -1) {
        parent[child] = {curr, type};
        q.push(child);
      }
    }
  }
}
class MainClass {
 public:
  int n, m;
  vector<string> grid;
  int grid_to_mask(int x, int y) {
    int mask = 0;
    if (grid[x][y] == '1') mask |= (1 << 3);
    if (grid[x][y + 1] == '1') mask |= (1 << 2);
    if (grid[x + 1][y] == '1') mask |= (1 << 1);
    if (grid[x + 1][y + 1] == '1') mask |= (1 << 0);
    return mask;
  }
  vector<pair<int, int>> conv(int x, int y, int type) {
    vector<pair<int, int>> res;
    type = (maxn - 1) ^ (1 << type);
    if (type & (1 << 3)) res.push_back({x, y});
    if (type & (1 << 2)) res.push_back({x, y + 1});
    if (type & (1 << 1)) res.push_back({x + 1, y});
    if (type & (1 << 0)) res.push_back({x + 1, y + 1});
    return res;
  }
  void solve() {
    cin >> n >> m;
    grid.resize(n);
    for (auto& i : grid) cin >> i;
    vector<vector<pair<int, int>>> ans;
    for (int i = 0; i <= n - 2; i++) {
      for (int j = 0; j <= m - 2; j++) {
        debug() << " ["
                << "i"
                   ": "
                << (i)
                << "] "
                   " ["
                << "j"
                   ": "
                << (j) << "] ";
        for (int curr = grid_to_mask(i, j); curr != 0;
             curr = parent[curr].first) {
          ans.push_back(conv(i, j, parent[curr].second));
        }
        grid[i][j] = '0';
        grid[i][j + 1] = '0';
        grid[i + 1][j] = '0';
        grid[i + 1][j + 1] = '0';
        if (j < m - 3) j++;
      }
      if (i < n - 3) i++;
    }
    for (int i = 0; i <= n - 1; i++)
      for (int j = 0; j <= m - 1; j++) assert(grid[i][j] == '0');
    cout << ans.size() << "\n";
    for (auto i : ans) {
      for (auto j : i) cout << j.first + 1 << " " << j.second + 1 << " ";
      cout << "\n";
    }
    return;
  }
};
int main() {
  pre();
  int test = 1;
  cin >> test;
  for (int tst = 1; tst <= test; ++tst) {
    MainClass Ausmosian;
    Ausmosian.solve();
    cout << "\n";
  }
  return 0;
}
