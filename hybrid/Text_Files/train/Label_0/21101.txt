#include <bits/stdc++.h>
using namespace std;
int n, k;
int f[2][1005], g[2][1005];
vector<int> v[100005];
int main() {
  cin >> n >> k;
  for (int x, y, i = 1; i <= k; i++) {
    scanf("%d%d", &x, &y);
    v[y].push_back(x + 1 - y);
  }
  for (int i = 1; i <= n; i++) sort(v[i].begin(), v[i].end());
  int pre = 0, cur = 1;
  for (int i = 1; i <= n; i++) {
    f[cur][0] = g[cur][0] = g[pre][1] + 3 * (v[i].end() - v[i].begin());
    for (int k, j = 1; j <= min(n - i + 1, 1000); j++) {
      k = 3 *
          (upper_bound(v[i].begin(), v[i].end(), n - i + 1 - j) - v[i].begin());
      f[cur][j] = 2 + j * (j + 1) / 2 + g[pre][j] + k;
      if (j < 1000 && i > 1) f[cur][j] = min(f[cur][j], f[pre][j + 1] + k);
      g[cur][j] = min(g[cur][j - 1], f[cur][j]);
    }
    swap(pre, cur);
  }
  printf("%d\n", min(f[pre][0], f[pre][1]));
}
