inputs = input().split(' ')
(numberOfSeats, programmers, athletes) = (int(inputs[0]), int(inputs[1]), int(inputs[2]))
seats = input()
result = 0
lastSit = ''
for c in seats:
	if c == '.':
		if programmers > athletes and programmers > 0:
			if lastSit != 'p':
				programmers -= 1
				result += 1
				lastSit = 'p'
			elif athletes > 0:
				athletes -= 1
				result += 1
				lastSit = 'a'
			else:
				lastSit = ''
		elif athletes >= programmers and athletes > 0:
			if lastSit != 'a':
				athletes -= 1
				result += 1
				lastSit = 'a'
			elif programmers > 0:
				programmers -= 1
				result += 1
				lastSit = 'p'
			else:
				lastSit = ''
	else:
		lastSit = ''
print(result)
