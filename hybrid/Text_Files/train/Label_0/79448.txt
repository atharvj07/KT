for _ in range(int(input())):
	(l, r) = [int(i) for i in input().split()]
	s = 0
	if l == r:
		if l % 2:
			print(-l)
		else:
			print(l)
	else:
		if l % 2 == 0:
			s += l
			l += 1
		if r % 2:
			s -= r
			r -= 1
		if l == r:
			if l % 2:
				s -= l
			else:
				s += l
		else:
			n = r - l + 1
			n //= 2
			s += n * (l + 1 + r) // 2
			s -= n * (l + r - 1) // 2
		print(s)
