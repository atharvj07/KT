#include <bits/stdc++.h>
using namespace std;
inline void open(const char *s) {}
inline int rd() {
  static int x, f;
  x = 0;
  f = 1;
  char ch = getchar();
  for (; ch < '0' || ch > '9'; ch = getchar())
    if (ch == '-') f = -1;
  for (; ch >= '0' && ch <= '9'; ch = getchar()) x = x * 10 + ch - '0';
  return f > 0 ? x : -x;
}
const int N = 200010;
int n, m;
pair<int, int> a[N];
map<pair<int, int>, bool> mp;
inline bool gao(int d) {
  for (int i = 1; i <= m; i++) {
    int x = (a[i].first - 1 + d) % n + 1;
    int y = (a[i].second - 1 + d) % n + 1;
    if (x > y) swap(x, y);
    if (!mp.count(pair<int, int>(x, y))) return 0;
  }
  return 1;
}
int main() {
  n = rd();
  m = rd();
  for (int i = 1; i <= m; i++) {
    int x = rd(), y = rd();
    if (x > y) swap(x, y);
    mp[a[i] = pair<int, int>(x, y)] = 1;
  }
  for (int i = 1; i <= n - 1; i++)
    if (n % i == 0 && gao(i)) return puts("Yes"), 0;
  puts("No");
  return 0;
}
