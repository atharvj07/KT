import java.util.*;

public class javatest {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t=sc.nextInt();
		for (int i=0;i<t;i++) {
			int a=sc.nextInt();
			int b=sc.nextInt();
			String s=sc.next();
			System.out.println(find(s,a,b));
		}
	}
	
	public static String find(String s,int cnt0,int cnt1) {
		StringBuilder sb=new StringBuilder(s);
		int len=cnt1+cnt0;
		int start=0,end=len-1;
		int[] cnt= {cnt0,cnt1};
		while (start<len/2) {
			if (sb.charAt(start)!='?'&&sb.charAt(end)!='?') {
				if (sb.charAt(start)!=sb.charAt(end)) return "-1";
				cnt[(int)(sb.charAt(start)-'0')]-=2;
			}
			else if (sb.charAt(start)!='?'){
				sb.setCharAt(end, sb.charAt(start));
				cnt[(int)(sb.charAt(start)-'0')]-=2;
			}
			else if (sb.charAt(end)!='?') {
				sb.setCharAt(start, sb.charAt(end));
				cnt[(int)(sb.charAt(end)-'0')]-=2;
			}
			start++;end--;
		}
		start=0;end=len-1;
		while (start<len/2) {
			if (sb.charAt(start)=='?') {
				if (cnt[0]>=2) {
					sb.setCharAt(start, '0');
					sb.setCharAt(end, '0');
				}
				else {
					sb.setCharAt(start, '1');
					sb.setCharAt(end, '1');
				}
				cnt[(int)(sb.charAt(start)-'0')]-=2;
			}
			start++;end--;	
		}
		if (sb.charAt(len/2)=='?'&&cnt[0]!=0) {
			cnt[0]--;
			sb.setCharAt(len/2, '0');
		}
		else if (sb.charAt(len/2)=='?'&&cnt[1]!=0) {
			cnt[1]--;
			sb.setCharAt(len/2, '1');
		}
		else if (len%2!=0)  cnt[(int)(sb.charAt(len/2)-'0')]--;
		if (cnt[0]==0&&cnt[1]==0)
			return sb.toString();
		return "-1";
	}
	
}
