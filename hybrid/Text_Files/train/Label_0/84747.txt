s = input()
l = []
n = len(s)
for i in range(n):
	l.append(int(s[i]))
for i in range(n):
	l[i] = l[i] % 3
d = {1: 0, 2: 0, 0: 0}
cs = 0
for i in range(n):
	if l[i] == 0:
		cs += 1
		(d[0], d[1], d[2]) = (0, 0, 0)
		i += 1
	elif l[i] == 1:
		d[1] += 1
		if d[1] % 3 == 0 or d[2] != 0:
			cs += 1
			(d[1], d[2]) = (0, 0)
	else:
		d[2] += 1
		if 2 * d[2] % 3 == 0 or d[1] != 0:
			cs += 1
			(d[1], d[2]) = (0, 0)
print(cs)
