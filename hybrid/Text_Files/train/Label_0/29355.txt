#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main(){

    int n;
    cin >> n;
    vector<int> vec(n);
    for(int i = 0; i < n; ++i){
        cin >> vec[i];
    }

    vector<int> origin = vec;

    if(prev_permutation(vec.begin(), vec.end())){
        for(auto it = vec.begin(); it != vec.end(); ++it){
            if(it != vec.begin()){
                cout << " ";
            }
            cout << *it;
        }
        cout << endl;
    }
    
    for(auto it = origin.begin(); it != origin.end(); ++it){
        if(it != origin.begin()){
            cout << " ";
        }
        cout << *it;
    }
    cout << endl;

    if(next_permutation(origin.begin(), origin.end())){
        for(auto it = origin.begin(); it != origin.end(); ++it){
            if(it != origin.begin()){
                cout << " ";
            }
            cout << *it;
        }
        cout << endl;
    }

    return 0;
}
