#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;

#define  ROT(dr) dr=(dr+1)%4;

bool f[205][205];

int W,H;
int dx[]={-1,0,1,0};
int dy[]={0,-1,0,1};

int wdx[]={-1,-1, 0, 1, 1,  1,  0,-1};
int wdy[]={ 0, 1, 1, 1, 0, -1, -1,-1};

void setw(int x1, int y1, int x2, int y2)
{
	x1*=2; y1*=2; x2*=2; y2*=2;
	y1=H*2-y1; y2=H*2-y2; 
	if(x1>x2) swap(x1,x2);
	if(y1>y2) swap(y1,y2);
	
	for(int i=y1; i<=y2; i++)
	for(int j=x1; j<=x2; j++)
		f[j][i]=1;
}

int main()
{
	int N;
	while(cin >> W >> H >> N,(W||H||N))
	{
		memset(f,0,sizeof(f));
		
		setw(0,0,0,H);
		setw(0,0,W,0);
		setw(W,0,W,H);
		setw(0,H,W,H);
		
		while(N--)
		{
			int x1,y1,x2,y2;
			cin >> x1 >> y1 >> x2 >> y2;
			setw(x1,y1,x2,y2);
		}
		
		int ux,uy,vx,vy,gx,gy;
		cin >> ux >> uy >> vx >> vy >> gx >> gy;
		int sx = max(ux,vx)*2;
		int sy = max(uy,vy)*2;
		int dr = 0;
		gx*=2; gx++; gy*=2; gy++;
		gy=H*2-gy;
		
		
		if(sx!=min(ux,vx)*2) sx--;
		if(sy!=min(uy,vy)*2) sy--;
		
		sy=H*2-sy;
		
		f[sx][sy]=0;
		
		bool v[205][205][4]={0};
		
		for(int i=0; i<4; i++)
		{
			int tx=sx+dx[i], ty=sy+dy[i];
			if(tx<0||ty<0||tx>=2*W+1||ty>=2*H+1) continue;
			if(f[tx][ty]) continue;
			
			dr=i;
			break;
		}
		/*
		for(int i=0; i<=H*2; i++)
		{
			for(int j=0; j<=W*2; j++)
			{
				cout << f[j][i];
			}
			cout << endl;
		}
		*/
		
		int c=0;
		bool g=false;
		while(1)
		{
			if(v[sx][sy][dr]) break;
			v[sx][sy][dr]=1;
			
			if(sx==gx&&sy==gy)
			{
				g=true;
				cout << c << endl;
				break;
			}
			
			int tx=sx+dx[dr], ty=sy+dy[dr];

			if(f[tx][ty]) 
			{
				for(int i=0; i<4; i++)
				{
					int wx=sx+dx[(dr+3+i)%4], wy=sy+dy[(dr+3+i)%4];
					if(f[wx][wy]==0)
					{
						dr=(dr+3+i)%4;
						break;
					}
				}
			}
			else
			{
				int wx=sx+dx[(dr+3)%4], wy=sy+dy[(dr+3)%4];
				
				if(f[wx][wy]==0)
				{
					dr=(dr+3)%4;
				}
			}

			sx+=dx[dr]; sy+=dy[dr];
			if(c!=0)
			{
				sx+=dx[dr]; sy+=dy[dr];
			}
			if(sx<0||sy<0||sx>=2*W+1||sy>=2*H+1) break;
			c++;
			
		}
		
		if(!g) cout << "Impossible" << endl;
	}
}
