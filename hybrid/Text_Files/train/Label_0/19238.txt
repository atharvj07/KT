#include <bits/stdc++.h>
#define MAX 200005
#define ll long long
#define P ((ll)1e9+7)
#define INF 0x3f3f3f3f
using namespace std;

template<typename T>
void read(T &n){
	n = 0;
	char c = getchar();
	int f = 1;
	while(!isdigit(c) && c != '-') c = getchar();
	if(c == '-') f = -1, c = getchar();
	while(isdigit(c)){
		n = (n<<1)+(n<<3)+c-'0';
		c = getchar();
	}
	n *= f;
}

int n, A, B, o1, o2;
int c[MAX], w[MAX], vis[MAX], a[MAX], cnt[MAX], mn[MAX];
ll fac[MAX], inv[MAX], ans = 1;

ll qpow(ll a, ll n){
	ll res = 1;
	while(n > 0){
		if(n&1) res = (res*a)%P;
		a = a*a%P;
		n >>= 1;
	}
	return res;
}

void init(){
	fac[0] = 1;
	for(int i = 1; i < MAX; i++){
		fac[i] = fac[i-1]*i%P;
	}
	inv[MAX-1] = qpow(fac[MAX-1], P-2);
	for(int i = MAX-2; i >= 0; i--){
		inv[i] = inv[i+1]*(i+1)%P;
	}
}

ll C(ll n, ll m){
	if(n < m) return 0;
	return fac[n]*inv[m]%P * inv[n-m]%P;
}

int main()
{
	init();
	cin >> n >> A >> B;
	for(int i = 1; i <= n; i++){
		read(c[i]), read(w[i]);
		cnt[c[i]]++;
		mn[i] = INF;
	}
	for(int i = 1; i <= n; i++){
		mn[c[i]] = min(mn[c[i]], w[i]);
	}
	for(int i = 1; i <= n; i++){
		if(mn[i] == INF) continue;
		if(!o1 || mn[i] < mn[o1]){
			o2 = o1;
			o1 = i;
		}
		else if(!o2 || mn[i] < mn[o2]) o2 = i;
	}
	
	for(int i = 1; i <= n; i++){
		if(mn[c[i]] != w[i] && w[i]+mn[c[i]] <= A) continue;
		if(c[i]==o1 && w[i]+mn[o2] <= B) continue;
		if(c[i]!=o1 && w[i]+mn[o1] <= B) continue;
		cnt[c[i]]--;
	}
	
	ll ans = 1, s = cnt[o1];
	for(int i = 1; i <= n; i++){
		if(i != o1 && mn[i]+mn[o1] <= B){
			s += cnt[i];
			ans = ans * C(s, cnt[i]) % P;
		}
	}
	cout << ans << endl;
	
	return 0;
}
