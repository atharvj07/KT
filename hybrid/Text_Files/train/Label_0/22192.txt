#include <bits/stdc++.h>
using namespace std;
long double n, s, e, mid, x[60005], v[60005];
bool can(long double t) {
  long double l = -1e18, r = 1e18;
  for (int i = 0; i < n; i++)
    l = max(l, x[i] - v[i] * t), r = min(r, x[i] + v[i] * t);
  return l <= r;
}
int main() {
  ios::sync_with_stdio(0);
  cout << fixed << setprecision(12);
  cin >> n;
  for (int i = 0; i < n; i++) cin >> x[i];
  for (int i = 0; i < n; i++) cin >> v[i];
  s = 0, e = 1e12;
  for (int i = 0; i < 200; i++) {
    mid = (s + e) / 2;
    if (can(mid))
      e = mid - 1e-15;
    else
      s = mid + 1e-15;
  }
  cout << s << endl;
  return 0;
}
