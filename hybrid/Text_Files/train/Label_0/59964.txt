
import java.io.*;
import java.text.DecimalFormat;
import java.util.*;
public class CF_741B {
	public static void main(String[] args) {
		new CF_741B().run();
	}
	public void run(){
		FastScanner sc = new FastScanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = sc.nextInt();
		int m = sc.nextInt();
		int w = sc.nextInt();
		
		int[] everyW = sc.readIntArray(n);
		int[] everyB = sc.readIntArray(n);
		
		DSU dsu = new DSU(n);
		//用并查集构造连通性
		for(int i = 0;i<m;i++){
			int a = sc.nextInt();
			int b = sc.nextInt();
			
			dsu.union(a-1, b-1);
		}
		int[] dp = new int[w+1];
		for(int i = 0;i<n;i++){
			if(dsu.get(i)!=i)
				continue;
			int num = 0;
			ArrayList<Integer> indexes = new ArrayList<Integer>();
			for(int j = 0;j<n;j++){
				//记录同一伙的下标
				if(dsu.get(j)==i)
					indexes.add(j);
			}
			//对每一个部落里，进行背包动态规划
			int[] next = dp.clone();
			int sumW = 0;
			int sumB = 0;
			for(int k = 0;k<indexes.size();k++){
				int index = indexes.get(k);
				int curW = everyW[index];
				int curB = everyB[index];
				sumW+=curW;
				sumB+=curB;
				
				//要么只加一个最大的
				for(int cw = w;cw>=curW;cw--){
					next[cw] = Math.max(next[cw], dp[cw-curW]+curB);
				}
			}
			//要么全部加
			for(int cw = w;cw>=sumW;cw--){
				next[cw] = Math.max(next[cw], dp[cw-sumW]+sumB);
			}
			dp = next;
		}
		out.println(dp[w]);
		out.close();
	}
}

class DSU {
	int[] p;

	public DSU(int n) {
		p = new int[n];
		clear();
	}

	public void clear() {
		for (int i = 0; i < p.length; i++) {
			p[i] = i;
		}
	}

	public int get(int x) {
		int y = x;
		while (y != p[y]) {
			y = p[y];
		}
		// 压缩
		while (x != p[x]) {
			int t = p[x];
			p[x] = y;
			x = t;
		}
		return y;
	}

	public boolean union(int a, int b) {
		a = get(a);
		b = get(b);
		p[a] = b;
		return a != b;
	}

}

class FastScanner {
	public BufferedReader br;
	public StringTokenizer st;

	public FastScanner(InputStream in) {
		br = new BufferedReader(new InputStreamReader(in));
		st = null;
	}

	public boolean hasMoreTokens() {
		while (st == null || !st.hasMoreTokens()) {
			String line = null;
			try {
				line = br.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}
			if (line == null)
				return false;
			st = new StringTokenizer(line);
		}
		return true;
	}

	public String next() {
		if (!hasMoreTokens())
			return null;
		return st.nextToken();
	}

	public int nextInt() {
		return Integer.parseInt(next());
	}

	public long nextLong() {
		return Long.parseLong(next());
	}

	//读一个int类型的数组
	public int[] readIntArray(int n) {
		int[] array = new int[n];
		for (int i = 0; i < n; i++) {
			array[i] = nextInt();
		}
		return array;
	}

	//读一个long数组
	public long[] readLongArray(int n) {
		long[] array = new long[n];
		for (int i = 0; i < n; i++) {
			array[i] = nextLong();
		}
		return array;
	}

	//读一个字符串数组
	public String[] readStringArray(int n) {
		String[] a = new String[n];
		for (int i = 0; i < n; i++) {
			a[i] = next();
		}
		return a;
	}

	//打乱一个数组，便于排序
	public void shuffleIntArray(int[] a) {
		Random rand = new Random();
		for (int i = 0; i < a.length; i++) {
			int x = rand.nextInt(i + 1); // 0~i
			int t = a[i];
			a[i] = a[x];
			a[x] = t;
		}
	}

	// 只适合比较小的数，太大的数会越界
	public long factorial(long n) {
		if (n == 0)
			return 1L;
		long res = 1;
		for (long i = 2; i <= n; i++) {
			res *= i;
		}
		return res;
	}

	// 只适合比较小的数，太大的数会越界
	public long combination(long mom, long son) {
		return factorial(mom) / (factorial(son) * factorial(mom - son));
	}

	//求最小公倍数
	public long gcd(long a, long b) {
		return b == 0 ? a : gcd(b, a % b);
	}

	//求最大公约数
	public long lcm(long a, long b) {
		return a * b / (gcd(a, b));
	}

	//格式化double类型的数
	public double format(String pattern, double d) {
		DecimalFormat df = new DecimalFormat(pattern);
		return Double.parseDouble(df.format(d));
	}

	//判断一个数是否是2的指数
	public boolean isPowerOfTwo(long v) {
		return (v & (v - 1)) == 0;
	}
}
