import sys
import re

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())

def solve():
	s = minp()
	while s == '':
		s = minp()
	(n, m) = map(int, s.split())
	a = [0] * m
	best = (-1, -1, -1)
	for i in range(m):
		(x, y) = mints()
		a[i] = (x, y)
	a.sort()
	for i in range(m):
		(x, y) = a[i]
		best = max(best, (y, x, i))
	p = [0] * (m + 1)
	for i in range(1, m + 1):
		p[i] = p[i - 1] + a[i - 1][0]
	a1 = p[m] - p[max(m - n, 0)] + max(n - m, 0) * best[0]
	if n < 1:
		print(a1)
	for ii in range(m):
		a2 = (n - 1) * a[ii][1] + a[ii][0]
		b = a[ii][1]
		(l, r) = (-1, m)
		while r - l > 1:
			c = (l + r) // 2
			if a[c][0] > b:
				r = c
			else:
				l = c
		r = max(m - (n - 1), r)
		if r <= ii:
			a2 += -(m - r) * b + p[m] - p[r] + a[ii][1] - a[ii][0]
		else:
			a2 += -(m - r) * b + p[m] - p[r]
		a1 = max(a1, a2)
	print(a1)
for i in range(mint()):
	solve()
