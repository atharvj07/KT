using namespace std;

/* constant */

const int MAX_N = 50000;
const int MAX_M = 50000;

const int NSC = 12;

const string scs[NSC] = {
  "C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"
};

/* typedef */

typedef map<string,int> msi;

/* global variables */

msi scmap;
int n, m;
int tis[MAX_N + 1], sis[MAX_M + 1];

/* subroutines */

inline int mod(int k) { return (k + NSC) % NSC; }

bool check(int ti) {
  for (int si = m - 1; si >= 0; si--) {
    if (ti < 0 || ti >= n) return false;

    int& t = tis[ti];
    int& s = sis[si];

    if (t == s) ti--;
    else if (mod(t + 1) == s) ti -= 2;
    else if (mod(t - 1) == s) ti++;
    else return false;
  }

  return (ti == -1);
}

/* main */

int main() {
  for (int i = 0; i < NSC; i++) scmap[scs[i]] = i;

  int tn;
  cin >> tn;

  while (tn--) {
    cin >> n >> m;
    string sc;
    
    for (int i = 0; i < n; i++) {
      cin >> sc;
      tis[i] = scmap[sc];
    }

    for (int i = 0; i < m; i++) {
      cin >> sc;
      sis[i] = scmap[sc];
    }

    bool ok = check(n - 1) || check(n - 2);
    cout << (ok ? "Yes" : "No") << endl;
  }

  return 0;
}
