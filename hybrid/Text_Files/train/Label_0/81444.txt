import sys
from math import *

def input():
	return sys.stdin.readline().strip()

def input_line():
	return map(int, input().split())
n = int(input())
a = list(input_line())
d = dict()
p = dict()
o = [0] * n
q = [0] * n
r = (0, 0)
for i in range(n):
	x = a[i]
	v = d.get(x - 1, 0) + 1
	r = max(r, (v, i))
	o[i] = v
	q[i] = p.get(x - 1, -1)
	d[x] = o[i]
	p[x] = i
print(r[0])
z = []
x = r[1]
while len(z) < r[0]:
	z.append(x + 1)
	x = q[x]
print(*z[::-1])
