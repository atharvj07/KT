n = int(input())
l = list(map(int, input().split()))
z = min(l)
posn = []
for i in range(n):
	if l[i] == z:
		posn.append(i)
ans = z * n
maxi = -inf
for i in range(len(l)):
	z = bisect_left(posn, i)
	if z == len(posn):
		maxi = max(n - i + posn[0], maxi)
	else:
		maxi = max(posn[z] - i, maxi)
print(ans + maxi)
