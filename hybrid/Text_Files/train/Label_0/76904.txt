import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
inin = lambda : int(input())
inar = lambda : list(map(int, input().split()))
inst = lambda : input().decode().rstrip('\n\r')
INF = float('inf')
_T_ = inin()
for _t_ in range(_T_):
	c = inar()
	odd = 0
	even = 0
	for i in c:
		if i % 2:
			odd += 1
		else:
			even += 1
	if odd <= 1:
		print('Yes')
	elif all(c[0:3]):
		for i in range(3):
			c[i] -= 1
		c[3] += 3
		odd = 0
		even = 0
		for i in c:
			if i % 2:
				odd += 1
			else:
				even += 1
		if odd <= 1:
			print('Yes')
		else:
			print('No')
	else:
		print('No')
