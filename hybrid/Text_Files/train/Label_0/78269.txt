NUMBER = 10 ** 9 + 7

def factorial(n):
	M = NUMBER
	f = 1
	for i in range(1, n + 1):
		f = f * i % M
	return f

def mult(a, b):
	return a * b % NUMBER

def minus(a, b):
	return (a - b) % NUMBER

def plus(a, b):
	return (a + b) % NUMBER

def egcd(a, b):
	if a == 0:
		return (b, 0, 1)
	else:
		(g, y, x) = egcd(b % a, a)
		return (g, x - b // a * y, y)

def modinv(a):
	m = NUMBER
	(g, x, y) = egcd(a, m)
	if g != 1:
		raise Exception('modular inverse does not exist')
	else:
		return x % m

def bootstrap(f, stack=[]):

	def wrappedfunc(*args, **kwargs):
		if stack:
			return f(*args, **kwargs)
		else:
			to = f(*args, **kwargs)
			while True:
				if type(to) is GeneratorType:
					stack.append(to)
					to = next(to)
				else:
					stack.pop()
					if not stack:
						break
					to = stack[-1].send(to)
			return to
	return wrappedfunc

def dfs(g, timeIn, timeOut, depths, parents):
	cnt = 0
	stack = [[1, 0, 0, 0]]
	while stack:
		(v, neig_i, parent, depth) = stack[-1]
		parents[v] = parent
		depths[v] = depth
		if neig_i == 0:
			timeIn[v] = cnt
			cnt += 1
		while neig_i < len(g[v]):
			u = g[v][neig_i]
			if u == parent:
				neig_i += 1
				continue
			stack[-1][1] = neig_i + 1
			stack.append([u, 0, v, depth + 1])
			break
		if neig_i == len(g[v]):
			stack.pop()
			timeOut[v] = cnt
			cnt += 1

def get_num_2_5(n):
	twos = 0
	fives = 0

		n //= 2

def bfs(adj):
	s = set()
	d = defaultdict(set)
	q = deque([])
	s.add(0)
	d[0].add(0)
	q.append((0, 0))
	while len(q) > 0:
		(v, depth) = q.popleft()
		for u in adj[v]:
			if u not in s:
				s.add(u)
				d[depth + 1].add(u)
				q.append((u, depth + 1))
	return d

def solution(adj, n, k):
	if k == 1:
		return n - 1
	if n == 2:
		return 0
	leaves = [set() for i in range(n)]
	stack = []
	for v in range(n):
		if len(adj[v]) == 1:
			leaves[adj[v].pop()].add(v)
	for v in range(n):
		if len(leaves[v]) >= k:
			stack.append(v)
	cnt = 0
	while len(stack) > 0:
		cnt += 1
		v = stack.pop()
		ls = set()
		try:
			for i in range(k):
				ls.add(leaves[v].pop())
		except:
			return f'{cnt}000{n}000{k}'
		adj[v] -= ls
		if len(adj[v]) == 0:
			continue
		elif len(adj[v]) == 1:
			u = adj[v].pop()
			leaves[u].add(v)
			if len(leaves[u]) == k:
				stack.append(u)
		elif len(leaves[v]) >= k:
			stack.append(v)
	return cnt
