factor_dict = {}

def factor(n):
	n2 = n
	if n in factor_dict:
		return factor_dict[n]
	d = {}
	for i in range(2, n):
		if i * i > n:
			break
		if n % i == 0:
			c = 0
			while n % i == 0:
				c += 1
				n = n // i
			d[i] = c
	if n > 1:
		d[n] = 1
	factor_dict[n2] = d
	return d

def process(A):
	primes = {}
	answer = 1
	n = len(A)
	for x in A:
		d = factor(x)
		for p in d:
			if p not in primes:
				primes[p] = []
			primes[p].append(d[p])
	for p in primes:
		if len(primes[p]) == n - 1:
			answer = answer * p ** min(primes[p])
		elif len(primes[p]) == n:
			mins = [float('inf'), 0]
			for i in range(n):
				entry = [primes[p][i], i]
				mins = min(entry, mins)
			mins2 = [float('inf'), 0]
			for i in range(n):
				if i != mins[1]:
					entry = [primes[p][i], i]
					mins2 = min(entry, mins2)
			answer = answer * p ** mins2[0]
	return answer
n = int(input())
A = [int(x) for x in input().split()]
print(process(A))
