using namespace std;


struct Info{
	double x,y,z,dist;
};

int N;
Info info[500];
bool check[500];

double calc_dist(Info info){
	return sqrt(info.x*info.x+info.y*info.y+info.z*info.z);
}

void func(){

	for(int i = 0; i < N; i++){
		scanf("%lf %lf %lf",&info[i].x,&info[i].y,&info[i].z);
		info[i].dist = calc_dist(info[i]);
	}

	for(int i = 0; i < N; i++)check[i] = false;

	int M;
	scanf("%d",&M);

	Info scope;
	double degree,degree2;

	for(int loop = 0; loop < M; loop++){
		scanf("%lf %lf %lf %lf",&scope.x,&scope.y,&scope.z,&degree);
		scope.dist = calc_dist(scope);

		for(int i = 0; i < N; i++){
			degree2 = acos((info[i].x*scope.x+info[i].y*scope.y+info[i].z*scope.z)/(info[i].dist*scope.dist));
			if(degree2 <= degree){
				check[i] = true;
			}
		}
	}

	int ans = 0;
	for(int i = 0; i < N; i++){
		if(check[i])ans++;
	}

	printf("%d\n",ans);
}


int main(){

	while(true){
		scanf("%d",&N);
		if(N == 0)break;

		func();
	}

	return 0;
}
