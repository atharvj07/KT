INF = 1000000000000.0
mx = [-1, -1, 0, 1, 1, 1, 0, -1]
my = [0, 1, 1, 1, 0, -1, -1, -1]

def canSolve(x, y, j):
	if mx[j] == 0:
		return x == 0 and y // my[j] > 0
	elif my[j] == 0:
		return x // mx[j] > 0 and y == 0
	else:
		return y // my[j] == x // mx[j] and y // my[j] > 0

def main():
	global INF
	n = int(input())
	(x0, y0) = map(int, input().split())
	close = [(INF, 'a')] * 8
	for i in range(n):
		(piece, x, y) = input().split()
		x = int(x) - x0
		y = int(y) - y0
		for j in range(8):
			if canSolve(x, y, j):
				aux = abs(x * mx[j] + y * my[j])
				if aux < close[j][0]:
					close[j] = (aux, piece)
	can = False
	for i in range(8):
		if i % 2 == 0 and (close[i][1] == 'R' or close[i][1] == 'Q'):
			can = True
			break
		elif i % 2 != 0 and (close[i][1] == 'B' or close[i][1] == 'Q'):
			can = True
			break
	if can:
		print('YES')
	else:
		print('NO')
main()
