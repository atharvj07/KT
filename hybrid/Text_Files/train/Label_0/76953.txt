MOD = 998244353
(N,) = map(int, input().split())
X = [0] + list(map(int, input().split()))
Y = [0] * (N + 1)
stack = [(-1, 0)]
for i in range(1, N + 1):
	while stack and stack[-1][0] > X[i]:
		stack.pop()
	Y[i] = stack[-1][1]
	stack.append((X[i], i))
dp = [0] * (N + 1)
dp[0] = 1
S = [[0, 0] for _ in range(N + 1)]
for i in range(1, N + 1):
	x = X[i]
	k = Y[i]
	if i % 2:
		dp[i] = (S[i - 1][0] - S[k][0]) * x
	else:
		dp[i] = (S[i - 1][1] - S[k][1]) * x
	if k > 0:
		dp[i] += pow(-1, i - k + 1) * dp[k] * (x - 1)
	else:
		dp[i] += pow(-1, i - k + 1) * (dp[k] * x)
	dp[i] %= MOD
	S[i][0] += S[i - 1][0] + pow(-1, i) * dp[i]
	S[i][1] += S[i - 1][1] + pow(-1, i + 1) * dp[i]
	S[i][0] %= MOD
	S[i][1] %= MOD
print(dp[-1] % MOD)
