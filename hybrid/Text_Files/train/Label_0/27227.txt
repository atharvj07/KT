#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 500;
int n, m;
int dp[maxn];
int L[maxn], R[maxn];
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    int x, s;
    scanf("%d", &x), scanf("%d", &s);
    L[i] = max(1, x - s);
    R[i] = min(x + s, m);
  }
  dp[m + 1] = 0;
  for (int i = 1; i <= m; i++) dp[i] = m - i + 1;
  for (int i = m; i >= 1; i--) {
    for (int j = 1; j <= n; j++) {
      if (L[j] <= i && i <= R[j]) {
        dp[i] = min(dp[i], dp[i + 1]);
      } else if (i < L[j]) {
        dp[i] = min(dp[i], dp[min(R[j] + L[j] - i, m) + 1] + L[j] - i);
      }
    }
  }
  cout << dp[1] << endl;
}
