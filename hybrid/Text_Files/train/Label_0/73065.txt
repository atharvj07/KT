import math

def f(m):
	global a, b, x, y
	f = m // (a * b // math.gcd(a, b))
	w = m // a - f
	e = m // b - f
	t = 0
	for i in range(f):
		t += x * p[i] // 100
		t += y * p[i] // 100
	for i in range(f, f + w):
		t += x * p[i] // 100
	for i in range(w + f, e + w + f):
		t += y * p[i] // 100
	return t
q = int(input())
for i in range(q):
	n = int(input())
	p = [int(i) for i in input().split()]
	(x, a) = [int(i) for i in input().split()]
	(y, b) = [int(i) for i in input().split()]
	k = int(input())
	p.sort()
	p = p[::-1]
	l = 0
	r = n
	if y > x:
		(x, y) = (y, x)
		(a, b) = (b, a)
	if f(r) < k:
		print(-1)
	else:
		while r - l > 1:
			m = (l + r) // 2
			if f(m) < k:
				l = m
			else:
				r = m
		print(r)
