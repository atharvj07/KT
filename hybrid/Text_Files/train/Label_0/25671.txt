#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, M = 1e6 + 7;
;
struct Hash_Tab {
  unsigned long long K[N];
  pair<long long, int> V[N];
  int nxt[N], fir[M], tot;
  Hash_Tab() {
    tot = 0;
    memset(fir, 0, sizeof fir);
  }
  inline pair<long long, int> operator[](unsigned long long key) {
    for (int i = fir[key % M]; i; i = nxt[i])
      if (K[i] == key) return V[i];
    return make_pair(0LL, 0);
  }
  inline void Upd(unsigned long long key, pair<long long, int> val) {
    for (int i = fir[key % M]; i; i = nxt[i])
      if (K[i] == key) return (void)(V[i] = val);
    nxt[++tot] = fir[key % M];
    fir[key % M] = tot;
    K[tot] = key;
    V[tot] = val;
  }
} T;
int n, m;
unsigned long long z, a[N], b[N], c[N], d[N];
inline unsigned long long ran() {
  return ((unsigned long long)rand()) + (((unsigned long long)rand()) << 16) +
         (((unsigned long long)rand()) << 32) +
         (((unsigned long long)rand()) << 48);
}
inline void Ins(int x) {
  pair<long long, int> tmp = T[d[x + 1] ^ c[x + 1] ^ z];
  tmp.first += x;
  tmp.second++;
  T.Upd(d[x + 1] ^ c[x + 1] ^ z, tmp);
}
pair<long long, int> tmp;
long long ans, id;
int l, r, cnt;
int main() {
  register int i, j;
  srand(19260817);
  scanf("%d %d", &n, &m);
  for (i = 1; i <= n; i++) {
    scanf("%d %d", &l, &r);
    id = ran();
    a[l] ^= id;
    a[r + 1] ^= id;
    d[l] ^= id;
  }
  for (i = 1; i <= m; i++) {
    b[i] = b[i - 1] ^ a[i];
    c[i] = c[i - 1] ^ b[i];
    if (!b[i]) cnt++;
    if (i == m || b[i]) {
      for (j = 1; j <= cnt; j++) ans -= 1LL * j * (cnt + 1 - j);
      cnt = 0;
    }
  }
  Ins(0);
  for (i = 1; i <= m; i++) {
    z ^= d[i];
    tmp = T[c[i] ^ z];
    ans += 1LL * tmp.second * i - tmp.first;
    Ins(i);
  }
  printf("%lld", ans);
  return 0;
}
