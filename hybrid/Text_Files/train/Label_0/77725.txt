def main():
	from array import array
	from sys import stdout, stdin
	n = int(stdin.readline())
	zero = array('i', (0,))
	ns = zero * 1000001
	ks = zero * n
	inp = [array('i')] * n
	for i in range(n):
		(k, a) = stdin.readline().split(maxsplit=1)
		a = array('i', map(int, a.split()))
		for ai in a:
			ns[ai] += 1
		ks[i] = int(k)
		inp[i] = a
	ans = 0
	for i in range(n):
		cur = 0
		for inp_i in inp[i]:
			cur = (cur + ns[inp_i]) % 998244353
		ans = (ans + pow(ks[i], 998244351, 998244353) * cur % 998244353) % 998244353
	stdout.write(f'{pow(n, 998244350, 998244353) * ans % 998244353}')
main()
