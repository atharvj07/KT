import java.math.*;
import java.util.*;
 
public class Main {
	public static BigDecimal[][] C = new BigDecimal[110][];
	public static int[] a = new int[110];
	public static BigDecimal[][] dp = new BigDecimal[110][];
		
	public static void main(String[] args) {
		for (int i = 0; i < 110; i++) C[i] = new BigDecimal [110];
		for (int i = 0; i < 110; i++) dp[i] = new BigDecimal [11000];
		for (int i = 0; i < 110; i++) for (int j = 0; j < 110; j++) C[i][j] = BigDecimal.valueOf(0);
		for (int i = 0; i < 110; i++) for (int j = 0; j < 11000; j++) dp[i][j] = BigDecimal.valueOf(0);
		Scanner scanner = new Scanner(System.in);
		for (int i = 0; i < 110; i++) {
			for (int j = 0; j <= i; j++) {
				if (j == 0) {
					C[i][j] = BigDecimal.valueOf(1);
					continue;
				}
				C[i][j] = C[i - 1][j - 1].add(C[i - 1][j]);
			}
		}
		int n, x;
		n = scanner.nextInt();
		x = scanner.nextInt();
		for (int i = 0; i < n; i++) a[i] = scanner.nextInt();
		dp[0][0] = BigDecimal.valueOf(1);
		for (int i = 0; i < n; i++) {
			for (int j = n; j >= 0; j--) for (int k = 0; k <= 10000; k++) if (dp[j][k] != BigDecimal.valueOf(0)) {
				dp[j + 1][k + a[i]] = dp[j + 1][k + a[i]].add(dp[j][k]);
			}
		}
		BigDecimal ans = BigDecimal.valueOf(x);
		for (int i = 2; i <= n; i++) {
			int m = i - 1;
			BigDecimal cur = BigDecimal.valueOf(0);
			BigDecimal cnt = BigDecimal.valueOf(x * (2 * n - m)).divide(BigDecimal.valueOf(2 * (n - m)), 20, RoundingMode.UP);
			for (int j = 0; j <= 10000; j++) {
				BigDecimal now = BigDecimal.valueOf(j).divide(BigDecimal.valueOf(n + 1 - i), 20, RoundingMode.UP);
				if (cnt.compareTo(now) < 0) now = cnt;
				cur = cur.add(now.multiply(dp[n + 1 - i][j]));
			}
			cur = cur.divide(C[n][m], 20, RoundingMode.UP);
			ans = ans.add(cur);
		}
		System.out.println(ans);
		scanner.close();
	}
}
