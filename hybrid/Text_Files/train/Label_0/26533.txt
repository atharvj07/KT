using namespace std;
const int maxn = 2e5 + 10;
const long long mod = 1e9 + 7;
inline int inverse(int a) {
  return a <= 1 ? 1 : (long long)(mod - mod / a) * inverse(mod % a) % mod;
}
int gcd(int a, int b) { return b ? gcd(b, a % b) : a; }
long long fsp(long long a, long long b) {
  long long ans = 1;
  for (; b; a = a * a % mod, b >>= 1)
    if (b & 1) ans = ans * a % mod;
  return ans % mod;
}
inline void read(int &x) {
  x = 0;
  int f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch <= '9' && ch >= '0') {
    x = x * 10 + ch - '0';
    ch = getchar();
  }
}
struct segtree {
  long long f[maxn * 4], mn[maxn * 4];
  void push_up(int rt) {
    int ls = rt << 1, rs = rt << 1 | 1;
    mn[rt] = min(mn[ls], mn[rs]);
    f[rt] = f[ls] + f[rs];
  }
  void build(int rt, int l, int r) {
    if (l == r) {
      f[rt] = mn[rt] = l;
      return;
    }
    int m = (l + r) / 2, ls = rt << 1, rs = rt << 1 | 1;
    build(ls, l, m);
    build(rs, m + 1, r);
    push_up(rt);
  }
  void push_down(int rt, int l, int r, int val) {
    if (mn[rt] >= val) return;
    if (l == r) {
      mn[rt] = f[rt] = val;
      return;
    }
    int m = (l + r) / 2, ls = rt << 1, rs = rt << 1 | 1;
    if (mn[ls] < val) push_down(ls, l, m, val);
    if (mn[rs] < val) push_down(rs, m + 1, r, val);
    push_up(rt);
  }
  void change(int rt, int l, int r, int L, int R, int val) {
    if (l >= L && r <= R) {
      push_down(rt, l, r, val);
      return;
    }
    int m = (l + r) / 2, ls = rt << 1, rs = rt << 1 | 1;
    if (m >= L) change(ls, l, m, L, R, val);
    if (m < R) change(rs, m + 1, r, L, R, val);
    push_up(rt);
  }
} st;
long long h[maxn];
int n, a[maxn];
vector<int> v[maxn], k[maxn];
void work() {
  st.build(1, 1, n);
  memset(h, 0, sizeof(h));
  cerr << "st.f[1]"
       << "   " << st.f[1] << endl;
  ;
  int mx = 0;
  for (int i = 1; i <= n; ++i) {
    int x;
    scanf("%d", &x);
    mx = max(mx, x);
    a[i] = x;
    for (auto it : v[x]) k[it].push_back(i);
  }
  h[mx + 1] = 1LL * (n + 1) * n - st.f[1];
  for (int i = mx; i >= 1; i--) {
    if (k[i].size() >= 2) {
      int sz = (int)k[i].size();
      st.change(1, 1, n, 1, k[i][0], k[i][sz - 2]);
      st.change(1, 1, n, k[i][0] + 1, k[i][1], k[i][sz - 1]);
      if (k[i][1] + 1 <= n) st.change(1, 1, n, k[i][1] + 1, n, n + 1);
    }
    h[i] = 1LL * (n + 1) * n - st.f[1];
  }
  long long ans = 0;
  for (int i = 1; i <= mx; ++i) {
    ans += 1LL * (h[i + 1] - h[i]) * i;
  }
  cout << ans << endl;
}
int main() {
  for (int i = 1; i <= 200000; ++i)
    for (int j = i; j <= 200000; j += i) v[j].push_back(i);
  while (~scanf("%d", &n)) work();
  return 0;
}
