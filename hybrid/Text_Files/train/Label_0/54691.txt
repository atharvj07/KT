import java.util.Scanner;

public class p266b {
	public static void main(String args[]) {
		Scanner in = new Scanner(System.in);
		int N = in.nextInt();
		int T = in.nextInt();
		char[] arr = in.next().toCharArray();
		for (int t = 0; t < T; t++) {
			for (int i = 0; i+1 < N; i++) {
				if (arr[i + 1] == 'G') {
					char tmp = arr[i];
					arr[i] = arr[i + 1];
					arr[i + 1] = tmp;
					i++;
				}
			}
		}

		System.out.println(new String(arr));
	}
}
