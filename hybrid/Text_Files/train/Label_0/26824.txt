#include <bits/stdc++.h>
using namespace std;
void solve() {
  long long int n, a, b, k;
  cin >> n >> a >> b >> k;
  vector<long long int> v(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  long long int ans = 0;
  priority_queue<int, vector<int>, greater<int>> q;
  for (int i = 0; i < n; i++) {
    long long int z = v[i] % (a + b);
    if (z > 0 && z <= a) {
      ans++;
    } else if (z == 0) {
      v[i] = b;
      long long int x = (v[i] + a - 1) / a;
      q.push(x);
    } else {
      v[i] = (v[i] % (a + b));
      v[i] -= a;
      long long int x = (v[i] + a - 1) / a;
      q.push(x);
    }
  }
  while (!q.empty()) {
    if (k < q.top())
      break;
    else {
      k -= q.top();
      ans++;
      q.pop();
    }
  }
  cout << ans << endl;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t = 1;
  while (t--) {
    solve();
  }
  return 0;
}
