(r, c) = map(int, input().split())
matrix = [list(input()) for _ in range(r)]
output = True
for i in range(r):
	for j in range(c):
		if matrix[i][j] == 'S':
			if i > 0:
				if matrix[i - 1][j] == '.':
					matrix[i - 1][j] = 'D'
				elif matrix[i - 1][j] == 'W':
					output = False
					break
			if i < r - 1:
				if matrix[i + 1][j] == '.':
					matrix[i + 1][j] = 'D'
				elif matrix[i + 1][j] == 'W':
					output = False
					break
			if j > 0:
				if matrix[i][j - 1] == '.':
					matrix[i][j - 1] = 'D'
				elif matrix[i][j - 1] == 'W':
					output = False
					break
			if j < c - 1:
				if matrix[i][j + 1] == '.':
					matrix[i][j + 1] = 'D'
				if matrix[i][j + 1] == 'W':
					output = False
					break
if output:
	print('Yes')
	for i in range(r):
		out = ''
		for j in range(c):
			out += matrix[i][j]
		print(out)
else:
	print('No')
