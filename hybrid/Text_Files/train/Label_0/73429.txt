q = int(input())
l = [list(map(int, input().split())) for _ in range(q)]
k = []
if q == 3:
	for j in l[0]:
		for n in l[1]:
			for h in l[2]:
				k.append(h)
				k.append(n)
				k.append(j)
				k.append(int(str(h) + str(n)))
				k.append(int(str(n) + str(h)))
				k.append(int(str(j) + str(h)))
				k.append(int(str(h) + str(j)))
				k.append(int(str(j) + str(n)))
				k.append(int(str(n) + str(j)))
elif q == 2:
	for j in l[0]:
		for n in l[1]:
			k.append(n)
			k.append(j)
			k.append(int(str(j) + str(n)))
			k.append(int(str(n) + str(j)))
else:
	for j in l[0]:
		k.append(j)
for n in range(1, 100):
	if n not in k:
		print(n - 1)
		quit()
