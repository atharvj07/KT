from math import hypot
(y1, y2, yw, xb, yb, r) = map(int, input().split())
yw -= r
(y1, y2) = (yw * 2 - y2, yw * 2 - y1)

def xww(y):
	return (y - yw) * xb / (y - yb)

def dd(y):
	xw = xww(y)
	return (y - y1) / hypot(1, (yw - y) / xw)

def binary_search():
	(a, b) = (y1 + r, 10000000000.0)
	for i in range(200):
		m = (a + b) / 2
		if dd(m) < r:
			a = m
		else:
			b = m
	return (a + b) / 2
m = binary_search()
if m + r > y2:
	print('-1')
else:
	m = (m + y2 - r) / 2
	print(xww(m))
