(a, b, c) = map(int, input().split())
(z, z1, z2) = (a // 3, b // 2, c // 2)
k = min(z, z1, z2)
ans = 7 * k
a -= 3 * k
b -= 2 * k
c -= 2 * k
d = [0, 1, 2, 0, 2, 1, 0]
ma = 0
for i in range(7):
	k = [a, b, c]
	j = i
	while k[d[j % 7]]:
		k[d[j % 7]] -= 1
		j += 1
	ma = max(ma, j - i)
print(ans + ma)
