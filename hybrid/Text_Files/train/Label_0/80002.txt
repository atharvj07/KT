def pho(n, k, a):
	aset = set(a)
	if len(aset) > k:
		print(-1)
	elif len(aset) == k:
		print(n * k)
		aset = list(aset)
		for i in range(n):
			for j in range(k):
				print(aset[j], end=' ')
		print()
	else:
		ns = []
		for x in range(1, k + 1):
			ns.append(x)
		q = 0
		while len(aset) != k:
			aset.add(ns[q])
			q += 1
		aset = list(aset)
		print(n * k)
		for i in range(n):
			for j in range(k):
				print(aset[j], end=' ')
		print()
t = int(input())
l = []
for i in range(t):
	(n, k) = list(map(int, input().split()))
	l1 = list(map(int, input().split()))
	l.append([n, k, l1])
for j in l:
	pho(j[0], j[1], j[2])
