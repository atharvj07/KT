#include <bits/stdc++.h>
using namespace std;
int N, K, C, *arr;
int presents() {
  int sum = 0, the_last_one = 0, j = 0;
  for (int i = 1; i <= N; i++) {
    if (i == arr[j]) {
      the_last_one = i;
      sum++;
      j++;
    } else if (i == the_last_one + K) {
      the_last_one = i;
      sum++;
    }
  }
  return sum;
}
int main() {
  cin >> N >> K;
  cin >> C;
  arr = new int[C];
  for (int i = 0; i < C; i++) {
    cin >> arr[i];
  }
  cout << presents();
}
