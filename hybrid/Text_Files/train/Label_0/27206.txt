//あみだくじは「置換」で表せる。これは6通りしかないので、実質N ≦ 6。よって全部試せる。
//同じ「置換」を表すあみだくじを2つ以上使うケースは存在するか？　→ 存在するかもしれない…。    
//嘘解法？：同じあみだくじを4つ以上使うケースがないものとして考える。　→　書いてみよう
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

int n;
int w, a;
map<vector<int>, int> mp;
vector<vector<int>> ts;

bool dfs(int used, vector<int> perm) {
	int i, j;
	
	if (used > 0 && perm[0] == 0 && perm[1] == 1 && perm[2] == 2) return true;
	for (i = 0; i < ts.size(); i++) {
		if ((used >> i) & 1) continue;
		
		vector<int> nP;
		for (j = 0; j < 3; j++) {
			nP.push_back(ts[i][perm[j]]);
		}
		bool res = dfs(used + (1 << i), nP);
		if (res) return true;
	}
	return false;
}

int main() {
	int i, j;
	
	cin >> n;
	for (i = 0; i < n; i++) {
		cin >> w;
		int hoge[3] = {0, 1, 2};
		for (j = 0; j < w; j++) {
			cin >> a;
			swap(hoge[a], hoge[a + 1]);
		}
		
		vector<int> tikan;
		tikan.resize(3);
		for (j = 0; j < 3; j++) tikan[hoge[j]] = j;
		if (mp.find(tikan) == mp.end()) mp[tikan] = 0;
		mp[tikan]++;
	}
	
	for (map<vector<int>, int>::iterator it = mp.begin(); it != mp.end(); it++) {
		for (i = 0; i < min((it->second), 3); i++) {
			ts.push_back(it->first);
		}
	}
	
	//for (i = 0; i < ts.size(); i++) { for (j = 0; j < 3; j++) cout << ts[i][j] << " " ; cout << endl; }
	bool res = dfs(0, {0, 1, 2});
	if (res) cout << "yes" << endl;
	else cout << "no" << endl;
	return 0;
}
