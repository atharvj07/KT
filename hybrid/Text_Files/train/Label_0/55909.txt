import java.util.*;
import java.io.*;
import java.lang.*;

public class CF1433A {
	public static void main (String[] args) {
		Scanner sc = new Scanner (System.in);
		int t = sc.nextInt();
		while (t-- > 0) {
			int x = sc.nextInt();
			int d = x%10 - 1;
			int y = 0;
			while (x!=0) {
				x = x - (x%10);
				x/=10;
				y++;
			}
			System.out.println(((y)*(y+1))/2 + 10 * d);
		}
	}
}
