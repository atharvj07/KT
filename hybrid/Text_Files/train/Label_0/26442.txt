#include <bits/stdc++.h>

using namespace std;

int main()
{
  string s;
  cin >> s;
  int r = 0;
  int st = 0;
  int en = s.length()-1;
  while (st < en) {
    if (s[st] == s[en]) {
      st++;
      en--;
    } else if (s[st] == 'x') {
      st++;
      r++;
    } else if (s[en] == 'x') {
      en--;
      r++;
    } else {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << r << endl;
}
