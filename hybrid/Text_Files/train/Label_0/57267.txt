import java.io.*;
import java.util.*;

/*
procrastinating
*/

public class Main {

    static FastReader in;
    static PrintWriter out;
    static Random rand = new Random();
    static final int INF = (int) (1e9 + 10), MOD = (int) (1e9 + 7), LOGN = 60;
    static final long IINF = (long) (2e18 + 10);

    static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    static void solve() {
        int a = in.nextInt();
        int b = in.nextInt();
        int q = in.nextInt();

        int lcm = a * b / gcd(a, b);
        int[] arr = new int[lcm];
        for (int i = 0; i < lcm; i++) {
            if ((i % a) % b != (i % b) % a) {
                arr[i] = 1;
            }
        }
        int[] pre = new int[lcm];
        pre[0] = arr[0];
        for (int i = 1; i < lcm; i++) {
            pre[i] = pre[i - 1] + arr[i];
        }
        int sum = pre[lcm - 1];

        for (int i = 0; i < q; i++) {
            long l = in.nextLong();
            long r = in.nextLong();
            long l1 = l / lcm;
            int l2 = (int) (l % lcm);
            long r1 = r / lcm;
            int r2 = (int) (r % lcm);

            if (l1 == r1) {
                out.print(pre[r2] - (l2 == 0 ? 0 : pre[l2 - 1]));
            } else {
                out.print(sum - (l2 == 0 ? 0 : pre[l2 - 1]) + pre[r2] + (long) sum * (r1 - l1 - 1));
            }
            out.print(' ');
        }
        out.println();

    }

    public static void main(String[] args) throws FileNotFoundException, InterruptedException {
        in = new FastReader(System.in);
//        in = new FastReader(new FileInputStream("sometext.txt"));
        out = new PrintWriter(System.out);
//        out = new PrintWriter(new FileOutputStream("output.txt"));

        Thread thread = new Thread(null, () -> {
            int tests = 1;
            tests = in.nextInt();

            while (tests-- > 0) {
                solve();
            }
        }, "Go", 1 << 28);
        thread.start();
        thread.join();

//        out.flush();
        out.close();
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        Integer nextInt() {
            return Integer.parseInt(next());
        }

        Long nextLong() {
            return Long.parseLong(next());
        }

        Double nextDouble() {
            return Double.parseDouble(next());
        }

        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }

        String nextLine() {
            String s = "";
            try {
                s = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return s;
        }
    }
}
