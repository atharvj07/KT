import sys
from sys import stdout
input = lambda : sys.stdin.readline().rstrip('\r\n')
P = lambda : list(map(int, input().split()))
from math import factorial as f
from collections import deque, Counter as C
from heapq import heapify, heappop, heappush, heappushpop, heapreplace, merge
mod = 10 ** 9 + 7
a = ord('A')
tc = int(input())
for _ in range(tc):
	(n, m) = P()
	lo = P()
	d = input().split()
	odd = []
	even = []
	for (a, b) in zip(lo, d):
		if a % 2:
			odd.append((a, b))
		else:
			even.append((a, b))
	dist = {}
	odd.sort()
	even.sort()
	ls = [odd, even]
	for lll in ls:
		l = []
		r = []
		for (a, b) in lll:
			if b == 'R':
				r.append(a)
			elif r:
				x = r.pop()
				dist[x] = dist[a] = (a - x) // 2
			else:
				l.append(a)
		for i in range(len(r) // 2):
			(a, b) = (r.pop(), r.pop())
			dist[a] = dist[b] = m - b + (b - a) // 2
		l = l[::-1]
		for i in range(len(l) // 2):
			(a, b) = (l.pop(), l.pop())
			dist[a] = dist[b] = a + (b - a) // 2
		while l and r:
			(a, b) = (l.pop(), r.pop())
			x = max(a, m - b)
			a1 = abs(a - x)
			b1 = b + x
			if b1 > m:
				b1 = m - (b1 - m)
			dist[a] = dist[b] = x + (b1 - a1) // 2
	for i in lo:
		print(dist.get(i, -1), end=' ')
	print()
