from bisect import bisect_left
from functools import cmp_to_key

def mp():
	return map(int, input().split())

def lt():
	return list(map(int, input().split()))

def pt(x):
	print(x)

def ip():
	return input()

def it():
	return int(input())

def sl(x):
	return [t for t in x]

def spl(x):
	return x.split()

def aj(liste, item):
	liste.append(item)

def bin(x):
	return '{0:b}'.format(x)
n = it()
visited = {}
E = {}
for i in range(n):
	(a, b) = mp()
	visited[a, b] = False
	if b - a in E:
		r = bisect_left(E[b - a], [a, b])
		if r == len(E[b - a]):
			E[b - a].append([a, b])
		else:
			E[b - a].insert(r, [a, b])
	else:
		E[b - a] = [[a, b]]
L = lt()[::-1]
res = []
for i in L:
	if not i in E or E[i] == []:
		bool = False
		break
	else:
		r = E[i].pop()
		(x, y) = (r[0], r[1])
		if (not (x + 1, y) in visited or visited[x + 1, y]) and (not (x, y + 1) in visited or visited[x, y + 1]):
			visited[x, y] = True
			res.append([x, y])
		else:
			bool = False
			break
res = res[::-1]
if bool:
	print('YES')
	for i in res:
		print(' '.join([str(x) for x in i]))
else:
	print('NO')
