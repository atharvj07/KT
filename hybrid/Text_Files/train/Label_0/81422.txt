def main():
	n = int(input())
	a = list(map(int, input().split()))
	if n == 1:
		print(n, n)
		print(-a[0])
		print(n, n)
		print(a[0])
		print(n, n)
		print(-a[0])
		return
	print(1, n)
	for i in a:
		print(-(i * n), end=' ')
	print()
	print(1, n - 1)
	for i in a:
		i -= i * n
	for i in range(n - 1):
		print(a[i] * (n - 1), end=' ')
	print()
	print(n, n)
	print(a[-1] * (n - 1))
main()
