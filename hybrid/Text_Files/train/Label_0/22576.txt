#include <bits/stdc++.h>
using namespace std;
long long a[301010];
long long b[500550], c[405060];
void solve() {
  int n, m;
  cin >> n >> m;
  string s;
  for (int i = 0; 20005 > i; i++) {
    a[i] = 0;
    b[i] = 0;
    c[i] = 0;
  }
  cin >> s;
  long long ans = 0;
  for (int i = 1; m >= i; i++) {
    cin >> a[i];
    c[a[i]]--;
    c[0]++;
  }
  c[0]++;
  for (int i = 0; n > i; i++) {
    ans += c[i];
    int r = s[i] - 'a';
    b[r] += ans;
  }
  for (int i = 0; 26 > i; i++) {
    cout << b[i] << " ";
  }
  cout << endl;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
