import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.StringTokenizer;


public class VKRound3A {
    static BufferedReader br = new BufferedReader(new InputStreamReader(
            System.in));
    static StringTokenizer st = new StringTokenizer("");
    static int nextInt() throws Exception {
        return Integer.parseInt(next());
    }
    static String next() throws Exception {
        while (true) {
            if (st.hasMoreTokens()) {
                return st.nextToken();
            }
            String s = br.readLine();
            if (s == null) {
                return null;
            }
            st = new StringTokenizer(s);
        }
    }
    static int n, m;
    static int state [];
    static ArrayList<Integer> graph1 [];
    static ArrayList<Integer> graph2 [];
    static boolean vis1 [], vis2 [];
    public static void dfs_source(int idx){
        vis1[idx] = true;
        for(Integer nxt : graph1[idx]){
            if(!vis1[nxt] && state[nxt] != 1)
                dfs_source(nxt);
        }
    }
    
    public static void dfs_dest(int idx){
        vis2[idx] = true;
        if(state[idx] == 1)
            return;
        for(Integer nxt : graph2[idx]){
            if(!vis2[nxt])
                dfs_dest(nxt);
        }
    }
    
    @SuppressWarnings("unchecked")
    public static void main(String[] args)throws Exception {
        n = nextInt();
        m = nextInt();
        state = new int [n + 1];
        vis1 = new boolean [n + 1];
        vis2 = new boolean [n + 1];
        graph1 = new ArrayList [n + 1];
        graph2 = new ArrayList [n + 1];
        for(int i = 0 ; i < n + 1 ; ++i)
            graph1[i] = new ArrayList<Integer>();
        for(int i = 0 ; i < n + 1 ; ++i)
            graph2[i] = new ArrayList<Integer>();
        for(int i = 1 ; i < n + 1 ; ++i)
            state[i] = nextInt();
        for(int i = 0 ; i < m ; ++i){
            int a = nextInt();
            int b = nextInt();
            graph1[a].add(b);
            graph2[b].add(a);
        }
        for(int i = 1 ; i < n + 1 ; ++i){
            if(state[i] == 1 && !vis1[i])
                dfs_source(i);
        }       
        for(int i = 1 ; i < n + 1 ; ++i){
            if(state[i] == 2 && !vis2[i])
                dfs_dest(i);
        }
        for(int i = 1 ; i < n + 1 ; ++i){
            if(vis1[i] && vis2[i])
                System.out.println(1);
            else
                System.out.println(0);
        }
    }
}
