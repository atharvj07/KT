voc = {'a': 'b', 'b': 'a'}

def ab(s):
	return s.count('ab')

def ba(s):
	return s.count('ba')

def change_first(s):
	return voc[s[0]] + s[1:]

def change_last(s):
	return voc[s[-1]] + s[:-1]
n = int(input())
list = []
for i in range(n):
	list.append(input())
for item in list:
	if ab(item) != ba(item):
		ch = change_first(item)
		if ab(ch) == ba(ch):
			print(ch)
		else:
			print(change_last(item))
	else:
		print(item)
