def solve(n, k):
	res = []
	if n % 2 == 0 and k % 2 == 0:
		if n < k:
			return res
		elif n < 2 * k:
			res.extend([1] * (k - 1))
			res.append(n - 1 * (k - 1))
			return res
		else:
			res.extend([2] * (k - 1))
			res.append(n - 2 * (k - 1))
			return res
	elif n % 2 == 0 and k % 2 != 0:
		if n < 2 * k:
			return res
		else:
			res.extend([2] * (k - 1))
			res.append(n - 2 * (k - 1))
			return res
	elif n % 2 == 1 and k % 2 == 0:
		return res
	elif n < k:
		return res
	else:
		res.extend([1] * (k - 1))
		res.append(n - 1 * (k - 1))
		return res
t = int(input())
for i in range(0, t):
	(n, k) = map(int, input().split())
	res = solve(n, k)
	if len(res) == 0:
		print('NO')
	else:
		print('YES')
		print(' '.join(map(str, res)))
