#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, temp, count = 0;
  int a[5] = {0};
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> temp;
    a[temp]++;
  }
  count += a[4];
  a[1] -= a[3];
  if (a[1] < 0) a[1] = 0;
  count += a[3];
  if (a[2] % 2 == 0) {
    count += (a[2] / 2);
    if (a[1] % 4)
      count += a[1] / 4 + 1;
    else
      count += a[1] / 4;
  } else {
    count += (a[2] / 2);
    count++;
    a[1] -= 2;
    if (a[1] < 0) a[1] = 0;
    if (a[1] % 4)
      count += a[1] / 4 + 1;
    else
      count += a[1] / 4;
  }
  cout << count << endl;
  return 0;
}
