#include <bits/stdc++.h>
using namespace std;
long long n, k, cnt[200100], a[200100], ans[200100], c = 1e9 + 7;
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    cnt[a[i]]++;
  }
  sort(a, a + n);
  for (int i = 0; i < n; i++) {
    if (cnt[a[i]] >= k) c = min(0LL, c);
  }
  for (int i = 0; i < n; i++) {
    if (cnt[a[i]] >= k) c = min(ans[a[i]], c);
    int j;
    for (j = 1; j < 30; j++) {
      if ((a[i] / (1 << j)) <= 0) break;
      cnt[a[i] / (1 << j)]++;
      ans[a[i] / (1 << j)] += j;
      if (cnt[a[i] / (1 << j)] >= k) {
        c = min(ans[a[i] / (1 << j)], c);
      }
    }
    cnt[0]++;
    ans[0] += j;
    if (cnt[0] >= k) c = min(ans[0], c);
  }
  cout << c << endl;
  return 0;
}
