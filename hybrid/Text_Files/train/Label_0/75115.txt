n = int(input())
l = 0
r = 9
stat = [0] * 10
a = input().strip()
for i in range(0, n):
	if a[i] == 'L':
		stat[l] = 1
		l = 0
		while stat[l] == 1:
			l += 1
			if l == 10:
				break
		r = 9
		while stat[r] == 1:
			r -= 1
			if r == -1:
				break
	elif a[i] == 'R':
		stat[r] = 1
		r = 9
		while stat[r] == 1:
			r -= 1
			if r == -1:
				break
		l = 0
		while stat[l] == 1:
			l += 1
			if l == 10:
				break
	else:
		stat[int(a[i])] = 0
		if int(a[i]) < l:
			l = int(a[i])
		if int(a[i]) > r:
			r = int(a[i])
for i in stat:
	print(i, end='')
print()
