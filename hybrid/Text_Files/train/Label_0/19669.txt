#include <bits/stdc++.h>

using namespace std;

typedef long long int64;

int64 N, k;

bool check(int64 val)
{
  for(int i = 0; i < N - 1; i++) {
    val -= val / k + 1;
  }
  return (val >= 0);
}

int main()
{
  cin >> N >> k;
  int64 low = 0, high = 1000000000000000000LL;
  while(high - low > 0) {
    int64 mid = (low + high) >> 1;
    if(check(mid)) high = mid;
    else low = mid + 1;
  }
  cout << low << endl;
}
