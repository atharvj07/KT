from sys import stdin, stdout
(a, b, c) = map(int, stdin.readline().split())
z = [list(map(int, stdin.readline().split())) for _ in ' ' * a]
k = [0] * a
m = [0] * b
ans = [0] * a
for _ in ' ' * c:
	(u, v) = map(int, stdin.readline().split())
	k[u - 1] += 1
	m[v - 1] += 1
for i in range(a):
	ans[i] = -k[i]
	for j in range(b):
		ans[i] += m[j] * z[i][j]
stdout.write(' '.join(map(str, ans)))
