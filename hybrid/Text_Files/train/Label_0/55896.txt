import java.util.*;
import java.math.*;
import java.io.*;

public class CF487B {
	class RMQ {
		int n, lg, sign;
		int[][] table;
		public RMQ(int nn, int ss) { // -1 for min, 1 for max
			sign = ss;
			lg = 32 - Integer.numberOfLeadingZeros(n = nn);
			table = new int[lg][n];
		}
		void init(int[] a) {
			for(int i = 0 ; i < n ; i++) table[0][i] = sign * a[i];
			for(int k = 1, len = 1 ; k < lg ; k++, len <<= 1)
				for(int i = 0 ; i < n ; i++)
					table[k][i] = Math.max(table[k - 1][i], table[k - 1][Math.min(i + len, n - 1)]);
		}
		void init(int i, int x) {
			table[0][i] = sign * x;
			for(int k = 1, len = 1 ; k < lg ; k++, len <<= 1)
				table[k][i] = Math.max(table[k - 1][i], table[k - 1][Math.min(i + len, n - 1)]);
		}
		int query(int i, int j) {
			int exp = Integer.highestOneBit(j - i + 1);
			int k = Integer.numberOfTrailingZeros(exp);
			return sign * Math.max(table[k][i], table[k][j - exp + 1]);
		}
	}
	final int oo = Integer.MAX_VALUE;
	int n, d, L;
	RMQ max, min;
	boolean good(int l, int r) { return max.query(l, r) - min.query(l, r) <= d; }
	public CF487B() {
		FS scan = new FS();
		n = scan.nextInt();  d = scan.nextInt();  L = scan.nextInt();
		int[] a = new int[n];
		for(int i = 0 ; i < n ; i++) a[i] = scan.nextInt();
		max = new RMQ(n, 1);
		min = new RMQ(n, -1);
		max.init(a);
		min.init(a);
		RMQ dpmin = new RMQ(n, -1);
		for(int i = n - 1 ; i > Math.max(-1, n - L - 1) ; i--) {
			if(n - i >= L && good(i, n - 1)) dpmin.init(i, 1);
			else dpmin.init(i, oo);
		}
		for(int i = n - L - 1 ; i >= 0 ; i--) {
			if(!good(i, i + L - 1)) {
				dpmin.init(i, oo);
				continue;
			}
			int lo = i + L - 1, hi = n - 1;
			while(lo < hi - 1) {
				int mid = (lo + hi >> 1);
				if(good(i, mid)) lo = mid;
				else hi = mid;
			}
			if(good(i, hi)) lo = hi;
			if(lo + 1 < n && dpmin.query(i + L, Math.min(lo + 1, n - 1)) == oo) {
				dpmin.init(i, oo);
				continue;
			}
			if(lo + 1 == n) dpmin.init(i, 1);
			else dpmin.init(i, dpmin.query(i + L, Math.min(lo + 1, n - 1)) + 1);
		}
		int res = dpmin.query(0, 0);
		if(res == oo) System.out.println(-1);
		else System.out.println(res);
	}
	class FS {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer("");
		public String next() {
			while (!st.hasMoreTokens()) {
				try { st = new StringTokenizer(br.readLine()); }
				catch (Exception e) { e.printStackTrace(); }
			}
			return st.nextToken();
		}
		public int nextInt() { return Integer.parseInt(next()); }
	}
	public static void main(String[] args) { new CF487B(); }
}
