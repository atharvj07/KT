from sys import stdin, stdout
stdin.readline

def mp():
	return list(map(int, stdin.readline().strip().split()))

def it():
	return int(stdin.readline().strip())
from collections import defaultdict as dd, Counter as C, deque
from math import ceil, gcd
for _ in range(it()):
	n = it()
	l = list(input())
	c = C(l)
	if l[0] == '0' or l[-1] == '0' or n & 1 or c['0'] & 1 or c['1'] & 1:
		print('NO')
		continue
	else:
		(a, b) = (['('], ['('])
		(o, z) = (1, 0)
		for i in range(1, n):
			if l[i] == '1':
				if o < c['1'] // 2:
					a.append('(')
					b.append('(')
				else:
					a.append(')')
					b.append(')')
				o += 1
			else:
				if z & 1:
					a.append('(')
					b.append(')')
				else:
					a.append(')')
					b.append('(')
				z += 1
		print('YES')
		print(''.join(a))
		print(''.join(b))
