import math
n = int(input())
s = input()
sum1 = 1000
for i in range(len(s) - 3):
	sum = 0
	sum += min(abs(ord(s[i]) - 65), abs(26 - abs(ord(s[i]) - 65)))
	sum += min(abs(ord(s[i + 1]) - 67), abs(26 - abs(ord(s[i + 1]) - 67)))
	sum += min(abs(ord(s[i + 2]) - 84), abs(26 - abs(ord(s[i + 2]) - 84)))
	sum += min(abs(ord(s[i + 3]) - 71), abs(26 - abs(ord(s[i + 3]) - 71)))
	if sum < sum1:
		sum1 = sum
print(sum1)
