from collections import *
from itertools import accumulate

def arr_sum(arr):
	arr.appendleft(0)
	return list(accumulate(arr, lambda x, y: x + y))

def solve():
	(ans1, ans0) = (0, 0)
	for i in range(n):
		if a[i]:
			ans1 += n - (i + 1) - (mem[-1] - mem[i + 1])
		else:
			ans0 += mem[i]
	return min(ans0, ans1)
(n, a) = (int(input()), deque(map(int, input().split())))
mem = arr_sum(a.copy())
print(solve())
