#include <bits/stdc++.h>
using namespace std;
template <typename T>
T gcd(T a, T b) {
  if (a == 0) return b;
  return gcd(b % a, a);
}
template <typename T>
T pow(T a, T b, long long int m) {
  T ans = 1;
  while (b > 0) {
    if (b % 2 == 1) ans = ((ans % m) * (a % m)) % m;
    b /= 2;
    a = ((a % m) * (a % m)) % m;
  }
  return ans % m;
}
template <typename T>
void tracev(vector<T> &a) {
  for (auto &i : a) {
    cout << i << ' ';
  }
  cout << '\n';
}
template <typename T>
void tracevv(vector<T> &a) {
  for (auto &i : a) {
    tracev(i);
  }
}
template <typename T>
void tracevp(vector<T> &a) {
  for (auto &i : a) {
    cout << "i.ff"
         << ": " << i.first << " | "
         << "i.ss"
         << ": " << i.second << "\n";
    ;
  }
  cout << '\n';
}
bool comp(vector<long long int> &a1, vector<long long int> &a2) {
  return a1.size() > a2.size();
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ;
  int n;
  cin >> n;
  vector<long long int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  if (n == 1) {
    cout << "0\n";
    return 0;
  }
  vector<vector<long long int>> ans(1);
  ans[0].push_back(a[0]);
  for (int i = 1; i < n; i++) {
    bool f = false;
    for (int j = 0; j < ans.size(); j++) {
      long long int g = gcd(a[i], ans[j][0]);
      if (!((a[i] / g) % 2 == 0 || (ans[j][0] / g) % 2 == 0)) {
        ans[j].push_back(a[i]);
        f = true;
      }
    }
    if (!f) {
      ans.push_back({a[i]});
    }
  }
  sort(ans.begin(), ans.end(), comp);
  cout << n - ans[0].size() << '\n';
  for (int i = 1; i < ans.size(); i++) {
    for (int j = 0; j < ans[i].size(); j++) {
      cout << ans[i][j] << '\n';
    }
  }
  return 0;
}
