	if novoAno != ano and novoAno[0] != novoAno[1] and (novoAno[0] != novoAno[2]) and (novoAno[0] != novoAno[3]) and (novoAno[1] != novoAno[2]) and (novoAno[1] != novoAno[3]) and (novoAno[2] != novoAno[3]):
		print(novoAno)
		break
	else:
		aux = int(novoAno)
		aux += 1
		novoAno = str(aux)
