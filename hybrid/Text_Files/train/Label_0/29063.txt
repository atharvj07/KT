#include<bits/stdc++.h>
using namespace std;

using ll=long long;
const ll INF=1e9;
void cmax(ll &lhs,ll rhs){
    lhs=max(lhs,rhs);
}
void cmin(ll &lhs,ll rhs){
    lhs=min(lhs,rhs);
}
int main(){
    string s;
    cin>>s;
    int n=s.size();
    vector<vector<ll>> dpmax(n,vector<ll>(n+1,-INF));
    vector<vector<ll>> dpmin(n,vector<ll>(n+1,INF));
    for(int k=1;k<=n;k++){
        for(int l=0;l+k<=n;l++){
            int r=l+k;
            if(k==1){
                if(isdigit(s[l])){
                    dpmax[l][r]=s[l]-'0';
                    dpmin[l][r]=s[l]-'0';
                }
            }
            else if(s[l]=='(' || s[r-1]==')'){
                int cnt=0;
                for(int i=l;i<r;i++){
                    cnt+=(int)(isdigit(s[i]));
                }
                if(cnt>=2){
                    if(s[l]=='('){
                        dpmax[l][r]=dpmax[l+1][r];
                        dpmin[l][r]=dpmin[l+1][r];
                    }
                    else{
                        dpmax[l][r]=dpmax[l][r-1];
                        dpmin[l][r]=dpmin[l][r-1];
                    }
                }
            }
            else{
                for(int i=l;i<r;i++){
                    if(s[i]=='+'){
                        cmax(dpmax[l][r],dpmax[l][i]+dpmax[i+1][r]);
                        cmin(dpmin[l][r],dpmin[l][i]+dpmin[i+1][r]);
                    }
                    if(s[i]=='-'){
                        cmax(dpmax[l][r],dpmax[l][i]-dpmin[i+1][r]);
                        cmin(dpmin[l][r],dpmin[l][i]-dpmax[i+1][r]);
                    }
                }

            }
        }
    }
    cout<<dpmax[0][n]<<endl;

    return 0;
}
