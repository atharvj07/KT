#include <bits/stdc++.h>
using namespace std;
const int INF = (int)2e9;
const double EPS = (double)1e-9;
const double PI = (double)acos(-1.0);
const int MOD = (int)1e9 + 7;
int main() {
  char c;
  int flag = 0;
  stack<int> tab;
  vector<int> res;
  res.clear();
  while (!tab.empty()) tab.pop();
  while (scanf("%c", &c) != EOF) {
    if (c == '/') flag = -100;
    if (!('a' <= c && c <= 'z')) continue;
    if (flag == -101) {
      if (c == 'a') res.push_back(tab.top()), tab.pop(), flag = 4;
      if (c == 'r') flag = 1;
      if (c == 'd') flag = 1;
    }
    if (flag == -100) flag = -101;
    if (flag == -10) {
      if (c == 'a') tab.push(0), flag = 4;
      if (c == 'r') flag = 1;
      if (c == 'd') tab.top()++, flag = 1;
    }
    if (flag == 0) flag = -10;
    if (flag > 0) flag--;
  }
  sort(res.begin(), res.end());
  for (vector<int>::iterator it = res.begin(); it != res.end(); it++) {
    printf("%d%c", *it, (it != res.end() - 1) ? ' ' : '\n');
  }
  return 0;
}
