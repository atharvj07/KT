import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.io.InputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author Xi Lin
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskF solver = new TaskF();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskF {
        int[] u;
        int[] v;
        char[] c;
        int[] merged;
        ArrayList<Integer>[] graph;
        ArrayList<Integer>[] level;

        int to(int i, int a) {
            return u[i] + v[i] - a;
        }

        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int n = in.nextInt();
            u = new int[n];
            v = new int[n];
            c = new char[n];
            graph = new ArrayList[n];
            level = new ArrayList[n + 1];
            merged = new int[n];
            for (int i = 0; i < n; i++) {
                graph[i] = new ArrayList<Integer>();
                level[i + 1] = new ArrayList<Integer>();
            }
            for (int i = 0; i < (n - 1); i++) {
                u[i] = in.nextInt() - 1;
                v[i] = in.nextInt() - 1;
                c[i] = in.next().charAt(0);
                graph[u[i]].add(i);
                graph[v[i]].add(i);
            }
            dfs(0, 0, 1);
            int p = -1, ret = 0, total = 0;
            for (int d = 1; d < n; ++d) {
                int tmp = total;
                for (int u : level[d]) {
                    tmp += merged[u];
                    if (merged[u] == 0) ++tmp;
                }
                if (p == -1 || tmp < ret) {
                    p = d;
                    ret = tmp;
                }
                total += level[d].size();
            }
            out.println(ret);
            out.println(p);
        }

        Node dfs(int u, int parent, int depth) {
            level[depth].add(u);
            Node rt = new Node();
            rt.sz = 1;
            Node tmp = null;
            for (int e : graph[u]) {
                int v = to(e, u);
                if (v == parent) continue;
                Node son = dfs(v, u, depth + 1);
                rt.ch[c[e] - 'a'] = merge(rt.ch[c[e] - 'a'], son);
                tmp = merge(tmp, son);
            }
            for (int i = 0; i < 26; i++) {
                rt.sz += size(rt.ch[i]);
            }
            merged[u] = size(tmp);
            return rt;
        }

        int size(Node u) {
            if (u == null) return 0;
            else return u.sz;
        }

        Node merge(Node u, Node v) {
            if (u == null) return v;
            if (v == null) return u;
            Node rt = new Node();
            rt.sz = 1;
            for (int i = 0; i < 26; i++) {
                rt.ch[i] = merge(u.ch[i], v.ch[i]);
                rt.sz += size(rt.ch[i]);
            }
            return rt;
        }

        class Node {
            Node[] ch;
            int sz;

            public Node() {
                ch = new Node[26];
                for (int i = 0; i < 26; i++) {
                    ch[i] = null;
                }
                sz = 0;
            }

            public Node clone() {
                Node o = new Node();
                o.sz = this.sz;
                for (int i = 0; i < 26; i++) {
                    o.ch[i] = this.ch[i];
                }
                return o;
            }

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
