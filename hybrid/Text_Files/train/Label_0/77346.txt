def get_ints():
	return [int(x) for x in input().split()]
t = int(input())
for _ in range(t):
	n = int(input())
	A = get_ints()
	As = sum(A)
	for c in range(n, 0, -1):
		if As % c == 0:
			ds = As // c
			good = True
			s = 0
			for j in range(n):
				s += A[j]
				if s == ds:
					s = 0
				elif s > ds:
					good = False
					break
			if good:
				print(n - c)
				break
