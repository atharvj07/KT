#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    bool flag = false;

    while (cin >> s) {
        int sz = s.size();
        if (s[sz - 1] == ',' || s[sz - 1] == '.') s = s.substr(0, (sz--) - 1);
        if (sz >= 3 && sz <= 6) {
            if (flag) cout << " ";
            cout << s;
            flag = true;
        }
    }
    cout << endl;
    
    return 0;
}
