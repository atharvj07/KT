def fib(n):
	f = [1, 1]
	for i in range(2, n):
		f.append(add(f[i - 1], f[i - 2]))
	return f[n - 1]

def add(a, b):
	a += b
	while a >= 998244353:
		a -= 998244353
	while a < 0:
		a += 998244353
	return a

def binpow(a, n):
	ans = 1
	while n > 0:
		if n % 2:
			ans = mul(ans, a)
		a = mul(a, a)
		n //= 2
	return ans
