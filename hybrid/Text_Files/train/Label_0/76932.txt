import math

def solve(n, A):
	A_diff = []
	a_prev = None
	for a in A:
		if a_prev is None:
			a_prev = a
		else:
			A_diff.append(a - a_prev)
			a_prev = a
	c_s = set(A_diff)
	if len(c_s) < 2:
		return 0
	elif len(c_s) == 2:
		m = 0
		for c in c_s:
			m += abs(c)
		c = c % m
		if all([a < m for a in A]):
			return f'{m} {c}'
		return -1
	return -1

def main():
	ans = []
	t = int(input())
	for _ in range(t):
		n = int(input())
		A = list(map(int, input().split(' ')))
		ans.append(solve(n, A))
	for a in ans:
		print(a)
main()
