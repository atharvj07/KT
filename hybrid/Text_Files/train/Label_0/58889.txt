import java.util.Scanner;

public class B {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int m = sc.nextInt();
        long r0 = sc.nextLong();
        int[] occur = new int[m+1];
        int res = -1;
        int idx = 1;
        while (true) {
            r0 = (r0 * a + b) % m;
            if (occur[(int) r0] != 0) {
                res = idx - occur[(int) r0];
                break;
            }
            occur[(int)r0]=idx;
            idx++;
        }
        System.out.println(res);
    }
}
