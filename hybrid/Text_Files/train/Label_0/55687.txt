import java.util.*;
import java.io.*;
public class Main {
static long bit [];
static long mod;
static void update(int pos,long val){
	while(pos < bit.length){
		bit[pos] += val;
		while(bit[pos] >= mod)bit[pos] -= mod;
		pos += pos&(-pos);
	}
}
static long read(int pos){
	long tmp = 0;
	while(pos > 0){
	 tmp += bit[pos];
	while(tmp >= mod)tmp -= mod;
	 pos -= pos&(-pos);	
	}
	return tmp;
}	
public static void main(String [] args)throws Exception {
	BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	int n = Integer.parseInt(br.readLine());
	StringTokenizer st = new StringTokenizer(br.readLine());
	mod = (long)1E9 + 7;
	int array[] = new int[n + 1];
	bit = new long[1000005];
	long ans = 0;
	long temp = 0;
	for(int i = 1;i <= n;i++){
		array[i] = Integer.parseInt(st.nextToken());
		long same = read(array[i]) - read(array[i] - 1);
		temp = (read(array[i])*array[i] + array[i] - same + mod)%mod;
		ans += temp;
		while(ans >= mod)ans -= mod;
		update(array[i],temp);
	}
	System.out.println(ans);
} 
}
