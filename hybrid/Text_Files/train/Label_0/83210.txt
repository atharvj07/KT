import math
n = int(input())
coor = []

def dis(x, y):
	return abs(x[0] - y[0]) + abs(y[1] - x[1])
for i in range(n):
	(u, v) = map(int, input().split())
	coor.append((u, v))
cost = list(map(int, input().split()))
k = list(map(int, input().split()))
d = {}
power_st = []
vis = [0] * n
connect = []
parent = [-1 for i in range(n)]
for i in range(n):
	d[i] = cost[i]
ans = 0
for i in range(n):
	m = min(d, key=d.get)
	if vis[m] == 0:
		power_st.append(m)
	else:
		connect.append([m, parent[m]])
	ans = ans + d[m]
	del d[m]
	for j in d.keys():
		t = dis(coor[m], coor[j]) * (k[m] + k[j])
		if t < d[j]:
			d[j] = t
			parent[j] = m
			vis[j] = 1
print(ans)
print(len(power_st))
for i in power_st:
	print(i + 1, end=' ')
print()
print(len(connect))
for i in connect:
	print(i[0] + 1, i[1] + 1)
