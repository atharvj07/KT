def correct(p):
	global typ
	if typ == '':
		typ = p
	elif typ == 'R':
		if p == 'L':
			return False
		if p == 'D' or p == 'U':
			typ = typ + p
	elif typ == 'L':
		if p == 'R':
			return False
		if p == 'D' or p == 'U':
			typ = typ + p
	elif typ == 'U':
		if p == 'D':
			return False
		if p == 'R' or p == 'L':
			typ = typ + p
	elif typ == 'D':
		if p == 'U':
			return False
		if p == 'R' or p == 'L':
			typ = typ + p
	elif typ == 'RU' or typ == 'UR':
		if p != 'R' and p != 'U':
			return False
	elif typ == 'RD' or typ == 'DR':
		if p != 'R' and p != 'D':
			return False
	elif typ == 'LU' or typ == 'UL':
		if p != 'L' and p != 'U':
			return False
	elif typ == 'LD' or typ == 'DL':
		if p != 'L' and p != 'D':
			return False
	return True
n = int(input())
s = input()
i = 0
k = 0
typ = ''
while i < n:
	while i < n:
		if correct(s[i]) == False:
			break
		else:
			i += 1
	k += 1
	typ = ''
print(k)
