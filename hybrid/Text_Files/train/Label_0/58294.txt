import java.util.*;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int k=sc.nextInt();

        while(k-->0){

            long n=sc.nextLong();
            long x=sc.nextLong();
            long t=sc.nextLong();

            if(x==t){
                System.out.println(n-1);
            }
            else if(x>t){
                System.out.println(0);
            }else {

                long z=Math.min(n-1,t/x);

                System.out.println(z*(z-1)/2+(z*(n-z)));

            }

        }
    }
}
