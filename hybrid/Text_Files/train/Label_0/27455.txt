#include <bits/stdc++.h>
#define fi first
#define se second
#define mp make_pair
using namespace std;
const int N = 2e5 + 5;

pair <int, int> q[N];

char a[N];
char b[N];

int st[4 * N];
int lz[4 * N];
int n,m;

void dolazy(int id, int l, int r)
{
    int mid = (l + r) >> 1;

    if (lz[id] != 0)
        lz[id << 1] = lz[id << 1 | 1] = lz[id];

    if (lz[id] > 0)
    {
        st[id << 1] = (mid - l + 1);
        st[id << 1 | 1] = (r - mid);
    } else if (lz[id] < 0)
        st[id << 1] = st[id << 1 | 1] = 0;
    lz[id] = 0;
    st[id] = st[id << 1] + st[id << 1 | 1];
}

void update(int id, int l, int r, int u, int v, int val)
{
    if (l > v || r < u)
        return;
    if (l >= u && r <= v)
    {
        lz[id] = val;
        if (val < 0)
            st[id] = 0;
        else
            st[id] = (r - l + 1);
        return;
    }
    int mid = (l + r) >> 1;
    dolazy(id, l, r);
    update(id << 1, l, mid, u, v, val);
    update(id << 1 | 1, mid + 1, r, u, v, val);
    st[id] = st[id << 1] + st[id << 1 | 1];
}

int getsum(int id, int l, int r, int u, int v)
{
    if (l > v || r < u)
        return 0;
    if (l >= u && r <= v)
        return st[id];
    int mid = (l + r) >> 1;
    dolazy(id, l, r);
    int lnode = getsum(id << 1, l, mid, u, v);
    int rnode = getsum(id << 1 | 1, mid + 1, r, u, v);
    return lnode + rnode;
}

void solve()
{
    for (int i = 1; i <= 4 * n; i++)
        st[i] = lz[i] = 0;

    for (int i = 1; i <= n; i++)
        update(1, 1, n, i, i, (b[i] - '0' == 1) ? 1 : -1);

    for (int i = m; i >= 1; i--)
    {
        int l = q[i].fi;
        int r = q[i].se;
        int cnt = getsum(1, 1, n, l, r);
        int len = (r - l + 1);
        if (2 * cnt < len)
            update(1, 1, n, l, r, -1);
        else if (2 * (len - cnt) < len)
            update(1, 1, n, l, r, 1);
        else {
            cout << "NO\n";
            return;
        }
    }

    for (int i = 1; i <= n; i++) if (getsum(1, 1, n, i, i) != a[i] - '0')
    {
        cout << "NO\n";
        return;
    }
    cout << "YES\n";
}

int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    //freopen("file.inp","r",stdin);
    int t;
    cin >> t;
    while (t--)
    {
        cin >> n >> m;
        for (int i = 1; i <= n; i++)
            cin >> a[i];

        for (int i = 1; i <= n; i++)
            cin >> b[i];

        for (int i = 1; i <= m; i++)
            cin >> q[i].fi >> q[i].se;

        solve();
    }
    return 0;
}
