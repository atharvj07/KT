for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	a = sorted(a)
	ans = 0
	for now in range(2 * n + 1):
		ta = 0
		i = 0
		j = n - 1
		while i < j:
			if a[i] + a[j] == now:
				ta += 1
				i += 1
				j -= 1
			elif a[i] + a[j] > now:
				j -= 1
			else:
				i += 1
		ans = max(ans, ta)
	print(ans)
