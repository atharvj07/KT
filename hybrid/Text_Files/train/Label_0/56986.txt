import java.io.OutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.PrintWriter;
import java.io.OutputStream;
import java.util.StringTokenizer;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.io.InputStream;
import java.io.BufferedOutputStream;

/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 *
 * @author AEoui
 */
public class C {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        Kattio in = new Kattio(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        TaskCentroid solver = new TaskCentroid();
        solver.solve(1, in, out);
        out.close();
    }

    static class TaskCentroid {
        private ArrayList<Integer>[] adjList;
        private int N;
        private int[] sz;
        private char[] ans;

        public void solve(int testNumber, Kattio in, PrintWriter out) {
            N = in.nextInt();

            adjList = new ArrayList[N];
            for (int i = 0; i < N; ++i) {
                adjList[i] = new ArrayList<>();
            }

            for (int i = 0; i < N - 1; ++i) {
                int u = in.nextInt() - 1;
                int v = in.nextInt() - 1;
                adjList[u].add(v);
                adjList[v].add(u);
            }

            sz = new int[N];
            ans = new char[N];
            markTree(0);

            for (int i = 0; i < N; ++i)
                out.print(ans[i] + " ");
        }

        private int getCentroid(int u, int p, int n) {
            boolean isCentroid = true;
            int hChild = -1;
            for (int v : adjList[u])
                if (v != p) {
                    if (!(ans[v] >= 'A' && ans[v] <= 'Z')) {
                        if (sz[v] > n / 2) {
                            isCentroid = false;
                        }

                        if (hChild == -1 || sz[v] > sz[hChild]) {
                            hChild = v;
                        }
                    }
                }

            if (isCentroid && n - sz[u] <= n / 2) {
                return u;
            }

            return getCentroid(hChild, u, n);
        }

        private int getCentroid(int u, char c) {
            dfsSize(u, -1);
            int centroid = getCentroid(u, -1, sz[u]);
            ans[centroid] = c;

            return centroid;
        }

        private void markTree(int root) {
            markTree(root, 'A');
        }

        private void markTree(int root, char c) {
            int centroid = getCentroid(root, c);
            ++c;
            for (int u : adjList[centroid]) {
                if (!(ans[u] >= 'A' && ans[u] <= 'Z')) {
                    markTree(u, c);
                }
            }
        }

        private void dfsSize(int u, int p) {
            sz[u] = 1;
            for (int v : adjList[u])
                if (v != p && !(ans[v] >= 'A' && ans[v] <= 'Z')) {
                    dfsSize(v, u);
                    sz[u] += sz[v];
                }
        }

    }

    static class Kattio extends PrintWriter {
        private BufferedReader r;
        private String line;
        private StringTokenizer st;
        private String token;

        public Kattio(InputStream i) {
            super(new BufferedOutputStream(System.out));
            r = new BufferedReader(new InputStreamReader(i));
        }

        public Kattio(InputStream i, OutputStream o) {
            super(new BufferedOutputStream(o));
            r = new BufferedReader(new InputStreamReader(i));
        }

        public int nextInt() {
            return Integer.parseInt(nextToken());
        }

        private String peekToken() {
            if (token == null)
                try {
                    while (st == null || !st.hasMoreTokens()) {
                        line = r.readLine();
                        if (line == null) return null;
                        st = new StringTokenizer(line);
                    }
                    token = st.nextToken();
                } catch (IOException e) {
                }
            return token;
        }

        private String nextToken() {
            String ans = peekToken();
            token = null;
            return ans;
        }

    }
}
