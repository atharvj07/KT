    void solve(int TC) throws Exception{
        int N = ni(), Q = ni();
        AhoCorasick aho = new AhoCorasick(26, (int)1e6);
        int[] index = new int[N];
        for(int i = 0; i< N; i++)index[i] = aho.addString(i, n().toCharArray());
        aho.buildSuffixLinks();
        int n = aho.ID;
        MyTreeSet<Integer>[] valueSet = new MyTreeSet[n];
        for(int i = 0; i< N; i++){
            if(valueSet[index[i]] == null)valueSet[index[i]] = new MyTreeSet<>();
            valueSet[index[i]].add(0);
        }

        int[][] e = aho.slt();
        int[] from = new int[e.length], to = new int[e.length];
        for(int i = 0; i< e.length; i++){
            from[i] = e[i][0];
            to[i] = e[i][1];
        }
        g = make(n, e.length, from, to, false);
        int root = aho.sentinal;

        sub = new int[n]; de = new int[n];par = new int[n];
        Arrays.fill(par, -1);

        dfs1(g, par, sub, de, root);
        chainHead = new int[n];
        Arrays.fill(chainHead, -1);
        treePos = new int[n];

        Arrays.fill(chainHead, -1);
        hld(root);
        SegmentTree segtree = new SegmentTree(n);
        for(int i = 0; i< N; i++)segtree.update(treePos[index[i]], 0);
        int[] A = new int[N];
        while (Q-->0){
            if(ni() == 1){
                int ind = ni()-1, val = ni();

                int maxVal = upd(valueSet, index[ind], A[ind], val);
                A[ind] = val;
                segtree.update(treePos[index[ind]], maxVal);
            }else{
                char[] text = n().toCharArray();
                int state = aho.root;
                long ans = -1;
                for(char i:text){
                    state = aho.move(state, i);
                    ans = Math.max(ans, queryUp(segtree, state));
                }
                pn(ans);
            }
        }
    }
    int upd(MyTreeSet<Integer>[] valueSet, int ind, int old, int newVal){
        valueSet[ind].remove(old);
        valueSet[ind].add(newVal);
        return valueSet[ind].last();
    }
    class MyTreeSet<T> implements Iterable<T>{
        private int size;
        private TreeMap<T, Integer> map;
        public MyTreeSet(){
            size = 0;
            map = new TreeMap<>();
        }
        public int size(){return size;}
        public int dsize(){return map.size();}
        public boolean isEmpty(){return size==0;}
        public void add(T t){
            size++;
            map.put(t, map.getOrDefault(t, 0)+1);
        }
        public boolean remove(T t){
            if(!map.containsKey(t))return false;
            size--;
            int c = map.get(t);
            if(c==1)map.remove(t);
            else map.put(t, c-1);
            return true;
        }
        public int freq(T t){return map.getOrDefault(t, 0);}
        public boolean contains(T t){return map.getOrDefault(t,0)>0;}
        public T ceiling(T t){return map.ceilingKey(t);}
        public T floor(T t){return map.floorKey(t);}
        public T first(){return map.firstKey();}
        public T last(){return map.lastKey();}
        public Iterator<T> iterator() {
            return new MyTreeSetIterator<>(this);
        }
        class MyTreeSetIterator<T> implements Iterator<T>{
            TreeMap<T, Integer> mp;
            T element = null;
            int cur = 0, freq = 0;
            MyTreeSetIterator(MyTreeSet<T> obj){
                this.mp = obj.map;
                if(!this.mp.isEmpty()){
                    element = mp.firstKey();
                    freq = mp.firstEntry().getValue();
                }
            }
            public boolean hasNext(){
                return element != null;
            }
            public T next(){
                T ret = element;
                cur++;
                if(cur == freq){
                    Map.Entry<T, Integer> e = mp.higherEntry(element);
                    if(e == null){
                        element = null;
                        freq = 0;
                    }else{
                        element = e.getKey();
                        freq = e.getValue();
                        cur = 0;
                    }
                }
                return ret;
            }
        }
    }
    long queryUp(SegmentTree segtree, int node){
        long ans = -1;
        while(node != -1){
            int ch = chainHead[node];
            ans = Math.max(ans, segtree.query(treePos[ch], treePos[node]));
            node = par[ch];
        }
        return ans;
    }
    void hld(int u){
        if(chainHead[u]==-1)chainHead[u] = u;
        treePos[u] = pos++;
        int ch = -1;
        for(int v:g[u]){
            if(v==par[u])continue;
            if(ch==-1 || sub[ch]<sub[v])ch = v;
        }
        if(ch>=0){
            chainHead[ch] = chainHead[u];
            hld(ch);
        }
        for(int v:g[u]){
            if(v==par[u] || v==ch)continue;
            hld(v);
        }
    }
    void dfs1(int[][] g, int[] par, int[] sub, int[] de, int u){
        sub[u] = 1;
        for(int v:g[u]){
            de[v] = de[u]+1;
            par[v] = u;
            dfs1(g, par, sub, de, v);
            sub[u]+=sub[v];
        }
    }


    class SegmentTree{
        private long initValue(){return -1;}
        private long update(long oldValue, long newValue){return newValue;}
        private long merge(long le, long ri){return Math.max(le, ri);}
        private long initQuery(){return -1;}

        int m= 1;
        long[] t;
        public SegmentTree(int n){
            while(m<n)m<<=1;
            t = new long[m<<1];
            Arrays.fill(t, initValue());
        }
        public SegmentTree(long[] a){
            while(m<a.length)m<<=1;
            t = new long[m<<1];
            Arrays.fill(t, initValue());
            for(int i = 0; i< a.length; i++)t[i+m] = a[i];
            for(int i = m-1; i>0; i--)t[i] = merge(t[i<<1], t[i<<1|1]);
        }
        public void update(int i, long val){
            t[i += m]  = update(t[i], val);
            for(i>>=1;i>0;i>>=1)t[i] = merge(t[i<<1], t[i<<1|1]);
        }
        long query(int l, int r){
            long lans = initQuery(), rans = initQuery();
            for(l+=m,r+=m+1;l<r;l>>=1,r>>=1){
                if((l&1)==1)lans = merge(lans, t[l++]);
                if((r&1)==1)rans = merge(t[--r], rans);
            }
            return merge(lans, rans);
        }
    }


    static class AhoCorasick {
        int root, sentinal, ID, A, maxSize;
        int[] next, fail, hit;
        int[] strIndex;
        public AhoCorasick(int A, int maxSize) {
            this.A = A;
            this.maxSize = maxSize + 2;
            next = new int[A * this.maxSize];
            fail = new int[this.maxSize];
//            hit = new int[this.maxSize];
            strIndex = new int[this.maxSize];
            Arrays.fill(strIndex, -1);
            Arrays.fill(fail, -1);
            Arrays.fill(next, -1);
            ID = 0;
            sentinal = ID++;
            root = ID++;
            fail[root] = sentinal;
            for (int i = 0; i < this.A; i++) next[sentinal * this.A + i] = root;
        }

        int f(char ch) {
            return ch - 'a';
        }

        int addString(int ind, char[] pat) {
            int node = root;
            for (char ch : pat) {
                int c = f(ch);
                if (next[node * A + c] == -1) next[node * A + c] = ID++;
                node = next[node * A + c];
            }
            strIndex[node] = ind;
            return node;
//            hit[node]++;
        }
        int move(int node, char ch){
            int c = ch-'a';
            while(next[node*A+c] == -1)node = fail[node];
            return next[node*A+c];
        }
        void buildSuffixLinks(){
            int[] Q = new int[ID];
            int c = 0;
            Q[c++] = root;
            for(int r = 0; r< c; r++){
                int node = Q[r];
                for(int i = 0; i< A; i++){
                    int nxt = next[node*A+i];
                    if(nxt == -1)continue;
                    Q[c++] = nxt;
                    for(int to = fail[node]; true; to = fail[to]){
                        int fch = next[to*A+i];
                        if(fch != -1){
                            fail[nxt] = fch;
//                            hit[nxt] += hit[fch];
                            break;
                        }
                    }
                }
            }
        }
        int[][] slt(){
            int cnt = 0;
            int[][] e = new int[ID][];
            for(int i = 0; i< ID; i++)if(fail[i] != -1)e[cnt++] = new int[]{fail[i], i};
            return Arrays.copyOf(e, cnt);
        }
        static char[] rev_(char[] a) {
            for (int i = 0, j = a.length - 1; i < j; i++, j--) {
                char c = a[i];
                a[i] = a[j];
                a[j] = c;
            }
            return a;
        }
    }

    int[][] make(int n, int e, int[] from, int[] to, boolean f){
        int[][] g = new int[n][];int[]cnt = new int[n];
        for(int i = 0; i< e; i++){
            cnt[from[i]]++;
            if(f)cnt[to[i]]++;
        }
        for(int i = 0; i< n; i++)g[i] = new int[cnt[i]];
        for(int i = 0; i< e; i++){
            g[from[i]][--cnt[from[i]]] = to[i];
            if(f)g[to[i]][--cnt[to[i]]] = from[i];
        }
        return g;
    }

    class FastReader{
        BufferedReader br;
        StringTokenizer st;
        public FastReader(){
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public FastReader(String s) throws Exception{
            br = new BufferedReader(new FileReader(s));
        }

        String next() throws Exception{
            while (st == null || !st.hasMoreElements()){
                try{
                    st = new StringTokenizer(br.readLine());
                }catch (IOException  e){
                    throw new Exception(e.toString());
                }
            }
            return st.nextToken();
        }

        String nextLine() throws Exception{
            String str;
            try{
                str = br.readLine();
            }catch (IOException e){
                throw new Exception(e.toString());
            }
            return str;
        }
    }
