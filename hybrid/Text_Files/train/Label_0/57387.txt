import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileNotFoundException;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class C {
    FastScanner scanner;
    PrintWriter writer;

    void solve() throws IOException {
        scanner = new FastScanner(System.in);
        writer = new PrintWriter(System.out);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        InputProvider ip = new IOInputProvider();

        int r = findGreater(ip, n, k);  // > k
        int l = findLower(ip, n, k);    // <= k

        if (r < l)
            r += n;
        while (r - l > 1) {
            int m = l + (r - l) / 2;
            if (ip.getNumberOfCards(m % n) > k) {
                r = m;
            } else {
                l = m;
            }
        }

        writer.println("! " + (l + 1));

        writer.close();
    }

    int findGreater(InputProvider ip, int n, int k) throws IOException {
        int d = (int)Math.sqrt(n) + 1;
        for (int i = 0;; i = (i + d) % n) {
            if (ip.getNumberOfCards(i) > k) {
                return i;
            }
        }
    }

    int findLower(InputProvider ip, int n, int k) throws IOException {
        int d = (int)Math.sqrt(n);
        for (int i = 0;; i = (i + d) % n) {
            if (ip.getNumberOfCards(i) <= k) {
                return i;
            }
        }
    }

    private int right(int pos, int n) {
        return (pos + 1) % n;
    }

    private int left(int pos, int n) {
        pos--;
        if (pos < 0)
            pos += n;
        return pos;
    }

    public static void main(String[] args) throws IOException {
        new C().solve();
    }

    interface InputProvider {
        int getNumberOfCards(int pos) throws IOException;
    }

    class TestInputProvider implements InputProvider {
        private final int k;
        private final int p;
        private int[] cards;
        private int reqs;

        public TestInputProvider(int n, int k, int p) {
            this.k = k;
            this.p = p;
            cards = new int[n];
            Arrays.fill(cards, k);
        }

        @Override public int getNumberOfCards(int pos) {
            reqs++;
            writer.println("Request #" + reqs + ": ? " + (pos + 1));
            int res = cards[pos];
            step();
            return res;
        }

        private void step() {
            int n = cards.length;
            int[] newCards = new int[n];
            for (int i = 0; i < n; i++) {
                if (left(i, n) == p) {
                    newCards[i] = cards[left(i, n)];
                } else {
                    newCards[i] = (cards[left(i, n)] + 1) / 2;
                }
                if (right(i, n) != p) {
                    newCards[i] += cards[right(i, n)] / 2;
                }
            }
            this.cards = newCards;
        }
    }

    class IOInputProvider implements InputProvider {
        @Override public int getNumberOfCards(int pos) throws IOException {
            writer.println("? " + (pos + 1));
            writer.flush();
            return scanner.nextInt();
        }
    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer tokenizer;

        FastScanner(InputStream is) {
            this.br = new BufferedReader(new InputStreamReader(is));
        }

        FastScanner(File file) throws FileNotFoundException {
            this(new FileInputStream(file));
        }

        String nextLine() throws IOException {
            return br.readLine();
        }

        String next() throws IOException {
            while (tokenizer == null || !tokenizer.hasMoreTokens())
                tokenizer = new StringTokenizer(br.readLine());

            return tokenizer.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }
}
