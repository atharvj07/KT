(n, M) = map(int, input().split())
a = [int(x) for x in input().split()]
a.append(0)
a.append(M)
a.sort()
b = [0 for i in range(0, n + 1)]
flag = 1
for i in range(0, n + 1):
	b[i] += b[i - 1] + flag * (a[i + 1] - a[i])
	flag += pow(-1, i + 1)
ans = b[n]
for i in range(0, n + 1):
	ans = max(ans, b[i] - 1 + M - a[i + 1] - (b[n] - b[i]))
print(ans)
