class SegmentTree:

	def __init__(self, arr, combinerFunction, isCommutative=True):
		self.n = len(arr)

		def cmb(a, b):
			if a == None:
				return b
			if b == None:
				return a
			return combinerFunction(a, b)
		self.cmb = cmb
		self.sz = self.n if isCommutative else pow(2, (self.n - 1).bit_length())
		self.t = [None] * (2 * self.sz)
		for i in range(self.n):
			self.t[i + self.sz] = arr[i]
		for i in range(self.sz - 1, 0, -1):
			self.pull(i)

	def pull(self, p):
		self.t[p] = self.cmb(self.t[2 * p], self.t[2 * p + 1])

	def update(self, idx, val):
		idx += self.sz
		self.t[idx] = val
		idx //= 2
		while idx > 0:
			self.pull(idx)
			idx //= 2

	def query(self, l, r):
		l += self.sz
		r += self.sz + 1
		a = b = None
		while l < r:
			if l % 2 == 1:
				a = self.cmb(a, self.t[l])
				l += 1
			if r % 2 == 1:
				r -= 1
				b = self.cmb(self.t[r], b)
			l //= 2
			r //= 2
		return self.cmb(a, b)

def getMaxSegTree(arr):
	return SegmentTree(arr, lambda a, b: max(a, b), False)

def getMinSegTree(arr):
	return SegmentTree(arr, lambda a, b: min(a, b), False)

def getSumSegTree(arr):
	return SegmentTree(arr, lambda a, b: a + b, False)
from collections import defaultdict

def countHorizontal(xes, onX):
	xSet = set(xes)
	for (x, y) in onX:
		xSet.add(x)
	xSorted = sorted(xSet)
	N = len(xSorted)
	xMap = {x: i for (i, x) in enumerate(xSorted)}
	for (i, x) in enumerate(xes):
		xes[i] = xMap[x]
	for (i, [x, y]) in enumerate(onX):
		onX[i][0] = xMap[x]
	yItems = defaultdict(list)
	for (x, y) in onX:
		yItems[y].append(x)
	for v in yItems.values():
		v.sort()
	segArr = [0] * (N + 1)
	for (x, y) in onX:
		segArr[x] += 1
	st = getSumSegTree(segArr)
	cnts = 0
	for (x, y) in onX:
		i = -1
		bb = len(xes)
		while bb > 0:
			while i + bb < len(xes) and xes[i + bb] < x:
				i += bb
			bb //= 2
		left = xes[i] + 1
		right = xes[i + 1] - 1
		include = st.query(left, right)
		arr = yItems[y]
		i = -1
		bb = len(arr)
		while bb > 0:
			while i + bb < len(arr) and arr[i + bb] < left:
				i += bb
			bb //= 2
		left2 = i + 1
		i = -1
		bb = len(arr)
		while bb > 0:
			while i + bb < len(arr) and arr[i + bb] <= right:
				i += bb
			bb //= 2
		right2 = i
		exclude = right2 - left2 + 1
		cnts += include - exclude
	return cnts

def readIntArr():
	return [int(x) for x in input().split()]

def makeArr(defaultValFactory, dimensionArr):
	dv = defaultValFactory
	da = dimensionArr
	if len(da) == 1:
		return [dv() for _ in range(da[0])]
	else:
		return [makeArr(dv, da[1:]) for _ in range(da[0])]
