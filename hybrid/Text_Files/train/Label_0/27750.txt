using namespace std;
char c1[55][55];
int main() {
  ios_base::sync_with_stdio(0);
  int a, b, c, d;
  cin >> a >> b >> c >> d;
  a--;
  b--;
  c--;
  d--;
  for (int i = 0; i < 25; i++) {
    for (int j = 0; j < 25; j++) {
      c1[i][j] = 'A';
    }
  }
  for (int i = 25; i < 50; i++) {
    for (int j = 0; j < 25; j++) {
      c1[i][j] = 'C';
    }
  }
  for (int i = 25; i < 50; i++) {
    for (int j = 25; j < 50; j++) {
      c1[i][j] = 'D';
    }
  }
  for (int i = 0; i < 25; i++) {
    for (int j = 25; j < 50; j++) {
      c1[i][j] = 'B';
    }
  }
  if (a > 0) {
    for (int i = 25; i < 50; i += 2) {
      for (int j = 25; j < 50; j += 2) {
        c1[i][j] = 'A';
        a--;
        if (a == 0) {
          break;
        }
      }
      if (a == 0) {
        break;
      }
    }
  }
  if (b > 0) {
    for (int i = 25; i < 50; i += 2) {
      for (int j = 0; j < 25; j += 2) {
        c1[i][j] = 'B';
        b--;
        if (b == 0) {
          break;
        }
      }
      if (b == 0) {
        break;
      }
    }
  }
  if (c > 0) {
    for (int i = 0; i < 25; i += 2) {
      for (int j = 25; j < 50; j += 2) {
        c1[i][j] = 'C';
        c--;
        if (c == 0) {
          break;
        }
      }
      if (c == 0) {
        break;
      }
    }
  }
  if (d > 0) {
    for (int i = 0; i < 25; i += 2) {
      for (int j = 0; j < 25; j += 2) {
        c1[i][j] = 'D';
        d--;
        if (d == 0) {
          break;
        }
      }
      if (d == 0) {
        break;
      }
    }
  }
  cout << 50 << " " << 50 << "\n";
  for (int i = 0; i < 50; i++) {
    for (int j = 0; j < 50; j++) {
      cout << c1[i][j];
    }
    cout << "\n";
  }
  return 0;
}
