n = int(input())
x = list(map(int, input().split()))
x.sort()
if n < 3:
	print(n)
else:
	for i in range(n):
		if x[i] == 1:
			if i != 0 and i != n - 2 and (x[i - 1] == 1) and (x[i + 1] > 1):
				x[i] += 1
			elif x[i + 1] > 1 and x[i - 1] == 1:
				x[i] += 1
		elif i == 0:
			x[i] -= 1
		elif i == n - 1:
			x[i] += 1
		elif x[i - 1] < x[i] - 1 and x[i + 1] < x[i] + 2:
			x[i] -= 1
		elif x[i] == x[i - 1] != x[i + 1]:
			x[i] += 1
	print(len(set(x)))
