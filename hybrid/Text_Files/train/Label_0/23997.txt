#include <bits/stdc++.h>
using namespace std;
const long double EPS = 1e-12;
double __t;
void quit();
set<pair<int, int> > s;
int n, k, q;
vector<int> a;
int main() {
  cin >> n >> k >> q;
  int t0, t1;
  for (int i = 0; i < (int)n; i++) {
    cin >> t0;
    a.push_back(-t0);
  }
  int cnt;
  for (int i = 0; i < (int)q; i++) {
    cin >> t0 >> t1;
    t1--;
    if (t0 == 1) {
      s.insert(make_pair(a[t1], t1));
    } else if (t0 == 2) {
      cnt = 1;
      for (auto j = s.begin(); (j != s.end()) && (cnt <= k); j++) {
        cnt++;
        if (j->second == t1) {
          cout << "YES\n";
          cnt = -1;
          break;
        }
      }
      if (cnt > 0) {
        cout << "NO\n";
      }
    }
  }
  quit();
}
void quit() { exit(0); }
