import java.io.File;
import java.io.FileNotFoundException;
import java.util.Scanner;

public class E {
  public static void main(String[] args) throws FileNotFoundException {
    Scanner scan = new Scanner(System.in);

    int n = scan.nextInt();
    int m = scan.nextInt();

    long[] heads = new long[n];
    long[] tracks = new long[m];

    for (int i = 0; i < n; i++) {
      heads[i] = scan.nextLong();
    }

    for (int i = 0; i < m; i++) {
      tracks[i] = scan.nextLong();
    }

    long lo = 0;
    long hi = Long.MAX_VALUE;

    while (true) {
      long time = (lo + hi) / 2;
      int i = 0;
      int j = 0;
      boolean pathRemained = false;

      boolean con = time == 0;

      for (; i < n && j < m; i++) {
        long h = heads[i];
        long t = tracks[j];

        long righPath = time;
        long leftPath = h - t;

        if (time < leftPath) {
          break;
        }

        if (h > t) {
          righPath = Math.max((time - leftPath) / 2, time - 2 * leftPath);
        }

        while ((tracks[j] < h) || (tracks[j] <= h + righPath)) {
          if (++j == m) {
            if (tracks[j - 1] < h + righPath) {
              pathRemained = true;
            }
            break;
          }
        }
      }

      if (lo >= hi) {
        System.out.println(time);
        break;
      }

      if (j < m) {
        lo = time + 1;
      } else {
        hi = time;
      }
    }
  }

}
