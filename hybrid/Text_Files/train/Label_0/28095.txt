#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, x, y, l[100000], ans = 0;
  vector<int> r;
  long long mod = 1e9 + 7;
  cin >> n >> x >> y;
  for (long long i = 0; i < n; i++) {
    long long x;
    cin >> l[i] >> x;
    r.push_back(x);
    ans += ((r[i] - l[i]) * y) % mod;
    ans %= mod;
  }
  sort(l, l + n);
  sort(r.begin(), r.end());
  long long i = 1, j = 0, cnt = 0;
  for (; i < n; i++) {
    int l1 = 0, r1 = r.size() - 1;
    while (l1 < r1) {
      j = (l1 + r1) / 2;
      if (r[j] < l[i] and (j == r.size() - 1 or r[j + 1] >= l[i])) break;
      if (r[j] < l[i]) l1 = j;
      if (r[j] >= l[i]) r1 = j;
    }
    if (r[j] < l[i] and (l[i] - r[j]) * y <= x) {
      cnt++;
      ans += ((l[i] - r[j]) * y) % mod;
      ans %= mod;
      r.erase(r.begin() + j, r.begin() + j + 1);
    }
  }
  ans += ((n - cnt) * x) % mod;
  cout << ans % mod;
  return 0;
}
