from math import *

def main():
	(n, m) = [int(x) for x in input().split()]
	if n == 0:
		print(0, 1)
		print(0, m)
		print(0, 0)
		print(0, m - 1)
	elif m == 0:
		print(1, 0)
		print(n, 0)
		print(0, 0)
		print(n - 1, 0)
	else:
		r1 = 2 * sqrt(n * n + m * m) + max(n, m)
		r2 = 2 * sqrt((n - 1) * (n - 1) + m * m) + sqrt(n * n + m * m)
		r3 = 2 * sqrt((m - 1) * (m - 1) + n * n) + sqrt(n * n + m * m)
		if r3 > max(r1, r2):
			print(n, 1)
			print(0, m)
			print(n, 0)
			print(0, m - 1)
		elif r2 > r1:
			print(n - 1, m)
			print(0, 0)
			print(n, m)
			print(1, 0)
		else:
			print(0, 0)
			print(n, m)
			if n < m:
				print(n, 0)
				print(0, m)
			else:
				print(0, m)
				print(n, 0)
main()
