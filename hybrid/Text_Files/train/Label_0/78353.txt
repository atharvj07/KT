import re
n = int(input())
temp = []
count = 0
for i in range(n):
	a = str(input())
	temp.append(a)
new = sorted(temp, key=len)[::-1]
for x in range(len(new) - 1):
	if re.search(str(new[x + 1]), str(new[x])) == None:
		break
	elif re.search(str(new[x + 1]), str(new[x])) != None:
		count += 1
if count != len(new) - 1:
	print('NO')
elif count == len(new) - 1:
	print('YES')
	new = new[::-1]
	for x in new:
		print(x)
