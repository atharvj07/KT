n = int(input())
pts = []
for i in range(n):
	(a, b, c) = map(int, input().split())
	pts.append((a, b, c, i))
dicter = [(float('inf'), 0)] * n
was2 = [0] * n
for el1 in pts:
	if not was2[el1[3]]:
		for el2 in pts:
			if not was2[el2[3]]:
				if el1[3] != el2[3]:
					ku = abs(el1[0] - el2[0]) + abs(el1[1] - el2[1]) + abs(el1[2] - el2[2])
					if ku < dicter[el1[3]][0]:
						dicter[el1[3]] = (ku, el2[3])
		was2[el1[3]] = 1
		was2[dicter[el1[3]][1]] = 1
was = [0] * n
for i in range(n):
	if dicter[i][0] != float('inf'):
		print(dicter[i][1] + 1, i + 1)
