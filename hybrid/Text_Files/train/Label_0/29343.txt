#include <bits/stdc++.h>
using namespace std;
const long long INF = 2000000000LL;
const long double EPS = 1e-9;
int prnt = 0;
struct BIT {
  long long tr[100000 + 7];
  int upto = 100000 + 7 - 1;
  void update(int p, long long x) {
    while (p <= upto) {
      tr[p] += x;
      p += (p & -p);
    }
  }
  long long query(int p) {
    long long sum = 0;
    while (p > 0) {
      sum += tr[p];
      p -= (p & -p);
    }
    return sum;
  }
} t, c;
long long cnt[100000 + 7], eat[100000 + 7], cost[100000 + 7];
long long id[100000 + 7], who[100000 + 7], dp[100000 + 7];
vector<int> g[100000 + 7];
int n;
void dfs(int u, long long tym) {
  t.update(id[u], cnt[u] * eat[u]);
  c.update(id[u], cnt[u]);
  tym -= 2LL * cost[u];
  int lo = 0, hi = n;
  while (lo < hi) {
    int mid = (lo + hi + 1) / 2;
    long long temp = t.query(mid);
    if (temp <= tym)
      lo = mid;
    else
      hi = mid - 1;
  }
  long long cur = c.query(lo);
  long long rem = tym - t.query(lo);
  if (lo < n) {
    int nx = who[lo + 1];
    cur += min(c.query(lo + 1) - cur, rem / eat[nx]);
  }
  dp[u] = cur;
  for (int v : g[u]) dfs(v, tym);
  t.update(id[u], -cnt[u] * eat[u]);
  c.update(id[u], -cnt[u]);
}
void go(int u) {
  long long best = 0, bet = 0;
  for (int v : g[u]) {
    go(v);
    if (dp[v] > best) {
      bet = best;
      best = dp[v];
    } else if (dp[v] > bet)
      bet = dp[v];
  }
  if (u == 1)
    dp[u] = max(dp[u], best);
  else
    dp[u] = max(dp[u], bet);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  long long T;
  cin >> n >> T;
  for (int i = 1; i <= n; ++i) cin >> cnt[i];
  for (int i = 1; i <= n; ++i) cin >> eat[i];
  for (int i = 2; i <= n; ++i) {
    int p;
    cin >> p >> cost[i];
    g[p].push_back(i);
  }
  cost[1] = 0;
  vector<pair<int, int> > v;
  for (int i = 1; i <= n; ++i) v.push_back(make_pair(eat[i], i));
  sort((v).begin(), (v).end());
  for (int i = 0; i < (int)v.size(); ++i)
    id[v[i].second] = i + 1, who[i + 1] = v[i].second;
  dfs(1, T);
  go(1);
  cout << dp[1] << "\n";
  return 0;
}
