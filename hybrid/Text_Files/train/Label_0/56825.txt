    public static class Task {
        static int mod = 1000000007;
        public static long modpow(long a, long b){
            if (b == 0) return 1;
            if (b == 1) return a % mod;
            if (b % 2 == 0) {
                return modpow(a * a % mod, b / 2) % mod;
            } else {
                return a * modpow(a * a % mod, b / 2) % mod;
            }
        }


        public static void solve(MyScanner sc, PrintWriter pw) throws IOException {

            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] S1 = new int[n];
            int[] S2 = new int[n];
            long p = 0;
            long qm = 0;
            for (int i = 0; i < n; i++) {
                S1[i] = sc.nextInt();
            }
            for (int i = 0; i < n; i++) {
                S2[i] = sc.nextInt();
            }
            for (int i = n - 1; i >= 0; i--) {
                int a = S1[i];
                int b = S2[i];
                if (a == 0 && b == 0) {
                    qm ++;
                    long pow = modpow(m, qm - 1);
                    long add = pow * (m - 1);
                    add %= mod;
                    p += add;
                    p %= mod;
                } else if (a == 0) {
                    int p1 = m - b;
                    qm++;
                    long pow = modpow(m, qm - 1);
                    long add = pow * 2 * p1;
                    add %= mod;
                    p += add;
                    p %= mod;

                } else if (b == 0) {
                    int p1 = a - 1;
                    qm++;
                    long pow = modpow(m, qm - 1);
                    long add = pow * 2 * p1;
                    add %= mod;
                    p += add;
                    p %= mod;
                } else {
                    if (a == b) {

                    } else if (a > b) {
                        p = 2;
                        qm = 0;
                    } else {
                        p = 0;
                        qm = 0;
                    }
                }
            }
            long q = modpow(m, qm);
            q *= 2; q %= mod;
            long q1 = modpow(q, mod - 2);
            long r = q1 * p;
            r %= mod;

            pw.println(r);
        }

    }




    static class MyScanner {
        StringTokenizer st;
        BufferedReader br;

        public MyScanner(InputStream s){  br = new BufferedReader(new InputStreamReader(s));}

        public MyScanner(FileReader s) throws FileNotFoundException {br = new BufferedReader(s);}

        public String next() throws IOException
        {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        public int nextInt() throws IOException {return Integer.parseInt(next());}

        public long nextLong() throws IOException {return Long.parseLong(next());}

        public String nextLine() throws IOException {return br.readLine();}

        public double nextDouble() throws IOException { return Double.parseDouble(next()); }

        public boolean ready() throws IOException {return br.ready();}
    }
