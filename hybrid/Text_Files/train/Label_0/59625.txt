import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner (System.in);
		int arr[] = new int[277];
		Arrays.fill(arr,-1);
		int n=sc.nextInt(), k = sc.nextInt();
		for(int i=0;i<n;i++){
		    
		    System.out.println();
			int t = sc.nextInt();
			if(arr[t]!=-1){
				System.out.print(arr[t]+" ");
			} else{
				for(int ii=0;ii<=t;ii++){
					if(Math.abs(ii-t)>=k) continue;
					if(arr[ii]!=-1 && arr[ii]!=ii) continue;
					for(int l=ii;l<=t;l++){
						arr[l]=ii;
					}
				}
				System.out.print(arr[t]+" ");
			}
		}
		System.out.println();//
	}
}
