import math

def gcd(a, b):
	if a < b:
		(a, b) = (b, a)
	while a % b != 0:
		(a, b) = (b, a % b)
	return b
t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	if n == 1:
		print(0)
		continue
	print(n)
	for i in range(1, n):
		if a[i] < a[i - 1]:
			print(i, i + 1, 10 ** 9 + 7 + i - 1, a[i])
		else:
			print(i, i + 1, 10 ** 9 + 7 + i - 1, a[i - 1])
			a[i] = a[i - 1]
		a[i - 1] = 10 ** 9 + 7 - i - 1
	print(n - 1, n, 1073676287, a[-1])
