from bisect import bisect_left
mod = 998244353
n = int(input())
a = [int(x) for x in input().split()]
s = sorted(a)
a = [bisect_left(s, x) for x in a]
(first, last) = ([n] * n, [-1] * n)
for i in range(n):
	first[a[n - 1 - i]] = n - 1 - i
	last[a[i]] = i
ans = 1
cur = last[a[0]]
for i in range(n):
	if first[a[i]] == i:
		if i > cur:
			ans <<= 1
			if ans > mod:
				ans -= mod
		cur = max(cur, last[a[i]])
print(ans)
