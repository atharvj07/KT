import java.util.Scanner;

public class temp {

	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int n = s.nextInt();
		int[] a = new int[n];
		
		long sum = 0;
		for (int i =0 ; i<n ; i++){
			a[i] = s.nextInt();
			sum = sum + a[i];			
		}
        long sumt = sum/3;
        if (sum%3!=0)
            System.out.print(0);
        else{
            long ans = 0;
            int count = 0;
            long sumtemp = 0;
            int[] endcount = new int[n];
            for (int i =n-1 ; i>=0 ; i--){
                sumtemp += a[i];
                if (sumtemp==sumt){
                    count++;                    
                }     
                endcount[i] = count;
            }
            
            sumtemp = 0;
            for (int i =0 ; i<n-2 ; i++){
                sumtemp += a[i];
                if (sumtemp == sumt){
                    ans += endcount[i+2];
                }
            }
            System.out.print(ans);
        }    
	}

}
