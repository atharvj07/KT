import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.StringTokenizer;

/**
 * @author Don Li
 */
public class DNAEvolution {
    
    int N = (int) 1e5 + 10;
    String P = "ATGC";
    
    int[][][] bit = new int[4][10][N];
    
    int n;
    
    void solve() {
        char[] s = in.nextToken().toCharArray();
        n = s.length;
        
        for (int len = 1; len <= 10; len++) {
            for (int i = 0; i < len; i++) {
                for (int j = i; j < n; j += len) {
                    int x = P.indexOf(s[j]);
                    update(x, len, j, 1);
                }
            }
        }
        
        int T = in.nextInt();
        while (T-- > 0) {
            int type = in.nextInt();
            if (type == 1) {
                int x = in.nextInt() - 1;
                char c = in.nextToken().charAt(0);
                if (s[x] == c) continue;
                for (int len = 1; len <= 10; len++) {
                    update(P.indexOf(s[x]), len, x, -1);
                    update(P.indexOf(c), len, x, 1);
                }
                s[x] = c;
            } else {
                int l = in.nextInt() - 1, r = in.nextInt();
                char[] e = in.nextToken().toCharArray();
                int sum = 0, m = e.length;
                if (r - l < m) {
                    for (int i = 0; i < r - l; i++) if (s[l + i] == e[i]) sum++;
                } else {
                    for (int i = 0; i < m; i++) {
                        int start = map(m, l + i), repeat = (r - l) / m;
                        if (i < (r - l) % m) repeat++;
                        sum += query(P.indexOf(e[i]), m, start, start + repeat - 1);
                    }
                }
                out.println(sum);
            }
        }
    }
    
    void update(int type, int len, int pos, int delta) {
        int i = map(len, pos);
        for (int x = i; x < N; x = x | (x + 1)) {
            bit[type][len - 1][x] += delta;
        }
    }
    
    int query(int type, int len, int l, int r) {
        return sum(type, len, r) - sum(type, len, l - 1);
    }
    
    int sum(int type, int len, int i) {
        int ret = 0;
        for (int x = i; x >= 0; x = (x & (x + 1)) - 1) {
            ret += bit[type][len - 1][x];
        }
        return ret;
    }
    
    int map(int len, int pos) {
        int q = pos / len, r = pos % len;
        int prev = 0;
        if (r <= n % len) {
            prev += r * (n / len + 1);
        } else {
            prev += (n % len) * (n / len + 1) + (r - n % len) * (n / len);
        }
        return prev + q;
    }
    
    public static void main(String[] args) {
        in = new FastScanner(new BufferedReader(new InputStreamReader(System.in)));
        out = new PrintWriter(System.out);
        new DNAEvolution().solve();
        out.close();
    }
    
    static FastScanner in;
    static PrintWriter out;
    
    static class FastScanner {
        BufferedReader in;
        StringTokenizer st;
        
        public FastScanner(BufferedReader in) {
            this.in = in;
        }
        
        public String nextToken() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    st = new StringTokenizer(in.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
        
        public int nextInt() {
            return Integer.parseInt(nextToken());
        }
        
        public long nextLong() {
            return Long.parseLong(nextToken());
        }
        
        public double nextDouble() {
            return Double.parseDouble(nextToken());
        }
    }
}
