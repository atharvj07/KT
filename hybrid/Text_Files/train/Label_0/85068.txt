import io
import os
from collections import Counter, defaultdict, deque

class UnionFind:

	def __init__(self, N):
		self.p = [-1 for i in range(N)]

	def find(self, i):
		if self.p[i] >= 0:
			self.p[i] = self.find(self.p[i])
			return self.p[i]
		else:
			return i

	def union(self, i, j):
		root1 = self.find(j)
		root2 = self.find(i)
		if root1 == root2:
			return
		size1 = -self.p[root1]
		size2 = -self.p[root2]
		if size1 < size2:
			self.p[root1] = root2
			self.p[root2] = -(size1 + size2)
		else:
			self.p[root2] = root1
			self.p[root1] = -(size1 + size2)

	def getComponentSize(self, i):
		return -self.p[self.find(i)]

def solve(N, S):
	graph = [[] for i in range(2 * (N + 1))]
	for i in range(N):
		j = i + 1
		if S[i] == 'L':
			graph[j].append(N + 1 + i)
			graph[N + 1 + i].append(j)
		else:
			graph[i].append(N + 1 + j)
			graph[N + 1 + j].append(i)
	uf = UnionFind(len(graph))
	for u in range(len(graph)):
		for v in graph[u]:
			uf.union(u, v)
	return ' '.join((str(uf.getComponentSize(u)) for u in range(N + 1)))
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
TC = int(input())
for tc in range(1, TC + 1):
	(N,) = [int(x) for x in input().split()]
	S = input().decode().rstrip()
	ans = solve(N, S)
	print(ans)
