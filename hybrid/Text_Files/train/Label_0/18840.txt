#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5;
int N, res;
int ar[MAXN];
int main() {
  cin >> N;
  if (N % 4 > 1) {
    cout << -1 << endl;
    return 0;
  }
  for (int i = 1, k = 0; k + 4 <= N; i += 2, k += 4) {
    ar[i] = i + 1;
    ar[i + 1] = N - i + 1;
    ar[N - i + 1] = N - i;
    ar[N - i] = i;
  }
  if (N % 2) ar[N / 2 + 1] = N / 2 + 1;
  for (int i = 1; i <= N; i++) printf("%d ", ar[i]);
  return 0;
}
