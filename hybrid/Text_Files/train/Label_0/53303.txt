
import java.util.*;

//	import java.text.*;

public class helloworld{
	
	public static void main(String args[]) {
		
		Scanner scan = new Scanner(System.in); 
		
		int a,b,c;
		
		int max,ans;
		
		a = scan.nextInt();
		
		b = scan.nextInt();
		
		c = scan.nextInt();
		
		max = a;
		
		if(max<b) max=b;
		
		if(max<c) max=c;
		
		if((ans=2*max-(a+b+c))<0) System.out.println(0);
		
		else System.out.println(ans+1);
		
		scan.close();
		
		Runtime.getRuntime().exit(0);
			
	}
	
}
