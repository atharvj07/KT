import math
(n, k) = map(int, input().split())
s = input()
alfa = [0 for i in range(26)]
set = []
for i in s:
	if alfa[ord(i) - 97] == 0:
		set.append(i)
		alfa[ord(i) - 97] += 1
set = sorted(set)
change = False
if n < k:
	news = ''
	news = s
	news += set[0] * (k - n)
	print(news)
elif n > k:
	newv = ''
	ind = k - 1
	ultima = s[ind]
	while change == False:
		if ultima == set[len(set) - 1]:
			ind -= 1
			ultima = s[ind]
		else:
			for i in range(len(set)):
				if set[i] == ultima:
					ultima = set[i + 1]
					change = True
					break
	newv += s[0:ind]
	newv += ultima
	newv += set[0] * (k - 1 - ind)
	print(newv)
else:
	flag = False
	ultima = ''
	for i in range(n - 1, -1, -1):
		novov = ''
		ultimo = i
		if s[i] != set[len(set) - 1]:
			for j in range(len(set)):
				if set[j] == s[i]:
					ultima = set[j + 1]
					flag = True
					break
		if flag:
			novov += s[0:i]
			novov += ultima
			novov += set[0] * (k - 1 - i)
			print(novov)
			break
