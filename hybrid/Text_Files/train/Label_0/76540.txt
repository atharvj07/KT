import sys
input = sys.stdin.readline

class Node:

	def __init__(self):
		self.data = 0
		self.parent = None
		self.rank = 0
d = {}

def makeSet(data):
	node = Node()
	node.data = data
	node.parent = node
	d[data] = node

def get(val):
	return val * (val - 1) // 2

def union(data1, data2, ele, res):
	node1 = d[data1]
	node2 = d[data2]
	parent1 = find(node1)
	parent2 = find(node2)
	temp = res
	temp -= get(ele[parent1.data])
	temp -= get(ele[parent2.data])
	if parent1.rank >= parent2.rank:
		parent1.rank = parent1.rank + 1 if parent1.rank == parent2.rank else parent1.rank
		parent2.parent = parent1
		ele[parent1.data] += ele[parent2.data]
	else:
		parent1.parent = parent2
		ele[parent2.data] += ele[parent1.data]
	temp += get(ele[parent1.parent.data])
	return temp

def findData(data):
	return find(d[data]).data

def find(node):
	parent = node.parent
	if parent == node:
		return node
	node.parent = find(node.parent)
	return node.parent
(n, q) = map(int, input().split())
ar = []
for i in range(n - 1):
	(a, b, w) = map(int, input().split())
	ar.append((a, b, w))
qr = [int(x) for x in input().split()]
for i in range(q):
	qr[i] = (qr[i], i)
ar.sort(key=lambda x: x[2])
qr.sort(key=lambda x: x[0])
for i in range(1, n + 1):
	makeSet(i)
ele = [1] * (n + 1)
ans = []
res = 0
pos = 0
for i in range(q):
	while pos < n - 1 and ar[pos][2] <= qr[i][0]:
		res = union(ar[pos][0], ar[pos][1], ele, res)
		pos += 1
	ans.append((str(res), qr[i][1]))
ans.sort(key=lambda x: x[1])
r = []
for i in ans:
	r.append(i[0])
print(' '.join(r))
