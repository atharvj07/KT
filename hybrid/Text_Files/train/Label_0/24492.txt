#include <iostream>
#include <cstdio>
#include <cstdlib>
using namespace std;

int a[10], b[10];

int main()
{
//	freopen("input.txt", "r", stdin);
//	freopen("output.txt", "w", stdout);

	for (int i = 0; i < 6; i++)
		scanf("%d", &a[i]);
	for (int i = 0; i < 6; i++)
		scanf("%d", &b[i]);


	double ans = 0;
	for (int id = 0; id < 6; id++)
	{
		if (a[id] + b[id] == 0) continue;
		for (int mask = 0; mask < 64; mask++)
		{
			double sum = 0;
			double res = 0;
			for (int i = 0; i < 6; i++)
			{
				if (id == i) continue;
				sum -= (a[i] + b[i]) * ((mask >> i) & 1) - b[i];
				res += a[i] * ((mask >> i) & 1);
			}
			sum += b[id];
			double q = a[id] + b[id];
			if (sum < 0 || sum > q) continue;
			double p = sum / q;
			res += a[id] * p;
			ans = max(ans, res / 100);
		}
	}
	printf("%.14lf\n", ans);

	return 0;
}
