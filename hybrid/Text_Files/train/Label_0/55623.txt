
import java.util.Scanner;

public class Solution {


    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int n = scan.nextInt();
        int h = scan.nextInt();
        int counter = 0 ;
        for(int i = 0 ;i<n ;i++) {
            int temp = scan.nextInt();
            counter += (temp>h)?2:1;
        }
        System.out.println(counter);
    }
}
