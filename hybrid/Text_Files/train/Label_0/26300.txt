#include <bits/stdc++.h>
#pragma GCC optimize("Ofast")
using namespace std;
int n, k;
char a[2010][2010];
int pre[2010][2010];
bool ok[2010][2010];
bool cand[2010][2010];
int main() {
  cin.tie(0);
  ios_base::sync_with_stdio(false);
  cin >> n >> k;
  for (int i = 0; i < int(n); i++)
    for (int j = 0; j < int(n); j++) {
      char x;
      cin >> x;
      a[i][j] = x;
    }
  int maxi = -1;
  vector<pair<int, int> > v;
  for (int i = 0; i < int(n); i++) {
    for (int j = 0; j < int(n); j++) {
      int ans = 0;
      if (a[i][j] == 'a') ans++;
      int addmax = 0;
      if (i > 0) addmax = max(addmax, pre[i - 1][j]);
      if (j > 0) addmax = max(addmax, pre[i][j - 1]);
      pre[i][j] = ans + addmax;
    }
  }
  for (int i = 0; i < int(n); i++) {
    for (int j = 0; j < int(n); j++) {
      int need = (i + j + 1) - pre[i][j];
      if (need <= k) {
        ok[i][j] = 1;
        if (i + j > maxi) {
          maxi = i + j;
          v.clear();
          v.push_back(make_pair(i, j));
        } else if (i + j == maxi)
          v.push_back(make_pair(i, j));
      }
    }
  }
  if (maxi == -1) v.push_back(make_pair(0, 0));
  for (int i = 0; i < int(maxi + 1); i++) cout << 'a';
  if (maxi == -1) cout << a[0][0];
  for (int i = 0; i < int(v.size()); i++) cand[v[i].first][v[i].second] = 1;
  int lvl = max(maxi, 0);
  while (lvl < n * 2 - 2) {
    char minichar = 'z' + 1;
    for (int i = max(0, lvl + 1 - (n - 1)); i <= min(n - 1, lvl + 1); i++) {
      int j = lvl + 1 - i;
      if ((i > 0 && cand[i - 1][j]) || (j > 0 && cand[i][j - 1]))
        minichar = min(minichar, a[i][j]);
    }
    cout << minichar;
    for (int i = max(0, lvl + 1 - (n - 1)); i <= min(n - 1, lvl + 1); i++) {
      int j = lvl + 1 - i;
      if (a[i][j] == minichar &&
          ((i > 0 && cand[i - 1][j]) || (j > 0 && cand[i][j - 1])))
        cand[i][j] = 1;
    }
    lvl++;
  }
  cout << endl;
}
