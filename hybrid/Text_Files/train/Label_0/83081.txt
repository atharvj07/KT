def parse():
	inputs = []
	nbCase = int(input())
	for i in range(nbCase):
		line = input()
		inputs.append(list(map(int, line.split())))
	return inputs

def write_tab(tab):
	w = ''
	for i in range(len(tab)):
		w += str(tab[i]) + ' '
	return w

def solve():
	inputs = parse()
	for case in inputs:
		rounds = decomposition(case[0])
		print(len(rounds))
		print(write_tab(rounds))

def decomposition(number):
	rounds = []
	b = 1
	while number != 0:
		round_b = number % 10 * b
		if round_b != 0:
			rounds.append(round_b)
		number = number // 10
		b *= 10
	return list(rounds)
solve()
