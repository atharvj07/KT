#include <bits/stdc++.h>
using namespace std;
char s[10010];
int n, fail[10010], f[10010];
int getlen(int x) {
  if (x == 0) return 1;
  int re = 0;
  while (x != 0) {
    re++;
    x /= 10;
  }
  return re;
}
int main() {
  scanf("%s", s + 1);
  n = strlen(s + 1);
  for (int i = 1; i <= n; i++) f[i] = 10000000;
  for (int i = 1; i <= n; i++) {
    fail[i - 1] = i - 2;
    for (int j = i; j <= n; j++) {
      int p = fail[j - 1];
      while (p != i - 2 && s[p + 1] != s[j]) p = fail[p];
      fail[j] = p + 1;
      int d = fail[j];
      int len = j - i + 1, len1 = j - d;
      if (d >= i && len1 * 2 <= len && len % len1 == 0)
        f[j] = min(f[j], f[i - 1] + len1 + getlen(len / len1));
      f[j] = min(f[j], j - i + 1 + f[i - 1] + 1);
    }
  }
  printf("%d\n", f[n]);
  return 0;
}
