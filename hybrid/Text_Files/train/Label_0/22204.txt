#include <bits/stdc++.h>
using namespace std;
long long abs1(long long h) {
  if (h < 0)
    return -h;
  else
    return h;
}
int main() {
  long long i, j, n, m, e, k, ot, x, y, i1, i2, i3, i4;
  vector<int> a, q, g;
  vector<vector<int> > w;
  vector<int> d;
  cin >> n;
  for (i = 0; i <= n + 4; i++) {
    q.push_back(0);
    w.push_back(d);
  }
  q[1] = 1;
  q[n + 1] = 1;
  for (i = 0; i < 2 * n; i++) {
    cin >> x >> y;
    w[x].push_back(y);
    w[y].push_back(x);
  }
  for (i = 1; i <= n; i++) {
    if (w[i].size() != 4) {
      cout << -1;
      return 0;
    }
  }
  for (i1 = 1; i1 <= 4; i1++) {
    for (i2 = 1; i2 <= 4; i2++) {
      for (i3 = 1; i3 <= 4; i3++) {
        for (i4 = 1; i4 <= 4; i4++) {
          if (i1 != i2 && i1 != i3 && i1 != i4 && i2 != i3 && i2 != i4 &&
              i3 != i4) {
            g.clear();
            for (i = 0; i < n + 2; i++) g.push_back(0);
            q[2] = w[1][i1 - 1];
            q[3] = w[1][i2 - 1];
            q[n] = w[1][i3 - 1];
            q[n - 1] = w[1][i4 - 1];
            q[0] = q[n];
            q[n + 2] = q[2];
            int z = 0;
            for (i = 2; i <= n - 2; i++) {
              e = 0;
              k = 0;
              for (j = 0; j < 4; j++) {
                if (w[q[i]][j] != q[i + 1] && w[q[i]][j] != q[i - 1] &&
                    w[q[i]][j] != q[i - 2]) {
                  e++;
                  k = w[q[i]][j];
                }
              }
              if (e != 1) {
                z = 1;
                break;
              } else
                q[i + 2] = k;
            }
            q[0] = q[n];
            q[n + 2] = q[2];
            if (z == 0) {
              z = 0;
              for (i = 2; i <= n; i++) {
                e = 0;
                for (j = 0; j < 4; j++) {
                  if (w[q[i]][j] != q[i + 1] && w[q[i]][j] != q[i - 1] &&
                      w[q[i]][j] != q[i - 2] && w[q[i]][j] != q[i + 2])
                    e++;
                }
                if (e != 0) {
                  z = 1;
                  break;
                }
              }
              for (i = 1; i <= n; i++) {
                g[q[i]]++;
                if (g[q[i]] == 2) {
                  z = 1;
                  break;
                }
              }
              if (z == 0) {
                for (i = 1; i <= n; i++) cout << q[i] << " ";
                return 0;
              }
            }
          }
        }
      }
    }
  }
  cout << -1;
  return 0;
}
