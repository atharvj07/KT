#include <bits/stdc++.h>
using namespace std;
using pii = pair<int, int>;
int64_t k, b, n, t;
int main() {
  ios_base::sync_with_stdio(false), cin.tie(0), cout << fixed;
  cin >> k >> b >> n >> t;
  int64_t z = 1;
  int i;
  for (i = 0; i < n; i++) {
    if (k * z + b > t) break;
    z = k * z + b;
  }
  cout << n - i << endl;
  return 0;
}
