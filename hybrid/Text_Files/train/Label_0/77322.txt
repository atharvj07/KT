def solution(x, h):
	if len(x) == 1:
		return 1
	last_x = x[0]
	ans = 2
	for i in range(1, N - 1):
		if x[i] - h[i] > last_x:
			last_x = x[i]
			ans += 1
		elif x[i] + h[i] < x[i + 1]:
			last_x = x[i] + h[i]
			ans += 1
		else:
			last_x = x[i]
	return ans
N = int(input())
x = []
h = []
for i in range(N):
	(a, b) = map(int, input().split())
	x.append(a)
	h.append(b)
print(solution(x, h))
