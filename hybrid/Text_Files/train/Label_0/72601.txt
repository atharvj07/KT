def hello():
	s = list(input())
	l = len(s)
	import string
	upper = list(string.ascii_uppercase)
	copy = upper[:]
	for i in range(l - 25):
		copy = upper[:]
		count = 0
		index = []
		for j in range(i, i + 26):
			if s[j] == '?':
				count += 1
				index.append(j)
			elif s[j] in copy:
				copy.remove(s[j])
			else:
				break
		if count == len(copy):
			for k in range(count):
				s[index[k]] = copy.pop()
			for o in range(l):
				if s[o] == '?':
					s[o] = 'A'
			break
	else:
		print(-1)
		return
	print(''.join(s))
hello()
