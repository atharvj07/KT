#include <iostream>
#include <cstdint>
#include <vector>
#include <utility>

using vec64 = std::vector<int64_t>;

int64_t solve(int n,const vec64 &as) {
  vec64 maxs(n+1);
  int64_t min=0,max=0;
  for ( int i=n; i>=0; i-- ) {
    auto e=as[i];
    min=(min+1)/2+e;
    maxs[i]=max+=e;
  }
  if ( min>1||max<1 ) {
    return -1;
  }
  int64_t t=1,ans=1;
  for ( int i=1; i<=n; i++ ) {
    ans += t = std::min((t-as[i-1])*2,maxs[i]);
  }
  return ans;
}

int main() {
  int n;
  std::cin >> n;

  vec64 as(n+1);
  for ( auto &e: as ) {
    std::cin >> e;
  }

  std::cout << solve(n,as) << std::endl;
}
