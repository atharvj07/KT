long long expo(long long x, long long y) {
  long long res = 1;
  while (y) {
    if (y % 2) res = (res * x % 1000000007) % 1000000007;
    x = (x * x) % 1000000007;
    y /= 2;
  }
  return res;
}
bool isPrime(long long n) {
  if (n <= 1) return false;
  if (n <= 3) return true;
  if (n % 2 == 0 || n % 3 == 0) return false;
  for (long long i = 5; i * i <= n; i = i + 6)
    if (n % i == 0 || n % (i + 2) == 0) return false;
  return true;
}
long long nextPrime(long long N) {
  if (N <= 1) return 2;
  long long prime = N;
  bool found = false;
  while (!found) {
    prime++;
    if (isPrime(prime)) found = true;
  }
  return prime;
}

long long fpow(long long x, long long y, long long p = 1000000007) {
  x %= p;
  long long sum = 1;
  while (y) {
    if (y & 1) sum = sum * x;
    sum %= p;
    y = y >> 1;
    x = x * x;
    x %= p;
  }
  return sum;
}
bool isPerSquare(long double a) {
  if (a < 0) return false;
  long long sr = sqrt(a);
  return (sr * sr == a);
}
void facto() {
  fact[0] = 1;
  fact[1] = 1;
  for (long long i = 2; i < 1000006; i++)
    fact[i] = (fact[i - 1] * i) % 1000000007;
}
void solve() {
  long long n;
  cin >> n;
  ;
  long long k;
  cin >> k;
  ;
  std::vector<long long> a(n);
  for (long long i = 0; i < n; i++) cin >> a[i];
  ;
  string s;
  cin >> s;
  ;
  long long l = 0, r = 0, ans = 0;
  while (l < n) {
    while (l < n && r < n && s[l] == s[r]) r++;
    long long i = l, j = r;
    sort(a.begin() + i, a.begin() + j);
    reverse(a.begin() + i, a.begin() + j);
    for (long long p = i; p < min(j, i + k); p++) ans += a[p];
    l = r;
  }
  print(ans);
}
