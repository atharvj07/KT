from math import factorial
t = int(input())
for i in range(t):
	n = int(input())
	A = list(map(int, input().split()))
	k = 0
	cnt = A.count(0)
	for j in range(1, cnt + 1):
		k += factorial(cnt) // (factorial(j) * factorial(cnt - j))
	if k == 0:
		print(A.count(1))
	else:
		print(A.count(1) * (k + 1))
