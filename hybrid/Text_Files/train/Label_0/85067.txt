import sys
from collections import deque
from math import gcd
for _ in range(int(input())):
	n = int(input())
	s = input()
	A = [[1, 1] for _ in range(n + 1)]
	ans = [0] * (n + 1)
	for i in range(1, n + 1):
		if s[i - 1] == 'L':
			A[i][0] = 1 + A[i - 1][1]
		else:
			A[i][1] = 1 + A[i - 1][0]
		ans[i] += A[i][0]
	A = [[1, 1] for _ in range(n + 1)]
	for i in range(n - 1, -1, -1):
		if s[i] == 'R':
			A[i][0] = 1 + A[i + 1][1]
		else:
			A[i][1] = 1 + A[i + 1][0]
		ans[i] += A[i][0] - (1 if i != 0 else 0)
	print(*ans)
