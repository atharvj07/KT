#include<iostream>
#include<cstdlib>

using namespace std;

int main(){
  int n,m;
  cin>>n>>m;
  int d[100000]={};
  for(int i=1;i<n;i++){
    cin>>d[i];
    d[i]+=d[i-1];
  }
  int p=0;
  int s=0;
  while(m--){
    int a;
    cin>>a;
    s=(s+abs(d[p]-d[p+a]))%100000;
    p+=a;
  }
  cout<<s<<endl;
  return 0;
}
