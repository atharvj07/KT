N = str(input())
S = ''
a = 0
b = 0
c = 0
d = 0
for i in range(len(N)):
	if N[i] == 'h':
		S = S + N[i]
		a = i
		break
for j in range(a + 1, len(N)):
	if N[j] == 'e':
		S = S + N[j]
		b = j
		break
for k in range(b + 1, len(N)):
	if N[k] == 'i':
		S = S + N[k]
		c = k
		break
for l in range(c + 1, len(N)):
	if N[l] == 'd':
		S = S + N[l]
		d = l
		break
for m in range(d + 1, len(N)):
	if N[m] == 'i':
		S = S + N[m]
		break
if S == 'heidi':
	print('YES')
else:
	print('NO')
