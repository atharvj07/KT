int main(){
    int n;
    string s;
    cin>>n>>s;
    int ans=0;
    rep(i,n-1){
        int X[26]={0},Y[26]={0};
        for(int j=0;j<=i;j++) X[s[j]-'a']++;
        for(int j=i+1;j<n;j++) Y[s[j]-'a']++;
        int m=0;

    }
