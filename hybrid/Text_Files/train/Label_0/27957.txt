#include <bits/stdc++.h>
using namespace std;
const int Z = (int)3e3 + 228;
const int N = (int)3e5 + 228;
const int INF = (int)1e9 + 228;
const int MOD = (int)1e9 + 7;
map<int, vector<int> > mp;
int a[N], t[4 * N], ans[N];
void build(int v, int tl, int tr) {
  if (tl == tr) {
    t[v] = 1;
    return;
  }
  int mid = (tl + tr) / 2;
  build(2 * v, tl, mid);
  build(2 * v + 1, mid + 1, tr);
  t[v] = t[2 * v] + t[2 * v + 1];
}
void update(int v, int tl, int tr, int pos) {
  if (tl == tr) {
    t[v] = 0;
    return;
  }
  int mid = (tl + tr) / 2;
  if (pos <= mid)
    update(2 * v, tl, mid, pos);
  else
    update(2 * v + 1, mid + 1, tr, pos);
  t[v] = t[2 * v] + t[2 * v + 1];
}
int get_kth(int v, int tl, int tr, int kth) {
  if (tl == tr) return a[tl];
  int mid = (tl + tr) / 2;
  if (kth <= t[2 * v])
    return get_kth(2 * v, tl, mid, kth);
  else
    return get_kth(2 * v + 1, mid + 1, tr, kth - t[2 * v]);
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  vector<int> b;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    mp[a[i]].push_back(i);
    b.push_back(a[i]);
  }
  build(1, 1, n);
  vector<pair<pair<int, int>, int> > q;
  int m;
  cin >> m;
  for (int i = 1; i <= m; i++) {
    int x, y;
    cin >> x >> y;
    q.push_back({{x, y}, i});
  }
  sort(q.rbegin(), q.rend());
  sort(b.begin(), b.end());
  int k = n, ind = 0;
  for (int i = 0; i < m; i++) {
    while (k > q[i].first.first) {
      k--;
      update(1, 1, n, mp[b[ind]].back());
      mp[b[ind]].pop_back();
      ind++;
    }
    ans[q[i].second] = get_kth(1, 1, n, q[i].first.second);
  }
  for (int i = 1; i <= m; i++) cout << ans[i] << "\n";
  return 0;
}
