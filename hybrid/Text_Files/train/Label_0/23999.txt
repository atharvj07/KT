#include <bits/stdc++.h>
using namespace std;
template <class T>
T sqr(T x) {
  return x * x;
}
long double pi = 3.1415926535897932384626433832795;
const int N = (int)155333;
int t[N], n, k, q;
set<pair<int, int> > s;
int main() {
  cin >> n >> k >> q;
  for (int i = 0; i < n; i++) scanf("%d", &t[i]);
  while (q--) {
    int type, a;
    scanf("%d%d", &type, &a);
    if (type == 1)
      s.insert(make_pair(-t[a - 1], a));
    else {
      set<pair<int, int> >::iterator it = s.begin();
      int alr = 0;
      bool ok = 0;
      while (it != s.end() && alr < k) {
        alr++;
        if ((*it).second == a) ok = 1;
        it++;
      }
      if (ok)
        printf("YES\n");
      else
        printf("NO\n");
    }
  }
  return 0;
}
