#include <bits/stdc++.h>
using namespace std;
int k;
void solve() {
  if (k % 2 == 0) {
    puts("NO");
    return;
  }
  puts("YES");
  if (k == 1) {
    puts("2 1");
    puts("1 2");
    return;
  }
  int n = k + 2, m = n * k;
  printf("%d %d\n", n << 1, m);
  for (int i = 1; i <= n; ++i) {
    for (int j = i + 1; j <= n; ++j) {
      if (i == 1 && j == n) continue;
      if (j == i + 1 && (i & 1)) continue;
      printf("%d %d\n", i, j);
      printf("%d %d\n", i + n, j + n);
    }
  }
  printf("%d %d\n", 1, n + 1);
}
int main() {
  scanf("%d", &k);
  solve();
  return 0;
}
