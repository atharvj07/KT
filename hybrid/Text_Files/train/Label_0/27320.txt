#include <bits/stdc++.h>
using namespace std;
const int INF = 0x7f7f7f7f;
int n, m, f[45][1 << 6][1 << 6];
int main() {
  scanf("%d%d", &n, &m);
  if (n < m) swap(n, m);
  memset(f, 0x7f, sizeof(f));
  for (int k = 0; k < (1 << m); k++) f[1][0][k] = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 0; j < (1 << m); j++)
      for (int k = 0; k < (1 << m); k++)
        if (f[i][j][k] < INF)
          for (int l = 0; l < (1 << m); l++) {
            int t = k | (((1 << m) - 1) & ~(j | l | k | (k << 1) | (k >> 1)));
            f[i + 1][t][l] =
                min(f[i + 1][t][l], f[i][j][k] + __builtin_popcount(t));
          }
  int ans = n * m;
  for (int i = 0; i < (1 << m); i++) ans = min(ans, f[n + 1][i][0]);
  printf("%d\n", n * m - ans);
  return 0;
}
