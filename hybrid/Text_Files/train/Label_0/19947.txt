#include <bits/stdc++.h>
using namespace std;
const int MAX_N = 200000;
const int INF = 1 << 30;
int as[MAX_N], ks[MAX_N];
int cs[MAX_N];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; i++) scanf("%d", as + i), as[i]--;
  int ksum = 0, un = 0;
  for (int i = 0; i < m; i++) {
    scanf("%d", ks + i);
    ksum += ks[i];
    if (ks[i] == 0) un++;
  }
  int mind = INF;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < n && un < m) {
      if (++cs[as[j]] == ks[as[j]]) un++;
      j++;
    }
    if (un >= m) {
      int d = (j - i) - ksum;
      if (mind > d) mind = d;
    }
    if (cs[as[i]]-- == ks[as[i]]) un--;
  }
  printf("%d\n", (mind >= INF) ? -1 : mind);
  return 0;
}
