#include <bits/stdc++.h>
using namespace std;
long n = 0, i = 0, x = 0, y = 0, s = 0, viz[1000], a[1000][1000];
void dfs(long v) {
  viz[v] = 1;
  for (long j = 1; j <= n; j++)
    if ((viz[j] == 0) && ((a[v][1] == a[j][1]) || (a[v][2] == a[j][2]))) dfs(j);
}
int main() {
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> x >> y;
    a[i][1] = x;
    a[i][2] = y;
  }
  s = 0;
  for (i = 1; i <= n; i++)
    if (viz[i] == 0) {
      s++;
      dfs(i);
    }
  s--;
  cout << s;
  return 0;
}
