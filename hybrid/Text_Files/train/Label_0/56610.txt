import java.util.Scanner;

public class maxin {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        boolean[] written = new boolean[n + 1];
        int[] l = new int[n + 1];
        int[] r = new int[n + 1];
        for (int i = 1; i <= n; ++i) {
            l[i] = in.nextInt();
            r[i] = in.nextInt();
        }

        //Initialization
        for (int i = 1; i <= n; ++i)
            written[i] = false;

        int last = -1;
        for (int i = 1; i <= n; ++i) {
            if (!written[i] && l[i] == 0) {
                if (last == -1)
                    last = writeList(i, l, r, written);
                else {
                    r[last] = i;
                    l[i] = last;
                    last = writeList(i, l, r, written);
                }
            }
        }

        for (int i = 1; i <= n; ++i)
            System.out.println(l[i] + " " + r[i]);
    }

    static int writeList(int i, int[] l, int[] r, boolean[] written) {
        written[i] = true;
        int last = i;
        while (r[i] != 0) {
            i = r[i];
            written[i] = true;
            last = i;
        }
        return last;
    }
}
