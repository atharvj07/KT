import java.util.Scanner;

public class DTickets {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int m = s.nextInt();
        int k = s.nextInt();
        double result = 1.0;
        for (int i = 0; i <= k; i++) {
            result *= (double)(m - i) / (double)(n + 1 + i);
        }
        result = 1.0 - result;
        if (result <= 0.0) {
            System.out.println(0.0);
            return;
        }
        System.out.println(result);

    }

}
