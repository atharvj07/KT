from collections import defaultdict
(n, k) = map(int, input().split())
s = input()
d = defaultdict(int)
i = 0
j = 0
count = 0
while j < n:
	if s[i] == s[j]:
		j += 1
		count += 1
		if j == n:
			d[s[i]] += count // k
			break
	else:
		d[s[i]] += count // k
		i = j
		count = 0
		if j == n - 1:
			d[s[j]] += 1 // k
			break
ans = 0
for i in range(ord('a'), ord('z') + 1):
	ans = max(ans, d[chr(i)])
print(ans)
