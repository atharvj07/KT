import sys

def minp():
	return sys.stdin.readline().strip()

def mint():
	return int(minp())

def mints():
	return map(int, minp().split())

def solve1(b, n, k):
	sl = 0
	sr = 0
	for i in b:
		sr += i[0] * i[1]
	i = 0
	j = 0
	m = len(b)
	nl = 0
	nr = n
	r = int(2 * 1000000000.0 + 5)
	while i < m:
		sl += b[i][0] * b[i][1]
		nl += b[i][1]
		while j < m - 1 and (j < i or b[i][0] * nl - sl + sr - b[j][0] * nr > k):
			sr -= b[j][0] * b[j][1]
			nr -= b[j][1]
			j += 1
		hm = k - (b[i][0] * nl - sl + sr - b[j][0] * nr)
		if hm < 0:
			break
		if i == j:
			return 0
		if i < j:
			r = min(r, b[j][0] - min(b[i][0] + hm // nl, b[i + 1][0]))
			r = min(r, max(b[j][0] - hm // nr, b[j - 1][0]) - b[i][0])
		i += 1
	return r

def solve():
	(n, k) = mints()
	a = list(mints())
	a.sort()
	b = []
	i = 0
	while i < n:
		j = i + 1
		while j < n and a[j] == a[i]:
			j += 1
		b.append((a[i], j - i))
		i = j
	r1 = solve1(b, n, k)
	b.reverse()
	for i in range(len(b)):
		b[i] = (-b[i][0], b[i][1])
	print(min(r1, solve1(b, n, k)))
solve()
