int main() {
  for (;;) {
    int n;
    cin >> n;
    if (n == 0) break;

    int min_sum = 1000, max_sum = -1;
  
    for (int i = 0; i < n; i++) {
      int sum = 0, s;
      for (int j = 0; j < 5; j++) {
	cin >> s;
	sum += s;
      }

      if (min_sum > sum) min_sum = sum;
      if (max_sum < sum) max_sum = sum;
    }

    printf("%d %d\n", max_sum, min_sum);
  }

  return 0;
}
