#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  vector<pair<long long int, long long int>> point;
  long long int i = 0;
  while (i < n) {
    long long int x, y;
    cin >> x >> y;
    point.push_back({x, y});
    i++;
  }
  map<double, set<double>> m3;
  set<long long int> p;
  for (long long int i = 0; i < n; i++) {
    for (long long int j = i + 1; j < n; j++) {
      if (point[i].first == point[j].first) {
        long long int c = point[i].first;
        p.insert(c);
      } else {
        double m = (double)(point[i].second - point[j].second) /
                   (point[i].first - point[j].first);
        double c = (double)point[i].second - (m * point[i].first);
        m3[m].insert(c);
      }
    }
  }
  n = p.size();
  for (auto it : m3) n += it.second.size();
  long long int ans = (n * (n - 1)) / 2;
  ans -= ((p.size() - 1) * p.size()) / 2;
  for (auto it : m3) ans -= ((it.second.size() - 1) * it.second.size()) / 2;
  ;
  cout << ans;
}
