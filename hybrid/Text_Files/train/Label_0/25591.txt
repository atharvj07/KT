#include <bits/stdc++.h>
using namespace std;
const int N = 2005;
char M[N][N];
int data[N], rdata[N];
int main() {
  int n;
  scanf("%d", &n);
  int p = 0;
  for (int i = (1); i < (n + 1); ++i) {
    scanf("%d", &data[i]);
    rdata[data[i]] = i;
    if (data[i] != i) p = 1;
  }
  if (p == 0) {
    printf("%d\n", n);
    for (int i = (0); i < (n); ++i) {
      for (int j = (0); j < (n); ++j) printf(".");
      puts("");
    }
    return 0;
  }
  printf("%d\n", n - 1);
  for (int i = (1); i < (n + 1); ++i)
    for (int j = (1); j < (n + 1); ++j) M[i][j] = '.';
  int f = n, P = 0;
  for (int i = (1); i < (n + 1); ++i)
    if (data[i] == 1) P = i;
  M[f][1] = M[f][P] = '/';
  f--;
  swap(data[1], data[P]);
  for (int i = (1); i < (n + 1); ++i) rdata[data[i]] = i;
  for (int cas = (2); cas < (n + 1); ++cas)
    if (data[cas] != cas) {
      M[f][1] = '\\';
      vector<int> V;
      V.push_back(cas);
      int now = rdata[cas];
      while (now != cas) {
        V.push_back(now);
        now = rdata[now];
      }
      for (int i = (1); i < (((int)(V).size())); ++i) {
        swap(data[V[i]], data[V[i - 1]]);
        if (V[i] > V[i - 1])
          M[f][V[i]] = M[f][V[i - 1]] = '\\';
        else
          M[f][V[i]] = M[f][V[i - 1]] = '/';
        f--;
      }
      M[f][1] = '/', M[f][V.back()] = '/';
      f--;
    }
  for (int i = (1); i < (n + 1); ++i) {
    for (int j = (1); j < (n + 1); ++j) printf("%c", M[i][j]);
    puts("");
  }
}
