n = int(input())
l = [int(x) for x in input().split()]
r = [int(x) for x in input().split()]
flag = 1
a = [n - l[i] - r[i] for i in range(n)]
for i in range(n):
	cl = 0
	cr = 0
	for j in range(i):
		if a[j] > a[i]:
			cl += 1
	for k in range(i + 1, n):
		if a[k] > a[i]:
			cr += 1
	if cl != l[i] or cr != r[i]:
		print('NO')
		flag = 0
		break
if flag:
	print('YES')
	print(*a)
