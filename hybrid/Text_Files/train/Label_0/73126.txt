def rec(copy, t):
	count = 0
	while copy != list(sorted(arr)):
		if copy == arr and count:
			return -1
		if count % 2 == t:
			for i in range(n):
				(copy[i], copy[i + n]) = (copy[i + n], copy[i])
		else:
			for i in range(0, 2 * n, 2):
				(copy[i], copy[i + 1]) = (copy[i + 1], copy[i])
		count += 1
	return count
n = int(input())
arr = [int(x) for x in input().split()]
res1 = rec(arr.copy(), 0)
res2 = rec(arr.copy(), 1)
if res1 > -1 and res2 > -1:
	print(min(res1, res2))
else:
	print(max(res1, res2))
