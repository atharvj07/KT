for i in range(int(input())):
	length_of_values = int(input())
	values = list(map(int, input().split()))
	types = list(map(int, input().split()))
	count_of_1s = types.count(1)
	count_of_0s = types.count(0)
	if sorted(values) == values:
		print('Yes')
	elif count_of_0s == length_of_values:
		print('No')
	elif count_of_1s == length_of_values:
		print('No')
	else:
		print('Yes')
