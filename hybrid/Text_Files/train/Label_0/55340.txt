import java.math.BigInteger;
import java.util.Scanner;

public class Rectangle {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int one_row = 0;
        int zero_row = 0;
        int one_col=0;
        int zero_col=0;
        BigInteger sets = BigInteger.ZERO;
        int total[][] = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                total[i][j] = in.nextInt();
            }
        }
        for (int i = 0; i < n; i++) {
            one_row=0;

            for (int j = 0; j < m; j++) {
                if (total[i][j] == 1)
                    one_row++;
            }
            zero_row = m - one_row;

            sets=sets.add(BigInteger.valueOf((long) ((Math.pow(2,one_row)-1)+(Math.pow(2,zero_row)-1))));

        }
        for (int i = 0; i <m ; i++) {
            one_col=0;

            for (int j = 0; j <n ; j++) {
                if (total[j][i]==1)
                    one_col++;
            }
            zero_col=n-one_col;
            sets=sets.add(BigInteger.valueOf((long) ((Math.pow(2,one_col)-1)+(Math.pow(2,zero_col)-1)-n)));
        }
        System.out.println(sets);
    }
}
