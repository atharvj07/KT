#include <bits/stdc++.h>
using namespace std;

int main() {
  cin.tie(nullptr);
  ios::sync_with_stdio(false);
  long long n, k;
  cin >> n >> k;
  auto chk = [&](long long x) {
    long long sum = 0;
    for (int i = 0; i < n; ++i) {
      sum += x;
      x /= 2;
      if (x == 0) {
        break;
      }
    }
    return sum <= k;
  };
  long long ok = 0, ng = k + 1;
  while (ng - ok > 1) {
    long long mid = (ok + ng) / 2;
    (chk(mid) ? ok : ng) = mid;
  }
  cout << ok << '\n';
}
