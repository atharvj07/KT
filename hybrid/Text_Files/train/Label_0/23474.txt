#include <bits/stdc++.h>
using namespace std;
int n, a[250100], ans[510], tot, t;
priority_queue<int> heap;
int gcd(int x, int y) {
  for (; t = x % y; x = y, y = t)
    ;
  return y;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n * n; ++i) scanf("%d", &a[i]);
  sort(a + 1, a + 1 + n * n);
  for (int i = n * n; i >= 1; --i)
    if (heap.empty() || heap.top() != a[i]) {
      for (int j = 1; j <= tot; ++j)
        heap.push(gcd(ans[j], a[i])), heap.push(gcd(ans[j], a[i]));
      printf("%d ", ans[++tot] = a[i]);
    } else
      heap.pop();
  printf("\n");
  return 0;
}
