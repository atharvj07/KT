#include <bits/stdc++.h>
using namespace std;
int sq[100][100], r2, c2;
void dfs(int i, int j, int n, bool vis[100][100]) {
  if (i >= n || j >= n || i < 0 || j < 0 || sq[i][j] == 1) {
    return;
  }
  if (vis[i][j]) {
    return;
  }
  vis[i][j] = true;
  dfs(i + 1, j, n, vis);
  dfs(i, j + 1, n, vis);
  dfs(i - 1, j, n, vis);
  dfs(i, j - 1, n, vis);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, r1, c1;
  bool vis[100][100];
  bool vis2[100][100];
  cin >> n >> r1 >> c1 >> r2 >> c2;
  r1--;
  c1--;
  r2--;
  c2--;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      vis[i][j] = false;
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      vis2[i][j] = false;
    }
  }
  string s[n];
  for (int i = 0; i < n; ++i) {
    cin >> s[i];
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      sq[i][j] = s[i][j] - '0';
    }
  }
  dfs(r1, c1, n, vis);
  if (vis[r2][c2]) {
    cout << 0;
    return 0;
  }
  dfs(r2, c2, n, vis2);
  long long min = INT_MAX;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      for (int k = 0; k < n; ++k) {
        for (int l = 0; l < n; ++l) {
          if (vis[i][j] && vis2[k][l]) {
            if (min > (i - k) * (i - k) + (j - l) * (j - l)) {
              min = (i - k) * (i - k) + (j - l) * (j - l);
            }
          }
        }
      }
    }
  }
  cout << min;
  return 0;
}
