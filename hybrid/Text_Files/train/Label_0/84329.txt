import sys, io, os
from collections import defaultdict
from heapq import heappush, heappop
if os.environ['USERNAME'] == 'kissz':
	inp = open('in2.txt', 'r').readline

	def debug(*args):
		print(*args, file=sys.stderr)
else:
	inp = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline

	def debug(*args):
		pass
for _ in range(int(inp())):
	inp()
	neighbors = defaultdict(list)
	(n, m) = map(int, inp().split())
	for i in range(m):
		(u, v) = map(int, inp().split())
		neighbors[u].append(v)
	dists = {1: 0}
	Q = []
	heappush(Q, (0, 1))
	while Q:
		(dist, node) = heappop(Q)
		if dists[node] == dist:
			for next_node in neighbors[node]:
				if next_node not in dists or dists[next_node] > dist + 1:
					dists[next_node] = dist + 1
					heappush(Q, (dist + 1, next_node))
	sol = {k: v for (k, v) in dists.items()}
	falls = []
	climbs = defaultdict(list)
	for (node, next_nodes) in neighbors.items():
		for next_node in next_nodes:
			if dists[node] < dists[next_node]:
				climbs[next_node].append(node)
			elif dists[node] > dists[next_node]:
				falls.append((dists[next_node], node))
	debug(neighbors)
	debug(climbs)
	debug(falls)
	debug(sol)
	for (dist, fall_node) in sorted(falls):
		debug(dist, fall_node)
		if sol[fall_node] > dist:
			sol[fall_node] = dist
			Q = [fall_node]
			while Q:
				node = Q.pop()
				for prev_node in climbs[node]:
					if sol[prev_node] > dist:
						sol[prev_node] = dist
						Q.append(prev_node)
		debug(sol)
	print(*[sol[i] for i in range(1, n + 1)])
