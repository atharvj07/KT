for z in range(int(input())):
	n = int(input())
	a = [[0, 0]]
	for i in range(n):
		a.append(list(map(int, input().split())))
	tm = list(map(int, input().split()))
	t = 0
	for i in range(1, n + 1):
		t += a[i][0] - a[i - 1][1] + tm[i - 1]
		if i != n:
			t = max(a[i][1], t - (a[i][0] - a[i][1]) // 2)
	print(t)
