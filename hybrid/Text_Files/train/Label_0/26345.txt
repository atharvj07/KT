void addmod(int& x, int y, int mod = 998244353) {
  assert(y >= 0);
  x += y;
  if (x >= mod) x -= mod;
  assert(x >= 0 && x < mod);
}

long long fastPow(long long x, long long y, int mod = 998244353) {
  long long ans = 1;
  while (y > 0) {
    if (y & 1) ans = (x * ans) % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ans;
}

long long w[300135];
vector<pair<int, pair<long long, long long>>> mp[300135];
int d[300135], p[300135][20];
long long pt[300135][20][2][2];
void dfs1(int x, int pa) {
  d[x] = d[pa] + 1;
  p[x][0] = pa;
  for (auto cp : mp[x])
    if (cp.first != pa) {
      dfs1(cp.first, x);
      long long w0 = cp.second.first, w1 = cp.second.second;
      mintt(w[x], w[cp.first] + w0 + w1);
    }
}
void dfs2(int x, int pa) {
  for (auto cp : mp[x])
    if (cp.first != pa) {
      long long w0 = cp.second.first, w1 = cp.second.second;
      mintt(w[cp.first], w[x] + w0 + w1);
      dfs2(cp.first, x);
    }
}
void dfs3(int x, int pa) {
  for (auto cp : mp[x])
    if (cp.first != pa) {
      dfs3(cp.first, x);
      long long w1 = cp.second.first, w2 = cp.second.second;
      pt[cp.first][0][0][0] = min(w1, w2 + w[x] + w[cp.first]);
      pt[cp.first][0][1][1] = min(w2, w1 + w[x] + w[cp.first]);
      pt[cp.first][0][0][1] = min(w1 + w[x], w2 + w[cp.first]);
      pt[cp.first][0][1][0] = min(w2 + w[x], w1 + w[cp.first]);
    }
}

int lca(int a, int b) {
  if (d[a] > d[b]) swap(a, b);
  int f = d[b] - d[a];
  for (int i = 0; (1 << i) <= f; i++)
    if ((1 << i) & f) b = p[b][i];
  if (a != b) {
    for (int i = 19; i >= 0; i--)
      if (p[a][i] != p[b][i]) {
        a = p[a][i];
        b = p[b][i];
      }
    a = p[a][0];
  }
  return a;
}
long long dp[2], dpv[2];
long long cal(int x, int y) {
  int u = (x + 1) / 2, v = (y + 1) / 2;
  int fa = lca(u, v);
  long long tp0, tp1;
  if (x & 1) {
    dp[0] = 0;
    dp[1] = w[u];
  } else {
    dp[1] = 0;
    dp[0] = w[u];
  }
  for (int j = 19; j >= 0; j--) {
    if (d[p[u][j]] >= d[fa]) {
      tp0 = dp[0];
      tp1 = dp[1];
      dp[0] = min(tp0 + pt[u][j][0][0], tp1 + pt[u][j][1][0]);
      dp[1] = min(tp0 + pt[u][j][0][1], tp1 + pt[u][j][1][1]);
      u = p[u][j];
    }
  }
  for (int(z) = 0; (z) < (int)(2); (z)++) dpv[z] = dp[z];
  u = v;
  if (y & 1) {
    dp[0] = 0;
    dp[1] = w[u];
  } else {
    dp[1] = 0;
    dp[0] = w[u];
  }
  for (int j = 19; j >= 0; j--) {
    if (d[p[u][j]] >= d[fa]) {
      tp0 = dp[0];
      tp1 = dp[1];
      dp[0] = min(tp0 + pt[u][j][0][0], tp1 + pt[u][j][1][0]);
      dp[1] = min(tp0 + pt[u][j][0][1], tp1 + pt[u][j][1][1]);
      u = p[u][j];
    }
  }
  return min(dp[0] + dpv[0], dp[1] + dpv[1]);
}
