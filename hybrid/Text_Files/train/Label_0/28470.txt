#include <bits/stdc++.h>
using namespace std;

int main(){
  int n;
  
  while(cin>>n,n){
    int mp[101][101]={};
    for(int i=0;i<n;i++)
      for(int j=0;j<n;j++) cin>>mp[i][j];

    int flg[101][101]={};  
    for(int i=0;i<n;i++){
      int mnx=1e9,mxy=0;
      int xi,xj,yi,yj;
      for(int j=0;j<n;j++) {
	if(mnx>mp[i][j]) mnx=mp[i][j];
	if(mxy<mp[j][i]) mxy=mp[j][i];
      }
      for(int j=0;j<n;j++){
	if(mnx==mp[i][j])flg[i][j]++;
	if(mxy==mp[j][i])flg[j][i]++;
      }
    }
    
    int f=0,ans=0;
    for(int i=0;i<n;i++)
      for(int j=0;j<n;j++) if(flg[i][j]==2)ans=mp[i][j],f++;
    
    if(f) cout << ans<<endl;
    else cout <<0<<endl;
  }
  return 0;
}
