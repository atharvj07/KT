from collections import defaultdict
n = int(input())
mp = defaultdict(list)
edges = dict()
input_list = []
for i in range(n - 1):
	(a, b) = list(map(int, input().split()))
	(a, b) = (min(a, b), max(a, b))
	edges[a, b] = -1
	input_list.append((a, b))
	mp[a].append(b)
	mp[b].append(a)
solved = False
leaves_edges = []
for i in range(1, n + 1):
	if len(mp[i]) > 2:
		solved = True
		for j in range(len(mp[i])):
			(a, b) = (min(i, mp[i][j]), max(i, mp[i][j]))
			edges[a, b] = j
		break
	elif len(mp[i]) == 1:
		leaves_edges.append((min(i, mp[i][0]), max(i, mp[i][0])))
if not solved:
	for i in range(len(leaves_edges)):
		if edges[leaves_edges[i]] == -1:
			edges[leaves_edges[i]] = i
cnt = n - 2
for e in edges:
	if edges[e] == -1:
		edges[e] = cnt
		cnt -= 1
for i in input_list:
	print(edges[i])
