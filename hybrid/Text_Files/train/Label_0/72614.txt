def ndivide(k, a, b):
	return (b - a) // k + (b % k < a % k) + (a % k == 0)
(n, k) = (int(x) for x in input().split())
A = [int(x) for x in input().split()]
B = [int(x) for x in input().split()]
N = 1
for i in range(n // k):
	a = B[i] * 10 ** (k - 1)
	b = (B[i] + 1) * 10 ** (k - 1) - 1
	N *= (ndivide(A[i], 0, 10 ** k - 1) - ndivide(A[i], a, b)) % (10 ** 9 + 7)
	N %= 10 ** 9 + 7
print(N)
