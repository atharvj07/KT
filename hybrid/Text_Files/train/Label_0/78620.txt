(n, m) = map(int, input().split())
a = []
b = []
ans = []
p = []
for i in range(n):
	(c, d) = map(str, input().split())
	d += ';'
	a.append(c)
	a.append(d)
for i in range(m):
	b.append(list(map(str, input().split())))
for i in range(m):
	p.append(b[i][0])
	p.append(b[i][1])
	p.append('#' + a[a.index(b[i][1]) - 1])
	ans.append(p)
	p = []
for i in range(m):
	print(*ans[i])
