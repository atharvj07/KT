#include <bits/stdc++.h>
using namespace std;
const long long MAXN = 50000;
const long long INF = 1e14;
long long n, k, p;
long long l, r, mid;
long long a[MAXN], c[MAXN];
long long tf;
signed main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  cin >> n;
  for (long long i = 0; i < n; i++) cin >> a[i];
  if (n % 7 == 0) {
    p = INF;
    for (long long i = 0; i < n; i++) {
      if (i % 7 != 6 && a[i] < p) k = i + 1, p = a[i];
    }
    cout << k << endl;
    return 0;
  }
  l = 0, r = INF;
  while (l < r) {
    mid = (l + r + 1) >> 1;
    k = mid / (7 * n);
    for (long long i = 0; i < n; i++) c[i] = a[i] - 6 * k;
    for (long long i = 0; i < mid % (7 * n); i++)
      if (i % 7 != 6) c[i % n]--;
    tf = 2;
    for (long long i = 0; i < n; i++) {
      if (c[i] < 0)
        tf = 0;
      else if (c[i] == 0)
        tf--;
    }
    if (tf <= 0)
      r = mid - 1;
    else
      l = mid;
  }
  k = l / (7 * n);
  for (long long i = 0; i < n; i++) c[i] = a[i] - 6 * k;
  for (long long i = 0; i < l % (7 * n); i++)
    if (i % 7 != 6) c[i % n]--;
  for (long long i = 0; i < n; i++)
    if (!c[i]) cout << i + 1 << endl;
  return 0;
}
