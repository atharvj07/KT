#include<bits/stdc++.h>

#define MAXN 1000010
#define inf (1<<25)

typedef long long ll;

using namespace std;

int tree[MAXN],n,vis[129];
bool have[MAXN];
char str[MAXN];
vector<int>v[129];

int lowbit(int x){return x&(x^(x-1));}

int sum(int x){
    int ret=0;
    while(x>0){
        ret+=tree[x];
        x-=lowbit(x);
    }
    return ret;
}

void add(int x,int k){
    while(x<=n){
        tree[x]+=k;
        x+=lowbit(x);
    }
}

void solve(){
    memset(tree,0,sizeof(tree));
    for(int i=1; i<=n+1; i++)add(i,1);
    ll ans=0;
    int u,sz;
    char s;
    memset(have,0,sizeof(have));
    for(int i=1; i<=n; i++){
        if(!have[i]){
            s = str[i];
            sz = v[s].size();
            u = v[s][sz-1];
            if(i==u)ans+=(sum(n)-sum(i))/2;
            else ans+=sum(n)-sum(u);
            add(u,-1);
            add(i,-1);
            have[i]=have[u]=1;
            v[s].erase(v[s].end()-1);
        }
    }
    printf("%lld\n",ans);
}

int main(){
    scanf("%s",str+1);
    n=strlen(str+1);
    memset(vis,0,sizeof(vis));
    for(int i=1; i<129; i++)v[i].clear();
    for(int i=1; i<=n; i++){
        vis[str[i]]++;
        v[str[i]].push_back(i);
    }
    int flag=0;
    for(int i=0; i<26; i++){
        if(vis['a'+i]%2)flag++;
        if(vis['A'+i]%2)flag++;
    }
    if(flag>1){
        printf("-1\n");
        return 0;
    }
    else solve();
    
    return 0;
}
