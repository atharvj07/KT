#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, k;
  cin >> n >> k;
  int a[n + 1], b[n + 1], dp[n + 1][2];
  dp[0][0] = 0;
  dp[0][1] = k;
  for (int i = 1; i < n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) cin >> b[i];
  cout << "0 ";
  for (int i = 1; i < n; i++) {
    dp[i][0] = min(dp[i - 1][0] + a[i], dp[i - 1][1] + a[i]);
    dp[i][1] = min(dp[i - 1][0] + b[i] + k, dp[i - 1][1] + b[i]);
    cout << min(dp[i][0], dp[i][1]) << " \n"[i == n - 1];
  }
}
