#include<stdio.h>
#include<algorithm>
#include<vector>
#include<math.h>
using namespace std;
double lx[100];
double ly[100];
double rx[100];
double ry[100];
double EPS=1e-9;
vector<pair<double,double> > v;
double ABS(double a){return max(a,-a);}
double calc(double a,double b){
	return -4.9*b*b+a*b;
}
int main(){
	int a;
	double b,c,d;
	scanf("%d%lf%lf%lf",&a,&b,&c,&d);
	for(int i=0;i<a;i++)scanf("%lf%lf%lf%lf",lx+i,ly+i,rx+i,ry+i);
	v.push_back(make_pair(c,d));
	for(int i=0;i<a;i++){
		rx[i]=min(rx[i],c);
		//v.push_back(make_pair(lx[i],ly[i]));
		v.push_back(make_pair(lx[i],ry[i]));
		//v.push_back(make_pair(min(rx[i],c),ly[i]));
		v.push_back(make_pair(rx[i],ry[i]));
	}
	for(int i=0;i<v.size();i++){
		int x=v[i].first;
		int y=v[i].second;
		double D=(9.8*y-b*b)*(9.8*y-b*b)-9.8*9.8*(x*x+y*y);
		if(D>-EPS)D=max(D,0.0);
		if(D<0)continue;
		double t,t2,vx,vy;
		bool ok;
		t2=(((double)b*b-9.8*y)+sqrt(D))*2/9.8/9.8;
		if(t2>-EPS)t2=max(t2,0.0);
		if(t2<0)continue;
		t=sqrt(t2);
		vx=(double)x/t;
		vy=sqrt((double)b*b-vx*vx);
		ok=true;
		if(vx-EPS>b)ok=false;
		double hg=calc(vy,(double)c/vx);
		for(int j=0;j<a;j++){
			if(lx[j]>=c-EPS)continue;
			if(ABS(rx[j]-c)<EPS&&d<=ry[j]&&ly[j]<=hg)ok=false;
			double p=calc(vy,(double)lx[j]/vx);
			double q=calc(vy,(double)rx[j]/vx);
			if(ly[j]<p-EPS&&p+EPS<ry[j])ok=false;
			if(ly[j]<q-EPS&&q+EPS<ry[j])ok=false;
			if(q+EPS<ry[j]&&ry[j]+EPS<p)ok=false;
			if(p+EPS<ly[j]&&ly[j]+EPS<q)ok=false;
			
			if(p+EPS<ry[j]&&ry[j]+EPS<q)ok=false;
			if(q+EPS<ly[j]&&ly[j]+EPS<p)ok=false;
			double th=calc(vy,vy/9.8);
		//	if(th-EPS>ly[j]){
				if(EPS+lx[j]<vx*vy/9.8&&vx*vy/9.8+EPS<rx[j]&&th>ly[j]+EPS&&ly[j]+EPS>calc(vy,lx[j]/vx)){
					ok=false;
				}
		//	}
			
		}
		if(hg<(double)d-EPS)ok=false;
		if(ok&&ABS(vx)>EPS){
			//printf("%f %f\n",vx,vy);
			printf("Yes\n");return 0;
		}
		t2=(((double)b*b-9.8*y)-sqrt(D))*2/9.8/9.8;
		if(t2>-EPS)t2=max(t2,0.0);
		if(t2<0)continue;
		t=sqrt(t2);
		vx=(double)x/t;
		vy=sqrt((double)b*b-vx*vx);
		ok=true;
		if(vx-EPS>b)ok=false;
		hg=calc(vy,(double)c/vx);
		for(int j=0;j<a;j++){
			if(lx[j]>=c-EPS)continue;
			if(ABS(rx[j]-c)<EPS&&d<=ry[j]&&ly[j]<=hg)ok=false;
			double p=calc(vy,(double)lx[j]/vx);
			double q=calc(vy,(double)rx[j]/vx);
			if(ly[j]<p-EPS&&p+EPS<ry[j])ok=false;
			if(ly[j]<q-EPS&&q+EPS<ry[j])ok=false;
			if(q+EPS<ry[j]&&ry[j]+EPS<p)ok=false;
			if(p+EPS<ly[j]&&ly[j]+EPS<q)ok=false;
			
			if(p+EPS<ry[j]&&ry[j]+EPS<q)ok=false;
			if(q+EPS<ly[j]&&ly[j]+EPS<p)ok=false;
			double th=calc(vy,vy/9.8);
		//	if(th-EPS>ly[j]){
				if(EPS+lx[j]<vx*vy/9.8&&vx*vy/9.8+EPS<rx[j]&&th>ly[j]+EPS&&ly[j]+EPS>calc(vy,lx[j]/vx)){
					ok=false;
				}
		//	}
			
		}
		if(hg<(double)d-EPS)ok=false;
		if(ok&&ABS(vx)>EPS){
			//printf("%f %f\n",vx,vy);
			printf("Yes\n");return 0;
		}
	}
	printf("No\n");
}
