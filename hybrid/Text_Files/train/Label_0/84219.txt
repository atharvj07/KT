import atexit
import io
import sys
_INPUT_LINES = sys.stdin.read().splitlines()
input = iter(_INPUT_LINES).__next__
_OUTPUT_BUFFER = io.StringIO()
sys.stdout = _OUTPUT_BUFFER

@atexit.register
def write():
	sys.__stdout__.write(_OUTPUT_BUFFER.getvalue())

def row_passable(grid, row):
	r = grid[row]
	if sum(r) == 0:
		return False
	elif sum(r) == 1 and grid[min(len(grid) - 1, row + 1)][grid[row].index(1)] == 0:
		return False
	elif sum(r) == 1 and grid[max(0, row - 1)][grid[row].index(1)] == 0:
		return False
	return True
(n, q) = [int(x) for x in input().split()]
grid = [[1, 1] for _ in range(n)]
impassable = set()
for _ in range(q):
	(r, c) = [int(x) - 1 for x in input().split()]
	grid[c][r] = int(not grid[c][r])
	for x in range(c - 1, c + 2):
		if 0 <= x <= n - 1:
			p = row_passable(grid, x)
			if not p:
				impassable.add(x)
			elif x in impassable:
				impassable.remove(x)
	if impassable:
		print('No')
	else:
		print('Yes')
