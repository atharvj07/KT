#include <bits/stdc++.h>
using namespace std;
inline long long mod(long long n, long long m) {
  long long ret = n % m;
  if (ret < 0) ret += m;
  return ret;
}
long long gcd(long long a, long long b) {
  return (b == 0LL ? a : gcd(b, a % b));
}
long long exp(long long a, long long b, long long m) {
  if (b == 0LL) return 1LL;
  if (b == 1LL) return mod(a, m);
  long long k = mod(exp(a, b / 2, m), m);
  if (b & 1LL) {
    return mod(a * mod(k * k, m), m);
  } else
    return mod(k * k, m);
}
const int N = 20;
int reach[N + 10];
int can[(1 << N) + 100];
struct dsu {
  vector<int> pai;
  int n;
  dsu() {}
  dsu(int n) : n(n) {
    pai = vector<int>(n + 1, 0);
    for (int i = 0; i <= n; i++) pai[i] = i;
  }
  int f(int x) { return pai[x] = (pai[x] == x ? x : f(pai[x])); }
  void join(int a, int b) { pai[f(a)] = f(b); }
};
void solve() {
  int nn;
  cin >> nn;
  string A, B;
  cin >> A >> B;
  map<pair<int, int>, int> e;
  dsu D(N);
  for (int i = 0; i < nn; i++) {
    if (A[i] != B[i]) {
      e[pair<int, int>(A[i] - 'a', B[i] - 'a')] = 1;
      D.join(A[i] - 'a', B[i] - 'a');
    }
  }
  int res = 0;
  for (int a = 0; a < 20; a++) {
    if (D.f(a) != a) continue;
    map<int, int> here;
    int cur = 0;
    for (int j = 0; j < 20; j++) {
      if (D.f(j) == a) here[j] = cur++;
    }
    for (int i = 0; i < cur; i++) {
      reach[i] = 0;
    }
    for (auto z : e) {
      pair<int, int> it = z.first;
      if (here.count(it.first)) {
        reach[here[it.first]] |= (1 << here[it.second]);
      }
    }
    for (int i = 0; i < cur; i++) {
      can[(1 << i)] = 1;
    }
    for (int i = 1; i < (1 << cur); i++) {
      for (int v = 0; v < cur; v++) {
        can[i | (1 << v)] |= (can[i] and (reach[v] & i) == 0);
      }
    }
    int big = 0;
    for (int i = 0; i < (1 << cur); i++) {
      if (can[i]) big = max(big, __builtin_popcount(i));
      can[i] = 0;
    }
    res += 2 * cur - 1 - big;
  }
  cout << res << "\n";
}
int32_t main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
}
