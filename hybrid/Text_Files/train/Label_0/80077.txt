from sys import stdin, stdout
import heapq
from collections import defaultdict
import math
import bisect
import io, os

def ncr(n, r, p):
	num = den = 1
	for i in range(r):
		num = num * (n - i) % p
		den = den * (i + 1) % p
	return num * pow(den, p - 2, p) % p

def main():
	t = int(stdin.readline())
	for _ in range(t):
		n = int(stdin.readline())
		arr = list(map(int, stdin.readline().split()))
		is_ok = True
		for i in range(n):
			if arr[i] % 2 == i % 2:
				is_ok = False
				break
		if not is_ok:
			stdout.write('-1\n')
			continue
		ans = []
		for i in range(0, n - 2, 2):
			x = n - i
			y = n - i - 1
			pos_x = pos_y = 0
			for j in range(n - i):
				if arr[j] == x:
					pos_x = j
				if arr[j] == y:
					pos_y = j
				if pos_x and pos_y:
					break
			if abs(pos_x - pos_y) != 1:
				if pos_x != 0:
					ans.append(pos_x + 1)
					for k in range(pos_x // 2):
						(arr[k], arr[pos_x - k]) = (arr[pos_x - k], arr[k])
					if pos_y < pos_x:
						pos_y = pos_x - pos_y
					pos_x = 0
				ans.append(pos_y)
				for k in range(pos_y // 2):
					(arr[k], arr[pos_y - 1 - k]) = (arr[pos_y - 1 - k], arr[k])
				pos_x = pos_y - 1
			if pos_y > pos_x and pos_x != 0:
				ans.append(pos_y + 2)
				for k in range((pos_y + 1) // 2):
					(arr[k], arr[pos_y + 1 - k]) = (arr[pos_y + 1 - k], arr[k])
				pos_y = 1
				pos_x = 2
			if pos_x > pos_y and pos_x != n - i - 1:
				ans.append(pos_x + 1)
				for k in range(pos_x // 2):
					(arr[k], arr[pos_x - k]) = (arr[pos_x - k], arr[k])
				pos_y = pos_x - pos_y
				pos_x = 0
			if pos_x == 0:
				ans.append(n - i)
				for k in range((n - i) // 2):
					(arr[k], arr[n - i - k - 1]) = (arr[n - i - k - 1], arr[k])
		if len(ans):
			stdout.write(str(len(ans)) + '\n')
			stdout.write(' '.join([str(x) for x in ans]) + '\n')
		else:
			stdout.write('2\n3 3\n')
main()
