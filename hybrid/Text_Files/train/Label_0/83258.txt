def solve(n, m, energy, link):
	total = 0
	for (x, y) in link:
		total += min(energy[x - 1], energy[y - 1])
	return total
(n, m) = list(map(int, input().split()))
ener = list(map(int, input().split()))
mat = []
for _ in range(m):
	mat.append(list(map(int, input().split())))
print(solve(n, m, ener, mat))
