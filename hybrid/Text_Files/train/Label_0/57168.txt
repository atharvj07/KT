import java.io.*;
import java.util.*;

public class CF632F extends PrintWriter {
	CF632F() { super(System.out, true); }
	static class Scanner {
		Scanner(InputStream in) { this.in = in; } InputStream in;
		int k, l; byte[] bb = new byte[1 << 15];
		byte getc() {
			if (k >= l) {
				k = 0;
				try { l = in.read(bb); } catch (IOException e) { l = 0; }
				if (l <= 0) return -1;
			}
			return bb[k++];
		}
		int nextInt() {
			byte c = 0; while (c <= 32) c = getc();
			int a = 0;
			while (c > 32) { a = a * 10 + c - '0'; c = getc(); }
			return a;
		}
	}
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF632F o = new CF632F(); o.main(); o.flush();
	}

	int[] dsu;
	int find(int i) {
		return dsu[i] < 0 ? i : (dsu[i] = find(dsu[i]));
	}
	int join(int i, int j) {
		i = find(i);
		j = find(j);
		if (i == j)
			return 0;
		int cnt = dsu[i] * dsu[j];
		if (dsu[i] > dsu[j]) {
			dsu[j] += dsu[i];
			dsu[i] = j;
		} else {
			dsu[i] += dsu[j];
			dsu[j] = i;
		}
		return cnt;
	}
	static class E {
		int i, j, a;
		E(int i, int j, int a) {
			this.i = i; this.j = j; this.a = a;
		}
	}
	void main() {
		int n = sc.nextInt();
		int[][] aa = new int[n][n];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
				aa[i][j] = sc.nextInt();
		for (int i = 0; i < n; i++) {
			if (aa[i][i] != 0) {
				println("NOT MAGIC");
				return;
			}
			for (int j = i + 1; j < n; j++)
				if (aa[i][j] != aa[j][i]) {
					println("NOT MAGIC");
					return;
				}
		}
		dsu = new int[n]; Arrays.fill(dsu, -1);
		int m = n * (n - 1) / 2;
		E[] ee = new E[m];
		m = 0;
		for (int i = 0; i < n; i++)
			for (int j = i + 1; j < n; j++)
				ee[m++] = new E(i, j, aa[i][j]);
		Arrays.sort(ee, (e, f) -> e.a - f.a);
		for (int h = 0, h_; h < m; ) {
			h_ = h + 1;
			while (h_ < m && ee[h_].a == ee[h].a)
				h_++;
			int cnt = h_ - h;
			while (h < h_) {
				E e = ee[h];
				cnt -= join(e.i, e.j);
				h++;
			}
			if (cnt != 0) {
				println("NOT MAGIC");
				return;
			}
		}
		println("MAGIC");
	}
}
