T = int(input())
for t in range(T):
	(n, m) = map(int, input().split())
	if m < n:
		print(0)
		continue
	N = bin(n)[2:]
	P = bin(m + 1)[2:]
	k = ['0'] * max(len(N), len(P))
	N = '0' * (len(P) - len(N)) + N
	for i in range(len(P)):
		if N[i] == '1' and P[i] == '0':
			break
		if P[i] == N[i]:
			k[i] = '0'
		else:
			k[i] = '1'
	print(int(''.join(k), 2))
