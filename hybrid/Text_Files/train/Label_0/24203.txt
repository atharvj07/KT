#include "bits/stdc++.h"
using namespace std;
int dp[2019][2];
signed main()
{
	string s;
	cin >> s;
	int ans = 0;
	for(int i = 0; i < s.size(); ++i)
	{
		for(int j = 0; j < 2019; ++j)
		{
			dp[(j * 10 + s[i] - '0') % 2019][i % 2] += dp[j][!(i % 2)];
			dp[j][!(i % 2)] = 0;
		}
		++dp[s[i] - '0'][i % 2];
		ans += dp[0][(i % 2)];
	}
	cout << ans << "\n";
}
