(a, b, c) = [int(x) for x in input().split()]
w = min(a // 3, b // 2, c // 2)
r = [a - 3 * w, b - 2 * w, c - 2 * w]
m = [0, 0, 1, 2, 0, 2, 1]
hi = 0
for i in range(7):
	rr = r[:]
	d = 0
	while rr[m[(i + d) % 7]] > 0:
		rr[m[(i + d) % 7]] = rr[m[(i + d) % 7]] - 1
		d = d + 1
	hi = max(hi, d)
print(7 * w + hi)
