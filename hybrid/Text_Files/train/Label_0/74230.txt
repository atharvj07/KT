N = int(input())
l = list(map(int, input().strip().split()))
q = [(0, 0)] * N
idx = 0
for x in reversed(l):
	xn = x
	xd = 1
	while idx:
		(n, d) = q[idx - 1]
		if xn * d < n * xd:
			break
		q[idx - 1] = (0, 0)
		xn += n
		xd += d
		idx -= 1
	q[idx] = (xn, xd)
	idx += 1
for (xn, xd) in reversed(q):
	if not xd:
		continue
	s = f'{xn / xd:.10f}'
	for _ in range(xd):
		print(s)
