#include <bits/stdc++.h>
using namespace std;
const int MOD = int(1e9 + 7);
long long n, m, dp[2005][3000], ans = 0;
string s;
int main() {
  cin >> n >> m >> s;
  dp[0][0] = 1;
  for (int i = 1; i <= n - m; i++) {
    dp[i][0] = dp[i - 1][1];
    for (int j = 1; j <= i; j++) {
      dp[i][j] = (dp[i - 1][j + 1] + dp[i - 1][j - 1]) % MOD;
    }
  }
  int b = 0, mmin = int(1e9);
  for (int i = 0; i < m; i++) {
    if (s[i] == '(')
      b++;
    else
      b--;
    mmin = min(mmin, b);
  }
  for (int i = 0; i <= n - m; i++) {
    for (int j = 0; j <= i; j++) {
      if (j >= -mmin) {
        if (j + b <= n - m && j + b >= 0)
          ans += (dp[i][j] * dp[n - m - i][j + b]) % MOD;
        ans %= MOD;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
