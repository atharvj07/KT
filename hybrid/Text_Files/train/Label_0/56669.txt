

import java.util.Scanner;


public class JavaApplication1 {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        
        
        int a = input.nextInt();
        int b = input.nextInt();
        int c = input.nextInt();


        int total =0;
        while(a >= 1 && b >= 2 && c >= 4){

            a--;
            b-=2;
            c-=4;
            total+=7;
        }
        
       System.out.println(total);
        
        
        
    
     }
    

}
