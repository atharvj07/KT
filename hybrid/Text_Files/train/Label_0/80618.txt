n = int(input())
grid = []
for i in range(n):
	tmp = input().split()
	tmp = [int(x) for x in tmp]
	grid.append(tmp)
a = int((grid[0][1] * grid[0][2] // grid[1][2]) ** 0.5)
ans = [a]
for i in range(1, n):
	ans.append(grid[0][i] // a)
outputBuffer = ''
for x in ans:
	outputBuffer += str(x) + ' '
print(outputBuffer)
