n = int(input())
l = list(map(int, input().split()))
index_position = []
minimum = min(l)
for i in range(len(l)):
	if l[i] == minimum:
		index_position.append(i)
index_minimum = index_position[0]
index_minimimu = n - 1 - index_position[-1]
maxi = index_minimimu + index_minimum
for i in range(1, len(index_position)):
	if index_position[i] - index_position[i - 1] - 1 > maxi:
		maxi = index_position[i] - index_position[i - 1] - 1
print(n * minimum + maxi)
