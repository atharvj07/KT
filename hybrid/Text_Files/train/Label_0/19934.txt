#include <bits/stdc++.h>
using namespace std;
bool isPow(int n) {
  double pot = pow(2, round(log2(n)));
  return pot == n;
}
int main() {
  int n, b, p;
  cin >> n >> b >> p;
  int bottles = 0;
  int towels = n * p;
  if (isPow(n)) {
    int cant = pow(2, log2(n)) - 1;
    bottles += cant * b * 2 + cant;
  } else {
    int k = n - pow(2, round(log2(n)));
    int cant = pow(2, log2(n - k)) - 1;
    bottles += cant * b * 2 + cant;
    bottles += k * b * 2 + k;
  }
  cout << bottles << " " << towels << endl;
  return 0;
}
