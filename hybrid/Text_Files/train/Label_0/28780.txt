#include <bits/stdc++.h>
using namespace std;
int n, bom[1001], sum[200001];
int maxi;
int main() {
  cin >> n;
  int i, j;
  for (int i = 1; i <= n; i++) cin >> bom[i];
  for (i = 1; i < n; i++)
    for (j = i + 1; j <= n; j++) {
      sum[bom[i] + bom[j]]++;
      maxi = max(maxi, sum[bom[i] + bom[j]]);
    }
  cout << maxi;
  return 0;
}
