#include <bits/stdc++.h>
using namespace std;
int res(int n) {
  int ans = 0;
  while (n != 0) {
    ans += n % 10;
    n /= 10;
  }
  return ans;
}
void solve() {
  int n;
  cin >> n;
  int ans = 0;
  while (n) {
    ans++;
    if (res(ans) == 10) n--;
  }
  cout << ans;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  solve();
}
