#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int cntx[105] = {}, cnty[105]{};
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      char ch;
      cin >> ch;
      if (ch == '*') {
        cntx[i + 1]++;
        cnty[j + 1]++;
      }
    }
  }
  int ansx, ansy;
  for (int i = 1; i <= max(n, m); i++) {
    if (cntx[i] == 1) ansx = i;
    if (cnty[i] == 1) ansy = i;
  }
  cout << ansx << " " << ansy;
}
