import os
from io import BytesIO
input = BytesIO(os.read(0, os.fstat(0).st_size)).readline

def update(arr, index, value):
	while index <= len(arr) - 1:
		arr[index] += value
		index += index & -index

def sum(arr, index):
	result = 0
	while index > 0:
		result += arr[index]
		index -= index & -index
	return result

def init(fenwick, char, items):
	if char not in fenwick:
		fenwick[char] = [0] * (items + 1)

def main():
	s = list(input().strip())
	fenwick = {}
	for (index, char) in enumerate(s):
		init(fenwick, char, len(s))
		update(fenwick[char], index + 1, 1)
	q = int(input())
	while q:
		q -= 1
		(a, b, c) = input().split()
		a = int(a)
		b = int(b)
		if a == 2:
			c = int(c)
			count = 0
			for tree in fenwick.values():
				if sum(tree, c) - sum(tree, b - 1) > 0:
					count += 1
			print(count)
		else:
			c = ord(c)
			init(fenwick, s[b - 1], len(s))
			init(fenwick, c, len(s))
			update(fenwick[s[b - 1]], b, -1)
			update(fenwick[c], b, 1)
			s[b - 1] = c
main()
