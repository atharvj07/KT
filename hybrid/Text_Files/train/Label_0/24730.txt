#include <bits/stdc++.h>
using namespace std;
int main() {
  int x, y;
  while (~scanf("%d%d", &x, &y)) {
    double z = sqrt(x * x * 1.0 + y * y * 1.0);
    int k = (int)z;
    if (z == k) {
      printf("black\n");
      continue;
    }
    if (x > 0 && y > 0 || x < 0 && y < 0) {
      if (k & 1)
        printf("white\n");
      else
        printf("black\n");
    } else {
      if (k & 1)
        printf("black\n");
      else
        printf("white\n");
    }
  }
}
