from sys import stdin, setrecursionlimit
from bisect import bisect_right
setrecursionlimit(15000)

def check(num, arr):
	j = 0
	for i in range(len(arr)):
		if min(arr[i], arr[j]) < num * abs(i - j):
			return False
	j = len(arr) - 1
	for i in range(len(arr)):
		if min(arr[i], arr[j]) < num * abs(i - j):
			return False
	return True
n = int(stdin.readline().strip())
arr = list(map(int, stdin.readline().split()))
start = 0
end = 10 ** 9
while True:
	if start >= end:
		if check(start, arr):
			pass
		else:
			start -= 1
		break
	mid = (start + end) // 2
	if check(mid, arr):
		start = mid + 1
	else:
		end = mid - 1
if check(start + 1, arr):
	print(max(0, start + 1))
else:
	print(max(0, start))
