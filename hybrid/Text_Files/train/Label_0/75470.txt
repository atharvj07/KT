import atexit
import io
import math
import sys
_INPUT_LINES = sys.stdin.read().splitlines()
input = iter(_INPUT_LINES).__next__
_OUTPUT_BUFFER = io.StringIO()
sys.stdout = _OUTPUT_BUFFER

@atexit.register
def write():
	sys.__stdout__.write(_OUTPUT_BUFFER.getvalue())

def main():
	t = int(input())
	for _ in range(t):
		x = int(input())
		if x == 0:
			print(1, 1)
			continue
		n = math.ceil(x ** 0.5)
		while True:
			if n * n - (n // 2) ** 2 > x:
				print(-1)
				break
			t = math.floor((n * n - x) ** 0.5)
			if t > 0 and t * t == n * n - x:
				m = n // t
				if t == n // m:
					print(n, m)
					break
			n += 1
main()
