#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  int mn = 1000000000;
  sort(a.begin(), a.end());
  for (int i = 0; i <= (n - 1) / 2; ++i) {
    mn = min(mn, a[n - 1 - i] - a[(n - 1) / 2 - i]);
  }
  cout << mn;
  return 0;
}
