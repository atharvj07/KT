#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, c = 0, count = 0;
  cin >> n;
  string *s = new string[n];
  for (int i = 0; i < n; i++) cin >> s[i];
  for (int i = 0; i < s[0].size(); i++) {
    for (int j = 0; j < n - 1; j++) {
      if (s[j][i] == s[j + 1][i]) {
        c++;
      }
    }
    if (c == n - 1) {
      count++;
      c = 0;
    } else {
      break;
    }
  }
  cout << count << endl;
  return 0;
}
