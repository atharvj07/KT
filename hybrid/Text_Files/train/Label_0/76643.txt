t = int(input())
for z in range(t):
	n = int(input())
	arr = list(map(int, input().split()))
	if len(set(arr)) == 1:
		print('NO')
	else:
		print('YES')
		rep = []
		for i in range(1, n):
			if arr[0] == arr[i]:
				rep.append(i)
			else:
				print('1', i + 1)
				k = i
		for num in rep:
			print(k + 1, num + 1)
