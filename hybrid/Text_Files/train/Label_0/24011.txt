#include <bits/stdc++.h>

class Solve {
private:
	struct Segment {
		int64_t num{}, length{};
		std::vector<bool> left, right;
	};

	std::string S, Q;
	int Slength, Qlength;
	std::vector<Segment> charSegment;
	Segment zeroSegment;
	
	Segment calcGenomes(int& s_i)
	{
		Segment ret{zeroSegment};
		while (s_i != Slength && S[s_i] != ')')
			ret = segmentMultiple(ret, calcOneGenome(s_i));
		return ret;
	}

	Segment calcOneGenome(int& s_i)
	{
		Segment ret;
		if (isupper(S[s_i]))
		{
			ret = charSegment[S[s_i] - 'A'];
			s_i++;
		}
		else
		{
			int64_t exponential{calcNumber(s_i)};
			if (S[s_i] == '(')
			{
				s_i++;
				Segment base{calcGenomes(s_i)};
				s_i++;
				ret = binExponent(base, exponential);
			}
			else
			{
				Segment base{charSegment[S[s_i] - 'A']};
				s_i++;
				ret = binExponent(base, exponential);
			}
		}
		return ret;
	}

	int64_t calcNumber(int& s_i)
	{
		int64_t ret{};
		while (isdigit(S[s_i]))
		{
			ret = 10 * ret + S[s_i] - '0';
			s_i++;
		}
		return ret;
	}

	Segment binExponent(const Segment& base, const int64_t exponential)
	{
		Segment ret{zeroSegment}, power{base};
		for (int64_t i{exponential}; i > 0; i >>= 1, power = segmentMultiple(power, power))
			if (i & 1)
				ret = segmentMultiple(ret, power);
		return ret;
	}

	Segment segmentMultiple(const Segment& a, const Segment& b)
	{
		Segment ret;
		ret.left = a.left;
		ret.right = b.right;
		ret.num = a.num + b.num;
		ret.length = a.length + b.length;

		for (int i{1}; i < Qlength; i++)
		{
			if (a.right[i] && b.left[i])
			{
				if (i - a.length <= 0 && Qlength <= i + b.length)
					ret.num++;
			}
			else if (a.right[i])
			{
				if (i - a.length >= 0)
					ret.left[i - a.length] = false;
			}
			else if (b.left[i])
			{
				if (i + b.length <= Qlength)
					ret.right[i + b.length] = false;
			}
		}
		return ret;
	}

public:
	bool is_last_query{};
	Solve()
	{
		std::cin >> S;
		if (S == "#")
		{
			is_last_query = true;
			return;
		}
		std::cin >> Q;
		Slength = (int)S.size();
		Qlength = (int)Q.size();
		
		charSegment.resize(26);
		for (int c_i{}; c_i < 26; c_i++)
		{
			charSegment[c_i].left.resize(Qlength + 1);
			charSegment[c_i].right.resize(Qlength + 1);
			charSegment[c_i].length = 1;
			if (Q.size() == 1 && Q.front() == 'A' + c_i)
				charSegment[c_i].num = 1;
			for (int q_i{}; q_i < Qlength; q_i++)
				if (Q[q_i] == 'A' + c_i)
					charSegment[c_i].left[q_i] = charSegment[c_i].right[q_i + 1] = true;
		}
		zeroSegment.left.resize(Qlength + 1, true);
		zeroSegment.right.resize(Qlength + 1, true);

		int s_i{};
		std::cout << calcGenomes(s_i).num << std::endl;
	}
};

int main()
{
	while (!Solve().is_last_query);

	return 0;
}
