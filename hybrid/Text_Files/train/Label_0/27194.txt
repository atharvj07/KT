long long n;
int getans() {
  int ret = 0;
  if (n % 3) return 0;
  n /= 3;
  for (long long i = 2; i * i * i <= n; i++) {
    if (n % i) continue;
    for (long long j = (long long)sqrt(n / i); j >= i; j--) {
      long long k = (n / i) / j;
      if (i + j <= k) break;
      if ((n / i) % j) continue;
      if ((i + j + k) & 1) continue;
      ret += (i == j && j == k) ? 1 : ((i == j || j == k || i == k) ? 3 : 6);
    }
  }
  return ret;
}
