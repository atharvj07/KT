def solution():
	(n0, n1, n2) = map(lambda x: int(x), input().split())
	answer = ''
	if n1 == 0:
		if n0 > 0:
			answer = ''.join(['0'] * (n0 + 1))
		else:
			answer = ''.join(['1'] * (n2 + 1))
	else:
		for i in range(n1 + 1):
			if i % 2 == 0:
				answer = ''.join([answer, '1'])
			else:
				answer = ''.join([answer, '0'])
		answer = ''.join([answer[:1], '0' * n0, answer[1:]])
		answer = ''.join(['1' * n2, answer])
	print(answer)
t = int(input())
for _ in range(t):
	solution()
