R = lambda : list(map(int, input().split()))
n = int(input())
l = R()
r = R()
q = []
for i in range(n):
	q.append(n - l[i] - r[i])
fl = True
for i in range(n):
	lx = 0
	rx = 0
	for j in range(n):
		if q[i] < q[j]:
			if i < j:
				rx += 1
			else:
				lx += 1
	if lx != l[i] or rx != r[i]:
		fl = False
		break
if not fl:
	print('NO')
else:
	print('YES')
	print(' '.join((str(e) for e in q)))
