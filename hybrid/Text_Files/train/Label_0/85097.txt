(ax, ay, bx, by, cx, cy) = list(map(int, input().split()))
dab = (ax - bx) ** 2 + (ay - by) ** 2
dbc = (bx - cx) ** 2 + (by - cy) ** 2
if dab != dbc:
	print('No')
elif ax == bx or bx == cx:
	if ax == bx and bx == cx:
		print('No')
	else:
		print('Yes')
else:
	mab = (ay - by) / (ax - bx)
	mbc = (by - cy) / (bx - cx)
	if mab == mbc:
		print('No')
	else:
		print('Yes')
