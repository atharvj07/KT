#include <iostream>
using namespace std;
int n,m,x,y;
int f[100005];
long long answer[100005],Size[100005];
struct Edge
{
	int u,v;
}e[100005];
int find_set(int x)
{
	if(x!=f[x]) f[x]=find_set(f[x]);
	return f[x];
}
int main()
{
	cin>>n>>m;
	for(int i=1;i<=n;i++) f[i]=i,Size[i]=1;
	for(int i=1;i<=m;i++)
	{
		cin>>x>>y;
		e[i].u=x;
		e[i].v=y;
	}
	answer[m]=(long long)n*(n-1)/2;
	for(int i=m;i>=1;i--)
	{
		if(find_set(e[i].u)==find_set(e[i].v))
		    answer[i-1]=answer[i];
	    else
	    {
	    	answer[i-1]=answer[i]-Size[find_set(e[i].u)]*Size[find_set(e[i].v)];
			Size[find_set(e[i].v)]+=Size[find_set(e[i].u)];
			f[find_set(e[i].u)]=find_set(e[i].v);
		}
	}
	for(int i=1;i<=m;i++) cout<<answer[i]<<endl;
	return 0;
}
