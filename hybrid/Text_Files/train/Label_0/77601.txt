import sys
import copy
input = sys.stdin.readline
from math import *
(n, m) = map(int, input().split())
a = [int(x) for x in input().split()]
d = {}
s = 1
p = []
for i in a:
	if i % m in d:
		print(0)
		exit(0)
	else:
		d[i % m] = 1
		p.append(i % m)
for i in range(len(p)):
	for j in range(i + 1, len(p)):
		s *= abs(a[i] - a[j])
		s %= m
print(s % m)
