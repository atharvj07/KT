n = int(input())
parent = [0] + [x - 1 for x in list(map(int, input().split()))]
citizen = list(map(int, input().split()))
sz = [1] * n
for i in range(1, n):
	sz[parent[i]] = 0
for i in range(n - 1, 0, -1):
	citizen[parent[i]] += citizen[i]
	sz[parent[i]] += sz[i]
ans = 0
for i in range(n):
	ans = max(ans, (citizen[i] + sz[i] - 1) // sz[i])
print(ans)
