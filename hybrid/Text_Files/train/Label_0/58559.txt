import static java.lang.Math.abs;
import java.util.*;
public class Le {

    public static void main(String[] args) {
        Scanner s=new Scanner(System.in);
        int n=s.nextInt();
        int m=s.nextInt();
        int b[]=new int[n];
        int g[]=new int[m];
        int bo=s.nextInt();
        for(int i=0;i<bo;i++){
            b[s.nextInt()]=1;
        }
        int gi=s.nextInt();
        for(int i=0;i<gi;i++){
            g[s.nextInt()]=1;
        }
        //int gc=1;
        int pr=3*m*n;
        for(int i=0;i<=pr;i++){
             if(b[(i%n)]==1 || g[(i%m)]==1){
                 b[(i%n)]=1;
                 g[(i%m)]=1;
             }
        }
        if(gi==0 && bo==0){
             System.out.println("No");
        }
        else{
             int i,j;
             for( i=0;i<n;i++){
                if(b[i]==0){
                     break;
                }
             }
             if(i<n){
                System.out.println("No");
                 //System.out.println("hey");;
              }
              else{
                for( j=0;j<m;j++){
                     if(g[j]==0){
                         break;
                     }
                }
                if(j<m){
                    System.out.println("No");
                    //System.out.println("bc");
                }
                else{
                  System.out.println("Yes");
                }
              }
         }  
    }
}
