from sys import stdin, stdout
t = int(stdin.readline())
for _ in range(t):
	(n, k) = map(int, stdin.readline().strip().split())
	s = input()
	arr = list(s)
	pre = []
	suf = []
	if arr[0] == '1':
		pre.append(0)
	else:
		pre.append(-1)
	for i in range(1, n):
		if arr[i] == '1':
			pre.append(i)
		else:
			pre.append(pre[-1])
	if arr[-1] == '1':
		suf.append(n - 1)
	else:
		suf.append(-1)
	for i in range(n - 2, -1, -1):
		if arr[i] == '1':
			suf.append(i)
		else:
			suf.append(suf[-1])
	suf = suf[::-1]
	ans = 0
	rec = -1
	for i in range(n):
		x = pre[i]
		y = suf[i]
		if x != -1 and y != -1:
			if abs(i - x) > k and abs(i - y) > k:
				if rec == -1:
					ans += 1
					rec = i
				elif abs(rec - i) > k:
					ans += 1
					rec = i
		elif x != -1 and y == -1:
			if abs(i - x) > k:
				if rec == -1:
					ans += 1
					rec = i
				elif abs(rec - i) > k:
					ans += 1
					rec = i
		elif x == -1 and y != -1:
			if abs(i - y) > k:
				if rec == -1:
					ans += 1
					rec = i
				elif abs(rec - i) > k:
					ans += 1
					rec = i
		elif rec == -1:
			ans += 1
			rec = i
		elif abs(rec - i) > k:
			ans += 1
			rec = i
	print(ans)
