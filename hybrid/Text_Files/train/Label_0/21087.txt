#include <bits/stdc++.h>
using namespace std;
int main(void) {
  string a, b, s;
  cin >> a >> b >> s;
  char a_layout[26]{};
  for (int i = 0; i < 26; ++i) a_layout[a[i] - 'a'] = i;
  string o;
  for (char c : s) {
    if (islower(c))
      o += b[a_layout[c - 'a']];
    else if (isupper(c))
      o += toupper(b[a_layout[tolower(c) - 'a']]);
    else
      o += c;
  }
  cout << o << endl;
  return 0;
}
