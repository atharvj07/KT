#include<iostream>
using namespace std;

int main(){
  int h,w;
  long long l;
  int x,y,d;
  string g[100];

  int vis[100][100][4];
  char dir[] = {'N','E','S','W'};
  int dy[] = {-1,0,1,0}, dx[] = {0,1,0,-1};

  int px[40100],py[40100],pd[40100];

  while(cin >> h >> w >> l,h){
    for(int i=0;i<h;i++)cin >> g[i];

    for(int i=0;i<h;i++){
      for(int j=0;j<w;j++){
	for(int k=0;k<4;k++){
	  if(g[i][j] == dir[k]){
	    y = i; x = j; d = k;
	  }
	  vis[i][j][k] = -1;
	}
      }
    }

    int step = 0;
    for(;;){
      if(vis[y][x][d]>=0)break;
      vis[y][x][d] = step;
      
      py[step] = y; px[step] = x; pd[step] = d;
      step++;

      for(int i=0;i<4;i++){
	int nd = (d+i)%4;
	int ny = y+dy[nd], nx = x+dx[nd];
	if(ny<0 || nx<0 || ny>=h || nx>=w)continue;
	if(g[ny][nx] == '#')continue;
	y = ny; x = nx; d = nd;
	break;
      }
    }

    long long head = vis[y][x][d], now = l;
    if(step<l){
      now -= head;
      now %= step-head;
      now += head;
    }
    cout << py[now]+1 << " " << px[now]+1 << " " << dir[pd[now]] << endl;
  }
}
