from queue import Queue

def task1():
	p = []
	buffer = Queue()
	p.append(q[0])
	for x in range(1, q[0]):
		buffer.put(x)
	for i in range(n - 1):
		if q[i + 1] == q[i]:
			temp = buffer.get()
			p.append(temp)
		else:
			p.append(q[i + 1])
			for x in range(q[i] + 1, q[i + 1]):
				buffer.put(x)
	for val in p:
		print(val, end=' ')
	print()

def task2():
	p = []
	buffer = []
	p.append(q[0])
	for x in range(1, q[0]):
		buffer.append(x)
	for i in range(n - 1):
		if q[i + 1] == q[i]:
			temp = buffer.pop()
			p.append(temp)
		else:
			p.append(q[i + 1])
			for x in range(q[i] + 1, q[i + 1]):
				buffer.append(x)
	for val in p:
		print(val, end=' ')
	print()
t = int(input())
for _ in range(t):
	n = int(input())
	q = list(map(int, input().strip().split()))
	task1()
	task2()
