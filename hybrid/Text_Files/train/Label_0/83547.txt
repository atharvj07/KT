from collections import Counter
T = int(input())
for i in range(T):
	n = int(input())
	s = []
	a = list(map(int, input().strip().split()))
	c = Counter(a)
	ans = [-1] * (n + 1)
	ans[0] = c[0]
	(st, cp) = (0, 0)
	for j in range(n):
		cp += c[j]
		if cp <= j:
			break
		if c[j] == 0:
			st += j - s.pop()
		ans[j + 1] = st + c[j + 1]
		s += [j] * (c[j] - 1)
	for j in range(n + 1):
		print(ans[j], end=' ')
	print('')
