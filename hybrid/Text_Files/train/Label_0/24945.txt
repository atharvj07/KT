#include<iostream>
#include<queue>
#include<string>
#define rep(i,n) for(int i=0;i<n;i++)
#define ck(a,b) (0<=(a)&&(a)<b)
using namespace std;
int dx[]={-1,1,0,0},dy[]={0,0,-1,1};
struct S{
	int lx,ly,rx,ry;
	S(int a,int b,int c,int d):lx(a),ly(b),rx(c),ry(d){}
};
bool V[50][50][50][50];
int main(){
	int w,h;
	while(cin>>w>>h,w){
		int lx,ly,rx,ry;
		string cavel[h],caver[h];
		rep(i,h)cin>>cavel[i]>>caver[i];
		rep(i,h)rep(j,w){
			if(cavel[i][j]=='L')ly=i,lx=j,cavel[i][j]='.';
			if(caver[i][j]=='R')ry=i,rx=j,caver[i][j]='.';
		}
		queue<S> F; rep(i,h)rep(j,w)rep(k,h)rep(l,w)V[i][j][k][l]=0;
		F.push(S(lx,ly,rx,ry)); V[ly][lx][ry][rx]=1;
		int goal=0;
		while(!F.empty()&&!goal){
			S s=F.front(); F.pop();
			rep(d,4){
				int clx=s.lx,cly=s.ly,crx=s.rx,cry=s.ry,nlx=clx+dx[d],nly=cly+dy[d],nrx=crx-dx[d],nry=cry+dy[d];
				if(!ck(nlx,w)||!ck(nly,h)||cavel[nly][nlx]=='#')nlx=clx,nly=cly;
				if(!ck(nrx,w)||!ck(nry,h)||caver[nry][nrx]=='#')nrx=crx,nry=cry;
				if(nlx==clx&&nly==cly&&nrx==crx&&nry==cry||(cavel[nly][nlx]=='%'^caver[nry][nrx]=='%'))continue;
				if(cavel[nly][nlx]=='%'&&caver[nry][nrx]=='%'){
					goal=1; goto END;
				}
				if(!V[nly][nlx][nry][nrx]){
					F.push(S(nlx,nly,nrx,nry));
					V[nly][nlx][nry][nrx]=1;
				}
			}
		}
		END:cout<<(goal?"Yes":"No")<<endl;
	}
	return 0;
}
