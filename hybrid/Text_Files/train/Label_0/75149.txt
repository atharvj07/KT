from collections import Counter, defaultdict
from sys import stdin, stdout
input = stdin.readline
I = lambda : int(input())
M = lambda : map(int, input().split())
LI = lambda : list(map(int, input().split()))

def solve():
	n = I()
	l = LI()
	d = Counter(l)
	a = list(d.values())
	m = max(a)
	c = a.count(m)
	ans = n - m - c + 1
	print(ans // (m - 1))
for _ in range(I()):
	solve()
