a = list(map(int, input().split()))
sum1 = a[0] + a[1] + a[2] + a[3]
if sum1 % 2 != 0:
	print('NO')
elif a[0] + a[1] == a[2] + a[3]:
	print('YES')
elif a[0] + a[2] == a[1] + a[3]:
	print('YES')
elif a[0] + a[3] == a[1] + a[2]:
	print('YES')
elif a[0] == sum1 // 2:
	print('YES')
elif a[1] == sum1 // 2:
	print('YES')
elif a[2] == sum1 // 2:
	print('YES')
elif a[3] == sum1 // 2:
	print('YES')
else:
	print('NO')
