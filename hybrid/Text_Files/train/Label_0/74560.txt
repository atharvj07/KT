from math import factorial as fact
from itertools import combinations
from collections import Counter

def multinomial(L):
	res = fact(sum(L))
	for x in L:
		res //= fact(x)
	if L[0] == 0:
		return res
	L[0] -= 1
	lead0 = fact(sum(L))
	for x in L:
		lead0 //= fact(x)
	L[0] += 1
	return res - lead0
