#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
long long int l[maxn], r[maxn];
int main() {
  long long int n, m;
  cin >> n >> m;
  for (long long int i = 1; i <= n; i++) cin >> l[i] >> r[i];
  long long int ans = m + r[1] - l[1], t = 0, j = 1;
  for (long long int i = 2; i <= n; i++) {
    t += l[i] - r[i - 1];
    while (t >= m) t -= l[j + 1] - r[j], j++;
    long long int res = r[i] - l[j] + m - t;
    ans = max(ans, res);
  }
  cout << ans << endl;
  return 0;
}
