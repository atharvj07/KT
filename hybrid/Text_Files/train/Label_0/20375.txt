#include <bits/stdc++.h>
using namespace std;
int n, m;
vector<int> v[200005];
int main() {
  int i, j;
  int st, dr, mid;
  scanf("%d", &n);
  for (i = 0; i < n; i++) {
    scanf("%d", &m);
    st = 1;
    dr = n;
    while (st <= dr) {
      mid = (st + dr) >> 1;
      v[mid].size() == 0 || v[mid].back() < m ? (st = mid + 1) : (dr = mid - 1);
    }
    v[dr].push_back(m);
  }
  for (i = n; i > 0; i--) {
    if (v[i].size() > 0) {
      for (j = 0; j < v[i].size(); ++j) {
        cout << v[i][j] << ' ';
      }
      cout << "\n";
    }
  }
  return 0;
}
