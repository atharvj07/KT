/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
	public static void main(String[] args) {
		InputStream inputStream = System.in;
		OutputStream outputStream = System.out;
		TokenizerReader in = new TokenizerReader(inputStream);
		PrintWriter out = new PrintWriter(outputStream);
		TaskD solver = new TaskD();
		solver.solve(1, in, out);
		out.close();
	}
}

class TaskD {
    final static long MOD = 1000000000 + 7;
    public void solve(@SuppressWarnings("UnusedParameters") int testNumber, TokenizerReader in, PrintWriter out) {
        int n = in.nextInt(), k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = in.nextInt() / 2;
        }

        int MAX_MASK = 1 << (k+1);
        long[][] dp = new long[n+1][MAX_MASK];
        for (int mask = 0; mask < MAX_MASK; ++mask)
            if ((mask & (1<<k)) > 0)
                dp[n][mask] = 1;

        for (int i = n-1; i >= 0; --i) {
            for (int mask = 0; mask < MAX_MASK; ++mask) {
                if (a[i] == 0)
                    dp[i][mask] = f(dp, i, mask, 1, k) + f(dp, i, mask, 2, k);
                else
                    dp[i][mask] = f(dp, i, mask, a[i], k);
                dp[i][mask] %= MOD;
            }
        }

        out.println(dp[0][0]);
    }

    private long f(long[][] dp, int n, int mask, int incoming, int k) {
        if ((mask & (1<<k)) == 0) {
            while (Integer.lowestOneBit(mask) == (1 << incoming)) {
                mask ^= (1 << incoming);
                ++incoming;
            }
            if (Integer.lowestOneBit(mask) < (1<<incoming))
                mask = 1<<incoming;
            else
                mask ^= (1<<incoming);
        }
        return dp[n+1][mask];
    }
}

class TokenizerReader extends StreamTokenizer {
    public TokenizerReader(InputStream in) {
        super(new BufferedReader(new InputStreamReader(in)));
        defaultConfig();
    }

    public String nextString() {
        try {
            nextToken();
        } catch (IOException e) {
            throw new RuntimeException("nextString: exception in line " + lineno(), e);
        }
        return sval;
    }

    public int nextInt() {
        return Integer.parseInt(nextString());
    }

    public void defaultConfig() {
        resetSyntax();
        wordChars(33, 126);
        whitespaceChars(0, ' ');
    }

}
