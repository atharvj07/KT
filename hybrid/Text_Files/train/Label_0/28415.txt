#include <bits/stdc++.h>
using namespace std;
long long int modulo = 1e9 + 7;
vector<vector<long long int> > operator*(vector<vector<long long int> > m1,
                                         vector<vector<long long int> > m2) {
  int n = int(m1.size());
  vector<vector<long long int> > r(n, vector<long long int>(n, 0));
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      for (int k = 0; k < n; k++)
        r[i][j] = (r[i][j] + m1[i][k] * m2[k][j]) % modulo;
  return r;
}
vector<long long int> operator*(vector<vector<long long int> > m,
                                vector<long long int> v) {
  int n = int(m.size());
  vector<long long int> r(n, 0);
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) r[i] = (r[i] + m[i][j] * v[j]) % modulo;
  return r;
}
vector<vector<long long int> > eleva(vector<vector<long long int> > m, int e) {
  if (e == 0) {
    int n = int(m.size());
    vector<vector<long long int> > r(n, vector<long long int>(n, 0));
    for (int i = 0; i < n; i++) r[i][i] = 1;
    return r;
  }
  if (e % 2 == 0) return eleva(m * m, e / 2);
  return m * eleva(m * m, e / 2);
}
int main() {
  int casoant = 0;
  vector<long long int> maneras(1, 1);
  for (int caso = 1; caso <= 7; caso++) {
    int w;
    cin >> w;
    if (w > 0) {
      vector<long long int> nextmaneras(1 << caso, 0);
      for (int c = 0; c < 1 << casoant; c++) {
        for (int cver = 0; cver < 1 << casoant; cver++) {
          if (c & cver) continue;
          for (int chor = 0; chor < 1 << (caso - 1); chor++) {
            int chortop = chor | (1 << (caso - 1));
            int chordown = (chor << 1) | 1;
            int cverext = cver | ((1 << caso) - (1 << casoant));
            nextmaneras[chortop & chordown & cverext] =
                (nextmaneras[chortop & chordown & cverext] + maneras[c]) %
                modulo;
          }
        }
      }
      vector<vector<long long int> > m(1 << caso,
                                       vector<long long int>(1 << caso, 0));
      for (int c = 0; c < 1 << caso; c++) {
        for (int cver = 0; cver < 1 << caso; cver++) {
          if (c & cver) continue;
          for (int chor = 0; chor < 1 << (caso - 1); chor++) {
            int chortop = chor | (1 << (caso - 1));
            int chordown = (chor << 1) | 1;
            m[chortop & chordown & cver][c]++;
            m[chortop & chordown & cver][c] %= modulo;
          }
        }
      }
      maneras = eleva(m, w - 1) * nextmaneras;
      casoant = caso;
    }
  }
  cout << maneras[0] << endl;
}
