def solve():
	(n, m) = [int(x) for x in input().split()]
	a = []
	for i in range(n):
		v = [int(x) for x in input().split()]
		a.append(v)
	xorsums = [0] * (n + m - 1)
	for i in range(n):
		for j in range(m):
			xorsums[i + j] ^= a[i][j]
	f = 0
	for i in range(n + m - 1):
		if xorsums[i] != 0:
			f = 1
	if f == 1:
		print('Ashish')
	else:
		print('Jeel')

def main():
	t = int(input())
	for i in range(t):
		solve()
main()
