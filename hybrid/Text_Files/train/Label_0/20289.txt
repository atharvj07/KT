#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC optimize("unroll-loops")
#pragma GCC target("avx2")
using namespace std;
const int MOD = 998244353;
int C[5001][5001];
int fact[5001];
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  fact[0] = 1;
  for (int i = 1; i <= 5000; i++) {
    fact[i] = (long long)fact[i - 1] * i % MOD;
  }
  for (int i = 0; i <= 5000; i++) {
    C[i][i] = 1;
    C[i][0] = 1;
    for (int j = 1; j < i; j++) {
      C[i][j] = (C[i - 1][j] + C[i - 1][j - 1]) % MOD;
    }
  }
  int a, b, c;
  cin >> a >> b >> c;
  int ans1 = 0, ans2 = 0, ans3 = 0;
  for (int i = 0; i <= min(a, b); i++) {
    ans1 += (long long)C[a][i] * C[b][i] % MOD * fact[i] % MOD;
    ans1 %= MOD;
  }
  for (int i = 0; i <= min(a, c); i++) {
    ans2 += (long long)C[a][i] * C[c][i] % MOD * fact[i] % MOD;
    ans2 %= MOD;
  }
  for (int i = 0; i <= min(c, b); i++) {
    ans3 += (long long)C[c][i] * C[b][i] % MOD * fact[i] % MOD;
    ans3 %= MOD;
  }
  cout << (long long)ans1 * ans2 % MOD * ans3 % MOD << '\n';
  return 0;
}
