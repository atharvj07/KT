x = int(input())
y = input().split()
z = []
for i in y:
	z.append(int(i))
flips = []
for i in z:
	if i >= 0:
		flips.append(-i - 1)
	else:
		flips.append(i)
if x % 2 == 0:
	flips = flips
else:
	val = min(flips)
	judge = 0
	for j in range(x):
		if judge == 0:
			if flips[j] == val:
				flips[j] = -flips[j] - 1
				judge = 1
out = str(flips[0])
if x == 1:
	print(out)
else:
	for i in range(1, x):
		out += ' ' + str(flips[i])
	print(out)
