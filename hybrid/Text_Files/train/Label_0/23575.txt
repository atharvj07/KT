#include <bits/stdc++.h>
using namespace std;
int n, fix[100000], par[100000], a[200005], b[200005];
pair<int, pair<int, int> > las[100000];
vector<pair<int, pair<int, int> > > ans, la;
void fin() {
  cout << "YES\n";
  cout << ans.size() << '\n';
  for (int i = 0; i < ans.size(); i++)
    cout << ans[i].first << " " << ans[i].second.first << " "
         << ans[i].second.second << '\n';
  exit(0);
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    b[i] = a[i];
  }
  if (n <= 10) goto br;
  for (int i = n; i > 8; i--) {
    if (i <= 12) {
      if (a[i]) {
        a[i] ^= 1;
        a[i - 1] ^= 1;
        a[i - 2] ^= 1;
        ans.push_back(make_pair(i - 2, make_pair(i - 1, i)));
      }
    }
    if (!a[i]) continue;
    if (a[i] & a[i - 1] & a[i - 2]) {
      a[i] ^= 1;
      a[i - 1] ^= 1;
      a[i - 2] ^= 1;
      ans.push_back(make_pair(i - 2, make_pair(i - 1, i)));
      continue;
    }
    if (!a[i - 1]) {
      if (a[i - 2]) {
        a[i] ^= 1;
        a[i - 2] ^= 1;
        a[i - 4] ^= 1;
        ans.push_back(make_pair(i - 4, make_pair(i - 2, i)));
        continue;
      }
      a[i] ^= 1;
      a[i - 3] ^= 1;
      a[i - 6] ^= 1;
      ans.push_back(make_pair(i - 6, make_pair(i - 3, i)));
      continue;
    }
    int sum = a[i - 3] + a[i - 4] + a[i - 5];
    if (sum == 2) {
      int fi = 0, se = 0;
      if (a[i - 5]) {
        if (!fi)
          fi = i - 5;
        else
          se = i - 5;
      }
      if (a[i - 4]) {
        if (!fi)
          fi = i - 4;
        else
          se = i - 4;
      }
      if (a[i - 3]) {
        if (!fi)
          fi = i - 3;
        else
          se = i - 3;
      }
      a[se - (i - se)] ^= 1;
      a[fi - (i - 1 - fi)] ^= 1;
      for (int j = i; j >= i - 5; j--) a[j] = 0;
      ans.push_back(make_pair(se - (i - se), make_pair(se, i)));
      ans.push_back(make_pair(fi - (i - 1 - fi), make_pair(fi, i - 1)));
      continue;
    }
    if (sum == 1) {
      int fi = 0, se = 0;
      if (a[i - 5]) fi = i - 5;
      if (a[i - 4]) fi = i - 4;
      if (a[i - 3]) fi = i - 3;
      for (int j = i; j >= i - 5; j--) a[j] = 0;
      a[fi - (i - fi)] ^= 1;
      a[i - 11] ^= 1;
      a[i - 6] ^= 1;
      ans.push_back(make_pair(fi - (i - fi), make_pair(fi, i)));
      ans.push_back(make_pair(i - 11, make_pair(i - 6, i - 1)));
      continue;
    }
    if (sum == 0) {
      a[i - 12] ^= 1;
      a[i - 11] ^= 1;
      a[i - 1] ^= 1;
      a[i] ^= 1;
      ans.push_back(make_pair(i - 12, make_pair(i - 6, i)));
      ans.push_back(make_pair(i - 11, make_pair(i - 6, i - 1)));
      continue;
    }
    ans.push_back(make_pair(i - 5, make_pair(i - 3, i - 1)));
    ans.push_back(make_pair(i - 8, make_pair(i - 4, i)));
    a[i - 8] ^= 1;
    for (int j = i; j >= i - 5; j--) a[j] = 0;
  }
  n = min(n, 8);
br:
  fix[0] = 1;
  queue<int> q;
  q.push(0);
  while (q.size()) {
    int x = q.front();
    q.pop();
    for (int i = 0; i < n; i++)
      for (int j = i + 1; j < n; j++) {
        int z = j + j - i;
        if (z >= n) continue;
        int fi = (x ^ (1 << i) ^ (1 << j) ^ (1 << z));
        if (!fix[fi]) {
          las[fi] = make_pair(i + 1, make_pair(j + 1, z + 1));
          par[fi] = x;
          fix[fi] = 1;
          q.push(fi);
        }
      }
  }
  int num = 0;
  for (int i = 0; i < n; i++)
    if (a[i + 1]) num += (1 << i);
  if (fix[num] == 0) {
    cout << "NO";
    return 0;
  }
  int x = num;
  while (x) {
    la.push_back(las[x]);
    x = par[x];
  }
  for (int i = 0; i < la.size(); i++) ans.push_back(la[i]);
  fin();
  return 0;
}
