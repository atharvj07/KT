#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1 << 18;
int N;
int p[MAXN];
int data[MAXN];
int find(int u) { return (data[u] < 0) ? u : data[u] = find(data[u]); }
void join(int u, int v) {
  u = find(u);
  v = find(v);
  if (u != v) {
    data[u] += data[v];
    data[v] = u;
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  memset(data, -1, sizeof data);
  cin >> N;
  for (int i = 0; i < N; ++i) {
    cin >> p[i];
    p[i]--;
  }
  string S;
  cin >> S;
  for (int i = 0; i < (int)S.size(); ++i) {
    if (S[i] == '1') join(i, i + 1);
  }
  for (int i = 0; i < N; ++i) {
    if (find(p[i]) != find(i)) {
      cout << "NO\n";
      return 0;
    }
  }
  cout << "YES\n";
  return 0;
}
