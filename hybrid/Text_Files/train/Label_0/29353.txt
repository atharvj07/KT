#include <bits/stdc++.h>
using namespace std;
int main(void) {
  {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
  };
  int n, i, j, a, b, k;
  cin >> n;
  vector<pair<int, int>> v;
  for (i = 0; i < n; i++) {
    cin >> a >> b;
    v.push_back({a, b});
  }
  long double dp[n + 1][n + 1];
  for (i = 0; i <= n; i++) {
    for (j = 0; j <= n; j++) dp[i][j] = 1e18;
  }
  set<long double> s1[n + 1];
  vector<long double> s;
  for (i = 0; i < n; i++) {
    for (j = i + 1; j < n; j++) {
      if (v[j].first - v[i].first != 0) {
        dp[i + 1][j + 1] = ((long double)(v[j].second - v[i].second)) /
                           (v[j].first - v[i].first);
        if (!s1[i + 1].count(dp[i + 1][j + 1])) s.push_back(dp[i + 1][j + 1]);
        s1[i + 1].insert(dp[i + 1][j + 1]);
        s1[j + 1].insert(dp[i + 1][j + 1]);
      } else {
        if (!s1[i + 1].count(dp[i + 1][j + 1])) s.push_back(1e18);
        s1[i + 1].insert(1e18);
        s1[j + 1].insert(1e18);
      }
    }
  }
  long long int ans = 0, cr = 0;
  int len = s.size();
  bool mark[len + 1][len + 1];
  for (i = 0; i <= len; i++) {
    for (j = 0; j <= len; j++) mark[i][j] = 0;
  }
  for (i = 0; i < len; i++) {
    for (j = i + 1; j < len; j++) {
      if (s[i] != s[j]) mark[i + 1][j + 1] = 1;
    }
  }
  for (i = 0; i <= len; i++) {
    for (j = 0; j <= len; j++) ans += mark[i][j];
  }
  cout << ans << endl;
}
