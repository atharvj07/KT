#include <bits/stdc++.h>
int main() {
  int x, d1, u1, q, u2, d2, r = 1, l = 1;
  scanf("%d%d%d%d%d%d", &x, &d1, &u1, &q, &u2, &d2);
  int ans = 0, now;
  now = 2 * x - 1;
  for (int c = 1; c <= d1 + u1; c++) {
    if (c == d1 + 1) {
      r = -1;
    } else {
      now += r;
    }
    if (c == d2 + 1) {
      l = -1;
    } else {
      now += l;
    }
    ans += now;
  }
  printf("%d", ans);
}
