#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx")
using namespace std;
inline pair<int, int> operator-(const pair<int, int> &a,
                                const pair<int, int> &b) {
  return make_pair(a.first - b.first, a.second - b.second);
}
inline long long vec(const pair<int, int> &a, const pair<int, int> &b) {
  return (long long)a.first * b.second - (long long)b.first * a.second;
}
inline long long scal(const pair<int, int> &a, const pair<int, int> &b) {
  return (long long)a.first * b.first + (long long)a.second * b.second;
}
int a, b, c, d, n, m, k;
pair<int, int> tel[8];
pair<int, int> mas[1002];
vector<int> bet[8][1002];
unordered_map<unsigned long long, bool> mem;
inline static int _popcnt(unsigned a) { return __builtin_popcount(a); }
inline static int _ctz(unsigned a) { return __builtin_ctz(a); }
bool solve(vector<int> kill, int mask);
int rc[1002];
bool rec(int cm, const vector<int> &kill, int killed, int rem,
         vector<int> &nx) {
  if (_popcnt(rem) < (int)((nx).size())) return 0;
  if (!cm) {
    if (killed != (1 << (int)((kill).size())) - 1) return 0;
    return solve(nx, rem);
  }
  int r = _popcnt(cm);
  int ck = _popcnt(killed);
  if (r + ck < (int)((kill).size())) return 0;
  int t = _ctz(cm);
  if (r + ck > (int)((kill).size())) {
    if (rec(cm ^ 1 << t, kill, killed, rem, nx)) return 1;
  }
  int old = (int)((nx).size());
  for (int _n(((int)((kill).size())) - 1), i(0); i <= _n; i++) {
    if (killed & 1 << i) continue;
    int v = kill[i];
    for (int _n(((int)((bet[t][v]).size())) - 1), j(0); j <= _n; j++) {
      int w = bet[t][v][j];
      if (!rc[w]) {
        nx.push_back(bet[t][v][j]);
      }
      ++rc[w];
    }
    bool cr = rec(cm ^ 1 << t, kill, killed | 1 << i, rem ^ 1 << t, nx);
    for (int _n(((int)((bet[t][v]).size())) - 1), j(0); j <= _n; j++) {
      int w = bet[t][v][j];
      --rc[w];
    }
    nx.resize(old);
    if (cr) return 1;
  }
  return 0;
}
bool solve(vector<int> kill, int mask) {
  if (kill.empty()) return 1;
  int rem = _popcnt(mask);
  if ((int)((kill).size()) > rem) return 0;
  sort((kill).begin(), (kill).end());
  unsigned long long h = 1;
  for (int _n(((int)((kill).size())) - 1), i(0); i <= _n; i++) {
    h = h * 1000000007 + kill[i];
  }
  h = h * 1234567891 + mask;
  auto it = mem.find(h);
  if (it != mem.end()) {
    return it->second;
  }
  vector<int> nx;
  bool ans = rec(mask, kill, 0, mask, nx);
  return mem[h] = ans;
}
int main() {
  scanf("%d%d", &k, &n);
  for (int _n((k)-1), i(0); i <= _n; i++) {
    scanf("%d%d", &tel[i].first, &tel[i].second);
  }
  for (int _n((n)-1), i(0); i <= _n; i++) {
    scanf("%d%d", &mas[i].first, &mas[i].second);
  }
  for (int _n((k)-1), i(0); i <= _n; i++) {
    for (int _n((n)-1), j(0); j <= _n; j++) {
      for (int _n((n)-1), z(0); z <= _n; z++) {
        if (z == j) continue;
        if (vec(mas[j] - tel[i], mas[z] - tel[i]) != 0) continue;
        if (scal(mas[j] - mas[z], tel[i] - mas[z]) > 0) continue;
        bet[i][j].push_back(z);
        if ((int)((bet[i][j]).size()) > k) break;
      }
    }
  }
  int ans = 0;
  memset((rc), 0, sizeof(rc));
  for (int _n((n)-1), i(0); i <= _n; i++) {
    vector<int> kill;
    kill.push_back(i);
    if (solve(kill, (1 << k) - 1)) {
      ++ans;
    }
  }
  printf("%d\n", ans);
}
