#include <bits/stdc++.h>
using namespace std;
long long p1 = 1e6 + 3, p2 = 1616161, p3 = 3959297, p4 = 7393931;
int n, m;
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> m;
  if (n == 2) {
    cout << "2 2\n";
    cout << "1 2 2";
    return 0;
  }
  long long usd = 2 + (n - 3) * 3;
  long long y = p4 - usd;
  cout << 2 << " " << p4 << "\n";
  cout << "1 " << n << " 2\n";
  cout << "1 2 " << y << "\n";
  for (int i = 2; i + 1 < n; ++i) {
    cout << i << " " << i + 1 << " 3\n";
  }
  m -= n - 1;
  if (m == 0) {
    return 0;
  }
  cout << n - 1 << " " << n << " " << y + 1 << "\n";
  --m;
  for (int i = 1; i <= n && m > 0; ++i) {
    for (int j = i + 2; j <= n && m > 0; ++j) {
      if (i == 1 && j == n) continue;
      cout << i << " " << j << " " << y + 1 << "\n";
      --m;
    }
  }
  return 0;
}
