#include<stdio.h>
#include<stdlib.h>

void solve()
{
    int n,m,k;
    scanf("%d %d %d",&n,&m,&k);

    int sum = 0;

    int diffx = n - 1;
    int diffy = m - 1;
    if(m >= n)
    {
        sum = sum + diffy*1;
        sum = sum + diffx*m; 
    }

    else if(n > m)
    {
        sum = sum + diffx*1;
        sum = sum + diffy*n;
    }
    if(sum != k)
    {
        printf("NO\n");
    }
    else
    {
        printf("YES\n");
    }
}

int main()
{
    int t;
    scanf("%d",&t);
    while(t > 0)
    {
        solve();
        t--;
    }
    return 0;
}
