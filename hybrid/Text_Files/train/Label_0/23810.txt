#include <bits/stdc++.h>
using namespace std;
const bool debug = 0;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int n, m;
  cin >> n >> m;
  vector<map<int, long long int> > g(n + 1), rg(n + 1);
  vector<long long int> ind(n + 1), oud(n + 1);
  while (m--) {
    int u, v;
    long long int d;
    cin >> u >> v >> d;
    g[u][v] += d;
    rg[v][u] += d;
  }
  set<pair<long long int, long long int> > st[2];
  for (int v = 1; v <= n; ++v) {
    for (auto x : g[v]) oud[v] += x.second;
    for (auto x : rg[v]) ind[v] += x.second;
    st[oud[v] > ind[v]].insert({v, abs(oud[v] - ind[v])});
  }
  vector<pair<int, pair<long long int, long long int> > > ans;
  while (!st[1].empty()) {
    auto pp = *st[1].begin();
    st[1].erase(pp);
    while (pp.second) {
      auto it = *st[0].begin();
      st[0].erase(it);
      while (!it.second) {
        it = *st[0].begin();
        st[0].erase(it);
      }
      long long int d = min(it.second, pp.second);
      ans.push_back({pp.first, {it.first, d}});
      pp.second -= d;
      it.second -= d;
      if (it.second) st[0].insert({it.first, it.second});
    }
  }
  cout << (int)(ans).size() << '\n';
  for (auto pp : ans)
    cout << pp.first << " " << pp.second.first << " " << pp.second.second
         << '\n';
  return 0;
}
