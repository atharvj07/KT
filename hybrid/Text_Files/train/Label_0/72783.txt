def s():
	(n, c) = [int(x) for x in input().split(' ')]
	p = [int(x) for x in input().split(' ')]
	t = [int(x) for x in input().split(' ')]
	limak_score = 0
	rad_score = 0
	limak_elapsed_time = 0
	rad_elapsed_time = 0
	for i in range(n):
		limak_score += max(0, p[i] - c * (limak_elapsed_time + t[i]))
		limak_elapsed_time += t[i]
		rad_score += max(0, p[n - 1 - i] - c * (rad_elapsed_time + t[n - 1 - i]))
		rad_elapsed_time += t[n - 1 - i]
	if limak_score > rad_score:
		print('Limak')
	elif rad_score > limak_score:
		print('Radewoosh')
	else:
		print('Tie')
s()
