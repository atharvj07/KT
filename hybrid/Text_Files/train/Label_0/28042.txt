#include <bits/stdc++.h>
using namespace std;
const long MAX_N = 1e5 + 1;
long n, m;
vector<pair<long, short> > g[MAX_N];
queue<long> q;
bool v[MAX_N];
long d[MAX_N];
long r[MAX_N];
long mn[MAX_N];
set<pair<pair<long, long>, short> > ans;
void add_edge(long a, long b, short c) {
  g[a].push_back(make_pair(b, c));
  g[b].push_back(make_pair(a, c));
  if (c == 1) {
    ans.insert(make_pair(make_pair(min(a, b), max(a, b)), 0));
  }
}
void bfs(long s) {
  v[s] = true;
  d[s] = 0;
  mn[s] = 0;
  q.push(s);
  while (!q.empty()) {
    long a = q.front();
    q.pop();
    for (size_t i = 0; i < g[a].size(); i++) {
      long b = g[a][i].first;
      short z = g[a][i].second;
      if (!v[b]) {
        mn[b] = mn[a] + (1 - z);
        d[b] = d[a] + 1;
        r[b] = a;
        v[b] = true;
        q.push(b);
      } else if (d[b] > d[a]) {
        if (mn[a] + (1 - z) < mn[b]) {
          mn[b] = mn[a] + (1 - z);
          r[b] = a;
        }
      }
    }
  }
}
void dfs(long a, long e) {
  if (a == e) return;
  long b = r[a];
  pair<pair<long, long>, short> p =
      make_pair(make_pair(min(a, b), max(a, b)), 0);
  if (ans.find(p) != ans.end()) {
    ans.erase(p);
  } else {
    p.second = 1;
    ans.insert(p);
  }
  dfs(b, e);
}
int main() {
  cin >> n >> m;
  for (long i = 0; i < m; i++) {
    long a, b;
    short c;
    cin >> a >> b >> c;
    add_edge(a, b, c);
  }
  bfs(1);
  dfs(n, 1);
  cout << ans.size() << endl;
  for (set<pair<pair<long, long>, short> >::iterator it = ans.begin();
       it != ans.end(); it++) {
    cout << it->first.first << " " << it->first.second << " " << it->second
         << endl;
  }
  return 0;
}
