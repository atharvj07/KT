import array
s = input()
p = array.array('i', [])
su = array.array('i', [])
for i in range(len(s) + 2):
	p.append(0)
	su.append(0)
i = 1
while i <= len(s):
	p[i] = p[i - 1] + int(s[i - 1])
	i += 1
i = len(s)
while i > 0:
	su[i] = su[i + 1] + int(s[i - 1])
	i -= 1
cnt = 0
l = 0
t_sum = p[len(s)]
for i in range(len(s) + 1):
	j = l
	while j < i:
		sum = t_sum - (p[j] + su[i + 1])
		if sum % 3 == 0:
			l = i
			cnt += 1
			break
		j += 1
print(cnt)
