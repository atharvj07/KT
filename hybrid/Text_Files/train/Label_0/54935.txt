import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;


public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] answer = new int[n];
        Arrays.fill(answer, 1);
        char[] marks = br.readLine().toCharArray();
        
        boolean solving = true;
        while(solving) {
            solving = false;
            
            for (int i = 0; i < marks.length; i++) {
                if (marks[i] == '=' && answer[i] != answer[i+1]) {
                    answer[i] = answer[i+1] = Math.max(answer[i], answer[i+1]);
                    solving = true;
                } else if (marks[i] == 'L' && answer[i] <= answer[i+1]) {
                    answer[i] = answer[i+1] + 1;
                    solving = true;
                } else if (marks[i] == 'R' && answer[i] >= answer[i+1]) {
                    answer[i+1] = answer[i] + 1;
                    solving = true;
                }
            }
        }
        
        for(int a : answer) {
            System.out.print(a+" ");
        }
        System.out.println("");

    }

}
