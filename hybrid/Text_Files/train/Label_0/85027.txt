def solve():
	(n, x, m) = map(int, input().split())
	r = []
	for i in range(m):
		r.append(list(map(int, input().split())))
	end = x
	start = x
	for i in range(0, m):
		if r[i][0] < start and r[i][1] < start:
			continue
		if end < r[i][0] and end < r[i][1]:
			continue
		start = min(start, r[i][0])
		end = max(end, r[i][1])
	return end - start + 1
t = int(input())
for _ in range(t):
	r1 = solve()
	print(r1)
