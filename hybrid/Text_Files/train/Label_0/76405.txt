for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	f = 1
	for i in range(n):
		if i == n - 1:
			if f == 1:
				print('First')
			else:
				print('Second')
			break
		elif a[i] == 1:
			if f == 0:
				f = 1
			else:
				f = 0
		elif a[i] > 1:
			if f == 1:
				print('First')
			else:
				print('Second')
			break
