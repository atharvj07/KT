#pragma GCC optimize("O3")
#include<bits/stdc++.h>
using namespace std;
#define SET0(a) memset(a,0,sizeof(a))
#define FOR(i,a,b) for(int i=(a);i<=(b);i++)
#define DWN(i,a,b) for(int i=(a);i>=(b);i--)
#define INF 0x3f3f3f3f
typedef long long ll;

int main(){
    int T; cin>>T;
    while(T--){
        int x,y,n; cin>>x>>y>>n;
        int l,r;
        l=1; r=1;
        while(!(x&1)){
            x>>=1;
            l<<=1;
        }
        while(!(y&1)){
            y>>=1;
            r<<=1;
        }
        if(l*r>=n) cout<<"YES"<<endl;
        else cout<<"NO"<<endl;
    }
    return 0;
}
