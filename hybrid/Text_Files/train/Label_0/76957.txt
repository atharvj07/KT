t = int(input())
while t != 0:
	n = int(input())
	A = []
	for i in range(n):
		(l, r) = map(int, input().split())
		A.append([r - l + 1, l, r])
	A.sort()
	net = [0] * n
	ans = []
	for i in A:
		(l, r) = (i[1], i[2])
		for j in range(l, r + 1):
			if net[j - 1] == 0:
				net[j - 1] = 1
				print(l, r, j)
				break
	print()
	t -= 1
