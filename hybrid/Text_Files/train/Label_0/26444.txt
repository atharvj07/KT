#include <bits/stdc++.h>
using namespace std;
int n, a[500010];
const int MAX = 500002;
class node {
 public:
  int mx, lzy;
};
class segment_tree {
 public:
  node t[4000010];
  void pushnow(int p, int v) {
    t[p].mx += v;
    t[p].lzy += v;
  }
  void pushdown(int p) {
    if (t[p].lzy) {
      pushnow((p << 1), t[p].lzy);
      pushnow(((p << 1) | 1), t[p].lzy);
      t[p].lzy = 0;
    }
  }
  void build(int p, int l, int r, bool ismin = 0) {
    if (l == r) {
      t[p].mx = l;
      return;
    }
    int mid = (l + r) >> 1;
    build((p << 1), l, mid, ismin);
    build(((p << 1) | 1), mid + 1, r, ismin);
    if (!ismin)
      t[p].mx = max(t[(p << 1)].mx, t[((p << 1) | 1)].mx);
    else
      t[p].mx = min(t[(p << 1)].mx, t[((p << 1) | 1)].mx);
  }
  void modify(int p, int l, int r, int ml, int mr, int v, bool ismin = 0) {
    if (ml <= l && r <= mr) {
      pushnow(p, v);
      return;
    }
    pushdown(p);
    int mid = (l + r) >> 1;
    if (ml <= mid) modify((p << 1), l, mid, ml, mr, v, ismin);
    if (mr > mid) modify(((p << 1) | 1), mid + 1, r, ml, mr, v, ismin);
    if (!ismin)
      t[p].mx = max(t[(p << 1)].mx, t[((p << 1) | 1)].mx);
    else
      t[p].mx = min(t[(p << 1)].mx, t[((p << 1) | 1)].mx);
  }
  int query(int p, int l, int r) {
    if (l == r) return l;
    pushdown(p);
    int mid = (l + r) >> 1;
    if (t[(p << 1)].mx >= 0) return query((p << 1), l, mid);
    return query(((p << 1) | 1), mid + 1, r);
  }
  int query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return t[p].mx;
    pushdown(p);
    int mid = (l + r) >> 1, ans = 0x3f3f3f3f;
    if (ql <= mid) ans = min(ans, query((p << 1), l, mid, ql, qr));
    if (qr > mid) ans = min(ans, query(((p << 1) | 1), mid + 1, r, ql, qr));
    return ans;
  }
} T1, T2;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &a[i]);
  }
  T1.build(1, -MAX, 0);
  T2.build(1, -MAX, MAX, 1);
  for (int i = 1; i <= n; i++) {
    if (a[i] < 0) T1.modify(1, -MAX, 0, a[i], 0, 1);
    T2.modify(1, -MAX, MAX, -MAX, a[i] - 1, 1, 1);
    int x = T1.query(1, -MAX, 0);
    printf("%d\n", T2.query(1, -MAX, MAX, x, MAX));
  }
  return 0;
}
