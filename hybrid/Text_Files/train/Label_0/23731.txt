#include<iostream>
#include<string>
#include<algorithm>
#include<vector>
using namespace std;
#define REP(i,b,n) for(int i=b;i<n;i++)
#define rep(i,n)   REP(i,0,n)
#define pb push_back

#define NEWPATH 0
#define OLDPATH 1
#define SAME    2

typedef vector<string> vecs;
const int INF = (1 << 21);
int cost[10][10];
string dp[1 << 10][10];

string solve(vecs in){
  int n = in.size();
  string inf;
  rep(i,n)inf+=in[i];
  inf = inf+inf;
  rep(i,1 << n)
    rep(j,n)dp[i][j] =inf;
  rep(i,n)dp[1 << i][i] = in[i];

  rep(i,(1 << n)){
    rep(j,n){
      if ( (i & (1 << j)) == 0)continue;
      rep(k,n){
	if ( (i & (1 << k)) != 0)continue;
	int next = i | (1 << k);
	string nextstring = dp[i][j] + in[k].substr(in[k].size() - cost[j][k]);
	if ( nextstring.size() < dp[next][k].size() ){
	  dp[next][k] = nextstring;
	}else if ( nextstring.size() == dp[next][k].size() &&
		   nextstring < dp[next][k]){	   
	  dp[next][k] = nextstring;
	}
      }
    }
  }

  string ans=inf+inf;
  int res = (1 << n) -1;
  rep(i,n){
    if ( dp[res][i].size() < ans.size() ||
	 (dp[res][i].size() == ans.size() && dp[res][i] < ans))
      ans = dp[res][i];
  }
  
  
  return ans;
}

void calc_cost(vecs in){
  int n = in.size();
  rep(i,n){
    rep(j,n){
      if ( i == j ){cost[i][j] = INF;continue;}
      cost[i][j] = in[j].size();
      rep(k,in[i].size() && k < in[j].size()){
	bool flag = true;
	rep(l,k){ 
	  if (in[i][in[i].size() - k + l] != in[j][l]){
	    flag =false;
	    break;
	  }
	}
	if ( flag == true)cost[i][j] = in[j].size() - k;
      }
    }
  }
}

vecs del(vecs in){
  vecs ret;
  bool tmp[in.size()];
  fill(tmp,tmp+in.size(),true);
  rep(i,in.size()){
    bool flag = true;
    rep(j,in.size()){
      if ( i == j)continue;
      if (tmp[j] == true && search(in[j].begin(),in[j].end(),in[i].begin(),in[i].end()) != in[j].end()){
	flag = false;
	break;
      }
    }
    if ( flag == true)ret.pb(in[i]);
    tmp[i]=flag;
  }
  return ret;
}
