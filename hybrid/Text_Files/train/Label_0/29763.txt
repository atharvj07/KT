#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string a[n];
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    bool check = 0;
    for (int j = 0; j < i; ++j) {
      if (a[i] == a[j]) {
        check = 1;
      }
    }
    if (check) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
}
