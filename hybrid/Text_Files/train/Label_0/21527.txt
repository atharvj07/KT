#include <bits/stdc++.h>
using namespace std;
int a[105];
int s[5];
int main() {
  int n, i;
  while (scanf("%d", &n) != EOF) {
    memset(s, 0, sizeof(s));
    int num = 0;
    for (i = 0; i < n; i++) {
      scanf("%d", &a[i]);
      if (a[i] == 0)
        s[0] = 1;
      else if (a[i] == 100)
        s[1] = 1;
      else if (a[i] < 10)
        s[2] = 1;
      else if (a[i] % 10 == 0)
        s[3] = 1;
      else
        s[4] = 1;
    }
    printf("%d\n", max(s[0] + s[1] + s[2] + s[3], s[0] + s[1] + s[4]));
    if (s[0]) printf("0 ");
    if (s[1]) printf("100 ");
    if (s[3] || s[2]) {
      if (s[3])
        for (i = 0; i < n; i++)
          if (a[i] % 10 == 0 && a[i] != 0 && a[i] != 100) {
            printf("%d ", a[i]);
            break;
          }
      if (s[2])
        for (i = 0; i < n; i++)
          if (a[i] && a[i] < 10) {
            printf("%d ", a[i]);
            break;
          }
    } else {
      for (i = 0; i < n; i++)
        if (a[i] != 0 && a[i] != 100) {
          printf("%d ", a[i]);
          break;
        }
    }
    printf("\n");
  }
  return 0;
}
