#include <bits/stdc++.h>
#pragma warning(disable : 4996)
using namespace std;
int n, m, s;
pair<int, int> prob[111111];
pair<pair<int, int>, int> stud[111111], tem;
bool cmp1(pair<pair<int, int>, int> a, pair<pair<int, int>, int> b) {
  if (a.first.first == b.first.first) return a.first.second < b.first.second;
  return a.first.first > b.first.first;
}
struct cmp {
  bool operator()(pair<pair<int, int>, int> a, pair<pair<int, int>, int> b) {
    if (a.first.second == b.first.second) return a.first.first < b.first.first;
    return a.first.second > b.first.second;
  }
};
priority_queue<pair<pair<int, int>, int>, vector<pair<pair<int, int>, int> >,
               cmp>
    q;
int ans[111111];
bool check(int x) {
  int i, j, k;
  long long re = 0;
  for (i = j = 0; i < m; i += x) {
    for (; j < n; ++j) {
      if (stud[j].first.first >= prob[i].first)
        q.push(stud[j]);
      else
        break;
    }
    if (!q.empty()) {
      tem = q.top(), q.pop();
      re += tem.first.second;
      for (int k = (int)(i); k <= (int)(min(i + x, m) - 1); ++k)
        ans[prob[k].second] = tem.second;
    } else
      return false;
  }
  while (!q.empty()) q.pop();
  if (re <= s)
    return true;
  else
    return false;
}
int main() {
  int i, j, k, x, y, z, ca = 1, re = 0;
  scanf("%d%d%d", &(n), &(m), &(s));
  for (int i = 0; i < (int)(m); ++i)
    scanf("%d", &(x)), prob[i] = make_pair(x, i);
  sort(prob, prob + m);
  reverse(prob, prob + m);
  for (int i = 0; i < (int)(n); ++i) scanf("%d", &(stud[i].first.first));
  for (int i = 0; i < (int)(n); ++i)
    scanf("%d", &(stud[i].first.second)), stud[i].second = i;
  sort(stud, stud + n, cmp1);
  int st = 0, en = m + 1, fl = 0;
  while (st + 1 < en) {
    int mid = (st + en) >> 1;
    if (check(mid))
      fl = 1, en = mid;
    else
      st = mid;
  }
  if (fl) {
    puts("YES");
    check(en);
    for (int i = 0; i < (int)(m); ++i) printf("%d ", ans[i] + 1);
  } else
    puts("NO");
  return 0;
}
