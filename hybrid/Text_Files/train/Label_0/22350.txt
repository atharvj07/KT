#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e6 + 5;
int scnt, ecnt, che[maxn], a[maxn], n, k;
long long stot, etot, ans;
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", a + i);
    if (a[i] <= i)
      ++scnt, stot += (i - a[i]);
    else
      ++ecnt, etot += (a[i] - i), ++che[a[i] - i];
  }
  ans = stot + etot;
  k = 0;
  for (int i = 1; i < n; ++i) {
    etot -= ecnt;
    ecnt -= che[i];
    stot += scnt;
    scnt += che[i];
    stot -= n + 1 - a[n - i + 1];
    --scnt;
    if (a[n - i + 1] > 1)
      ++che[a[n - i + 1] - 1 + i], etot += a[n - i + 1] - 1, ++ecnt;
    else
      ++scnt;
    if (ans > stot + etot) ans = stot + etot, k = i;
  }
  printf("%lld %d\n", ans, k);
  return 0;
}
