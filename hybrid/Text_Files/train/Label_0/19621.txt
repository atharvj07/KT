#include <bits/stdc++.h>
using namespace std;
const int maxn = 5005;
string s;
int q, n, i, j, a, b, l;
int dp[maxn][maxn], fix[maxn][maxn], A[maxn];
int main() {
  cin >> s;
  n = s.size();
  for (i = 1; i <= n; i++) fix[i][i] = 1;
  for (i = 1; i <= n - 1; i++) fix[i][i + 1] = (s[i] == s[i - 1]);
  for (l = 2; l <= n - 1; l++)
    for (j = 1; j + l <= n; j++) {
      a = j;
      b = j + l;
      fix[a][b] = (fix[a + 1][b - 1] && s[a - 1] == s[b - 1]);
    }
  for (i = 1; i <= n; i++) dp[i][i] = 1;
  for (i = 1; i <= n - 1; i++) dp[i][i + 1] = 2 + fix[i][i + 1];
  for (i = 1; i <= n - 1; i++) A[i] = fix[i][i + 1] + fix[i][i];
  for (l = 2; l <= n - 1; l++) {
    for (j = 1; j + l <= n; j++) A[j] = A[j] + fix[j][j + l];
    for (j = 1; j + l <= n; j++) {
      a = j;
      b = j + l;
      dp[a][b] = dp[a + 1][b] + A[a];
    }
  }
  cin >> q;
  while (q--) {
    cin >> a >> b;
    cout << dp[a][b] << endl;
  }
}
