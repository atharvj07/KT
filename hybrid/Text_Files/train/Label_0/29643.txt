	while (isdigit(c)) {x=x*10+c-'0'; c=getchar();} x*=f;
}
template<class T> inline void umin(T &x, T y){x=x<y?x:y;}

inline ui R() {
	static ui seed=416;
	return seed^=seed>>5,seed^=seed<<17,seed^=seed>>13;
}
const int N = 1<<18|3;
int n,all,a[N],l[N],r[N],id[N];bool vis[N];
void GG(){puts("NO");exit(0);}
bool cmp(int x, int y){return l[x]<l[y];}
priority_queue<Pii,vector<Pii>,greater<Pii> >heap;
int main() {
	read(n);all=(1<<n)-1;
	rep(i,0,all){
		read(a[i]),a[i]--;
		if(a[i]>=0)l[i]=r[i]=a[i];
		else l[i]=0,r[i]=all;
	}
	rep(s,0,all)rep(i,0,n-1)if(!(s>>i&1))umax(l[s^(1<<i)],l[s]+1);
	per(s,all,0)rep(i,0,n-1)if(s>>i&1)umin(r[s^(1<<i)],r[s]-1);
	rep(s,0,all)if(l[s]>r[s])GG();
	rep(s,0,all)id[s]=s;sort(id,id+all+1,cmp);
//	rep(s,0,all)printf("%d:%d %d\n",s,l[s],r[s]);
	for(int s=0,p=-1;s<=all;s++){
		while(p+1<=all&&l[id[p+1]]<=s)p++,heap.push(Pii(r[id[p]],id[p]));
	//	while(!heap.empty()&&vis[heap.top().se])heap.pop();
		if(heap.empty())GG();int u=heap.top().se;heap.pop();
		if(s<l[u]||s>r[u])GG();a[u]=s;//vis[u]=1;
	}
//	rep(s,0,all)printf("%d:%d %d %d\n",s,l[s],r[s],a[s]);
	for(int i=1,j=0;i<=all;i++){
		for(int k=1<<(n-1);!((j^=k)&k);k>>=1);
	//	printf("%d %d\n",i,j);
		if(i<j)swap(a[i],a[j]);
	}
	printf("YES\n");rep(i,0,all)printf("%d ",1+a[i]);
	return 0;
}
