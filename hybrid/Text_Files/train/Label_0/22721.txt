#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
char s1[N], s2[N];
int n1, n2, f[N];
long long dp[N], ans;
void getFail() {
  f[0] = f[1] = 0;
  for (int i = 1; i < n2; i++) {
    int j = f[i];
    while (j && s2[i] != s2[j]) j = f[j];
    f[i + 1] = s2[i] == s2[j] ? j + 1 : 0;
  }
}
void find() {
  int j = 0;
  for (int i = 0; i < n1; i++) {
    while (j && s2[j] != s1[i]) j = f[j];
    if (s2[j] == s1[i]) j++;
    if (j) {
      if (i != j - 1)
        ans += dp[j];
      else
        ans += dp[f[j]];
    }
    if (j == n2) j = f[j];
  }
}
inline void work() {
  scanf("%s%s", s1, s2);
  n1 = strlen(s1);
  n2 = strlen(s2);
  getFail();
  for (int i = 1; i <= n2; i++) dp[(i - f[i])] += f[i] != 0;
  for (int i = 1; i <= n2; i++) dp[i] += dp[f[i]];
  find();
  printf("%lld\n", 1LL * n1 * n2 - ans);
}
int main() {
  work();
  return 0;
}
