
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    
    public static void main(String[]args){
        
        Scanner in = new Scanner(System.in);
    
        int n = in.nextInt();
        int d = in.nextInt();
        
        int [] h = new int [n];
        
        for(int i=0 ; i<n ; i++)
            h[i] = in.nextInt();
        
        Arrays.sort(h);
        
        int cnt=0;
        for(int i=0 ; i<n ; i++)
            for(int j=0 ; j<n ; j++)
            if(Math.abs(h[j] - h[i]) <= d && j != i)
                cnt++;
        
        System.out.println(cnt);
     }
   }
