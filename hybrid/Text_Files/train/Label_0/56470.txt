import java.util.*;
import java.io.*;

public class Solution{

	static int mod=1_000_000_007, oo=Integer.MAX_VALUE, _oo=Integer.MAX_VALUE;



	public static void main(String args[]) throws IOException {
 
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));	Scanner in=new Scanner(System.in); StringBuilder out=new StringBuilder();
		
		int n=in.nextInt();

		char s[]=in.next().toCharArray();

		int A=-1, B=-1;

		long count=0;

		for(int i=0; i<n; i++) {

			if(s[i]=='A') {
				count+=solve(A, i, B);
				A=i;
			}
			else {
				count+=solve(B, i, A);
				B=i;
			}
		}

		System.out.println(count);
	}

	public static int solve(int X, int i, int Y) {

		if(X==-1)
			return 0;

		if(i-X!=1 || Y==-1)
			return X+1;

		return X;
	}
}
