import java.io.*; import java.util.*;
public class CF1264D2 {

    static long MOD=998244353;
    public static void main(String[] args) throws IOException {
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        String str=br.readLine(); int len=str.length();
        if(len==1){
            System.out.println(0);return;
        }
        int[] left=new int[len];//Prefix
        int[] right=new int[len];
        int[] question=new int[len];
        if(str.charAt(0)=='('){
             left[0]=1;
        }else if(str.charAt(0)==')'){
            right[0]=1;
        }else{
            question[0]=1;
        }
        for (int i = 1; i < len; i++) {
            left[i]=left[i-1]; right[i]=right[i-1]; question[i]=question[i-1];
            if(str.charAt(i)=='('){
                left[i]++;
            }else if(str.charAt(i)==')'){
                right[i]++;
            }else{
                question[i]++;
            }
        }
        
        if(question[len-1]==0){
            int ans=0;
            int rp=len-1;
            for (int i = 0; i < len; i++) {
                if(str.charAt(i)=='('){
                    while(rp>i && str.charAt(rp)=='('){
                        rp--;
                    }
                    if(rp<=i){
                        System.out.println(ans);return;
                    }
                    rp--; ans++;
                    
                }
            }
            System.out.println(0);return;
        }
        int k=question[len-1];
        long[] prefixk1=new long [k];//Computes prefix for binomial coefficients with top k-1
        long cur=1; prefixk1[0]=1;
        for (int i = 1; i <k; i++) {
            cur*=(k-i); cur%=MOD; cur*=inv(i,MOD); cur%=MOD;
            prefixk1[i]=prefixk1[i-1]+cur; prefixk1[i]%=MOD;
        }
        
        long[] prefixk=new long[k+1];
        prefixk[0]=1; prefixk[k]=(2*prefixk1[k-1])%MOD;
        for (int i = 1; i < k; i++) {
            prefixk[i]=(prefixk1[i-1]+prefixk1[i])%MOD;//Pascal identity
        }
        int a=0;  int y=0;
        int b=right[len-1]-right[0];
        int z=question[len-1]-question[0];
        
        long ans=0;
        if(str.charAt(0)=='('){
               //a=0, y=0
               ans+=prefixk[Math.min(b-a+z-1,k)];
         }else if(str.charAt(0)=='?'){
             
                ans+=prefixk1[Math.min(b-a+z-1,k-1)];
         }
        ans%=MOD;
        //System.out.println(a+" "+b+" "+y+" "+z);
        //System.out.println(ans);
        for (int i = 1; i < len; i++) {
            a=left[i-1];
            y=question[i-1];
            b=right[len-1]-right[i];
            z=question[len-1]-question[i];
            //System.out.println(a+" "+b+" "+y+" "+z);
            if(b-a+z>0){
                if(str.charAt(i)=='('){
                    ans+=prefixk[Math.min(b-a+z-1,k)];
                }else if(str.charAt(i)=='?'){
                    ans+=prefixk1[Math.min(b-a+z-1,k-1)];
                }
            }
            ans%=MOD;
            //System.out.println(ans);
        }
        System.out.println(ans);
    }
    public static long inv(long a, long b){//Computes modular inverse of a mod b
        //Assume b>a and gcd(a,b)=1, 
        if(a==1)return 1;
        long q=b/a;
        long r=b-q*a;
        long y=-inv(r,a);
        return ((b*y+1)/a)+b;
        
    }
}
