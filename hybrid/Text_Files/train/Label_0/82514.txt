import sys, os, io
I = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
for tc in range(1, 1 + int(I())):
	n = int(I())
	s = I().rstrip().decode('utf-8')
	if s.find('<') < 0 or s.find('>') < 0:
		print(n)
	else:
		r = [0] * n
		for i in range(n):
			if s[i] == '-':
				r[i] = 1
				r[(i + 1) % n] = 1
		print(sum(r))
