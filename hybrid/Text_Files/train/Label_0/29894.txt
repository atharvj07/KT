#include <bits/stdc++.h>
#define fastio      ios_base::sync_with_stdio(false); cin.tie(NULL)
#define debug(x)    cout << "Line " << __LINE__ << ", " << #x << " is " << x << endl
#define fi          first
#define se          second
#define mp          make_pair
#define pb          push_back
#define ll          long long
#define ull         unsigned long long
#define ld          long double
#define pii         pair<int, int>
#define pl          pair<ll, ll>
using namespace std;
ll X, Y, A, B, le, ri, mi1, mi2, ans=0;
ld skor(ll x){
    return x+min((X-A*x)*1.0/B, (Y-B*x)*1.0/A);
}
int main(){
    fastio;
    int tc;
    cin >> tc;
    while(tc--){
        cin >> X >> Y >> A >> B;
        le=0, ri=min(X/A, Y/B);
        ans=0;
        while(ri-le>2){
            mi1=(le+ri+le)/3, mi2=(le+ri+ri)/3;
            if(skor(mi1)<=skor(mi2))
                le=mi1;
            else
                ri=mi2;
        }
        //cout << "ans";
        for(int i=le; i<=ri; i++)
            ans=max(ans, (ll)skor(i));
        cout << ans << endl;
    }
}
