#include <bits/stdc++.h>
using namespace std;
long long INF = 1e9 + 7;
const int N = 1e5 + 5;
int e_c[N];
long long edges[N];
long long sum = 0;
int dfs(vector<vector<pair<int, int>>> &graph, vector<bool> &isVisited, int n,
        int src) {
  isVisited[src] = true;
  int cnt = 0;
  for (int i = 0; i < graph[src].size(); i++) {
    if (!isVisited[graph[src][i].first]) {
      int id = graph[src][i].second;
      int val = dfs(graph, isVisited, n, graph[src][i].first);
      long long w = edges[id];
      e_c[id] = val;
      cnt += val;
    }
  }
  return max(cnt, 1);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int tests = 1;
  cin >> tests;
  while (tests--) {
    int n;
    cin >> n;
    long long k = 0;
    cin >> k;
    vector<vector<pair<int, int>>> graph(n);
    for (int i = 0; i < n - 1; i++) {
      int a, b, w;
      cin >> a >> b >> w;
      edges[i] = w;
      graph[a - 1].push_back({b - 1, i});
      graph[b - 1].push_back({a - 1, i});
    }
    sum = 0;
    vector<bool> isVisited(n, false);
    int cnt = dfs(graph, isVisited, n, 0);
    priority_queue<pair<long long, int>> maxHeap;
    for (int i = 0; i < n - 1; i++) {
      long long diff = (edges[i] - (edges[i] / 2)) * e_c[i];
      sum += (edges[i]) * e_c[i];
      maxHeap.push({diff, i});
    }
    int steps = 0;
    while (sum > k) {
      int id = maxHeap.top().second;
      sum -= maxHeap.top().first;
      maxHeap.pop();
      steps++;
      edges[id] = edges[id] / 2;
      if (edges[id] > 0) {
        long long diff = (edges[id] - (edges[id] / 2)) * e_c[id];
        maxHeap.push({diff, id});
      }
    }
    cout << steps << endl;
  }
  return 0;
}
