#include <bits/stdc++.h>
using namespace std;
const int N = 4e5 + 6;
int n, trie[2][N][26], cnt, fail[N];
long long ans = 0, f[2][N], num[2][N];
char t[N], s[N];
void insert(int o) {
  int now = 0;
  int len = strlen(s + 1);
  for (int i = 1; i <= len; i++) {
    int c = s[i] - 'a';
    if (!trie[o][now][c]) trie[o][now][c] = ++cnt;
    now = trie[o][now][c];
  }
  num[o][now]++;
}
void build(int o) {
  queue<int> q;
  for (int i = 0; i < 26; i++)
    if (trie[o][0][i]) q.push(trie[o][0][i]);
  while (!q.empty()) {
    int now = q.front();
    q.pop();
    for (int i = 0; i < 26; i++)
      if (trie[o][now][i]) {
        fail[trie[o][now][i]] = trie[o][fail[now]][i];
        q.push(trie[o][now][i]);
        num[o][trie[o][now][i]] += num[o][fail[trie[o][now][i]]];
      } else
        trie[o][now][i] = trie[o][fail[now]][i];
  }
}
void query(int o) {
  int len = strlen(t + 1), now = 0;
  for (int i = 1, c; i <= len; i++) {
    c = t[i] - 'a';
    now = trie[o][now][c];
    if (now) f[o][i] = num[o][now];
  }
}
int main() {
  scanf("%s %d", t + 1, &n);
  while (n--) {
    scanf("%s", s + 1);
    int len = strlen(s + 1);
    insert(0);
    reverse(s + 1, s + 1 + len);
    insert(1);
  }
  build(0);
  build(1);
  query(0);
  int len = strlen(t + 1);
  reverse(t + 1, t + 1 + len);
  query(1);
  for (int i = 1; i < len; i++) {
    ans += f[0][i] * f[1][len - i];
  }
  cout << ans;
  return 0;
}
