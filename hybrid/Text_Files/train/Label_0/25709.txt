#include <bits/stdc++.h>
using namespace std;
int main() {
  long long int n;
  cin >> n;
  long long int arr[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  string s;
  cin >> s;
  vector<long long int> store;
  set<char> st;
  set<char>::iterator itr;
  long long int lmin = -pow(10, 9), lmax = pow(10, 9), rmin = -pow(10, 9),
                rmax = pow(10, 9), l, r;
  for (int i = 4; i < n; i++) {
    st.clear();
    store.clear();
    for (int j = i - 4; j <= i - 1; j++) {
      st.insert(s[j]);
    }
    if (st.size() == 2) {
      continue;
    } else {
      for (int j = i - 4; j <= i; j++) {
        store.push_back(arr[j]);
      }
      itr = st.begin();
      if (*itr == '0') {
        if (s[i] == '1') {
          sort(store.begin(), store.end());
          l = store[4] + 1;
          lmin = max(l, lmin);
        } else if (s[i] == '0') {
          sort(store.begin(), store.end());
          l = store[4];
          lmax = min(l, lmax);
        }
      } else if (*itr == '1') {
        if (s[i] == '0') {
          sort(store.begin(), store.end());
          r = store[0] - 1;
          rmax = min(r, rmax);
        } else if (s[i] == '1') {
          sort(store.begin(), store.end());
          r = store[0];
          rmin = max(r, rmin);
        }
      }
    }
  }
  cout << lmin << " " << rmax << endl;
}
