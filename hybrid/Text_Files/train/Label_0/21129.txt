#include <bits/stdc++.h>
using namespace std;

int a[50], b[50], c[50], d[50];
int n, m, q;
int ans = 0;
int arr[10];

void f(int idx, int low) {
  if (idx == n) {
    int value = 0;
    for (int i=0; i<q; i++) {
      if (arr[b[i]] - arr[a[i]] == c[i]) {
        value += d[i];
      }
    }
    ans = max(value, ans);
    return;
  }
  for (int i=low; i<=m; i++) {
    arr[idx] = i;
    f(idx+1, i);
  }
}
int main() {
  cin>>n>>m>>q;
  for (int i=0; i<q; i++) {
    cin>>a[i]>>b[i]>>c[i]>>d[i];
    --a[i];
    --b[i];
  }
  f(0, 1);
  cout<<ans<<endl;
}
