def fastpow(base, exp, mod):
	res = 1
	while exp > 0:
		if exp % 2 == 1:
			res = res * base % mod
		base = base * base % mod
		exp //= 2
	return res % mod
(a, b, c) = input().split()
a = int(a)
b = int(b)
c = int(c)
ans = fastpow(2, (a - 1) * (b - 1), 1000000007)
if (a + b) % 2 == 1:
	if c == -1:
		ans = 0
print(ans)
