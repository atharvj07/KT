import sys
(n, m, k) = map(int, input().split())
p = []
for _ in range(m):
	(u, v, l) = map(int, input().split())
	p.append((u, v, l))
if k != 0:
	d = {int(x): 1 for x in input().split()}
	mini = sys.maxsize
	for i in range(m):
		(u, v, l) = p[i]
		if d.get(u) and (not d.get(v)) or (not d.get(u) and d.get(v)):
			if l < mini:
				mini = l
	if mini != sys.maxsize:
		print(mini)
	else:
		print('-1')
else:
	print('-1')
