def rotate(sliced):
	rotated = [sliced[len(sliced) - 1]]
	for i in range(len(sliced) - 1):
		rotated.append(sliced[i])
	return rotated
t = int(input())
for i in range(t):
	ans = []
	shifts = []
	n = int(input())
	a = list(map(int, input().split()))
	for i in range(len(a)):
		for j in range(len(a) - 1):
			end = 0
			start = 0
			if a[j + 1] < a[j]:
				end = j + 1
				for x in range(end - 1, -1, -1):
					if a[x] >= a[end]:
						start = x
					else:
						break
			if end > start:
				ans.append([start + 1, end + 1])
				a[start:end + 1] = rotate(a[start:end + 1])
				shifts.append(len(a[start:end + 1]) - 1)
				break
	print(len(ans))
	for p in range(len(ans)):
		ans[p].append(shifts[p])
		for z in range(len(ans[p])):
			print(ans[p][z], end=' ')
		print('')
