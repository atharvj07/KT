#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 4;
const int INF = 4e5 + 78;
struct data {
  int first;
  int edge_used;
  int parent;
  data() {
    first = 0;
    edge_used = 0;
    parent = 0;
  }
};
data nodes[N];
vector<pair<int, int>> graph[N];
bool visited[N] = {0};
pair<int, int> edges[N];
int dist[N];
bool edge[N];
inline int scan() {
  int x;
  scanf("%d", &x);
  return x;
}
int main(void) {
  int n = scan(), m = scan();
  for (int i = 1; i <= m; ++i) {
    int x = scan(), y = scan(), z = scan();
    graph[x].push_back(make_pair(y, i));
    graph[y].push_back(make_pair(x, i));
    edge[i] = z;
    edges[i] = make_pair(x, y);
  }
  memset(dist, INF, sizeof(dist));
  queue<int> q;
  bool aslivested[N] = {0};
  q.push(1);
  visited[1] = 1;
  dist[1] = 0;
  while (!q.empty()) {
    int node = q.front();
    q.pop();
    aslivested[node] = 1;
    int x = graph[node].size();
    int num = nodes[node].first;
    for (int i = 0; i < x; ++i) {
      int next = graph[node][i].first;
      if (aslivested[next]) continue;
      int num2 = num;
      if (edge[graph[node][i].second] == 0) {
        num2 = num2 + 1;
      }
      if (!visited[next]) {
        q.push(next);
        dist[next] = dist[node] + 1;
        nodes[next].edge_used = graph[node][i].second;
        nodes[next].parent = node;
        nodes[next].first = num2;
        visited[next] = 1;
      } else if (dist[next] >= dist[node] + 1) {
        if (num2 < nodes[next].first) {
          nodes[next].first = num2;
          nodes[next].edge_used = graph[node][i].second;
          nodes[next].parent = node;
        }
      }
    }
  }
  bool used[N] = {0};
  int cur = n;
  while (cur) {
    used[nodes[cur].edge_used] = 1;
    cur = nodes[cur].parent;
  }
  int counter = 0;
  queue<pair<pair<int, int>, bool>> ans;
  for (int i = 1; i <= m; i++) {
    if (!used[i] && edge[i] == 1) {
      counter++;
      ans.push(make_pair(edges[i], 0));
    } else if (used[i] && edge[i] == 0) {
      counter++;
      ans.push(make_pair(edges[i], 1));
    }
  }
  printf("%d\n", counter);
  while (!ans.empty()) {
    printf("%d %d %d\n", ans.front().first.first, ans.front().first.second,
           ans.front().second);
    ans.pop();
  }
}
