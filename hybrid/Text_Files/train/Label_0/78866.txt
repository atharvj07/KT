n = int(input())
l = list(map(int, input().split()))
j = max(l)
p = l.index(j)
p1 = p - 1
p2 = p + 1
flag = 0
for x in range(p1, 0, -1):
	q = l[x]
	w = l[x - 1]
	if q > w:
		pass
	else:
		flag = 1
		break
for x in range(p2, n - 1):
	q = l[x]
	w = l[x + 1]
	if q > w:
		pass
	else:
		flag = 1
		break
if flag == 1:
	print('NO')
else:
	print('YES')
