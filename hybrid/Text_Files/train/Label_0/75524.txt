def flip(bit):
	if bit == 0:
		return 1
	else:
		return 0
(n, m) = map(int, input().split())
A = []
B = []
for i in range(n):
	A.append(list(map(int, input().split())))
for i in range(n):
	B.append(list(map(int, input().split())))
for i in range(n - 1):
	for j in range(m - 1):
		if A[i][j] != B[i][j]:
			A[i][j] = flip(A[i][j])
			A[i + 1][j] = flip(A[i + 1][j])
			A[i][j + 1] = flip(A[i][j + 1])
			A[i + 1][j + 1] = flip(A[i + 1][j + 1])
flag = 0
for i in range(n):
	for j in range(m):
		if A[i][j] != B[i][j]:
			print('No')
			flag = 1
			break
	if flag == 1:
		break
else:
	print('Yes')
