from queue import PriorityQueue
cas = int(input())
while cas:
	cas -= 1
	a = input()
	cnt = [0 for i in range(26)]
	vis = [0 for i in range(26)]
	for i in range(len(a)):
		c = ord(a[i]) - ord('a')
		cnt[c] += 1
	st = [30]
	s = 0
	for i in range(len(a)):
		tmp = ord(a[i]) - ord('a')
		if vis[tmp] == 0:
			while s and st[s] < tmp and (cnt[st[s]] >= 1):
				vis[st[s]] = 0
				st.pop(s)
				s -= 1
			st.append(tmp)
			s += 1
			vis[tmp] = 1
		cnt[tmp] -= 1
	for i in range(1, len(st)):
		c = chr(ord('a') + st[i])
		if i == len(st) - 1:
			print(c)
		else:
			print(c, end='')
