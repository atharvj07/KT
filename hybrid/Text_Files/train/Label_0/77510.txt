(n, k) = map(int, input().split())
s = input()
ss = ''
sss = [ord(i) for i in s]
for i in range(0, n):
	k1 = 122 - sss[i]
	k2 = sss[i] - 97
	if k1 >= k2:
		go = sss[i] + min(k1, k)
		k -= min(k1, k)
		ss += chr(go)
	else:
		go = sss[i] - min(k2, k)
		k -= min(k2, k)
		ss += chr(go)
print(-1 if k else ss)
