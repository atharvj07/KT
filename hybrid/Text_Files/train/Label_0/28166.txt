#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
const int maxn = 222222;
int n;
int d[maxn];
int BIT[maxn];
void ins(int x, int v) {
  while (x < maxn) {
    BIT[x] += v;
    x += x & -x;
  }
}
int get(int x) {
  int ret = 0;
  while (x > 0) {
    ret += BIT[x];
    x -= x & -x;
  }
  return ret;
}
int main(int argc, char const *argv[]) {
  scanf("%d", &(n));
  for (int(i) = 0; i < (int)(n); ++(i)) {
    int x;
    scanf("%d", &(x));
    int ll = x - get(x + 1);
    d[i] += ll;
    ins(x + 1, 1);
  }
  memset(BIT, 0, sizeof(BIT));
  for (int(i) = 0; i < (int)(n); ++(i)) {
    int x;
    scanf("%d", &(x));
    int ll = x - get(x + 1);
    d[i] += ll;
    ins(x + 1, 1);
  }
  for (int i = n - 1; i > 0; --i) {
    d[i - 1] += d[i] / (n - i);
    d[i] = d[i] % (n - i);
  }
  d[0] = d[0] % n;
  memset(BIT, 0, sizeof(BIT));
  for (int(i) = 0; i < (int)(n); ++(i)) ins(i + 1, 1);
  for (int i = 0; i < n; ++i) {
    int t = d[i] + 1;
    int L = 0, R = n - 1;
    while (L < R) {
      int M = (R + L) / 2;
      int r = get(M + 1);
      if (r < t)
        L = M + 1;
      else
        R = M;
    }
    printf("%d ", L);
    ins(L + 1, -1);
  }
  return 0;
}
