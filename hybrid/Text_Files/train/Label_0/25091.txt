#include <bits/stdc++.h>
using namespace std;
vector<int> v[200010];
vector<int> cnt;
int e[200010];
int vis[200010];
void dfs(int x) {
  vis[x] = 1;
  cnt.emplace_back(x);
  for (auto it : v[x])
    if (vis[it] == 0) dfs(it);
}
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= m; i++) {
    int x, y;
    scanf("%d%d", &x, &y);
    v[x].emplace_back(y);
    v[y].emplace_back(x);
    e[x]++;
    e[y]++;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (vis[i] == 0) {
      cnt.clear();
      dfs(i);
      int ok = 1;
      for (auto it : cnt) {
        if (e[it] != 2) {
          ok = 0;
          break;
        }
      }
      if (ok) ans++;
    }
  }
  printf("%d\n", ans);
  return 0;
}
