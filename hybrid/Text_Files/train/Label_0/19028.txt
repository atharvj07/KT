#include <bits/stdc++.h>
using namespace std;
vector<string> split(const string& s, char c) {
  vector<string> v;
  stringstream ss(s);
  string x;
  while (getline(ss, x, c)) v.emplace_back(x);
  return move(v);
}
void err(vector<string>::iterator it) {}
template <typename T, typename... Args>
void err(vector<string>::iterator it, T a, Args... args) {
  cerr << it->substr((*it)[0] == ' ', it->length()) << " = " << a << '\n';
  err(++it, args...);
}
static constexpr int inf = std::numeric_limits<int>::max();
const int M = 500010;
int n, m;
struct segment_tree {
  vector<pair<int, int> > T;
  void resize(int n) { T.resize(n + 1); }
  pair<int, int> find_max(pair<int, int> a, pair<int, int> b) {
    return a.first >= b.first ? a : b;
  }
  void build(int* A, int i, int l, int r) {
    if (l == r)
      T[i] = make_pair(A[l], l);
    else {
      int m = (l + r) / 2;
      build(A, 2 * i, l, m);
      build(A, 2 * i + 1, m + 1, r);
      T[i] = find_max(T[2 * i], T[2 * i + 1]);
    }
  }
  void update(int i, int l, int r, int x, int val) {
    if (l == r && r == x)
      T[i] = make_pair(val, l);
    else if (x >= l && x <= r) {
      int m = (l + r) / 2;
      update(2 * i, l, m, x, val);
      update(2 * i + 1, m + 1, r, x, val);
      T[i] = find_max(T[2 * i], T[2 * i + 1]);
    }
  }
  pair<int, int> query(int i, int l, int r, int st, int ed) {
    if (l >= st && r <= ed)
      return T[i];
    else if (l > ed || r < st)
      return make_pair(-inf, -1);
    else {
      int m = (l + r) / 2;
      return find_max(query(2 * i, l, m, st, ed),
                      query(2 * i + 1, m + 1, r, st, ed));
    }
  }
};
segment_tree T[6];
int find_sum(int l, int r) {
  int sum = 0;
  for (int i = 0; i < m; i++) sum += T[i].query(1, 0, n - 1, l, r).first;
  return sum;
}
int main() {
  int i, j, k, t, maxx = 0, l, h, mid;
  scanf("%d%d%d", &n, &m, &k);
  for (i = 0; i < m; i++) T[i].resize(M);
  for (i = 0; i < n; i++) {
    for (j = 0; j < m; j++) {
      scanf("%d", &t);
      T[j].update(1, 0, n - 1, i, t);
    }
  }
  pair<int, int> ans;
  for (i = 0; i < n; i++) {
    if (find_sum(i, i) > k)
      continue;
    else {
      l = i;
      h = n - 1;
      while (l < h) {
        mid = (l + h + 1) / 2;
        if (find_sum(i, mid) <= k)
          l = mid;
        else
          h = mid - 1;
      }
      if (l - i + 1 > maxx) {
        maxx = l - i + 1;
        ans = make_pair(i, l);
      }
    }
  }
  if (maxx == 0)
    for (i = 0; i < m; i++) printf("%d ", 0);
  else
    for (i = 0; i < m; i++)
      printf("%d ", T[i].query(1, 0, n - 1, ans.first, ans.second).first);
  return 0;
}
