#include <bits/stdc++.h>
using namespace std;
void solve() {
  std::ios_base::sync_with_stdio(false);
  int n;
  cin >> n;
  vector<long long> vec(n / 2 + 1);
  for (int i = 0; i < n / 2; i++) {
    long long temp;
    cin >> temp;
    vec[i + 1] = temp;
  }
  vector<long long> ans(n + 1);
  ans[1] = 0;
  ans[n] = vec[1];
  for (int i = 2; i <= n / 2; i++) {
    long long value = vec[i];
    long long dif = value - ans[i - 1] - ans[n - i + 2];
    if (dif >= 0) {
      ans[i] = ans[i - 1] + dif;
      ans[n - i + 1] = ans[n - i + 2];
    } else {
      ans[i] = ans[i - 1];
      ans[n - i + 1] = ans[n - i + 2] + dif;
    }
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
}
int main() {
  solve();
  cin.get();
}
