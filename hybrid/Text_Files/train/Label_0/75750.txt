from sys import stdin, stdout

def INPUT():
	return list((int(i) for i in stdin.readline().split()))

def inp():
	return stdin.readline()

def out(x):
	return stdout.write(x)
import math
import random
import sys

def calc(n):
	ans = 0
	i = 1
	while n:
		m = 0
		if n % 2 == 0 and n // 2 % 2 == 1 or n == 4:
			n //= 2
			m = n
		else:
			n -= 1
			m = 1
		if i:
			ans += m
		i ^= 1
	return ans
for _ in range(int(input())):
	n = int(inp())
	t = calc(n)
	out(str(t) + '\n')
