#include <bits/stdc++.h>
int f[100201][18][16], n, m, k, cc[100201];
int main() {
  scanf("%d%d%d", &n, &k, &m);
  for (register int i = 1; i < 100201; ++i) cc[i] = cc[i >> 1] + (i & 1);
  f[0][0][0] = 1;
  for (register int i = 1; i <= n; ++i)
    for (register int j = 0; j < (1 << m); ++j)
      for (register int a = 0; a <= k; ++a) {
        f[i][j >> 1][a] = (f[i][j >> 1][a] + f[i - 1][j][a]) % 1000000007;
        f[i][(j >> 1) | (1 << m - 1)][a + 1] =
            (f[i][(j >> 1) | (1 << m - 1)][a + 1] +
             1LL * f[i - 1][j][a] * (cc[j] + 1)) %
            1000000007;
      }
  register int ans = 0;
  for (register int j = 0; j < (1 << m); ++j)
    ans = (ans + f[n][j][k]) % 1000000007;
  std::cout << ans;
}
