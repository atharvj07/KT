for tt in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	ans = 0
	for i in range(n):
		if sum(l) % (n - i) == 0:
			curr = 0
			x = sum(l) / (n - i)
			for j in range(n):
				curr += l[j]
				if curr == x:
					curr = 0
				elif curr > x:
					break
			if curr == 0:
				ans = i
				break
			else:
				continue
	print(ans)
