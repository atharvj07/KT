(A, B, C, D) = map(int, input().split())
ans = 0
howmany = [0 for i in range(10 ** 6 + 1)]
cnt = 0
for i in range(A + B + 1):
	if A <= i <= B and B <= A + B - i <= C:
		cnt += 1
howmany[A + B] = cnt
for z in range(A + B + 1, B + C + 1):
	if z <= min(A + C, 2 * B):
		cnt += 1
	elif min(A + C, 2 * B) < z <= max(A + C, 2 * B):
		pass
	elif max(A + C, 2 * B) < z:
		cnt -= 1
	howmany[z] = cnt
for i in range(1, 10 ** 6 + 1):
	howmany[i] += howmany[i - 1]
for i in range(C, D + 1):
	ans += howmany[10 ** 6] - howmany[i]
print(ans)
