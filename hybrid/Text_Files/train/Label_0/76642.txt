t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	k = a[0]
	flag = 0
	for i in range(n):
		if a[i] != k:
			flag = 1
			break
	if flag == 0:
		print('NO')
		continue
	print('YES')
	dic = {}
	for i in range(n):
		if a[i] in dic:
			dic[a[i]].append(i + 1)
		else:
			dic[a[i]] = [i + 1]
	listofgangs = []
	for i in dic:
		listofgangs.append(i)
	for i in range(len(listofgangs) - 1):
		l = dic[listofgangs[i + 1]][0]
		for j in dic[listofgangs[i]]:
			print(j, l)
	l = dic[listofgangs[0]][0]
	for j in range(1, len(dic[listofgangs[-1]])):
		print(dic[listofgangs[-1]][j], l)
