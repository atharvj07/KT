ar = list(map(int, input()))
ar1 = []
ar2 = []
k = 0
MOD = 10 ** 9 + 7
bebe = 1
for i in range(len(ar)):
	if ar[i] % 5 == 0:
		k += bebe
		k %= MOD
	bebe *= 2
	bebe %= MOD
day = int(input())
num1 = pow(2, len(ar) * day, MOD) - 1
if num1 == -1:
	num1 = MOD - 1
num2 = pow(2, len(ar), MOD) - 1
if num2 == -1:
	num2 = MOD - 1
print(num1 * k % MOD * pow(num2, MOD - 2, MOD) % MOD)
