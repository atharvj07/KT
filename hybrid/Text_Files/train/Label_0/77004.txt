t = int(input())
for _ in range(t):
	(a, b, c, r) = map(int, input().split())
	if a > b:
		(a, b) = (b, a)
	total = b - a
	right = c + r
	left = c - r
	if c < a:
		covered = max(0, right - a)
	elif c > b:
		covered = max(0, b - left)
	else:
		covered = min(b, right) - max(a, left)
	print(max(total - covered, 0))
