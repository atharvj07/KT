for _ in range(int(input())):
	(n, x) = map(int, input().split())
	arr = list(map(int, input().split()))
	if sum(arr) == x:
		print('NO')
		continue
	l = []
	s = 0
	reject_list = []
	for i in range(n):
		s += arr[i]
		if s == x:
			reject_list.append(arr[i])
			s -= arr[i]
		else:
			l.append(arr[i])
	l.extend(reject_list)
	del reject_list, arr, n, x
	print('YES')
	print(' '.join(map(str, l)))
