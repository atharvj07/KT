(n, k) = [int(el) for el in input().split()]
s = input()
count = {}
for i in s:
	if i in count.keys():
		count[i] += 1
	else:
		count[i] = 1
m = float('+inf')
for i in range(k):
	j = i + 65
	q = chr(j)
	w = count.get(q)
	if w == None:
		w = 0
		count[q] = 0
	m = min(m, w)
out = 0
for i in range(k):
	j = i + 65
	if count.get(chr(j)) > m:
		out += count.get(chr(j)) - m
print(n - out)
