#define reps(i, s, n) for(int i = (int)(s); i < (int)(n); i++)

template<class T1, class T2> void chmin(T1 &a, T2 b){if(a>b)a=b;}
template<class T1, class T2> void chmax(T1 &a, T2 b){if(a<b)a=b;}

using pint = pair<int, int>;
using tint = tuple<int, int, int>;
using vint = vector<int>;

const int inf = 1LL << 55;
const int mod = 1e9 + 7;

int n;

int dx[] = {-1, -1, -1, 0, 0, 1, 1, 1};
int dy[] = {-1, 0, 1, -1, 1, -1, 0, 1};

map<string, int> mp;

bool in(int x, int y) {
  return 0 <= x && x < n && 0 <= y && y < n;
}

bool isinfe(int x, int y, string s) {
  int cnt = 0;
  rep(i, 8) {
    int nx = x + dx[i], ny = y + dy[i];
    if(!in(nx, ny)) continue;
    if(s[nx+ny*n] == '#') cnt++;
    else if(s[nx+ny*n] == '@') cnt++;
  }
  //cout<<x<<" "<<y<< " "<<cnt<<endl;
  if(s[x+y*n] == '.') return cnt == 3;
  else if(s[x+y*n] == '#') return cnt == 2 || cnt == 3;
  assert(false);
}

string infect(string s) {
  string t = "";
  rep(i, s.size()) {
    if(s[i] == '@') t += '@';
    else if(isinfe(i%n, i/n, s)) t += '#';
    else t += '.';
  }
  return t;
}

bool check(string s) {
  //cout<<s<<endl;
  rep(i, s.size()) if(s[i] == '#') return false;
  return true;
}

int findvehi(string s) {
  rep(i, s.size()) if(s[i] == '@') return i;
  assert(false);
}

int bfs(string s) {
  mp.clear();
  queue<string> que;
  mp[s] = 0;
  que.push(s);
  while(!que.empty()) {
    string t = que.front(); que.pop();
    if(check(t)) return mp[t];
    int idx = findvehi(t);
    int x = idx%n, y = idx/n;
    rep(i, 8) {
      string u = t;
      int nx = x + dx[i], ny = y + dy[i];
      int ni = nx+ny*n;
      if(!in(nx, ny)) continue;
      if(u[ni] != '.') continue;
      u[idx] = '.';
      u[ni] = '@';
      u = infect(u);
      if(mp.count(u)) continue;
      mp[u] = mp[t] + 1;
      que.push(u);
    }
  }
  return -1;
}

signed main()
{
  cin.tie(0);

  while(cin >> n, n) {
    string s = "";
    rep(i, n) {
      string t;
      cin >> t;
      s += t;
    }
    cout << bfs(s) << endl;
  }

  return 0;
}
