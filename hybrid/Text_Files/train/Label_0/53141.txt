// practice with rainboy
import java.io.*;
import java.util.*;

public class CF601D extends PrintWriter {
	CF601D() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF601D o = new CF601D(); o.main(); o.flush();
	}

	Random rand = new Random();
	static final int MD = 0x7fffffff;
	int X = rand.nextInt(MD - 26) + 26;
	int Y = rand.nextInt(MD - 26) + 26;
	int[] next, jj; int $ = 1;
	int link(int l, int j) {
		next[$] = l; jj[$] = j;
		return $++;
	}
	int[] ao, cc; byte[] ss;
	int[] ll, rr; int time; long[] key;
	Integer[] ii; int[] pp;
	int[] tr;
	void init(int n, int m) {
		next = new int[1 + m * 2]; jj = new int[1 + m * 2];
		ao = new int[n];
		cc = new int[n];
		ll = new int[n]; rr = new int[n]; key = new long[n];
		ii = new Integer[n]; for (int i = 0; i < n; i++) ii[i] = i;
		pp = new int[n];
		tr = new int[n];
	}
	void update(int i, int n, int x) {
		while (i < n) {
			tr[i] += x;
			i |= i + 1;
		}
	}
	int query(int i) {
		int x = 0;
		while (i >= 0) {
			x += tr[i];
			i &= i + 1; i--;
		}
		return x;
	}
	void dfs(int p, int i, int x, int y) {
		x = (int) (((long) x * X + ss[i]) % MD);
		y = (int) (((long) y * Y + ss[i]) % MD);
		ll[i] = time++;
		key[ll[i]] = (long) x * MD + y;
		for (int l = ao[i]; l != 0; l = next[l]) {
			int j = jj[l];
			if (j != p)
				dfs(i, j, x, y);
		}
		rr[i] = time;
	}
	void main() {
		int n = sc.nextInt();
		init(n, n - 1);
		for (int i = 0; i < n; i++)
			cc[i] = sc.nextInt();
		ss = sc.next().getBytes();
		for (int h = 0; h < n - 1; h++) {
			int i = sc.nextInt() - 1;
			int j = sc.nextInt() - 1;
			ao[i] = link(ao[i], j);
			ao[j] = link(ao[j], i);
		}
		dfs(-1, 0, 0, 0);
		Arrays.sort(ii, (i, j) -> key[i] == key[j] ? i - j : key[i] < key[j] ? -1 : 1);
		for (int i = 0; i < n; i++)
			pp[ii[i]] = i == 0 || key[ii[i - 1]] != key[ii[i]] ? -1 : ii[i - 1];
		Arrays.sort(ii, (i, j) -> rr[i] - rr[j]);
		int c_ = -1, k = 0;
		for (int i = 0, r_ = 0; i < n; i++) {
			int i_ = ii[i], l = ll[i_], r = rr[i_];
			while (r_ < r) {
				if (pp[r_] != -1)
					update(pp[r_], n, -1);
				update(r_, n, 1);
				r_++;
			}
			cc[i_] += query(r - 1) - query(l - 1);
			if (c_ < cc[i_]) {
				c_ = cc[i_];
				k = 1;
			} else if (c_ == cc[i_])
				k++;
		}
		println(c_);
		println(k);
	}
}
