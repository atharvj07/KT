(n, s, t) = map(int, input().split())
l = list(map(int, input().split()))
ans = 0
original_s = s
flag = 0
while flag == 0:
	if s == t:
		print(ans)
		flag = 1
	elif l[s - 1] == t:
		ans += 1
		print(ans)
		flag = 1
	elif l[s - 1] != t and l[s - 1] != original_s:
		ans += 1
		s = l[s - 1]
	else:
		print(-1)
		flag = 1
