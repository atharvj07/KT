class FastScanner {

    BufferedReader br;
    StringTokenizer st;

    FastScanner(InputStreamReader reader) {
        br = new BufferedReader(reader);
        st = new StringTokenizer("");
    }

    String next() {
        while (!st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (Exception ex) {
                return null;
            }
        }
        return st.nextToken();
    }

    int nextInt() {
        return Integer.parseInt(next());
    }

    long nextLong() {
        return Long.parseLong(next());
    }

    double nextDouble() {
        return Double.parseDouble(next());
    }

    String nextLine() {
        try {
            return br.readLine();
        } catch (IOException ex) {
            return null;
        }
    }
}

    FastScanner sc;

    final long inf = (long) 1e18;

    class Rib {

        int to;
        long w;

        public Rib(int to, long w) {
            this.to = to;
            this.w = w;
        }
    }

    class Para implements Comparable<Para> {

        long dp;
        int cnt;

        public Para(long dp, int cnt) {
            this.dp = dp;
            this.cnt = cnt;
        }

        @Override
        public int compareTo(Para x) {
            int res = Long.compare(dp, x.dp);
            if (res != 0) {
                return res;
            }
            return Integer.compare(cnt, x.cnt);
        }

        @Override
        public String toString() {
            return "dp,cnt: " + dp + ", " + cnt;
        }

    }

    int distSz;

    int[] idDist(int s, int n, ArrayList<Rib>[] gr) {
        long[] dist = new long[n + 1];
        fill(dist, inf);
        dist[s] = 0;

        TreeSet<Integer> Q = new TreeSet<>((a, b) -> {
            int res = Long.compare(dist[a], dist[b]);
            if (res != 0) {
                return res;
            }
            return a - b;
        });
        Q.add(s);

        while (!Q.isEmpty()) {
            int v = Q.pollFirst();
            for (Rib rib : gr[v]) {
                if (dist[rib.to] > dist[v] + rib.w) {
                    Q.remove(rib.to);
                    dist[rib.to] = dist[v] + rib.w;
                    Q.add(rib.to);
                }
            }
        }

        err.println("dist: " + Arrays.toString(dist));

        TreeMap<Long, Integer> mp = new TreeMap<>();
        mp.put(-1L, null); // ?????????????????????????????????????????????????
        for (int v = 1; v <= n; v++) {
            mp.put(dist[v], null);
        }
        distSz = mp.size();
        Iterator<Map.Entry<Long, Integer>> iter = mp.entrySet().iterator();
        int id = 0;
        while (iter.hasNext()) {
            Map.Entry<Long, Integer> next = iter.next();
            next.setValue(id++);
        }
        err.println("mp: " + mp);

        int[] res = new int[n + 1];
        for (int v = 1; v <= n; v++) {
            res[v] = mp.get(dist[v]);
        }
        return res;
    }

    void solve() {
        int n, m;
        int A, B;
        long[] p;
        ArrayList<Rib>[] gr;

        n = sc.nextInt();
        m = sc.nextInt();

        A = sc.nextInt();
        B = sc.nextInt();

        p = new long[n + 1];
        for (int v = 1; v <= n; v++) {
            p[v] = sc.nextInt();
        }

        gr = new ArrayList[n + 1];
        for (int v = 1; v <= n; v++) {
            gr[v] = new ArrayList<>(1); // ??????????????????? 0 ? 3 4 5
        }
        for (int iter = 0; iter < m; iter++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int w = sc.nextInt();
            gr[a].add(new Rib(b, w));
            gr[b].add(new Rib(a, w));
        }

        int[] aIdDist = idDist(A, n, gr);
        int aSz = distSz;

        int[] bIdDist = idDist(B, n, gr);
        int bSz = distSz;

        long[][] sum = new long[aSz + 1][bSz + 1];
        int[][] cnt = new int[aSz + 1][bSz + 1];

        for (int v = 1; v <= n; v++) {
            sum[aIdDist[v]][bIdDist[v]] += p[v];
            cnt[aIdDist[v]][bIdDist[v]]++;
        }
        for (int a = aSz - 1; 0 <= a; a--) {
            for (int b = bSz - 1; 0 <= b; b--) {
                sum[a][b] += sum[a + 1][b] + sum[a][b + 1] - sum[a + 1][b + 1];
                cnt[a][b] += cnt[a + 1][b] + cnt[a][b + 1] - cnt[a + 1][b + 1];
            }
        }

        TreeSet<Para>[][] rGo1Min = new TreeSet[aSz][bSz]; //for (long[] x : rGo1Min ) fill(x,inf);
        TreeSet<Para>[][] upGo0Min = new TreeSet[aSz][bSz]; //for (long[] x : upGo0Min) fill(x,inf);

        for (int a = 0; a < aSz; a++) {
            upGo0Min[a][bSz - 1] = new TreeSet<>();
            upGo0Min[a][bSz - 1].add(new Para(0, 0));
        }
        for (int b = 0; b < bSz; b++) {
            rGo1Min[aSz - 1][b] = new TreeSet<>();
            rGo1Min[aSz - 1][b].add(new Para(0, 0));
        }

        long go0_0_0 = 228;
        for (int a = aSz - 2; 0 <= a; a--) {
            for (int b = bSz - 2; 0 <= b; b--) {

                    if (cnt[a + 1][b + 1] != para.cnt) {
                        go0ab = max(go0ab, sum[a + 1][b + 1] - para.dp);
                    }
                }
                if (go0ab == Long.MIN_VALUE) {
                    go0ab = 0;
                }

                long go1ab = Long.MIN_VALUE;
                for (Para para : upGo0Min[a][b + 1]) {
                    if (cnt[a + 1][b + 1] != para.cnt) {
                        go1ab = max(go1ab, sum[a + 1][b + 1] - para.dp);
                    }
                }
                if (go1ab == Long.MIN_VALUE) {
                    go1ab = 0;
                }

                if (a == 0 && b == 0) {
                    go0_0_0 = go0ab;

//                rGo1Min [a][b] = min(go1ab, rGo1Min [a+1][b]);
                rGo1Min[a][b] = rGo1Min[a + 1][b];

                while (3 <= rGo1Min[a][b].size()) {

//                upGo0Min[a][b] = min(go0ab, upGo0Min[a][b+1]);
                upGo0Min[a][b] = upGo0Min[a][b + 1];

                    upGo0Min[a][b].pollLast();
                }

        }
