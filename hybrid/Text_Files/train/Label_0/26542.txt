#include <bits/stdc++.h>
using namespace std;

const int N = 200005;

int n, a[N];
vector<int> z;
priority_queue < int, vector<int>, greater<int> > pq[N];
long long res;

void merge(int u, int v) { // merge u -> v
    if (pq[u].size() > pq[v].size()) pq[u].swap(pq[v]);
    while(!pq[u].empty()) pq[v].push(pq[u].top()), pq[u].pop();
}

int main() {
    ios_base::sync_with_stdio(false); cin.tie(0);
    cin >> n;
    for (int i = 1; i <= n; ++i) cin >> a[i], z.push_back(a[i]);
    sort(z.begin(), z.end());
    z.erase(unique(z.begin(), z.end()), z.end());

    for (int i = 1; i <= n; ++i) {
        a[i] = lower_bound(z.begin(), z.end(), a[i]) - z.begin() + 1;
    }

    deque <int> dq;
    for (int i = 1; i <= n; ++i) dq.push_back(i), pq[i].push(a[i]);

    for (int i = 1; i <= n; ++i) {
        while(!dq.empty() && pq[dq.front()].top() != i) {
            int u = dq.front(); dq.pop_front();
            int v = dq.front();
            merge(u, v);
        }
        
        int x = dq.front(); dq.pop_front();
        pq[x].pop();

        res += pq[x].size();

        if (pq[x].size()) dq.push_back(x);
    }

    cout << res << endl;
}
