int query(int l, int r) {
  if (l == r) return -1;
  cout << "? " << l << " " << r << endl;
  cout.flush();
  int x;
  cin >> x;
  return x;
}

void report(int x) {
  cout << "! " << x << endl;
  cout.flush();
}

void solve(){
  int n; cin >> n;
  int k = query(1, n);
  if (k != 1 && query(1, k) == k) {
    int lo = 1, hi = k;
    while (lo < hi) {
      int m = lo + hi >> 1;
      if (query(m, k) != k) {
        hi = m;
      }else{
        lo = m + 1;
      }
    }
    report(lo-1);
  }else{
    int lo = k+1, hi = n;
    while (lo < hi) {
      int m = lo + hi >> 1;
      if (query(k, m) == k) hi = m;
      else lo = m + 1;
    }
    report(hi);
  }
}

int main(){
  ios::sync_with_stdio(false);
  cin.tie(NULL); cout.tie(NULL);

  int T = 1;
  // cin >> T;
  while (T--) solve();
  return 0;
}
