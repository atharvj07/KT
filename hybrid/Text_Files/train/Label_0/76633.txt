import sys
input = sys.stdin.readline
(n, m) = map(int, input().split())
strs = [input() for i in range(n)]
bycol = [0] * m
for i in range(1, n):
	for j in range(1, m):
		if strs[i - 1][j] == 'X' and strs[i][j - 1] == 'X':
			bycol[j] += 1
sums = [bycol[0]]
for i in range(1, m):
	sums.append(sums[-1] + bycol[i])
for i in range(int(input())):
	(a, b) = map(int, input().split())
	if a == m:
		print('YES')
	else:
		print('YES') if sums[b - 1] - sums[a - 1] == 0 else print('NO')
