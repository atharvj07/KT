def bubblesort(sta, n):
	for i in range(n):
		noswap = 1
		for j in range(len(sta) - 1, i, -1):
			if sta[j][1] > sta[j - 1][1]:
				(sta[j], sta[j - 1]) = (sta[j - 1], sta[j])
				noswap = 0
		if noswap:
			break
	return sta
(n, m) = [int(i) for i in input().split()]
reg = []
for i in range(m):
	reg.append([])
for i in range(n):
	p = list(input().split())
	p[1] = int(p[1])
	p[2] = int(p[2])
	reg[p[1] - 1].append([p[0], p[2]])
for i in range(m):
	a = bubblesort(reg[i], 3)
	if len(a) == 2:
		print(a[0][0], a[1][0])
	elif a[1][1] == a[2][1]:
		print('?')
	else:
		print(a[0][0], a[1][0])
