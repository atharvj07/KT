#include <bits/stdc++.h>
using namespace std;
int dfs(const vector<vector<pair<int, int> > >& g, int w, vector<int>& c,
        int u) {
  int ret = 1;
  for (int i = 0; i < int(g[u].size()); ++i) {
    if (g[u][i].second == w) {
      int v = g[u][i].first;
      if (c[v] == -1) {
        c[v] = c[u];
        ret += dfs(g, w, c, v);
      }
    }
  }
  return ret;
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  int n, m, a, b;
  cin >> n >> m >> a >> b;
  vector<vector<pair<int, int> > > g(n);
  for (int i = 0; i < m; ++i) {
    int u, v, w;
    cin >> u >> v >> w;
    --u, --v;
    g[u].push_back(pair<int, int>(v, w));
    g[v].push_back(pair<int, int>(u, w));
  }
  vector<int> c(n, -1);
  vector<int> csz;
  for (int i = 0; i < n; ++i) {
    if (c[i] == -1) {
      c[i] = int(csz.size());
      csz.push_back(dfs(g, a, c, i));
    }
  }
  vector<int> id(csz.size(), -1);
  int ids = 0;
  for (int i = 0; i < int(csz.size()); ++i) {
    if (csz[i] > 3) {
      id[i] = ids;
      ++ids;
    }
  }
  vector<vector<int> > dist(n, vector<int>(1 << ids, -1));
  priority_queue<pair<int, pair<int, int> >,
                 vector<pair<int, pair<int, int> > >,
                 greater<pair<int, pair<int, int> > > >
      pq;
  int vis0 = 0;
  if (id[c[0]] != -1) vis0 |= 1 << id[c[0]];
  dist[0][vis0] = 0;
  pq.push(make_pair(0, pair<int, int>(0, vis0)));
  for (; !pq.empty();) {
    int u = pq.top().second.first;
    int vis = pq.top().second.second;
    int cur_d = pq.top().first;
    pq.pop();
    if (cur_d > dist[u][vis]) continue;
    for (int i = 0; i < int(g[u].size()); ++i) {
      int v = g[u][i].first;
      int nxt_vis = vis;
      if (g[u][i].second == b) {
        if (c[v] == c[u]) continue;
        if (id[c[v]] != -1) {
          if (((vis >> id[c[v]]) & 1) == 1) continue;
          nxt_vis |= 1 << id[c[v]];
        }
      }
      int d = dist[u][vis] + g[u][i].second;
      if (dist[v][nxt_vis] == -1 || d < dist[v][nxt_vis]) {
        dist[v][nxt_vis] = d;
        pq.push(make_pair(d, pair<int, int>(v, nxt_vis)));
      }
    }
  }
  for (int i = 0; i < n; ++i) {
    int mn = -1;
    for (int j = 0; j < int(dist[i].size()); ++j) {
      if (dist[i][j] != -1 && (mn == -1 || dist[i][j] < mn)) {
        mn = dist[i][j];
      }
    }
    if (i > 0) cout << " ";
    cout << mn;
  }
  cout << endl;
}
