(n, k) = map(int, input().split())
l = list(map(int, input().split()))
l.sort()
out = 10000000000 + 7
d = [[] for i in range(0, max(l) + 1)]
if k != 1:
	for i in l:
		c = 0
		while i:
			d[i].append(c)
			c += 1
			i //= 2
	for i in d:
		if len(i) >= k:
			out = min(out, sum(i[:k]))
	print(out)
else:
	print(0)
