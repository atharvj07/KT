(n, k) = [int(i) for i in input().split()]

def ok(a):
	print('YES')
	print(*a)
if k * (k + 1) // 2 > n:
	print('NO')
else:
	v = (n - k * (k + 1) // 2) // k
	a = [v + i for i in range(1, k + 1)]
	if v == 0:
		if k == 2:
			if sum(a) != n:
				print('NO')
			else:
				ok(a)
		elif k == 3:
			if n == 6:
				ok(a)
			elif n == 7:
				ok([1, 2, 4])
			else:
				print('NO')
		else:
			if n - sum(a) >= 2:
				a[-2] += 1
			a[-1] += n - sum(a)
			ok(a)
	else:
		a[-1] += n - sum(a)
		ok(a)
