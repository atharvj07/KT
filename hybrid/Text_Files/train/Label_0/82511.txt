test = int(input())
for _ in range(test):
	n = int(input())
	s = input()
	hasw = False
	hascw = False
	for i in s:
		if i == '<':
			hasw = True
		if i == '>':
			hascw = True
	if hasw and hascw:
		s += s[0]
		ans = 0
		for i in range(n):
			if s[i] == '-' or s[i + 1] == '-':
				ans += 1
		print(ans)
	else:
		print(n)
