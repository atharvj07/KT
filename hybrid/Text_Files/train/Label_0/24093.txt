#include <bits/stdc++.h>
using namespace std;
long long a, b, n, x, y, z;
int main() {
  scanf("%I64d %I64d %I64d", &a, &b, &n);
  for (int i = 0; i < n; i++) {
    scanf("%I64d %I64d %I64d", &x, &y, &z);
    long long st = x;
    long long ed = 10000000LL;
    long long zz = a * (st - 1) + b * ((st - 1) * (st - 2) / 2);
    while (st <= ed) {
      long long mid = (st + ed) / 2;
      long long xx = a + b * (mid - 1);
      long long yy = a * mid + b * (mid * (mid - 1) / 2) - zz;
      if (xx <= y && yy <= y * z) {
        st = mid + 1;
      } else {
        ed = mid - 1;
      }
    }
    --st;
    if (st < x) {
      st = -1;
    }
    printf("%I64d\n", st);
  }
  return 0;
}
