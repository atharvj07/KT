#include <cstdio>
#include <cstring>

int p[2][100001],q[2][100001],n,k;
int tem[100001],A[100001],ans[100001],con[100001],top,f[7][100001];
bool vis[100001];
void mul(int *a,int *b,int *c){
	for(int i=1;i<=n;i++)tem[i]=a[b[i]];
	for(int i=1;i<=n;i++)c[i]=tem[i];
}
int main(){
	scanf("%d%d",&n,&k);
	for(int i=1;i<=n;i++)scanf("%d",&p[0][i]),p[1][p[0][i]]=i;
	for(int i=1;i<=n;i++)scanf("%d",&q[0][i]),q[1][q[0][i]]=i;
	for(int i=1;i<=n;i++)A[i]=ans[i]=i;
	mul(A,q[0],A);
	mul(A,p[1],A);
	mul(A,q[1],A);
	mul(A,p[0],A);
	for(int i=1;i<=n;i++)
		if(!vis[i]){
			con[(top=0)++]=i;
			int Tem=A[i];
			vis[i]=1;
			while(Tem!=i){
				vis[Tem]=1;
				con[top++]=Tem;
				Tem=A[Tem];
			}
			for(int i=0;i<top;i++)
				tem[con[i]]=con[(i+((k-1)/6))%top];
		}
	for(int i=1;i<=n;i++)A[i]=tem[i];
	mul(ans,A,ans);
	memcpy(f[1],p[0],sizeof p[0]);
	memcpy(f[2],q[0],sizeof q[0]);
	for(int i=3;i<=(k-1)%6+1;i++){
		for(int j=1;j<=n;j++)tem[f[i-2][j]]=j;
		for(int j=1;j<=n;j++)f[i-2][j]=tem[j];
		mul(f[i-1],f[i-2],f[i]);
	}
	mul(ans,f[(k-1)%6+1],ans);
	for(int i=1;i<=n;i++)tem[A[i]]=i;
	for(int i=1;i<=n;i++)A[i]=tem[i];
	mul(ans,A,ans);
	for(int i=1;i<=n;i++)printf("%d ",ans[i]);
}
