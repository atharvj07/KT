from math import ceil, sqrt
for i in range(int(input())):
	n = int(input())
	spec = []
	m = n
	while m > 2:
		spec.append(m)
		m = ceil(sqrt(m))
	if len(spec) > 0 and spec[-1] != 2:
		spec.append(2)
	ans = []
	used = [0] * n
	for e in spec:
		used[e - 1] = 1
	for i in range(1, n - 1):
		if not used[i]:
			ans.append((i + 1, n))
	for i in range(1, len(spec)):
		ans.append((spec[i - 1], spec[i]))
		ans.append((spec[i - 1], spec[i]))
	print(len(ans))
	for e in ans:
		print(*e)
