for _ in range(int(input())):
	path = input()
	track = {}
	track[0, 0] = [0, 0, 0, 0]
	coordinate = (0, 0)
	time = 0
	for p in path:
		if p == 'N':
			if track[coordinate][0] == 1:
				time += 1
			else:
				time += 5
				track[coordinate][0] = 1
			t1 = coordinate[0]
			t2 = coordinate[1] + 1
			coordinate = (t1, t2)
			if coordinate in track.keys():
				track[coordinate][2] = 1
			else:
				track[coordinate] = [0, 0, 1, 0]
		elif p == 'S':
			if track[coordinate][2] == 1:
				time += 1
			else:
				time += 5
				track[coordinate][2] = 1
			t1 = coordinate[0]
			t2 = coordinate[1] - 1
			coordinate = (t1, t2)
			if coordinate in track.keys():
				track[coordinate][0] = 1
			else:
				track[coordinate] = [1, 0, 0, 0]
		elif p == 'E':
			if track[coordinate][1] == 1:
				time += 1
			else:
				time += 5
				track[coordinate][1] = 1
			t1 = coordinate[0] + 1
			t2 = coordinate[1]
			coordinate = (t1, t2)
			if coordinate in track.keys():
				track[coordinate][3] = 1
			else:
				track[coordinate] = [0, 0, 0, 1]
		else:
			if track[coordinate][3] == 1:
				time += 1
			else:
				time += 5
				track[coordinate][3] = 1
			t1 = coordinate[0] - 1
			t2 = coordinate[1]
			coordinate = (t1, t2)
			if coordinate in track.keys():
				track[coordinate][1] = 1
			else:
				track[coordinate] = [0, 1, 0, 0]
	print(time)
