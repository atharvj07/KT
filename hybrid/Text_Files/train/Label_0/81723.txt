N = 105
graph = [[] for _ in range(N)]
component = [0] * N
visited = [False] * N

def explore(node):
	global cnt
	visited[node] = True
	component[node] = cnt
	for (neighbour, color) in graph[node]:
		if visited[neighbour] == False:
			explore(neighbour)

def dfs(n):
	global cnt
	cnt = 0
	for i in range(1, n + 1):
		if visited[i] == False:
			cnt += 1
			explore(i)
