def bin(x):

def printlist(l):
	print(''.join([str(x) for x in l]))

def listring(l):
	return ''.join([str(x) for x in l])
(n, m) = mp()
if n == 1:
	print(1)
	exit()
if n == 2:
	print(3 - m)
	exit()
if n % 2 == 0:
	a = n // 2
	if a == m:
		print(a + 1)
	elif a < m:
		print(m - 1)
	else:
		print(m + 1)
else:
	a = n // 2 + 1
	if m == a:
		print(a - 1)
	elif m > a:
		print(m - 1)
	else:
		print(m + 1)
