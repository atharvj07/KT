#include<bits/stdc++.h>
typedef long long int ll;
typedef unsigned long long int ull;
#define BIG_NUM 2000000000
#define HUGE_NUM 99999999999999999
#define MOD 1000000007
#define EPS 0.000000001
using namespace std;


int num_row,col_half,N;

int main(){

	scanf("%d %d %d",&num_row,&col_half,&N);

	int goal_row = num_row,goal_col = col_half+1;

	vector<int> V;

	int tmp_row,tmp_col;
	for(int i = 0; i < N; i++){

		scanf("%d %d",&tmp_row,&tmp_col);

		if(tmp_col > col_half){
			tmp_col += 1;
		}
		V.push_back(goal_row-tmp_row+abs(goal_col-tmp_col));
	}
	sort(V.begin(),V.end());

	int current = 0;

	for(int i = 0; i < V.size(); i++){

		current = max(V[i],current+1);
	}

	printf("%d\n",current+1);

	return 0;
}
