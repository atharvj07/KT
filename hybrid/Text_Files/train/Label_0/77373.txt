import math

def thanosSort(arr):
	mas = []
	t = 1
	length = 1
	for i in range(int(math.log(len(arr)) / math.log(2))):
		t *= 2
		for j in range(len(arr) // t):
			start = j * t
			stop = j * t + t
			mas.append(arr[start:stop])
	for e in mas:
		if e == sorted(e):
			if len(e) > length:
				length = len(e)
	return length
n = int(input())
arr = list(map(int, input().split()))
print(thanosSort(arr))
