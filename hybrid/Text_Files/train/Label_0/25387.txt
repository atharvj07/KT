#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, i, k;
  string str = "", buff;
  cin >> n;
  while (n--) {
    cin >> buff;
    if (buff[0] == 'p')
      cout << str + '/' << endl;
    else {
      cin >> buff;
      if (buff[0] == '/')
        str = buff;
      else
        str += '/' + buff;
    }
    while ((k = str.find('.')) != -1) {
      i = ++k - 3;
      while (i >= 0 && str[i--] != '/')
        ;
      str.erase(++i, k - i + 1);
    }
  }
  return 0;
}
