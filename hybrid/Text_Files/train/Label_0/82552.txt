(n, m) = map(int, input().split())
arrA = list(map(int, input().split()))
arrB = list(map(int, input().split()))
ans = 511
for val in range(510, -1, -1):
	temp = True
	for a in arrA:
		for b in arrB:
			if a & b | val == val:
				break
		else:
			temp = False
			break
	if temp:
		ans = val
print(ans)
