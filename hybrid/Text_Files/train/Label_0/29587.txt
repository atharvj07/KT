#include <bits/stdc++.h>
using namespace std;
double MDST(vector<vector<int> >& w, int n) {
  for (int k = 0; k < (n); ++k)
    for (int i = 0; i < (n); ++i)
      for (int j = 0; j < (n); ++j)
        if (w[i][k] + w[k][j] < w[i][j]) w[i][j] = w[i][k] + w[k][j];
  double mx = 1e9;
  for (int u = 0; u < (n); ++u)
    for (int v(u + 1); v < (n); ++v) {
      vector<pair<int, int> > z, fin;
      for (int i = 0; i < (n); ++i) z.push_back(make_pair(w[i][u], w[i][v]));
      sort(z.begin(), z.end());
      for (int i = 0; i < (z.size()); ++i) {
        while (fin.size() && fin.back().second < z[i].second) fin.pop_back();
        fin.push_back(z[i]);
      }
      double tx = max(min(fin[0].first + w[u][v], fin[0].second),
                      min(fin[0].first + w[u][v], fin[0].second));
      tx = max<double>(max(min(fin.back().first + w[u][v], fin.back().second),
                           min(fin.back().first + w[u][v], fin.back().second)),
                       tx);
      for (int i = 0; i < (fin.size() - 1); ++i) {
        const int& a = fin[i].first;
        const int& b = fin[i + 1].second;
        double pos = (0.5 + (b - a) / (2.0 * w[u][v]));
        double val = a + pos * w[u][v];
        tx = min(tx, val);
      }
      mx = min(tx, mx);
    }
  return mx;
}
int main() {
  int n, m;
  cin >> n >> m;
  const int inf = 1e9;
  vector<vector<int> > w(n, vector<int>(n, inf));
  for (int i = 0; i < (n); ++i) w[i][i] = 0;
  for (int i = 0; i < (m); ++i) {
    int a, b;
    cin >> a >> b;
    --a;
    --b;
    cin >> w[a][b];
    w[b][a] = w[a][b];
  }
  cout << MDST(w, n) << endl;
  return 0;
}
