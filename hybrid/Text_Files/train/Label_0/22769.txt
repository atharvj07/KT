#include<iostream>

int memo[500001];

int main(){
	int n, m;
	while(std::cin >> n >> m, n){
		for(int i=0;i<500001;i++){
			memo[i] = 0;
		}

		bool flag = false;
		char prev_c, c;
		int combo = 0;
		for(int i=0;i<m;i++){
			std::cin >> c;

			if(flag){
				if(prev_c == c){//清算処理
					for(int j=1;j<=combo;j++){
						memo[j] += combo - (j-1);
					}
					flag = false;
					combo = 0;
				}else{
					combo += (c=='I')?1:0;
				}
			}

			if(!flag && c == 'I'){
				flag = true;
			}

			prev_c = c;
		}

		std::cout << memo[n] << std::endl;
	}
}
