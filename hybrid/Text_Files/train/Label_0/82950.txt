t = int(input())
n = []
for _ in range(t):
	k = int(input())
	a = int(k ** 0.5)
	if a ** 2 == k:
		stolbec = 1
		stroka = a
	else:
		d = k - a ** 2
		if d == a + 1:
			stroka = a + 1
			stolbec = a + 1
		elif d < a + 1:
			stolbec = a + 1
			stroka = d
		else:
			stroka = a + 1
			stolbec = a + 1 - (d - (a + 1))
	n.append(stroka)
	n.append(stolbec)
for i in range(0, len(n) - 1, 2):
	print(n[i], n[i + 1])
