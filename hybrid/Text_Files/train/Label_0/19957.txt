#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  scanf("%d", &T);
  while (T--) {
    int n, m;
    scanf("%d%d", &n, &m);
    int sum = 0;
    for (int i = 1; i <= n; i++) {
      int x;
      scanf("%d", &x);
      sum += x;
    }
    if (n == 2) {
      printf("-1\n");
      continue;
    }
    if (m < n)
      printf("-1\n");
    else {
      printf("%d\n", 2 * sum);
      for (int i = 1; i < n; i++) {
        printf("%d %d\n", i, i + 1);
      }
      printf("%d %d\n", 1, n);
    }
  }
  return 0;
}
