#include <bits/stdc++.h>
using namespace std;
long long inv(long long tar) {
  long long base[40];
  base[0] = tar;
  for (int i = 1; i < 40; i++) {
    base[i] = base[i - 1] * base[i - 1] % 1000000007LL;
    if ((1LL << i) > 1000000007LL - 2) break;
  }
  long long ret = 1, left = 1000000007LL - 2;
  for (int i = 39; i >= 0; i--) {
    if (left >= (1LL << i)) {
      left -= (1LL << i);
      ret = ret * base[i] % 1000000007LL;
    }
  }
  return ret;
}
int main(void) {
  long long n;
  scanf("%I64d", &n);
  long long cur = 1, fact = 1;
  for (int i = 0; i < n; i++) {
    cur = cur * (n * 2 - 1 - i) % 1000000007LL;
    fact = fact * (i + 1) % 1000000007LL;
  }
  cur = cur * inv(fact) % 1000000007LL;
  cur = cur * 2 % 1000000007LL;
  cur = (cur - n + 1000000007LL) % 1000000007LL;
  printf("%I64d\n", cur);
  return 0;
}
