#include <bits/stdc++.h>
using namespace std;
set<int> prime;
void SieveOfEratosthenes(int n) {
  bool prme[n + 1];
  memset(prme, true, sizeof(prme));
  for (int p = 2; p * p <= n; p++) {
    if (prme[p] == true) {
      for (int i = p * p; i <= n; i += p) prme[i] = false;
    }
  }
  for (int p = 2; p <= n; p++)
    if (prme[p]) prime.insert(p);
}
void solve() {
  int n, m;
  int nxt[100];
  cin >> n >> m;
  string s;
  cin >> s;
  for (int i = 0; i < 26; ++i) nxt[i] = i;
  while (m--) {
    char c1, c2;
    cin >> c1 >> c2;
    int x = c1 - 'a';
    int y = c2 - 'a';
    for (int i = 0; i < 26; ++i)
      if (nxt[i] == x)
        nxt[i] = y;
      else if (nxt[i] == y)
        nxt[i] = x;
  }
  for (int i = 0; i < n; ++i) cout << char('a' + nxt[s[i] - 'a']);
  cout << '\n';
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  solve();
}
