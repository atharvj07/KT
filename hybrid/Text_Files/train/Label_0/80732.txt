import sys
from collections import defaultdict, deque

def buildTree(root):
	q = deque()
	q.append((root, root))
	vis = [0 for _ in range(n + 1)]
	vis[root] = 1
	while q:
		(v, fa) = q.pop()
		for x in preedge[v]:
			if vis[x] != 1:
				vis[x] = 1
				if col[x] == col[fa]:
					q.append((x, fa))
				else:
					edge[fa].append(x)
					edge[x].append(fa)
					q.append((x, x))

def getDeep(root):
	vis = [0 for _ in range(n + 1)]
	d = {}
	q = deque()
	q.append(root)
	vis[root] = 1
	d[root] = 0
	while q:
		u = q.pop()
		for x in edge[u]:
			if vis[x] == 0:
				vis[x] = 1
				d[x] = d[u] + 1
				q.append(x)
	return d
n = int(input())
precol = [int(_) for _ in input().split()]
preedge = [[] for _ in range(n + 1)]
for _ in range(n - 1):
	(x, y) = [int(_) for _ in input().split()]
	preedge[x].append(y)
	preedge[y].append(x)
root = (n + 1) // 2
edge = [[] for _ in range(n + 1)]
col = [0 for _ in range(n + 1)]
for x in range(1, n + 1):
	col[x] = precol[x - 1]
buildTree(root)
d = getDeep(root)
u = max(d.keys(), key=lambda k: d[k])
print((max(getDeep(u).values()) + 1) // 2)
