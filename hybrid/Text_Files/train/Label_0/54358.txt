import java.io.BufferedReader;
import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {

	static class Pair implements Comparable<Pair> {

		int v;
		long d;

		Pair(int v, long d) {
			this.v = v;
			this.d = d;
		}

		@Override
		public int compareTo(Pair o) {
			return Long.compare(d, o.d);
		}

		@Override
		public String toString() {
			return "(" + v + ", " + d + ")";
		}

	}

	int MOD = 1000 * 1000 * 1000 + 7;

	class Edge implements Comparable<Edge> {
		int to;
		long w;

		public Edge(int to, long w) {
			this.to = to;
			this.w = w;
		}

		@Override
		public int compareTo(Edge o) {
			return Long.compare(w, o.w);
		}
	}

	int[] tree;
	int n, q, s;
	int OFFSET;

	private ArrayList<Edge>[] G;
	private long oo = 100_000_000_000_000L;

	private void solve() {
		n = in.nextInt();
		q = in.nextInt();
		s = in.nextInt();
		OFFSET = 4 * n;
		G = (ArrayList<Edge>[]) new ArrayList[8 * n];
		for (int i = 0; i < 8 * n; i++) {
			G[i] = new ArrayList<Edge>();
		}
		buildTree(1, 1, n);

		for (int i = 0; i < q; i++) {
			int t = in.nextInt();
			if (t == 1) {
				int v = in.nextInt();
				int u = in.nextInt();
				int w = in.nextInt();
				int vId = getId(v, v).get(0);
				int uId = getId(u, u).get(0);
				G[vId].add(new Edge(uId, w));
				continue;
			}
			if (t == 2) {
				int v = in.nextInt();
				int l = in.nextInt();
				int r = in.nextInt();
				int w = in.nextInt();
				int vId = getId(v, v).get(0);
				ArrayList<Integer> ids = getId(l, r);
				for (int to : ids) {
					G[vId].add(new Edge(to, w));
				}
				continue;
			}
			if (t == 3) {
				int v = in.nextInt();
				int l = in.nextInt();
				int r = in.nextInt();
				int w = in.nextInt();
				int vId = getId(v, v).get(0);
				ArrayList<Integer> ids = getId(l, r);
				for (int to : ids) {
					G[to + OFFSET].add(new Edge(vId, w));
				}

			}
		}

		int sId = getId(s, s).get(0);
		long[] dist = dijkstra(sId);
		for (int i = 1; i <= n; i++) {
			int id = getId(i, i).get(0);
			long d = dist[id];
			if (d == oo) {
				d = -1;
			}
			out.print(d + " ");
		}
	}

	private long[] dijkstra(int sId) {
		long[] dist = new long[8 * n];
		Arrays.fill(dist, oo);
		PriorityQueue<Pair> Q = new PriorityQueue<Pair>();
		Q.add(new Pair(sId, 0));
		boolean[] visited = new boolean[8 * n];
		dist[sId] = 0;

		while (!Q.isEmpty()) {
			Pair p = Q.poll();

			if (visited[p.v] || p.d > dist[p.v]) {
				continue;
			}

			visited[p.v] = true;

			for (Edge e : G[p.v]) {
				if (dist[p.v] + e.w < dist[e.to]) {
					dist[e.to] = dist[p.v] + e.w;
					Q.add(new Pair(e.to, p.d + e.w));
				}
			}
		}
		return dist;
	}

	private ArrayList<Integer> getId(int idxL, int idxR) {
		ArrayList<Integer> res = new ArrayList<Integer>();
		getId(idxL, idxR, 1, 1, n, res);
		return res;
	}

	private void getId(int idxL, int idxR, int node, int l, int r, ArrayList<Integer> res) {
		if (r < idxL || l > idxR) {
			return;
		}

		if (idxL <= l && r <= idxR) {
			res.add(node);
			return;
		}

		int mid = l + (r - l) / 2;
		getId(idxL, idxR, 2 * node, l, mid, res);
		getId(idxL, idxR, 2 * node + 1, mid + 1, r, res);

		return;
	}

	private void buildTree(int node, int l, int r) {
		if (r == l) {
			G[node].add(new Edge(node + OFFSET, 0));
			return;
		}

		int mid = l + (r - l) / 2;
		buildTree(2 * node, l, mid);
		buildTree(2 * node + 1, mid + 1, r);

		G[node].add(new Edge(2 * node, 0));
		G[node].add(new Edge(2 * node + 1, 0));

		G[2 * node + OFFSET].add(new Edge(node + OFFSET, 0));
		G[2 * node + 1 + OFFSET].add(new Edge(node + OFFSET, 0));
	}

	private FastScanner in;
	private PrintWriter out;

	void runIO() {

		in = new FastScanner(System.in);
		out = new PrintWriter(System.out);

		solve();

		out.close();
	}

	class FastScanner {
		BufferedReader br;
		StringTokenizer st;

		public FastScanner(File f) {
			try {
				br = new BufferedReader(new FileReader(f));
			} catch (FileNotFoundException e) {
				e.printStackTrace();
			}
		}

		public FastScanner(InputStream f) {
			br = new BufferedReader(new InputStreamReader(f));
		}

		String next() {
			while (st == null || !st.hasMoreTokens()) {
				String s = null;
				try {
					s = br.readLine();
				} catch (IOException e) {
					e.printStackTrace();
				}
				if (s == null)
					return null;
				st = new StringTokenizer(s);
			}
			return st.nextToken();
		}

		boolean hasMoreTokens() {
			while (st == null || !st.hasMoreTokens()) {
				String s = null;
				try {
					s = br.readLine();
				} catch (IOException e) {
					e.printStackTrace();
				}
				if (s == null)
					return false;
				st = new StringTokenizer(s);
			}
			return true;
		}

		int nextInt() {
			return Integer.parseInt(next());
		}

		long nextLong() {
			return Long.parseLong(next());
		}

		double nextDouble() {
			return Double.parseDouble(next());
		}
	}

	public static void main(String[] args) {
		new Main().runIO();
	}
}
