from collections import defaultdict
import math

def prime(n):
	global d
	while n % 2 == 0:
		d[2] += 1
		n = n // 2
	for i in range(3, int(math.sqrt(n)) + 1, 2):
		while n % i == 0:
			d[i] += 1
			n = n // i
	if n > 2:
		d[n] += 1
for _ in range(int(input())):
	n = int(input())
	x = n
	d = defaultdict(int)
	prime(n)
	answer = None
	s = 0
	for (i, j) in d.items():
		if j > s:
			s = j
			answer = (i, j)
	print(answer[1])
	for i in range(answer[1] - 1):
		print(answer[0], end=' ')
		x = x // answer[0]
	if x > 1:
		print(x)
	else:
		print()
