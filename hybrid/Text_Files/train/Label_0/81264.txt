import sys
input = sys.stdin.buffer.readline
t = int(input())
for _ in range(t):
	n = int(input())
	adj = [[] for _ in range(n + 1)]
	indegree = [0 for _ in range(n + 1)]
	for _ in range(n):
		(u, v) = [int(x) for x in input().split()]
		adj[u].append(v)
		adj[v].append(u)
		indegree[u] += 1
		indegree[v] += 1
	visited = [False for _ in range(n + 1)]
	leaves = []
	for node in range(1, n + 1):
		if indegree[node] == 1:
			leaves.append(node)
	subTreeSizes = [1 for _ in range(n + 1)]
	while leaves:
		leaf = leaves.pop()
		visited[leaf] = True
		indegree[leaf] -= 1
		for nex in adj[leaf]:
			if visited[nex] == False:
				subTreeSizes[nex] += subTreeSizes[leaf]
				indegree[nex] -= 1
				if indegree[nex] == 1:
					leaves.append(nex)
	rootsAtCycle = []
	for node in range(1, n + 1):
		if visited[node] == False:
			rootsAtCycle.append(node)
	intraTreeWays = 0
	interTreeWays = 0
	for root in rootsAtCycle:
		size = subTreeSizes[root]
		intraTreeWays += size * (size - 1)
		interTreeWays += 2 * (n - size) * size
	ans = intraTreeWays + interTreeWays
	ans //= 2
	print(ans)
