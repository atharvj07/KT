import java.util.*;
import java.lang.Math;
public class Test {

	public static void main(String[] args) {
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		
		for (int i = 0; i < t; i++) {
			int n=sc.nextInt();
			int[] a=new int[n];
			for (int j = 0; j < a.length; j++) {
				a[j]=sc.nextInt();
			}
			int flag=0;
			int flag1=0;
			int ff=0;
			for (int j = 0; j < a.length; j++) {
				if(a[j]%2==0)flag++;
				else flag1++;
				if(flag>=1 && flag1>=1) {
					ff=1;
					System.out.println("NO");
					break;
				}
			}
			if(ff==0)System.out.println("YES");
		}
	}

}
