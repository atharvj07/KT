#include <bits/stdc++.h>
using namespace std;
vector<vector<int> > G(1000, vector<int>(1000, 0));
void set4(int i, int j) {
  G[i][j] = 1;
  G[i + 1][j] = 1;
  G[i - 1][j] = 1;
  G[i][j + 1] = 1;
  G[i][j - 1] = 1;
}
void solve() {
  int N;
  cin >> N;
  vector<pair<int, int> > ans;
  if (N == 1) {
    set4(1, 1);
    G[0][0] = G[2][2] = 1;
  } else if (N == 2) {
    set4(2, 1);
    set4(3, 1);
    G[5][1] = G[5][2] = G[5][3] = G[5][4] = G[4][4] = G[3][4] = G[2][4] =
        G[1][4] = G[0][4] = G[0][3] = G[0][2] = G[0][1] = 1;
  } else if (N % 3 == 1) {
    int count = 1;
    set4(1, 1);
    int i = 2;
    while (count != N) {
      set4(i, i);
      set4(i - 1, i);
      set4(i, i - 1);
      count += 3;
      i++;
    }
  } else if (N % 3 == 2) {
    int count = 2;
    set4(2, 3);
    set4(3, 2);
    G[3][1] = G[3][0] = G[2][0] = G[1][0] = G[0][0] = G[0][1] = G[0][2] =
        G[0][3] = 1;
    int i = 3;
    while (count != N) {
      set4(i, i);
      set4(i + 1, i);
      set4(i, i + 1);
      count += 3;
      i++;
    }
    G[i + 2][i - 1] = G[i + 2][i] = G[i + 2][i + 1] = G[i + 2][i + 2] =
        G[i + 1][i + 2] = G[i][i + 2] = G[i - 1][i + 2] = 1;
  } else if (N % 3 == 0) {
    int count = 3;
    set4(1, 1);
    set4(1, 2);
    set4(2, 1);
    int i = 2;
    while (count != N) {
      set4(i, i);
      set4(i + 1, i);
      set4(i, i + 1);
      count += 3;
      i++;
    }
    G[i + 2][i - 1] = G[i + 2][i] = G[i + 2][i + 1] = G[i + 2][i + 2] =
        G[i + 1][i + 2] = G[i][i + 2] = G[i - 1][i + 2] = 1;
  }
  for (int i = 0; i < 1000; i++) {
    for (int j = 0; j < 1000; j++) {
      if (G[i][j]) ans.push_back({i, j});
    }
  }
  cout << ans.size() << "\n";
  ;
  for (pair<int, int> &p : ans) cout << p.first << " " << p.second << "\n";
  return;
}
int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin.exceptions(cin.failbit);
  solve();
}
