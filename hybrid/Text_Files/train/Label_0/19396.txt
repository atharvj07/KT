#include <array>
#include <climits>
#include <cstdlib>
#include <iostream>
#include <vector>
using namespace std;

template<class T> inline void chmax(T& a, const T& b) { if(b > a) a = b; }
typedef vector<vector<int>> matrix;

array<int, 6> dx = {-1, 0, 1, 1, 0, -1};
array<int, 6> dy = {-1, -1, 0, 1, 1, 0};

inline bool out(int x, int y, int n) {
	return x < 0 || y < 0 || x > y || y >= n;
}

int dfs(int x, int y, int v, const matrix &field, matrix &visited) {
	int res = 1;
	visited[y][x] = true;
	for(int d = 0; d < 6; ++d) {
		const int nx = x + dx[d];
		const int ny = y + dy[d];
		if(out(nx, ny, field.size()) || field[ny][nx] != v || visited[ny][nx])
			continue;

		res += dfs(nx, ny, v, field, visited);
	}

	return res;
}

int calc_score(int x, int y, const matrix &field) {
	const int n = field.size();
	int res = n * (n + 1) / 2;
    matrix visited(n, vector<int>(n, false));

	for(int i = 0; i < n; ++i) {
		for(int j = 0; j <= i; ++j) {
			if(field[i][j] == 0) {
				visited[i][j] = true;
				--res;
				for(int d = 0; d < 6; ++d) {
					const int nx = j + dx[d];
					const int ny = i + dy[d];
					if(out(nx, ny, n) || !field[ny][nx] || visited[ny][nx])
						continue;

					res -= dfs(nx, ny, field[ny][nx], field, visited);
				}
			}
		}
	}

	if(!visited[y][x])
		res -= 2 * dfs(x, y, field[y][x], field, visited);

	return res;
}

int main() {
	cin.tie(nullptr);
	ios::sync_with_stdio(false);

	for(int n, c; cin >> n >> c && n;) {
		matrix field(n);
		for(int i = 0; i < n; ++i) {
			for(int j = 0; j <= i; ++j) {
				int value;
				cin >> value;
				field[i].emplace_back(value);
			}
		}

		int ans = -INT_MAX;
		for(int i = 0; i < n; ++i) {
			for(int j = 0; j <= i; ++j) {
				if(field[i][j] == 0) {
					field[i][j] = c;
					chmax(ans, calc_score(j, i, field));
					field[i][j] = 0;
				}
			}
		}

		cout << ans << endl;
	}

	return EXIT_SUCCESS;
}
