def find(mat):
	count = 0
	for i in range(n - 1, -1, -1):
		for j in range(m - 1, -1, -1):
			if mat[i][j] == '.':
				mat[i][j] = 0
			elif j == 0 or j == m - 1 or i == n - 1:
				mat[i][j] = 1
			else:
				min_val = min(mat[i + 1][j - 1], mat[i + 1][j], mat[i + 1][j + 1])
				mat[i][j] = min_val + 1
			count += mat[i][j]
	return count
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	mat = []
	for i in range(n):
		mat.append(list(input().strip()))
	print(find(mat))
