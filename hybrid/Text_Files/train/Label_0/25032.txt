long long read() {
  long long d, f = 1;
  char c;
  while (!isdigit(c = getchar()))
    if (c == '-') f = -1;
  d = c ^ '0';
  while (isdigit(c = getchar())) d = (d * 10) + (c ^ '0');
  return d * f;
}
const int N = 1e5 + 5;
int a[N], last[N], nxt[N], tot, flag[N * 12], stamp, vis[N];
long long fenw[N], ans[N * 12];
struct node {
  int v, a, b, id;
  node(int _v = 0, int _a = 0, int _b = 0, int _id = 0)
      : v(_v), a(_a), b(_b), id(_id) {}
};
node op[N * 12], temp[N * 12];
set<int> pos[N];
void modify(int k, int v) {
  if (v == a[k]) return;
  auto it = pos[a[k]].find(k), lit = it, rit = it;
  if (it != pos[a[k]].end()) {
    if (it != pos[a[k]].begin()) {
      --lit;
      op[++tot] = node(-(k - last[k]), last[k], k, 0);
    }
    if (*it != *pos[a[k]].rbegin()) {
      ++rit;
      op[++tot] = node(-(nxt[k] - k), k, nxt[k], 0);
    }
    if (it != pos[a[k]].begin() && *it != *pos[a[k]].rbegin()) {
      op[++tot] = node(*rit - *lit, *lit, *rit, 0);
      nxt[*lit] = *rit;
      last[*rit] = *lit;
    }
    pos[a[k]].erase(it);
  }
  a[k] = v;
  pos[a[k]].insert(k);
  it = pos[a[k]].find(k), lit = it, rit = it;
  if (it != pos[a[k]].begin()) {
    --lit;
    nxt[*lit] = *it;
    last[*it] = *lit;
    op[++tot] = node((k - last[k]), last[k], k, 0);
  }
  if (*it != *pos[a[k]].rbegin()) {
    ++rit;
    last[*rit] = *it;
    nxt[*it] = *rit;
    op[++tot] = node((nxt[k] - k), k, nxt[k], 0);
  }
  if (it != pos[a[k]].begin() && *it != *pos[a[k]].rbegin())
    op[++tot] = node(-(*rit - *lit), *lit, *rit, 0);
}
void add(int n, int k, int v) {
  while (k <= n) {
    if (vis[k] != stamp) {
      vis[k] = stamp;
      fenw[k] = v;
    } else
      fenw[k] += v;
    k += k & -k;
  }
}
long long sum(int k) {
  long long ans = 0;
  while (k) {
    if (vis[k] == stamp) ans += fenw[k];
    k -= k & -k;
  }
  return ans;
}
void merge_sort(int n, int l, int r) {
  if (l == r) return;
  int m = (l + r) >> 1;
  merge_sort(n, l, m);
  merge_sort(n, m + 1, r);
  ++stamp;
  for (int i = l, j = l, k = m + 1; i <= r; ++i) {
    if (k > r || (j <= m && op[j].b <= op[k].b)) {
      temp[i] = op[j++];
      if (!temp[i].id) add(n, n - temp[i].a + 1, temp[i].v);
    } else {
      temp[i] = op[k++];
      if (temp[i].id) ans[temp[i].id] += sum(n - temp[i].a + 1);
    }
  }
  for (int i = (l); i <= (r); ++i) op[i] = temp[i];
}
int main() {
  int n, m;
  while (cin >> n >> m) {
    tot = 0;
    for (int i = (1); i <= (n); ++i) {
      int v;
      scanf("%d", &v);
      modify(i, v);
    }
    for (int i = (1); i <= (m); ++i) {
      int type, a, b;
      scanf("%d%d%d", &type, &a, &b);
      if (type == 1)
        modify(a, b);
      else {
        ++tot, op[tot] = node(0, a, b, tot);
        flag[tot] = 1;
      }
    }
    stamp = 0;
    merge_sort(n, 1, tot);
    for (int i = (1); i <= (tot); ++i)
      if (flag[i]) printf("%lld\n", ans[i]);
  }
  return 0;
}
