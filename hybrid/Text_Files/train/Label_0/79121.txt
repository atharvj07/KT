from math import *
sa = lambda : input()

sd = lambda : list(map(int, input().split()))
se = lambda : float(input())
sf = lambda : list(input())

def hnbhai():
	(n, r) = sd()
	ans = radians(180 / n)
	ans = sin(ans)
	print(r * ans / (1 - ans))
for _ in range(1):
	hnbhai()
