#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 10;
double dat[MAXN], x, y, kx;
int n, k;
double dist(int pos) { return hypot(x - dat[pos], y); }
double cal1(int l, int r) { return dat[r] - dat[l] + min(dist(l), dist(r)); }
double cal2(int l, int r) {
  return dat[r] - dat[l] +
         min(dist(l) + fabs(kx - dat[r]), dist(r) + fabs(kx - dat[l]));
}
int main() {
  scanf("%d%d", &n, &k);
  for (int i = 1; i <= n; i++) scanf("%lf", &dat[i]);
  scanf("%lf%lf", &x, &y);
  kx = dat[k];
  sort(dat + 1, dat + n + 1);
  double res;
  if (k == n + 1)
    res = cal1(1, n);
  else {
    res = cal2(1, n);
    for (int i = 2; i <= n; i++)
      res = min(res,
                min(cal1(1, i - 1) + cal2(i, n), cal2(1, i - 1) + cal1(i, n)));
  }
  printf("%.10f", res);
  return 0;
}
