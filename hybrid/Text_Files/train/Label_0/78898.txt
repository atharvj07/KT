a = []
n = int(input())
c = []
f = 0
z1 = []
j = -1
for _ in ' ' * n:
	for i in map(int, input().split()):
		j += 1
		z1 += [i]
		if i != 1:
			c += [(f, j, i)]
	f += 1
	a += [z1]
	z1 = []
	j = -1
for _ in c:
	if not any((a[_[0]][i] + a[j][_[1]] == _[2] for i in range(n) for j in range(n))):
		print('NO')
		exit()
print('yes')
