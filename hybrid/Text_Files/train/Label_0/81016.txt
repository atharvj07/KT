t = int(input())
while t > 0:
	(n, s) = map(int, input().split())
	A = list(map(int, input().split()))
	c_max = -1
	c_ind = 0
	summ = 0
	if sum(A) <= s:
		print(0)
	else:
		for i in range(0, n):
			summ += A[i]
			if A[i] > c_max:
				c_max = A[i]
				c_ind = i
			if summ > s:
				break
		print(c_ind + 1)
	t = t - 1
