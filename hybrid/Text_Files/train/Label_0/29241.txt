#include <bits/stdc++.h>
using namespace std;
const int dx[] = {0, 0, -1, 1, -1, 1, 1, -1};
const int dy[] = {-1, 1, 0, 0, -1, -1, 1, 1};
const int N = 1e5 + 50;
const long long mod = 1e9 + 7;
int n, u, v, ans;
vector<int> adj[N];
vector<int> vtx(N, 0);
bool vis[N];
int dfs(int node) {
  vis[node] = true;
  for (auto i : adj[node])
    if (!vis[i]) vtx[node] += (dfs(i) + 1);
  if (vtx[node] && (vtx[node] + 1) % 2 == 0 && node != 1) ans++;
  return vtx[node];
}
int main() {
  cin >> n;
  if (n % 2) return puts("-1");
  n--;
  for (int i = 0; i < n; ++i) {
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1);
  cout << ans << "\n";
  return 0;
}
