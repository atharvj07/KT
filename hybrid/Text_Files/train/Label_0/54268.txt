import java.util.regex.Matcher;  
import java.util.regex.Pattern;  
import java.util.Scanner;

public class Main{
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int kol=0;
        for (int i=1;i<=n;i++){
            if(x%i==0 && x/i<=n)
                kol++;
        }
        System.out.println(kol);
    }
}
