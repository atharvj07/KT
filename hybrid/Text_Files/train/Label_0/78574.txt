from collections import Counter

def solve(n, k, cur, cur_set):
	if len(cur) > len(str(n)) or len(cur_set) > k or cur < str(n)[:len(cur)]:
		return None
	if len(cur) == len(str(n)) and cur >= str(n):
		return cur
	for i in range(10):
		if i not in cur_set:
			cur_set.add(i)
			ans = solve(n, k, cur + str(i), cur_set)
			cur_set ^= {i}
		else:
			ans = solve(n, k, cur + str(i), cur_set)
		if ans != None:
			return ans
t = int(input())
for _ in range(t):
	(n, k) = map(int, input().split())
	print(solve(n, k, '', set()))
