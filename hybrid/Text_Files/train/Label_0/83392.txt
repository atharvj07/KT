n = int(input())
dirs = []
strdir = {'West': 0, 'North': 1, 'East': 2, 'South': 3}
for x in range(0, n):
	line = input().split()
	dirs.append([strdir[line[1]], int(line[0])])
flag = True
cy = 0
for dir in dirs:
	if (cy == 0 or cy == 20000) and (dir[0] == 0 or dir[0] == 2):
		flag = False
		break
	if dir[0] == 1:
		cy = cy - dir[1]
	elif dir[0] == 3:
		cy = cy + dir[1]
	if cy < 0 or cy > 20000:
		flag = False
		break
if flag == True and cy != 0:
	flag = False
if flag:
	print('YES')
else:
	print('NO')
