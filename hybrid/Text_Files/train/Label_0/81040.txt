def magic(a, b, x):
	if x > b:
		return False
	if b % a == 0:
		return x % a == 0
	if a < x:
		return (b - x) % a == 0
	r = b % a
	return magic(r, a, x)
total = int(input())
for i in range(total):
	line = input().split()
	a = int(line[0])
	b = int(line[1])
	x = int(line[2])
	if a <= b:
		wow = magic(a, b, x)
	else:
		wow = magic(b, a, x)
	if wow:
		print('YES')
	else:
		print('NO')
