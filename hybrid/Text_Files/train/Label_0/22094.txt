#include <bits/stdc++.h>
int main() {
  long long T, a, b, m;
  scanf("%lld", &T);
  while (T--) {
    scanf("%lld%lld%lld", &a, &b, &m);
    if (a == b) {
      printf("1 %lld\n", a);
      continue;
    }
    int n = -1;
    for (int i = 2; i <= 50; i++) {
      long long tmp = 1LL << (i - 2);
      if (b <= tmp * (a + m)) {
        if (tmp * (a + 1) <= b) n = i;
        break;
      }
    }
    if (n == -1) {
      puts("-1");
      continue;
    }
    long long k = (b >> (n - 2)) - a;
    printf("%d %lld", n, a);
    long long s = a;
    for (int i = n - 3; ~i; i--) {
      long long r = k + ((b >> i) & 1);
      printf(" %lld", s + r);
      s = (s << 1) + r;
    }
    printf(" %lld\n", b);
  }
  return 0;
}
