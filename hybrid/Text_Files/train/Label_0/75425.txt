mishika = 0
chris = 0
for roll in range(int(input())):
	(m, c) = [int(a) for a in input().split()]
	if m > c:
		mishika += 1
	elif m < c:
		chris += 1
	else:
		chris += 1
		mishika += 1
if mishika > chris:
	print('Mishka')
elif mishika < chris:
	print('Chris')
else:
	print('Friendship is magic!^^')
