bool primes[N + 1];

void sieve()
{
    memset(primes, true, sizeof(primes));
    primes[0] = primes[1] = false;
    for (int i = 2; i * i <= N; i++){
        if (primes[i]){
            for (int j = i * i; j <= N; j += i) primes[j] = false;
        }
    }
}

void moveOneStep(ll& n, ll& d, ll q)
{
    ll next = n - d * q;
    n = d;
    d = next;
}

bool isPrime(ll n)
{
    if (n <= 1) return false;
    if (n == 2) return true;
    fe (i, 2, sqrt(n)) if (n % i == 0) return false;
    return true;
}

bool palindrome(const string& s)
{
    string reversedS = s;
    reverse(all(reversedS));
    if (s == reversedS) return true;
    else return false;
}

ll fastPow(ll base, ll power)
{
    if (!power) return 1;
    if (power == 1) return base;
    ll val = fastPow(base, power / 2);
    return (power % 2 == 0) ? val * val : val * val * base;
}

ll modPow(ll base, ll power, ll M)
{
    if (power == 1) return base;
    ll val = modPow(base, power / 2, M);
    return (power % 2 == 0) ? (val * val) % M : (((val * val) % M) * base) % M;
}

ll gcd(ll a, ll b)
{
    if (!b) return a;
    return gcd(b, a % b);
}

ll lcm(ll a, ll b)
{
    return a * b / gcd(a, b);
}

ll eGCD(ll r0, ll r1, ll& x0, ll& y0)
{
    ll x1, y1;
    x0 = y1 = 1;
    x1 = y0 = 0;
    while (r1) {
        ll q = r0 / r1;
        moveOneStep(x0, x1, q);
        moveOneStep(y0, y1, q);
        moveOneStep(r0, r1, q);
    }
    return r0;
}

ll modInv(ll c, ll M)
{
    ll x, y;
    ll g = eGCD(c, M, x, y);
    if (g != 1) return -1;
    return (x + M) % M;
}

ll modInvFer(ll c, ll m)
{
    return modPow(c, m - 2, m);
}

ll npr(ll n, ll r)
{
    ll res = 1; fe (i, n - r + 1, n) res *= i;
    return res;
}

ll ncr(ll n, ll r)
{
    ll res = 1; ll rFact = 1;
    fe (i, n - r + 1, n){
        res *= i;
        if (res % rFact == 0 && rFact <= r) res /= rFact, rFact++;
    }
    return res;
}

sll getDivisors(ll val)
{
    sll st;
    fe (i, 1, sqrt(val) + 7) if (val % i == 0) st.insert(i), st.insert(val / i);
    return st;
}
