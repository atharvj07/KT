#include <bits/stdc++.h>
double const PI = 4 * atan(1.0);
using namespace std;
int a[5];
bool toan1() {
  for (int i = 1; i <= 4; i++)
    if (a[i] != 1) return false;
  return true;
}
bool doi2() {
  for (int i = 1; i <= 4; i++)
    if (a[i] % 2 == 0 && a[i % 4 + 1] % 2 == 0) {
      a[i] /= 2;
      a[i % 4 + 1] /= 2;
      printf("/%d\n", i);
      return true;
    }
  return false;
}
bool doile() {
  for (int i = 1; i <= 4; i++)
    if (a[i] % 2 != 0 && a[i % 4 + 1] % 2 != 0 && a[i] + a[i % 4 + 1] > 2) {
      a[i]++;
      a[i % 4 + 1]++;
      printf("+%d\n", i);
      return true;
    }
  return false;
}
void tangkhac1() {
  for (int i = 1; i <= 4; i++)
    if (a[i] > 1) {
      a[i]++;
      a[i % 4 + 1]++;
      printf("+%d\n", i);
      return;
    }
}
int main() {
  for (int i = 1; i <= 4; i++) scanf("%d", &a[i]);
  while (!toan1()) {
    if (doi2())
      ;
    else if (doile())
      ;
    else
      tangkhac1();
  }
}
