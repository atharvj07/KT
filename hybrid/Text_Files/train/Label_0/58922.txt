import java.io.IOException;
import java.io.PrintWriter;
import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class ProblemC {
	
	
	public static void main(String[] args) throws IOException {
		Scanner s = new Scanner(System.in);
		PrintWriter out = new PrintWriter(System.out);
		
		int n = s.nextInt();
		long[] a = new long[n];
		for (int i = 0 ; i < n ; i++) {
			a[i] = s.nextInt();
		}
		
		long[] sum = new long[n+1];
		for (int i = 0 ; i < n ; i++) {
			sum[i+1] = sum[i] + a[i];
		}
		
		long[] dp_prefix = new long[n+1];
		Arrays.fill(dp_prefix, Long.MIN_VALUE);
		dp_prefix[0] = 0;
		for (int i = 1 ; i <= n ; i++) {
			dp_prefix[i] = Math.max(dp_prefix[i-1] + a[i-1], -sum[i]);
		}
		
		
		long[] dp_suffix = new long[n+1];
		Arrays.fill(dp_suffix, Long.MIN_VALUE);
		dp_suffix[n] = 0;
		for (int i = n-1 ; i >= 0 ; i--) {
			dp_suffix[i] = Math.max(dp_suffix[i+1] + a[i], -(sum[n] - sum[i]));
		}
		
		// debug(dp_prefix);
		// debug(dp_suffix);
		
		long ans = Long.MIN_VALUE;
		for (int i = 0 ; i <= n ; i++) {
			ans = Math.max(ans, dp_prefix[i] + dp_suffix[i]);
		}
		out.println(ans);
		out.flush();
	}


	public static void debug(Object... os){
		System.err.println(Arrays.deepToString(os));
	}
}
