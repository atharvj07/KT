import java.io.*;
import java.util.*;

public class p426b
{
    public static void main(String[] args)
    {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int a[][] = new int[n][m];
        String str[] = new String[n];
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                a[i][j] = sc.nextInt();
            }
        }
        if(n%2!=0) System.out.println(n);
        else
        {
            for(int i=0;i<n;i++)
            {
                str[i] = "";
                for(int j=0;j<m;j++)
                {
                    str[i] += Integer.toString(a[i][j]);
                }
            }
            int result = n;
            P:
            for (int i=n;i>0 && i%2==0;i=i/2)
            {
                for(int j=0;j<=i/2-1;j++)
                {
                    if(str[j].equals(str[i-j-1])) continue;
                    else break P;
                }
                result = i/2;
            }
            System.out.println(result);
        }
    }
}
