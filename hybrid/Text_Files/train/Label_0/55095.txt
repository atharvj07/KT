import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class E {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);

        char[] source = s.next().toCharArray();
        char[] target = s.next().toCharArray();
        int n = s.nextInt();
        List<List<Integer>> tasks = IntStream.range(0, n).boxed().map(x -> {
            ArrayList<Integer> abcd = new ArrayList<>(4);
            abcd.add(s.nextInt());
            abcd.add(s.nextInt());
            abcd.add(s.nextInt());
            abcd.add(s.nextInt());
            return abcd;
        }).collect(Collectors.toList());

        int[] sourceCounts = new int[source.length + 1];
        sourceCounts[0] = 0;
        for (int i = 0; i < source.length; i++) sourceCounts[i+1] = sourceCounts[i] + ((source[i] == 'A') ? 0 : 1);

        int[] targetCounts = new int[target.length + 1];
        targetCounts[0] = 0;
        for (int i = 0; i < target.length; i++) targetCounts[i+1] = targetCounts[i] + ((target[i] == 'A') ? 0 : 1);

        for (List<Integer> task : tasks) {
            int sbc = sourceCounts[task.get(1)] - sourceCounts[task.get(0) - 1];
            int tbc = targetCounts[task.get(3)] - targetCounts[task.get(2) - 1];
            int sa = trailingACount(sourceCounts, task.get(0) - 1, task.get(1));
            int ta = trailingACount(targetCounts, task.get(2) - 1, task.get(3));
            int diffA = ((sa - ta) % 3 == 0) ? 0 : 1;
            boolean result = sa >= ta && (sbc + 2*diffA) <= tbc && ((tbc - sbc) % 2 == 0);
            System.out.print(result ? 1 : 0);
        }
    }

    static int trailingACount(int[] counts, int l, int r) {
        int origR = r;
        while (r - l > 1) {
            int c = (r+l)/2;
            if (counts[c] == counts[origR]) r = c;
            else l = c;
        }
        return origR - l;
    }
}
