n = int(input())
for b in range(n):
	already_answered = False
	y = input().split(' ')
	arr = [int(x) for x in y]
	matches = [[arr[0], arr[1]], [arr[2], arr[3]]]
	top = max(arr)
	arr.remove(max(arr))
	second = max(arr)
	for c in matches:
		if top in c and second in c:
			print('NO')
			already_answered = True
	if not already_answered:
		print('YES')
