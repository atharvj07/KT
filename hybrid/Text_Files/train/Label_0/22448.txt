#include <bits/stdc++.h>
#pragma GCC target("sse4.2")
using namespace std;
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
const long long mod = int(1e9) + 7;
const int INF = 1e9;
const long long LINF = 2e18;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  cout << setprecision(18) << fixed;
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto& x : a) cin >> x;
  vector<array<int, 3>> res;
  if (n % 2 == 0) {
    int x = a[0];
    for (int i = 1; i < n; i++) {
      x ^= a[i];
    }
    if (x != 0) {
      cout << "NO\n";
      return 0;
    }
    n--;
  }
  for (int i = 0; i + 2 < n; i += 2) {
    int cur = a[i] ^ a[i + 1] ^ a[i + 2];
    a[i] = a[i + 1] = a[i + 2] = cur;
    res.push_back({i + 1, i + 2, i + 3});
  }
  for (int i = 0; i + 1 < n; i += 2) {
    int cur = a[i] ^ a[i + 1] ^ a[n - 1];
    a[i] = a[i + 1] = a[n - 1] = cur;
    res.push_back({i + 1, i + 2, n});
  }
  cout << "YES\n";
  cout << (int)(res).size() << "\n";
  for (auto& x : res) cout << x[0] << " " << x[1] << " " << x[2] << "\n";
  return 0;
}
