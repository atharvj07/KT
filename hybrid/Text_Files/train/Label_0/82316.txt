import sys
input = sys.stdin.readline
from collections import Counter
T = int(input())
for _ in range(T):
	n = int(input())
	a = list(map(int, input().split()))
	cnt = Counter(a)
	l = []
	flag = False
	for (k, v) in cnt.items():
		if v >= 4:
			print(k, k, k, k)
			flag = True
			break
		elif v >= 2:
			l.append(k)
	if flag:
		continue
	l.sort()
	for i in range(len(l) - 1):
		(a, b) = (l[i], l[i + 1])
		if i == 0:
			(ans1, ans2) = (a, b)
		elif (ans1 ** 2 + ans2 ** 2) * a * b > (a ** 2 + b ** 2) * ans1 * ans2:
			(ans1, ans2) = (a, b)
	print(ans1, ans1, ans2, ans2)
