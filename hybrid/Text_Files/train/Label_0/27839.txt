#include <iostream>
using namespace std;
int A[101];
int N,M;
int D;

int main() {
	cin>>N>>M;
	for(int i=0;i<N;i++){
		cin>>A[i];
	}

	for(int j=1;j<M+1;j++){
		for(int i=0;i<N-1;i++){
			if(A[i]%j>A[i+1]%j){
				D=A[i];
				A[i]=A[i+1];
				A[i+1]=D;
			}
		}
	}


	for(int i=0;i<N;i++){
		cout<<A[i]<<endl;
	}
	return 0;
}
