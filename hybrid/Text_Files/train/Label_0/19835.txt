#include<iostream>
#include<cstdio>

using namespace std;

int main(void){
	int s[5];
	int maxs,mins;
	int n;
	int a;
	while(1){
		cin >> n;
		if(n==0)break;
		cin >> s[0] >> s[1] >> s[2] >> s[3] >> s[4];
		maxs=s[0]+s[1]+s[2]+s[3]+s[4];
		mins=s[0]+s[1]+s[2]+s[3]+s[4];
		for(a=1;a<n;a++){
			cin >> s[0] >> s[1] >> s[2] >> s[3] >> s[4];
			maxs=max(maxs,s[0]+s[1]+s[2]+s[3]+s[4]);
			mins=min(mins,s[0]+s[1]+s[2]+s[3]+s[4]);
		}
		cout << maxs << ' ' << mins << endl;
	}
	return 0;
}
