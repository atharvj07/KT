#include<cstdio>
#include<cstdlib>
#include<algorithm>

#define rep(i,n) for(int i=0;i<(n);i++)

using namespace std;

typedef long long ll;

const ll INF=1LL<<61;

// x0 に降りて、最後に x_last まで行くときの x 方向の総コスト
ll f(int x0,int x_last,int n,const int *x,const ll *x_sum){
	int p=lower_bound(x,x+n,x0)-x;
	ll res=0;
	res+=(ll)p*x0-x_sum[p];                // x0 より左にある家のコスト
	res+=(x_sum[n]-x_sum[p])-(ll)(n-p)*x0; // x0 より右にある家のコスト
	return 2*res-abs(x_last-x0);
}

void solve(int n,const int *x,int *x0,ll *ans){
	static int x_sorted[100000];
	rep(i,n) x_sorted[i]=x[i];
	sort(x_sorted,x_sorted+n);

	static ll x_sum[100001];
	rep(i,n) x_sum[i+1]=x_sum[i]+x_sorted[i];

	rep(i,n){
		int lo=x_sorted[0],hi=x_sorted[n-1];
		while(lo<hi){
			int mi1=(lo+hi)/2,mi2=mi1+1;
			if(f(mi1,x[i],n,x_sorted,x_sum)>f(mi2,x[i],n,x_sorted,x_sum)) lo=mi1+1; else hi=mi2-1;
		}
		x0[i]=lo;
		ans[i]=f(lo,x[i],n,x_sorted,x_sum);
	}
}

int main(){
	int n; scanf("%*d%*d%d",&n);
	static int x[100000],y[100000];
	rep(i,n) scanf("%d%d",x+i,y+i);

	// x0[i] := ( 最後に家 i を訪れるとしたときの、最適な降りる位置の x 座標 )
	// ans_x[i] := ( 最後に家 i を訪れるとして、x0[i] に降りたときの x 方向の総コスト )
	static int x0[100000],y0[100000];
	static ll ans_x[100000],ans_y[100000];
	solve(n,x,x0,ans_x);
	solve(n,y,y0,ans_y);

	static ll ans[100000];
	rep(i,n) ans[i]=ans_x[i]+ans_y[i];

	int i_ans=0;
	rep(i,n){
		if(ans[i_ans]>ans[i]
		|| ans[i_ans]==ans[i] && x0[i_ans]>x0[i]
		|| ans[i_ans]==ans[i] && x0[i_ans]==x0[i] && y0[i_ans]>y0[i]) i_ans=i;
	}
	printf("%lld\n%d %d\n",ans[i_ans],x0[i_ans],y0[i_ans]);

	return 0;
}
