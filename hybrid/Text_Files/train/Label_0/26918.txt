#include <bits/stdc++.h>
using namespace std;
map<double, int> mp;
double a[200500];
int main() {
  int m;
  cin >> m;
  for (int i = 0; i < m; i++) {
    char c = getchar();
    char c1, c2, c3, c4;
    int x, y, z;
    scanf("%c%d%c%d%c%c%d", &c1, &x, &c2, &y, &c3, &c4, &z);
    double d1 = double(x + y);
    double d2 = double(z);
    double tmp = double(d1 / d2);
    a[i] = tmp;
    mp[tmp]++;
  }
  for (int i = 0; i < m; i++) {
    cout << mp[a[i]] << ' ';
  }
  return 0;
}
