#include <bits/stdc++.h>
using namespace std;
int main() {
  int i, j, n, k, right, left;
  string s;
  scanf("%d%d", &n, &k);
  cin >> s;
  if (n == 1) {
    cout << "PRINT " << s[0] << '\n';
    return 0;
  }
  right = n - k;
  left = k - 1;
  if (right < left) {
    for (i = k; i <= n; i++) {
      cout << "PRINT " << s[i - 1] << '\n';
      if (i != n) cout << "RIGHT" << '\n';
    }
    for (i = n; i >= k; i--) cout << "LEFT" << '\n';
    for (i = k - 1; i > 0; i--) {
      cout << "PRINT " << s[i - 1] << '\n';
      if (i != 1) cout << "LEFT" << '\n';
    }
  } else {
    for (i = k; i > 0; i--) {
      cout << "PRINT " << s[i - 1] << '\n';
      if (i != 1) cout << "LEFT" << '\n';
    }
    for (i = 1; i <= k; i++) cout << "RIGHT" << '\n';
    for (i = k + 1; i <= n; i++) {
      cout << "PRINT " << s[i - 1] << '\n';
      if (i != n) cout << "RIGHT" << '\n';
    }
  }
  printf("");
  return 0;
}
