#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
int read()
{
	int out = 0, sgn = 1;
	char jp = getchar();
	while (jp != '-' && (jp < '0' || jp > '9'))
		jp = getchar();
	if (jp == '-')
		sgn = -1, jp = getchar();
	while (jp >= '0' && jp <= '9')
		out = out * 10 + jp - '0', jp = getchar();
	return out * sgn;
}
const int N = 2e3 + 10, inf = 1e9;
vector<int> G[N];
int n, ans = inf, a[N], tot[N], cnt[N];
void dfs(int u, int fa)
{
	cnt[u] = a[u], tot[u] = 0;
	for (int v : G[u])
		if (v != fa)
		{
			dfs(v, u);
			cnt[u] += cnt[v];
			tot[u] += tot[v] + cnt[v];
		}
}
int calc(int u, int fa)
{
	int mx = 0;
	for (int v : G[u])
		if (v != fa)
			if (tot[v] >= tot[mx])
				mx = v;
	if (mx == 0)
		return 0;
	int k = calc(mx, u) + cnt[mx];
	if (tot[u] - tot[mx] - cnt[mx] >= k)
		return tot[u] & 1;
	return k - (tot[u] - tot[mx] - cnt[mx]);
}
char buf[N];
int main()
{
	n = read();
	scanf("%s", buf + 1);
	for (int i = 1; i <= n; ++i)
		if (buf[i] == '1')
			a[i] = 1;
	for (int i = 1; i < n; ++i)
	{
		int u = read(), v = read();
		G[u].push_back(v);
		G[v].push_back(u);
	}
	for (int rt = 1; rt <= n; ++rt)
	{
		dfs(rt, 0);
		if (calc(rt, 0) == 0)
			ans = min(ans, tot[rt] >> 1);
	}
	printf("%d\n", ans < inf ? ans : -1);
	return 0;
}
