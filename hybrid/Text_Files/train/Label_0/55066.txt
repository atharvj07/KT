import java.io.*;
import java.util.*;

public class Main {

    static FastReader in;
    static PrintWriter out;
    static Random rand = new Random();
    static final int oo = (int) 1e9 + 10;
    static final long OO = (long) 2e18 + 10;
    static final int MOD = (int) 1e9 + 7;
    static final int N = (int) 2e5 + 10;

    static int n, m;
    static ArrayList<Integer>[] g;
    static HashSet<Integer>[] g1;

    static void dfs0(int v, boolean[] vis) {
        vis[v] = true;
        for (int u : g1[v]) {
            if (!vis[u])
                dfs0(u, vis);
        }
    }

    static void dfs(int v, ArrayList<Integer> path) {
        ArrayList<Integer> adj = new ArrayList<>(g1[v]);
        for (int u : adj) {
            if (!g1[v].contains(u))
                continue;
            g1[v].remove(u);
            g1[u].remove(v);
            dfs(u, path);
        }
        path.add(v);
    }

    static ArrayList<Integer> findEulerPath(int start) {
        ArrayList<Integer> path = new ArrayList<>();
        boolean[] vis = new boolean[n];
        dfs0(start, vis);
        for (int v = 0; v < n; v++) {
            if (g1[v].size() > 0 && !vis[v])
                return path;
        }
        int cntOdd = 0;
        for (int v = 0; v < n; v++) {
            if (g1[v].size() % 2 == 1)
                cntOdd++;
        }
        if (cntOdd > 2)
            return path;
        dfs(start, path);
        return path;
    }

    static void answer(ArrayList<Integer> path, HashSet<Integer> star) {
        int c = path.get(path.size() - 1);
        ArrayList<Integer> ans = new ArrayList<>(path);
        if (!star.isEmpty())
            ans.add(-1);
        for (int v : star) {
            ans.add(v);
            ans.add(c);
        }
        out.println(ans.size());
        for (int v : ans) {
            out.print((v == -1 ? -1 : (v + 1)) + " ");
        }
        out.println();
    }

    static void solve() {
        n = in.nextInt();
        m = in.nextInt();
        g = new ArrayList[n]; Arrays.setAll(g, i -> new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int v = in.nextInt() - 1;
            int u = in.nextInt() - 1;
            g[v].add(u);
            g[u].add(v);
        }

        g1 = new HashSet[n]; Arrays.setAll(g1, i -> new HashSet<>());
        for (int c = 0; c < n; c++) {
            boolean[] adj = new boolean[n];
            adj[c] = true;
            for (int v : g[c]) {
                adj[v] = true;
            }
            int cntOdd = 0;
            for (int v = 0; v < n; v++) {
                if (!adj[v] && g[v].size() % 2 == 1)
                    cntOdd++;
            }
            if (cntOdd > 1)
                continue;
            {
                HashSet<Integer> star = new HashSet<>();
                for (int v = 0; v < n; v++) {
                    g1[v].clear();
                    for (int u : g[v]) {
                        if ((v == c && g[u].size() % 2 == 1) || (u == c && g[v].size() % 2 == 1)) {
                            if (v == c)
                                star.add(u);
                            continue;
                        }
                        g1[v].add(u);
                    }
                }
                ArrayList<Integer> path = findEulerPath(c);
                if (!path.isEmpty()) {
                    if (path.get(path.size() - 1) != c)
                        Collections.reverse(path);
                    answer(path, star);
                    return;
                }
            }
            if (cntOdd == 1)
                continue;
            for (int start : g[c]) {
                HashSet<Integer> star = new HashSet<>();
                for (int v = 0; v < n; v++) {
                    g1[v].clear();
                    for (int u : g[v]) {
                        if ((v == c && g[u].size() % 2 == 1) || (u == c && g[v].size() % 2 == 1)) {
                            if (v == c)
                                star.add(u);
                            continue;
                        }
                        g1[v].add(u);
                    }
                }
                if (g[start].size() % 2 == 1) {
                    g1[c].add(start);
                    g1[start].add(c);
                    star.remove(start);
                } else {
                    g1[c].remove(start);
                    g1[start].remove(c);
                    star.add(start);
                }
                ArrayList<Integer> path = findEulerPath(c);
                if (!path.isEmpty()) {
                    if (path.get(path.size() - 1) != c)
                        Collections.reverse(path);
                    answer(path, star);
                    return;
                }
            }
        }
        out.println(0);

    }

    public static void main(String[] args) {
        in = new FastReader();
        out = new PrintWriter(System.out);

        int T = 1;
//        T = in.nextInt();
        while (T-- > 0)
            solve();

        out.flush();
        out.close();
    }

    static class FastReader {
        BufferedReader br;
        StringTokenizer st;

        FastReader() {
            this(System.in);
        }
        FastReader(String file) throws FileNotFoundException {
            this(new FileInputStream(file));
        }
        FastReader(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        int nextInt() {
            return Integer.parseInt(next());
        }
        long nextLong() {
            return Long.parseLong(next());
        }
        double nextDouble() {
            return Double.parseDouble(next());
        }
        String next() {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(nextLine());
            }
            return st.nextToken();
        }
        String nextLine() {
            String line;
            try {
                line = br.readLine();
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
            return line;
        }
    }
}
