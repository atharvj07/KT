#include <bits/stdc++.h>
using namespace std;
const int N = 410000;
long long a[N], s[N];
int n, k, b, c, ok;
long long sum;
priority_queue<long long> h;
void add(long long x) {
  if (h.size() < k) {
    h.push(-x);
    sum += x;
    ok |= h.size() == k;
  } else {
    long long mi = -(h.top());
    if (x <= mi)
      return;
    else {
      sum = sum - mi + x;
      h.pop();
      h.push(-x);
    }
  }
}
long long ans;
void getans(int cen) {
  while (!h.empty()) h.pop();
  sum = 0;
  ok = 0;
  for (int i = 1; i <= n; i++) {
    long long val = 0;
    if (a[i] % 5 <= cen)
      val = a[i] / 5 * b + a[i] % 5 * c;
    else
      val = (a[i] / 5 + 1) * b - (5 - a[i] % 5) * c;
    add(val);
    for (int j = 0; j <= 4; j++)
      if ((a[i] + j) % 5 == cen) {
        long long now = a[i] + j;
        long long val = now / 5 * b + now % 5 * c;
        if (ok) ans = min(ans, val * k - sum);
      }
  }
}
int main() {
  ans = 1e18;
  scanf("%d%d%d%d", &n, &k, &b, &c);
  int mi = 0;
  for (int i = 1; i <= n; i++) {
    int x;
    scanf("%d", &x);
    mi = min(mi, x);
    a[i] = x;
  }
  for (int i = 1; i <= n; i++) a[i] -= mi;
  sort(a + 1, a + 1 + n);
  for (int i = 0; i <= 4; i++) getans(i);
  for (int i = 1; i <= n; i++) s[i] = s[i - 1] + a[i];
  for (int i = k; i <= n; i++)
    ans = min(ans, (a[i] * k - (s[i] - s[i - k])) * c);
  cout << ans;
}
