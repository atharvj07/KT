(n, m) = map(int, input().split())
q = min(n, m)
w = max(n, m)
if q == 1 and w <= 3 or q == 2 == w:
	print('0')
elif q == 1 and w == 4:
	print('2')
elif q == 1 and w == 5:
	print('4')
elif q == 1 and w == 6:
	print('6')
elif q == 2 and w == 3:
	print('4')
elif q == 1:
	r = w // 6
	if w % 6 == 0:
		print(w)
	elif w % 6 <= 3:
		print(r * 6)
	elif w % 6 == 4:
		print(r * 6 + 2)
	elif w % 6 == 5:
		print(r * 6 + 4)
elif q == 2 and w == 7:
	print('12')
elif q * w % 2 == 0:
	print(q * w)
else:
	print(q * w - 1)
