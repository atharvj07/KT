#include <bits/stdc++.h>
int in() {
  int r = 0, c;
  for (c = getchar(); c <= 32; c = getchar())
    ;
  if (c == '-') return -in();
  for (; c > 32; r = (r << 1) + (r << 3) + c - '0', c = getchar())
    ;
  return r;
}
using namespace std;
const int INF = int(1e9 + 7);
double prob[20][20];
double dp[(1 << 19)][20];
int n;
double f(int mask, int sith) {
  if (mask + 1 == (1 << n) - 1) return prob[0][sith];
  if (dp[mask][sith] != -1) return dp[mask][sith];
  double bestProb = 0;
  for (int i = 1; i < n; i++) {
    if (mask & (1 << i)) continue;
    double x = 0;
    x += prob[i][sith] * f((mask | (1 << i)), i);
    x += prob[sith][i] * f((mask | (1 << i)), sith);
    bestProb = max(bestProb, x);
  }
  return dp[mask][sith] = bestProb;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      cin >> prob[i][j];
    }
  }
  if (n == 1) {
    cout << 1;
    return 0;
  }
  double res = 0;
  for (int i = 0; i < (1 << n); i++)
    for (int j = 0; j < n; j++) dp[i][j] = -1;
  for (int i = 1; i < n; i++) {
    double x = f((1 << i), i);
    res = max(res, x);
  }
  cout << fixed << setprecision(7) << res << endl;
  return 0;
}
