t = int(input())
res = [[] for i in range(10)]
for i in range(1, 10):
	for j in range(1, 10):
		k = i * j
		s = str(k)
		res[i].append([s[-1], k])
for i in range(t):
	(n, d) = map(int, input().split())
	b = list(map(int, input().split()))
	for j in range(n):
		curr = b[j]
		s = str(curr)[-1]
		f = 0
		for k in res[d]:
			if k[0] == s[0] and curr >= k[1]:
				f = 1
				break
		if f or curr >= 10 * d:
			print('YES')
		else:
			print('NO')
