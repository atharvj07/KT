n = int(input())
t = list(map(int, input().split()))
segments = []
curr = t[0]
ln = 0
for i in range(n):
	if t[i] == curr:
		ln += 1
	else:
		segments.append(ln)
		curr = t[i]
		ln = 1
segments.append(ln)
longest = 0
for i in range(0, len(segments) - 1):
	longest = max(longest, min(segments[i], segments[i + 1]))
print(longest * 2)
