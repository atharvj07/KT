import copy
n = int(input())
l = list()
a = list()

def findprime(N):
	i = 2
	count = 0
	while i * i <= N:
		if i * i == N:
			count = count + 1
		elif N % i == 0:
			count = count + 2
		i = i + 1
	if count == 0 and N != 1:
		l.append(N)
for j in range(2, n + 1):
	findprime(j)
a = copy.copy(l)
for i in l:
	k = 2
	while i ** k <= n:
		a.append(i ** k)
		k = k + 1
print(len(a))
print(*a)
