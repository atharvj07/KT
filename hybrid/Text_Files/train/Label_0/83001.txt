import sys
(k, a, b) = map(int, input().split())
next_a = [[0] * 4] + [[0] + list(map(int, input().split())) for _ in range(3)]
next_b = [[0] * 4] + [[0] + list(map(int, input().split())) for _ in range(3)]
dp = [[-1] * 4 for _ in range(4)]
dp_score = [[[0, 0] for _ in range(4)] for _ in range(4)]
score = [0, 0]
i = k
while i:
	if i < k and dp[a][b] != -1:
		m = i // (dp[a][b] - i)
		score[0] += (score[0] - dp_score[a][b][0]) * m
		score[1] += (score[1] - dp_score[a][b][1]) * m
		i %= dp[a][b] - i
	if i:
		dp[a][b] = i
		dp_score[a][b] = score[:]
		if a == 3 and b == 2 or (a == 2 and b == 1) or (a == 1 and b == 3):
			score[0] += 1
		elif b == 3 and a == 2 or (b == 2 and a == 1) or (b == 1 and a == 3):
			score[1] += 1
		(a, b) = (next_a[a][b], next_b[a][b])
		i -= 1
print(*score)
