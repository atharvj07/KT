import sys
import math
from decimal import *

def I():
	return int(sys.stdin.readline())

def IL():
	return list(map(int, sys.stdin.readline().strip().split()))

def ILS():
	return list(map(str, sys.stdin.readline().strip().split()))
tt = I()
for ii in range(tt):
	(n, m) = IL()
	mat = []
	for i in range(n):
		l = IL()
		mat.append(l)
	nm = [[0] * m for i in range(n)]
	for i in range(0, n):
		nm[i][0] = i
	for i in range(0, m):
		nm[0][i] = i
	ans = 0
	for i in range(1, n):
		for j in range(1, m):
			nm[i][j] = nm[i - 1][j] + 1
			ans = max(ans, nm[i][j])
	d = {}
	for i in range(ans + 1):
		d[i] = {}
	for i in range(n):
		for j in range(m):
			x = mat[i][j]
			y = nm[i][j]
			if x not in d[y]:
				d[y][x] = 1
			else:
				d[y][x] += 1
	cnt = 0
	for i in range((ans + 1) // 2):
		k = set(d[i].keys())
		for xx in list(d[ans - i].keys()):
			k.add(xx)
		tx = None
		kl = list(k)
		for x in kl:
			tmp = 0
			for y in kl:
				if x != y and y in d[i]:
					tmp += d[i][y]
				if x != y and y in d[ans - i]:
					tmp += d[ans - i][y]
			if tx == None:
				tx = tmp
			else:
				tx = min(tx, tmp)
		cnt += tx
	print(cnt)
