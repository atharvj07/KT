def solve(s, n):
	a = 0
	ab = 0
	abc = 0
	aTimes = 1
	for i in range(s):
		if n[i] == 'a':
			a += aTimes
			a = a % mod
		elif n[i] == 'b':
			ab += a
			ab = ab % mod
		elif n[i] == 'c':
			abc += ab
			abc = abc % mod
		elif n[i] == '?':
			abc = (3 * abc + ab) % mod
			ab = (3 * ab + a) % mod
			a = (3 * a + aTimes) % mod
			aTimes = aTimes * 3 % mod
	return abc
mod = 10 ** 9 + 7
s = int(input())
n = input()
print(solve(s, n))
