def cal(i):
	a = [x for x in m]
	t = a[i]
	for j in range(i - 1, -1, -1):
		a[j] = min(t, a[j])
		t = a[j]
	t = a[i]
	for j in range(i + 1, n):
		a[j] = min(t, a[j])
		t = a[j]
	return a
n = int(input())
m = [int(x) for x in input().split()]
res = cal(0)
for i in range(1, n):
	a = cal(i)
	if sum(res) < sum(a):
		res = a
print(*res)
