ll a[100000];
ll b[100000];

ll t[100000];

int main() {
  ll n;
  ll m,C;
  while(input(n,m,C)) {
    REP(i,n) {
      input(a[i]);
      b[i] = a[i];
    }
    reverse(b,b+n);
    ll low = -1, high = n+1;
    while(low+1<high) {
      ll mid = (low+high)/2;
      deque<pii> D;
      REP(i,n) {
        while(D.size()) {
          pii &p = D.front();
          if (p.second < i-mid) {
            D.pop_front();
          } else break;
        }
        while(D.size()) {
          pii &p = D.back();
          if (p.first - (i-p.second)*C < a[i]) {
            D.pop_back();
          } else break;
        }
        D.push_back(pii(a[i], i));
        pii &p = D.front();
        t[i] = p.first - (i-p.second)*C;
        // REP(j,D.size()) {
        //   assert(D[j].first-(i-D[j].second)*C <= t[i]);
        // }
      }
      // REP(i,n) {
      //   ll tmp = a[i];
      //   for (int j=max(0LL,i-mid); j<=i; ++j) {
      //     chmax(tmp, a[j] - abs(i-j)*C);
      //   }
      //   assert(tmp == t[i]);
      // }      
      D.clear();
      REP(i,n) {
        while(D.size()) {
          pii &p = D.front();
          if (p.second < i-mid) {
            D.pop_front();
          } else break;
        }
        while(D.size()) {
          pii &p = D.back();
          if (p.first - (i-p.second)*C < b[i]) {
            D.pop_back();
          } else break;
        }
        D.push_back(pii(b[i], i));
        pii &p = D.front();
        chmax(t[n-i-1], p.first - (i-p.second)*C);
      }
      ll sum = 0;
      REP(i,n) sum += t[i];
      
      // cout << mid << " " << sum << endl;
      // output(t,n);
      if (sum >= m) {
        high = mid;
      } else {
        low = mid;
      }
    }
    if (high == n+1) {
      cout << -1 << endl;
    } else {
      cout << high << endl;
    }
  
  }
}
