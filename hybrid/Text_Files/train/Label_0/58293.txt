import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class E implements Runnable {

    public static void main(String[] args) throws IOException {
//        System.out.println("10000 10000");
//        for (int i = 0; i < 10000; i++) {
//            System.out.println((i + 1) + " " + (((i + 1) % 10000) + 1));
//        }
        new Thread(null, new E(), "a", 10000 * 100).start();
    }

    public void run() {
        try {
            solve();
        } catch (IOException e1) {
            e1.printStackTrace();
            throw new RuntimeException(e1);
        }
    }

    StringTokenizer st = new StringTokenizer("");
    BufferedReader br;
    List<List<Edge>> e;
    Edge[] prev;
    boolean[] active;
    boolean[] visited;
    boolean[] visitedEdge;

    boolean[] even;
    int oddCycleCount;
    int[] oddCycle;

    void solve() throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        int n = nextInt();
        int m = nextInt();
        e = new ArrayList<List<Edge>>();
        for (int i = 0; i < n; i++) {
            e.add(new ArrayList<Edge>());
        }
        for (int i = 0; i < m; i++) {
            int from = nextInt() - 1;
            int to = nextInt() - 1;
            e.get(from).add(new Edge(from, to, i));
            e.get(to).add(new Edge(to, from, i));
        }
        br.close();

        visitedEdge = new boolean[m];
        visited = new boolean[n];
        active = new boolean[n];
        prev = new Edge[n];
        even = new boolean[m];
        oddCycleCount = 0;
        oddCycle = new int[m];
        Arrays.fill(oddCycle, 0);
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                rec(i);
            }
        }
        List<Integer> res = new ArrayList<Integer>();
        if (oddCycleCount > 0) {
            for (int i = 0; i < m; i++) {
                if (!even[i] && oddCycle[i] == oddCycleCount) {
                    res.add(i);
                }
            }
        } else {
            for (int i = 0; i < m; i++) {
                res.add(i);
            }
        }
        PrintWriter pw = new PrintWriter(System.out);
        pw.println(res.size());
        boolean first = true;
        for (int edge : res) {
            if (!first) {
                pw.print(" ");
            }
            first = false;
            pw.print(edge + 1);
        }
        pw.close();
    }

    void rec(final int k) {
        visited[k] = true;
        active[k] = true;
        for (final Edge edge : e.get(k)) {
            if (visitedEdge[edge.id]) {
                continue;
            }
            visitedEdge[edge.id] = true;
            if (visited[edge.to]) {
                List<Edge> cycle = new ArrayList<Edge>();
                if (active[edge.to]) {
                    Edge e = edge;
                    while (e.from != edge.to) {
                        cycle.add(e);
                        e = prev[e.from];
                    }
                    cycle.add(e);
                } else {
                    Edge e = prev[edge.to];
                    while (!active[e.from]) {
                        cycle.add(e);
                        e = prev[e.from];
                    }
                    cycle.add(e);
                    Collections.reverse(cycle);

                    int root = e.from;

                    e = edge;
                    while (e.from != root) {
                        cycle.add(e);
                        e = prev[e.from];
                    }
                    cycle.add(e);
                }
                handle(cycle);
            } else {
                prev[edge.to] = edge;
                rec(edge.to);
            }
        }
        active[k] = false;
    }

    void handle(List<Edge> cycle) {
        if (cycle.size() % 2 == 0) {
            for (Edge edge : cycle) {
                even[edge.id] = true;
            }
        } else {
            oddCycleCount++;
            for (Edge edge : cycle) {
                oddCycle[edge.id]++;
            }
        }
    }

    int nextInt() throws IOException {
        while (!st.hasMoreTokens()) {
            st = new StringTokenizer(br.readLine());
        }
        return Integer.parseInt(st.nextToken());
    }

    static class Edge {
        final int from;
        final int to;
        final int id;

        Edge(int from, int to, int id) {
            this.from = from;
            this.to = to;
            this.id = id;
        }

        @Override
        public String toString() {
            return String.valueOf(id);
        }
    }

}
