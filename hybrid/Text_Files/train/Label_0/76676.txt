import sys
stdin = sys.stdin
ns = lambda : stdin.readline().rstrip()
ni = lambda : int(stdin.readline().rstrip())
nm = lambda : map(int, stdin.readline().split())
nl = lambda : list(map(int, stdin.readline().split()))
t = ni()
for _ in range(t):
	n = ni()
	g = 2 ** n + 2 ** (n // 2) - 2
	print(g * 2 - 2 ** (n + 1) + 2)
