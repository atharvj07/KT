table = [0] * 17

def maxsum(x):
	b = bin(x)[2:]
	ans = 0
	for (i, v) in enumerate(reversed(b)):
		if v == '1':
			table[i] += 1
			for n in range(i):
				table[n] += 2 ** (i - n - 1)
	for (i, v) in enumerate(table):
		ans += 2 ** i * v
	return ans

def lowbit(x):
	b = bin(x)[2:]
	for (i, v) in enumerate(reversed(b)):
		if v == '1':
			return 2 ** i
(s, l) = [int(x) for x in input().split()]
candi = []
mm = maxsum(l)
if s > mm:
	print('-1')
else:
	flag = True
	while s:
		for i in reversed(range(len(table))):
			if table[i] > 0 and 2 ** i <= s:
				s -= 2 ** i
				table[i] -= 1
				candi.append(table[i] * 2 ** (i + 1) + 2 ** i)
				break
		else:
			flag = False
			break
	if flag:
		print(len(candi))
		for x in candi:
			print(x, end=' ')
		print()
	else:
		print('-1')
