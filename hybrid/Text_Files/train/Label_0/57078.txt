import java.io.*;
import java.util.*;


public class Cf {
	public static void main(String[] args) throws IOException {
		//BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		Scanner sc = new Scanner(System.in);
		int a = sc.nextInt();
		int b = sc.nextInt();
		int c = sc.nextInt();
		int t[] = new int[1000001];
		for(int i=1;i<=1000000;i++)
			for(int j=i;j<=1000000;j=j+i)
				t[j]++;
		long sum = 0;
		long mod = 1073741824;
		for(int i=1;i<=a;i++){
			for(int j=1;j<=b;j++){
				for(int k =1;k<=c;k++){
					int num = i*j*k;
					sum=sum+t[num];
					sum=sum%mod;
				}
			}
		}
		System.out.println(sum);
		// end of main
	}
}
