#include <bits/stdc++.h>
using namespace std;
int n, m, r, i, j, k, p, x;
int a[60][60], b[1001][60][60];
int main() {
  scanf("%d%d%d", &n, &m, &r);
  for (j = 0; j < n; j++)
    for (k = 0; k < n; k++) b[0][j][k] = 1000000007;
  for (i = 0; i < m; i++) {
    for (j = 0; j < n; j++)
      for (k = 0; k < n; k++) scanf("%d", &a[j][k]);
    for (p = 0; p < n; p++)
      for (j = 0; j < n; j++)
        for (k = 0; k < n; k++)
          if (a[j][k] > a[j][p] + a[p][k]) a[j][k] = a[j][p] + a[p][k];
    for (j = 0; j < n; j++)
      for (k = 0; k < n; k++)
        b[0][j][k] = ((b[0][j][k]) < (a[j][k]) ? (b[0][j][k]) : (a[j][k]));
  }
  for (i = 1; i <= 1000; i++)
    for (j = 0; j < n; j++)
      for (k = 0; k < n; k++) {
        b[i][j][k] = b[i - 1][j][k];
        for (p = 0; p < n; p++)
          b[i][j][k] = ((b[i][j][k]) < (b[0][j][p] + b[i - 1][p][k])
                            ? (b[i][j][k])
                            : (b[0][j][p] + b[i - 1][p][k]));
      }
  for (i = 0; i < r; i++) {
    scanf("%d%d%d", &k, &p, &x);
    printf("%d\n", b[x][k - 1][p - 1]);
  }
  return 0;
}
