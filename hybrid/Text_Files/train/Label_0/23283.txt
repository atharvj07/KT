#include <bits/stdc++.h>
using namespace std;
long long int n, q, arr[200005], sqr[500], grupsize, grupno, grupcnt, mins, ans,
    upd[500];
void update(long long int left, long long int right, long long int value) {
  long long int lgrup = (left / grupsize + (left % grupsize > 0));
  long long int rgrup = (right / grupsize + (right % grupsize > 0));
  if (rgrup == lgrup) {
    for (int i = left; i <= right; i++) {
      arr[i] += value;
      sqr[lgrup] = min(sqr[lgrup], arr[i]);
    }
    sqr[lgrup] = LLONG_MAX;
    for (int i = (rgrup - 1) * grupsize + 1; i <= rgrup * grupsize; i++)
      sqr[lgrup] = min(sqr[lgrup], arr[i]);
  } else {
    for (int i = lgrup + 1; i < rgrup; i++) upd[i] += value;
    for (int i = left; i <= lgrup * grupsize; i++) arr[i] += value;
    for (int i = (rgrup - 1) * grupsize + 1; i <= right; i++) arr[i] += value;
    sqr[lgrup] = LLONG_MAX;
    for (int i = (lgrup - 1) * grupsize + 1; i <= lgrup * grupsize; i++)
      sqr[lgrup] = min(sqr[lgrup], arr[i]);
    sqr[rgrup] = LLONG_MAX;
    for (int i = (rgrup - 1) * grupsize + 1; i <= rgrup * grupsize; i++)
      sqr[rgrup] = min(sqr[rgrup], arr[i]);
  }
}
long long int query(long long int left, long long int right) {
  long long int lgrup = (left / grupsize + (left % grupsize > 0));
  long long int rgrup = (right / grupsize + (right % grupsize > 0));
  ans = LLONG_MAX;
  if (rgrup == lgrup) {
    for (int i = left; i <= right; i++) {
      ans = min(ans, arr[i] + upd[lgrup]);
    }
  } else {
    for (int i = lgrup + 1; i < rgrup; i++) ans = min(ans, sqr[i] + upd[i]);
    for (int i = left; i <= lgrup * grupsize; i++)
      ans = min(ans, arr[i] + upd[lgrup]);
    for (int i = (rgrup - 1) * grupsize + 1; i <= right; i++)
      ans = min(ans, arr[i] + upd[rgrup]);
  }
  return ans;
}
void solve() {
  cin >> n;
  grupcnt = (long long int)sqrt(n);
  grupsize = (long long int)ceil(sqrt(n));
  grupno = 1;
  mins = LLONG_MAX;
  for (int i = 1; i <= n; i++) {
    cin >> arr[i];
    mins = min(arr[i], mins);
    if (i % grupsize == 0) {
      sqr[grupno] = mins;
      mins = LLONG_MAX;
      grupno++;
    }
  }
  cin >> q;
  string s;
  getline(std::cin, s);
  while (q--) {
    int l, r, val = -999999999;
    getline(std::cin, s);
    istringstream iss(s);
    iss >> l >> r >> val;
    l++;
    r++;
    if (val == 0) continue;
    if (val == -999999999) {
      if (l == r) {
        cout << arr[l] + upd[(l / grupsize + (l % grupsize > 0))] << endl;
        continue;
      } else {
        long long int c = LLONG_MAX;
        if (l <= r)
          c = min(c, query(l, r));
        else {
          c = min(c, query(1, r));
          c = min(c, query(l, n));
        }
        cout << c << endl;
      }
    } else {
      if (l <= r)
        update(l, r, val);
      else {
        update(1, r, val);
        update(l, n, val);
      }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  solve();
}
