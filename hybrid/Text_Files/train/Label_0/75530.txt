t = int(input())
for z in range(t):
	n = int(input())
	l = list(map(int, input().split()))
	ans = -1
	max_ = max(l)
	check = False
	for i in range(n):
		if l[i] == max_:
			if n > i + 1 >= 0 and l[i + 1] != max_ or (n > i - 1 >= 0 and l[i - 1] != max_):
				check = True
				ans = i + 1
				break
	print(ans)
