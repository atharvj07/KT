#include <bits/stdc++.h>
using namespace std;
const int N = 4e5 + 10;
const long long MOD = 100000001659;
const long long MOD2 = 998244353;
int sg[4 * N], ban[4 * N], ans[N];
vector<pair<int, int> > v, qv[N];
void update(int v, int tl, int tr, int ind, int val) {
  if (tl == tr) {
    sg[v] = 1;
    ban[v] = val;
    return;
  }
  int tm = (tl + tr) / 2;
  if (ind <= tm)
    update(v + v, tl, tm, ind, val);
  else
    update(v + v + 1, tm + 1, tr, ind, val);
  sg[v] = sg[v + v] + sg[v + v + 1];
}
int query(int v, int tl, int tr, int x) {
  if (tl == tr) {
    return ban[v];
  }
  int tm = (tl + tr) / 2;
  if (sg[v + v] >= x) return query(v + v, tl, tm, x);
  return query(v + v + 1, tm + 1, tr, x - sg[v + v]);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ;
  mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
  int n;
  cin >> n;
  for (int i = 0; i < (n); ++i) {
    int tv;
    cin >> tv;
    v.push_back({tv, -i});
  }
  int q;
  cin >> q;
  for (int i = 0; i < (q); ++i) {
    int ind, val;
    cin >> ind >> val;
    qv[ind].push_back({val, i});
  }
  sort(v.begin(), v.end());
  for (int i = (v.size()) - 1; i >= 0; i--) {
    update(1, 0, n - 1, -v[i].second, v[i].first);
    for (auto tv : qv[n - i]) {
      ans[tv.second] = query(1, 0, n - 1, tv.first);
    }
  }
  for (int i = 0; i < (q); ++i) {
    cout << ans[i] << endl;
  }
  return 0;
}
