from sys import stdin, stdout

def ncr(n, r, p):
	num = den = 1
	for i in range(r):
		num = num * (n - i) % p
		den = den * (i + 1) % p
	return num * pow(den, p - 2, p) % p

def main():
	n = int(stdin.readline())
	arr = list(map(int, stdin.readline().split()))
	ans = 0
	for i in range(0, n, 2):
		k = kn = 0
		for j in range(i + 1, n, 2):
			ans += max(0, min(arr[i], arr[j] - k) - max(0, -kn - 1))
			if j + 1 >= n:
				continue
			k -= arr[j]
			kn = min(k, kn)
			k += arr[j + 1]
	print(ans)
main()
