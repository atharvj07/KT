#include <bits/stdc++.h>
using namespace std;
int main() {
  int *arr, n, numOfTeams = 0, numOfTeam = 0;
  short k;
  cin >> n >> k;
  arr = new int[n];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
    if ((arr[i] + k) <= 5) {
      numOfTeam += 1;
      if (numOfTeam == 3) {
        numOfTeams += 1;
        numOfTeam = 0;
      }
    }
  }
  cout << numOfTeams;
  return 0;
}
