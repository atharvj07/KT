import java.util.*;
public class Second {
    public static void main(String[] args){
        Scanner sc= new Scanner(System.in);
        int r = sc.nextInt();
        int d = sc.nextInt();
        int n = sc.nextInt();
        int c=0;
        for(int i=0;i<n;i++){
            int xi = sc.nextInt();
            int yi = sc.nextInt();
            int ri = sc.nextInt();
            double dist = distance(xi,yi);
            if(dist>=(r-d+ri)){
                if(dist<=Math.abs(r-ri)){
                    c++;
                }
            }
        }
        System.out.println(c);
    }
    public static double distance(int xi,int yi){
        double x = xi;
        double y = yi;
        x*=xi;
        y*=yi;
        double m = x+y;
        return Math.sqrt(m);
    }
}
