import sys
(length, count) = map(int, input().split())
string = input()
pref = []
for s in ['abc', 'acb', 'bca', 'cab', 'cba', 'bac']:
	pref_sum = [0] * (length + 1)
	for i in range(length):
		pref_sum[i + 1] = pref_sum[i] + int(string[i] != s[i % 3])
	pref.append(tuple(pref_sum))
ans = [0] * count
for i in range(count):
	(a, b) = map(int, sys.stdin.readline().split())
	ans[i] = str(min([el[b] - el[a - 1] for el in pref]))
sys.stdout.write('\n'.join(ans))
