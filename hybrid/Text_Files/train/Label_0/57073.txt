import java.io.*;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.StringTokenizer;

public class mainD {
    public static PrintWriter out = new PrintWriter(System.out);
    public static FastScanner enter = new FastScanner(System.in);
    public static HashMap<Long, Integer> hm=new HashMap<>();
    public static void main(String[] args) throws IOException {
        int n=enter.nextInt();
        long summ=0;
        for (int i = 0; i <n ; i++) {
            long tmp=enter.nextLong();
            if(hm.get(tmp)!=null){
                hm.put(tmp, hm.get(tmp)+1);
            }
            else{
                hm.put(tmp,1);
            }
            summ+=tmp;
        }

        int kol_vo=0;
        LinkedList<Long> ll=new LinkedList<>();
        for (long key:hm.keySet()) {
            if(hm.get(key)>1){
                kol_vo++;
                ll.addFirst(key);
            }
        }

        if(kol_vo>1){
            System.out.println("cslnb");
            return;
        }

        if(kol_vo==1){
            Long who =ll.peekFirst();
            if(hm.get(who-1)!=null || hm.get(who)>2 || who==0){
                System.out.println("cslnb");
                return;
            }
        }
        long oorAns=n*(n-1);
        oorAns/=2;
        summ-=oorAns;
        if(summ%2==0){
            System.out.println("cslnb");
        }
        else{
            System.out.println("sjfnb");
        }



    }

    static class FastScanner {
        BufferedReader br;
        StringTokenizer stok;

        FastScanner(InputStream is) {
            br = new BufferedReader(new InputStreamReader(is));
        }

        String next() throws IOException {
            while (stok == null || !stok.hasMoreTokens()) {
                String s = br.readLine();
                if (s == null) {
                    return null;
                }
                stok = new StringTokenizer(s);
            }
            return stok.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        long nextLong() throws IOException {
            return Long.parseLong(next());
        }

        double nextDouble() throws IOException {
            return Double.parseDouble(next());
        }

        char nextChar() throws IOException {
            return (char) (br.read());
        }

        String nextLine() throws IOException {
            return br.readLine();
        }
    }
}
