#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2 * 1000 * 100 + 10;
const double EPS = 1e-9;
const long long MOD = 1000 * 1000 * 1000 + 7;
long long ans = 0;
int n, k;
struct el {
  long long vert_count, dist;
};
vector<el> D[MAXN];
vector<el> dfs(int v, vector<vector<int> > &g, int p = -1) {
  vector<el> result(k);
  for (int i = 0; i < g[v].size(); ++i) {
    int to = g[v][i];
    if (to == p) continue;
    auto tmp = dfs(to, g, v);
    result[min(k - 1, 1)].vert_count++;
    result[min(k - 1, 1)].dist++;
    for (int j = 0; j < k; ++j) {
      result[(j + 1) % k].vert_count += tmp[j].vert_count;
      result[(j + 1) % k].dist += tmp[j].dist + tmp[j].vert_count;
    }
  }
  return D[v] = result;
}
long long dfs2(int v, vector<vector<int> > &g, vector<el> from_parent,
               int p = -1) {
  long long result = 0;
  for (int j = 0; j < k; ++j) {
    result += (D[v][j].dist - D[v][j].vert_count * j) / k;
    if (j != 0) {
      result += D[v][j].vert_count;
    }
  }
  if (p == -1) {
    for (int i = 0; i < g[v].size(); ++i) {
      int to = g[v][i];
      vector<el> my = D[v];
      my[min(k - 1, 1)].vert_count--;
      my[min(k - 1, 1)].dist--;
      for (int j = 0; j < k; ++j) {
        my[j].vert_count -= D[to][(j - 1 + k) % k].vert_count;
        my[j].dist -=
            D[to][(j - 1 + k) % k].dist + D[to][(j - 1 + k) % k].vert_count;
      }
      vector<el> my2 = my;
      for (int j = 0; j < k; ++j) {
        my[j] = my2[(j - 1 + k) % k];
      }
      my[min(k - 1, 1)].vert_count++;
      for (int j = 0; j < k; ++j) {
        my[j].dist += my[j].vert_count;
      }
      auto tmp = dfs2(to, g, my, v);
      result += tmp;
    }
  } else {
    for (int i = 0; i < from_parent.size(); ++i) {
      result += (from_parent[i].dist - from_parent[i].vert_count * i) / k;
      if (i != 0) {
        result += from_parent[i].vert_count;
      }
    }
    for (int i = 0; i < g[v].size(); ++i) {
      int to = g[v][i];
      if (to == p) continue;
      vector<el> my = D[v];
      my[min(k - 1, 1)].vert_count--;
      my[min(k - 1, 1)].dist--;
      for (int j = 0; j < k; ++j) {
        my[j].vert_count -= D[to][(j - 1 + k) % k].vert_count;
        my[j].dist -=
            D[to][(j - 1 + k) % k].dist + D[to][(j - 1 + k) % k].vert_count;
      }
      vector<el> my2 = my;
      for (int j = 0; j < k; ++j) {
        my[j] = my2[(j - 1 + k) % k];
      }
      my[min(k - 1, 1)].vert_count++;
      for (int j = 0; j < k; ++j) {
        my[j].dist += my[j].vert_count;
      }
      for (int j = 0; j < k; ++j) {
        my[j].vert_count += from_parent[(j - 1 + k) % k].vert_count;
        my[j].dist += from_parent[(j - 1 + k) % k].dist +
                      from_parent[(j - 1 + k) % k].vert_count;
      }
      auto tmp = dfs2(to, g, my, v);
      result += tmp;
    }
  }
  return result;
}
int main(int argc, char *argv[]) {
  ios::sync_with_stdio(0);
  cin.tie();
  scanf("%d%d", &n, &k);
  vector<vector<int> > g(n);
  for (int i = 0; i < n - 1; ++i) {
    int f, t;
    scanf("%d%d", &f, &t);
    f--;
    t--;
    g[f].push_back(t);
    g[t].push_back(f);
  }
  dfs(0, g);
  cout << dfs2(0, g, vector<el>()) / (long long)2;
  return 0;
}
