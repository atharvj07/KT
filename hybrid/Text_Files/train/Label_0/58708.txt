import java.util.*;

public class Main{

    public static int c = 1;
    public static boolean[] boo = new boolean[4];

    public static void clear(){
        c++;
        for (int j = 0; j<4; j++)boo[j]=false;
    }

    public static void main(String[] args){
        Scanner p = new Scanner(System.in);
        int n = p.nextInt();
        String s = p.next();
        for (int i = 0; i<n; i++){
            if (s.charAt(i) == 'U') {
                if (!boo[1])
                    boo[0] = true;
                else {
                    clear();
                    boo[0] = true;
                }

            }
            if (s.charAt(i) == 'D') {
                if (!boo[0])
                    boo[1] = true;
                else {
                    clear();
                    boo[1] = true;
                }
            }
            if (s.charAt(i) == 'L') {
                if (!boo[3])
                    boo[2] = true;
                else {
                    clear();
                    boo[2] = true;
                }
            }
            if (s.charAt(i) == 'R') {
                if (!boo[2])
                    boo[3] = true;
                else {
                    clear();
                    boo[3] = true;
                }
            }

        }
        System.out.print(c);
    }

}
