#include <bits/stdc++.h>
using namespace std;

void solve()
{
    int n,m;cin>>n>>m;
    vector<vector<int>> g(n);
    for(int i=0;i<m;i++)
    {
        int x,y;cin>>x>>y;x--;y--;
        g[x].push_back(y);
        g[y].push_back(x);
    }
    vector<int> q;
    vector<int> col(n,-1);
    col[0] = 1;
    for(auto x:g[0])
    {
        q.push_back(x);
        col[x] = 0;
    }
    for(int i=0;i<q.size();i++)
    {
        for(int v:g[q[i]])
        {
            if(col[v] == -1)
            {
                col[v] = 1;
                for(int x:g[v])
                {
                    if(col[x] == -1)
                    {
                        col[x] = 0;
                        q.push_back(x);
                    }
                }
            }
        }
    }
    if (*min_element(col.begin(), col.end()) == -1) 
    {
        cout << "NO" << '\n';
        return;
    }
    cout << "YES" << '\n';
    vector<int> ret;
    for (int i = 0; i < n; i++) 
    {
        if (col[i] == 1) 
        {
            ret.push_back(i);
        }
    }
    cout << ret.size() << endl;
    for(int x:ret)
    {
        cout << x+1 << " ";
    }
    cout << endl;
}

int main() {
    int t;cin>>t;
    while(t--)
    {
        solve();
    }
    return 0;
}
