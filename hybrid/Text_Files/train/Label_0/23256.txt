#include <bits/stdc++.h>
using namespace std;
signed main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  ;
  long long int n;
  cin >> n;
  long long int a[n];
  long long int b[n / 2];
  for (long long int i = 0; i < n / 2; i++) cin >> b[i];
  a[0] = 0;
  a[n - 1] = b[0];
  for (long long int i = 1; i < n / 2; i++) {
    if (b[i] <= a[n - i]) {
      if (0 == a[i - 1]) {
        a[i] = 0;
        a[n - i - 1] = b[i];
      } else {
        a[i] = a[i - 1];
        a[n - i - 1] = b[i] - a[i];
      }
    } else {
      if (b[i] - a[n - i] >= a[i - 1]) {
        a[n - i - 1] = a[n - i];
        a[i] = b[i] - a[n - i - 1];
      } else {
        a[i] = a[i - 1];
        a[n - i - 1] = b[i] - a[i];
      }
    }
  }
  for (long long int i = 0; i < n; i++) cout << a[i] << " ";
  return 0;
}
