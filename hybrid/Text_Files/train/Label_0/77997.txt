import sys
import math
import bisect

def solve(A):
	n = len(A)
	m = len(A[0])
	row = [0] * n
	col = [0] * m
	total = 0
	for i in range(n):
		for j in range(m):
			if A[i][j] == '*':
				row[i] += 1
				col[j] += 1
				total += 1
	for i in range(n):
		for j in range(m):
			val = row[i] + col[j]
			if A[i][j] == '*':
				val -= 1
			if val == total:
				return (i, j)
	return None

def main():
	(n, m) = map(int, input().split())
	A = []
	for i in range(n):
		A.append(input())
	ans = solve(A)
	if ans:
		print('YES')
		print('%d %d' % (ans[0] + 1, ans[1] + 1))
	else:
		print('NO')
main()
