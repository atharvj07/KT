for _ in range(int(input())):
	(n, q) = map(int, input().split())
	a = list(map(int, input().split()))
	a.append(0)
	count = 0
	Pokemon = []
	for i in range(n):
		if a[i] > a[i + 1] and count % 2 == 0:
			Pokemon.append(a[i])
			count += 1
		if a[i] < a[i + 1] and count % 2 == 1:
			Pokemon.append(-a[i])
			count += 1
		Pokemon.append(a[-1])
	if len(Pokemon) % 2 == 0:
		Pokemon.pop()
	print(sum(Pokemon))
