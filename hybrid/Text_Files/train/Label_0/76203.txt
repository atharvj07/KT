import sys
input = sys.stdin.readline
t = int(input())
for i in range(t):
	n = int(input())
	a = map(int, input().split())
	b = map(int, input().split())
	zipp_list = zip(a, b)
	sorted_pair = sorted(zipp_list)
	tuples = zip(*sorted_pair)
	(restoran, petya) = [list(tuple) for tuple in tuples]
	restoran = [-1] + restoran
	petya = [-1] + petya
	lo = 0
	hi = n
	naj = float('inf')
	while lo <= hi:
		mid = (lo + hi) // 2
		dostavljac = max(restoran[:mid + 1])
		auto = sum(petya[mid + 1:])
		rj = max(dostavljac, auto)
		if rj < naj:
			naj = rj
		if dostavljac > auto:
			hi = mid - 1
		elif auto > dostavljac:
			lo = mid + 1
		else:
			lo = hi = mid
			break
	print(naj)
