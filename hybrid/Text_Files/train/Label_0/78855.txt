from math import ceil
for _ in range(1):
	n = int(input())
	if n <= 3:
		for i in range(n):
			if i == n - 1:
				print(i + 1)
				break
			print(1, end=' ')
		continue
	count = 1
	while 2 ** count <= n:
		count += 1
	count -= 1
	curr = 2 ** count
	total = 0
	add = 0
	ans = []
	while curr > 0:
		add = n // curr
		for j in range(add - total):
			ans.append(curr)
		total = add
		curr = curr >> 1
	ans = ans[::-1]
	c = ans[-2]
	d = n - n % c
	ans.pop()
	ans.append(d)
	print(*ans)
