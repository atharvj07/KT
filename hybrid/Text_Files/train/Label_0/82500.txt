s = str(input())
x = b = 0
r = []
for i in range(len(s)):
	if s[i] == '(':
		b += 1
	elif s[i] == ')':
		b -= 1
	else:
		b -= 1
		x = b
		r.append(1)
	if b < 0:
		print(-1)
		quit()
if b <= x:
	r[-1] += b
	x = b = 0
	for i in range(len(s)):
		if s[i] == '(':
			b += 1
		elif s[i] == ')':
			b -= 1
		else:
			b -= r[x]
			x += 1
		if b < 0:
			print(-1)
			quit()
else:
	print(-1)
	quit()
for i in range(len(r)):
	print(r[i])
