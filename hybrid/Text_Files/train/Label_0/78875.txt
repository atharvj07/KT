from itertools import product

def sq(i, j):
	return ([i - 1, j - 1], [])
(n, m, k) = [int(v) for v in input().split()]
a = [] * k
black = [[0 for i in range(m + 2)] for j in range(n + 2)]
for q in range(k):
	a = [int(v) for v in input().split()]
	black[a[0]][a[1]] = 1
	(i, j) = a
	if sum([black[u][v] for (u, v) in product([i - 1, i], [j - 1, j])]) == 4 or sum([black[u][v] for (u, v) in product([i - 1, i], [j, j + 1])]) == 4 or sum([black[u][v] for (u, v) in product([i, i + 1], [j - 1, j])]) == 4 or (sum([black[u][v] for (u, v) in product([i, i + 1], [j, j + 1])]) == 4):
		print(q + 1)
		raise SystemExit
print(0)
