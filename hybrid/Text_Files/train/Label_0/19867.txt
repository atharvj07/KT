#include <bits/stdc++.h>
using namespace std;
struct str {
  int v, nex;
  long long c;
} edge[101000 << 1];
const long long Mo = 1000000007LL;
long long C[201000];
int A[201000], B[201000], mark[201000];
int R, fa[101000], fst[101000], n, m, tp, e;
long long f[101000][2], co[201000], xo[101000], okay[70], ans;
vector<int> vec[101000];
int find(int a) { return (fa[a] == a ? a : (fa[a] = find(fa[a]))); }
void make_edge(int a, int b, long long c) {
  edge[++e].nex = fst[a];
  fst[a] = e;
  edge[e].v = b;
  edge[e].c = c;
  edge[++e].nex = fst[b];
  fst[b] = e;
  edge[e].v = a;
  edge[e].c = c;
}
void dfs(int u) {
  for (int i = fst[u]; i; i = edge[i].nex) {
    int v = edge[i].v;
    if (fa[u] == v) continue;
    fa[v] = u;
    xo[v] = xo[u] ^ edge[i].c;
    dfs(v);
  }
  return;
}
int get_dim() {
  int q = 0;
  for (int i = 0; i <= 60; i++) {
    int u = 0;
    for (int j = q + 1; j <= tp; j++)
      if ((co[j] >> i) & 1LL) {
        u = j;
        continue;
      }
    if (!u) continue;
    q++;
    swap(co[q], co[u]);
    for (int j = q + 1; j <= tp; j++) {
      if ((co[j] >> i) & 1LL) co[j] ^= co[q];
    }
  }
  return q;
}
void Dfs(int u, int k) {
  f[u][0] = 1;
  f[u][1] = 0;
  for (int i = fst[u]; i; i = edge[i].nex) {
    int v = edge[i].v;
    if (fa[u] == v) continue;
    Dfs(v, k);
    if ((edge[i].c >> k) & 1LL) swap(f[v][0], f[v][1]);
    if (okay[k]) {
      ans += (f[u][0] + f[u][1]) * (f[v][0] + f[v][1]) % Mo *
             ((1LL << (R - 1)) % Mo) % Mo * ((1LL << k) % Mo) % Mo;
      ;
      ans %= Mo;
    } else {
      ans += (f[u][0] * f[v][1] + f[v][0] * f[u][1]) % Mo * ((1LL << R) % Mo) %
             Mo * ((1LL << k) % Mo) % Mo;
      ans %= Mo;
    }
    f[u][0] += f[v][0];
    f[u][0] %= Mo;
    f[u][1] += f[v][1];
    f[u][1] %= Mo;
  }
  return;
}
void solve(int base) {
  dfs(base);
  tp = 0;
  for (int i = 0; i < (int)vec[base].size(); i++) {
    int k = vec[base][i];
    co[++tp] = (xo[A[k]] ^ xo[B[k]] ^ C[k]);
  }
  R = get_dim();
  for (int i = 0; i <= 60; i++) {
    okay[i] = 0;
    for (int j = 1; j <= R; j++) okay[i] |= (co[j] >> i) & 1LL;
  }
  for (int i = 0; i <= 60; i++) {
    Dfs(base, i);
  }
}
int main() {
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= m; i++) {
    scanf("%d %d %I64d", &A[i], &B[i], &C[i]);
  }
  for (int i = 1; i <= n; i++) fa[i] = i;
  for (int i = 1; i <= m; i++) {
    int a = A[i], b = B[i];
    find(a);
    find(b);
    if (fa[a] == fa[b]) continue;
    mark[i] = 1;
    if (fa[a] > fa[b]) swap(a, b);
    fa[fa[b]] = fa[a];
    make_edge(a, b, C[i]);
  }
  for (int i = 1; i <= m; i++) {
    if (mark[i]) continue;
    int a = A[i];
    find(a);
    vec[fa[a]].push_back(i);
  }
  for (int i = 1; i <= n; i++) fa[i] = 0;
  for (int i = 1; i <= n; i++) {
    if (!fa[i]) solve(i);
  }
  cout << ans << endl;
  return 0;
}
