#include <cstdio>
#include <cstring>
#include <cstdlib>
#include <algorithm>
using namespace std;
const int Maxn = 510;
int n, m;
char s[Maxn][Maxn];
int main() {
	int i, j, k;
	scanf("%d%d", &n, &m);
	for(i = 1; i <= n; i++) scanf("%s", s[i]+1);
	for(i = 1; i <= n; i++){
		printf("#");
		if(i&1) for(j = 2; j < m; j++) printf("#");
		else for(j = 2; j < m; j++) printf("%c", s[i][j]);
		printf(".\n");
	}
	printf("\n");
	for(i = 1; i <= n; i++){
		printf(".");
		if(!(i&1)) for(j = 2; j < m; j++) printf("#");
		else for(j = 2; j < m; j++) printf("%c", s[i][j]);
		printf("#\n");
	}
	return 0;
}
