import java.util.*;
import java.io.*;


public class Main{

    public static void main(String[] args) throws java.io.IOException{
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while (t-->0)
        {
            int n=sc.nextInt();
            if(n==2)
            {
                System.out.println(-1);
                continue;
            }
            int[][] mat=new int[n][n];
            int count=1;
            int i=0;
            while(count<=n*n)
            {
                if(i>=n*n)
                {
                    i=1;
                    continue;
                }
                int row = i/n;
                int col = i%n;
                mat[row][col]=count;
                count++;
                i+=2;
            }
            for(int j=0;j<n;++j) {
                for (int k = 0; k < n; k++)
                    System.out.print(mat[j][k] + " ");
                System.out.println();
            }
        }
    }
}
