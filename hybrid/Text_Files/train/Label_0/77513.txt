import sys

def techno_c():
	a = list(set(map(int, sys.stdin.readline().split())))
	n = int(sys.stdin.readline())
	b = list(set(map(int, sys.stdin.readline().split())))
	n = len(b)
	m = len(a)
	raz = []
	for j in range(m):
		for i in range(n):
			raz.append([b[i] - a[j], i])
	raz.sort()
	if m == 1:
		return raz[-1][0] - raz[0][0]
	v = [-1] * n
	s = set()
	for i in range(n * m):
		if v[raz[i][1]] >= 0:
			raz[v[raz[i][1]]][0] = -1
		v[raz[i][1]] = i
		s.add(raz[i][1])
		if len(s) == n:
			break
	j = i
	i = min(v)
	ans = raz[j][0] - raz[i][0]
	while j < n * m - 1:
		if len(s) == n:
			ans = min(ans, raz[j][0] - raz[i][0])
			if ans == 0:
				return 0
			k = raz[i][1]
			raz[v[k]][0] = -1
			v[k] = -1
			s.remove(k)
			i += 1
			while raz[i][0] == -1:
				i += 1
		else:
			j += 1
			if v[raz[j][1]] >= 0:
				if v[raz[j][1]] == i:
					i += 1
					while raz[i][0] == -1:
						i += 1
				raz[v[raz[j][1]]][0] = -1
			v[raz[j][1]] = j
			s.add(raz[j][1])
	return ans
rez = techno_c()
sys.stdout.write(str(rez) + '\n')
