from sys import stdin, stdout
nmbr = lambda : int(stdin.readline())
lst = lambda : list(map(int, stdin.readline().split()))
for _ in range(1):
	n = nmbr()
	a = lst()
	bits = [[0 for _ in range(34)] for _ in range(n)]
	for i in range(n):
		for j in range(34):
			bits[i][33 - j] = a[i] >> j & 1
	bit = -1
	x = (1 << 32) - 1
	for bp in range(34):
		nd = x
		for i in range(n):
			if bits[i][33 - bp]:
				nd &= a[i]
		if nd % (1 << bp) == 0:
			bit = bp
	if bit == -1:
		print(-1)
	else:
		ans = []
		for i in range(n):
			if bits[i][33 - bit]:
				ans += [a[i]]
		print(len(ans))
		print(*ans)
