#include <bits/stdc++.h>
using namespace std;
int main() {
  long long ct[200004], a, j, n, x, i;
  scanf("%lld", &n);
  x = 0;
  for (i = 0; i < n; i++) {
    scanf("%lld", &ct[i]);
  }
  for (i = 1; i < n; i++) {
    if (2 * ct[i - 1] < ct[i]) ct[i - 1] = -1;
  }
  ct[i - 1] = -1;
  x = i = 0;
  a = -1;
  for (i = 0; i < n; i++) {
    if (ct[i] == -1) {
      x = max(x, (i - a));
      a = i;
    }
  }
  cout << x << endl;
  return 0;
}
