#include <bits/stdc++.h>
using namespace std;
const int maxn = 1000 + 5;
int mp[maxn][maxn];
int n, m, k;
void Input() {
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      scanf("%d", &mp[i][j]);
    }
  }
}
int solve() {
  int Min = 999999999;
  if (n > k) {
    int sum;
    for (int i = 0; i < n; i++) {
      sum = 0;
      for (int j = 0; j < n; j++) {
        if (i == j) continue;
        int a = 0, b = 0;
        for (int l = 0; l < m; l++) {
          if (mp[i][l] == mp[j][l])
            a++;
          else
            b++;
        }
        sum += min(a, b);
      }
      Min = min(Min, sum);
    }
  } else {
    for (int i = 0; i < (1 << n); i++) {
      int tmp = 0, b[maxn];
      for (int j = 0; j < n; j++) {
        if (i & (1 << j)) {
          b[j] = 1;
          if (mp[j][0] == 0) tmp++;
        } else {
          b[j] = 0;
          if (mp[j][0] == 1) tmp++;
        }
      }
      for (int j = 1; j < m; j++) {
        int cnt1 = 0, cnt2 = 0;
        for (int t = 0; t < n; t++) {
          if (mp[t][j] == b[t])
            cnt1++;
          else
            cnt2++;
        }
        tmp += min(cnt1, cnt2);
      }
      Min = min(Min, tmp);
    }
  }
  return Min;
}
int main() {
  while (~scanf("%d%d%d", &n, &m, &k)) {
    Input();
    int ans = solve();
    if (ans > k)
      puts("-1");
    else
      printf("%d\n", ans);
  }
  return 0;
}
