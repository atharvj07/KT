#include<bits/stdc++.h>
using namespace std;
const int N = 3e3 + 3;
int dp[N][N];
int main(){
	string s, t, ans = "";
	cin >> s >> t;
	int n = s.size(), m = t.size();
	for(int i = n-1; i >= 0; --i)
		for(int j = m-1; j >= 0; --j)
			if(s[i] == t[j])
				dp[i][j] = 1 + dp[i+1][j+1];
			else
				dp[i][j] = max(dp[i][j+1], dp[i+1][j]);
	int i = 0, j = 0;
	while(i < n || j < m)
		if(s[i] == t[j]){
			ans.push_back(s[i]);
			++i; ++j;
		}
		else if(dp[i][j+1] >= dp[i+1][j] && j < m)
			++j;
		else
			++i;
	cout << ans;
	return 0;
}
