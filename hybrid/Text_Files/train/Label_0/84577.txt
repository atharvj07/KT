mot = ' ' + input() + ' '
n = len(mot)
dp = [0] * n
dp[0] = 1
if n == 1:
	print(1)
else:
	if mot[0] == mot[1] and (mot[0] == 'n' or mot[0] == 'u'):
		dp[1] = 2
	else:
		dp[1] = 1
	for x in range(2, n):
		if mot[x] == 'u':
			if mot[x - 1] == 'u':
				dp[x] = (dp[x - 1] + dp[x - 2]) % 1000000007
			else:
				dp[x] = dp[x - 1]
		elif mot[x] == 'n':
			if mot[x - 1] == 'n':
				dp[x] = (dp[x - 1] + dp[x - 2]) % 1000000007
			else:
				dp[x] = dp[x - 1]
		else:
			dp[x] = dp[x - 1]
if mot.count('w') > 0 or mot.count('m') > 0:
	print(0)
else:
	print(dp[n - 1])
