#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int n, q;
  string name, mno;
  vector<string> v;
  cin >> n;
  map<string, vector<string> > map;
  for (int i = 0; i < n; ++i) {
    v.clear();
    cin >> name >> q;
    auto it = map.find(name);
    int j;
    if (it == map.end()) {
      while (q--) {
        cin >> mno;
        for (j = 0; j < (int)v.size(); ++j) {
          if (mno.size() >= v[j].size()) {
            int found = mno.rfind(v[j]);
            if ((mno.length() - found) == v[j].length()) {
              v.erase(v.begin() + j);
              j--;
            }
          } else {
            int found = v[j].rfind(mno);
            if ((v[j].length() - found) == mno.length()) {
              break;
            }
          }
        }
        if (j == (int)v.size()) v.push_back(mno);
      }
      map.insert(make_pair(name, v));
    } else {
      while (q--) {
        cin >> mno;
        for (j = 0; j < (int)(*it).second.size(); ++j) {
          if (mno.size() >= (*it).second[j].size()) {
            if ((mno.size() - mno.rfind((*it).second[j])) ==
                (*it).second[j].size()) {
              (*it).second.erase((*it).second.begin() + j);
              j--;
            }
          } else {
            if (((*it).second[j].size() - (*it).second[j].rfind(mno)) ==
                mno.size()) {
              break;
            }
          }
        }
        if (j == (int)(*it).second.size()) (*it).second.push_back(mno);
      }
    }
  }
  cout << map.size() << endl;
  for (auto it = map.begin(); it != map.end(); ++it) {
    cout << it->first << " " << it->second.size();
    for (int i = 0; i < (int)it->second.size(); ++i) {
      cout << " " << it->second[i];
    }
    cout << endl;
  }
  return 0;
}
