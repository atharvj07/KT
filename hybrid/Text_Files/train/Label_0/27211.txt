#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  while (cin >> n) {
    long long city[100000 + 10] = {};
    for (int i = 0; i < n; i++) cin >> city[i];
    long long min, max;
    for (int j = 0; j < n; j++) {
      long long a, b;
      if (j == 0)
        min = city[j + 1] - city[j];
      else if (j == n - 1)
        min = city[j] - city[j - 1];
      else {
        a = city[j] - city[j - 1];
        b = city[j + 1] - city[j];
        if (a < b)
          min = a;
        else
          min = b;
      }
      a = city[j] - city[0];
      b = city[n - 1] - city[j];
      if (a > b)
        max = a;
      else
        max = b;
      cout << min << " " << max << endl;
    }
  }
  return 0;
}
