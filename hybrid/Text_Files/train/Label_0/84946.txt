for _ in range(int(input())):
	length = int(input())
	array = list(map(int, input().split()))
	array.sort()
	cnt = [0] * (max(array) + 1)
	dp = [0] * (max(array) + 1)
	for x in array:
		cnt[x] += 1
	for i in range(1, len(dp)):
		dp[i] += cnt[i]
		for j in range(i, len(dp), i):
			dp[j] = max(dp[j], dp[i])
	print(length - max(dp))
