    final static int dx[] = {0,0,-1,1}; //1,1,-1,1};
    final static int dy[] = {-1,1,0,0}; //1,-1,1,-1};

    final static int MAXN = 100010;
    final static int MAXL = 128*1024+3;
    final static int MAXD = 2*128*1024+3;

    static class Pair implements Comparable<Pair> {
      int first;

      int second;

      Pair(int f, int s)
      {
        first = f;
        second = s;
      }
      
      Pair(Pair o)
      {
        copy(o);
      }
      
      void copy(Pair o)
      {
        first = o.first;
        second = o.second;
      }

      public int compareTo(Pair o)
      {
        int r = first - o.first;

        if(r == 0)
          r = second - o.second;

        return r;
      }
    }
    
    int n;
    @SuppressWarnings("unchecked")
    ArrayList<Pair>[] E = new ArrayList[MAXN];

    class rooted_tree {
      int k;
      int[] P = new int[MAXN], W = new int[MAXN];
      int[] L = new int[MAXN], S = new int[MAXN], J = new int[MAXN], T = new int[MAXN];
      int[] R = new int[MAXN], PS = new int[MAXN];
      int dfs(int u){
        int p = u; T[u] = 0;
        for(Pair e : E[u]) if(P[u] != e.first){
          P[e.first] = u; W[e.first] = e.second;
          int c = dfs(e.first);
          J[c] = u; T[c] += e.second;
          if(T[c] > T[p]) p = c;
        }
        L[u] = p;
        S[u] = T[p];
        return p;
      }
      void init(int u){
        P[u] = 0;
        W[u] = 0;
        dfs(u);
        ArrayList<Integer> paths = new ArrayList<>();
        for(int i = 1; i < n+1; i++) if(L[i] == i) paths.add(i);
        Collections.sort(paths, (a, b) -> -Integer.compare(T[a], T[b]));
        k = paths.size();
        PS[0] = 0;
        for(int i = 0; i < k; i++){
          int pi = paths.get(i);
          R[pi] = i;
          PS[i+1] = PS[i] + T[pi];
        }
      }
      int get_ans(int x, int y){
        y = Math.min(2*y-1, k);
//    DEBUG printf("%d %d\n", x, y);
        x = L[x];
        if(R[x] < y) return PS[y];
        int tx = 0;
        while(R[L[x]] >= y){
          tx += T[L[x]] - S[x];
          x = J[L[x]];
        }
        return Math.max(
          PS[y-1] + tx,
          PS[y] - S[x] + tx
        );
      }
    };
    rooted_tree A = new rooted_tree(), B = new rooted_tree();

    int[] V = new int[MAXN];

    Pair dfs(int u){
      V[u] = 1;
      Pair res = new Pair(0, u);
      for(Pair e : E[u]) if(V[e.first] == 0){
        Pair can = dfs(e.first);
        can.first += e.second;
        res = res.compareTo(can) < 0 ? can : res;
      }
      return res;
    }
    public void solve(int test, Reader input, PrintWriter out)
    {
        MyScanner in = new MyScanner(input);
        
        int q;
        n = in.nextInt();
        q = in.nextInt();
        for(int i = 0; i <= n; i++)
          E[i] = new ArrayList<>();
        for(int i = 1; i < n; i++){
            int a, b, c;
            a = in.nextInt();
            b = in.nextInt();
            c = in.nextInt();
            E[a].add(new Pair(b,c));
            E[b].add(new Pair(a,c));
        }
        int a = dfs(1).second;
        for(int i = 1; i < n+1; i++) V[i] = 0;
        int b = dfs(a).second;
        A.init(a);
        B.init(b);
        int ans = 0;
        for(int i = 0; i < q; i++){
            int x, y;
            x = in.nextInt();
            y = in.nextInt();
            x = (x+ans-1)%n + 1;
            y = (y+ans-1)%n + 1;
            ans = Math.max(A.get_ans(x, y), B.get_ans(x, y));
            out.println(ans);
        }
    }

    //-----------MyScanner class for faster input----------
    public static class MyScanner {
        BufferedReader br;
        StringTokenizer st;

        public MyScanner(Reader reader) {
            br = new BufferedReader(reader);
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }

        double nextDouble() {
            return Double.parseDouble(next());
        }

        String nextLine(){
            String str = "";
            try {
                str = br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
            }
            return str;
        }

    }
