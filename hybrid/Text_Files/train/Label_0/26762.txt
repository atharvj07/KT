#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
using pll = pair<ll, ll>;
ll modpow(ll a, ll p, ll MOD) {
  ll res = 1;
  for (ll i = 0; (p >> i) > 0; i++) {
    if ((p >> i) & 1) res = res * a % MOD;
    a = a * a % MOD;
  }
  return res;
}
int main() {
  ll N;
  cin >> N;
  const ll MOD = 1e9 + 7;
  vector<ll> fac(N + 1, 1), invfac(N + 1, 1);
  for (ll i = 2; i <= N; i++) fac[i] = fac[i - 1] * i % MOD;
  invfac[N] = modpow(fac[N], MOD - 2, MOD);
  for (ll i = N; i >= 2; i--) invfac[i - 1] = invfac[i] * i % MOD;
  auto comb = [&](ll n, ll k) {
    return (fac[n] * invfac[n - k] % MOD) * invfac[k] % MOD;
  };
  vector<ll> A(N);
  for (auto& a : A) cin >> a;
  vector<ll> table(N + 1, 0);
  ll P = 0, Q = 0;
  for (auto a : A)
    if (a != -1) table[a]++;
  for (ll i = 0; i < N; i++) {
    if (A[i] == -1) P++;
    if (A[i] == -1 && table[i + 1] == 0) Q++;
  }
  ll res = 0, sign = 1;
  for (ll i = 1; i <= Q; i++) {
    res += (comb(Q, i) * fac[P - i] % MOD) * sign;
    if (res < 0) res += MOD;
    res %= MOD;
    sign *= -1;
  }
  res = fac[P] - res;
  if (res < 0) res += MOD;
  cout << res << endl;
  return 0;
}
