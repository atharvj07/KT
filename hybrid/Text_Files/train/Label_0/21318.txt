#include <bits/stdc++.h>
using namespace std;
const int MN = 100 + 10;
int arr[MN][MN];
int main() {
  int n, m;
  cin >> n >> m;
  if (n == 1 && m == 1) {
    cout << 1;
    return 0;
  }
  if ((n == 1 && m <= 3) || (n <= 3 && m == 1) || (n == 2 && m == 2)) {
    cout << -1;
    return 0;
  }
  int k = 1;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if ((i + j) % 2) arr[i][j] = k++;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (!((i + j) % 2)) arr[i][j] = k++;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) cout << arr[i][j] << " ";
    cout << endl;
  }
  return 0;
}
