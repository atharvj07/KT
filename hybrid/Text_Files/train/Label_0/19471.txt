#include<iostream>
#include<vector>
#include<string>
#include<algorithm>
using namespace std;
long long rui(long long n, long long k, long long p) {
	if (k == 0) {
		return 1;
	}
	else {
		long long a = rui(n, k / 2, p);
		if (k % 2 == 0) {
			return (a*a) % p;
		}
		return (((a*a) % p)*n) % p;
	}
}
int main(){
	long long h, w;
	cin >> h >> w;
	long long ans = 0;
	for (long long i = 1; i <= h; i++) {
		for (long long j = 1; j <= w; j++) {
			long long a;
			cin >> a;
			ans += a * i*(h - i + 1)*j*(w - j + 1);
		}
	}
	cout << ans << endl;
}
