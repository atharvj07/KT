typedef long long ll;
const int INF = 1e9;
typedef pair<int, int> PII;
int dx[4]={0, -1, 1, 0}, dy[4]={-1, 0, 0, 1};

int V, E;
vector<vector<int>> G;
vector<int> in_deg;
vector<bool> done;

bool solve()
{
    priority_queue<int, vector<int>, greater<int>> Q;

    rep(i, V) {
        if (in_deg[i] == 0) Q.push(i);
    }

    while (Q.size()) {
        int v = Q.top(); Q.pop();
        done[v] = true;

        for (int u : G[v]) {
            if (--in_deg[u] == 0) Q.push(u);
        }
    }

    rep(i, V) {
        if (in_deg[i] > 0) return true;
    }
    
    return false;
}

int main()
{
    cin >> V >> E;
    G.resize(V);
    in_deg.resize(V);
    done.resize(V);

    rep(i, E) {
        int S, T;
        cin >> S >> T;
        G[S].push_back(T);
        in_deg[T]++;
    }

    cout << solve() << endl;
}
