import sys
input = lambda : sys.stdin.readline().rstrip('\r\n')
from collections import defaultdict
(n1, n2, n3, n4) = map(int, input().split())
A = list(map(int, input().split()))
B = list(map(int, input().split()))
C = list(map(int, input().split()))
D = list(map(int, input().split()))
a = sorted(enumerate(A), key=lambda x: x[1])
b = sorted(enumerate(B), key=lambda x: x[1])
c = sorted(enumerate(C), key=lambda x: x[1])
d = sorted(enumerate(D), key=lambda x: x[1])
m1 = int(input())
m1set = set()
for i in range(m1):
	(x, y) = map(int, input().split())
	m1set.add((x, y))
m2 = int(input())
m2set = set()
for i in range(m2):
	(x, y) = map(int, input().split())
	m2set.add((x, y))
m3 = int(input())
m3set = set()
for i in range(m3):
	(x, y) = map(int, input().split())
	m3set.add((x, y))
bDic = defaultdict(lambda : -1)
for i in range(n2):
	for j in a:
		if (j[0] + 1, i + 1) in m1set:
			continue
		bDic[i] = j[0]
		break
cDic = defaultdict(lambda : -1)
cc = []
for i in range(n3):
	for j in d:
		if (i + 1, j[0] + 1) in m3set:
			continue
		cDic[i] = j[0]
		cc.append((i, C[i] + j[1]))
		break
out = 10 ** 9
cc.sort(key=lambda x: x[1])
for i in range(n2):
	if bDic[i] != -1:
		for j in cc:
			if (i + 1, j[0] + 1) in m2set:
				continue
			out = min(out, j[1] + B[i] + A[bDic[i]])
			break
if out < 10 ** 9:
	print(out)
else:
	print(-1)
