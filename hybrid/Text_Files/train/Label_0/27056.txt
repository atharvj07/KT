#include <bits/stdc++.h>
#define rep(i,n) for(int i=0;i<(int)(n);i++)
#define rep1(i,n) for(int i=1;i<=(int)(n);i++)
#define all(c) c.begin(),c.end()
#define pb push_back
#define fs first
#define sc second
#define show(x) cout << #x << " = " << x << endl
#define chmin(x,y) x=min(x,y)
#define chmax(x,y) x=max(x,y)
using namespace std;
int main(){
	bool x[100]={};
	int N,M;
	cin>>N>>M;
	rep(i,M){
		int a,b;
		cin>>a>>b;
		rep(j,b) x[(a+j)%N]=1;
	}
	int ans[101]={};
	int s=-1;
	rep(i,N) if(!x[i]) s=i;
	if(s==-1){
		cout<<N<<" "<<1<<endl;
		return 0;
	}
	int now=0;
	rep(i,N){
		int p=(s+i)%N;
		if(x[p]) now++;
		else{
			if(now>0) ans[now]++;
			now=0;
		}
	}
	if(now>0) ans[now]++;
	for(int i=N;i>0;i--){
		if(ans[i]) cout<<i<<" "<<ans[i]<<endl;
	}
}
