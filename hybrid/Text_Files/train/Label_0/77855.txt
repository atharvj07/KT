def isprime(a):
	if a < 2:
		return False
	if a == 2:
		return True
	for h in range(2, int(a ** 0.5) + 1):
		if a % h == 0:
			return False
	return True
(a, b, k) = map(int, input().split())
arr = []
for g in range(a, b + 1):
	if isprime(g):
		arr += [g]
check = a
ptr1 = -1
ptr2 = 0
count = 0
n = len(arr)
maxm = 0
if n < k:
	print(-1)
else:
	while ptr2 < n:
		count += 1
		if count >= k:
			if count == k:
				maxm = max(maxm, arr[ptr2] - check + 1)
			maxm = max(maxm, arr[ptr2] - check)
			ptr1 += 1
			check = arr[ptr1]
		ptr2 += 1
	if arr[-1] < b:
		maxm = max(maxm, b - check + 1)
	print(maxm)
