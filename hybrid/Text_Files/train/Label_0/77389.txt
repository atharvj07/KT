from sys import stdin

def main():
	Q = int(input())
	for _ in [0] * Q:
		N = int(input())
		P = list(map(int, input().split()))
		i = P.index(1)
		p1 = tuple(P[i:] + P[:i])
		p2 = tuple(P[i + 1:] + P[:i + 1])
		ans1 = tuple((i for i in range(1, N + 1)))
		ans2 = tuple((i for i in range(N, 0, -1)))
		if p1 == ans1 or p2 == ans2:
			print('YES')
		else:
			print('NO')
input = lambda : stdin.readline()
main()
