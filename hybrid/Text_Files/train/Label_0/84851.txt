n = int(input())
for i in range(n):
	l = [int(x) for x in input().split()]
	l.sort()
	if l[0] + l[1] == l[2]:
		print('YES')
	elif l[0] == l[1] or l[1] == l[2]:
		if l[0] == l[1] and l[2] % 2 == 0 or (l[1] == l[2] and l[0] % 2 == 0):
			print('YES')
		else:
			print('NO')
	elif l[0] != l[1] and l[1] != l[2]:
		if l[0] + l[1] == l[2]:
			print('YES')
		else:
			print('NO')
