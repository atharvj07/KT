ch = input()
s = ''
k = ''
for i in range(len(ch)):
	if ch[i] != 'a':
		s = s + ch[i]
test = True
if len(s) % 2 == 1:
	print(':(')
else:
	for i in range(len(s) // 2):
		test = test and s[i] == s[i + len(s) // 2]
		test = test and s[i] == ch[len(ch) - len(s) // 2 + i]
	if test:
		for i in range(len(ch) - len(s) // 2):
			k = k + ch[i]
	else:
		print(':(')
print(k)
