import bisect
t = int(input())
for i in range(t):
	n = int(input())
	a = [int(x) for x in input().split()]
	k = 0
	for j in range(len(a)):
		if a[j] % 3 == 0:
			a[j] = 0
			k = k + 1
	while 0 in a:
		a.remove(0)
	a.sort()
	p = 0
	q = len(a) - 1
	while p < q:
		if (a[p] + a[q]) % 3 == 0:
			k = k + 1
			a.remove(a[q])
			a.remove(a[p])
			p = 0
			q = len(a) - 1
			continue
		else:
			q = q - 1
	while len(a) > 2:
		j = 3
		m = sum(a[:j])
		if m % 3 == 0:
			k = k + 1
			for q in range(j):
				a.pop(0)
			bisect.bisect(a, m)
	print(k)
