
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class C {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int inf = Integer.MAX_VALUE;
        String s = in.readLine();
        int n = s.length();
        int[][] next = new int[n][26];
        for (int i = 0; i < n; i++)
            Arrays.fill(next[i], inf);
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 0; j < 26; j++)
                next[i][j] = next[i + 1][j];
            next[i][s.charAt(i + 1) - 'a'] = i + 1;
        }
        long[] res = new long[27];
        int d = 0;
        for (int i = 0; i < n; i++) {
            boolean[] seen = new boolean[26];
            seen[s.charAt(i) - 'a'] = true;
            int cnt = 1;
            int current = i;
            while (true) {
                d = Math.max(d, cnt);
                int nextMin = inf;
                for (int j = 0; j < 26; j++)
                    if (!seen[j])
                        nextMin = Math.min(nextMin, next[i][j]);
                if (nextMin == inf) {
                    res[cnt] += n - current;
                    break;
                } else {
                    seen[s.charAt(nextMin) - 'a'] = true;
                    res[cnt] += (nextMin - current);
                    current = nextMin;
                    cnt += 1;
                }
            }
        }
        System.out.println(d);
        for (int i = 1; i <= d; i++)
            System.out.println(res[i]);
    }
}
