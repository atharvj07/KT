#include <bits/stdc++.h>
using namespace std;
const int MAXN = 22;
const int MAXM = 1 << 22;
int e[MAXN + 1];
int n, m;
int full;
int a[MAXM + 1];
bool ext[MAXM + 1];
bool bj[MAXM + 1];
int ans;
void Dfs(int value) {
  if (bj[value]) return;
  bj[value] = true;
  if (ext[value]) Dfs(full ^ value);
  for (int i = 0; i <= n; i++)
    if (value & e[i]) Dfs(value ^ e[i]);
}
int main(int argc, char const *argv[]) {
  for (int i = 0; i <= MAXN; i++) e[i] = 1 << i;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= m; i++) {
    scanf("%d", &a[i]);
    ext[a[i]] = true;
  }
  full = e[n] - 1;
  for (int i = 1; i <= m; i++)
    if (!bj[a[i]]) {
      ans++;
      Dfs(full ^ a[i]);
    }
  printf("%d\n", ans);
  return 0;
}
