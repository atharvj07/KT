TC = int(input())
for tc in range(TC):
	N = int(input())
	A = list(map(int, input().split()))
	X = len(A)
	j = len(A)
	sets = []
	while X > 0:
		for i in range(j - 1, -1, -1):
			if A[i] == X:
				sets.append(j - i)
				j = i
				break
		X -= 1
	for i in range(j):
		sets.append(1)
	dp = [[False for _ in range(N + 1)] for _ in range(len(sets) + 1)]
	for i in range(1, len(sets) + 1):
		dp[i][0] = True
	for i in range(1, len(sets) + 1):
		for j in range(1, N + 1):
			if j < sets[i - 1]:
				dp[i][j] = dp[i - 1][j]
			if j >= sets[i - 1]:
				dp[i][j] = dp[i - 1][j] or dp[i - 1][j - sets[i - 1]]
	if dp[len(sets)][N]:
		print('Yes')
	else:
		print('No')
