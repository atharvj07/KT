#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1005;
int N, mx[MAXN];
string s, t;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  cin >> s;
  N = s.size();
  t = "";
  for (int i = 0; i < N; i++) {
    t += "0";
  }
  stack<int> ind;
  for (int i = 0; i < N; i++) {
    if (s[i] == '1')
      ind.push(i);
    else {
      if (ind.size()) {
        int temp = ind.top();
        ind.pop();
        t[temp] = '1';
      }
    }
  }
  cout << t;
}
