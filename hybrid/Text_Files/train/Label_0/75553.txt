from datetime import date, timedelta
in_ = input().split()
x = int(in_[0])
cur = date(2016, 1, 1)
ans = 0
while cur.year != 2017:
	if in_[2] == 'week':
		if x - 1 == cur.weekday():
			ans += 1
	elif in_[2] == 'month':
		if x == cur.day:
			ans += 1
	else:
		assert False
	cur += timedelta(1)
print(ans)
