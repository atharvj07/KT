#include <bits/stdc++.h>
using namespace std;
int a[100001];
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    for (int j = 1; j * j <= x; j++) {
      if (x % j == 0) {
        if (j == x / j) {
          a[j] += 1;
        } else {
          a[j] += 1;
          a[x / j] += 1;
        }
      }
    }
  }
  int ans = 1;
  for (int i = 2; i <= 100000; i++) {
    ans = max(ans, a[i]);
  }
  cout << ans << '\n';
}
