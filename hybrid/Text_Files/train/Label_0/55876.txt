import java.io.*;
import java.util.*;

public class D {
	static int L = 3000;
	static int i(String s) { return Integer.parseInt(s); }
	public static void main(String[] args) throws Exception {
		BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
		
		int[][] AREA = new int[L][L];
		int[][] R0 = new int[L][L];
		int[][] C0 = new int[L][L];
		int[][] R1 = new int[L][L];
		int[][] C1 = new int[L][L];
		
		boolean[][] G = new boolean[3000][3000];
		int n = i(in.readLine());
		int[][] R = new int[n][4];
		for(int i=0; i<n; i++) {
			String[] arr = in.readLine().split(" ");
			int c0 = i(arr[0]);
			int r0 = i(arr[1]);
			int c1 = i(arr[2]);
			int r1 = i(arr[3]);
			R[i] = new int[]{c0, r0, c1-1, r1-1};
			for(int r=r0; r<r1; r++)
				for(int c=c0; c<c1; c++)
					AREA[r][c]=1;
			for(int c=c0; c<c1; c++) {
				R0[r0][c] = 1;
				R1[r1-1][c] = 1;
			}
			for(int r=r0; r<r1; r++) {
				C0[r][c0] = 1;
				C1[r][c1-1] = 1;
			}
		}
		
		int[][] _AREA = PREFIX(AREA);
		int[][] _R0 = PREFIX(R0);
		int[][] _C0 = PREFIX(C0);
		int[][] _R1 = PREFIX(R1);
		int[][] _C1 = PREFIX(C1);
		I: for(int i=0; i<n; i++) {
			int c0 = R[i][0];
			int r0 = R[i][1];
			for(int sz=1; c0+sz-1<L && r0+sz-1<L; sz++) {
				int c1 = c0+sz-1;
				int r1 = r0+sz-1;
				if(sum(_AREA, c0, r0, c1, r1) < sz*sz) continue I;
				if(sum(_R0, c0, r0, c1, r0) < sz) continue I;
				if(sum(_C0, c0, r0, c0, r1) < sz) continue I;
				if(sum(_R1, c0, r1, c1, r1)==sz &&
				   sum(_C1, c1, r0, c1, r1)==sz) {
					List<Integer> A = new ArrayList<Integer>();
					for(int ii=0; ii<n; ii++) {
						if(R[ii][0]>=c0 && R[ii][1]>=r0 && R[ii][2]<=c1 && R[ii][3]<=r1)
							A.add(ii);
					}
					StringBuilder out = new StringBuilder();
					out.append(String.format("YES %d\n", A.size()));
					for(int ii=0; ii<A.size(); ii++) {
						out.append(A.get(ii)+1);
						out.append(ii==A.size()-1?"\n":" ");
					}
					System.out.print(out);
					return;
				}
			}
		}
		System.out.println("NO");
	}
	static int sum(int[][] B, int c0, int r0, int c1, int r1) {
		return B[r1][c1]-(r0==0?0:B[r0-1][c1])-(c0==0?0:B[r1][c0-1])+(r0==0||c0==0?0:B[r0-1][c0-1]);
		//return get(B,r1,c1)-get(B,r0-1,c1)-get(B,r1,c0-1)+get(B,r0-1,c0-1);
	}
	static int[][] PREFIX(int[][] A) {
		int[][] B = new int[L][L];
		for(int r=0; r<L; r++)
			for(int c=0; c<L; c++)
				B[r][c] = A[r][c]+get(B,r-1,c)+get(B,r,c-1)-get(B,r-1,c-1);
		return B;
	}
	static int get(int[][] A, int r, int c) {
		if(r<0 || c<0 || r>=3000 || c>=3000) return 0;
		return A[r][c];
	}
}
