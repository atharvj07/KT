#include <bits/stdc++.h>
using namespace std;
using namespace rel_ops;
using ll = int64_t;
using Pii = pair<int, int>;
using ull = uint64_t;
using Vi = vector<int>;
void run();
int main() {
  cin.sync_with_stdio(0);
  cin.tie(0);
  cout << fixed << setprecision(10);
  run();
  return 0;
}
int n;
void answer(int first) {
  cout << "! " << first + 1 << endl;
  exit(0);
}
int ask(int first) {
  cout << "? " << first + 1 << endl;
  int second;
  cin >> second;
  return second;
}
int f(int first) { return ask(first) - ask((first + n / 2) % n); }
void run() {
  cin >> n;
  int begin = 0, end = n / 2;
  int first = f(begin);
  if (first == 0) answer(first);
  bool sign = (first > 0);
  while (begin + 1 < end) {
    int mid = (begin + end) / 2;
    int ans = f(mid);
    if (ans == 0) answer(mid);
    bool s = (ans > 0);
    if (s == sign) {
      begin = mid + 1;
    } else {
      end = mid;
    }
  }
  if (f(begin) == 0) {
    answer(begin);
  } else {
    answer(-2);
  }
}
