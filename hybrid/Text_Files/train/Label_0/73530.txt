for tc in range(int(input())):
	n = int(input())
	a = []
	for i in range(n):
		a.append(list(input()))
	t = [i.count('*') for i in a]
	if 2 in t:
		on = t.index(2)
		dot = [i for i in range(len(a[on])) if a[on][i] == '*']
		if on == 0:
			for i in dot:
				a[1][i] = '*'
			for i in a:
				print(''.join(i))
		if on != 0:
			for i in dot:
				a[0][i] = '*'
			for i in a:
				print(''.join(i))
	else:
		k = [i for i in range(n) if t[i] == 1]
		k1 = a[k[0]].index('*')
		k2 = a[k[1]].index('*')
		if k1 == k2:
			if k1 == 0:
				a[k[0]][1] = '*'
				a[k[1]][1] = '*'
			else:
				a[k[0]][0] = '*'
				a[k[1]][0] = '*'
			for i in a:
				print(''.join(i))
		else:
			a[k[0]][k2] = '*'
			a[k[1]][k1] = '*'
			for i in a:
				print(''.join(i))
