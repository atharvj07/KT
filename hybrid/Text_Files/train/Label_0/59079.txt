import java.util.*;
import java.io.*;

public class Functions 
{
	static class FastReader
    {
        BufferedReader br;
        StringTokenizer st;
 
        public FastReader()
        {
            br = new BufferedReader(new
                     InputStreamReader(System.in));
        }
 
        String next()
        {
            while (st == null || !st.hasMoreElements())
            {
                try
                {
                    st = new StringTokenizer(br.readLine());
                }
                catch (IOException  e)
                {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }
 
        int nextInt()
        {
            return Integer.parseInt(next());
        }
 
        long nextLong()
        {
            return Long.parseLong(next());
        }
 
        double nextDouble()
        {
            return Double.parseDouble(next());
        }
 
        String nextLine()
        {
            String str = "";
            try
            {
                str = br.readLine();
            }
            catch (IOException e)
            {
                e.printStackTrace();
            }
            return str;
        }
    }
	
	public static void main(String[] args) 
	{
		FastReader reader = new FastReader();
		
		int i, size = reader.nextInt();
		long[] oldArray = new long[size];
		
		long[] array = new long[size-1];
		
		for (i=0; i<size; i++)
			oldArray[i] = reader.nextInt();
		
		long[] sum = new long[size-1];
		for (i=1; i<size; i++)
			array[i-1] = Math.abs (oldArray[i] - oldArray[i-1]);
		sum[0] = array[0];
		
		long globalHigh = sum[0];
		
		for (i=1; i<size; i++)
			globalHigh = Math.max (globalHigh, array[i-1]);
		
		long evenHigh = sum[0];
		long oddLow = Integer.MAX_VALUE;
		
		
		for (i=1; i<size-1; i++)
		{
			sum[i] = sum[i-1];	
			
			if (i%2 == 0)
			{
				sum[i] += array[i];
				
				globalHigh = Math.max (globalHigh, evenHigh-sum[i]);
				globalHigh = Math.max (globalHigh, sum[i]-oddLow);
				
				evenHigh = Math.max (evenHigh, sum[i]);
			}
			else
			{
				sum[i] -= array[i];
				
				globalHigh = Math.max (globalHigh, evenHigh-sum[i]);
				globalHigh = Math.max (globalHigh, sum[i]-oddLow);
				
				oddLow = Math.min (oddLow, sum[i]);
			}
				
			globalHigh = Math.max (globalHigh , sum[i]);
		}
		
		System.out.println (globalHigh);
	}

}
