#include <bits/stdc++.h>
using namespace std;
int main() {
  char c, d, a[1002] = {}, b[1002] = {}, A[1002] = {}, S[1002] = {},
             W[1002] = {};
  int i, t = 0, k = 0, x, m, n;
  gets(a);
  gets(b);
  x = strlen(a);
  for (i = 0; i < x; i++) {
    c = a[i];
    m = (int)c;
    d = b[i];
    n = (int)d;
    if ((A[m] == 0) && (A[n] == 0)) {
      A[m] = d;
      A[n] = c;
      if (d != c) {
        k++;
        S[k] = c;
        W[k] = d;
      }
    } else if ((A[m] != 0) && (A[n] != 0)) {
      if ((A[m] == d) && (A[n] == c)) {
      } else {
        t++;
        break;
      }
    } else if (((A[m] != 0) && (A[n] == 0)) || ((A[m] == 0) && (A[n] != 0))) {
      t++;
      break;
    }
  }
  if (t > 0)
    cout << "-1";
  else if ((t == 0) && (k == 0))
    cout << "0";
  else {
    cout << k << "\n";
    for (i = 1; i <= k; i++) {
      cout << S[i] << " " << W[i] << "\n";
    }
  }
  return 0;
}
