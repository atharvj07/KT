#include <bits/stdc++.h>
using namespace std;
long long int n;
int m;
int w[101000];
long long int calcula(long long int x) {
  if (x == 1) return 1;
  if (x % 2 == 1) {
    return ((x - 1) * x) / 2 + 1;
  } else {
    return (x * (x - 2)) / 2 + x;
  }
}
int main() {
  cin >> n >> m;
  for (int i = 0; i < (m); i++) {
    int q;
    scanf("%d %d", &q, &w[i]);
  }
  sort(w, w + m);
  reverse(w, w + m);
  int maior = 0;
  for (int i = 1; i <= m; i++) {
    if (calcula(i) <= n) {
      maior = i;
    }
  }
  long long int res = 0;
  for (int i = 0; i < maior; i++) {
    res += w[i];
  }
  cout << res << endl;
  return 0;
}
