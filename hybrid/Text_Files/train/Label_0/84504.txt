from collections import *
from functools import reduce
import sys
input = sys.stdin.readline

def factors(n):
	return set(reduce(list.__add__, ([i, n // i] for i in range(1, int(n ** 0.5) + 1) if n % i == 0)))

def li():
	return [int(i) for i in input().rstrip('\n').split(' ')]

def st():
	return input().rstrip('\n')

def val():
	return int(input())
for _ in range(val()):
	(n, m) = li()
	a = li()
	if n == 2:
		print(-1)
		continue
	for i in range(n):
		a[i] = [a[i], i]
	a.sort(key=lambda x: x[0])
	ans = []
	d = defaultdict(set)
	tot = 0
	for i in range(n):
		j = (i + 1) % n
		tot += a[i][0] + a[j][0]
		d[i].add(j)
		d[j].add(i)
		m -= 1
	if m < 0:
		print(-1)
		continue
	anset = set()
	for i in d:
		for j in d[i]:
			anset.add(tuple(sorted([a[i][1], a[j][1]])))
	while m:
		tot += a[0][0] + a[1][0]
		ans.append((a[0][1], a[1][1]))
		m -= 1
	print(tot)
	for i in anset:
		print(i[0] + 1, i[1] + 1)
	for i in ans:
		print(i[0] + 1, i[1] + 1)
