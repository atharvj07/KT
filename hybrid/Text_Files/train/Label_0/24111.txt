#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9;
int id[110][210];
int n, m, S, T;
int ans;
int l[110], r[110], a[110], b[110], c[110];
int d[200010];
int tot, num;
int cur[200010], head[200010];
int vel[2000010], key[2000010], ne[2000010], addnum;
void add(int u, int v, int k) {
  addnum++, vel[addnum] = v, key[addnum] = k, ne[addnum] = head[u],
            head[u] = addnum;
  addnum++, vel[addnum] = u, key[addnum] = 0, ne[addnum] = head[v],
            head[v] = addnum;
}
bool bfs() {
  queue<int> q;
  for (int i = 0; i <= tot; i++) d[i] = -1;
  d[S] = 0;
  q.push(S);
  while (q.size()) {
    int u = q.front();
    q.pop();
    for (int e = head[u]; e; e = ne[e]) {
      if (!key[e]) continue;
      int v = vel[e];
      if (d[v] == -1) {
        d[v] = d[u] + 1;
        q.push(v);
      }
    }
  }
  return d[T] != -1;
}
int dfs(int u, int k) {
  if (u == T) return k;
  int used = 0;
  for (int &e = cur[u]; e; e = ne[e]) {
    if (!key[e]) continue;
    int v = vel[e];
    if (d[v] == d[u] + 1) {
      int t = dfs(v, min(k - used, key[e]));
      key[e] -= t;
      key[e ^ 1] += t;
      used += t;
      if (used == k) return k;
    }
  }
  return used;
}
int main() {
  addnum = 1;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) scanf("%d%d%d", &a[i], &b[i], &c[i]);
  for (int i = 1; i <= n; i++) scanf("%d%d", &l[i], &r[i]);
  tot = 1;
  S = 0;
  T = 1;
  num = 1e6;
  for (int i = 1; i <= n; i++) {
    for (int j = l[i]; j <= r[i]; j++) {
      id[i][j - l[i]] = ++tot;
      add(tot, tot + 1, num - (a[i] * j * j + b[i] * j + c[i]));
    }
    add(S, id[i][0], inf);
    id[i][r[i] - l[i] + 1] = ++tot;
    add(tot, T, inf);
  }
  for (int i = 1; i <= m; i++) {
    int x, y, d;
    scanf("%d%d%d", &x, &y, &d);
    for (int j = max(l[x] - d, l[y]); j <= min(r[x] - d, r[y]) + 1; j++)
      add(id[x][j + d - l[x]], id[y][j - l[y]], inf);
  }
  while (bfs()) {
    for (int i = 0; i <= tot; i++) cur[i] = head[i];
    ans += dfs(S, inf);
  }
  printf("%d\n", num * n - ans);
}
