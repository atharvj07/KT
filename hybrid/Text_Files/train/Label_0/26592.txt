#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e3 + 3;
int dp[MAXN][2 * MAXN];
char go[MAXN][MAXN];
int n, k;
char S[MAXN];
char ans[MAXN];
int val(char c) {
  if (c == 'W') return 1;
  if (c == 'L') return -1;
  return 0;
}
int f(int p, int d) {
  if (p == n) {
    if (abs(d) == k)
      return 1;
    else
      return 0;
  } else if (abs(d) == k)
    return 0;
  int &ret = dp[p][d + MAXN];
  if (ret != -1) return ret;
  if (S[p] != '?')
    return ret = f(p + 1, d + val(S[p]));
  else {
    if (f(p + 1, d)) {
      go[p][d + MAXN] = 'D';
      return ret = 1;
    } else if (f(p + 1, d + 1)) {
      go[p][d + MAXN] = 'W';
      return ret = 1;
    } else if (f(p + 1, d - 1)) {
      go[p][d + MAXN] = 'L';
      return ret = 1;
    } else {
      return ret = 0;
    }
  }
}
int main() {
  scanf("%d", &n), scanf("%d", &k);
  scanf("%s", S);
  memset(dp, -1, sizeof(dp));
  if (f(0, 0) == 0) {
    printf("NO\n");
  } else {
    int d = 0;
    for (int i = 0; i < (int)n; i++) {
      if (S[i] != '?') {
        ans[i] = S[i];
      } else {
        ans[i] = go[i][d + MAXN];
      }
      d += val(ans[i]);
    }
    printf("%s\n", ans);
  }
}
