import sys
input = sys.stdin.readline
from collections import deque

def process(a, n):
	b = [0] * n
	b[0] = a[0]
	for i in range(1, n):
		b[i] = a[i] - b[i - 1]
	return b

def check(a, b, n):
	c = deque([])
	c.append([0, 1])
	d = deque([])
	count = 0
	for i in range(n):
		if i % 2 == 0:
			while c and c[-1][0] > b[i]:
				c.pop()
			if c and c[-1][0] == b[i]:
				count += c[-1][1]
				c[-1][1] += 1
			else:
				c.append([b[i], 1])
			while d and b[i] < -d[0][0]:
				d.popleft()
			if d and b[i] == -d[0][0]:
				count += d[0][1]
		else:
			while c and b[i] < -c[0][0]:
				c.popleft()
			if c and b[i] == -c[0][0]:
				count += c[0][1]
			while d and d[-1][0] > b[i]:
				d.pop()
			if d and d[-1][0] == b[i]:
				count += d[-1][1]
				d[-1][1] += 1
			else:
				d.append([b[i], 1])
	return count
for _ in range(int(input())):
	n = int(input())
	a = [int(i) for i in input().split()]
	print(check(a, process(a, n), n))
