import sys
import math
T = int(sys.stdin.readline())
for _ in range(T):
	n = int(sys.stdin.readline())
	a = list(map(int, sys.stdin.readline().split()))
	pos = [[] for _ in range(27)]
	for (i, x) in enumerate(a):
		pos[x].append(i + 1)
	pre_sum = [[0] * (n + 1) if pos[i] else [] for i in range(27)]
	for (i, x) in enumerate(a):
		pre_sum[x][i + 1] = 1
	for x in range(1, 27):
		if pre_sum[x]:
			for i in range(1, n + 1):
				pre_sum[x][i] += pre_sum[x][i - 1]
	ans = 0
	for x in range(1, 27):
		if pos[x]:
			x_num = len(pos[x])
			ans = max(ans, x_num)
			mid = x_num // 2
			for j in range(mid):
				left = j
				right = x_num - 1 - j
				left_pos = pos[x][left]
				right_pos = pos[x][right]
				for y in range(1, 27):
					if pos[y] and y != x:
						ans = max(ans, pre_sum[y][right_pos] - pre_sum[y][left_pos] + (j + 1) * 2)
	print(ans)
