import java.util.HashMap;
import java.util.Scanner;


public class Main {
	static HashMap<Character, Integer> val = new HashMap<Character, Integer>();
	static boolean[][][] [][][] v;
	static int cnt = 0;
	
	public static void main(String[] args) {
		Scanner r = new Scanner(System.in);
		
		char[] ac = r.next().toCharArray();
		int id = 0;
		for(int i = 0; i < ac.length; i++)
			if(!val.containsKey(ac[i]))val.put(ac[i], id++);
		
		int[] a = new int[ac.length];
		for(int i = 0; i < 6; i++)
			a[i] = val.get(ac[i]);
		
		v = new boolean[6][6][6] [6][6][6];
		
		perm(0, a, new boolean[6], new int[6]);
		
		System.out.println(cnt);
	}
	private static void perm(int i, int[] a, boolean[] t, int[] p) {
		if(i == 6){
			if(!v[p[0]][p[1]][p[2]] [p[3]][p[4]][p[5]]){
				//ROTATIONS.
				rotateAndMark(p[0], p[1], p[2], p[3], p[4], p[5]);				
				cnt++;
				
			}
		}else{
			for(int j = 0; j < 6; j++)if(!t[j]){
				t[j] = true;
				p[i] = a[j];
				perm(i+1, a, t, p);
				t[j] = false;
			}
		}
	}
	private static void rotateAndMark(int i, int j, int k, int l, int m, int n) {
		if(v[i][j][k] [l][m][n])return;
		
		v[i][j][k] [l][m][n] = true;
		
		rotateAndMark(i, j, m, n, l, k);
		rotateAndMark(i, j, n, m, k, l);
		
		rotateAndMark(l, k, i, j, m, n);
		rotateAndMark(k, l, j, i, m, n);
		
		rotateAndMark(n, m, k, l, i, j);
		rotateAndMark(m, n, k, l, j, i);
	}
}
