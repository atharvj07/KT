vector<long long> z_function(string s) {
  long long n = s.length();
  vector<long long> z(n);
  for (long long i = 1, l = 0, r = 0; i < n; ++i) {
    if (i <= r) {
      z[i] = min(r - i + 1, z[i - l]);
    }
    while (i + z[i] < n && s[z[i]] == s[i + z[i]]) {
      ++z[i];
    }
    if (i + z[i] - 1 > r) {
      l = i;
      r = i + z[i] - 1;
    }
  }
  return z;
}
pair<long long, long long> find_max_podotrsok(vector<long long> a) {
  long long ans = a[0], ans_l = 0, ans_r = 0, sum = 0, minus_pos = -1;
  for (long long r = 0; r < a.size(); ++r) {
    sum += a[r];
    if (sum > ans) {
      ans = sum;
      ans_l = minus_pos + 1;
      ans_r = r;
    }
    if (sum == ans && ans_r == r) {
      ans_l = max(ans_l, minus_pos + 1);
    }
    if (0 >= sum) {
      sum = 0;
      minus_pos = r;
    }
  }
  return {ans_l, ans_r};
}
bool is_prost(long long n) {
  if (n == 0) {
    return true;
  }
  long long i = 2;
  while (i * i <= n) {
    if (n % i == 0) {
      return false;
    }
    ++i;
  }
  return true;
}
vector<long long> used, a, b, ans, dop_ans;
vector<vector<long long> > g;
long long ans_chislo = 0;
void dfs(long long v, long long p) {
  if (used[v]) {
    return;
  }
  used[v] = 1;
  for (auto to : g[v]) {
    if (to == p) {
      continue;
    }
    dfs(to, v);
  }
  if (a[v] >= 0) {
    ans.push_back(v);
    ans_chislo += a[v];
    if (b[v] != -1) {
      a[b[v]] += a[v];
    }
  } else {
    dop_ans.push_back(v);
  }
}

  }
  g.resize(n);
  vector<long long> moget_byt_otprav(n, 1);
  for (long long i = 0; i < n; ++i) {
    cin >> b[i];
    if (b[i] != -1) {
      g[--b[i]].push_back(i);
      moget_byt_otprav[i] = 0;
    }
  }
  used.resize(n, 0);
  for (long long i = 0; i < n; ++i) {
    if (!used[i] && moget_byt_otprav[i]) {
      dfs(i, -1);
    }
  }
  reverse(dop_ans.begin(), dop_ans.end());
  for (auto y : dop_ans) {
    ans.push_back(y);
    ans_chislo += a[y];
  }
  cout << ans_chislo << '\n';
  for (auto i : ans) {
    cout << ++i << ' ';
  }
}
