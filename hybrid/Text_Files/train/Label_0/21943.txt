typedef long long ll;
typedef pair<int, int> pint;

const int dx[] = { 0, 1, 0, -1 };
const int dy[] = { 1, 0, -1, 0 };


const ll inf = ten(9);
class SegTree
{
private:
    vector<vector<ll>> data;
    int n;
 
    ll _query(int a, int b, int k, int l, int r, ll d)
    {
        if (r <= a || b <= l)
            return inf;
 
        if (a <= l && r <= b)
        {
            auto it = upper_bound(all(data[k]), d);
            ll diff = inf;
            ll x = *it - d;
            ll y = d - *--it;
            assert(x >= 0);
            assert(y >= 0);
            return min(x, y);
        }
        else
        {
            ll left = _query(a, b, 2 * k + 1, l, (l + r) / 2, d);
            ll right = _query(a, b, 2 * k + 2, (l + r) / 2, r, d);
            return min(left, right);
        }
    }

    void update(int k, ll v)
    {
        k += n - 1;
        data[k].push_back(v);
        while (k > 0)
        {
            k = (k - 1) / 2;
            data[k].push_back(v);
        }
    }
public:
 
    SegTree(const vector<ll>& a)
    {
        int n = a.size();
        this->n = 1;
        while (this->n < n)
            this->n *= 2;
        data = vector<vector<ll>>(this->n * 2, {-inf, inf});

        rep(i, n)
            update(i, a[i]);

        for (auto& v : data)
        {
//             dump(v);
            sort(all(v));
        }
    }

    ll query(int a, int b, ll d) { return _query(a, b, 0, 0, n, d); }
};
int main()
{
    int n;
    scanf("%d", &n);
    vector<ll> a(n);
    rep(i, n)
        scanf("%lld", &a[i]);

    SegTree seg(a);

    int q;
    scanf("%d", &q);
    rep(qi, q)
    {
        int l, r, d;
        scanf("%d%d%d", &l, &r, &d);

        ll diff = seg.query(l, r + 1, d);
        assert(diff < ten(8));
        cout << diff << endl;
    }
}
