import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Scanner;

public class PriorityQueue {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        PrintWriter pw = new PrintWriter(System.out);
        int t = 1;
        for (int i = 0; i < t; i++) {
            solve(sc, pw);
        }
        pw.close();
    }
    static int mod = 998244353;
    static void solve(Scanner in, PrintWriter out){
        int n = in.nextInt();
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            char c = in.next().charAt(0);
            if (c == '+'){
                int v = in.nextInt();
                arr[i] = new int[]{1, v};
            }else{
                arr[i] = new int[]{-1, 0};
            }
        }
        long cnt = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i][0] == -1) continue;
            long[][] dp = new long[501][501];
            for (int j = 0; j < dp.length; j++) {
                Arrays.fill(dp[j], -1);
            }
            cnt = (cnt + (dfs(0, 0, arr[i][1], i, n, dp, arr) * 1l * arr[i][1]) % mod) % mod;
        }
        out.println(cnt);
    }

    static long dfs(int idx, int curSmall, int compareNumber, int cmpidx, int n, long[][] dp, int[][] arr){
        if (idx == n) return 1;
        int former = curSmall;
        if (dp[idx][curSmall] != -1) return dp[idx][curSmall];
        long res;
        if (idx == cmpidx){
            return dfs(idx + 1, curSmall, compareNumber, cmpidx, n, dp, arr);
        }else if (idx < cmpidx){
            res = dfs(idx + 1, curSmall, compareNumber, cmpidx, n, dp, arr);
            if (arr[idx][0] == -1){
                res = (res + dfs(idx + 1, Math.max(0, curSmall - 1), compareNumber, cmpidx, n, dp, arr)) % mod;
            }else{
                if (arr[idx][1] <= compareNumber) {
                    curSmall++;
                }
                res = (res + dfs(idx + 1, curSmall, compareNumber, cmpidx, n, dp, arr)) % mod;
            }
        }else{
            res = dfs(idx + 1, curSmall, compareNumber, cmpidx, n, dp, arr);
            if (arr[idx][0] == -1){
                if (curSmall > 0){
                    res = (res + dfs(idx + 1, curSmall - 1, compareNumber, cmpidx, n, dp, arr)) % mod;
                }
            }else{
                if (arr[idx][1] < compareNumber) {
                    curSmall++;
                }
                res = (res + dfs(idx + 1, curSmall, compareNumber, cmpidx, n, dp, arr)) % mod;
            }
        }
        return dp[idx][former] = res;
    }
}
