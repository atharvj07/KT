#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, ln, i, j, ck = 0;
  cin >> n;
  int* arr = new int[n];
  for (i = 0; i < n; i++) cin >> arr[i];
  sort(arr, arr + n);
  if (n % 2 != 0)
    cout << arr[n / 2] << endl;
  else {
    ck = (n / 2) - 1;
    cout << arr[ck] << endl;
  }
  return 0;
}
