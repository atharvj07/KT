from sys import stdin, stdout
readline = stdin.readline
write = stdout.write
s = input()
M = []
U = []
for (i, ch) in enumerate(s):
	if ch == '0':
		if U:
			z = U.pop()
			z.append(i + 1)
			M.append(z)
		else:
			M.append([i + 1])
	elif not M:
		print(-1)
		exit()
	else:
		z = M.pop()
		z.append(i + 1)
		U.append(z)
if U:
	print(-1)
	exit()
print(len(M))
for z in M:
	write(str(len(z)) + ' ' + ' '.join((str(i) for i in z)) + '\n')
