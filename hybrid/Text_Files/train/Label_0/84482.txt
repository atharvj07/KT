import sys
input = sys.stdin.readline
from collections import Counter

def is_palind(s):
	if len(s) <= 1:
		return True
	f = s[:len(s) // 2]
	b = s[(len(s) + 1) // 2:][::-1]
	return b == f
t = int(input())
orda = ord('a')
for _ in range(t):
	s = input().rstrip()
	if s.count('a') == len(s):
		print('NO')
		continue
	t1 = 'a' + s
	print('YES')
	if is_palind(t1):
		print(s + 'a')
	else:
		print(t1)
