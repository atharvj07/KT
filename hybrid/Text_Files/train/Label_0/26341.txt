#include <bits/stdc++.h>
using namespace std;
const double EPS = 1e-9;
const double PI = 3.141592653589793238462;
const int dr[] = {1, 1, 0, -1, -1, -1, 0, 1};
const int dc[] = {0, 1, 1, 1, 0, -1, -1, -1};
const int dx[] = {1, 0, -1, 0};
const int dy[] = {0, 1, 0, -1};
unsigned long long a[100005], b[100005], dp[100005];
vector<unsigned long long> M, B;
bool bad(int l1, int l2, int l3) {
  return (double)((B[l3] - B[l1]) * (double)(M[l1] - M[l2])) <
         (double)((B[l2] - B[l1]) * (double)(M[l1] - M[l3]));
}
unsigned long long eval(int ind, unsigned long long x) {
  return M[ind] * x + B[ind];
}
void add(unsigned long long m, unsigned long long b) {
  M.push_back(m);
  B.push_back(b);
  while (M.size() > 2 && bad(M.size() - 3, M.size() - 2, M.size() - 1)) {
    M.erase(M.end() - 2);
    B.erase(B.end() - 2);
  }
}
int pointer;
unsigned long long query(unsigned long long x) {
  if (pointer >= M.size()) pointer = M.size() - 1;
  while (pointer < M.size() - 1 && eval(pointer + 1, x) < eval(pointer, x)) {
    pointer++;
  }
  return eval(pointer, x);
}
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  dp[0] = 0;
  add(b[0], dp[0]);
  for (int i = 1; i < n; i++) {
    dp[i] = query(a[i]);
    add(b[i], dp[i]);
  }
  cout << dp[n - 1] << '\n';
  return 0;
}
