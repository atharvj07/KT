n = int(input())
if n % 2 == 0:
	b = int(4)
	a = n + 4
	print(a, b)
else:

	def comp(x):
		for i in range(2, int(x / 2)):
			if x % i == 0:
				return 1
				break
		return 0
	b = int(4)
	while 1:
		a = b + n
		cp = comp(a)
		if cp == 1:
			print(a, b)
			break
		b += 1
		while 1:
			if comp(b) == 0:
				b += 1
			else:
				break
