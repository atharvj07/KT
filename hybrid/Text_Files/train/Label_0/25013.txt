#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
const int N = 2e5;
ll fact[N];
ll bpow(ll a, int n) {
  ll res = 1;
  while (n > 0) {
    if (n & 1) {
      res *= a;
      --n;
      res %= mod;
    } else {
      n /= 2;
      a *= a;
      a %= mod;
    }
  }
  return res;
}
ll C(int n, int k) {
  if (n == k) return 1;
  ll a = fact[n];
  ll b = fact[k] * fact[n - k];
  b %= mod;
  return (a * bpow(b, mod - 2)) % mod;
}
int main() {
  ios::sync_with_stdio(0);
  cin.tie(nullptr);
  vector<int> primes;
  vector<bool> f(71, true);
  fact[0] = 1;
  for (int i = 1; i < N; ++i) {
    fact[i] = fact[i - 1] * i;
    fact[i] %= mod;
  }
  for (int i = 2; i < f.size(); ++i) {
    if (f[i]) {
      primes.push_back(i);
      for (int j = i * i; j < f.size(); j += i) {
        f[j] = false;
      }
    }
  }
  int n;
  cin >> n;
  map<int, int> m;
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    int mask = 0;
    for (int j = 0; j < primes.size(); ++j) {
      while (x % primes[j] == 0) {
        x /= primes[j];
        mask ^= (1 << j);
      }
    }
    ++m[mask];
  }
  vector<pair<int, ll>> a;
  for (auto w : m) {
    a.push_back(w);
  }
  vector<ll> st(n + 1);
  st[0] = 1;
  for (int i = 1; i <= n; ++i) {
    st[i] = st[i - 1] * 2;
    st[i] %= mod;
  }
  vector<ll> dp(1 << primes.size());
  for (int i = 0; i < a.size(); ++i) {
    ll x = 0, y = 0;
    for (int j = 1; j <= a[i].second; ++j) {
      if (j & 1) {
        x = (x + C(a[i].second, j)) % mod;
      } else {
        y = (y + C(a[i].second, j)) % mod;
      }
    }
    auto dp1 = dp;
    for (int j = 0; j < (1 << primes.size()); ++j) {
      dp1[a[i].first ^ j] += (x * dp[j]) % mod;
      dp1[a[i].first ^ j] %= mod;
      dp1[j] += (y * dp[j]) % mod;
      dp1[j] %= mod;
    }
    dp = dp1;
    dp[0] += y;
    dp[a[i].first] += x;
    dp[0] %= mod;
    dp[a[i].first] %= mod;
  }
  cout << dp[0] << '\n';
}
