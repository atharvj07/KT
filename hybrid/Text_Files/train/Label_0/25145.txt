#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
    int n,even,odd;
    cin>>n;
    vector<int>e(100001,0),o(100001,0);
    for(int i=0;i<n/2;i++){cin>>even>>odd;e[even]++,o[odd]++;}
    auto m_e=max_element(e.begin(),e.end()),m_o=max_element(o.begin(),o.end());
    if(m_e-e.begin()!=m_o-o.begin())cout<<n-*m_e-*m_o<<endl;
    else{
        even=*m_e,odd=*m_o;
        *m_e=0,m_e=max_element(e.begin(),e.end());
        *m_o=0,m_o=max_element(o.begin(),o.end());
        cout<<n-max(*m_e+odd,even+*m_o)<<endl;
    }
    return 0;
}
