(x, y) = [int(i) for i in input().split()]
if x >= 0:
	if y > x:
		zone = 3
	elif y <= x and y > -x + 1:
		zone = 2
	else:
		zone = 1
elif y >= -x:
	zone = 3
elif y < -x and y >= x:
	zone = 4
else:
	zone = 1
if zone == 1:
	ans = -y * 4
elif zone == 2:
	ans = 1 + (x - 1) * 4
elif zone == 3:
	ans = 2 + (y - 1) * 4
else:
	ans = 3 + -1 * (x + 1) * 4
print(ans)
