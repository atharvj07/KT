#include <bits/stdc++.h>
using namespace std;
int a[((int)(1e5 + 5))];
int main() {
  int n, k;
  scanf("%d %d", &n, &k);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  a[n] = a[n - 1];
  int ans = 1, pre;
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] == a[i + 1])
      pre = 1;
    else {
      ans = max(ans, pre + 1);
      pre++;
    }
  }
  printf("%d\n", ans);
  return 0;
}
