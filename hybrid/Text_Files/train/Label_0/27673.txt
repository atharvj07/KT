#include <bits/stdc++.h>
using namespace std;
#define inf 1e9
int x[100050],ans=inf;
int main() {
    int n,k;
    cin >> n >> k;
    for (int i=1;i<=n;i++) cin >> x[i];
    for (int i=1;i<=n-k+1;i++) {
        int dis = min(abs(x[i]),abs(x[i+k-1])) + x[i+k-1] - x[i];
        if (dis<ans) ans = dis;
    }
    cout << ans << endl;
    return 0;
}
