#include <bits/stdc++.h>
using namespace std;
int divp(int n, int p, string ch) {
  int i = p;
  int j = 0;
  while (i <= n) {
    while (j < i) {
      cout << ch[j];
      j++;
    }
    cout << endl;
    j = i;
    i += p;
  }
  return 0;
}
int divq(int n, int q, string ch) {
  int i = q;
  int j = 0;
  while (i <= n) {
    while (j < i) {
      cout << ch[j];
      j++;
    }
    cout << endl;
    j = i;
    i += q;
  }
  return 0;
}
int main() {
  int n, p, q, i, j;
  string ch, ch1;
  cin >> n >> p >> q;
  cin >> ch;
  if (!(n % p)) {
    cout << (n / p) << endl;
    divp(n, p, ch);
  } else if (!(n % q)) {
    cout << (n / q) << endl;
    divq(n, q, ch);
  } else if (!((n % p) % q)) {
    cout << (n / p) + ((n % p) / q) << endl;
    divp(((n / p) * p), p, ch);
    for (int j = ((n / p) * p); j <= n; j++) ch1.push_back(ch[j]);
    divq(n % p, q, ch1);
  } else {
    for (i = 0; (i <= n / p); i++) {
      if (!((n - i * p) % q)) {
        j = (n - i * p) / q;
        break;
      }
    }
    if ((i * p + q * j) > n)
      cout << "-1" << endl;
    else {
      cout << i + j << endl;
      divp((i * p), p, ch);
      for (int c = (i * p); c <= n; c++) ch1.push_back(ch[c]);
      divq((j * q), q, ch1);
    }
  }
  return 0;
}
