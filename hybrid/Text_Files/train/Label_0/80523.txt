def lock(ar, i, sm, n):
	if i == n:
		if sm == 0 or sm % 360 == 0:
			return True
		return False
	return lock(ar, i + 1, sm + ar[i], n) | lock(ar, i + 1, sm - ar[i], n)
n = int(input())
ar = []
for i in range(n):
	ar.append(int(input()))
b = lock(ar, 0, 0, n)
if b == False:
	print('NO')
else:
	print('YES')
