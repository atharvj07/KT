t = int(input())

def ans(n, k, p):
	res = 0
	x = 0
	den = p[0]
	i = 1
	while i < n:
		if p[i] / den > k / 100:
			if p[i] * 100 % k == 0:
				x = 100 * p[i] // k - den
			else:
				x = 100 * p[i] // k - den + 1
			res += x
			den += x
		den += p[i]
		i += 1
	return int(res)
for i in range(t):
	a = list(map(int, input().strip().split()))[:2]
	n = a[0]
	k = a[1]
	p = list(map(int, input().strip().split()))[:n]
	pj = ans(n, k, p)
	print(pj)
