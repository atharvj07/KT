import java.util.*;
public class Main {

    public static void main(String[] args) {

        Scanner scan = new Scanner(System.in);

        long a,b;
        a= scan.nextLong();
        b= scan.nextLong();
        long counter=0;
        long temp;

        while( b % a != 0  )
        {
            counter+= a/b;
            a = a%b;
            if(b > a)
            {
                temp =a;
                a=b;
                b= temp;
            }
        }

        if(a==b)
            counter++;

        System.out.println(counter);
    }
}
