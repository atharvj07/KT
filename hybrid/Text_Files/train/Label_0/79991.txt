import sys
sys.setrecursionlimit(2147483647)
input = sys.stdin.readline
import math

def calc(arr):
	(m, d, w) = arr
	max_x = min(m, d)
	if (d - 1) % w == 0:
		return max_x * (max_x - 1) // 2
	else:
		s = math.gcd(d - 1, w)
		w = w // s
		res = max_x % w
		b = max_x // w
		ans = b * (b - 1) // 2 * w + res * b
		return ans
(n, arr) = (int(input()), [])
for _ in range(n):
	arr.append(calc(list(map(int, input().split(' ')))))
for a in arr:
	print(a)
