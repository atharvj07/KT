#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  map<string, int> mpa, mpb;
  cin >> n >> m;
  string str;
  for (int i = 0; i < n; i++) {
    cin >> str;
    mpa[str]++;
  }
  for (int i = 0; i < m; i++) {
    cin >> str;
    mpb[str]++;
  }
  if (n > m) {
    cout << "YES" << endl;
  } else if (n < m)
    cout << "NO" << endl;
  else {
    int c = 0;
    for (auto it1 = mpa.begin(), it2 = mpb.begin(); it1 != mpa.end();
         it1++, it2++) {
      if (it1->second == mpb[it1->first]) c++;
    }
    if (c % 2)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
