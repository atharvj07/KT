import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;
public class Main {
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Integer> input =  new ArrayList<Integer>();
        input.clear();
        List<Integer> first = new  ArrayList<Integer>();
        List<Integer> second = new ArrayList<Integer>();
        input.clear();
        for (int i = 0; i < n; i ++) {
            input.add(scanner.nextInt());
        }
        if (input.size() == 2) {
            System.out.println(input.get(0));
            System.out.println(input.get(1));
        } else {
            boolean flag = false;
            if (solve(input, first, second, input.get(1) - input.get(0), 0)) {
                flag = true;
            }
            if (flag == false) {
                flag = solve(input, first, second, input.get(2) - input.get(1), 1);
            }
            if (flag == false) {
                flag = solve(input, first, second, input.get(2) - input.get(0), 0);
            }
            if (flag) {
                print(first);
                print(second);
            } else {
                System.out.println("No solution");
            }
        }
    }
    public static void print(List<Integer> list) {
        for (int i = 0; i < list.size(); i ++) {
            if (i > 0) System.out.print(" ");
            System.out.print(list.get(i));
        }
        System.out.println();
    }
    public static boolean solve(List<Integer> input, List<Integer> first,List<Integer> second,int num,int start) {

        first.clear();
        second.clear();
        for (int i = 0; i < start; i ++) second.add(input.get(i));
        first .add(input.get(start));
        int pre = input.get(start);        
        boolean res = true;
        int lastPos = start;
        for (int i = start + 1; i < input.size(); i ++) {
            if (input.get(i) == pre + num) {
                first.add(input.get(i));
                pre = input.get(i);
                lastPos = i;
            } else {
                second.add(input.get(i));
            }
        }
        if (!isIt(second)) {
            first.clear();
            second.clear();
            //System.out.println(lastPos);
            for (int i = 0; i < start; i ++) second.add(input.get(i));
            first .add(input.get(start));
            pre = input.get(start);        
            for (int i = start + 1; i < input.size(); i ++) {
                if (i == lastPos) {
                    second.add(input.get(i));                
                    lastPos = -1;
                    continue;
                }
                if (lastPos == -1)
                    second.add(input.get(i));
                else {
                    if (input.get(i) == pre + num) {
                        first.add(input.get(i));
                        pre = input.get(i);
                       // lastPos = i;
                    } else {
                        second.add(input.get(i));
                    }
                }
            }
            
        }
        return isIt(second) && isIt(first);
    }
    
    public static  boolean isIt(List<Integer> list) {
        if (list.size() == 0) return false;
        else if (list.size() == 1 || list.size() == 2) return true;
        else {
            int num = list.get(1) - list.get(0);
            for (int i = 2; i < list.size(); i ++) {
                if (list.get(i ) != list.get(i - 1) + num) return false; 
            }
            return true;
        }
    }

}
