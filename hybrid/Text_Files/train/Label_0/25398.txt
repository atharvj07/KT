#include <bits/stdc++.h>
int main() {
  char c[101];
  int i, d, n = 1;
  scanf("%s", c);
  for (i = 1; c[i] != '\0'; i++) {
    for (d = 0; d < i; d++) {
      if (c[i] == c[d])
        break;
      else
        continue;
    }
    if (d == i) n++;
  }
  if (n % 2 == 0)
    printf("CHAT WITH HER!");
  else
    printf("IGNORE HIM!");
}
