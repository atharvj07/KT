//package ACM_code;

import java.util.Scanner;
public class Main{
    public static void main(String[] args){
    Scanner cin = new Scanner(System.in);
    long n = cin.nextLong();
    long m = cin.nextLong();
    long k = cin.nextLong();
    long l = cin.nextLong();
    long ans = (l+k)/m;
    if(ans*m<l+k) ans++;
    if(ans*m<=n||ans==0) System.out.println(ans);
    else System.out.println(-1);
    }
}
