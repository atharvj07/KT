#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int max_n=100005;
int n,a[max_n],b[max_n],c[max_n];
int main(){
    scanf("%d",&n);
    for(int i=1;i<=n;i++)scanf("%d",a+i);
    for(int i=1;i<=n;i++)scanf("%d",b+i);
    for(int i=1;i<=n;i++)scanf("%d",c+i);
    sort(a+1,a+n+1),sort(b+1,b+n+1),sort(c+1,c+n+1);
    ll ans=0;
    for(int i=1;i<=n;i++){
        int x=lower_bound(a+1,a+n+1,b[i])-a-1,y=upper_bound(c+1,c+n+1,b[i])-c;
        y=n-y+1;
        ans+=1ll*x*y;
    }
    printf("%lld\n",ans);
    return 0;
}
