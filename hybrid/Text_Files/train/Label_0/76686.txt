a = input()
list1 = a.split(' ')
n = int(list1[0])
m = int(list1[1])
i = 1
teleportlocation = [0]
teleportrange = [0]
while i <= n:
	tempstring = input()
	list1 = tempstring.split(' ')
	list1[0] = int(list1[0])
	list1[1] = int(list1[1])
	teleportlocation.insert(i, list1[0])
	teleportrange.insert(i, list1[1])
	i = i + 1
testlist = [0] * m
i = 1
while i <= n:
	x = teleportlocation[i]
	y = teleportrange[i]
	testlist[x:y] = [1] * (y - x)
	i = i + 1
final = 1
for i in testlist:
	if i == 0:
		final = 0
		break
if final == 0:
	print('NO')
else:
	print('YES')
