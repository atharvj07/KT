def main():
	n = int(input())
	mines = []
	(minx, miny, maxx, maxy) = (1000000000.0, 11000000000.0, -1000000000.0, -1000000000.0)
	for _ in range(n):
		(x, y) = map(int, input().split())
		minx = min(minx, x)
		maxx = max(maxx, x)
		miny = min(miny, y)
		maxy = max(maxy, y)
	sidex = maxx - minx
	sidey = maxy - miny
	res = max(sidex, sidey) ** 2
	print(res)
main()
