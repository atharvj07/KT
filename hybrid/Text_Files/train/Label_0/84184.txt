from sys import stdin, stdout
N = 998244353

def egcd(a, b):
	if a == 0:
		return (b, 0, 1)
	else:
		(g, y, x) = egcd(b % a, a)
		return (g, x - b // a * y, y)

def modinv(a, m=998244353):
	(g, x, y) = egcd(a, m)
	if g != 1:
		raise Exception('modular inverse does not exist')
	else:
		return x % m
n = int(stdin.readline())
if n == 1:
	domino = stdin.readline().strip()
	if domino[0] == '?' and domino[1] == '?':
		stdout.write('2\n')
	elif domino[0] == '?' or domino[1] == '?':
		stdout.write('1\n')
	elif domino == 'BB' or domino == 'WW':
		stdout.write('0\n')
	else:
		stdout.write('1\n')
else:
	count = {'BB': 0, 'BW': 0, 'B?': 0, 'WB': 0, 'WW': 0, 'W?': 0, '?B': 0, '?W': 0, '??': 0}
	for _ in range(n):
		count[stdin.readline().strip()] += 1
	count_b = 2 * count['BB'] + count['BW'] + count['WB'] + count['B?'] + count['?B']
	count_w = 2 * count['WW'] + count['BW'] + count['WB'] + count['W?'] + count['?W']
	if count_b > n or count_w > n:
		stdout.write('0\n')
	else:
		answer = 1
		top = 2 * n - count_b - count_w
		bottom = 1
		while bottom <= n - count_w:
			answer = answer * top * modinv(bottom) % N
			top -= 1
			bottom += 1
		if count['BB'] == 0 and count['WW'] == 0:
			deduction = 1
			for i in range(count['??']):
				deduction = deduction * 2 % N
			if count['BB'] + count['BW'] + count['B?'] + count['WW'] + count['?W'] == 0:
				deduction -= 1
			if count['WW'] + count['WB'] + count['W?'] + count['BB'] + count['?B'] == 0:
				deduction -= 1
			answer = (answer - deduction) % N
		stdout.write(str(answer) + '\n')
