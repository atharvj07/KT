import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;

public class C {

  private final static class P {
    int w, h, a; P(int aw, int ah) {w = aw; h = ah; a = w * h;}
  }

  private void solve() {
    int n = ni();
    int m = ni();
    char[][] grid = new char[n][];
    for (int i = 0; i < n; ++i)
      grid[i] = nt().toCharArray();
    int prec[][] = new int[n + 1][m + 1];
    int i0 = 0, j0 = 0;
    boolean ok = true;
    for (int i = 0; i < n; ++i)
      for (int j = 0; j < m; ++j) {
        prec[i + 1][j + 1] = prec[i][j + 1] + prec[i + 1][j] - prec[i][j];
        if (grid[i][j] == 'X') {
          if (ok) {
            i0 = i;
            j0 = j;
            ok = false;
          }
          ++prec[i + 1][j + 1];
        }
      }

    int h = 1, w = 1;
    while (i0 + h < n && grid[i0 + h][j0] == 'X') h++;
    while (j0 + w < m && grid[i0][j0 + w] == 'X') w++;

    if (count(prec, i0, j0, w, h) != w * h) { prln("-1"); return; }
    int bi = i0;
    int bj = j0;
    PriorityQueue<P> pq = new PriorityQueue<P>(h + w + 10, new Comparator<P>() {
      public int compare(P o1, P o2) { return  o1.a - o2.a;} });
    for (int i = 1; i <= h; ++i) pq.add(new P(w, i));
    for (int i = 1; i <= w; ++i) pq.add(new P(i, h));

    while (!pq.isEmpty()) {
      P p = pq.poll();
      w = p.w;
      h = p.h;
      int cnt = w * h;
      j0 = bj;
      i0 = bi;
      while (true) {
        if (i0 + h < n && grid[i0 + h][j0] == 'X') {
          if (count(prec, i0 + h , j0, w, 1) != w) break;
          cnt += w;
          ++i0;
        } else if (j0 + w < m && grid[i0][j0 + w] == 'X') {
          if (count(prec, i0, j0 + w, 1, h) != h) break;
          cnt += h;
          ++j0;
        } else
          break;
      }

      if (cnt == prec[n][m]) {
        prln(w * h);
        return;
      }
    }
    prln("-1");
  }

  private int count(int[][] gr, int i, int j, int w, int h) {
    return gr[i + h][j + w] - gr[i + h][j] - gr[i][j + w] + gr[i][j];
  }

  public static void main(String ... args) { new C().run(); }

  private PrintWriter pw;
  private BufferedReader br;
  private StringTokenizer st;

  private void run() {
    pw = new PrintWriter(System.out);
    br = new BufferedReader(new InputStreamReader(System.in));
    solve();
    try {
      br.close();
    } catch (IOException e) {
      e.printStackTrace();
      return;
    }
    pw.close();
  }

  private void pr(Object o) { pw.print(o); }
  private void prln(Object o) {pw.println(o);}
  private void prln() {pw.println();}

  private long nl() { return Long.parseLong(nt());	}

  private double nd() { return Double.parseDouble(nt()); }

  private int ni() { return Integer.parseInt(nt()); }

  private int[] nia(int n) {
    int res[] = new int[n];
    for (int i = 0; i < n; ++i) res[i] = ni() - 1; return res;
  }

  private long[] nla(int n) {
    long res[] = new long[n];
    for (int i = 0; i < n; ++i) res[i] = nl(); return res;
  }

  private String[] nsa(int n) {
    String[] res = new String[n];
    for (int i = 0; i < n; ++i) res[i] = nt(); return res;
  }

  private String nt() {
    if (st == null || !st.hasMoreTokens())
      try {
        st = new StringTokenizer(br.readLine());
      } catch (IOException e) {
        System.exit(666);
      }
    return st.nextToken();
  }
}
