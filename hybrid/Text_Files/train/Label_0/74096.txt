for _ in range(int(input())):
	n = int(input())
	a = list(map(int, input().split()))
	if n <= 2:
		print(0)
		continue
	flip = 0
	last = 1
	if a[-2] - a[-1] < 0:
		flip = 1
		last = -1
	ans = 0
	for i in range(n - 3, -1, -1):
		if a[i] - a[i + 1] > 0 and last > 0:
			continue
		elif a[i] - a[i + 1] < 0 and last > 0:
			flip += 1
			last = -1
		elif a[i] - a[i + 1] > 0:
			flip += 1
		elif a[i] == a[i + 1]:
			continue
		if flip > 1:
			ans = i + 1
			break
	print(ans)
