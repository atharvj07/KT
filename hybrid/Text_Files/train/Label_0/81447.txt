import sys
input = sys.stdin.readline
out = sys.stdout

def main():
	n = int(input())
	a = list(map(int, input().split()))
	dp = {a[i]: 0 for i in range(n)}
	(ans, lst) = (0, 0)
	for i in range(n):
		x = a[i]
		if x - 1 in dp:
			dp[x] = dp[x - 1] + 1
		else:
			dp[x] = 1
		if ans < dp[x]:
			ans = dp[x]
			lst = x
	res = []
	for i in range(n - 1, -1, -1):
		if a[i] == lst:
			res.append(i)
			lst -= 1
	res.sort()
	print(ans)
	for k in res:
		out.write(str(k + 1) + ' ')
	return
main()
