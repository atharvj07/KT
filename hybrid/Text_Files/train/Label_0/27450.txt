#include <bits/stdc++.h>
using namespace std;

const long long MAX_N = 1e5 + 5e4 + 5, inf = 1e9 + 7;
long long n;
long long Arr[MAX_N];
long long P[MAX_N];
long long psum[MAX_N];
long long cnt[MAX_N];
long long dp[MAX_N];

void process(long long d) {
	memset(cnt, 0, sizeof cnt);
	memset(dp, 0, sizeof dp);
	for (long long i = 1; i <= n; i++) {
		P[i] = (d <= Arr[i]? +1: -1);
		psum[i] = P[i] + psum[i - 1];
	}
}

long long calc(long long d) {
	long long res = 0;
	cnt[50000] = 1;
	
	for (long long i = 1; i <= n; i++) {
		dp[i] = dp[i - 1];
		if (P[i] == +1)
			dp[i] += cnt[50000 + psum[i - 1]];
		else
			dp[i] -= cnt[50000 + psum[i]];
			
		res += dp[i] + cnt[50000+ psum[i]];
		cnt[50000+ psum[i]]++;
	}
	
	return res;
}

int main() {
	scanf("%d", &n);
	for (long long i = 1; i <= n; i++)
		scanf("%d", &Arr[i]);
	
	long long s = 1, e = inf;
	while (e - s > 1) {
		long long mid = e + s >> 1;
		
		process(mid);
		if (calc(mid) < ((((n * (n + 1)) >> 1) + 1) >> 1))
			e = mid;
		else
			s = mid;
	}
	
	printf("%d\n", s);
	return 0;
}
