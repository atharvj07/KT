import sys
input = sys.stdin.readline
for _ in range(int(input())):
	s = input().strip()
	n = len(s)
	odd = [0]
	even = [0]
	for i in range(n):
		if s[i] == '[' or s[i] == ']':
			if i % 2 == 0:
				even.append(even[-1] + 1)
				odd.append(odd[-1])
			else:
				odd.append(odd[-1] + 1)
				even.append(even[-1])
		else:
			even.append(even[-1])
			odd.append(odd[-1])
	for query in range(int(input())):
		(l, r) = [int(x) for x in input().split(' ')]
		oddS = odd[r] - odd[l - 1]
		evenS = even[r] - even[l - 1]
		print(abs(oddS - evenS))
