#include <bits/stdc++.h>
using namespace std;
long long f[2001][2001], g[2001][2001], n, m, i, j, kq;
const long long mod = 998244853;
int main() {
  cin >> n >> m;
  f[0][0] = g[0][0] = 1;
  for (i = 0; i <= n; i++)
    for (j = 0; j <= m; j++)
      if (i - j > 0) {
        if (i > 0) f[i][j] = (f[i][j] + f[i - 1][j]) % mod;
        if (j > 0) f[i][j] = (f[i][j] + f[i][j - 1]) % mod;
      } else {
        if (i > 0) g[i][j] = (g[i][j] + g[i - 1][j]) % mod;
        if (j > 0) g[i][j] = (g[i][j] + g[i][j - 1]) % mod;
      }
  for (i = 0; i <= n; i++)
    for (j = 0; j <= m; j++)
      kq = (kq + (i - j) * f[i][j] % mod * g[n - i][m - j] % mod) % mod;
  cout << kq;
}
