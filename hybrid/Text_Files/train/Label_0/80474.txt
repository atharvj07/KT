(r, c) = [int(i) for i in input().split(' ')]
ls = list()
for i in range(r):
	ls_ = [-(c + r) if s == 'S' else 1 for s in input()]
	ls.append(ls_)
result = 0
for i in range(r):
	sum_ = sum([s_ for s_ in ls[i]])
	if sum_ < 0:
		continue
	result += sum_
	for j in range(c):
		if ls[i][j] >= 0:
			ls[i][j] = 0
for i in range(c):
	sum_ = sum([ls[s_][i] for s_ in range(r)])
	if sum_ < 0:
		continue
	result += sum_
	for j in range(r):
		if ls[j][i] >= 0:
			ls[j][i] = 0
print(result)
