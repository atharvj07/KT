n = int(input())
a = list(map(int, input().split()))
o = []
t = []
if a[0] == 1:
	o.append(1)
	t.append(0)
else:
	o.append(0)
	t.append(1)
y = 1
while y < len(a):
	if a[y] == 1:
		if a[y - 1] == 1:
			o.append(o[y - 1] + 1)
		else:
			o.append(1)
		t.append(0)
	else:
		if a[y - 1] == 2:
			t.append(t[y - 1] + 1)
		else:
			t.append(1)
		o.append(0)
	y += 1
c1 = 0
c2 = 0
y = 1
while y < len(a):
	if o[y] != 0 and c1 == 0:
		c1 = 1
	if t[y] != 0 and c2 == 0:
		c2 = 1
	if o[y] == 0 and c1 != 0:
		o[y] = o[y - 1]
	if t[y] == 0 and c2 != 0:
		t[y] = t[y - 1]
	y += 1
y = 0
mx = 2
while y < len(a):
	mx = max(mx, min(o[y], t[y]) * 2)
	y += 1
print(mx)
