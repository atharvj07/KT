#include <bits/stdc++.h>
using namespace std;
long long t, n, arr[200005];
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    long long has = 0;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    sort(arr + 1, arr + n + 1);
    long long a = 1, b = 1, c = 1;
    for (int i = n; i >= n - 4; i--) a *= arr[i];
    for (int i = 1; i <= 2; i++) b *= arr[i];
    for (int i = n; i >= n - 2; i--) b *= arr[i];
    for (int i = 1; i <= 4; i++) c *= arr[i];
    c *= arr[n];
    has = max(a, max(b, c));
    cout << has << endl;
  }
}
