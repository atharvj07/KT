import sys
input = sys.stdin.readline
for f in range(int(input())):
	n = int(input())
	miners = []
	mines = []
	for i in range(2 * n):
		(x, y) = map(int, input().split())
		if x:
			miners.append(abs(x))
		else:
			mines.append(abs(y))
	miners.sort()
	mines.sort()
	d = 0
	for i in range(n):
		d += (miners[i] ** 2 + mines[i] ** 2) ** 0.5
	print(d)
