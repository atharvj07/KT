def solve(cake):
	(rows, collumns) = (set(), set())
	for (i, row) in enumerate(cake):
		for (j, cell) in enumerate(row):
			if cell == 'S':
				rows.add(i)
				collumns.add(j)
	res = 0
	for (i, row) in enumerate(cake):
		for (j, cell) in enumerate(row):
			if cell == 'S':
				continue
			if i not in rows or j not in collumns:
				res += 1
	return res
(r, c) = map(int, input().split())
l = []
for _ in range(r):
	l.append(input())
print(solve(l))
