#include <bits/stdc++.h>
using namespace std;
long long INF = 1e9;
struct P {
  long long x, y;
  P(long long _x = 0, long long _y = 0) : x(_x), y(_y) {}
  bool operator<(const P& rhs) const {
    if (x != rhs.x) return x < rhs.x;
    return y > rhs.y;
  }
};
P p[100005];
int n;
long long prefy[100005];
long long prefY[100005];
long long suffy[100005];
long long suffY[100005];
bool ok(long long k) {
  if (min((prefy[n] - prefY[n]) * (prefy[n] - prefY[n]),
          (p[n].x - p[1].x) * (p[n].x - p[1].x)) <= k)
    return 1;
  long long y;
  long long Y;
  long long ans = 1e17;
  for (int i = 1; i <= n; ++i) {
    long long tmp = 0;
    y = min(prefy[i - 1], suffy[i + 1]);
    Y = max(prefY[i - 1], suffY[i + 1]);
    tmp = max(tmp, (Y - y) * (Y - y));
    tmp = max(tmp, p[i].x * p[i].x + max(y * y, Y * Y));
    ans = min(ans, tmp);
  }
  int a = 1;
  int b = 1;
  while (a < n) {
    if (a < b && (p[a].x - p[b].x) * (p[a].x - p[b].x) <= k) {
      long long tmp = (p[a].x - p[b].x) * (p[a].x - p[b].x);
      y = min(prefy[a - 1], suffy[b + 1]);
      Y = max(prefY[a - 1], suffY[b + 1]);
      tmp = max(tmp, (Y - y) * (Y - y));
      tmp = max(tmp, p[a].x * p[a].x + max(y * y, Y * Y));
      tmp = max(tmp, p[b].x * p[b].x + max(y * y, Y * Y));
      ans = min(ans, tmp);
    }
    if (b <= a || (b < n && (p[a].x - p[b + 1].x) * (p[a].x - p[b + 1].x) <= k))
      ++b;
    else
      ++a;
  }
  return ans <= k;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) scanf("%lld%lld", &p[i].x, &p[i].y);
  if (n == 1) {
    puts("0");
    return 0;
  }
  long long ans = 1e17;
  sort(p + 1, p + n + 1);
  prefy[0] = INF;
  prefY[0] = -INF;
  for (int i = 1; i <= n; ++i) {
    prefy[i] = min(p[i].y, prefy[i - 1]);
    prefY[i] = max(p[i].y, prefY[i - 1]);
  }
  suffy[n + 1] = INF;
  suffY[n + 1] = -INF;
  for (int i = n; i >= 1; --i) {
    suffy[i] = min(p[i].y, suffy[i + 1]);
    suffY[i] = max(p[i].y, suffY[i + 1]);
  }
  long long l = 0;
  long long r = 1e17;
  while (l < r) {
    long long mid = (l + r) >> 1;
    if (ok(mid)) {
      r = mid;
    } else {
      l = mid + 1;
    }
  }
  ans = min(ans, l);
  for (int i = 1; i <= n; ++i) swap(p[i].x, p[i].y);
  sort(p + 1, p + n + 1);
  prefy[0] = INF;
  prefY[0] = -INF;
  for (int i = 1; i <= n; ++i) {
    prefy[i] = min(p[i].y, prefy[i - 1]);
    prefY[i] = max(p[i].y, prefY[i - 1]);
  }
  suffy[n + 1] = INF;
  suffY[n + 1] = -INF;
  for (int i = n; i >= 1; --i) {
    suffy[i] = min(p[i].y, suffy[i + 1]);
    suffY[i] = max(p[i].y, suffY[i + 1]);
  }
  l = 0;
  r = 1e17;
  while (l < r) {
    long long mid = (l + r) >> 1;
    if (ok(mid)) {
      r = mid;
    } else {
      l = mid + 1;
    }
  }
  ans = min(ans, l);
  cout << ans << endl;
}
