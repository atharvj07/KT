def power(bottom, up):
	if up == 1:
		return bottom
	elif up == 0:
		return 1
	tmp = power(bottom, int(up // 2))
	if up % 2 == 0:
		return tmp * tmp
	else:
		return tmp * tmp * bottom

def ccw(p1, p2, p3):
	a = p2[0] - p1[0]
	b = p2[1] - p1[1]
	c = p3[0] - p1[0]
	d = p3[1] - p1[1]
	tmp = a * d - b * c
	if tmp > 0:
		return 1
	elif tmp == 0:
		return 0
	else:
		return -1
