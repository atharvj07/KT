n = int(input())
numbers = list(map(int, input().split()))
(even_numbers, odd_numbers) = ([], [])
for i in numbers:
	even_numbers.append(i) if i % 2 == 0 else odd_numbers.append(i)
odd_numbers.sort()
if len(odd_numbers) % 2 == 0:
	print(sum(even_numbers) + sum(odd_numbers))
else:
	print(sum(even_numbers) + sum(odd_numbers[1:]))
