class Scrl:

	def __init__(self, s, v, e):
		self.sp = s
		self.vp = v
		self.ep = e

	def spm(self, move):
		d1 = abs(self.vp - self.sp)
		d2 = abs(self.ep - self.vp)
		if d2 % move == 0:
			if d1 % move == 0:
				return (d1 + d2) // move
			else:
				return (d1 + d2) // move + 1
		else:
			return None
for t in range(int(input())):
	(n, x, y, d) = map(int, input().split())
	m = list()
	for o in (Scrl(x, x, y), Scrl(x, 1, y), Scrl(x, n, y)):
		m.append(o.spm(d))
		if m[-1] == None:
			m.pop()
	if m:
		print(min(m))
	else:
		print(-1)
