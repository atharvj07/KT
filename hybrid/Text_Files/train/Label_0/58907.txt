import java.util.*;
public class A{
    public static void main(String args[])
    {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int a[]=new int[n*2];
        boolean flag=false;
        int cnt=1;
        for(int i=0;i<n*2;i++)
        {
            a[i]=sc.nextInt();
            if(i!=0 && a[0]==a[i])
            {
                cnt++;
            }
        }
        if(cnt==2*n)
        {
            System.out.println("-1");
        }
        else
        {
            Arrays.sort(a);
            for(int i:a)
            {
                System.out.print(i+" ");
            }
        }
    }
    
}
