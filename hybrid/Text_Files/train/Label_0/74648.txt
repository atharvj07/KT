def checkBalancedArray(string, ch):
	stack = 0
	for i in string:
		if stack < 0:
			return 'NO'
		if i in ch:
			stack += 1
		else:
			stack -= 1
	return 'NO' if stack else 'YES'
for _ in range(int(input())):
	s = input()
	count_A = s.count('A')
	count_B = s.count('B')
	count_C = s.count('C')
	c = s[0]
	if count_A + count_B == count_C:
		print(checkBalancedArray(s, [['A', 'B'], [c]][c == 'C']))
	elif count_B + count_C == count_A:
		print(checkBalancedArray(s, [['B', 'C'], [c]][c == 'A']))
	elif count_C + count_A == count_B:
		print(checkBalancedArray(s, [['C', 'A'], [c]][c == 'B']))
	else:
		print('NO')
