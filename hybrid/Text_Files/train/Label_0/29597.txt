        string str;
        cin >> str;

        int invariant[2] = {-1,-1};

        ll ans = 0;
        for (int i = 0; i < str.size(); ++i) {
            if(str[i] != '?'){
                invariant[ (i%2)^(str[i]-'0') ] = i; // keep maximum invariant index
            }
            int mn = min(invariant[0],invariant[1]);
            ans += i-mn;
        }

        cout << ans close
