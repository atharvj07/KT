#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m, ma = 0, c, x, y;
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> x;
    y = x / m;
    if (x % m) y++;
    if (y >= ma) {
      ma = y;
      c = i + 1;
    }
  }
  cout << c << '\n';
}
