s = input()
t = input()
diff = []
ans = [None] * len(s)
for i in range(len(s)):
	if s[i] != t[i]:
		diff.append(i)
	else:
		ans[i] = s[i]
if len(diff) % 2 == 1:
	print('impossible')
else:
	for i in range(len(diff) // 2):
		ans[diff[i]] = t[diff[i]]
	for i in range(len(diff) // 2, len(diff)):
		ans[diff[i]] = s[diff[i]]
	print(''.join(ans))
