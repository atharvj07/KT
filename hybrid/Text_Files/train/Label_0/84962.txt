from math import gcd
from math import ceil
g = int(input())
p = 0
if g % 2 != 0:
	x1 = (2 + g - 1) / 2 - 1
	x2 = x1 * (x1 + 1)
	y1 = ceil(g / 2) / 2 * (1 + g)
	x = list(map(str, range(1, g + 1, 2)))
	y = list(map(str, range(2, g, 2)))
elif g % 2 == 0:
	x1 = (2 + g) / 2 - 1
	x2 = x1 * (x1 + 1)
	y1 = g / 2 / 2 * g
	x = list(map(str, range(1, g, 2)))
	y = list(map(str, range(2, g + 1, 2)))
if gcd(int(x2), int(y1)) > 1:
	print('Yes')
	print(' '.join([str(len(y))] + y))
	print(' '.join([str(len(x))] + x))
	p += 1
elif p != 1:
	print('No')
