#include <bits/stdc++.h>
using namespace std;
struct xy {
  int x, y;
  xy() : x(), y() {}
  xy(int x_, int y_) : x(x_), y(y_) {}
  xy operator-(const xy &b) const { return xy(x - b.x, y - b.y); }
  xy operator+(const xy &b) const { return xy(x + b.x, y + b.y); }
  int operator*(const xy &b) const { return x * b.x + y * b.y; }
  int length2() const { return sqrt(x * x + y * y); }
  int dist2(const xy &b) const { return (*this - b).length2(); }
  bool operator<(const xy &b) const {
    if (x == b.x) return y < b.y;
    return x < b.x;
  }
};
int cross(const xy &a, const xy &b) { return a.x * b.y - a.y * b.x; }
xy p[8];
bool rect(vector<xy> pts, bool square = false) {
  xy A = pts[0], B = pts[1], C = pts[2], D = pts[3];
  return ((B - A) * (A - D) == 0) and ((A - B) * (B - C) == 0) and
         ((B - C) * (C - D) == 0) and ((C - D) * (D - A) == 0) and
         (A.dist2(B) > 0) and (B.dist2(C) > 0) and
         (not square or A.dist2(B) == B.dist2(C));
}
bool checkAll(vector<xy> pts, bool square = false) {
  sort((pts).begin(), (pts).end());
  do {
    if (rect(pts, square)) return true;
  } while (next_permutation((pts).begin(), (pts).end()));
  return false;
}
int main() {
  for (int i = (0); i < (8); ++i) {
    scanf("%d%d", &p[i].x, &p[i].y);
  }
  for (int m = (0); m < (256); ++m) {
    if (__builtin_popcount(m) != 4) continue;
    vector<int> i1, i2;
    vector<xy> s1, s2;
    for (int i = (0); i < (8); ++i) {
      if (m & (1 << i)) {
        s1.push_back(p[i]);
        i1.push_back(i);
      } else {
        s2.push_back(p[i]);
        i2.push_back(i);
      }
    }
    if (checkAll(s1, true) and checkAll(s2)) {
      printf("YES\n");
      for (int i = (0); i < (4); ++i)
        printf("%d%c", i1[i] + 1, ((i == 3) ? '\n' : ' '));
      for (int i = (0); i < (4); ++i)
        printf("%d%c", i2[i] + 1, ((i == 3) ? '\n' : ' '));
      return 0;
    }
  }
  printf("NO\n");
  return 0;
}
