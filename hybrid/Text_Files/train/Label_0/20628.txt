#include<iostream>
#include<algorithm>
using namespace std;

int main()
{
  int n,m;
  for(;cin>>n>>m;)
    {
      long long int map[n+2];
      long long int sum=0;
      map[1]=0;
      for(int i=2;i<=n;i++)
	{
	  int tmp;
	  cin>>tmp;
	  sum+=tmp;
	  map[i]=sum;
	  //	  cout<<"map["<<i<<"]="<<map[i]<<endl;
	}
      long long int ans=0;
      int now=1;
      for(int i=1;i<=m;i++)
	{
	  int tmp;
	  cin>>tmp;
	  int old=now;
	  now+=tmp;
	  ans+=abs(map[now]-map[old]);
	}
      cout<<ans%100000<<endl;
    }
}
