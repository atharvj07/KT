#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define INF 0x3f3f3f3f
int t;
char tu[505][505];
int main()
{
    scanf("%d",&t);
    while(t--)
    {
        int n,m;
        scanf("%d%d",&n,&m);
        for(int i=0;i<n;i++)scanf("%s",tu[i]);
        if(m==1)for(int i=0;i<n;i++)tu[i][0]='X';
        for(int i=0;i<m;i+=3)
        {
            for(int j=0;j<n;j++)
                tu[j][i]='X';
            if(i+2>=m)break;
            int pos=0;
            for(int j=0;j<n;j++)
                if(tu[j][i+1]=='X'||tu[j][i+2]=='X')
                pos=j;
            tu[pos][i+1]=tu[pos][i+2]='X';
        }
        if(m%3==0)
        {
            for(int i=0;i<n;i++)
                if(tu[i][m-1]=='X')tu[i][m-2]='X';
        }
        for(int i=0;i<n;i++)
            printf("%s\n",tu[i]);
    }
    return 0;
}
