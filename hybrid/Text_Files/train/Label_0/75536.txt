entrada = input()
n = 0
e = 0
i = 0
t = 0
for k in range(len(entrada)):
	if entrada[k] == 'n':
		n += 1
	if entrada[k] == 'e':
		e += 1
	if entrada[k] == 'i':
		i += 1
	if entrada[k] == 't':
		t += 1
if n % 2 == 0:
	n -= 1
n = (n - 3) // 2 + 1
if n == -1:
	n = 0
e = e // 3

def fmenor(a, b, c, d):
	menor: None
	if a <= b and a <= c and (a <= d):
		menor = a
	elif b <= a and b <= c and (b <= d):
		menor = b
	elif c <= d and c <= a and (c <= b):
		menor = c
	else:
		menor = d
	return menor
print(fmenor(n, e, i, t))
