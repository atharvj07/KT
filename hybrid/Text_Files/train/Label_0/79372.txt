import math
import sys
inp = input().split()
a = int(inp[0])
b = int(inp[1])
if a == 0:
	print(str(-b ** 2))
	out = 'x' * b
	print(out)
elif b == 0:
	print(str(a ** 2))
	out = 'o' * a
	print(out)
else:
	ind = -1
	buk = -1
	quo = -1
	rem = -1
	sco = -sys.maxsize - 1
	for i in range(a):
		buk_i = i + 2
		quo_i = int(b / buk_i)
		rem_i = b % buk_i
		sco_i = (a - i) ** 2 + i - rem_i * (quo_i + 1) ** 2 - (buk_i - rem_i) * quo_i ** 2
		if sco_i > sco:
			buk = buk_i
			quo = quo_i
			rem = rem_i
			sco = sco_i
			ind = i
	print(sco)
	out = 'x' * quo + 'o' * (a - ind - 1)
	for i in range(rem):
		out += 'o' + 'x' * (quo + 1)
	for i in range(buk - rem - 1):
		out += 'o' + 'x' * quo
	print(out)
