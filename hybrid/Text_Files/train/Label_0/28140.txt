#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main() {

  // input
  int n, K; cin >> n >> K;
  int sum_b = 0;
  int sum_r = 0;

  vector<int> a(n);
  vector<int> acc(n + 1, 0);
  for(int i = 0; i < n; i++){
    cin >> a[i];
    acc[i + 1] = a[i];
    acc[i + 1] += acc[i];
  }
  vector<int> r(K);
  for(int i = 0; i < K; i++){
    cin >> r[i];
    sum_r += r[i];
  }
  vector<int> b(K);
  for(int i = 0; i < K; i++){
    cin >> b[i];
    sum_b += b[i];
  }
  sort(a.rbegin(), a.rend());
  sort(b.rbegin(), b.rend());
  sort(r.rbegin(), r.rend());

  int inf = 10101010;
  vector<vector<vector<int>>> dp(n + 1, vector<vector<int>> (K + 1, vector<int> (sum_r + 1, inf)));
  dp[0][0][0] = 0;

  for(int i = 0; i < n; i++) {
    for(int j = 0; j < K; j++) {
      for(int k = 0; k < sum_r; k++) {
        if(dp[i][j][k] == inf) continue;


        int l = i - j;
        int used_r = k;
        int used_b = acc[i] - used_r;

        // use red
        int nxt_used_r = used_r + a[i];
        int nxt_cost = max(0, a[i] - r[j]);
        if(nxt_used_r <= sum_r) dp[i + 1][j + 1][nxt_used_r] = min(dp[i + 1][j + 1][nxt_used_r], dp[i][j][k] + nxt_cost);


        // use blue
        int nxt_used_b = used_b + a[i];
        if(l >= K) continue;
        nxt_cost = max(0, a[i] - b[l]);
        if(nxt_used_b <= sum_b) dp[i + 1][j][used_r] = min(dp[i + 1][j][used_r], dp[i][j][k] + nxt_cost);

      }
    }
  }

  int ans = inf;
  for(int j = 0; j <= K; j++)
    for(int k = 0; k <= sum_r; k++)
      ans = min(ans, dp[n][j][k]);

  if(ans == inf) ans = -1;

  cout << ans << endl;
  return 0;
}
