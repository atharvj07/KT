import sys
input = sys.stdin.readline
for _ in range(int(input())):
	N = int(input())
	res = [i for i in range(1, N + 1)]
	qs = []
	c = 0
	for i in range(N // 2, N):
		qs.append((i + 1, 1))
		(res[0], res[i]) = (res[i], res[0])
		c += i ** 2
	for i in range(1, N // 2):
		qs.append((i + 1, N))
		(res[-1], res[i]) = (res[i], res[-1])
		c += (N - i - 1) ** 2
	print(c)
	print(*res)
	qs.reverse()
	print(len(qs))
	for q in qs:
		print(*q)
