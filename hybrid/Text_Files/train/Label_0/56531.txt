class TaskD {
    public void solve(int testNumber, Scanner in, PrintWriter out) {
        long d = in.nextLong();
        long n = in.nextLong();
        int m = in.nextInt();
        GasStation[] stations = new GasStation[m+1];
        for ( int i = 0; i < m; i++) {
            int x = in.nextInt();
            int p = in.nextInt();
            stations[i] = new GasStation(x, p);
        }
        stations[m] = new GasStation(d, 0);
        Arrays.sort(stations);
        long prev = 0;
        long cost = 0;
        LinkedList<Fuel> tank = new LinkedList<Fuel>();
        tank.add(new Fuel(n, 0));
        for ( GasStation station: stations ) {
            long travel = station.x - prev;
            if ( travel > n ) {
                cost = -1;
                break;
            }
            long currentGas = n;
            while ( travel > 0 ) {
                Fuel f = tank.poll();
                long use = Math.min(f.amount, travel);
                cost += use*f.price;
                f.amount -= use;
                travel -= use;
                currentGas -= use;
                if ( f.amount > 0 ) tank.addFirst(f);
            }
            while ( tank.size() > 0 && tank.peekLast().price > station.price) {
                Fuel f = tank.pollLast();
                currentGas -= f.amount;
            }
            if ( tank.size() > 0 && tank.peekLast().price == station.price ) {
                tank.peekLast().amount += n - currentGas;
            } else tank.addLast(new Fuel(n-currentGas, station.price));
            prev = station.x;
        }
        System.out.println(cost);

    }

    class Fuel {
        long amount;
        long price;
        public Fuel(long a, long p) {
            amount = a;
            price = p;
        }
        public String toString() { return String.format("price %d amount %d", price, amount); }
    }
    class GasStation implements Comparable<GasStation> {
        long x, price;
        public GasStation(long x, long p) {
            this.x = x;
            price = p;
        }

        public int compareTo(GasStation o) {
            return Long.compare(x, o.x);
        }
        public String toString() { return String.format("location %d price %d", x, price); }
    }
}
