static int lca(int u, int v) throws  IOException{
    if(h[u]>h[v]){
        u^=v;v^=u;u^=v;
    }
    int diff=h[v]-h[u];
    for(int i=19;i>=0;i--){
        if((diff&(1<<i))!=0){
            v=f[v][i];
        }
    }
    for(int i=19;i>=0;i--){
        if(f[u][i]!=f[v][i]){
            u=f[u][i];
            v=f[v][i];
        }
    }
    return u==v?u:f[v][0];
}
static int dist(int u,int v) throws  IOException{
    int a=lca(u,v);
    return h[u]-h[a]+h[v]-h[a];
}

}
