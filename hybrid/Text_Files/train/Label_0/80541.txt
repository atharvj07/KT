(n, l, r, x) = map(int, input().split())
c = list(map(int, input().split()))
c.sort()
qtd = 0
for b in range(int('1' * n, base=2) + 1):
	binary = bin(b)[2:]
	binary = '{:0>{n}}'.format(int(binary), n=n)
	vetor = []
	for (i, d) in enumerate(binary):
		if d == '1':
			vetor.append(c[i])
	vetor.sort()
	if l <= sum(vetor) <= r and vetor[-1] - vetor[0] >= x:
		qtd += 1
print(qtd)
