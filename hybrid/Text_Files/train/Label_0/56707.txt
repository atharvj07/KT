import java.io.*;
import java.util.Arrays;
import java.util.Scanner;
import java.util.StringTokenizer;

public class Problem3 implements Runnable {
    private int currentLength = 4;
    PrintWriter writer = new PrintWriter(System.out);
    private boolean [] isCorrect = new boolean[10];

    private int answer = 0;
    private StringBuffer currentLine;
    private void rec(int pos, int length, int mode) {
        if (pos == length) {
            for (int it = 0; it < 2; it++) {
                String [] values = new String[4];
                StringBuffer ip = new StringBuffer(currentLine.toString());
                if (it == 0) {
                    if (length == 6) {
                        continue;
                    }
                    for (int ch = 0; ch < 10; ch++) {
                        if (isCorrect[ch]) {

                            ip = new StringBuffer(currentLine.toString());
                            ip.append(ch);

                            for (int i = length - 1; i > -1; i--) {
                                ip.append(currentLine.charAt(i));
                            }
                            boolean [] b = new boolean[10];
                            Arrays.fill(b, false);
                            for (int index = 0; index < length + 1; index++) {
                                b[ip.charAt(index) - '0'] = true;
                            }
                            int cont = 0;
                            for (int index = 0; index < 10; index++) {
                                if (b[index] != isCorrect[index]) {
                                    cont = 1;
                                }
                            }
                            if (cont == 1) {
                                continue;
                            }


                            for (int i = 0; i < Math.min(length * 2 - 2, 3); i++) {
                                for (int j = i + 1; j < Math.min(length * 2 - 1, i + 4); j++) {
                                    for (int k = j + 1; k < Math.min(j + 4, length * 2); k++) {

                                        values[0] = ip.substring(0, i + 1);
                                        values[1] = ip.substring(i + 1, j + 1);
                                        values[2] = ip.substring(j + 1, k + 1);
                                        values[3] = ip.substring(k + 1);

                                        if (isGoodValues(values)) {

                                            answer += 1;

                                            if (mode == 1) {
                                                writer.println(values[0] + "." + values[1] + "." + values[2] + "." + values[3]);
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                } else {
                    for (int i = length - 1; i > -1; i--) {
                        ip.append(currentLine.charAt(i));
                    }
                    boolean [] b = new boolean[10];
                    Arrays.fill(b, false);
                    for (int index = 0; index < length + 1; index++) {
                        b[ip.charAt(index) - '0'] = true;
                    }
                    for (int index = 0; index < 10; index++) {
                        if (b[index] != isCorrect[index]) {
                            return;
                        }
                    }


                    for (int i = 0; i < Math.min(length * 2, 3); i++) {
                        for (int j = i + 1; j < Math.min(length * 2, i + 4); j++) {
                            for (int k = j + 1; k < Math.min(j + 4, length * 2); k++) {

                                values[0] = ip.substring(0, i + 1);
                                values[1] = ip.substring(i + 1, j + 1);
                                values[2] = ip.substring(j + 1, k + 1);
                                values[3] = ip.substring(k + 1);

                                if (isGoodValues(values)) {

                                    answer += 1;

                                    if (mode == 1) {
                                        writer.println(values[0] + "." + values[1] + "." + values[2] + "." + values[3]);
                                    }
                                }
                            }
                        }
                    }

                }

            }
            return;
        }
        for (int i = 0; i < 10; i++) {
            if (isCorrect[i]) {
                currentLine.append(i);
                rec(pos + 1, length, mode);
                currentLine.deleteCharAt(currentLine.length() - 1);
            }
        }

    }
    public boolean isGoodValues(String [] values) {

        for (int i = 0; i < values.length; i++) {
            if (values[i].length() == 0) return false;
            if (values[i].equals("" + Long.parseLong(values[i]))) {
                if ((Long.parseLong(values[i]) < 256) && ((Long.parseLong(values[i]) >= 0))) {
                    // no operations
                } else return false;
            } else return false;
        }
        return true;
    }
    public void run() {
        InputReader reader = new InputReader(System.in);
        reader.nextLine();
        int n = reader.nextInt();
        reader.nextLine();
        Arrays.fill(isCorrect, false);
        for (int i = 0; i < n; i++) {
            isCorrect[reader.nextInt()] = true;
        }
        if (n > 6) {
            writer.println(0);
            writer.close();
            return;
        }
        for (int i = 2; i < 7; i++) {
            currentLine = new StringBuffer("");
            rec(0, i, 0);
        }
        writer.println(answer);
        for (int i = 2; i < 7; i++) {
            currentLine = new StringBuffer("");
            rec(0, i, 1);
        }


        writer.close();
    }
    public static void main(String[] args) {
        new Problem3().run();
    }
    private class InputReader {
        BufferedReader reader;
        StringTokenizer token;

        private InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream));
        }

        private String next() {
            return token.nextToken();
        }

        private int nextInt() {
            return Integer.parseInt(this.next());
        }

        private double nextDouble() {
            return Double.parseDouble(this.next());
        }

        private long nextLong() {
            return Long.parseLong(this.next());
        }

        private String nextLine() {
            String line = "";
            try {
                line = reader.readLine();
                token = new StringTokenizer(line, " ");

            } catch(IOException e) {

            }
            return line;
        }
    }
}
