n = int(input())
while n:
	s = input()
	a = s.count('ab')
	b = s.count('ba')
	if a == b:
		print(s)
	else:
		if a > b:
			if s[-1] == 'b':
				t = list(s)
				t[-1] = 'a'
				s = ''.join(t)
				a -= 1
			if a > b:
				if s[0] == 'b':
					t = list(s)
					t[0] = 'a'
					s = ''.join(t)
					a -= 1
		else:
			if s[-1] == 'a':
				t = list(s)
				t[-1] = 'b'
				s = ''.join(t)
				b -= 1
			if b > a:
				if s[0] == 'a':
					t = list(s)
					t[0] = 'b'
					s = ''.join(t)
					b -= 1
		print(s)
	n -= 1
