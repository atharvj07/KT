from collections import defaultdict
n = [int(i) for i in input()]
st = frozenset(n)
ini = [n.count(i) for i in range(10)]
dp = defaultdict(int)

def dfs(state):
	key = tuple(state)
	if key in dp:
		return dp[key]
	if all((state[i] > 0 for i in st)):
		dp[key] += 1
	for i in range(10):
		if state[i] < ini[i]:
			state[i] += 1
			dp[key] += dfs(state)
			state[i] -= 1
	return dp[key]
ans = 0
s = [0] * 10
for i in range(1, 10):
	if ini[i]:
		s[i] = 1
		ans += dfs(s)
		s[i] = 0
print(ans)
