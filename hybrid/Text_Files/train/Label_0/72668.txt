import bisect
test = int(input())
for _ in range(test):
	(n, k) = map(int, input().split())
	x = list(map(int, input().split()))
	yp = list(map(int, input().split()))
	x.sort()
	ans = 0
	ma = [0] * (n + 3)
	for i in range(n - 1, -1, -1):
		(l, r) = (x[i], x[i] + k)
		ll = bisect.bisect_left(x, l)
		rr = bisect.bisect_right(x, r)
		now = rr - ll
		ma[i] = max(now, ma[i + 1])
		ans = max(ans, now + ma[rr])
	print(ans)
