#include <iostream>
using namespace std;

int main(int argc, char *argv[]){
    int n,m;
    cin >> n >> m;
    int a=(n+1)/2;
    int c=0;
    for(int i=0;i<m;i++){
        cout << 1+i << " " << n-i << endl;
        c++;
        if(c==m) break;
        cout << a-1-i << " " << a+1+i << endl;
        c++;
        if(c==m) break;
    }
    return 0;
}
