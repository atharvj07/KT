def plus(a, b):
	c = ''
	i = len(a) - 1
	arg = 0
	while i >= 0:
		k = (arg + ord(a[i]) + ord(b[i])) % 26
		if arg + ord(a[i]) + ord(b[i]) < 26:
			arg = 0
		else:
			arg = 1
		i -= 1
		c = chr(k) + c
	if arg:
		c = chr(1) + c
	return c

def minus(a):
	c = ''
	i = 0
	arg = 0
	while i < len(a):
		c += chr((arg * 26 + ord(a[i])) // 2)
		arg = ord(a[i]) % 2
		i += 1
	if arg:
		c = c[:len(c) - 1] + chr(ord(c[len(c) - 1]) + 1)
	return c
