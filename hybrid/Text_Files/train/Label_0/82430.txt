import sys
input = sys.stdin.readline

def print(x):
	sys.stdout.write(x)
	sys.stdout.write('\n')

def diff(a, b, c, d):
	if len(a) + int(b) > len(c) + int(d):
		return 1
	if len(a) + int(b) < len(c) + int(d):
		return -1
	(a, b, c, d) = (int(a), int(b), int(c), int(d))
	if abs(b - d) > 10:
		return b - d
	if b > d:
		return a * 10 ** (b - d) - c
	return -(c * 10 ** (d - b) - a)

def solve():
	n = int(input())
	arr = list(map(lambda x: int(x), input().split(' ')))
	m = min(arr)
	t = 0
	for a in arr:
		if t >= n // 2:
			break
		if a != m:
			print(f'{a} {m}')
			t += 1
cases = int(input())
for _ in range(cases):
	solve()
