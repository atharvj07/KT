import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[5];
        int sum = 0;
        for(int i=0;i<n;i++) {
            int num = sc.nextInt();
            sum += num;
            nums[num]++;
        }

        if(sum<3||sum==5) {
            System.out.println(-1);
            return;
        }

        int num4=sum%3;
        int num3=sum/3-num4;

        int result = n+1;
        while(num3>=0) {
            int one = solution(nums, num4, num3);
            result = Math.min(one, result);
            num4+=3;
            num3-=4;
        }
        System.out.println(result);

    }

    public static int solution(int[] nums, int num4, int num3) {
        int curr4=0;
        int curr3=0;
        int cnt=0;
        for(int i=4;i>=1;i--) {
            if(curr4==num4 && curr3==num3) {
                break;
            }

            int tnum = nums[i];

            if(curr4<num4) {
                int tneed = num4-curr4;
                int tmin = Math.min(tnum,tneed);
                curr4+=tmin;
                tnum-=tmin;

                cnt+=(4-i)*tmin;
            }

            if(curr3<num3) {
                if(i==4) {
                    curr3+=tnum;
                    continue;
                }

                int tneed = num3-curr3;
                int tmin = Math.min(tnum,tneed);
                curr3+=tmin;
                tnum-=tmin;

                cnt+=(3-i)*tmin;
            }
        }
        return cnt;
    }
}
