import java.io.*;
import java.util.*;

public class C {
	public static void main (String[] args) { new C(); }

	public C() {
		FastScanner fs = new FastScanner();
		PrintWriter out = new PrintWriter(System.out);
		System.err.println("");

		work();
		
		int n = fs.nextInt();
		int q = fs.nextInt();
		int[] a = fs.nextIntArray(n);
		
		int[][] sums = new int[101][n];
		for(int i = 0; i < q; i++) {
			
			int L = fs.nextInt()-1, R = fs.nextInt()-1;
			int K = fs.nextInt();
			int len = R-L+1, other = 0;
			
			sums[K][L] = add(sums[K][L], 1);
			for(int j = K; j >= 0; j--, other++) {
				if(R+1 < n) {
					sums[j][R+1] = sub(sums[j][R+1], choose(len+other-1, other));
				}
			}
		}
		
		int[][] fin = new int[101][n];
		for(int i = 100; i >= 0; i--) {
			for(int j = 0; j < n; j++) {
				fin[i][j] = sums[i][j];
				if(j > 0) fin[i][j] = add(fin[i][j], fin[i][j-1]);
				if(i+1 <= 100) fin[i][j] = add(fin[i][j], fin[i+1][j]);
			}
		}
		
		for(int i = 0; i < n; i++) {
			a[i] = add(a[i], fin[0][i]);
		}
		
		for(int i = 0; i < n; i++) {
			out.print(a[i] + " ");
		}
		
		out.close();
	}
	
	int MOD = (int)1e9+7;
	int[] fact, invFact;
	void work() {
		int MAX = 100100;
		fact = new int[MAX];
		fact[0] = 1;
		for(int i = 1; i < MAX; i++) {
			fact[i] = mult(fact[i-1], i);
		}
		
		int[] invs = new int[MAX];
		invs[0] = invs[1] = 1;
		for(int i = 2; i < invs.length; i++) {
			invs[i] = sub(MOD, mult(MOD/i, invs[MOD%i]));
		}
		
		invFact = new int[MAX];
		invFact[0] = invs[0];
		for(int i = 1; i < MAX; i++) {
			invFact[i] = mult(invFact[i-1], invs[i]);
		}
		
	}
	
	int choose(int n, int k) {
		int num = fact[n];
		int den = mult(invFact[k], invFact[n-k]);
		return mult(num, den);
	}
	
	int add(int a, int b) {
		a += b;
		if(a >= MOD) a -= MOD;
		return a;
	}
	int mult(long a, long b) {
		a *= b;
		if(a >= MOD) a %= MOD;
		return (int)a;
	}
	int sub(int a, int b) {
		a -= b;
		if(a < 0) a += MOD;
		return a;
	}

	class FastScanner {
		public int BS = 1<<16;
		public char NC = (char)0;
		byte[] buf = new byte[BS];
		int bId = 0, size = 0;
		char c = NC;
		double num = 1;
		BufferedInputStream in;

		public FastScanner() {
			in = new BufferedInputStream(System.in, BS);
		}

		public FastScanner(String s) {
			try {
				in = new BufferedInputStream(new FileInputStream(new File(s)), BS);
			}
			catch (Exception e) {
				in = new BufferedInputStream(System.in, BS);
			}
		}

		public char nextChar(){
			while(bId==size) {
				try {
					size = in.read(buf);
				}catch(Exception e) {
					return NC;
				}                
				if(size==-1)return NC;
				bId=0;
			}
			return (char)buf[bId++];
		}

		public int nextInt() {
			return (int)nextLong();
		}

		public long nextLong() {
			num=1;
			boolean neg = false;
			if(c==NC)c=nextChar();
			for(;(c<'0' || c>'9'); c = nextChar()) {
				if(c=='-')neg=true;
			}
			long res = 0;
			for(; c>='0' && c <='9'; c=nextChar()) {
				res = (res<<3)+(res<<1)+c-'0';
				num*=10;
			}
			return neg?-res:res;
		}

		public double nextDouble() {
			double cur = nextLong();
			return c!='.' ? cur:cur+nextLong()/num;
		}

		public String next() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c>32) {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public String nextLine() {
			StringBuilder res = new StringBuilder();
			while(c<=32)c=nextChar();
			while(c!='\n') {
				res.append(c);
				c=nextChar();
			}
			return res.toString();
		}

		public boolean hasNext() {
			if(c>32)return true;
			while(true) {
				c=nextChar();
				if(c==NC)return false;
				else if(c>32)return true;
			}
		}
		
		public int[] nextIntArray(int n) {
			int[] res = new int[n];
			for(int i = 0; i < n; i++) res[i] = nextInt();
			return res;
		}
		
	}

}
