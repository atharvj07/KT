from typing import List

def maximum_alternating_subsequence(values: List[int]) -> int:
	(maximums, minimums) = ([0] * (len(values) + 1), [0] * (len(values) + 1))
	for index in range(len(values)):
		maximums[index + 1] = max(maximums[index], values[index] - minimums[index])
		minimums[index + 1] = min(minimums[index], values[index] - maximums[index])
	return maximums[-1]

def read_input():
	cases = int(input())
	while cases:
		cases = cases - 1
		(_, _) = input().split(' ')
		values = list(map(int, input().split(' ')))
		print(maximum_alternating_subsequence(values))

def main():
	read_input()
main()
