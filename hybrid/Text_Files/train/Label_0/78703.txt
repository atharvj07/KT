(n, m) = map(lambda x: int(x), input().split())
c = 0
l = []
for i in range(m):
	l.append(set(map(lambda x: int(x), input().split())))
for e in l[0]:
	for i in range(0, m):
		if e not in l[i]:
			for el in l[i]:
				for j in range(i, m):
					if el not in l[j] and e not in l[j]:
						c += 1
						break
			break
if c == 4:
	print('NO')
else:
	print('YES')
