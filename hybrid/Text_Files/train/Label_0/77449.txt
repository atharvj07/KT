n = int(input())
suml = 0
sumr = 0
biggestleftinc = 0
blindex = -1
biggestrightinc = 0
brindex = -1
for i in range(n):
	(l, r) = (int(x) for x in input().split())
	suml += l
	sumr += r
	diff = abs(l - r)
	if l > r and diff > biggestleftinc:
		biggestleftinc = diff
		blindex = i
	elif r > l and diff > biggestrightinc:
		biggestrightinc = diff
		brindex = i
if suml == sumr:
	if biggestleftinc == biggestrightinc and biggestleftinc == 0:
		print(0)
	elif biggestleftinc > biggestrightinc:
		print(blindex + 1)
	else:
		print(brindex + 1)
elif sumr > suml:
	righttest = biggestrightinc - (sumr - suml)
	if biggestleftinc > righttest and biggestleftinc > 0:
		print(blindex + 1)
	elif righttest > biggestleftinc:
		print(brindex + 1)
	else:
		print(0)
else:
	lefttest = biggestleftinc - (suml - sumr)
	if biggestrightinc > lefttest and biggestrightinc > 0:
		print(brindex + 1)
	elif lefttest > biggestrightinc:
		print(blindex + 1)
	else:
		print(0)
