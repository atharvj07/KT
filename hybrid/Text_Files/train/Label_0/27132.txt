#include <bits/stdc++.h>
using namespace std;
const int N = (int)1e6 + 1;
const int mod = (int)1e9 + 7;
int n, k, l = 1, in = 1;
string a[60], b[800], c[60];
int main() {
  cin >> n >> k;
  for (int i = 1; i <= n - k + 1; i++) cin >> a[i];
  for (int i = 'A'; i <= 'Z'; i++) {
    for (int j = 'a'; j <= 'z'; j++) {
      b[l] += char(i);
      b[l] += char(j);
      l++;
    }
  }
  if (a[1] == "NO") {
    c[1] = b[in];
    c[2] = b[in];
    in++;
    for (int i = 3; i <= k; i++) c[i] = b[in], in++;
  } else {
    for (int i = 1; i <= k; i++) c[i] = b[in], in++;
  }
  for (int i = 2; i <= n - k + 1; i++) {
    if (a[i] == "NO") {
      if (a[i - 1] == "YES")
        c[i + k - 1] = c[i];
      else
        c[i + k - 1] = c[i];
    } else {
      c[i + k - 1] = b[in], in++;
    }
  }
  for (int i = 1; i <= n; i++) cout << c[i] << " ";
  return 0;
}
