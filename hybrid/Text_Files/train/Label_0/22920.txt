#include <bits/stdc++.h>
using namespace std;
const long long mod = 1000000000 + 7;
const int M = 1 << 9;
const int N = 100000 + 5;
vector<int> g[N];
vector<int> dp[N], val[N];
int ncr[M][M];
int multi[M];
int k;
long long mpow(long long a, long long b) {
  long long ans = 1;
  for (; b; b >>= 1) {
    if (b & 1) ans = ans * a % mod;
    a = a * a % mod;
  }
  return ans;
}
inline int mul(int a, int b) { return (1LL * a * b) % mod; }
inline int add(int a, int b) { return a + b >= mod ? a + b - mod : a + b; }
inline int sub(int a, int b) { return a < b ? a - b + mod : a - b; }
vector<int> mul(vector<int>& a, vector<int>& b) {
  int length = a.size() + b.size();
  if (length > k + 2) length = k + 2;
  vector<int> c(length, 0);
  for (int i = 0; i < a.size(); i++) {
    for (int j = 0; j < b.size() && i + j <= k; j++) {
      int v = mul(a[i], b[j]);
      c[i + j] = add(c[i + j], v);
      c[i + j + 1] = add(c[i + j + 1], v);
    }
  }
  if (c.size() > k + 1) c.resize(k + 1);
  return c;
}
void add(vector<int>& a, vector<int>& b) {
  a.resize(max(a.size(), b.size()), 0);
  for (int i = 0; i < b.size(); i++) {
    a[i] = add(a[i], b[i]);
  }
}
void dfs(int x, int p) {
  dp[x] = {2};
  for (auto it : g[x]) {
    if (it == p) continue;
    dfs(it, x);
    vector<int> temp;
    temp = mul(dp[x], dp[it]);
    add(dp[x], temp);
  }
  dp[x][0]--;
  if (dp[x][0] < 0) dp[x][0] += mod;
  val[x] = dp[x];
  for (auto it : g[x]) {
    if (it == p) continue;
    for (int i = 1; i < dp[x].size() && i - 1 < dp[it].size(); i++) {
      if (i < dp[it].size()) val[x][i] = sub(val[x][i], dp[it][i]);
      val[x][i] = sub(val[x][i], dp[it][i - 1]);
    }
  }
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  ncr[0][0] = 1;
  for (int i = 1; i < M; i++) {
    ncr[i][0] = 1;
    for (int j = 1; j <= i; j++) {
      ncr[i][j] = add(ncr[i - 1][j], ncr[i - 1][j - 1]);
    }
  }
  int n;
  cin >> n >> k;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1, 0);
  for (int i = 1; i < M; i++) {
    multi[i] = mpow(i, k);
    for (int j = 1; j < i; j++) {
      multi[i] = sub(multi[i], mul(multi[j], ncr[i][j]));
    }
  }
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j < val[i].size(); j++) {
      ans += 1LL * multi[j] * val[i][j];
      ans %= mod;
    }
  }
  cout << ans << endl;
}
