def sv():
	N = int(input())
	lf = [False] * N
	mt = [False] * N
	for n in range(N):
		(a, b) = map(int, input().split())
		sp = False
		if a > N:
			(a, b) = (b, a)
			sp = True
		if a > N or b <= N:
			return -1
		a -= 1
		b = 2 * N - b
		lf[a] = b
		if sp:
			mt[a] = True
	g0 = -1
	g1 = -1
	gm = -1
	cc = 0
	cs = 0
	ans = 0
	for n in range(N):
		cc += 1
		if lf[n] > g0 and (g0 >= g1 or lf[n] <= g1):
			g0 = lf[n]
			if mt[n]:
				cs += 1
		elif lf[n] > g1:
			g1 = lf[n]
			if not mt[n]:
				cs += 1
		else:
			return -1
		gm = max(gm, lf[n])
		if gm == n:
			ans += min(cs, cc - cs)
			cc = 0
			cs = 0
			g0 = n
			g1 = n
	return ans
print(sv())
