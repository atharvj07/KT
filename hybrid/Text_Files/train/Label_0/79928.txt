import sys, collections, bisect
for _ in range(int(sys.stdin.readline())):
	(n, m, k) = [int(i) for i in sys.stdin.readline().split()]
	x = [int(i) for i in sys.stdin.readline().split()]
	y = [int(i) for i in sys.stdin.readline().split()]
	(sx, sy) = (set(x), set(y))
	px = collections.defaultdict(lambda : collections.defaultdict(int))
	py = collections.defaultdict(lambda : collections.defaultdict(int))
	for i in range(k):
		(a, b) = [int(i) for i in sys.stdin.readline().split()]
		if not (a in sx and b in sy):
			if a in sx:
				px[bisect.bisect(y, b)][a] += 1
			else:
				py[bisect.bisect(x, a)][b] += 1
	r = sum((sum(dd.values()) ** 2 - sum((x * x for x in dd.values())) for dd in px.values())) + sum((sum(dd.values()) ** 2 - sum((x * x for x in dd.values())) for dd in py.values()))
	sys.stdout.write(str(r // 2) + '\n')
