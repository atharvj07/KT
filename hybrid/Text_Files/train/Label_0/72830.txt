import sys
input = sys.stdin.readline
for _ in range(int(input())):
	(s, n, k) = map(int, input().split())
	if s < k:
		print('NO')
		continue
	if s == k:
		print('YES')
		continue
	if n == 1:
		print('NO')
		continue
	r = s - n
	if k & 1 and r >= n and ((r - n) % 2 == 0):
		print('NO')
		continue
	if n < k:
		print('NO')
		continue
	if n >= k:
		if s - n >= n // k * k:
			print('NO')
			continue
	print('YES')
