def main():
	n = int(input())
	aa = list((tuple(map(int, input().split())) for _ in range(n)))
	cnt = []
	for row in aa:
		l = [0] * n
		for x in row:
			l[x] += 1
		cnt.append(l)
	res = [0] * n
	for x in range(1, n):
		i = max(range(n), key=lambda i: cnt[i][x])
		res[i] = x
	for (i, x) in enumerate(res):
		if not x:
			res[i] = n - 1
		elif x == n - 1:
			res[i] = n
	print(*res)
main()
