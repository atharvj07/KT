#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  set<string> s;
  for (int i = 1; i <= n; i++) {
    string str;
    cin >> str;
    s.insert(str);
    if (s.size() == i)
      cout << "NO" << endl;
    else {
      cout << "YES" << endl;
      i--;
      n--;
    }
  }
  return 0;
}
