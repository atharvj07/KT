import java.io.*;
import java.util.*;

public class E implements Runnable {
    private MyScanner in;
    private PrintWriter out;

    final int[] dx = { -1, 0, 1, 0 };
    final int[] dy = { 0, -1, 0, 1 };

    int n;
    int[][] map;
    boolean[][] seen;
    Deque<Integer> queue;
    Point up, down, left, right;
    int cup, cdown, cleft, cright;

    boolean valid(int x, int y) {
        return x >= 0 && x < n && y >= 0 && y < n;
    }

    int bfs(int i, int j) {
        queue.addLast(i);
        queue.addLast(j);
        int ans = 0;
        seen[i][j] = true;
        up = new Point(0, Integer.MIN_VALUE);
        down = new Point(0, Integer.MAX_VALUE);
        left = new Point(Integer.MAX_VALUE, 0);
        right = new Point(Integer.MIN_VALUE, 0);
        while (!queue.isEmpty()) {
            ++ans;
            int x = queue.removeFirst();
            int y = queue.removeFirst();
            if (x < left.x) {
                left.x = x;
                left.y = y;
                cleft = 1;
            } else if (x == left.x) {
                left.y += y;
                ++cleft;
            }
            if (x > right.x) {
                right.x = x;
                right.y = y;
                cright = 1;
            } else if (x == right.x) {
                right.y += y;
                ++cright;
            }
            if (y < down.y) {
                down.x = x;
                down.y = y;
                cdown = 1;
            } else if (y == down.y) {
                down.x += x;
                ++cdown;
            }
            if (y > up.y) {
                up.x = x;
                up.y = y;
                cup = 1;
            } else if (y == up.y) {
                up.x += x;
                ++cup;
            }
            for (int dir = 0; dir < 4; ++dir) {
                int nx = x + dx[dir];
                int ny = y + dy[dir];
                if (valid(nx, ny) && !seen[nx][ny] && map[nx][ny] == 1) {
                    seen[nx][ny] = true;
                    queue.addLast(nx);
                    queue.addLast(ny);
                }
            }
        }
        return ans;
    }

    void repairPicture() {
        int d = 4;
        int[][] near = new int[n][n], ones = new int[n][n];
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                if (map[i][j] == 1) {
                    ++ones[i][j];
                }
                for (int ni = i - d; ni <= i + d; ++ni) {
                    if (ni < 0) {
                        continue;
                    }
                    if (ni >= n) {
                        break;
                    }
                    for (int nj = j - d; nj <= j + d; ++nj) {
                        if (nj < 0) {
                            continue;
                        }
                        if (nj >= n) {
                            break;
                        }
                        ++near[i][j];
                        if (map[ni][nj] == 1) {
                            ++ones[i][j];
                        }
                    }
                }
            }
        }
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                map[i][j] = (ones[i][j] >= 0.4 * near[i][j]) ? 1 : 0;
            }
        }
    }

    private void solve() {
        n = in.nextInt();
        map = new int[n][n];
        for (int i = 0; i < n; ++i) {
            String temp = in.nextLine();
            for (int j = 0; j < n; ++j) {
                map[i][j] = temp.charAt(2 * j) - '0';
            }
        }
        for (int i = 0; i < 1; ++i) {
            repairPicture();
        }
        int[][] toRight = new int[n][];
        for (int i = 0; i < n; ++i) {
            int[] cur = toRight[i] = new int[n + 1];
            cur[n] = 0;
            for (int j = n - 1; j >= 0; --j) {
                if (map[i][j] == 0) {
                    cur[j] = 0;
                } else {
                    cur[j] = cur[j + 1] + 1;
                }
            }
        }
        queue = new ArrayDeque<Integer>();
        seen = new boolean[n][n];
        int circles = 0, squares = 0;
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                if (seen[i][j]) {
                    continue;
                }
                if (toRight[i][j] == 0) {
                    seen[i][j] = true;
                    continue;
                }
                int sqr = bfs(i, j);
                if (sqr < 150) {
                    continue;
                }
                up.x /= cup;
                down.x /= cdown;
                left.y /= cleft;
                right.y /= cright;
                double r = (right.x - left.x) / 2;
                double s = Math.PI * r * r;
                double minFrac = 0.8, maxFrac = 1.1;
                double fraction = s / sqr;
                double delta = 0.4 * r;
                // System.err.println(i + " " + j);
                // System.err.println(up.x - down.x);
                // System.err.println(left.y - right.y);
                // System.err.println(fraction);
                // System.err.println(delta);
                if (Math.abs(left.y - right.y) <= delta
                        && Math.abs(up.x - down.x) <= delta
                        && fraction >= minFrac && fraction <= maxFrac) {
                    ++circles;
                } else {
                    ++squares;
                }
            }
        }
        out.println(circles + " " + squares);
    }

    @Override
    public void run() {
        in = new MyScanner();
        out = new PrintWriter(System.out);
        solve();
        in.close();
        out.close();
    }

    public static void main(String[] args) {
        new E().run();
    }

    class Point {
        public double x, y;

        public Point(double x, double y) {
            this.x = x;
            this.y = y;
        }
    }

    class MyScanner {
        private BufferedReader br;
        private StringTokenizer st;

        public MyScanner() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public void close() {
            try {
                br.close();
            } catch (IOException e) {
                e.printStackTrace();
            }
        }

        public boolean hasNext() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    String s = br.readLine();
                    if (s == null) {
                        return false;
                    }
                    st = new StringTokenizer(s);
                } catch (IOException e) {
                    e.printStackTrace();
                    return false;
                }
            }
            return st != null && st.hasMoreTokens();
        }

        private String next() {
            while (st == null || !st.hasMoreTokens()) {
                try {
                    String s = br.readLine();
                    if (s == null) {
                        return null;
                    }
                    st = new StringTokenizer(s);
                } catch (IOException e) {
                    e.printStackTrace();
                    return null;
                }
            }
            return st.nextToken();
        }

        public String nextLine() {
            try {
                st = null;
                return br.readLine();
            } catch (IOException e) {
                e.printStackTrace();
                return null;
            }
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

        public long nextLong() {
            return Long.parseLong(next());
        }

        public double nextDouble() {
            return Double.parseDouble(next());
        }
    }
}
