#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, sum = 0;
    cin >> n;
    if (n == 1)
      cout << 0 << endl;
    else {
      float x = sqrt(n);
      int y = sqrt(n);
      if (x == y) {
        sum += 2 * y - 2;
      } else {
        if (x - y < 0.5) {
          sum += 2 * y - 1;
        } else {
          sum += 2 * y;
        }
      }
      cout << sum << endl;
    }
  }
}
