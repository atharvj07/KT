#include <bits/stdc++.h>
using namespace std;
int dp[1005];
int main() {
  int n, m, c0, d0;
  cin >> n >> m >> c0 >> d0;
  memset(dp, 0, sizeof(dp));
  for (int i = c0; i <= n; i++) {
    dp[i] = i / c0 * d0;
  }
  int a, b, c, d;
  for (int i = 1; i <= m; i++) {
    cin >> a >> b >> c >> d;
    for (int j = 1; j <= min(a / b, n / c); j++) {
      for (int k = n; k >= c; k--) {
        dp[k] = max(dp[k - c] + d, dp[k]);
      }
    }
  }
  cout << dp[n] << endl;
  return 0;
}
