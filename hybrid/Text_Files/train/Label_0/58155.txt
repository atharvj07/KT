import java.io.*;
import java.util.*;
import java.math.*;

public class Main implements Runnable {
    int mod;

    long tmpr, tmps;
    
    void calc(long l, long r, long u, long v, long n) {
        u = Math.max(u, 1);
        v = Math.min(v, n);

        if (u > v) { tmpr = tmps = 0; return; }
        if (l==1 && r==n) {
            long a1 = u+v, a2 = (v-u+1); if (a1%2 == 0) a1 /= 2; else a2 /= 2;
            tmpr = ((a1%mod) * (a2%mod))%mod;
            tmps = v-u+1;
            return;
        }
        
        long m = (n+1)/2;
        long resr = 0, ress = 0;
        
        if (l <= m) { calc(l, Math.min(m, r), u/2+1, (v+1)/2, m); resr += 2*tmpr-tmps; ress += tmps; }
        if (r > m) { calc(Math.max(1, l-m), r-m, (u+1)/2, v/2, n-m); resr += 2*tmpr; ress += tmps; }
        
        tmpr = resr%mod; if (tmpr < 0) tmpr += mod; tmps = ress;
    }
    
    public void run() {
        long n = nextLong();
        int m = nextInt();
        mod = nextInt();
        
        for (int i = 0; i < m; ++i) {
            long l = nextLong(), r = nextLong(), u = nextLong(), v = nextLong();
            
            calc(l, r, u, v, n);
            output.println(tmpr);
        }

        output.flush();
    }

    public static void main(String[] args) {
        //try { input = new BufferedReader(new InputStreamReader(new FileInputStream("input.txt"))); } catch (Exception e) {}
        new Main().run();
    }

    static BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
    PrintWriter output = new PrintWriter(System.out);
    StringTokenizer tokenizer = null;

    int nextInt() {
        return Integer.parseInt(nextToken());
    }

    long nextLong() {
        return Long.parseLong(nextToken());
    }

    double nextDouble() {
        return Double.parseDouble(nextToken());
    }

    String nextToken() {
        try
        {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                tokenizer = new StringTokenizer(input.readLine());
            }

            return tokenizer.nextToken();
        } catch (Exception e) {
            return null;
        }
    }
}
