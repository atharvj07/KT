#include <bits/stdc++.h>
using namespace std;
int main() {
  long long tc, n, m, a[1005];
  cin >> tc;
  while (tc--) {
    cin >> n >> m;
    vector<pair<long long, long long> > v;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      v.push_back(make_pair(a[i], i));
    }
    if (n <= 2 || m < n) {
      cout << "-1" << endl;
      continue;
    }
    long long ans = 0;
    sort(v.begin(), v.end());
    vector<pair<long long, long long> > vv;
    for (int i = 1; i <= m; i++) {
      if (i > n) {
        vv.push_back(make_pair(v[0].second, v[1].second));
        ans += v[0].first + v[1].first;
      } else {
        int y = i + 1;
        if (y > n) {
          y = 1;
        }
        ans += a[i] + a[y];
        vv.push_back(make_pair(i, y));
      }
    }
    cout << ans << endl;
    for (int i = 0; i < m; i++) {
      cout << vv[i].first << " " << vv[i].second << endl;
    }
  }
  return 0;
}
