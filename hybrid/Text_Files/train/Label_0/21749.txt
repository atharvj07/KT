#include<bits/stdc++.h>
using namespace std;
typedef long long ll;



signed main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout << fixed << setprecision(20);
    
    ll n,k;
    cin>>n>>k;
    if(n==1){
        cout << k << endl;
        return 0;
    }
    ll now = 1;
    ll cnt=0;
    while(now < k){
        now *=2;
        cnt++;
    }
    n = min(n,cnt);
    long long low=-1,up=k+1,mid;
    while(up-low>1){
        mid=(up+low)/2;
        ll now = mid;
        ll sum = 0;
        for(int i=0;i<n;i++){
            sum += now;
            now/=2;
        }
        if(sum <= k){
            low=mid;
        }
        else{
            up=mid;
        }
    }
    cout << low << endl;
}
