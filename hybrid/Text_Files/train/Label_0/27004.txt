#include <iostream>
#include <cstdio>
using namespace std;

typedef long long ll;
ll a, b, i, cnt, g;
ll f(ll p, ll q) {
	if (p % q == 0) return q;
	f(q, p % q);
}
int main()
{
	cin >> a >> b;
	g = f(a, b);
	for (i = 2; i * i <= g; i++) {
		if (g % i == 0) {
			cnt++;
		}
		while (g % i == 0) g /= i;
	}
	if (g > 1) cnt++;
	cout << cnt + 1 << endl;
    return 0;
}
