(x, y, l, r) = map(int, input().split())
xi = [1] * 62
yi = [1] * 62
a = 0
b = 0
while 1000000000000000000 > xi[a]:
	xi[a + 1] = x ** a
	a += 1
while 1000000000000000000 > yi[b]:
	yi[b + 1] = y ** b
	b += 1
a += 1
b += 1
n = a * b
timei = [0] * n
for i in range(a):
	for j in range(b):
		timei[i * b + j] = xi[i] + yi[j]
timei.sort()
i = 0
while n > i and l > timei[i]:
	i += 1
ans = min(timei[i] - 1, r) - l + 1
last_p = timei[i]
while n > i and r > timei[i]:
	ans = max(ans, timei[i] - last_p - 1)
	last_p = timei[i]
	i += 1
ans = max(ans, min(timei[i] - 1, r) - last_p)
print(ans)
