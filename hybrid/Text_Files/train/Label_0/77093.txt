import sys
n = int(input())
a = list(map(int, input().split()))
s = []
r = []
for i in range(0, n):
	x = input()
	s.append(x)
	r.append(x[::-1])
for i in range(0, len(s) - 1):
	if not (s[i] <= s[i + 1] or s[i] <= r[i + 1] or r[i] <= s[i + 1] or (r[i] <= r[i + 1])):
		print(-1)
		sys.exit(0)
dp = [[100000000000000 for _ in range(2)] for _ in range(n)]
dp[-1][0] = 0
dp[-1][1] = a[-1]
flag = 1
for i in range(n - 2, -1, -1):
	if s[i] <= s[i + 1]:
		dp[i][0] = min(dp[i][0], dp[i + 1][0])
	if s[i] <= r[i + 1]:
		dp[i][0] = min(dp[i][0], dp[i + 1][1])
	if r[i] <= s[i + 1]:
		dp[i][1] = min(dp[i][1], dp[i + 1][0] + a[i])
	if r[i] <= r[i + 1]:
		dp[i][1] = min(dp[i][1], dp[i + 1][1] + a[i])
ans = min(dp[0][1], dp[0][0])
if ans == 100000000000000:
	print(-1)
else:
	print(ans)
