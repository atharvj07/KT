import sys
input = sys.stdin.readline

def inp():
	return int(input())

def input_list():
	return list(map(int, input().split()))

def input_string():
	s = input()
	return list(s[:len(s) - 1])

def input_int_gen():
	return map(int, input().split())
tests = inp()
for _ in range(tests):
	(a, b, c) = input_int_gen()
	halfsize = abs(a - b)
	size = halfsize * 2
	d = c + halfsize if c <= halfsize else c - halfsize
	if a > size or b > size or c > size:
		print(-1)
	else:
		print(d)
