def sumdig(n):
	sum_ = 0
	n = str(n)
	for i in range(len(n)):
		sum_ += int(n[i])
	return sum_

def fun(n):
	sum_ = 0
	for i in range(len(n)):
		ans = 0
		sum_ += int(n[i])
	if sum_ % 4 == 0:
		ans = int(n)
	elif int(n) <= 4:
		ans = 4
	else:
		t = int(n)
		for i in range(10):
			s = sumdig(t)
			if s % 4 == 0:
				ans = t
				break
			t = t + 1
	return ans
n = input()
print(fun(n))
