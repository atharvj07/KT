#include <bits/stdc++.h>
using namespace std;
int ar[200005];
map<long long, long long> vis;
int main() {
  long long n, i, ans, pre = 1, suf = 0, mx = 0, ok = 0, j;
  cin >> n;
  for (i = 1; i <= n; i++) {
    cin >> ar[i];
  }
  ans = n - 1;
  for (i = 1; i <= n; i++) {
    bool f = true;
    for (j = 1; j < i; j++) {
      if (vis[ar[j]] == 0) {
        vis[ar[j]] = 1;
      } else {
        f = false;
        break;
      }
    }
    suf = n + 1;
    for (j = n; j >= i; j--) {
      if (vis[ar[j]] == 0) {
        suf = j;
        vis[ar[j]] = 1;
      } else
        break;
    }
    if (f) {
      ans = min(ans, suf - i);
    }
    vis.clear();
  }
  cout << ans << endl;
}
