#include <stdio.h>
#include <utility>
#include <algorithm>
#include <set>
using namespace std;
typedef long long ll;

int main(void) {
  ll i, j, k, n, q, a;
  char c[10];
  scanf("%lld%lld", &n, &q);
  pair<ll, ll> tm[n];
  ll pos[n], s[n];
  for(i = 0; i < n; ++i) {
    scanf("%lld", &tm[i].first);
    tm[i].second = i;
  }
  sort(tm, tm + n);
  for(i = 0; i < n; ++i) s[i] = tm[i].first, pos[tm[i].second] = i;
  multiset<ll> ms;
  for(i = 0; i < q; ++i) {
    scanf("%s%lld", c, &a);
    if(c[0] == 'A') --a, ms.insert(s[pos[a]]);
    else if(c[0] == 'R') --a, ms.erase(ms.lower_bound(s[pos[a]]));
    else {
      multiset<ll>::iterator it;
      if(ms.empty()) {
        if(a != n) printf("NA\n");
        else printf("0\n");
        continue;
      }
      it = ms.end();
      it--;
      if(s + n - upper_bound(s, s + n, *it) > a) {
        printf("NA\n");
        continue;
      }
      ll t = 1e9, h = -1;
      while(t != h + 1) {
        ll now = (t + h) / 2, las = -1, ans = 0;
        for(it = ms.begin(); it != ms.end(); ++it) {
          int f = lower_bound(s, s + n, *it - now) - s, b = upper_bound(s, s + n, *it) - s;
          if(f > las) ans += b - f;
          else ans += b - las - 1;
          las = b - 1;
        }
        if(ans >= n - a) t = now;
        else h = now;
      }
      printf("%lld\n", t);
    }
  }
  return 0;
}
