import java.io.*;
import java.util.*;

public class E {
	
	// disjoint occurences
	// so we could determine for every position if a occurence is possible or not.
	// ??????? t=3
	// yyyyynn
	// ??????aa????? -> ?????a and a?????
	// yyyyynnyyyynn
	// until you find 2 (t-1) n's in a row, all occurences can overlap.
	// we want to maximize the number of occurences, 
	// SO we have 3 kinds of strings:
	// - all jokers: ????
	// - a-based string: ?b?bab??aba??
	// - b-based string: ?a??????
	// DP:
	// maintain:
	// max_beauty [ n ]   = max beautiy that can be made using first n chars
	// replacements [ n ] = min number of replacements necessary to achieve max_beauty[n]
	// calculate [n] = {
	//      // can we place t at the end of first n chars? 
	//      // if yes, how many replacements do we need? r
	//      max_beauty[n]   = max_beauty[n-t] + 1;
	//      replacements[n] = replacements[n-t] + r;
	//
	//      // or, just don't place t at the end
	//      max_beauty[n] = max_beauty[n-1];
	//      replacements[n] = replacements[n-1];
	// }
	// Let's do this bottom up. Starting point:
	// max_beauty[0] = 0;
	// replacements[0] = 0;
	// now to determine if we can place t at position p:
	//    ???a??a???baba???
	//    ???f??t???ffff???
	// t: 32106543210000321   (fill backwards, start with 0 for every f)
	// f: 6543210T987654321   (fill backwards, start with 0 for every t)
	// How many question marks are there in the next t chars?
	//    ???a??a???baba???
	//    32222223210012321   (fill backwards, diff is look at cur char and char at cur+t-1)
	// check if l_(p is odd or even?)[p] >= t
	// if yes, check question_marks_count for r.
		
	public static void main(String[] args) throws Exception {
		Scanner sc = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
		
		int n = sc.nextInt();
		char[] c = sc.next().toCharArray();
		int t = sc.nextInt();
		
		int[] type = new int[n]; // regular = 0, goofy = 1, ? = 2
		for (int i=0; i<n; i++) {
			if (c[i] == '?') {
				type[i] = 2;
			} else if (c[i] == 'a') {
				type[i] = (i%2) == 0 ? 0 : 1;
			} else if (c[i] == 'b') {
				type[i] = (i%2) == 1 ? 0 : 1;
			}
		}
		
		//System.out.println(Arrays.toString(type));
		
		int[][] lengthSameType = new int[2][n];
		for (int i=0; i<2; i++) {
			int last = 0;
			for (int j=n-1; j>=0; j--) {
				if (type[j] == i || type[j] == 2) {
					lengthSameType[i][j] = last+1;
					last++;
				} else {
					lengthSameType[i][j] = 0;
					last = 0;
				}
			}
		}

		//System.out.println(Arrays.toString(lengthSameType[0]));
		//System.out.println(Arrays.toString(lengthSameType[1]));
		
		int[] questionMarks = new int[n];
		int qmlast = 0;
		for (int i=n-1; i>=0; i--) {
			int add = type[i] == 2 ? 1 : 0;
			int end = (i+t);
			int subtract = (end < n && type[end] == 2) ? 1 : 0;
			questionMarks[i] = qmlast + add - subtract;
			qmlast = questionMarks[i];
		}

		//System.out.println(Arrays.toString(questionMarks));
		
		int[] maxBeauty = new int[n+1];
		int[] minReplacements = new int[n+1];
		maxBeauty[0] = 0;
		minReplacements[0] = 0;
		for(int i=1; i<=n; i++) {
			maxBeauty[i] = maxBeauty[i-1];
			minReplacements[i] = minReplacements[i-1];
			
			int p = i-t;
			int typeAtP = p%2;
			boolean canBePlacedAtEnd = p >= 0 && lengthSameType[typeAtP][p] >= t;
			if (canBePlacedAtEnd) {
				int r = questionMarks[p];
				
				int trial = maxBeauty[i-t] + 1;
				int trialr = minReplacements[i-t] + r;
				if (trial > maxBeauty[i] || (trial == maxBeauty[i] && trialr < minReplacements[i])) {
					maxBeauty[i] = trial;
					minReplacements[i] = trialr;					
				}
			}
		}
		
		System.out.println(minReplacements[n] + "");
	}

}
