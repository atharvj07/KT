#include <bits/stdc++.h>
using namespace std;
int Mod = 1000000007;
int C[107][107];
int first[107][17];
bool isCalc[107][17];
int A[17];
int ans(int n, int k) {
  if (isCalc[n][k]) return first[n][k];
  isCalc[n][k] = true;
  if (k == 9) {
    if (n >= A[k])
      first[n][k] = 1;
    else
      first[n][k] = 0;
    return first[n][k];
  }
  for (int i = A[k]; i <= n; i++) {
    if (k == 0)
      first[n][k] = (first[n][k] + 1LL * ans(n - i, k + 1) * C[n - 1][i]) % Mod;
    else
      first[n][k] = (first[n][k] + 1LL * ans(n - i, k + 1) * C[n][i]) % Mod;
  }
  return first[n][k];
}
int main() {
  int i, j, k, l, m, n, x, y, z, a, b, r;
  C[0][0] = 1;
  for (i = 1; i <= 100; i++) {
    C[i][0] = C[i][i] = 1;
    for (j = 1; j < i; j++) {
      C[i][j] = (C[i - 1][j - 1] + C[i - 1][j]) % Mod;
    }
  }
  scanf("%d", &n);
  for (x = 0, i = 0; i < 10; i++) {
    scanf("%d", &A[i]);
    x += A[i];
  }
  a = 0;
  for (i = x; i <= n; i++) {
    a = (a + ans(i, 0)) % Mod;
  }
  printf("%d\n", a);
  return 0;
}
