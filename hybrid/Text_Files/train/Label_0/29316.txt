#include <bits/stdc++.h>
using namespace std;
const int MAXN = 500005;
int N, M, C, ban[MAXN], ans[MAXN], done[MAXN];
set<int> pset;
vector<int> color[MAXN], E[MAXN];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  ;
  cin >> N >> M;
  C = 0;
  int rt = 0;
  for (int i = (1); i <= (N); i++) {
    int K;
    cin >> K;
    color[i].resize(K);
    for (int j = 0; j < (K); j++) cin >> color[i][j];
    if (K > C) {
      C = K;
      rt = i;
    }
    sort(begin(color[i]), end(color[i]));
  }
  for (int _ = 0; _ < (N - 1); _++) {
    int u, v;
    cin >> u >> v;
    E[u].push_back(v);
    E[v].push_back(u);
  }
  queue<int> que;
  for (int i = (1); i <= (C); i++) {
    ans[color[rt][i - 1]] = i;
    pset.insert(i);
  }
  done[rt] = 1;
  que.push(rt);
  while (!que.empty()) {
    int u = que.front();
    que.pop();
    for (auto it : color[u]) ban[it] = 1;
    for (auto v : E[u]) {
      if (done[v]) continue;
      done[v] = 1;
      for (auto it : color[v]) {
        if (ban[it]) {
          pset.erase(ans[it]);
          assert(ans[it]);
        }
      }
      for (auto it : color[v]) {
        if (ans[it]) continue;
        assert(!pset.empty());
        ans[it] = *pset.begin();
        pset.erase(pset.begin());
      }
      que.push(v);
      for (auto it : color[v]) {
        pset.insert(ans[it]);
      }
    }
    for (auto it : color[u]) ban[it] = 0;
  }
  for (int i = (1); i <= (M); i++) {
    if (!ans[i]) ans[i] = 1;
  }
  C = max(C, 1);
  cout << C << endl;
  for (int i = (1); i <= (M); i++) cout << ans[i] << " \n"[i == M];
  for (int i = (1); i <= (N); i++) {
    int k = ((int)((color[i]).size()));
    for (int j = 0; j < (k - 1); j++)
      assert(ans[color[i][j]] != ans[color[i][j + 1]]);
  }
  return 0;
}
