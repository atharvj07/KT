(n, x) = tuple(map(int, input().split()))

def get_4_n_numbrs(n):
	(a, b, c, d) = (5, 10, 12, 3)
	res = []
	for i in range(n):
		pref = '1{0:014b}'.format(i)
		pref += '{0:04b}'
		res.append(pref.format(a))
		res.append(pref.format(b))
		res.append(pref.format(c))
		res.append(pref.format(d))
	return res

def get_3_nums(x):
	if x not in [1, 2]:
		x = 1 ^ 2 ^ x
		return [1, 2, x]
	else:
		x = 8 ^ 4 ^ x
		return [8, 4, x]

def get_4_nums(x):
	if x not in [1, 2, 3]:
		return [1, 2, 3, x]
	else:
		xx = 8 ^ 16 ^ 32 ^ x
		return [8, 16, 32, xx]

def get_2_nums(x):
	if x == 0:
		return [33, 3, 6, 12, 24, 48]
	else:
		return [0, x]

def get_numbers(n, x):
	if n == 1:
		return [x]
	fours = (n - 1) // 4
	res = []
	left = n - fours * 4
	if left == 2 and x == 0:
		if n == 2:
			return -1
		fours -= 1
	res.extend([int(xx, 2) for xx in get_4_n_numbrs(fours)])
	if left == 1:
		res.append(x)
	if left == 2:
		res.extend(get_2_nums(x))
	if left == 3:
		res.extend(get_3_nums(x))
	if left == 4:
		res.extend(get_4_nums(x))
	return res
res = get_numbers(n, x)
if res != -1:
	print('YES')
	print(' '.join(map(str, get_numbers(n, x))))
else:
	print('NO')
