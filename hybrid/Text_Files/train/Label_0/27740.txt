#include <bits/stdc++.h>
using namespace std;
vector<int> ans(10000010, 1000000000);
int main() {
  long long a, n;
  cin >> a >> n;
  for (long long i = 1; i * i <= a + n; ++i) {
    long long s = i * i;
    for (long long j = 1; s * j < a + n; ++j) {
      if (s * j >= a) {
        ans[s * j] = j;
      }
    }
  }
  long long an = 0;
  for (long long y = a; y < a + n; ++y) {
    an += ans[y];
  }
  cout << an << endl;
  return 0;
}
