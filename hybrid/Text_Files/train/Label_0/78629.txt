def main():

	def solve(n, m):
		if n == m == 1:
			return 0
		if n == 1 or m == 1:
			return 1
		return 2
	t = int(input())
	ans = []
	for _ in range(t):
		(n, m) = [int(x) for x in input().split()]
		ans.append(solve(n, m))
	print('\n'.join(map(str, ans)))
import sys, threading
input = sys.stdin.readline
thread = threading.Thread(target=main)
thread.start()
thread.join()
