t = int(input())
for _ in range(t):
	n = int(input())
	a = list(map(int, input().split()))
	a.append(float('-inf'))
	l = []
	x = 1
	for i in range(1, n):
		if a[i] < a[i + 1]:
			x += 1
			continue
		l.append(x)
		x = 1
	l.reverse()
	x = l.pop()
	h = 1
	while len(l) > 0:
		ans = 0
		while x > 0 and len(l) > 0:
			ans += l.pop()
			x -= 1
		h += 1
		x = ans
	print(h)
