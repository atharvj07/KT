c = list(map(int, input().split()))
(n, m) = map(int, input().split())
b = list(map(int, input().split()))
t = list(map(int, input().split()))
d = 0
for x in b:
	if x * c[0] <= c[1]:
		d += x * c[0]
	else:
		d += c[1]
if d > c[2]:
	d = c[2]
f = 0
for x in t:
	if x * c[0] < c[1]:
		f += x * c[0]
	else:
		f += c[1]
if f > c[2]:
	f = c[2]
if d + f < c[3]:
	print(d + f)
else:
	print(c[3])
