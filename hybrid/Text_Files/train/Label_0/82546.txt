from sys import stdin, stdout
Q = int(input())
mini = 0
maxi = 0
for i in range(Q):
	arr = [x for x in stdin.readline().split()]
	arr[1] = int(arr[1])
	arr[2] = int(arr[2])
	if arr[0] == '+':
		mini = max(mini, min(arr[1], arr[2]))
		maxi = max(maxi, max(arr[1], arr[2]))
	elif arr[0] == '?':
		if min(arr[1], arr[2]) >= mini and max(arr[1], arr[2]) >= maxi:
			print('YES')
		else:
			print('NO')
