def check(arrayList1, arrayList2):
	f = 0
	minList1 = min(arrayList1)
	arrayList2.sort()
	for item in range(0, len(arrayList1)):
		if arrayList1[item] != arrayList2[item] and arrayList1[item] % minList1:
			f = 1
	if f:
		print('NO')
	else:
		print('YES')
t = int(input())
for i in range(0, t):
	n = int(input())
	lists1 = list((int(num) for num in input().strip().split()))[:n]
	lists2 = lists1.copy()
	check(lists1, lists2)
