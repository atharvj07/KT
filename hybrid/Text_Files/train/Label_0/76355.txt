import math
for _ in range(int(input())):
	(n, m) = map(int, input().split())
	ans = n * m
	ans1 = n * (m - 1)
	ans2 = n * 1
	if ans1 + ans2 == ans and ans1 > 0 and (ans2 > 0) and (ans > 0):
		if ans1 == ans2:
			print('YES')
			print(ans1 * 3, ans2 * 1, ans * 2)
		else:
			print('YES')
			print(ans1, ans2, ans)
	else:
		print('NO')
