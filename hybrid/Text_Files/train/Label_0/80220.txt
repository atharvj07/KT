import sys, math
input = sys.stdin.readline
t = int(input())
for tttt in range(t):
	(n, q) = map(int, input().split())
	s = input().strip('\n')
	for qq in range(q):
		(l, r) = map(int, input().split())
		l -= 1
		r -= 1
		flag = False
		for i in range(0, l):
			if s[i] == s[l]:
				flag = True
				break
		for i in range(r + 1, n):
			if s[i] == s[r]:
				flag = True
				break
		if flag == True:
			print('YES')
		else:
			print('NO')
