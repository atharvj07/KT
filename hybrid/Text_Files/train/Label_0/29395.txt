int m;
vector<bool> v;
int a[10];

int dfs(int now, int x) {
  if (now == v.size()) {
    return x % 10 == 0;
  }
  int res = 0;
  REP(i,m) {
    int t = a[i];
    if (v[now]) t*=2;
    res += dfs(now+1, x+t/10+t%10);
  }
  return res;
}

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  bool f = 0;
  int x = 0;
  v.clear();
  for (int i=s.size()-1; i>=0; --i) {
    if (s[i] != '*') {
      int a = s[i]-'0';
      if (f) a*=2;
      x += a / 10 + a % 10;
    } else {
      v.push_back(f);
    }
    f = !f;
  }
  cin >> m;
  REP(i,m) {
    cin >> a[i];
  }
  cout << dfs(0,x) << endl;
}
