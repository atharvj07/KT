#include <bits/stdc++.h>
using namespace std;
int n, s, h, m;
int t[105];
int main() {
  cin >> n >> s;
  for (int i = 1; i <= n; i++) {
    cin >> h >> m;
    t[i] = h * 60 + m;
  }
  t[0] = -s - 1;
  int ans = t[n] + s + 1;
  for (int i = 0; i < n; i++) {
    if (t[i + 1] - t[i] > 2 * s + 1) {
      ans = t[i] + s + 1;
      break;
    }
  }
  cout << ans / 60 << ' ' << ans % 60 << endl;
  return 0;
}
