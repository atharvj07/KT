#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e7;
long long power[5000], a[maxn + 10];
void get() {
  for (long long i = 1; i * i <= 1e7 + 5; i++) {
    power[i] = i * i;
  }
}
int main() {
  get();
  memset(a, 0x3f3f3f3f, sizeof(a));
  for (long long i = 1; i <= maxn; i++) {
    for (int j = 1; j <= 4000; j++) {
      if (i * power[j] > maxn) break;
      a[i * power[j]] = min(a[i * power[j]], i);
    }
  }
  long long ans = 0;
  int n, c;
  cin >> n >> c;
  for (int i = n; i <= n + c - 1; i++) {
    ans += a[i];
  }
  cout << ans << endl;
  return 0;
}
