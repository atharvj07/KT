#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  string s;
  cin >> n >> m >> s;
  string ans = "abcdefghijklmnopqrstuvwxyz";
  for (int i = 0; i < m; i++) {
    char u, v;
    cin >> u >> v;
    for (int i = 0; i < ans.size(); i++) {
      if (ans[i] == u)
        ans[i] = v;
      else if (ans[i] == v)
        ans[i] = u;
    }
  }
  for (int i = 0; i < s.size(); i++) {
    s[i] = ans[s[i] - 'a'];
  }
  cout << s << endl;
  return 0;
}
