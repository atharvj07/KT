t = int(input())
while t:
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	b = list(map(int, input().split()))
	res = []
	a.sort()
	b.sort(reverse=True)
	if a[0] > b[0]:
		print(sum(a))
		t -= 1
		continue
	s = 0
	for i in range(n):
		if k > 0:
			if a[i] >= b[i]:
				s += a[i]
				continue
			else:
				s += b[i]
				k -= 1
				continue
		s += a[i]
	print(s)
	t -= 1
