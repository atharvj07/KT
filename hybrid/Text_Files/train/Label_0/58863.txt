import java.util.*;

public final class  cont686A
{
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int T=sc.nextInt();
        while(T>0)
        {
            int n=sc.nextInt();
            int []a=new int[n+1];
            List<Integer>adj[]=new ArrayList[n+1];
            HashMap<Integer,Integer>mp=new HashMap<>();
            for(int i=1;i<=n;i++)
                adj[i]=new ArrayList<>();
            for(int i=1;i<=n;i++)
            {
                    a[i]=sc.nextInt();
                    mp.put(a[i],mp.getOrDefault(a[i],0)+1);
                    adj[a[i]].add(i);
            }
            
            int min=Integer.MAX_VALUE;

            
            for(int x:mp.keySet())
            {
                List<Integer>lst=adj[x];
                int cnt=0;
                int c1=lst.get(0)-1;
                if(c1>0)
                    cnt++;
                int c2=n-lst.get(lst.size()-1);
                if(c2>0)
                    cnt++;
                for(int i=0;i<lst.size()-1;i++)
                {
                    int c=lst.get(i+1)-lst.get(i)-1;
                    if(c>0)
                        cnt++;
                }
                min=Math.min(min,cnt);

            }

            System.out.println(min);

            T--;
        }
        
    }
}
