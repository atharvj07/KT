template<class d>
struct maxflow{
	struct E{
		int to,rev;
		d cap;
	};
	vvc<E> g;
	vi itr,lv;
	maxflow(int n):g(n),itr(n),lv(n){}
	void ae(int s,int t,d c){
		g[s].pb({t,(int)g[t].size(),c});
		g[t].pb({s,(int)g[s].size()-1,0});
	}
	void bfs(int s){
		fill(all(lv),-1);
		lv[s]=0;
		queue<int> q;
		q.push(s);
		while(q.size()){
			int v=q.front();q.pop();
			for(auto e:g[v])if(e.cap>0&&lv[e.to]==-1){
				lv[e.to]=lv[v]+1;
				q.push(e.to);
			}
		}
	}
	d dfs(int v,int t,d f){
		if(v==t)return f;
		d res=0;
		for(int&i=itr[v];i<(int)g[v].size();i++){
			E& e=g[v][i];
			if(e.cap>0&&lv[e.to]==lv[v]+1){
				d a=dfs(e.to,t,min(f,e.cap));
				if(a>0){
					e.cap-=a;
					g[e.to][e.rev].cap+=a;
					res+=a;
					f-=a;
					if(f<=0)break;
				}
			}
		}
		return res;
	}
	d calc(int s,int t){
		d f=0;
		while(1){
			bfs(s);
			if(lv[t]==-1)
				return f;
			fill(all(itr),0);
			f+=dfs(s,t,inf);
		}
	}
};

const int nmax=45;
//r,c,col,vh
int id[nmax][nmax][2][2];

signed main(){
	cin.tie(0);
	ios::sync_with_stdio(0);
	cout<<fixed<<setprecision(20);
	
	one(id);
	
	int n,m,a,b,c;cin>>n>>m>>a>>b>>c;
	vc<string> w(n);
	rep(i,n)cin>>w[i];
	
	vc<tuple<int,int,int>> es;
	int s=2;
	int ans=0;
	
	rep(i,n)rep(j,m){
		if(w[i][j]=='#'){
			int d=s++;
			int e=s++;
			id[i][j][0][0]=d;
			id[i][j][0][1]=e;
			
			es.eb(0,d,c);
			es.eb(d,1,a+b);
			es.eb(0,e,a+b);
			es.eb(e,1,c);
			es.eb(d,e,c);
			
			ans-=c;
		}else{
			int d=s++;
			int e=s++;
			id[i][j][0][0]=d;
			id[i][j][0][1]=e;
			int f=s++;
			int g=s++;
			id[i][j][1][0]=f;
			id[i][j][1][1]=g;
			
			es.eb(d,1,a+b);
			es.eb(0,e,a+b);
			es.eb(0,f,a+b);
			es.eb(g,1,a+b);
			
			es.eb(d,e,inf);
			es.eb(d,f,inf);
			es.eb(d,g,c);
			es.eb(f,e,c);
			es.eb(g,e,inf);
		}
	}
	
	maxflow<int> mf(s);
	for(auto e:es){
		int x,y,z;tie(x,y,z)=e;
		mf.ae(x,y,min(z*2,inf));
	}
	ans*=2;
	auto addleft=[&](int x,int y){
		mf.ae(x,y,b);
		mf.ae(y,x,b);
		mf.ae(x,1,b);
		mf.ae(y,1,b);
		ans-=b*2;
	};
	auto addright=[&](int x,int y){
		mf.ae(x,y,b);
		mf.ae(y,x,b);
		mf.ae(0,x,b);
		mf.ae(0,y,b);
		ans-=b*2;
	};
	
	rep(i,n-1)rep(j,m)rep(k,2){
		int x=id[i][j][k][0];
		int y=id[i+1][j][k][0];
		if(x!=-1&&y!=-1){
			if(k==0)addright(x,y);
			else addleft(x,y);
		}
	}
	rep(i,n)rep(j,m-1)rep(k,2){
		int x=id[i][j][k][1];
		int y=id[i][j+1][k][1];
		if(x!=-1&&y!=-1){
			if(k==0)addleft(x,y);
			else addright(x,y);
		}
	}
	
	ans+=mf.calc(0,1);
	
	cout<<ans/2<<endl;
}
