n = int(input())
n1 = []
for i in range(n):
	(a, b, c) = map(int, input().split())
	summa = a + b + c
	if summa % 9 == 0:
		summa = int(summa / 9)
		a -= summa
		b -= summa
		c -= summa
		if a >= 0 and b >= 0 and (c >= 0):
			n1 += ['YES']
		else:
			n1 += ['NO']
	else:
		n1 += ['NO']
for i in n1:
	print(i)
