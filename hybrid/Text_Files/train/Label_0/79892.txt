n = int(input())
dict = [set() for i in range(n + 1)]
for _ in range(n - 1):
	(a, b) = map(int, input().split())
	dict[a].add(b)
	dict[b].add(a)
seq = list(map(int, input().split()))
i = 1
j = 0
if seq[0] != 1:
	print('No')
	exit()
while i < n and j < n:
	if seq[i] in dict[seq[j]]:
		i += 1
	else:
		j += 1
if i == n:
	print('Yes')
else:
	print('No')
