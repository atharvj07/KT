#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, m, q;
int ans[N], a[N], b[N], nt[N], last[N];
int f[N][20];
int main() {
  scanf("%d%d%d", &n, &m, &q);
  int tt = m + 1;
  for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
  a[n + 1] = a[1];
  for (int i = 1; i <= m; i++) scanf("%d", &b[i]);
  for (int i = 1; i <= n; i++) nt[a[i]] = a[i + 1];
  for (int i = 1; i <= n; i++) last[i] = m + 1;
  for (int i = 0; i <= 19; i++) f[m + 1][i] = m + 1;
  for (int i = m; i >= 1; i--) {
    f[i][0] = last[nt[b[i]]];
    last[b[i]] = i;
    for (int j = 1; j <= 19; j++) f[i][j] = f[f[i][j - 1]][j - 1];
    int len = n - 1, p = i;
    for (int j = 19; j >= 0; j--) {
      if (len >= (1 << j)) {
        len -= 1 << j;
        p = f[p][j];
      }
    }
    tt = min(tt, p);
    ans[i] = tt;
  }
  for (int i = 1; i <= q; i++) {
    int l, r;
    scanf("%d%d", &l, &r);
    if (ans[l] <= r)
      printf("1");
    else
      printf("0");
  }
  return 0;
}
