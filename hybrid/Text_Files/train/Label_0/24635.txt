#include <bits/stdc++.h>
using namespace std;
const long long int INF = (long long int)(8e18);
const long long int MOD = 1e9 + 7;
int a[5010], ans[5010];
int freq[5010];
int main() {
  ios ::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  int n;
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    int maxi = 0;
    int val = 0;
    for (int j = i; j < n; j++) {
      freq[a[j]]++;
      if (freq[a[j]] > maxi) {
        maxi = freq[a[j]];
        val = a[j];
      } else {
        if (freq[a[j]] == maxi) {
          if (a[j] < val) {
            val = a[j];
          }
        }
      }
      ans[val]++;
    }
    memset(freq, 0, sizeof freq);
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
  cout << '\n';
}
