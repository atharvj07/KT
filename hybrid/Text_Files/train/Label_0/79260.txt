import sys
import bisect
input = sys.stdin.readline

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	arr = [a]
	while True:
		d = {}
		for i in arr[-1]:
			if i in d:
				d[i] += 1
			else:
				d[i] = 1
		arr.append([d[arr[-1][i]] for i in range(n)])
		for i in range(n):
			if arr[-1][i] != arr[-2][i]:
				break
		else:
			break
	for q in range(int(input())):
		(x, k) = map(int, input().split())
		k = min(k, len(arr) - 1)
		print(arr[k][x - 1])
t = int(input())
for _ in range(t):
	solve()
