#include <bits/stdc++.h>
using namespace std;
const int64_t MAX = 5005;
int64_t calc[MAX][MAX], ans[MAX];
int32_t main() {
  ios_base::sync_with_stdio();
  cin.tie(0);
  cout.tie(0);
  string s;
  cin >> s;
  int64_t n = ((int64_t)s.size());
  for (int64_t len = 1; len < n + 1; ++len)
    for (int64_t i = 0; i < n; ++i) {
      if (i + len > n) break;
      if (s[i] != s[i + len - 1]) continue;
      if (len <= 2)
        ++calc[i][i + len - 1];
      else if (calc[i + 1][i + len - 2])
        ++calc[i][i + len - 1];
    }
  for (int64_t len = 2; len < n + 1; ++len)
    for (int64_t i = 0; i < n; ++i) {
      if (i + len > n) break;
      int64_t j = i + len - 1;
      if (calc[i][j] && calc[i][i + len / 2 - 1])
        calc[i][j] =
            1 + max(calc[i][i + len / 2 - 1], calc[j - len / 2 + 1][j]);
      if (calc[i][j]) ans[calc[i][j] - 1]++;
    }
  int64_t cur = 0;
  for (int64_t i = n - 1; i >= 0; --i) {
    cur += ans[i];
    ans[i] = cur;
  }
  ans[0] += n;
  for (int64_t i = 0; i < n; ++i) cout << ans[i] << " ";
  return 0;
}
