// C - Sequence
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define rp(i,n) for(int i=0;i<(n);i++)

ll counter(vector<ll> &A, int N, int sign){//sign: (+)start -> 1, (-)start -> -1
	ll c = 0, s = 0;
	rp(i,N){
		s += A[i];
		if(sign*s<=0){
			c += abs(s) + 1;
			s = sign;
		}
		sign *= -1;
	}
	return c;
}

int main(){
	int N; cin>>N;
	vector<ll> A(N);
	rp(i,N) cin>>A[i];
	ll a = counter(A,N, 1);
	ll b = counter(A,N,-1);
	cout<< (a<b?a:b) <<endl;
}
