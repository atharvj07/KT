import collections
import sys
import functools
import heapq
import bisect
import collections
import math
input = sys.stdin.readline
mod = 10 ** 9 + 7
t = int(input())
for _ in range(t):
	n = int(input().strip())
	s = input().strip()
	right = [[0, 0] for i in range(n)]
	left = [[0, 0] for i in range(n)]
	if s[-1] == 'L':
		right[-1][1] = 1
	else:
		right[-1][0] = 1
	for i in range(n - 2, -1, -1):
		if s[i] == 'L':
			right[i][1] = 1 + right[i + 1][0]
		else:
			right[i][0] = 1 + right[i + 1][1]
	if s[0] == 'L':
		left[0][1] = 1
	else:
		left[0][0] = 1
	for i in range(1, n):
		if s[i] == 'L':
			left[i][1] = 1 + left[i - 1][0]
		else:
			left[i][0] = 1 + left[i - 1][1]
	for i in range(n + 1):
		if i == 0:
			print(1 + right[i][0], end=' ')
		elif i == n:
			print(1 + left[i - 1][1], end=' ')
		else:
			print(1 + left[i - 1][1] + right[i][0], end=' ')
	print()
