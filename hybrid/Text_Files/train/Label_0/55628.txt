import java.util.*;
import java.io.*;

// dijkstra
public class cf20c {
  static long INF = Long.MAX_VALUE;
  static int V, E;
  static List<Node>[] adj;
  static long[] dist;
  static int[] pre;

  public static void main(String[] args) throws Exception {
    // Scanner in = new Scanner(new File("in"));
    Scanner in = new Scanner(System.in);

    V = in.nextInt();
    E = in.nextInt();
    adj = new LinkedList[V + 1];
    dist = new long[V + 1];
    pre = new int[V + 1];
    Arrays.fill(dist, INF);
    for (int i = 1; i <= V; i++)
      adj[i] = new LinkedList<>();

    for (int i = 0; i < E; i++) {
      int u = in.nextInt();
      int v = in.nextInt();
      int w = in.nextInt();
      adj[u].add(new Node(v, w));
      adj[v].add(new Node(u, w));
    }

    // System.out.println(Arrays.toString(adj));
    // ---------------------------------------------------

    PriorityQueue<Node> q = new PriorityQueue<>((a, b) -> a.w < b.w ? -1 : 1);
    q.add(new Node(1, 0));
    dist[1] = 0;

    while (!q.isEmpty()) {
      Node cur = q.remove();
      if (cur.w > dist[cur.v])
        continue;

      for (Node next : adj[cur.v]) {

        long w = dist[cur.v] + next.w;
        if (dist[next.v] > w) {
          dist[next.v] = w;
          pre[next.v] = cur.v;
          q.add(new Node(next.v, w));
        }
      }
    }

    if (dist[V] == INF) {
      System.out.println(-1);
      return;
    }

    Deque<Integer> stack = new LinkedList<>();
    int node = V;
    stack.add(node);
    while (node != 1) {
      node = pre[node];
      stack.add(node);
    }

    while (!stack.isEmpty()) {
      System.out.print(stack.removeLast() + " ");
    }

  }

  static class Node {
    int v;
    long w;

    public Node(int v, long w) {
      this.v = v;
      this.w = w;
    }
  }
}
