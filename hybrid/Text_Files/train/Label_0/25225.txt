#include <bits/stdc++.h>
using namespace std;
int i, n, v, a, b, x[3005], ans, one, two;
int min(int xi, int v) {
  if (xi > v) return v;
  return xi;
}
int main() {
  cin >> n >> v;
  for (i = 0; i < n; i++) {
    cin >> a >> b;
    x[a] += b;
  }
  for (i = 0; i < 3002; i++) {
    one = min(x[i], v);
    ans += one;
    two = min(x[i + 1], v - one);
    ans += two;
    x[i + 1] -= two;
  }
  cout << ans << endl;
  return 0;
}
