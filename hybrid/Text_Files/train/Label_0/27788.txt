#include <bits/stdc++.h>
using namespace std;
int n;
char a[105][105];
bool vis[105][105];
bool valid(int x, int y) {
  bool ok = (x > 0 && y > 0 && x < n - 1 && y < n - 1 && a[x][y] == '#');
  for (int i = -1; i < 2; i++)
    ok &= (a[x + i][y] == a[x][y + i] && a[x + i][y] == '#' && !vis[x + i][y] &&
           !vis[x][y + i]);
  return ok;
}
void cmp(int x, int y) {
  for (int i = -1; i < 2; i++) vis[x + i][y] = vis[x][y + i] = 1;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      if (valid(i, j) && !vis[i][j]) cmp(i, j);
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      if (!vis[i][j] && a[i][j] == '#') return puts("NO");
  puts("YES");
  return 0;
}
