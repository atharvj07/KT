import collections as col

from types import GeneratorType

def solve():

	def bootstrap(f, stack=[]):

		def wrappedfunc(*args, **kwargs):
			if stack:
				return f(*args, **kwargs)
			else:
				to = f(*args, **kwargs)
				while True:
					if type(to) is GeneratorType:
						stack.append(to)
						to = next(to)
					else:
						stack.pop()
						if not stack:
							break
						to = stack[-1].send(to)
				return to
		return wrappedfunc

	@bootstrap
	def dfs(graph, visited, node, visited_dict):
		if node not in visited:
			visited.append(node)
			visited_dict[node] = 1
		for neighbour in graph[node]:
			if neighbour not in visited_dict:
				yield dfs(graph, visited, neighbour, visited_dict)
		yield visited

	@bootstrap
	def bfs(graph, visited, node, queue, visited_dict, nodes):
		level = 1
		visited.append((node, level))
		visited_dict[node] = 1
		nodes.remove(node)
		queue.append((node, level))
		queue = col.deque(queue)
		while queue:
			(s, level) = queue.popleft()
			for neighbour in graph[s]:
				if neighbour not in visited_dict:
					visited.append((neighbour, level + 1))
					visited_dict[neighbour] = 1
					nodes.remove(neighbour)
					queue.append((neighbour, level + 1))
		yield (visited, nodes)
	(N, M) = getInts()
	graph = col.defaultdict(set)
	edges = []
	for n in range(M):
		(U, V) = getInts()
		graph[U].add(V)
		graph[V].add(U)
		edges.append((U, V))
	nodes = set([j + 1 for j in range(N)])
	connected_components = []
	(dict_A, dict_B) = (col.defaultdict(int), col.defaultdict(int))
	(A, B) = ([], [])
	while nodes:
		node = next(iter(nodes))
		(visited, nodes) = bfs(graph, [], node, [], {}, nodes)
		if len(visited) == 1:
			continue
		for (node, level) in visited:
			if level % 2 == 0:
				dict_A[node] = node
				A.append(node)
			else:
				dict_B[node] = node
				B.append(node)
	for (U, V) in edges:
		if U in dict_A and V in dict_A or (U in dict_B and V in dict_B):
			print(-1)
			return
	print(len(A))
	print(*A)
	print(len(B))
	print(*B)
	return
solve()
