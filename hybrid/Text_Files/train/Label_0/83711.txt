s = input()
t = input()
arrS = [0] * 70
arrT = [0] * 70
for i in range(len(s)):
	arrS[ord(s[i]) - ord('A')] += 1
for i in range(len(t)):
	arrT[ord(t[i]) - ord('A')] += 1
yay = 0
whoops = 0
for i in range(70):
	temp = min(arrS[i], arrT[i])
	yay += temp
	arrT[i] -= temp
	arrS[i] -= temp
for i in range(28):
	temp = min(arrS[i], arrT[i + 32])
	arrT[i + 32] -= temp
	arrS[i] -= temp
	whoops += temp
for i in range(32, 68):
	temp = min(arrS[i], arrT[i - 32])
	arrT[i - 32] -= temp
	arrS[i] -= temp
	whoops += temp
print('{} {}'.format(yay, whoops))
