#include <bits/stdc++.h>
using namespace ::std;
const int maxn = 2500;
const int mod = 1e9 + 7;
const int inf = 2e9 + 100;
string s[maxn];
int a[maxn];
int n, m;
unsigned int pr[maxn];
int dp[maxn][maxn];
pair<int, int> id(int x) {
  int r = lower_bound(pr, pr + m + 1, x) - (&pr[0]);
  if (pr[r] == x) {
    return make_pair(r, x - pr[r]);
  }
  return make_pair(r - 1, x - pr[r - 1]);
}
unsigned int find_nxt_out[maxn][26];
vector<unsigned int> find_nxt_in[maxn][26];
int find_nxt(int x, char c) {
  if (x >= inf) {
    return inf;
  }
  pair<int, int> u = id(x);
  if (u.first >= m) {
    return inf;
  }
  return min(find_nxt_in[a[u.first]][c][u.second] + pr[u.first],
             find_nxt_out[u.first + 1][c]);
}
string okeye(string a) {
  for (int i = 0; i < a.size(); i++) {
    a[i] -= 'a';
  }
  return a;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    s[i] = okeye(s[i]);
  }
  for (int i = 0; i < n; i++) {
    for (int j = -1; j < (int)s[i].size(); j++) {
      for (int k = 0; k < 26; k++) {
        find_nxt_in[i][k].push_back(inf);
      }
    }
    for (int j = s[i].size() - 1; j >= 0; j--) {
      for (int k = 0; k < 26; k++) {
        find_nxt_in[i][k][j] = find_nxt_in[i][k][j + 1];
      }
      find_nxt_in[i][s[i][j]][j] = j;
    }
  }
  cin >> m;
  int pp = 0;
  for (int i = 0; i < m; i++) {
    pr[i] = pp;
    cin >> a[i];
    a[i]--;
    pp += s[a[i]].size();
  }
  pr[m] = pp;
  for (int k = 0; k < 26; k++) {
    find_nxt_out[m][k] = inf;
  }
  for (int i = m - 1; i >= 0; i--) {
    for (int k = 0; k < 26; k++) {
      find_nxt_out[i][k] =
          min(find_nxt_out[i + 1][k], find_nxt_in[a[i]][k][0] + pr[i]);
    }
  }
  string t;
  cin >> t;
  t = okeye(t);
  int tol = t.size();
  {
    int i = 0;
    dp[i][0] = -1;
    dp[i][1] = find_nxt(0, t[0]);
    for (int j = 2; j <= tol; j++) {
      dp[i][j] = inf;
    }
  }
  for (int i = 1; i < tol; i++) {
    dp[i][0] = -1;
    for (int j = 1; j <= tol; j++) {
      dp[i][j] = min(dp[i - 1][j], find_nxt(dp[i - 1][j - 1] + 1, t[i]));
    }
  }
  for (int j = tol; j >= 0; j--) {
    if (dp[tol - 1][j] < inf) {
      cout << j;
      return 0;
    }
  }
}
