n = int(input())
c = list(map(int, input().split(' ')))
t = list(map(int, input().split(' ')))
check = c[0] == t[0]
c_diff = [c[i] - c[i - 1] for i in range(1, n)]
t_diff = [t[i] - t[i - 1] for i in range(1, n)]
c_diff.sort()
t_diff.sort()
if check and c_diff == t_diff:
	print('Yes')
else:
	print('No')
