class fenwick:

	def __init__(self, n):
		self.n = n
		self.arr = [0] * (n + 1)

	def update(self, ind, x):
		if ind <= 0:
			return
		while ind <= self.n:
			self.arr[ind] += x
			ind += ind & -ind

	def query(self, ind):
		s = 0
		while ind > 0:
			s += self.arr[ind]
			ind -= ind & -ind
		return s

def solve(n, s):
	fen = fenwick(n + 1)
	inv = 0
	t = s[::-1]
	chars = collections.defaultdict(collections.deque)
	for i in range(n):
		chars[ord(s[i]) - ord('a')].append(i)
	for i in range(n):
		v = chars[ord(t[i]) - ord('a')].popleft()
		fen.update(v + 1, 1)
		v += i + 1 - fen.query(v + 1)
		inv += v - i
	return inv
n = int(input())
s = input()
print(solve(n, s))
