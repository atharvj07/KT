from collections import Counter, defaultdict, deque
import sys
input = sys.stdin.readline

def solve():
	lens = [int(x) for x in input().split()]
	s1 = sorted([int(x) for x in input().split()])
	s2 = sorted([int(x) for x in input().split()])
	s3 = sorted([int(x) for x in input().split()])
	s1 = s1[:1] + ([sum(s1[1:])] if sum(s1[1:]) != 0 else [])
	s2 = s2[:1] + ([sum(s2[1:])] if sum(s2[1:]) != 0 else [])
	s3 = s3[:1] + ([sum(s3[1:])] if sum(s3[1:]) != 0 else [])
	ss = sorted(s1 + s2 + s3)
	best = ss[0] + ss[1]
	if lens[0] == 1:
		best = min(best, s1[0])
	if lens[1] == 1:
		best = min(best, s2[0])
	if lens[2] == 1:
		best = min(best, s3[0])
	print(sum(ss) - 2 * best)
tt = 1
for test in range(tt):
	solve()
