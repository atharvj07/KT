#include <bits/stdc++.h>
using namespace std;
const int maxn = 5e3 + 10;
const int maxm = 500 * 500 + 10;
const int inf = 1e9;
const long long mod = 1e9 + 7;
int getint() {
  char c;
  while ((c = getchar()) && !(c >= '0' && c <= '9') && c != '-')
    ;
  int ret = c - '0', sgn = 0;
  if (c == '-') sgn = 1, ret = 0;
  while ((c = getchar()) && c >= '0' && c <= '9') ret = ret * 10 + c - '0';
  if (sgn) ret = -ret;
  return ret;
}
struct node {
  int x, id;
  bool operator<(const node a) const { return x < a.x; }
} p[maxn];
int ans[maxn], fa[maxn], c[maxn][maxn];
vector<pair<int, int> > g[maxn];
void dfs(int u, int pa) {
  fa[u] = pa;
  int sz = g[u].size();
  for (int i = 0; i < sz; i++) {
    int v = g[u][i].first;
    if (v == pa) continue;
    dfs(v, u);
  }
}
int gg[maxn][maxn];
int main() {
  int n = getint();
  for (int i = 1; i <= n; i++) p[i].x = getint(), p[i].id = i;
  sort(p + 1, p + n + 1);
  for (int i = 1; i < n; i++) {
    int u = getint(), v = getint(), cc = getint();
    g[u].push_back(make_pair(v, cc));
    g[v].push_back(make_pair(u, cc));
    c[u][v] = c[v][u] = cc;
  }
  dfs(1, -1);
  for (int i = 1; i <= n; i++) {
    int t = 0;
    int u = p[i].id;
    while (u != 1) {
      int cost = c[u][fa[u]];
      if (gg[u][t] < cost) {
        gg[u][t]++;
        t++;
        u = fa[u];
      } else
        t++;
    }
    ans[p[i].id] = t;
  }
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  return 0;
}
