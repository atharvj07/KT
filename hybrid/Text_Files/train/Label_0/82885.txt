import math
n = int(input())
l1 = []
l2 = []
for i in range(1, math.floor(math.sqrt(n)) + 1):
	if n % i == 0:
		l = 1 + (n // i - 1) * i
		p1 = n // i * (1 + l) // 2
		l1.append(p1)
		l = 1 + (i - 1) * (n // i)
		p = int(i / 2 * (1 + l))
		if p1 != p:
			l2.append(p)
for j in range(len(l2)):
	print(l2[j], end=' ')
for j in range(len(l1)):
	print(l1[len(l1) - j - 1], end=' ')
