#include <bits/stdc++.h>
using namespace std;
const int INF = 2147483647;
const long long LLINF = 9223372036854775807LL;
const int maxn = 1000010;
int nxt[maxn][26] = {};
int len[maxn] = {};
pair<int, int> tt[10010];
int vv[10010];
int ans[10010];
int m;
int find(int val) {
  int l = -1;
  int r = m;
  while (r - l > 1) {
    int mid = (l + r) >> 1;
    if (tt[mid].first <= val)
      l = mid;
    else
      r = mid;
  }
  if (l == -1 || tt[l].first != val) return -1;
  return l;
}
int main() {
  string s;
  cin >> s;
  int n = s.length();
  for (int i = n - 1; i >= 0; --i) {
    if (i < n - 1 && s[i + 1] != s[i]) nxt[i][len[i]++] = i + 1;
    for (int j = 0; j < len[i + 1]; ++j)
      if (s[nxt[i + 1][j]] != s[i]) nxt[i][len[i]++] = nxt[i + 1][j];
  }
  scanf("%d", &m);
  for (int i = 0; i < m; ++i) {
    string t;
    cin >> t;
    int msk = 0;
    for (int j = 0; j < int((t).size()); ++j) msk |= 1 << (t[j] - 'a');
    tt[i] = make_pair(msk, i);
    vv[i] = msk;
  }
  sort(tt, tt + m);
  for (int i = 0; i < n; ++i) {
    if (i && s[i - 1] == s[i]) continue;
    int msk = 1 << (s[i] - 'a');
    int pos = find(msk);
    if (pos != -1) ans[tt[pos].second]++;
    for (int j = 0; j < len[i]; ++j) {
      if (i > 0 && s[nxt[i][j]] == s[i - 1]) break;
      msk |= 1 << (s[nxt[i][j]] - 'a');
      pos = find(msk);
      if (pos != -1) ans[tt[pos].second]++;
    }
  }
  for (int i = 0; i < m; ++i) {
    int pos = find(vv[i]);
    printf("%d\n", ans[tt[pos].second]);
  }
  return 0;
}
