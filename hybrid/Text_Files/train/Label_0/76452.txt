T = int(input())
while T > 0:
	T -= 1
	(h, c, t) = map(int, input().split())
	if 2 * t <= h + c:
		print(2)
		continue
	k = (h - t) // (2 * t - h - c)
	if abs((2 * k + 3) * t - k * h - 2 * h - k * c - c) * (2 * k + 1) < abs((2 * k + 1) * t - k * h - h - k * c) * (2 * k + 3):
		print(2 * k + 3)
	else:
		print(2 * k + 1)
