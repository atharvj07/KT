(b, c) = input().split()
c = int(c)
stroka = input()
n = 26
spisok = [0] * n
for i in stroka:
	spisok[ord(i) - 97] += 1
stroka = stroka[::-1]
for i in range(0, len(spisok)):
	if c > 0:
		if int(spisok[i]) >= c:
			spisok[i] = int(spisok[i]) - c
			c = 0
		elif (int(spisok[i]) < c) & (int(spisok[i]) > 0):
			c = c - spisok[i]
			spisok[i] = 0
stroka_f = ''
for i in stroka:
	if spisok[ord(i) - 97] > 0:
		stroka_f += i
		spisok[ord(i) - 97] -= 1
print(stroka_f[::-1])
