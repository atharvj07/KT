import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        boolean beforePlus = true, beforeEqualAfterPlus = false, afterEqual = false;
        String str = sc.nextLine();
        int length = str.length();
        int count1 = 0, count2 = 0, count3 = 0;
        for(int i=0; i<length; i++) {
            if(beforePlus == true) {
                if(str.charAt(i)=='+') {
                    beforeEqualAfterPlus = true;
                    beforePlus = false;
                    continue;
                } else {
                    count1++;
                }
            } else if (beforeEqualAfterPlus == true) {
                if(str.charAt(i) == '=') {
                    afterEqual = true;
                    beforeEqualAfterPlus = false;
                    continue;
                } else {
                    count2++;
                }
            } else if (afterEqual == true) {
                count3++;
            }
        }
        if(count1+count2 == count3) {
            System.out.println(str);
        } else if (count1+count2 == count3-2) {
            String sub = str.substring(0, length-1);
            System.out.println("|"+sub);
        } else if (count1+count2 == count3+2) {
            if(count1>1) {
                String sub = str.substring(1, length);
                System.out.println(sub + "|");
            } else {
                System.out.print("|+|");
                for(int i=1; i<count2-1; i++) {
                    System.out.print("|");
                }
                System.out.print("=");
                for(int i=0; i<count3; i++) {
                    System.out.print("|");
                }
                System.out.print("|");
            }
        } else {
            System.out.print("Impossible");
        }
    }
}
