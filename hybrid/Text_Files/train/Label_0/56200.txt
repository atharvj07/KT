import java.util.Scanner;

public class Main {

    public static int count(int a, int b){
        int moves = 0;
        if(a == 1){
            if(b == 1){
                return 2;
            } else {
                return 1;
            }
        } else if (a < b){
            return 1;
        }
        if(b > 1)
            moves += Math.ceil(Math.log(a+0.1)/Math.log(b));
        else
            moves += 1+Math.ceil(Math.log(a+0.1)/Math.log(b+1));

        for(int i = 1; i < moves; i++){
            if(i+Math.ceil(Math.log(a+0.1)/Math.log(b+i)) < moves){
                moves = i + (int)Math.ceil(Math.log(a+0.1)/Math.log(b+i));
            } else if(i+1 + (int)(Math.ceil(Math.log((a+0.1)/(b+i))/Math.log(b+i))) < moves){
                moves = i+1 + (int)(Math.ceil(Math.log((a+0.1)/(b+i))/Math.log(b+i)));
            }
        }
        return moves;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for(int i = 0; i < n; i++){
            int a = sc.nextInt();
            int b = sc.nextInt();
            System.out.println(count(a,b));
        }
    }
}
