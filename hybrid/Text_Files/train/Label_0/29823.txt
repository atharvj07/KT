#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9 + 10, maxN = 1e5 + 10;
struct TNode {
  TNode *lef, *rig;
  int l, r, val, lazy;
  TNode() {
    l = r = val = lazy = 0;
    lef = rig = nullptr;
  }
  void set(int le, int ri) {
    l = le;
    r = ri;
    val = INF;
    lazy = 0;
    lef = rig = nullptr;
  }
} * root, *base;
int n, k, q, a[maxN];
void Down(TNode *par) {
  par->lef->val = par->lef->lazy = par->lazy;
  par->rig->val = par->rig->lazy = par->lazy;
  par->lazy = 0;
}
int Get(TNode *par, int u, int v);
int GetFromBase(int u, int v) {
  int su = (u + n - 1) / n, sv = (v + n - 1) / n;
  int iu = u % n, iv = v % n;
  if (!iu) iu = n;
  if (!iv) iv = n;
  if (su == sv) return Get(base, iu, iv);
  if (su + 1 == sv) return min(Get(base, iu, n), Get(base, 1, iv));
  return base->val;
}
void Update(TNode *par, int u, int v, int val) {
  int l = par->l, r = par->r;
  if (v < l || r < u) return;
  if (u <= l && r <= v) {
    par->val = val;
    par->lazy = val;
    return;
  }
  int mid = l + r >> 1;
  if (par->lef == nullptr) {
    par->lef = new TNode();
    par->lef->set(l, mid);
    par->lef->val = GetFromBase(l, mid);
  }
  if (par->rig == nullptr) {
    par->rig = new TNode();
    par->rig->set(mid + 1, r);
    par->rig->val = GetFromBase(mid + 1, r);
  }
  if (par->lazy) Down(par);
  Update(par->lef, u, v, val);
  Update(par->rig, u, v, val);
  par->val = min(par->lef->val, par->rig->val);
}
int Get(TNode *par, int u, int v) {
  int l = par->l, r = par->r;
  if (v < l || r < u) return INF;
  if (u <= l && r <= v) return par->val;
  int mid = l + r >> 1;
  if (par->lef == nullptr) {
    par->lef = new TNode();
    par->lef->set(l, mid);
    par->lef->val = GetFromBase(l, mid);
  }
  if (par->rig == nullptr) {
    par->rig = new TNode();
    par->rig->set(mid + 1, r);
    par->rig->val = GetFromBase(mid + 1, r);
  }
  if (par->lazy) Down(par);
  return min(Get(par->lef, u, v), Get(par->rig, u, v));
}
void Build(TNode *par) {
  int l = par->l, r = par->r;
  if (l == r) {
    par->val = a[l];
    return;
  }
  int mid = l + r >> 1;
  par->lef = new TNode();
  par->lef->set(l, mid);
  par->rig = new TNode();
  par->rig->set(mid + 1, r);
  Build(par->lef);
  Build(par->rig);
  par->val = min(par->lef->val, par->rig->val);
}
int main() {
  scanf("%d%d", &n, &k);
  root = new TNode();
  root->set(1, n * k);
  for (int i = 1; i <= n; ++i) scanf("%d", &a[i]);
  base = new TNode();
  base->set(1, n);
  Build(base);
  for (int i = 1; i <= n; ++i) Update(root, i, i, a[i]);
  scanf("%d", &q);
  for (int i = 1; i <= q; ++i) {
    int t, l, r, x;
    scanf("%d%d%d", &t, &l, &r);
    if (t == 1) {
      scanf("%d", &x);
      Update(root, l, r, x);
    } else
      printf("%d\n", Get(root, l, r));
  }
}
