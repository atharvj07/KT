primes = set()
for i in range(2, 20400):
	primes.add(i)
for div in range(2, 20400):
	j = 2 * div
	while j <= 20400:
		if j in primes:
			primes.remove(j)
		j += div

def isprime(i):
	return i in primes
for _ in range(int(input())):
	d = int(input())
	p1 = -1
	for num in range(d + 1, d + 1000):
		if isprime(num):
			p1 = num
			break
	p2 = -1
	for num in range(p1 + d, p1 + d + 1000):
		if isprime(num):
			p2 = num
			break
	print(p1 * p2)
