import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
right = [(i + 1) % n for i in range(n)]
c = []
cnt = [0] * n
for bb in b:
	cnt[bb] += 1
for aa in a:
	tar = (n - aa) % n
	while cnt[tar] == 0:
		if cnt[right[tar]] == 0:
			right[tar] = right[right[tar]]
		tar = right[tar]
	cnt[tar] -= 1
	c.append((aa + tar) % n)
print(*c)
