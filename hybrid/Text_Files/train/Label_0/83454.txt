def solve(a, b, s):
	n = a + b
	m = n // 2
	for i in range(m):
		j = n - i - 1
		x = s[i]
		y = s[j]
		if x == '?' and y == '?':
			continue
		elif x == '0' and y == '0':
			a -= 2
		elif x == '0' and y == '1':
			return -1
		elif x == '0' and y == '?':
			s[j] = '0'
			a -= 2
		elif x == '1' and y == '0':
			return -1
		elif x == '1' and y == '1':
			b -= 2
		elif x == '1' and y == '?':
			s[j] = '1'
			b -= 2
		elif x == '?' and y == '0':
			s[i] = '0'
			a -= 2
		elif x == '?' and y == '1':
			s[i] = '1'
			b -= 2
	if n % 2 == 1:
		if s[m] == '0':
			a -= 1
		if s[m] == '1':
			b -= 1
	for i in range(m):
		j = n - i - 1
		if s[i] == '?':
			if a > 1:
				s[i] = '0'
				s[j] = '0'
				a -= 2
			elif b > 1:
				s[i] = '1'
				s[j] = '1'
				b -= 2
	if n % 2 == 1 and s[m] == '?':
		if a > 0:
			s[m] = '0'
			a -= 1
		elif b > 0:
			s[m] = '1'
			b -= 1
	if a < 0 or b < 0 or '?' in s:
		return -1
	return ''.join(s)
t = int(input())
while t > 0:
	(a, b) = map(int, input().split())
	s = list(input())
	print(solve(a, b, s))
	t -= 1
