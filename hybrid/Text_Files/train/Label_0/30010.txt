#include <bits/stdc++.h>
using namespace std;
int r[110], b[110];
int main() {
  int n;
  cin >> n;
  int rs = 0, bs = 0;
  for (int i = 0; i < n; ++i) {
    cin >> r[i];
    rs += r[i];
  }
  int rw = 0, bw = 0;
  for (int i = 0; i < n; ++i) {
    cin >> b[i];
    bs += b[i];
    if (b[i] == 0 && r[i] == 1) rw++;
    if (b[i] == 1 && r[i] == 0) bw++;
  }
  if (rw == 0)
    cout << -1 << endl;
  else if (rs > bs)
    cout << 1 << endl;
  else {
    int mark = 0;
    do {
      rs++;
      ++mark;
    } while (rs <= bs);
    cout << ceil((double)mark / rw) + 1 << endl;
  }
  return 0;
}
