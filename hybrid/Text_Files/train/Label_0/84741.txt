	(n, k, m) = tuple(map(int, input().split(' ')))
	ml = set(map(int, input().split(' ')))
	havitada = []
	for i in range(1, n + 1):
		if i not in ml:
			havitada.append(i)
	saab = False
	if len(havitada) % (k - 1) != 0:
		print('no')
		continue
	for i in range(k // 2 - 1, len(havitada) - k // 2):
		if havitada[i + 1] - havitada[i] != 1:
			print('yes')

	else:
		print('no')
