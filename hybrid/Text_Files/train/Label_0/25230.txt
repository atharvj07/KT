#include <bits/stdc++.h>
using namespace std;
string s;
int place[4];
char ch[4];
int ans[4];
int main() {
  cin >> s;
  ch[0] = 'R';
  ch[1] = 'B';
  ch[2] = 'Y';
  ch[3] = 'G';
  for (int i = 0; i < s.size(); i++) {
    for (int j = 0; j < 4; j++)
      if (s[i] == ch[j]) place[j] = i;
  }
  for (int i = 0; i < 4; i++) {
    for (int j = place[i] + 4; j < s.size(); j += 4)
      if (s[j] == '!') ans[i]++;
    for (int j = place[i] - 4; j >= 0; j -= 4)
      if (s[j] == '!') ans[i]++;
  }
  cout << ans[0] << " " << ans[1] << " " << ans[2] << " " << ans[3];
  return 0;
}
