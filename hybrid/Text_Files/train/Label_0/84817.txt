(n, q) = map(int, input().split())
arr = list(map(int, input().split()))
n0 = arr.count(0)
n1 = n - n0
for _ in range(q):
	(t, x) = map(int, input().split())
	if t == 1:
		if arr[x - 1] == 1:
			n0 += 1
			n1 -= 1
			arr[x - 1] = 0
		else:
			n0 -= 1
			n1 += 1
			arr[x - 1] = 1
	elif x <= n1:
		print(1)
	else:
		print(0)
