s = list(map(int, input().split(' ')))
n = s[0]
k = s[1]
cnt = 0
r = n * 2
if r % k == 0:
	cnt = cnt + r // k
elif r % k > 0:
	cnt = cnt + r // k + 1
g = n * 5
if g % k == 0:
	cnt = cnt + g // k
elif g % k > 0:
	cnt = cnt + g // k + 1
b = n * 8
if b % k == 0:
	cnt = cnt + b // k
elif b % k > 0:
	cnt = cnt + b // k + 1
print(cnt)
