using namespace ::std;
long long a[10000];
int main() {
  int n;
  cin >> n;
  a[0] = 0;
  a[1] = 1;
  if (n == 0) {
    cout << "0 0 0";
    return 0;
  }
  if (n == 1) {
    cout << "0 0 1";
    return 0;
  }
  if (n == 2) {
    cout << "0 1 1";
    return 0;
  }
  for (int i = 2; i < 45; i++) {
    a[i] = a[i - 1] + a[i - 2];
  }
  for (int i = 1; i < 45; i++)
    for (int j = 1; j < 45; j++)
      for (int k = 1; k < 45; k++) {
        if (a[i] + a[j] + a[k] == n) {
          cout << a[i] << " " << a[j] << " " << a[k];
          return 0;
        }
      }
  cout << "I'm too stupid to solve this problem";
}
