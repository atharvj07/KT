#include <bits/stdc++.h>
#define maxn 100009
using namespace std;
long long L[maxn],R[maxn],sum[maxn];
long long a[maxn];
int n,k;
int main(){
	cin>>n>>k;
	for(int i=1;i<=n;i++){
		cin>>a[i];
		if(a[i]>=0)
			L[i]=L[i-1]+a[i];
		else
			L[i]=L[i-1];
		sum[i]=sum[i-1]+a[i];
	}
	for(int i=n;i>=1;i--){
		R[i]=R[i+1]+max(a[i],0LL);
	}
	long long ans=0;
	for(int i=1;i+k-1<=n;i++){
		ans = max(ans,L[i-1]+R[i+k]);
		ans = max(ans,L[i-1]+R[i+k]+sum[i+k-1]-sum[i-1]);
	}
	cout<<ans<<endl;
	//system("pause");
	return 0;
}
