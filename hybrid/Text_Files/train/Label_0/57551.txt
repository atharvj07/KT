import java.util.*;
import java.io.*;

public class Solution{
     
     public static void main(String []args){
        Scanner sc = new Scanner(System.in);

            int t = sc.nextInt();
            
            
            
            while(t > 0){
                
                long n = sc.nextLong();
                List<long[]> pairs = new ArrayList<>();
            
                long i = 2 ;
                
               while(i * i <= n){
                    long[] pair = new long[2];
                    
                    while(n % i == 0){
                        
                        pair[0] = i;
                        pair[1]++;
                        
                        n /= i;
                        
                        
                    }
                    
                    pairs.add(pair);
                    i++;
                    
                    
                }
                
                if(n > 0){
                    pairs.add(new long[]{n , 1});
                }
                
                Collections.sort(pairs, (a,b)->Long.compare(b[1],a[1]));
                
                long[] k = pairs.get(0);
                List<Long> res = new ArrayList<>();
                System.out.println(k[1]);
                
                for(int j = 0 ; j < k[1] ; j++){
                    res.add(k[0]);
                }
                
                int last = res.size() - 1;
                
                for(int j = 1 ; j < pairs.size() ; j++){
                    
                    long[] curr = pairs.get(j);
                    
                    long val = (long)Math.pow(curr[0] , curr[1]);
                    
                    res.set(last , res.get(last) * val);
                    
                }
                
                for(long j : res){
                    System.out.print(j + " ");
                }
                System.out.println();
                t--;
                
            }
           
      
     }
}
