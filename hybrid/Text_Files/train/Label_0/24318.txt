#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, c, h[1000] = {0};
  cin >> n >> k >> c;
  for (int i(0); i < (c); i++) {
    int d;
    cin >> d;
    h[d] = 1;
  }
  int t = 0, ans = 0;
  for (int i(1); i < (n + 1); i++) {
    ans += h[i];
    if (h[i] == 0)
      t++;
    else
      t = 0;
    if (t == k) {
      ans++;
      t = 0;
    }
  }
  cout << ans;
}
