#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
map<int, int> mx, my;
int n;
set<int> pos[420000];
int mkr[420000];
int arr[420000][2];
vector<int> adj[420000];
int mkd[430000];
pair<int, int> dfs(int x, int dad = -1) {
  mkd[x] = 1;
  pair<int, int> ret = {0, 1};
  for (auto aaa : adj[x]) {
    if (aaa != dad) {
      if (mkd[aaa])
        ret.first = 1;
      else {
        pair<int, int> xxx = dfs(aaa, x);
        ret.first |= xxx.first;
        ret.second += xxx.second;
      }
    }
  }
  return ret;
}
int main() {
  cin >> n;
  int k = 1;
  for (int i = 0; i < n; i++) {
    int x, y;
    cin >> x >> y;
    arr[i][0] = x;
    arr[i][1] = y;
    if (!mx[x]) mx[x] = k++;
    if (!my[y]) my[y] = k++;
    adj[mx[x]].push_back(my[y]);
    adj[my[y]].push_back(mx[x]);
  }
  long long ans = 1;
  for (int i = 1; i < k; i++)
    if (!mkd[i]) {
      pair<int, int> ret = dfs(i);
      long long addd = 1;
      for (int i = 0; i < ret.second; i++) addd = (addd * 2) % mod;
      addd = (addd + mod - 1 + ret.first) % mod;
      ans = (ans * addd) % mod;
    }
  cout << ans << endl;
}
