import java.io.*;
import java.util.*;

public class R1D
{
    public static StringTokenizer st;
    public static void nextLine(BufferedReader br) throws IOException
    {
        st = new StringTokenizer(br.readLine());
    }
    
    public static String next()
    {
        return st.nextToken();
    }
    
    public static int nextInt()
    {
        return Integer.parseInt(st.nextToken());
    }
    
    public static long nextLong()
    {
        return Long.parseLong(st.nextToken());
    }
    
    public static double nextDouble()
    {
        return Double.parseDouble(st.nextToken());
    }
    
    static Line[] nslines, ewlines;
    
    public static void main(String[] args) throws IOException
    {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        nextLine(br);
        int n = nextInt();
        int m = nextInt();
        nslines = new Line[n];
        ewlines = new Line[m];
        
        for (int i = 0; i < n; i++)
        {
            nextLine(br);
            int x = nextInt();
            int y = nextInt();
            int l = nextInt();
            nslines[i] = new Line(x, y, l, i, true);
        }
        for (int i = 0; i < m; i++)
        {
            nextLine(br);
            int x = nextInt();
            int y = nextInt();
            int l = nextInt();
            ewlines[i] = new Line(x, y, l, i, false);
        }
        
        int lb = 0;
        int ub = 1000000000;
        while (lb < ub)
        {
            int mid = (lb + ub + 1) / 2;
            if (solve(mid))
            {
                lb = mid;
            }
            else
            {
                ub = mid - 1;
            }
        }
        System.out.println(lb);
    }
    
    private static boolean solve(int len)
    {
        PriorityQueue<Event> pq = new PriorityQueue<Event>();
        for (int i = 0; i < nslines.length; i++)
        {
            if (nslines[i].l < len * 2)
            {
                continue;
            }
            pq.offer(new Event(nslines[i].y + len, i, 0, true));
            pq.offer(new Event(nslines[i].y + nslines[i].l - len, i, 2, true));
        }
        for (int i = 0; i < ewlines.length; i++)
        {
            if (ewlines[i].l < len * 2)
            {
                continue;
            }
            pq.offer(new Event(ewlines[i].y, i, 1, false));
        }
        
        TreeSet<Long> tree = new TreeSet<Long>();
        while (!pq.isEmpty())
        {
            Event ev = pq.poll();
            if (ev.type == 0)
            {
                tree.add(nslines[ev.index].x);
            }
            else if (ev.type == 2)
            {
                tree.remove(nslines[ev.index].x);
            }
            else
            {
                Line hl = ewlines[ev.index];
                Long x = tree.ceiling(hl.x + len);
                if (x != null && x <= hl.x + hl.l - len)
                {
                    return true;
                }
            }
        }
        return false;
    }
    
    private static class Line implements Comparable<Line>
    {
        long x, y, l;
        int index;
        boolean ns;
        
        public Line(long x, long y, long l, int index, boolean ns)
        {
            this.x = x;
            this.y = y;
            this.l = l;
            this.index = index;
            this.ns = ns;
        }

        @Override
        public int compareTo(Line o)
        {
            return (int)(x - o.x);
        }
        
    }
    
    private static class Event implements Comparable<Event>
    {
        int index, type;
        boolean ns;
        long y;
        
        public Event(long y, int index, int type, boolean ns)
        {
            this.y = y;
            this.index = index;
            this.type = type;
            this.ns = ns;
        }
        
        @Override
        public int compareTo(Event o)
        {
            if (y == o.y)
            {
                return type - o.type;
            }
            return (int)(y - o.y);
        }
        
    }
}
