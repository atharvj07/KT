import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        FastScanner in = new FastScanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        new Main().run(in, out);
        out.close();
    }

    public static long mod = 17352642619633L;

    // dinic

    int N, M;
    class FlowEdge {
        int u;
        int v;
        int cap, flow;
        FlowEdge(int u, int v, int cap) {
            this.u = u;
            this.v = v;
            this.cap = cap;
            this.flow = 0;
        }
    }

    List<FlowEdge> edges = new ArrayList<>();
    List<Integer>[] adj;
    int[] levels;
    int SRC, SNK;

    int[] a;
    int[] b;
    int sz = 0;

    void NO() {
        System.out.println("NO");
        System.exit(0);
    }

    void addEdge(int u, int v, int cap) {
        edges.add(new FlowEdge(u, v, cap));
        edges.add(new FlowEdge(v, u, 0));
        adj[u].add(sz++);
        adj[v].add(sz++);
    }


    void run(FastScanner in, PrintWriter out) {
        N = in.nextInt();
        M = in.nextInt();
        adj = new List[2*N+4];
        for (int i = 0; i < adj.length; i++) adj[i] = new ArrayList<>();

        levels = new int[2*N+4];
        ptr = new int[2*N+2];
        a = new int[N+2];
        b = new int[N+2];

        SRC = 0;
        SNK = 2*N+1;

        int asum = 0;
        int bsum = 0;
        for (int i = 1; i <= N; i++) {
            a[i] = in.nextInt();
            asum += a[i];
        }
        for (int i = 1; i <= N; i++) {
            b[i] = in.nextInt();
            bsum += b[i];
        }

        if (asum != bsum) NO();

        for (int i = 0; i < M; i++) {
            int u = in.nextInt();
            int v = in.nextInt();

            addEdge(u, N+v, oo);
            addEdge(v, N+u, oo);
        }

        for (int i = 1; i <= N; i++) {
            addEdge(i, i+N, oo);
            addEdge(SRC, i, a[i]);
            addEdge(N+i, SNK, b[i]);
        }


        int maxflow = 0;
        while (true) {
            getLevels();
            if (levels[SNK] == -1) break;

            Arrays.fill(ptr, 0);
            int flow = 0;
            while ((flow = dfs(SRC, oo)) != 0) {
                maxflow += flow;
            }
        }

        if (maxflow != asum) NO();

        int[][] ret = new int[N][N];

        for (FlowEdge fe : edges) {
            if (fe.u == SRC || fe.v == SRC) continue;
            if (fe.u == SNK || fe.v == SNK) continue;
            if (fe.flow < 0) continue;

            if (fe.u <= N && fe.v > N) {
                ret[fe.u-1][fe.v-1-N] = fe.flow;
            }

        }

        out.println("YES");
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                out.print(ret[i][j] + " ");
            }
            out.println();
        }

    }

    int oo = 987654321;

    int[] ptr;
    void getLevels() {
        Arrays.fill(levels, -1);
        levels[SRC] = 0;
        LinkedList<Integer> ll = new LinkedList<>();
        ll.offer(SRC);
        while (!ll.isEmpty()) {
            int u = ll.pollFirst();
            for (int k : adj[u]) {
                FlowEdge fe = edges.get(k);
                if (levels[fe.v] != -1 || fe.cap - fe.flow <= 0) continue;
                levels[fe.v] = levels[u]+1;
                ll.offerLast(fe.v);
            }
        }
    }

    int dfs(int u, int currflow) {
        if (u == SNK) return currflow;
        for (int p = ptr[u]; p < adj[u].size(); p = ++ptr[u]) {
            int k = adj[u].get(p);

            FlowEdge fe = edges.get(k);
            if (levels[fe.v] != levels[u]+1 || fe.cap - fe.flow <= 0) continue;
            int ret = dfs(fe.v, Math.min(currflow, fe.cap - fe.flow));
            if (ret != 0) {
                fe.flow += ret;
                edges.get(k^1).flow -= ret;
                return ret;
            }
        }
        return 0;
    }



    static class FastScanner {
        BufferedReader br;
        StringTokenizer st;

        public FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
            st = null;
        }

        String next() {
            while (st == null || !st.hasMoreElements()) {
                try {
                    st = new StringTokenizer(br.readLine());
                } catch (IOException e) {
                    e.printStackTrace();
                }
            }
            return st.nextToken();
        }

        int nextInt() {
            return Integer.parseInt(next());
        }

        long nextLong() {
            return Long.parseLong(next());
        }
    }
}
