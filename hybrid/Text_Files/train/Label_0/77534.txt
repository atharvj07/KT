m = int(input())
ans = []
for i in range(m):
	s = list(input())
	s.remove('(')
	s.remove(')')
	a = ''
	k = -1
	for i in range(len(s)):
		if s[i] != '+':
			a += s[i]
		else:
			k = i
			break
	b = ''
	l = 0
	for i in range(k + 1, len(s)):
		if s[i] != '/':
			b += s[i]
		else:
			l = i
			break
	c = ''
	for i in range(l + 1, len(s)):
		c += s[i]
	a = int(a)
	b = int(b)
	c = int(c)
	ans.append((a + b) / c)
b = list(set(ans))
dic = {}
for i in range(len(b)):
	dic[b[i]] = 0
for i in range(len(ans)):
	dic[ans[i]] += 1
for i in range(len(ans)):
	print(dic[ans[i]], end=' ')
