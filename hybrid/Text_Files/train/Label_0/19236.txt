#include<bits/stdc++.h>

#define N 200010
#define mod 1000000007

using namespace std;

int n,x,y,ans=1,tot,min1,min2,c[N],w[N],minc[N],numc[N];
int fac[N],inv[N];

void getmin()
{
	for(int i=1;i<=n;i++)
	{
		if(minc[i]<minc[min1])
		{
			min2=min1;
			min1=i;
		}
		else if(minc[i]<minc[min2])
			min2=i;
	}
}

int poww(int a,int b)
{
	int ans=1;
	while(b)
	{
		if(b&1)
			ans=(1ll*ans*a)%mod;
		a=(1ll*a*a)%mod;
		b>>=1;
	}
	return ans;
}

void preC()
{
	fac[0]=1;
    for(int i=1;i<=n;i++)
        fac[i]=1ll*i*fac[i-1]%mod;
    inv[n]=poww(fac[n],mod-2);
    for(int i=n-1;i>=0;i--)
        inv[i]=1ll*inv[i+1]*(i+1)%mod;
}

int C(int n,int m)
{
    return ((1ll*fac[m]*inv[m-n])%mod*inv[n])%mod;
}

int main()
{
	memset(minc,127,sizeof(minc));
	scanf("%d%d%d",&n,&x,&y);
	for(int i=1;i<=n;i++)
	{
		scanf("%d%d",&c[i],&w[i]);
		minc[c[i]]=min(minc[c[i]],w[i]);
	}
	getmin();
	for(int i=1;i<=n;i++)
	{
		if(w[i]+minc[c[i]]<=x)
			w[i]=minc[c[i]];
		if((c[i]==min1&&minc[min2]+w[i]<=y)||(c[i]!=min1&&minc[min1]+w[i]<=y))
		{
			numc[c[i]]++;
			tot++;
		}
	}
	preC();
	for(int i=1;i<=n;i++)
	{
		if(numc[i])
		{
			ans=(1ll*ans*C(numc[i],tot))%mod;
			tot-=numc[i];
		}
	}
	printf("%d\n",ans);
	return 0;
}
