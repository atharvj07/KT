import heapq
import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	l = list(map(int, input().split()))
	if n <= 2:
		print(0)
		continue
	ans = 0
	for i in range(1, n - 1):
		x = l[i]
		if x > l[i + 1] and x > l[i - 1] or (x < l[i + 1] and x < l[i - 1]):
			ans += 1
			continue
	cur = ans
	for i in range(n):
		if i == 0:
			x = l[i + 1]
			v1 = x
			v2 = x - 1
			v3 = x + 1
			o = [v1, v2, v3]
			c = 0
			if l[i + 1] > l[i] and l[i + 1] > l[i + 2] or (l[i + 1] < l[i] and l[i + 1] < l[i + 2]):
				c += 1
			if len(l) >= 3:
				for j in o:
					r1 = j
					r2 = l[i + 1]
					r3 = l[i + 2]
					e = 0
					if r2 > r1 and r2 > r3 or (r2 < r1 and r2 < r3):
						e += 1
					if e < c:
						d = c - e
						ans = min(ans, cur - d)
			continue
		if i == n - 1:
			x = l[i - 1]
			v1 = x
			v2 = x - 1
			v3 = x + 1
			o = [v1, v2, v3]
			c = 0
			if l[i - 1] > l[i] and l[i - 1] > l[i - 2] or (l[i - 1] < l[i] and l[i - 1] < l[i - 2]):
				c += 1
			if len(l) >= 3:
				for j in o:
					r1 = j
					r2 = l[i - 1]
					r3 = l[i - 2]
					e = 0
					if r2 > r1 and r2 > r3 or (r2 < r1 and r2 < r3):
						e += 1
					if e < c:
						d = c - e
						ans = min(ans, cur - d)
			continue
		x1 = l[i + 1]
		x2 = l[i - 1]
		v1 = x1
		v2 = x1 - 1
		v3 = x1 + 1
		v4 = x2
		v5 = x2 - 1
		v6 = x2 + 1
		o = [v1, v2, v3, v4, v5, v6]
		c = 0
		for j in range(i - 1, i + 2):
			if j - 1 > -1 and j + 1 < n:
				if l[j - 1] > l[j] and l[j + 1] > l[j] or (l[j - 1] < l[j] and l[j + 1] < l[j]):
					c += 1
					continue
		temp = l[i]
		for j in o:
			e = 0
			l[i] = j
			for k in range(i - 1, i + 2):
				if k - 1 > -1 and k + 1 < n:
					if l[k - 1] > l[k] and l[k + 1] > l[k] or (l[k - 1] < l[k] and l[k + 1] < l[k]):
						e += 1
						continue
			if c < e:
				continue
			d = c - e
			ans = min(ans, cur - d)
		l[i] = temp
	print(ans)
