import sys
import math
files = False
if files:
	sys.stdin = open('slalom.in')
sz = int(input())
(x, y) = [int(i) for i in input().split()]
tow = [[1, 0], [-1, 0], [0, -1], [0, 1]]
bis = [[1, 1], [-1, 1], [1, -1], [-1, -1]]
towd = [[1, 0, 'N'], [-1, 0, 'N'], [0, -1, 'N'], [0, 1, 'N']]
bisd = [[1, 1, 'N'], [-1, 1, 'N'], [1, -1, 'N'], [-1, -1, 'N']]
for n1 in range(sz):
	(l, xt, yt) = [i for i in input().split()]
	xt = int(xt) - x
	yt = int(yt) - y
	if xt == 0 or yt == 0:
		for (n, i1) in enumerate(tow):
			if i1[0] == 0 and xt != 0:
				continue
			if i1[1] == 0 and yt != 0:
				continue
			if xt * i1[0] >= 0 and yt * i1[1] >= 0:
				if towd[n][2] == 'N':
					towd[n][0] = xt
					towd[n][1] = yt
					towd[n][2] = l
					continue
				if abs(towd[n][0]) + abs(towd[n][1]) > abs(xt) + abs(yt):
					towd[n][0] = xt
					towd[n][1] = yt
					towd[n][2] = l
					continue
		continue
	if abs(xt) != abs(yt):
		continue
	for (n, i1) in enumerate(bis):
		if xt * i1[0] >= 0 and yt * i1[1] >= 0:
			if bisd[n][2] == 'N':
				bisd[n][0] = xt
				bisd[n][1] = yt
				bisd[n][2] = l
				break
			if abs(bisd[n][0]) > abs(xt):
				bisd[n][0] = xt
				bisd[n][1] = yt
				bisd[n][2] = l
				break
hit = 'NO'
for (n1, i1) in enumerate(bisd):
	if i1[2] == 'B' or i1[2] == 'Q':
		hit = 'YES'
for (n1, i1) in enumerate(towd):
	if i1[2] == 'R' or i1[2] == 'Q':
		hit = 'YES'
print(hit)
sys.stdout.close()
