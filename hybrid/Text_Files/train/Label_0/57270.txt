import java.io.BufferedReader;
import java.io.InputStreamReader;

public class C {

	private final int MAX = 1000010;

	BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	int a, b, k;
	boolean[] isComposite = new boolean[MAX];
	int[] cntPrimes = new int[MAX];

	private void readInput() {
		try {
			String[] tokens = reader.readLine().split(" ");
			a = Integer.parseInt(tokens[0]);
			b = Integer.parseInt(tokens[1]);
			k = Integer.parseInt(tokens[2]);
		} catch (Exception e) {
			e.printStackTrace();
		}
	}

	private void crossAll(int number) {
		int temp = number + number;
		while (temp <= b) {
			isComposite[temp] = true;
			temp += number;
		}
	}

	private void generatePrimes() {
		isComposite[1] = true;
		crossAll(2);
		crossAll(3);

		int possiblePrime = 5;
		while (possiblePrime <= b) {
			if (!isComposite[possiblePrime]) {
				crossAll(possiblePrime);
			}

			if (possiblePrime + 2 <= b && !isComposite[possiblePrime + 2]) {
				crossAll(possiblePrime + 2);
			}

			possiblePrime += 6;
		}

		cntPrimes[0] = 0;
		cntPrimes[1] = 0;
		cntPrimes[2] = 1;
		for (int i = 3; i <= b; i++) {
			cntPrimes[i] += cntPrimes[i - 1] + (isComposite[i] ? 0 : 1);
		}
	}
	
	private boolean possible(int l) {
		for (int start = a; start <= b - l + 1; start++) {
			int end = start + l - 1;
			if (cntPrimes[end] - cntPrimes[start - 1] < k) {
				return false;
			}
		}
		
		return true;
	}
	
	private void binarySearch() {
		int left = 0, right = b - a + 2;
		while(left + 1 < right) {
			int mid = left + right; mid = mid >> 1;
			if (possible(mid)) {
				right = mid;
			} else {
				left = mid;
			}
		}
		
		if (right < b - a + 2 && right >= 1) {
			System.out.println(right);
		} else {
			System.out.println("-1");
		}
	}
	
	private void solve() {
		readInput();
		generatePrimes();
	
		binarySearch();
		
	}

	public static void main(String[] args) {
		new C().solve();
	}
}
