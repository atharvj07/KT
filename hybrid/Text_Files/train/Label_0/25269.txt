#include <bits/stdc++.h>
using namespace std;
double pow1(double a, int b) {
  double r = 1, base = a;
  while (b != 0) {
    if (b & 1) r *= base;
    base *= base;
    b >>= 1;
  }
  return r;
}
int main() {
  double n;
  int t;
  while (cin >> n >> t) {
    printf("%.30lf\n", n * pow1(1.000000011, t));
  }
  return 0;
}
