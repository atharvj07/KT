(k, l, r, t, x, y) = map(int, input().split())
if x == y:
	if k - x >= l or k + y <= r:
		print('Yes')
	else:
		print('No')
elif x > y:
	if k + y <= r:
		k = k + y
	k = k - (t - 1) * (x - y)
	k = k - x
	if k >= l:
		print('Yes')
	else:
		print('No')
elif x + y <= r - l:
	print('Yes')
else:
	dp = [0] * x
	z = 0
	while z < t:
		if k + y <= r:
			k = k + y
		p = (k - l) // x
		z = z + p
		k = k - x * p
		if p == 0:
			break
		if dp[k % x] == 1:
			z = t
		else:
			dp[k % x] = 1
	if z >= t:
		print('Yes')
	else:
		print('No')
