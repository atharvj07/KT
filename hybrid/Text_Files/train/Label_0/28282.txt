#include<bits/stdc++.h>
using namespace std;
int main()
{
	int ans[100000+5],turn[2];memset(ans,0,sizeof(ans));
	string s;
	cin>>s;
	for(int i=0;i<s.size()-1;i++)
	{
		if(s[i]=='R'&&s[i+1]=='L')
		{
			memset(turn,0,sizeof(turn));
			for(int j=i;j>=0;j--){
				if(s[j]=='R') turn[(i-j)%2]++;
			    else break;
			}
			
			for(int j=i+1;j<s.size();j++)
			{
				if(s[j]=='L')turn[(j-i)%2]++;
				else break;
			}
			ans[i]=turn[0];
			ans[i+1]=turn[1];
			
		}
	}
	for(int i=0;i<s.size();i++)cout<<ans[i]<<' ';
	return 0;
}
