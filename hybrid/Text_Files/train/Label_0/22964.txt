
#include<cstdio>
#include<iostream>
#include<algorithm>
#include<string>
#include<queue>
#include<vector>
#include<numeric>

using namespace std;

typedef long long ll;

int main()
{
	while (1) {
		string s;
		cin >> s;
		if (s == "0000")break;
		else if ((s[0] == s[1])&&(s[1]==s[2])&&(s[2]==s[3])) cout << "NA" << endl;
		else if (s == "6174")cout << 0 << endl;
		else {
			ll cnt = 0;
			while (1) {
				int d[4];
				d[0] = s[0]-'0', d[1] = s[1]-'0', d[2] = s[2]-'0', d[3] = s[3]-'0';
				sort(d,d+4);
				int S = d[0] * 1000 + d[1] * 100 + d[2] * 10 + d[3];
				reverse(d,d+4);
				int L = d[0] * 1000 + d[1] * 100 + d[2] * 10 + d[3];
				cnt++;
				int res = L - S;
				if (res == 6174)break;
				s = to_string(res);
				while (s.size() < 4) {
					s = '0' + s;
				}
			}
			cout << cnt << endl;
		}
	}
	return 0;
}
