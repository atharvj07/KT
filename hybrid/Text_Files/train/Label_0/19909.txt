#include <bits/stdc++.h>

using namespace std;

int main()
{
	int n;

	while(cin >> n && n)
	{
		int name,m[4],s[4];
		int ans[3]={};
		int max1=0,max2=0,min1=1000000,min2=1000000;
		int list[100000]={};
		int namelist[100000]={};
		int count=0;

		while(n--)
		{
			cin >> name;

			for(int i=0;i<4;i++)
			{
				cin >>m[i]>>s[i];
				list[count]+=m[i]*60+s[i];
			}

			namelist[count]=name;

			if(min1>=list[count])
			{
				min1=min(list[count],min1);
				ans[0]=name;
			}
			max1=max(max1,list[count]);

			count++;
		}

		for(int i=0;i<count;i++)
		{
			if( list[i]>=max2  && max1!=list[i])
			{
				max2=list[i];
				ans[2]=namelist[i];
			}
			if( list[i]<=min2  && min1!=list[i])
			{
				min2=list[i];
				ans[1]=namelist[i];
			}
		}
		for(int i=0;i<3;i++)cout << ans[i] <<endl;
	}

    return 0;
}
