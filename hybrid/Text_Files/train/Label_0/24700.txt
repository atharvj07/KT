#include <bits/stdc++.h>

using namespace std;

#define rep(i,n) for(int i = 0; i < n; i++)
#define FOR(i,a,n) for(int i = a; i < n; i++)
typedef long long ll;




/*
// s1の(i-1)文字目までの文字列に1文字追加する場合
dp[i][j] = dp[i-1][j] + 1;
//s1のj文字目を削除する場合
dp[i][j] = dp[i][j-1] + 1;
// s1のi文字目とs2のj文字目が同じ文字になるように置換する場合
dp[i][j] = dp[i-1][j-1] + cost;
// -> i文字目とj文字目が等しければコストはかからない
cost = (s[i-1] == t[j-1]) ? 0 : 1;
*/
// cost[0] : 追加
// coat[1] : 削除
// cost[2] : 置き換え
// cost[3] : 先頭から末尾へ

string s,t;

ll cost[4];
const ll INF = 1e16;
void solve(){

  ll n = s.size() ,m = t.size();
  ll dp[s.size()+1][t.size()+1];

  ll ans = INF;
  ll c;
  rep(k,n){ // k回 末尾移動操作をした
    /* dpテーブル初期化 */
    rep(i,n+1) rep(j,m+1) dp[i][j] = INF;
    dp[0][0] = 0;

    /* dpテーブル埋め(貰うDP) */
    rep(i, n+1) rep(j, m+1) {
      // 追加する場合
      if(j > 0) dp[i][j] = min(dp[i][j], dp[i][j-1] + cost[0]);

      // 削除する場合
      if(i > 0){
        c = (n-k < i) ? cost[3] : 0;
        dp[i][j] = min(dp[i][j], dp[i-1][j] + cost[1] - c);
      }

      // 置き換える場合
      if(i > 0 && j > 0){
        c = (s[i-1] == t[j-1]) ? 0 : cost[2];
        dp[i][j] = min(dp[i][j], dp[i-1][j-1] + c);
      }

    }

    ans = min(ans,dp[n][m] + cost[3] * k);
    // スワップ回数
    s += s[0];
    s.erase(s.begin());
  }
  cout << ans << endl;
}

int main(){

  while(true){
    cin >> s;
    if(s == "#") break;
    cin >> t;
    rep(i,4) cin >> cost[i];
    solve();
    s.clear();
    t.clear();
  }
  return 0;
}
