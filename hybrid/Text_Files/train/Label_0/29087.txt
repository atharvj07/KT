#include <iostream>
#include <stdio.h>
#include <string.h>
#include <algorithm>
using namespace std;

const int M=3005;

int n,t;
struct Node
{
	int a,b;
	void read()
	{
		scanf("%d%d",&a,&b);
	}
	bool operator<(const Node &B) const
	{
		return a<B.a;
	}
}node[M];
int dp[M];

int main()
{
	scanf("%d%d",&n,&t);
	for(int i=0;i<n;i++)
		node[i].read();
	sort(node,node+n);
	for(int i=0;i<n;i++)
	{
		for(int j=t-1;j>=max(t-node[i].a,0);j--)
			dp[t]=max(dp[t],dp[j]+node[i].b);
		for(int j=t-1;j>=node[i].a;j--)
			dp[j]=max(dp[j],dp[j-node[i].a]+node[i].b);
	}
	printf("%d\n",dp[t]);
	return 0;
}
