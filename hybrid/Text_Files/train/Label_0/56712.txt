import java.util.Scanner;

public class CF237A
{
	public static void main(String[] args)
	{
		Scanner in = new Scanner(System.in);
		int n = in.nextInt(), ret = 0, cnt = 0, last = -1;
		for (int i = 0; i < n; ++i)
		{
			int h = in.nextInt(), m = in.nextInt(), t = h*60+m;
			if (t == last) cnt++;
			else cnt = 1;
			last = t;
			ret = Math.max(ret, cnt);
		}
		System.out.println(ret);
	}
}
