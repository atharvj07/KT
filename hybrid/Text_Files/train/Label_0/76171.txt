def Prefixes_and_Suffixes():
	size_str = int(input())
	str_arr = []
	for i in range(2 * size_str - 2):
		str_arr.append(input())
	first_found = False
	for i in range(len(str_arr)):
		if len(str_arr[i]) == size_str - 1 and (not first_found):
			first_found = True
			st1 = str_arr[i]
		elif len(str_arr[i]) == size_str - 1:
			st2 = str_arr[i]
	if st1[1:] == st2[:len(st2) - 1]:
		orginal_String = st1 + st2[-1]
		output = ''
		done = [False for i in range(size_str - 1)]
		for i in range(len(str_arr)):
			if str_arr[i][0] == orginal_String[0]:
				j = 0
				is_prefix = True
				while j < len(str_arr[i]):
					if str_arr[i][j] != orginal_String[j]:
						output += 'S'
						is_prefix = False
						break
					j += 1
				if is_prefix and (not done[len(str_arr[i]) - 1]):
					output += 'P'
					done[len(str_arr[i]) - 1] = True
				elif is_prefix and done[len(str_arr[i]) - 1]:
					output += 'S'
			else:
				output += 'S'
		if output.count('P') == output.count('S'):
			return output
		else:
			orginal_String = st2 + st1[-1]
			output = ''
			done = [False for i in range(size_str - 1)]
			for i in range(len(str_arr)):
				if str_arr[i][0] == orginal_String[0]:
					j = 0
					is_prefix = True
					while j < len(str_arr[i]):
						if str_arr[i][j] != orginal_String[j]:
							output += 'S'
							is_prefix = False
							break
						j += 1
					if is_prefix and (not done[len(str_arr[i]) - 1]):
						output += 'P'
						done[len(str_arr[i]) - 1] = True
					elif is_prefix and done[len(str_arr[i]) - 1]:
						output += 'S'
				else:
					output += 'S'
			return output
	else:
		orginal_String = st2 + st1[-1]
		output = ''
		done = [False for i in range(size_str - 1)]
		for i in range(len(str_arr)):
			if str_arr[i][0] == orginal_String[0]:
				j = 0
				is_prefix = True
				while j < len(str_arr[i]):
					if str_arr[i][j] != orginal_String[j]:
						output += 'S'
						is_prefix = False
						break
					j += 1
				if is_prefix and (not done[len(str_arr[i]) - 1]):
					output += 'P'
					done[len(str_arr[i]) - 1] = True
				elif is_prefix and done[len(str_arr[i]) - 1]:
					output += 'S'
			else:
				output += 'S'
		return output
remained_test_cases = 1
while remained_test_cases > 0:
	print(Prefixes_and_Suffixes())
