#include <bits/stdc++.h>
using namespace std;
inline int abs(int a) { return (a > 0 ? a : a * -1); }
int main() {
  int n, arr[205];
  int pc = 0, nc = 0, sum = 0;
  int mini = 1005;
  cin >> n;
  for (int i = 0; i < 2 * n - 1; i++) {
    cin >> arr[i];
    if (arr[i] < 0) nc++;
    if (arr[i] >= 0) pc++;
    mini = min(mini, abs(arr[i]));
    sum += abs(arr[i]);
  }
  if (nc % 2 == 0)
    cout << sum << "\n";
  else if ((n + nc) % 2 == 0)
    cout << sum << "\n";
  else
    cout << sum - 2 * mini;
  return 0;
}
