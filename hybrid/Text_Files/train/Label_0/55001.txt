import java.util.*;
public class Main {
    public static void main(String args[]) {
        Scanner scn = new Scanner(System.in);
        int n = scn.nextInt();
        int b = scn.nextInt();
        int a = scn.nextInt();
        int in = a;
        int count =0;
        for(int i=1;i<=n;++i){

            int temp = scn.nextInt();
            if(temp == 0){
                if(a-1>=0){
                   a--;
                   count++;
                }else if(b-1>=0){
                    b--;
                    count++;
                }else{
                    break;
                }
            }else{
                if(b-1>=0||in==0){
                  if(a+1<=in){
                   b--;
                   a++;
                  }else{
                      a--;
                  }
                   count++;
                }else if(a-1>=0){
                    a--;
                    count++;
                }else{
                    break;
                }
            }
            // System.out.println(b+" "+a);
            }
            System.out.println(count);

        
    }
}
