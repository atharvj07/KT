(a, b, n, x) = input().split(' ')
(a, b, n, x) = (int(a), int(b), int(n), int(x))
if a == 1:
	print((x + n * b) % (10 ** 9 + 7))
else:
	print((pow(a, n, 10 ** 9 + 7) * x % (10 ** 9 + 7) + (pow(a, n, 10 ** 9 + 7) - 1) * b % (10 ** 9 + 7) * pow(a - 1, 10 ** 9 + 5, 10 ** 9 + 7) % (10 ** 9 + 7)) % (10 ** 9 + 7))
