public class Main {

    static long  lastSq = 0;

    public static void main(String[] args) {
        //BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        //String line = null;

        Scanner scanner = new Scanner(System.in);
        long x = scanner.nextLong();


        List<String> sim = new ArrayList<String>();
        List<String> right = new ArrayList<String>();
        for (long m = 1;; ++ m){
            long pro = ((1 + m) * m) / 2;
            long chis = x + (m * (pro - m)) - (sqSum(m - 1));
            if (chis / pro < m){
                //System.out.println(m);
                break;
            }
            if (chis  % pro == 0){
                if (chis/ pro != m){
                    sim.add(chis / pro + " " + m);
                }
                right.add(m + " " + chis / pro);
            }
        }

        System.out.println(right.size() + sim.size());
        for (String r: right){
            System.out.println(r);
        }

        for (int i = sim.size() - 1; i >= 0; -- i){
            System.out.println(sim.get(i));
        }
    }

    public static long sqSum(long c){
        lastSq = lastSq + (c*c);
        return lastSq;
    }
}
