#include<bits/stdc++.h>

#define REP(i,s,n) for(int i=s;i<n;i++)
#define rep(i,n) REP(i,0,n)
#define MAX_N 10010

using namespace std;

typedef long long ll;
ll N,K[MAX_N];

int main(){

  cin >> N;
  rep(i,N)cin >> K[i];
  sort(K,K+N);
  ll sum = 0;
  rep(i,N){
    if( K[i] > sum+1LL ){
      cout << sum+1LL << endl;
      return 0;
    }
    sum += K[i];
  }
  cout << sum+1LL << endl;
  return 0;
}
