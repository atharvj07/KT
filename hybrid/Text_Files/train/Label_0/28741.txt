#include <bits/stdc++.h>
using namespace std;
int x, y, cnt = 0, j, p1, p2, ff[15005], tot = 0, seq[15005], dfn[15005],
          nw = 0, n, i, vis[15005], siz[15005], dp[2][15005];
vector<int> s[15005];
long long anm = 0, ans[2][15005], fnl = 0;
void dfs1(int p, int f) {
  int i, v, res;
  ff[p] = f;
  dfn[p] = ++cnt;
  for (i = 0; i < s[p].size(); i++) {
    v = s[p][i];
    if (v == f) continue;
    if (dfn[v] > dfn[p]) continue;
    if (dfn[v] && dfn[v] < dfn[p]) {
      vis[v] = 1;
      seq[++tot] = v;
      res = p;
      while (res != v) {
        vis[res] = 1;
        seq[++tot] = res;
        res = ff[res];
      }
      continue;
    }
    dfs1(v, p);
  }
}
void dfs(int p, int f) {
  int i, v, ok = 0, sum = 0, mx = 0;
  siz[p] = 1;
  for (i = 0; i < s[p].size(); i++) {
    v = s[p][i];
    if (v == f || vis[v]) continue;
    dfs(v, p);
    siz[p] += siz[v];
    sum += dp[0][v];
    mx = max(mx, dp[1][v] - dp[0][v] + n - siz[v]);
    ok = 1;
  }
  dp[0][p] = sum + siz[p];
  if (ok == 1)
    dp[1][p] = mx + sum;
  else
    dp[1][p] = n;
}
int main() {
  scanf("%d", &n);
  for (i = 1; i <= n; i++) {
    scanf("%d%d", &x, &y);
    x++;
    y++;
    s[x].push_back(y);
    s[y].push_back(x);
  }
  dfs1(1, 0);
  for (i = 1; i <= tot; i++) dfs(seq[i], 0), anm += dp[0][seq[i]];
  anm += 1LL * (tot - 2) * n;
  nw = 0;
  for (i = 2; i < tot; i++) {
    nw = !nw;
    for (j = 1; j <= tot; j++) {
      p1 = (j + 1 <= tot) ? j + 1 : 1;
      p2 = (j + i - 1 <= tot) ? j + i - 1 : j + i - 1 - tot;
      ans[nw][j] = min(1LL * (i - 1) * siz[seq[j]] + ans[!nw][p1],
                       1LL * (i - 1) * siz[seq[p2]] + ans[!nw][j]);
    }
  }
  for (i = 1; i <= tot; i++)
    fnl = max(fnl, anm - dp[0][seq[i]] + dp[1][seq[i]] -
                       ans[nw][i + 1 > tot ? 1 : i + 1] -
                       1LL * siz[seq[i]] * (tot - 2));
  printf("%lld\n", fnl);
  return 0;
}
