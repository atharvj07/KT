  std::ios::sync_with_stdio(false);
  std::cin.tie(0);
  ;
  std::string s;
  MACRO_VAR_Scan(s);
  ;
  ll t = s.size() / 2;
  std::string l, r;
  ll i = 0, j = s.size() - 1;
  while (i <= j) {
    if (i == j) {
      l += s[i];
      break;
    }
    if (s[i] == s[j]) {
      l += s[i];
      r += s[j];
      ++i;
      --j;
      continue;
    }
    if (i + 1 < j && s[i] == s[j - 1]) {
      l += s[i];
      r += s[j - 1];
      ++i;
      j -= 2;
      continue;
    }
    ++i;
  }
  std::reverse((r).begin(), (r).end());
  std::cout << (l);
  std::cout << (r);
  std::cout << "\n";
  ;
  return 0;
}
