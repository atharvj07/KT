n = int(input())
a = list(map(int, input().split()))
l = [0] * (n + 1)
flag = 0
b = []
c = []
import heapq
for i in range(n):
	if a[i] > i + 1:
		flag = 1
		break
	l[a[i]] += 1
for i in range(n + 1):
	if l[i] == 0:
		c.append(i)
heapq.heapify(c)
if flag == 1:
	print(-1)
else:
	for i in range(n):
		j = heapq.heappop(c)
		b.append(j)
		l[j] -= 1
		l[a[i]] -= 1
		if l[a[i]] == 0:
			heapq.heappush(c, a[i])
	print(' '.join(map(str, b)))
