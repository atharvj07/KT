#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, t, m;
  cin >> s >> t >> m;
  for (int i = 0; i < m.size(); i++) {
    if (m[i] <= 'Z' && m[i] >= 'A') {
      m[i] = m[i] - 'A' + 'a';
      int k = 0;
      while (s[k] != m[i]) k++;
      char c = t[k] + 'A' - 'a';
      cout << c;
    } else if (m[i] <= 'z' && m[i] >= 'a') {
      int k = 0;
      while (s[k] != m[i]) k++;
      cout << t[k];
    } else
      cout << m[i];
  }
}
