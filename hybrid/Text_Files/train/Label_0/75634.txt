from sys import stdin

def Readlines():
	for data in stdin.read().split('\n'):
		yield data

def main():
	(lines, result) = (Readlines(), list())
	readln = lambda : next(lines)
	for t in range(int(readln())):
		c = list(map(int, readln().split()))
		a = list(map(int, readln().split()))
		for i in range(5):
			b = min(c[i % 3], a[i])
			(c[i % 3], a[i]) = (c[i % 3] - b, a[i] - b)
		a[3] += a.pop()
		a[3] -= min(a[3], c[2])
		if sum(a):
			result.append('No')
		else:
			result.append('Yes')
	print(*result, sep='\n')
main()
