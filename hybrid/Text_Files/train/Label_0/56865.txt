import java.util.*;

public class LifeLine {
	public static void main(String[] args) {
		Scanner f = new Scanner(System.in);
		int n = f.nextInt();
		int[] deg = new int[n];
		boolean[][] connected = new boolean[n][n];
		for (int i = 0; i < n - 1; i++) {
			int a = f.nextInt() - 1, b = f.nextInt() - 1;
			connected[a][b] = true;
			connected[b][a] = true;
			deg[a]++; deg[b]++;
		}
		//System.out.println(Arrays.toString(deg));
		int lifeLine = 0;
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
				if (connected[i][j]) lifeLine += deg[j] - 1;
		System.out.println(lifeLine / 2);
		f.close();
	}
}
