/* package codechef; // don't place package name! */

import java.util.*;
import java.lang.*;
import java.io.*;

/* Name of the class has to be "Main" only if the class is public. */
public class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int p = in.nextInt();
		String s = in.next();
		boolean flag = false;
		char a[] = s.toCharArray();
		for(int i=0;i<n-p;i++){
		   if(a[i]==a[i+p] && a[i]!='.'){
		       continue;
		   }
		   else if(a[i]==a[i+p] && a[i]=='.'){
		       a[i]='0';
		       a[i+p]='1';
		       flag = true;
		       break;
		   }
		   else if(a[i]!=a[i+p]){
		       if(a[i]=='1' && a[i+p]=='0'){
		           flag = true;
		           continue;
		       }
		       else if(a[i]=='1' && a[i+p]=='.'){
		           a[i+p]='0';
		           flag = true;
		       }
		       else if(a[i]=='0' && a[i+p]=='1'){
		           flag = true;
		           continue;
		       }
		       else if(a[i]=='0' && a[i+p]=='.'){
		           a[i+p]='1';
		           flag = true;
		       }
		       else if(a[i]=='.' && a[i+p]=='1'){
		           a[i]='0';
		           flag = true;
		       }
		       else if(a[i]=='.' && a[i+p]=='0'){
		           a[i]='1';
		           flag = true;
		       }
		   }
		}
		if(flag){
		    for(int i=0;i<n;i++){
	            if(a[i]=='.'){
	                a[i]='0';
	            }
	        }
	        for(int i=0;i<n;i++){
	            System.out.print(a[i]);
	        }
		}
	    else{
	        System.out.println("No");
	    }
	}
}
