#include <bits/stdc++.h>
using namespace std;
int n;
vector<string> s;
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    string str;
    cin >> str;
    if (str == "pwd") {
      cout << "/";
      for (int i = 0; i < s.size(); i++) cout << s[i] << "/";
      cout << endl;
    } else {
      cin >> str;
      str += "/";
      string t;
      for (int i = 0; i < str.size(); i++)
        if (str[i] != '/')
          t += str[i];
        else {
          if (t == "")
            s.clear();
          else if (t == "..")
            s.pop_back();
          else
            s.push_back(t);
          t = "";
        }
    }
  }
  return 0;
}
