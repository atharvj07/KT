n = int(input())
for i in range(n):
	x = int(input())
	k = x
	f = 0
	while x > 0:
		if x % 3 == 0 or x % 7 == 0:
			print('YES')
			f = 1
			break
		else:
			x -= 3
	x = k
	if f == 1:
		continue
	while x > 0:
		if x % 3 == 0 or x % 7 == 0:
			print('YES')
			f = 1
			break
		else:
			x -= 7
	if f == 1:
		continue
	print('NO')
