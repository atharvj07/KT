#include <complex>
#include <cmath>
#include <cstdio>
using namespace std;

typedef complex<double> xy_t;

double dot_product(xy_t a, xy_t b) { return (conj(a)*b).real(); }
double cross_product(xy_t a, xy_t b) { return (conj(a)*b).imag(); }
xy_t projection(xy_t p, xy_t b) { return b*dot_product(p,b)/norm(b); }

int main(){
  double x1, y1, x2, y2, xq, yq;
  while(~scanf("%lf,%lf,%lf,%lf,%lf,%lf", &x1,&y1,&x2,&y2,&xq,&yq)){
    xy_t P1(x1,y1), P2(x2,y2), Q(xq,yq);
    xy_t R = Q + xy_t(2, 0) * (projection(Q-P1, P2-P1) - (Q-P1));
    printf("%.6f %.6f\n", real(R), imag(R));
  }
}
