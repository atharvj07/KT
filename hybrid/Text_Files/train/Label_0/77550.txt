read = lambda : [int(i) for i in input().split()]
t = int(input())
for _ in range(t):
	(n, m) = read()
	pages = n // m
	cycle_cnt = 1
	cycle_sums = [m % 10]
	k = m * 2
	while k % 10 != m % 10:
		cycle_cnt += 1
		cycle_sums.append(cycle_sums[-1] + k % 10)
		k += m
	ans = cycle_sums[-1] * (pages // cycle_cnt)
	if pages % cycle_cnt != 0:
		ans += cycle_sums[pages % cycle_cnt - 1]
	print(ans)
