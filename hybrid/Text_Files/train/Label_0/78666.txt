import sys
input = sys.stdin.buffer.readline

def main():
	(n, m, p) = map(int, input().split())
	A = list(map(int, input().split()))
	B = list(map(int, input().split()))
	for (i, a) in enumerate(A):
		if a % p != 0:
			break
	for (j, b) in enumerate(B):
		if b % p != 0:
			break
	print(i + j)
main()
