#include<bits/stdc++.h>
using namespace std;
int main(){	
  long long n;
  cin >> n;
  for(long long i=1;i<=3500;i++){
  	for(long long j=i;j<=3500;j++){
  	  long long a=4*i*j-n*i-n*j;  	  
  	  if(a>0&&(n*i*j)%a==0){
  	  	long long k=(n*i*j)/a;
  	  	cout << i << ' ' << j << ' ' << k << endl;
  	  	return 0;
  	  }
  	}
  }
}
