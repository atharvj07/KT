n = int(input())
l = [*map(int, input().split())]
minim = [10 ** 18, 0]
for i in range(9):
	if l[i] <= minim[0]:
		minim = [l[i], i + 1]
(leng, rem) = divmod(n, minim[0])
if leng == 0:
	print(-1)
	exit()
fin = []
cnt = 0
for i in range(8, minim[1] - 1, -1):
	if rem + minim[0] >= l[i]:
		(tot, rem) = divmod(rem, l[i] - minim[0])
		fin += [str(i + 1)] * tot
		cnt += tot
print(''.join(fin + (leng - cnt) * [str(minim[1])]))
