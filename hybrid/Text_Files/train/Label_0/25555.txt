ll fastexpo(ll base,ll power)
{
  ll res=1;
  while(power)
  {
    if(power%2 == 0)
    {
     base*= base;
     power/=2;
    }
    else
    {
     res *= base;
     power--;
    }
  }
  return res;
}

ll modexpo(ll base,ll power,ll mod)
{
  ll res=1;
  while(power)
  {
    if(power%2 == 0)
    {
     base = (base*base)%mod;
     power/=2;
    }
    else
    {
     res = (base*res)%mod;
     power--;
    }
  }
  return res;
}

const ll SN = 1e6;

bool prime[SN + 1];
void Sieve(int n)
{ 
    memset(prime, true, sizeof(prime));
 
    for (int p = 2; p * p <= n; p++)
    {
        if (prime[p] == true) 
        {
            for (int i = p * p; i <= n; i += p)
                prime[i] = false;
        }
    }
}
