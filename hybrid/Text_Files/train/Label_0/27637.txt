#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define f(i, x, n) for (int i = x; i < (int)(n); ++i)

int const N = 50000, M = 100000;
int n, m, T;
bool al[N + 1];
char an[M + 1];
bitset<N> msk[N + 1];

struct E{
	int a, b;
	E(){}
	void sc() { scanf("%d%d", &a, &b); }
}ed[M];

void no() { printf("-1\n"); exit(0); }

void solve1(){
	f(i, 1, n + 1)msk[i][i] = true;
	for (int i = m - 1; i >= 0; --i)msk[ed[i].a] = msk[ed[i].b] = msk[ed[i].a] | msk[ed[i].b];
	f(i, 2, n + 1)msk[1] &= msk[i];
	int x = 0;
	f(i, 1, n + 1)if (msk[1][i]){
		x = i;
		break;
	}
	if (!x)no();
	al[x] = true;
	for (int i = m - 1; i >= 0; --i)if (al[ed[i].a] + (int)al[ed[i].b] == 1){
		an[i] = al[ed[i].a] ? '^' : 'v';
		al[ed[i].a] = al[ed[i].b] = true;
	}
	else an[i] = '^';
}

void J(int a, int b, char c){
	if (al[a] && al[b])--n;
	if (c == '^')al[a] = max(al[a], al[b]), al[b] = false;
	else al[b] = max(al[a], al[b]), al[a] = false;
}

void solve2(){
	if (n == 2)no();
	f(i, 1, n + 1)al[i] = true;
	int i = 0;
	while (i < m && n != 3)J(ed[i].a, ed[i].b, an[i] = '^'), ++i;
	if (n == 3)f(j, i, m - 1){
		int sm = al[ed[j].a] + (int)al[ed[j].b];
		if (sm == 0)an[j] = '^';
		else if (sm == 1)an[j] = al[ed[j].a] ? '^' : 'v';
		else {
			int k = j + 1;
			while (k < m - 1 && al[ed[k].a] + (int)al[ed[k].b] < 2)++k;
			an[j] = ed[j].a == ed[k].a || ed[j].a == ed[k].b ? 'v' : '^';
		}
	}
	an[m - 1] = '^';
}

int main(){
	scanf("%d%d%d", &n, &m, &T);
	f(i, 0, m)ed[i].sc();
	if (T == 1)solve1();
	else solve2();
	printf("%s\n", an);
}
