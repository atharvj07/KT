#include <bits/stdc++.h>
using namespace std;
const int N = 1e4 + 10;
const int MOD = 998244353;
int dis[20][20], mp[20][N], dp[20][20][1 << 16], dis_next[20][20];
int main() {
  int n, m;
  scanf("%d%d", &n, &m);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      scanf("%d", &mp[i][j]);
    }
  }
  if (n == 1) {
    int ans = 0x3f3f3f3f;
    for (int i = 1; i < m; i++) ans = min(ans, abs(mp[0][i] - mp[0][i - 1]));
    cout << ans << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      dis[i][j] = INT_MAX;
      dis_next[i][j] = INT_MAX;
      if (i == j) continue;
      for (int k = 0; k < m; k++) {
        dis[i][j] = min(dis[i][j], abs(mp[i][k] - mp[j][k]));
        if (k != 0)
          dis_next[i][j] = min(dis_next[i][j], abs(mp[i][k - 1] - mp[j][k]));
      }
    }
  }
  memset(dp, 0, sizeof(dp));
  for (int i = 0; i < n; i++) dp[i][i][1 << i] = INT_MAX;
  for (int i = 1; i < (1 << n); i++) {
    for (int s = 0; s < n; s++) {
      if ((i & (1 << s)) == 0) continue;
      for (int t = 0; t < n; t++) {
        if (s == t || (i >> t & 1) == 1) continue;
        for (int e = 0; e < n; e++) {
          if ((i & (1 << e)) == 0) continue;
          dp[s][t][i | (1 << t)] =
              max(dp[s][t][i | (1 << t)], min(dp[s][e][i], dis[e][t]));
        }
      }
    }
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i == j) continue;
      ans = max(ans, min(dp[i][j][(1 << n) - 1], dis_next[j][i]));
    }
  }
  cout << ans << endl;
  return 0;
}
