(n, M) = map(int, input().split())
a = [0] + list(map(int, input().split())) + [M]
dp = [0]
m = 0
for i in range(1, n + 2):
	dp.append(dp[i - 1] + (a[i] - a[i - 1]) * (i & 1))
for i in range(1, n + 2):
	if a[i] - a[i - 1] > 1:
		if i & 1:
			m = max(m, dp[i] * 2 + M - 1 - dp[n + 1] - a[i])
		else:
			m = max(m, dp[i] * 2 - a[i - 1] - 1 + M - dp[n + 1])
print(max(dp[n + 1], m))
