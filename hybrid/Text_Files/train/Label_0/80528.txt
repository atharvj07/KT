def solver():
	n = int(input())
	final = []
	writer = []
	changer = {}
	for i in range(n):
		writer.append(list(map(int, input().split())))
	for i in reversed(writer):
		if i[0] == 1:
			if i[1] not in changer:
				dummy = i[1]
			else:
				dummy = changer[i[1]]
			final.append(dummy)
		elif i[2] not in changer:
			changer[i[1]] = i[2]
		else:
			changer[i[1]] = changer[i[2]]
	print(*final[::-1])
solver()
