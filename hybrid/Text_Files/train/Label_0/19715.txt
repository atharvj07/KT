#include<iostream>
#include<string>
#include<set>
using namespace std;

int main(){
  int n,len = 2;
  string s;
  cin >> n >> s; n = 1<<n;
  set<string> a;
  while(len<=n){
    for(int i=0;i<n;i+=len){
      if(s.substr(i,len/2) != s.substr(i+len/2,len/2))
	a.insert(s.substr(i,len));
    }
    len <<= 1;
  }
  cout << a.size() << endl;
}
