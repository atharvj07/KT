#include <stdio.h>

int n, q, mx1, my1, mx2, my2, x1[10000], y1[10000], x2[10000], y2[10000], imos[10001];

int main()
{
	while (true)
	{
		scanf("%d", &n);
		scanf("%d", &q);

		if (n == 0 && q == 0) break;

		mx1 = 10001;
		my1 = 10001;
		mx2 = -1;
		my2 = -1;

		for (int i = 0; i < n; i++)
		{
			scanf("%d", &x1[i]); mx1 = mx1 < x1[i] ? mx1 : x1[i];
			scanf("%d", &y1[i]); my1 = my1 < y1[i] ? my1 : y1[i];
			scanf("%d", &x2[i]); mx2 = mx2 > x2[i] ? mx2 : x2[i];
			scanf("%d", &y2[i]); my2 = my2 > y2[i] ? my2 : y2[i];
		}

		int ret1 = 0;
		int ret2 = 0;

		for (int i = my1; i < my2; i++)
		{
			for (int j = 0; j < n; j++)
			{
				if (y1[j] <= i && i < y2[j])
				{
					imos[x1[j]]++;
					imos[x2[j]]--;
				}
			}

			int sum = 0;

			for (int j = mx1; j <= mx2; j++)
			{
				if ((sum == 0 && sum + imos[j] > 0) || (sum > 0 && sum + imos[j] == 0)) ret2++;

				sum += imos[j]; imos[j] = 0;

				if(sum > 0) ret1++;
			}
		}

		for (int i = mx1; i < mx2; i++)
		{
			for (int j = 0; j < n; j++)
			{
				if (x1[j] <= i && i < x2[j])
				{
					imos[y1[j]]++;
					imos[y2[j]]--;
				}
			}

			int sum = 0;

			for (int j = my1; j <= my2; j++)
			{
				if ((sum == 0 && sum + imos[j] > 0) || (sum > 0 && sum + imos[j] == 0)) ret2++;

				sum += imos[j]; imos[j] = 0;
			}
		}

		printf("%d\n", ret1);

		if (q == 2)
		{
			printf("%d\n", ret2);
		}
	}

	return 0;
}
