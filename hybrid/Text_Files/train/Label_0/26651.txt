#include <iostream>

using namespace std;

template<typename T>
T euler_phi(T n){
    T res = n;
    for(T k=2; k*k<=n; ++k){
        if(n % k)continue;
        for(res=res/k*(k-1); n%k == 0; n/=k);
    }
    return 1 < n? res / n * (n - 1): res;
}

int main(){
    int n;
    cin >> n;
    cout << euler_phi(n) << '\n';
    return 0;
}
