inline int gethash(int h[],int l,int r)
{
	if(r==0) return 0;
	opti=h[r-1]-1LL*(l?h[l-1]:0)*e[r-l]%mod;
	return (opti<0?opti+mod:opti);
}

inline void init(string &c,int h[])
{
	int len=c.size();
	h[0]=c[0];
	for(int i=1;i<len;i++)
	{
		h[i]=(1LL*h[i-1]*107+c[i])%mod;
	}
}

bool dfs(int x,int y)
{
	if(y<0) return false;
	if(x==n) return (y==0);
	if(f[x][y]!=-1) return f[x][y];
	f[x][y]=dfs(x+1,y-len[x]) | dfs(x+1,y);
	return f[x][y];
}

	}

	if(a.first>b.first)

bool compare(const pii &a,const pii &b)
{
	int len1=a.first+a.second*len[var];
	int len2=b.first+b.second*len[var];
	int l=min(a.first,b.first),r=min(len1,len2),mid,res=-1;
	if(l==r) a.first<b.first;
	while(l<=r)
	{
		mid=(l+r)>>1;
		if(same(a,b,mid))
		{
			res=mid;
			l=mid+1;
		}
		else
		{
			r=mid-1;
		}
	}
	if(res==min(len1,len2)) return len1>len2;
	if(res>=a.first && res>=b.first)
	{
		return str[var][res-a.first]<str[var][res-b.first];
	}
	if(res>=a.first)
	{
		return str[var][res-a.first]<cur[res];
	}
	if(res>=b.first)
	{
		return cur[res]<str[var][res-b.first];
	}
}

bool check(pii a,pii b)
{
	int len1=a.first+a.second*len[var];
	int len2=b.first+b.second*len[var];
	int l=min(a.first,b.first),r=min(len1,len2),mid,res=-1;
	if(len2>len1) return false;
	if(l==r) return true;
	while(l<=r)
	{
		mid=(l+r)>>1;
		if(same(a,b,mid))
		{
			res=mid;
			l=mid+1;
		}
		else
		{
			r=mid-1;
		}
	}
	if(res==min(len1,len2)) return true;
	return false;
}
