def bakchodi(x):
	cnt = 0
	while x % 3 == 0:
		x = x // 3
		cnt += 1
	return cnt
n = int(input())
ls = [int(x) for x in input().split()]
for j in range(n):
	for i in range(n - 1):
		if bakchodi(ls[i]) < bakchodi(ls[i + 1]):
			(ls[i], ls[i + 1]) = (ls[i + 1], ls[i])
		elif bakchodi(ls[i]) == bakchodi(ls[i + 1]):
			if ls[i] > ls[i + 1]:
				(ls[i], ls[i + 1]) = (ls[i + 1], ls[i])
for i in ls:
	print(i, end=' ')
