#include <bits/stdc++.h>
using namespace std;
const int N = 5007, inf = 0x3f3f3f3f;
vector<int> e[N];
int ene[N];
int vis[N];
int col[N];
int ok = 1;
vector<int> vec[4];
vector<int> group[N];
int tanto[N];
void dfs(int u, int c, int comp) {
  vis[u] = 1;
  col[u] = c;
  group[comp].push_back(u);
  vec[col[u]].push_back(u);
  for (int v : e[u]) {
    if (vis[v] and col[v] == col[u]) {
      ok = 0;
    }
    if (!vis[v]) dfs(v, 3 - c, comp);
  }
}
int ans[N];
int memo[N][N];
int comp;
int solve(int u, int cnt) {
  if (cnt < 0) return 0;
  if (u > comp) {
    return cnt == 0;
  }
  int& st = memo[u][cnt];
  if (st == -1) {
    st = 0;
    if (group[u].size() == 1) {
      st = solve(u + 1, cnt - 1) or solve(u + 1, cnt);
    } else {
      int ones = tanto[u];
      int twos = group[u].size() - ones;
      st = solve(u + 1, cnt - ones) or solve(u + 1, cnt - twos);
    }
  }
  return st;
}
void build(int u, int cnt) {
  if (cnt < 0) return;
  if (u > comp) return;
  if (group[u].size() == 1) {
    if (solve(u + 1, cnt - 1)) {
      ans[group[u][0]] = 2;
      build(u + 1, cnt - 1);
      return;
    }
    build(u + 1, cnt);
  } else {
    int ones = tanto[u];
    int twos = group[u].size() - ones;
    if (solve(u + 1, cnt - ones)) {
      for (int x : group[u]) {
        if (col[x] == 1) {
          ans[x] = 2;
        }
      }
      build(u + 1, cnt - ones);
      return;
    }
    for (int x : group[u]) {
      if (col[x] == 2) {
        ans[x] = 2;
      }
    }
    build(u + 1, cnt - twos);
  }
}
int main() {
  memset(memo, -1, sizeof memo);
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= 3; ++i) cin >> ene[i];
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    e[u].push_back(v);
    e[v].push_back(u);
  }
  for (int i = 1; i <= n; ++i) {
    if (!vis[i]) dfs(i, 1, ++comp);
  }
  for (int i = 1; i <= comp; i++) {
    tanto[i] = 0;
    for (int x : group[i]) tanto[i] += col[x] == 1;
  }
  if (!ok) {
    cout << "NO\n";
    return 0;
  }
  if (solve(1, ene[2])) {
    cout << "YES\n";
    build(1, ene[2]);
    for (int i = 1; i <= n; ++i) {
      if (ans[i] == 0) {
        if (ene[1])
          ans[i] = 1, ene[1]--;
        else
          ans[i] = 3;
      }
      cout << ans[i];
    }
    cout << '\n';
    return 0;
  }
  cout << "NO\n";
  return 0;
}
