#include <bits/stdc++.h>
using namespace std;
void run();
int main() {
  ios::sync_with_stdio(0);
  run();
}
struct song {
  int l;
  long double p;
};
bool should_swap(song a, song b) {
  return (b.p * (1 - a.p) * b.l) < (a.p * (1 - b.p) * a.l);
}
void run() {
  int n;
  long double t = 0, sum = 0;
  cin >> n;
  song s[n];
  for (int i = 0; i < n; i++) {
    cin >> s[i].l >> s[i].p;
    s[i].p /= 100.0;
  }
  sort(s, s + n, should_swap);
  for (int i = n - 1; i >= 0; i--) {
    t += s[i].l + s[i].l * (s[i].p * sum);
    sum += (1 - s[i].p);
  }
  cout.precision(20);
  cout << t << endl;
}
