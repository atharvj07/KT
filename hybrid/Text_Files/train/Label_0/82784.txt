from math import sqrt

def main():
	n = int(input())
	prime = []
	m = int(sqrt(n))
	if n % 2 == 0:
		prime.append(2)
		while n % 2 == 0:
			n = n // 2
	for i in range(3, m + 1, 2):
		if n < i:
			break
		if n % i == 0:
			prime.append(i)
			while n % i == 0:
				n = n // i
	if n > 1:
		prime.append(n)
	ans = 1
	for p in prime:
		ans = ans * p
	print(ans)
main()
