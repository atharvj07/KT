import sys

def get_ints():
	return list(map(int, sys.stdin.readline().strip().split()))
testcases = int(input())
for testcase in range(testcases):
	temparr = get_ints()
	a = temparr[0]
	b = temparr[1]
	ans = []
	if a == 4:
		print('2 1 1')
	elif a & 1:
		ans.append(str(1))
		a -= 1
		diff = a >> 1
		ans.append(str(diff))
		ans.append(str(diff))
		print(' '.join(ans))
	else:
		diff = a >> 1
		ans = []
		if diff & 1:
			ans.append(str(2))
			ans.append(str(diff - 1))
			ans.append(str(diff - 1))
			print(' '.join(ans))
		else:
			halfhalf = diff >> 1
			ans.append(str(diff))
			ans.append(str(halfhalf))
			ans.append(str(halfhalf))
			print(' '.join(ans))
