import sys
input = sys.stdin.readline
t = int(input())
results = []
for i in range(t):
	(a, b, c) = map(int, input().rstrip().split())
	if a < b:
		(a, b) = (b, a)
	people = (a - b) * 2
	if a > people or c > people:
		results.append(-1)
	elif c + (a - b) > people:
		results.append((c + a - b) % people)
	else:
		results.append(c + a - b)
for i in results:
	print(i)
