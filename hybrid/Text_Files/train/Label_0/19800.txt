#include <bits/stdc++.h>
using namespace std;
#define FOR(i,a,b) for(int i=(a);i<(b);++i)
#define rep(i,n)   FOR(i,0,n)
#define pb push_back
#define mp make_pair
typedef long long ll;
typedef pair<int,int> pint;

int d[11][11];
bool used[11][11];
vector<int> route;
int ans,ns,nl;
void dfs(int ds,int c,vector<int> v){
    int cnt=0;
    FOR(i,1,ns+1){
        if(!used[ds][i]){
            ++cnt;continue;
        }
        used[ds][i]=used[i][ds]=false;
        vector<int> v2=v;
        v2.pb(i);
        dfs(i,c+d[ds][i],v2);
        used[ds][i]=used[i][ds]=true;
    }
    if(cnt==ns&&ans<c){
        ans=c;route=v;
    }
}
int main(){
    while(cin>>ns>>nl,ns){
        ans=0;
        memset(used,0,sizeof(used));
        int src,dst,dis;
        rep(i,nl){
            cin>>src>>dst>>dis;
            used[src][dst]=used[src][dst]=true;
            d[src][dst]=d[dst][src]=dis;
        }
        FOR(i,1,ns+1){
            vector<int> vi2;
            vi2.pb(i);
            dfs(i,0,vi2);
        }
        cout<<ans<<endl;
        rep(i,route.size()-1) cout<<route[i]<<" ";
        cout<<route.back()<<endl;
    }
    return 0;
}
