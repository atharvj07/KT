#include<bits/stdc++.h>
#define pb push_back
#define _x(i) v1[p[i].x]
#define _y(i) v2[p[i].y]

using namespace std;
typedef long long ll;

const int N=100005,M=4000005,L=8888888,K=16666666;

int n,a,b,dis;
ll ans;
int v1[N],m1,v2[N],m2,va[N];
struct node{
	int x,y,id;
	inline bool friend operator<(const node&a,const node&b){return a.x<b.x||a.x==b.x&&a.y<b.y;}
}p[N];
inline int abs_(int x){return x<0?-x:x;}

map<int,int>R1,R2;
int ls[M],rs[M],sz[M],id[M],ndc;
int num,h[L],nxt[K],to[K],ce;
inline void addedge(int u,int v){nxt[++ce]=h[u],to[ce]=v,h[u]=ce;}
inline int copynode(int x){
	int y=++ndc;
	ls[y]=ls[x],rs[y]=rs[x],sz[y]=sz[x];
	id[y]=++num;
	if(x)addedge(num,id[x]);
	return y;
}
void insert(int&x,int l,int r,int p,int u){
	x=copynode(x);
	sz[x]++;addedge(id[x],u);
	if(l==r)return;
	int mid=(l+r)>>1;
	if(p<=mid)insert(ls[x],l,mid,p,u);else insert(rs[x],mid+1,r,p,u);
}
int link(int x,int l,int r,int s,int t,int u){
	if(!x||s>t)return 0;
	if(l==s&&r==t){addedge(u,id[x]);return sz[x];}
	int mid=(l+r)>>1;
	if(t<=mid)return link(ls[x],l,mid,s,t,u);
	if(s>mid)return link(rs[x],mid+1,r,s,t,u);
	return link(ls[x],l,mid,s,mid,u)+link(rs[x],mid+1,r,mid+1,t,u);
}

bool vi[L];
void dfs(int x){
	if(vi[x])return;
	vi[x]=true;
	if(x<=n)ans+=va[x];
	for(int i=h[x];i;i=nxt[i])dfs(to[i]);
}

int main(){
	scanf("%d%d%d",&n,&a,&b);
	for(int i=1;i<=n;i++)scanf("%d%d",&p[i].x,&p[i].y),v1[++m1]=p[i].x,v2[++m2]=p[i].y;
	dis=abs_(p[a].x-p[b].x)+abs_(p[a].y-p[b].y);
	sort(v1+1,v1+m1+1);m1=unique(v1+1,v1+m1+1)-v1-1;
	sort(v2+1,v2+m2+1);m2=unique(v2+1,v2+m2+1)-v2-1;
	for(int i=1;i<=n;i++){
		p[i].x=lower_bound(v1+1,v1+m1+1,p[i].x)-v1;
		p[i].y=lower_bound(v2+1,v2+m2+1,p[i].y)-v2;
		p[i].id=i;
	}
	sort(p+1,p+n+1);num=n;
	ndc=0;R1.clear();R2.clear();
	for(int i=1;i<=n;i++){
		int X=v1[p[i].x],Y=v2[p[i].y];
		va[i]+=link(R1[X+Y-dis],1,m2,1,p[i].y,i);
		va[i]+=link(R2[X-Y-dis],1,m2,p[i].y+1,m2,i);
		insert(R1[X+Y],1,m2,p[i].y,i);
		insert(R2[X-Y],1,m2,p[i].y,i);
	}
	ndc=0;R1.clear();R2.clear();
	for(int i=n;i>=1;i--){
		int X=v1[p[i].x],Y=v2[p[i].y];
		va[i]+=link(R1[Y-X-dis],1,m2,1,p[i].y,i);
		va[i]+=link(R2[-Y-X-dis],1,m2,p[i].y+1,m2,i);
		insert(R1[Y-X],1,m2,p[i].y,i);
		insert(R2[-Y-X],1,m2,p[i].y,i);
	}
	for(int i=1;i<=n;i++)if(p[i].id==a||p[i].id==b)dfs(i);
	printf("%lld\n",ans/2);
	return 0;
}
