class TaskG2 {
    int[] p;
    int k;

    public void solve(int testNumber, InputReader in, PrintWriter out) {

        int n = in.nextInt();
        k = in.nextInt();
        p = new int[n];

        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }

        int pairsCnt = ((n * (n - 1) / 2) + n);
        double oneProb = 1d / pairsCnt;

        double[][][] dp = new double[n][n][k + 1];

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                dp[i][j][0] = 1;
            }
        }

        for (int step = 1; step <= k; step++) {

            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {

                    int usedIntervals = 0;
                    for (int newIPos = 0; newIPos < n; newIPos++) {

                        int ls = Math.min(i, newIPos);
                        int rs = Math.max(i, newIPos);

                        int jNotMovedIntervals = Math.min(j - rs, ls + 1);
                        if (jNotMovedIntervals > 0) {
                            usedIntervals += jNotMovedIntervals;
                            dp[i][j][step] += dp[newIPos][j][step - 1] * jNotMovedIntervals * oneProb;
                        }

                        int rsJMoved = Math.max(rs, j);
                        int lsJMoved = ls - (rsJMoved - rs);
                        int jMovedIntervals = Math.min(lsJMoved + 1, n - rsJMoved);
                        int newJPos = newIPos - (j - i);
                        if (jMovedIntervals > 0) {
                            usedIntervals += jMovedIntervals;
                            if (newJPos < 0) throw new RuntimeException();

                            dp[i][j][step] += (1 - dp[newJPos][newIPos][step - 1]) * jMovedIntervals * oneProb;
                        }
                    }
                    //only J in interval
                    for (int newJPos = 0; newJPos < n; newJPos++) {
                        int ls = Math.min(j, newJPos);
                        int rs = Math.max(j, newJPos);

                        int iNotMovedIntervals = Math.min(ls - i, n - rs);
                        if (iNotMovedIntervals > 0) {
                            usedIntervals += iNotMovedIntervals;

                            dp[i][j][step] += dp[i][newJPos][step - 1] * iNotMovedIntervals * oneProb;
                        }
                    }

                    dp[i][j][step] += dp[i][j][step - 1] * (pairsCnt - usedIntervals) * oneProb;

//                    for (int l = 0; l < n; l++) {
//                        for (int r = l; r < n; r++) {
//
//                            int newI = getNewCoordinate(i, l, r);
//                            int newJ = getNewCoordinate(j, l, r);
//                            if (newI < newJ) {
//                                dp[i][j][step] += dp[newI][newJ][step - 1] * oneProb;
//                            } else {
//                                dp[i][j][step] += (1 - dp[newJ][newI][step - 1]) * oneProb;
//                            }
//                        }
//                    }
                }
            }
        }

        double res = 0.;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (p[i] < p[j]) {
                    res += 1 - dp[i][j][k];
                } else {
                    res += dp[i][j][k];
                }
            }
        }

        out.println(String.format(Locale.US, "%.15f", res));
    }

}

class InputReader {
    private BufferedReader reader;
    private String[] currentArray;
    private int curPointer;

    public InputReader(InputStream inputStream) {
        reader = new BufferedReader(new InputStreamReader(inputStream));
    }


    public int nextInt() {
        if ((currentArray == null) || (curPointer >= currentArray.length)) {
            try {
                currentArray = reader.readLine().split(" ");
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
            curPointer = 0;
        }
        return Integer.parseInt(currentArray[curPointer++]);
    }

}
