from sys import stdin, stdout
import bisect

def main():
	n = int(stdin.readline())
	arr = [int(el) for el in stdin.readline().split()]
	i = 0
	res = []
	while i < n:
		if arr[i] - 1 == i:
			i += 1
		else:
			pos2 = arr[i] - 1
			if abs(pos2 - i) >= n // 2:
				res.append([min(i + 1, pos2 + 1), max(i + 1, pos2 + 1)])
				temp = arr[i]
				arr[i] = arr[pos2]
				arr[pos2] = temp
			elif i < n // 2 and pos2 < n // 2:
				res.append([i + 1, n])
				res.append([pos2 + 1, n])
				res.append([i + 1, n])
				temp = arr[i]
				arr[i] = arr[pos2]
				arr[pos2] = temp
			elif i < n // 2 and pos2 >= n // 2:
				res.append([i + 1, n])
				res.append([1, pos2 + 1])
				res.append([1, n])
				res.append([1, pos2 + 1])
				res.append([i + 1, n])
				temp = arr[i]
				arr[i] = arr[pos2]
				arr[pos2] = temp
			else:
				res.append([1, i + 1])
				res.append([1, pos2 + 1])
				res.append([1, i + 1])
				temp = arr[i]
				arr[i] = arr[pos2]
				arr[pos2] = temp
	print(len(res))
	for i in range(len(res)):
		print(*res[i])
main()
