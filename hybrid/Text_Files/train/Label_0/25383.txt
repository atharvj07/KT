long long fastPow(long long x, long long y, int mod = 1000000007) {
  long long ans = 1;
  while (y > 0) {
    if (y & 1) ans = (x * ans) % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ans;
}
struct Point {
  long long x, y;
  Point() {}
  Point(long long x, long long y) : x(x), y(y) {}
  Point operator+(const Point &o) const { return Point(x + o.x, y + o.y); }
  Point operator-(const Point &o) const { return Point(x - o.x, y - o.y); }
  bool operator<(const Point &o) const {
    return x < o.x || (x == o.x && y < o.y);
  }
  friend bool operator==(const Point &r1, const Point &r2) {
    return r1.x == r2.x && r1.y == r2.y;
  }
  long long cross(Point b) const { return x * b.y - b.x * y; }
  void readin() { scanf("%lld%lld", &x, &y); }
  void pp() { printf("%lld %lld\n", x, y); }
} p[200105];
vector<Point> convex(vector<Point> &vp) {
  sort(vp.begin(), vp.end());
  int N = unique(vp.begin(), vp.end()) - vp.begin();
  vector<Point> ans1, ans2;
  for (int(i) = 0; (i) < (int)(N); (i)++) {
    while (ans1.size() > 1) {
      Point o = ans1[ans1.size() - 2], a = ans1.back();
      if ((a - o).cross(vp[i] - o) <= 0) {
        ans1.pop_back();
      } else
        break;
    }
    ans1.push_back(vp[i]);
  }
  for (int i = N - 1; i >= 0; i--) {
    while (ans2.size() > 1) {
      Point o = ans2[ans2.size() - 2], a = ans2.back();
      if ((a - o).cross(vp[i] - o) <= 0) {
        ans2.pop_back();
      } else
        break;
    }
    ans2.push_back(vp[i]);
  }
  for (int i = 1; i < ans2.size() - 1; i++) ans1.push_back(ans2[i]);
  return ans1;
}

bool isIn(long long x, long long y) {
  for (int i = 0; i <= 1; i++)

          yr[x + i] < y + j)
        return 0;
  return 1;
}
