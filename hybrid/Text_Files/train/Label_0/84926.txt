def main():
	(p, q, l, r) = map(int, input().split())
	(ab, cd) = ([False] * (1001 - l), [False] * 1001)
	for _ in range(p):
		(a, b) = map(int, input().split())
		for i in range(1000 - b, 1001 - max(a, l)):
			ab[i] = True
	for _ in range(q):
		(a, b) = map(int, input().split())
		for i in range(a, b + 1):
			cd[i] = True
	res = 0
	for _ in range(l, r + 1):
		if any((a and b for (a, b) in zip(reversed(ab), cd))):
			res += 1
		del ab[-1]
	print(res)
main()
