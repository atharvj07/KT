for _ in range(int(input())):
	n = int(input())
	dic = [[] for _ in range(n + 1)]
	gress = [0] * (n + 1)
	gress[1] += 1
	father = [0] * (n + 1)
	now = [1]
	s = [[] for _ in range(n + 1)]
	leaf = []
	for _ in range(n - 1):
		(u, v) = map(int, input().split())
		dic[u].append(v)
		dic[v].append(u)
	while now:
		node = now.pop()
		for child in dic[node]:
			if child != father[node]:
				gress[node] += 1
				father[child] = node
				now.append(child)
		if gress[node] == 0:
			leaf.append(node)
	while leaf:
		node = leaf.pop()
		f = father[node]
		if not s[node]:
			s[f].append((1, 0))
		elif len(s[node]) == 1:
			(d, k) = s[node][0]
			s[f].append((d + 1, k))
		else:
			d = min((p[0] for p in s[node])) + 1
			k = max(max((p[1] for p in s[node])), max((p[0] for p in s[node])) + 1)
			s[f].append((d, k))
		gress[f] -= 1
		if gress[f] == 0:
			leaf.append(f)
	node = 1
	if len(s[node]) == 1:
		print(max(s[node][0]))
	else:
		k = max((p[1] for p in s[node]))
		tmp = [p[0] for p in s[node]]
		m = max(tmp)
		tmp.remove(m)
		print(max(max(tmp) + 1, m, k))
