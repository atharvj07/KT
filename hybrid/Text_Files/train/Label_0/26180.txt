#include <bits/stdc++.h>

typedef long long ll;
typedef long double ld;
#define FASTIO ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
#define PRECISION std::cout << std::fixed << std::setprecision(20);

using namespace std;

const int inf = 2e9;

template <typename SegNode>
struct SegTree {
    vector<SegNode> tree, arr;
    int n;

    SegTree(int n, int val): n(n) {
        tree.assign(4 * n, val);
    }

    SegTree(int n, vector<SegNode> arr): n(n) {
        tree.assign(4 * n, 0);
        arr = arr;
        build(0, n - 1, 0);
    }

    SegNode merge(SegNode l, SegNode r) {
        return max(l, r);
    }

    void build(int ss, int se, int si) {
        if (ss == se) {
            tree[si] = arr[ss];
        } else {
            int mid = ss + se >> 1;
            build(ss, mid, 2 * si + 1);
            build(mid + 1, se, 2 * si + 2);
            tree[si] = merge(tree[2 * si + 1], tree[2 * si + 2]);
        }
    }

    SegNode query(int qs, int qe) {
        return query(qs, qe, 0, n - 1, 0);
    }
 
    SegNode query(int qs, int qe, int ss, int se, int si) {
        if (ss >= qs && se <= qe) {
            return tree[si];
        }

        int mid = ss + se >> 1;
        if (qe <= mid) {
            return query(qs, qe, ss, mid, 2 * si + 1);
        } else if (qs >= mid + 1) {
            return query(qs, qe, mid + 1, se, 2 * si + 2);
        } else {
            return merge(query(qs, qe, ss, mid, 2 * si + 1), query(qs, qe, mid + 1, se, 2 * si + 2));
        }
    }

    void update(int qi, SegNode val) {
        update(qi, val, 0, n - 1, 0);
    }
 
    void update(int qi, SegNode val, int ss, int se, int si) {
        if (ss == se) {
            tree[si] = max(tree[si], val);
        } else {
            int mid = ss + se >> 1;
            if (qi <= mid)
                update(qi, val, ss, mid, 2 * si + 1);
            else
                update(qi, val, mid + 1, se, 2 * si + 2);
            tree[si] = merge(tree[2 * si + 1], tree[2 * si + 2]);
        }
    }
};

void solve() {
    int n;
    cin >> n;
    vector<int> arr(n), brr(n), store;
    for (int i = 0; i < n; i++) cin >> arr[i], store.push_back(arr[i]);
    for (int i = 0; i < n; i++) cin >> brr[i], store.push_back(brr[i]);

    sort(store.begin(), store.end());
    store.resize(unique(store.begin(), store.end()) - store.begin());

    int m = store.size();
    map<int, int> id;
    for (int i = 0; i < m; i++) id[store[i]] = i;

    ll ans = 0;

    vector<array<int, 2>> X, Y;
    for (int i = 0; i < n; i++) {
        if (arr[i] < brr[i]) {
            X.push_back({arr[i], brr[i]});
        } else {
            Y.push_back({brr[i], arr[i]});
        }
        ans += abs(arr[i] - brr[i]);
    }

    int ma = 0;

    for (int itr = 0; itr < 2; itr++) {
        {
            SegTree<int> tree(m, -inf);
            for (auto y: Y) {
                tree.update(id[y[0]], y[1]);
            }

            for (auto x: X) {
                int r = tree.query(0, id[x[0]]);
                if (r != -inf) {
                    r = min(r, x[1]);
                    ma = max(ma, r - x[0]);
                }
            }
        }

        {
            SegTree<int> tree(m, -inf);
            for (auto y: Y) {
                tree.update(id[y[1]], -y[0]);
            }

            for (auto x: X) {
                int l = -tree.query(id[x[1]], m - 1);
                if (l != inf) {
                    l = max(l, x[0]);
                    ma = max(ma, x[1] - l);
                }
            }
        }

        X.swap(Y);
    }

    cout << ans - 2 * ma << "\n";
}

int main() {
    FASTIO;
    PRECISION;

    int t = 1;
    // cin >> t;

    for (int i = 1; i <= t; i++) {
        solve();
    }
}
