import java.util.*;
import java.io.*;
public class a{
   
    public static void main(String [] args)throws IOException{
        BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
        String [] data=br.readLine().trim().split(" ");
        int n=Integer.parseInt(data[0]);
        int m=Integer.parseInt(data[1]);
        data=br.readLine().trim().split(" ");
        int [] a=new int[n];
        Queue<Integer> q=new LinkedList<>();
        TreeSet<Integer> tset=new TreeSet<>();
        Map<Integer,Integer> map=new HashMap<>();

        for(int i=0;i<n;i++){
            a[i]=Integer.parseInt(data[i]);
            q.add(a[i]);
            map.put(a[i],1);
            tset.add(a[i]);
        }

        int cnt=0;
        List<Integer> list=new ArrayList<>();
        while(!q.isEmpty() && cnt<m){
            int u=q.poll();
            if(!map.containsKey(u+1) && cnt<m){
                q.add(u+1);
                list.add(u+1);
                map.put(u+1,1);
                cnt++;
            }
            if(!map.containsKey(u-1) && cnt<m){
                q.add(u-1);
                list.add(u-1);
                map.put(u-1,1);
                cnt++;
            }
        }
        long sum=0;
        for(int i : list){
            Integer lower=tset.lower(i);
            Integer higher=tset.higher(i);
            if(lower!=null && higher !=null)
                sum+=Math.min(Math.abs(i-lower),Math.abs(i-higher));
            else if(lower!=null)
                sum+=Math.abs(i-lower);
            else  
               sum+=Math.abs(i-higher);

        }
        System.out.println(sum);
        StringBuilder ss=new StringBuilder("");
        for(int i : list){
            ss.append(i+" ");
        }

        System.out.println(ss.toString());

    }
}
