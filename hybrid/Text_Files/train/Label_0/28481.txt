#include <bits/stdc++.h>
using namespace std;
int n, t, flag, i;
char str[500005];
int main() {
  scanf("%d\n", &t);
  while (t--) {
    scanf("%d", &n);
    scanf("\n%s", str);
    if (n < 11) {
      puts("NO");
      continue;
    }
    n = n - 11 + 1;
    flag = 0;
    for (i = 0; i < n; i++)
      if (str[i] == '8') flag = 1;
    if (flag)
      puts("YES");
    else
      puts("NO");
  }
  return 0;
}
