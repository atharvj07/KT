import java.io.*;
import java.util.*;

import static java.lang.Math.*;

public class CF1535E {
    public static void main(String[] args) throws Exception {
        final int LOG = 20;
        int q = nextInt();

        int[][] parent = new int[q + 1][LOG];
        int[] gold = new int[q + 1];
        int[] price = new int[q + 1];
        int[] depth = new int[q + 1];

        gold[0] = nextInt();
        price[0] = nextInt();
        depth[0] = 0;

        for (int i = 1; i <= q; i++) {
            int op = nextInt();

            if (op == 1) {
                int p = nextInt();
                int a = nextInt();
                int c = nextInt();

                parent[i][0] = p;
                gold[i] = a;
                price[i] = c;
                depth[i] = depth[p] + 1;
                processNode(i, parent);
            } else {
                long amount = 0, cost = 0;
                int node = nextInt();
                int requiredGold = nextInt();
                while (requiredGold > 0) {
                    int p = getFirstNonEmptyParent(node, parent, gold);
                    if (gold[p] == 0) {
                        break;
                    }
                    int min = min(gold[p], requiredGold);
                    gold[p] -= min;
                    requiredGold -= min;
                    amount += min;
                    cost += (long) min * price[p];
                }
                System.out.println(amount + " " + cost);
            }
        }
    }

    static int nextInt() {
        try {
            boolean in = false;
            int res = 0;
            for (; ; ) {
                int b = System.in.read() - '0';
                if (b >= 0) {
                    in = true;
                    res = 10 * res + b;
                } else if (in)
                    return res;
            }
        } catch (IOException e) {
            throw new Error(e);
        }
    }

    static int getFirstNonEmptyParent(int node, int[][] parent, int[] gold) {
        int p = node;
        int LOG = parent[node].length;
        for (int j = LOG - 1; j >= 0; j--) {
            if (gold[parent[p][j]] > 0) {
                p = parent[p][j];
            }
        }
        return p;
    }

    static int getKthParent(int node, int k, int[][] parent) {
        if (k == 0) {
            return node;
        }
        int LOG = parent[node].length;
        int p = node;
        for (int i = LOG - 1; i >= 0; i--) {
            if (k >= (1 << i)) {
                p = parent[p][i];
                k -= (1 << i);
            }
        }
        return p;
    }

    static void processNode(int node, int[][] parent) {
        int LOG = parent[node].length;
        for (int i = 1; i < LOG; i++) {
            parent[node][i] = parent[parent[node][i - 1]][i - 1];
        }
    }

    static void debug(Object... obj) {
        System.err.println(Arrays.deepToString(obj));
    }
}
