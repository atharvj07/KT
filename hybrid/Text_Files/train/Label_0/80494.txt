def main():
	n = int(input().strip())
	st = input().strip()
	if n % 4:
		print('===')
		return
	li = [0] * 5
	t = n // 4
	for i in st:
		if i == '?':
			li[0] += 1
		elif i == 'A':
			li[1] += 1
		elif i == 'C':
			li[2] += 1
		elif i == 'G':
			li[3] += 1
		else:
			li[4] += 1
	for i in li[1:]:
		if i > t:
			print('===')
			return
	string = ''
	for i in st:
		d = i
		if d == '?':
			for p in range(1, 5):
				if li[p] < t:
					li[p] += 1
					if p == 1:
						d = 'A'
					elif p == 2:
						d = 'C'
					elif p == 3:
						d = 'G'
					else:
						d = 'T'
					break
		string += d
	print(string)
main()
