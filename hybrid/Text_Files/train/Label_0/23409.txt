#include <bits/stdc++.h>
using namespace std;
const long long INF = (long long)(1e18 + 7);
const double EPS = 1e-6;
const int MAXN = (int)(1e5 + 500);
long long n, k, del;
int main() {
  srand(time(0));
  int ttt;
  cin >> ttt;
  while (ttt--) {
    cin >> n >> k;
    del = n;
    for (int i = 2; i * i <= n; i++)
      if (n % i == 0) {
        del = i;
        break;
      }
    n += del;
    k--;
    n += (k * 2);
    cout << n << endl;
  }
  return 0;
}
