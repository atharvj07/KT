import math
n = int(input())
for e in range(n):
	t = int(input())
	l = list(map(int, input().split()))
	mx = max(l)
	if mx == 0:
		ans = []
		for i in range(1, t + 1):
			ans.append(i)
		print(*ans)
	else:
		q = int(math.log(mx, 2)) + 1
		a = []
		for i in range(q):
			c = 0
			for j in l:
				if j & 1 << i != 0:
					c += 1
			a.append(c)
		ans = []
		for i in range(1, max(a) + 1):
			chk = 0
			for j in a:
				if j % i == 0:
					continue
				else:
					chk = 1
					break
			if chk == 0:
				ans.append(i)
		print(*ans)
