#include <iostream>
#include <vector>
#include <map>
#include <algorithm>
#include <climits>
#include <queue>
using namespace std;
typedef pair<int,pair<int,int> >NODE;
#define name first
#define val second
#define CST first
#define POS second.first
#define BIT second.second
#define rep(i,n) for(int i=0;i<n;i++)

int main(){
	int n;
	while(1){
		cin >> n;if(n==0)break;
		int N = n+1;
		vector< vector< pair<string,int> > > info(n);
		
		for(int i=0;i<n;i++){
			int k; cin >> k;
			for(int j=0;j<k;j++){
				string s;int value;
				cin >> s >> value;
				info[i].push_back(make_pair(s,value));
			}
		}
		int q; cin >> q; if(q==0){cout << 0 << endl;continue;}
		vector<string> wantList(q);
		rep(i,q)cin >> wantList[i];
		sort(wantList.begin(),wantList.end());
		
		vector< pair<int,int> > ret;
		
		for(int bit=0;bit<(1<<n);bit++){
			map<string,int> data;
			rep(i,q)data[wantList[i]] = 10001;
			for(int j=0;j<n;j++){
				if(bit & (1<<j) ){
					rep(i,info[j].size()){
						int value = info[j][i].val;
						string name = info[j][i].name;
						if(count(wantList.begin(),wantList.end(),name)){
							data[name] = min(data[name],value);
						}
					}
				}
			}
			bool f = false;
			int sum = 0;
			rep(i,q){
				if(data[wantList[i]] == 10001)f=true;
				sum += data[wantList[i]];
			}
			if(f)continue;
			ret.push_back( make_pair(sum,bit) );
		}
		sort(ret.begin(),ret.end());
		
				
		int m; cin >> m;
		int wf[11][11]; rep(x,11)rep(y,11)wf[x][y] = (1<<21);
		rep(i,11)wf[i][i] = 0;
		
		rep(i,m){
			int a,b,c;
			cin >> a >> b >> c;
			wf[a][b] = wf[b][a] = c;
		}
		
		int homo = INT_MAX;
		
		rep(k,11)rep(i,11)rep(j,11)
			wf[i][j] = min(wf[i][j],wf[i][k]+wf[k][j]);
			
		int mostMin; if(ret.size())mostMin = ret[0].first;
		
		int dp[1<<11][11];
		
		rep(x,1<<11)rep(y,11)dp[x][y]=(1<<21);
		
		dp[1][0] = 0;
		rep(bit,1<<N){
			rep(pos,N){
				if(bit & (1<<pos) == 0)continue;
				rep(next,N){
					if((1<<next) & bit)continue;
					int nextbit = bit | (1<<next);
					dp[nextbit][next] = min(dp[nextbit][next], dp[bit][pos]+wf[pos][next]);
				}
			}
		}
		
		/*rep(i,1<<N){
			rep(j,N)
				cout << dp[i][j] << " ";
			cout << endl;
		}*/
		rep(t,ret.size()){
			if(ret[t].first != mostMin )break;
			int bitch = (ret[t].second<<1)+1;
			for(int i=1;i<n;i++)
				homo = min(homo,dp[bitch][i]+wf[i][0]);
		}
		if(homo == INT_MAX)cout << "impossible" << endl;
		else cout << mostMin << " " << homo << endl;
	}
	
}
