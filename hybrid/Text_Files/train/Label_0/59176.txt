import java.util.Scanner;

public class Adding_Digits {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int a = scan.nextInt();
		int b = scan.nextInt();
		int n = scan.nextInt();
		int ans = a;
		for (int i = 0 ; i < 10 ; i++) {
			if (Integer.parseInt("" + a + i) % b == 0) {
				ans = Integer.parseInt("" + a + i);
				break;
			}
		}
		if (ans == a)
			System.out.println(-1);
		else {
			System.out.print(ans);
			for (int i = 0 ; i < n-1 ; i++)
				System.out.print(0);
		}
	}

}
