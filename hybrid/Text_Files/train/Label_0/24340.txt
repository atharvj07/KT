#include <bits/stdc++.h>
using namespace std;
long long su(long long a) {
  long long s = 0;
  while (a) {
    s += a % 10;
    a /= 10;
  }
  return s;
}
long long pow(long long a, long long b) {
  if (b == 0)
    return 1;
  else
    return a * pow(a, b - 1);
}
int main() {
  long long a, b, c, i, x = 0, h, z;
  cin >> a >> b >> c;
  vector<long long> v;
  for (i = 1; i <= 81; i++) {
    h = b * pow(i, a) + c;
    z = su(h);
    if (z == i && h < 1000000000) {
      x++;
      v.push_back(h);
    }
  }
  cout << v.size() << endl;
  for (i = 0; i < x; i++) cout << v[i] << endl;
}
