#include <cstdio>

#include <string>

#include <iostream>
#include <string>
#include <map>

#include <functional>
#include <iostream>
#define MOD 1000000007LL
using namespace std;
typedef long long ll;
typedef pair<int,int> P;

int n,k;
deque<P> que;

int main(void){
	scanf("%d%d",&n,&k);
	string str;
	cin >> str;
	for(int i=0;i<n;i++){
		if(str[i]=='A')que.push_back(P(0,0));
		else que.push_back(P(1,0));
	}
	int flip=0;
	for(int i=0;i<min(n*2,k);i++){
		int g=que[0].first;
		if((flip-que[0].second)%2==1)g=1-g;
		if(g==0){
			que.pop_front();
			que.push_front(P(1,flip));
		}else{
			flip++;
			que.pop_front();
			que.push_back(P(0,flip));
		}
	}
	for(int i=0;i<n;i++){
		if((flip-que[i].second)%2==1){
			que[i].first=1-que[i].first;
		}
	}
	if(n*2<k){
		int rest=k-n*2;
		if(n%2==1 && rest%2==1){
			que[0].first=1-que[0].first;
		}
	}
	for(int i=0;i<n;i++){
		printf("%c",que[i].first==0?'A':'B');
	}
	printf("\n");
	return 0;
}
