import java.io.*;
import java.util.*;

public class CF1263E extends PrintWriter {
	CF1263E() { super(System.out, true); }
	Scanner sc = new Scanner(System.in);
	public static void main(String[] $) {
		CF1263E o = new CF1263E(); o.main(); o.flush();
	}

	void main() {
		int n = sc.nextInt();
		int[] pmin = new int[n + 3];
		int[] pmax = new int[n + 3];
		int[] psum = new int[n + 3];
		int[] qmin = new int[n + 3];
		int[] qmax = new int[n + 3];
		int[] qsum = new int[n + 3];
		byte[] cc = sc.next().getBytes();
		byte[] aa = new byte[n + 3];
		int h = 1;
		for (int i = 0; i < n; i++) {
			if (cc[i] == 'L') {
				if (h > 1)
					h--;
			} else if (cc[i] == 'R') {
				h++;
			} else
				aa[h] = cc[i];
			psum[h] = psum[h - 1] + (aa[h] == ')' ? -1 : aa[h] == '(' ? 1 : 0);
			pmin[h] = Math.min(pmin[h - 1], psum[h]);
			pmax[h] = Math.max(pmax[h - 1], psum[h]);
			qsum[h] = qsum[h + 1] + (aa[h] == '(' ? -1 : aa[h] == ')' ? 1 : 0);
			qmin[h] = Math.min(qmin[h + 1], qsum[h]);
			qmax[h] = Math.max(qmax[h + 1], qsum[h]);
			int ans = -1;
			if (pmin[h] >= 0 && qmin[h + 1] >= 0 && psum[h] == qsum[h + 1])
				ans = Math.max(pmax[h], qmax[h + 1]);
			print(ans + " ");
		}
		println();
	}
}
