from decimal import Decimal, getcontext
getcontext().prec = 123456
o = input().split(' ')
a = Decimal(o[0])
b = Decimal(o[1])
c = Decimal(o[2])
d = str(a / b).split('.')
if len(d) == 2:
	d = d[1]
	for (p, q) in enumerate(d):
		if int(q) == int(c):
			if p == 123455:
				print(-1)
			else:
				print(p + 1)
			exit()
	if int(c) == 0 and len(d) < 123456:
		print(len(d) + 1)
		exit()
	print(-1)
	exit()
elif c == 0:
	print(1)
else:
	print(-1)
