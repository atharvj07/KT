import java.io.*;
import java.util.*;

@SuppressWarnings("unchecked")
public class Problem_E {
	public static void main(String[] args) {
		InputReader in = new InputReader();
		StringBuilder out = new StringBuilder();

		int T = in.nextInt();
		while (T-- > 0) {
			int N = in.nextInt();
			int M = in.nextInt();
			List<Integer>[] G = new List[N + 1];
			for (int i = 1; i <= N; i++) {
				G[i] = new ArrayList<>();
			}
			while (M-- > 0) {
				int x = in.nextInt();
				int y = in.nextInt();
				G[y].add(x);
			}
			int[] V = new int[N + 1];
			for (int i = 1; i <= N; i++) {
				for (int u : G[i]) {
					V[i] = Math.max(V[i], V[u] + 1);
				}
				if (V[i] == 2) {
					V[i] = -1;
				}
			}
			int k = 0;
			for (int i = 1; i <= N; i++) {
				if (V[i] == -1) {
					k++;
				}
			}
			out.append(k).append('\n');
			if (k == 0) {
				out.append('\n');
				continue;
			}
			for (int i = 1; i <= N; i++) {
				if (V[i] == -1) {
					out.append(i).append(' ');
				}
			}
			out.setCharAt(out.length() - 1, '\n');
		}

		System.out.print(out);
	}

	static class InputReader {
		public BufferedReader reader;
		public StringTokenizer st;

		public InputReader() {
			reader = new BufferedReader(new InputStreamReader(System.in));
		}

		public String next() {
			while (st == null || !st.hasMoreTokens()) {
				st = new StringTokenizer(nextLine());
			}
			return st.nextToken();
		}

		public String nextLine() {
			try {
				return reader.readLine();
			} catch (IOException e) {
				e.printStackTrace();
			}
			return null;
		}

		public int nextInt() {
			return Integer.parseInt(next());
		}
	}
}
