t = int(input())
for i in range(t):
	s1 = input()
	s2 = input()
	l1 = []
	if len(s1) >= len(s2):
		i = 0
		j = 0
		count = 0
		while j < len(s2):
			if i == j:
				if s1.find(s2[i]) >= 0:
					count += 1
					j += 1
				else:
					l1.append(count)
					count = 0
					i += 1
					j += 1
			elif s1.find(s2[i:j + 1]) >= 0:
				count += 1
				j += 1
			else:
				l1.append(count)
				count = 0
				i = i + 1
				j = i
	else:
		i = 0
		j = 0
		count = 0
		while j < len(s1):
			if i == j:
				if s2.find(s1[i]) >= 0:
					count += 1
					j += 1
				else:
					l1.append(count)
					count = 0
					i += 1
					j += 1
			elif s2.find(s1[i:j + 1]) >= 0:
				count += 1
				j += 1
			else:
				l1.append(count)
				count = 0
				i = i + 1
				j = i
	if count != 0:
		l1.append(count)
	print(len(s1) + len(s2) - 2 * max(l1))
