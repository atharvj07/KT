#include <bits/stdc++.h>
using namespace std;
vector<pair<int, int>> tree[100010];
int dp[100010][2];
int k;
void dfs(int u, int fa) {
  vector<pair<int, int>> vec;
  for (auto o : tree[u]) {
    if (o.first != fa) {
      dfs(o.first, u);
      vec.push_back({dp[o.first][1] + o.second, dp[o.first][0] + o.second});
    }
  }
  sort(vec.begin(), vec.end(), greater<pair<int, int>>());
  int ksum = 0, kksum = 0;
  for (int i = 0; i < k - 1 && i < vec.size(); i++) ksum += vec[i].first;
  for (int i = 0; i < k && i < vec.size(); i++) kksum += vec[i].first;
  dp[u][1] = ksum;
  dp[u][0] = kksum;
  ksum = 0;
  for (int i = 0; i < vec.size(); i++) {
    if (i >= k)
      dp[u][0] = max(dp[u][0], ksum + max(vec[i].first, vec[i].second));
    else
      dp[u][0] = max(dp[u][0], kksum - vec[i].first + vec[i].second);
  }
}
int main() {
  int n, u, v, c;
  cin >> n >> k;
  for (int i = 1; i < n; i++) {
    cin >> u >> v >> c;
    tree[u].push_back({v, c});
    tree[v].push_back({u, c});
  }
  dfs(0, -1);
  cout << max(dp[0][1], dp[0][0]);
  return 0;
}
