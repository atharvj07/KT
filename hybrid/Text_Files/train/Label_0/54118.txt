import java.io.*;
import java.util.*;
public class Solution{
	public static void main(String args[]){
		Scanner sc=new Scanner(System.in);
		int t=sc.nextInt();
		while(t-->0){
			int n=sc.nextInt();
			String s=sc.next();
			int idx=0;
			int cnt=0;
			for(int i=0;i<n;i++){
				if(s.charAt(i)=='8'){
					cnt++;
					idx=i;
					break;
				}
			}
			if(n-idx<11||cnt==0)
				System.out.println("NO");
			else
				System.out.println("YES");
		}
		
	}
	public static String foo(int n,int m,int[] a,int[] b){
		int[] freq=new int[300001];
		int[] x=new int[2];
		x[0]=a[0];
		x[1]=b[0];
		for(int xi:x){
			int cnt=0;
			for(int i=1;i<m;i++){
				if(a[i]!=xi && b[i]!=xi){
					freq[a[i]]++;
					freq[b[i]]++;
					cnt++;
				}
			}
			System.out.println(cnt+" "+freq[2]+" "+freq[3]+" "+freq[4]);
			for(int i=1;i<m;i++){
				if(freq[a[i]]==cnt || freq[b[i]]==cnt)
					return "YES";
			}
			Arrays.fill(freq,0);
		}
		if(m==1)
			return "YES";
		return "NO";
		
		
	}
}
