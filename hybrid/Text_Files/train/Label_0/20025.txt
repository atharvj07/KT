#include <bits/stdc++.h>
using namespace std;
int a[4];
int main() {
  int n, j = 0;
  char c[1111];
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> c[i];
    switch (c[i]) {
      case 'A':
        a[0]++;
        break;
      case 'G':
        a[1]++;
        break;
      case 'C':
        a[2]++;
        break;
      case 'T':
        a[3]++;
        break;
    }
  }
  for (int i = 0; i < 4; i++) {
    if (a[i] > n / 4) {
      cout << "===";
      return 0;
    }
    a[i] = n / 4 - a[i];
  }
  if (n % 4) {
    cout << "===";
    return 0;
  }
  for (int i = 0; i < n; i++)
    if (c[i] != '?')
      cout << c[i];
    else {
      while (!a[j]) ++j;
      switch (j) {
        case 0:
          cout << 'A';
          break;
        case 1:
          cout << 'G';
          break;
        case 2:
          cout << 'C';
          break;
        case 3:
          cout << 'T';
          break;
      }
      a[j]--;
    }
  return 0;
}
