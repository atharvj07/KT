import math
T = int(input())
for _ in range(T):
	N = int(input())
	S = input()
	B = [[-1, -1, -1] for i in range(N)]
	for i in range(N):
		if S[i] == 'L':
			B[i][0] = i - 1
			if i == 0:
				B[i][0] = N - 1
		else:
			B[i][0] = (i + 1) % N
		b = B[i][0]
		if B[b][1] == -1:
			B[b][1] = i
		else:
			B[b][2] = i
	ans = 0
	C = [0 for i in range(N)]
	for i in range(N):
		val = B[i]
		if val[1] != -1 and val[2] == -1 and (val[0] != val[1]):
			C[i] = 1
	flag = -1
	for i in range(N):
		if C[i] == 0:
			flag = i
			break
	if flag == -1:
		ans = math.ceil(N / 3)
	else:
		D = []
		for i in range(N):
			D.append(C[(flag + i) % N])
		D.append(0)
		for i in range(1, N + 1):
			if D[i] == 1:
				D[i] = D[i] + D[i - 1]
			if D[i] == 0:
				ans = ans + math.ceil(D[i - 1] / 3)
	print(ans)
