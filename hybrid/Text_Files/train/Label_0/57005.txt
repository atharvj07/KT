import java.util.*;
import java.io.*;
public class MaxAndMin
{
	public static void main(String[] args) throws Exception
	{
		FastScanner in = new FastScanner(System.in);
		
		int n = in.nextInt();
		int m = in.nextInt();
		
		int sx = in.nextInt();
		int sy = in.nextInt();
		
		HashSet<Vector> hs = new HashSet<Vector>();
		long mx = 0;
		long my = 0;
		for(int x = 0; x < n + m; x++)
		{
			Vector current = new Vector(in.nextInt(), in.nextInt(), x);
			
			mx = Math.max(mx, current.x);
			my = Math.max(my, current.y);

			hs.add(current);
		}
		
		hs.add(new Vector(mx, 0, n + m));
		hs.add(new Vector(0, my, n + m + 1));
		hs.add(new Vector(0, 0, n + m + 2));
		
		Vector[] pts = new Vector[hs.size()];
		int index = 0;
		for(Vector v : hs)
		{
			pts[index] = v;
			index++;
		}
		
		Arrays.sort(pts);
		
		Vector[] hull = convexHull(pts);
		
		boolean result = false;
		for(Vector v : hull)
		{
			result |= (v.index < n);
		}
		
		System.out.println(result ? "Max" : "Min");
	}
	
	public static Vector[] convexHull(Vector[] p)
	{
		int k = 0;
		Vector[] h = new Vector[p.length*2];
		for(int i = 0; i<p.length; h[k++] = p[i++])
			for(;k>1 && turn(h[k-2], h[k-1], p[i]) > 0; k--);
		for(int i = p.length-2, t = k; i>=0; h[k++] = p[i--])
			for(;k>t && turn(h[k-2], h[k-1], p[i]) > 0; k--);
		return Arrays.copyOf(h, k-1);
	}
	
	public static long turn(Vector a, Vector b, Vector c)
	{
		return (b.x-a.x)*(c.y-a.y) - (b.y-a.y)*(c.x-a.x);
	}
	
	static class Vector implements Comparable<Vector>
	{
		long x;
		long y;
		int index;
		
		public Vector(long x, long y, int i)
		{
			this.x = x;
			this.y = y;
			index = i;
		}
		
		public int compareTo(Vector o)
		{
			if(x != o.x)
			{
				return Long.compare(x, o.x);
			}
			else
			{
				return Long.compare(y, o.y);
			}
		}
		
		public boolean equals(Object o)
		{
			Vector v = (Vector)o;
			return x == v.x && y == v.y;
		}
		
		public int hashCode()
		{
			long[] array = {x, y};
			return Arrays.hashCode(array);
		}
	}
	
	static class FastScanner
	{
		BufferedReader br;
		StringTokenizer st;
		
		public FastScanner(InputStream input)
		{
			br = new BufferedReader(new InputStreamReader(input));
			st = new StringTokenizer("");
		}
		
		public String next() throws IOException
		{
			if(st.hasMoreTokens())
			{
				return st.nextToken();
			}
			else
			{
				st = new StringTokenizer(br.readLine());
				return next();
			}
		}
		
		public int nextInt() throws IOException
		{
			return Integer.parseInt(next());
		}
	}
}
