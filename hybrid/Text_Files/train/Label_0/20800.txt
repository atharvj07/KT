#include <cstdio>
#include <cstdint>
#include <vector>
#include <algorithm>
#include <string>

struct cmp {
  size_t k;
  cmp(size_t k): k(k) {}

  bool operator ()(std::string const& s, std::string const& t) const {
    return s.compare(0, k, t, 0, k) < 0;
  }
};

int main() {
  size_t n, m;
  scanf("%zu %zu", &n, &m);

  std::vector<std::string> word(n), wrev(n);
  for (size_t i = 0; i < n; ++i) {
    char buf[256];
    scanf("%s", buf);
    word[i] = buf;
    wrev[i] = std::string(word[i].rbegin(), word[i].rend());
  }

  std::sort(word.begin(), word.end());
  std::sort(wrev.begin(), wrev.end());

  for (size_t i = 0; i < m; ++i) {
    char buf[256];
    scanf("%s", buf);
    std::string s = buf;

    bool rev = false;
    bool fixed = true;
    if (s[0] == '*') {
      std::reverse(s.begin(), s.end());
      rev = true;
      fixed = false;
      s.pop_back();
    } else if (s.back() == '*') {
      fixed = false;
      s.pop_back();
    }

    size_t q = -1;
    for (size_t j = 0; j < s.length(); ++j)
      if (s[j] == '?') q = j;

    std::vector<std::string> const& crit = (rev? wrev:word);
    int res = 0;
    if (fixed) {
      if (q+1 == 0) {
        auto lb = std::lower_bound(crit.begin(), crit.end(), s);
        auto ub = std::upper_bound(crit.begin(), crit.end(), s);
        // fprintf(stderr, "%td:%td\n", lb-crit.begin(), ub-crit.begin());
        res = ub-lb;
      } else {
        for (size_t c = 'a'; c <= 'z'; ++c) {
          s[q] = c;
          auto lb = std::lower_bound(crit.begin(), crit.end(), s);
          auto ub = std::upper_bound(crit.begin(), crit.end(), s);
          // fprintf(stderr, "%td:%td\n", lb-crit.begin(), ub-crit.begin());
          res += ub-lb;
        }
      }
    } else {
      if (q+1 == 0) {
        auto lb = std::lower_bound(crit.begin(), crit.end(), s, cmp(s.length()));
        auto ub = std::upper_bound(crit.begin(), crit.end(), s, cmp(s.length()));
        // fprintf(stderr, "%td:%td\n", lb-crit.begin(), ub-crit.begin());
        res = ub-lb;
      } else {
        for (size_t c = 'a'; c <= 'z'; ++c) {
          s[q] = c;
          auto lb = std::lower_bound(crit.begin(), crit.end(), s, cmp(s.length()));
          auto ub = std::upper_bound(crit.begin(), crit.end(), s, cmp(s.length()));
          // fprintf(stderr, "%td:%td\n", lb-crit.begin(), ub-crit.begin());
          res += ub-lb;
        }
      }
    }

    printf("%d\n", res);
  }
}
