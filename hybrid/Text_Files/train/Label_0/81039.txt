import sys
input = sys.stdin.readline

def solve():
	(a, b, x) = map(int, input().split())
	(a, b) = (max(a, b), min(a, b))
	if x > a:
		return 'NO'
	while b > 0 and a >= x:
		if (a - x) % b == 0:
			return 'YES'
		(a, b) = (b, a % b)
	return 'NO' if a != x else 'YES'
for _ in range(int(input())):
	print(solve())
