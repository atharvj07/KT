q = int(input())
for _ in range(q):
	n = int(input())
	a = [int(a) for a in input().split()]
	len_prog = 0
	for i in range(n):
		dct = {}
		for j in range(i + 1, n):
			name = (a[i] - a[j]) / (i - j)
			if name not in dct:
				dct[name] = {i, j}
			else:
				dct[name].add(j)
		for (k, v) in dct.items():
			if len_prog < len(v):
				len_prog = len(v)
	if n > 1:
		print(n - len_prog)
	else:
		print(0)
