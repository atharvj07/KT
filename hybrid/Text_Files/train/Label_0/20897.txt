const double eps = 1e-7;
bool feq(double x, double y) { return abs(x - y) <= eps; }
bool inrange(ll x, ll t) { return x >= 0 && x < t; }
bool inrange(vll xs, ll t) {
  for (auto x : xs)
    if (!(x >= 0 && x < t)) return false;
  return true;
}
int ceillog2(ll x) {
  int ret = 0;
  x--;
  while (x > 0) {
    ret++;
    x >>= 1;
  }
  return ret;
}
ll rndf(double x) { return (ll)(x + (x >= 0 ? 0.5 : -0.5)); }
ll floorsqrt(ll x) {
  ll m = (ll)sqrt((double)x);
  return m + (m * m <= x ? 0 : -1);
}
ll ceilsqrt(ll x) {
  ll m = (ll)sqrt((double)x);
  return m + (x <= m * m ? 0 : 1);
}
ll rnddiv(ll a, ll b) { return (a / b + (a % b * 2 >= b ? 1 : 0)); }
ll ceildiv(ll a, ll b) { return (a / b + (a % b == 0 ? 0 : 1)); }
ll gcd(ll m, ll n) {
  if (n == 0)
    return m;
  else
    return gcd(n, m % n);
}
ll lcm(ll m, ll n) { return m * n / gcd(m, n); }
int main() {
  int n, m;
  cin >> n >> m;
  vi a(m + 1);
  unordered_map<int, vi> sum_mp, dif_mp;
  for (int i = 1; i <= int(m); i++) {
    cin >> a[i];
    sum_mp[i + a[i]].push_back(i);
    dif_mp[i - a[i]].push_back(i);
  }
  if (n == 1) {
    cout << 0 << endl;
    return 0;
  }
  vi br(m + 1, INT_MIN), bl(m + 1, INT_MAX);
  ll ans = 0;
  for (int j = 1; j <= int(n); j++) {
    int pi, pj;
    stack<int> record;
    pi = 0;
    pj = j;
    while (pi <= m) {
      auto itr =
          lower_bound(sum_mp[pi + pj].begin(), sum_mp[pi + pj].end(), pi);
      if (itr == sum_mp[pi + pj].end()) {
        int k = min(m + 1 - pi, pj - 1);
        pi = m + 1;
        pj -= k;
      } else {
        if (br[*itr] == INT_MIN) {
          record.push(*itr);
          pi = *itr;
          pj = a[*itr] + 1;
        } else {
          pi = m + 1;
          pj = br[*itr];
          break;
        }
      }
    }
    while (record.size()) {
      int x = record.top();
      record.pop();
      br[x] = pj;
    }
    int ll = pj;
    pi = 0;
    pj = j;
    while (pi <= m) {
      auto itr =
          lower_bound(dif_mp[pi - pj].begin(), dif_mp[pi - pj].end(), pi);
      if (itr == dif_mp[pi - pj].end()) {
        int k = min(m + 1 - pi, n - pj);
        pi += m + 1;
        pj += k;
      } else {
        if (bl[*itr] == INT_MAX) {
          record.push(*itr);
          pi = *itr;
          pj = a[*itr] - 1;
        } else {
          pi = m + 1;
          pj = bl[*itr];
          break;
        }
      }
    }
    while (record.size()) {
      int x = record.top();
      record.pop();
      bl[x] = pj;
    }
    int rr = pj;
    ans += rr - ll + 1;
  }
  cout << ans << endl;
}
