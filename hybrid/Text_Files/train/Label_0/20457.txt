typedef vector<double> vec;
typedef vector<vec> mat;

vec gauss_jordan(const mat& A,const vec& b){
  int n=A.size();
  mat B(n,vec(n+1));
  for(int i=0;i<n;i++){
    for(int j=0;j<n;j++){
      B[i][j]=A[i][j];
    }
    B[i][n]=b[i];
  }

  for (int k = 0; k < n-1; k++) {
    for (int i = k + 1; i < n; i++) {
      double d = B[i][k] / B[k][k];
      for (int j = k + 1; j <= n; j++){
        B[i][j] -= B[k][j] * d;
      }
    }
  }

  for (int i = n - 1; i >= 0; i--) {
    double d = B[i][n];
    for (int j = i + 1; j < n; j++){
      d -= B[i][j] * B[j][n];
    }
    B[i][n] = d / B[i][i];
  }

  vec x(n);
  for(int i=0;i<n;i++)x[i]=B[i][n];
  return x;
}

int H,W,N;
vector<vector<int> > ps;
vector<int> crt;
vector<string> sb,tb;
double p;

vector<int> calc(const vector<string>& a){
  vector<int> res;
  int mn=H;
  for(int j=W-1;j>=0;j--){
    int tmp=0;
    for(int i=H-1;i>=0;i--){
      if(tb[i][j]!=a[i][j]){
        break;
      }
      tmp++;
    }
    minch(mn,tmp);
    res.push_back(mn);
  }
  reverse(all(res));
  return res;
}

void predfs(int i,int j){
  if(i==W){
    ps.push_back(crt);
    return ;
  }
  repl(k,j,H+1){
    crt.push_back(k);
    predfs(i+1,k);
    crt.pop_back();
  }
}
void dfs(int i0,int j0,mat& A,vec &b){
  if(i0==W){
    vector<string> crtb;
    crtb.resize(H);
    rep(i,H)crtb[i]=tb[i];
    rep(j,W){
      rep(i,H-crt[j])crtb[i][j]='.';
    }
    int crtidx=lower_bound(all(ps),crt)-ps.begin();
    rep(i,H)rep(j,W)rep(c,2){
      vector<string> tmp(H);
      rep(ni,H)tmp[ni]=crtb[ni];
      rep(ni,i+1)rep(nj,j+1){
        if(c==0) tmp[ni][nj]='W';
        else tmp[ni][nj]='B';
      }
      int nxtidx=lower_bound(all(ps),calc(tmp))-ps.begin();
      A[crtidx][nxtidx]-=p; b[crtidx]+=p*(double)(i+1)*(double)(j+1);
    }
    return ;
  }
  repl(k,j0,H+1){
    crt.push_back(k);
    dfs(i0+1,k,A,b);
    crt.pop_back();
  }
}

int main(){
  cin.tie(0);
  ios::sync_with_stdio(false);

  while(1){
    cin>>H>>W;
    if(H==0)break;

    ps.clear(); p=1.0/(double)(H*W*2.0);
    sb.resize(H); tb.resize(H);

    rep(i,H)cin>>sb[i];
    rep(i,H)cin>>tb[i];

    predfs(0,0);
    N=ps.size();
    mat A(N,vec(N,0)); vec b(N,0.0);
    dfs(0,0,A,b);

    int startidx=lower_bound(all(ps),calc(sb))-ps.begin();
    int endidx=lower_bound(all(ps),calc(tb))-ps.begin();
    rep(i,N)A[i][i]+=1.0;
    rep(j,N)A[endidx][j]=0.0;
    A[endidx][endidx]=1.0; b[endidx]=0.0;
    vec res=gauss_jordan(A,b);
    printf("%.10Lf\n", res[startidx]);
  }

  return 0;
}
