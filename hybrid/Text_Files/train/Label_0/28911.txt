#include <bits/stdc++.h>
long long i, j, n, qc, m;
long long a[200010];
long long q[200010][3];
void printn(long long v) { printf("%I64d", v); }
int main() {
  long long p, l, r, t;
  scanf("%I64d", &n);
  ;
  scanf("%I64d", &qc);
  ;
  scanf("%I64d", &m);
  ;
  for (i = (1); i < (n + 1); i++) scanf("%I64d", &a[i]);
  ;
  for (i = (0); i < (qc); i++) {
    scanf("%I64d", &q[i][0]);
    ;
    scanf("%I64d", &q[i][1]);
    ;
    scanf("%I64d", &q[i][2]);
    ;
  }
  for (i = (0); i < (m); i++) {
    scanf("%I64d", &p);
    ;
    for (j = (0); j < (qc); j++) {
      t = q[qc - j - 1][0];
      l = q[qc - j - 1][1];
      r = q[qc - j - 1][2];
      if (p >= l && p <= r)
        if (t == 1)
          if (p == l)
            p = r;
          else
            p -= 1;
        else
          p = r - (p - l);
    }
    printn(a[p]);
    putchar(' ');
  }
}
