import sys
t = int(input())
for i in range(t):
	(n, k) = sys.stdin.readline().split()
	n = n.lstrip('000000000')
	k = int(k)
	L = []
	for s in n:
		if int(s) not in L:
			L.append(int(s))
	if len(L) <= k:
		print(n)
	else:
		L = L[:k]
		Num = list(map(int, n))
		ind = Num.index(L[-1])
		maxL = max(L)
		bada = False
		i = 0
		while i < len(n):
			if bada:
				Num[i] = 0
			elif Num[i] > maxL:
				bada = True
				while Num[i] + 1 > maxL and i > ind:
					i -= 1
				Num[i] += 1
				if i == ind:
					L[-1] += 1
					if L[-1] in L[:-1]:
						L[-1] = 0
			elif Num[i] not in L:
				bada = True
			i += 1
		L.sort()
		Go = dict()
		ind = 0
		for i in range(L[-1] + 1):
			while i > L[ind]:
				ind += 1
			Go[i] = str(L[ind])
		minL = str(L[0])
		bada = False
		for i in range(len(Num)):
			if bada == True:
				Num[i] = minL
			else:
				if Num[i] not in L:
					bada = True
				Num[i] = Go[Num[i]]
		print(int(''.join(Num)))
