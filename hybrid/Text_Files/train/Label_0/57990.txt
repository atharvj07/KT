import java.util.*;

// B. Serial Time!
// http://codeforces.com/problemset/problem/60/B
public class Week_2 {

    static final int[][] DIRECTIONS = {{-1, 0, 0}, {1, 0, 0}, {0, -1, 0}, {0, 0, 1}, {0, 1, 0}, {0, 0, -1}};

    static int k, n, m;
    static boolean[][][] filled;
    static int result;

    public static void main(String[] args) {
        result = 0;
        Scanner in = new Scanner(System.in);
        k = in.nextInt();
        n = in.nextInt();
        m = in.nextInt();
        in.nextLine();
        // just a weird representation of a graph, but goal is ultimately to find the number of nodes in the
        // component of the starting node
        filled = new boolean[k][n][m];
        for (int l = 0; l < k; l++) {
            in.nextLine();
            for (int r = 0; r < n; r++) {
                String s = in.nextLine();
                for (int c = 0; c < m; c++) {
                    filled[l][r][c] = s.charAt(c) != '.';
                }
            }
        }
        Point start = new Point(0, in.nextInt()-1, in.nextInt()-1);
        getResult(start);
        System.out.println(result);
    }

    static void getResult(Point start) {
        Stack<Point> s = new Stack<>();
        s.push(start);
        while (!s.isEmpty()) {
            Point current = s.pop();
            if (!current.isFilled()) {
                current.fill();
                result++;
                for (int[] direction : DIRECTIONS) {
                    int newL = current.l + direction[0];
                    int newR = current.r + direction[1];
                    int newC = current.c + direction[2];
                    if (inBounds(newL, newR, newC) && !filled[newL][newR][newC]) {
                        s.push(new Point(newL, newR, newC));
                    }
                }
            }
        }
    }

    static boolean inBounds(int l, int r, int c) {
        return 0 <= l && l < k && 0 <= r && r < n && 0 <= c && c < m;
    }

    private static class Point {
        int l, r, c;

        Point(int layer, int row, int column) {
            l = layer;
            r = row;
            c = column;
        }

        void fill() {
            filled[l][r][c] = true;
        }

        boolean isFilled() {
            return filled[l][r][c];
        }
    }


}
