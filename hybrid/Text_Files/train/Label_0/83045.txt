from sys import stdin, stdout, setrecursionlimit
from gc import disable
gets = input
puts = print
input = stdin.readline
print = stdout.write

def main() -> int:
	disable()
	(n, m) = map(int, input().split())
	a = list(map(int, input().split()))
	y = 0
	for i in range(n):
		y += a[i]
		ans = y // m
		y %= m
		print('%i ' % ans)
	return 0
main()
