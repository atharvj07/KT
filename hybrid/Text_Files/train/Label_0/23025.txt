#include <bits/stdc++.h>
using namespace std;
const int maxn = (int)(2e6 + 6);
const long long mod = (long long)(998244353);
int a[maxn], fact[maxn], inv_fact[maxn];
int mul(long long a, long long b) {
  long long ret = (a * b);
  if (ret >= mod) {
    ret %= mod;
  }
  return (int)ret;
}
int add(long long a, long long b) {
  long long ret = a + b;
  if (ret >= mod) {
    ret %= mod;
  }
  return (int)ret;
}
int sub(long long a, long long b) {
  long long ret = a - b;
  if (ret < 0) {
    ret += mod;
  }
  return (int)ret;
}
static int poww(long long a, long long b) {
  long long x = 1, y = a;
  while (b > 0) {
    if (b % 2) {
      x = mul(x, y);
    }
    y = mul(y, y);
    b /= 2;
  }
  return (int)(x % mod);
}
void build() {
  fact[0] = 1;
  for (int i = 1; i < maxn; i++) {
    fact[i] = mul(fact[i - 1], i);
  }
  inv_fact[maxn - 1] = poww(fact[maxn - 1], mod - 2);
  for (int i = maxn - 2; i >= 0; i--) {
    inv_fact[i] = mul(inv_fact[i + 1], i + 1);
  }
}
int C(int n, int r) {
  if (n < r || min(n, r) < 0) {
    return 0;
  }
  int val1 = fact[n], val2 = mul(inv_fact[r], inv_fact[n - r]);
  return mul(val1, val2);
}
void print(int* x, int n) {
  for (int i = 0; i < n; i++) {
    cerr << *(x + i) << " ";
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  build();
  int n, m, k;
  cin >> n;
  m = n;
  k = 3;
  long long ans = sub(poww(k, n), k);
  ans = poww(ans, m);
  for (int i = 1; i <= n; i++) {
    long long val = mul(C(n, i), k);
    long long zz = sub(poww(k, n - i), 1);
    val = mul(val, poww(zz, m));
    long long val2 = C(n, i), zz2 = sub(poww(k, i), k);
    val2 = mul(val2, zz2);
    long long qq = poww(k, n - i);
    val2 = mul(val2, poww(qq, m));
    if (i > 1) {
      val = add(val, val2);
    }
    if (i % 2) {
      ans = sub(ans, val);
    } else {
      ans = add(ans, val);
    }
  }
  long long tot = n * 1ll * n;
  long long res = sub(poww(3, tot), ans);
  cout << res << endl;
  return 0;
}
