def func():
	n = int(input())
	l = list(map(int, input().split()))
	pois = 0
	health = 0
	neg_pos = []
	for potion in l:
		if potion >= 0:
			pois += 1
			health += potion
		elif potion + health >= 0:
			pois += 1
			health += potion
			neg_pos.append(potion)
		elif neg_pos:
			mn = min(neg_pos)
			if potion > mn:
				health = health - mn + potion
				neg_pos.remove(mn)
				neg_pos.append(potion)
	return pois
t = 1
for i in range(t):
	print(func())
