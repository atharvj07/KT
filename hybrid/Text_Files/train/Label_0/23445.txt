#include <bits/stdc++.h>
using namespace std;
vector<int> v[100000];
int p[100000];
int s = 1;
int f(int i, int l) {
  for (int j = 0; j < v[i].size(); j++) {
    if (p[v[i][j]] == l) {
      s = 0;
      return 0;
    } else {
      if (p[v[i][j]] == -1) {
        p[v[i][j]] = 1 - l;
        f(v[i][j], 1 - l);
      }
    }
  }
}
int main() {
  int n, m;
  cin >> n >> m;
  int x, y;
  vector<pair<int, int> > t;
  for (int i = 0; i < m; i++) {
    cin >> x >> y;
    x--;
    y--;
    t.push_back({x, y});
    v[x].push_back(y);
    v[y].push_back(x);
  }
  for (int i = 0; i < n; i++) p[i] = -1;
  for (int i = 0; i < n; i++) {
    if (p[i] == -1) {
      p[i] = 0;
      f(i, 0);
      if (s == 0) {
        cout << "-1";
        return 0;
      }
    }
  }
  vector<int> o[2];
  for (int i = 0; i < n; i++) {
    o[p[i]].push_back(i + 1);
  }
  cout << o[0].size() << endl;
  for (int i = 0; i < o[0].size(); i++) {
    cout << o[0][i] << " ";
  }
  cout << endl;
  cout << o[1].size() << endl;
  for (int i = 0; i < o[1].size(); i++) {
    cout << o[1][i] << " ";
  }
}
