import java.util.Scanner;

public class TestJava {
    public static  void main(String[] args){
        Scanner sc = new Scanner(System.in);
        String w = sc.next();

        for(char c : w.toLowerCase().toCharArray()){
            if(c != 'a' && c!='o' && c!='u' && c!='y' && c!='e' &&  c !='i'  ){
                System.out.print('.');
                System.out.print(c);
            }
        }
    }
}
