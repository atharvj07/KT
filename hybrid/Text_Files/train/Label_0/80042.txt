T = int(input())
for _ in range(T):
	x = input()
	y = input()
	x = list(reversed(x))
	y = list(reversed(y))
	first_one_in_y = None
	for i in range(len(y)):
		if y[i] == '1':
			first_one_in_y = i
			break
	if first_one_in_y is None:
		print(0)
		continue
	first_one_in_x = None
	for j in range(first_one_in_y, len(x)):
		if x[j] == '1':
			first_one_in_x = j
			break
	if first_one_in_x is None:
		print(0)
	else:
		print(first_one_in_x - first_one_in_y)
