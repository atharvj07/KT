#include <bits/stdc++.h>
using namespace std;
int n, m, x, q;
vector<string> v;
string t;
vector<int> a;
bool y[256];
void hz(int r, int c) {
  int l;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
      if (v[i][j] != 'S') {
        a[v[i][j] - 'a'] =
            min(a[v[i][j] - 'a'], (r - i) * (r - i) + (c - j) * (c - j));
      }
}
int main() {
  cin >> n >> m >> x;
  x *= x;
  memset(y, false, sizeof(y));
  v.resize(n);
  a.resize(26, 1 << 30);
  for (int i = 0; i < n; i++) cin >> v[i];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++) {
      if (v[i][j] == 'S') hz(i, j);
      y[v[i][j]] = true;
    }
  cin >> q >> t;
  int z = 0;
  for (int i = 0; i < q; i++)
    if (t[i] >= 'A' && t[i] <= 'Z') {
      if (a[t[i] - 'A'] > x) {
        if (!y[tolower(t[i])] || !y['S']) {
          cout << -1;
          return 0;
        }
        z++;
      }
    } else {
      if (!y[t[i]]) {
        cout << -1;
        return 0;
      }
    }
  cout << z;
}
