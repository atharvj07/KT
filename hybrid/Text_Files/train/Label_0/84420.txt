x = int(input())
for i in range(x):
	y = int(input())
	A = input().split()
	B = [0] * y
	maxi = 0
	if y == 1:
		print('0')
	else:
		for j in range(y):
			B[j] = int(A[j])
		for j in range(y - 1):
			C = []
			D = []
			for k in range(j + 1, y):
				a = (B[k] - B[j]) / (k - j)
				C.append(float(a))
			C.sort()
			max_count = 1
			curr_count = 1
			for i in range(1, len(C)):
				if C[i] == C[i - 1]:
					curr_count += 1
				else:
					curr_count = 1
				if curr_count > max_count:
					max_count = curr_count
			if max_count + 1 > maxi:
				maxi = max_count + 1
		print(len(A) - maxi)
