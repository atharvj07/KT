#include <bits/stdc++.h>
using namespace std;
const int M = 105;
const double eps = 1e-9;
int n;
int X[M], Y[M];
int dis[M][M];
int sqr(int x) { return x * x; }
long long cal(int x, int y, int z) {
  return (long long)dis[x][y] + dis[x][z] - dis[y][z];
}
double ans;
void check(double x, double y, double dis) {
  if (dis < ans - eps) return;
  for (int i = 1; i <= n; i++)
    if (sqrt((X[i] - x) * (X[i] - x) + (Y[i] - y) * (Y[i] - y)) < dis - eps)
      return;
  ans = dis;
}
int main() {
  ans = -1;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d %d", &X[i], &Y[i]);
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      dis[i][j] = dis[j][i] = sqr(X[i] - X[j]) + sqr(Y[i] - Y[j]);
  for (int i = 1; i <= n; i++)
    for (int j = i + 1; j <= n; j++)
      for (int k = j + 1; k <= n; k++) {
        if (cal(i, j, k) <= 0 || cal(j, k, i) <= 0 || cal(k, i, j) <= 0)
          continue;
        static int x1, x2, x3, y1, y2, y3;
        static double k12, k13, b12, b13, xx, yy;
        x1 = X[i], y1 = Y[i];
        x2 = X[j], y2 = Y[j];
        x3 = X[k], y3 = Y[k];
        if (y1 == y2) swap(x1, x3), swap(y1, y3);
        if (y1 == y3) swap(x1, x2), swap(y1, y2);
        k12 = -1.0 * (x2 - x1) / (y2 - y1);
        b12 = 0.5 * (y1 + y2) - k12 * 0.5 * (x1 + x2);
        k13 = -1.0 * (x3 - x1) / (y3 - y1);
        b13 = 0.5 * (y1 + y3) - k13 * 0.5 * (x1 + x3);
        if (fabs(k12 - k13) > eps) {
          xx = (b13 - b12) / (k12 - k13);
          yy = xx * k12 + b12;
          check(xx, yy, sqrt((xx - x1) * (xx - x1) + (yy - y1) * (yy - y1)));
        }
      }
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (i != j)
        for (int k = 1; k <= n; k++)
          if (i != k && j != k)
            for (int l = 1; l <= n; l++)
              if (i != l && j != l && k != l)
                if (dis[i][j] == dis[l][k] && dis[j][k] == dis[i][l] &&
                    dis[i][j] == dis[i][k] - dis[j][k])
                  check(0.5 * (X[i] + X[k]), 0.5 * (Y[i] + Y[k]),
                        0.5 * sqrt(dis[i][k]));
  if (ans < 0)
    puts("-1");
  else
    printf("%.6f\n", ans);
  return 0;
}
