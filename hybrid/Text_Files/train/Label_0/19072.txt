#include <bits/stdc++.h>
using namespace std;
const int N_MAX = 1e5 + 10;
int arr[N_MAX];
int main() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", arr + i);
  int x, y;
  scanf("%d%d", &x, &y);
  sort(arr, arr + n);
  n = unique(arr, arr + n) - arr;
  int ptr = n - 1, ans = 0;
  while (x != y) {
    while (ptr >= 0 && x - x % arr[ptr] < y) ptr--;
    if (ptr < 0) {
      ans += (x - y);
      x = y;
    } else {
      int tmp = x - 1;
      for (int i = 0; i <= ptr; i++) {
        int div = x - x % arr[i];
        tmp = (div >= y) ? min(tmp, div) : tmp;
      }
      x = tmp;
      ans++;
    }
  }
  printf("%d\n", ans);
  return 0;
}
