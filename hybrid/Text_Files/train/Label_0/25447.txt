#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int f[N][13][16], ans, n, m, k;
inline void mo(int &p, const int q) {
  p += q, p = (p >= 1000000007 ? p - 1000000007 : p);
}
int main() {
  scanf("%d%d%d", &n, &k, &m), f[0][0][0] = 1;
  const int lim = (1 << m) - 1;
  register int i, j, s, nows;
  for (i = 0; i < n; ++i)
    for (j = 0; j <= k; ++j)
      for (s = 0; s <= lim; ++s)
        nows = (s << 1) & lim, mo(f[i + 1][j][nows], f[i][j][s]),
        j < k ? f[i + 1][j + 1][nows | 1] =
                    (f[i + 1][j + 1][nows | 1] +
                     (__builtin_popcount(s) + 1ll) * f[i][j][s]) %
                    1000000007
              : 0;
  for (s = 0; s <= lim; ++s) mo(ans, f[n][k][s]);
  printf("%d", ans);
  return 0;
}
