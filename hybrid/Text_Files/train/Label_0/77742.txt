n = int(input())
starts = dict()
ends = dict()
start_points = set()
for i in range(n):
	(st, en) = map(int, input().split())
	if st == 0:
		start_points.add(i + 1)
	starts[i + 1] = st
	ends[i + 1] = en
lists = []
it = iter(start_points)
x = next(it)
while ends[x] != 0:
	x = ends[x]
for st in it:
	starts[st] = x
	ends[x] = st
	while ends[x] != 0:
		x = ends[x]
for i in range(1, n + 1):
	print(starts[i], ends[i], sep=' ')
