#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int tt;
  cin >> tt;
  while (tt--) {
    int n;
    cin >> n;
    vector<int> p;
    for (int i = 0; i < n; ++i) {
      int u;
      cin >> u;
      p.push_back(u / 100);
    }
    int x, a, y, b;
    cin >> x >> a >> y >> b;
    if (y > x) {
      swap(x, y);
      swap(a, b);
    }
    long long k;
    cin >> k;
    sort(p.begin(), p.end());
    int l = 0, r = n + 1;
    bool co = 1;
    while (l + 1 < r) {
      int m = l + (r - l) / 2;
      int c1 = 0, c2 = 0, c3 = 0;
      for (int i = 1; i <= m; ++i) {
        if (i % a == 0 && i % b == 0) {
          c1++;
        } else if (i % a == 0) {
          c2++;
        } else if (i % b == 0) {
          c3++;
        }
      }
      long long cost = 0;
      int ptr = n - 1;
      for (int i = 0; i < c1; ++i) {
        if (ptr < 0) break;
        cost += (long long)p[ptr--] * (x + y);
      }
      for (int i = 0; i < c2; ++i) {
        if (ptr < 0) break;
        cost += (long long)p[ptr--] * x;
      }
      for (int i = 0; i < c3; ++i) {
        if (ptr < 0) break;
        cost += (long long)p[ptr--] * y;
      }
      if (cost >= k)
        r = m;
      else
        l = m;
    }
    cout << (r == n + 1 ? -1 : r) << "\n";
  }
  return 0;
}
