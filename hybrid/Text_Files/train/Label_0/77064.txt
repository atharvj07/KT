(n, k) = map(int, input().split(' '))
s = list(map(int, list(input())))
l = s[:k]
flag = 0
f = 0
d = 0
for i in range(n):
	if s[i] < l[i % k]:
		f = 1
	if s[i] > l[i % k] and f != 1:
		d = 1
if d == 1:
	t = k - 1
	while t >= 0 and l[t] == 9:
		t -= 1
	if t == -1:
		flag = 1
		print(len(s))
		for j in range(len(s)):
			print(l[j % k], end='')
	else:
		l[t] += 1
		flag = 1
		for j in range(t + 1, len(l)):
			l[j] = 0
		print(len(s))
		for j in range(len(s)):
			print(l[j % k], end='')
if flag == 0:
	print(len(s))
	for j in range(len(s)):
		print(l[j % k], end='')
