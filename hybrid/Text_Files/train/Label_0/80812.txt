from sys import stdin
t = int(stdin.readline().strip())
for i in range(t):
	lst = [int(x) for x in stdin.readline().strip().split()]
	n = lst[0]
	l = lst[1]
	r = lst[2]
	s = lst[3]
	d = r - l + 1
	e = n - d
	if d * (d + 1) // 2 > s:
		print(-1)
		continue
	if n * (n + 1) // 2 - e * (e + 1) // 2 < s:
		print(-1)
		continue
	av = s // d
	missing = s - av * d
	liste = []
	halb = d // 2
	for k in range(halb):
		liste.append(av - halb + k)
	for k in range(halb):
		liste.append(av + k + 1)
	if d % 2 == 1:
		liste.insert(halb, av)
	result = [x + 1 for x in range(n)]
	if d % 2 == 0:
		for k in range((d - 1) // 2, -1, -1):
			if missing > 0:
				liste[k] += 1
				missing -= 1
	for k in range(len(liste) - 1, -1, -1):
		if missing > 0:
			liste[k] += 1
			missing -= 1
		del result[liste[k] - 1]
	for k in range(len(result)):
		if k + 1 == l:
			for elem in liste:
				print(elem, '', end='')
		print(result[k], '', end='')
	if r == n:
		for elem in liste:
			print(elem, '', end='')
	print()
