import sys
input = sys.stdin.readline

def solve():
	n = int(input())
	a = list(map(int, input().split()))
	T = [(-1, -1)] * (2 * n)
	for i in range(n - 1):
		T[i + n] = (a[i], i)
	T[n + n - 1] = (n, n - 1)
	for i in range(n - 1, -1, -1):
		T[i] = max(T[2 * i], T[2 * i + 1])
	dp = [0] * n
	res = 0
	for i in range(n - 2, -1, -1):
		l = i + n
		r = a[i] - 1 + n
		v = (-1, -1)
		while l <= r:
			if l % 2 == 1:
				v = max(v, T[l])
			if r % 2 == 0:
				v = max(v, T[r])
			l = (l + 1) // 2
			r = (r - 1) // 2
		dp[i] = dp[v[1]] + (n - v[1]) + (v[1] - i) - (a[i] - 1 - v[1] + 1)
		res += dp[i]
	print(res)
solve()
