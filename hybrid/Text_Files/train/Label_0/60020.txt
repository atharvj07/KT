import java.util.Arrays;
import java.util.Scanner;


public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    String st = sc.next();
    sc.close();
    // System.out.println(st);
    int[] arr = new int[k];
    for(int i = 0; i < n; i++){
      char ch = st.charAt(i);
      int index = ch - 65;
      arr[index]++;
    }
    Arrays.sort(arr);
    // System.out.println(Arrays.toString(arr));
    System.out.println((arr[0]*k));
  }

}
