#include <bits/stdc++.h>
using namespace std;
#define FNAME "A"
#define ll long long
bool isFile = 0;
void faster() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    if (isFile) {
        freopen(FNAME".in", "r", stdin);
        freopen(FNAME".out", "w", stdout);
    }
}
void solve() {
    long n,m;
    cin>>n>>m;
    string s;
    cin>>s;
    s='0'+s+'0';
    for(int k=1;k<=m;k++){
        string temp=s;
        for(int i=1;i<=n;i++){
            if(temp[i]=='0'){
                if(temp[i+1]=='0'&&temp[i-1]=='1'){
                    s[i]='1';
                }
                if(temp[i-1]=='0'&&temp[i+1]=='1'){
                    s[i]='1';
                }
            }
        }
        if(s==temp){
            break;
        }
    }
    for(int i=1;i<=n;i++){
        cout<<s[i];
    }
    cout<<endl;
}
int main() {
    faster();
    int t = 1;
    cin>>t;
    while(t--) solve();
}
