#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll dp[105][105];
ll f(string x, string y, ll a, ll e, ll s, ll r) {
    //cerr << x << " " << y << endl;
    int n = x.size();
    int m = y.size();
    x = "#" + x;
    y = "#" + y;
    dp[0][0] = 0;
    for(int i = 1; i <= n; i++) dp[i][0] = i * a;
    for(int i = 1; i <= m; i++) dp[0][i] = i * e;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            dp[i][j] = min(dp[i][j-1] + e, min(dp[i-1][j-1] + s + r, dp[i-1][j] + a));
            if(x[i] == y[j]) dp[i][j] = min(dp[i][j], dp[i-1][j-1] + r);
        }
    }
    //cerr << x << " " << y << " " << dp[n][m] << endl;
    return dp[n][m];
}


int main() {
    while(true) {
        string X, Y;
        cin >> X;
        if(X == "#") break;
        cin >> Y;
        swap(X, Y);
        ll A, E, S, R;
        cin >> A >> E >> S >> R;
        ll ans = 1e18;
        for(int i = 0; i <= X.size(); i++) {
            for(int j = 0; j <= Y.size(); j++) {
                string x1 = X.substr(0, i);
                string x2 = X.substr(i, X.size() - i);
                string y1 = Y.substr(0, j);
                string y2 = Y.substr(j, Y.size() - j);
                //f(x1, y2, A, E, S, R);
                ans = min(ans, f(x1, y2, A, E, S, 0) + f(x2, y1, A, E, S, R));
            }
        }
        cout << ans << endl;
    }
}
