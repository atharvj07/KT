import java.util.*;

public class MissileSilos {

	public static void main(String args[]) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		int s = sc.nextInt();
		List<Edge>[] edges = new List[n + 1];

		for (int i = 0; i <= n; i++) {
			List<Edge> temp = new ArrayList<Edge>();
			edges[i] = temp;
		}

		for (int i = 0; i < m; i++) {
			int v = sc.nextInt();
			int u = sc.nextInt();
			int w = sc.nextInt();
			// System.out.println(a + " " + b + " " + c);

			edges[v].add(new Edge(u, w));
			edges[u].add(new Edge(v, w));
		}
		int dist = sc.nextInt();

		int result = findMissile(s, dist, n, edges);
		System.out.println(result);
		sc.close();
	}

	public static int findMissile(int s, int dist, int n, List<Edge>[] edges) {
		int distance[] = new int[n + 1];
		int result = 0;
		PriorityQueue<Edge> pq = new PriorityQueue<Edge>();
		Map<String, Integer> m = new HashMap<String, Integer>();

		Arrays.fill(distance, Integer.MAX_VALUE);
		distance[s] = 0;
		pq.add(new Edge(s, 0));

		while (!pq.isEmpty() && pq.peek().weight <= dist) {
			Edge current = pq.poll();
			int dest = current.dest;
			int current_weight = current.weight;
			if (current_weight != distance[dest])
				continue;

			if (current_weight == dist)
				result++;

			for (Edge e : edges[dest]) {
				int next = e.dest;
				int next_weight = e.weight;
				if (dist > current_weight && dist < next_weight + current_weight) {
					int high = Math.max(dest, next);
					int low = Math.min(dest, next);
					String key = low + " " + high;
					if (dest == low) {
						int fromLow = dist - current_weight;
						if (m.get(key) != null) {
							int fromLow2 = m.get(key);
							if (fromLow2 <= fromLow)
								result--;
							if (fromLow2 >= fromLow)
								result++;
						} else {
							if (distance[next] == Integer.MAX_VALUE || distance[next] + fromLow >= dist) {
								m.put(key, fromLow);
								result++;
							}
						}
					} else {
						int fromLow = next_weight - (dist - current_weight);
						if (m.get(key) != null) {
							int fromLow2 = m.get(key);
							if (fromLow2 >= fromLow)
								result--;
							if (fromLow2 <= fromLow)
								result++;
						} else {
							if (distance[next] == Integer.MAX_VALUE || distance[next] + dist - current_weight >= dist) {
								m.put(key, fromLow);
								result++;
							}
						}
					}
				}
				if (distance[next] > current_weight + next_weight) {
					distance[next] = current_weight + next_weight;
					pq.add(new Edge(next, current_weight + next_weight));
				}
			}
		}
		return result;
	}

}

class Edge implements Comparable<Edge> {
	int dest;
	int weight;

	public Edge(int t, int w) {
		dest = t;
		weight = w;
	}

	public int compareTo(Edge e) {
		return this.weight - e.weight;
	}
}
