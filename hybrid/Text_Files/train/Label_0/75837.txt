(n, k) = map(int, input().split())
h = list(map(int, input().split()))
j = 0
m = 0
sum = 0
s = float('inf')
if n == k:
	print(1)
else:
	for i in range(n):
		sum += h[i]
		m += 1
		if m == k:
			a = s
			s = min(s, sum)
			sum -= h[j]
			j += 1
			m -= 1
			if a != s:
				index = j
		else:
			continue
	print(index)
