#include<stdio.h>
#include<algorithm>
#include<math.h>
using namespace std;
int UF[100];
int x[100][4];
int y[100][4];
int FIND(int a){
	if(UF[a]<0)return a;
	return UF[a]=FIND(UF[a]);
}
void UNION(int a,int b){
	a=FIND(a);
	b=FIND(b);
	if(a==b)return;
	UF[a]+=UF[b];
	UF[b]=a;
}
bool inclusive(int ax,int ay,int bx,int by,int cx,int cy,int px,int py){
	bool ok1=true;
	bool ok2=true;
	if((bx-ax)*(py-ay)-(px-ax)*(by-ay)<0)ok1=false;
	if((bx-ax)*(py-ay)-(px-ax)*(by-ay)>0)ok2=false;
	if((cx-bx)*(py-by)-(px-bx)*(cy-by)<0)ok1=false;
	if((cx-bx)*(py-by)-(px-bx)*(cy-by)>0)ok2=false;
	if((ax-cx)*(py-cy)-(px-cx)*(ay-cy)<0)ok1=false;
	if((ax-cx)*(py-cy)-(px-cx)*(ay-cy)>0)ok2=false;
	return ok1||ok2;
}

bool intersect(int ax,int ay,int bx,int by,int cx,int cy,int dx,int dy){
	long long ta=(cx-dx)*(ay-cy)+(cy-dy)*(cx-ax);
	long long tb=(cx-dx)*(by-cy)+(cy-dy)*(cx-bx);
	long long tc=(ax-bx)*(cy-ay)+(ay-by)*(ax-cx);
	long long td=(ax-bx)*(dy-ay)+(ay-by)*(ax-dx);
	if(ta*tb<0LL&&tc*td<0LL)return true;
	else return false;
}

int main(){
	int a;
	while(scanf("%d",&a),a){
		while(a--){
			int b;
			scanf("%d",&b);
			for(int i=0;i<b;i++){
				for(int j=0;j<4;j++)scanf("%d%d",&x[i][j],&y[i][j]);
			}
			for(int i=0;i<b;i++)UF[i]=-1;
			for(int i=0;i<b;i++){
				for(int j=i+1;j<b;j++){
					
					bool ok=false;
					for(int k=0;k<4;k++){
						if(inclusive(x[i][0],y[i][0],x[i][1],y[i][1],x[i][2],y[i][2],x[j][k],y[j][k]))ok=true;
						if(inclusive(x[i][0],y[i][0],x[i][3],y[i][3],x[i][2],y[i][2],x[j][k],y[j][k]))ok=true;
						if(inclusive(x[j][0],y[j][0],x[j][1],y[j][1],x[j][2],y[j][2],x[i][k],y[i][k]))ok=true;
						if(inclusive(x[j][0],y[j][0],x[j][3],y[j][3],x[j][2],y[j][2],x[i][k],y[i][k]))ok=true;
					}
					for(int k=0;k<4;k++){
						for(int l=0;l<4;l++)if(intersect(x[i][k],y[i][k],x[i][(k+1)%4],y[i][(k+1)%4],x[j][l],y[j][l],x[j][(l+1)%4],y[j][(l+1)%4]))ok=true;
					}
					if(ok)UNION(i,j);
				}
			}
			int ret=0;
			for(int i=0;i<b;i++)if(UF[i]<0)ret++;
			printf("%d\n",ret);
		}
	}
}
