#include<bits/stdc++.h>
#define ll long long
#define pii pair<int,int>
#define mp make_pair
#define fst first
#define snd second
#define pb push_back
using namespace std;
const int N=310;
const ll P=1e9+7;
vector<pii>q[N];
ll dp[N][N][N];
int main(){
	int n,m;
	scanf("%d%d",&n,&m);
	while(m--){
		int x,y,z;
		scanf("%d%d%d",&x,&y,&z);
		q[y].pb(mp(x,z));
	}
	dp[1][0][0]=3;//注意dp[1][0][0]有3种取色方案
	for(int i=1;i<=n;++i){
		for(int j=0;j<q[i].size();++j){
			int x=q[i][j].fst,z=q[i][j].snd;
			for(int k=0;k<i;++k)
				for(int p=0;p<=max(0,k-1);++p){
					if(z==1&&k>=x)dp[i][k][p]=-1;
					if(z==2&&(k<x||p>=x))dp[i][k][p]=-1;
					if(z==3&&p<x)dp[i][k][p]=-1;
			}
		}
		if(i==n)break;
		for(int j=0;j<i;++j)
			for(int k=0;k<=max(0,j-1);++k){
				if(dp[i][j][k]==-1)continue;
				(dp[i+1][j][k]+=dp[i][j][k])%=P;
			(dp[i+1][i][k]+=dp[i][j][k])%=P;
				(dp[i+1][i][j]+=dp[i][j][k])%=P;
		}
	}
	ll ans=0;
	for(int j=0;j<n;++j)
		for(int k=0;k<=max(0,j-1);++k)
			if(dp[n][j][k]!=-1)
				(ans+=dp[n][j][k])%=P;
	printf("%lld\n",ans);
	return 0;
}
