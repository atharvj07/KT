#include <bits/stdc++.h>
using namespace std;
const int MAX = 2e9;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    map<int, vector<int>> mp;
    for (int i = 0; i < n; i++) {
      int k;
      cin >> k;
      mp[k].push_back(i + 1);
    }
    vector<int> ans(n + 1);
    for (int i = 1; i <= n; i++) ans[i] = -1;
    int hi = n;
    for (auto k : mp) {
      vector<int> v = k.second;
      int maxt = (int)-1e9;
      for (int i = 0; i < v.size(); i++) {
        if (i == 0) maxt = max(maxt, v[i]);
        if (i == v.size() - 1) maxt = max(maxt, n + 1 - v[i]);
        if (i != v.size() - 1) {
          maxt = max(maxt, v[i + 1] - v[i]);
        }
      }
      bool isf = 0;
      for (int i = maxt; i <= hi; i++) {
        ans[i] = k.first;
        isf = 1;
      }
      if (isf) hi = maxt - 1;
    }
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << "\n";
  }
}
