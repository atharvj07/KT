import sys

def answer(n, s):
	num_changes = 0
	choices = {'R', 'G', 'B'}
	ans = [s[0]]
	for i in range(1, n):
		if s[i] != ans[-1]:
			ans.append(s[i])
		else:
			temp = {'R', 'G', 'B'}
			temp.discard(s[i])
			if i < n - 1:
				temp.discard(s[i + 1])
			ans.append(temp.pop())
			num_changes += 1
	print(num_changes)
	print(''.join(ans))
	return

def main():
	n = int(sys.stdin.readline())
	s = sys.stdin.readline().rstrip()
	answer(n, s)
	return
main()
