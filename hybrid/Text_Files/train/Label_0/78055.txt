def weight(state):
	state = state.rstrip('=')
	return 0 if len(state) < 1 else sum((int(state[i]) * (i + 1) if state[i] != '=' else 0 for i in range(len(state))))

def main():
	[left, right] = input().split('^')
	left = left[::-1]
	(weight_left, weight_right) = (weight(left), weight(right))
	print('balance' if weight_left == weight_right else 'left' if weight_left > weight_right else 'right')
main()
