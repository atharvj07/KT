#include <bits/stdc++.h>
using namespace std;
const int INF = 1 << 30;
const double EPS = 1e-7;
long long data[200005];
long long query(int x) {
  long long sum = 0;
  for (; x > 0; x -= (x & -x)) sum += data[x];
  return sum;
}
void update(int x, int val) {
  for (; x < 200005; x += (x & -x)) data[x] += val;
}
void update(int x, int y, int k) {
  update(x, k);
  update(y + 1, -k);
}
int main() {
  ios::sync_with_stdio(false);
  int n;
  cin >> n;
  double p = 0;
  int cant = 1;
  for (int i = 0; i < n; i++) {
    int t;
    cin >> t;
    if (t == 1) {
      int a, x;
      cin >> a >> x;
      update(1, a, x);
      p += (1.0 * a * x) / cant;
    } else if (t == 2) {
      int k;
      cin >> k;
      update(cant + 1, cant + 1, k);
      p = (p * cant + k) / (cant + 1);
      cant++;
    } else {
      long long val = query(cant);
      p = (p * cant - val) / (cant - 1);
      update(cant, cant, -val);
      cant--;
    }
    cout.precision(6);
    cout << fixed << p << endl;
  }
  return 0;
}
