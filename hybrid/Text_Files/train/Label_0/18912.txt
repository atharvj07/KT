#include <bits/stdc++.h>
inline int in() {
  int x = 0;
  char c = getchar();
  bool f = false;
  while (c < '0' || c > '9') f |= c == '-', c = getchar();
  while (c >= '0' && c <= '9')
    x = (x << 1) + (x << 3) + (c ^ 48), c = getchar();
  return f ? -x : x;
}
const int N = 1e5 + 5;
struct answer {
  int a, b;
};
std::vector<answer> ans;
int n;
char a[N], b[N], r[N];
inline void good_game() {
  puts("-1");
  exit(0);
}
inline void sub(const int i);
inline void add(const int i) {
  if (i == n) good_game();
  if (a[i + 1] == '9') sub(i + 1);
  ans.push_back((answer){i, 1});
  ++a[i], ++a[i + 1];
}
inline void sub(const int i) {
  if (i == n) good_game();
  if (a[i + 1] == '0') add(i + 1);
  ans.push_back((answer){i, -1});
  --a[i], --a[i + 1];
}
inline void calc() {
  for (int i = 1; i < n; ++i) {
    while (a[i] < b[i] && ans.size() < (int)1e5) add(i);
    while (a[i] > b[i] && ans.size() < (int)1e5) sub(i);
  }
}
inline long long work() {
  long long tot = 0;
  static int x[N], y[N];
  for (int i = 1; i <= n; ++i) {
    x[i] = a[i] - '0';
    y[i] = b[i] - '0';
  }
  for (int i = 1, d; i < n; ++i) {
    if (x[i] < 0) {
      d = -x[i];
      x[i] += d;
      x[i + 1] += d;
      tot += d;
    }
    if (x[i] > 9) {
      d = x[i] - 9;
      x[i] -= d;
      x[i + 1] -= d;
      tot += d;
    }
    while (x[i] < y[i]) {
      ++x[i];
      ++x[i + 1];
      ++tot;
    }
    while (x[i] > y[i]) {
      --x[i];
      --x[i + 1];
      ++tot;
    }
  }
  if (x[n] != y[n]) good_game();
  return tot;
}
int main() {
  n = in();
  scanf("%s %s", a + 1, b + 1);
  strcpy(r + 1, a + 1);
  calc();
  strcpy(a + 1, r + 1);
  printf("%lld\n", work());
  for (int i = 0; i < (int)ans.size() && i < (int)1e5; ++i)
    printf("%d %d\n", ans[i].a, ans[i].b);
  return 0;
}
