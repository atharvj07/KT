#include <bits/stdc++.h>
#pragma GCC optimize("O3")
using namespace std;
const int MAX = 1e5 + 5;
const long long MAX2 = 11;
const long long MOD = 998244353;
const long long MOD2 = 1000005329;
const long long INF = 2e18;
const int dr[] = {1, 0, -1, 0, 1, 1, -1, -1, 0};
const int dc[] = {0, 1, 0, -1, 1, -1, 1, -1, 0};
const double pi = acos(-1);
const double EPS = 1e-9;
const int block = 2000;
long long n, m, k, jk[MAX], dp[105][MAX], ans, a, b, x[MAX], z[MAX];
void f(int rem, int l, int r, int optl, int optr) {
  if (l > r) return;
  int mid = l + r >> 1;
  long long opt = INF, c;
  int optid = mid;
  for (int i = optl; i <= mid && i <= optr; ++i) {
    c = x[mid] * (mid - i + 1) - (z[mid] - z[i - 1]);
    if (dp[rem - 1][i - 1] + c < opt) {
      opt = dp[rem - 1][i - 1] + c;
      optid = i;
    }
  }
  dp[rem][mid] = opt;
  f(rem, l, mid - 1, optl, optid);
  f(rem, mid + 1, r, optid, optr);
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m >> k;
  for (int i = 2; i <= n; ++i) cin >> jk[i], jk[i] += jk[i - 1];
  for (int i = 1; i <= m; ++i) {
    cin >> a >> b;
    x[i] = b - jk[a];
  }
  sort(x + 1, x + 1 + m);
  for (int i = 1; i <= m; ++i) z[i] = z[i - 1] + x[i];
  for (int i = 1; i <= m; ++i) dp[1][i] = x[i] * i - z[i];
  for (int i = 2; i <= k; ++i) f(i, i, m, i, m);
  cout << ans + dp[k][m] << '\n';
  return 0;
}
