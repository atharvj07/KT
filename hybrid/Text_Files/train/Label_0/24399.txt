#include <bits/stdc++.h>
using namespace std;
int n;
char st[110];
bool condition1, condition2, condition3, condition4;
int main() {
  gets(st);
  n = strlen(st);
  if (n >= 5) condition1 = true;
  for (int i = 0; i < n; i++) {
    if (st[i] >= 'A' && st[i] <= 'Z') condition2 = true;
    if (st[i] >= 'a' && st[i] <= 'z') condition3 = true;
    if (st[i] >= '0' && st[i] <= '9') condition4 = true;
  }
  if (condition1 && condition2 && condition3 && condition4)
    printf("Correct\n");
  else
    printf("Too weak\n");
  return 0;
}
