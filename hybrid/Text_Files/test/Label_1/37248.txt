n, m = map(int, input().split())
a = list(map(int, input().split()))
cnt = [[0] * (n + 1) for _ in range(m + 1)]
for i in range(m):
    l, r = map(int, input().split())
    cnt[i + 1][l] += 1
    cnt[i + 1][r + 1] -= 1
for i in range(1, m + 1):
    for j in range(1, n + 1):
        cnt[i][j] += cnt[i - 1][j]
        cnt[i][j] *= a[j - 1]
print(max(sum(line) for line in cnt))
