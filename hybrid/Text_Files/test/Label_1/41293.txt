import sys
input = sys.stdin.readline

n, m = map(int, input().split())
s = [input().strip() for _ in range(n)]
a = [[int(x) for x in input().split()] for _ in range(n)]

ans = 0
for j in range(m):
    d = {}
    for i in range(n):
        if s[i][j] not in d:
            d[s[i][j]] = i
            continue
        ans += min(a[i][j], a[d[s[i][j]]][j])
        s[d[s[i][j]]][j] = chr(ord('a')+26)
    
print(ans)
