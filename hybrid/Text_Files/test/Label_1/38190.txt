import bisect
n=int(input())
pos=list()
app=list()
for i in range(n):
    x,a=map(int,input().split())
    bisect.insort(pos,x)
    app.append(a)
pos.append(1e9)
ans=0
l=0
r=n-1
now=0
while l<=r:
    if pos[l]<pos[r]:
        if abs(now)>pos[l]:
            ans-=abs(now)-pos[l]
        now=pos[l]
        ans+=app[l]
        l+=1
    else:
        if abs(now)>pos[r]:
            ans-=abs(now)-pos[r]
        now=pos[r]
        ans+=app[r]
        r-=1
print(ans)
