t = int(input())
s = input()
d = {0:0, 1:1, 2:5, 3:3, 4:4, 5:2, 6:9, 7:7, 8:8, 9:6}
ds = [d[int(x)] for x in s]
cnt = [0 for x in range(10)]
for x in ds:
    cnt[x] += 1

ans = 0
if cnt[t] > 0:
    ans += cnt[t] - 1

if t in [0, 2, 4, 6, 8]:
    t_alt = 5 if t%2==0 else 2
    ans += cnt[t_alt]

if t in [1, 3, 7, 9]:
    t_alt = 9 if t%2==0 else 1
    ans += cnt[t_alt]

if t == 5:
    ans += cnt[2]

print(ans//2)
