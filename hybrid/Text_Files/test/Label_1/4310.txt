#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

int main() {
	int n;
	cin >> n;
	vector<ll> a(n);
	for (int i = 0; i < n; i++) {
		cin >> a[i];
	}
	ll ans1 = 1e18, ans2 = -1e18;
	for (int i = 0; i < n; i++) {
		ans1 = min(ans1, abs(a[i] - a[0]));
		ans2 = max(ans2, abs(a[i] - a[n - 1]));
	}
	for (int i = 0; i < n; i++) {
		cout << ans1 << " " << ans2 << endl;
	}
	return 0;
}
