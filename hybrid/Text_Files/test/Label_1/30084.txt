import java.util.*;
class Main {
    static int n,m,k;
    static int[][] a;
    static int[][] ship;
    static boolean[][] used;
    static int[] dx = { -1, 0, 0, 1 };
    static int[] dy = { 0, 1, -1, 0 };
    static int ans;
    static void dfs(int x,int y) {
        if (used[x][y]) return;
        ans += a[x][y];
        ship[x][y] = 1;
        used[x][y] = true;
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx>=0 && nx<n && ny>=0 && ny<m && !used[nx][ny] && ship[nx][ny] == 0) {
                dfs(nx,ny);
            }
        }
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        a = new int[n][m];
        ship = new int[n][m];
        used = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = in.nextInt();
            }
        }
        int[] x = new int[k];
        int[] y = new int[k];
        for (int i = 0; i < k; i++) {
            x[i] = in.nextInt() - 1;
            y[i] = in.nextInt() - 1;
        }
        ans = 0;
        for (int i = 0; i < k; i++) {
            if (!used[x[i]][y[i]]) {
                dfs(x[i],y[i]);
            }
        }
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (ship[i][j] == 0) System.out.print(".");
                else System.out.print("X");
            }
            System.out.println();
        }
    }
}
