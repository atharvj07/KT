#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, k, a, b, l = 1, p, ans = 1e18;
    cin >> n >> k >> a >> b;
    while (l < 1 << n) l *= 2;

    set<long long> s;
    for (int i = 0; i < k; i++) {
        cin >> p;
        s.insert(p);
    }

    for (int i = 0; i < (1 << n); i++) {
        long long tot = 0;
        for (int j = 0; j < n; j++) {
            if ((1 << j) & i) {
                if (s.find((j + 1) * l - 1 - tot) != s.end()) tot += b; else tot += a;
            }
        }
        ans = min(ans, tot);
    }
    cout << ans << endl;
    return 0;
}
