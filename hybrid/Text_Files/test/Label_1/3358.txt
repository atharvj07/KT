#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<string, vector<int>>> v(n);
  vector<int> pos(1e6 + 1, -1);
  for (int i = 0; i < n; i++) {
    string s;
    int k, x;
    cin >> s >> k;
    v[i].first = s;
    while (k--) {
      cin >> x;
      v[i].second.push_back(x);
      pos[x] = i;
    }
  }
  string ans;
  for (int i = 1; i <= 1e6; i++) {
    if (pos[i] != -1) {
      ans += v[pos[i]].first;
      i += v[pos[i]].first.length() - 1;
    }
  }
  cout << ans << endl;
  return 0;
}
