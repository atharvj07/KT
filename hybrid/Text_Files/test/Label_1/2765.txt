#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  
  std::vector<int> t(n);
  for (int& x : t) {
    std::cin >> x;
  }
  
  std::sort(t.begin(), t.end());
  
  int ans = 0;
  long long time = 0;
  for (int x : t) {
    if (time + x <= x) {
      ans++;
      time += x;
    }
  }
  
  std::cout << ans << '\n';
}
