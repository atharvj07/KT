import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int[][] dp = new int[n][2];
        dp[0][0] = 1;
        dp[0][1] = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < 2; j++) {
                for (int k = 0; k < i; k++) {
                    if (c[i] != c[k] && c[k] != -1 && c[i] != -1 && k < i) {
                        dp[i][j] = (dp[i][j] + dp[k][j ^ (c[i] ^ c[k])]) % 1000000007;
                    }
                }
                if (c[i] == -1) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j] + dp[i - 1][j ^ 1]) % 1000000007;
                }
            }
        }
        System.out.println(dp[n - 1][p]);
    }
}
