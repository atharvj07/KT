#include <bits/stdc++.h>
using namespace std;
string solve(string s, string t) {
    string t0 = t;
    sort(s.begin(), s.end());
    int n = s.size();
    for (int i = 0; i < n;) {
        char c = s[i];
        bool ok = false;
        while (i < n && s[i] == c) {
            if (t.find(c) != string::npos) {
                ok = true;
                t.erase(t.find(c), 1);
            }
            i++;
        }
        if (ok) {
            char c0 = *t.begin();
            t = t0;
            for (int j = 0; j < i; j++) {
                if (s[j] != c) {
                    swap(s[j], s[i - 1]);
                    if (s[j] == c0) {
                        swap(s[j], s[i - 2]);
                        t.erase(t.find(c0), 1);
                    }
                    break;
                }
            }
            break;
        }
    }
    return s;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        string s, t;
        cin >> s >> t;
        cout << solve(s, t) << '\n';
    }
}
