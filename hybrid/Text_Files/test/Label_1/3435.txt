#include <bits/stdc++.h>
#define pb emplace_back
#define mk make_pair
#define fst first
#define scd second
using namespace std;
typedef pair<int, int> pi;
const int N = 1005, mod = 1e9 + 7;
int n, m, b[N][N], Ans, dp[N][3][1 << 12];
int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        b[x][y] = 1;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (b[i][j])
                continue;
            if ((i == 1 || i == n) && (j == 1 || j == n))
                continue;
            if (i == 1) {
                dp[i][0][1 << (j - 1)] = 1;
                int msk = 1 << (j - 2) | 1 << j;
                dp[i][2][msk] = 1;
            }
            if (i == n) {
                dp[i][0][1 << (j - 1)] = 1;
                int msk = 1 << (j - 2) | 1 << j;
                dp[i][1][msk] = 1;
            }
            if (j == 1) {
                dp[i][0][1 << (i - 1)] = 1;
                int msk = 1 << (i - 2) | 1 << i;
                dp[i][2][msk] = 1;
            }
            if (j == n) {
                dp[i][0][1 << (i - 1)] = 1;
                int msk = 1 << (i - 2) | 1 << i;
                dp[i][1][msk] = 1;
            }
        }
    }
    for (int i = 2; i <= n - 1; i++) {
        for (int j = 1; j <= n; j++) {
            for (int msk = 0; msk < (1 << (n - 1)); msk++) {
                bool prv = 0, prv2 = 0;
                if (i == 2) {
                    if (j == 1)
                        prv = 0, prv2 = 0;
                    else
                        prv = dp[i - 1][0][msk & ~(1 << (j - 2))],
                        prv2 = dp[i - 1][2][msk & ~(1 << (j - 2) | 1 << (j - 1))];
                } else if (j == 1)
                    prv = dp[i - 1][0][msk & ~(1 << (i - 2))],
                    prv2 = dp[i - 1][2][msk & ~(1 << (i - 2) | 1 << (i - 1))];
                else
                    prv = dp[i - 1][0][msk & ~(1 << (i - 2) | 1 << (j - 2))],
                    prv2 = dp[i - 1][2][msk & ~(1 << (i - 2) | 1 << (i - 1) | 1 << (j - 2) | 1 << (j - 1))];
                int ntk = msk | 1 << (i - 1);
                dp[i][0][ntk] |= prv;
                ntk = msk | 1 << (j - 1);
                dp[i][0][ntk] |= prv;
                ntk = msk | 1 << (i - 1) | 1 << (j - 1);
                dp[i][1][ntk] |= prv;
                dp[i][1][ntk] |= prv2;
            }
        }
    }
    for (int msk = 0; msk < (1 << (n - 1)); msk++) { Ans = max(Ans, __builtin_popcount(msk)); }
    cout << Ans << "\n";
}
