h, w = map(int, input().split())
picture = [input() for _ in range(h)]

def is_single_plus(picture):
    stars = [(i, j) for i in range(h) for j in range(w) if picture[i][j] == '*']
    
    if len(stars) < 5:
        return False
    
    center = None
    for i, j in stars:
        neighbors = [(i+di, j+dj) for di, dj in [(0, -1), (0, 1), (-1, 0), (1, 0)]]
        if all(0 <= ni < h and 0 <= nj < w and picture[ni][nj] == '*' for ni, nj in neighbors):
            if center is not None:
                return False
            center = (i, j)
    
    if center is None:
        return False
    
    ci, cj = center
    for di, dj in [(0, -1), (0, 1), (-1, 0), (1, 0)]:
        ni, nj = ci + di, cj + dj
        while 0 <= ni < h and 0 <= nj < w and picture[ni][nj] == '*':
            if (ni, nj) not in stars:
                return False
            ni += di
            nj += dj
    
    return True

print("YES" if is_single_plus(picture) else "NO")
