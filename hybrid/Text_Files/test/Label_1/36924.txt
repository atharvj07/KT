import sys, math
n = int(input())
a = [int(i) for i in input().split()]
cnt = {}
for x in a:
    if x not in cnt: cnt[x] = 0
    cnt[x] += 1
ans = 0
mod = int(1e9) + 7
for x in cnt:
    for i in range((int)(math.log2(cnt[x]))+1):
        if (2**i)%7 != 1: ans = (ans + math.factorial(cnt[x])/(math.factorial(cnt[x]-i)*math.factorial(i)))%mod
print(ans)
