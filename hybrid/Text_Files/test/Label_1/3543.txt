#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<vector<vector<int>>> schedule(k, vector<vector<int>>(m));
    int small = n / m, big = (n + m - 1) / m;
    int cur = 0;
    for (int i = 0; i < k; i++) {
      for (int j = 0; j < m; j++) {
        if (i % 2 == 0 || j % 2 == 0) {
          schedule[i][j].push_back(small);
          for (int l = 0; l < small; l++) {
            schedule[i][j].push_back(cur + l + 1);
          }
        } else {
          schedule[i][j].push_back(big);
          for (int l = 0; l < big; l++) {
            schedule[i][j].push_back(cur + l + 1);
          }
          cur += big - small;
        }
      }
    }
    for (int i = 0; i < k; i++) {
      for (int j = 0; j < m; j++) {
        cout << schedule[i][j][0];
        for (int l = 1; l < schedule[i][j].size(); l++) {
          cout << ' ' << schedule[i][j][l];
        }
        cout << endl;
      }
      cout << endl;
    }
  }
  return 0;
}
