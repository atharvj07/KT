import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        sc.close();
        int i = 0;
        while(true) {
            if(7 * i % k == 0) {
                System.out.println(i + 1);
                return;
            }
            i++;
        }
    }
}
