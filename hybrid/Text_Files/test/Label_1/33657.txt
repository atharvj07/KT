import java.util.*;

public class Main {
    static int t, n;
    static long[] a, b;
    static List<Long> ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            n = sc.nextInt();
            a = new long[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextLong();
            }
            solve();
            for (long x : ans) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }

    static void solve() {
        b = new long[n];
        ans = new ArrayList<>();
        Arrays.sort(a);
        long sum = 0;
        for (long x : a) sum += x;
        long cur = a[n - 1], cnt = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (sum / cur >= 1) {
                cnt = sum / cur;
                break;
            } else {
                sum -= a[i];
                cur = a[i];
            }
        }
        ans.add(cur);
        long pre = cur;
        for (int i = 1; i < n; i++) {
            if (cnt > 0) {
                ans.add(cur);
                cnt--;
            } else {
                pre = lcm(pre, a[i]);
                ans.add(pre);
            }
        }
    }

    static long gcd(long a, long b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }

    static long lcm(long a, long b) {
        return (a * b) / gcd(a, b);
    }
}
