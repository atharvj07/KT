#include <bits/stdc++.h>
using namespace std;

const int mxN = 2e5 + 5;
long long c[mxN], d[mxN];
int main() {
    long long n, rat = 1899;
    cin >> n;
    for (long long i = 0; i < n; ++i) {
        cin >> c[i] >> d[i];
        if (d[i] == 1) rat = max(rat + c[i], 1899ll);
    }
    if (rat > 1899)
        cout << "Infinity";
    else if (rat < 1900)
        cout << "Impossible";
    else
        cout << rat;
}
