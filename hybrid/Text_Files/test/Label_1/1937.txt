#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int main() {
    int n, q;
    cin >> n >> q;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    while (q--) {
        int t, x;
        cin >> t >> x;
        if (t == 1) a[x] = 1 - a[x];
        else {
            vector<int> b;
            for (int i = 1; i <= n; i++) b.push_back(a[i]);
            sort(b.begin(), b.end());
            reverse(b.begin(), b.end());
            cout << b[x - 1] << endl;
        }
    }
    return 0;
}
