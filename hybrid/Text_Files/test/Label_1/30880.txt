import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> songs = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            int t = sc.nextInt();
            songs.put(s, t);
        }
        String x = sc.next();
        int sum = 0;
        for (Map.Entry<String, Integer> entry : songs.entrySet()) {
            if (entry.getKey().equals(x)) {
                break;
            }
            sum += entry.getValue();
        }
        System.out.println(sum);
    }
}
