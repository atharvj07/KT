p=[('zero','one','two','three','four','five','six','seven','eight','nine')]
t=[('ten','eleven','twelve','thirteen','fourteen','fifteen','sixteen','seventeen','eighteen','nineteen')]
ty=[('','','twenty','thirty','forty','fifty','sixty','seventy','eighty','ninety')]
def convert_to_words(num):
    if num==0:
        return 'zero'
    elif num<10:
        return p[num][0]
    elif num<20:
        return t[num-10][0]
    elif num<100:
        return ty[num//10][0]+'-'+p[num%10][0]
    elif num==100:
        return 'one'+'-'+'hundred'
    elif num<1000:
        return p[num//100][0]+'-'+'hundred'+'-'+convert_to_words(num%100)
    else:
        return 'one'+'-'+'thousand'
num = int(input())
print(convert_to_words(num))
