import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int[][] dp = new int[n + 1][n + 1];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = i; j < n; j++) {
                dp[i][j] = Integer.MIN_VALUE;
                for (int k = i + 1; k <= j + 1; k++) {
                    int sum = 0;
                    for (int l = i; l < k; l++) {
                        sum += a[l];
                    }
                    dp[i][j] = Math.max(dp[i][j], sum - dp[k][j]);
                }
            }
        }
        System.out.println(dp[0][n - 1]);
    }
}
