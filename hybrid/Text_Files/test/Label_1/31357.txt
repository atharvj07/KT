import java.util.ArrayList;
import java.util.Scanner;

class Main {
    public static int[] restoreDiary(int n, int[] diary) {
        int maxBook = 0;
        int currBook = 0;
        int days = 0;
        ArrayList<Integer> restoredDiary = new ArrayList<>();
        
        for (int i = 0; i < n; i++) {
            if (diary[i] != 0) {
                if (diary[i] == currBook) {
                    days++;
                } else {
                    currBook = diary[i];
                    days = 1;
                }
                maxBook = Math.max(maxBook, currBook);
            }
            
            if (days >= 2 && days <= 5) {
                restoredDiary.add(currBook);
            } else {
                return new int[] {-1};
            }
        }
        
        while (restoredDiary.size() < n) {
            restoredDiary.add(maxBook);
        }
        
        int[] restoredDiaryArray = new int[restoredDiary.size()];
        for (int i = 0; i < restoredDiaryArray.length; i++) {
            restoredDiaryArray[i] = restoredDiary.get(i);
        }
        
        return restoredDiaryArray;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        int n = scanner.nextInt();
        int[] diary = new int[n];
        for (int i = 0; i < n; i++) {
            diary[i] = scanner.nextInt();
        }
        
        int[] restoredDiary = restoreDiary(n, diary);
        if (restoredDiary[0] == -1) {
            System.out.println("-1");
        } else {
            System.out.println(restoredDiary[0]);
            for (int i = 1; i < restoredDiary.length; i++) {
                System.out.print(restoredDiary[i] + " ");
            }
            System.out.println();
        }
    }
}
