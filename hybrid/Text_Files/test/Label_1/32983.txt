import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), d = sc.nextInt();
        int[] c = new int[m];
        for (int i = 0; i < m; i++) c[i] = sc.nextInt();
        if (!canReach(n, m, d, c)) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int i = 0; i < n + 2; i++) {
                if (i < c[0]) System.out.print("1 ");
                else {
                    for (int j = 0; j < m; j++) {
                        if (i >= c[j] && i < c[j] + c[j]) System.out.print((j + 1) + " ");
                    }
                    if (i >= c[m - 1] + c[m - 1]) System.out.print("0 ");
                }
            }
        }
    }

    static boolean canReach(int n, int m, int d, int[] c) {
        int[] dp = new int[n + 2];
        dp[0] = 1;
        for (int i = 0; i <= n + 1; i++) {
            if (dp[i] == 1) {
                for (int j = 1; j <= d; j++) {
                    if (i + j <= n + 1) {
                        for (int k = 0; k < m; k++) {
                            if (i + j >= c[k] && i + j < c[k] + c[k]) dp[i + j] = 1;
                        }
                    }
                }
            }
        }
        return dp[n + 1] == 1;
    }
}
