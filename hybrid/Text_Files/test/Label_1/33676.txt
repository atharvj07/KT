import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    static int n, k, a, b;
    static int[] orders, pre;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        a = Integer.parseInt(st.nextToken());
        b = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        orders = new int[n + 1];
        pre = new int[n + 1];

        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            if (type == 1) {
                orders[Integer.parseInt(st.nextToken())] += Integer.parseInt(st.nextToken());
            } else {
                System.out.println(calc(Integer.parseInt(st.nextToken())));
            }
        }
    }

    static int calc(int x) {
        Arrays.fill(pre, 0);
        pre[k] = Math.min(a * k, b * (k + 1) + orders[k]);
        for (int i = k + 1; i <= n; i++) {
            pre[i] = pre[i - 1] + Math.min(a, b + orders[i]);
        }
        int ans = 0;
        for (int i = x; i <= x + k - 1; i++) {
            ans += pre[Math.min(i + k, n)] - pre[i - 1];
        }
        return ans;
    }
}
