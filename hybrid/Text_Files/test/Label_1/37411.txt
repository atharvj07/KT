def min_additional_lights(n, lights):
    def helper(node):
        if node >= len(lights):
            return 0
        left_sum = helper(2 * node + 1)
        right_sum = helper(2 * node + 2)
        added_lights = max(left_sum, right_sum) - min(left_sum, right_sum)
        lights[node] += added_lights
        total_sum = lights[node] + max(left_sum, right_sum)
        return total_sum
    
    lights = [0] + lights
    total = helper(1)
    return sum(lights) - total

n = int(input().strip())
lights = list(map(int, input().strip().split()))
print(min_additional_lights(n, lights))
