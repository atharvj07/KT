#include <iostream>
#include <vector>
using namespace std;

int main() {
  // Process input
  int n;
  cin >> n;
  vector<pair<int, int>> a(n);
  vector<int> freq(n + 1); // frequency of each element in the sequence
  for (auto& [x, y] : a) { cin >> x >> y; freq[x]++; freq[y]++; }

  long ans = 1LL * n * (n - 1) * (n - 2);
  int mx = 0; // maximum frequency of any element in the sequence
  for (int i = 1; i <= n; i++) mx = max(mx, freq[i]);
  ans -= 1LL * mx * (mx - 1) * (mx - 2); // subtract the number of bad permutations where the element with maximum frequency is sorted
  ans %= 998244353;
  cout << ans << endl;
  return 0;
}
