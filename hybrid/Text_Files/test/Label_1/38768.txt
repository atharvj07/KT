import sys as s
order = []
for _ in range(int(input())):
    order.append([])
    for j in range(int(input())):order[-1].append(tuple(map(int,input().split())))
    try:
        co=[(0,0)]
        order[-1].sort()
        for x,y in order[-1]:
            while x>co[-1][0]:co.append((co[-1][0]+1,co[-1][1]))
            while y>co[-1][1]:co.append((co[-1][0],co[-1][1]+1))
        s.stdout.write("YES\n"+"".join(['R'if co[i][0]>co[i-1][0]else'U'for i in range(1,len(co))])+"\n")
    except:s.stdout.write("NO\n")
