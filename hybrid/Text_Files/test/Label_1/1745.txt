#include <iostream>
using namespace std;

const int MAX_SIZE = 50;

// count digit's frequency
int freq[MAX_SIZE];

int main() {
  int n, x;
  cin >> n;
  while (n > 0) {
    x = n % 10;
    freq[x]++;
    n /= 10;
  }
  int odd_count = 0;
  for (int i = 0; i < MAX_SIZE; i++) {
    odd_count += (freq[i] & 1);
  }
  if (odd_count > 1) {
    cout << -1;
  } else {
    cout << odd_count;
  }
  return 0;
}
