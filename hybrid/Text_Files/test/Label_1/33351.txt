import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        Map<String, Integer> suits = new HashMap<>();
        Map<String, Integer> ranks = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String card = sc.next();
            String rank = card.substring(0, 1);
            String suit = card.substring(1, 2);

            suits.put(suit, suits.getOrDefault(suit, 0) + 1);
            ranks.put(rank, ranks.getOrDefault(rank, 0) + 1);
        }

        int minSuits = Integer.MAX_VALUE;
        for (String suit : suits.keySet()) {
            minSuits = Math.min(minSuits, suits.get(suit));
        }

        int minRanks = Integer.MAX_VALUE;
        for (String rank : ranks.keySet()) {
            minRanks = Math.min(minRanks, ranks.get(rank));
        }

        if (minSuits >= 2 || minRanks >= 3) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
