#include <iostream>
#include <queue>
#include <vector>
using namespace std;
struct Laundry {
  int t_start;
  int t_wait;
  int t_end;
};
int main() {
  priority_queue<int, vector<int>, greater<int>> T;
  int k, n1, n2, n3, t1, t2, t3;
  cin >> k >> n1 >> n2 >> n3 >> t1 >> t2 >> t3;
  vector<Laundry> laundry(k);
  for (int i = 0; i < k; i++) {
    if (i < n1) {
      laundry[i].t_start = 0;
      laundry[i].t_end = t1;
      T.push(t1);
    } else {
      laundry[i].t_start = T.top();
      T.pop();
      laundry[i].t_end = laundry[i].t_start + t1;
      T.push(laundry[i].t_end);
    }
    T.pop();
    while (!T.empty() && T.top() < laundry[i].t_end) {
      T.pop();
    }
    if (T.empty()) {
      laundry[i].t_wait = t2;
      T.push(laundry[i].t_end + t2);
    } else {
      laundry[i].t_wait = T.top() - laundry[i].t_end;
      T.push(laundry[i].t_end + t2);
    }
    T.pop();
    while (!T.empty() && T.top() < laundry[i].t_end + laundry[i].t_wait) {
      T.pop();
    }
    if (T.empty()) {
      laundry[i].t_end += laundry[i].t_wait + t3;
    } else {
      laundry[i].t_end = T.top() + t3;
      T.push(laundry[i].t_end);
    }
  }
  cout << laundry[k - 1].t_end << endl;
}
