import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        if (l == r || r - l < 2) {
            System.out.println(-1);
        } else if (l % 2 == 0) {
            System.out.println(l + " " + (l + 1) + " " + (l + 2));
        } else {
            if ((l + 1) % 3 == 0) {
                System.out.println(l + " " + (l + 1) + " " + (l + 3));
            } else {
                System.out.println(l + " " + (l + 2) + " " + (l + 3));
            }
        }
    }
}
