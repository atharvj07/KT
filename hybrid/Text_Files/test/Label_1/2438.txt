#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n);
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    vector<int> a(n);
    a[0] = b[0];
    int l = 1, r = n - 1;
    for (int i = 1; i < n; i++) {
      if (i % 2 == 1) {
        a[l] = b[i];
        l++;
      } else {
        a[r] = b[i];
        r--;
      }
    }
    for (int x : a) { cout << x << ' '; }
    cout << '\n';
  }
  return 0;
}
