from collections import deque

n=int(input())
a=[]
b=[]
for i in range(n):
    x,c=input().split()
    a.append(int(x))
    b.append(c)
a=[[x,i] for i,x in enumerate(a)]
a.sort()
ans=0
st=deque()
for x,i in a:
    if b[i]=='P':
        st.append([x,i])
    elif b[i]=='B':
        while st and st[0][1]<i:
            st.popleft()
        if st:
            ans+=x-st[0][0]
            st.append([x,i])
st.clear()
for x,i in a[::-1]:
    if b[i]=='P':
        st.append([x,i])
    elif b[i]=='R':
        while st and st[0][1]<i:
            st.popleft()
        if st:
            ans+=st[0][0]-x
            st.append([x,i])
print(ans)
