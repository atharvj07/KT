#include <iostream>
#include <stack>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  stack<char> st;
  for (int i = 0; i < n; i++) {
    if (s[i] == ')' && !st.empty() && st.top() == '(')
      st.pop();
    else if (s[i] == ']') {
      if (!st.empty() && st.top() == '[') {
        st.pop();
      }
    } else if (s[i] == '}') {
      if (!st.empty() && st.top() == '{') {
        st.pop();
      }
    } else {
      st.push(s[i]);
    }
    if (st.empty()) {
      cout << 0 << " ";
    } else {
      cout << -1 << " ";
    }
  }
  cout << endl;
  return 0;
}
