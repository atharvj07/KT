import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt(), c = sc.nextInt();
        int diff = c - a;
        int ans = 0;
        int i = 1;
        while (diff > 0) {
            if (diff % 3 != 0) {
                ans += diff % 3 * i;
            }
            diff /= 3;
            i *= 10;
        }
        System.out.println(ans);
    }
}
