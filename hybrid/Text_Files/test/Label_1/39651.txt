t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    color_count = [0] * (n+1)
    for i in a:
        color_count[i] += 1
    min_possible = sum(min(color_count[i], k) for i in set(a)) // k * k
    color_assignments = [0] * n
    current_color = 1
    colored = 0
    for i in range(n):
        if color_count[a[i]] > 0 and color_count[a[i]] <= k:
            color_assignments[i] = current_color
            color_count[a[i]] -= 1
            colored += 1
            current_color = current_color % k + 1
            if colored == min_possible:
                break
    print(" ".join(map(str, color_assignments)))
