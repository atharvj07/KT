import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    String[] nm = br.readLine().split("\\s");
    int n = Integer.parseInt(nm[0]);
    int m = Integer.parseInt(nm[1]);

    String[] yx = br.readLine().split("\\s");
    int ys = Integer.parseInt(yx[0]);
    int xs = Integer.parseInt(yx[1]);
    int yf = Integer.parseInt(yx[2]);
    int xf = Integer.parseInt(yx[3]);

    int[][] loc = new int[m][2];
    for (int i = 0; i < m; i++) {
      String[] xy = br.readLine().split("\\s");
      loc[i] = new int[] { Integer.parseInt(xy[0]), Integer.parseInt(xy[1]) };
    }

    int ans = Integer.MAX_VALUE;

    for (int i = 0; i < m; i++) {
      if (loc[i][0] == ys || loc[i][1] == xs) {
        ans = Math.min(ans, Math.abs(xf - loc[i][0]) + Math.abs(yf - loc[i][1]));
      }
    }

    ans = Math.min(ans, Math.abs(xf - xs) + Math.abs(yf - ys));

    System.out.println(ans);
  }
}
