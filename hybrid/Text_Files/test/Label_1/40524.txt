n = int(input())

result = ''
for i in range (n):
  if i == 0:
    result += 'a'
  elif i == 1:
    result += 'b'
  else:
    if result[i - 1] == 'c':
      if result[i - 2] == 'b':
        result += 'a'
      else:
        result += 'c'
    else:
      if result[i - 2] == 'a':
        result += 'b'
      else:
        result += 'c'

print (result)
