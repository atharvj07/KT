import sys
input = sys.stdin.readline

a, b, h, w, n = map(int, input().split())
ext = list(map(int, input().split()))
ext.sort()

if a <= h and b <= w:
    print(0)
    exit()

ans = float('inf')

for i in ext:
    if i >= a and h * i >= b:
        ans = min(ans, 1)
    elif i >= b and w * i >= a:
        ans = min(ans, 1)

if w * h >= a * b:
    ans = min(ans, 0)

for i in ext:
    if h * i >= b:
        h *= i
        ans = min(ans, 2)
        break

for i in ext:
    if w * i >= a:
        w *= i
        ans = min(ans, 2)
        break

if a <= h and b <= w:
    print(ans)
else:
    print(-1)
