import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int Q = sc.nextInt();
        for (int i = 0; i < Q; i++) {
            long n = sc.nextLong();
            int ans = 0;
            while (n > 9) {
                long max = 0;
                for (int j = 1; j < n; j *= 10) {
                    long num1 = (n / j) % 10;
                    long num2 = n % j;
                    max = Math.max(max, num1 * num2);
                }
                n = max;
                ans++;
            }
            System.out.println(ans == Integer.MAX_VALUE ? -1 : ans);
        }
    }
}
