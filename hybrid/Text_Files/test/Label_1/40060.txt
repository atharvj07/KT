import math
n=int(input())
A=[list(map(int,input().split())) for i in range(n)]
a=[]
for i in range(n):
    for j in range(n):
        for k in range(n):
            if i!=j and j!=k and i!=k:
                x=[A[i][p]-A[k][p] for p in range(5)]
                y=[A[j][p]-A[k][p] for p in range(5)]
                dot=0
                Modx=0
                Mody=0
                for p in range(5):
                    dot+=x[p]*y[p]
                    Modx+=x[p]**2
                    Mody+=y[p]**2
                dot=(dot/(Modx**0.5*Mody**0.5))
                if dot<-0.999 and dot>-1.001: a.append(i)
print(len(set(a)))
for i in set(a): print(i+1)
