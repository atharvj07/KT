def apply_permutation(perm, original):
    return [original[i - 1] for i in perm]

def find_inverse_permutation(perm):
    inverse_perm = [0] * len(perm)
    for i, p in enumerate(perm):
        inverse_perm[p - 1] = i + 1
    return inverse_perm

def is_possible(n, k, q, s):
    original_perm = list(range(1, n + 1))
    inverse_q = find_inverse_permutation(q)
    permutations_set = set()
    
    current_perm = original_perm
    for _ in range(k):
        current_perm = apply_permutation(q, current_perm)
        if tuple(current_perm) in permutations_set:
            return "NO"
        permutations_set.add(tuple(current_perm))
        q, inverse_q = inverse_q, q  # Switch the permutation applied each move
    
    if current_perm == s:
        return "YES"
    else:
        return "NO"

n, k = map(int, input().strip().split())
q = list(map(int, input().strip().split()))
s = list(map(int, input().strip().split()))

print(is_possible(n, k, q, s))
