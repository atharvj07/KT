#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
const int arsz=1e6+1;
ll a[arsz],pre[arsz],suf[arsz],cnt[arsz];

int main(){
    ll t,i,j,n,ans;
    cin>>t;
    while(t--){
        cin>>n;
        for(i=0;i<arsz;++i) pre[i]=suf[i]=cnt[i]=0;
        for(i=1;i<=n;++i){
            cin>>a[i];
            ++cnt[a[i]];
        }
        pre[a[1]]=1;
        for(i=2;i<=n;++i) pre[a[i]]=pre[a[i]]+cnt[a[i]]-1;
        suf[a[n]]=1;
        for(i=n-1;i>=1;--i) suf[a[i]]=suf[a[i]]+cnt[a[i]]-1;
        ans=0;
        for(i=1;i<=n;++i){
            ans+=pre[a[i]]*suf[a[i]];
        }
        cout<<ans<<'\n';
    }
    return 0;
}
