#include <iostream>
#include <string>
using namespace std;
int main() {
  string s;
  cin >> s;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] != '9' && s[i] > (char)('0' + 9 - (s[i] - '0'))) s[i] = (char)('0' + 9 - (s[i] - '0'));
  }
  while (s.length() > 1 && s[0] == '0') s.erase(s.begin());
  cout << s << endl;
  return 0;
}
