n, q = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))
st = [0] * n
ans = []
for i in range(q):
    tot = 0
    for j in range(n):
        if st[j] == 0:
            tot += 1
            a[j] -= k[i]
            if a[j] <= 0:
                st[j] = 1
    ans.append(tot)
    if tot == 0:
        st = [0] * n
print(*ans, sep = '\n')
