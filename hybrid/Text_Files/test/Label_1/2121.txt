#include<bits/stdc++.h>
using namespace std;
int n,a[100001],pos[100001],dp[100001],l,r,ans,c,x,y;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){cin>>x;pos[x]=i;}
    for(int i=1;i<=n;i++){cin>>x;a[i]=pos[x];}
    l=1;r=0;
    for(int i=1;i<=n;i++){
        if(a[i]<a[r])continue;
        while(l<=r&&a[r]<a[i])r--;
        dp[i]=r;l=i;r=i;
    }
    for(int i=1;i<=n;i++)ans+=!dp[i];
    cout<<ans;
    return 0;
}
