#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int r, g, b;
    cin >> r >> g >> b;
    if (r > 1 && g > 1 && b > 1) {
      cout << "No" << endl;
      continue;
    }
    if (r == 1 && g == 1) {
      cout << "No" << endl;
      continue;
    }
    if (g == 1 && b == 1) {
      cout << "No" << endl;
      continue;
    }
    if (r == 1 && b == 1) {
      cout << "No" << endl;
      continue;
    }
    cout << "Yes" << endl;
  }
  return 0;
}
