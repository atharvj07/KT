void solve(){
    ll n,k;
    cin>>n>>k;
    vector<int> ans;
    for(int i=0;i<n;i++){
        ans.pb(i+1);
    }
    for(int i=n;i<2*n;i++){
        ans.pb(i+1);
    }
    int temp=n-1;
    int cnt=0;
    for(int i=0;i<n;i++){
        if(cnt>=k)break;
        swap(ans[i],ans[temp]);
        cnt++;
        temp--;
    }
    for(int i=0;i<2*n;i++){
        cout<<ans[i]<<" ";
    }
    cout<<endl;
    return;
}
int main(){
    int t=1;
    // cin>>t;
    while(t--){
        solve();
    }
    return 0;
}
