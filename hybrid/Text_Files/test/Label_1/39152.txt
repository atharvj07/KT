import sys
input = sys.stdin.readline

n = int(input())
a, g, chk = [0]*n, [0]*n, [0]*n

for i in range(n):
    x, y = map(int, input().split())
    a[i], g[i] = x, y
    if x > y: chk[i] = 1

a.sort()
g.sort(reverse=True)

sa, sg = 0, 0
for i in range(n):
    if chk[i]:
        sa += a[i]
        sg += g[i]
    else:
        sa += g[i]
        sg += a[i]

if abs(sa-sg) <= 500:
    ans = ''.join(['G' if chk[i] else 'A' for i in range(n)])
    print(ans)
else: print(-1)
