import sys
for t in range(int(input())):
    n, x = map(int, input().split())
    s = input()
    c0, c1 = 0, 0
    cnt = []
    ans = 0
    for chr in s:
        if chr == '0': c0 += 1
        else: c1 += 1
        cnt.append(c0-c1)
    if x == 0: ans = n
    else: 
        for i in range(1, len(s)):
            if cnt[i] == x: ans += 1
            if cnt[i] == cnt[i-1]: ans += cnt[i-1]==x
        if s[0] == '0':
            if x == cnt[0]: ans += 1
        else:
            if x == -cnt[0]: ans += 1
        if x in cnt: ans += 1
    print(-1 if ans == 0 else ans)
