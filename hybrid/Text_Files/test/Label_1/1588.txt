#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> c(n), t(n);
    for (int i = 0; i < n; i++) { cin >> c[i]; }
    for (int i = 0; i < n; i++) { cin >> t[i]; }
    vector<bool> possible(n);
    possible[0] = possible[n - 1] = true;
    for (int i = 1; i < n - 1; i++) { possible[i] = (c[i] == t[i]); }
    for (int i = 1; i < n - 1; i++) {
        if (possible[i - 1] && possible[i + 1] && 2 * c[i] == c[i + 1] + c[i - 1]) {
            possible[i] = true;
        }
    }
    cout << (possible[n - 2] ? "Yes" : "No") << endl;
    return 0;
}
