import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] x1 = new int[n];
        int[] y1 = new int[n];
        int[] x2 = new int[n];
        int[] y2 = new int[n];
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            x1[i] = Integer.parseInt(line[0]);
            y1[i] = Integer.parseInt(line[1]);
            x2[i] = Integer.parseInt(line[2]);
            y2[i] = Integer.parseInt(line[3]);
        }
        // Horizontal segments
        Map<Integer, Integer> horizontal = new HashMap<>();
        // Vertical segments
        Map<Integer, Integer> vertical = new HashMap<>();
        for (int i = 0; i < n; i++) {
            if (x1[i] == x2[i]) {
                vertical.put(x1[i], vertical.getOrDefault(x1[i], 0) + 1);
            } else {
                horizontal.put(y1[i], horizontal.getOrDefault(y1[i], 0) + 1);
            }
        }
        int ans = 0;
        // For each horizontal segment
        for (int h : horizontal.keySet()) {
            // For each vertical segment
            for (int v : vertical.keySet()) {
                // Count the number of intersections
                int cnt = horizontal.get(h) * vertical.get(v);
                ans += cnt * (cnt - 1) / 2;
            }
        }
        System.out.println(ans);
    }
}
