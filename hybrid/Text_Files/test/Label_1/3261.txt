#include <iostream>
#include <vector>
using namespace std;
int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int>f(n);
  if (s[0] == ')') f[0] = 0;
  else f[0] = 1;
  for (int i = 1; i < n; i++) {
    if (s[i] == ')' && s[i - 1] == '(') {
      f[i] = f[i - 2] + 2;
    } else if (s[i] == ')') {
      f[i] = f[i - 1] - 1;
    } else {
      f[i] = f[i - 1] + 1;
    }
  }
  int q;
  cin >> q;
  while (q--) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    cout << f[r] - (l == 0 ? 0 : f[l - 1]) << endl;
  }
}
