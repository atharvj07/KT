import java.util.*;

public class Main {
    static char[][] a;
    static int n, m;
    static int[] ox = {0, 0, 1, 1};
    static int[] oy = {0, 1, 0, 1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                Set<Character> set = new HashSet<>();
                set.add(a[i][j]);
                for (int k = 0; k < 4; k++) {
                    int x = i + ox[k];
                    int y = j + oy[k];
                    if (0 <= x && x < n && 0 <= y && y < m) {
                        set.add(a[x][y]);
                    }
                }
                if (set.size() < 4) {
                    for (int k = 0; k < 4; k++) {
                        int x = i + ox[k];
                        int y = j + oy[k];
                        if (0 <= x && x < n && 0 <= y && y < m) {
                            if (!set.contains(a[x][y])) {
                                a[i][j] = a[x][y];
                                break;
                            }
                        }
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(a[i][j]);
            }
            System.out.println();
        }
    }
}
