import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        Set<Integer> s = new HashSet<>();
        for (int i = 0; i < n; i++) {
            if (c[i] != a) {
                s.add(c[i]);
            }
        }
        if (s.size() == 0) {
            System.out.println("-1");
        } else {
            System.out.println(s.iterator().next());
        }
    }
}
