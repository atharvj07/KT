def construct_tree(n, d, k):
    if k == 1:
        if d + 1 == n:
            print("YES")
            for i in range(1, n):
                print(i, i + 1)
        else:
            print("NO")
        return
    if d >= n or 2 * (k - 1) * (1 - pow(k - 1, d // 2)) // (1 - (k - 1)) + 1 < n:
        print("NO")
        return
    print("YES")
    levels = [[] for _ in range(d + 1)]
    levels[0].append(1)
    vertex = 2
    for i in range(1, d + 1):
        for u in levels[i - 1]:
            if len(levels[i]) == (k - 1) * len(levels[i - 1]):
                break
            levels[i].append(vertex)
            print(u, vertex)
            vertex += 1
    for i in range(d + 1):
        for u in levels[i]:
            while len(levels[i]) < k and vertex <= n:
                if i > 0 and len(levels[i - 1]) < k:
                    break
                levels[i].append(vertex)
                print(u, vertex)
                vertex += 1

n, d, k = map(int, input().split())
construct_tree(n, d, k)
