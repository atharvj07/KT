import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int result = 0;
        while (true) {
            int max = 0;
            int maxCount = 0;
            for (int element : a) {
                maxCount = Math.max(maxCount, ++element);
            }
            if (maxCount >= k) {
                break;
            }
            result++;
        }

        System.out.println(result);
    }
}
