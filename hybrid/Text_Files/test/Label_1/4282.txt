#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin >> t;
    while(t--){
        string a, b;
        cin >> a >> b;
        int n = a.size(), m = b.size();
        if(a == b){
            cout << "---\n";
            continue;
        }
        for(int i=0; i<n; i++){
            for(int j=i+1; j<n; j++){
                if(a[i] > a[j]){
                    swap(a[i], a[j]);
                    if(a < b){
                        cout << a << '\n';
                        goto loop;
                    }
                    swap(a[i], a[j]);
                }
            }
        }
        cout << "---\n";
        loop:;
    }
    return 0;
}
