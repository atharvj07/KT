from sys import stdin, stdout
import math

n, k = map(int, stdin.readline().split())
w = list(map(int, stdin.readline().split()))
k50 = 0
for j in w:
    if j == 100:
        stdout.write("-1\n0")
        exit()
    k50 += j // 50
print(math.ceil(k50 / (k // 50)), pow(2, k50 - (k50 // (k // 50)), 9) % 1000000007)
