#include <iostream>
using namespace std;
int main() {
  int n, s1 = 0, s2 = 0;
  cin >> n;
  while (n--) {
    int m, c;
    cin >> m >> c;
    if (m > c)
      s1++;
    else if (c > m)
      s2++;
  }
  if (s1 > s2)
    cout << "Mishka";
  else if (s2 > s1)
    cout << "Chris";
  else
    cout << "Friendship is magic!^^";
  return 0;
}
