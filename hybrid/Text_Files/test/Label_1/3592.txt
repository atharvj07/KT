#include <bits/stdc++.h>
using namespace std;
vector<int> divs(int x){
    vector<int>ret;
    for(int i=1; i*i<=x; ++i){
        if(x%i==0){
            ret.push_back(i);
            if(x/i!=i)ret.push_back(x/i);
        }
    }
    return ret;
}
int main() {
    int n; cin >> n;
    vector<int>a(n);
    for(int i=0; i<n; ++i)cin >> a[i];
    int ans = -1;
    for(int i=0; i<n; ++i){
        vector<int>d = divs(a[i]);
        for(int j=0; j<d.size(); ++j){
            bool ok = true;
            for(int k=0; k<n; ++k)if(a[k]%d[j]!=0)ok=false;
            if(ok){
                ans = d[j];
                break;
            }
        }
        if(ans!=-1)break;
    }
    cout << ans
         << endl;
}
