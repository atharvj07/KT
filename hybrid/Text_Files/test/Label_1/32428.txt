import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
    private static final int MOD = 1000000007;
    private static List<Integer>[] adjList;
    private static Map<Long, Integer> dp;
    private static int[] cycleRoots;
    private static int[] vertexToCycleRoot;
    private static int[] subTreeSize;
    private static int[] parent;
    private static boolean[] visited;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        adjList = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) adjList[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            adjList[a].add(b);
            adjList[b].add(a);
        }
        dp = new HashMap<>();
        cycleRoots = new int[n + 1];
        vertexToCycleRoot = new int[n + 1];
        subTreeSize = new int[n + 1];
        parent = new int[n + 1];
        visited = new boolean[n + 1];
        findCycleRoots(1, 0);
        preProcess(1, 0);
        int k = Integer.parseInt(br.readLine());
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            if (vertexToCycleRoot[x] != vertexToCycleRoot[y]) {
                System.out.println(0);
                continue;
            }
            long key = (long)x << 32 | y;
            if (dp.containsKey(key)) {
                System.out.println(dp.get(key));
                continue;
            }
            int res = 0;
            if (x == y) res = subTreeSize[x];
            else {
                res = subTreeSize[y] - subTreeSize[parent[x]];
                res = addMod(res, subTreeSize[x] - subTreeSize[parent[y]]);
            }
            dp.put(key, res);
            System.out.println(res);
        }
    }

    private static void findCycleRoots(int u, int p) {
        visited[u] = true;
        for (int v : adjList[u]) {
            if (v == p) continue;
            if (visited[v]) cycleRoots[u] = v;
            else findCycleRoots(v, u);
        }
    }

    private static void preProcess(int u, int p) {
        subTreeSize[u] = 1;
        parent[u] = p;
        vertexToCycleRoot[u] = cycleRoots[u];
        for (int v : adjList[u]) {
            if (v == p) continue;
            if (cycleRoots[u] == cycleRoots[v]) {
                vertexToCycleRoot[v] = cycleRoots[u];
            }
            preProcess(v, u);
            subTreeSize[u] += subTreeSize[v];
        }
    }

    private static int addMod(int a, int b) {
        return ((a % MOD) + (b % MOD)) % MOD;
    }
}
