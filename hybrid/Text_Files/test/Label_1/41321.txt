import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

vis = [False] * (n+1)
ans = 0

def dfs(u, par):
    global ans
    vis[u] = True
    mx = a[u-1]
    chosen = False

    for v in g[u]:
        if not vis[v]:
            dfs(v, u)
            if not chosen:
                ans += mx
                chosen = True
            else:
                ans += max(0, a[v-1])
            mx = max(mx, a[v-1])
    
dfs(1, 1)
print(ans if ans != 0 else "Impossible")
