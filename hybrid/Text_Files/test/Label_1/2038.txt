#include <bits/stdc++.h>
using namespace std;
#define P pair<int, int>
#define rep(i, n) for (int i = 0; i < n; i++)
vector<P> ans;
bool ok[200005];
int main() {
  int n, q;
  cin >> n >> q;
  vector<int> a(n);
  rep(i, n) cin >> a[i];
  int cnt = 0;
  rep(i, n) {
    if (a[i] == 0) cnt++;
  }
  if (cnt > q) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  int lst = 0, val = 0;
  rep(i, n) {
    if (a[i] == 0) {
      if (lst == 0) {
        val = 1;
        lst = 1;
      }
      a[i] = val;
    } else if (lst == 0) {
      lst = a[i];
      val = a[i];
    } else if (lst == a[i]) {
      continue;
    } else {
      ans.push_back({lst, val});
      lst = a[i];
      val = a[i];
    }
  }
  if (lst != 0) ans.push_back({lst, val});
  cout << a[0];
  for (int i = 1; i < n; i++) { cout << " " << a[i]; }
  cout << endl;
}
