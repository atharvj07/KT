import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = in.next().toCharArray();
        }
        boolean[][] visited = new boolean[n][m];
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 'W' && !visited[i][j]) {
                    boolean found = false;
                    for (int[] d : new int[][]{{0, 1}, {0, -1}, {1, 0}, {-1, 0}}) {
                        int x = i + d[0];
                        int y = j + d[1];
                        if (x >= 0 && x < n && y >= 0 && y < m && grid[x][y] == 'P' && !visited[x][y]) {
                            visited[x][y] = true;
                            found = true;
                            count++;
                            break;
                        }
                    }
                    if (!found) {
                        visited[i][j] = true;
                    }
                }
            }
        }
        System.out.println(count);
    }
}
