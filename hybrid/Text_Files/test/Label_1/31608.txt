import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] changes = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                changes[i][j] = scanner.nextInt();
            }
        }
        int[] result = new int[n];
        int[] sum = new int[3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                sum[j] += changes[i][j];
            }
        }
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < 3; i++) {
            min = Math.min(min, Math.abs(sum[i]));
        }
        if (min % 2 != 0) {
            System.out.println("Impossible");
            return;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                int[] temp = sum.clone();
                temp[j] -= changes[i][j];
                if (Math.abs(temp[j]) == min) {
                    result[i] = j;
                    break;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (result[i] == 0) {
                System.out.print("L");
            } else if (result[i] == 1) {
                System.out.print("M");
            } else {
                System.out.print("W");
            }
            for (int j = 0; j < 3; j++) {
                if (j != result[i]) {
                    System.out.print(" " + (j + 1));
                }
            }
            System.out.println();
        }
    }
}
