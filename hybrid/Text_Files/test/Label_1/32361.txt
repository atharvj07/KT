import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        try(Scanner sc = new Scanner(System.in)) {
            long h = sc.nextLong();
            long m = sc.nextLong();
            long k = sc.nextLong();
            long h1 = sc.nextLong();
            long m1 = sc.nextLong();
            long h2 = sc.nextLong();
            long m2 = sc.nextLong();
            
            long lastM = (h2 + (m2 / m)) * m;
            long commonM = h1 * m + m1;
            
            if(commonM >= lastM) {
                commonM -= m;
                lastM -= m;
            }
            
            long lastH = lastM / m;
            long commonH = commonM / m;
            
            long add = 0;
            if(lastH == commonH) {
                add = (m2 - m1) / k;
                if((m2 - m1) % k != 0) add++;
            }
            
            long tot = lastH - commonH + add;
            System.out.println(tot);
        }
    }
}
