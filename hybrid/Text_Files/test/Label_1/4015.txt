#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<char> t;
    for (int i = 0; i < n; i++) {
        if (t.empty() || t.back() != s[i]) {
            t.push_back(s[i]);
        } else {
            t.pop_back();
        }
    }
    cout << t.size() << "\n";
    return 0;
}
