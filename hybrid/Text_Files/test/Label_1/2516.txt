#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> a[i][j];
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] < 0 && (i == 0 || a[i - 1][j] >= 0) &&
            (j == 0 || a[i][j - 1] >= 0)) {
          ans -= a[i][j];
          a[i][j] = -a[i][j];
          if (i > 0) a[i - 1][j] = -a[i - 1][j];
          if (j > 0) a[i][j - 1] = -a[i][j - 1];
        }
        ans += a[i][j];
      }
    }
    cout << ans << endl;
  }
  return 0;
}
