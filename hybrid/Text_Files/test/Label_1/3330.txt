#include <cstdio>
#include <cstring>
#include <algorithm>
using namespace std;

const int N = 51;
char s[N];
int dp[N][N];

int main() {
  scanf("%s", s);
  int k = strlen(s);
  for (int d = 1; d < k; ++d) {
    for (int i = 0; i < k - d; ++i) {
      int j = i + d;
      dp[i][j] = d + 1;
      for (int t = i; t < j; ++t)
        if (s[t] != s[j - t + i])
          dp[i][j] = min(dp[i][j], dp[i][t] + dp[t + 1][j]);
    }
  }
  printf("%d\n", dp[0][k - 1] >= 2 * k ? 0 : dp[0][k - 1] - 1);
}
