import java.util.*;
import java.io.*;

class Main{
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long[] ab = Arrays.stream(in.nextLine().split(" ")).mapToLong(Long::parseLong).toArray();
        long ans[] = EEA(ab[0], ab[1]);
        System.out.println(ans[0] + " " + ans[1]);
    }
    public static  long[] EEA(long a, long b) {
        if (b == 0) return new long[]{1, 0};
        long[] tmp = EEA(b, a % b);
        long x = tmp[0];
        long y = tmp[1];
        return new long[]{y, x - y * (a / b)};
    }
}
