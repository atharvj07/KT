n = int(input().strip())
a = list(map(int, input().strip().split()))

left = [0] * n
right = [0] * n

for i in range(1, n):
    left[i] = max(left[i-1] + 1, a[i])

for i in range(n-2, -1, -1):
    right[i] = max(right[i+1] + 1, a[i])

res = float('inf')
for i in range(n):
    res = min(res, max(left[i], right[i]))

additional_requests = sum(max(left[i], right[i]) - a[i] for i in range(n)) - (sum(left) + sum(right) - res * n)

print(additional_requests)
