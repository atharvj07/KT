n,l,r=map(int,input().split())
x=int((l+r)%3)
if(n%3==x):
    #dp[rem][atleast1][pos] : no. of ways to get rem with atleast1 as prefix at pos
    dp=[[[-1 for k in range(n+1)] for j in range(2)] for i in range(3)]
    mod=10**9+7
    dp[x][1][0]=1
    for rem in range(3):
        for at in range(2):
            for pos in range(n):
                if(dp[rem][at][pos]==-1):
                    continue
                #choose l
                dp[rem][at][pos+1]=(dp[rem][at][pos+1]+dp[rem][at][pos]*((l+r-x)%3))%mod
                #choose r
                dp[rem][at][pos+1]=(dp[rem][at][pos+1]+dp[rem][at][pos]*(x))%mod
                #choose ?
                rem_=(rem+l+r-x)%3
                dp[rem_][1][pos+1]=(dp[rem_][1][pos+1]+dp[rem][at][pos])%mod
    print(dp[0][1][n])
else:
    print(0)
