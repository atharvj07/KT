import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    arr = list(map(int, input().split()))

    if sum(arr) % 2 == 0 or k % 2 == 0 or n < k:
        print("NO")
        continue

    print("YES")
    sub_len = n//k + bool(n % k)

    for i in range(0, n, sub_len):
        print(i + sub_len, end=' ')
    print()
