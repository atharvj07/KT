import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[3];

        int n = sc.nextInt();
        for (int i = 0; i < 3; i++)
            arr[i] = sc.nextInt();

        long[][] dp = new long[n + 1][arr[0] + 1];
        for (int i = 0; i <= arr[0]; i++) {
            dp[0][i] = 1;
        }

        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= arr[0]; j++) {
                if (i >= 5) {
                    dp[i][j] += dp[i - 5][j];
                }
                if (i >= 1 && j > 0) {
                    dp[i][j] += dp[i - 1][j - 1];
                }
                if (i >= 2 && j >= 2) {
                    dp[i][j] += dp[i - 2][j - 2];
                }
            }
        }
        long ans = 0;
        for (int i = 0; i <= arr[0]; i++) {
            for (int j = 0; j <= arr[1]; j++) {
                for (int k = 0; k <= arr[2]; k++) {
                    int sum = i * 5 + j * 1 + k * 2;
                    if (sum == n) {
                        ans += dp[i][j] * dp[j][k];
                    }
                }
            }
        }
        System.out.println(ans);
        sc.close();
    }
}
