#include<bits/stdc++.h>
using namespace std;
const int MAXN = 200;
int a[MAXN];
int main() {
    int n;
    scanf("%d",&n);
    int l=2*n-1,sum=0;
    for(int i=0;i<l;++i) {
        scanf("%d",&a[i]);
        sum+=abs(a[i]);
    }
    sort(a,a+l);
    for(int i=0;i<n;++i) {
        sum-=min(2*a[i],a[l-1-i]);
    }
    printf("%d\n",sum);
    return 0;
}
