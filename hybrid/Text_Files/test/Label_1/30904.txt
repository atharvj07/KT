import java.util.*;

public class Main {
    static final long MOD = 1000000007l;

    static class Edge implements Comparable<Edge> {
        int to, weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }

        @Override
        public int compareTo(Edge o) {
            return Integer.compare(weight, o.weight);
        }
    }

    static long dijkstra(int s, int t, List<List<Edge>> graph) {
        int n = graph.size();
        long[] dist = new long[n + 1];
        Arrays.fill(dist, Long.MAX_VALUE);
        dist[s] = 0;
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(s, 0));
        while (!pq.isEmpty()) {
            Edge cur = pq.poll();
            if (cur.to == t) return cur.weight;
            if (dist[cur.to] < cur.weight) continue;
            for (Edge edge : graph.get(cur.to)) {
                long newDist = cur.weight + edge.weight;
                if (newDist < dist[edge.to]) {
                    dist[edge.to] = newDist;
                    pq.add(new Edge(edge.to, (int)newDist));
                }
            }
        }
        return -1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<List<Edge>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) graph.add(new ArrayList<>());
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            graph.get(u).add(new Edge(v, w * 2));
            graph.get(v).add(new Edge(u, w * 2));
        }
        int s = sc.nextInt();
        int t = sc.nextInt();
        long ans = dijkstra(s, t, graph);
        System.out.println(ans == -1 ? -1 : ans % MOD);
        if (ans != -1) {
            List<Integer> path = new ArrayList<>();
            int cur = t;
            while (cur != s) {
                path.add(cur);
                for (Edge edge : graph.get(cur)) {
                    if (dijkstra(s, t, graph) - edge.weight == dijkstra(s, edge.to, graph) + edge.weight) {
                        cur = edge.to;
                        break;
                    }
                }
            }
            path.add(cur);
            Collections.reverse(path);
            System.out.println(path.size());
            for (int p : path) System.out.print(p + " ");
            System.out.println();
        }
    }
}
