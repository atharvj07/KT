#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
typedef pair<ll, ll> pll;
#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()
#define clr(x) memset(x, 0, sizeof(x))
#define sz(x) (int)x.size()
#define umap unordered_map
#define uset unordered_set
#define nl '\n'

const int mod = 998244353;
const int N = (int)2e5 + 4;

void solve() {
  ll n, k;
  cin >> n >> k;
  vi(int) c(n);
  for (int i = 0; i < n; ++i) cin >> c[i];
  ll dp[n + 1][k + 1];
  for (int i = 0; i <= n; ++i) fill(dp[i], dp[i] + k + 1, LLONG_MIN);
  dp[0][0] = 0;
  for (int i = 0; i < n; ++i)
    for (int j = 0; j <= k; ++j)
      if (dp[i][j] >= 0) { dp[i + 1][j] = max(dp[i + 1][j], dp[i][j]);
        if (j + 1<=k) dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + c[i]); }
  cout << *max_element(dp[n], dp[n] + k + 1) << endl;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int t = 1;
  // cin >> t;
  while (t--) solve();
  return 0;
}
