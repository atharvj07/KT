t = int(input())
for _ in range(t):
    a, b, k = map(int, input().split())
    boys = list(map(int, input().split()))
    girls = list(map(int, input().split()))

    boy_count = [0] * (a + 1)
    girl_count = [0] * (b + 1)

    for boy in boys:
        boy_count[boy] += 1
    for girl in girls:
        girl_count[girl] += 1

    total_ways = 0
    for i in range(k):
        total_ways += (k - 1 - boy_count[boys[i]] - girl_count[girls[i]] + 2)

    print(total_ways // 2)
