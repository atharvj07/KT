import java.util.*;
import java.io.*;

class Main {
    static final int INF = 1 << 28;
    static int[][] dis;
    static Queue<Integer> q = new LinkedList<>();

    static void bfs(int n, int x, int y) {
        dis[x][y] = 0;
        q.add(x * n + y);
        while (!q.isEmpty()) {
            int u = q.poll();
            int x1 = u / n;
            int y1 = u % n;
            for (int i = -1; i <= 1; i += 2) {
                int x2 = x1 + i;
                if (x2 >= 0 && x2 < n) {
                    if (dis[x2][y1] > dis[x1][y1] + 1) {
                        dis[x2][y1] = dis[x1][y1] + 1;
                        q.add(x2 * n + y1);
                    }
                }
                int y2 = y1 + i;
                if (y2 >= 0 && y2 < n) {
                    if (dis[x1][y2] > dis[x1][y1] + 1) {
                        dis[x1][y2] = dis[x1][y1] + 1;
                        q.add(x1 * n + y2);
                    }
                }
            }
        }
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");

        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        line = br.readLine().split(" ");
        int sx = Integer.parseInt(line[0]) - 1;
        int sy = Integer.parseInt(line[1]) - 1;
        int fx = Integer.parseInt(line[2]) - 1;
        int fy = Integer.parseInt(line[3]) - 1;

        dis = new int[n][n];

        for (int i = 0; i < n; i++) {
            Arrays.fill(dis[i], INF);
        }
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int x = Integer.parseInt(line[0]) - 1;
            int y = Integer.parseInt(line[1]) - 1;
            dis[x][y] = 0;
            q.add(x * n + y);
        }

        bfs(n, sx, sy);

        System.out.println(dis[fx][fy] == INF ? -1 : dis[fx][fy]);
    }
}
