import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt();
        long[] arr = new long[n];
        for(int i = 0; i < n; i++){
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        long sum = 0;
        for(int i = 0; i < n ; i++){
            sum += arr[i] * Math.min(x, arr[i]);
            x = Math.max(x - 1, 1);
        }
        System.out.println(sum);
    }
}
