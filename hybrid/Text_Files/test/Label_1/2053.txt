#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> t(n);
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    sort(t.begin(), t.end());
    long long ans = 0, sum = 0;
    for (int i = 0; i < n; i++) {
        if (sum + t[i] <= t[i]) {
            ans++;
            sum += t[i];
        }
    }
    cout << ans << endl;
    return 0;
}
