import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long cnt1 = in.nextLong(), cnt2 = in.nextLong(), x = in.nextLong(), y = in.nextLong();
        if (x == y) {
            System.out.println((cnt1 + cnt2 + 1) / 2);
            return;
        }
        long res = Math.max(x + 1, y + 1);
        if (res < cnt1 + cnt2) ++res;
        System.out.println(res);
    }
}
