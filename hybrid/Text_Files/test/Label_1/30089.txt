import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        char[][] a = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j);
            }
        }
        int ans = -1;
        for (int l = 0; l < n; l++) {
            for (int r = l; r < n; r++) {
                for (int c = 0; c < m; c++) {
                    for (int d = c; d < m; d++) {
                        boolean ok = true;
                        for (int i = l; i <= r; i++) {
                            for (int j = c; j <= d; j++) {
                                if (a[i][j] == 'W') {
                                    ok = false;
                                }
                            }
                        }
                        for (int i = 0; i < l; i++) {
                            for (int j = 0; j < m; j++) {
                                if (a[i][j] == 'B') {
                                    ok = false;
                                }
                            }
                        }
                        for (int i = r + 1; i < n; i++) {
                            for (int j = 0; j < m; j++) {
                                if (a[i][j] == 'B') {
                                    ok = false;
                                }
                            }
                        }
                        for (int i = 0; i < n; i++) {
                            for (int j = 0; j < c; j++) {
                                if (a[i][j] == 'B') {
                                    ok = false;
                                }
                            }
                        }
                        for (int i = 0; i < n; i++) {
                            for (int j = d + 1; j < m; j++) {
                                if (a[i][j] == 'B') {
                                    ok = false;
                                }
                            }
                        }
                        if (ok) {
                            ans = (r - l + 1) * (d - c + 1);
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
