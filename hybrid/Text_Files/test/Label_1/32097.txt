import java.util.*;

public class Main {
    static char arr[][];
    static boolean rowCrossed[][];
    static boolean colCrossed[][];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        arr = new char[n][m];
        rowCrossed = new boolean[n][m];
        colCrossed = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.next().toCharArray();
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                rowCrossed[i][j] = false;
                colCrossed[i][j] = false;
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!rowCrossed[i][j]) {
                    crossRow(i, j);
                }
            }
        }

        for (int j = 0; j < m; j++) {
            for (int i = 0; i < n; i++) {
                if (!colCrossed[i][j]) {
                    crossCol(i, j);
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!rowCrossed[i][j] && !colCrossed[i][j]) {
                    sb.append(arr[i][j]);
                }
            }
        }

        System.out.println(sb.toString());
    }

    public static void crossRow(int i, int j) {
        for (int k = j + 1; k < arr[0].length; k++) {
            if (arr[i][j] == arr[i][k]) {
                rowCrossed[i][k] = true;
            }
        }
    }

    public static void crossCol(int i, int j) {
        for (int k = i + 1; k < arr.length; k++) {
            if (arr[i][j] == arr[k][j]) {
                colCrossed[k][j] = true;
            }
        }
    }
}
