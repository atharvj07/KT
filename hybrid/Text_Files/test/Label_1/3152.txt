#include <bits/stdc++.h>
using namespace std;

int n, m, k, cnt[101][101][101], ans;
char mp[101][101][101];

int get(int x, int y, int z) {
  if (x < 1 || y < 1 || z < 1 || x > n || y > m || z > k) return 0;
  return mp[x][y][z] - '0';
}

void dfs(int x, int y, int z) {
  if (get(x, y, z)) {
    cnt[x][y][z] = 1;
    dfs(x + 1, y, z);
    dfs(x, y + 1, z);
    dfs(x, y, z + 1);
  }
}

int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      for (int l = 1; l <= k; l++) cin >> mp[i][j][l];
    }
    cin.ignore();
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      for (int l = 1; l <= k; l++) {
        if (!cnt[i][j][l]) dfs(i, j, l);
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      for (int l = 1; l <= k; l++) {
        if (cnt[i][j][l] == 1) {
          if (!get(i + 1, j, l) && !get(i, j + 1, l) && !get(i, j, l + 1)) ans++;
        }
      }
    }
  }
  cout << ans << '\n';
}
