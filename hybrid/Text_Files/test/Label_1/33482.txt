import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        int[] min = new int[n];
        int[] max = new int[n];
        for (int i = 0; i < n; i++) {
            min[i] = i + 1;
            max[i] = i + 1;
        }
        for (int i = 0; i < m; i++) {
            int friend = a[i] - 1;
            if (min[friend] > 1) {
                min[friend]--;
            }
            if (max[friend] < n) {
                max[friend]++;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(min[i] + " " + max[i]);
        }
    }
}
