import sys

input = sys.stdin.read()
n, *a = map(int, input.split())
r, c = 0, 0
b = []
for i in range(0, n, 1):
    k = a[i]
    v = []
    for j in range(i, i+k, 1):
        v += [a[j]]
    v.sort()
    b += [v]

for i in b:
    s = 0
    for j in i:
        if j == 10:
            s *= 2
        s += j%3
        s = min(s, 3)
    r += s
    c += 1
    if c == 10:
        c = 0

print(r)
