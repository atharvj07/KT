t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    total_sum = sum(a)
    i = 0
    while i < len(a):
        if a[i] % x == 0:
            total_sum += (a[i] // x) * x
            a.extend([a[i] // x] * x)
        else:
            break
        i += 1
    print(total_sum)
