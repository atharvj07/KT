import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        boolean[] fountains = new boolean[m + 1];
        int last = -1;
        for (int i = 0; i < n; i++) {
            int l = sc.nextInt(), r = sc.nextInt();
            if (l > r) {
                int tmp = l;
                l = r;
                r = tmp;
            }
            if (last != l) {
                for (int j = last + 1; j <= l; j++) {
                    fountains[j] = !fountains[j];
                }
            }
            for (int j = l + 1; j <= r; j++) {
                fountains[j] = !fountains[j];
            }
            last = r;
        }
        long ans = 0;
        int count = 0;
        for (int i = 1; i <= m; i++) {
            if (fountains[i])
                count++;
            else {
                ans += count * (count + 1) / 2;
                count = 0;
            }
        }
        if (count > 0)
            ans += count * (count + 1) / 2;
        System.out.println(ans);
    }
}
