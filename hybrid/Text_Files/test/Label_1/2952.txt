#include <bits/stdc++.h>
using namespace std;

int n, m, p[100001], sz[100001];

int f(int x) { return p[x] == x; }

int find(int x) { return f(x) ? x : p[x] = find(p[x]); }

void uni(int a, int b) {
  if ((a = find(a)) == (b = find(b))) return;
  if (sz[a] < sz[b]) swap(a, b);
  sz[a] += sz[b];
  p[b] = a;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n >> m;
  for (int i = 0; i <= n; i++) { p[i] = i, sz[i] = 1; }
  for (int i = 0, a, b; i < m; i++) {
    cin >> a >> b;
    if (find(a) != find(b)) uni(a, b);
  }
  unordered_map<int, int> mp;
  for (int i = 1; i <= n; i++) mp[find(i)]++;
  if (mp.size() != 3) {
    cout << -1;
    return 0;
  }
  for (int i = 1; i <= n; i++) {
    for (auto it : mp) {
      if (find(i) != it.first) {
        if (i == 1) {
          cout << 3;
        } else {
          cout << ' ' << 3;
        }
        mp.erase(it.first);
        break;
      }
      if (i == 1) {
        cout << 1;
      } else {
        cout << ' ' << 1;
      }
    }
  }
  cout << '\n';
  return 0;
}
