#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  int m;
  cin >> s >> m;
  vector<int> pref(s.size() + 1);
  for (int i = 0; i < s.size(); i++) pref[i + 1] = pref[i] + (s[i] == '.');
  while (m--) {
    int l, r;
    cin >> l >> r;
    cout << pref[r] - pref[l] << '\n';
  }
}
