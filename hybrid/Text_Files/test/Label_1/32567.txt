import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	    Scanner in = new Scanner(System.in);
	    int n = in.nextInt();
	    String colors = "ROYGBIV";
	    String result = "";
	    
	    for (int i = 0; i < n; i++) {
	        result += colors.charAt(i % 7);
	    }
	    
	    System.out.println(result);
	}
}
