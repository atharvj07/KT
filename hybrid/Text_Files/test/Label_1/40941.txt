from collections import deque

def bfs(start, graph, n):
    distance = [-1] * (n + 1)
    queue = deque([start])
    distance[start] = 0
    while queue:
        current = queue.popleft()
        for neighbor in graph[current]:
            if distance[neighbor] == -1:
                distance[neighbor] = distance[current] + 1
                queue.append(neighbor)
    return distance

def max_match(bipartite_graph, n, m):
    match = [-1] * (m + 1)
    result = 0
    for i in range(1, n + 1):
        visited = [False] * (m + 1)
        if bpm(bipartite_graph, i, visited, match):
            result += 1
    return result

def bpm(bp_graph, u, visited, match):
    for v in range(1, len(bp_graph[u])):
        if bp_graph[u][v] and not visited[v]:
            visited[v] = True
            if match[v] == -1 or bpm(bp_graph, match[v], visited, match):
                match[v] = u
                return True
    return False

def solve(n, m, edges, s, b, k, h, spaceships, bases):
    graph = [[] for _ in range(n + 1)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    bipartite_graph = [[0] * (b + 1) for _ in range(s + 1)]

    for i, (x, a, f) in enumerate(spaceships, start=1):
        distances = bfs(x, graph, n)
        for j, (base_x, d) in enumerate(bases, start=1):
            if a >= d and f >= distances[base_x]:
                bipartite_graph[i][j] = 1

    matched = max_match(bipartite_graph, s, b)
    return min(matched * k, s * h)

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
s, b, k, h = map(int, input().split())
spaceships = [list(map(int, input().split())) for _ in range(s)]
bases = [list(map(int, input().split())) for _ in range(b)]

print(solve(n, m, edges, s, b, k, h, spaceships, bases))
