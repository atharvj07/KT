import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] l = new long[n];
        for (int i = 0; i < n; i++) {
            l[i] = in.nextLong();
        }
        String s = in.next();
        long[] t = new long[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'G') {
                t[i] = 5;
            } else if (s.charAt(i) == 'W') {
                t[i] = 3;
            } else {
                t[i] = 1;
            }
        }
        long ans = 0;
        long sta = 0;
        for (int i = 0; i < n; i++) {
            long diff = t[i] - sta;
            if (diff > 0) {
                ans += diff;
            } else {
                ans += t[i];
            }
            sta = t[i];
            ans += l[i];
        }
        System.out.println(ans);
    }
}
