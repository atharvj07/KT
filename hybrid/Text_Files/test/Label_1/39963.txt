mod=998244353
n=int(input())
s=input()
t=sorted(s)
ans=0
for i in range(n):
    for j in range(n):
        if i==j:continue
        if s[i]==s[j]:
            x=t[:]
            x.pop(i)
            x.pop(j-1)
            if len(set(x))<=1:
                ans+=1
                ans%=mod
print(ans)
