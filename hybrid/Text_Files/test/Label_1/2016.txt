#include<bits/stdc++.h>
using namespace std;
const double eps = 1e-9;
int main() {
    int n, p; cin >> n >> p;
    multiset<double> t;
    double ans = -1, sum = 0;
    for (int i = 0, a, b; i < n; i++) {
        cin >> a >> b;
        if (a > p) {
            t.insert((double)b / (a - p));
        }
        else sum += (double)b / a;
    }
    while (!t.empty()) {
        double cur = *t.begin();
        t.erase(t.begin());
        if (sum >= cur) {
            ans = max(ans, cur);
            sum -= cur;
        }
        else {
            ans = max(ans, sum + (cur - sum) / 2);
            break;
        }
    }
    if (ans > eps) cout << setprecision(9) << fixed << ans << endl;
    else cout << -1 << endl;
}
