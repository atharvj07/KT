from functools import reduce
import sys

input = sys.stdin.readline

n, m = map(int, input().split())
fac = [0] * (n + 2)
inv = [0] * (n + 2)

def powmod(a: int, p: int, mod: int):
    if p == 0: return 1
    if p == 1: return a
    val = pow(a, p >> 1, mod)
    val = (val * val) % mod
    if p % 2 != 0: val = (val * a) % mod
    return val

def pre():
    fac[0] = 1
    for i in range(1, n + 1):
        fac[i] = (i * fac[i - 1]) % m
    inv[n] = powmod(fac[n], m - 2, m)
    for i in range(n - 1, -1, -1):
        inv[i] = (inv[i + 1] * (i + 1)) % m

pre()
print((fac[n] * powmod(2, n * (n + 1) // 2 - n, m)) % m)
