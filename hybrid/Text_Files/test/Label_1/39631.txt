import sys

input = sys.stdin.readline
n, s, m, k = map(int, input().split())
a = list(map(int, input().split()))
seg = [[] for i in range(n + 1)]
for i in range(s):
    l, r = map(int, input().split())
    seg[l].append(r)

for i in range(1, n + 1):
    seg[i].extend(seg[i - 1])

ans = 1e9
for i in range(m, s + 1):
    b = []
    for j in range(n + 1):
        if len(seg[j]) >= i:
            b.append(a[j - 1])
    if len(b) < k: continue
    b.sort()
    ans = min(ans, b[k - 1])

if ans == 1e9: print(-1)
else: print(ans)
