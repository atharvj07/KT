import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        System.out.println(solve(w));
    }

    private static String solve(int w) {
        ArrayList<Integer> weights = new ArrayList<>();
        for (int i = 0; i < 4; i++) {
            weights.add((int) Math.pow(3, i));
        }
        int n = weights.size();
        StringBuilder sb = new StringBuilder();
        for (int i = n - 1; i >= 0; i--) {
            if (w >= weights.get(i)) {
                sb.append('+');
                w -= weights.get(i);
            } else if (w > 0) {
                sb.append('-');
            } else {
                sb.append('0');
            }
        }
        while (sb.charAt(0) == '0') {
            sb.deleteCharAt(0);
        }
        return sb.toString();
    }
}
