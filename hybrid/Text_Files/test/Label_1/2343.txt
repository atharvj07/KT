#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> c(n);
    for (int i = 0; i < n; i++) cin >> c[i];
    long long mx = *max_element(c.begin(), c.end());
    long long ans = 0;
    for (int i = 0; i < n; i++) ans += (mx - c[i] + 1) / 2;
    cout << ans << endl;
}
