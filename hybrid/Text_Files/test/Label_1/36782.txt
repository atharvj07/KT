def solve():
    q = int(input())
    for _ in range(q):
        n, k = map(int, input().split())
        s = input()
        mx_pref = [-1] * n
        mn_suff = [n] * n
        mn_pref = mx_suff = [0] * n
        cur = 3
        for i in range(n):
            if i and s[i] == s[i - 1]:
                mx_pref[i] = mx_pref[i - 1]
            if (i + 1) % 3 == 0:
                mx_pref[i] = i
            if (i >= 2 and s[i] == s[i - 1] == s[i - 2]):
                mn_pref[i] = mn_pref[i - 1]
            if s[i] != cur:
                mn_pref[i] = i
            cur = (cur + 1) % 3
        cur = 0
        for i in range(n - 1, -1, -1):
            if i < n - 1 and s[i] == s[i + 1]:
                mn_suff[i] = mn_suff[i + 1]
            if (n - i) % 3 == 0:
                mn_suff[i] = i
            if (i <= n - 3 and s[i] == s[i + 1] == s[i + 2]):
                mx_suff[i] = mx_suff[i + 1]
            if s[i] != cur:
                mx_suff[i] = i
            cur = (cur + 1) % 3
        ans = n
        for i in range(n - k + 1):
            if mn_suff[i] <= mx_pref[i + k - 1]:
                ans = min(ans, mn_pref[i + k - 1] - mn_suff[i])
        print(ans)

solve()
