#include <iostream>
using namespace std;
int main() {
  int n, bad = 0;
  cin >> n;
  string s;
  while (n--) {
    cin >> s;
    for (int i = 0; i < s.size(); i++)
      if (s[i] == 'x' && (i <= 0 || s[i - 1] != 'o') && (i != s.size() - 1 || s[i + 1] != 'o') && (i <= 0 || i == s.size() - 1 || s[i - 1] != 'o' || s[i + 1] != 'o')) bad = 1;
  }
  cout << (bad ? "NO" : "YES") << endl;
  return 0;
}
