import java.util.Scanner;
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> x = new HashMap<>();
        Map<Integer, Integer> y = new HashMap<>();
        Map<Integer, Character> u = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            int x_i = sc.nextInt();
            int y_i = sc.nextInt();
            char u_i = sc.next().charAt(0);
            x.put(i, x_i);
            y.put(i, y_i);
            u.put(i, u_i);
        }
        boolean collision = false;
        int time = -1;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (u.get(i) == u.get(j)) {
                    continue;
                }
                if (u.get(i) == 'U' && u.get(j) == 'D') {
                    if (x.get(i) == x.get(j)) {
                        if (y.get(i) > y.get(j)) {
                            int t = (y.get(i) - y.get(j)) / 2;
                            if (t < time || time == -1) {
                                time = t;
                            }
                        }
                    }
                } else if (u.get(i) == 'R' && u.get(j) == 'L') {
                    if (y.get(i) == y.get(j)) {
                        if (x.get(i) > x.get(j)) {
                            int t = (x.get(i) - x.get(j)) / 2;
                            if (t < time || time == -1) {
                                time = t;
                            }
                        }
                    }
                } else if (u.get(i) == 'D' && u.get(j) == 'U') {
                    if (x.get(i) == x.get(j)) {
                        if (y.get(i) < y.get(j)) {
                            int t = (y.get(j) - y.get(i)) / 2;
                            if (t < time || time == -1) {
                                time = t;
                            }
                        }
                    }
                } else if (u.get(i) == 'L' && u.get(j) == 'R') {
                    if (y.get(i) == y.get(j)) {
                        if (x.get(i) < x.get(j)) {
                            int t = (x.get(j) - x.get(i)) / 2;
                            if (t < time || time == -1) {
                                time = t;
                            }
                        }
                    }
                }
            }
        }
        if (time == -1) {
            System.out.println("SAFE");
        } else {
            System.out.println(time);
        }
    }
}
