#include <algorithm>
#include <cmath>
#include <iostream>
#include <vector>

using namespace std;

struct Point { long long x, y; };

int n, r;
vector<Point> points;

bool cmp_x(Point a, Point b) { return a.x < b.x; }
bool cmp_y(Point a, Point b) { return a.y < b.y; }

int main() {
    cin >> n >> r;
    points.resize(n);
    for (int i = 0; i < n; i++) cin >> points[i].x >> points[i].y;
    sort(points.begin(), points.end(), cmp_x);
    int ans = 1;
    for (int i = 0, j = 0; i < n; i++) {
        while (j < n && points[j].x - points[i].x <= r) j++;
        ans = max(ans, j - i);
    }
    sort(points.begin(), points.end(), cmp_y);
    ans = 1;
    for (int i = 0, j = 0; i < n; i++) {
        while (j < n && points[j].y - points[i].y <= r) j++;
        ans = max(ans, j - i);
    }
    cout << ans << endl;
    return 0;
}
