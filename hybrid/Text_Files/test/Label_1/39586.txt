h, w = map(int, input().split())
a = [[i for i in input()] for _ in range(h)]
x = any([a[i][j] == '*' and ((0 <= i - 1 < h and a[i - 1][j] == '*') or (0 <= i + 1 < h and a[i + 1][j] == '*') or (0 <= j - 1 < w and a[i][j - 1] == '*') or (0 <= j + 1 < w and a[i][j + 1] == '*')) for i in range(h) for j in range(w)])
y = sum([sum([1 for j in range(w) if a[i][j] == '*']) for i in range(h)]) == 1
print('YES' if x and y else 'NO')
