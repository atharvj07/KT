import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt(), t = sc.nextInt(), r = sc.nextInt(), ans = 0, time = 0;
        int[] visits = new int[m];
        for (int i = 0; i < m; i++) {
            visits[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            if (visits[i] >= time) {
                time = visits[i] + t;
            } else {
                time++;
                ans++;
                time += t;
            }
        }
        System.out.println(ans >= r ? ans : -1);
    }
}
