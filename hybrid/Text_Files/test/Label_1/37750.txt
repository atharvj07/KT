import sys

# Read the number of test cases
t = int(sys.stdin.readline())

for _ in range(t):
    # Read the number of players
    n = int(sys.stdin.readline())

    # Read the string of attack directions
    s = sys.stdin.readline().strip()

    # Count the number of players that are not attacking logically
    cnt = 0
    for i in range(n):
        if s[i] == 'R' and s[(i + 1) % n] == 'L':
            cnt += 1
        elif s[i] == 'L' and s[(i - 1) % n] == 'R':
            cnt += 1

    # Print the minimum number of players to talk to
    print(cnt // 2)
