import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            y[i] = sc.nextInt();
        }
        int xor = 0;
        for (int i = 0; i < n; i++) {
            xor ^= x[i] ^ y[i];
        }
        System.out.println(xor == 0 ? "Koyomi" : "Karen");
    }
}
