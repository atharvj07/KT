#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    vector<vector<int>> grid(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> grid[i][j];
        }
    }
    vector<int> row_max(n);
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < m; j++) {
            cnt += grid[i][j];
        }
        row_max[i] = cnt;
    }
    int ans = 0;
    for (int i = 0; i < q; i++) {
        int x, y;
        cin >> x >> y;
        x--, y--;
        grid[x][y] ^= 1;
        int cnt = 0;
        for (int j = 0; j < m; j++) {
            cnt += grid[x][j];
        }
        row_max[x] = max(row_max[x], cnt);
        ans = max(ans, row_max[x]);
        cout << ans << endl;
    }
    return 0;
}
