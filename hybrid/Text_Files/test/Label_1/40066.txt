from collections import deque, defaultdict
import sys

n, a, b = int(input()), str(input()), str(input())
if a == b:
    print(0)
else:
    q = deque([(a, 0)])
    d = defaultdict(bool)
    d[a] = True
    while q:
        s, c = q.popleft()
        if s == b:
            print(c)
            for i in range(min(c, 100000)):
                d1, s1 = q[i]
                print(d1, s1)
            sys.exit(0)
        for i in range(n - 1):
            if d[s] == 0:
                if s[i] != '0' or s[i + 1] != '9':
                    d[s] = True
                    q.append((s[:i] + str(int(s[i]) + 1) + s[i + 1] + s[i + 2:], c + 1))
                if s[i] != '9' or s[i + 1] != '0':
                    d[s] = True
                    q.append((s[:i] + str(int(s[i]) - 1) + s[i + 1] + s[i + 2:], c + 1))
    print(-1)
