#include <bits/stdc++.h>
#define int long long
using namespace std;

const int maxN = 2e5 + 5;
vector<pair<int, int>> frq(maxN);

int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, ans = 0, tpc[maxN] = {0}, diff[maxN] = {0};
        cin >> n;
        for (int i = 0; i < n; i++) {
            int a, b;
            cin >> a >> b;
            frq[a].first++;
            frq[b].second++;
            tpc[a]++;
            diff[b]++;
        }
        for (int i = 1; i <= n; i++) {
            ans += frq[i].first * (frq[i].first - 1) * (frq[i].first - 2) / 6;
            ans += frq[i].second * (frq[i].second - 1) * (frq[i].second - 2) / 6;
        }
        for (int i = 1; i <= n; i++) {
            ans += tpc[i] * (tpc[i] - 1) * (tpc[i] - 2) / 6;
            ans += diff[i] * (diff[i] - 1) * (diff[i] - 2) / 6;
        }
        cout << ans << '\n';
        for (int i = 1; i <= n; i++) {
            frq[i].first = frq[i].second = 0;
            tpc[i] = diff[i] = 0;
        }
    }
}
