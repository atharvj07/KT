r = int(input())

# We start by solving for x in the equation x^2 + 2xy + x + 1 = r
# This simplifies to x^2 + x(2y + 1) + 1 - r = 0, which is a quadratic equation in terms of x
# Applying the quadratic formula, we solve for x

found = False
for x in range(1, int(r**0.5) + 1):
    # We check if the resulting value of y is an integer by ensuring the discriminant is a perfect square
    # and (r - 1 - x) is divisible by (2x + 1)
    if (r - 1 - x) % (2*x + 1) == 0:
        y = (r - 1 - x) // (2*x + 1)
        if y > 0:
            print(x, y)
            found = True
            break

if not found:
    print("NO")
