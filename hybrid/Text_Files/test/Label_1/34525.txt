import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    public static class Creature {
        private int num;
        private int[] characs;

        public Creature(int num, int[] chars) {
            this.num = num;
            this.characs = chars;
        }

        public int getCharac(int i) {
            return characs[i];
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());

        List<Creature> creatures = new ArrayList<>();
        int cnt = 1;
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            int[] chars = new int[n];
            for (int j = 0; j < n; j++) {
                chars[j] = Integer.parseInt(st.nextToken());
            }
            creatures.add(new Creature(cnt++, chars));
        }

        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());

            if (type == 1) {
                Creature c1 = creatures.get(x - 1);
                Creature c2 = creatures.get(y - 1);
                int[] newChars = new int[n];
                for (int j = 0; j < n; j++) {
                    newChars[j] = Math.max(c1.getCharac(j), c2.getCharac(j));
                }
                creatures.add(new Creature(cnt++, newChars));
            } else if (type == 2) {
                Creature c1 = creatures.get(x - 1);
                Creature c2 = creatures.get(y - 1);
                int[] newChars = new int[n];
                for (int j = 0; j < n; j++) {
                    newChars[j] = Math.min(c1.getCharac(j), c2.getCharac(j));
                }
                creatures.add(new Creature(cnt++, newChars));
            } else {
                System.out.println(creatures.get(x - 1).getCharac(y - 1));
            }
        }
    }
}
