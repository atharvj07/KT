import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = reader.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        String[] a = reader.readLine().split(" ");
        boolean incorrect = false;
        List<Integer> restoredA = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (a[i] == "?") {
                int sum = restoredA.stream().mapToInt(Integer::intValue).sum();
                restoredA.add(k * i - sum);
            } else {
                int num = Integer.parseInt(a[i]);
                int sum = restoredA.stream().mapToInt(Integer::intValue).sum();
                if ((k * i - sum) != num) {
                    incorrect = true;
                    break;
                }
                restoredA.add(num);
            }
        }
        int sumA = Math.abs(restoredA.stream().mapToInt(Integer::intValue).sum());
        if (incorrect || (k > 1 && restoredA.stream().anyMatch(num -> num < 0))) {
            System.out.println("Incorrect sequence");
        } else {
            System.out.println(restoredA.stream().map(Object::toString).collect(Collectors.joining(" ")));
        }
    }
}
