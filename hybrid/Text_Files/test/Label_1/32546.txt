import java.util.Scanner;

public class BerryBaskets {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int maxBaskets = 0;
        for (int i = 0; i < n; i++) {
            int red = scanner.nextInt();
            int blue = scanner.nextInt();

            // Count baskets for berries from the same shrub
            maxBaskets += Math.min(red / k, blue / k);

            // Count baskets for red berries
            maxBaskets += red / k;

            // Count baskets for blue berries
            maxBaskets += blue / k;
        }

        System.out.println(maxBaskets);
    }
}
