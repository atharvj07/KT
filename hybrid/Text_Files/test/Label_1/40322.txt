import sys

n = int(input())
vasya = list(map(int, input().split()))
petya = list(map(int, input().split()))
problems = [0]*5
for i in range(1, n):
    for j in range(5):
        if petya[j] != -1:
            problems[j] += 1
for j in range(5):
    problems[j] *= 1000
problems[0] -= (problems[0]*vasya[0]/250)
problems[1] -= (problems[1]*vasya[1]/250)
problems[2] -= (problems[2]*vasya[2]/250)
problems[3] -= (problems[3]*vasya[3]/250)
problems[4] -= (problems[4]*vasya[4]/250)
problems.sort()
if petya[0] == -1: problems[0] = 0
if petya[1] == -1: problems[1] = 0
if petya[2] == -1: problems[2] = 0
if petya[3] == -1: problems[3] = 0
if petya[4] == -1: problems[4] = 0
ans = 0
for i in range(5):
    if problems[i] <= petya[i]:
        ans += (petya[i]-problems[i])/1000+1
        problems[i] = petya[i]+1
if ans > 1e9+7: ans = -1
print(int(ans))
