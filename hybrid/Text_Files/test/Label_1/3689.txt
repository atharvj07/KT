#include<bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, a, b;
        cin >> n >> a >> b;
        if ((a + b + 2 - n) & 1) {
            cout << -1 << endl;
        } else {
            vector<int> v;
            for (int i = 1; i <= n; i++) {
                if (i != a && i != b) v.push_back(i);
            }
            v.push_back(b);
            v.push_back(a);
            for (auto x : v) cout << x << " ";
            cout << endl;
        }
    }
    return 0;
}
