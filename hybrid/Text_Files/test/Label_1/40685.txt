import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
dp = [0] * n
dp[n - 1] = 1 if a[n - 1] != 0 else 0
for i in range(n - 2, -1, -1):
    if a[i] != 0: dp[i] = dp[i + 1] + 1
    elif dp[i + 1] > 0: dp[i] = 1
ans = 0
for i in range(n):
    if i + dp[i] > n - 1: ans += dp[i]
    elif a[i + dp[i]] != 0: ans += dp[i]
print(ans)
