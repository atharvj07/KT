#include <iostream>
#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define vl vector<long long>
#define vll vector<long long>
bool compare(const pair<long long, long long> &a, const pair<long long, long long> &b) { return a.second > b.second; }
int main() {
  ll w, n, m;
  cin >> n >> m;
  vector<pair<long long, long long>> v;
  for (ll i = 0; i < n; i++) {
    ll c;
    cin >> w >> c;
    v.push_back({w, c});
  }
  sort(v.begin(), v.end(), compare);
  long long sum = 0ll;
  for (pair<long long, long long> x : v) {
    if (m - x.first >= 0) {
      sum += x.second;
      m -= x.first;
    }
  }
  cout << sum << endl;
  return 0;
}
