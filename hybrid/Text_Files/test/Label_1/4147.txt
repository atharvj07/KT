#include <bits/stdc++.h>

#define rep(i, a, b) for (int i = a; i < (b); ++i)
using namespace std;
using ll = long long;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, x, y;
    cin >> n >> x >> y;
    string a;
    cin >> a;
    int j = n - y - 1;
    int k = n - x - 1;
    int ans = 0;
    for (int i = n - 1; i >= 0; --i) {
        if (i == j and a[i] == '1') {
            ans += 1;
            j -= x - y;
        }
        if (i == k and a[i] == '1') {
            ans += 1;
            k -= x - y;
        }
        if (i <= j || i <= k) {
            if (a[i] == '0') {
                ans += 1;
                a[i] = '1';
            }
        }
    }
    cout << ans << endl;
    return 0;
}
