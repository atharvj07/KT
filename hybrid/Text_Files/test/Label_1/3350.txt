#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<string> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<vector<int>> vis(n, vector<int>(m, 0));
    bool ok = true;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] == '*') {
          if (i - k < 0 || j - k < 0 || i + k >= n || j + k >= m) {
            ok = false;
            break;
          }
          for (int l = -k; l <= k; l++) {
            if (a[i + l][j + l] != '*') {
              ok = false;
              break;
            }
          }
          if (!ok) break;
          for (int l = -k; l <= k; l++) {
            for (int r = -k; r <= k; r++) vis[i + l][j + r] = 1;
          }
        }
      }
      if (!ok) break;
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] == '*' && !vis[i][j]) {
          ok = false;
          break;
        }
      }
      if (!ok) break;
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
  return 0;
}
