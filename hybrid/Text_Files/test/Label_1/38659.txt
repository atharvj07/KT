from collections import defaultdict

n, m = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]
adj = defaultdict(list)

for a, b in edges:
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n + 1)
used = set()

def dfs(u, par):
    vis[u] = True
    for v in adj[u]:
        if not vis[v]:
            used.add((min(u, v), max(u, v)))
            dfs(v, u)

dfs(1, 1)
for a, b in sorted(used):
    print(a, b)
