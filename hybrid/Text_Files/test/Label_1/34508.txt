import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        if (s.charAt(0) == s.charAt(n - 1)) {
            System.out.println(1);
        } else {
            System.out.println(0);
        }
    }
}
