import sys
input = sys.stdin.readline

N,K = map(int,input().split())
X = sorted(list(map(int,input().split())))
A = int(input())
C = list(map(int,input().split()))

s = 0
ans = 1e9
for i in range(len(C)):
    if X[i]>K:
        ans = min(ans,s+C[i])
    K+=A
    s+=C[i]

if ans == 1e9:
    ans = -1
print(ans)
