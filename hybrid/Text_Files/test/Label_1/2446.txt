#include <bits/stdc++.h>
#define EPS 1e-9
using namespace std;
const int N = 1005;
struct point {
  double x, y;
  point() : x(0), y(0) {}
  point(double X, double Y) : x(X), y(Y) {}
};
double dist(point A, point B) { return sqrt((A.x - B.x) * (A.x - B.x) + (A.y - B.y) * (A.y - B.y)); }
int n, r;
double x[N], y[N];
int main() {
  cin >> n >> r;
  for (int i = 1; i <= n; i++) cin >> x[i];
  y[1] = r;
  for (int i = 2; i <= n; i++) {
    point p(x[i], 1e100);
    for (int j = 1; j < i; j++) {
      if (dist(point(x[j], y[j]), p) <= 2.0 * r) {
        double temp = dist(point(x[j], y[j]), p);
        y[i] = max(y[i], sqrt((temp * temp) - (4.0 * r * r)) + y[j]);
      }
    }
    y[i] += r;
  }
  for (int i = 1; i <= n; i++) {
    if (fabs(y[i] - r) < EPS)
      cout << 0 << " ";
    else
      cout << fixed << setprecision(12) << y[i] << " ";
  }
  return 0;
}
