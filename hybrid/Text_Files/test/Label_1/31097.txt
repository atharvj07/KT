import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        if (l + 2 > r) {
            System.out.println(-1);
        } else if (l + 1 == r) {
            System.out.println(l + " " + (l + 1));
        } else {
            System.out.println(l + " " + (l + 1) + " " + (l + 2));
        }
    }
}
