import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        List<Integer> b = new ArrayList<>();
        for (int i = 0; i < n;) {
            int j = i + 1;
            while (j < n && a[j] == a[i]) {
                j++;
            }
            if (j - i > 1) {
                i = j - (j - i) / 2;
            } else {
                b.add(a[i]);
                i = j;
            }
        }
        System.out.println(b.size());
        for (int x : b) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
