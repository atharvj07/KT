import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[][] pts = new int[n][2];
        for (int i = 0; i < n; i++) {
            pts[i][0] = s.nextInt();
            pts[i][1] = s.nextInt();
        }
        s.close();
        Arrays.sort(pts, (a, b) -> Integer.compare(a[0], b[0]));
        long ans = 0;
        for (int i = 1; i < n; i++) {
            int dx = pts[i][0] - pts[i - 1][0];
            int dy = pts[i][1] - pts[i - 1][1];
            ans += Math.min(dx, dy);
        }
        System.out.println(ans);
    }
}
