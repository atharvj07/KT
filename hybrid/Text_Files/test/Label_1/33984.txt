import java.util.*;
import java.io.*;

public class Main {
    static int n, k;
    static int[][] d;
    static List<Integer>[] adj;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        d = new int[k][n];
        adj = new List[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(in.readLine());
            for (int j = 0; j < n; j++) {
                d[i][j] = Integer.parseInt(st.nextToken());
            }
        }
        boolean ok = true;
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                int dist = 0;
                for (int l = 0; l < n; l++) {
                    dist += Math.abs(d[i][l] - d[j][l]);
                }
                if (dist % 2 == 1) {
                    ok = false;
                }
            }
        }
        if (!ok) {
            System.out.println(-1);
            return;
        }
        int[] deg = new int[n];
        for (int i = 0; i < k; i++) {
            for (int j = i + 1; j < k; j++) {
                if (!adj[i].contains(j)) {
                    adj[i].add(j);
                    adj[j].add(i);
                    deg[i]++;
                    deg[j]++;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (deg[i] == 1) {
                for (int j : adj[i]) {
                    adj[j].remove((Integer) i);
                    deg[j]--;
                }
                adj[i] = null;
            }
        }
        List<Integer> leaves = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (adj[i] != null) {
                leaves.add(i);
            }
        }
        while (leaves.size() > 1) {
            int a = leaves.remove(0);
            int b = leaves.remove(0);
            System.out.println(a + 1 + " " + (b + 1));
            adj[a].remove((Integer) b);
            adj[b].remove((Integer) a);
            if (adj[a].size() == 1) {
                leaves.add(a);
            }
            if (adj[b].size() == 1) {
                leaves.add(b);
            }
        }
    }
}
