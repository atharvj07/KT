n = int(input())
grid = [list(map(int, input().split())) for _ in range(n)]

# Find the row and column of the empty cell
empty_row, empty_col = 0, 0
for i in range(n):
    for j in range(n):
        if grid[i][j] == 0:
            empty_row, empty_col = i, j
            break

# If n == 1, it's a special case, output 1
if n == 1:
    print(1)
else:
    # Calculate the target sum for each row/column/diagonal
    target_sum = 0
    for i in range(n):
        if i != empty_row:
            target_sum = sum(grid[i])
            break
    
    # Calculate the sum for the row and column with the empty cell
    sum_row = sum(grid[empty_row])
    sum_col = sum(grid[i][empty_col] for i in range(n))

    # Find the value that needs to be filled
    x = target_sum - sum_row

    # Fill in the value and check if the grid becomes a magic square
    grid[empty_row][empty_col] = x

    is_magic_square = True
    # Check rows and columns
    for i in range(n):
        if sum(grid[i]) != target_sum or sum(grid[j][i] for j in range(n)) != target_sum:
            is_magic_square = False
            break

    # Check diagonals
    if sum(grid[i][i] for i in range(n)) != target_sum or sum(grid[i][n-i-1] for i in range(n)) != target_sum:
        is_magic_square = False

    # Output result
    if is_magic_square and x > 0:
        print(x)
    else:
        print(-1)
