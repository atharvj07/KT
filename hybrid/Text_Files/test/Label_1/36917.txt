import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

if min(a) == 0:
    print(-1)
    exit()

queue = [1]
used = [False] * (n + 1)
used[1] = True
ans = []

while queue:
    u = queue.pop(0)
    for v in range(2, n + 1):
        if used[v] or a[u - 1] == 0:
            continue
        a[u - 1] -= 1
        ans.append([u, v])
        used[v] = True
        queue.append(v)

print(len(ans))
for u, v in ans:
    print(u, v)
