import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[][] points = new long[n][5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 5; j++) {
                points[i][j] = input.nextInt();
            }
        }
        int goodPoints = 0;
        for (int i = 0; i < n; i++) {
            boolean isGood = true;
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                for (int k = 0; k < n; k++) {
                    if (j == k || i == k) continue;
                    long dotProduct = 0;
                    for (int l = 0; l < 5; l++) {
                        dotProduct += points[i][l] * points[j][l] - points[i][l] * points[k][l];
                    }
                    if (dotProduct > 0) {
                        isGood = false;
                        break;
                    }
                }
            }
            if (isGood) {
                goodPoints++;
            }
        }
        System.out.println(goodPoints);
        for (int i = 0; i < n; i++) {
            boolean isGood = true;
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                for (int k = 0; k < n; k++) {
                    if (j == k || i == k) continue;
                    long dotProduct = 0;
                    for (int l = 0; l < 5; l++) {
                        dotProduct += points[i][l] * points[j][l] - points[i][l] * points[k][l];
                    }
                    if (dotProduct > 0) {
                        isGood = false;
                        break;
                    }
                }
            }
            if (isGood) {
                System.out.println(i + 1);
            }
        }
    }
}
