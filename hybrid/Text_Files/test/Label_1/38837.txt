n = int(input())
a = [int(i) for i in input().split()]
d = [0 for i in range(n)]
for i in range(n-1):
    d[i] = a[i+1] - a[i]
for i in range(n-2):
    if d[i]>0 and d[i+1]<0:
        for j in range(i+1,n-1):
            if d[j]<0:
                l = j
                break
        for j in range(i,-1,-1):
            if d[j]>0:
                r = j
                break
        print(l+1,r+1)
        for j in range(l,r-1):
            d[j] = d[j+1]
        d[r-1] = 0
