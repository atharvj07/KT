import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            y[i] = sc.nextInt();
        }
        int[] xor = new int[2 * n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                xor[i ^ j]++;
            }
        }
        int count = 0;
        for (int i = 0; i < 2 * n; i++) {
            if (xor[i] % 2 == 1)
                count++;
        }
        System.out.println(count % 2 == 0 ? "Karen" : "Koyomi");
    }
}
