#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1e9+7;
ll bigmod(ll a,ll b)
{
    if(!b) return 1;
    if(b==1) return a%mod;
    ll x = bigmod(a,b/2);
    x = (x*x)%mod;
    if(b%2) x=(x*a)%mod;
    return x;
}
int32_t main()
{
    ll a,b,p,x;
    cin>>a>>b>>p>>x;
    ll ans=0;
    for(ll i = 1 ;i<=x ;i++)
    {
        ll temp = i*a;
        temp%=p;
        if(temp==b) ans++;
    }
    cout<<ans<<endl;
}
