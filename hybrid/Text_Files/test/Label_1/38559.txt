import sys
input = sys.stdin.readline

n = int(input())
s = [input() for _ in range(n)]
S = set(s)
a = []
for t in S:
    st = []
    ok = 1
    for i in t:
        if i == '(':
            st.append(i)
        elif not st:
            ok = 0
            break
        else:
            st.pop()
    a.append([ok,t])
ans = 0
for i in range(len(a)):
    if not a[i][0]:
        continue
    for j in range(i+1,len(a)):
        if a[i][1]+a[j][1] in S:
            ans += 1
print(ans)
