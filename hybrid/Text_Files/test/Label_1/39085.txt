t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    prefix_sum = [0] * n
    prefix_sum[0] = a[0]
    for i in range(1, n):
        prefix_sum[i] = prefix_sum[i - 1] + a[i]
    possible_winners = []
    for i in range(n - 1):
        if prefix_sum[i] * 2 >= a[i + 1]:
            possible_winners.append(i)
    possible_winners.append(n - 1)
    print(len(possible_winners))
    print(*(i + 1 for i in possible_winners))
