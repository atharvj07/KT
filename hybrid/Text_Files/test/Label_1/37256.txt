s=input()
ans=0
las=[0]*10
p=[0]*len(s)
las[int(s[len(s)-1])]=len(s)-1
for i in range(len(s)-2,-1,-1):
    x=int(s[i])
    if (x%4==0):
        ans+=las[0]+1
    if (x*10+las[x%10]%4==0):
        ans+=las[(x*10+las[x%10]%4)%10]-las[x%10]
    if (x*100+las[(x*10+las[x%10]%4)%10]%4==0):
        ans+=las[(x*100+las[(x*10+las[x%10]%4)%10]%4)%10]-las[(x*10+las[x%10]%4)%10]
    if (x*1000+las[(x*100+las[(x*10+las[x%10]%4)%10]%4)%10]%4==0):
        ans+=las[(x*1000+las[(x*100+las[(x*10+las[x%10]%4)%10]%4)%10]%4)%10]-las[(x*100+las[(x*10+las[x%10]%4)%10]%4)%10]
    las[x]=i
print(ans)
