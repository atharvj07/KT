import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] deg = new int[n + 1];
    for (int i = 0; i < n - 1; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      deg[a]++;
      deg[b]++;
    }
    long ans = 0;
    for (int i = 1; i <= n; i++) {
      ans += 1L * deg[i] * (deg[i] - 1) / 2;
    }
    System.out.println(ans);
  }
}
