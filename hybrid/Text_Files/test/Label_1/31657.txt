import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = input.nextInt();
        int xor = 0;
        for (int i = 0; i < n; i++)
            xor ^= a[i];
        if (xor == 0 || n % 2 == 0)
            System.out.println("BitAryo");
        else
            System.out.println("BitLGM");
    }
}
