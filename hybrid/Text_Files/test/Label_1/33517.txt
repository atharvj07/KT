import java.util.*;

public class Main {

  static final int MAX_N = 2000;
  static int[][] count = new int[MAX_N][MAX_N];

  public static void main(String[] args) {
    try (Scanner sc = new Scanner(System.in)) {
      int n = Integer.parseInt(sc.nextLine());
      int[] a = new int[n];
      int[] b = new int[n];
      String[] lineA = sc.nextLine().split(" ");
      String[] lineB = sc.nextLine().split(" ");
      for (int i = 0; i < n; i++) {
        a[i] = Integer.parseInt(lineA[i]);
        b[i] = Integer.parseInt(lineB[i]);
      }

      calcCount(a, b, n);

      int sum = 0;
      for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
          sum += count[i][j];
        }
      }

      System.out.println(sum % 2 == 0 ? "Karen" : "Koyomi");
    }
  }

  private static void calcCount(int[] a, int[] b, int n) {
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        int c = a[i] ^ b[j];
        for (int k = 0; k < n; k++) {
          count[i][k] += (c == a[k]) ? 1 : 0;
          count[j][k] += (c == b[k]) ? 1 : 0;
        }
      }
    }
  }

}
