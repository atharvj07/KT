#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define ll long long

const int N = 2e5 + 5;
vector<int> adj[N];
bool vis[N];
vector<pair<int, int>> edges;
string ans;

bool dfs(int u, int p) {
    vis[u] = 1;
    for (int v : adj[u]) {
        if (v == p) continue;
        if (vis[v]) return false;
        if (!dfs(v, u)) return false;
    }
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        edges.push_back({u, v});
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    bool ok = true;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            if (!dfs(i, i)) {
                ok = false;
                break;
            }
        }
    }
    if (!ok) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for (auto e : edges) {
        if (e.first < e.second) {
            ans += '0';
        }
        else {
            ans += '1';
        }
    }
    cout << ans << endl;
    return 0;
}
