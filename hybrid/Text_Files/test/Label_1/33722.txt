import java.util.Scanner;

public class PetyaAndSorting {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }

        int minSwaps = n - 1;
        int ways = 0;

        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                if (permutation[i] > permutation[j]) {
                    int[] newPermutation = permutation.clone();
                    swap(newPermutation, i, j);
                    int swaps = countSwaps(newPermutation);
                    if (swaps < minSwaps) {
                        minSwaps = swaps;
                        ways = 1;
                    } else if (swaps == minSwaps) {
                        ways++;
                    }
                }
            }
        }

        System.out.println(minSwaps + " " + ways);
    }

    private static void swap(int[] array, int i, int j) {
        int temp = array[i];
        array[i] = array[j];
        array[j] = temp;
    }

    private static int countSwaps(int[] permutation) {
        int swaps = 0;
        for (int i = 1; i < permutation.length; i++) {
            int j = i;
            while (j > 0 && permutation[j] < permutation[j - 1]) {
                swap(permutation, j, j - 1);
                j--;
                swaps++;
            }
        }
        return swaps;
    }
}
