from collections import deque
from itertools import zip_longest
from string import ascii_lowercase

def minify(reserved, source):
    tokens = ['']
    for line in source:
        line = line.split('#', 1)[0].strip()
        tokens.extend(line.split())
    tokens = [token for token in tokens if token]

    def is_num(token):
        return token.isdigit()

    def is_word(token):
        return all(32 < ord(char) < 127 for char in token) and not is_num(token) and token not in reserved

    def rename_word(token, target_words):
        if is_word(token) and token not in reserved:
            return target_words.popleft()
        return token

    # Iterate through the source tokens.
    target_words = deque(ascii_lowercase)
    renamed = []
    for token in tokens:
        if token not in reserved:
            token = rename_word(token, target_words)
        renamed.append(token)

    # Construct the minified source.
    minified = ' '.join(renamed).replace(' )', ')').replace(' (', '(')
    return minified

# Read the input.
n = int(input())
reserved = input().split()
m = int(input())
source = []
for _ in range(m):
    source.append(input())

# Minify the source.
minified = minify(reserved, source)

# Print the minified source.
print(minified)
