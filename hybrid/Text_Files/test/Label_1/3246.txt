#include <iostream>
#include <vector>
#include <map>
using namespace std;
#define ll long long

int main() {
  int n;
  cin >> n;
  map<ll, int> x, y;
  for (int i = 0; i < n; i++) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    x[a]++, x[c]--, y[b]++, y[d]--;
  }
  ll xa = 0, ya = 0, xb = 0, yb = 0;
  for (auto i : x) {
    xa = i.first;
    xb += i.second;
    if (xb == n - 1) break;
  }
  for (auto i : y) {
    ya = i.first;
    yb += i.second;
    if (yb == n - 1) break;
  }
  cout << xa << ' ' << ya << endl;
}
