#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, m; cin >> n >> m;
    string s; cin >> s;
    int ans = 0;
    for (int i = 0; i < n - 1; i++) {
        if (s[i] == s[i + 1] && s[i] == '.') ans++;
    }
    for (int i = 0; i < m; i++) {
        int p; char c; 
        cin >> p >> c; p--;
        int ch = 0;
        if (s[p] == '.' && s[p + 1] == '.') ch = 1;
        if (s[p] == '.' && c != '.' && s[p + 1] == '.') ch = -1;
        if (s[p] != '.' && s[p + 1] == '.' && c == '.') ch = -1;
        if (s[p] != '.' && s[p + 1] == '.' && c != '.') ch = 1;
        s[p] = c;
        ans += ch;
        cout << ans << endl;
    }
}
