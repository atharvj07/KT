#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
ll a[1005];
int main(){
    ll t,n,i,ans,al,bo;
    cin>>t;
    while(t--){
        ans=al=bo=0;
        cin>>n;
        for(i=0;i<n;i++)cin>>a[i];
        for(i=0;i<n;i++){
            if(al>bo){
                bo+=a[i];
                ans++;
            }
            else{
                al+=a[i];
                ans++;
            }
            if(al==bo)break;
        }
        cout<<ans<<" "<<al<<" "<<bo<<endl;
    }
}
