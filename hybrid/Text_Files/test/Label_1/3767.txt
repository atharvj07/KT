#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    while(n--){
        int a,flag=1;
        cin>>a;
        int p=a,c=0;
        while(p>0){
            c+=p%10;
            p/=10;
        }
        if(c<=k) flag=0;
        while(a>0){
            if(a%10>k)  {flag=1; break;}
            a/=10;
        }
        if(!flag){ cout<<1; return 0;}
    }
    cout<<0;
}
