import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int L = sc.nextInt();
        int sum = L*(N+1)/2;
        if(N%2==1){
            sum -= N/2+1;
        }
        System.out.println(sum);
    }
}
