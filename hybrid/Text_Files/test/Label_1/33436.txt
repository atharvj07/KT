import java.util.Scanner;

public class MaxOnes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();
            System.out.println(maxOnes(l, r));
        }
    }

    private static long maxOnes(long l, long r) {
        long maxOnes = 0;
        long result = 0;

        for (long i = l; i <= r; i++) {
            long ones = countOnes(i);
            if (ones > maxOnes) {
                maxOnes = ones;
                result = i;
            }
        }

        return result;
    }

    private static long countOnes(long x) {
        long count = 0;
        while (x > 0) {
            count += x & 1;
            x >>= 1;
        }
        return count;
    }
}
