#include <vector>
#include <cmath>
#include <iostream>
#include <algorithm>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
int main()
{
    int n, k;
    cin >> n >> k;
    vector<int> d(n);
    for (int &x : d)
        cin >> x;
    int ans = 0;
    vector<pi> edges;
    for (int i = 0; i < n; i++)
    {
        for (int j = i + 1; j < n; j++)
        {
            if (d[i] != d[j])
            {
                if (d[j] == d[i] + 1 || d[i] == d[j] + 1)
                {
                    ans++;
                    edges.push_back({ i, j });
                }
                else
                {
                    cout << -1 << endl;
                    return 0;
                }
            }
        }
    }
    cout << ans << endl;
    for (auto e : edges)
        cout << e.first + 1 << " " << e.second + 1 << endl;
    return 0;
}
