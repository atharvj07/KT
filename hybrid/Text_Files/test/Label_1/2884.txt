#include <iostream>
using namespace std;
long long m;

int main() {
  cin >> m;
  if (m <= 7)
    cout << m << " " << m << endl;
  else {
    long long l = 1, r = m;
    while (l < r) {
      long long mid = (l + r + 1) / 2;
      if (mid * mid * mid <= m)
        l = mid;
      else
        r = mid - 1;
    }
    long long x = l * l * l;
    long long ans = l + (m - x) / (l + 1);
    cout << ans << " " << x << endl;
  }
  return 0;
}
