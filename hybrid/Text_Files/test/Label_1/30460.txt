import java.util.*;

public class Main {
    static class Clue {
        int clue;
        List<Clue> links;
        Clue(int clue) {
            this.clue = clue;
            links = new ArrayList<>();
        }
    }

    private static long modPow(long base, long n) {
        if (n == 0) return 1;
        long v = modPow(base, n/2);
        v = (v * v) % MOD;
        if (n % 2 == 1) {
            v = (v * base) % MOD;
        }
        return v;
    }

    private static final long MOD = 1000000000;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int m = in.nextInt();
        long k = in.nextLong();

        Map<Integer, Clue> clues = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            clues.put(i, new Clue(i));
        }

        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();

            Clue clueA = clues.get(a);
            Clue clueB = clues.get(b);

            clueA.links.add(clueB);
            clueB.links.add(clueA);
        }

        // Calculate the number of connected components
        int components = 0;
        Queue<Clue> queue = new LinkedList<>();
        Set<Integer> visited = new HashSet<>();

        for (Clue clue : clues.values()) {
            if (visited.contains(clue.clue)) continue;
            components++;

            queue.offer(clue);
            visited.add(clue.clue);

            while (!queue.isEmpty()) {
                Clue current = queue.poll();
                for (Clue link : current.links) {
                    if (!visited.contains(link.clue)) {
                        queue.offer(link);
                        visited.add(link.clue);
                    }
                }
            }
        }

        // Calculate the number of ways to add T links
        long numWays = modPow(components, n - components - m - 1);
        numWays = (numWays * components) % MOD;

        System.out.println(numWays);
    }
}
