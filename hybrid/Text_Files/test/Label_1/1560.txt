#include <iostream>

using namespace std;

typedef long long ll;

const int MAXN = 100000;

ll n, a[MAXN];

int main() {
    cin >> n;
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
    }
    ll ans = 1;
    for (ll i = 0; i < n; i++) {
        ans *= a[i];
        if (ans == 0) {
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
