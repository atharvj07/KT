import sys
input1 = sys.stdin.readline
for _ in range(int(input1())):
    n, m = map(int, input1().split())
    A = []
    for i in range(n):
        A.append(input1().strip())
    B = [bin(x)[2:].rjust(m, '0') for x in range(1 << m)]
    for x in A:
        B.remove(x)
    B.sort()
    print(B[len(B) // 2])
