import java.util.*;
import java.lang.*;

public class Main {
  static int N, X, Y;
  static List<Integer>[] adj = new List[2005];
  static int[] dist;
  static boolean[] vis;

  static void dfs(int v, int d) {
    vis[v] = true;
    dist[v] = d;
    for (int u : adj[v]) {
      if (!vis[u]) dfs(u, d + 1);
    }
  }

  static void solve() {
    for (int k = 1; k <= N - 1; k++) {
      int cnt = 0;
      for (int i = 1; i <= N; i++) {
        for (int j = i + 1; j <= N; j++) {
          if (Math.abs(dist[i] - dist[j]) == k) cnt++;
        }
      }
      System.out.println(cnt);
    }
  }

  public static void main(String[] args) throws Exception {
    Scanner sc = new Scanner(System.in);
    N = sc.nextInt();
    X = sc.nextInt();
    Y = sc.nextInt();
    for (int i = 1; i <= N; i++) adj[i] = new ArrayList<>();
    for (int i = 1; i <= N - 1; i++) adj[i].add(i + 1);
    adj[X].add(Y);
    adj[Y].add(X);
    dist = new int[2005];
    vis = new boolean[2005];
    dfs(1, 0);
    solve();
  }
}
