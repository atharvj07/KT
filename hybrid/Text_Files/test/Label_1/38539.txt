n,p=map(int,input().split())
if(p<=0):
    if p==0:
        if(n%2==0):print(n//2)
        else:print(-1)
    else:
        if((n-1)%abs(p)==0 and (n-1)//abs(p)>=0):print((n-1)//abs(p))
        else:print(-1)
else:
    ans=0
    while(n>0):
        temp=0
        while(2**temp+p<=n):
            temp+=1
        n-=2**temp+p
        ans+=1
    print(ans)
