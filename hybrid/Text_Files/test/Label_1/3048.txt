#include<bits/stdc++.h>
using namespace std;
int main() {
    int x, y, a, b;
    cin >> x >> y >> a >> b;
    int ans = 0;
    int vasya_max = min(x, a + y - b), vasya_min = a;
    int petya_max = y - (vasya_max - a), petya_min = max(0, b - x);
    for(int vs = vasya_min; vs <= vasya_max; vs++) {
        int ps = max(petya_min, b - (vs - a));
        cout << vs << ' ' << ps << endl;
        ans++;
    }
    cout << ans << endl;
    return 0;
}
