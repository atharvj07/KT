import java.util.*;

public class Main {

  static long[] a;

  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int n = s.nextInt();
    a = new long[n];
    long[] d = new long[n];
    Map<Long, Integer> m = new HashMap<>();
    long ans = n;
    for (int i = 0; i < n; i++) {
      a[i] = s.nextInt();
      int cnt = m.getOrDefault(a[i], 0);
      m.put(a[i], cnt + 1);
      d[i] = a[i];
      if (a[i] * 2 <= 1000000000000000000L) {
        int c = m.getOrDefault(a[i] * 2, 0);
        if (c > 0) {
          d[i] = 0;
          ans--;
          m.put(a[i], cnt - 1);
          m.put(a[i] * 2, c - 1);
        }
      }
    }
    Arrays.sort(d);
    for (int i = n - 1; i >= 0; i--) {
      if (d[i] == 0) continue;
      ans--;
      break;
    }
    System.out.println(ans);
  }
}
