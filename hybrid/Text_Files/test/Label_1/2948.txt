#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, k;
  string s;
  cin >> n >> k;
  cin >> s;
  int cnt = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] != s[i - 1]) { cnt++; }
    if (cnt > k) { ans = 1; }
  }
  cout << (ans ? "YES" : "NO") << endl;
  return 0;
}
