import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            boolean ok = true;
            while (x > 0) {
                if (x % 10 > k) {
                    ok = false;
                }
                x /= 10;
            }
            if (ok) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
