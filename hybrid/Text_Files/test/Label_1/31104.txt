import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Read input
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        // Solve
        int ans = Integer.MAX_VALUE;
        int[] dp = new int[1 << n]; // dp[i] = minimum time to get all n colors, where i is the bitmask of the colors already obtained
        Arrays.fill(dp, -1);
        dp[0] = 0; // Base case: no colors obtained
        for (int i = 0; i < (1 << n); i++) { // Iterate over all possible bitmasks
            if (dp[i] == -1) {
                continue; // Skip if the current bitmask is not valid
            }
            for (int j = 0; j < n; j++) { // Iterate over all colors
                if ((i & (1 << j)) == 0) { // If the current color is not obtained
                    int next = i | (1 << j); // Get the bitmask of the colors obtained after catching the current color
                    if (dp[next] == -1 || dp[next] > dp[i] + a[j]) { // If the next bitmask is not valid or the time to get the next bitmask is greater than the current time
                        dp[next] = dp[i] + a[j]; // Update the time to get the next bitmask
                    }
                }
            }
            if (i == (1 << n) - 1) { // If all colors are obtained
                ans = Math.min(ans, dp[i]); // Update the minimum time
            }
        }

        // Output
        System.out.println(ans);
    }
}
