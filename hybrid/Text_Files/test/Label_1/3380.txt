#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int k, n, m;
    cin >> k >> n >> m;
    vector<int> a(n), b(m);
    for (int &x : a) cin >> x;
    for (int &x : b) cin >> x;
    vector<int> last(k + 1);
    for (int i = 1; i <= k; i++) last[i] = k + 1;
    vector<int> ans;
    bool ok = true;
    for (int i = 0, j = 0; i < n || j < m;) {
      if (i < n && a[i] <= k && last[a[i]] <= i) {
        ok = false;
        break;
      }
      if (j < m && b[j] <= k && last[b[j]] <= i + j) {
        ok = false;
        break;
      }
      if (i < n && (j == m || a[i] < b[j] || (a[i] == 0 && b[j] == 0))) {
        if (a[i] > 0) last[a[i]] = i;
        ans.push_back(a[i++]);
      } else {
        if (b[j] > 0) last[b[j]] = i + j;
        ans.push_back(b[j++]);
      }
    }
    if (!ok) {
      cout << -1 << endl;
    } else {
      for (int x : ans) cout << x << ' ';
      cout << endl;
    }
  }
  return 0;
}
