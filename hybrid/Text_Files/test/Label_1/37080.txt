import numpy as np
def check(a,n):
    s=np.sum(a)
    if np.all(np.sum(a,axis=0)==s) and np.all(np.sum(a,axis=1)==s) and np.sum(np.diag(a))==s and np.sum(np.diag(np.rot90(a)))==s:
        return True
    else:
        return False
def solve(a,n):
    for i in range(n):
        for j in range(n):
            if a[i][j]==0:
                for x in range(1,n**2+1):
                    a[i][j]=x
                    if check(a,n):
                        return a
                    else:
                        a[i][j]=0
    return None
n=int(input())
a=np.zeros((n,n),dtype=np.int64)
for i in range(n):
    a[i]=[int(x) for x in input().split()]
r=solve(a,n)
if r is None:
    print(-1)
else:
    print(r[np.where(r==0)])
