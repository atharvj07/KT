#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  vector<int> v;
  for (int i = 0; i < t; i++) {
    int n, x;
    cin >> n;
    v.clear();
    for (int j = 0; j < n; j++) {
      cin >> x;
      v.push_back(x);
    }
    int a = v[0], b = 0;
    int ma = a, mb = 0;
    bool fl = true;
    for (int j = 1; j < n; j++) {
      int tmp;
      if (fl) {
        tmp = ma + 1;
        while (tmp <= mb) tmp += v[j];
        a += tmp - ma;
        ma = tmp;
        fl = false;
      } else {
        tmp = mb + 1;
        while (tmp <= ma) tmp += v[n - j - 1];
        b += tmp - mb;
        mb = tmp;
        fl = true;
      }
    }
    cout << (n + 1) / 2 << " " << a << " " << b << endl;
  }
}
