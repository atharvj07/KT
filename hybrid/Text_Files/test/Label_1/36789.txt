import sys
input = sys.stdin.readline

def f(arr1, arr2, n, m):
    ans = 0
    for x1 in arr1:
        for x2 in arr2:
            if (x1+x2)&1==1: continue
            if (x1+x2)//2>=0 and (x1+x2)//2<=1000000000: ans += 1
    return ans
for _ in range(int(input())):
    n = int(input())
    arr1 = list(map(int, input().split()))
    m = int(input())
    arr2 = list(map(int, input().split()))
    print(f(arr1, arr2, n, m))
