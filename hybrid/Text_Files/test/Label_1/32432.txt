import java.util.*;
import java.io.*;

public class Main {
    static long mod = 1000000007;
    static int n, k;
    static ArrayList<Integer>[] adj, adj2;
    static long[][] dist;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        adj = new ArrayList[n + 1];
        adj2 = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
            adj2[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            String[] line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            int c = Integer.parseInt(line[2]);
            adj[a].add(b);
            if (c == 0)
                adj[b].add(a);
            adj2[b].add(a);
            if (c == 0)
                adj2[a].add(b);
        }
        k = Integer.parseInt(br.readLine());
        String[] line = br.readLine().split(" ");
        int[] stops = new int[k];
        for (int i = 0; i < k; i++) {
            stops[i] = Integer.parseInt(line[i]);
        }
        dist = new long[n + 1][k + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(dist[i], -1);
        }
        System.out.println(dfs(1, 0, stops));
    }

    public static long dfs(int u, int idx, int[] stops) {
        if (idx == k)
            return 0;
        if (dist[u][idx] != -1)
            return dist[u][idx];
        dist[u][idx] = Long.MAX_VALUE;
        int v = stops[idx];
        dist[u][idx] = Math.min(dist[u][idx], dfs(v, idx + 1, stops));
        for (int x : adj2[u]) {
            if (x != v)
                dist[u][idx] = Math.min(dist[u][idx], dfs(x, idx, stops) + 1000);
        }
        long ans = dist[u][idx];
        for (int x : adj[u]) {
            if (x != v)
                ans = Math.min(ans, dfs(x, idx, stops) + 2000);
        }
        dist[u][idx] = ans % mod;
        return dist[u][idx];
    }
}
