n, m, k = map(int, input().split())
a = [int(i) for i in input().split()]
ans = 1e9
for l in range(max(m - k // 10, 1), min(m, n - k // 10) + 1):
    r = l + k // 10
    tmp = 1e9
    for i in range(l, r + 1):
        if not i == m:
            tmp = min(tmp, a[i - 1])
    ans = min(ans, tmp)
print(ans * 10)
