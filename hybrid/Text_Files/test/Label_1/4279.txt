#include <bits/stdc++.h>

using namespace std;

const long long MOD = 1e9 + 7;
const long long NMAX = 1000005;

long long powmod(long long a, long long e = MOD - 2) {
    long long p = 1;
    while (e > 0) {
        if (e & 1) p = (p * a) % MOD;
        a = (a * a) % MOD;
        e >>= 1;
    }
    return p;
}

long long f[NMAX];
long long invf[NMAX];

void init() {
    f[0] = 1;
    invf[0] = 1;
    for (long long i = 1; i < NMAX; i++) {
        f[i] = i * f[i - 1];
        f[i] %= MOD;
        invf[i] = powmod(f[i]);
    }
}

long long choose(long long n, long long k) {
    if (n < k) return 0;
    return f[n] * invf[k] % MOD * invf[n - k] % MOD;
}

long long solve(long long g, long long y) {
    if (y % g != 0) return 0;
    long long n = y / g;
    return (choose(n + g - 3, n - 1) - choose(g - 1, g - 2) + MOD) % MOD;
}

int main() {
    long long x, y;
    cin >> x >> y;
    init();
    cout << solve(x, y) << endl;
    return 0;
}
