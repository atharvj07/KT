#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> p(n + 1), inv_p(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> p[i];
    inv_p[p[i]] = i;
  }
  while (m--) {
    int l, r, x;
    cin >> l >> r >> x;
    int ans = inv_p[x];
    if (l <= ans && ans <= r) {
      cout << "No\n";
    } else {
      cout << "Yes\n";
    }
  }
  return 0;
}
