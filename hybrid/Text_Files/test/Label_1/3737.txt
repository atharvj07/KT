#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vector<int> color(n + 1, 0);
    bool possible = true;
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        if (color[i] == 0) {
            q.push(i);
            color[i] = 1;
            while (!q.empty()) {
                int u = q.front();
                q.pop();
                for (int v : adj[u]) {
                    if (color[v] == 0) {
                        q.push(v);
                        color[v] = 3 - color[u];
                    } else if (color[v] == color[u]) {
                        possible = false;
                        break;
                    }
                }
            }
        }
    }
    if (!possible)
        cout << -1 << endl;
    else {
        for (int i = 1; i <= n; i++) {
            cout << color[i] << " ";
        }
        cout << endl;
    }
    return 0;
}
