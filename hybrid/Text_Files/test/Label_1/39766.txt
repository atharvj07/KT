from functools import reduce
from operator import mul
n = int(input())
a = [int(x) for x in input().split()]
b = sum(1 for x in a if x == -1)
c = 0
for x in a:
    if x > 0:
        c = max(c, x)
ans = 1
if c > 0:
    num = [1] * (c + 1)
    den = [1] * (c + 1)
    for x in a:
        if x > 0:
            num[x] *= b + 1
            den[x] *= c - x
    ans = reduce(mul, num) * pow(reduce(mul, den), -1, 1000000007)
print(ans)
