q = int(input().strip())
for _ in range(q):
    n = int(input().strip())
    if n == 1 or n == 2 or n == 3:
        print(-1)
    elif n % 4 == 0:
        print(n // 4)
    elif n % 4 == 1 or n % 4 == 2 or n % 4 == 3:
        if n < 12:
            print(-1)
        else:
            print(n // 4 - 1)
