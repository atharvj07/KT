n = int(input())
d = []
for i in range(n):
    a, g = map(int, input().split())
    d.append((a - g, i))
d.sort()
a, g = 0, 0
ans = [''] * n
for _, i in d:
    if abs(a - g) <= 500:
        if a <= g:
            ans[i] = 'A'
            a += d[i][0]
        else:
            ans[i] = 'G'
            g += d[i][0]
    else:
        ans = ['-1']
        break
print(''.join(ans))
