import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        int num = 0;
        int carry = 0;
        for (int i = n - 1; i >= 0; i--) {
            carry += arr[i];
            if (carry <= 4) {
                num++;
            } else {
                carry -= 4;
            }
        }
        System.out.println(num);
    }
}
