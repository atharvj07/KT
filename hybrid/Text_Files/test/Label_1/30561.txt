import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            System.out.println(Math.max(solve(a, b, c), solve(c, b, a)));
        }
    }

    public static int solve(int a, int b, int c) {
        int ans = 0;
        while (b >= 2 && c >= 2) {
            ans += 3;
            b -= 2;
            c -= 2;
        }
        while (b >= 1 && a >= 1) {
            ans += 3;
            b -= 1;
            a -= 1;
        }
        return ans;
    }
}
