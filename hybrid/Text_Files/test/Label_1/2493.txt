#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, a, b;
    cin >> n >> a >> b;
    if (a > b)
      swap(a, b);
    if (n % 2 != 0 || a > n / 2 || b < n / 2 + 1) {
      cout << "-1" << endl;
      continue;
    }
    int arr[n + 1];
    for (int i = 1; i <= n; i++) {
      arr[i] = i;
    }
    int pos = 0;
    for (int i = n / 2 + 1; i <= n; i++) {
      if (arr[i] == a) {
        pos = i;
        break;
      }
    }
    if (pos == 0) {
      for (int i = 1; i <= n; i++) {
        if (arr[i] == b) {
          pos = i;
          break;
        }
      }
    }
    arr[pos] = b;
    int temp = arr[n / 2 + 1];
    arr[n / 2 + 1] = arr[pos];
    arr[pos] = temp;
    for (int i = 1; i <= n; i++) {
      cout << arr[i] << " ";
    }
    cout << endl;
  }
  return 0;
}
