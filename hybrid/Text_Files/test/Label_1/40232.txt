from math import gcd
from collections import defaultdict

MOD = 10**9 + 7

def dfs(tree, beauty, u, parent, path_gcd, result):
    new_path_gcd = {beauty[u]: 1}
    for g, count in path_gcd.items():
        # Find gcd for current path and add to new paths
        current_gcd = gcd(g, beauty[u])
        new_path_gcd[current_gcd] = new_path_gcd.get(current_gcd, 0) + count
    # Sum the result with all path gcds ending at this vertex
    for g, count in new_path_gcd.items():
        result[0] = (result[0] + g * count) % MOD
    # Recurse for children
    for v in tree[u]:
        if v != parent:
            dfs(tree, beauty, v, u, new_path_gcd, result)

def solve(n, beauty, edges):
    tree = defaultdict(list)
    for a, b in edges:
        tree[a].append(b)
        tree[b].append(a)
    result = [0]
    dfs(tree, beauty, 1, -1, {}, result)
    return result[0]

# Input reading
n = int(input().strip())
beauty = [0] + list(map(int, input().strip().split()))
edges = [tuple(map(int, input().strip().split())) for _ in range(n - 1)]

print(solve(n, beauty, edges))
