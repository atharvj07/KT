#include<iostream>
#include<vector>

using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  vector<string> s(r);
  for (auto &x : s) cin >> x;
  int ans = 0;
  for (int j = 0; j < c; j++) {
    for (int i = 0; i < r; i++) {
      if (s[i][j] == 'B') {
        if (i == 0 || s[i - 1][j] == '.') {
          ans++;
        }
        if (j < c - 1 && s[i][j + 1] == '.') {
          ans++;
        }
        i = r;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
