import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        switch (sc.nextInt()) {
            case 1:
            case 2:
                System.out.println(1);
                break;
            case 3:
            case 4:
                System.out.println(2);
                break;
            case 5:
                System.out.println(3);
                break;
        }
    }
}
