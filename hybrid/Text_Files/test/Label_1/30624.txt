import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int d = Integer.parseInt(st.nextToken());
        int s = Integer.parseInt(st.nextToken());

        Robot[] robots = new Robot[n];
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int ci = Integer.parseInt(st.nextToken());
            int fi = Integer.parseInt(st.nextToken());
            int li = Integer.parseInt(st.nextToken());
            robots[i] = new Robot(ci, fi, li);
        }

        // Sort the robots by their fuel efficiency (fuel / li).
        Arrays.sort(robots, Comparator.comparingDouble(r -> (double) r.fi / r.li));

        // Calculate the minimum amount of fuel needed to move the robots.
        int minFuel = 0;
        List<Robot> movingRobots = new ArrayList<>();
        for (Robot robot : robots) {
            if (robot.li < d) {
                continue;
            }
            if (minFuel + robot.fi <= s) {
                minFuel += robot.fi;
                movingRobots.add(robot);
            } else {
                break;
            }
        }

        // Calculate the maximum number of robots that can be transported.
        int maxRobots = 0;
        PriorityQueue<Robot> pq = new PriorityQueue<>(Comparator.comparingInt(r -> -r.ci));
        pq.addAll(movingRobots);
        while (!pq.isEmpty()) {
            Robot robot = pq.poll();
            maxRobots++;
            for (int i = 0; i < robot.ci; i++) {
                pq.add(robots[robot.carriedRobots[i]]);
            }
        }

        // Output the maximum number of robots and the minimum amount of fuel needed.
        System.out.println(maxRobots + " " + minFuel);
    }

    static class Robot {
        int ci;
        int fi;
        int li;
        int[] carriedRobots;

        public Robot(int ci, int fi, int li) {
            this.ci = ci;
            this.fi = fi;
            this.li = li;
            this.carriedRobots = new int[ci];
        }
    }
}
