#include <bits/stdc++.h>
using namespace std;
int n, z;
vector<int> x;
int main() {
    cin >> n >> z;
    x.resize(n);
    for (int i = 0; i < n; i++) { cin >> x[i]; }
    sort(x.begin(), x.end());
    int ans = 0, j = 0;
    for (int i = 0; i < n; i++) {
        while (j < n && x[j] <= x[i] + z) j++;
        if (j == n) break;
        ans++;
        j++;
    }
    cout << ans << endl;
    return 0;
}
