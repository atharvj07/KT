#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long

int main() {
    ll n, k;
    cin >> n >> k;
    vi a(n);
    for (auto &x : a) {
        cin >> x;
    }
    int even_total = 0;
    ll total = 0;
    for (auto x : a) {
        if (x % 2 == 0)
            even_total += x / 2;
        total += x;
    }
    if (total & 1)
        cout << "Nicky\n";
    else if (even_total > 0 || k == 1)
        cout << "Kevin\n";
    else
        cout << "Nicky\n";
}
