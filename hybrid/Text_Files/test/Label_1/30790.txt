import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] a = new int[3][n + 1];
        for (int i = 1; i <= 2; i++) {
            for (int j = 1; j <= n; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        for (int j = n - 1; j >= 1; j--) {
            a[2][j] += a[2][j + 1];
            a[1][j] += Math.max(a[1][j + 1], a[2][j + 1]);
        }
        System.out.println(a[1][1]);
    }
}
