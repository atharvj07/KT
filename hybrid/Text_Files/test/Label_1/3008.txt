#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

int main() {
  string s;
  int n;
  cin >> n;
  unordered_map<string, string> m;
  vector<vector<string>> v;
  while (n--) {
    cin >> s;
    int pos = s.find('=');
    string l = s.substr(0, pos);
    string r = s.substr(pos + 1);
    if (r.size() > 1 && r[1] == '=') {
      m[r.substr(0, 2)] = l;
      continue;
    }
    for (int i = 0; i < r.size(); i += 2) {
      if (r[i] == '$' || r[i] == '^' || r[i] == '#' || r[i] == '&') {
        m[r.substr(i - 2, 2)] = l;
        break;
      }
    }
    v.push_back({l, r});
  }
  cout << v.size() << '\n';
  for (auto x : v) {
    cout << x[0] << '=' << x[1] << '\n';
  }
  return 0;
}
