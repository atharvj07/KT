import sys

n = int(input())
w = input().split('*')
m = int(input())
lst = []
for _ in range(m): lst.append(input())

a = []

for i in range(n):
    if w[i] == '*': a.extend(lst)
    else: a = [x for x in a if x[i] == w[i]]
    if len(a) == 1: break

count = 0

for i in range(n):
    if w[i] == '*' and len(set([x[i] for x in a])) > 1: count += 1

print(count)
