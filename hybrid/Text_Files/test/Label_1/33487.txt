import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int x = sc.nextInt();
        int[][] p = new int[d][n];
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < n; j++) {
                p[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[d][x + 1];
        for (int i = 0; i <= x; i++) {
            dp[0][i] = 0;
        }
        for (int i = 1; i < d; i++) {
            for (int j = 0; j <= x; j++) {
                for (int k = 0; k < n; k++) {
                    if (j - p[i - 1][k] >= 0) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - p[i - 1][k]] + p[i][k]);
                    }
                }
                dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
            }
        }
        System.out.println(dp[d - 1][x]);
    }
}
