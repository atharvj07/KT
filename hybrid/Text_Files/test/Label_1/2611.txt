#include <bits/stdc++.h>
#define rep(i,a,n) for(int i = (a); i < (n); i++)
#define ull unsigned long long
using namespace std;
const int N = 2e5+5;
int n, q, a[N];

bool dfs(int l, int r, int x) {
	a[l] = x;
	rep(i,l,r+1) if(!a[i] && i != l) {
		bool ok = false;
		rep(j,1,x) ok |= dfs(i,r,j);
		if(!ok) return false;
	}
	return true;
}
int main() {
	cin >> n >> q;
	rep(i,0,n) cin >> a[i];
	int x = *max_element(a+1, a+n+1);
	cout << (dfs(0,n-1,x) ? "YES\n":"NO\n");
	if(dfs(0,n-1,x)) {
		rep(i,0,n) cout << a[i] << " "; cout << "\n";
	}
	return 0;
}
