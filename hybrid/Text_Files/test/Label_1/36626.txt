h, w = map(int, input().split())
r = list(map(int, input().split()))
c = list(map(int, input().split()))

m = 10 ** 9 + 7
dp = [[[-1] * (w + 1) for _ in range(h + 1)] for _ in range(h + 1)]
def solve(idx, row, pre_c):
    if idx == h:
        if row == pre_c: return 1
        else: return 0

    ret = dp[idx][row][pre_c]
    if ret != -1: return ret

    ret = 0
    if row == pre_c:
        ret = solve(idx + 1, row, pre_c)
        if row + 1 <= r[idx]:
            ret += solve(idx + 1, row + 1, pre_c)
    else:
        if row + 1 <= r[idx]:
            ret = solve(idx + 1, row + 1, pre_c)
        if row <= c[pre_c]:
            ret += solve(idx + 1, row, pre_c)

    dp[idx][row][pre_c] = (ret % m)
    return ret

print(solve(0, 0, 0))
