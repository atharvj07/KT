n=int(input());a=[int(x) for x in input().split()];b=[int(x) for x in input().split()];A=[[]for i in range(60)];B=[[]for i in range(60)]
for i in range(n):
  for j in range(60):
    if a[i]&(1<<j):A[j].append(b[i])
    else:B[j].append(b[i])
x=0
for i in range(60):
  A[i].sort();B[i].sort(reverse=True)
  A[i]=A[i][:max(len(A[i])-(len(A[i]+B[i]))//2,0)];B[i]=B[i][:max(len(B[i])-(len(A[i]+B[i]))//2,0)]
  x+=sum(A[i])
print(x)
