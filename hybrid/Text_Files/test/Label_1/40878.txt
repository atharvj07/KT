n = int(input())
a = list(map(int, input().split()))
a.sort()
if n <= 2: print(0); exit()
ans = []
for i in range(1, n):
    if abs(a[i] - a[i - 1]) > 1:
        ans.append((a[i-1], a[i]))
if len(ans) == 0: print(0); exit()
if len(ans) == 1: print(1); exit()
if abs(ans[0][1] - ans[1][0]) == 1: print(1); exit()
print(-1)
