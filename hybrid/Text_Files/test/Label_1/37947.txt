s = input()
t = input()

if len(s) != len(t):
    print("No")
    exit()

for i in range(len(s)):
    if s[i] != t[i]:
        if (s[i] in ['a', 'e', 'i', 'o', 'u'] and t[i] in ['a', 'e', 'i', 'o', 'u']) or (s[i] not in ['a', 'e', 'i', 'o', 'u'] and t[i] not in ['a', 'e', 'i', 'o', 'u']):
            continue
        else:
            print("No")
            exit()

print("Yes")
