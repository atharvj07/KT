#include <algorithm>
#include <climits>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<char, int>> cards;
  for (int i = 0; i < n; ++i) {
    char c;
    int num;
    cin >> c >> num;
    cards.push_back({c, num});
  }
  int max_color = 0, max_num = 0;
  vector<int> color_count(5, 0), num_count(5, 0);
  for (auto &p : cards) {
    color_count[p.first - 'R']++;
    num_count[p.second - 1]++;
    max_color = max(max_color, color_count[p.first - 'R']);
    max_num = max(max_num, num_count[p.second - 1]);
  }
  int ans = max(max_color, max_num);
  cout << ans << endl;
  return 0;
}
