import java.util.*;
import java.lang.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        long n = s.nextLong();
        long m = s.nextLong();
        long sx = s.nextLong();
        long sy = s.nextLong();
        long fx = s.nextLong();
        long fy = s.nextLong();
        Map<Long, List<Long>> map = new HashMap<>();
        for (long i = 0; i < m; i++) {
            long x1 = s.nextLong();
            long y1 = s.nextLong();
            List<Long> list = map.getOrDefault(x1, new ArrayList<>());
            list.add(y1);
            map.put(x1, list);
            list = map.getOrDefault(y1, new ArrayList<>());
            list.add(x1);
            map.put(y1, list);
        }
        Queue<long[]> q = new LinkedList<>();
        q.offer(new long[]{sx, sy, 0});
        boolean[][] visited = new boolean[(int) n + 1][(int) n + 1];
        visited[(int) sx][(int) sy] = true;
        while (!q.isEmpty()) {
            long[] cur = q.poll();
            long x = cur[0];
            long y = cur[1];
            long t = cur[2];
            if (x == fx && y == fy) {
                System.out.println(t);
                return;
            }
            if (map.containsKey(x)) {
                for (long j : map.get(x)) {
                    if (!visited[(int) j][(int) y]) {
                        q.offer(new long[]{j, y, t + 1});
                        visited[(int) j][(int) y] = true;
                    }
                }
            }
            if (map.containsKey(y)) {
                for (long j : map.get(y)) {
                    if (!visited[(int) x][(int) j]) {
                        q.offer(new long[]{x, j, t + 1});
                        visited[(int) x][(int) j] = true;
                    }
                }
            }
            if (x - 1 >= 1 && !visited[(int) (x - 1)][(int) y]) {
                q.offer(new long[]{x - 1, y, t + 1});
                visited[(int) (x - 1)][(int) y] = true;
            }
            if (x + 1 <= n && !visited[(int) (x + 1)][(int) y]) {
                q.offer(new long[]{x + 1, y, t + 1});
                visited[(int) (x + 1)][(int) y] = true;
            }
            if (y - 1 >= 1 && !visited[(int) x][(int) (y - 1)]) {
                q.offer(new long[]{x, y - 1, t + 1});
                visited[(int) x][(int) (y - 1)] = true;
            }
            if (y + 1 <= n && !visited[(int) x][(int) (y + 1)]) {
                q.offer(new long[]{x, y + 1, t + 1});
                visited[(int) x][(int) (y + 1)] = true;
            }
        }
        System.out.println(-1);
    }
}
