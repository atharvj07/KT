import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();

        List<Integer> inc = new ArrayList<>();
        List<Integer> dec = new ArrayList<>();

        boolean ok = true;
        for (int i = 1; i < n; i++) {
            if (a[i] < a[i - 1]) {
                if (!dec.isEmpty()) {
                    ok = false;
                    break;
                }
                dec.add(a[i - 1]);
            } else if (a[i] > a[i - 1]) {
                if (!inc.isEmpty()) {
                    ok = false;
                    break;
                }
                inc.add(a[i - 1]);
            } else {
                inc.add(a[i - 1]);
                dec.add(a[i - 1]);
            }
        }

        if (inc.isEmpty() || dec.isEmpty() || !ok) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        System.out.println(inc.size());
        System.out.println(inc.stream().map(String::valueOf).collect(Collectors.joining(" ")));
        System.out.println(dec.size());
        System.out.println(dec.stream().map(String::valueOf).collect(Collectors.joining(" ")));
    }
}
