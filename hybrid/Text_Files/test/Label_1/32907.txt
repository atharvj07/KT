import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] g = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            g[i] = input.nextInt();
        }
        if (n % 2 == 0 && a[0] == g[0]) {
            System.out.println("-1");
        } else {
            String s = "";
            for (int i = 0; i < n; i++) {
                if (a[i] > g[i]) {
                    s += "A";
                } else if (a[i] < g[i]) {
                    s += "G";
                } else {
                    s += (i % 2 == 0) ? "A" : "G";
                }
            }
            System.out.println(s);
        }
    }
}
