import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long p = sc.nextLong();
        long w = sc.nextLong();
        long d = sc.nextLong();
        long x = (p-n*d)/(w-d);
        long y = (p-x*w)/d;
        long z = n-x-y;
        if(x>=0 && y>=0 && z>=0) System.out.println(x+" "+y+" "+z);
        else System.out.println(-1);
    }
}
