import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[] robots = new int[n];
        for (int i = 0; i < n; i++) {
            robots[i] = input.nextInt();
        }
        int[] exits = new int[m];
        for (int j = 0; j < m; j++) {
            exits[j] = input.nextInt();
        }
        System.out.println(solve(robots, exits));
        input.close();
    }

    static int solve(int[] robots, int[] exits) {
        int MOD = 1000000007;
        int n = robots.length;
        int m = exits.length;
        int[][] dp = new int[n][m + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = -1;
            }
        }
        return f(robots, exits, 0, 0, dp) % MOD;
    }

    static int f(int[] robots, int[] exits, int i, int j, int[][] dp) {
        if (i == robots.length) {
            return j == exits.length ? 1 : 0;
        }
        if (dp[i][j] != -1) {
            return dp[i][j];
        }
        long ans = 0;
        if (robots[i] == exits[j]) {
            ans = f(robots, exits, i + 1, j + 1, dp);
        } else {
            ans = (long) f(robots, exits, i + 1, j, dp) + (long) f(robots, exits, i + 1, j + 1, dp);
        }
        dp[i][j] = (int) (ans % 1000000007);
        return dp[i][j];
    }
}
