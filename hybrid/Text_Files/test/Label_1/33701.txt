import java.util.*;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Integer> cards = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            cards.add(sc.nextInt());
        }

        Set<List<Integer>> stairs = new HashSet<>();
        for (Integer card : cards) {
            boolean added = false;
            for (List<Integer> stair : stairs) {
                if (stair.size() == 0 || stair.get(stair.size() - 1) < card) {
                    List<Integer> newStair = new ArrayList<>(stair);
                    newStair.add(card);
                    stairs.add(newStair);
                    added = true;
                    break;
                }
            }
            if (!added) {
                stairs.add(Arrays.asList(card));
            }
        }

        int maxSize = 0;
        List<Integer> maxStair = null;
        for (List<Integer> stair : stairs) {
            if (stair.size() > maxSize) {
                maxSize = stair.size();
                maxStair = stair;
            }
        }

        System.out.println(maxSize);
        System.out.println(maxStair.stream().map(String::valueOf).collect(Collectors.joining(" ")));
    }
}
