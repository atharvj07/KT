#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int one_trip_cost = 20;
const int ninety_min_cost = 50;
const int one_day_cost = 120;

int main() {
  int n;
  cin >> n;

  vector<int> times(n);
  for (int i = 0; i < n; ++i) {
    cin >> times[i];
  }

  vector<int> charges(n);
  charges[0] = one_trip_cost;

  int ninety_min_start = times[0];
  int one_day_start = times[0];

  for (int i = 1; i < n; ++i) {
    int current_time = times[i];

    // Check if a 90-minute ticket is cheaper
    if (current_time < ninety_min_start + 90) {
      charges[i] = charges[i - 1];
    } else {
      int cost_ninety = charges[i - 1] + ninety_min_cost;
      if (cost_ninety < one_day_cost) {
        charges[i] = cost_ninety;
        ninety_min_start = current_time;
      } else {
        int cost_one_day = charges[i - 1] + one_day_cost;
        if (cost_one_day < one_trip_cost) {
          charges[i] = cost_one_day;
          one_day_start = current_time;
        } else {
          charges[i] = charges[i - 1] + one_trip_cost;
        }
      }
    }
  }

  for (int i = 0; i < n; ++i) {
    cout << charges[i] << endl;
  }

  return 0;
}
