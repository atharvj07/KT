#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 0, lp = 0, rp = n - 1;
    while (lp <= rp) {
        if (a[lp] <= k) { ans++; lp++; }
        else if (a[rp] <= k) { ans++; rp--; }
        else { break; }
    }
    cout << ans << "\n";
}
