
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] colors = new int[n];
        for (int i = 0; i < n; i++) {
            colors[i] = scanner.nextInt();
        }

        int[] dominantColors = new int[n];
        for (int i = 0; i < n; i++) {
            int start = i;
            int end = i;
            int maxCount = 1;
            int maxColor = colors[i];
            int currCount = 1;
            for (int j = i + 1; j < n; j++) {
                if (colors[j] == colors[i]) {
                    currCount++;
                } else if (currCount > maxCount) {
                    maxCount = currCount;
                    maxColor = colors[j - 1];
                    end = j - 1;
                } else if (currCount == maxCount) {
                    maxColor = Math.min(maxColor, colors[j - 1]);
                    end = j - 1;
                } else {
                    break;
                }
            }

            for (int k = start; k <= end; k++) {
                dominantColors[colors[k] - 1]++;
            }

            i = end;
        }

        for (int i = 0; i < n; i++) {
            System.out.print(dominantColors[i] + " ");
        }
    }
}
