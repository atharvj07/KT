s=input()
q=[0]
for i in s:
    q.append(q[-1]+(i=='0'))
n=len(s)
ans=[]
cur=[0]
for i in range(1,n):
    if q[i]-q[cur[-1]]==len(cur):
        cur.pop()
    cur.append(i)
if len(cur)==1:
    print(-1)
else:
    print(len(cur))
    for l in cur:
        print(l-cur[len(cur)-cur.index(l)-1],*list(range(cur[len(cur)-cur.index(l)-1]+1,l+1)))
