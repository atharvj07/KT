#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
const int mxN = 2e5 + 5;
const ll inf = 1e18;

ll n, m;
ll a[mxN];
ll dp[mxN];

int main() {
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
    }

    vector<pair<ll, ll>> v;
    for (ll i = 0; i < m; i++) {
        ll l, r;
        cin >> l >> r;
        v.push_back({l, r});
    }

    sort(v.begin(), v.end());

    ll ans = 0;

    for (ll i = 0; i <= n; i++) {
        dp[i] = a[i];
    }

    for (pair<ll, ll> p : v) {
        ll l = p.first, r = p.second;
        dp[r] = max(dp[r], dp[l - 1] + a[r]);
        ans = max(ans, dp[r]);
    }

    cout << ans << endl;
}
