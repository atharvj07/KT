import sys, heapq
input = sys.stdin.readline

n = int(input())
a = tuple(map(int, input().split()))
m = int(input())
dragons = [(x,y) for x,y in map(lambda x: tuple(map(int, input().split())), range(m))]

b = []
heapq.heapify(b)
for i in a:
    heapq.heappush(b, -i)

for x,y in dragons:
    total = 0
    while len(b) > 0 and (-b[0] < x):
        total += 1
        x += 1
        heapq.heappop(b)
    while len(b) > 0 and (-b[0] < y):
        total += 1
        y += 1
        heapq.heappop(b)
    if len(b) > 0: heapq.heappop(b)
    heapq.heappush(b, -(x+y))
    print(total)
