#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define N 100005
int n, k, main_courses[N];
vector<int> courses[N], ord;
bool vis[N], undone[N];

void dfs(int node) {
  vis[node] = true;
  for (int next : courses[node]) {
    if (!vis[next])
      dfs(next);
  }
  ord.push_back(node);
}

int main() {
  cin >> n >> k;
  for (int i = 1; i <= k; i++) {
    int x;
    cin >> x;
    main_courses[x] = true;
  }
  for (int i = 1; i <= n; i++) {
    int t;
    cin >> t;
    for (int j = 0; j < t; j++) {
      int a;
      cin >> a;
      courses[i].push_back(a);
    }
  }
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      dfs(i);
    }
  }
  unordered_set<int> st;
  reverse(ord.begin(), ord.end());
  int cnt = 0;
  for (int x : ord) {
    if (!courses[x].empty()) {
      bool ok = true;
      for (int c : courses[x]) {
        if (st.find(c) == st.end()) ok = false;
      }
      if (ok) {
        st.insert(x);
        cnt++;
        if (main_courses[x]) undone[x] = true;
      }
    } else if (main_courses[x])
      undone[x] = true;
  }
  int tot = 0;
  for (int i = 1; i <= n; i++) {
    if (undone[i]) tot++;
  }
  if (tot != k)
    cout << -1 << endl;
  else {
    cout << cnt << endl;
    for (int x : ord) {
      if (undone[x]) cout << x << " ";
    }
    cout << '\n';
  }
  return 0;
}
