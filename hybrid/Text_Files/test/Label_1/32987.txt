import java.util.*;

class Main {
  static class SegmentTree {
    int start, end;
    int value;

    SegmentTree left;
    SegmentTree right;

    public SegmentTree(int start, int end) {
      this.start = start;
      this.end = end;
      this.value = 0;
    }

    public void update(int index, int value) {
      if (index < start || index > end) return;
      if (start == end) {
        this.value = value;
        return;
      }
      int mid = (start + end) / 2;
      if (index <= mid) {
        if (left == null) left = new SegmentTree(start, mid);
        left.update(index, value);
      } else {
        if (right == null) right = new SegmentTree(mid + 1, end);
        right.update(index, value);
      }
      this.value = left == null ? right.value : right == null ? left.value : left.value + right.value;
    }

    public int query(int start, int end) {
      if (this.start > end || this.end < start) return 0;
      if (this.start >= start && this.end <= end) return this.value;
      int mid = (this.start + this.end) / 2;
      int leftValue = left == null ? 0 : left.query(start, end);
      int rightValue = right == null ? 0 : right.query(start, end);
      return leftValue + rightValue;
    }
  }

  static SegmentTree segmentTree;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int q = in.nextInt();
    segmentTree = new SegmentTree(1, n);

    for (int i = 0; i < q; i++) {
      int type = in.nextInt();
      if (type == 0) {
        int start = in.nextInt();
        int end = in.nextInt();
        System.out.println(segmentTree.query(start, end));
      } else {
        segmentTree.update(type, type % 2 == 0 ? 0 : 1);
      }
    }
  }
}
