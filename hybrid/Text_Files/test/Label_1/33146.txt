import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			String s = sc.next();
			int sum = 0;
			for (char c : s.toCharArray())
				sum += c - '0';
			if (sum % 2 == 1)
				System.out.println("-1");
			else {
				StringBuilder sb = new StringBuilder();
				for (int i = n - 1; i >= 0; i--) {
					char c = s.charAt(i);
					sb.insert(0, c);
					int digit = c - '0';
					sum -= digit;
					if (sum % 2 == 1 || digit % 2 == 1)
						break;
				}
				if (sum % 2 == 1)
					System.out.println("-1");
				else
					System.out.println(sb.toString());
			}
		}
	}
}
