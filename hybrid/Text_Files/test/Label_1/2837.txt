#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int cnt[26] = {0};
  for (int i = 0; i < n; ++i)
    cnt[s[i] - 'a']++;
  for (int i = 0; i < 26; ++i)
    if (cnt[i] > k) {
      cout << "NO" << endl;
      return 0;
    }
  cout << "YES" << endl;
}
