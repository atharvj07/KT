import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int[][] p = new int[n][8];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 8; j++) {
                p[i][j] = input.nextInt();
            }
        }

        for (int i = 0; i < n; i++) {
            boolean[] v = new boolean[4];
            int ans = 0;
            for (int j = 0; j < 4; j++) {
                int x = p[i][j * 2] - p[i][4];
                int y = p[i][j * 2 + 1] - p[i][5];
                if (Math.abs(x) + Math.abs(y) <= 1) {
                    v[j] = true;
                }
            }
            int cnt0 = 0, cnt1 = 0;
            for (int j = 0; j < 4; j++) {
                if (v[j]) {
                    cnt0++;
                } else {
                    cnt1++;
                }
            }
            if (cnt0 == 4 || cnt0 == 2 && cnt1 == 2) {
                ans = 0;
            } else if (cnt0 == 3 || cnt0 == 1) {
                ans = 1;
            } else if (v[0] != v[1] && v[1] != v[2] && v[2] != v[3]) {
                ans = 2;
            } else if (v[0] == v[1] && v[2] == v[3] && v[0] != v[2]) {
                ans = 2;
            } else {
                ans = 3;
            }
            System.out.println(ans == 0 ? 0 : ans + 1);
        }
    }
}
