n=int(input())
a=list(map(int,input().split()))
if max(a)!=n or min(a)<=0:print(-1);exit()
b=[0 for i in range(0,n+1)]
for i in range(0,n):b[a[i]]+=1
if sum(b[1:n+1])!=n:print(-1);exit()
print(n)
print(*[i for i in range(1,n+1) for j in range(0,b[i])],sep=' ')
print(*[a.index(i)+1 for i in range(1,n+1)],sep=' ')
