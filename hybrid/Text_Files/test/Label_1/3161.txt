#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, q;
  cin >> n >> m;
  vector<string> s(n), t(m);
  for (int i = 0; i < n; ++i) cin >> s[i];
  for (int i = 0; i < m; ++i) cin >> t[i];
  cin >> q;
  for (int i = 0; i < q; ++i) {
    int y;
    cin >> y;
    --y;
    int a = y % n, b = y % m;
    cout << s[a] << t[b] << endl;
  }
}
