#include<iostream>
#include<map>
using namespace std;
map<char,char>mp;
string s,t;
int main()
{
	int n,i; cin>>n;
	for(i=1;i<=n;i++)
	{
		cin>>s,t;
		if(!t.empty()&&mp.find(t[0])==mp.end())
			mp[t[0]]=s[0];
	}
	cout<<mp.size()<<'\n';
	for(map<char,char>::iterator it=mp.begin();it!=mp.end();it++)
		cout<<it->second<<it->first;
	return 0;
}
