#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> m(n);
  for (int i = 0; i < n; i++) cin >> m[i];
  vector<int> ans;
  stack<pair<int, int>> s;
  for (int i = n - 1; i >= 0; i--) {
    while (!s.empty() && s.top().first < m[i]) { ans.push_back(s.top().second); s.pop(); }
    s.push({m[i], i});
  }
  while (!s.empty()) { ans.push_back(s.top().second); s.pop(); }
  for (int i : ans) cout << m[i] << " ";
  cout << endl;
}
