import sys
s = sys.stdin.readline().strip()
n = int(sys.stdin.readline().strip())
rules = [tuple(map(str, sys.stdin.readline().strip().split())) for i in range(n)]
cnt = 0
for i in range(len(s)):
    for j in range(i + 1, len(s) + 1):
        t = s[i: j]
        compliance = True
        for p, l, r in rules:
            k = t.count(p)
            if l > k or k > r:
                compliance = False
                break
        if compliance:
            cnt += 1
print(cnt)
