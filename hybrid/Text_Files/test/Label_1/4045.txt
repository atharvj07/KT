#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
  vector<int> a;
  int n, x;
  cin >> n;
  while (n--) {
    cin >> x;
    a.push_back(x);
  }
  sort(a.begin(), a.end());
  long long sum = 0;
  for (auto it : a) sum += it;
  if (sum % 2 != 0) a[0]++;
  sum /= 2;
  for (auto &it : a) {
    while (it > sum) it -= 2;
    while (it < sum) it += 2;
  }
  for (auto it : a) cout << it << endl;
  return 0;
}
