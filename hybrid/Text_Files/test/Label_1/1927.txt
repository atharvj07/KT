#include <bits/stdc++.h>
using namespace std;
int n, a[200005];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> a[i];
        long long ans = 0;
        for (int i = 1; i <= n; i++) {
            int l = i + 1, r = n;
            while (l < r) {
                int mid = (l + r) >> 1;
                if (a[mid] - a[i] <= 2) r = mid;
                else l = mid + 1;
            }
            ans += max(0, l - i - 1);
        }
        cout << ans << endl;
    }
    return 0;
}
