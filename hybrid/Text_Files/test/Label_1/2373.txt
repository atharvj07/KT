#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    vector<int> ans;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] + sum > 0) {
        ans.push_back(i + 1);
      }
      sum += a[i];
    }
    cout << ans.size() << endl;
    for (int u : ans) cout << u << " ";
    cout << endl;
  }
}
