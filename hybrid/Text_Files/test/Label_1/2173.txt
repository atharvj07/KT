#include <iostream>
#include <vector>
#include <algorithm>
#include <cstring>

using namespace std;

int n, k;
vector<int> a;
int dp[100005];

int main() {
	cin >> n >> k;
	a.resize(n + 1);
	for (int i = 1; i <= n; i++) cin >> a[i];
	dp[0] = 0;
	for (int i = 1; i <= n; i++) {
		int sum = 0;
		for (int j = i; j >= 1; j--) {
			sum += a[j];
			if (sum % 2 == 0 && sum / 2 >= k) dp[i] = 1;
		}
		if (!dp[i]) {
			dp[i] = -1;
			continue;
		}
		if (n - i & 1) dp[i] = -dp[i];
	}
	if (dp[n] == 1) cout << "Kevin" << endl;
	else cout << "Nicky" << endl;
	return 0;
}
