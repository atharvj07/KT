#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mxN = 1e5 + 5;
ll a[mxN];
int main() {
    ll n, k;
    cin >> n >> k;
    for (ll i = 0; i < n; i++) cin >> a[i];
    for (ll i = 0; i < n; i++) {
        if ((a[i] & 1 || a[i] == 2) && n > 1) {
            cout << "Kevin" << endl;
            return 0;
        }
        if ((a[i] & 1) && a[i] != 1) {
            cout << "Nicky" << endl;
            return 0;
        }
    }
    if (n > 1 || (n == 1 && a[0] == 1)) {
        cout << "Kevin" << endl;
    } else {
        cout << "Nicky" << endl;
    }
    return 0;
}
