import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        List<Map<Integer, Integer>> d = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            String[] di = br.readLine().split(" ");
            Map<Integer, Integer> dist = new HashMap<>();
            for (int j = 0; j < n; j++) {
                dist.put(j + 1, Integer.parseInt(di[j]));
            }
            d.add(dist);
        }
        boolean[][] adj = new boolean[n + 1][n + 1];
        for (Map<Integer, Integer> dist1 : d) {
            for (Map<Integer, Integer> dist2 : d) {
                if (dist1 == dist2) {
                    continue;
                }
                int u = 0;
                int v = 0;
                int minDist = Integer.MAX_VALUE;
                for (int i = 1; i <= n; i++) {
                    int dist = Math.abs(dist1.get(i) - dist2.get(i));
                    if (dist < minDist) {
                        u = i;
                        v = i;
                        minDist = dist;
                    }
                    for (int j = i + 1; j <= n; j++) {
                        dist = Math.abs(dist1.get(i) - dist2.get(j));
                        if (dist < minDist) {
                            u = i;
                            v = j;
                            minDist = dist;
                        }
                        dist = Math.abs(dist1.get(j) - dist2.get(i));
                        if (dist < minDist) {
                            u = j;
                            v = i;
                            minDist = dist;
                        }
                    }
                }
                adj[u][v] = true;
                adj[v][u] = true;
            }
        }
        for (int i = 1; i <= n; i++) {
            int count = 0;
            for (int j = 1; j <= n; j++) {
                if (adj[i][j]) {
                    count++;
                }
            }
            if (count > 1) {
                System.out.println(-1);
                return;
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (adj[i][j]) {
                    System.out.println(i + " " + j);
                }
            }
        }
    }
}
