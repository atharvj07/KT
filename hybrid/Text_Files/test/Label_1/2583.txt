#include <iostream>
using namespace std;
int main() {
  int x, y, ans = 0;
  cin >> x >> y;
  if (x == y) {
    cout << "YES" << endl;
    return 0;
  }
  for (int i = 0; i < 62; i++) {
    if ((y >> i) & 1 != (x >> i) & 1) {
      if ((x >> i) & 1) {
        x = (x >> i) | ((1 << i) - 1);
        ans++;
      }
      if (((x >> i) & 1) ^ ((y >> i) & 1)) {
        x = x + (1 << i);
        ans++;
      }
    }
  }
  if (x == y)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
