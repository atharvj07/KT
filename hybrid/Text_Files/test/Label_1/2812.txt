#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
int main() {
  string x;
  cin >> x;
  int n = x.size();
  long long ans = 0;
  for (int i = 0; i < n; ++i) {
    if (i < n - 1 && x[i] == 'a' && x[i + 1] == 'b') {
      ans++;
      ++i;
    }
  }
  cout << (ans * 2ll % mod) << endl;
  return 0;
}
