#include <cstdio>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int n, z;
  scanf("%d%d", &n, &z);
  vector<int> v(n);
  for (int i=0; i<n; i++) { scanf("%d", &v[i]); }
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i=0, j=i+1; i<n; i++) {
    if (v[i] >= v[j]) { j++; }
    while (j<n && v[j] - v[i] < z) { j++; }
    if (j<n && v[j] - v[i] >= z) { ans++; j++; }
  }
  printf("%d\n", ans);
  return 0;
}
