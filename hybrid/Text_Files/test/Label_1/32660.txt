import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] m = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
            sum += m[i];
        }
        int avg = sum / n;
        int res = 0;
        for (int x : m) {
            res += Math.abs(x - avg);
        }
        res /= 2;
        System.out.println(res);
    }
}
