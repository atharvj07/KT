import java.util.*;
import java.io.*;

public class Main {

    static int n, m, k;
    static int[] p;
    static int[][] dp;
    static int[][] took;

    static int solve(int idx, int pairsLeft) {
        if (idx == n || pairsLeft == 0)
            return 0;

        if (dp[idx][pairsLeft] != -1)
            return dp[idx][pairsLeft];

        int ans = solve(idx + 1, pairsLeft);

        if (idx + m - 1 < n) {
            int sum = 0;

            for (int i = idx; i < idx + m; i++) {
                sum += p[i];
            }

            int take = solve(idx + m, pairsLeft - 1) + sum;

            if (take > ans) {
                ans = take;
                took[idx][pairsLeft] = 1;
            }
        }
        
        return dp[idx][pairsLeft] = ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();

        p = new int[n];

        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }

        dp = new int[n][k + 1];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = -1;
            }
        }

        took = new int[n][k + 1];

        int ans = solve(0, k);

        System.out.println(ans);

        for (int i = 0, pairsLeft = k; i < n; i++) {
            if (took[i][pairsLeft] == 1) {
                System.out.print((i + 1) + " ");
                i += m - 1;
                pairsLeft--;
            }
        }
    }
}
