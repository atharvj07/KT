n,M=map(int,input().split())
a=list(map(int,input().split()))
maxi=0
for i in range(n):
    for j in range(n+1):
        b=[]
        b.extend(a)
        if j!=n:
            b.insert(j,M-b[-1])
        else:
            b.append(M-b[-1])
        sum=b[1]-b[0]
        for j in range(1,n+1):
            sum+=b[j]-b[j-1]
        maxi=max(maxi,sum)
print(maxi)
