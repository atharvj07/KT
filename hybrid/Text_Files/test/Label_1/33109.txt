import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int max = (n-1) / k;
        int min = (n <= k) ? 1 : (n % k == 0 ? n / k : n / k + 1);
        System.out.println(max - min);
    }
}
