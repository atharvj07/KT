import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int iceCream = x;
        int distress = 0;
        for (int i = 0; i < n; i++) {
            char c = sc.next().charAt(0);
            int d = sc.nextInt();
            if (c == '+') {
                iceCream += d;
            } else {
                if (iceCream >= d) {
                    iceCream -= d;
                } else {
                    distress++;
                }
            }
        }
        System.out.println(iceCream + " " + distress);
    }
}
