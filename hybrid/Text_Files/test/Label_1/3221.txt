#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, d, m, ans = 0, a[110];
    cin >> n >> d >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    for (int i = 1; i <= m && i <= n; i++) ans += a[i];
    for (int i = n + 1; i <= m; i++) ans -= d;
    cout << ans << endl;
    return 0;
}
