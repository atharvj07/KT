for t in range(int(input())):
    n, *p = map(int, input().split())
    p.sort()
    m, *q = map(int, input().split())
    q.sort()
    ans = 0
    for x in p:
        l, r = 0, m-1
        while l <= r:
            mid = (l + r) // 2
            if q[mid] + x < 0:
                l = mid + 1
            else:
                r = mid - 1
        ans += r 
    print(ans)
