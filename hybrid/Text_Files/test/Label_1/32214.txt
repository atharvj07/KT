import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        Map<String, Integer> ai = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] line = in.readLine().split(" ");
            ai.put(line[0], Integer.parseInt(line[1]));
        }
        int[] order = new int[n];
        int[] hi = new int[n];
        for (int i = 0; i < n; i++) {
            int j = 0;
            while (j < n && ai.getOrDefault(order[j] + "", -1) != i) j++;
            if (j == n) {
                System.out.println(-1);
                return;
            }
            order[j] = i;
        }
        int height = 1;
        for (int i = 0; i < n; i++) {
            int j = 0;
            while (j < n && ai.getOrDefault(order[j] + "", -1) != i) {
                if (hi[j] == 0) hi[j] = height++;
                j++;
            }
            if (hi[j] == 0) hi[j] = height++;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(order[i] + " " + hi[i]);
        }
    }
}
