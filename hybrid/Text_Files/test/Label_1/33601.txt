import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        String t = sc.next();
        int[] ans = solve(s, t);
        if (ans == null) {
            System.out.println(-1);
        } else {
            System.out.println(ans.length);
            for (int x : ans) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }

    public static int[] solve(String s, String t) {
        int n = s.length();
        int[] p = new int[n];
        int[] q = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = s.charAt(i) - 'a';
            q[i] = t.charAt(i) - 'a';
        }
        int[] ans = new int[n];
        int j = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] == q[i]) {
                continue;
            }
            int k = -1;
            for (int l = i + 1; l < n; l++) {
                if (p[l] == q[i]) {
                    k = l;
                    break;
                }
            }
            if (k == -1) {
                return null;
            }
            ans[j++] = k - i;
            for (int l = k; l > i; l--) {
                p[l] = p[l - 1];
            }
            p[i] = q[i];
        }
        return Arrays.copyOf(ans, j);
    }
}
