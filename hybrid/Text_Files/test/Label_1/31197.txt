import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[] b = new int[n];
        Arrays.sort(x);
        for (int i = 0; i < n; i++) {
            int l = 0, r = n - 1;
            while (l <= r) {
                int mid = (l + r) / 2;
                if (x[mid] <= x[i]) {
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            b[i] = x[(l + r) / 2];
        }
        for (int bi : b) {
            System.out.println(bi);
        }
    }
}
