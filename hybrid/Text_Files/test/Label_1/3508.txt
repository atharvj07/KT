#include<bits/stdc++.h>
using namespace std;
int n,m;
char g[1010][1010];
bool f[1010][1010];
int main(){
	cin>>n>>m;
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
			cin>>g[i][j];
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
			if(g[i][j]=='.'&&g[i-1][j]=='#'&&g[i][j-1]=='#'&&g[i+1][j]=='#'&&g[i][j+1]=='#'&&!f[i-1][j]&&!f[i][j-1]&&!f[i+1][j]&&!f[i][j+1]){
				f[i-1][j]=f[i][j-1]=f[i+1][j]=f[i][j+1]=true;
				g[i][j]='#';
				j++;
			}
	bool ok=0;
	for(int i=1;i<=n;i++)
		for(int j=1;j<=m;j++)
			ok|=g[i][j]=='.';
	puts(ok?"YES":"NO");
}
