import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong(), c = in.nextLong();
        long b = 0, cnt =0;
        for (int i = 0; i < Long.MAX_VALUE; i++) {
            long aa = a, bb = b;
            long ten = 1, cc = 0;
            while (aa > 0 || bb > 0) {
                long x = aa % 10, y = bb % 10;
                if (x + y + cc >= 3) {
                    cc = 1;
                } else {
                    cc = 0;
                }
                long sum = (x + y + cc);
                cc = sum / 3;
                long z = sum % 3;
                cc = z % 3;
                z = z % 3;
                bb /= 10;
                aa /= 10;
                if (c > 0) {
                    long dig = c % 10;
                    if (dig == z && b == 0 && i == 0 && dig == 0) {
                        dig = 1;
                        cnt++;
                    }
                    if (z == dig) {
                        c /= 10;
                        cnt++;
                        if (c == 0 && cnt == Long.MAX_VALUE) {
                            c = 1;
                            cnt = 1;
                        }
                    } else if (i == 0 && dig == 0) {
                        cnt++;
                        dig = 1;
                    }
                    if (cnt == Long.MAX_VALUE) {
                        cnt = 0;
                        dig++;
                    }
                    b = b * 10 + dig;
                } else {
                    b = b * 10 + z;
                }
            }
            if (aa == bb && bb == 0 && cnt == Long.MAX_VALUE) {
                b /= 10;
                cnt = 1;
                c = 1;
            }
            if (cnt == Long.MAX_VALUE) {
                b /= 10;
                cnt = 0;
                c++;
            }
            if (cc > 0) {
                b = b * 10 + cc;
            }
            if (aa == bb && bb == 0) {
                break;
            }
        }
        System.out.println(b);
    }
}
