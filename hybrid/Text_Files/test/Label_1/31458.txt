import java.util.*;
import java.io.*;

class Main {
    static void print(String s) {
        System.out.print(s);
    }
    static void println(String s) {
        System.out.println(s);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();

        println(n > 1 ? "Petya" : "Vasya");

        in.close();
    }
}
