#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, u;
  cin >> n >> u;
  vector<long long> e(n);
  for (int i = 0; i < n; i++) { cin >> e[i]; }
  double ans = -1;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      int k = upper_bound(e.begin(), e.end(), e[i] + u) - e.begin();
      if (k <= j) continue;
      ans = max(ans, (double)(e[k] - e[j]) / (e[k] - e[i]));
    }
  }
  cout << ans << endl;
  return 0;
}
