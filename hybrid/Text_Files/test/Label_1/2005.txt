#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int al = 0, bl = 0, ans = 0;
    for (int l = 0, r = n - 1; l <= r;) {
      if (al <= bl) {
        al += a[l++];
        ans++;
      } else {
        bl += a[r--];
        ans++;
      }
    }
    cout << ans << " " << al << " " << bl << '\n';
  }
  return 0;
}
