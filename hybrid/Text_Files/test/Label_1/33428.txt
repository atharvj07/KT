import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long k = sc.nextLong();
        long[] p = new long[(int) m];
        for (int i = 0; i < m; i++) {
            p[i] = sc.nextLong();
        }

        long ops = 0;
        long cur = 0;
        long page = 1;
        boolean hasSpecial = false;
        for (int i = 0; i < m; i++) {
            if (p[i] > cur + k) {
                ops++;
                page = (p[i] - 1) / k + 1;
                cur = (page - 1) * k;
            }
            if (p[i] >= cur + 1 && p[i] <= cur + k) {
                hasSpecial = true;
            }
        }
        if (hasSpecial) {
            ops++;
        }
        System.out.println(ops);
    }
}
