#include <iostream>
#include<vector>
#include<climits>
using namespace std;
int n,s,t,ans;
vector<int> d;
int main(){
    cin>>n;
    d.resize(n+1);
    d[0]=d[n]=INT_MAX;
    for(int i=1;i<n;i++) cin>>d[i];
    cin>>s>>t;
    if(s==t) cout<<0;
    else{
        int a=s+1, b=(s==1)?n:s-1;
        if(s<t){
            ans=d[a];
            while(a!=t) {a++; if(a==n+1) a=1; ans+=d[a];}
        }
        else{
            ans=d[b];
            while(b!=t) {b--; if(b==0) b=n; ans+=d[b];}
        }
        cout<<ans;
    }
    return 0;
}
