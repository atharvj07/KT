#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
int n, t[100005], ans[100005];
int main() {
    cin.tie(0), cout.tie(0), ios::sync_with_stdio(0);
    cin >> n;
    for (int i = 0; i < n; i++) cin >> t[i];
    priority_queue<pair<int, int>> pq;
    pq.push({-INF, 0});
    int st = 0, last = 0;
    for (int i = 0; i < n; i++) {
        while (last < t[i]) {
            pq.push({-last, st});
            st += 20;
            last += 90;
        }
        pair<int, int> p = pq.top();
        pq.pop();
        ans[i] = st - (-p.first);
        pq.push({-t[i], st});
    }
    for (int i = 0; i < n; i++) cout << ans[i] << endl;
}
