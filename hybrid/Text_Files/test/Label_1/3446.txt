#include <iostream>

int n, a[100001], s[100001], dp[100001], mx;

int main() {
    std::cin >> n;
    for (int i = 0; i < n; i++) std::cin >> a[i];
    for (int i = 0; i < n; i++) { char ch; std::cin >> ch; s[i] = ch - '0'; }
    for (int i = 0; i < (1 << n); i++) {
        int sum = 0;
        for (int j = 0; j < n; j++) { if (i % (1 << (n - j - 1)) == 0) sum += a[j] * s[j]; }
        mx = std::max(mx, sum);
    }
    std::cout << mx << std::endl;
}
