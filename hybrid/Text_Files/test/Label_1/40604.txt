import sys
from collections import Counter
input = sys.stdin.readline

n = int(input())
A = [[int(x) for x in input().split()] for _ in range(n)]
m = int(input())
for sqdist in [Counter(map(int, input().split())) for _ in range(m)]:
    k = 0; B = []
    for i, c in sqdist.items():
        uu = [a for a, b in A if b*b == i]
        if len(uu) != c: break
        if not B: B = uu
        else: B = [b if not (a and b and a != b[0]) else [min(x) for x in zip(a, b)] for a, b in zip(B, uu)]
    if B: k, B = len(B), sorted(B)
    print(k, *B)
