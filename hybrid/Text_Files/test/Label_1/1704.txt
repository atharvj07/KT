#include <iostream>
#include <vector>
using namespace std;

typedef long long ll;

const ll M = 1e9+7;
const vector<ll> P{1,1,2,6,24,120,720,5040,40320,362880};

void solve() {
    ll k;
    cin >> k;
    ll ans = 0;
    for (int i = 8; i >= 0; i--) {
        ll t = (k-1) / P[i];
        ans = (ans + t * (i+1)) % M;
        k -= t * P[i];
    }
    ans = (ans + 1) % 10;
    if (ans == 0) ans = 10;
    cout << ans << '\n';
}

int main() {
    ll q;
    cin >> q;
    while (q--) solve();
    return 0;
}
