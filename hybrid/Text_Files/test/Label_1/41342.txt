q = int(input())
l, r= 0, 0
for _ in range(q):
    op, k = input().split()
    k = int(k)
    if op[0] == 'L':
        l = k
    elif op[0] == 'R':
        r = k
    else:
        if k <= l:
            print(0)
        elif k >= r:
            print(0)
        else:
            print(min(k - l, r - k))
