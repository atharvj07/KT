#include <bits/stdc++.h>

using namespace std;

int n, m, k;
int p[105][105], q[105], ans[105];

int main() {
    cin >> n >> m >> k;
    for (int j = 1; j <= n; j++) {
        for (int i = 1; i <= m; i++) cin >> p[i][j];
        q[j] = 0;
    }
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++)
            if (!q[j] && p[i][j]) {
                if (q[p[i][j]])
                    q[j] = q[p[i][j]];
                else
                    q[j] = i;
            }
    }
    for (int i = 1; i <= n; i++) {
        if (!q[i])
            ans[i] = 0;
        else
            ans[i] = q[i];
        cout << ans[i] << endl;
    }
    return 0;
}
