import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        boolean[] visited = new boolean[n];
        List<Integer>[] graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            graph[i].add(p[i]);
        }

        if (!isWeaklyConnected(graph, visited, 0)) {
            System.out.println("IMPOSSIBLE");
            return;
        }

        int[] a = new int[n];
        Arrays.fill(a, -1);
        a[0] = 0;
        if (!dfs(graph, 0, a)) {
            System.out.println("IMPOSSIBLE");
            return;
        }

        System.out.println("POSSIBLE");
    }

    private static boolean isWeaklyConnected(List<Integer>[] graph, boolean[] visited, int start) {
        visited[start] = true;
        for (int next : graph[start]) {
            if (!visited[next]) {
                if (!isWeaklyConnected(graph, visited, next)) {
                    return false;
                }
            }
        }
        return true;
    }

    private static boolean dfs(List<Integer>[] graph, int u, int[] a) {
        for (int v : graph[u]) {
            if (a[v] == -1) {
                a[v] = a[u] + 1;
                if (!dfs(graph, v, a)) {
                    return false;
                }
            } else if (a[v] != a[u] + 1) {
                return false;
            }
        }
        return true;
    }
}
