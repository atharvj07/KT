#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MAXN = 2205;
ll c[2][MAXN];
ll dp[MAXN][MAXN];
ll f(ll pos,ll rem){
    if(pos == MAXN)  return rem?1e18:0;
    ll &ans=dp[pos][rem];
    if(ans!=-1) return ans;
    ans=f(pos+1,rem);
    if(rem) ans=min(ans,c[0][pos]+f(pos+1,rem-1));
    if(rem+1<=pos-1) ans=min(ans,c[1][pos]+f(pos+1,rem+1));
    return ans;
}
int main(){
    memset(dp,-1,sizeof dp);
    ll n,k;
    cin >> n >> k;
    for(int i=0;i<n;i++)  cin >> c[0][i];
    for(int i=0;i<n;i++)  cin >> c[1][i];
    cout << f(1,k) << endl;
    return 0;
}
