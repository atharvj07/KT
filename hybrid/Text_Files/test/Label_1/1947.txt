#include <iostream>
#include <unordered_set>
#define LL long long

using namespace std;

int main() {
    int n, m, x;
    LL sum = 0;
    bool found = false;
    LL prev = 0;
    unordered_set<int> uniq;

    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        sum += x - prev;
        prev = x;
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> x;
        uniq.emplace(x);
    }
    for (auto u : uniq) {
        sum -= (prev - (u - 1));
        prev = u;
    }
    cout << sum << "\n";
    return 0;
}
