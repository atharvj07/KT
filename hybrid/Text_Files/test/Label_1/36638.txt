s=input()
x,y=map(int,input().split())
n=len(s)
dp=[[1e9 for j in range(2)] for i in range(n+1)]
dp[n][0]=dp[n][1]=0
for i in range(n-1,-1,-1):
    if s[i]=='?':
        dp[i][0]=min(dp[i+1][0],dp[i+1][1]+x)
        dp[i][1]=min(dp[i+1][1],dp[i+1][0]+y)
    else:
        dp[i][int(s[i])]=min(dp[i+1][0],dp[i+1][1])
print(dp[0][0])
