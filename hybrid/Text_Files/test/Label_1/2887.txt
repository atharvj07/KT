#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> cards(n);
    for (int i = 0; i < n; i++) {
        int m;
        cin >> m;
        cards[i].resize(m);
        for (int j = 0; j < m; j++) {
            cin >> cards[i][j];
        }
    }
    vector<int> balls(101, 0);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < cards[i].size(); j++) {
            balls[cards[i][j]]++;
        }
        bool win = true;
        for (int j = 0; j < cards[i].size(); j++) {
            if (balls[cards[i][j]] > 1) {
                win = false;
                break;
            }
        }
        if (win) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
        for (int j = 0; j < cards[i].size(); j++) {
            balls[cards[i][j]]--;
        }
    }
    return 0;
}
