n, m = map(int, input().split())
a = list(map(int, input().split()))
c = list(map(int, input().split()))

orders = [list(map(int, input().split())) for _ in range(m)]

# Pair each dish type with its cost and initial amount, and sort by cost then by index
dishes = sorted([(cost, amount, i) for i, (amount, cost) in enumerate(zip(a, c))])

# To keep track of the cheapest available dish
cheapest_idx = 0

for t, d in orders:
    t -= 1  # Adjust index
    total_cost = 0
    # Serve the dishes of the requested type if available
    if a[t] > 0:
        served = min(a[t], d)
        total_cost += served * c[t]
        a[t] -= served
        d -= served

    # Serve dishes of the cheapest available type if the request is not yet fulfilled
    while d > 0 and cheapest_idx < n:
        # Find the next available cheapest dish
        while cheapest_idx < n and dishes[cheapest_idx][1] == 0:
            cheapest_idx += 1
        if cheapest_idx == n:
            break  # No more dishes available

        # Serve the cheapest dish
        dish_cost, dish_amount, dish_index = dishes[cheapest_idx]
        served = min(dish_amount, d)
        total_cost += served * dish_cost
        d -= served
        dishes[cheapest_idx] = (dish_cost, dish_amount - served, dish_index)
        a[dish_index] -= served

    if d > 0:  # If the request is not fully served
        print(0)
    else:
        print(total_cost)
