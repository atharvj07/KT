import math
n = int(input())
table = list(map(int, input().split()))
a = [0]*n
for i in range(n):
    for j in range(n):
        if table[i*n+j] == table[j*n+i]:
            a[i] = table[i*n+j]
for i in range(n):
    for j in range(i+1, n):
        a[j] = table[i*n+j] // math.gcd(a[i], a[j])
print(' '.join(map(str, a)))
