import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a1 = scanner.nextInt();
        int b1 = scanner.nextInt();
        int a2 = scanner.nextInt();
        int b2 = scanner.nextInt();
        int a3 = scanner.nextInt();
        int b3 = scanner.nextInt();

        boolean[][] connected = new boolean[5][5];
        connected[a1][b1] = connected[b1][a1] = true;
        connected[a2][b2] = connected[b2][a2] = true;
        connected[a3][b3] = connected[b3][a3] = true;

        boolean[] visited = new boolean[5];
        visited[1] = true;
        int count = 1;

        while (count < 4) {
            for (int i = 1; i <= 4; i++) {
                if (!visited[i]) {
                    for (int j = 1; j <= 4; j++) {
                        if (connected[i][j] && visited[j]) {
                            visited[i] = true;
                            count++;
                            break;
                        }
                    }
                }
            }
        }

        System.out.println(count == 4 ? "YES" : "NO");
    }
}
