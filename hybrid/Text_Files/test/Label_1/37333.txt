import sys
input = sys.stdin.readline

n, k = map(int, input().split())
arr = []
for i in range(n):
    l, r, a = map(int, input().split())
    arr.append((l, r, a))

ans = 0
cur = 0
for l, r, a in arr:
    if cur + a <= k:
        cur += a
    else:
        ans += (cur + a) // k
        cur = (cur + a) % k
ans += (cur + k - 1) // k
if ans > 1e9:
    print(-1)
else:
    print(ans)
