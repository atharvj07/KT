import java.util.*;
import java.math.BigInteger;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long x = in.nextLong();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        Arrays.sort(arr);
        BigInteger ans = BigInteger.valueOf(x).multiply(BigInteger.valueOf(arr[n - 1]));
        for (int i = n - 2; i >= 0; i--) {
            x = Math.max(x - 1, 1);
            ans = ans.add(BigInteger.valueOf(x).multiply(BigInteger.valueOf(arr[i])));
        }
        System.out.println(ans);
    }
}
