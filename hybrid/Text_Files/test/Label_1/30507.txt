import java.util.HashMap;
import java.util.Scanner;

public class Main {
    private static HashMap<String, String> types = new HashMap<>();

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine(); // consume the newline character

        for (int i = 0; i < n; i++) {
            String line = in.nextLine();
            String[] parts = line.split(" ");
            if (parts[0].equals("typedef")) {
                typedef(parts[1], parts[2]);
            } else if (parts[0].equals("typeof")) {
                System.out.println(typeof(parts[1]));
            }
        }
    }

    private static void typedef(String a, String b) {
        types.put(b, a);
    }

    private static String typeof(String a) {
        if (types.containsKey(a)) {
            a = types.get(a);
        }

        int p = 0;
        while (a.charAt(p) == '*') {
            p++;
        }

        return "void" + "*".repeat(Math.max(0, p - 1));
    }
}
