#include<bits/stdc++.h>
using namespace std;
#define int long long
#define vi vector<int>
#define pii pair<int,int>
#define vii vector<pii>
#define pb push_back
#define ff first
#define ss second
const int inf=1e18;
const int mod=1e9+7;
const int N=2e5+5;
vi adj[N];
int a[N];
int sub[N];
int ans[N];
void dfs(int node,int par)
{
	sub[node]=a[node];
	for(auto x:adj[node])
	{
		if(x==par) continue;
		dfs(x,node);
		sub[node]+=sub[x];
	}
    ans[node]=sub[node];
	for(auto x:adj[node])
	{
		if(x==par) continue;
		int val=sub[node]-sub[x]+2*sub[x];
		ans[node]=max(ans[node],val);
	}
	ans[node]-=sub[node];
}
int32_t main()
{
	int n;cin>>n;
	for(int i=0;i<n;i++) cin>>a[i];
	for(int i=0;i<n-1;i++)
	{
		int x,y;cin>>x>>y;
		adj[x].pb(y);adj[y].pb(x);
	}
	dfs(1,1);
	for(int i=1;i<=n;i++) cout<<ans[i]<<" ";
	return 0;
}
