import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt();
    long[] a = new long[n];
    TreeSet<Long> t = new TreeSet<>();
    for (int i = 0; i < n; i++) {
      a[i] = cin.nextLong();
      t.add(a[i]);
    }
    long gcd = 0;
    for (long x : t) {
      boolean flag = true;
      for (long y : t) {
        if (x % y != 0) {
          flag = false;
          break;
        }
      }
      if (flag) {
        gcd = x;
      }
    }
    System.out.println(gcd);
  }
}
