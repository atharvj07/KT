#include<bits/stdc++.h>
using namespace std;
int n,a[110];
bool ok;
void dfs(int cur,int cnt,int sum){
	if(ok) return;
	if((cur==n+1&&cnt==4)||(cur>n+1&&cnt==3)){
		if(sum==0) ok=1;
		return;
	}
	for(int i=1;i<=n;i++){
		if(a[i]==0){
			a[i]=cur;
			if(cnt==2) dfs(cur+1,cnt+1,sum-(i));
			else if(cnt==1) dfs(cur+1,cnt+1,sum+i);
			else dfs(cur+1,cnt,sum);
			a[i]=0;
		}
	}
}
int main(){
	string s;
	cin>>s;
	n=count(s.begin(),s.end(),'?');
	dfs(1,0,0);
	if(!ok) cout<<"Impossible\n";
	else{
		cout<<"Possible\n";
		for(int i=1;i<=n;i++){
			cout<<a[i]<<" ";
		}
		cout<<"\n";
	}
	return 0;
}
