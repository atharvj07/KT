n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
mx = 0
ans = 0
for i in range(n):
    l = 1
    r = b[i]
    while l <= r:
        m = (l + r) // 2
        if m <= a[i] and b[i] - m <= a[i]:
            ans = max(ans, m * (b[i] - m))
            mx = max(mx, b[i] - m)
            l = m + 1
        else:
            r = m - 1
    if mx > 0:
        ans -= mx
print(ans)
