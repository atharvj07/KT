#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k, l1, r1, l2, r2, ans = 0;
    cin >> n >> k >> l1 >> r1 >> l2 >> r2;
    while (k > 0 && l1 < r2) {
      if (l2 <= l1) {
        k -= (l1 - l2) * n;
        l2 = l1;
      } else {
        k -= (r2 - r1) * n;
        r1 = r2;
      }
      ans++;
    }
    cout << ans << '\n';
  }
}
