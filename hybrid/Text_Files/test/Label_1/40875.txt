import sys
read = sys.stdin.readline
for t in range(int(read())):
    n = int(read())
    g = [[]] + [[] for i in range(n)]
    for i in range(n-1):
        x, y = map(int,read().split())
        g[x].append(y)
        g[y].append(x)
    cnt = [0 for i in range(n+1)]
    leaf = 0
    def dfs(v, p):
        nonlocal leaf
        if len(g[v]) == 1 and p != -1:
            leaf += 1
            return 1
        ans = 0
        for u in g[v]:
            if u==p: continue
            ans += dfs(u, v)
        if p!=-1 and ans == len(g[v])-1:
            cnt[v] = 1
        return ans
    dfs(1, -1)
    print(leaf - min(cnt))
