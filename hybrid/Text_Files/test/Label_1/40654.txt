n=int(input())
t=[int(a) for a in input().split()]
c=[0 for i in range(n)]
l=[[] for i in range(n)]
for i in range(n):
    for k in range(i+1):
        l[t[i]-1].append([k,i])
for i in range(n):
    for a in l[i]:
        x=a[0]
        y=a[1]
        q=0;b=0
        for j in range(x,y+1):
            if t[j]==i+1:q+=1
            else:b+=1
        if q>b:c[i]+=1
for i in range(n):
    print(c[i])
