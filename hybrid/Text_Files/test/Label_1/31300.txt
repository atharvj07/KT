import java.util.Scanner;

public class Main {
    static int n, idx[];
    static long ans;

    static void dfs(int u, int l, int r, int t, int b) {
        if (u == n * n + 1) {
            ans++;
            return;
        }
        for (int i = l; i <= r; i++) {
            for (int j = t; j <= b; j++) {
                if (i + j - 1 < idx[u]) continue;
                int cnt = 0;
                if (i > l) cnt += r - i + 1;
                if (j > t) cnt += b - j + 1;
                if (i > l && j > t) cnt--;
                dfs(u + 1, i, r, j, b);
                r = i - cnt - 1;
                b = j - cnt - 1;
            }
        }
    }

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        n = cin.nextInt();
        idx = new int[n * n + 1];
        for (int i = 1; i <= n * n; i++) idx[cin.nextInt()] = i;
        dfs(1, 1, n, 1, n);
        System.out.println(ans);
    }
}
