#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];

  set<int> s;
  for (int i = 0; i < n; i++) s.insert(a[i]);

  if (s.size() > 7) {
    cout << -1 << endl;
    return 0;
  }

  vector<int> ans;
  for (int i = 1; i <= 7; i++) {
    if (!s.count(i)) continue;
    for (int j = i; j <= 7; j++) {
      if (!s.count(j)) continue;
      for (int k = j; k <= 7; k++) {
        if (!s.count(k)) continue;
        if (i < j && j < k && i % j == 0 && j % k == 0) {
          ans.push_back(i);
          ans.push_back(j);
          ans.push_back(k);
          s.erase(i);
          s.erase(j);
          s.erase(k);
        }
      }
    }
  }

  if (ans.size() != n) cout << -1 << endl;
  else {
    for (int i = 0; i < n; i += 3) {
      cout << ans[i] << ' ' << ans[i + 1] << ' ' << ans[i + 2] << ' ';
    }
    cout << endl;
  }

  return 0;
}
