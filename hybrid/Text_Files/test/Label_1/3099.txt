#include<bits/stdc++.h>
#define endl "\n"
using namespace std;
string s;
int pre[1000005], suf[1000005];
int solve(int n){
    int l = 0, r = 0, ans = 0;
    for(int i = 0; i < n; i++){
        if(s[i] == '(')  l++;
        else{
            if(l)  l--, ans++, r++;
            else   r++;
        }
    }
    if(l == 0){
        if(r - 1 > 0)    ans += r - 1;
    }
    return ans;
}
int main() {
    int n;
    cin >> n;
    cin >> s;
    cout << solve(n) << endl;
    return 0;
}
