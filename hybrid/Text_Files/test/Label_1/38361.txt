def solve():
    _ = int(input())
    for _ in range(_):
        n = int(input())
        s = input()
        a = [0] * n
        for i in range(n):
            if s[i] == '1':
                a[i] = 2
            elif i > 0 and a[i - 1] == 2:
                a[i] = 2
        ans = 0
        for i in range(n):
            if a[i] != 0:
                ans += 1
        m = [0] * 2
        for i in range(n):
            m[a[i] - 1] += 1
        ans += max(m)
        print(ans)

solve()
