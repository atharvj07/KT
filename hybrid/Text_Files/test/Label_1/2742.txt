#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007

long long p[300];
long long dp[300][300];

int main() {
  int n, k;
  cin >> n >> k;
  p[0] = 1;
  for (int i = 1; i <= n; i++)
    p[i] = (1LL * p[i - 1] * k) % mod;
  for (int i = 1; i <= n; i++)
    dp[i][1] = p[n - i];
  for (int i = 1; i <= n; i++) {
    for (int j = 2; j <= n; j++) {
      dp[i][j] = (1LL * dp[i][j - 1] * (k - j + 1)) % mod;
    }
  }
  cout << dp[n][n] << endl;
  return 0;
}
