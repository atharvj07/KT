n = int(input())
arr = list(map(int, input().split()))

count = {}
first_pos = {}
last_pos = {}

for i in range(n):
    if arr[i] in count:
        count[arr[i]] += 1
        last_pos[arr[i]] = i
    else:
        count[arr[i]] = 1
        first_pos[arr[i]] = i
        last_pos[arr[i]] = i

max_count = max(count.values())

candidates = [num for num, cnt in count.items() if cnt == max_count]

min_length = n
result = (0, 0)

for num in candidates:
    length = last_pos[num] - first_pos[num] + 1
    if length < min_length:
        min_length = length
        result = (first_pos[num] + 1, last_pos[num] + 1)

print(result[0], result[1])
