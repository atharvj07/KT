import sys 
input = sys.stdin.readline
n = int(input())
c1 = list(map(int, input().split()))
c2 = list(map(int, input().split()))
c3 = list(map(int, input().split()))
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    p, q = map(int, input().split())
    g[p].append(q)
    g[q].append(p)
vis = [0] * (n+1)
ans = [0] * (n+1)
def dfs(cur, par):
    vis[cur] = 1
    a = [c1[cur-1], c2[cur-1], c3[cur-1]]
    if par != -1:
        a.remove(ans[par])
    ans[cur] = min(a)
    for u in g[cur]:
        if vis[u] == 0:
            dfs(u, cur)
dfs(1, -1)
min_ans = min(ans)
if min_ans == 0:
    print(-1)
else:
    print(min_ans)
    for x in ans:
        if x == min_ans:
            print(1, end = ' ')
        elif x == min_ans * 2:
            print(2, end = ' ')
        elif x == min_ans * 3:
            print(3, end = ' ')
