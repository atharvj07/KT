import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        ArrayList<Integer> events = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int type = scanner.nextInt();
            if (type >= 3) {
                events.add(type);
            }
        }
        int maxSpeed = 300;
        int numNoOvertake = 0;
        int ans = 0;
        for (int event : events) {
            switch (event) {
                case 3:
                    maxSpeed = scanner.nextInt();
                    break;
                case 5:
                    maxSpeed = 300;
                    break;
                case 6:
                    numNoOvertake++;
                    break;
                case 4:
                    numNoOvertake = 0;
                    break;
            }
            if (event == 3 && numNoOvertake > 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
