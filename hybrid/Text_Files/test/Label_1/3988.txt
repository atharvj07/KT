#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
int main(){
    ll n,s,k,i,j,ans=INT_MAX;
    cin>>n>>s>>k;
    ll a[n+5],b[n+5],mn=INT_MAX;
    for(i=1;i<=n;++i)cin>>a[i];
    for(i=1;i<=n;++i){
        cin>>b[i];
        if(b[i]==b[s])mn=min(mn,a[i]);
    }
    if(mn+a[s]<k)cout<<-1<<'\n';
    else{
        for(i=s+1;i<=n;++i){
            if(b[i]!=b[i-1]&&a[i]>a[i-1]){
                if(mn+a[i]>=k){
                    ans=min(ans,i-s);
                    break;
                }
                mn=a[i];
            }
        }
        mn=INT_MAX;
        for(i=s-1;i>=1;--i){
            if(b[i]!=b[i+1]&&a[i]>a[i+1]){
                if(mn+a[i]>=k){
                    ans=min(ans,s-i);
                    break;
                }
                mn=a[i];
            }
        }
        if(ans==INT_MAX)cout<<-1<<'\n';
        else cout<<ans<<'\n';
    }
    return 0;
}
