#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    if (n % 2 == 1) {
        cout << -1 << endl;
        return 0;
    }
    vector<int> p(n + 1);
    for (int i = 1; i <= n; i += 2) {
        p[i] = i + 1;
        p[i + 1] = n - i + 1;
    }
    for (int i = 1; i <= n; i++) {
        cout << p[i] << " ";
    }
    cout << endl;
    return 0;
}
