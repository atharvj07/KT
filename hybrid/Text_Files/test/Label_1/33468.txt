import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                sb.append('+');
            } else {
                sb.append('-');
                a[i]--;
            }
            for (int j = i + 1; j < n; j++) {
                if (a[j] > a[j - 1]) {
                    sb.append('+');
                    a[j]--;
                } else {
                    sb.append('-');
                }
            }
            break;
        }
        System.out.println(sb);
    }
}
