import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        int min = sum / n;
        int max = n - min + 1;
        if (min % 2 == sum % 2) {
            for (int i = min; i <= max; i += 2) {
                System.out.print(i + " ");
            }
        } else {
            System.out.print("-1");
        }
        System.out.println();
    }
}
