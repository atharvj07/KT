import math

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    lg = [0] * (int(math.log2(max(a)) + 3))
    for i in range(len(lg)):
        lg[i] = 1<<i
    dp = [[[math.inf] * 3 for _ in range(m + 1)] for j in range(n + 1)]
    dp[0][0][0] = 0
    for i in range(n + 1):
        for j in range(m + 1):
            dp[i][j][0] = min(dp[i][j][0], dp[i][j - 1][0])
            for k in range(len(lg)):
                if i + lg[k] <= n:
                    dp[i][j][1] = min(dp[i][j][1], dp[i + lg[k]][j - 1][0] + 1)
                    dp[i][j][1] = min(dp[i][j][1], dp[i][j - 1][1] + 1)
                    dp[i][j][1] = min(dp[i][j][1], dp[i + lg[k]][j - 1][1] + 1)
                    dp[i][j][2] = min(dp[i][j][2], dp[i + lg[k]][j - 1][1])
                    dp[i][j][2] = min(dp[i][j][2], dp[i + lg[k]][j - 1][2])
                    dp[i][j][2] = min(dp[i][j][2], dp[i][j - 1][2])
    print(-1 if dp[n][m][0] == math.inf else dp[n][m][0])
