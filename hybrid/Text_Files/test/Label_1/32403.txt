import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        boolean triple = false;
        int prev1 = in.nextInt();
        int prev2 = in.nextInt();
        for (int i = 2; i < n; i++) {
            int next1 = in.nextInt();
            int next2 = in.nextInt();
            if (prev1 == prev2 && prev1 == next1 && next1 == next2)
                triple = true;
            prev1 = next1;
            prev2 = next2;
        }
        System.out.println(triple ? "Yes" : "No");
    }
}
