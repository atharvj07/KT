m, n = map(int, input().split())
times = [list(map(int, input().split())) for _ in range(m)]

for i in range(m):
    for j in range(1, n):
        times[i][j] += times[i][j-1]
    if i > 0:
        times[i][0] += times[i-1][0]

for i in range(1, m):
    for j in range(1, n):
        times[i][j] = max(times[i][j], times[i-1][j]) + times[i][j] - times[i][j-1]

result = [times[i][-1] for i in range(m)]
print(" ".join(map(str, result)))
