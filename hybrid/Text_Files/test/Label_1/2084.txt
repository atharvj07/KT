#include <iostream>

using namespace std;

bool all_distinct(int x){
    bool vis[10]={};
    while(x){
        if(vis[x%10]) return false;
        vis[x%10]=1;
        x/=10;
    }
    return true;
}

int main(){
    long long n,m;
    cin>>n>>m;
    if(n<m) swap(n,m);
    long long ans=0;
    for(long long i=0; i*m<n; i++){
        for(long long j=0; j<m; j++){
            if(all_distinct(i) && all_distinct(j)) ans++;
        }
    }
    cout<<ans<<endl;
}
