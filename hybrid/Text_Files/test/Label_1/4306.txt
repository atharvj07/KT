#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<long long> b;
  long long mx = -1;
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] >= mx) {
      b.push_back(a[i]);
      mx = a[i];
    }
  }
  if (b.size() == n) {
    for (int i = n - 1; i >= 0; i--) cout << b[i] << " ";
  } else {
    cout << ":(";
  }
  return 0;
}
