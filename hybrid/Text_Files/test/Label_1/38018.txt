n, k = map(int, input().split())
a = list(map(int, input().split()))
q = int(input())
for i in range(q):
    x = int(input())
    ans = -1
    for j in range(k + 1):
        for l in range(k - j + 1):
            if a[j] * l + a[l + j] * (k - l - j) == x:
                ans = k - j - l
                break
        if ans != -1:
            break
    print(ans)
