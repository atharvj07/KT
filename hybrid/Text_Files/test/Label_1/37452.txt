import sys
input = sys.stdin.readline

n = int(input())
s = list(input().rstrip())

for i in range(1, n):
    if s[i] < s[i-1]:
        if i%2 == 0:
            s[i-1], s[i] = s[i], s[i-1]

if any(s[i] < s[i-1] for i in range(1, n)):
    print('NO')
else:
    print('YES')
    print(''.join('01'[i%2] for i in range(n)))
