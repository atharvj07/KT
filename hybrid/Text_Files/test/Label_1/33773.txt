import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        long ans = 0;
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            pq.add((long) a[i]);
        }
        while (pq.size() > 1) {
            long x = pq.poll();
            long y = pq.poll();
            ans += x + y;
            pq.add(x + y);
        }
        System.out.println(ans);
    }
}
