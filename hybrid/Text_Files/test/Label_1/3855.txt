#include <bits/stdc++.h>

using namespace std;

#define all(x) x.begin(), x.end()

typedef long long int ll;

int main() {
    ll ans = 0;
    ll n;
    cin >> n;
    vector<ll> a(n);
    for (ll &x : a) cin >> x;
    priority_queue<pair<ll, ll>> q;
    multiset<ll> ms;
    for (ll i = 0; i < n; i++) ms.insert(a[i]);
    for (ll i = 0; i < n; i++) {
        if (a[i] > 0) {
            q.push({a[i], i + 1});
        }
    }
    while (!q.empty()) {
        auto [sz, ele] = q.top();
        q.pop();
        ans++;
        auto it = ms.find(ele);
        if (it != ms.end()) {
            ms.erase(it);
            a[ele - 1]--;
            if (a[ele - 1] > 0) q.push({a[ele - 1], ele});
        }
        for (ll i = ele + 1; i <= n; i++) {
            it = ms.find(i);
            if (it != ms.end()) {
                ms.erase(it);
                a[i - 1]--;
                if (a[i - 1] > 0) q.push({a[i - 1], i});
            }
        }
    }
    cout << ans << endl;
    return 0;
}
