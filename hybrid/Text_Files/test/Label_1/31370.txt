import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long x = scanner.nextLong();
        boolean found = false;
        long max = -1;
        for (long b = 2; b < x; b++) {
            long num = 0;
            for (long i = 0; i < n; i++) {
                num *= b;
                num += b - 1;
            }
            boolean rotatable = true;
            for (long i = 1; i <= n; i++) {
                long rotated = num;
                for (long j = 0; j < i; j++) {
                    long lastDigit = rotated % 10;
                    rotated /= 10;
                    rotated += lastDigit * Math.pow(10, n - 1);
                }
                if (rotated != num * i) {
                    rotatable = false;
                    break;
                }
            }
            if (rotatable) {
                found = true;
                max = b;
            }
        }
        System.out.println(found ? max : -1);
    }
}
