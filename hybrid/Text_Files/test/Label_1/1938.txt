#include <iostream>
#include <vector>
using namespace std;

int n, m, a[101], good[101][101], ans[101], gcd[1000001];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }
    for (int i = 1; i <= m; ++i) {
        int x, y;
        cin >> x >> y;
        good[x][y] = good[y][x] = 1;
    }
    for (int x = 1; x <= n; ++x) {
        for (int y = x + 1; y <= n; ++y) {
            if (good[x][y]) {
                for (int v = 2; v * v <= a[x]; ++v) {
                    if (a[x] % v == 0 && a[y] % v == 0) {
                        ++ans[v];
                    }
                }
            }
        }
    }
    int res = 0;
    for (int i = 1; i <= n; ++i) { res = max(res, ans[i]); }
    cout << res << endl;
    return 0;
}
