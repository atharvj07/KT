#include <iostream>
#include <vector>
using namespace std;
const int mo=1e9+7;
typedef long long ll;
const int mxn=1e5;
ll a[mxn+5];
struct node{ll sum,cnt;node(){sum=0,cnt=0;}};
node tree[4*mxn+5];
inline node combine(node l,node r){node tmp;tmp.cnt=l.cnt+r.cnt;tmp.sum=(l.sum+r.sum)%mo;return tmp;}
void build(int idx,int l,int r){
	if(l==r){tree[idx].cnt=1;tree[idx].sum=a[l];return;}
	int mid=(l+r)/2;
	build(idx*2,l,mid);
	build(idx*2+1,mid+1,r);
	tree[idx]=combine(tree[idx*2],tree[idx*2+1]);
}
node query(int idx,int l,int r,int x){
	if(l>x)return node();
	if(r<=x)return tree[idx];
	int mid=(l+r)/2;
	return combine(query(idx*2,l,mid,x),query(idx*2+1,mid+1,r,x));
}
int main(){
	int n;cin>>n;
	for(int i=1;i<=n;i++)cin>>a[i];
	build(1,1,n);
	ll sum=0;
	for(int i=1;i<=n;i++){
		node tmp=query(1,1,n,i);
		if(tmp.cnt==0)continue;
		sum+=tmp.sum*query(1,1,n,tmp.cnt-1).cnt;
		sum%=mo;
	}
	cout<<sum<<endl;
}
