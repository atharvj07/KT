import java.util.*;

public class Main {
    static final int MAX_LEN = 501;
    static int[][] dp = new int[MAX_LEN][MAX_LEN];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next(), p = sc.next();
        int n = s.length(), m = p.length();

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (s.charAt(i - 1) == p.charAt(j - 1))
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                else
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
        int[] ans = new int[n + 1];
        for (int i = 0; i <= n; i++)
            ans[i] = dp[i][m];
        for (int x : ans)
            System.out.print(x + " ");
    }
}
