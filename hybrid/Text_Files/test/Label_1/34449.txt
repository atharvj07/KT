import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {

  static int n, m, k;
  static List<List<Integer>> adj;
  static boolean[] vis;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());

    adj = new ArrayList<>();
    for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());

    for (int i = 0; i < m; i++) {
      st = new StringTokenizer(br.readLine());
      int a = Integer.parseInt(st.nextToken());
      int b = Integer.parseInt(st.nextToken());
      adj.get(a).add(b);
      adj.get(b).add(a);
    }

    st = new StringTokenizer(br.readLine());
    k = Integer.parseInt(st.nextToken());

    vis = new boolean[n + 1];
    boolean possible = true;
    List<String> res = new ArrayList<>();
    for (int i = 1; i <= n; i++) {
      if (!vis[i] && dfs(i, -1)) possible = false;
      vis[i] = false;
    }

    if (possible) {
      for (int i = 1; i <= n; i++) {
        if (dfs2(i, -1, "")) res.add(i + " " + (sb.length() == 0 ? "-1" : sb));
      }
      System.out.println("Yes");
      for (String line : res) System.out.println(line);
    } else {
      System.out.println("No");
    }
  }

  static boolean dfs(int node, int prev) {
    vis[node] = true;

    for (int next : adj.get(node)) {
      if (next == prev) continue;
      if (vis[next]) return true;
      if (dfs(next, node)) return true;
    }

    return false;
  }

  static StringBuilder sb = new StringBuilder();

  static boolean dfs2(int node, int prev, String dir) {
    sb.append(node + " ");

    for (int next : adj.get(node)) {
      if (next == prev) continue;

      if (!vis[next]) {
        if (dfs2(next, node, dir + next + " ")) return true;
      } else if (sb.toString().contains(" " + next + " ")) {
        return true;
      }
    }

    sb.setLength(sb.length() - (node + " ").length());
    return false;
  }
}
