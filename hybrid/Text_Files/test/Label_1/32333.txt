import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] lastSeen = new int[k + 1];
        for (int i = 0; i <= k; i++) {
            lastSeen[i] = -1;
        }
        int maxLen = 0;
        int lastDiff = -1;
        for (int i = 0; i < n; i++) {
            if (lastSeen[a[i]] != lastDiff) {
                maxLen = Math.max(maxLen, i - lastSeen[a[i]]);
                lastDiff = lastSeen[a[i]];
            }
            lastSeen[a[i]] = i;
        }
        System.out.println(maxLen);
    }
}
