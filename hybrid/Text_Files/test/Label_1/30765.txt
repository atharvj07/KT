import java.util.*;
import java.math.*;

public class Main {
    static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        BigInteger ans = BigInteger.ONE;
        for (int i = 1; i <= n; i++) {
            BigInteger ways = (BigInteger.valueOf(i).subtract(BigInteger.valueOf(m)).multiply(BigInteger.valueOf(i).subtract(BigInteger.valueOf(m)).subtract(BigInteger.ONE)));
            ways = ways.divide(BigInteger.TWO);
            ways = ways.mod(BigInteger.valueOf(MOD));
            ans = ans.multiply(ways);
            ans = ans.mod(BigInteger.valueOf(MOD));
        }
        System.out.println(ans);
    }
}
