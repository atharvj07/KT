import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class GraphColoring {

    static class Node {
        int id;
        List<Node> neighbors;

        public Node(int id) {
            this.id = id;
            this.neighbors = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }
        for (int i = 1; i <= n; i++) {
            int p = scanner.nextInt();
            nodes[i].neighbors.add(nodes[p]);
        }

        int[] colors = new int[n + 1];
        if (isColorable(nodes, colors)) {
            System.out.println("POSSIBLE");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }

    static boolean isColorable(Node[] nodes, int[] colors) {
        for (int i = 1; i <= nodes.length - 1; i++) {
            if (colors[i] == 0) {
                if (!tryColor(nodes, colors, i, 1)) {
                    return false;
                }
            }
        }
        return true;
    }

    static boolean tryColor(Node[] nodes, int[] colors, int node, int color) {
        if (colors[node] != 0) {
            return true;
        }
        colors[node] = color;
        for (Node neighbor : nodes[node].neighbors) {
            if (colors[neighbor.id] == color) {
                return false;
            }
            if (!tryColor(nodes, colors, neighbor.id, color + 1)) {
                return false;
            }
        }
        return true;
    }
}
