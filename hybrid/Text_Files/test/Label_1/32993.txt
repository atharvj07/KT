import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }

        Arrays.sort(arr);

        int k = 0;
        int[] res = new int[n];

        for (int i = 0; i < n; i++) {
            if (i == 0 || arr[i] - arr[i - 1] <= 1) {
                res[k] = arr[i];
                k++;
            }
        }

        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
