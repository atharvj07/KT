def mult(x, y, m):
    return ((x % m) * (y % m)) % m

def fastpow(x, p, m):
    if p == 0:
        return 1

    if p == 1:
        return x

    val = fastpow(x, int(p / 2), m)
    val = mult(val, val, m)

    if p % 2 != 0:
        val = mult(val, x, m)

    return val

def get_min_periodic(l, a):
    ans = []
    
    for i in range(l):
        if i == 0:
            ans.append(str((a // fastpow(10, i, 1000000007)) % 10 + 1))
        else:
            ans.append('0')
    
    return int(''.join(ans))

l = int(input())
a = int(input())
print(get_min_periodic(l, a))
