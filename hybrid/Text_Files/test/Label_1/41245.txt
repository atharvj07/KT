n = int(input())
balls = list(map(int, input().split()))

dominant_counts = [0] * n

for left in range(n):
    colors = [0] * n
    for right in range(left, n):
        colors[balls[right] - 1] += 1
        dominant_color = colors.index(max(colors))
        dominant_counts[dominant_color] += 1

print(*dominant_counts)
