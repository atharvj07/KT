import java.util.*;
import java.io.*;

public class Main {
    static int[] p = new int[500001];
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        int[] a = new int[n + 1], ans = new int[q + 1];
        for (int i = 1; i <= n; i++)
            a[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j += i) {
                p[j]++;
            }
        }
        for (int i = 1; i <= q; i++) {
            int x = sc.nextInt();
            ans[i] = ans[i - 1] - (p[x] * (p[x] - 1) / 2);
            p[x] = (p[x] == 0) ? 1 : 0;
            ans[i] += (p[x] * (p[x] - 1) / 2);
        }
        for (int i = 1; i <= q; i++)
            System.out.println(ans[i]);
    }
}
