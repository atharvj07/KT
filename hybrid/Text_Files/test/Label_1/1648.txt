#include<bits/stdc++.h>

using namespace std;
const int N = 1005;
typedef pair<int, int> pi;
#define vi(x) vector<x>
#define pb push_back

int main() {
	int n, s;
	cin >> n >> s;
	priority_queue<pi> buy;
	priority_queue<pi, vi(pi), greater<pi>> sell;
	while (n--) {
		char c; int p, q; cin >> c >> p >> q;
		if (c == 'B') {
			buy.push({p, q});
		}
		else {
			sell.push({p, q});
		}
	}
	while (s--) {
		if (buy.size() >= 1) {
			pi tmp = buy.top(); buy.pop();
			cout << 'B' << ' ' << tmp.first << ' ' << tmp.second << '\n';
		}
		if (sell.size() >= 1) {
			pi tmp = sell.top(); sell.pop();
			cout << 'S' << ' ' << tmp.first << ' ' << tmp.second << '\n';
		}
	}
	return 0;
}
