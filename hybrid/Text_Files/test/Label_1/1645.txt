#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll kMaxN = 1e5, kMaxK = 1e5, kMaxM = 1e7;
ll n, k, m, a[kMaxN], b[kMaxN];

ll Solve() {
  multiset<ll> ms;
  for (ll i = 0; i < n; i++) { ms.insert(a[i]); }
  ll s = 0;
  for (ll i = 0; i < m; i++) {
    ms.insert(*ms.begin() + 1);
    ms.erase(ms.begin());
  }
  for (const auto& x : ms) { s += x; }
  return s * 1.0 / (n - m);
}

int main() {
  cin >> n >> k >> m;
  for (ll i = 0; i < n; i++) { cin >> a[i]; }
  cout << fixed << setprecision(12) << Solve() << endl;
  return 0;
}
