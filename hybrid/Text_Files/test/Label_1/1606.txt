#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;

  while (t--) {
    string s;
    cin >> s;

    int n = s.length();
    int ans = n;
    for (int i = 0; i + 1 < n; i++) {
      if (s[i] == s[i + 1]) {
        ans = min(ans, min(i, n - i - 1));
      }
    }

    cout << ans << endl;
  }

  return 0;
}
