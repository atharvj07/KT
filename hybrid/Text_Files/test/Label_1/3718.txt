#include <bits/stdc++.h>
using namespace std;
int n;
vector<int> v;
int main() {
    cin >> n;
    v.resize(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    int ans = 0;
    int lct = -1;
    for (int i = 0; i < n; i++) {
        if (v[i]) {
            if (lct != -1) { ans = max(ans, i - lct); }
            lct = i;
        }
    }
    ans = max(ans, n - lct);
    cout << ans << endl;
    return 0;
}
