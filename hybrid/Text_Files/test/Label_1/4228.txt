#include <bits/stdc++.h>
using namespace std;
pair<string, string> a[305];
int l[305], d[305], s[305], p[305], m;
int main() {
  int n, q, w, e, r, ans = 1e9;
  cin >> n >> q >> w >> e >> r;
  for (int i = 1; i <= n; i++) {
    cin >> a[i].first >> a[i].second;
  }
  for (int i = 1; i <= n; i++) {
    if (i != 1 && a[i].first == a[i - 1].second) d[i] = 1;
    l[i] = l[i - 1] + (!d[i]) * q;
  }
  for (int i = n; i >= 1; i--) {
    if (i != n && a[i].second == a[i + 1].first) s[i] = 1;
    p[i] = p[i + 1] + (!s[i]) * q;
  }
  for (int i = 0; i <= e; i++) {
    m = i * r;
    for (int j = 1; j <= n; j++) {
      m = min(m, p[j + 1] + l[j - 1] + w * (d[j] + s[j]));
    }
    ans = min(ans, m);
  }
  cout << ans << endl;
}
