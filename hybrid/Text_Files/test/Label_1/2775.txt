#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n);
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  vector<int> ans(n, -1);
  vector<bool> vis(n, false);
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (!vis[i]) {
      if (cnt == 3) {
        cout << -1 << endl;
        return 0;
      }
      ans[i] = cnt + 1;
      vis[i] = true;
      vector<int> q;
      q.push_back(i);
      while (!q.empty()) {
        int u = q.back();
        q.pop_back();
        for (auto v : adj[u]) {
          if (!vis[v]) {
            ans[v] = cnt + 1;
            vis[v] = true;
            q.push_back(v);
          } else if (ans[v] != ans[u]) {
            cout << -1 << endl;
            return 0;
          }
        }
      }
      cnt++;
    }
  }
  if (cnt != 3) {
    cout << -1 << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
