s = input()
x, y = 0, 0
op = 0
for i in s:
  if i == 'L':
    x -= 1
  elif i == 'R':
    x += 1
  elif i == 'U':
    y += 1
  else:
    y -= 1
  if abs(x) + abs(y) > abs(y):
    op += abs(y) - abs(x)
    if x < 0:
      s = s.replace('R', 'L', abs(y) - abs(x), 1)
    else:
      s = s.replace('L', 'R', abs(y) - abs(x), 1)
print(-1 if x != 0 or y != 0 else op)
