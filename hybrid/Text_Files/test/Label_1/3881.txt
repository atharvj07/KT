#include <iostream>
#include <vector>
#include <stack>
#include <set>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::vector<int>> g(n);
    for (int i = 0; i < m; ++i) {
        int u, v;
        std::cin >> u >> v;
        --u, --v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    std::vector<int> deg(n);
    std::vector<bool> used(n);
    std::stack<int> st;
    st.push(0);
    used[0] = true;
    while (!st.empty()) {
        int v = st.top();
        st.pop();
        for (auto u : g[v]) {
            if (!used[u]) {
                ++deg[v];
                ++deg[u];
                st.push(u);
                used[u] = true;
            }
        }
    }
    int mxdeg = 0;
    std::set<int> ans;
    for (int i = 0; i < n; ++i) {
        mxdeg = std::max(mxdeg, deg[i]);
    }
    for (int i = 0; i < n; ++i) {
        if (deg[i] == mxdeg) {
            ans.insert(i);
        }
    }
    for (auto i : ans) {
        for (auto j : g[i]) {
            if (deg[i] + deg[j] == mxdeg * 2) {
                std::cout << i + 1 << ' ' << j + 1 << '\n';
                deg[i]--;
                deg[j]--;
            }
        }
    }
    return 0;
}
