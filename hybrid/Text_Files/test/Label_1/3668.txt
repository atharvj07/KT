#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  vector<int> u;
  for (int i = n - 1; i >= 0; i--) {
    if (u.empty() || u.back() != v[i]) { u.push_back(v[i]); }
  }
  cout << u.size() << endl;
  for (int i = u.size() - 1; i >= 0; i--) { cout << u[i] << " "; }
  cout << endl;
  return 0;
}
