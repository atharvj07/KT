#include <bits/stdc++.h>
using namespace std;

int main() {
    int A, B, a[2], b[2];
    cin >> A >> B;
    for (int i = 0; i < 2; i++) {
        cin >> a[i] >> b[i];
        swap(a[i], b[i]);
    }
    sort(a, a + 2);
    sort(b, b + 2);
    cout << (a[0] <= A && b[0] <= B ? "YES" : "NO") << endl;
}
