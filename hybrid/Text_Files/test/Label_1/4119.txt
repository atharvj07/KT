#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long int ll;

bool cmp(ll a, ll b) {
    return a > b;
}

int main() {
    ll n1, n2, n3;
    cin >> n1 >> n2 >> n3;
    vector<ll> v1(n1), v2(n2), v3(n3);
    for (ll i = 0; i < n1; i++) cin >> v1[i];
    for (ll i = 0; i < n2; i++) cin >> v2[i];
    for (ll i = 0; i < n3; i++) cin >> v3[i];
    sort(v1.begin(), v1.end(), cmp);
    sort(v2.begin(), v2.end(), cmp);
    sort(v3.begin(), v3.end(), cmp);
    ll ans = 0;
    while (v2.size() > 0 && v3.size() > 0) {
        if (v2.back() > v3.back()) {
            ans += v2.back();
            v2.pop_back();
        } else {
            ans += v3.back();
            v3.pop_back();
        }
    }
    while (v1.size() > 0 && v2.size() > 0) {
        if (v1.back() > v2.back()) {
            ans += v1.back();
            v1.pop_back();
        } else {
            ans += v2.back();
            v2.pop_back();
        }
    }
    while (v1.size() > 0 && v3.size() > 0) {
        if (v1.back() > v3.back()) {
            ans += v1.back();
            v1.pop_back();
        } else {
            ans += v3.back();
            v3.pop_back();
        }
    }
    cout << ans << endl;
    return 0;
}
