n, s = map(int, input().split())
landings = [list(map(int, input().split())) for _ in range(n)]

def can_takeoff(before, after, s):
    end_before = (before[0]*60 + before[1] + s + 1) # End time of before landing plus safety interval
    start_after = (after[0]*60 + after[1]) # Start time of after landing
    return start_after - end_before >= s + 1 # Check if gap is enough for takeoff and safety intervals

# Check if takeoff can be inserted before the first landing
if landings[0][0] > 0 or (landings[0][0] == 0 and landings[0][1] > s):
    print(0, 0)
else:
    for i in range(n-1):
        if can_takeoff(landings[i], landings[i+1], s):
            takeoff_time = (landings[i][0]*60 + landings[i][1] + s + 1)
            print(takeoff_time//60, takeoff_time%60)
            break
    else: # If the loop completes without finding a slot, place takeoff after last landing
        last_landing_end = (landings[-1][0]*60 + landings[-1][1] + s + 1)
        print(last_landing_end//60, last_landing_end%60)
