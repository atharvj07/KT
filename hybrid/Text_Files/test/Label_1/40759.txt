from heapq import *

n, k, q = map(int, input().split())
a = [0] * (n + 3)
t = [0] * (n + 3)
for i in range(1, n + 1):
    t[i] = int(input())

for i in range(q):
    x, y = map(int, input().split())
    if x == 1:
        heappush(a, -t[y])
        if len(a) > k:
            heappop(a)
    else:
        print("YES" if -a[0] == t[y] else "NO")
