#include <iostream>
#include <vector>
using namespace std;

const int MOD = 1e9 + 7;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k;
    cin >> n >> k;
    long long dp[k + 1];
    dp[k] = 1;
    for (int i = k - 1; i >= 0; i--) {
      dp[i] = ((dp[i + 1] * (n + 1)) % MOD + dp[i + 1] - 1 + MOD) % MOD;
    }
    cout << dp[0] << endl;
  }
  return 0;
}
