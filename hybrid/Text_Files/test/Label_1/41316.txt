import sys

n = int(input())
b = list(map(int, input().split()))
a = list(map(int, input().split()))
x = [0] + [int(i) - 1 for i in input().split()]
k = [0] + [int(i) for i in input().split()]

for i in range(n - 1, 0, -1):
    b[x[i]] += b[i] // k[i] * (k[i] - 1)
    b[i] = (b[i] - 1) % k[i] + 1

for i in range(n):
    if b[i] < a[i]:
        print("NO")
        sys.exit(0)

print("YES")
