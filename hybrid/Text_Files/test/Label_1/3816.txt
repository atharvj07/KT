#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
const int maxn = 1005;

vi days[maxn];
int ans[maxn][maxn];

int main() {
    int n, k, d;
    cin >> n >> k >> d;

    for (int i = 0; i < d; i++) {
        days[i].resize(n);
        for (int j = 0; j < n; j++) {
            cin >> days[i][j];
        }
    }
    int first = days[0][0];
    int cnt1 = 0, cntk = 0;
    for (int i = 0; i < n; i++) {
        if (days[0][i] == first) {
            cnt1++;
            ans[0][i] = 1;
        } else {
            cntk++;
            ans[0][i] = k;
        }
    }
    if (cnt1 * cntk == 0) {
        cout << -1 << endl;
        return 0;
    }
    for (int i = 1; i < d; i++) {
        for (int j = 0; j < n; j++) {
            if (days[i][j] == first) {
                ans[i][j] = 1;
            } else {
                ans[i][j] = k;
            }
        }
    }
    for (int i = 0; i < d; i++) {
        for (int j = 0; j < n; j++) {
            cout << ans[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
