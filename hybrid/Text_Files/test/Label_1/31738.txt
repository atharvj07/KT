import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        long temp = r;
        long cnt = 0;
        if (l == 0)
            cnt++;
        while (temp > 0) {
            if (temp % 10 == (temp / getPow(temp, 10)) % 10)
                cnt++;
            temp /= 10;
        }
        System.out.println(cnt);
    }

    static long getPow(long x, long pow) {
        long ans = 1;
        for (int i = 0; i < pow; i++) {
            ans *= x;
        }
        return ans;
    }
}
