import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int[] cards;
    static boolean[] visited;
    static boolean solved;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int n = input.nextInt();
            if (n == 0) {
                break;
            }
            cards = new int[n];
            for (int i = 0; i < n; i++) {
                cards[i] = input.nextInt();
            }
            visited = new boolean[53];
            solved = false;
            Arrays.sort(cards);
            solve(0, 0);
            System.out.println(solved ? Arrays.toString(cards).replaceAll("[\\[\\],]", "") : "No");
        }
        input.close();
    }

    static void solve(int sum, int count) {
        if (solved) {
            return;
        }
        if (count == cards.length) {
            solved = true;
            return;
        }
        for (int i = 0; i < cards.length; i++) {
            if (visited[cards[i]]) {
                continue;
            }
            if (count == 0 || sum % cards[i] == 0) {
                visited[cards[i]] = true;
                solve(sum + cards[i], count + 1);
                visited[cards[i]] = false;
            }
        }
    }
}
