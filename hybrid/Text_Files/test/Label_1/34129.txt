import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] p = new int[n];
    for (int i = 0; i < n; i++) {
      p[i] = sc.nextInt();
    }
    int[] f = new int[n];
    for (int i = 0; i < n; i++) {
      f[i] = sc.nextInt();
    }
    if (p[0] != 1 || p[n - 1] != n) {
      System.out.println("NO");
      return;
    }
    for (int i = 0; i < n; i++) {
      if (f[i] > n - i - 1) {
        System.out.println("NO");
        return;
      }
    }
    System.out.println("YES");
  }
}
