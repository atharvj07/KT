# include <bits/stdc++.h>
using namespace std;
int main() {
    int q; cin >> q;
    for (int i = 1; i <= q; i++) {
        int n, ans = 0; cin >> n;
        vector<int> a(n);
        for (int& x : a) { cin >> x; ans += x; }
        sort(begin(a), end(a));
        cout << ceil(1.0 * ans / n) << endl;
    }
    return 0;
}
