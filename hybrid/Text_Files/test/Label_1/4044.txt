#include <bits/stdc++.h>
using namespace std;

long long ans;
int main(){
    int n;
    cin >> n;
    vector<int> a(n);
    for(int i = 0; i < n; i++) cin >> a[i];
    stack<int> st;
    for(int i = 0; i < n; i++){
        while(!st.empty() && a[st.top()] <= a[i]) st.pop();
        if(!st.empty()) ans += st.top() - i;
        st.push(i);
    }
    cout << ans << endl;
}
