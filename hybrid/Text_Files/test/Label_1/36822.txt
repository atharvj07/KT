import queue
def swap(a, i, j):
    aa = list(a)
    aa[i], aa[j] = aa[j], aa[i]
    return "".join(aa)

n = int(input())
a = input()
b = input()
q = queue.Queue()
vis = [False] * (10**n)
q.put((a, 0))
vis[int(a)] = True
while not q.empty():
    aa, cnt = q.get()
    if aa==b:
        print(cnt)
        break
    for i in range(n-1):
        for j in [1, -1]:
            c = int(swap(aa, i, i+1))
            if c>=10**(n-1) and not vis[c]:
                vis[c] = True
                q.put((swap(aa, i, i+1), cnt+j))
else:
    print(-1)
