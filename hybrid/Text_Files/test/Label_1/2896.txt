#include <cstdio>
#include <algorithm>
#include <cstring>
using namespace std;
int a[5], b[5], cnt[101];
int main() {
  int ans = 0;
  for (int i = 0; i < 5; i++) {
    scanf("%d", &a[i]);
    ans += a[i];
  }
  sort(a, a + 5);
  memcpy(b, a, sizeof(a));
  for (int i = 0; i < 5; i++) {
    memset(cnt, 0, sizeof(cnt));
    for (int j = i; j < 5; j++) cnt[b[j]]++;
    int sum = 0, num = 0;
    for (int j = 1; j <= 100; j++) {
      if (cnt[j] > 0) {
        sum += j;
        num++;
      }
      if (cnt[j] >= 3) num = 3;
      if (cnt[j] >= 2) num = max(num, 2);
      if (num == 3) break;
    }
    if (num == 3) ans = min(ans, sum);
    if (num == 2) ans = min(ans, sum * 2);
  }
  printf("%d\n", ans);
  return 0;
}
