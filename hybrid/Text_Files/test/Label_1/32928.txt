import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int x = sc.nextInt();
    int[] q = new int[n];
    int c = 0, d = 0;
    for (int i = 0; i < n; i++) {
      String s = sc.next();
      int v = sc.nextInt();
      if(s.equals("+")) {
        q[i] = v;
        c++;
      } else {
        q[i] = -v;
        d++;
      }
    }
    Arrays.sort(q);
    for (int i = 0; i < n; i++) {
      if (q[i] < 0 && x >= -q[i]) {
        d--;
        x += q[i];
      } else if (q[i] > 0) {
        x += q[i];
        c--;
      }
    }
    System.out.println(x+" "+d);
  }
}
