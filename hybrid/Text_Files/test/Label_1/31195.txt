import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = br.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);

        String[] staminas0 = br.readLine().split(" ");
        List<Integer> st0 = new ArrayList<>();
        for (String sta : staminas0) {
            st0.add(Integer.parseInt(sta));
        }

        String[] staminas1 = br.readLine().split(" ");
        List<Integer> st1 = new ArrayList<>();
        for (String sta : staminas1) {
            st1.add(Integer.parseInt(sta));
        }

        st0.sort((a, b) -> Integer.compare(a, b));
        st1.sort((a, b) -> Integer.compare(a, b));

        boolean possible = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (st0.isEmpty() || st1.isEmpty()) {
                    possible = false;
                    break;
                }
                int maxStamina = Math.max(st0.remove(st0.size() - 1), st1.remove(st1.size() - 1));
                if (maxStamina < i + j + 1) {
                    possible = false;
                    break;
                }
            }
            if (!possible) {
                break;
            }
        }

        if (possible) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
