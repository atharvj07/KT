#include <bits/stdc++.h>
#define ll long long
const ll m = 998244353;
ll dp[505][505][2][2], n, k;
int main() {
  std::cin >> n >> k;
  dp[0][0][0][0] = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      for (int x = 0; x < 2; x++) {
        for (int y = 0; y < 2; y++) {
          dp[i + 1][j][x][y] = (dp[i + 1][j][x][y] + dp[i][j][x][y]) % m;
          dp[i][j + 1][x][y] = (dp[i][j + 1][x][y] + dp[i][j][x][y]) % m;
          if (x == y) {
            dp[i + 1][j + 1][0][y] = (dp[i + 1][j + 1][0][y] + dp[i][j][x][y]) % m;
            if (j - i >= k) dp[i + 1][j + 1][1][y] =
                (dp[i + 1][j + 1][1][y] + dp[i][j][x][y]) % m;
          } else {
            dp[i + 1][j + 1][1][0] = (dp[i + 1][j + 1][1][0] + dp[i][j][x][y]) % m;
            if (j - i >= k) dp[i + 1][j + 1][1][1] =
                (dp[i + 1][j + 1][1][1] + dp[i][j][x][y]) % m;
          }
        }
      }
    }
  }
  std::cout << dp[n][n][1][0] << '\n';
  return 0;
}
