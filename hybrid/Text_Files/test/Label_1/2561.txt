#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<pair<char, int>> cards(n);
  for (int i = 0; i < n; i++) {
    char c, v;
    cin >> c >> v;
    cards[i] = {c, v - '0'};
  }

  int min_hints = 5;
  for (int i = 0; i < n; i++) {
    int cnt_color = 0, cnt_val = 0;
    for (int j = 0; j < n; j++) {
      if (cards[i].first == cards[j].first) cnt_color++;
      if (cards[i].second == cards[j].second) cnt_val++;
    }
    if (cnt_color == 1) min_hints--;
    if (cnt_val == 1) min_hints--;
  }
  cout << min_hints << endl;
}
