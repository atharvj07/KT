from collections import deque
from sys import stdin

n, q = map(int, stdin.readline().split())
a = deque(map(int, stdin.readline().strip().split()))
qs = [int(stdin.readline()) for _ in range(q)]

for m in qs:
    m %= n * 2 - 2  # period
    if m < n:
        print(a[m], a[m + 1])
    else:
        print(a[n - 1 - (m - n)], a[n - (m - n)])
