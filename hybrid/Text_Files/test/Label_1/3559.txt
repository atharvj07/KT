#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    cout << a[1] - a[0] << ' ' << (upper_bound(a.begin(), a.end(), a[1]) - lower_bound(a.begin(), a.end(), a[1])) << '\n';
    return 0;
}
