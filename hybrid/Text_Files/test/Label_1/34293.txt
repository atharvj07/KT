import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = br.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        long x = Long.parseLong(br.readLine());
        Edge[] edges = new Edge[m];
        for (int i = 0; i < m; i++) {
            String[] uvw = br.readLine().split(" ");
            edges[i] = new Edge(Integer.parseInt(uvw[0]), Integer.parseInt(uvw[1]), Integer.parseInt(uvw[2]));
        }
        long ans = solve(n, m, x, edges);
        System.out.println(ans);
    }
    private static long solve(int n, int m, long x, Edge[] edges) {
        Arrays.sort(edges, (a, b) -> Long.compare(a.w, b.w));
        List<Edge> white = new ArrayList<>();
        List<Edge> black = new ArrayList<>();
        long sum = 0;
        for (Edge edge : edges) {
            if (x % edge.w == 0) {
                long cnt = x / edge.w;
                white.add(edge);
                black.add(edge);
                sum += edge.w * cnt;
                x -= edge.w * cnt;
            }
        }
        if (x != 0) {
            return 0;
        }
        long ans = 1;
        for (Edge edge : white) {
            ans *= 2;
            ans %= 1000000007;
        }
        for (Edge edge : black) {
            ans *= 2;
            ans %= 1000000007;
        }
        ans %= 1000000007;
        return ans;
    }
    private static class Edge {
        int u;
        int v;
        long w;
        Edge(int u, int v, long w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }
}
