import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    static final int MOD = 1000000007;
    static int n, q;
    static int[] a;
    static boolean[] swapped;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer strtok = new StringTokenizer(br.readLine());
        n = Integer.parseInt(strtok.nextToken());
        q = Integer.parseInt(strtok.nextToken());
        a = new int[n + 1];
        swapped = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(br.readLine());
        }
        int[][] queries = new int[q][2];
        for (int i = 0; i < q; i++) {
            strtok = new StringTokenizer(br.readLine());
            queries[i][0] = Integer.parseInt(strtok.nextToken());
            queries[i][1] = Integer.parseInt(strtok.nextToken());
        }
        long ans = 0;
        for (int i = 0; i < (1 << q); i++) {
            int[] b = a.clone();
            for (int j = 0; j < q; j++) {
                if ((i & (1 << j)) != 0) {
                    swapped[queries[j][0]] = !swapped[queries[j][0]];
                    swapped[queries[j][1]] = !swapped[queries[j][1]];
                    int tmp = b[queries[j][0]];
                    b[queries[j][0]] = b[queries[j][1]];
                    b[queries[j][1]] = tmp;
                }
            }
            int[] c = new int[n + 1];
            for (int j = 1; j <= n; j++) {
                if (swapped[j]) {
                    c[j] = b[n + 1 - j];
                } else {
                    c[j] = b[j];
                }
            }
            long cnt = mergeSort(c, 1, n);
            ans += cnt;
        }
        System.out.println(ans % MOD);
    }

    static long mergeSort(int[] c, int start, int end) {
        if (start >= end) {
            return 0;
        }
        int mid = (start + end) >> 1;
        long left = mergeSort(c, start, mid);
        long right = mergeSort(c, mid + 1, end);
        int[] tmp = new int[end - start + 2];
        int p1 = start, p2 = mid + 1, p = 0;
        long ans = 0;
        while (p1 <= mid && p2 <= end) {
            if (c[p1] > c[p2]) {
                ans += mid - p1 + 1;
                tmp[p++] = c[p2++];
            } else {
                tmp[p++] = c[p1++];
            }
        }
        while (p1 <= mid) {
            tmp[p++] = c[p1++];
        }
        while (p2 <= end) {
            tmp[p++] = c[p2++];
        }
        for (int i = 0; i < p; i++) {
            c[start + i] = tmp[i];
        }
        return left + right + ans;
    }
}
