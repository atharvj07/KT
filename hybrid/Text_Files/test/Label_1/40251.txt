from collections import defaultdict, deque
n,m=map(int,input().split())
e=list(map(int,input().split()))
g=defaultdict(list)
for _ in range(m):
    a,b=map(int,input().split())
    g[b].append(a)
ind=list(range(n))
for i in g:
    for j in g[i]:
        ind[j]-=1
q=deque()
for i in range(n):
    if not ind[i]:
        q.append(i)
ans=0
while q:
    x=[]
    while q:
        i=q.popleft()
        for j in g[i]:
            ind[j]+=1
            if ind[j]==1:
                q.append(j)
    for i in range(n):
        if not ind[i] and not e[i]:
            x.append(i)
        elif not ind[i] and e[i]:
            ans+=1
            x.append(i)
    for i in x:
        ind[i]=-1
print(ans)
