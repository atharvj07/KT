from collections import defaultdict, deque

def min_coprocessor_calls():
    n, m = map(int, input().split())
    exec_env = list(map(int, input().split()))
    dependencies = defaultdict(list)
    indegree = [0] * n
    
    for _ in range(m):
        u, v = map(int, input().split())
        dependencies[v].append(u)
        indegree[u] += 1
    
    queue = deque([i for i in range(n) if indegree[i] == 0])
    coprocessor_calls = 0
    while queue:
        next_round = []
        for task in queue:
            if exec_env[task] == 1:
                coprocessor_calls += 1
                for dep in dependencies[task]:
                    indegree[dep] -= 1
                    if indegree[dep] == 0:
                        next_round.append(dep)
            else:
                for dep in dependencies[task]:
                    indegree[dep] -= 1
                    if indegree[dep] == 0:
                        next_round.append(dep)
        queue = deque(next_round)
    
    return coprocessor_calls

print(min_coprocessor_calls())
