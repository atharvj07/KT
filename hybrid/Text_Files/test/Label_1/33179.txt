import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[][] judgment = new int[4][2];
        String line;
        while ((line = br.readLine()) != null) {
            String[] s = line.split(" ");
            double left = Double.parseDouble(s[0]);
            double right = Double.parseDouble(s[1]);
            if (left >= 1.1) judgment[0][0]++;
            else if (left >= 0.6) judgment[1][0]++;
            else if (left >= 0.2) judgment[2][0]++;
            else judgment[3][0]++;
            if (right >= 1.1) judgment[0][1]++;
            else if (right >= 0.6) judgment[1][1]++;
            else if (right >= 0.2) judgment[2][1]++;
            else judgment[3][1]++;
        }
        for (int i = 0; i < 4; i++) {
            System.out.println(judgment[i][0] + " " + judgment[i][1]);
        }
    }
}
