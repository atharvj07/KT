#include <bits/stdc++.h>
using namespace std;
const int N=102, M=102;
int n,m,a[N*M],z[N*M],l[N*M],trie[N*M][2],pw[50];
char s[N];

void update(int id,int val) {
    a[id]=val;
    if (l[id]>0) l[id]=max({l[trie[id][0]],l[trie[id][1]],l[id]});
}

int find(int id,int d) {
    int ans=0;
    int p=id;
    for (;d>=0;d--) {
        int v=1-pw[d]&(z[id]&pw[d]);
        !l[trie[p][v]]?d=0:ans=d,v^=1,id=trie[id][v],p=id;
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    pw[0]=1;
    for (int i=1;i<50;i++) pw[i]=pw[i-1]*2;
    cin>>n;
    for (int i=1;i<=n;i++) {
        cin>>s;
        int len=strlen(s);
        int p=0;
        for (int j=0;j<len;j++) {
            if (!trie[p][s[j]-'0']) trie[p][s[j]-'0']=++z[0];
            p=trie[p][s[j]-'0'];
        }
        l[p]=len;
    }
    for (int i=1;i<=z[0];i++) for (int j=0;j<2;j++) if (trie[i][j]) {
        update(trie[i][j],a[i]+find(trie[i][j],l[i]));
    }
    cin>>m;
    while (m--) {
        int u,v; cin>>u>>v;
        update(z[0],max(a[u],a[v]));
        cout<<a[z[0]]<<"\n";
    }
    return 0;
}
