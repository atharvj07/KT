n = int(input())

dp = [0] * (n + 1)

for turn in range(n):
    k = int(input())
    arr = []
    for _ in range(k):
        c, d = map(int, input().split())
        arr.append((c, d))
    arr.sort(reverse = True)

    for i in range(len(arr)):
        if i % 10 == 9:
            arr[i] = (0, arr[i][1] * 2)

    for j in range(n - 1, turn - 1, -1):
        dp[j] = max(dp[j], dp[j - 1])
    for c, d in arr:
        if dp[turn] + c <= 3:
            dp[turn + 1] = max(dp[turn + 1], dp[turn] + d)

print(max(dp))
