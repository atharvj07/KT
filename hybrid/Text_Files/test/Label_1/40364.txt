n=int(input())
v=[int(i) for i in input().split()]
L=[0]*n
R=[0]*n
a=[]
for i in range(n):
    while a and v[a[-1]]>=v[i]:a.pop()
    if a: L[i]=a[-1]+1
    a.append(i)
a=[]
for i in range(n-1,-1,-1):
    while a and v[a[-1]]>=v[i]:a.pop()
    if a: R[i]=a[-1]-1
    a.append(i)
for i in range(n):
    print(max(v[L[i]:R[i]+1]))
