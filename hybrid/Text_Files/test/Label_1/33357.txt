import java.util.*;

public class ElectionBribes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] votes = new int[m + 1];
        int[] costs = new int[n];
        for (int i = 0; i < n; i++) {
            int party = scanner.nextInt();
            int cost = scanner.nextInt();
            votes[party]++;
            costs[i] = cost;
        }

        Arrays.sort(costs);

        int minCost = 0;
        for (int i = 0; i < n; i++) {
            minCost += costs[i];
            votes[1]++;
            if (votes[1] > votes[2]) {
                break;
            }
        }

        System.out.println(minCost);
    }
}
