#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
typedef long long ll;

int main() {
    ll a, b, w, x, c;
    cin >> a >> b >> w >> x >> c;
    ll ans = 0, tmp;
    while (c > a) {
        ans++;
        if (b >= x) b -= x;
        else tmp = x - b, a--, b = w - tmp;
        c--;
    }
    cout << ans << endl;
    return 0;
}
