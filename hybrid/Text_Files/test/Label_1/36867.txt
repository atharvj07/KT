import sys

for t in range(int(input())):
    n = int(input())
    st = set()
    cur = set()
    mx = 0
    for _ in range(n):
        w = input()
        cur = cur | set(w)
        if len(cur) < len(w):
            mx = max(mx, len(cur))
            cur = set()
            cur.add(w)
    print(mx)
