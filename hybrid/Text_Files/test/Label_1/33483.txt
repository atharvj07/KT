import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        int ghd = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == a[(n + 1) / 2 - 1]) {
                ghd = a[i];
                break;
            }
        }
        System.out.println(ghd);
    }
}
