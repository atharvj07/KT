import java.util.*;

public class Main {
  static ArrayList<Integer> gems = new ArrayList<>(Arrays.asList(1, 2, 3, 4, 5, 6, 7, 8, 9));
  static int[] r = new int[2];
  static int[] c = new int[2];
  static int[] d = new int[2];
  static int[][] ans = new int[2][2];

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    r[0] = sc.nextInt();
    r[1] = sc.nextInt();
    c[0] = sc.nextInt();
    c[1] = sc.nextInt();
    d[0] = sc.nextInt();
    d[1] = sc.nextInt();
    System.out.println(solve() ? ans[0][0] + " " + ans[0][1] + "\n" + ans[1][0] + " " + ans[1][1] : -1);
  }

  static boolean solve() {
    if (r[0] + c[0] + d[0] != r[1] + c[1] + d[1]) return false;
    int rc = (r[0] + c[0] + d[0]) / 2;
    if (r[0] > rc || r[1] > rc || c[0] > rc || c[1] > rc || d[0] > rc || d[1] > rc)
      return false;
    int[] row1 = new int[3];
    row1[0] = (r[0] - c[0] + d[0]) / 2;
    row1[1] = rc - row1[0];
    row1[2] = r[0] - row1[0] - row1[1];
    int[] row2 = new int[3];
    row2[0] = rc - row1[2];
    row2[1] = row1[1];
    row2[2] = row1[0];
    Arrays.sort(row1);
    Arrays.sort(row2);
    for (int a : row1)
      for (int b : row2) {
        if (!gems.removeIf(e -> e == a)) return false;
        if (!gems.removeIf(e -> e == b)) return false;
        ans[0][0] = a;
        ans[0][1] = b;
        return true;
      }
    return false;
  }
}
