#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int r, b, k;
        std::cin >> r >> b >> k;
        if (r == b && k % 2 == 0)
            std::cout << "REBEL" << std::endl;
        else
            std::cout << "OBEY" << std::endl;
    }
    return 0;
}
