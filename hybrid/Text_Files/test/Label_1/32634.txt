import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();
        int k = scanner.nextInt();
        List<Set<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new HashSet<>());
        }
        for (int i = 0; i < q; i++) {
            int t = scanner.nextInt();
            if (t == 1) {
                int x = scanner.nextInt();
                int y = scanner.nextInt();
                graph.get(x).add(y);
                graph.get(y).add(x);
            } else if (t == 2) {
                int z = scanner.nextInt();
                Set<Integer> infected = new HashSet<>();
                dfs(z, z, infected, graph, k);
                System.out.println(infected.size());
            } else if (t == 3) {
                for (int j = 1; j <= n; j++) {
                    graph.get(j).clear();
                }
            }
        }
        scanner.close();
    }

    private static void dfs(int root, int z, Set<Integer> infected, List<Set<Integer>> graph, int k) {
        if (infected.contains(z)) {
            return;
        }
        infected.add(z);
        for (int neighbor : graph.get(z)) {
            dfs(root, neighbor, infected, graph, k - 1);
        }
    }
}
