def can_i_pop(c, t):
    for i in range(len(t)):
        if c[i] != t[i]: return False
    return True

def zuma(n, c):
    ret = 0
    while len(c) > 0:
        t = []
        for x in c:
            if len(t) == 0 or t[-1] == x: t.append(x)
            else: break
        c = c[len(t):]
        ind = 0
        while ind < len(c):
            if not can_i_pop(t, c[ind : ind + len(t)]): ind += 1
            else: break
        c = c[ind + len(t):]
        ret += 1
    return ret

n = int(input())
c = list(map(int, input().split()))
print(zuma(n, c))
