import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = i + 1;
        }
        int[] i = new int[n];
        for (int k = 2; k <= n; k++) {
            int g = 0;
            for (int j = 0; j < k; j++) {
                for (int l = j + 1; l < k; l++) {
                    g = Math.max(g, gcd(a[j], a[l]));
                }
            }
            i[k - 2] = g;
        }

        for (int j = 0; j < n - 1; j++) {
            System.out.print(i[j] + " ");
        }
    }

    static int gcd(int a, int b) {
        if (a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }
}
