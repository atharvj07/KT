import sys
input = sys.stdin.readline

n, L, a = map(int, input().split())
times = []
for _ in range(n):
    x = list(map(int, input().split()))
    times.append(x)

count = 0
free = 0
for t, l in times:
    free = max(free, t)
    if free <= t:
        count += 1
        free = t + l + a
    else:
        free += l + a
print(count)
