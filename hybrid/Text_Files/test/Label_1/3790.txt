#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, m, k;
    cin >> n >> m >> k;
    vector<int> p(n), s(n), c(k);
    for (int i = 0; i < n; i++) cin >> p[i];
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < k; i++) cin >> c[i];
    vector<pair<int, int>> v;
    for (int i = 0; i < n; i++) v.push_back(make_pair(p[i], s[i]));
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 0; i < k; i++) {
        bool flag = false;
        for (int j = 0; j < n; j++) {
            if (v[j].first == c[i]) {
                flag = true;
                if (v[j].second != v[n - 1].second) ans++;
            }
        }
        if (!flag) ans++;
    }
    cout << ans << endl;
    return 0;
}
