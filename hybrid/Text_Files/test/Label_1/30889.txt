import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int length = s.length();
        int cntLinks = 0, cntPearls = 0;
        int i = 0;
        while(i < length) {
            if(s.charAt(i) == '-') cntLinks++;
            else cntPearls++;
            i++;
        }
        if(cntLinks == 0 || cntPearls == 0) {
            System.out.println("NO");
            return;
        }
        int avgLinks = cntLinks / cntPearls;
        if(cntLinks % cntPearls != 0) {
            System.out.println("NO");
            return;
        }
        int removedLinks = 0;
        List<Integer> links = new ArrayList<>();
        i = 0;
        int pearl = 0;
        while(i < length) {
            if(s.charAt(i) == '-') {
                if(pearl == avgLinks) {
                    removedLinks++;
                    pearl = 0;
                } else {
                    links.add(i);
                    pearl++;
                }
            }
            i++;
        }
        if(removedLinks == cntPearls) {
            System.out.println("YES");
            return;
        }
        System.out.println("NO");
    }
}
