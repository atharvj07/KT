import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader bf = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(bf.readLine());
        int[] books = new int[n];
        String[] line = bf.readLine().split(" ");
        Map<Integer, List<Integer>> map = new HashMap<>();
        int day = 1;
        for (int i = 0; i < n; i++) {
            int book = Integer.parseInt(line[i]);
            if (book > 0) {
                map.computeIfAbsent(book, k -> new ArrayList<>()).add(day);
                day++;
            }
        }
        int max = 0;
        for (int i = 1; i <= map.size(); i++) {
            List<Integer> days = map.get(i);
            if (days.size() >= 2 && days.size() <= 5) {
                max = i;
            }
        }
        if (max == 0) {
            System.out.println(-1);
        } else {
            List<Integer> days = map.get(max);
            boolean[] read = new boolean[n + 1];
            for (int i : days) {
                read[i] = true;
            }
            int num = max;
            for (int i = 1; i <= n; i++) {
                if (!read[i]) {
                    read[i] = true;
                    num = num != max ? num : max - 1;
                    days.add(i);
                }
            }
            System.out.println(num);
            for (int i = 1; i <= n; i++) {
                int book = read[i] ? days.get(num) : 0;
                System.out.print(book + " ");
                if (book > 0) {
                    num--;
                }
            }
        }
    }
}
