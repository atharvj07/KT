n, m, k = map(int, input().split())
a = list(map(int, input().split()))
d = [0] * (m + 1)
for i in range(m):
    l, r, di = map(int, input().split())
    d[l - 1] += di
    d[r] -= di
q = []
for i in range(k):
    x, y = map(int, input().split())
    q.append((x, y))

for x, y in q:
    for i in range(x, y + 1):
        a[i - 1] += d[i - 1]
        d[i] = 0

print(*a)
