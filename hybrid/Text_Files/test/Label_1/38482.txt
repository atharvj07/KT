import math
s, x1, x2 = map(int, input().split())
t1, t2 = map(int, input().split())
p, d = map(int, input().split())
def f(x):
    if d == 1:
        return math.ceil((x - p) / t1) * t1 + t2 * abs(x2 - x)
    else:
        return math.ceil((s - p) / t1) * t1 + t2 * abs(x - x2)
print(min(f(x1), f(x2)))
