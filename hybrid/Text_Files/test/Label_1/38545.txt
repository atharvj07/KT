def min_operations_to_genome(n, s):
    def distance(a, b):
        return min(abs(ord(a) - ord(b)), 26 - abs(ord(a) - ord(b)))
    
    min_operations = float('inf')
    for i in range(n - 3):
        current_operations = sum(distance(s[i+j], "ACTG"[j]) for j in range(4))
        min_operations = min(min_operations, current_operations)
    return min_operations

n = int(input().strip())
s = input().strip()
print(min_operations_to_genome(n, s))
