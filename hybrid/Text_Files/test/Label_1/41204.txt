import math

s = input()
n = int(input())

dict = {}
for char in s:
    if char not in dict:
        dict[char] = 0
    dict[char] += 1

keys = list(dict.keys())
keys.sort()

k = 0
sheets = 0
ans = ''

while k < len(keys) and sheets * n < len(s):
    if sheets * n + dict[keys[k]] >= len(s):
        ans += keys[k] * math.ceil((len(s) - sheets * n) / dict[keys[k]])
    else:
        ans += keys[k] * dict[keys[k]]
    sheets += 1
    k += 1

if sheets * n < len(s):
    print(-1)
else:
    print(sheets)
    print(ans)
