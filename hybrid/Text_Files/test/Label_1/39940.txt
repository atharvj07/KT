import sys
input = sys.stdin.readline

n, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for _ in range(q):
    l, r = map(int, input().split())
    dif = 0
    for i in range(l - 1, r):
        dif += abs(a[i] - b[i])
    if dif % 2: print(-1)
    else: print(dif // 2)
