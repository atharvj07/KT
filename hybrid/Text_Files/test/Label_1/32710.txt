import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        long curSum = 0;
        for (int i = 1; i <= n; i++) {
            curSum += l + i - 1;
        }
        long minDiff = Long.MAX_VALUE;
        long minFlavor = 0;
        for (int i = 1; i <= n; i++) {
            long diff = Math.abs(curSum - l - i + 1 - (curSum - (l + i - 1)));
            if (diff < minDiff) {
                minDiff = diff;
                minFlavor = curSum - (l + i - 1);
            }
        }
        System.out.println(minFlavor);
    }
}
