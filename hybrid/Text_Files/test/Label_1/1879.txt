#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

void solve() {
  int n, r;
  cin >> n >> r;
  vector<int> x(n);
  for (int &i : x) cin >> i;
  sort(x.begin(), x.end());
  int ans = 0;
  int cur = 0;
  for (int i = 0; i < n; i++) {
    if (x[i] <= cur) continue;
    ans++;
    if (cur > 0) cur -= r;
    cur += x[i] + r;
  }
  cout << ans << endl;
}

int main() {
  int q;
  cin >> q;
  while (q--) solve();
}
