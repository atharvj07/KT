def find_median_binary_string(n, m, removed_strings):
    total_strings = 2**m
    remaining_strings = total_strings - n
    median_index = (remaining_strings - 1) // 2

    removed_indices = set(int(r, 2) for r in removed_strings)
    left, right = 0, 2**m - 1

    for _ in range(m):
        mid = (left + right) // 2
        left_count = mid - left + 1 - sum(1 for i in range(left, mid + 1) if i in removed_indices)
        
        if median_index >= left_count:
            median_index -= left_count
            left = mid + 1
        else:
            right = mid

    return "{:0{}b}".format(left, m)

t = int(input().strip())
for _ in range(t):
    n, m = map(int, input().split())
    removed_strings = [input().strip() for _ in range(n)]
    print(find_median_binary_string(n, m, removed_strings))
