import java.util.*;
import java.util.stream.*;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = s.nextInt();
        }

        int max = 0;
        for (int i = 0; i < n; i++) {
            max += a[i];
        }
        if (max == 0) {
            System.out.println(-1);
            return;
        }
        int min = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                min++;
            }
        }

        for (int i = 1; i <= max; i++) {
            if (i <= min || i > max) {
                continue;
            }
            System.out.println(i);
        }
    }
}
