import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        in.nextLine();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        in.nextLine();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }
        int ans = 1;
        while (true) {
            int x = ans;
            boolean ok = false;
            while (x > 0) {
                int d = x % 10;
                for (int i = 0; i < n; i++) {
                    if (a[i] == d) {
                        ok = true;
                        break;
                    }
                }
                for (int i = 0; i < m; i++) {
                    if (b[i] == d) {
                        ok = true;
                        break;
                    }
                }
                if (ok) {
                    break;
                }
                x /= 10;
            }
            if (ok) {
                System.out.println(ans);
                break;
            }
            ans++;
        }
    }
}
