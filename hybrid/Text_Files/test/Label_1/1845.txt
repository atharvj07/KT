#include <iostream>

using namespace std;

int main() {
  int t[5];
  cin >> t[0] >> t[1] >> t[2] >> t[3] >> t[4];

  bool discarded[5] = {false, false, false, false, false};
  int same = 0;
  for (int i = 0; i < 5; i++) {
    for (int j = 0; j < 5; j++) {
      if (i == j) continue;
      if (t[i] == t[j] && !discarded[i] && !discarded[j]) {
        same++;
        discarded[i] = true;
        discarded[j] = true;
        break;
      }
    }
  }

  int sum = 0;
  for (int i = 0; i < 5; i++) {
    if (!discarded[i]) sum += t[i];
  }

  if (same > 2) sum -= t[0];

  cout << sum << endl;

  return 0;
}
