#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
ll dp[100010][2]; // dp[i][0] = don't reverse, dp[i][1] = reverse

int32_t main() {
    int n; cin >> n;
    vector<pair<ll, ll> > v(n);
    vector<string> s(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].second;
        cin >> s[i];
    }
    for (int i = 1; i <= n; i++) {
        dp[i][0] = dp[i - 1][0];
        dp[i][1] = dp[i - 1][1];
        if (s[i - 1] < s[i - 1]) {
            dp[i][0] = max(dp[i][0], dp[i - 1][1] + v[i - 1].second);
        } else {
            dp[i][1] = max(dp[i][1], dp[i - 1][0] + v[i - 1].second);
        }
    }
    cout << dp[n][dp[n][0] > dp[n][1] ? 0 : 1];
    return 0;
}
