from sys import stdin

x, y = tuple(map(int, stdin.readline().split()))
grid = [list(stdin.readline().strip()) for _ in range(x)]

def valid(r, c):
    for i in range(r - s, r + s + 1):
        for j in range(c - s, c + s + 1):
            if not (0 <= i < x and 0 <= j < y and (i == r or j == c or abs(i - r) == abs(j - c)) and grid[i][j] == '*'):
                return False
    return True

stars = []
for i in range(x):
    for j in range(y):
        for s in range(1, min(x, y)):
            if valid(i, j):
                stars.append((i + 1, j + 1, s))
                for l in range(i - s, i + s + 1):
                    for k in range(j - s, j + s + 1):
                        if l == i or k == j or abs(l - i) == abs(k - j):
                            grid[l][k] = '.'
                break
if any(gr != '.' for gr in sum(grid, [])):
    print(-1)
else:
    print(len(stars))
    for r, c, s in stars:
        print(r, c, s)
