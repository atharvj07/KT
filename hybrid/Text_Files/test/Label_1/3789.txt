#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
	ll t;
	cin>>t;
	while(t--){
		string s;
		cin>>s;
		ll n=s.size(), ans=0;
		for(ll i=0; i<n; i++){
			ll x=0,y=0;
			for(ll j=i; j<n; j++){
				x<<=1;
				x|=s[j]-'0';
				y++;
				if(x==y) ans++;
			}
		}
		cout<<ans<<endl;
	}
	return 0;
}
