#include <bits/stdc++.h>
using namespace std;

const int mxN = 1e5 + 5;
int n, m, a[mxN], b[mxN], c[mxN];
vector<pair<int, int>> v[mxN];

int32_t main() {
    cin >> n >> m;
    for (int i = 0; i < m; ++i) {
        cin >> a[i] >> b[i];
        --a[i], --b[i];
        if (a[i] > b[i]) swap(a[i], b[i]);
        v[a[i]].emplace_back(b[i], i);
    }
    set<int> all;
    for (int i = 0; i < n; ++i) for (int j = i + 1; j < n; ++j) {
        if (v[i].size() == 0 || v[j].size() == 0) continue;
        if (v[i].back().first < j || v[j].back().first < i) continue;
        int idx1 = v[i].back().second, idx2 = v[j].back().second;
        if ((a[idx1] == i && b[idx1] == j) || (a[idx2] == i && b[idx2] == j)) {
            c[i]++, c[j]++;
            all.insert(i), all.insert(j);
        }
    }
    if (all.size() < n) cout << "NO";
    else {
        vector<int> ua, ub;
        cout << "YES\n";
        for (int i = 0; i < n; ++i) ua.push_back(all.count(i) ? 1 : n);
        for (int i = 0; i < n; ++i) ub.push_back(c[i] ? 1 : n);
        for (int i = 0; i < n; ++i) cout << ua[i] << ' ';
        cout << '\n';
        for (int i = 0; i < n; ++i) cout << ub[i] << ' ';
    }
}
