#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int del = 1;
    char c = s[0];
    for (int i = 1; i < n; i++) {
        if (c != s[i]) {
            del++;
            c = s[i];
        }
    }
    cout << del
         << "\n";
    return 0;
}
