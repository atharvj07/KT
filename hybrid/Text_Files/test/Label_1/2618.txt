#include <bits/stdc++.h>
using namespace std;
char s[2001], a[2001], b[2001];
int main() {
  cin >> s;
  int n = strlen(s);
  for (int i = 0; i < n; i++) a[i] = s[i];
  int mx = 0, k = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == '0') k++;
    mx = max(k, mx);
    a[i] = s[i];
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == '0' && k > 0) {
      k--;
      b[i] = '0';
    } else {
      b[i] = s[i];
    }
  }
  cout << b;
  return 0;
}
