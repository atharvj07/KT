#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    map<char, set<char>> mp;
    bool ok = true;
    for (int i = 0; i < n / 2; i++) {
      if (s[i] != s[n - 1 - i]) ok = false;
      mp[s[i]].insert(s[n - 1 - i]);
    }
    for (auto x : mp) {
      if (x.second.size() > 1) ok = false;
    }
    if (ok)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
