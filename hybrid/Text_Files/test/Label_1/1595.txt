#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> dp(n * 2);
    dp[0] = 1;
    int ans = 1;
    for (int i = 1; i < n * 2; i++) {
      dp[i] = 1;
      for (int j = 0; j < i; j++) {
        if (a[i % n] > a[j % n]) dp[i] = max(dp[i], dp[j] + 1);
      }
      ans = max(ans, dp[i]);
    }
    cout << ans << '\n';
  }
  return 0;
}
