import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        if (n % 2 == 0) {
            if (m <= n / 2) {
                System.out.println(m + 1);
            } else {
                System.out.println(m - 1);
            }
        } else {
            System.out.println(m);
        }
    }
}
