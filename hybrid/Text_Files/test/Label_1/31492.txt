import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            long mask = 0;
            for (int j = 0; j < k; j++) {
                mask |= 1L << sc.nextInt();
            }
            masks[i] = mask;
        }
        int q = sc.nextInt();
        while (q-- > 0) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            switch (op) {
                case 1:
                    System.out.println(test(masks, m));
                    break;
                case 2:
                    set(masks, m);
                    break;
                case 3:
                    clear(masks, m);
                    break;
                case 4:
                    flip(masks, m);
                    break;
                case 5:
                    System.out.println(all(masks, m));
                    break;
                case 6:
                    System.out.println(any(masks, m));
                    break;
                case 7:
                    System.out.println(none(masks, m));
                    break;
                case 8:
                    System.out.println(count(masks, m));
                    break;
                case 9:
                    System.out.println(val(masks, m));
                    break;
            }
        }
    }

    public static int test(long[] masks, int m) {
        return (masks[m] & (1L << 63)) > 0 ? 1 : 0;
    }

    public static void set(long[] masks, int m) {
        masks[m] |= (1L << 63);
    }

    public static void clear(long[] masks, int m) {
        masks[m] &= ~(1L << 63);
    }

    public static void flip(long[] masks, int m) {
        masks[m] ^= (1L << 63);
    }

    public static int all(long[] masks, int m) {
        return (masks[m] & (1L << 63)) == (1L << 63) ? 1 : 0;
    }

    public static int any(long[] masks, int m) {
        return (masks[m] & (1L << 63)) > 0 ? 1 : 0;
    }

    public static int none(long[] masks, int m) {
        return (masks[m] & (1L << 63)) == 0 ? 1 : 0;
    }

    public static int count(long[] masks, int m) {
        int cnt = 0;
        for (int i = 0; i < 63; i++) {
            if ((masks[m] & (1L << i)) > 0) {
                cnt++;
            }
        }
        return cnt;
    }

    public static int val(long[] masks, int m) {
        int v = 0;
        for (int i = 0; i < 63; i++) {
            if ((masks[m] & (1L << i)) > 0) {
                v |= (1 << i);
            }
        }
        return v;
    }
}
