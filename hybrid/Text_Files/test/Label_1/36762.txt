n = int(input())
grades = list(map(int, input().split()))

total = sum(grades)
needed_total = 4.5 * n

if total >= needed_total:
    print(0)
else:
    grades.sort()
    count = 0
    for grade in grades:
        total += 5 - grade
        count += 1
        if total >= needed_total:
            print(count)
            break
