import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static char[][] grid;
    static int[][] dp;
    static int[] dx = {0, 1, 0, 1};
    static int[] dy = {0, 0, 1, 1};

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        grid = new char[n][n];
        dp = new int[n][n];
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < n; j++) {
                grid[i][j] = line.charAt(j);
            }
        }
        dp[n - 1][n - 1] = grid[n - 1][n - 1] == '#' ? 1 : 0;
        for (int i = n - 2; i >= 0; i--) {
            dp[i][n - 1] = grid[i][n - 1] == '#' ? dp[i + 1][n - 1] + 1 : dp[i + 1][n - 1];
        }
        for (int j = n - 2; j >= 0; j--) {
            dp[n - 1][j] = grid[n - 1][j] == '#' ? dp[n - 1][j + 1] + 1 : dp[n - 1][j + 1];
        }
        for (int i = n - 2; i >= 0; i--) {
            for (int j = n - 2; j >= 0; j--) {
                dp[i][j] = Math.min(dp[i + 1][j], dp[i][j + 1]) + (grid[i][j] == '#' ? 1 : 0);
                for (int k = 0; k < 4; k++) {
                    int nx = i + dx[k];
                    int ny = j + dy[k];
                    if (nx < n && ny < n) {
                        dp[i][j] = Math.min(dp[i][j], dp[nx][ny] + 1);
                    }
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
