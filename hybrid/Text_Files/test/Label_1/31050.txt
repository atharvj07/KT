import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long m = sc.nextLong();
        if (m <= 2) {
            System.out.println(m + " " + m);
            return;
        }
        long n = 0;
        for (long i = 2; i * i * i <= m; i++) {
            n = i;
        }
        long x = n * n * n;
        long cnt = 0;
        for (long i = 1; i < n; i++) {
            cnt += n;
        }
        cnt += m / x;
        System.out.println(cnt + " " + x);
    }
}
