from bisect import bisect_left

n = int(input())
a = list(map(int, input().split()))
m = int(input())
qs = []
for _ in range(m):
    qs.append([int(_) for _ in input().split()])
a.sort()
cnt = 0
for j, (k, pos) in enumerate(qs):
    j += 1
    if not a:
        print(-1)
    else:
        idx = bisect_left(a, a[cnt - 1] + 1, lo=cnt)
        while idx < len(a) and len(a) - idx < k:
            idx += 1
        if idx == len(a):
            idx = cnt
        ans = a[idx]
        cnt = idx
        if len(a) - cnt + 1 < k:
            print(-1)
        else:
            print(ans)
