import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long k = sc.nextInt();
            long l = sc.nextInt();
            long r = sc.nextInt();
            long p = sc.nextInt();
            long ans = 1;
            for (long j = k * k + l + 1; j <= k * k + r + 1; j++) {
                ans = lcm(ans, j);
            }
            System.out.println(ans % p);
        }
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }

    public static long lcm(long a, long b) {
        return a * b / gcd(a, b);
    }
}
