import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    String[] line = br.readLine().split(" ");
    int n = Integer.parseInt(line[0]);
    int m = Integer.parseInt(line[1]);

    String[] strings = new String[n];
    for (int i = 0; i < n; i++) {
      strings[i] = br.readLine();
    }

    int[][] costs = new int[n][m];
    for (int i = 0; i < n; i++) {
      line = br.readLine().split(" ");
      for (int j = 0; j < m; j++) {
        costs[i][j] = Integer.parseInt(line[j]);
      }
    }

    System.out.println(solve(strings, costs));
  }

  private static int solve(String[] strings, int[][] costs) {
    int n = strings.length;
    int m = strings[0].length();

    int[][] minCost = new int[n][26];
    for (int i = 0; i < n; i++) {
      Arrays.fill(minCost[i], Integer.MAX_VALUE);
    }

    List<List<Integer>> charPositions = new ArrayList<>();
    for (int i = 0; i < 26; i++) {
      charPositions.add(new ArrayList<>());
    }

    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        charPositions.get(strings[i].charAt(j) - 'a').add(i);
      }
    }

    int answer = 0;
    for (int i = 0; i < 26; i++) {
      int minPos = -1;
      int minCostPos = -1;
      for (int j : charPositions.get(i)) {
        if (minCost[j][i] < minCostPos) {
          minPos = j;
          minCostPos = minCost[j][i];
        }
      }

      if (minPos != -1) {
        for (int j = 0; j < m; j++) {
          minCost[minPos][strings[minPos].charAt(j) - 'a'] = Integer.MAX_VALUE;
        }
        answer += costs[minPos][i];
      }
    }

    return answer;
  }
}
