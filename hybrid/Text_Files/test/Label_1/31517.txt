import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        HashMap<Integer, Integer> fareMap = new HashMap<>();
        fareMap.put(1, 550);
        fareMap.put(2, 450);
        fareMap.put(3, 400);
        fareMap.put(4, 300);
        fareMap.put(5, 250);
        fareMap.put(6, 200);
        fareMap.put(7, 150);
        while (true) {
            int d = sc.nextInt();
            if (d == 0) {
                break;
            }
            int hd = sc.nextInt();
            int md = sc.nextInt();
            int a = sc.nextInt();
            int ha = sc.nextInt();
            int ma = sc.nextInt();
            int distance = Math.abs(a - d);
            int fare = fareMap.get(distance);
            if (isInDiscountTime(hd, md) || isInDiscountTime(ha, ma)) {
                fare = (int) Math.ceil(fare * 0.5);
            }
            System.out.println(fare);
        }
    }

    private static boolean isInDiscountTime(int h, int m) {
        return (h >= 17 && h <= 19 && m >= 30) || (h == 17 && m == 0) || (h == 19 && m == 30);
    }
}
