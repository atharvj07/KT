import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static class Table implements Comparable<Table> {
        private final int[] values;

        Table(int[] values) {
            this.values = values;
        }

        @Override
        public int compareTo(Table o) {
            return Arrays.compare(values, o.values);
        }
    }

    private static boolean isPossible(int n, int m, Table[] a, Table[] b) {
        Arrays.sort(a);
        Arrays.sort(b);
        return Arrays.equals(a, b);
    }

    private static List<Integer> findSequence(int n, int m, Table[] a, Table[] b) {
        List<Integer> sequence = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            Table[] sortedByColumn = new Table[n];
            for (int j = 0; j < n; j++) {
                sortedByColumn[j] = new Table(Arrays.copyOf(a[j].values, m));
                sortedByColumn[j].values[i] = b[j].values[i];
            }
            if (isPossible(n, m, a, sortedByColumn)) {
                sequence.add(i + 1);
                a = sortedByColumn;
            }
        }
        return sequence;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        Table[] a = new Table[n];
        Table[] b = new Table[n];
        for (int i = 0; i < n; i++) {
            int[] rowA = new int[m];
            int[] rowB = new int[m];
            for (int j = 0; j < m; j++) {
                rowA[j] = scanner.nextInt();
            }
            for (int j = 0; j < m; j++) {
                rowB[j] = scanner.nextInt();
            }
            a[i] = new Table(rowA);
            b[i] = new Table(rowB);
        }
        if (!isPossible(n, m, a, b)) {
            System.out.println(-1);
            return;
        }
        List<Integer> sequence = findSequence(n, m, a, b);
        System.out.println(sequence.size());
        for (int column : sequence) {
            System.out.print(column + " ");
        }
    }
}
