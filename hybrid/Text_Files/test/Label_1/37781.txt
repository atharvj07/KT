n,m=[int(x) for x in input().split()]
adj=[[] for x in range(n+1)]
for a,b in [list(map(int, input().split())) for x in range(m)]:
    adj[a].append(b)
    adj[b].append(a)
    
flag=False
for a in range(1,n+1):
    for b in adj[a]:
        for c in adj[b]:
            if a!=c:
                flag=True
                break
        if flag:break
    if flag:break
if flag: print('YES')
else: print('NO')
