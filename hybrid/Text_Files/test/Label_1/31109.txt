import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scn = new Scanner(System.in);
        int n = scn.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = scn.nextInt();
        int j = -1;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] == 0)
                j = i;
            else if (j != -1)
                --j;
        }
        System.out.println((j == -1) ? n : j + 1);
    }
}
