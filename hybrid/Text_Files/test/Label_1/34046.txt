import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int testCases = scanner.nextInt();

        while (testCases-- > 0) {
            int people = scanner.nextInt();
            int wealthThreshold = scanner.nextInt();
            long totalSavings = 0;
            int wealthyPeople = 0;

            for (int i = 0; i < people; i++) {
                long savings = scanner.nextInt();
                totalSavings += savings;
                if (savings >= wealthThreshold) {
                    wealthyPeople++;
                }
            }

            long maxWealthyPeople = Math.min(people, (int) (totalSavings / wealthThreshold));
            System.out.println(Math.max(wealthyPeople, maxWealthyPeople));
        }
    }
}
