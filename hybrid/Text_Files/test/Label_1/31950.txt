import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            int k = sc.nextInt();
            if (n % 2 != k % 2 || n < k) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                long each = n / k;
                for (int i = 0; i < k - 1; i++) {
                    System.out.print(each + " ");
                }
                System.out.println(n - each * (k - 1));
            }
        }
    }
}
