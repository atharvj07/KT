import sys
readline = sys.stdin.readline

for t in range(int(readline())):
    n = int(readline())
    a = list(map(int, readline().split()))
    n1 = sorted([*a])
    n2 = sorted([*a], reverse=True)
    ans = n1[0]*n1[1]*n1[2]*n1[3]*n1[4]
    ans = max(ans, n2[0]*n2[1]*n2[2]*n2[3]*n2[4])
    print(ans)
