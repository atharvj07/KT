def find_spanning_tree(n, edges):
    from collections import defaultdict

    # Create a graph from edges
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    # Find the vertex with the maximum degree
    max_degree_vertex = max(graph, key=lambda x: len(graph[x]))
    
    # Use DFS to find a spanning tree starting from the vertex with the max degree
    def dfs(v, visited, tree):
        visited.add(v)
        for neighbor in graph[v]:
            if neighbor not in visited:
                tree.append((v, neighbor))
                dfs(neighbor, visited, tree)
                
    visited = set()
    spanning_tree = []
    dfs(max_degree_vertex, visited, spanning_tree)
    
    return spanning_tree

# Reading input
n, m = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]

# Find and print the spanning tree
spanning_tree = find_spanning_tree(n, edges)
for edge in spanning_tree:
    print(*edge)
