import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                cnt++;
            } else {
                ans += (cnt + 1) / 2;
                cnt = 0;
            }
        }
        ans += (cnt + 1) / 2;
        System.out.println(ans - 1);
    }
}
