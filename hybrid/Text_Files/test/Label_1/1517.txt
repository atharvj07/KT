#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[100009], b = 0;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) {
        if (b == 0 or a[i] != a[i - 1]) b++;
        if (b > 6) break;
    }
    for (int i = n; i >= 1; i--) {
        if (b == 0 or a[i] != a[i + 1]) b++;
        if (b > 6) break;
        else n = i;
    }
    cout << n << endl;
}
