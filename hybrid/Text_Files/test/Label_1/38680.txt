n, m = map(int, input().split())
s = input()
ans = 0
for i in s:
    if i == '.':
        ans += 1
    else:
        ans = 0
print(ans)
for _ in range(m):
    pos, c = map(str, input().split())
    pos = int(pos) - 1
    if s[pos] == '.':
        ans += 1 - (c == '.')
    elif c == '.':
        ans -= 1
    s = s[:pos] + c + s[pos + 1:]
    print(ans)
