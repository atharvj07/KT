import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        long a = scanner.nextInt();
        long b = scanner.nextInt();
        long ans = (b / k) - (a - 1) / k;
        System.out.println(ans);
    }
}
