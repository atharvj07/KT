import java.util.*;
import java.io.*;

public class TreeDecomposition {

    static class Edge {
        int u, v;

        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Path {
        int u, v;

        Path(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        ArrayList<Edge>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(new Edge(u, v));
            adj[v].add(new Edge(u, v));
        }

        if (!isTree(adj)) {
            System.out.println("No");
            return;
        }

        ArrayList<Path> paths = new ArrayList<>();
        dfs(1, -1, adj, paths);

        System.out.println("Yes");
        System.out.println(paths.size());
        for (Path path : paths) {
            System.out.println(path.u + " " + path.v);
        }
    }

    static boolean isTree(ArrayList<Edge>[] adj) {
        int[] visited = new int[adj.length];
        if (dfsUtil(1, -1, adj, visited) != adj.length - 1) {
            return false;
        }
        for (int i = 1; i < adj.length; i++) {
            if (visited[i] == 0) {
                return false;
            }
        }
        return true;
    }

    static int dfsUtil(int node, int parent, ArrayList<Edge>[] adj, int[] visited) {
        visited[node] = 1;
        int count = 1;
        for (Edge edge : adj[node]) {
            if (edge.v != parent) {
                count += dfsUtil(edge.v, node, adj, visited);
            }
        }
        return count;
    }

    static void dfs(int node, int parent, ArrayList<Edge>[] adj, ArrayList<Path> paths) {
        for (Edge edge : adj[node]) {
            if (edge.v != parent) {
                paths.add(new Path(node, edge.v));
                dfs(edge.v, node, adj, paths);
            }
        }
    }
}
