import java.util.*;

public class Main {
    static class Event implements Comparable<Event> {
        int type, value;
        boolean seen;

        Event(int type, int value) {
            this.type = type;
            this.value = value;
            this.seen = false;
        }

        public int compareTo(Event other) {
            return -Integer.compare(this.value, other.value);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        PriorityQueue<Event> events = new PriorityQueue<>();

        for (int i = 0; i < n; i++) {
            int type = sc.nextInt();
            if (type == 1 || type == 3) {
                events.add(new Event(type, sc.nextInt()));
            } else {
                events.add(new Event(type, 0));
            }
        }

        int speed = 0;
        boolean overtakeAllowed = true;
        int unseenCount = 0;

        while (!events.isEmpty()) {
            Event event = events.poll();
            if (event.seen) {
                continue;
            }
            event.seen = true;
            if (event.type == 1) {
                speed = event.value;
            } else if (event.type == 3) {
                speed = Math.min(speed, event.value);
            } else if (event.type == 4) {
                overtakeAllowed = true;
            } else if (event.type == 5) {
                speed = 0;
            } else if (event.type == 6) {
                overtakeAllowed = false;
            } else if (event.type == 2 && !overtakeAllowed) {
                unseenCount++;
            }
        }

        System.out.println(unseenCount);
    }
}
