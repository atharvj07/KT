import java.util.*;
 
public class Main {
  static final int MAXN = 200000 + 10;
  static int[] b = new int[MAXN];
 
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    int m = n >> 1;
 
    for (int i = 0; i < n; ++i)
      a[i] = sc.nextInt();
 
    Arrays.sort(a);
 
    for (int i = m - 1; i >= 0; --i)
      b[i] = a[n - i - 1];
    for (int i = m; i < n; ++i)
      b[i] = a[i - m];
 
    for (int i = 0; i < n; ++i)
      System.out.println(b[i]);
  }
}
