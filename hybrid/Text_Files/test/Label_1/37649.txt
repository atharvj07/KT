n, k = map(int, input().split())
p = list(map(int, input().split()))

# Check if the given sequence is stack-sortable
def is_stack_sortable(p):
    stack = []
    expected = 1
    for i in p:
        while stack and stack[-1] == expected:
            stack.pop()
            expected += 1
        if i == expected:
            expected += 1
        elif not stack or stack[-1] > i:
            stack.append(i)
        else:
            return False
    return True

# Construct the lexicographically maximal permutation
def construct_max_permutation(n, p):
    if not is_stack_sortable(p):
        return -1
    remaining = set(range(1, n+1)) - set(p)
    max_perm = p[:]
    while remaining:
        next_max = max(remaining)
        if max_perm and next_max < max_perm[-1]:
            max_perm.append(next_max)
        else:
            return -1
        remaining.remove(next_max)
    return max_perm

# Main
max_perm = construct_max_permutation(n, p)
if max_perm == -1:
    print(-1)
else:
    print(' '.join(map(str, max_perm)))
