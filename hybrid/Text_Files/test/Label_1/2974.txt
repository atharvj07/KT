#include <bits/stdc++.h>

using namespace std;

int n, m, ans;
char a[105][105];
int sum[105][105];

int get(int x1, int y1, int x2, int y2){
    return sum[x2][y2] - sum[x2][y1 - 1] - sum[x1 - 1][y2] + sum[x1 - 1][y1 - 1];
}

int main(){
    cin >> n >> m;
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            cin >> a[i][j];
            sum[i][j] = sum[i - 1][j] + sum[i][j - 1] + (a[i][j] == 'W') - (a[i][j] == 'B');
        }
    }
    for(int i = 1; i <= n; ++i){
        for(int j = 1; j <= m; ++j){
            for(int i2 = i; i2 <= n; ++i2){
                for(int j2 = j; j2 <= m; ++j2){
                    if((get(i, j, i2, j2) == 0) && (i != i2 || j != j2)){
                        if(i == 1 && j == 1){
                            ans += 2;
                            continue;
                        }
                        for(int k = i2; k <= n; ++k){
                            int sm = get(k, 1, k, j2);
                            if(sm == 0){
                                ans += 2;
                                break;
                            }
                            else if(sm > 0){
                                ans++;
                                break;
                            }
                        }
                        for(int k = i2; k <= n; ++k){
                            int sm = get(k, j2 + 1, k, m);
                            if(sm == 0){
                                ans += 2;
                                break;
                            }
                            else if(sm < 0){
                                ans++;
                                break;
                            }
                        }
                        for(int k = j2; k <= m; ++k){
                            int sm = get(1, k, i2, k);
                            if(sm == 0){
                                ans += 2;
                                break;
                            }
                            else if(sm < 0){
                                ans++;
                                break;
                            }
                        }
                        for(int k = j2; k <= m; ++k){
                            int sm = get(i, k, i2 - 1, k);
                            if(sm == 0){
                                ans += 2;
                                break;
                            }
                            else if(sm > 0){
                                ans++;
                                break;
                            }
                        }
                        break;
                    }
                }
            }
        }
    }
    cout << ans;
}
