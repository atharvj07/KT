import sys

n, k, m, t = map(int, sys.stdin.readline().split())

l = n
ex = [0] * (m + 1)
ex[k] = 1

for i in range(t):
    a, b = map(int, sys.stdin.readline().split())
    if a:
        if b <= k:
            k += 1
        l += 1
    else:
        if ex[b + 1]:
            ex[b + 1] = 0
            k -= 1
        if ex[b]:
            ex[b] = 0
            k -= 1
        l -= 1
    ex[k] = 1
    print(l, k)
