#include <iostream>
#include <iomanip>

int main() {
    int N, P;
    double p = 0.0;
    std::cin >> N >> P;
    for (int i = 1; i < N; i++) {
        p += (double)(N - i) / N * (N - i - 1) / (N - 1) * (N - i - 2) / (N - 2);
        if (p >= P) {
            std::cout << i << std::endl;
            return 0;
        }
    }
    std::cout << N << std::endl;
    return 0;
}
