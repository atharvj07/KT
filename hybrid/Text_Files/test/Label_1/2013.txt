#include<iostream>
#include<vector>
using namespace std;
int main()
{
    int n,k;
    cin>>n>>k;
    vector<int> a(n);
    for(int i=0;i<n;++i)
        cin>>a[i];
    int prv=k,add=0;
    for(int i=0;i<n;++i)
    {
        if(a[i]>=prv)
            prv=a[i]+1;
        else
        {
            add+=prv-a[i];
            prv=a[i]+1;
        }
    }
    cout<<add<<endl;
    for(int i=0;i<n;++i)
    {
        cout<<max(a[i],prv-1)<<" ";
        prv=max(a[i],prv-1)+1;
    }
    cout<<endl;
    return 0;
}
