import sys

n, m, s, t = map(int, sys.stdin.readline().split())

graph = [[] for _ in range(n+1)]

for _ in range(m):
    a, b = map(int, sys.stdin.readline().split())
    graph[a].append(b)
    graph[b].append(a)

dis = [-1] * (n+1)

def dfs(cur, d):
    dis[cur] = d
    for nxt in graph[cur]:
        if dis[nxt] == -1:
            dfs(nxt, d+1)

dfs(s, 0)
ans = 0
for i in range(1, n+1):
    if dis[i] == -1 or dis[i] > dis[t]:
        for j in graph[i]:
            if dis[j] == -1 or dis[j] > dis[t]:
                ans += 1
print(ans)
