from sys import stdin
n, m = map(int, stdin.readline().split())
a = [[int(x) for x in stdin.readline().split()] for _ in range(n)]
b = [0 for _ in range(m)]
for i in range(n):
  for j in range(m):
    b[j] ^= a[i][j]
if 0 in b:
  print("NIE")
else:
  print("TAK")
  print(*[i+1 for i in range(m) if b[i] == 0])
