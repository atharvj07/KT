n = int(input())
p = list(map(int, input().split()))
a = [1]
l = [[] for _ in range(n+1)]
for i in range(n):
    l[p[i]].append(i)
for i in range(1, n+1):
    prev = -1
    for j in l[i]:
        if prev > -1 and j < prev:
            a[-1] += 1
        prev = j
    a.append(a[-1])
for ans in a:
    print(ans)
