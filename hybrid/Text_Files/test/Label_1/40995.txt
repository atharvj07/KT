import queue
import sys

input = sys.stdin.readline

def bfs(adj, n, start, dist):
    q = queue.Queue()
    q.put((start, 0))
    dist[start] = 0
    while not q.empty():
        cur, d = q.get()
        if d > dist[cur]: continue
        for nxt in adj[cur]:
            if d+1 < dist[nxt]:
                dist[nxt] = d+1
                q.put((nxt, d+1))

tc = int(input())
for _ in range(tc):
    n = int(input())
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        a,b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)
    dist = [sys.maxsize] * (n+1)
    bfs(adj, n, 1, dist)
    k = max(dist[1:])
    print(k)
