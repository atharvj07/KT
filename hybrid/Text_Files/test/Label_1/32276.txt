import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        if (a[0] == 0) {
            System.out.println("cslnb");
        } else {
            int gcd = a[0];
            for (int i = 1; i < n; i++) {
                gcd = gcd(gcd, a[i]);
            }
            if (gcd == 1) {
                System.out.println("sjfnb");
            } else {
                System.out.println("cslnb");
            }
        }
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        } else {
            return gcd(b, a % b);
        }
    }
}
