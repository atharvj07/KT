import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
        }
        int sum = 0;
        for (int i : m) {
            sum += i;
        }
        int ans = x / sum * n;
        int remain = x % sum;
        for (int i = 0; i < n && remain > 0; i++) {
            while (remain >= m[i]) {
                ans++;
                remain -= m[i];
            }
        }
        System.out.println(ans);
    }
}
