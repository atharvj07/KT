import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] c = new long[n];
        long[] t = new long[n];
        for (int i = 0; i < n; i++) c[i] = input.nextLong();
        for (int i = 0; i < n; i++) t[i] = input.nextLong();
        boolean valid = true;
        for (int i = 1; i < n - 1; i++) {
            if (2 * c[i] != c[i + 1] + c[i - 1] + t[i]) {
                valid = false;
                break;
            }
        }
        if (n >= 2 && c[0] + c[1] != t[0] + t[1]) valid = false;
        if (n >= 2 && c[n - 1] + c[n - 2] != t[n - 1] + t[n - 2]) valid = false;
        System.out.println(valid ? "Yes" : "No");
    }
}
