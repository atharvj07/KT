#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  int n, m;
  cin >> n >> m;
  vector<vector<int>> g(n);
  vector<bool> used(n);
  set<pair<int, int>> ans;
  set<int> leaf;
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    g[a].push_back(b);
    g[b].push_back(a);
  }
  auto dfs = [&](auto self, int cur, int prv) -> void {
    if (used[cur]) return;
    leaf.insert(cur);
    used[cur] = true;
    for (auto x : g[cur]) {
      if (x == prv) continue;
      self(self, x, cur);
      if (leaf.count(x) && leaf.count(cur)) {
        ans.insert({min(cur, x), max(x, cur)});
        leaf.erase(x);
        leaf.erase(cur);
      }
    }
  };
  dfs(dfs, 0, -1);
  for (auto x : ans) cout << x.first + 1 << ' ' << x.second + 1 << '\n';
  return 0;
}
