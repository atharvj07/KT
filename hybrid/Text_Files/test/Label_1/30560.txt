import java.io.*;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.StringTokenizer;

class Main {
    public static void main(String[] args) throws IOException {
        InputReader ir = new InputReader();
        int n = ir.nextInt();
        int k = ir.nextInt();
        int[] col = new int[n];
        Arrays.fill(col, 0);

        HashMap<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            int c = ir.nextInt();
            col[i] = c;
            if (map.containsKey(c)) {
                List<Integer> temp = map.get(c);
                temp.add(i);
                map.put(c, temp);
            } else {
                List<Integer> temp = new ArrayList<>();
                temp.add(i);
                map.put(c, temp);
            }
        }

        List<Integer>[] adj = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = ir.nextInt() - 1;
            int v = ir.nextInt() - 1;
            adj[u].add(v);
            adj[v].add(u);
        }

        int ans = countNiceCut(0, col, map, adj, -1);
        
        ans %= 998244353;
        System.out.println(ans);
    }

    static int countNiceCut(int u, int[] col, HashMap<Integer, List<Integer>> map, List<Integer>[] adj, int par) {
        int count = 0;
        for (int v : adj[u]) {
            if (v != par) {
                int c = col[v];
                map.get(c).remove((Integer) v);
                if (map.get(c).size() == 1) {
                    count += countNiceCut(v, col, map, adj, u);
                    count %= 998244353;
                }
                map.get(c).add(v);
            }
        }
        count++;
        return count;
    }

    static class InputReader {
        BufferedReader br;
        StringTokenizer st;

        public InputReader() {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        public String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) {
                st = new StringTokenizer(br.readLine());
            }
            return st.nextToken();
        }

        public int nextInt() throws IOException {
            return Integer.parseInt(next());
        }

        public long nextLong() throws IOException {
            return Long.parseLong(next());
        }
    }
}
