#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  ll n, m, x, y, sum1 = 0, sum2 = 0;
  cin >> n;
  for (ll i = 0; i < n; i++) {
    cin >> x;
    sum1 += x;
  }
  cin >> m;
  for (ll i = 0; i < m; i++) {
    cin >> y;
    sum2 += y;
  }
  if (sum1 != sum2) {
    cout << "-1";
    return 0;
  }
  cout << n;
  return 0;
}
