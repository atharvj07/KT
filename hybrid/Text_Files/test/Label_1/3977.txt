#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long
#define mod 1000000007
using namespace std;
ll fact[1000001];
ll binpow(ll a, ll b){
    if(b==0)return 1;
    if(b%2)return (binpow(a%mod,b%mod)*a)%mod;
    return binpow(a%mod,b%mod)%mod;
}
int main() {
    ll n;
    cin >> n;
    vector < ll > a(n);
    for (ll i = 0; i < n; i++)cin >> a[i];
    fact[1] = 1;
    for (ll i = 2; i <= n; i++){
        fact[i] = (fact[i - 1] % mod * i % mod) % mod;
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++){
        ll cur = 0, cnt = 0;
        for (ll j = i+1; j < n; j++){
            if (a[j] <= a[i])continue;
            cur += a[j];
            cnt++;
        }
        ans += (cur % mod * fact[cnt] % mod) % mod;
        ans %= mod;
    }
    cout << ans << endl;
    return 0;
}
