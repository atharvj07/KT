def solve():
    s = input()
    L, R, U, D, cnt = 0, 0, 0, 0, 0
    for c in s:
        if c == 'L':
            L += 1
        elif c == 'R':
            R += 1
        elif c == 'U':
            U += 1
        elif c == 'D':
            D += 1
    if L == R and U == D:
        return 0
    elif L + R == U + D:
        return abs(L - U)
    else:
        return -1
        

print(solve())
