import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] c = new int[2 * n];
        int[] d = new int[n + 1];
        int[] e = new int[n + 1];
        for (int i = 0; i < 2 * n - 1; i++) {
            if (s.charAt(i) == '1') {
                if (d[c[i]] == 0) {
                    e[c[i]] = i + 1;
                }
                d[c[i]]++;
            }
        }
        int j = 0;
        for (int i = 1; i <= n; i++) {
            if (d[i] != 2) {
                System.out.println("No");
                return;
            }
            c[j++] = i;
            c[j++] = i;
        }
        System.out.println("Yes");
        for (int i = 0; i < 2 * n; i++) {
            System.out.print(c[i] + " ");
        }
        System.out.println();
    }
}
