import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] time = new int[4];
        for (int i = 0; i < 4; i++) {
            time[i] = sc.nextInt();
        }
        String a = sc.nextLine();
        String b = sc.nextLine();
        int m = a.length();
        int n = b.length();
        int[][] dp = new int[m + 1][n + 1];
        for (int i = 0; i <= m; i++) {
            dp[i][0] = i * time[1];
        }
        for (int i = 0; i <= n; i++) {
            dp[0][i] = i * time[0];
        }
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                if (a.charAt(i - 1) == b.charAt(j - 1)) {
                    dp[i][j] = dp[i - 1][j - 1];
                } else {
                    dp[i][j] = Math.min(dp[i - 1][j] + time[1], dp[i][j - 1] + time[0]);
                    dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - 1] + time[2]);
                    if (i > 1 && j > 1 && a.charAt(i - 2) == b.charAt(j - 1) && a.charAt(i - 1) == b.charAt(j - 2)) {
                        dp[i][j] = Math.min(dp[i][j], dp[i - 2][j - 2] + time[3]);
                    }
                }
            }
        }
        System.out.println(dp[m][n]);
    }
}
