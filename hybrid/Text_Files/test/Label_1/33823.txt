import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        List<List<Integer>> switches = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int k = scanner.nextInt();
            List<Integer> switchList = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                switchList.add(scanner.nextInt());
            }
            switches.add(switchList);
        }
        List<Integer> states = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            states.add(scanner.nextInt());
        }
        int count = 0;
        for (int i = 0; i < (1 << n); i++) {
            int[] switchCount = new int[n];
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) != 0) {
                    for (int k = 0; k < switches.get(j).size(); k++) {
                        switchCount[switches.get(j).get(k) - 1]++;
                    }
                }
            }
            boolean isOn = true;
            for (int j = 0; j < m; j++) {
                if (switchCount[j] % 2 != states.get(j)) {
                    isOn = false;
                    break;
                }
            }
            if (isOn) {
                count++;
            }
        }
        System.out.println(count);
    }
}
