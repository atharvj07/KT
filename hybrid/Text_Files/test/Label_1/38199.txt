from collections import Counter
for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    cnt = Counter(a)
    c = [0] * n
    idx = 1
    for i in sorted(cnt.keys()):
        while idx <= k and cnt[i] > 0:
            for j in range(n):
                if a[j] == i and c[j] == 0:
                    c[j] = idx
                    cnt[i] -= 1
                    break
            idx += 1
    print(*c)
