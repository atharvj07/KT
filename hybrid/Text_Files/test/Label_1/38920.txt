s=input()
r=[-1]
i=0
for j in s:
    r.append((-r[i]-1)**int(j))
    i+=1
if -1 in r[1:]:print(-1)
else:
    for i in r[1:]:
        print(r.index(i)-1,end=' ')
        r[r.index(i):]=[i]*len(r[r.index(i):])
    print()
    for i in r[1:]:
        print(i,end=' ')
        print(*[j+1 for j in range(len(r)) if r[j]==i])
