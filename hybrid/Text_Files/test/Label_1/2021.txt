#include <bits/stdc++.h>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int m, n;
  cin >> m >> n;
  vector<vector<int>> t(m, vector<int>(n));
  for (auto &v : t)
    for (auto &x : v)
      cin >> x;

  vector<int> ans(m);
  for (int i = 0; i < m; ++i) {
    for (int j = 0, sum = 0; j < n; ++j) {
      sum += t[i][j];
      ans[i] = max(ans[i], sum);
    }
  }

  for (auto &x : ans)
    cout << x << " ";
  cout << "\n";
  return 0;
}
