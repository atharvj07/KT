def is_adorable(s):
    letter_counts = {}
    for letter in s:
        if letter in letter_counts:
            letter_counts[letter] += 1
        else:
            letter_counts[letter] = 1
    distinct_letter_counts = len(letter_counts)

    if distinct_letter_counts > 4:
        return "No"
    elif distinct_letter_counts == 4 or distinct_letter_counts == 3:
        return "Yes"
    elif distinct_letter_counts == 2:
        for count in letter_counts.values():
            if count == 1:
                return "No"
        return "Yes"
    elif distinct_letter_counts == 1 and len(s) > 1:
        return "No"
    return "Yes"

s = input().strip()
print(is_adorable(s))
