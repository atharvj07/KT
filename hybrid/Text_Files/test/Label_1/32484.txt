import java.util.Scanner;

public class GoodSubsegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;
            System.out.println(countGoodSubsegments(p, l, r));
        }
    }

    private static int countGoodSubsegments(int[] p, int l, int r) {
        int[] min = new int[r - l + 1];
        int[] max = new int[r - l + 1];
        min[0] = max[0] = p[l];
        for (int i = 1; i <= r - l; i++) {
            min[i] = Math.min(min[i - 1], p[l + i]);
            max[i] = Math.max(max[i - 1], p[l + i]);
        }
        int count = 0;
        for (int i = 0; i <= r - l; i++) {
            for (int j = i; j <= r - l; j++) {
                if (min[j - i] == p[l + i] && max[j - i] == p[l + j]) {
                    count++;
                }
            }
        }
        return count;
    }
}
