import java.util.*;

public class Main {
    static long[] solve(int[] a) {
        int n = a.length;
        long ans = 0;
        int rem = 0;
        for (int i = n - 1; i >= 0; i--) {
            int cur = a[i];
            if (i > 0)
                cur -= rem;
            if (i > 0)
                rem += cur / 4;
            rem %= 100;
            int digit = cur % 10;
            if (i == 0) {
                if (digit == 4) {
                    ans = ans * 10 + 4;
                } else if (digit == 7) {
                    ans = ans * 10 + 7;
                } else {
                    ans = -1;
                    break;
                }
            } else {
                if (i == 1) {
                    if (digit == 4 && rem == 0) {
                        ans = ans * 100 + 47;
                    } else if (digit == 7 && rem == 0) {
                        ans = ans * 100 + 74;
                    } else {
                        ans = -1;
                        break;
                    }
                } else {
                    if (digit == 4 && rem == 0) {
                        ans = ans * 100 + 47;
                    } else if (digit == 7) {
                        if (rem == 0) {
                            ans = ans * 100 + 74;
                        } else if (rem == 47) {
                            ans = ans * 100 + 447;
                        } else {
                            ans = -1;
                            break;
                        }
                    } else {
                        ans = -1;
                        break;
                    }
                }
            }
        }
        return new long[] { ans };
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] a = new int[4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextInt();
        }
        long[] ans = solve(a);
        if (ans[0] == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ans[0]);
        }
    }
}
