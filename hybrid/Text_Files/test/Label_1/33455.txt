import java.util.*;

public class Main {
    static boolean[] vis;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        vis = new boolean[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++)
            adj[i] = new ArrayList<Integer>();
        while (m-- > 0) {
            int u = in.nextInt(), v = in.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        System.out.println(check(1, -1) ? "YES" : "NO");
    }

    static boolean check(int node, int par) {
        vis[node] = true;
        for (int child : adj[node]) {
            if (!vis[child]) {
                if (!check(child, node))
                    return false;
            } else if (child != par)
                return false;
        }
        return true;
    }
}
