import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        boolean[][] graph = new boolean[n + 1][n + 1];
        boolean[] visited = new boolean[n + 1];
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == '1') {
                graph[(i / 2) + 1][(i / 2) % 2 + 1] = true;
                graph[(i / 2) % 2 + 1][(i / 2) + 1] = true;
            }
        }
        int[] ans = new int[2 * n];
        boolean ok = true;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                int cnt = 0;
                for (int j = 1; j <= n; j++) {
                    if (graph[i][j]) {
                        cnt++;
                        if (visited[j]) {
                            ok = false;
                            break;
                        }
                    }
                }
                if (cnt % 2 == 1) {
                    ok = false;
                    break;
                } else if (cnt == 0) {
                    ans[s.length()] = i;
                }
            }
            visited[i] = true;
        }
        if (ok) {
            System.out.println("Yes");
            for (int i = 0; i < 2 * n; i++) {
                if (ans[i] == 0) {
                    for (int j = 0; j <= n; j++) {
                        if (graph[j][i + 1] && !visited[j]) {
                            ans[i] = j;
                            visited[j] = true;
                            break;
                        }
                    }
                }
            }
            for (int i : ans) {
                System.out.print(i + " ");
            }
            System.out.println();
        } else {
            System.out.println("No");
        }
    }
}
