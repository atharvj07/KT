import sys

n, s = map(int, sys.stdin.readline().split())

times = []
for _ in range(n):
    h, m = map(int, sys.stdin.readline().split())
    times.append((h, m))

times.sort()

ans_h = 0
ans_m = 0

for h, m in times:
    if ans_m + 1 + s > m:
        ans_m = m + 1
    else:
        ans_h, ans_m = h, m + 1
        break

if ans_m == 60:
    ans_h += 1
    ans_m = 0

print(ans_h, ans_m)
