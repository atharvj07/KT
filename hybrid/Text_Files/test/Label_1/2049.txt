#include<cstdio>
#include<cmath>
using namespace std;
const double eps=1e-4;
int main(){
	int n,i,j;
	double w[188][188],a[188][188],b[188][188];
	scanf("%d",&n);
	for(i=0;i<n;i++){
		for(j=0;j<n;j++){
			scanf("%lf",&w[i][j]);
			a[i][j]=w[i][j];
			b[i][j]=0;
		}
	}
	for(i=0;i<n;i++){
		for(j=i+1;j<n;j++){
			double delta=w[i][j]-w[j][i];
			if(fabs(delta)>eps){
				if(fabs(a[i][i])>eps)a[i][j]=a[j][i]=(w[i][j]-w[j][i]+a[i][i])/2.0;
				else a[j][i]=a[i][j]=(w[j][i]-w[i][j]+a[j][j])/2.0;
				if(fabs(a[j][j])>eps)b[i][j]=b[j][i]=(w[i][j]-w[j][i]+a[j][j])/2.0;
				else b[i][j]=b[j][i]=(w[j][i]-w[i][j]+a[i][i])/2.0;
			}
		}
	}
	for(i=0;i<n;i++){
		for(j=0;j<n;j++)printf("%.8f ",a[i][j]);
		printf("\n");
	}
	for(i=0;i<n;i++){
		for(j=0;j<n;j++)printf("%.8f ",b[i][j]);
		printf("\n");
	}
}
