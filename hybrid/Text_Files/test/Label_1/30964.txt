import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner input = new Scanner(System.in);
		int a = input.nextInt(), b = input.nextInt(), s = input.nextInt();
		if ((s - Math.abs(a) - Math.abs(b)) % 2 == 0 && s >= Math.abs(a) + Math.abs(b)) {
			System.out.println("Yes");
		} else {
			System.out.println("No");
		}
		input.close();
	}
}
