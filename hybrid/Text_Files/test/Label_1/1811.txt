#include <cstdio>
#include <algorithm>
using namespace std;
int main() {
  int n, m, k, *arr, prefix[5003], ans;
  scanf("%d %d %d", &n, &m, &k);
  arr = new int[n + 3];
  for (int i = 1; i <= n; i++) {
    scanf("%d", &arr[i]);
    prefix[i] = prefix[i - 1] + arr[i];
  }
  sort(arr + 1, arr + n + 1);
  for (int i = m; i <= n; i++) {
    ans = max(ans, prefix[i] - prefix[i - m]);
  }
  for (int i = 1; i <= k; i++) {
    ans += arr[n - m * i];
  }
  printf("%d", ans);
  return 0;
}
