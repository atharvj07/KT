#include <iostream>
#include <vector>

using namespace std;

bool is_prime(long long int n) {
  if (n <= 1) return false;
  for (long long int i = 2; i * i <= n; ++i) {
    if (n % i == 0) return false;
  }
  return true;
}

int main() {
  long long int n;
  cin >> n;
  vector<long long int> primes;
  for (long long int i = 3; i * i <= n; ++i) {
    if (is_prime(i)) {
      primes.push_back(i);
    }
  }
  if (n <= 5) {
    cout << 1 << endl;
    cout << n << endl;
  } else if (n <= 11) {
    cout << 2 << endl;
    cout << 3 << ' ' << (n - 3) << endl;
  } else {
    long long int sum = 0;
    for (long long int i = 0; i < primes.size(); ++i) {
      if (sum + primes[i] <= n) {
        sum += primes[i];
        cout << primes[i] << ' ';
      } else if (sum + primes[i] > n) {
        cout << endl;
        cout << n - sum << endl;
        break;
      }
    }
  }
  return 0;
}
