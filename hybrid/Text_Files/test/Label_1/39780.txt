n, m = tuple(int(x) for x in input().split())
plants = {int(s): float(x) for s, x in ((int(s), float(x)) for s, x in (inp.split() for inp in (input() for _ in range(n))))}
previous_x = [0] * (m + 1)
x = [0] * (n + 2)
for i in sorted(plants):
    x[i] = plants[i] - previous_x[i - 1]
    previous_x[i] = plants[i]
x[n + 1] = 1
need_to_replant = 0
for i in range(1, n + 2):
    if x[i] < x[i - 1]:
        need_to_replant += 1
print(need_to_replant)
