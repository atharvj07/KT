n = int(input())
a = list(map(int, input().split()))
dp = [0] * n
for i in range(n):
    for j in range(i):
        if a[i] & a[j] == a[j]:
            dp[i] = max(dp[i], dp[j] + 1)
k = max(dp)
ans = []
for i in range(n - 1, -1, -1):
    if dp[i] == k:
        ans.append(a[i])
        k -= 1
print(len(ans))
print(*ans)
