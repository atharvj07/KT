#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        vector<int> p(k + 1);
        for (int i = 0; i <= k; i++) p[i] = i;
        int l = 1, r = k, inv = 0;
        for (int i = 0, j = 0; i < n; i++) {
            while (j != n && j < i + k && a[j] != l) j++;
            while (j != n && j < i + k && a[j] != r) j++;
            if (j != n && j < i + k) {
                if (a[j] == l) {
                    inv += i - (j + 1);
                    p[a[j]] = r;
                    r--;
                } else {
                    p[a[j]] = l;
                    l++;
                }
                j++;
            }
        }
        for (int i = 1; i <= k; i++) cout << p[i] << ' ';
        cout << '\n';
    }
}
