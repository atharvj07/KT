import sys
input=sys.stdin.readline
mod=998244353
for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=[x>1 for x in a]
    a=[x%2 for x in a]
    odd=sum(a)
    if odd%2==1:
        print(0)
    else:
        ans=1
        for x in b:
            if x:
                ans*=2
                ans%=mod
        print(ans)
