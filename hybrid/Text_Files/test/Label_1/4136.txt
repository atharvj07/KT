#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) cin >> a[i];

    long long sum = 0;
    for (long long i = 0; i < n; i++) sum += a[i];

    sort(a.begin(), a.end());
    long long res = a[n - 1] - a[0];
    vector<pair<long long, long long>> ans;
    while (res > 0 && k) {
        if (a[0] < a[n - 1]) {
            a[0]++;
            a[n - 1]--;
            sum--;
            res--;
            ans.push_back({n, 1});
        }
        k--;
    }
    cout << res << ' ' << ans.size() << endl;
    for (auto &i : ans) cout << i.first << ' ' << i.second << endl;
}
