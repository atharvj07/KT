import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long s = sc.nextLong();
            long ans = 0;
            while (s >= 10) {
                ans += s - (s % 10);
                s = (s / 10) + (s % 10);
            }
            ans += s;
            System.out.println(ans);
        }
    }
}
