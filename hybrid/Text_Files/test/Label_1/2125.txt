#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> beacons(n);
  for (int i = 0; i < n; ++i) {
    cin >> beacons[i].first >> beacons[i].second;
  }
  sort(beacons.begin(), beacons.end());
  int ans = n;
  for (int i = 0, j; i < n; ++i) {
    if (i != 0) {
      j = i;
      while (j < n && beacons[j].second <= beacons[i].second) {
        ++j;
      }
      ans = min(ans, n - j);
    }
  }
  cout << ans << '\n';
  return 0;
}
