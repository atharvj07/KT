from collections import defaultdict
from heapq import *

n, m = map(int, input().split())
edges = defaultdict(list)
for _ in range(m):
    a, b, w = map(int, input().split())
    edges[a].append((b, w))

dist = [0] * (n + 1)
prv = [0] * (n + 1)
vis = [False] * (n + 1)
q = [(1, 1)]
ans = 0

while len(q) > 0:
    x, l = heappop(q)
    if vis[x]:
        continue
    vis[x] = True
    ans = max(ans, l)
    for u, w in edges[x]:
        if w > dist[x]:
            dist[u] = w
            prv[u] = x
            heappush(q, (u, l + 1))

print(ans)
