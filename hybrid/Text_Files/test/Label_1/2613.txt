#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

const int MAXN=3e5+10;
LL a[MAXN],b[MAXN];
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int n,m;
    cin>>n;
    LL sum1=0;
    for(int i=0; i<n; i++){
        cin>>a[i];
        sum1+=a[i];
    }
    cin>>m;
    LL sum2=0;
    for(int i=0; i<m; i++){
        cin>>b[i];
        sum2+=b[i];
    }
    if(sum1!=sum2){
        cout<<"-1\n";
        return 0;
    }
    vector<LL> A,B;
    LL cur=a[0];
    LL cnt=1;
    for(int i=1; i<n; i++){
        if(cur!=a[i]) {A.push_back(cur);A.push_back(cnt); cur=a[i]; cnt=0;}
        cnt++;
    }
    A.push_back(cur);A.push_back(cnt);

    cur=b[0];
    cnt=1;
    for(int i=1; i<m; i++){
        if(cur!=b[i]) {B.push_back(cur);B.push_back(cnt); cur=b[i]; cnt=0;}
        cnt++;
    }
    B.push_back(cur);B.push_back(cnt);
    if(A==B){
        cout<<A.size()/2<<endl;
    }
    else cout<<"-1\n";
    return 0;
}
