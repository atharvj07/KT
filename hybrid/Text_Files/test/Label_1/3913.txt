#include <bits/stdc++.h>
using namespace std;
#define mp make_pair
#define ll long long
#define rep(i, l, r) for (int i = (l); i <= (r); ++i)
#define lrep(i, l, r) for (ll i = (l); i <= (r); ++i)
typedef vector<int> vi;
typedef pair<int, int> pi;
typedef vector<pi> vpi;
#define pb push_back
#define F first
#define S second
#define foreach(i, a) for (auto&(i) : (a))
const int mod = 998244353;
int pw(int x, int p) {
    if (p == 0) return 1;
    if (p == 1) return x;
    int t = pw(x, p / 2);
    return 1LL * t * t % mod * (p % 2 ? x : 1) % mod;
}
int fac[100005], ifac[100005];
void init() {
    fac[0] = 1;
    rep(i, 1, 100000) fac[i] = 1LL * i * fac[i - 1] % mod;
    ifac[100000] = pw(fac[100000], mod - 2);
    for (int i = 99999; i >= 0; --i) ifac[i] = 1LL * ifac[i + 1] * (i + 1) % mod;
}
int C(int n, int m) { return 1LL * fac[n] * ifac[m] % mod * ifac[n - m] % mod; }
int lin[100005], lout[100005], rin[100005], rout[100005], a[100005], b[100005];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    init();
    int n;
    cin >> n;
    rep(i, 1, n) {
        char l, r;
        cin >> l >> r;
        if (l == '?')
            lin[i] = 1;
        else if (l == 'B')
            a[i] = 1;
        if (r == '?')
            lout[i] = 1;
        else if (r == 'B')
            b[i] = 1;
    }
    int totl = 0, totr = 0, free = 0;
    rep(i, 1, n) {
        totl += lin[i];
        totr += lout[i];
        free += (lin[i] + lout[i]) == 0;
    }
    int ans = 1LL * pw(2, free) % mod * C(totl + totr + free, totl + free) % mod;
    vector<int> cycles;
    int cystart = -1, cyend = -1;
    rep(i, 1, n) {
        if (!lin[i] && !lout[i]) continue;
        if (cystart == -1) {
            cystart = i;
            cyend = i;
        } else {
            cyend = i;
            cycles.pb(cyend - cystart + 1);
            cystart = -1;
        }
    }
    if (cystart != -1) cycles.pb(cyend - cystart + 1);
    foreach(x, cycles) {
        if (x % 2 == 0)
            ans = 1LL * ans * 2 % mod;
        else
            ans = 1LL * ans * (mod + 1) % mod;
    }
    cout << ans << '\n';
}
