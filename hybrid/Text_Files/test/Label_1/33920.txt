import java.util.*;
import java.io.*;
class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] weights = new int[n];
    for (int i = 0; i < n; i++) {
      weights[i] = in.nextInt();
    }
    Arrays.sort(weights);
    int minSteps = 0;
    while (n > 0) {
      int weight = weights[n - 1];
      while (weight >= 1) {
        weight /= 2;
        n--;
        if (n == 0) {
          break;
        }
      }
      minSteps++;
    }
    System.out.println(minSteps);
  }
}
