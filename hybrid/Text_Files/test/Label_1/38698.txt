import sys

def solve():
    n = int(input())
    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)

    vis = [False] * (n + 1)
    par = [-1] * (n + 1)
    stack = [1]
    vis[1] = True
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if not vis[v]:
                vis[v] = True
                par[v] = u
                stack.append(v)
    
    deg = [0] * (n + 1)
    for u in range(1, n + 1):
        for v in adj[u]:
            if par[u] != v:
                deg[u] += 1
    
    cnt = 0
    for u in range(1, n + 1):
        if deg[u] == 1:
            cnt += 1
    
    if cnt % 2 == 0:
        print("YES")
    else:
        print("NO")

solve()
