import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) break;
            Set<Integer> set = new HashSet<>();
            for (int i = 0; i < m; i++) {
                int p = sc.nextInt();
                for (int j = p; j <= n; j += p) {
                    set.add(j);
                }
            }
            double ans = 1.0 * n / (2 * n - 1);
            if (set.isEmpty()) {
                System.out.printf("%.10f\n", ans);
            } else {
                System.out.println(0.0);
            }
        }
    }
}
