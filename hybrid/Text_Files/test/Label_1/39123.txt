t = int(input())
for _ in range(t):
    n = int(input())
    s = input()
    subsequences = []
    for char in s:
        placed = False
        for subseq in subsequences:
            if subseq[-1] != char:  # Can place char here
                subseq.append(char)
                placed = True
                break
        if not placed:
            subsequences.append([char])
    print(len(subsequences))
    answer = [0] * n
    for i, subseq in enumerate(subsequences, 1):
        pos = 0
        for char in subseq:
            while s[pos] != char or answer[pos] != 0:
                pos += 1
            answer[pos] = i
            pos += 1
    print(*answer)
