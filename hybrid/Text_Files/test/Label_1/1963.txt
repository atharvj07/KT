#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        a[i] += a[i - 1];
    }
    if (a[n] == 0) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    int l = 1, r = 1;
    while (r <= n) {
        while (r <= n && a[r] - a[l - 1] == 0) {
            r++;
        }
        if (r <= n) {
            cout << l << " " << r << endl;
            l = r + 1;
            r = l;
        }
    }
    return 0;
}
