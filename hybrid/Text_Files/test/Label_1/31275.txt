import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] arr = new int[10];
        for (int i = 0; i < 10; i++) {
            arr[i] = (i % 3 + i / 3) % 3;
        }
        for (int i = 1; i < n; i++) {
            if (arr[s.charAt(i) - '0'] != arr[s.charAt(i - 1) - '0']) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
