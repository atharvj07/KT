#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x, a, sum = 0, odd = 0;
    cin >> n >> x;
    for (int i = 0; i < n; i++) {
      cin >> a;
      sum += a;
      if (a & 1) odd++;
    }
    if (sum & 1) {
      cout << "Yes" << endl;
      continue;
    }
    if (odd == 0)
      cout << "No" << endl;
    else if (x & 1)
      cout << "Yes" << endl;
    else if ((odd >= 1) && (odd <= x))
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
}
