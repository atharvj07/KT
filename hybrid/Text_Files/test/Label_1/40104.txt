import sys
input = sys.stdin.readline

n, m = map(int, input().split())
st = [[] for _ in range(n+3)]
for _ in range(m):
    a, b = map(int, input().split())
    st[a].append(b)

vis=[0]*(n+3)
ans=0
now=1
flag=0
while True:
    ans+=1
    tmp = []
    for i in range(1,n+1):
        if vis[i] != ans: continue
        for j in st[i]:
            if vis[j] == ans:
                flag=1
                break
            tmp.append(j)
        vis[i]=ans+1
    if flag:
        ans=-1
        break
    if not tmp: break
    st=tmp
print(ans)
