#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    int a[n],b[m];
    for(int i=0;i<n;i++) cin>>a[i];
    for(int i=0;i<m;i++) cin>>b[i];
    set<int>t(b,b+m);
    int c=n;
    for(int i=0;i<n;i++){
        if(t.find(a[i])==t.end()) c--;
    }
    cout<<c<<endl;
    return 0;
}
