import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int k = in.nextInt();
        String s = in.next();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'Y') {
                a[i] = 1;
            } else if (s.charAt(i) == 'N') {
                a[i] = -1;
            }
        }

        int max = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                cur++;
            } else {
                max = Math.max(max, cur);
                cur = 0;
            }
        }
        max = Math.max(max, cur);

        if (max == k) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
