from sys import stdin
input = stdin.readline

n, x = map(int, input().split())
a = sorted([list(map(int, input().split())) for i in range(n)], key=lambda x: x[1])
y = [0] * n
y[0] = 1 if a[0][0] == 1 else 0
for i in range(1, n):
    if a[i][0] == a[i - 1][0]: continue
    y[i] = y[i - 1] + 1
ans = 1
cur = 0
for i in range(n):
    if x < a[i][1]: continue
    if y[i] != cur:
        ans += 1
        cur = y[i]
    x += a[i][2]
print(ans)
