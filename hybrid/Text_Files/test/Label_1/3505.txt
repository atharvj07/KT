#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxN = 1e5 + 10;
int n, a[maxN];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    int gcd = a[1];
    for (int i = 2; i <= n; i++) gcd = __gcd(gcd, a[i]);
    cout << gcd << endl;
    return 0;
}
