import java.util.Arrays;
import java.util.Scanner;

class Main {
    private static class Task {
        int time, ki, l;

        Task(int time, int ki, int l) {
            this.time = time;
            this.ki = ki;
            this.l = l;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = i + 1;
        }
        int q = sc.nextInt();
        Task[] tasks = new Task[q];
        for (int i = 0; i < q; i++) {
            tasks[i] = new Task(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        Arrays.sort(tasks, (x, y) -> Integer.compare(x.time, y.time));
        for (Task task : tasks) {
            int[] b = a.clone();
            int sum = 0;
            boolean ok = true;
            for (int i = 0; i < task.ki && i < n; i++) {
                if (b[i] == 0) {
                    ok = false;
                    break;
                }
                sum += b[i];
                b[i] = 0;
            }
            if (ok) {
                System.out.println(sum);
                for (int i = 1; i <= task.l; i++) {
                    for (int j = 0; j < n; j++) {
                        if (b[j] == 0) {
                            b[j] = a[j];
                        } else {
                            b[j]--;
                        }
                    }
                }
                a = b.clone();
            } else {
                System.out.println(-1);
            }
        }
    }
}
