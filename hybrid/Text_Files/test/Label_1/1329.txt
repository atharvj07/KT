#include <iostream>
#include <string>
using namespace std;

int main() {
  string x;
  cin >> x;
  for (int i = 0; i < x.size(); i++) {
    if (x[i] != '9') {
      x[i] = '9' - (x[i] - '0');
    }
  }
  while (x.size() > 1 && x[0] == '0') {
    x.erase(0, 1);
  }
  cout << x << endl;
  return 0;
}
