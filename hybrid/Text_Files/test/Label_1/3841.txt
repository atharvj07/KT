#include<bits/stdc++.h>
using namespace std;
#define pb push_back

int main() {
	int n, ans = 0; cin >> n;
	map<string, int> a, b;
	for (int i = 0; i < n; i++) {
		string x; cin >> x;
		a[x]++;
	}
	for (int i = 0; i < n; i++) {
		string x; cin >> x;
		b[x]++;
	}
	for (auto [x, y] : a) {
		ans += abs(y - b[x]);
		b[x] = 0;
	}
	cout << ans / 2 << endl;
}
