n, a, b = map(int, input().split())
t = list(map(int, input().split()))
deny = 0
for client in t:
    if client == 1:
        if a > 0:
            a -= 1
        elif b > 0:
            b -= 1
        elif 2 * b > 0:
            b -= 1
        else:
            deny += 1
    else:
        if b > 0:
            b -= 1
        else:
            deny += 2
print(deny)
