from collections import defaultdict
import sys

input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))

freq = defaultdict(int)
l, r = 0, 0
ans = n
for j in range(1, m + 1):
    cnt = 0
    for i in range(l, n):
        if freq[a[i]] == 0:
            r += 1
        freq[a[i]] += 1
        while r - l >= k[j - 1]:
            freq[a[l]] -= 1
            if freq[a[l]] == 0:
                cnt += 1
            l += 1
    if (l == n) or (r - l < k[j - 1]):  # impossible
        ans = -1
        break
    ans = min(ans, cnt)
print(ans)
