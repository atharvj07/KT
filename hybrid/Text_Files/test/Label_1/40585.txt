import math

def find_smallest_magical_box(n, boxes):
    max_k = 0
    for k_i, a_i in boxes:
        # Calculate the effective k needed to fit all a_i boxes of size 2^k_i
        effective_k = k_i + math.ceil(math.log2(a_i) / 2)
        max_k = max(max_k, effective_k)
    return max_k

n = int(input())
boxes = [list(map(int, input().split())) for _ in range(n)]
print(find_smallest_magical_box(n, boxes))
