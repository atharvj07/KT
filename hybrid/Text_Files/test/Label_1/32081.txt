import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++)
            a[i] = sc.nextInt();
        Arrays.sort(a, 1, n + 1);
        int sum = 0;
        int odd = 0;
        for (int i = 1; i <= n; i++) {
            sum += a[i];
            if (a[i] % 2 == 1)
                odd++;
        }
        if (odd == 0)
            System.out.println("cslnb");
        else if (odd == 1 && sum % 2 == 0)
            System.out.println("cslnb");
        else
            System.out.println("sjfnb");
    }
}
