from sys import stdin

n, m = map(int, stdin.readline().split())
a = [stdin.readline() for i in range(n)]
b = [[int(j) for j in stdin.readline().split()] for i in range(n)]
ans = 0
for i in range(m):
    L = []
    for j in range(n):
        L.append((a[j][i], b[j][i]))
    L.sort()
    for j in range(n - 1):
        if L[j][0] == L[j+1][0]:
            ans += min(L[j][1], L[j+1][1])
print(ans)
