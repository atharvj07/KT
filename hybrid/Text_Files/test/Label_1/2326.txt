#include <iostream>
#include <map>
#include <vector>
using namespace std;

map<pair<int, int>, char> mp;
int32_t main() {
    int n;
    cin >> n;
    int x, y;
    cin >> x >> y;
    for (int i = 0; i < n; i++) {
        char c;
        int x1, y1;
        cin >> c >> x1 >> y1;
        if (x1 - x == y1 - y || x1 - x == y - y1 || x1 == x || y1 == y) {
            if (mp.count({x, y}) == 0 || mp[{x, y}] == 'R')
                mp[{x, y}] = c;
        }
    }
    cout << (mp.count({x, y}) && mp[{x, y}] != 'R' ? "YES" : "NO");
    return 0;
}
