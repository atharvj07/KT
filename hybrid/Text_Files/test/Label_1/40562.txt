from collections import deque

def bfs(u, p):
    q = deque([(u, p)])
    while q:
        u, p = q.popleft()
        for v in edges[u]:
            if v != p:
                yield (u, v)
                q.append((v, u))

n = int(input())
edges = [[] for i in range(n+1)]
for i in range(n-1):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)

cycle = []
for u in range(1, n+1):
    if len(edges[u]) == 1:
        if p is None:
            p = u
            continue
        else:
            for u, v in bfs(u, p):
                cycle.append((u, v))
            break

if len(cycle) % 2 == 1:
    print("NO")
else:
    print("YES")
