from collections import deque
n,m=map(int,input().split())
d=list(map(int,input().split()))
g={i:[] for i in range(1,n+1)}
for _ in range(m):
    u,v=map(int,input().split())
    g[u].append(v)
    g[v].append(u)
q=deque([(1,-1,0)])
vis=[0]*(n+1)
ans=[0]*m
vis[1]=1
cnt=0
f=0
while q:
    v,p,mex=q.popleft()
    for u in g[v]:
        if u!=p and not vis[u]:
            vis[u]=1
            if d[u-1]>0:
                if mex==0:
                    mex=1
                else:
                    f=1
                    break
            else:
                if mex==1:
                    mex=0
                else:
                    f=1
                    break
            q.append((u,v,mex))
    if f:
        break
    if mex==0:
        cnt+=1
        ans[v]=cnt
if f or d[0]==1:
    print(-1)
else:
    print(cnt)
    for i in ans:
        if i>0:
            print(i)
