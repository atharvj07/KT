import java.util.*;
import java.lang.*;

class Main {
    static int n, m;
    static int[] a, b;
    static int[][] dp;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        n = input.nextInt();
        a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = input.nextInt();
        }
        m = input.nextInt();
        b = new int[m];
        for (int j = 0; j < m; ++j) {
            b[j] = input.nextInt();
        }
        dp = new int[n + 1][m + 1];
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < m; ++j) {
                if (a[i] == b[j]) {
                    dp[i + 1][j + 1] = dp[i][j] + 1;
                } else {
                    dp[i + 1][j + 1] = Math.max(dp[i + 1][j], dp[i][j + 1]);
                }
            }
        }
        int k = dp[n][m];
        int[] c = new int[k + 1];
        int ia = n, ja = m;
        while (ia > 0 && ja > 0) {
            if (a[ia - 1] == b[ja - 1]) {
                c[k--] = a[ia - 1];
                --ia;
                --ja;
            } else if (dp[ia - 1][ja] > dp[ia][ja - 1]) {
                --ia;
            } else {
                --ja;
            }
        }
        System.out.println(k);
        for (int i = 1; i <= k; ++i) {
            System.out.print(c[i] + " ");
        }
    }
}
