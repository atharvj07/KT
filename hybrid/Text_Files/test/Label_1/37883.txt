n = int(input())
books = [tuple(map(int, input().split())) for _ in range(n)]

thickness_1 = sum(t for t, w in books if t == 1)
thickness_2 = sum(t for t, w in books if t == 2)

min_thickness = thickness_1 + thickness_2
widths = sorted([w for t, w in books if t == 1], reverse=True)

for i in range(len(widths) + 1):
    horizontal_width_sum = sum(widths[:i])
    if horizontal_width_sum <= thickness_2:
        min_thickness = min(min_thickness, thickness_2)
    else:
        needed_thickness_1 = horizontal_width_sum - thickness_2
        if needed_thickness_1 <= thickness_1:
            min_thickness = min(min_thickness, thickness_2 + needed_thickness_1)

print(min_thickness)
