import numpy as np

n = int(input())
grid = np.array([list(map(int, input().split())) for i in range(n)])
s = (1 + n) * n // 2
for i in range(n):
    for j in range(n):
        if grid[i][j] == 0:
            grid[i][j] = s - np.sum(grid[i]) - np.sum(grid[:, j])
            break
print(grid[i][j] if (grid[i][j] > 0 and grid[:, j].min() > 0 and grid[i].min() > 0) else -1)
