#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100005;
bool vis[MAXN];
int a[MAXN];
int main() {
  int n, m;
  cin >> n >> m;
  set<int> s;
  for (int i = 0; i < m; ++i) {
    int l, r;
    cin >> l >> r;
    for (int j = l; j <= r; ++j) { s.insert(a[j]); }
  }
  for (int i = 0; i < MAXN; ++i) {
    if (!s.count(i)) {
      cout << i << "\n";
      break;
    }
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j <= 1000000000; ++j) {
      if (!vis[j]) {
        a[i] = j;
        vis[j] = 1;
        break;
      }
    }
  }
  for (int i = 0; i < n; ++i) { cout << a[i] << " "; }
  cout << "\n";
  return 0;
}
