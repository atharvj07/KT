#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 1e5+5;

int main() {
    int T, n;
    int a[MAX_N];
    cin >> T;
    while (T--) {
        cin >> n;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a, a + n, greater<int>());
        int k = 0;
        for (int i = 0; i < n; i++) {
            if (k + 2 <= a[i]) {
                k++;
            }
        }
        cout << k << endl;
    }
    return 0;
}
