MOD = 998244353

def mod_pow(base, exponent, modulus):
    result = 1
    base = base % modulus
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus
        exponent = exponent >> 1
        base = (base * base) % modulus
    return result

def solve(n, m, L, R):
    total_cells = n * m
    total_numbers = R - L + 1
    if total_cells % 2 == 1:
        return mod_pow(total_numbers, total_cells, MOD)
    else:
        even_numbers = (total_numbers + 1) // 2
        odd_numbers = total_numbers // 2
        even_config = mod_pow(even_numbers + odd_numbers, total_cells, MOD)
        odd_config = mod_pow(even_numbers - odd_numbers, total_cells, MOD)
        return (even_config + odd_config) * pow(2, MOD - 2, MOD) % MOD

n, m, L, R = map(int, input().split())
print(solve(n, m, L, R))
