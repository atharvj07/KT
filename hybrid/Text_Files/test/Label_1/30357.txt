import java.util.*;
import java.lang.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        double[] r = new double[n];
        for (int i = 0; i < n; i++) {
            r[i] = in.nextInt();
        }
        double area = 0;
        for (int i = 0; i < n - 1; i++) {
            area += Math.PI * (r[i] + r[i + 1]) * (r[i] + r[i + 1]) - Math.PI * r[i + 1] * r[i + 1];
        }
        area += Math.PI * r[n - 1] * r[n - 1];
        System.out.println(area);
    }
}
