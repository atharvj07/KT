import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int l = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int maxArea = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] >= l) {
                int num = 0;
                for (int j = 0; j < n; j++) {
                    if (a[j] >= l) {
                        num++;
                    }
                }
                if (num == n) {
                    int width = n;
                    int height = a[i];
                    maxArea = Math.max(maxArea, width * height);
                }
            }
            for (int j = i - 1; j >= 0; j--) {
                int num = 0;
                for (int k = 0; k < n; k++) {
                    if (a[k] >= l) {
                        num++;
                    }
                }
                if (num == n) {
                    int width = a[i] + a[j];
                    int height = l;
                    maxArea = Math.max(maxArea, width * height);
                }
            }
            for (int j = i - 1; j >= 0; j--) {
                for (int k = j - 1; k >= 0; k--) {
                    int num = 0;
                    for (int p = 0; p < n; p++) {
                        if (a[p] >= l) {
                            num++;
                        }
                    }
                    if (num == n) {
                        int width = a[i] + a[j] + a[k];
                        int height = l;
                        maxArea = Math.max(maxArea, width * height);
                    }
                }
            }
        }
        System.out.println(maxArea);
    }
}
