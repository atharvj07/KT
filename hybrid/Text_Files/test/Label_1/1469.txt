#include <iostream>
using namespace std;
int main() {
    long long n, a[100000], b[100000];
    cin >> n;
    for (long long i = 0; i < n; i++) cin >> a[i];
    b[0] = (a[0] + a[1]) / 2;
    for (long long i = 1; i < n; i++) b[i] = a[i] - a[i - 1];
    for (long long i = 0; i < n; i++) cout << b[i] << " ";
    return 0;
}
