import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s = sc.nextLine();
    int n = s.length();
    int[][] dp = new int[n + 1][3]; // dp[i][j]: number of strings ending at position i with state j
    dp[0][0] = 1;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 3; j++) {
        if (s.charAt(i) == 'a') {
          // if current character is 'a'
          // can stay in the same state
          dp[i + 1][j] = (dp[i + 1][j] + dp[i][j]) % 1000000007;
          // or append 'a' to previous 'a'
          if (j > 0)
            dp[i + 1][j - 1] = (dp[i + 1][j - 1] + dp[i][j]) % 1000000007;
        } else {
          // if current character is 'b'
          // can stay in the same state
          dp[i + 1][j] = (dp[i + 1][j] + dp[i][j]) % 1000000007;
          // or append 'b' to previous 'b'
          if (j < 2)
            dp[i + 1][j + 1] = (dp[i + 1][j + 1] + dp[i][j]) % 1000000007;
        }
      }
    }
    System.out.println(dp[n][0]);
  }
}
