import java.util.*;

class Main {
  static int MOD = 1000000007;
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int[][] dp = new int[n + 1][m + 1];
    dp[n][0] = 1;
    for (int i = n - 1; i > 0; i--) {
      for (int j = 0; j < m; j++) {
        dp[i][j] =
            (dp[i + 1][j] + (dp[i + 1][j + 1] * j % MOD) % MOD) % MOD;
      }
    }
    System.out.println(dp[1][0]);
  }
}
