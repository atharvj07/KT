import java.io.*;
import java.util.*;

public class Main {
  static final long mod = 998244353;
  static char[] s;
  static int n;
  static long[] dp;

  public static void main(String[] args) {
    try (Scanner scanner = new Scanner(System.in)) {
      s = scanner.nextLine().toCharArray();
      n = s.length;
      dp = new long[n + 1];
      solve();
    }
  }

  public static void solve() {
    // dp[i] = sum of depths of all bracket sequences obtained by replacing all characters '?'
    // in s[1..i] by either '(' or ')'
    dp[0] = 0;
    for (int i = 1; i <= n; i++) {
      if (s[i - 1] == '?')
        dp[i] = (2 * dp[i - 1]) % mod;
      else if (s[i - 1] == '(')
        dp[i] = dp[i - 1] + 1;
      else
        dp[i] = dp[i - 1] - 1;
    }
    long answer = 0;
    for (int i = 0; i <= n; i++) {
      answer = (answer + dp[i]) % mod;
    }
    System.out.println(answer);
  }
}
