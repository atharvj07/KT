import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            int l = Integer.parseInt(line[2]);
            int r = Integer.parseInt(line[3]);
            edges.add(new Edge(a, b, l, r));
        }
        int maxLoyalty = 0;
        for (int i = 1; i <= 100000; i++) {
            if (isReachable(i, edges, n)) {
                maxLoyalty++;
            }
        }
        if (maxLoyalty == 0) {
            System.out.println("Nice work, Dima!");
        } else {
            System.out.println(maxLoyalty);
        }
    }

    public static boolean isReachable(int x, List<Edge> edges, int n) {
        int[] visited = new int[n + 1];
        return dfs(x, edges, n, visited, 1);
    }

    public static boolean dfs(int x, List<Edge> edges, int n, int[] visited, int node) {
        if (node == n) {
            return true;
        }
        visited[node] = 1;
        for (Edge edge : edges) {
            if (visited[edge.a] == 0 && edge.l <= x && x <= edge.r) {
                if (dfs(x, edges, n, visited, edge.a)) {
                    return true;
                }
            }
            if (visited[edge.b] == 0 && edge.l <= x && x <= edge.r) {
                if (dfs(x, edges, n, visited, edge.b)) {
                    return true;
                }
            }
        }
        return false;
    }

    public static class Edge {
        int a;
        int b;
        int l;
        int r;

        public Edge(int a, int b, int l, int r) {
            this.a = a;
            this.b = b;
            this.l = l;
            this.r = r;
        }
    }
}
