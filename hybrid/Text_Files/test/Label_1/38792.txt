import sys
input = sys.stdin.readline

n = int(input())
grid = [list(map(int, input().split())) for i in range(n)]
mp = {}
for i in range(n):
    for j in range(n):
        if grid[i][j]==0:
            r,c=i,j
        mp[grid[i][j]]=mp.get(grid[i][j],0)+1
if len(mp.keys())>=n*n:
    print(-1)
    exit()
for i in range(1,n*n+1):
    if i not in mp.keys():
        ans=i
        break
sum1 = 0
for j in range(n):
    sum1+=grid[r][j]
sum2 = 0
for i in range(n):
    sum2+=grid[i][c]
cnt=0
for i in range(n):
    if grid[i][i]!=0:
        sum1-=grid[i][i]
        cnt+=1
for i in range(n):
    if grid[i][n-i-1]!=0:
        sum2-=grid[i][n-i-1]
        cnt+=1
if cnt==0:
    print(ans)
elif cnt==1:
    if sum1==sum2:
        print(ans)
    else:
        print(-1)
else:
    print(-1)
