def mod_inverse(a, m):
    return pow(a, m - 2, m)

def solve(n, probabilities):
    M = 998244353
    numerator = 1
    denominator = 1
    total_prob = 1
    for p in probabilities:
        numerator = (numerator * 100) % M
        numerator = (numerator + total_prob * (100 - p)) % M
        denominator = (denominator * p) % M
        total_prob = (total_prob * p) % M

    answer = (numerator * mod_inverse(denominator, M)) % M
    return answer

n = int(input().strip())
probabilities = list(map(int, input().strip().split()))
print(solve(n, probabilities))
