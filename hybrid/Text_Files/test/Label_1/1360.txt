#include <iostream>
#include <algorithm>
typedef long long ll;
const ll mod = 1000000007;
inline ll mul(ll a, ll b) { return (a * b) % mod; }
inline ll add(ll a, ll b) { return (a + b) % mod; }
int main() {
    std::string s;
    std::cin >> s;
    int n = s.length();
    ll dp[n + 2][3][2][3]; // dp[i][bombs][rest][last]
    dp[n][0][0][0] = 1;
    for (int i = n - 1; i >= 0; --i) {
        for (int bombs = 0; bombs <= 2; ++bombs) {
            for (int rest = 0; rest <= 1; ++rest) {
                for (int last = 0; last <= 2; ++last) {
                    int cur = s[i] - '0';
                    if (cur == bombs + rest) {
                        dp[i][bombs][rest][last] = add(dp[i + 1][bombs][rest][last], dp[i + 1][bombs][rest][cur]);
                    } else if (cur == bombs + rest + 1) {
                        dp[i][bombs][rest][last] = add(dp[i + 1][bombs][rest][last], dp[i + 1][bombs][rest][cur]);
                        dp[i][bombs][rest][last] = add(dp[i][bombs][rest][last], dp[i + 1][bombs][1][cur]);
                        dp[i][bombs][rest][last] = add(dp[i][bombs][rest][last], dp[i + 1][bombs + 1][0][cur]);
                    } else if (cur == bombs + rest + 2) {
                        dp[i][bombs][rest][last] = add(dp[i + 1][bombs][rest][last], dp[i + 1][bombs][1][cur]);
                        dp[i][bombs][rest][last] = add(dp[i][bombs][rest][last], dp[i + 1][bombs + 1][0][cur]);
                        dp[i][bombs][rest][last] = add(dp[i][bombs][rest][last], dp[i + 1][bombs + 2][0][cur]);
                    }
                }
            }
        }
    }
    std::cout << dp[0][0][0][0] << std::endl;
    return 0;
}
