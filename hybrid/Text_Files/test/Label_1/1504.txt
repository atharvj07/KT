#include <bits/stdc++.h>
#define int long long
using namespace std;
vector<int> a;
int32_t main() {
  int n, m;
  cin >> n >> m;
  a.resize(n);
  for (int &c : a) cin >> c;
  vector<int> ans(n);
  while (m--) {
    int l, r;
    cin >> l >> r;
    reverse(a.begin() + l - 1, a.begin() + r);
    for (int i = l; i <= r; i++) ans[a[i]] += (i + (i == r * (r & 1))) & 1;
  }
  for (int i = 1; i <= n; i++) {
    if ((ans[i] - i) & 1) {
      cout << "odd\n";
    } else
      cout << "even\n";
  }
  return 0;
}
