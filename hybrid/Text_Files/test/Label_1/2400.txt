#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> b(n);
        for (int i = 0; i < n; i++) {
            cin >> b[i];
        }
        bool impossible = false;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) {
            if (!impossible) {
                a[i] = b[i] / (n - i);
                if (a[i] <= 0 || (i > 0 && a[i] != a[i - 1])) {
                    impossible = true;
                }
            }
        }
        if (impossible) {
            cout << "NO" << endl;
        } else {
            cout << "YES" << endl;
            for (int i = 0; i < n; i++) {
                cout << a[i] << " ";
            }
            cout << endl;
        }
    }
    return 0;
}
