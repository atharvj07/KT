#include<bits/stdc++.h>
#define ll long long
const ll mod = 1e9 + 7;
using namespace std;
ll fastpow(ll num, ll p) {
    if (!p) return 1ll;
    if (p == 1)return num;
    ll ans = fastpow(num, p / 2);
    ans = (1ll * ans * ans) % mod;
    if (p % 2) ans = (ans * num) % mod;
    return ans;
}
int main() {
    ll n, q, ans = 0;
    cin >> n >> q;
    string s; cin >> s;
    vector<ll> sum(n + 1, 0), pre(n + 1, 0);
    sum[0] = 1;
    for (int i = 1; i <= n; ++i) {
        ll x = (s[i - 1] - '0');
        sum[i] = sum[i - 1] + x;
        pre[i] = (pre[i - 1] + (x * (sum[i - 1])) % mod) % mod;
    }
    while (q--) {
        ll l, r;
        cin >> l >> r;
        ans = (pre[r] - pre[l - 1]) % mod;
        cout << ans << '\n';
    }
}
