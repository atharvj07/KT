import sys, math

def main():
    n = int(input())
    a = [int(x) for x in input().split()]

    # Check if the path is possible
    for i in range(1, n):
        if (a[i] - a[i - 1]) % 2 != 0 or a[i] - a[i - 1] > 1:
            print('NO')
            return

    # Find possible values of x and y
    x = max(a)
    y = int((a[-1] + 1) / 2)

    # Print the result
    print('YES')
    print(x, y)

main()
