#include<bits/stdc++.h>
#define ll long long
using namespace std;

int main(){
    ll n, m;
    cin >> n >> m;
    vector<ll> a(m);
    for(auto &i : a) cin >> i;
    sort(a.begin(), a.end());
    ll sum = 0;
    for(ll i = 1; i < m; i++) sum += abs(a[i] - a[i - 1]);
    ll ans = sum;
    for(ll i = 0; i < m - 1; i++){
        if(a[i] == a[i + 1]) continue;
        ans = min(ans, sum - abs(a[i + 1] - a[i]) + abs(a[i + 1] - a[i - 1]) + abs(a[i] - a[i + 2]));
    }
    cout << ans << endl;
    return 0;
}
