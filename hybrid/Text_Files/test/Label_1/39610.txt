import sys
input=sys.stdin.readline

n=int(input())
a=[]
for i in range(n):
    k=int(input())
    b=[]
    for j in range(k):
        x,y=[int(x) for x in input().split()]
        b.append((x,y))
    a.append(sorted(b))
ans=0
g=0
for k in a:
    l=len(k)
    if g:
        ans+=k[l-1][1]
        g-=1
        l-=1
    for i in range(l):
        if i%10==9:
            g+=1
        ans+=k[i][1]
print(ans)
