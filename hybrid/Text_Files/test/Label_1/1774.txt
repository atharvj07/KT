#include<bits/stdc++.h>

using namespace std;

#define ll long long int

const int mxN = 505;

int grid[mxN][mxN], dp[mxN][mxN], preSum[mxN][mxN], row[mxN], col[mxN];

int main(){
    int n, m, q;
    cin >> n >> m >> q;

    for(int i=1; i<=n; i++){
        for(int j=1; j<=m; j++){
            cin >> grid[i][j];
            preSum[i][j] = preSum[i][j-1] + grid[i][j];
        }
    }

    while(q--){
        int x, y;
        cin >> x >> y;
        if(grid[x][y]==0){
            grid[x][y] = 1;
            int in = preSum[x][y] - preSum[x][y-1];
            row[x] = max(row[x], in);
            dp[x][y] = in;
            if(y>1) dp[x][y] = max(dp[x][y], in + dp[x][y-1]);
            if(y<m) dp[x][y] = max(dp[x][y], in + dp[x][y+1]);
            col[y] = max(col[y], dp[x][y]);
        }
        else{
            grid[x][y] = 0;
            int in = preSum[x][y] - preSum[x][y-1];
            row[x] = max(row[x], in);
            dp[x][y] = in;
            if(y>1) dp[x][y] = max(dp[x][y], in + dp[x][y-1]);
            if(y<m) dp[x][y] = max(dp[x][y], in + dp[x][y+1]);
            col[y] = max(col[y], dp[x][y]);
        }

        int mxr = 0, mxc = 0;
        for(int i=1; i<=n; i++) mxr = max(mxr, row[i]);
        for(int i=1; i<=m; i++) mxc = max(mxc, col[i]);

        cout << max(mxr, mxc) << endl;
    }

    return 0;
}
