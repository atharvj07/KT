from collections import defaultdict, deque
n, m = map(int, input().split())
s = input()
g = defaultdict(list)
for i in range(m):
    a, b = map(lambda x: int(x) - 1, input().split())
    g[a].append(b)
ind = [0] * n
for i in range(n):
    for j in g[i]:
        ind[j] += 1
q = deque()
ans = 0
vis = [False] * n
for i in range(n):
    if ind[i] == 0:
        q.append(i)
        ans = max(ans, 1)
        vis[i] = True
while len(q) > 0:
    now = q.popleft()
    for i in g[now]:
        ind[i] -= 1
        if ind[i] == 0 and not vis[i]:
            q.append(i)
            ans = max(ans, 1 + (s[i] == s[now]))
            vis[i] = True
if ans > n:
    print(-1)
else:
    print(ans)
