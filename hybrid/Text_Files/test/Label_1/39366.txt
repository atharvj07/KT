t = int(input())
for _ in range(t):
    n = int(input())
    packages = []
    for _ in range(n):
        x, y = map(int, input().split())
        packages.append((x, y))
    packages.sort()

    path = ''
    impossible = False
    current_x, current_y = 0, 0
    for x, y in packages:
        if y < current_y:  # If the package is below the current position, it's impossible to reach.
            impossible = True
            break
        path += 'R' * (x - current_x) + 'U' * (y - current_y)  # Move right to x and then up to y.
        current_x, current_y = x, y

    if impossible:
        print("NO")
    else:
        print("YES")
        print(path)
