#include <bits/stdc++.h>
#define int long long
using namespace std;
const int inf = 1e18;
int solve(vector<int>& v, int n) {
  int l = 0, r = 0, ans = 0;
  unordered_map<int, int> mp;
  while (r < n) {
    mp[v[r]]++;
    while (mp[v[r]] > 1) mp[v[l++]]--;
    ans = max(ans, r - l + 1);
    r++;
  }
  return ans;
}
signed main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (int& vi : v) cin >> vi;
  cout << solve(v, n) << endl;
  return 0;
}
