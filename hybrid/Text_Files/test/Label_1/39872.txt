t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    letters = {}
    for _ in range(n):
        s = input().strip()
        for char in s:
            if char in letters:
                letters[char] += 1
            else:
                letters[char] = 1
    if all(value % n == 0 for value in letters.values()):
        print("YES")
    else:
        print("NO")
