import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long h = scanner.nextLong();
        long m = scanner.nextLong();
        long k = scanner.nextLong();
        long h1 = scanner.nextLong();
        long m1 = scanner.nextLong();
        long h2 = scanner.nextLong();
        long m2 = scanner.nextLong();
        long hDiff = h2 - h1;
        long mDiff = m2 - m1;
        long totalMinutes = hDiff * m + mDiff;
        long answer = 0;
        for (long i = h1 * m + m1 + 1; i <= h2 * m + m2; i++) {
            long newHours = ((i + m - 1) / m);
            if (newHours != i / m || (i % m + k - 1) / m != i % m / m) {
                answer++;
            }
        }
        System.out.println(answer);
    }
}
