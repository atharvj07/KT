MOD = 10**9 + 7


def count_ways(n, k, a):
    a.sort(reverse=True)

    dp = [[[-1] * 3 for _ in range(k + 1)] for _ in range(n + 1)]
    dp[n][k][0] = 1

    def get_dp(i, j, called):
        if dp[i][j][called] != -1:
            return dp[i][j][called]
        res = 0
        if i < n:
            res += get_dp(i + 1, j, called)
            res %= MOD
            if j > 0:
                if not called:
                    res += get_dp(i + 1, j - 1, True)
                    res %= MOD
                else:
                    if a[i] == a[i - 1]:
                        res += get_dp(i + 1, j - 1, True)
                        res %= MOD
                    else:
                        res += get_dp(i + 1, j - 1, False)
                        res %= MOD
        dp[i][j][called] = res
        return res

    return get_dp(0, k, False)


for i in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    print(count_ways(n, k, a))
