n, m = map(int, input().split())
ans = []
if n >= 1:
    ans.append(1)
    n -= 1
if m >= 1:
    ans.append(0)
    m -= 1
flag = False
while n > 0 or m > 0:
    if n > 0 and (not flag or ans[-1] == 0):
        ans.append(1)
        n -= 1
        flag = True
    elif m > 0 and ans[-1] == 1:
        ans.append(0)
        m -= 1
        flag = False
    elif ans[-1] == 1 and n > 0 and m == 0:
        ans.append(1)
        n -= 1
        flag = True
    else:
        break
if len(ans) == n + m:
    print(*ans, sep='')
else:
    print(-1)
