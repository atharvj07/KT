import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        String s = in.next();
        int[] ans = new int[n];

        int open = 0;
        int close = 0;
        Stack<Integer> stack = new Stack<>();

        for (int i = 0; i < n; i++) {
            char ch = s.charAt(i);
            if (ch == '(') {
                open++;
                stack.push(i);
            } else if (ch == ')') {
                if (!stack.isEmpty()) {
                    int j = stack.pop();
                    ans[j] = ans[i] = 1;
                    open--;
                    close++;
                } else {
                    ans[i] = -1;
                }
            } else {
                ans[i] = 1;
            }

            if (open > close || ans[i] == -1) {
                ans[i] = -1;
            }

            if (i > 0) {
                if (ans[i] != -1) {
                    ans[i] = Math.max(ans[i], ans[i - 1]);
                }
            }
        }

        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
