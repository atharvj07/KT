#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;
    while (q--) {
      int l, r;
      cin >> l >> r;
      int cnt = 0, last = -1;
      for (int i = l - 1; i < r - 1; i++) {
        if (s[i] == s[r - 1]) {
          if (last != i - 1) cnt++;
          last = i;
        }
      }
      if (cnt > 0) cout << "YES" << endl;
      else cout << "NO" << endl;
    }
  }
  return 0;
}
