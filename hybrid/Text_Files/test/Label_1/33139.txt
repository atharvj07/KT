import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long l = input.nextLong();
        long r = input.nextLong();
        long ans = r - l + 1;
        while (r > 0) {
            r /= 10;
            if (r == 0)
                break;
            l /= 10;
            if (l == 0)
                break;
            if (r % 10 != l % 10) {
                ans--;
                break;
            }
        }
        System.out.println(ans);
    }
}
