import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int n = sc.nextInt();
        boolean[][] reclaimed = new boolean[r + 2][3];
        for (int i = 0; i < n; i++) {
            int ri = sc.nextInt();
            int ci = sc.nextInt();
            reclaimed[ri][ci] = true;
            reclaimed[ri][3 - ci] = true;
        }
        boolean turn = true;
        for (int i = 1; i <= r; i++) {
            boolean canReclaim = false;
            for (int j = 1; j <= 2; j++) {
                if (!reclaimed[i][j] && !reclaimed[i - 1][3 - j] && !reclaimed[i + 1][3 - j]) {
                    canReclaim = true;
                    reclaimed[i][j] = turn;
                    reclaimed[i][3 - j] = turn;
                    turn = !turn;
                    break;
                }
            }
            if (!canReclaim) {
                System.out.println(turn ? "WIN" : "LOSE");
                return;
            }
        }
    }
}
