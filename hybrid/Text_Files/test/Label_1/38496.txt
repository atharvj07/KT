t = int(input())
for i in range(t):
    n, k = list(map(int, input().split()))
    if k == 1:
        print('a' * n)
    elif k == 2:
        print('ab' * (n // 2) + 'a' * (n % 2))
    else:
        s = ''
        for j in range(k):
            s += chr(ord('a') + j % k)
        print(s * (n // k) + s[0 : n % k])
