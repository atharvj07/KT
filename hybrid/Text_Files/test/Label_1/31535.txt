import java.util.*;
import java.io.*;

public class Main {
  static int n, ans;
  static int[] a;

  public static void dfs(int id, int cnt) {
    if (id == n + 1) {
      ans = Math.min(ans, cnt);
      return;
    }
    if (a[id] == 0) {
      dfs(id + 1, cnt);
    } else {
      if (id == 1 || a[id] % 2 != a[id - 1] % 2) {
        dfs(id + 1, cnt + 1);
      }
      if (id == n || a[id] % 2 != a[id + 1] % 2) {
        dfs(id + 1, cnt + 1);
      }
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    a = new int[n + 2];
    for (int i = 1; i <= n; i++) {
      a[i] = sc.nextInt();
    }
    ans = n;
    dfs(1, 0);
    System.out.println(ans);
  }
}
