def min_max_b_c(n, a, q, changes):
    def min_max_for_sequence(a):
        left = max(a[0], 0)
        right = sum(max(x, 0) for x in a)
        while left < right:
            mid = (left + right) // 2
            if check(a, mid):
                right = mid
            else:
                left = mid + 1
        return left

    def check(a, max_val):
        b = [min(a[0], max_val)]
        for i in range(1, n):
            b.append(min(max(b[i-1], a[i] - (max_val - b[i-1])), max_val))
            if b[i] + (max_val - b[i]) < a[i]:
                return False
        return True

    results = [min_max_for_sequence(a)]
    for l, r, x in changes:
        for i in range(l-1, r):
            a[i] += x
        results.append(min_max_for_sequence(a))
    return results

n = int(input().strip())
a = list(map(int, input().strip().split()))
q = int(input().strip())
changes = [list(map(int, input().strip().split())) for _ in range(q)]

results = min_max_b_c(n, a, q, changes)
for result in results:
    print(result)
