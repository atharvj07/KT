import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n - k + 1; i++) {
            long tmp = 0;
            for (int j = 0; j < k; j++) {
                tmp += Math.abs(a[i + j]);
            }
            sum = Math.max(sum, tmp);
        }
        System.out.println(sum);
    }
}
