import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] teams = new String[n];
        String[] cities = new String[n];
        for (int i = 0; i < n; i++) {
            teams[i] = sc.next();
            cities[i] = sc.next();
        }
        boolean[] used = new boolean[26];
        for (int i = 0; i < n; i++) {
            String team = teams[i];
            String city = cities[i];
            if (team.charAt(0) != city.charAt(0)) {
                String shortName = team.substring(0, 3);
                if (used[shortName.charAt(0) - 'A']) {
                    System.out.println("NO");
                    return;
                }
                used[shortName.charAt(0) - 'A'] = true;
                System.out.println("YES");
                System.out.println(shortName);
            } else {
                String shortName = team.substring(0, 2) + city.charAt(0);
                if (used[shortName.charAt(0) - 'A']) {
                    System.out.println("NO");
                    return;
                }
                used[shortName.charAt(0) - 'A'] = true;
                System.out.println("YES");
                System.out.println(shortName);
            }
        }
    }
}
