import java.util.*;

public class Main {
    static int n, k;
    static int[] dx = {-1, 1, 0, 0};
    static int[] dy = {0, 0, -1, 1};
    static int[][] map;
    static boolean[][] vis;
    static Queue<int[]> q;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        map = new int[101][101];
        vis = new boolean[101][101];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            vis[50 + x][50 + y] = true;
        }
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            map[50 + x][50 + y] = 1;
        }
        q = new LinkedList<>();
        int res = -1;
        for (int i = 0; i < 101; i++) {
            for (int j = 0; j < 101; j++) {
                if (vis[i][j] && map[i][j] == 0) {
                    q.add(new int[]{i, j, 0});
                    res = 0;
                }
            }
        }
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            int time = cur[2];
            if (map[x][y] == 1) {
                res = time;
                break;
            }
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx >= 0 && nx < 101 && ny >= 0 && ny < 101 && !vis[nx][ny]) {
                    vis[nx][ny] = true;
                    q.add(new int[]{nx, ny, time + 1});
                }
            }
        }
        System.out.println(res);
    }
}
