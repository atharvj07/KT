import java.util.*;

public class Main {
    static class MagicBox {
        private int r1, r2, c1, c2, d1, d2;
        private ArrayList<Integer> gems = new ArrayList<>();

        public MagicBox(Scanner sc) {
            r1 = sc.nextInt();
            r2 = sc.nextInt();
            c1 = sc.nextInt();
            c2 = sc.nextInt();
            d1 = sc.nextInt();
            d2 = sc.nextInt();
        }

        public int getR1() {
            return r1;
        }

        public int getR2() {
            return r2;
        }

        public int getC1() {
            return c1;
        }

        public int getC2() {
            return c2;
        }

        public int getD1() {
            return d1;
        }

        public int getD2() {
            return d2;
        }

        public void addGem(int gem) {
            gems.add(gem);
        }

        public List<Integer> getGems() {
            return gems;
        }

        public boolean isSolved() {
            return !gems.isEmpty() 
                    && r1 == gems.get(0) + gems.get(1) 
                    && r2 == gems.get(2) + gems.get(3) 
                    && c1 == gems.get(0) + gems.get(2) 
                    && c2 == gems.get(1) + gems.get(3) 
                    && d1 == gems.get(0) + gems.get(3) 
                    && d2 == gems.get(1) + gems.get(2);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        MagicBox box = new MagicBox(sc);
        for (int i = 1; i <= 9; i++) {
            box.addGem(i);
        }
        List<Integer> gems = box.getGems();
        if (solve(box, gems, 0)) {
            for (int gem : box.getGems()) {
                System.out.print(gem + " ");
            }
            System.out.println();
        } else {
            System.out.println("-1");
        }
    }

    private static boolean solve(MagicBox box, List<Integer> gems, int index) {
        if (index == 4) {
            return box.isSolved();
        }
        for (int i = index; i < 4; i++) {
            Collections.swap(gems, index, i);
            if (solve(box, gems, index + 1)) {
                return true;
            }
            Collections.swap(gems, index, i);
        }
        return false;
    }
}
