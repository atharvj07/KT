import java.util.Scanner;

public class BerlandCalendar {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int n = scanner.nextInt();

        long p = 0;
        for (int i = 1; i <= n; i++) {
            int daysInYear = a + i - 1;
            int monthsInYear = 0;
            int daysInMonth = 0;
            for (int j = 1; j * j <= daysInYear; j++) {
                if (daysInYear % j == 0) {
                    if (j > daysInMonth) {
                        daysInMonth = j;
                        monthsInYear = daysInYear / j;
                    }
                    int otherMonthDays = daysInYear / j;
                    if (otherMonthDays > daysInMonth) {
                        daysInMonth = otherMonthDays;
                        monthsInYear = j;
                    }
                }
            }
            p += daysInYear / daysInMonth;
        }

        System.out.println(p);
    }
}
