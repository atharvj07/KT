import java.util.*;
import java.lang.*;
import static java.lang.Math.min;
import static java.lang.Math.max;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] a = new int[m], p = new int[n + 1], mi = new int[n + 1], ma = new int[n + 1];
        Arrays.fill(mi, n + 1);
        for (int i = 1; i <= n; i++) p[i] = i;
        for (int i = 0; i < m; i++) {
            a[i] = in.nextInt();
            int x = p[a[i]];
            mi[a[i]] = min(mi[a[i]], x);
            ma[a[i]] = max(ma[a[i]], x);
            for (int j = x; j > 1; j--) p[j] = p[j - 1];
            p[1] = x;
        }
        for (int i = 1; i <= n; i++) System.out.println(mi[i] + " " + ma[i]);
    }
}
