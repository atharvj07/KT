#include <bits/stdc++.h>
using namespace std;

int main() {
    string s; getline(cin, s);
    s.erase(remove(s.begin(), s.end(), '{'), s.end());
    s.erase(remove(s.begin(), s.end(), '}'), s.end());
    s.erase(remove(s.begin(), s.end(), ' '), s.end());
    sort(s.begin(), s.end());
    int n = unique(s.begin(), s.end()) - s.begin();
    cout << n << endl;
    return 0;
}
