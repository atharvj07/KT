import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = i + 1;
        }
        ArrayList<Integer> b = new ArrayList<>();
        int m = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (gcd(a[i], a[j]) > 1) {
                    b.add(a[i]);
                    b.add(a[j]);
                    m++;
                    i = j;
                    break;
                }
            }
        }
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            System.out.println(b.get(2 * i) + " " + b.get(2 * i + 1));
        }
    }

    public static int gcd(int a, int b) {
        while (b > 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
