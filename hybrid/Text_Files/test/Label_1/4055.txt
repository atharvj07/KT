#include <iostream>
#include <vector>

void solve() {
    int num_tests;
    std::cin >> num_tests;

    for (int test_num = 0; test_num < num_tests; test_num++) {
        long long n;
        std::cin >> n;

        std::vector<int> digits;
        while (n > 0) {
            digits.push_back(n % 10);
            n /= 10;
        }

        for (int i = digits.size() - 1; i > 0; i--) {
            if (digits[i] % 2 != digits[i - 1] % 2) {
                std::swap(digits[i], digits[i - 1]);
            }
        }

        for (int d : digits) {
            std::cout << d;
        }
        std::cout << std::endl;
    }
}

int main() {
    solve();
    return 0;
}
