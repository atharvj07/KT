import java.util.*;
import java.lang.*;
class Main {

    static int n;
    static int m;

    static class Concert {
        int a;
        int b;
        int c;
        int d;

        Concert(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
            this.d = 0;
        }
    }

    public static void main(String[] args) {
        Scanner stdIn = new Scanner(System.in);
        n = stdIn.nextInt();
        m = stdIn.nextInt();

        Concert[] songs = new Concert[n];

        for (int i = 0; i < n; i++) {
            songs[i] = new Concert(stdIn.nextInt(), stdIn.nextInt(), stdIn.nextInt());
        }

        int maxScore = search(songs, 0, m, 0);
        System.out.println(maxScore);
    }

    private static int search(Concert[] songs, int index, int sticks, int score) {
        if (index == n) return score;

        // level 1 type glowsticks
        int scoreIfLevel1 = search(songs, index + 1, sticks - 1, score + songs[index].b);

        // level 2 type glowsticks
        int scoreIfLevel2 = search(songs, index + 1, sticks - 1, score + songs[index].a);

        // no glowsticks
        int scoreIfNoGlowsticks = search(songs, index + 1, sticks, score + songs[index].c);

        return Math.max(Math.max(scoreIfLevel1, scoreIfLevel2), scoreIfNoGlowsticks);
    }
}
