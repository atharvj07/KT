import static java.lang.Math.max;
import static java.lang.System.in;
import java.io.PrintWriter;
import java.util.*;


class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(in);
        PrintWriter out = new PrintWriter(System.out);

        int n = sc.nextInt();

        int[] b = new int[n];
        int[] i = new int[n];
        int[] r = new int[n];
        for (int j = 0; j < n; ++j) { b[j] = sc.nextInt(); }
        for (int j = 0; j < n; ++j) { i[j] = sc.nextInt(); }
        for (int j = 0; j < n; ++j) { r[j] = sc.nextInt(); }

        int[] bMax = new int[n];
        int[] iMax = new int[n];
        int[] rMax = new int[n];

        bMax[n - 1] = b[n - 1];
        iMax[n - 1] = i[n - 1];
        rMax[n - 1] = r[n - 1];

        for (int j = n - 2; j >= 0; --j) {
            bMax[j] = max(bMax[j + 1], b[j]);
            iMax[j] = max(iMax[j + 1], i[j]);
            rMax[j] = max(rMax[j + 1], r[j]);
        }

        int ans = 0;
        for (int j = 0; j < n; ++j) {
            ans += (b[j] < bMax[j] && i[j] < iMax[j] && r[j] < rMax[j]) ? 1 : 0;
        }

        out.println(ans);
        out.close();
    }
}
