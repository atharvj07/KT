import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] balls = new int[n];
        for (int i = 0; i < n; i++) {
            balls[i] = sc.nextInt();
        }
        int[] res = new int[n];
        int[] freq = new int[n + 1];
        int max_freq = 0;
        int l = 0;
        for (int r = 0; r < n; r++) {
            freq[balls[r]]++;
            if (freq[balls[r]] > max_freq) {
                max_freq = freq[balls[r]];
                l = r;
            } else if (freq[balls[r]] == max_freq) {
                l = Math.min(l, r);
            }
            res[balls[r] - 1] += r - l + 1;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
