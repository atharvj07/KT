import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        double[] pos = new double[n];
        int[] species = new int[n];
        for (int i = 0; i < n; i++) {
            species[i] = sc.nextInt();
            pos[i] = sc.nextDouble();
        }
        int[] count = new int[m+1];
        for (int i = 0; i < n; i++) {
            count[species[i]]++;
        }
        int answer = 0;
        for (int i = 1; i <= m; i++) {
            answer += Math.abs(count[i] - (i == m ? 0 : 1));
        }
        System.out.println(answer);
    }
}
