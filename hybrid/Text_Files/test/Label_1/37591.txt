n, w = map(int, input().split())
capacities = list(map(int, input().split()))
capacities.sort()
girl_capacity = capacities[n]
boy_capacity = capacities[0] * 2

if boy_capacity > girl_capacity:
    total_water = min(w, 3 * n * girl_capacity)
else:
    total_water = min(w, 3 * n * boy_capacity / 2)

print(total_water / 3 * n)
