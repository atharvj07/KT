a1 = int(input())
a2 = int(input())
k1 = int(input())
k2 = int(input())
n = int(input())

# Maximum number of players that could have been thrown out
max_out = min(n // k1, a1) + min((n - min(n // k1, a1) * k1) // k2, a2)

# Minimum number of players that could have been thrown out
min_out = max(0, n - (k1 - 1) * a1 - (k2 - 1) * a2)

# In case k1 and k2 need to be swapped for max calculation
if k1 > k2:
    max_out = min(n // k2, a2) + min((n - min(n // k2, a2) * k2) // k1, a1)

print(min_out, max_out)
