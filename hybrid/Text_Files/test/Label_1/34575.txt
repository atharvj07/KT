import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int R = sc.nextInt();
        int Q = sc.nextInt();

        // Initialize the rule list
        List<Rule> rules = new ArrayList<>();
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            rules.add(new Rule(a, b, c));
        }

        // Initialize the hand list
        List<Hand> hands = new ArrayList<>();
        for (int i = 0; i < Q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            int z = sc.nextInt();
            hands.add(new Hand(x, y, z));
        }

        // Initialize the memoization table
        int[][][] memo = new int[N + 1][N + 1][Q + 1];
        for (int i = 0; i <= N; i++) {
            for (int j = 0; j <= N; j++) {
                Arrays.fill(memo[i][j], -1);
            }
        }

        // Calculate the maximum amount of money that can be increased
        int max = 0;
        for (Hand hand : hands) {
            max = Math.max(max, dfs(hand.x, hand.y, 0, rules, hands, memo, R));
        }

        // Print the maximum amount of money that can be increased
        System.out.println(max);
    }

    private static int dfs(int x, int y, int i, List<Rule> rules, List<Hand> hands, int[][][] memo, int R) {
        // Check if the memoization table has already been calculated
        if (memo[x][y][i] != -1) {
            return memo[x][y][i];
        }

        // Calculate the reward for the current hand
        int reward = 0;
        if ((x + y) % R == 0) {
            reward = hands.get(i).z;
        }

        // Try all possible rules
        int max = reward;
        for (Rule rule : rules) {
            // Check if the rule can be applied
            if (rule.a == x || rule.a == y) {
                // Apply the rule
                int newX = rule.a == x ? rule.b : x;
                int newY = rule.a == y ? rule.b : y;

                // Calculate the maximum amount of money that can be increased after applying the rule
                max = Math.max(max, dfs(newX, newY, i + 1, rules, hands, memo, R) - rule.c + reward);
            }
        }

        // Store the maximum amount of money that can be increased in the memoization table
        memo[x][y][i] = max;

        // Return the maximum amount of money that can be increased
        return max;
    }

    private static class Rule {
        int a;
        int b;
        int c;

        public Rule(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    private static class Hand {
        int x;
        int y;
        int z;

        public Hand(int x, int y, int z) {
            this.x = x;
            this.y = y;
            this.z = z;
        }
    }
}
