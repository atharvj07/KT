#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> a[i][j];
      }
    }
    bool ok = false;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] == 0) {
          int cnt = 0;
          for (int k = 0; k < m; k++) {
            cnt += (a[i][k] > 0);
          }
          for (int k = 0; k < n; k++) {
            cnt += (a[k][j] > 0);
          }
          if (cnt == 0) {
            ok = true;
            break;
          }
        }
      }
    }
    if (ok)
      cout << "Ashish" << endl;
    else
      cout << "Vivek" << endl;
  }
  return 0;
}
