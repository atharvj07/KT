import java.util.Scanner;

public class MakeItOne {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        int result = findMinSubset(arr);
        System.out.println(result);
    }

    private static int findMinSubset(int[] arr) {
        int gcd = arr[0];
        for (int i = 1; i < arr.length; i++) {
            gcd = gcd(gcd, arr[i]);
        }

        if (gcd != 1) {
            return -1;
        }

        return findMinSubsetHelper(arr, 0, 0);
    }

    private static int findMinSubsetHelper(int[] arr, int index, int subsetSize) {
        if (index == arr.length) {
            return subsetSize;
        }

        int withoutCurrent = findMinSubsetHelper(arr, index + 1, subsetSize);
        int withCurrent = findMinSubsetHelper(arr, index + 1, subsetSize + 1);

        if (withCurrent != -1) {
            return withCurrent;
        }

        return withoutCurrent;
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
