#include<bits/stdc++.h>
using namespace std;
int main() {
    long long n; cin >> n;
    long long sm1 = 0, sm2 = 0, ans = 0;
    string vec;
    while (n--) {
        long long x; cin >> x;
        if (x > 0) sm1 += x;
        else if (x < 0) sm2 -= x;
        if (sm1 == sm2) {
            if (!vec.empty() && vec.back() == 's') ans = 1;
            else ans = -1;
        }
        if (ans == 1) vec += "s";
        else if (ans == -1) vec += "f";
    }
    if (ans == 0) {
        if (sm1 > sm2) ans = 1;
        else if (sm2 > sm1) ans = -1;
    }
    if (ans == 1) cout << "first" << endl;
    else cout << "second" << endl;
}
