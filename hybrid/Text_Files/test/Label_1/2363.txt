#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef vector<int> vi;
typedef pair<int,int> pi;
const ll mod=1e9+7;
#define pb push_back
#define mp make_pair

int main() {
	ll n,k; cin>>n>>k;
	vector<ll> v(n+10);
	for(ll i=1; i<=n; i++) cin>>v[i];
	for(ll i=1; i<=n-k+1; i++){
		v[i]+=v[i+k];
	}
	for(ll i=n-k+2; i<=n; i++) v[i]=0;
	ll mx=0, pos=0;
	for(ll i=1; i<=n-k+1; i++){
		if(v[i]>mx){
			mx=v[i];
			pos=i;
		}
	}
	cout<<pos<<" "<<pos+k<<endl;
	return 0;
}
