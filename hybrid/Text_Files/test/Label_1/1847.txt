#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin >>n;
    
    vector<int> arr;
    while(n>0){
        int cur = 0;
        while((1<<cur)<=n) cur++;
        cur--;
        arr.push_back(1<<cur);
        n-=(1<<cur);
    }
    cout <<arr.size()<<endl;
    for(auto x:arr)cout <<x<<" ";
    cout <<endl;
}
