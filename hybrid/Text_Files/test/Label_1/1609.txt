#include<bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
#define mp(a,b) make_pair(a,b)
priority_queue<pii>q;
map<int,priority_queue<pii>>m;
int main(){
    int n;cin>>n;
    priority_queue<pii>pq;
    for(int i=1;i<=n;++i){
        string s;int k,x;cin>>s>>k;
        pq.push(mp(-k,i));
        while(k--){
            cin>>x;
            m[i].push(mp(x,0));
        }
    }
    string ans="";
    while(!pq.empty()){
        auto now=pq.top();pq.pop();
        int id=now.second;
        if(m[id].empty()){
            cout<<"Impossible";
            return 0;
        }
        now=m[id].top();m[id].pop();
        ans.push_back('a'+id-1);
        if(now.first!=1)m[id].push(mp(now.first-1,0));
        q.push(mp(now.first,id));
    }
    while(!q.empty()){
        auto now=q.top();q.pop();
        int id=now.second;
        while(!m[id].empty()){
            ans.push_back('a'+id-1);
            auto newnow=m[id].top();m[id].pop();
            if(newnow.first!=1)m[id].push(mp(newnow.first-1,0));
            q.push(mp(newnow.first,id));
        }
    }
    cout<<ans;
    return 0;
}
