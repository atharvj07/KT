#include <iostream>
#include <vector>
using namespace std;
using ll = long long;

int main() {
  ll T;
  cin >> T;
  while (T--) {
    string s;
    cin >> s;
    ll n = s.size();

    if (s[0] == s[n - 1]) {
      cout << 0 << '\n';
    } else {
      ll a[n + 1], b[n + 1];
      a[n] = b[0] = 0;
      for (ll i = 0; i <= n - 1; i++) {
        a[i + 1] = a[i];
        b[i + 1] = b[i];
        if (s[i] == s[0]) a[i + 1]++;
        if (s[i] == s[n - 1]) b[i + 1]++;
      }
      ll ans = n;
      for (ll i = 0; i < n; i++) {
        ans = min(ans, a[i] + n - 1 - b[i + 1]);
        ans = min(ans, b[i] + n - 1 - a[i + 1]);
      }
      cout << ans << '\n';
    }
  }
  return 0;
}
