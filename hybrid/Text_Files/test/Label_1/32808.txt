import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            int ans = 0;
            for (int i = 1; i <= a; i++) {
                for (int j = i; j <= b; j++) {
                    for (int k = j; k <= c; k++) {
                        if (a % i == 0 && b % j == 0 && c % k == 0) {
                            ans++;
                        }
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
