import sys

MAX_LIMIT = 2 * 10 ** 5

n, t = map(int, input().split())
problems = []
for i in range(n):
    a, b = map(int, input().split())
    problems.append((i + 1, a, b))

problems.sort(key=lambda x: (x[1], x[2], x[0]), reverse=True)
dp = [0] * (MAX_LIMIT + 1)
for i in range(n):
    idx = problems[i][0]
    limit = problems[i][1]
    time = problems[i][2]
    if i + 1 <= limit:
        for j in range(t - time, limit - 1, -1):
            dp[j] = max(dp[j], dp[j - time] + 1)
print(dp[t])
k = dp[t]
ans = []
for i in range(n - 1, -1, -1):
    idx = problems[i][0]
    limit = problems[i][1]
    time = problems[i][2]
    if k <= limit and dp[t - time] + 1 == k:
        ans.append(idx)
        k -= 1
        t -= time
print(k)
print(*ans)
