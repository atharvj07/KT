from collections import deque

def read_input():
    n, m = map(int, input().split())
    field = [list(input().strip()) for _ in range(n)]
    return n, m, field

def bfs(field, x, y, n, m):
    visited = [[False] * m for _ in range(n)]
    queue = deque([(x, y)])
    visited[x][y] = True
    size = 1

    while queue:
        cx, cy = queue.popleft()
        for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nx, ny = cx + dx, cy + dy
            if 0 <= nx < n and 0 <= ny < m and not visited[nx][ny] and field[nx][ny] == '.':
                queue.append((nx, ny))
                visited[nx][ny] = True
                size += 1
    return size

def solve(n, m, field):
    components = [[0 if field[i][j] == '*' else -1 for j in range(m)] for i in range(n)]
    for i in range(n):
        for j in range(m):
            if field[i][j] == '*' and components[i][j] == 0:
                components[i][j] = bfs(field, i, j, n, m) % 10

    for i in range(n):
        for j in range(m):
            if field[i][j] == '.':
                components[i][j] = '.'

    for row in components:
        print(''.join(map(str, row)))

n, m, field = read_input()
solve(n, m, field)
