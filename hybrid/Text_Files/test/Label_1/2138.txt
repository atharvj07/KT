#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  int t = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (cnt == 0) {
      t++;
    }
    if (a[i] == 1) {
      cnt++;
    } else {
      cnt--;
    }
  }
  cout << t << endl;
  cnt = 0;
  for (int i = 0; i < n; i++) {
    if (cnt == 0) {
      cout << a[i] << " ";
    }
    if (a[i] == 1) {
      cnt++;
    } else {
      cnt--;
    }
  }
  return 0;
}
