import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]);
        int k = Integer.parseInt(tokens[1]);
        int a = Integer.parseInt(tokens[2]);
        int b = Integer.parseInt(tokens[3]);
        int q = Integer.parseInt(tokens[4]);
        int[] orders = new int[n + 1];
        int[] production = new int[n + 1];

        for (int i = 0; i < q; i++) {
            tokens = br.readLine().split(" ");
            if (tokens[0].equals("1")) {
                orders[Integer.parseInt(tokens[1])] += Integer.parseInt(tokens[2]);
            } else {
                int pi = Integer.parseInt(tokens[1]);
                int start = Math.max(1, pi - k + 1);
                int left = a * (pi - start + 1) - (b * (n - pi));
                for (int j = 1; j <= pi; j++) {
                    left -= orders[j];
                }
                System.out.println(Math.max(0, left));
            }
        }
    }
}
