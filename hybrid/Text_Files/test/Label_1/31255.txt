import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int N = scan.nextInt();

        // This map stores the power of each number.
        HashMap<Long, Integer> numPowers = new HashMap<>();

        for (int i = 0; i < N; i++) {
            long num = scan.nextLong();

            int power = 1;
            while (num % 3 == 0) {
                power *= 3;
                num /= 3;
            }

            numPowers.put(num, Math.max(numPowers.getOrDefault(num, 0), power));
        }

        // Count the number of numbers with different powers.
        int maxNum = 0;
        for (int power : numPowers.values()) {
            maxNum += (power == 1 ? 1 : 0);
        }

        System.out.println(numPowers.size() + maxNum);
    }
}
