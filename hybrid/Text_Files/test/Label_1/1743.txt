#include <bits/stdc++.h>
#define int long long
using namespace std;

int gcd(int a, int b)
{
	if (a == 0)
		return b;
	return gcd(b % a, a);
}

int32_t main()
{
	int t;
	cin >> t;
	while (t--)
	{
		int n;
		cin >> n;
		vector<int> a(n);
		for (int i = 0; i < n; i++)
		{
			cin >> a[i];
		}
		int ans = 0;
		while (1)
		{
			vector<int> b(n);
			for (int i = 0; i < n; i++)
			{
				b[i] = gcd(a[i], a[(i + 1) % n]);
			}
			bool f = 1;
			for (int i = 0; i < n; i++)
			{
				if (b[i] != b[(i + 1) % n])
				{
					f = 0;
					break;
				}
			}
			if (f)
				break;
			a = b;
			ans++;
		}
		cout << ans << '\n';
	}
	return 0;
}
