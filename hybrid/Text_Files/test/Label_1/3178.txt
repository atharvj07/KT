#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

int perm[100000];
int main(){
    int n;
    scanf("%d",&n);
    if(n==6){
        puts("NO");
    }
    else{
        puts("YES");
        for(int i=2;i<=n;i++)perm[i]=(i-1)&1?n-i+1:i-1;
        swap(perm[2],perm[n]);
        perm[1]=n;
        for(int i=1;i<=n;i++)printf("%d\n",perm[i]);
    }
    return 0;
}
