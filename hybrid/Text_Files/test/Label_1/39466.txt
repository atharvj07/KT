h, w = map(int, input().split())
pic = [input() for _ in range(h)]
cnt = 0
for i in range(h):
    for j in range(w):
        if pic[i][j] == '*':
            flag = True
            for x, y in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                ni, nj = i + x, j + y
                while 0 <= ni < h and 0 <= nj < w and pic[ni][nj] == '*':
                    ni += x
                    nj += y
                if 0 <= ni < h and 0 <= nj < w and pic[ni][nj] == '.':
                    flag = False
            if flag:
                cnt += 1
print('YES' if cnt == 1 else 'NO')
