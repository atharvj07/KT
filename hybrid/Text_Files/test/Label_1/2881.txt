#include <iostream>
#include <bits/stdc++.h>
using namespace std;

int main() {
  int tc;
  cin >> tc;
  while (tc--) {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    if (a > 0)
      cout << "Ya ";
    else
      cout << "Tidak ";
    if (b > 0)
      cout << "Ya ";
    else
      cout << "Tidak ";
    if (c > 0)
      cout << "Ya ";
    else
      cout << "Tidak ";
    if (d > 0)
      cout << "Ya ";
    else
      cout << "Tidak ";
    cout << endl;
  }
}
