import sys

grid = []
for _ in range(5):
    grid.append(list(map(int, sys.stdin.readline().split())))

happiness = [[0] * 5 for _ in range(5)]
for i in range(5):
    for j in range(5):
        happiness[i][j] = grid[i][j] + grid[j][i]

maximum = 0

for i in range(5):
    for j in range(i + 1, 5):
        for k in range(j + 1, 5):
            maximum = max(maximum, happiness[i][j] + happiness[j][k] + happiness[k][i])
            maximum = max(maximum, happiness[i][k] + happiness[k][j] + happiness[j][i])

print(maximum)
