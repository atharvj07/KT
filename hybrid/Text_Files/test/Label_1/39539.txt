import sys
s   = sys.stdin.readline()[:-1]
n   = int(sys.stdin.readline())
S   = [set() for j in range(len(s) + 1)]
cnt = 0
for _ in range(n):
    p, l, r = sys.stdin.readline().split()
    l, r= int(l), int(r)
    cnt -= sum(l <= len(j) <= r for j in S[len(s) - len(p) :])
    for j in range(len(s) - len(p) + 1):
        if s[j:j + len(p)] == p:
            S[j].add(len(p))
            cnt += l <= len(p) <= r
print(cnt)
