import java.util.*;

public class Main {

    static int n, m;
    static int[][] map;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        map = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                map[i][j] = sc.nextInt();
            }
        }

        while (m-- > 0) {
            int o = sc.nextInt();
            if (o == 0) {
                int r = sc.nextInt();
                int c = sc.nextInt();
                int size = sc.nextInt();
                int angle = sc.nextInt();
                rotate(r - 1, c - 1, size, angle);
            } else if (o == 1) {
                int r = sc.nextInt();
                int c = sc.nextInt();
                int size = sc.nextInt();
                reverse(r - 1, c - 1, size);
            } else if (o == 2) {
                int r = sc.nextInt();
                leftShift(r - 1);
            } else if (o == 3) {
                int r = sc.nextInt();
                rightShift(r - 1);
            } else if (o == 4) {
                int r = sc.nextInt();
                int c = sc.nextInt();
                islandReverse(r - 1, c - 1);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(map[i][j] + " ");
            }
            System.out.println();
        }

        sc.close();
    }

    static void rotate(int r, int c, int size, int angle) {
        int[][] temp = new int[size][size];
        for (int i = 0; i < size; i++) {
            for (int j = 0; j < size; j++) {
                temp[i][j] = map[r + i][c + j];
            }
        }

        if (angle == 90) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    map[r + j][c + size - 1 - i] = temp[i][j];
                }
            }
        } else if (angle == 180) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    map[r + size - 1 - i][c + size - 1 - j] = temp[i][j];
                }
            }
        } else if (angle == 270) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    map[r + size - 1 - j][c + i] = temp[i][j];
                }
            }
        }
    }

    static void reverse(int r, int c, int size) {
        for (int i = 0; i < size; i++) {
            for (int j = 0; j < size; j++) {
                map[r + i][c + j] ^= 1;
            }
        }
    }

    static void leftShift(int r) {
        int[] temp = new int[n];
        for (int i = 0; i < n; i++) {
            temp[i] = map[r][i];
        }
        for (int i = 1; i < n; i++) {
            map[r][i - 1] = temp[i];
        }
        map[r][n - 1] = temp[0];
    }

    static void rightShift(int r) {
        int[] temp = new int[n];
        for (int i = 0; i < n; i++) {
            temp[i] = map[r][i];
        }
        for (int i = n - 2; i >= 0; i--) {
            map[r][i + 1] = temp[i];
        }
        map[r][0] = temp[n - 1];
    }

    static void islandReverse(int r, int c) {
        Queue<int[]> queue = new LinkedList<>();
        boolean[][] visited = new boolean[n][n];

        int[] dr = {-1, 0, 1, 0};
        int[] dc = {0, 1, 0, -1};

        map[r][c] ^= 1;
        queue.offer(new int[]{r, c});
        visited[r][c] = true;

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            for (int i = 0; i < 4; i++) {
                int nr = current[0] + dr[i];
                int nc = current[1] + dc[i];
                if (nr >= 0 && nr < n && nc >= 0 && nc < n && !visited[nr][nc] && map[nr][nc] == map[current[0]][current[1]]) {
                    map[nr][nc] ^= 1;
                    queue.offer(new int[]{nr, nc});
                    visited[nr][nc] = true;
                }
            }
        }
    }
}
