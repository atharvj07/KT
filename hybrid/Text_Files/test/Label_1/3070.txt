#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int g = a[0];
    for (int x : a) g = __gcd(g, x);
    int ans = 0;
    for (int x : a) ans += x != g;
    cout << ans << "\n";
  }
  return 0;
}
