import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n; i++) {
            graph.get(p[i]).add(i);
        }

        int[] values = new int[n];
        for (int i = 0; i < n; i++) {
            values[i] = -1;
        }

        if (assignValues(graph, values, 0)) {
            System.out.println("POSSIBLE");
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }

    private static boolean assignValues(List<List<Integer>> graph, int[] values, int vertex) {
        if (values[vertex] != -1) {
            return true;
        }

        for (int i = 0; i < graph.size(); i++) {
            if (graph.get(vertex).contains(i)) {
                if (!assignValues(graph, values, i)) {
                    return false;
                }
            }
        }

        for (int i = 0; ; i++) {
            boolean valid = true;
            for (int neighbor : graph.get(vertex)) {
                if (values[neighbor] == i) {
                    valid = false;
                    break;
                }
            }

            if (valid) {
                values[vertex] = i;
                return true;
            }
        }
    }
}
