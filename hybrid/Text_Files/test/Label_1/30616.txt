import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long k = scanner.nextLong();
        long a = scanner.nextLong();
        long b = scanner.nextLong();
        long start = (a + k - 1) / k;
        long end = b / k;
        System.out.println(end - start + 1);
    }
}
