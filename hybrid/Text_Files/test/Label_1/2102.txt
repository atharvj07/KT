#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 15;
int n, m, bx, by;
char x[maxn], y[maxn];

int main() {
	int s[2] = {0, 0};
	cin >> n >> bx; cin >> x;
	cin >> m >> by; cin >> y;
	for (int i = 0; i < n; i++) {
		s[0] = s[0] * bx + x[i] - '0';
	}
	for (int i = 0; i < m; i++) {
		s[1] = s[1] * by + y[i] - '0';
	}
	cout << ((s[0] < s[1]) ? '<' :
			(s[0] > s[1]) ? '>' : '=');
	return 0;
}
