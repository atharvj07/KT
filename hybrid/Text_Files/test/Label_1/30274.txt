import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1 || a[i] == 3 || a[i] == 5 || a[i] == 7 || a[i] == 9 || a[i] == 11 || a[i] == 13 || a[i] == 15 || a[i] == 17 || a[i] == 19 || a[i] == 21 || a[i] == 23 || a[i] == 25 || a[i] == 27 || a[i] == 29 || a[i] == 31) {
                count++;
            }
        }
        System.out.println(count);
    }
}
