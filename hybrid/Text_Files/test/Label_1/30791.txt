import java.io.*;
import java.util.*;
import java.util.stream.*;
import static java.util.stream.Collectors.toList;

class Main {
    static class UnionFindSet {
        int[] parent;
        int[] size;
        int components;

        UnionFindSet(int n) {
            parent = new int[n];
            size = new int[n];
            components = n;
            for (int i = 0; i < n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }

        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }

        boolean union(int p, int q) {
            int rootP = find(p);
            int rootQ = find(q);
            if (rootP == rootQ) return false;
            if (size[rootP] < size[rootQ]) {
                parent[rootP] = rootQ;
                size[rootQ] += size[rootP];
            } else {
                parent[rootQ] = rootP;
                size[rootP] += size[rootQ];
            }
            components--;
            return true;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        UnionFindSet ufs = new UnionFindSet(n);
        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) adjList.add(new ArrayList<>());

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int p = Integer.parseInt(st.nextToken()) - 1;
            int q = Integer.parseInt(st.nextToken()) - 1;
            ufs.union(p, q);
            adjList.get(p).add(q);
            adjList.get(q).add(p);
        }

        if (ufs.components != 2) {
            System.out.println(-1);
            return;
        }

        int[] bobPath = new int[n];
        int[] alexPath = new int[n];
        Queue<Integer> bobQueue = new LinkedList<>();
        Queue<Integer> alexQueue = new LinkedList<>();
        bobQueue.offer(0);
        alexQueue.offer(n - 1);

        int steps = 0;
        boolean[] visitedBob = new boolean[n];
        boolean[] visitedAlex = new boolean[n];
        visitedBob[0] = visitedAlex[n - 1] = true;

        while (!bobQueue.isEmpty() && !alexQueue.isEmpty()) {
            int sizeBob = bobQueue.size();
            int sizeAlex = alexQueue.size();

            for (int i = 0; i < sizeBob; i++) {
                int node = bobQueue.poll();
                bobPath[steps] = node;
                for (int next : adjList.get(node)) {
                    if (!visitedBob[next]) {
                        visitedBob[next] = true;
                        if (next == n - 1) {
                            System.out.println(++steps);
                            System.out.println(Arrays.stream(bobPath).limit(steps + 1).mapToObj(String::valueOf).collect(Collectors.joining(" ")) + " ");
                            System.out.println(Arrays.stream(alexPath).limit(steps + 1).mapToObj(String::valueOf).collect(Collectors.joining(" ")));
                            return;
                        }
                        bobQueue.offer(next);
                    }
                }
            }

            for (int i = 0; i < sizeAlex; i++) {
                int node = alexQueue.poll();
                alexPath[steps] = node;
                for (int next : adjList.get(node)) {
                    if (!visitedAlex[next]) {
                        visitedAlex[next] = true;
                        if (next == 0) {
                            System.out.println(++steps);
                            System.out.println(Arrays.stream(bobPath).limit(steps + 1).mapToObj(String::valueOf).collect(Collectors.joining(" ")) + " ");
                            System.out.println(Arrays.stream(alexPath).limit(steps + 1).mapToObj(String::valueOf).collect(Collectors.joining(" ")));
                            return;
                        }
                        alexQueue.offer(next);
                    }
                }
            }

            steps++;
        }

        System.out.println(-1);
    }
}
