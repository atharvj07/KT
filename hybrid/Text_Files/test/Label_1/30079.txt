import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int s = sc.nextInt();
        int v1 = sc.nextInt();
        int v2 = sc.nextInt();
        int t1 = sc.nextInt();
        int t2 = sc.nextInt();
        long time1 = (long) (s * v1) + t1 + t1;
        long time2 = (long) (s * v2) + t2 + t2;
        if (time1 < time2) {
            System.out.println("First");
        } else if (time1 > time2) {
            System.out.println("Second");
        } else {
            System.out.println("Friendship");
        }
    }
}
