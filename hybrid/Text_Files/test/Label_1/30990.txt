import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) {
                break;
            }
            boolean[][] grid = new boolean[n][n];
            for (int i = 0; i < n; i++) {
                String line = scanner.next();
                for (int j = 0; j < n; j++) {
                    grid[i][j] = line.charAt(j) == '*';
                }
            }
            int side = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (!grid[i][j]) {
                        int temp = 1;
                        boolean ok = true;
                        while (ok && i + temp < n && j + temp < n) {
                            for (int k = 0; k < temp; k++) {
                                if (grid[i + k][j + temp] || grid[i + temp][j + k]) {
                                    ok = false;
                                    break;
                                }
                            }
                            if (ok) {
                                temp++;
                            }
                        }
                        side = Math.max(side, temp - 1);
                    }
                }
            }
            System.out.println(side);
        }
        scanner.close();
    }
}
