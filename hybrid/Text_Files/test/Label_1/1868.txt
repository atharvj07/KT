#include <iostream>
#include <vector>
#include <queue>
#include <set>
#include <algorithm>
using namespace std;

const int MAXN = 3e5 + 5;

struct Edge { int to, w; };

vector<Edge> adj[MAXN];
vector<int> police;
int dist[MAXN], ans[MAXN];
int n, k, d, cnt;

void dfs(int x, int p, int dis){
    dist[x] = dis;
    for(Edge e : adj[x]){
        if(e.to != p && dis+e.w <= d) dfs(e.to, x, dis+e.w);
    }
}

int main(){
    cin >> n >> k >> d;
    for(int i = 0; i < k; i++){
        int x; cin >> x;
        police.push_back(x);
    }
    for(int i = 0; i < n-1; i++){
        int u, v; cin >> u >> v;
        adj[u].push_back({v, 1});
        adj[v].push_back({u, 1});
    }
    for(int x : police){
        dfs(x, -1, 0);
    }
    priority_queue<pair<int, int>> pq;
    for(int i = 1; i <= n; i++){
        if(dist[i] <= d) continue;
        set<int> st;
        for(Edge e : adj[i]){
            int to = e.to;
            if(dist[to] <= d) continue;
            if(!st.count(e.to)) st.insert(e.to), pq.push({dist[to], i});
        }
    }
    while(!pq.empty()){
        int d = pq.top().first;
        int u = pq.top().second; pq.pop();
        if(dist[u] <= d) continue;
        dist[u] = d;
        cnt++;
        for(Edge e : adj[u]){
            int to = e.to;
            if(dist[to] <= d) continue;
            ans[cnt] = e.to;
            pq.push({d-1, to});
        }
    }
    cout << cnt << endl;
    for(int i = 1; i <= cnt; i++) cout << ans[i] << ' ';
}
