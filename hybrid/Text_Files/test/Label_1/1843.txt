#include <bits/stdc++.h>

using namespace std;

bool is_set(long long x, int p) {
  return ((x & (1LL << p)) != 0);
}

const int MOD = 1e9 + 7;
int main() {
  long long x;
  cin >> x;
  int n = 0;
  while (x > 0) {
    n++;
    x /= 2;
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if ((is_set(x, i) && !is_set(x, j)) || (!is_set(x, i) && is_set(x, j))) {
        ans++;
        ans %= MOD;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
