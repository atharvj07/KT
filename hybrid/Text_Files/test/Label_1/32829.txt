import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int a = input.nextInt();
        int colour[] = new int[n];
        Map<Integer, Integer> freq = new HashMap<>();
        for(int i=0; i<n; i++) {
            colour[i] = input.nextInt();
            freq.put(colour[i], freq.getOrDefault(colour[i], 0)+1);
        }
        for(int i=1; i<=1000000; i++) {
            if(i==a) continue;
            boolean ok = true;
            int ca = a;
            int cb = i;
            for(int j=0; j<n; j++) {
                if(colour[j]==a) { ca++; }
                else if(colour[j]==i) { cb++; }
                if(cb>=ca) { ok = false; break; }
            }
            if(ok) { System.out.println(i); return; }
        }
        System.out.println(-1);
    }
}
