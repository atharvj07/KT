#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int mod = 1e9 + 7;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 1; i < s.size(); i++) {
    if (s[i] == 'b' && s[i - 1] == 'a') {
      ans++;
      i++;
    }
  }
  cout << ans << '\n';
  return 0;
}
