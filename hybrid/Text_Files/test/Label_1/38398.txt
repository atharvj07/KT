import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    s = input().strip()
    a = []
    for ch in s:
        if ch == '>':
            a.append(1)
        elif ch == '<':
            a.append(-1)
        else:
            a.append(0)
    suf = [0] * n
    tot = 0
    for i in range(n - 1, -1, -1):
        tot += a[i]
        suf[i] = tot
    ans = 0
    pre = 0
    for i in range(n):
        if pre + suf[i] == 0:
            ans += 1
        pre += a[i]
    print(ans)
