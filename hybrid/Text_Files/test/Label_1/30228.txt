import java.util.*;

public class TreeMaxProduct {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            graph.get(u).add(v);
            graph.get(v).add(u);
        }

        long[] result = new long[1];
        dfs(graph, -1, 0, n, result);
        System.out.println(result[0]);
    }

    private static void dfs(List<List<Integer>> graph, int parent, int node, int n, long[] result) {
        int subtreeSize = 1;
        long subtreeProduct = 1;

        for (int child : graph.get(node)) {
            if (child != parent) {
                dfs(graph, node, child, n, result);
                subtreeSize += result[0];
                subtreeProduct = Math.max(subtreeProduct, (long) subtreeSize * (n - subtreeSize));
                result[0] = subtreeSize;
            }
        }

        result[0] = Math.max(result[0], (long) n * (n - subtreeSize));
        result[0] = Math.max(result[0], subtreeProduct);
    }
}
