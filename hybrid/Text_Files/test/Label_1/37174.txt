t = int(input())
for _ in range(t):
    n = int(input())
    degrees = [0] * (n + 1)
    for __ in range(n - 1):
        u, v = map(int, input().split())
        degrees[u] += 1
        degrees[v] += 1
    leaves = sum(1 for d in degrees if d == 1)
    buds = 0
    for d in degrees:
        if d == 2:
            buds += 1
    print(max(1, leaves - buds))
