def rotate90(matrix):
    return [list(reversed(x)) for x in zip(*matrix)]

def flip_vertical(matrix):
    return [row[::-1] for row in matrix]

def flip_horizontal(matrix):
    return matrix[::-1]

def are_maps_identical(map1, map2):
    if map1 == map2:
        return True
    for _ in range(3):
        map1 = rotate90(map1)
        if map1 == map2:
            return True
    map1 = flip_vertical(map1)
    if map1 == map2:
        return True
    for _ in range(3):
        map1 = rotate90(map1)
        if map1 == map2:
            return True
    return False

n = int(input().strip())
map1 = [list(input().strip()) for _ in range(n)]
map2 = [list(input().strip()) for _ in range(n)]

if are_maps_identical(map1, map2):
    print("Yes")
else:
    print("No")
