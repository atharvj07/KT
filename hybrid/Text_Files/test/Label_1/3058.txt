#include <bits/stdc++.h>
using namespace std;
int main() {
    long long int n, m, sum = 0;
    cin >> n >> m;
    long long int a[n + 1];
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        sum += a[i];
    }
    long long int b[m + 1];
    for (int i = 1; i <= m; i++) {
        cin >> b[i];
    }
    long long int p = 1, temp, cnt = 0;
    for (int i = 1; i <= m; i++) {
        temp = b[i] - cnt;
        while (temp > a[p]) {
            temp -= a[p];
            p++;
        }
        cnt = b[i];
        cout << p << " " << temp << endl;
    }
    return 0;
}
