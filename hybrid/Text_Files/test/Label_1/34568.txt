import java.util.*;
import java.math.*;

public class Main {

    static final long MOD = 1000000007L;

    static long binpow(long a, long b) {
        if (b == 0) return 1L;
        if (b == 1) return a;
        long t = binpow(a, b / 2) % MOD;
        if (b % 2 == 0) { return (t * t) % MOD; }
        return (((t * t) % MOD) * a) % MOD;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        String[] s = new String[n];
        for (int i = 0; i < n; i += 1) { s[i] = in.next(); }
        int[][] cnt = new int[m][26];
        for (int j = 0; j < m; j += 1) {
            for (int i = 0; i < n; i += 1) {
                cnt[j][s[i].charAt(j) - 'A'] += 1;
            }
        }
        long ans = 1L;
        for (int j = 0; j < m; j += 1) {
            long sum = 0;
            for (int i = 0; i < 26; i += 1) { sum += cnt[j][i]; }
            for (int i = 0; i < 26; i += 1) {
                if (cnt[j][i] > 0) {
                    ans = (ans * sum) % MOD;
                    ans = (ans * binpow(cnt[j][i], MOD - 2)) % MOD;
                    sum -= cnt[j][i];
                }
            }
        }
        System.out.println(ans);
    }
}
