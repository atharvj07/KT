import java.util.*;

public class Main {
    private static final long[] masks = new long[10];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            long m = 0;
            for (int j = 0; j < k; j++) {
                m |= 1L << sc.nextInt();
            }
            masks[i] = m;
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            long m = sc.nextInt();
            switch (op) {
                case 0:
                    System.out.println(test(m));
                    break;
                case 1:
                    set(m);
                    break;
                case 2:
                    clear(m);
                    break;
                case 3:
                    flip(m);
                    break;
                case 4:
                    System.out.println(all(m));
                    break;
                case 5:
                    System.out.println(any(m));
                    break;
                case 6:
                    System.out.println(none(m));
                    break;
                case 7:
                    System.out.println(count(m));
                    break;
                case 8:
                    System.out.println(val(m));
                    break;
            }
        }
    }

    private static int test(long m) {
        return (masks[0] & m) != 0 ? 1 : 0;
    }

    private static void set(long m) {
        masks[0] |= m;
    }

    private static void clear(long m) {
        masks[0] &= ~m;
    }

    private static void flip(long m) {
        masks[0] ^= m;
    }

    private static int all(long m) {
        return (masks[0] & m) == m ? 1 : 0;
    }

    private static int any(long m) {
        return (masks[0] & m) != 0 ? 1 : 0;
    }

    private static int none(long m) {
        return (masks[0] & m) == 0 ? 1 : 0;
    }

    private static int count(long m) {
        return Long.bitCount(masks[0] & m);
    }

    private static long val(long m) {
        return masks[0] & m;
    }
}
