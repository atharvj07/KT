import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int V = in.nextInt(), E = in.nextInt();
        long[][] d = new long[V][V];
        for (int i = 0; i < V; i++)
            Arrays.fill(d[i], Long.MAX_VALUE);
        for (int i = 0; i < E; i++)
            d[in.nextInt()][in.nextInt()] = in.nextLong();
        long[] cost = new long[V];
        for (int k = 0; k < V; k++)
            for (int i = 0; i < V; i++)
                for (int j = 0; j < V; j++)
                    d[i][j] = Math.min(d[i][j], d[i][k] + d[k][j]);
        int negcycle = 0;
        for (int i = 0; i < V; i++)
            if (d[i][i] < 0)
                negcycle = 1;
        if (negcycle == 1)
            System.out.println("NEGATIVE CYCLE");
        else
            for (int i = 0; i < V; i++) {
                System.out.print(d[i][0]);
                for (int j = 1; j < V; j++)
                    System.out.print(" " + (d[i][j] == Long.MAX_VALUE ? "INF" : d[i][j]));
                System.out.println();
            }
    }
}
