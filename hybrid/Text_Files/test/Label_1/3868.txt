#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    bool ok = true;
    for (int i = 0; i < n - 1; i++)
      if (a[i] != a[i + 1]) ok = false;
    cout << (ok ? "NO" : "YES") << endl;
  }
  return 0;
}
