#include <iostream>
#include <cstring>

const int MOD = 1e9 + 7;

int main() {
  char s[1000005];
  std::cin >> s;

  int n = strlen(s);

  int pref[n + 1];
  memset(pref, 0, sizeof(pref));

  for (int i = 1; i <= n; ++i) {
    if (s[i - 1] == '*') pref[i] = pref[i - 1] + 1;
    else pref[i] = pref[i - 1];
  }

  int dp[n + 1];
  memset(dp, 0, sizeof(dp));
  dp[0] = 1;

  for (int i = 1; i <= n; ++i) {
    if (s[i - 1] == '?') {
      if (pref[i] > 2) dp[i] = 0;
      else if (pref[i] == 2) dp[i] = dp[i - 1] * 2 % MOD;
      else if (pref[i] == 1) dp[i] = dp[i - 1] * 3 % MOD;
      else dp[i] = dp[i - 1];
    } else {
      if ((s[i - 1] - '0') != pref[i] - pref[i - 1]) dp[i] = 0;
      else dp[i] = dp[i - 1];
    }
  }

  std::cout << dp[n] << std::endl;

  return 0;
}
