#include <bits/stdc++.h>
using namespace std;
const int maxn=1e5+1;
long long int u,n,a[maxn],dp[maxn][4];
int main(){
	cin>>n>>u;
	for(int i=1;i<=n;++i)cin>>a[i];
	for(int i=0;i<=n;++i)for(int j=0;j<=4;++j)dp[i][j]=1e18;
	dp[1][1]=0;
	for(int i=1;i<=n;++i)
		for(int j=1;j<=3;++j)
			for(int k=i+1;k<=n;++k)
				if(a[k]-a[i]<=u)
					dp[k][j+1]=min(dp[k][j+1],dp[i][j]+a[k]-a[i]);
	if(dp[n][3]==1e18)return cout<<-1<<endl,0;
	cout<<fixed<<setprecision(9)<<(double)(a[n]-a[n-1])/(double)(a[n]-a[1])<<endl;
}
