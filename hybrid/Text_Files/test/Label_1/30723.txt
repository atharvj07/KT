import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n - 2; i++) {
            if (a[i] == a[i + 1]) {
                continue;
            }
            int j = i + 1;
            int k = n - 1;
            while (j < k) {
                if (a[i] * a[j] * a[k] < a[i] * a[j + 1] * a[k - 1]) {
                    k--;
                } else {
                    j++;
                }
                ans += j - i - 1;
            }
        }
        System.out.println(ans);
    }
}
