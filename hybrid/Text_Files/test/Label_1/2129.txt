#include<bits/stdc++.h>
using namespace std;
#define ll long long

map<ll,ll> mp;

ll bigmod(ll a, ll b, ll m){
    if(b==0)return 1;
    ll x=bigmod(a,b/2,m);
    x=(x*x)%m;
    if(b%2)x=(x*a)%m;
    return x;
}

ll inverse(ll a, ll m){
    return bigmod(a,m-2,m);
}

int main(){
    ll a,b,p,x,ans=0;
    cin>>a>>b>>p>>x;
    ll x0 = bigmod(a,x,p);
    if(x0==b){
        cout<<1<<endl;
        return 0;
    }
    for(ll i=1; i<=x; i++){
        ll a1 = bigmod(a,i,p);
        ll b1 = (b*inverse(x0,p))%p;
        if(a1==b1){
            mp[i]+=1;
            if(x%i==0){
                ll rem = x/i;
                mp[rem]+=1;
            }
        }
    }
    for(auto x:mp)ans+=x.second;
    cout<<ans<<endl;
    return 0;
}
