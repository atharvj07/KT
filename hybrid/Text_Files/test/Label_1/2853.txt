#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(n);
  for (auto &ai : a) cin >> ai;
  ll m = 0;
  ll ans = 0;
  for (ll i = 0, j = 0, t = 0; i < n; ++i) {
    while (j < n && j - i + 1 <= k) {
      j += 1;
      t += a[j - 1];
      m += (ll)(t <= 100 * (j - i)) * (ll)(t >= 99 * (j - i));
    }
    ans += (ll)(t - a[i] <= 100 * (j - i) && t - a[i] >= 99 * (j - i));
    t -= a[i];
    m -= (ll)(t <= 100 * (j - i - 1)) * (ll)(t >= 99 * (j - i - 1));
  }
  cout << ans << endl;
  return 0;
}
