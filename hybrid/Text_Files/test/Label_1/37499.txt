import sys

input = sys.stdin.readline
n, q = map(int, input().split())
a, t = [int(x) for x in input().split()], [0 for i in range(q)]
for i in range(q):
    t[i] = sum(a)
    for j in range(n):
        if a[j] <= 0:
            continue
        t[i] = min(t[i], (a[j] - t[i - j - 1]) / q)
    for j in range(n):
        a[j] -= q * int(t[i])
    print(sum(1 for ai in a if ai > 0))
