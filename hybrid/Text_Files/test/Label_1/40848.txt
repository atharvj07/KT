from collections import deque
n = int(input())
pos = [None] + [[] for i in range(2*n)]
for i in range(n):
  l, r = map(int, input().split())
  pos[l].append(i)
  pos[r].append(-i)
stack = deque()
lst = []
ok = True
for i in range(1, 2*n+1):
  if ok:
    while pos[i]:
      j = pos[i].pop()
      if j < 0:
        if i - (-j) not in pos[-j]:
          ok = False
        else:
          pos[-j].remove(i - (-j))
        stack.append(lst.pop())
      else:
        stack.append(j)
        lst.append(i)
    if len(stack) % 2:
      ok = False
    else:
      while stack:
        j = stack.pop()
        if j < 0:
          lst.append(i + (-j))
        else:
          i = j
if ok:
  print(''.join(')' if i > 0 else '(' for i in lst))
else:
  print('IMPOSSIBLE')
