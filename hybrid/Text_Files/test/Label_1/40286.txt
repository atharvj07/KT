def max_possible_number(n, k, digit_states):
    # Encoding of all digits from 0 to 9 with their respective segment's state
    digits = ["1110111", "0010010", "1011101", "1011011", "0111010", "1101011", "1101111", "1010010", "1111111", "1111011"]
    # Calculate the cost to turn a current state into a digit 0-9
    cost = [[0 for _ in range(10)] for _ in range(n)]
    for i in range(n):
        for j in range(10):
            for bit in range(7):
                if digit_states[i][bit] == '1' and digits[j][bit] == '0':
                    cost[i][j] = -1
                    break
                if digit_states[i][bit] == '0' and digits[j][bit] == '1':
                    cost[i][j] += 1
            if cost[i][j] != -1 and cost[i][j] > k:
                cost[i][j] = -1

    # DP to check if it's possible to obtain a valid number
    dp = [[False for _ in range(k+1)] for _ in range(n+1)]
    dp[n][0] = True
    for i in range(n-1, -1, -1):
        for used in range(k+1):
            for d in range(10):
                if cost[i][d] != -1 and used + cost[i][d] <= k and dp[i+1][used]:
                    dp[i][used + cost[i][d]] = True

    # Construct the maximum number
    if not dp[0][k]:
        return -1
    ans = ""
    for i in range(n):
        for d in range(9, -1, -1):
            if cost[i][d] != -1 and k - cost[i][d] >= 0 and dp[i+1][k-cost[i][d]]:
                ans += str(d)
                k -= cost[i][d]
                break
    return ans

# Reading input
n, k = map(int, input().split())
digit_states = [input() for _ in range(n)]

# Output the result
print(max_possible_number(n, k, digit_states))
