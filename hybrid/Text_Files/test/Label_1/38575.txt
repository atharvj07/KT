import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, x = map(int, input().split())
    s = input()
    if (x + n) % 2:
        print(-1)
    else:
        z = s.count('0')
        o = s.count('1')
        ans = 0
        cnt = (x + n) // 2 - o
        for c in s:
            if c == '0':
                cnt -= 1
            else:
                cnt += 1
            if not cnt:
                ans += 1
        print(ans * (10 ** 9 + 7))
