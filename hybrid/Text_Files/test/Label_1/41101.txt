from sys import stdin
from bisect import bisect_right as upper_bound

n = int(stdin.readline())
sol_cnt = []
for _ in range(n):
    sol_cnt.append([0] * 5)
    for i in map(int, stdin.readline().split()[1:]):
        if i != -1: sol_cnt[-1][i // 25] += 1
max_pts = [500, 1000, 1500, 2000, 2500, 3000]
vasya = [0, 0]
petya = [0, 0]
for prob, sc in enumerate(sol_cnt):
    sc.sort()
    idx = upper_bound(sc, n / 4) - 1
    if idx < 0:
        petya[prob] = max_pts[5]
    else:
        petya[prob] = max_pts[idx]
for prob, sc in enumerate(sol_cnt[1:]):
    for j in range(n):
        if sc[j] == 0: vasya[prob] += max_pts[j]
print(-1 if vasya[0] <= petya[0] else 1)
