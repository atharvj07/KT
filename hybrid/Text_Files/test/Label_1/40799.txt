import sys

n, m, k = map(int, sys.stdin.readline().split())
p = list(map(int, sys.stdin.readline().split()))
a = [list(map(int, sys.stdin.readline().split())) for i in range(n)]

t = 0
for ord in a:
    for item in ord:
        t += p[item - 1]
        for j in range(item, k):
            p[j - 1], p[j] = p[j], p[j - 1]
        k -= 1

print(t)
