import sys

n = int(input())
s = input()
for i in range(1, n):
    if s[i - 1] != s[i]:
        if ((s[i - 1] == '(' and s[i] == ')') or (s[i - 1] == ')' and (i + 1 < n and s[i + 1] == '('))) and (i == 1 or s[i - 2] == '(' and s[i + 1] == ')'):
            print("Yes")
            sys.exit()
print("No")
