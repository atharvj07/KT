#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    for (int x = 0; x < t; x++) {
        int n;
        cin >> n;
        vector<int64_t> a(n);
        int64_t g = 0;
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            g = __gcd(g, a[i]);
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != g) { ans = 1; }
        }
        cout << ans << endl;
    }
    return 0;
}
