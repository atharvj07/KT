import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int minHeight = 0;
        if (m > 0) {
            minHeight = 3 * m;
        }
        if (n > 0) {
            if (minHeight == 0) {
                minHeight = 2 * n;
            } else {
                while (minHeight % 2 == 0 && n > 0) {
                    minHeight++;
                    n--;
                }
            }
        }
        System.out.println(minHeight);
    }
}
