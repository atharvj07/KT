import java.util.*;

public class TreeQueries {

    static class Node {
        int id;
        List<Node> children;

        Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Node root = new Node(1);
        Map<Integer, Node> nodes = new HashMap<>();
        nodes.put(1, root);

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();

            Node parent = nodes.get(u);
            Node child = nodes.get(v);

            if (child == null) {
                child = new Node(v);
                nodes.put(v, child);
            }

            parent.children.add(child);
        }

        for (int i = 0; i < m; i++) {
            int k = scanner.nextInt();
            int[] query = new int[k];

            for (int j = 0; j < k; j++) {
                query[j] = scanner.nextInt();
            }

            System.out.println(solve(root, query) ? "YES" : "NO");
        }
    }

    private static boolean solve(Node node, int[] query) {
        Set<Integer> visited = new HashSet<>();
        visited.add(node.id);

        for (int v : query) {
            if (node.id == v) {
                continue;
            }

            boolean found = false;

            for (Node child : node.children) {
                if (child.id == v || dfs(child, v, visited)) {
                    found = true;
                    break;
                }
            }

            if (!found) {
                return false;
            }
        }

        return true;
    }

    private static boolean dfs(Node node, int target, Set<Integer> visited) {
        if (node.id == target) {
            return true;
        }

        if (visited.contains(node.id)) {
            return false;
        }

        visited.add(node.id);

        for (Node child : node.children) {
            if (dfs(child, target, visited)) {
                return true;
            }
        }

        return false;
    }
}
