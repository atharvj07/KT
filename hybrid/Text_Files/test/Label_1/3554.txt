#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, t;
  cin >> n >> t;
  string s;
  cin >> s;
  int point = -1;
  for (int i = 0; i < n; i++) {
    if (s[i] == '.') {
      point = i;
      break;
    }
  }
  if (t >= n - point) {
    cout << s << "\n";
    return 0;
  }
  long long sum = 0;
  for (int i = 0; i < n; i++) {
    sum = sum * 10 + (s[i] - '0');
    if (i == point) continue;
    if (t == 0) break;
    t--;
    int bit = (s[i + 1] - '0') * 10;
    if (sum % 10 < 5) {
      sum = sum / 10;
    } else if (bit >= 5) {
      sum = sum / 10 + 1;
    } else {
      sum = sum / 10;
    }
  }
  cout << fixed << setprecision(0) << (double)sum / 10 << "\n";
  return 0;
}
