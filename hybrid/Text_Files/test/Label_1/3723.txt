#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  int a[26] = {};
  vector<int> q;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '?') q.push_back(i);
    if (s[i] >= 'A' && s[i] <= 'Z') a[s[i] - 'A'] = 1;
  }
  int mx = *max_element(a, a + 26);
  if (mx < 26 - q.size()) {
    cout << -1 << endl;
    return 0;
  }
  for (int i = 0; i < q.size(); i++) {
    for (int j = 0; j < 26; j++) {
      if (!a[j]) {
        a[j] = 1;
        s[q[i]] = char(j + 'A');
        break;
      }
    }
  }
  cout << s << endl;
  return 0;
}
