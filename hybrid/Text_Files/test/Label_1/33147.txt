import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int[] time = new int[4];
		for (int i = 0; i < 4; i++)
			time[i] = scan.nextInt();
		String a = scan.nextLine();
		a = scan.nextLine();
		String b = scan.nextLine();
		int n = a.length();
		int[][] dp = new int[n + 1][n + 1];
		for (int i = 0; i <= n; i++)
			dp[i][i] = 0;
		for (int l = 2; l <= n; l++)
			for (int i = 0; i + l - 1 < n; i++) {
				dp[i][i + l - 1] = Integer.MAX_VALUE;
				for (int j = i; j + 1 <= i + l - 1; j++)
					dp[i][i + l - 1] = Math.min(dp[i][i + l - 1], dp[i][j] + dp[j + 1][i + l - 1] + ((a.charAt(i) == b.charAt(i + l - 1)) ? time[2] : time[3]));
				if (l % 2 == 0)
					for (int j = i; j + l / 2 - 1 < i + l - 1; j++)
						dp[i][i + l - 1] = Math.min(dp[i][i + l - 1], dp[i][j] + dp[j + 1][i + l - 1] + time[0]);
				for (int j = i; j + l - 1 < i + l; j++)
					dp[i][i + l - 1] = Math.min(dp[i][i + l - 1], dp[i][j] + dp[j + 1][i + l - 1] + ((a.charAt(j) == b.charAt(i + l - 1)) ? time[1] : time[3]));
				if (l % 2 == 0)
					for (int j = i; j + l / 2 < i + l - 1; j++)
						dp[i][i + l - 1] = Math.min(dp[i][i + l - 1], dp[i][j] + dp[j + 1][i + l - 1] + time[0]);
			}
		System.out.println(dp[0][n - 1]);
	}
}
