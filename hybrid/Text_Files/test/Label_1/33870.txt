import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int p = input.nextInt(), q = input.nextInt(), r = input.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = input.nextInt();
    }
    long max = Long.MIN_VALUE;
    for (int i = 0; i < n; i++) {
      long sum = p * a[i];
      for (int j = i; j < n; j++) {
        sum += q * a[j];
        for (int k = j; k < n; k++) {
          sum += r * a[k];
          max = Math.max(max, sum);
        }
      }
    }
    System.out.println(max);
  }
}
