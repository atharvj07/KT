def find_k(n, m, battles):
    graph = [[] for _ in range(n+1)]
    in_degree = [0] * (n+1)
    for u, v in battles:
        graph[u].append(v)
        in_degree[v] += 1
    queue = [i for i in range(1, n+1) if in_degree[i] == 0]

    count, last_battle = 0, 0
    while queue:
        if len(queue) > 1:
            return -1
        current = queue.pop(0)
        count += 1
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
                last_battle = count
    return last_battle if count == n else -1

n, m = map(int, input().split())
battles = [list(map(int, input().split())) for _ in range(m)]
print(find_k(n, m, battles))
