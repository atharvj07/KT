#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define vi vector<int>
#define ll long long
int n;
bool G[55][55], vis[55][55], f, a[55][55];
char c[55][55];
ll dr[]={-1,0,1,-1,1,-1,0,1};
ll dc[]={-1,-1,-1,0,0,1,1,1};
void dfs(int x, int y){
  if(f) return;
  vis[x][y]=1;
  for(int i=0;i<8;++i){
    int nx=x+dr[i], ny=y+dc[i];
    if(0<=nx && nx<n && 0<=ny && ny<n){
      if(!vis[nx][ny]){
        if(c[nx][ny]=='o' || (c[nx][ny]=='x' && !G[nx][ny]))
          dfs(nx, ny);
      }
      else if(!a[nx][ny])
        f=true;
    }
  }
}
int main(){
  cin>>n;
  for(int i=0;i<n;++i){
    for(int j=0;j<n;++j){
      cin>>c[i][j];
      if(c[i][j]=='o')
        G[i][j]=1;
    }
  }
  bool ok=0;
  for(int i=0;i<n;++i){
    for(int j=0;j<n;++j){
      if(G[i][j] && !ok){
        memset(vis, 0, sizeof vis);
        dfs(i, j);
        ok=1;
      }
    }
  }
  if(f) cout<<"NO\n";
  else{
    cout<<"YES\n";
    for(int i=0;i<n;++i){
      for(int j=0;j<n;++j){
        for(int k=0;k<8;++k){
          int nx=i+dr[k], ny=j+dc[k];
          if(0<=nx && nx<n && 0<=ny && ny<n)
            a[nx][ny]=1;
        }
      }
    }
    for(int i=0;i<n;++i){
      for(int j=0;j<n;++j){
        if(a[i][j] && !G[i][j])
          cout<<'x';
        else if(!a[i][j] && G[i][j])
          cout<<'o';
        else
          if(c[i][j]=='.')
            cout<<'.';
          else
            cout<<'o';
      }
      cout<<endl;
    }
  }
  return 0;
}
