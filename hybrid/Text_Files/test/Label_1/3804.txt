#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n;
    cin >> n;
    if (n <= 10) {
        cout << (n / 2) * ((n + 1) / 2) << endl;
        return 0;
    }
    int x = n / 10, y = n % 10;
    long long ans = (9LL * x + y - 1LL) * x;
    if (ans > 9LL * (x - 1LL) + 9LL)
        ans = ans - 9LL * y + 9LL;
    cout << ans << endl;
    return 0;
}
