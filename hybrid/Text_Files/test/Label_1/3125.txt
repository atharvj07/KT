#include <bits/stdc++.h>
using namespace std;
#define FIO ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
#define ll long long
const int N = 1000 + 10;
vector<int> G[N];
int vis[N];
void dfs(int u) {
  vis[u] = 1;
  for (auto v : G[u]) {
    if (vis[v] == 0)
      dfs(v);
  }
}
int main() {
  FIO;
  int n, m;
  cin >> n >> m;
  while (m--) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    G[u].push_back(v);
    G[v].push_back(u);
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (vis[i] == 0) {
      dfs(i);
      ans++;
    }
  }
  cout << ans - 1 << endl;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (vis[i] != vis[j]) {
        if (cnt == ans - 1)
          break;
        cout << i + 1 << " " << j + 1 << endl;
        cnt++;
      }
    }
  }
  return 0;
}
