#include <iostream>

using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;
    if (n > (1LL << k) - 1) {
        cout << "NO\n";
    } else {
        cout << "YES\n";
        for (int i = 0; i < k; ++i) {
            long long a = (n >> i) & 1;
            if (i > 0) a = max(a * 2 + 1, a);
            cout << a << ' ';
        }
        cout << '\n';
    }
}
