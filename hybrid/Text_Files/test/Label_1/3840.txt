#include <vector>
#include <cstdio>

void eval(std::vector<int>&a, std::vector<int>&b, int k) {
	int mx = 0;
	for (int i : a) mx = std::max(mx, i);
	for (int i : b) mx = std::max(mx, i);
	int m = 0;
	int ans = 0;
	for (int len = mx; len >= 1; --len) {
		int sum = 0;
		for (int i = 0; i < (int)a.size(); ++i) {
			if (a[i] >= len) ++sum;
			if (b[i] >= len) ++sum;
		}
		int got = sum / 2;
		if (got * len >= k) {
			if (got > m) ans += (got - m) * len;
			break;
		}
		m = got;
	}
	printf("%d\n", ans);
}

int main() {
	int t;
	scanf("%d", &t);
	while (t--) {
		int n, k;
		scanf("%d %d", &n, &k);
		std::vector<int>a(n), b(n);
		for (int i = 0; i < n; ++i) scanf("%d %d", &a[i], &b[i]);
		eval(a, b, k);
	}
	return 0;
}
