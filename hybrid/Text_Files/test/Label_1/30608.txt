import java.util.*;
import java.io.*;

public class Main {
    static long mod = (long)1e9 + 7;
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] in = br.readLine().split(" ");
        int n = Integer.parseInt(in[0]);
        int m = Integer.parseInt(in[1]);
        long[] arr = new long[n];
        in = br.readLine().split(" ");
        for(int i=0;i<n;i++)
            arr[i] = Long.parseLong(in[i]);
        long ansA = 0;
        long ansB = 0;
        long ansF = 0;
        long ansS = 0;
        for(int a=1;a<=m;a++){
            for(int b=1;b<=m;b++){
                long A_won = 0;
                long B_won = 0;
                for(long x: arr){
                    if(A_won==0 && x-a>=0)
                        A_won = 1;
                    if(B_won==0 && x-b>=0)
                        B_won = 1;
                }
                if(A_won==1 && B_won==0)
                    ansA++;
                if(A_won==0 && B_won==1)
                    ansB++;
                if(A_won==1 && B_won==1)
                    ansF++;
                if(A_won==0 && B_won==0){
                    ansS++;
                    ansS %= mod;
                }
            }
        }
        System.out.println(ansA+" "+ansB+" "+ansF+" "+ansS);
    }
}
