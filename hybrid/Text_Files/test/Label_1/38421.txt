import sys
for _ in range(int(input())):
    n = int(input())
    ans = []
    for i in range(n - 2, 1, -1):
        ans.append([i, i + 1])
        ans.append([i + 1, i])
    if n > 3:
        ans.append([3, n])
        ans.append([n, 3])
    ans.append([2, n])
    ans.append([n, 2])
    print(len(ans))
    for e in ans:
        print(*e)
