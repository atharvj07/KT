import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long ans = (long)Math.ceil(7.0 / k) * k;
        if (ans == 7) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
