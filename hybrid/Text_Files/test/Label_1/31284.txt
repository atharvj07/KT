import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        int mod = 1000000007;
        long n = new Scanner(System.in).nextLong();
        long ans = 1L;
        for (long i = 1L; i <= n; i++) {
            ans = ans * i % mod;
        }
        System.out.println(ans);
    }
}
