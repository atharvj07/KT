for _ in range(int(input())):
    n, m = map(int, input().split())
    s = input()
    x, y = -1, -1
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            px, py = i, j
            cur = 0
            for ch in s:
                if ch == 'L' and py > 1:
                    py -= 1
                    cur += 1
                elif ch == 'R' and py < m:
                    py += 1
                    cur += 1
                elif ch == 'U' and px > 1:
                    px -= 1
                    cur += 1
                elif ch == 'D' and px < n:
                    px += 1
                    cur += 1
            if cur > 0:
                x, y = i, j
    print(x, y)
