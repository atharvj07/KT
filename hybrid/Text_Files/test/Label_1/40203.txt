MOD = 10**9 + 7
s = input()
n = len(s)
lnum = [0] * n
rnum = [0] * n
dp = [0] * n
for i in range(n):
    if i > 0: lnum[i] = lnum[i-1]
    if s[i] == 'a': lnum[i] += 1
for i in range(n-1,-1,-1):
    if i < n-1: rnum[i] = rnum[i+1]
    if s[i] == 'b': rnum[i] += 1
for i in range(n):
    if s[i] == 'a':
        dp[i] = (dp[i] + 1 + (lnum[i] * rnum[i] % MOD)) % MOD
    else:
        dp[i] = dp[i] + dp[i-1]
print(dp[-1])
