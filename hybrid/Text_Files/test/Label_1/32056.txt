import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();

        // dp[i][j] = number of ways to write the numbers on the houses' plaques so that:
        // i = current house number
        // j = 0 if the penguin can walk to house number 1, 1 otherwise
        long[][] dp = new long[n + 1][2];

        // Base case: the penguin can walk to house number 1 from any house indexed from 1 to k, inclusive
        for (int i = 1; i <= k; i++) {
            dp[i][0] = 1;
        }

        // Iterate over the remaining houses
        for (int i = k + 1; i <= n; i++) {
            // The penguin cannot walk to house number 1 from any house indexed from k + 1 to n, inclusive
            dp[i][1] = 1;

            // Iterate over all possible numbers that can be written on the plaque of house i
            for (int j = 1; j <= n; j++) {
                // If the penguin can walk to house number 1 from house j
                if (dp[j][0] > 0) {
                    // The penguin can also walk to house number 1 from house i
                    dp[i][0] += dp[j][0];
                }
            }
        }

        // Output the number of ways to write the numbers on the houses' plaques modulo 1000000007 (10^9 + 7)
        System.out.println(dp[n][0] % 1000000007);
    }
}
