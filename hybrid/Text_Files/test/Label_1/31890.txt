import java.util.ArrayDeque;
import java.util.Arrays;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static void addEdge(int u, int v, int w, int[][] g) {
        g[u][v] = g[v][u] = w;
    }

    public static int minSpanningTree(int[][] g, int n) {
        int[] key = new int[n + 1];
        int[] parent = new int[n + 1];
        boolean[] mstSet = new boolean[n + 1];

        for (int i = 1; i <= n; i++)
            key[i] = Integer.MAX_VALUE;

        key[1] = 0;
        parent[1] = -1;

        for (int count = 0; count < n - 1; count++) {
            int u = -1;
            for (int i = 1; i <= n; i++)
                if (!mstSet[i] && (u == -1 || key[i] < key[u]))
                    u = i;

            mstSet[u] = true;

            for (int v = 1; v <= n; v++)
                if (g[u][v] != 0 && !mstSet[v] && g[u][v] < key[v]) {
                    parent[v] = u;
                    key[v] = g[u][v];
                }
        }

        int sum = 0;
        for (int i = 2; i <= n; i++)
            sum += g[parent[i]][i];
        return sum;
    }

    public static void main(String[] z) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] g = new int[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            addEdge(u, v, w, g);
        }

        int oddCount = 0;
        Queue<int[]> q = new ArrayDeque<>();
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (g[i][j] == 0) {
                    q.add(new int[]{i, j});
                    oddCount++;
                }
            }
        }

        oddCount %= 2;

        if (oddCount == 0) {
            System.out.println(minSpanningTree(g, n));
            return;
        }

        int xor = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (g[i][j] == 0) {
                    xor ^= 1;
                    addEdge(i, j, 1, g);
                }
            }
        }

        if (xor == 0)
            System.out.println(minSpanningTree(g, n));
        else
            System.out.println(-1);
    }
}
