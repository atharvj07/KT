#include <bits/stdc++.h>
using namespace std;

#define pb push_back
#define mp make_pair
#define ll long long
#define ld long double
const int NMAX = 1e5 + 10;

vector<ll> a, b;
ll n, m;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    a.resize(n);
    for (auto &i : a) cin >> i;
    cin >> m;
    b.resize(m);
    for (auto &i : b) cin >> i;

    ll ans = 0;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    for (auto &i : a) {
      int x = lower_bound(b.begin(), b.end(), -i) - b.begin();
      int y = upper_bound(b.begin(), b.end(), -i) - b.begin();
      ans += y - x;
    }
    cout << ans << '\n';
    a.clear();
    b.clear();
  }
  return 0;
}
