#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    int cnt = 0;
    for (int i = 1; i <= 5; i++) {
        if ((sum + i) % (n + 1) != 0) cnt++;
    }
    cout << cnt << endl;
    return 0;
}
