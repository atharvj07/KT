#include <bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
typedef vector<int> vi;

const int N = 1e3+1, M = 2e3+1;

int n, m, a[N], par[N], comp[N], sz[N], chains[N][N];

int find(int x) { return (par[x] == x ? x : par[x] = find(par[x])); }
void join(int a, int b) { a = find(a); b = find(b); if (a == b) return; if (sz[a] < sz[b]) swap(a, b); par[b] = a; sz[a] += sz[b]; }

bool ok()
{
    iota(par, par+n+1, 0);
    memset(sz, 0, sizeof sz);
    memset(chains, 0, sizeof chains);
    for (int i=1; i<=n; i++) {
        for (int j=i+1; j<=n; j++) {
            chains[i][j] = chains[j][i] = a[i] + a[j];
        }
    }

    for (int i=0; i<m; i++) {
        int u, v; cin >> u >> v;
        u = find(u); v = find(v);
        if (u == v) return false;
        join(u, v);
        chains[u][v] = chains[v][u] = 0;
    }

    for (int i=1; i<=n; i++) comp[find(i)] = i;
    for (int i=1; i<=n; i++) {
        int p = find(i);
        for (int j=1; j<=n; j++) {
            if (i == j || p == find(j)) continue;
            if (chains[i][j] == 0) return false;
        }
    }

    return true;
}

int main()
{
    int t; cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i=1; i<=n; i++) cin >> a[i];
        if (!ok()) {
            cout << -1 << '\n';
            continue;
        }
        ll ans = 0;
        for (int i=1; i<=n; i++) {
            for (int j=i+1; j<=n; j++) {
                if (chains[i][j]) {
                    cout << i << ' ' << j << '\n';
                    ans += chains[i][j];
                }
            }
        }
        cout << ans << '\n';
    }
}
