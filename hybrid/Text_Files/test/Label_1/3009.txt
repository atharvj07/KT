#include <bits/stdc++.h>
using namespace std;
#define int long long
vector<int> vx {0, -1, -1, 0, 1, 1};
vector<int> vy {0, 1, 0, -1, 1, -1};
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<string> arr(n + 2);
        for (int i = 0; i < n + 2; i++) arr[i] = '.' + string(m + 2, '.') + '.';
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> arr[i][j];
            }
        }
        bool ok = true;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (arr[i][j] == '*') {
                    ok = false;
                    for (int c = 0; c < 6; c++) {
                        int x = i + vx[c], y = j + vy[c];
                        if (arr[x][y] == '*') {
                            ok = true;
                            int len = 0;
                            while (1) {
                                x += vx[c], y += vy[c];
                                if (arr[x][y] == '.') break;
                                len++;
                            }
                            x = i + vx[c], y = j + vy[c];
                            while (1) {
                                x -= vx[c], y -= vy[c];
                                if (arr[x][y] == '.') break;
                                len++;
                            }
                            if (len <= k) {
                                ok = false;
                                break;
                            }
                        }
                    }
                    if (!ok) break;
                }
            }
            if (!ok) break;
        }
        cout << (ok ? "YES" : "NO") << '\n';
    }
}
