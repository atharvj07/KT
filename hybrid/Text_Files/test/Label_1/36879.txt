d = {'U': (1, 0), 'R': (0, 1), 'D': (-1, 0), 'L': (0, -1)}
n = int(input())
s = input()
p = [d[c] for c in s]
su = [0] * 4
cnt = 0
for x, y in p:
    su[0] += x
    su[1] += y
    su[2] -= x
    su[3] -= y
    cnt += sum(su[i] == 0 for i in range(4)) * (sum(su[i] == 0 for i in range(4)) - 1) // 2
print(cnt)
