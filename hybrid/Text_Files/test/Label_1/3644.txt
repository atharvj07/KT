#include <iostream>
#include <deque>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;
    deque<char> dq;
    for (auto c : s) dq.push_back(c);
    string u;
    while (!dq.empty()) {
        if (dq.front() < dq.back()) {
            u += dq.front();
            dq.pop_front();
        } else {
            u += dq.back();
            dq.pop_back();
        }
    }
    cout << u << endl;
    return 0;
}
