import java.util.Scanner;

public class BitwiseXOR {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long s = scanner.nextLong();
        long x = scanner.nextLong();

        // Check if a and b are both odd or both even
        if ((s & 1) != (x & 1)) {
            System.out.println(0); // No solutions
            return;
        }

        // Calculate the range of possible values for a
        long aMin = (s - x) / 2;
        long aMax = (s + x) / 2;

        // Check if the range is valid
        if (aMin < 0 || aMax < 0 || aMax - aMin > s) {
            System.out.println(0); // No solutions
            return;
        }

        // Calculate the number of possible values for a
        long numSolutions = aMax - aMin + 1;

        System.out.println(numSolutions);
    }
}
