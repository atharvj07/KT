n = int(input())
balance = {}
for _ in range(n):
    s = input()
    b = 0
    min_b = 0
    for c in s:
        if c == '(':
            b += 1
        else:
            b -= 1
        min_b = min(min_b, b)
    if b >= 0 and min_b == 0:
        balance[b] = balance.get(b, 0) + 1
    elif b < 0 and min_b == b:
        balance[b] = balance.get(b, 0) + 1

pairs = balance.get(0, 0) // 2
for k in balance:
    if k > 0:
        pairs += min(balance[k], balance.get(-k, 0))

print(pairs)
