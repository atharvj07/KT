import java.util.*;

public class Main {
  
  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int n = s.nextInt();
    long[] a = new long[n];
    long sum = 0L;
    for (int i = 0; i < n; i++) {
      a[i] = s.nextLong();
      sum += a[i];
    }
    if (sum != 0) {
      System.out.println(-1);
    } else {
      long ans = 0L;
      long cur = Integer.MAX_VALUE;
      int idx = -1;
      for (int i = 0; i < n; i++) {
        if (a[i] != 0L) {
          if (idx != -1) {
            ans += Math.min(cur + a[i], -a[idx]);
            cur = 0L;
          }
          idx = i;
          cur = a[i];
        }
      }
      System.out.println(ans);
    }
  }
}
