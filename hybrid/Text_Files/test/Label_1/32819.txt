import java.util.Scanner;

public class TownTraversal {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = scanner.nextInt();
            roads[i][1] = scanner.nextInt();
        }

        boolean[][] graph = new boolean[5][5];
        for (int[] road : roads) {
            graph[road[0]][road[1]] = true;
            graph[road[1]][road[0]] = true;
        }

        boolean[] visited = new boolean[5];
        if (dfs(graph, visited, 1)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean dfs(boolean[][] graph, boolean[] visited, int node) {
        if (visited[node]) {
            return true;
        }
        visited[node] = true;
        for (int i = 1; i < 5; i++) {
            if (graph[node][i] && !dfs(graph, visited, i)) {
                return false;
            }
        }
        return true;
    }
}
