#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<vector<int>> dp(n + 1, vector<int>(101, INT_MAX));
  dp[0][0] = 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j <= 100; j++)
      for (int k = 0; k <= 100; k++)
        if (dp[i][j] != INT_MAX) dp[i + 1][k] = min(dp[i + 1][k], dp[i][j] + abs(k - a[i]));
  int ans = INT_MAX, t;
  for (int i = 1; i <= 100; i++) {
    if (dp[n][i] < ans) ans = dp[n][i], t = i;
  }
  cout << t << ' ' << ans << '\n';
}
