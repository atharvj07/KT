#include <bits/stdc++.h>

using namespace std;

const int maxN = 3e5 + 5;

int n, k;
vector<int> pre[maxN], ord;
vector<pair<int, int>> special;

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        if (x == 0) {
            ord.push_back(i);
            continue;
        }
        pre[i].push_back(x);
    }
    for (int i = 1; i <= k; i++) {
        int x, y;
        cin >> x >> y;
        special.push_back({x, y});
    }
    for (auto [x, y] : special) {
        pre[y].push_back(x);
    }
    vector<int> q(n + 1);
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (q[i]) continue;
        vector<int> ord_cur;
        int x = i;
        while (!q[x]) {
            q[x] = 1;
            ord_cur.push_back(x);
            for (int y : pre[x]) {
                if (!q[y]) x = y;
            }
        }
        if (ord_cur.size() > ord.size()) cnt++;
        ord_cur.insert(ord_cur.end(), ord.begin(), ord.end());
        for (int y : ord_cur) {
            if (q[y]) continue;
            q[y] = 1;
            ord.push_back(y);
        }
    }
    if (cnt > 1) {
        cout << 0 << endl;
        return 0;
    }
    for (int x : ord) {
        cout << x << " ";
    }
    cout << endl;
    return 0;
}
