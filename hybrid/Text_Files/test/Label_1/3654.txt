#include <bits/stdc++.h>
using namespace std;
#define ll long long

const int mxN = 2e5 + 10;
int n, a[3][mxN], freq[3][mxN], ans;
string s[3];

int main() {
    cin >> n;
    for (int i = 0; i < 3; i++) {
        cin >> s[i];
        for (int j = 0; j < s[i].size(); j++) {
            a[i][j] = s[i][j] - 'a';
            freq[i][a[i][j]]++;
        }
    }
    for (int i = 0; i < s[0].size(); i++) {
        for (int j = 0; j < 3; j++) freq[j][a[j][i]]--;
        int cur = 0;
        for (int j = 0; j < 26; j++) {
            int mx = 0;
            for (int k = 0; k < 3; k++) mx = max(mx, freq[k][j]);
            cur += mx;
        }
        ans = max(ans, cur);
        for (int j = 0; j < 3; j++) freq[j][a[j][i]]++;
    }
    if (ans % 3 != 0) cout << "Kuro" << endl;
    else if (ans % 3 == 0) cout << "Draw" << endl;
}
