#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int ans = 0;
    for (int i = 1; i < s.size(); i++) {
      if (s[i - 1] == s[i]) {
        ans++;
        s[i] = 'x';
      }
    }
    cout << ans << endl;
  }
  return 0;
}
