import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n + 1];
        int[] rev = new int[n + 1];
        String[] t = br.readLine().split(" ");
        for (int i = 1; i <= n; i++) {
            a[i] = Integer.parseInt(t[i - 1]);
            rev[a[i]] = i;
        }
        int m = Integer.parseInt(br.readLine());
        boolean odd = false;
        PrintWriter pw = new PrintWriter(System.out);
        while (m-- > 0) {
            t = br.readLine().split(" ");
            int l = Integer.parseInt(t[0]), r = Integer.parseInt(t[1]);
            odd ^= (r - l) % 2 == 1;
            while (l < r) {
                int ai = a[l], aj = a[r];
                int ali = l, ajr = r;
                a[l++] = aj;
                rev[ai] = ajr;
                a[r--] = ai;
                rev[aj] = ali;
            }
            pw.println(odd ? "odd" : "even");
        }
        pw.flush();
    }
}
