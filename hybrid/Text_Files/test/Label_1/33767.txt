import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        Arrays.sort(a);
        if (a[n - 1] - a[0] > d) {
            System.out.println("NO");
        } else {
            int cur = a[0], cnt = 0, ans = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] - cur <= d) {
                    cnt++;
                } else {
                    cur = a[i];
                    cnt = 1;
                }
                if (cnt >= k) ans++;
            }
            System.out.println(ans >= n / k ? "YES" : "NO");
        }
    }
}
