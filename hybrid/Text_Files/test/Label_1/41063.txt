import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    and_a = [0] * 64
    for x in a:
        for i in range(64):
            if (1 << i) & x:
                and_a[i] += 1
    ans = []
    for i in range(64):
        if all(x % (1 << i) == 0 for x in a):
            ans.append(1 << i)
    print(*ans)
