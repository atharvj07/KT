import java.util.Scanner;

public class LongestGreedySubsequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] result = solve(a, k);
        for (int r : result) {
            System.out.print(r + " ");
        }
    }

    private static int[] solve(int[] a, int k) {
        int n = a.length;
        int[] result = new int[n - k + 1];

        // dp[i][j] stores the length of the longest greedy subsequence
        // ending at index j in the subarray a[i...j].
        int[][] dp = new int[n][n];

        // Initialize the first row of dp with 1.
        for (int j = 0; j < n; j++) {
            dp[0][j] = 1;
        }

        // Iterate over the subarrays.
        for (int i = 1; i < n - k + 1; i++) {
            // Iterate over the indices in the subarray.
            for (int j = i + k - 1; j < n; j++) {
                // Find the maximum length of a greedy subsequence ending at index j.
                int maxLength = 1;
                for (int p = i; p < j; p++) {
                    if (a[p] < a[j] && dp[i - 1][p] + 1 > maxLength) {
                        maxLength = dp[i - 1][p] + 1;
                    }
                }
                dp[i][j] = maxLength;
            }
        }

        // Copy the last row of dp to the result array.
        System.arraycopy(dp[n - k], 0, result, 0, n - k + 1);
        return result;
    }
}
