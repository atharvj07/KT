import java.util.*;
public class Main {
    public static void main(String[] aaa) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int dom = 0, j = 0;
        for (int i = 0; i < n && j < n; i++) {
            if (a[j] >= 2) {
                a[j] -= 2;
                dom++;
                if (a[j] > 0)
                    j++;
            } else {
                j++;
            }
        }
        System.out.println(dom);
    }
}
