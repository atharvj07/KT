import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Point, Integer> map = new HashMap<>();
        for(int i=0;i<n;++i){
            int x = sc.nextInt(), y = sc.nextInt();
            map.put(new Point(x, y), map.getOrDefault(new Point(x, y), 0)+1);
        }
        sc.close();
        long ans = 0;
        for(Point p:map.keySet()){
            for(Point q:map.keySet()){
                if(p.equals(q)){
                    int f = map.get(p);
                    ans += (long)f*(f-1)*(f-2)/6;
                }else{
                    int f = map.get(p);
                    int s = map.get(q);
                    ans+=(long)f*s*(s-1)/2;
                }
            }
        }
        System.out.println(ans);
    }
}

class Point{
    int x, y;
    Point(int x, int y){
        this.x = x;
        this.y = y;
    }
    @Override
    public boolean equals(Object obj){
        Point p = (Point)obj;
        return this.x == p.x && this.y == p.y;
    }

    @Override
    public int hashCode() {
        return (x*100000+y)%1000000;
    }
}
