import sys

t = int(input())

for _ in range(t):
    a = input()
    b = list(a)
    even = []; odd = []
    for i in b:
        if int(i) % 2 == 0:
            even.append(i)
        else:
            odd.append(i)
    
    i = 0; j = 0
    while i < len(even) and j < len(odd):
        if b[i] > odd[j]:
            b[i], b[j+1] = b[j+1], b[i]
            j += 1
        i += 1
    
    print(''.join(b).lstrip('0'))
