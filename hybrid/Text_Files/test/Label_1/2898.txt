#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false); cin.tie(nullptr);
  int n, k, A, B;
  cin >> n >> k >> A >> B;

  vector<bool> p(1 << n);
  for (int i = 0; i < k; ++i) {
    int a; cin >> a; --a;
    p[a] = true;
  }
  int ans = 1e9;
  for (int mask = 0; mask < (1 << n); ++mask) {
    int cnt = 0;
    for (int i = 0; i < n; ++i)
      cnt += (mask >> i) & 1;
    if (cnt <= 1) {
      ans = min(ans, A * (1 << n));
      continue;
    }
    int num = 0;
    for (int i = 0; i < (1 << n); ++i) {
      if (mask & i) num += !p[i];
    }
    ans = min(ans, B * cnt * (1 << __builtin_ffs(mask)) + num * A);
  }
  cout << ans << endl;
  return 0;
}
