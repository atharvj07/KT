import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int A = scanner.nextInt();
        int B = scanner.nextInt();

        int count = 0;
        while (B > 0) {
            B -= A - 1;
            count++;
        }

        System.out.println(count);
    }
}
