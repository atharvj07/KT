#include <iostream>
using namespace std;
int main() {
    int n, x[101], y[101], up = 0, down = 0;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> x[i] >> y[i];
        up += x[i];
        down += y[i];
    }
    if ((up & 1) != (down & 1)) {
        cout << -1;
    } else {
        int ans = 0;
        for (int i = 1; i <= n; ++i)
            if (x[i] != y[i])
                ++ans;
        cout << ans / 2;
    }
}
