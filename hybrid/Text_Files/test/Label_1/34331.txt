import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Segment[] segs = new Segment[n];
        for (int i = 0; i < n; i++) {
            int x1 = sc.nextInt(), y1 = sc.nextInt(), x2 = sc.nextInt(), y2 = sc.nextInt();
            segs[i] = new Segment(x1, y1, x2, y2);
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (segs[i].isHorizontal() && segs[j].isVertical() && segs[i].intersects(segs[j])) {
                    for (int k = j + 1; k < n; k++) {
                        for (int l = k + 1; l < n; l++) {
                            if (segs[k].isHorizontal() && segs[l].isVertical() && segs[i].intersects(segs[k]) && segs[i].intersects(segs[l]) && segs[j].intersects(segs[k]) && segs[j].intersects(segs[l])) {
                                ans++;
                            }
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }

    static class Segment {
        int x1, y1, x2, y2;

        public Segment(int x1, int y1, int x2, int y2) {
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
        }

        public boolean isHorizontal() {
            return y1 == y2;
        }

        public boolean isVertical() {
            return x1 == x2;
        }

        public boolean intersects(Segment other) {
            if (isVertical() && other.isHorizontal()) {
                return x1 <= other.x1 && other.x1 <= x2 && other.y1 <= y1 && y1 <= other.y2;
            } else if (isHorizontal() && other.isVertical()) {
                return other.x1 <= x1 && x1 <= other.x2 && y1 <= other.y1 && other.y1 <= y2;
            } else {
                return false;
            }
        }
    }
}
