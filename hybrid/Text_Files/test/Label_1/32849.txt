import java.util.Scanner;

public class Main {
    static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long[] x = new long[6];
        long[] y = new long[6];
        for (int i = 0; i < 6; i++) {
            x[i] = sc.nextLong();
        }
        for (int i = 0; i < 6; i++) {
            y[i] = sc.nextLong();
        }
        long ans = (x[1] - x[0]) * (y[2] - y[1]) * (x[4] - x[3]) * (y[5] - y[4]);
        System.out.println(ans % MOD);
    }
}
