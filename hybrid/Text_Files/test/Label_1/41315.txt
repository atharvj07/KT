n = int(input())
qu = []
for i in range(n):
    l = list(map(int, input().split()))
    if l[0] == 1:
        if len(qu) == 0 or qu[-1][1] < l[1]:
            qu.append(l[1:])
            continue
        for j in range(len(qu)):
            if j == 0:
                if qu[j][1] < l[1]:
                    qu = [l[1:]] + qu
                    break
            elif qu[j][1] < l[1] and l[1] < qu[j - 1][1]:
                qu = qu[:j] + [l[1:]] + qu[j:]
                break
    elif l[0] == 2 and l[1] != l[2]:
        t = 0
        for j in range(len(qu)):
            if l[1] > qu[j][0] and l[1] < qu[j][1]:
                t = j + 1
            elif l[2] > qu[j][0] and l[2] < qu[j][1]:
                t -= 1
        if t != 0:
            print('YES')
        else:
            print('NO')
