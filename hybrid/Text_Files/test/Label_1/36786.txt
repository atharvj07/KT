import sys

input = sys.stdin.readline

n, q = map(int, input().split())

a = list(map(int, input().split()))
b = list(map(int, input().split()))

for _ in range(q):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    if a[l] != b[l] or a[r] != b[r]:
        print(-1)
    else:
        ans = 0
        if a[l:r+1] == b[l:r+1]:
            print(ans)
        else:
            for i in range(l, r):
                if a[i] != b[i]:
                    ans += 1
            if ans % 2 == 0:
                print(ans//2)
            else:
                print(-1)
