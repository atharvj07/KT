#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
vector<ll> dp;
int main() {
  int n;
  cin >> n;
  dp.resize(n + 1);
  dp[3] = 6;
  for (int i = 4; i <= n; i += 2) {
    dp[i] = (dp[i - 2] * i) % mod;
  }
  cout << dp[n] << endl;
  return 0;
}
