import sys

n, k = map(int, input().split())
prereq = list(map(int, input().split()))
spec = [list(map(int, input().split())) for _ in range(k)]
vis = [False] * (n + 1)
res = []
stack = []

for i in range(1, n+1):
    if not vis[i]:
        stack.append(i)
        while stack:
            node = stack[-1]
            if vis[node]:
                stack.pop()
                res.append(node)
            else:
                vis[node] = True
                if prereq[node-1] != 0 and not vis[prereq[node-1]]:
                    stack.append(prereq[node-1])
                else:
                    can = True
                    for x, y in spec:
                        if x == node and vis[y]:
                            can = False
                            break
                    if can:
                        stack.pop()
                        res.append(node)
                    else:
                        res = []
                        break
if len(res) == n:
    print(*res)
else:
    print(0)
