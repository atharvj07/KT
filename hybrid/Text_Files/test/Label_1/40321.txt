from sys import stdin

n = int(stdin.readline())
lis = list(map(int, stdin.readline().split()))

ans = 1
ind = -1

for i in range(n):
    if lis[i] == -1:
        if ind != -1:
            ans *= i - ind
            ind = i
        else:
            ind = i
    elif ind != -1:
        ans *= i - ind

print(ans % 1000000007)
