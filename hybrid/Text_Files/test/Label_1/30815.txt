import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long x = sc.nextLong();
            boolean flag = false;
            for (long a = 1; a * a * a <= x; a++) {
                double b = Math.cbrt(x - a * a * a);
                if (b == (int) b) {
                    flag = true;
                    break;
                }
            }
            System.out.println(flag ? "YES" : "NO");
        }
    }
}
