import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static long mod = (int) (Math.pow(10, 9) + 7);

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]), k = Integer.parseInt(nk[1]);
        List<Integer>[] adj = new List[n + 1];
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            String[] ab = br.readLine().split(" ");
            int a = Integer.parseInt(ab[0]), b = Integer.parseInt(ab[1]);
            adj[a].add(b);
            adj[b].add(a);
        }

        long[] dp = new long[n + 1];
        boolean[] vis = new boolean[n + 1];
        dfs1(1, -1, dp, vis, adj);

        long[] dp1 = new long[n + 1];
        Arrays.fill(vis, false);
        dfs2(1, -1, dp, dp1, vis, adj, k);

        long sum = 0;
        for (int i = 1; i <= n; i++) {
            sum = (sum + dp1[i]) % mod;
        }
        System.out.println(sum);
    }

    private static void dfs1(int node, int parent, long[] dp, boolean[] vis, List<Integer>[] adj) {
        vis[node] = true;
        Map<Long, Integer> freq = new HashMap<>();
        for (int child : adj[node]) {
            if (child == parent) continue;
            dfs1(child, node, dp, vis, adj);
            long childDp = dp[child];
            dp[node] = dp[node] + childDp + 1;
            freq.put(childDp, freq.getOrDefault(childDp, 0) + 1);
        }
        long mx = 0;
        for (long f : freq.keySet()) {
            mx = Math.max(mx, f);
        }
        dp[node] -= mx;
    }

    private static void dfs2(int node, int parent, long[] dp, long[] dp1, boolean[] vis, List<Integer>[] adj, int k) {
        vis[node] = true;
        for (int child : adj[node]) {
            if (child == parent) continue;
            if (dp[node] - dp[child] - 1 > dp[child]) {
                dp1[child] = (dp1[child] + dp1[node] + dp[child] * (k - 1) % mod + 1) % mod;
            } else {
                dp1[child] = (dp1[child] + dp1[node] + dp[node] * (k - 1)) % mod;
            }
            dfs2(child, node, dp, dp1, vis, adj, k);
        }
    }
}
