from itertools import permutations

s = input()
d = {}
for i in s:
    if i not in d:
        d[i] = 0
    d[i] += 1
ans = [-1] * len(s)
odd = 0
oddc = ''
for i in d:
    if d[i] % 2 == 1:
        if odd > 0:
            break
        odd += 1
        oddc = i
    for j in range(d[i] // 2):
        ans[len(s) // 2 - j - 1] = i
        ans[len(s) // 2 + j] = i
if odd > 0:
    ans[len(s) // 2] = oddc
pm = sorted(permutations(d))
for p in pm:
    st = ''
    for i in p:
        for j in range(d[i]):
            st += i
    if st < ''.join(ans):
        ans = list(st)
print(''.join(ans))
