#include <iostream>

using namespace std;

typedef long long LL;

void solve() {
    LL r, b, k;
    cin >> r >> b >> k;
    if (r == 1 || b == 1 || r % k == 0 || b % k == 0) cout << "REBEL" << endl;
    else cout << "OBEY" << endl;
}

int main() {
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
