#include <bits/stdc++.h>
#define ll long long
#define ld long double
#define forab(i, a, b) for (ll i = a; i <= b; i++)
#define forba(i, a, b) for (ll i = a; i >= b; i--)
#define all(a) a.begin(), a.end()
#define S second
#define F first
const ll mod = 1e9 + 7;
using namespace std;
int main() {
  ll n, p;
  cin >> n >> p;
  vector<pair<ll, ll>> a(n);
  forab(i, 0, n - 1) { cin >> a[i].F >> a[i].S; }
  sort(all(a));
  ld ans = 0;
  ll sum = 0;
  forab(i, 0, n - 1) {
    if (a[i].S <= sum) {
      ld x = 1.0 * (sum - a[i].S) / (a[i].F - p);
      sum += (x * p - x * a[i].F);
      ans = max(ans, x);
    } else {
      ld x = 1.0 * (a[i].S - sum) / (p - a[i].F);
      sum += (x * p - x * a[i].F);
      ans = max(ans, x);
    }
  }
  if (sum <= 0) {
    cout << -1 << endl;
  } else {
    cout << fixed;
    cout << setprecision(10) << ans;
    cout << endl;
  }
  return 0;
}
