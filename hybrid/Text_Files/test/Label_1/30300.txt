import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        long L = sc.nextLong();
        int N = sc.nextInt();
        long[] X = new long[N];
        for (int i = 0; i < N; i++) {
            X[i] = sc.nextLong();
        }
        sc.close();

        long ans = 0;
        for (long x : X) {
            ans += Math.min(x, L - x);
        }
        ans *= 2;

        System.out.println(ans);
    }
}
