import sys

T = int(sys.stdin.readline())

for _ in range(T):
    N, M = map(int, sys.stdin.readline().split())
    S = sys.stdin.readline().rstrip()
    L = len(S)
    x, d = 1, [1, 0, -1, 0]
    y = 1
    for c in S:
        if c == 'D':
            x += d[2]
        elif c == 'U':
            x += d[0]
        elif c == 'L':
            y += d[1]
        elif c == 'R':
            y += d[3]
        if x <= 0 or x > N or y <= 0 or y > M:
            break
    print(x, y)
