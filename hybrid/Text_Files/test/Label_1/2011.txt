#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long cur = 1, ans = 0;
  for (auto x : a) {
    if (x < cur) break;
    ans += min(cur, x);
    cur++;
  }
  cout << ans << '\n';
}
