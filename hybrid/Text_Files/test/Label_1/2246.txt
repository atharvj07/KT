#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  vector<int> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      for (int j = i + 1; j < n; j++) {
        if (s[j] == t[i]) {
          ans.push_back(j);
          swap(s[i], s[j]);
          ans.push_back(i);
          break;
        }
      }
    }
  }
  if (s == t) {
    cout << ans.size() << endl;
    for (int x : ans) { cout << x << " "; }
  } else {
    cout << -1 << endl;
  }
}
