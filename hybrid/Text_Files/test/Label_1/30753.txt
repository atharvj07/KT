import java.util.Scanner;

public class DiverseStrings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline after the integer

        for (int i = 0; i < n; i++) {
            String str = scanner.nextLine();
            boolean diverse = isDiverse(str);
            System.out.println(diverse ? "Yes" : "No");
        }
    }

    private static boolean isDiverse(String str) {
        int prev = -1;
        for (char ch : str.toCharArray()) {
            int curr = ch - 'a';
            if (curr - prev > 1 || curr == prev) {
                return false;
            }
            prev = curr;
        }
        return true;
    }
}
