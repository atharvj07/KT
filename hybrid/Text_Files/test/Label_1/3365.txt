#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) { cin >> p[i]; }
    vector<int> ans(n + 1, 1);
    for (int i = 1; i <= n; i++) {
        int last = 0;
        for (int j = 1; j <= n; j++) {
            if (p[j - 1] == i) {
                ans[i] = max(ans[i], last + 1);
                last = ans[i];
            }
        }
    }
    for (int i = 0; i <= n; i++) { cout << ans[i] << ' '; }
    return 0;
}
