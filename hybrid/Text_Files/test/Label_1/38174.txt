def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def phi(n):
    result = n
    p = 2
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    if n > 1:
        result -= result // n
    return result

def is_primitive_root(x, p):
    if gcd(x, p) != 1:
        return False
    phi_p = phi(p)
    fact = []
    q = 2
    while q * q <= phi_p:
        if phi_p % q == 0:
            fact.append(q)
            while phi_p % q == 0:
                phi_p //= q
        q += 1
    if phi_p > 1:
        fact.append(phi_p)
    for factor in fact:
        if pow(x, phi(p) // factor, p) == 1:
            return False
    return True

def count_primitive_roots(p):
    count = 0
    for x in range(1, p):
        if is_primitive_root(x, p):
            count += 1
    return count

p = int(input())
print(count_primitive_roots(p))
