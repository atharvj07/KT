import java.util.*;

public class Main {
    private static final long MOD = (long) (1e9 + 7);
    private static long[] dp;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        dp = new long[n];
        Arrays.fill(dp, -1);
        long ans = solve(s, 0);
        System.out.println(ans);
    }

    private static long solve(String s, int i) {
        if (i == s.length()) {
            return 1;
        }
        if (dp[i] != -1) {
            return dp[i];
        }
        long ans = 0;
        if (s.charAt(i) == 'a') {
            ans = (ans + solve(s, i + 1)) % MOD;
            if (i + 1 < s.length() && s.charAt(i + 1) == 'a') {
                ans = (ans + solve(s, i + 2)) % MOD;
            }
        } else {
            ans = (ans + solve(s, i + 1)) % MOD;
            if (i + 1 < s.length() && s.charAt(i + 1) == 'b') {
                ans = (ans + solve(s, i + 2)) % MOD;
            }
        }
        dp[i] = ans;
        return ans;
    }
}
