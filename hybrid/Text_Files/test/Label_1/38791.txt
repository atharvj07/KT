MOD = 1000000007

def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result = (result * i) % MOD
    return result

def mod_inverse(a, m):
    return pow(a, m - 2, m)

def comb(n, r, mod):
    numerator = factorial(n)
    denominator = (factorial(r) * factorial(n - r)) % mod
    return (numerator * mod_inverse(denominator, mod)) % mod

k = int(input())
colors = [int(input()) for _ in range(k)]

total_ways = 1
total_balls = 0
for c in colors:
    total_ways = (total_ways * comb(total_balls + c, c, MOD)) % MOD
    total_balls += c

print(total_ways)
