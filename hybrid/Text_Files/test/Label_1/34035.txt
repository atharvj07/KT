import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

public class GoodString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = scanner.nextLine();
        }

        String result = findGoodString(strings);
        System.out.println(result);
    }

    private static String findGoodString(String[] strings) {
        // Find all substrings of all strings.
        Set<String> allSubstrings = new HashSet<>();
        for (String string : strings) {
            for (int i = 0; i < string.length(); i++) {
                for (int j = i + 1; j <= string.length(); j++) {
                    allSubstrings.add(string.substring(i, j));
                }
            }
        }

        // Count occurrences of each substring in each string.
        Map<String, Map<String, Integer>> substringCounts = new HashMap<>();
        for (String string : strings) {
            Map<String, Integer> counts = new HashMap<>();
            for (String substring : allSubstrings) {
                counts.put(substring, countOccurrences(string, substring));
            }
            substringCounts.put(string, counts);
        }

        // Find all good substrings.
        List<String> goodSubstrings = new ArrayList<>();
        for (String substring : allSubstrings) {
            boolean isGood = true;
            for (Map<String, Integer> counts : substringCounts.values()) {
                int maxCount = Collections.max(counts.values());
                if (counts.get(substring) < maxCount) {
                    isGood = false;
                    break;
                }
            }
            if (isGood) {
                goodSubstrings.add(substring);
            }
        }

        // Find the lexicographically minimum good string.
        if (goodSubstrings.isEmpty()) {
            return "NO";
        }
        String result = goodSubstrings.get(0);
        for (String substring : goodSubstrings) {
            if (substring.compareTo(result) < 0) {
                result = substring;
            }
        }
        return result;
    }

    private static int countOccurrences(String string, String substring) {
        int count = 0;
        int index = string.indexOf(substring);
        while (index != -1) {
            count++;
            index = string.indexOf(substring, index + substring.length());
        }
        return count;
    }
}
