def max_digit_product(n):
    if n < 10:
        return n
    
    digits = list(map(int, str(n)))
    max_product = 1
    
    for i in range(len(digits)):
        if digits[i] == 0:
            continue
        
        temp_digits = digits[:]
        temp_digits[i] -= 1
        
        for j in range(i + 1, len(temp_digits)):
            temp_digits[j] = 9
        
        product = 1
        for num in temp_digits:
            product *= num
            
        max_product = max(max_product, product)
        
    return max_product

n = int(input())
print(max_digit_product(n))
