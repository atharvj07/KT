t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))

    zeros = a.count(0)
    ones = a.count(1)

    if zeros >= n // 2:
        print(n // 2)
        print('0 ' * (n // 2))
    else:
        if (n // 2) % 2 == 0:
            print(n // 2)
            print('1 ' * (n // 2))
        else:
            print(n // 2 + 1)
            print('1 ' * (n // 2 + 1))
