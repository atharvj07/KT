def max_projects(n, r, projects):
    pos = [p for p in projects if p[1] >= 0]
    neg = [p for p in projects if p[1] < 0]

    pos.sort(key=lambda x: x[0])
    neg.sort(key=lambda x: x[0] + x[1])

    completed = 0

    for p in pos:
        if r >= p[0]:
            r += p[1]
            completed += 1

    dp = [0] * (r + 1)
    for a, b in neg:
        for i in range(r, a - 1, -1):
            if i + b >= 0:
                dp[i + b] = max(dp[i + b], dp[i] + 1)

    return completed + max(dp)

n, r = map(int, input().split())
projects = [list(map(int, input().split())) for _ in range(n)]

print(max_projects(n, r, projects))
