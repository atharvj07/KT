n, m, q = map(int, input().split())
a = [[0] * m for i in range(n)]
row = [0] * n
col = [0] * m
for i in range(q):
    t, *args = map(int, input().split())
    if t == 3:
        r, c, x = args
        a[r - 1][c - 1] = x
    else:
        x = args[0] - 1
        if t == 1:
            row[x] = (row[x] + 1) % m
        else:
            col[x] = (col[x] + 1) % n
for i in range(n):
    for j in range(m):
        print(a[(i + row[i]) % n][(j + col[j]) % m], end=' ')
    print()
