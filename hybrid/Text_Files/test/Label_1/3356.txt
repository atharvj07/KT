#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    long long n, m;
    cin >> n >> m;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    unordered_map<long long, long long> freq;
    for (int i = 0; i < n; i++) {
        freq[a[i]]++;
    }
    long long ans = 0;
    for (auto it : freq) {
        ans += it.second * (it.second - 1) / 2;
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (a[i] != a[j]) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
