#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k, d;
  cin >> n >> k >> d;
  if (n > k * d) {
    cout << -1 << endl;
    return 0;
  }
  vector<vector<int>> ans(d, vector<int>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < d; j++) { ans[j][i] = (i % k) + 1; }
  }
  for (int i = 0; i < d; i++) {
    for (int j = 0; j < n; j++) { cout << ans[i][j] << " "; }
    cout << endl;
  }
  return 0;
}
