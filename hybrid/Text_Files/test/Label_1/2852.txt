#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    long long mn = *min_element(a.begin(), a.end());
    sort(a.begin(), a.end());
    bool ok = true;
    for (int i = 1; i < n; i++) {
      if (a[i] < a[i - 1]) {
        if (__gcd(a[i], a[i - 1]) != mn) ok = false;
      }
    }
    cout << (ok ? "YES\n" : "NO\n");
  }
  return 0;
}
