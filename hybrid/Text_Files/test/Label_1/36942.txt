import sys

def is_digit(c):
    return '0' <= c <= '9'

def is_word(c):
    return c.isalpha() or is_digit(c) or c in '$_'

def is_hash(c):
    return c == '#'

def is_reserved(c):
    return c in reserved

def minify():
    words = []
    for line in source:
        for token in tokenize(line):
            if not is_reserved(token):
                words.append(token)
    words = sorted(set(words))
    words = [word for word in words if not is_reserved(word)]
    words = sorted(words, key=lambda w: (len(w), w))
    min_str = ''
    for line in source:
        new_line = ''
        for token in tokenize(line):
            if not is_reserved(token):
                idx = words.index(token)
                token = letters[idx] if idx < len(letters) else '\{' + token + '\}'
            new_line += token + ' '
        new_line = new_line.strip()
        min_str = min_str + new_line + '\n'
    return min_str.strip()

def tokenize(line):
    token = ''
    in_comment = False
    for c in line:
        if in_comment:
            if is_hash(c):
                in_comment = False
            continue
        if is_hash(c):
            in_comment = True
            continue
        if is_space(c):
            if token:
                yield token
            token = ''
        else:
            token += c
    if token:
        yield token

def is_space(c):
    return c == ' ' or c == '\t'

n = int(input())
reserved = set(input().split())
m = int(input())
source = [input().strip() for i in range(m)]
letters = 'abcdefghijklmnopqrstuvwxyz'
print(minify())
