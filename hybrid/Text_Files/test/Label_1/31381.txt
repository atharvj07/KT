import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n];
            int xor = 0;
            for(int i=0; i<n; i++) {
                a[i] = sc.nextInt();
                xor ^= a[i];
            }
            if(xor == 0) {
                System.out.println("DRAW");
            } else if(n%2 == 0) {
                System.out.println("LOSE");
            } else {
                System.out.println("WIN");
            }
        }
    }
}
