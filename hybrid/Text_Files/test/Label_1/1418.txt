#include <bits/stdc++.h>
using namespace std;
const long long MAXN = 2e5;
const long long INF = 1e18;
long long n, s, a[MAXN + 5], c[MAXN + 5];
int main() {
    cin >> n >> s;
    for (int i = 0; i < n; i++) cin >> a[i], c[a[i]]++;
    long long cnt = 0, sum = 0;
    for (long long i = 1; i <= MAXN; i++) {
        cnt += c[i];
        sum += c[i] * i;
        if (cnt >= (n + 1) / 2) {
            if (sum / cnt == s) {
                cout << 0 << endl;
                return 0;
            }
            if (sum / cnt < s) {
                long long ans = INF;
                for (long long j = sum / cnt + 1; j <= s; j++) {
                    if ((j - sum / cnt) * cnt <= n) ans = min(ans, (j - sum / cnt) * cnt);
                    else
                        ans = min(ans, n + (j - (sum / cnt + 1)) * cnt);
                }
                cout << ans << endl;
                return 0;
            } else {
                long long ans = INF;
                for (long long j = s; j < sum / cnt; j++) {
                    if ((sum / cnt - j) * cnt <= n) ans = min(ans, (sum / cnt - j) * cnt);
                    else
                        ans = min(ans, n + (sum / cnt - (j + 1)) * cnt);
                }
                cout << ans << endl;
                return 0;
            }
        }
    }
    return 0;
}
