import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n - 1];
        for (int i = 0; i < n - 1; i++) d[i] = sc.nextInt();
        int a = sc.nextInt() - 1;
        int b = sc.nextInt() - 1;
        int ans = d[a];
        for (int i = a + 1; i < b; i++) ans += d[i];
        System.out.println(ans);
    }
}
