import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    static class Graph {
        private List<List<Integer>> adjList;
        private int n;
        private int[] visited;
        private int[] path;
        private boolean hasHamiltonianPath;

        public Graph(int n) {
            this.n = n;
            adjList = new ArrayList<>(n);
            for (int i = 0; i < n; i++) {
                adjList.add(new ArrayList<>());
            }
            visited = new int[n];
            path = new int[n];
            hasHamiltonianPath = false;
        }

        public void addEdge(int u, int v) {
            adjList.get(u - 1).add(v - 1);
            adjList.get(v - 1).add(u - 1);
        }

        public boolean hasHamiltonianPath() {
            dfs(0);
            return hasHamiltonianPath;
        }

        private void dfs(int v) {
            if (hasHamiltonianPath) {
                return;
            }
            visited[v] = 1;
            path[v] = 1;
            for (int u : adjList.get(v)) {
                if (visited[u] == 0) {
                    dfs(u);
                }
            }
            if (visited[0] == 1 && visited[n - 1] == 1) {
                hasHamiltonianPath = true;
            }
            path[v] = 0;
            visited[v] = 0;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        Graph graph = new Graph(n);
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            graph.addEdge(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
        }
        if (graph.hasHamiltonianPath()) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
