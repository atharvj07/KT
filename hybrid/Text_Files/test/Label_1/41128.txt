n,a,b=map(int,input().split())
if n%2==(a*b)%2:
    c=[[0]*b for i in range(a)]
    p=[i for i in range(1,n+1) if i%2==(a*b)%2]
    q=[i for i in range(1,n+1) if i%2!=(a*b)%2]
    for i in range(n):
        for j in range(b):
            if i%2==0:c[i][j]=p.pop(0)
            else:c[i][j]=q.pop(0)
    for i in range(a):
        print(*c[i])
else:print(-1)
