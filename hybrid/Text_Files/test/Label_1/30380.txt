import java.util.*;
class Main {
    static int solve(int n, int k, int[] a) {
        int mx = 1, cnt = 1, lst = a[0];
        for (int i = 1; i < n; i++) {
            if (a[i] != lst) {
                cnt++;
            } else {
                mx = Math.max(mx, cnt);
                cnt = 1;
            }
            lst = a[i];
        }
        return Math.max(mx, cnt);
    }
	public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(solve(n, k, a));
    }
}
