import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long diff = n - m;
        System.out.println(m == diff || m == diff + 1 ? m + 1 : m - 1);
    }
}
