import sys

for _ in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))

    if a == sorted(a):
        print('YES')
        continue
    
    swap = False
    for i in range(1, n):
        if a[i] < a[i - 1]:
            swap = True
            break
    
    if swap:
        print('YES')
    else:
        print('NO')
