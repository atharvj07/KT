#include<bits/stdc++.h>
using namespace std ;
bool ok( string s , int &a , int &b ){
  int n = s.size() ;
  for( int i = 0 ; i < n ; i++ ){
    if( s[i] == s[n-i-1] || s[i] == '?' || s[n-i-1] == '?' ) continue ;
    else return 0 ;
  }
  int cnt0 = 0 , cnt1 = 0 ;
  for( char ch : s ) cnt0 += ch == '0' , cnt1 += ch == '1' ;
  if( abs( cnt0 - a ) + abs( cnt1 - b ) > 2 ) return 0 ;
  return 1 ;
}
int main(){
  int t ; cin >> t ;
  while( t-- ){
    int a , b ; cin >> a >> b ;
    string s ; cin >> s ;
    if( !ok( s , a , b ) ) cout << "-1\n" ;
    else{
      int n = s.size() ;
      int cur0 = 0 , cur1 = 0 ;
      for( int i = 0 ; i < n ; i++ ){
        if( s[i] == '?' ){
          if( cur1 < b ){
            cout << '1' , cur1++ ;
          }
          else{
            cout << '0' , cur0++ ;
          }
        }
        else cout << s[i] ;
      }
      cout << '\n' ;
    }
  }
  return 0 ;
}
