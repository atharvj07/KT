L = int(input())
A = int(input())

def find_periodic_number(L, A):
    P = A // (10 ** L) + 1
    X = int(str(P) * (L + 1))  # Try to construct a periodic number with one extra period
    # Check if constructed number is actually bigger than A
    if X > A:
        return int(str(P) * (len(str(A)) // L))  # Adjust the periodic number to match or exceed A in size
    else:
        return int(str(P + 1) * (len(str(A)) // L))

print(find_periodic_number(L, A))
