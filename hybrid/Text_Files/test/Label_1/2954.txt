#include <iostream>
using namespace std;

int f(long long n) {
    if (n < 10) return n;
    long long x = n / 10;
    long long y = n % 10;
    return (x - 1) % 10 + 1 + (y - 1) % 10 + 1;
}

int main() {
    long long n;
    cin >> n;
    while (n--) {
        long long k;
        cin >> k;
        --k;
        long long ans = 0;
        while (true) {
            ans += f(k + 1);
            k = k / 10;
            if (k == 0) break;
        }
        ans = (ans + 10) % 10;
        cout << ans << '\n';
    }
    return 0;
}
