import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long m = in.nextLong();
        long sum = 0;
        int n = 0;
        for (long i = (int) Math.cbrt(m); i >= 1; i--) {
            if (sum + i * i * i <= m) {
                sum += i * i * i;
                n++;
            }
        }
        System.out.println(n + " " + sum);
    }
}
