import java.util.*;
import java.io.*;

class Main {
    static boolean[] NotPrime = new boolean[200001];
    static int[] Count = new int[200001];

    private static void sieve(int n) {
        // sieve prime numbers
        NotPrime[0] = NotPrime[1] = true;
        int lim = (int) Math.sqrt(n) + 1;
        for (int p = 2; p < lim; p++)
            if (!NotPrime[p])
                for (int i = p * p; i <= n; i += p)
                    NotPrime[i] = true;
    }

    private static int query(int[] a, int l, int r) {
        // count number of primes in subarray
        return Count[r] - ((l > 0) ? Count[l - 1] : 0);
    }

    public static void main(String[] args) throws IOException {
        sieve(200000);
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
            Count[i + 1] = Count[i];
            if (a[i] == 1)
                Count[i + 1]++;
        }
        int[] b = new int[n];
        int[] ans = new int[n];
        for (int i = 0; i < n; i++)
            b[i] = a[i];
        Arrays.sort(b);
        int l = 0, r = n - 1;
        for (int i = 0; i < n; i++) {
            if (query(b, l, r) < query(b, l + 1, r)) {
                ans[i] = b[l++];
            } else {
                ans[i] = b[r--];
            }
        }
        for (int j : ans)
            System.out.print(j + " ");
    }
}
