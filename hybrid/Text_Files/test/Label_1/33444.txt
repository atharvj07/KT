import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int q = sc.nextInt();
    for (int i = 0; i < q; i++) {
      int n = sc.nextInt();
      int[] t = new int[n];
      for (int j = 0; j < n; j++) {
        t[j] = sc.nextInt();
      }
      Arrays.sort(t);
      int ans = 0;
      for (int j = 0; j < n; j++) {
        ans += Math.abs(t[j] - (j + 1));
      }
      System.out.println(ans);
    }
  }
}
