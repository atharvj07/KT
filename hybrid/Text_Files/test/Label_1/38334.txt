n = int(input())

# The sum of the first n natural numbers is n*(n+1)/2
total_sum = n * (n + 1) // 2

# If the total sum is odd, the minimum possible difference is 1
# If the total sum is even, the sets can be divided such that the difference is 0
if total_sum % 2 == 0:
    print(0)
else:
    print(1)
