import math

def distance_on_avenue(a, b, c, x1, y1, x2, y2):
    # Calculating the perpendicular distance of the points from the avenue
    d1 = abs(a*x1 + b*y1 + c) / math.sqrt(a**2 + b**2)
    d2 = abs(a*x2 + b*y2 + c) / math.sqrt(a**2 + b**2)
    # Calculating the distance between the points along the avenue
    d_avenue = math.sqrt((x2-x1)**2 + (y2-y1)**2 - (d2-d1)**2)
    return d1 + d2 + d_avenue

def min_distance(a, b, c, x1, y1, x2, y2):
    # Straight line distance if not using the avenue
    min_dist = abs(x1-x2) + abs(y1-y2)
    # Check if both points are on the same side of the avenue or on the avenue
    if (a*x1 + b*y1 + c)*(a*x2 + b*y2 + c) <= 0:
        min_dist = min(min_dist, distance_on_avenue(a, b, c, x1, y1, x2, y2))
    return min_dist

a, b, c = map(int, input().split())
x1, y1, x2, y2 = map(int, input().split())
print('{:.10f}'.format(min_distance(a, b, c, x1, y1, x2, y2)))
