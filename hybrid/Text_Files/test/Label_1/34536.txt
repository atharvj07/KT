import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int min = a[n - 1] - a[1];
        for (int i = 1; i < n - 1; i++) {
            min = Math.min(min, Math.max(a[i + 1] - a[i - 1], a[n - 1] - a[i]));
        }
        System.out.println(min);
    }
}
