#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  long long x, y;
  cin >> x >> y;
  if (x == y) {
    cout << "Impossible" << endl;
    return 0;
  }
  if (x < y) {
    swap(x, y);
  }
  string ans;
  while (x > 0 && y > 0) {
    if (x > y) {
      ans += "B";
      x -= y;
    } else {
      ans += "A";
      y -= x;
    }
  }
  cout << ans << endl;
  return 0;
}
