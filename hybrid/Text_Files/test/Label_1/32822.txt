import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] deg = new int[n + 1];
        List<Integer>[] edgeList = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            edgeList[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            edgeList[u].add(v);
            edgeList[v].add(u);
            deg[u]++;
            deg[v]++;
        }
        int[] sortedDeg = Arrays.copyOf(deg, deg.length);
        Arrays.sort(sortedDeg);
        int k = (n + m + 1) / 2;
        Map<Integer, List<Integer>> degToEdgeMap = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            List<Integer> edgeListForDeg = degToEdgeMap.getOrDefault(deg[i], new ArrayList<>());
            edgeListForDeg.add(i);
            degToEdgeMap.put(deg[i], edgeListForDeg);
        }
        List<Integer> remainingEdges = new ArrayList<>();
        for (int i = sortedDeg.length - 1; k > 0; i--) {
            if (sortedDeg[i] == 0) {
                break;
            }
            List<Integer> edgeListForDeg = degToEdgeMap.getOrDefault(sortedDeg[i], new ArrayList<>());
            for (int u : edgeListForDeg) {
                if (deg[u] == 0) {
                    continue;
                }
                for (int v : edgeList[u]) {
                    if (deg[v] == 0) {
                        continue;
                    }
                    remainingEdges.add(u);
                    remainingEdges.add(v);
                    deg[u]--;
                    deg[v]--;
                    k--;
                    if (k == 0) {
                        break;
                    }
                }
                if (k == 0) {
                    break;
                }
            }
            if (k == 0) {
                break;
            }
        }
        List<Integer> edgeToRemove = new ArrayList<>();
        List<Integer> vertexToRemove = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (deg[i] > sortedDeg[deg[i] / 2]) {
                edgeToRemove.addAll(edgeList[i]);
                vertexToRemove.add(i);
            }
        }
        for (int v : vertexToRemove) {
            for (int u : edgeList[v]) {
                edgeToRemove.remove(Integer.valueOf(u));
            }
        }
        System.out.println(remainingEdges.size() + edgeToRemove.size());
        for (Integer edge : remainingEdges) {
            System.out.println(edge + " " + edgeList[edge].get(0));
            edgeList[edge].remove(Integer.valueOf(edgeList[edge].get(0)));
        }
        for (Integer edge : edgeToRemove) {
            System.out.println(edge + " " + edgeList[edge].get(0));
            edgeList[edge].remove(Integer.valueOf(edgeList[edge].get(0)));
        }
    }
}
