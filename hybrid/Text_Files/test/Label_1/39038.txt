n = int(input())
fragments = list(map(int, input().split()))

# Create a mapping of fragment to its sector
fragment_to_sector = {fragment: i for i, fragment in enumerate(fragments, start=1)}

time = 0
for i in range(1, n):
    current_sector = fragment_to_sector[i]
    next_sector = fragment_to_sector[i + 1]
    time += abs(next_sector - current_sector)

print(time)
