import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int cnt = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur + arr[i] <= 4) {
                cur += arr[i];
            } else {
                cnt++;
                cur = arr[i];
            }
        }
        if (cur > 0) cnt++;
        System.out.println(cnt);
    }
}
