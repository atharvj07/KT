from collections import defaultdict, deque
import sys

n, k = map(int, input().split())
prereq = [0] + [int(x) for x in input().split()]
edges = defaultdict(list)
deg = [0] * (n + 1)
for i in range(k):
    x, y = map(int, input().split())
    edges[x].append(y)
    deg[y] += 1
q = deque([i for i in range(1, n + 1) if not deg[i]])
ans = []
while q:
    u = q.popleft()
    ans.append(u)
    for v in edges[u]:
        deg[v] -= 1
        if not deg[v]:
            q.append(v)
if len(ans) == n:
    print(' '.join(map(str, ans)))
else:
    print(0)
