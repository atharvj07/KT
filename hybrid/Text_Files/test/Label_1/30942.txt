import java.io.*;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<Point> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            list.add(new Point(x, y));
        }
        BigDecimal max = BigDecimal.ZERO;
        for (int i = 0; i < n; i++) {
            Point p1 = list.get(i);
            Point p2 = list.get((i + 1) % n);
            Point p3 = list.get((i + 2) % n);
            BigDecimal area = new BigDecimal(p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y));
            area = area.abs();
            BigDecimal d = area.divide(new BigDecimal(2 * Math.abs(p1.x * p2.y - p2.x * p1.y + p3.x * p1.y - p1.x * p3.y + p2.x * p3.y - p3.x * p2.y)), 2, RoundingMode.HALF_UP);
            if (d.compareTo(max) == 1) {
                max = d;
            }
        }
        System.out.println(max.doubleValue());
    }

    static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
