#include <bits/stdc++.h>
using namespace std;

int n, m;
unordered_map<string, int> mp;
vector<string> s, t;

int main() {
  cin >> n >> m;
  s.resize(n); t.resize(m);
  for (int i = 0; i < n; i++) { cin >> s[i]; mp[s[i]]++; }
  for (int i = 0; i < m; i++) { cin >> t[i]; mp[t[i]]++; }
  int q; cin >> q;
  while (q--) {
    int y; cin >> y;
    long long cur = y - 1;
    cout << s[cur % n] << t[cur % m] << '\n';
  }
  return 0;
}
