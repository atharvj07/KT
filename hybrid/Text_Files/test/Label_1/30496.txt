import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long l = input.nextLong();
        long r = input.nextLong();
        long ans = r / 10L;
        if (l > 9L) --ans;
        if (l == 0L && r == 0L) ++ans;
        System.out.println(ans);
    }
}
