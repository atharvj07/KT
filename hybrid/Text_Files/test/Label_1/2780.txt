#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 2e5 + 10;
int n, q;
ll a[maxn], l[maxn], r[maxn], ans;
inline ll read() {
  ll x = 0, f = 1;
  char ch = getchar();
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') {
    x = x * 10 + ch - 48;
    ch = getchar();
  }
  return x * f;
}
void solve() {
  n = read(), q = read();
  for (int i = 1; i <= n; i++) a[i] = read();
  for (int i = 1; i <= q; i++) l[i] = read(), r[i] = read();
  sort(a + 1, a + n + 1);
  sort(l + 1, l + q + 1);
  sort(r + 1, r + q + 1);
  int p = 1;
  for (int i = q; i >= 1; i--) {
    while (p <= n && a[p] < r[i] + 1) p++;
    ans += 1ll * p * (r[i] - l[i] + 1);
  }
  cout << ans << endl;
}
int main() { solve(); }
