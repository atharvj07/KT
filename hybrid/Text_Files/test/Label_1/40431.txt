import sys

for T in range(int(sys.stdin.readline())):
    n = int(sys.stdin.readline())
    dl = [int(x) for x in sys.stdin.readline().split()]
    m = int(sys.stdin.readline())
    jl = [int(x) for x in sys.stdin.readline().split()]
    
    dl.sort()
    jl.sort()
    l, r = 0, m - 1
    ans = 0

    for p in dl:
        while l < r and jl[l] + jl[r] < p:
            l += 1
        if l < r and jl[l] + jl[r] == p:
            ans += r - l
    
    print(ans)
