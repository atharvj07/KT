from collections import Counter

# Input reading
s = input().strip()
n = int(input())

# Count frequency of characters in s
char_freq = Counter(s)

# Calculate minimum sheets needed
min_sheets = -1
sheet_str = ""
if max(char_freq.values()) > n:
    min_sheets = -1
else:
    min_sheets = 0
    sheet_chars = []
    for char, freq in char_freq.items():
        required_sheets = -(-freq // n)  # Ceiling division to find out how many sheets needed for each character
        min_sheets = max(min_sheets, required_sheets)
        sheet_chars.append(char * required_sheets)
    sheet_str = ''.join(sheet_chars)[:n]  # Construct possible sheet of stickers

# Output
if min_sheets == -1:
    print(min_sheets)
else:
    print(min_sheets)
    print(sheet_str)
