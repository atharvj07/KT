n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

difference = [a[i] - b[i] for i in range(n)]
difference.sort()

total = sum(a)
for i in range(k):
    if difference[i] < 0:
        total -= a[i] - b[i]
for i in range(k, n):
    if difference[i] < 0:
        total -= difference[i]

print(total)
