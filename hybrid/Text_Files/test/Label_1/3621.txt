#include <map>
#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, m; cin >> n >> m;
    map<string, vector<string> > rowm;
    map<string, vector<string> > colm;
    vector<string> rows(n);
    for (int i = 0; i < n; i++) cin >> rows[i];
    for (int i = 0; i < n; i++) {
        string row;
        for (int j = 0; j < m; j++) {
            row.push_back(rows[j][i]);
        }
        rowm[rows[i]].push_back(row);
        colm[row].push_back(rows[i]);
    }
    sort(rows.begin(), rows.end());
    for (int i = 0; i < n; i++) {
        for (string& r : rowm[rows[i]]) {
            for (string& c : colm[r]) {
                if (rows[i] == c) cout << i + 1 << ' ' << rows[i].find(c) + 1 << endl;
            }
        }
    }
    return 0;
}
