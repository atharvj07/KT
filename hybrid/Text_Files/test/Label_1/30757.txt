import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        int n = s.length();
        int[] open = new int[n + 1];
        int[] close = new int[n + 1];
        int ans = 0;
        int[] l = new int[n + 1];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                open[i + 1] = open[i] + 1;
                close[i + 1] = close[i];
            } else if (s.charAt(i) == '[') {
                open[i + 1] = open[i];
                close[i + 1] = close[i] + 1;
            } else if (s.charAt(i) == ')') {
                if (open[i] > 0) {
                    ans = Math.max(ans, open[i] + close[i]);
                    l[i + 1] = i;
                }
                open[i + 1] = open[i] - 1;
                close[i + 1] = close[i];
            } else if (s.charAt(i) == ']') {
                if (close[i] > 0) {
                    ans = Math.max(ans, open[i] + close[i]);
                    l[i + 1] = i;
                }
                open[i + 1] = open[i];
                close[i + 1] = close[i] - 1;
            }
        }
        System.out.println(ans);
        StringBuilder sb = new StringBuilder();
        int i = n;
        while (i > 0) {
            if (open[i] > 0) {
                sb.append('(');
            } else if (close[i] > 0) {
                sb.append('[');
            }
            i = l[i];
            if (open[i] > 0) {
                sb.append(')');
            } else if (close[i] > 0) {
                sb.append(']');
            }
            i--;
        }
        System.out.println(sb.toString());
    }
}
