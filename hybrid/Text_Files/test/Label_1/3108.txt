#include <bits/stdc++.h>
using namespace std;

int n, s, k, ans = -1;
int a[55], b[55];
char c[55];

int main() {
  cin >> n >> s >> k;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    c[i] = getchar();
  }
  for (int i = 0; i < n; i++) {
    if (c[i] == c[s - 1]) continue;
    if (a[i] < a[s - 1]) continue;
    vector<int> v;
    v.push_back(s - 1);
    v.push_back(i);
    int lst = a[s - 1];
    s = i;
    for (int j = i + 1; j < n; j++) {
      if (c[j] == c[s - 1] || a[j] <= lst) continue;
      lst = a[j];
      v.push_back(j);
      s = j;
    }
    if (v.size() > 1 && accumulate(a + v[0], a + v[v.size() - 1] + 1, 0) >= k) {
      int tmp = v.size() - 1 + abs(v[0] - v[v.size() - 1]);
      if (ans == -1 || tmp < ans) ans = tmp;
    }
    i = s;
  }
  cout << ans << endl;
  return 0;
}
