import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        boolean[] vis = new boolean[n];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] > 0) {
                cnt++;
                vis[i] = true;
            }
        }
        if (cnt <= 2) {
            System.out.println("cslnb");
            return;
        }
        boolean win = false;
        for (int i = 0; i < n; i++) {
            if (!vis[i] || arr[i] == 0) {
                continue;
            }
            for (int j = 0; j < n; j++) {
                if (!vis[j] || arr[j] == 0 || i == j) {
                    continue;
                }
                if (arr[i] == arr[j]) {
                    vis[i] = false;
                    win = true;
                    break;
                } else if (arr[i] > arr[j]) {
                    arr[i] -= arr[j];
                    vis[j] = false;
                } else {
                    arr[j] -= arr[i];
                    vis[i] = false;
                }
            }
            if (win) {
                break;
            }
        }
        if (win) {
            System.out.println("cslnb");
        } else {
            System.out.println("sjfnb");
        }
    }
}
