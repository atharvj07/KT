import sys

t = int(input())

for _ in range(t):
    a, b, k = map(int, input().split())
    a_list = list(map(int, input().split()))
    b_list = list(map(int, input().split()))
    count = 0
    for i in range(k):
        for j in range(i + 1, k):
            if a_list[i] != a_list[j] and b_list[i] != b_list[j]:
                count += 1
    print(count)
