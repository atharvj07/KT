#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int N = 3e5 + 10;

ll pfx[N], sfx[N];

int main() {
  ll n, m;
  cin >> n;
  vector<ll> a(n);
  for (ll &x : a) cin >> x;
  cin >> m;
  vector<ll> b(m);
  for (ll &x : b) cin >> x;
  bool eq = 1;
  for (ll i = 0; i < n; i++) {
    if (a[i] != b[i]) { eq = 0; break; }
  }
  if (eq) {
    cout << n << endl;
    return 0;
  }
  ll sum = 0;
  for (ll i = 0; i < n; i++) {
    if (i) pfx[i] = pfx[i - 1];
    pfx[i] += a[i];
  }
  for (ll i = m - 1; i >= 0; i--) {
    if (i < m - 1) sfx[i] = sfx[i + 1];
    sfx[i] += b[i];
  }
  ll ans = -1;
  for (ll i = 0; i < n; i++) {
    for (ll j = 0; j < m; j++) {
      ll csum = 0;
      if (i > 0) csum = pfx[i - 1];
      csum += sfx[j];
      if (i < n - 1) csum += a[i + 1];
      if (j > 0) csum += b[j - 1];
      if (csum == a[i]) {
        ans = max(ans, i + m - j);
      }
    }
  }
  cout << ans << endl;
}
