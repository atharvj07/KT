import java.util.*;
public class Main {
public static void main(String[] args) {
     Scanner in = new Scanner(System.in);
     int n = in.nextInt();
     int[] a = new int[n];
     for (int i = 0; i < n; i++) {
         a[i] = in.nextInt();
     }
     int[] l = new int[n], r = new int[n], c = new int[n];
     if (a[0] == 0) {
         System.out.println("NO");
     } else {
         System.out.println("YES");
         int j = 0, ans = 0;
         for (int i = 0; i < n; i++) {
             if (a[i] == 0) {
                 l[j] = i;
                 r[j] = i;
                 j++;
                 if (ans > 0) {
                     c[ans - 1]++;
                 }
                 ans = 0;
             } else {
                 ans++;
             }
         }
         if (ans > 0) {
             c[ans - 1]++;
         }
         System.out.println(j);
         for (int i = 0; i < j; i++) {
             System.out.println((l[i] + 1) + " " + (r[i] + 1));
             for (int k = 0; k < c[i]; k++) {
                 if (i == 0) {
                     System.out.println((l[i] + 2) + " " + (r[i] + 1));
                 } else {
                     System.out.println((l[i] + 1) + " " + (r[i]));
                 }
             }
         }
     }
}
}
