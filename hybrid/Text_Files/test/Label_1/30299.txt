import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        outer: while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }
            int[][] edges = new int[n][4];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 4; j++) {
                    edges[i][j] = sc.nextInt();
                }
            }
            int k = sc.nextInt();
            for (int i = 0; i < k; i++) {
                int p = sc.nextInt();
                int q = sc.nextInt();
                int r = sc.nextInt();
                int[][] d = new int[m][2];
                for (int j = 0; j < m; j++) {
                    Arrays.fill(d[j], Integer.MAX_VALUE);
                }
                d[p - 1][r] = 0;
                while (true) {
                    boolean updated = false;
                    for (int[] edge : edges) {
                        int a = edge[0] - 1;
                        int b = edge[1] - 1;
                        int cost = edge[2];
                        int time = edge[3];
                        if (d[a][r] != Integer.MAX_VALUE && d[a][r] + cost < d[b][r]) {
                            d[b][r] = d[a][r] + cost;
                            updated = true;
                        }
                        if (d[b][r] != Integer.MAX_VALUE && d[b][r] + cost < d[a][r]) {
                            d[a][r] = d[b][r] + cost;
                            updated = true;
                        }
                        if (d[a][r] != Integer.MAX_VALUE && d[a][r] + time < d[b][r]) {
                            d[b][r] = d[a][r] + time;
                            updated = true;
                        }
                        if (d[b][r] != Integer.MAX_VALUE && d[b][r] + time < d[a][r]) {
                            d[a][r] = d[b][r] + time;
                            updated = true;
                        }
                    }
                    if (!updated) {
                        break;
                    }
                }
                System.out.println(d[q - 1][r]);
            }
        }
    }
}
