#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2100, MAXM = 220;
int N, M, mp[MAXN][MAXM], revmp[MAXN][MAXM], matched;
char g1[MAXN][MAXM], g2[MAXM][MAXN];
bool flg[MAXN][MAXN];
int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
    cin >> N >> M;
    for (int i = 1; i <= N; i++) {
        for (int j = 1; j <= M; j++) {
            cin >> g1[i][j];
            mp[i][j] = g1[i][j] - 'a' + 1;
        }
    }
    for (int i = 1; i <= M; i++) {
        for (int j = 1; j <= N; j++) {
            cin >> g2[i][j];
            revmp[i][j] = g2[i][j] - 'a' + 1;
        }
    }
    for (int i = 1; i <= N - M + 1; i++) {
        for (int j = 1; j <= N - M + 1; j++) {
            bool ok = 1;
            for (int p = 0; p < M; p++) {
                for (int q = 0; q < M; q++) {
                    if (mp[i + p][j + q] != revmp[q + 1][p + 1]) {
                        ok = 0;
                        break;
                    }
                }
            }
            if (ok) {
                cout << i << " " << j << endl;
                return 0;
            }
        }
    }
    return 0;
}
