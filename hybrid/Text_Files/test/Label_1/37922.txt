import sys

r = []
h = []
for _ in range(int(input())):
    a, b = map(int, sys.stdin.readline().split())
    r.append(a)
    h.append(b)

pi = 3.141592653589793
s = sorted(zip(h, r), reverse=True)

v, x = 0, 0.
for i in s:
    t = pi * i[1] ** 2 * i[0]
    if i[0] > x:
        x = i[0]
        v += t

print(v)
