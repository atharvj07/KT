import java.util.*;
import java.io.*;

public class Main {

    static class Point {
        long x, y;
        Point(long x, long y) {
            this.x = x;
            this.y = y;
        }
    }

    static long ccw(Point p1, Point p2, Point p3) {
        return (p2.x - p1.x) * (p3.y - p1.y) - (p3.x - p1.x) * (p2.y - p1.y);
    }

    static boolean convex(Point[] points) {
        int n = points.length;
        for (int i = 0; i < n; i++) {
            if (ccw(points[i], points[(i + 1) % n], points[(i + 2) % n]) < 0) {
                return false;
            }
        }
        return true;
    }

    static long perimeter(Point[] points, int k) {
        long ans = 0;
        for (int i = 0; i < k; i++) {
            ans += Math.abs(points[i].x - points[(i + 1) % k].x) + Math.abs(points[i].y - points[(i + 1) % k].y);
        }
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            String[] line = in.readLine().split(" ");
            points[i] = new Point(Long.parseLong(line[0]), Long.parseLong(line[1]));
        }
        if (!convex(points)) {
            System.out.println(-1);
            return;
        }
        long[] f = new long[n + 1];
        for (int k = 3; k <= n; k++) {
            for (int i = 0; i < n; i++) {
                f[k] = Math.max(f[k], perimeter(points, k) + f[k - 1]);
            }
        }
        for (int k = 3; k <= n; k++) {
            System.out.print(f[k] + " ");
        }
    }
}
