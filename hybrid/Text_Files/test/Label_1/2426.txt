#include<iostream>
#include<vector>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    long long sum = 0;
    for (int i = 0; i < n; i++)
        sum += a[i];
    long long avg = (sum + k - 1) / k;
    long long ans = 0;
    for (int i = 0; i < n; i++)
        if (a[i] > avg)
            ans += a[i] - avg;
    cout << ans << endl;
}
