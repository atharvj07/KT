#include <bits/stdc++.h>
using namespace std;
char a[105][10], b[105][10];
int main(){
    int n, ans = 0, p;
    cin >> n;
    for (int i = 1; i <= n; i++){
        cin >> a[i] >> b[i];
        p = strlen(a[i]);
        for (int j = 1; j < p; j++)
            if(a[i][j] == b[i][j] && isalpha(a[i][j])) continue;
            else{
                if(a[i][j] == b[i][j] - 32) ans++;
                else if(a[i][j] == b[i][j] + 32) ans += 2;
                else ans++;
            }        
    }
    cout << ans << endl;
}
