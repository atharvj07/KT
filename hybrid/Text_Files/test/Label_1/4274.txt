#include <iostream>

using namespace std;

int main() {
    long long n;
    cin >> n;
    long long ans = 0;
    for (long long i = 9; i * i <= n; i++) {
        while (n % i == 0) {
            if (ans == 0) ans = i;
            else ans *= i;
            n /= i;
        }
    }
    if (n > 1) {
        if (ans == 0) ans = n;
        else ans *= n;
    }
    cout << ans << endl;
    return 0;
}
