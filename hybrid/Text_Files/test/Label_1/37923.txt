import sys
t=int(input())
for _ in range(t):
    s=input()
    z=s.count('0')
    o=s.count('1')
    mn=min(z,o)
    l=[-1]*len(s)
    r=[-1]*len(s)
    st=[]
    for i in range(len(s)):
        if s[i]=='0':
            z-=1
            if z>=0 and z<=mn:
                l[i]=st[-1] if st else i
            st.append(i)
        else:
            o-=1
            if o>=0 and o<=mn:
                r[len(s)-i-1]=st[-1] if st else len(s)-i-1
            st.append(len(s)-i-1)
    for i in range(len(s)):
        if l[i]!=-1:
            sb=s[l[i]+1:r[i]+1]
            sb=sb[::-1]
            sb=sb.replace('0','2').replace('1','0').replace('2','1')
            s=s[:l[i]+1]+sb+s[r[i]+1:]
    print(s)
