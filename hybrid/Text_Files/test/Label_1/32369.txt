import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] c = in.next().toCharArray();
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (c[i] == 'A' || c[i] == 'F') {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
