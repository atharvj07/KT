import sys

n, m = [int(i) for i in sys.stdin.readline().split()]
arr = [list(i) for i in sys.stdin.read().splitlines()]

if any('*' in i for i in zip(*arr)):
    print(-1)
elif all('*' in i for i in arr):
    print(0)
else:
    print(n * m)
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if arr[i - 1][j - 1] == '.':
                print(i, j, 0)
