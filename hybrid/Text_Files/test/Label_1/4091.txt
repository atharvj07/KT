#include <bits/stdc++.h>

using namespace std;

const int MX = 4e4 + 5, INF = 1e9;

struct node {
    int mx, red, grn, dp[2][2];
    int ans;
} T[4 * MX];

int a, b, n, h[MX];

void build(int p, int s, int e) {
    if(s == e) {
        T[p].mx = h[s];
        T[p].dp[0][0] = T[p].dp[0][1] = T[p].dp[1][0] = 0;
        T[p].dp[1][1] = h[s];
        T[p].ans = h[s];
        return;
    }
    int mid = (s + e) / 2;
    build(p * 2, s, mid);
    build(p * 2 + 1, mid + 1, e);
    T[p].mx = max(T[p * 2].mx, T[p * 2 + 1].mx);
    T[p].dp[0][0] = T[p * 2].dp[0][0];
    T[p].dp[0][1] = max(T[p * 2].dp[0][1], T[p * 2].mx + T[p * 2 + 1].dp[0][0]);
    T[p].dp[1][0] = max(T[p * 2].dp[1][0], T[p * 2].mx + T[p * 2 + 1].dp[1][1]);
    T[p].dp[1][1] = max(max(T[p * 2].dp[1][1], T[p * 2 + 1].dp[1][1]), T[p * 2].mx + T[p * 2 + 1].dp[0][1]);
    T[p].ans = max(T[p * 2].ans, T[p * 2 + 1].ans);
}

void query(int p, int s, int e, int l, int r, int tp) {
    if(s > r || e < l)
        return;
    if(s >= l && e <= r) {
        if(tp)
            T[p].ans = max(T[p].ans, T[p].dp[0][1]);
        else
            T[p].ans = max(T[p].ans, T[p].dp[1][0]);
        return;
    }
    int mid = (s + e) / 2;
    query(p * 2, s, mid, l, r, tp);
    query(p * 2 + 1, mid + 1, e, l, r, tp);
}

int main() {
    cin >> n >> a >> b;
    for(int i = 1; i <= n; i++)
        cin >> h[i];
    build(1, 1, n);
    int ans = INF;
    for(int i = 1; i <= n; i++) {
        query(1, 1, n, 1, i - 1, 1);
        query(1, 1, n, i + 1, n, 0);
        ans = min(ans, T[1].ans + h[i]);
    }
    if(ans != INF)
        cout << ans << endl;
    else
        cout << -1 << endl;
    return 0;
}
