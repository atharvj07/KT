def solve():
    n = int(input())
    s = input()
    freq = {}
    for i in range(n):
        cur = s[i:] + s[:i]
        if cur not in freq: freq[cur] = 0
        freq[cur] += 1
    mx=0
    for v in freq.values(): mx = max(mx, v)
    ans = 0
    for v in freq.values():
        ans += v == mx
    return ans % 1000000007

print(solve())
