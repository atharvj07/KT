import java.util.*;

class Main {
    boolean[][] A;

    void dfs(int x, int y, boolean f) {
        A[x][y] = f;
        int[] dx = { -1, 0, 1, 0 };
        int[] dy = { 0, 1, 0, -1 };
        for (int k = 0; k < 4; k++) {
            int nx = x + dx[k];
            int ny = y + dy[k];
            if (nx < 0 || ny < 0 || nx >= A.length || ny >= A[nx].length)
                continue;
            if (A[nx][ny])
                continue;
            dfs(nx, ny, !f);
        }
    }

    void run() {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        A = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                A[i][j] = (s.charAt(j) == '#');
            }
        }
        boolean ok = true;
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                if (!A[i][j]) {
                    dfs(i, j, false);
                    ok = false;
                    break;
                }
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                if (A[i][j]) {
                    System.out.println("Yes");
                    return;
                }
        System.out.println("No");
    }

    public static void main(String[] args) {
        new Main().run();
    }
}
