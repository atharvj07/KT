#include <bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while(t--){
        int n, m, a[55][55]; cin >> n >> m;
        for(int i = 1; i <= n; i++) for(int j = 1; j <= m; j++) cin >> a[i][j];
        bool ok = 0, no = 0;
        for(int i = 1; i <= n; i++){
            bool row = 0;
            for(int j = 1; j <= m; j++){
                if(a[i][j]) row = 1;
            }
            if(!row){
                if(ok) no = 1;
                ok = 1;
            }
        }
        for(int j = 1; j <= m; j++){
            bool col = 0;
            for(int i = 1; i <= n; i++){
                if(a[i][j]) col = 1;
            }
            if(!col){
                if(ok) no = 1;
                ok = 1;
            }
        }
        if(no) cout << "Ashish" << endl;
        else cout << "Vivek" << endl;
    }
}
