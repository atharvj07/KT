import java.util.*;

class Main {
    static boolean[][] reach;
    static int[] p;
    static int[] pPos;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int d = sc.nextInt();
        int[] c = new int[m];
        for (int i = 0; i < m; i++) {
            c[i] = sc.nextInt();
        }
        reach = new boolean[n + 2][d + 1];
        p = new int[n + 2];
        pPos = new int[m + 1];
        int l = 1, r = m;
        while (r - l > 1) {
            int mid = (l + r) / 2;
            if (check(c, mid, n, d)) {
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        if (r == m) {
            System.out.println("YES");
            for (int i = 1; i < n + 1; i++) {
                System.out.print(p[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }

    public static boolean check(int[] c, int cnt, int n, int d) {
        int lft = 1;
        for (int i = 1; i <= cnt; i++) {
            lft += c[i - 1];
        }
        Arrays.fill(reach[0], false);
        reach[0][0] = true;
        for (int i = 1; i < n + 2; i++) {
            Arrays.fill(reach[i], false);
            p[i] = 0;
            if (i >= d + 1) {
                reach[i][0] = true;
            }
            for (int j = 0; j <= d; j++) {
                if (reach[i][j] == false) {
                    continue;
                }
                if (i + j <= n) {
                    reach[i + j][j] = true;
                }
                if (i - j >= 1) {
                    reach[i - j][j] = true;
                }
                for (int k = 1; k <= cnt; k++) {
                    if (i + j >= lft && i + j < lft + c[k - 1]) {
                        reach[i + j][j] = true;
                        p[i + j] = k;
                    }
                    if (i - j > pPos[k] && i - j <= pPos[k] + c[k - 1]) {
                        reach[i - j][j] = true;
                        p[i - j] = k;
                    }
                }
            }
            for (int k = 1; k <= cnt; k++) {
                lft += c[k - 1];
                pPos[k] = lft;
            }
        }
        for (int i = 0; i <= d; i++) {
            if (reach[n + 1][i]) {
                return true;
            }
        }
        return false;
    }
}
