#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
ll n,x;
vi color;
int f[100005][11],pre[100005][11],L,R;

int main()
{
  cin>>n;
  color.pb(0);
  for(int i=0;i<n;i++)
  {
     int a;
     cin>>a;
     color.pb(a);
  }
  for(int i=1;i<=n;i++)
  {
     for(int j=1;j<=10;j++)
     {
        pre[i][j]=pre[i-1][j];
        if(color[i]==j)pre[i][j]++;
     }
  }
   for(int i=n;i>=1;i--)
   {
      for(int j=1;j<=10;j++)
      {
         f[i][j]=f[i+1][j];
         if(color[i]==j)f[i][j]++;
      }
   }
   for(int i=1;i<=n;i++)
   {
      for(int j=1;j<=n;j++)
      {
         bool ok=1;
         for(int k=1;k<=10;k++)
         {
            ok&=f[j][k]-pre[i-1][k]==pre[j][k]-pre[i-1][k];
         }
         if(ok)
         {
            if(j-i+1>x)
            {
               x=j-i+1;
               L=i,R=j;
            }
         }
      }
   }
   cout<<x<<endl;
   return 0;
}
