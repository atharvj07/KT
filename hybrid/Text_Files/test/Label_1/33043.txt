import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int prevX = sc.nextInt();
        int prevY = sc.nextInt() + h;
        double dist = 0;
        for (int i = 1; i < n; i++) {
            int curX = sc.nextInt();
            int curY = sc.nextInt() + h;
            double dy = curY - prevY;
            double dx = curX - prevX;
            double len = Math.sqrt(dy * dy + dx * dx);
            if (dy / dx >= 1.0) {
                dist += len;
            } else {
                double interX = prevX + (dx * h) / dy;
                if (interX >= prevX && interX <= curX) {
                    dist += len;
                }
            }
            prevX = curX;
            prevY = curY;
        }
        System.out.println(dist);
    }
}
