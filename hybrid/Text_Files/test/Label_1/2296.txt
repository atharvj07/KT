#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<string> ans;
    for (int i = 0; i < n; i++) {
      string s;
      s += '(';
      for (int j = 1; j < 2 * n; j++) {
        if (j % 2) {
          s += ')';
        } else {
          int x = rand() % 2;
          if (x) {
            s += '(';
          } else {
            s += ')';
          }
        }
      }
      ans.push_back(s);
    }
    for (auto x : ans) {
      cout << x << endl;
    }
  }
  return 0;
}
