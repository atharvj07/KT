n, k = map(int, input().split())
a = sorted(list(map(int, input().split())))
x = [[] for _ in range(n)]
for i in range(n*k):
    x[i//n].append(i+1)
for i in range(n*k):
    for j in range(n):
        if a[j]==x[j][i//n]:
            x[j][i//n], x[j][n-1-i//k] = x[j][n-1-i//k], x[j][i//n]
for i in x:
    print(*i)
