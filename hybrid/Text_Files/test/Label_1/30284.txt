import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int T, t[], N, D[], M[];
        while (true) {
            T = in.nextInt();
            if (T == 0) break;
            t = new int[T];
            for (int i = 0; i < T; i++) t[i] = in.nextInt();
            N = in.nextInt();
            D = new int[N];
            M = new int[N];
            for (int i = 0; i < N; i++) {
                D[i] = in.nextInt();
                M[i] = in.nextInt();
            }
            int ans = 0;
            for (int i = 0; i < N; i++) {
                if (D[i] < ans) ans += T;
                ans = (ans + D[i]) % T + 1;
                if (t[ans - 1] < M[i]) ans++;
            }
            System.out.println(ans);
        }
    }
}
