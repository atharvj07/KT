#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        string s;
        cin >> s; n = s.size();
        for(int i = 0; i < n - 1; i++) {
            if(s[i] % 2 != s[i + 1] % 2) {
                swap(s[i], s[i + 1]);
                break;
            }
        }
        while(s.size() > 1 && s[0] == '0') s.erase(0, 1);
        cout << s << "\n";
    }
    return 0;
}
