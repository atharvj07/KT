#include <iostream>
#include <limits>

using namespace std;

int main() {
    string s;
    getline(cin, s);
    bool upper = false;
    bool lower = false;
    bool digit = false;
    bool special = false;
    for (char c : s) {
        upper |= isupper(c);
        lower |= islower(c);
        digit |= isdigit(c);
        special |= c == '!' || c == '?' || c == '.' || c == ',' || c == '_';
    }
    cout << ((s.size() >= 5 && upper && lower && digit && special) ? "Correct" : "Too weak") << '\n';
    return 0;
}
