s=input();n=int(input())
a=[0]*26
for x in s:a[ord(x)-ord('a')]+=1
b=a+[0]
for i in range(25,-1,-1):
	k=a[i]//n
	b[i]=k
	for j in range(i+1,26):a[j]-=k*min(a[j],n)
if sum(b)<len(s):print(-1)
else:
	r=[]
	for i in range(26):
		for j in range(b[i]):r+=chr(ord('a')+i)
	print(len(s)//n+1)
	print(''.join(r))
