#include<bits/stdc++.h>
using namespace std;
int main() {
    int x1, y1, x2, y2, x3, y3, x4, y4, x5, y5, x6, y6;
    cin >> x1 >> y1 >> x2 >> y2 >> x3 >> y3 >> x4 >> y4 >> x5 >> y5 >> x6 >> y6;
    if ((x1 < x3 && x2 < x3) || (x1 > x4 && x2 > x4) || (y1 < y3 && y2 < y3) || (y1 > y4 && y2 > y4) ||
        (x1 < x5 && x2 < x5) || (x1 > x6 && x2 > x6) || (y1 < y5 && y2 < y5) || (y1 > y6 && y2 > y6))
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
}
