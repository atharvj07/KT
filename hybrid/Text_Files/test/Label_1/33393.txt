import java.util.*;

public class Main {
    private static class Edge {
        int to;
        int val;

        public Edge(int to, int val) {
            this.to = to;
            this.val = val;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<List<Edge>> adj = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }
        Set<Integer> vals = new HashSet<>();
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            int val = scanner.nextInt();
            adj.get(u).add(new Edge(v, val));
            adj.get(v).add(new Edge(u, val));
            vals.add(val);
        }
        List<Integer> valsList = new ArrayList<>(vals);
        Collections.sort(valsList);
        int[][] dp = new int[n][valsList.size() + 1];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        if (dfs(0, -1, adj, valsList, dp) == 0) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            List<String> operations = new ArrayList<>();
            int u = 0;
            int val = 0;
            for (int i = 1; i < n; i++) {
                if (dp[i][val] != dp[u][val]) {
                    operations.add((u + 1) + " " + (i + 1) + " " + (valsList.get(val) - valsList.get(dp[i][val])));
                    u = i;
                    val = dp[i][val];
                }
            }
            System.out.println(operations.size());
            for (String operation : operations) {
                System.out.println(operation);
            }
        }
    }

    private static int dfs(int u, int p, List<List<Edge>> adj, List<Integer> valsList, int[][] dp) {
        if (dp[u][0] != -1) {
            return dp[u][0];
        }
        dp[u][0] = 0;
        for (Edge edge : adj.get(u)) {
            if (edge.to == p) {
                continue;
            }
            for (int i = 0; i < valsList.size(); i++) {
                int val = valsList.get(i);
                if (dp[edge.to][i] != -1 && edge.val == valsList.get(dp[edge.to][i])) {
                    dp[u][i + 1] = i;
                    break;
                }
            }
        }
        return dp[u][0];
    }
}
