from collections import Counter

n=int(input())
A=[int(x) for x in input().split()]
ans=[0]*n
for i in range(n):
    B=[]
    C=Counter()
    for j in range(i,n):
        B.append(A[j])
        C[A[j]]+=1
        if C[abs(A[j])]&1==0: C[abs(A[j])]-=1
        else: C[abs(A[j])]=1
        ans[len(B)-C[0]]+=1
print(*ans)
