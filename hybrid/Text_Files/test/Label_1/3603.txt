#include <bits/stdc++.h>
using namespace std;
int n;
string s, t;
int main() {
    cin >> n;
    cin >> s >> t;
    bool f = 0;
    for (int i = 0; i < n; ++i) {
        if (s[i] != t[i]) {
            f = 1;
            break;
        }
    }
    if (f) {
        puts("-1");
    } else {
        puts("0");
    }
}
