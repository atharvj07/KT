#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int last = 0, ans = 0, pre = 0;
    vector<int> prv(x, -1);
    for (int i = 0; i < n; i++) {
      pre = (pre + a[i]) % x;
      if (prv[pre] != -1) {
        ans = max(ans, i - prv[pre]);
      } else
        prv[pre] = i;
    }
    if (prv[0] != -1) ans = max(ans, n - prv[0]);
    if (ans == 0)
      cout << -1 << endl;
    else
      cout << ans << endl;
  }
  return 0;
}
