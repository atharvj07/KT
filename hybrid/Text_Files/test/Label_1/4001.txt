#include <iostream>

int main() {
    std::string s;
    std::cin >> s;

    int c0 = 0, c6 = 0;
    for (char c : s) {
        if (c == '0') {
            c0++;
        } else {
            c6++;
        }
    }

    while (c6 >= 6) {
        c6 -= 6;
        c0++;
    }

    if (c0 % 2 == 0 && c6 == 0) {
        std::cout << "yes" << std::endl;
    } else {
        std::cout << "no" << std::endl;
    }

    return 0;
}
