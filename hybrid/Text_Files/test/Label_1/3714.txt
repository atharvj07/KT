#include<iostream>
#include<unordered_map>
using namespace std;

int main(){
    int cases;
    cin>>cases;
    while(cases--){
        int n;
        cin>>n;
        unordered_map<char,int> freq;
        string str;
        cin>>str;

        bool flag=true;
        for(int i=0;i<n;++i){
            freq[str[i]]++;
        }
        int sum=0;
        for(auto i:freq){
            if(i.second%2 == 1) ++sum;
        }
        if(sum<=1 || (freq[str[0]]%2 == 1 && sum==2)) cout<<sum/2<<'\n';
        else cout<<-1<<'\n';
    }
    return 0;
}
