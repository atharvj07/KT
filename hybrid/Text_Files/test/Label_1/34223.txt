import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int q = scan.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = scan.nextInt();
        }
        int[] cumulativeAnd = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            cumulativeAnd[i] = cumulativeAnd[i - 1] & a[i];
        }
        for (int i = 0; i < q; i++) {
            int x = scan.nextInt();
            int y = scan.nextInt();
            if (cumulativeAnd[y] > cumulativeAnd[x - 1]) {
                System.out.println("Shi");
            } else {
                System.out.println("Fou");
            }
        }
    }
}
