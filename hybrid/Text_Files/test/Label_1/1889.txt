#include<bits/stdc++.h>
#define int long long int
using namespace std;
const int mod=1e9+7;

int p[100005];
int lz(int x){
    return (x&(-x));
}
void update(int x,int val,int n){
    while(x<=n){
        p[x]+=val;
        x+=lz(x);
    }
}
int query(int x){
    int sum=0;
    while(x>0){
        sum+=p[x];
        x-=lz(x);
    }
    return sum;
}

int32_t main(){
    int sum,n;
    cin>>sum>>n;
    if(sum>n*(n+1)/2){
        cout<<-1<<endl;
        return 0;
    }
    for(int i=1;i<=n;i++){
        int low=0,high=n+1,pos=-1;
        while(low<=high){
            int mid=(low+high)/2;
            int val=query(mid);
            if(val==sum){
                pos=mid;
                break;
            }
            if(val<sum)low=mid+1;
            else high=mid-1;
        }
        if(pos==-1){
            cout<<-1<<endl;
            return 0;
        }
        update(pos,1,n);
        cout<<pos<<" ";
        sum-=pos;
    }
}
