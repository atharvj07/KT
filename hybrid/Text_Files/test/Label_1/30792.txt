import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        int maxSum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (arr[i][j] != 0) {
                    maxSum = Math.max(maxSum, dfs(arr, i, j, 0, 0));
                }
            }
        }
        System.out.println(maxSum);
    }

    public static int dfs(int[][] arr, int x, int y, int sum, int dir) {
        if (x < 0 || x >= arr.length || y < 0 || y >= arr[0].length || arr[x][y] == 0) {
            return sum;
        }
        sum += arr[x][y];
        arr[x][y] = 0;
        int[] dx = {0, 1, 0, -1};
        int[] dy = {1, 0, -1, 0};
        return dfs(arr, x + dx[dir], y + dy[dir], sum, dir) + dfs(arr, x + dx[(dir + 1) % 4], y + dy[(dir + 1) % 4], sum, (dir + 1) % 4);
    }
}
