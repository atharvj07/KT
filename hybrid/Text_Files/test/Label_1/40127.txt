from sys import stdin

n = int(stdin.readline())
s, g, ans = [], [], []
for _ in range(n):
    a, b = map(int, stdin.readline().split())
    s.append(a), g.append(b)

for i in range(n):
    if s[i] == s[max(0, i - 1)]:
        ans.append(s[i])
    elif s[i] == s[i - 1] + 1:
        ans.append(s[i - 1])
    else:
        diff = s[i] - s[i - 1]
        ans.append(s[i - 1] - diff // 2)

for i in range(n - 1):
    if ans[i] == ans[i + 1] and ans[i] != s[i] and ans[i] != s[i] + g[i]:
        if g[i] > g[i + 1]:
            ans[i] += min(g[i] - g[i + 1], g[i] // 2)
        else:
            ans[i + 1] += min(g[i + 1] - g[i], g[i + 1] // 2)

for i in range(n):
    g[i] -= ans[i] - s[i]

if max(g) > min(g):
    print(-1)
else:
    print(sum(g))
    print(*ans)
