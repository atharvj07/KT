#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;

  string s;
  cin >> s;

  if (k > (n + 1) / 2) {
    cout << -1 << endl;
    return 0;
  }

  sort(s.begin(), s.end());

  int ans = 0;
  for (int i = 0; i < k; i++) {
    ans += s[i] - 'a' + 1;
  }

  cout << ans << endl;

  return 0;
}
