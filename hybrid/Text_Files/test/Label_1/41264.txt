n=int(input())
a=list()
c=list()
for i in range(n):
  x,y=list(map(int,input().split()))
  a+=[x]
  c+=[y]
inf=10**18
g=[[inf for i in range(n)] for _ in range(n)]
for i in range(n):
  for j in range(n):
    if i!=j:
      g[i][j]=max(c[i],a[j]-a[i])
for k in range(n):
  for i in range(n):
    for j in range(n):
      g[i][j]=min(g[i][j],g[i][k]+g[k][j])
print(min(g[0]))
