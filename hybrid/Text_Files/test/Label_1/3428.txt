#include<bits/stdc++.h>
using namespace std;
#define MAXN 105
#define MAXM 55
char a[MAXN][MAXM];
int dp[MAXN][MAXN][2];
int n, m, len;

int plps(int l, int r, int f) {
    if(l == r) { return f ? 1 : 0; }
    if(l + 1 == r) { return f ? (a[l][0] == a[r][0]) : 0; }
    int &ret = dp[l][r][f];
    if(~ret) return ret;
    ret = 0;
    if(a[l][len - 1] == a[r][len - 1]) {
        if(l == 0 || r == n - 1) {
            ret = max(ret, 2 + plps(l + 1, r - 1, 1));
        } else {
            int temp = plps(l + 1, r - 1, 1);
            if(temp != -1) { ret = max(ret, 2 + temp); }
        }
    }
    if(f == 0) {
        if(l + 1 <= r) ret = max(ret, plps(l + 1, r, 0));
        if(l <= r - 1) ret = max(ret, plps(l, r - 1, 0));
    }
    if(ret) len = ret;
    return ret;
}

string ans;

void print(int l, int r, int f) {
    if(l == r) {
        if(f) ans += a[l][len - 1];
        return;
    }
    if(l + 1 == r) {
        if(f) ans += a[l][len - 1] + a[r][len - 1];
        return;
    }
    if(a[l][len - 1] == a[r][len - 1]) {
        if(l == 0 || r == n - 1) {
            ans += a[l][len - 1];
            print(l + 1, r - 1, 1);
            ans += a[r][len - 1];
            return;
        } else {
            int temp = plps(l + 1, r - 1, 1);
            if(temp != -1) {
                ans += a[l][len - 1];
                print(l + 1, r - 1, 1);
                ans += a[r][len - 1];
                return;
            }
        }
    }
    if(f == 0) {
        if(l + 1 <= r) {
            print(l + 1, r, 0);
        }
        if(l <= r - 1) {
            print(l, r - 1, 0);
        }
    }
}

int main() {
    memset(dp, -1, sizeof(dp));
    scanf("%d %d", &n, &m);
    len = m;
    for(int i = 0; i < n; i++) {
        scanf("%s", a[i]);
    }
    int ret = plps(0, n - 1, 0);
    if(ret) {
        printf("%d\n", ret);
        print(0, n - 1, 0);
        printf("%s\n", ans.c_str());
    }
    else puts("0");
    return 0;
}
