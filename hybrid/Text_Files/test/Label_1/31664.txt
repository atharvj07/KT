import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long k = input.nextLong();
        long a = input.nextLong();
        long b = input.nextLong();
        if (a < b) {
            System.out.println(-1);
        } else {
            long ans = (k + b - 1) / (a - b);
            if (ans % 2 == 0) {
                ans++;
            }
            System.out.println(ans);
        }
    }
}
