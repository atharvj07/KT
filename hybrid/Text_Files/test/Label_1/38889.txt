import sys

def input(): return sys.stdin.readline().strip()
def read_int_list(): return list(map(int, input().split()))

n, h, m, k = read_int_list()
x = [read_int_list() for i in range(n)]
k = min(m//2, k)
x.sort()
l = 0
r = (m + 1)//2
ans = n
ans_t = 0
while(l <= r):
    t = (l + r)//2
    c = 0
    for i in x:
        if((i[1] >= t and i[1] < t + k) or (i[1] < k and i[0] == h-1)):
            c += 1
    if(c <= ans):
        ans = c
        ans_t = t
    if(c > 0):
        r = t - 1
    else:
        l = t + 1

print(ans, ans_t)
for i in x:
    if((i[1] >= ans_t and i[1] < ans_t + k) or (i[1] < k and i[0] == h-1)):
        print(i[0]+1, i[1]+1)
