import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    static class Gem {
        int x, y;

        public Gem(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static class Command {
        char d;
        int l;

        public Command(char d, int l) {
            this.d = d;
            this.l = l;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }

            List<Gem> gems = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                int y = sc.nextInt();
                gems.add(new Gem(x, y));
            }

            int m = sc.nextInt();
            List<Command> commands = new ArrayList<>();
            for (int i = 0; i < m; i++) {
                char d = sc.next().charAt(0);
                int l = sc.nextInt();
                commands.add(new Command(d, l));
            }

            boolean[][] visited = new boolean[21][21];
            visited[10][10] = true;

            for (Command command : commands) {
                int dx = 0, dy = 0;
                switch (command.d) {
                    case 'N':
                        dy = -1;
                        break;
                    case 'E':
                        dx = 1;
                        break;
                    case 'S':
                        dy = 1;
                        break;
                    case 'W':
                        dx = -1;
                        break;
                }

                for (int i = 0; i < command.l; i++) {
                    int x = 10 + dx * i;
                    int y = 10 + dy * i;
                    if (x < 0 || x > 20 || y < 0 || y > 20) {
                        break;
                    }

                    visited[x][y] = true;
                }
            }

            boolean allCollected = true;
            for (Gem gem : gems) {
                if (!visited[gem.x][gem.y]) {
                    allCollected = false;
                }
            }

            System.out.println(allCollected ? "Yes" : "No");
        }
    }
}
