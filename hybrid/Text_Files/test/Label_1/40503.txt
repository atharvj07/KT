for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    s = sum(a)
    b = [a[0]]
    for i in range(1, n):
        if b[-1] % a[i] == 0:
            b.append(b[-1])
        elif a[i] % b[-1] == 0:
            b.append(a[i])
        else:
            b.extend([a[i], b[-1]])
    for i in range(1, n):
        b[i] = min(b[i], s - b[i])
    print(*b)
