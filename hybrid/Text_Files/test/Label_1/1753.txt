#include <bits/stdc++.h>
#define ll long long

using namespace std;

const ll MOD = 1e9 + 7;
const int MAX = 2 * 1e5 + 5;
ll dp[MAX][6];
bool vis[MAX];

vector<pair<ll, char>> graph[MAX];
ll n, k;
ll res = 6 * 6 * 6;

void dfs(ll root, ll par) {
    vis[root] = 1;
    for (auto i: graph[root]) {
        ll child = i.first;
        char color = i.second;
        if (vis[child]) continue;
        if (color == 'W') dp[child][0] = dp[child][1] = dp[child][2] = 0;
        else if (color == 'Y') dp[child][0] = dp[child][1] = dp[child][3] = 0;
        else if (color == 'G') dp[child][0] = dp[child][2] = dp[child][4] = 0;
        else if (color == 'B') dp[child][0] = dp[child][2] = dp[child][5] = 0;
        else if (color == 'R') dp[child][1] = dp[child][3] = dp[child][4] = 0;
        else dp[child][1] = dp[child][3] = dp[child][5] = 0;
        dfs(child, root);
        dp[root][0] = (dp[root][0] + dp[child][1] + dp[child][2] + dp[child][3] + dp[child][4] + dp[child][5]) % MOD;
        dp[root][1] = (dp[root][1] + dp[child][0] + dp[child][2] + dp[child][3] + dp[child][4] + dp[child][5]) % MOD;
        dp[root][2] = (dp[root][2] + dp[child][0] + dp[child][1] + dp[child][3] + dp[child][4] + dp[child][5]) % MOD;
        dp[root][3] = (dp[root][3] + dp[child][0] + dp[child][1] + dp[child][2] + dp[child][4] + dp[child][5]) % MOD;
        dp[root][4] = (dp[root][4] + dp[child][0] + dp[child][1] + dp[child][2] + dp[child][3] + dp[child][5]) % MOD;
        dp[root][5] = (dp[root][5] + dp[child][0] + dp[child][1] + dp[child][2] + dp[child][3] + dp[child][4]) % MOD;
    }
}

int main() {
    cin >> k >> n;
    k++;
    for (ll i = 0; i < (1 << k); i++) dp[i][0] = dp[i][1] = dp[i][2] = dp[i][3] = dp[i][4] = dp[i][5] = 1;
    for (ll i = 0; i < n; i++) {
        ll p;
        string s;
        cin >> p >> s;
        if (s == "white") graph[p].push_back({p, 'W'});
        else if (s == "yellow") graph[p].push_back({p, 'Y'});
        else if (s == "green") graph[p].push_back({p, 'G'});
        else if (s == "blue") graph[p].push_back({p, 'B'});
        else if (s == "red") graph[p].push_back({p, 'R'});
        else graph[p].push_back({p, 'O'});
    }
    dfs(1, 0);
    cout << dp[1][0] << endl;
}
