def min_complexity(n, bulbs):
    missing = sorted(set(range(1, n + 1)) - set(bulbs))
    even, odd = 0, 0
    for i in missing:
        if i % 2 == 0:
            even += 1
        else:
            odd += 1

    dp = [[[float('inf')] * 2 for _ in range(2)] for _ in range(n + 1)]
    dp[0][0][0] = dp[0][1][1] = 0

    for i in range(1, n + 1):
        for e in range(2):
            for o in range(2):
                if bulbs[i - 1] == 0:
                    if even > 0:
                        dp[i][0][e] = min(dp[i][0][e], dp[i - 1][1][e] + (e == 1), dp[i - 1][0][e])
                    if odd > 0:
                        dp[i][1][o] = min(dp[i][1][o], dp[i - 1][0][o] + (o == 0), dp[i - 1][1][o])
                else:
                    if bulbs[i - 1] % 2 == 0:
                        dp[i][0][e] = min(dp[i][0][e], dp[i - 1][1][e] + (e == 1), dp[i - 1][0][e])
                    else:
                        dp[i][1][o] = min(dp[i][1][o], dp[i - 1][0][o] + (o == 0), dp[i - 1][1][o])

    return min(dp[n][0][0], dp[n][1][0], dp[n][0][1], dp[n][1][1])

n = int(input().strip())
bulbs = list(map(int, input().strip().split()))
print(min_complexity(n, bulbs))
