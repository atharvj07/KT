import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        try {
            int n = Integer.parseInt(br.readLine());
            int[] p = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int q = Integer.parseInt(br.readLine());
            int[] l = new int[n + 1], r = new int[n + 1];
            for (int i = 0; i < n; i++) {
                l[p[i]] = i;
                r[p[i]] = i;
            }
            for (int i = 0; i < q; i++) {
                String[] s = br.readLine().split(" ");
                if (s[0].equals("1")) {
                    System.out.println(Math.max(0, r[Integer.parseInt(s[1])] - l[Integer.parseInt(s[1])] + l[Integer.parseInt(s[2])] - r[Integer.parseInt(s[2])] + 1));
                } else {
                    int a = Integer.parseInt(s[1]), b = Integer.parseInt(s[2]);
                    int t = l[a]; l[a] = l[b]; l[b] = t;
                    t = r[a]; r[a] = r[b]; r[b] = t;
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
