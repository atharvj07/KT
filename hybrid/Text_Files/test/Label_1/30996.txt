import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        in.close();
        boolean[] canWin = new boolean[300];
        for (int i = 1; i < 300; i++) {
            for (int j = 0; j < n; j++) {
                if (a[j] >= i && !canWin[a[j] - i]) {
                    canWin[i] = true;
                    break;
                }
            }
        }
        boolean bitLGM = false;
        for (int i = 0; i < n; i++) {
            if (canWin[a[i]]) {
                bitLGM = !bitLGM;
            }
        }
        System.out.println(bitLGM ? "BitLGM" : "BitAryo");
    }
}
