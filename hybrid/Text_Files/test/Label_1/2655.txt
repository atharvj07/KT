#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
const int inf=1e9;
ll bit[1<<(17+1)],a[1<<(17+1)],n,m;
ll gc(ll x,ll y){
    return x?gc(y%x,x):y;
}
ll get(ll i){
    ll s=0;
    while(i){
        s^=bit[i];
        i-=(i&(-i));
    }
    return s;
}
void update(ll x,ll y){
    ll val=y^a[x];
    a[x]=y;
    while(x<=n){
        bit[x]^=val;
        x+=(x&(-x));
    }
}
int main(){
    ios_base::sync_with_stdio(0);cin.tie(0);
    cin>>n>>m;
    n=1<<n;
    while(m--){
        ll x,y;
        cin>>x>>y;
        update(x,y);
        cout<<get(n)<<'\n';
    }
    return 0;
}
