from collections import defaultdict

s = input()
cnt = 0
ln = len(s)
mp = defaultdict(int)
for i in s:
    if i == 'w':
        cnt += 1
        mp['u'] += 1
    elif i == 'm':
        cnt += 1
        mp['n'] += 1

pw = 1
mod = 10**9 + 7
for i in range(1, cnt + 1):
    pw = (pw * i) % mod

pwn = 1
for i in mp.values():
    pwn = (pwn * i) % mod

print((pw * pwn) % mod)
