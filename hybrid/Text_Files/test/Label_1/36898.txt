import sys

n, t = map(int, sys.stdin.readline().split())
a = [0]
b = [[], []]
for i in range(n):
    ai, ti = map(int, sys.stdin.readline().split())
    a.append(ai)
    b[ti <= t].append(ai)
b[0].sort()
b[1].sort(reverse=True)
s = 0
j = len(b[1]) - 1
for x in b[0]:
    while j >= 0 and x <= b[1][j]:
        j -= 1
        s += 1
    if s >= n:
        break
print(s)
print(s)
for x in b[1][:s]:
    print(a.index(x), end=" ")
