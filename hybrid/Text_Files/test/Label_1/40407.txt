import sys

input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for i in range(1, n + 1):
        line = input().strip()
        for j in range(1, n + 1):
            if line[j - 1] != '*':
                adj[i].append((j, line[j - 1]))
    if m % 2 == 1:
        print("NO")
        continue
    m //= 2
    vis = [False] * (n + 1)
    vis[1] = True
    path = [1]
    flag = False
    def dfs(node, cnt):
        nonlocal flag
        if cnt == m:
            flag = True
            path.append(node)
            print("YES")
            print(*path)
            return
        for u, w in adj[node]:
            if not vis[u]:
                vis[u] = True
                path.append(u)
                dfs(u, cnt + 1)
                vis[u] = False
                path.pop()
    dfs(1, 0)
    if not flag:
        print("NO")
