import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            points[i] = new Point(sc.nextInt(), sc.nextInt());
        }
        sc.close();
        double d = 0;
        for (int i = 0; i < n; i++) {
            Point p1 = points[i], p2 = points[(i + 1) % n], p3 = points[(i + 2) % n];
            double vp = (p2.y - p1.y) * (p3.x - p2.x) - (p2.x - p1.x) * (p3.y - p2.y);
            if (vp < 0) {
                double h = Math.abs(vp) / Math.sqrt(Math.pow(p2.x - p1.x, 2) + Math.pow(p2.y - p1.y, 2));
                d = Math.max(d, h);
            }
        }
        System.out.println(d);
    }
    private static class Point {
        int x, y;
        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
