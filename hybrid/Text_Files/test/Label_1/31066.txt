import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        int max = 0;
        for (int i = 0; i <= n; i++) {
            int cal = c + i * b;
            int cost = a + i * b;
            int val = cal / cost;
            if (val > max) {
                max = val;
            }
        }
        System.out.println(max);
    }
}
