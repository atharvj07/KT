#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll nax = 2e5 + 5;
ll n, c[nax];
int d[nax];

int main() {
    ios_base::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);
    cin >> n;
    ll f = 0, mx = LLONG_MIN;
    for (ll i = 1; i <= n; i++) {
        cin >> c[i] >> d[i];
        mx = max(mx, d[i] == 1 ? c[i] : 0LL);
        f += c[i];
    }
    if (f <= mx)cout << "Infinity\n";
    else if (mx == LLONG_MIN)cout << "Impossible\n";
    else cout << mx << "\n";
    return 0;
}
