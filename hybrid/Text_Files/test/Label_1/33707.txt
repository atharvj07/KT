import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] dice = new int[n * 2];
        for (int i = 0; i < n * 2; i++) {
            dice[i] = sc.nextInt();
        }
        boolean found = false;
        for (int i = 0; i < n - 2; i++) {
            if (dice[i * 2] == dice[i * 2 + 1] && dice[i * 2 + 2] == dice[i * 2 + 3] && dice[i * 2 + 4] == dice[i * 2 + 5]) {
                found = true;
                break;
            }
        }
        System.out.println(found ? "Yes" : "No");
    }
}
