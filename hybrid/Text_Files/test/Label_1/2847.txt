#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s, c;
    cin >> s >> c;
    int n = s.size();
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        string t = s;
        swap(t[i], t[j]);
        if (t < c) {
          cout << t << endl;
          goto end;
        }
      }
    }
    cout << "---" << endl;
  end:;
  }
  return 0;
}
