n = int(input())
a = [int(x) for x in input().split()]
ans = []
for i in range(1, n+1):
    t = i
    for j in range(1, n+1):
        s = j
        cur = []
        for w in a:
            cur.append(w)
            cur = cur[-s*t:]
            if cur.count(1) < t or cur.count(2) < t:
                break
            if cur.count(1) == s*t or cur.count(2) == s*t:
                ans.append((s, t))
                break
print(len(ans))
for s, t in ans:
    print(s, t)
