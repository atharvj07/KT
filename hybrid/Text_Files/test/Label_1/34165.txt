import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int h = in.nextInt();
        int m = in.nextInt();
        int n = in.nextInt();
        int[] table = new int[h];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            char op = in.next().charAt(0);
            if (op == '+') {
                int id = in.nextInt();
                int hash = in.nextInt();
                int idx = hash;
                int cnt = 0;
                while (table[idx] != 0) {
                    idx = (idx + m) % h;
                    cnt++;
                }
                table[idx] = id;
                ans += cnt;
            } else {
                int id = in.nextInt();
                int idx = 0;
                while (table[idx] != id) {
                    idx++;
                }
                table[idx] = 0;
            }
        }
        System.out.println(ans);
    }
}
