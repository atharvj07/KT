#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, m, q;
    cin >> n >> m;
    vector<string> s(n), t(m);
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < m; i++) cin >> t[i];
    cin >> q;
    vector<string> res(q);
    for (int i = 0; i < q; i++) {
        int y;
        cin >> y;
        y--;
        res[i] = s[(y / m) % n] + t[y % m];
    }
    for (string x : res) cout << x << endl;
    return 0;
}
