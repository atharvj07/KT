import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), x = input.nextInt(), c[] = new int[n];
        BigInteger total = BigInteger.valueOf(0), cur = BigInteger.valueOf(x);
        for (int i = 0; i < n; i++) {
            c[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            total = total.add(cur.multiply(BigInteger.valueOf(c[i])));
            cur = cur.subtract(BigInteger.ONE);
            if (cur.compareTo(BigInteger.ONE) < 0) cur = BigInteger.ONE;
        }
        System.out.println(total);
    }
}
