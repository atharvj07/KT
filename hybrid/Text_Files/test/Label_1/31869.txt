import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if ((i == 1 && j == 1) || (i == n && j == 1) || (i == 1 && j == m) || (i == n && j == m)) {
                    continue;
                }
                if (i == 1 || i == n || j == 1 || j == m) {
                    ans += 2;
                } else {
                    ans += 4;
                }
            }
        }
        System.out.println(ans / 4 * x);
    }
}
