import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] perm = new int[n + 1];
        List<Integer> even = new ArrayList<>();
        List<Integer> odd = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            perm[i] = sc.nextInt();
        }

        for (int i = 1; i <= n; i++) {
            if (perm[i] == 0) {
                if (even.isEmpty()) {
                    even.add(i);
                } else if (odd.isEmpty()) {
                    odd.add(i);
                } else if (even.size() < odd.size()) {
                    even.add(i);
                } else {
                    odd.add(i);
                }
            } else {
                if (perm[i] % 2 == 0) {
                    even.add(perm[i]);
                } else {
                    odd.add(perm[i]);
                }
            }
        }

        int ans = 0;
        int last = n % 2;
        for (int i = 0; i < even.size(); i++) {
            if (last != even.get(i) % 2) {
                ans++;
                last = even.get(i) % 2;
            }
        }
        for (int i = 0; i < odd.size(); i++) {
            if (last != odd.get(i) % 2) {
                ans++;
                last = odd.get(i) % 2;
            }
        }
        System.out.println(ans);
    }
}
