import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int X = in.nextInt();
        int[] m = new int[N];
        for (int i = 0; i < N; i++) {
            m[i] = in.nextInt();
        }
        Arrays.sort(m);

        int ans = 0;
        for (int i = 0; i < N; i++) {
            ans += Math.min(X / m[i], 1);
            X -= X / m[i] * m[i];
        }

        System.out.println(ans);
    }
}
