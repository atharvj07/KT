#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    string ans(n, '0');
    int team = 1;
    while (!a.empty()) {
        int mid = a.size() / 2;
        int l = max(0, mid - k);
        int r = min(n - 1, mid + k);
        for (int i = l; i <= r; i++) {
            ans[a[i] - 1] = team + '0';
        }
        a.erase(a.begin() + l, a.begin() + r + 1);
        team = 3 - team;
    }
    cout << ans << endl;
    return 0;
}
