import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        ArrayList<Integer>[] g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int a = in.nextInt(), b = in.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        System.out.println((n & 1) == 0 ? "YES" : "NO");
    }
}
