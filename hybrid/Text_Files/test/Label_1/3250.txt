#include <iostream>
using namespace std;
int main() {
    int m[5], w[5], s, u;
    cin >> m[0] >> m[1] >> m[2] >> m[3] >> m[4];
    cin >> w[0] >> w[1] >> w[2] >> w[3] >> w[4];
    cin >> s >> u;
    double ans = 0;
    for (int i = 0; i < 5; i++) {
        ans += max(0.3 * (i + 1) * 500, (1 - m[i] / 250.0) * (i + 1) * 500 - 50 * w[i]);
    }
    ans += s * 100 - u * 50;
    cout << int(ans) << endl;
    return 0;
}
