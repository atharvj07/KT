import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] v = new int[n];
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            v[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            w[i] = in.nextLong();
        }
        in.close();
        long[] dp = new long[n];
        for (int i = 0; i < n; i++) {
            dp[i] = v[i];
            if (i > 0) {
                dp[i] = Math.max(dp[i], dp[i - 1]);
            }
            for (int j = i - 1; j >= 0; j--) {
                if (Math.abs(w[j] - w[j + 1]) == 1 && 2 * w[j] - w[j + 1] - w[j - 1] >= 0) {
                    dp[i] = Math.max(dp[i], dp[j - 1] + v[i]);
                } else {
                    break;
                }
            }
        }
        long max = 0;
        for (long x : dp) {
            max = Math.max(max, x);
        }
        System.out.println(max);
    }
}
