import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] ropes = new int[m];
        for (int i = 0; i < m; i++) {
            ropes[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            ropes[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            System.out.println(findMax(x, y, ropes));
        }
    }

    private static int findMax(int x, int y, int[] ropes) {
        int max = 0;
        for (int i = 0; i < ropes.length; i++) {
            if (x <= ropes[i] && ropes[i] <= y && ropes[i] > max) {
                max = ropes[i];
            }
        }
        return max;
    }
}
