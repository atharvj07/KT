import sys
n = int(input())
s = input()
k, r = 0, ''
if s[0] == s[-1]:
    k, r = k + 1, s[0]
else:
    for i in range(n):
        if s[i] == r:
            r = '0' if s[i] == '1' else '1'
            k += 1
print(k)
for i in range(k):
    print(s[(i * n // k):((i + 1) * n // k)], end=' ')
    sys.stdout.write(' ')
