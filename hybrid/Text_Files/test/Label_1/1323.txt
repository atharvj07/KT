#include <bits/stdc++.h>
using namespace std;

const int mxN = 1e5;
const int inf = 1e9;

int n, m;
int typ[mxN];
vector<int> adj[mxN], ord;
bool vis[mxN];

int col[mxN];
bool ok = true;

void dfs(int u, int c) {
    vis[u] = true;
    col[u] = c;
    ord.push_back(u);
    for (int v : adj[u]) {
        if (!vis[v])
            dfs(v, c ^ typ[v]);
        else if (col[u] != col[v] ^ typ[v])
            ok = false;
    }
}

int main() {
    cin >> n >> m;
    while (m--) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
        typ[a]++, typ[b]++;
    }
    for (int i = 0; i < n; i++) {
        if (!vis[i]) {
            dfs(i, 0);
        }
    }
    if (!ok)
        cout << "NO";
    else {
        cout << "YES\n";
        int a[n], b[n];
        int ca = 1;
        int cb = 1;
        for (int i : ord) {
            if (col[i] == 0) {
                a[i] = ca;
                ca++;
            } else {
                a[i] = cb;
                cb++;
            }
        }
        for (int i : ord) {
            if (col[i] == 0) {
                b[i] = cb;
                cb++;
            } else {
                b[i] = ca;
                ca++;
            }
        }
        for (int i = 0; i < n; i++) {
            cout << a[i] << " ";
        }
        cout << "\n";
        for (int i = 0; i < n; i++) {
            cout << b[i] << " ";
        }
        cout << "\n";
    }
}
