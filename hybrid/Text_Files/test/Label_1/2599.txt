#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  int ans = 1;
  int cnt = 1;
  for (int i = 1; i < n; ++i) {
    if (a[i] == a[i - 1])
      cnt = 1;
    else
      cnt++;
    ans = max(ans, cnt);
  }
  cout << ans << '\n';
  return 0;
}
