import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long cnt1 = sc.nextLong();
        long cnt2 = sc.nextLong();
        long x = sc.nextLong();
        long y = sc.nextLong();
        long v = 0;
        while (cnt1 > 0 || cnt2 > 0) {
            v++;
            if (v % x != 0 && cnt1 > 0) {
                cnt1--;
            } else if (v % y != 0 && cnt2 > 0) {
                cnt2--;
            }
        }
        System.out.println(v);
    }
}
