n, m = map(int, input().split())
a = [list(input()) for _ in range(n)]
c = [['.'] * m for _ in range(n)]
for x in range(n):
    for y in range(m):
        if a[x][y] == '.':
            continue
        f = [[x, y]]
        while f:
            nx, ny = f.pop()
            if 0 <= nx < n and 0 <= ny < m and a[nx][ny] == '.':
                a[nx][ny] = '*'
                c[nx][ny] = str((c[nx][ny] + 1) % 10)
                f += [(nx + 1, ny), (nx - 1, ny), (nx, ny + 1), (nx, ny - 1)]
print(*[''.join(cc) for cc in c], sep='\n')
