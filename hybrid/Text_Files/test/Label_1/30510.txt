import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        char[] chars = s.toCharArray();
        boolean[] colored = new boolean[n];
        boolean[] canColor = new boolean[26];
        int firstUncolored = -1;
        int firstDifferent = -1;
        for (int i = 0; i < n; i++) {
            if (!colored[i]) {
                if (firstUncolored == -1 || chars[i] <= chars[firstUncolored]) {
                    firstUncolored = i;
                }
                if (i > 0 && chars[i] != chars[i - 1]) {
                    firstDifferent = i;
                }
            }
        }
        if (firstUncolored != -1) {
            if (firstDifferent != -1 && chars[firstUncolored] > chars[firstDifferent]) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                colored[firstUncolored] = true;
                canColor[chars[firstUncolored] - 'a'] = true;
                for (int i = 0; i < n; i++) {
                    if (!colored[i]) {
                        if (canColor[chars[i] - 'a']) {
                            colored[i] = true;
                            for (char c : s.substring(i + 1).toCharArray()) {
                                canColor[c - 'a'] = false;
                            }
                        }
                    }
                }
                StringBuilder builder = new StringBuilder();
                for (boolean b : colored) {
                    builder.append(b ? '1' : '0');
                }
                System.out.println(builder);
            }
        } else {
            System.out.println("NO");
        }
    }
}
