import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        ArrayDeque<Integer> showcase = new ArrayDeque<>();
        TreeSet<Integer> prices = new TreeSet<>();
        
        for (int i = 0; i < 2 * n; i++) {
            String[] line = reader.readLine().split(" ");
            if (line[0].equals("+")) {
                prices.add(n - i);
                showcase.addLast(n - i);
            } else {
                int price = Integer.parseInt(line[1]);
                if (showcase.isEmpty() || !showcase.contains(price) || price != prices.first()) {
                    System.out.println("NO");
                    return;
                }
                prices.remove(price);
                showcase.remove(price);
            }
        }
        
        if (showcase.isEmpty()) {
            System.out.println("YES");
            for (Integer price : prices) {
                System.out.print(price + " ");
            }
        } else {
            System.out.println("NO");
        }
    }
}
