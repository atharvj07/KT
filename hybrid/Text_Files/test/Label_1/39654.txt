MOD = 10**9 + 9

def mod_pow(base, exponent, modulus):
    result = 1
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus
        exponent = exponent >> 1
        base = (base * base) % modulus
    return result

def solve(n, a, b, k, s):
    sequence = [1 if c == '+' else -1 for c in s]
    sum_sequence = sum(sequence[i] * mod_pow(a, n - i, MOD) * mod_pow(b, i, MOD) for i in range(k)) % MOD

    total_periods = (n + 1) // k
    abk = mod_pow(a, k, MOD) * mod_pow(mod_pow(b, k, MOD), MOD-2, MOD) % MOD
    
    if abk == 1:
        result = (sum_sequence * total_periods) % MOD
    else:
        result = (sum_sequence * (1 - mod_pow(abk, total_periods, MOD)) * mod_pow(1 - abk, MOD-2, MOD)) % MOD

    return result

n, a, b, k = map(int, input().split())
s = input().strip()

print(solve(n, a, b, k, s))
