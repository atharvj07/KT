#include <bits/stdc++.h>
typedef long long LL;
using namespace std;
const int MOD = int(1e9) + 7;
int n, cnt, pre[200005], suf[200005];
int pw[200005];
char s[200005];
int main() {
  pw[0] = 1;
  for (int i = 1; i <= 200000; i++) pw[i] = 1LL * pw[i - 1] * 3 % MOD;
  scanf("%d", &n);
  scanf("%s", s + 1);
  for (int i = 1; i <= n; i++)
    if (s[i] == '?') {
      s[i] = 'a';
      if (i > 1 && s[i] == s[i - 1]) {
        cnt++;
        pre[i] = pre[i - 1];
        suf[i] = suf[i - 1];
      } else if (i == 1 || s[i] != s[i - 1]) {
        pre[i] = i;
        suf[i] = i;
      }
    }
  if (cnt == 0) {
    printf("0\n");
    return 0;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (s[i] == 'b') {
      ans = (1LL * pre[i - 1] * suf[i + 1] + ans) % MOD;
      if (cnt == 1) ans = (1LL * ans * pw[n - 1 - cnt] + ans) % MOD;
    }
  }
  printf("%d\n", ans);
  return 0;
}
