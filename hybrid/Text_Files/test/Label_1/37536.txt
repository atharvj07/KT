n, m = map(int, input().split())
a = list(map(int, input().split()))

a.sort()

prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sum[i] = prefix_sum[i - 1] + a[i - 1]

result = [0] * n
for k in range(1, n + 1):
    total_penalty = 0
    for day in range(1, (k // m) + 2):
        sweets_eaten = min(k - (day - 1) * m, m)
        total_penalty += day * (prefix_sum[min(day * m, k)] - prefix_sum[(day - 1) * m])
    result[k - 1] = total_penalty

print(' '.join(map(str, result)))
