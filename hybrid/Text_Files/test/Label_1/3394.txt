#include <bits/stdc++.h>
 
using namespace std;
 
const int MAXN = 4e5 + 5;
const int MAX_DEG = 32;
 
vector<int> G[MAXN];
bool vis[MAXN];
int deg[MAXN];
 
int main() {
  int n, d, k;
  cin >> n >> d >> k;
  if (d > n - 1 || d > k * (n - 1) || (d & 1) != (n & 1)) {
    cout << "NO\n";
    return 0;
  }
  cout << "YES\n";
  int u = 1, v, steps = 0;
  for (int i = 2; i <= n; i++) {
    G[u].push_back(i);
    deg[u]++;
    deg[i]++;
    if (steps < d / 2) {
      u = i;
      steps++;
    } else if (steps == d / 2) {
      v = i;
      steps = 0;
    }
  }
  if (d & 1) {
    u = 1;
    for (int i = 2; i <= n; i++) {
      if (i != v && deg[i] < k) {
        G[u].push_back(i);
        deg[u]++;
        deg[i]++;
        u = i;
        break;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j : G[i]) {
      cout << i << ' ' << j << '\n';
    }
  }
  return 0;
}
