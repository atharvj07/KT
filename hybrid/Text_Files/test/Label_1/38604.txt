import sys
n, r, avg = map(int, input().split())
a = []
b = []
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
sum = 0
for i in range(n):
    sum += a[i]
add = max(0, (avg * n - sum + a[i] - 1) // a[i])
ans = 0
for i in range(n):
    if add > 0:
        if b[i] <= add * a[i]:
            ans += b[i]
            add -= b[i] // a[i]
        else:
            ans += add * a[i]
            add = 0
print(ans)
