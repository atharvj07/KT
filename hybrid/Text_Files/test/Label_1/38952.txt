for _ in range(int(input())):
    n, x = map(int, input().split())
    bal = 0
    a = [bal]
    for i in input():
        bal += int(i)
        a.append(bal)
    ans = 0
    for i in a:
        j = i - x
        if j >= 0: ans += a.count(j)
    if ans == 0: print(1)
    elif ans == float('inf'): print(-1)
    else: print(ans)
