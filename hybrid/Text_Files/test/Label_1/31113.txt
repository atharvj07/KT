import java.io.*;
import java.util.*;

public class Main {

    private static int n, root;
    private static int[] w, par, sz;
    private static List<int[]>[] adj;
    private static long ans;

    private static int find(int node) {
        return par[node] = (par[node] == node) ? node : find(par[node]);
    }

    private static void merge(int a, int b) {
        a = find(a);
        b = find(b);
        if (a != b) {
            if (sz[a] > sz[b]) {
                par[b] = a;
                sz[a] += sz[b];
            } else {
                par[a] = b;
                sz[b] += sz[a];
            }
        }
    }

    private static void dfs(int node, int p, int depth) {
        ans += (long) w[node] * sz[find(node)];
        for (int[] edge : adj[node]) {
            int next = edge[0], weight = edge[1];
            if (next != p) {
                dfs(next, node, depth + 1);
                merge(node, next);
                ans += (long) Math.ceil(Math.log(depth + 1) / Math.log(2)) * Math.min(w[node], w[next]);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        StringTokenizer tokenizer = new StringTokenizer(in.readLine());
        w = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            w[i] = Integer.parseInt(tokenizer.nextToken());
        }
        int minWeight = Integer.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            if (w[i] < minWeight) {
                minWeight = w[i];
                root = i;
            }
        }
        par = new int[n + 1];
        sz = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            par[i] = i;
            sz[i] = 1;
        }
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            tokenizer = new StringTokenizer(in.readLine());
            int a = Integer.parseInt(tokenizer.nextToken());
            int b = Integer.parseInt(tokenizer.nextToken());
            adj[a].add(new int[]{b, 0});
            adj[b].add(new int[]{a, 0});
        }
        dfs(root, root, 0);
        System.out.println(ans);
    }
}
