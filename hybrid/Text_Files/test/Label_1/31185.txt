import java.io.*;
import java.util.*;

public class Main {
    static long mod = 1000000007;
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) c[i] = input.nextInt();
        int[] b = new int[n-1];
        for (int i = 0; i < n-1; i++) b[i] = input.nextInt();
        int q = input.nextInt();
        int[] x = new int[q];
        for (int i = 0; i < q; i++) x[i] = input.nextInt();
        long[] ans = new long[q];
        for (int k = 0; k < q; k++) {
            long good = 0;
            for (int l = 0; l < n; l++) c[l] = (int)(Math.max(0, c[l] - x[k]));
            boolean zero = true;
            for (int i = 0; i < n; i++) if (c[i] != 0) zero = false;
            if (zero) {
                good = 1;
                for (int i = 0; i < n-1; i++)
                    if (c[i] < b[i] || c[i+1] < b[i]) good = 0;
            }
            ans[k] = good;
        }
        for (int i = 0; i < q; i++) System.out.println(ans[i]);
    }
}
