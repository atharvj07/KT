import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), x = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += (long)(a[i] * Math.max(x, 1));
            x--;
        }
        System.out.println(ans);
    }
}
