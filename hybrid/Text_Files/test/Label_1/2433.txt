#include <bits/stdc++.h>
using namespace std;

const int mxN = 101;
const int MX = 1e5;

int a[mxN][4];

int main() {
    int n, ans = 0;
    cin >> n;
    for(int i = 1; i <= n; i++) {
        cin >> a[i][1] >> a[i][2] >> a[i][3];
    }
    for(int i = 1; i <= n; i++) {
        swap(a[i][1], a[i][3]);
    }
    for(int i = 1; i <= n; i++) {
        int x = a[i][2];
        for(int j = i + 1; j <= n; j++) {
            if(x != a[j][1]) continue;
            int val = max(a[i][2], a[j][2]);
            for(int k = j + 1; k <= n; k++) {
                if(val == a[j][2]) val = max(val, a[k][2]);
                else val = max(val, a[k][3]);
                if(val == a[k][3]) val = max(val, a[k][1]);
                if(val == a[k][1]) val = max(val, a[i][3]);
                ans = max(ans, val);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
