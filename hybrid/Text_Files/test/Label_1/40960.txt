from collections import defaultdict, deque
from sys import stdin

input = stdin.readline
n, m = map(int, input().split())
s = input()
adj = defaultdict(list)
for _ in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
mx = 0

def bfs():
    vis = [False] * (n + 1)
    max_freq = [0] * (n + 1)
    vis[1] = True
    q = deque([(1, 0)])
    while q:
        x, f = q.popleft()
        max_freq[x] = max(max_freq[x], f)
        for y in adj[x]:
            if not vis[y]:
                q.append((y, f + (s[y - 1] == s[x - 1])))
                vis[y] = True
    return max(max_freq)

ans = bfs()

if not ans:
    print(-1)
else:
    print(ans)
