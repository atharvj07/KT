import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int I = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int k = 0;
        for (long i : a) {
            k = Math.max(k, Long.numberOfTrailingZeros(i) + 1);
        }
        int m = 0;
        for (long i : a) {
            m = Math.max(m, 64 - Long.numberOfLeadingZeros(i));
        }
        long ans = (long)n * k;
        long l = -1, r = -1;
        for (int i = 0; i <= 64; i++) {
            for (int j = i + 1; j <= 64; j++) {
                long nl = 0, nr = 0;
                for (long x : a) {
                    nl += Math.max(0, i - Long.numberOfTrailingZeros(x));
                    nr += Math.max(0, j - Long.numberOfTrailingZeros(x));
                }
                long nk = 0;
                if (i != 0) {
                    nk += Math.max(0, i - Long.numberOfLeadingZeros(l));
                }
                if (j != 64) {
                    nk += Math.max(0, j - Long.numberOfLeadingZeros(r));
                }
                if (nl + nr + nk <= I * 8) {
                    if (nl + nr < ans) {
                        ans = nl + nr;
                        l = 1L << i;
                        r = 1L << j;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
