import java.util.*;

class Main {
   public static void main(String[] args) {
      Scanner in = new Scanner(System.in);
      int m = in.nextInt();
      int t = in.nextInt();
      int r = in.nextInt();
      int[] wi = new int[m];
      for (int i = 0; i < m; i++) {
         wi[i] = in.nextInt();
      }
      int ans = 0;
      int l = 0;
      int r2 = 0;
      while (l < m) {
         while (r2 < m && wi[r2] - wi[l] <= t - 1) {
            r2++;
         }
         ans = Math.max(ans, r2 - l);
         l++;
      }
      if (ans < r) {
         System.out.println(-1);
      } else {
         System.out.println(ans);
      }
   }
}
