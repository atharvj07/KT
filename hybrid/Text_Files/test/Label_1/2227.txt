#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m;
  std::cin >> n >> m;
  int base = n / m;
  std::vector<int> ans(m, base);
  for (int i = 0; i < n % m; i++) ans[i]++;
  std::sort(ans.begin(), ans.end());
  for (auto a : ans) std::cout << a << ' ';
  std::cout << '\n';
}
