import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = i;
        }
        for (int i = 2; i <= n; i++) {
            for (int j = i; j <= n; j += i) {
                p[j] = i;
            }
        }
        int m = 0;
        for (int i = 1; i <= n; i++) {
            m += n / p[i] / 2;
        }
        System.out.println(m);
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j += p[i]) {
                if (j % p[i] == 0) {
                    System.out.println(i + " " + j);
                }
            }
        }
    }
}
