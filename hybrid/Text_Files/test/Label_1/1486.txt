#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int32_t main() {
    int n, m, a[N], pre[N] = {}, ans[N] = {};
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) pre[i] += pre[i - 1] + a[i];
    int lst = 0;
    while (m--) {
        int t;
        cin >> t;
        if (t == 3) {
            int q;
            cin >> q;
            ans[lst++] = pre[q] - pre[q - 1];
        } else if (t == 2) {
            int y;
            cin >> y;
            pre[n] += y * n;
        } else {
            int v, x;
            cin >> v >> x;
            pre[v] = pre[v] - a[v] + x;
            a[v] = x;
        }
    }
    for (int i = 0; i < lst; i++) cout << ans[i] << "\n";
    return 0;
}
