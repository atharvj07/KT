import java.util.*;

public class Main {
    static int n, m;
    static int[][] adj;
    static int[] a, b;
    static boolean ans;
    static int[][] sol;

    static void dfs(int u, int p, int fa, int fb) {
        int sum = a[u] - fa + fb;
        if (sum == b[u]) {
            sol[u][u] = fa;
            for (int v : adj[u])
                if (v != p)
                    dfs(v, u, fa, sum - a[v]);
            return;
        }
        if (sum < b[u]) {
            for (int v : adj[u])
                if (v != p && a[v] >= b[u] - sum) {
                    dfs(v, u, fa, sum - a[v]);
                    return;
                }
        }
        ans = false;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        a = new int[n + 1];
        b = new int[n + 1];
        for (int i = 1; i <= n; i++)
            a[i] = in.nextInt();
        for (int i = 1; i <= n; i++)
            b[i] = in.nextInt();
        adj = new int[n + 1][];
        for (int i = 1; i <= n; i++)
            adj[i] = new int[0];
        for (int i = 1; i <= m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            int[] t = adj[u];
            adj[u] = new int[t.length + 1];
            System.arraycopy(t, 0, adj[u], 0, t.length);
            adj[u][t.length] = v;
            t = adj[v];
            adj[v] = new int[t.length + 1];
            System.arraycopy(t, 0, adj[v], 0, t.length);
            adj[v][t.length] = u;
        }
        ans = true;
        sol = new int[n + 1][n + 1];
        dfs(1, 0, 0, 0);
        if (ans) {
            System.out.println("YES");
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++)
                    System.out.print(sol[i][j] + " ");
                System.out.println();
            }
        } else
            System.out.println("NO");
        in.close();
    }
}
