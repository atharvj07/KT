#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int k;
  cin >> k;

  string ad;
  getline(cin, ad);
  getline(cin, ad);

  ad.insert(0, 1, ' ');
  ad.push_back(' ');

  int n = ad.size();
  vector<vector<int>> dp(n, vector<int>(k + 1, 1e9));

  for (int i = n - 2; i >= 0; i--) {
    for (int j = 1; j <= k; j++) {
      for (int l = i + 1; l < n; l++) {
        if (ad[l] == ' ' || ad[l] == '-') {
          dp[i][j] = min(dp[i][j], dp[l][j - (l - i)] + l - i);
        }
      }
    }
  }

  cout << dp[0][k] << endl;

  return 0;
}
