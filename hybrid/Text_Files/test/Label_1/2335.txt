#include <iostream>
#include <vector>
using namespace std;

const int MAX = 105;

int n, m, q;
int row[MAX], col[MAX];
vector<pair<int,int>> mat[MAX][MAX];

void solve() {
    for (int i = 0; i < q; i++) {
        int t, r, c, x;
        cin >> t;
        if (t == 1) cin >> r, row[r]++;
        else if (t == 2) cin >> c, col[c]++;
        else cin >> r >> c >> x, mat[r][c].push_back({row[r] + col[c], x});
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            for (auto it : mat[i][j]) {
                if (it.first == row[i] + col[j]) {
                    cout << it.second << ' ';
                    break;
                }
            }
        }
    }
    cout << endl;
}

int main() {
    cin >> n >> m >> q;
    solve();
    return 0;
}
