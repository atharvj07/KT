import sys

n,q = map(int,input().split())
a = list(map(int,input().split()))
b = list(map(int,input().split()))

for x in range(q):
    l,r = map(int,input().split())
    if min(a[l-1:r]) == max(b[l-1:r]):
        if sum(a[l-1:r]) == sum(b[l-1:r]): print(0)
        else:
            if (sum(a[l-1:r]) - sum(b[l-1:r])) % 2 == 0: print(1)
            else: print(-1)
    else: print(-1)
