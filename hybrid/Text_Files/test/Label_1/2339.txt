#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 3e5+5;
int n, k;
ll a[N];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    if (k <= n - k) {
        ll ans = 0;
        for (int i = 1; i <= k; i++) {
            ans += a[i + k] - a[i];
        }
        cout << ans << '\n';
    } else {
        ll ans = 0;
        for (int i = k + 1; i <= n; i++) {
            ans += a[i] - a[i - k];
        }
        cout << ans << '\n';
    }
    return 0;
}
