from collections import defaultdict,deque
n=int(input())
l1=defaultdict(list);r1=defaultdict(list)
for i in range(n):
    l,r=map(int,input().split())
    l1[l].append(i);r1[r].append(i)
stk=[]
ans=[None for j in range(2*n)]
for i in range(1,2*n+1):
    if not stk:stk.append([i,1])
    elif i-stk[-1][0]==stk[-1][1]:
        ans[stk[-1][0]-1]='('
        ans[i-1]=')'
        stk.pop()
    else:stk.append([i,1])
for i in l1[1]:
    ans[i-1]='('
    stk.append([i,2])
for i in range(2,2*n+1):
    if not stk:stk.append([i,1])
    elif i-stk[-1][0]==stk[-1][1]:
        ans[stk[-1][0]-1]='('
        ans[i-1]=')'
        stk.pop()
    elif r1[i]:
        ans[r1[i][0]-1]=')'
        stk.pop()
    else:stk.append([i,2])
if not stk:print(''.join(ans))
else:print('IMPOSSIBLE')
