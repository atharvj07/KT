#include <bits/stdc++.h>
using namespace std;
#define all(x) x.begin(), x.end() 
#define vi vector<int>

int main() {
    int n; cin >> n; 
    vi ans(n);
    for (int i = 0; i < n; i++) {
        int x; cin >> x;
        if (i == 0) ans[i] = x;
        else if (i == 1) ans[i] = 10;
        else {
            int t = ans[i - 1];
            int tt = (t > 9); 
            while (x != 0) {
                t++; 
                tt += (t > 9); 
                x--;
            }
            ans[i] = t;
        }
    }
    for (int x : ans) cout << x << endl;
    return 0;
}
