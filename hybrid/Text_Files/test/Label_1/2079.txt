#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define pb push_back
typedef vector<ll> vll;
typedef vector<int> vi;
typedef map<int,int> mii;
ll mod=1e9+7;
const ll inf = 1e18;
const ld eps = 1e-9;
#define endl "\n"
#define IOS ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
int dx[]={-1,0,1,0};
int dy[]={0,1,0,-1};

ll solve(ll x1,ll y1,ll x2,ll y2,string s,ll n){
	queue<pair<ll,ll>>q;
	q.push({x1,y1});
	ll vis[1001][1001]={0};
	vis[x1][y1]=1;
	ll day=1;
	while(!q.empty()){
		ll sz=q.size();
		while(sz--){
			ll x=q.front().first;
			ll y=q.front().second;
			q.pop();
			if(x==x2 && y==y2) return day;
			for(ll i=0;i<4;i++){
				ll nx=x+dx[i];
				ll ny=y+dy[i];
				if(nx>=0 && ny>=0 && vis[nx][ny]==0){
					vis[nx][ny]=1;
					q.push({nx,ny});
				}
			}
			vis[x][y]=0;
		}
		day++;
		for(ll i=0;i<4;i++){
			ll nx=x1+dx[i];
			ll ny=y1+dy[i];
			if(nx>=0 && ny>=0){
				if(s[(day-1)%n]=='U' && ny==y2){
					return day;
				}
				else if(s[(day-1)%n]=='D' && ny==y2){
					return day;
				}
				else if(s[(day-1)%n]=='L' && nx==x2){
					return day;
				}
				else if(s[(day-1)%n]=='R' && nx==x2){
					return day;
				}
			}
		}
	}
	return -1;
}

int main(){
	IOS;
	ll x1,y1,x2,y2;
	cin>>x1>>y1>>x2>>y2;
	string s;
	cin>>s;
	ll n=s.length();
	cout<<solve(x1,y1,x2,y2,s,n)<<endl;
	return 0;
}
