#include <bits/stdc++.h>
using namespace std;

namespace DSU {
    vector<int> par;
    void init(int n) { par.assign(n, -1); }
    int find(int a) { return par[a] < 0 ? a : (par[a] = find(par[a])); }
    void unite(int a, int b) { par[find(a)] = find(b); }
};

const int N = 3e5 + 10, inf = 1 << 29;
vector<int> adj[2][N];
vector<pair<int, int>> uv;

int main() {
    int n, m, s;
    cin >> n >> m >> s;
    DSU::init(n + 1);
    for (int i = 0; i < m; i++) {
        int t, u, v;
        cin >> t >> u >> v;
        if (t == 1) {
            adj[0][u].push_back(v);
            adj[1][v].push_back(u);
        } else {
            uv.push_back({u, v});
        }
    }
    queue<int> q[2];
    vector<int> sz(n + 1, 1);
    vector<bool> v(n + 1, false);
    q[0].push(s);
    v[s] = true;
    while (!q[0].empty() || !q[1].empty()) {
        for (int k = 0; k < 2; k++) {
            while (!q[k].empty()) {
                int u = q[k].front();
                q[k].pop();
                for (int x : adj[k][u]) {
                    if (!v[x]) {
                        q[k].push(x);
                        v[x] = true;
                        if (k == 0) sz[DSU::find(x)]++;
                        else sz[DSU::find(u)]++;
                    }
                }
            }
        }
        for (auto u : uv) {
            int p1 = DSU::find(u.first), p2 = DSU::find(u.second);
            if (p1 != p2) {
                if (sz[p1] > sz[p2]) adj[0][u.first].push_back(u.second);
                else adj[0][u.second].push_back(u.first);
                q[0].push(u.first);
                q[0].push(u.second);
                DSU::unite(p1, p2);
            }
        }
        for (auto u : uv) {
            int p1 = DSU::find(u.first), p2 = DSU::find(u.second);
            if (p1 != p2) {
                if (sz[p1] < sz[p2]) adj[1][u.first].push_back(u.second);
                else adj[1][u.second].push_back(u.first);
                q[1].push(u.first);
                q[1].push(u.second);
                DSU::unite(p1, p2);
            }
        }
    }
    cout << sz[DSU::find(s)] << '\n';
    for (auto e : uv) {
        if (adj[0][e.first].size()) cout << "+";
        else cout << "-";
    }
    cout << '\n';
    cout << sz[DSU::find(s)] << '\n';
    for (auto e : uv) {
        if (adj[1][e.first].size()) cout << "+";
        else cout << "-";
    }
    cout << '\n';
}
