import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] init = sc.nextLine().split(" ");
        int days = sc.nextInt();
        sc.nextLine();
        String[][] murders = new String[days][2];
        for (int i = 0; i < days; i++) {
            murders[i] = sc.nextLine().split(" ");
        }
        String[] victims = init;
        for (int i = 0; i < days; i++) {
            System.out.println(victims[0] + " " + victims[1]);
            for (String[] murder : murders) {
                if (murder[0].equals(victims[0])) {
                    victims[0] = murder[1];
                    break;
                } else if (murder[0].equals(victims[1])) {
                    victims[1] = murder[1];
                    break;
                }
            }
        }
        System.out.println(victims[0] + " " + victims[1]);
    }
}
