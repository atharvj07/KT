q = int(input())
for _ in range(q):
    s = input()
    l, r, u, d = s.count('L'), s.count('R'), s.count('U'), s.count('D')
    min_lr = min(l, r)
    min_ud = min(u, d)
    if min_lr == 0 and min_ud == 0:
        print(0)
        print()
    elif min_lr == 0:
        print(2)
        print("UD")
    elif min_ud == 0:
        print(2)
        print("LR")
    else:
        print(2 * (min_lr + min_ud))
        print("L" * min_lr + "U" * min_ud + "R" * min_lr + "D" * min_ud)
