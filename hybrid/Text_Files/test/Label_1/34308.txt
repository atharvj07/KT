import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] a = new int[n], b = new int[n];
        int[] c = new int[4000000];
        for (int i = 0; i < n; ++i) { a[i] = s.nextInt(); c[(int)a[i]]++; }
        for (int i = 0; i < n; ++i) { b[i] = s.nextInt(); c[(int)b[i]]++; }
        int cnt = 0;
        for (int i = 0; i < 4000000; ++i) if (c[i] == 1) cnt++;
        System.out.println(((cnt & 1) == 0) ? ("Karen") : ("Koyomi"));
    }
}
