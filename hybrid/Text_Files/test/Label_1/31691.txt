import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        input.nextLine();
        String[] a = new String[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextLine();
        }
        int[][] cost = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cost[i][j] = input.nextInt();
            }
        }
        input.close();
        int[][] minCost = new int[n][26];
        for (int i = 0; i < n; i++) {
            Arrays.fill(minCost[i], Integer.MAX_VALUE);
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int c = a[i].charAt(j) - 'a';
                minCost[i][c] = Math.min(minCost[i][c], cost[i][j]);
            }
        }
        int ans = 0;
        for (int i = 0; i < 26; i++) {
            int min = Integer.MAX_VALUE;
            for (int j = 0; j < n; j++) {
                min = Math.min(min, minCost[j][i]);
            }
            if (min != Integer.MAX_VALUE) {
                ans += min;
            }
        }
        System.out.println(ans);
    }
}
