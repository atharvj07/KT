#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a;
  cin >> n;
  vector<int> v(n);
  for (auto &x : v) cin >> x;
  int s, t;
  cin >> s >> t;
  if (s > t) swap(s, t);
  long long ans1 = 0, ans2 = 0;
  for (int i = s; i < t; i++) ans1 += v[i];
  for (int i = s; i >= 1; i--) ans2 += v[i - 1];
  for (int i = n; i > t; i--) ans2 += v[i - 1];
  cout << min(ans1, ans2) << endl;
  return 0;
}
