#include <iostream>
#include <vector>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<int> pref(n + 1);
    for (int i = 1; i < n; i++) {
        pref[i] = pref[i - 1] + (s[i] == '.' && s[i - 1] == '#');
    }
    int m;
    cin >> m;
    while (m--) {
        int l, r;
        cin >> l >> r;
        cout << pref[r - 1] - pref[l - 1] << '\n';
    }
    return 0;
}
