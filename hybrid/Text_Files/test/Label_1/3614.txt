#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  long long n, sum, avg, x;
  cin >> t;
  while (t--) {
    cin >> n;
    sum = 0;
    for (int i = 0; i < n; i++) {
      cin >> x;
      sum += x;
    }
    avg = sum / n;
    x = sum % n;
    cout << (x == 0 ? 0 : n - x) << endl;
  }
  return 0;
}
