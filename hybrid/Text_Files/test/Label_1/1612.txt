#include <cstdio>
#include <ctime>
#include <cmath>

int a, ta, b, tb;
int h, m, h1, m1;
double d1, d2;
int main(void) {
  scanf("%d %d %d %d", &a, &ta, &b, &tb);
  scanf("%d:%d", &h, &m);
  if (((h * 60 + m) % a) == 0)
    ++d1;
  while (d1 < 1159) {
    d1 += a;
    ++d2;
  }
  if (((600 % b) == 0) && ((h * 60 + m) > 600))
    --d2;
  if (((h * 60 + m - 600) % b) == 0)
    ++d1;
  h1 = (int)floor(d1 / 60);
  m1 = (int)d1 - 60 * h1;
  printf("%d\n", (h1 - h) * 60 + m1 - m - d2);
  return 0;
}
