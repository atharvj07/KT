import sys

n, k = map(int, sys.stdin.readline().split())
p = [list(map(int, sys.stdin.readline().split())) for i in range(k)]

lcs = [[0] * (n + 1) for i in range(k + 1)]
for i in range(k):
    for j in range(n):
        lcs[i + 1][j + 1] = max(lcs[i][j + 1], lcs[i + 1][j])
        if p[i][j] == j + 1:
            lcs[i + 1][j + 1] = max(lcs[i + 1][j + 1], lcs[i][j] + 1)

print(lcs[k][n])
