import sys

n, w = map(int, input().split())
cups = sorted(map(int, input().split()))
girls = cups[: n]
boys = cups[n :] + [0]
ans = 0
for girl in girls:
    x = min(w // (n + 1), girl // 2)
    ans += x
    w -= x
    for boy in boys:
        if w >= 2 * x and boy >= 2 * x:
            ans += 2 * x
            w -= 2 * x
        else:
            break
print(ans)
