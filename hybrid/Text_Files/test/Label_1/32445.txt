import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        int now = 0;
        for (int i = 0; i < n; i++) {
            if (now == 0) {
                if (a[i] >= 3) {
                    now = 3;
                    a[i] -= 3;
                    ans++;
                } else if (a[i] > 0) {
                    now = a[i];
                    a[i] = 0;
                    ans++;
                }
            } else {
                if (a[i] >= now) {
                    a[i] -= now;
                    ans++;
                } else {
                    now = 0;
                    i--;
                }
            }
        }
        System.out.println(ans);
    }
}
