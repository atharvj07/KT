#include <bits/stdc++.h>

using namespace std;
using ll=long long;

const ll N=5005;

int main()
{
    ll n,i,a[N];
    cin>>n;
    for(i=1;i<=n;++i) cin>>a[i];

    ll cur=0,ans=0;
    for(i=1;i<=n;++i)
    {
        if(a[i]==0) continue;
        if(cur<i)
        {
            ans+=(i-cur+1)/2;
            cur=i;
        }
        cur+=a[i];
    }
    if(cur>n) ans++;
    cout<<ans<<endl;
    return 0;
}
