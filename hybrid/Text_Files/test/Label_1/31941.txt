import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }
        long maxSum = 0;
        int[] cut = new int[n];
        int k = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] == a[j] && (i == 0 || j == n - 1 || a[i - 1] != a[i] || a[j + 1] != a[j])) {
                    long sum = 0;
                    for (int l = 0; l < n; l++) {
                        if (l != i && l != j) {
                            sum += a[l];
                        }
                    }
                    if (sum > maxSum) {
                        maxSum = sum;
                        k = 0;
                        cut[k++] = i + 1;
                        cut[k++] = j + 1;
                    }
                }
            }
        }
        System.out.println(maxSum + " " + k);
        for (int i = 0; i < k; i++) {
            System.out.print(cut[i] + " ");
        }
        System.out.println();
    }
}
