import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int l = sc.nextInt();
            int[][] mn = new int[12][2];
            if (l == 0) {
                break;
            }
            for (int i = 0; i < 12; i++) {
                mn[i][0] = sc.nextInt();
                mn[i][1] = sc.nextInt();
            }
            int sum = 0;
            int month = -1;
            for (int i = 0; i < 12; i++) {
                sum += mn[i][0] - mn[i][1];
                if (sum >= l) {
                    month = i + 1;
                    break;
                }
            }
            if (month != -1) {
                System.out.println(month);
            } else {
                System.out.println("NA");
            }
        }
    }
}
