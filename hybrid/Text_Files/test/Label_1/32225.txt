import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    private static void dfs(int v, int p, int[] c, int[] d, List<Long>[] g, int n, Queue<Long> q) {
        int delta = 0;
        if (d[v]%2==0 && c[v]%2==1) delta = 1;
        if (d[v]%2==1 && c[v]%2==0) delta = -1;

        q.add((long)delta);
        for (long w: g[v])
            if (w!=p) {
                dfs((int)w, v, c, d, g, n, q);
                d[(int)w]+=delta;
            }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] line = in.readLine().split(" ");
        int n = Integer.parseInt(line[0]), m = Integer.parseInt(line[1]);
        int[] c = new int[n+1], d = new int[n+1];
        line = in.readLine().split(" ");
        for (int i=1; i<=n; i++) {
            c[i] = Integer.parseInt(line[i-1]);
            d[i] = n;
        }

        List<Long>[] g = new ArrayList[n+1];
        for (int i=1; i<=n; i++)
            g[i] = new ArrayList<>();
        for (int i=0; i<m; i++) {
            line = in.readLine().split(" ");
            int a = Integer.parseInt(line[0]), b = Integer.parseInt(line[1]);
            g[a].add((long)b);
            g[b].add((long)a);
        }

        boolean ok = true;
        for (int i=1; i<=n; i++)
            if (c[i]%2==1 && d[i]%2==0) ok = false;
            else if (c[i]%2==0 && d[i]%2==1) ok = false;

        if (!ok) {
            System.out.println("NO");
            return;
        }

        int[] p = new int[n+1];
        for (int i=1; i<=n; i++)
            if (p[i]==0) {
                Queue<Long> q = new LinkedList<>();
                dfs(i, -1, c, d, g, n, q);
                if (q.size()!=m) ok = false;
                else {
                    long sum = 0;
                    while (!q.isEmpty())
                        sum+=q.poll();
                    if (sum!=0) ok = false;
                }
            }

        if (!ok) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int i=0; i<m; i++) {
            line = in.readLine().split(" ");
            int a = Integer.parseInt(line[0]), b = Integer.parseInt(line[1]);
            if (p[a]==0) {
                Queue<Long> q = new LinkedList<>();
                dfs(a, -1, c, d, g, n, q);
                while (!q.isEmpty()) {
                    long w = q.poll();
                    if (q.isEmpty())
                        System.out.println(w);
                    else {
                        p[a] = 1;
                        System.out.println(w);
                        p[(int)-w] = 1;
                    }
                }
            }
        }
    }
}
