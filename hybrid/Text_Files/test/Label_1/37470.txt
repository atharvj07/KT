import sys
input = sys.stdin.readline

n = int(input())
cards = []
dmg = 0
count = 1

for _ in range(n):
    cards.append([])
    k = int(input())
    for _ in range(k):
        c, d = map(int, input().split())
        cards[-1].append((c,d))
        
while cards:
    L = []
    for card in cards[0]:
        if card[1] > count:
            L.append(card)
        else:
            dmg += 2 * card[1]
    cards.pop(0)
    if cards:
        L.sort(key = lambda x : x[0])
        cost = 0
        while L and cost + L[0][0] <= 3:
            cost += L[0][0]
            dmg += L[0][1]
            L.pop(0)
    count += 1
    cards = cards[1:]
    
print(dmg)
