import java.util.*;
import java.lang.*;
public class Main {
   public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);
      int n = sc.nextInt();
      int a[] = new int[n];
      for (int i = 0; i < n; i++) a[i] = sc.nextInt();
      int l = 0;
      for (int i = 0; i < n; i++) if (a[i] != 0) l++;
      if (l % 2 == 0) l++;
      if (l <= n) {
         for (int i = 0; i < l; i++) {
            System.out.print(i + " ");
         }
      } else {
         System.out.println(-1);
      }
   }
}
