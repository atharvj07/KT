import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class RookPath {

    static class VerticalSpell {
        int x;

        VerticalSpell(int x) {
            this.x = x;
        }
    }

    static class HorizontalSpell {
        int x1, x2, y;

        HorizontalSpell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<VerticalSpell> verticalSpells = new ArrayList<>();
        List<HorizontalSpell> horizontalSpells = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            verticalSpells.add(new VerticalSpell(scanner.nextInt()));
        }

        for (int i = 0; i < m; i++) {
            horizontalSpells.add(new HorizontalSpell(scanner.nextInt(), scanner.nextInt(), scanner.nextInt()));
        }

        horizontalSpells.sort(Comparator.comparingInt(spell -> spell.y));

        int minSpellsToRemove = 0;
        for (HorizontalSpell horizontalSpell : horizontalSpells) {
            if (verticalSpells.stream().noneMatch(spell -> spell.x >= horizontalSpell.x1 && spell.x <= horizontalSpell.x2)) {
                minSpellsToRemove++;
            }
        }

        System.out.println(minSpellsToRemove);
    }
}
