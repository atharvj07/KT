def max_taste(n, k, tastes, calories):
    dp = [-1]*(n*100+1)
    dp[0] = 0
    for i in range(n):
        for j in range(n*100, tastes[i]-1, -1):
            if dp[j-tastes[i]] != -1:
                dp[j] = max(dp[j], dp[j-tastes[i]] + calories[i])
    answer = -1
    for i in range(1, n*100+1):
        if dp[i] != -1 and dp[i] * k == i:
            answer = max(answer, i)
    return answer

n, k = map(int, input().split())
tastes = list(map(int, input().split()))
calories = list(map(int, input().split()))
print(max_taste(n, k, tastes, calories))
