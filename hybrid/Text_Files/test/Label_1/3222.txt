#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++)
            cin >> a[i];
        sort(a.begin(), a.end());
        long long sum = 0;
        for (int i = 0; i < n - 1; i++)
            sum += a[n - 1] - a[i];
        cout << sum << endl;
    }
    return 0;
}
