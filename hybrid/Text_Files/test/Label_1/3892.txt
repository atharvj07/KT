#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main(){ll n,k;cin>>n>>k;vector<ll>a(n),ans(n);for(ll&i:a)cin>>i;
  sort(a.begin(),a.end());
  if(k>n){cout<<"NO\n";exit(0);}
  for(ll i=0;i<n;){
    for(ll j=0,rk=0;j<k;){
      if(i==n||a[i]!=a[rk]){j++;a[rk]=i+1;rk++;}
      else{ans[i]=j+1;i++;}
    }
  }
  cout<<"YES\n";
  for(ll&i:ans)cout<<i<<" ";cout<<endl;}
