#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5, M = 2e5 + 5, OO = 0x3f3f3f3f;
int n, m;
bool found;
vector<int> g[N];
vector<pair<int, int>> ans;
unordered_map<long long, int> u;

void dfs(int node, int par = 0, int x = 0, int y = 0) {
    u[1ll * x * OO + y] = 1;
    ans.push_back({x, y});
    for (int nei : g[node]) {
        if (nei == par) continue;
        if (!found) {
            bool ok = 1;
            for (int i = 1; i <= 3; ++i) {
                if (u[(1ll * x + i) * OO + y] ||
                    u[1ll * x * OO + y + i] ||
                    u[1ll * x * OO + y - i]) {
                    ok = 0;
                }
            }
            if (ok) {
                dfs(nei, node, x + 1, y);
                if (found) return;
            }
        }
        if (!found) {
            bool ok = 1;
            for (int i = 1; i <= 3; ++i) {
                if (u[(1ll * x - i) * OO + y] ||
                    u[1ll * x * OO + y + i] ||
                    u[1ll * x * OO + y - i]) {
                    ok = 0;
                }
            }
            if (ok) {
                dfs(nei, node, x - 1, y);
                if (found) return;
            }
        }
        if (!found) {
            bool ok = 1;
            for (int i = 1; i <= 3; ++i) {
                if (u[1ll * x * OO + y + i] ||
                    u[1ll * x + i * OO + y] ||
                    u[1ll * x - i * OO + y]) {
                    ok = 0;
                }
            }
            if (ok) {
                dfs(nei, node, x, y + 1);
                if (found) return;
            }
        }
        if (!found) {
            bool ok = 1;
            for (int i = 1; i <= 3; ++i) {
                if (u[1ll * x * OO + y - i] ||
                    u[1ll * x + i * OO + y] ||
                    u[1ll * x - i * OO + y]) {
                    ok = 0;
                }
            }
            if (ok) {
                dfs(nei, node, x, y - 1);
                if (found) return;
            }
        }
    }
}

int main() {
    cin >> n;
    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1);
    if (!found)
        cout << "NO" << endl;
    else {
        cout << "YES" << endl;
        for (const auto &p : ans)
            cout << p.first << " " << p.second << endl;
    }
    return 0;
}
