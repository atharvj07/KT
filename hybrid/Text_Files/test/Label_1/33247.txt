import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt(), c = sc.nextInt(), n = sc.nextInt();
        boolean[][] grid = new boolean[r + 1][c + 1];
        int ans = 0;
        while (n-- > 0) {
            int x = sc.nextInt(), y = sc.nextInt();
            if (!grid[x][y] && check(grid, x, y)) {
                grid[x][y] = true;
                ans++;
            }
        }
        System.out.println(ans);
    }

    private static boolean check(boolean[][] grid, int x, int y) {
        for (int i = 1; i <= grid[0].length - 1; i++) {
            if (grid[1][i] && grid[grid.length - 1][i]) {
                return true;
            }
        }
        for (int i = 1; i <= grid.length - 1; i++) {
            if (grid[i][1] && grid[i][grid[0].length - 1]) {
                return true;
            }
        }
        return false;
    }
}
