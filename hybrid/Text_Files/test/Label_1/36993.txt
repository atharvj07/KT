import sys

def solve():
    n = int(input())
    packages = [list(map(int, input().split())) for _ in range(n)]

    packages.sort()

    path = ""

    for x, y in packages:
        while x > 0:
            path += "R"
            x -= 1
        while y > 0:
            path += "U"
            y -= 1

    if len(path) == sum(x + y for x, y in packages):
        print("YES")
        print(path)
    else:
        print("NO")

for _ in range(int(input())):
    solve()
