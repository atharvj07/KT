#include<bits/stdc++.h>
using namespace std;
int n,a[210],b[210],ans;
int main(){
    cin>>n;
    for(int i=0;i<2*n-1;i++)cin>>a[i],b[i]=a[i];
    for(int i=0;i<2*n-1;i++){
        int tmp=0;
        for(int j=0;j<2*n-1;j++)if(j!=i)tmp+=max(0,a[j]);
        ans=max(ans,tmp);
        a[i]=0-a[i];
    }
    cout<<ans<<endl;
    return 0;
}
