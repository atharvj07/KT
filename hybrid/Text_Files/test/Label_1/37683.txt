import sys
input = sys.stdin.readline

n,m=map(int,input().split())
edges=[]
for _ in range(m):
    a,b=map(int,input().split())
    edges.append((a,b))

adj=[[] for _ in range(n+1)]

for e in edges:
    a,b=e
    adj[a].append(b)
    adj[b].append(a)

vis=[False for _ in range(n+1)]
ans=[]

def dfs(node,par):
    vis[node]=True
    for u in adj[node]:
        if vis[u]==False:
            ans.append((node,u))
            dfs(u,node)

dfs(1,-1)
for e in ans:
    print(*e)
