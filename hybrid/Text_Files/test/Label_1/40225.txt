import sys

digits = sys.stdin.read().splitlines()
n = int(digits[0])
a, b = int(digits[1]), int(digits[2])
if b < a or 10**n - 1 < b:
    print(-1)
    sys.exit(0)
ans = []
while a != b:
    for i in range(len(str(a)) - 1):
        if int(str(a)[i:i + 2]) < int(str(b)[i:i + 2]):
            if str(b)[i - 1] != str(0):
                inc = 1
                ans.append([i, inc])
                a += inc * 10**(len(str(a)) - i - 2)
            else:
                inc = -1
                ans.append([i + 1, inc])
                a += inc * 10**(len(str(a)) - i - 2)
        elif int(str(a)[i:i + 2]) > int(str(b)[i:i + 2]):
            if str(b)[i - 1] != str(9):
                inc = -1
                ans.append([i, inc])
                a += inc * 10**(len(str(a)) - i - 2)
            else:
                inc = 1
                ans.append([i + 1, inc])
                a += inc * 10**(len(str(a)) - i - 2)
        else:
            continue
    if a == b:
        break
    for i in range(len(str(a)) - 1, 0, -1):
        if int(str(a)[i - 1:i + 1]) < int(str(b)[i - 1:i + 1]):
            if str(b)[i - 2] != str(0):
                inc = 1
                ans.append([i - 1, inc])
                a += inc * 10**(len(str(a)) - i - 1)
            else:
                inc = -1
                ans.append([i, inc])
                a += inc * 10**(len(str(a)) - i - 1)
        elif int(str(a)[i - 1:i + 1]) > int(str(b)[i - 1:i + 1]):
            if str(b)[i - 2] != str(9):
                inc = -1
                ans.append([i - 1, inc])
                a += inc * 10**(len(str(a)) - i - 1)
            else:
                inc = 1
                ans.append([i, inc])
                a += inc * 10**(len(str(a)) - i - 1)
        else:
            continue
print(len(ans))
for anss in ans[0:min(len(ans), 10**5)]:
    print(anss[0] + 1, anss[1])
