#include<cstdio>
#include<cstring>
#include<vector>
#include<algorithm>
using namespace std;
const int maxn=2e5+10;
const long long inf=1e18;
typedef long long ll;
char s[maxn];
int main()
{
	int n=0;
	scanf("%s",s);
	n=strlen(s);
	vector<char> t;
	int cnt[26];
	for(int i=0;i<26;i++) cnt[i]=0;
	for(int i=0;i<n;i++)
	{
		cnt[s[i]-'a']++;
		t.push_back(s[i]);
	}
	int ans=0;
	for(int i=0;i<26;i++)
	{
		ans+=cnt[i]%2;
	}
	if(ans>1)
	{
		printf("NO\n");
		return 0;
	}
	int pos=-1;
	for(int i=0;i<26;i++)
	{
		if(cnt[i]%2)
		{
			pos=i;
		}
	}
	vector<vector<int> > c(26,vector<int>());
	for(int i=0;i<t.size();i++)
	{
		char x=t[i];
		x-='a';
		c[x].push_back(i);
	}
	vector<char> p;
	int l=0,r=t.size()-1;
	while(l<r)
	{
		if(t[l]==t[r])
		{
			l++,r--;
			continue;
		}
		int x=t[l]-'a';
		int now=upper_bound(c[x].begin(),c[x].end(),r)-c[x].begin();
		if(c[x].size()==now)
		{
			x=t[r]-'a';
			now=upper_bound(c[x].begin(),c[x].end(),l)-c[x].begin();
		}
		int pos1=c[x][now];
		c[x].erase(c[x].begin()+now);
		swap(t[l],t[pos1]);
		l++;
		r--;
		p.push_back(t[pos1]);
	}
	if(t.size()%2)
	{
		char x=pos+'a';
		p.push_back(x);
	}
	reverse(p.begin(),p.end());
	for(int i=0;i<t.size();i++)
	{
		t[i]=p[i];
	}
	for(int i=0;i<t.size();i++)
	{
		printf("%c",t[i]);
	}
	printf("\n");
	return 0;
}
