#include <iostream>

using namespace std;

int main() {
  char s[4][5];
  for (int i = 0; i < 4; i++) cin >> s[i];
  int cc = 0;
  for (int i = 0; i < 4; i++)
    for (int j = 0; j < 4; j++) if (s[i][j] == '#') cc++;
  if (cc == 4 || cc == 8 || cc % 2 == 0) cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
