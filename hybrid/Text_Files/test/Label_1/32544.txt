import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt(), num = 0;
        char[][] a = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j);
            }
        }
        outer:
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i][j] == '*') {
                    for (int r = 1; i - r >= 0 && i + r < n && j - r >= 0 && j + r < m && a[i - r][j] == '*' && a[i + r][j] == '*' && a[i][j - r] == '*' && a[i][j + r] == '*'; r++) {
                        num++;
                        if (num == k) {
                            System.out.println(i + 1 + " " + j);
                            System.out.println(i - r + 1 + " " + j);
                            System.out.println(i + r + 1 + " " + j);
                            System.out.println(i + " " + (j - r));
                            System.out.println(i + " " + (j + r));
                            break outer;
                        }
                    }
                }
            }
        }
        if (num < k) System.out.println(-1);
    }
}
