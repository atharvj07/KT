#include <iostream>
#include <vector>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long a, b, m;
    cin >> a >> b >> m;
    vector<long long> v = {a};
    long long sum = a;
    while (sum < b) {
      sum += m;
      v.push_back(sum - v.back());
    }
    if (sum == b) {
      cout << v.size() << ' ';
      for (long long x : v) cout << x << ' ';
      cout << endl;
    } else
      cout << -1 << endl;
  }
}
