c, d = map(int, input().split())
n, m = map(int, input().split())
k = int(input())

required_winners = n*m - k
problems = 0

if required_winners > 0:
    main_rounds = required_winners // n
    remainder_winners = required_winners % n
    
    if remainder_winners > 0:
        if d < c * remainder_winners:
            problems = main_rounds * c + d
        else:
            problems = (main_rounds + 1) * c
    else:
        problems = main_rounds * c

print(problems)
