import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = sc.nextInt();
        int[] b = new int[k];
        for (int i = 0; i < k; i++) {
            b[i] = sc.nextInt();
        }

        List<String> actions = new ArrayList<>();
        boolean[] eaten = new boolean[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = a[i];
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                if (!eaten[j] && w[j] > 0 && w[j] <= b[i]) {
                    if (j > 0 && w[j - 1] <= w[j]) {
                        actions.add((j + 1) + " L");
                        w[j - 1] += w[j];
                        eaten[j] = true;
                    } else if (j + 1 < n && w[j + 1] <= w[j]) {
                        actions.add((j + 1) + " R");
                        w[j + 1] += w[j];
                        eaten[j] = true;
                    }
                }
            }
        }

        boolean possible = true;
        for (int i = 0; i < n; i++) {
            if (!eaten[i]) {
                possible = false;
                break;
            }
        }

        if (possible) {
            System.out.println("YES");
            for (String action : actions) {
                System.out.println(action);
            }
        } else {
            System.out.println("NO");
        }
    }
}
