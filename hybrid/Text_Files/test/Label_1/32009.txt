import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = input.nextInt();
        }
        Arrays.sort(x);
        for (int i = 0; i < n; i++) {
            int m = (n + 1) / 2;
            if (n % 2 == 0) {
                if (i < m - 1) {
                    System.out.println(x[m - 1]);
                } else if (i == m - 1 || i == m) {
                    System.out.println(x[m]);
                } else {
                    System.out.println(x[m + 1]);
                }
            } else {
                if (i < m - 1) {
                    System.out.println(x[m - 1]);
                } else if (i == m - 1) {
                    System.out.println(x[m]);
                } else {
                    System.out.println(x[m + 1]);
                }
            }
        }
    }
}
