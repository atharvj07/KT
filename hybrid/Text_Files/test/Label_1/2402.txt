#include <bits/stdc++.h>

using namespace std;

const int MAXN = 100;
int n, m;
char f[MAXN][MAXN + 1];
long long pref[MAXN][MAXN];

signed main() {
    cin >> n >> m;
    for (int i = 0; i < n; ++i) { cin >> f[i]; }
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) { pref[i + 1][j + 1] = pref[i][j + 1] + pref[i + 1][j] - pref[i][j]; }
        for (int j = 0; j < m; ++j) { pref[i + 1][j + 1] += (f[i][j] == 'W' ? 1LL : -1LL); }
    }

    int ans = n * m;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            for (int x = i; x <= n; ++x) {
                for (int y = j; y <= m; ++y) {
                    long long sum = pref[x][y] - pref[x][j - 1] - pref[i - 1][y] + pref[i - 1][j - 1];
                    ans = min(ans, 2 + (sum != 0));
                }
            }
        }
    }

    cout << ans << endl;
    return 0;
}
