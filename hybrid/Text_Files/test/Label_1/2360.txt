#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, ans = 0;
    cin >> n;
    vector<long long> a(n), b(n);
    for (long long i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (long long i = 0; i < n; i++) {
        cin >> b[i];
    }
    vector<long long> lst(n, 0);
    for (long long i = n - 1; i >= 0; i--) {
        for (long long j = 0; j < i; j++) {
            if (a[j] >= a[i]) { lst[i] = j; }
        }
    }
    for (long long i = 0; i < n; i++) {
        ans += b[lst[i]];
    }
    cout << ans << '\n';
}
