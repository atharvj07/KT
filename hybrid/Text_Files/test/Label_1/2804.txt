#include <bits/stdc++.h>

using namespace std;

const int MOD = 1000000007;

int32_t main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);
  int n, m;
  cin >> n >> m;
  vector<int> b(n);
  vector<int> cnt(m + 1, 0);
  for (int i = 0; i < n; ++i) {
    cin >> b[i];
    ++cnt[b[i]];
  }
  long long ans = (1LL * n * (n - 1)) / 2;
  for (int i = 1; i <= m; ++i) {
    ans -= (1LL * cnt[i] * (cnt[i] - 1)) / 2;
  }
  cout << ans << '\n';
}
