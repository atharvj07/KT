import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[3];
        int[] b = new int[3];
        if (sc.hasNextInt()) {
            for (int i = 0; i < 3; i++) {
                a[i] = sc.nextInt();
            }
            for (int i = 0; i < 3; i++) {
                b[i] = sc.nextInt();
            }
            int[] c = new int[9];
            for (int i = 0; i < 3; i++) {
                for (int j = 0; j < 3; j++) {
                    c[i * 3 + j] = sc.nextInt();
                    if (c[i * 3 + j] != a[i] + b[j]) {
                        System.out.println("No");
                        return;
                    }
                }
            }
            System.out.println("Yes");
        }
    }
}
