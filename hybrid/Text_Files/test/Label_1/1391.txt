#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;

string s;
int main() {
    cin >> s;
    int las = 0;
    while (las != s.size()) {
        int len = s.size() - las;
        int cnt = 0;
        for (int i = las; i < s.size() - 1; ++i)
            if (s[i] == s[i + 1]) ++cnt;
        if (cnt == 0) {
            cout << len << ' ' << s.substr(las, len) << '\n';
            las = s.size();
            continue;
        }
        int pos;
        for (int i = las; i < s.size() - 1; ++i)
            if (s[i] == s[i + 1]) {
                pos = i;
                break;
            }
        if (len <= 10)
            cout << len << ' ' << s.substr(las, len) << '\n';
        else
            cout << len << ' ' << s.substr(las, 5) << "..." << s.substr(len - 2, 2)
                 << '\n';
        s.erase(s.begin() + pos, s.begin() + pos + 2);
    }
    return 0;
}
