answer = 0
MOD = 10**9 + 7
base = 64
s = input().strip()
value = 0
cnt = 0
for ch in s:
    value <<= 6
    if ch >= '0' and ch <= '9':
        value += ord(ch) - ord('0')
    elif ch >= 'A' and ch <= 'Z':
        value += ord(ch) - ord('A') + 10
    elif ch >= 'a' and ch <= 'z':
        value += ord(ch) - ord('a') + 36
    elif ch == '-':
        value += 62
    elif ch == '_':
        value += 63
    cnt += 1
mask = (base**cnt) - 1
answer = (answer + 1) % MOD
for i in range(1, cnt):
    mask = mask >> 6
    submask = value & mask
    if submask == 0:
        answer = (answer + pow(2, cnt - i - 1, MOD)) % MOD
print(answer)
