import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int N = in.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = in.nextInt();
        }
        int[] gcd = new int[N];
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                gcd[i] = gcd(a[i], gcd[i]);
                gcd[j] = gcd(a[j], gcd[j]);
                gcd[i] = gcd(gcd[i], gcd[j]);
            }
        }
        int ans = 0;
        for (int i = 0; i < N; i++) {
            ans = ans + a[i] / gcd[i];
        }
        System.out.println(ans);
    }

    public static int gcd(int a, int b) {
        while (b > 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
