from collections import defaultdict

def solve():
    n = int(input())
    a = list(map(int, input().split()))

    if sum(a) < n - 1:
        print("NO")
        return

    g = defaultdict(list)

    for i in range(1, n):
        if a[i] == 0:
            print("NO")
            return
        for j in range(i + 1, min(i + a[i] + 1, n) + 1):
            if a[i] == 0 or a[j] == 0:
                print("NO")
                return
            g[i].append(j)
            g[j].append(i)
            a[i] -= 1
            a[j] -= 1

    print("YES", n - 1)
    print(sum(len(g[i]) for i in range(1, n + 1)))

    for i in range(1, n + 1):
        for j in g[i]:
            print(i, j)

solve()
