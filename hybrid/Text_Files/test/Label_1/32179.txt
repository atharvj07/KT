import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] grid = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                grid[i][j] = sc.nextInt();
            }
        }

        int[] a = new int[3];
        int[] b = new int[3];
        for (int i = 0; i < 3; i++) {
            a[i] = grid[i][0];
            for (int j = 1; j < 3; j++) {
                a[i] -= grid[i][j];
            }
        }
        for (int j = 0; j < 3; j++) {
            b[j] = grid[0][j];
            for (int i = 1; i < 3; i++) {
                b[j] -= grid[i][j];
            }
        }

        boolean correct = true;
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                if (grid[i][j] != a[i] + b[j]) {
                    correct = false;
                }
            }
        }

        System.out.println(correct ? "Yes" : "No");
    }
}
