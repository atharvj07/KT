#include <iostream>
#include <cstring>
using namespace std;

int n, m, a, b;
bool p[300002];

int main() {
    cin >> n >> m;
    while (m--) {
        cin >> a >> b;
        p[a] = p[b] = true;
    }
    n = (n + 1) / 2;
    for (int i = 0; i <= n; ++i)
        if (!p[i] && !p[n + i]) {
            cout << "NO" << endl;
            return 0;
    }
    cout << "YES" << endl;
    return 0;
}
