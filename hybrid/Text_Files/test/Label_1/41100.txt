n, m = map(int, input().split())
s = input()
f = [0] * (n + 1)
for i in range(n - 1, 0, -1):
    f[i] = f[i + 1]
    if s[i] == s[i - 1] == '.':
        f[i] += 1
a = [0] * (n + 1)
for i in range(1, n):
    if s[i] == s[i - 1] == '.':
        a[i - 1] = 1
for i in range(m):
    p, c = input().split()
    p = int(p)
    s = s[:p - 1] + c + s[p:]
    if p > 1 and s[p - 1] == s[p - 2] == '.':
        a[p - 2] = 1
    f[p - 1] -= a[p - 1]
    if p < n and s[p] == s[p - 1] == '.':
        a[p - 1] = 1
    f[p - 1] += a[p - 1]
    print(f[p - 1])
