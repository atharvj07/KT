import java.util.Scanner;

public class VasilyTheBear {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        long[][] dp = new long[n + 1][k + 1];
        dp[0][0] = 1;

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= k; j++) {
                // Choose 3 points from the current and previous layer
                for (int x1 = 0; x1 < 4 * i; x1++) {
                    for (int y1 = 0; y1 < 4 * i; y1++) {
                        for (int x2 = 0; x2 < 4 * (i - 1); x2++) {
                            for (int y2 = 0; y2 < 4 * (i - 1); y2++) {
                                for (int x3 = 0; x3 < 4 * (i - 1); x3++) {
                                    for (int y3 = 0; y3 < 4 * (i - 1); y3++) {
                                        if (isValidTriangle(x1, y1, x2, y2, x3, y3)) {
                                            dp[i][j] = (dp[i][j] + dp[i - 1][j - 1]) % MOD;
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        System.out.println(dp[n][k]);
    }

    // Check if the triangle formed by the given points is valid
    private static boolean isValidTriangle(int x1, int y1, int x2, int y2, int x3, int y3) {
        // Check if the points are distinct
        if (x1 == x2 && y1 == y2) return false;
        if (x1 == x3 && y1 == y3) return false;
        if (x2 == x3 && y2 == y3) return false;

        // Check if any two points are connected by a segment
        if ((x1 == 0 && y1 == 0) || (x2 == 0 && y2 == 0) || (x3 == 0 && y3 == 0)) {
            return true;
        }

        if (x1 == x2 || y1 == y2) {
            return true;
        }

        if (x1 == x3 || y1 == y3) {
            return true;
        }

        if (x2 == x3 || y2 == y3) {
            return true;
        }

        return false;
    }
}
