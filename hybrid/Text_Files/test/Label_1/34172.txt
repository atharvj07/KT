import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int[] dp = new int[n];
        int[] prev = new int[n];
        int max = 0;
        int index = -1;
        for (int i = 0; i < n; i++) {
            dp[i] = 1;
            prev[i] = -1;
            for (int j = 0; j < i; j++) {
                if (arr[i] > arr[j] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    prev[i] = j;
                }
            }
            if (dp[i] > max) {
                max = dp[i];
                index = i;
            }
        }

        int[] ans = new int[max];
        int curr = index;
        for (int i = max - 1; i >= 0; i--) {
            ans[i] = arr[curr];
            curr = prev[curr];
        }

        System.out.println(max);
        for (int i = 0; i < max; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
