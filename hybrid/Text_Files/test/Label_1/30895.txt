import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int T = input.nextInt();
            if (T == 0) {
                break;
            }
            int[] t = new int[T + 1];
            for (int i = 1; i <= T; i++) {
                t[i] = input.nextInt();
            }
            int N = input.nextInt();
            int[] D = new int[N + 1];
            int[] M = new int[N + 1];
            for (int i = 1; i <= N; i++) {
                D[i] = input.nextInt();
                M[i] = input.nextInt();
            }
            int ans = 0;
            int day = 1;
            int hour = 0;
            for (int i = 1; i <= N; i++) {
                while (true) {
                    hour += t[day];
                    day++;
                    if (day > T) {
                        day = 1;
                    }
                    if (hour >= M[i] && day <= D[i]) {
                        break;
                    }
                    if (day == D[i] && hour > M[i]) {
                        ans++;
                        hour = 0;
                        day = 1;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
