import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] cnt = new int[100001];
        int[] l = new int[100001];
        int[] r = new int[100001];
        for (int i = 0; i < n; i++) {
            cnt[a[i]]++;
            if (cnt[a[i]] == 1) {
                l[a[i]] = i;
            }
            r[a[i]] = i;
        }
        int[] ans = new int[n];
        int len = 0;
        for (int i = 0; i < n; i++) {
            if (cnt[a[i]] <= 1) {
                ans[len++] = a[i];
            } else {
                int ll = l[a[i]];
                int rr = r[a[i]];
                int c = 1;
                while (ll < rr) {
                    if (a[ll] == a[rr]) {
                        ll++;
                        rr--;
                        c++;
                    } else {
                        break;
                    }
                }
                if (c * 2 == rr - ll + 1) {
                    for (int j = ll; j <= rr; j++) {
                        cnt[a[j]]--;
                    }
                } else {
                    ans[len++] = a[i];
                }
            }
        }
        System.out.println(len);
        for (int i = 0; i < len; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
