import sys

n = int(input())
l = sorted(map(int, input().split()))
if l[0] == l[-1]: print(-1); sys.exit(0)
if n == 1: print(-1); sys.exit(0)
d = []
for i in range(1, n):
    d.append(l[i] - l[i - 1])

def solve():
    if len(set(d)) > 1: return -1
    else: return l[0] - d[0]

if solve() == -1:
    print(-1)
else:
    ans = [solve()]
    sum = ans[0]
    for i in d:
        sum += i
        ans.append(sum)
    print(len(ans))
    print(' '.join(map(str, ans)))
