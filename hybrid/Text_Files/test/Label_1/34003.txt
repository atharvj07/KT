import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            boolean canClear = true;
            for (int j = 0; j < n; j++) {
                if (a[j] % 2 == 1) {
                    canClear = false;
                    break;
                }
            }
            System.out.println(canClear ? "YES" : "NO");
        }
    }
}
