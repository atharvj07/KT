#include <bits/stdc++.h>
using namespace std;

const int MM = 2010;
char g1[MM][MM], g2[MM][MM];
int N, M;
int main() {
	cin >> N >> M;
	for (int i = 1; i <= N; i++)
		for (int j = 1; j <= M; j++) cin >> g1[i][j];
	for (int i = 1; i <= M; i++)
		for (int j = 1; j <= N; j++) cin >> g2[i][j];
	for (int x = 1; x <= N - M + 1; x++) {
		for (int y = 1; y <= M - M + 1; y++) {
			bool got = 1;
			for (int i = 0; i < M; i++) {
				for (int j = 0; j < M; j++) {
					if (g1[x + i][y + j] != g2[i + 1][y + j]) { got = 0; }
				}
			}
			if (got) {
				cout << x << ' ' << y << endl;
				return 0;
			}
		}
	}
	return 0;
}
