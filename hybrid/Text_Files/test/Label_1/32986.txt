import java.util.Scanner;
import java.util.HashMap;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        HashMap<Integer, Integer> lastVisit = new HashMap<>();
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            lastVisit.put(a[i], i);
        }
        int answer = 0;
        int last = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (lastVisit.get(a[i]) < last) {
                answer = a[i];
                break;
            }
            last = lastVisit.get(a[i]);
        }
        System.out.println(answer);
    }
}
