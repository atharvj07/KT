#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    bool used[26] = {false};
    for (int i = 0; i < n; i++) {
        if ('a' <= s[i] && s[i] <= 'z') {
            used[s[i] - 'a'] = true;
        } else {
            used[s[i] - 'A'] = true;
        }
    }
    bool isPangram = true;
    for (int i = 0; i < 26; i++) {
        isPangram &= used[i];
    }
    cout << (isPangram ? "YES" : "NO") << endl;
    return 0;
}
