for t in range(int(input())):
    n,m=map(int,input().split())
    a=list(map(int,input().split()))
    b=list(map(int,input().split()))
    i,j=0,0
    while i<n and j<m:
        if b[j]==a[i]:
            j+=1
        i+=1
    print('YES'if j==m else'NO')
    if j==m:
        print(j,end=' ')
        i,j=0,0
        while i<n and j<m:
            if b[j]==a[i]:
                print(b[j],end=' ')
                j+=1
            i+=1
        print()
