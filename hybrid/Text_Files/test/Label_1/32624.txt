import java.util.Scanner;

public class ZebraChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] photo = new int[n];
        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        boolean isZebra = true;
        int stripeWidth = -1;
        for (int i = 0; i < n; i++) {
            if (photo[i] == 0) {
                if (stripeWidth == -1) {
                    stripeWidth = 1;
                } else if (stripeWidth % 2 != 0) {
                    isZebra = false;
                    break;
                } else {
                    stripeWidth++;
                }
            } else {
                if (stripeWidth == -1) {
                    stripeWidth = 0;
                } else if (stripeWidth % 2 == 0) {
                    isZebra = false;
                    break;
                } else {
                    stripeWidth++;
                }
            }
        }

        System.out.println(isZebra ? "YES" : "NO");
    }
}
