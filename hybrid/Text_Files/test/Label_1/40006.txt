def solve_sequence_operations(n, a):
    a.sort()
    if a[0] != 1:
        return -1
    
    m = 1
    s = [1]
    for i in range(1, n):
        x = a[i]
        sum_set = set()
        for j in range(len(s)):
            for k in range(j, len(s)):
                sum_set.add(s[j] + s[k])
        if x not in sum_set:
            m += 1
            s.append(x)
        else:
            s.append(x)
            s.sort()
            while len(s) > m:
                del s[0]
    return m

n = int(input())
a = list(map(int, input().split()))
print(solve_sequence_operations(n, a))
