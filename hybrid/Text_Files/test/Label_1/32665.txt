import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = in.nextLong();
        }
        Arrays.sort(b);
        long[] a = new long[n];
        a[0] = b[0];
        for (int i = 1; i < n; i++) {
            a[i] = a[i - 1] ^ b[i];
        }
        boolean valid = true;
        for (int i = 1; i < n; i++) {
            valid &= a[i] > a[i - 1];
        }
        if (!valid) {
            System.out.println("No");
            return;
        }
        System.out.println("Yes");
        for (int i = 0; i < n; i++) {
            System.out.print(b[i] + " ");
        }
    }
}
