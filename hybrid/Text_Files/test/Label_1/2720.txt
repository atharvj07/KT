#include <iostream>
#include <set>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  string home;
  cin >> home;
  set<string> airports;
  airports.insert(home);
  while (n--) {
    string s;
    cin >> s;
    int pos = s.find("->");
    string a = s.substr(0, pos);
    string b = s.substr(pos + 2);
    if (airports.count(a)) {
      airports.erase(a);
    } else {
      airports.insert(a);
    }
    if (airports.count(b)) {
      airports.erase(b);
    } else {
      airports.insert(b);
    }
  }
  if (airports.count(home)) {
    cout << "home" << endl;
  } else {
    cout << "contest" << endl;
  }
  return 0;
}
