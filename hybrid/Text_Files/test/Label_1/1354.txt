#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int get(vector<int>&a,int m,int n){
	int ans=0;
	for(int i=0;i<n;i++){
		int cnt=0;
		for(int j=i;j<n;j++){
			cnt+=(a[j]<=m);
			if(cnt*2>j-i+1)ans++;
		}
	}
	return ans;
}
int main(){
	int n,m;cin>>n>>m;
	vector<int>a(n);
	for(int i=0;i<n;i++)cin>>a[i];
	cout<<get(a,m,n);
	return 0;
}
