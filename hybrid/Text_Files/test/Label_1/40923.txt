def solve(n):
    a = [0] * n
    segments = [(-n, 0, n-1)] # Using a min heap where each element is a tuple (-segment_length, left_index, right_index)
    for i in range(1, n+1):
        _, l, r = heapq.heappop(segments)
        if (r-l) % 2 == 0: # odd segment length
            mid = (l+r)//2
        else: # even segment length
            mid = (l+r-1)//2
        a[mid] = i
        if mid-1 >= l:
            heapq.heappush(segments, (-(mid-1-l), l, mid-1))
        if mid+1 <= r:
            heapq.heappush(segments, (-(r-mid-1), mid+1, r))
    return a

import heapq
t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    print(*solve(n))
