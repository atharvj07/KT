#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main() {
    int k;
    cin >> k;
    string s;
    cin >> s;
    int n = s.size();
    vector<int> dp(n + 1, n + 1);
    dp[0] = 0;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j < i; ++j) {
            if (dp[j] + 1 + i - j <= k) {
                dp[i] = min(dp[i], dp[j] + 1 + i - j);
            }
            if (s[j] == '-' && dp[j] + i - j <= k) {
                dp[i] = min(dp[i], dp[j] + i - j);
            }
        }
    }
    cout << dp[n] << endl;
    return 0;
}
