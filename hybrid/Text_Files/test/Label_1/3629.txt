#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    int r[n], b[n];
    int cnt_r = 0, cnt_b = 0;
    for (int i = 0; i < n; i++) {
        cin >> r[i];
        cnt_r += r[i];
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
        cnt_b += b[i];
    }
    if (cnt_r <= cnt_b)
        cout << -1 << endl;
    else
        cout << 1 << endl;
    return 0;
}
