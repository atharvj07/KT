n, m = map(int, input().split())
x = list(map(int, input().split()))
p = list(map(int, input().split()))

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

# Calculate the gcd of differences between all the events
diff_gcd = x[1] - x[0]
for i in range(2, n):
    diff_gcd = gcd(diff_gcd, x[i] - x[i-1])

# Try to find a matching interval in the provided options
for j in range(m):
    if diff_gcd % p[j] == 0:
        print("YES")
        print(f"{x[0]} {j+1}")
        break
else:
    print("NO")
