import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner in = new Scanner(System.in);
        String s = in.next();
        int n = s.length();
        int[][] lps = new int[n][n + 1];
        for (int i = n - 1; i >= 0; i--) {
            lps[i][i + 1] = 1;
            for (int j = i + 1; j < n; j++) {
                if (s.charAt(i) == s.charAt(j)) {
                    lps[i][j + 1] = lps[i + 1][j] + 2;
                } else {
                    lps[i][j + 1] = Math.max(lps[i][j], lps[i + 1][j]);
                }
            }
        }
        int max = 0;
        int start = 0;
        int end = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (lps[i][j] > max && (j - i + 1) % 2 == 0) {
                    max = lps[i][j];
                    start = i;
                    end = j;
                }
            }
        }
        if (max < 100) {
            max = 0;
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    if (lps[i][j] > max) {
                        max = lps[i][j];
                        start = i;
                        end = j;
                    }
                }
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int i = start; i <= end; i++) {
            sb.append(s.charAt(i));
        }
        System.out.println(sb);
    }
}
