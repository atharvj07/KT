#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++)
      cin >> a[i];
    bool ok = true;
    if (n == 2)
      ok = (a[0] == a[1]);
    else {
      int x = a[0];
      for (int i = 1; i < n; i++) {
        x ^= a[i];
        if (i > 1 && x != 0)
          ok = false;
      }
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
  return 0;
}
