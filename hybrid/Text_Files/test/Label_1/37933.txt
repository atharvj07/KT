import sys

MOD = 998244353

def solve():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    a.sort(reverse=True)
    if a[0] > n - 1:
        print(0)
        return

    dp = [[0] * (n + 1) for _ in range(n + 1)]
    dp[0][0] = 1
    for i in range(n):
        for j in range(n + 1):
            if j + a[i] <= n:
                dp[i + 1][j + a[i]] = (dp[i + 1][j + a[i]] + dp[i][j]) % MOD
            dp[i + 1][j] = (dp[i + 1][j] + dp[i][j]) % MOD

    print(dp[n][n - 1])

for _ in range(int(sys.stdin.readline())):
    solve()
