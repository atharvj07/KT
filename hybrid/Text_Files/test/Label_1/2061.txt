#include <bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
vector<pii>adj[200001];
vector<int>path;
bool vis[200001],check[200001];
void dfs(int s, int p){
    vis[s] = 1;
    path.push_back(s);
    for(pii x : adj[s]){
        if(x.first != p){
            dfs(x.first,s);
        }
    }
}
bool ans[200001];
int main(){
    int n, m,a,b,k;
    cin >> n >> m;
    for(int i=0;i<n-1;++i){
        cin >> a >> b;
        adj[a].push_back({b,1});
        adj[b].push_back({a,1});
    }
    dfs(1,0);
    for(int i=0;i<m;++i){
        cin >> k;
        while(k--){
            cin >> a;
            for(int x : path){
                if(a==x){
                    check[i] = 1;
                }
                else if(adj[x][0].first == a || adj[x][0].second == a){
                    check[i] = 1;
                }
            }
        }
    }
    for(int i=0;i<m;++i){
        if(check[i])cout << "YES" << endl;
        else cout << "NO" << endl;
    }
}
