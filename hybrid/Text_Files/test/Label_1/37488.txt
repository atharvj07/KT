t = int(input())
for _ in range(t):
    n = int(input())
    friends = [tuple(map(int, input().split())) for _ in range(n)]
    modified_friends = [(min(h, w), max(h, w), i) for i, (h, w) in enumerate(friends)]
    modified_friends.sort()

    result = [-1] * n
    min_width = float('inf')
    candidate = -1

    for h, w, i in modified_friends:
        if w < min_width:
            min_width = w
            candidate = i
        else:
            result[i] = candidate + 1

    print(*result)
