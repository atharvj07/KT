#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> sticks(n);
  for (int i = 0; i < n; i++) cin >> sticks[i];

  int min_cost = 1e9, min_t = -1;
  for (int t = 1; t <= 100; t++) {
    int cost = 0;
    for (int i = 0; i < n; i++) {
      cost += min(abs(sticks[i] - t), abs(sticks[i] - (t + 1)));
    }
    if (cost < min_cost) {
      min_cost = cost;
      min_t = t;
    }
  }

  cout << min_t << " " << min_cost << endl;
  return 0;
}
