import sys
# sys.setrecursionlimit(10**7)

for _ in range(int(input())):
    n = int(input())
    arr = list(map(int, input().split()))
    freq = {}
    for i in arr:
        if i not in freq: freq[i] = 0
        freq[i] += 1
    
    if any([x > (n + 1) // 2 for x in freq.values()]): print(-1)
    else:
        arr.sort()
        ans = 0
        i = 0
        j = n - 1
        while i < j:
            if arr[i] == arr[i + 1]:
                arr[j], arr[i + 1] = arr[i + 1], arr[j]
                j -= 1
                ans += 1
            i += 1
        
        print(ans)
