#include <bits/stdc++.h>
using namespace std;
#define ll long long
struct data {
    int val, id;
    bool operator<(const data &p) const {
        return val > p.val;
    }
} v[200005];
int n, k, ans, rem[200005];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> k;
        for (int i = 1; i <= n; ++i) {
            cin >> v[i].val;
            v[i].id = i;
        }
        sort(v + 1, v + 1 + n);
        priority_queue<int, vector<int>, greater<int>> pq;
        for (int i = 1; i <= n; ++i) {
            if (i <= n - k * 2) rem[v[i].id] = 1;
            if (k && v[i].val > 1) pq.push(v[i].id);
        }
        for (int i = n - k * 2 + 1; i <= n; ++i) {
            while (!pq.empty() && v[pq.top()].val % v[i].val) pq.pop();
            if (!pq.empty()) {
                int x = pq.top();
                pq.pop();
                if (rem[x] == 0) continue;
                rem[x] = 0;
                k--;
                ans += v[i].val / v[x].val;
            }
        }
        for (int i = 1; i <= n; ++i) ans += rem[i] * v[i].val;
        cout << ans << '\n';
        ans = 0;
    }
    return 0;
}
