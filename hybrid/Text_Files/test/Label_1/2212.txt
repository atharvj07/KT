#include <bits/stdc++.h>
#define ll long long int
using namespace std;

vector <ll> g[123456];
bool mrk[123456];
ll dis[123456], ans = -1;

void bfs(ll n) {
	priority_queue <pair<ll, ll>> pq;
	pq.push({0, n});
	dis[n] = 0;
	mrk[n] = true;
	while(!pq.empty()) {
		ll now = pq.top().second;
		ll cost = -pq.top().first;
		pq.pop();
		if(cost > dis[now])
			continue;
		for(ll x : g[now]) {
			if(!mrk[x]) {
				pq.push({-(cost + 1), x});
				dis[x] = cost + 1;
				mrk[x] = true;
			}
		}
	}
}

int main() {
	ll n, m, last, sum_ans = 0;
	cin >> n >> m;
	while(m--) {
		ll x;
		cin >> x;
		mrk[x] = true;
	}
	for(ll i = 1; i < n; i++) {
		ll x, y;
		cin >> x >> y;
		g[x].push_back(y);
		g[y].push_back(x);
	}
	for(ll i = 1; i <= n; i++) {
		if(!mrk[i]) {
			bfs(i);
			if(ans == -1 || dis[i] < dis[ans]) {
				ans = i;
			}
			for(ll j = 1; j <= n; j++) {
				if(mrk[j] && dis[ans] < dis[j]) {
					sum_ans += dis[j] - dis[ans];
				}
			}
		}
	}
	cout << ans << endl;
	cout << sum_ans << endl;
	return 0;
}
