#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i].first;
    v[i].second = i;
    p[i] = i;
  }
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (v[i].second == p[i]) continue;
    ans++;
    swap(p[v[i].second], p[i]);
    cout << p[i] << " " << v[i].second << endl;
  }
  cout << ans << endl;
}
