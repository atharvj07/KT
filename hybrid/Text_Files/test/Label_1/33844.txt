import java.util.Scanner;

public class PermutationWithPeaks {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();

            if (k > n - 1 || k < 0) {
                System.out.println(-1);
                continue;
            }

            int[] a = new int[n];

            // Construct the first k peaks
            for (int i = 0; i < k; i++) {
                a[i * 2 + 1] = n - i;
            }

            // Fill the remaining elements in increasing order
            int index = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] == 0) {
                    a[i] = ++index;
                }
            }

            // Print the permutation
            for (int num : a) {
                System.out.print(num + " ");
            }
            System.out.println();
        }
    }
}
