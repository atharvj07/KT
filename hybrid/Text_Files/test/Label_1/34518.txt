import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    private static final int RIGHT_HANDED = 0;
    private static final int LEFT_HANDED = 1;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] dominantHands = new int[n];
        int[] frustrations = new int[n];

        String[] line = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            dominantHands[i] = Integer.parseInt(line[i]);
        }

        line = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            frustrations[i] = Integer.parseInt(line[i]);
        }

        int minFrustration = Integer.MAX_VALUE;
        int[] bestOrder = null;

        // Try all possible permutations of the order of the sages
        for (int[] permutation : permutations(dominantHands)) {
            int frustration = 0;
            for (int i = 0; i < n; i++) {
                if (permutation[i] == RIGHT_HANDED && permutation[(i + 1) % n] == LEFT_HANDED) {
                    frustration += frustrations[i];
                } else if (permutation[i] == LEFT_HANDED && permutation[(i - 1 + n) % n] == RIGHT_HANDED) {
                    frustration += frustrations[i];
                }
            }

            if (frustration < minFrustration) {
                minFrustration = frustration;
                bestOrder = permutation;
            }
        }

        System.out.println(minFrustration);
    }

    // Generates all permutations of a given array
    private static int[][] permutations(int[] arr) {
        int n = arr.length;
        int[][] permutations = new int[n][n];

        int[] temp = new int[n];
        for (int i = 0; i < n; i++) {
            temp[i] = i;
        }

        permutations[0] = temp;

        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                temp = permutations[j].clone();
                int tempVar = temp[i];
                temp[i] = temp[j];
                temp[j] = tempVar;
                permutations[i] = temp;
            }
        }

        return permutations;
    }
}
