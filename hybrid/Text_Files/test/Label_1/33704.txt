import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (vis[i]) {
                continue;
            }
            int cnt = 1;
            vis[i] = true;
            int j = i;
            while (p[j] != i + 1) {
                j = p[j] - 1;
                cnt++;
                vis[j] = true;
            }
            if (cnt != d[i]) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
