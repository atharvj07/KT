#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int dp[n + 1] = {0};
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    dp[i] = dp[i - 1] * 4;
    if (i >= 2) dp[i] += dp[i - 2];
    if (i >= 5) dp[i] += dp[i - 5];
    if (i >= 10) dp[i] += dp[i - 10];
    if (i >= 50) dp[i] += dp[i - 50];
  }
  cout << dp[n] << endl;
  return 0;
}
