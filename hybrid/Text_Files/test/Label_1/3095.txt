#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define fo(i, n) for (int i = 0; i < n; i++)
#define deb(x) cout << #x << "=" << x << endl
#define deb2(x, y) cout << #x << "=" << x << ", " << #y << "=" << y << endl
long long get_ones(long long n, vector<pair<ll, ll>> &arr) {
  vector<bool> vis(n + 1, false);
  for (int i = 0; i < arr.size(); i++) {
    if (arr[i].first == arr[i].second) continue;
    if (vis[arr[i].first] == vis[arr[i].second]) {
      vis[arr[i].first] = !vis[arr[i].first];
      vis[arr[i].second] = !vis[arr[i].second];
    }
  }
  ll ans = 0;
  for (auto x : vis) ans += x;
  return ans;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    ll n, x, m;
    cin >> n >> x >> m;
    vector<pair<ll, ll>> arr(m);
    fo(i, m) cin >> arr[i].first >> arr[i].second;
    cout << get_ones(n, arr) << endl;
  }
}
