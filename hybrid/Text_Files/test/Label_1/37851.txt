import sys

n = int(input())
grid = [list(map(int, input().split())) for i in range(n)]
rsum = [sum(row) for row in grid]
csum = [sum(col) for col in zip(*grid)]
dsum = sum([grid[i][i] for i in range(n)]) + sum([grid[i][n-i-1] for i in range(n)])
if min(rsum) != max(rsum) or min(csum) != max(csum) or dsum != min(rsum):
    print(-1)
    sys.exit()
for i in range(n):
    for j in range(n):
        if grid[i][j] == 0:
            print(min(rsum) - sum(grid[i]) + sum(grid[j]) - grid[j][j] + dsum - grid[i][i] - grid[i][n-i-1])
            sys.exit()
