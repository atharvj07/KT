import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        int[] min = new int[n + 1];
        int[] max = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            min[i] = i;
            max[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int j = a[i];
            if (min[j] == max[j]) {
                continue;
            }
            min[j] = Math.min(min[j], j);
            max[j] = Math.max(max[j], j);
            for (int k = j - 1; k >= min[j]; k--) {
                min[k + 1] = min[k];
                max[k + 1] = max[k];
            }
            min[j] = j;
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(min[i] + " " + max[i]);
        }
        scanner.close();
    }
}
