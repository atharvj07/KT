import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int[] suf = new int[n];
        int sum = 0;
        for(int i = n - 1; i >= 0; i--) {
            suf[i] = sum;
            sum += a[i];
        }
        if(sum == 0) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        int k = 0;
        for(int i = 0; i < n; i++) {
            sum += a[i];
            if(sum == 0 || i == n - 1) {
                System.out.println(++k);
                System.out.println((i == 0 ? 1 : suf[i] + 1) + " " + (i + 1));
            }
        }
    }
}
