import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner scanner = new java.util.Scanner(System.in);

        int n = scanner.nextInt();

        int[] xs = new int[n];
        for (int i = 0; i < n; i++) {
            xs[i] = scanner.nextInt();
        }

        int[] ys = new int[n];
        for (int i = 0; i < n; i++) {
            ys[i] = scanner.nextInt();
        }

        int count = 0;
        Arrays.sort(xs);
        Arrays.sort(ys);

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if ((xs[i] ^ ys[j]) < 2 * n) {
                    count++;
                }
            }
        }

        System.out.println(count % 2 == 0 ? "Karen" : "Koyomi");
    }
}
