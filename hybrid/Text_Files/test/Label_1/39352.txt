n=int(input())
S=list(map(int,input().split()))
S.sort()
m=min(S)
M=max(S)
p={m,m+1}
ans=0
if m!=M:
    for i in range(m,M+2):
        x=0
        y=0
        for j in S:
            if j==i: continue
            if j<i:
                x+=i-j
            else:
                y+=j-i
        if x+y<ans or ans==0:
            ans=x+y
            p={i}
        if x+y==ans: p.add(i)
print(*p,ans)
