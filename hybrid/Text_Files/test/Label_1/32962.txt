import java.util.*;

public class Main {
  static final int MOD = 1000000007;

  static int[] switches;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    switches = new int[n + 1];
    int[][] bulbs = new int[m][];
    for (int i = 0; i < m; i++) {
      int k = sc.nextInt();
      bulbs[i] = new int[k];
      for (int j = 0; j < k; j++) {
        bulbs[i][j] = sc.nextInt();
      }
    }
    int[] p = new int[m];
    for (int i = 0; i < m; i++) {
      p[i] = sc.nextInt();
    }
    System.out.println(solve(m, bulbs, p));
  }

  static int solve(int m, int[][] bulbs, int[] p) {
    return dfs(m, bulbs, p, 0);
  }

  static int dfs(int m, int[][] bulbs, int[] p, int i) {
    if (i == m) {
      boolean ok = true;
      for (int[] bulb : bulbs) {
        int count = 0;
        for (int s : bulb) {
          count += switches[s];
        }
        ok &= (count % 2 == p[0]);
      }
      return ok ? 1 : 0;
    }
    int ret = (dfs(m, bulbs, p, i + 1) + dfs(m, bulbs, p, i + 1) * 2) % MOD;
    for (int s : bulbs[i]) {
      switches[s]++;
      ret = (ret + dfs(m, bulbs, p, i + 1)) % MOD;
      switches[s]--;
    }
    return ret;
  }
}
