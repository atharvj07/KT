from collections import Counter
n, k, m, a = map(int, input().split())
g = Counter([int(x) for x in input().split()])
vote = [0] * (n + 1)
for x in g:
    vote[x] += g[x]

ans = []
for x in range(1, n + 1):
    if vote[x] + (m - a) >= k:
        ans.append(1)
    elif vote[x] > 0:
        ans.append(2)
    else:
        ans.append(3)

print(*ans)
