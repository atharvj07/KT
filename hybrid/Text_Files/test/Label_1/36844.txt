n, c, d = map(int,input().split())
diamonds = []
coins = []
for _ in range(n):
    b,p,t = input().split()
    b, p = int(b), int(p)
    if t == 'D': diamonds.append((b,p))
    else: coins.append((b,p))
coins.sort(key=lambda x:x[1])
diamonds.sort(key = lambda x:x[1], reverse = True)
ans = 0
if c >= coins[0][1]:
    ans += coins[0][0]
    c -= coins[0][1]
if d >= diamonds[0][1]:
    ans += diamonds[0][0]
    d -= diamonds[0][1]
if c >= coins[1][1] and ans != 0:
    ans += coins[1][0]
print(ans)
