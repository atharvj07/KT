import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = 0;
        int[] b = new int[100000];
        for (int i = 0; i < n; i++) {
            int x = Math.abs(a[i]);
            if (x > 0 && (x & (x - 1)) == 0) {
                b[k++] = a[i];
            }
        }
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(b[i] + " ");
        }
        System.out.println();
    }
}
