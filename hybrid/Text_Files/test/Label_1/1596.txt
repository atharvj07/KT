#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    vector<int> m(5), w(5);
    cin >> m[0] >> m[1] >> m[2] >> m[3] >> m[4];
    cin >> w[0] >> w[1] >> w[2] >> w[3] >> w[4];
    int hs, hu;
    cin >> hs >> hu;
    int score = 0, inc = hs * 100, dec = hu * 50;
    for (int i = 0; i < 5; i++) {
        score += max(0.3 * (i + 1) * 500, ((1 - 1.0 * m[i] / 250) * (i + 1) * 500 - 50 * w[i]));
    }
    cout << score + inc - dec << endl;
    return 0;
}
