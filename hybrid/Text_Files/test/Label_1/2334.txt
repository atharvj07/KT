#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            if (a[i] == 0 || b[i] == 0 || a[(i + 1) % n] == b[i] ||
                a[(i - 1 + n) % n] == b[i]) {
                cout << "YES" << endl;
                return 0;
            }
        }
    }
    cout << "NO" << endl;
    return 0;
}
