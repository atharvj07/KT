#include<iostream>
#include <bits/stdc++.h>

using namespace std;

int main(){
    int n;
    cin >> n;

    int first = 0, second = 0;
    int last_first = 0, last_second = 0;

    string winner = "first";

    while(n--){
        int x;
        cin >> x;

        if(x > 0){
            first += x;
            last_first = x;
        }else{
            second += abs(x);
            last_second = abs(x);
        }

        if(first < second){
            winner = "second";
        }else if(first > second){
            winner = "first";
        }else{
            if(last_first > last_second){
                winner = "first";
            }else{
                winner = "second";
            }
        }
    }

    cout << winner << endl;
    return 0;
}
