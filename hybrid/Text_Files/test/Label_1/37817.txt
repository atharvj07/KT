def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def solve(k, n):
    n_str = str(n)
    # Check if removing all but one digit can yield a non-prime
    for digit in n_str:
        if not is_prime(int(digit)):
            print(1)
            print(digit)
            return
    # Check for two digit non-primes if single digit removal doesn't work
    if k > 1:
        for i in range(k):
            for j in range(i + 1, k):
                if not is_prime(int(n_str[i] + n_str[j])):
                    print(2)
                    print(n_str[i] + n_str[j])
                    return
    # In case all digits are prime, we remove the last digit as guaranteed by problem statement
    print(k-1)
    print(n_str[:-1])

t = int(input().strip())
for _ in range(t):
    k = int(input().strip())
    n = int(input().strip())
    solve(k, n)
