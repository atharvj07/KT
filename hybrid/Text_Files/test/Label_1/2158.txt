#pragma GCC optimize("Ofast","unroll-loops","omit-frame-pointer","inline")
#pragma GCC target("avx2")
#include<bits/stdc++.h>
using namespace std;
typedef long long LL;
const int N=1e5+5,M=1e9+7;
int main(){
	ios::sync_with_stdio(false);cin.tie(nullptr);
	int n;cin>>n;
	vector<int> a(n);
	for(auto& x:a)cin>>x;
	sort(a.begin(),a.end());
	LL ans=0;
	for(auto& x:a)if(x%2==0)ans+=x;
	if(ans==0){
		ans=a[0]%2;
		for(int i=1;i<n;++i){
			if(a[i]%2==1){
				ans+=a[i];break;
			}
		}
	}
	cout<<ans<<endl;
	return 0;
}
