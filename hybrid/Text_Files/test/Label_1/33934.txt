import java.util.*;
import java.lang.*;

class Solution {
    public static void main(String[] args) throws java.lang.Exception {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int maxSideLength = 0;
        for (int i = 0; i < n; i++) {
            int sideLength = (int) Math.pow(2, scanner.nextInt());
            int count = scanner.nextInt();
            maxSideLength = Math.max(maxSideLength, sideLength * count);
        }
        int p = (int) (Math.log(maxSideLength) / Math.log(2));
        System.out.println(p);
    }
}
