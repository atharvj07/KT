#include <bits/stdc++.h>
#define ll long long

using namespace std;

struct node {
    int val, pos;
};

int n, m;
vector<int> p, nxt;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m;
    p.resize(n + 1);
    nxt.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
        nxt[p[i]] = i;
    }
    while (m--) {
        int u, v;
        cin >> u >> v;
        int pu = nxt[u], pv = nxt[v];
        if (pu > pv)
            swap(pu, pv);
        swap(p[pu], p[pv]);
        nxt[p[pv]] = pv;
        nxt[p[pu]] = pu;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (p[i] == n) {
            ans = i - 1;
            break;
        }
    }
    cout << ans << "\n";
}
