import java.util.Scanner;

public class VestDistribution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] soldierSizes = new int[n];
        for (int i = 0; i < n; i++) {
            soldierSizes[i] = scanner.nextInt();
        }

        int[] vestSizes = new int[m];
        for (int i = 0; i < m; i++) {
            vestSizes[i] = scanner.nextInt();
        }

        int k = 0;
        int[] soldierIndex = new int[n];
        int[] vestIndex = new int[m];
        for (int i = 0, j = 0; i < n && j < m; ) {
            if (soldierSizes[i] - x <= vestSizes[j] && vestSizes[j] <= soldierSizes[i] + y) {
                soldierIndex[k] = i;
                vestIndex[k] = j;
                k++;
                i++;
                j++;
            } else if (soldierSizes[i] < vestSizes[j]) {
                i++;
            } else {
                j++;
            }
        }

        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.println((soldierIndex[i] + 1) + " " + (vestIndex[i] + 1));
        }

        scanner.close();
    }
}
