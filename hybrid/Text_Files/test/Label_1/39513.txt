from sys import stdin

n, m = map(int, stdin.readline().split())
sa = list(map(int, stdin.readline().split()))
sb = list(map(int, stdin.readline().split()))
c = 0
l = 0
mod = 1000000007

for i in range(n):
    if not sa[i]:
        l += 1
    elif not sb[i]:
        c += l
        l = 0
print((c * m % mod) * pow(m - 1, mod - 2, mod) % mod)
