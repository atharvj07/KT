import random

# Read input.
t = int(input())
for _ in range(t):
    n = int(input())
    arr = list(map(int, input().split()))
    arr.sort()

    # Check if all heights are the same.
    all_same = True
    for i in range(1, n):
        if arr[i] != arr[i - 1]:
            all_same = False
            break

    # If all heights are the same, output any order.
    if all_same:
        for height in arr:
            print(height, end=' ')
        print()
    # Otherwise, output a random permutation of the array.
    else:
        random.shuffle(arr)
        for height in arr:
            print(height, end=' ')
        print()
