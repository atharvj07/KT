N, M = map(int, input().split())
D = list(map(int, input().split()))
G = [[] for _ in range(N)]
for _ in range(M):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    G[u].append(v)
    G[v].append(u)

odd = [0] * N
for u in range(N):
    for v in G[u]:
        if v > u:
            odd[u] ^= 1
            odd[v] ^= 1

if any(d == 1 and o == 0 for d, o in zip(D, odd)) or any(d == -1 and o == 1 for d, o in zip(D, odd)):
    print(-1)
    exit()

for i in range(N):
    if D[i] != -1 and odd[i] != D[i]:
        print(-1)
        exit()

ans = []
for u in range(N):
    for v in G[u]:
        if v > u and odd[u] == odd[v]:
            ans.append(u + 1)
            ans.append(v + 1)
            odd[u] ^= 1
            odd[v] ^= 1

print(len(ans) // 2)
for i in range(0, len(ans), 2):
    print(ans[i], ans[i + 1])
