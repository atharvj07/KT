import sys
input = sys.stdin.readline
n = int(input())
a = [int(x) for x in input().split()]
s = sorted(list(enumerate(a)), key = lambda x: x[1])
for i in range(n - 1):
    if s[i][1] > s[i + 1][1]:
        if (s[i][0] + 1) % 2 == 0 and (s[i + 1][0] - s[i][0]) % 2 == 0:
            print(s[i][0] + 1, s[i + 1][0] + 1)
            s[i + 1], s[i] = s[i], s[i + 1]
        else:
            print(s[i + 1][0] + 1, s[i][0] + 1)
            s[i + 1], s[i] = s[i], s[i + 1]
