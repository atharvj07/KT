import sys

n = int(input())
a = list(map(int, input().split()))
edges = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)

vis = [False] * (n + 1)
sub_sum = [0] * (n + 1)

def dfs(node, par):
    vis[node] = True
    sub_sum[node] = a[node - 1]
    for u in edges[node]:
        if u == par:
            continue
        dfs(u, node)
        sub_sum[node] += sub_sum[u]

dfs(1, 1)

mx = -sys.maxsize
for i in range(1, n + 1):
    for u in edges[i]:
        if vis[u]:
            mx = max(mx, sub_sum[i] + sub_sum[u])

if mx > 0:
    print(mx)
else:
    print("Impossible")
