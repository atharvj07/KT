import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        String[][] photo = new String[n][m];
        boolean isColor = false;
        for (int r = 0; r < n; r++) {
            for (int c = 0; c < m; c++) {
                photo[r][c] = input.next();
                if (photo[r][c].equals("C") || photo[r][c].equals("M") || photo[r][c].equals("Y")) {
                    isColor = true;
                }
            }
        }
        System.out.println(isColor ? "#Color" : "#Black&White");
    }
}
