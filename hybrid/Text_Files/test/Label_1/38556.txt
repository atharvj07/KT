from fractions import Fraction as fr
mod = int(1e9)+7
n, m = map(int, input().split())
a = [int(i) for i in input().split()]
b = [int(i) for i in input().split()]
p = q = 1
for i in a+b:
    if i==0:
        p*=m
        q*=m-1
    elif i==0 and b[i]<a[i]:
        exit(print(0))
for x in a:
    if x==0:
        p = p*x//fr(m, m-1)
p = p%mod
q = q%mod
r = (fr(p, q))%mod
print(r)
