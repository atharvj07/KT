#include <iostream>
#include <vector>
using namespace std;

typedef long long int ll;

ll p[101][4];
int main() {
    int n, k;
    cin >> n >> k;
    vector<pair<int, int>> v;
    for (int i = 0; i < k; i++) {
        int x, y;
        char op;
        cin >> x >> op >> y;
        if (op == '=') {
            v.push_back({x, y});
            v.push_back({y, x});
        } else if (op == '<') {
            v.push_back({y, x});
        } else if (op == '>') {
            v.push_back({x, y});
        } else if (op == '<=') {
            v.push_back({x, y});
        } else {
            v.push_back({y, x});
        }
    }
    for (int i = 0; i < v.size(); i++) {
        if (v[i].first > v[i].second)
            swap(v[i].first, v[i].second);
    }
    for (int i = 0; i < 101; i++) {
        for (int j = 0; j < 4; j++)
            p[i][j] = 0;
    }
    p[0][0] = 1;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < 3; j++) {
            for (int l = 1; l <= 2; l++) {
                int ok = 1;
                for (int u = 0; u < v.size(); u++) {
                    if (i * 2 - j * 2 + l == v[u].first &&
                        i * 2 - j * 2 + l + 1 == v[u].second)
                        ok = 0;
                }
                if (j + l <= 3 && ok)
                    p[i][j + l] += p[i - 1][j];
            }
        }
    }
    cout << p[n][3] << endl;
    return 0;
}
