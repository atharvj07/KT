n, k, m, t = map(int, input().split())
l = n

for _ in range(t):
    op, x = map(int, input().split())
    if op == 1:
        k = k + (x <= k)
        l += 1
    else:
        if x == k:
            k = min(k + (l - k + 1 == n), l - 1)
        else:
            k = max(k - (k <= x), 1)
        l -= 1
    print(l, k)
