n, c, d = map(int, input().split())
fountains = []
for _ in range(n):
    b, p, t = input().split()
    fountains.append((int(b), int(p), t == "C"))

fountains.sort(key=lambda x: x[0], reverse=True)

max_beauty = 0
for i in range(n):
    for j in range(i + 1, n):
        if (fountains[i][2] and fountains[j][2] and c < fountains[i][1] + fountains[j][1]) or (not fountains[i][2] and not fountains[j][2] and d < fountains[i][1] + fountains[j][1]):
            continue
        max_beauty = max(max_beauty, fountains[i][0] + fountains[j][0])

print(max_beauty)
