#include <bits/stdc++.h>
#define endl '\n'
using namespace std;

const int maxn = 1e5;
int n, k;
vector<int> pre[maxn + 1];
vector<int> main_course;
stack<int> ans;
int topo[maxn + 1];

void top_sort(int u) {
  topo[u] = 1;
  for (auto &v : pre[u]) {
    if (!topo[v]) { top_sort(v); }
  }
  ans.push(u);
}

int main() {
  cin >> n >> k;
  for (int i = 0; i < k; i++) {
    int x;
    cin >> x;
    main_course.push_back(x);
  }
  for (int i = 1; i <= n; i++) {
    int cnt;
    cin >> cnt;
    for (int j = 0; j < cnt; j++) {
      int x;
      cin >> x;
      pre[i].push_back(x);
    }
  }
  for (auto &x : main_course) {
    if (!topo[x]) {
      top_sort(x);
    }
  }
  vector<bool> vis(n + 1, 0);
  vector<int> tmp;
  while (!ans.empty()) {
    int u = ans.top();
    ans.pop();
    if (!vis[u]) {
      vis[u] = 1;
      tmp.push_back(u);
    }
  }
  if (tmp.size() != n) {
    cout << "-1" << endl;
    return 0;
  }
  cout << tmp.size() << endl;
  for (auto &u : tmp) { cout << u << endl; }
}
