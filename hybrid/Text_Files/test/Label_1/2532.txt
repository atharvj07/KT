#include <iostream>
using namespace std;
int sum(int n) {
  int s = 0;
  while (n > 0) {
    s += n % 10;
    n /= 10;
  }
  return s;
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (sum(i + 1) < sum(i)) ans++;
    }
    cout << ans << endl;
  }
}
