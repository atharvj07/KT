import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

public class Main {
    List<Integer>[] tree;
    List<Integer> order;

    void dfs(int vertex, int parent) {
        if (degree[vertex] % 2 == 0) {
            order.add(vertex + 1);
            for (int adj : tree[vertex]) {
                if (adj != parent) {
                    dfs(adj, vertex);
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Main main = new Main();
        main.tree = new ArrayList[n];
        main.degree = new int[n];
        main.order = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            main.tree[i] = new ArrayList<>();
        }
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            if (p[i] != 0) {
                main.tree[i].add(p[i] - 1);
                main.tree[p[i] - 1].add(i);
                main.degree[i]++;
                main.degree[p[i] - 1]++;
            }
        }
        boolean isPossible = true;
        for (int i = 0; i < n; i++) {
            if (main.degree[i] % 2 == 1) {
                isPossible = false;
            }
        }
        if (isPossible) {
            main.dfs(0, -1);
            System.out.println("YES");
            for (int vertex : main.order) {
                System.out.println(vertex);
            }
        } else {
            System.out.println("NO");
        }
    }

    int[] degree;
}
