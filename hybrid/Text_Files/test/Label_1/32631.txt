import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] primes = new long[n];
        for (int i = 0; i < n; i++) primes[i] = in.nextInt();
        long k = in.nextInt();
        int[] freq = new int[32];
        while (k > 0) {
            freq[(int)(k % 32)]++;
            k /= 32;
        }
        long ans = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < freq[i]; j++) {
                ans *= primes[i];
            }
        }
        System.out.println(ans);
    }
}
