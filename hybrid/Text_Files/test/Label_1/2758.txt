#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, x;
    cin >> n >> x;
    vector<int> d(n);
    for (auto &dt : d) { cin >> dt; }
    vector<long long> sum_till_month(n, 0);
    sum_till_month[0] = d[0];
    for (int i = 1; i < n; i++) { sum_till_month[i] = sum_till_month[i - 1] + d[i]; }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
        int low = i, high = n - 1, lg = -1;
        while (low <= high) {
            int mid = (low + high) / 2;
            if (sum_till_month[mid] - (i ? sum_till_month[i - 1] : 0) <= x) { lg = mid; }
            if (sum_till_month[mid] - (i ? sum_till_month[i - 1] : 0) <= x) { low = mid + 1; }
            else { high = mid - 1; }
        }
        if (lg != -1) {
            ans = max(sum_till_month[lg] - (i ? sum_till_month[i - 1] : 0) + (i ? d[i] : 0), ans);
        }
    }
    cout << ans << endl;
    return 0;
}
