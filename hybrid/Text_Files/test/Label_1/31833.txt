import java.util.*;
import java.io.*;

class Main {

    static class Edge {
        int a, b, c;

        public Edge(int _a, int _b, int _c) {
            a = _a;
            b = _b;
            c = _c;
        }
    }

    static int N, M;
    static List<Edge> edges;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        edges = new ArrayList<>();
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            edges.add(new Edge(a, b, c));
        }

        int[] dist = new int[N];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[0] = 0;

        boolean updated = true;
        for (int k = 0; updated && k < N; k++) {
            updated = false;
            for (Edge e : edges) {
                if (dist[e.a] < Integer.MAX_VALUE && dist[e.a] + e.c < dist[e.b]) {
                    dist[e.b] = dist[e.a] + e.c;
                    updated = true;
                }
            }
        }

        for (Edge e : edges) {
            if (dist[e.a] < Integer.MAX_VALUE && dist[e.a] + e.c < dist[e.b]) {
                System.out.println("NEGATIVE CYCLE");
                return;
            }
        }

        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                if (dist[i] == Integer.MAX_VALUE || dist[i] + dist[j] == Integer.MAX_VALUE) {
                    System.out.print("INF ");
                } else {
                    System.out.print((dist[i] + dist[j]) + " ");
                }
            }
            System.out.println();
        }
    }
}
