import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        Set<Integer> taps = new HashSet<>();
        Set<Integer> tanks = new HashSet<>();
        Map<Integer, List<Edge>> graph = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            graph.put(i, new ArrayList<>());
        }
        for (int i = 0; i < p; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int d = sc.nextInt();
            graph.get(a).add(new Edge(b, d));
        }
        for (int i = 1; i <= n; i++) {
            if (graph.get(i).size() > 0) {
                List<Edge> edges = graph.get(i);
                if (edges.size() == 1) {
                    if (edges.get(0).to == i) {
                        taps.add(i);
                    } else {
                        tanks.add(i);
                    }
                }
            }
        }
        List<TankTapPair> pairs = new ArrayList<>();
        for (int tank : tanks) {
            Set<Integer> visited = new HashSet<>();
            int maxFlow = Integer.MIN_VALUE;
            int tap = -1;
            dfs(tank, graph, visited, maxFlow, tap, pairs);
        }
        Collections.sort(pairs);
        System.out.println(pairs.size());
        for (TankTapPair pair : pairs) {
            System.out.println(pair.tank + " " + pair.tap + " " + pair.flow);
        }
    }

    private static void dfs(int tank, Map<Integer, List<Edge>> graph, Set<Integer> visited, int maxFlow, int tap, List<TankTapPair> pairs) {
        visited.add(tank);
        int curFlow = maxFlow;
        for (Edge edge : graph.get(tank)) {
            if (!visited.contains(edge.to)) {
                if (edge.to != tank) {
                    if (tap == -1) {
                        tap = edge.to;
                    }
                    if (curFlow != -1) {
                        curFlow = Math.min(curFlow, edge.flow);
                    }
                }
                dfs(edge.to, graph, visited, curFlow, tap, pairs);
            }
        }
        if (tank != tap && tap != -1) {
            pairs.add(new TankTapPair(tank, tap, curFlow));
        }
    }

    private static class Edge {
        int to;
        int flow;

        public Edge(int to, int flow) {
            this.to = to;
            this.flow = flow;
        }
    }

    private static class TankTapPair implements Comparable<TankTapPair> {
        int tank;
        int tap;
        int flow;

        public TankTapPair(int tank, int tap, int flow) {
            this.tank = tank;
            this.tap = tap;
            this.flow = flow;
        }

        @Override
        public int compareTo(TankTapPair o) {
            return tank - o.tank;
        }
    }
}
