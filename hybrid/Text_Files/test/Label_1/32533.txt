import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] dx = new int[n], dy = new int[n];
        for (int i = 0; i < n; i++) {
            dx[i] = sc.nextInt();
            dy[i] = sc.nextInt();
        }
        int[] sx = new int[k], sy = new int[k];
        for (int i = 0; i < k; i++) {
            sx[i] = sc.nextInt();
            sy[i] = sc.nextInt();
        }
        int[][] dist = new int[n][k];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                dist[i][j] = Math.abs(dx[i] - sx[j]) + Math.abs(dy[i] - sy[j]);
            }
        }
        int[] minDist = new int[n];
        for (int i = 0; i < n; i++) {
            minDist[i] = Integer.MAX_VALUE;
            for (int j = 0; j < k; j++) {
                minDist[i] = Math.min(minDist[i], dist[i][j]);
            }
        }
        int maxDist = 0;
        for (int i = 0; i < n; i++) {
            maxDist = Math.max(maxDist, minDist[i]);
        }
        System.out.println(maxDist);
    }
}
