n, m, k, q = map(int, input().split())
treasures = [tuple(map(int, input().split())) for _ in range(k)]
safe_cols = set(map(int, input().split()))

# Group treasures by row and find leftmost and rightmost treasure in each row
row_treasures = {}
for r, c in treasures:
    if r in row_treasures:
        row_treasures[r] = (min(row_treasures[r][0], c), max(row_treasures[r][1], c))
    else:
        row_treasures[r] = (c, c)

# Sort rows by the row number
sorted_rows = sorted(row_treasures.keys())

# Initialize variables
current_col = 1
total_moves = 0
for row in sorted_rows:
    leftmost, rightmost = row_treasures[row]
    if current_col not in safe_cols:
        # Move to the nearest safe column if not already in one
        nearest_safe_col = min(safe_cols, key=lambda x: abs(x - current_col))
        total_moves += abs(nearest_safe_col - current_col)
        current_col = nearest_safe_col
    if leftmost < current_col:
        # If the leftmost treasure is to the left, go there, then to the rightmost
        total_moves += (current_col - leftmost) + (rightmost - leftmost)
    else:
        # If the leftmost treasure is to the right, just go to the rightmost
        total_moves += (rightmost - current_col)
    current_col = rightmost  # Update current column to the rightmost treasure's column
    total_moves += 1  # Move up

print(total_moves)
