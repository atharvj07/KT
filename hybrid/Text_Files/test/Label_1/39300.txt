import sys
input = sys.stdin.readline

n = int(input())
arr = list(map(int, input().split()))
sm = sum(arr)

if sm % 2 == 0:
    avg = sm // 2
    for i in range(0, n**2, 2):
        if arr[i] + arr[i+1] == avg:
            arr[i], arr[i+1] = avg, arr[i+1]
    arr = [arr[i:i+n] for i in range(0, n**2, n)]
    print("YES")
    for i in arr:
        print(*i)
else:
    print("NO")
