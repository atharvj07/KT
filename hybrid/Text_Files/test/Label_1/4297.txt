#include <cstdio>
#include <vector>

using namespace std;

int main() {
  int n, k;
  scanf("%d%d", &n, &k);
  
  vector<bool> used(n+1, false);
  vector<int> skills(n);
  vector<int> ans(n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &skills[i]);
  }

  int first_team = 1, second_team = 2;

  while (true) {
    bool found = false;
    for (int i = 0; i < n; ++i) {
      if (!used[skills[i]]) {
        int low = max(0, skills[i]-k);
        int high = min(n-1, skills[i]+k);
        bool possible = true;
        for (int j = low; j <= high; ++j) {
          possible &= !used[skills[j]];
        }
        if (possible) {
          found = true;
          for (int j = low; j <= high; ++j) {
            ans[skills[j]-1] = first_team;
            used[skills[j]] = true;
          }
          swap(first_team, second_team);
          break;
        }
      }
    }
    if (!found) {
      break;
    }
  }

  for (int i = 0; i < n; ++i) {
    printf("%d", ans[i]);
  }
  printf("\n");

  return 0;
}
