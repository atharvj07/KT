import java.util.*;

class Main {
    static int n, m, v;
    static ArrayList<Integer>[] g;
    static boolean[] vis;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        v = sc.nextInt();

        if (2 * (n - 1) <= m) { // Check if it is possible to connect the servers in the required way
            System.out.println(-1);
            return;
        }

        g = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) { // Connect the servers in the required way
            g[i].add(i + 1);
            g[i + 1].add(i);
        }
        g[v].clear(); // Remove the connections from the server that fails

        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        vis = new boolean[n + 1];
        vis[1] = true;
        int cnt = 0;

        while (!q.isEmpty()) { // Count the number of connected components
            int u = q.poll();
            cnt++;

            for (int adj : g[u]) {
                if (!vis[adj]) {
                    vis[adj] = true;
                    q.add(adj);
                }
            }
        }

        if (cnt >= 3) { // Check if there are at least three connected components
            for (int i = 1; i <= n; i++) {
                for (int adj : g[i]) {
                    System.out.println(i + " " + adj); // Output the connections
                }
            }
        } else {
            System.out.println(-1); // If there are less than three connected components, it is impossible
        }
    }
}
