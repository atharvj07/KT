import sys

n = int(input())
s = input()

freq = {}
for c in s:
    freq[c] = freq.get(c, 0) + 1

if len(freq) == n:
    print(0)
    sys.exit()

ans = -1
for c in freq:
    if freq[c] > 1:
        if ans == -1:
            ans = freq[c] - 1
        else:
            ans = min(ans, freq[c] - 1)

if ans != -1:
    print(ans)
else:
    print(-1)
