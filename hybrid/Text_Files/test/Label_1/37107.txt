MOD = 1000000007

def power(x, y, p):
    res = 1  
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, f):
    return (power(f, n, MOD) - power(f-1, n, MOD)) % MOD

q = int(input())
for _ in range(q):
    n, f = map(int, input().split())
    print(solve(n, f))
