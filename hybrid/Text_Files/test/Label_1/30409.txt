import java.util.Scanner;

public class GoodSequence {

    private static final long MOD = 998244353L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        long[][] dp = new long[n + 1][n + 1];
        long[] prefix = new long[n + 1];
        for (int i = 1; i < n; i++) {
            prefix[i + 1] = prefix[i] + (arr[i] > 0 ? 1 : 0);
        }
        dp[n][0] = 1;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < n; j++) {
                if (dp[i + 1][j] == 0) {
                    dp[i][j] = 0;
                    continue;
                }
                if (arr[i] == 0 || arr[i] == n - i - 1) {
                    dp[i][j] = (dp[i + 1][j] + dp[i + 1][j + 1]) % MOD;
                } else if (arr[i] < n - i - 1) {
                    dp[i][j] = (dp[i + 1][j + 1] + dp[i + 1][j + 1] * prefix[i] % MOD) % MOD;
                } else {
                    dp[i][j] = 0;
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
