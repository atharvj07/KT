import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String s = sc.next();
    char[] c = s.toCharArray();
    Arrays.sort(c);
    String sorted = new String(c);
    if (!sorted.equals(s)) {
      System.out.println("NO");
    } else {
      System.out.println("YES");
      for (int i = 0; i < n; i++)
        System.out.print("0");
      System.out.println();
    }
  }
}
