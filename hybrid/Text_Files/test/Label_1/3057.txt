#define _USE_MATH_DEFINES
#include <iostream>
#include <vector>
#include <queue>
#include <cmath>
using namespace std;
int main()
{
    int n, m;
    cin >> n >> m;
    vector<vector<int>> adj(n + 1, vector<int>());
    for (int i = 0; i < m; i++)
    {
        int a, b;
        cin >> a >> b;
        if (a > b)
            swap(a, b);
        adj[a].push_back(b);
    }
    vector<int> dist(n + 1, -1);
    queue<int> q;
    q.push(1);
    dist[1] = 0;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        for (auto v : adj[u])
        {
            if (dist[v] == -1)
            {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
    for (int i = 1; i <= n; i++)
    {
        if (dist[i] > 2)
        {
            cout << -1 << endl;
            return 0;
        }
    }
    vector<pair<int, int>> edges;
    for (int i = 1; i <= n; i++)
    {
        for (auto j : adj[i])
        {
            if (dist[j] == 2 && dist[i] == 0)
            {
                edges.push_back({i, j});
            }
        }
    }
    cout << edges.size() << endl;
    for (auto e : edges)
    {
        cout << e.first << " " << e.second << endl;
    }
}
