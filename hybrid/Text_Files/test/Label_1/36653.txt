import sys
from decimal import Decimal, getcontext

getcontext().prec = 500

def abs_(x): return Decimal(abs(x))
def sqrt_(x): return Decimal(x).sqrt()
def sign_(x): return x / abs_(x)

a, b, c = map(int, input().split())
x1, y1, x2, y2 = map(int, input().split())

if a == 0 or b == 0:
    print(abs_(x1 - x2) + abs_(y1 - y2))
else:
    dist = lambda x1, y1, x2, y2: sqrt_((x1 - x2) ** 2 + (y1 - y2) ** 2)
    if a * x1 + b * y1 + c > 0 and a * x2 + b * y2 + c > 0:
        print(dist(x1, y1, x2, y2))
    elif a * x1 + b * y1 + c > 0:
        print(min(dist(x1, y1, x2, y2),
                    dist(x1, y1, (-c / a, -c / b)),
                    dist(-c / a, -c / b, x2, y2)))
    elif a * x2 + b * y2 + c > 0:
        print(min(dist(x1, y1, x2, y2),
                    dist(x1, y1, (-c / a, -c / b)),
                    dist(-c / a, -c / b, x2, y2)))
    else:
        print(dist(x1, y1, (-c / a, -c / b)) + dist((-c / a, -c / b), x2, y2))
