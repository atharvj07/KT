import java.util.Scanner;

public class ZebraPhoto {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] photo = new int[n];

        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        if (isZebraPhoto(photo)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static boolean isZebraPhoto(int[] photo) {
        int stripeWidth = -1;
        int currentColor = photo[0];
        int stripeCount = 0;

        for (int color : photo) {
            if (color != currentColor) {
                if (stripeWidth == -1) {
                    stripeWidth = currentColor == 0 ? 1 : 2;
                } else if (stripeWidth % 2 != 0 && color == currentColor) {
                    return false;
                } else {
                    stripeCount++;
                }
                currentColor = color;
            }
        }

        return stripeCount > 0 && stripeWidth % 2 == 0;
    }
}
