import java.util.Scanner;

public class MaxOnes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();

            long result = findMaxOnes(l, r);
            System.out.println(result);
        }

        scanner.close();
    }

    private static long findMaxOnes(long l, long r) {
        long maxOnes = 0;
        long result = 0;

        for (long i = l; i <= r; i++) {
            long currentOnes = countOnes(i);
            if (currentOnes > maxOnes) {
                maxOnes = currentOnes;
                result = i;
            } else if (currentOnes == maxOnes) {
                // If there are multiple numbers with the same number of ones, find the smallest one.
                result = Math.min(result, i);
            }
        }

        return result;
    }

    private static long countOnes(long x) {
        long count = 0;
        while (x > 0) {
            if ((x & 1) == 1) {
                count++;
            }
            x >>= 1;
        }
        return count;
    }
}
