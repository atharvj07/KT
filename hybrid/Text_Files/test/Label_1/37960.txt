h,w=map(int,input().split());c=list(map(int,input().split()))
r=list(map(int,input().split()));m=10**9+7
for i in range(w):
    if c[i]>h:print(0);exit()
for i in range(1,w):c[i]+=c[i-1]
for i in range(h):
    if r[i]<=0:continue
    if i:r[i]+=r[i-1]
    t=(c[w-r[i]-1] if r[i]<w else c[w-1])
    t-=c[w-r[i]-1-max(0,r[i]-1)]if r[i]>0 else 0
    r[i]=t%m
print(r[h-1]%m)
