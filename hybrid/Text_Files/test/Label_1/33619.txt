import java.util.*;
import java.lang.*;

class Main {
  static double eps = 1e-9;

  static double cross(double x1, double y1, double x2, double y2) {
    return x1 * y2 - x2 * y1;
  }

  static double area(double[] x, double[] y, int n) {
    double ans = 0.0;
    for (int i = 0; i < n; i++) {
      ans += cross(x[i], y[i], x[(i + 1) % n], y[(i + 1) % n]);
    }
    ans = Math.abs(ans);
    return ans / 2.0;
  }

  static boolean convex(double[] x, double[] y, int n) {
    for (int i = 0; i < n; i++) {
      if (cross(x[i], y[i], x[(i + 1) % n], y[(i + 1) % n]) > eps) {
        return false;
      }
    }
    return true;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    double[] x = new double[n];
    double[] y = new double[n];

    for (int i = 0; i < n; i++) {
      x[i] = sc.nextInt();
      y[i] = sc.nextInt();
    }

    double l = 0.0;
    double r = 1.0;
    for (int i = 0; i < 100; i++) {
      double m = (l + r) / 2.0;
      double[] nx = new double[n];
      double[] ny = new double[n];

      for (int j = 0; j < n; j++) {
        nx[j] = x[j] + m * (x[(j + 1) % n] - x[j]);
        ny[j] = y[j] + m * (y[(j + 1) % n] - y[j]);
      }

      if (convex(nx, ny, n) && area(nx, ny, n) >= area(x, y, n) - eps) {
        l = m;
      } else {
        r = m;
      }
    }

    System.out.println(l);
  }
}
