#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxN = 1e9 + 7;

ll X[100005], Y[100005], A[100005], B[100005];
ll n, m, c, h, ans;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m;
    cin >> c;
    for (ll i = 1; i <= c; i++) {
        cin >> X[i] >> Y[i];
    }
    cin >> h;
    for (ll i = 1; i <= h; i++) {
        cin >> A[i] >> B[i];
    }
    ans = maxN;
    for (ll i = 1; i <= h; i++) {
        for (ll j = 1; j <= c; j++) {
            ans = min(ans, abs(X[j] - A[i]) + abs(Y[j] - B[i]));
        }
    }
    cout << ans << '\n';
    for (ll i = 1; i <= h; i++) {
        for (ll j = 1; j <= c; j++) {
            if (ans == abs(X[j] - A[i]) + abs(Y[j] - B[i])) {
                cout << i << '\n';
                return 0;
            }
        }
    }
}
