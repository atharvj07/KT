#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
    ll n,y,k;
    cin>>y>>k>>n;
    set<ll>s;
    for(ll i=k;i<=n;i+=k)
    {
        if((i-y)>=1 && (i-y)%k==0)
        {
            s.insert(i-y);
        }
    }
    if(s.empty())
        cout<<-1;
    else
    {
        for(auto u:s)
            cout<<u<<" ";
    }
    return 0;
}
