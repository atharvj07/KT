#include <cstdio>
int main() {
  int r, c, n, k, ans = 0;
  scanf("%d%d%d%d", &r, &c, &n, &k);
  bool vio[11][11] = {};
  for (int i = 0; i < n; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    vio[x][y] = 1;
  }
  for (int i = 0; i < (1 << r); ++i) {
    for (int j = 0; j < (1 << c); ++j) {
      int cnt = 0;
      for (int x = 0; x < r; ++x) {
        for (int y = 0; y < c; ++y) {
          if ((i >> x) & 1) continue;
          if ((j >> y) & 1) continue;
          cnt += vio[x + 1][y + 1];
        }
      }
      if (cnt >= k) ++ans;
    }
  }
  printf("%d\n", ans);
  return 0;
}
