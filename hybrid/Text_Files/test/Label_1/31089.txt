import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int n, m;
    static int[] xs, ys;
    static int[][] dist;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        xs = new int[m + 2];
        ys = new int[m + 2];

        dist = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) for (int j = 1; j <= n; j++) dist[i][j] = -1;

        int sx = sc.nextInt();
        int sy = sc.nextInt();
        int fx = sc.nextInt();
        int fy = sc.nextInt();

        for (int i = 1; i <= m; i++) {
            xs[i] = sc.nextInt();
            ys[i] = sc.nextInt();
        }

        xs[m + 1] = sx;
        ys[m + 1] = sy;

        Queue<Integer> q = new LinkedList<>();
        q.add(sx);
        q.add(sy);
        dist[sx][sy] = 0;

        while (!q.isEmpty()) {
            int x = q.poll();
            int y = q.poll();

            if (x == fx && y == fy) {
                System.out.println(dist[x][y]);
                return;
            }

            for (int i = 1; i <= m + 1; i++) {
                if (x == xs[i] || y == ys[i]) {
                    if (dist[xs[i]][ys[i]] == -1) {
                        dist[xs[i]][ys[i]] = dist[x][y] + 1;
                        q.add(xs[i]);
                        q.add(ys[i]);
                    }
                }
            }

            if (x + 1 <= n && dist[x + 1][y] == -1) {
                dist[x + 1][y] = dist[x][y] + 1;
                q.add(x + 1);
                q.add(y);
            }
            if (x - 1 >= 1 && dist[x - 1][y] == -1) {
                dist[x - 1][y] = dist[x][y] + 1;
                q.add(x - 1);
                q.add(y);
            }
            if (y + 1 <= n && dist[x][y + 1] == -1) {
                dist[x][y + 1] = dist[x][y] + 1;
                q.add(x);
                q.add(y + 1);
            }
            if (y - 1 >= 1 && dist[x][y - 1] == -1) {
                dist[x][y - 1] = dist[x][y] + 1;
                q.add(x);
                q.add(y - 1);
            }
        }

        System.out.println(-1);
    }
}
