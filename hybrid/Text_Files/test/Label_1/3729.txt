#include <bits/stdc++.h>
#include <vector>
using namespace std;
int main(){
    long long n,k;
    cin>>n>>k;
    long long a[n];
    for(long long i=0; i<n; i++){
        cin>>a[i];
    }
    sort(a,a+n);
    if(k>=n-1){
        cout<<"0"; return 0;
    }
    long long ans = a[n-1]-a[0];
    k = k - (n-1);
    if(k>0){
        if(k%2==1){
            if(a[1]-a[0]>a[n-1]-a[n-2]){
                a[0] = a[1];
                ans = a[n-1]-a[0];
            }
            else{
                a[n-1] = a[n-2];
                ans = a[n-1]-a[0];
            }
            k--;
        }
        while(k>0){
            a[0]++;
            a[n-1]--;
            ans = a[n-1]-a[0];
            k = k-2;
        }
    }
    cout<<ans;
}
