#include <bits/stdc++.h>
using namespace std;

const int N = 3e5 + 5;
long long a[N], b[N], pre[N];
vector<int> all[N];
int main() {
  int n, m, q[N];
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> m;
  for (int i = 1; i <= m; i++) cin >> q[i];
  sort(a + 1, a + 1 + n, greater<int>());
  sort(q + 1, q + 1 + m);
  for (int l = 1; l <= m; l++) {
    int r = l;
    while (r <= m && q[l] == q[r]) r++;
    for (int i = l; i < r; i++) all[q[i]].push_back(r - l);
    l = r;
  }
  for (int i = 1; i <= n; i++) b[i] = a[i];
  for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
  long long ans = a[1];
  for (int i = 1; i <= n; i++) {
    int s = all[i].size();
    if (s > 0) sort(all[i].begin(), all[i].end());
    for (int j = 0; j < s; j++) {
      ans = min(ans, pre[i - 1] + pre[n] - pre[n - all[i][j] * i]);
    }
    if (s == 0) ans = min(ans, pre[n] - pre[n - i]);
  }
  for (int i = 1; i <= m; i++) cout << ans + b[q[i]] - a[q[i]] << ' ';
  return 0;
}
