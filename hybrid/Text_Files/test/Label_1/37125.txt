n, k = map(int, input().split())
p = [list(map(int, input().split())) for _ in range(k)]
dp = [[0] * (n + 1) for _ in range(k + 1)]
for i in range(1, k + 1):
    for j in range(1, n + 1):
        dp[i][j] = max(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1] + (p[i - 1][j - 1] == p[i - 2][j - 2]))
print(dp[k][n])
