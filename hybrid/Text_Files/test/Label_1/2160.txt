#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const long long MOD = 998244353;

int main() {
    int n; cin >> n;
    vector<long long> a(2 * n);
    for (int i = 0; i < 2 * n; ++i) { cin >> a[i]; }
    sort(a.begin(), a.end());
    
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += (2 * i + 1) * (a[2 * n - 1 - i] - a[i]);
    }
    
    cout << ans % MOD;
    return 0;
}
