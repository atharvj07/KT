import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        int t = scanner.nextInt();
        int[] q = new int[t];
        for (int i = 0; i < t; i++) {
            q[i] = scanner.nextInt() - 1;
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = p[i];
        }
        for (int i = 0; i < t; i++) {
            x[q[i]] = -p[q[i]];
        }
        boolean valid = true;
        int balance = 0;
        for (int i = 0; i < n; i++) {
            balance += x[i];
            if (balance < 0) {
                valid = false;
                break;
            }
        }
        if (balance != 0) {
            valid = false;
        }
        if (!valid) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(x[i] + " ");
            }
            System.out.println();
        }
    }
}
