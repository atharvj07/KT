n = int(input().strip())
a = list(map(int, input().strip().split()))

def is_real(n, a):
    for i in range(n):
        expected = (a[0] + i) % n
        if (i % 2 == 0 and a[i] != expected) or (i % 2 == 1 and a[i] != (n + expected - i) % n):
            return "No"
    return "Yes"

print(is_real(n, a))
