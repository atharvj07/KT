#include <bits/stdc++.h>

using namespace std;

long long n, m, w[100001], c[100001];

int main() {
  cin >> n >> m;
  for (int i = 0; i < n; i++) { cin >> w[i] >> c[i]; }
  pair<int, int> p[n];
  for (int i = 0; i < n; i++) { p[i] = {c[i], w[i]}; }
  sort(p, p + n);
  reverse(p, p + n);
  int sum = 0;
  for (int i = 0; i < n && m > 0; i++) {
    if (p[i].second <= m) {
      sum += p[i].first;
      m -= p[i].second;
    } else {
      sum += p[i].first * ((long long)m / p[i].second);
      m = 0;
    }
  }
  cout << sum << endl;
}
