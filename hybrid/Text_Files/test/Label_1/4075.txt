#include <bits/stdc++.h>
using namespace std;

int n, a[200001], h;
vector<vector<int>> g;

void bfs() {
    queue<int> q;
    q.push(1);
    int cur = 1;
    while (!q.empty()) {
        int p = q.front();
        q.pop();
        if (a[p] == cur) {
            cur++;
            for (auto i : g[p]) q.push(i);
        }
    }
    h = cur - 1;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        g.resize(n + 1);
        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = 2; i <= n; i++) {
            int p;
            cin >> p;
            g[p].push_back(i);
        }
        bfs();
        cout << h << endl;
        g.clear();
    }
}
