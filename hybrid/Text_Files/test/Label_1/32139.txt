import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();
        long res = 0;
        while (a % 2 == 0 && b % 2 == 0 && c % 2 == 0) {
            res += 1;
            a /= 2;
            b /= 2;
            c /= 2;
        }
        if (a % 2 == 1 || b % 2 == 1 || c % 2 == 1) {
            System.out.println(res);
        } else {
            System.out.println(-1);
        }
    }
}
