#include <vector>
#include <queue>
#include <iostream>

int main() {
  int n, p;
  std::cin >> n >> p;
  std::vector<int> t(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> t[i];
  }
  std::priority_queue<int, std::vector<int>, std::greater<int>> q;
  std::vector<int> ans(n);
  int cur = 0;
  for (int i = 0; i < n; ++i) {
    while (!q.empty() && q.top() <= t[i]) {
      q.pop();
    }
    q.push(t[i] + p);
    ans[i] = q.top();
  }
  for (int i = 0; i < n; ++i) {
    std::cout << ans[i] << " ";
  }
  std::cout << "\n";
  return 0;
}
