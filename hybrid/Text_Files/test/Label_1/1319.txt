#include <iostream>
#include <cstring>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    bool appeared[26] = { false };
    for (int i = 0; i < n; i++) {
        int idx = s[i] - 'a';
        if (idx >= 0 && idx < 26) appeared[idx] = true;
        idx = s[i] - 'A';
        if (idx >= 0 && idx < 26) appeared[idx] = true;
    }
    bool pangram = true;
    for (int i = 0; i < 26; i++) {
        pangram &= appeared[i];
    }
    cout << (pangram ? "YES" : "NO") << endl;
    return 0;
}
