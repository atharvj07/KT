#include <bits/stdc++.h>
using namespace std;
const int MAXN = 300, MAXK = 300;
int n, a, b, k, f, dist[MAXN][MAXN], dp[MAXN][MAXN][MAXK + 1];
int main() {
    ios_base::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);
    cin >> n >> a >> b >> k >> f;
    memset(dist, 0x3f, sizeof dist);
    string s, t;
    map<string, int> mp;
    for (int i = 0; i < n; i++) {
        cin >> s >> t;
        if (!mp.count(s)) mp[s] = mp.size(), dist[mp.size() - 1][mp.size() - 1] = 0;
        if (!mp.count(t)) mp[t] = mp.size(), dist[mp.size() - 1][mp.size() - 1] = 0;
        dist[mp[s]][mp[t]] = dist[mp[t]][mp[s]] = 1;
    }
    for (int i = 0; i < mp.size(); i++)
        for (int j = 0; j < mp.size(); j++)
            for (int k = 0; k < mp.size(); k++)
                if (dist[j][k] > dist[j][i] + dist[i][k]) dist[j][k] = dist[j][i] + dist[i][k];
    memset(dp, 0x3f, sizeof dp);
    dp[0][0][0] = 0;
    for (int i = 0; i < mp.size(); i++)
        for (int j = 0; j <= n; j++)
            for (int p = 0; p <= k; p++) {
                if (dp[i][j][p] == 0x3f3f3f3f) continue;
                int &ret = dp[i][j][p];
                if (p) ret = min(ret, dp[i][j][p - 1]);
                dp[i][j + 1][p] = min(dp[i][j + 1][p], ret + a);
                for (int l = 0; l < mp.size(); l++) {
                    if (l == i) continue;
                    dp[l][j + dist[i][l]][p] = min(dp[l][j + dist[i][l]][p], ret + b);
                    if (dist[i][l] != 1e9) dp[l][j][p + 1] = min(dp[l][j][p + 1], ret + f);
                }
            }
    cout << dp[0][n][k] << "\n";
}
