#include <bits/stdc++.h>

using namespace std;

bool pangram(string s)
{
    bitset<26> b;
    for (char c : s)
        b[tolower(c) - 'a'] = 1;
    return b.count() == 26;
}

int main()
{
    int n;
    cin >> n;
    string s;
    cin >> s;
    cout << (pangram(s) ? "YES" : "NO") << endl;
    return 0;
}
