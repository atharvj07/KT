#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int MN = 3e5;
int N[3];
ll A[3][MN], sm[3];
int main() {
    for (int tc = 0; tc < 3; ++tc) {
        int &n = N[tc];
        cin >> n;
        for (int j = 0; j < n; ++j) cin >> A[tc][j];
        sort(A[tc], A[tc] + n);
    }
    sm[0] = accumulate(A[0], A[0] + N[0], 0LL);
    sm[1] = accumulate(A[1], A[1] + N[1], 0LL);
    sm[2] = accumulate(A[2], A[2] + N[2], 0LL);
    for (int tc = 0; tc < 3; ++tc) {
        int &n = N[tc];
        if (sm[tc] > sm[(tc + 1) % 3] + sm[(tc + 2) % 3]) {
            cout << sm[tc] << '\n';
            return 0;
        }
    }
    cout << sm[0] + sm[1] + sm[2] << '\n';
}
