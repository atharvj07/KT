#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
int main() {
  ll n, k;
  cin >> n >> k;
  vector<ll> a(k);
  for (ll i = 0; i < k; i++) { cin >> a[i]; }
  ll ans = n * (n - 1) / 2;
  for (ll i = 0; i < k - 1; i++) {
    if (a[i] == a[i + 1]) { ans--; }
  }
  cout << ans << endl;
}
