#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
typedef long long ll;
ll maxcandies(ll a, ll b, ll c)
{
    ll sum=a+b+c;
    if(sum%2==0)
    {
        return sum/2;
    }
    else
    {
        sum--;
        return sum/2;
    }
}
int main()
{
    int q;
    cin>>q;
    while(q--)
    {
        ll a,b,c;
        cin>>a>>b>>c;
        cout<<maxcandies(a,b,c)<<endl;
    }
    return 0;
}
