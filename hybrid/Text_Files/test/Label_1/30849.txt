import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }
        scanner.close();
        long result = 0;
        long h = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] < h) {
                result += h - a[i];
            } else {
                h = a[i];
            }
        }
        System.out.println(result);
    }
}
