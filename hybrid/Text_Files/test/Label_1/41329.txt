def lexicographically_max_string(s):
    last_occurrence = {char: i for i, char in enumerate(s)}
    seen = set()
    stack = []
    for i, char in enumerate(s):
        if char not in seen:
            while stack and char > stack[-1] and i < last_occurrence[stack[-1]]:
                seen.remove(stack.pop())
            stack.append(char)
            seen.add(char)
    return ''.join(stack)

t = int(input().strip())
for _ in range(t):
    s = input().strip()
    print(lexicographically_max_string(s))
