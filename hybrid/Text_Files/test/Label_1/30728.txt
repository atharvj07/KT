import java.util.Scanner;

public class ConveyorBelt {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            char[][] grid = new char[n][m];
            for (int i = 0; i < n; i++) {
                String row = scanner.next();
                for (int j = 0; j < m; j++) {
                    grid[i][j] = row.charAt(j);
                }
            }
            System.out.println(solve(grid));
        }
    }

    static int solve(char[][] grid) {
        int n = grid.length;
        int m = grid[0].length;
        boolean[][] visited = new boolean[n][m];
        int changes = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!visited[i][j] && !dfs(grid, visited, i, j, n, m)) {
                    changes++;
                }
            }
        }
        return changes;
    }

    static boolean dfs(char[][] grid, boolean[][] visited, int i, int j, int n, int m) {
        if (i == n - 1 && j == m - 1) {
            return true;
        }
        if (i < 0 || i >= n || j < 0 || j >= m || visited[i][j]) {
            return false;
        }
        visited[i][j] = true;
        if (grid[i][j] == 'R') {
            return dfs(grid, visited, i, j + 1, n, m);
        } else {
            return dfs(grid, visited, i + 1, j, n, m);
        }
    }
}
