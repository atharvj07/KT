#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,l,r;
    cin>>n>>l>>r;
    vector<int>a(n),b(n);
    for(int i=0;i<n;i++)cin>>a[i];
    for(int i=0;i<n;i++)cin>>b[i];
    for(int i=0;i<l-1;i++)if(a[i]!=b[i])return puts("LIE"),0;
    sort(a.begin()+l-1,a.begin()+r);
    sort(b.begin()+l-1,b.begin()+r);
    for(int i=0;i<n;i++)if(a[i]!=b[i])return puts("LIE"),0;
    puts("TRUTH");
    return 0;
}
