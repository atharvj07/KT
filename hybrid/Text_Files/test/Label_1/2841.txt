#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) std::cin >> a[i];
    int ans = 0;
    for (int i = 1; i < n; i++) {
      if (a[i] % 2 == a[i - 1] % 2) ans++;
    }
    if (ans % 2 == 0)
      std::cout << ans / 2 << "\n";
    else
      std::cout << -1 << "\n";
  }
  return 0;
}
