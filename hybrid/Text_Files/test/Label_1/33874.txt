import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt() * 60 + sc.nextInt();
        }
        Arrays.sort(arr);
        int ans = 0, cur = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] >= cur) {
                ans++;
                cur = arr[i] + 1;
            }
        }
        System.out.println(ans);
    }
}
