#include<iostream>
#include<unordered_map>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    unordered_map<int, bool> mp;

    for(int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        if(a < b) {
            if(mp[b] == 1) {
                cout << "YES" << endl;
                return 0;
            }
            mp[a] = 1;
        } else {
            if(mp[a] == 1) {
                cout << "YES" << endl;
                return 0;
            }
            mp[b] = 1;
        }
    }

    cout << "NO" << endl;
    return 0;
}
