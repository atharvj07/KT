import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n + 2];
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextInt();
        }
        boolean[] changed = new boolean[n + 2];
        int cnt = 0;
        while (true) {
            cnt++;
            Arrays.fill(changed, false);
            for (int i = 2; i <= n - 1; i++) {
                int[] b = {a[i - 1], a[i], a[i + 1]};
                Arrays.sort(b);
                if (a[i] != b[1]) {
                    a[i] = b[1];
                    changed[i] = true;
                }
            }
            boolean allZero = true;
            for (int i = 1; i <= n; i++) {
                if (a[i] != 0) {
                    allZero = false;
                }
            }
            if (allZero) {
                System.out.println(cnt);
                for (int i = 1; i <= n; i++) {
                    System.out.print(0 + " ");
                }
                System.out.println();
                return;
            }
            boolean ok = true;
            for (int i = 1; i <= n; i++) {
                if (changed[i]) {
                    ok = false;
                }
            }
            if (ok) {
                System.out.println(cnt);
                for (int i = 1; i <= n; i++) {
                    System.out.print(a[i] + " ");
                }
                System.out.println();
                return;
            }
        }
    }
}
