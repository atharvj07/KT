import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long n = sc.nextLong();
            int k = sc.nextInt();
            if (n % 2 != k % 2 || n < k) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                long a = n / k;
                long b = n % k;
                for (int j = 0; j < b; j++) {
                    System.out.print(a + 1 + " ");
                }
                for (int j = 0; j < k - b; j++) {
                    System.out.print(a + " ");
                }
                System.out.println();
            }
        }
        sc.close();
    }
}
