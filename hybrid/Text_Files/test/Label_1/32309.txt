import java.util.*;
import java.math.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    BigInteger l = sc.nextBigInteger();
    BigInteger r = sc.nextBigInteger();
    int n = r.toString().length();
    BigInteger[] pow10 = new BigInteger[n + 1];
    pow10[0] = BigInteger.ONE;
    for (int i = 1; i <= n; i++) {
      pow10[i] = pow10[i - 1].multiply(BigInteger.TEN);
    }
    BigInteger ans = BigInteger.ZERO;
    for (int i = 1; i < n; i++) {
      BigInteger left = pow10[i - 1].max(l);
      BigInteger right = pow10[i].subtract(BigInteger.ONE);
      ans = ans.add(right.subtract(left).add(BigInteger.ONE));
    }
    if (l.compareTo(BigInteger.ZERO) == 0) {
      ans = ans.add(BigInteger.ONE);
    }
    System.out.println(ans);
  }
}
