#include <bits/stdc++.h>

#define all(v) v.begin(), v.end()

using namespace std;

int inv;
vector<pair<int, int>> st;
vector<int> a;

void upd(int pos, int val) {
    if (pos < 0 || pos >= a.size()) return;
    st[pos + a.size()] = {val, pos};
    for (pos += a.size() / 2; pos > 0; pos /= 2)
        st[pos] = min(st[2 * pos], st[2 * pos + 1]);
}

pair<int, int> get_min(int l, int r) {
    pair<int, int> ans = {1e9, -1};
    for (l += a.size(), r += a.size(); l < r; l /= 2, r /= 2) {
        if (l % 2) ans = min(ans, st[l++]);
        if (r % 2) ans = min(ans, st[--r]);
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    a.resize(n);
    st.resize(2 * n);
    for (auto &x : a) cin >> x;
    int m;
    cin >> m;
    for (int l, r, i = 0; i < m; i++) {
        cin >> l >> r;
        l--, r--;
        if ((l + r) % 2 == 0) inv += (get_min(l, r).second > get_min(r + 1, n).second);
        else inv -= (get_min(l, r).second < get_min(r + 1, n).second);
        reverse(all(a));
        for (int j = l; j <= r; j++) {
            upd(j, a[j]);
            a[j] = a[r - (j - l)];
        }
        cout << (inv % 2 ? "odd" : "even") << '\n';
    }
}
