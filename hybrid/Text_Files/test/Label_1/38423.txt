import sys
input = sys.stdin.readline
n = int(input())
d = {}
for _ in range(n-1):
    x, y = map(int, input().split())
    if not x in d: d[x]=[x]
    if not y in d: d[y]=[y]
    d[x].append(y)
    d[y].append(x)
num = list(range(1,n+1))
for x in d:
    for y in d[x]:
        if not y in d[x]:
            d[x].append(y)
            d[y].append(x)
            num[d[x].index(x)],num[d[x].index(y)]=num[d[x].index(y)],num[d[x].index(x)]
print(*num)
