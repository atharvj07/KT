for _ in range(int(input())):
    n, m = map(int, input().split())
    rows = [list(map(int, input().split())) for _ in range(n)]
    cols = [list(map(int, input().split())) for _ in range(m)]
    for i in range(n):
        for j in range(m):
            if rows[i][j] in cols[j]:
                print(*rows[i])
                break
