n, m, k = map(int, input().split())
a = list(map(int, input().split()))
p = {}
for i in range(k):
    p[a[i]]=i+1
ans = 0
for _ in range(n):
    b = list(map(int, input().split()))
    for x in b:
        ans+= p[x]
        p[x]=p[a[-1]]
        a.pop()
        p[a[-1]]=x
        a.append(x)
print(ans)
