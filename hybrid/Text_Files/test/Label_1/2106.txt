#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m, q;
    cin >> n >> m >> q;
    vector<vector<int>> matrix(n, vector<int>(m, 0));
    for (int i = 0; i < q; i++) {
        int t;
        cin >> t;
        if (t == 1) {
            int r;
            cin >> r;
            r--;
            for (int j = 0; j < m; j++) {
                int tmp = matrix[r][j];
                matrix[r][j] = matrix[r][(j + 1) % m];
                matrix[r][(j + 1) % m] = tmp;
            }
        } else if (t == 2) {
            int c;
            cin >> c;
            c--;
            for (int j = 0; j < n; j++) {
                int tmp = matrix[j][c];
                matrix[j][c] = matrix[(j + 1) % n][c];
                matrix[(j + 1) % n][c] = tmp;
            }
        } else {
            int r, c, x;
            cin >> r >> c >> x;
            r--;
            c--;
            matrix[r][c] = x;
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cout << matrix[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
