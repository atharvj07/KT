import sys
import heapq

class Node:
    def __init__(self):
        self.u = []
    def add(self, v, w):
        self.u.append((v, w))

n = int(input())
g = [Node() for i in range(n)]
dist = [[sys.maxsize] * n for i in range(n)]
for i in range(n-1):
    a, b, c = map(int, input().split())
    a -= 1
    b -= 1
    g[a].add(b, c)
    g[b].add(a, c)
    dist[a][b] = dist[b][a] = c
x = list(map(int, input().split()))

for i in range(n):
    dist[i][i] = 0

for k in range(n):
    for i in range(n):
        for j in range(n):
            dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])

for i in range(n):
    for j in range(n):
        if i == j: continue
        dist[i][j] -= x[i] * x[j]

ans = 0
for i in range(n):
    ans = max(ans, min(dist[i]))

print(ans)
