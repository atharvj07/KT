k, a, b, v = map(int, input().split())

sections_per_box = k - 1  # Maximum additional sections per box due to divisors
total_sections = 0
boxes = 0

while a > 0:
    used_divisors = min(b, sections_per_box)
    current_sections = used_divisors + 1
    b -= used_divisors
    nuts_in_current_box = current_sections * v
    a -= nuts_in_current_box
    boxes += 1

print(boxes)
