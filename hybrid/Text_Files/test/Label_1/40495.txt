n = int(input().strip())
segments = [list(map(int, input().split())) + [i + 1] for i in range(n)]  # Reading and storing segments with indices

segments.sort(key=lambda x: (x[0], -x[1]))  # Sorting based on left end, then right end in descending

index_i, index_j = -1, -1  # Initializing default values for i and j
max_right = -1  # To keep track of the maximum right end seen so far

for left, right, index in segments:
    if right <= max_right:  # If current segment's right end is within the max right end seen so far
        index_i = index  # Found a segment that lies within another
        break
    max_right = max(max_right, right)  # Update max right end

if index_i != -1:
    # To find the index_j, go through the segments again to find the one that contains segment at index_i
    for left, right, index in segments:
        if segments[index_i - 1][0] >= left and segments[index_i - 1][1] <= right:
            index_j = index
            break

print(index_i, index_j)
