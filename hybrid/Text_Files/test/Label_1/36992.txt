import sys
input = sys.stdin.readline
n=int(input())
a=list(map(int, input().split()))
b=[0]*n
c=[0]*n
if a[0]>0: b[0]=1
for i in range(1, n):
    if a[i]*a[i-1]>0: b[i]=b[i-1]+1
    else: b[i]=1
if a[n-1]<0: c[n-1]=1
for i in range(n-2, -1, -1):
    if a[i]*a[i+1]>0: c[i]=c[i+1]+1
    else: c[i]=1
ans=0
for i in range(n):
    if b[i]==c[i]: ans+=b[i]
print(ans)
