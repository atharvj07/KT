#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int64_t> p(n);
  for (int i = 0; i < n; ++i) { std::cin >> p[i]; }
  int64_t q = 0;
  for (int i = 0; i < n; ++i) {
    int64_t tmp = p[i];
    for (int j = 0; j < n; ++j) { tmp ^= j % (i + 1); }
    q ^= tmp;
  }
  std::cout << q << '\n';
}
