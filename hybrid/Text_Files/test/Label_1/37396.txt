n = int(input())
s = list(input())
a0 = s.count('0')
a1 = s.count('1')
a2 = s.count('2')
for i in range(n):
    if s[i] == '2':
        if a1 > a0:
            s[i] = '0'
            a0 += 1
            a2 -= 1
        else:
            s[i] = '1'
            a1 += 1
            a2 -= 1
print(''.join(s))
