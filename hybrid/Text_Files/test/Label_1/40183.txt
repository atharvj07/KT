def calculate_points(p, t):
    return max(3*p/10, p - p/250*t)

a, b, c, d = map(int, input().split())

misha_points = calculate_points(a, c)
vasya_points = calculate_points(b, d)

if misha_points > vasya_points:
    print("Misha")
elif vasya_points > misha_points:
    print("Vasya")
else:
    print("Tie")
