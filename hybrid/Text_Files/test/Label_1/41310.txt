import sys

for _ in range(int(input())):
    n, k = map(int, input().split())
    s = list(str(n))
    if len({c for c in s}) <= k:
        print(n)
        continue
    i = len(s) - 1
    while i >= 0 and s[i] == '9':
        i -= 1
    if i == -1:
        print(''.join(['1'] * (k + 1)))
    else:
        s[i] = '9'
        i += 1
        for j in range(i, len(s)):
            s[j] = str(int(s[j]) - 1)
        while '0' in s:
            s.remove('0')
        print(''.join(s))
