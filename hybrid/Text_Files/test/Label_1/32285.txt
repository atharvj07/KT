import java.util.*;
import java.io.*;

public class Chessboard {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        // Create a set to store the vertical spells
        Set<Integer> verticalSpells = new TreeSet<>();

        // Read and store the vertical spells
        for (int i = 0; i < n; i++) {
            verticalSpells.add(Integer.parseInt(br.readLine()));
        }

        // Create a list to store the horizontal spells
        List<Spell> horizontalSpells = new ArrayList<>();

        // Read and store the horizontal spells
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int x1 = Integer.parseInt(st.nextToken());
            int x2 = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            horizontalSpells.add(new Spell(x1, x2, y));
        }

        // Sort the horizontal spells by y-coordinate
        horizontalSpells.sort(Comparator.comparingInt(Spell::getY));

        // Find the minimum number of spells to remove
        int minSpellsToRemove = 0;

        // Iterate over the horizontal spells
        for (Spell spell : horizontalSpells) {
            // Check if the rook can reach the spell's y-coordinate without removing any spells
            if (canReach(verticalSpells, spell.getY())) {
                continue;
            }

            // Remove the spell
            verticalSpells.removeAll(spell.getBlockedColumns());
            minSpellsToRemove++;

            // Check if the rook can now reach the spell's y-coordinate
            if (canReach(verticalSpells, spell.getY())) {
                break;
            }
        }

        // Print the minimum number of spells to remove
        System.out.println(minSpellsToRemove);
    }

    // Check if the rook can reach the given y-coordinate without removing any spells
    private static boolean canReach(Set<Integer> verticalSpells, int y) {
        // The rook can reach the y-coordinate if there are no vertical spells blocking its path
        return !verticalSpells.contains(y);
    }

    // Represents a horizontal spell
    private static class Spell {
        private final int x1;
        private final int x2;
        private final int y;

        public Spell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }

        public int getY() {
            return y;
        }

        public Set<Integer> getBlockedColumns() {
            Set<Integer> blockedColumns = new TreeSet<>();
            for (int i = x1; i <= x2; i++) {
                blockedColumns.add(i);
            }
            return blockedColumns;
        }
    }
}
