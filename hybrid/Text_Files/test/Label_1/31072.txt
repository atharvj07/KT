import java.util.*;
import java.io.*;

public class Main {
    static int MOD = 1000000007;
    static int n, m, t;
    static List<Integer> tanks;
    static long[][] dp;

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        t = sc.nextInt();
        tanks = new ArrayList<>();
        for (int i = 0; i < t; i++) {
            tanks.add(sc.nextInt());
        }
        dp = new long[t + 1][2 * n * m + 1];

        long ans = solve(0, 0);
        if (ans < 0) ans += MOD;
        System.out.println(ans);
    }

    static long solve(int idx, int rem) {
        if (idx == t) {
            if (rem == 0) return 1;
            return 0;
        }
        if (dp[idx][rem] != 0) return dp[idx][rem];
        long res = solve(idx + 1, rem);
        if (rem + tanks.get(idx) <= 2 * n * m) res = (res + solve(idx + 1, rem + tanks.get(idx))) % MOD;
        dp[idx][rem] = res;
        return dp[idx][rem];
    }
}
