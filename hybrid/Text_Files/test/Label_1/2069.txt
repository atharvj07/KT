#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      a[i]--;
    }
    vector<int> b = a;
    sort(b.begin(), b.end());
    if (a == b) {
      cout << 0 << endl;
    } else if (a[0] == 0) {
      int p = -1;
      for (int i = 1; i < n; i++) {
        if (a[i] > a[i - 1]) {
          p = i;
          break;
        }
      }
      if (p == -1) {
        cout << -1 << endl;
      } else {
        cout << 2 << endl;
        cout << p << " " << n - p << endl;
      }
    } else {
      cout << -1 << endl;
    }
  }
  return 0;
}
