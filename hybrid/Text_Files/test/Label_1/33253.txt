import java.util.Scanner;

public class CoinPayment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            int p = scanner.nextInt();
            if (p == 0) {
                break;
            }
            int n1 = scanner.nextInt();
            int n5 = scanner.nextInt();
            int n10 = scanner.nextInt();
            int n50 = scanner.nextInt();
            int n100 = scanner.nextInt();
            int n500 = scanner.nextInt();

            int totalCoins = 0;
            int remainingAmount = p;

            totalCoins += Math.min(n500, remainingAmount / 500);
            remainingAmount %= 500;
            totalCoins += Math.min(n100, remainingAmount / 100);
            remainingAmount %= 100;
            totalCoins += Math.min(n50, remainingAmount / 50);
            remainingAmount %= 50;
            totalCoins += Math.min(n10, remainingAmount / 10);
            remainingAmount %= 10;
            totalCoins += Math.min(n5, remainingAmount / 5);
            remainingAmount %= 5;
            totalCoins += Math.min(n1, remainingAmount / 1);

            System.out.println(totalCoins);
        }
    }
}
