#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) { cin >> x; }
    sort(a.begin(), a.end());
    // if 1 and n are adjacent, we only need 1 step
    cout << (a[1] == a[0] + 1 ? 1 : 2) << '\n';
  }
  return 0;
}
