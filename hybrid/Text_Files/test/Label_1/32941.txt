import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class TreeGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 1; i < n; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            graph.get(a).add(b);
            graph.get(b).add(a);
        }

        long[] result = new long[1];
        dfs(graph, 1, -1, result);
        System.out.println(result[0]);
    }

    private static long dfs(List<List<Integer>> graph, int node, int parent, long[] result) {
        long subtreeSize = 1;
        long maxChildSize = 0;
        for (int child : graph.get(node)) {
            if (child != parent) {
                long childSize = dfs(graph, child, node, result);
                subtreeSize += childSize;
                maxChildSize = Math.max(maxChildSize, childSize);
            }
        }
        result[0] = Math.max(result[0], subtreeSize * (graph.size() - subtreeSize));
        return Math.max(maxChildSize, subtreeSize);
    }
}
