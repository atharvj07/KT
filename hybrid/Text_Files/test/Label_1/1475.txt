#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int INF = (1 << 30);

int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int t;
  cin >> t;
  unordered_map<char, int> m;
  while (t--) {
    string s1, s2;
    cin >> s1 >> s2;
    for (int i = 0; i < 26; i++) m[s1[i]] = i;
    int ans = 0;
    int pos = 0;
    for (char ch : s2) ans += abs(pos - m[ch]), pos = m[ch];
    cout << ans << '\n';
  }
  return 0;
}
