#include <iostream>
#include <vector>

const int MX = 105;

int n, a, x, b, y;
std::vector<int> pos[MX];

int main() {
  std::cin >> n >> a >> x >> b >> y;
  pos[a].push_back(0);
  pos[b].push_back(0);
  for (int i = 1; i <= n; i++) {
    if (i <= x) { pos[a].push_back(i); }
    if (i >= b) { pos[b].push_back(i); }
  }
  bool ans = false;
  for (const auto &p : pos[a]) {
    for (const auto &q : pos[b]) {
      ans |= p == q;
    }
  }
  std::cout << (ans ? "YES" : "NO") << std::endl;
}
