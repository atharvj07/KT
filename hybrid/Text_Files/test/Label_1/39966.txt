import sys
for q in range(int(input())):
    n = int(input())
    s = [input() for i in range(n)]
    ans = 0
    for i in range(n):
        fl = 0
        for j in range(len(s[i])//2):
            if s[i][j]!=s[i][len(s[i])-1-j]:
                fl = 1
        if fl==0:
            ans+=1
    print(ans)
