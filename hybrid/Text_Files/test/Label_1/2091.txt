#include<iostream>
#include<algorithm>
using namespace std;
int main()
{
    string a,b,s;
    int i,k,n,m,j;
    cin>>a>>b;
    n=a.length();
    m=b.length();
    j=0;
    for(i=0; i<n; i++)
    {
        if(j<m and a[i]==b[j])
        {
            s+=a[i];
            j++;
        }
    }
    for(i=0; i<n; i++)
    {
        if(j<m and a[i]!=b[j])
        {
            s+=a[i];
            j++;
        }
    }
    k=0;
    for(i=0; i<n; i++)
    {
        if(k<m and s[i]==b[k])
        {
            k++;
        }
    }
    if(k!=m)
    {
        s=b;
    }
    cout<<s<<endl;
    return 0;
}
