import java.util.*;
public class Main {
    static int n, m;
    static int[] a, b;
    static ArrayList<Integer>[] adj;
    static int[][] ans;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n+1];
        b = new int[n+1];
        for (int i = 1; i <= n; i++) a[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) b[i] = sc.nextInt();
        adj = new ArrayList[n+1];
        for (int i = 1; i <= n; i++) adj[i] = new ArrayList();
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            adj[x].add(y);
            adj[y].add(x);
        }
        boolean flag = false;
        ans = new int[n+1][n+1];
        for (int i = 1; i <= n; i++) {
            if (a[i] != b[i]) {
                if (b[i] - a[i] > 1 || b[i] - a[i] < -1) {
                    flag = true;
                    break;
                }
                else {
                    int cnt = 0;
                    for (int j : adj[i]) {
                        if (a[j] + b[i] - a[i] > b[j]) cnt++;
                    }
                    if (cnt == 0) {
                        flag = true;
                        break;
                    }
                }
            }
        }
        if (flag) System.out.println("NO");
        else {
            System.out.println("YES");
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    ans[i][j] += b[i] - a[i];
                    if (i == j) ans[i][j] += a[i];
                }
            }
            for (int i = 1; i <= n; i++) {
                for (int j : adj[i]) {
                    if (b[j] - a[j] > 0) ans[i][j] -= b[j] - a[j];
                    else ans[j][i] += b[j] - a[j];
                }
            }
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    System.out.print(ans[i][j] + " ");
                }
                System.out.println();
            }
        }
    }
}
