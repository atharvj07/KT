n,k=map(int,input().split())
a=[]
for i in range (1,n+1):
    for j in range (1,5):
        if (i+j<=k) or (i+j>=n+1) or ((i+j)%(k+1)==0) or (i+j==1): continue
        a.append([i,i+j])
f=[[-1]*(k+1) for i in range(n+1)]
mod=1000000007
def dfs(u,cnt):
    if u==1: return 1
    if f[u][cnt]!=-1: return f[u][cnt]
    res=0
    for [v,w] in a:
        if v==u and w==1: continue
        if cnt+1<=k: res+=dfs(w,cnt+1)
        else: res+=dfs(w,cnt)
    res%=mod
    f[u][cnt]=res
    return res
print(dfs(1,1))
