def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(k, arr):
    MOD = 10**9 + 7
    n = 1
    for a in arr:
        n = (n * a) % (MOD - 1)  # Fermat's little theorem for power calculation under modulo
    p = pow(2, n - 1, MOD)
    q = pow(2, n, MOD)
    g = gcd(p, q)
    return f"{p // g}/{q // g}"

k = int(input())
arr = list(map(int, input().split()))
print(solve(k, arr))
