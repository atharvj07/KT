#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll a[1003];
int main() {
  int n, k;
  cin >> n >> k;
  ll sum = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    sum += a[i];
  }
  sort(a, a + n);
  ll ans = 0, rem = sum, tt = 0;
  int p = 0;
  while (rem) {
    rem -= a[p];
    tt++;
    if (tt * k >= sum - rem) {
      ans++;
    }
    p++;
  }
  cout << ans << endl;
  return 0;
}
