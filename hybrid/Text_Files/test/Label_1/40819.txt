import sys

input = sys.stdin.readline

n = int(input())

adj = [[] for _ in range(n + 1)]

for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

vis = [False] * (n + 1)
ans = []

def dfs(x):
    vis[x] = True
    ans.append(x)
    for y in adj[x]:
        if not vis[y]:
            dfs(y)

dfs(1)
print(*ans)
