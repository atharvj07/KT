import java.io.*;
import java.util.*;

public class SonyaMultiset {

    private static class IntegerWithCount {
        int value;
        int count;

        public IntegerWithCount(int value, int count) {
            this.value = value;
            this.count = count;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter writer = new PrintWriter(System.out);

        int t = Integer.parseInt(reader.readLine());
        List<IntegerWithCount> multiset = new ArrayList<>();

        for (int i = 0; i < t; i++) {
            String[] query = reader.readLine().split(" ");
            char operation = query[0].charAt(0);

            switch (operation) {
                case '+':
                    int addValue = Integer.parseInt(query[1]);
                    add(multiset, addValue);
                    break;
                case '-':
                    int removeValue = Integer.parseInt(query[1]);
                    remove(multiset, removeValue);
                    break;
                case '?':
                    String pattern = query[1];
                    int count = countMatchingIntegers(multiset, pattern);
                    writer.println(count);
                    break;
            }
        }

        writer.flush();
    }

    private static void add(List<IntegerWithCount> multiset, int value) {
        for (IntegerWithCount item : multiset) {
            if (item.value == value) {
                item.count++;
                return;
            }
        }
        multiset.add(new IntegerWithCount(value, 1));
    }

    private static void remove(List<IntegerWithCount> multiset, int value) {
        for (IntegerWithCount item : multiset) {
            if (item.value == value) {
                item.count--;
                if (item.count == 0) {
                    multiset.remove(item);
                }
                return;
            }
        }
    }

    private static int countMatchingIntegers(List<IntegerWithCount> multiset, String pattern) {
        int count = 0;
        for (IntegerWithCount item : multiset) {
            if (matchesPattern(item.value, pattern)) {
                count += item.count;
            }
        }
        return count;
    }

    private static boolean matchesPattern(int value, String pattern) {
        String binaryValue = Integer.toBinaryString(value);
        int lengthDifference = pattern.length() - binaryValue.length();
        if (lengthDifference > 0) {
            binaryValue = "0".repeat(lengthDifference) + binaryValue;
        }

        for (int i = 0; i < pattern.length(); i++) {
            if (pattern.charAt(i) == '0' && binaryValue.charAt(i) % 2 == 1) {
                return false;
            }
            if (pattern.charAt(i) == '1' && binaryValue.charAt(i) % 2 == 0) {
                return false;
            }
        }
        return true;
    }
}
