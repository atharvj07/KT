#include <bits/stdc++.h>
using namespace std;
int a[3005], b[3005];
int main(){
    int n, ans = 0;
    cin >> n;
    for(int i = 0; i < n; i++){
        cin >> a[i];
        b[i] = a[i];
    }
    sort(b, b + n);
    for(int i = 0; i < n; i++){
        if(a[i] != b[i]){
            for(int j = i + 1; j < n; j++){
                if(a[j] == b[i]){
                    swap(a[i], a[j]);
                    printf("%d %d\n", i, j);
                    ans++;
                }
            }
        }
    }
    printf("%d\n", ans);
}
