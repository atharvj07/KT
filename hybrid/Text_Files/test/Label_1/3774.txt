#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'
ll n, m, ans;
map<ll, ll> a, b;
int main() {
    cin >> n;
    vector<pair<ll, ll>> chem, top;
    for(int i = 0; i < n; i++) {
        ll x, y; cin >> x >> y; chem.push_back({y, x});
    }
    cin >> m;
    for(int i = 0; i < m; i++) {
        ll x, y; cin >> x >> y; top.push_back({y, x});
    }
    sort(chem.begin(), chem.end(), greater<pair<ll, ll>>());
    sort(top.begin(), top.end(), greater<pair<ll, ll>>());
    for(auto i: chem) a[i.second] = i.first;
    for(auto i: top) b[i.second] = i.first;
    for(auto i: a) {
        if(!b.count(i.first)) ans += i.second;
    }
    for(auto i: b) {
        if(!a.count(i.first)) ans += i.second;
    }
    cout << ans << endl;
    return 0;
}
