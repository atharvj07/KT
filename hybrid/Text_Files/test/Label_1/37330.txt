import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    seg = []
    for i in range(n):
        l, r = map(int, input().split())
        seg.append([l, r])

    seg.sort()
    ans = 1
    mn = seg[0][1]
    for l, r in seg[1:]:
        if l <= mn:
            mn = min(mn, r)
        else:
            ans += 1
            mn = r
    print(ans)
