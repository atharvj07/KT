import java.util.*;
import java.io.*;

public class Main {
    static int[] x, y;
    static final long UN = (long)1e18;
    static int bound = 2200;

    static class Interval {
        int l, r;

        Interval() {
            this(0, 0);
        }

        Interval(int a, int b) {
            l = a;
            r = b;
        }
    }

    static ArrayList<Interval>[] intervals = new ArrayList[bound * 2 + 10];

    static long eval(int index, long a) {
        return a * a + x[index] * a + y[index];
    }

    static long intersect(int index, int a, int b) {
        if (eval(index, a) > eval(index, b))
            return UN;
        long delta = b - a;
        return eval(index, a) + (2 * x[index] + delta) * delta;
    }

    static void add(int index, int a, int b) {
        intervals[index + bound].add(new Interval(a, b));
    }

    static boolean check(int index) {
        ArrayList<Interval> list = intervals[index + bound];
        if (list.isEmpty())
            return true;
        int a = list.get(0).l, b = list.get(0).r;
        for (int i = 1; i < list.size(); i++) {
            if (intersect(index, b, list.get(i).l) < b)
                return false;
            a = Math.min(a, list.get(i).l);
            b = Math.max(b, list.get(i).r);
        }
        return true;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        x = new int[n + 20];
        y = new int[n + 20];
        for (int i = 0; i < n; i++) {
            st = new StringTokenizer(in.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            x[i] = a;
            y[i] = b;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (x[i] == x[j])
                    continue;
                int a = x[i], b = x[j], c = (y[i] * b - y[j] * a) / (b - a);
                if (c >= -bound && c <= bound) {
                    add(c, a, b);
                    if (check(c))
                        ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
