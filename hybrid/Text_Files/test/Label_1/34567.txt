class Main {
    public static void main(String[] args) throws java.lang.Exception {
        java.util.Scanner sc= new java.util.Scanner(System.in);
        int n = sc.nextInt(),m =sc.nextInt(), c=0;
        char [][]cross= new char[n][m];
        for(int i=0;i<n;i++){
            cross[i]=sc.next().toCharArray();
        }
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                char cc=cross[i][j];
                for(int k=j+1;k<m;k++){
                    if(cc==cross[i][k]){
                        c++;
                        cross[i][k]='0';
                    }
                }
                for(int k=i+1;k<n;k++){
                    if(cc==cross[k][j]){
                        c++;
                        cross[k][j]='0';
                    }
                }
            }
        }
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(cross[i][j]!='0')
                    System.out.print(cross[i][j]);
            }
        }
    }
}
