#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    char c;
    cin >> n >> c;
    string s;
    cin >> s;
    vector<int> ans;
    for (int i = n / 2; i >= 1; i--) {
      if (s[i - 1] != c) {
        ans.push_back(i);
        for (int j = i + i; j <= n; j += i) {
          s[j - 1] = c;
        }
      }
    }
    cout << ans.size() << "\n";
    for (int i : ans) cout << i << " ";
    cout << "\n";
  }
  return 0;
}
