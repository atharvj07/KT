#include <bits/stdc++.h>
#define ll long long
#define vi vector<int>
#define pb push_back
using namespace std;
const int maxn = 2e5 + 5;

int n, m, a[maxn], p[maxn], s[maxn];
ll c[maxn];

int main() {
    int t;
    cin >> t;
    for (int cs = 1; cs <= t; cs++) {
        cin >> n;
        for (int i = 0; i < n; i++) cin >> a[i];
        cin >> m;
        for (int i = 0; i < m; i++) {
            cin >> p[i] >> s[i];
            c[p[i]] = s[i];
        }
        priority_queue<ll> q;
        sort(a, a + n);
        reverse(a, a + n);
        int day = 0;
        bool flag = 0;
        for (int i = 0; i < n; i++) {
            int l = 0, r = m - 1, pos = -1;
            while (l <= r) {
                int mid = (l + r) / 2;
                if (p[mid] >= a[i]) {
                    pos = mid;
                    r = mid - 1;
                } else
                    l = mid + 1;
            }
            if (pos != -1) {
                if (c[p[pos]] > 0) {
                    c[p[pos]]--;
                    day++;
                    continue;
                }
            }
            flag = 1;
            break;
        }
        if (flag)
            cout << "-1" << endl;
        else
            cout << day << endl;
    }
}
