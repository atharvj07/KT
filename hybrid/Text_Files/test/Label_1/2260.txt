#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  string events;
  cin >> n >> events;
  string rooms(10, '0');
  for (int i = 0; i < n; i++) {
    char event = events[i];
    if (event == 'L') {
      for (int j = 0; j < 10; j++) {
        if (rooms[j] == '0') {
          rooms[j] = '1';
          break;
        }
      }
    } else if (event == 'R') {
      for (int j = 9; j >= 0; j--) {
        if (rooms[j] == '0') {
          rooms[j] = '1';
          break;
        }
      }
    } else {
      rooms[event - '0'] = '0';
    }
  }
  cout << rooms << endl;

  return 0;
}
