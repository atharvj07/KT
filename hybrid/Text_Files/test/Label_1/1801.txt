#include<bits/stdc++.h>
using namespace std;

const int N = 3e5 + 5, M = 3e5 + 5;
typedef long long ll;

struct edge {
    int u, v, w, id, di;
}E[M];

int n, m, k, p[N], dis[N], ans[M];

bool comp(edge a, edge b) {
    return a.di < b.di;
}

int findp(int x) {
    if (x == p[x]) return x;
    return p[x] = findp(p[x]);
}

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        E[i] = {u, v, w, i, 0};
    }
    iota(p, p + N, 0);
    int e = 0;
    for (int i = 1; i <= n; i++) cin >> dis[i];
    for (int i = 1; i <= m; i++) {
        int x = findp(E[i].u), y = findp(E[i].v);
        if (x == y) continue;
        E[i].di = dis[E[i].u] + E[i].w + dis[E[i].v];
        p[x] = p[y];
        ++e;
    }
    sort(E + 1, E + 1 + m, comp);
    e = min(e, k);
    cout << e << endl;
    for (int i = 1; i <= e; i++) cout << E[i].id << ' ';
}
