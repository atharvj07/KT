#include <iostream>
const int MAXN = 1 << 19;
char out[MAXN];
int main() {
  long long int n, m;
  std::cin >> n >> m;
  if (n < m || (n - m) % 3) {
    std::cout << "-1\n";
    return 0;
  }
  for (int i = 0; i < n; i++) {
    out[i] = '0';
  }
  for (int i = 0; i < m; i++) {
    out[i] = '1';
  }
  int t = (n - m) / 3;
  int p = 0;
  while (t--) {
    out[p] = out[p + 1] = out[p + 2] = '1';
    p += 3;
  }
  std::cout << out << '\n';
  return 0;
}
