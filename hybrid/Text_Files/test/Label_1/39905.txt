import sys
input=sys.stdin.readline

for _ in range(int(input())):
    n=int(input())
    a=[0]*n
    for i in range(n):
        a[i]=[int(j) for j in input().split()]
    b,c=0,0
    for i in range(1,5):
        d,e=0,0
        for j in range(n):
            d+=a[j][i-1]==0
            e+=a[j][i-1]==1
        if d>n/2 or e>n/2: continue
        if e==n/2: b=i
        if d==n/2: c=i
    if b>0 and c>0: print('YES')
    else: print('NO')
