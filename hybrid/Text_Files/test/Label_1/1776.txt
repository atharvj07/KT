#include <cstdio>
#include <vector>
#include <algorithm>
int main() {
  char s[1001];
  scanf("%s", s);
  int i, j, ans = 1, n = 0, ord[1001] = {0}, h[1001] = {0};
  for (i = 0; i < 1001; i++) ord[i] = 500;
  for (i = 0; (s[i]); i++) {
    j = s[i] - 'A';
    if (ord[j] == 500) { ord[j] = n; h[n++] = i; }
    else h[ord[j]] = i;
  }
  for (i = 0; i < n; i++) ans = std::max(ans, h[i] - ord[i] + 1);
  printf("%d\n", ans);
  return 0;
}
