#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> p(n), s(n), c(k);
  for (int &x : p) cin >> x;
  for (int &x : s) cin >> x;
  for (int &x : c) cin >> x;
  sort(p.begin(), p.end());
  sort(c.begin(), c.end());
  for (int &x : s)
    x = lower_bound(p.begin(), p.end(), x) - p.begin() + 1;
  for (int &x : c)
    x = lower_bound(p.begin(), p.end(), x) - p.begin() + 1;
  vector<int> cnt(m + 1, 0);
  for (int x : s) { cnt[x]++; }
  sort(cnt.begin(), cnt.end(), greater<int>());
  int ans = 0;
  for (int x : cnt) {
    if (!x) { ans++; } else { k--; }
    if (!k) break;
  }
  cout << ans << endl;
  return 0;
}
