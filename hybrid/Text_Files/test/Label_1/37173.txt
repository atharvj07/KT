from collections import deque

def bfs(start, grid, n, m):
    visited = [[False] * m for _ in range(n)]
    q = deque([start])
    visited[start[0]][start[1]] = True
    distance = [[0] * m for _ in range(n)]
    while q:
        x, y = q.popleft()
        for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
            nx, ny = x + dx, y + dy
            if 0 <= nx < n and 0 <= ny < m and not visited[nx][ny] and grid[nx][ny] != '#':
                visited[nx][ny] = True
                distance[nx][ny] = distance[x][y] + 1
                q.append((nx, ny))
    return visited, distance

def min_road_to_connect_states(n, m, grid):
    state_positions = {i: [] for i in '123'}
    for i in range(n):
        for j in range(m):
            if grid[i][j] in state_positions:
                state_positions[grid[i][j]].append((i, j))
    
    min_distance = float('inf')
    for i in '123':
        for start in state_positions[i]:
            visited, distance = bfs(start, grid, n, m)
            total_distance = 0
            for j in '123':
                if j != i:
                    state_distance = min(distance[x][y] for x, y in state_positions[j] if visited[x][y])
                    total_distance += state_distance - 1
            min_distance = min(min_distance, total_distance)
    
    return -1 if min_distance == float('inf') else min_distance

n, m = map(int, input().split())
grid = [input() for _ in range(n)]
print(min_road_to_connect_states(n, m, grid))
