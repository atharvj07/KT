import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long c = sc.nextLong();
        String as = Long.toString(a, 3);
        String cs = Long.toString(c, 3);
        int n = Math.max(as.length(), cs.length());
        as = String.format("%" + n + "s", as).replace(' ', '0');
        cs = String.format("%" + n + "s", cs).replace(' ', '0');
        char[] ans = new char[n];
        for (int i = 0; i < n; i++) {
            ans[i] = tor(as.charAt(i), cs.charAt(i));
        }
        String anss = new String(ans);
        anss = anss.replaceFirst("^0+(?!$)", "");
        if (anss.equals("")) anss = "0";
        System.out.println(Long.parseLong(anss, 3));
    }

    public static char tor(char a, char b) {
        int aa = Integer.parseInt(String.valueOf(a));
        int bb = Integer.parseInt(String.valueOf(b));
        int ans = (aa + bb) % 3;
        return String.valueOf(ans).charAt(0);
    }
}
