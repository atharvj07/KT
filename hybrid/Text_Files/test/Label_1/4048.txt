#include<bits/stdc++.h>
using namespace std;
map<pair<int,char>,int>mp;
vector<pair<string,pair<int,int>>>t;
int main(){
	string h,a;
	cin>>h>>a;
	int n;
	cin>>n;
	int hr=0,ar=0;
	while(n--){
		int x;
		char c,s;
		int y;
		cin>>x>>c>>y>>s;
		if(s=='y'){
			mp[{y,c}]++;
			if(mp[{y,c}]==2) (c=='h'?hr:ar)++,t.push_back({c=='h'?h:a,{y,x}});
		}
		else {
			if(mp[{y,c}]==0) (c=='h'?hr:ar)++,t.push_back({c=='h'?h:a,{y,x}});
			else if(mp[{y,c}]==1) t.push_back({c=='h'?h:a,{y,x}});
		}
	}
	sort(t.begin(),t.end());
	for(auto x:t) {
		cout<<x.first<<" "<<x.second.first<<" "<<x.second.second<<'\n';
	}
}
