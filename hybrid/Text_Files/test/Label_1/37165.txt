import sys
input=sys.stdin.readline
for t in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    b = sorted(range(n), key=lambda x: a[x])
    print(min(n-1, max(a[b[i]]-a[b[i-1]] if i>0 else a[b[i]] for i in range(n-1))-1, max(a[b[i-1]]-a[b[i]] if i<n-1 else a[b[i]] for i in range(n-1))-1))
