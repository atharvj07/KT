#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 5;
ll pre[MAXN], suf[MAXN], a[MAXN];
ll solve(int n, int q) {
    for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
    for (int i = n; i >= 1; i--) suf[i] = suf[i + 1] + a[i];
    ll res = 0;
    while (q--) {
        int l, r;
        cin >> l >> r;
        res += max(pre[r] - pre[l - 1], suf[l] - suf[r + 1]);
    }
    return res;
}
int main() {
    int n, q;
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cout << solve(n, q) << endl;
    return 0;
}
