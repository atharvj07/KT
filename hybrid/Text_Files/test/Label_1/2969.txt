#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5, M = 1e5 + 5;
int p[N], s[N], n, m;
signed main() {
  cin >> n >> m;
  for (int i = 1; i <= m; i++) cin >> p[i];
  for (int i = 1; i <= n; i++) {
    if (s[i]) continue;
    int mx = 0, ps;
    for (int j = 1; j <= m; j++) {
      if (p[j] == i) continue;
      if (!s[p[j]] && mx < p[j]) mx = p[j], ps = j;
    }
    if (mx && s[mx]) continue;
    if (mx) s[i] = mx, s[ps] = i;
  }
  long long ans = 0;
  for (int i = 1; i <= m - 1; i++) {
    if (s[p[i]] == s[p[i + 1]])
      continue;
    else
      ans += abs(p[i] - p[i + 1]);
  }
  cout << ans << endl;
  return 0;
}
