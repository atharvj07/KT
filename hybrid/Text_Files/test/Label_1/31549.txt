import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        boolean[] inc = new boolean[200001];
        boolean[] dec = new boolean[200001];
        List<Integer> incList = new ArrayList<>();
        List<Integer> decList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (inc[a[i]]) {
                System.out.println("NO");
                return;
            }
            if (dec[a[i]]) {
                System.out.println("NO");
                return;
            }
            inc[a[i]] = true;
            dec[a[i]] = true;
            if (i > 0 && a[i] < a[i - 1]) {
                decList.add(a[i]);
            } else if (i > 0 && a[i] > a[i - 1]) {
                incList.add(a[i]);
            }
        }
        if (incList.size() == 0 || decList.size() == 0) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        System.out.println(incList.size());
        for (int x : incList) {
            System.out.print(x + " ");
        }
        System.out.println();
        System.out.println(decList.size());
        for (int x : decList) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
