#include <iostream>
#include <vector>
#include <algorithm>
#include <string>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &i : a) cin >> i;
        vector<string> s(n + 1);
        s[0] = string(max(a[0], 1), 'a');
        for (int i = 0; i < n; i++) {
            s[i + 1] = s[i].substr(0, a[i]) + char('a' + (i + 1) % 26);
        }
        for (string &i : s) cout << i << endl;
    }
}
