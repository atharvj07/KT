#include <bits/stdc++.h>
using namespace std;

using i64 = long long;

int main() {
  i64 n, k;
  cin >> n >> k;
  vector<i64> c(n);
  for (i64 &x : c) cin >> x;
  sort(c.rbegin(), c.rend());
  i64 ans = 0, bonus = 0;
  for (i64 i = 0; i < n; i++) {
    if (bonus + c[i] > 0) {
      bonus += c[i];
      ans += bonus;
    } else {
      if (k == 0) break;
      k--;
      ans += bonus + c[i];
      bonus = 0;
    }
  }
  cout << ans << '\n';
  return 0;
}
