import sys

for _ in range(int(input())):
    n, m, x = map(int, input().split())
    a = sorted(map(int, input().split()))
    if a[-1] > x: print('NO'); continue
    print('YES')
    b = [[]] * m
    for i in a:
        if not b[0]: t = 0
        else: t = max(b[0][-1], b[-1][-1])
        if i + t <= x: b[0].append(i)
        else: b[-1].append(i)
    sys.stdout.write(' '.join(map(str, sum(b, []))))
    print()
