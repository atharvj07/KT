import java.util.*;

public class Main {
    static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[][] dp = new long[n + 1][k + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= k; j++) {
                dp[i][j] = -1;
            }
        }
        solve(n, k, dp);
        System.out.println(dp[n][k]);
    }

    static long solve(int n, int k, long[][] dp) {
        if (dp[n][k] != -1) {
            return dp[n][k];
        }
        if (k == 0) {
            return 0;
        }
        if (n == 1) {
            return k;
        }
        dp[n][k] = 0;
        for (int i = k; i >= 1; i--) {
            dp[n][k] = (dp[n][k] + solve(n - 1, k - i, dp) + k) % MOD;
        }
        return dp[n][k];
    }
}
