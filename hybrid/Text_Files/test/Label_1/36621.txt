from sys import stdin
from heapq import heappush, heappop
input = stdin.readline

for _ in range(int(input())):
    k = int(input())
    G = 0
    B = 0
    adj = [[] for _ in range(2*k+1)]
    for _ in range(2*k-1):
        a, b, w = map(int,input().split())
        adj[a].append((w,b))
        adj[b].append((w,a))
    q = []
    for i in range(1,2*k+1):
        if len(adj[i]) <= 1:
            heappush(q, (0,i))
    while q:
        cur, u = heappop(q)
        for w, v in adj[u]:
            if (u,v) not in adj[v]:
                continue
            adj[v].remove((w,u))
            if len(adj[v]) <= 1:
                heappush(q,(cur+w, v))
                G += cur+w
                B += cur+w
    print(G,B)
