import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        long[] dp = new long[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            b[i] = input.nextInt();
            dp[i] = a[i] + b[i];
        }
        for (int i = 1; i < n; i++) {
            dp[i] = Math.max(dp[i], dp[i - 1] + a[i] + b[i]);
        }
        System.out.println(dp[n - 1]);
    }
}
