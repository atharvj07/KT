import sys
import heapq

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = [a.pop()]
    hq = []
    
    for x in a:
        if x>ans[-1]:
            heapq.heappush(hq, x)
        else:
            ans.append(x) 
            
    while hq:
        ans.append(heapq.heappop(hq))
        
    print(*ans)
