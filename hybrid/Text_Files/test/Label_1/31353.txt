import java.util.LinkedList;
import java.util.Queue;

public class Labyrinth {

    static class Block {
        int x, y;
        int time;

        Block(int x, int y, int time) {
            this.x = x;
            this.y = y;
            this.time = time;
        }
    }

    static int minMinutes(char[][] labyrinth, int xT, int yT, int xM, int yM) {
        int n = labyrinth.length;
        int m = labyrinth[0].length;

        int[][] visited = new int[n][m];

        Queue<Block> queue = new LinkedList<>();
        queue.offer(new Block(xT, yT, 0));
        visited[xT][yT] = 1;

        int[][] dx = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

        while (!queue.isEmpty()) {
            Block current = queue.poll();

            if (current.x == xM && current.y == yM) {
                return current.time;
            }

            for (int[] dir : dx) {
                int nextX = current.x + dir[0];
                int nextY = current.y + dir[1];

                if (nextX >= 0 && nextX < n && nextY >= 0 && nextY < m &&
                        labyrinth[nextX][nextY] != '*' && visited[nextX][nextY] == 0) {

                    if (labyrinth[current.x][current.y] == '|' && dir[0] != 0 ||
                            labyrinth[current.x][current.y] == '-' && dir[1] != 0 ||
                            labyrinth[current.x][current.y] == '+' ||
                            labyrinth[nextX][nextY] == '|' && dir[0] != 0 ||
                            labyrinth[nextX][nextY] == '-' && dir[1] != 0 ||
                            labyrinth[nextX][nextY] == '+') {
                        visited[nextX][nextY] = 1;
                        queue.offer(new Block(nextX, nextY, current.time));
                    } else {
                        visited[nextX][nextY] = 2;
                        queue.offer(new Block(nextX, nextY, current.time + 1));
                    }
                }
            }
        }

        return -1;
    }

    public static void main(String[] args) {
        char[][] labyrinth = {
                {'+', '-', '+', '-', '+'},
                {'-', '+', '-', '+', '-'},
                {'+', '-', '+', '-', '+'},
                {'-', '+', '-', '+', '-'},
                {'+', '-', '+', '-', '+'}
        };

        int xT = 1, yT = 1;
        int xM = 5, yM = 1;

        int minMinutes = minMinutes(labyrinth, xT, yT, xM, yM);

        System.out.println(minMinutes);
    }
}
