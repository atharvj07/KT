#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ld long double
#define pb push_back
#define vi vector<int>
const ll inf = 1 << 30;
const ll mod = 1000000007;

void solve() {
	int n; cin >> n; vi a(n); 
	for(int&x:a) cin >> x;
	bool fl1=1, fl2=1;
	for(int i=0;i<n;i++) {
		if(a[i]!=i+1) fl1=0;
		if(a[i]!=n-i) fl2=0;
	}
	if(fl1) cout << "YES\n";
	else if(fl2) cout << "YES\n";
	else cout << "NO\n";
}

int main() {
	ios_base :: sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	int t; cin >> t; while(t--) solve();
	return 0;
}
