import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int a = scan.nextInt(), b = scan.nextInt(), c = scan.nextInt(), l = scan.nextInt();

        if(a + b <= c || a + c <= b || b + c <= a) {
            System.out.println(0);
            return;
        }

        int ans = 0;
        if(a < b && b < c) {
            ans = (l / 3) + 1;
            if(c - b > l) ans = l;
        } else if(b < a && a < c) {
            ans = (l / 3) + 1;
            if(c - a > l) ans = l;
        } else {
            ans = (l / 3) + 1;
            if(b - a > l) ans = l;
        }
        System.out.println(ans);
    }
}
