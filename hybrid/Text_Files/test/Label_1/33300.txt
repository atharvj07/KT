import java.util.*;

public class MostFrequentSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        Set<String> strings = new HashSet<>();
        for (int i = 0; i < n; i++) {
            strings.add(scanner.nextLine());
        }

        String result = findGoodString(strings);
        System.out.println(result);
    }

    static String findGoodString(Set<String> strings) {
        Trie trie = new Trie();
        for (String str : strings) {
            trie.insert(str);
        }

        String minString = null;
        for (String str : strings) {
            if (isGoodString(str, trie, strings)) {
                if (minString == null || str.compareTo(minString) < 0) {
                    minString = str;
                }
            }
        }

        return minString != null ? minString : "NO";
    }

    static boolean isGoodString(String str, Trie trie, Set<String> strings) {
        for (String subStr : strings) {
            if (!isMostFrequent(subStr, str, trie)) {
                return false;
            }
        }
        return true;
    }

    static boolean isMostFrequent(String subStr, String str, Trie trie) {
        int count = trie.countOccurrences(subStr, str);
        for (String otherSubStr : trie.getAllSubstrings(str)) {
            if (otherSubStr.length() == subStr.length() && trie.countOccurrences(otherSubStr, str) > count) {
                return false;
            }
        }
        return true;
    }

    static class Trie {
        private TrieNode root;

        Trie() {
            root = new TrieNode();
        }

        void insert(String str) {
            TrieNode node = root;
            for (char ch : str.toCharArray()) {
                node = node.children.computeIfAbsent(ch, c -> new TrieNode());
                node.count++;
            }
            node.isWord = true;
        }

        int countOccurrences(String subStr, String str) {
            TrieNode node = root;
            for (char ch : subStr.toCharArray()) {
                node = node.children.get(ch);
                if (node == null) {
                    return 0;
                }
            }
            return node.count - (str.length() - subStr.length()); // Subtract occurrences outside the main string
        }

        List<String> getAllSubstrings(String str) {
            List<String> substrings = new ArrayList<>();
            for (int i = 0; i < str.length(); i++) {
                StringBuilder sb = new StringBuilder();
                for (int j = i; j < str.length(); j++) {
                    sb.append(str.charAt(j));
                    substrings.add(sb.toString());
                }
            }
            return substrings;
        }
    }

    static class TrieNode {
        Map<Character, TrieNode> children = new HashMap<>();
        int count = 0;
        boolean isWord = false;
    }
}
