#include <bits/stdc++.h>
#define int long long
using namespace std;

signed main() {
    int n, l, x, y;
    cin >> n >> l >> x >> y;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    bool b1 = false, b2 = false;
    int ans = 2, p1 = -1, p2 = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] == x) {
            b1 = true;
        }
        if (a[i] > x && a[i] < y) {
            p1 = a[i];
        }
        if (a[i] == y) {
            b2 = true;
        }
    }
    if (b1 && b2) {
        cout << 0 << endl;
    } else if (b1 || b2) {
        cout << 1 << endl;
        if (b1)
            cout << y << endl;
        else
            cout << x << endl;
    } else {
        cout << ans << endl;
        if (p1 != -1) {
            cout << x << " " << y << endl;
        } else {
            cout << x << " " << x + (y - x) / 2 << " " << y << endl;
        }
    }
    return 0;
}
