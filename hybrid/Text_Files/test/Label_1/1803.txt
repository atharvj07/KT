#include <iostream>
using namespace std;
int main() {
    int x, y, z;
    cin >> x >> y >> z;
    if (!z) {
        if (x > y)
            cout << '+' << endl;
        else if (x < y)
            cout << '-' << endl;
        else
            cout << '0' << endl;
    } else
        cout << '?' << endl;
}
