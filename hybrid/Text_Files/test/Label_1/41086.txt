for _ in range(int(input())):
    n = int(input())
    a = [int(i) for i in input().split()]
    s1 = sum(a[0::2])
    s2 = sum(a[1::2])
    if s1 == s2:
        print(n)
        print(*a)
    else:
        if s1 < s2:
            for i in range(1 << n):
                sa1 = 0
                sa2 = 0
                for j in range(n):
                    if (i >> j) & 1:
                        sa1 += a[j]
                    else:
                        sa2 += a[j]
                if sa1 == sa2:
                    k = i.bit_count()
                    print(k)
                    for j in range(n):
                        if (i >> j) & 1:
                            print(a[j], end=' ')
                    print()
                    break
        else:
            for i in range(1 << n):
                sa1 = 0
                sa2 = 0
                for j in range(n):
                    if (i >> j) & 1:
                        sa2 += a[j]
                    else:
                        sa1 += a[j]
                if sa1 == sa2:
                    k = i.bit_count()
                    print(k)
                    for j in range(n):
                        if (i >> j) & 1:
                            print(a[j], end=' ')
                    print()
                    break
