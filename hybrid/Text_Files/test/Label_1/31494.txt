import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int n = scanner.nextInt();
        if (m > n && (m - n) % 2 != 0) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        while (m > 0 || n > 0) {
            if (m >= n) {
                m -= n;
                ans += 1;
            } else {
                n -= m;
                ans += 1;
            }
        }
        System.out.println(ans);
    }
}
