#include <iostream>
#include <unordered_map>
#include <cstring>
using namespace std;
int main()
{
    int n, a, b, k, f;
    cin >> n >> a >> b >> k >> f;
    unordered_map<string, int> freq;
    while (n--)
    {
        string s, t;
        cin >> s >> t;
        freq[s]++, freq[t]++;
    }
    int ans = 0;
    for (auto &p : freq)
        if (p.second > 1)
            ans += min(f, b * p.second);
    cout << ans << '\n';
}
