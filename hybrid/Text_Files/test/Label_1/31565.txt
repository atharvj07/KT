import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String p1 = sc.next();
        String p2 = sc.next();
        int[] f1 = new int[n];
        int[] f2 = new int[n];
        f1[0] = 0;
        f2[0] = 0;
        for (int i = 1; i < n; i++) {
            f1[i] = f1[i - 1];
            f2[i] = f2[i - 1];
            if (p1.charAt(i - 1) == 'N') f1[i]++;
            if (p1.charAt(i - 1) == 'S') f1[i]--;
            if (p1.charAt(i - 1) == 'E') f1[i]++;
            if (p1.charAt(i - 1) == 'W') f1[i]--;
            if (p2.charAt(i - 1) == 'N') f2[i]++;
            if (p2.charAt(i - 1) == 'S') f2[i]--;
            if (p2.charAt(i - 1) == 'E') f2[i]++;
            if (p2.charAt(i - 1) == 'W') f2[i]--;
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (f1[i] != f2[i]) ok = false;
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
