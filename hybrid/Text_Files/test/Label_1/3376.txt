#include <bits/stdc++.h>
#define ll long long int
using namespace std;

const int mod = 1e9 + 7;
vector<ll> F1, F2;
vector<bool> vis;

void pre() {
  F1.assign(1e6 + 1, 0);
  F2.assign(1e6 + 1, 0);
  vis.assign(1e6 + 1, false);
  F1[1] = 1;
  for (int i = 2; i <= 1e6; i++) {
    if (!vis[i]) {
      F1[i] = 1;
      for (int j = i + i; j <= 1e6; j += i) {
        F1[j] *= 1LL * i % mod;
        F1[j] %= mod;
        vis[j] = true;
      }
    }
  }
  for (int i = 1; i <= 1e6; i++) {
    for (int j = i; j <= 1e6; j += i) { F2[j] += F1[i]; }
  }
}

ll fastpow(ll x, ll y) {
  ll ans = 1;
  while (y) {
    if (y & 1) { ans *= x; }
    x *= x;
    y >>= 1;
    x %= mod, ans %= mod;
  }
  return ans;
}

vector<ll> sieve;
void gen_sieve() {
  sieve.assign(1e6 + 1, 0);
  for (ll i = 1; i <= 1e6; i++) {
    for (ll j = i; j <= 1e6; j += i) { sieve[j]++; }
  }
}

ll F(ll n, ll k, bool type) {
  if (n <= 1e6) {
    if (k & 1)
      return type ? F1[n] : F2[n];
    else
      return type ? F2[n] : F1[n];
  }
  if (type) {
    ll ans = 0;
    for (ll i = 1; i * i <= n; i++) {
      if (n % i == 0) {
        ans += F1[i];
        if (i != n / i) ans += F1[n / i];
        ans %= mod;
      }
    }
    return ans;
  } else {
    ll ans = 0;
    for (ll i = 1; i * i <= n; i++) {
      if (n % i == 0) {
        ans += F2[i];
        if (i != n / i) ans += F2[n / i];
        ans %= mod;
      }
    }
    return ans;
  }
}

int main() {
  pre();
  gen_sieve();
  ll n, k;
  cin >> n >> k;
  ll ans;
  if (k == 1)
    ans = F(n, k, 1);
  else {
    ll l = 1;
    while (l * sieve[n] <= k) l *= sieve[n];
    if (l * sieve[n] != k) ans = F(n, l * sieve[n], 1);
    else {
      k /= sieve[n];
      if (k & 1)
        ans = F(n, k, 0);
      else
        ans = F(n, k, 1);
    }
  }
  cout << ans << endl;
  return 0;
}
