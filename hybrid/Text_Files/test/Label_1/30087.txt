import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class HanhsTree {

    private static final int MOD = 998244353;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int q = Integer.parseInt(line[1]);

        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }

        for (int i = 0; i < n - 1; i++) {
            line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            adjList.get(u).add(v);
            adjList.get(v).add(u);
        }

        int[] subtreeSize = new int[n + 1];
        calculateSubtreeSize(adjList, subtreeSize, 1, 0);

        long[] expectedValue = new long[n + 1];
        long[] sum = new long[n + 1];
        for (int i = 0; i < q; i++) {
            line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);

            if (type == 1) {
                int d = Integer.parseInt(line[2]);
                update(adjList, subtreeSize, expectedValue, sum, 1, 0, v, d);
            } else {
                long ans = expectedValue[v];
                long denominator = subtreeSize[v];
                ans %= MOD;
                denominator %= MOD;
                long inv = modInverse(denominator, MOD);
                ans *= inv;
                ans %= MOD;
                System.out.println(ans);
            }
        }
    }

    private static void update(List<List<Integer>> adjList, int[] subtreeSize, long[] expectedValue, long[] sum, int node, int parent, int v, int d) {
        subtreeSize[node]++;
        expectedValue[node] += (d * (subtreeSize[node] - 1L)) % MOD;
        sum[node] += d;
        sum[node] %= MOD;

        for (int child : adjList.get(node)) {
            if (child != parent) {
                update(adjList, subtreeSize, expectedValue, sum, child, node, v, d);
            }
        }

        if (node == v) {
            expectedValue[node] += d;
            sum[node] += d;
            sum[node] %= MOD;
        } else {
            long contribution = ((subtreeSize[node] - 1L) * (subtreeSize[node] - 2L) / 2) % MOD;
            contribution *= d;
            contribution %= MOD;
            expectedValue[node] -= contribution;
            expectedValue[node] %= MOD;
        }
    }

    private static void calculateSubtreeSize(List<List<Integer>> adjList, int[] subtreeSize, int node, int parent) {
        for (int child : adjList.get(node)) {
            if (child != parent) {
                calculateSubtreeSize(adjList, subtreeSize, child, node);
                subtreeSize[node] += subtreeSize[child];
            }
        }
    }

    private static long modInverse(long a, long m) {
        long m0 = m;
        long y = 0, x = 1;

        if (m == 1) {
            return 0;
        }

        while (a > 1) {
            long q = a / m;
            long t = m;
            m = a % m;
            a = t;
            t = y;
            y = x - q * y;
            x = t;
        }

        if (x < 0) {
            x += m0;
        }

        return x;
    }
}
