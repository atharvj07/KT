import sys

def calc_dist(graph, n, police, dist):
    visited = [False] * (n + 1)
    distance = [0] * (n + 1)
    for p in police:
        visited[p] = True

    q = [p for p in police]
    while len(q) > 0:
        u = q.pop(0)
        for v in graph[u]:
            if visited[v] == False:
                q.append(v)
                distance[v] = distance[u] + 1
                visited[v] = True

    count = 0
    for i in range(1, n + 1):
        if distance[i] > dist:
            count += 1
    return count

input = sys.stdin.readlines()
n, k, d = map(int, input[0].split())
police = list(map(int, input[1].split()))
graph = [[] for i in range(n + 1)]
for i in range(2, n + 1):
    u, v = map(int, input[i].split())
    graph[u].append(v)
    graph[v].append(u)

maxi = n - k - 1
res = []
for i in range(2, n + 1):
    u, v = map(int, input[i].split())
    graph[u].remove(v)
    graph[v].remove(u)
    dist = calc_dist(graph, n, police, d)
    if dist > 0:
        maxi -= 1
        res.append(i - 1)
    graph[u].append(v)
    graph[v].append(u)

print(maxi)
for i in range(maxi):
    print(res[i], end=" ")
