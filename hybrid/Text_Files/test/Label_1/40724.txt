import sys

n = int(input())
L = []
for _ in range(n):
    x, y = input().split()
    L.append((x, int(y)))

L.sort(key=lambda x: x[1], reverse=True)

a = 0
b = 0
ans = 0
for x, y in L:
    if x=="11":
        ans += y
        a += 1
        b += 1
        continue
    if x=="10":
        ans += y
        a += 1
        if a >= n//2 and b >= n//2:
            break
        continue
    if x=="01":
        ans += y
        b += 1
        if a >= n//2 and b >= n//2:
            break
        continue
    if x=="00":
        if a < n//2:
            ans += y
            a += 1
            if a >= n//2 and b >= n//2:
                break
        elif b < n//2:
            ans += y
            b += 1
            if a >= n//2 and b >= n//2:
                break
        else:
            break
print(ans)
