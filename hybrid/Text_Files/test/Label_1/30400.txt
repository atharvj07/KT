import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long x = sc.nextInt(), packs = x, distressed = 0;
        while (n-- > 0) {
            char op = sc.next().charAt(0);
            long d = sc.nextInt();
            if (op == '+') packs += d;
            else {
                if (packs >= d) packs -= d;
                else distressed++;
            }
        }
        System.out.println(packs + " " + distressed);
    }
}
