import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(in.readLine());

        long[] a = new long[n];
        long[] b = new long[n];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(st.nextToken());
        }

        long[] sorted = a.clone();
        Arrays.sort(sorted);
        int mid = (n + 1) / 2;
        for (int i = 0; i < n; i++) {
            b[i] = sorted[mid - 1];
            for (int j = i; j < n; j++) {
                if (a[j] == a[i]) {
                    int index = Arrays.binarySearch(sorted, j);
                    if (mid <= index) {
                        b[i] = sorted[mid];
                    } else {
                        b[i] = sorted[mid - 1];
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(b[i]);
        }
    }
}
