#include<bits/stdc++.h>
using namespace std;
string a,s[200005],ans;
int ansl,l,r,n;
int main()
{
    cin>>a;
    n=a.length();
    for(r=n-1,l=n-1;r>=0;r--)
    {
        while(r>=0 && a[r]==a[l]) r--;
        s[++ansl]=a.substr(r+1,l-r);
        l=r;
    }
    for(int i=ansl;i>=1;i--)
    {
        l=s[i].length();
        if(l>10) ans=s[i].substr(0,5)+"..."+s[i].substr(l-2,2);
        else ans=s[i];
        cout<<l<<" "<<ans<<endl;
    }
    return 0;
}
