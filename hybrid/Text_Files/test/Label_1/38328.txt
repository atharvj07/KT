t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    total_sum = sum(a)
    mean = total_sum / n
    target_sum = 2 * mean
    counts = {}
    pairs = 0
    for element in a:
        pairs += counts.get(target_sum - element, 0)
        counts[element] = counts.get(element, 0) + 1
    print(pairs)
