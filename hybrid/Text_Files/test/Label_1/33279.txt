import java.util.*;
import java.io.*;

public class Main {
    static int n, m, k;
    static int[] p;
    static long[][] dp;
    static List<ArrayList<Integer>> adj;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        p = new int[n + 1];
        dp = new long[n + 1][m + 1];
        for (int i = 1; i <= n; i++) p[i] = i;
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) adj.add(new ArrayList<Integer>());
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            p[find(u)] = find(v);
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        dfs(1, 0);
        System.out.println(dp[1][m] % k);
    }
    static int find(int x) {
        if (x == p[x]) return x;
        return p[x] = find(p[x]);
    }
    static void dfs(int u, int par) {
        dp[u][0] = 1;
        for (Integer v : adj.get(u)) {
            if (v == par) continue;
            dfs(v, u);
            for (int i = m - 1; i >= 0; i--) {
                for (int j = 0; j <= i; j++) {
                    dp[u][i] += dp[v][j] * dp[u][i - j] % k;
                    dp[u][i] %= k;
                }
            }
        }
    }
}
