n = int(input())
a = list(map(int, input().split()))
pref = [0] * (n + 1)
for i in range(n):
    pref[i + 1] = pref[i]
    if a[i] == 2:
        pref[i + 1] += 1
dp = [0] * (n + 1)  
for i in range(n - 1, -1, -1):
    dp[i] = dp[i + 1]
    if a[i] == 1:
        dp[i] = max(dp[i], pref[i + 1])
for i in range(n):
    if a[i] == 2:
        dp[i] = max(dp[i], dp[i + 1] + pref[i + 1] - pref[i])
print(max(dp))
