import java.util.*;
import java.io.*;

public class Main {

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int p = in.nextInt();
    int[][] a = new int[n][m];
    int[][] b = new int[n][m];
    for (int i = 0; i < n; i++) {
      String line = in.next();
      for (int j = 0; j < m; j++) {
        if (line.charAt(j) == '1') {
          a[i][j]++;
          b[i][j]++;
        }
      }
    }
    int[] ans = new int[m];
    for (int i = 0; i < m; i++) {
      for (int j = 0; j < n; j++) {
        if (a[j][i] >= (n + 1) / 2) {
          ans[i]++;
        }
      }
    }
    for (int i = 0; i < m; i++) {
      if (ans[i] > 0) {
        System.out.print(1);
      } else {
        System.out.print(0);
      }
    }
    System.out.println();
  }
}
