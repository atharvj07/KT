import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        int m = Integer.parseInt(s[1]);
        int[] emps = new int[n + 1];
        int[][] empLang = new int[n + 1][m + 1];

        for (int i = 1; i <= n; i++) {
            s = br.readLine().split(" ");
            emps[i] = Integer.parseInt(s[0]);
            for (int j = 1; j <= emps[i]; j++)
                empLang[i][Integer.parseInt(s[j])] = 1;
        }

        boolean[] vis = new boolean[n + 1];
        int ans = 0;

        for (int i = 1; i <= n; i++) {
            if (!vis[i]) {
                vis[i] = true;
                Queue<Integer> q = new LinkedList<>();
                q.add(i);

                while (!q.isEmpty()) {
                    int u = q.poll();
                    for (int j = 1; j <= m; j++)
                        if (empLang[u][j] == 1) {
                            for (int k = 1; k <= n; k++)
                                if (empLang[k][j] == 1 && !vis[k]) {
                                    vis[k] = true;
                                    q.add(k);
                                }
                        }
                }
                ans++;
            }
        }

        System.out.println(--ans);
    }
}
