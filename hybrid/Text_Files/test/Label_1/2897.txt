#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  sort(s.begin(), s.end());
  int ans = 0;
  for (int i = 0; i < k; i++) {
    if (i != 0 && s[i] == s[i - 1]) {
      cout << -1 << endl;
      return 0;
    }
    ans += s[i] - 'a' + 1;
  }
  cout << ans << endl;
  return 0;
}
