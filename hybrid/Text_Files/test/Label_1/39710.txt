def get_min_rides_and_count(people, k):
    n = len(people)
    dp = [[0] * 2 for _ in range(k + 1)]
    dp[0][0] = 1
    dp[50][0] = 1
    for i in range(1, k + 1):
        for j in range(1, n + 1):
            if dp[i][0] > 0:
                if i + people[j - 1] <= k:
                    dp[i + people[j - 1]][0] = max(dp[i + people[j - 1]][0], dp[i][0] + 1)
                if i + 2 * people[j - 1] <= k:
                    dp[i + 2 * people[j - 1]][0] = max(dp[i + 2 * people[j - 1]][0], dp[i][0] + 1)
            if dp[i][1] > 0:
                if i + people[j - 1] <= k:
                    dp[i + people[j - 1]][1] = max(dp[i + people[j - 1]][1], dp[i][1] + 1)
    if dp[-1][0] > 0:
        res = dp[-1][0]
        cnt = 0
        for i in range(k + 1):
            if dp[i][0] == res:
                cnt += dp[i][1]
        return res, cnt % 1000000007
    return -1, 0

n, k = map(int, input().split())
people = list(map(int, input().split()))
min_rides, ways = get_min_rides_and_count(people, k)
print(min_rides)
print(ways)
