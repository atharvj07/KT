import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] sum = new int[n];
        sum[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            sum[i] = sum[i + 1] + a[i];
        }
        boolean[] dp = new boolean[sum[0] + 1];
        dp[0] = true;
        for (int i = 0; i < n; i++) {
            for (int j = sum[0]; j >= a[i]; j--) {
                dp[j] |= dp[j - a[i]];
            }
        }
        System.out.println(dp[sum[0]] ? "BitLGM" : "BitAryo");
    }
}
