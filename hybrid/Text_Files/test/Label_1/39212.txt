n=int(input())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
p=[0]*n
for i in range(n):
    if a[i]==b[i]: p[i]=a[i]
    else: break
for i in range(n):
    if a[i]!=b[i]:
        if p[a[i]-1]==0: p[a[i]-1]=b[i]
        else: p[b[i]-1]=a[i]
        break
for i in range(n):
    if p[i]==0:
        for j in range(1,n+1):
            if j not in p:
                p[i]=j
                break
print(*p)
