import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] rooms = new int[n][3];
        for (int i = 0; i < n; i++) {
            rooms[i][0] = scanner.nextInt();
            rooms[i][1] = scanner.nextInt();
            rooms[i][2] = scanner.nextInt();
        }
        int m = scanner.nextInt();
        int[][] wallpapers = new int[m][3];
        for (int i = 0; i < m; i++) {
            wallpapers[i][0] = scanner.nextInt();
            wallpapers[i][1] = scanner.nextInt();
            wallpapers[i][2] = scanner.nextInt();
        }
        int minCost = Integer.MAX_VALUE;
        for (int i = 0; i < m; i++) {
            int cost = 0;
            for (int[] room : rooms) {
                int perimeter = 2 * (room[0] + room[1]);
                int numStrips = (int) Math.ceil((double) perimeter / wallpapers[i][0]);
                cost += numStrips * wallpapers[i][2];
            }
            minCost = Math.min(minCost, cost);
        }
        System.out.println(minCost);
    }
}
