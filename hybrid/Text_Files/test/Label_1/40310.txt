n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]

# Dictionary to keep track of which kids each kid remembers
memories = {}
for i in range(n):
    memories[i + 1] = a[i]

# Start with the first kid and find its remembered kids
p = [1]
next_possible = set(memories[1])

while len(p) < n:
    last = p[-1]
    next_kids = memories[last]
    for kid in next_kids:
        if kid in next_possible:
            p.append(kid)
            next_possible = set(memories[kid]) - {last}
            break

print(' '.join(map(str, p)))
