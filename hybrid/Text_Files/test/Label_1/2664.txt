#include <bits/stdc++.h>
using namespace std;
#define ll long long int
vector<ll> graph[200005];
ll color[200005];
ll check=0;
void dfs(ll node,ll c){
	ll adj=1;
	color[node]=c;
	for(ll i=0;i<graph[node].size();i++){
		if(color[graph[node][i]]==-1){
			if(adj==c) adj=3-c;
			dfs(graph[node][i],adj);
		}else if(color[graph[node][i]]==color[node]){
			check=1;
			return;
		}
	}
}
int main()
{
	ll t,n,l,r,i,j,k,cnt=0;
	cin>>t;
	while(t--){
		check=0;
		for(i=0;i<=200005;i++){
			graph[i].clear();
			color[i]=-1;
		}
		cin>>n;
		for(i=0;i<n;i++){
			cin>>l>>r;
			graph[l].push_back(r);
			graph[r].push_back(l);
		}
		for(i=1;i<=200005;i++){
			if(color[i]==-1){
				dfs(i,1);
			}
		}
		if(check==1) cout<<-1<<"\n";
		else{
			for(i=1;i<=200005;i++){
				if(color[i]==1) cout<<1<<" ";
				else cout<<2<<" ";
			}
			cout<<"\n";
		}
	}
}
