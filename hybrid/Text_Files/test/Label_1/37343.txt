n = int(input())
h = [int(input()) for _ in range(n)]

time = h[0] + 1  # Time to climb the first tree and eat the nut
for i in range(1, n):
    time += 1  # Jump to the next tree
    if h[i] > h[i-1]:
        time += (h[i] - h[i-1])  # Climb up
    else:
        time += (h[i-1] - h[i])  # Climb down
    time += 1  # Eat the nut
print(time)
