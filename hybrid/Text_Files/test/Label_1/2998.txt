#include <cstdio>
#include <cstring>

int main() {
    int t; scanf("%d", &t);
    while(t--) {
        char s[100005], ls; int ans = 0, n; scanf("%s", s);
        n = strlen(s); ls = s[0]; int f = 1;
        for(int i=1; i<n; i++) {
            if(s[i] == ls) {
                ans++;
                ls = s[i];
                if(f) f = 0;
                else f = 1;
            }
            if(f) ls = s[i];
        }
        printf("%d\n", ans);
    }
    return 0;
}
