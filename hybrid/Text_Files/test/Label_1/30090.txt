import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long s = sc.nextLong();
        if (Math.abs(a + b) <= s && s % 2 == Math.abs(a + b) % 2){
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
