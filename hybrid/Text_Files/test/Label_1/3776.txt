#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, h, v;
    cin >> n;
    set<pair<int, int>> s;
    for (int i = 1; i <= n * n; i++) {
        cin >> h >> v;
        if (!s.count({h, v}) && !s.count({v, h})) {
            cout << i << ' ';
            s.insert({h, v});
            s.insert({v, h});
        }
    }
    return 0;
}
