n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]

for i in range(n):
    for j in range(m):
        if a[i][j] != 0:
            print('TAK')
            print(' '.join(map(str, [j+1 for j in range(m) if any(a[k][j] != 0 for k in range(n))])), flush=True)
            exit()

print('NIE')
