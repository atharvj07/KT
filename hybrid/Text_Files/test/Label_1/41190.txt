n = int(input())
a = list(map(int, input().split()))

odd = sum(x % 2 for x in a)
even = n - odd

# dp[i][j][k] - maximum number of bouquets using first i flowers,
# with j mod 3 odd flowers and k mod 3 even flowers
dp = [[[-float('inf')] * 3 for _ in range(3)] for _ in range(2)]
dp[0][0][0] = 0

for i in range(n):
    cur = i % 2
    nxt = 1 - cur
    for j in range(3):
        for k in range(3):
            dp[nxt][j][k] = max(dp[nxt][j][k], dp[cur][j][k])
            if a[i] % 2 == 1:
                dp[nxt][(j + 1) % 3][k] = max(dp[nxt][(j + 1) % 3][k], dp[cur][j][k] + (j == 2))
            else:
                dp[nxt][j][(k + 1) % 3] = max(dp[nxt][j][(k + 1) % 3], dp[cur][j][k] + (k == 2))

    dp[cur] = [[-float('inf')] * 3 for _ in range(3)]

answer = dp[n % 2][0][0]
print(answer)
