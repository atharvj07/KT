import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(); // n liters to be bought
        int a = input.nextInt(); // number of 0.5-liter bottles
        int b = input.nextInt(); // number of 1-liter bottles
        int c = input.nextInt(); // number of 2-liter bottles

        int count = 0;
        for (int i = 0; i <= a; i++) { // i bottles of 0.5-liter
            for (int j = 0; j <= b; j++) { // j bottles of 1-liter
                for (int k = 0; k <= c; k++) { // k bottles of 2-liter
                    if (0.5 * i + 1 * j + 2 * k == n) {
                        count++;
                    }
                }
            }
        }

        System.out.println(count);
    }
}
