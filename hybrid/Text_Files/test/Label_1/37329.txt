n,m=map(int,input().split())
a=list(map(int,input().split()))
b=[0]*n;c=[False]*n
for i in range(1,n):
 b[i]=(a[i]-a[i-1]+m)%m
for i in range(n):
 a[i]=(a[i]+a[0])%m
for i in range(1,n):
 if b[i]%2==0 and c[i]==False:
  c[i]=True;c[n-i]=True
  for j in range(1,n//i+1):
   b[i*j]%=(m*m)
   b[n-i*j]%=(m*m)
prod=1
for i in b:
 prod=(prod*i)%m
print(prod)
