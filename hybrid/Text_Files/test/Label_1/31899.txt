import java.io.*;
import java.util.*;

public class Main {
    static class Node {
        int val;
        long add;
        int min;

        public Node() {}

        public Node(int val, long add, int min) {
            this.val = val;
            this.add = add;
            this.min = min;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        Node[] tree = new Node[(n + 1) * 4];
        buildTree(tree, 1, 1, n);
        boolean valid = true;
        loop: for (int i = 0; i < m; i++) {
            st = new StringTokenizer(in.readLine());
            int type = Integer.parseInt(st.nextToken());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            int x = 0;
            if (type == 1) {
                x = Integer.parseInt(st.nextToken());
            } else {
                int expectedMax = Integer.parseInt(st.nextToken());
                int max = getMax(tree, 1, 1, n, l, r);
                if (max != expectedMax) {
                    valid = false;
                    break loop;
                }
            }
            increment(tree, 1, 1, n, l, r, x);
        }
        if (valid) {
            System.out.println("YES");
            StringBuilder sb = new StringBuilder();
            for (int i = 1; i <= n; i++) {
                sb.append(get(tree, 1, 1, n, i)).append(' ');
            }
            System.out.println(sb.toString());
        } else {
            System.out.println("NO");
        }
    }

    private static void buildTree(Node[] tree, int node, int start, int end) {
        if (start == end) {
            tree[node] = new Node(0, 0, 0);
            return;
        }
        int mid = (start + end) / 2;
        buildTree(tree, node * 2, start, mid);
        buildTree(tree, node * 2 + 1, mid + 1, end);
        tree[node] = merge(tree[node * 2], tree[node * 2 + 1]);
    }

    private static Node merge(Node a, Node b) {
        return new Node(Math.min(a.val, b.val), 0, Math.min(a.min, b.min));
    }

    private static void increment(Node[] tree, int node, int start, int end, int l, int r, int x) {
        if (r < start || end < l) {
            return;
        }
        if (l <= start && end <= r) {
            tree[node].val += x;
            tree[node].add += x;
            tree[node].min += x;
            return;
        }
        int mid = (start + end) / 2;
        increment(tree, node * 2, start, mid, l, r, x);
        increment(tree, node * 2 + 1, mid + 1, end, l, r, x);
        tree[node] = merge(tree[node * 2], tree[node * 2 + 1]);
    }

    private static int getMax(Node[] tree, int node, int start, int end, int l, int r) {
        if (r < start || end < l) {
            return Integer.MAX_VALUE;
        }
        if (l <= start && end <= r) {
            return tree[node].min;
        }
        int mid = (start + end) / 2;
        int leftMax = getMax(tree, node * 2, start, mid, l, r);
        int rightMax = getMax(tree, node * 2 + 1, mid + 1, end, l, r);
        return Math.min(leftMax, rightMax);
    }

    private static int get(Node[] tree, int node, int start, int end, int idx) {
        if (start == end) {
            return tree[node].val;
        }
        int mid = (start + end) / 2;
        if (idx <= mid) {
            return get(tree, node * 2, start, mid, idx) + (int) tree[node].add;
        } else {
            return get(tree, node * 2 + 1, mid + 1, end, idx) + (int) tree[node].add;
        }
    }
}
