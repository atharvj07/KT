#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll ln = 1e9 + 7;
const ll mod = 1e9 + 7;
const ll maxn = 2e5 + 5;
ll n, m, c, h, x, y;
pair<ll, ll> hotels[maxn], restaurants[maxn];
int main() {
  cin >> n >> m >> c;
  for (int i = 0; i < c; i++)cin >> hotels[i].first >> hotels[i].second;
  cin >> h;
  for (int i = 0; i < h; i++)cin >> restaurants[i].first >> restaurants[i].second;
  sort(hotels, hotels + c);
  sort(restaurants, restaurants + h);
  ll low = 0, high = n + m;
  while (low <= high) {
    ll mid = low + (high - low) / 2;
    bool ans = false;
    for (int i = 0; i < c; i++) {
      bool fl = false;
      for (int j = 0; j < h; j++) {
        if (abs(hotels[i].first - restaurants[j].first) + abs(hotels[i].second - restaurants[j].second) <= mid) {
          fl = true;
        }
      }
      ans |= fl;
    }
    if (ans)
      high = mid - 1;
    else
      low = mid + 1;
  }
  bool flg = false;
  ll ans = ln;
  for (int i = 0; i < c; i++) {
    for (int j = 0; j < h; j++) {
      if (abs(hotels[i].first - restaurants[j].first) + abs(hotels[i].second - restaurants[j].second) == low) {
        if (!flg) {
          flg = true;
          ans = restaurants[j].first;
        } else
          ans = min(ans, restaurants[j].first);
      }
    }
  }
  cout << low << "\n";
  cout << ans << "\n";
}
