import java.util.*;
import java.io.*;

public class Main {
    static int[] degree;
    static List<Integer>[] graph;
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        degree = new int[n + 1];
        graph = new List[n + 1];
        for (int i = 0; i <= n; i++) graph[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int v = scanner.nextInt();
            int u = scanner.nextInt();
            degree[u]++;
            graph[v].add(u);
        }
        int[] ans = topologicalSort(n);
        if (ans == null) System.out.println(-1);
        else {
            for (int x : ans) System.out.print(x + " ");
            System.out.println();
        }
    }

    static int[] topologicalSort(int n) {
        int[] ans = new int[n];
        Queue<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++)
            if (degree[i] == 0) q.add(i);
        int l = 0;
        while (!q.isEmpty()) {
            int v = q.poll();
            ans[l++] = v;
            for (int u : graph[v]) {
                degree[u]--;
                if (degree[u] == 0) q.add(u);
            }
        }
        if (l == n) return ans;
        else return null;
    }
}
