#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }

  vector<int> freq(k + 1, 0);
  for (int i = 0; i < n; i++) {
    freq[a[i]]++;
  }

  int min_freq = n + 1;
  int ans = -1;
  for (int i = 1; i <= k; i++) {
    if (freq[i] < min_freq) {
      min_freq = freq[i];
      ans = i;
    }
  }

  cout << ans << endl;

  return 0;
}
