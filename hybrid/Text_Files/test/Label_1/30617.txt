import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt(), mod = (int) 1e9 + 7;
        long[] a = new long[n], w = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        for (int i = 0; i < n; i++) w[i] = sc.nextLong();

        Map<Long, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) map.put(a[i], i);

        while (q-- > 0) {
            int type = sc.nextInt();

            if (type > 0) {
                int l = type - 1, r = sc.nextInt() - 1;

                int left = map.get(a[l]);
                int right = map.get(a[r]);

                long ans = 0;

                for (int i = left + 1; i < right; i++) {
                    long gap = a[i] - a[i - 1];
                    ans = (ans + gap * (w[i - 1] + w[i])) % mod;
                }

                System.out.println(ans);
            } else {
                int id = -type;
                long nw = sc.nextLong();
                w[id - 1] = nw;
            }
        }
    }
}
