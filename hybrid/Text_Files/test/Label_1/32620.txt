import java.util.Scanner;

public class BerryBaskets {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int maxBaskets = 0;

        for (int i = 0; i < n; i++) {
            int redBerries = scanner.nextInt();
            int blueBerries = scanner.nextInt();

            // Count baskets for berries of the same color
            maxBaskets += Math.min(redBerries, k) / k;
            maxBaskets += Math.min(blueBerries, k) / k;

            // Count baskets for berries from the same shrub
            maxBaskets += Math.min(redBerries + blueBerries, k) / k;
        }

        System.out.println(maxBaskets);
    }
}
