#include <iostream>
#include <queue>
using namespace std;
#define MAXN 105
#define MAXM 105
struct cell {
  int x, cycle;
  cell(int xx = 0, int cc = 0) { x = xx, cycle = cc; }
};
queue<cell> Q;
int n, m, k, x, locked[MAXN], lockCycle[MAXN];
bool canDo[MAXM][MAXN];
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> x;
      if (x) Q.push(cell(x, j));
    }
  }
  while (!Q.empty()) {
    cell CC = Q.front();
    Q.pop();
    if (locked[CC.x]) continue;
    locked[CC.x] = 1;
    if (canDo[CC.x][CC.cycle] == 0) canDo[CC.x][CC.cycle] = 1; else {
      lockCycle[CC.cycle] = 1;
      for (int i = CC.cycle; i <= m; i++) {
        if (canDo[CC.x][i]) { locked[CC.x] = 0; }
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    if (locked[i])
      cout << 1 << endl;
    else {
      for (int j = 1; j <= m; j++) {
        if (canDo[i][j] == 0) {
          cout << j << endl;
          break;
        }
      }
    }
  }
  return 0;
}
