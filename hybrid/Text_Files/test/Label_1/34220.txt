import java.util.*;
class Main {
    void solve(Scanner in) {
        int n = in.nextInt(), l = in.nextInt(), k = in.nextInt()-1;
        char[] s = in.next().toCharArray();
        Arrays.sort(s);
        while (--n >= 0) {
            for (int i = k * l, j = k * l + l; i < j; i++)
                System.out.print(s[i]);
            System.out.println();
        }
    }
    public static void main(String[] args) {
        new Main().solve(new Scanner(System.in));
    }
}
