#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
#define Ass(a, b) a = b
#define MAX 100005
bool ans = true;
int n, k, cnt, dp[MAX];
string s;
int fun(int i) {
    if(i >= n) {
        if(ans && cnt == k) {
            cnt = 0;
            ans = false;
            return 0;
        }
        return INT_MAX;
    }
    int &ret = dp[i];
    if(ret != -1) return ret;
    ret = fun(i+1);
    if(ans && (i + 1 == n || s[i+1] - s[i] > 1)) {
        ret = min(ret, s[i] - 'a' + 1 + fun(i+1));
        cnt++;
    }
    return ret;
}
int main() {
    cin >> n >> k;
    cin >> s;
    memset(dp, -1, sizeof(dp));
    int anss = fun(0);
    if(ans) cout << "-1\n";
    else cout << anss << endl;
}
