#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
      if (i > 0 && s[i] > s[i - 1]) {
        s[i - 1] = 'a';
      }
      if (s[i] == 'a') {
        s[i] = 'z';
      } else {
        s[i]--;
      }
    }
    cout << s << endl;
  }
  return 0;
}
