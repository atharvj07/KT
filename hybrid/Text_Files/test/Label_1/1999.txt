#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

typedef long long ll;
typedef long double ld;
typedef pair<int,int> pii;

const int N = 2e5 + 5;
ll a[N], pre[N][11];
ll k[11];

void solve() {
    int n, m; cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 0; i <= m; i++) cin >> k[i];
    ll sum[m + 1]; memset(sum, 0, sizeof(sum));
    for (int j = 1; j <= m; j++) {
        for (int i = 1; i <= n; i++) {
            if (a[i] == j) sum[j]++;
        }
    }
    for (int j = 1; j <= m; j++) {
        for (int i = 1; i <= n; i++) {
            pre[i][j] = pre[i - 1][j] + (a[i] == j);
        }
    }
    int cnt = 0;
    for (int l = 1; l <= n; l++) {
        memset(sum, 0, sizeof(sum));
        for (int r = l; r <= n; r++) {
            cnt = 0;
            for (int i = 1; i <= m; i++) {
                sum[i] += (pre[r][i] - pre[l - 1][i]);
                if (sum[i] < k[i]) {
                    cnt++;
                }
            }
            if (cnt == 0) {
                cout << l - 1 << endl;
                return;
            }
        }
    }
    cout << -1 << endl;
}

int main () {
    solve();
    return 0;
}
