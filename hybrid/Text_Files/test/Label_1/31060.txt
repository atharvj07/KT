import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        Map<String, Integer> map = new HashMap<>();
        for (int i = 0; i < N; i++) {
            String s = sc.next();
            int t = sc.nextInt();
            map.put(s, t);
        }
        String X = sc.next();
        int ans = 0;
        for (String s : map.keySet()) {
            if (s.equals(X)) {
                break;
            }
            ans += map.get(s);
        }
        System.out.println(ans);
    }
}
