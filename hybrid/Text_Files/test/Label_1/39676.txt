n = int(input())
a, c = [], []
for i in range(n):
    aa, cc = map(int, input().split())
    a.append(aa)
    c.append(cc)
ans = 1e18
a.sort()
c.sort()
for i in range(n):
    cost = 0
    for j in range(n):
        cost += max(c[(j - i) % n], a[j] - a[(j - i) % n])
    ans = min(ans, cost)
print(int(ans))
