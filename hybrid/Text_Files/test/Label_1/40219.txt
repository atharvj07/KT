n = int(input())
s = input()

has_red = 'R' in s
has_green = 'G' in s
has_blue = 'B' in s

if (has_red + has_green + has_blue) >= 2:
    if (has_red and has_green and has_blue) or (s.count('R') % 2 + s.count('G') % 2 + s.count('B') % 2) > 1:
        print('BGR')
    else:
        if s.count('R') % 2:
            print('R'*(has_red) + 'G'*(has_green) + 'B'*(not has_blue))
        else:
            print('R'*(not has_red) + 'G'*(has_green and not has_green % 2) + 'B'*(has_blue and not has_blue % 2))
else:
    print('R'*has_red + 'G'*has_green + 'B'*has_blue)
