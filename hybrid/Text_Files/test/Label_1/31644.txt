import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        String s = in.next();

        char[] ch = s.toCharArray();
        for (int i = 0; i < n && k > 0; i++) {
            if (ch[i] > '0') {
                ch[i] = '0';
                k--;
            }
        }

        if (k > 0) { // change all remaining digits to '0'.
            for (int i = n - 1; i >= 0 && k > 0; i--) {
                if (ch[i] != '0') {
                    ch[i] = '0';
                    k--;
                }
            }
        }

        while (n > 1 && ch[n - 1] == '0') { // remove all leading '0'.
            n--;
        }

        System.out.println(new String(ch).substring(0, n));
    }
}
