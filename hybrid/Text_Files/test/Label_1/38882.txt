import sys
n = 1001
ans_num = 1
ans = ''
s = input()
t = input()
c = [0 for i in range(n)]
for i in s:
    c[ord(i)] += 1
for i in t:
    c[ord(i)] -= 1
    if c[ord(i)] < 0:
        ans_num = -1
        break
if ans_num != -1:
    for i in range(26):
        while c[i] > 0:
            c[i] -= 1
            ans += chr(i+ord('a'))
    print(ans_num)
    print(ans)
else:
    print(-1)
