import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] c = new int[n + 1];
        while (m-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            c[a]++;
            c[b]++;
        }
        long ans = 1;
        for (int i = 1; i <= n; i++) {
            ans = ans * (c[i] <= 1 ? c[i] * 2 : c[i]) % k;
        }
        System.out.println(ans);
    }
}
