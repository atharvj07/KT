import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class SerejaAndSubsequences {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        // Calculate the number of subsequences for each element
        int[] dp = new int[n + 1];
        dp[0] = 1;
        for (int i = 1; i <= n; i++) {
            dp[i] = (dp[i - 1] * 2) % MOD;
        }

        // Calculate the number of sequences that do not exceed each subsequence
        List<List<Integer>> subsequences = new ArrayList<>();
        subsequences.add(new ArrayList<>()); // Empty subsequence
        for (int i = 0; i < n; i++) {
            int size = subsequences.size();
            for (int j = 0; j < size; j++) {
                List<Integer> subsequence = new ArrayList<>(subsequences.get(j));
                if (subsequence.isEmpty() || a[i] >= subsequence.get(subsequence.size() - 1)) {
                    subsequence.add(a[i]);
                    subsequences.add(subsequence);
                }
            }
        }

        // Calculate the total number of sequences
        long answer = 0;
        for (List<Integer> subsequence : subsequences) {
            answer += (long) Math.pow(2, n - subsequence.size());
        }

        // Print the answer modulo MOD
        System.out.println(answer % MOD);
    }
}
