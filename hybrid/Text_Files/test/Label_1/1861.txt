#include <bits/stdc++.h>

using namespace std;

typedef int64_t ll;
typedef pair<int, int> pi;

const int MAXN = 1e5 + 5;
const int INF = 1e9 + 7;
const int LOGN = 20;
const ll MOD = 998244353;

int n;
string a, b;

void solve() {
    int l = 0, r = 0;
    while (l < a.size() && r < b.size()) {
        if (a[l] == b[r]) {
            l++;
            r++;
        } else {
            l = 0;
            r++;
        }
    }
    if (l == a.size()) {
        cout << b.size() << '\n';
    } else if (r == b.size()) {
        cout << a.size() << '\n';
    } else {
        cout << "-1\n";
    }
}

int main() {
    cin >> a >> b;
    solve();
    return 0;
}
