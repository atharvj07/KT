import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][][] sofas = new int[n][m][2];
        for (int i = 0; i < d; i++) {
            int x1 = sc.nextInt() - 1;
            int y1 = sc.nextInt() - 1;
            int x2 = sc.nextInt() - 1;
            int y2 = sc.nextInt() - 1;
            if (x1 == x2) {
                if (y1 > y2) {
                    int tmp = y1;
                    y1 = y2;
                    y2 = tmp;
                }
                for (int j = y1; j <= y2; j++) {
                    sofas[x1][j][0]++;
                    sofas[x1][j][1] = i + 1;
                }
            } else {
                if (x1 > x2) {
                    int tmp = x1;
                    x1 = x2;
                    x2 = tmp;
                }
                for (int j = x1; j <= x2; j++) {
                    sofas[j][y1][1]++;
                    sofas[j][y1][0] = i + 1;
                }
            }
        }
        int cntl = sc.nextInt();
        int cntr = sc.nextInt();
        int cntt = sc.nextInt();
        int cntb = sc.nextInt();
        int ans = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (sofas[i][j][0] == cntl && sofas[i][j][1] != 0 && sofas[i][j][1] <= d) {
                    if (i + 1 >= cntt + 1 && n - i >= cntb + 1) {
                        if (j + 1 >= cntl + 1 && m - j >= cntr + 1) {
                            ans = sofas[i][j][1];
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
