#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9+7;
const ll LL_INF = 0x3f3f3f3f3f3f3f3f;
const int INF = 0x3f3f3f3f;
const double PI = acos(-1);

struct Edge{
	int u, v;
};

vector<Edge> parseEdge(int m) {
	vector<Edge> edges;
	for(int i=0;i<m;i++) {
		int u, v;
		cin >> u >> v;
		edges.push_back({u, v});
	}
	return edges;
}

void solve(int n, int m, vector<Edge> edges) {
	vector<vector<int>> g(n+1);
	for(Edge e: edges) {
		g[e.u].push_back(e.v);
		g[e.v].push_back(e.u);
	}
	vector<int> vis(n+1);
	int cnt = 0;
	vector<int> ans;
	for(int i=1;i<=n;i++) {
		if(!vis[i]) {
			cnt++;
			queue<int> q;
			q.push(i);
			while(!q.empty()) {
				int u = q.front();
				q.pop();
				if(vis[u]) continue;
				ans.push_back(cnt);
				vis[u] = cnt;
				for(int adj: g[u]) {
					if(vis[adj]) continue;
					q.push(adj);
				}
			}
		}
	}
	if(cnt==1) {
		for(int x: ans) cout << x << " ";
		return;
	}
	if(cnt==2) {
		int sz1=0, sz2=0;
		for(int x: ans) {
			if(x==1) sz1++;
			else sz2++;
		}
		if(sz1>sz2) {
			int c=0;
			for(int x: ans) {
				if(c==sz1) {
					x=3;
					cout << x << " ";
					continue;
				}
				cout << x << " ";
				c++;
			}
		}
		else {
			int c=0;
			for(int x: ans) {
				if(c==sz2) {
					x=3;
					cout << x << " ";
					continue;
				}
				cout << x << " ";
				c++;
			}
		}
		return;
	}
	cout << "-1" << endl;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int n, m;
	cin >> n >> m;
	solve(n, m, parseEdge(m));
	return 0;
}
