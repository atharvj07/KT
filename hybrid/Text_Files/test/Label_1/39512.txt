import queue
inf = float('inf')

n, m = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

ships, bases, cost, maxcost = map(int, input().split())

spaceships = []
for _ in range(ships):
    x, a, f = map(int, input().split())
    spaceships.append((x, a, f))

strongholds = []
for _ in range(bases):
    x, d = map(int, input().split())
    strongholds.append((x, d))

dist = [[inf] * (n + 1) for _ in range(n + 1)]
for i in range(1, n + 1):
    dist[i][i] = 0
    q = queue.SimpleQueue()
    q.put((i, 0))
    while not q.empty():
        cur, d = q.get()
        for nxt in g[cur]:
            if d + 1 < dist[i][nxt]:
                q.put((nxt, d + 1))
                dist[i][nxt] = d + 1

cur = 0
ans = inf
for h in range(0, n + 1):
    cost = h * maxcost
    spaceships_left = ships
    strongholds_left = bases
    match = [False] * ships
    match_strongholds = [-1] * ships
    while spaceships_left > 0 and strongholds_left > 0:
        can = -1
        can_stronghold = -1
        for i in range(ships):
            if match[i]: continue
            for j in range(bases):
                if match_strongholds[i] != j and spaceships[i][0] != strongholds[j][0] and spaceships[i][2] >= dist[spaceships[i][0]][strongholds[j][0]] and spaceships[i][1] >= strongholds[j][1]:
                    if can == -1 or dist[spaceships[i][0]][strongholds[j][0]] < dist[spaceships[can][0]][strongholds[can_stronghold][0]]:
                        can = i
                        can_stronghold = j
        if can == -1: break
        match[can] = True
        match_strongholds[can] = can_stronghold
        spaceships_left -= 1
        strongholds_left -= 1
        cost += cost
    ans = min(ans, cost)

print(ans)
