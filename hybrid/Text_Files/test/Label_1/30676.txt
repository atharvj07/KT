import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int h = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();

        int h1 = sc.nextInt();
        int m1 = sc.nextInt();

        int h2 = sc.nextInt();
        int m2 = sc.nextInt();

        int hh = h * m;
        int mm = 60;
        int c = 0;
        for (int i = h1 * m + m1; i <= h2 * m + m2; i++) {
            int t = i;
            int ct = 0;
            while (t > 0) {
                if (t % 10 != (i % 10)) ct++;
                t /= 10;
                i /= 10;
            }
            if (ct >= k) c++;
        }
        System.out.println(c);
    }
}
