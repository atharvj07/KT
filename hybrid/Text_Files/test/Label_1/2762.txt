#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(1 << n);
  for (int i = 2; i < (1 << n); i++) { cin >> a[i]; }
  vector<int> c(1 << n);
  for (int i = (1 << n) - 1; i >= 1; i--) {
    if (i % 2 == 0) {
      c[i] = a[i / 2] - max(c[i / 2], c[i / 2 + 1]);
      a[i / 2] = max(c[i / 2], c[i / 2 + 1]);
    }
  }
  cout << -c[1] << endl;
  return 0;
}
