#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long n, m;
  cin >> n >> m;
  vector<bool> dirty(n + 1, false);
  for (long long i = 0; i < m; i++) {
    long long x;
    cin >> x;
    dirty[x] = true;
  }
  if (dirty[1] || dirty[n])
    cout << "NO" << endl;
  else {
    for (long long i = 3; i <= n; i++)
      if (dirty[i - 1] && dirty[i - 2]) {
        cout << "NO" << endl;
        return 0;
      }
    cout << "YES" << endl;
  }
  return 0;
}
