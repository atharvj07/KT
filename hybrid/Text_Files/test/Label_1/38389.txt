x1, y1, x2, y2, x3, y3 = map(int, input().split())
if x1*y1 + x2*y2 <= x3*y3:
  print(-1)
else:
  n = 1
  while n*n < x1*y1 + x2*y2:
    n += 1
  print(n)
  for i in range(n):
    a = ""
    for j in range(n):
      if j < x1 and i < y1: a += "A"
      elif j < x1 + x2 and i < y2: a += "B"
      else: a += "C"
    print(a)
