import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt(), v = sc.nextInt();

        System.out.println(Math.max((a + v - 1) / v, (a + b + k - 2) / (k - 1)));
    }
}
