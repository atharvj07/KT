import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] friends = new int[n];
        for (int i = 0; i < n; i++) {
            friends[i] = in.nextInt();
        }
        Arrays.sort(friends);
        if (friends[0] == friends[n - 1]) {
            System.out.println(friends[0]);
        } else if (friends[0] == 0 && friends[n - 1] == n - 1) {
            System.out.println(n - 2);
        } else if (friends[0] == 0 && friends[n - 1] == n) {
            System.out.println(n - 1);
        } else if (friends[0] == 0 && friends[n - 2] == n - 1) {
            System.out.println(n);
        } else if (friends[0] + 1 == friends[n - 1]) {
            System.out.println(friends[0] + 1);
        } else {
            System.out.println(-1);
        }
        in.close();
    }
}
