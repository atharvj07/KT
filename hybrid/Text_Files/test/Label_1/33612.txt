import java.util.*;
public class Main {
    static ArrayList<Integer> g[];
    static int n, m, cnt, deg[];
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        g = new ArrayList[n + 1];
        deg = new int[n + 1];
        for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            g[u].add(v);
            g[v].add(u);
            deg[u]++;
            deg[v]++;
        }
        int rem = (n + m) / 2, ans = 0;
        PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> deg[b] - deg[a]);
        for (int i = 1; i <= n; i++) {
            if (deg[i] > rem) pq.add(i);
            else if (deg[i] > 0) ans++;
        }
        while (!pq.isEmpty() && rem > 0) {
            int u = pq.poll();
            rem--;
            deg[u]--;
            if (deg[u] > rem) pq.add(u);
            else if (deg[u] > 0) ans++;
        }
        System.out.println(ans);
        for (int u = 1; u <= n; u++) {
            for (Integer v : g[u]) {
                if (deg[u] <= 0 || deg[v] <= 0) continue;
                System.out.println(u + " " + v);
                deg[u]--;
                deg[v]--;
                ans--;
                if (ans == 0) return;
            }
        }
    }
}
