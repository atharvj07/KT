n, m, k = map(int, input().split())
lst = [0] + list(map(int, input().split()))
ans = []
for i in range(m, 0, -1):
    if not lst[i]:
        continue
    if lst[i] <= k:
        ans.append(10 * (m - i))
        break
for i in range(m + 1, n + 1):
    if not lst[i]:
        continue
    if lst[i] <= k:
        ans.append(10 * (i - m))
        break
print(min(ans))
