import sys

n, m, w = map(int, input().split())
a = [int(x) for x in input().split()]
a_sorted = sorted(a)
pre = [0] * n
for i in range(1, n):
    pre[i] = pre[i - 1] + a[i] - a[i - 1]
# print(pre)

dp = [0] * (m + 1)
for i in range(1, m + 1):
    l = max(i - w, 0)
    r = i - 1
    dp[i] = a[l] + pre[r] - (pre[l - 1] if l > 0 else 0)
    dp[i] = max(dp[i], dp[i - 1])

print(dp[m])
