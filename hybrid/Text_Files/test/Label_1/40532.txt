from itertools import combinations

def count_intersecting_pairs(n, poles):
    # Function to calculate the slope between two points
    def slope(p1, p2):
        if p1[0] == p2[0]:  # Vertical line
            return float('inf')
        return (p2[1] - p1[1]) / (p2[0] - p1[0])
    
    lines = {}
    for (p1, p2) in combinations(poles, 2):
        m = slope(p1, p2)
        # Using a set to avoid duplicate points on the same line
        if m in lines:
            lines[m].add(p1)
            lines[m].add(p2)
        else:
            lines[m] = {p1, p2}
    
    # Count the intersection pairs
    intersections = 0
    for line1 in lines.values():
        for line2 in lines.values():
            if line1 is not line2 and not line1.isdisjoint(line2):
                intersections += 1
    return intersections // 2  # Each pair is counted twice

# Input reading
n = int(input())
poles = [tuple(map(int, input().split())) for _ in range(n)]

# Output the result
print(count_intersecting_pairs(n, poles))
