#include<iostream>
#include<tuple>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
  int n, m;
  cin >> n >> m;
  vector<tuple<int,int>> souvenir(n);
  for(int i = 0; i < n; ++i){
    int w, c;
    cin >> w >> c;
    souvenir[i] = {c,w};
  }
  sort(souvenir.begin(), souvenir.end(), greater<tuple<int, int>>());
  int ans=0;
  for(auto s:souvenir){
    if(get<1>(s) > m) break;
    ans += get<0>(s);
    m -= get<1>(s);
  }
  cout << ans << endl;
  return 0;
}
