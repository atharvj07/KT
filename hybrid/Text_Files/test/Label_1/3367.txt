#include <bits/stdc++.h>
#define mod 998244353

using namespace std;

long long dp[1000005];

long long solve(long long n) {
  if (n == 1) return 1;
  if (dp[n] != -1) return dp[n];
  long long ans = 0;
  for (long long i = 2; i <= n; i += 2) {
    ans = (ans + solve(i - 1) * solve(n - i)) % mod;
  }
  return dp[n] = ans;
}

int main() {
  long long n;
  cin >> n;
  memset(dp, -1, sizeof(dp));
  cout << solve(n) << "\n";
  return 0;
}
