n, k = map(int, input().split())
s = input()
t = input()
ans = 0
for i in range(n):
    for j in range(i + 1, n + 1):
        flag = True
        for letter in s[i:j]:
            if letter not in t:
                flag = False
                break
        if flag:
            ans += 1
print(ans)
