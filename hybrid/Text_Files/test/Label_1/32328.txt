import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long ans = 0;
        while (n-- > 0) {
            long t = sc.nextInt();
            long a = sc.nextInt();
            ans += (a * (ans / a) + 1) * t / gcd(t, a);
        }
        System.out.println(ans);
    }

    static long gcd(long a, long b) {
        if (a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }
}
