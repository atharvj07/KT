import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int ans = 0;
        while (n > 0) {
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] > 0) {
                    cnt++;
                    a[i]--;
                    if (a[i] == 0) {
                        n--;
                    }
                }
            }
            ans++;
        }
        System.out.println(ans);
    }
}
