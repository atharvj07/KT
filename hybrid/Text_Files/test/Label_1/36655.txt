import sys
input = sys.stdin.readline

for _ in range(int(input())):
    s = input().rstrip()
    t = []
    for i in s:
        t.append(int(i) % 2)
    odd = []
    for i in range(len(t)):
        if t[i] == 0:
            odd.append(i)
    ans = [0] * len(s)
    for i in range(len(odd) - 1, -1, -1):
        j = odd[i]
        while j > 0 and t[j - 1] == t[j]:
            ans[j] = s[j]
            j -= 1
        if j > 0:
            j -= 1
            t[j], t[j + 1] = t[j + 1], t[j]
            ans[j] = s[j + 1]
            ans[j + 1] = s[j]
    for i in range(len(s)):
        if ans[i] == 0:
            ans[i] = s[i]
    print(''.join(ans))
