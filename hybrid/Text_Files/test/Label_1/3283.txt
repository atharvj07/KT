#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1e9 + 7;
int32_t main() {
    ll m[5], w[5], c = 0, x;
    for (ll i = 0; i < 5; i++) cin >> m[i];
    for (ll i = 0; i < 5; i++) cin >> w[i];
    cin >> x >> c;
    ll sum = 0;
    for (ll i = 0; i < 5; i++) {
        ll a[5] = {500, 1000, 1500, 2000, 2500};
        sum += max(0.3 * a[i], (1 - (double(m[i]) / 250)) * a[i] - 50 * w[i]);
    }
    cout << sum + 100 * x - 50 * c << '\n';
    return 0;
}
