#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 1;
  int l = 0, r = 0;
  vector<int> last(k + 1, -1);
  while (r < n) {
    if (last[a[r]] != -1 && last[a[r]] >= l) {
      l = last[a[r]] + 1;
    }
    last[a[r]] = r;
    ans = max(ans, r - l + 1);
    r++;
  }
  cout << ans << endl;
  return 0;
}
