class Room:
    def __init__(self):
        self.seats = []
        for i in range(6):
            self.seats.append([ch for ch in input()])

    def find_best_seat(self):
        best_seat = None
        best_convenience = -1
        for i in range(6):
            for j in range(len(self.seats[i])):
                if self.seats[i][j] == '.':
                    convenience = self.calculate_convenience(i, j)
                    if convenience > best_convenience:
                        best_convenience = convenience
                        best_seat = (i, j)
        return best_seat

    def calculate_convenience(self, row, col):
        convenience = 6 - row
        if col == 1:
            convenience += 1
        return convenience

    def update_room(self, best_seat):
        row, col = best_seat
        self.seats[row][col] = 'P'

    def print_room(self):
        for row in self.seats:
            print(''.join(row))

room = Room()
best_seat = room.find_best_seat()
room.update_room(best_seat)
room.print_room()
