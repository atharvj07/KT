import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static int[] d;
    static int[] path;
    static boolean[] vis;

    static boolean bfs(List<Integer>[] adjList, int s) {
        boolean ok = true;
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        vis[s] = true;
        d[s] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adjList[u]) {
                if (!vis[v]) {
                    vis[v] = true;
                    d[v] = d[u] ^ 1;
                    path[v] = u;
                    q.add(v);
                } else if (d[v] == d[u]) {
                    ok = false;
                }
            }
        }
        return ok;
    }

    static void printPath(int u) {
        if (u == 0) {
            return;
        }
        printPath(path[u]);
        System.out.print(u + " ");
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Integer>[] adjList = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adjList[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            adjList[u].add(v);
            adjList[v].add(u);
        }
        d = new int[n + 1];
        path = new int[n + 1];
        vis = new boolean[n + 1];
        if (!bfs(adjList, 1)) {
            System.out.println("Impossible");
        } else {
            int days = 0;
            for (int i = 1; i <= n; i++) {
                if (!vis[i]) {
                    if (!bfs(adjList, i)) {
                        System.out.println("Impossible");
                        return;
                    }
                    days++;
                }
            }
            System.out.println(days);
            for (int i = 1; i <= n; i++) {
                if (d[i] == 0) {
                    printPath(i);
                    System.out.println();
                }
            }
        }
    }
}
