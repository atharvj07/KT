#include <bits/stdc++.h>
using namespace std;
#define ll long long
struct coo {
    ll x, y;
    bool operator<(coo b) { return x < b.x || (x == b.x && y < b.y); };
};
vector<set<ll>> ban(1005);
ll n, m, ans;
int main() {
    cin >> n >> m;
    for (ll i = 0, x, y; i < m; i++) {
        cin >> y >> x;
        ban[x].insert(y);
    }
    for (ll i = 1; i <= n; i++) {
        set<ll> used;
        for (ll j = 1; j < n; j++) {
            if (!ban[i].count(j) && !ban[n - i + 1].count(n - j + 1)) {
                if (!used.count(j) && !used.count(n - j + 1)) {
                    ans++;
                    used.insert(j);
                    used.insert(n - j + 1);
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
