#include <vector>
#include <iostream>
#include <climits>
#include <cmath>

double dist(int x1, int y1, int x2, int y2) {
    return sqrt((x2 - x1)*(x2 - x1) + (y2 - y1)*(y2 - y1));
}

int main() {
    int n, r;
    std::cin >> n >> r;
    std::vector<int> x(n);
    for (int i = 0; i < n; i++)
        std::cin >> x[i];
    std::vector<double> y(n, 0);
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            double d = dist(x[i], 1000000000, x[j], 1000000000);
            if (d <= 2*r) {
                double theta = acos((2*r*2*r - d*d) / (2*2*r*2*r));
                double tmp = r*sin(theta);
                if (y[i] < tmp) y[i] = tmp;
                tmp = r*sin(acos((2*r*2*r - d*d) / (2*2*r*2*r)));
                if (y[j] < tmp) y[j] = tmp;
            }
        }
    }
    for (int i = 0; i < n; i++)
        std::cout << (y[i] == 0 ? 0 : r*r / y[i]) << " ";
    return 0;
}
