import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) {
                return;
            }
            Set<Point> gems = new HashSet<>();
            for (int i = 0; i < n; i++) {
                int x = scanner.nextInt();
                int y = scanner.nextInt();
                gems.add(new Point(x, y));
            }
            int m = scanner.nextInt();
            Robot robot = new Robot(10, 10);
            for (int i = 0; i < m; i++) {
                char direction = scanner.next().charAt(0);
                int distance = scanner.nextInt();
                robot.move(direction, distance);
                gems.removeAll(robot.getCollectedGems());
            }
            if (gems.isEmpty()) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }
    }

    private static class Point {
        int x;
        int y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public boolean equals(Object obj) {
            if (this == obj) {
                return true;
            }
            if (obj == null || getClass() != obj.getClass()) {
                return false;
            }
            Point point = (Point) obj;
            return x == point.x && y == point.y;
        }

        @Override
        public int hashCode() {
            return 31 * x + y;
        }
    }

    private static class Robot {
        int x;
        int y;
        Set<Point> collectedGems;

        Robot(int x, int y) {
            this.x = x;
            this.y = y;
            collectedGems = new HashSet<>();
        }

        void move(char direction, int distance) {
            switch (direction) {
                case 'N':
                    y += distance;
                    break;
                case 'E':
                    x += distance;
                    break;
                case 'S':
                    y -= distance;
                    break;
                case 'W':
                    x -= distance;
                    break;
            }
            for (int i = 1; i <= distance; i++) {
                switch (direction) {
                    case 'N':
                        collectedGems.add(new Point(x, y - i));
                        break;
                    case 'E':
                        collectedGems.add(new Point(x + i, y));
                        break;
                    case 'S':
                        collectedGems.add(new Point(x, y + i));
                        break;
                    case 'W':
                        collectedGems.add(new Point(x - i, y));
                        break;
                }
            }
            collectedGems.add(new Point(x, y));
        }

        Set<Point> getCollectedGems() {
            return collectedGems;
        }
    }
}
