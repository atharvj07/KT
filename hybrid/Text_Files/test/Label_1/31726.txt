import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long cur = a[i] * (n - 1 - i);
            ans += cur;
            sum -= a[i];
        }
        System.out.println(ans - sum);
    }
}
