#include <algorithm>
#include <vector>
#include <iostream>

using namespace std;

typedef long long LL;
typedef vector<int> VI;
typedef vector<LL> VLL;
typedef pair<int, int> PI;
typedef pair<LL, LL> PLL;

#define FOR(i, b, e) for(int i = (b); i < (e); ++i)
#define REP(i, n) FOR(i, 0, n)
#define REV(i, a, b) for(int i = (a); i >= (b); --i)

const int INF = 1e9;
const LL LINF = 1e18;
const double EPS = 1e-9;
const LL MOD = 1e9 + 7;

int main() {
    int n;
    cin >> n;
    VI v(n), a(n);
    REP(i, n) {
        cin >> v[i];
    }
    REP(i, n) {
        cin >> a[i];
        a[i]--;
    }
    VI d(n, 0), b(n, 0);
    REP(i, n) {
        if (d[i] == 0) {
            int j = i;
            while (true) {
                d[j] = 1;
                j = a[j];
                if (j == i) {
                    break;
                }
            }
            int minCost = INF;
            int minJ = -1;
            j = i;
            while (true) {
                if (v[j] < minCost) {
                    minCost = v[j];
                    minJ = j;
                }
                j = a[j];
                if (j == i) {
                    break;
                }
            }
            if (minJ >= 0) {
                b[minJ] = 1;
            }
        }
    }
    int ans = 0;
    REP(i, n) {
        ans += b[i];
    }
    cout << ans << endl;
    return 0;
}
