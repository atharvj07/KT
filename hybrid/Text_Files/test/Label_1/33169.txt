import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        StringBuilder sb = new StringBuilder();
        int[] cnt = new int[4];
        for (char c : s.toCharArray()) {
            if (c == '+') {
                continue;
            }
            cnt[c - '0']++;
        }
        for (int i = 1; i <= 3; i++) {
            while (cnt[i]-- > 0) {
                sb.append(i).append('+');
            }
        }
        sb.deleteCharAt(sb.length() - 1);
        System.out.println(sb);
    }
}
