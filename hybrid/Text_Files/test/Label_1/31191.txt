import java.io.*;
import java.util.*;

public class Main {
  static int N;
  static int M;
  static int R;
  static int[][][] TIMES;
  static int[][][] DIST;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    N = Integer.parseInt(st.nextToken());
    M = Integer.parseInt(st.nextToken());
    R = Integer.parseInt(st.nextToken());
    TIMES = new int[M][N][N];
    for (int m = 0; m < M; m++) {
      for (int i = 0; i < N; i++) {
        st = new StringTokenizer(br.readLine());
        for (int j = 0; j < N; j++) {
          TIMES[m][i][j] = Integer.parseInt(st.nextToken());
        }
      }
    }
    DIST = new int[R][M][N];
    for (int r = 0; r < R; r++) {
      st = new StringTokenizer(br.readLine());
      int start = Integer.parseInt(st.nextToken()) - 1;
      int end = Integer.parseInt(st.nextToken()) - 1;
      int changes = Integer.parseInt(st.nextToken());
      dijkstra(r, start, end, changes);
      System.out.println(DIST[r][0][end]);
    }
  }

  static void dijkstra(int r, int start, int end, int changes) {
    for (int[] dist : DIST[r]) {
      Arrays.fill(dist, Integer.MAX_VALUE);
    }
    PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[2]));
    pq.offer(new int[]{start, 0, 0});
    DIST[r][0][start] = 0;
    while (!pq.isEmpty()) {
      int[] curr = pq.poll();
      int u = curr[0];
      int car = curr[1];
      int dist = curr[2];
      if (u == end) {
        return;
      }
      if (dist > DIST[r][car][u]) {
        continue;
      }
      for (int v = 0; v < N; v++) {
        if (u == v) {
          continue;
        }
        int newDist = dist + TIMES[car][u][v];
        if (newDist < DIST[r][car][v]) {
          DIST[r][car][v] = newDist;
          pq.offer(new int[]{v, car, newDist});
        }
        if (car != 0 && newDist < DIST[r][0][v]) {
          DIST[r][0][v] = newDist;
          pq.offer(new int[]{v, 0, newDist});
        }
        for (int i = 0; i < M && changes > 0; i++) {
          if (i == car) {
            continue;
          }
          newDist = dist + TIMES[i][u][v];
          if (newDist < DIST[r][i][v]) {
            DIST[r][i][v] = newDist;
            pq.offer(new int[]{v, i, newDist});
          }
        }
      }
    }
  }
}
