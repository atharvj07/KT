import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt();
        char[] arr = sc.next().toCharArray();
        int[] pre = new int[n];
        if (arr[a - 1] != arr[b - 1])
            pre[a - 1] = 1;
        for (int i = a; i < b; i++) {
            if (arr[i] != arr[i - 1])
                pre[i] = pre[i - 1] + 1;
            else
                pre[i] = pre[i - 1];
        }
        int ans = pre[b - 1];
        System.out.println(ans);
    }
}
