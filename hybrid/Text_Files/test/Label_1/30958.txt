import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        for (int i = 1; i <= q; i++) {
            long k = in.nextLong();
            long d = (long) Math.ceil((Math.sqrt(8 * k + 1) - 1) / 2.0);
            System.out.println((d * (d + 1) / 2 - k) % 10);
        }
    }
}
