import java.util.Scanner;
import java.math.BigDecimal;
import java.math.RoundingMode;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        long n = scan.nextLong(), w = scan.nextLong();
        Item[] items = new Item[(int)n];
        for (int i = 0; i < n; i++) {
            items[i] = new Item(scan.nextLong(), scan.nextLong());
        }
        BigDecimal ans = new BigDecimal(0);
        for (Item item : items) {
            if (w >= item.w) {
                ans = ans.add(new BigDecimal(item.v));
                w -= item.w;
            } else {
                ans = ans.add(new BigDecimal(item.v).multiply(new BigDecimal(w)).divide(new BigDecimal(item.w), 10, RoundingMode.HALF_UP));
                break;
            }
        }
        System.out.println(ans);
    }
    
    private static class Item {
        public long v, w;
        public Item(long v, long w) {
            this.v = v;
            this.w = w;
        }
    }
}
