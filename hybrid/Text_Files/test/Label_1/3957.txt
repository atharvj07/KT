#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> b(n);
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += a[i]; }
  if (sum % 2 != 0) {
    for (int i = 0; i < n; i++) {
      if (a[i] % 2 != 0) {
        b[i] = (a[i] + 1) / 2;
        sum -= a[i] + 1;
        break;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    if (sum == 0) { break; }
    if (a[i] % 2 == 1) {
      b[i] = a[i] / 2;
      sum += a[i] / 2;
    }
  }
  for (int i = 0; i < n; i++) { b[i] += (sum / 2); }
  for (int i = 0; i < n; i++) { cout << b[i] << endl; }
  return 0;
}
