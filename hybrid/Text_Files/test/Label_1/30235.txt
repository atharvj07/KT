import java.util.*;
    
public class Main {
    static int n, m;
    static boolean[] l = new boolean[1000001];
    static boolean[] r = new boolean[1000001];
    static boolean[] v = new boolean[1000001];
    static boolean found;
    
    static void dfs(int x) {
        v[x] = true;
        if (!l[x] && !r[x] && found) {
            found = false;
            return;
        }
        if (l[x]) {
            if (v[x << 1]) found = false;
            else dfs(x << 1);
        }
        if (r[x]) {
            if (v[x << 1 | 1]) found = false;
            else dfs(x << 1 | 1);
        }
        System.out.print(x + " ");
    }
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            if (sc.next().equals("LEFT")) l[a] = true;
            else r[a] = true;
        }
        found = true;
        dfs(1);
        if (!found) System.out.println("IMPOSSIBLE");
    }
}
