import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            long d = input.nextInt();
            long k = input.nextInt();
            System.out.println((d - k) * (d - k) > d * d ? "Ashish" : "Utkarsh");
        }
    }
}
