#include <iostream>
#include <cmath>

int main() {
    long long x;
    std::cin >> x;
    std::string s = std::to_string(x), ans = s;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] != '9') {
            ans[i] = '9' - (s[i] - '0');
            break;
        }
    }
    if (ans[0] == '0') ans = '1' + ans.substr(1);
    std::cout << ans << std::endl;
    return 0;
}
