import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < m; j++) {
                switch (line.charAt(j)) {
                    case '.':
                        grid[i][j] = 0;
                        break;
                    case 'P':
                        grid[i][j] = 1;
                        break;
                    case 'W':
                        grid[i][j] = 2;
                        break;
                }
            }
        }
        int maxEaten = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 2) {
                    for (int k = 0; k < 4; k++) {
                        int nx = i + dx[k];
                        int ny = j + dy[k];
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && grid[nx][ny] == 1) {
                            grid[nx][ny] = 0;
                            maxEaten++;
                            break;
                        }
                    }
                }
            }
        }
        System.out.println(maxEaten);
    }

    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};
}
