n=int(input());s=list(input());i=0;j=n-1;ans=0
while i<j:
    if s[i]==s[j]:
        i+=1;j-=1
    else:
        for k in range(i+1,n):
            if s[k]==s[i]:
                for l in range(k,j):
                    s[l],s[l+1]=s[l+1],s[l]
                ans+=k-i;break
        i+=1;j-=1
print(ans)
