import java.io.*;
import java.util.*;

public class Main {
    static int n, m, k, s;
    static int[][] arr;
    static int[] notes;
    static int[] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        s = sc.nextInt();
        arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        notes = new int[s];
        for (int i = 0; i < s; i++) {
            notes[i] = sc.nextInt();
        }
        dp = new int[s];
        Arrays.fill(dp, -1);
        System.out.println(solve(0));
    }

    static int solve(int idx) {
        if (idx == s) {
            return 0;
        }
        if (dp[idx] != -1) {
            return dp[idx];
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (arr[i][j] == notes[idx]) {
                    ans = Math.max(ans, j + 1);
                }
            }
        }
        ans = Math.max(ans, solve(idx + 1));
        return dp[idx] = ans;
    }
}
