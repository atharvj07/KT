import java.util.*;

public class IntegerPair {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long s = scanner.nextLong();
        long x = scanner.nextLong();

        // Check for invalid input
        if (s < 2 || s > 1000000000000L || x < 0 || x > 1000000000000L) {
            System.out.println(0);
            return;
        }

        // If the sum is odd and the XOR is even, there are no solutions.
        if ((s & 1) == 1 && (x & 1) == 0) {
            System.out.println(0);
            return;
        }

        // If the sum is even and the XOR is odd, there are no solutions.
        if ((s & 1) == 0 && (x & 1) == 1) {
            System.out.println(0);
            return;
        }

        // If the sum and XOR are both even, the number of solutions is 2^(s/2).
        if ((s & 1) == 0 && (x & 1) == 0) {
            System.out.println((long) Math.pow(2, s / 2));
            return;
        }

        // If the sum and XOR are both odd, the number of solutions is 2^((s-1)/2).
        if ((s & 1) == 1 && (x & 1) == 1) {
            System.out.println((long) Math.pow(2, (s - 1) / 2));
            return;
        }
    }
}
