import java.util.Scanner;

public class WallSegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int rows = scanner.nextInt();
        int cols = scanner.nextInt();
        scanner.nextLine();

        // Initialize a 2D array to represent the wall
        char[][] wall = new char[rows][cols];

        // Read the wall description from input
        for (int i = 0; i < rows; i++) {
            String row = scanner.nextLine();
            for (int j = 0; j < cols; j++) {
                wall[i][j] = row.charAt(j);
            }
        }

        // Count the wall segments
        int segments = 0;
        boolean inSegment = false;

        for (int i = 0; i < cols; i++) {
            for (int j = 0; j < rows; j++) {
                if (wall[j][i] == 'B') {
                    if (!inSegment) {
                        segments++;
                        inSegment = true;
                    }
                } else {
                    inSegment = false;
                }
            }
        }

        // Print the number of wall segments
        System.out.println(segments);
    }
}
