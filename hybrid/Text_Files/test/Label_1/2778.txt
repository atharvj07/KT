#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  int n;
  cin >> n;
  vector<long long> x(n);
  for (auto &c : x) cin >> c;
  int min = x[1] - x[0], max = x[n - 1] - x[0];
  for (int i = 1; i < n - 1; i++) {
    if (x[i + 1] - x[i] < min) min = x[i + 1] - x[i];
    if (x[i + 1] - x[i] > max) max = x[i + 1] - x[i];
  }
  cout << min << ' ' << max << '\n';
  for (int i = 1; i < n - 1; i++) {
    if (x[i] - x[0] < min) min = x[i] - x[0];
    if (x[i] - x[0] > max) max = x[i] - x[0];
    if (x[n - 1] - x[i] < min) min = x[n - 1] - x[i];
    if (x[n - 1] - x[i] > max) max = x[n - 1] - x[i];
    cout << min << ' ' << max << '\n';
  }
  cout << min << ' ' << max << '\n';
  return 0;
}
