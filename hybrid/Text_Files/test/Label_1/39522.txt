import heapq

n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
ans = a[-1] - a[0]
ans1 = []

while k > 0:
    if a[0] == a[-1]:
        break
    ans -= 1
    k -= 1
    ans1.append((a[0], a[-1]))
    a[0] += 1
    a[-1] -= 1
    a.sort()

print(ans, len(ans1))
for i in ans1:
    print(*i)
