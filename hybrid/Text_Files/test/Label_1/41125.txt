import sys

w1, h1, w2, h2, w3, h3 = map(int, input().split())
maxw = max([w1, w2, w3])
maxh = max([h1, h2, h3])
if maxw * maxh < w1 * h1 + w2 * h2 + w3 * h3:
    print(-1)
    sys.exit()
for n in range(maxw + maxh, 101):
    for i in range(n - maxw + 1):
        for j in range(n - maxh + 1):
            if all(a == b for a, b in zip(
                [
                    [j <= x < j + w1 and i <= y < i + h1 for x in range(n)] for y in range(n)
                ],
                [
                    [j <= x < j + wi and i <= y < i + hi for x in range(n)] for wi, hi in [(w2, h2), (w3, h3)]
                ],
            )):
                print(n)
                for y in range(n):
                    s = ""
                    for x in range(n):
                        if j <= x < j + w1 and i <= y < i + h1:
                            s += "A"
                        elif j <= x < j + w2 and i <= y < i + h2:
                            s += "B"
                        elif j <= x < j + w3 and i <= y < i + h3:
                            s += "C"
                        else:
                            s += "."
                    print(s)
                sys.exit()
print(-1)
