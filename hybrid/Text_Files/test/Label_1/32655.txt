import java.util.Scanner;

public class GoodSegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;
            System.out.println(countGoodSegments(permutation, l, r));
        }
    }

    static int countGoodSegments(int[] permutation, int l, int r) {
        int[] minLeft = new int[r - l + 1];
        int[] maxRight = new int[r - l + 1];
        minLeft[0] = permutation[l];
        maxRight[r - l] = permutation[r];
        for (int i = l + 1; i <= r; i++) {
            minLeft[i - l] = Math.min(minLeft[i - l - 1], permutation[i]);
            maxRight[r - i] = Math.max(maxRight[r - i + 1], permutation[i]);
        }
        int count = 0;
        for (int i = l; i <= r; i++) {
            if (minLeft[i - l] <= permutation[i] && permutation[i] <= maxRight[r - i]) {
                count++;
            }
        }
        return count;
    }
}
