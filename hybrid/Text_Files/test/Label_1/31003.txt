import java.util.HashMap;
import java.util.Scanner;

public class DistinctArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        int minSize = 0;

        // Track the last occurrence of each element
        HashMap<Integer, Integer> lastOccurrence = new HashMap<>();

        // Iterate through the array
        for (int i = 0; i < n; i++) {
            int element = arr[i];

            // If the element already exists in the map
            if (lastOccurrence.containsKey(element)) {
                // Calculate the size of the subsegment to remove
                int subsegmentSize = i - lastOccurrence.get(element);

                // Update the minimum size
                minSize = Math.max(minSize, subsegmentSize);
            }

            // Update the last occurrence of the element
            lastOccurrence.put(element, i);
        }

        // Print the minimum size of the subsegment to remove
        System.out.println(minSize);
    }
}
