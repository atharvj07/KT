import java.util.*;

class Main {

    static class Lady implements Comparable<Lady> {
        int b, i, r;

        Lady(int b, int i, int r) {
            this.b = b;
            this.i = i;
            this.r = r;
        }

        @Override
        public int compareTo(Lady o) {
            if (this.b != o.b) {
                return this.b - o.b;
            } else if (this.i != o.i) {
                return this.i - o.i;
            } else {
                return this.r - o.r;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Lady[] ladies = new Lady[n];
        for (int i = 0; i < n; i++) {
            ladies[i] = new Lady(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        Arrays.sort(ladies);
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (ladies[i].b < ladies[j].b && ladies[i].i < ladies[j].i && ladies[i].r < ladies[j].r) {
                    cnt++;
                    break;
                }
            }
        }
        System.out.println(cnt);
    }
}
