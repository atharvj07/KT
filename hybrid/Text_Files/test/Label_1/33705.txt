import java.util.*;
import static java.lang.Math.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = input.nextInt();
        boolean[] seen = new boolean[2 * arr[n - 1] + 2];
        for (int i = 0; i < n; i++) {
            int x = 0, y = 0;
            for (int j = 0; j <= log(arr[i]) / log(2); j++) {
                int z = (int) pow(2, j);
                if (x + z <= arr[i]) {
                    seen[x + z] = true;
                    x += z;
                } else {
                    y = x + z - arr[i];
                    if (y == 0 || seen[y]) {
                        seen[x] = true;
                        break;
                    }
                    seen[y] = true;
                    break;
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= 2 * arr[n - 1] + 1; i++) ans += seen[i] ? 1 : 0;
        System.out.println(ans == n ? -1 : ans);
    }
}
