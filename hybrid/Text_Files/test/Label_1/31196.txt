import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.PriorityQueue;
import java.util.Scanner;

class Main {
        static HashMap<Integer, List<Edge>> graph;
        public static void main(String[] args) {
                Scanner scanner = new Scanner(System.in);
                int n = scanner.nextInt();
                int m = scanner.nextInt();
                int[] mush = new int[n + 1];
                int max = 0;
                graph = new HashMap<>();
                for (int i = 1; i <= n; i++) {
                        graph.put(i, new ArrayList<>());
                }
                for (int i = 0; i < m; i++) {
                        int a = scanner.nextInt();
                        int b = scanner.nextInt();
                        int c = scanner.nextInt();
                        graph.get(a).add(new Edge(b, c));
                        mush[a] += c;
                        mush[b] += c;
                        max += c;
                }
                PriorityQueue<Node> pq = new PriorityQueue<>();
                pq.add(new Node(1, mush[1]));
                int min = mush[1];
                while (!pq.isEmpty()) {
                        Node curr = pq.poll();
                        max -= min;
                        if (max == 0) {
                                break;
                        }
                        min = Integer.MAX_VALUE;
                        for (Edge edge : graph.get(curr.ind)) {
                                if (mush[edge.dest] > 0) {
                                        min = Math.min(min, mush[edge.dest] -= edge.w);
                                        pq.add(new Node(edge.dest, mush[edge.dest]));
                                }
                        }
                }
                System.out.println(max);
        }
}

class Node implements Comparable<Node> {
        int ind;
        int cost;

        public Node(int ind, int cost) {
                this.ind = ind;
                this.cost = cost;
        }

        @Override
        public int compareTo(Node other) {
                return this.cost - other.cost;
        }
}

class Edge {
        int dest;
        int w;

        public Edge(int dest, int w) {
                this.dest = dest;
                this.w = w;
        }
}
