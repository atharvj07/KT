import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong(), m = input.nextLong();
        long sx = input.nextLong(), sy = input.nextLong(), fx = input.nextLong(), fy = input.nextLong();
        Set<Long> x = new HashSet<>(), y = new HashSet<>();
        for (int i = 0; i < m; i++) {
            long a = input.nextLong(), b = input.nextLong();
            x.add(a); y.add(b);
        }
        long ans = Math.abs(fx - sx) + Math.abs(fy - sy);
        if (x.contains(sx) || x.contains(fx)) ans -= Math.min(Math.abs(fx - sx), Math.abs(fy - sy));
        if (y.contains(sy) || y.contains(fy)) ans -= Math.min(Math.abs(fx - sx), Math.abs(fy - sy));
        System.out.println(ans);
    }
}
