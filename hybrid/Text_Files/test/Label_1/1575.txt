#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    long long ans = 1;
    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;
        int cnt = 0;
        for (char c : s) cnt += (c == '1');
        if (cnt > 1) ans = 0;
    }
    cout << ans << endl;
    return 0;
}
