#include <vector>
#include <limits.h>
#include <queue>
#include <iostream>
#include <algorithm>
using namespace std;
vector<pair<int, int> >adj[71];
int dis[71];
bool vis[71];
int main() {
  int n, m, a, b;
  cin >> n >> m >> a >> b;
  for (int i = 0; i < m; i++) {
    int x, y, w;
    cin >> x >> y >> w;
    if (w == a) {
      adj[x].push_back({y, a});
      adj[y].push_back({x, a});
    } else {
      adj[x].push_back({y, b});
      adj[y].push_back({x, b});
    }
  }
  int s = 1;
  for (int p = 1; p <= n; p++) {
    priority_queue<pair<int, int> >pq;
    fill(dis, dis + 71, INT_MAX);
    pq.push({0, s});
    dis[s] = 0;
    while (!pq.empty()) {
      int d = -pq.top().first;
      int u = pq.top().second;
      pq.pop();
      if (vis[u] == true) {
        continue;
      }
      vis[u] = true;
      if (u == p) {
        break;
      }
      for (pair<int, int>e : adj[u]) {
        int v = e.first;
        int w = e.second;
        if (dis[u] + w < dis[v]) {
          pq.push({-(dis[u] + w), v});
          dis[v] = dis[u] + w;
        }
      }
    }
    if (dis[p] < INT_MAX) {
      cout << dis[p] << " ";
    } else {
      cout << "-1 ";
    }
    for (int i = 1; i <= n; i++) {
      vis[i] = false;
    }
  }
  return 0;
}
