import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Pair[] p = new Pair[n];
        for (int i = 0; i < n; i++) {
            p[i] = new Pair(sc.nextInt(), sc.nextInt());
        }
        Arrays.sort(p);
        int j = 0;
        for (int i = 1; i < n; i++) {
            if (p[i].x == p[i - 1].x) {
                p[j].y = Math.max(p[j].y, p[i].y);
            } else {
                p[++j] = p[i];
            }
        }
        n = j + 1;
        boolean[] ans = new boolean[n];
        for (int i = 0; i < n; i++) {
            boolean ok = true;
            for (int k = i + 1; k < n; k++) {
                if (p[k].y > p[i].y) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                ans[i] = true;
            }
        }
        for (int i = 0; i < n; i++) {
            if (ans[i]) {
                System.out.print((i + 1) + " ");
            }
        }
        System.out.println();
    }

    static class Pair implements Comparable<Pair> {
        int x, y;

        Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }

        @Override
        public int compareTo(Pair o) {
            return this.x - o.x;
        }
    }
}
