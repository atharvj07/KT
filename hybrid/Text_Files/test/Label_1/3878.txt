#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  map<int, pair<int, int>> a;  // map<width, {first empty seat, first extro seat}>
  vector<int> ans(2 * n);
  for (int i = 1; i <= 2 * n; i++) {
    int p;
    cin >> p;
    if (p == 0) {
      auto [l, r] = a.begin()->second;
      ans[i - 1] = l;
      a.begin()->second.first = r;
      if (l == r) a.erase(a.begin());
    } else {
      int w;
      cin >> w;
      if (!a.count(w)) a[w] = {2, 1};
      auto &[l, r] = a[w];
      ans[i - 1] = r;
      r = l;
      if (l == r) a.erase(w);
    }
  }
  for (int x : ans) cout << x << ' ';
  cout << endl;
  return 0;
}
