import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt(), q = sc.nextInt();
        long[] order = new long[n + 1];
        while (q-- > 0) {
            if (sc.nextInt() == 1) {
                order[sc.nextInt()] += sc.nextInt();
            } else {
                long ans = 0, cur = 0;
                for (int j = sc.nextInt(); j <= n; j++) {
                    cur += Math.min(a, order[j]);
                    if (j >= k) cur -= Math.min(b, order[j - k]);
                    ans = Math.max(ans, cur);
                }
                System.out.println(ans);
            }
        }
    }
}
