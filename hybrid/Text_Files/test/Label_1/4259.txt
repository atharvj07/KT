#include <bits/stdc++.h>
using namespace std;

int main() {
    int m, n;
    cin >> m >> n;
    int pic[m][n];
    for (int i = 0; i < m; i++)
        for (int j = 0; j < n; j++) cin >> pic[i][j];
    int ans[m];
    for (int i = 0; i < m; i++) {
        ans[i] = pic[i][0];
        for (int j = 1; j < n; j++) ans[i] += pic[i][j];
    }
    for (int i = 0; i < m; i++) cout << ans[i] << ' ';
    cout << endl;
    return 0;
}
