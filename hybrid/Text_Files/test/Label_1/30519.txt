import java.util.Scanner;

public class Main {
    private static final long MOD = 998244353;
    private static long[][] dp;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n + 1];
        dp = new long[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        dp[n][n] = 1;
        for (int i = n - 1; i >= 1; i--) {
            for (int j = i; j <= n; j++) {
                if (a[i] == j - 1 || a[i] <= 0) {
                    dp[i][j] = (dp[i + 1][j] + dp[i + 1][i - 1]) % MOD;
                }
            }
        }
        System.out.println(dp[1][n]);
    }
}
