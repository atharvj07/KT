#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 3e5 + 5;
int p[N], q[N], pr[N], pp[N], nxt[N], fs[N], ls[N];
vector<int> g[N];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
        g[p[i]].push_back(i);
    }
    for (int i = 1; i <= k; i++) {
        int x, y;
        cin >> x >> y;
        nxt[x] = y;
    }
    int cnt = 0;
    for (int i = 0; i <= n; i++) {
        if (g[i].size()) pr[++cnt] = i;
    }
    for (int i = 1; i <= cnt; i++) {
        int v = pr[i];
        for (int j = 0; j < g[v].size(); j++) {
            int u = g[v][j];
            pp[u] = v;
            if (!fs[u]) fs[u] = i;
            ls[u] = i;
        }
    }

    vector<int> ans;
    for (int i = 1; i <= cnt; i++) {
        int v = pr[i];
        while (fs[v] == i) {
            ans.push_back(v);
            v = pp[v];
        }
        if (nxt[v]) {
            int u = nxt[v];
            ans.push_back(v);
            if (ls[u] <= i) {
                cout << 0;
                return 0;
            }
            if (nxt[u]) {
                int z = nxt[u];
                if (fs[z] == i) {
                    cout << 0;
                    return 0;
                }
            }
            while (u) {
                ans.push_back(u);
                u = pp[u];
            }
        }
    }
    if (ans.size() < n) {
        cout << 0;
        return 0;
    }
    reverse(ans.begin(), ans.end());
    for (int i = 0; i < n; i++) cout << ans[i] << ' ';
}
