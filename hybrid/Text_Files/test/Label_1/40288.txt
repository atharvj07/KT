def max_gift_sets(x, y, a, b):
    if a < b:
        a, b = b, a
    if a == b:
        return (x + y) // (a + b)
    lo, hi = 0, min(x, y) + 1
    while lo < hi:
        mid = (lo + hi) // 2
        if x >= mid * a + max(0, (mid - 1)) * b and y >= mid * b + max(0, (mid - 1)) * a:
            lo = mid + 1
        else:
            hi = mid
    return lo - 1

t = int(input().strip())
for _ in range(t):
    x, y, a, b = map(int, input().strip().split())
    print(max_gift_sets(x, y, a, b))
