import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next(), c = sc.next();
            int n = s.length();
            StringBuilder sb = new StringBuilder(s);
            int i = -1, j = -1;
            for (int k = 0; k < n; ++k) {
                if (s.charAt(k) > c.charAt(k)) {
                    i = k;
                    break;
                }
            }
            if (i != -1) {
                for (int k = i + 1; k < n; ++k) {
                    if (s.charAt(k) < c.charAt(i)) {
                        j = k;
                        break;
                    }
                }
            }
            if (i != -1 && j != -1) {
                char tmp = sb.charAt(i);
                sb.setCharAt(i, sb.charAt(j));
                sb.setCharAt(j, tmp);
            }
            if (sb.toString().compareTo(c) < 0) {
                System.out.println(sb.toString());
            } else {
                System.out.println("---");
            }
        }
    }
}
