#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s, c;
    cin >> s >> c;
    for (int i = 0; i < s.size(); i++) {
      for (int j = i + 1; j < s.size(); j++) {
        swap(s[i], s[j]);
        if (s < c) {
          cout << s << endl;
          goto END;
        }
        swap(s[i], s[j]);
      }
    }
    cout << "---" << endl;
  END:;
  }
  return 0;
}
