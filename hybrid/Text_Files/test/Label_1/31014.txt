import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] c = new long[n - 1];
        long[] s = new long[n - 1];
        long[] f = new long[n - 1];
        for (int i = 0; i < n - 1; i++) {
            c[i] = sc.nextInt();
            s[i] = sc.nextInt();
            f[i] = sc.nextInt();
        }
        long[] time = new long[n];
        time[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            time[i] = time[i + 1] + c[i] + (s[i] + f[i] - 1) / f[i] * f[i];
        }
        for (int i = 0; i < n; i++) {
            System.out.println(time[i]);
        }
        sc.close();
    }
}
