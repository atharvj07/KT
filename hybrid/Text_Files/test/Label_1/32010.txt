import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.*;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

public class Main {
    static class Edge {
        int to, w;

        public Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    static int n, m;
    static int[] d;
    static List<List<Edge>> adjList = new ArrayList<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        d = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            d[i] = sc.nextInt();
        }
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adjList.get(u).add(new Edge(v, 0));
            adjList.get(v).add(new Edge(u, 0));
        }
        int[] color = new int[n + 1];
        int[] w = new int[m + 1];
        if (!dfs(1, 1, color, w)) {
            System.out.println(-1);
            return;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i <= n; i++) {
            sb.append(color[i] == 1 ? "W" : "B");
        }
        sb.append("\n");
        for (int i = 1; i <= m; i++) {
            sb.append(w[i]).append("\n");
        }
        System.out.print(sb);
    }

    static boolean dfs(int u, int c, int[] color, int[] w) {
        color[u] = c;
        for (Edge e : adjList.get(u)) {
            if (color[e.to] == 0) {
                if (!dfs(e.to, c ^ 1, color, w)) {
                    return false;
                }
                w[e.to] = d[u];
            } else if (color[e.to] == c) {
                if (d[u] != d[e.to]) {
                    return false;
                }
            } else {
                if (d[u] != w[e.to]) {
                    return false;
                }
            }
        }
        return true;
    }
}
