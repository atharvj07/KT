import sys
from collections import deque

n, k, d = map(int, input().split())
p = list(map(int, input().split()))
u, v = [[] for _ in range(n + 1)], [[] for _ in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    u[a].append(b)
    v[b].append(a)

lev = [-1] * (n + 1)
que = deque(p)
while que:
    now = que.popleft()
    if lev[now] != -1:
        continue
    lev[now] = 0
    for nxt in u[now] + v[now]:
        if lev[nxt] != -1:
            continue
        que.append(nxt)
        lev[nxt] = lev[now] + 1

tot = 0
for x in range(1, n + 1):
    if lev[x] <= d:
        tot += 1

ans = []
que2 = deque([1])
while que2:
    now = que2.popleft()
    if now == 1 and len(ans) == tot:
        break
    if lev[now] == d + 1:
        for to in u[now] + v[now]:
            if lev[to] == lev[now] - 1:
                ans.append(now)
                que2.append(to)
                break

print(len(ans))
print(*ans)
