#include<bits/stdc++.h>
using namespace std;
int main() {
	int n, t = 0, mx = 0, ans[1005];
	cin >> n;
	for (int i = 1; i <= n; i++) {
		int x; cin >> x; mx = max(mx, x);
	}
	for (int i = 1; i <= mx; i++) {
		int cnt = 0;
		for (int j = i; j <= n; j += (i + 1))if (ans[j] == 0 && j <= n) {
			cnt++; ans[j] = 1;
		}
		if (cnt)t++;
	}
	cout << t << endl;
	for (int i = 1; i <= mx; i++) {
		int cnt = 0;
		for (int j = i; j <= n; j += i + 1)if (ans[j] == 1)cnt++;
		if (cnt)cout << cnt << " ";
	}
	cout << endl;
}
