import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n + 1];
        boolean ok = true;
        for (int i = 0; i < m; i++) {
            int t = in.nextInt();
            int l = in.nextInt();
            int r = in.nextInt();
            if (t == 1) {
                int d = in.nextInt();
                for (int j = l; j <= r; j++) {
                    a[j] += d;
                    if (Math.abs(a[j]) > 1e9) {
                        ok = false;
                    }
                }
            } else {
                int mx = Integer.MIN_VALUE;
                for (int j = l; j <= r; j++) {
                    mx = Math.max(mx, a[j]);
                }
                if (mx != in.nextInt()) {
                    ok = false;
                }
            }
        }
        if (ok) {
            System.out.println("YES");
            for (int i = 1; i <= n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }
}
