import bisect

n = int(input())
a = list(map(int, input().split()))

b = sorted(a)
l, r = 0, 0
for i in range(1, n):
    if b[i] != a[i]:
        l = i
        break
for i in range(n - 1, -1, -1):
    if b[i] != a[i]:
        r = i
        break
if b[l - 1:r + 1] == a[r:l - 1:-1]:
    print("yes")
    print(l, r)
else:
    print("no")
