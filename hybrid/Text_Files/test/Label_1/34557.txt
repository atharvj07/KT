import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] arr = new String[n];
        for (int i = 0; i < n; i++) {
            arr[i] = br.readLine();
        }

        String res = "";
        int mx = 0;
        for (String s : arr) {
            int[] cnt = new int[26];
            for (int i = 0; i < s.length(); i++) {
                cnt[s.charAt(i) - 'a']++;
            }
            int mxf = 0;
            for (int i = 0; i < 26; i++) {
                mxf = Math.max(mxf, cnt[i]);
            }
            if (mxf > mx) {
                res = s;
                mx = mxf;
            }
        }
        if (res.isEmpty()) {
            res = "NO";
        } else {
            for (String s : arr) {
                if (!isSubstr(s, res, mx)) {
                    res = "NO";
                    break;
                }
            }
        }
        System.out.println(res);
    }

    static boolean isSubstr(String s, String t, int mx) {
        int[] cnt = new int[26];
        for (int i = 0; i < t.length(); i++) {
            cnt[t.charAt(i) - 'a']++;
        }
        int cur = 0;
        for (int i = 0; i < mx; i++) {
            if (cnt[t.charAt(i) - 'a'] == 1) {
                cur++;
            }
        }
        for (int i = mx; i < t.length(); i++) {
            if (cnt[t.charAt(i) - 'a'] == 1) {
                cur++;
            }
            if (cnt[t.charAt(i - mx) - 'a'] == 1) {
                cur--;
            }
            if (cur == 0) {
                return false;
            }
        }
        for (int i = 0; i < s.length(); i++) {
            cnt[s.charAt(i) - 'a']++;
            if (cnt[s.charAt(i) - 'a'] == mx) {
                int j = i;
                boolean ok = true;
                for (int k = 0; k < t.length() - 1; k++) {
                    j++;
                    if (j >= s.length()) {
                        j = 0;
                    }
                    if (s.charAt(j) != t.charAt(k)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    return true;
                }
            }
        }
        return false;
    }
}
