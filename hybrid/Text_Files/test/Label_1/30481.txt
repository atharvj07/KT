import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        long[] x = new long[n + 1];
        for (int i = 0; i < n; i++)
            x[i] = s.charAt(i) - '0';
        for (int i = 2; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                x[j - 1] = Math.abs(x[j - 1] - x[j]);
            }
        }
        System.out.println(x[0]);
    }
}
