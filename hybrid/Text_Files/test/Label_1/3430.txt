#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=5e5+5,mo=1e9+7;
int n,k,a[maxn];
ll dp[maxn][2][2];
int main(){
	cin>>n>>k;
	for(int i=1;i<=n;i++) cin>>a[i];
	dp[0][1][1]=0;
	for(int i=1;i<=n;i++){
		for(int j=0;j<=1;j++){
			if(j)
				for(int l=0;l<=1;l++){
					dp[i][j][l]=max(dp[i][j][l],dp[i-1][j][l]);
					dp[i][j][l]=max(dp[i][j][l],dp[i-1][j-l][!l]+a[i]);
				}
			else{
				dp[i][j][0]=max(dp[i][j][0],dp[i-1][j][0]);
				dp[i][j][0]=max(dp[i][j][0],dp[i-1][j][1]+a[i]);
			}
		}
		if(i<=k+1){
			for(int j=0;j<=1;j++){
				dp[i][j][1]=max(dp[i][j][1],dp[i-1][j][1]);
				dp[i][j][1]=max(dp[i][j][1],dp[i-1][j][0]+a[i]);
			}
		}
	}
	cout<<dp[n][1][1]<<endl;
	return 0;
}
