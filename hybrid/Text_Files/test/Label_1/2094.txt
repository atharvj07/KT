#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> b(n);
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  vector<int> a(n);
  int cur = 1;
  for (int i = n - 1; i >= 0; i--) {
    while (cur <= 9 && cur <= b[i]) {
      if (cur % 10 == b[i]) {
        a[i] = cur;
        break;
      }
      cur++;
    }
    cur++;
  }
  for (int i = 0; i < n; i++) { cout << a[i] << endl; }
}
