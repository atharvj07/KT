#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll m=1e9+7,f[300005],i,j,n,ans;
const ll maxn=3e5+5;
ll pow(ll x, ll y){
    ll res=1;
    while (y>0){
        if (y%2==1) res=(res*x)%m;
        x=(x*x)%m;
        y/=2;
    }
    return res;
}
int main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    cin>>n;
    for (i=1; i<=n; i++) {
        cin>>j;
        ans=ans+(f[j+1]*(i-f[j]+m))%m;
        ans%=m;
        f[j]=i;
    }
    cout<<ans<<endl;
}
