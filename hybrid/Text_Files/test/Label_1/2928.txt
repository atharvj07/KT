#include<bits/stdc++.h>
using namespace std;
vector<int> adj[31];
long long int pos[31][2];
bool vis[31];
bool ok=true;
void dfs(int node, long long int x, long long int y){
    if(vis[node]==true){
        return;
    }
    vis[node]=true;
    pos[node][0]=x;
    pos[node][1]=y;
    for(int i=0;i<adj[node].size();i++){
        if(vis[adj[node][i]]==false){
            if(x==y){
                dfs(adj[node][i], x+1, y);
            }else{
                dfs(adj[node][i], x, y+1);
            }
        }
    }
}
int32_t main() {
  int n;
  cin>>n;
  for(int i=1;i<n;i++){
      int u,v;
      cin>>u>>v;
      adj[u].push_back(v);
      adj[v].push_back(u);
  }
  dfs(1, 0, 0);
  for(int i=1;i<=n;i++){
      for(int j=1;j<=n;j++){
          if(i!=j){
              if(pos[i][0]==pos[j][0] && pos[i][1]==pos[j][1]){
                  ok=false;
              }
          }
      }
  }
  if(ok){
      cout<<"YES"<<endl;
      for(int i=1;i<=n;i++){
          cout<<pos[i][0]<<" "<<pos[i][1]<<endl;
      }
  }else{
      cout<<"NO"<<endl;
  }
  return 0;
}
