import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] count = new int[4][2];
        while (sc.hasNext()) {
            double l = sc.nextDouble(), r = sc.nextDouble();
            int li = (l <= 0.2) ? 3 : (l < 0.6) ? 2 : (l < 1.1) ? 1 : 0;
            int ri = (r <= 0.2) ? 3 : (r < 0.6) ? 2 : (r < 1.1) ? 1 : 0;
            count[li][0]++;
            count[ri][1]++;
        }
        sc.close();
        for (int i = 0; i < 4; i++)
            System.out.println(count[i][0] + " " + count[i][1]);
    }
}
