import java.util.Scanner;

public class MagicalBoxes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long maxSideLength = 0;
        for (int i = 0; i < n; i++) {
            int sideLength = scanner.nextInt();
            long count = scanner.nextLong();
            maxSideLength = Math.max(maxSideLength, sideLength * count);
        }
        int smallestBoxSideLength = 0;
        while (Math.pow(2, smallestBoxSideLength) < maxSideLength) {
            smallestBoxSideLength++;
        }
        System.out.println(smallestBoxSideLength);
    }
}
