import sys

input = sys.stdin.readlines()
n, k = map(int, input[0].split())
grid = [list(r) for r in input[1:]]
for i in range(n):
    for j in range(n):
        if grid[i][j] == 'W':
            continue
        for ii in range(max(i - k + 1, 0), min(i + k, n)):
            for jj in range(max(j - k + 1, 0), min(j + k, n)):
                grid[ii][jj] = 'W'

white_lines = n * 2
for i in range(n):
    white_lines -= 'B' in grid[i]
    white_lines -= 'B' in [row[i] for row in grid]
print(white_lines)
