from collections import defaultdict

def solve(n, m, edges):
    graph = defaultdict(list)
    for v, u, c in edges:
        graph[v].append((u, c))
    
    # Sort edges by character to ensure lexicographical order during traversal
    for node in graph:
        graph[node].sort(key=lambda x: x[1])
    
    # Memoization table to store results of subproblems
    # dp[position][prev_char][turn] = can the current player win from this state?
    dp = [[[None for _ in range(2)] for _ in range(26)] for _ in range(n + 1)]

    def can_win(pos, prev_char, turn):
        if dp[pos][prev_char][turn] is not None:
            return dp[pos][prev_char][turn]
        
        for next_pos, char in graph[pos]:
            # Convert char to index 0-25 (a-z)
            char_idx = ord(char) - ord('a')
            # If the character is lexicographically equal or greater
            if char_idx >= prev_char:
                # If the opponent cannot win from the next position, current player wins
                if not can_win(next_pos, char_idx, 1 - turn):
                    dp[pos][prev_char][turn] = True
                    return True
        
        # If no move leads to a win, the current player loses
        dp[pos][prev_char][turn] = False
        return False
    
    # Build results for each initial position
    results = [['B' for _ in range(n)] for _ in range(n)]
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            if i != j and can_win(i, 0, 0):
                results[i - 1][j - 1] = 'A'
    
    return '\n'.join([''.join(row) for row in results])

# Input processing
n, m = map(int, input().split())
edges = [list(input().split()) for _ in range(m)]
for i in range(m):
    edges[i][0], edges[i][1] = int(edges[i][0]), int(edges[i][1])

# Solve the problem
print(solve(n, m, edges))
