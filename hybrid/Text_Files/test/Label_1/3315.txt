#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main(){
    int tc;
    cin >> tc;
    for (int i = 1; i <= tc; i++){
        int n;
        cin >> n;
        unordered_map<int, int> mp;
        for (int j = 1; j <= n; j++){
            int x;
            cin >> x;
            mp[x]++;
        }
        int ind;
        for (int j = 1; j <= n; j++){
            if (mp[j] == 1){
                ind = j;
            }
        }
        for (int j = 1; j <= n; j++){
            if (ind != j){
                cout << j << endl;
                break;
            }
        }
    }
}
