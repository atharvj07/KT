import sys
import math
from collections import defaultdict

def rceil(a, b):
    return (a + b - 1) // b


n = int(input())
graph = defaultdict(set)
for _ in range(n - 1):
    a, b = map(int, input().split())
    graph[a].add(b)
    graph[b].add(a)

p = [0] * n
p[0] = 1

for i in range(1, n):
    for j in graph[i]:
        if j < i:
            p[i] += p[j] / 2
    p[i] /= len(graph[i])


for x in p:
    print(x)
