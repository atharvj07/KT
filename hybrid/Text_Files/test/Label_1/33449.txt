import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long c = sc.nextLong();

        if ((a + b + c) % 3 != 0) {
            System.out.println(-1);
            return;
        }

        long count = 0;
        while (a % 2 == 0 && b % 2 == 0 && c % 2 == 0) {
            a /= 2;
            b /= 2;
            c /= 2;
            count++;
        }

        System.out.println(count);
    }
}
