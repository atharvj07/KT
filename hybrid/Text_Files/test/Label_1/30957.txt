import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int m = scanner.nextInt();
    long[] v = new long[n];
    for (int i = 0; i < n; i++) {
      v[i] = scanner.nextLong();
    }
    int w_a = 0, w_b = 0, w_f = 0, w_s = 0;
    for (int a = 1; a <= m; a++) {
      for (int b = 1; b <= m; b++) {
        boolean AliceWins = true, BobWins = true;
        for (int i = 0; i < n; i++) {
          AliceWins &= v[i] >= a;
          BobWins &= v[i] >= b;
        }
        if (AliceWins && BobWins) {
          if (a == b) {
            w_a++;
          } else {
            w_b++;
          }
        } else if (AliceWins) {
          w_f++;
        } else if (BobWins) {
          w_s++;
        }
      }
    }
    System.out.println(w_a + " " + w_b + " " + w_f + " " + w_s);
  }
}
