import sys

MX = sys.maxsize

n = int(input())
c1 = list(map(int, input().split()))
c2 = list(map(int, input().split()))
c3 = list(map(int, input().split()))
g = [[] for i in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)

par = [0] * (n + 1)
col = [0] * (n + 1)


def dfs(u, p):
    mn = [MX, MX, MX]
    for v in g[u]:
        if v == p:
            continue
        dfs(v, u)
        mn = [min(mn[0], mn[1] + c1[v - 1]), min(mn[1], mn[2] + c2[v - 1]), min(mn[2], mn[0] + c3[v - 1])]
    col[u] = mn.index(min(mn)) + 1
    par[u] = p


dfs(1, 1)

if len(set(col)) < 3:
    print(-1)
else:
    print(min(c1[0], c2[0], c3[0]) + sum(mn) - min(mn))
    print(*col)
