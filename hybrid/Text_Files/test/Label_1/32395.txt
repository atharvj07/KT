import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String line;
        while (!(line = sc.nextLine()).equals("0")) {
            int l = Integer.parseInt(line);
            long[] m = new long[12];
            long[] n = new long[12];
            for (int i = 0; i < 12; i++) {
                line = sc.nextLine();
                String[] s = line.split(" ");
                m[i] = Long.parseLong(s[0]);
                n[i] = Long.parseLong(s[1]);
            }
            long savings = 0;
            int months = 0;
            for (int i = 0; i < 12; i++) {
                savings += m[i] - n[i];
                months++;
                if (savings >= l) {
                    System.out.println(months);
                    break;
                }
            }
            if (months == 12) {
                System.out.println("NA");
            }
        }
    }
}
