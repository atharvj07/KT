#include <bits/stdc++.h>

using namespace std;

string solve(string a, string b) {
    int len = a.length();
    if (a == b) return a;
    int dif = b.length() - a.length();
    int idx = -1;
    for (int i = 0; i < len; i++) {
        char c = a[i];
        bool found = false;
        for (int j = i; j < len; j++) {
            if (a[j] > c && !found) {
                c = a[j];
                idx = j;
                found = true;
            }
        }
        if (found) {
            char tmp = a[i];
            a[i] = c;
            a[idx] = tmp;
            if (a <= b) return a;
            else {
                for (int j = 0; j < dif; j++) a.pop_back();
                while (a[a.length() - 1] == '0') a.pop_back();
                return a;
            }
        }
    }
    return a;
}

int main() {
    string a, b;
    cin >> a >> b;
    cout << solve(a, b);
    return 0;
}
