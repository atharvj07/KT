#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 1, lst = -1;
  for (int i = 0; i < n; i++) {
    if (a[i] == -1) continue;
    if (lst != -1 && a[i] - lst != 1) ans++;
    lst = a[i];
  }
  cout << ans << endl;
  return 0;
}
