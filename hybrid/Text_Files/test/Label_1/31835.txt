import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        // Read the switch information
        int[][] switches = new int[m][];
        for (int i = 0; i < m; i++) {
            int k = sc.nextInt();
            switches[i] = new int[k];
            for (int j = 0; j < k; j++) {
                switches[i][j] = sc.nextInt();
            }
        }

        // Read the bulb information
        int[] bulbs = new int[m];
        for (int i = 0; i < m; i++) {
            bulbs[i] = sc.nextInt();
        }

        // Count the number of combinations
        int count = 0;
        for (int i = 0; i < (1 << n); i++) {
            boolean[] on = new boolean[n];
            for (int j = 0; j < n; j++) {
                on[j] = ((i >> j) & 1) == 1;
            }

            boolean ok = true;
            for (int j = 0; j < m; j++) {
                int cnt = 0;
                for (int k = 0; k < switches[j].length; k++) {
                    if (on[switches[j][k] - 1]) {
                        cnt++;
                    }
                }

                if (cnt % 2 != bulbs[j]) {
                    ok = false;
                    break;
                }
            }

            if (ok) {
                count++;
            }
        }

        System.out.println(count);
    }
}
