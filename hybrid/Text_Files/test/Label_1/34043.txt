import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int[] l = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
        }
        Arrays.sort(l);
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (i == j || j == k || i == k) {
                        continue;
                    }
                    int[] l2 = l.clone();
                    l2[i] = a;
                    l2[j] = b;
                    l2[k] = c;
                    Arrays.sort(l2);
                    int mp = 0;
                    for (int m = 0; m < n; m++) {
                        if (l2[m] < l[m]) {
                            mp += (l[m] - l2[m]);
                        } else if (l2[m] > l[m]) {
                            mp += ((l2[m] - l[m]) + 1) * 10;
                        }
                    }
                    ans = Math.min(ans, mp);
                }
            }
        }
        System.out.println(ans);
    }
}
