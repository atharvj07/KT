import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] x = new int[N];
        for(int i=0; i<N; i++){
            x[i] = sc.nextInt();
        }
        int L = sc.nextInt();
        int Q = sc.nextInt();
        for(int i=0; i<Q; i++){
            int a = sc.nextInt()-1;
            int b = sc.nextInt()-1;
            System.out.println((x[b]-x[a])/L + ((x[b]-x[a])%L==0 ? 0 : 1));
        }
    }
}
