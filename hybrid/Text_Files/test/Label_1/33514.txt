import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int p = sc.nextInt();
    int q = sc.nextInt();
    int r = sc.nextInt();
    int[] a = new int[n];
    long mx = 0;
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    for (int i = 0; i < n; i++) {
      for (int j = i; j < n; j++) {
        for (int k = j; k < n; k++) {
          mx = Math.max(mx, p * a[i] + q * a[j] + r * a[k]);
        }
      }
    }
    System.out.println(mx);
  }
}
