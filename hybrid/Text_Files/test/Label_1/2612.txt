#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int n = s.length(), m = t.length();
  vector<int> dp(n + 1);
  for (int i = 1; i <= n; i++) {
    for (int j = m; j >= 1; j--) {
      if (t[j - 1] == s[i - 1]) dp[i] = max(dp[i], dp[i - 1] + 1);
    }
  }
  int idx = n;
  string ans;
  for (int j = m; j >= 1; j--) {
    while (idx > 0 && dp[idx] != dp[idx - 1] + 1) idx--;
    ans += s[idx - 1];
    idx--;
  }
  cout << ans << endl;
  return 0;
}
