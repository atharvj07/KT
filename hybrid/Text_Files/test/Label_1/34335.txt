import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        for (int i = 0; i < 3; i++) {
            long sum = 0;
            int l = 0, r = 0;
            for (int j = 0; j < n; j++) {
                sum += a[j];
                if (sum == 0) {
                    l = j + 1;
                    r = j + 1;
                } else if (sum % (j + 1) == 0) {
                    l = 1;
                    r = j + 1;
                }
            }
            if (l == 0) {
                l = n;
                r = n;
            }
            System.out.println(l + " " + r);
            long b = -sum / (r - l + 1);
            for (int j = l - 1; j < r; j++) {
                System.out.print(b + " ");
            }
            System.out.println();
            for (int j = l - 1; j < r; j++) {
                a[j] += b;
            }
        }
    }
}
