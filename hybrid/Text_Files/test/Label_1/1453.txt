#include<bits/stdc++.h>
using namespace std;
int a[6000],b[30000],c[6000];
int main()
{
	int n,w,ans=0;
	cin>>n;
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
	}
	for(int i=1;i<=n;i++)
	{
		if(a[i]==1){b[++b[0]]=i;}
		if(a[i]==2){c[++c[0]]=i;}
	}
	w=min(b[0],min(c[0],n/3));
	cout<<w<<'\n';
	for(int i=1;i<=w;i++)
	{
		cout<<b[i]<<' '<<c[i]<<' '<<b[b[0]-i+1]<<'\n';
	}
	return 0;
}
