import java.util.*;

public class Main {
    static int[] a;
    static long[] sum;

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt(), m = s.nextInt();
        a = new int[n + 1];
        sum = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = s.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + a[i];
        }
        while (m-- > 0) {
            int type = s.nextInt();
            int l = s.nextInt(), r = s.nextInt(), x = s.nextInt();
            if (type == 1) {
                for (int i = l; i <= r && a[i] > x; i++) {
                    a[i] -= x;
                    sum[i] -= x;
                }
            } else {
                System.out.println(sum[r] - sum[l - 1]);
            }
        }
    }
}
