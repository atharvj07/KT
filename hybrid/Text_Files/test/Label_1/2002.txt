#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e9+7;
bool solve(int n) {
    if(n%2 == 0) {
        cout << "YES\n";
        cout << "1\n";
        cout << n/2 << " " << n << endl;
        return true;
    }
    if(n%3 == 0) {
        cout << "YES\n";
        cout << "2\n";
        cout << n/3 << " " << n << endl;
        cout << (n/3) * 2 << " " << n << endl;
        return true;
    }
    for(int d = 4; d*d <= n; d++) {
        if(n%d == 0) {
            int x = n/d;
            if(x%2 == 0 || x%3 == 0) {
                cout << "YES\n";
                cout << "3\n";
                cout << n/x << " " << n << endl;
                if(x%2 == 0) cout << n/2 << " " << n << endl;
                else cout << (n/3) * 2 << " " << n << endl;
                cout << x - n/x - ((x%2 == 0 || x%3 == 0) ? 1 : 0) << " " << n << endl;
                return true;
            }
        }
    }
    cout << "NO\n";
    return false;
}
int main() {
    int n; cin >> n;
    solve(n);
}
