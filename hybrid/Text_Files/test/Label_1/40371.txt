import sys
MAX_A = 10**12
MAX_N = 10**5

s = set()
a = list(map(lambda x: int(x), sys.stdin.readline().split()))

for i, ai in enumerate(a):
    k = 0
    p = 1
    while p <= ai:
        s.add(ai - p)
        k += 1
        p *= 2
    for j in range(i + 1, len(a)):
        p = 1
        while p * (k + 1) <= a[j]:
            s.add(a[j] - p * (k + 1))
            p *= 2
if len(s) == 0:
    print(-1)
else:
    mn = min(s)
    mx = max(s)
    for i in range(mn, mx + 1):
        if i not in s:
            print(i - 1)
            exit(0)
    print(mx)
