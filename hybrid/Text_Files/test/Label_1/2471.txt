#include <iostream>

int main() {
    long long n, a, b, p, q;
    std::cin >> n >> a >> b >> p >> q;
    long long ans = ((n / a) + (n / b) - (n / (a * b))) * std::min(p, q);
    if (a < b) {
        ans += n % a == 0 ? p : q;
    } else if (a > b) {
        ans += n % b == 0 ? q : p;
    } else {
        ans += (n % a == 0 && n % b == 0) ? 0 : std::min(p, q);
    }
    std::cout << ans << std::endl;
}
