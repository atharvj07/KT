#include <cstdio>
#include <vector>
using namespace std;
const int MAXN = 250005;
#define ll long long 

ll n, m;
int p[MAXN];

int main() {
	scanf("%lld%lld", &n, &m);
	for (int i = 1; i <= n; i++) p[i] = i;
	ll ans = 0;
	for (int i = 1; i <= n; i++) ans += (i * (n - i + 1) % m) * (n - i + 1) % m;
	ans %= m;
	printf("%lld\n", ans);
	return 0;
}
