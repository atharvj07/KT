import sys
n, t = map(int, input().split())
a, ti = [], []
for _ in range(n):
    aa, tt = map(int, input().split())
    a.append(aa)
    ti.append(tt)

a.sort(reverse=True)
ti.sort()
i = 0
ans = 0
time = 0
l = []
while i < n and time + ti[i] <= t:
    if a[i] >= i + 1:
        l.append(i + 1)
        ans += 1
        time += ti[i]
    i += 1
print(ans)
print(len(l))
print(*l)
