#include <bits/stdc++.h>

const int N = 1e6+5;
int g[N], h[N];

bool is_same(int x, int y) {
  return g[x] == g[y] && h[x] == h[y];
}

int main() {
  int n, m, q;
  scanf("%d%d", &n, &m);
  char s[N];
  for (int i = 0; i < n; i++) {
    scanf("%s", s);
    for (int j = 0; j < m; j++) {
      if(s[j]=='.') g[j]++, h[i]++;
    }
  }
  scanf("%d", &q);
  while (q--) {
    int l, r;
    scanf("%d%d", &l, &r);
    l--;
    int ansl = g[l], ansr = g[r];
    int anslu = h[0], ansld = h[n - 1];
    for (int i = l; i < r; i++) {
      ansl += (s[i] == '.');
      ansr += (s[i] == '.');
    }
    if (is_same(ansl, anslu) && is_same(ansr, ansld))
      printf("YES\n");
    else
      printf("NO\n");
  }
  return 0;
}
