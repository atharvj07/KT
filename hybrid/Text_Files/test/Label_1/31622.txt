import java.util.*;
public class Main {
    static class Bus implements Comparable<Bus> {
        int id, l, r, time;
        public Bus(int id, int l, int r, int time) {
            this.id = id;
            this.l = l;
            this.r = r;
            this.time = time;
        }
        @Override
        public int compareTo(Bus o) {
            return this.time - o.time;
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Bus[] b = new Bus[n];
        for(int i = 0; i < n; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int time = sc.nextInt();
            b[i] = new Bus(i + 1, l, r, time);
        }
        Arrays.sort(b);
        int[] ans = new int[m];
        for(int i = 0; i < m; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int time = sc.nextInt();
            int bus = -1;
            for(int j = 0; j < n; j++) {
                if(l >= b[j].l && r <= b[j].r && time <= b[j].time) {
                    bus = b[j].id;
                    break;
                }
            }
            ans[i] = bus;
        }
        for(int a : ans) System.out.print(a + " ");
    }
}
