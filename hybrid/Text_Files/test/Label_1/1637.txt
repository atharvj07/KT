#include <map>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;

  std::map<int, int> counts;
  for (int i = 0; i < n; i++) {
    int t;
    std::cin >> t;
    counts[t]++;
  }

  int w = std::min(counts[1], std::min(counts[2], counts[3]));
  std::cout << w << "\n";

  for (int i = 0; i < w; i++) {
    for (int j = 1; j <= 3; j++) {
      if (counts[j] > 0) {
        std::cout << i * 3 + j << " ";
        counts[j]--;
      }
    }
    std::cout << "\n";
  }
  return 0;
}
