import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Map<String, Integer> cnt = new HashMap<>();
        int[] top = new int[n];
        Set<String> s = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String cur = in.next();
            if (s.contains(cur)) {
                System.out.println("NO");
                return;
            }
            s.add(cur);
            top[i] = cnt.getOrDefault(cur, 0);
            cnt.put(cur, cnt.getOrDefault(cur, 0) + 1);
        }
        for (int i = n - 1; i >= 0; i--) {
            if (top[i] == 0) {
                continue;
            }
            int cur = top[i];
            for (int j = i - 1; j >= 0; j--) {
                if (top[j] == 0) {
                    top[j] = cur;
                    cnt.put(s.iterator().next(), cnt.get(s.iterator().next()) - 1);
                    break;
                }
                if (s.contains(s.iterator().next() + top[j])) {
                    top[j] = cur;
                    cnt.put(s.iterator().next(), cnt.get(s.iterator().next()) - 1);
                    break;
                }
            }
            top[i] = 0;
        }
        for (int i = 0; i < n; i++) {
            if (top[i] > 0) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
