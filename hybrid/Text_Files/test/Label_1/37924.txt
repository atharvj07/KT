from collections import deque
import sys

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = set(map(int, input().split())) if k else set()

if a[0] > a[1]:
    print(-1)
    sys.exit()

d = deque()
d.append(0)
ans = 0

for i in range(1, n):
    if i in b:
        continue

    while d and a[d[-1]] > a[i]:
        d.pop()
        ans += 1

    d.append(i)

if len(d) == n:
    print(ans)
else:
    print(-1)
