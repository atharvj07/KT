from queue import PriorityQueue
import sys

getline = sys.stdin.readline

def dijkstra(st, n, dist, adj):
    q = PriorityQueue()
    q.put((0, st))
    dist[st] = 0
    while not q.empty():
        w, u = q.get()
        if w > dist[u] : continue
        for v, w_ in adj[u]:
            if w_ + w < dist[v]:
                dist[v] = w_ + w
                q.put((dist[v], v))

for t in range(int(getline())):
    n = int(getline())
    dist = [1e9] * (n + 1)
    adj = [[] for i in range(n + 1)]
    for i in range(n - 1):
        u, v = map(int, getline().split())
        adj[u].append((v, 1))
        adj[v].append((u, 1))
    dijkstra(1, n, dist, adj)
    print(dist[n])
