#include<bits/stdc++.h>
using namespace std;
const int N=55;
int s,n,k,a[N],b[N],ans;
char c[N];
bool eat(int x,int col){
    if(a[x]==0)return 0;
    if(a[x]<k)return 0;
    if(a[x]>b[x])return 0;
    if(c[x]==c[x-1]&&x>s)return 0;
    b[x]=a[x];
    a[x]=0;
    if(x==s){
        k-=a[x];
        if(k<=0)return 1;
    }
    if(eat(x+1,c[x]))return 1;
    if(eat(x-1,c[x]))return 1;
    return 0;
}
int main(){
    cin>>n>>s>>k;
    for(int i=1;i<=n;i++)cin>>a[i];
    cin>>c;
    if(eat(s,-1))cout<<ans<<endl;
    else cout<<-1<<endl;
    return 0;
}
