#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;

#define PB push_back

const int MAXN = 10005;
vector<int> V[MAXN];

int main() {
    int n, m;
    scanf(" %d %d ", &n, &m);
    for (int i = 1; i <= m; i++) {
        int k; scanf(" %d ", &k);
        for (int j = 1; j <= k; j++) {
            int x; scanf(" %d ", &x);
            V[i].PB(x);
        }
    }
    for (int i = 1; i <= m; i++) {
        for (int j = 0; j < V[i].size(); j++) {
            int x = V[i][j];
            if (x < 0) x = -x;
            for (int k = j + 1; k < V[i].size(); k++) {
                int y = V[i][k];
                if (y < 0) y = -y;
                if (x == y) {
                    printf("YES\n");
                    return 0;
                }
            }
        }
    }
    printf("NO\n");
    return 0;
}
