#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 6;
vector<int> adj[N];
int par[N];
bool vis[N];
int n;

void mark_parent(int node, int p) {
  vis[node] = true;
  par[node] = p;
  for (int next : adj[node]) {
    if (!vis[next]) {
      mark_parent(next, node);
    }
  }
}
int main() {
  cin >> n;
  for (int i = 0; i < n - 1; ++i) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  mark_parent(1, 1);
  set<pair<int, int>> edges;
  for (int i = 2; i <= n; i++) {
    edges.insert({par[i], i});
  }
  int sz = edges.size();
  if (((sz & 1) && n % 2 == 0) || ((n & 1) && sz % 2 == 0)) {
    cout << "NO\n";
  } else {
    cout << "YES\n";
  }
}
