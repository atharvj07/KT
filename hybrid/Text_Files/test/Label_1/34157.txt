import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        if (B > A) {
            int temp = A;
            A = B;
            B = temp;
        }
        int gcd = 1;
        for (int i = 2; i <= B; i++) {
            if (A % i == 0 && B % i == 0) {
                gcd *= i;
                A /= i;
                B /= i;
            }
        }
        System.out.println(gcd);
    }
}
