#include <iostream>
#include <vector>
using namespace std;
int p[101];
int main() {
  int n, k;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int l, r;
    cin >> l >> r;
    p[i] = r;
  }
  cin >> k;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (p[i] < k) cnt++;
  }
  cout << cnt << endl;
  return 0;
}
