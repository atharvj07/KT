import java.util.*;
class Main {
    static boolean canWin(int r, int n) {
        return n % 2 == 0 || (r == n && n % 2 == 1);
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int n = sc.nextInt();
        System.out.println(canWin(r, n) ? "WIN" : "LOSE");
    }
}
