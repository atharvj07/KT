import java.io.*;
import java.util.*;

public class Main {
    static int[] a, p;
    static boolean[] vis;

    public static void main(String[] args) throws IOException {
        Reader in = new InputStreamReader(System.in);
        StreamTokenizer tok = new StreamTokenizer(in);
        tok.nextToken();
        int n = (int) tok.nval;
        tok.nextToken();
        int q = (int) tok.nval;
        a = new int[n + 1];
        p = new int[n + 1];
        vis = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            tok.nextToken();
            a[i] = (int) tok.nval;
        }
        for (int i = 1; i <= q; i++) {
            tok.nextToken();
            int x = (int) tok.nval;
            tok.nextToken();
            int y = (int) tok.nval;
            System.out.println(check(x, y) ? "Shi" : "Fou");
        }
    }

    static boolean check(int x, int y) {
        if (x == y)
            return true;
        if (a[y] == 0)
            return false;
        if (vis[y])
            return p[y] == x;
        vis[y] = true;
        p[y] = x;
        return (a[x] & a[y]) > 0 && check(x, y - 1);
    }
}
