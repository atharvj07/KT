import java.io.*;
import java.util.*;
import java.math.BigInteger;
import java.text.DecimalFormat;

public class Main {
  private static final long MOD = BigInteger.valueOf(1000000007).longValue();
  private static final double DELTA = 1e-6;

  private static boolean[][] graph;
  private static long[] color;
  private static long[] weight;
  private static long[] dp;

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] parts = br.readLine().split(" ");
    int n = Integer.parseInt(parts[0]);
    int m = Integer.parseInt(parts[1]);
    long x = Long.parseLong(br.readLine());

    graph = new boolean[n + 1][n + 1];
    color = new long[n + 1];
    weight = new long[n + 1];
    dp = new long[1 << n];

    for (int i = 0; i < m; i++) {
      parts = br.readLine().split(" ");
      int u = Integer.parseInt(parts[0]);
      int v = Integer.parseInt(parts[1]);
      long w = Long.parseLong(parts[2]);
      graph[u][v] = true;
      graph[v][u] = true;
      weight[u] += w;
      weight[v] += w;
    }

    dfs(1, x, 0, 0, 0);

    System.out.println(dp[dp.length - 1]);
  }

  private static void dfs(int u, long x, long curWeight, long curMask, long curCount) {
    long nextWeight = curWeight + weight[u];
    long nextMask = curMask | (1 << (u - 1));

    if (nextWeight == x && curCount > 0 && curCount < dp.length) {
      dp[(int) curCount] = (dp[(int) curCount] + 1) % MOD;
    }

    for (int v = u + 1; v <= graph.length - 1; v++) {
      if (graph[u][v] && (curMask & (1 << (v - 1))) == 0) {
        dfs(v, x, nextWeight, nextMask, curCount + 1);
      }
    }
  }
}
