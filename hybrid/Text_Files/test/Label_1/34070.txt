import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int[][] data = new int[n][3];
        for (int i = 0; i < n; i++) {
            data[i][0] = scanner.nextInt();
            data[i][1] = scanner.nextInt();
            data[i][2] = scanner.nextInt();
        }

        String program = "";
        int x = 0;
        int y = 0;
        boolean found = false;
        for (int i = 0; i < l; i++) {
            for (int j = 0; j < n; j++) {
                int time = data[j][0];
                int x1 = data[j][1];
                int y1 = data[j][2];
                if (time % l != i) {
                    continue;
                }
                if (x == x1 && y == y1) {
                    continue;
                }
                if (x == x1) {
                    if (y1 > y) {
                        program += "D";
                        y++;
                    } else {
                        program += "U";
                        y--;
                    }
                } else if (y == y1) {
                    if (x1 > x) {
                        program += "R";
                        x++;
                    } else {
                        program += "L";
                        x--;
                    }
                } else {
                    found = true;
                    break;
                }
            }
            if (found) {
                break;
            }
        }

        if (found) {
            System.out.println("NO");
        } else {
            System.out.println(program);
        }
    }
}
