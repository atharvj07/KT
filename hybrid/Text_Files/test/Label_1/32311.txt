import java.util.Scanner;

public class LuckyArray {
    private static final int MOD = 777777777;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        boolean[][] isGood = new boolean[4][4];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                isGood[i + 1][j + 1] = scanner.nextInt() == 1;
            }
        }

        int[] a = new int[n + 1];
        for (int i = 1; i <= m; i++) {
            int v = scanner.nextInt();
            int t = scanner.nextInt();
            a[v] = t;

            long count = countLuckyArrays(a, isGood);
            System.out.println(count % MOD);
        }
    }

    private static long countLuckyArrays(int[] a, boolean[][] isGood) {
        int n = a.length - 1;
        long[][] dp = new long[n + 1][4];

        dp[n][0] = 1; // Base case: empty array is always lucky
        for (int i = n - 1; i >= 1; i--) {
            if (a[i] == 0) {
                for (int j = 1; j <= 3; j++) {
                    for (int k = 1; k <= 3; k++) {
                        if (isGood[j][k]) {
                            dp[i][j] = (dp[i][j] + dp[i + 1][k]) % MOD;
                        }
                    }
                }
            } else {
                int j = a[i];
                for (int k = 1; k <= 3; k++) {
                    if (isGood[j][k]) {
                        dp[i][j] = (dp[i][j] + dp[i + 1][k]) % MOD;
                    }
                }
            }
        }

        long count = 0;
        for (int j = 1; j <= 3; j++) {
            count = (count + dp[1][j]) % MOD;
        }

        return count;
    }
}
