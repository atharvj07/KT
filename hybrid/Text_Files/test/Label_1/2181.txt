#include <vector>
#include <iostream>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<long long> a(n);
    for (int i = 0; i < n; ++i)
        std::cin >> a[i];
    long long g = *std::min_element(a.begin(), a.end());
    int ans = 1;
    for (int i = 0; i < n; ++i)
        if (a[i] == g) {
            ans = i + 1;
            break;
        }
    std::cout << ans << '\n';
    return 0;
}
