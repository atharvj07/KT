import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long N = input.nextInt();
        HashMap<Long, Long> hm = new HashMap<>();
        for (long i = 0; i < N; i++) {
            long a = input.nextInt();
            long old = 0;
            if (hm.containsKey(a)) {
                old = hm.get(a);
            }
            hm.put(a, old + 1);
        }
        long ans = 0;
        for (long value : hm.values()) {
            if (value < 2) {
                ans += value;
            } else {
                ans += value - value / 2;
            }
        }
        System.out.println(ans);
    }
}
