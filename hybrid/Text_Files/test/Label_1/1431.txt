#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;

int32_t main() {
	ll p, k;
	cin >> p >> k;
	if (p == 0) {
		cout << "1\n0" << endl;
		return 0;
	}
	vector<ll> pre(k);
	vector<ll> sum(k, 0);
	pre[0] = 1;
	for (int i = 1; i < k; i++) pre[i] = (pre[i - 1] * i) % MOD;
	for (int i = 1; i < k; i++) sum[i] = (sum[i - 1] + pre[i]) % MOD;
	ll sum_inv = p % MOD;
	sum_inv = (sum_inv * k) % MOD;
	ll cur = 0;
	for (int i = k - 1; i > 0; i--) {
		ll val = (sum_inv - cur) * pre[k - i - 1] % MOD;
		val = (val * sum[i]) % MOD;
		if (val <= 0) val += MOD;
		cur = (cur + val) % MOD;
		sum_inv = (sum_inv - val) % MOD;
		if (sum_inv < 0) sum_inv += MOD;
	}
	if (sum_inv != 0) {
		cout << "-1" << endl;
		return 0;
	}
	int d = 0;
	while (cur != 0) {
		if (cur % k != 0) break;
		cur /= k;
		d++;
	}
	cout << d << endl;
	for (int i = d - 1; i >= 0; i--) {
		cout << cur % k << " ";
		cur /= k;
	}
	cout << endl;
}
