#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, d, sum = 0;
  cin >> n >> d;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i], sum += a[i];
  sort(a.begin(), a.end());
  cin >> n;
  for (int i = 0; i < n; i++)
    if (a[i] <= d) d -= a[i], sum -= a[i];
  cout << sum << endl;
}
