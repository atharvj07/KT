import sys

def solve():
    t = int(input())
    for _ in range(t):
        n, m = map(int, input().split())

        a = set()
        for _ in range(n):
            a.add(input())

        all = 1 << m
        k = all - n

        lst = []
        for i in range(all):
            s = bin(i)[2:]
            s = '0' * (m - len(s)) + s
            if s not in a:
                lst.append(s)

        lst.sort()
        print(lst[k // 2])

solve()
