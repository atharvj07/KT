#include <iostream>
#include <string>
#include <vector>
#include <map>

int main() {
  int n, t;
  std::cin >> n >> t;
  std::string s1, s2;
  std::cin >> s1 >> s2;
  if (t > n) {
    std::cout << "-1" << std::endl;
    return 0;
  }
  std::map<char, int> m1, m2;
  for (int i = 0; i < n; i++) {
    m1[s1[i]]++;
    m2[s2[i]]++;
  }
  if (m1 != m2) {
    std::cout << "-1" << std::endl;
    return 0;
  }
  std::vector<char> diff;
  for (auto it : m1) {
    if (it.second != t) {
      diff.push_back(it.first);
    }
  }
  std::string s3;
  for (int i = 0; i < n; i++) {
    if (s1[i] != s2[i]) {
      s3 += diff[0];
      diff.erase(diff.begin());
    } else {
      s3 += s1[i];
    }
  }
  std::cout << s3 << std::endl;
  return 0;
}
