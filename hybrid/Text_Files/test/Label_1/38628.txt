mod = 1000000007

def dfs(i, j, ways):
    if (i, j) == (h - 1, w - 1):
        return 1
    if (i, j) in ways:
        return ways[(i, j)]

    ans = 0
    if i + 1 < h and (i + 1, j) not in black:
        ans += dfs(i + 1, j, ways)
    if j + 1 < w and (i, j + 1) not in black:
        ans += dfs(i, j + 1, ways)
    ways[(i, j)] = ans % mod
    return ways[(i, j)]

h, w, n = map(int, input().split())
black = set()
for _ in range(n):
    x, y = map(int, input().split())
    black.add((x - 1, y - 1))

ways = {}
print(dfs(0, 0, ways))
