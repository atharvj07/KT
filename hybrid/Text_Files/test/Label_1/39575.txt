import sys
input = sys.stdin.readline

n = int(input())

sites = {}
for _ in range(n):
    split = input().strip('http://').split('/')
    hostname = split[0]
    path = ''
    if len(split) > 1:
        path = split[1]
    if hostname not in sites:
        sites[hostname] = set()
    sites[hostname].add(path)

result = []

for hostname in sites:
    if len(sites[hostname]) > 1:
        for next_hostname in sites:
            if hostname in sites[next_hostname] and next_hostname in sites[hostname]:
                group = set()
                for path in sites[hostname]:
                    group.add(hostname + '/' + path)
                for path in sites[next_hostname]:
                    group.add(next_hostname + '/' + path)
                result.append(group)

print(len(result))
for group in result:
    print(*group, sep=' ')
