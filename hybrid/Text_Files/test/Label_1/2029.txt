#include <iostream>

using namespace std;

int main() {
  int a, b;
  cin >> a >> b;
  int win1 = 0, win2 = 0, draw = 0;
  for (int i = 1; i <= 6; i++) {
    if (abs(i - a) <= abs(i - b)) {
      if (abs(i - a) == abs(i - b)) draw++;
      else win1++;
    } else win2++;
  }
  cout << win1 << " " << draw << " " << win2 << endl;
  return 0;
}
