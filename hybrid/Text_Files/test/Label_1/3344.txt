#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, x;
    cin >> n >> x;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<int> count(100001, 0);
    for (int i = 0; i < n; i++) {
        count[a[i]]++;
    }

    long long ans = 0;
    for (int i = 0; i < n; i++) {
        ans += count[x ^ a[i]];
        if (a[i] == x) {
            ans--;
        }
    }

    cout << ans / 2 << endl;

    return 0;
}
