#include <iostream>
#include <map>
using namespace std;
typedef long long ll;

int main()
{
    ll l, r;
    cin >> l >> r;
    map<ll, ll>c;
    for(ll i = l; i <= r; i++){
        for(ll d = 2; d * d <= i; d++){
            if(i % d == 0){
                c[d]++;
            }
        }
    }
    for(ll i = l; i <= r; i++){
        c[i]++;
    }
    ll ans = 0, mx = 0;
    for(auto it = c.begin(); it != c.end(); it++){
        if(it->second > mx){
            mx = it->second;
            ans = it->first;
        }
    }
    cout << ans << endl;
}
