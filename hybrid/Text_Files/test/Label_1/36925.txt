MOD = 10**9 + 7

def mod_inv(a, mod=MOD):
    return pow(a, mod-2, mod)

def nCr_mod(n, r, mod=MOD):
    if r > n or n < 0 or r < 0: return 0
    numerator = 1
    for i in range(n, n-r, -1):
        numerator = (numerator * i) % mod
    denominator = 1
    for i in range(1, r+1):
        denominator = (denominator * i) % mod
    return numerator * mod_inv(denominator, mod) % mod

def solve(h, w, n, black_cells):
    black_cells.sort()
    black_cells = [(1, 1)] + black_cells + [(h, w)]
    dp = [0] * (n+2)
    dp[0] = 1
    for i in range(1, n+2):
        dp[i] = nCr_mod(black_cells[i][0]+black_cells[i][1]-2, black_cells[i][0]-1)
        for j in range(i):
            if black_cells[j][0] <= black_cells[i][0] and black_cells[j][1] <= black_cells[i][1]:
                dp[i] -= dp[j] * nCr_mod(black_cells[i][0]-black_cells[j][0]+black_cells[i][1]-black_cells[j][1], black_cells[i][0]-black_cells[j][0])
                dp[i] %= MOD
    return dp[-1]

h, w, n = map(int, input().split())
black_cells = [tuple(map(int, input().split())) for _ in range(n)]
print(solve(h, w, n, black_cells))
