import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n], f = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt() - 1;
            f[i] = in.nextInt() - 1;
        }
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                int j = p[i];
                List<Integer> c = new ArrayList<>();
                while (!vis[j]) {
                    vis[j] = true;
                    c.add(j);
                    j = p[j];
                }
                if (c.size() > 1) {
                    for (int k = 0; k < c.size(); k++) {
                        if (f[c.get(k)] != f[c.get((k + 1) % c.size())]) {
                            System.out.println("NO");
                            return;
                        }
                    }
                }
            }
        }
        System.out.println("YES");
    }
}
