#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<int> dp(n, 1e9);
    dp[0] = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] == -1)
            dp[i] = dp[i - 1];
        else
            for (int j = 0; j < i; j++) {
                if (a[j] == -1 || 2 * a[i] > a[j] + a[i])
                    dp[i] = min(dp[i], dp[j] + 1);
            }
    }

    cout << dp[n - 1] << endl;
}
