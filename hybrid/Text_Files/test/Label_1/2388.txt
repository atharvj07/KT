#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = k - 1;
  for (int i = 0; i < n; i++) {
    if (i > 0) ans -= max(0, k - a[i - 1] - a[i]);
    ans += max(0, k - a[i]);
    a[i] += max(0, k - a[i]);
  }
  cout << ans << endl;
  for (int i = 0; i < n; i++) cout << a[i] << ' ';
  cout << endl;
  return 0;
}
