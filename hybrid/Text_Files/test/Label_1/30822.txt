import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int r = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextInt();
        }
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[a[i] - 1] = i;
        }
        int[] d = new int[n];
        for (int i = l - 1; i < r; i++) {
            d[c[b[i] - 1]] = 1;
        }
        boolean flag = false;
        for (int i = l - 1; i < r; i++) {
            if (d[i] == 0) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println("LIE");
        } else {
            System.out.println("TRUTH");
        }
    }
}
