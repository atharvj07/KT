#include <bits/stdc++.h>
using namespace std;
const int MX = 5e5+5;
int n, a, b, t, ans;
char s[MX];
int main() {
    cin >> n >> a >> b >> t;
    cin >> s + 1;
    for (int i = 1, now = 1, rot = 0; i <= t; i++) {
        if (now == 1 && s[now] == s[n])
            rot = 0;
        if (now == n && s[now] == s[1])
            rot = 0;

        if (s[now] == 'w') {
            if (!rot) {
                ans++;
                t -= a;
                now += a < t ? a : t;
            }
            else {
                t -= rot;
                rot = 0;
            }
        }
        else {
            if (rot == 1) {
                ans++;
                t -= a;
                now += a < t ? a : t;
            }
            else if (rot == 0)
                rot = b;
        }
        now %= n;
        if (!now)
            now = n;
    }
    cout << ans << endl;
    return 0;
}
