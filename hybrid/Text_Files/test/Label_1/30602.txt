import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        int ans = -1;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (c[i] != a) {
                if (cnt == 0) {
                    ans = c[i];
                }
                cnt++;
            }
        }
        System.out.println(ans);
    }
}
