def max_palindromic_strings(Q, test_cases):
    results = []
    for n, strings in test_cases:
        odd_length_count = sum(len(s) % 2 for s in strings)
        if odd_length_count > 0:
            results.append(n)
        else:
            counts = [0, 0]  # [count of 0's, count of 1's]
            for s in strings:
                counts[0] += s.count('0')
                counts[1] += s.count('1')
            if counts[0] % 2 == 0 and counts[1] % 2 == 0:
                results.append(n)
            else:
                results.append(n - 1)
    return results

Q = int(input().strip())
test_cases = []
for _ in range(Q):
    n = int(input().strip())
    strings = [input().strip() for _ in range(n)]
    test_cases.append((n, strings))

results = max_palindromic_strings(Q, test_cases)
for result in results:
    print(result)
