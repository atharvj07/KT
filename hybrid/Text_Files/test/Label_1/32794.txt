import java.util.Scanner;

public class TreeChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        boolean possible = checkTree(c);
        System.out.println(possible ? "YES" : "NO");
    }

    private static boolean checkTree(int[] c) {
        int totalChildren = 0;
        for (int count : c) {
            totalChildren += count;
        }

        if (totalChildren != 2 * (c.length - 1)) {
            return false; // Total children must be twice the number of internal nodes
        }

        int[] childrenCounts = new int[c.length + 1];
        for (int count : c) {
            childrenCounts[count]++;
        }

        if (childrenCounts[0] != 1 || childrenCounts[1] == 0) {
            return false; // Must have a root and at least one internal node
        }

        return true;
    }
}
