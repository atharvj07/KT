import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class GraphAssignment {

    static class Edge {
        int from, to;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] p = new int[N];
        for (int i = 0; i < N; i++) {
            p[i] = sc.nextInt();
        }

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < N; i++) {
            graph.get(p[i] - 1).add(i);
        }

        boolean[] visited = new boolean[N];
        boolean possible = true;

        for (int i = 0; i < N; i++) {
            if (!visited[i]) {
                possible &= dfs(graph, visited, i, -1);
            }
        }

        System.out.println(possible ? "POSSIBLE" : "IMPOSSIBLE");
    }

    static boolean dfs(List<List<Integer>> graph, boolean[] visited, int node, int parent) {
        visited[node] = true;
        boolean possible = true;
        for (int next : graph.get(node)) {
            if (!visited[next]) {
                possible &= dfs(graph, visited, next, node);
            } else if (next != parent) {
                possible = false;
            }
        }
        return possible;
    }
}
