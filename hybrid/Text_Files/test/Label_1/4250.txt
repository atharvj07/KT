#include <bits/stdc++.h>
using namespace std;
int n, m, a[105], vis[105][105], ans;
inline int read() {
  int x;
  cin >> x;
  return x;
}
int gcd(int a, int b) {
  if (!a) return b;
  return gcd(b % a, a);
}
int main() {
  n = read(), m = read();
  for (int i = 1; i <= n; ++i) a[i] = read();
  for (int i = 1; i <= n; ++i)
    for (int j = i + 1; j <= n; ++j) {
      if ((i + j) & 1) {
        if (gcd(a[i], a[j]) > 1) {
          vis[i][j] = 1;
          vis[j][i] = 1;
        }
      }
    }
  for (int i = 1; i <= m; ++i) {
    int x = read(), y = read();
    if (vis[x][y]) {
      ++ans;
      vis[x][y] = 0;
    }
  }
  cout << ans << endl;
}
