from collections import deque, defaultdict

n = int(input())
a = list(map(int, input().split()))
g = defaultdict(list)
for i in range(n-1):
    u, v = map(int, input().split())
    u -= 1; v -= 1
    g[u].append(v)
    g[v].append(u)

q = deque([(0,-1)])
sub = [0] * n
vis = [False] * n
while q:
    u, p = q.popleft()
    vis[u] = True
    sub[u] = a[u]
    for v in g[u]:
        if v == p: continue
        q.append((v,u))
        sub[u] += sub[v]

ans = -1e9
for i in range(n):
    for j in g[i]:
        if vis[j]: continue
        ans = max(ans, sub[i] + sub[j])

print("Impossible" if ans == -1e9 else ans)
