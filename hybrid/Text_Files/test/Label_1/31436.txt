import java.util.*;
import java.lang.*;
import java.io.*;
class Main
{

	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner scan = new Scanner(System.in);
		int size = scan.nextInt();
		int x = scan.nextInt();
		int y = scan.nextInt();
		long expo = (long) Math.pow(10, x - y);
		int [] freq = new int[10];
		long num = scan.nextLong() % expo;
		while (num > 0) {
			freq[(int)(num % 10)]++;
			num /= 10;
		}
		int ans = size;
		for (int i = 1; i < x; i++) {
			int count = freq[i] + freq[0];
			if (count >= y) {
				ans = Math.min(ans, count - y);
			}
		}
		System.out.println(ans);
	}
}
