import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        int day = 1;
        for (int i = 0; i < n; i++) {
            while (day > a[i]) {
                day -= a[i];
            }
            if (day != a[i]) {
                ans++;
                day++;
            }
            day++;
        }
        System.out.println(ans);
    }
}
