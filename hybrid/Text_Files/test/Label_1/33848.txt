import java.util.*;

public class Main {
    static int[][] m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n * n];
        for (int i = 0; i < n * n; i++)
            a[i] = sc.nextInt();
        Arrays.sort(a);
        if (!check(n, a))
            System.out.println("NO");
        else {
            m = new int[n][n];
            int l = 0, r = n * n - 1, cx = 0, cy = n / 2;
            while (l <= r) {
                m[cx++][cy] = a[l++];
                m[cx--][cy] = a[r--];
                if (cx == n) {
                    cx = 0;
                    cy--;
                }
            }
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++)
                    System.out.print(m[i][j] + " ");
                System.out.println();
            }
        }
    }

    static boolean check(int n, int[] a) {
        if (n % 2 == 1) {
            int mid = n / 2 * n + n / 2;
            return a[mid] == a[mid + 1];
        } else {
            int mid = n / 2 * n + n / 2 - 1;
            return a[mid] == a[mid + 1];
        }
    }
}
