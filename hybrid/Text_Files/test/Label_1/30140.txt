import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] ai = new int[n];
        for (int i = 0; i < n; ++i) {
            ai[i] = scanner.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; ++i) {
            ans += ai[i];
        }
        System.out.println(ans);
    }
}
