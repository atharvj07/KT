#include<bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    int pre[n];
    int last[2] = {-1, -1};
    int now = 0;
    char ans = 'D';
    for (int i = 0; i < n; i++) {
        char c; cin >> c;
        int c_int = c=='R';
        if(c_int!=now) {
            last[now] = -1;
            now = c_int;
        }
        pre[i] = last[now];
        last[now] = i;
    }
    for(int i = n - 1; i >= 0; i--) {
        if(pre[i] == -1) ans = 'D';
        else {
            if(ans == 'D') {
                if(pre[i]==-1) ans = 'R';
            } else {
                if(pre[i]!=-1){
                    ans = 'D';
                }  
            }
        }
    }
    cout << ans << endl;
    return 0;
}
