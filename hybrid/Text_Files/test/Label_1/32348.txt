import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        long l = sc.nextInt();
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int ans = (int) Math.ceil((double) (x[b] - x[a]) / l);
            System.out.println(ans);
        }
    }
}
