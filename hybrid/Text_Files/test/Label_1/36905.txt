n, d, k = map(int, input().split())
if d%2 != n-1 or n-1 > d or k < n-1:
    print('NO')
else:
    print('YES')
    d -= n - 1
    if d&1:
        print(2,1)
        d -= 1
    for i in range(2, n):
        print(i+1,1)
    if d > 0:
        print(2,3,2,n)
    for i in range(n-1, 1, -1):
        print(i,i-1)
