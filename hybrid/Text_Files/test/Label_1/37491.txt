def solve():
    s = input()
    up = down = 0
    ans = 0
    for c in s:
        if c == 'U':
            up += 1
        elif c == 'D':
            up -= 1
            if up < 0:
                ans += 1
                up = 0
    for c in s:
        if c == 'D':
            down += 1
        elif c == 'U':
            down -= 1
            if down < 0:
                ans += 1
                down = 0
    print(ans)


solve()
