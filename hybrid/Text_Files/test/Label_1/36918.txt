import sys

n, m = map(int, input().split())
L = list(map(int, input().split()))
d = {int(i): [] for i in range(1, n + 1)}
for i in range(m):
    a, b = map(int, input().split())
    d[a].append(b)
    d[b].append(a)
ans = 0
for i in range(1, n + 1):
    vis = [False] * (n + 1)
    q = [i]
    vis[i] = True
    cur = L[i - 1]
    while q:
        t = q.pop(0)
        for j in d[t]:
            if not vis[j]:
                q.append(j)
                vis[j] = True
                cur = min(cur, L[j - 1])
    for j in range(1, n + 1):
        if i != j:
            ans += cur
print(ans / (n * (n - 1)))
