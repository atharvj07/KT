#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 75; // 2 * n
int n, k;
int a[MAXN], b[MAXN], c[MAXN], d[MAXN];
vector<pair<int, int>> g[MAXN];

bool check() {
  int mx[2] = {0, 0}, mn[2] = {MAXN, MAXN};
  for (int i = 0; i < 2 * n; ++i) {
    int j = i < n ? 0 : 1;
    mx[j] = max(mx[j], a[i]);
    mn[j] = min(mn[j], a[i]);
  }
  return mx[0] < mn[1];
}

int dfs(int u, int p) {
  int ans = 0;
  for (auto [v, t] : g[u]) {
    if (t == 0) {
      if (v == p) continue;
      ans += dfs(v, u);
    } else {
      a[c[v]] = d[v];
      if (check()) ans += dfs(v, u);
      a[c[v]] = 0;
    }
  }
  return ans;
}

int main() {
  cin >> n >> k;
  for (int i = 0; i < k; ++i) {
    int x, y, z;
    char op;
    cin >> x >> op >> y;
    x--, y--;
    z = (op == '=') ? 0 : ((op == '<') ? 1 : ((op == '>') ? 2 : (op == '<=' ? 3 : 4)));
    g[x].push_back({y, z});
    c[y] = x;
    d[y] = a[x];
  }
  a[0] = 0;
  cout << dfs(0, 0) << "\n";
  return 0;
}
