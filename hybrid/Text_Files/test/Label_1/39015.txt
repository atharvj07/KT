from collections import defaultdict

def dfs(adj, node, parent, depth, friends, max_depth, critical_node):
    depth[node] = depth[parent] + 1
    for child in adj[node]:
        if child != parent:
            dfs(adj, child, node, depth, friends, max_depth, critical_node)
    if node in friends:
        if depth[node] > max_depth[0]:
            max_depth[0] = depth[node]
            critical_node[0] = node

def solve(n, k, friends, corridors):
    adj = defaultdict(list)
    for u, v in corridors:
        adj[u].append(v)
        adj[v].append(u)

    depth = [0] * (n + 1)
    max_depth = [-1]
    critical_node = [-1]
    dfs(adj, 1, 0, depth, set(friends), max_depth, critical_node)

    depth = [0] * (n + 1)
    max_depth = [-1]
    second_node = [-1]
    dfs(adj, critical_node[0], 0, depth, set(), max_depth, second_node)

    return "YES" if max_depth[0] <= k else "NO"

t = int(input().strip())
for _ in range(t):
    input()
    n, k = map(int, input().strip().split())
    friends = list(map(int, input().strip().split()))
    corridors = [list(map(int, input().strip().split())) for _ in range(n-1)]
    print(solve(n, k, friends, corridors))
