import java.io.*;
import java.util.Arrays;
public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String [] arr = br.readLine().split(" ");
        int n = Integer.parseInt(arr[0]);
        int l = Integer.parseInt(arr[1]);
        int k = Integer.parseInt(arr[2]);
        char [] chs = br.readLine().toCharArray();
        Arrays.sort(chs);
        int idx = 0;
        for(int i = 1; i < n; i++) {
            while(idx < chs.length && chs[idx] <= chs[i * l]) {
                idx++;
            }
            if(idx >= chs.length) {
                break;
            }
            char temp = chs[i * l];
            chs[i * l] = chs[idx];
            chs[idx] = temp;
            idx++;
        }
        for(int i = 1; i < k; i++) {
            int x = (i - 1) * l;
            int y = x + l - 1;
            while(x < y) {
                char temp = chs[x];
                chs[x] = chs[y];
                chs[y] = temp;
                x++;
                y--;
            }
        }
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < l; j++) {
                System.out.print(chs[i * l + j]);
            }
            System.out.println();
        }
    }
}
