#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll inf = 1e18;
ll dp[300005][3];
int main() {
    ll n;
    cin >> n;
    ll i, j, k, a[n + 1], b[n + 1];
    for (i = 1; i <= n; i++)
        cin >> a[i];
    for (i = 1; i <= n; i++)
        cin >> b[i];
    for (i = n; i >= 1; i--) {
        dp[i][0] = max({dp[i + 1][0], dp[i + 1][1] + a[i], dp[i + 1][2] + b[i]});
        dp[i][1] = max({dp[i + 1][0] + a[i], dp[i + 1][1], dp[i + 1][2] + b[i]});
        dp[i][2] = max({dp[i + 1][0] + a[i], dp[i + 1][1] + b[i], dp[i + 1][2]});
    }
    cout << dp[1][0] << "\n";
    return 0;
}
