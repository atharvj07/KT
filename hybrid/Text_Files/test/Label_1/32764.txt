import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] ev = new int[n][2];
        for (int i = 0; i < n; i++) {
            ev[i][0] = sc.nextInt();
            ev[i][1] = sc.nextInt();
        }
        int V = sc.nextInt();
        Arrays.sort(ev, (a, b) -> a[1] - b[1]);
        int max1 = 0, max2 = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur <= ev[i][1]) {
                max1++;
                cur = ev[i][1] + Math.abs(ev[i][0]);
            }
        }
        cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur <= ev[i][1] - Math.abs(ev[i][0])) {
                max2++;
                cur = ev[i][1];
            }
        }
        System.out.println(max1 + " " + max2);
    }
}
