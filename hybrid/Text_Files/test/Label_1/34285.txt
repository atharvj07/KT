import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        sc.close();
        int[] f = new int[n + 1];
        int[] last = new int[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            f[i] = f[i + 1] + 1;
            last[p[i]] = i;
            if (i + 1 < n) {
                f[i] = Math.min(f[i], f[i + 2] + (p[i + 1] != p[i] - 1 ? 1 : 0));
            }
            for (int j = p[i] + 1; j <= n; j++) {
                if (last[j] > i) {
                    f[i] = Math.min(f[i], f[last[j]] + (p[i] != j - 1 ? 1 : 0));
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(f[i] + " ");
        }
        System.out.println();
    }
}
