n = int(input())
a = input()
m = int(input())
words = [input() for _ in range(m)]
d = {}
for i in range(n):
    if a[i] == '*':
        for w in words:
            if w[i] in d:
                d[w[i]] -= 1
            else:
                d[w[i]] = -1
cnt = 0
for v in d.values():
    if v == 0:
        cnt += 1
print(cnt)
