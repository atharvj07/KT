import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextLong();
        }
        long sum = 0;
        int idx = 0;
        while (idx < n && arr[idx] == 0) {
            idx++;
        }
        for (int i = idx; i < n; i++) {
            sum += arr[i];
        }
        sum *= 2;
        int ans = 0;
        while (sum > 0) {
            long cur = sum;
            while (cur % 2 == 0) {
                cur /= 2;
            }
            if (cur == 1) {
                break;
            }
            ans++;
            sum++;
        }
        System.out.println(ans);
    }
}
