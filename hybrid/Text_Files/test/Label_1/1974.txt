#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define ll long long
ll n,m,q;
string a[1000001];
bool vis[1000001]={0};
vi adj[1000001];
bool ex[1000001]={0};
queue<int> q1;
int main(){
    cin>>n>>m;
    for (int i=1;i<=n;++i){
        cin>>a[i];
    }
    for (int i=1;i<=n;++i){
        for (int j=1;j<=m;++j){
            if (a[i][j]=='.'){
                if (j==1||(j>1&&a[i][j-1]=='.')||(i>1&&a[i-1][j]=='.')||(i==n&&j==m)){
                    ex[i*m+j]=1;
                    q1.push(i*m+j);
                }
            }
        }
    }
    while(!q1.empty()){
        int u=q1.front();
        q1.pop();
        int x=u/m,y=u%m;
        if (x!=1&&a[x-1][y]=='.'){
            if (!ex[(x-1)*m+y]){
                ex[(x-1)*m+y]=1;
                q1.push((x-1)*m+y);
            }
        }
        if (x!=n&&a[x+1][y]=='.'){
            if (!ex[(x+1)*m+y]){
                ex[(x+1)*m+y]=1;
                q1.push((x+1)*m+y);
            }
        }
        if (y!=1&&a[x][y-1]=='.'){
            if (!ex[x*m+y-1]){
                ex[x*m+y-1]=1;
                q1.push(x*m+y-1);
            }
        }
        if (y!=m&&a[x][y+1]=='.'){
            if (!ex[x*m+y+1]){
                ex[x*m+y+1]=1;
                q1.push(x*m+y+1);
            }
        }
    }
    cin>>q;
    while(q--){
        int l,r;
        cin>>l>>r;
        int f=0,fl=0;
        for (int j=l;j<=r;++j){
            if (ex[j]){
                f=1;
            }
            if (!ex[j]){
                fl=1;
            }
        }
        f+=fl;
        if (f==1){
            cout<<"YES"<<endl;
        }
        else if (f==2){
            cout<<"NO"<<endl;
        }
        else{
            cout<<"YES"<<endl;
        }
    }
}
