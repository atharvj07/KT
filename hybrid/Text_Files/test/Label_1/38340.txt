import sys

n = int(input())
arr = list(map(int, input().split()))

if n == 1:
    print("YES")
    print(arr[0])
    sys.exit()

arr.sort()
if arr[0] == arr[-1]:
    print("YES")
    for i in range(n):
        for j in range(n):
            print(arr[0], end=" ")
        print()
    sys.exit()

if n % 2 == 0:
    if arr[n // 2] != arr[n // 2 - 1]:
        print("NO")
        sys.exit()
    mid1 = arr[n // 2]
    mid2 = arr[n // 2 - 1]
else:
    mid1 = arr[n // 2]
    mid2 = mid1

i = 0
j = n * n - 1
mat = [[0] * n for i in range(n)]
while i < j:
    mat[i][i] = mat[j][j] = arr[i]
    mat[i][j] = mat[j][i] = arr[j]
    i += 1
    j -= 1

mat[n // 2][n // 2] = mid1
if n % 2 != 0:
    mat[(n // 2 + 1) % n][n // 2] = mat[n // 2][(n // 2 + 1) % n] = mid2

print("YES")
for i in mat:
    for j in i:
        print(j, end=" ")
    print()
