n=int(input())
aa=[0]*n;x=[0]*n
for i in range(n):
    aa[i],x[i]=map(int, input().split())
m=int(input())
bb=[0]*m;y=[0]*m
for i in range(m):
    bb[i],y[i]=map(int, input().split())
aa.sort();bb.sort()
aa=list(set(aa));bb=list(set(bb))
ans=0
i=n-1;j=m-1
while i>=0 and j>=0:
    if aa[i]>bb[j]:
        ans+=x[i];i-=1
    else:
        ans+=y[j];j-=1
print(ans)
