#include<bits/stdc++.h>
using namespace std;
const int maxn=71,maxm=201;
struct Edge{
	int from,to,w;
	bool operator < (const Edge &t)const{
		return w<t.w;
	}
}; vector<Edge>e;
int n,m,a,b,p[maxn],r[maxn];
void init(int n){
	for(int i=1;i<=n;++i)r[i]=i,p[i]=i;
}
int find(int x){
	return x==r[x]?x:(r[x]=find(r[x]));
}
void unite(int u,int v){
	u=find(u),v=find(v);
	r[v]=u;
}
bool same(int u,int v){
	return find(u)==find(v);
}
int solve(){
	int tot=0;
	sort(e.begin(),e.end());
	for(Edge u : e){
		if(same(u.from,u.to))continue;
		tot+=u.w;
		unite(u.from,u.to);
	}
	return tot;
}
int dis[maxn][maxn];
int d[maxn]; bool vis[maxn];
void dij(int st){
	for(int i=1;i<=n;++i)d[i]=1e9;
	for(Edge u : e){
		if(same(st,u.from))dis[st][u.to]=u.w;
		if(same(st,u.to))dis[st][u.from]=u.w;
	}
	d[st]=0; vis[st]=true;
	for(int i=1;i<=n;++i){
		int u=-1;
		for(int j=1;j<=n;++j)
			if(!vis[j]&&(u==-1||d[j]<d[u]))
				u=j;
		if(u==-1)break;
		vis[u]=true;
		for(Edge j : e){
			if(!same(u,j.from)&&!same(u,j.to))continue;
			int v=(same(u,j.from)?j.to:j.from);
			dis[st][v]=min(dis[st][v],d[u]+j.w);
		}
	}
}
int main(){
	cin>>n>>m>>a>>b;
	init(n);
	for(int i=1;i<=m;++i){
		int u,v,w;
		cin>>u>>v>>w;
		e.push_back({u,v,w});
	}
	int ans=solve();
	for(int i=1;i<=n;++i)dij(i);
	for(int i=1;i<=n;++i)
		cout<<(i==1?0:dis[1][i])<<' ';
	return 0;
}
