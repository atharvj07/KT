import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();

            long totalWealth = 0;
            int wealthyPeople = 0;

            for (int j = 0; j < n; j++) {
                long wealth = scanner.nextLong();
                totalWealth += wealth;

                if (wealth >= x) {
                    wealthyPeople++;
                }
            }

            wealthyPeople = Math.max(wealthyPeople, (int) (totalWealth / x));

            System.out.println(wealthyPeople);
        }
    }
}
