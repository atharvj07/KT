import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            long mask = 0;
            for (int j = 0; j < k; j++) {
                mask |= 1L << sc.nextInt();
            }
            masks[i] = mask;
        }
        int q = sc.nextInt();
        while (q-- > 0) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            long mask = masks[m];
            switch (op) {
                case 0:
                    System.out.println((mask >> sc.nextInt() & 1));
                    break;
                case 1:
                    System.out.println(mask |= mask);
                    break;
                case 2:
                    System.out.println(mask &= ~mask);
                    break;
                case 3:
                    System.out.println(mask ^= mask);
                    break;
                case 4:
                    System.out.println((mask & mask) == mask ? 1 : 0);
                    break;
                case 5:
                    System.out.println((mask & mask) > 0 ? 1 : 0);
                    break;
                case 6:
                    System.out.println((mask & mask) == 0 ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(mask & mask));
                    break;
                case 8:
                    System.out.println(Long.toBinaryString(mask & mask));
                    break;
            }
        }
    }
}
