#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

int main() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for(ll i = 0; i < n; i++) cin >> a[i];
    vector<pair<ll, pair<ll, ll>>> ans;
    ll ans1 = 0, ans2 = 0;
    for(ll i = 0; i + 1 < n; i++) {
        if(a[i] == a[i + 1]) continue;
        if(a[i] > a[i + 1]) {
            ans1 += a[i] - a[i + 1];
            ans.push_back({1, {i + 1, i + 2}});
        } else {
            ans2 += a[i + 1] - a[i];
            ans.push_back({2, {i + 1, i + 2}});
        }
    }
    cout << ans1 + ans2 << endl;
    for(auto x : ans) cout << x.first << ' ' << x.second.first << ' ' << x.second.second << endl;
    return 0;
}
