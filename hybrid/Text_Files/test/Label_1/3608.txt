#include <cstdio>
#include <vector>
const int N = 3e5 + 5;
int n, k, d, p[N], u[N], v[N], g[N], ans[N], s;
std::vector<int> adj[N];
int main() {
  scanf("%d%d%d", &n, &k, &d);
  for (int i = 0; i < k; i++) { scanf("%d", p + i); }
  for (int i = 1; i < n; i++) {
    scanf("%d%d", u + i, v + i);
    adj[u[i]].push_back(i);
    adj[v[i]].push_back(i);
  }
  for (int i = 0; i < k; i++) {
    for (int j = 0; j < adj[p[i]].size(); j++) {
      int id = adj[p[i]][j];
      if (!g[id]) {
        g[id] = g[u[id]] & g[v[id]] | 1 << i;
        if (__builtin_popcount(g[id]) > d) { ans[++s] = id; }
      }
    }
  }
  printf("%d\n", s);
  for (int i = 1; i <= s; i++) { printf("%d ", ans[i]); }
  puts("");
  return 0;
}
