import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = input.nextInt();
            }
        }

        int max = 0;
        for (int x = 0; x < n; x++) {
            for (int y = 0; y < m; y++) {
                int sum = 0;
                boolean[][] visited = new boolean[n][m];
                int i = x, j = y;
                while (i >= 0 && i < n && j >= 0 && j < m && !visited[i][j]) {
                    sum += a[i][j];
                    visited[i][j] = true;
                    if (i == x && j < m - 1 && !visited[i][j + 1]) {
                        j++;
                    } else if (j == m - 1 && i < n - 1 && !visited[i + 1][j]) {
                        i++;
                    } else if (i == n - 1 && j > 0 && !visited[i][j - 1]) {
                        j--;
                    } else if (j == 0 && i > 0 && !visited[i - 1][j]) {
                        i--;
                    } else if (i == x && j < m - 1 && !visited[i][j + 1]) {
                        j++;
                    }
                }
                max = Math.max(max, sum);
            }
        }
        System.out.println(max);
    }
}
