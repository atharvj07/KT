import java.util.*;
class Main {
  private static int[] arr;
  private static int[][] memo;
  private static boolean wins(int l, int r) {
    if (l > r)
      return false;
    if (memo[l][r] != -1)
      return memo[l][r] == 1;
    boolean can_win = false;
    for (int i = l; i <= r; i++) {
      for (int j = i; j <= r; j++) {
        if (arr[j] == arr[i] && i != j)
          can_win |= !wins(i + 1, j - 1);
        if (!can_win)
          can_win |= !wins(i + 1, r);
      }
    }
    memo[l][r] = can_win ? 1 : 0;
    return can_win;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    arr = new int[n];
    memo = new int[n][n];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++)
        memo[i][j] = -1;
      arr[i] = sc.nextInt();
    }
    sc.close();
    System.out.println(wins(0, n - 1) ? "sjfnb" : "cslnb");
  }
}
