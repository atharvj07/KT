#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v;
  while (n--) {
    int id;
    cin >> id;
    bool found = false;
    for (int i = 0; i < v.size(); i++) {
      if (v[i] == id) {
        found = true;
        for (int j = i; j > 0; j--) { v[j] = v[j - 1]; }
        v[0] = id;
        break;
      }
    }
    if (!found) {
      v.insert(v.begin(), id);
      if (v.size() > k) v.pop_back();
    }
  }
  cout << v.size() << endl;
  for (int i = 0; i < v.size(); i++) cout << v[i] << " ";
}
