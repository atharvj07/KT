from collections import deque
n, q = map(int, input().split())
a = deque(map(int, input().split()))
for i in range(q):
    t, x = map(int, input().split())
    if t == 1:
        a[x - 1] ^= 1
    else:
        for i in sorted(a, reverse=True):
            x -= 1
            if x == 0:
                print(i)
                break
