import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        boolean color = false;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                char c = in.next().charAt(0);
                if (c == 'C' || c == 'M' || c == 'Y') {
                    color = true;
                    break;
                }
            }
        }
        System.out.println(color ? "#Color" : "#Black&White");
    }
}
