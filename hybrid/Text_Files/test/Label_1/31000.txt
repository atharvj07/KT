import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int t = in.nextInt();
        in.close();
        System.out.println(countPolylines(n, t));
    }

    private static long countPolylines(int n, int t) {
        if (n < 3 || t > 10) {
            return 0;
        }

        long[][] dp = new long[n][t + 1];
        dp[0][0] = 1;

        for (int i = 1; i < n; i++) {
            for (int j = 0; j <= t; j++) {
                dp[i][j] = dp[i - 1][j];
                if (j > 0) {
                    dp[i][j] += dp[i - 1][j - 1] * Math.abs(4 - j);
                }
                if (j < t) {
                    dp[i][j] += dp[i - 1][j + 1] * Math.abs(j + 1 - 1);
                }
            }
        }

        return dp[n - 1][t];
    }
}
