#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
vector<ll> suff(N), pref(N);
int n, k;
ll get(int l, int r) { return suff[l - 1] - suff[r]; }
int32_t main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    ll x;
    cin >> x;
    pref[i] = pref[i - 1] + x;
  }
  for (int i = n; i >= 1; i--) {
    suff[i] = suff[i + 1] + pref[i];
  }
  ll ans = LLONG_MIN;
  for (int l = k; l <= n; l++) {
    ans = max(ans, get(l - k + 1, l));
    // cout << ans << ' ' << l << endl;
  }
  cout << ans << endl;
}
