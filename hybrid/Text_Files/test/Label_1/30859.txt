import java.util.Scanner;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Stack<Integer> speedLimits = new Stack<>();
        Stack<Boolean> overtakeAllowed = new Stack<>();
        overtakeAllowed.push(true);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int type = in.nextInt();
            switch (type) {
                case 1:
                    speedLimits.clear();
                    speedLimits.push(in.nextInt());
                    break;
                case 2:
                    if (overtakeAllowed.peek()) {
                        break;
                    }
                    ans++;
                    break;
                case 3:
                    speedLimits.push(in.nextInt());
                    break;
                case 4:
                    overtakeAllowed.push(true);
                    break;
                case 5:
                    speedLimits.clear();
                    break;
                case 6:
                    overtakeAllowed.push(false);
                    break;
            }
        }
        System.out.println(ans);
    }
}
