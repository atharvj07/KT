import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int t = scanner.nextInt();
        int r = scanner.nextInt();
        int[] w = new int[m];
        for (int i = 0; i < m; i++) {
            w[i] = scanner.nextInt();
        }
        int candles = 0;
        int burning = 0;
        int time = 0;
        for (int i = 0; i < m; i++) {
            if (burning < r) {
                int need = r - burning;
                candles += need;
                burning += need;
                time = Math.max(time, w[i] - 1);
            }
            burning--;
            time++;
            if (time >= w[i]) {
                burning++;
            }
        }
        if (burning < r) {
            System.out.println(-1);
        } else {
            System.out.println(candles);
        }
    }
}
