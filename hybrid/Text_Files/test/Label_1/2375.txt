#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, p;
    cin >> n >> p;
    vector<long long> t(n);
    for (auto &e : t) cin >> e;

    vector<long long> ans(n);
    vector<int> q;
    for (int i = 0; i < n; ++i) {
        while (!q.empty() && t[q.back()] <= t[i] - p) q.pop_back();
        ans[i] = !q.empty() ? q.back() * p + p : t[i];
        q.push_back(i);
    }
    for (const auto &e : ans) cout << e << ' ';
    return 0;
}
