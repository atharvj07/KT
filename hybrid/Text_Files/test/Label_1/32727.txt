import java.io.BufferedReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static long[] A, P;
    static int N;

    void solve(InputStream in) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(in));
        int m = Integer.parseInt(br.readLine());
        A = new long[m + 1];
        String[] strs = br.readLine().split(" ");
        for (int i = 0; i < m; ++i)
            A[i + 1] = Long.parseLong(strs[i]);
        P = new long[m + 1];
        for (int i = 1; i <= m; ++i) {
            P[i] = P[i - 1] ^ A[i];
        }
        int[] id = new int[m + 1];
        Arrays.fill(id, -1);
        for (int i = 1; i < m; ++i) {
            for (int j = i + 1; j <= m; ++j) {
                if (P[i] == P[j]) id[i] = j;
            }
        }
        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; ++i) {
            int l, r;
            {
                String[] strs2 = br.readLine().split(" ");
                l = Integer.parseInt(strs2[0]);
                r = Integer.parseInt(strs2[1]);
            }
            long ans = 0L;
            if (id[l - 1] != -1 && id[l - 1] >= r) ans = P[l - 1];
            else ans = P[r];
            System.out.println(ans);
        }
    }

    public static void main(String[] args) throws Exception {
        Main main = new Main();
        main.solve(System.in);
    }
}
