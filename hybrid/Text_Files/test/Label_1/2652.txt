#include<bits/stdc++.h>
using namespace std;
int main() {
    int n,k,id; cin>>n>>k;
    vector<int> a(k+1);
    int j=1;
    while(n--) {
        cin>>id;
        for(int i=j; i>=1; i--) {
            if(id==a[i]) {
                for(int x=i; x<j; x++) a[x]=a[x+1];
                a[j]=id;
                break;
            } else {
                a[i+1]=a[i];
            }
        }
        a[1]=id;
        j++;
        if(j>k) j=k;
    }
    cout<<j<<endl;
    for(int i=1; i<=j; i++) cout<<a[i]<<' ';
    cout<<endl;
}
