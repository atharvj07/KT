#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;
  bool good = false;
  while (n--) {
    std::string handle;
    int before, after;
    std::cin >> handle >> before >> after;
    if (before < 2400 && after >= 2400) {
      good = true;
    }
  }
  std::cout << (good ? "YES" : "NO") << std::endl;
  return 0;
}
