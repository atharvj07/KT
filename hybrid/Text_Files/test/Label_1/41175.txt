def find_digit(k):
    block_size = 1
    num_blocks = 9
    while k > block_size * num_blocks:
        k -= block_size * num_blocks
        block_size += 1
        num_blocks *= 10
    
    number = (k - 1) // block_size
    digit_in_number = (k - 1) % block_size
    starting_number = 10 ** (block_size - 1)

    final_number = starting_number + number
    return str(final_number)[digit_in_number]

q = int(input())
for _ in range(q):
    k = int(input())
    print(find_digit(k))
