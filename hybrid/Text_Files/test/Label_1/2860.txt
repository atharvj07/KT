#include <set>
#include <vector>
#include <iostream>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<set<int>> emp(n);
    for (int i = 0; i < n; ++i) {
        int k;
        cin >> k;
        for (int j = 0; j < k; ++j) {
            int l;
            cin >> l;
            emp[i].insert(l);
        }
    }
    set<int> all;
    for (auto& e : emp) {
        all.insert(e.begin(), e.end());
    }

    int ans = 0;
    for (auto& l : all) {
        int cnt = 0;
        for (int i = 0; i < n; ++i) {
            if (!emp[i].count(l)) {
                ++cnt;
            }
        }
        ans += cnt > 0;
    }

    cout << ans << endl;

    return 0;
}
