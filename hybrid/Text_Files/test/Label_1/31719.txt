import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        // Read the size of the permutation and the permutation itself
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        // Read the number of queries
        int m = in.nextInt();

        // Initialize the inversion count to 0
        long inv = 0;

        // Initialize the array of prefix sums for inversions
        long[] invP = new long[n + 1];

        // Calculate the prefix sums of inversions
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[i] > a[j]) {
                    inv++;
                }
            }
            invP[i + 1] = inv;
        }

        // Process the queries
        for (int i = 0; i < m; i++) {
            // Read the indices of the segment to reverse
            int l = in.nextInt() - 1;
            int r = in.nextInt() - 1;

            // Reverse the segment
            for (int j = 0; j < (r - l + 1) / 2; j++) {
                int tmp = a[l + j];
                a[l + j] = a[r - j];
                a[r - j] = tmp;
            }

            // Update the prefix sums of inversions
            inv -= invP[r + 1] - invP[l];
            for (int j = l; j < r; j++) {
                for (int k = j + 1; k <= r; k++) {
                    if (a[j] > a[k]) {
                        inv++;
                    }
                }
            }
            invP[l] = inv;

            // Print whether the number of inversions is odd or even
            if (inv % 2 == 0) {
                System.out.println("even");
            } else {
                System.out.println("odd");
            }
        }
    }
}
