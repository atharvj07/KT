import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();

        Point[] p = new Point[n];
        for (int i = 0; i < n; i++) {
            p[i] = new Point(in.nextInt(), in.nextInt());
        }

        int[] ans = new int[k];
        for (int i = 0; i < k; i++) {
            ans[i] = i + 1;
        }

        double curDiam = maxDiam(p, ans);

        int[] curAns = ans.clone();

        for (int i = k; i < n; i++) {
            ans[k - 1] = i + 1;
            double newDiam = maxDiam(p, ans);
            if (newDiam < curDiam) {
                curDiam = newDiam;
                curAns = ans.clone();
            }
        }

        for (int i = 0; i < k; i++) {
            System.out.print(curAns[i] + " ");
        }
        System.out.println();
    }

    public static double maxDiam(Point[] p, int[] ans) {
        double maxDist = 0;

        for (int i = 0; i < ans.length; i++) {
            for (int j = i + 1; j < ans.length; j++) {
                double dist = p[ans[i] - 1].dist(p[ans[j] - 1]);
                maxDist = Math.max(dist, maxDist);
            }
        }

        return maxDist;
    }

    public static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public double dist(Point other) {
            double dx = x - other.x;
            double dy = y - other.y;
            return Math.sqrt(dx * dx + dy * dy);
        }
    }
}
