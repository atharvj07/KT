import java.util.Scanner;
import java.util.ArrayList;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ArrayList<Integer> piles = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            piles.add(sc.nextInt());
        }
        sc.close();

        boolean win = false;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (piles.get(i) == piles.get(j)) {
                    win = true;
                }
            }
        }
        System.out.println((win ? "sjfnb" : "cslnb"));
    }
}
