#include <bits/stdc++.h>
using namespace std;

const int MAXN = 2e5 + 5, MAXE = MAXN;

vector<int> adj[MAXN];
int sub_size[MAXN], ans = 0;
bool vis[MAXN];

void dfs(int node, int parent) {
    vis[node] = true;
    sub_size[node] = 1;
    for (int nbr : adj[node]) {
        if (vis[nbr]) continue;
        dfs(nbr, node);
        sub_size[node] += sub_size[nbr];
    }
}

void get_ans(int node, int parent, int sz, int odd) {
    if (sub_size[node] % 2 == 1) odd = 1;
    bool ok = false;
    for (int nbr : adj[node]) {
        if (nbr == parent) continue;
        if (sub_size[nbr] % 2 == 1) {
            get_ans(nbr, node, sz, 1);
            ok = true;
            continue;
        }
        get_ans(nbr, node, sz, odd);
    }

    if (!ok && odd) {
        ans = max(ans, sz - sub_size[node]);
        return;
    }

    if (!ok) {
        int mx = -1, node = -1;
        for (int nbr : adj[node]) {
            if (nbr == parent) continue;
            if (sub_size[nbr] > mx) {
                mx = sub_size[nbr];
                node = nbr;
            }
        }
        if (node != -1) ans = max(ans, sz - sub_size[node]);
    }
}

int main() {
    int n;
    cin >> n;

    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    dfs(1, -1);

    if (sub_size[1] % 2 == 1) {
        cout << -1 << endl;
        return 0;
    }

    get_ans(1, -1, n, 0);

    cout << ans << endl;
}
