import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n;
    while ((n = sc.nextInt()) != 0) {
      int[] x = new int[n];
      int[] y = new int[n];
      for (int i = 0; i < n; i++) {
        x[i] = sc.nextInt();
        y[i] = sc.nextInt();
      }
      int[] d = new int[n];
      for (int i = 0; i < n; i++) {
        d[i] = sc.nextInt();
        y[i] += d[i];
      }
      int z = Arrays.stream(y).max().getAsInt();
      System.out.println(z / 10000 + (z % 10000 > 0 ? 1 : 0));
    }
  }
}
