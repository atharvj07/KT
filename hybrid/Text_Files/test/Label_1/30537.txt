import java.util.ArrayList;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        input.close();

        ArrayList<Integer> list = new ArrayList<>();
        for (int i = 31; i >= 0; i--) {
            if ((n & (1 << i)) > 0) {
                list.add(1 << i);
                n -= 1 << i;
            }
        }

        System.out.println(list.size());
        for (int x : list) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
