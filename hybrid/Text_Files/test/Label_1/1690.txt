#include <iostream>
#define N 3
using namespace std;
int main() {
  string s;
  cin >> s;
  int f[N + 1] = {0};
  for (int i = 0; i < s.size(); i++) {
    char c = s[i];
    if (c == '+') continue;
    int num = (int)c - (int)'0';
    f[num]++;
  }
  for (int i = 1; i <= N; i++) {
    for (int j = 1; j <= f[i]; j++) {
      cout << i << (j < f[i] ? '+' : '\n');
    }
  }
  return 0;
}
