import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean[] check = new boolean[360];

        // Iterate over each rotation
        for (int i = 0; i < n; i++) {
            int angle = sc.nextInt();
            check[360 - angle] = true; // Mark the angle in the opposite direction
            check[angle] = true; // Mark the angle in the original direction
        }

        // Check if there are any unmarked angles
        for (boolean b : check) {
            if (!b) {
                System.out.println("NO");
                return;
            }
        }

        System.out.println("YES");
    }
}
