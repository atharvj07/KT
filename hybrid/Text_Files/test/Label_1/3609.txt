#include<bits/stdc++.h>
using namespace std;

int main() {
    int n, k, x;
    cin >> n >> k;
    int a[n], b[k];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < k; i++) {
        cin >> b[i];
    }
    sort(b, b + k);
    int cnt = 0, idx = 0, ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) cnt++;
        else if (a[i] < b[idx]) {
            ans = 1;
            break;
        } else if (a[i] == b[idx]) idx++;
    }
    if (cnt != idx) ans = 1;
    cout << (ans == 0 ? "Yes" : "No") << endl;
    return 0;
}
