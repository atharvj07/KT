import java.util.*;

public class Main {
    static class Edge {
        int to, cost, time;
        Edge(int to, int cost, int time) {
            this.to = to;
            this.cost = cost;
            this.time = time;
        }
    }

    static class Node {
        int no, cost, time;
        Node(int no, int cost, int time) {
            this.no = no;
            this.cost = cost;
            this.time = time;
        }
    }

    static int N, M, K;
    static List<List<Edge>> G;
    static int[] dist;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            N = sc.nextInt();
            M = sc.nextInt();
            if (N == 0 && M == 0) break;
            G = new ArrayList<>();
            for (int i = 0; i <= M; i++)
                G.add(new ArrayList<>());
            for (int i = 0; i < N; i++) {
                int a = sc.nextInt();
                int b = sc.nextInt();
                int c = sc.nextInt();
                int t = sc.nextInt();
                G.get(a).add(new Edge(b, c, t));
                G.get(b).add(new Edge(a, c, t));
            }
            K = sc.nextInt();
            for (int i = 0; i < K; i++) {
                int p = sc.nextInt();
                int q = sc.nextInt();
                int r = sc.nextInt();
                dijkstra(p, r);
                System.out.println(dist[q]);
            }
        }
    }

    static void dijkstra(int s, int r) {
        dist = new int[M + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[s] = r == 0 ? 0 : Integer.MAX_VALUE;
        Queue<Node> que = new PriorityQueue<>(Comparator.comparing(x -> x.cost));
        que.offer(new Node(s, 0, 0));
        while (!que.isEmpty()) {
            Node u = que.poll();
            if (dist[u.no] < u.cost) continue;
            for (Edge e : G.get(u.no)) {
                int v = e.to;
                int cost = r == 0 ? e.cost : e.time;
                if (dist[v] > u.cost + cost) {
                    dist[v] = u.cost + cost;
                    que.offer(new Node(v, u.cost + cost, u.time + e.time));
                }
            }
        }
    }
}
