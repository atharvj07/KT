#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, x[1001], y[1001];
  cin >> n >> m;
  vector<string> a(n);
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < m; ++j) {
      if (a[i][j] == '*') {
        x[i]++;
        y[j]++;
      }
    }
  }
  int sum_x = 0, sum_y = 0;
  for (int i = 0; i < n; ++i) {
    if (x[i] == m) {
      sum_x++;
    }
  }
  for (int j = 0; j < m; ++j) {
    if (y[j] == n) {
      sum_y++;
    }
  }
  if (sum_x == 0 || sum_y == 0) {
    cout << "NO";
  } else {
    cout << "YES";
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m; ++j) {
        if (x[i] == m && y[j] == n) {
          cout << i + 1 << ' ' << j + 1 << endl;
          return 0;
        }
      }
    }
  }
  return 0;
}
