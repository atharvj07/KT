#include <iostream>
using namespace std;
typedef unsigned long long ll;
long long sum[1000005], pos[1000005];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a;
    for (int i = 0; i < n; i++) {
      cin >> a;
      sum[i] = sum[i - 1];
      pos[i] = i;
      if (i && a == pos[i - 1]) {
        sum[i] = sum[i - 1] + (2 * i - pos[i - 1] - 1);
        pos[i] = pos[i - 1] - 1;
      } else {
        sum[i]++;
      }
    }
    cout << sum[n - 1] << endl;
  }
  return 0;
}
