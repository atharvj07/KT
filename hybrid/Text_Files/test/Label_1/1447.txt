#include<bits/stdc++.h>
using namespace std;
int main(){
    int a,cnt=2e9;
    string s;
    unordered_map<char,int>mp;
    for(int i=0;i<3;i++){ cin>>s; mp[s[1]]++; }
    for(int i='m';i<='s';i++){
        int d=mp[i],k;
        if(d==0){ k=3; }
        else if(d==1){ k=2; }
        else if(d==2){ k=1; }
        else k=0;
        cnt=min(cnt,k);
    }
    cout<<cnt<<endl;
}
