import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int x = sc.nextInt();
    int numDistressedKids = 0;
    for (int i = 0; i < n; i++) {
      char sign = sc.next().charAt(0);
      int amount = sc.nextInt();
      if (sign == '+') {
        x += amount;
      } else {
        if (x >= amount) {
          x -= amount;
        } else {
          numDistressedKids++;
        }
      }
    }
    System.out.println(x + " " + numDistressedKids);
  }
}
