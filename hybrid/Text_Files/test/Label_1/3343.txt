#include <bits/stdc++.h>

using namespace std;

bool pos_on_dir[3][4], b_exist;
bool kingx[1000005], kingy[1000005], cross[1000005];

int main() {
  int n;
  cin >> n;
  int a, b;
  cin >> a >> b;
  while (n--) {
    char c;
    cin >> c >> a >> b;
    if (a == 0 and b == 0) b_exist = 1;
    if (kingx[a + b]) cross[a + b] = 1;
    kingx[a + b] = kingy[a - b] = 1;
    pos_on_dir[c == 'Q' || c == 'R'][a != 0] |=
        (c == 'Q' || c == 'B') && abs(a) == abs(b);
  }
  cout << ((pos_on_dir[0][0] || pos_on_dir[1][0] || pos_on_dir[2][0] ||
                pos_on_dir[0][1] || pos_on_dir[1][1] || pos_on_dir[2][1] ||
                (b_exist && cross[a + b]))
               ? "YES"
               : "NO");
}
