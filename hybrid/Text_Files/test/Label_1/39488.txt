import sys

def f(n):
    if n == 1: return (1, [1])
    col = [0] * (n + 1)
    for i in range(2, n + 1):
        if not col[i]:
            for j in range(i*i, n + 1, i):
                col[j] = col[i] + 1
    c = max(col) + 1
    return (c, col)

n = int(input())
c, col = f(n)
print(c)
print(*col[1:])
