import sys

inpt = sys.stdin.read().split('\n')
inpt_ = iter(inpt)

def input():
    return next(inpt_)

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(input())
    a_ = a.copy()
    a.sort()
    ans = 0
    i = 0
    while i < n:
        j = i
        while j < n and a[j] == a[i]:
            j += 1
        ans = max(ans, min(k, j - i))
        i = j
        x = n - ans
        if x <= 0: break
        ans = max(ans, min(k, x + n - 1))
    x = n - ans
    if x <= 0: print(ans); continue
    y = int(x / k)
    z = x % k
    ans = max(ans, min(k, y * k + z + 1))
    print(ans)
