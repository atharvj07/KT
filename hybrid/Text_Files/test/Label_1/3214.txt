#include<bits/stdc++.h>
using namespace std;
const int N = 1005;
int n, t;
int dp[N][N];
struct task {
    int time, q;
};
vector<task> v;
int solve(int i, int rem) {
    if (i == n) return 0;
    if (dp[i][rem]!=-1) return dp[i][rem];
    int op1 = 0, op2 = 0;
    if (v[i].time<=rem) {
        op1 = v[i].q + solve(i+1, rem - v[i].time);
    }
    op2 = solve(i+1, rem);
    return dp[i][rem] = max(op1, op2);
}
int main() {
    cin >> n >> t;
    v.resize(n+1);
    memset(dp, -1, sizeof(dp));
    for (int i = 0; i < n; i++) {
        int a, b; cin >> a >> b;
        v[i] = {a,b};
    }
    cout << solve(0, t) << '\n';
    return 0;
}
