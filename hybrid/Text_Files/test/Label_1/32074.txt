import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static int n, m, k;
    private static int[][] grid;
    private static int[][][] memo;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        k = scanner.nextInt();
        grid = new int[n + 2][m + 2];
        for (int i = 1; i <= n; i++) {
            String line = scanner.next();
            for (int j = 1; j <= m; j++) {
                grid[i][j] = line.charAt(j - 1) - '0';
            }
        }
        memo = new int[n + 2][m + 2][k + 1];
        for (int i = 0; i <= n + 1; i++) {
            for (int j = 0; j <= m + 1; j++) {
                Arrays.fill(memo[i][j], -1);
            }
        }
        System.out.println(count(1, 1, k));
    }

    private static int count(int x, int y, int stars) {
        if (x > n || y > m || memo[x][y][stars] != -1) {
            return memo[x][y][stars];
        }
        if (stars == 0) {
            return 1;
        }
        int res = 0;
        if (grid[x][y] == 1) {
            res += count(x + 1, y - 1, stars - 1);
            res += count(x + 1, y, stars - 1);
            res += count(x + 1, y + 1, stars - 1);
            res += count(x, y + 1, stars - 1);
            res += count(x - 1, y + 1, stars - 1);
            res += count(x - 1, y, stars - 1);
            res += count(x - 1, y - 1, stars - 1);
            res += count(x, y - 1, stars - 1);
        }
        return memo[x][y][stars] = res;
    }
}
