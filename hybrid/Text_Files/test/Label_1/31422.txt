import java.util.*;

class Main {
    private final int MOD = 1000000007;
    private ArrayList<Integer>[] adj;
    private int[] ans;
    private int[] depth;
    private long[] facts;
    private int n;

    public static void main(String[] args) {
        Main m = new Main();
        m.solve();
    }

    private void solve() {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        ans = new int[n + 1];
        depth = new int[n + 1];
        facts = new long[n + 2];
        facts[0] = facts[1] = 1;
        for (int i = 2; i <= n + 1; i++) {
            facts[i] = (facts[i - 1] * i) % MOD;
        }
        dfs(1, -1, 0);
        int minDepth = depth[1];
        for (int i = 2; i <= n; i++) {
            minDepth = Math.min(minDepth, depth[i]);
        }
        int cntMinDepth = 0;
        for (int i = 1; i <= n; i++) {
            if (depth[i] == minDepth) {
                cntMinDepth++;
            }
        }
        ans[minDepth] = 1;
        for (int i = minDepth + 1; i <= n; i++) {
            if (cntMinDepth > 1) {
                ans[i] = (int) ((facts[cntMinDepth] * fastPower(cntMinDepth, MOD - 2)) % MOD);
            } else {
                ans[i] = (int) ((facts[n - 1] * fastPower(n - 1, MOD - 2)) % MOD);
            }
        }
        System.out.println(ans[n]);
    }

    private void dfs(int curr, int parent, int d) {
        depth[curr] = d;
        for (int next : adj[curr]) {
            if (next == parent) {
                continue;
            }
            dfs(next, curr, d + 1);
        }
    }

    private long fastPower(int base, int p) {
        if (p == 0) {
            return 1;
        }
        if (p == 1) {
            return base;
        }
        long val = fastPower(base, p / 2);
        val = (val * val) % MOD;
        if (p % 2 != 0) {
            val = (val * base) % MOD;
        }
        return val % MOD;
    }
}
