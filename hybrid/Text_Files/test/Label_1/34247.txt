import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int T;
        int[] t;
        int N;
        int[][] interview;
        while ((T = in.nextInt()) != 0) {
            t = new int[T];
            for (int i = 0; i < T; i++) {
                t[i] = in.nextInt();
            }
            N = in.nextInt();
            interview = new int[N][2];
            for (int i = 0; i < N; i++) {
                interview[i][0] = in.nextInt();
                interview[i][1] = in.nextInt();
            }

            int[] dp = new int[103];
            for (int i = 1; i <= 102; i++) {
                dp[i] = Integer.MAX_VALUE;
            }
            dp[1] = 0;
            for (int i = 1; i + T <= 102; i++) {
                for (int j = 1; j <= T; j++) {
                    int wakeTime = (i - 1) % T + 1;
                    if (t[wakeTime - 1] <= 23 - interview[j - 1][1]) {
                        dp[i + T] = Math.min(dp[i + T], dp[i] + 1);
                    }
                }
            }
            int minCaffeine = Integer.MAX_VALUE;
            for (int i = 1; i <= 102; i++) {
                if (dp[i] != Integer.MAX_VALUE) {
                    minCaffeine = Math.min(minCaffeine, dp[i]);
                }
            }
            System.out.println(minCaffeine);
        }
    }
}
