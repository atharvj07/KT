n, t, k = map(int, input().split())
a = list(map(int, input().split()))

# Calculate the minimum and maximum number of dead-end cities possible
min_dead_ends = sum(a[:-1]) - (t - 1)
max_dead_ends = sum([x - 1 for x in a])

if k < min_dead_ends or k > max_dead_ends:
    print(-1)
else:
    # Initial setup
    roads = []
    current_label = 2
    last_level_nodes = [1]  # The capital
    dead_ends_needed = k

    # Construct roads level by level
    for i, num_cities in enumerate(a):
        current_level_nodes = []
        for node in last_level_nodes:
            children = min(num_cities, dead_ends_needed - (len(a) - i - 1)) if i < len(a) - 1 else num_cities
            for _ in range(children):
                roads.append((node, current_label))
                current_level_nodes.append(current_label)
                current_label += 1
                num_cities -= 1

                if i < len(a) - 1:  # If not the last level, each node can potentially be a dead-end
                    dead_ends_needed -= 1

            if num_cities == 0:
                break

        # Fill the rest of the cities in the level if any
        if num_cities > 0:
            for _ in range(num_cities):
                # Connect remaining cities to the last node in the current_level_nodes
                roads.append((current_level_nodes[-1], current_label))
                current_level_nodes.append(current_label)
                current_label += 1

        last_level_nodes = current_level_nodes

    print(n)
    for road in roads:
        print(road[0], road[1])
