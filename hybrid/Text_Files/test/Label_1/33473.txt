import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        List<Edge> edges = new ArrayList<>();
        int[] count = new int[2];
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            edges.add(new Edge(Integer.parseInt(line[0]), Integer.parseInt(line[1]) == 1));
            count[Integer.parseInt(line[1])]++;
        }
        edges.sort((e1, e2) -> e1.weight - e2.weight);
        if (count[1] != n - 1 || count[0] != m - n + 1) {
            System.out.println(-1);
            return;
        }
        int[] parent = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            parent[i] = i;
        }
        int[] rank = new int[n + 1];
        Arrays.fill(rank, 1);
        for (Edge edge : edges) {
            if (!edge.inTree) {
                continue;
            }
            int root1 = findParent(edge.u, parent);
            int root2 = findParent(edge.v, parent);
            if (root1 != root2) {
                if (rank[root1] > rank[root2]) {
                    parent[root2] = root1;
                    rank[root1]++;
                } else {
                    parent[root1] = root2;
                    rank[root2]++;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (findParent(i, parent) != findParent(1, parent)) {
                System.out.println(-1);
                return;
            }
        }
        for (Edge edge : edges) {
            if (!edge.inTree) {
                int root1 = findParent(edge.u, parent);
                int root2 = findParent(edge.v, parent);
                if (root1 != root2) {
                    System.out.println(edge.u + " " + edge.v);
                    parent[root2] = root1;
                    rank[root1]++;
                }
            }
        }
    }

    private static int findParent(int node, int[] parent) {
        if (parent[node] == node) {
            return node;
        }
        return parent[node] = findParent(parent[node], parent);
    }

    private static class Edge {
        int weight;
        boolean inTree;
        int u;
        int v;

        public Edge(int weight, boolean inTree) {
            this.weight = weight;
            this.inTree = inTree;
        }
    }
}
