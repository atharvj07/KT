import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] edges = Arrays.stream(scanner.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        Arrays.sort(edges);
        System.out.println((edges[0] == edges[1] && edges[2] == edges[3]) || edges[0] + edges[1] + edges[2] > edges[3] ? "yes" : "no");
    }
}
