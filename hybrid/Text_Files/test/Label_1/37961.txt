def solve():
    t = int(input())
    for _ in range(t):
        n, m, a, b = map(int, input().split())
        s = sorted([(i, j) for i, j in enumerate(map(int, input().split()))], key=lambda x: x[1])
        ans, idx = 0, 0
        while a!=b:
            if idx==m: break
            if s[idx][1]<=abs(a-b)-1:
                ans+=1
                idx+=1
                a+=1 if a<b else -1
            else: break
        print(ans)

solve()
