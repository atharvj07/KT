import java.util.Arrays;
import java.util.Scanner;

public class Main {
  private static class File implements Comparable<File> {
    int raw, compressed;

    File(int raw, int compressed) {
      this.raw = raw;
      this.compressed = compressed;
    }

    public int compareTo(File o) {
      return this.compressed - o.compressed;
    }
  }

  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    while (true) {
      int numOfFiles = input.nextInt();
      int maxSize = input.nextInt();
      if (numOfFiles == 0 && maxSize == 0) {
        break;
      }
      File[] files = new File[numOfFiles];
      for (int i = 0; i < numOfFiles; i++) {
        int rawSize = input.nextInt();
        int compressedSize = input.nextInt();
        files[i] = new File(rawSize, compressedSize);
      }
      Arrays.sort(files);
      int usedMemory = 0, counter = 0;
      for (File file : files) {
        if (usedMemory + file.compressed <= maxSize) {
          usedMemory += file.compressed;
          counter++;
        }
      }
      if (counter == numOfFiles) {
        System.out.println(counter);
      } else {
        System.out.println("Impossible");
      }
    }
  }
}
