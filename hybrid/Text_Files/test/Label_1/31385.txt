import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n + 2];
        String s = sc.next();
        for (int i = 1; i <= n; i++) a[i] = s.charAt(i - 1) - '0';
        for (int i = n; i >= 2; i--)
            for (int j = 1; j <= n + 1 - i; j++) a[j] = Math.abs(a[j] - a[j + 1]);
        System.out.println(a[1]);
    }
}
