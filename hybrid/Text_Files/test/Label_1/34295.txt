import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();

        int[] capacities = new int[k];
        for (int i = 0; i < k; i++) {
            capacities[i] = sc.nextInt();
        }

        int size = 1; long max = 0; int type = 0;
        for (int i = 0; i < k; i++) {
            long cur = (n / capacities[i]);
            if (cur > max) {
                max = cur;
                size = capacities[i];
                type = i + 1;
            }
        }

        System.out.println(type + " " + max);
    }
}
