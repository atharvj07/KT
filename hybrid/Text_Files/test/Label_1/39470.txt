t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    total_substrings = n * (n + 1) // 2
    zero_substrings = (n - m) * ((n - m) + 1) // 2
    parts = m + 1
    min_zeros_per_part = (n - m) // parts
    extra_zeros = (n - m) % parts
    max_value = total_substrings - (parts - extra_zeros) * min_zeros_per_part * (min_zeros_per_part + 1) // 2 - extra_zeros * (min_zeros_per_part + 1) * (min_zeros_per_part + 2) // 2
    print(max_value)
