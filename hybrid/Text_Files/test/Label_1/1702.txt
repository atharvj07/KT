#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;
const int LOG = 30;
int n, a[MAXN], cnt[LOG];
bool f;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    int v = 0;
    while ((1 << v) < a[i]) v++;
    cnt[v]++;
  }
  for (int j = 0; j < LOG; j++) {
    if (cnt[j]) {
      if (f) cnt[j + 1] += cnt[j] / 2;
      f |= (cnt[j] & 1);
      cnt[j] &= 1;
    }
  }
  cout << cnt[LOG - 1] + f << endl;
  return 0;
}
