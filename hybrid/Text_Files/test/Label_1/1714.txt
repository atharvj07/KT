#include <bits/stdc++.h>

using namespace std;

const int N = 501;
int pre[N][N], suf[N][N];

int main() {
    int n, k;
    cin >> n >> k;
    int sum = n * (n + 1) / 2;
    cout << sum << endl;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            pre[i][j] = pre[i - 1][j] + pre[i][j - 1] - pre[i - 1][j - 1];
            suf[i][j] = suf[i - 1][j] + suf[i][j - 1] - suf[i - 1][j - 1];
            int t;
            if (i % 2 == j % 2) {
                t = pre[i][j] + (i + j - 1) * (n - j + 1) + suf[n][k] - suf[i][k] - suf[n][j] + suf[i][j];
            } else {
                t = pre[i][j] + (i + j - 1) * (j - 1) + suf[n][k] - suf[i][k] - suf[n][j] + suf[i][j];
            }
            if (t == sum) {
                for (int ii = 1; ii <= n; ii++) {
                    for (int jj = 1; jj <= n; jj++) {
                        int tt;
                        if (ii % 2 == jj % 2) {
                            tt = pre[ii][jj] + (ii + jj - 1) * (n - jj + 1) + suf[n][k] - suf[ii][k] - suf[n][jj] + suf[ii][jj];
                        } else {
                            tt = pre[ii][jj] + (ii + jj - 1) * (jj - 1) + suf[n][k] - suf[ii][k] - suf[n][jj] + suf[ii][jj];
                        }
                        if (tt == t) cout << pre[ii][jj] << " ";
                    }
                    cout << endl;
                }
                return 0;
            }
        }
    }
    cerr << "impossible" << endl;
    return 0;
}
