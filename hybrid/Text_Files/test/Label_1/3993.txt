#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll ma = 200005;
ll arr[ma], o[ma], c[ma], dp[ma], sum[ma], ans = 0, n;
int main() {
    scanf("%lld", &n);
    for (ll i = 0; i<n; i++) {
        string x;
        cin >> x;
        arr[i] = x.size();
    }
    for (ll i = 0; i<n; i++) {
        if (arr[i] % 2 == 0) {
            o[i] = arr[i] / 2;
            c[i] = arr[i] / 2;
        } else {
            o[i] = arr[i] / 2;
            c[i] = (arr[i] / 2) + 1;
        }
    }
    for (ll i = 0; i<n; i++) {
        if (o[i] == 0) {
            continue;
        }
        memset(dp, 0, sizeof(dp));
        dp[i] = o[i];
        sum[i] = c[i];
        ans = max(ans, dp[i]);
        for (ll j = i + 1; j<n; j++) {
            if (o[i] == c[j] && o[i] != 0) {
                dp[j] = dp[i] + o[j];
            } else if (c[i] == o[j] && o[i] != 0) {
                dp[j] = dp[i] + c[j];
            }
            sum[j] = sum[i] + c[j];
            ans = max(ans, dp[j]);
        }
        for (ll j = i + 1; j<n; j++) {
            if (sum[j]>dp[j]) {
                dp[j] = 0;
            } else {
                ll temp = sum[j] - dp[j];
                dp[j] = dp[j] - temp;
            }
            ans = max(ans, dp[j]);
        }
    }
    printf("%lld\n", ans);
    return 0;
}
