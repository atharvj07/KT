def solve():
    test_num = int(input())
    for _ in range(test_num):
        days = int(input())
        arr = list(map(int, input().split()))
        height = 1
        died = False
        for i, water in enumerate(arr):
            if i - 2 > -1 and not arr[i - 1] and not arr[i - 2]:
                died = True
                break
            if water:
                height += 5 if i > 0 and arr[i - 1] else 1
        print(-1 if died else height)

solve()
