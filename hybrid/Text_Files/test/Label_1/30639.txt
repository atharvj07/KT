import java.io.BufferedReader;
import java.io.InputStreamReader;

public class Main {

  public static long f(long n, long k, long p) {
    if (k == 0) {
      return n;
    }
    if (k % 2 == 1) {
      return (f(n, k / 2, p) * 2 % p + n) % p;
    }
    return (f(n, k / 2, p) * f(n, k / 2, p) % p) % p;
  }

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] s = br.readLine().split(" ");
    long n = Long.parseLong(s[0]);
    long x = Long.parseLong(s[1]);
    long y = Long.parseLong(s[2]);
    long p = Long.parseLong(s[3]);
    s = br.readLine().split(" ");
    long sum = 0;
    for (int i = 0; i < n; i++) {
      sum = (sum + Long.parseLong(s[i])) % p;
    }
    System.out.println(((sum * f(2, x + y, p)) % p - sum) % p);
  }
}
