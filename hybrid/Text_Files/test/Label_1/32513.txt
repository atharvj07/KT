import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] f = new int[n+1], rf = new int[n+2];
        Arrays.fill(f, 1);
        Arrays.fill(rf, 1);
        for (int i=2; i<=n; ++i) {
            f[i] = (int)((long)i*f[i-1]%1000000007);
        }
        for (int i=1; i<=n; ++i) {
            rf[i] = (int)((long)rf[i-1]*f[i]%1000000007);
        }
        int ans = (int)((long)f[n]*rf[n-1]%1000000007);
        System.out.println(ans);
    }
}
