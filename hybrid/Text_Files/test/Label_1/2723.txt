#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int n, l, x, y;
  cin >> n >> l >> x >> y;
  vector<int> a(n + 1);
  for (int i = 0; i <= n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  vector<int> ans;
  if (a.back() < y) { ans.push_back(y); }
  if (a[1] > x) { ans.push_back(x); }
  for (int i = 1; i < n; i++) {
    if (a[i + 1] - a[i] > y - x) {
      ans.push_back(a[i] + x);
      break;
    }
  }
  cout << ans.size() << endl;
  for (auto e : ans) { cout << e << ' '; }
  cout << endl;
  return 0;
}
