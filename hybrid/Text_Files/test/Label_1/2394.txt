#include <cstdio>
#include <algorithm>
using namespace std;
int n, a[1005];
int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    sort(a, a + n);
    int sum = 0;
    int cnt0 = 0, cnt5 = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (a[i] == 0) cnt0++;
        else cnt5++;
        sum += a[i];
    }
    if (sum == 0) {
        printf("0\n");
        return 0;
    }
    if ((sum % 9 != 0) || (cnt0 == n)) {
        printf("-1\n");
        return 0;
    }
    if (cnt5 == n) {
        for (int i = 0; i < n; i++) printf("5");
        printf("\n");
        return 0;
    }
    for (int i = n - 1; i >= 0; i--) {
        if (a[i] == 0) {
            if (cnt0 == 1) continue;
            printf("0");
            cnt0--;
        } else {
            printf("5");
        }
    }
    printf("\n");
}
