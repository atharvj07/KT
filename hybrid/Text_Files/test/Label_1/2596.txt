#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int p[n];
    for (int i = 0; i < n; i++) p[i] = i;
    sort(p, p + n, [&](int x, int y) { return a[x] < a[y]; });
    int ans = 0;
    for (int i = 0; i < n; i++) if (p[i] != i) ans++;
    cout << ans << "\n";
    for (int i = 0; i < n; i++) if (p[i] != i) cout << i << " " << p[i] << "\n";
    return 0;
}
