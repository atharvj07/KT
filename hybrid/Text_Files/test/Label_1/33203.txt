import java.util.*;

public class Main {

    static class Edge {
        int a, b;
        long w;
        Edge(int a, int b, long w) {
            this.a = a; this.b = b; this.w = w;
        }
    }

    static int n;
    static int[] a;
    static List<Edge> edges = new ArrayList<>();
    static HashMap<Integer, Integer> c = new HashMap<>();

    static long dfs1(int u, int p) {
        long ans = a[u - 1];
        c.put(u, 1);
        for (Edge edge : edges) {
            if (edge.a != u && edge.b != u) continue;
            int v = edge.a == u ? edge.b : edge.a;
            if (v == p) continue;
            ans += dfs1(v, u);
            c.put(u, c.get(u) + c.get(v));
        }
        return ans;
    }

    static long dfs2(int u, int p, long sum, long dist, long d) {
        long ans = (sum + (n - 2 - c.get(u)) * a[u - 1]) * d + (a[u - 1] * c.get(u) - sum) * dist;
        for (Edge edge : edges) {
            if (edge.a != u && edge.b != u) continue;
            int v = edge.a == u ? edge.b : edge.a;
            if (v == p) continue;
            ans += dfs2(v, u, sum + c.get(u) * a[u - 1], dist, (dist + 1) << 1);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        for (int i = 0; i < n - 1; i++) {
            int u = in.nextInt(), v = in.nextInt();
            edges.add(new Edge(u, v, 0));
        }
        dfs1(1, 1);
        System.out.println(dfs2(1, 1, 0, 0, 1));
    }
}
