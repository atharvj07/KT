import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
        }
        Arrays.sort(m);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (x >= m[i]) {
                ans++;
                x -= m[i];
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
