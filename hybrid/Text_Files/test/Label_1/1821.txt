#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 2e5 + 5;
int n, a[N], ans[N], g[N];
vector<int> adj[N];

void dfs(int u, int p) {
    for (auto &v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        g[u] = __gcd(g[u], g[v]);
    }
    g[u] = __gcd(g[u], a[u]);
    ans[u] = max(ans[u], g[u]);
    for (auto &v : adj[u]) {
        if (v == p) continue;
        ans[v] = max(ans[v], a[u]);
    }
}

signed main() {
    ios::sync_with_stdio(false), cin.tie(nullptr);
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i], ans[i] = a[i]; }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, -1);
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    cout << "\n";
    return 0;
}
