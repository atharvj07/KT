import java.util.*;

public class Main {

    static int getNumberOfGoodSubarrays(int[] a) {
        int n = a.length;
        int[][] sumXOR = new int[n][n];
        for (int i = 0; i < n; i++) {
            sumXOR[i][i] = a[i];
            for (int j = i + 1; j < n; j++) {
                sumXOR[i][j] = a[j] ^ sumXOR[i][j - 1];
            }
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 2; j < n; j++) {
                if ((sumXOR[i][j - 1] ^ sumXOR[i][j]) == sumXOR[i + 1][j - 1]) {
                    count++;
                }
            }
        }
        return count;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        System.out.println(getNumberOfGoodSubarrays(a));
    }
}
