import sys
input = sys.stdin.readline

for _ in range(int(input())):
    h, n = map(int, input().split())
    a = sorted([[-(h - ll), ll] for ll in map(int, input().split())] + [[-2, 1]])
    ans = 0
    i = 0
    while i + 1 < len(a):
        if a[i][1] + 2 != a[i + 1][1]:
            ans += 1
            i += 1
        i += 1
    print(ans)
