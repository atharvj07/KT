MOD = 998244353

def modinv(a, m):
    return pow(a, m - 2, m)

n = int(input())
total_wishes = 0
item_counts = {}
for _ in range(n):
    k, *items = map(int, input().split())
    total_wishes += k
    for item in items:
        if item not in item_counts:
            item_counts[item] = 0
        item_counts[item] += 1

valid_decisions = sum(v * v for v in item_counts.values())
probability = valid_decisions * modinv(total_wishes * n, MOD) % MOD
print(probability)
