#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        int cnt0 = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '0') cnt0++;
        }
        if (cnt0 == n) {
            cout << "NO\n";
            continue;
        }
        int fst = 0, lst = n - 1;
        while (fst < n && s[fst] == '0') fst++;
        while (lst >= 0 && s[lst] == '0') lst--;
        if (fst == n || lst == -1) {
            cout << "NO\n";
            continue;
        }
        string sub = s.substr(fst, lst - fst + 1);
        int f20 = sub.find("20");
        int f2020 = sub.find("2020");
        if (f20 != -1 || f2020 != -1) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}
