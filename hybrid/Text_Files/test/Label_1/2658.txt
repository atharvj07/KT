#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n), a(n);
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    a[0] = b[n - 1];
    a[n - 1] = b[0];
    int j = 1, k = n - 2;
    for (int i = 1; i < n - 1; i++) {
      if (i % 2 == 1) {
        a[j] = b[i];
        j++;
      } else {
        a[k] = b[i];
        k--;
      }
    }
    for (int i = 0; i < n; i++) { cout << a[i] << " "; }
    cout << endl;
  }
  return 0;
}
