import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();

        long[] fib = new long[46];
        fib[0] = 0;
        fib[1] = 1;
        for (long i = 2; i <= n; i++) {
            fib[(int) i] = fib[(int) (i - 1)] + fib[(int) (i - 2)];
        }

        System.out.println(fib[(int) n]);
    }
}
