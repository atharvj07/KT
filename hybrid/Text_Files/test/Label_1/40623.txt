from collections import defaultdict

def find_parent(parent, i):
    if parent[i] == i:
        return i
    return find_parent(parent, parent[i])

def union(parent, rank, x, y):
    xroot = find_parent(parent, x)
    yroot = find_parent(parent, y)
    if rank[xroot] < rank[yroot]:
        parent[xroot] = yroot
    elif rank[xroot] > rank[yroot]:
        parent[yroot] = xroot
    else:
        parent[yroot] = xroot
        rank[xroot] += 1

n, m, k = map(int, input().split())
govs = list(map(int, input().split()))

parent = list(range(n + 1))
rank = [0] * (n + 1)
for _ in range(m):
    u, v = map(int, input().split())
    union(parent, rank, u, v)

components = defaultdict(int)
for i in range(1, n + 1):
    components[find_parent(parent, i)] += 1

max_component = 0
total_nodes = 0
for gov in govs:
    gov_component = components[find_parent(parent, gov)]
    if gov_component > max_component:
        max_component = gov_component
    total_nodes += gov_component

total_nodes -= max_component
max_edges = (max_component * (max_component - 1)) // 2 + (total_nodes * (total_nodes + 1)) // 2
max_edges += total_nodes * max_component
max_edges -= m

print(max_edges)
