#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, l, x, y;
  cin >> n >> l >> x >> y;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  if (a[n - 1] < y) {
    cout << 1 << endl;
    cout << y << endl;
    return 0;
  }
  if (a[0] >= x) {
    cout << 0 << endl;
    return 0;
  }
  int j = 0;
  while (a[j] < x) j++;
  cout << 1 << endl;
  cout << a[j] - x << endl;
  return 0;
}
