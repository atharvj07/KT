from sys import stdin

n = int(stdin.readline())
price = []
have = [set() for i in range(3)]
for i in range(n):
    c, s = stdin.readline().split()
    c = int(c)
    for j in list(s):
        have[ord(j)-ord('A')].add(i)
    price.append((c, i))

have[0] = have[0].intersection(*have[1:])
if not have[0]: print(-1)
else:
    have[0] = min(price[i][0] for i in have[0])
    print(have[0])
