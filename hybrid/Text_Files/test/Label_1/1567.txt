#include <iostream>
#include <string>
#include <map>
using namespace std;
int main() {
  map<char, int> a;
  char suit[3];
  int num[3];
  for (int i = 0; i < 3; i++) {
    cin >> suit[i] >> num[i];
    a[suit[i]]++;
  }
  if (a.size() == 1 || (a.size() == 2 && (a['m'] > 0) && (a['p'] > 0))) {
    cout << 0 << endl;
    return 0;
  }
  map<char, int>::iterator it;
  int max = 0;
  for (it = a.begin(); it != a.end(); it++)
    if (it->second > max) max = it->second;
  if (a.size() == 3)
    cout << (3 - max) << endl;
  else
    cout << (2 - max) << endl;
  return 0;
}
