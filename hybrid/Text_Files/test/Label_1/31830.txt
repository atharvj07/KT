import java.util.Scanner;

public class Main {
  static final long MOD = 1000000007;
  static int maxS = 52;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    long[][] f = new long[maxS][maxS];
    long[] g = new long[maxS];

    f[0][0] = g[0] = 1;
    for (int i = 1; i <= n; i++) {
      for (int j = 0; j <= i - 1; j++)
        f[i][j] = (f[i][j] + g[j] * f[i - 1][j] % MOD) % MOD;
      g[i] = i * f[i][i] % MOD;
      for (int j = i + 1; j <= i - 1; j++)
        f[i][j] = (f[i][j] + g[j] * f[i - 1][j - 1] % MOD) % MOD;
    }
    long ans = (f[n][m] - f[n][m - 1] + MOD) % MOD;
    System.out.println(ans);
  }
}
