#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 10;
const int inf = 1e9;
int n, a[maxn], l[maxn], r[maxn];
int main() {
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    a[i] = s[i - 1] == '*';
    cnt += a[i] == 1;
  }
  l[1] = r[n] = inf;
  int suml = 0, sumr = 0;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == 'P') {
      l[i] = suml;
      suml = 0;
    } else
      suml += (s[i - 1] == '*');
  }
  for (int i = n; i >= 1; i--) {
    if (s[i - 1] == 'P') {
      r[i] = sumr;
      sumr = 0;
    } else
      sumr += (s[i - 1] == '*');
  }
  int ans = inf;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == 'P')
      ans = min(ans, l[i] + r[i]);
  }
  if (ans == inf)
    ans = -1;
  cout << ans << endl;
  return 0;
}
