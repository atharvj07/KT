import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arrivalTimes = new int[n];
        for (int i = 0; i < n; i++) {
            arrivalTimes[i] = scanner.nextInt() * 60 + scanner.nextInt();
        }
        Arrays.sort(arrivalTimes);
        int maxConcurrentCustomers = 1;
        int currentConcurrentCustomers = 1;
        int currentTime = arrivalTimes[0];
        for (int i = 1; i < n; i++) {
            if (arrivalTimes[i] == currentTime) {
                currentConcurrentCustomers++;
            } else {
                maxConcurrentCustomers = Math.max(maxConcurrentCustomers, currentConcurrentCustomers);
                currentConcurrentCustomers = 1;
                currentTime = arrivalTimes[i];
            }
        }
        maxConcurrentCustomers = Math.max(maxConcurrentCustomers, currentConcurrentCustomers);
        System.out.println(maxConcurrentCustomers);
    }
}
