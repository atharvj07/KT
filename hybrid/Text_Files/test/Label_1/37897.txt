from collections import defaultdict

n = int(input())
edges = list(map(lambda x: tuple(map(int, x.split())), [input() for _ in range(n - 1)]))
g = defaultdict(set)
for u, v in edges:
    g[u].add(v)
    g[v].add(u)

p = [1 / n] * n
for _ in range(n - 1):
    np = [0] * n
    for u in g:
        for v in g[u]:
            np[u - 1] += p[v - 1]
    p = [0.5 * sum(u) for u in zip(p, np)]
print(*p)
