#include <bits/stdc++.h>

using namespace std;

int n;
int a[200005], res[200005];
vector<vector<int> > g(200005);

int dfs(int u, int p) {
  int w = a[u], b = (a[u] ? 0 : 1);
  for(int v: g[u]) {
    if(v == p) continue;
    w += dfs(v, u);
    b += dfs(v, u);
  }
  res[u] = w - b;
  return a[u];
}

int main() {
  cin >> n;
  for(int i = 1; i <= n; i++) cin >> a[i];
  for(int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
    g[v].push_back(u);
  }
  dfs(1, 1);
  for(int i = 1; i <= n; i++) cout << res[i] << ' ';
  return 0;
}
