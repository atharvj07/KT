#include <iostream>
#include <algorithm>

int main() {
    long long n, k;
    std::cin >> n >> k;

    long long a[n];
    for (long long &x : a)
        std::cin >> x;

    std::sort(a, a + n);

    long long ans = a[n - 1] - a[0];

    if (k >= n - 1)
        ans = 0;
    else if (k >= (n - 1) / 2)
        ans /= 2;
    else
        ans -= k * 2;

    std::cout << ans << std::endl;

    return 0;
}
