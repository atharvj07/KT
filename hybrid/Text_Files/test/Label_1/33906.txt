import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            long k = scanner.nextLong();
            int d = 0;
            while (k > d * 9 * (d + 1) / 2) {
                k -= d * 9 * (d + 1) / 2;
                d++;
            }
            k--;
            long m = k / d;
            long n = k % d;
            int ans = (int) (m + 1 + n);
            System.out.println(ans % 10);
        }
    }
}
