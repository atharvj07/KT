import sys
input = sys.stdin.readline

def dfs(u, par, res):
    if u in res:
        return res[u]
    res[u] = par
    for v in adj[u]:
        if v != par:
            res[u] = dfs(v, u, res)
    return res[u]

n = int(input())
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)
res = dfs(1, -1, {})
print("YES" if res == {} else "NO")
