#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int s = 0, t = 0, ans[1001];
  for (int i = 1, x; i <= n; i++) {
    cin >> x;
    s += x;
    if (s == i) {
      ans[t++] = x;
      s = 0;
    }
  }
  cout << t << endl;
  for (int i = 0; i < t; i++) cout << ans[i] << ' ';
  return 0;
}
