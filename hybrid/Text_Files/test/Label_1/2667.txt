#include <iostream>
#include <vector>
#include <algorithm>
#include <tuple>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> w(n);
    for (int i = 0; i < n; i++) { cin >> w[i]; }
    sort(w.begin(), w.end());
    string s; cin >> s;
    vector<int> ans(2 * n);
    int j = 0;
    for (int i = 0; i < 2 * n; i += 2) {
        if (s[i] == '0') { ans[i] = w[j++]; }
        else {
            int l = j, r = n - 1;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (w[mid] == ans[i - 1]) {
                    l = mid + 1;
                } else { ans[i] = w[mid], j++; r = mid - 1; }
            }
        }
        ans[i + 1] = ans[i];
    }
    for (int i = 0; i < 2 * n; i++) { cout << ans[i] << ' '; }
    cout << endl;
}
