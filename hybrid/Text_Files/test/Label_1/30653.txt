import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] argv) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int k = in.nextInt();
    int[] a = new int[k];
    for (int i = 0; i < k; i++) a[i] = in.nextInt();
    int l = in.nextInt();
    int[] b = new int[l];
    for (int i = 0; i < l; i++) b[i] = in.nextInt();
    in.close();
    Arrays.sort(a);
    Arrays.sort(b);
    boolean flag = true;
    for (int i = 0; i < k; i++) {
      if (a[i] < i + 1) {
        flag = false;
        break;
      }
    }
    for (int i = 0; i < l; i++) {
      if (b[i] < n - i) {
        flag = false;
        break;
      }
    }
    if (flag) System.out.println("YES");
    else System.out.println("NO");
  }
}
