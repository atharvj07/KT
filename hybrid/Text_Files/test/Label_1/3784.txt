#include<bits/stdc++.h>

using namespace std;

int n;
string str;
bool chk[4];

int main()
{
    cin>>n;

    while(n--){
        int x;
        cin>>x>>str;

        if(str=="North") chk[1]=1;
        if(str=="South") chk[0]=1,chk[2]=0;
        if(str=="East") chk[3]=1;
        if(str=="West") chk[2]=1;
    }

    for(int i=0;i<4;i++){
        if(chk[i]==0){
            cout<<"NO"<<endl;
            return 0;
        }
    }

    cout<<"YES"<<endl;

    return 0;
}
