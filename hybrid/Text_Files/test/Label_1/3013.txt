#include <bits/stdc++.h>
using namespace std;
#define MAXN 12
#define vi vector<int>
#define pb push_back
vi r[MAXN], c[MAXN];
int main() {
  int n, m, p, k;
  cin >> n >> m >> p >> k;
  for (int i = 0; i < p; i++) {
    int a, b;
    cin >> a >> b;
    r[a].pb(b);
    c[b].pb(a);
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    sort(r[i].begin(), r[i].end());
    for (int j = 1; j <= m; j++) {
      sort(c[j].begin(), c[j].end());
      int cnt = 0;
      for (int a : r[i]) {
        if (a == j) cnt++;
        if (cnt == k) break;
      }
      for (int a : c[j]) {
        if (a == i) cnt++;
        if (cnt == k) break;
      }
      if (cnt >= k) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
