n, k = map(int, input().split())
a = list(map(int, input().split()))
stck = []
b = []
for x in a:
    while stck and stck[-1] > x:
        b.append(stck.pop())
    stck.append(x)
while stck:
    b.append(stck.pop())
sz = len(b)
if sz != n - k:
    print(-1)
else:
    for x in a:
        b.remove(x)
    ans = a + sorted(b)
    print(*ans)
