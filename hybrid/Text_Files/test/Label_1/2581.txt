#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m), b(m);
  for (int i = 0; i < m; i++) {
    cin >> a[i] >> b[i];
  }
  bool flag = false;
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      bool aok = false, bok = false;
      for (int k = 0; k < m; k++) {
        if (j == a[k]) aok = true;
        if (j == b[k]) bok = true;
        if (i == a[k]) aok = true;
        if (i == b[k]) bok = true;
      }
      if (aok && bok) {
        flag = true;
        break;
      }
    }
  }
  if (flag)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
