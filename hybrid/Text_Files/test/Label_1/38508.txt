from collections import defaultdict, deque
s=input();g=defaultdict(list);lst=[]
for i in range(len(s)):
    g[s[i]].append(i)
    lst.append(i)
u=deque();c=0;ans=lst.copy()
for i in s:
    if u and u[-1]>g[i][0]:
        c+=1
        u.clear()
    u.append(g[i][0])
    g[i].pop(0)
    ans[u[-1]]=c+1
print(c+1)
print(*ans)
