import sys

m = int(input())
s = ""
nst = [0] * 4
mod = 10**9 + 7
for i in range(m):
    c = sys.stdin.read(1)
    s += c
    nst[len(s)] += 1
    nst[len(s)] -= int(''.join(reversed(s)) in ('0011', '0101', '1110', '1111'))
    tot = 0
    for i in nst:
        tot += (1 << i) - 1
    print(tot % mod)
