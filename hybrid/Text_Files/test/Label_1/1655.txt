#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
      cin >> a[i];
    }
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < n; ++i) {
      sum += a[i];
    }
    cout << (sum + n - 1) / n << endl;
  }
  return 0;
}
