import sys

n = int(input())
adj = [[] for i in range(n+1)]
visited = [False] * (n+1)
dp = [1.0] * (n+1)
size = [0] * (n+1)
for i in range(n-1):
    a,b = map(int,input().split())
    adj[a].append(b)
    adj[b].append(a)

def dfs(u,p):
    size[u] = 1
    for v in adj[u]:
        if v == p or visited[v]:
            continue
        visited[v] = True
        dfs(v,u)
        size[u] += size[v]
        dp[u] += dp[v] + size[v]

visited[1] = True
dfs(1,1)
print(dp[1] / n)
