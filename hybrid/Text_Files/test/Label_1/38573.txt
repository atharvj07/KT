import sys
input = sys.stdin.readline

N, M, s, t = map(int, input().split())
adj = [[] for i in range(N+1)]
for i in range(M):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

dis = [0] * (N+1)
vis = [0] * (N+1)
def dfs1(x, d):
    vis[x] = 1
    dis[x] = d
    for y in adj[x]:
        if vis[y] == 0:
            dfs1(y, d+1)
dfs1(s, 0)

ans = 0
def dfs2(x, d):
    global ans
    vis[x] = 1
    dis[x] = d
    for y in adj[x]:
        if vis[y] == 0:
            dfs2(y, d+1)
            if dis[s] >= dis[t]:
                ans += 1
dfs2(t, 0)
print(ans)
