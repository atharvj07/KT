def calculate_score(minutes, max_points):
    if minutes == -1:
        return 0
    return max_points * (1 - minutes / 250)

def calculate_points(n, submissions, solvers):
    scores = []
    for i in range(5):
        solvers_count = solvers[i]
        if solvers_count > n / 2:
            max_points = 500
        elif solvers_count > n / 4:
            max_points = 1000
        elif solvers_count > n / 8:
            max_points = 1500
        elif solvers_count > n / 16:
            max_points = 2000
        elif solvers_count > n / 32:
            max_points = 2500
        else:
            max_points = 3000
        scores.append([calculate_score(submissions[j][i], max_points) for j in range(n)])
    return scores

def min_new_accounts_to_win(n, submissions):
    solvers = [0] * 5
    for i in range(n):
        for j in range(5):
            if submissions[i][j] != -1:
                solvers[j] += 1

    base_scores = calculate_points(n, submissions, solvers)
    vasya_score, petya_score = sum(base_scores[i][0] for i in range(5)), sum(base_scores[i][1] for i in range(5))
    if vasya_score > petya_score:
        return 0

    for new_accounts in range(1, 10**5):
        new_solvers = [solvers[i] + (1 if submissions[0][i] != -1 else new_accounts) for i in range(5)]
        new_scores = calculate_points(n + new_accounts, submissions, new_solvers)
        new_vasya_score = sum(new_scores[i][0] for i in range(5))
        if new_vasya_score > petya_score:
            return new_accounts

    return -1

n = int(input().strip())
submissions = [list(map(int, input().strip().split())) for _ in range(n)]
print(min_new_accounts_to_win(n, submissions))
