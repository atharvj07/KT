#include <iostream>
#include <vector>

using namespace std;

int main() {
  vector<int> factors(101, 0);
  int a;
  while (cin >> a) factors[a] += 1;

  int min_factor = 101;
  for (int i = 0; i < 101; i++) {
    if (factors[i] == 0) continue;
    min_factor = min(min_factor, i);
  }

  if (min_factor == 101) min_factor = -1;
  cout << min_factor << endl;

  return 0;
}
