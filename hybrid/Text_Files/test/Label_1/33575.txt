import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt(), a = in.nextInt(), b = in.nextInt(), v = in.nextInt();
        int ans = (a + v - 1) / v;
        if (b + k >= ans) ans = 1;
        else ans = (ans + k - b - 1) / k + 1;
        System.out.println(ans);
    }
}
