#include <iostream>
#include <vector>
const int64_t mod = 998244353;
int main() {
  int64_t n, k;
  std::cin >> n >> k;
  std::vector<int64_t> dp(n + 1);
  dp[2] = 6;
  for (int i = 3; i <= n; i++) {
    dp[i] = (dp[i - 2] * (dp[i - 1] * 2 % mod - 2) % mod + 2) % mod;
  }
  int64_t ans = ((dp[n] - 2) * (dp[n] - 2) % mod + 2) % mod;
  k -= 2;
  while (k > 0) {
    int64_t t = 1;
    for (int i = 1; i <= k; i++) {
      t = (t * i) % mod;
    }
    ans = (ans - t + mod) % mod;
    k -= 2;
  }
  std::cout << ans << "\n";
  return 0;
}
