MOD = 10**9 + 7

def char_to_int(c):
    if '0' <= c <= '9':
        return ord(c) - ord('0')
    if 'A' <= c <= 'Z':
        return ord(c) - ord('A') + 10
    if 'a' <= c <= 'z':
        return ord(c) - ord('a') + 36
    if c == '-':
        return 62
    if c == '_':
        return 63

def count_bits(n):
    count = 0
    while n:
        count += n & 1
        n >>= 1
    return count

def solve(s):
    bit_counts = [0] * 64
    for c in s:
        bit_counts[char_to_int(c)] += 1

    result = 1
    for c in s:
        total = 0
        target = char_to_int(c)
        for i in range(64):
            if i & target == target:
                total += bit_counts[i]
        result = (result * total) % MOD
    return result

s = input().strip()
print(solve(s))
