#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int inf = 1e9;

using vi = vector<int>;
using vvi = vector<vi>;

struct DSU {
  vi par, siz;
  int comps;
  DSU(int n) {
    par.resize(n + 1);
    iota(par.begin(), par.end(), 0);
    siz.assign(n + 1, 1);
    comps = n;
  }
  int find(int x) {
    if (x != par[x]) par[x] = find(par[x]);
    return par[x];
  }
  bool join(int a, int b) {
    a = find(a), b = find(b);
    if (a == b) return false;
    if (siz[a] >= siz[b])
      par[b] = a, siz[a] += siz[b];
    else
      par[a] = b, siz[b] += siz[a];
    comps--;
    return true;
  }
  bool same_comp(int a, int b) { return find(a) == find(b); }
};

vvi g;
int n, m;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> m;
  g.resize(n + 1);
  DSU dsu(n);
  while (m--) {
    int a, b;
    cin >> a >> b;
    g[a].push_back(b);
    g[b].push_back(a);
    dsu.join(a, b);
  }
  if (dsu.comps > 1)
    cout << "NO\n";
  else {
    cout << "YES\n";
    for (int i = 2; i <= n; i++) {
      int u = 1, v = i, p = 1;
      while (u != v) {
        if (!dsu.same_comp(u, v)) { p = 0; break; }
        u = g[u][0];
        v = g[v][0];
      }
      cout << p;
    }
    cout << '\n';
  }
}
