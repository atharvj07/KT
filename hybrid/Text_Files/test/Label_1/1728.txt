#include <bits/stdc++.h>
using namespace std;
int n, m;
vector<vector<bool>> g;
vector<bool> vis;
void dfs(int u) {
  if (vis[u]) return;
  vis[u] = 1;
  for (int i = 1; i <= n; i++) {
    if (i == u || !g[u][i]) continue;
    dfs(i);
  }
}
int main() {
  cin >> n >> m;
  g.resize(n + 1, vector<bool>(n + 1, 0));
  while (m--) {
    int x, y;
    cin >> x >> y;
    g[x][y] = 1;
  }
  int ans = 0;
  vis.resize(n + 1, 0);
  for (int i = 1; i <= n; i++) {
    if (!vis[i]) {
      dfs(i);
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
