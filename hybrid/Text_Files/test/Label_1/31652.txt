import java.util.*;

public class GeoRepaintings {
    static int n, m;
    static int[][] colors, symbols;
    static int[] dx = {-1, 0, 1, 0}, dy = {0, -1, 0, 1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        colors = new int[n][m];
        symbols = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                colors[i][j] = sc.nextInt();
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                symbols[i][j] = sc.nextInt();
            }
        }

        int x = sc.nextInt() - 1;
        int y = sc.nextInt() - 1;

        System.out.println(solve(x, y));
    }

    static int solve(int x, int y) {
        Deque<int[]> q = new ArrayDeque<>();
        q.add(new int[]{x, y});
        boolean[][] visited = new boolean[n][m];
        int repaintings = 0;

        while (!q.isEmpty()) {
            int[] current = q.pollFirst();
            int cx = current[0];
            int cy = current[1];

            if (visited[cx][cy]) continue;
            visited[cx][cy] = true;

            if (colors[cx][cy] != 0 && colors[cx][cy] != symbols[cx][cy]) {
                repaintings++;
                for (int i = 0; i < dx.length; i++) {
                    int nx = cx + dx[i];
                    int ny = cy + dy[i];
                    if (nx >= 0 && nx < n && ny >= 0 && ny < m && colors[nx][ny] == colors[cx][cy]) {
                        q.add(new int[]{nx, ny});
                    }
                }
            }

            if (symbols[cx][cy] != -1) {
                symbols[cx][cy] = -1;
                q.add(new int[]{cx, cy});
            }
        }

        return repaintings;
    }
}
