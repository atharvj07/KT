#include <bits/stdc++.h>
using namespace std;
const int MXN = 1e6 + 5;
int n, pre[MXN][2];
int32_t main() {
    cin >> n;
    string s, t;
    cin >> s >> t;
    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j < 2; j++) {
            pre[i][j] = pre[i + 1][j];
            pre[i][s[i] - '0']++;
        }
    }
    int ans = INT_MAX;
    for (int i = 0; i < n; i++) {
        int flip = pre[0][t[i] - '0'] - pre[i + 1][t[i] - '0'];
        int non_flip = i - pre[i][t[i] - '0'];
        ans = min({ans, flip, non_flip});
    }
    if (ans == INT_MAX) ans = -1;
    cout << ans << '\n';
}
