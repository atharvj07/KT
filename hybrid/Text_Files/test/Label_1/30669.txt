import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int l = sc.nextInt();
    long ans = 0;
    boolean positive = false;
    for (int i = 0; i < n; i++) {
      int a = l + i;
      if (i != n - 1) {
        if (positive && a < 0) {
          ans += l + i - 1;
        } else if (!positive && a > 0) {
          ans += l + i + 1;
          positive = true;
        } else {
          ans += a;
        }
      } else {
        if (i % 2 == 0) {
          ans += a;
        }
      }
    }
    System.out.println(ans);
  }
}
