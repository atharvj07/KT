#include <bits/stdc++.h>
using namespace std;

#define pb push_back

vector<pair<long long, long long>> h, f;
long long manhattan = 0;

int main() {
    long long n, m, c, h1;
    cin >> n >> m >> c;
    for (int i = 0; i < c; ++i) {
        long long x, y;
        cin >> x >> y;
        manhattan = max(manhattan, max(x, n - x + 1) + max(y, m - y + 1));
        f.pb({x, y});
    }
    cin >> h1;
    for (int i = 0; i < h1; ++i) {
        long long x, y;
        cin >> x >> y;
        manhattan = min(manhattan, max(x, n - x + 1) + max(y, m - y + 1));
        h.pb({x, y});
    }

    long long ans, ansh;
    for (auto h_i : h) {
        long long tmp = 1LL * 1e18;
        for (auto f_i : f) {
            tmp = min(tmp, abs(h_i.first - f_i.first) + abs(h_i.second - f_i.second));
        }
        if (tmp < ans || !ans) {
            ans = tmp;
            ansh = distance(h.begin(), find(h.begin(), h.end(), h_i));
        }
    }

    cout << ans << '\n' << ansh + 1 << '\n';

    return 0;
}
