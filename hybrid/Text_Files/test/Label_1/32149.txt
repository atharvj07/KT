import java.util.Scanner;

public class BeautifulString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();
            String s = scanner.next();

            String result = findBeautifulString(s, k);
            System.out.println(result);
        }
    }

    private static String findBeautifulString(String s, int k) {
        // Check if any substring of length k is all 0s.
        for (int i = 0; i <= s.length() - k; i++) {
            boolean allZeros = true;
            for (int j = 0; j < k; j++) {
                if (s.charAt(i + j) == '1') {
                    allZeros = false;
                    break;
                }
            }
            if (allZeros) {
                return "NO";
            }
        }

        // Build the lexicographically smallest beautiful string.
        char[] result = new char[k];
        for (int i = 0; i < k; i++) {
            result[i] = '1';
        }

        // Check each character in the string.
        for (int i = 0; i < s.length() - k; i++) {
            for (int j = 0; j < k; j++) {
                if (s.charAt(i + j) == '0' && result[j] == '1') {
                    result[j] = '0';
                }
            }
        }

        return "YES\n" + new String(result);
    }
}
