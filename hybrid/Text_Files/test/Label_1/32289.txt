import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if ((n & 1) == 0)
            System.out.println(-1);
        else if ((n / 2) % 2 == 0)
            System.out.println(n / 4 + 1);
        else
            System.out.println(n / 2);
    }
}
