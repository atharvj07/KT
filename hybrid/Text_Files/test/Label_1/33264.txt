import java.util.*;
import java.io.*;

class Main {
    static int[] p;

    static int root(int x) {
        if (x == p[x])
            return x;
        return p[x] = root(p[x]);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int m = in.nextInt();

        p = new int[n + 1];
        for (int i = 1; i <= n; i++)
            p[i] = i;

        int[] color = new int[n + 1];
        color[1] = 1;
        for (int i = 0; i < n - 1; i++) {
            int a = in.nextInt();
            int b = in.nextInt();

            if (root(a) != root(b))
                p[root(b)] = root(a);
        }

        for (int i = 0; i < m; i++) {
            int type = in.nextInt();
            int x = in.nextInt();

            if (type == 1)
                color[x] = 1;
            else {
                int min = n;
                for (int j = 1; j <= n; j++) {
                    if (color[j] == 1)
                        min = Math.min(min, getDist(j, x));
                }
                System.out.println(min == n ? -1 : min);
            }
        }
    }

    static int getDist(int a, int b) {
        int ra = root(a);
        int rb = root(b);

        if (ra == rb)
            return getLevel(a) + getLevel(b);
        return -1;
    }

    static int getLevel(int x) {
        int level = 0;
        int y = x;
        while (y != root(y)) {
            y = root(y);
            level++;
        }
        return level;
    }
}
