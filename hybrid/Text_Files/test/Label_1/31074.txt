import java.util.Scanner;

public class TownTraversal {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int a1 = scanner.nextInt();
        int b1 = scanner.nextInt();
        int a2 = scanner.nextInt();
        int b2 = scanner.nextInt();
        int a3 = scanner.nextInt();
        int b3 = scanner.nextInt();

        // Check if the roads form a connected graph
        boolean[][] connected = new boolean[5][5];
        connected[a1][b1] = connected[b1][a1] = true;
        connected[a2][b2] = connected[b2][a2] = true;
        connected[a3][b3] = connected[b3][a3] = true;

        boolean isConnected = true;
        for (int i = 1; i <= 4; i++) {
            for (int j = 1; j <= 4; j++) {
                if (i != j && !connected[i][j]) {
                    isConnected = false;
                    break;
                }
            }
        }

        // Print the result
        if (isConnected) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
