import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = scanner.nextInt();
        }

        int[] dp = new int[m];
        Arrays.fill(dp, 1);
        for (int i = 1; i < m; i++) {
            for (int j = 0; j < i; j++) {
                if (b[i] > b[j] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                }
            }
        }

        int maxLen = 0;
        int maxIdx = 0;
        for (int i = 0; i < m; i++) {
            if (dp[i] > maxLen) {
                maxLen = dp[i];
                maxIdx = i;
            }
        }

        System.out.println(maxLen);
        int[] stairs = new int[maxLen];
        int idx = maxIdx;
        for (int i = maxLen - 1; i >= 0; i--) {
            stairs[i] = b[idx];
            idx--;
        }

        for (int i = 0; i < maxLen; i++) {
            System.out.print(stairs[i] + " ");
        }
        System.out.println();
    }
}
