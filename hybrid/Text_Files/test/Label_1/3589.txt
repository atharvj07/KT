#include <iostream>
#include <vector>
using namespace std;
vector<int> divs[300005];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        char c;
        cin >> n >> c;
        string s;
        cin >> s;
        int ans = 0;
        vector<int> xs;
        for (int x = 2; x <= n; x++) {
            bool ok = true;
            for (int y : divs[x]) {
                if (s[y - 1] != s[y] || s[y] != c) {
                    ok = false;
                }
            }
            if (ok) {
                xs.push_back(x);
                for (int y : divs[x]) { s[y - 1] = c; }
                ans++;
            }
        }
        cout << ans << endl;
        for (int x : xs) { cout << x << " "; }
        cout << endl;
    }
    return 0;
}
