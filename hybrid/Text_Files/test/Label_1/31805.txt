import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }

        int[] change = new int[n + 1];
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (a[i] != i) {
                change[i]++;
            }
            if (p[a[i]] == 0) {
                p[a[i]] = i;
            } else {
                change[i]++;
            }
        }

        int min = n;
        for (int i = 1; i <= n; i++) {
            min = Math.min(min, change[i]);
        }

        System.out.println(min);
        for (int i = 1; i <= n; i++) {
            if (change[i] == min) {
                System.out.print(p[a[i]] + " ");
            } else {
                System.out.print(a[i] + " ");
            }
        }
    }
}
