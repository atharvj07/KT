#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e9 + 10;

int n, m, ans;

int main() {
    cin >> n >> m;
    if ((n + m) % 3 == 0) {
        ans = (n + m) / 3;
    } else if ((n + m) % 3 == 2) {
        ans = (n + m) / 3;
        if (n > m)
            n -= 3;
        else
            m -= 3;
        if (n > 0 && m > 0) ans++;
    } else if ((n + m) % 3 == 1) {
        n -= 4, m -= 4;
        if (n < 0 || m < 0)
            ans = 0;
        else
            ans = (n + m) / 3 + 2;
    }
    cout << ans << endl;
    return 0;
}
