from math import factorial

n = int(input())
a = list(map(int, input().split()))

mod = 10**9 + 7

# dp[i][j] stores the sum of f(a) over all permutations of a[i:j]
dp = [[0] * (n+1) for _ in range(n+1)]

# base case: dp[i][i+1] = 0 for all i
for i in range(n):
    dp[i][i+1] = 0

# inductive case: dp[i][j] = dp[i][k] + dp[k+1][j] for all i < k < j
for l in range(2, n+1):
    for i in range(n-l+1):
        j = i + l
        for k in range(i, j):
            dp[i][j] = (dp[i][j] + dp[i][k] + dp[k+1][j] + a[k]) % mod

# the answer is dp[0][n]
print(dp[0][n])
