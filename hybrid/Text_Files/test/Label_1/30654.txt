import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] days = new int[n * 2];
        for (int i = 0; i < n; i++) {
            days[2 * i] = scanner.nextInt();
            days[2 * i + 1] = scanner.nextInt();
        }
        Arrays.sort(days);
        int day = 0;
        for (int i = 0; i < n; i++) {
            if (days[day++] < days[2 * i + 1]) {
                day = 2 * i + 1;
            }
        }
        System.out.println(day % 2 == 1 ? days[day] : -1);
    }
}
