#include <bits/stdc++.h>
#define int long long
using namespace std;
const int inf = 1ll << 60;
const int Maxn = 300005;
vector<int> g[Maxn];
vector<int> w[Maxn];
int d[Maxn];
void dijkstra() {
    priority_queue<pair<int, int>> pq;
    memset(d, 0x3f, sizeof d);
    d[1] = 0;
    pq.push({0, 1});
    while (pq.size()) {
        int cur = pq.top().second;
        int dis = -pq.top().first;
        pq.pop();
        if (dis > d[cur]) continue;
        for (int i = 0; i < g[cur].size(); i++) {
            int u = g[cur][i];
            int ndis = dis + w[cur][i];
            if (ndis < d[u]) {
                d[u] = ndis;
                pq.push({-ndis, u});
            }
        }
    }
}
int32_t main() {
    int n, m, k;
    cin >> n >> m >> k;
    while (m--) {
        int u, v, weight;
        cin >> u >> v >> weight;
        g[u].push_back(v);
        g[v].push_back(u);
        w[u].push_back(weight);
        w[v].push_back(weight);
    }
    dijkstra();
    vector<int> ans;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < g[i].size(); j++) {
            int u = g[i][j];
            int ndis = d[u] + w[i][j];
            if (ndis == d[i]) {
                if (ans.size() < k) ans.push_back(j);
            }
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) { cout << ans[i] + 1 << " "; }
}
