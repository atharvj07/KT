import heapq
n, T = map(int, input().split())
a = [0] * (n+1)
t = [0] * (n+1)
for i in range(1, n+1):
    a[i], t[i] = map(int, input().split())
q = [(t[i],i) for i in range(1, n+1)]
cnt = 0
cur = 0
ans = []
heapq.heapify(q)
while q and (cur + q[0][0] <= T):
    tmp = heapq.heappop(q)
    cur += tmp[0]
    if cnt < a[tmp[1]]:
        cnt += 1
        ans.append(tmp[1])
print(cnt)
print(len(ans))
print(*ans)
