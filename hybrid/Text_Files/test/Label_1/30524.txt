import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        String t = sc.next();
        int[] ans = solve(s, t);
        if (ans == null) {
            System.out.println(-1);
            return;
        }
        System.out.println(ans.length);
        for (int i : ans) {
            System.out.print(i + " ");
        }
        System.out.println();
    }

    public static int[] solve(String s, String t) {
        int n = s.length();
        int[] lps = new int[n + 1];
        lps[0] = 0;
        int i = 1;
        int j = 0;
        while (i < n) {
            if (s.charAt(i) == s.charAt(j)) {
                lps[i] = j + 1;
                i++;
                j++;
            } else {
                if (j == 0) {
                    lps[i] = 0;
                    i++;
                } else {
                    j = lps[j - 1];
                }
            }
        }
        i = 0;
        j = 0;
        List<Integer> ans = new ArrayList<>();
        while (i < n) {
            if (s.charAt(i) == t.charAt(j)) {
                i++;
                j++;
            } else {
                if (j == 0) {
                    ans.add(n - i);
                    i++;
                } else {
                    j = lps[j - 1];
                }
            }
            if (j == n) {
                break;
            }
        }
        if (j != n) {
            return null;
        }
        int[] ansArr = new int[ans.size()];
        for (int k = 0; k < ans.size(); k++) {
            ansArr[k] = ans.get(k);
        }
        return ansArr;
    }
}
