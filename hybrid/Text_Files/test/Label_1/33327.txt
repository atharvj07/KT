import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] deg = new int[n];
        for (int i = 0; i < n - 1; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            deg[x - 1]++;
            deg[y - 1]++;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) ans += deg[i] * (long)(deg[i] - 1);
        System.out.println(ans / 2);
    }
}
