import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class BerlandElections {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Voter> voters = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int preferredParty = scanner.nextInt();
            int cost = scanner.nextInt();
            voters.add(new Voter(preferredParty, cost));
        }

        // Sort the voters in ascending order of cost
        Collections.sort(voters);

        // Initialize the variables for tracking the number of votes for each party
        int[] partyVotes = new int[m + 1];
        int maxVotes = 0;

        // Iterate through the sorted voters
        for (Voter voter : voters) {
            // Increase the vote count for the preferred party
            partyVotes[voter.preferredParty]++;

            // Update the maximum vote count
            maxVotes = Math.max(maxVotes, partyVotes[voter.preferredParty]);

            // Check if the United Party can win by buying this voter's vote
            if (partyVotes[1] + 1 > maxVotes) {
                System.out.println(voter.cost);
                return;
            }
        }

        // If the United Party cannot win by buying votes, output -1
        System.out.println(-1);
    }

    static class Voter implements Comparable<Voter> {
        int preferredParty;
        int cost;

        Voter(int preferredParty, int cost) {
            this.preferredParty = preferredParty;
            this.cost = cost;
        }

        @Override
        public int compareTo(Voter other) {
            return Integer.compare(cost, other.cost);
        }
    }
}
