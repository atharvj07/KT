from collections import defaultdict
import sys
input = sys.stdin.readline

def ccw(p1, p2, p3):
    return p1[0] * (p2[1] - p3[1]) + p2[0] * (p3[1] - p1[1]) + p3[0] * (p1[1] - p2[1])

def is_convex(hull):
    n = len(hull)
    for i in range(n):
        if ccw(hull[i], hull[(i + 1) % n], hull[(i + 2) % n]) < 0:
            return False
    return True
