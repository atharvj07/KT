from collections import defaultdict
n = int(input())
s = [int(x) for x in input().split()]
cnt = defaultdict(int)
st = 0
en = 0
ans = 0
occs = 0
for x in s:
    cnt[x] += 1
    if cnt[x] <= occs + 1:
        en += 1
    else:
        while occs < cnt[x]:
            cnt[s[st]] -= 1
            if cnt[s[st]] < occs:
                occs -= 1
            st += 1
        cnt[x] -= 1
        if cnt[x] <= occs:
            en += 1
        else:
            occs += 1
            en += 1
    ans = max(ans, en - st)
print(ans)
