import java.util.Arrays;
import java.util.Scanner;
import java.util.List;
import java.util.ArrayList;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int m = input.nextInt();
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = input.nextInt();
        }
        int[][] dp = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dp[i][j] = 0;
            }
        }
        int maxLen = 0;
        int maxLenIndex = -1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (a[i] == b[j]) {
                    dp[i][j] = 1;
                    if (i > 0 && j > 0 && dp[i - 1][j - 1] > 0) {
                        dp[i][j]++;
                    }
                    if (dp[i][j] > maxLen) {
                        maxLen = dp[i][j];
                        maxLenIndex = i;
                    }
                }
            }
        }
        List<Integer> lcs = new ArrayList<>();
        while (maxLenIndex >= 0) {
            lcs.add(a[maxLenIndex]);
            maxLenIndex--;
        }
        System.out.println(maxLen);
        for (int i = lcs.size() - 1; i >= 0; i--) {
            System.out.print(lcs.get(i) + " ");
        }
    }
}
