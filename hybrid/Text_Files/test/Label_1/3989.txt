#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1005;

ll ar[MAXN];
ll hashes[MAXN][MAXN];
ll p = 1000000007;

void cal_hash(const int n) {
  for (int i = 1; i <= n; i++) hashes[i][i] = ar[i];
  for (int len = 2; len <= n; len++) {
    for (int i = 1; i <= n - len + 1; i++) {
      hashes[i][i + len - 1] = ((hashes[i][i + len - 2] * p) % p + ar[i + len - 1]) % p;
    }
  }
}

bool check(const int n) {
  for (int l1 = 1; l1 <= n; l1++) {
    for (int r1 = l1; r1 <= n; r1++) {
      for (int l2 = r1 + 1; l2 <= n; l2++) {
        for (int r2 = l2; r2 <= n; r2++) {
          if (hashes[l1][r1] == hashes[l2][r2]) return true;
        }
      }
    }
  }
  return false;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
      cin >> ar[i];
    }
    cal_hash(n);
    if (check(n)) cout << "YES" << endl;
    else cout << "NO" << endl;
  }
}
