n = int(input())
a = list(map(int, input().split()))

b = a[::-1]

f = [[1 for i in range(n+1)] for j in range(n+1)]

for i in range(1, n+1):
    for j in range(1, n+1):
        if a[i-1] <= b[j-1]:
            f[i][j] = f[i-1][j-1] + 1
        f[i][j] = max(f[i][j], f[i-1][j], f[i][j-1])

print(f[n][n])
