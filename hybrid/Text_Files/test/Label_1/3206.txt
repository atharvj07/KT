int arr[3][N];
struct FenwickTree
{
    int bit[N];
    void update(int idx, int val)
    {
        for (; idx <= N; idx += idx & -idx)
            bit[idx] += val;
    }
    int query(int idx)
    {
        int sum = 0;
        for (; idx > 0; idx -= idx & -idx)
            sum += bit[idx];
        return sum;
    }
} ft;
int get(int l, int r)
{
    return ft.query(r) - (l > 0 ? ft.query(l - 1) : 0);
}
