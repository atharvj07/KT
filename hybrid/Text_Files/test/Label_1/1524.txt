#include <vector>
#include <iostream>
#include <algorithm>

struct TextBreak
{
    size_t pos;
    bool is_zero_width;
    TextBreak() = default;
    TextBreak(size_t pos_, bool is_zero_width_) : pos(pos_), is_zero_width(is_zero_width_) {}
};

std::vector<TextBreak> get_breaks(const std::string& text)
{
    std::vector<TextBreak> result;

    size_t line_start = 0;
    size_t line_end = 0;

    while (line_start < text.size()) {
        line_end = line_start;
        bool is_hyphen_found = false;
        while (line_end < text.size() && ((text[line_end] == '-' && !is_hyphen_found) || text[line_end] != ' ')) {
            if (text[line_end] == '-') {
                is_hyphen_found = true;
            }
            line_end++;
        }
        result.emplace_back(is_hyphen_found ? line_end - 1 : line_end, is_hyphen_found);
        line_start = line_end + 1;
    }

    return result;
}

int get_width(const std::string& text, size_t max_width)
{
    auto breaks = get_breaks(text);
    std::reverse(breaks.begin(), breaks.end());

    int w = 0;
    int h = 0;
    for (size_t i = 0; i < breaks.size(); i++) {
        const auto& break_ = breaks[i];
        size_t line_width = break_.is_zero_width ? 0 : break_.pos + 1;

        if (w + line_width <= max_width) {
            w += line_width;
        } else {
            h++;
            w = line_width;
        }
    }

    h++;

    return std::max(w, h);
}

int main()
{
    size_t max_width;
    std::cin >> max_width;

    std::string text;
    std::getline(std::cin, text);
    std::getline(std::cin, text);

    std::cout << get_width(text, max_width);

    return 0;
}
