#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll d[100001];
ll f[100001];
main() {
  ll n, k, ans = 0;
  cin >> n >> k;
  ll sum = 0;
  for (ll i = 0; i < n; i++) {
    ll x;
    cin >> x;
    d[i] = x;
    sum += x;
  }
  f[n] = sum;
  for (ll i = n - 1; i >= 0; i--) {
    f[i] = f[i + 1] + d[i];
  }
  for (ll i = 0; i < n; i++) {
    ll x;
    x = f[i + 1] + (k - 1) * d[i];
    ans = max(ans, x);
  }
  cout << ans << "\n";
}
