#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
  ll n, k, x, mx = 0, mn = 1e9, a[100001];
  cin >> n >> k >> x;
  for (ll i = 0; i < n; ++i) { cin >> a[i]; }
  sort(a, a + n);
  for (ll i = 0; i < n; i += 2) {
    if (k % 2 == 0)
      continue;
    a[i] ^= x;
  }
  for (ll i = 0; i < n; i++) {
    mx = max(mx, a[i]);
    mn = min(mn, a[i]);
  }
  cout << mx << " " << mn << '\n';
  return 0;
}
