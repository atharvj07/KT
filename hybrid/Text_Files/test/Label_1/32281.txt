import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        BigInteger[] a = new BigInteger[n];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = new BigInteger(st.nextToken());
        }

        Map<BigInteger, Integer> freq = new HashMap<>();
        BigInteger gcd = new BigInteger("0");
        for (BigInteger bi : a) {
            freq.put(bi, freq.getOrDefault(bi, 0) + 1);
            gcd = gcd.gcd(bi);
        }
        if (gcd.equals(BigInteger.ZERO)) gcd = new BigInteger("1");
        for (Map.Entry<BigInteger, Integer> entry : freq.entrySet()) {
            int count = entry.getValue();
            BigInteger f = factorial(count);
            gcd = gcd.multiply(f).mod(new BigInteger("998244353"));
        }
        System.out.println(gcd);
    }
    public static BigInteger factorial(int n) {
        BigInteger f = new BigInteger("1");
        for (int i = 1; i <= n; i++) {
            f = f.multiply(new BigInteger(String.valueOf(i)));
        }
        return f;
    }
}
