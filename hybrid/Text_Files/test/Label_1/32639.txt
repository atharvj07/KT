import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int m = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }
            boolean[] p = new boolean[n];
            for (int j = 0; j < m; j++) {
                p[scanner.nextInt() - 1] = true;
            }
            boolean sorted = true;
            for (int j = 0; j < n - 1; j++) {
                if (a[j] > a[j + 1]) {
                    if (!p[j]) {
                        sorted = false;
                        break;
                    }
                    int tmp = a[j];
                    a[j] = a[j + 1];
                    a[j + 1] = tmp;
                    j--;
                }
            }
            System.out.println(sorted ? "YES" : "NO");
        }
    }
}
