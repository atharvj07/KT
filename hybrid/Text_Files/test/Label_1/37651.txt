from collections import defaultdict, deque
import sys
MOD = 10**9 + 7

def bfs(graph, root, n):
    queue = deque([root])
    sizes = [0] * (n + 1)
    visited = [False] * (n + 1)
    visited[root] = True
    while queue:
        node = queue.pop()
        sizes[node] = 1
        for neighbor in graph[node]:
            if not visited[neighbor]:
                visited[neighbor] = True
                queue.appendleft(neighbor)
                sizes[node] += sizes[neighbor]
    return sizes

def solve(n, edges, prime_factors):
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    sizes = bfs(graph, 1, n)
    edge_weights = [min(sizes[u], sizes[v]) * (n - max(sizes[u], sizes[v])) for u, v in edges]
    edge_weights.sort(reverse=True)
    
    prime_factors.sort()
    while len(prime_factors) > len(edge_weights):
        max_factor = prime_factors.pop()
        prime_factors[-1] *= max_factor
        prime_factors[-1] %= MOD
    
    for i in range(len(prime_factors)):
        edge_weights[i] = (edge_weights[i] * prime_factors[i]) % MOD
    
    for i in range(len(prime_factors), len(edge_weights)):
        edge_weights[i] = (edge_weights[i] * 1) % MOD
    
    return sum(edge_weights) % MOD

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    edges = [list(map(int, input().strip().split())) for _ in range(n-1)]
    m = int(input().strip())
    prime_factors = list(map(int, input().strip().split()))
    print(solve(n, edges, prime_factors))
