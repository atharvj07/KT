#include <iostream>
#include <vector>

using namespace std;

typedef long long int llint;
#define MOD 1000000009

llint powmod(llint a, llint b) {
    if (b == 0) return 1;
    if (b == 1) return a;
    llint ans = powmod(a, b / 2);
    if (b % 2 == 1) return ans * ans % MOD * a % MOD;
    else return ans * ans % MOD;
}

llint fact[250001];
llint invfact[250001];

llint modinv(llint x) {
    return powmod(x, MOD - 2);
}

void pre() {
    fact[0] = 1;
    for (int i = 1; i < 250001; i++) {
        fact[i] = i * fact[i - 1];
        fact[i] %= MOD;
    }
    
    invfact[0] = 1;
    for (int i = 1; i < 250001; i++) {
        invfact[i] = modinv(fact[i]);
    }
}

int main() {
    llint n, m;
    cin >> n >> m;

    pre();

    llint ans = fact[n];

    for (int i = 1; i <= n; i++) {
        ans -= fact[n - i] * invfact[n - 2 * i] % MOD;
        ans %= MOD;
        ans += MOD;
        ans %= MOD;
    }

    cout << ans << endl;

    return 0;
}
