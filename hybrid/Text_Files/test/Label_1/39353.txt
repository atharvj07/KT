n,c,d = input().split(' ')
n = int(n);c = int(c);d = int(d)
f = []
for i in range(n):
    b,p,t = input().split(' ')
    b = int(b);p = int(p)
    if t=='D': f.append([b,p,1])
    else: f.append([b,p,0])
f.sort()
f.reverse()
ans,k = 0,1
for i in f:
    if i[2]==0 and k<=2 and i[1]<=c:
        c -= i[1]
        ans += i[0]
        k += 1
    elif i[2]==1 and k<=2 and i[1]<=d:
        d -= i[1]
        ans += i[0]
        k += 1
    else: break
if ans==0:print(0)
else:print(ans)
