#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll oo = 1e18;
int n, a[100005], col[100005], low[100005], mxcol, cy, vs[100005];
vector<int> adj[100005];
ll ans = oo;
void vis(int u, int p){
    col[u] = mxcol;
    low[u] = mxcol++;
    vs[u] = 1;
    for(int v: adj[u])
        if(!col[v]){
            vis(v, u);
            if(low[v] >= col[u])
                cy = 1;
            low[u] = min(low[u], low[v]);
            if(low[v] > col[u])
                ans = min(ans, (ll)1 + low[v] - col[u]);
        }
        else if(v != p && low[v] >= col[u])
            cy = 1;
}
int32_t main(){
    cin >> n;
    for(int i=1;i<=n;++i) cin >> a[i];
    for(int i=1;i<=n;++i)
        for(int j=i+1;j<=n;++j)
            if(a[i] & a[j])
                adj[i].push_back(j), adj[j].push_back(i);
    for(int i=1;i<=n;++i)
        if(!col[i]){
            vis(i, i);
            if(cy == 0) ans = -1;
        }
    cout << ans;
    return 0;
}
