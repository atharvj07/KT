def max_interesting_words(t, test_cases):
    results = []

    for n, words in test_cases:
        word_counts = [[word.count(char) for char in "abcde"] for word in words]
        max_count = 0
        
        for i in range(5):
            diffs = [2*word_count[i] - sum(word_count) for word_count in word_counts]
            diffs.sort(reverse=True)
            count, total = 0, 0
            
            for diff in diffs:
                if total + diff > 0:
                    count += 1
                    total += diff
                else:
                    break
            max_count = max(max_count, count)
        
        results.append(max_count)
    return results

# Reading input and calling the function
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    words = [input().strip() for _ in range(n)]
    test_cases.append((n, words))

results = max_interesting_words(t, test_cases)
for result in results:
    print(result)
