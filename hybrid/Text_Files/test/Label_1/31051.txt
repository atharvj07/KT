import java.util.Arrays;
import java.util.Scanner;

public class ValerieShark {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt();
            teeth[i][1] = scanner.nextInt();
        }

        Arrays.sort(teeth, (t1, t2) -> t2[1] - t1[1]);

        int maxCrucians = 0;
        int[] rowUsage = new int[m];
        for (int[] tooth : teeth) {
            int row = tooth[0] - 1;
            int viability = tooth[1];

            int remainingViability = viability - rowUsage[row];
            if (remainingViability > 0) {
                int cruciansToEat = Math.min(remainingViability, k - maxCrucians);
                rowUsage[row] += cruciansToEat;
                maxCrucians += cruciansToEat;
            }
        }

        System.out.println(maxCrucians);
    }
}
