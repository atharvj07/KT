#include <iostream>
#include <map>
using namespace std;
map<string, int>mp;
int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        int h, m, ans;
        cin >> h >> m;
        mp["hh"] = h, mp["mm"] = m;
        if(mp["hh"] == 0 && mp["mm"] > 0) ans = (1440 - mp["mm"]);
        else if(mp["hh"] > 0 || mp["mm"] > 0) ans = ((24 - mp["hh"] - 1) * 60 + (60 - mp["mm"]));
        cout << ans << endl;
    }
}
