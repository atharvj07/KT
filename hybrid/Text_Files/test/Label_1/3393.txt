#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  int change = 0;
  for (int i : a) {
    if (i == 25)
      change++;
    else if (i == 50) {
      change++;
      change--;
    } else {
      if (change >= 2) {
        change -= 2;
        change++;
      } else if (change == 1) {
        change -= 1;
      } else {
        cout << "NO" << endl;
        return 0;
      }
    }
  }
  cout << "YES" << endl;
  return 0;
}
