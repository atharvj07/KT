#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e5 + 7;
int a[N], par[N];
vector<int> adj[N];
ll sub[N], ans[N];

void dfs(int node) {
    sub[node] = a[node];
    for (int child : adj[node]) {
        if (child != par[node]) {
            par[child] = node;
            dfs(child);
            sub[node] += sub[child];
        }
    }
    ans[node] = sub[node];
}

void dfs2(int node) {
    for (int child : adj[node]) {
        if (child != par[node]) {
            ans[child] = ans[node] - 2 * sub[child] + sub[node];
            dfs2(child);
        }
    }
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1);
    dfs2(1);
    for (int i = 1; i <= n; i++) cout << ans[i] << " ";
    return 0;
}
