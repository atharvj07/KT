import java.util.*;

class Main {
    static int n, p;
    static List<Pipe>[] G;
    static int[] type;
    static List<Pair> ans;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        p = sc.nextInt();
        G = new List[n + 1];
        ans = new ArrayList<>();
        for (int i = 1; i <= n; i++) G[i] = new ArrayList<>();
        for (int i = 0; i < p; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            G[a].add(new Pipe(b, c));
            G[b].add(new Pipe(a, c));
        }
        type = new int[n + 1];
        Arrays.fill(type, -1);
        for (int i = 1; i <= n; i++) {
            if (type[i] == -1) dfs(i, i);
        }
        System.out.println(ans.size());
        for (Pair pair : ans) {
            System.out.println(pair.a + " " + pair.b + " " + pair.c);
        }
    }

    static void dfs(int u, int root) {
        type[u] = root;
        for (Pipe pipe : G[u]) {
            if (type[pipe.to] != root) {
                if (type[pipe.to] == -1) {
                    Pair pair = new Pair(root, pipe.to, pipe.w);
                    ans.add(pair);
                    dfs(pipe.to, root);
                }
            }
        }
    }

    static class Pipe {
        int to, w;

        public Pipe(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    static class Pair {
        int a, b, c;

        public Pair(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }
}
