#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> a(n), b(n);
    unordered_map<string, int> cnta, cntb;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        cnta[a[i]]++;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i];
        cntb[b[i]]++;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            if (a[i][0] != b[i][0]) ans++;
            else {
                if (a[i].size() == b[i].size()) ans++;
                else ans += 2;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
