import java.io.*;
import java.util.*;

class Main {
    static final int B = (int)1e9 + 7;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] a = new int[n][n];
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            StringTokenizer tokenizer = new StringTokenizer(line);
            for (int j = 0; j < n; j++) {
                a[i][j] = Integer.parseInt(tokenizer.nextToken());
            }
        }

        boolean[][] can = new boolean[n][n];
        int[] deg = new int[n];
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if ((i != j) && (i != k) && (j != k) && a[i][k] > 0 && a[k][j] > 0) {
                        can[i][j] = true;
                        deg[i]++;
                        deg[j]++;
                    }
                }
            }
        }

        boolean[] vis = new boolean[n];
        long ans = 1;
        for (int v = 0; v < n; v++) {
            if (!vis[v]) {
                int cur = 0;
                long sum = 0;
                for (int i = 0; i < n; i++) {
                    if (!vis[i] && can[v][i]) {
                        cur++;
                        sum += deg[i];
                        vis[i] = true;
                    }
                }
                sum = (sum - 2 * cur + 2) % B;
                if (sum < 0) sum += B;
                ans = (ans * sum) % B;
            }
        }

        System.out.println(ans);
    }
}
