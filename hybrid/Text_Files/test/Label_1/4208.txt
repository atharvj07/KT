#include <cstdio>

int main() {
    char a[4][5];
    int cnt = 0;
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            scanf(" %c", &a[i][j]);
            cnt += (a[i][j] == '#');
        }
    }
    if (cnt & 1)
        printf("NO");
    else
        printf("YES");
}
