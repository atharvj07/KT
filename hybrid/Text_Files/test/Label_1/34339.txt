import java.util.*;

public class Main {
    static int[] color;
    static List<Integer>[] adj;
    static int[] components;
    static int[] componentSize;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();

        while (t-- > 0) {
            int n = input.nextInt();

            color = new int[n + 1];
            adj = new List[n + 1];

            for (int i = 0; i <= n; i++) {
                adj[i] = new ArrayList<>();
            }

            for (int i = 1; i <= n; i++) {
                color[i] = input.nextInt();
            }

            for (int i = 0; i < n - 1; i++) {
                int u = input.nextInt();
                int v = input.nextInt();
                adj[u].add(v);
                adj[v].add(u);
            }

            components = new int[n + 1];
            componentSize = new int[n + 1];
            int componentCount = 0;

            for (int i = 1; i <= n; i++) {
                if (components[i] == 0) {
                    dfs1(i, ++componentCount);
                }
            }

            int minOperations = n;

            for (int i = 1; i <= componentCount; i++) {
                if (componentSize[i] > 1) {
                    minOperations = Math.min(minOperations, componentSize[i]);
                } else {
                    minOperations = Math.min(minOperations, componentSize[i] + 1);
                }
            }

            System.out.println(minOperations);
        }
    }

    static void dfs1(int node, int component) {
        components[node] = component;
        componentSize[component]++;

        for (int next : adj[node]) {
            if (components[next] == 0) {
                dfs1(next, component);
            }
        }
    }
}
