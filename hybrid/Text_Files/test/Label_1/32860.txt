import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Election {

    static class Voter {
        int party;
        int cost;

        Voter(int party, int cost) {
            this.party = party;
            this.cost = cost;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Voter[] voters = new Voter[n];
        for (int i = 0; i < n; i++) {
            voters[i] = new Voter(scanner.nextInt(), scanner.nextInt());
        }

        Arrays.sort(voters, Comparator.comparingInt(v -> v.cost));

        int[] votes = new int[m + 1];
        for (Voter voter : voters) {
            votes[voter.party]++;
        }

        int maxVotes = 0;
        for (int i = 1; i <= m; i++) {
            maxVotes = Math.max(maxVotes, votes[i]);
        }

        int minCost = 0;
        for (int i = 0; i < n && votes[1] <= maxVotes; i++) {
            votes[voters[i].party]--;
            votes[1]++;
            minCost += voters[i].cost;
            maxVotes = Math.max(maxVotes, votes[1]);
        }

        System.out.println(minCost);
    }
}
