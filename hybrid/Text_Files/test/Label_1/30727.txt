import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] s = new int[n];
        int cars = 0;
        for (int i = 0; i < n; i++) {
            s[i] = in.nextInt();
            cars += (s[i] + 3) / 4;
        }
        System.out.println(cars);
    }
}
