from collections import defaultdict, deque
import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

for i in range(1, n + 1): g[i].sort()

vis = [False] * (n + 1)
f = []

def dfs(v, p, max_deg):
    vis[v] = True
    f.append((v, p))
    deg = 1
    if p != -1: deg += 1
    for u in g[v]:
        if vis[u]:
            if u != p or deg == 1:
                max_deg = max(max_deg, deg)
            continue
        max_deg = max(max_deg, dfs(u, v, deg))
    return max_deg

max_deg = 0
for i in range(1, n + 1):
    if vis[i]: continue
    max_deg = max(max_deg, dfs(i, -1, 0))

print(max_deg)

for u, p in f:
    if p == -1: continue
    print(u, p)
