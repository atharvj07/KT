import sys

input = sys.stdin.readline

n, w, v, u = map(int, input().split())
xa, ya = [], []
for _ in range(n):
    x, y = map(int, input().split())
    xa.append(x)
    ya.append(y)
xa.append(xa[0])
ya.append(ya[0])
mn = w / u
for i in range(n):
    x1 = xa[i]
    y1 = ya[i]
    x2 = xa[i + 1]
    y2 = ya[i + 1]
    if x2 - x1 == 0:
        continue
    a = (y2 - y1) / (x2 - x1)
    b = y1 - a * x1
    if a * v <= u and b <= w and b >= 0:
        t = -x1 * a / (u * a + v)
        if t > 0 and t < mn:
            mn = t
print("{:.9f}".format(mn))
