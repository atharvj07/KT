int main() {
    int q;
    cin >> q;
    while (q--) {
        int n, r;
        cin >> n >> r;
        vector<long long int> v(n);
        for (int i = 0; i < n; i++) {
            cin >> v[i];
        }
        sort(v.begin(), v.end());
        int ans = 0;
        long long int cur = -r - 1;
        for (int i = 0; i < n; i++) {
            if (v[i] <= cur) {
                continue;
            }
            ans++;
            cur = v[i] + r;
        }
        cout << ans << endl;
    }
    return 0;
}
