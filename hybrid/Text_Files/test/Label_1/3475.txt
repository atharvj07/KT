#include<bits/stdc++.h>
using namespace std;

#define ll long long
const int mod = 1e9 + 7;

void solve() {
	ll n, i, ans = 0;
	cin >> n;
	priority_queue<ll, vector<ll>, greater<ll> > pq;
	for (i = 0; i < n; i++) {
		ll x;
		cin >> x;
		pq.push(x);
	}
	while (pq.size() > 1) {
		ll a = pq.top(); pq.pop();
		ll b = pq.top(); pq.pop();
		ans += (a - b);
		pq.push(a - b);
	}
	cout << ans << "\n";
}

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	ll t = 1;
	while (t--) {
		solve();
	}
	return 0;
}
