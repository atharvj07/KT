import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] a = sc.next().toCharArray();
        char[] b = sc.next().toCharArray();
        int[][][] dp = new int[n + 1][n + 1][n + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                Arrays.fill(dp[i][j], -1);
            }
        }
        System.out.println(go(a, b, 0, 0, dp, 0));
    }

    static int go(char[] a, char[] b, int ai, int bi, int[][][] dp, int op) {
        if (ai == a.length) {
            if (bi == b.length) {
                return op;
            } else {
                return -1;
            }
        }
        if (dp[ai][bi][op] != -1) {
            return dp[ai][bi][op];
        }
        int ans = -1;
        if (a[ai] == b[bi]) {
            ans = go(a, b, ai + 1, bi + 1, dp, op);
        }
        for (int i = ai; i < a.length; i++) {
            int cnt = 0;
            for (int j = bi; j < b.length; j++) {
                if (a[i] == b[j]) {
                    cnt++;
                }
            }
            if (cnt > 0) {
                int nxt = go(a, b, i + 1, bi, dp, op + 1);
                if (nxt != -1) {
                    ans = nxt;
                    break;
                }
            }
        }
        dp[ai][bi][op] = ans;
        return ans;
    }
}
