#include <bits/stdc++.h>
using namespace std;

#define ll long long
const ll INF = 1e18;

int main() {
    int t; cin >> t;
    while (t--) {
        ll x, y; cin >> x >> y;
        ll c[6];
        for (ll &i : c) cin >> i;
        ll dir[2] = {y, x - y};
        ll ans = INF;
        for (ll i = 0; i < 6; i++) {
            ll tmp[2] = {dir[0] - (i % 3), dir[1] - ((6 - i) % 3)};
            if (tmp[0] >= 0 && tmp[1] >= 0) ans = min(ans, c[i] * (tmp[0] + tmp[1]));
        }
        cout << ans << endl;
    }
    return 0;
}
