import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] start = scanner.nextLine().split(":");
        String[] end = scanner.nextLine().split(":");
        int h1 = Integer.parseInt(start[0]), m1 = Integer.parseInt(start[1]);
        int h2 = Integer.parseInt(end[0]), m2 = Integer.parseInt(end[1]);
        int m = (m1 + m2) / 2;
        int h = h1 + (m1 + m2) / 60;
        h += m / 60;
        m %= 60;
        System.out.printf("%02d:%02d", h, m);
    }
}
