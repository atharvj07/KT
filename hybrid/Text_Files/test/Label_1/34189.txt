import java.util.HashMap;
import java.util.Scanner;

public class MinimalSegment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int l = 0, r = 0;
        int minLength = Integer.MAX_VALUE;
        int distinctCount = 0;
        HashMap<Integer, Integer> frequencyMap = new HashMap<>();

        for (r = 0; r < n; r++) {
            frequencyMap.put(a[r], frequencyMap.getOrDefault(a[r], 0) + 1);
            if (frequencyMap.get(a[r]) == 1) {
                distinctCount++;
            }

            while (distinctCount > k) {
                frequencyMap.put(a[l], frequencyMap.get(a[l]) - 1);
                if (frequencyMap.get(a[l]) == 0) {
                    distinctCount--;
                }
                l++;
            }

            if (distinctCount == k && r - l + 1 < minLength) {
                minLength = r - l + 1;
                l = r - minLength + 1;
            }
        }

        if (minLength == Integer.MAX_VALUE) {
            System.out.println("-1 -1");
        } else {
            System.out.println(l + " " + r);
        }
    }
}
