#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<string> s(n + 1);
    s[0] = string(a[0], 'a');
    for (int i = 0; i < n; i++) {
      s[i + 1] = s[i];
      for (int j = a[i]; j < s[i].size(); j++) s[i + 1][j] = s[i][j] + 1;
    }
    for (int i = 0; i <= n; i++) cout << s[i] << '\n';
  }
}
