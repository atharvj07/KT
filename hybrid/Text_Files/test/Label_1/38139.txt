import sys

def solve(n, a):
    p = [-1] * n
    for i in range(n):
        j = a[i][0]
        if p[j-1] == -1 or a[i][1] == p[j-1]:
            p[j-1] = i+1
        else:
            p[j-1] = a[i][1]
    print(' '.join(map(str, p)))

n = int(input())
a = []
for _ in range(n):
    a.append(list(map(int, input().split())))
solve(n, a)
