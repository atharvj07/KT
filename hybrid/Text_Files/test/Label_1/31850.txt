import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int t = sc.nextInt();
        int[] q = new int[t];
        for (int i = 0; i < t; i++) {
            q[i] = sc.nextInt() - 1;
        }
        int[] x = new int[n];
        for (int i = 0; i < t; i++) {
            x[q[i]] = -p[q[i]];
        }
        int balance = 0;
        for (int i = 0; i < n; i++) {
            if (balance == 0) {
                x[i] = p[i];
            } else if (balance == p[i]) {
                x[i] = -p[i];
            } else if (balance < p[i]) {
                System.out.println("NO");
                return;
            } else {
                x[i] = p[i];
            }
            balance += x[i];
        }
        if (balance == 0) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(x[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }
}
