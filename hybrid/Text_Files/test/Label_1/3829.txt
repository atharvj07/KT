#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> v(n), suf(n);
        for (int i = 0; i < n; i++) cin >> v[i];
        sort(v.begin(), v.end());
        suf[n - 1] = v[n - 1];
        for (int i = n - 2; i >= 0; i--) suf[i] = suf[i + 1] * v[i];
        long long ans = 0;
        for (int i = 0; i < n - 4; i++) ans = max(ans, 1LL * v[i] * v[i + 1] * v[i + 2] * v[i + 3] * suf[i + 4]);
        cout << ans << '\n';
    }
}
