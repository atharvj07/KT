import java.util.*;

public class BombDestroyer {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        Point[] bombs = new Point[n];
        for (int i = 0; i < n; i++) {
            bombs[i] = new Point(scanner.nextInt(), scanner.nextInt());
        }

        List<String> operations = solve(bombs);
        System.out.println(operations.size());
        for (String operation : operations) {
            System.out.println(operation);
        }
    }

    private static List<String> solve(Point[] bombs) {
        List<String> operations = new ArrayList<>();

        // Sort bombs by distance from the origin
        Arrays.sort(bombs, (p1, p2) -> Integer.compare(p1.distSq(), p2.distSq()));

        // Destroy bombs one by one
        Point current = new Point(0, 0);
        for (Point bomb : bombs) {
            // Move to the bomb
            operations.addAll(move(current, bomb));

            // Pick up the bomb
            operations.add("2");

            // Move back to the origin
            operations.addAll(move(bomb, current));

            // Destroy the bomb
            operations.add("3");

            current = new Point(0, 0);
        }

        return operations;
    }

    private static List<String> move(Point start, Point end) {
        List<String> operations = new ArrayList<>();

        int dx = end.x - start.x;
        int dy = end.y - start.y;

        if (dx > 0) {
            operations.add(String.format("1 %d R", dx));
        } else if (dx < 0) {
            operations.add(String.format("1 %d L", -dx));
        }

        if (dy > 0) {
            operations.add(String.format("1 %d U", dy));
        } else if (dy < 0) {
            operations.add(String.format("1 %d D", -dy));
        }

        return operations;
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public int distSq() {
            return x * x + y * y;
        }
    }
}
