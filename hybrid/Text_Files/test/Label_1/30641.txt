import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int a = input.nextInt();
            int b = input.nextInt();
            int n = input.nextInt();
            System.out.println(((n & 1) == 1) ? b : a);
        }
        input.close();
    }
}
