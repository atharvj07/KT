import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String[] arr = new String[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.next();
        }
        Set<String> set = new HashSet<>();
        for (String s : arr) {
            set.add(s);
        }
        if (set.size() > 1) {
            System.out.println(-1);
        } else {
            String s = arr[0];
            int ans = s.length() - 1;
            for (int i = 1; i < n; i++) {
                if (!arr[i].equals(s)) {
                    ans = -1;
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
