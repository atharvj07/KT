#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define ll long long int
#define pb push_back
#define mp make_pair
const ll mod = 1e9 + 7;
bool can(ll n, ll k, ll l, vector<ll> &d) {
  vector<ll> p(2 * k);
  for (int i = 0; i < k; ++i) p[i] = i;
  for (int i = k; i < 2 * k; ++i) p[i] = k - (i - k);
  for (ll i = 0;; ++i) {
    ll x = 0;
    while (x <= n && d[x] + p[i % (2 * k)] <= l) {
      x++;
    }
    if (x == n + 1) return true;
    if (p[i % (2 * k)] == 0) return false;
  }
  return false;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ll t;
  cin >> t;
  while (t--) {
    ll n, k, l;
    cin >> n >> k >> l;
    vector<ll> d(n);
    for (auto &it : d) {
      cin >> it;
    }
    if (can(n, k, l, d))
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
  return 0;
}
