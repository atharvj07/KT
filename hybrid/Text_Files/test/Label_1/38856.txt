import sys
input=sys.stdin.readline

for t in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    b=[[] for i in range(n+1)]
    for i in range(n):
        b[a[i]].append(i)
    ans=0
    for l in b:
        for x in range(1,len(l)):
            ans=max(l[x]-l[x-1]-1,ans)
    print(ans)
