#include<bits/stdc++.h>
using namespace std;
int main(){
    int n, m, x, y, dp[105][105], a[105];
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    while (m--) {
        cin >> x >> y;
        dp[x][y] = dp[x][y - 1] + dp[x + 1][y] - dp[x + 1][y - 1] + a[x];
    }
    int ans = -1e9;
    for (int l = 1; l <= n; l++) {
        for (int r = 1; r <= n; r++) {
            ans = max(dp[l][r], ans);
        }
    }
    cout << ans << "\n";
}
