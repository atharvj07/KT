import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }
        int t = scanner.nextInt();
        int[] q = new int[t];
        for (int i = 0; i < t; i++) {
            q[i] = scanner.nextInt();
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                x[i] = p[i];
            } else if (i > 0) {
                x[i] = x[i - 1] + p[i];
            }
        }
        for (int i = 0; i < t; i++) {
            x[q[i] - 1] = -x[q[i] - 1];
        }
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < t; j++) {
                if (i == q[j] - 1) {
                    if (x[i] > 0) {
                        flag = false;
                    }
                }
            }
            if (i > 0) {
                if (x[i] <= x[i - 1]) {
                    flag = false;
                }
            }
        }
        if (flag) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(x[i] + " ");
            }
        } else {
            System.out.println("NO");
        }
    }
}
