import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[] s = new int[m], a = new int[k];
        for (int i = 0; i < m; i++) s[i] = sc.nextInt();
        for (int i = 0; i < k; i++) a[i] = sc.nextInt();
        Arrays.sort(s);
        Arrays.sort(a);
        int i = 0, j = 0, ans = 0, cur = 0;
        while (i < m && j < k) {
            if (s[i] < cur + a[j]) i++;
            else {
                ans += a[j];
                cur += a[j++];
            }
        }
        while (j < k) {
            ans += a[j++];
            cur += a[j - 1];
        }
        if (cur >= n) System.out.println(ans);
        else System.out.println(-1);
    }
}
