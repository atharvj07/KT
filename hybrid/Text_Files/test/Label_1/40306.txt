import sys
input = sys.stdin.readline

x1, y1 = map(int, input().split())
x2, y2 = map(int, input().split())
n = int(input())
ls = []
for i in range(n):
    a, b, c = map(int, input().split())
    ls.append([a, b, c])
ds = abs(abs(x1 - x2) - abs(y1 - y2))
for i in ls:
    a, b, c = i
    if (a * x1 + b * y1 + c) * (a * x2 + b * y2 + c) < 0:
        ds = 1

print(ds)
