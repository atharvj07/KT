import sys
 
n = int(input())
data = [list(map(int, inp.split())) for inp in sys.stdin.readlines()]
 
ans = 0
for i in range(1, n):
    if data[i][1] > data[i - 1][1] and data[i][0] > data[i - 1][0]:
        ans += 1
    if data[i][1] < data[i - 1][1] and data[i][0] < data[i - 1][0]:
        ans += 1
 
print(ans)
