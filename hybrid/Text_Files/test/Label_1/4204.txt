#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    string s;
    cin >> n >> s;

    vector<int> ans(n);
    int bal = 0, col = 0, lst_col = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'L') {
            if (bal > 0) bal--;
        } else if (s[i] == 'R') {
            bal++;
        } else {
            if (s[i] == '(') {
                if (bal == 0) {
                    col++;
                    lst_col = col;
                }
                bal++;
            } else {
                if (bal == 0) col = lst_col;
                bal--;
            }
        }
        ans[i] = bal == 0 ? col : -1;
    }

    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;

    return 0;
}
