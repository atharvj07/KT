import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int[] a = new int[10];
        for (int i = 0; i < n; i++) {
            a[s.charAt(i) - '0']++;
        }
        int target = 0;
        for (int i = 0; i < 10; i++) {
            if (a[i] > 0) {
                target = i;
                break;
            }
        }
        int sum = 0;
        int count = 0;
        for (int i = 0; i < n; i++) {
            int digit = s.charAt(i) - '0';
            sum += digit;
            a[digit]--;
            if (sum == target && count > 0) {
                System.out.println("YES");
                return;
            }
            if (a[digit] == 0) {
                sum = 0;
                count++;
            }
        }
        System.out.println("NO");
    }
}
