import java.util.*;

public class Main {
    private static final int N = 1001;
    private static final int MAX_V = 1000;
    private static final int MAX_D = 150;

    // precalculated Fibonacci modulo N for 0 to MAX_V
    private static final int[] fib = new int[MAX_V + 1];

    static {
        // initialize Fibonacci for 0 and 1
        fib[0] = 1;
        fib[1] = 1;
        // calculate Fibonacci for 2 to MAX_V
        for (int i = 2; i <= MAX_V; i++) {
            fib[i] = (fib[i - 1] + fib[i - 2]) % N;
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (in.hasNext()) {
            int v = in.nextInt();
            int d = in.nextInt();
            System.out.println(countConnectedSubsets(v, d));
        }
        in.close();
    }

    private static int countConnectedSubsets(int v, int d) {
        // initialize connected subsets count
        int count = 0;
        // initialize a set to track visited nodes
        Set<Integer> visited = new HashSet<>();
        // loop through each node
        for (int i = 1; i <= v; i++) {
            // if the node is not visited
            if (!visited.contains(i)) {
                // visit the node
                visited.add(i);
                // perform depth first search to find all connected nodes
                dfs(i, v, d, visited);
                // increment connected subsets count
                count++;
            }
        }
        return count;
    }

    private static void dfs(int node, int v, int d, Set<Integer> visited) {
        // loop through all other nodes
        for (int j = 1; j <= v; j++) {
            // if the other node is not visited and is within distance d from the current node
            if (!visited.contains(j) && Math.abs(fib[node] - fib[j]) < d) {
                // visit the other node
                visited.add(j);
                // perform depth first search to find all connected nodes
                dfs(j, v, d, visited);
            }
        }
    }
}
