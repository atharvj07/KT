import java.util.Scanner;

public class ValerieAndCrucians {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Total amount of teeth
        int m = scanner.nextInt(); // Amount of tooth rows
        int k = scanner.nextInt(); // Amount of crucians in Valerie's portion for dinner

        int[][] teeth = new int[n][2]; // teeth[i][0] - row index, teeth[i][1] - residual viability

        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt() - 1; // Adjust row index to 0-based indexing
            teeth[i][1] = scanner.nextInt();
        }

        int maxCrucians = 0;
        while (k > 0) {
            // Find the row with the maximum residual viability
            int maxRow = 0;
            for (int i = 1; i < m; i++) {
                if (teeth[i][1] > teeth[maxRow][1]) {
                    maxRow = i;
                }
            }

            // Eat crucians until the viability of the row becomes 0
            int cruciansEaten = 0;
            while (teeth[maxRow][1] > 0 && k > 0) {
                teeth[maxRow][1]--;
                k--;
                cruciansEaten++;
            }

            maxCrucians += cruciansEaten;
        }

        System.out.println(maxCrucians);
    }
}
