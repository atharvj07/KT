n, d = map(int, input().split())
x = list(map(int, input().split()))

possible_locations = 2  # At the start and end of the line

for i in range(1, n):
    if x[i] - x[i-1] >= 2*d:
        possible_locations += 1
        if x[i] - x[i-1] > 2*d:
            possible_locations += 1

print(possible_locations)
