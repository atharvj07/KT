import sys

n = int(input())
a = list(map(int, input().split()))

mns = [1] * n
for i in range(n - 1):
    if a[i] < a[i + 1]:
        mns[i + 1] = mns[i] + 1

d = {}
l = []
for i in range(n - 1, -1, -1):
    if a[i] not in d:
        d[a[i]] = i
        l.append(i)
mx = 0
for i in l:
    if mx < mns[i]:
        mx = mns[i]
        s = i

ans = []
for i in range(mx):
    ans.append(s - i)

print(mx)
print(*ans)
