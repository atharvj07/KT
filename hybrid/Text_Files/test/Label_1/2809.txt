#include <iostream>
#include <algorithm>
#include <vector>

int main() {
    int n, L;
    std::cin >> n >> L;
    std::vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
        std::cin >> b[i];
    }
    std::sort(a.begin(), a.end());
    std::sort(b.begin(), b.end());
    bool flag = true;
    for (int i = 0; i < n; i++) {
        if (a[i] != b[i]) {
            flag = false;
        }
    }
    if (flag) {
        std::cout << "YES" << std::endl;
    } else {
        std::cout << "NO" << std::endl;
    }
    return 0;
}
