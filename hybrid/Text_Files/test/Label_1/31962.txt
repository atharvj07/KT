import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] arr = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int sumRow = 0;
                int sumCol = 0;
                for (int k = 0; k < n; k++) {
                    sumRow += arr[i][k];
                    sumCol += arr[k][j];
                }
                if (sumCol > sumRow) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
