import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), c = sc.nextInt();
            int ans = 0;
            for (int j = 1; j <= a; j++) {
                for (int k = j; k <= b; k++) {
                    for (int l = k; l <= c; l++) {
                        if (a % j == 0 && b % k == 0 && c % l == 0) {
                            ans++;
                        }
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
