import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int[] count = new int[26];
        int[] prefSum = new int[26];
        for (int i = 0; i < n; i++) {
            count[s.charAt(i) - 'a']++;
        }
        prefSum[0] = count[0];
        for (int i = 1; i < 26; i++) {
            prefSum[i] = prefSum[i - 1] + count[i];
        }
        long ans = 1;
        for (int i = 0; i < n; i++) {
            int l = i > 0 ? prefSum[s.charAt(i - 1) - 'a'] : 0;
            int r = prefSum[s.charAt(i) - 'a'] - count[s.charAt(i) - 'a'];
            ans = (ans * (l + r + 1)) % 1000000007;
        }
        System.out.println(ans);
    }
}
