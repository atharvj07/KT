#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;

  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }

  int p;
  std::cin >> p;

  std::sort(a.begin(), a.end());

  int l = 0;
  int r = 0;
  int min_size = a[0];

  for (int i = 0; i < n; i++) {
    while (r < n && l <= r && a[l] + a[r] <= p) {
      r++;
    }
    r--;

    min_size = std::min(min_size, r - l + 1);

    l++;
  }

  std::cout.setf(std::ios::fixed);
  std::cout.precision(9);

  std::cout << (double)min_size / n << '\n';

  return 0;
}
