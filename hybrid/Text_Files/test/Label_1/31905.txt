import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt();
        int[] a = new int[n], b = new int[k];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < k; i++) {
            b[i] = scanner.nextInt();
        }
        int[] c = new int[n], last = new int[201];
        for (int i = 0; i < n; i++) {
            c[i] = a[i];
        }
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                last[a[i]] = i;
            }
        }
        Arrays.sort(b);
        for (int i = 0, j = 0; i < n; i++) {
            if (a[i] == 0) {
                c[i] = b[j++];
                for (int x = 1; x <= 200; x++) {
                    if (x != c[i] && x != a[i] && last[x] > i) {
                        System.out.println("NO");
                        return;
                    }
                }
            }
        }
        System.out.println("YES");
    }
}
