#include <bits/stdc++.h>
using namespace std;
inline void solve(){
    int n, x; string s;
    cin >> s >> x;
    n = s.size();
    vector<int> ans(n);
    for(int i = 0; i < n; i++){
        int l = max(0, i - x), r = min(n - 1, i + x);
        int ok = 0;
        for(int j = l; j <= r; j++) ok |= (s[j] == '1');
        ans[i] = ok;
    }
    for(int i = 0; i < n; i++){
        int l = max(0, i - x), r = min(n - 1, i + x);
        int ok = 0;
        for(int j = l; j <= r; j++) ok |= (ans[j] == 1);
        if(s[i] != (ok + '0')) goto fail;
    }
    for(auto &x : ans) cout << x;
    cout << endl;
    return;
fail:
    cout << "-1\n";
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int t; cin >> t;
    while(t--) solve();
}
