#include <bits/stdc++.h>
#define nl "\n"
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (int &x : a) cin >> x;
        int oddCnt = 0;
        for (int x : a) oddCnt += ((__builtin_popcount(x)) % 2 == 1);
        if (oddCnt % 2 == 1)
            cout << "NO" << nl;
        else
            cout << "YES" << nl;
    }
    return 0;
}
