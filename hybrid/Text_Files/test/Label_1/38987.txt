from itertools import accumulate
from sys import stdin, setrecursionlimit

setrecursionlimit(1 << 30)
input = stdin.readline

n = int(input())
p = []
for _ in range(n):
    x, y, s = map(int, input().split())
    p.append((x, y, s))
p.sort()

ans = 0
t = [0] * n
for i, (x, y, s) in enumerate(p):
    t[i] = x - 1
    if s or i > 0:
        t[i] -= t[i - 1]
    if not s:
        ans += t[i]
    ans %= 998244353
print(ans)
