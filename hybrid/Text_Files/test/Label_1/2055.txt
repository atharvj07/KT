#include <algorithm>
#include <vector>
#include <set>
#include <iostream>
using namespace std;
int main(){
    int n;
    cin >> n;
    struct Point{
        long long x,y;
        Point(){};
        Point(long long _x,long long _y):x(_x),y(_y){};
    } P;
    vector < pair<Point,Point> > points(n);
    for(int f=0;f<n;f++){
        cin >> P.x >> P.y;
        points[f].first=P;
        cin >> P.x >> P.y;
        points[f].second=P;
    }
    long long ans_x=-1,ans_y=-1;
    for(int f=0;f<n;f++){ // O(N log N)
        set <long long> x_axis,y_axis;
        vector <long long> tmp_x,tmp_y;
        for(int g=0;g<n;g++){
            long long x1=points[g].first.x,y1=points[g].first.y,x2=points[g].second.x,y2=points[g].second.y;
            tmp_x.push_back(x1);
            tmp_x.push_back(x2);
            tmp_y.push_back(y1);
            tmp_y.push_back(y2);
        }
        sort(tmp_x.begin(),tmp_x.end());
        sort(tmp_y.begin(),tmp_y.end());
        tmp_x.erase(unique(tmp_x.begin(),tmp_x.end()),tmp_x.end());
        tmp_y.erase(unique(tmp_y.begin(),tmp_y.end()),tmp_y.end());
        for(int g=0;g<tmp_x.size();g++) x_axis.insert(tmp_x[g]);
        for(int g=0;g<tmp_y.size();g++) y_axis.insert(tmp_y[g]);
        for(set<long long>::const_iterator it=x_axis.begin();it!=--x_axis.end();){
            set<long long>::const_iterator it2=it;
            it2++;
            for(set<long long>::const_iterator it3=y_axis.begin();it3!=--y_axis.end();){
                if(x_axis.count(*it) and y_axis.count(*it3) and x_axis.count(*it2) and y_axis.count(*it3)){
                    int cnt=0;
                    for(int g=0;g<n;g++){
                        long long x1=points[g].first.x,y1=points[g].first.y,x2=points[g].second.x,y2=points[g].second.y;
                        if(x1<=*it and y1<=*it3 and x2>=*it2 and y2>=*it3){
                            cnt++;
                        }
                    }
                    if(cnt>=n-1){
                        ans_x=*it;
                        ans_y=*it3;
                        goto aa;
                    }
                }
                it3++;
            }
            it++;
        }
    }
    aa:
    cout << ans_x << ' ' << ans_y << '\n';
}
