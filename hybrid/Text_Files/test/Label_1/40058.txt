import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    s = input().strip()
    x, y = 0, 0
    for i in range(n):
        if s[i] == 'L':
            x -= 1
        elif s[i] == 'R':
            x += 1
        elif s[i] == 'U':
            y += 1
        else:
            y -= 1
    if x == y == 0:
        print(-1)
    else:
        l, r = 0, n
        for i in range(n):
            if s[i] == 'L' or s[i] == 'R':
                if s[i] == 'L':
                    x += 1
                else:
                    x -= 1
                if x == y == 0:
                    r = i
                    break
            else:
                if s[i] == 'U':
                    y -= 1
                else:
                    y += 1
                if x == y == 0:
                    l = i + 1
                    break
        print(l + 1, r)
