import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static int n, m;
    private static int[] c;
    private static int[] deg;
    private static List<Integer>[] adj;
    private static boolean[] vis;
    private static long[] ans;

    private static void dfs1(int v, int p) {
        vis[v] = true;
        deg[v] = 1;
        for (int u : adj[v]) {
            if (u == p) continue;
            dfs1(u, v);
            deg[v] += deg[u];
        }
    }

    private static boolean dfs2(int v, int p, long sum) {
        if (sum > 2L * n * n || sum < -2L * n * n) return false;
        vis[v] = true;
        ans[v] = sum;
        boolean ok = true;
        for (int u : adj[v]) {
            if (u == p) continue;
            ok &= dfs2(u, v, sum + c[u] - c[v]);
        }
        return ok;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        c = new int[n + 1];
        deg = new int[n + 1];
        adj = new List[n + 1];
        vis = new boolean[n + 1];
        ans = new long[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i <= n; i++) {
            c[i] = scanner.nextInt();
            if (c[i] % 2 != deg[i] % 2) {
                System.out.println("NO");
                return;
            }
        }
        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        dfs1(1, 0);
        if (c[1] != 0) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        dfs2(1, 0, 0);
        for (long w : ans) {
            System.out.println(w);
        }
    }
}
