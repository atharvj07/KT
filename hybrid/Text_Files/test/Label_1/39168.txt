def dfs(node, parent):
    ways[node] = 1
    for child in tree[node]:
        if child != parent:
            dfs(child, node)
            ways[node] *= 0.5 * (ways[child] + 1)

n = int(input())
tree = {i: [] for i in range(1, n + 1)}
for _ in range(n - 1):
    u, v = map(int, input().split())
    tree[u].append(v)
    tree[v].append(u)

ways = [0] * (n + 1)
dfs(1, 0)
total_ways = ways[1]

for node in range(1, n + 1):
    print(ways[node] / total_ways)
