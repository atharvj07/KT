from typing import List

def gcd(a: int, b: int) -> int:
    while b:
        a, b = b, a % b
    return a

def solve(arr: List[int]) -> int:
    ans, n = 0, len(arr)
    for i in range(n - 1):
        g = gcd(arr[i], arr[(i + 1) % n])
        ans += (arr[i] - g) // g
        arr[i] = g
    return ans

t = int(input())
for _ in range(t):
    arr = [int(x) for x in input().split()]
    arr = arr[1:]
    print(solve(arr))
