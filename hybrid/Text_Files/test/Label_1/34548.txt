import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.close();
        int x = n / 7 * 10;
        n = n - n / 7 * 7;
        if (n <= 4) x += n;
        else x -= (n - 4);
        if (x > 0) System.out.println(x);
        else System.out.println(-1);
    }
}
