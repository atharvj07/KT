#include <bits/stdc++.h>
using namespace std;
#define int long long
#define pd pair<int, int>
#define pdi pair<int, pd>
const int mod = 1e9 + 7;
#define FOR(i, a, b) for (int i = a; i < b; i++)
#define FORD(i, a, b) for (int i = a; i >= b; i--)
int32_t main()
{
    int n, m;
    cin >> n >> m;
    vector<int> arr[n + 1];
    vector<int> brr[n + 1];
    vector<bool> vis(n + 1, 0);
    while (m--)
    {
        int a, b;
        cin >> a >> b;
        arr[a].push_back(b);
        arr[b].push_back(a);
        brr[a].push_back(b);
        brr[b].push_back(a);
    }
    int cnt = 0;
    queue<int> q;
    FOR(i, 1, n + 1)
    {
        if (!vis[i])
        {
            q.push(i);
            cnt++;
            while (!q.empty())
            {
                int s = q.front();
                q.pop();
                vis[s] = 1;
                for (auto u : arr[s])
                {
                    if (!vis[u])
                    {
                        q.push(u);
                    }
                }
            }
        }
    }
    cout << cnt - 1 << endl;
    FOR(i, 1, n)
    {
        for (auto u : brr[i])
        {
            if (vis[u] ^ vis[i])
            {
                cout << i << " " << u << endl;
            }
        }
    }
    return 0;
}
