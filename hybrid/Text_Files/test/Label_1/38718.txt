def can_conduct_experiment(n, b, a, transformations):
    extra = 0
    for i in range(n - 1, 0, -1):
        b[i] += extra
        if b[i] < a[i]:
            needed = a[i] - b[i]
            transform_cost = transformations[i - 1][1]
            min_base_needed = (needed + transform_cost - 1) // transform_cost
            base_material = transformations[i - 1][0] - 1
            if b[base_material] < min_base_needed:
                return "NO"
            b[base_material] -= min_base_needed
            extra = min_base_needed * transform_cost - needed
        else:
            extra = b[i] - a[i]
    if b[0] + extra < a[0]:
        return "NO"
    return "YES"

n = int(input().strip())
b = list(map(int, input().strip().split()))
a = list(map(int, input().strip().split()))
transformations = [list(map(int, input().strip().split())) for _ in range(n - 1)]

print(can_conduct_experiment(n, b, a, transformations))
