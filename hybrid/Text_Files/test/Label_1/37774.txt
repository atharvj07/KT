n = int(input())
ciel, jiro = 0, 0
middle_cards = []

for _ in range(n):
    pile = list(map(int, input().split()))[1:]
    if len(pile) % 2 == 1:
        middle_cards.append(pile[len(pile) // 2])
    ciel += sum(pile[:(len(pile) + 1) // 2])
    jiro += sum(pile[len(pile) // 2:])

middle_cards.sort(reverse=True)
for i in range(len(middle_cards)):
    if i % 2 == 0:
        ciel += middle_cards[i]
    else:
        jiro += middle_cards[i]

print(ciel, jiro)
