#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> b(n), a(n);
  for (int i = 0; i < n; ++i) { std::cin >> b[i]; }
  a[0] = b[0];
  for (int i = 1; i < n; ++i) {
    a[i] = std::max(a[i - 1] + 1, b[i] - (b[i] % 10));
  }
  for (int i = 0; i < n; ++i) { std::cout << a[i] << std::endl; }
  return 0;
}
