import sys

def solve():
    t = int(input())

    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))

        ans = []

        for i in range(1, n + 1):
            can = True

            for j in range(n):
                if a[j] % i != 0:
                    can = False
                    break

            if can:
                ans.append(i)

        print(*ans)

solve()
