import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int x = sc.nextInt();
        System.out.println((n - 1) * (m - 1) + ((n - 2) * (m - 2) - 1) * (x - 1));
    }
}
