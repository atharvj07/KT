#include <vector>
#include <unordered_set>
#include <iostream>
using namespace std;
int main() {
    int n, k, d;
    cin >> n >> k >> d;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    unordered_set<int> seen;
    for (int i = 0; i < n; i++) {
        int cur = a[i] % k;
        if (!seen.count(cur)) seen.insert(cur);
    }
    vector<int> ans;
    for (int i = 0; i < k; i++) {
        if (seen.count((d - i) % k)) {
            ans.push_back(i);
        }
    }
    cout << ans.size() << endl;
    for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
    cout << endl;
}
