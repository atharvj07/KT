t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    if sum(a) < n:
        print(-1)
        continue
    a.sort(reverse=True)
    count, i = 0, 0
    while n > 0 and i < m:
        if a[i] <= n:
            n -= a[i]
        else:
            a[i] //= 2
            count += 1
            continue
        i += 1
    if n > 0:
        print(-1)
    else:
        print(count)
