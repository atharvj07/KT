import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int x = 0; x < 3; x++) {
            long sum = 0;
            int l = -1, r = -1;
            for (int i = 0; i < n; i++) {
                if (a[i] != 0 && (l == -1 || a[i] * sum < 0)) {
                    l = i;
                    sum = a[i];
                }
                if (l != -1 && a[i] == 0) {
                    r = i - 1;
                    break;
                }
                sum += a[i];
            }
            if (l == -1) {
                l = 0;
                r = n - 1;
            }
            System.out.println(l + 1 + " " + (r + 1));
            for (int i = l; i <= r; i++) {
                System.out.print(-sum / (r - l + 1) + " ");
            }
            System.out.println();
            for (int i = l; i <= r; i++) {
                a[i] = 0;
            }
        }
    }
}
