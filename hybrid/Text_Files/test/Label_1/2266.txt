#include<bits/stdc++.h>
using namespace std;
int main(){
    int sum,limit;cin>>sum>>limit;int cnt=0;
    for(int i=limit;i>0;i--){
        if((1<<i)<=sum){sum-=(1<<i);cnt++;}
    }
    cout<<(sum==0?cnt:-1)<<endl;
    for(int i=limit;i>0;i--){
        if((1<<i)<=sum){sum-=(1<<i);cout<<i<<" ";}
    }
}
