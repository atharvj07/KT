#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int dx[] = {0, 0, -1, 1};
const int dy[] = {1, -1, 0, 0};
const char dc[] = {'U', 'D', 'L', 'R'};

int n, m;
vector<int> d;
string s;

int main() {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2 >> n;
    cin >> s;
    m = s.size();
    d.resize(n + 1);
    queue<pair<int, pair<int, int>>> q;
    q.push({0, {x1, y1}});
    while (q.size()) {
        int t = q.front().first;
        int x = q.front().second.first;
        int y = q.front().second.second;
        q.pop();
        if (x == x2 && y == y2) {
            cout << t << endl;
            return 0;
        }
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];
            if (nx >= 0 && nx <= x2 && ny >= 0 && ny <= y2) {
                int nt = t + ((char)s[(t % m)] == dc[i] ? 0 : 1);
                if (d[nt] == 0) {
                    d[nt] = 1;
                    q.push({nt, {nx, ny}});
                }
            }
        }
    }
    cout << -1 << endl;
    return 0;
}
