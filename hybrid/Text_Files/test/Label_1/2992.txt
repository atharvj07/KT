#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  vector<int> v(5);
  for (int &x : v) cin >> x;
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = 0; i < 3; i++) ans += v[i];
  for (int i = 3; i < 5; i++) ans += v[i] / 2;
  cout << ans << "\n";
  return 0;
}
