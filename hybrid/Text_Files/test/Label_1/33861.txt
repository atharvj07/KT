import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long h = input.nextLong();
        long m = input.nextLong();
        int k = input.nextInt();
        long h1 = input.nextLong();
        long m1 = input.nextLong();
        long h2 = input.nextLong();
        long m2 = input.nextLong();
        long res = 0;
        for (long i = h1; i < h2; i++) {
            res += count(i, m, k);
        }
        for (long i = m1; i < m2; i++) {
            res += count(h2, i, k);
        }
        System.out.println(res);
    }

    private static long count(long h, long m, int k) {
        int hLen = (int) Math.log10(h) + 1;
        int mLen = (int) Math.log10(m) + 1;
        int diffCnt = 0;
        for (long i = 1; i <= m; i*=10) {
            if (((h / i) % 10) != ((h / (i*10)) % 10)) diffCnt++;
            if (((m / i) % 10) != ((m / (i*10)) % 10)) diffCnt++;
        }
        return diffCnt >= k ? 1 : 0;
    }
}
