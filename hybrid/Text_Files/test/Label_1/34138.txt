import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int h = input.nextInt();
        int m = input.nextInt();
        int n = input.nextInt();
        long result = 0;
        int[] table = new int[h];
        Arrays.fill(table, -1);
        while (n-- > 0) {
            char op = input.next().charAt(0);
            int id = input.nextInt();
            if (op == '+') {
                int hash = input.nextInt();
                int t = hash;
                long dummy = 0;
                while (table[t] != -1) {
                    t = (t + m) % h;
                    dummy++;
                }
                table[t] = id;
                result += dummy;
            } else if (op == '-') {
                int t = 0;
                while (table[t] != id) {
                    t = (t + m) % h;
                }
                table[t] = -1;
            }
        }
        System.out.println(result);
    }
}
