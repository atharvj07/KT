import sys
n, m = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
a.sort()
x = [0] * n
for i in range(n):
    x[i] = a[i] * min(i + 1, m)
    for j in range(m - 1, 0, -1):
        if i + j < n:
            x[i] += a[i + j]
        else:
            break
print(*x)
