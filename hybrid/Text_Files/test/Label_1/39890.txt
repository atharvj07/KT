n = int(input().strip())
a = list(input().strip())
ans = ""

num0 = 0
num5 = 0
for i in a:
    if i == '5':
        num5 += 1
    else:
        num0 += 1

if num0 == n:
    print(0)
    exit()

if num5 < num0:
    num5, num0 = num0, num5

if num5 == num0:
    ans = '5' * num5

elif num5 > num0:
    ans = '5' * num0 + '0' * (num5 - num0)

if int(ans) % 90 == 0:
    print(ans)
else:
    print(-1)
