import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.PriorityQueue;
import java.util.Queue;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int d = Integer.parseInt(input[1]);
        int s = Integer.parseInt(input[2]);
        Robot[] robots = new Robot[n];
        for (int i = 0; i < n; i++) {
            input = br.readLine().split(" ");
            robots[i] = new Robot(
                    Integer.parseInt(input[0]),
                    Integer.parseInt(input[1]),
                    Integer.parseInt(input[2])
            );
        }
        Arrays.sort(robots, Comparator.comparingInt(Robot::getFuel));
        Queue<Robot> pq = new PriorityQueue<>(Comparator.comparingInt(Robot::getCapacity));
        int numRobots = 0;
        int fuelUsed = 0;
        for (Robot robot : robots) {
            if (robot.getDistance() >= d) {
                if (robot.getFuel() <= s) {
                    numRobots++;
                    pq.add(robot);
                    fuelUsed += robot.getFuel();
                    s -= robot.getFuel();
                }
            }
        }
        while (s > 0 && !pq.isEmpty()) {
            Robot robot = pq.poll();
            if (robot.getCapacity() > 0) {
                for (int i = 0; i < robot.getCapacity() && !robots[i].used; i++) {
                    if (robots[i].getDistance() >= d) {
                        numRobots++;
                        fuelUsed += robots[i].getFuel();
                        s -= robots[i].getFuel();
                        pq.add(robots[i]);
                        robots[i].used = true;
                    }
                }
            }
        }
        robots[0].used = true;
        System.out.println(numRobots + " " + fuelUsed);
    }

    static class Robot {
        private int capacity;
        private int fuel;
        private int distance;
        private boolean used;

        public Robot(int capacity, int fuel, int distance) {
            this.capacity = capacity;
            this.fuel = fuel;
            this.distance = distance;
            this.used = false;
        }

        public int getCapacity() {
            return capacity;
        }

        public int getFuel() {
            return fuel;
        }

        public int getDistance() {
            return distance;
        }
    }
}
