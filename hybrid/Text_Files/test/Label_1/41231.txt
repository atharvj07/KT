from collections import defaultdict
n=int(input())
f=list(map(int,input().split()))
if len(set(f))<n:
    print(-1); exit()
g,h=defaultdict(int),defaultdict(int)
for i in range(n):
    g[f[i]]=i+1; h[i+1]=f[i]
m=len(set(f))
print(m)
for i in f: print(g[i],end=" ")
print()
for i in range(1,m+1): print(h[i],end=" ")
