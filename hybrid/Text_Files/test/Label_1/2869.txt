#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> a(n), ans(n), b(n);
  for (int i = 0; i < n; i++) a[i] = s[i] - '0';
  for (int i = 0; i < n; i++) ans[i] = 0;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (cnt == k) {
      b[i] = ans[i - k];
      ans[i] = b[i];
      cnt++;
      if (cnt == k) cnt = 0;
    } else {
      b[i] = a[i];
      ans[i] = b[i];
      cnt++;
    }
  }
  int j = 0;
  while (ans[j] == 0) j++;
  string res = "";
  for (int i = j; i < n; i++) res += to_string(ans[i]);
  cout << res.size() << endl;
  cout << res << endl;
  return 0;
}
