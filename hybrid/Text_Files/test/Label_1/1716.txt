#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (n == 4 && s == "2020")
      cout << "YES" << endl;
    else if (n == 6 && s.substr(2, 4) == "2020")
      cout << "YES" << endl;
    else if (n >= 6 && s.substr(0, 4) == "2020")
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
