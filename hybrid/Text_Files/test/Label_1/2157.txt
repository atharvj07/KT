#include <iostream>
#include <vector>
#include <string>

int main() {
    int k;
    std::cin >> k;
    std::string s;
    std::cin >> s;

    std::vector<std::string> words;
    std::string word;
    for (char c : s) {
        if (c == ' ') {
            words.push_back(word);
            word = "";
            continue;
        }
        word.push_back(c);
    }
    words.push_back(word);

    std::vector<std::vector<int>> dp(words.size() + 1, std::vector<int>(k + 1, 0));
    for (int i = 0; i <= k; i++) {
        dp[0][i] = 0;
    }
    for (int i = 1; i <= words.size(); i++) {
        for (int j = 1; j <= k; j++) {
            dp[i][j] = dp[i - 1][j];
            int width = words[i - 1].size();
            if (j >= width) {
                dp[i][j] = std::max(dp[i][j], dp[i - 1][j - width] + width);
            }
        }
    }

    std::cout << dp[words.size()][k] << std::endl;
    return 0;
}
