#include<cstdio>
#include<vector>
using namespace std;
typedef pair<int, int> pii;
typedef pair<pii, int> piii;
const int N=25;
char hh[N], vv[N];
bool vis[N][N][4];
vector<piii> g[N][N];
void dfs(int x, int y, int c, int d){
	if(vis[x][y][c]!=0){
		return;
	}
	vis[x][y][c]=1;
	if(c==0)for(auto to: g[x][y]){
		if(to.first.first==x&&to.first.second==y){
			dfs(to.first.first, to.first.second, to.second, (d+1)%4);
		}
	}
	if(c==1)for(auto to: g[x][y]){
		if(to.first.first==x+1&&to.first.second==y){
			dfs(to.first.first, to.first.second, to.second, d);
		}
	}
	if(c==2)for(auto to: g[x][y]){
		if(to.first.first==x&&to.first.second==y+1){
			dfs(to.first.first, to.first.second, to.second, (d+1)%4);
		}
	}
	if(c==3)for(auto to: g[x][y]){
		if(to.first.first==x&&to.first.second==y-1){
			dfs(to.first.first, to.first.second, to.second, d);
		}
	}
}
int main(){
	int n, m;
	scanf("%d%d", &n, &m);
	scanf("%s%s", hh+1, vv+1);
	for(int i=1;i<=n;++i){
		int d=hh[i]=='>';
		g[i-1][0].push_back({{i, 0}, d});
		g[i][m-1].push_back({{i, m-1}, d});
	}
	for(int i=1;i<=m;++i){
		int d=vv[i]=='^';
		g[0][i-1].push_back({{0, i}, d});
		g[n-1][i].push_back({{n-1, i}, d});
	}
	dfs(0, 0, 0, 0);	// 0 -> right; 1 -> down; 2 -> left; 3 -> up
	printf("%s", vis[n-1][m-1][0]?"YES":"NO");
}
