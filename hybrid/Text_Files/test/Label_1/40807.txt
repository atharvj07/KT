def minimum_possible_kicks(t, test_cases):
    results = []
    for s in test_cases:
        score1, score2 = 0, 0
        possible1, possible2 = 5, 5  # Possible scores left for each team
        min_kicks = 10
        for i in range(10):
            if s[i] == '1':
                if i % 2 == 0:
                    score1 += 1
                else:
                    score2 += 1
            if i % 2 == 0:
                possible1 -= 1
            else:
                possible2 -= 1
            # Check if the game can be decided
            if score1 > score2 + possible2 or score2 > score1 + possible1:
                min_kicks = i + 1
                break
        results.append(min_kicks)
    return results

# Read input
t = int(input().strip())
test_cases = [input().strip() for _ in range(t)]

# Get results
results = minimum_possible_kicks(t, test_cases)

# Print results
for result in results:
    print(result)
