import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] nums = new int[12];
    for (int i = 0; i < 12; i++) {
      nums[i] = sc.nextInt();
    }
    int ans = 0;
    for (int i = nums[0]; i <= nums[1]; i++) {
      for (int j = nums[2]; j <= nums[3]; j++) {
        for (int k = nums[4]; k <= nums[5]; k++) {
          for (int l = nums[6]; l <= nums[7]; l++) {
            if (i == k || j == l || isCross(i, j, k, l)) {
              continue;
            }
            if (i < k && j < l) {
              ans += (l - j) * (k - i);
            }
          }
        }
      }
    }
    System.out.println(ans % (int) (1e9 + 7));
  }

  private static boolean isCross(int x1, int y1, int x2, int y2) {
    return x1 <= x2 && y1 >= y2 || x1 >= x2 && y1 <= y2;
  }
}
