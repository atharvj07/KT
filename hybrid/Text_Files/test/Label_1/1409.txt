#include <bits/stdc++.h>
using namespace std;
#define pb push_back
vector<vector<int>>adj;
const int MAXN = 2e5 + 5;
int sub[MAXN], dist[MAXN];
bool visited[MAXN];
 
void dfs1(int u, int p){
    sub[u] = 1;
    for(const auto& v : adj[u]){
        if(v == p) continue;
        dfs1(v, u);
        sub[u] += sub[v];
    }
}
 
void dfs2(int u, int p, int d){
    visited[u] = true;
    dist[u] = d;
    for(const auto& v : adj[u]){
        if(v == p || visited[v]) continue;
        dfs2(v, u, d + 1);
    }
}
 
int main(){
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        adj.assign(n + 1, vector<int>());
        for(int i = 1; i < n; i++){
            int u, v;
            cin >> u >> v;
            adj[u].pb(v);
            adj[v].pb(u);
        }
        dfs1(1, 1);
        int centroid = -1;
        for(int i = 1; i <= n; i++){
            if(sub[i] * 2 <= n){
                centroid = i;
            }
        }
        dfs2(centroid, centroid, 0);
        int mx = 0;
        for(int i = 1; i <= n; i++){
            mx = max(mx, dist[i]);
        }
        cout << mx << '\n';
    }
    return 0;
}
