import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            int n = sc.nextInt();
            int s = sc.nextInt();
            int[][] dp = new int[n + 1][s + 1];
            dp[0][0] = 1;
            for (int i = 1; i <= n; i++) {
                for (int j = 0; j <= s; j++) {
                    for (int k = 0; k <= 9 && j + i * k <= s; k++) {
                        dp[i][j + i * k] += dp[i - 1][j];
                    }
                }
            }
            System.out.println(dp[n][s]);
        }
    }
}
