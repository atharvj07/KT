t = int(input())

for _ in range(t):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    apps = sorted(zip(a, b), key=lambda x: (x[1], -x[0]))
    
    memory_freed = 0
    points_lost = 0
    result = float('inf')
    
    regular_apps = []
    for mem, points in apps:
        if points == 1:
            while regular_apps and memory_freed < m:
                memory_freed += regular_apps.pop()
                points_lost += 1
        memory_freed += mem
        points_lost += points
        if memory_freed >= m:
            result = min(result, points_lost)
        if points == 1:
            regular_apps.append(mem)
    
    print(result if result != float('inf') else -1)
