import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int k = scanner.nextInt();
        char[] letters = scanner.next().toCharArray();
        Arrays.sort(letters);
        StringBuilder[] words = new StringBuilder[n];
        StringBuilder smallest = new StringBuilder();
        StringBuilder last = new StringBuilder();
        for (int i = 0; i < n; i++) {
            words[i] = new StringBuilder();
            for (int j = 0; j < l; j++) {
                if (i == k - 1) {
                    smallest.append(letters[i * l + j]);
                }
                words[i].append(letters[i * l + j]);
                if (i > 0) {
                    if (words[i].charAt(j) != words[i - 1].charAt(j)) {
                        if (words[i].charAt(j) < words[i - 1].charAt(j)) {
                            smallest = new StringBuilder(words[i]);
                            break;
                        } else {
                            last = new StringBuilder(words[i]);
                            break;
                        }
                    }
                }
            }
        }
        if (last.length() == 0) {
            last = smallest;
        }
        Comparator<StringBuilder> comparator = Comparator.comparing(StringBuilder::toString);
        Arrays.sort(words, comparator);
        words[k - 1] = smallest;
        for (StringBuilder word : words) {
            System.out.println(word);
        }
    }
}
