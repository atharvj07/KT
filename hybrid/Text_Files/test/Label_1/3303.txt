#include <bits/stdc++.h>
using namespace std;
#define ld long double
#define ll long long
#define pb push_back
#define mp make_pair
#define F first
#define S second

const ll MAXN = 300005;
const ll INF = 123456789;
const ll MOD = 1000000007;

ll n[3], x[3][MAXN], i, j, k, ans;
map<ll, ll> cnt;

int main() {
    for (i = 0; i < 3; i++) {
        cin >> n[i];
        for (j = 0; j < n[i]; j++) {
            cin >> x[i][j];
            cnt[x[i][j]]++;
        }
    }
    for (map<ll, ll>::iterator it = cnt.begin(); it != cnt.end(); it++) {
        if (it->S == 1) ans += it->F;
    }
    cout << ans << endl;
    return 0;
}
