#include <algorithm>
#include <cstdio>
#include <cstring>
#include <vector>

using namespace std;

#define MAXN 100000

int n;
int cost[MAXN + 1];
char s[MAXN + 1][2000];
bool rev[MAXN + 1];
int dp[MAXN + 1][2];
int nxt[MAXN + 1][26];

int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &cost[i]);
  for (int i = 0; i < n; i++) scanf("%s", s[i]);
  for (int i = 0; i < n; i++) {
    int len = strlen(s[i]);
    for (int j = 0; j < len / 2; j++) swap(s[i][j], s[i][len - j - 1]);
    rev[i] = (s[i][0] > s[i][1]);
  }
  for (int i = 0; i < n; i++) {
    int len = strlen(s[i]);
    for (int j = 0; j < 26; j++) {
      bool ok = true;
      for (int k = 0; k < len / 2; k++) ok &= (s[i][k] >= 'a' + j);
      if ((len & 1) && ok) ok &= (s[i][len / 2] >= 'a' + j);
      nxt[i][j] = (ok ? i : -1);
    }
  }
  dp[n][0] = dp[n][1] = 0;
  for (int i = n - 1; i >= 0; i--) {
    dp[i][0] = dp[i][1] = 1e9;
    for (int j = 0; j < 26; j++) {
      if (nxt[i][j] == -1) continue;
      int c = (rev[i] ? cost[i] : 0);
      if (nxt[i][j] == i) {
        dp[i][0] = min(dp[i][0], c + dp[nxt[i][j]][0]);
        dp[i][1] = min(dp[i][1], c + dp[nxt[i][j]][1]);
      } else {
        dp[i][1] = min(dp[i][1], c + dp[nxt[i][j]][0]);
      }
    }
  }
  int ans = dp[0][0];
  if (ans >= 1e9) ans = -1;
  printf("%d", ans);
  return 0;
}
