import java.util.*;

public class Main {

    static int n, m;
    static int[][] gardens;
    static int[] pref;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        gardens = new int[n][2];
        for (int i = 0; i < n; i++) {
            gardens[i][0] = sc.nextInt();
            gardens[i][1] = sc.nextInt();
        }
        pref = new int[m + 2];
        for (int[] garden : gardens) {
            pref[garden[0]]++;
            pref[garden[1] + 1]--;
        }
        for (int i = 1; i <= m; i++) {
            pref[i] += pref[i - 1];
        }
        long ans = 0;
        for (int i = 1; i <= m; i++) {
            if (pref[i] % 2 == 1) {
                ans += i;
            }
        }
        System.out.println(ans);
    }
}
