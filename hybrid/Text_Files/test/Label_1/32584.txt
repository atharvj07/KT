import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    private static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int[] array = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] fibs = new int[n + 1];
        fibs[0] = 1;
        fibs[1] = 1;
        for (int i = 2; i <= n; i++) {
            fibs[i] = (fibs[i - 1] + fibs[i - 2]) % MOD;
        }
        int[][] segtree = new int[4 * n][2];
        build(array, 0, 0, n - 1, segtree);
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int l = Integer.parseInt(line[1]) - 1;
            int r = Integer.parseInt(line[2]) - 1;
            if (type == 1) {
                int x = Integer.parseInt(line[3]);
                update(l, r, x, 0, 0, n - 1, segtree);
            } else {
                System.out.println(query(l, r, 0, 0, n - 1, segtree, fibs));
            }
        }
    }

    private static int query(int l, int r, int node, int tl, int tr, int[][] segtree, int[] fibs) {
        if (tl == l && tr == r) {
            return segtree[node][0] * fibs[r - l + 1] % MOD;
        }
        if (l > tr || r < tl) {
            return 0;
        }
        int mid = (tl + tr) / 2;
        return (query(l, Math.min(r, mid), 2 * node + 1, tl, mid, segtree, fibs) +
                query(Math.max(l, mid + 1), r, 2 * node + 2, mid + 1, tr, segtree, fibs)) % MOD;
    }

    private static void update(int l, int r, int val, int node, int tl, int tr, int[][] segtree) {
        if (tl == l && tr == r) {
            segtree[node][0] = (segtree[node][0] + val) % MOD;
            segtree[node][1] = (segtree[node][1] + val) % MOD;
            return;
        }
        if (l > tr || r < tl) {
            return;
        }
        int mid = (tl + tr) / 2;
        update(l, Math.min(r, mid), val, 2 * node + 1, tl, mid, segtree);
        update(Math.max(l, mid + 1), r, val, 2 * node + 2, mid + 1, tr, segtree);
        segtree[node][0] = (segtree[2 * node + 1][0] + segtree[2 * node + 2][0]) % MOD;
        segtree[node][1] = (segtree[2 * node + 1][1] + segtree[2 * node + 2][1]) % MOD;
    }

    private static void build(int[] array, int node, int tl, int tr, int[][] segtree) {
        if (tl == tr) {
            segtree[node] = new int[]{array[tl], array[tl]};
        } else {
            int mid = (tl + tr) / 2;
            build(array, 2 * node + 1, tl, mid, segtree);
            build(array, 2 * node + 2, mid + 1, tr, segtree);
            segtree[node][0] = (segtree[2 * node + 1][0] + segtree[2 * node + 2][0]) % MOD;
            segtree[node][1] = (segtree[2 * node + 1][1] + segtree[2 * node + 2][1]) % MOD;
        }
    }
}
