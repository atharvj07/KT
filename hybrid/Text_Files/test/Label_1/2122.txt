#include <bits/stdc++.h>
using namespace std;
const int N = 5e4 + 5;
int n, a[N], pref[N];
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> a[i];
        int x;
        cin >> x;
        for (int i = 1; i <= n; i++) pref[i] = a[i] + pref[i - 1];
        int mx = 0;
        for (int i = 1; i <= n; i++) {
            int l = i, r = n, ans = i;
            while (l <= r) {
                int mid = (l + r) >> 1;
                int sum = pref[mid] - pref[i - 1];
                if (sum >= (mid - i + 1) * x) {
                    ans = mid;
                    l = mid + 1;
                } else
                    r = mid - 1;
            }
            mx = max(mx, ans - i + 1);
        }
        cout << mx << endl;
    }
    return 0;
}
