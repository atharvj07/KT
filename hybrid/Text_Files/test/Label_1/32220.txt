import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();

    int[] limits = new int[m + 1];
    for (int i = 0; i < n; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      limits[a] = Math.max(limits[a], b);
    }

    boolean[] visited = new boolean[m + 1];
    for (int i = m; i >= 0; i--) {
      if (visited[i]) {
        continue;
      }
      for (int j = i; j <= limits[i]; j++) {
        visited[j] = true;
      }
    }

    System.out.println(visited[m] ? "YES" : "NO");
  }
}
