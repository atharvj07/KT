import java.util.*;

class Main {

    static class Node {
        int data;
        Node left;
        Node right;

        Node(int data) {
            this.data = data;
            this.left = null;
            this.right = null;
        }
    }

    static class BinaryTree {
        Node root;

        BinaryTree() {
            this.root = null;
        }

        void insert(int data) {
            if (this.root == null) {
                this.root = new Node(data);
                return;
            }

            Node current = this.root;
            Node parent = null;
            while (current != null) {
                parent = current;
                if (data < current.data) {
                    current = current.left;
                } else {
                    current = current.right;
                }
            }

            if (data < parent.data) {
                parent.left = new Node(data);
            } else {
                parent.right = new Node(data);
            }
        }

        void inOrder(List<Integer> list) {
            inOrder(this.root, list);
        }

        private void inOrder(Node node, List<Integer> list) {
            if (node == null) {
                return;
            }

            inOrder(node.left, list);
            list.add(node.data);
            inOrder(node.right, list);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int c = scanner.nextInt();

        Map<Integer, List<Integer>> left = new HashMap<>();
        Map<Integer, List<Integer>> right = new HashMap<>();

        for (int i = 0; i < c; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            String direction = scanner.next();

            if (direction.equals("LEFT")) {
                if (!left.containsKey(a)) {
                    left.put(a, new ArrayList<>());
                }
                left.get(a).add(b);
            } else {
                if (!right.containsKey(a)) {
                    right.put(a, new ArrayList<>());
                }
                right.get(a).add(b);
            }
        }

        BinaryTree tree = new BinaryTree();
        boolean possible = true;
        for (int i = 1; i <= n; i++) {
            if (left.containsKey(i)) {
                for (int j : left.get(i)) {
                    if (j < i) {
                        tree.insert(j);
                    } else {
                        possible = false;
                    }
                }
            }

            tree.insert(i);

            if (right.containsKey(i)) {
                for (int j : right.get(i)) {
                    if (j > i) {
                        tree.insert(j);
                    } else {
                        possible = false;
                    }
                }
            }
        }

        if (possible) {
            List<Integer> inOrderList = new ArrayList<>();
            tree.inOrder(inOrderList);
            for (int i = 0; i < inOrderList.size(); i++) {
                System.out.print(inOrderList.get(i));
                if (i < inOrderList.size() - 1) {
                    System.out.print(" ");
                }
            }
            System.out.println();
        } else {
            System.out.println("IMPOSSIBLE");
        }
    }
}
