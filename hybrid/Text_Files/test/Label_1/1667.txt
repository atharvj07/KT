#include <bits/stdc++.h>
#include <iostream>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(k);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i < k; i++) {
        cin >> b[i];
    }
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    int i = 0, j = 0;
    bool ok = false;
    while (i < n && j < k) {
        if (a[i] == 0) {
            if (j == 0 || b[j] > b[j - 1]) {
                a[i] = b[j];
                j++;
            } else {
                ok = true;
                break;
            }
        } else {
            if (b[j] < a[i]) {
                ok = true;
                break;
            }
        }
        i++;
    }
    if (i == n && j == k) {
        ok = true;
    }
    if (ok) {
        cout << "Yes" << endl;
    } else {
        cout << "No" << endl;
    }
    return 0;
}
