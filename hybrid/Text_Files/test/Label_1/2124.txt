#include<bits/stdc++.h>
using namespace std;
int main(){
    int n; cin >> n;
    while(n--){
      string s, t; cin >> s >> t;
      int ss[26] = {0};
      for(char c : s) ss[c-'a']++;
      bool f = true;
      for(char c : t){
        if(ss[c-'a'] == 0){
          f = false;
          break;
        }
        ss[c-'a']--;
      }
      if(f) cout << "YES\n";
      else cout << "NO\n";
    }
    return 0;
}
