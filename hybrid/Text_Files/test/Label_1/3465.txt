#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    long long n, a, b, p, q;
    cin >> n >> a >> b >> p >> q;

    long long lcm = (a * b) / __gcd(a, b);

    long long total_chocolates = (n/a)*p + (n/b)*q;

    if (lcm <= n) {
        total_chocolates -= ((n/lcm)*(min(p, q)));
        total_chocolates += (n/lcm)*max(p, q);
    }

    cout << total_chocolates << "\n";

    return 0;
}
