import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] pieces = sc.nextLine().split(" ");
        int numRooks = 0;
        for (String piece : pieces) {
            if (piece.charAt(0) == 'r') {
                numRooks++;
            }
        }
        String rook1 = pieces[0];
        String rook2 = pieces[1];
        String whiteKing = pieces[2];
        String blackKing = pieces[3];
        char blackKingRow = blackKing.charAt(1);
        char blackKingCol = blackKing.charAt(0);
        boolean checkmate = false;
        if (numRooks == 2) {
            if (whiteKing.charAt(0) == 'h') {
                if (whiteKing.charAt(1) == '8') {
                    if (rook1.charAt(0) == 'h' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'g' && rook2.charAt(1) == '8') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'g' && rook1.charAt(1) == '8') {
                        if (rook2.charAt(0) == 'h' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    }
                } else if (whiteKing.charAt(1) == '7') {
                    if (rook1.charAt(0) == 'h' && rook1.charAt(1) == '6') {
                        if (rook2.charAt(0) == 'g' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'g' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'h' && rook2.charAt(1) == '6') {
                            checkmate = true;
                        }
                    }
                }
            } else if (whiteKing.charAt(0) == 'a') {
                if (whiteKing.charAt(1) == '8') {
                    if (rook1.charAt(0) == 'a' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'b' && rook2.charAt(1) == '8') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'b' && rook1.charAt(1) == '8') {
                        if (rook2.charAt(0) == 'a' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    }
                } else if (whiteKing.charAt(1) == '7') {
                    if (rook1.charAt(0) == 'a' && rook1.charAt(1) == '6') {
                        if (rook2.charAt(0) == 'b' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'b' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'a' && rook2.charAt(1) == '6') {
                            checkmate = true;
                        }
                    }
                }
            } else if (whiteKing.charAt(0) == 'c') {
                if (whiteKing.charAt(1) == '7') {
                    if (rook1.charAt(0) == 'c' && rook1.charAt(1) == '6') {
                        if (rook2.charAt(0) == 'd' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'd' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'c' && rook2.charAt(1) == '6') {
                            checkmate = true;
                        }
                    }
                }
            } else if (whiteKing.charAt(0) == 'f') {
                if (whiteKing.charAt(1) == '7') {
                    if (rook1.charAt(0) == 'f' && rook1.charAt(1) == '6') {
                        if (rook2.charAt(0) == 'e' && rook2.charAt(1) == '7') {
                            checkmate = true;
                        }
                    } else if (rook1.charAt(0) == 'e' && rook1.charAt(1) == '7') {
                        if (rook2.charAt(0) == 'f' && rook2.charAt(1) == '6') {
                            checkmate = true;
                        }
                    }
                }
            }
        } else if (numRooks == 1) {
            if (blackKingCol == rook1.charAt(0)) {
                if (blackKingRow == rook1.charAt(1) + 1 || blackKingRow == rook1.charAt(1) - 1) {
                    checkmate = true;
                }
            } else if (blackKingRow == rook1.charAt(1)) {
                if (blackKingCol == rook1.charAt(0) + 1 || blackKingCol == rook1.charAt(0) - 1) {
                    checkmate = true;
                }
            }
        }
        System.out.println(checkmate ? "CHECKMATE" : "OTHER");
    }
}
