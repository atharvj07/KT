#include <iostream>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        long long n, k;
        cin >> n >> k;

        long long aPos = k % n;
        if (aPos == 0) aPos = n;

        long long bPos = k % (n - 1);
        if (bPos == 0) bPos = n - 1;

        if (bPos >= aPos) bPos++;

        cout << bPos << endl;
    }

    return 0;
}
