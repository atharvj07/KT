import sys

t = int(input())
for _ in range(t):
    n = int(input())
    words = [list(input()) for _ in range(n)]
    freq = [0] * 5
    for word in words:
        for letter in word:
            freq[ord(letter) - ord('a')] += 1
    flag = False
    for i in range(5):
        if freq[i] > sum(freq) - freq[i]:
            flag = True
            break
    if flag:
        print(n)
    else:
        print(0)
