T = int(input())
for _ in range(T):
    n = int(input())
    a = list(map(int, input().split()))
    count = {}
    for item in a:
        if item in count:
            count[item] += 1
        else:
            count[item] = 1
    max_count = max(count.values())
    num_max_count = list(count.values()).count(max_count)
    ans = (n - num_max_count) // (max_count - 1) - 1
    print(ans)
