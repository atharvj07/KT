import sys
input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[], ] * n

for _ in range(m):
    a, b, w = map(int, input().split())
    a -= 1
    b -= 1
    adj[a] += [(b, w)]

dp = [[0 for j in range(100001)] for i in range(n)]

for x in range(n):
    for w in range(1, 100001):
        for u, ww in adj[x]:
            if ww > w:
                dp[x][w] = max(dp[x][w], dp[u][ww] + 1)

print(max(max(x) for x in dp))
