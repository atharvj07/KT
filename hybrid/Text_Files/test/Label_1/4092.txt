#include <iostream>
#include <vector>

using namespace std;

bool check(vector<long long int> a, long long int d) {
  for (int i = 0; i < a.size() - 1; i++) {
    if ((a[i] % d == 0) == (a[i + 1] % d == 0)) {
      return false;
    }
  }
  return true;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    bool found = false;
    for (long long int d = 1; d <= 1000000000000000000; d++) {
      if (check(a, d)) {
        cout << d << endl;
        found = true;
        break;
      }
    }
    if (!found) {
      cout << 0 << endl;
    }
  }
  return 0;
}
