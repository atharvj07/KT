import java.util.*;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] seq1 = new int[n];
        for (int i = 0; i < n; i++) {
            seq1[i] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[] seq2 = new int[m];
        for (int i = 0; i < m; i++) {
            seq2[i] = sc.nextInt();
        }
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (seq1[i - 1] == seq2[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }
        int k = dp[n][m];
        int[] lcs = new int[k];
        int i = n;
        int j = m;
        while (i > 0 && j > 0) {
            if (seq1[i - 1] == seq2[j - 1]) {
                lcs[k - 1] = seq1[i - 1];
                k--;
                i--;
                j--;
            } else if (dp[i - 1][j] > dp[i][j - 1]) {
                i--;
            } else {
                j--;
            }
        }
        System.out.println(k);
        System.out.println(Arrays.stream(lcs).mapToObj(String::valueOf).collect(Collectors.joining(" ")));
    }
}
