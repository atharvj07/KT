n = int(input().strip())
a = list(map(int, input().strip().split()))

prefix_sum = {0: -1}
current_sum = 0
last_invalid_index = -1
good_subarrays = 0

for i in range(n):
    current_sum += a[i]
    if current_sum in prefix_sum:
        last_invalid_index = max(last_invalid_index, prefix_sum[current_sum])
    good_subarrays += i - last_invalid_index
    prefix_sum[current_sum] = i

print(good_subarrays)
