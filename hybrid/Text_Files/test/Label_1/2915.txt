#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b, p;
    cin >> a >> b >> p;
    string s;
    cin >> s;
    int pr = -1;
    int n = s.size();
    for (int i = 0; i < n; i++) {
      if (s[i] != s[0]) pr = i;
    }
    if (pr == -1 || a <= p)
      cout << 1 << '\n';
    else
      cout << pr + 1 << '\n';
  }
  return 0;
}
