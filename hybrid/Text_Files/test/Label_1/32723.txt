import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    long h = scan.nextLong();
    long m = scan.nextLong();
    int k = scan.nextInt();
    long h1 = scan.nextLong();
    long m1 = scan.nextLong();
    long h2 = scan.nextLong();
    long m2 = scan.nextLong();

    long dif = (h2 - h1) * m + (m2 - m1) + 1;

    long[] d = new long[20];
    long x = dif;
    for (int i = 0; x > 0; i++) {
      d[i] = x % 10;
      x /= 10;
    }

    int s = 0;
    int l = (int) x;
    if (l + 1 <= k) {
      s = 1;
    }

    Arrays.sort(d);

    for (int i = 0; i <= l; i++) {
      if (d[i] <= s) {
        continue;
      }
      if (s + d[i] >= k) {
        s += d[i];
        continue;
      }
      s += 2;
    }

    System.out.println(s);
  }
}
