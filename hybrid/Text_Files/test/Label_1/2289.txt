#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long x, y;
    cin >> x >> y;
    long long ans = 0;
    for (long long i = 1; i <= y; i++) {
      long long low = 1, high = x, mid;
      while (low <= high) {
        mid = (low + high) / 2;
        if (mid / i == mid % i) {
          ans++;
          low = mid + 1;
        } else {
          high = mid - 1;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
