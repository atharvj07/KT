n = int(input())
positions = list(map(int, input().split()))
positions.sort()

# Calculate target positions for both colors
black_targets = [i for i in range(1, n+1, 2)]
white_targets = [i for i in range(2, n+1, 2)]

# Calculate minimum moves to place all pieces on black or white cells
moves_to_black = sum(abs(positions[i] - black_targets[i]) for i in range(n//2))
moves_to_white = sum(abs(positions[i] - white_targets[i]) for i in range(n//2))

# Output the minimum of two possible moves
print(min(moves_to_black, moves_to_white))
