#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, l, r;
    scanf("%d%d%d", &n, &l, &r);
    vector<int> a(n), p(n), ans(n, -1);
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    for (int i = 0; i < n; i++) scanf("%d", &p[i]);
    for (int i = n - 1; i >= 1; i--) {
        for (int j = i - 1; j >= 0; j--) {
            if (p[i] > p[j]) {
                ans[i] = r;
                ans[j] = r - (p[i] - p[j] - 1);
            }
        }
        if (ans[i] == -1) return puts("-1"), 0;
        if (ans[i] < l) return puts("-1"), 0;
    }
    for (int i = 0; i < n; i++) printf("%d ", ans[i]);
    return 0;
}
