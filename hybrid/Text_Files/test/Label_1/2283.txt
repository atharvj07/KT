#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  int k;
  cin >> s >> k;
  int n = s.size();
  int l = 2; 
  for (; l <= k + n; l++) {
    int i = 0, j = l;
    bool f = 1;
    while (j < n + k) {
      if (s[i % n] != s[j % n]) {
        f = 0;
        break;
      }
      i++;
      j++;
    }
    if (f) break;
  }
  cout << l - 2 << endl;
  return 0;
}
