#include <bits/stdc++.h>
using namespace std;
const int N = 1e3 + 5, M = 1e5 + 5;
int n, m;
int x[M], y[M];
bool blocked[N][N];
bool vis[N][N][4];  // 0=top, 1=right, 2=bottom, 3=left
bool valid(int i, int j, int dir) {
  if (i <= 0 || i > n || j <= 0 || j > n) return false;
  if (blocked[i][j]) return false;
  if (dir >= 2) return vis[i][j][dir - 2];
  return vis[i][j][dir + 2];
}
int dfs(int i, int j, int dir) {
  if (!valid(i, j, dir)) return 0;
  if (vis[i][j][dir]) return -1;
  vis[i][j][dir] = true;
  int ans = 1;
  if (dir == 0 || dir == 2) {
    ans = max(ans, 1 + dfs(i + (dir - 1), j, 1));
    ans = max(ans, 1 + dfs(i + (dir - 1), j, 3));
  } else {
    ans = max(ans, 1 + dfs(i, j + (dir - 2), 0));
    ans = max(ans, 1 + dfs(i, j + (dir - 2), 2));
  }
  return ans;
}
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> m;
  for (int i = 0; i < m; i++) {
    cin >> x[i] >> y[i];
    blocked[x[i]][y[i]] = true;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
      if (!blocked[i][j]) {
        memset(vis, false, sizeof(vis));
        ans = max(ans, dfs(i, j, 0));
        ans = max(ans, dfs(i, j, 1));
        ans = max(ans, dfs(i, j, 2));
        ans = max(ans, dfs(i, j, 3));
      }
  cout << ans << "\n";
  return 0;
}
