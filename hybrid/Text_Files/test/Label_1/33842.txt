import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        char[] s = new char[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            s[i] = (sum <= a[0]) ? '+' : '-';
        }
        System.out.println(s);
    }
}
