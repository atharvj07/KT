import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int t = sc.nextInt();
            if (t == 0) {
                break;
            }

            int[] cycle = new int[t];
            for (int i = 0; i < t; i++) {
                cycle[i] = sc.nextInt();
            }

            int n = sc.nextInt();
            int[][] interviews = new int[n][2];
            for (int i = 0; i < n; i++) {
                interviews[i][0] = sc.nextInt();
                interviews[i][1] = sc.nextInt();
            }

            int minCaffeine = Integer.MAX_VALUE;
            for (int i = 0; i < t; i++) {
                int caffeine = 0;
                int currentDay = i;
                int currentHour = 0;

                for (int j = 0; j < n; j++) {
                    int interviewDay = interviews[j][0];
                    int interviewHour = interviews[j][1];

                    if (currentDay == interviewDay) {
                        if (currentHour + cycle[currentDay] < interviewHour) {
                            caffeine++;
                            currentHour = 0;
                        }
                    } else if (currentDay < interviewDay) {
                        int daysToInterview = interviewDay - currentDay;
                        int hoursToInterview = interviewHour;

                        for (int k = 0; k < daysToInterview; k++) {
                            hoursToInterview += cycle[(currentDay + k) % t];
                        }

                        if (hoursToInterview > 23) {
                            caffeine++;
                            hoursToInterview -= 23;
                        }

                        currentDay = interviewDay;
                        currentHour = hoursToInterview;
                    } else {
                        int daysToInterview = interviewDay + t - currentDay;
                        int hoursToInterview = interviewHour;

                        for (int k = 0; k < daysToInterview; k++) {
                            hoursToInterview += cycle[(currentDay + k) % t];
                        }

                        if (hoursToInterview > 23) {
                            caffeine++;
                            hoursToInterview -= 23;
                        }

                        currentDay = interviewDay;
                        currentHour = hoursToInterview;
                    }
                }

                minCaffeine = Math.min(minCaffeine, caffeine);
            }

            System.out.println(minCaffeine);
        }
    }
}
