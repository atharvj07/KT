#include <bits/stdc++.h>
using namespace std;
inline long long minSwap(string s, int i, int n){
    if(i==n)
        return stoll(s);
    string ss=s;
    swap(s[i],s[i+1]);
    long long a=minSwap(s,i+1,n);
    s=ss;
    return min(a,minSwap(s,i+1,n));
}
int main(){
    int t;
    cin>>t;
    while(t--){
        string s;
        cin>>s;
        cout<<minSwap(s,0,s.size()-1)<<'\n';
    }
    return 0;
}
