import sys

t = int(input())
for _ in range(t):
    s = input()
    n = len(s)
    ab = 0
    ba = 0
    for i in range(n):
        if s[i] == 'a':
            ab += i == 0 or s[i - 1] == 'b'
        else:
            ba += i == 0 or s[i - 1] == 'a'
    if ab == ba:
        print(s)
    else:
        for i in range(n):
            if s[i] == 'a':
                if ab > ba:
                    s = s[:i] + 'b' + s[i + 1:]
                    ab -= 1
                    ba += i != 0 and s[i - 1] == 'a'
            else:
                if ba > ab:
                    s = s[:i] + 'a' + s[i + 1:]
                    ba -= 1
                    ab += i != 0 and s[i - 1] == 'b'
            if ab == ba:
                break
        print(s)
