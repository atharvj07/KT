import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] rook1 = sc.next().toCharArray();
        char[] rook2 = sc.next().toCharArray();
        char[] whiteKing = sc.next().toCharArray();
        char[] blackKing = sc.next().toCharArray();
        int rook1x = rook1[0] - 'a';
        int rook1y = rook1[1] - '1';
        int rook2x = rook2[0] - 'a';
        int rook2y = rook2[1] - '1';
        int whiteKingx = whiteKing[0] - 'a';
        int whiteKingy = whiteKing[1] - '1';
        int blackKingx = blackKing[0] - 'a';
        int blackKingy = blackKing[1] - '1';
        if ((rook1x == blackKingx || rook2x == blackKingx || rook1y == blackKingy || rook2y == blackKingy) && (whiteKingx != blackKingx || whiteKingy != blackKingy)) {
            System.out.println("CHECKMATE");
        } else {
            System.out.println("OTHER");
        }
    }
}
