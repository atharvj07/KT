import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int c = sc.nextInt();
        int v0 = sc.nextInt();
        int v1 = sc.nextInt();
        int a = sc.nextInt();
        int l = sc.nextInt();
        int total = 0;
        int days = 0;
        while (total < c) {
            days++;
            if (v0 + a * (days - 1) >= v1) {
                total += v1 - l;
            } else {
                total += v0 + a * (days - 1) - l;
            }
        }
        System.out.println(days);
    }
}
