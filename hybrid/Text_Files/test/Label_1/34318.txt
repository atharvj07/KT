import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[][] field = new char[4][4];
        for (int i = 0; i < 4; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < 4; j++) {
                field[i][j] = line.charAt(j);
            }
        }
        boolean win = false;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (field[i][j] == '.') {
                    field[i][j] = 'x';
                    if (checkWin(field)) {
                        win = true;
                        break;
                    }
                    field[i][j] = '.';
                }
            }
            if (win) break;
        }
        System.out.println(win ? "YES" : "NO");
    }

    private static boolean checkWin(char[][] field) {
        for (int i = 0; i < 4; i++) {
            if (field[i][0] != '.' && field[i][0] == field[i][1] && field[i][0] == field[i][2] && field[i][0] == field[i][3]) {
                return true;
            }
            if (field[0][i] != '.' && field[0][i] == field[1][i] && field[0][i] == field[2][i] && field[0][i] == field[3][i]) {
                return true;
            }
        }
        if (field[0][0] != '.' && field[0][0] == field[1][1] && field[0][0] == field[2][2] && field[0][0] == field[3][3]) {
            return true;
        }
        if (field[0][3] != '.' && field[0][3] == field[1][2] && field[0][3] == field[2][1] && field[0][3] == field[3][0]) {
            return true;
        }
        return false;
    }
}
