#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> knows(n);
  for (int i = 0; i < n; i++) {
    int k;
    cin >> k;
    for (int j = 0; j < k; j++) {
      int a;
      cin >> a;
      knows[i].push_back(a);
    }
  }
  vector<int> needs(m + 1, 0);
  for (int i = 0; i < n; i++) {
    int cnt = 0;
    for (int j = 0; j < knows[i].size(); j++) {
      needs[knows[i][j]] = 1;
      cnt++;
    }
    if (cnt == 0) {
      needs[0] = 1;
    }
  }
  int ans = 0;
  for (int i = 0; i < m + 1; i++) {
    if (needs[i] == 1) {
      ans++;
    }
  }
  cout << ans - 1 << endl;
  return 0;
}
