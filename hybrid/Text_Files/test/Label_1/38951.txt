from collections import defaultdict

def max_beauty(k, n):
    # Store strings and their beauty
    string_beauty = defaultdict(list)
    for _ in range(k):
        s, beauty = input().split()
        string_beauty[s].append(int(beauty))

    max_beauty = 0
    used = set()  # Keep track of used string pairs

    # Process for palindrome pairs and single palindromes
    for s, beauties in string_beauty.items():
        beauties.sort(reverse=True)
        rev = s[::-1]

        if s == rev:  # It's a palindrome itself
            max_beauty += sum(beauty for i, beauty in enumerate(beauties) if i % 2 == 0)
        elif s not in used and rev in string_beauty:
            used.add(s)
            used.add(rev)
            # Pair strings with their reverse counterparts
            for beauty_s, beauty_rev in zip(beauties, string_beauty[rev]):
                max_beauty += max(0, beauty_s + beauty_rev)

    print(max_beauty)

# Read input
k, n = map(int, input().split())
max_beauty(k, n)
