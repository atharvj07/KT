a = list(map(int,input().split()))
n = m = int(len(a)**0.5)
assert n*m == len(a)
d = sorted(zip(a,range(n*m)))
z = d[0][0]
if z == 0:
    print(n,m)
    print(n//2+1,m//2+1)
    exit()
for i in range(1,n*m):
    if d[i][0] != z + i:
        print(-1)
        exit()
print(n,m)
print((d[0][1]//m)+1,(d[0][1]%m)+1)
