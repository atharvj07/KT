import sys

n = int(input())

adj = {i: [] for i in range(1, n + 1)}
for i in range(n - 1):
    a, b = list(map(int, input().split()))
    adj[a].append(b)
    adj[b].append(a)

exp = [0] * (n + 1)
vis = [False] * (n + 1)

def dfs(cur, par):
    vis[cur] = True
    size = 1
    for u in adj[cur]:
        if u == par or vis[u]: continue
        size += dfs(u, cur)
    exp[cur] = (exp[par] * (n - size) + 1 + size) / n
    return size

dfs(1, 0)
print(exp[1])
