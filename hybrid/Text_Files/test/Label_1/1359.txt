#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
#define vvi vector<vi>

int main() {
  int n, m;
  cin >> n >> m;
  vi p(n);
  for (int &x : p) cin >> x;
  vvi adj(n);
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    a--;
    b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  vi d(n, -1);
  queue<int> q;
  p[n - 1]--;
  q.push(p[n - 1]);
  d[p[n - 1]] = 0;
  int ans = 0;
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    ans = max(ans, d[u]);
    for (int v : adj[u]) {
      if (d[v] == -1) {
        d[v] = d[u] + 1;
        q.push(v);
      }
    }
  }
  cout << ans << endl;
}
