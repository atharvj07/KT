import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (scanner.hasNextLine()) {
            String input = scanner.nextLine();
            int r1 = 0;
            int r2 = 0;
            char op = ' ';
            boolean error = false;
            for (char c : input.toCharArray()) {
                if (Character.isDigit(c)) {
                    r2 = r2 * 10 + c - '0';
                } else if (c == '+') {
                    if (op != ' ') {
                        r1 = calculate(r1, r2, op);
                    } else {
                        r1 = r2;
                    }
                    op = c;
                    r2 = 0;
                } else if (c == '-') {
                    if (op != ' ') {
                        r1 = calculate(r1, r2, op);
                    } else {
                        r1 = r2;
                    }
                    op = c;
                    r2 = 0;
                } else if (c == '*') {
                    if (op != ' ') {
                        r1 = calculate(r1, r2, op);
                    } else {
                        r1 = r2;
                    }
                    op = c;
                    r2 = 0;
                } else if (c == '=') {
                    if (op != ' ') {
                        r1 = calculate(r1, r2, op);
                    } else {
                        r1 = r2;
                    }
                    if (r1 < 0 || r1 > 9999) {
                        error = true;
                    }
                    break;
                }
            }
            System.out.println(error ? "E" : r1);
        }
        scanner.close();
    }

    private static int calculate(int r1, int r2, char op) {
        switch (op) {
            case '+':
                return r1 + r2;
            case '-':
                return r1 - r2;
            case '*':
                return r1 * r2;
            default:
                return r2;
        }
    }
}
