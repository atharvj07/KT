def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

m = int(input().strip())
s = list(map(int, input().strip().split()))

if m == 1:
    print(1)
    print(s[0])
else:
    if s[0] != 1:
        print(-1)
    else:
        n = m + 1
        sequence = [0] * n
        sequence[0] = s[-1]
        for i in range(1, n):
            sequence[i] = (s[-1] // s[i - 1]) * s[i - 1]
        print(n)
        print(' '.join(map(str, sequence)))
