from collections import Counter
for _ in range(int(input())):
    n = int(input())
    words = [Counter(input()) for _ in range(n)]
    ch = ord('a')
    while ch <= ord('e'):
        a = -1
        b = 0
        for word in words:
            a = max(a, word[chr(ch)])
            b += word[chr(ch)]
        if 2 * a > b:
            print(b)
            break
        ch += 1
    else:
        print(0)
