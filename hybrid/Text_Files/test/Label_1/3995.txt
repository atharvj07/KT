#include <iostream>
#include <set>
#include <map>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> p(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
  }
  vector<int> s(n + 1);
  map<int, vector<int>> mp;
  for (int i = 1; i <= n; ++i) {
    cin >> s[i];
    mp[s[i]].push_back(p[i]);
  }
  vector<int> c(k + 1);
  set<int> st;
  for (int i = 1; i <= k; ++i) {
    cin >> c[i];
    st.insert(s[c[i]]);
  }
  for (auto& x : mp) {
    sort(x.second.rbegin(), x.second.rend());
  }
  int ans = st.size();
  for (int i = 1; i <= k; ++i) {
    auto& v = mp[s[c[i]]];
    int pos = lower_bound(v.begin(), v.end(), p[c[i]]) - v.begin();
    if (pos != 0) {
      ans -= 1;
      break;
    }
  }
  cout << ans << endl;
  return 0;
}
