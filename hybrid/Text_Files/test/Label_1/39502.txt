n, m = map(int, input().split())
grid = [input() for _ in range(n)]

def solve(n, m, grid):
    for i in range(n):
        for j in range(m):
            if grid[i][j] == '#':
                row_has_black = any(grid[i][k] == '#' for k in range(m) if k != j)
                col_has_black = any(grid[k][j] == '#' for k in range(n) if k != i)
                if not row_has_black and not col_has_black:
                    return "No"
    return "Yes"

print(solve(n, m, grid))
