n = int(input())
a = []
for i in range(n):
    x, y = map(int, input().split())
    a.append([x, y])
cnt = [0 for i in range(100001)]
ans = []
for x, y in a:
    cnt[x] += 1
    cnt[y] += 1
for x, y in a:
    if cnt[x] == 1:
        ans.append((n - 1, 1))
    else:
        ans.append((n - 1 - cnt[x] // 2, cnt[x] // 2))
for i in ans:
    print(*i)
