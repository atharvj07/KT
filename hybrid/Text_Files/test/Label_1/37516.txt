import heapq
import bisect
n=int(input())
a=list(map(int,input().split()))
m=int(input())
q=[(0,0,[])]
for _ in range(m):
    k,j=map(int,input().split())
    k-=1
    j-=1
    heapq.heappush(q,(a[j]-a[k]*(k+1-j),k,a[j:]))
r=[0]*m
for i in range(m):
    s,l,x=heapq.heappop(q)
    r[i]=x.pop(0)
    if j:
        for y in x:
            heapq.heappush(q,(s+y-l-1,l+1,x[:bisect.bisect_left(x,y)]+[y]))
    j+=1
print(*r,sep='\n')
