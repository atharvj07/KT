import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = input.nextInt();
        int ans = 0;
        for (int i = 1; i < n; i++) {
            int x = Math.min(-a[i - 1], a[i]);
            ans += x;
            a[i] -= x;
            a[i - 1] += x;
        }
        System.out.println(ans);
    }
}
