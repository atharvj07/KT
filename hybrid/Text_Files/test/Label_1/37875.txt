import sys

def solve():
    t = int(input())
    for _ in range(t):
        k = int(input())
        n = int(input())
        st = str(n)
        sz = len(st)
        for i in range(0, k):
            if sz > 1:
                print(sz - 1)
                print(st[:i] + st[i + 1:])
                return
        print(0)

solve()
