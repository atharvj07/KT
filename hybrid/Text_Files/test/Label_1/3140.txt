#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int M=1e9+7;
int n,a[1000005],mn[1000005],sum;
int32_t main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    for(int i=n;i>=1;i--){
        mn[i]=a[i];
        for(int j=i+1;j<=n;j++){
            mn[i]=min(mn[i],a[j]);
        }
    }
    for(int i=1;i<=n;i++){
        sum=(sum+1ll*mn[i])%M;
    }
    cout<<sum<<endl;
    return 0;
}
