import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int l = in.nextInt();
        int[] time = new int[n];
        int[][] pos = new int[n][2];
        for (int i = 0; i < n; i++) {
            time[i] = in.nextInt();
            pos[i][0] = in.nextInt();
            pos[i][1] = in.nextInt();
        }
        int[] dx = new int[]{0, -1, 0, 1};
        int[] dy = new int[]{1, 0, -1, 0};
        String ans = "NO";
        for (char c = 'U'; c <= 'R'; c++) {
            for (char d = 'U'; d <= 'R'; d++) {
                boolean ok = true;
                int x = 0, y = 0;
                for (int i = 0; i < n; i++) {
                    int t = time[i] % (l * 2);
                    if (t >= l) {
                        x += dx[d - 'U'];
                        y += dy[d - 'U'];
                        t -= l;
                    }
                    x += dx[c - 'U'] * (t / 2);
                    y += dy[c - 'U'] * (t / 2);
                    if (x != pos[i][0] || y != pos[i][1]) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    ans = "" + c + d;
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
