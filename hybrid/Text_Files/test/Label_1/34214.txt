import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int h = sc.nextInt();
            int m = sc.nextInt();
            int s = sc.nextInt();
            if (h == -1 && m == -1 && s == -1) {
                break;
            }
            int totalSeconds = h * 3600 + m * 60 + s;
            int remainingSeconds = 120 * 60 - totalSeconds;
            int h1 = remainingSeconds / 3600;
            int m1 = (remainingSeconds % 3600) / 60;
            int s1 = remainingSeconds % 60;
            remainingSeconds = remainingSeconds * 3;
            int h2 = remainingSeconds / 3600;
            int m2 = (remainingSeconds % 3600) / 60;
            int s2 = remainingSeconds % 60;
            System.out.printf("%02d:%02d:%02d\n", h1, m1, s1);
            System.out.printf("%02d:%02d:%02d\n", h2, m2, s2);
        }
    }
}
