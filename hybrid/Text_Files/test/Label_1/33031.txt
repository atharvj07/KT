import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);

        long maxSum = 0;
        for (int i = 0; i < n - 1; i++) {
            maxSum += Math.min(arr[i], arr[i + 1]);
        }

        System.out.println(maxSum);
    }
}
