import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[][] p = new int[n][2];
        int minx = 0, miny = 0, maxx = 0, maxy = 0;
        for (int i = 0; i < n; i++) {
            p[i][0] = input.nextInt();
            p[i][1] = input.nextInt();
            minx = Math.min(minx, p[i][0]);
            miny = Math.min(miny, p[i][1]);
            maxx = Math.max(maxx, p[i][0]);
            maxy = Math.max(maxy, p[i][1]);
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (p[i][0] == p[j][0])
                    continue;
                int b = (p[j][1] - p[i][1]) / (p[j][0] - p[i][0]);
                int c = p[i][1] - p[i][0] * b;
                boolean valid = true;
                for (int k = 0; k < n; k++) {
                    if (k == i || k == j)
                        continue;
                    if (p[k][1] > p[k][0] * b + c) {
                        valid = false;
                        break;
                    }
                }
                if (valid)
                    ans++;
            }
        }
        System.out.println(ans);
    }
}
