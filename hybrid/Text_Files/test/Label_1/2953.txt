#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<int> b(n);
    int cnt = 0;
    while (a != b) {
        b = a;
        for (int i = 1; i < n - 1; i++) {
            a[i] = (a[i - 1] + a[i] + a[i + 1]) / 2;
        }
        cnt++;
    }
    if (cnt == 0) {
        cout << 1 << endl;
    } else {
        cout << cnt << endl;
        for (int i = 0; i < n; i++) {
            cout << a[i] << ' ';
        }
        cout << endl;
    }
    return 0;
}
