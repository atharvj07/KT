import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int sum1 = 0;
        int sum2 = 0;

        for (int i = 0; i < 3; i++) {
            sum1 += s.charAt(i) - '0';
            sum2 += s.charAt(i + 3) - '0';
        }

        if (sum1 != sum2) {
            System.out.println(Math.abs(sum1 - sum2));
        } else {
            System.out.println(0);
        }
    }
}
