n, m, k, q = map(int, input().split())
t = []
for _ in range(k):
  r, c = map(int, input().split())
  t.append((r, c))
b = [int(x) for x in input().split()]
t.sort(key=lambda x: x[0])
ans = 0
r = 1
while t:
  _, c = t[0]
  if c in b:
    ans += r - 1
    r = 1
    while t and t[0][1] == c:
      t.pop(0)
  else:
    r += 1
    t.pop(0)
print(ans)
