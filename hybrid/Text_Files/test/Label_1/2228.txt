#include <bits/stdc++.h>
using namespace std;

const int maxn = 3e5 + 5;
const long long inf = 1e18;

struct edge {
    int to, w;
};

vector<edge> adj[maxn];
vector<long long> uf(maxn), bu(maxn), wu(maxn);
int n;

void dfs(int u, int p) {
    if (p != -1) {
        uf[u] += uf[p];
        if (bu[p] > 0) {
            wu[u] = uf[u] + bu[p] - wu[p];
            bu[u] = wu[u] + wu[p];
            uf[u] = -inf;
        } else
            bu[u] += bu[p];
    }
    for (auto& e : adj[u]) {
        if (e.to != p)
            dfs(e.to, u);
    }
}

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> wu[i]; }
    for (int i = 1; i < n; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }
    dfs(1, -1);
    cout << *max_element(wu.begin() + 1, wu.begin() + n + 1) << endl;
    return 0;
}
