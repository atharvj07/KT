n, m = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
l, r = [[int(x) for x in input().split()] for i in range(m)]
ans = 0
for i in range(1 << m):
    cur = 0
    for j in range(m):
        if (i >> j) & 1:
            for k in range(l[j] - 1, r[j]):
                cur += a[k]
    ans = max(ans, cur)
print(ans)
