#include <iostream>
#include <cstring>

using namespace std;

int main() {
  int n;
  cin >> n;
  char s[n + 1];
  cin >> s;

  int room[10] = {0};

  for (int i = 0; i < n; i++) {
    if (s[i] == 'L') {
      for (int j = 0; j < 10; j++) {
        if (!room[j]) {
          room[j] = 1;
          break;
        }
      }
    } else if (s[i] == 'R') {
      for (int j = 9; j >= 0; j--) {
        if (!room[j]) {
          room[j] = 1;
          break;
        }
      }
    } else {
      room[s[i] - '0'] = 0;
    }
  }

  for (int i = 0; i < 10; i++) {
    cout << room[i];
  }

  return 0;
}
