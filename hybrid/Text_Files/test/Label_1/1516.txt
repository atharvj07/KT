#include <iostream>
#include <vector>
using namespace std;

int main() {
    long long nA, nB, k, m;
    cin >> nA >> nB;
    cin >> k >> m;
    vector<long long> a(nA), b(nB);
    for (long long i = 0; i < nA; i++) cin >> a[i];
    for (long long i = 0; i < nB; i++) cin >> b[i];
    cout << ((k + m <= nA + nB && a[k - 1] < b[m - 1]) ? "YES" : "NO")
         << endl;
    return 0;
}
