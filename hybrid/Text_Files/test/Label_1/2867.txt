#include <bits/stdc++.h>
using namespace std;
#define int long long
vector<int> p;
vector<bool> vis;
int32_t main() {
  int n, z;
  cin >> n >> z;
  p.resize(n);
  vis.resize(n, 0);
  for (int i = 0; i < n; i++) cin >> p[i];
  sort(p.begin(), p.end());
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (vis[i]) continue;
    int x = p[i];
    ans++;
    for (int j = i + 1; j < n; j++) {
      if (p[j] >= x + z) {
        vis[j] = 1;
        x = p[j];
      }
    }
  }
  cout << ans << endl;
}
