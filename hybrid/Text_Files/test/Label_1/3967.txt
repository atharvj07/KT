#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, d, k; cin >> n >> d >> k; if (n == 1) { cout << "YES\n"; return 0; } if (n <= 3 || d > n - 1 || d % 2 != n % 2 || d > 2 * k) { cout << "NO\n"; return 0; } cout << "YES\n"; int l = (n + 1) / 2, m = n - l; for (int i = 2; i <= m; i++) { cout << "1 " << i << "\n"; } for (int i = m + 2; i <= l; i++) { cout << "2 " << i << "\n"; } cout << "2 " << m + 1 << "\n";
}
