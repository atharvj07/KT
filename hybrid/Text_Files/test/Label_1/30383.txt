import java.util.*;
import java.io.*;

class Main {

    static int mod = 998244353;

    static int dfs(int node, int par, int[] subTreeColor, List<Integer>[] adj) {
        Set<Integer> color = new HashSet<>();
        color.add(subTreeColor[node]);
        for (int child : adj[node]) {
            if (child != par) {
                int childColor = dfs(child, node, subTreeColor, adj);
                if (childColor == -1) return -1;
                color.add(childColor);
            }
        }
        if (node != 1 && color.size() == 1) return -1;
        int ans = 1;
        for (int c : color) ans = (ans * c) % mod;
        return ans;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());

        int[] subTreeColor = new int[n + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) subTreeColor[i] = Integer.parseInt(st.nextToken());

        List<Integer>[] adj = new List[n + 1];
        for (int i = 1; i <= n; i++) adj[i] = new ArrayList<>();

        for (int i = 1; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj[u].add(v);
            adj[v].add(u);
        }
        int ans = dfs(1, 1, subTreeColor, adj);
        if (ans == -1) ans = 0;
        System.out.println(ans);
    }
}
