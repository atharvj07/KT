cnt1, cnt2, x, y = map(int, input().split())

if x == y:
    print(max(cnt1 + 1, cnt2 + 1))
else:
    need = cnt1 + cnt2
    if need & 1:
        need += 1
    if need <= x*y:
        print(need)
    else:
        need2 = (need + 1) // 2
        if need2 % x == 0 or need2 % y == 0:
            need2 += 1
        print(need2)
