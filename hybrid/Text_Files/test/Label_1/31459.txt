import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long K = sc.nextLong();
        long A = sc.nextLong();
        long B = sc.nextLong();

        if (A <= B) {
            System.out.println(-1);
            return;
        }

        long ans = 0;
        long rating = 0;
        while (rating < K) {
            rating += A;
            ans += 1;
            rating -= B;
        }

        System.out.println(ans);
    }
}
