#include <iostream>
#include <vector>

using namespace std;

const int MX = 1e5 + 5;

int sub[MX], vis[MX];
int a[MX];

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    int l, r;
    cin >> l >> r;
    sub[l]++, sub[r + 1]--;
  }
  int sum = 0;
  for (int i = 1; i <= n; i++) {
    sum += sub[i];
    vis[sum]++;
  }
  int cnt = 0;
  for (int i = 0; i <= n; i++) {
    if (!vis[i]) cnt++;
  }
  cout << cnt << endl;
  for (int i = 1; i <= n; i++) {
    if (vis[a[i] ^ 0] || vis[a[i] ^ 1])
      a[i] = vis[a[i] ^ 0] ? a[i] ^ 1 : a[i] ^ 0;
    else
      a[i] = 0;
    cout << a[i] << ' ';
  }
  cout << endl;
  return 0;
}
