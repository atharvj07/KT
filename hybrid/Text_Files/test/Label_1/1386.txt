#include <bits/stdc++.h>
using namespace std;
using ll=long long;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> p(n);
    for (int i = 0; i < n; ++i) cin >> p[i];
    sort(p.begin(), p.end());
    int key = 0, diff = k;
    for (int i = 0; i < n; ++i) {
        if (p[i] > key + diff) {
            key = p[i];
            diff = k;
        }
        cout << key << ' ';
        diff -= p[i] - key;
    }
    cout << '\n';
    return 0;
}
