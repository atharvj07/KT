h, w = map(int, input().split())
r = list(map(int, input().split()))
c = list(map(int, input().split()))
mod = 1000000007

ans = 1
for i in range(h):
    x = min(r[i], w - c[0] - c[w - 1])
    ans = ans * (x + 1) % mod
    for j in range(1, w - 1):
        x = min(r[i], w - c[j - 1] - c[j])
        ans = ans * (x + 1) % mod
print(ans)
