def mod_pow(base, pwr, mod):
    if pwr == 0:
        return 1
    if pwr == 1:
        return base
    val = mod_pow(base, int(pwr / 2), mod)
    val = (val * val) % mod
    if pwr % 2 != 0:
        val = (val * base) % mod
    return val % mod


h, w = map(int, input().split())
r = list(map(int, input().split()))
c = list(map(int, input().split()))

dp = [[0] * (w + 1) for _ in range(h + 1)]
dp[0][0] = 1
for i in range(h):
    for j in range(w):
        if r[i] >= j and c[j] >= i:
            dp[i + 1][j + 1] = (dp[i][j + 1] + dp[i + 1][j]) % (10**9 + 7)

print(dp[h][w])
