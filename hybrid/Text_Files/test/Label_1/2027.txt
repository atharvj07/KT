#include <iostream>
#include <cstdio>
#define ll long long
#define ld long double
using namespace std;
int main() {
    ll n, p, q, a[100005], b[100005];
    ld ans = 0;
    cin >> n >> p >> q;
    ll xp = 0, mny = 0;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        xp += a[i];
        mny += b[i];
    }
    if (xp == 0) {cout << -1 << endl; return 0;}
    if (mny >= q) cout << (ld)p / xp << endl;
    else {
        ld tmp = (ld)(q - mny) / xp;
        ll tot = tmp * xp;
        for (ll i = 1; i <= n; i++) {
            ll aa = (a[i] * tot) / xp, bb = b[i] * tot / xp;
            if ((tmp + (ld)aa / xp) >= (ld)p / xp) {
                tmp += (ld)aa / xp;
                ans += (ld)bb / b[i];
                break;
            } else {
                ans += tmp;
                tmp = (ld)p / xp;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
