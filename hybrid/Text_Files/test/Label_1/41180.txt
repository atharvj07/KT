n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
b = [[0, 0] for _ in range(n + 1)]
ans = 0
for i in range(1, n + 1):
  b[i][0] = b[i][1] = -1
  for j in range(1, m + 1):
    if a[i - 1][j - 1]:
      if b[i][0] == -1:
        b[i][0] = j
      b[i][1] = j
      if b[i - 1][0] == -1:
        b[i - 1][0] = j
      b[i - 1][1] = j
  if b[i][0] != -1:
    for j in range(b[i][0], b[i][1] + 1):
      if a[i - 1][j - 1]:
        if b[i][0] == -1:
          b[i][0] = j
        b[i][1] = j
        if b[i - 1][0] == -1:
          b[i - 1][0] = j
        b[i - 1][1] = j
for i in range(1, n + 1):
  for j in range(1, m + 1):
    if a[i - 1][j - 1] and b[i][0] != -1 and b[i][1] != -1 and (b[i][0], b[i][1]) != (j, j):
      ans += 1
print(ans)
