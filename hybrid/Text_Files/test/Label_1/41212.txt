import sys  
input = sys.stdin.readline

def mod(x):
    return x % 998244353

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    mx = max(a)
    cur = sum(a)
    ans = 1
    for i in range(1,mx+1):
        if cur <= 0:
            ans = 0
            break
        cur -= i*n
        ans = mod(ans * n)
    print(ans)
