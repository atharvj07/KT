import java.util.*;

public class Main {

    static long mod = (long) (1e9 + 7);

    static Map<Integer, List<Edge>> graph;
    static int[] dist;
    static int[] last;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        graph = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int w = sc.nextInt();
            graph.computeIfAbsent(a, k -> new ArrayList<>()).add(new Edge(b, w));
            graph.computeIfAbsent(b, k -> new ArrayList<>()).add(new Edge(a, w));
        }
        int k = sc.nextInt();
        int[] stops = new int[k];
        for (int i = 0; i < k; i++) {
            stops[i] = sc.nextInt() - 1;
        }
        dist = new int[n];
        Arrays.fill(dist, Integer.MAX_VALUE);
        last = new int[n];
        Arrays.fill(last, -1);
        dist[0] = 0;
        Queue<Integer> queue = new LinkedList<>();
        queue.add(0);
        while (!queue.isEmpty()) {
            int u = queue.poll();
            for (Edge e : graph.getOrDefault(u, List.of())) {
                int v = e.to;
                int w = e.weight * (1 << last[u]);
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    last[v] = u;
                    queue.add(v);
                }
            }
        }
        long ans = 0;
        for (int i = 0; i < k - 1; i++) {
            int u = stops[i];
            int v = stops[i + 1];
            if (dist[v] > dist[u]) {
                ans += dist[v] - dist[u];
            }
        }
        System.out.println(ans % mod);
    }

    static class Edge {
        int to, weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }
}
