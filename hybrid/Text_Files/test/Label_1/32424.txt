import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Set;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        char[] grid = reader.readLine().toCharArray();

        int n = grid.length;
        int[][] dp = new int[n + 1][3];
        Set<Integer> bombs = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            Arrays.fill(dp[i], -1);
            if (grid[i - 1] == '*') {
                bombs.add(i);
            }
        }

        System.out.println(countWays(1, 0, bombs, grid, dp));
    }

    private static int countWays(int i, int prev, Set<Integer> bombs, char[] grid, int[][] dp) {
        if (i > grid.length) {
            return 1;
        }

        if (dp[i][prev] != -1) {
            return dp[i][prev];
        }

        int ways = 0;
        if (grid[i - 1] == '?' || grid[i - 1] == '0') {
            ways = countWays(i + 1, 0, bombs, grid, dp);
        } else if (grid[i - 1] == '1') {
            if (bombs.contains(i - 1) || bombs.contains(i) || bombs.contains(i + 1)) {
                ways = 0;
            } else {
                ways = countWays(i + 1, 1, bombs, grid, dp);
            }
        } else if (grid[i - 1] == '2') {
            if (!bombs.contains(i - 1) && !bombs.contains(i) && !bombs.contains(i + 1) && prev == 0) {
                ways = countWays(i + 1, 2, bombs, grid, dp);
            } else if (bombs.contains(i - 1) || bombs.contains(i + 1)) {
                ways = 0;
            } else {
                ways = countWays(i + 1, 1, bombs, grid, dp);
            }
        }

        dp[i][prev] = ways % MOD;
        return ways;
    }
}
