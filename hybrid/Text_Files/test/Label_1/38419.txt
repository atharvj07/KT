n = int(input())
c = list(map(int, input().split()))
dp = [n] * (n + 1)
dp[0] = dp[1] = 0
for i in range(2, n + 1):
    dp[i] = dp[i - 1] + 1
    if c[i - 2] == c[i - 1]:
        dp[i] = min(dp[i], dp[i - 2] + 1)
    for j in range(i - 3, -1, -1):
        if c[j] == c[i - 1]:
            dp[i] = min(dp[i], dp[j] + 1)
print(dp[n])
