#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0, cur = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) ans = max(ans, cur), cur = 0;
    else cur++;
  }
  ans = max(ans, cur);
  cout << ans << endl;
}
