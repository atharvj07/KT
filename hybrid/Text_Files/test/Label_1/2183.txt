#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;

int n, a[N], p[N], s[N];

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  p[1] = a[1];
  for (int i = 2; i <= n; i++) p[i] = p[i - 1] + a[i];
  s[n] = a[n];
  for (int i = n - 1; i >= 1; i--) s[i] = s[i + 1] + a[i];
  int ans = -2e9;
  for (int i = 1; i < n; i++) {
    ans = max(ans, p[i] + s[i + 1] - a[i]);
    ans = max(ans, p[i - 1] + s[i + 1]);
  }
  cout << ans << '\n';
}
