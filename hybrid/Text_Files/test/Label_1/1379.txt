#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

int main() {
    ll n;
    cin >> n;
    if (n % 2 == 0) {
        cout << "YES" << endl;
        cout << 1 << endl;
        cout << n / 2 << " " << n << endl;
    } else if (n % 3 == 0) {
        cout << "YES" << endl;
        cout << 2 << endl;
        cout << n / 3 << " " << n << endl;
        cout << 2 * n / 3 << " " << n << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
