#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n, m;
    cin >> n >> m;
    vector<ll> a(n), b(m);
    for (ll &i : a) cin >> i;
    for (ll &i : b) cin >> i;
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    ll ans = 1e18;
    vector<int> x, y;
    for (int sw = 0; sw <= 2; sw++) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                ll sum_a = 0, sum_b = 0;
                vector<ll> aa = a;
                vector<ll> bb = b;
                for (int z = 0; z < sw; z++) {
                    swap(aa[i], bb[j]);
                }
                for (ll i : aa) sum_a += i;
                for (ll i : bb) sum_b += i;
                if (abs(sum_a - sum_b) < ans) {
                    x.clear(), y.clear();
                    ans = abs(sum_a - sum_b);
                    for (int z = 0; z < sw; z++) {
                        x.push_back(i + 1);
                        y.push_back(j + 1);
                    }
                }
            }
        }
    }
    cout << ans << endl;
    cout << x.size() << endl;
    for (int i = 0; i < (int)x.size(); i++) {
        cout << x[i] << " " << y[i] << endl;
    }
    return 0;
}
