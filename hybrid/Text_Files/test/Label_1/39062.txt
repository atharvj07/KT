n = int(input())
a = input()
b = ['0'] * n
c = ['1'] * n
d = ['2'] * n
ans = list(a)
for i in range(n):
    if a[i] == '0' or ((i % 3) and (a[(i - 1) % n] == '1') and (a[(i + 1) % n] == '2')): continue
    if a[i] == '1':
        if a[(i - 1) % n] == '2': b[i] = '0'
        else: d[i] = '0'
    else:
        if a[(i - 1) % n] == '1': c[i] = '0'
        else: b[i] = '0'
    if b[i] != a[i]: ans[i] = b[i]; break
    if c[i] != a[i]: ans[i] = c[i]; break
    if d[i] != a[i]: ans[i] = d[i]; break
print(''.join(ans))
