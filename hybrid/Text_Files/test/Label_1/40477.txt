from collections import Counter
from itertools import permutations

s = input()
c = Counter(s)
odd = [x for x in c if c[x] % 2]
if len(odd) > 1: print(-1); exit()
if len(odd):
    m, odd = c[odd[0]], odd[0]
    for x in c:
        if x != odd and c[x] % 2: print(-1); exit()
        c[x] = c[x]//2
else:
    m = 0
    for x in c: c[x] = c[x]//2
a = []
for x in c:
    for y in range(c[x]): a.append(x)
if m: a.append(odd*m)
b = sorted(a)
print(''.join(b + b[::-1]))
