import java.util.Arrays;
import java.util.Scanner;

public class Main {

    private static final long MOD = 1000000007;
    private static final long MODINV = 562500004;
    private static long[][] dp;
    private static int n;
    private static int[][] p;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        p = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                p[i][j] = in.nextInt();
            }
        }
        in.close();

        dp = new long[1 << n][n];
        for (int i = 0; i < (1 << n); i++) {
            Arrays.fill(dp[i], -1);
        }
        long result = dfs(0, 0);
        System.out.println(result);
    }

    private static long dfs(int mask, int last) {
        if (mask == (1 << n) - 1) {
            return 1;
        }
        if (dp[mask][last] != -1) {
            return dp[mask][last];
        }
        dp[mask][last] = 0;
        for (int i = 0; i < n; i++) {
            if ((mask & (1 << i)) == 0) {
                dp[mask][last] = (dp[mask][last] + p[last][i] * dfs(mask | (1 << i), i)) % MOD;
            }
        }
        return dp[mask][last] * MODINV % MOD;
    }
}
