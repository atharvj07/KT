import sys,bisect
inp=list(map(int,sys.stdin.read().split()))
lsts=[inp[4*i:4*(i+1)]for i in range((len(inp)-4)//4)]
if len(lsts)<inp[-1]:sys.stdout.write("-1")
else:
    lsts.sort(key=lambda x:len(x));ans=[];j=0
    for i in range(inp[-1]):
        while j<len(lsts)and lsts[j][1]<=i:j+=1
        if j<len(lsts):ans.append(lsts[j][0])
    sys.stdout.write(str(sorted(ans)[inp[-1]-1]))
