#include <iostream>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    bool f = true;
    for (int j = 0; j <= k; j++) {
      if (a % 10 == j) {
        a /= 10;
        continue;
      } else {
        f = false;
        break;
      }
    }
    ans += f;
  }
  cout << ans << endl;
  return 0;
}
