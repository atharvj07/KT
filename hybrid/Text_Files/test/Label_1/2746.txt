#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int maxn = 1e5 + 5;
ll x[maxn];

int main() {
  ll n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> x[i];
  for (int i = 1; i <= n - 1; i++) cout << x[i + 1] - x[i] << ' ';
  cout << "0\n";
  for (int i = 2; i <= n; i++) cout << x[i] - x[1] << ' ';
  cout << "0\n";
  return 0;
}
