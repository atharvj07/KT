import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            int n = s.length();
            long ans = 0;
            int[] cnt = new int[26];
            for (int i = 0; i < n; i++) {
                ans += ++cnt[s.charAt(i) - 'a'];
            }
            System.out.println(ans);
        }
    }
}
