import java.util.Scanner;

public class DominoCarpet {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine(); // Consume the newline

        char[][] carpet = new char[4 * n + 1][4 * m + 1];
        for (int i = 0; i < 4 * n + 1; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < 4 * m + 1; j++) {
                carpet[i][j] = line.charAt(j);
            }
        }

        long[][][] dp = new long[n + 1][m + 1][2]; // dp[i][j][k] = number of ways to cover the first i rows and j columns,
                                                    // where the last domino placed is horizontal (k = 0) or vertical (k = 1)

        dp[0][0][0] = dp[0][0][1] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (carpet[4 * i + 2][4 * j + 1] == 'O' && carpet[4 * i + 1][4 * j + 2] == 'O') {
                    dp[i + 1][j][0] = (dp[i][j][0] + dp[i][j][1]) % MOD; // Place a horizontal domino
                    dp[i][j + 1][1] = (dp[i][j][0] + dp[i][j][1]) % MOD; // Place a vertical domino
                } else if (carpet[4 * i + 2][4 * j + 1] == 'O') {
                    dp[i][j + 1][1] = (dp[i][j][0] + dp[i][j][1]) % MOD; // Place a vertical domino
                } else if (carpet[4 * i + 1][4 * j + 2] == 'O') {
                    dp[i + 1][j][0] = (dp[i][j][0] + dp[i][j][1]) % MOD; // Place a horizontal domino
                }
            }
        }

        System.out.println((dp[n][m][0] + dp[n][m][1]) % MOD);
    }
}
