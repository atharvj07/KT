import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int A = sc.nextInt();
        int B = sc.nextInt();
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int l = sc.nextInt();
            int t = sc.nextInt();
            int m = sc.nextInt();
            long r = l - 1 + (long) (Math.min(t, m) - 1) * m;
            if (r < l || (r - l) % B != 0 || (r - l + A) / B < m) {
                System.out.println(-1);
            } else {
                System.out.println(r);
            }
        }
    }
}
