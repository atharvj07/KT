import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt(), d = scanner.nextInt();
        int[] c = new int[m];
        for (int i = 0; i < m; i++) c[i] = scanner.nextInt();
        int[] plat = new int[n + 2];
        for (int i : c) for (int j = 1; i > 0; i--) plat[j++] = plat[j - 1] + 1;
        int[] a = new int[n + 2]; Arrays.fill(a, -1);
        int l = 0;
        for (int i = 1; i <= n + 1;) {
            int r = Math.min(n + 1, i + d);
            for (; i <= r && plat[i] > 0; i++) {
                if (a[i] != -1) break;
                a[i] = plat[i];
            }
            if (i > r) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
        for (int i = 1; i <= n + 1; i++) {
            if (a[i] == -1) a[i] = 0;
            System.out.print(a[i] + " ");
        }
    }
}
