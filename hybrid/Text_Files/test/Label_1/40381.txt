n = int(input())
a = list(map(int, input().split()))
b = [0] * n
for i in range(n):
    if i < a[i]:
        b[i] = i
    else:
        b[i] = a[i] + 1
        for j in range(a[i]):
            if j not in a[:i+1]:
                b[i] = j
                break
if len({x for x in b}) == n:
    print(*b)
else:
    print(-1)
