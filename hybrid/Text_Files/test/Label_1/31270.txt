import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        int[] b = new int[n];
        int count = 0;
        do {
            count++;
            System.arraycopy(a, 0, b, 0, n);
            for (int i = 1; i < n - 1; i++) {
                a[i] = (b[i - 1] + b[i] + b[i + 1]) / 2;
            }
        } while (!java.util.Arrays.equals(a, b));

        if (count == 1) {
            System.out.println(-1);
        } else {
            System.out.println(count - 1);
            for (int i : a) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
