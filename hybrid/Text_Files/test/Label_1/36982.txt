import sys

s = input()
q = int(input())
pref = [[0] * 26 for _ in range(len(s) + 1)]
for i in range(len(s)):
    pref[i + 1] = pref[i].copy()
    pref[i + 1][ord(s[i]) - ord('a')] += 1
for _ in range(q):
    t, *args = map(int, input().split())
    if t == 2:
        l, r = args
        print(sum([pref[r][i] - pref[l - 1][i] for i in range(26)]))
    else:
        c, i = args
        s = s[:i - 1] + c + s[i:]
        pref[i] = pref[i - 1].copy()
        pref[i][ord(c) - ord('a')] += 1
