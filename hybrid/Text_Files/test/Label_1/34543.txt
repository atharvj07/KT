import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long ans;
        if (k % 7 == 0) {
            ans = 1;
        } else {
            ans = k / 7 + 1;
        }
        System.out.println(ans);
    }
}
