for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [list(map(int, input().split())) for _ in range(n)]
    ans = 0
    for i in range(n):
        for j in range(m // 2):
            ans += abs(a[i][j] - a[i][m - j - 1])
    for j in range(m):
        for i in range(n // 2):
            ans += abs(a[i][j] - a[n - i - 1][j])
    print(ans)
