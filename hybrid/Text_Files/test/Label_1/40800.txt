n, a, b = map(int, input().split())
s = input()
k = 0
i = 0
while i < n:
    if s[i] == '.':
        if k % 2 == 0:
            if a > 0:
                k += 1
                a -= 1
            elif b > 0:
                k += 2
                b -= 1
        else:
            if b > 0:
                k += 2
                b -= 1
            elif a > 0:
                k += 1
                a -= 1
    i += 1
print(k)
