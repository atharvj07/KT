import java.util.*;

public class Main {
    static int[] last, ans;
    static int n, m, k;
    static char[] s;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        s = sc.next().toCharArray();
        last = new int[k];
        ans = new int[m + 1];
        Arrays.fill(last, -1);
        for (int i = 0; i < n; i++) {
            last[s[i] - 'a'] = i;
        }
        for (int i = 1; i <= m; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt();
                char c = sc.next().charAt(0);
                Arrays.fill(last, -1);
                for (int j = l; j < r; j++) {
                    s[j] = c;
                }
                last[c - 'a'] = r - 1;
            } else {
                String p = sc.next();
                int[] pPos = new int[k];
                for (int j = 0; j < k; j++) {
                    pPos[p.charAt(j) - 'a'] = j;
                }
                int d = n + 1;
                for (int j = 0; j < k; j++) {
                    if (last[pPos[j]] != -1) {
                        d = Math.min(d, last[pPos[j]] + 1);
                    }
                }
                ans[i] = d;
                System.out.println(d);
            }
        }
        sc.close();
    }
}
