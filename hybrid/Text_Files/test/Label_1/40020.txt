n = int(input())
t = list(map(int, input().split()))
c = {}
for x in t:
    c[x] = 0
for i in range(n):
    for j in range(i, n):
        d = {}
        for k in range(i, j + 1):
            if t[k] not in d:
                d[t[k]] = 0
            d[t[k]] += 1
        mx = max(d.values())
        for k in d:
            if d[k] == mx:
                c[k] += 1
                break
for x in t:
    print(c[x], end=' ')
