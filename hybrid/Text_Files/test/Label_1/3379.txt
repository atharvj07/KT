#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
  int n, m, r;
  cin >> n >> m >> r;
  vector<ll> s(n), b(m);
  for (auto &x : s) cin >> x;
  for (auto &x : b) cin >> x;
  sort(s.rbegin(), s.rend());
  sort(b.rbegin(), b.rend());
  vector<ll> dp(r + 1, -1e18);
  dp[0] = r;
  for (ll x : s) {
    for (ll i = r; i >= x; --i) dp[i] = max(dp[i], dp[i - x] + x);
  }
  ll ans = 0;
  for (ll x : b) {
    for (ll i = 0; i <= r; ++i) {
      if (dp[i] < 0) continue;
      ans = max(ans, dp[i] + i - x);
    }
  }
  cout << ans << endl;
}
