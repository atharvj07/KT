n, k = map(int, input().split())
a = list(map(int, input().split()))
b = []
for i in range(n - 1):
    b.append(a[i + 1] - a[i] - k)
    if b[-1]:
        if b[-1] > 0:
            a[i + 1] -= b[-1]
            b[-1] = 0
        else:
            a[i] += b[-1]
            b[-1] = 0
print(len(b))
for i in b:
    if i:
        print(("+ {} {}".format(i + 1, abs(int(i))) if i > 0 else ("- {} {}".format(i + 1, abs(int(i))))))
