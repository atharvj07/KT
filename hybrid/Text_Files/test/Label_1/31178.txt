import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = reader.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]);
        int q = Integer.parseInt(tokens[1]);
        FenwickTree fenwickTree = new FenwickTree(n);
        Map<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < q; i++) {
            tokens = reader.readLine().split(" ");
            if (tokens[0].equals("0")) {
                int left = Integer.parseInt(tokens[1]) - 1;
                int right = Integer.parseInt(tokens[2]);
                System.out.println(fenwickTree.rangeSum(left, right));
            } else {
                int id = Integer.parseInt(tokens[0]);
                if (!map.containsKey(id)) {
                    int hook = fenwickTree.findFirstAvailableHook();
                    fenwickTree.add(hook, 1);
                    map.put(id, hook);
                } else {
                    int hook = map.get(id);
                    fenwickTree.add(hook, -1);
                    map.remove(id);
                }
            }
        }
    }

    static class FenwickTree {
        int[] tree;
        int n;

        public FenwickTree(int n) {
            this.n = n;
            tree = new int[n + 1];
        }

        public void add(int index, int value) {
            while (index <= n) {
                tree[index] += value;
                index += (index & (-index));
            }
        }

        public int rangeSum(int left, int right) {
            return rangeSum(right) - rangeSum(left - 1);
        }

        private int rangeSum(int index) {
            int sum = 0;
            while (index > 0) {
                sum += tree[index];
                index -= (index & (-index));
            }
            return sum;
        }

        public int findFirstAvailableHook() {
            int left = 1;
            int right = n;
            while (left < right) {
                int mid = (left + right) / 2;
                if (rangeSum(mid) == 0) {
                    left = mid + 1;
                } else {
                    right = mid;
                }
            }
            return left;
        }
    }
}
