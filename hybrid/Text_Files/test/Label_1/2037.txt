#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int n, b, m;
    ll ansx = 0, ansy = 0;
    cin >> n >> b;
    for (int i = n - 1, mul = 1; i >= 0; i--) {
        int x;
        cin >> x;
        ansx = ansx + (x * mul);
        mul *= b;
    }
    cin >> m >> b;
    for (int i = m - 1, mul = 1; i >= 0; i--) {
        int x;
        cin >> x;
        ansy = ansy + (x * mul);
        mul *= b;
    }
    if (ansx < ansy)
        cout << '<';
    else if (ansx > ansy)
        cout << '>';
    else
        cout << '=';
    return 0;
}
