def max_total_comfort(n, city_codes):
    city_codes_dict = {}
    for i, code in enumerate(city_codes):
        if code in city_codes_dict:
            city_codes_dict[code].append(i)
        else:
            city_codes_dict[code] = [i]
    intervals = []
    for indices in city_codes_dict.values():
        intervals.append((indices[0], indices[-1]))
    intervals.sort()

    merged = []
    for interval in intervals:
        if not merged or merged[-1][1] < interval[0]:
            merged.append(interval)
        else:
            merged[-1] = (merged[-1][0], max(merged[-1][1], interval[1]))

    result = 0
    for interval in merged:
        distinct_codes = set(city_codes[interval[0]:interval[1]+1])
        xor_sum = 0
        for code in distinct_codes:
            xor_sum ^= code
        result += xor_sum
    return result

n = int(input().strip())
city_codes = list(map(int, input().strip().split()))
print(max_total_comfort(n, city_codes))
