#include<iostream>
using namespace std;
typedef int i64;
i64 main() {
    i64 v;
    cin >> v;
    i64 m = v, ans = v;
    for (i64 i = 2; i * i <= v; i++) {
        while (v % i == 0) {
            m = m / i;
            ans = ans - ans / i;
        }
    }
    if (v > 1) {
        m = m / v;
        ans = ans - ans / v;
    }
    cout << ans << endl;
    return 0;
}
