import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            grid[i] = input.next().toCharArray();
        }
        int min = n * m;
        int[][] minCell = new int[n][m];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = m - 1; j >= 0; j--) {
                if (grid[i][j] == 'W') {
                    continue;
                }
                if (i + 1 < n && j + 1 < m && grid[i + 1][j] == 'B' && grid[i][j + 1] == 'B' && grid[i + 1][j + 1] == 'B') {
                    minCell[i][j] = Math.min(Math.min(minCell[i + 1][j], minCell[i][j + 1]), minCell[i + 1][j + 1]) + 1;
                } else {
                    minCell[i][j] = 1;
                }
                min = Math.min(min, minCell[i][j]);
            }
        }
        if (min == n * m) {
            System.out.println(-1);
        } else {
            System.out.println(min);
        }
    }
}
