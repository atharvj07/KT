import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
    int n = Integer.parseInt(tokenizer.nextToken());
    int m = Integer.parseInt(tokenizer.nextToken());
    long k = Long.parseLong(tokenizer.nextToken());
    long[] colors = new long[m + 1];
    StringTokenizer colorTokenizer = new StringTokenizer(reader.readLine());
    for (int i = 0; i < n; i++) {
      colors[Integer.parseInt(colorTokenizer.nextToken())]++;
    }
    k = Math.min(k, m);
    long[][] dp = new long[m + 1][(int) k + 1];
    dp[0][0] = 1;
    for (int i = 1; i <= m; i++) {
      for (int j = 0; j <= k; j++) {
        long tmp = 0;
        for (int l = 0; l <= colors[i] && l <= j; l++) {
          tmp += dp[i - 1][j - l] * combinations(colors[i], l) % 1009;
          tmp %= 1009;
        }
        dp[i][j] = tmp;
      }
    }

    System.out.println(dp[m][(int) k]);
  }

  private static long combinations(long n, long k) {
    long numerator = 1;
    long denominator = 1;
    for (long i = 1; i <= k; i++) {
      numerator = numerator * (n - i + 1) % 1009;
      denominator = denominator * i % 1009;
    }
    return (numerator * fastPow(denominator, 1007)) % 1009;
  }

  private static long fastPow(long base, long pow) {
    if (pow == 0) {
      return 1;
    }
    if (pow == 1) {
      return base;
    }
    long temp = fastPow(base, pow / 2);
    temp = temp * temp % 1009;
    if (pow % 2 == 0) {
      return temp;
    } else {
      return temp * base % 1009;
    }
  }
}
