import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.next().toCharArray();
        int n = s.length;
        int ans = 0;
        int bal = 0;
        for (char c : s) {
            bal += c - '0';
            if (bal == 0) ans++;
            else if (bal < 0) {
                ans++;
                bal = 1;
            }
        }
        System.out.println(ans);
    }
}
