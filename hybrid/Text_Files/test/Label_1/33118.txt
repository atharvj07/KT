import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = a[i - 1] + sc.nextInt();
        }
        long sum = a[n];
        if (sum % 3 != 0) {
            System.out.println(0);
            return;
        }
        long mid = sum / 3;
        long count = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] == mid) {
                count += (a[n] - a[i] - mid) / mid;
            }
        }
        System.out.println(count);
    }
}
