import java.util.*;

public class Cinema {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        long s = scanner.nextLong();
        int t = scanner.nextInt();

        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        Arrays.sort(gasStations);

        long minPrice = -1;
        for (Car car : cars) {
            if (canReachCinema(car, gasStations, s, t)) {
                minPrice = Math.min(minPrice, car.price);
            }
        }

        System.out.println(minPrice);
    }

    private static boolean canReachCinema(Car car, int[] gasStations, long s, int t) {
        int currentFuel = car.capacity;
        int timeSpent = 0;

        for (int gasStation : gasStations) {
            long distanceToGasStation = gasStation;
            int timeToGasStation = (int) Math.ceil((double) distanceToGasStation / car.normalSpeed);
            timeSpent += timeToGasStation;

            if (timeSpent > t) {
                return false;
            }

            currentFuel -= distanceToGasStation;
            if (currentFuel < 0) {
                return false;
            }

            currentFuel = car.capacity;
        }

        long distanceToCinema = s - gasStations[gasStations.length - 1];
        int timeToCinema = (int) Math.ceil((double) distanceToCinema / car.normalSpeed);
        timeSpent += timeToCinema;

        return timeSpent <= t;
    }

    private static class Car {
        int price;
        int capacity;
        int normalSpeed = 1;
        int acceleratedSpeed = 2;

        public Car(int price, int capacity) {
            this.price = price;
            this.capacity = capacity;
        }
    }
}
