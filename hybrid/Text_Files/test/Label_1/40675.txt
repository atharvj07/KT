for t in range(int(input())):
    n = int(input())
    ans = [0] * n
    cur = 0
    for i in range(n):
        l = r = i
        while l >= 0 and r < n:
            if ans[l] == ans[r]:
                l -= 1
                r += 1
            else:
                break
        if (r - l - 1) % 2 == 1:
            ans[(l + r) // 2] = i + 1
        else:
            ans[(l + r - 1) // 2] = i + 1
    print(*ans)
