#include <bits/stdc++.h>

using namespace std;

const int MAXN = 200005;

int n,m,q;
long long a[MAXN],b[MAXN],ans[MAXN];

int main()
{
    cin>>n>>m>>q;
    for (int i=0; i<n; i++) cin>>a[i];
    for (int i=0; i<m; i++) cin>>b[i];
    for (int i=0; i<q; i++)
    {
        long long k;
        cin>>k;
        sort(a,a+n); sort(b,b+m);
        long long cura=0,curb=0;
        int j=0;
        for (int i=0; i<n; i++)
        {
            while (j<m&&b[j]<=a[i]+k)
            {
                if (b[j]>curb) curb=b[j];
                j++;
            }
            if (curb>cura)
            {
                cura=curb;
                ans[i]=cura;
            }
            else ans[i]=cura;
        }
        long long best=0;
        for (int i=0; i<n; i++) best=max(best,ans[i]);
        cout<<best<<endl;
    }
}
