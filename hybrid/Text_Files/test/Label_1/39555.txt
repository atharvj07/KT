MOD = 998244353

n = int(input())
portals = [list(map(int, input().split())) for _ in range(n)]

dp = [0] * n
for i in range(n):
    x, y, s = portals[i]
    if i == 0:
        dp[i] = x
    else:
        dp[i] = (dp[i - 1] + x - portals[i - 1][0]) % MOD
    if s == 1:
        left = 0
        right = i - 1
        while left <= right:
            mid = (left + right) // 2
            if portals[mid][0] >= y:
                right = mid - 1
            else:
                left = mid + 1
        dp[i] = (dp[i] + ((dp[right] if right >= 0 else 0) + MOD - y) % MOD) % MOD

answer = (dp[n - 1] + 1) % MOD
print(answer)
