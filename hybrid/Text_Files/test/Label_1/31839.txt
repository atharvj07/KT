import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if (n % 4 == 0) {
            System.out.print(String.format("%0" + n / 4 + "d", 4));
        } else if (n % 7 == 0) {
            System.out.print(String.format("%0" + n / 7 + "d", 7));
        } else if (n % (4 + 7) == 0) {
            System.out.print(String.format("%0" + n / (4 + 7) + "d", 47));
        } else {
            System.out.println(-1);
        }
    }
}
