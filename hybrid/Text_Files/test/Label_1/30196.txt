import java.io.*;
import java.util.*;

public class Main {
    private static final int NOT_VISITED = -1;

    private static BufferedReader br;
    private static StringTokenizer st;
    private static PrintWriter pw;

    private static int n;
    private static List<Integer>[] adj;

    public static void main(String[] args) throws IOException {
        br = new BufferedReader(new InputStreamReader(System.in));
        pw = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
        st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        adj = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            adj[a].add(b);
            adj[b].add(a);
        }
        int[] parent = new int[n + 1];
        int[] mark = new int[n + 1];
        int[] leaf = new int[n + 1];
        int[] leafParent = new int[n + 1];
        Arrays.fill(parent, NOT_VISITED);
        Arrays.fill(mark, NOT_VISITED);
        Arrays.fill(leaf, NOT_VISITED);
        Arrays.fill(leafParent, NOT_VISITED);
        dfs(1, 0, parent, mark, leaf, leafParent);
        int leafCount = 0;
        for (int i = 1; i <= n; i++) {
            if (leaf[i] == NOT_VISITED) {
                continue;
            }
            leafCount++;
            if (leafCount >= 5) {
                pw.println("NO");
                pw.close();
                return;
            }
        }
        pw.println("YES");
        pw.close();
    }

    private static void dfs(int cur, int p, int[] parent, int[] mark, int[] leaf, int[] leafParent) {
        parent[cur] = p;
        mark[cur] = 0;
        if (adj[cur].size() == 1) {
            leaf[cur] = 0;
            leafParent[cur] = p;
            return;
        }
        for (int next : adj[cur]) {
            if (next == p) {
                continue;
            }
            dfs(next, cur, parent, mark, leaf, leafParent);
        }
        int childLeaf = 0;
        int childLeafParent = NOT_VISITED;
        for (int next : adj[cur]) {
            if (next == p) {
                continue;
            }
            if (leaf[next] != NOT_VISITED) {
                childLeaf++;
                childLeafParent = leafParent[next];
            }
        }
        if (childLeaf == 1) {
            leaf[cur] = childLeaf;
            leafParent[cur] = childLeafParent;
        }
    }
}
