#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  while (cin >> n >> k) {
    map<char, int> doors;
    string ent;
    cin >> ent;
    int maxEnt = 0;
    for (int i = 0; i < n; i++) {
      char d = ent[i];
      doors[d]++;
      maxEnt = max(maxEnt, doors[d]);
      if (maxEnt > k) {
        cout << "YES" << endl;
        return 0;
      }
    }
    cout << "NO" << endl;
  }
  return 0;
}
