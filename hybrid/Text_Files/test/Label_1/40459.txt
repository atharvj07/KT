for _ in range(int(input())):
    n, m = map(int, input().split())
    s = input()
    e = (0, 0)
    for i in s:
        if i == 'R': e = ((e[0] + 1) % n, e[1])
        if i == 'L': e = ((e[0] - 1) % n, e[1])
        if i == 'U': e = (e[0], (e[1] - 1) % m)
        if i == 'D': e = (e[0], (e[1] + 1) % m)
    print(e[0] + 1, e[1] + 1)
