import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        int[] prev = new int[n];
        dp[n - 1] = p[n - 1];
        prev[n - 1] = n - 1;
        for (int i = n - 2; i >= 0; i--) {
            for (int j = i + 1; j <= i + k && j < n; j++) {
                if (dp[j] + p[i] < dp[i]) {
                    dp[i] = dp[j] + p[i];
                    prev[i] = j;
                }
            }
        }
        System.out.println(prev[0] + 1);
    }
}
