import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    while (true) {
      long n = in.nextLong();
      int k = in.nextInt();
      if (n == 0 && k == 0) {
        break;
      }
      StringBuilder sb = new StringBuilder();
      long start = 0;
      while (n > 0) {
        start *= 10;
        start += n % 10;
        n /= 10;
      }
      for (int i = 1; i <= k; i++) {
        sb.append(start % 10);
        start++;
      }
      System.out.println(sb);
    }
  }
}
