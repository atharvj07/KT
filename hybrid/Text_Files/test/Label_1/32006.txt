import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong(); in.nextLine();
        long[] a = new long[(int)n];
        for (int i=0; i<n; ++i) {
            a[i] = Integer.parseInt(in.next());
        }
        long positives = 0;
        for (long x: a) {
            if (x > 0) {
                positives += x;
            }
        }
        long negsum = 0;
        long ans = 0;
        for (int i=0; i<n; ++i) {
            if (a[i] < 0) {
                if (positives > 0) {
                    long tt = Math.min(positives, - a[i]);
                    positives -= tt;
                    negsum -= tt;
                    ans += tt;
                } else {
                    negsum += a[i];
                }
            }
        }
        ans += (-negsum + 1) / 2;
        System.out.println(ans);
    }
}
