#include <bits/stdc++.h>
using namespace std;
#define endl "\n"

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    map<char, int> m;
    for (char c : s) m[c]++;
    bool a = true, b = true, c = true;
    if (m['A'] % 2 == 0) {
      if (m['B'] % 2 == 0 && m['C'] % 2 == 0) {
        a = false;
      } else if (m['B'] > 0 && m['C'] > 0) {
        a = false;
      }
    }
    if (m['B'] % 2 == 0) {
      if (m['A'] % 2 == 0 && m['C'] % 2 == 0) {
        b = false;
      } else if (m['A'] > 0 && m['C'] > 0) {
        b = false;
      }
    }
    if (m['C'] % 2 == 0) {
      if (m['A'] % 2 == 0 && m['B'] % 2 == 0) {
        c = false;
      } else if (m['A'] > 0 && m['B'] > 0) {
        c = false;
      }
    }
    if (a || b || c) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
}
