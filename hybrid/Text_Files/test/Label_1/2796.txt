int main()
{
    fast;
    int n, m;
    cin >> n >> m;
    vi p(n);
    for (int &x : p) cin >> x;

    ll ans = 0;
    for (int i = 0; i < n; i++) {
        if (p[i] == m) ans++;
        for (int j = i + 1; j < n; j++) {
            if (p[i] == m && p[j] > m) ans++;
            if (p[i] > m && p[j] == m) ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
