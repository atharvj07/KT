#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long

signed main()
{
    string s;
    cin >> s;
    int c = 0;
    vector<int> v;
    for (auto x : s)
    {
        if (x == '(')
            c++;
        else if (x == ')')
            c--;
        else
            v.push_back(c);
    }
    if (c != 0)
        cout << -1 << endl;
    else
    {
        for (int x : v)
            cout << -x << endl;
    }
}
