import sys

# Read the input.
n = int(sys.stdin.readline())
reserved_tokens = set(sys.stdin.readline().split())
m = int(sys.stdin.readline())
source_lines = []
for i in range(m):
    source_lines.append(sys.stdin.readline())
# Tokenize the input.
tokens = []
for line in source_lines:
    i = 0
    while i < len(line):
        if line[i] == '#':
            i = line.find('\n', i)
            if i == -1:
                i = len(line)
        elif line[i].isspace():
            i += 1
        else:
            j = i + 1
            while j < len(line) and not line[j].isspace() and line[j] != '#':
                j += 1
            token = line[i:j]
            if token not in reserved_tokens:
                tokens.append(token)
            i = j
# Rename the words.
words = [word for word in tokens if word.islower()]
words.sort(key=lambda word: (len(word), word))
rename_map = {word: chr(97 + i) for i, word in enumerate(words)}
# Minify the source.
minified_source = []
for token in tokens:
    if token in rename_map:
        minified_source.append(rename_map[token])
    else:
        minified_source.append(token)
minified_source = ' '.join(minified_source)
# Print the output.
print(minified_source)
