n = int(input())
a = list(map(int, input().split()))
if n == 1:
    print("YES")
elif n == 2:
    print("YES") if a[0] <= a[1] else print("NO")
else:
    inc, dec = False, False
    for i in range(n - 1):
        if a[i] < a[i + 1]:
            inc = True
            dec = False
        elif a[i] > a[i + 1]:
            inc = False
            dec = True
    if inc and dec:
        print("YES")
    else:
        print("NO")
