import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		int t = s.nextInt();
		while (t > 0) {
			t--;
			int a = s.nextInt();
			int b = s.nextInt();
			int q = s.nextInt();
			if (a > b) {
				int temp = a;
				a = b;
				b = temp;
			}
			int gcd = a;
			while (b > 0) {
				int temp = b;
				b = a % b;
				a = temp;
			}
			gcd = gcd / a;
			while (q > 0) {
				q--;
				long l = s.nextLong();
				long r = s.nextLong();
				if (l % gcd == 0 && (r - l + 1) % 2 == 1) {
					System.out.println((r - l + 1) / 2);
				} else if ((l + (gcd - 1)) % gcd == 0 && (r - (l + (gcd - 1)) + 1) % 2 == 1) {
					System.out.println((r - (l + (gcd - 1)) + 1) / 2);
				} else {
					System.out.println(0);
				}
			}
		}
	}
}
