import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), c = in.nextInt();
        int[] a = new int[n - 1], b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            b[i] = in.nextInt();
        }
        long[] t = new long[n];
        long[] dt = new long[n];
        t[0] = 0;
        t[1] = c + b[0];
        for (int i = 2; i < n; i++) {
            dt[i] = Math.min(dt[i - 1] + a[i - 1], dt[i - 2] + c + b[i - 1]);
            t[i] = Math.min(t[i - 1] + a[i - 1], t[i - 2] + c + b[i - 1]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(t[i] + " ");
        }
        System.out.println();
    }
}
