A, B, H, W, N = map(int, input().split())
A_ = list(map(int, input().split()))
if A > H and B > W:
    print(-1)
elif A > H:
    H = H*max(A_, key=lambda x:(H*x<=A))
    if H >= A: print(1)
    else: print(-1)
elif B > W:
    W = W*max(A_, key=lambda x:(W*x<=B))
    if W >= B: print(1)
    else: print(-1)
else:
    print(0)
