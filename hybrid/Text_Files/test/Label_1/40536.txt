n = int(input())
sticks = list(map(int, input().split()))

sticks_count = {}
for stick in sticks:
    if stick in sticks_count:
        sticks_count[stick] += 1
    else:
        sticks_count[stick] = 1

pairs = []
for length, count in sticks_count.items():
    if count >= 2:
        pairs.extend([length] * (count // 2))

pairs.sort()

area = 0
while len(pairs) > 1:
    second = pairs.pop()
    first = pairs.pop()
    if second - first <= 1:
        area += first * second
    else:
        pairs.append(first)

print(area)
