#include <iostream>
#include <vector>
using namespace std;

long long solve(long long x, long long y, long long n) {
    return (y + n/x *x);
}
int main() {
    long long t;
    cin >> t;
    while (t--) {
        long long x, y, n;
        cin >> x >> y >> n;
        cout << solve(x, y, n) << endl;
    }
    return 0;
}
