#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n], b[n], x, y, sum1 = 0, sum2 = 0;
  for (int i = 0; i < n; i++) {
    cin >> x >> y;
    a[i] = x, b[i] = y;
    sum1 += x;
    sum2 += y;
  }
  if (sum1 % 2 != sum2 % 2) {
    cout << -1 << endl;
    return 0;
  }
  int ans = INT_MAX;
  int cnt1 = 0, cnt2 = 0;
  for (int i = 0; i < n; i++) {
    if ((a[i] % 2) != (sum1 % 2)) cnt1++;
    if ((b[i] % 2) != (sum2 % 2)) cnt2++;
  }
  if (cnt1 == n || cnt2 == n) {
    cout << -1;
    return 0;
  }
  ans = min(ans, cnt1);
  ans = min(ans, cnt2);
  if (sum1 % 2 == 0 && sum2 % 2 == 0) {
    int cntx = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] % 2 == 0) cntx++;
    }
    ans = min(ans, cntx);
  }
  cout << ans << endl;
  return 0;
}
