#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<double> p(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
    }

    double ans = 0;
    for (int i = 0; i < n;) {
        double cur = p[i];
        int cnt = 0;
        while (i < n && cur > 0) {
            cur *= p[i] / (p[i] + 1 - p[i]);
            cnt++;
            i++;
        }
        ans += cnt * cnt;
    }

    cout.precision(15);
    cout.setf(ios::fixed);
    cout << ans << endl;

    return 0;
}
