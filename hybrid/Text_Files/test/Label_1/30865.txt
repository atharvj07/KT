import java.util.Scanner;
import java.util.Arrays;

public class Main {

    private static int longestSegment(int[] colors, int n, int k) {
        int[] lastOccurrence = new int[k + 1];
        Arrays.fill(lastOccurrence, -1);

        int maxLength = 0;
        int start = 0;

        for (int end = 0; end < n; end++) {
            if (lastOccurrence[colors[end]] >= start) {
                start = lastOccurrence[colors[end]] + 1;
            }
            lastOccurrence[colors[end]] = end;
            maxLength = Math.max(maxLength, end - start + 1);
        }

        return maxLength;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] colors = new int[n];
        for (int i = 0; i < n; i++) {
            colors[i] = scanner.nextInt();
        }
        System.out.println(longestSegment(colors, n, k));
    }
}
