#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string a, b;
        cin >> a >> b;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i] - b[i];
        }
        if (sum == 0)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
}
