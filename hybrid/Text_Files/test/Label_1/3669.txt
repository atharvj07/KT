#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;cin>>t;
    while(t--){
        string s;cin>>s;
        int n=s.size(),ans=0;
        for(int i=0;i<n;i++){
            int cur=0,x=1;
            for(int j=i;j<n;j++){
                cur=cur*2+s[j]-'0';
                if(j-i+1==cur)ans++;
            }
        }
        cout<<ans<<endl;
    }
}
