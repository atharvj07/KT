#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
int main()
{
    ll a[120005], n, cnt=0, mx=0, ex;
    cin>>n;
    for(ll i=1; i<=n; i++)
    {
        cin>>a[i];
        mx=max(mx, a[i]);
    }
    for(ll i=2; i<=1000000000000000000; i*=2){
        ex=0;
        for(ll j=1; j<=n; j++){
            if(a[j]==i){ex=1;break;}
        }
        if(ex==0){
            for(ll j=1; j<=n; j++){
                if((i-a[j])>0 && (i-a[j])<=mx){cnt++;break;}
            }
        }
    }
    cout<<cnt<<endl;
    return 0;
}
