import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] grid = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                grid[i][j] = in.nextInt();
            }
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int rowSum = 0;
                int colSum = 0;
                for (int k = 0; k < n; k++) {
                    rowSum += grid[i][k];
                    colSum += grid[k][j];
                }
                if (colSum > rowSum) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
