#include <iostream>
#include <string>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a, b;
    cin >> a >> b;
    int ans = 0;
    int i = 0, j = 0;
    while (i < a.size() && j < b.size()) {
      if (a[i] == b[j]) {
        i++;
        j++;
        continue;
      }
      if (a[i] > b[j]) {
        i++;
        ans++;
        continue;
      }
      if (a[i] < b[j]) {
        j++;
        ans++;
        continue;
      }
    }
    while (i < a.size()) {
      i++;
      ans++;
    }
    while (j < b.size()) {
      j++;
      ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
