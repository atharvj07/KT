import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        int maxSum = 0;
        int maxSumStart = -1;
        int maxSumEnd = -1;
        for (int i = 0; i < n; i++) {
            int sum = 0;
            for (int j = i; j < n; j++) {
                sum += a[j];
                if (sum > maxSum || (sum == maxSum && j - i > maxSumEnd - maxSumStart)) {
                    maxSum = sum;
                    maxSumStart = i;
                    maxSumEnd = j;
                }
            }
        }

        int cutCount = 0;
        for (int i = 0; i < maxSumStart; i++) {
            if (a[i] != a[maxSumStart]) {
                cutCount++;
            }
        }
        for (int i = maxSumEnd + 1; i < n; i++) {
            if (a[i] != a[maxSumStart]) {
                cutCount++;
            }
        }

        System.out.println(maxSum + " " + cutCount);
        for (int i = 0; i < maxSumStart; i++) {
            if (a[i] != a[maxSumStart]) {
                System.out.print(i + 1 + " ");
            }
        }
        for (int i = maxSumEnd + 1; i < n; i++) {
            if (a[i] != a[maxSumStart]) {
                System.out.print(i + 1 + " ");
            }
        }
        System.out.println();
    }
}
