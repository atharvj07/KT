from collections import defaultdict

n = int(input())
g = defaultdict(list)
for _ in range(n-1):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

vis = [False] * (n+1)
ans = []
def dfs(u):
    vis[u] = True
    ans.append(u)
    for v in g[u]:
        if not vis[v]:
            dfs(v)

dfs(1)
print(*ans)
