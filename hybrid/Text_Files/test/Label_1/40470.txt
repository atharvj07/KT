from bisect import bisect_left

def min_time(n, m, c_l, c_e, v, stairs, elevators, queries):
    results = []
    for x1, y1, x2, y2 in queries:
        # Calculate the horizontal distance
        horizontal_dist = abs(y1 - y2)
        # Calculate the vertical distance
        vertical_dist = abs(x1 - x2)
        
        if x1 == x2:  # If on the same floor
            results.append(horizontal_dist)
            continue
        
        min_time = abs(x1 - x2) + horizontal_dist  # Assume walking, no elevator or stairs
        
        # Check stairs
        for stair in stairs:
            time_with_stair = abs(y1 - stair) + abs(x1 - x2) + abs(stair - y2)
            if time_with_stair < min_time:
                min_time = time_with_stair

        # Check elevators
        for elevator in elevators:
            floors_with_elevator = (vertical_dist + v - 1) // v  # Ceiling division
            time_with_elevator = abs(y1 - elevator) + floors_with_elevator + abs(elevator - y2)
            if time_with_elevator < min_time:
                min_time = time_with_elevator
        
        results.append(min_time)
    return results

n, m, c_l, c_e, v = map(int, input().split())
stairs = list(map(int, input().split())) if c_l > 0 else []
elevators = list(map(int, input().split())) if c_e > 0 else []
q = int(input())
queries = [list(map(int, input().split())) for _ in range(q)]

for result in min_time(n, m, c_l, c_e, v, stairs, elevators, queries):
    print(result)
