import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;
import java.util.Queue;

class Main {
  static int n, q, lvl[], parent[], subTreeSize[], manager[];
  static LinkedList<Integer>[] adj;
  static Queue<Integer> q1;

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    String s[] = in.readLine().split(" ");
    n = Integer.parseInt(s[0]);
    q = Integer.parseInt(s[1]);
    s = in.readLine().split(" ");
    lvl = new int[n + 1];
    parent = new int[n + 1];
    subTreeSize = new int[n + 1];
    manager = new int[n + 1];
    adj = new LinkedList[n + 1];
    for (int i = 0; i <= n; i++)
      adj[i] = new LinkedList<Integer>();
    for (int i = 2; i <= n; i++) {
      int p = Integer.parseInt(s[i - 2]);
      parent[i] = p;
      adj[p].add(i);
    }
    lvl[1] = 0;
    q1 = new LinkedList<Integer>();
    q1.add(1);
    bfs1();
    dfs1(1, 0);
    for (int i = 1; i <= n; i++) {
      if (adj[i].size() == 0)
        continue;
      int mxL = 0, node = 0;
      for (int j : adj[i]) {
        if (mxL < lvl[j]) {
          mxL = lvl[j];
          node = j;
        }
      }
      manager[i] = node;
    }
    for (int i = 0; i < q; i++) {
      s = in.readLine().split(" ");
      int x = Integer.parseInt(s[0]), y = Integer.parseInt(s[1]);
      int ansNode = 0, maxLvl = 0;
      for (int j = x; j <= y; j++) {
        int temp = manager[j];
        if (x <= temp && temp <= y) {
          if (maxLvl < lvl[temp]) {
            maxLvl = lvl[temp];
            ansNode = temp;
          }
        }
      }
      int rem = 0;
      for (int j = x; j <= y; j++) {
        if (manager[j] == ansNode) {
          rem = j;
          break;
        }
      }
      System.out.println(rem + " " + maxLvl);
    }
  }

  static void bfs1() {
    while (!q1.isEmpty()) {
      int p = q1.poll();
      for (int i : adj[p]) {
        if (lvl[i] == 0) {
          lvl[i] = lvl[p] + 1;
          q1.add(i);
        }
      }
    }
  }

  static void dfs1(int u, int l) {
    subTreeSize[u] = 1;
    for (int v : adj[u]) {
      lvl[v] = lvl[u] + 1;
      dfs1(v, u);
      subTreeSize[u] += subTreeSize[v];
    }
  }
}
