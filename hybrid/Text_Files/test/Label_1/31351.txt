import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long h = sc.nextLong();
        long m = sc.nextLong();
        int k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong() * m + sc.nextLong();
        }
        Arrays.sort(a);
        long ans = k * h;
        for (int i = 0; i < n; i++) {
            if (a[i] <= ans) ans++;
            else if (a[i] > ans && a[i] - ans >= k) ans = a[i] - k + 1;
            else {
                System.out.println(i + 1);
                for (int j = 0; j < i; j++) {
                    System.out.print((j + 1) + " ");
                }
                System.exit(0);
            }
        }
        System.out.println("0 0");
    }
}
