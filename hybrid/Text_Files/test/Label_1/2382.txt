#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 1;

int n, m;
long long a[MAXN], b[MAXN];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; i++) {
        cin >> b[i];
    }
    while (m--) {
        int type;
        cin >> type;
        if (type == 1) {
            int x, y, k;
            cin >> x >> y >> k;
            for (int i = 0; i < k; i++) {
                b[y + i] = a[x + i];
            }
        } else if (type == 2) {
            int x;
            cin >> x;
            cout << b[x] << "\n";
        }
    }
    return 0;
}
