import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[][] points = new int[n][5];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 5; j++) {
        points[i][j] = scanner.nextInt();
      }
    }
    int count = 0;
    for (int i = 0; i < n; i++) {
      boolean found = false;
      for (int j = 0; j < n; j++) {
        if (i == j) continue;
        for (int k = j + 1; k < n; k++) {
          if (i == k) continue;
          int[] vec1 = new int[5];
          int[] vec2 = new int[5];
          for (int l = 0; l < 5; l++) {
            vec1[l] = points[j][l] - points[i][l];
            vec2[l] = points[k][l] - points[i][l];
          }
          int dotProduct = 0;
          int len1 = 0;
          int len2 = 0;
          for (int l = 0; l < 5; l++) {
            dotProduct += vec1[l] * vec2[l];
            len1 += vec1[l] * vec1[l];
            len2 += vec2[l] * vec2[l];
          }
          double cosine = dotProduct / (Math.sqrt(len1) * Math.sqrt(len2));
          if (cosine > 0) {
            found = true;
            break;
          }
        }
      }
      if (!found) {
        count++;
        System.out.println(i + 1);
      }
    }
    System.out.println(count);
  }
}
