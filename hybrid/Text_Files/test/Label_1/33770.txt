import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] pebbles = new int[n];
        for (int i = 0; i < n; i++) {
            pebbles[i] = in.nextInt();
        }
        in.close();
        Arrays.sort(pebbles);
        int total = 0;
        int days = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (total + pebbles[i] <= k) {
                total += pebbles[i];
            } else {
                total = pebbles[i];
                days++;
            }
        }
        System.out.println(days + 1);
    }
}
