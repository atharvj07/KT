#include <bits/stdc++.h>
using namespace std;
int n, a, b;
string s;
int main() {
  cin >> n >> a >> b >> s;
  int ans = 1e9;
  if (s[a - 1] == s[b - 1]) ans = 0;
  if (a > b) swap(a, b);
  for (int i = a - 1; i < b; i++) {
    if (s[i] != s[i + 1]) {
      ans = 0;
      break;
    }
  }
  if (ans) {
    for (int i = a - 1; i < b; i++) {
      if (s[i] != s[i + 1]) ans++;
    }
  }
  cout << ans << endl;
}
