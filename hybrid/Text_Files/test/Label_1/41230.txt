n = int(input())
a = list(map(int, input().split()))
m = int(input())
queries = [tuple(map(int, input().split())) for _ in range(m)]

# Preprocessing
indices = sorted(range(n), key=lambda i: (-a[i], i))
prefix_sums = [0] * (n + 1)
for i in range(n):
    prefix_sums[i + 1] = prefix_sums[i] + a[indices[i]]

# Answer queries
for k, pos in queries:
    optimal_indices = sorted(indices[:k], key=lambda i: i)
    print(a[optimal_indices[pos - 1]])
