#include <bits/stdc++.h>
using namespace std;
#define pb push_back
typedef long long ll;
const int mod=1e9+7;
vector<int>a,v;
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n,q,x;
    cin>>n>>q;
    a.resize(n);
    for(int i=0;i<n;i++){
        cin>>x;
        a[i]=x;
    }
    v.pb(0);
    for(int i=0;i<n;i++){
        if(a[i])
            v.pb(i+1);
    }
    for(int i=0;i<v.size();i++){
        v[i]+=v[i-1];
    }
    while(q--){
        int l,r;
        cin>>l>>r;
        l--;r--;
        int ans=0;
        if(l>0)ans+=v[l-1];
        ans+=v[r]-v[l];
        cout<<ans<<endl;
    }
    return 0;
}
