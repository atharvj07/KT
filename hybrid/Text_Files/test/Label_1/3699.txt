#include <cstdio>
#include <algorithm>
#include <vector>
const int siz = 1e5 + 100;
int a[siz];
int main() {
    int t;
    scanf("%d", &t);
    while(t--) {
        int n;
        scanf("%d", &n);
        for(int i = 1; i <= n; ++i) {
            scanf("%d", &a[i]);
            a[i + n] = a[i];
        }
        int len = 0, Max = 0;
        for(int i = 1; i <= 2 * n; ++i)
            a[i] > len ? ++len : len = 1, Max = std :: max(len, Max);
        printf("%d\n", Max);
    }
}
