import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int tests = sc.nextInt();
        for (int t = 0; t < tests; t++) {
            int d = sc.nextInt();
            int k = sc.nextInt();
            if (d <= k || d % 2 != k % 2) {
                System.out.println("Utkarsh");
            } else {
                System.out.println("Ashish");
            }
        }
    }
}
