s=input()
s='('+s+')'
n=len(s)
l=[0]*n
r=[0]*n
st=[[0]*n for _ in range(n)]
ans=0
for i in range(1,n-1):
    if s[i]=='?':
        l[i]=l[i-1]+1
    else:
        l[i]=l[i-1]
for i in range(n-2,-1,-1):
    if s[i]=='?':
        r[i]=r[i+1]+1
    else:
        r[i]=r[i+1]
for i in range(1,n-1):
    for j in range(i+1,n-1):
        if s[i]==s[j]=='?' or s[i]!=s[j] and s[i]!='?' and s[j]!='?':
            st[i][j]=st[i][j-1]+st[i+1][j]-st[i+1][j-1]
            if st[i][j]>0:
                ans+=l[i-1]*(r[j+1]-r[j])+(l[j]-l[i])*r[j+1]
print(ans)
