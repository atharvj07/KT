for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    odd, even = 0, 0
    for i in range(n):
        if i % 2 == 0:
            even += a[i]
        else:
            odd += a[i]
    need = (even - odd) // 2
    ans = []
    k = 0
    for i in range(n):
        if i % 2 == 0:
            if k == need:
                break
            if a[i] == 1:
                ans.append(0)
                a[i] = 0
                k += 1
        else:
            if k == need:
                break
            if a[i] == 0:
                ans.append(1)
                a[i] = 1
                k += 1
    for i in range(n):
        if a[i] == 1:
            ans.append(1)
    print(len(ans))
    print(' '.join(map(str, ans)))
