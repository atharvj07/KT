import java.util.*;

class Main {
    static final int MX = 200000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt(), q = sc.nextInt();
        int[] p = new int[MX], cnt = new int[MX];
        for (int i = 0; i < q; i++) {
            if (sc.nextInt() == 1) {
                int di = sc.nextInt() - 1, ai = sc.nextInt();
                for (int j = Math.max(0, di - k - 1); j <= Math.min(n - 1, di); j++) {
                    p[j] += ai;
                }
            } else {
                int pi = sc.nextInt() - 1;
                int sum = 0;
                for (int j = Math.max(0, pi); j <= Math.min(n - 1, pi + k - 1); j++) {
                    sum += Math.min(b, cnt[j] + p[j]);
                }
                for (int j = Math.max(0, pi + k); j <= Math.min(n - 1, pi + k + k - 1); j++) {
                    sum += Math.min(a, cnt[j] + p[j]);
                }
                cnt = Arrays.copyOf(p, n);
                p = new int[MX];
                System.out.println(sum);
            }
        }
        sc.close();
    }
}
