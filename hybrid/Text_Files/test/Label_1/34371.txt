import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<List<Edge>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            g.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            int c = in.nextInt();
            g.get(u).add(new Edge(v, c));
            g.get(v).add(new Edge(u, c));
        }
        int[] maxDepth = new int[n];
        boolean[] visited = new boolean[n];
        dfs(0, -1, g, maxDepth, visited, 0);
        int max = 0;
        for (int i = 0; i < n; i++) {
            max = Math.max(max, maxDepth[i]);
        }
        System.out.println(max);
    }

    private static void dfs(int u, int parent, List<List<Edge>> g, int[] maxDepth, boolean[] visited, int depth) {
        visited[u] = true;
        maxDepth[u] = depth;
        for (Edge e : g.get(u)) {
            if (e.to == parent) {
                continue;
            }
            if (!visited[e.to]) {
                dfs(e.to, u, g, maxDepth, visited, depth + e.cost);
            }
        }
    }

    private static class Edge {

        int to;
        int cost;

        public Edge(int to, int cost) {
            this.to = to;
            this.cost = cost;
        }
    }
}
