import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    segs = [tuple(map(int,input().split())) for _ in range(n)]
    segs.sort()
    l,r = -1e9, -1e9
    ans = 0
    for x,y in segs:
        if x<=l and y>=r:
            ans += 1
        l = x
        r = y
    print(ans)
