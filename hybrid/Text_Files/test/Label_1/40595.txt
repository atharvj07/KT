import sys
def dfs(g, v, p):
    dp = [0, 0]
    for u in g[v]:
        if u != p:
            x = dfs(g, u, v)
            dp[0] += x[0]
            dp[1] += x[1]
    dp[1] += 1
    ans[v] = dp
    return dp

for _ in range(int(input())):
    n, m, a, b = map(int, input().split())
    g = [[] for i in range(n+1)]
    for _ in range(m):
        x, y = map(int, input().split())
        g[x].append(y)
        g[y].append(x)
    ans = [0 for i in range(n+1)]
    dfs(g, a, -1)
    dfs(g, b, -1)
    res = 0
    for i in range(1, n+1):
        if i != a and i != b:
          res += ans[i][0] * ans[i][1]
    print(res)
