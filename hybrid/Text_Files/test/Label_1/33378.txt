import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arr = new int[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
            if (arr[i] > 0)
                sum += arr[i];
        }
        long max = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] > 0)
                max = Math.max(max, sum + arr[i]);
            else
                sum -= arr[i];
        }
        System.out.println(max);
        scanner.close();
    }
}
