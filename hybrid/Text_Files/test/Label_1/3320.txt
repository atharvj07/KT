#include<bits/stdc++.h>
using namespace std;
int n,a,b,k,f;
vector<string>v[305];
map<pair<string,string>,int>mp;
int vis[305],vis2[305],cost[305];
vector<int>ed[305];
int bfs(int s){
    for(int i=1;i<=n;i++){
        vis[i]=0;
        cost[i]=1e9;
    }
    queue<int>q;
    q.push(s);
    vis[s]=1;
    cost[s]=0;
    while(!q.empty()){
        int u=q.front();
        q.pop();
        for(int i=0;i<(int)ed[u].size();i++){
            int x=ed[u][i];
            if(vis[x]==0){
                vis[x]=1;
                q.push(x);
                cost[x]=cost[u]+1;
            }
        }
    }
    int mx=0;
    for(int i=1;i<=n;i++){
        if(vis[i]){
            if(cost[i]>mx) mx=cost[i];
        }
    }
    return mx;
}
int main(){
    cin>>n>>a>>b>>k>>f;
    int cnt=0;
    for(int i=1;i<=n;i++){
        string s,t;
        cin>>s>>t;
        v[i].push_back(s);
        v[i].push_back(t);
        if(i>1){
            if(s==v[i-1][1]) cnt+=b;
            else cnt+=a;
        }
        mp[make_pair(s,t)]=i;
    }
    for(int i=1;i<=n;i++){
        int x,y;
        string s=v[i][0];
        string t=v[i][1];
        for(int j=i+1;j<=n;j++){
            string ss=v[j][0];
            string tt=v[j][1];
            if(t==ss){
                x=i;
                y=j;
                ed[x].push_back(y);
                ed[y].push_back(x);
            }
            if(s==tt){
                x=i;
                y=j;
                ed[x].push_back(y);
                ed[y].push_back(x);
            }
        }
    }
    int ans=0;
    for(int i=1;i<=n;i++){
        if(vis2[i]==0){
            ans+=f;
            int x=bfs(i);
            vis2[i]=1;
            for(int j=1;j<=k;j++){
                if((int)ed[i].size()>=j){
                    int mx=0;
                    for(int q=0;q<(int)ed[i].size();q++){
                        if(cost[ed[i][q]]>mx){
                            mx=cost[ed[i][q]];
                            x=ed[i][q];
                        }
                    }
                }
            }
            for(int j=1;j<=n;j++){
                if(j!=i){
                    if(vis[j]==1){
                        x=j;
                        break;
                    }
                }
            }
            x=bfs(x);
            ans+=x*b;
        }
    }
    cout<<ans<<endl;
}
