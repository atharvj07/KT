import java.util.Scanner;

public class GoodSegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;
            System.out.println(countGoodSegments(permutation, l, r));
        }
    }

    static int countGoodSegments(int[] permutation, int l, int r) {
        int minIndex = l, maxIndex = r;
        for (int i = l + 1; i <= r; i++) {
            if (permutation[i] < permutation[minIndex]) {
                minIndex = i;
            }
            if (permutation[i] > permutation[maxIndex]) {
                maxIndex = i;
            }
        }
        int goodSegments = 0;
        for (int i = l; i <= maxIndex; i++) {
            for (int j = i; j <= r; j++) {
                if (permutation[i] <= permutation[j]) {
                    goodSegments++;
                }
            }
        }
        return goodSegments;
    }
}
