import java.util.*;

class Main {

    static class Node {
        int value;
        Node left;
        Node right;

        public Node(int value) {
            this.value = value;
        }
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        Node root = new Node(a[0]);
        for (int i = 1; i < n; i++) {
            Node node = root;
            while (true) {
                if (a[i] < node.value) {
                    if (node.left == null) {
                        node.left = new Node(a[i]);
                        break;
                    }
                    node = node.left;
                } else {
                    if (node.right == null) {
                        node.right = new Node(a[i]);
                        break;
                    }
                    node = node.right;
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 1; i < n; i++) {
            sb.append(findParent(root, a[i]).value).append(" ");
        }
        System.out.println(sb.toString());
    }

    static Node findParent(Node root, int value) {
        if (root == null) {
            return null;
        }
        if (root.left != null && root.left.value == value) {
            return root;
        }
        if (root.right != null && root.right.value == value) {
            return root;
        }
        if (value < root.value) {
            return findParent(root.left, value);
        } else {
            return findParent(root.right, value);
        }
    }
}
