#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];

  int ans = 0;
  int mn = p[0];
  for (int i = 1; i < n; i++) {
    ans += max(p[i] - mn, 0);
    mn = min(mn, p[i]);
  }
  cout << ans << endl;
  return 0;
}
