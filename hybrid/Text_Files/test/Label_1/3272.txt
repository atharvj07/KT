#include <iostream>
#include <vector>

using namespace std;

int main() {
    int sum, limit;
    cin >> sum >> limit;
    if (sum == 0) {
        cout << 1 << endl;
        cout << 1 << endl;
        return 0;
    }
    if (sum > limit || (sum & 1) ^ ((limit - sum) & 1)) {
        cout << -1 << endl;
        return 0;
    }
    vector<int> ans;
    int cnt = 0;
    for (int i = limit; i >= 1; i--) {
        if (sum & 1) {
            ans.push_back(i);
            cnt++;
            sum--;
        }
        if (cnt >= 2) break;
        if ((i - 1) & i) continue;
        ans.push_back(i);
        cnt++;
        sum -= 2;
    }
    for (int i = limit; i >= 1; i--) {
        if (sum <= 0) break;
        if (i & 1) continue;
        ans.push_back(i);
        sum -= 2;
    }
    if (sum != 0) {
        cout << -1 << endl;
        return 0;
    }
    cout << ans.size() << endl;
    for (auto x : ans) cout << x << " ";
    cout << endl;
    return 0;
}
