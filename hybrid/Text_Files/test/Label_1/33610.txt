import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] perm = new int[n + 1];
        for (int i = 0; i < n; i++) {
            perm[sc.nextInt()] = i + 1;
        }
        TreeSet<Integer> set = new TreeSet<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            set.add(Math.min(perm[a], perm[b]) * n + Math.max(perm[a], perm[b]));
        }
        for (int i = 1; i <= n; i++) {
            int pos = -1;
            for (int j : set) {
                if (j / n == i) {
                    pos = j % n;
                    break;
                }
            }
            if (pos != -1) {
                int cur = perm[i];
                perm[i] = perm[pos];
                perm[pos] = cur;
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(perm[i] + " ");
        }
    }
}
