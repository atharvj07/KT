#include <bits/stdc++.h>
using namespace std;

const int maxn = 54321;
vector<int> adj[maxn];
int vis[maxn], ans[maxn];

int dfs(int u) {
  int ret = 0;
  vis[u] = 1;
  for (int v : adj[u]) {
    if (!vis[v]) ret = max(ret, dfs(v));
  }
  return ans[u] = ret + 1;
}

int main() {
  int n;
  cin >> n;
  for (int i = 1; i < n; i++) { adj[i].push_back(i + 1); }
  adj[n].push_back(1);
  cout << dfs(1) << endl;
  return 0;
}
