#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int x, y;
    cin >> x >> y;
    string s;
    cin >> s;
    int l = 0, r = 0, u = 0, d = 0;
    for (int i = 0; i < s.length(); i++) {
      if (s[i] == 'L')
        l++;
      else if (s[i] == 'R')
        r++;
      else if (s[i] == 'U')
        u++;
      else
        d++;
    }
    if (abs(x - r + l) + abs(y - u + d) <= max({abs(x), abs(y), abs(x - r + l), abs(y - u + d)}))
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
