import java.util.*;
import java.io.*;

public class Main {
    static long mod = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long s = in.nextLong();
        long[] f = new long[n];
        for (int i = 0; i < n; i++) f[i] = in.nextLong();
        Arrays.sort(f);
        long ans = 1;
        for (int i = 0; i < n; i++) {
            if (s < f[i]) break;
            ans = (ans * (s - f[i] + 1)) % mod;
            s -= f[i];
        }
        System.out.print(ans);
    }
}
