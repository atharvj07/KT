#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans = 0;
  while (!a.empty()) {
    int l = *min_element(a.begin(), a.end());
    int r = *max_element(a.begin(), a.end());
    if (r - l + 1 <= a[r]) {
      a.erase(remove(a.begin(), a.end(), l), a.end());
      a.erase(remove(a.begin(), a.end(), r), a.end());
      ans++;
    } else {
      a.erase(remove(a.begin(), a.end(), l), a.end());
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
