import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] id = new int[n];
        for (int i = 0; i < n; i++) id[i] = sc.nextInt();
        int[] stack = new int[k];
        int top = -1;
        for (int i = 0; i < n; i++) {
            int cur = id[i];
            boolean found = false;
            for (int j = 0; j <= top; j++) {
                if (stack[j] == cur) {
                    found = true;
                    break;
                }
            }
            if (found) continue;
            if (top + 1 >= k) {
                for (int j = 0; j < top; j++) stack[j] = stack[j + 1];
            } else {
                top++;
            }
            stack[top] = cur;
        }
        System.out.println(Math.min(top + 1, k));
        for (int i = 0; i < Math.min(top + 1, k); i++) System.out.print(stack[i] + " ");
        sc.close();
    }
}
