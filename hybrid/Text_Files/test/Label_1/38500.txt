t = int(input())
for _ in range(t):
    s = input()
    ab, ba = 0, 0
    for i in range(len(s) - 1):
        if s[i:i+2] == "ab": ab += 1
        elif s[i:i+2] == "ba": ba += 1
    if ab == ba:
        print(s)
    elif ab > ba:
        for i in range(len(s) - 1):
            if s[i:i+2] == "ab":
                s = s[:i] + 'b' + s[i+1:]
                ab -= 1
                ba += 1
                break
        print(s)
    elif ba > ab:
        for i in range(len(s) - 1):
            if s[i:i+2] == "ba":
                s = s[:i] + 'a' + s[i+1:]
                ab += 1
                ba -= 1
                break
        print(s)
