import java.util.*;

class Main {
  static int n, m;
  static ArrayList<Integer>[] g;
  static boolean[] vis;
  static int ans;

  static void dfs(int u, int p, int roads) {
    vis[u] = true;
    if (roads == m - 2) {
      ans++;
      vis[u] = false;
      return;
    }
    for (int v : g[u]) {
      if (v == p) continue;
      dfs(v, u, roads + 1);
    }
    vis[u] = false;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    g = new ArrayList[n + 1];
    for (int i = 1; i <= n; i++) g[i] = new ArrayList<>();
    vis = new boolean[n + 1];
    ans = 0;
    for (int i = 1; i <= m; i++) {
      int u = sc.nextInt();
      int v = sc.nextInt();
      g[u].add(v);
      g[v].add(u);
    }
    for (int i = 1; i <= n; i++) {
      if (g[i].size() >= 2) dfs(i, i, 0);
    }
    System.out.println(ans);
  }
}
