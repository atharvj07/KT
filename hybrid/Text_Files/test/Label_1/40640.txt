n = int(input())
s = input()

# Initialize color list with zeros
color = ['0'] * n

# Flag to check if it's possible to sort by coloring and swapping
possible = True

for i in range(1, n):
    if s[i] < s[i-1]:
        color[i] = '1'
    elif s[i] == s[i-1] and color[i-1] == '1':
        color[i] = '1'
    # Check if there's a decrease after coloring the same letters
    if i > 1 and s[i] >= s[i-1] and color[i] < color[i-1] and s[i-1] < s[i-2] and color[i-1] == color[i-2]:
        possible = False
        break

if possible:
    print("YES")
    print(''.join(color))
else:
    print("NO")
