n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
for i in range(n):
    a[i] = bin(a[i])[2:]
    sz = len(a[i])
    while sz < 60:
        a[i] = '0' + a[i]
        sz += 1
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        flag = 1
        for k in range(60):
            if int(a[i][k]) != int(a[j][k]) and int(a[j][k]) == 1:
                flag = 0
        if flag == 0:
            if b[i] > b[j]:
                b[j] = 0
            else:
                b[i] = 0
print(max(b))
