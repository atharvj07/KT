import sys
n, m = map(int, sys.stdin.readline().split())
s = sys.stdin.readline()[:-1]
t = sys.stdin.readline()[:-1]
changes = 0
pos = []
for i in range(n):
    if s[i] != t[i]:
        pos.append(i + 1)
        changes += 1
        s = s[:i] + '?' + s[i + 1:]

print(changes)
for p in pos:
    print(p, end=' ')
