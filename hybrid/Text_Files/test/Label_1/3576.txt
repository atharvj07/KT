#include <bits/stdc++.h>
using namespace std;
long long a[100005], p[100005];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) p[i] = p[i - 1] + a[i];
  long long ans = (long long)1e18;
  for (int i = 1; i <= n; i++) {
    long long l = p[i - 1], r = p[n] - p[i];
    ans = min(ans, i * i - 2 * i * (l - r) + l * l + r * r);
  }
  cout << ans << endl;
}
