import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            long n = sc.nextLong();
            boolean flag = false;

            for (long a = 1; a * a * a <= n; a++) {
                long b = (long)Math.cbrt(n - a * a * a);
                if (a * a * a + b * b * b == n) {
                    flag = true;
                    break;
                }
            }

            System.out.println(flag ? "YES" : "NO");
        }
    }
}
