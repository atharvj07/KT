#include <iostream>
#include <vector>
#include <queue>

using namespace std;

struct Edge {
    int u, v, w;
};

int main() {
    int n, m;
    cin >> n >> m;

    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<Edge> edges(m);
    for (int i = 0; i < m; i++) {
        cin >> edges[i].u >> edges[i].v >> edges[i].w;
        edges[i].u--, edges[i].v--;
    }

    vector<long long> dist(n, 1e18);
    dist[0] = 0;

    priority_queue<pair<long long, int>> pq;
    pq.push({0, 0});

    while (!pq.empty()) {
        long long d = -pq.top().first;
        int u = pq.top().second;
        pq.pop();

        if (d > dist[u]) continue;

        for (Edge e : edges) {
            if (e.u == u) {
                long long nd = d + e.w - a[e.v];
                if (nd < dist[e.v]) {
                    dist[e.v] = nd;
                    pq.push({-nd, e.v});
                }
            }
            if (e.v == u) {
                long long nd = d + e.w - a[e.u];
                if (nd < dist[e.u]) {
                    dist[e.u] = nd;
                    pq.push({-nd, e.u});
                }
            }
        }
    }

    long long ans = 0;
    for (long long x : dist) {
        if (x < 1e18) ans += x;
    }

    cout << ans << endl;

    return 0;
}
