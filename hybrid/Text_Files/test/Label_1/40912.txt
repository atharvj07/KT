import sys

def solve():
    n, m = map(int, sys.stdin.readline().strip().split())
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        x, y = map(int, sys.stdin.readline().strip().split())
        adj[x].append(y)
    vis = [0] * (n+1)
    ans = []
    def dfs(u):
        vis[u] = 1
        for v in adj[u]:
            if vis[v] == 0:
                dfs(v)
                if len(ans) < 4*n//7:
                    ans.append(v)
    dfs(1)
    print(len(ans))
    print(*ans)

for _ in range(int(sys.stdin.readline())):
    solve()
