#include <bits/stdc++.h>
using namespace std;
int main(){
    int t,n;
    string s;
    cin>>t;
    for(int testcase=1;testcase<=t;++testcase){
        bool swapped=false;
        cin>>s;
        n=s.size();
        for(int i=n-2;i>=0;--i){
            if(s[i]!=s[i+1]){
                swapped=true;
                if(s[i+1]>'0')
                    swap(s[i],s[i+1]);
            }
        }
        if(!swapped)
            s.erase(s.find_first_not_of("0"),1);
        cout<<s<<"\n";
    }
}
