import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long n = input.nextLong(), p = input.nextLong(), w = input.nextLong(), d = input.nextLong();
        if (p > n * w) {
            System.out.println(-1);
        } else {
            long x = (p + d - w * n) / (w - d);
            if (x >= 0 && x * w + d * (n - x - x) == p) {
                System.out.println(x + " " + (n - x - x) + " " + x);
            } else
                System.out.println(-1);
        }
    }
}
