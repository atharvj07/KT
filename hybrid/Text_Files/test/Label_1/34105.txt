import java.util.*;
import java.lang.*;

class Main {
    public static void subtask(int n, int[] input) {
        long[] prefix = new long[n + 1];
        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + input[i];
        }
        for (int i = 0; i < n; i++) {
            long p = prefix[i] % 1000000007;
            input[i] = (int) ((prefix[n] - p) % 1000000007);
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(input[i]);
            if (i + 1 < n) {
                sb.append(" ");
            }
        }
        System.out.println(sb.toString());
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        subtask(n, arr);
    }
}
