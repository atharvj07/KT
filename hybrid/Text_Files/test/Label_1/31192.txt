import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int d = sc.nextInt();
        int e = sc.nextInt();
        int f = sc.nextInt();
        int suits1 = Math.min(a, d);
        int suits2 = Math.min(Math.min(b, c), d);
        System.out.println(suits1 * e + suits2 * f);
    }
}
