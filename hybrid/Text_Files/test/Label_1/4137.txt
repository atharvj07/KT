#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  int w[n + 1];
  for (int i = 1; i <= n; i++) cin >> w[i];
  int days = 0, p = 0, q = 0;
  for (int i = 1; i <= n; i++) {
    if (p + w[i] <= k)
      p += w[i];
    else if (q + w[i] <= k)
      q = w[i];
    else
      p = w[i], days++, q = 0;
  }
  if (p + q > 0) days++;
  cout << days << endl;
}
