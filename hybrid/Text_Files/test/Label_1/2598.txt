#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<ll> a(n);
  for (ll &x : a) { cin >> x; }
  ll ans = 0, prv = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] < a[i - 1]) {
      ans = i + 1;
      prv = (a[i] ^ a[i - 1]) + 1;
      break;
    }
  }
  bool ok = 0;
  for (int i = ans; i < n; i++) {
    if (a[i] < prv) {
      ok = 1;
      break;
    }
  }
  if (ok)
    cout << ans << endl;
  else
    cout << -1 << endl;
}
