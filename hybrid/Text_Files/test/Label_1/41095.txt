import sys

n, x = map(int, input().strip().split())
candies = [tuple(map(int, input().strip().split())) for _ in range(n)]

candies.sort(key=lambda v: v[1])

eaten = [False] * n
max_height = x
ans = 0

for t, h, m in candies:
  if (not eaten[t] and h <= max_height):
    max_height += m
    eaten[t] = True
    ans += 1

print(ans)
