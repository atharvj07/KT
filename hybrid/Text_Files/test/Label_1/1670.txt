#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;

ll gcd(ll a, ll b) { if(b == 0) return a; return gcd(b, a % b); }

int main() {
    ll n; cin >> n;
    vector<ll> a;
    ll non_zero = -1, sum = 0;
    for(ll i = 0, ai; i < n; i++) {
        cin >> ai;
        a.push_back(ai);
        sum += ai;
        if(ai) non_zero = i;
    }
    if(non_zero == -1) { cout << -1; return 0; }
    if(sum == 0) { cout << 0; return 0; }
    ll g = a[non_zero];
    for(ll i = 0; i < n; i++) if(a[i]) g = gcd(g, a[i]);
    cout << sum / g - 1;
    return 0;
}
