#include <bits/stdc++.h>
using namespace std;

const int maxn = 200001;

struct node {
    int p, c;
    bool operator<(const node& b) const { return p < b.p; }
};

int main() {
    int n, k;
    cin >> n >> k;
    vector<node> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i].p; }
    for (int i = 0; i < n; i++) { cin >> a[i].c; }
    sort(a.begin(), a.end());
    vector<long long> f(n);
    f[n - 1] = a[n - 1].c;
    for (int i = n - 2; i >= 0; i--) {
        f[i] = f[i + 1];
        if (k > 0) {
            f[i] = max(f[i], f[min(n - 1, i + k + 1)] + a[i].c);
            k--;
        }
    }
    for (int i = 0; i < n; i++) { cout << f[i] << ' '; }
    cout << endl;
}
