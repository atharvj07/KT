import java.util.*;

class Main {
    static class MemoryManager {
        private int[] memory;
        private int[] state;
        private int allocCnt;
        private int freeCnt;

        MemoryManager(int m) {
            memory = new int[m];
            state = new int[m];
            allocCnt = 0;
            freeCnt = 0;
        }

        int alloc(int n) {
            if (freeCnt + n <= memory.length) {
                int start = freeCnt;
                freeCnt += n;
                for (int i = start; i < start + n; i++) {
                    state[i] = 1;
                }
                allocCnt++;
                return allocCnt;
            } else {
                return -1;
            }
        }

        void erase(int x) {
            if (x <= 0 || x > allocCnt) {
                System.out.println("ILLEGAL_ERASE_ARGUMENT");
            } else {
                int start = 0;
                int end = 0;
                for (int i = 0; i < memory.length; i++) {
                    if (state[i] == 0) {
                        start = i;
                        end = i;
                        while (end + 1 < memory.length && state[end + 1] == 0) {
                            end++;
                        }
                        if (end - start + 1 >= x) {
                            for (int j = start; j < start + x; j++) {
                                state[j] = 0;
                            }
                            freeCnt = start + x;
                            break;
                        }
                    }
                }
            }
        }

        void defragment() {
            int start = 0;
            int end = 0;
            for (int i = 0; i < memory.length; i++) {
                if (state[i] == 0) {
                    start = i;
                    end = i;
                    while (end + 1 < memory.length && state[end + 1] == 0) {
                        end++;
                    }
                    if (end - start + 1 >= freeCnt) {
                        for (int j = start; j < start + freeCnt; j++) {
                            state[j] = 1;
                        }
                        for (int j = start + freeCnt; j <= end; j++) {
                            state[j] = 0;
                        }
                        break;
                    }
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        int m = scanner.nextInt();
        MemoryManager manager = new MemoryManager(m);
        for (int i = 0; i < t; i++) {
            String command = scanner.next();
            if (command.equals("alloc")) {
                int n = scanner.nextInt();
                int result = manager.alloc(n);
                System.out.println(result);
            } else if (command.equals("erase")) {
                int x = scanner.nextInt();
                manager.erase(x);
            } else if (command.equals("defragment")) {
                manager.defragment();
            }
        }
        scanner.close();
    }
}
