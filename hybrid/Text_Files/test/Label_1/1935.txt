#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+5;
int ans[maxn], a[maxn];
int32_t main()
{
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    for (int i = 1, j = 1, p = n; i <= n; i++)
    {
        if (a[i] > a[j])
        {
            j = i;
        }
        if (i - j > k)
        {
            ans[j] = 2;
            j++;
        }
        if (p - i > k)
        {
            ans[p] = 2;
            p--;
        }
        if (j > p)
            break;
        ans[i] = 1;
    }
    for (int i = 1; i <= n; i++)
        cout << ans[i];
}
