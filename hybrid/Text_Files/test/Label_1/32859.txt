import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.nextLine();
    int maxLen = 0;
    int cnt = 0;
    for (char ch : s.toCharArray()) {
      if (ch == 'A' || ch == 'C' || ch == 'G' || ch == 'T') cnt++;
      else cnt = 0;
      maxLen = Math.max(maxLen, cnt);
    }
    System.out.println(maxLen);
  }
}
