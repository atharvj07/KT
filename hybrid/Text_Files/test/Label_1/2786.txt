#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, w;
  cin >> n >> w;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  long long ans = 0;
  long long cur = w;
  for (int i = n - 1; i >= 0; i--) {
    cur -= a[i];
    if (cur < 0 || cur > w) {
      ans = 0;
      break;
    }
    if (i == 0) ans++;
  }
  cout << ans << endl;
  return 0;
}
