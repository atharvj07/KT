def min_operations(n, m, arr):
    def is_valid(mid):
        cur = 0
        for i in range(n):
            if arr[i] <= cur:
                cur = max(cur, (arr[i] + mid) % m)
            elif arr[i] + mid >= m + cur:
                continue
            else:
                return False
        return True

    l, r = 0, m
    while l < r:
        mid = (l + r) // 2
        if is_valid(mid):
            r = mid
        else:
            l = mid + 1
    return l

n, m = map(int, input().split())
arr = list(map(int, input().split()))
print(min_operations(n, m, arr))
