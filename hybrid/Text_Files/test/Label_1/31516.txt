import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] x = new long[n];
        long[] y = new long[m];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            y[i] = sc.nextInt();
        }
        int[] pos = new int[n];
        for (int i = 0; i < n; i++) {
            pos[i] = Arrays.binarySearch(y, x[i]);
            if (pos[i] < 0) {
                pos[i] = -(pos[i] + 1);
            }
        }
        long ans = 1;
        for (int i = 0; i < n; i++) {
            ans = (ans * pos[i]) % 1000000007;
        }
        System.out.println(ans);
    }
}
