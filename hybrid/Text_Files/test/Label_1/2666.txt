#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long int n;
    cin >> n;
    if (n % 2 == 0) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    vector<pair<long long int, long long int>> ans;
    long long int sum = 0;
    for (long long int i = n / 2; i > 1; i--) {
        if (n % i == 0) {
            ans.push_back({n / i - 1, n / i});
            sum += n / i - 1;
            n = i;
            if (sum == n - 1) {
                break;
            }
        }
    }
    cout << ans.size() << endl;
    for (auto it : ans) {
        cout << it.first << " " << it.second << endl;
    }
    return 0;
}
