N = int(input())
soft_count = 0
hard_count = 0
for _ in range(N):
    _, cheese_type = input().split()
    if cheese_type == "soft":
        soft_count += 1
    else:
        hard_count += 1

size = 0
while size * size < N or (min(soft_count, hard_count) > (size * size) // 2):
    size += 1

print(size)
