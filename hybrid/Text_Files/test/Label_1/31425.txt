import java.util.*;
import java.io.*;

class Main{
    long [] tree;
    long [] arr;

    void build(int node, int start, int end){
        if(start == end){
            tree[node] = arr[start];
            return;
        }

        int mid = (start+end)/2;
        build(2*node, start, mid);
        build(2*node+1, mid+1, end);

        tree[node] = tree[2*node] + tree[2*node+1];
    }

    // 1-based indexing
    long query(int node, int start, int end, int l, int r){
        if(start > end || start > r || end < l){
            return 0;
        }
        if(start >= l && end <= r){
            return tree[node];
        }

        int mid = (start+end)/2;
        long L = query(2*node, start, mid, l, r);
        long R = query(2*node+1, mid+1, end, l, r);

        return L+R;
    }

    // 1-based indexing
    void update(int node, int start, int end, int l, int r, long val){
        if(start > end || start > r || end < l){
            return;
        }
        if(start >= l && end <= r){
            tree[node] = tree[node]^val;
            return;
        }

        int mid = (start+end)/2;
        update(2*node, start, mid, l, r, val);
        update(2*node+1, mid+1, end, l, r, val);

        tree[node] = tree[2*node] + tree[2*node+1];
    }

    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String [] inp = br.readLine().split(" ");
        long [] arr = new long[n+1];
        for(int i=1; i<=n; i++){
            arr[i] = Long.parseLong(inp[i-1]);
        }

        int m = Integer.parseInt(br.readLine());
        Main m1 = new Main();
        m1.arr = arr;
        m1.tree = new long[4*n];
        m1.build(1, 1, n);

        while(m-->0){
            inp = br.readLine().split(" ");
            int type = Integer.parseInt(inp[0]);
            if(type == 1){
                int l = Integer.parseInt(inp[1]);
                int r = Integer.parseInt(inp[2]);
                System.out.println(m1.query(1, 1, n, l, r));
            }
            else{
                int l = Integer.parseInt(inp[1]);
                int r = Integer.parseInt(inp[2]);
                long val = Long.parseLong(inp[3]);
                m1.update(1, 1, n, l, r, val);
            }
        }
    }
}
