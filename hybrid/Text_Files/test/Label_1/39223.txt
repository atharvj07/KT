def find_stars(n, m, grid):
    stars = []
    for x in range(1, n - 1):
        for y in range(1, m - 1):
            if grid[x][y] == '*':
                size = 0
                while True:
                    size += 1
                    if x - size < 0 or x + size >= n or y - size < 0 or y + size >= m:
                        size -= 1
                        break
                    if grid[x - size][y] == '*' and grid[x + size][y] == '*' and grid[x][y - size] == '*' and grid[x][y + size] == '*':
                        continue
                    else:
                        size -= 1
                        break
                if size > 0:
                    stars.append((x + 1, y + 1, size))
                    for i in range(-size, size + 1):
                        grid[x + i][y] = '.'
                        grid[x][y + i] = '.'
                    grid[x][y] = '.'  # Remove the center last to avoid premature deletion.

    # Check if there are any '*' left
    for row in grid:
        if '*' in row:
            return -1

    return stars

n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]

stars = find_stars(n, m, grid)
if stars == -1:
    print(-1)
else:
    print(len(stars))
    for star in stars:
        print(*star)
