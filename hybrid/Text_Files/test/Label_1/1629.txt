#include <bits/stdc++.h>
using namespace std;

long long mod = 1e9 + 7;
int main() {
  long long x, y;
  cin >> x >> y;
  if (y < x) {
    cout << 0 << endl;
    return 0;
  }
  if ((y - x + 1) % x) {
    cout << 0 << endl;
    return 0;
  }
  long long ans = 1, tmp = y - x + 1;
  for (; tmp % x == 0; tmp /= x) ans = ans * 2 % mod;
  cout << ans << endl;
  return 0;
}
