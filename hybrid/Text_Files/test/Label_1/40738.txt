import sys
input = sys.stdin.readline

n, h, m, k = map(int, input().split())
a = []
for i in range(n):
    x, y = map(int, input().split())
    if x == h:
        a.append(y)
        continue
    a.append(m * x + y)
a.sort()
need = [0] * m
for x in a:
    need[x % m] = 1
ans = min(need[:m - k] + need[k:])
print(ans, min(i for i in range(m - k) if not need[i]))
for x in a:
    if x % m in need and ans:
        need[x % m] = 0
        print(x // m + 1, end=' ')
        ans -= 1
