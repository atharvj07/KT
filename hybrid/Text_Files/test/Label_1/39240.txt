import bisect

n=int(input())
a=list(map(int,input().split()))
m=int(input())

d=[]
for i in reversed(range(n)):
    d.append([a[i],i])
    a[i]+=a[i+1] if i+1<n else 0

req=[]
for j in range(m):
    k,p=list(map(int,input().split()))
    req.append([k,p])

req.sort()
d.sort(reverse=True)
ans=[]
for k,p in req:
    i=bisect.bisect_right(d,[-a[p-1]*2-1,n])
    ans.append(d[i][0])

print(*ans,sep='\n')
