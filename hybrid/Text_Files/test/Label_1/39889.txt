n = int(input())
s = input()

# Define the keyboard layout
keyboard = {'1': (0, 0), '2': (0, 1), '3': (0, 2),
            '4': (1, 0), '5': (1, 1), '6': (1, 2),
            '7': (2, 0), '8': (2, 1), '9': (2, 2),
            '0': (3, 1)}

# Calculate the vector movements
movements = []
for i in range(1, n):
    x1, y1 = keyboard[s[i-1]]
    x2, y2 = keyboard[s[i]]
    movements.append((x2-x1, y2-y1))

# Check if there is any other number with the same movements
for key, value in keyboard.items():
    if key not in s:
        x, y = value
        valid = True
        for dx, dy in movements:
            x += dx
            y += dy
            if (x, y) not in keyboard.values():
                valid = False
                break
        if valid:
            print("NO")
            break
else:
    print("YES")
