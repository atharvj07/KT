#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> w(n);
        for (int &w_i : w) { cin >> w_i; }
        sort(w.begin(), w.end());
        bool ok = true;
        int sum = 0;
        for (int w_i : w) {
            sum += w_i;
            if (sum == x) { ok = false; }
        }
        cout << (ok ? "YES\n" : "NO\n");
        if (ok) {
            for (int w_i : w) { cout << w_i << ' '; }
            cout << '\n';
        }
    }
    return 0;
}
