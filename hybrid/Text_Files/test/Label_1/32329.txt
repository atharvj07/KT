import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] dist = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dist[i][j] = sc.nextInt();
            }
        }
        int k = sc.nextInt();
        int[] ans = new int[k];
        for (int i = 0; i < k; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int c = sc.nextInt();
            ans[i] = getDist(dist, n);
            dist[a][b] = dist[b][a] = Math.min(dist[a][b], c);
        }
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }

    private static int getDist(int[][] dist, int n) {
        int sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                sum += dist[i][j];
            }
        }
        return sum;
    }
}
