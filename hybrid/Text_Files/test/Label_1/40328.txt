from collections import Counter
import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    c = Counter()
    for i in range(n):
        row = list(input().split())
        for j in row:
            c[j] += 1
    ans = 0
    for k, v in c.items():
        if v % 2:
            ans += v - 1
    print(ans)
