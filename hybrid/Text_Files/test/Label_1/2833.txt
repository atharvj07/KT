#include <bits/stdc++.h>
using namespace std;
#define i64 long long
i64 n, x, a[100001], p[100001];
int main() {
    cin >> n >> x;
    for (i64 i = 0; i < n; i++) {
        cin >> a[i];
        p[a[i] ^ x]++;
    }
    i64 sum = 0;
    for (i64 i = 0; i < n; i++) { sum += p[a[i]]; }
    cout << sum << endl;
    return 0;
}
