#include<bits/stdc++.h>
using namespace std;
#define nl '\n'
typedef long long ll;

int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);

    int n, k; cin >> n >> k;

    vector<int> skills(n);
    for(auto &skill : skills) cin >> skill;

    sort(skills.begin(), skills.end());

    string ans(n, '1');

    for(int i = 0; i < n;) {
        int pivot = i + k;
        for(int j = pivot; j <= i + k && j < n; j++) {
            if(skills[j] < skills[i]) {
                ans[j] = '2';
                pivot = j;
            }
        }
        i = pivot + k + 1;
    }

    cout << ans << nl;

    return 0;
}
