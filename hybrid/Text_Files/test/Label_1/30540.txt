import java.util.Scanner;
import java.util.Arrays;
import java.util.HashSet;

public class Main {
    static int solve(int n, int[] x, int[] y) {
        int[] z = new int[2 * n];
        System.arraycopy(x, 0, z, 0, n);
        System.arraycopy(y, 0, z, n, n);
        Arrays.sort(z);
        int m = 0;

        for (int i = 0; i < 2 * n; i++) {
            int j = i + 1;
            while (j < 2 * n && z[i] == z[j]) j++;
            m += (j - i) * (j - i - 1) / 2;
            i = j - 1;
        }

        return m % 2 == 0 ? 1 : 2;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] x = new int[n], y = new int[n];

        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            y[i] = scanner.nextInt();
        }

        int winner = solve(n, x, y);
        String name = winner == 1 ? "Koyomi" : "Karen";
        System.out.println(name);
    }
}
