#include <bits/stdc++.h>
using namespace std;
#define int long long int
int n;
vector<int>B;
bool vis[50005];
main() {
	cin >> n;
	B.resize(n);
	for (int i = 0; i < n; i++) { cin >> B[i]; }
	sort(B.begin(), B.end());
	vector<int>A;
	for (int i = 0; i < n; i++) {
		if (!vis[i]) {
			A.push_back(i);
			vis[i] = 1;
			vis[B[i]] = 1;
		}
	}
	for (int i = 0; i < A.size(); i++) { cout << A[i] << ' '; }
	cout << endl;
}
