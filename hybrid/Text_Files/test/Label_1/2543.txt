#include <iostream>
using namespace std;

int main() {
  int a, b, c;
  cin >> a >> b >> c;
  if (a == 0 || b == 0 || c == 0)
    cout << 0;
  else if (a > b && a > c)
    cout << b + 2 * min(b, c);
  else if (b > a && b > c)
    cout << a + 2 * min(a, c);
  else
    cout << c + 2 * min(a, b);
  return 0;
}
