n = int(input())
s = input()
ans = 1e9
for i in range(n-3):
    lst = []
    for j in range(i, i+4):
        lst.append(s[j])
    if lst == ['A', 'C', 'T', 'G']:
        ans = 0
        break
    for j in range(i, i+4):
        c = s[j]
        if c == 'A': s[j] = 'Z'
        else: s[j] = chr(ord(s[j])-1)
        lst = []
        for j in range(i, i+4):
            lst.append(s[j])
        if lst == ['A', 'C', 'T', 'G']:
            ans = min(ans, j-i+1)
        s[j] = c
print(ans)
