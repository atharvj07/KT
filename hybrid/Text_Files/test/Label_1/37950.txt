import sys

for _ in range(int(sys.stdin.readline())):
    n, m = map(int, sys.stdin.readline().split())
    s = sys.stdin.readline().strip()
    L = 0  # left
    R = m - 1  # right
    D = 0  # down
    U = n - 1  # up
    for c in s:
        if c == 'L':
            L += 1
        elif c == 'R':
            R -= 1
        elif c == 'D':
            D += 1
        elif c == 'U':
            U -= 1
        if L > R or D > U:
            break
    print(U + 1, L + 1)
