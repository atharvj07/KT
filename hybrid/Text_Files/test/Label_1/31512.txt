import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String str = "";
        for (int i = 0; i < n; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            if (a < b) {
                str += "A";
            } else {
                str += "G";
            }
        }
        if (str.equals("")) {
            System.out.println("-1");
        } else {
            System.out.println(str);
        }
    }
}
