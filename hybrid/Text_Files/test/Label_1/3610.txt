#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < s.size(); i++) {
    for (int j = i + 1; j < s.size(); j++) {
      bool ok = true;
      for (int k = i; k < (j + i + 1) / 2; k++) {
        if (s[k] != s[j - k + i]) {
          ok = false;
          break;
        }
      }
      if (ok) {
        ans = max(ans, j - i + 1);
      }
    }
  }
  cout << ans << '\n';
  return 0;
}
