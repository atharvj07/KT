#include <iostream>
#include <vector>
#include <deque>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        deque<char> n;
        char c;
        while (cin >> c) {
            if (c != '0' || n.empty()) n.push_back(c);
        }
        for (int i = 1; i + 1 < n.size(); i++) {
            if (n[i] % 2 != n[i + 1] % 2) {
                swap(n[i], n[i + 1]);
                break;
            }
        }
        for (char c : n) cout << c;
        cout << '\n';
    }
    return 0;
}
