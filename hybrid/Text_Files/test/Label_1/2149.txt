#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
const int maxn = 3e5 + 5;
int a[maxn];
ll dp[maxn][64];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = n; i >= 1; i--) {
        for (int j = 0; j < 64; j++) {
            dp[i][j] = dp[i + 1][j];
            int b = (1LL << j);
            if ((a[i] & b) > 0) {
                dp[i][j] = max(dp[i][j], dp[i + 1][j ^ 1] + (b / 2));
            }
        }
    }
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            if (dp[i][0] == dp[j + 1][0]) { ans++; }
        }
    }
    cout << ans << "\n";
    return 0;
}
