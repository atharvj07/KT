md=int(1e9+7)
n=int(input())
s=input()
a=[0]*n
for i in range(n):a[i]=0 if s[i]==s[(i+1)%n] else 1
b=[0]*n
c=[0]*n
for i in range(n-1,-1,-1):
	b[i]=a[i]
	if i!=0:b[i]+=b[i-1]
for j in range(1,n):
	i=j
	while i<n:
		c[i]+=a[i]
		if i!=j:c[i]+=c[i-1]
		i+=j
ans=1
for i in range(n):
	ans*=b[i]**c[i]
	ans%=md
print(ans)
