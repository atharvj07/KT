import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        Map<Integer, Set<Integer>> adjList = new HashMap<>();

        for (int i = 0; i < k; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            adjList.computeIfAbsent(u, s -> new HashSet<>()).add(v);
            adjList.computeIfAbsent(v, s -> new HashSet<>()).add(u);
        }
        for (int i = 0; i < n; i++) {
            adjList.putIfAbsent(i + 1, new HashSet<>());
        }

        int m = in.nextInt();
        for (int i = 0; i < m; i++) {
            int u = in.nextInt();
            int v = in.nextInt();
            adjList.computeIfAbsent(u, s -> new HashSet<>()).remove(v);
            adjList.computeIfAbsent(v, s -> new HashSet<>()).remove(u);
        }

        boolean[] visited = new boolean[n + 1];
        int max = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                max = Math.max(max, dfs(adjList, visited, i));
            }
        }

        System.out.println(max);
    }

    private static int dfs(Map<Integer, Set<Integer>> adjList, boolean[] visited, int node) {
        if (visited[node]) {
            return 0;
        }
        visited[node] = true;
        int count = 1;
        for (int adj : adjList.get(node)) {
            count += dfs(adjList, visited, adj);
        }
        return count;
    }
}
