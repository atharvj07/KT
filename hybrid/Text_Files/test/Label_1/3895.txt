#include <bits/stdc++.h>
using namespace std;
#define mod 1000000007LL
int32_t main() {
  int n;
  cin >> n;
  map<int, int> cnt;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    cnt[x]++;
  }
  int ans = 1;
  for (auto el : cnt) {
    if (el.second >= el.first) {
      ans = (ans * 1LL * (el.second - el.first + 1)) % mod;
    }
  }
  cout << ans << endl;
  return 0;
}
