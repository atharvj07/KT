#include<bits/stdc++.h>
using namespace std;
const int MAXN = 2e5+1;
int32_t main() {
    int64_t n, k, s, t, x, minc = 0x3f3f3f3f, c, v, cm;
    cin >> n >> k >> s >> t;
    for (int64_t i = 0; i < n; i++) {
        cin >> c >> v;
        cm = ((s * 2 - t) + v - 1) / v;
        if (cm <= k) minc = min(minc, c);
    }
    cout << (minc == 0x3f3f3f3f ? -1 : minc) << endl;
    return 0;
}
