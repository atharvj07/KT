#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long n, m, k;
    cin >> n >> m >> k;
    if (n * m - 1 < k) {
        cout << -1 << endl;
    } else {
        long long ans = (n + m - k - 1 + (n + m - k - 1) % 2) / 2;
        cout << ans * ans << endl;
    }
    return 0;
}
