import heapq

for _ in range(int(input())):
    n, p, k = map(int, input().split())
    a = list(map(int, input().split()))
    b = []
    for i in range(n):
        if a[i] % k==0:
            b.append((a[i] // k, 1, [i]))
        else:
            b.append(((a[i] + k - 1) // k, 2, [i]))
    b.sort()
    ans = 0
    for cost, t, ind in b:
        if not p: break
        if t==1:
            if p >= cost:
                ans+=1
                p -= cost
                for x in ind: a[x]=-1
        else:
            c = 0
            for x in ind:
                if a[x]==-1: continue
                if p < cost or c==k-1: break
                p -= cost
                a[x]=-1
                c+=1
            ans+=c
    print(ans)
