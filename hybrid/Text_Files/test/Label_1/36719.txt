import sys

input = [int(x) for x in sys.stdin.read().strip().split()]

n = input[0]
x = input[1:]
d = {}
for i in x:
	d[i] = {}
for i in x:
	for j in x:
		if i == j:
			continue
		dis = abs(i - j)
		while dis not in d[i]:
			d[i][dis] = 0
		d[i][dis] += 1

ans = 0
a = []
for i in x:
	for j in d[i]:
		if d[i][j] > d[i].get(2*j, 0):
			if (j >> 1) not in d[i] or d[i][j] > 2 * d[i][j >> 1]:
				if ans < d[i][j]:
					ans = d[i][j]
					a = [i for _ in range(d[i][j])]

print(ans, *a)
