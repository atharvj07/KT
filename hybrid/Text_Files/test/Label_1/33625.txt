import java.util.*;

public class Main {
    static void solve() {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int m = in.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            int[] p = new int[m];
            for (int i = 0; i < m; i++) {
                p[i] = in.nextInt() - 1;
            }
            boolean[] vis = new boolean[n - 1];
            for (int i : p) {
                vis[i] = true;
            }
            boolean[] sorted = new boolean[n];
            for (int i = 0; i < n - 1; i++) {
                if (!vis[i] && a[i] > a[i + 1]) {
                    System.out.println("NO");
                    return;
                }
                if (a[i] == a[i + 1]) {
                    sorted[i] = true;
                    sorted[i + 1] = true;
                }
            }
            for (boolean b : sorted) {
                if (!b) {
                    System.out.println("NO");
                    return;
                }
            }
            System.out.println("YES");
        }
    }

    public static void main(String[] args) {
        solve();
    }
}
