import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int mod = 1000000007;
        int ans = 1;
        for (int i = 1; i <= n; i++) {
            ans = (ans * i) % mod;
        }
        System.out.println(ans);
    }
}
