import sys
input = sys.stdin.readline

n, m, c_l, c_e, v = map(int, input().split())
l = [int(i) for i in input().split()]
e = [int(i) for i in input().split()]
q = int(input())
for i in range(q):
    x1, y1, x2, y2 = map(int, input().split())
    if x1 == x2:
        print(abs(y1 - y2))
    else:
        d = abs(x1 - x2) // v
        if abs(x1 - x2) % v == 0:
            d -= 1
        ans = d * 2
        if x2 - x1 > 0:
            if y1 < max(e, key=lambda i: i < y1 and max(e, key=lambda i: i <= y1)):
                ans += 1
            if y2 < max(e, key=lambda i: i < y2 and max(e, key=lambda i: i <= y2)):
                ans += 1
        else:
            if y1 > min(e, key=lambda i: i > y1 and min(e, key=lambda i: i >= y1)):
                ans += 1
            if y2 > min(e, key=lambda i: i > y2 and min(e, key=lambda i: i >= y2)):
                ans += 1
        print(ans)
