#include <bits/stdc++.h>
using namespace std;
unordered_map<char, int> C;
unordered_map<char, int> V;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    C[s[i]]++;
    V[s[i + 1]]++;
    i++;
  }
  int ans = max(C.size(), V.size());
  cout << ans << endl;
}
