def check(n, a, b):
    if n - 1 < a + b:
        return -1

    if a == 0 and b == 0:
        return [i for i in range(1, n + 1)]

    ans = []
    for i in range(1, n - a - b + 1):
        ans.extend([i] * 2)
    for i in range(n - a - b + 1, n + 1):
        ans.append(i)
    return ans

for t in range(int(input())):
    n, a, b = map(int, input().split())
    res = check(n, a, b)
    if res != -1:
        print(*res)
    else:
        print(-1)
