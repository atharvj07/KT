import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] sofas = new int[d][4];
        for (int i = 0; i < d; i++) {
            sofas[i][0] = sc.nextInt();
            sofas[i][1] = sc.nextInt();
            sofas[i][2] = sc.nextInt();
            sofas[i][3] = sc.nextInt();
        }
        int cntl = sc.nextInt();
        int cntr = sc.nextInt();
        int cntt = sc.nextInt();
        int cntb = sc.nextInt();
        int ans = -1;
        for (int i = 0; i < d; i++) {
            int[] sofa = sofas[i];
            int l = 0, r = 0, t = 0, b = 0;
            for (int j = 0; j < d; j++) {
                int[] other = sofas[j];
                if (sofa[0] > other[0] && sofa[1] == other[1]) {
                    l++;
                }
                if (sofa[0] < other[0] && sofa[1] == other[1]) {
                    r++;
                }
                if (sofa[1] > other[1] && sofa[0] == other[0]) {
                    t++;
                }
                if (sofa[1] < other[1] && sofa[0] == other[0]) {
                    b++;
                }
            }
            if (l == cntl && r == cntr && t == cntt && b == cntb) {
                ans = i + 1;
                break;
            }
        }
        System.out.println(ans);
    }
}
