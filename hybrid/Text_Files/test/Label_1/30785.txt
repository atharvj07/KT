
import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {
    static final int MAX_WEIGHT = 150;
    static final int TIME_RANGE = 231;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int n = scanner.nextInt();
            if (n == 0) {
                break;
            }
            Bridge[] bridges = new Bridge[n];
            for (int i = 0; i < n; i++) {
                int w = scanner.nextInt();
                int s = scanner.nextInt();
                int e = scanner.nextInt();
                bridges[i] = new Bridge(w, s, e);
            }
            Arrays.sort(bridges, Comparator.comparingInt(bridge -> bridge.start));
            int[] weight = new int[TIME_RANGE];
            for (Bridge bridge : bridges) {
                for (int i = bridge.start; i < bridge.end; i++) {
                    weight[i] += bridge.weight;
                }
            }
            boolean isBroken = false;
            for (int i = 0; i < TIME_RANGE; i++) {
                if (weight[i] > MAX_WEIGHT) {
                    isBroken = true;
                    break;
                }
            }
            System.out.println(isBroken ? "NG" : "OK");
        }
        scanner.close();
    }

    static class Bridge {
        int weight;
        int start;
        int end;

        Bridge(int weight, int start, int end) {
            this.weight = weight;
            this.start = start;
            this.end = end;
        }
    }
}
