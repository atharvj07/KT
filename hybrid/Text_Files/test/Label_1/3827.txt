#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());
        int teams = 0, cur_min = 1e9;
        for (int i = n - 1; i >= 0; i--) {
            cur_min = min(cur_min, a[i]);
            if (cur_min * (n - i) >= x) {
                teams++;
            } else {
                break;
            }
        }
        cout << teams << '\n';
    }
    return 0;
}
