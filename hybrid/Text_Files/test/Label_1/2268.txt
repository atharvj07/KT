#include <iostream>

using ll = long long;

ll a, b;

int main() {
  std::cin >> a >> b;
  ll n = a / 3;
  std::cout << n << "\n";
  for (ll i = 1; i <= n; i++) std::cout << 3 * i << " ";
  std::cout << "\n" << a - 3 * n << "\n";
  for (ll i = 1; i <= a - 3 * n; i++) std::cout << i << " ";
  std::cout << "\n";
}
