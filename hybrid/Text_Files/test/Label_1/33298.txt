import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] a = new int[2][n];
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[2][n];
        dp[0][0] = a[0][0];
        dp[1][0] = a[1][0];
        for (int j = 1; j < n; j++) {
            dp[0][j] = dp[0][j - 1] + a[0][j];
            dp[1][j] = dp[1][j - 1] + a[1][j];
        }
        for (int i = 1; i < 2; i++) {
            for (int j = 1; j < n; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]) + a[i][j];
            }
        }
        System.out.println(dp[1][n - 1]);
    }
}
