n, k = map(int, input().split())
light_levels = list(map(int, input().split()))

# Pair each light level with its hour index, sort by light level
indexed_light_levels = sorted((value, index) for index, value in enumerate(light_levels, start=1))

# Select the last k hours after sorting, which will have the highest light levels
selected_hours = indexed_light_levels[-k:]

# Minimum light level among the selected hours
min_light_level = selected_hours[0][0]

# Extract the indexes of the selected hours
selected_indexes = [index for _, index in selected_hours]

print(min_light_level)
print(' '.join(map(str, selected_indexes)))
