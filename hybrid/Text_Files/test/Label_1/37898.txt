import heapq
import sys
INF=int(1e9)

def dijkstra(s,n,Edges):
    dist=[INF]*(n+1)
    dist[s]=0
    pq=[(0,s)]
    while pq:
        dis,cur=heapq.heappop(pq)
        if dis>dist[cur]:
            continue
        for neb,w in Edges[cur]:
            new_d=dis+w
            if new_d<dist[neb]:
                dist[neb]=new_d
                heapq.heappush(pq,(new_d,neb))
    return dist

n,m,k=map(int,sys.stdin.readline().split())
Edges=[[] for _ in range(n+1)]
for _ in range(m):
    a,b,w=map(int,sys.stdin.readline().split())
    Edges[a].append((b,w))
    Edges[b].append((a,w))
if k==0:
    sys.stdout.write("-1")
    sys.exit()
storages=list(map(int,sys.stdin.readline().split()))
ans=INF
for s in storages:
    dist=dijkstra(s,n,Edges)
    for i in range(1,n+1):
        if i not in storages:
            ans=min(ans,dist[i])
if ans==INF:
    sys.stdout.write("-1")
else:
    sys.stdout.write(str(ans))
