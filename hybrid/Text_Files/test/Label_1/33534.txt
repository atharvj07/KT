import java.util.Scanner;

public class Main {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int n = sc.nextInt();
        int hCnt = 0, vCnt = 0, rCnt = 0;
        int[][] h = new int[n + 1][4];
        int[][] v = new int[n + 1][4];
        for (int i = 1; i <= n; i++) {
            int x1, y1, x2, y2;
            x1 = sc.nextInt();
            y1 = sc.nextInt();
            x2 = sc.nextInt();
            y2 = sc.nextInt();
            if (x1 == x2) {
                v[++vCnt][0] = x1;
                v[vCnt][1] = y1;
                v[vCnt][2] = x2;
                v[vCnt][3] = y2;
            } else {
                h[++hCnt][0] = x1;
                h[hCnt][1] = y1;
                h[hCnt][2] = x2;
                h[hCnt][3] = y2;
            }
        }
        for (int i = 1; i <= hCnt; i++) {
            for (int j = 1; j <= vCnt; j++) {
                if ((h[i][1] <= v[j][3] && h[i][1] >= v[j][1]) && (v[j][0] <= h[i][2] && v[j][0] >= h[i][0])) {
                    rCnt++;
                }
            }
        }
        for (int i = 1; i <= hCnt; i++) {
            for (int j = 1; j <= vCnt; j++) {
                if ((h[i][3] <= v[j][3] && h[i][3] >= v[j][1]) && (v[j][0] <= h[i][0] && v[j][0] >= h[i][2])) {
                    rCnt++;
                }
            }
        }
        System.out.println(rCnt / 4);
    }
}
