#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(2 * n - 1);
  for (int i = 0; i < 2 * n - 1; ++i) {
    std::cin >> a[i];
  }
  std::sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    ans += std::abs(a[i]);
  }
  for (int i = n; i < 2 * n - 1; ++i) {
    ans += std::max(std::abs(a[i]), 0);
  }
  std::cout << ans << std::endl;
  return 0;
}
