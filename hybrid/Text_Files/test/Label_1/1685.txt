#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long a, b;
  cin >> a >> b;
  string s;
  cin >> s;
  int n = s.length();
  vector<int> dx = {0, 1, 0, -1}, dy = {-1, 0, 1, 0};
  for (long long x = 0, y = 0, i = 0, t = 0;; t++) {
    char c = s[i];
    for (int j = 0; j < 4; j++) {
      if (c == "ULDR"[j]) {
        x += dx[j], y += dy[j];
        break;
      }
    }
    if (x == a && y == b) {
      cout << "Yes" << endl;
      return 0;
    }
    i = (i + 1) % n;
    if (i == 0 && t > 0) break;
  }
  cout << "No" << endl;
  return 0;
}
