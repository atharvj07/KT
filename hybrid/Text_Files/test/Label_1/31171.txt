import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] piles = new long[n];
        for (int i = 0; i < n; i++) {
            piles[i] = sc.nextLong();
        }

        long lastPossible = m;
        while (lastPossible > 0) {
            long cntAliceWins = 0, cntBobWins = 0, cntFirstWins = 0, cntSecondWins = 0;
            for (int a = 1; a <= m; a++) {
                for (int b = 1; b <= m; b++) {
                    boolean aliceWins = true;
                    boolean bobWins = true;
                    for (long pile : piles) {
                        if (pile >= b) {
                            pile -= b;
                        } else {
                            bobWins = false;
                        }
                        if (pile >= a) {
                            pile -= a;
                        } else {
                            aliceWins = false;
                        }
                    }
                    if (aliceWins && bobWins) {
                        cntFirstWins++;
                    } else if (aliceWins) {
                        cntAliceWins++;
                    } else if (bobWins) {
                        cntBobWins++;
                    } else {
                        cntSecondWins++;
                    }
                }
            }
            System.out.printf("%d %d %d %d\n", cntAliceWins, cntBobWins, cntFirstWins, cntSecondWins);
            lastPossible /= 2;
            for (int i = 0; i < n; i++) {
                piles[i] /= 2;
            }
        }
    }
}
