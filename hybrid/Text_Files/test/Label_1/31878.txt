import java.util.Scanner;

public class SharkTeeth {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt() - 1;
            teeth[i][1] = scanner.nextInt();
        }

        int[] rowViability = new int[m];
        for (int i = 0; i < n; i++) {
            rowViability[teeth[i][0]] += teeth[i][1];
        }

        int maxCrucians = 0;
        for (int i = 0; i < m; i++) {
            maxCrucians = Math.max(maxCrucians, Math.min(k, rowViability[i]));
        }

        System.out.println(maxCrucians);
    }
}
