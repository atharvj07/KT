#include<iostream>
#include<vector>

using namespace std;

int main() {
	int n, m, t, v, x, y, q;
	cin >> n >> m;
	vector<int> a(n);
	for (int i = 0; i < n; ++i) { cin >> a[i]; }
	while (m--) {
		cin >> t;
		switch (t) {
			case 1: {
				cin >> v >> x; --v; a[v] = x;
			} break;
			case 2: {
				cin >> y; for (int i = 0; i < n; ++i) { a[i] += y; }
			} break;
			case 3: {
				cin >> q; --q; cout << a[q] << endl;
			} break;
		}
	}
	return 0;
}
