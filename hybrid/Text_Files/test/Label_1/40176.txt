import math

n = int(input())
for i in range(n):
    p, q, b = map(int, input().split())
    if p == 0:
        print("Finite")
        continue

    p = str(p)
    q = str(q)
    if b == 10:
        res = p + '/' + q
    else:
        res = str(int(p, 10) // int(q, 10))

    if res[-1] == '0' or res.count('.') + res.count(',') >= 2:
        print("Finite")
    else:
        print("Infinite")
