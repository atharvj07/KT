#include<bits/stdc++.h>
using namespace std;
bool check(string s,string t){
    int n=s.size(),m=t.size();
    if(n<m)return 0;
    int i=0,j=0,c=0;
    while(j<m){
        if(s[i]==t[j])c++;
        i++;
        j++;
    }
    if(c==m){
        while(i<n){
            if(s[i]=='0')c++;
            i++;
        }
        if(c==n)return 1;
    }
    return 0;
}
int main(){
    string x,y;
    cin>>x>>y;
    string a=x,b=y;
    reverse(a.begin(),a.end());
    reverse(b.begin(),b.end());
    if(a==b)cout<<"YES"<<endl;
    else if(check(a,b))cout<<"YES"<<endl;
    else cout<<"NO"<<endl;
    return 0;
}
