n=int(input());s=input();a=[0]+[int(x) for x in input().split()];
dp=[[0]*len(a) for _ in range(n+1)];
for i in range(n-1,-1,-1):
  for j in range(min(n-i,len(a))):
    if i+j>=n:continue;
    dp[i][j]=max(dp[i+j][k]+(a[j] if s[i]==s[i+j-1] else 0)for k in range(len(a)));
print(max(max(dp[i])for i in range(n+1)));
