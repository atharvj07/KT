import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < q; i++) {
            int t = sc.nextInt();
            int x = sc.nextInt();
            int y = sc.nextInt();
            if (t == 1) {
                for (int j = 0; j < x; j++) {
                    a[j] = Math.max(a[j], y);
                }
            } else {
                int ans = 0;
                for (int j = x - 1; j < n; j++) {
                    if (y >= a[j]) {
                        y -= a[j];
                        ans++;
                    }
                }
                System.out.println(ans);
            }
        }
    }
}
