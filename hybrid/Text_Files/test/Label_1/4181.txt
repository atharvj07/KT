#include <bits/stdc++.h>
using namespace std;

const int maxn = 100+5;
char f[maxn][maxn];
int p[maxn][maxn];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) {
            cin >> f[i][j];
            p[i][j] = 0;
        }
    int ans = n*m;
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) {
            if (f[i][j] == 'W')
                p[i][j] = 1;
            else p[i][j] = -1;
        }
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) {
            if (!i && !j) continue;
            else if (!i) p[i][j] += p[i][j-1];
            else if (!j) p[i][j] += p[i-1][j];
            else p[i][j] += p[i-1][j] + p[i][j-1] - p[i-1][j-1];
        }
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j) {
            for (int ii = i; ii < n; ++ii)
                for (int jj = j; jj < m; ++jj) {
                    int res = p[ii][jj];
                    if (i) res -= p[i-1][jj];
                    if (j) res -= p[ii][j-1];
                    if (i && j) res += p[i-1][j-1];
                    ans = min(ans, res+res);
                }
        }
    cout << ans << endl;
}
