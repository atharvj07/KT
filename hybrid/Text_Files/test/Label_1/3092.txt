#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 255;
const ll MOD = 1e9 + 7;
int main() {
    int N, K;
    cin >> N >> K;
    ll dp[MAXN][MAXN] = {};
    dp[0][0] = 1;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            for (int k = 1; k <= K; k++) {
                if (i > 0) dp[i][j] = (dp[i][j] + dp[i - 1][j] * max(k - 1, 1)) % MOD;
                if (j > 0) dp[i][j] = (dp[i][j] + dp[i][j - 1] * max(k - 1, 1)) % MOD;
            }
        }
    }
    cout << dp[N - 1][N - 1] << endl;
}
