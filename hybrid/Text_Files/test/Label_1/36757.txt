n = int(input())
a = [float(input()) for _ in range(n)]

b = [int(x) for x in a]
difference = sum(b)

for i in range(n):
    if difference == 0:
        break
    if a[i] % 1 != 0:
        if difference > 0 and a[i] > 0:
            b[i] = int(a[i])
        elif difference < 0 and a[i] < 0:
            b[i] = int(a[i]) + 1
        else:
            b[i] = int(round(a[i]))
        difference = sum(b)

for num in b:
    print(num)
