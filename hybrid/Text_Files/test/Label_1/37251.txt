n = int(input())
a = list(map(int, input().split()))

handshakes = [0] * n
for shake in a:
    handshakes[shake] += 1

order = []
current = 0

for i in range(n):
    if handshakes[current] == 0:
        print("Impossible")
        break
    order.append(a.index(current) + 1)
    handshakes[current] -= 1
    a[a.index(current)] = -1
    if current > 0 and handshakes[current - 1] > 0:
        current -= 1
    else:
        current += 2
else:
    print("Possible")
    print(' '.join(map(str, order)))
