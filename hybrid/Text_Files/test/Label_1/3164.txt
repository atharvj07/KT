#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  bool one[101] = {};
  bool two[101] = {};
  bool found = false;
  while (n--) {
    int t;
    cin >> t;
    if (!one[t])
      one[t] = true;
    else if (!two[t])
      two[t] = true;
    else
      found = true;
  }
  if (found)
    cout << "NO" << endl;
  else {
    int a = -1, b = -1;
    for (int i = 1; i < 101; i++) {
      if (one[i]) {
        if (a == -1)
          a = i;
        else if (b == -1)
          b = i;
      }
    }
    cout << "YES" << endl;
    cout << a << " " << b << endl;
  }
}
