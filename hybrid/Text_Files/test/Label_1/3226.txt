#include <bits/stdc++.h>
using namespace std;

int main()
{
    int n, k, l, ans = 0;
    char x;
    cin >> n >> k;
    string s;
    cin >> s;
    for (int i = 0; i < n; i++)
    {
        if (i == 0 or s[i] != s[i - 1])
            l = 1;
        else
            l++;
        if (l == k)
            ans++, l = 0;
    }
    cout << ans << endl;
    return 0;
}
