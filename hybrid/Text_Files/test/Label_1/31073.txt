import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] time = new int[n];
        for (int i = 0; i < n; i++) {
            int h = in.nextInt();
            int m = in.nextInt();
            time[i] = h * 60 + m;
        }
        Arrays.sort(time);
        int ans = 0;
        int cur = 0;
        for (int t : time) {
            if (t >= cur) {
                ans++;
                cur = t + 1;
            }
        }
        System.out.println(ans);
    }
}
