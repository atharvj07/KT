import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int t = sc.nextInt();
        String s = sc.next();

        int cnt = 0;
        int sum = 0;
        boolean[] vis = new boolean[n + 1];
        int[] ori = new int[n + 1];
        for (int i = 0; i < n; i++) {
            ori[i + 1] = (s.charAt(i) == 'w') ? 0 : 1;
        }

        int cur = 1;
        while (sum + a <= t) {
            if (!vis[cur]) {
                if (ori[cur] == 0) {
                    sum += 1;
                } else {
                    sum += b + 1;
                }
                cnt++;
            }
            vis[cur] = true;
            sum += a;
            cur++;
            if (cur > n) {
                cur = 1;
            }
        }

        System.out.println(cnt);
    }
}
