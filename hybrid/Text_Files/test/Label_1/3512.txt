#include <iostream>
#include <vector>
using namespace std;
typedef long long LL;
const int mod = 1e9 + 7;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<LL> dp(n, -1);
    dp[n - 1] = 1;
    for (int i = n - 2; ~i; --i) {
        if (s[i] == '*')
            continue;
        int c = 0;
        if (i + 1 < n && s[i + 1] != '?' && s[i + 1] - '0' > 0)
            c++;
        if (i + 2 < n && s[i + 2] != '?' && s[i + 2] - '0' > 0)
            c++;
        if (s[i] == '?' && c == 0)
            continue;
        if (s[i] != '?' && s[i] - '0' != c)
            continue;
        dp[i] = (dp[i + 1] + ((s[i] == '?' ? 3 : 1) * dp[i + max(c, 1)])) % mod;
    }
    cout << dp[0] << endl;
    return 0;
}
