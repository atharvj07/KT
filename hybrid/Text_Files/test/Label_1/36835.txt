n = int(input())
a, b = map(int, input().split())
h = list(map(int, input().split()))
h.sort()
ans = 2e9
i = 0
j = n - 1
while i < j:
  if a >= h[i] and b >= h[j]:
    ans = min(ans, h[i] + h[j])
    a -= h[i]
    b -= h[j]
    i += 1
    j -= 1
  elif a >= h[i]:
    ans = min(ans, h[i] + b)
    a -= h[i]
    i += 1
  elif b >= h[j]:
    ans = min(ans, a + h[j])
    b -= h[j]
    j -= 1
  else:
    break
if a > 0 or b > 0:
  ans = -1
print(ans)
