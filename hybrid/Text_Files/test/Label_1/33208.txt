import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }

        long score = 0;
        long maxPrefix = 0;
        for (int i = 0; i < k; i++) {
            maxPrefix += a[i];
        }
        score = Math.max(score, maxPrefix);

        for (int i = k; i < n; i++) {
            maxPrefix = maxPrefix - a[i - k] + a[i];
            score = Math.max(score, maxPrefix);
        }

        System.out.println(score);
    }
}
