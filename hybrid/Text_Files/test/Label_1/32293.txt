import java.util.*;

public class Main {

    public static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    public static long lcm(long a, long b) {
        return (a * b) / gcd(a, b);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            long k = in.nextLong();
            long l = in.nextLong();
            long r = in.nextLong();
            long p = in.nextLong();
            long n = lcm(k * k + 1, k * k + 2);
            for (long j = k * k + 3; j <= k * k + r - l + 1; j++) {
                n = lcm(n, j);
            }
            System.out.println(n % p);
        }
    }
}
