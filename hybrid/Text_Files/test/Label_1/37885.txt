import sys
input = sys.stdin.readline

n = int(input())
s, g, s_tot, g_tot = [], [], 0, 0
for _ in range(n):
    x, y = map(int, input().split())
    s.append(x)
    g.append(y)
    s_tot += x
    g_tot += y
if n == 1 or s_tot + max(g) - min(g) <= n - 1:
    print(g_tot)
    print(*[s_tot])
else:
    print(-1)
