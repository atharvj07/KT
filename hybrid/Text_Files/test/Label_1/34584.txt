import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int[] edge = new int[3];
            edge[0] = Integer.parseInt(line[0]);
            edge[1] = Integer.parseInt(line[1]);
            edge[2] = Integer.parseInt(line[2]);
            edges.add(edge);
        }
        boolean[] visited = new boolean[n + 1];
        int[] distance = new int[n + 1];
        boolean consistent = true;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                if (!dfs(edges, i, 0, visited, distance)) {
                    consistent = false;
                    break;
                }
            }
        }
        System.out.println(consistent ? "Yes" : "No");
    }

    private static boolean dfs(List<int[]> edges, int node, int dist, boolean[] visited, int[] distance) {
        visited[node] = true;
        distance[node] = dist;
        for (int[] edge : edges) {
            if (edge[0] == node) {
                if (!visited[edge[1]] && !dfs(edges, edge[1], dist + edge[2], visited, distance)) {
                    return false;
                } else if (visited[edge[1]] && distance[edge[1]] != dist + edge[2]) {
                    return false;
                }
            } else if (edge[1] == node) {
                if (!visited[edge[0]] && !dfs(edges, edge[0], dist - edge[2], visited, distance)) {
                    return false;
                } else if (visited[edge[0]] && distance[edge[0]] != dist - edge[2]) {
                    return false;
                }
            }
        }
        return true;
    }
}
