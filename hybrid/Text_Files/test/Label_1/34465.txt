import java.util.Arrays;
import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] a = new int[2 * n];
        HashMap<Integer, Integer> map = new HashMap<>();

        for (int i = 0; i < 2 * n; ++i) {
            a[i] = s.nextInt();
            map.put(a[i], i);
        }

        int ans = 0;
        for (int i = 0; i < 2 * n; ++i) {
            if (a[i] == a[i + 1]) {
                ++i;
                continue;
            }

            int j = map.get(a[i]);
            map.put(a[i], i + 1);
            map.put(a[i + 1], j);
            a[j] = a[i + 1];
            a[i + 1] = a[i];
            ++ans;
        }

        System.out.println(ans);
    }
}
