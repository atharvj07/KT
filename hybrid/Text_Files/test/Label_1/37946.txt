import sys
input = sys.stdin.readline

n = int(input())
strings = [input().strip() for _ in range(n)]
blocks = [i for i in range(n)]
ans = []

for _ in range(int(input())):
    a, b = map(int, input().split())
    a, b = blocks[a-1], blocks[b-1]
    if a != b:
        strings[b] += strings[a]
        strings[a] = ''
        blocks[b] = a
    ans.append('0' if strings[b].count('0') == len(strings[b]) else str(max(-1, int(len(strings[b])/2))))

print(*ans, sep='\n')
