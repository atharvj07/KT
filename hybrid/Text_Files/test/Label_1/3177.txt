#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 10;
int n;
char cur[MAXN], pre[MAXN];
int beauty[MAXN];
int main() {
  cin >> n;
  string Kuro, Shiro, Katie;
  cin >> Kuro >> Shiro >> Katie;
  for (int i = 0; i < n; ++i) {
    pre[i] = (i == 0 ? Kuro[0] : pre[i - 1]);
    cur[i] = (i == 0 ? Shiro[0] : cur[i - 1]);
    beauty[i] = 0;
    for (int j = 0; i - j >= 0; ++j) {
      if (pre[i - j] != cur[i]) { continue; }
      beauty[i] = max(beauty[i], beauty[i - j] + 1);
    }
    if (cur[i] != Katie[i]) { beauty[i] = max(beauty[i], beauty[i - 1] + 1); }
    cur[i] = Katie[i];
  }
  if (beauty[n - 1] == beauty[n - 2]) {
    cout << "Draw";
  } else if (beauty[n - 1] == beauty[n - 3]) {
    cout << "Kuro";
  } else {
    cout << "Shiro";
  }
  return 0;
}
