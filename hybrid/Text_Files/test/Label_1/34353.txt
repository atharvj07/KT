import java.util.*;

public class Main {
    static int n, q, k;
    static List<Integer>[] contacts;
    static Map<Integer, Integer> visitDays;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        q = in.nextInt();
        k = in.nextInt();
        contacts = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            contacts[i] = new ArrayList<>();
        }
        visitDays = new HashMap<>();
        for (int i = 0; i < q; i++) {
            int t = in.nextInt();
            if (t == 1) {
                int x = in.nextInt();
                int y = in.nextInt();
                contacts[x].add(y);
                contacts[y].add(x);
                visitDays.put(x, i);
                visitDays.put(y, i);
            } else if (t == 2) {
                int z = in.nextInt();
                System.out.println(countContacts(z));
            } else {
                visitDays.clear();
            }
        }
    }

    private static int countContacts(int z) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(z);
        Set<Integer> visited = new HashSet<>();
        visited.add(z);
        int count = 1;
        while (!queue.isEmpty()) {
            int curr = queue.poll();
            for (int contact : contacts[curr]) {
                if (visitDays.getOrDefault(contact, -1) >= visitDays.getOrDefault(z, 0) - k && !visited.contains(contact)) {
                    visited.add(contact);
                    queue.add(contact);
                    count++;
                }
            }
        }
        return count;
    }
}
