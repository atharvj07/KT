#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll a[505][505];
ll n, k;
int main() {
  cin >> n >> k;
  ll ans = n * (n + 1) / 2 * n;
  for (ll i = 1; i <= n; ++i) {
    for (ll j = 1; j <= n; ++j) {
      a[j][i] = (i - 1) * n + j;
    }
  }
  for (ll i = 1; i <= n; ++i) swap(a[i][k], a[i][n]);
  cout << ans << endl;
  for (ll i = 1; i <= n; ++i) {
    for (ll j = 1; j <= n; ++j) cout << a[i][j] << " ";
    cout << endl;
  }
  return 0;
}
