n = int(input())
p = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
m = int(input())
c = list(map(int, input().split()))
for x in c:
  t = []
  for i in range(n):
    if a[i] == x or b[i] == x:
      t.append(i)
  if t:
    print(min([p[i] for i in t]))
  else:
    print(-1)
