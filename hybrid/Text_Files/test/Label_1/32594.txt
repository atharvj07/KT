import java.util.*;
import java.io.*;
 
public class Main {
 
    public static void main(String[] args) {
        int n, t;
        long min;
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
 
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
 
        Arrays.sort(a);
        min = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] >= i + 1) {
                t = i + 1;
            } else if (i + 1 - a[i] > n) {
                t = n;
            } else {
                t = i + 1 - a[i];
            }
 
            if (t < 0) {
                t = 0;
            }
            min += 2 * (t - i);
        }
 
        System.out.println(min);
    }
}
