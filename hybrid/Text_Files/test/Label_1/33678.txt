import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class ChessboardSpells {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        List<Integer> verticalSpells = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            verticalSpells.add(Integer.parseInt(reader.readLine()));
        }

        List<Spell> horizontalSpells = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int x1 = Integer.parseInt(line[0]);
            int x2 = Integer.parseInt(line[1]);
            int y = Integer.parseInt(line[2]);
            horizontalSpells.add(new Spell(x1, x2, y));
        }

        Collections.sort(horizontalSpells);

        int minSpellsToRemove = 0;
        int currentRow = 1;
        int currentColumn = 1;

        for (Spell spell : horizontalSpells) {
            if (spell.y == currentRow) {
                // Horizontal spell blocking the current row
                if (spell.x1 <= currentColumn && currentColumn <= spell.x2) {
                    // Rook cannot go up, so it needs to remove the spell
                    minSpellsToRemove++;
                    currentColumn = spell.x2 + 1;
                }
            } else if (spell.y > currentRow) {
                // Horizontal spell above the current row
                if (currentColumn <= spell.x1) {
                    // Rook can go up without hitting the spell
                    currentRow = spell.y;
                } else {
                    // Rook needs to remove the spell to go up
                    minSpellsToRemove++;
                    currentRow = spell.y;
                    currentColumn = spell.x1;
                }
            }
        }

        // Check if rook needs to remove any vertical spells
        for (int i = 0; i < verticalSpells.size(); i++) {
            int x = verticalSpells.get(i);
            if (currentColumn == x) {
                // Vertical spell blocking the current column
                minSpellsToRemove++;
                if (i + 1 < verticalSpells.size()) {
                    currentColumn = verticalSpells.get(i + 1);
                } else {
                    currentColumn = 1000000000;
                }
            }
        }

        System.out.println(minSpellsToRemove);
    }

    static class Spell implements Comparable<Spell> {
        int x1;
        int x2;
        int y;

        public Spell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }

        @Override
        public int compareTo(Spell other) {
            return Integer.compare(y, other.y);
        }
    }
}
