t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    roads = [tuple(map(int, input().split())) for _ in range(m)]

    # Since the problem does not specify a way to choose the orientation
    # for maximizing the number of cities, we simply orient roads from the 
    # city with the smaller number to the city with the larger number.
    # This will not necessarily maximize the number of such cities,
    # but it will provide a valid orientation of roads.

    oriented_roads = [(min(u, v), max(u, v)) for u, v in roads]

    # Output 
    print(n)  # In this simple approach, we are not actually maximizing, so this is a placeholder
    for road in oriented_roads:
        print(road[0], road[1])
