#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n), s(n + 1);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      s[i + 1] = s[i] + a[i];
    }
    bool ok = false;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          for (int l = k + 1; l < n; l++) {
            if (s[j] - s[i] == s[l] - s[k]) ok = true;
          }
        }
      }
    }
    cout << (ok ? "YES" : "NO") << "\n";
  }
}
