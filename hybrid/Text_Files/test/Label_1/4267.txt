#include <iostream>
#include <cstring>

using namespace std;
char t[200001];
int main() {
  int n, i, j= 0, l;
  cin >> t;
  l = strlen(t);
  for(i = 0;i<l;i++){
    while(i < l && t[i] == t[i+1] && t[i] == t[i+2])i++;
    while(i < l && t[i] == t[i+1])i++;
    t[j++] = t[i];
  }
  t[j] = 0;
  cout << t;
  return 0;
}
