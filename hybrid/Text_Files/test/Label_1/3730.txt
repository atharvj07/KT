#include <bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for(int i = a; i < (b); ++i)
#define all(x) begin(x), end(x)
using ll = long long;

int32_t main() {
    int r, c, n, k;
    cin >> r >> c >> n >> k; 
    vector<bool> row(r), col(c);
    rep(i, 0, n) {
        int x, y; cin >> x >> y;
        row[x-1] = true;
        col[y-1] = true;
    }
    int rowsm = 0, colsm = 0;
    rep(i, 0, r) rowsm += row[i];
    rep(i, 0, c) colsm += col[i];

    int ans = (rowsm >= k) * (colsm >= k) * r * c - (rowsm > k) * (colsm < k) * rowsm - (rowsm < k) * (colsm > k) * colsm + (rowsm < k) * (colsm < k) * rowsm * colsm;
    cout << ans << endl;
    return 0;
}
