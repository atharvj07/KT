import sys

input = sys.stdin.readline
n, m = map(int, input().split())
w = [list(map(int, input().split())) for _ in range(m)]

w.sort(key=lambda a: -a[1])

ans = 0
was = [False] * n

for x, d in w:
    for i in range(n):
        if not was[i]:
            was[i] = True
            ans += x + d * i
            break
print("{:.12f}".format(ans / n))
