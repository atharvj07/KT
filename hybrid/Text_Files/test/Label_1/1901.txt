#include <iostream>
#include <vector>
using namespace std;
int main()
{
    int n;
    cin >> n;
    vector<int> nums(n);
    int cnt = 1, t = 0;
    for (int i = 0; i < n; i++)
    {
        cin >> nums[i];
        if (i && nums[i] == 1)
        {
            cnt++;
            t = max(t, cnt);
        }
    }
    cout << cnt << endl;
    for (int i = 0; i < cnt; i++) cout << t-- << " ";
    return 0;
}
