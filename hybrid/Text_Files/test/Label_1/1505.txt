#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mxN = 2e5 + 5;
int n;
ll arr[mxN], ans[mxN];
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) cin >> arr[i];
        ll res = 0, g;
        for (int i = 0; i < n; i++) { res += arr[i]; }
        g = res / n;
        for (int i = 0; i < n; i++) { ans[i] = g - arr[i]; }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (ans[i] % arr[i] == 0) cnt++;
        }
        if (cnt == n)
            cout << 0 << '\n';
        else
            cout << 1 << '\n';
    }
    return 0;
}
