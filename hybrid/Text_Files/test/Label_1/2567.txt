#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main(){
    ios::sync_with_stdio(0); cin.tie(0);
    int n; cin >> n;
    vector<int> a(n);
    for(int &i : a) cin >> i;
    vector<int> f(n + 1, -1), l(n + 1, -1);
    ll ans = 0;
    for(int i = 0; i < n; i++){
        if(l[ a[i] ] != -1){
            ans -= (f[a[i]] > l[a[i]]);
        }
        if(f[ a[i] ] == -1){ f[a[i]] = i; }
        l[ a[i] ] = i;
        if(f[ a[i] ] == l[a[i]]){ ans++; }
    }
    cout << ans << "\n";
    return 0;
}
