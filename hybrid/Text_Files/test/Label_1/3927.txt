#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> times(n);
  for (int i = 0; i < n; i++) { cin >> times[i]; }
  if (n == 0) {
    cout << 90 << endl;
    return 0;
  }
  int ans = times[0];
  for (int i = 1; i < n; i++) {
    if (times[i] - times[i - 1] > 15) {
      ans = times[i];
    }
  }
  cout << ans + 15 << endl;
  return 0;
}
