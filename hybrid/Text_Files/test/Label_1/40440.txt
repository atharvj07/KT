import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    x = sum(a) // 2
    i = 0
    while x > 0:
        if x >= a[i]:
            x -= a[i]
            a[i] = -1
        i += 1
    b = []
    for i in range(n):
        if a[i] == -1:
            if b and b[-1] % 2 == 0:
                b.append(b[-1] // 2)
            else:
                b += [b[-1] // 2, 2]
        else:
            b.append(a[i])
    print(*b)
