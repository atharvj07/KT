import java.util.Scanner;

public class Main {

    private static final int MAX_N = 100;
    private static final int MAX_SIZE = 105;
    private static final int MAX_SCORE = 1000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int hl = sc.nextInt();
        int hr = sc.nextInt();
        int n = sc.nextInt();

        int[][] mirrors = new int[MAX_N + 1][4];

        for (int i = 1; i <= n; i++) {
            mirrors[i][0] = sc.nextInt();
            char c = sc.next().charAt(0);
            mirrors[i][1] = (c == 'T') ? 0 : 1;
            mirrors[i][2] = sc.nextInt();
            mirrors[i][3] = sc.nextInt();
        }

        int[][] dp = new int[MAX_SIZE][MAX_SIZE];

        for (int i = 0; i < MAX_SIZE; i++) {
            for (int j = 0; j < MAX_SIZE; j++) {
                dp[i][j] = -1;
            }
        }

        dp[hl][0] = 0;

        for (int x = 0; x < MAX_SIZE; x++) {
            for (int y = 0; y < MAX_SIZE; y++) {
                if (dp[y][x] == -1) {
                    continue;
                }

                for (int i = 1; i <= n; i++) {
                    int mirrorX = mirrors[i][2];
                    int mirrorY = mirrors[i][1] * (MAX_SIZE - 1 - mirrors[i][3]) + mirrors[i][1];
                    if (mirrorY == y && mirrorX >= x) {
                        int ny = y;
                        int nx = x;
                        if (mirrors[i][1] == 0) {
                            ny = MAX_SIZE - 1 - ny;
                        }
                        if (ny > y) {
                            nx = mirrorX;
                        } else if (ny < y) {
                            nx = mirrorX - 1;
                        }
                        if (nx >= 0 && nx < MAX_SIZE) {
                            dp[ny][nx] = Math.max(dp[ny][nx], dp[y][x] + mirrors[i][0]);
                        }
                    }
                }
            }
        }

        int maxScore = 0;
        for (int i = 0; i < MAX_SIZE; i++) {
            maxScore = Math.max(maxScore, dp[hr][i]);
        }

        System.out.println(maxScore);
    }
}
