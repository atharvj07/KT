import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner cin=new Scanner(System.in);
        int n=cin.nextInt(),m=cin.nextInt();
        int ed[][]=new int [n+1][n+1],deg[]=new int [n+1];
        for(int i=1;i<=m;i++){
            int u=cin.nextInt(),v=cin.nextInt();
            ed[u][v]=1;
            ed[v][u]=1;
            deg[u]++;
            deg[v]++;
        }
        int rem=(n+m)/2;
        System.out.println(rem);
        Random ran=new Random();
        for(int i=1;i<=n;i++){
            if(rem==0)break;
            boolean flag=true;
            while(flag) {
                int r=ran.nextInt(n) + 1;
                if (r == i || deg[i] == 0 || deg[r] == 0)continue;
                if (ed[i][r] == 1) {
                    System.out.println(i + " " + r);
                    deg[i]--;
                    deg[r]--;
                    rem--;
                    ed[i][r] = ed[r][i] = 0;
                    flag=false;
                }
            }
        }
    }
}
