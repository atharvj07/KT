import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        int[][] dp = new int[n][n];
        dp[0][0] = a[0];
        for (int i = 1; i < n; i++) {
            dp[i][0] = dp[i - 1][0] + a[i];
        }
        for (int i = 1; i < n; i++) {
            dp[0][i] = dp[0][i - 1] + b[i];
        }
        for (int i = 1; i < n; i++) {
            for (int j = 1; j < n; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]) + Math.max(a[i], b[j]);
            }
        }
        System.out.println(dp[n - 1][n - 1]);
    }
}
