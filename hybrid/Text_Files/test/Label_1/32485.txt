import java.util.*;
import java.lang.*;
class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int d = scanner.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = scanner.nextInt();
    }
    Arrays.sort(a);
    int l = 0, r = n - 1, ans = n;
    while (l < r) {
      if (a[r] - a[l] > d) {
        ans--;
        l++;
      } else {
        r--;
      }
    }
    System.out.println(ans);
  }
}
