import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {

  static class Pair implements Comparable<Pair> {
    int a, b;

    Pair(int a, int b) {
      this.a = a;
      this.b = b;
    }

    @Override
    public int compareTo(Pair o) {
      return Integer.compare(a, o.a);
    }
  }

  static class PrimeFactor {
    int prime, power;

    PrimeFactor(int prime, int power) {
      this.prime = prime;
      this.power = power;
    }
  }

  static long gcd(long a, long b) {
    if (a == 0) {
      return b;
    }
    return gcd(b % a, a);
  }

  static int solve(int n, int q, int[] a, int[] queries) {
    int[] cnt = new int[n + 1];
    long sum = 0;

    for (int x : queries) {
      cnt[x] ^= 1;
      if (cnt[x] == 0) {
        sum -= (long) a[x - 1] * (a[x - 1] - 1) / 2;
      } else {
        sum += (long) a[x - 1] * (a[x - 1] - 1) / 2;
      }

      Set<Pair> primes = new TreeSet<>();
      int m = 1;

      for (int i = 1; i <= n; i++) {
        if (cnt[i] > 0) {
          m++;

          int p = a[i - 1];
          ArrayList<PrimeFactor> factors = new ArrayList<>();
          for (int j = 2; j * j <= p; j++) {
            int power = 0;
            while (p % j == 0) {
              p /= j;
              power++;
            }
            if (power > 0) {
              factors.add(new PrimeFactor(j, power));
            }
          }
          if (p > 1) {
            factors.add(new PrimeFactor(p, 1));
          }

          for (PrimeFactor f : factors) {
            int prime = f.prime, power = f.power;
            boolean found = false;
            for (Pair pp : primes) {
              if (pp.a == prime) {
                found = true;
                pp.b = Math.max(power, pp.b);
              }
            }
            if (!found) {
              primes.add(new Pair(prime, power));
            }
          }
        }
      }

      long product = 1;
      for (Pair p : primes) {
        product *= (long) Math.pow(p.a, p.b - 1) * (p.a - 1);
      }

      for (int i = 1; i <= n; i++) {
        if (cnt[i] > 0) {
          int p = a[i - 1];
          ArrayList<PrimeFactor> factors = new ArrayList<>();
          for (int j = 2; j * j <= p; j++) {
            int power = 0;
            while (p % j == 0) {
              p /= j;
              power++;
            }
            if (power > 0) {
              factors.add(new PrimeFactor(j, power));
            }
          }
          if (p > 1) {
            factors.add(new PrimeFactor(p, 1));
          }

          int divisorCount = 1;
          for (PrimeFactor f : factors) {
            divisorCount *= f.power + 1;
          }
          divisorCount--;

          long g = gcd(product, a[i - 1]);
          sum += (long) (divisorCount - (divisorCount * g / a[i - 1])) * cnt[i] * (m - cnt[i]);
        }
      }
      System.out.println(sum);
    }

    return 0;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    int n = Integer.parseInt(st.nextToken());
    int q = Integer.parseInt(st.nextToken());
    int[] a = new int[n];
    st = new StringTokenizer(br.readLine());
    for (int i = 0; i < n; i++) {
      a[i] = Integer.parseInt(st.nextToken());
    }
    int[] queries = new int[q];
    for (int i = 0; i < q; i++) {
      queries[i] = Integer.parseInt(br.readLine());
    }
    solve(n, q, a, queries);
  }
}
