import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        while (T-- > 0) {
            String s = sc.next();
            int[] cnt = new int[26];
            for (int i = 0; i < s.length(); i++) {
                cnt[s.charAt(i) - 'a']++;
            }
            long ans = 0;
            for (int i = 0; i < 26; i++) {
                ans += (long) cnt[i] * (cnt[i] + 1) / 2;
            }
            System.out.println(ans);
        }
    }
}
