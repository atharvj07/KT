import java.util.*;
import java.io.*;

class Main {
    static final int MAXN = 100000 + 10;
    static final int MOD = 1000000000;
    static int n, m, pow2[];
    static List<Integer> adj[];
    static long dp[];

    public static void main(String[] args) {
        InputStreamReader isr = new InputStreamReader(System.in);
        BufferedReader br = new BufferedReader(isr);
        StringTokenizer st;

        try {
            st = new StringTokenizer(br.readLine());
            n = Integer.parseInt(st.nextToken());
            m = Integer.parseInt(st.nextToken());
            pow2 = new int[2 * n];
            for (int i = 0; i < 2 * n; i++) {
                pow2[i] = (i % 2 == 0) ? 1 : 2;
            }

            adj = new List[2 * n];
            for (int i = 0; i < 2 * n; i++) {
                adj[i] = new ArrayList<>();
            }
            dp = new long[2 * n];
            Arrays.fill(dp, -1);
            for (int i = 0; i < m; i++) {
                st = new StringTokenizer(br.readLine());
                int x = Integer.parseInt(st.nextToken());
                int y = Integer.parseInt(st.nextToken());
                add_edge(x - 1, y - 1, false);
                add_edge(n + x - 1, n + y - 1, false);
                add_edge(x - 1, n + y - 1, true);
                add_edge(n + x - 1, y - 1, true);
            }
            System.out.println(recur(0, n * 2, m));
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static void add_edge(int x, int y, boolean flag) {
        adj[x].add(y);
        if (!flag) {
            adj[y].add(x);
        }
    }

    public static long recur(int node, int parent, int count) {
        if (dp[node] != -1) {
            return dp[node];
        }
        int connected_components = 0;
        for (int u: adj[node]) {
            if (u != parent) {
                connected_components += recur(u, node, count);
            }
        }
        int potential_edges = connected_components * (connected_components - 1) / 2 + connected_components;
        int real_edges = count + potential_edges;
        long res = ((1L * real_edges * (real_edges - 1) / 2) % MOD) * pow2[n - connected_components - 1];
        res %= MOD;
        return dp[node] = res;
    }
}
