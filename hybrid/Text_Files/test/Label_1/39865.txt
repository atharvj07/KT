def can_transform(s, t, p):
    # First, check if s can be transformed into t by reordering and adding characters
    si, ti = 0, 0
    while si < len(s) and ti < len(t):
        if s[si] == t[ti]:
            si += 1
        ti += 1
    if si < len(s):
        return "NO"
    
    # If s can potentially be transformed into t, check if p has enough characters to do so
    combined = s + p
    combined_counts = {}
    t_counts = {}
    
    for char in combined:
        if char in combined_counts:
            combined_counts[char] += 1
        else:
            combined_counts[char] = 1
    
    for char in t:
        if char in t_counts:
            t_counts[char] += 1
        else:
            t_counts[char] = 1
    
    for char in t_counts:
        if char not in combined_counts or t_counts[char] > combined_counts[char]:
            return "NO"
    
    return "YES"

# Main
q = int(input())
for _ in range(q):
    s = input()
    t = input()
    p = input()
    print(can_transform(s, t, p))
