n = int(input())
prefixes = []
suffixes = []
for i in range(2*n-2):
    s = input()
    if len(s) == n-1:
        if s not in prefixes:
            prefixes.append(s)
        else:
            suffixes.append(s)
    else:
        if s not in suffixes:
            suffixes.append(s)
        else:
            prefixes.append(s)
print(''.join('P' if s in prefixes else 'S' for s in prefixes + suffixes))
