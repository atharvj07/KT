import java.util.Scanner;

public class ColoredCircle {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        String s = scanner.next();

        long[][] dp = new long[n][m + 1];

        // Base case: if m is 0, there's only one way to paint the circle
        for (int i = 0; i < n; i++) {
            dp[i][0] = 1;
        }

        // Iterate over the moves (m) and the starting positions (i)
        for (int i = 0; i < n; i++) {
            for (int j = 1; j <= m; j++) {
                // Move clockwise and counter-clockwise from the current position
                int clockwise = (i + 1) % n;
                int counterClockwise = (i - 1 + n) % n;

                // Check if the colors match the current move
                if (s.charAt(j - 1) == 'R') {
                    dp[i][j] = (dp[i][j] + dp[clockwise][j - 1] + dp[counterClockwise][j - 1]) % MOD;
                } else {
                    dp[i][j] = (dp[i][j] + dp[clockwise][j - 1] + dp[counterClockwise][j - 1] - 2 * dp[i][j - 1] + MOD) % MOD;
                }
            }
        }

        long result = 0;
        for (int i = 0; i < n; i++) {
            result = (result + dp[i][m]) % MOD;
        }

        System.out.println(result);
    }
}
