#include <algorithm>
#include <iostream>

int main() {
    std::string s;
    int k;
    std::cin >> s >> k;
    int l = 0;
    for (int i = 0, j; i < s.length(); ++i) {
        for (j = i + 1; j < s.length() && j - i <= k; ++j) {
            if (s[j] != s[j - i])
                break;
            l = std::max(l, j - i);
        }
    }
    std::cout << l * 2 << '\n';
}
