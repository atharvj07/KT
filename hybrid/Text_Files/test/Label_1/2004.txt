#include <bits/stdc++.h>
using namespace std;
const int maxn = 3e5+5;
vector<int> adj[maxn];
vector<bool> vis(maxn);
vector<int> ord;
vector<int> p(maxn);
vector<pair<int,int>> sp(maxn);

void dfs(int node) {
  vis[node] = true;
  for (auto u: adj[node]) {
    if (!vis[u]) dfs(u);
  }
  ord.push_back(node);
}

int main() {
  int n, k; cin >> n >> k;
  for (int i = 1; i <= n; ++i) {
    cin >> p[i];
    if (p[i] == 0) p[i] = n+1;
  }
  for (int i = 0; i < k; ++i) {
    int x, y; cin >> x >> y;
    adj[x].push_back(y);
    sp[i] = make_pair(x, y);
  }
  dfs(n+1);
  reverse(ord.begin(), ord.end());
  vector<bool> used(n+1);
  for (int i = 0; i < n; ++i) {
    int x = ord[i];
    if (used[x]) {
      cout << 0 << endl;
      return 0;
    }
    used[x] = true;
    int nx;
    for (int j = 0; j < k; ++j) {
      tie(nx, ignore) = sp[j];
      if (nx == x) {
        int y; tie(ignore,y) = sp[j];
        if (!used[y]) {
          cout << 0 << endl;
          return 0;
        }
      }
    }
    if (p[x] != n+1) {
      if (!used[p[x]]) {
        cout << 0 << endl;
        return 0;
      }
    }
  }
  for (auto u: ord) cout << u << ' ';
  cout << endl;
  return 0;
}
