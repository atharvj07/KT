#include <iostream>
using namespace std;
int main() {
  int n, m, c;
  int mis = 0, chr = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> m >> c;
    mis += m > c;
    chr += c > m;
  }
  if (mis > chr)
    cout << "Mishka";
  else if (chr > mis)
    cout << "Chris";
  else
    cout << "Friendship is magic!^^";
  return 0;
}
