import sys
import heapq

for testcase in range(int(input())):
    input()
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        a, b = map(int, input().split())
        adj[a].append(b)
    dist = [float('inf')] * (n + 1)
    q = [(0, 1, False)]
    while q:
        d, u, used = heapq.heappop(q)
        if dist[u] != float('inf'):
            continue
        dist[u] = d
        if not used:
            heapq.heappush(q, (d, u, True))
            for v in adj[u]:
                if dist[v] == float('inf'):
                    heapq.heappush(q, (d + 1, v, False))
    for x in dist[1:]:
        sys.stdout.write(str(x) + ' ')
    sys.stdout.write('\n')
