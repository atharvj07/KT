n = int(input())
a = list(map(int, input().split()))
dp = [1] * n
for i in range(1, n):
    for j in range(i):
        if a[i] - a[j] == 1 and dp[i] < dp[j] + 1:
            dp[i] = dp[j] + 1
l = max(dp)
r = []
for i in range(n - 1, -1, -1):
    if dp[i] == l:
        r.append(i + 1)
        l -= 1
print(r[-1] - r[0] + 1)
print(*r)
