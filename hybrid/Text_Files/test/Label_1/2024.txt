#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int solve(int n, vector<int>& p, int m, vector<int>& q) {
    int ans = 0;
    sort(p.begin(), p.end());
    sort(q.begin(), q.end());
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (p[i] >= q[j] || i > 0 && p[i] == p[i - 1]) {
                break;
            }
            int x = (p[i] + q[j]) / 2;
            if ((p[i] + q[j]) % 2 == 0 && x >= 0 && p[i] == x + p[i] - q[j]) {
                ans++;
            }
        }
    }
    return ans;
}
int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> p(n);
        for (int i = 0; i < n; i++) { cin >> p[i]; }
        int m; cin >> m;
        vector<int> q(m);
        for (int i = 0; i < m; i++) { cin >> q[i]; }
        cout << solve(n, p, m, q) << endl;
    }
    return 0;
}
