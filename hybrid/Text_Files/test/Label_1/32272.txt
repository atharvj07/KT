import java.util.*;
import java.math.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int N = Integer.parseInt(br.readLine());
        int[] X = new int[N];
        int[] Y = new int[N];
        char[] U = new char[N];
        for (int i = 0; i < N; i++) {
            String[] s = br.readLine().split(" ");
            X[i] = Integer.parseInt(s[0]);
            Y[i] = Integer.parseInt(s[1]);
            U[i] = s[2].charAt(0);
        }

        // Create a map that maps each direction to a list of (X, Y) coordinates.
        Map<Character, List<int[]>> map = new HashMap<>();
        map.put('U', new ArrayList<>());
        map.put('R', new ArrayList<>());
        map.put('D', new ArrayList<>());
        map.put('L', new ArrayList<>());
        for (int i = 0; i < N; i++) {
            map.get(U[i]).add(new int[] { X[i], Y[i] });
        }

        // Check if there is a pair of airplanes that will collide with each other.
        for (char direction : map.keySet()) {
            // Get the list of (X, Y) coordinates for the current direction.
            List<int[]> coordinates = map.get(direction);
            // Check if there is a pair of airplanes that are on a collision course.
            for (int[] p1 : coordinates) {
                for (int[] p2 : coordinates) {
                    if (p1 != p2) {
                        // Calculate the time it will take for the two airplanes to collide.
                        long dx = p1[0] - p2[0];
                        long dy = p1[1] - p2[1];
                        BigDecimal time = new BigDecimal(Math.sqrt(dx * dx + dy * dy) / 0.1);
                        // Check if the two airplanes will collide.
                        if (time.compareTo(new BigDecimal(0)) > 0) {
                            // Print the time it will take for the two airplanes to collide and exit the program.
                            System.out.println(time);
                            return;
                        }
                    }
                }
            }
        }

        // If there is no pair of airplanes that will collide, print "SAFE".
        System.out.println("SAFE");
    }
}
