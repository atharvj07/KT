#include<iostream>
#include<vector>
using namespace std;

int main(){
  string s;
  getline(cin,s);
  vector<int>a;
  int sum = 0;
  for(char c : s){
    if(isdigit(c))
      a.push_back(c-'0');
    else if(c=='+' || c=='-' || c=='='){
      a.push_back(c);
      int x = a.back();
      sum += x * (x=='='?-1:1);
      a.pop_back();
    }
    else
      a.push_back(0);
  }
  for(int i=0;i<a.size()-1;++i)
    if(a[i]==0)
      a[i] = (sum / a.size()) + ((abs(sum) + i) % a.size() ? 1 : 0);

  if(sum == 0){
    cout<<"Possible";
    for(int x : a)
      cout<<(x?' ':'+')<<x;
    cout<<endl;
  }
  else
    cout<<"Impossible"<<endl;
  return 0;
}
