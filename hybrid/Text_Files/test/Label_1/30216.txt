import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), bx = input.nextInt(), m = input.nextInt(), by = input.nextInt();
        long x = 0, y = 0;
        for (int i = 0; i < n; i++) {
            x = x * bx + input.nextInt();
        }
        for (int i = 0; i < m; i++) {
            y = y * by + input.nextInt();
        }
        System.out.println(x == y ? "=" : x < y ? "<" : ">");
    }
}
