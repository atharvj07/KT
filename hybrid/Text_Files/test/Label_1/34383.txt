import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        PriorityQueue<Integer> q = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            q.add(sc.nextInt());
        }
        int cost = 0;
        while (q.size() > 1) {
            int a = q.poll();
            int b = q.poll();
            cost += a + b;
            q.add(a + b);
        }
        System.out.println(cost);
    }
}
