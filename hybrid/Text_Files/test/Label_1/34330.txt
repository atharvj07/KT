import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            pq.add(sc.nextLong());
        }
        long ans = 0;
        while (pq.size() > 1) {
            long x = pq.poll();
            long y = pq.poll();
            pq.add(x + y);
            ans += x + y;
        }
        System.out.println(ans);
    }
}
