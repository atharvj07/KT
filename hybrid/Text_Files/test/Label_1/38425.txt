def min_xor_value(n, a):
    trie = {}
    for number in a:
        node = trie
        for i in range(29, -1, -1):
            bit = (number >> i) & 1
            if bit not in node:
                node[bit] = {}
            node = node[bit]
    
    def query(x):
        node = trie
        result = 0
        for i in range(29, -1, -1):
            bit = (x >> i) & 1
            if 1-bit in node:
                node = node[1-bit]
                result |= (1-bit) << i
            else:
                node = node[bit]
                result |= bit << i
        return result
    
    min_value = float('inf')
    for number in a:
        min_value = min(min_value, query(number) ^ number)
    return min_value

n = int(input())
a = list(map(int, input().split()))
print(min_xor_value(n, a))
