m, n = map(int, input().split());
a = [list(input()) for _ in range(m)]
ans = [[0]*n for _ in range(m)]
for i in range(m):
    for j in range(n):
        if a[i][j] == '*':
            q = [(i, j, 0)]; a[i][j] = '.'
            while q:
                x, y, d = q.pop(0)
                for dx, dy in ((0, 1), (0, -1), (-1, 0), (1, 0)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < m and 0 <= ny < n and a[nx][ny] == '.' and (nx, ny, d + 1) not in q:
                        q += [(nx, ny, d + 1)]
                        a[nx][ny] = '*'
                if d != 0:
                    ans[i][j] = str((ans[i][j] + d) % 10)
            a[i][j] = '*'
print(''.join(''.join(e for e in f) for f in a))
