import java.util.Scanner;

public class BinaryHeapProperties {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int H = scanner.nextInt();
        int[] keys = new int[H + 1];
        for (int i = 1; i <= H; i++) {
            keys[i] = scanner.nextInt();
        }

        for (int i = 1; i <= H; i++) {
            System.out.print("node " + i + ": key = " + keys[i] + ", ");
            int parent = i / 2;
            if (parent >= 1) {
                System.out.print("parent key = " + keys[parent] + ", ");
            }
            int left = 2 * i;
            if (left <= H) {
                System.out.print("left key = " + keys[left] + ", ");
            }
            int right = 2 * i + 1;
            if (right <= H) {
                System.out.print("right key = " + keys[right] + ", ");
            }
            System.out.println();
        }
    }
}
