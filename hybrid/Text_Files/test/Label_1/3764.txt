#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      if (i == j) continue;
      ll tmp = (ll)a[i] * pow(10, (int)log10(a[j]) + 1) + a[j];
      ans += tmp % k == 0;
    }
  }
  cout << ans << endl;
}
