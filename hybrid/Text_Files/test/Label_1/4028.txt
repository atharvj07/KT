#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<pair<int, int>> subarrays(m);
  for (int i = 0; i < m; i++) {
    cin >> subarrays[i].first >> subarrays[i].second;
  }

  sort(subarrays.begin(), subarrays.end());

  vector<int> a(n);
  int mex = 0;
  for (int i = 0; i < m; i++) {
    int l = subarrays[i].first - 1;
    int r = subarrays[i].second - 1;

    if (l == r) {
      if (a[l] == 0) {
        mex = 1;
      }
      continue;
    }

    while (l <= r && a[l] <= mex) {
      l++;
    }

    if (l <= r) {
      a[l] = mex;
    } else {
      a[l] = mex + 1;
      mex += 2;
    }
  }

  cout << mex << endl;

  for (int i = 0; i < n; i++) {
    cout << a[i] << " ";
  }
  cout << endl;

  return 0;
}
