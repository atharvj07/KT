#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int n, k;
    cin >> n >> k;
    vector<int> v(n);
    for(int &i : v){
        cin >> i;
    }
    sort(v.begin(), v.end());
    reverse(v.begin(), v.end());
    int x = 0;
    int ans = 0;
    for(const int i : v){
        int tmp = i;
        int cnt = 0;
        while(tmp >= 2){
            tmp /= 2;
            cnt++;
        }
        if(cnt >= ans){
            x++;
            ans = cnt;
        }
        if(x == k){
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
