#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> a(n);
        for (long long &x : a) cin >> x;
        bool ok = false;
        for (long long i = 2; i <= 1e9; i++) {
            bool valid = true;
            for (long long j = 0; j < n - 1; j++) {
                if ((a[j] % i != 0 && a[j + 1] % i != 0) ||
                    (a[j] % i == 0 && a[j + 1] % i == 0)) {
                    valid = false;
                }
            }
            if (valid) {
                cout << i << endl;
                ok = true;
                break;
            }
        }
        if (!ok) cout << 0 << endl;
    }
    return 0;
}
