#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> ans(n);
  for (int i = 0; i < n; i++) {
    ans[i] = 0;
    for (char c : a[i]) {
      if (c != '?') ans[i] *= 10, ans[i] += c - '0';
    }
  }
  int p = 0;
  for (int i = 0; i < n; i++) {
    while (p < n && ans[i] == ans[p]) p++;
    if (p == n) ans[i] = 0;
    p++;
  }
  if (ans[0] == 0) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  for (int x : ans) cout << x << endl;
}
