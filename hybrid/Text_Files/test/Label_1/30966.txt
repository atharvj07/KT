import java.util.*;
import java.lang.*;

class Main
{
    static void printMaxDist(int arr[], int n)
    {
        HashMap<Integer, Integer> lastIndex = new HashMap<>();
        int max = 0;
        for (int i = 0; i < n; i++)
        {
            int last = lastIndex.getOrDefault(arr[i], -1);
            if (last >= 0)
                max = Math.max(max, i - last - 1);
            lastIndex.put(arr[i], i);
        }
        System.out.println(max);
    }
    public static void main(String[] args)
    {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++)
        {
            int n = scanner.nextInt();
            int arr[] = new int[n];
            for (int j = 0; j < n; j++)
                arr[j] = scanner.nextInt();
            printMaxDist(arr, n);
        }
        scanner.close();
    }
}
