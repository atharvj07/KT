#include<bits/stdc++.h>
#define ll long long
using namespace std;
int n,a[200],ans=0;
int main()
{
    cin>>n;
    vector<int>adj[200];
    for(int i=2; i<=1<<n; i++) cin>>a[i];
    for(int i=1<<n; i>1; --i)
    {
        for(int j=i<<1; j<=(i<<1)+1; j++)
            adj[i].push_back(j);
    }
    queue<int>q;
    q.push(1);
    vector<int>dist;
    while(!q.empty())
    {
        int x=q.front();
        q.pop();
        dist.push_back(a[x]);
        for(int v:adj[x])
            q.push(v);
    }
    int mex=0;
    for(int x:dist)
    {
        if(mex==x) ++mex;
    }
    for(int x:dist) ans+=mex-x;
    cout<<ans<<endl;
}
