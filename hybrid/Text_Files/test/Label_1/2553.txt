#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s = to_string(n);
  int ans = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] % 2) {
      ans++;
      if ((i + 1 < s.size() && s[i + 1] % 2 == 0) || (i > 0 && s[i - 1] % 2 == 0))
        s.erase(i--, 1);
      else {
        cout << -1 << endl;
        return 0;
      }
    }
  }
  s.erase(remove(s.begin(), s.end(), '0'), s.end());
  cout << ans + s.size() / 2 << endl;
  return 0;
}
