#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, ans = 0; cin >> n;
    vector<pair<int, int>> a(n);
    for (auto& i : a) cin >> i.first, cin >> i.second;
    sort(a.rbegin(), a.rend());
    for (int i = 1, r = a[0].first + a[0].second; i < n; i++) {
        if (a[i].first > r) {
            ans++;
            r = max(a[i].first, r + a[i].second);
        }
    }
    cout << ans << '\n';
}
