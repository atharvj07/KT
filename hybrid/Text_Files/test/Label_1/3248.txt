#include <iostream>
using namespace std;

int main() {
  int n, ans = 0;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    ans += s[i] == 'x';
  }
  cout << (ans == n / 2 ? 0 : 1) << '\n';
  if (ans != n / 2) {
    for (int i = 0; i < n; i++) {
      s[i] = s[i] == 'x' ? 'X' : 'x';
      ans++;
      if (ans == n / 2) break;
    }
  }
  cout << s << '\n';
}
