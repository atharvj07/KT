import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] w = new long[n];
    for (int i = 0; i < n; i++) {
      w[i] = sc.nextLong();
    }
    int p = sc.nextInt();
    for (int i = 0; i < p; i++) {
      int a = sc.nextInt() - 1;
      int b = sc.nextInt();
      long sum = w[a];
      for (int j = a + b; j < n; j += b) {
        sum += w[j];
      }
      System.out.println(sum);
    }
  }
}
