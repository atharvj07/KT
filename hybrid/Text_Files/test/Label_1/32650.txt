import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int k = cin.nextInt();
        String s = cin.next();
        char[] ch = s.toCharArray();
        int i;
        for (i = 0; i < n && k > 0; i++)
            if (ch[i] > '0') {
                ch[i] = '0';
                k--;
            }
        int j;
        for (j = i, i = 0; i < n; i++, j++) {
            while (j < n && ch[j] == '0') j++;
            if (j == n) break;
            ch[i] = ch[j];
        }
        for (; i < n; i++) ch[i] = '0';
        System.out.println(String.valueOf(ch));
        cin.close();
    }
}
