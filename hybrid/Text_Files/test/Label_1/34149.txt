import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s1 = sc.nextLine(), s2 = sc.nextLine(), virus = sc.nextLine();
    int n1 = s1.length(), n2 = s2.length(), n3 = virus.length();
    int[][] dp = new int[n1 + 1][n2 + 1];
    for (int i = 1; i <= n1; i++) {
      for (int j = 1; j <= n2; j++) {
        if (s1.charAt(i - 1) == s2.charAt(j - 1) && (i <= n3 || j <= n3 || virus.charAt(i - 1) != s1.charAt(i - 1))) {
          dp[i][j] = dp[i - 1][j - 1] + 1;
        } else {
          dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
        }
      }
    }
    if (dp[n1][n2] == 0) {
      System.out.println(0);
    } else {
      int i = n1, j = n2;
      while (i > 0 && j > 0) {
        if (s1.charAt(i - 1) == s2.charAt(j - 1)) {
          System.out.print(s1.charAt(i - 1));
          i--;
          j--;
        } else {
          if (dp[i - 1][j] > dp[i][j - 1]) {
            i--;
          } else {
            j--;
          }
        }
      }
      System.out.println();
    }
  }
}
