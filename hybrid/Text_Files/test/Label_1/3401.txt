#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int N = 2e5 + 7;
ll a[N], b[N], c[N];

int main() {
    ll n, m;
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        cin >> a[i];
        c[i] = a[i];
    }
    for (ll i = 1; i <= n; i++) {
        cin >> b[i];
    }
    while (m--) {
        ll t;
        cin >> t;
        if (t == 1) {
            ll x, y, k;
            cin >> x >> y >> k;
            for (ll i = 0; i < k; i++) {
                b[y + i] += a[x + i];
            }
        } else {
            ll x;
            cin >> x;
            cout << b[x] << endl;
        }
    }
}
