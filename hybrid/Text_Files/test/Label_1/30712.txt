import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int totalPages = input.nextInt();
        int initialPages = input.nextInt();
        int maxPages = input.nextInt();
        int increase = input.nextInt();
        int pagesReread = input.nextInt();

        int currentPage = initialPages;
        int day = 1;

        while (currentPage < totalPages) {
            currentPage += Math.min(maxPages, totalPages - currentPage);
            currentPage -= Math.min(currentPage, pagesReread);
            day++;
            maxPages += increase;
        }

        System.out.println(day);
    }
}
