#include <bits/stdc++.h>
using namespace std;
int v[300001];
int main() {
  int n, p;
  cin >> n >> p;
  for (int i = 1; i <= n; i++) {
    int x, y;
    cin >> x >> y;
    if (x != y) {
      v[x]++;
      v[y]++;
    }
  }
  long long ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += (n - p) * 1ll * v[i];
    ans -= v[i] * 1ll * (v[i] - 1) / 2;
  }
  ans /= 2;
  cout << ans << endl;
  return 0;
}
