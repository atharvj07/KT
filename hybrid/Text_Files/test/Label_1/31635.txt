import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] m = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
            d[i] = i - m[i];
        }
        Arrays.sort(d);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += d[i];
        }
        System.out.println(ans);
    }
}
