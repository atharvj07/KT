#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MAXN = 5010;
const ll INF = 0x3f3f3f3f3f3f3f3f;
ll n, fa[MAXN], tot;
vector<ll> ans;
ll find(ll x) { return x == fa[x] ? x : fa[x] = find(fa[x]); }
ll merge(ll x, ll y) { fa[find(y)] = find(x); }
ll check(vector<ll> &a) {
  for (ll i = 1; i <= n; i++) fa[i] = i;
  for (ll i = 1; i < n; i++)
    if (a[i] == a[i + 1]) { tot = 1; return 0; }
  return 1;
}
int main() {
  ll T;
  cin >> T;
  while (T--) {
    cin >> n;
    ans.clear();
    vector<ll> a(n + 1);
    for (ll i = 1; i <= n; i++) cin >> a[i];
    tot = 0;
    if (!check(a)) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
      for (ll i = 1; i < n; i++) {
        if (find(i) != find(i + 1)) {
          ans.push_back(i);
          ans.push_back(i + 1);
          merge(i, i + 1);
        }
      }
      for (auto &x : ans) cout << x << " ";
      cout << endl;
    }
  }
  return 0;
}
