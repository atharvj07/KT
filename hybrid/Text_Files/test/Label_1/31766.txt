import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    private static int w, n, m;
    private static int[] a;
    private static long[] ans;
    private static String[] gates;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        w = Integer.parseInt(s[0]);
        n = Integer.parseInt(s[1]);
        m = Integer.parseInt(s[2]);
        a = new int[n];
        String[] t = br.readLine().split(" ");
        for (int i = 0; i < n; i++)
            a[i] = Integer.parseInt(t[i]);
        ans = new long[m];
        gates = new String[m];
        for (int i = 0; i < m; i++) {
            gates[i] = br.readLine();
            int[] gate = new int[w];
            for (int j = 0; j < w; j++) {
                char ch = gates[i].charAt(j);
                switch (ch) {
                    case 'A':
                        gate[j] = 0;
                        break;
                    case 'O':
                        gate[j] = 1;
                        break;
                    case 'X':
                        gate[j] = 2;
                        break;
                    case 'a':
                        gate[j] = 3;
                        break;
                    case 'o':
                        gate[j] = 4;
                        break;
                    case 'x':
                        gate[j] = 5;
                        break;
                }
            }
            for (int x = 0; x < n; x++)
                for (int y = 0; y < n; y++) {
                    int r = 0;
                    for (int j = 0; j < w; j++) {
                        int p = (a[x] >> j) & 1;
                        int q = (a[y] >> j) & 1;
                        switch (gate[j]) {
                            case 0:
                                r |= (p & q) << j;
                                break;
                            case 1:
                                r |= (p | q) << j;
                                break;
                            case 2:
                                r |= (p ^ q) << j;
                                break;
                            case 3:
                                r |= ((p & q) ^ 1) << j;
                                break;
                            case 4:
                                r |= ((p | q) ^ 1) << j;
                                break;
                            case 5:
                                r |= ((p ^ q) ^ 1) << j;
                                break;
                        }
                    }
                    if (r == 0)
                        ans[i]++;
                }
        }
        for (long an : ans)
            System.out.println(an);
    }
}
