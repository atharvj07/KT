#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll maxn = 3e5 + 5;
#define pb push_back

ll n, a[maxn], b[maxn], dp[maxn][3];
vector<ll> g[3];

ll solve(ll x, ll pre) {
    if (x > n) {
        return 0;
    }
    if (dp[x][pre] != -1) {
        return dp[x][pre];
    }
    ll ret = 0;
    if (pre != 1) {
        ret = max(ret, a[x] + solve(x + 1, 1));
    }
    if (pre != 2) {
        ret = max(ret, b[x] + solve(x + 1, 2));
    }
    if (!g[pre].empty() && g[pre].back() != x) {
        ret = max(ret, solve(x + 1, pre));
    }
    return dp[x][pre] = ret;
}
int main() {
    memset(dp, -1, sizeof(dp));
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; i++) {
        cin >> b[i];
    }
    g[1].pb(0);
    for (int i = 1; i <= n; i++) {
        if (a[i] > b[i]) {
            g[1].pb(i);
        } else if (a[i] < b[i]) {
            g[2].pb(i);
        } else {
            g[1].pb(i);
            g[2].pb(i);
        }
    }
    cout << solve(1, 0) << endl;
    return 0;
}
