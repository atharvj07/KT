import sys
board = []; b=[['W']*8 for i in range(8)]
for i in range(8):
    a = input().strip()
    board.append(list(a))

for i in range(8):
    for j in range(8):
        if board[i][j] == b[i][j] and (i + j)%2:
            b[i][j] = 'B'
for i in board:
    for j in i:
        if j == b[0][0]:
            print('YES')
            sys.exit()
print('NO')
