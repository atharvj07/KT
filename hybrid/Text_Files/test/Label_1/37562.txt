s = input()
t = input()
n = len(s)

diff_positions = [i for i in range(n) if s[i] != t[i]]

if len(diff_positions) % 2 != 0:
    print("impossible")
else:
    p = list(s)
    half_diff = len(diff_positions) // 2
    for i in range(half_diff):
        p[diff_positions[i]] = t[diff_positions[i]]
    print("".join(p))
