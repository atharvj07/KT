import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int ans = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += s[i];
            if (sum > 4) {
                ans++;
                sum = s[i];
            }
        }
        if (sum > 0) {
            ans++;
        }
        System.out.println(ans);
    }
}
