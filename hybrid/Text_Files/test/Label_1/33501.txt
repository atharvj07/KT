import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int count = 0;
        while (a % 2 == 0 && b % 2 == 0 && c % 2 == 0) {
            a /= 2;
            b /= 2;
            c /= 2;
            count++;
        }
        if (a % 2 == 0 || b % 2 == 0 || c % 2 == 0) {
            System.out.println(count);
        } else {
            System.out.println(-1);
        }
    }
}
