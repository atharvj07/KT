import java.util.*;
import java.math.*;

public class Main {
    static Scanner sc = new Scanner(System.in);
    static int m, t, r;

    public static void main(String[] args) {
        m = sc.nextInt(); t = sc.nextInt(); r = sc.nextInt();
        int[] w = new int[m];
        for(int i = 0;i < m;++i) w[i] = sc.nextInt();

        int ans = -1, l = 0;
        for(int i = 0;i < m;++i){
            int cur = (w[i] - l + t - 1) / t;
            if(cur + ans < r){
                ans = -1; break;
            }
            l = w[i] + 1;
        }

        System.out.println(ans);
    }
}
