import java.util.Scanner;

public class Main {
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        int[] x = new int[n];
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            a[i] = sc.nextInt();
        }
        long[] dp = new long[1 << n];
        dp[0] = 1;
        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) == 0) {
                    int pos = x[j];
                    while (pos < l) {
                        dp[i | (1 << j)] += dp[i];
                        dp[i | (1 << j)] %= MOD;
                        pos += a[j];
                    }
                }
            }
        }
        System.out.println(dp[(1 << n) - 1]);
    }
}
