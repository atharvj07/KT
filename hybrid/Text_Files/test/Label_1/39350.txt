def max_union_segments(t, test_cases):
    results = []
    for i in range(t):
        n, segments = test_cases[i]
        segments.sort()
        prefix = [0] * n
        suffix = [0] * n
        prefix[0] = 1
        for j in range(1, n):
            if segments[j][0] > segments[j-1][1]:
                prefix[j] = prefix[j-1] + 1
            else:
                prefix[j] = prefix[j-1]
        suffix[n-1] = 1
        for j in range(n-2, -1, -1):
            if segments[j][1] < segments[j+1][0]:
                suffix[j] = suffix[j+1] + 1
            else:
                suffix[j] = suffix[j+1]
        max_segments = 1
        for j in range(n):
            left_union = prefix[j-1] if j > 0 else 0
            right_union = suffix[j+1] if j < n-1 else 0
            if j > 0 and j < n-1 and segments[j-1][1] < segments[j+1][0]:
                max_segments = max(max_segments, left_union + right_union + 1)
            else:
                max_segments = max(max_segments, left_union + right_union)
        results.append(max_segments)
    return results

# Input reading and processing
t = int(input().strip())
test_cases = []
for _ in range(t):
    n = int(input().strip())
    segments = [list(map(int, input().strip().split())) for _ in range(n)]
    test_cases.append((n, segments))

# Function call and output printing
results = max_union_segments(t, test_cases)
for result in results:
    print(result)
