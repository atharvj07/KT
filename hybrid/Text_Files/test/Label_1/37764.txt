def find_s(t):
    count_a = t.count('a')
    if (len(t) - count_a) % 2 != 0:
        return ":("
    half_len = (len(t) - count_a) // 2
    s, s_prime = t[:half_len + count_a], t[half_len + count_a:]
    if s.replace('a', '') == s_prime:
        return s
    else:
        return ":("

t = input().strip()
print(find_s(t))
