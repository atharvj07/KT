#include <bits/stdc++.h>
using namespace std;
#define vi vector<int>
int32_t main() {
  int n, a, b;
  cin >> n >> a >> b;
  string s;
  cin >> s;
  vi id[2];
  for (int i = 1; i <= n; i++) {
    id[s[i - 1] - '0'].push_back(i);
  }
  if (id[0].empty() || id[1].empty()) {
    cout << -1 << endl;
    return 0;
  }
  int ans = (a <= b) ? b - a : a - b;
  cout << ans << endl;
}
