import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.List;
import java.util.Arrays;
 
class Main {
    private static final List<String> winningOrder = Arrays.asList("Rock", "Fire", "Scissors", "Snake", "Human", "Tree", "Wolf", "Sponge", "Paper", "Air", "Water", "Dragon", "Devil", "Lightning", "Gun");
    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        while(true){
            int n = Integer.parseInt(input.readLine());
            if(n == 0) break;
            String[] hands = new String[n];
            for(int i = 0; i < n; i++){
                hands[i] = input.readLine();
            }
            boolean[] defeated = new boolean[n];
            Arrays.fill(defeated, false);
            String winner = null;
            for(int i = 0; i < n; i++){
                for(int j = i + 1; j < n; j++){
                    int index1 = winningOrder.indexOf(hands[i]);
                    int index2 = winningOrder.indexOf(hands[j]);
                    if(index1 < index2){
                        defeated[j] = true;
                    }else if(index1 > index2){
                        defeated[i] = true;
                    }
                }
            }
            boolean draw = true;
            for(int i = 0; i < n; i++){
                if(!defeated[i]){
                    draw = false;
                    winner = hands[i];
                    break;
                }
            }
            if(draw) System.out.println("Draw");
            else System.out.println(winner);
        }
    }
}
