n, k = map(int, input().split())
a = sorted([(i, x) for i, x in enumerate(map(int, input().split()))])
if a[-1][1] - a[0][1] <= k:
    print(0, k)
    for i in range(k):
        print(a[-1][0] + 1, a[0][0] + 1)
        a[-1][1] -= 1
        a[0][1] += 1
    exit()
i = 0
ans = []
while k:
    if a[i][1] < a[i + 1][1] - 1:
        a[i][1] += 1
        a[i + 1][1] -= 1
        ans.append((a[i + 1][0] + 1, a[i][0] + 1))
        k -= 1
    i += 1
print(a[-1][1] - a[0][1], len(ans))
for i, j in ans:
    print(i, j)
