import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int ans = -1;
        int cntA = 0, cntB = 0;
        for (int i = 0; i < n; i++) {
            if (c[i] == a) cntA++;
            else if (cntA == 0) { ans = c[i]; break; }
        }
        for (int i = 0; i < n; i++) {
            if (c[i] == a) cntA++;
            else if (c[i] == ans) cntB++;
            if (cntB >= cntA) { ans = -1; break; }
        }
        System.out.println(ans);
    }
}
