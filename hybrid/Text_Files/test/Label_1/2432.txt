#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    v[i] = {a, i + 1};
  }
  sort(v.begin(), v.end());
  int ans = v[k - 1].first;
  cout << ans << endl;
  for (int i = 0; i < k; i++) {
    cout << v[i].second << " ";
  }
  return 0;
}
