#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(k);
    for (int i = 0; i < k; i++) cin >> a[i];
    int ans = 0;
    for (int x = 1; x <= n; x++) {
        for (int y = 1; y <= n; y++) {
            int now = x;
            bool flag = true;
            for (int i = 0; i < k; i++) {
                if (now == a[i]) {
                    if (x != y) {
                        now = y;
                    } else {
                        flag = false;
                        break;
                    }
                } else {
                    now += (a[i] > now);
                    now -= (a[i] < now);
                    if (now < 1 || now > n) {
                        flag = false;
                        break;
                    }
                }
            }
            ans += flag;
            ans %= mod;
        }
    }
    cout << ans << endl;
    return 0;
}
