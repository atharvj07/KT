n = int(input())
weights = list(map(int, input().split()))

weights.sort()
min_instability = float('inf')

for i in range(2 * n):
    for j in range(i + 1, 2 * n):
        temp_weights = weights[:i] + weights[i+1:j] + weights[j+1:]
        instability = sum(abs(temp_weights[k] - temp_weights[k+1]) for k in range(0, len(temp_weights) - 1, 2))
        min_instability = min(min_instability, instability)

print(min_instability)
