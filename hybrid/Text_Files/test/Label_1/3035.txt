#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, d, a[101], m;
    cin >> n >> d;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    cin >> m;
    int ans = 0;
    for (int i = 0; i < min(n, m); i++) ans += a[i];
    if (n < m) ans -= d * (m - n);
    cout << ans << endl;
    return 0;
}
