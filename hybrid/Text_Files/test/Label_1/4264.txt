#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k,i,j,ans[1005],a[1005],mx=0,x=0;
    cin>>n>>k;
    for(i=1;i<=n;i++)cin>>a[i];
    sort(a+1,a+n+1);
    for(i=n;i>=n-k+1;i--){
        if(a[i]>mx)mx=a[i];
        ans[x++]=i;
    }
    cout<<mx<<'\n';
    for(i=x-1;i>=0;i--)cout<<ans[i]<<" ";
    return 0;
}
