import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> m = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            m.put(s, m.getOrDefault(s, 0) + 1);
        }
        if (m.size() == 1) System.out.print("YES");
        else System.out.print("NO");
    }
}
