t = int(input())
for _ in range(t):
    n = int(input())
    segments = [list(map(int, input().split())) for _ in range(n)]
    left_max = max(segments, key=lambda x: x[0])[0]
    right_min = min(segments, key=lambda x: x[1])[1]
    if left_max <= right_min:
        print(0)
    else:
        print(min(n - sum(left_max <= r and l <= right_min for l, r in segments), n - 1))
