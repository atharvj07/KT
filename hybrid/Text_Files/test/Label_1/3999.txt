#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(k);
    for (int& x : a) cin >> x;
    for (int& x : b) cin >> x;
    sort(b.begin(), b.end());
    bool ok = false;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            if (i == 0 || a[i - 1] <= b[0]) {
                ok = true;
            } else break;
        } else if (i > 0 && a[i] < a[i - 1]) break;
    }
    cout << (ok ? "Yes" : "No") << endl;
}
