import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length();
        boolean[][] dp = new boolean[n][n];
        int[][] pal = new int[n][n];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = i; j < n; j++) {
                if (i == j) {
                    dp[i][j] = true;
                    pal[i][j] = 1;
                } else {
                    if (s.charAt(i) == s.charAt(j) && ((j - i <= 2) || dp[i + 1][j - 1])) {
                        dp[i][j] = true;
                        if (i + 1 < n && j - 1 >= 0 && pal[i + 1][j - 1] > 0) {
                            pal[i][j] = pal[i + 1][j - 1] + 2;
                        } else {
                            pal[i][j] = 2;
                        }
                    }
                }
            }
        }
        if (pal[0][n - 1] == 100) {
            int i = 0, j = n - 1;
            StringBuilder sb = new StringBuilder();
            while (i <= j) {
                sb.append(s.charAt(i));
                i++;
                j--;
            }
            System.out.println(sb.toString());
        } else {
            int len = 0;
            int l = -1, r = -1;
            for (int i = 0; i < n; i++) {
                for (int j = i; j < n; j++) {
                    if (pal[i][j] > len && dp[i][j]) {
                        len = pal[i][j];
                        l = i;
                        r = j;
                    }
                }
            }
            StringBuilder sb = new StringBuilder();
            for (int i = l; i <= r; i++) {
                sb.append(s.charAt(i));
            }
            System.out.println(sb.toString());
        }
    }
}
