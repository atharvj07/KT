import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            int min = 0;
            for (int j = 1; j < n; j++) {
                if (a[j] < a[j - 1]) {
                    min = Math.max(min, (a[j - 1] - a[j] + 1) / 2);
                }
            }
            System.out.println(min);
        }
    }
}
