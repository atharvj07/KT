import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] h = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = in.nextInt();
        }
        Arrays.sort(h);
        int k = 0;
        int res = 0;
        for (int i = 0; i < n - 1; i++) {
            if (h[i + 1] - h[i] <= 1) {
                k++;
                res = Math.max(res, k);
            } else {
                k = 0;
            }
        }
        if (h[n - 1] - h[0] <= 1) {
            res = Math.max(res, n);
        }
        System.out.println(res);
        if (res == n) {
            for (int i = 0; i < n; i++) {
                System.out.print(h[i] + " ");
            }
        } else {
            int j = 0;
            for (int i = 0; i < n - 1; i++) {
                if (h[i + 1] - h[i] <= 1) {
                    j++;
                } else {
                    break;
                }
            }
            if (h[n - 1] - h[j] <= 1) {
                for (int i = j; i < n; i++) {
                    System.out.print(h[i] + " ");
                }
                for (int i = 0; i <= j; i++) {
                    System.out.print(h[i] + " ");
                }
            } else {
                for (int i = 0; i <= j; i++) {
                    System.out.print(h[i] + " ");
                }
            }
        }
        System.out.println();
    }
}
