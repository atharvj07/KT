import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList());
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            graph.get(a).add(c);
            graph.get(b).add(c);
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int[] visited = new int[m + 1];
            dfs(u, v, visited, graph);
            int ans = 0;
            for (int j = 1; j <= m; j++) {
                if (visited[j] == 2) {
                    ans++;
                }
            }
            System.out.println(ans);
        }
    }

    public static void dfs(int u, int v, int[] visited, List<List<Integer>> graph) {
        visited[0] = 1;
        if (u == v) {
            visited[0] = 2;
            return;
        }
        for (int c : graph.get(u)) {
            if (visited[c] == 0) {
                visited[c] = 1;
                dfs(v, u, visited, graph);
                if (visited[0] == 2) {
                    visited[c] = 2;
                    return;
                }
            }
        }
    }
}
