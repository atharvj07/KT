import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();

            long x = findMaxBits(l, r);
            System.out.println(x);
        }
    }

    private static long findMaxBits(long l, long r) {
        long x = l;
        long maxBits = Long.bitCount(l);

        for (long i = l + 1; i <= r; i++) {
            long currentBits = Long.bitCount(i);

            if (currentBits > maxBits) {
                maxBits = currentBits;
                x = i;
            } else if (currentBits == maxBits) {
                x = Math.min(x, i);
            }
        }

        return x;
    }
}
