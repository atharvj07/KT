#include <iostream>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string a, b;
    std::cin >> a >> b;
    bool ok = true;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) {
        if (a[i] == '1') cnt++;
        else cnt--;
      }
      if (cnt < 0 || cnt > i + 1) ok = false;
    }
    if (cnt == 0) std::cout << "YES\n";
    else std::cout << "NO\n";
  }
  return 0;
}
