import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long l = in.nextLong();
        long r = in.nextLong();
        long ans = r - l + 1;
        String max = String.valueOf(r);
        char[] c = max.toCharArray();
        for (int i = 0; i < c.length; i++) {
            if (c[i] != '0') {
                ans--;
                break;
            }
        }
        System.out.println(ans);
    }
}
