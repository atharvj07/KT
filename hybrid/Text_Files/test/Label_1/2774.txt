#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int cnt[10] = {};
    for (char c : s) {
      cnt[c - '0']++;
    }
    int odd = 0;
    for (int i = 0; i < 10; i++) {
      if (cnt[i] % 2) odd++;
    }
    cout << (odd + 1) / 2 << '\n';
  }
  return 0;
}
