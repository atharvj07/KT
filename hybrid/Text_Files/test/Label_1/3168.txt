#include <iostream>
#include <string>
using ull = unsigned long long;
const ull mod = 998244353;
ull powm(ull base, ull p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  ull r = powm(base, p / 2);
  r = (r * r) % mod;
  if (p % 2) r = (r * base) % mod;
  return r;
}
std::string a;
int main() {
  ull l, r;
  std::cin >> a >> l >> r;
  ull ans = 0, p = 1;
  for (ull i = 0; i < a.size(); ++i) {
    int d = a[i] - '0';
    if (i == a.size() - 1) {
      if (l <= d && d <= r) ans += p;
    } else {
      if (d >= l) ans += p;
      if (d > 0) p = (p * 10) % mod;
    }
  }
  std::cout << ans % mod << std::endl;
}
