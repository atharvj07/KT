import math

def gcd_of_array(arr):
    gcd = arr[0]
    for num in arr[1:]:
        gcd = math.gcd(gcd, num)
    return gcd

def count_divisors(n):
    count = 0
    for i in range(1, int(math.sqrt(n)) + 1):
        if n % i == 0:
            if n / i == i:
                count += 1
            else:
                count += 2
    return count

n = int(input())
arr = list(map(int, input().split()))
gcd = gcd_of_array(arr)
print(count_divisors(gcd))
