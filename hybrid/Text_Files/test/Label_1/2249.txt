#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m, mi, ma;
  cin >> n >> m >> mi >> ma;
  vector<int> t(m), t2;
  t2.reserve(n);
  for (int i = 0; i < m; i++) cin >> t[i];
  int has[101] = {0}, cnt = 0;
  for (int x : t) has[x] = 1;
  for (int i = mi; i <= ma; i++) if (!has[i]) cnt++;
  int add = 0;
  if (cnt + m < n)
    cout << "Incorrect";
  else {
    for (int x : t)
      if (x == mi || x == ma) { add++; }
      else
        t2.push_back(x);
    for (int i = mi; i <= ma; i++)
      if (!has[i] && add <= n - m) { t2.push_back(i); }
    cout << "Correct";
  }
}
