#include <iostream>
#include <unordered_set>

int main() {
    std::unordered_set<char> unique_chars;
    char c;
    std::cin.get(c);  // skip '{'
    while ((c = std::cin.get()) != '}') {
        if (c != ',') unique_chars.insert(c);
    }
    std::cout << unique_chars.size() << '\n';
    return 0;
}
