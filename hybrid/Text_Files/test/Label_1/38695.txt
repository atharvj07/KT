from fractions import Fraction
def binpow(x, n, mod):
    if n == 0: return 1
    if n == 1: return x % mod
    if n % 2 == 0:
        return binpow(x, n//2, mod)**2 % mod
    else:
        return x * binpow(x, n-1, mod) % mod

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
ans = 0
for i in range(n):
    if a[i] == b[i] == 0:
        ans += (m-1) * (m-2) * binpow(m, n-i-1, 10**9+7)
    elif a[i] == 0:
        ans += (m-1) * (m-b[i]-1) * binpow(m, n-i-1, 10**9+7)
    elif b[i] == 0:
        ans += (m-a[i]-1) * (m-1) * binpow(m, n-i-1, 10**9+7)
print(ans % (10**9+7))
