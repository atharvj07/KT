#include <iostream>
#include <algorithm>
using namespace std;
int main() {
  int n, m, a[100001], b[100001], x[100001], y[100001], ans = 0;
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i] >> x[i]; }
  cin >> m;
  for (int i = 0; i < m; i++) { cin >> b[i] >> y[i]; }
  sort(a, a + n);
  sort(b, b + m);
  int l1 = 0, l2 = 0, c1 = 0, c2 = 0;
  while (l1 < n && l2 < m) {
    if (a[l1] == b[l2]) {
      l1++;
      l2++;
    } else if (a[l1] < b[l2]) {
      if (!c1) ans += x[l1], c1 = 1;
      l1++;
    } else {
      if (!c2) ans += y[l2], c2 = 1;
      l2++;
    }
  }
  while (l1 < n) { if (!c1) ans += x[l1], c1 = 1; l1++; }
  while (l2 < m) { if (!c2) ans += y[l2], c2 = 1; l2++; }
  cout << ans << endl;
}
