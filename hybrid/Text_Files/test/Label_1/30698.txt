import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int k = 0;
        Set<Integer> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            if (!set.contains(a[i])) {
                set.add(a[i]);
                k++;
            }
        }
        System.out.println(k);
        for (int i : set) {
            System.out.print(i + " ");
        }
    }
}
