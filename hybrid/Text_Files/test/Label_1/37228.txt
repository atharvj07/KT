import sys

t = int(input())
for _ in range(t):
    n = int(input())
    mask = []
    for _ in range(n):
        mask.append(tuple(map(int, input().split())))
    ans = 'YES' if any(all(mask[i][j] == mask[k][j] for i in range(n) for k in range(n) if i != k and mask[i][j] == 1) for j in range(5)) else 'NO'
    sys.stdout.write(ans + '\n')
