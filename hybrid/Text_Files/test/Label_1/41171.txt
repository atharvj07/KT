n=int(input())
a=list(map(int,input().split()))
ans=[]
for s in range(1,n+1):
    t=n//s
    if n%s==0:
        ok=1
        b=[0]*s
        for i in range(n):
            b[i%s]+=a[i]==1
            if b[i%s]>t:
                ok=0
                break
        if ok:
            ans.append([s,t])
print(len(ans))
for x in ans:
    print(*x)
