#include<iostream>
#include<vector>
#include<algorithm>
int main() {
 int t; std::cin >> t;
 while(t--) {
  int n; std::cin >> n;
  std::vector<int> a(n);
  for(auto& w: a) std::cin >> w;
  std::sort(a.begin(), a.end());
  int l = 0, r = n - 1, res = 0, sum = 0;
  while(l < r) {
   sum = a[l] + a[r];
   if(sum < a[l + 1] + a[r - 1]) {
    l++;
   } else if(sum > a[l + 1] + a[r - 1]) {
    r--;
   } else {
    res++;
    l++;
    r--;
   }
  }
  std::cout << res << '\n';
 }
}
