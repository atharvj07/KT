#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> b(n);
    for (int i = 0; i < n; i++) { cin >> b[i]; }
    bool flag = true;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
      if (b[i] % n != 0) {
        flag = false;
      }
      a[i] = b[i] / n;
    }
    if (flag) {
      cout << "YES" << endl;
      for (long long x : a) { cout << x << " "; }
      cout << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
