#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    long long n;
    cin >> n;
    bool vis[10] = {false};
    int cnt = 0;
    while (!(n == 0 || n == 1)) {
        long long tmp = n;
        while (tmp > 0) {
            if (!vis[tmp % 10]) {
                vis[tmp % 10] = true;
                cnt++;
            }
            tmp /= 10;
        }
        if (n % 10 == 0) {
            n /= 10;
        } else {
            n++;
        }
    }
    cout << cnt + 1 << endl;
    return 0;
}
