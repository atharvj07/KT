import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int L = sc.nextInt();
        long sum = 0;
        for (int i = 1; i <= N; i++) {
            sum += L + i - 1;
        }
        long minDiff = Long.MAX_VALUE;
        for (int i = 1; i <= N; i++) {
            long diff = Math.abs(sum - (L + i - 1) * 2);
            if (diff < minDiff) {
                minDiff = diff;
            }
        }
        System.out.println(sum - minDiff);
    }
}
