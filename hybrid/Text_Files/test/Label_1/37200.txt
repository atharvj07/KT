import sys
input = sys.stdin.readline
def dot(a, b): return sum([x * y for x, y in zip(a, b)])
n = int(input())
ants = [[int(x) for x in input().split()] for i in range(n)]
mq = int(input())
queries = [[int(x) for x in input().split()] for j in range(mq)]

for query in queries:
    dist = [(query[i]-ants[i][0])**2+(query[i]-ants[i][1])**2 for i in range(n)]
    dist.sort()
    cmb = [[]]
    for i in range(n):
        nxt = []
        for s in cmb:
            nxt.append(s + [dist[i]])
            nxt.append(s + [-(dist[i])])
        cmb = nxt
    ans = []
    for c in cmb:
        x, y = 0, 0
        for d in c:
            x += d
            y += -d**0.5
        if x == 0 and y.is_integer():
            ans.append([int(y), 0])
    ans.sort()
    print(len(ans))
    print(*ans)
