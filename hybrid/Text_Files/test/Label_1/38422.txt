import sys

input = sys.stdin.readline

mod = 1000000007
n, k = map(int,input().split())
a = []
for i in range(n):
    l,r = map(int,input().split())
    a.append((l,r))
a.sort()
ans = 0
for i in range(n):
    l1,r1 = a[i]
    for j in range(i+1,n):
        l2,r2 = a[j]
        l = max(l1,l2)
        r = min(r1,r2)
        if l<=r:
            ans = (ans + r - l + 1) % mod
print(ans)
