import sys

mod = 10 ** 9 + 7
fact = [1 for i in range(72)]

for i in range(2, 72):
  fact[i] = (fact[i - 1] * i) % mod

def power(x, y):
  if y == 0:
    return 1
  elif y == 1:
    return x
  elif y % 2 == 0:
    return power((x ** 2) % mod, y // 2)
  else:
    return (x * power(x, y - 1)) % mod

n = int(sys.stdin.readline())
arr = [int(x) for x in sys.stdin.readline().split()]

count = [0 for i in range(72)]
for x in arr:
  count[x] += 1

ans = 0
for i in range(1, 72):
  if count[i] > 0:
    ans = (ans + (fact[count[i] + count[i % 35]] * power((fact[count[i]] * fact[count[i % 35]]) % mod, mod - 2)) % mod)

print(ans)
