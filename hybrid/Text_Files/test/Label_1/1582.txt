#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    long long arr[n];
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    sort(arr, arr + n);
    long long ans = arr[n - 1] - arr[0];
    for (int i = 0; i < n - 1 && k > 0; i++) {
        if (arr[i + 1] - arr[i] <= k) {
            ans -= arr[i + 1] - arr[i];
            k -= arr[i + 1] - arr[i];
            arr[i + 1] = arr[i];
        } else {
            ans -= k;
            k = 0;
        }
    }
    cout << ans << endl;
    return 0;
}
