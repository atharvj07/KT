import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    static int[] indegree, outdegree;
    static List<Edge>[] adj;
    static int n, p;
    static int[] tank, tap;
    static int[] diameter;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        p = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) { adj[i] = new ArrayList<>(); }
        diameter = new int[p + 1];
        indegree = new int[n + 1];
        outdegree = new int[n + 1];
        for (int i = 1; i <= p; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int d = sc.nextInt();
            diameter[i] = d;
            adj[a].add(new Edge(b, i));
            indegree[b]++;
            outdegree[a]++;
        }
        init();
        int t = 0;
        for (int i = 1; i <= n; i++) {
            if (indegree[i] == 0 && outdegree[i] > 0) {
                tank[t] = i;
                tap[t] = -1;
                t++;
            }
            if (indegree[i] > 0 && outdegree[i] == 0) {
                tank[t] = -1;
                tap[t] = i;
                t++;
            }
        }
        System.out.println(t);
        for (int i = 0; i < t; i++) {
            int max = Integer.MIN_VALUE;
            if (tank[i] != -1) {
                for (Edge e : adj[tank[i]]) {
                    max = Math.max(diameter[e.id], max);
                }
                System.out.println(tank[i] + " " + tap[i] + " " + max);
            }
        }
    }

    static void init() {
        tank = new int[n + 1];
        tap = new int[n + 1];
        Arrays.fill(tank, -1);
        Arrays.fill(tap, -1);
    }

    static class Edge {
        int to;
        int id;

        Edge(int to, int id) {
            this.to = to;
            this.id = id;
        }
    }
}
