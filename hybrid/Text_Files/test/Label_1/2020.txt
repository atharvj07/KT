#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int Pow(int x, int y) {
	int ans = 1;
	x = x % mod;
	while (y > 0) {
		if (y & 1) {
			ans = ((ans % mod) * (x % mod)) % mod;
		}
		x = ((x % mod) * (x % mod)) % mod;
		y >>= 1;
	}
	return ans;
}
int f[1000001] = {1};
int main() {
	int n; cin >> n;
	for (int i = 1; i <= n; i++) {
		f[i] = f[i - 1] * i;
		f[i] %= mod;
	}
	if (n % 2 == 1) {
		cout << 0 << endl;
		return 0;
	}
	n /= 2;
	int res = Pow(2, n);
	res = res * res;
	res %= mod;
	int val = f[n];
	val = Pow(val, mod - 2);
	val %= mod;
	res = res * val;
	res %= mod;
	cout << res << endl;
	return 0;
}
