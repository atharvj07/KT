import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static final long MOD = 1_000_000_007;

    private static int n, k;
    private static List<List<Edge>> adj;
    private static long[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        adj = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            int x = sc.nextInt();
            adj.get(u).add(new Edge(v, x));
            adj.get(v).add(new Edge(u, x));
        }
        dp = new long[n][k + 1];
        long ans = dfs(0, 0, -1);
        System.out.println(ans);
    }

    private static long dfs(int u, int steps, int prev) {
        if (steps == k) {
            return 1;
        }
        if (dp[u][steps] != 0) {
            return dp[u][steps];
        }
        long ans = 0;
        for (Edge e : adj.get(u)) {
            int v = e.to;
            int color = e.color;
            if (v == prev) {
                continue;
            }
            if (color == 1 || steps > 0) {
                ans = (ans + dfs(v, steps + 1, u)) % MOD;
            }
        }
        return dp[u][steps] = ans;
    }

    private static class Edge {
        int to;
        int color;

        public Edge(int to, int color) {
            this.to = to;
            this.color = color;
        }
    }
}
