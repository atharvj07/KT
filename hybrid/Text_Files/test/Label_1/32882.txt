import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long r = sc.nextLong();
        if ((r - l) < 2) {
            System.out.println(-1);
        } else if ((r - l) <= 5) {
            System.out.println(l + 1 + " " + (l + 1) + " " + (l + 2));
        } else {
            System.out.println(l + 2 + " " + (l + 3) + " " + (l + 4));
        }
    }
}
