#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;
const int max_n = 2e5 + 10;

ll n, m, ans;
vector<ll> a;
vector<tuple<ll, ll, ll>> e;
vector<ll> p;

ll find(ll node) {
    if (p[node] == node) return node;
    return p[node] = find(p[node]);
}

ll join(ll x, ll y) {
    x = find(x);
    y = find(y);
    p[x] = y;
}

int main() {
    cin >> n >> m;
    a.resize(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    e.resize(m);
    for (int i = 0; i < m; i++) {
        ll v, u, w;
        cin >> v >> u >> w;
        v--;
        u--;
        e[i] = {w, v, u};
    }
    sort(e.rbegin(), e.rend());
    ans = 0;
    p.resize(n);
    for (int i = 0; i < n; i++) p[i] = i;
    for (int i = 0; i < m; i++) {
        ll w, v, u;
        tie(w, v, u) = e[i];
        if (find(v) != find(u)) {
            join(v, u);
            ans += w - a[v] - a[u];
        }
    }
    cout << ans << endl;
    return 0;
}
