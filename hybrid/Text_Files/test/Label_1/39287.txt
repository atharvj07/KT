def solve():
    s = input()
    openLeft = closeRight = 0
    ans = 0
    MOD = 10 ** 9 + 7
    for ch in s:
        if ch == '(':
            openLeft += 1
            ans += closeRight
        else:
            closeRight += 1
            ans += openLeft
        ans %= MOD
    print(ans)

solve()
