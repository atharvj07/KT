import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n * n];
        for (int i = 0; i < n * n; i++) {
            arr[i] = sc.nextInt();
        }
        int[] gcd = new int[n];
        for (int i = 0; i < n; i++) {
            gcd[i] = arr[i * n + i];
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            ans[i] = gcd[i];
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                ans[i] = gcd(ans[i], gcd[j]);
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
