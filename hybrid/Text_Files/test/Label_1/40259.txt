def find_parent(parents, x):
    if parents[x] != x:
        parents[x] = find_parent(parents, parents[x])
    return parents[x]

def union(parents, ranks, a, b):
    rootA = find_parent(parents, a)
    rootB = find_parent(parents, b)
    
    if rootA != rootB:
        if ranks[rootA] < ranks[rootB]:
            rootA, rootB = rootB, rootA
        parents[rootB] = rootA
        if ranks[rootA] == ranks[rootB]:
            ranks[rootA] += 1

def solve(n, m, roads):
    parents = [i for i in range(n+1)]
    ranks = [0] * (n+1)
    separate_cities = n
    
    for x, y in roads:
        if find_parent(parents, x) != find_parent(parents, y):
            separate_cities -= 1
            union(parents, ranks, x, y)
    
    return separate_cities

n, m = map(int, input().split())
roads = [list(map(int, input().split())) for _ in range(m)]
print(solve(n, m, roads))
