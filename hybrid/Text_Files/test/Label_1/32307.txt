import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = sc.nextInt();
        sc.nextLine();
        int[] d = new int[26];
        Arrays.fill(d, -1);
        for (int i = 0; i < n; i++) {
            String p = sc.nextLine();
            d[p.charAt(0) - 'a'] = p.charAt(1) - 'a';
            d[p.charAt(1) - 'a'] = p.charAt(0) - 'a';
        }
        int ans = 0;
        for (int i = 1; i < s.length(); i++) {
            if (d[s.charAt(i) - 'a'] == s.charAt(i - 1) - 'a') {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
