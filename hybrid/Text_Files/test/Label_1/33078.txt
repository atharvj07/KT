import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();

        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += arr[i];
        }
        if (sum != 0) {
            System.out.println(-1);
            return;
        }

        int ans = 0;
        ArrayList<Integer> neg = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (arr[i] < 0) {
                neg.add(i);
            }
        }

        for (int i = 0; i < neg.size(); i++) {
            int idx = neg.get(i);
            int val = arr[idx];
            if (val == 0) {
                continue;
            }
            int pos = (idx + 1) % n;
            while (val < 0) {
                arr[idx] = 0;
                arr[pos] += val;
                ans++;
                val++;
                pos = (pos + 1) % n;
            }
        }

        System.out.println(ans);
    }
}
