import java.util.Scanner;

public class Tree {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        if (isValidTree(c)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static boolean isValidTree(int[] c) {
        int n = c.length;
        int[] degree = new int[n];
        for (int i = 0; i < n; i++) {
            degree[i] = c[i] - 1;
        }

        for (int i = 0; i < n; i++) {
            if (degree[i] == 0) {
                return false; // No root node
            }
            if (degree[i] > 1) {
                for (int j = 0; j < n; j++) {
                    if (degree[j] == 0 && i != j) {
                        degree[i]--;
                        degree[j]++;
                        break;
                    }
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (degree[i] != 1) {
                return false; // Not all internal nodes have at least two sons
            }
        }

        return true;
    }
}
