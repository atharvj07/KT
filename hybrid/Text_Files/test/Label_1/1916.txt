#include <bits/stdc++.h>
#define MAXN (1<<17)
using namespace std;
int n;
int t[MAXN];
int f[MAXN][2];
int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) {
    cin >> t[i];
  }
  memset(f, -1, sizeof(f));
  f[1][0] = f[1][1] = 1;
  for (int i = 2; i <= n; ++i) {
    f[i][0] = f[i - 1][0];
    if (t[i - 1] >= t[i]) f[i][0] = max(f[i][0], f[i - 1][1] + 1);
    f[i][1] = f[i - 1][0];
    if (t[i-1]<=t[i]) f[i][1] = max(f[i][1], f[i - 1][1] + 1);
  }
  cout << max(f[n][0], f[n][1]) << endl;
  return 0;
}
