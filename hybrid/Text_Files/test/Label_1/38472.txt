for _ in range(int(input())):
    s = list(input())
    a, b = 0, 0
    for c in s:
        if c == 'a':
            a += 1
        else:
            b += 1
    cur = s[0]
    ans = ''
    ch = 'a' if a > b else 'b'
    for i in range(len(s)):
        if cur == s[i]:
            ans += cur
            continue
        if cur == 'a':
            ans += 'b'
            a -= 1
            b += 1
        else:
            ans += 'a'
            b -= 1
            a += 1
        cur = s[i]
    print(ans)
