  int n, a[100005];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  int t = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 25)
      t++;
    else if (a[i] == 50) {
      if (t > 0)
        t--;
      else {
        cout << "NO";
        return 0;
      }
    } else {
      if (t > 0)
        t--;
      else {
        cout << "NO";
        return 0;
      }
      if (t > 0)
        t--;
      else {
        cout << "NO";
        return 0;
      }
    }
  }

  return 0;
}
