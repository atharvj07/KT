#include <iostream>
#include <cstdio>

using namespace std;

int main() {
  int n;
  cin >> n;
  char a[n][n];
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++) cin >> a[i][j];
  int ans = 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      int cnt = 0;
      if (a[i][j] == 'o') continue;
      if ((i - 1 >= 0 && a[i - 1][j] == 'o')) cnt++;
      if ((i + 1 < n && a[i + 1][j] == 'o')) cnt++;
      if ((j - 1 >= 0 && a[i][j - 1] == 'o')) cnt++;
      if ((j + 1 < n && a[i][j + 1] == 'o')) cnt++;
      if (cnt == 0) ans = 0;
    }
  }
  ans ? cout << "YES\n" : cout << "NO\n";
  return 0;
}
