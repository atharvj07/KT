#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
typedef long long ll;
using namespace std;
ll gcd(ll x, ll y) {
	if (y == 0) return x;
	return gcd(y, x % y);
}
int main() {
	ll a, b, c, g, p, q;
	cin >> a >> b >> c;
	g = gcd(b, c);
	p = a / (b / g);
	q = a / (c / g);
	g = gcd(p, q);
	p /= g;
	q /= g;
	cout << p << '/' << q << endl;
	return 0;
}
