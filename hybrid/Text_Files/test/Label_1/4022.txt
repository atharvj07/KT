#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> a(k);
  for (int i = 0; i < k; i++) cin >> a[i];

  vector<int> output(k);
  int curr = 1;
  for (int i = 0; i < k; i++) {
    for (long long j = 0; j < a[i] % n; j++) { curr++; }
    output[i] = curr;
    curr++;
    curr %= n;
  }

  for (int i = 0; i < k; i++) cout << output[i] << endl;
  return 0;
}
