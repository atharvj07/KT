import java.util.*;

public class TreeDecomposition {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] edges = new int[n - 1][2];
        for (int i = 0; i < n - 1; i++) {
            edges[i][0] = scanner.nextInt() - 1;
            edges[i][1] = scanner.nextInt() - 1;
        }

        List<int[]> decomposition = decomposeTree(n, edges);

        if (decomposition == null) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            System.out.println(decomposition.size());
            for (int[] path : decomposition) {
                System.out.println((path[0] + 1) + " " + (path[1] + 1));
            }
        }
    }

    private static List<int[]> decomposeTree(int n, int[][] edges) {
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int[] edge : edges) {
            graph.get(edge[0]).add(edge[1]);
            graph.get(edge[1]).add(edge[0]);
        }

        int[] degree = new int[n];
        for (int i = 0; i < n; i++) {
            degree[i] = graph.get(i).size();
        }

        List<int[]> decomposition = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (degree[i] == 1) {
                int neighbor = graph.get(i).get(0);
                degree[i]--;
                degree[neighbor]--;
                decomposition.add(new int[]{i, neighbor});
            }
        }

        for (int i = 0; i < n; i++) {
            if (degree[i] > 2) {
                return null;
            }
        }

        return decomposition;
    }
}
