mod = int(1e9) + 7
def fastpow(x, p):
    if p == 0: return 1
    if p == 1: return x
    val = fastpow(x, int(p / 2)) % mod
    val = (val * val) % mod
    if p % 2 != 0: val = (val * x) % mod
    return val

n, l, r = map(int, input().split())
if (r - l + 1) % 3 == 0 and n % 3 == 0:
    a = (r - l + 1) // 3
    b = n // 3
    ans = fastpow(3, b)
    ans = (ans * fastpow(a, b)) % mod
    print(ans)
else: print(0)
