b = []
for x in a:
    if not b or x > b[-1]:
        b += x,
i = 0
ans = ''
for x in a:

        if (i == 0 or x > ans[-1]) and (i == len(b)-1 or x < ans[-1]):
            ans += 'L'
            i += 1
