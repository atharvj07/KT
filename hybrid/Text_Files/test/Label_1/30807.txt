import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) break;

            int[][] files = new int[n][2];
            for (int i = 0; i < n; i++) {
                files[i][0] = sc.nextInt();
                files[i][1] = sc.nextInt();
            }

            Arrays.sort(files, (a, b) -> b[1] - a[1]);

            int cnt = 0;
            int size = 0;
            for (int[] file : files) {
                if (size + file[1] <= m) {
                    size += file[1];
                    cnt++;
                }
            }

            System.out.println(cnt == n ? cnt : "Impossible");
        }
    }
}
