#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m = 0, cnt = 0;
  vector<int> t;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    t.push_back(x);
    if (i > 0 && x - m > 15) {
      cnt = 0;
    }
    cnt++;
    if (cnt >= 15) break;
    m = x;
  }
  cout << m << endl;
}
