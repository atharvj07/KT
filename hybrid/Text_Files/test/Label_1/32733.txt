import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static class Event implements Comparable<Event> {
        int x, y1, y2, type;
        public Event(int x, int y1, int y2, int type) {
            this.x = x;
            this.y1 = y1;
            this.y2 = y2;
            this.type = type;
        }
        @Override
        public int compareTo(Event that) {
            if (x == that.x) return type - that.type;
            return x - that.x;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        Event[] events = new Event[2 * (n + m)];
        int idx = 0;
        for (int i = 0; i < n; i++) {
            int x1 = sc.nextInt(), x2 = sc.nextInt(), y = sc.nextInt();
            events[idx++] = new Event(x1, y, y, 1);
            events[idx++] = new Event(x2 + 1, y, y, -1);
        }
        for (int j = 0; j < m; j++) {
            int x = sc.nextInt(), y1 = sc.nextInt(), y2 = sc.nextInt();
            events[idx++] = new Event(x, y1, y2, 1);
            events[idx++] = new Event(x, y2 + 1, y1, -1);
        }
        Arrays.sort(events);
        int cnt = 0, ans = 0;
        for (Event event : events) {
            if (event.type == 1) cnt++;
            else cnt--;
            if (cnt == 1) ans += event.y2 - event.y1 + 1;
        }
        if (ans == 0) System.out.println("INF");
        else System.out.println(ans);
    }
}
