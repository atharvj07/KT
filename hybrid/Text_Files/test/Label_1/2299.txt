#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 998244353;
int pow(int a, int b) {
    int ret = 1;
    for (; b; b >>= 1) {
        if (b & 1) { ret = ret * a % mod; }
        a = a * a % mod;
    }
    return ret;
}
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(2 * n);
    for (int i = 0; i < 2 * n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    reverse(a.begin() + n, a.end());
    int sum = 0, ans = 0;
    for (int i = 0; i < 2 * n; i++) {
        sum += a[i], ans += sum;
    }
    ans -= 2 * a[n - 1] * pow(a[n] - a[n - 1], n - 1) % mod;
    cout << (ans + mod) % mod << endl;
}
