#include <bits/stdc++.h>

using namespace std;

int n, m;

int dx[]{-1, 1, 0, 0};
int dy[]{0, 0, -1, 1};

bool vis[505][505];
bool chk[505][505];
char grid[505][505];

bool valid(int x, int y) {
  return (x >= 1 && x <= n && y >= 1 && y <= m);
}

bool dfs(int x, int y) {
  if (x == n && y == m) return true;
  vis[x][y] = true;
  bool ret = false;
  for (int k = 0; k < 4; k++) {
    int nx = x + dx[k], ny = y + dy[k];
    if (valid(nx, ny) && grid[nx][ny] != 'X') {
      if (!vis[nx][ny] && dfs(nx, ny)) ret = true;
      if (vis[nx][ny] && chk[nx][ny]) ret = true;
    }
  }
  chk[x][y] = true;
  return ret;
}

int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> grid[i][j];
    }
  }
  int x1, y1, x2, y2;
  cin >> x1 >> y1 >> x2 >> y2;
  if (grid[x1][y1] == '.') { cout << "NO" << endl; } else {
    if (dfs(x1, y1))
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
