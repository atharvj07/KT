n, m = map(int, input().split())
plants = [tuple(map(float, input().split())) for _ in range(n)]

species_indices = [[] for _ in range(m + 1)]
for i, (s, _) in enumerate(plants):
    species_indices[int(s)].append(i)

# Longest Increasing Subsequence (LIS) problem
# Transform problem into finding LIS of plant positions (their indices in the input list),
# grouped by species, to minimize the number of plants that need to be replanted.

def find_LIS(sequence):
    # Dynamic programming approach for LIS
    dp = [1] * len(sequence)
    for i in range(len(sequence)):
        for j in range(i):
            if sequence[i] > sequence[j]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)

# Concatenate indices to form the sequence for LIS problem
indices_sequence = []
for species in species_indices[1:]:
    indices_sequence.extend(species)

# Number of plants to replant is total plants minus length of LIS
replant_min = n - find_LIS(indices_sequence)
print(replant_min)
