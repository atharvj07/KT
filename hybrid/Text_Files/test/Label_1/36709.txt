import math

def game(n):
    ans = 0
    sq = int(math.sqrt(n))
    for i in range(2, sq + 1):
        if n % i == 0:
            ans ^= 1
            while n % i == 0:
                n //= i
    if n > 1:
        ans ^= 1
    if ans == 0:
        return "Bob"
    return "Alice"

for _ in range(int(input())):
    n = int(input())
    print(game(n))
