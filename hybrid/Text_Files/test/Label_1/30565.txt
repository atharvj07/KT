import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[][] dp = new int[n][2];
        dp[0][0] = Math.max(0, a[0]);
        dp[0][1] = Math.max(0, -a[0]);
        for (int i = 1; i < n; i++) {
            for (int j = 0; j <= 1; j++) {
                if (i >= k) {
                    if (j == 0) {
                        dp[i][j] = Math.max(dp[i - 1][j], dp[i - k][1] + Math.max(0, a[i]));
                    } else {
                        dp[i][j] = Math.max(dp[i - 1][j], dp[i - k][0] + Math.max(0, -a[i]));
                    }
                } else {
                    dp[i][j] = dp[i - 1][j];
                }
            }
        }

        System.out.println(Math.max(dp[n - 1][0], dp[n - 1][1]));
    }
}
