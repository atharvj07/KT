#include <iostream>
#include <unordered_set>
using namespace std;

int main() {
  int n;
  string home;
  cin >> n >> home;
  unordered_set<string> i, v;
  for (int j = 0; j < n; ++j) {
    string a, b;
    cin >> a >> b;
    if (a == home)
      v.insert(b);
    else if (b == home)
      i.insert(a);
  }
  cout << (i == v ? "home" : "contest") << endl;
}
