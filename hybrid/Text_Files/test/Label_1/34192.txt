import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashSet;
import java.util.Set;

public class Main {
    static final int[] dr = {0, 0, 1, 1};
    static final int[] dc = {0, 1, 1, 0};

    public static void main(String[] args) throws IOException {
        try (var br = new BufferedReader(new InputStreamReader(System.in))) {
            var rc = br.readLine().split(" ");
            int r = Integer.parseInt(rc[0]), n = Integer.parseInt(rc[1]);
            Set<Long> reclaimed = new HashSet<>();
            for (int i = 0; i < n; i++) {
                var pos = br.readLine().split(" ");
                int rr = Integer.parseInt(pos[0]), cc = Integer.parseInt(pos[1]);
                reclaimed.add(((long) rr << 32) | cc);
                for (int j = 0; j < 4; j++) {
                    int tr = rr + dr[j], tc = ((cc - 1 + 2) % 2) + dc[j];
                    reclaimed.add(((long) tr << 32) | tc);
                }
            }

            if (((r & 1) ^ (n & 1)) == 0) {
                System.out.println("WIN");
            } else {
                System.out.println("LOSE");
            }
        }
    }
}
