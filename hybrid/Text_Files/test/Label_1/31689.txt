import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n;
        while ((n = scanner.nextInt()) != 0) {
            int[] ages = new int[n];
            for (int i = 0; i < n; i++) {
                ages[i] = scanner.nextInt();
            }
            int[] counts = new int[7];
            for (int age : ages) {
                if (age < 10) {
                    counts[0]++;
                } else if (age < 20) {
                    counts[1]++;
                } else if (age < 30) {
                    counts[2]++;
                } else if (age < 40) {
                    counts[3]++;
                } else if (age < 50) {
                    counts[4]++;
                } else if (age < 60) {
                    counts[5]++;
                } else {
                    counts[6]++;
                }
            }
            for (int count : counts) {
                System.out.println(count);
            }
        }
        scanner.close();
    }
}
