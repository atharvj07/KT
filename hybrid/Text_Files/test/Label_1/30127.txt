import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int ans = 0;
            Stack<Character> st = new Stack<>();
            for (int i = 0; i < n; i++) {
                if (!st.isEmpty() && st.peek() == s.charAt(i))
                    st.pop();
                else
                    st.push(s.charAt(i));
            }
            System.out.println(st.size());
        }
    }
}
