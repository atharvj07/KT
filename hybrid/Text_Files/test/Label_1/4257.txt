#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;

long long binpow(long long x, long long y) {
    long long ans = 1;
    while (y) {
        if (y & 1) ans = ans * x % MOD;
        x = x * x % MOD;
        y >>= 1;
    }
    return ans;
}

int main() {
    long long n, k;
    cin >> n >> k;
    long long ans = 1;
    for (long long L = 1, R; L <= n; L = R + 1) {
        R = n / (n / L);
        if (R - L + 1 >= k) {
            ans = (ans * binpow(n / L, k)) % MOD;
            break;
        } else {
            k -= R - L + 1;
        }
    }
    cout << ans << '\n';
    return 0;
}
