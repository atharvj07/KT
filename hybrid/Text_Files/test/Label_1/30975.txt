import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        for (int i = 1; i <= k; i++) {
            pq.add(a[i]);
        }
        int mx = -1;
        int mn = Integer.MAX_VALUE;
        for (int i = k + 1; i <= n; i++) {
            pq.add(a[i]);
            mx = Math.max(mx, pq.peek());
            mn = Math.min(mn, pq.peek());
            pq.remove(a[i - k]);
        }
        if (q == 1) {
            System.out.println(0);
        } else {
            System.out.println(mx - mn);
        }
    }
}
