n = int(input().strip())
a = list(map(int, input().strip().split()))

a.sort()

prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sum[i] = prefix_sum[i - 1] + a[i - 1]

def weakness(x):
    modified = [ai - x for ai in a]
    prefix_mod = [0] * (n + 1)
    for i in range(1, n + 1):
        prefix_mod[i] = prefix_mod[i - 1] + modified[i - 1]
    min_prefix = 0
    max_weakness = 0
    for i in range(1, n + 1):
        max_weakness = max(max_weakness, abs(prefix_mod[i] - min_prefix))
        min_prefix = min(min_prefix, prefix_mod[i])
    return max_weakness

left, right = -10000, 10000
while right - left > 1e-7:
    mid1 = left + (right - left) / 3
    mid2 = right - (right - left) / 3
    if weakness(mid1) < weakness(mid2):
        right = mid2
    else:
        left = mid1

print(f'{weakness(left):.15f}')
