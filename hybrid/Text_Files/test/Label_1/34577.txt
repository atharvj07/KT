import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int m = sc.nextInt();
        int a = sc.nextInt();

        int[] v = new int[n];
        int[] t = new int[n];

        for (int i = 0; i < a; i++) {
            int x = sc.nextInt() - 1;
            v[x]++;
            t[x] = i + 1;
        }

        List<Integer> res = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            if (v[i] >= k) res.add(1);
            else if (v[i] + (m - a) >= k) res.add(2);
            else res.add(3);
        }

        for (int x : res) System.out.print(x + " ");
    }
}
