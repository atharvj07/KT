#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

int main() {
    ll m, n;
    cin >> m >> n;
    vector<vector<ll>> t(m, vector<ll>(n));
    for (auto &i : t) {
        for (auto &j : i)
            cin >> j;
    }
    for (ll i = 0; i < m; i++) {
        ll cur = 0;
        for (ll j = 0; j < n; j++) {
            cur += t[i][j];
            cout << cur << ' ';
        }
    }
    return 0;
}
