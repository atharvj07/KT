import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;

public class BlogsNetwork {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        List<Integer>[] graph = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            graph[a].add(b);
            graph[b].add(a);
        }

        line = reader.readLine().split(" ");
        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = Integer.parseInt(line[i - 1]);
        }

        int[] order = solve(graph, desiredTopics);
        if (order == null) {
            System.out.println(-1);
        } else {
            for (int i = 1; i <= n; i++) {
                System.out.print(order[i] + " ");
            }
        }
    }

    private static int[] solve(List<Integer>[] graph, int[] desiredTopics) {
        int n = graph.length - 1;
        int[] result = new int[n + 1];
        boolean[] visited = new boolean[n + 1];
        Queue<Integer> queue = new LinkedList<>();

        for (int i = 1; i <= n; i++) {
            if (graph[i].size() == 0) {
                queue.offer(i);
                visited[i] = true;
            }
        }

        int[] availableTopics = new int[n + 1];
        Arrays.fill(availableTopics, 1);

        while (!queue.isEmpty()) {
            int current = queue.poll();
            result[current] = desiredTopics[current];
            visited[current] = true;

            for (int neighbor : graph[current]) {
                if (!visited[neighbor]) {
                    for (int topic : graph[neighbor]) {
                        if (visited[topic]) {
                            availableTopics[desiredTopics[topic]] = 0;
                        }
                    }
                    int nextTopic = findAvailableTopic(availableTopics);
                    if (nextTopic == -1) {
                        return null;
                    }
                    queue.offer(neighbor);
                    visited[neighbor] = true;
                    availableTopics[nextTopic] = 0;
                }
            }
        }

        return result;
    }

    private static int findAvailableTopic(int[] availableTopics) {
        for (int i = 1; i < availableTopics.length; i++) {
            if (availableTopics[i] == 1) {
                return i;
            }
        }
        return -1;
    }
}
