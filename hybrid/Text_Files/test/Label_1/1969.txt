#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
int main() {
    string s;
    cin >> s;
    int n = s.length();
    vector<ll> dp(n + 1, 0);
    dp[n] = 1;
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '?') {
            if (i + 1 < n && s[i + 1] == '0') {
                dp[i] = (2 * dp[i + 1]) % mod;
            } else {
                dp[i] = (dp[i + 1] + dp[i + 2]) % mod;
            }
        } else if (s[i] >= '0' && s[i] <= '2') {
            if (i + 1 < n && s[i + 1] == '0') {
                dp[i] = dp[i + 1];
            } else if (s[i] - '0' + i + 1 < n && s[i + s[i] - '0' + 1] == '0') {
                dp[i] = dp[i + 1];
            } else {
                dp[i] = dp[i + s[i] - '0' + 1];
            }
        } else {
            dp[i] = 0;
        }
    }
    cout << dp[0] << endl;
    return 0;
}
