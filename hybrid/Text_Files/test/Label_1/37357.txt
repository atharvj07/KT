import sys

n = int(input())
pts = [list(map(int, input().split())) for _ in range(n)]

def dist2(p1, p2):
    d2 = 0
    for i in range(5):
        d2 += (p2[i] - p1[i]) ** 2
    return d2

ans = []
for i in range(n):
    bad = False
    for j in range(n):
        if i == j: continue
        for k in range(n):
            if i == k or j == k: continue
            d1 = dist2(pts[i], pts[j])
            d2 = dist2(pts[i], pts[k])
            d3 = dist2(pts[j], pts[k])
            theta = d1 * d2 + d2 * d3 + d3 * d1
            theta = theta / ((d1 * d2 * d3) ** 0.5)
            if theta > 0:
                bad = True
                break
        if bad:
            break
    if not bad:
        ans.append(i+1)

print(len(ans))
for i in ans:
    print(i)
