#include <bits/stdc++.h>
using namespace std;
#define mod(a, b) (a % b + b) % b
int main() {
  long long y, k, n;
  cin >> y >> k >> n;
  long long ans = -1;
  for (long long x = 1; x * 2 <= n; x++) {
    long long a = mod(x + y, k);
    long long b = mod(n, k);
    if (a == b) { ans = x; }
  }
  cout << ans << '\n';
}
