#include <bits/stdc++.h>
#define vi vector<int>
using namespace std;
const int N = 2e5 + 10;
int a[N];
int main(void) {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vi b(2 * n + 10);
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      b[a[i]]++;
    }
    int ans = 0;
    vi v;
    for (int i = 1; i <= n; i++) {
      if (!b[a[i]])
        continue;
      int w = true;
      for (int j = a[i] + 1; j <= 2 * n; j++) {
        if (b[j]) {
          w = false;
          break;
        }
      }
      if (w) {
        ans++;
        v.push_back(i);
      }
    }
    cout << ans << endl;
    for (int i : v) {
      cout << i << ' ';
    }
    cout << endl;
  }
}
