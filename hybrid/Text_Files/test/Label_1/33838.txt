import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), bx = sc.nextInt(), m = sc.nextInt(), by = sc.nextInt();
        Integer[] x = new Integer[n], y = new Integer[m];
        for (int i = 0; i < n; i++)
            x[i] = sc.nextInt();
        for (int i = 0; i < m; i++)
            y[i] = sc.nextInt();
        long numX = toDecimal(x, bx), numY = toDecimal(y, by);
        System.out.println(numX < numY ? '<' : numX == numY ? '=' : '>');
    }

    public static long toDecimal(Integer[] a, int b) {
        long num = 0;
        for (int i = a.length - 1, f = 1; i >= 0; i--, f *= b)
            num += f * a[i];
        return num;
    }
}
