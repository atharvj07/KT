import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[][] dp = new int[n][2];
        dp[0][0] = a[0];
        for (int i = 1; i < n; i++) {
            dp[i][0] = dp[i - 1][0] + a[i];
            int max = 0;
            for (int j = 2; j <= i; j++) {
                max = Math.max(max, -dp[i - j][1] + dp[i][0] - dp[i - j][0]);
            }
            dp[i][1] = max;
        }

        System.out.println(dp[n - 1][1]);
    }
}
