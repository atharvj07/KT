import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a[] = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = in.nextInt();
        int dp[] = new int[n];
        dp[0] = 1;
        int max = 1, ind = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (a[i] > a[j])
                    dp[i] = Math.max(dp[i], dp[j] + 1);
            }
            if (dp[i] > max) {
                max = dp[i];
                ind = i;
            }
        }
        System.out.println(max);
        List<Integer> ans = new ArrayList<>();
        while (ind >= 0) {
            ans.add(a[ind]);
            ind--;
            max--;
            if (max == 0)
                break;
        }
        Collections.reverse(ans);
        for (int i : ans)
            System.out.print(i + " ");
        System.out.println();
    }
}
