import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class VertexAssignment {
    static class Edge {
        int from, to;

        public Edge(int from, int to) {
            this.from = from;
            this.to = to;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] p = new int[N];
        for (int i = 0; i < N; i++) {
            p[i] = sc.nextInt() - 1;
        }

        List<Edge>[] edges = new List[N];
        for (int i = 0; i < N; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < N; i++) {
            edges[p[i]].add(new Edge(p[i], i));
        }

        boolean[] visited = new boolean[N];
        int[] color = new int[N];
        Arrays.fill(color, -1);

        if (!dfs(0, visited, color, edges)) {
            System.out.println("IMPOSSIBLE");
            return;
        }

        System.out.println("POSSIBLE");
    }

    static boolean dfs(int v, boolean[] visited, int[] color, List<Edge>[] edges) {
        visited[v] = true;
        for (Edge edge : edges[v]) {
            int to = edge.to;
            if (color[to] != -1) {
                if (color[to] == color[v]) {
                    return false;
                }
            } else {
                color[to] = 1 - color[v];
                if (!dfs(to, visited, color, edges)) {
                    return false;
                }
            }
        }
        return true;
    }
}
