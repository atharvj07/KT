from collections import deque
from heapq import heappush, heappop

t = int(input())
for _ in range(t):
    n = int(input())
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)
    dist = [[float('inf') for _ in range(n+1)] for _ in range(n+1)]
    dist[1][1] = 0
    pq = [(0, 1)]
    while pq:
        d, node = heappop(pq)
        if d > dist[node][1]:
            continue
        for nei in adj[node]:
            if d + 1 < dist[nei][1]:
                dist[nei][1] = d + 1
                heappush(pq, (d+1, nei))
    ans = 0
    for i in range(1, n+1):
        ans = max(ans, dist[i][1])
    print(ans)
