#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=2e5+10;
int n;
ll q[N], l[N];
int main() {
  cin>>n;
  for(int i=1; i<=n; i++) cin>>q[i];
  l[n]=q[n];
  for(int i=n-1; i; i--) l[i]=l[i+1]|q[i];
  int ans=0;
  for(int i=1; i<n; i++) ans+=(l[i+1]>q[i]);
  cout<<ans<<endl;
  return 0;
}
