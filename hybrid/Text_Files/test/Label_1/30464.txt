import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] count = new int[m + 1];
        for (int i = 0; i < n; i++) {
            count[sc.nextInt()]++;
        }
        int ans = 1;
        for (int i = 1; i <= m; i++) {
            ans = (ans * choose(count[i] + k - 1, k)) % 1009;
        }
        System.out.println(ans);
    }

    private static int choose(int n, int k) {
        int res = 1;
        for (int i = n; i > n - k; i--) {
            res = (res * i) % 1009;
        }
        return res;
    }
}
