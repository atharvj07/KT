#include<bits/stdc++.h>
using namespace std;
 
int main() {
    int n;
    cin >> n;
    vector<bool> vroad(n + 1, false);
    vector<bool> hroad(n + 1, false);
    for(int i = 0; i < n * n; ++i) {
        int h, v;
        cin >> h >> v;
        if(!vroad[v]) vroad[v] = true, hroad[h] = true;
    }
    for(int i = 1; i <= n; ++i) if(vroad[i] || hroad[i]) cout << i << ' ';
    cout << endl;
    return 0;
}
