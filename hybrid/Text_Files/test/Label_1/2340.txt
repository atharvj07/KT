#include <bits/stdc++.h>
using namespace std;
map<int,int> mx,mn;
int main() {
    int t; cin >> t;
    while(t--) {
        int n, k; cin >> n >> k;
        bool flg = 1;
        for(int i=1;i<=n;i++) {
            int x; cin >> x;
            if(mn.count(x) || mx.count(x)) flg=0;
            else mn[x]=i, mx[x]=i;
        }
        for(auto it:mn) {
            if(mn[it.first]!=mx[it.first]) flg=0;
        }
        if(k==1 || flg) cout << "YES\n";
        else cout << "NO\n";
        mn.clear(), mx.clear();
    }
}
