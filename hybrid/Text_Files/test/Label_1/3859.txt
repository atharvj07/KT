#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=5005;
ll a[maxn],f[maxn][maxn],pre[maxn];
int main()
{
	int n,m,k;
	cin>>n>>m>>k;
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
		pre[i]=pre[i-1]+a[i];
	}
	for(int i=1;i<=k;i++)
		for(int j=i*m;j<=n;j++)
		{
			f[i][j]=max(f[i][j],f[i][j-1]);
			int l=max(j-m+1,1);
			f[i][j]=max(f[i][j],f[i-1][l-1]+pre[j]-pre[l-1]);
		}
	cout<<f[k][n]<<endl;
	return 0;
}
