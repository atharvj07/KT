import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] r = new int[h];
        int[] c = new int[w];
        for (int i = 0; i < h; i++) r[i] = sc.nextInt();
        for (int i = 0; i < w; i++) c[i] = sc.nextInt();
        long[][] dp = new long[h + 1][w + 1];
        dp[0][0] = 1;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if (dp[i][j] == 0) continue;
                if (i != h - 1 && r[i] == r[i + 1]) dp[i + 1][j] += dp[i][j];
                else dp[i + 1][j] = dp[i][j];
                if (j != w - 1 && c[j] == c[j + 1]) dp[i][j + 1] += dp[i][j];
                else dp[i][j + 1] = dp[i][j];
            }
        }
        System.out.println(dp[h][w] % 1000000007);
    }
}
