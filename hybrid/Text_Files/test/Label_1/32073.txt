import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int c = in.nextInt();
        int d = in.nextInt();
        int e = in.nextInt();
        int f = in.nextInt();
        System.out.println(Math.max(a / 2 * e + Math.min(b, c) * f, (d - Math.min(a, d) / 2) * e + Math.min(a, d) / 2 * f));
    }
}
