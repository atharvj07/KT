import java.util.Scanner;

class Main {

    private static int[][] movements = {
            {1, 1}, {2, 2}, {1, 3},
            {2, 3}, {3, 3}, {3, 2},
            {4, 2}, {5, 2}, {4, 3},
            {3, 4}
    };

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] dp = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dp[i][j] = -1;
            }
        }
        String number = scanner.next();
        boolean unique = true;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (unique && dp[i][j] == -1) {
                    int result = check(i, j, number, dp);
                    if (result == -1) {
                        unique = false;
                        break;
                    }
                }
            }
        }
        System.out.println(unique ? "YES" : "NO");
    }

    private static int check(int i, int j, String number, int[][] dp) {
        if (i == j) {
            return 0;
        }
        int result = dp[i][j];
        if (result != -1) {
            return result;
        }
        result = number.charAt(i) != number.charAt(j) ? -1 : 0;
        int[] movementI = movements[number.charAt(i) - '0'];
        int[] movementJ = movements[number.charAt(j) - '0'];
        result += movementI[0] == movementJ[0] || movementI[1] == movementJ[1] ? 0 : -1;
        if (i + 1 < j) {
            int temp = check(i + 1, j, number, dp);
            if (temp != -1) {
                result += temp;
            } else {
                result = -1;
            }
        }
        if (result != -1 && i < j - 1) {
            int temp = check(i, j - 1, number, dp);
            if (temp != -1) {
                result += temp;
            } else {
                result = -1;
            }
        }
        dp[i][j] = dp[j][i] = result;
        return result;
    }
}
