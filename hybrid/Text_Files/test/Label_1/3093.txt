#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
vector<int>arr, t;
int n, q;
void build(int node, int start, int end) {
  if(start == end) { t[node] = arr[start]; return; }
  int m = (start + end) / 2;
  build(node * 2, start, m);
  build(node * 2 + 1, m + 1, end);
  t[node] = max(t[node * 2], t[node * 2 + 1]);
}
int query(int node, int start, int end, int l, int r, int k) {
  if(start > r || end < l) return 0;
  if(start >= l && end <= r) {
    int cnt = t[node] - t[node * 2];
    if(cnt >= k) return query(node * 2, start, (start + end) / 2, l, r, k);
    else return query(node * 2 + 1, (start + end) / 2 + 1, end, l, r, k - cnt);
  }
  int a = query(node * 2, start, (start + end) / 2, l, r, k);
  int b = query(node * 2 + 1, (start + end) / 2 + 1, end, l, r, k);
  return max(a, b);
}
void update(int node, int start, int end, int idx, int val) {
  if(start == end) { t[node] = val; return; }
  int m = (start + end) / 2;
  if(idx <= m) update(node * 2, start, m, idx, val);
  else update(node * 2 + 1, m + 1, end, idx, val);
  t[node] = max(t[node * 2], t[node * 2 + 1]);
}
int main() {
  ios::sync_with_stdio(0), cin.tie(0);
  cin >> n >> q;
  arr.resize(n + 1); t.resize(4 * n + 4);
  for(int i = 1; i <= n; i++) cin >> arr[i];
  build(1, 1, n);
  while(q--) {
    int type, x; cin >> type >> x;
    if(type == 1) {
      if(arr[x] == 1) arr[x] = 0, update(1, 1, n, x, 0);
      else arr[x] = 1, update(1, 1, n, x, 1);
    } else {
      cout << query(1, 1, n, 1, n, x) << '\n';
    }
  }
  return 0;
}
