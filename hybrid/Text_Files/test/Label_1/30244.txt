import java.util.*;

public class FindKey {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] ai = new int[n];
        for (int i = 0; i < n; i++) {
            ai[i] = scanner.nextInt();
        }

        int[] bi = new int[n];
        for (int i = 0; i < n; i++) {
            bi[i] = ai[i];
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (gcd(ai[i], ai[j]) != 1) {
                    int temp = bi[i];
                    bi[i] = bi[j];
                    bi[j] = temp;
                }
            }
        }

        for (int i : bi) {
            System.out.print(i + " ");
        }
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
