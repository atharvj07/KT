import sys
input = sys.stdin.readline

n, k, x = map(int, input().split())
a = sorted(int(x) for x in input().split())
i, ans, mx = 0, 0, 0
for j in range(n):
    while i < n and mx - a[i] <= x:
        mx = max(mx, a[i])
        i += 1
    if i - j > 0:
        ans += 1
    if i == n:
        break
print(ans + max(0, (len(a) + k - i)//(i - j)))
