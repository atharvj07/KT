import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        int sum = 0;
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
        }
        int avg = sum / N;
        int dist = 101;
        int ans = -1;
        for (int i = 0; i < N; i++) {
            if (Math.abs(a[i] - avg) < dist) {
                dist = Math.abs(a[i] - avg);
                ans = i;
            }
        }
        System.out.println(ans);
    }
}
