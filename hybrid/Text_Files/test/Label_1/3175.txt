#include <bits/stdc++.h>

using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
const int mx = 2e5 + 5;

void solve() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n + 5);
    map<pair<int, int>, int> mp;
    for (int i = 1; i <= n; i++) {
        int x, y;
        cin >> x >> y;
        v[i] = {x, y};
        mp[{x, y}]++;
    }
    ll cnt1 = 0, cnt2 = 0, cnt3 = 0;
    for (int i = 1; i <= n; i++) {
        cnt1 += (mp[v[i]] == 1);
    }
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (v[i].first == v[j].first) cnt2++;
            if (v[i].second == v[j].second) cnt3++;
        }
    }
    ll ans = cnt1 * (cnt1 - 1) * (cnt1 - 2) / 6;
    ans += cnt2 * (cnt2 - 1) * (cnt2 - 2) / 6;
    ans += cnt3 * (cnt3 - 1) * (cnt3 - 2) / 6;
    cout << ans << endl;
}

int main() {
    int t;
    cin >> t;
    while (t--) { solve(); }
    return 0;
}
