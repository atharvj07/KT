#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int q, a[300001][3];
void getsc(ll a, ll b, ll m, vector<ll>& v) {
    if (a == b) { v.push_back(a); return; }
    ll l = max(1LL, b - m);
    while (l <= (b + 1) / 2) {
        ll j = (a + l - 1) / l;
        if (l * j <= b) {
            v.push_back(l * j);
            getsc(a, b - l * j, m, v);
            return;
        }
        ++l;
    }
    cout << -1 << endl;
    exit(0);
}
int main() {
    cin >> q;
    for (int i = 1; i <= q; ++i) cin >> a[i][0] >> a[i][1] >> a[i][2];
    for (int i = 1; i <= q; ++i) {
        vector<ll> v;
        getsc(a[i][0], a[i][1], a[i][2], v);
        cout << v.size();
        for (auto x : v) cout << ' ' << x;
        cout << endl;
    }
}
