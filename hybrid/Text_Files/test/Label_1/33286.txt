import java.util.Scanner;

public class PresentsForHedgehog {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int days = scanner.nextInt();
        int maxDaysBetweenPresents = scanner.nextInt();
        int holidaysCount = scanner.nextInt();
        int[] holidays = new int[holidaysCount];
        for (int i = 0; i < holidaysCount; i++) {
            holidays[i] = scanner.nextInt();
        }

        int lastPresentDay = -1;
        int presentsCount = 0;

        for (int i = 0; i < days; i++) {
            if (isHoliday(i, holidays) || i - lastPresentDay >= maxDaysBetweenPresents) {
                presentsCount++;
                lastPresentDay = i;
            }
        }

        System.out.println(presentsCount);
    }

    private static boolean isHoliday(int day, int[] holidays) {
        for (int holiday : holidays) {
            if (holiday == day) {
                return true;
            }
        }
        return false;
    }
}
