from collections import defaultdict

n = int(input())
a = [input() for i in range(n)]
d = defaultdict(int)
for i in range(n):
    for j in range(len(a[i])):
        if a[i][j] != '?':
            d[i, j] = int(a[i][j])
            if j > 0 and (i == 0 or d[i, j] > d[i-1, j]):
                d[i-1, j] = d[i, j] - 1
    for j in range(len(a[i]) - 1, -1, -1):
        if a[i][j] != '?':
            d[i, j] = int(a[i][j])
            if j < len(a[i]) - 1 and (i == 0 or d[i, j] < d[i-1, j+1]):
                d[i-1, j+1] = d[i, j] + 1
if any(d[i, 0] > 9 or d[i, 0] != d[i, len(a[i]) - 1] for i in range(n)):
    print("NO")
else:
    print("YES")
    for i in range(n):
        print(''.join(str(d[i, j] if a[i][j] == '?' else a[i][j]) for j in range(len(a[i]))))
