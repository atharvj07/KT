import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    static int n;
    static int[] apples;
    static List<List<Integer>> adj;
    static int[] subTreeWeight;
    static boolean[] vis;
    static int ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        apples = new int[n + 1];
        String[] line = br.readLine().split(" ");
        for (int i = 1; i <= n; i++) {
            apples[i] = Integer.parseInt(line[i - 1]);
        }
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            line = br.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        subTreeWeight = new int[n + 1];
        vis = new boolean[n + 1];
        dfs(1, 0);
        Arrays.sort(subTreeWeight);
        int mid = subTreeWeight[n / 2];
        for (int i = 0; i < n; i++) {
            ans += Math.abs(mid - subTreeWeight[i]);
        }
        System.out.println(ans);
    }

    private static void dfs(int u, int p) {
        vis[u] = true;
        subTreeWeight[u] = apples[u];
        for (int v : adj.get(u)) {
            if (!vis[v]) {
                dfs(v, u);
                subTreeWeight[u] += subTreeWeight[v];
            }
        }
    }
}
