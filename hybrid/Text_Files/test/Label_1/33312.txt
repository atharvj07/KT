import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            boolean win = false;
            for (int j = 0; j < n; j++) {
                if (a[j] == 2048) {
                    win = true;
                    break;
                }
            }
            for (int j = 0; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (a[j] == a[k]) {
                        win = true;
                        break;
                    }
                }
            }
            System.out.println(win ? "YES" : "NO");
        }
    }
}
