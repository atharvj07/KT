import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long ans = (m + n) / 2;
        if ((m + n) % 2 != 0) {
            ans--;
        }
        System.out.println(ans);
    }
}
