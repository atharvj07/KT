#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;

    std::string s;
    std::cin >> s;

    std::vector<int> cnt(n, 0);

    int ans = 0;
    int pref = 0;
    for (int i = 0; i < n; ++i) {
        pref += s[i] == '1' ? 1 : -1;
        if (pref == 0) { ans = i + 1; }
        else if (pref + cnt[pref - 1] == 0) { ans = std::max(ans, i - cnt[pref - 1]); }
        cnt[pref]++;
    }

    std::cout << ans << '\n';
    return 0;
}
