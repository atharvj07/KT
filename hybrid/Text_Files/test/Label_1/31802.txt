import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < a[i]; j++) {
                sum += j - i;
            }
        }
        System.out.println(sum);
    }
}
