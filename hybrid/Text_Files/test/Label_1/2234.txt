#include <bits/stdc++.h>
using namespace std;
int n; string s, ans;
int main(){
    getline(cin, s); n=s.size(); ans=s;
    for(int i=n-1, cnt=0; i>=0; i--){
        if(s[i]==s[i+1]) cnt++;
        if(cnt){
            ans.erase(i, 2);
            cnt--;
        }
    }
    for(int len=n; len; len--){
        if(ans.size()<=len) cout<<ans.size()<<" "<<ans<<"\n";
        else cout<<len<<" "<<ans.substr(0,5)<<"..."+ans.substr(ans.size()-2,2)<<"\n";
    }
    return 0;
}
