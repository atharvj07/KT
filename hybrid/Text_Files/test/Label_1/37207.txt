n, m, q = map(int, input().split())
grid = [list(map(int, input().split())) for _ in range(n)]

def calculate_score():
    max_score = 0
    for row in grid:
        count = max_consecutive = 0
        for cell in row:
            if cell == 1:
                count += 1
                max_consecutive = max(max_consecutive, count)
            else:
                count = 0
        max_score = max(max_score, max_consecutive)
    return max_score

for _ in range(q):
    i, j = map(int, input().split())
    grid[i-1][j-1] = 1 - grid[i-1][j-1]  # Toggle the state
    print(calculate_score())
