import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] sizes = new int[n];
        for (int i = 0; i < n; i++) {
            sizes[i] = sc.nextInt();
        }
        int ans = 0;
        int cnt = 0;
        for (int size : sizes) {
            if (ans + size <= sizes[n - 1]) {
                ans += size;
                cnt++;
            } else {
                break;
            }
        }
        if (cnt >= k) {
            System.out.println(ans);
        } else {
            System.out.println(-1);
        }
    }
}
