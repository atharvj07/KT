import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static ArrayList<Integer>[] adjList;
    static int[] distBob;
    static int[] distAlex;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        adjList = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adjList[i] = new ArrayList<>();
        }
        while (m-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adjList[a].add(b);
            adjList[b].add(a);
        }
        distBob = new int[n + 1];
        distAlex = new int[n + 1];
        Arrays.fill(distBob, -1);
        Arrays.fill(distAlex, -1);
        distBob[1] = 0;
        distAlex[n] = 0;
        bfs(1, distBob);
        bfs(n, distAlex);
        if (distBob[n] != -1) {
            System.out.println(distBob[n]);
            ArrayList<Integer> pathBob = new ArrayList<>();
            ArrayList<Integer> pathAlex = new ArrayList<>();
            int curr = n;
            while (curr != 1) {
                pathBob.add(curr);
                for (int neighbor : adjList[curr]) {
                    if (distBob[neighbor] == distBob[curr] - 1) {
                        curr = neighbor;
                        break;
                    }
                }
            }
            pathBob.add(1);
            curr = 1;
            while (curr != n) {
                pathAlex.add(curr);
                for (int neighbor : adjList[curr]) {
                    if (distAlex[neighbor] == distAlex[curr] - 1) {
                        curr = neighbor;
                        break;
                    }
                }
            }
            pathAlex.add(n);
            for (int i = 0; i <= distBob[n]; i++) {
                System.out.print(pathBob.get(i) + " ");
            }
            System.out.println();
            for (int i = distAlex[n]; i >= 0; i--) {
                System.out.print(pathAlex.get(i) + " ");
            }
        } else {
            System.out.println(-1);
        }
    }

    static void bfs(int start, int[] dist) {
        Queue<Integer> q = new LinkedList<>();
        q.add(start);
        while (!q.isEmpty()) {
            int curr = q.poll();
            for (int neighbor : adjList[curr]) {
                if (dist[neighbor] == -1) {
                    q.add(neighbor);
                    dist[neighbor] = dist[curr] + 1;
                }
            }
        }
    }
}
