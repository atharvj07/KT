def min_different_letters(a, b, l, r):
    cycle = 2 * (a + b)
    if a >= b:
        if l <= a + b:
            return min(a, r - l + 1)
        else:
            return 1
    else:
        full_cycles_before_l = (l - 1) // cycle
        l_in_cycle = (l - 1) % cycle + 1
        r_in_cycle = (r - 1) % cycle + 1
        full_cycles_between_l_r = (r - 1) // cycle - full_cycles_before_l
        
        if full_cycles_between_l_r > 0:
            return min(a + 1, a + b)
        
        if l_in_cycle <= a + b:
            if r_in_cycle <= a + b:
                return min(a, r_in_cycle - l_in_cycle + 1)
            else:
                return min(a + 1, a + b)
        else:
            if r_in_cycle <= a + b:
                return 1
            else:
                return min(r_in_cycle - a - b, a)

a, b, l, r = map(int, input().split())
print(min_different_letters(a, b, l, r))
