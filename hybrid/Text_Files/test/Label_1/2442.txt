#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const ll mod = 1e9 + 7;
int n, x;
vector<ll> a, pre;

ll power(ll base, ll p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  ll val = power(base, p / 2);
  val = (val * val) % mod;
  if (p % 2) val = (val * base) % mod;
  return val;
}

int main() {
  int tt;
  cin >> tt;
  while (tt--) {
    cin >> n >> x;
    a.resize(n);
    pre.resize(n + 1);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      pre[i + 1] = pre[i] + a[i];
    }
    int ans = 0;
    unordered_map<ll, int> mp;
    mp[0] = -1;
    for (int i = 0; i <= n; i++) {
      ll sum = pre[i] % x;
      if (mp.count(sum)) {
        ans = max(ans, i - mp[sum]);
      } else {
        mp[sum] = i;
      }
    }
    cout << (ans ? ans : -1) << endl;
  }
  return 0;
}
