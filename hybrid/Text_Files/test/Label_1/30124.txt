import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Main {
    static final int M = 1000000007;
    int n, k;
    List<Edge>[] g;
    long[] dp;
    long ans = 0;

    class Edge {
        int to;
        long w;

        Edge(int t, long _w) {
            to = t;
            w = _w;
        }
    }

    public Main() {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), x = sc.nextInt();
            if (x == 1) {
                g[a].add(new Edge(b, 10000));
                g[b].add(new Edge(a, 0));
            } else {
                g[a].add(new Edge(b, 0));
                g[b].add(new Edge(a, 0));
            }
        }
        k = sc.nextInt();
        int[] s = new int[k];
        for (int i = 0; i < k; i++) {
            s[i] = sc.nextInt();
        }
        Arrays.sort(s);
        dp = new long[n + 1];
        Arrays.fill(dp, -1);
        dfs(1, 1, s[0]);
        System.out.println(ans % M);
    }

    void dfs(int u, long cost, int dest) {
        dp[u] = cost;
        if (u == dest) {
            ans = Math.min(ans, cost);
            return;
        }
        for (Edge e : g[u]) {
            if (e.to != u && dp[e.to] != -1) {
                continue;
            }
            dfs(e.to, cost + e.w, dest);
        }
    }

    public static void main(String[] args) {
        new Main();
    }
}
