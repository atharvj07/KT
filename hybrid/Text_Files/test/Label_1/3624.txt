#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m;
    cin >> n;
    while (n--) {
        cin >> m;
        int h = 1;
        int la = 0, lb = 0;
        bool d = true;
        for (int i = 0, a; i < m; ++i) {
            cin >> a;
            if (a) {
                if (lb)
                    h += 5;
                else
                    h += 1;
            } else if (la)
                d = false;
            la = lb;
            lb = a;
        }
        cout << (d ? h : -1) << '\n';
    }
    return 0;
}
