#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    cin.tie(0);
    ios::sync_with_stdio(0);
    int t;
    cin >> t;
    while (t--) {
        ll a, b, n, m;
        cin >> a >> b >> n >> m;
        if (min(a, b) < min(n, m))
            cout << "No" << '\n';
        else if ((a <= n && b >= m) || (a >= m && b <= n))
            cout << "Yes" << '\n';
        else
            cout << "No" << '\n';
    }
}
