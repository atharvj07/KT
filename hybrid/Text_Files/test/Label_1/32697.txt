import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int maxLength = 0;
        int currentLength = 0;
        String ACGT = "ACGT";
        for (char c : s.toCharArray()) {
            if (ACGT.indexOf(c) >= 0) {
                currentLength++;
            } else {
                maxLength = Math.max(maxLength, currentLength);
                currentLength = 0;
            }
        }
        maxLength = Math.max(maxLength, currentLength);
        System.out.println(maxLength);
    }
}
