import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int ignored = 0;
        int speed = 0;
        boolean overtakeAllowed = true;
        boolean speedLimit = false;
        boolean limitedSpeed = false;
        for (int i = 0; i < n; i++) {
            int type = sc.nextInt();
            switch (type) {
                case 1:
                    speed = sc.nextInt();
                    break;
                case 2:
                    if (!overtakeAllowed) {
                        ignored++;
                    }
                    break;
                case 3:
                    if (speedLimit) {
                        ignored++;
                    }
                    limitedSpeed = true;
                    speedLimit = true;
                    speed = sc.nextInt();
                    break;
                case 4:
                    if (limitedSpeed) {
                        ignored++;
                    }
                    overtakeAllowed = true;
                    limitedSpeed = false;
                    speedLimit = false;
                    break;
                case 5:
                    if (speedLimit) {
                        ignored++;
                    }
                    speedLimit = false;
                    break;
                case 6:
                    overtakeAllowed = false;
                    break;
            }
        }
        System.out.println(ignored);
    }
}
