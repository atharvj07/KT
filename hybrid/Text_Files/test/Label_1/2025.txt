#include <iostream>
using namespace std;

#define ll long long

ll n, m, k;

ll gcd(ll a, ll b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}

int main() {
  cin >> n >> m >> k;
  ll g = gcd(n, m);
  if (k >= g + 1)
    cout << -1 << endl;
  else
    cout << (n / g) * (m / g) << endl;
  return 0;
}
