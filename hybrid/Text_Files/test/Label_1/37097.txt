for _ in range(int(input())):
    n, l, r, s = map(int, input().split())
    ans = [0] * (r - l + 1)
    t = s - n * (n + 1) // 2
    for i in range(r - l + 1):
        if not 0 <= t <= r - l:
            print(-1)
            break
        t -= i == r - l
        ans[i] = t
        t += 1
    else:
        lst = [i for i in range(1, l) if i not in ans] + [i for i in range(r + 1, n + 1) if i not in ans] + ans
        print(*lst)
