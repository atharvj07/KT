#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;

bool sieve[MAXN];
int mu[MAXN];
int ans;
vector<int> p;

void mobius_gen() {
    mu[1] = 1;
    for (int i = 2; i < MAXN; i++) {
        if (!sieve[i]) {
            mu[i] = -1;
            p.push_back(i);
        }
        for (int j = 0; j < p.size() && i * p[j] < MAXN; j++) {
            sieve[i * p[j]] = true;
            if (i % p[j] == 0) {
                mu[i * p[j]] = 0;
                break;
            } else
                mu[i * p[j]] = -mu[i];
        }
    }
}

vector<int> get_div(int n) {
    vector<int> ret;
    for (int i = 1, j = 0; i * i <= n; i = j + 1) {
        if (n % i == 0) {
            ret.push_back(i);
            if (i * i != n) ret.push_back(n / i);
        }
        j = n / i;
    }
    return ret;
}

int32_t main() {
    mobius_gen();
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int even = 0;
        for (int i = 0; i < n; i++) {
            int a;
            cin >> a;
            vector<int> divs = get_div(a);
            for (int d : divs) {
                if (mu[d] == 1) even++;
            }
        }
        ans = 0;
        if (even >= 2) {
            cout << n << endl;
            for (int i = 1; i <= n; i++) cout << i << " ";
            cout << endl;
        } else {
            vector<int> vec;
            for (int i = 0; i < n; i++) {
                int a;
                cin >> a;
                vector<int> divs = get_div(a);
                bool ok = false;
                for (int d : divs) {
                    if (mu[d] == 1) ok = true;
                }
                if (ok) vec.push_back(i + 1);
            }
            ans = vec.size();
            cout << ans << endl;
            for (int x : vec) cout << x << " ";
            cout << endl;
        }
    }
}
