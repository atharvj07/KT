import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while(t-- > 0) {
            long x = sc.nextLong();
            long cb = (long) Math.cbrt(x);
            if(cb * cb * cb == x) System.out.println("YES");
            else System.out.println("NO");
        }
    }
}
