#include <iostream>
using namespace std;

int main() {
  long long t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    string a, b;
    cin >> a >> b;
    long long ans = 0;
    bool check = false;
    for (long long i = 0; i < n; i++) {
      if (a[i] != b[i]) { ans++; }
      if (a[i] == '1') { check = true; }
    }
    cout << (check ? ans : -1) << endl;
  }
  return 0;
}
