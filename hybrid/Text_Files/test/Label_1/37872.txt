import sys
input = sys.stdin.readline

n = int(input())
s = input().strip()
p = [(0, 0)]
for c in s:
    if c == 'U':
        p.append((p[-1][0]+1, p[-1][1]))
    elif c == 'D':
        p.append((p[-1][0]-1, p[-1][1]))
    elif c == 'L':
        p.append((p[-1][0], p[-1][1]-1))
    elif c == 'R':
        p.append((p[-1][0], p[-1][1]+1))
p = set(p)
print(len(p))
