#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, W;
    cin >> n >> W;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    sort(v.begin(), v.end());
    reverse(v.begin(), v.end());
    int ans = 0, used = 0;
    for (int i : v) {
      if (used + i <= W) {
        used += i;
        ans++;
      }
    }
    cout << ans << endl;
  }
}
