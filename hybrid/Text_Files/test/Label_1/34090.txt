import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class HanhTree {

    static final int MOD = 998244353;
    static final int ROOT = 1;

    static class Edge {
        int to, weight;

        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    static long pow(long x, long p) {
        if (p == 0) return 1;
        if (p == 1) return x;
        long half = pow(x, p / 2);
        half = (half * half) % MOD;
        if (p % 2 == 1) half = (half * x) % MOD;
        return half;
    }

    static long inv(long x) {
        return pow(x, MOD - 2);
    }

    static class Node {
        int id;
        List<Edge> edges = new ArrayList<>();
        long size, sum;

        Node(int id) {
            this.id = id;
        }
    }

    static class Tree {
        Node[] nodes;

        Tree(int n) {
            nodes = new Node[n + 1];
            for (int i = 1; i <= n; i++) {
                nodes[i] = new Node(i);
            }
        }

        void addEdge(int u, int v, int weight) {
            nodes[u].edges.add(new Edge(v, weight));
            nodes[v].edges.add(new Edge(u, weight));
        }

        void dfs(int node, int parent, int size, long sum) {
            nodes[node].size = size;
            nodes[node].sum = sum;
            for (Edge edge : nodes[node].edges) {
                if (edge.to != parent) {
                    dfs(edge.to, node, size + 1, (sum + size * edge.weight) % MOD);
                }
            }
        }

        void solve() {
            dfs(ROOT, ROOT, 1, 0);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter pw = new PrintWriter(new OutputStreamWriter(System.out));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        Tree tree = new Tree(n);
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            tree.addEdge(u, v, 0);
        }
        tree.solve();
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            if (type == 1) {
                int d = Integer.parseInt(st.nextToken());
                long invSize = inv(tree.nodes[v].size);
                long sum = tree.nodes[v].sum;
                for (Edge edge : tree.nodes[v].edges) {
                    sum = (sum - tree.nodes[edge.to].size * edge.weight) % MOD;
                }
                sum = (sum + MOD) % MOD;
                sum = (sum * invSize) % MOD;
                for (Edge edge : tree.nodes[v].edges) {
                    tree.nodes[edge.to].sum = (tree.nodes[edge.to].sum + sum * edge.weight) % MOD;
                }
            } else {
                pw.println((tree.nodes[v].sum * inv(tree.nodes[v].size)) % MOD);
            }
        }
        pw.close();
    }
}
