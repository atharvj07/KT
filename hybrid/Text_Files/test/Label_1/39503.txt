n = int(input())
f = []; s = []; p = []
for i in range(n):
  fn, sn = list(input().split())
  f.append(fn); s.append(sn)
for i in range(n):
  p.append(int(input())-1)
flag = True
for i in range(n):
  if i!=p[i] and (f[i]<f[p[i]] or s[i]<f[p[i]] or s[i]<s[p[i]]):
    flag = False
print(['NO','YES'][flag])
