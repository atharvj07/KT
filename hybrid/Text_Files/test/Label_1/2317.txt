#include<bits/stdc++.h>
using namespace std;
const int maxn=5e5+5;
int a[maxn],ans[maxn];
int main(){
    int n;cin>>n;
    for(int i=0;i<n;++i)cin>>a[i];
    bool vis[600];
    memset(vis,false,sizeof(vis));
    int cnt=0;
    for(int i=0;i<n;++i){
        memset(vis,false,sizeof(vis));
        for(int j=0;j<i;++j)vis[a[j]]=true;
        int mex;
        for(mex=0;mex<=i;++mex)if(!vis[mex])break;
        ans[i]=mex;
        if(mex>a[i])a[i]=a[i]+1;
    }
    for(int i=0;i<n;++i){
        if(ans[i]>a[i]){cout<<-1<<endl;return 0;}
        cout<<ans[i]<<' ';
    }
}
