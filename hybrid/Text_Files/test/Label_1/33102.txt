import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        if (n < 4) {
            System.out.println(-1);
            return;
        }
        if (n % 2 == 0) {
            int[] arr = new int[n];
            Arrays.fill(arr, 1);
            for (int i = 2; i < n; i += 2) {
                arr[i] = 2;
            }
            System.out.println(Arrays.toString(arr).replace(",", "").replace("[", "").replace("]", ""));
        } else {
            int[] arr = new int[n];
            Arrays.fill(arr, 1);
            arr[0] = 2;
            for (int i = 2; i < n; i += 2) {
                arr[i] = 3;
            }
            System.out.println(Arrays.toString(arr).replace(",", "").replace("[", "").replace("]", ""));
        }
    }
}
