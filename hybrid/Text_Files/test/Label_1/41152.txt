from collections import defaultdict

def dfs(graph, start, visited, exclude):
    visited[start] = True
    count = 1
    for neighbour in graph[start]:
        if not visited[neighbour] and neighbour != exclude:
            count += dfs(graph, neighbour, visited, exclude)
    return count

def find_pairs(n, m, a, b, roads):
    graph = defaultdict(list)
    for u, v in roads:
        graph[u].append(v)
        graph[v].append(u)

    visited_a = [False] * (n + 1)
    visited_b = [False] * (n + 1)

    count_a = dfs(graph, a, visited_a, b) - 1
    count_b = dfs(graph, b, visited_b, a) - 1

    return count_a * count_b

t = int(input().strip())
for _ in range(t):
    n, m, a, b = map(int, input().strip().split())
    roads = [tuple(map(int, input().strip().split())) for _ in range(m)]
    print(find_pairs(n, m, a, b, roads))
