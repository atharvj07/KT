#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, r;
  cin >> n >> r;
  vector<pair<int, int>> v(n);
  for (auto& p : v) { cin >> p.first >> p.second; }
  sort(v.begin(), v.end());
  int ans = 0, rating = r;
  for (auto& p : v) {
    if (rating >= p.first) {
      rating += p.second;
      ++ans;
    }
  }
  cout << ans << endl;
  return 0;
}
