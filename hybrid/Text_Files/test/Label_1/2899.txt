#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &ai : a) cin >> ai;
    sort(a.begin(), a.end());
    int sum = 0;
    for (int ai : a) sum += ai;
    vector<int> b(n);
    for (int i = 0; i < n; i++) {
        b[i] = (a[i] + sum) / 2;
    }
    for (int &bi : b) cout << bi << '\n';
}
