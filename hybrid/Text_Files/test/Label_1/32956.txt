import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int d = sc.nextInt();
        char[][] grid = new char[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if ((i + j) % d == 0) {
                    grid[i][j] = 'R';
                } else if ((i - j) % d == 0) {
                    grid[i][j] = 'Y';
                } else if ((i + j + d) % d == 0) {
                    grid[i][j] = 'G';
                } else if ((i - j + d) % d == 0) {
                    grid[i][j] = 'B';
                }
            }
        }
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                System.out.print(grid[i][j]);
            }
            System.out.println();
        }
    }
}
