#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
  int t; cin >> t;
  while (t--) {
    int n, w;
    cin >> n >> w;
    set<int> ms;
    for (int i = 0, x; i < n; i++) {
      cin >> x;
      ms.insert(x);
    }
    int sum = 0, ans = 0;
    for (int x : ms) {
      sum += x;
      ans += (sum > w);
    }
    cout << ans + 1 << endl;
  }
  return 0;
}
