#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, l, r;
  cin >> n >> l >> r;
  vector<int> a(n), p(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n; ++i) cin >> p[i];
  for (int i = 0; i < n; ++i) a[i] = l + p[i] - a[i];
  for (int i = 0; i < n; ++i)
    if (a[i] < l || a[i] > r) {
      cout << "-1" << endl;
      return 0;
    }
  cout << a[0];
  for (int i = 1; i < n; ++i) cout << ' ' << a[i];
  cout << endl;
  return 0;
}
