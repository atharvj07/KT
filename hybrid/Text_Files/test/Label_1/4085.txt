#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }
  int sum1 = 0, sum2 = 0;
  for (int i = 0; i < n; i++) {
    if (sum1 <= sum2) {
      sum1 += w[i];
    } else {
      sum2 += w[i];
    }
  }
  if (sum1 == sum2) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
