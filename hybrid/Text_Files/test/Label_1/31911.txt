import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int a = scanner.nextInt(), b = scanner.nextInt(), c = scanner.nextInt();
            int ans = 0;
            while (b >= 2 && c >= 1) {
                b -= 2;
                c -= 1;
                ans += 3;
            }
            while (a >= 1 && b >= 1) {
                a -= 1;
                b -= 2;
                ans += 3;
            }
            System.out.println(ans);
        }
        scanner.close();
    }
}
