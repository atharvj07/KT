#include <bits/stdc++.h>
using namespace std;
using LL = long long;

const int MX = 3e5+5;
int p[MX], id[MX], deg[MX], q[MX], pt[MX];
vector<int> adj[MX];

bool topo(int n, int& ord) {
	queue<int> q;
	for (int i = 1; i <= n; i++) if (!deg[i]) q.push(i);
	while (!q.empty()) {
		int u = q.front(); q.pop();
		id[ord++] = u;
		for (int v: adj[u]) if (!(--deg[v])) q.push(v);
	}
	return ord == n;
}

int main() {
	int n, k; cin >> n >> k;
	for (int i = 1; i <= n; i++) {
		cin >> p[i];
		if (!p[i]) pt[i] = 0;
		else ++deg[p[i]], adj[p[i]].push_back(i);
	}
	while (k--) {
		int x,y; cin >> x >> y;
		++deg[x], adj[x].push_back(y);
	}
	int ord = 0;
	if (!topo(n,ord)) { cout << 0; return 0; }
	for (int i = 0; i < n; i++) q[pt[id[i]]] = id[i];
	for (int i = 1; i <= n; i++) cout << q[i] << ' ';
	return 0;
}
