from copy import deepcopy

n, m, q = map(int, input().split())
grid = [[0] * m for i in range(n)]
rows, cols = [0] * n, [0] * m
for _ in range(q):
    t, *args = map(int, input().split())
    if t == 3: i, j, x = args; grid[i - 1][j - 1] = x
    else: r = args[0] - 1; exec(f'rows[{r}] -= {t - 1}; cols[{r}] -= {t - 1}')
for i in range(n):
    for j in range(m): grid[i][j] = grid[i][j - cols[i] % m]
for i in range(m):
    for j in range(n): grid[j][i] = grid[j - rows[j] % n][i]
for i in grid: print(' '.join(map(str, i)))
