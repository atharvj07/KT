from collections import deque

def bfs(adj, start):
    n = len(adj)
    dist = [-1] * n
    dist[start] = 0
    queue = deque([start])
    while queue:
        v = queue.popleft()
        for u in adj[v]:
            if dist[u] == -1:
                dist[u] = dist[v] + 1
                queue.append(u)
    return dist

n = int(input())
color = list(map(int, input().split()))
adj = [[] for _ in range(n)]
for _ in range(n - 1):
    u, v = map(int, input().split())
    adj[u - 1].append(v - 1)
    adj[v - 1].append(u - 1)

# Find the farthest node from node 0
dist_from_0 = bfs(adj, 0)
farthest_node = dist_from_0.index(max(dist_from_0))

# Find the farthest node from the previously found farthest node
# This will give us one of the endpoints of the diameter of the tree
dist_from_farthest = bfs(adj, farthest_node)
diameter_end = dist_from_farthest.index(max(dist_from_farthest))

# The minimum number of operations needed is the ceil of the half of the diameter of the tree
diameter = dist_from_farthest[diameter_end]
print((diameter + 1) // 2)
