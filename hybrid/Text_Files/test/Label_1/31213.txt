import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String S = sc.nextLine();
        int N = S.length();
        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < N; i++) {
            if (S.charAt(i) == '0') {
                cnt++;
            } else {
                ans += Math.min(cnt, N - i - 1);
                cnt--;
            }
        }
        System.out.println(ans);
    }
}
