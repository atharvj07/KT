#include <algorithm>
#include <iostream>
int main() {
  int k, n1, n2, n3, t1, t2, t3, tmp;
  std::cin >> k >> n1 >> n2 >> n3 >> t1 >> t2 >> t3;
  tmp = (t2 + t3 - t1) / t1 * n1 + (n2 - n1) / n2 * t2;
  tmp = std::min(tmp, (t1 + t3 - t2) / t2 * n2 + (n1 - n2) / n1 * t1);
  tmp = std::min(tmp, (t1 + t2 - t3) / t3 * n3 + (n1 - n3) / n1 * t1);
  tmp = std::max(tmp, (t2 + t3) / t2 * n2 + (n3 - n2) / n3 * t3);
  tmp = std::max(tmp, (t1 + t3) / t3 * n3 + (n2 - n3) / n2 * t2);
  tmp = std::max(tmp, (t1 + t2) / t1 * n1 + (n3 - n1) / n3 * t3);
  tmp = std::max(tmp, t1 + t2 + t3);
  std::cout << tmp * k << std::endl;
  return 0;
}
