import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        int steps = 0;
        while (true) {
            boolean changed = false;
            for (int i = 0; i < n - 1; i++) {
                if (arr[i] > arr[i + 1]) {
                    arr[i + 1] = 0;
                    changed = true;
                }
            }
            if (!changed) {
                break;
            }
            steps++;
        }
        System.out.println(steps);
    }
}
