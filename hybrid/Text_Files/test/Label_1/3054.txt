#include <bits/stdc++.h>
#define int long long
using namespace std;

const int MAXN = 80, MAXM = 250;

int n, m, a, b;
int g[MAXN][MAXN], dist[MAXN], par[MAXN];
bool vis[MAXN];

void dijkstra(int src) {
  memset(dist, 63, sizeof(dist));
  dist[src] = 0;

  for (int i = 1; i <= n; i++) {
    int node = -1;
    for (int j = 1; j <= n; j++) {
      if (!vis[j] && (node == -1 || dist[j] < dist[node]))
        node = j;
    }

    vis[node] = 1;

    for (int j = 1; j <= n; j++) {
      if (g[node][j] && !vis[j]) {
        int wt = g[node][j];
        if (dist[node] + wt < dist[j]) {
          dist[j] = dist[node] + wt;
          par[j] = node;
        }
      }
    }
  }
}

int32_t main() {
  memset(g, 63, sizeof(g));

  cin >> n >> m >> a >> b;

  for (int i = 0; i < m; i++) {
    int x, y, wt;
    cin >> x >> y >> wt;
    g[x][y] = wt;
    g[y][x] = wt;
  }

  dijkstra(1);

  for (int i = 1; i <= n; i++) {
    int ans = 0;

    vector<int> path;
    int node = i;

    while (node != 1) {
      path.push_back(node);
      node = par[node];
    }

    path.push_back(1);

    int mn = 1e9;

    for (int j = 1; j < path.size(); j++) {
      mn = min(mn, g[path[j]][path[j - 1]]);
    }

    for (int j = 1; j < path.size(); j++) {
      ans += (g[path[j]][path[j - 1]] == mn ? a : b);
    }

    cout << ans << ' ';
  }

  return 0;
}
