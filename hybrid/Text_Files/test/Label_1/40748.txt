import sys

input = sys.stdin.readline
for _ in range(int(input())):
    n = int(input())
    a = [1]*(n+1)
    b = [int(x) for x in input().split()]
    for i in range(1, n+1):
        if b[i-1] % n:
            print('NO')
            break
        a[i] = b[i-1]//n
    else:
        print('YES')
        print(*a[1:])
