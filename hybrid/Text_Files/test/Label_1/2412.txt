#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    map<string, int> mp;
    for (int i = 0; i < n; i++) {
      mp[s.substr(i)]++;
    }
    for (int i = 0; i < n; i++) {
      string r = s.substr(0, i + 1);
      reverse(r.begin(), r.end());
      if (mp[r] > 0 && i + 2 * r.length() <= n &&
          (i == 0 || mp[s.substr(i + r.length())] > 0)) {
        cout << "YES\n";
        goto end;
      }
    }
    cout << "NO\n";
  end:;
  }
  return 0;
}
