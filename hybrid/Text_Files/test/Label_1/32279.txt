import java.util.*;
import java.io.*;

public class Main {
    private static final int MOD = 1000000009;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int w = sc.nextInt();
        int b = sc.nextInt();
        System.out.println(((long) choose2(n + w - 1) + (long) choose2(n + b - 1) - (long) choose2(n - 1)) % MOD);
    }

    private static long choose2(int n) {
        long result = n;
        result = result * (n - 1) % MOD;
        result = result * power(2, MOD - 2);
        return result % MOD;
    }

    private static long power(long base, long exponent) {
        if (exponent == 0) {
            return 1;
        }
        if (exponent == 1) {
            return base;
        }
        long result = power(base, exponent / 2);
        result = result * result;
        result = result % MOD;
        if (exponent % 2 == 1) {
            result = result * base;
            result = result % MOD;
        }
        return result;
    }
}
