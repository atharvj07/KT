import math

def read_int():
    return int(input())

def read_ints():
    return list(map(int, input().split()))

def area(r):
    return math.pi * r * r

n = read_int()
circles = [read_ints() for _ in range(n)]
circles.sort(key=lambda x: -x[2])

total_area = 0
for i in range(n):
    if i % 2 == 0:
        total_area += area(circles[i][2])
    else:
        total_area -= area(circles[i][2])

print(f"{total_area:.9f}")
