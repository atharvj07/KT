#include <iostream>
#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxN = 3e5 + 5;
int n, r;
pair<ll, ll> p[maxN];
int main() {
    cin >> n >> r;
    for (int i = 0; i < n; i++) cin >> p[i].first >> p[i].second;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) {
            if (abs(p[i].first - p[j].first) + abs(p[i].second - p[j].second) <= r) cnt++;
        }
        ans = max(ans, cnt);
    }
    cout << ans << endl;
}
