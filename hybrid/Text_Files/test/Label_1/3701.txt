#include<bits/stdc++.h>
using namespace std;

int main(){
    long long t, n;
    string a, b;
    cin >> t;
    while(t--){
        cin >> n;
        cin >> a >> b;
        if(a==b) cout << "YES\n";
        else if(n%2){
            long long x=0, y=0;
            for(long long i=0; i<n; i++){
                x+= (a[i]=='1');
                y+= (b[i]=='1');
            }
            if(x==y) cout << "YES\n";
            else cout << "NO\n";
        }
        else cout << "NO\n";
    }
    return 0;
}
