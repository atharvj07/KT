import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args){
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        long sum = 0;
        long[] a = new long[n];
        for(int i = 0; i < n; i++){
            a[i] = input.nextLong();
            sum += a[i];
        }
        Arrays.sort(a);
        int p = k;
        long ans = 0;
        for(int i = n-1; i >= 0; i--){
            if(p == 0)break;
            ans += a[i]*a[i];
            a[i] = a[i] - 1;
            if(a[i] > 0){
                p--;
            }
            Arrays.sort(a);
        }
        System.out.println(ans);
    }
}
