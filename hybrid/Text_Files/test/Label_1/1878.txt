#include <bits/stdc++.h>
using namespace std;

int main() {
  string n;
  cin >> n;
  int ans = 0;
  for (int i = 0; i < n.size(); i++) {
    if (n[i] >= '4') ans++, n[i] = '0';
  }
  reverse(n.begin(), n.end());
  while (!n.empty() && n.back() == '0') ans++, n.pop_back();
  if (n.empty())
    cout << ans << endl;
  else
    cout << -1 << endl;
  return 0;
}
