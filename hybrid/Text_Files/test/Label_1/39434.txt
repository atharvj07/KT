def is_possible_to_zero(n, rotations):
    def dfs(idx, total):
        if idx == n:
            return total % 360 == 0
        return dfs(idx + 1, total + rotations[idx]) or dfs(idx + 1, total - rotations[idx])
    
    return dfs(0, 0)

n = int(input().strip())
rotations = [int(input().strip()) for _ in range(n)]

print("YES" if is_possible_to_zero(n, rotations) else "NO")
