n, k = map(int, input().split())
table = [[0] * n for _ in range(n)]

num = 1
for i in range(n):
    for j in range(k - 1):
        table[i][j] = num
        num += 1

for i in range(n):
    for j in range(k - 1, n):
        table[i][j] = num
        num += 1

col_sum = sum(table[i][k - 1] for i in range(n))
print(col_sum)
for row in table:
    print(' '.join(map(str, row)))
