import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, x = map(int, input().split())
    s = input().strip()
    cnt = 0
    ans = 0
    freq = [0] * n
    for i in range(n - 1, -1, -1):
        if s[i] == '1':
            cnt += 1
        freq[i] = cnt
    for i in range(n):
        lo, hi = i, n
        pos = -1
        while lo <= hi:
            mid = (lo + hi) // 2
            if x <= freq[mid] - (freq[(mid - 1) % n] if i > 0 else 0):
                pos = mid
                hi = mid - 1
            else:
                lo = mid + 1
        if pos == -1:
            print(-1)
            break
        ans += pos - i + 1
    else:
        print(ans)
