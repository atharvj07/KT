for _ in range(int(input())):
    n,k=map(int,input().split());a=list(map(int,input().split()))
    print('YES'if sum(a)%2else'NO');a=[0]+a
    if k==1 and sum(a)%2:
        print(n)
        continue
    if sum(a)%2==0:print('NO');continue
    ans=[];s=0
    for i in range(1,n+1):
        s+=a[i]
        if s%2 or i==n:
            ans.append(str(i));s=0
    print(' '.join(ans))
