#include <iostream>
#include <limits>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long a, b, n, m;
    cin >> a >> b >> n >> m;
    bool yes = false;
    if(n == 0 || m == 0){
        yes = true;
    } else if (n == m) {
      if (a >= n) { yes = true; }
    } else if (a > m && b >= n) {
      yes = true;
    } else if (b > m && a >= n) {
      yes = true;
    }
    cout << (yes ? "Yes" : "No") << endl;
  }
  return 0;
}
