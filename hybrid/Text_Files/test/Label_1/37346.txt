def find_increasing_sequence(n, b):
    b.sort()
    a = [b[0]]
    for i in range(1, n):
        for j in range(i, n):
            if a[-1] ^ b[j] > a[-1]:
                a.append(a[-1] ^ b[j])
                b[i], b[j] = b[j], b[i]
                break
        else:
            return "No"
    return "Yes\n" + " ".join(map(str, b))

n = int(input())
b = list(map(int, input().split()))

print(find_increasing_sequence(n, b))
