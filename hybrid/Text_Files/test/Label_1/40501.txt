n, h, m, k = map(int, input().split())
m //= 2

trains = []
for i in range(n):
    h_i, m_i = map(int, input().split())
    trains.append(((m_i % m), i+1))

trains.sort()

min_canceled = n
best_time = 0
canceled_trains = []

for start in range(n):
    current_time = (trains[start][0] + k) % m
    canceled = []
    for end in range(start, start + n):
        end_mod = end % n
        if (trains[end_mod][0] - current_time) % m < 2*k % m:
            canceled.append(trains[end_mod][1])
    if len(canceled) < min_canceled:
        min_canceled = len(canceled)
        best_time = current_time
        canceled_trains = canceled

print(min_canceled, best_time * 2)
print(*sorted(canceled_trains))
