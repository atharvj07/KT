n = int(input().strip())
a = list(map(int, input().strip().split()))

result = 0
left = 0
frequency = [0] * (n + 1)
max_value = 0
min_value = 1

for right in range(n):
    frequency[a[right]] += 1
    max_value = max(max_value, a[right])

    while frequency[a[right]] > 1 or max_value - min_value != right - left:
        frequency[a[left]] -= 1
        if frequency[a[left]] == 0 and a[left] == min_value:
            min_value += 1
            while min_value <= n and frequency[min_value] > 0:
                min_value += 1
        left += 1
        max_value = max(a[left:right+1])

    if max_value - min_value == right - left and max_value != 0:
        result += 1

print(result)
