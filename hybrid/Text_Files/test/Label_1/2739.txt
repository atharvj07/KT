#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin >> n;
    vector<int> p(n), cnt(n), ans(n);
    for (int i = 0; i < n; i++) {
        cin >> p[i];
        p[i]--;
        cnt[p[i]]++;
    }
    int last = -1;
    for (int i = 0; i < n; i++) {
        if (cnt[i] == 0) continue;
        if (last != -1) ans[last] += (i - last - 1) * cnt[i];
        last = i;
        cnt[i]--;
        if (cnt[i] == 0 && last != -1) { ans[last] += cnt[last]; last = -1; }
    }
    for (int i = 0; i < n; i++) {
        cout << (i ? ans[i - 1] : 0) << ' ';
    }
    cout << '\n';
}
