import java.util.*;
import java.math.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long N = Long.parseLong(sc.next());
        long ans = 1L;
        for (int i = 1; i <= N; i++) {
            ans = (ans * i) % 1000000007;
        }
        System.out.println(ans);
    }
}
