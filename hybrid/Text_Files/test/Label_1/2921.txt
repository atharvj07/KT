#include <bits/stdc++.h>

using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  vector<string> cake(r);
  for (int i = 0; i < r; i++) {
    cin >> cake[i];
  }
  vector<vector<int>> dp(r + 1, vector<int>(c + 1, -1));
  dp[r][0] = dp[0][c] = 0;
  for (int i = r - 1; i >= 0; i--) {
    for (int j = c - 1; j >= 0; j--) {
      if (cake[i][j] == 'S') continue;
      dp[i][j] = max(dp[i + 1][j], dp[i][j + 1]) + 1;
    }
  }
  cout << dp[0][0] << endl;
  return 0;
}
