import sys
input = sys.stdin.readline

def max_goods(n, p, k, a):
    a.sort()
    i = 0
    cnt = 0
    while i < n and p >= a[i]:
        p -= a[i]
        cnt += 1
        i += k - 1
    return cnt

for _ in range(int(input())):
    n, p, k = map(int, input().split())
    a = list(map(int, input().split()))
    print(max_goods(n, p, k, a))
