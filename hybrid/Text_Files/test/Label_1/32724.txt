import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int diskSize = scanner.nextInt();
    int[] intensities = new int[n];
    for (int i = 0; i < n; i++) {
      intensities[i] = scanner.nextInt();
    }
    int numBits = calculateNumberOfBits(intensities);
    int numBytes = (numBits + 7) / 8;
    int bitsPerByte = 8;
    int numChanged = 0;
    if (numBytes > diskSize) {
      int[] left = new int[n];
      int[] right = new int[n];
      Arrays.fill(left, 1000000000);
      Arrays.fill(right, -1);
      for (int i = 0; i < n; i++) {
        left[i] = Math.min(left[i], intensities[i]);
        right[i] = Math.max(right[i], intensities[i]);
      }
      int[] intervals = new int[n];
      int intervalCount = 0;
      for (int i = 0; i < n; i++) {
        if (i == 0 || intensities[i] != intensities[i - 1]) {
          intervals[intervalCount] = intensities[i];
          intervalCount++;
        }
      }
      int[] compressed = new int[intervalCount];
      for (int i = 0; i < intervalCount; i++) {
        compressed[i] = intervals[i];
      }
      int[] leftCompressed = new int[intervalCount];
      int[] rightCompressed = new int[intervalCount];
      Arrays.fill(leftCompressed, 1000000000);
      Arrays.fill(rightCompressed, -1);
      for (int i = 0; i < intervalCount; i++) {
        leftCompressed[i] = Math.min(leftCompressed[i], compressed[i]);
        rightCompressed[i] = Math.max(rightCompressed[i], compressed[i]);
      }
      int numBytesCompressed = ((calculateNumberOfBits(compressed) + 7) / 8) * intervalCount;
      numChanged = 0;
      if (numBytesCompressed <= diskSize) {
        for (int i = 0; i < n; i++) {
          if (intensities[i] < leftCompressed[0]) {
            numChanged++;
          } else if (intensities[i] > rightCompressed[0]) {
            numChanged++;
          }
        }
      } else {
        numChanged = n;
      }
    }
    System.out.println(numChanged);
  }

  private static int calculateNumberOfBits(int[] intensities) {
    int maxValue = 0;
    for (int intensity : intensities) {
      maxValue = Math.max(maxValue, intensity);
    }
    int numBits = 0;
    while (maxValue > 0) {
      numBits++;
      maxValue >>= 1;
    }
    return numBits;
  }
}
