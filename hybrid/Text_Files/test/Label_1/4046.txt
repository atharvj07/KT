#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int n, a[N];
bool ok(int k) {
    int f[k + 1] = {0}, s[k + 1] = {0};
    for (int i = 0; i < n; i++) {
        if (a[i] > k) {
            f[0] = 1;
            if (a[i] >= k + 1) {
                s[0] = 1;
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = k; j >= 1; j--) {
            if (a[i] >= j && f[j - 1]) {
                f[j] = 1;
                if (a[i] >= j + 1) {
                    s[j] = 1;
                }
            }
        }
    }
    return f[k] && s[k - 1];
}
int main() {
    cin.tie(NULL);
    ios_base::sync_with_stdio(0);
    int Test, t = 0;
    cin >> Test;
    while (Test--) {
        t++;
        bool flag = false;
        cin >> n;
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (ok(i)) {
                ans = i;
                break;
            }
        }
        cout << ans << endl;
    }
}
