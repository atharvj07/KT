import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long sum = 0;
        for (long i = 0; i < m; i++) {
            sum += sc.nextLong();
        }
        System.out.println(sum <= n ? n - sum : -1);
    }
}
