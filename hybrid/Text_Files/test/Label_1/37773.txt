def count_balanced_prefixes(n, x, s):
    balance = 0
    cnt = {0: 1}
    for char in s:
        if char == '0':
            balance += 1
        else:
            balance -= 1
        if balance in cnt:
            cnt[balance] += 1
        else:
            cnt[balance] = 1

    if balance == 0:
        if x in cnt:
            return -1
        else:
            return 0

    ans = 0
    for k, v in cnt.items():
        if (x - k) % balance == 0 and (x - k) // balance >= 0:
            ans += v
    return ans

T = int(input().strip())
for _ in range(T):
    n, x = map(int, input().strip().split())
    s = input().strip()
    print(count_balanced_prefixes(n, x, s))
