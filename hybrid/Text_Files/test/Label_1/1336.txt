#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = ll(1e9) + 7;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    vector<pair<ll, ll>> x(1);
    vector<pair<ll, ll>> y(1);
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        pair<ll, ll> l, r;
        cin >> l.first >> l.second >> r.first >> r.second;
        x.push_back(l), x.push_back(r);
        y.push_back(l), y.push_back(r);
    }
    sort(x.begin(), x.end()), sort(y.begin(), y.end());
    ll m = 0, mx = 0, l = 0, r = 0;
    for (int i = 1; i < x.size(); i++) {
        if (x[i] == x[i - 1]) r++;
        else
            r = 1;
        if (r > mx) mx = r, m = x[i].first;
    }
    m = 0, mx = 0, l = 0, r = 0;
    for (int i = 1; i < y.size(); i++) {
        if (y[i] == y[i - 1]) r++;
        else
            r = 1;
        if (r > mx) mx = r, m = y[i].first;
    }
    cout << m << " " << m << endl;
}
