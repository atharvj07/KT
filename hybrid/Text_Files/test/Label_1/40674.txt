import sys

def rotate_90(m):
    return list(zip(*m[::-1]))

def flip_v(m):
    return m[::-1]

def flip_h(m):
    return [list(i) for i in zip(*m)]

def match(m1, m2):
    return m1 == m2 or m1 == rotate_90(m2) or m1 == rotate_90(rotate_90(m2)) or m1 == rotate_90(rotate_90(rotate_90(m2))) or m1 == flip_v(m2) or m1 == flip_h(m2) or m1 == flip_v(flip_h(m2))

n = int(sys.stdin.readline())
m1 = [list(sys.stdin.readline().strip()) for i in range(n)]
m2 = [list(sys.stdin.readline().strip()) for i in range(n)]
sys.stdout.write("Yes" if match(m1, m2) else "No")
