import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
  private static List<Integer>[] adjList;
  private static int[] degree;
  private static boolean[] vis;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int[] nk = Arrays.stream(br.readLine().split("\\s+")).mapToInt(Integer::parseInt).toArray();
    int n = nk[0], m = nk[1];
    adjList = new List[n + 1];
    degree = new int[n + 1];
    vis = new boolean[n + 1];
    for (int i = 1; i <= n; i++) {
      adjList[i] = new ArrayList<>();
    }
    for (int i = 0; i < m; i++) {
      int[] tmp = Arrays.stream(br.readLine().split("\\s+")).mapToInt(Integer::parseInt).toArray();
      adjList[tmp[0]].add(tmp[1]);
      adjList[tmp[1]].add(tmp[0]);
      degree[tmp[0]]++;
      degree[tmp[1]]++;
    }
    int longest = 0;
    for (int i = 0; i < m; i++) {
      vis = new boolean[n + 1];
      int tmp = dfs(1, -1, 0);
      if (tmp == -1) {
        System.out.println(-1);
      } else {
        longest = Math.max(longest, tmp);
        System.out.println(longest);
      }
    }
  }

  private static int dfs(int node, int pre, int distance) {
    vis[node] = true;
    int ret = distance;
    int oddNum = 0;
    List<Integer> curDegree = new ArrayList<>();
    for (int u : adjList[node]) {
      if (u == pre) {
        continue;
      }
      if (vis[u]) {
        curDegree.add(degree[u]);
        if (degree[u] % 2 == 1) {
          oddNum++;
        }
      } else {
        int tmp = dfs(u, node, distance + 1);
        if (tmp == -1) {
          return -1;
        }
        ret = Math.max(ret, tmp);
        if (degree[u] % 2 == 1) {
          oddNum++;
        }
        curDegree.add(degree[u]);
      }
    }
    if (pre != -1) {
      degree[node] = degree[pre] - 1;
      for (int d : curDegree) {
        degree[node] -= d;
      }
      if (oddNum % 2 != degree[node] % 2) {
        return -1;
      }
    }
    return ret;
  }
}
