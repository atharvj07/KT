#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  cin >> n;
  unordered_map<int, int> count;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    count[x]++;
  }
  if (count.size() <= 2) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    int petya = 0, vasya = 0;
    for (auto &p : count) {
      if (petya == 0) {
        petya = p.first;
      } else if (petya != p.first && vasya == 0) {
        vasya = p.first;
      } else {
        cout << "NO" << endl;
        return 0;
      }
    }
    cout << petya << " " << vasya << endl;
  }
  return 0;
}
