import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String prophesy = sc.nextLine();
        int max = 0;
        String date = "";
        int[] days = new int[]{0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        for (int i = 2013; i <= 2015; i++)
            for (int j = 1; j <= 12; j++)
                for (int k = 1; k <= days[j]; k++) {
                    String d = (k < 10 ? "0" : "") + k + "-" + (j < 10 ? "0" : "") + j + "-" + i;
                    int c = 0;
                    for (int l = 0; l <= prophesy.length() - d.length(); l++) {
                        String temp = prophesy.substring(l, l + d.length());
                        if (d.equals(temp)) {
                            c++;
                        }
                    }
                    if (c > max) {
                        max = c;
                        date = d;
                    }
                }
        System.out.println(date);
    }
}
