import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] x = new int[n];
        String[] temp = br.readLine().split(" ");
        for (int i = 0; i < n; i++)
            x[i] = Integer.parseInt(temp[i]);
        int[] mini = new int[n];
        int[] maxi = new int[n];
        for (int i = 0; i < n; i++) {
            int min = Math.abs(x[i] - x[(i > 0) ? i - 1 : 0]);
            int max = Math.abs(x[i] - x[n - 1]);
            mini[i] = min;
            maxi[i] = max;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(mini[i] + " " + maxi[i]);
        }
    }
}
