import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        char[] chars = s.toCharArray();
        int sumLeft = 0;
        int sumRight = 0;
        for (int i = 0; i < 3; i++) {
            sumLeft += chars[i] - '0';
            sumRight += chars[i + 3] - '0';
        }
        int ans = Math.abs(sumLeft - sumRight);
        System.out.println(ans);
    }
}
