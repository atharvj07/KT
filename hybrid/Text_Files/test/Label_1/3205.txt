#include <bits/stdc++.h>
using namespace std;
const int mxN = 1e4;
int a[mxN], b[mxN];
int main() {
  int k;
  cin >> k;
  for (int i = 2; i * i <= k; i++) {
    if (k % i == 0) {
      a[k / i]++, b[i]++;
    }
  }
  vector<int> r, c;
  for (int i = 1; i <= k; i++) {
    if (a[i] && b[i])
      r.push_back(i), c.push_back(i);
  }
  if (!r.empty()) {
    for (int i = 0; i < r.size(); i++) {
      for (int j = 0; j < r[i]; j++) {
        for (int aa = 0; aa < c.size(); aa++) {
          for (int bb = 0; bb < c[aa]; bb++) {
            cout << "aeiou"[bb % 5];
          }
        }
        cout << endl;
      }
    }
  } else {
    cout << "-1\n";
  }
}
