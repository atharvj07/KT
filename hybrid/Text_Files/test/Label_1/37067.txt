import sys

n, m = map(int, input().split())
a = list(map(int, input().split()))
req = list(map(int, input().split()))

d = {}

for x in range(0, n):
    d[a[x]] = d.get(a[x], 0) + 1

# check if it's possible to get k 
for i in range(1, m + 1):
    if d.get(i, 0) < req[i - 1]:
        print(-1)
        sys.exit()

# get the first interval that satisfies the requirement
l = r = 0
cnt = [0] * (m + 2)
ans = n + 1
while True:
    # extend the range if needed
    while r < n and cnt[a[r]] < req[a[r] - 1]:
        cnt[a[r]] += 1
        r += 1
    
    # shrink the range if possible
    while l < r and cnt[a[l]] > req[a[l] - 1]:
        cnt[a[l]] -= 1
        l += 1

    # check if the current range is valid
    if cnt == req:
        ans = min(ans, r - l)

    # break if we reach the end of the array
    if r == n:
        break

    # extend the range
    cnt[a[r]] += 1
    r += 1

if ans == n + 1:
    print(-1)
else:
    print(ans)
