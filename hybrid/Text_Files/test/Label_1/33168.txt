import java.util.*;
import java.io.*;

class Main {

    static int n, q;
    static Task[] tasks;
    static boolean[] free;

    static class Task {
        int t, k, d;

        public Task(int t, int k, int d) {
            this.t = t;
            this.k = k;
            this.d = d;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        q = Integer.parseInt(st.nextToken());
        tasks = new Task[q];
        free = new boolean[n + 1];
        int res = 0;
        Arrays.fill(free, true);
        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int t = Integer.parseInt(st.nextToken());
            int k = Integer.parseInt(st.nextToken());
            int d = Integer.parseInt(st.nextToken());
            tasks[i] = new Task(t, k, d);
        }
        Arrays.sort(tasks, Comparator.comparingInt(a -> a.t));
        for (Task task : tasks) {
            int sum = 0;
            int cnt = 0;
            for (int i = 1; i <= n; i++) {
                if (free[i] && cnt < task.k) {
                    free[i] = false;
                    sum += i;
                    cnt++;
                }
            }
            if (cnt == task.k)
                res += sum;
            else
                res += -1;
            for (int i = 0; i < task.d; i++) {
                if (task.t + i >= n)
                    break;
                free[task.t + i] = true;
            }
        }
        System.out.println(res);
    }
}
