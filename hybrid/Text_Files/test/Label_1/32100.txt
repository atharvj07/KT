import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int[][] c = new int[3][3];
        for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j) c[i][j] = cin.nextInt();
        boolean ok = true;
        for (int i = 0; i < 3 && ok; ++i) for (int j = 1; j < 3; ++j)
            if (c[i][j] != c[i][j - 1]) ok = false;
        for (int j = 0; j < 3 && ok; ++j) for (int i = 1; i < 3; ++i)
            if (c[i][j] != c[i - 1][j]) ok = false;
        if (!ok) System.out.println("No");
        else System.out.println("Yes");
    }
}
