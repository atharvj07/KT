import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int I = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        Arrays.sort(a);
        int l = 0, r = 0;
        int min = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            while (l < n && a[l] <= a[i] - 2) {
                l++;
            }
            while (r < n && a[r] <= a[i] + 2) {
                r++;
            }
            min = Math.min(min, l + n - r);
        }
        System.out.println(min);
    }
}
