import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt();
        int k = in.nextInt();
        int[] favoriteActors = new int[k];
        for (int i = 0; i < k; i++) {
            favoriteActors[i] = in.nextInt();
        }
        int n = in.nextInt();
        int[] actorsCount = new int[n];
        List<List<Integer>> actorsList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String title = in.next();
            actorsCount[i] = in.nextInt();
            List<Integer> actors = new ArrayList<>();
            for (int j = 0; j < actorsCount[i]; j++) {
                actors.add(in.nextInt());
            }
            actorsList.add(actors);
        }
        for (int i = 0; i < n; i++) {
            int favoriteActorsCount = 0;
            for (int actor : actorsList.get(i)) {
                if (actor > 0) {
                    for (int favoriteActor : favoriteActors) {
                        if (actor == favoriteActor) {
                            favoriteActorsCount++;
                        }
                    }
                }
            }
            if (favoriteActorsCount > 0 && favoriteActorsCount == actorsCount[i]) {
                System.out.println("0");
            } else if (favoriteActorsCount == 0) {
                System.out.println("1");
            } else {
                System.out.println("2");
            }
        }
    }
}
