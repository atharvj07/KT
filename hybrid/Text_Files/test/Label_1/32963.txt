import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    String[] input = scan.nextLine().split(" ");
    String[] rook1 = input[0].split("");
    int rook1X = rook1[0].charAt(0) - 'a' + 1;
    int rook1Y = Integer.parseInt(rook1[1]);
    String[] rook2 = input[1].split("");
    int rook2X = rook2[0].charAt(0) - 'a' + 1;
    int rook2Y = Integer.parseInt(rook2[1]);
    String[] king1 = input[2].split("");
    int king1X = king1[0].charAt(0) - 'a' + 1;
    int king1Y = Integer.parseInt(king1[1]);
    String[] king2 = input[3].split("");
    int king2X = king2[0].charAt(0) - 'a' + 1;
    int king2Y = Integer.parseInt(king2[1]);

    boolean checkmate = false;

    // Check if the black king is attacked by either rook
    if (rook1X == king2X || rook1Y == king2Y || rook2X == king2X || rook2Y == king2Y) {
      checkmate = true;
    }

    // Check if the black king can move to an unbeaten position
    if (!checkmate) {
      if (king2X - 1 >= 1 && king2Y - 1 >= 1 && !(king1X == king2X - 1 && king1Y == king2Y - 1)) {
        checkmate = false;
      } else if (king2X - 1 >= 1 && !(king1X == king2X - 1 && king1Y == king2Y)) {
        checkmate = false;
      } else if (king2X - 1 >= 1 && king2Y + 1 <= 8 && !(king1X == king2X - 1 && king1Y == king2Y + 1)) {
        checkmate = false;
      } else if (king2Y - 1 >= 1 && !(king1X == king2X && king1Y == king2Y - 1)) {
        checkmate = false;
      } else if (king2Y + 1 <= 8 && !(king1X == king2X && king1Y == king2Y + 1)) {
        checkmate = false;
      } else if (king2X + 1 <= 8 && king2Y - 1 >= 1 && !(king1X == king2X + 1 && king1Y == king2Y - 1)) {
        checkmate = false;
      } else if (king2X + 1 <= 8 && !(king1X == king2X + 1 && king1Y == king2Y)) {
        checkmate = false;
      } else if (king2X + 1 <= 8 && king2Y + 1 <= 8 && !(king1X == king2X + 1 && king1Y == king2Y + 1)) {
        checkmate = false;
      }
    }

    // Print the result
    System.out.println(checkmate ? "CHECKMATE" : "OTHER");
  }
}
