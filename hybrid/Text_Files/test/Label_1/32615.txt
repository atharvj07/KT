import java.util.Scanner;

public class TestGenerator {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s1 = scanner.nextLine();
        String s2 = scanner.nextLine();
        String s3 = scanner.nextLine();

        // Find the longest common prefix of s1 and s2
        int prefixLength = 0;
        while (prefixLength < s1.length() && prefixLength < s2.length() && s1.charAt(prefixLength) == s2.charAt(prefixLength)) {
            prefixLength++;
        }

        // Construct the test string
        String testString = s1 + s2.substring(prefixLength) + s3;

        // Output the length of the test string
        System.out.println(testString.length());
    }
}
