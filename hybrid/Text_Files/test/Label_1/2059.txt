#include <iostream>
#include <vector>
typedef long long ll;
using namespace std;
vector<ll> ans;
int main() {
  ll l, r, k;
  cin >> l >> r >> k;
  for (ll p = 1;; p *= k) {
    if (p >= l && p <= r) ans.push_back(p);
    else if (p > r) break;
  }
  if (ans.size() == 0)
    cout << "-1" << endl;
  else {
    for (ll x : ans) cout << x << " ";
    cout << endl;
  }
}
