import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
            y[i] = scanner.nextInt();
        }
        int[] left = new int[n];
        int[] right = new int[n];
        for (int i = 0; i < n; i++) {
            int j = i - 1;
            while (j >= 0 && x[j] > x[i]) {
                j--;
            }
            if (j < 0) {
                left[i] = -1;
            } else {
                left[i] = j;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            int j = i + 1;
            while (j < n && x[j] < x[i]) {
                j++;
            }
            if (j == n) {
                right[i] = -1;
            } else {
                right[i] = j;
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (left[i] != -1 && right[i] != -1 && (y[i] > y[left[i]] || y[i] > y[right[i]])) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
