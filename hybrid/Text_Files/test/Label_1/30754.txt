import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int n = input.nextInt();
            String s = input.next();
            int sum = 0;
            for (int i = 0; i < n; i++) {
                sum += s.charAt(i) - '0';
            }
            if (sum % 2 != 0) {
                System.out.println("-1");
            } else {
                StringBuilder ans = new StringBuilder();
                for (int i = 0; i < n; i++) {
                    int digit = s.charAt(i) - '0';
                    if (digit % 2 == 0) {
                        ans.append(digit);
                    } else if (sum % 2 != 0) {
                        ans.append(digit);
                        sum -= digit;
                    }
                }
                if (ans.length() == 0) {
                    System.out.println("-1");
                } else {
                    System.out.println(ans);
                }
            }
        }
    }
}
