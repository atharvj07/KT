from collections import defaultdict

def count_vowels(word):
    vowels = "aeiou"
    count = 0
    last_vowel = ''
    for char in word:
        if char in vowels:
            count += 1
            last_vowel = char
    return count, last_vowel

def solve_lyrics(words):
    # Group words by the number of vowels and last vowel
    groups = defaultdict(list)
    for word in words:
        vowel_count, last_vowel = count_vowels(word)
        groups[(vowel_count, last_vowel)].append(word)

    # Intermediate groups for words with same vowel count but different last vowels
    potential_pairs = defaultdict(list)
    complete_pairs = []

    for key, group in groups.items():
        while len(group) >= 2:
            first_word = group.pop()
            second_word = group.pop()
            potential_pairs[key[0]].append((first_word, second_word))

    for count, pairs in potential_pairs.items():
        while len(pairs) >= 2:
            first_pair = pairs.pop()
            second_pair = pairs.pop()
            complete_pairs.append((first_pair[0], second_pair[0]))
            complete_pairs.append((first_pair[1], second_pair[1]))

    print(len(complete_pairs) // 2)
    for i in range(0, len(complete_pairs), 2):
        print(complete_pairs[i][0], complete_pairs[i][1])
        print(complete_pairs[i + 1][0], complete_pairs[i + 1][1])

n = int(input())
words = [input().strip() for _ in range(n)]
solve_lyrics(words)
