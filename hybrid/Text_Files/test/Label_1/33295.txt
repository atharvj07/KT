import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = input.nextInt();
        }
        int k = 0;
        int t = 0;
        while (k < n) {
            int max = 0;
            int idx = 0;
            for (int i = 0; i < n; i++) {
                if (b[i] - a[i] >= max) {
                    max = b[i] - a[i];
                    idx = i;
                }
            }
            k++;
            t += max;
            a[idx] = b[idx];
            for (int i = 0; i < n; i++) {
                if (i != idx) {
                    int diff = b[idx] - a[idx];
                    if (diff <= 0) {
                        continue;
                    }
                    int p = Math.min(diff, b[i] - a[i]);
                    t += p;
                    a[i] += p;
                    a[idx] -= p;
                }
            }
        }
        System.out.println(k + " " + t);
    }
}
