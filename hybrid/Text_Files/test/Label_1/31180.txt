import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) p[a[i] - 1] = i;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] == i) continue;
            int j = p[i];
            while (j > i) {
                if (j - i <= 5) {
                    ans++;
                    System.out.println(j - i);
                    for (int k = j - 1; k >= i; k--) {
                        System.out.print(a[k] + " ");
                        p[a[k] - 1] = k;
                    }
                    System.out.println();
                    for (int k = i; k < j; k++) {
                        System.out.print(a[k] + " ");
                    }
                    System.out.println();
                    j -= 5;
                } else {
                    ans++;
                    System.out.println(5);
                    for (int k = j - 1; k >= j - 5; k--) {
                        System.out.print(a[k] + " ");
                        p[a[k] - 1] = k;
                    }
                    System.out.println();
                    for (int k = j - 6; k < j; k++) {
                        System.out.print(a[k] + " ");
                    }
                    System.out.println();
                    j -= 5;
                }
            }
        }
        System.out.println(ans);
    }
}
