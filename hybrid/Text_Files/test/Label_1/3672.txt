#include<bits/stdc++.h>
#include<vector>
using namespace std;
typedef long long ll;
typedef pair<int,int> pii;
#define pb push_back
#define mp make_pair
#define sz(s) (int)s.size()
#define rep(i,a,b) for(int i=(a);i<=(b);++i)
#define per(i,a,b) for(int i=(a);i>=(b);--i)
#define FOIO ios_base::sync_with_stdio(false); cin.tie(NULL);
#define MEMS(arr,val) memset(arr,val,sizeof(arr));
#define all(v) v.begin(), v.end()
const int MOD = 1e9 + 7, MAXN = 1e6;

int n;
vector<string> v;
vector<ll> cost;
ll dp[100005];

bool cmp(const string& p, const string& q) {
    int x = sz(p), y = sz(q);
    if (x == y)
        return p < q;
    if (x > y) {
        return p.substr(0, y) < q;
    }
    return p < q.substr(0, x);
}

ll solve(int idx) {
    if (idx >= n)
        return 0;
    if (dp[idx] != -1)
        return dp[idx];

    ll opt1 = LLONG_MAX, opt2 = LLONG_MAX;
    if (idx == 0) {
        opt1 = solve(idx + 1);
    } else {
        if (cmp(v[idx], v[idx - 1]))
            opt1 = solve(idx + 1);
        if (cmp(v[idx - 1], v[idx]))
            opt2 = cost[idx] + solve(idx + 1);
    }
    dp[idx] = min(opt1, opt2);
    return dp[idx];
}

int main() {
    cin >> n;
    v.resize(n + 1), cost.resize(n + 1);
    rep(i, 1, n)
        cin >> cost[i];
    rep(i, 1, n)
        cin >> v[i];
    MEMS(dp, -1);
    ll res = solve(0);
    if (res == LLONG_MAX)
        res = -1;
    cout << res << endl;
    return 0;
}
