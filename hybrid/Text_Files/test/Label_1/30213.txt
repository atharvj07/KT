import java.io.*;
import java.util.*;

public class Main {
    static int n, k;
    static long[] a, b;
    static boolean[] vis;
    static ArrayList<String> ans;

    static void dfs(int x, long l, long r) {
        if (x == k) {
            Collections.reverse(ans);
            System.out.println("YES");
            for (String s : ans) System.out.println(s);
            System.exit(0);
        }
        vis[x] = true;
        if (l != -1 && a[x] > l && !vis[x + 1]) {
            ans.add(x + " L");
            dfs(x + 1, a[x], r);
            ans.remove(ans.size() - 1);
        }
        if (r != -1 && a[x] > r && !vis[x - 1]) {
            ans.add(x + " R");
            dfs(x - 1, l, a[x]);
            ans.remove(ans.size() - 1);
        }
        vis[x] = false;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(in.readLine());
        a = new long[n];
        String[] sa = in.readLine().split(" ");
        for (int i = 0; i < n; i++) a[i] = Long.parseLong(sa[i]);
        k = Integer.parseInt(in.readLine());
        b = new long[k];
        vis = new boolean[n];
        sa = in.readLine().split(" ");
        for (int i = 0; i < k; i++) b[i] = Long.parseLong(sa[i]);
        ans = new ArrayList<>();
        dfs(0, -1, -1);
        System.out.println("NO");
    }
}
