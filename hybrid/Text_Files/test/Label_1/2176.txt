#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        ll ans = (ll)sqrt(n);
        ans *= ans;
        ll ans2 =  (ll)cbrt(n);
        ans2 *= ans2 * ans2;
        cout << ans + ans2 - (ans == ans2) << endl;
    }
}
