#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int sum = a[i];
      int idx = i;
      while (idx + a[idx] <= n) {
        sum += a[idx + a[idx]];
        idx += a[idx];
      }
      ans = max(ans, sum);
    }
    cout << ans << '\n';
  }
  return 0;
}
