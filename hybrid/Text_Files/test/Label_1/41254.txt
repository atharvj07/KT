import sys

n, bx = map(int, sys.stdin.readline().split())
x = [int(i) for i in sys.stdin.readline().split()]

m, by = map(int, sys.stdin.readline().split())
y = [int(i) for i in sys.stdin.readline().split()]

if n == m and bx == by:
    if x == y:
        print('=')
    elif x < y:
        print('<')
    else:
        print('>')
elif n == m:
    if bx < by:
        print('<')
    else:
        print('>')
elif n < m:
    print('<')
else:
    print('>')
