def max_value(s):
    n = len(s)
    dp = [[0] * n for _ in range(n)]
    for i in range(n):
        dp[i][i] = int(s[i])
    # Digit 1 character
    for i in range(1, n, 2):
        for l in range(0, n - i, 2):
            r = l + i
            for j in range(l, r):
                op1 = dp[l][j] * dp[j + 2][r] if s[j + 1] == '*' else dp[l][j] + dp[j + 2][r]
                op2 = dp[l][j] if s[j + 1] == '*' else dp[l][j] * dp[j + 2][r]
                dp[l][r] = max(dp[l][r], op1, op2)
    return dp[0][n - 1]

s = input()
print(max_value(s))
