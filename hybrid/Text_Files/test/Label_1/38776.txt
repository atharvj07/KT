stack = list(input())

while 1:
    i = 0
    while i < len(stack) - 1:
        if stack[i] == 'y' and stack[i+1] == 'x':
            stack[i], stack[i+1] = stack[i+1], stack[i]
            break
        elif stack[i] == 'x' and stack[i+1] == 'y':
            stack.pop(i)
            stack.pop(i)
            break
        i += 1
    else:
        break
print(''.join(stack))
