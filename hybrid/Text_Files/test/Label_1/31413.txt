import java.util.*;

class Main {

    private static class Client implements Comparable<Client> {
        private final int hour;
        private final int minute;

        public Client(int hour, int minute) {
            this.hour = hour;
            this.minute = minute;
        }

        @Override
        public int compareTo(Client other) {
            if (this.hour < other.hour) {
                return -1;
            } else if (this.hour > other.hour) {
                return 1;
            } else {
                return Integer.compare(this.minute, other.minute);
            }
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Client[] clients = new Client[n];
        for (int i = 0; i < n; i++) {
            int hour = scanner.nextInt();
            int minute = scanner.nextInt();
            clients[i] = new Client(hour, minute);
        }
        scanner.close();
        Arrays.sort(clients);

        int maxCashes = 0;
        int currentCashes = 0;
        for (Client client : clients) {
            if (client.hour == 0 && client.minute == 0) {
                currentCashes++;
                if (currentCashes > maxCashes) {
                    maxCashes = currentCashes;
                }
            } else {
                currentCashes--;
                if (currentCashes < 0) {
                    currentCashes = 1;
                }
                if (currentCashes > maxCashes) {
                    maxCashes = currentCashes;
                }
            }
        }
        System.out.println(maxCashes);
    }
}
