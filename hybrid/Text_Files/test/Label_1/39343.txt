n, m = [int(i) for i in input().split()]
g = [[int(i) for i in input().split()] for _ in range(n)]

def solve():
    ans = []
    for i in range(n):
        for j in range(m):
            if g[i][j] % 2 != 0:
                return -1
            if i % 2 == j % 2:
                ans.append("row %d" % (i+1))
            else:
                ans.append("col %d" % (j+1))
    return len(ans), ans

ans = solve()
if ans == -1:
    print(ans)
else:
    k, moves = ans
    print(k)
    for move in moves:
        print(move)
