#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }
  sort(w.begin(), w.end());
  int ans = 0, cnt = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (cnt + w[i] <= k) {
      cnt += w[i];
    } else {
      ans++;
      cnt = w[i];
    }
  }
  cout << ans << endl;
  return 0;
}
