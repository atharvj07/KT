#include<iostream>
#include<vector>
#include<set>
#include<map>
#include<algorithm>
using namespace std;
int main()
{
	int n,x; cin>>n;
	vector<int>v,ans(n);
	set<pair<int,int>>st;
	for(int i=0; i<n*n; i++)
	{
		cin>>x;
		if(st.count({x,i}) == 0)
		{
			v.push_back(x);
			st.insert({x,i});
		}
	}
	sort(v.begin(),v.end());
	for(int i=0; i<n; i++)
	{
		for(int j=0; j<n; j++)
		{
			if(i == j) ans[i] = v[0];
			else
			{
				int tmp = __gcd(v[0],v[j+1]);
				if(st.count({tmp,i}) && st.count({tmp,j})) ans[i] = v[0], ans[j] = v[j+1];
			}
		}
	}
	for(auto &k:ans) cout<<k<<" ";
}
