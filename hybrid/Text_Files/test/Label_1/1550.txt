#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  int ans = 0;
  for (int i = 1; i * i <= n; ++i) {
    if (n % i == 0) {
      int t1 = i;
      int t2 = n / i;
      if (t1 != t2 && t2 > 1) {
        ++ans;
      }
      if (t1 > 1) {
        ++ans;
      }
    }
  }
  std::cout << ans << '\n';
  return 0;
}
