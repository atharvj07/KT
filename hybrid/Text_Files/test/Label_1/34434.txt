import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    static long[][] dp;
    static long[] prefix;
    static long[] ans;
    static long mod = (long) 1e9 + 7;

    static long orientedArea(long[] p, long[] q, long[] r) {
        return (q[1] - p[1]) * (r[0] - q[0]) - (q[0] - p[0]) * (r[1] - q[1]);
    }

    static long getPerimeter(List<long[]> points, int from, int to) {
        if (to - from + 1 <= 0) return 0;
        long ret = prefix[to] - (from == 0 ? 0 : prefix[from - 1]);
        if (to - from + 1 == 3) return ret;
        return ret + Math.abs(points.get(to)[0] - points.get(from)[0]) + Math.abs(points.get(to)[1] - points.get(from)[1]);
    }

    static void solve(List<long[]> points, int n) {
        dp[0][0] = getPerimeter(points, 0, 0);
        dp[0][1] = getPerimeter(points, 0, 1);
        dp[0][2] = getPerimeter(points, 0, 2);
        dp[1][0] = 0;
        dp[1][1] = getPerimeter(points, 1, 1);
        dp[1][2] = getPerimeter(points, 1, 2);
        if (n == 3) {
            ans[3] = Math.max(dp[0][0], Math.max(dp[0][1], dp[0][2]));
        } else {
            for (int end = 3; end <= n; end++) {
                for (int start = 0; start <= end - 3; start++) {
                    if (orientedArea(points.get(start), points.get(end - 1), points.get((start + end + 2) % n)) < 0) continue;
                    long curr = getPerimeter(points, start, end - 1);
                    long prev = start == 0 ? 0 : dp[start - 1][end - 2];
                    dp[start][end - 1] = Math.max(dp[start][end - 1], prev + curr);
                }
            }
            for (int i = 3; i <= n; i++) {
                for (int j = 0; j <= i - 3; j++) {
                    ans[i] = Math.max(ans[i], dp[j][i - 2]);
                }
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        dp = new long[n][n];
        prefix = new long[n];
        ans = new long[n + 1];
        List<long[]> points = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] parts = reader.readLine().split(" ");
            long[] point = new long[2];
            point[0] = Long.parseLong(parts[0]);
            point[1] = Long.parseLong(parts[1]);
            points.add(point);
        }
        for (int i = 1; i < n; i++) {
            long[] p = points.get(i - 1);
            long[] q = points.get(i % n);
            prefix[i] = prefix[i - 1] + Math.abs(q[0] - p[0]) + Math.abs(q[1] - p[1]);
        }
        solve(points, n);
        for (int i = 3; i <= n; i++) {
            System.out.println(ans[i] % mod);
        }
    }
}
