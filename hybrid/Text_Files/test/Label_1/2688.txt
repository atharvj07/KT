#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    int lon, lat;
    bool north = false, south = false;
    for (int i = 0; i < n; i++) {
        int d; string s;
        cin >> d >> s;
        if (s == "South" && lon == 0) {
            south = true;
            lat -= d;
            if (lat < -40000) return cout << "NO", 0;
        } else if (s == "North" && lon == 0) {
            north = true;
            lat += d;
            if (lat > 40000) return cout << "NO", 0;
        } else if (s == "East") {
            lon += d;
            if (lon > 40000 || (lon == 40000 && (north || south))) return cout << "NO", 0;
        } else if (s == "West") {
            lon -= d;
            if (lon < 0 || (lon == 0 && (north || south))) return cout << "NO", 0;
        }
    }
    if (lon != 0 || lat != 0) return cout << "NO", 0;
    cout << "YES";
}
