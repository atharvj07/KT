import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Long, Integer> map_x = new HashMap<>();
        Map<Long, Integer> map_y = new HashMap<>();
        for (int i = 0; i < n; i++) {
            long x = sc.nextInt();
            long y = sc.nextInt();
            map_x.put(x, map_x.getOrDefault(x, 0) + 1);
            map_y.put(y, map_y.getOrDefault(y, 0) + 1);
        }
        long ans = 0;
        for (long x : map_x.keySet()) {
            if (map_x.getOrDefault(x + 1, 0) > 0 && map_x.getOrDefault(x - 1, 0) > 0) {
                ans += map_x.getOrDefault(x, 0) * map_x.getOrDefault(x + 1, 0) * map_x.getOrDefault(x - 1, 0);
            }
        }
        for (long y : map_y.keySet()) {
            if (map_y.getOrDefault(y + 1, 0) > 0 && map_y.getOrDefault(y - 1, 0) > 0) {
                ans += map_y.getOrDefault(y, 0) * map_y.getOrDefault(y + 1, 0) * map_y.getOrDefault(y - 1, 0);
            }
        }
        System.out.println(ans / 3);
    }
}
