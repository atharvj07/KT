import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine(), t = sc.nextLine();
        int m = s.length(), n = t.length();
        int[][] dp = new int[m + 1][n + 1];
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                dp[i][j] = (s.charAt(i - 1) == t.charAt(j - 1) ? dp[i - 1][j - 1] + 1 : dp[i - 1][j]);
            }
        }
        long ans = 0;
        for (int[] d : dp) {
            for (int v : d) {
                ans += v;
            }
        }
        System.out.println(ans % (long) (1e9 + 7));
    }
}
