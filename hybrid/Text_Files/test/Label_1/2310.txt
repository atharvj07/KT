#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int ans = 0;
    bool l = 0, r = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '*') {
        if (!l) {
          l = 1;
          ans++;
        } else if (i == n - 1) {
          ans++;
        } else if (i - ans + 1 > k) {
          ans++;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
