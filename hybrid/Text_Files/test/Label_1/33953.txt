import java.util.*;
class Main{
    static long mod=998244353L;
    static long f(Long[]p){
        if(p==null||p.length==0)return 1;
        int n=p.length;
        p=Arrays.stream(p).distinct().sorted().toArray(Long[]::new);
        long res=1;
        for(long x:p){
            long cnt=0,i=x;
            while(i>1){
                i/=x;
                cnt++;
            }
            res=(res*(cnt+1))%mod;
        }
        return res;
    }
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        Long[]P=new Long[n];
        for(int i=0;i<n;i++)P[i]=sc.nextLong();
        System.out.println(f(P));
    }
}
