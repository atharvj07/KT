import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        long[] times = new long[n];
        long[][] coords = new long[n][2];
        for (int i = 0; i < n; i++) {
            times[i] = sc.nextLong();
            coords[i][0] = sc.nextLong();
            coords[i][1] = sc.nextLong();
        }
        Map<Long, Long> firstMoves = new HashMap<>();
        Map<Long, String> programs = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            long timeDiff = times[i + 1] - times[i];
            long xDiff = coords[i + 1][0] - coords[i][0];
            long yDiff = coords[i + 1][1] - coords[i][1];
            if (timeDiff % l != 0 || Math.abs(xDiff) % l != 0 || Math.abs(yDiff) % l != 0) {
                System.out.println("NO");
                return;
            }
            long period = timeDiff / l;
            long firstTime = times[i] % period;
            if (firstMoves.containsKey(period) && firstMoves.get(period) != firstTime) {
                System.out.println("NO");
                return;
            }
            firstMoves.put(period, firstTime);
            StringBuilder program = new StringBuilder();
            for (int j = 0; j < l; j++) {
                if (xDiff > 0) {
                    program.append("R");
                    xDiff--;
                } else if (yDiff > 0) {
                    program.append("U");
                    yDiff--;
                } else if (xDiff < 0) {
                    program.append("L");
                    xDiff++;
                } else {
                    program.append("D");
                    yDiff++;
                }
            }
            if (programs.containsKey(period) && !programs.get(period).equals(program.toString())) {
                System.out.println("NO");
                return;
            }
            programs.put(period, program.toString());
        }
        System.out.println(programs.get(times[0] % firstMoves.get(times[0] % firstMoves.size())));
    }
}
