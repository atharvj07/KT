import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class GeoPainting {

    static int[][] grid;
    static int[][] symbols;
    static boolean[][] visited;
    static int n, m;
    static int startX, startY;
    static int[] dx = {0, 1, 0, -1};
    static int[] dy = {1, 0, -1, 0};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        grid = new int[n][m];
        symbols = new int[n][m];
        visited = new boolean[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                symbols[i][j] = scanner.nextInt();
            }
        }

        startX = scanner.nextInt() - 1;
        startY = scanner.nextInt() - 1;

        System.out.println(paint());
    }

    static int paint() {
        Queue<Pair> queue = new LinkedList<>();
        int color = symbols[startX][startY];
        queue.offer(new Pair(startX, startY));
        visited[startX][startY] = true;
        int repaintCount = 0;

        while (!queue.isEmpty()) {
            Pair current = queue.poll();
            int x = current.x;
            int y = current.y;

            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];

                if (isValid(nx, ny) && !visited[nx][ny] && grid[nx][ny] == grid[x][y]) {
                    visited[nx][ny] = true;

                    if (symbols[nx][ny] != -1 && symbols[nx][ny] != color) {
                        queue.offer(new Pair(nx, ny));
                    } else if (symbols[nx][ny] == -1) {
                        symbols[nx][ny] = color;
                        queue.offer(new Pair(nx, ny));
                        repaintCount++;
                    }
                }
            }
        }

        return repaintCount;
    }

    static boolean isValid(int x, int y) {
        return x >= 0 && x < n && y >= 0 && y < m;
    }

    static class Pair {
        int x;
        int y;

        public Pair(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
