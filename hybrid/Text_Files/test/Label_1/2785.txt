#include <bits/stdc++.h>
using namespace std;
const int N = 100;
vector<pair<pair<int, int>, int>> blocks[N];
int n,dp[N][N][N][5],vis[N][N][N][5];
int func(int l,int r,int pre,int lastclr)
{
	if(l>r) return 0;
	int &ret = dp[l][r][pre][lastclr];
	if(vis[l][r][pre][lastclr]==1) return ret;
	vis[l][r][pre][lastclr] = 1;
	///take l
	if(lastclr==0) ret = max(ret,func(l+1,r,l,blocks[l][pre].second));
	else if(lastclr==blocks[l][pre].second) ret = max(ret,blocks[l][pre].first.first+func(l+1,r,pre,lastclr));
	else if(lastclr==blocks[l][pre].first.second) ret = max(ret,blocks[l][pre].first.first+func(l+1,r,pre,blocks[l][pre].second));
	///take r
	if(lastclr==0) ret = max(ret,func(l,r-1,r,blocks[r][pre].first.second));
	else if(lastclr==blocks[r][pre].second) ret = max(ret,blocks[r][pre].first.first+func(l,r-1,pre,lastclr));
	else if(lastclr==blocks[r][pre].first.second) ret = max(ret,blocks[r][pre].first.first+func(l,r-1,pre,blocks[r][pre].first.second));
	///skip
	ret = max(ret,func(l+1,r,pre,lastclr));
	ret = max(ret,func(l,r-1,pre,lastclr));
	return ret;
}
int main()
{
	cin>>n;
	for(int i=1;i<=n;i++)
	{
		int x,y,z;
		cin>>x>>y>>z;
		blocks[i].push_back({{x,z},y});
		blocks[i].push_back({{z,x},y});
	}
	memset(dp,-1,sizeof dp);
	cout<<func(1,n,0,0)<<endl;
	return 0;
}
