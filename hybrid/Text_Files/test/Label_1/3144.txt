#include <bits/stdc++.h>
using namespace std;
using ll = long long;
// use sieve of Eratosthenes to compute all divisors of k

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }

  vector<bool> is_divisible(k);

  for (int i = 0; i < n; ++i) {
    int rem = 0;
    ll ai = a[i];
    while (ai > 0) {
      rem = ((10 * rem) + (ai % 10)) % k;
      if (rem == 0) {
        is_divisible[a[i]] = true;
        break;
      }
      ai = ai / 10;
    }
  }

  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      if (is_divisible[__gcd(a[i], a[j])] == true) {
        ans += 1;
      }
    }
  }
  cout << ans << "\n";
}
