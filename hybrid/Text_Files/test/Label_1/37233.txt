t = int(input())
for _ in range(t):
    n = int(input())
    r = list(map(int, input().split()))
    a, b = 0, 0
    for x in r:
        if x == 1:
            a += 1
            b += 1
        elif x == 2:
            a -= 1
            b -= 1
        else:
            if a > b:
                a += 1
            else:
                b += 1
    print(a + b)
