import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            String s = scanner.next();
            String t = scanner.next();
            int[] pos = new int[26];
            for (int j = 0; j < n; j++) {
                pos[s.charAt(j) - 'a'] = j;
            }
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) == t.charAt(j)) {
                    continue;
                }
                int p = pos[t.charAt(j) - 'a'];
                if (p < j) {
                    ans += j - p;
                    for (int k = p; k < j; k++) {
                        pos[s.charAt(k) - 'a'] = k + 1;
                    }
                } else {
                    ans += n - 1 - p + j;
                    for (int k = p; k > j; k--) {
                        pos[s.charAt(k) - 'a'] = k - 1;
                    }
                }
                s = s.charAt(j) + s.substring(0, j) + s.substring(j + 1);
                pos[s.charAt(j) - 'a'] = j;
            }
            if (s.equals(t)) {
                System.out.println(ans);
            } else {
                System.out.println(-1);
            }
        }
    }
}
