#include <iostream>
#include <vector>
#include <set>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<int> a(n);
    for (int i = 0; i < n; ++i) std::cin >> a[i];
    std::vector<std::set<int>> edges(n);
    for (int i = 0; i < m; ++i) {
        int x, y;
        std::cin >> x >> y;
        --x; --y;
        edges[x].insert(y);
        edges[y].insert(x);
    }
    double ans = 0.;
    for (int i = 0; i < n; ++i) {
        std::vector<int> dist(n, -1);
        std::vector<int> min_on_path(n, -1);
        std::set<std::pair<int, int>> que;
        que.insert({0, i});
        dist[i] = 0;
        min_on_path[i] = a[i];
        while (!que.empty()) {
            auto [d, v] = *que.begin();
            que.erase(que.begin());
            for (int u : edges[v]) {
                if (dist[u] == -1) {
                    dist[u] = d + 1;
                    min_on_path[u] = std::min(min_on_path[v], a[u]);
                    que.insert({d + 1, u});
                }
            }
        }
        for (int j = 0; j < n; ++j) {
            if (i != j) {
                if (dist[j] != -1) ans += (double)min_on_path[j];
            }
        }
    }
    ans /= (double)(n - 1);
    ans /= (double)n;
    std::cout.precision(9);
    std::cout << ans << std::endl;
    return 0;
}
