#include <bits/stdc++.h>
#define ll long long
#define MAXN 100005
using namespace std;
ll t[MAXN],v[MAXN],a[MAXN];
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    ll tc;
    cin >> tc;
    while (tc--) {
        ll n, x;
        cin >> n >> x;
        ll pre = 0;
        ll ans = -1;
        for (ll i = 0; i < n; i++) {
            cin >> a[i];
            pre += a[i];
            v[pre % x]++;
        }
        for (ll i = 0; i < x; i++) {
            if (v[i])
                ans = n;
            for (ll j = 0; j < v[i]; j++)
                ans = max(ans, n - v[i]);
        }
        cout << ans << "\n";
        memset(t, 0, sizeof t);
        memset(v, 0, sizeof v);
        memset(a, 0, sizeof a);
    }
    return 0;
}
