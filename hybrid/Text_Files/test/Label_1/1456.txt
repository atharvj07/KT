#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, i, j, f = 0;
    cin >> n >> m;
    char a[n][m + 1];
    for (i = 0; i < n; i++) cin >> a[i];
    for (i = 1; i + 1 < n; i++) {
        for (j = 1; j + 1 < m; j++) {
            if (a[i][j] == '.' && a[i - 1][j] == '#' && a[i + 1][j] == '#' && a[i][j - 1] == '#' && a[i][j + 1] == '#') {
                cout << "YES" << endl;
                return 0;
            }
        }
    }
    cout << "NO" << endl;
    return 0;
}
