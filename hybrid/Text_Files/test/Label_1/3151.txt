#include <bits/stdc++.h>
typedef long long int ll;
using namespace std;
int a, b, x = 0, y = 0;
string s;
void go(char c) {
    if (c == 'R')
        x++;
    else if (c == 'L')
        x--;
    else if (c == 'U')
        y++;
    else
        y--;
}
int main() {
    cin >> a >> b >> s;
    for (int i = 0;; i = (i + 1) % s.size()) {
        go(s[i]);
        if (x == a && y == b)
            return cout << "Yes", 0;
    }
    cout << "No";
}
