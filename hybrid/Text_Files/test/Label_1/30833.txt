import java.util.*;
import java.io.*;

class Main {
  static final int mod = 1000000007;

  int n, m, k;
  char[][] a;

  long dfs(int x, int y) {
    if (x < 1 || x > n || y < 1 || y > m) return 1;
    if (a[x][y] == '0') return 1;
    if (a[x][y] == '2') return 0;
    a[x][y] = '2';
    return (dfs(x - 1, y) + dfs(x + 1, y) + dfs(x, y - 1) + dfs(x, y + 1)) % mod;
  }

  public static void main(String[] args) throws Exception {
    Main m = new Main();
    Scanner in = new Scanner(System.in);
    m.n = in.nextInt();
    m.m = in.nextInt();
    m.k = in.nextInt();
    m.a = new char[m.n + 2][m.m + 2];
    for (int i = 1; i <= m.n; i++) m.a[i] = ('0' + in.next()).toCharArray();
    long res = 0;
    for (int i = 1; i <= m.n; i++) {
      for (int j = 1; j <= m.m; j++) {
        if (m.a[i][j] == '1') {
          res = (res + m.dfs(i, j)) % mod;
        }
      }
    }
    System.out.println(res);
  }
}
