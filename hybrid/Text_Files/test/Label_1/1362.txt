#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k, d;
  cin >> n >> k >> d;
  if (d > 1) {
    for (int i = 0; i < d; i++) {
      for (int j = 0; j < n / k; j++) {
        for (int p = 0; p < k; p++) cout << j * k + p + 1 << " ";
      }
      for (int p = 0; p < n % k; p++) cout << n / k * k + p + 1 << " ";
      cout << endl;
    }
  } else
    cout << "-1\n";
}
