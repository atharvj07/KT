n = int(input())
a = list(map(int, input().split()))

b = [-1] * n
current_mex = 0
used = set()

for i in range(n):
    if a[i] != current_mex:
        if current_mex not in used:
            b[i] = current_mex
            used.add(current_mex)
            current_mex += 1
            while current_mex in used:
                current_mex += 1
        else:
            print(-1)
            exit()
    else:
        for new_value in range(0, 1000001):
            if new_value not in used and new_value != current_mex:
                b[i] = new_value
                used.add(new_value)
                break

        if a[i] == a[i - 1]:
            continue
        current_mex += 1
        while current_mex in used:
            current_mex += 1

if -1 in b:
    print(-1)
else:
    print(' '.join(map(str, b)))
