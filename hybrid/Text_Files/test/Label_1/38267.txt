n, q = map(int, input().split())
a = list(map(int, input().split()))

# Find the maximum element and its position
max_a = max(a)
max_pos = a.index(max_a)

# Precompute the pairs before the list becomes stable
precomputed_pairs = []
for i in range(max_pos):
    A = a.pop(0)
    B = a.pop(0)
    precomputed_pairs.append((A, B))
    if A > B:
        a.insert(0, A)
        a.append(B)
    else:
        a.insert(0, B)
        a.append(A)

# Process each query
for _ in range(q):
    m = int(input())
    if m <= max_pos:
        print(*precomputed_pairs[m-1])
    else:
        print(max_a, a[(m - max_pos - 1) % (n - 1) + 1])
