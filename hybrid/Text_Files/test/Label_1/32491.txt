import java.util.Scanner;

public class GreedySubsequence {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] result = new int[n - k + 1];
        for (int i = 0; i <= n - k; i++) {
            result[i] = solve(a, i, k);
        }

        for (int i = 0; i < n - k + 1; i++) {
            System.out.print(result[i] + " ");
        }
    }

    static int solve(int[] a, int start, int k) {
        int[] d = new int[k];
        int[] p = new int[k];
        int length = 0;
        for (int i = start; i < start + k; i++) {
            int pos = -1;
            for (int j = 0; j < length; j++) {
                if (a[d[j]] < a[i]) {
                    pos = j;
                    break;
                }
            }
            if (pos == -1) {
                if (length > 0) {
                    p[length] = d[length - 1];
                }
                d[length++] = i;
            } else {
                if (pos == length - 1) {
                    p[length] = d[length - 1];
                } else {
                    p[length] = d[pos];
                }
                d[pos] = i;
            }
        }
        return length;
    }
}
