#include<bits/stdc++.h>
#define ll long long
#define mod 1000000007
using namespace std;
ll dp[100005][1005];
ll a[100005];
int main() {
    ll n,i,j;
    cin>>n;
    for(i=1;i<=n;i++) cin>>a[i];
    dp[0][1]=1;
    dp[0][0]=1;
    for(i=1;i<=n;i++) {
        for(j=1;j<=1000;j++) {
            dp[i][j] = (dp[i-1][j] + dp[i-1][j/__gcd(a[i],j)])%mod;
        }
    }
    cout<<(dp[n][1]-1+mod)%mod;
}
