#include <vector>
#include <map>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, h;
    cin >> n >> h;
    vector<int> a(n);
    for (int &x: a) {
        cin >> x;
    }
    int mx = *max_element(a.begin(), a.end());
    cout << (mx <= h ? n : 2 * (n - (upper_bound(a.begin(), a.end(), h) - a.begin())));
    return 0;
}
