import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int t = sc.nextInt();
        char[] c = sc.next().toCharArray();
        int last = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (last == 0 || c[i] == c[last - 1]) {
                if (t >= a) {
                    t -= a;
                    last = i + 1;
                    ans++;
                }
            } else {
                if (t >= b + a) {
                    t -= b + a;
                    last = i + 1;
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
