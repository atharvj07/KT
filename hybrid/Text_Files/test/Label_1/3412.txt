#include <iostream>
#include <string>
#include <cmath>
using namespace std;
bool cmp_val(string a, string b) {
    long double va = 0, vb = 0;
    long double q = (sqrt(5) + 1) / 2, q_powi = 1;
    int len_a = a.length(), len_b = b.length();
    for (int i = len_a - 1; i >= 0; i--) {
        va += (a[i] - '0') * q_powi;
        q_powi *= q;
    }
    q_powi = 1;
    for (int i = len_b - 1; i >= 0; i--) {
        vb += (b[i] - '0') * q_powi;
        q_powi *= q;
    }
    if (va > vb) return true;
    if (va < vb) return false;
    return true;
}
int main() {
    string a, b;
    cin >> a >> b;
    if (cmp_val(a, b)) cout << ">";
    else if (cmp_val(b, a)) cout << "<";
    else cout << "=";
    cout << endl;
}
