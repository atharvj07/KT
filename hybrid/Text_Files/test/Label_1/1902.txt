#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  int n;
  cin >> n;
  vector<vector<ll>> w(n, vector<ll>(n)), a(n, vector<ll>(n)), b(n, vector<ll>(n));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cin >> w[i][j]; }
  }

  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      a[i][j] = (w[i][j] % n + n) % n;
      b[i][j] = (a[i][j] - w[i][j] + n) % n;
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cout << a[i][j] << " "; }
    cout << endl;
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) { cout << b[i][j] << " "; }
    cout << endl;
  }
}
