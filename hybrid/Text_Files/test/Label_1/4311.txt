#include<bits/stdc++.h>
using namespace std;

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int q;
    cin >> q;
    while (q--) {
        string s;
        cin >> s;
        if (s.size() <=2 || s == "1")
            cout << -1 << '\n';
        else if (s.size() % 2 == 1)
            cout << s.size()/2 << endl;
        else
            cout << s.size()/2 - 1 << endl;
    }
}
