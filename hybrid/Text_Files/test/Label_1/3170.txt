#include<iostream>
using namespace std;
int a[105], b[105];
int main() {
    int m, n, min, max;
    cin >> n >> m >> min >> max;
    for (int i = 1; i <= m; i++) { cin >> a[i]; b[a[i]]++; }
    if (b[min] && b[max]) { cout << "Correct" << endl; }
    else { cout << "Incorrect" << endl; }
    return 0;
}
