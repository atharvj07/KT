#include <bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
#define fi first
#define se second
int main(){
	ios::sync_with_stdio(0),cin.tie(0);
	int n,r;
	cin>>n>>r;
	vector<pii> v(n);
	for(int i=0;i<n;i++){
		cin>>v[i].fi>>v[i].se;
	}
	sort(v.begin(),v.end());
	int ans=0;
	for(int i=0;i<n;i++){
		if(r>=v[i].fi){
			r+=v[i].se;
			ans++;
		}
		else break;
	}
	cout<<ans<<endl;
}
