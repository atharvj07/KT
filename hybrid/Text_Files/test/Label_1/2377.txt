#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  vector<int> ans(n + 1, 1);
  for (int i = 0; i < n; i++) {
    if (p[i] > 1) {
      ans[i + 1] = ans[i] + 1;
    }
    for (int j = p[i]; j < n; j++) {
      if (p[j] < p[i]) {
        ans[i + 1] = ans[i] + 1;
      }
    }
  }
  for (int i = 0; i < n + 1; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
