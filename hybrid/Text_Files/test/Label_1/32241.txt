import java.util.Scanner;

public class CoinPayment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            int p = scanner.nextInt();
            if (p == 0) {
                break;
            }
            int n1 = scanner.nextInt();
            int n5 = scanner.nextInt();
            int n10 = scanner.nextInt();
            int n50 = scanner.nextInt();
            int n100 = scanner.nextInt();
            int n500 = scanner.nextInt();

            int minCoins = calculateMinCoins(p, n1, n5, n10, n50, n100, n500);
            System.out.println(minCoins);
        }
    }

    private static int calculateMinCoins(int p, int n1, int n5, int n10, int n50, int n100, int n500) {
        int[] coinValues = {500, 100, 50, 10, 5, 1};
        int[] coinCounts = {n500, n100, n50, n10, n5, n1};

        int paidCoins = 0;
        int returnedCoins = 0;

        for (int i = 0; i < coinValues.length; i++) {
            int coinValue = coinValues[i];
            int coinCount = coinCounts[i];

            int maxCoins = p / coinValue;
            if (maxCoins > coinCount) {
                maxCoins = coinCount;
            }

            p -= maxCoins * coinValue;
            paidCoins += maxCoins;
        }

        returnedCoins = p;

        return paidCoins + returnedCoins;
    }
}
