for tc in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    s = sum(a)
    k = n//2 + s%2
    for i in range(n):
        if i%2==s%2 and a[i]==0:
            k -= 1
        elif i%2!=s%2 and a[i]:
            k -= 1
    print(k)
    for i in range(n):
        if i%2==s%2 and a[i]==0 or i%2!=s%2 and a[i]:
            print(a[i], end=' ')
    print()
