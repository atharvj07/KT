#include <bits/stdc++.h>
using namespace std;
const int N = 26;
typedef long long ll;
ll A[N], B[N];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> A[i];
    sort(A, A + n, greater<ll>());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int j = 1;
        while (j * (j + 1) / 2 <= A[i]) j++;
        ans += j - 1;
        A[i] -= (j - 1) * j / 2;
        if (A[i] == 0) continue;
        B[j - 1] = max(B[j - 1], A[i]);
    }
    for (int i = 0; i < N; i++) ans += B[i] / (i + 1);
    cout << ans;
    return 0;
}
