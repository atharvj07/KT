#include <iostream>

typedef long long ll;
const ll MOD = 1e9 + 7;
int main() {
  std::string s;
  std::cin >> s;
  int n = s.size();
  int l = 0, r = 0;
  ll ans = 1;
  for(int i = 0;i < n;++i){
    if(s[i] == '('){
      l++;
      ans = (ans + 2) % MOD;
    }else if(l){
      l--;
      r++;
      ans = (ans + 1) % MOD;
    }
  }
  std::cout << ans << '\n';
  return 0;
}
