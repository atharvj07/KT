# Read the board
board = [input() for i in range(8)]

# Check if player A can win
for i in range(8):
    if board[0][i] == 'W':
        print('A')
        exit()

# Check if player B can win
for i in range(8):
    if board[7][i] == 'B':
        print('B')
        exit()

# Check if player A can block player B
for i in range(8):
    for j in range(8):
        if board[j][i] == 'B' and j != 0:
            if board[j - 1][i] == '.':
                print('A')
                exit()

# Otherwise, player B wins
print('B')
