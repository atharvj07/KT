import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] gates = new int[m][w];
        for (int i = 0; i < m; i++) {
            String s = sc.next();
            for (int j = 0; j < w; j++) {
                gates[i][j] = s.charAt(j) - 'A';
            }
        }
        for (int i = 0; i < m; i++) {
            int count = 0;
            for (int j = 0; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    int x = a[j];
                    int y = a[k];
                    int z = 0;
                    for (int l = 0; l < w; l++) {
                        int xl = (x >> l) & 1;
                        int yl = (y >> l) & 1;
                        int gl = gates[i][l];
                        int zl = 0;
                        switch (gl) {
                            case 0:
                                zl = xl & yl;
                                break;
                            case 1:
                                zl = xl | yl;
                                break;
                            case 2:
                                zl = xl ^ yl;
                                break;
                            case 3:
                                zl = 1 ^ (xl & yl);
                                break;
                            case 4:
                                zl = 1 ^ (xl | yl);
                                break;
                            case 5:
                                zl = 1 ^ (xl ^ yl);
                                break;
                        }
                        z |= zl << l;
                    }
                    if (z == 0) {
                        count++;
                    }
                }
            }
            System.out.println(count);
        }
    }
}
