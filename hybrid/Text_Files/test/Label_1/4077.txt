#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; ++i) { cin >> a[i]; }
        sort(a.begin(), a.end());
        if (k == 1) {
            cout << "YES" << endl;
        } else if (k > 1) {
            cout << "YES" << endl;
        } else {
            bool check = true;
            for (int i = 1; i < n; ++i) {
                if (a[i] < a[i - 1]) {
                    check = false;
                    break;
                }
            }
            if (check) {
                cout << "YES" << endl;
            } else {
                cout << "NO" << endl;
            }
        }
    }
    return 0;
}
