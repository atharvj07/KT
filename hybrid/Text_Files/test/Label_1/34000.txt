import java.util.Scanner;
import java.util.ArrayList;
import java.math.BigInteger;

public class MakeItOne {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        ArrayList<Integer> numbers = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            numbers.add(scanner.nextInt());
        }

        // Find the greatest common divisor of all numbers
        BigInteger gcd = BigInteger.valueOf(numbers.get(0));
        for (int i = 1; i < n; i++) {
            gcd = gcd.gcd(BigInteger.valueOf(numbers.get(i)));
        }

        // If the gcd is 1, output the size of the smallest subset with gcd equal to 1
        if (gcd.equals(BigInteger.ONE)) {
            System.out.println(numbers.size());
        } else {
            System.out.println(-1);
        }
    }
}
