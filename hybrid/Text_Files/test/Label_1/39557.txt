t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    x = int(input())
    
    dp = [0] * n
    dp[0] = 1
    sum_segment = a[0]
    len_segment = 1
    
    for i in range(1, n):
        sum_segment += a[i]
        len_segment += 1
        if sum_segment >= len_segment * x:
            dp[i] = dp[i - 1] + 1
        else:
            sum_segment -= (a[i - len_segment + 1] + a[i])
            len_segment -= 2
            dp[i] = dp[i - 1]
            
            if i - len_segment > 0:
                dp[i] = max(dp[i], dp[i - len_segment - 1] + len_segment + 1)
            else:
                dp[i] = max(dp[i], len_segment + 1)
                
    print(dp[-1])
