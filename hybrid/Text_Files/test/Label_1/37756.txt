import heapq

n, m = map(int, input().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

s, b, k, h = map(int, input().split())
space = []
for i in range(s):
    c, d, f = map(int, input().split())
    space.append((f, -d, c))
base = []
for i in range(b):
    c, d = map(int, input().split())
    base.append((d, c))

heapq.heapify(space)
base.sort()
ans = 0
while base:
    bd, bi = base.pop()
    while space and -space[0][1] >= bd:
        _, sd, si = heapq.heappop(space)
        if adj[si][0] == bi:
            continue
        dist = len(adj[si]) // 2
        if -sd >= dist:
            ans += k
            break
    else:
        ans += h
print(ans)
