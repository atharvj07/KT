#include <iostream>
#include <unordered_map>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, x, y;
  cin >> n >> x >> y;
  unordered_map<int, bool> col;
  unordered_map<int, bool> diag1;
  unordered_map<int, bool> diag2;
  col[x] = true;
  diag1[x - y] = true;
  diag2[x + y] = true;
  while (n--) {
    char type;
    int x, y;
    cin >> type >> x >> y;
    if (type == 'R') {
      col[x] = true;
    } else if (type == 'B') {
      diag1[x - y] = true;
      diag2[x + y] = true;
    } else if (type == 'Q') {
      col[x] = true;
      diag1[x - y] = true;
      diag2[x + y] = true;
    }
  }
  if (col[x] || diag1[x - y] || diag2[x + y]) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
