import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        List<Song> songs = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            String s = sc.next();
            int t = sc.nextInt();
            songs.add(new Song(s, t));
        }
        String X = sc.next();
        int idx = -1;
        for (int i = 0; i < songs.size(); i++) {
            if (songs.get(i).s.equals(X)) {
                idx = i;
            }
        }
        int ans = 0;
        for (int i = idx + 1; i < songs.size(); i++) {
            ans += songs.get(i).t;
        }
        System.out.println(ans);
    }

    static class Song {
        String s;
        int t;

        Song(String s, int t) {
            this.s = s;
            this.t = t;
        }
    }
}
