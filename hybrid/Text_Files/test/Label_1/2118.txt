#include <bits/stdc++.h>

typedef long long ll;

int main() {
  int n;
  std::cin >> n;
  std::vector<std::array<ll, 3>> a(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 3; j++) std::cin >> a[i][j];
  }
  std::vector<std::vector<ll>> g(n);
  std::vector<bool> vis(n);
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      bool is_good = true;
      for (int k = 0; k < n; k++) {
        if (k == i || k == j) continue;
        if (a[k][0] >= std::min(a[i][0], a[j][0]) &&
            a[k][0] <= std::max(a[i][0], a[j][0]) &&
            a[k][1] >= std::min(a[i][1], a[j][1]) &&
            a[k][1] <= std::max(a[i][1], a[j][1]) &&
            a[k][2] >= std::min(a[i][2], a[j][2]) &&
            a[k][2] <= std::max(a[i][2], a[j][2])) {
          is_good = false;
          break;
        }
      }
      if (is_good) {
        g[i].push_back(j);
        g[j].push_back(i);
      }
    }
  }
  std::vector<ll> ans;
  std::function<void(ll)> dfs = [&](ll u) {
    vis[u] = true;
    for (ll v : g[u]) {
      if (!vis[v]) {
        ans.push_back(u);
        ans.push_back(v);
        dfs(v);
      }
    }
  };
  dfs(0);
  for (ll x : ans) std::cout << x + 1 << " ";
  std::cout << "\n";
}
