import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner scanner=new Scanner(System.in);
        int a=scanner.nextInt();
        switch(a)
        {
            case 1:
                System.out.println(2);
                break;
            case 2:
                System.out.println(1);
                break;
            case 3:
                System.out.println(1);
                break;
            case 4:
                System.out.println(3);
                break;
            case 5:
                System.out.println(2);
                break;
        }
        scanner.close();
    }
}
