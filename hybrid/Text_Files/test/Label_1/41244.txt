import heapq

n, m = map(int, input().split())
edges = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)
s, b, k, h = map(int, input().split())
spaceships = []
for _ in range(s):
    x, a, f = map(int, input().split())
    spaceships.append((a, f))
bases = []
for _ in range(b):
    x, d = map(int, input().split())
    bases.append((d, x))
bases.sort()
spaceships.sort(reverse=True)
answer = []
cur = 0
for d, x in bases:
    while cur < len(spaceships) and spaceships[cur][0] >= d:
        if spaceships[cur][1] >= x - 1:
            heapq.heappush(answer, spaceships[cur][1])
            cur += 1
        else:
            break
    if len(answer) < len(bases):
        heapq.heappush(answer, 0)
if len(answer) < len(bases):
    print(b * k)
else:
    print(k * len(bases) + h * (len(bases) - len(answer)))
