#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, m, i, j, x[100001], y[100001], cnt = 0;
    cin >> n;
    for (i = 1; i <= n; i++) cin >> x[i];
    cin >> m;
    for (i = 1; i <= m; i++) cin >> y[i];
    for (i = 1; i <= n; i++)
      for (j = 1; j <= m; j++)
        if ((x[i] + y[j]) % 2 == 0) cnt++;
    cout << cnt << endl;
  }
}
