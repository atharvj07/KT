import numpy as np

def is_identical(map1, map2):
    for rot in range(4):
        for flip in range(2):
            if np.array_equal(map1, map2):
                return True
            map2 = np.rot90(map2)
        map2 = np.fliplr(map2)
    return False

n = int(input())
map1 = [input() for _ in range(n)]
map2 = [input() for _ in range(n)]
if is_identical(map1, map2):
    print("Yes")
else:
    print("No")
