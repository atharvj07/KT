import math

def check(t, x, y, vx, vy, n, X, Y):
    xx = x + vx * t
    yy = y + vy * t
    for i in range(n):
        x1 = X[i]
        y1 = Y[i]
        x2 = X[(i + 1) % n]
        y2 = Y[(i + 1) % n]
        if (y1 <= yy <= y2 or y2 <= yy <= y1) and (min(x1, x2) <= xx <= max(x1, x2)):
            a = (y2 - y1) / (x2 - x1)
            b = y1 - a * x1
            if a * xx + b - yy <= 1e-9:
                return True
    return False

n, w, v, u = map(int, input().split())
X = []
Y = []
for i in range(n):
    x, y = map(int, input().split())
    X.append(x)
    Y.append(y)
l = 0
r = w / u
ans = w / u
while abs(r - l) > 1e-9:
    t = (l + r) / 2
    if check(t, 0, 0, -v, u, n, X, Y):
        r = t
    else:
        l = t
        ans = t
print(ans)
