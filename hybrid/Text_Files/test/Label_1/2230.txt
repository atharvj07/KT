#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<unordered_map<int, bool>> cards(n);
    for (int i = 0; i < n; i++) {
        int m;
        cin >> m;
        for (int j = 0; j < m; j++) {
            int number;
            cin >> number;
            cards[i][number] = true;
        }
    }
    for (int i = 0; i < n; i++) {
        bool canWin = true;
        for (int j = 1; j <= 100; j++) {
            if (cards[i].find(j) == cards[i].end()) {
                canWin = false;
                break;
            }
        }
        if (canWin) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
