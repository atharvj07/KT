import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        String s = input.next();
        int cnt = 0;
        for (int i = 0; i < n - 10; i++) {
            if (s.charAt(i) == '8') {
                boolean valid = true;
                for (int j = i + 1; j < i + 11; j++) {
                    if (s.charAt(j) < '0' || s.charAt(j) > '9') {
                        valid = false;
                    }
                }
                if (valid) {
                    cnt++;
                }
            }
        }
        System.out.println(cnt);
    }
}
