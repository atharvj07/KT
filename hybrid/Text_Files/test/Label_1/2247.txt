#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 3e5 + 5;
int a[maxn], prea[maxn], suf[maxn];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        for (int i = 1; i <= n; i++) cin >> a[i];
        prea[1] = a[1];
        for (int i = 2; i <= n; i++) prea[i] = max(prea[i - 1], a[i]);
        suf[n] = a[n];
        for (int i = n - 1; i >= 1; i--) suf[i] = min(suf[i + 1], a[i]);
        cout << (k == 1 ? "NO" : (prea[n / k] >= suf[(n + k - 1) / k] ? "YES" : "NO")) << endl;
    }
    return 0;
}
