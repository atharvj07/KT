n = int(input().strip())
a = list(map(int, input().strip().split()))

ones = a.count(1)
twos = a.count(2)
max_len = max(ones, twos)

# To store the number of twos before each position
pref_twos = [0] * (n + 1)
for i in range(1, n + 1):
    pref_twos[i] = pref_twos[i-1] + (a[i-1] == 2)

# Iterate through all possible intervals and calculate the maximum length
for l in range(1, n + 1):
    for r in range(l, n + 1):
        ones_in_interval = (r - l + 1) - (pref_twos[r] - pref_twos[l-1])
        twos_before_interval = pref_twos[l-1]
        twos_after_interval = pref_twos[n] - pref_twos[r]
        max_len = max(max_len, ones_in_interval + twos_before_interval + twos_after_interval)

print(max_len)
