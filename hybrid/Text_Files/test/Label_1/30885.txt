import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][][] ranges = new int[n][2][2]; // {x,y} {start,end}
        for(int i = 0; i < n; i++){
            for(int j = 0; j < 2; j++){
                for(int k = 0; k < 2; k++){
                    ranges[i][j][k] = in.nextInt();
                }
            }
        }
        int minX = Integer.MAX_VALUE;
        int minY = Integer.MAX_VALUE;
        int maxX = Integer.MIN_VALUE;
        int maxY = Integer.MIN_VALUE;
        for(int i = 0; i < n; i++){
            minX = Math.min(minX, ranges[i][0][0]);
            minY = Math.min(minY, ranges[i][1][0]);
            maxX = Math.max(maxX, ranges[i][0][1]);
            maxY = Math.max(maxY, ranges[i][1][1]);
        }
        boolean isSquare = true;
        for(int i = 0; i < n; i++){
            if(ranges[i][0][0] != minX && ranges[i][0][1] != maxX && ranges[i][1][0] != minY && ranges[i][1][1] != maxY){
                isSquare = false;
            }
        }
        System.out.println(isSquare ? "YES" : "NO");
    }
}
