#include <iostream>
#include <set>
#include <string>

using namespace std;

int main() {
  int n;
  string home;
  cin >> n >> home;
  set<string> airports;
  airports.insert(home);
  while (n--) {
    string s;
    cin >> s;
    int pos = s.find('-');
    if (airports.count(s.substr(pos + 1))) {
      airports.erase(s.substr(0, pos));
    } else {
      airports.insert(s.substr(0, pos));
    }
  }
  cout << (*airports.begin() == home ? "home" : "contest") << endl;
  return 0;
}
