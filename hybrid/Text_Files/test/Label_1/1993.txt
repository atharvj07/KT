#include <bits/stdc++.h>

using namespace std;

#define ll long long int

int main() {
    ll n;
    cin >> n;
    ll a[n], b[n];
    for (ll i = 0; i < n; i++) cin >> b[i];
    ll sum[n + 1];
    sum[n] = 0;
    for (ll i = n - 1; i >= 0; i--) sum[i] = sum[i + 1] + b[i];
    bool flag = true;
    for (ll i = 1; i <= 9; i++) {
        a[n - 1] = i;
        if (i <= sum[n - 1]) {
            flag = false;
            break;
        }
    }
    if (flag) {
        cout << -1 << endl;
        return 0;
    }
    for (ll i = n - 2; i >= 0; i--) {
        ll s = b[i] - (a[i + 1] % 10);
        for (ll j = max(0ll, s - 9); j < 10; j++) {
            if (sum[i + 1] - j >= a[i + 1] / 10) {
                a[i] = j;
                sum[i] = sum[i + 1] - j;
                break;
            }
        }
    }
    for (ll i = 0; i < n; i++) cout << a[i] << endl;
    return 0;
}
