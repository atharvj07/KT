#include <bits/stdc++.h>
using namespace std;

int main() {
  string n;
  cin >> n;
  int dig = n.length();
  if (dig % 2) {
    if (n[dig - 1] == '0')
      n.pop_back(), dig--;
    else
      return cout << -1, 0;
  }
  if (n.back() % 2) return cout << -1, 0;
  for (int i = 0; i < dig; i += 2) {
    if (n[i] > n[i + 1])
      return cout << -1, 0;
  }
  int del = 0;
  for (int i = 0; i < dig; i++) {
    int cur = n[i] - '0';
    int nx = (i + 1 < dig) ? n[i + 1] - '0' : -1;
    del += (i + 1 >= dig || cur * 10 + nx > 25);
  }
  cout << del;
}
