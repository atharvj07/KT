import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class SonyaAndMultiset {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(reader.readLine());

        List<Long> multiset = new ArrayList<>();

        for (int i = 0; i < t; i++) {
            StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            char operation = tokenizer.nextToken().charAt(0);
            if (operation == '+') {
                multiset.add(Long.parseLong(tokenizer.nextToken()));
            } else if (operation == '-') {
                long value = Long.parseLong(tokenizer.nextToken());
                multiset.remove(value);
            } else {
                String pattern = tokenizer.nextToken();
                long count = 0;
                for (long value : multiset) {
                    if (matchesPattern(value, pattern)) {
                        count++;
                    }
                }
                System.out.println(count);
            }
        }
    }

    private static boolean matchesPattern(long value, String pattern) {
        String valueString = Long.toBinaryString(value);
        int patternLength = pattern.length();
        int valueLength = valueString.length();
        int diff = Math.abs(patternLength - valueLength);

        if (patternLength > valueLength) {
            valueString = "0".repeat(diff) + valueString;
        } else {
            pattern = "0".repeat(diff) + pattern;
        }

        for (int i = 0; i < pattern.length(); i++) {
            if (pattern.charAt(i) == '1' && valueString.charAt(i) != '1') {
                return false;
            }
        }

        return true;
    }
}
