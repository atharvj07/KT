import sys
input = sys.stdin.readline

t = int(input())
for tt in range(t):
    k = int(input())
    edges = []
    for i in range(2*k-1):
        a, b, c = map(int, input().split())
        edges.append((a, b, c))
    
    G = [0]*k
    B = [0]*k
    
    adj = [[] for i in range(2*k+1)]
    for a, b, c in edges:
        adj[a].append((b, c))
        adj[b].append((a, c))
    
    def dfs(node, parent, g, b, d):
        for next, w in adj[node]:
            if next != parent:
                dfs(next, node, g+d*w, b-d*w, d*-1)
    
    dfs(1, 1, 0, 0, 1)
    
    print(*G, *B)
