s = input()
q = int(input())
cnt = [0] * 26
for i in range(26):
    l = []
    for j in range(len(s)):
        if s[j] == chr(97 + i):
            l.append(j)
    cnt[i] = l
for _ in range(q):
    t, l, r = map(int, input().split())
    l -= 1
    r -= 1
    if t == 2:
        print(len(set(s[l : r + 1])))
    else:
        s = s[:l] + chr(97 + r) + s[l + 1:]
        cnt[ord(s[l]) - 97] = [l]
        cnt[ord(r) - 97].remove(l)
        if len(cnt[ord(r) - 97]) == 0:
            cnt[ord(r) - 97] = []
