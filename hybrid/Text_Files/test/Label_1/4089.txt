#include<bits/stdc++.h>
using namespace std;
char c[110][110];
int main(){
    int n;
    cin>>n;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            cin>>c[i][j];
        }
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=n;j++){
            int ans=0;
            if(c[i][j]=='o'){
                if(c[i-1][j-1]=='o')ans++;
                if(c[i-1][j]=='o')ans++;
                if(c[i-1][j+1]=='o')ans++;
                if(c[i][j-1]=='o')ans++;
                if(c[i][j+1]=='o')ans++;
                if(c[i+1][j-1]=='o')ans++;
                if(c[i+1][j]=='o')ans++;
                if(c[i+1][j+1]=='o')ans++;
                if(ans%2)return cout<<"NO",0;
            }
        }
    }
    cout<<"YES";
}
