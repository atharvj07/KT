import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            boolean valid = true;
            int prevPlays = 0, prevClears = 0;

            while (n-- > 0) {
                int plays = sc.nextInt();
                int clears = sc.nextInt();

                if (plays < prevPlays || clears < prevClears || clears > plays) {
                    valid = false;
                }

                prevPlays = plays;
                prevClears = clears;
            }

            System.out.println(valid ? "YES" : "NO");
        }
    }
}
