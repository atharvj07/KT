#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> p, q;

int main() {
  cin >> n;
  p.resize(n + 1);
  for (int i = 1; i <= n; i++) { cin >> p[i]; }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (p[p[j]] == i) { q.push_back(j); }
    }
  }
  if (q.size() == n) {
    for (int i : q) { cout << i << " "; }
  } else {
    cout << "-1" << endl;
  }
  return 0;
}
