import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt(), q = input.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = input.nextLong();
        }
        List<Integer>[] sets = new List[m + 1];
        for (int i = 1; i <= m; i++) {
            int k = input.nextInt();
            sets[i] = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                sets[i].add(input.nextInt());
            }
        }
        int[] cnt = new int[n + 1];
        long[] sum = new long[m + 1];
        for (int i = 1; i <= m; i++) {
            for (int index : sets[i]) {
                if (cnt[index] == 0) {
                    sum[i] += a[index];
                }
                cnt[index]++;
            }
        }
        while (q-- > 0) {
            String query = input.next();
            char type = query.charAt(0);
            int k = input.nextInt();
            if (type == '?') {
                System.out.println(sum[k]);
            } else {
                long x = input.nextLong();
                for (int index : sets[k]) {
                    cnt[index]--;
                    if (cnt[index] == 0) {
                        sum[k] -= a[index];
                        a[index] += x;
                        sum[k] += a[index];
                    } else {
                        a[index] += x;
                    }
                }
            }
        }
    }
}
