#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int n, k;
vector<int> pixels;
vector<int> keys;

int main() {
    cin >> n >> k;
    pixels.resize(n);
    keys.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> pixels[i];
    }
    sort(pixels.begin(), pixels.end());
    int group = 0;
    int key = 0;
    for (int i = 0; i < n; i++) {
        if (i > 0 && pixels[i] > pixels[i - 1] + k) {
            group++;
            key = pixels[i];
        }
        keys[i] = key;
    }
    for (int i = 0; i < n; i++) {
        cout << keys[i] << " ";
    }
    cout << endl;
    return 0;
}
