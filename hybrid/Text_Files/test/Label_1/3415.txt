#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  string s;
  cin >> n >> k >> s;
  vector<int> cost(n + 1, INT_MAX);
  cost[0] = 0;
  for (int i = 1; i <= n; i++) {
    if (s[i - 1] == '1') {
      for (int j = max(1, i - k); j <= min(n, i + k); j++) {
        cost[j] = min(cost[j], cost[i - 1] + i);
      }
    }
  }
  cout << cost[n] << endl;
}
