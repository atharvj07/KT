def min_changes_needed(n, s):
    if s.count('L') == n or s.count('R') == n:
        return n // 3
    s += s[0]  # Make the string circular
    segments, count = [], 1
    for i in range(1, n + 1):
        if s[i] == s[i - 1]:
            count += 1
        else:
            segments.append(count)
            count = 1
    segments[0] += count - 1  # Merge the first and last segments if they are the same
    return sum(seg // 3 for seg in segments)

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    print(min_changes_needed(n, s))
