import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            long k = sc.nextLong();
            long ans = (k + 8) % 9;
            if (ans == 0) ans = 9;
            System.out.println(ans);
        }
    }
}
