#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>

using namespace std;
using namespace __gnu_pbds;

typedef tree<int,null_type,less<int>,rb_tree_tag,tree_order_statistics_node_update> ordered_set;
typedef tree<pair<int,int>,null_type,less<pair<int,int>>,rb_tree_tag,tree_order_statistics_node_update> ordered_pset;

int n,k,lst[100001],deg[100001];
vector<int>adj[100001];
ordered_set s[100001];

int32_t main()
{
    cin>>n>>k;
    for(int i=1;i<=n;++i){
        int x;
        cin>>x;
        s[x].insert(i);
    }
    for(int dist=1;dist<n;++dist){
        for(int i=0;i<n;++i){
            auto it=s[dist].upper_bound(i);
            if(it==s[dist].end()) continue;
            int j=*it;
            lst[dist+1]=j;
            deg[j]++;
            deg[i]++;
            s[dist+1].insert(i);
            adj[i].push_back(j);
            adj[j].push_back(i);
            s[dist].erase(j);
        }
        for(int i=0;i<n;++i){
            int fr=lst[dist],to=lst[dist+1];
            if(deg[fr]>k || deg[to]>k || adj[fr].size()>k || adj[to].size()>k){
                cout<<-1<<endl;
                return 0;
            }
        }
    }
    cout<<(n-1)*2<<endl;
    for(int i=1;i<=n;++i){
        for(int v:adj[i]){
            if(i<v) cout<<i<<' '<<v<<endl;
        }
    }
    return 0;
}
