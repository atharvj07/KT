#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, x;
    cin >> n >> x;
    vector<array<int, 3>> v(n);
    for (auto &i : v) cin >> i[0] >> i[1] >> i[2];

    sort(v.begin(), v.end(), [](auto &a, auto &b) { return a[1] - a[0] < b[1] - b[0]; });

    int ans = 1e9;
    for (int i = 0, j = n - 1; i < j;) {
        int sum = v[i][1] - v[i][0] + 1 + v[j][1] - v[j][0] + 1;
        if (sum == x) {
            ans = min(ans, v[i][2] + v[j][2]);
            i++, j--;
        } else if (sum < x)
            i++;
        else
            j--;
    }
    if (ans == 1e9) ans = -1;
    cout << ans << '\n';

    return 0;
}
