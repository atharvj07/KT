import java.util.*;
import java.io.*;

public class Main {
    
    static final long MOD = 1000000007L;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        char[] arr = s.toCharArray();
        int[] right = new int[n];
        right[n - 1] = n - 1;
        for(int i = n - 2; i >= 0; i--){
            if(arr[i] > arr[right[i + 1]]){
                right[i] = i;
            } else {
                right[i] = right[i + 1];
            }
        }
        long[] pow = new long[n + 1];
        pow[0] = 1;
        for(int i = 1; i <= n; i++){
            pow[i] = (pow[i - 1] * 26) % MOD;
        }
        long ans = 0;
        for(int l = 0; l < n; l++){
            long r = right[l] - l;
            if(r + 1 >= k){
                ans = (ans + (pow[(int)r + 1] - 1)) % MOD;
            }
        }
        System.out.println(ans);
    }
}
