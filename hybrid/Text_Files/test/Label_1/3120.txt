#include <bits/stdc++.h>
using namespace std;
bool ok[50][50];
int main() {
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < s.length(); i++) {
    for (int j = i; j < s.length(); j++) {
      bool pal = true;
      for (int k = i; k <= (i + j) / 2; k++) {
        pal &= (s[k] == s[i + j - k]);
      }
      if (!pal) ans = max(ans, j - i + 1);
      ok[i][j] = pal;
    }
  }
  cout << ans << endl;
}
