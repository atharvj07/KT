import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int cur_mex = 1;
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = cur_mex++;
            while (cur_mex < a[i]) {
                cur_mex++;
            }
        }
        for (int i = 0; i < n; i++) {
            if (b[i] > 1000000) {
                System.out.println(-1);
                return;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(b[i] + " ");
        }
        System.out.println();
    }
}
