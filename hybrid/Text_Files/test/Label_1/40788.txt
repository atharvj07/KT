import sys
input = sys.stdin.readline
n, m = map(int, input().split())
A = []
for i in range(n):
    A.append(list(map(int, input().split())))
vis_col = [0 for i in range(m)]
B = []
for i in range(n):
    for j in range(m):
        B.append([A[i][j], i, j])
B.sort()
k = 0
ans = []
for a, i, j in B:
    if not vis_col[j] and (i == 0 or A[i-1][j] == a or i == n-1 or A[i+1][j] == a):
        k += 1
        ans.append('col {}'.format(j+1))
        vis_col[j] = 1
    else:
        ans.append('row {}'.format(i+1))
print(k)
for a in ans:
    print(a)
