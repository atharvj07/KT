#include<bits/stdc++.h>
using namespace std;
int main(int argv,char **argc)
{
    long long int x,y;
    cin>>x>>y;
    x+=1;
    if((x+1)/2<y)
        cout<<-1<<endl;
    else
    {
        for(long long int i=1;i<=y;i++)
        {
            cout<<1;
            if(i%2==0)
                cout<<0;
        }
    }
    cout<<endl;
    return 0;
}
