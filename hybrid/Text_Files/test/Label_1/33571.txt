import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Set<Integer> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                set.add(a[i]);
                set.add(-a[i]);
            }
        }
        int k = 0;
        List<Integer> list = new ArrayList<>();
        for (int x : set) {
            int cnt = 0;
            int y = x;
            while (y != 0) {
                cnt++;
                y /= 2;
            }
            if (cnt > k) {
                k = cnt;
                list.clear();
                list.add(x);
            } else if (cnt == k) {
                list.add(x);
            }
        }
        System.out.println(k);
        for (int x : list) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
