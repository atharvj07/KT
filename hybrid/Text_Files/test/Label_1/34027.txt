import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int sum = 0;
        int i = 0;
        int j = n - 1;
        boolean turn = true; // Alice's turn
        while (i <= j) {
            if (turn) {
                if (a[i] > a[j]) {
                    sum += a[i];
                    i++;
                } else {
                    sum += a[j];
                    j--;
                }
            } else {
                if (a[i] < a[j]) {
                    sum -= a[i];
                    i++;
                } else {
                    sum -= a[j];
                    j--;
                }
            }
            turn = !turn;
        }
        System.out.println(sum);
    }
}
