#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> topic(n), difficulty(n);
    unordered_map<int, int> topic_freq, difficulty_freq;
    for (int i = 0; i < n; ++i) {
      cin >> topic[i] >> difficulty[i];
      ++topic_freq[topic[i]];
      ++difficulty_freq[difficulty[i]];
    }
    long long ans = 1ll * n * (n - 1) * (n - 2) / 6;
    for (int i = 0; i < n; ++i) {
      ans -= 1ll * topic_freq[topic[i]] * (topic_freq[topic[i]] - 1) * (topic_freq[topic[i]] - 2) / 6;
      ans -= 1ll * difficulty_freq[difficulty[i]] * (difficulty_freq[difficulty[i]] - 1) * (difficulty_freq[difficulty[i]] - 2) / 6;
    }
    cout << ans << '\n';
  }
  return 0;
}
