#include <bits/stdc++.h>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n, k, i, j;
    cin >> n >> k;
    vector<int> a(2 * k), p(k + 1);
    for (i = 1; i <= n; i++) cin >> a[i];
    iota(p.begin() + 1, p.end(), 1);
    j = k;
    for (i = n; i >= n - k + 1; i--) {
      swap(p[a[i]], p[a[j]]);
      j--;
    }
    for (i = 1; i <= k; i++) cout << p[i] << ' ';
    cout << '\n';
  }
}
