import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class Main {
    private static Map<Integer, Integer> countMap = new HashMap<Integer, Integer>();

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int q = Integer.parseInt(br.readLine());
        initCountMap(a, n);
        for (int i = 0; i < q; i++) {
            String[] line = br.readLine().split(" ");
            int b = Integer.parseInt(line[0]);
            int e = Integer.parseInt(line[1]);
            int k = Integer.parseInt(line[2]);
            System.out.println(count(b, e, k));
        }
    }

    private static void initCountMap(int[] a, int n) {
        for (int i = 0; i < n; i++) {
            int count = countMap.getOrDefault(a[i], 0);
            count++;
            countMap.put(a[i], count);
        }
    }

    private static int count(int b, int e, int k) {
        return countMap.getOrDefault(k, 0);
    }
}
