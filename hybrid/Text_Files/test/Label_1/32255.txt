import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        String ad = scanner.nextLine().trim();
        String[] words = ad.split(" ");
        int[][] dp = new int[words.length + 1][k + 1];
        for (int i = 0; i <= k; i++) {
            dp[0][i] = 0;
        }
        for (int i = 0; i <= words.length; i++) {
            dp[i][0] = 0;
        }
        for (int i = 1; i <= words.length; i++) {
            for (int j = 1; j <= k; j++) {
                int width = words[i - 1].length() + 1;
                dp[i][j] = dp[i - 1][j];
                if (width <= j) {
                    dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - width] + width);
                }
            }
        }
        System.out.println(dp[words.length][k]);
    }
}
