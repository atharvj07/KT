#include <iostream>
#include <cmath>
#include <vector>

using ll = long long;
using ull = unsigned ll;
constexpr ll MOD = 998244353;

int main() {
    ll n, k;
    std::cin >> n >> k;
    std::vector<ll> fact(502, 1);
    for (ll i = 2; i <= 501; ++i) fact[i] = fact[i - 1] * i % MOD;
    std::vector<ll> invfact(502, 1);
    invfact[501] = ll(std::pow(fact[501], MOD - 2)) % MOD;
    for (ll i = 499; i >= 0; --i) invfact[i] = invfact[i + 1] * (i + 1) % MOD;

    auto comb = [&](ll p, ll q) -> ll {
        if (p < q || p < 0 || q < 0) return 0ll;
        return (fact[p] * invfact[p - q] % MOD) * invfact[q] % MOD;
    };

    auto calc = [&](ll n, ll m, ll k) -> ll {
        if (k > n * m) return 0ll;
        ll ans = comb((n + m - 2) / 2, (n - 1) / 2) * comb((n + m - 2) / 2, (m - 1) / 2) % MOD;
        if ((n + m) % 2 == 0) {
            ll rem = comb((n + m) / 2 - 1, (n - 1) / 2) * comb((n + m) / 2 - 1, (m - 1) / 2) % MOD;
            ans = (ans - rem + MOD) % MOD;
        }
        return ans;
    };

    ll ans = 0;
    for (ll i = 0; i <= k; ++i)
        ans = (ans + calc(n, n, i)) % MOD;
    std::cout << ans << std::endl;
    return 0;
}
