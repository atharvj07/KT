#include <bits/stdc++.h>
using namespace std;
const int MAXN = 2e5 + 5;
vector<int> adj[MAXN];
int sub_tree[MAXN], dist[MAXN], dep[MAXN];
int n, k, d, ans;
bool visited[MAXN];
void dfs1(int cur, int prev) {
  sub_tree[cur] = 1;
  for (const int &nxt : adj[cur]) {
    if (nxt == prev or visited[nxt]) continue;
    dfs1(nxt, cur);
    sub_tree[cur] += sub_tree[nxt];
  }
}
void dfs2(int cur, int prev, int sum) {
  dist[cur] = sum;
  for (const int &nxt : adj[cur]) {
    if (nxt == prev or visited[nxt]) continue;
    dfs2(nxt, cur, sum + 1);
  }
}
void dfs3(int cur, int prev, int sum) {
  ans = max(ans, sum);
  for (const int &nxt : adj[cur]) {
    if (nxt == prev or visited[nxt]) continue;
    dfs3(nxt, cur, sum + sub_tree[nxt] * (n - sub_tree[nxt]));
  }
}
int main() {
  cin >> n >> k;
  unordered_map<int, int> mp;
  for (int i = 1; i <= 2 * k; i++) {
    int x;
    cin >> x;
    mp[x] = i;
  }
  for (int i = 1; i < n; i++) {
    int x, y;
    cin >> x >> y;
    adj[x].push_back(y);
    adj[y].push_back(x);
  }
  dfs1(1, 1);
  dfs2(1, 1, 0);
  for (const auto &x : mp) {
    visited[x.first] = true;
    dfs3(x.first, x.first, dist[x.first] - sub_tree[x.first]);
  }
  cout << ans << '\n';
  return 0;
}
