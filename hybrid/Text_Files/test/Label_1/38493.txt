def solve_fgh(n, f):
    unique_f = list(set(f))
    unique_f.sort()
    m = len(unique_f)
    g = [0] * n
    h = [0] * m
    
    f_to_h = {f_val: i+1 for i, f_val in enumerate(unique_f)}
    h_to_f = {i+1: f_val for i, f_val in enumerate(unique_f)}
    
    for i in range(n):
        g[i] = f_to_h[f[i]]
    
    for i in range(m):
        h[i] = h_to_f[i+1]
    
    return m, g, h

n = int(input().strip())
f = list(map(int, input().strip().split()))

res = solve_fgh(n, f)
if res:
    m, g, h = res
    print(m)
    print(" ".join(map(str, g)))
    print(" ".join(map(str, h)))
else:
    print(-1)
