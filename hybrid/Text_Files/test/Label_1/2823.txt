#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> perm(n);
  for (int i = 0; i < n; i++) {
    cin >> perm[i];
  }
  for (int i = 0; i < m; i++) {
    int l, r, x;
    cin >> l >> r >> x;
    int idx = -1;
    for (int j = l - 1; j < r; j++) {
      if (perm[j] == x) {
        idx = j;
        break;
      }
    }
    if (idx == -1 || idx == x - 1) {
      cout << "Yes" << endl;
    } else {
      cout << "No" << endl;
    }
  }
  return 0;
}
