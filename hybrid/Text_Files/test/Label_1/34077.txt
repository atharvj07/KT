import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (in.hasNext()) {
            String input = in.nextLine();
            int r1 = 0, r2 = 0;
            char op = 0;
            for (char c : input.toCharArray()) {
                if (c >= '0' && c <= '9') {
                    r2 = r2 * 10 + (c - '0');
                } else if (c == '+' || c == '-' || c == '*') {
                    if (op != 0) {
                        if (op == '+') {
                            r1 += r2;
                        } else if (op == '-') {
                            r1 -= r2;
                        } else if (op == '*') {
                            r1 *= r2;
                        }
                    } else {
                        r1 = r2;
                    }
                    r2 = 0;
                    op = c;
                } else if (c == '=') {
                    if (op != 0) {
                        if (op == '+') {
                            r1 += r2;
                        } else if (op == '-') {
                            r1 -= r2;
                        } else if (op == '*') {
                            r1 *= r2;
                        }
                    }
                    r2 = 0;
                    op = 0;
                    if (r1 < 0 || r1 > 9999) {
                        System.out.println("E");
                    } else {
                        System.out.println(r1);
                    }
                }
            }
        }
    }
}
