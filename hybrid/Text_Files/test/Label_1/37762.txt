n = int(input())
lab = [list(map(int, input().split())) for _ in range(n)]

def check_good_lab(lab):
    for x in range(n):
        for y in range(n):
            if lab[x][y] != 1:
                found = False
                for s in range(n):
                    for t in range(n):
                        if lab[x][y] == lab[x][s] + lab[t][y] and s != y and t != x:
                            found = True
                            break
                    if found:
                        break
                if not found:
                    return "No"
    return "Yes"

print(check_good_lab(lab))
