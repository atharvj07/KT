#include<bits/stdc++.h>
using namespace std;

#define ll long long
#define ld long double
#define endl "\n"
#define fo(i,n) for(ll i=0;i<n;++i)
#define all(a) a.begin(),a.end()

const ll mod=1e9+7;

void solve(){
ll n;cin>>n;
vector<ll> a(n);fo(i,n) cin>>a[i];
unordered_map<ll,ll> mp;
fo(i,n){
if(a[i]>0) mp[a[i]]++;
}
if(mp.empty()) {cout<<-1<<endl;return;}
ll g=0,cnt=0;
for(auto x:mp) g=__gcd(g,x.first),cnt=x.second;
if(g==1) {cout<<-1<<endl;return;}
if(cnt==1) {cout<<0<<endl;return;}
cout<<n-cnt<<endl;
}

int main(){
ios_base::sync_with_stdio(false);
cin.tie(NULL);
ll t;cin>>t;
while(t--) solve();
return 0;
}
