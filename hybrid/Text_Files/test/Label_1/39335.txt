n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split())) if k else []
b = set(b)

ans = 0
prev = 0
for i in range(n):
    if i in b:
        continue
    else:
        if a[i] < prev:
            ans += 1
        prev = a[i]
if ans > n - k - 1:
    print(-1)
else:
    print(ans)
