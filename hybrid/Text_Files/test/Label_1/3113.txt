#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int maxN = 105;
const int maxM = 105;
int a[maxN][maxM];
int d[maxN][maxM];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        while (x--) {
            int y;
            cin >> y;
            a[i][y] = 1;
        }
    }

    int ans = 0;
    for (int j = 1; j <= m; j++) {
        int cnt = 0;
        for (int i = 0; i < n; i++) cnt += a[i][j];
        if (cnt * n + n - cnt > 0) ans++;
    }

    cout << ans << endl;
    return 0;
}
