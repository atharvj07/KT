#include<bits/stdc++.h>
using namespace std;
const int mxn=2e5+1;
int n,a[mxn],sa[mxn];
vector<bool>vis(1<<21);
int32_t main(){
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    cin>>n;
    for(int i=0;i<n;++i) { cin>>a[i]; sa[i]=a[i]; }
    sort(sa,sa+n);
    int cnt=0,msk=0;
    for(int i=0;i<n;++i){
        msk^=sa[i];
        if(!vis[msk]) { vis[msk]=true; ++cnt; }
    }
    if(cnt<n) return cout<<0<<endl<<0<<endl,0;
    int x=0;
    while(cnt>1){
        cnt>>=1;
        ++x;
    }
    cout<<x<<endl;
    for(int i=0;i<n;++i) cout<<(a[i]^sa[i])<<" ";
    return 0;
}
