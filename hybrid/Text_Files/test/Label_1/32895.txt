import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

class Main {
  private static class Club {
    String name, city;
    String shortName1, shortName2;

    Club(String name, String city) {
      this.name = name;
      this.city = city;
      shortName1 = name.substring(0, 3);
      shortName2 = name.substring(0, 2) + city.substring(0, 1);
    }

    String getShortName() {
      return shortName1;
    }

    void setShortName(String shortName) {
      shortName1 = shortName;
      shortName2 = shortName;
    }

    @Override
    public String toString() {
      return String.format("%s, %s, %s, %s", name, city, shortName1, shortName2);
    }
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    scanner.nextLine();

    List<Club> clubs = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      String[] line = scanner.nextLine().split(" ");
      clubs.add(new Club(line[0], line[1]));
    }
    scanner.close();

    Map<String, String> usedShortName = new HashMap<>();
    for (Club club : clubs) {
      String shortName1 = club.getShortName();
      if (usedShortName.containsKey(shortName1)) {
        club.setShortName(club.shortName2);
        String shortName2 = club.getShortName();
        if (usedShortName.containsKey(shortName2)) {
          System.out.println("NO");
          return;
        }
        usedShortName.put(shortName2, shortName2);
      } else {
        usedShortName.put(shortName1, shortName1);
      }
    }

    System.out.println("YES");
    for (Club club : clubs) {
      System.out.println(club.getShortName());
    }
  }
}
