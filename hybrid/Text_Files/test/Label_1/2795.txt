#include <bits/stdc++.h>

using namespace std;

#define ll long long
const ll mod = 1e9 + 7;

ll fastpow(ll a, ll b) {
    ll ans = 1;
    while (b > 0) {
        if (b % 2 == 1) {
            ans = ans * a % mod;
        }
        a = a * a % mod;
        b >>= 1;
    }
    return ans % mod;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    string s;
    cin >> s;
    int n = s.size();

    vector<ll> pref(n + 1, 0);
    for (int i = 0; i < n; ++i) {
        pref[i + 1] = pref[i];
        if (s[i] == '(') {
            pref[i + 1]++;
        }
    }

    ll ans = 0;
    for (int i = 0; i < n; ++i) {
        if (s[i] == ')') {
            ll cnt = pref[i];
            ll rem = n - i - cnt;
            ans = (ans + fastpow(2, cnt - 1) * fastpow(2, rem - 1) % mod) % mod;
        }
    }

    cout << ans << endl;

    return 0;
}
