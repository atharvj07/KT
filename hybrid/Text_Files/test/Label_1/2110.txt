#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> p(n);
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') {
            p[cnt++] = i;
        }
    }
    int ans = n;
    for (int i = 0; i < cnt; i++) {
        int l = max(1, p[i] - k);
        int r = min(n, p[i] + k);
        ans = min(ans, i + n - r + l - 1);
    }
    cout << ans << endl;
    return 0;
}
