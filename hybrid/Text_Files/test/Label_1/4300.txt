#include <iostream>
#include <vector>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);

  int t;
  cin >> t;
  while (t--) {
    int x, y;
    cin >> x >> y;
    vector<int> c(6);
    for (int& i : c) cin >> i;

    int up = 0, down = 0;
    if (y < 0) {
      up = -y + abs(x % 2);
      y = 0;
    }
    if (abs(x) >= abs(y)) {
      x -= up * (-1) + down * 1;
      y += up * 0 + down * 1;
    }
    if (abs(x) >= abs(y)) {
      x -= up * 1 + down * (-1);
      y += up * 1 + down * 0;
    }
    while (x != 0 && y != 0) {
      if (y > 0)
        up++, x -= 1, y -= 1;
      else
        down++, x -= 1, y += 1;
      if (abs(x) >= abs(y)) {
        x -= up * (-1) + down * 1;
        y += up * 0 + down * 1;
      }
      if (abs(x) >= abs(y)) {
        x -= up * 1 + down * (-1);
        y += up * 1 + down * 0;
      }
    }
    while (y > 0)
      up++, x -= 1, y -= 1;
    while (x > 0)
      down++, x -= 1, y += 1;
    cout << up * c[1] + down * c[3] << '\n';
  }
}
