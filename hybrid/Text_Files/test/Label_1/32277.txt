import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }
            int[] b = new int[n];
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
                a[i] = sc.nextInt();
            }
            int[][] dp = new int[n + 1][m + 1];
            for (int i = 1; i <= n; i++) {
                for (int j = 0; j <= m; j++) {
                    dp[i][j] = dp[i - 1][j];
                    if (j >= a[i - 1] && dp[i - 1][j - a[i - 1]] + 1 > dp[i][j]) {
                        dp[i][j] = dp[i - 1][j - a[i - 1]] + 1;
                    }
                }
            }
            if (dp[n][m] == 0) {
                System.out.println("Impossible");
            } else {
                System.out.println(dp[n][m]);
            }
        }
    }
}
