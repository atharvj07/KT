def minJediRemoval(n, m, colors, desired):
    from collections import Counter
    
    def check(mid):
        current = Counter()
        total = 0
        for i in range(mid):
            current[colors[i]] += 1
            if current[colors[i]] <= desired[colors[i]-1]:
                total += 1
        if total == sum(desired):
            return True
        for i in range(mid, n):
            current[colors[i]] += 1
            if current[colors[i]] <= desired[colors[i]-1]:
                total += 1
            current[colors[i-mid]] -= 1
            if current[colors[i-mid]] < desired[colors[i-mid]-1]:
                total -= 1
            if total == sum(desired):
                return True
        return False

    left, right = sum(desired), n
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if check(mid):
            answer = n - mid
            right = mid - 1
        else:
            left = mid + 1
    return answer

n, m = map(int, input().split())
colors = list(map(int, input().split()))
desired = list(map(int, input().split()))
print(minJediRemoval(n, m, colors, desired))
