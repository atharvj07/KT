from collections import deque
from sys import stdin
input = stdin.readline

n, m = map(int, input().split())
s = input().strip()
palin = deque()
ans = []
for i in range(n):
    if i > 0 and s[i] == s[i - 1]:
        palin.append(i - 1)
    while palin and palin[0] + 1 < i - 1 and s[palin[0]] == s[i - 1]:
        palin.popleft()
    if palin and palin[0] + 1 == i - 1:
        ans.append(1)
    else:
        ans.append(2)
for _ in range(m):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    print(ans[r] if l == 0 else ans[r] - ans[l - 1])
