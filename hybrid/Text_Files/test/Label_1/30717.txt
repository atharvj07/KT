import java.util.Scanner;

public class Main {
  private static String[] notes = {"C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "B", "H"};

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] idx = new int[3];
    for (int i = 0; i < 3; i++) {
      idx[i] = index(sc.next(), notes);
    }
    int[] distances = new int[3];
    for (int i = 0; i < 3; i++) {
      distances[i] = idx[(i + 1) % 3] - idx[i];
      if (distances[i] < 0) distances[i] += 12;
    }
    sc.close();

    boolean major =
        (distances[0] == 4 && distances[1] == 3) || (distances[1] == 4 && distances[2] == 3);
    boolean minor =
        (distances[0] == 3 && distances[1] == 4) || (distances[1] == 3 && distances[2] == 4);
    if (major && minor) System.out.println("strange");
    else System.out.println(major ? "major" : "minor");
  }

  private static int index(String note, String[] notes) {
    for (int i = 0; i < notes.length; i++) {
      if (note.equals(notes[i])) return i;
    }
    return -1;
  }
}
