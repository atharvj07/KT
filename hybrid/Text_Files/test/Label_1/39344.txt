from collections import defaultdict
from math import sqrt
import sys
input = sys.stdin.readline
print = sys.stdout.write

t = int(input())
a = list(map(int, input().split()))

if any(x == 0 for x in a):
    print('-1')  # Check if 0 exists
else:
    n = int(sqrt(t))
    m = t // n

    d = defaultdict(list)
    for i in range(t):
        d[a[i]].append(i)

    x0, y0 = 0, 0
    for i in range(0, t, n):
        for j in range(m):
            if len(d[i + j]) == 1:
                x0 = i // n + 1
                y0 = j + 1
                break
        if x0 != 0:
            break

    x0, y0 = x0 - 1, y0 - 1

    pos = 0
    for i in range(n):
        for j in range(m):
            a[d[pos][0]] = i - x0 + j - y0
            pos += 1

    print(n, m)
    print(x0 + 1, y0 + 1)
