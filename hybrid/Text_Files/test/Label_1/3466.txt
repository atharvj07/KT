#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  int g = *min_element(a.begin(), a.end());
  for (int i = 0; i < n; ++i) {
    if (a[i] % g != 0) {
      g = -1;
      break;
    }
  }
  cout << g << endl;
  return 0;
}
