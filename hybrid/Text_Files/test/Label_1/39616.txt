t = int(input())
for _ in range(t):
    n = int(input())
    tags = list(map(int, input().split()))
    tag_counts = {}
    for tag in tags:
        if tag in tag_counts:
            tag_counts[tag] += 1
        else:
            tag_counts[tag] = 1
    max_count = max(tag_counts.values())
    distinct_tags = len(tag_counts)
    
    if max_count > (n + 1) // 2:
        print(-1)
    else:
        print(max(0, 2 * (max_count - 1) - (n - distinct_tags)))
