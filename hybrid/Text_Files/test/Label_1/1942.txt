#include <bits/stdc++.h>
using namespace std;
const int inf = 1e9+7;
const int maxN = 1e5+5;
int par[maxN];
int son[maxN];
vector<int>graph[maxN];
bool vis[maxN];
vector<vector<int>>ans;
void dfs(int cur){
    for(auto adj:graph[cur]){
        if(par[cur]!=adj && !vis[adj]){
            par[adj]=cur;
            dfs(adj);
        }
    }
}
void Path_Finding(int u,int v){
    vector<int>path;
    while(u){
        path.push_back(u);
        u=par[u];
        if(u==v){
            path.push_back(v);
            break;
        }
    }
    reverse(path.begin(),path.end());
    ans.push_back(path);
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin>>n;
    for(int i=0;i<n-1;i++){
        int a,b;
        cin>>a>>b;
        graph[a].push_back(b);
        graph[b].push_back(a);
    }
    dfs(1);
    bool ok=0;
    for(int i=1;i<=n;i++){
        for(auto adj:graph[i]){
            if(adj==par[i])
                continue;
            if(!vis[adj]){
                vis[adj]=1;
                vis[par[adj]]=1;
                son[i]++;
                son[adj]++;
            }
        }
    }
    int root=1;
    for(int i=1;i<=n;i++){
        if(son[i]>1)
            ok=1;
        if(son[i]>2){
            cout<<"No"<<endl;
            return 0;
        }
        if(son[i]==1)
            root=i;
    }
    if(!ok){
        cout<<"No"<<endl;
        return 0;
    }
    vector<int>ch;
    for(auto adj:graph[root]){
        if(adj==par[root])
            continue;
        ch.push_back(adj);
    }
    int sz=ch.size();
    if(sz==1){
        Path_Finding(1,ch[0]);
        Path_Finding(ch[0],1);
        cout<<"Yes"<<endl;
        cout<<ans.size()<<endl;
        for(auto path:ans){
            cout<<path[0]<<" "<<path[1]<<endl;
        }
        return 0;
    }
    for(int i=0;i<sz;i++){
        Path_Finding(1,ch[i]);
    }
    cout<<"Yes"<<endl;
    cout<<ans.size()<<endl;
    for(auto path:ans){
        cout<<path[0]<<" "<<path[1]<<endl;
    }
    return 0;
}
