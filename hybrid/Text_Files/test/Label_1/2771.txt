#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,a[2005][205],b[205][2005],c[2005][205];
    cin>>n>>m;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            cin>>a[i][j];
        }
    }
    for(int i=0;i<m;i++){
        for(int j=0;j<n;j++){
            cin>>b[i][j];
        }
    }
    for(int i=0;i+m<=n;i++){
        for(int j=0;j+m<=n;j++){
            bool f=1;
            for(int p=i;p<i+m;p++){
                for(int q=j;q<j+m;q++){
                    if(a[p][q]!=b[q-j][p-i]){
                        f=0;
                        break;
                    }
                }
            }
            if(f){
                cout<<i+1<<' '<<j+1<<endl;
                return 0;
            }
        }
    }
}
