import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;
import java.util.stream.Collectors;

public class Main {
    static class Node {
        int color;
        List<Node> neighbors = new ArrayList<>();

        Node(int color) {
            this.color = color;
        }
    }

    static class Graph {
        List<Node> nodes;
        int blackCount = 0;

        Graph(int n) {
            nodes = new ArrayList<>(n);
            for (int i = 0; i < n; i++) {
                nodes.add(new Node(0));
            }
        }

        void addEdge(int u, int v) {
            nodes.get(u - 1).neighbors.add(nodes.get(v - 1));
            nodes.get(v - 1).neighbors.add(nodes.get(u - 1));
        }

        void setColors(int[] colors) {
            for (int i = 0; i < colors.length; i++) {
                nodes.get(i).color = colors[i];
                if (colors[i] == 2) {
                    blackCount++;
                }
            }
        }

        int minRemovals() {
            if (blackCount == nodes.size()) {
                return 0;
            }
            List<Integer> whiteNodes = nodes.stream().filter(node -> node.color == 1).map(node -> nodes.indexOf(node) + 1).collect(Collectors.toList());
            List<Integer> grayNodes = nodes.stream().filter(node -> node.color == 0).map(node -> nodes.indexOf(node) + 1).collect(Collectors.toList());
            int removals = blackCount;
            removals += Math.ceil((double) whiteNodes.size() / 2);
            removals += (grayNodes.size() % 2 == 0) ? grayNodes.size() / 2 : grayNodes.size() / 2 + 1;
            return removals;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t; i++) {
            int n = Integer.parseInt(br.readLine());
            StringTokenizer st = new StringTokenizer(br.readLine());
            int[] colors = new int[n];
            for (int j = 0; j < n; j++) {
                colors[j] = Integer.parseInt(st.nextToken());
            }
            Graph graph = new Graph(n);
            for (int j = 0; j < n - 1; j++) {
                st = new StringTokenizer(br.readLine());
                int u = Integer.parseInt(st.nextToken());
                int v = Integer.parseInt(st.nextToken());
                graph.addEdge(u, v);
            }
            graph.setColors(colors);
            sb.append(graph.minRemovals()).append("\n");
        }
        System.out.println(sb);
    }
}
