#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define INF 1e9

int n, m, dx[2] = {1, 0}, dy[2] = {0, 1};
vector<pair<int, int>> volcano;
vector<vector<int>> dist;

bool valid(int x, int y) {
    if (x < 1 || x > n || y < 1 || y > n) return false;
    for (auto v : volcano) if (v.first == x && v.second == y) return false;
    return true;
}
int main() {
    cin >> n >> m;
    volcano.resize(m);
    dist.resize(n + 1, vector<int>(n + 1, INF));
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        volcano[i] = {x, y};
    }
    dist[1][1] = 0;
    queue<pair<int, int>> q;
    q.push({1, 1});
    while (!q.empty()) {
        int x = q.front().first;
        int y = q.front().second;
        q.pop();
        for (int k = 0; k < 2; k++) {
            int nx = x + dx[k], ny = y + dy[k];
            if (!valid(nx, ny)) continue;
            if (dist[nx][ny] > dist[x][y] + 1) {
                dist[nx][ny] = dist[x][y] + 1;
                q.push({nx, ny});
            }
        }
    }
    if (dist[n][n] == INF) cout << -1 << "\n";
    else cout << dist[n][n] << "\n";
    return 0;
}
