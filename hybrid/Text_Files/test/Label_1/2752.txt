#include <bits/stdc++.h>

using namespace std;

const int N = 503;
// dx, dy: each cell has 4 neighbours
const int dx[] = {-1, 0, 1, 0};
const int dy[] = {0, 1, 0, -1};

char g[N][N];
bool vis[N][N];

int main() {
  int n, m, x1, y1, x2, y2;
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= m; ++j) cin >> g[i][j];
  }
  cin >> x1 >> y1 >> x2 >> y2;
  queue<pair<int, int>> q;
  q.push({x1, y1});
  vis[x1][y1] = true;
  while (!q.empty()) {
    int x = q.front().first;
    int y = q.front().second;
    q.pop();
    if (x == x2 && y == y2) {
      cout << "YES" << endl;
      return 0;
    }
    for (int i = 0; i < 4; ++i) {
      int nx = x + dx[i];
      int ny = y + dy[i];
      if (nx <= 0 || nx > n || ny <= 0 || ny > m) continue;
      if (!vis[nx][ny] && g[nx][ny] != 'X') {
        vis[nx][ny] = true;
        q.push({nx, ny});
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
