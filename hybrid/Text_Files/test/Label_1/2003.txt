#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<bool> v(26, 0);
  for (int i = 0; i < n; i++) {
    if (s[i] >= 'a' && s[i] <= 'z') {
      v[s[i] - 'a'] = 1;
    } else {
      v[s[i] - 'A'] = 1;
    }
  }
  bool flag = true;
  for (int i = 0; i < 26; i++) {
    if (v[i] == 0) {
      flag = false;
    }
  }
  if (flag) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
