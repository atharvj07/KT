#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
using namespace std;

#define ll long long

ll gcd(ll a, ll b) {
    if (a == 0)
        return b;
    return gcd(b % a, a);
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        
        ll d = -1;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (gcd(a[i], a[j]) == 1) {
                    d = 0;
                }
            }
        }
        
        if (d == 0) { cout << 0 << '\n'; continue; }
        
        ll mx = *max_element(a.begin(), a.end());
        
        for (ll i = 1; i * i <= mx; i++) {
            if (mx % i == 0) {
                bool ok = true;
                for (int j = 0; j < n; j++) {
                    if (a[j] % i == 0) {
                        if (j > 0 && a[j - 1] % i == 0) { ok = false; }
                        if (j < n - 1 && a[j + 1] % i == 0) { ok = false; }
                    }
                }
                if (ok) {
                    d = max(d, i);
                    break;
                }
                
                if (mx % (mx / i) == 0) {
                    bool ok = true;
                    for (int j = 0; j < n; j++) {
                        if (a[j] % (mx / i) == 0) {
                            if (j > 0 && a[j - 1] % (mx / i) == 0) { ok = false; }
                            if (j < n - 1 && a[j + 1] % (mx / i) == 0) { ok = false; }
                        }
                    }
                    if (ok) {
                        d = max(d, mx / i);
                        break;
                    }
                }
            }
        }
        
        cout << d << '\n';
    }
}
