import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
g = [[] for _ in range(n+1)]
b = [0] * (n+1)
for _ in range(n-1):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

def dfs(x, p, sum):
    b[x] = a[x]
    for y in g[x]:
        if y != p:
            b[x] = max(b[x], dfs(y,x, a[x]))
    sum += b[x]
    return sum

print(dfs(1, 1, 0) % (10**9 + 7))
