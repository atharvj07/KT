#include <bits/stdc++.h>
#define ll long long
#define INF 0x3f3f3f3f
using namespace std;

const ll N = 1e6 + 5;
ll a[N], cnt[N], p[N]; // p[i] stores the expected value of cnt[i]

int main() {
    ll n;
    cin >> n;
    for (ll i = 1; i <= n; i++) { cin >> a[i]; }
    // Calculate cumulative sum
    for (ll i = 1; i <= n; i++) { cnt[a[i]]++; }
    for (ll i = 1; i <= n; i++) { p[i] = p[i - 1] + 1LL * cnt[i] * (cnt[i] - 1) / 2; }
    // Calculate expected value
    ll ans = 0;
    for (ll i = 1; i <= n; i++) {
        ans += 1LL * (p[n] - p[i - 1]) * cnt[a[i]] + 1LL * (cnt[a[i]] - 1) * (cnt[a[i]] - 2) / 2;
    }
    ans /= 2.0;
    printf("%.6f\n", 1.0 * ans / n / (n - 1));
    return 0;
}
