#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long int n;
    cin>>n;
    string s;
    cin>>s;
    long long int ans=0;
    stack<char>st;
    for(long long int i=0;i<n;i++)
    {
        if(s[i]=='(')
        {
            st.push(s[i]);
        }
        else
        {
            if(!st.empty())
                st.pop();
            else
                ans++;
        }
    }
    cout<<st.size()+ans<<endl;

    return 0;
}
