import java.util.*;
import java.math.BigInteger;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int t = scanner.nextInt();
        List<Integer> tanks = new ArrayList<>();
        for (int i = 0; i < t; i++) {
            tanks.add(scanner.nextInt());
        }
        scanner.close();

        // dp[i] = number of ways to form a valid patrol using tanks[0], ..., tanks[i]
        long[] dp = new long[t];
        dp[0] = 1;
        int gcd = 0;
        for (int i = 1; i < t; i++) {
            for (int j = 0; j < i; j++) {
                int diff = Math.abs(tanks.get(i) - tanks.get(j));
                if (diff > m) {
                    gcd = gcd(gcd, diff);
                }
            }
            dp[i] = dp[i - 1] * 2;
            dp[i] %= MOD;
        }

        System.out.println(dp[t - 1]);
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
