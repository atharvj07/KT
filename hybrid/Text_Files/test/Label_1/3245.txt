#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k, l;
    cin >> n >> k >> l;
    vector<int> d(n);
    for (int i = 0; i < n; i++) { cin >> d[i]; }
    int p = 2 * k;
    bool ok = 1;
    for (int i = 0; i < n; i++) {
      ok &= d[i] + max(p / 2, p / 2 + (i % p >= p / 2) - (i % p < p / 2)) <= l;
    }
    cout << (ok ? "Yes" : "No") << endl;
  }
  return 0;
}
