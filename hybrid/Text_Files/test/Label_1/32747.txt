import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int sum = 0;
            boolean odd = false;
            for (int i = 0; i < n; i++) {
                sum += s.charAt(i) - '0';
                if ((s.charAt(i) - '0') % 2 == 1) {
                    odd = true;
                }
            }
            if (sum % 2 == 0 && odd) {
                System.out.println(s);
            } else {
                System.out.println(-1);
            }
        }
    }
}
