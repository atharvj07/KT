#include<bits/stdc++.h>
using namespace std;
int main(){
ios_base::sync_with_stdio(false);
cin.tie(nullptr);
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int arr[n+1];
        for(int i=0;i<n;i++)
            cin>>arr[i];
        int l=0, r=n-1, ans[n+1];
        for(int i=0;i<n;i++){
            if(i%2==0){
                ans[l]=arr[i];
                l++;
            }
            else{
                ans[r]=arr[i];
                r--;
            }
        }
        for(int i=0;i<n;i++)
            cout<<ans[i]<<" ";
        cout<<endl;
    }
    return 0;
}
