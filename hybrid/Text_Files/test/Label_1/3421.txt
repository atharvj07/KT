#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  std::string s;
  std::cin >> s;
  std::vector<int> nums;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] >= '0' && s[i] <= '9') nums.push_back(s[i] - '0');
  }
  std::sort(nums.begin(), nums.end());
  for (int i = 0; i < nums.size(); i++) {
    std::cout << nums[i];
    if (i != nums.size() - 1) std::cout << "+";
  }
  std::cout << std::endl;
  return 0;
}
