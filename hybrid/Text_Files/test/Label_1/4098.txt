#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k, temp = 100;
    cin >> n >> k;
    vector<int> v(n), res;
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    sort(v.begin(), v.end());
    for (int i = 0; i < n - k + 1; i++) {
        temp = min(temp, v[i + k - 1] - v[i]);
    }
    for (int i = 0; i < n - k + 1; i++) {
        if (temp == v[i + k - 1] - v[i]) {
            for (int j = 0; j < k; j++) res.push_back(i + j + 1);
            break;
        }
    }
    cout << temp << endl;
    for (auto& i : res) cout << i << ' ';
    return 0;
}
