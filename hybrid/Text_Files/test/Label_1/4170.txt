#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int n,k;
vector<int> g[200001];
int d[200001];
vector<int> used;

void bfs(int x, int dep){
    queue<int> que;
    d[x] = dep;
    que.push(x);
    while(!que.empty()){
        int tmpx = que.front();
        que.pop();
        for(int i = 0; i < g[tmpx].size();++i){
            if(d[g[tmpx][i]] == -1){
                d[g[tmpx][i]] = dep + 1;
                que.push(g[tmpx][i]);
            }
        }
    }
}

int main(){
    cin >> n >> k;
    used.resize(n + 1);
    used.assign(n + 1,false);
    for(int i = 0; i < 2 * k;++i){
        int tmp;
        cin >> tmp;
        used[tmp] = true;
    }
    for(int i = 0; i < n - 1;++i){
        int u,v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    int mxdis,ans = 0;
    for(int i = 1; i <= n;++i){
        if(used[i]){
            for(int j = 0; j <= n;++j){
                d[j] = -1;
            }
            mxdis = 0;
            bfs(i,0);
            for(int j = 1; j <= n;++j){
                if(used[j]){
                    mxdis = max(mxdis,d[j]);
                }
            }
            ans += mxdis;
        }
    }
    cout << ans << endl;
    return 0;
}
