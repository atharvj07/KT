#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> c(n);
  for (int i = 0; i < n; i++) {
    cin >> c[i];
  }

  sort(c.begin(), c.end());
  long long ans = 0, bonus = 0, sum = 0;
  for (int i = 0; i < n; i++) {
    sum += c[i];
    if (sum >= 0) {
      ans += sum;
      bonus += sum;
      sum = 0;
    } else {
      if (k == 0) break;
      k--;
      bonus = 0;
      i--;
    }
  }
  cout << ans << endl;
  return 0;
}
