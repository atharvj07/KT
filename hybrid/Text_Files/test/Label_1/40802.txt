for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [list(map(int, input().split())) for i in range(n)]
    b = [[a[i][j] + n - i - 1 for j in range(m)] for i in range(n)]
    c = [[b[i][j] + m - j - 1 for j in range(m)] for i in range(n)]
    c = [[max(a[i][j], b[i][j], c[i][j]) for j in range(m)] for i in range(n)]
    ans = sum(sum(j - i for i in a[k] for j in b[k]) for k in range(n))
    for i in range(n):
        for j in range(m):
            ans = min(ans, ans + j - i + sum(min(a[i][k], b[i][k]) for k in range(m)) + sum(min(j, a[k][j]) for k in range(n)))
            ans = min(ans, ans + n - i - 1 - j + sum(min(a[i][k], c[i][k]) for k in range(m)) + sum(min(n - i - 1, a[k][j]) for k in range(n)))
    print(ans)
