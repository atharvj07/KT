import collections
n,s=map(int,input().split())
a=[[]for i in range(n+1)]
for i in range(n-1):
    x,y=map(int,input().split())
    a[x].append(y)
    a[y].append(x)
vis=[False for i in range(n+1)]
def dfs(x,p):
    d=[0 for i in range(n+1)]
    q=[(x,0)]
    while q:
        x,u=q.pop()
        if vis[x]:continue
        vis[x]=True
        d[x]=u
        for i in a[x]:
            if i!=p:q.append((i,u+1))
    return d
d1=dfs(1,0)
x=d1.index(max(d1))
vis=[False for i in range(n+1)]
d2=dfs(x,0)
print(s/d2[d2.index(max(d2))])
