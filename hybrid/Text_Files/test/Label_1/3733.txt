#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  long long int a, b, n, m;
  cin >> n >> m;
  vector<long long int> A(n), B(m);
  for (long long int i = 0; i < n; i++) { cin >> A[i]; }
  for (long long int i = 0; i < m; i++) { cin >> B[i]; }
  bool flag = false;
  long long int maxa = 0, maxb = 0;
  for (auto &it : A) {
    maxa = max(maxa, it);
  }
  for (auto &it : B) {
    maxb = max(maxb, it);
  }
  if (maxa != maxb) {
    cout << "-1" << endl;
    return 0;
  }
  vector<long long int> c;
  long long int temp = -1;
  for (long long int i = 0; i < n; i++) {
    if (A[i] != temp) {
      c.push_back(0);
      temp = A[i];
    }
    c.back() += A[i];
  }
  temp = -1;
  vector<long long int> d;
  for (long long int i = 0; i < m; i++) {
    if (B[i] != temp) {
      d.push_back(0);
      temp = B[i];
    }
    d.back() += B[i];
  }
  if (c.size() != d.size()) {
    cout << "-1" << endl;
    return 0;
  }
  for (long long int i = 0; i < c.size(); i++) {
    if (c[i] != d[i]) {
      cout << "-1" << endl;
      return 0;
    }
  }
  cout << maxa << endl;
}
