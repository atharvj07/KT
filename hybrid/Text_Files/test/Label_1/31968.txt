import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        int[] y = new int[n];
        char[] u = new char[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
            u[i] = sc.next().charAt(0);
        }

        int ans = -1;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (u[i] == u[j]) {
                    if (u[i] == 'U' || u[i] == 'D') {
                        if (x[i] == x[j]) {
                            int dy = Math.abs(y[i] - y[j]) - 1;
                            if (dy > 0) {
                                ans = Math.max(ans, dy / 2);
                            }
                        }
                    } else {
                        if (y[i] == y[j]) {
                            int dx = Math.abs(x[i] - x[j]) - 1;
                            if (dx > 0) {
                                ans = Math.max(ans, dx / 2);
                            }
                        }
                    }
                } else if ((u[i] == 'U' || u[i] == 'D') && (u[j] == 'R' || u[j] == 'L')) {
                    int dx = x[j] - x[i];
                    int dy = y[j] - y[i];
                    if (dx * dx + dy * dy > 0) {
                        ans = Math.max(ans, dx / 2 + dy / 2);
                    }
                } else if ((u[i] == 'R' || u[i] == 'L') && (u[j] == 'U' || u[j] == 'D')) {
                    int dx = x[i] - x[j];
                    int dy = y[i] - y[j];
                    if (dx * dx + dy * dy > 0) {
                        ans = Math.max(ans, dx / 2 + dy / 2);
                    }
                }
            }
        }

        if (ans == -1) {
            System.out.println("SAFE");
        } else {
            System.out.println(ans);
        }
    }
}
