n, m, d = map(int, input().split())
adj = [set() for i in range(n + 1)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    adj[a].add(b)
    adj[b].add(a)

affected = set(map(int, input().split()))
dist = [-1] * (n + 1)
q = [i for i in affected]
dist[0] = 0
while q:
    v = q.pop(0)
    for u in adj[v]:
        if dist[u] == -1:
            dist[u] = dist[v] + 1
            if dist[u] <= d:
                q.append(u)

print(sum(1 for i in range(1, n + 1) if dist[i] <= d and i not in affected))
