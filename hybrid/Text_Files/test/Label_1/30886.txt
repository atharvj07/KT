import java.util.Scanner;

public class RoadTrip {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] roads = new int[3][2];

        for (int i = 0; i < 3; i++) {
            roads[i][0] = scanner.nextInt();
            roads[i][1] = scanner.nextInt();
        }

        boolean[][] connected = new boolean[5][5];
        for (int[] road : roads) {
            connected[road[0]][road[1]] = true;
            connected[road[1]][road[0]] = true;
        }

        boolean[] visited = new boolean[5];
        boolean isPossible = dfs(connected, visited, 1);

        System.out.println(isPossible ? "YES" : "NO");
    }

    static boolean dfs(boolean[][] connected, boolean[] visited, int current) {
        visited[current] = true;

        for (int i = 1; i < 5; i++) {
            if (connected[current][i] && !visited[i]) {
                if (!dfs(connected, visited, i)) {
                    return false;
                }
            }
        }

        return true;
    }
}
