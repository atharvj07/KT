import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        Arrays.sort(s);
        int ans = 0;
        int remain = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (remain + s[i] <= 4) {
                remain += s[i];
            } else {
                remain = s[i];
                ans++;
            }
        }
        System.out.println(ans);
    }
}
