import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long k = sc.nextLong();
        char[][] map = new char[n][m];
        for (int i = 0; i < n; i++) {
            map[i] = sc.next().toCharArray();
        }
        int[][] cross = new int[n][m];
        for (int i = 1; i < n - 1; i++) {
            for (int j = 1; j < m - 1; j++) {
                if (map[i][j] == '*') {
                    int radius = 1;
                    while (i + radius < n && i - radius >= 0 && j + radius < m && j - radius >= 0 &&
                            map[i + radius][j] == '*' && map[i - radius][j] == '*' && map[i][j + radius] == '*' && map[i][j - radius] == '*') {
                        radius++;
                    }
                    radius--;
                    cross[i][j] = radius;
                }
            }
        }
        int[][] crossCnt = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int[] dir : new int[][]{{1, 0}, {-1, 0}, {0, 1}, {0, -1}}) {
                    int x = i + dir[0], y = j + dir[1];
                    while (x >= 0 && x < n && y >= 0 && y < m) {
                        crossCnt[x][y]++;
                        x += dir[0];
                        y += dir[1];
                    }
                }
            }
        }
        int cnt = 0;
        int[] ans = new int[5];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (cross[i][j] != 0) {
                    if (crossCnt[i][j] == 5) {
                        cnt++;
                        ans[0] = i + 1;
                        ans[1] = i + 1 - cross[i][j];
                        ans[2] = i + 1 + cross[i][j];
                        ans[3] = j + 1 - cross[i][j];
                        ans[4] = j + 1 + cross[i][j];
                        if (cnt == k) {
                            for (int x : ans) {
                                System.out.print(x + " ");
                            }
                            return;
                        }
                    }
                }
            }
        }
        System.out.println(-1);
    }
}
