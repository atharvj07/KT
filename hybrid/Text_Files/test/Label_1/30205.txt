import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        long[] piles = new long[n];
        for (int i = 0; i < n; i++) {
            piles[i] = scanner.nextLong();
        }
        int winAlice = 0;
        int winBob = 0;
        int winFirst = 0;
        int winSecond = 0;
        for (int a = 1; a <= m; ++a) {
            for (int b = 1; b <= m; ++b) {
                boolean aliceWin;
                if (a + b <= m) {
                    aliceWin = true;
                } else {
                    long sum = 0;
                    for (long pile : piles) {
                        sum += Math.max(pile / b, pile / a);
                    }
                    aliceWin = (sum & 1) == 0;
                }
                if (aliceWin) {
                    ++winAlice;
                    if (a == b) {
                        ++winFirst;
                    } else {
                        ++winSecond;
                    }
                } else {
                    ++winBob;
                    if (a == b) {
                        ++winSecond;
                    } else {
                        ++winFirst;
                    }
                }
            }
        }
        System.out.println(winAlice + " " + winBob + " " + winFirst + " " + winSecond);
    }
}
