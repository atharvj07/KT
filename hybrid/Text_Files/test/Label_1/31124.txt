import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long result = 1;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            if (x > 0) {
                result *= x;
            }
        }
        System.out.println(result);
    }
}
