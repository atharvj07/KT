import sys

n, m, k, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
gadgets = []
for i in range(m):
    t, c = map(int, input().split())
    if t == 1:
        gadgets.append((c, a))
    else:
        gadgets.append((c, b))

gadgets.sort()

days = []
for i in range(n):
    cost = 0
    for j in range(m):
        if cost + gadgets[j][1][i] <= s:
            cost += gadgets[j][1][i]
            days.append((j + 1, i + 1))
    if cost >= s:
        break

if len(days) < k:
    print(-1)
else:
    print(days[k - 1][1])
    for i in range(k):
        print(days[i][0], days[i][1])
