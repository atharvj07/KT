n,k=map(int,input().split())
c=list(map(int,input().split()))
dp=[[False]*501 for i in range(n+1)]
dp[0][0]=True
for i in range(n):
    for j in range(k+1):
        dp[i+1][j]=dp[i+1][j] or dp[i][j]
        dp[i+1][j+c[i]]=dp[i+1][j+c[i]] or dp[i][j]
res=[]
for i in range(k+1):
    if dp[n][i]:res+=[i]
print(len(res))
print(*res)
