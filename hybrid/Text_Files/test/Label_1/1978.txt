#include <bits/stdc++.h>
using namespace std;
const int MAXN = 52;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> stages;
  for (int i = 0; i < n; i++) {
    int weight = s[i] - 'a' + 1;
    if (i < n - 1 && weight + 2 <= s[i + 1] - 'a' + 1) {
      continue;
    }
    stages.push_back(weight);
  }
  int m = stages.size();
  if (m < k) {
    cout << -1 << endl;
    return 0;
  }
  sort(stages.begin(), stages.end());
  int ans = 0;
  for (int i = 0; i < k; i++) {
    ans += stages[i];
  }
  cout << ans << endl;
  return 0;
}
