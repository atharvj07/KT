def solve():
    n=int(input())
    uf=[i for i in range(n+1)]
    def find(x):
        if x!=uf[x]:uf[x]=find(uf[x])
        return uf[x]
    def uni(x,y):
        uf[find(x)]=find(y)

    ans=[1]*n
    for i in range(n-1):
        x,y=map(int,input().split())
        uni(x,y)
        for j in range(1,n+1):
            if find(j)==find(x):
                ans[find(x)-1]=j
    return ans

print(*solve())
