n = int(input())
c = list(map(int, input().split()))

s = []
alphabet = 'abcdefghijklmnopqrstuvwxyz'

idx = 0
for i in range(n):
    if c[i] > 0:
        length = i + 1
        count = c[i] // length
        for _ in range(count):
            s.append(alphabet[idx] * length)
            idx = (idx + 1) % 26
        for j in range(i, n):
            c[j] -= count * length * (j - i + 1)

print(''.join(s))
