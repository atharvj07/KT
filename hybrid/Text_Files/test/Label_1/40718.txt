n, m = map(int, input().split())
good_cells = [list(map(int, input().split())) for _ in range(n)]

# Check if any good cell is on the border
top = any(good_cells[0])
bottom = any(good_cells[-1])
left = any(row[0] for row in good_cells)
right = any(row[-1] for row in good_cells)

if top or bottom or left or right:
    print(2)
else:
    print(4)
