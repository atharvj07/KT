from collections import defaultdict

s = input()
n = int(input())
rules = []
for i in range(n):
    rule, l, r = map(str, input().split())
    rules.append((rule, int(l), int(r)))
good = []
for i in range(1, len(s) + 1):
    for j in range(i):
        t = s[j:i]
        cnt = 0
        for rule, l, r in rules:
            occurrences = 0
            for k in range(len(rule) - len(t) + 1):
                if rule[k:k + len(t)] == t:
                    occurrences += 1
            if occurrences not in range(l, r + 1):
                break
        else:
            good.append(t)
print(len(set(good)))
