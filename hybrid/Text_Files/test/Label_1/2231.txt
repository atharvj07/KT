#include <cstdio>
#define ll long long

const ll mod = 1e9 + 9;
const ll inv2 = 500000447;

ll fastpow(ll x,ll p){
    if(p == 0) return 1;
    if(p == 1) return x;

    ll tmp = fastpow(x, p / 2);
    tmp = tmp * tmp % mod;
    if(p & 1) tmp = tmp * x % mod;

    return tmp;
}

ll C(ll nn,ll m){
    if(nn < m) return 0;
    ll den = 1;
    for(ll i = 1;i <= m;i++) den = den * i % mod;
    ll num = 1;
    for(ll i = nn;i >= nn - m + 1;i--) num = num * i % mod;

    return num * fastpow(den,mod-2) % mod;
}

int main(){
    ll n,m;
    scanf("%lld%lld",&n,&m);

    if(n == 1){
        printf("1\n");
        return 0;
    }

    ll ans = fastpow(n,n) % mod;
    ans = ans + (n - 1) * n % mod * fastpow(n-1,n-2) % mod;
    ans = ans * inv2 % mod;
    ans = ans - 1;
    ans = (ans % m + m) % m;
    printf("%lld\n",ans);
}
