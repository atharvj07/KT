import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int t = sc.nextInt();
        int[] q = new int[t];
        for (int i = 0; i < t; i++) {
            q[i] = sc.nextInt();
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = p[i];
        }
        for (int i = 0; i < t; i++) {
            x[q[i] - 1] = -x[q[i] - 1];
        }
        boolean valid = true;
        int balance = 0;
        for (int i = 0; i < n; i++) {
            balance += x[i];
            if (balance < 0) {
                valid = false;
                break;
            }
        }
        if (balance != 0) {
            valid = false;
        }
        if (valid) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(x[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
        sc.close();
    }
}
