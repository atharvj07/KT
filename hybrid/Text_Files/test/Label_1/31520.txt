import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] teams = new String[n];
        String[] towns = new String[n];
        HashMap<String, Integer> firstOptions = new HashMap<>();
        HashMap<String, Integer> secondOptions = new HashMap<>();
        for (int i = 0; i < n; i++) {
            teams[i] = sc.next();
            towns[i] = sc.next();
            String firstOption = teams[i].substring(0, 3);
            String secondOption = teams[i].substring(0, 2) + towns[i].charAt(0);
            if (firstOptions.containsKey(firstOption) || secondOptions.containsKey(firstOption)) {
                System.out.println("NO");
                return;
            }
            if (secondOptions.containsKey(secondOption)) {
                System.out.println("NO");
                return;
            }
            firstOptions.put(firstOption, i);
            secondOptions.put(secondOption, i);
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            String firstOption = teams[i].substring(0, 3);
            String secondOption = teams[i].substring(0, 2) + towns[i].charAt(0);
            if (secondOptions.containsKey(firstOption)) {
                System.out.println(secondOption);
            } else {
                System.out.println(firstOption);
            }
        }
    }
}
