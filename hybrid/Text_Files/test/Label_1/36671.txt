t = int(input())
for _ in range(t):
    n = int(input())
    a = list(range(1, n + 1))
    ans = []
    for idx in range(n - 1):
        if a[idx] > 1:
            a[idx], a[n - 1] = a[n - 1], a[idx]
            ans.append([n - 1, idx])
            a[n - 1] = 1
    print(len(ans))
    for x, y in ans:
        print(x, y)
