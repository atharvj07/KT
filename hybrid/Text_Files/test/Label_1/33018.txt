import java.util.Scanner;

public class Main {
    private static int[] a, dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new int[n];
        dp = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            dp[i] = dp[i - 1];
            if (a[i] >= a[i - 1]) {
                dp[i]++;
            }
        }

        int ans = 0;
        for (int i = n - 2; i >= 0; i--) {
            if (a[i] >= a[i + 1]) {
                ans++;
            } else {
                ans += dp[i + 1];
                break;
            }
        }

        System.out.println(ans);
    }
}
