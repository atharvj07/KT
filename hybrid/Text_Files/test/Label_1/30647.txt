import java.util.Scanner;

public class HedgehogPresents {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int holidays = scanner.nextInt();
        int[] holidayDays = new int[holidays];
        for (int i = 0; i < holidays; i++) {
            holidayDays[i] = scanner.nextInt();
        }

        int presents = 0;
        int lastPresentDay = 0;

        for (int day = 1; day <= n; day++) {
            if (day == lastPresentDay + 1 || isHoliday(day, holidayDays)) {
                presents++;
                lastPresentDay = day;
            } else if (day > lastPresentDay + k) {
                presents++;
                lastPresentDay = day;
            }
        }

        System.out.println(presents);
    }

    private static boolean isHoliday(int day, int[] holidayDays) {
        for (int holidayDay : holidayDays) {
            if (day == holidayDay) {
                return true;
            }
        }
        return false;
    }
}
