#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> s(n);
  for (auto &x : s) cin >> x;
  bool ok = 0;
  for (int x = 0; x < n; ++x)
    for (int y = 0; y < m; ++y)
      if (!ok && s[x][y] == '.' && s[x][y - 1] == '.' && s[x][y + 1] == '.' &&
          s[x + 1][y] == '.' && s[x + 1][y - 1] == '.' && s[x + 1][y + 1] == '.' &&
          s[x - 1][y] == '.' && s[x - 1][y - 1] == '.' && s[x - 1][y + 1] == '.' &&
          x + 1 < n && x > 0 && y + 1 < m && y > 0)
        ok = 1;
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
