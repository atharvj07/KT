import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    while (true) {
      int n = in.nextInt();
      if (n == 0) {
        break;
      }
      String[] words = new String[n];
      for (int i = 0; i < n; i++) {
        words[i] = in.next();
      }
      boolean isSorted = true;
      for (int i = 1; i < n; i++) {
        if (words[i - 1].compareTo(words[i]) > 0) {
          isSorted = false;
          break;
        }
        if (words[i - 1].length() > words[i].length()) {
          String prefix = words[i - 1].substring(0, words[i].length());
          if (prefix.compareTo(words[i]) > 0) {
            isSorted = false;
            break;
          }
        }
      }
      System.out.println(isSorted ? "yes" : "no");
    }
  }
}
