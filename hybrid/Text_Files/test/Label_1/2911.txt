#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string a;
    cin >> a;
    int n = a.size();
    int ones = 0;
    bool ans = 1;
    for (int i = 0; i < n; i++) {
      if (a[i] == '1') {
        ones++;
      } else {
        if (ones > 0) {
          ans &= (ones % 2 == 0);
          ones = 0;
        }
      }
    }
    if (ones > 0) {
      ans &= (ones % 2 == 0);
    }
    if (ans) {
      cout << "YES" << endl;
    } else {
      cout << "NO" << endl;
    }
  }
  return 0;
}
