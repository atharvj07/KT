import java.util.Scanner;

public class ValerieTheShark {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt() - 1;
            teeth[i][1] = scanner.nextInt();
        }

        int maxCrucians = 0;
        for (int i = 0; i < m; i++) {
            int currentRowCrucians = 0;
            for (int j = 0; j < n; j++) {
                if (teeth[j][0] == i) {
                    currentRowCrucians = Math.min(currentRowCrucians, teeth[j][1]);
                }
            }
            maxCrucians += currentRowCrucians;
        }

        System.out.println(Math.min(maxCrucians, k));
    }
}
