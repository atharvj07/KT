import java.util.List;
import java.util.Arrays;
import java.util.ArrayList;

class Main {
    private static long calculatePeriod(int base, int num) {
        long val = num;
        int period = 0;

        do {
            val = val * num % base;
            period++;
        } while (val != num);

        return period;
    }

    private static List<Integer> getRotatableInBase(int base, int num, int len) {
        long curVal = num;
        List<Integer> ret = new ArrayList<Integer>();

        for (int i = 0; i <= len; i++) {
            ret.add((int) curVal);
            curVal = curVal * base % (base * pow(10, len - i));
        }

        return ret;
    }

    private static int pow(int base, int exp) {
        if (exp == 0) {
            return 1;
        }

        int ret = base;
        for (int i = 1; i < exp; i++) {
            ret *= base;
        }

        return ret;
    }

    public static void main(String[] args) {
        java.util.Scanner reader = new java.util.Scanner(System.in);
        int n = reader.nextInt();
        int x = reader.nextInt();
        boolean found = false;

        for (int base = x - 1; base > 1; base--) {
            int num = pow(10, n - 1);

            while (num < pow(base, n)) {
                List<Integer> rotatables = getRotatableInBase(base, num, n);
                int period = (int) calculatePeriod(base, num);
                
                boolean isRotatable = true;
                for (int i = 1; i <= n; i++) {
                    if (rotatables.get(i) % base != rotatables.get(i % period)) {
                        isRotatable = false;
                        break;
                    }
                }

                if (isRotatable) {
                    System.out.println(base);
                    found = true;
                    break;
                }

                num++;
            }

            if (found) {
                break;
            }
        }

        if (!found) {
            System.out.println(-1);
        }
    }
}
