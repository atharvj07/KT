import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long m = sc.nextLong();
        long mod = 1000000007;
        long ans = 0;
        if (m % 2 == 0) {
            ans = (m - 2) * (m - 2) % mod;
        } else {
            ans = (m - 1) * (m - 1) % mod;
        }
        System.out.println(ans);
    }
}
