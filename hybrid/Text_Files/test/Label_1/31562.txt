import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int max = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (a[j] > a[i]) {
                    a[i] = a[j] - a[i];
                    a[j] = 0;
                } else {
                    a[j] = a[i] - a[j];
                    a[i] = 0;
                }
                max = Math.max(max, a[i]);
                max = Math.max(max, a[j]);
            }
        }

        System.out.println(max);
    }
}
