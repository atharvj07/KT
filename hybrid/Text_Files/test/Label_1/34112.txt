import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int R = sc.nextInt();
        int Q = sc.nextInt();
        int[] a = new int[M];
        int[] b = new int[M];
        int[] c = new int[M];
        int[] x = new int[Q];
        int[] y = new int[Q];
        int[] z = new int[Q];
        for (int i = 0; i < M; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        for (int i = 0; i < Q; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
            z[i] = sc.nextInt();
        }
        int[][] dp = new int[Q + 1][N + 1];
        for (int i = 0; i <= Q; i++) {
            for (int j = 0; j <= N; j++) {
                dp[i][j] = -1;
            }
        }
        dp[0][0] = 0;
        for (int i = 0; i < Q; i++) {
            for (int j = 0; j <= N; j++) {
                if (dp[i][j] == -1) {
                    continue;
                }
                dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j]);
                for (int k = 0; k < M; k++) {
                    if (a[k] == j) {
                        dp[i + 1][b[k]] = Math.max(dp[i + 1][b[k]], dp[i][j] - c[k]);
                    } else if (b[k] == j) {
                        dp[i + 1][a[k]] = Math.max(dp[i + 1][a[k]], dp[i][j] - c[k]);
                    }
                }
                if (j % R == x[i] % R && j % R == y[i] % R) {
                    dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j] + z[i]);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i <= N; i++) {
            ans = Math.max(ans, dp[Q][i]);
        }
        System.out.println(ans);
    }
}
