#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> last(n + 1, -1);
  int ans = 0;
  for (int i = 0, r = 0; i < n; i++) {
    if (last[a[i]] < r) { ans++; }
    r = max(r, last[a[i]] + 1);
    last[a[i]] = i;
  }
  cout << ans << endl;
  return 0;
}
