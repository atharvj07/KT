n, k = map(int, input().split())
s = input()
if k % 2 == 0:
    print(s)
else:
    t = ''
    for i in range(n):
        j = (i - 1 + n) % n
        l = (i + 1) % n
        if s[i] == s[j] and s[i] == s[l]:
            t += s[i]
        else:
            t += 'W' if s[i] == 'B' else 'B'
    print(t)
