import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long[] a = new long[4];
        for (int i = 0; i < 4; i++) {
            a[i] = sc.nextLong();
        }
        long n = a[0];
        long l = a[1];
        long r = a[2];
        long k = a[3];
        long ans = 2 * ((k + (l - r - 1 + n) % n + 1) / n);
        if (ans > n) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }
}
