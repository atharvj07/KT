#include <cstdio>
#include <vector>
using namespace std;
typedef long long ll;
template <class T>
void vec(vector<T> &v, int n, T ini) { v.resize(n, ini); }
typedef vector<ll> vi;
const int MX = 50+5;
bool vert[MX], horz[MX];
int main() {
  int n;
  scanf("%d", &n);
  for (int d = 0; d < n*n; d++) {
    int h, v;
    scanf("%d %d", &h, &v);
    if (!vert[v] || !horz[h]) {
      printf("%d ", d+1);
      vert[v] = horz[h] = 1;
    }
  }
  printf("\n");
  return 0;
}
