import java.util.*;
import java.io.*;

class Main {
    static final int MAX_N = 2000, INF = 99999999;
    private static Scanner sc = new Scanner(System.in);
    private static PrintWriter pw = new PrintWriter(System.out);
    private static int[][] nxt, dist;
    private static int[] pth;
    private static char[] u, v;

    public static void main(String[] args) {
        u = sc.next().toCharArray();
        v = sc.next().toCharArray();
        int n = u.length;
        nxt = new int[n + 1][26];
        dist = new int[n + 1][26];
        pth = new int[n + 1];
        Arrays.fill(pth, -1);
        Arrays.fill(dist[n], INF);
        dist[n][25] = 0;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 26; j++) {
                if (i + 1 == n) {
                    nxt[i][j] = n;
                    dist[i][j] = INF;
                } else {
                    nxt[i][j] = nxt[i + 1][j];
                    dist[i][j] = dist[i + 1][j];
                    if (u[i + 1] == (char) (j + 'a')) {
                        nxt[i][j] = i + 1;
                        dist[i][j] = dist[i + 1][j] + 1;
                    }
                }
            }
        }
        int l = 0;
        for (int r = n - 1; r >= 0; r--) {
            for (int j = 0; j < 26; j++) {
                if (v[l] == (char) (j + 'a')) {
                    int d = dist[r][j] + n - l - r;
                    if (d < dist[0][25]) {
                        dist[0][25] = d;
                        pth[0] = r;
                    }
                }
            }
            l++;
        }
        if (dist[0][25] == INF) {
            pw.println(-1);
        } else {
            List<Integer> ans = new ArrayList<>();
            for (int i = 0; i < 25; i++) {
                int r = pth[i];
                if (r == -1) {
                    break;
                }
                while (nxt[r][v[i] - 'a'] != r) {
                    ans.add(n - r);
                    r = nxt[r][v[i] - 'a'];
                }
                for (int j = 0; j < 26; j++) {
                    if (nxt[r][j] < r) {
                        ans.add(n - r);
                        r = nxt[r][j];
                        break;
                    }
                }
            }
            pw.println(ans.size());
            for (int i : ans) {
                pw.print(i + " ");
            }
            pw.println();
        }
        pw.close();
    }
}
