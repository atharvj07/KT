MOD = 1000000007

# Function to calculate complexity
def calculate_complexity(x):
    n = len(x)
    complexity = 0
    for i in range(n):
        if x[i] == '1':
            complexity += pow(2, (2 * (n - i - 1)), MOD)
            complexity %= MOD
    return complexity

# Input processing
x = input().strip()
# Output the complexity
print(calculate_complexity(x))
