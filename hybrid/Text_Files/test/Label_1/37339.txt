cnt = [0 for _ in range(30)]

for _ in range(int(input())):
    n = int(input())
    s = input()
    t = input()
    for i in range(n):
        cnt[ord(s[i]) - ord('a')] += 1
        cnt[ord(t[i]) - ord('a')] -= 1
    
    k = 0
    for i in range(26):
        k += abs(cnt[i])
        if k > 2:
            break
        
    if k <= 2:
        print("Yes")
    else:
        print("No")
