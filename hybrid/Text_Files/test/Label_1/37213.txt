from collections import deque
n, m = map(int, input().split())
a = []
vis = [[False] * m for _ in range(n)]
d = [[-1] * m for _ in range(n)]
dx = [1, -1, 0, 0]
dy = [0, 0, 1, -1]
for i in range(n):
    a.append(input())

ans = -1
for i in range(n):
    for j in range(m):
        if a[i][j] != '.':
            vis[i][j] = True
            d[i][j] = 0
            que = deque()
            que.append((i, j))
            while que:
                x, y = que.popleft()
                for k in range(4):
                    nx, ny = x + dx[k], y + dy[k]
                    if 0 <= nx < n and 0 <= ny < m and not vis[nx][ny] and a[nx][ny] != '#':
                        vis[nx][ny] = True
                        d[nx][ny] = d[x][y] + 1
                        que.append((nx, ny))
            ans = max(ans, d[i][j])
print(ans)
