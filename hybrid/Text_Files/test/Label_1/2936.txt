#include <bits/stdc++.h>
#include <iostream>
using namespace std;
typedef long long ll;
const ll inf = 1LL << 60;
const ll maxn = 1e5;
struct goods {
  ll w, v;
};
ll n, m;
goods a[maxn + 1];
ll dp[maxn + 1][3005]; // dp[i][j] maximum value of the first i-th goods with j weight remaining

void solve() {
  cin >> n >> m;
  for (ll i = 1; i <= n; i++) {
    cin >> a[i].w >> a[i].v;
  }
  for (ll i = 1; i <= n; i++) {
    for (ll j = m; j >= a[i].w; j--) {
      dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - a[i].w] + a[i].v);
    }
  }
  cout << dp[n][m] << endl;
}
int main() {
  solve();
  return 0;
}
