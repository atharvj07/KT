import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = scanner.nextInt();
            }
        }
        while (q-- > 0) {
            int a1 = scanner.nextInt() - 1;
            int b1 = scanner.nextInt() - 1;
            int c1 = scanner.nextInt() - 1;
            int d1 = scanner.nextInt() - 1;
            int h = scanner.nextInt();
            int w = scanner.nextInt();
            for (int i = a1; i < a1 + h; i++) {
                for (int j = b1; j < b1 + w; j++) {
                    int temp = arr[i][j];
                    arr[i][j] = arr[c1 + i - a1][d1 + j - b1];
                    arr[c1 + i - a1][d1 + j - b1] = temp;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(arr[i][j] + " ");
            }
            System.out.println();
        }
    }
}
