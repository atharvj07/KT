n, a, b = map(int, input().split())
seats = input()

def max_students(n, a, b, seats):
    students_placed = 0
    i = 0
    while i < n:
        if seats[i] == '.':
            if i+1 < n and seats[i+1] == '.':
                if a > b:
                    a -= 1
                else:
                    b -= 1
                students_placed += 1
                i += 1  # Skip next seat as it would be adjacent
            else:
                if a > 0 or b > 0:
                    students_placed += 1
                    if a >= b:
                        a -= 1
                    else:
                        b -= 1
        i += 1
    return students_placed

print(max_students(n, a, b, seats))
