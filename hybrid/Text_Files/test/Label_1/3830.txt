#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int MOD = 1e9 + 7;

ll fastpow(ll x, ll n) {
  ll res = 1;
  x %= MOD;
  while (n) {
    if (n & 1) res = res * x % MOD;
    x = x * x % MOD;
    n >>= 1;
  }
  return res;
}

int main() {
  string s;
  cin >> s;
  ll n = s.size();
  ll ans = 0;
  for (ll i = 0; i < n; i++) {
    if (s[i] == '1') {
      ll p = (1LL << i) % MOD;
      ans = (ans + p * (2LL << i - 1) % MOD) % MOD;
    }
  }
  cout << ans << '\n';
  return 0;
}
