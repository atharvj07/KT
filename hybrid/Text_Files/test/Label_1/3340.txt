#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> x, y;
    for (int i = 0; i < n; ++i) {
        int xi, yi;
        cin >> xi >> yi;
        x.push_back(xi);
        y.push_back(yi);
    }
    sort(x.begin(), x.end());
    sort(y.begin(), y.end());
    int ans = 0;
    if (unique(x.begin(), x.end()) == x.end() && unique(y.begin(), y.end()) == y.end()) {
        ans = 2;
    } else if (unique(x.begin(), x.end()) == x.end()) {
        ans = 3;
    } else if (unique(y.begin(), y.end()) == y.end()) {
        ans = 3;
    } else {
        ans = 4;
    }
    cout << ans << endl;
    return 0;
}
