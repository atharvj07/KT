#include<bits/stdc++.h>
using namespace std;
int32_t main()
{
    int32_t n,m,k;
    cin>>n>>m>>k;
    int32_t l=1,r=n*m;
    while(l<r)
    {
        int32_t mid=(l+r)>>1;
        int32_t cnt=0;
        for(int32_t i=1;i<=n;i++)
        {
            cnt+=(min(m,mid/i));
        }
        if(cnt<k)
            l=mid+1;
        else
            r=mid;
    }
    cout<<l<<endl;
}
