#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, w;
    cin >> n >> w;
    vector<int> arr(n);
    for (int &i : arr) { cin >> i; }
    sort(arr.begin(), arr.end(), greater<int>());
    int ans = 1, curr = 0;
    for (int x : arr) {
      if (curr + x <= w)
        curr += x;
      else
        curr = x, ans++;
    }
    cout << ans << endl;
  }
}
