import java.util.*;

class Main {

    static final int INF = (int) 1e9;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        for (int i = 0; i < n; i++) b[i] = sc.nextInt();
        List<Integer>[] g = new ArrayList[n];
        for (int i = 0; i < n; i++) g[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt() - 1, q = sc.nextInt() - 1;
            g[p].add(q);
            g[q].add(p);
        }
        int[] deg = new int[n];
        for (int i = 0; i < n; i++) deg[i] = g[i].size();
        long sum = 0;
        for (int i = 0; i < n; i++) sum += a[i] - b[i];
        if (sum % 2 != 0 || sum < 0) {
            System.out.println("NO");
            return;
        }
        long toAdd = sum / 2;
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            if (deg[i] == 0) {
                continue;
            }
            int dif = b[i] - a[i];
            if (dif == 0) {
                continue;
            }
            if (dif > 0) {
                long cur = Math.min(toAdd, dif);
                toAdd -= cur;
                ans[i] = (int) cur;
            } else {
                ans[i] = dif;
            }
        }
        for (int i = 0; i < n; i++) {
            if (deg[i] == 0) {
                ans[i] = b[i];
                continue;
            }
            int dif = b[i] - a[i];
            if (dif < 0) {
                ans[i] += -dif;
            }
        }
        if (toAdd != 0) {
            System.out.println("NO");
            return;
        }
        for (int i = 0; i < n; i++) {
            if (ans[i] < 0) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    System.out.print(ans[i] + " ");
                    continue;
                }
                if (deg[i] == 0) {
                    System.out.print("0 ");
                    continue;
                }
                int dif = ans[i] - ans[j];
                if (dif > 0) {
                    System.out.print(dif + " ");
                    continue;
                }
                System.out.print("0 ");
            }
            System.out.println();
        }
    }
}
