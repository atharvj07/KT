import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        System.out.println(getPassengerOrder(n, m));
    }

    public static String getPassengerOrder(int n, int m) {
        StringBuilder sb = new StringBuilder();
        boolean[][] seats = new boolean[2 * n][2];
        int[] windowSeats = new int[2 * n];
        int[] aisleSeats = new int[2 * n];
        int[] backWindowSeats = new int[2 * n];
        int[] backAisleSeats = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            windowSeats[i] = 2 * i + 1;
            aisleSeats[i] = 2 * i + 2;
            backWindowSeats[2 * n - 1 - i] = 2 * i + 1;
            backAisleSeats[2 * n - 1 - i] = 2 * i + 2;
        }
        int count = 0;
        for (int i = 0; i < 2 * n; i++) {
            for (int j = 0; j < 2; j++) {
                if (count < m) {
                    if (i < n) {
                        seats[i][j] = true;
                        sb.append(windowSeats[i]).append(' ');
                    } else if (count < n * 2) {
                        seats[i][j] = true;
                        sb.append(aisleSeats[i - n]).append(' ');
                    } else if (count < n * 3) {
                        seats[i][j] = true;
                        sb.append(backWindowSeats[i - n]).append(' ');
                    } else {
                        seats[i][j] = true;
                        sb.append(backAisleSeats[i - n]).append(' ');
                    }
                    count++;
                }
            }
        }
        for (int i = 0; i < 2 * n; i++) {
            for (int j = 0; j < 2; j++) {
                if (count >= m) {
                    break;
                }
                if (!seats[i][j]) {
                    if (i < n) {
                        sb.append(windowSeats[i]).append(' ');
                    } else {
                        sb.append(aisleSeats[i - n]).append(' ');
                    }
                    count++;
                }
            }
        }
        sb.setLength(sb.length() - 1);
        return sb.toString();
    }
}
