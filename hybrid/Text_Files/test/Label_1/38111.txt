t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    matrix = [list(map(int, input().split())) for _ in range(n)]
    operations = 0
    for i in range((n+1)//2):
        for j in range((m+1)//2):
            values = [matrix[i][j], matrix[i][m-j-1], matrix[n-i-1][j], matrix[n-i-1][m-j-1]]
            values.sort()
            median = values[1] if len(values) == 4 else values[0]
            operations += sum(abs(median - value) for value in values)
    print(operations)
