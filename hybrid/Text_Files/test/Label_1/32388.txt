import java.util.Scanner;

public class Main {
  private static final long MOD = (long)1e9 + 7;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextLong();
    }

    // Sort the monsters in decreasing order of health.
    java.util.Arrays.sort(a);

    // The minimum possible final health of the last monster alive is the sum of the health of the two strongest monsters.
    long minFinalHealth = (a[n - 2] + a[n - 1]) % MOD;

    System.out.println(minFinalHealth);
  }
}
