#include <bits/stdc++.h>
using namespace std;
int a[31][31];
int c1[31][31], c2[31][31];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m;
        cin >> n >> m;
        memset(c1, 0, sizeof(c1));
        memset(c2, 0, sizeof(c2));
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) {
                cin >> a[i][j];
                if ((i + j) % 2 == a[i][j]) {
                    c1[i][j] = c1[i - 1][j] + c1[i][j - 1] - c1[i - 1][j - 1];
                    c2[i][j] = c2[i - 1][j] + c2[i][j - 1] - c2[i - 1][j - 1];
                } else {
                    c1[i][j] = min(c1[i - 1][j], c1[i][j - 1]) + 1;
                    c2[i][j] = min(c2[i - 1][j], c2[i][j - 1]) + 1;
                }
            }
        cout << min(c1[n][m], c2[n][m]) << '\n';
    }
    return 0;
}
