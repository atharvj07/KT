#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 10e5 + 5;
ll a[maxn];
int main() {
    ll n, sum = 0;
    cin >> n;
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
        sum += a[i];
    }
    sort(a, a + n);
    ll ans = LLONG_MAX;
    ans = min(ans, abs(a[n / 2] + (n - n / 2) - sum));
    ll pos = 0;
    for (ll i = n / 2 + 1; i < n; i++) {
        pos += (a[i] - a[i - 1]) * (n - i);
    }
    pos += a[n / 2] - a[0];
    ans = min(ans, pos);
    cout << ans << endl;
    return 0;
}
