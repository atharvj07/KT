#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        int a[n+1];
        map<int,int>mp;
        for(int i=1;i<=n;i++)
        {
            cin>>a[i];
            mp[a[i]]=i;
        }
        int p=1,f=0;
        for(int i=2;i<=n;i++)
        {
            if(mp[i]!=i)
            {
                if(p==1)
                {
                    cout<<"NO\n";
                    f=1;
                    break;
                }
            }
            else
            p=i;
        }
        if(f==0)
        cout<<"YES\n";
    }
}
