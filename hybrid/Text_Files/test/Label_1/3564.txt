#include <bits/stdc++.h>
using namespace std;

long long X, Y, x, y;
string s;

int main() {
  cin >> X >> Y >> s;
  for (char t : s) {
    if (t == 'U') y++;
    if (t == 'D') y--;
    if (t == 'L') x--;
    if (t == 'R') x++;
    if (x == X && y == Y) { cout << "Yes" << endl; return 0; }
  }
  cout << "No" << endl;
  return 0;
}
