from itertools import permutations

def calculate_happiness(order, g):
    happiness = 0
    # First pair before anyone showers
    happiness += g[order[0]-1][order[1]-1] + g[order[1]-1][order[0]-1]
    # Second pair before anyone showers
    if len(order) > 2:
        happiness += g[order[2]-1][order[3]-1] + g[order[3]-1][order[2]-1]
    # Pairs while the shower is being used
    for i in range(1, len(order)-1):
        happiness += g[order[i]-1][order[i+1]-1] + g[order[i+1]-1][order[i]-1]
        if i+2 < len(order):
            happiness += g[order[i+1]-1][order[i+2]-1] + g[order[i+2]-1][order[i+1]-1]
    return happiness

g = [list(map(int, input().split())) for _ in range(5)]

max_happiness = 0
for order in permutations(range(1, 6)):
    happiness = calculate_happiness(order, g)
    if happiness > max_happiness:
        max_happiness = happiness

print(max_happiness)
