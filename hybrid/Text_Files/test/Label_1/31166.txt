import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int[][] s = new int[m][];
        for (int i = 0; i < m; i++) {
            int size = scanner.nextInt();
            s[i] = new int[size];
            for (int j = 0; j < size; j++) {
                s[i][j] = scanner.nextInt() - 1;
            }
        }
        int[] sum = new int[m];
        for (int i = 0; i < m; i++) {
            for (int j : s[i]) {
                sum[i] += a[j];
            }
        }
        for (int i = 0; i < q; i++) {
            String query = scanner.next();
            if (query.equals("?")) {
                System.out.println(sum[scanner.nextInt() - 1]);
            } else {
                int k = scanner.nextInt() - 1;
                int x = scanner.nextInt();
                for (int j : s[k]) {
                    a[j] += x;
                }
                sum[k] += x * s[k].length;
            }
        }
    }
}
