def solve(n, m, s):
    best_r, best_c = 1, 1
    max_steps = 0
    for r in range(1, n + 1):
        for c in range(1, m + 1):
            steps = 0
            nr, nc = r, c
            for ch in s:
                if ch == 'L' and nc > 1:
                    nc -= 1
                    steps += 1
                elif ch == 'R' and nc < m:
                    nc += 1
                    steps += 1
                elif ch == 'U' and nr > 1:
                    nr -= 1
                    steps += 1
                elif ch == 'D' and nr < n:
                    nr += 1
                    steps += 1
                else:
                    break
            if steps > max_steps:
                max_steps = steps
                best_r, best_c = r, c
    return best_r, best_c


for _ in range(int(input())):
    n, m = map(int, input().split())
    s = input()
    r, c = solve(n, m, s)
    print(r, c)
