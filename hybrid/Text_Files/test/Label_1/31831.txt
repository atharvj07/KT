import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class TheseusAndMinotaur {

    static int n, m;
    static char[][] labyrinth;
    static int startX, startY;
    static int endX, endY;
    static boolean[][] visited;

    static final int[] dx = {0, 0, 1, -1};
    static final int[] dy = {1, -1, 0, 0};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        labyrinth = new char[n][m];
        scanner.nextLine();
        for (int i = 0; i < n; i++) {
            labyrinth[i] = scanner.nextLine().toCharArray();
        }
        startX = scanner.nextInt() - 1;
        startY = scanner.nextInt() - 1;
        endX = scanner.nextInt() - 1;
        endY = scanner.nextInt() - 1;

        visited = new boolean[n][m];
        int minMinutes = bfs();
        System.out.println(minMinutes == Integer.MAX_VALUE ? -1 : minMinutes);
    }

    static int bfs() {
        Queue<State> queue = new LinkedList<>();
        queue.offer(new State(startX, startY, 0));
        visited[startX][startY] = true;

        while (!queue.isEmpty()) {
            State state = queue.poll();
            if (state.x == endX && state.y == endY) {
                return state.minutes;
            }

            for (int i = 0; i < 4; i++) {
                int newX = state.x + dx[i];
                int newY = state.y + dy[i];
                if (isValidMove(newX, newY) && !visited[newX][newY]) {
                    visited[newX][newY] = true;
                    queue.offer(new State(newX, newY, state.minutes + 1));
                }
            }

            if (canRotate(state.x, state.y)) {
                for (int i = 0; i < 4; i++) {
                    int newX = state.x + dx[i];
                    int newY = state.y + dy[i];
                    if (isValidMove(newX, newY) && !visited[newX][newY]) {
                        visited[newX][newY] = true;
                        queue.offer(new State(newX, newY, state.minutes + 2));
                    }
                }
            }
        }

        return Integer.MAX_VALUE;
    }

    static boolean isValidMove(int x, int y) {
        return x >= 0 && x < n && y >= 0 && y < m && labyrinth[x][y] != '*';
    }

    static boolean canRotate(int x, int y) {
        return labyrinth[x][y] != '+' && labyrinth[x][y] != 'L' && labyrinth[x][y] != 'R' && labyrinth[x][y] != 'U' && labyrinth[x][y] != 'D';
    }

    static class State {
        int x;
        int y;
        int minutes;

        State(int x, int y, int minutes) {
            this.x = x;
            this.y = y;
            this.minutes = minutes;
        }
    }
}
