#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    cin>>n>>k;
    string s;
    cin>>s;
    map <char,int>p;
    for(int i=0;i<s.length();i++){
        p[s[i]]++;
    }
    set <char> openings;
    for(int i=0;i<s.length();i++){
        openings.insert(s[i]);
        p[s[i]]--;
        if(p[s[i]]==0) openings.erase(s[i]);
        if(openings.size()>k){
            cout<<"YES"<<endl;
            return 0;
        }
    }
    cout<<"NO"<<endl;
    return 0;
}
