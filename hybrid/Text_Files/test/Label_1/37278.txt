from math import gcd
from sys import stdin

def max_beauty(n, k, a):
    g = a[0]
    for i in range(1, n):
        g = gcd(g, a[i])
    min_value = min(a)
    max_possible_gcd = (min_value - 1) // k * k + g
    for i in range(max_possible_gcd, 0, -k):
        for j in a:
            if j % i > k:
                break
        else:
            return i
    return 1

n, k = map(int, stdin.readline().split())
a = list(map(int, stdin.readline().split()))
print(max_beauty(n, k, a))
