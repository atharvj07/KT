#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    vector<int> b2k(n);
    for (int &x : b2k)
        cin >> x;
    vector<int> b2k1(n);
    for (int i = 0; i < n; i++)
        b2k1[(i + b2k[i] % n) % n] = (i + b2k[i] % n) % n;
    for (int &x : b2k1)
        cout << x << " ";
}
