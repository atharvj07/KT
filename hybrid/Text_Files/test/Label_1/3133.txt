#include <cstdio>

const int M = 1000000007;
const int N = 1000007;

char s[N];

int main() {
  long long ans = 0;
  scanf("%s", s);
  for (int i = 0; s[i];) {
    if (s[i] == 'a') {
      ++i;
      if (s[i] == 'b') {
        ++i;
        ++ans;
      }
    } else {
      ++i;
    }
  }
  printf("%lld\n", ans);
  return 0;
}
