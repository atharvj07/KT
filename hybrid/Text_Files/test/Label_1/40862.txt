n = int(input())
a = [[0] * 100 for i in range(100)]
for i in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    for x in range(x1 - 1, x2):
        for y in range(y1 - 1, y2):
            a[x][y] += 1
ans = 0
for x in range(100):
    for y in range(100):
        ans += a[x][y]
print(ans)
