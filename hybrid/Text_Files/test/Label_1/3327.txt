#include <bits/stdc++.h>
using namespace std;

unordered_map<int,int> fup,fup2, fupc, fupc2;
vector<int> ua, ua2, ub, ub2;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m, i, x, y, a, b;
  bool c = true;
  cin >> n >> m;
  for (i = 1; i <= n; i++) {
    fup[i] = i;
    fup2[i] = i;
  }
  for (i = 0; i < m; i++) {
    cin >> x >> y, a = fup[x], b = fup[y];
    if (a != b)
      fup[a] = fup[b], ua.push_back(x), ub.push_back(y);
    else
      c = false;
    a = fup2[x], b = fup2[y];
    if (a != b)
      fup2[a] = fup2[b], ua2.push_back(x), ub2.push_back(y);
    else
      c = false;
  }
  if (c) {
    cout << "NO";
    return 0;
  }
  fupc[ua[0]] = 1, fupc[ub[0]] = 1;
  fupc2[ua2[0]] = 1, fupc2[ub2[0]] = 1;
  for (i = 1; i < (int)ua.size(); i++) {
    if (fupc[ua[i]] == 0) {
      if (fupc[ub[i]] == 1)
        fupc[ua[i]] = 2;
      else
        fupc[ua[i]] = 1;
    }
    if (fupc[ub[i]] == 0) {
      if (fupc[ua[i]] == 1)
        fupc[ub[i]] = 2;
      else
        fupc[ub[i]] = 1;
    }
    if (fupc2[ua2[i]] == 0) {
      if (fupc2[ub2[i]] == 1)
        fupc2[ua2[i]] = 2;
      else
        fupc2[ua2[i]] = 1;
    }
    if (fupc2[ub2[i]] == 0) {
      if (fupc2[ua2[i]] == 1)
        fupc2[ub2[i]] = 2;
      else
        fupc2[ub2[i]] = 1;
    }
  }
  cout << "YES" << endl;
  for (auto &it : fupc) cout << it.second << " ";
  cout << endl;
  for (auto &it : fupc2) cout << it.second << " ";
  return 0;
}
