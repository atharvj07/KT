import java.util.*;

public class Main {
    static class DSU {
        int[] parent, size;

        DSU(int n) {
            parent = new int[n + 1];
            size = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }

        int find(int x) {
            if (parent[x] == x) return x;
            return parent[x] = find(parent[x]);
        }

        void union(int a, int b) {
            int rootA = find(a);
            int rootB = find(b);
            if (rootA != rootB) {
                if (size[rootA] > size[rootB]) {
                    parent[rootB] = rootA;
                    size[rootA] += size[rootB];
                } else {
                    parent[rootA] = rootB;
                    size[rootB] += size[rootA];
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        DSU dsu = new DSU(n);
        int[] max = new int[m];
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            dsu.union(x, y);
            max[i] = 0;
            for (int j = 1; j <= n; j++) {
                if (dsu.find(j) == dsu.find(x)) {
                    max[i] = Math.max(max[i], dsu.size[dsu.find(j)]);
                }
            }
            if (max[i] < k) max[i] = 0;
        }
        for (int i : max) {
            System.out.println(i);
        }
    }
}
