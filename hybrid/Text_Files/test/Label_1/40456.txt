import sys
input = sys.stdin.readline

n=int(input())
a=[[] for i in range(n+1)]
for i in range(n-1):
    x,y,z=map(int,input().split())
    a[x].append((y,z))
    a[y].append((x,z))
x=[int(input()) for i in range(n)]
d=[-1]*(n+1)
v=[False]*(n+1)

def dfs(p,l,now):
    if v[now]:return
    d[now]=l
    v[now]=True
    for y,z in a[now]:
        if y!=p:
            dfs(now,max(l,z),y)

ans=0
for i in range(1,n+1):
    if not v[i]:dfs(i,0,i)
    ans=max(ans,min(d[i] for i in range(1,n+1) if d[i]!=-1 and x[i-1]>0))
    for j in range(1,n+1):
        if v[j]:x[j-1]-=1
    v=[False]*(n+1)
print(ans)
