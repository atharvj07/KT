import bisect
n,r=map(int,input().split())
a=list()
b=list()
for i in range(n):
    x,y=map(int,input().split())
    a.append(x);b.append(y)
p=[]
for i in range(n):
    if r>=a[i]:
        r+=b[i]
        p.append(i)
p.sort(key=lambda x:a[x])
ans=len(p)
for i in p:
    if r<a[i]:
        ans-=1
    else:r+=b[i]
print(ans)
