n = int(input().strip())
colors = list(map(int, input().strip().split()))

dp = [[0] * n for _ in range(n)]

for length in range(1, n + 1):
    for start in range(n - length + 1):
        end = start + length - 1
        if length == 1:
            dp[start][end] = 1
        elif length == 2:
            dp[start][end] = 1 if colors[start] == colors[end] else 2
        else:
            if colors[start] == colors[end]:
                dp[start][end] = dp[start + 1][end - 1] if start + 1 <= end - 1 else 1
            else:
                dp[start][end] = min(dp[start + 1][end], dp[start][end - 1]) + 1
            for mid in range(start, end):
                dp[start][end] = min(dp[start][end], dp[start][mid] + dp[mid + 1][end])

print(dp[0][n - 1])
