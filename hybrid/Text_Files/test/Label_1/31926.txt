import java.util.*;

public class Main {
    static int[] p = new int[901];
    static int[] sz = new int[901];
    static int n, m, k;
    static List<Integer>[] adj = new List[901];

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();

        for (int i = 1; i <= n; i++) {
            p[i] = i;
            sz[i] = 1;
        }

        for (int i = 0; i <= n; i++)
            adj[i] = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            adj[a].add(b);
            adj[b].add(a);
            join(a, b);
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int v: adj[i]) {
                if (find(v) == i) {
                    System.out.println(-1);
                    return;
                }
            }
        }

        for (int i = 1; i <= n; i++) {
            for (int v: adj[i]) {
                if (find(v) != i) {
                    ans++;
                    addEdge(i, v);
                }
            }
        }
        System.out.println(ans);
    }

    static int find(int x) {
        if (x == p[x])
            return x;
        return p[x] = find(p[x]);
    }

    static void join(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            if (sz[x] < sz[y]) {
                int tmp = x;
                x = y;
                y = tmp;
            }
            p[y] = x;
            sz[x] += sz[y];
        }
    }

    static void addEdge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            k++;
            System.out.println(x + " " + y);
        }
    }
}
