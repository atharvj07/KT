import sys

n = int(input())
g = [[] for i in range(n+1)]

for i in range(n-1):
    a, b = map(int, sys.stdin.readline().split())
    g[a].append(b)
    g[b].append(a)

cnt = [0] * (n+1)
def dfs(cur, p):
    cnt[cur] = 1
    for i in g[cur]:
        if i == p:
            continue
        dfs(i, cur)
        cnt[cur] += cnt[i]

dfs(1, 1)
if max(cnt)/n > 2. or sum(cnt) % 2 == 1:
    print("NO")
else:
    print("YES")
