import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        String line = in.next();
        char[] arr = line.toCharArray();

        int hCnt = 0;
        int tCnt = 0;
        for (char c : arr) {
            if (c == 'H') hCnt++;
            else tCnt++;
        }

        int ans = hCnt - 1 + tCnt - 1;
        System.out.println(ans);
    }
}
