#include <bits/stdc++.h>
using namespace std;
long long n, k, ans = 0, sumc = 0, cnt = 0, vec[200005], arr[200005];
vector<pair<int, int>> cards[200005];
bool cmp(pair<int, int> a, pair<int, int> b) { return a.second > b.second; }
int main() {
  cin >> n;
  for (int l = 1; l <= n; l++) {
    cin >> k;
    for (int i = 1; i <= k; i++) {
      int x, y;
      cin >> x >> y;
      cards[l].push_back({x, y});
    }
    sort(cards[l].begin(), cards[l].end(), cmp);
    sumc = 0;
    for (int i = 0; i < cards[l].size(); i++) {
      if (sumc + cards[l][i].first <= 3) {
        sumc += cards[l][i].first;
        vec[l] += cards[l][i].second;
        cnt++;
        if (cnt % 10 == 0) vec[l] += cards[l][i].second;
      }
    }
  }
  sort(vec + 1, vec + n + 1);
  for (int i = n; i >= 1; i--) ans += vec[i];
  cout << ans;
  return 0;
}
