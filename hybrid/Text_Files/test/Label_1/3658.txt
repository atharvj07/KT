#include<bits/stdc++.h>
using namespace std;
int32_t main()
{
	int n, k;
	cin >> n >> k;
	vector<int> a(n);
	for (int &x : a) cin >> x;
	sort(a.begin(), a.end());
	while (k--)
	{
		int mn = *lower_bound(a.begin(), a.end(), 1);
		if (!mn) break;
		cout << mn << '\n';
		for (int &x : a) if(x) x -= mn;
	}
}
