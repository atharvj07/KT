n = int(input())
arr = [[0] * 101 for i in range(101)]
for i in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    for k in range(y1, y2 + 1):
        for j in range(x1, x2 + 1):
            arr[k][j] += 1
ans = 0
for i in range(1, 101):
    for j in range(1, 101):
        ans += arr[i][j]
print(ans)
