import sys

t = int(input())

for tc in range(t):
    s = input()
    lst = []
    ans = 0

    for i in s:
        if i == '?': ans += 1
        else: lst.append(int(i))

    lst.sort(reverse=True)
    for i in lst:
        if ans == 0: break
        if i == 1: ans -= 1
        elif i == 0: ans -= 2
    
    print(ans + len(lst))
