import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long[] a = new long[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }

        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (long aa : a) {
            pq.add(aa);
        }

        long ans = 0;
        while (pq.size() > 1) {
            long x = pq.poll();
            long y = pq.poll();
            ans += x + y;
            pq.add(x + y);
        }

        System.out.println(ans);
    }
}
