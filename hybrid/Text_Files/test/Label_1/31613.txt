import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int n, t;
    static int[] memo = new int[21];
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        t = sc.nextInt();
        Arrays.fill(memo, -1);
        System.out.println(calc(1));
    }
    static int calc(int pos) {
        if(pos == n) return 1;
        int ret = memo[pos];
        if(ret != -1) return ret;
        ret = 0;
        for(int i = 1; i <= 4; i++)
            if(pos == 1 || i != memo[pos - 1])
                for(int j = 1; j <= 4; j++)
                    if(pos == 1 || j != i)
                        if(pos > 1)
                            if(i < j && j < memo[pos - 1]) continue;
                            else if(i > j && j > memo[pos - 1]) continue;
                        ret += calc(pos + 1);
        memo[pos] = ret;
        return ret;
    }
}
