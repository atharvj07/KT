#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MAXN = 3e5 + 5;
ll n, a[MAXN], l[MAXN], r[MAXN], ans[MAXN];
int main() {
  ll t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (ll i = 1; i <= n; i++) cin >> a[i];
    ll mx = 0;
    for (ll i = 1; i <= n; i++) {
      l[i] = 1;
      if (a[i - 1] < a[i]) l[i] = l[i - 1] + 1;
    }
    for (ll i = n; i >= 1; i--) {
      r[i] = 1;
      if (a[i + 1] < a[i]) r[i] = r[i + 1] + 1;
    }
    for (ll i = 1; i <= n; i++) { mx = max(mx, l[i] + r[i] - 1); }
    bool flag = false;
    for (ll i = 1; i <= n; i++) {
      if (l[i] + r[i] - 1 == mx) {
        if (flag) {
          flag = false;
          break;
        } else
          flag = true;
      }
    }
    if (flag)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
