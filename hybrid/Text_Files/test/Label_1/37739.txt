import sys

n, k = map(int, input().split())
s = input()
rooms = [int(x) for x in s]

# dp[i] contains the minimum cost of connecting the first i rooms
dp = [float('inf') for _ in range(n + 1)]

# Set the cost of connecting the first room to the minimum cost
dp[1] = 0

for i in range(2, n + 1):
    # Check if it is possible to place a router in the current room
    if rooms[i - 1] == 1:
        # Cost of placing a router in the current room
        cost = i
        
        # Calculate the cost of connecting all rooms within range of the router
        for j in range(max(1, i - k), min(n, i + k) + 1):
            cost += min(j, dp[j - 1])
        
        # Store the minimum cost of connecting the first i rooms
        dp[i] = min(dp[i], cost)
    
    # Otherwise, connect the current room directly
    dp[i] = min(dp[i], i)

# Output the minimum cost of connecting all n rooms
print(dp[n])
