#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int N = 2e5 + 7;

int n, k;
ll a[N];
bool vis[N];

ll cal_prefix(ll x) {
    ll cnt = 0;
    while (x > 0) {
        vis[x % k] = true;
        x /= k;
        cnt++;
    }
    return cnt;
}
ll cal_suffix(ll x) {
    ll cnt = 0;
    ll last = 0;
    while (x > 0) {
        last = x % k;
        if (vis[last]) {
            cnt += vis[last];
        }
        x /= k;
    }
    return cnt;
}
int main() {
    scanf(" %d %d", &n, &k);
    for (int i = 1; i <= n; i++) { scanf(" %lld", &a[i]); }
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        memset(vis, false, sizeof(vis));
        ll pref = cal_prefix(a[i]);
        memset(vis, false, sizeof(vis));
        ll suff = cal_suffix(a[i]);
        ans += (pref * suff);
    }
    printf("%lld\n", ans);
    return 0;
}
