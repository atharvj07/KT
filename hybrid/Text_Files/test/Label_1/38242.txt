import sys

t = int(sys.stdin.readline().strip())
for _ in range(t):
    n = int(sys.stdin.readline().strip())
    a = list(map(int, sys.stdin.readline().strip().split()))
    if sum(a) % 3 > 0:
        print("NO")
        continue
    print("YES")
    k = n // 3
    ops = []
    for i in range(n - 3):
        if i % 3 == (sum(a[i:i + 3]) % 3):
            ops.append(i + 1)
    print(k)
    print(*ops)
