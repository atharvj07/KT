from collections import deque
n=int(input())
a=deque()
b=deque()
for _ in range(n):
    x,y=map(int,input().split())
    a.append(x)
    b.append(y)
x=[0]*n
y=[0]*n
x[0]=a[0]
y[0]=b[0]
j=0
while j<n-1:
    if x[j]==x[j+1] or x[j]==x[j+1]-1:
        r=a[j+1]
        if x[j]+1==r:
            x[j+1]=r
            y[j+1]=0
            j+=1
        elif r-x[j+1]>=1:
            x[j+1]=r
            y[j+1]=0
            j+=1
        else:
            break
    elif x[j]==x[j+1]+1:
        l=b[j+1]
        if l-y[j+1]>=1:
            y[j+1]=l
            x[j+1]=r=a[j+1]+l
            j+=1
        else:
            break
    else:
        break

if x[j-1]==x[j] or x[j-1]==x[j]-1:
    pass
else:
    j=-1
if j==-1:
    print(-1)
else:
    print(sum(y))
    print(*x)
