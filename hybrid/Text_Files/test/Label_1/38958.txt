n = int(input())
data = {}
max_length = 0

for _ in range(n):
    temp = input().split()
    t, k = temp[0], int(temp[1])
    indexes = list(map(int, temp[2:]))
    for index in indexes:
        if index not in data or len(data[index]) < len(t):
            data[index] = t
    max_length = max(max_length, indexes[-1] + len(t) - 1)

result = [''] * max_length
for index, t in sorted(data.items()):
    for i, char in enumerate(t):
        if index + i - 1 < max_length and (result[index + i - 1] == '' or result[index + i - 1] > char):
            result[index + i - 1] = char

for i in range(max_length):
    if result[i] == '':
        result[i] = 'a'

print(''.join(result))
