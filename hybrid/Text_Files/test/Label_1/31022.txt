import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int k = scanner.nextInt();
        int t = scanner.nextInt();
        long[][] dp = new long[t + 1][2 * k + 1];
        dp[0][k] = 1;
        for (int i = 1; i <= t; i++) {
            for (int j = 0; j <= 2 * k; j++) {
                for (int l = -k; l <= k; l++) {
                    if (j + l < 0 || j + l > 2 * k) continue;
                    if (l == 0) continue;
                    if (i == t && j != k) continue;
                    if (l < 0) {
                        if (j + l > k) dp[i][j] = (dp[i][j] + dp[i - 1][j + l]) % 1000000007;
                    } else {
                        if (j + l < k) dp[i][j] = (dp[i][j] + dp[i - 1][j + l]) % 1000000007;
                    }
                }
            }
        }
        System.out.println(dp[t][k]);
    }
}
