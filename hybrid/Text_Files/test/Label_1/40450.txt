import sys
input = sys.stdin.readline
sys.setrecursionlimit(10**9)

def mex(a):
    b=[0]*1000001
    for x in a:
        b[x]=1
    for i in range(1000001):
        if b[i]==0:
            return i

q,x=map(int,input().split())
a=[]
for i in range(q):
    y=int(input())
    a.append(y)
    ans=mex(a)
    while ans-x in a:
        ans+=x
    print(ans)
