#include<bits/stdc++.h>
#define ll long long
#define mod 1000000007LL
using namespace std;
int n,dp[1010];
string s;
vector<int>a;
ll pw(ll b,ll p)
{
	if(p==0)return 1;
	if(p==1)return b;
	ll x=pw(b,p/2);
	x*=x;
	x%=mod;
	if(p%2==1)x*=b,x%=mod;
	return x;
}
int main()
{
	int i,j,k;
	cin>>n;
	cin>>s;
	a.resize(26,0);
	for(i=0;i<n;i++)
	{
		int x=s[i]-'a';
		a[x]++;
	}
	for(i=0;i<26;i++)
	{
		ll x=pw(2,a[i]);
		x-=1;
		x%=mod;
		if(x<0)x+=mod;
		dp[a[i]]=x;
	}
	ll ans=1;
	for(i=0;i<n;i++)
	{
		int x=s[i]-'a';
		ans*=dp[a[x]];
		ans%=mod;
	}
	ll mx=-1,mn=n;
	for(i=0;i<26;i++)
	{
		if(a[i]>0)
		{
			mx=max(mx,(ll)a[i]);
			mn=min(mn,(ll)a[i]);
		}
	}
	cout<<ans<<endl<<mx<<endl<<mn<<endl;
	return 0;
}
