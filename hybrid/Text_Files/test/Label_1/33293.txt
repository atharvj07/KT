import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String time = scanner.nextLine();
        String[] parts = time.split(":");
        int hours = Integer.parseInt(parts[0], 36);
        int minutes = Integer.parseInt(parts[1], 36);
        List<Integer> radixes = new ArrayList<>();
        for (int radix = 2; radix <= 36; radix++) {
            if (hours >= 0 && hours < radix && minutes >= 0 && minutes < radix) {
                radixes.add(radix);
            }
        }
        if (radixes.isEmpty()) {
            System.out.println(0);
        } else if (radixes.size() == 36) {
            System.out.println(-1);
        } else {
            Collections.sort(radixes);
            for (int radix : radixes) {
                System.out.print(radix + " ");
            }
        }
    }
}
