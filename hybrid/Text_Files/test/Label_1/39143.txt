n, m, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
k_queries = list(map(int, input().split()))

a.sort()
b.sort()

# Prefix sum arrays for a and b
a_prefix_sum = [0] * (n + 1)
b_prefix_sum = [0] * (m + 1)

for i in range(1, n + 1):
    a_prefix_sum[i] = a_prefix_sum[i - 1] + a[i - 1]

for i in range(1, m + 1):
    b_prefix_sum[i] = b_prefix_sum[i - 1] + b[i - 1]

for k in k_queries:
    max_total_cost = 0
    j = 0
    for price in a:
        while j < m and b[j] <= price + k:
            j += 1
        max_total_cost += b_prefix_sum[j]
    print(max_total_cost)
