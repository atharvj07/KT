import math
import sys
n = int(input())
cakes = [tuple(map(int,input().split())) for i in range(n)]
cakes.sort(key = lambda x : x[1], reverse=True)
dp = [-math.inf] * n

for i in range(n) :
    volume = math.pi * cakes[i][0] ** 2 * cakes[i][1]
    for j in range(i) :
        if cakes[j][1] < cakes[i][1] and cakes[j][0] < cakes[i][0] :
            volume = max(volume, dp[j] + volume)
    dp[i] = volume

print(max(dp))
