#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    int n; cin>>n;
    string s; cin>>s;
    vector<int> divs;
    for(int i=1;i*i<=n;i++){
        if(n%i==0){
            divs.push_back(i);
            if(i!=n) divs.push_back(n/i);
        }
    }
    reverse(divs.begin(),divs.end());
    for(int d:divs){
        for(int i=0;i<d/2;i++){
            swap(s[i],s[d-i-1]);
        }
    }
    cout<<s<<'\n';
    return 0;
}
