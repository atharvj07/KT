#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long int n,k,t,ans;
    cin >> n >> k >> t;
    if(t <= k)
        ans = t;
    else if(t > n)
        ans = k;
    else
    {
        long long int num_full_cycles = (t-1) / n * k;
        long long int remain = (t-1) % n;
        ans = num_full_cycles + min(remain,k);
    }
    cout << ans << endl;
    return 0;
}
