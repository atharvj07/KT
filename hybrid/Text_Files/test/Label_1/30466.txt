import java.util.*;

public class HosParty {

    static class Hos {
        int weight;
        int beauty;
        List<Integer> friends;

        Hos(int weight, int beauty) {
            this.weight = weight;
            this.beauty = beauty;
            this.friends = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int w = scanner.nextInt();

        Hos[] hoses = new Hos[n];
        for (int i = 0; i < n; i++) {
            hoses[i] = new Hos(scanner.nextInt(), scanner.nextInt());
        }

        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            hoses[x].friends.add(y);
            hoses[y].friends.add(x);
        }

        int maxBeauty = 0;
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                maxBeauty = Math.max(maxBeauty, dfs(hoses, i, visited, 0, w));
            }
        }

        System.out.println(maxBeauty);
    }

    static int dfs(Hos[] hoses, int u, boolean[] visited, int currentWeight, int remainingWeight) {
        visited[u] = true;
        int maxBeauty = hoses[u].beauty;

        for (int v : hoses[u].friends) {
            if (!visited[v] && currentWeight + hoses[v].weight <= remainingWeight) {
                maxBeauty = Math.max(maxBeauty, dfs(hoses, v, visited, currentWeight + hoses[v].weight, remainingWeight));
            }
        }

        return maxBeauty;
    }
}
