import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    private static char[][] grid;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = in.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = in.readLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = row.charAt(j);
            }
        }
        int minChanges = Integer.MAX_VALUE;
        char[][] bestGrid = null;
        for (char a = 'A'; a <= 'T'; a++) {
            for (char b = 'A'; b <= 'T'; b++) {
                for (char c = 'A'; c <= 'T'; c++) {
                    for (char d = 'A'; d <= 'T'; d++) {
                        if (a == b || a == c || a == d || b == c || b == d || c == d) continue;
                        char[][] newGrid = new char[n][m];
                        for (int i = 0; i < n; i++) {
                            for (int j = 0; j < m; j++) {
                                newGrid[i][j] = grid[i][j];
                            }
                        }
                        int changes = 0;
                        for (int i = 0; i < n; i++) {
                            for (int j = 0; j < m; j++) {
                                if (i % 2 == 0) {
                                    if (j % 2 == 0) {
                                        if (newGrid[i][j] != a) {
                                            newGrid[i][j] = a;
                                            changes++;
                                        }
                                    } else {
                                        if (newGrid[i][j] != b) {
                                            newGrid[i][j] = b;
                                            changes++;
                                        }
                                    }
                                } else {
                                    if (j % 2 == 0) {
                                        if (newGrid[i][j] != c) {
                                            newGrid[i][j] = c;
                                            changes++;
                                        }
                                    } else {
                                        if (newGrid[i][j] != d) {
                                            newGrid[i][j] = d;
                                            changes++;
                                        }
                                    }
                                }
                            }
                        }
                        if (isNice(newGrid) && changes < minChanges) {
                            minChanges = changes;
                            bestGrid = newGrid;
                        }
                    }
                }
            }
        }
        for (char[] row : bestGrid) {
            System.out.println(Arrays.toString(row).replaceAll("[\\[\\], ]", ""));
        }
    }

    private static boolean isNice(char[][] grid) {
        for (int i = 0; i < grid.length; i += 2) {
            for (int j = 0; j < grid[i].length; j += 2) {
                char[] chars = {grid[i][j], grid[i][j + 1], grid[i + 1][j], grid[i + 1][j + 1]};
                Arrays.sort(chars);
                if (!(chars[0] == 'A' && chars[1] == 'C' && chars[2] == 'G' && chars[3] == 'T')) {
                    return false;
                }
            }
        }
        return true;
    }
}
