import queue
n, m, d = map(int, input().split())
g = [[] for i in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)
p = set(map(int, input().split()))
q = queue.Queue()
q.put((1, 0))
ans = 0
vis = [False] * (n + 1)
while not q.empty():
    now, dis = q.get()
    if dis > d:
        break
    if not vis[now]:
        ans += 1
        vis[now] = True
        for i in g[now]:
            if not vis[i]:
                q.put((i, dis + 1))
print(ans)
