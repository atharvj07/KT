import sys

input = sys.stdin.readline

n = int(input())
a = []
for i in range(n):
    a.append(input().strip())
a.sort()

ans = "YES"
for i in range(1, n):
    if a[i][:len(a[i - 1])] != a[i - 1]:
        ans = "NO"
        break
    if len(a[i]) > len(a[i - 1]):
        for j in range(len(a[i - 1]), len(a[i])):
            if a[i][j] != '?':
                a[i - 1] += a[i][j]

for i in range(n):
    a[i] = a[i].replace('?', '0')

if ans == "YES":
    print(ans)
    for i in a:
        print(i)
else:
    print(ans)
