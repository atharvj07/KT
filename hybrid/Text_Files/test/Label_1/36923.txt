import sys
import math

n, t, k = map(int, input().split())
dist = list(map(int, input().split()))
dist[0] -= 1
if sum(dist) != n - 1:
    print(-1)
    sys.exit()
if t > n - 1 - k:
    print(-1)
    sys.exit()
if dist[0] <= k:
    print(-1)
    sys.exit()

Map = [0] * (n + 1)
for i in range(2, n + 1):
    if Map[i] == 0:
        roads = []
        for j in range(t):
            for _ in range(dist[j]):
                if dist[j] == 0:
                    continue
                dist[j] -= 1
                roads.append(j + 2)
        if len(roads) < 2:
            print(-1)
            sys.exit()
        Map[i] = Map[roads.pop()] = 1
        for road in roads:
            Map[road] = 1
            print(i, road)
            if k == 0:
                break

for i in range(2, n + 1):
    if Map[i] == 0:
        for j in range(i + 1, n + 1):
            if Map[j] == 0:
                print(i, j)
                Map[i] = Map[j] = 1
                break
