for _ in range(int(input())):
    n = int(input())
    a = []
    b = []
    for i in range(n):
        x, y = map(int, input().split())
        a.append(min(x, y))
        b.append(max(x, y))
    ans = [-1] * n
    for j in range(n - 1, -1, -1):
        for k in range(n):
            if a[k] < a[j] and b[k] < b[j]:
                ans[j] = k + 1
                break
    print(*ans)
