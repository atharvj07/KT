n, m, min_temp, max_temp = map(int, input().split())
temperatures = list(map(int, input().split()))

if min_temp not in temperatures and max_temp not in temperatures and n - m < 2:
    print("Incorrect")
elif min_temp not in temperatures and n - m < 1:
    print("Incorrect")
elif max_temp not in temperatures and n - m < 1:
    print("Incorrect")
elif min(temperatures) > min_temp or max(temperatures) < max_temp:
    print("Incorrect")
else:
    print("Correct")
