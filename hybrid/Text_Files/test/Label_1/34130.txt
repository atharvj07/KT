import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            String s = in.next();
            StringBuilder sb = new StringBuilder(s);
            boolean alice = true;
            for (int i = 0; i < s.length(); i++) {
                char min = s.charAt(i);
                int idx = i;
                for (int j = i + 1; j < s.length(); j++) {
                    if (s.charAt(j) <= min) {
                        min = s.charAt(j);
                        idx = j;
                    }
                }
                if (min != s.charAt(i)) {
                    sb.setCharAt(idx, s.charAt(i));
                    sb.setCharAt(i, min);
                    alice = !alice;
                }
            }
            if (alice) {
                for (int i = 0; i < s.length(); i++) {
                    if (sb.charAt(i) == s.charAt(i)) {
                        sb.setCharAt(i, (char) (s.charAt(i) + 1));
                        break;
                    }
                }
            }
            System.out.println(sb);
        }
    }
}
