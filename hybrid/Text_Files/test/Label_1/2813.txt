#include <iostream>
#include <string>
using namespace std;

int main() {
  long long n;
  int k;
  cin >> n >> k;
  string s = to_string(n);
  int l = s.size();
  int r = (l - 1) % k + 1;
  for (int i = l - 1; i >= 0; i--) {
    if (i % k == r - 1 && s[i] != '0') { r++; }
    if (i % k == r - 1 && s[i] == '0') {
      while (i > 0 && s[i - 1] == '0') { i--; }
      if (i == 0 && k == 1) { break; }
      if (s[i - 1] == '0') { continue; }
      r++;
    }
  }
  cout << r << endl;
  return 0;
}
