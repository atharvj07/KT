import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        long[][] dp = new long[N + 1][4];
        dp[0][3] = 1;
        for (int i = 1; i <= N; i++) {
            dp[i][0] = (dp[i - 1][1] + dp[i - 1][2] + dp[i - 1][3]) % 998244353;
            dp[i][1] = (dp[i - 1][0] + dp[i - 1][2]) % 998244353;
            dp[i][2] = (dp[i - 1][0] + dp[i - 1][1] + dp[i - 1][3]) % 998244353;
            dp[i][3] = (dp[i - 1][0] + dp[i - 1][1] + dp[i - 1][2]) % 998244353;
        }
        System.out.println(dp[N][3]);
    }
}
