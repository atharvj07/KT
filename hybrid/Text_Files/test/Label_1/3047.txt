#include <iostream>
using namespace std;
#define ll long long
int main() {
    ll l, r, ans = 1;
    cin >> l >> r;
    for (int i = 2; i * i <= r; ++i) {
        int cnt = 0;
        while (!(l % i)) cnt += (l /= i), l++;
        while (!(r % i)) cnt += (r /= i);
        cnt += (r / i - (l - 1) / i);
        if (cnt) ans = i;
    }
    cout << ans << endl;
}
