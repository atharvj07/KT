import java.util.*;

class Main {
    static double cross(double[] a, double[] b, double[] c) {
        return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double[][] p = new double[n][2];
        for (int i = 0; i < n; i++) {
            p[i][0] = sc.nextDouble();
            p[i][1] = sc.nextDouble();
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }

                for (int k = 0; k < n; k++) {
                    if (i == k || j == k) {
                        continue;
                    }

                    if (cross(p[i], p[j], p[k]) > 0) {
                        System.out.println((i + 1) + " " + (j + 1) + " " + (k + 1));
                        return;
                    }
                }
            }
        }
    }
}
