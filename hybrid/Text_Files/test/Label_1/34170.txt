import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int r = sc.nextInt();
        int[] a = new int[1 << n];
        for (int i = 0; i < (1 << n); i++) a[i] = sc.nextInt();
        double[][] dp = new double[r + 1][1 << n];
        Arrays.fill(dp[r], 1.0);
        for (int i = r - 1; i >= 0; i--) {
            int z = sc.nextInt();
            int g = sc.nextInt();
            for (int j = 0; j < (1 << n); j++) {
                if ((j >> (n - 1)) == z) {
                    dp[i][j] = g;
                    continue;
                }
                int x = j ^ (1 << (n - 1));
                dp[i][j] = (dp[i + 1][j] + dp[i + 1][x]) * 0.5;
            }
        }
        for (int i = 0; i < (1 << n); i++) {
            dp[0][i] = (dp[0][i] + a[i]) * 0.5;
        }
        for (int i = 0; i <= r; i++) {
            System.out.println(dp[i][0]);
        }
    }
}
