s=list(input())
L=[-1 for i in range(len(s))]
R=[-1 for i in range(len(s))]
mod=1000000007
dp=[0]*len(s)
stack=[]
for i in range(len(s)):
    if s[i]=='(':
        stack.append(i)
    else:
        if len(stack)>0:
            x=stack.pop()
            L[i]=x
            R[x]=i
for i in range(len(s)-1,-1,-1):
    if s[i]=='(':
        l=-1
        r=-1
        if L[i]!=-1:
            l=L[i]
        if r[i]!=-1:
            r=R[i]
        if l!=-1 and r!=-1 and s[l]=='(' and s[r]==')':
            dp[i]=(dp[r+1]+1)%mod
        else:
            dp[i]=0
    else:
        dp[i]=0
print(dp[0])
