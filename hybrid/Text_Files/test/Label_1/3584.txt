#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> l(n);
  vector<int> r(n);
  int idx = -1;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) idx = i;
    else {
      if (idx == -1) l[i] = -1;
      else l[i] = i - idx;
    }
  }
  idx = -1;
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] == 0) idx = i;
    else {
      if (idx == -1) r[i] = -1;
      else r[i] = idx - i;
    }
  }
  for (int i = 0; i < n; i++)
    cout << min(l[i], r[i]) << " ";
  cout << endl;
}
