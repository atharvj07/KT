import sys

n, m, k = map(int, input().split())
adj = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v, l = map(int, input().split())
    adj[u].append((v, l))
    adj[v].append((u, l))
if k == 0:
    print(-1)
    sys.exit()
dist = [-1] * (n + 1)
storages = list(map(int, input().split()))
q = [storages[0]]
dist[storages[0]] = 0
while q:
    v = q.pop(0)
    for u, l in adj[v]:
        if dist[u] == -1:
            dist[u] = dist[v] + l
            q.append(u)
ans = float('inf')
for v in storages:
    if dist[v] == -1:
        ans = -1
        break
    for u in range(1, n + 1):
        if v != u and dist[u] != -1:
            ans = min(ans, dist[u] + dist[v])
print(ans)
