import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int D = sc.nextInt();
        int[] superiors = new int[n];
        for (int i = 1; i < n; i++) {
            superiors[i] = sc.nextInt() - 1;
        }
        long[][] dp = new long[n][D + 1];
        dp[0][0] = 1;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j <= D; j++) {
                for (int k = 1; k <= D; k++) {
                    if (j == 0 || k > j) {
                        dp[i + 1][k] += dp[i][j];
                    }
                }
            }
            for (int j = 0; j <= D; j++) {
                int superior = superiors[i + 1];
                dp[superior][j] += dp[i][j];
            }
        }
        long ans = 0;
        for (int i = 0; i <= D; i++) {
            ans += dp[n - 1][i];
        }
        System.out.println(ans % (1000000007));
    }
}
