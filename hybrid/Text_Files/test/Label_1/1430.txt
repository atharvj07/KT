#include <bits/stdc++.h>
#define vi vector<int>
#define ll long long
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vi a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    if (n & 1) {
      cout << -1 << '\n';
      continue;
    }
    ll ans = 0, odd = 0;
    for (int i = 0; i < n; i++) {
      if (i & 1)
        odd += a[i] % 2;
      else
        ans += a[i] % 2;
    }
    ans += (odd + 1) / 2;
    cout << ans << '\n';
  }
  return 0;
}
