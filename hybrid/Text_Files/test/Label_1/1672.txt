#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main()
{
    int n;
    cin >> n;
    while (n--)
    {
        int l;
        cin >> l;
        vector<int> a(l);
        for (int &i : a)
            cin >> i;
        sort(a.begin(), a.end());
        for (int i = 0; i < 4; i++)
            cout << a[i] << " ";
        cout << endl;
    }
}
