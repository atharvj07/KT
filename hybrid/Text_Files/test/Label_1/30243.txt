import java.util.Scanner;

public class MaxOnes {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();

            long x = findMaxOnes(l, r);
            System.out.println(x);
        }
    }

    private static long findMaxOnes(long l, long r) {
        long maxOnes = 0;
        long maxOnesNumber = 0;

        for (long x = l; x <= r; x++) {
            long currentOnes = countOnes(x);
            if (currentOnes > maxOnes) {
                maxOnes = currentOnes;
                maxOnesNumber = x;
            }
        }

        return maxOnesNumber;
    }

    private static long countOnes(long x) {
        long count = 0;
        while (x > 0) {
            if ((x & 1) == 1) {
                count++;
            }
            x >>= 1;
        }
        return count;
    }
}
