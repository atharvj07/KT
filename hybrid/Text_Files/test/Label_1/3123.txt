#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll n, m;
char c;
string s;
vector<ll> vct;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> m;
  while (m--) {
    cin >> n >> c;
    cin >> s;
    vct.clear();
    ll g = 0, can = 1;
    for (ll i = 1; i <= n; i++) {
      if (g == 0) {
        g = __gcd(i, n);
        if (s[i - 1] != c) can = 0;
      }
      if (i % g) continue;
      if (s[i - 1] != c) vct.push_back(i);
    }
    cout << (can ? vct.size() : -1) << '\n';
    for (auto x : vct) cout << x << ' ';
    cout << '\n';
  }
  return 0;
}
