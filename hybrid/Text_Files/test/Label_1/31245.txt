import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            long m = 0;
            for (int j = 0; j < k; j++) {
                int b = sc.nextInt();
                m |= (1L << b);
            }
            masks[i] = m;
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            switch (op) {
                case 0:
                    System.out.println(((1L << m) & masks[0]) > 0 ? 1 : 0);
                    break;
                case 1:
                    masks[0] |= masks[m];
                    break;
                case 2:
                    masks[0] &= ~masks[m];
                    break;
                case 3:
                    masks[0] ^= masks[m];
                    break;
                case 4:
                    System.out.println((masks[0] & masks[m]) == masks[m] ? 1 : 0);
                    break;
                case 5:
                    System.out.println((masks[0] & masks[m]) > 0 ? 1 : 0);
                    break;
                case 6:
                    System.out.println((masks[0] & masks[m]) == 0 ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(masks[0] & masks[m]));
                    break;
                case 8:
                    System.out.println(Long.bitCount(masks[0] & masks[m]));
                    break;
            }
        }
    }
}
