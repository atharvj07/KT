#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        bool flag = false;
        for (int i = 1; i < n; i++) {
            if (s[i] < s[i - 1]) {
                flag = true;
                break;
            }
        }
        if (!flag) {
            cout << "YES" << endl;
            continue;
        }
        int cnt0 = 0, cnt1 = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '0')
                cnt0++;
            else
                cnt1++;
        }
        if (cnt0 == 0 || cnt1 == 0) {
            cout << "NO" << endl;
            continue;
        }
        cout << "YES" << endl;
    }
    return 0;
}
