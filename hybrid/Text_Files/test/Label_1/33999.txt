import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] heights = new int[n];
    for (int i = 0; i < n; i++) {
      heights[i] = sc.nextInt();
    }
    int count = 0;
    for (int i = 1; i <= 300000; i++) {
      int currCount = 0;
      for (int j = 0; j < n; j++) {
        if (heights[j] < i) {
          break;
        }
        if (heights[j] >= i * 2) {
          currCount += 2;
          heights[j] -= 2 * i;
        } else {
          currCount++;
          heights[j] -= i;
        }
      }
      count = Math.max(count, currCount);
    }
    System.out.println(count);
  }
}
