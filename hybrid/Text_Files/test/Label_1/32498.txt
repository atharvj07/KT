import java.util.*;
import java.lang.*;

class Main {
    static int MAXV = 101;
    static int MAXE = 101;
    static Map<Integer, Integer>[] adj = new HashMap[MAXV];
    static int[] color = new int[MAXE];
    static int[][] dis = new int[MAXV][MAXV];

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt(), M = sc.nextInt();
        for (int i = 1; i <= N; i++) {
            adj[i] = new HashMap<Integer, Integer>();
            for (int j = 1; j <= N; j++) {
                dis[i][j] = (int) 1e9;
            }
        }
        for (int i = 1; i <= M; i++) {
            int u = sc.nextInt(), v = sc.nextInt(), c = sc.nextInt();
            adj[u].put(v, c);
            adj[v].put(u, c);
            color[c] = 1;
        }

        for (int k = 1; k <= N; k++) {
            for (int i = 1; i <= N; i++) {
                for (int j = 1; j <= N; j++) {
                    if (i == j) {
                        dis[i][j] = 0;
                    } else if (adj[i].containsKey(j)) {
                        dis[i][j] = 1;
                    }
                }
            }
        }

        for (int k = 1; k <= N; k++) {
            for (int i = 1; i <= N; i++) {
                for (int j = 1; j <= N; j++) {
                    if (dis[i][j] == (int) 1e9) {
                        if (dis[i][k] != (int) 1e9 && dis[k][j] != (int) 1e9) {
                            dis[i][j] = dis[i][k] + dis[k][j];
                        }
                    }
                }
            }
        }

        int Q = sc.nextInt();
        while (Q-- > 0) {
            int u = sc.nextInt(), v = sc.nextInt();
            if (dis[u][v] == (int) 1e9) {
                System.out.println(0);
            } else {
                int ans = 0;
                for (int i = 1; i <= M; i++) {
                    if (ans == -1) break;
                    if (adj[u].containsKey(v)) {
                        ans = color[adj[u].get(v)];
                        break;
                    } else {
                        Set<Map.Entry<Integer, Integer>> entries = adj[u].entrySet();
                        for (Map.Entry<Integer, Integer> entry : entries) {
                            int a = entry.getKey(), b = entry.getValue();
                            if (dis[a][v] == dis[u][v] - 1 && dis[u][a] == 1) {
                                ans = b;
                                break;
                            }
                        }
                    }
                    if (adj[v].containsKey(u)) {
                        break;
                    } else {
                        Set<Map.Entry<Integer, Integer>> entries = adj[v].entrySet();
                        for (Map.Entry<Integer, Integer> entry : entries) {
                            int a = entry.getKey(), b = entry.getValue();
                            if (dis[a][u] == dis[v][u] - 1 && dis[v][a] == 1) {
                                ans = b;
                                break;
                            }
                        }
                    }
                }
                if (ans == 0) {
                    System.out.println(-1);
                } else {
                    System.out.println(ans);
                }
            }
        }
    }
}
