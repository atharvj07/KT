#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'

const ll MAX = 1e5 + 10;
ll a[MAX], op[MAX][3];

int main()
{
    ll n, m;
    cin>>n>>m;
    
    for(ll i = 1; i <= n; i++) cin>>a[i];
    
    for(ll i = 1; i <= m; i++)
    {
        ll x, y, z;
        cin>>x;
        op[i][0] = x;
        
        if(x == 1)
        {
            cin>>y>>z;
            op[i][1] = y;
            op[i][2] = z;
        }
        else if(x == 2)
        {
            cin>>y;
            op[i][1] = y;
        }
        else
        {
            cin>>y;
            op[i][1] = y;
        }
    }
    
    for(ll i = 1; i <= m; i++)
    {
        ll x = op[i][0], y = op[i][1], z = op[i][2];
        if(x == 1) a[y] = z;
        else if(x == 2) for(ll j = 1; j <= n; j++) a[j] += y;
        else cout<<a[y]<<endl;
    }
    
    return 0;
}
