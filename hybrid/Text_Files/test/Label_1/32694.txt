import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[n + 1];
        int[] y = new int[n + 1];
        for (int i = 0; i < n; i++) {
            x[i + 1] = scanner.nextInt();
            y[i + 1] = scanner.nextInt();
        }
        int[] mx = new int[101];
        int[] my = new int[101];
        for (int i = 0; i < n; i++) {
            mx[x[i + 1]] = 1;
            my[y[i + 1]] = 1;
        }
        int ansx = 0, ansy = 0;
        for (int i = 0; i <= 100; i++) {
            ansx += mx[i];
            ansy += my[i];
        }
        System.out.println(Math.min(ansx, ansy));
    }
}
