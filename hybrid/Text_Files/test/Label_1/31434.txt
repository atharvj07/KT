import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }

        int count = 0;
        for (int k = 1; k <= 100000; k++) {
            int arcCount = 0;
            int abcCount = 0;
            for (int i = 0; i < n; i++) {
                if (d[i] >= k) {
                    arcCount++;
                } else {
                    abcCount++;
                }
            }
            if (arcCount == abcCount) {
                count++;
            }
        }

        System.out.println(count);
    }
}
