#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    std::string a, b;
    std::cin >> a >> b;
    bool flip = false;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) {
        if (a[i] == '1') flip = !flip;
        ans++;
      }
    }
    if (!flip) std::cout << ans << std::endl;
    else std::cout << -1 << std::endl;
  }
  return 0;
}
