def check(deg, n):
    a = [0] * (n + 1)
    for i in deg:
        a[i] += 1
    for i in range(1, n + 1):
        if a[i] == 0 or a[i] > 2:
            return False
        if a[i] == 2:
            c = i
    for i in range(1, n + 1):
        if a[i] == 1:
            b = i
            break
    print("YES")
    for i in range(n - 1):
        if i == n - 2:
            print(b, c)
        else:
            print(c, i + 2)


def solve():
    n, d, k = map(int, input().split())
    if d % 2 != 0 or max(d, k) > 8 or (k == 2 and d > 3) or (d > k and d % 2 == 0):
        print("NO")
    elif d <= k:
        for i in range(d, k + 1):
            deg = []
            for j in range(1, n + 1):
                if i == 1 or i == j:
                    deg.append(1)
                else:
                    deg.append(2)
            check(deg, n)
    else:
        deg = []
        for i in range(1, int(n / 2) + 2):
            if i == 1:
                deg.append(1)
            else:
                deg.append(3)
        for i in range(int(n / 2) + 2, n + 1):
            deg.append(1)
        check(deg, n)


solve()
