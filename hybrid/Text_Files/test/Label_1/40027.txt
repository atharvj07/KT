mod = 10**9 +7
dp = [[0]*2**4 for _ in range(4)]
dp[0][0] = 1
for i in range(3):
    for j in range(2**4):
        dp[i+1][j | 1<<(3-i)] += dp[i][j]
        dp[i+1][j | 1<<(3-i)] %= mod
        dp[i+1][j] += dp[i][j]
        dp[i+1][j] %= mod
m = int(input())
s = [int(input()) for _ in range(m)]
for c in s:
    print(dp[3][c])
