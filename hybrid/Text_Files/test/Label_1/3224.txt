#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, b, a;
  cin >> n >> b >> a;
  vector<int> s(n);
  for (int i = 0; i < n; i++) cin >> s[i];
  int bat = b, acc = a, ans = 0;
  for (int i = 0; i < n; i++) {
    if (bat) {
      ans++;
      bat--;
      if (!s[i]) acc++;
    } else if (acc) {
      ans++;
      acc--;
    } else
      break;
  }
  cout << ans << endl;
}
