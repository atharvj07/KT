import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static final int INF = Integer.MAX_VALUE;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            int w = scanner.nextInt();
            edges.add(new Edge(u, v, w));
        }
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;
            System.out.println(getMinimumHardness(n, edges, l, r));
        }
    }

    private static int getMinimumHardness(int n, List<Edge> edges, int l, int r) {
        int[] dist = new int[n];
        for (int i = 0; i < n; i++) {
            dist[i] = INF;
        }
        dist[0] = 0;
        for (int i = 0; i < n - 1; i++) {
            for (Edge edge : edges) {
                if (edge.weight < 0) {
                    continue;
                }
                int u = edge.from;
                int v = edge.to;
                if (dist[u] + edge.weight < dist[v]) {
                    dist[v] = dist[u] + edge.weight;
                }
                if (dist[v] + edge.weight < dist[u]) {
                    dist[u] = dist[v] + edge.weight;
                }
            }
        }
        int hardness = -1;
        for (Edge edge : edges) {
            if (l <= edge.index && edge.index <= r) {
                continue;
            }
            if (hardness == -1 || hardness < dist[edge.from] + dist[edge.to] - edge.weight) {
                hardness = dist[edge.from] + dist[edge.to] - edge.weight;
            }
        }
        return hardness;
    }

    private static class Edge {
        int from;
        int to;
        int weight;
        int index;

        public Edge(int from, int to, int weight) {
            this.from = from;
            this.to = to;
            this.weight = weight;
        }
    }
}
