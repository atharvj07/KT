import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int q = sc.nextInt();
        int[] freq = new int[26];
        for (int i = 0; i < n; i++) {
            freq[s.charAt(i) - 'a']++;
        }
        int ans = 0;
        for (int i = 0; i < q; i++) {
            int m = sc.nextInt();
            char c = sc.next().charAt(0);
            ans = Math.max(ans, (freq[c - 'a'] + m) / (freq[c - 'a'] + m + n - freq[c - 'a'] - m) * n);
        }
        System.out.println(ans);
    }
}
