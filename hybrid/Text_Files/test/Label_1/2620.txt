#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long a, b;
  cin >> a >> b;
  long long ans1 = 0, ans2 = 0;
  vector<long long> v1, v2;
  for (long long i = 1; i * i <= a; i++) {
    if (a % i == 0) {
      v1.push_back(i);
      if (i * i != a) v1.push_back(a / i);
    }
  }
  for (long long i = 1; i * i <= b; i++) {
    if (b % i == 0) {
      v2.push_back(i);
      if (i * i != b) v2.push_back(b / i);
    }
  }
  for (int i = 0; i < v1.size(); i++) {
    for (int j = 0; j < v2.size(); j++) {
      if (v1[i] + v2[j] <= a + b) {
        if (ans1 + ans2 < v1[i] + v2[j]) {
          ans1 = v1[i];
          ans2 = v2[j];
        }
      }
    }
  }
  cout << ans1 << "\n";
  for (long long i = 1; i <= ans1; i++) cout << i * i << " ";
  cout << "\n" << ans2 << "\n";
  for (long long i = 1; i <= ans2; i++) cout << i * i << " ";
  return 0;
}
