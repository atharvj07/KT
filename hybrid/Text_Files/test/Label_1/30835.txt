import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        // Read the input
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] l = new int[n];
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }

        // Find the minimum required number of additional chairs
        int minAdditionalChairs = 0;
        for (int i = 0; i < n; i++) {
            if (l[i] > r[i]) {
                minAdditionalChairs++;
            }
        }

        // Print the minimum required number of additional chairs
        System.out.println(minAdditionalChairs);
    }
}
