import java.util.*;

class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = sc.nextInt();
    }

    int[] dp = new int[n];
    Arrays.fill(dp, 1);

    for (int i = n - 2; i >= 0; i--) {
      for (int j = i + 1; j < n; j++) {
        if (arr[j] % arr[i] == 0) {
          dp[i] = Math.max(dp[i], dp[j] + 1);
        }
      }
    }

    int res = 0;
    for (int i = 0; i < n; i++) {
      res = Math.max(res, dp[i]);
    }

    System.out.println(res);
  }
}
