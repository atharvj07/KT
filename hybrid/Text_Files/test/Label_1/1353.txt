#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n), res(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int inc = 1, dec = 1e9, indinc = -1, inddec = -1;
  for (int i = 0; i < n; i++) {
    if (inc < a[i]) {
      indinc = i;
      inc = a[i];
    } else if (dec > a[i]) {
      inddec = i;
      dec = a[i];
    } else {
      cout << "NO" << endl;
      return 0;
    }
  }
  if (indinc > inddec) {
    cout << "NO" << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (i <= inddec)
      res[i] = 1;
    else
      res[i] = 0;
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) { cout << res[i] << " "; }
  cout << endl;
  return 0;
}
