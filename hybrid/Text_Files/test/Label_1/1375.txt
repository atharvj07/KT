#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N = (1 << 19);
int n, a[N], dp[N][20], pw[20];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    for (int j = 0; j < 19; j++) {
      if ((1LL << j) & a[i]) {
        dp[i][j] = 1;
      }
    }
  }
  pw[0] = 1;
  for (int i = 1; i < 19; i++) { pw[i] = pw[i - 1] << 1; }
  for (int j = 0; j < 19; j++) {
    for (int i = 0; i < n; i++) {
      dp[i][j + 1] = dp[i][j] | dp[dp[i][j]][j];
    }
  }
  int ans = n + 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < 19; j++) {
      if ((dp[i][j] & dp[dp[i][j]][j]) != 0 && (dp[i][j] & dp[dp[i][j]][j]) != pw[j]) {
        ans = min(ans, j + 3);
      }
    }
  }
  if (ans > n)
    ans = -1;
  cout << ans << endl;
}
