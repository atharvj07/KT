import java.util.*;

public class Main {
    public static final int MAX = 100000;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int q = in.nextInt();
        int[] a = new int[MAX + 1];
        for (int i = 0; i < q; i++) {
            int l = in.nextInt();
            int r = in.nextInt();
            int x = in.nextInt();
            a[l] += x;
            a[r + 1] -= x;
        }

        int[] cnt = new int[MAX + 1];
        for (int i = 1; i <= MAX; i++) {
            cnt[i] = cnt[i - 1] + a[i];
        }

        Set<Integer> set = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            set.add(cnt[i]);
        }

        List<Integer> list = new ArrayList<>(set);
        Collections.sort(list);
        System.out.println(list.size());
        for (int x : list) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
