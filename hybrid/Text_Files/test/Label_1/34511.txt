import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] b = new int[n];
        int[] i = new int[n];
        int[] r = new int[n];
        for (int j = 0; j < b.length; j++)
            b[j] = sc.nextInt();
        for (int j = 0; j < i.length; j++)
            i[j] = sc.nextInt();
        for (int j = 0; j < r.length; j++)
            r[j] = sc.nextInt();
        int sum = 0;
        for (int j = 0; j < n; j++)
            for (int k = 0; k < n; k++)
                if (b[k] > b[j] && i[k] > i[j] && r[k] > r[j]) {
                    sum++;
                    break;
                }
        System.out.println(sum);
    }
}
