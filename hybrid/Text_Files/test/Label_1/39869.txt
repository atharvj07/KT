import sys
input = sys.stdin.readline

def main():
    t = int(input())
    for _ in range(t):
        n, a, b = map(int, input().split())
        if a >= n - 1 or b >= n - 1 or a + b > n - 2:
            print(-1)
            continue
        ans = [1]
        for i in range(2, n + 1):
            if a > 0 and b > 0:
                if ans[-1] > i:
                    ans.append(i)
                    b -= 1
                else:
                    ans.insert(-1, i)
                    a -= 1
            elif a > 0:
                ans.append(i)
                a -= 1
            else:
                ans.insert(-1, i)
                b -= 1
        print(*ans)

main()
