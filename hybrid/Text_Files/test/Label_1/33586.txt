import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] array = new int[ n - 1 ];
        for (int i = 0; i < n - 1; i++) {
            array[ i ] = input.nextInt();
        }
        int a = input.nextInt();
        int b = input.nextInt();
        int years = 0;
        for (int i = a - 1; i < b - 1; i++) {
            years += array[ i ];
        }
        System.out.println(years);
    }
}
