n, m = map(int, input().split())
exams = []
for _ in range(m):
    s, d, c = map(int, input().split())
    exams.append((s, d, c, _ + 1))
exams.sort(key=lambda x: x[1])

schedule = [0] * (n + 1)
exam_days = {d: i for _, d, _, i in exams}

for s, d, c, i in exams:
    preparation_days = c
    for day in range(s, d):
        if schedule[day] == 0 and preparation_days > 0:
            schedule[day] = i
            preparation_days -= 1
    if preparation_days > 0:
        print(-1)
        break
    else:
        schedule[d] = m + 1
else:
    print(*schedule[1:])
