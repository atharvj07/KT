import java.util.Scanner;

public class CircleColoring {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        String s = scanner.next();

        long[][] dp = new long[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dp[i][j] = -1;
            }
        }

        long[][][] prefixSum = new long[2][n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (s.charAt(j - 1) == 'R') {
                    prefixSum[0][i][j] = prefixSum[0][i][j - 1] + 1;
                    prefixSum[1][i][j] = prefixSum[1][i][j - 1];
                } else {
                    prefixSum[0][i][j] = prefixSum[0][i][j - 1];
                    prefixSum[1][i][j] = prefixSum[1][i][j - 1] + 1;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            dp[i][0] = 1;
        }

        for (int j = 1; j < m; j++) {
            for (int i = 0; i < n; i++) {
                int prev = (i - 1 + n) % n;
                int next = (i + 1) % n;
                long count = 0;

                if (s.charAt(j - 1) == 'R') {
                    count += (prefixSum[1][n][j] - prefixSum[1][i][j]) % MOD;
                    count += (prefixSum[0][prev][j] - prefixSum[0][0][j]) % MOD;
                    count += (prefixSum[0][next][j] - prefixSum[0][i][j]) % MOD;
                } else {
                    count += (prefixSum[0][n][j] - prefixSum[0][i][j]) % MOD;
                    count += (prefixSum[1][prev][j] - prefixSum[1][0][j]) % MOD;
                    count += (prefixSum[1][next][j] - prefixSum[1][i][j]) % MOD;
                }

                dp[i][j] = count % MOD;
            }
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans = (ans + dp[i][m - 1]) % MOD;
        }

        System.out.println(ans);
    }
}
