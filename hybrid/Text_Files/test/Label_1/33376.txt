import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
        }
        int ans = x;
        for (int i = 0; i < n; i++) {
            ans = Math.min(ans, x / m[i]);
        }
        System.out.println(ans);
    }
}
