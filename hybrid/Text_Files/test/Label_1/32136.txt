import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            boolean valid = true;
            int p = 0, c = 0;
            for (int i = 0; i < n; i++) {
                int np = in.nextInt();
                int nc = in.nextInt();
                if (p > np || c > nc) {
                    valid = false;
                }
                p = np;
                c = nc;
            }
            System.out.println(valid ? "YES" : "NO");
        }
    }
}
