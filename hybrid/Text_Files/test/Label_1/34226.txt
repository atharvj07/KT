import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
	 
    static int[] arr;
    static long[] p;
    static int n, k, a, b, q;

    static void fillPrefix(int start, int end, int val) {
        for (int i = start; i < end; i++) {
            p[i] += val;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        n = Integer.parseInt(s[0]);
        k = Integer.parseInt(s[1]);
        a = Integer.parseInt(s[2]);
        b = Integer.parseInt(s[3]);
        q = Integer.parseInt(s[4]);
        arr = new int[n];
        p = new long[n + 1];
        List<Integer> queries = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            s = br.readLine().split(" ");
            if (s[0].equals("1")) {
                int day = Integer.parseInt(s[1]) - 1;
                int num = Integer.parseInt(s[2]);
                arr[day] += num;
                fillPrefix(day, day + 1, num);
            } else {
                queries.add(Integer.parseInt(s[1]) - 1);
            }
        }
        for (int i = 1; i < n; i++) {
            p[i] += p[i - 1];
        }
        for (int day : queries) {
            long ord = p[day] + Math.min((long) arr[day], b);
            if (day + k < n) {
                ord += p[day + k] - p[day];
            }
            ord += (long) a * (n - day - k);
            System.out.println(ord);
        }
    }
}
