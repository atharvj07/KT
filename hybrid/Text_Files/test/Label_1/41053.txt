MOD = 998244353

def solve(n, k, h):
    count = 0
    for i in range(n):
        if h[i] != h[i - 1]:
            count += 1
    power = pow(k - 2, n - count, MOD) if n - count else 1
    return count * power * pow(k, MOD - 2, MOD) % MOD

n, k = map(int, input().split())
h = list(map(int, input().split()))
print(solve(n, k, h))
