import sys

n, k = map(int, input().split())
q = list(map(int, input().split()))
s = list(map(int, input().split()))

if q == s:
    print('NO')
else:
    if k % 2 == 1:
        if q == s[::-1]:
            print('YES')
        else:
            print('NO')
    else:
        if q == s:
            print('YES')
        else:
            print('NO')
