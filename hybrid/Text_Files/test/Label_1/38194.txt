n = int(input())
cards = list(map(int, input().split()))

min_card = min(cards)
total_picks = 0
while cards:
    top_card = cards.pop(0)
    total_picks += 1
    if top_card == min_card:
        if cards:
            min_card = min(cards)
    else:
        cards.append(top_card)

print(total_picks)
