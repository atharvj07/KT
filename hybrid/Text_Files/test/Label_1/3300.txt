#include <iostream>
#include <string>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::string s;
    std::cin >> n >> s;
    bool found = false;
    for (int i = 0; i < n - 3; i++) {
      if (s[i] == '2' && s[i + 1] == '0' && s[i + 2] == '2' && s[i + 3] == '0') {
        found = true;
        break;
      }
    }
    if (found || s == "2020")
      std::cout << "YES\n";
    else
      std::cout << "NO\n";
  }
  return 0;
}
