#include <iostream>
#include <vector>
 
using namespace std;
 
const int MAX_N = 1e5 + 5;
 
int n, a[MAX_N], dp[MAX_N];
 
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      dp[i] = 1;
    }
    int ans = 1;
    for (int i = 2; i <= n; i++) {
      for (int j = 1; j < i; j++) {
        if (a[i] > a[j]) {
          dp[i] = max(dp[i], dp[j] + 1);
        }
      }
      ans = max(ans, dp[i]);
    }
    ans = max(ans, dp[1] * n);
    cout << ans << endl;
  }
  return 0;
}
