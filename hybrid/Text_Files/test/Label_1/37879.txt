import sys

n, a, b = map(int, input().split())
if n % 2 == 0 or a*b < n:
    print(-1)
    sys.exit()

if n % 2:
    n += 1
    for i in range(1, n, 2):
        for j in range(1, b+1):
            if i<=a:
                print(i, end = " ")
            else:
                print(0, end = " ")
        print()
    n -= 1
    for i in range(2, n, 2):
        for j in range(1, b+1):
            if i<=a:
                print(i, end = " ")
            else:
                print(0, end = " ")
        print()
else:
    for i in range(1, n, 2):
        for j in range(1, b+1):
            if i<=a:
                print(i, end = " ")
            elif j == 1:
                print(n, end = " ")
            else:
                print(0, end = " ")
        print()
    for i in range(2, n, 2):
        for j in range(1, b+1):
            if i<=a:
                print(i, end = " ")
            else:
                print(0, end = " ")
        print()
