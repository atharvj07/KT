n = int(input())
t = ['', 'one', 'two', 'three', 'four', 'five', 'six', 'seven', 'eight', 'nine']
t1 = ['ten', 'eleven', 'twelve', 'thirteen', 'fourteen', 'fifteen', 'sixteen', 'seventeen', 'eighteen', 'nineteen']
t2 = ['', '', 'twenty', 'thirty', 'forty', 'fifty', 'sixty', 'seventy', 'eighty', 'ninety']
if 0 <= n < 10: print(t[n]);
if 10 <= n < 20: print(t1[n - 10]);
if 20 <= n < 100: print(t2[n // 10] + '-' + t[n % 10]);
if n == 100: print('one-hundred');
