#include<bits/stdc++.h>
using namespace std;
#define ll long long

const int N=2005;
ll a[N], b[N];
ll ans, as, bs, n, m;

void solve(){
  cin>>n;
  for(int i=0; i<n; i++) cin>>a[i];
  cin>>m;
  for(int i=0; i<m; i++) cin>>b[i];
  as=accumulate(a, a+n, 0ll);
  bs=accumulate(b, b+m, 0ll);
  ans=as<bs? bs-as: as-bs;
  int num=0;
  for(int i=0; i<n; i++){
    for(int j=0; j<m; j++){
      ll sum=as-a[i]+b[j];
      if(as<bs && sum>=as && sum<bs && ans>sum-bs){
        ans=sum-bs;
        num=1;
        cout<<1<<'\n';
        cout<<i+1<<' '<<j+1<<'\n';
      }
      else if(as>bs && sum>bs && sum<=as && ans>as-sum){
        ans=as-sum;
        num=1;
        cout<<1<<'\n';
        cout<<i+1<<' '<<j+1<<'\n';
      }
    }
  }
  for(int i=0; i<n; i++){
    for(int j=0; j<m; j++){
      ll sum=as-a[i]+b[j];
      for(int k=0; k<n; k++){
        if(i==k) continue;
        for(int l=0; l<m; l++){
          if(j==l) continue;
          ll sum2=sum-b[j]+a[k];
          if(as<bs && sum2>=bs && ans>sum2-bs){
            ans=sum2-bs;
            num=2;
            cout<<2<<'\n';
            cout<<i+1<<' '<<j+1<<'\n';
            cout<<k+1<<' '<<l+1<<'\n';
          }
          else if(as>bs && sum2<=bs && ans>bs-sum2){
            ans=bs-sum2;
            num=2;
            cout<<2<<'\n';
            cout<<i+1<<' '<<j+1<<'\n';
            cout<<k+1<<' '<<l+1<<'\n';
          }
        }
      }
    }
  }
  if(num==0){
    cout<<0<<'\n';
    cout<<0<<'\n';
  }
}

int main(){
  solve();
  return 0;
}
