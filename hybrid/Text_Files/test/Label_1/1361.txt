#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const ll MAXP = 1e6 + 3;
const ll MAXN = 1e12;
vector<ll> phi(MAXP), pr(MAXP);
ll p[MAXP];

void sieve() {
  for (ll i = 1; i < MAXP; i++) phi[i] = i;
  for (ll i = 2; i < MAXP; i++) {
    if (phi[i] == i) {
      p[++p[0]] = i;
      pr[i] = (i - 1) * i;
    }
    for (ll j = 1; j <= p[0]; j++) {
      ll d = p[j];
      if (i * d > MAXP - 1) break;
      if (i % d == 0) {
        phi[i * d] = phi[i] * d;
      } else {
        phi[i * d] = phi[i] * (d - 1);
      }
      if (i % d == 0) break;
    }
  }
}

ll qpow(ll a, ll x, ll p) {
  ll res = 1;
  while (x) {
    if (x & 1) res = res * a % p;
    a = a * a % p;
    x >>= 1;
  }
  return res;
}

ll solve(ll a, ll b, ll p, ll x) {
  ll Ans = 0;
  for (ll i = 1; i * i <= x; i++) {
    if (x % i == 0) {
      ll l = i, r = x / i;
      if (qpow(a, l, p) % p == b % p && l <= phi[p]) Ans++;
      if (r != l && qpow(a, r, p) % p == b % p && r <= phi[p]) Ans++;
    }
  }
  return Ans;
}

int main() {
  sieve();
  ll a, b, p, x;
  cin >> a >> b >> p >> x;
  cout << solve(a, b, p, x);
  return 0;
}
