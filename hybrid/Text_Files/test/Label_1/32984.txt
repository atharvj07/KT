import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] a = new int[5];
        for (int i = 0; i < 5; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 0; i < 4; i++) {
            ans = (ans + a[i] + 9) / 10 * 10;
        }
        ans += a[4];
        System.out.println(ans);
    }
}
