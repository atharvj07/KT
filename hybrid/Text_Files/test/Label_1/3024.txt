#include <iostream>
#include <vector>
#include <set>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<pair<int, int>, int>> points(n);
  for (int i = 0; i < n; i++) {
    int x, y, z;
    cin >> x >> y >> z;
    points[i] = make_pair(make_pair(x, y), z);
  }
  set<int> not_removed;
  for (int i = 0; i < n; i++) {
    not_removed.insert(i);
  }
  vector<pair<int, int>> ans;
  while (not_removed.size() > 0) {
    // Find two points that can form a perfectly balanced pair.
    int a = -1, b = -1;
    for (auto it1 = not_removed.begin(); it1 != not_removed.end(); it1++) {
      int i = *it1;
      for (auto it2 = it1; it2 != not_removed.end(); it2++) {
        int j = *it2;
        if (i == j) continue;
        bool valid = true;
        for (auto k : not_removed) {
          if (k == i || k == j) continue;
          int x = points[i].first.first;
          int y = points[i].first.second;
          int z = points[i].second;
          int x2 = points[j].first.first;
          int y2 = points[j].first.second;
          int z2 = points[j].second;
          int x3 = points[k].first.first;
          int y3 = points[k].first.second;
          int z3 = points[k].second;
          if (x <= x3 && x3 <= x2 && y <= y3 && y3 <= y2 && z <= z3 && z3 <= z2) {
            valid = false;
            break;
          }
          if (x <= x3 && x3 <= x2 && y2 <= y3 && y3 <= y && z2 <= z3 && z3 <= z) {
            valid = false;
            break;
          }
          if (x2 <= x3 && x3 <= x && y <= y3 && y3 <= y2 && z <= z3 && z3 <= z2) {
            valid = false;
            break;
          }
          if (x2 <= x3 && x3 <= x && y2 <= y3 && y3 <= y && z2 <= z3 && z3 <= z) {
            valid = false;
            break;
          }
          if (x <= x3 && x3 <= x2 && y <= y3 && y3 <= y2 && z2 <= z3 && z3 <= z) {
            valid = false;
            break;
          }
          if (x <= x3 && x3 <= x2 && y2 <= y3 && y3 <= y && z <= z3 && z3 <= z2) {
            valid = false;
            break;
          }
        }
        if (valid) {
          a = i;
          b = j;
          break;
        }
      }
      if (a != -1) break;
    }
    if (a == -1 || b == -1) {
      cout << -1 << endl;
      return 0;
    }
    ans.push_back(make_pair(a + 1, b + 1));
    not_removed.erase(a);
    not_removed.erase(b);
  }
  for (auto p : ans) {
    cout << p.first << " " << p.second << endl;
  }
  return 0;
}
