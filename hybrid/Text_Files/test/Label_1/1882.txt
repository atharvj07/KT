#include<iostream>
#include<cmath>
using namespace std;
int gcd(int a, int b)
{
    if(!b)
        return a;
    return gcd(b, a % b);
}
int main()
{
    long long int t, w, b;
    cin >> t >> w >> b;
    int p = t / b;
    int q = t / w + 1;
    int f = gcd(p, q);
    p /= f, q /= f;
    cout << p << "/" << q << endl;
}
