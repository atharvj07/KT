#include <bits/stdc++.h>
#define all(x) (x).begin(), (x).end()
using namespace std;

int n, m;
vector<int> adj[200005];
vector<int> directed;

bool dfs(int u, int p) {
    for (int v : adj[u]) {
        if (v == p) continue;
        if (directed[v - 1] == -1) {
            directed[v - 1] = (directed[u - 1] == 0 ? 1 : 0);
        } else if (directed[v - 1] == directed[u - 1]) {
            return false;
        }
        if (!dfs(v, u)) return false;
    }
    return true;
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    directed.resize(m, -1);
    bool valid = dfs(1, 1);
    if (!valid)
        cout << "NO" << endl;
    else {
        cout << "YES" << endl;
        for (int d : directed) cout << d;
        cout << endl;
    }
    return 0;
}
