import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int i, count = 0;
        for (i = 0; i < n - 1; i++) {
            if (s.charAt(i) == 'H' && s.charAt(i + 1) == 'T') {
                count++;
            }
        }
        if (s.charAt(n - 1) == 'H' && s.charAt(0) == 'T') {
            count++;
        }
        System.out.println(count);
    }
}
