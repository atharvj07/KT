#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<bool> vis(1001, false);
    vector<int> b;
    for (int i = n - 1; i >= 0; i--) {
        if (!vis[a[i]]) {
            vis[a[i]] = true;
            b.push_back(a[i]);
        }
    }
    cout << b.size() << endl;
    for (int i = b.size() - 1; i >= 0; i--) { cout << b[i] << " "; }
    cout << endl;
    return 0;
}
