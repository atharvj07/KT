import java.util.Scanner;

public class ValerieAndCrucians {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        int[] rows = new int[m];
        int[] viabilities = new int[n];

        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt() - 1;
            teeth[i][1] = scanner.nextInt();
        }

        for (int i = 0; i < n; i++) {
            rows[teeth[i][0]]++;
        }

        for (int i = 0; i < n; i++) {
            viabilities[i] = teeth[i][1];
        }

        int totalCrucians = 0;
        while (k > 0) {
            int minViability = Integer.MAX_VALUE;
            int minRow = -1;
            for (int i = 0; i < n; i++) {
                if (viabilities[i] > 0 && teeth[i][0] != minRow) {
                    if (viabilities[i] < minViability) {
                        minViability = viabilities[i];
                        minRow = teeth[i][0];
                    }
                }
            }

            if (minViability == Integer.MAX_VALUE) {
                break;
            }

            int crucians = Math.min(k, rows[minRow] * minViability);
            totalCrucians += crucians;
            k -= crucians;

            for (int i = 0; i < n; i++) {
                if (teeth[i][0] == minRow) {
                    viabilities[i] -= minViability;
                }
            }
        }

        System.out.println(totalCrucians);
    }
}
