t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()
    t_count, m_count = 0, 0
    possible = True
    for char in s:
        if char == 'T':
            t_count += 1
        elif char == 'M':
            m_count += 1
        if m_count > t_count:
            possible = False
            break
    if t_count != 2 * m_count:
        possible = False
    t_count, m_count = 0, 0
    for char in reversed(s):
        if char == 'T':
            t_count += 1
        elif char == 'M':
            m_count += 1
        if m_count > t_count:
            possible = False
            break
    print("YES" if possible else "NO")
