n, a, b = map(int, input().split())
s = input()
i = 0
cnt = 0
while i < n and a > 0 and b > 0:
    if s[i] == '.':
        if (i == 0 or s[i - 1] == '*') and (i == n - 1 or s[i + 1] == '*'):
            if cnt % 2 == 0:
                a -= 1
                cnt += 1
            else:
                b -= 1
                cnt += 1
    i += 1
print(cnt)
