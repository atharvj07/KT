#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> b(n);
  for (int i = 0; i < n; i++) cin >> b[i];
  vector<int> ans;
  for (int a = 1; !ans.empty() || a <= b[0]; a++) {
    int sum = a;
    while (sum) sum -= sum % 10, sum /= 10, a++;
    ans.push_back(a - 1);
    if (ans.size() == n) break;
    vector<int> tmp2;
    for (int i = ans.size() - 1; i >= 0; i--) {
      int sum = ans[i];
      while (sum) sum -= sum % 10, sum /= 10, ans[i]++;
      if (ans[i] > b[i]) ans.erase(ans.begin() + i);
      else
        break;
    }
  }
  for (int x : ans) cout << x << endl;
}
