#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int n = s.size();
    long double ans = 0;
    for (int i = 0; i < n; ++i) {
        if (s[i] == 'I' || s[i] == 'E' || s[i] == 'A' || s[i] == 'O' || s[i] == 'U' || s[i] == 'Y')
            ans += n - i;
    }
    cout << fixed << setprecision(8) << ans << '\n';
    return 0;
}
