import java.util.*;

public class Main {
    static int N, M;
    static int[] D;
    static int[] U, V, C;
    static boolean[] visited;
    static boolean[] color;
    static boolean impossible;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        D = new int[N + 1];
        U = new int[M + 1];
        V = new int[M + 1];
        C = new int[M + 1];
        visited = new boolean[N + 1];
        color = new boolean[N + 1];
        impossible = false;
        for (int i = 1; i <= N; i++) {
            D[i] = sc.nextInt();
        }
        for (int i = 1; i <= M; i++) {
            U[i] = sc.nextInt();
            V[i] = sc.nextInt();
        }
        solve(1);
        if (impossible) {
            System.out.println(-1);
            return;
        }
        StringBuilder sb = new StringBuilder(N);
        for (int i = 1; i <= N; i++) {
            sb.append(color[i] ? 'B' : 'W');
        }
        System.out.println(sb);
        for (int i = 1; i <= M; i++) {
            System.out.println(C[i]);
        }
    }

    static void solve(int v) {
        visited[v] = true;
        for (int i = 1; i <= M; i++) {
            if (!visited[U[i]] && U[i] == v) {
                solve(U[i]);
                if (impossible) {
                    return;
                }
                color[v] = !color[U[i]];
                C[i] = D[v];
            } else if (!visited[V[i]] && V[i] == v) {
                solve(V[i]);
                if (impossible) {
                    return;
                }
                color[v] = !color[V[i]];
                C[i] = D[v];
            }
        }
        if (color[v]) {
            for (int i = 1; i <= M; i++) {
                if (visited[U[i]] && U[i] == v) {
                    if (C[i] != D[v]) {
                        impossible = true;
                        return;
                    }
                } else if (visited[V[i]] && V[i] == v) {
                    if (C[i] != D[v]) {
                        impossible = true;
                        return;
                    }
                }
            }
        } else {
            for (int i = 1; i <= M; i++) {
                if (visited[U[i]] && U[i] == v) {
                    if (C[i] != D[U[i]]) {
                        impossible = true;
                        return;
                    }
                } else if (visited[V[i]] && V[i] == v) {
                    if (C[i] != D[V[i]]) {
                        impossible = true;
                        return;
                    }
                }
            }
        }
    }
}
