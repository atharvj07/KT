import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Map<String, Integer> suit = new HashMap<>();
        Map<String, Integer> value = new HashMap<>();
        String[] suits = {"S", "D", "H", "C"};
        String[] values = {"2", "3", "4", "5", "6", "7", "8", "9", "T", "J", "Q", "K", "A"};
        for (int i = 0; i < n; i++) {
            String card = scanner.next();
            String s = card.substring(0, 1);
            String v = card.substring(1, 2);
            suit.put(s, suit.getOrDefault(s, 0) + 1);
            value.put(v, value.getOrDefault(v, 0) + 1);
        }
        for (String s : suits) {
            if (suit.getOrDefault(s, 0) > 1) {
                System.out.println("YES");
                return;
            }
        }
        for (String v : values) {
            if (value.getOrDefault(v, 0) > 1) {
                System.out.println("YES");
                return;
            }
        }
        if (n % 2 == 0) {
            System.out.println("YES");
            return;
        } else {
            System.out.println("NO");
            return;
        }
    }
}
