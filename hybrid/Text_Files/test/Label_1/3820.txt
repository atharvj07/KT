#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> v(n);
        for (int i = 0; i < n; i++) { cin >> v[i]; }
        long long ans = 0;
        for (long long i = 0; i < n; i++) {
            for (long long j = i + 1; j < n; j++) {
                if ((v[i] & v[j]) >= (v[i] ^ v[j])) { ans++; }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
