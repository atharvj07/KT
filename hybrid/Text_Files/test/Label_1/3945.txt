#include<bits/stdc++.h>
using namespace std;
int main() {
     string s;
     long long x, y, cnt1 = 0, cnt0 = 0;
     cin >> s >> x >> y;
     for (char c : s) {
          if (c == '1') cnt1++;
          else if (c == '0') cnt0++;
     }
     long long ans = cnt0 * x + cnt1 * y;
     for (int i = 0; i < s.size(); i++) {
          if (s[i] == '?') {
               if (x < y) ans -= x, cnt0++;
               else ans -= y, cnt1++;
          }
     }
     cout << ans;
     return 0;
}
