#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> a(1 << (n + 1));
  for (int i = 2; i < (1 << (n + 1)); i++) cin >> a[i];

  int sum = 0;
  for (int i = (1 << n); i < (1 << (n + 1)); i++) sum += a[i];

  int ans = 0;
  for (int i = (1 << n) - 1; i >= 1; i--) {
    ans += max(0, sum - a[i] * (1 << n));
    sum -= a[i];
    sum *= 2;
  }

  cout << ans << endl;

  return 0;
}
