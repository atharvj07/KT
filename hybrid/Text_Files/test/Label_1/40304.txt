import sys

n = int(sys.stdin.readline())
if n == 1:
    print("No")
else:
    print("Yes")
    if n % 2 == 0:
        print("1", 2)
        print(n - 1, end=" ")
        for i in range(3, n + 1):
            print(i, end=" ")
    else:
        print("2", 1, 2)
        print(n - 2, end=" ")
        for i in range(3, n + 1):
            print(i, end=" ")
