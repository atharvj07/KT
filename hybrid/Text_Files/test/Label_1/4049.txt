#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, m, a[200005], b[200005], psum[200005], l = 0;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        psum[i + 1] = psum[i] + a[i];
    }
    for (int i = 0; i < m; i++) {
        cin >> b[i];
        l = lower_bound(psum + 1, psum + n + 1, b[i]) - psum;
        cout << l << " " << b[i] - psum[l - 1] << endl;
    }
}
