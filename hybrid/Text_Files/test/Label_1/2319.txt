#include <bits/stdc++.h>

using namespace std;

const int MAXN = 200 + 10;

int n, a, b;
int h[MAXN];

int dp[MAXN][2][2];

int solve(int idx, bool painted_red, bool painted_green) {
  if (idx > n) {
    return 0;
  }

  int &ret = dp[idx][painted_red][painted_green];
  if (ret != -1) return ret;

  int ans = INT_MAX;

  if (idx > 1 && painted_red && painted_green) {
    ans = min(ans, solve(idx + 1, painted_red, painted_green) + h[idx] - 1);
  }

  if (painted_red && a >= h[idx]) {
    ans = min(ans, solve(idx + 1, painted_red, true) + h[idx] - 1);
    a -= h[idx];
  }

  if (painted_green && b >= h[idx]) {
    ans = min(ans, solve(idx + 1, true, painted_green) + h[idx] - 1);
    b -= h[idx];
  }

  return ret = ans;
}

int main() {
  memset(dp, -1, sizeof dp);

  cin >> n >> a >> b;
  for (int i = 1; i <= n; i++) cin >> h[i];

  int ans = solve(1, false, false);

  if (ans == INT_MAX) ans = -1;

  cout << ans << "\n";

  return 0;
}
