import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int k = in.nextInt();
        boolean[] blocked = new boolean[n];
        for (int i = 0; i < m; i++) {
            blocked[in.nextInt()] = true;
        }
        long[] costs = new long[k];
        for (int i = 0; i < k; i++) {
            costs[i] = in.nextInt();
        }
        long[] dp = new long[n];
        Arrays.fill(dp, Long.MAX_VALUE);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            if (!blocked[i]) {
                for (int j = 0; j < k; j++) {
                    if (i + j < n) {
                        dp[i + j] = Math.min(dp[i + j], dp[i] + costs[j]);
                    }
                }
            }
        }
        System.out.println(dp[n - 1] == Long.MAX_VALUE ? -1 : dp[n - 1]);
    }
}
