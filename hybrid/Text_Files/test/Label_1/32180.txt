import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        if (a[n - 1] < n) {
            b[n - 1] = n;
            for (int i = n - 2; i >= 0; i--) {
                int mex = 0;
                while (mex <= a[i] || mex < b[i + 1]) {
                    mex++;
                }
                b[i] = mex;
            }
            for (int i = 0; i < n; i++) {
                System.out.print(b[i] + " ");
            }
        } else {
            System.out.println("-1");
        }
    }
}
