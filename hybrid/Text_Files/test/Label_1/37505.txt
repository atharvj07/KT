import sys
input = sys.stdin.readline

n, m = map(int, input().split())
w = list(map(int, input().split()))
vis = [False] * (n+1)
a = []
for _ in range(m):
    x, y = map(int, input().split())
    a.append((x,y))

ff = True
for x,y in a:
    if w[x-1] == 0 and w[y-1] == 0:
        ff = False

for x,y in a:
    if w[x-1] > 0 or w[y-1] > 0:
        vis[x] = True
        vis[y] = True
        w[x-1] -= 1
        w[y-1] -= 1
        break

for x,y in a:
    if vis[x] and vis[y]:
        continue
    elif w[x-1] > 0 and w[y-1] == 0:
        vis[x] = True
        w[x-1] -= 1
    elif w[y-1] > 0 and w[x-1] == 0:
        vis[y] = True
        w[y-1] -= 1
    else:
        ff = False
if not vis.count(True) == m:
    ff = False

if ff:
    print("ALIVE")
    for x,y in a:
        if vis[x] and vis[y]:
            print(x, end=" ")
            continue
        elif w[x-1] > 0:
            print(x, end=" ")
            w[x-1] -= 1
        elif w[y-1] > 0:
            print(y, end=" ")
            w[y-1] -= 1
else:
    print("DEAD")
