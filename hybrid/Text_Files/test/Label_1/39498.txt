import sys
input = sys.stdin.readline
sys.setrecursionlimit(10**6)

n, m, c_l, c_e, v = map(int, input().split())
l = list(map(int, input().split()))
e = list(map(int, input().split()))
q = int(input())
from collections import deque
def dijkstra():
    d = [[10**18] * (m+1) for i in range(n+1)]
    vis = [[False] * (m+1) for i in range(n+1)]
    q = deque([(1, 1, 0)])
    vis[1][1] = True
    d[1][1] = 0
    while q:
        x, y, dist = q.popleft()
        for v, b, w in [(0,0,x),(1,0,x)]:
            if 1 <= v + x <= n and not vis[v+x][y]:
                q.append((v+x, y, dist+w))
                vis[v+x][y] = True
                d[v+x][y] = dist+w
        if y > 1 and not vis[x][y-1]:
            q.append((x, y-1, dist+1))
            vis[x][y-1] = True
            d[x][y-1] = dist+1
        if y < m and not vis[x][y+1]:
            q.append((x, y+1, dist+1))
            vis[x][y+1] = True
            d[x][y+1] = dist+1
    return d

d = dijkstra()
for i in range(q):
    x1, y1, x2, y2 = map(int, input().split())
    print(d[x2][y2]-d[x1][y1])
