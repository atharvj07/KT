#include <iostream>

int main() {
  std::string s;
  std::cin >> s;

  int n = s.size(), ans = 0;
  for (int i = 0; i < n; i++) {
    for (int j = n; j > i; j--) {
      bool palindrome = true;
      for (int k = i; k < (i + j) / 2; k++) {
        if (s[k] != s[i + j - k - 1]) {
          palindrome = false;
          break;
        }
      }
      if (!palindrome) {
        ans = std::max(ans, j - i);
      }
    }
  }
  std::cout << ans << '\n';
  return 0;
}
