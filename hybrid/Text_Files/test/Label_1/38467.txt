from collections import deque
n, q = map(int, input().split())
dq = deque(map(int, input().split()))
for _ in range(q):
    x = int(input())
    for _ in range(x - 1):
        A, B = dq.popleft(), dq.popleft()
        if A > B:
            dq.appendleft(B)
            dq.append(A)
        else:
            dq.appendleft(A)
            dq.append(B)
    print(*dq[:2])
