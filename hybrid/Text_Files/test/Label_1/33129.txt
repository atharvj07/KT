import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String[] start = in.nextLine().split(":");
        String[] end = in.nextLine().split(":");
        int h1 = Integer.parseInt(start[0]),
            m1 = Integer.parseInt(start[1]),
            h2 = Integer.parseInt(end[0]),
            m2 = Integer.parseInt(end[1]),
            h3 = (h1 + h2) / 2,
            m3 = (m1 + m2) / 2;
        if((m1 + m2) % 2 == 1) {
            m3++;
            if(m3 > 59) {
                m3 -= 60;
                h3++;
            }
        }
        System.out.printf("%02d:%02d", h3, m3);
    }
}
