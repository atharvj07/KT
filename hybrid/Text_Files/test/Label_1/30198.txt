import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();

        while (t-- > 0) {
            int n = in.nextInt();
            long k = in.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }

            long max = Long.MIN_VALUE;
            for (long x : a) {
                max = Math.max(max, x);
            }

            for (int i = 0; i < n; i++) {
                a[i] = max - a[i];
            }

            for (long x : a) {
                System.out.print(x + " ");
            }
            System.out.println();
        }
    }
}
