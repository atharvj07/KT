import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
	Scanner s = new Scanner(System.in);
	int N = s.nextInt();
	int[] A = new int[N];
	for (int i = 0; i < N; i++) {
	    A[i] = s.nextInt();
	}
	s.close();

	int cnt = 0;
	int min = 1;
	for (int i = 0; i < N; i++) {
	    if (A[i] != min) {
		cnt++;
	    } else {
		min++;
	    }
	}
	System.out.println(cnt);
    }
}
