#include <bits/stdc++.h>
#define rep(i, a, b) for (int i = (a), tmp = (b); i <= tmp; ++i)
#define per(i, a, b) for (int i = (a), tmp = (b); i >= tmp; --i)
using namespace std;
typedef long long ll;
typedef long double ld;
#define INF ((int)1e9)
#define N ((int)1e5 + 100)
#define MOD ((int)1e9 + 7)
char s[N];
int n, x, y, dp[N];
int main() {
    scanf("%s%d%d", s + 1, &x, &y);
    n = strlen(s + 1);
    memset(dp, INF, sizeof(dp));
    dp[0] = dp[1] = 0;
    rep(i, 2, n) {
        if (s[i] == '0')
            dp[i] = min({dp[i - 1], dp[i - 2] + x});
        else if (s[i] == '1')
            dp[i] = min({dp[i - 1], dp[i - 2] + y});
        else {
            dp[i] = min({dp[i - 1], dp[i - 2] + x, dp[i - 2] + y});
        }
    }
    cout << dp[n] << endl;
    return 0;
}
