for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    i = j = 0
    ans = []
    while i < n and j < m:
        if a[i] == b[j]:
            ans.append(a[i])
            i += 1
            j += 1
        elif a[i] < b[j]:
            i += 1
        else:
            j += 1
    if ans:
        print("YES")
        print(len(ans))
        print(*ans)
    else:
        print("NO")
