from collections import defaultdict
from math import gcd

n = int(input())
g = [int(x) for x in input().split()]
a = [0] * n

for i in range(0, n):
    for j in range(i, n):
        if i == j:
            a[i] = g[i]
        else:
            a[i] = a[j] * g[i + j + 1] // g[i] // g[j + 1]

print(*a)
