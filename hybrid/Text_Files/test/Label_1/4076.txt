#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k,i;
    cin>>n>>k;
    char a[n+1];
    cin>>a;
    if(a[0]=='#'||a[n-1]=='#'){
        cout<<"NO";
        return 0;
    }
    for(i=0;i<n-1;){
        while(a[i+1]=='#'&&i<n-1)
            i++;
        if(i==n-1) break;
        i=min(n,i+k);
        if(i==n-1) break;
    }
    if(i==n-1) cout<<"YES";
    else cout<<"NO";
}
