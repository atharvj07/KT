#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int k = 0, n1 = 0, n2 = 0, n3 = 0, t1 = 0, t2 = 0, t3 = 0;
    vector<int> machineCount = {0, 0, 0};
    vector<int> timeTaken = {0, 0, 0};
    int startTime = 0;
    
    cin >> k >> n1 >> n2 >> n3 >> t1 >> t2 >> t3;
    
    machineCount[0] = n1;
    machineCount[1] = n2;
    machineCount[2] = n3;
    
    timeTaken[0] = t1;
    timeTaken[1] = t2;
    timeTaken[2] = t3;
    
    for(int i = 0 ; i < k; i++) {
        for(int j = 0; j < 3; j++) {
            if(machineCount[j] > 0) {
                machineCount[j]--;
                startTime = max(startTime, timeTaken[j] + startTime);
                break;
            }
        }
    }
    
    cout << startTime << endl;
    return 0;
}
