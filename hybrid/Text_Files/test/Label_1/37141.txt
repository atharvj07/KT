n=int(input())
s=input()
z=s.count('0')
o=s.count('1')
t=s.count('2')
p=min(z,o,t)
z-=p
o-=p
t-=p
ans=[]
for i in range(n):
    if s[i]=='0' and z:
        ans.append('1')
        z-=1
    elif s[i]=='1' and o:
        ans.append('2')
        o-=1
    elif s[i]=='2' and t:
        ans.append('0')
        t-=1
    else:
        ans.append(s[i])
print(''.join(ans))
