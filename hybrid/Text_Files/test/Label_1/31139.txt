import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long[] masks = new long[10];
        int n, q;
        n = input.nextInt();
        for (int i = 0; i < n; i++) {
            int k = input.nextInt();
            long mask = 0;
            for (int j = 0; j < k; j++) {
                long b = input.nextInt();
                mask |= 1 << b;
            }
            masks[i] = mask;
        }
        q = input.nextInt();
        for (int i = 0; i < q; i++) {
            int opcode = input.nextInt();
            long mask = input.nextInt();
            switch (opcode) {
                case 0:
                    System.out.println(((masks[0] >> mask) & 1));
                    break;
                case 1:
                    masks[0] |= mask;
                    break;
                case 2:
                    masks[0] &= ~mask;
                    break;
                case 3:
                    masks[0] ^= mask;
                    break;
                case 4:
                    System.out.println(((masks[0] & mask) == mask) ? 1 : 0);
                    break;
                case 5:
                    System.out.println(((masks[0] & mask) != 0) ? 1 : 0);
                    break;
                case 6:
                    System.out.println(((masks[0] & mask) == 0) ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(masks[0] & mask));
                    break;
                case 8:
                    System.out.println(masks[0] & mask);
                    break;
            }
        }
    }
}
