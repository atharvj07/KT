def is_good_lab(grid):
    n = len(grid)
    for i in range(n):
        for j in range(n):
            if grid[i][j] != 1:
                found = False
                for k in range(n):
                    if grid[i][k] + grid[k][j] == grid[i][j]:
                        found = True
                        break
                if not found:
                    return False
    return True

n = int(input())
grid = []
for i in range(n):
    grid.append(list(map(int, input().split())))
if is_good_lab(grid):
    print('Yes')
else:
    print('No')
