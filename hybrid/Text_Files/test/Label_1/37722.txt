class node:
    def __init__(self, s, e, i):
        self.s, self.e = s, e
        self.i = i
        self.left = self.right = None

    def insert(self, s, e, i):
        if s >= self.e:
            if not self.right:
                self.right = node(s, e, i)
            else:
                self.right.insert(s, e, i)
        elif e <= self.s:
            if not self.left:
                self.left = node(s, e, i)
            else:
                self.left.insert(s, e, i)

    def path(self, a, b):
        if a >= self.s and b <= self.e:
            return True
        elif a >= self.e or b <= self.s:
            return False
        if a < self.s:
            return self.right.path(a, b)
        else:
            return self.left.path(a, b)

n = int(input())
root = None
for i in range(1, n + 1):
    q = list(map(int, input().split()))
    if q[0] == 1:
        t = node(q[1], q[2], i)
        if not root:
            root = t
        else:
            root.insert(q[1], q[2], i)
    else:
        print(['NO', 'YES'][root.path(q[1], q[2])])
