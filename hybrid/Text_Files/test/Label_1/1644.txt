#include <cstdio>
#include <algorithm>
using namespace std;

const int MAXN = 2e5 + 5;

typedef long long ll;

struct card { ll c, d; };

card a[MAXN];

int main() {
  int n, k, cnt = 0;
  ll ans = 0;
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &k);
    for (int j = 1; j <= k; j++) {
      scanf("%lld %lld", &a[++cnt].c, &a[cnt].d);
    }
    sort(a + 1, a + cnt + 1, [](const card &x, const card &y) -> bool { return x.c < y.c; });
    ll now = a[1].c * 3, sum = 0;
    for (int j = 1; j <= cnt; j++) {
      if (now >= a[j].c) {
        now -= a[j].c;
        sum += a[j].d;
        if (j % 10 == 0) sum += a[j].d;
      } else
        break;
    }
    ans += sum;
    cnt = 0;
  }
  printf("%lld\n", ans);
  return 0;
}
