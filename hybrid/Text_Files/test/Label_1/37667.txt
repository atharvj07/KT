n, A = map(int, input().split())
colors = list(map(int, input().split()))

cnt_A = 0
cnt_colors = {}

for color in colors:
    if color == A:
        cnt_A += 1
    else:
        if color in cnt_colors:
            cnt_colors[color] += 1
        else:
            cnt_colors[color] = 1
        if cnt_colors[color] < cnt_A:
            cnt_colors[color] = -1

potential_colors = [color for color, count in cnt_colors.items() if count >= cnt_A]
if not potential_colors:
    print(-1)
else:
    for color in potential_colors:
        if colors.count(color) >= cnt_A:
            print(color)
            break
