#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    a[i]--;
  }
  string s;
  for (int i = 0; i < n - 1; i++) {
    if (a[i] + 1 < a[i + 1])
      s += "-";
    else
      s += "+";
  }
  s += "+";
  cout << s << endl;
  return 0;
}
