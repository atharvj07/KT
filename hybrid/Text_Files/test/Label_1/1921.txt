#include <iostream>
#include <vector>
using namespace std;
int main() {
    int x, y, a, b;
    cin >> x >> y >> a >> b;
    vector<pair<int, int>> v;
    for (int i = a; i <= x; i++) {
        for (int j = b; j <= y; j++) {
            if (i > j) v.push_back(make_pair(i, j));
        }
    }
    cout << v.size() << endl;
    for (auto c : v) {
        cout << c.first << " " << c.second << endl;
    }
}
