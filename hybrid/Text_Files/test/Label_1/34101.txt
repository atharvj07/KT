import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[][] grid = new char[4][4];
        for (int i = 0; i < 4; i++) {
            grid[i] = in.nextLine().toCharArray();
        }

        boolean win = false;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (grid[i][j] == '.') {
                    char[][] newGrid = new char[4][4];
                    for (int k = 0; k < 4; k++) {
                        for (int l = 0; l < 4; l++) {
                            newGrid[k][l] = grid[k][l];
                        }
                    }
                    newGrid[i][j] = 'x';
                    if (checkWin(newGrid)) {
                        win = true;
                        break;
                    }
                }
            }
            if (win) {
                break;
            }
        }

        System.out.println(win ? "YES" : "NO");
    }

    private static boolean checkWin(char[][] grid) {
        for (int i = 0; i < 4; i++) {
            if (grid[i][0] == grid[i][1] && grid[i][1] == grid[i][2] && grid[i][2] == grid[i][3] && grid[i][0] != '.') {
                return true;
            }
            if (grid[0][i] == grid[1][i] && grid[1][i] == grid[2][i] && grid[2][i] == grid[3][i] && grid[0][i] != '.') {
                return true;
            }
        }
        if (grid[0][0] == grid[1][1] && grid[1][1] == grid[2][2] && grid[2][2] == grid[3][3] && grid[0][0] != '.') {
            return true;
        }
        if (grid[0][3] == grid[1][2] && grid[1][2] == grid[2][1] && grid[2][1] == grid[3][0] && grid[0][3] != '.') {
            return true;
        }
        return false;
    }
}
