#include <bits/stdc++.h>
using namespace std;
map<tuple<int, int, int>, int> vis;
int n, k, init[5][51], des[5][51], dist[5][51], step[5][51];
int dx[4] = {-1, 0, 1, 0}, dy[4] = {0, 1, 0, -1};
int bfs(int r, int c) {
    queue<tuple<int, int, int>> q;
    q.emplace(r, c, 0);
    while (!q.empty()) {
        auto [r, c, s] = q.front();
        q.pop();
        if (vis[{r, c, s}]) continue;
        vis[{r, c, s}] = 1;
        dist[r][c] = s, step[r][c] = init[r][c];
        for (int i = 0; i < 4; i++) {
            int nr = r + dx[i], nc = c + dy[i];
            if (nr < 1 || nr > 4 || nc < 1 || nc > n) continue;
            if (init[nr][nc])
                q.emplace(nr, nc, 0);
            else if (!vis[{nr, nc, init[r][c]}])
                q.emplace(nr, nc, s + 1);
        }
    }
    int ans = k * 20001;
    for (int i = 2; i <= 3; i++) {
        for (int j = 1; j <= n; j++) {
            if (des[i][j] && init[i][j]) {
                ans = min(ans, dist[i][j] + abs(des[i][j] - init[i][j]));
            }
        }
    }
    return ans;
}
int main() {
    cin >> n >> k;
    for (int i = 1; i <= 4; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> init[i][j];
            if (i == 2 || i == 3) step[i][j] = init[i][j];
        }
    }
    for (int i = 1; i <= 4; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> des[i][j];
        }
    }
    int ans = bfs(2, 1);
    if (ans > 20000)
        cout << -1;
    else {
        cout << ans << endl;
        while (step[2][1]) {
            int x = step[2][1], y = 1;
            for (int i = 2; i <= 3; i++) {
                x = min(des[i][x], step[i][x]);
                y = x;
            }
            step[2][1] = step[x][y];
            cout << x << ' ' << y << endl;
        }
    }
    return 0;
}
