#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        map<int, int> mp;
        int mx = 0;
        for (int i = 0; i < n; i++) {
            int w;
            cin >> w;
            mp[w]++;
            mx = max(mx, w);
        }
        int ans = 0;
        for (int j = 1; j <= mx; j++) {
            int grp = 0;
            for (int k = j; k <= mx; k += j) {
                grp += min(mp[k], mp[j + k - k]);
            }
            ans = max(ans, grp / 2);
        }
        cout << ans << endl;
    }
    return 0;
}
