import java.util.*;
import java.io.*;

public class Main {
    static final long MOD = 1000000007L;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] x = new long[n], y = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
        }

        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 2; j < n; j++) {
                long area1 = 0, area2 = 0;
                for (int k = 0; k < n; k++) {
                    if (k != i && k != j) {
                        long ax = j - i, bx = k - i, cx = j - k;
                        long ay = y[j] - y[i], by = y[k] - y[i], cy = y[j] - y[k];
                        long area = Math.abs(ax * (by * cx - cy * bx) + bx * (cy * ax - cx * ay) + cx * (ay * bx - by * ax));
                        if ((k - i) % 2 == 1) {
                            area1 += area;
                        } else {
                            area2 += area;
                        }
                    }
                }
                sum += ((area1 - area2) % MOD + MOD) % MOD;
                sum %= MOD;
            }
        }

        System.out.println(sum);
    }
}
