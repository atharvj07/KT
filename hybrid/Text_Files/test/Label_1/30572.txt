import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        int k = scanner.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextLong();
        }
        long min = Long.MAX_VALUE;
        int minIndex = -1;
        for (int i = 0; i < k; i++) {
            if (a[i] > 0 && a[i] <= n && a[i] < min) {
                min = a[i];
                minIndex = i + 1;
            }
        }
        System.out.println(minIndex + " " + n / min);
    }
}
