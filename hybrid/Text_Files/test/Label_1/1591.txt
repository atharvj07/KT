#include <iostream>
#include <vector>

using namespace std;
typedef long long ll;

const int MAXN = 105;
int n, m, q;
vector<vector<ll>>a;
vector<ll>rows, cols;
vector<string>ops;
vector<vector<ll>>opsv;

void swap(ll &x, ll &y) {
  ll t = x;
  x = y;
  y = t;
}

void shift_row(ll x) {
  ll f = a[x][0];
  for (ll j = 0; j < m - 1; j++) a[x][j] = a[x][j + 1];
  a[x][m - 1] = f;
}

void shift_col(ll x) {
  ll f = a[0][x];
  for (ll i = 0; i < n - 1; i++) a[i][x] = a[i + 1][x];
  a[n - 1][x] = f;
}

int main() {
  cin >> n >> m >> q;
  a.resize(n, vector<ll>(m));
  rows.resize(n), cols.resize(m), opsv.resize(q);
  for (ll i = 0; i < n; i++) { cin >> rows[i]; }
  for (ll i = 0; i < m; i++) { cin >> cols[i]; }
  for (ll i = 0; i < q; i++) {
    string op;
    ll x, y, z;
    cin >> op;
    ops.push_back(op);
    if (op == "1" || op == "2") { cin >> x;  opsv[i].push_back(x); }
    if (op == "3") {
      cin >> x >> y >> z;
      opsv[i].push_back(x);
      opsv[i].push_back(y);
      opsv[i].push_back(z);
    }
  }
  for (ll i = 0; i < q; i++) {
    if (ops[i] == "1") { shift_row(opsv[i][0] - 1); }
    if (ops[i] == "2") { shift_col(opsv[i][0] - 1); }
    if (ops[i] == "3") { a[opsv[i][0] - 1][opsv[i][1] - 1] = opsv[i][2]; }
  }
  for (ll i = 0; i < n; i++) {
    for (ll j = 0; j < m; j++) { cout << a[i][j] << " "; }
    cout << endl;
  }
  return 0;
}
