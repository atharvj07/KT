#include <cstdio>
#include <algorithm>
using namespace std;

int main() {
    int n, m, min, max, t;
    scanf("%d%d%d%d", &n, &m, &min, &max);
    int temp[1001];
    for (int i = 0; i < m; ++i) {
        scanf("%d", &temp[i]);
    }
    sort(temp, temp + m);
    int p = 0, q = m - 1, yes = 0;
    for (int i = 0; i < n - m; ++i) {
        if (temp[p] == min) {
            ++p;
        } else if (temp[q] == max) {
            --q;
        } else {
            yes = 1;
            break;
        }
    }
    printf("%s", yes ? "Incorrect" : "Correct");
    return 0;
}
