#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    for (int i = 0; i < s.size(); i++) {
      for (int j = 'a'; j <= 'z'; j++) {
        if (j < s[i]) {
          s[i] = j;
          break;
        }
      }
    }
    cout << s << endl;
  }
  return 0;
}
