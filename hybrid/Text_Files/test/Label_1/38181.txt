import sys

t = int(input())

for _ in range(t):
    n = int(input())
    
    ans = []
    
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            ans.append(i)
            n //= i
            
            if n % i == 0:
                ans.append(i)
                n //= i
    
    if n > 1:
        ans.append(n)
    
    if len(ans) < 3:
        print("NO")
    else:
        print("YES")
        print(*ans)
