#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const int maxn=2e5+1;
int n,q;
ll a[maxn],b[maxn],sum[maxn];
int main(){
	cin>>n>>q;
	for(int i=1;i<=n;++i)cin>>a[i],b[i]=a[i];
	sort(b+1,b+n+1);for(int i=1;i<=n;++i)sum[i]=sum[i-1]+b[i];
	ll ans=LLONG_MIN;
	while(q--){
		int l,r;cin>>l>>r;
		ans=max(ans,sum[r]-sum[l-1]+1ll*(r-l+1)*a[l]);
	}
	cout<<ans<<'\n';
	return 0;
}
