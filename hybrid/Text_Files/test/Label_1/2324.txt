#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e5 + 5;
ll a[MAXN];
int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a, a + n);
        ll ans = 0;
        for (int i = 0; i < n - 2 * k; i++) ans += a[i];
        for (int i = n - 2 * k; i < n; i += 2) {
            ll x = a[i], y = a[i + 1];
            ll p = x / y;
            ans += max(x % y, p);
        }
        cout << ans << "\n";
    }
    return 0;
}
