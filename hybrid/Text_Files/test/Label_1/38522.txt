import sys
input = sys.stdin.readline

n, m, k, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = []
for i in range(m):
    t, p = map(int, input().split())
    c.append((p, t))
c.sort(key=lambda x: x[0])
ans = []
cur = 0
day = 0
for p, t in c:
    if t == 1:
        if cur + p // a[day] >= k:
            ans.append((k - cur, day + 1))
            break
        else:
            ans.append((p // a[day], day + 1))
            cur += p // a[day]
            s -= p // a[day] * a[day]
    else:
        if cur + p // b[day] >= k:
            ans.append((k - cur, day + 1))
            break
        else:
            ans.append((p // b[day], day + 1))
            cur += p // b[day]
            s -= p // b[day] * b[day]
    day += 1
if len(ans) == k:
    print(ans[0][1])
    for gadget, day in ans:
        print(gadget, day)
else:
    print(-1)
