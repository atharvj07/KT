import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }

        int ans = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += w[i];
            if (sum == 0) {
                ans++;
                sum = 0;
            } else if (sum > 0) {
                for (int j = i; j >= 0; j--) {
                    sum -= w[j];
                    if (sum == 0) {
                        ans++;
                        sum = 0;
                        break;
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
