n,s = input()
row = int(n)
seconds = 0
if row % 3 == 0:
    seconds += 6 + row // 3 * 4
    if s in ['a', 'b', 'f']:
        seconds += 6
    elif s in ['c', 'd']:
        seconds += 3
elif row % 3 == 1:
    seconds += 1+ (row-1) // 3 * 4
    if s in ['a', 'b', 'c']:
        seconds += 4
    elif s in ['e', 'f']:
        seconds += 9
elif row % 3 == 2:
    seconds += 6+ row // 3 * 4
    if s in ['c', 'd', 'e']:
        seconds += 6
    elif s in ['a', 'b', 'f']:
        seconds += 3
print(seconds)
