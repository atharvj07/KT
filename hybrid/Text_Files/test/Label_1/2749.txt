#include <bits/stdc++.h>
#define ll long long
using namespace std;

ll solve() {
  ll n;
  cin >> n;
  vector<vector<pair<int, int>>> a(n);
  ll mx = 0;
  ll x = 0;
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    ll k;
    cin >> k;
    for (int j = 0; j < k; j++) {
      int x, y;
      cin >> x >> y;
      a[i].push_back({x, y});
    }
    sort(a[i].begin(), a[i].end());
  }
  for (int i = 0; i < n; i++) {
    int j = 0;
    ll sm = 0;
    for (auto x : a[i]) {
      if (j && j % 10 == 0) {
        x.second *= 2;
      }
      if (sm + x.first <= 3) {
        sm += x.first;
        ans += x.second;
        j++;
      } else {
        break;
      }
    }
  }
  return ans;
}
int main() {
  cout << solve() << endl;
  return 0;
}
