#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int M = 1e9 + 7;
int main() {
    int n;
    cin >> n;
    priority_queue<pair<int, int>> p;
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        p.push({a + b, a - b});
    }
    int c = 0;
    ll ans = LLONG_MAX;
    while (!p.empty()) {
        ll a = p.top().first;
        ll b = p.top().second;
        p.pop();
        ans = min(ans, c + max(a - b, 0LL));
        c += (b > 0);
    }
    cout << ans << endl;
    return 0;
}
