n = int(input())
cards = list(map(int, input().split()))

counts = [0] * 9  # Index 0 is unused, 1-8 for card numbers
for card in cards:
    counts[card] += 1

# Find the minimum and maximum counts ignoring zeros
min_count = min(filter(lambda x: x > 0, counts))
max_count = max(counts)

# Calculate the maximum length satisfying both conditions
length = 0
for count in counts:
    if count == max_count:
        length += count
    elif count > 0:
        length += min_count
print(length)
