#include <cstdio>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  long long n; scanf("%lld", &n);
  vector<long long> a(n);
  for (long long i = 0; i < n; ++i) scanf("%lld", &a[i]);
  sort(a.begin(), a.end());
  for (long long i = 0; i < n; ++i) if (a[i] & 1) a[i] = 0;
  long long Ans = 0;
  for (long long i = 0; i < n; ++i) Ans += a[i];
  printf("%lld\n", Ans);
  return 0;
}
