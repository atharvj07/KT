#include <bits/stdc++.h>
using namespace std;
const int maxn = 35, masks = 1 << 11;
int n, k, f[maxn][masks][2], g[maxn][masks][2], p[maxn];
char c[105];
int main() {
  cin >> n >> k;
  for (int i = 1; i < n * 2; i++) { cin >> p[i]; }
  for (int i = 1; i < 2 * n; i++) {
    for (int j = 1; j < 1 << n; j++) {
      if (i > 1) {
        f[i][j][0] = max(f[i - 1][j][0], f[i - 1][j][1]);
        g[i][j][0] = max(g[i - 1][j][0], g[i - 1][j][1]);
      }
      if (p[i] + 1 <= n) {
        if ((j >> p[i]) & 1 || j == 0) {
          f[i][j][0] = max(f[i][j][0], g[i - 1][j ^ (1 << p[i])][1] + 1);
        }
      }
      if (p[i] > 0) {
        if ((j >> (p[i] - 1)) & 1 || j == 0) {
          g[i][j][0] = max(g[i][j][0], f[i - 1][j ^ (1 << (p[i] - 1))][0] + 1);
        }
      }
    }
    for (int j = 1; j < 1 << n; j++) {
      f[i][j][1] = f[i][j][0];
      g[i][j][1] = g[i][j][0];
      if (p[i] + 1 <= n) {
        f[i][j][1] = max(f[i][j][1], f[i - 1][j][1]);
        if ((j >> p[i]) & 1) {
          f[i][j][1] = max(f[i][j][1], g[i - 1][j ^ (1 << p[i])][0]);
        }
      }
      if (p[i] > 0) {
        g[i][j][1] = max(g[i][j][1], g[i - 1][j][1]);
        if ((j >> (p[i] - 1)) & 1) {
          g[i][j][1] = max(g[i][j][1], f[i - 1][j ^ (1 << (p[i] - 1))][1]);
        }
      }
    }
  }
  for (int j = 1; j < 1 << n; j++) {
    if (__builtin_popcount(j) == n) {
      f[n * 2][j][0] = max(f[n * 2][j][0], g[n * 2 - 1][j][1] + 1);
    }
  }
  cout << g[n * 2 - 1][(1 << n) - 1][1] << endl;
  return 0;
}
