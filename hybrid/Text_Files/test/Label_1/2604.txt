#include <iostream>
#include <vector>

using namespace std;

vector<bool> status(10, false);

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'L') {
      int j = 0;
      while (status[j]) j++;
      status[j] = true;
    } else if (s[i] == 'R') {
      int j = 9;
      while (status[j]) j--;
      status[j] = true;
    } else {
      status[s[i] - '0'] = false;
    }
  }
  for (int i = 0; i < 10; i++) {
    cout << (status[i] ? '1' : '0');
  }
  return 0;
}
