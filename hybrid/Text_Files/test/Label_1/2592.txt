#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    sort(s.begin(), s.end());
    if (s[n - 1] == s[n - 2]) {
        if (k >= 3) cout << "YES" << endl;
        else cout << "NO" << endl;
    } else {
        if (k >= 2) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
}
