import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            Arrays.sort(a);
            long sum = 0;
            boolean ok = false;
            for (int j = 0; j < n; j++) {
                sum += a[j] * (n - j);
            }
            if (sum == m) {
                ok = true;
            }
            System.out.println(ok ? "YES" : "NO");
        }
    }
}
