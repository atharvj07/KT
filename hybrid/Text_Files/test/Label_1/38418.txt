n = int(input())
t = list(map(int, input().split())) + [0]
l, r = [0] * (n + 1), [0] * (n + 1)
for i in range(n):
    num = t[i]
    l[num] = max(l[num], i)
    r[num] = i
ans = [0] * (n + 1)
for i in range(1, n + 1):
    for j in range(1, n + 1):
        if l[j] >= l[i] and r[j] <= r[i]:
            ans[j] += 1
for i in range(1, n + 1):
    print(ans[i])
