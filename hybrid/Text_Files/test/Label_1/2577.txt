#include <bits/stdc++.h>
using namespace std;
typedef int64_t ll;
const int mod = 1e9 + 7;

mt19937 rng((uint64_t)chrono::steady_clock::now().time_since_epoch().count());

ll k, n, m, mn, ans;
map<ll, ll> mp;

int main() {
    cin >> n >> m;
    while (n--) {
        ll x;
        cin >> x;
        mp[x]++;
    }
    while (m--) {
        ll x;
        cin >> x;
        k += x;
    }
    mn = n;
    for (auto& i : mp) {
        if (k <= i.second) mn = min(mn, i.second - k);
    }
    cout << (mn == n ? -1 : mn) << endl;
    return 0;
}
