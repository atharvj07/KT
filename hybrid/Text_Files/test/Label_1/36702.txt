import sys
N, M, K, S = list(map(int, input().split()))
cost_doll = list(map(int, input().split()))
cost_pow = list(map(int, input().split()))
need = []
for _ in range(M):
    tp, c = map(int, input().split())
    if tp == 1:
        need.append(((c + cost_doll[0] - 1) // cost_doll[0], -1))
    else:
        need.append(((c + cost_pow[0] - 1) // cost_pow[0], 1))
need.sort()
ans = []
rem = S
for c, tp in need:
    if tp == -1:
        rem -= (c * cost_doll[0])
    else:
        rem -= (c * cost_pow[0])
    if rem < 0:
        print(-1)
        sys.exit(0)
    ans.append((c, tp))
    if len(ans) == K:
        break
day = 1
ans = reversed(ans)
for c, tp in ans:
    print(day)
    if tp == -1:
        for i in range(c):
            print(need.index((c, -1)) + 1, day)
            need[need.index((c, -1))] = (-1, -1)
    else:
        for i in range(c):
            print(need.index((c, 1)) + 1, day)
            need[need.index((c, 1))] = (-1, -1)
    day += 1
