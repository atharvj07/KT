#include <bits/stdc++.h>
using namespace std;
const int max_n = 105, max_m = 1e6 + 5;
int n, k, ans, a[max_n], f[max_m];
int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  sort(a + 1, a + n + 1);
  int sum = 0;
  for (int i = n; i >= 1; i -= 2) { sum += (a[i] != a[i - 1]); }
  cout << max(0, k - sum + 1) << endl;
  return 0;
}
