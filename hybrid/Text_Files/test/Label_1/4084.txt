#include<iostream>
#include<vector>
#include<climits>
using namespace std;
typedef long long ll;

int main() {
	int t; cin >> t;
	while (t--) {
		int n; cin >> n;
		vector<int> v(n + n);
		for (int i = 0; i < n; i++) {
			cin >> v[i];
			v[i + n] = v[i];
		}

		ll LIS = 0;
		vector<int> dp(n + n, 1);
		for (int i = 1; i < n + n; i++) {
			for (int j = 0; j < i; j++) {
				if (v[i] > v[j]) dp[i] = max(dp[i], 1 + dp[j]);
			}
			LIS = max(LIS, (ll)dp[i]);
		}

		cout << LIS << endl;
	}
}
