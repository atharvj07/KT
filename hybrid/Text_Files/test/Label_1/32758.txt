import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long d = sc.nextInt();
            long k = sc.nextInt();
            if ((d - k) % 2 == 0)
                System.out.println("Utkarsh");
            else
                System.out.println("Ashish");
        }
    }
}
