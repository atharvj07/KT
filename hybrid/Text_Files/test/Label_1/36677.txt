import sys
input = sys.stdin.readline

n = int(input())
color = [int(x) for x in input().split()]
edges = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)

vis = [False] * (n+1)
cnt = [0, 0]

def dfs(node, par):
    vis[node] = True
    cnt[color[node-1]] += 1
    for u in edges[node]:
        if not vis[u]:
            dfs(u, node)

dfs(1, 1)
ans = min(cnt)
print(ans)
