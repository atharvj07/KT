import java.util.*;
import java.io.*;

class Main {
    static HashMap<Long,Long> val = new HashMap<Long,Long>();
    static long gcd(long a,long b ){
        if(b==0)
            return a;
        else
            return gcd(b,a%b);
    }
    static long fact(long a){
        if(a==0||a==1)
            return 1;
        if(val.containsKey(a))
            return val.get(a);
        else{
            val.put(a,a*fact(a-1));
            return val.get(a);
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long A = sc.nextLong();
        long B = sc.nextLong();
        System.out.println(gcd(fact(A),fact(B)));
    }
}
