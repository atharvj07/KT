import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), k = input.nextInt();
        char[] digits = input.next().toCharArray();

        int i = 0;
        // find the first digit that can be decreased
        while (i < n && digits[i] == '1') {
            i++;
        }

        // decrease the first digit that can be decreased
        if (i < n && k > 0) {
            digits[i] = '0';
            k--;
        }

        // change the remaining k digits to 0
        while (k > 0) {
            digits[i] = '0';
            i++;
            k--;
        }

        // print the resulting number
        System.out.println(digits);
    }
}
