import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int c8 = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '8') {
                c8++;
            }
        }
        if (c8 == 0 || n - c8 < 10) {
            System.out.println(0);
        } else {
            System.out.println(c8 + (n - c8) / 10);
        }
    }
}
