import java.util.*;
import java.io.*;

public class Main {
    static int[][] a, b;
    static int n, m;
    static boolean[][] u;

    static int dfs(int x, int y, int dx, int dy) {
        int ret = a[x][y];
        u[x][y] = true;
        while (true) {
            int nx = x + dx;
            int ny = y + dy;
            if (nx < 0 || nx >= n || ny < 0 || ny >= m || u[nx][ny] || (x == nx && y == ny))
                break;
            ret += a[nx][ny];
            x = nx;
            y = ny;
            u[nx][ny] = true;
        }
        return ret;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n][m];
        b = new int[n][m];
        u = new boolean[n][m];
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < m; ++j)
                a[i][j] = sc.nextInt();
        int ans = 0;
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < m; ++j) {
                if (u[i][j])
                    continue;
                ans = Math.max(ans, dfs(i, j, 1, 0));
                ans = Math.max(ans, dfs(i, j, 0, 1));
                ans = Math.max(ans, dfs(i, j, -1, 0));
                ans = Math.max(ans, dfs(i, j, 0, -1));
            }
        System.out.println(ans);
    }
}
