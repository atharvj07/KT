#include <bits/stdc++.h>
using namespace std;
struct data{
    int val,num;
};
bool comp(data a,data b){
    return a.val<b.val;
}
int main(){
    int n,s,i,ans=0;
    cin>>n>>s;
    vector<data> v(n);
    for(i=0;i<n;i++){
        cin>>v[i].val;
        v[i].num=i-n/2;
    }
    sort(v.begin(),v.end(),comp);
    for(i=1;i<v.size();i++){
        if(v[i].val==v[i-1].val) ans+=abs(v[i].num);
        else{
            ans+=abs(v[i].num-v[i-1].num);
            v[i].num=v[i-1].num+v[i].num;
        }
    }
    int carry=(v[i-1].val<s)?1:0;
    ans+=carry;
    cout<<ans<<'\n';
    return 0;
}
