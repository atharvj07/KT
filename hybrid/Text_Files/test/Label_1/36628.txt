h, w = map(int, input().split())
pic = [list(input()) for _ in range(h)]

def valid(y, x):
    if y < 0 or y >= h or x < 0 or x >= w or pic[y][x] == '.':
        return False
    return True

def dfs(y, x):
    if not valid(y, x):
        return False
    pic[y][x] = '.'
    if dfs(y + 1, x):
        return True
    if dfs(y - 1, x):
        return True
    if dfs(y, x + 1):
        return True
    if dfs(y, x - 1):
        return True
    
    return False

cnt = 0
for i in range(h):
    for j in range(w):
        if pic[i][j] == '*':
            if cnt > 0:
                print('NO')
                exit()
            cnt += 1
            if dfs(i, j):
                print('YES')
                exit()

print('NO')
