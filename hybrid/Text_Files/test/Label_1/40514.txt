import sys
input = sys.stdin.readline

n = int(input())
p = list(map(int,input().split()))
a = list(map(int,input().split()))
ans = a[0]
vis = [False] * (n+1)
vis[1] = True
st = [1]

while st:
  u = st.pop()
  vis[u] = False
  if a[u] > 0:
    ans += a[u] - 1
    a[u] = 0
    a[p[u]] += a[u]
    if not vis[p[u]]:
      st.append(p[u])
      vis[p[u]] = True
print(ans)
