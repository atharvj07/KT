#include <iostream>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int cnt = 0;
  char c = s[0];
  int len = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == c) {
      len++;
      if (len >= k) {
        cnt++;
        len = 0;
      }
    } else {
      len = 1;
      c = s[i];
    }
  }
  cout << cnt << endl;
  return 0;
}
