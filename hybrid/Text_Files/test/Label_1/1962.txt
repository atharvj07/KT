#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> grid(n);
  for (int i = 0; i < n; i++) { cin >> grid[i]; }
  int q;
  cin >> q;
  while (q--) {
    int l, r;
    cin >> l >> r;
    bool ok = true;
    for (int i = 0; i < n && ok; i++) {
      int j = l - 1;
      while (j <= r && grid[i][j] == '.')
        j++;
      if (j > r) ok = false;
    }
    cout << (ok ? "YES\n" : "NO\n");
  }
}
