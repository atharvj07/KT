#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    if (n == 1) {
      cout << "()" << endl;
      continue;
    }
    cout << "(";
    for (int i = 1; i < n; ++i) {
      cout << "()";
    }
    cout << ")()";
    for (int i = 1; i < n - 1; ++i) {
      cout << "()";
    }
    cout << ")" << endl;
  }
}
