import java.util.Scanner;

public class ValerieAndCrucians {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt();
            teeth[i][1] = scanner.nextInt();
        }

        int maxCrucians = 0;
        for (int i = 0; i < m; i++) {
            int row = i;
            int remainingViability = k;
            while (remainingViability > 0) {
                int currentTooth = 0;
                while (currentTooth < n && teeth[currentTooth][0] != row) {
                    currentTooth++;
                }
                if (currentTooth == n) {
                    break;
                }
                remainingViability -= teeth[currentTooth][1];
                maxCrucians++;
            }
        }

        System.out.println(maxCrucians);
    }
}
