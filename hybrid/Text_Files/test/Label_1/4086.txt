#include<bits/stdc++.h>
using namespace std;
#define loop(i,L,R) for(int i=(L);i<=(R);i++)
#define re_loop(i,L,R) for(int i=(L);i>=(R);i--)
typedef long long ll;
int dx[]={-1,1,0,0};
int dy[]={0,0,-1,1};
#define pi acos(-1)
#define pb push_back
#define endl "\n"
ll mod=1e9+7;
//const double re=1.570796;
const double eps=1e-6;
int dir[100009];
int main()
{
    ll x1,y1,x2,y2,n,ans=0,check=0;
    cin>>x1>>y1>>x2>>y2>>n;
    string s;
    map<pair<ll, ll>, ll> vis;
    cin>>s;
    loop(i,0,n-1)
    {
        if(s[i]=='U') dir[i]=0;
        else if(s[i]=='D') dir[i]=1;
        else if(s[i]=='L') dir[i]=2;
        else dir[i]=3;
    }
    loop(i,0,n-1)
    {
        x1+=dx[dir[i%n]];
        y1+=dy[dir[i%n]];
        if(x1==x2&&y1==y2)
        {
            check=1;
            ans=i+1;
            break;
        }
        if(vis[{x1,y1}]==1) break;
        vis[{x1,y1}]++;
    }
    if(check==0) ans=-1;
    cout<<ans<<endl;
    return 0;
}
