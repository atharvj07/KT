import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String S = in.nextLine();
        int a = 0, b = 0;
        for (char c : S.split(":")[0].toCharArray()) {
            a = a * 10 + getVal(c);
        }
        for (char c : S.split(":")[1].toCharArray()) {
            b = b * 10 + getVal(c);
        }
        if (b > 59) {
            System.out.println(0);
            return;
        }
        if (a == 24 && b == 0) {
            System.out.println(-1);
            return;
        }
        for (int r = 2; r * r <= a + b; r++) {
            if ((a + b) % r == 0) {
                System.out.print(r + " ");
            }
        }
    }
    static int getVal(char c) {
        if (c >= '0' && c <= '9') {
            return c - '0';
        }
        return c - 'A' + 10;
    }
}
