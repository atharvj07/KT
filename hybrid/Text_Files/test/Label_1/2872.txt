#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int ans = a[n - 1] - a[0];
  if (k >= (a[n - 1] - a[0]) / 2) {
    ans /= 2;
    k -= (a[n - 1] - a[0]) / 2;
  }
  if (k >= n) ans = 0;
  cout << ans << endl;
  return 0;
}
