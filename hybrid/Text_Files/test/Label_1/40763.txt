from collections import Counter, defaultdict
def solve():
    inp = input().strip()
    st, en = defaultdict(lambda: []), defaultdict(lambda: [])
    cnt = Counter()
    for i in range(len(inp)):
        cnt[inp[i]] += 1
        en[inp[i]].append(i)
    q = int(input().strip())
    res = []
    for _ in range(q):
        arr = input().strip().split()
        if arr[0] == '2':
            l, r = list(map(int, arr[1:]))
            l -= 1
            r -= 1
            a, b = st[inp[l]], en[inp[r]]
            if len(a) == len(b):
                ans = sum(cnt.values())
            else:
                ans = len(cnt)
            for i in range(len(a)):
                if a[i] < b[i]:
                    val = max(0, cnt[inp[b[i]]] - b[i] - (len(b) - 1))
                    ans -= val
                else:
                    val = max(0, cnt[inp[a[i]]] - l + (len(a) - 1))
                    ans -= val
            res.append(ans)
        else:
            inp = inp[:int(arr[1])-1] + arr[2] + inp[int(arr[1]):]
            st[inp[int(arr[1])-1]].clear()
            st[inp[int(arr[1])-1]].append(int(arr[1]) - 1)
            en[inp[int(arr[1])-1]][-1] = int(arr[1]) - 1
    return res

print(*solve(), sep='\n')
