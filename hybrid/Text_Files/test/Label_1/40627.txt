n, k = map(int, input().split())
a = list(map(int, input().split()))
t = list(map(int, input().split()))

best = 0
for i in range(n - k + 1):
    cur = sum(a[i:i + k])
    for j in range(i):
        cur += t[j] * a[j]
    for j in range(i + k, n):
        cur += t[j] * a[j]
    best = max(best, cur)

print(best)
