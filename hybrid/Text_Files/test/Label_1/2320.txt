#include <bits/stdc++.h>
using namespace std;

int n, m, s;
vector <int> adj[5010];
bool vis[5010];

void dfs(int cur) {
    vis[cur] = 1;
    for (int u: adj[cur]) {
        if (!vis[u]) {
            dfs(u);
        }
    }
}

int main() {
    cin >> n >> m >> s;
    for (int i = 1, u, v; i <= m; ++i) {
        cin >> u >> v;
        adj[u].push_back(v);
    }
    dfs(s);
    int cnt = 0;
    for (int i = 1; i <= n; ++i) {
        cnt += !vis[i];
    }
    cout << cnt << endl;
}
