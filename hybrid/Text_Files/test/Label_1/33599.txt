import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
        }
        assert sum == 0;
        int ans = 0;
        for (int i = 0; i < n;) {
            if (a[i] >= 0) {
                i++;
                continue;
            }
            int j = (i + 1) % n;
            while (a[j] == 0) {
                j = (j + 1) % n;
            }
            int x = Math.min(-a[i], a[j]);
            a[i] += x;
            a[j] -= x;
            ans++;
            i++;
        }
        System.out.println(ans);
    }
}
