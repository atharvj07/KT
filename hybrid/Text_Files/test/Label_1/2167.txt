#include <bits/stdc++.h>
using namespace std;
#define pb push_back
inline int read() {
	register int x;
	register int k;
	x = 0;
	k = 1;
	register char c;
	for (c = getchar(); !isdigit(c); c = getchar())
		if (c == '-') k = -1;
	for (; isdigit(c); c = getchar()) x = x * 10 + c - '0';
	return x * k;
}

const int MOD = 998244353;
const int maxn = 2e5 + 5;
vector<int> G[maxn];
int dis[maxn];
queue<int> q;

int main() {
	int t = read();
	while (t--) {
		int n = read();
		for (int i = 1; i <= n; i++) G[i].clear();
		for (int i = 0; i < n - 1; i++) {
			int x = read();
			int y = read();
			G[x].pb(y);
			G[y].pb(x);
		}
		for (int i = 1; i <= n; i++) dis[i] = 2e9;
		dis[1] = 0;
		q.push(1);
		while (!q.empty()) {
			int u = q.front();
			q.pop();
			for (int i = 0; i < G[u].size(); i++) {
				int v = G[u][i];
				if (dis[v] > dis[u] + 1) {
					dis[v] = dis[u] + 1;
					q.push(v);
				}
			}
		}
		int ans = 0;
		for (int i = 1; i <= n; i++) ans = max(ans, dis[i]);
		printf("%d\n", ans);
	}
	return 0;
}
