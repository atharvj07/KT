#include <bits/stdc++.h>

using namespace std;

const long long INF = 1e18;

int main() {
    int n, r;
    cin >> n >> r;
    vector<pair<int, int>> projects(n);
    for (auto& p : projects) cin >> p.first >> p.second;
    sort(projects.begin(), projects.end());
    int ans = 0;
    for (const auto& p : projects) {
        if (r >= p.first) {
            r += p.second;
            ans++;
        }
    }
    cout << ans << "\n";
    return 0;
}
