import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), a = in.nextInt(), b = in.nextInt();
    String[] companys = in.next().split("");
    int[] arr = new int[n];
    int cost = Math.abs(a - b);
    for (int i = 0; i < n; i++) {
      arr[i] = companys[i].equals("0") ? 0 : 1;
    }
    if (arr[a - 1] == arr[b - 1]) {
      cost = 0;
    } else {
      for (int i = 0; i < n; i++) {
        if (arr[i] == arr[a - 1]) {
          cost = Math.min(cost, Math.abs(i - b) + Math.abs(i - a));
        }
      }
    }
    System.out.println(cost);
  }
}
