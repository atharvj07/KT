n, q = map(int, input().split())
a = list(map(int, input().split()))

# To store the rightmost occurrence of each query number
rightmost = dict()
for i in range(n):
    if a[i] != 0:
        rightmost[a[i]] = i

# List to store the segments
segments = []

# Creating segments from the rightmost occurrences
for num in sorted(rightmost.keys()):
    segments.append((rightmost[num], num))

# Sort the segments by their end positions
segments.sort()

# Final array to be obtained
final_array = [0] * n

# Cover the segments
for i, num in segments:
    if all(final_array[j] == 0 or final_array[j] == num for j in range(i)):
        for j in range(i):
            final_array[j] = num
    else:
        print("NO")
        break
else:
    if all(x != 0 for x in final_array):
        print("YES")
        print(*final_array)
    else:
        # Fill in the blanks with the highest number possible
        last_num = 1
        for i in range(n):
            if final_array[i] == 0:
                final_array[i] = last_num
            else:
                last_num = final_array[i]
        print("YES")
        print(*final_array)
