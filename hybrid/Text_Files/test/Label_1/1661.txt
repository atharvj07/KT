#include <iostream>

int main() {
  long long q, k;
  std::cin >> q;
  while (q--) {
    std::cin >> k;
    long long block_len = 0;
    long long sum = 0;
    while (sum < k) {
      block_len += 1;
      sum += block_len;
    }
    long long pos = k - (sum - block_len);
    std::cout << pos << std::endl;
  }
  return 0;
}
