import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] chars = sc.next().toCharArray();
        int min = n;
        for (int i = 0; i < n / 2; i++) {
            if (chars[i] == chars[n - i - 1]) {
                min = Math.min(min, n - i - 1);
            }
        }
        System.out.println(min);
    }
}
