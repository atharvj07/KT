def main():
    s = input()
    n = len(s)
    if s[0] != s[n-1] or '0' not in s or '1' not in s:
        print(-1)
        return

    a = [i for i, v in enumerate(s) if v == '0']
    b = [i for i, v in enumerate(s) if v == '1']

    m = min(len(a), len(b))

    print(m)
    for i in range(m):
        print(b[i]-a[i]+1, *range(a[i]+1, b[i]+1))

main()
