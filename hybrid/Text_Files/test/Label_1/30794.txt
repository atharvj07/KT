import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();  
        int m = sc.nextInt(); 
        int[] sx = new int[m], sy = new int[m], dist = new int[n * n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        int sxy = sc.nextInt() - 1 + n * (sc.nextInt() - 1); 
        int fxy = sc.nextInt() - 1 + n * (sc.nextInt() - 1); 
        dist[sxy] = 0;
        for (int i = 0; i < m; i++) {
            sx[i] = sc.nextInt() - 1;
            sy[i] = sc.nextInt() - 1;
            dist[sx[i] + n * sy[i]] = 0;
        }
        Queue<Integer> q = new LinkedList<>();
        q.offer(sxy);
        while (!q.isEmpty()) {
            int xy = q.poll();
            int x = xy % n, y = xy / n;
            for (int i = 0; i < m; i++) {
                if (x == sx[i] || y == sy[i]) {
                    if (dist[xy] + 1 < dist[sx[i] + n * sy[i]]) {
                        dist[sx[i] + n * sy[i]] = dist[xy] + 1;
                        q.offer(sx[i] + n * sy[i]);
                    }
                }
            }
            if (x > 0 && dist[xy] + 1 < dist[xy - 1]) {
                dist[xy - 1] = dist[xy] + 1;
                q.offer(xy - 1);
            }
            if (x < n - 1 && dist[xy] + 1 < dist[xy + 1]) {
                dist[xy + 1] = dist[xy] + 1;
                q.offer(xy + 1);
            }
            if (y > 0 && dist[xy] + 1 < dist[xy - n]) {
                dist[xy - n] = dist[xy] + 1;
                q.offer(xy - n);
            }
            if (y < n - 1 && dist[xy] + 1 < dist[xy + n]) {
                dist[xy + n] = dist[xy] + 1;
                q.offer(xy + n);
            }
        }
        System.out.println(dist[fxy]);
    }
}
