#include<bits/stdc++.h>
using namespace std;
int m,n,x,a[100001],b[100001],c[100001],d[100001];
int main(){
    int T;
    cin>>T;
    for(int q=1;q<=T;q++){
        cin>>n>>x;
        for(int i=1;i<=n;i++){
            cin>>a[i];
            b[i]=a[i];
        }
        for(int i=1;i<=n;i++){
            if(b[i]%x!=0){
                d[i]=1;
            }
            d[i]+=d[i-1];
        }
        int w=0,ans=0;
        for(int i=1;i<=n;i++){
            for(int j=i;j<=n;j++){
                w+=a[j];
                if(w%x!=0){
                    int v=d[j];
                    if(i>1) v-=d[i-1];
                    ans=max(ans,j-i+1+v);
                }
            }
            w=0;
        }
        if(ans==0) cout<<-1<<endl;
        else cout<<ans<<endl;
    }
    return 0;
}
