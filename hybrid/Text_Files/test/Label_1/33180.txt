import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        Arrays.sort(a);
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum > n) {
                System.out.println(-1);
                return;
            }
        }
        for (long i = n - 1; i >= a[0]; i--) {
            if ((sum + i) % 2 == 0 && (sum + i) / 2 <= n) {
                System.out.print(i + " ");
            }
        }
        System.out.println();
    }
}
