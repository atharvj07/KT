import java.util.*;

public class Main {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long gcd = sc.nextInt();
        long lcm = sc.nextInt();
        for (int i = 2; i < n; i++) {
            long a = sc.nextInt(), b = sc.nextInt();
            gcd *= gcd(gcd, a * lcm / gcd(a, lcm));
            lcm = lcm * b / gcd(lcm, b);
        }
        System.out.println(lcm / gcd);
    }

    public static long gcd(long a, long b) {
        while (b != 0) {
            long t = a;
            a = b;
            b = t % b;
        }
        return a;
    }
}
