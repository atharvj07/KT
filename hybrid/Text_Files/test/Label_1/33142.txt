import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int m = s.nextInt(), n = s.nextInt();
        int count = 0;
        if (m / 2 > n || m % n == m - 1) {
            System.out.println(-1);
            return;
        }

        while (m > 0) {
            if (m >= n) {
                count += (m + n - 1) / n;
                m -= n - 1;
            } else {
                count += 2;
                m -= 1;
            }
        }
        System.out.println(count);
    }
}
