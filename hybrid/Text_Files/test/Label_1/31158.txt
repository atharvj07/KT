import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Edge> adj[] = new List[n + 1];
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int l = sc.nextInt();
            int r = sc.nextInt();
            adj[a].add(new Edge(b, l, r));
            adj[b].add(new Edge(a, l, r));
        }
        int[] dp = new int[n + 1];
        boolean[] vis = new boolean[n + 1];
        int ans = dfs(1, -1, dp, vis, adj);
        System.out.println(ans == 0 ? "Nice work, Dima!" : ans);
    }

    static int dfs(int node, int prev, int[] dp, boolean[] vis, List<Edge>[] adj) {
        vis[node] = true;
        dp[node] = 0;
        for (Edge edge : adj[node]) {
            if (!vis[edge.to] && edge.l <= edge.r) {
                dp[node] += dfs(edge.to, node, dp, vis, adj);
            } else if (edge.to == prev && edge.l <= edge.r) dp[node]++;
        }
        return dp[node];
    }

    static class Edge {
        int to, l, r;

        Edge(int to, int l, int r) {
            this.to = to;
            this.l = l;
            this.r = r;
        }
    }
}
