MOD = 1000000007

def mod_fact(n, mod):
    result = 1
    for i in range(1, n + 1):
        result = (result * i) % mod
    return result

def mod_inv(a, mod):
    return pow(a, mod - 2, mod)

def mod_comb(n, k, mod):
    numerator = mod_fact(n, mod)
    denominator = (mod_fact(k, mod) * mod_fact(n - k, mod)) % mod
    return (numerator * mod_inv(denominator, mod)) % mod

n = int(input())
a = list(map(int, input().split()))

missing_count = a.count(-1)
fixed_count = sum(1 for i in range(n) if a[i] == i + 1)

derangement = [0] * (missing_count + 1)
derangement[0] = 1
for i in range(1, missing_count + 1):
    derangement[i] = (i - 1) * (derangement[i - 1] + derangement[i - 2]) % MOD

result = mod_comb(missing_count, fixed_count, MOD) * derangement[missing_count - fixed_count] % MOD
print(result)
