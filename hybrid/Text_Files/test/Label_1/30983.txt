import java.util.*;
import java.io.*;

public class Main {
    static ArrayList<ArrayList<Edge>> adj;
    static int[] ranges;
    static boolean[] visited;
    static int maxLoyalty;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++)
            adj.add(new ArrayList<>());
        ranges = new int[m];
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            int l = Integer.parseInt(st.nextToken());
            int r = Integer.parseInt(st.nextToken());
            ranges[i] = r;
            adj.get(a).add(new Edge(b, i));
            adj.get(b).add(new Edge(a, i));
        }
        visited = new boolean[m];
        maxLoyalty = 0;
        dfs(1, n, 0);
        if (maxLoyalty == 0)
            System.out.println("Nice work, Dima!");
        else
            System.out.println(maxLoyalty);
    }

    static void dfs(int cur, int dest, int loyalty) {
        if (cur == dest) {
            maxLoyalty = Math.max(maxLoyalty, loyalty);
            return;
        }
        for (Edge e : adj.get(cur)) {
            if (!visited[e.idx]) {
                visited[e.idx] = true;
                dfs(e.node, dest, loyalty + (ranges[e.idx] - e.node + 1));
                visited[e.idx] = false;
            }
        }
    }

    static class Edge {
        int node, idx;

        Edge(int node, int idx) {
            this.node = node;
            this.idx = idx;
        }
    }
}
