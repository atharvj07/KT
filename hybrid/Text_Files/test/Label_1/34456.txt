import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int[] times = new int[5];
        for (int i = 0; i < 5; i++) {
            times[i] = scan.nextInt();
        }
        int lastTime = 0;
        for (int time : times) {
            lastTime = ((lastTime + time - 1) / 10) * 10 + time;
        }
        System.out.println(lastTime);
    }
}
