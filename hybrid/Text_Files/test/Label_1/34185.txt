import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int[] x = new int[q], y = new int[q];
        for (int i = 0; i < q; i++) {
            x[i] = sc.nextInt() - 1;
            y[i] = sc.nextInt() - 1;
        }
        long ans = 0;
        for (int i = 0; i < q; i++) {
            int xi = x[i], yi = y[i];
            if (xi > yi) {
                int t = xi; xi = yi; yi = t;
            }
            long cnt = 0;
            for (int j = 0; j < n; j++) if (j != xi && j != yi) {
                if (a[j] > a[xi]) cnt++;
                if (a[j] > a[yi]) cnt++;
            }
            long inv = (a[xi] - 1) * (a[yi] - a[xi] - 1) + (a[yi] - 1) * (n - a[yi]) + cnt;
            ans += inv;
        }
        System.out.println(ans % (1 << 30));
    }
}
