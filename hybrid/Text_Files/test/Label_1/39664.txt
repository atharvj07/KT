import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    adj = [[] for i in range(n + 1)]
    ind = [0] * (n + 1)

    for i in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        ind[v] += 1

    ans = 0
    for i in range(1, n + 1):
        if ind[i] == len(adj[i]):
            ans += 1

    print(ans)
    for i in range(1, n + 1):
        for j in adj[i]:
            if ind[j] > 0:
                sys.stdout.write("{} {}\n".format(i, j))
                ind[j] -= 1
