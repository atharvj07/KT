import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++) a[i] = in.nextInt();
        in.close();
        int maxL = 0, l = 0, r = 0;
        int[] c = new int[k+1];
        while (r < n) {
            if (c[a[r]] == 0) {
                maxL = Math.max(maxL, r - l + 1);
                c[a[r]]++;
                r++;
            }
            else {
                c[a[l]]--;
                l++;
            }
        }
        System.out.print(maxL);
    }
}
