import java.util.Scanner;
import java.util.ArrayList;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] adj = new int[n][n];
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            adj[a][b] = 1;
            adj[b][a] = 1;
        }
        int[] dist = new int[n];
        boolean[] vis = new boolean[n];
        Arrays.fill(dist, -1);
        int[] path = new int[n * n];
        int[] par = new int[n];
        Arrays.fill(par, -1);
        int[] newpath = new int[n * n];
        int[] newpar = new int[n];
        Arrays.fill(newpar, -1);
        int t = -1;

        for (int i = 0; i < n; i++) {
            if (dist[i] != -1)
                continue;
            dist[i] = 0;
            dfs(i, -1, adj, dist, path, par);
            int min = Integer.MAX_VALUE;
            for (int j = 0; j < n; j++) {
                if (dist[j] == -1)
                    continue;
                min = Math.min(min, dist[j]);
            }
            if (t == -1 || t > min) {
                t = min;
                System.arraycopy(dist, 0, newpath, 0, n);
                System.arraycopy(par, 0, newpar, 0, n);
            }
            for (int j = 0; j < n; j++) {
                if (dist[j] != -1)
                    vis[j] = false;
            }
        }

        int[] q = new int[n * n];
        int[] newq = new int[n * n];
        int l = 0;
        int r = 0;
        int nl = 0;
        int nr = 0;
        for (int i = 0; i < n; i++) {
            if (newpath[i] == t) {
                q[r++] = i;
            }
        }
        for (int i = 0; i < t; i++) {
            while (l < r) {
                int u = q[l++];
                for (int j = 0; j < n; j++) {
                    if (adj[u][j] == 1) {
                        if (newpath[j] == t - i - 1) {
                            newq[nr++] = u;
                            newq[nr++] = j;
                            break;
                        }
                    }
                }
            }
            System.arraycopy(q, l, newq, nr, r - l);
            nr += r - l;
            r = l = 0;
            System.arraycopy(newq, 0, q, 0, nr);
            r = nr;
            nr = nl = 0;
        }

        System.out.println(t);
        for (int i = 0; i < t - 1; i++) {
            int u = newq[i * 2];
            int v = newq[i * 2 + 1];
            int w = newpar[u];
            int x = newpar[v];
            adj[u][v] = 1;
            adj[v][u] = 1;
            adj[w][x] = 0;
            adj[x][w] = 0;
            System.out.println((u + 1) + " " + (v + 1) + " " + (w + 1) + " " + (x + 1));
        }
    }

    public static void dfs(int u, int p, int[][] adj, int[] dist, int[] path, int[] par) {
        for (int i = 0; i < adj.length; i++) {
            if (adj[u][i] == 1 && i != p) {
                if (dist[i] != -1) {
                    int len = dist[u] + dist[i] + 1;
                    if (len < dist[u]) {
                        par[u] = i;
                        dist[u] = len;
                        path[u] = i;
                    } else if (len == dist[u]) {
                        par[u] = Math.min(par[u], i);
                        dist[u] = len;
                        path[u] = Math.min(path[u], i);
                    }
                    continue;
                }
                par[i] = u;
                dist[i] = dist[u] + 1;
                dfs(i, u, adj, dist, path, par);
            }
        }
    }
}
