import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] x = new int[n + 2];
        for (int i = 0; i < s.length(); ++i) {
            x[i + 1] = s.charAt(i) - '0';
        }
        for (int i = 2; i <= n; ++i) {
            for (int j = 1; j <= n + 2 - i; ++j) {
                x[j] = Math.abs(x[j] - x[j + 1]);
            }
        }
        System.out.println(x[1]);
    }
}
