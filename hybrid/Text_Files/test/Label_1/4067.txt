#include <iostream>
#include <vector>
#include <string>
#include <cstring>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<string> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    string s;
    cin >> s;
    int j = 0;
    bool ok = true;
    for (int i = 0; i < s.size() && ok;) {
        if (s[i] == '<' && s[i + 1] == '3') {
            int cnt = 0;
            i += 2;
            while (s[i] != '<' || s[i + 1] != '3') {
                if (s[i] == '<' || s[i] == '>' || s[i] == '+' || s[i] == '-') {
                    ok = false;
                }
                i++;
                cnt++;
            }
            if (cnt != v[j].size()) {
                ok = false;
            }
            j++;
            i++;
        } else {
            ok = false;
        }
    }
    cout << (ok ? "yes" : "no") << endl;
    return 0;
}
