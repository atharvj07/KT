#include <iostream>
#include <vector>
#include <queue>
using namespace std;
int main() {
    int n, x, y;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
        cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    vector<int> bfs;
    for (int i = 0; i < n; i++) {
        cin >> x;
        bfs.push_back(x);
    }
    queue<int> q;
    vector<bool> vis(n + 1, false);
    q.push(1);
    vis[1] = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (bfs[q.size()] != u) { cout << "No" << endl; return 0; }
        for (int v : adj[u]) {
            if (!vis[v]) {
                vis[v] = true;
                q.push(v);
            }
        }
    }
    cout << "Yes" << endl;
}
