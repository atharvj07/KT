import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws IOException {
        InputStreamReader r = new InputStreamReader(System.in);
        BufferedReader br = new BufferedReader(r);

        int n = Integer.parseInt(br.readLine());
        int[] top = Arrays.stream(br.readLine().split("\\s")).mapToInt(Integer::parseInt).toArray();
        int[] bottom = Arrays.stream(br.readLine().split("\\s")).mapToInt(Integer::parseInt).toArray();
        long[][] dp = new long[n][3];

        dp[0][0] = top[0];
        dp[0][1] = Math.max(top[0], bottom[0]);
        dp[0][2] = Math.max(dp[0][1], bottom[0]);
        for (int i = 1; i < n; i++) {
            dp[i][0] = Math.max(dp[i - 1][1] + top[i], dp[i - 1][2] + top[i]);
            dp[i][1] = Math.max(dp[i - 1][0] + bottom[i], dp[i - 1][2] + bottom[i]);
            dp[i][2] = Math.max(dp[i][1], dp[i][0]);
        }

        System.out.println(dp[n - 1][2]);
    }
}
