n=int(input())
s=input()
cnt=[0]*3
for c in s:cnt[(ord(c)-ord('0'))]+=1
if cnt[0]==cnt[1]==cnt[2]:print(s);exit()
a0=cnt[0]-cnt[1]
a1=cnt[1]-cnt[2]
if a1==0:
    for i in range(n):
        if a0>0 and s[i]=='2':
            s=s[:i]+'1'+s[i+1:]
            a0-=1
        elif a0==0 and s[i]=='1':
            s=s[:i]+'0'+s[i+1:]
            a1+=1
elif a1>0:
    for i in range(n):
        if a1>0 and s[i]=='2':
            s=s[:i]+'1'+s[i+1:]
            a1-=1
        elif a1==0 and s[i]=='0':
            s=s[:i]+'1'+s[i+1:]
            a0+=1
else:
    for i in range(n):
        if a0>0 and s[i]=='1':
            s=s[:i]+'2'+s[i+1:]
            a0-=1
        elif a0==0 and s[i]=='0':
            s=s[:i]+'2'+s[i+1:]
            a1+=1
print(s)
