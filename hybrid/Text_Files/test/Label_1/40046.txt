def solve_good_subset(n, m, d, edges):
    if sum(d[i] == 1 for i in range(n)) % 2 == 1:
        return -1
    answer = []
    visited = [False] * n
    edge_included = [False] * m
    adj = [[] for _ in range(n)]

    for i, (u, v) in enumerate(edges):
        adj[u-1].append((v-1, i))
        adj[v-1].append((u-1, i))

    def dfs(v, parent):
        visited[v] = True
        for to, idx in adj[v]:
            if to == parent:
                continue
            if not visited[to]:
                if dfs(to, v):
                    edge_included[idx] = not edge_included[idx]
        return d[v] == 1 or (d[v] == -1 and any(edge_included[idx] for _, idx in adj[v]))

    if not dfs(0, -1):
        return -1

    for i, include in enumerate(edge_included):
        if include:
            answer.append(i+1)

    return [len(answer)] + answer

n, m = map(int, input().split())
d = list(map(int, input().split()))
edges = [tuple(map(int, input().split())) for _ in range(m)]

result = solve_good_subset(n, m, d, edges)
if result == -1:
    print(result)
else:
    print(result[0])
    for edge in result[1:]:
        print(edge)
