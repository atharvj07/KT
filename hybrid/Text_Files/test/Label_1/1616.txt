#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, l, m, p, lst = 1, dist = 0, ans = -1;
    cin >> n;
    vector<int> a(n, 0);
    vector<int> b(n * 2 + 5, 0);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> p >> l;
        b[p] += l;
        if (b[p] == 1) {
            dist += a[p - 1] - (a[lst - 1]);
            lst = p;
        }
    }
    ans = dist + a[n - 1] - a[lst - 1];
    cout << ans << endl;
    return 0;
}
