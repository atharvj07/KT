import java.util.Scanner;

public class VasyaAndBitwiseXor {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }

        long count = 0;
        for (int i = 0; i < n; i++) {
            long xor = a[i];
            for (int j = i + 1; j < n; j++) {
                xor ^= a[j];
                if (xor == 0) {
                    count += j - i + 1;
                }
            }
        }

        System.out.println(count);
    }
}
