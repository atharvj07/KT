import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int len = s.length(), ans = 0;
        for (int i = 0; i < len; i++) {
            if (i + 1 < len && s.charAt(i) != s.charAt(i + 1)) {
                i++;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
