#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
#define mod 998244353
ll n,k,f[505][505],g[505][505],h[505][505];
ll pw(ll x, ll y){
    if(y == 0)return 1;
    if(y == 1)return x;
    ll t=pw(x,y/2);
    t=(t*t)%mod;
    if(y%2 == 1){
        t=(t*x)%mod;
    }
    return t;
}
int main(){
    cin >> n >> k;
    for(ll i=0;i<n;i++){
        f[0][i]=1;
        f[1][i]=2;
        g[0][i]=1;
        h[0][i]=1;
    }
    for(ll i=2;i<=n;i++){
        for(ll j=0;j<n;j++){
            f[i][j]=(f[i-1][j]*g[i-1][j])%mod;
            g[i][j]=(f[i-1][j]+g[i-1][j])%mod;
            h[i][j]=(g[i][j]-f[i][j])%mod;
            if(h[i][j] < 0)h[i][j]+=mod;
        }
    }
    ll ans=0;
    for(ll i=0;i<n;i++){
        for(ll j=0;j<n;j++){
            ll x=h[i][j];
            if(n-1-i >= k){
                x=(x-f[i][j]+mod)%mod;
            }
            if(n-1-j >= k){
                x=(x-h[i][j]+mod)%mod;
                x=(x*g[i][n-1])%mod;
                x=(x*g[n-1][j])%mod;
            }
            ans=(ans+x)%mod;
        }
    }
    cout << ans << endl;
}
