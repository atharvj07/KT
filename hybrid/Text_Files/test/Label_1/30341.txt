import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        int min = (int) Math.ceil((double) m / (n - 1));
        if (n % 2 == 0 && m % 2 == 1) min++;
        if (2 * n < m) System.out.println(-1);
        else System.out.println(min);
    }
}
