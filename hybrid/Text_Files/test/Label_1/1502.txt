#include<bits/stdc++.h>
using namespace std;
int main()
{
    int a,b,c,d,e,f,g,h,i,j;
    cin>>a>>b>>c>>d;
    cin>>e>>f>>g>>h;
    cin>>i>>j>>c>>d;
    if(a<e&&c>g&&b<h&&d>f){cout<<"YES";}
    else if(a<i&&c>g&&b<j&&d>f){cout<<"YES";}
    else if(i<a&&c>g&&j<d&&f<b){cout<<"YES";}
    else if(e<a&&g>c&&h<d&&f<b){cout<<"YES";}
    else{cout<<"NO";}
    return 0;
}
