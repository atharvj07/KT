import java.io.*;
import java.lang.*;
import java.util.*;

public class Main {
	public static void main(String[] args) throws Exception {
		Scanner cin = new Scanner(System.in);
		char[] s = cin.nextLine().toCharArray();
		int m = 0;
		for (char ch : s)
			if (ch == '?')
				++m;
		int[] a = new int[m];
		int[] b = new int[m];
		for (int i = 0; i < m; ++i) {
			a[i] = cin.nextInt();
			b[i] = cin.nextInt();
		}

		int[][] dp = new int[m + 1][2];
		dp[m][0] = dp[m][1] = 0;
		for (int i = m - 1; i >= 0; --i) {
			for (int bal = 0; bal <= 1; ++bal) {
				if (s[i] == ')') {
					if (bal > 0)
						dp[i][bal] = dp[i + 1][bal - 1];
					else
						dp[i][bal] = -1;
				} else if (s[i] == '(') {
					if (bal == 0)
						dp[i][bal] = dp[i + 1][bal + 1];
					else
						dp[i][bal] = -1;
				} else {
					int cur = bal == 0 ? a[i] : b[i];
					if (dp[i + 1][bal + 1] >= 0)
						dp[i][bal] = Math.min(dp[i][bal], cur + dp[i + 1][bal + 1]);
					if (bal > 0 && dp[i + 1][bal - 1] >= 0)
						dp[i][bal] = Math.min(dp[i][bal], cur + dp[i + 1][bal - 1]);
					if (dp[i][bal] == 0)
						s[i] = bal == 0 ? '(' : ')';
					else if (dp[i][bal] < 0)
						s[i] = '!';
				}
			}
		}

		if (s[0] == '!')
			System.out.println(-1);
		else {
			System.out.println(dp[0][0]);
			System.out.println(new String(s));
		}
	}
}
