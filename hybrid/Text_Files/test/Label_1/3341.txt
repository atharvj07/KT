#include <bits/stdc++.h>
#define nl '\n'
using namespace std;

int main() {
  int t; cin >> t;
  while (t--) {
    string s; cin >> s;
    int n = s.length();
    vector<int> a;
    int cur = 0;
    char last = '0';
    for (int i = 0; i < n; ++i) {
      if (s[i] == '1') {
        if (last == '0') {
          a.push_back(i);
          last = '1';
        }
      }
      else {
        last = '0';
      }
    }
    if (a.size() <= 1) cout << "NO" << nl;
    else {
      bool sorted = true;
      for (int i = 1; i < a.size(); ++i) {
        if (s[a[i]] < s[a[i - 1]]) sorted = false;
      }
      if (sorted) cout << "NO" << nl;
      else cout << "YES" << nl;
    }
  }
}
