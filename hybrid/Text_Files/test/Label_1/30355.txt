import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        int k = input.nextInt();
        int[] colors = new int[n];
        for (int i = 0; i < n; i++) {
            colors[i] = input.nextInt();
        }
        long[][] dp = new long[n + 1][k + 1];
        dp[n][0] = 1;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = dp[i + 1][j];
                if (j > 0) {
                    dp[i][j] = (dp[i][j] + dp[i + 1][j - 1]) % 1009;
                }
                if (j > 0 && colors[i] <= m) {
                    dp[i][j] = (dp[i][j] + dp[i + 1][j - 1]) % 1009;
                }
            }
        }
        System.out.println(dp[0][k]);
    }
}
