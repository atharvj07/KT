#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
using namespace std;
using namespace __gnu_pbds;
typedef long long ll;
typedef pair<int, int> pii;
typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> ordered_set;
const int MAXN = 1e4 + 10;

int n, m, q;
char g[MAXN][MAXN];
int l[MAXN][MAXN];
int d[MAXN][MAXN];
bool vis[MAXN][MAXN];
inline bool valid(int x, int y) {
    return 1 <= x && x <= n && 1 <= y && y <= m;
}
int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) {
        scanf("%s", g[i] + 1);
        for (int j = 1; j <= m; j++) {
            l[i][j] = (g[i][j] == 'X' ? 0 : 1e9);
            d[i][j] = (g[i][j] == 'X' ? 0 : 1e9);
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (g[i][j] == '.') {
                if (valid(i - 1, j)) l[i][j] = min(l[i][j], l[i - 1][j] + 1);
                if (valid(i, j - 1)) l[i][j] = min(l[i][j], l[i][j - 1] + 1);
            }
        }
    }
    for (int i = n; i >= 1; i--) {
        for (int j = m; j >= 1; j--) {
            if (g[i][j] == '.') {
                if (valid(i + 1, j)) d[i][j] = min(d[i][j], d[i + 1][j] + 1);
                if (valid(i, j + 1)) d[i][j] = min(d[i][j], d[i][j + 1] + 1);
            }
        }
    }
    scanf("%d", &q);
    while (q--) {
        int a, b;
        scanf("%d%d", &a, &b);
        int mx_l = 0;
        int mx_r = 0;
        for (int i = 1; i <= n; i++) {
            mx_l = max({mx_l, l[i][a], d[i][b]});
            mx_r = max({mx_r, l[i][b], d[i][a]});
        }
        if (mx_l + mx_r <= n)
            puts("YES");
        else
            puts("NO");
    }
    return 0;
}
