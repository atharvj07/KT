import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int[] c = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        for (int i = 0; i < n - 1; i++) b[i] = sc.nextInt();
        for (int i = 0; i < n - 2; i++) c[i] = sc.nextInt();
        for (int i = 0; i < n; i++) {
            boolean found = false;
            for (int j = 0; j < n - 1; j++) {
                if (a[i] == b[j]) {
                    found = true;
                }
            }
            for (int j = 0; j < n - 2; j++) {
                if (a[i] == c[j]) {
                    found = true;
                }
            }
            if (!found) {
                System.out.println(a[i]);
                break;
            }
        }
        for (int i = 0; i < n - 1; i++) {
            boolean found = false;
            for (int j = 0; j < n - 2; j++) {
                if (b[i] == c[j]) {
                    found = true;
                }
            }
            if (!found) {
                System.out.println(b[i]);
                break;
            }
        }
    }
}
