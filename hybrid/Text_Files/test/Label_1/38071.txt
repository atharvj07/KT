n = int(input())
s = input()
ans = n*2
for i in range(len(s)):
    for j in range(i+1, len(s)):
        for k in range(j+1, len(s)):
            for l in range(k+1, len(s)):
                t = s[i:j+1]+s[k:l+1]
                if "ACTG" in t:
                    ans = min(ans, (j-i+1)+(l-k+1)-4)
print(ans)
