import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt();
        List<Integer> a = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            a.add(in.nextInt());
        }
        Collections.sort(a);
        int ans = 0;
        for (int i = 1; i < m; i++) {
            if (a.get(i) != a.get(i - 1)) {
                ans++;
            }
        }
        System.out.println(ans + 1);
        for (int i = 0; i < m; i++) {
            if (i > 0 && a.get(i) == a.get(i - 1)) {
                continue;
            }
            System.out.print(a.get(i) + " ");
        }
        System.out.println();
    }
}
