#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAX_OPS = 1e7 + 5;
typedef pair<int, int> pii;

int n, k, m;
int op[MAX_OPS];
vector<pii> v;

int main() {
    cin >> n >> k >> m;
    int a, cnt = 0;
    for (int i = 0; i < n; ++i) {
        cin >> a;
        v.push_back({a, cnt});
        op[cnt]++;
        if (op[cnt] == k) ++cnt;
    }

    sort(v.begin(), v.end());
    for (int i = 0; i < n && m; ++i) {
        int mx = v.back().first;
        if (mx == v[i].first) continue;
        m -= mx - v[i].first;
        op[v[i].second]--, op[cnt]++;
        if (op[cnt] == k) ++cnt;
        v[i].first = mx;
    }

    long long sum = 0;
    for (const auto& p : v) sum += p.first;
    long double ans = (long double) sum / n;
    cout << ans << endl;
    return 0;
}
