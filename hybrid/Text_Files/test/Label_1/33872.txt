import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int diff = a[n - 1] - a[0];
        for (int i = 1; i < n - 1; i++) {
            diff = Math.min(diff, a[i + 1] - a[i - 1]);
        }
        System.out.println(diff);
    }
}
