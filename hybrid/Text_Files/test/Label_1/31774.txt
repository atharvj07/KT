import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = sc.nextInt();
            }
            boolean[] happy = new boolean[n];
            for (int k = 1; k <= n; k++) {
                boolean isPermutation = true;
                Set<Integer> seen = new HashSet<>();
                for (int j = 0; j < n - k + 1; j++) {
                    int min = Integer.MAX_VALUE;
                    for (int l = j; l < j + k; l++) {
                        min = Math.min(min, a[l]);
                    }
                    if (seen.contains(min)) {
                        isPermutation = false;
                        break;
                    }
                    seen.add(min);
                }
                happy[k - 1] = isPermutation;
            }
            for (boolean h : happy) {
                System.out.print(h ? 1 : 0);
            }
            System.out.println();
        }
        sc.close();
    }
}
