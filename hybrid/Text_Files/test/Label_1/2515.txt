#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> a(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i].first >> b[i].second;
    }
    int k = 0;
    while (2 * k + 2 * (n - 2 * k) <= n) {
        k++;
    }
    k--;
    for (int i = 0; i < n; i++) {
        if (i < 2 * k) {
            cout << '1';
        } else {
            if (a[i].first < a[2 * k].first) {
                cout << '1';
            } else {
                cout << '0';
            }
        }
    }
    cout << '\n';
    for (int i = 0; i < n; i++) {
        if (i < 2 * k) {
            cout << '1';
        } else {
            if (b[i].first < b[2 * k].first) {
                cout << '1';
            } else {
                cout << '0';
            }
        }
    }
    cout << '\n';
    return 0;
}
