def has_solution(ax, ay, bx, by, cx, cy):
    # Check if the distances between a-b and b-c are equal
    # and if the midpoint between a and c is the same as b
    if (ax - bx)**2 + (ay - by)**2 == (bx - cx)**2 + (by - cy)**2 and (ax + cx) / 2 == bx and (ay + cy) / 2 == by:
        return "Yes"
    else:
        return "No"

ax, ay, bx, by, cx, cy = map(int, input().split())
print(has_solution(ax, ay, bx, by, cx, cy))
