n, a, b = map(int, input().split())
arr = list(input())
cnt = 0
for i in range(n):
    if arr[i] == '.':
        if i > 0 and arr[i - 1] == arr[i]:
            continue
        if i + 1 < n and arr[i + 1] == arr[i]:
            continue
        if arr[i] == '*':
            continue
        cnt += 1
        if cnt <= a:
            arr[i] = 'A'
        else:
            arr[i] = 'B'
print(cnt)
