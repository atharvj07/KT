import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        Map<Point, Integer> points = new HashMap<>();
        for (int i = 0; i < 8; i++) {
            Point point = new Point(scanner.nextInt(), scanner.nextInt(), scanner.nextInt());
            points.put(point, i);
        }

        if (!isCube(points)) {
            System.out.println("NO");
            return;
        }

        List<Point> vertices = new ArrayList<>(points.keySet());
        int[] permutation = new int[8];
        for (int i = 0; i < 8; i++) {
            permutation[i] = points.get(vertices.get(i));
        }

        System.out.println("YES");
        for (int[] vertex : getVertices(permutation)) {
            System.out.println(vertex[0] + " " + vertex[1] + " " + vertex[2]);
        }
    }

    private static boolean isCube(Map<Point, Integer> points) {
        if (points.size() != 8) {
            return false;
        }

        List<Point> vertices = new ArrayList<>(points.keySet());
        for (Point vertex : vertices) {
            if (vertex.x < 0 || vertex.y < 0 || vertex.z < 0) {
                return false;
            }

            for (Point otherVertex : vertices) {
                if (vertex.equals(otherVertex)) {
                    continue;
                }

                if (vertex.distanceTo(otherVertex) != 1) {
                    return false;
                }
            }
        }

        return true;
    }

    private static int[][] getVertices(int[] permutation) {
        int[][] vertices = new int[8][3];
        for (int i = 0; i < 8; i++) {
            vertices[i] = getVertex(permutation, i);
        }

        return vertices;
    }

    private static int[] getVertex(int[] permutation, int index) {
        int[] vertex = new int[3];
        for (int i = 0; i < 3; i++) {
            vertex[i] = permutation[index * 3 + i];
        }

        Arrays.sort(vertex);
        return vertex;
    }

    private static class Point {
        private int x;
        private int y;
        private int z;

        public Point(int x, int y, int z) {
            this.x = x;
            this.y = y;
            this.z = z;
        }

        public int distanceTo(Point other) {
            int dx = Math.abs(this.x - other.x);
            int dy = Math.abs(this.y - other.y);
            int dz = Math.abs(this.z - other.z);

            return dx + dy + dz;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Point point = (Point) o;
            return x == point.x && y == point.y && z == point.z;
        }

        @Override
        public int hashCode() {
            int result = x;
            result = 31 * result + y;
            result = 31 * result + z;
            return result;
        }
    }
}
