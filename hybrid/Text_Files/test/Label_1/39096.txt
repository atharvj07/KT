import numpy as np

g = np.zeros((5, 5), dtype=int)
for i in range(5):
    g[i, :] = list(map(int, input().split()))

t = np.zeros((5, 5), dtype=int)
for i in range(1, 5):
    for j in range(i, 5):
        t[i, j] = t[j, i] = g[i - 1, j - 1] + g[j - 1, i - 1]

ans = 0
for perm in range(120):
    s = perm
    cur = 0
    res = 0
    for _ in range(5):
        nxt = s % 5
        res += t[cur, nxt]
        s //= 5
        cur = nxt

    ans = max(ans, res)

print(ans)
