import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Scanner;

public class DancingParty {

    private static class Hos {
        int weight;
        int beauty;

        Hos(int weight, int beauty) {
            this.weight = weight;
            this.beauty = beauty;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int w = scanner.nextInt();

        Hos[] hoses = new Hos[n];
        for (int i = 0; i < n; i++) {
            hoses[i] = new Hos(scanner.nextInt(), scanner.nextInt());
        }

        List<List<Integer>> friendshipGroups = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;

            // Find the group containing x
            int groupX = -1;
            for (int j = 0; j < friendshipGroups.size(); j++) {
                if (friendshipGroups.get(j).contains(x)) {
                    groupX = j;
                    break;
                }
            }

            // Create a new group if x is not in any group
            if (groupX == -1) {
                groupX = friendshipGroups.size();
                friendshipGroups.add(new ArrayList<>());
            }

            // Add y to the group of x
            friendshipGroups.get(groupX).add(y);

            // Find the group containing y
            int groupY = -1;
            for (int j = 0; j < friendshipGroups.size(); j++) {
                if (friendshipGroups.get(j).contains(y)) {
                    groupY = j;
                    break;
                }
            }

            // Merge the group of y into the group of x
            if (groupY != -1 && groupY != groupX) {
                friendshipGroups.get(groupX).addAll(friendshipGroups.get(groupY));
                friendshipGroups.remove(groupY);
            }
        }

        // Calculate the maximum total beauty
        int maxBeauty = 0;
        for (List<Integer> group : friendshipGroups) {
            int groupWeight = 0;
            int groupBeauty = 0;
            for (int hosIndex : group) {
                groupWeight += hoses[hosIndex].weight;
                groupBeauty += hoses[hosIndex].beauty;
            }

            // Choose all hoses from the group if their total weight doesn't exceed w
            if (groupWeight <= w) {
                maxBeauty += groupBeauty;
            } else {
                // Choose the hose with the maximum beauty from the group
                int maxHosBeauty = 0;
                for (int hosIndex : group) {
                    maxHosBeauty = Math.max(maxHosBeauty, hoses[hosIndex].beauty);
                }
                maxBeauty += maxHosBeauty;
            }
        }

        System.out.println(maxBeauty);
    }
}
