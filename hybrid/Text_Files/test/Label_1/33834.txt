import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int speed = 0, speedLimit = 0, minSigns = 0;
        boolean canOvertake = true;
        for (int i = 0; i < n; i++) {
            int type = in.nextInt();
            switch (type) {
                case 1:
                    speed = in.nextInt();
                    break;
                case 2:
                    if (!canOvertake)
                        minSigns++;
                    break;
                case 3:
                    speedLimit = in.nextInt();
                    break;
                case 4:
                    canOvertake = true;
                    break;
                case 5:
                    speedLimit = 0;
                    break;
                case 6:
                    canOvertake = false;
                    break;
            }
            if (speed > speedLimit && speedLimit != 0)
                minSigns++;
        }
        System.out.println(minSigns);
    }
}
