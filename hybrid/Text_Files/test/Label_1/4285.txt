#include <iostream>
#include <vector>
#include <numeric>

using namespace std;

int main() {
    long long n;
    cin >> n;
    vector<int> v(n);
    for (long long i = 0; i < n; i++) {
        char x;
        cin >> x;
        v[i] = (x == 'a');
    }
    const long long mod = 1000000007;
    partial_sum(v.begin(), v.end(), v.begin());
    long long k = 0;
    long long ans = 0;
    for (long long i = 0; i < n; i++) {
        if (v[i] == 0)
            continue;
        if (k == 0)
            k = i + 1;
        else {
            ans += k * i + 2 * (v[i] - v[k - 1]);
            ans %= mod;
            k = i + 1;
        }
    }
    if (k > 0) {
        ans += k * n + 2 * (v[n - 1] - v[k - 1]);
        ans %= mod;
    }
    cout << ans << endl;
}
