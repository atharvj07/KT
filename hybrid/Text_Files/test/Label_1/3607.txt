#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<bool> road(n + 1, true);
  for (int i = 1; i <= n; i++) {
    char c;
    std::cin >> c;
    road[i] = c == '.';
  }

  std::vector<bool> dp(n + 1, false);
  dp[1] = true;
  for (int i = 1; i <= n; i++) {
    if (dp[i]) {
      for (int j = 1; j <= k && i + j <= n; j++) {
        if (road[i + j]) {
          dp[i + j] = true;
        }
      }
    }
  }

  if (dp[n]) {
    std::cout << "YES" << std::endl;
  } else {
    std::cout << "NO" << std::endl;
  }

  return 0;
}
