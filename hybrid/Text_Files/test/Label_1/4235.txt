#include <iostream>
#include <vector>
using namespace std;
const int MAX_N = 1e4;
int dp[MAX_N + 5];
vector<vector<int>> factors(MAX_N + 5);

int solve(int n) {
  if (n == 0 || n == 1) return -1;
  if (n == 2) return -1;
  if (dp[n] != -1) return dp[n];
  int ans = -1;
  for (int i = 2; i * i <= n; ++i) {
    if (n % i == 0) {
      int factor = n / i;
      vector<int> col_cnt(26), row_cnt(26);
      for (char ch = 'a'; ch <= 'u'; ++ch) {
        col_cnt[ch - 'a'] = (factor > 2);
        row_cnt[ch - 'a'] = (i > 2);
      }
      for (int j = 0; j < factors[factor].size(); ++j) {
        char ch = factors[factor][j] + 'a';
        if (ch >= 'a' && ch <= 'u') {
          row_cnt[ch - 'a'] += row_cnt[ch - 'a'] == 0;
          col_cnt[ch - 'a'] += col_cnt[ch - 'a'] == 0;
        }
      }
      for (int j = 0; j < factors[i].size(); ++j) {
        char ch = factors[i][j] + 'a';
        if (ch >= 'a' && ch <= 'u') {
          row_cnt[ch - 'a'] += row_cnt[ch - 'a'] == 0;
          col_cnt[ch - 'a'] += col_cnt[ch - 'a'] == 0;
        }
      }
      bool flag = true;
      for (int j = 0; j < 26; ++j) {
        if (row_cnt[j] == 0 || col_cnt[j] == 0) {
          flag = false;
        }
      }
      if (flag) {
        int res = solve(factor) * solve(i);
        if (res != -1) ans = res;
      }
    }
  }
  return dp[n] = ans;
}

int main() {
  for (int i = 1; i <= MAX_N; ++i) {
    for (int j = 1; j * j <= i; ++j) {
      if (i % j == 0) {
        factors[i].push_back(j);
        if (i / j != j) factors[i].push_back(i / j);
      }
    }
  }
  fill(dp, dp + MAX_N + 5, -1);
  dp[0] = 1;
  dp[1] = 1;
  dp[2] = 1;
  int n;
  cin >> n;
  int ans = solve(n);
  if (ans == -1) {
    cout << -1 << endl;
  } else {
    cout << ans << endl;
  }
  return 0;
}
