import java.io.*;
import java.util.*;

public class Main {
    static long[] ages;
    static int[] friends;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        ages = new long[n];
        String[] s = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            ages[i] = Long.parseLong(s[i]);
        }

        // Find friendships using bitwise AND
        friends = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if ((ages[i] & ages[j]) == 0) {
                    friends[i]++;
                    friends[j]++;
                }
            }
        }

        // Sort ages in descending order
        Arrays.sort(ages);

        // Initialize maximum gainings
        long maxGainings = 0;

        // Find maximum gainings by inviting friends
        for (int i = 0; i < n; i++) {
            if (friends[i] > 0) {
                maxGainings += ages[i] * friends[i];
            }
        }

        // Print maximum gainings
        System.out.println(maxGainings);
    }
}
