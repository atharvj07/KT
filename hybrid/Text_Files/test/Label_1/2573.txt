#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += s[i] == 'F';
  }
  for (int i = 0; i < n; i++) {
    if (s[i] == 'F' && i <= sum) cnt++;
  }
  cout << (cnt > sum - cnt ? "YES" : "NO") << endl;
  return 0;
}
