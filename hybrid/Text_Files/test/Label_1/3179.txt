#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    if (a[0] == a[n-1]) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    std::vector<int> ans(n, 0);
    int cur = 1;
    for (int i = 0; i < n; i++) {
        if (i > 0 && a[i] == a[i-1]) {
            ans[i] = ans[i-1];
        } else {
            ans[i] = cur++;
        }
    }
    if (cur > k) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    std::cout << "YES" << std::endl;
    for (int i = 0; i < n; i++) {
        std::cout << ans[i] << " ";
    }
    std::cout << std::endl;
    return 0;
}
