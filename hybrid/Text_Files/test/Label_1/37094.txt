MOD = 10**9 + 7

def power(x, y, p):
    res = 1  
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def find_cycles(n, roads):
    visited = [False] * n
    cycles_length = []
    for i in range(n):
        if not visited[i]:
            length = 0
            current = i
            while not visited[current]:
                visited[current] = True
                current = roads[current] - 1
                length += 1
            if length > 0:
                cycles_length.append(length)
    return cycles_length

def solve(n, roads):
    cycles_length = find_cycles(n, roads)
    answer = 1
    sum_cycles_length = 0
    for length in cycles_length:
        answer = (answer * (power(2, length, MOD) - 2)) % MOD
        sum_cycles_length += length
    answer = (answer * power(2, n - sum_cycles_length, MOD)) % MOD
    return answer

n = int(input().strip())
roads = list(map(int, input().strip().split()))
print(solve(n, roads))
