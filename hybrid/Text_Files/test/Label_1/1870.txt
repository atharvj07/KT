#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int MOD = 998244353;

int main(){
    int n; cin >> n;
    unordered_map<ll, int> mp;
    for(int i = 0; i < n; i++){
        int a, b; cin >> a >> b;
        ll val = (ll)a * 1e9 + b;
        mp[val]++;
    }
    ll ans = 1;
    int mx = 0;
    for(auto x : mp){
        ans = (ans * 1ll * x.second) % MOD;
        mx = max(mx, x.second);
    }
    if(mx > 1) ans = 0;
    cout << ans << endl;
}
