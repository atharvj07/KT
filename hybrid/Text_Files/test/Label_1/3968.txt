#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, b, cnt = 0, ans = 1e9;
  cin >> n >> a >> b;
  priority_queue<pair<int, int>> pq;
  for (int i = 1, h; i <= n; i++) {
    cin >> h;
    pq.push({h, i});
  }
  while (!pq.empty()) {
    if (a >= b)
      a -= pq.top().first, pq.pop(), cnt++;
    else
      b -= pq.top().first, pq.pop(), cnt--;
  }
  if (cnt <= 0)
    ans = 0;
  else
    for (int i = 1; i <= abs(cnt); i++) ans += i;
  cout << ans << '\n';
  return 0;
}
