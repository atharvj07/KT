import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int q = in.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < q; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            System.out.println(isReachable(a, x, y) ? "Shi" : "Fou");
        }
    }

    public static boolean isReachable(int[] a, int x, int y) {
        for (int i = x; i < y; i++) {
            if ((a[i] & a[i + 1]) == 0) {
                return false;
            }
        }
        return true;
    }
}
