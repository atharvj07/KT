n = int(input())
revealed = input()
words = set([input() for i in range(int(input()))])
h = [0 for i in range(26)]
for i in range(n):
    if revealed[i] == '*':
        for j in range(26):
            h[j] += (chr(ord('a') + j) in words and chr(ord('a') + j) == revealed[i - 1])
ans = 0
for i in range(26):
    if h[i] == 1:
        ans += 1
print(ans)
