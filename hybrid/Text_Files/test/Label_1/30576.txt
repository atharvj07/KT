import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int l = 0, r = n - 1;
        while (l < r && k > 0) {
            if (a[l] == a[r]) {
                l++;
                r--;
            } else if (a[l] < a[r]) {
                a[l]++;
                ans++;
                l++;
            } else {
                a[r]--;
                ans++;
                r--;
            }
            k--;
        }
        System.out.println(ans);
    }
}
