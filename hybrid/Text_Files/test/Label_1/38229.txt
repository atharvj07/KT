n, m = map(int, input().split())
pairs = [tuple(map(int, input().split())) for _ in range(m)]

def check(pairs, x):
    candidates = set()
    for a, b in pairs:
        if a != x and b != x:
            candidates.add(a)
            candidates.add(b)
    for y in candidates:
        if all(x in (a, b) or y in (a, b) for a, b in pairs):
            return True
    return False

a, b = pairs[0]
if check(pairs, a) or check(pairs, b):
    print("YES")
else:
    print("NO")
