#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  string t;
  cin >> n >> t;

  vector<int> divisors;
  for (int i = n; i >= 1; i--) {
    if (n % i == 0) {
      divisors.push_back(i);
    }
  }

  for (int i = divisors.size() - 1; i >= 0; i--) {
    int d = divisors[i];
    reverse(t.begin(), t.begin() + d);
  }

  cout << t << endl;

  return 0;
}
