import java.util.*;

class Main {
    static long solve(String s) {
        int n = s.length();
        long[] a = new long[n];
        char[] b = s.toCharArray();
        int depth = 0;
        for (int i = 0; i < n; i++) {
            if (b[i] == '(')
                depth++;
            else if (b[i] == ')')
                depth--;
            else
                a[i] = -1;
        }
        if (depth != 0)
            return -1;
        a[n - 1] = 1;
        for (int i = n - 2; i >= 0; i--) {
            if (a[i] == -1)
                continue;
            a[i] = a[i + 1] * 2 % 998244353;
        }
        long ans = 0;
        for (long l : a)
            ans += l;
        return ans % 998244353;
    }

    public static void main(String[] Args) {
        Scanner in = new Scanner(System.in);
        System.out.println(solve(in.nextLine()));
    }
}
