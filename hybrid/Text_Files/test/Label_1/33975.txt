import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();

        // Prefix sums for fast computation
        long[] prefixXor = new long[n];
        long[] prefixSum = new long[n];
        for (int i = 0; i < n; i++) {
            if (i > 0) {
                prefixXor[i] = prefixXor[i - 1];
                prefixSum[i] = prefixSum[i - 1];
            }
            prefixXor[i] ^= a[i];
            prefixSum[i] += a[i];
        }

        int ans = 0;
        for (int i = 1; i < n - 1; i++) {
            long leftXor = prefixXor[i - 1];
            long rightXor = prefixXor[n - 1] ^ prefixXor[i];
            long leftSum = prefixSum[i - 1];
            long rightSum = prefixSum[n - 1] - prefixSum[i];
            if (leftXor == rightSum) ans++;
            else if (rightXor == leftSum) ans++;
        }
        System.out.println(ans);
    }
}
