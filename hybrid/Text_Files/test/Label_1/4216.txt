#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int M=62,N=120010;
int n,p[M],c,ans=N;

int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        int x;cin>>x;
        for(int j=0;j<M;j++)if(x>>j&1)p[j]++,c|=(1<<j);
    }
    for(int i=0;i<M;i++){
        int m=p[i],num=c^1<<i;
        if(m&1)ans=0;
        m/=2;
        for(int j=0;j<M;j++)if(j!=i&&((1<<j)&num))m+=p[j]/2;
        ans=min(ans,m);
    }
    cout<<ans<<endl;
    return 0;
}
