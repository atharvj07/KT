def min_steps(t, test_cases):
    results = []
    for i in range(t):
        n, k = test_cases[i][0]
        l1, r1 = test_cases[i][1]
        l2, r2 = test_cases[i][2]

        if l1 > l2:
            l1, r1, l2, r2 = l2, r2, l1, r1

        intersection = max(0, min(r1, r2) - l2)
        initial_intersection = intersection * n

        if initial_intersection >= k:
            results.append(0)
            continue

        k -= initial_intersection

        # Calculate increase within and outside the current range
        inside = min(r2, r1) - l2
        outside = max(r2, r1) - l1 - inside

        steps = float('inf')
        for i in range(1, n + 1):
            current_steps = outside * (i - 1) + 2 * (k - inside * i)
            if k <= inside * i:
                current_steps = min(current_steps, k - inside * i + i)
            steps = min(steps, current_steps)

        results.append(steps)

    return results

# Reading input
t = int(input())
test_cases = []
for _ in range(t):
    n_k = list(map(int, input().split()))
    l1_r1 = list(map(int, input().split()))
    l2_r2 = list(map(int, input().split()))
    test_cases.append((n_k, l1_r1, l2_r2))

# Solving
results = min_steps(t, test_cases)

# Printing output
for result in results:
    print(result)
