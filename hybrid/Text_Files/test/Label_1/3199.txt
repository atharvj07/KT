#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    int a[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> a[i][j];
      }
    }
    int ans = 0;
    for (int i = 0; i < n / 2; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] != a[n - 1 - i][j]) {
          ans++;
        }
      }
    }
    for (int i = 0; i < m / 2; i++) {
      if (a[n / 2][i] != a[n / 2][m - 1 - i]) {
        ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
