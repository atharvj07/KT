def are_equivalent(a, b):
    if a == b:
        return True
    if len(a) % 2:
        return False
    mid = len(a) // 2
    a1, a2 = a[:mid], a[mid:]
    b1, b2 = b[:mid], b[mid:]
    return (are_equivalent(a1, b1) and are_equivalent(a2, b2)) or (are_equivalent(a1, b2) and are_equivalent(a2, b1))

a = input().strip()
b = input().strip()

if are_equivalent(a, b):
    print("YES")
else:
    print("NO")
