from collections import defaultdict
import sys

def lcp(s, t):
    mn = min(len(s), len(t))
    for i in range(mn):
        if s[i] != t[i]:
            return i
    return mn

n = int(input())
names = [input() for _ in range(n)]
ps = [input() for _ in range(n)]
ind_name = defaultdict(list)
ind_ps = defaultdict(list)

for i in range(n):
    ind_name[names[i]].append(i)
    ind_ps[ps[i]].append(i)

ans = -1
name = -1
ps_ = -1
mn = []
for i in range(n):
    t = []
    for j in range(n):
        if ind_name[names[i]] and ind_ps[ps[j]]:
            t.append((lcp(names[i], ps[j]), j))
    t.sort()
    mn.append(t[-1])
    if len(t) > 0:
        if t[-1][0] > ans:
            ans = t[-1][0]
            name = i
            ps_ = t[-1][1]

print(ans * n)
for i in range(n):
    for nm, lst in ind_name.items():
        if i in lst:
            ind_name[nm].remove(i)
    for nm, lst in ind_ps.items():
        if i in lst:
            ind_ps[nm].remove(i)
    for j in range(n):
        if ind_name[names[name]] and ind_ps[ps[j]]:
            print(name + 1, j + 1)
            ind_name[names[name]].remove(name)
            ind_ps[ps[j]].remove(j)
            break
