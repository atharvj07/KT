#include <iostream>
#include <vector>
int main() {
  long long n;
  std::cin >> n;
  if ((n - 1) % n != 0) {
    std::cout << "NO";
    return 0;
  };
  std::cout << "YES" << std::endl;
  std::vector<std::pair<long long, long long>> v;
  for (long long i = 2; i < n; ++i) {
    if (n % i == 0) {
      v.push_back({i - 1, i});
    };
  };
  size_t k = v.size();
  std::cout << k << std::endl;
  for (size_t i = 0; i < k; ++i) { std::cout << v[i].first << ' ' << v[i].second
                                            << std::endl; };
  return 0;
}
