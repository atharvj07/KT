#include <bits/stdc++.h>
using namespace std;

int n, m, k, mp[110][110][110], vis[110][110][110], res=0;
int dx[] = {1,0,0}, dy[] = {0,1,0}, dz[] = {0,0,1};

void dfs(int x, int y, int z) {
	if (vis[x][y][z]) return;
	vis[x][y][z] = 1;
	for (int i=0;i<3;i++) {
		int ax = x+dx[i], ay = y+dy[i], az = z+dz[i];
		if (ax>=1 && ax<=n && ay>=1 && ay<=m && az>=1 && az<=k && mp[ax][ay][az]) {
			dfs(ax, ay, az);
		}
	}
}

int main() {
	cin >> n >> m >> k;
	for (int i=1;i<=n;i++) {
		for (int j=1;j<=m;j++) {
			for (int t=1;t<=k;t++) {
				char x; cin >> x;
				if (x=='1') {
					mp[i][j][t] = 1;
				}
			}
		}
		cin.get();
	}
	for (int i=1;i<=n;i++) {
		for (int j=1;j<=m;j++) {
			for (int t=1;t<=k;t++) {
				if (mp[i][j][t] && !vis[i][j][t]) {
					dfs(i,j,t);
				}
			}
		}
	}
	for (int i=1;i<=n;i++) {
		for (int j=1;j<=m;j++) {
			for (int t=1;t<=k;t++) {
				if (mp[i][j][t]) {
					mp[i][j][t] = 0;
					memset(vis,0,sizeof vis);
					dfs(i,j,t);
					bool flag = 0;
					for (int i=1;i<=n;i++) {
						for (int j=1;j<=m;j++) {
							for (int t=1;t<=k;t++) {
								if (mp[i][j][t] && !vis[i][j][t]) {
									flag = 1;
								}
							}
						}
					}
					if (flag) res++;
					mp[i][j][t] = 1;
				}
			}
		}
	}
	cout << res << '\n';
	return 0;
}
