import java.util.*;

public class Main {
    static boolean checkmate(String[] pieces) {
        // Check if black king is in check
        if (isCheck(pieces[2], pieces[0], pieces[1])) {
            // If there is no rook in the same row or column as black king
            if (!isRookInSameLine(pieces[2], pieces[0], pieces[1])) {
                return true;
            }
            // Check if there is a rook that can protect the king
            for (String piece : pieces) {
                if (piece.equals(pieces[2])) continue;
                if (isRookInSameLine(piece, pieces[0], pieces[1])) {
                    return false;
                }
            }
            return true;
        }
        return false;
    }

    static boolean isCheck(String king, String rook1, String rook2) {
        return isSameRank(king, rook1) || isSameFile(king, rook1) || isSameRank(king, rook2) || isSameFile(king, rook2);
    }

    static boolean isSameRank(String piece1, String piece2) {
        return piece1.charAt(1) == piece2.charAt(1);
    }

    static boolean isSameFile(String piece1, String piece2) {
        return piece1.charAt(0) == piece2.charAt(0);
    }

    static boolean isRookInSameLine(String rook, String rook1, String rook2) {
        return isSameRank(rook, rook1) || isSameFile(rook, rook1) || isSameRank(rook, rook2) || isSameFile(rook, rook2);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] pieces = scanner.nextLine().split(" ");
        System.out.println(checkmate(pieces) ? "CHECKMATE" : "OTHER");
    }
}
