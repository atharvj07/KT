t = int(input())
for _ in range(t):
    n = int(input())
    tasks = input()
    seen = set()
    prev_task = ""
    suspicious = False
    for task in tasks:
        if task != prev_task:
            if task in seen:
                suspicious = True
                break
            seen.add(task)
        prev_task = task
    print("NO" if suspicious else "YES")
