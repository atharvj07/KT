import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            double D = sc.nextDouble();
            if (D == 0.0) {
                break;
            }
            double px = sc.nextDouble();
            double py = sc.nextDouble();
            double vx = sc.nextDouble();
            double vy = sc.nextDouble();
            if (isInCircle(D, px, py, vx, vy) || isInCircle(D, px, -py, vx, -vy)) {
                System.out.println("0.00000000");
                continue;
            }
            double t = (1.0 * px * vx + 1.0 * py * vy) / (vx * vx + vy * vy);
            double ux = vx - 2.0 * t * vx;
            double uy = vy - 2.0 * t * vy;
            if (isInCircle(D, px + t * vx, py + t * vy, ux, uy) || isInCircle(D, px + t * vx, -py - t * vy, ux, -uy)) {
                System.out.printf("%.9f\n", t);
                continue;
            }
            System.out.println("impossible");
        }
    }

    private static boolean isInCircle(double D, double x, double y, double vx, double vy) {
        double t = (D - Math.hypot(x, y)) / Math.hypot(vx, vy);
        return t >= 0.0 && t * vx + x <= 0.0 && t * vy + y <= 0.0;
    }
}
