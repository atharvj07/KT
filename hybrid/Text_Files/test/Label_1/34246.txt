import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    static final int MOD = 1000000007;
    static int[][] dp;
    static List<Integer>[] sub;
    static Map<Integer, Integer> cnt;
    static int[] prv;
    static int[] nxt;
    static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String s = in.readLine();
        String t = in.readLine();
        n = s.length();
        sub = new List[n];
        for (int i = 0; i < n; i++)
            sub[i] = new ArrayList<>();
        prv = new int[n];
        nxt = new int[n];
        Arrays.fill(prv, -1);
        Arrays.fill(nxt, n);
        dp = new int[n][t.length() + 1];
        cnt = new HashMap<>();
        for (int i = 0; i < t.length(); i++) {
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) != t.charAt(i))
                    continue;
                if (prv[j] != -1)
                    sub[prv[j]].add(j);
                prv[j] = j;
                nxt[j] = i + 1;
            }
        }
        int res = 0;
        for (int j = 0; j < n; j++) {
            dp(j, t.length());
            res = (res + dp[j][t.length()]) % MOD;
        }
        System.out.println(res);
    }

    static int dp(int i, int j) {
        if (dp[i][j] != 0)
            return dp[i][j];
        if (j == 0) {
            dp[i][j] = 1;
            return 1;
        }
        int cur = 0;
        if (prv[i] != -1 &&nxt[prv[i]] >= j)
            cur = (cur + cnt.getOrDefault(prv[i], 0)) % MOD;
        for (int k : sub[i]) {
            cur = (cur + dp(k, j - 1)) % MOD;
        }
        int aa = (cnt.getOrDefault(i, 0) + cur) % MOD;
        cnt.put(i, aa);
        dp[i][j] = aa;
        return aa;
    }
}
