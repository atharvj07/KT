#include<bits/stdc++.h>
using namespace std;
int n,a[100000];
int main()
{
    cin>>n;
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    sort(a,a+n);
    int sz=n/3;
    for(int i=0;i<sz;i++)
    {
        if(a[3*i]==a[3*i+1] && a[3*i+1]==a[3*i+2])
        {
            cout<<-1<<endl;
            return 0;
        }
    }
    for(int i=0;i<sz;i++)
    {
        for(int j=i+1;j<sz;j++)
        {
            for(int k=j+1;k<sz;k++)
            {
                if(a[i]<a[j] && a[j]<a[k])
                {
                    if(a[i]%a[j]==0 && a[j]%a[k]==0)
                    {
                        cout<<a[i]<<" "<<a[j]<<" "<<a[k]<<" ";
                    }
                }
            }
        }
    }
    return 0;
}
