import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] planes = new int[n][4];
        for (int i = 0; i < n; i++) {
            planes[i][0] = sc.nextInt();
            planes[i][1] = sc.nextInt();
            char dir = sc.next().charAt(0);
            if (dir == 'U') planes[i][2] = 1;
            else if (dir == 'R') planes[i][3] = 1;
            else if (dir == 'D') planes[i][2] = -1;
            else planes[i][3] = -1;
        }

        long minTime = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int dx = planes[i][0] - planes[j][0];
                int dy = planes[i][1] - planes[j][1];
                if (dx * planes[i][3] == dy * planes[i][2] && dx * planes[j][3] == dy * planes[j][2]) {
                    long time = (long) (Math.abs(dx) / (Math.sqrt(planes[i][2] * planes[i][2] + planes[i][3] * planes[i][3])));
                    if (time < minTime) minTime = time;
                }
            }
        }

        if (minTime == Long.MAX_VALUE) System.out.println("SAFE");
        else System.out.println(minTime);
    }
}
