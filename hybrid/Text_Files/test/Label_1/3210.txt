#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n, q;
    cin >> n >> q;

    vector<int> v(n);
    for(int i=0; i<n; i++)
    {
        cin >> v[i];
    }

    vector<int> bit_inv(n+1), bit_val(n+1);

    auto update1 = [&](int idx, int val)
    {
        while(idx <= n)
        {
            bit_inv[idx] += val;
            idx += idx&(-idx);
        }
    };

    auto query1 = [&](int idx)
    {
        int sum = 0;
        while(idx > 0)
        {
            sum += bit_inv[idx];
            idx -= idx&(-idx);
        }
        return sum;
    };

    auto update2 = [&](int idx, int val)
    {
        while(idx <= n)
        {
            bit_val[idx] += val;
            idx += idx&(-idx);
        }
    };

    auto query2 = [&](int idx)
    {
        int sum = 0;
        while(idx > 0)
        {
            sum += bit_val[idx];
            idx -= idx&(-idx);
        }
        return sum;
    };

    for(int i=0; i<q; i++)
    {
        int t, x;
        cin >> t >> x;

        if(t == 1)
        {
            update1(x, 1);
            int val = query1(x);
            if(val&1)
            {
                update2(x, -1);
            }
            else
            {
                update2(x, 1);
            }
        }
        else
        {
            int cnt = query1(n);
            if(cnt < x)
            {
                cout << 0 << endl;
            }
            else
            {
                int low = 1, high = n, mid, ans = 0;

                while(low <= high)
                {
                    mid = (low + high) / 2;

                    int cnt = query1(mid);

                    if(cnt >= x)
                    {
                        ans = mid;
                        high = mid - 1;
                    }
                    else
                    {
                        low = mid + 1;
                    }
                }

                int val = query2(ans);

                if(val&1)
                {
                    cout << 1 << endl;
                }
                else
                {
                    cout << 0 << endl;
                }
            }
        }
    }

    return 0;
}
