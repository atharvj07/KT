import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        boolean[][][] dp = new boolean[n][10][10];
        dp[0][s.charAt(0) - '0'][0] = dp[1][s.charAt(1) - '0'][s.charAt(0) - '0'] = true;
        for (int i = 2; i < n; i++) {
            for (int j = 0; j < 10; j++) {
                for (int k = 0; k < 10; k++) {
                    int a = s.charAt(i) - '0';
                    if ((k == a && dp[i - 1][a][j]) || (k != a && dp[i - 1][a][j] && Math.abs(k - j) != Math.abs(a - s.charAt(i - 1)))) {
                        dp[i][a][j] = dp[i - 1][a][k] = true;
                    }
                }
            }
        }
        boolean ans = false;
        for (int i = 0; i < 10; i++) {
            if (dp[n - 1][i][i]) {
                ans = true;
            }
        }
        System.out.println(ans ? "YES" : "NO");
    }
}
