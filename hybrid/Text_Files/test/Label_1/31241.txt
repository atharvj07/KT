import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int a = scanner.nextInt();
        int b = scanner.nextInt();
        int v = scanner.nextInt();
        int boxes = 0;
        int nuts = 0;
        int sections = 0;
        while (nuts < a) {
            sections++;
            if (sections == k) {
                sections = 0;
                boxes++;
            }
            nuts += v;
        }
        System.out.println(boxes);
    }
}
