import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        long sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
        }
        long average = sum / n;
        for (int i = 0; i < 3; i++) {
            System.out.println(1 + " " + n);
            for (int j = 0; j < n; j++) {
                System.out.print((a[j] - average) + " ");
            }
            System.out.println();
            for (int j = 0; j < n; j++) {
                a[j] -= average;
            }
        }
    }
}
