for _ in range(int(input())):
    n, k = map(int, input().split())
    ans = n
    for d in range(1, 10):
        if k == 1 and d == n % 10:
            break
        num = d
        while num <= n:
            if num >= n and num % 10 != d:
                num += 10
                k -= 1
            if k == 0:
                ans = num
            num = num * 10 + d
    print(ans)
