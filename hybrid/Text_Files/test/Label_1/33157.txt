import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        int c = 0, k = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '-')
                c++;
            else
                k++;
        }
        if (k == 0 || c == 0)
            System.out.println("NO");
        else {
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) == 'o') {
                    if (cnt % 2 == c % 2)
                        break;
                    cnt++;
                }
            }
            if (cnt % 2 == c % 2)
                System.out.println("YES");
            else
                System.out.println("NO");
        }
    }
}
