def min_burles_spent():
    c1, c2, c3, c4 = map(int, input().split())
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))

    bus_cost = sum(min(c1 * rides, c2) for rides in a)
    bus_cost = min(bus_cost, c3)
    
    trolley_cost = sum(min(c1 * rides, c2) for rides in b)
    trolley_cost = min(trolley_cost, c3)

    return min(bus_cost + trolley_cost, c4)

print(min_burles_spent())
