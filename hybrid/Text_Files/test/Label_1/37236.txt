import sys
input = sys.stdin.readline
import math

def find(x, p):
    while x != p[x]:
        p[x] = p[p[x]]
        x = p[x]
    return x
def union(x, y, p, r):
    rx = find(x, p)
    ry = find(y, p)
    if rx == ry: return
    if r[rx] < r[ry]:
        p[rx] = ry
    else:
        p[ry] = rx
        if r[rx] == r[ry]: r[rx] += 1

for test in range(int(input())):
    k = int(input())
    edge = []
    for _ in range(2*k-1):
        x, y, w = map(int,input().split())
        edge.append((w, x, y))
    edge.sort()
    p = [i for i in range(2*k+1)]
    r = [0 for i in range(2*k+1)]
    sum1 = 0
    sum2 = 0
    for w, x, y in edge:
        if find(x, p) != find(y, p):
            union(x, y, p, r)
            sum1 += w
            sum2 += w*2
    print(sum1, sum2)
