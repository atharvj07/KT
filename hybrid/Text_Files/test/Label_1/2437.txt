#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int n, k, a[200005], t[800005], lz[800005];
void build (int v, int s, int e) {
    if (s > e) return;
    if (s == e) {
        t[v] = a[s];
        return;
    }
    int mid = (s + e) >> 1;
    build (v << 1, s, mid);
    build ((v << 1) + 1, mid + 1, e);
    t[v] = t[v << 1] + t[(v << 1) + 1];
}
void push (int v, int s, int e) {
    if (!lz[v]) return;
    t[v] += (e - s + 1) * lz[v];
    if (s != e) {
        lz[v << 1] += lz[v];
        lz[(v << 1) + 1] += lz[v];
    }
    lz[v] = 0;
}
int get (int v, int s, int e, int l, int r) {
    if (s > r || e < l || r < s || l > e) return 0;
    push (v, s, e);
    if (s >= l && e <= r) return t[v];
    int mid = (s + e) >> 1;
    return get (v << 1, s, mid, l, r) + get ((v << 1) + 1, mid + 1, e, l, r);
}
void upd (int v, int s, int e, int l, int r, int x) {
    if (s > r || e < l || r < s || l > e) return;
    push (v, s, e);
    if (s >= l && e <= r) {
        lz[v] += x;
        push (v, s, e);
        return;
    }
    int mid = (s + e) >> 1;
    upd (v << 1, s, mid, l, r, x);
    upd ((v << 1) + 1, mid + 1, e, l, r, x);
    t[v] = t[v << 1] + t[(v << 1) + 1];
}
int main () {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        upd (1, 1, n, i, i, x);
    }
    int ans = -1;
    for (int l = 1; l + k - 1 <= n; l++) {
        ans = max(ans, get (1, 1, n, l, l + k - 1) / k);
    }
    cout << ans << "\n";
    return 0;
}
