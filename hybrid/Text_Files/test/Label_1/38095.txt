for _ in range(int(input())):
    n = int(input())
    w = [int(x) for x in input().split()]
    h = 1
    dead = 0
    for i in range(n):
        if dead == 0:
            if i>0 and w[i]==w[i-1]==1:
                h += 5
            elif w[i] == 1:
                h += 1
            else:
                dead += 1
    print(h if dead == 0 else -1)
