import java.util.*;

public class Main {
    static final int MAGIC = (int) 5e6;
    static ArrayList<Integer>[] adj;
    static ArrayList<Integer>[] radj;
    static int[] dist;
    static int n, m;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        n = in.nextInt();
        m = in.nextInt();

        adj = new ArrayList[n];
        radj = new ArrayList[n];
        for (int i = 0; i < n; i++) { adj[i] = new ArrayList<>(); radj[i] = new ArrayList<>(); }
        int[] proc = new int[n];
        for (int i = 0; i < n; i++) proc[i] = in.nextInt();

        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            adj[a].add(b);
            radj[b].add(a);
        }

        LinkedList<Integer> queue = new LinkedList<>();
        dist = new int[n];
        for (int i = 0; i < n; i++) if (proc[i] == 0) { queue.add(i); dist[i] = 0; }
        while (!queue.isEmpty()) {
            int v = queue.poll();
            for (Integer u : adj[v]) {
                if (dist[u] == MAGIC) {
                    dist[u] = dist[v] + 1;
                    queue.add(u);
                }
            }
        }

        int ans = 0;
        queue.clear();
        dist = new int[n];
        for (int i = 0; i < n; i++) if (proc[i] == 1) { queue.add(i); dist[i] = 0; }
        while (!queue.isEmpty()) {
            int v = queue.poll();
            ans = Math.max(ans, dist[v]);
            for (Integer u : radj[v]) {
                if (dist[u] == MAGIC) {
                    dist[u] = dist[v] + 1;
                    queue.add(u);
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (proc[i] == 1 && dist[i] == MAGIC) {
                ans = -1;
                break;
            }
        }

        System.out.println(ans);
    }
}
