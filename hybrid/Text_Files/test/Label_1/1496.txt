#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<pair<string, int>> jiro(n);
  vector<int> ciel(m);
  for (int i = 0; i < n; i++) {
    string s;
    int x;
    cin >> s >> x;
    jiro[i] = {s, x};
  }
  for (int i = 0; i < m; i++) {
    int x;
    cin >> x;
    ciel[i] = x;
  }

  sort(jiro.begin(), jiro.end(), [](pair<string, int> a, pair<string, int> b) {
    return a.second < b.second;
  });
  sort(ciel.begin(), ciel.end(), greater<int>());

  int ans = 0;
  int j = 0;
  for (int i = 0; i < m; i++) {
    if (j == n) {
      ans += ciel[i];
    } else {
      if (jiro[j].first == "ATK") {
        if (ciel[i] >= jiro[j].second) {
          ans += ciel[i] - jiro[j].second;
          j++;
        }
      } else {
        if (ciel[i] > jiro[j].second) {
          j++;
        }
      }
    }
  }
  cout << ans << endl;
}
