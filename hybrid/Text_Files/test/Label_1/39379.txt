import bisect, sys
rl = sys.stdin.readline
n = int(rl())
lr = []
for _ in range(n):
    x, y = map(int, rl().split())
    lr += [(abs(y) - x, x)]
lr.sort()

if len(lr) >= 2 and lr[0][1] > lr[1][1]:
    print(-1)
else:
    sm, ans = 0, 0
    for x, y in reversed(lr):
        sm += x
        if sm > y:
            ans += 1
            sm = y
    print(ans + 1)
