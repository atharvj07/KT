import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int p = in.nextInt();
    long[] a = new long[p];
    for (int i = 0; i < p; i++) {
      a[i] = in.nextInt();
    }
    long[][] h = new long[p][p];
    h[0][0] = 1;
    for (int i = 1; i < p; i++) {
      h[i][i] = 1;
      for (int j = 0; j < i; j++) {
        h[i][j] = (h[i][j] + h[i - 1][j] * i) % p;
      }
    }
    long[] b = new long[p];
    for (int i = 0; i < p; i++) {
      for (int j = (i + 1) % p; j < p; j++) {
        a[i] -= a[j] * h[j][i];
        a[i] = (a[i] % p + p) % p;
      }
      for (int j = 0; j < i; j++) {
        a[i] -= b[j] * h[j][i];
        a[i] = (a[i] % p + p) % p;
      }
      b[i] = a[i] * pow(h[i][i], p - i - 2, p) % p;
    }
    for (long bi : b) {
      System.out.print(bi + " ");
    }
  }

  static long pow(long a, long b, long mod) {
    a %= mod;
    long res = 1;
    while (b > 0) {
      if ((b & 1) != 0) {
        res = res * a % mod;
      }
      a = a * a % mod;
      b >>= 1;
    }
    return res;
  }
}
