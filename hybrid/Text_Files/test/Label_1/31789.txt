import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        // Read input
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();

        // Precompute gcd values
        int[][] gcd = new int[n][n];
        for (int i = 0; i < n; i++) for (int j = i + 1; j < n; j++) gcd[i][j] = gcd(a[i], a[j]);

        // Set up shelf
        boolean[] shelf = new boolean[n];

        // Process queries
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt() - 1;

            // Toggle shelf status
            shelf[x] = !shelf[x];

            // Calculate score
            int score = 0;
            for (int j = 0; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (shelf[j] && shelf[k]) score += gcd[j][k];
                }
            }

            // Print score
            System.out.println(score);
        }
    }

    // Calculate the greatest common divisor of a and b
    static int gcd(int a, int b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
