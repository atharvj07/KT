#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1e5 + 10;
int n;
int a[maxn];
string s[maxn];
bool rev[maxn];
bool cmp(int x, int y) {
  if (rev[x] != rev[y]) {
    return rev[x] < rev[y];
  }
  if (!rev[x]) {
    return s[x] < s[y];
  } else {
    return s[x] > s[y];
  }
}
main() {
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
  }
  for (int i = 0; i < n; ++i) {
    string t;
    cin >> t;
    s[i] = t;
  }
  vector<int> vec;
  for (int i = 0; i < n; ++i) {
    vec.push_back(i);
  }
  sort(vec.begin(), vec.end(), cmp);
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    ans += (int)rev[vec[i]] * a[vec[i]];
  }
  if (vec[0] != 0) {
    ans = -1;
  }
  cout << ans << '\n';
  return 0;
}
