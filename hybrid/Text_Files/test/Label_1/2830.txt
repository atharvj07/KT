#include <iostream>
#include <vector>
#include <utility>
#define endl "\n"

using namespace std;

int main() {
    long long x, y;
    cin >> x >> y;
    while (x - y and x % 3 and y % 3) {
        x -= 2;
        y -= 1;
    }
    while (x + y) {
        if (x >= 3) {
            cout << "B";
            x -= 2;
            y += 1;
        } else if (y >= 3) {
            cout << "A";
            x += 2;
            y -= 3;
        } else {
            cout << "Impossible";
            return 0;
        }
    }
    cout << endl;
    return 0;
}
