#include <iostream>
using namespace std;
#define ll long long
const ll MOD = 1e9 + 7;
int main() {
    ll n, m, k;
    cin >> n >> m >> k;
    if (n * m - 1 < k) {
        cout << -1 << endl;
        return 0;
    }
    if (n < m) {
        swap(n, m);
    }
    ll ans = min((k + 1) / 2, n - 1);
    ans = ans * min(m, k - ans);
    cout << ans << endl;
    return 0;
}
