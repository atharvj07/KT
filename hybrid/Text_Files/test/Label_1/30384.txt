import java.io.IOException;
import java.io.InputStreamReader;
import java.io.BufferedReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[] segments = new long[n];
        char[] terrains = new char[n];
        String[] tmp = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            segments[i] = Long.parseLong(tmp[i]);
        }
        terrains = br.readLine().toCharArray();
        long res = solve(segments, terrains);
        System.out.println(res);
    }

    public static long solve(long[] segments, char[] terrains) {
        long[] dp = new long[segments.length + 1];
        Arrays.fill(dp, Long.MAX_VALUE);
        dp[0] = 0;
        for (int i = 1; i <= segments.length; i++) {
            long byFoot = dp[i - 1] + 5 * segments[i - 1];
            if (terrains[i - 1] != 'L') {
                long bySwim = dp[i - 1] + 3 * segments[i - 1];
                if (bySwim < dp[i]) dp[i] = bySwim;
            }
            if (terrains[i - 1] != 'W') {
                long byFlight = dp[i - 1] + segments[i - 1];
                if (byFlight < dp[i]) dp[i] = byFlight;
            }
        }
        return dp[segments.length];
    }
}
