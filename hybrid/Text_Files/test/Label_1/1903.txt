#include <iostream>
#include <algorithm>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n, x;
    std::cin >> n >> x;
    int w[101];
    for (int i = 0; i < n; i++) std::cin >> w[i];
    std::sort(w, w + n);
    bool can = 1;
    for (int i = 0; i < n; i++) {
      if (w[i] >= x) can = 0;
      x -= w[i];
    }
    if (can) {
      std::cout << "YES\n";
      for (int i = 0; i < n; i++) std::cout << w[i] << " ";
      std::cout << "\n";
    } else
      std::cout << "NO\n";
  }
  return 0;
}
