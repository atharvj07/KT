#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, p;
  cin >> n >> p;

  vector<pair<int, int>> a(n + 1);
  vector<bool> used(n + 1, false);

  for (int i = 1; i <= n; ++i) {
    cin >> a[i].first >> a[i].second;
  }

  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (!used[i] && !used[a[i].first] && !used[a[i].second]) {
      used[i] = true;
      used[a[i].first] = true;
      used[a[i].second] = true;
      ++ans;
    }
  }

  cout << ans << endl;

  return 0;
}
