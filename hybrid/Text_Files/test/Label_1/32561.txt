import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        int[] divisors = new int[n];
        boolean[] isPrime = new boolean[5001];
        Arrays.fill(isPrime, true);
        isPrime[0] = false;
        isPrime[1] = false;
        for (int i = 2; i <= 5000; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j <= 5000; j += i) {
                    isPrime[j] = false;
                }
            }
        }
        int cnt = 0;
        for (int i = 2; i <= 5000 - n + 1; i++) {
            if (!isPrime[i]) {
                arr[cnt] = i;
                cnt++;
                if (cnt == n) {
                    break;
                }
            }
        }
        System.out.println(arr[0]);
        for (int i = 0; i < n; i++) {
            for (int j = 2; j <= 5000; j++) {
                if (arr[i] % j == 0 && j != arr[i]) {
                    divisors[i] = j;
                    break;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(divisors[i]);
        }
    }
}
