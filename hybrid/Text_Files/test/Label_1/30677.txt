import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in=new Scanner(System.in);
        String[] p=in.nextLine().split(" ");
        int r0=p[0].charAt(0)-'a';
        int r1=p[1].charAt(0)-'a';
        int c0=p[0].charAt(1)-'1';
        int c1=p[1].charAt(1)-'1';
        int kx=p[2].charAt(0)-'a';
        int ky=p[2].charAt(1)-'1';
        if(r0==r1 || c0==c1){
            if(r0==kx || c0==ky){
                System.out.println("CHECKMATE");
            }
            else{
                System.out.println("OTHER");
            }
        }
        else{
            System.out.println("OTHER");
        }
    }
}
