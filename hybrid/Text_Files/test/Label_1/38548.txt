import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a = []
for _ in range(n):
    a.append([int(c) for c in input().strip()])

if sum(map(sum, a)) != 7*(n+k):
    print(-1)
    sys.exit()

b = [[1 for i in range(7)] for _ in range(10)]
b[0][3] = b[0][6] = 0
b[1][2] = b[1][5] = 0
b[2][1] = b[2][3] = b[2][5] = b[2][6] = 0
b[3][1] = b[3][2] = b[3][3] = b[3][5] = 0
b[4][1] = b[4][2] = b[4][3] = b[4][6] = 0
b[5][0] = b[5][2] = b[5][3] = b[5][6] = 0
b[6][0] = b[6][2] = b[6][3] = b[6][4] = b[6][5] = 0
b[7][2] = b[7][5] = 0
b[8][6] = 0
b[9][1] = b[9][3] = 0

ans = []
for i in range(n):
    tot = sum(a[i])
    for j in range(10):
        if sum(x==1 for x in a[i]) + sum(x==1 for x in b[j]) == tot:
            flag = 1
            for k in range(7):
                if a[i][k] == 1 and b[j][k] == 0:
                    flag = 0
            if tot == 7:
                flag = 1
            if flag:
                ans.append(j)
                break
if len(ans) != n:
    print(-1)
    sys.exit()

for x in ans:
    print(x, end='')
