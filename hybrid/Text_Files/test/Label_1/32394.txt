import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    private static List<int[]> roads;
    private static Map<Integer, List<Integer>> graph;
    private static boolean hasCycle;
    private static boolean[] visited;
    private static boolean[] onStack;

    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);

        roads = new ArrayList<>();
        graph = new HashMap<>();
        for (int i = 0; i < n; i++) {
            graph.put(i + 1, new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            parts = reader.readLine().split(" ");
            int u = Integer.parseInt(parts[0]);
            int v = Integer.parseInt(parts[1]);
            roads.add(new int[]{u, v});
            graph.get(u).add(v);
            graph.get(v).add(u);
        }

        if (hasCycle()) {
            System.out.println(-1);
            return;
        }

        List<int[]> addedRoads = getAddedRoads();
        if (addedRoads == null) {
            System.out.println(0);
            return;
        }

        System.out.println(addedRoads.size());
        for (int[] road : addedRoads) {
            System.out.println(road[0] + " " + road[1]);
        }
    }

    private static boolean hasCycle() {
        visited = new boolean[graph.size() + 1];
        onStack = new boolean[graph.size() + 1];

        for (int i = 1; i <= graph.size(); i++) {
            if (!visited[i]) {
                if (dfs(i)) {
                    return true;
                }
            }
        }

        return false;
    }

    private static boolean dfs(int vertex) {
        visited[vertex] = true;
        onStack[vertex] = true;

        for (int neighbor : graph.get(vertex)) {
            if (!visited[neighbor]) {
                if (dfs(neighbor)) {
                    return true;
                }
            } else if (onStack[neighbor]) {
                return true;
            }
        }

        onStack[vertex] = false;
        return false;
    }

    private static List<int[]> getAddedRoads() {
        List<int[]> addedRoads = new ArrayList<>();

        for (int[] road : roads) {
            int u = road[0];
            int v = road[1];

            graph.get(u).remove(Integer.valueOf(v));
            graph.get(v).remove(Integer.valueOf(u));

            if (!hasCycle()) {
                addedRoads.add(road);
            } else {
                graph.get(u).add(v);
                graph.get(v).add(u);
            }
        }

        return addedRoads.isEmpty() ? null : addedRoads;
    }
}
