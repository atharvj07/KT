import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
		long n = Long.parseLong(new BufferedReader(new InputStreamReader(System.in)).readLine());
		System.out.println(NumberOfBonus(n));
		
    }
	
	public static long NumberOfBonus(long n)
	{
		long ans=0;
		for(long i=1; i<=n; i++)
		{
			int count = 0;
			for(long j=2; j<=10; j++)
				if(i%j==0)
					count++;
			
			if(count==9)
				ans++;
		}
		return ans;
	}
}
