#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> v;
    for (int i = 0; i < n; i++) {
        if (s[i] == '0') v.push_back(i);
    }
    int ans = v[1] - v[0];
    for (int i = 1; i < v.size() - 1; i++) {
        ans = max(ans, v[i + 1] - v[i]);
    }
    cout << ans << endl;
    return 0;
}
