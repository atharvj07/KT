#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    int n, ans = -1;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    sort(a.begin(), a.end());
    for (int l = 0, r = 0; r < n; r++) {
        while (l < r && a[r] > a[l] * 2)
            l++;
        ans = max(ans, max(n - l - (r - l % 2), r - l % 2));
    }
    cout << n - ans << endl;
    return 0;
}
