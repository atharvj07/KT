from collections import Counter

n, m = map(int, input().split())
s = [input() for _ in range(n)]
t = [list(map(int, input().split())) for _ in range(n)]

pos = []
for i in range(m):
    cnt = Counter()
    for j in range(n):
        cnt[s[j][i]] += 1
    for c, ccnt in cnt.items():
        if ccnt == 1: pos.append((c, i))

ans = 0
for c, i in pos:
    for j in range(n):
        if s[j][i] == c: continue
        ans += t[j][i]
print(ans)
