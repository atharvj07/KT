import sys
N,M=map(int,input().split())
G=[[] for i in range(N)]
for i in range(M):
	A,B=map(int,input().split())
	A-=1;B-=1
	G[A].append(B)
	G[B].append(A)
FLG=[-1]*N
FLG2=[]
for i in range(N):
	if FLG[i]!=-1:continue
	Q=[i]
	while Q:
		X=Q.pop()
		for y in G[X]:
			if FLG[X]==FLG[y]:
				print(-1)
				sys.exit(0)
			if FLG[y]==-1:
				FLG[y]=1-FLG[X]
				Q.append(y)
	FLG2.append([])
for i in range(N):
	FLG2[FLG[i]].append(i)
if len(FLG2)!=3:
	print(-1)
	sys.exit(0)
for x in FLG2:
	for y in x:print(FLG2.index(x)+1,end=" ")
