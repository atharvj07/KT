#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n + 1);
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  queue<int> q;
  vector<int> d(n + 1, -1);
  vector<int> attacked(n + 1);
  for (int i = 0; i < m; i++) {
    int city;
    cin >> city;
    attacked[city] = 1;
    q.push(city);
    d[city] = 0;
  }
  int ans_city = 1;
  int ans = 0;
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    for (int v : adj[u]) {
      if (d[u] + 1 < d[v] || d[v] == -1) {
        d[v] = d[u] + 1;
        q.push(v);
        if (attacked[v]) {
          ans = max(ans, d[v]);
          ans_city = v;
        }
      }
    }
  }
  cout << ans_city << endl;
  cout << ans << endl;
  return 0;
}
