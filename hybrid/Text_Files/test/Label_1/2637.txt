#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll N = 1e5 + 5;
ll n, q;
ll par[N], sz[N];
ll fin(ll x) {
    if(x == par[x]) return x;
    return par[x] = fin(par[x]);
}
void uni(ll a, ll b) {
    ll p = fin(a), q = fin(b);
    if(p != q) {
        if(sz[p] < sz[q]) swap(p, q);
        par[q] = p;
        sz[p] += sz[q];
    }
}
int main() {
    cin >> n >> q;
    for(ll i = 1; i <= n; i++) {
        par[i] = i;
        sz[i] = 1;
    }
    vector<ll> v;
    while(q--) {
        ll ty, x;
        cin >> ty >> x;
        if(ty == 1) {
            if(v.size() && v.back() == x) v.pop_back();
            else {
                v.push_back(x);
                uni(x, x + 1);
            }
        }
        else {
            if(x > v.size()) cout << "0\n";
            else cout << fin(v[x - 1]) << '\n';
        }
    }
    return 0;
}
