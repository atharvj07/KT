import java.util.*;

class Main {
    public static void main(String[] args) throws Exception {
        Scanner scan = new Scanner(System.in);
        int t = scan.nextInt();
        while (t-- > 0) {
            int n = scan.nextInt();
            int m = scan.nextInt();
            long s[] = new long[n];
            long p[] = new long[n + 1];
            for (int i = 0; i < n; ++i) {
                s[i] = scan.nextInt();
                p[i + 1] = p[i] + s[i] / (i + 1);
            }
            boolean ok = false;
            for (int i = 0; i < n; ++i) {
                if (p[n] - p[i] == m - p[i]) {
                    ok = true;
                    break;
                }
            }
            System.out.println(ok ? "YES" : "NO");
        }
    }
}
