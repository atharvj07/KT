import sys
import heapq
input = sys.stdin.readline

for _ in range(int(input())):
    k = int(input())
    G = [(0,0)]
    B = [(0,0)]
    for _ in range(2*k-1):
        G.append(heapq.heappop(B))
        a, b, t = map(int, input().split())
        heapq.heappush(G, (t, max(a,b)))
        heapq.heappush(B, (t, -min(a,b)))
    print(*[v[0] for v in G])
