import java.util.Scanner;

class Main {

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    String[] colors = new String[] {"C", "M", "Y"};
    boolean isColored = false;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        String color = in.next();
        if (isColored)
          continue;
        for (String c : colors)
          if (c.equals(color)) {
            isColored = true;
            break;
          }
      }
    }
    System.out.println(isColored ? "#Color" : "#Black&White");
  }
}
