import java.io.*;
import java.util.*;

public class Main {
    static class Node {
        int val;
        Node next;

        Node(int val) {
            this.val = val;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        LinkedList<Node> showcase = new LinkedList<>();
        boolean res = true;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 2 * n; i++) {
            String[] s = br.readLine().split(" ");
            if (s[0].equals("+")) {
                Node newNode = new Node(n - i);
                if (showcase.isEmpty() || showcase.peekLast().val >= newNode.val) {
                    showcase.addLast(newNode);
                } else {
                    res = false;
                    break;
                }
            } else {
                int val = Integer.parseInt(s[1]);
                if (!showcase.isEmpty() && showcase.peekFirst().val == val) {
                    showcase.removeFirst();
                } else {
                    res = false;
                    break;
                }
            }
        }
        if (res) {
            for (Node node : showcase) {
                sb.append(node.val).append(" ");
            }
            System.out.println("YES\n" + sb.toString().trim());
        } else {
            System.out.println("NO");
        }
    }
}
