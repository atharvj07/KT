#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, k;
    cin >> n >> k;

    vector<long long> a(2 * n);
    iota(a.begin(), a.end(), 1);
    for (long long i = 0; i < k; i++) {
        swap(a[2 * i], a[2 * i + 1]);
    }

    for (auto x : a) {
        cout << x << " ";
    }
    cout << "\n";

    return 0;
}
