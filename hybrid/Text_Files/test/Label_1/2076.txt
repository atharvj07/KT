#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

ll qikPow(ll b, ll p) {
  if (p == 0) {
    return 1;
  }
  if (p == 1) {
    return b;
  }
  ll t = qikPow(b, p / 2);
  if (p % 2 == 1) {
    return t * t % mod * b % mod;
  }
  return t * t % mod;
}

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
  }
  ll res = 0;
  ll sum = 0;
  for (ll i = 0; i < n; i++) {
    sum += a[i];
    sum %= mod;
    res += (i * a[i] % mod) * qikPow(sum, mod - 2) % mod;
    res %= mod;
  }
  cout << res << endl;
  return 0;
}
