#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
int n,k;
vector<int>ans;
ll solve(int x){
    if(x>n){
        if(ans.size()==k){
            for(int i=0;i<ans.size();i++)
                cout<<ans[i]<<" ";
            cout<<endl;
            return 0;
        }
        return 0;
    }
    solve(x+1);
    ans.pb(x);
    solve(x+1);
    ans.pop_back();
    return 0;
}
int main(){
    cin>>n>>k;
    if(k>(n/2+n%2))
        cout<<-1<<endl;
    else
        solve(1);
}
