import java.math.*;
import java.util.Arrays;

public class Main {
    private static final BigInteger MOD = new BigInteger("1000000007");
    private static BigInteger[][] dp = new BigInteger[101][101];

    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int k = sc.nextInt();
        int t = sc.nextInt();

        for (int i = 0; i < 101; i++) {
            Arrays.fill(dp[i], BigInteger.ZERO);
        }
        dp[0][0] = BigInteger.ONE;
        for (int i = 0; i < t; i++) {
            for (int j = 0; j < 101; j++) {
                for (int l = -k; l <= k; l++) {
                    int ni = j + l;
                    int nj = i + 1;
                    if (ni < 0 || ni > 100) {
                        continue;
                    }
                    if (ni > nj) {
                        dp[nj][ni] = dp[nj][ni].add(dp[i][j]);
                        dp[nj][ni] = dp[nj][ni].mod(MOD);
                    }
                }
            }
        }

        System.out.println(dp[t][0]);
    }
}
