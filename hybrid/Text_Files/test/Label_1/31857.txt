import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class BombDestroyer {

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Point> bombs = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = scanner.nextInt();
            int y = scanner.nextInt();
            bombs.add(new Point(x, y));
        }

        // Find the shortest sequence of operations
        List<String> operations = findShortestSequence(bombs);

        // Print the number of operations and the sequence
        System.out.println(operations.size());
        for (String operation : operations) {
            System.out.println(operation);
        }
    }

    private static List<String> findShortestSequence(List<Point> bombs) {
        // Sort bombs by distance to the origin (0, 0)
        bombs.sort((p1, p2) -> Integer.compare(p1.x * p1.x + p1.y * p1.y, p2.x * p2.x + p2.y * p2.y));

        // Find the shortest sequence of operations
        List<String> operations = new ArrayList<>();
        Point robotPosition = new Point(0, 0);
        for (Point bomb : bombs) {
            // Move to the bomb
            operations.addAll(moveTo(robotPosition, bomb));

            // Pick up the bomb
            operations.add("2");

            // Return to the origin
            operations.addAll(moveTo(bomb, new Point(0, 0)));

            // Destroy the bomb
            operations.add("3");

            // Update robot position
            robotPosition = new Point(0, 0);
        }

        return operations;
    }

    private static List<String> moveTo(Point start, Point end) {
        List<String> operations = new ArrayList<>();

        // Move horizontally
        int dx = end.x - start.x;
        if (dx > 0) {
            operations.add(String.format("1 %d R", dx));
        } else if (dx < 0) {
            operations.add(String.format("1 %d L", -dx));
        }

        // Move vertically
        int dy = end.y - start.y;
        if (dy > 0) {
            operations.add(String.format("1 %d U", dy));
        } else if (dy < 0) {
            operations.add(String.format("1 %d D", -dy));
        }

        return operations;
    }
}
