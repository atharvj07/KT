import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int s = sc.nextInt();
        System.out.println((Math.abs(a) + Math.abs(b)) <= s && (s - Math.abs(a) - Math.abs(b)) % 2 == 0 ? "Yes" : "No");
    }
}
