#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) ++cnt;
    }
    if (cnt == 0) {
      cout << 0 << endl;
    } else if (cnt > 1) {
      cout << -1 << endl;
    } else {
      cout << 1 << endl;
    }
  }
  return 0;
}
