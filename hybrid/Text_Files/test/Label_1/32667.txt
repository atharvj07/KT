import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean[] a = new boolean[n + 1];
        int[] p = new int[n + 1];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            if (p[i] > 0) {
                a[p[i]] = true;
            }
        }
        int[] b = new int[n];
        int j = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i]) {
                b[j++] = i;
            }
        }
        int[] c = new int[n];
        int[] d = new int[n];
        j = 0;
        for (int i = 0; i < n; i++) {
            if (b[i] % 2 == 0) {
                c[j++] = b[i];
            }
        }
        j = 0;
        for (int i = 0; i < n; i++) {
            if (b[i] % 2 == 1) {
                d[j++] = b[i];
            }
        }
        int m = Math.min(j, n - j);
        for (int i = 0; i < m; i++) {
            b[i] = c[i];
        }
        for (int i = m; i < n; i++) {
            b[i] = d[i - m];
        }
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (b[i] % 2 != b[i - 1] % 2) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
