#include <bits/stdc++.h>

using namespace std;

const int N = 105;
int n, m, k;
int p[N], s[N];
int c[N], cp[N];

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) cin >> p[i];
    for (int i = 1; i <= n; i++) cin >> s[i];
    for (int i = 1; i <= k; i++) cin >> c[i];

    memset(cp, -1, sizeof(cp));
    for (int i = 1; i <= k; i++) cp[s[c[i]]] = c[i];

    int ans = 0;
    for (int i = 1; i <= m; i++) {
        if (cp[i] == -1) ans++;
    }

    cout << ans << endl;

    return 0;
}
