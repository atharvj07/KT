import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    static class Block {
        int start;
        int size;

        public Block(int start, int size) {
            this.start = start;
            this.size = size;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = reader.readLine().split(" ");
        int t = Integer.parseInt(firstLine[0]);
        int m = Integer.parseInt(firstLine[1]);

        List<Block> blocks = new ArrayList<>();
        blocks.add(new Block(0, m));

        for (int i = 0; i < t; i++) {
            String[] command = reader.readLine().split(" ");
            switch (command[0]) {
                case "alloc":
                    int size = Integer.parseInt(command[1]);
                    int allocated = allocate(blocks, size);
                    System.out.println(allocated == -1 ? "NULL" : allocated);
                    break;
                case "erase":
                    int token = Integer.parseInt(command[1]);
                    boolean erased = erase(blocks, token);
                    System.out.println(erased ? token : "ILLEGAL_ERASE_ARGUMENT");
                    break;
                case "defragment":
                    defragment(blocks);
                    break;
            }
        }
    }

    private static int allocate(List<Block> blocks, int size) {
        for (int i = 0; i < blocks.size(); i++) {
            Block block = blocks.get(i);
            if (block.size >= size) {
                int start = block.start;
                block.start += size;
                block.size -= size;
                if (block.size == 0) {
                    blocks.remove(i);
                }
                return start + 1;
            }
        }
        return -1;
    }

    private static boolean erase(List<Block> blocks, int token) {
        for (int i = 0; i < blocks.size(); i++) {
            Block block = blocks.get(i);
            if (block.start + 1 == token) {
                block.start = token - 1;
                block.size += token - block.start - 1;
                if (i > 0) {
                    Block prevBlock = blocks.get(i - 1);
                    if (prevBlock.start + prevBlock.size == block.start) {
                        prevBlock.size += block.size;
                        blocks.remove(i);
                    }
                }
                return true;
            }
        }
        return false;
    }

    private static void defragment(List<Block> blocks) {
        for (int i = 0; i < blocks.size(); i++) {
            Block block = blocks.get(i);
            if (i > 0) {
                Block prevBlock = blocks.get(i - 1);
                if (prevBlock.start + prevBlock.size == block.start) {
                    prevBlock.size += block.size;
                    blocks.remove(i);
                    i--;
                }
            }
        }
    }
}
