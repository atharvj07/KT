import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] c = new int[n];
        int[] s = new int[n];
        int[] f = new int[n];
        for(int i = 0; i < n - 1; i++) {
            c[i] = in.nextInt();
            s[i] = in.nextInt();
            f[i] = in.nextInt();
        }
        int[] ans = new int[n];
        ans[n - 1] = 0;
        for(int i = n - 2; i >= 0; i--) {
            ans[i] = ans[i + 1] + c[i] + (s[i] + f[i] - 1) / f[i] * f[i];
        }
        for(int i = 0; i < n; i++) {
            System.out.println(ans[i]);
        }
    }
}
