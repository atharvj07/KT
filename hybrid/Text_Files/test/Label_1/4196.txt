#include <iostream>

int main() {
  int a, ta, b, tb;
  std::cin >> a >> ta >> b >> tb;
  std::string arrival;
  std::cin >> arrival;
  int arrival_h = (arrival[0] - '0') * 10 + (arrival[1] - '0'),
      arrival_m = (arrival[3] - '0') * 10 + (arrival[4] - '0');
  int minutes = arrival_h * 60 + arrival_m, ans = 0;
  for (int i = 0; i * ta < minutes; ++i) {
    int time = i * ta;
    if (time >= 5 * 60 && time < 23 * 60) {
      ans += minutes / b - (time + tb - 1) / b;
    }
  }
  std::cout << ans << std::endl;
}
