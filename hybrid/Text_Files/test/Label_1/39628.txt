n, m, d = map(int, input().split())
breaks = list(map(int, input().split()))
breaks_sorted = sorted((b, i) for i, b in enumerate(breaks))

days = [0] * n
current_day = 1
last_break = -1
for break_time, index in breaks_sorted:
    if break_time - last_break > d:
        days[index] = current_day
        last_break = break_time
    else:
        current_day += 1
        days[index] = current_day
        last_break = break_time

print(max(days))
print(*days)
