import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int k = cin.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = cin.nextLong();
        }
        long ans = 1;
        long tot = 0;
        for (int i = 0; i < k; i++) {
            tot += a[i];
        }
        for (int i = 0; i < k; i++) {
            if (a[i] == tot) {
                ans = ans * a[i] % 998244353;
            } else {
                ans = ans * (a[i] + 1) % 998244353;
            }
        }
        System.out.println(ans);
    }
}
