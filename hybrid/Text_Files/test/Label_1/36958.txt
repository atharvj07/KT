import sys
input = sys.stdin.readline

n = int(input())

def isPrime(n):
  if n < 2: return False
  for i in range(2, int(n**0.5) + 1):
    if n % i == 0:
      return False
  return True

prime = []
vis = [False] * 100010
for i in range(2, 100010):
  if not vis[i]:
    vis[i] = True
    prime.append(i)
    for j in range(i + i, 100010, i):
      vis[j] = True
      
ans = [0] * (n + 1)
cnt = 0
for i in range(2, n + 1):
  if prime[cnt] > i: break
  if i % prime[cnt] == 0: ans[i] = cnt + 1
  cnt += 1
  
cnt = 0
for i in reversed(range(2, n + 1)):
  if ans[i] == 0:
    cnt += 1
    for j in range(i, n + 1, i):
      if ans[j] == 0: ans[j] = cnt

print(cnt)
print(*ans[2:])
