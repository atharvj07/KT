import java.util.*;
import java.lang.*;

class Main {
    static int N, M;
    static int[] S, T;
    static long mod = (long) (1e9 + 7);
    static long[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        S = new int[N + 1];
        T = new int[M + 1];
        dp = new long[N + 1][M + 1];
        for (int i = 1; i <= N; i++) {
            S[i] = sc.nextInt();
        }
        for (int i = 1; i <= M; i++) {
            T[i] = sc.nextInt();
        }
        System.out.println(solve(N, M));
    }

    static long solve(int n, int m) {
        if (n == 0 || m == 0) {
            return 1;
        }
        if (dp[n][m] != 0) {
            return dp[n][m];
        }
        dp[n][m] = solve(n - 1, m);
        if (S[n] == T[m]) {
            dp[n][m] = (dp[n][m] + solve(n - 1, m - 1)) % mod;
        }
        return dp[n][m];
    }
}
