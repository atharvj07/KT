from sys import stdin
n = int(stdin.readline())
a = [-1] * n
for i in range(n):
    b = stdin.readline().split()
    for j in range(n):
        if b[j] != '-1' and a[j] == -1:
            a[j] = int(b[j])
b = [-1] * n
for i in range(n):
    b[i] = a[i]
    for j in range(n):
        if b[j] != -1 and b[i] != -1 and b[j] != b[i] & a[j]:
            b[i] = -1
            break
print(' '.join(map(str, b)))
