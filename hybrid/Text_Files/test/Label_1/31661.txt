import java.util.*;
  
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int c = sc.nextInt();
        int n = sc.nextInt();
        boolean[][] grid = new boolean[r + 1][c + 1];
        int[] dr = { 0, 1, 0, -1 };
        int[] dc = { 1, 0, -1, 0 };
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            if (grid[x][y])
                continue;
            boolean ok = false;
            for (int j = 0; j < 4; j++) {
                int nx = x + dr[j], ny = y + dc[j];
                if (nx >= 1 && nx <= r && ny >= 1 && ny <= c && !grid[nx][ny]) {
                    ok = true;
                }
            }
            if (!ok)
                continue;
            grid[x][y] = true;
            ans++;
            int cnt = 0;
            for (int j = 1; j <= c; j++) {
                if (grid[1][j]) {
                    cnt++;
                    for (int k = 1; k <= r; k++) {
                        if (grid[k][j]) {
                            cnt--;
                        }
                    }
                    if (cnt == 0) {
                        ok = false;
                        break;
                    }
                }
            }
            if (!ok) {
                ans--;
                grid[x][y] = false;
            }
        }
        System.out.println(ans);
    }
}
