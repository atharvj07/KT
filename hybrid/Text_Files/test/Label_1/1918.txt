#include<iostream>
#include<cstring>
using namespace std;
int main(){
    int r,c;
    cin>>r>>c;
    char grid[r][c+1], cur, lst;
    for(int i=0;i<r;i++)cin>>grid[i];
    int cnt=0;
    lst=grid[r-1][0];
    for(int j=1;j<=c;j++)if((grid[r-1][j-1] == 'B') ^ (grid[r-1][j] == 'B'))cnt++,lst=grid[r-1][j];
    for(int i=r-2;i>=0;i--){
        cur=grid[i][0];
        for(int j=1;j<=c;j++){
            if(cur^grid[i][j]){
                if(cur=='B'){
                    if(lst=='B')cnt--, lst=grid[i][j];
                    else{cnt++;lst=grid[i][j];}
                }
                else{
                    if(lst=='B'){cnt++; lst=grid[i][j];}
                }
            }
            cur=grid[i][j];
        }
    }
    cout<<cnt<<endl;
    return 0;
}
