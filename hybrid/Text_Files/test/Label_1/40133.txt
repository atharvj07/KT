for _ in range(int(input())):
    n = int(input())
    a = [0] + list(map(int, input().split()))
    if max(a) > n:
        print(-1)
    else:
        ans = 0
        last = -1
        for i in a[1:]:
            if i != last:
                last = i
            else:
                ans += 1
        print(ans)
