#include <bits/stdc++.h>
#define int long long
using namespace std;
const int MAXN = 3e5 + 1;
int n, k, d, p[300001], u[300001], v[300001], par[300001], vis[300001], dis[300001], t, ans[300001], cnt;
vector<int> adj[300001];
queue<int> q;
int find(int x) {
    if (x == par[x]) return x;
    return par[x] = find(par[x]);
}
void merge(int a, int b) { par[find(b)] = find(a); }
main() {
    cin >> n >> k >> d;
    for (int i = 1; i <= k; i++) cin >> p[i];
    for (int i = 1; i < n; i++) { cin >> u[i] >> v[i]; adj[u[i]].push_back(v[i]); adj[v[i]].push_back(u[i]); }
    for (int i = 1; i <= n; i++) par[i] = i;
    q.push(p[1]);
    vis[p[1]] = 1;
    while (!q.empty()) {
        t = q.front();
        q.pop();
        for (int u : adj[t]) {
            if (vis[u]) continue;
            vis[u] = 1;
            merge(t, u);
            if (dis[find(u)] > dis[find(t)] + 1) q.push(u), dis[find(u)] = dis[find(t)] + 1;
        }
    }
    for (int i = 1; i < n; i++) {
        if (find(u[i]) != find(v[i])) {
            ans[cnt++] = i;
            merge(u[i], v[i]);
        }
    }
    cout << cnt << '\n';
    for (int i = 0; i < cnt; i++) cout << ans[i] << ' ';
}
