#include <bits/stdc++.h>

using namespace std;

int n, x;
vector<int> d;
int dp[200005];

int solve(int day, int prevdays) {
  if (day == (int)d.size()) return 0;
  int & res = dp[day * 100 + prevdays];
  if (res != -1) return res;

  int dayInMonth = prevdays % d[day] + 1;
  res = dayInMonth;
  if (dayInMonth + x <= d[day]) res = max(res, solve(day, prevdays + x) + x);
  else res = max(res, solve(day + 1, prevdays + d[day] - dayInMonth + x) + d[day]);
  return res;
}

int main() {
  cin >> n >> x;
  d.resize(n);
  for (auto& dd : d) cin >> dd;
  memset(dp, -1, sizeof dp);
  cout << solve(0, 0) << endl;
  return 0;
}
