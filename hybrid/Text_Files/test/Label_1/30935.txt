import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] lengths = new long[n];
        for (int i = 0; i < n; i++) lengths[i] = input.nextLong();
        char[] terrain = input.next().toCharArray();
        long time = 0;
        int stamina = 0;
        for (int i = 0; i < n; i++) {
            if (terrain[i] == 'G') {
                stamina += lengths[i];
                time += lengths[i] * 5;
            } else if (terrain[i] == 'W') {
                if (stamina > 0) {
                    stamina--;
                    time += lengths[i] * 3;
                } else {
                    if (i == n - 1 || terrain[i + 1] == 'G') {
                        stamina++;
                        time += lengths[i] * 3 + 1;
                    } else {
                        time += lengths[i] * 5 + 1;
                    }
                }
            } else {
                if (stamina >= 2) {
                    stamina -= 2;
                    time += lengths[i];
                } else {
                    if (i == n - 1 || terrain[i + 1] == 'G') {
                        time += lengths[i] * 5;
                    } else {
                        time += lengths[i] * 3 + 1;
                    }
                }
            }
        }
        System.out.println(time);
    }
}
