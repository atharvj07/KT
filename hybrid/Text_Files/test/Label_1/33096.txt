import java.util.*;

class Main {
    static Scanner sc = new Scanner(System.in);

    static String solve() {
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();

        if (a[0] % 2 == n % 2) return "BitLGM";
        return "BitAryo";
    }

    public static void main(String[] args) {
        System.out.println(solve());
    }
}
