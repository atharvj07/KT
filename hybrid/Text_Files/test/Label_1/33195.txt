import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (in.hasNext()) {
            String line = in.nextLine();
            int r1 = 0, r2 = 0;
            char op = '\0';
            for (char c : line.toCharArray()) {
                if (c >= '0' && c <= '9') {
                    r2 = r2 * 10 + c - '0';
                } else {
                    switch (c) {
                        case '+':
                        case '-':
                        case '*':
                            if (op != '\0') {
                                switch (op) {
                                    case '+':
                                        r1 += r2;
                                        break;
                                    case '-':
                                        r1 -= r2;
                                        break;
                                    case '*':
                                        r1 *= r2;
                                        break;
                                }
                            } else {
                                r1 = r2;
                            }
                            r2 = 0;
                            op = c;
                            break;
                        case '=':
                            if (op != '\0') {
                                switch (op) {
                                    case '+':
                                        r1 += r2;
                                        break;
                                    case '-':
                                        r1 -= r2;
                                        break;
                                    case '*':
                                        r1 *= r2;
                                        break;
                                }
                            }
                            if (r1 < 0 || r1 > 9999) {
                                System.out.println("E");
                            } else {
                                System.out.println(r1);
                            }
                            r1 = 0;
                            r2 = 0;
                            op = '\0';
                            break;
                    }
                }
            }
        }
        in.close();
    }
}
