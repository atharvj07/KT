#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
const int M = 1e9 + 7;
const double PI = 3.14159265358979323846;
#define endl '\n'
#define all(a) a.begin(), a.end()
#define x first
#define y second
int main()
{
    int n;
    cin >> n;
    vi A(n);
    for (int i = 0; i < n; i++)
        cin >> A[i];
    vi B(n);
    sort(all(A));
    int j = 0;
    for (int i = 0; i < n; i++)
        if (A[i] != A[i + 1] && j < n - 1)
            B[j++] = A[i];
    B[j] = A[n - 1];
    for (int i = 0; i < n; i++)
        cout << B[i] << " ";
    cout << endl;
}
