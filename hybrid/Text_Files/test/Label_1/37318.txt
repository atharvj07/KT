for t in range(int(input())):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    res = sum(a)
    i = 0
    while i < n:
        if a[i] % x == 0:
            res += x * (a[i] // x - 1)
            a.append(a[i] // x)
            a.append(a[i] // x)
        else:
            break
        i += 1
    print(res)
