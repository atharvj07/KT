MOD = int(1e9 + 7)
s = input()
n = len(s)
pref = [0] * (n + 1)
for i in range(n):
    pref[i + 1] = pref[i]
    if s[i] == '(':
        pref[i + 1] += 1
dp = [[0] * (n + 1) for _ in range(n + 1)]
dp[0][0] = 1
for i in range(1, n + 1):
    for j in range(n + 1):
        dp[i][j] = dp[i - 1][j]
        if s[i - 1] == '(':
            dp[i][j] = (dp[i][j] + dp[i - 1][j - 1]) % MOD
        elif j > 0 and pref[i] - pref[i - 1] >= j:
            dp[i][j] = (dp[i][j] + dp[i - 1][j + 1]) % MOD
print(dp[n][0])
