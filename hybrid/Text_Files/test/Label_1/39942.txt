import sys
input = sys.stdin.readline
n = int(input())
lst = list(map(int, input().split()))
lst.sort()
ans = 0
for i in range(n):
    l, r = i, n - 1
    cnt = 0
    while l <= r:
        m = (l + r) // 2
        if lst[m] == lst[i]:
            cnt = m - i + 1
            l = m + 1
        else: r = m - 1
    ans += cnt / n
print(ans)
