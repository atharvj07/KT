import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int k = 0;
        for (int i = 0; i < n - 1; i++) {
            if (a[i + 1] - a[i] <= 1) {
                k++;
            }
        }
        if (a[n - 1] - a[0] <= 1) {
            k++;
        }
        System.out.println(k);
        if (k >= 2) {
            for (int i = 0; i < k; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println(a[0]);
        }
    }
}
