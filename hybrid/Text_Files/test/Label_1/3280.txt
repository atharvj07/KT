#include <iostream>
#include <unordered_set>
using namespace std;
unordered_set<string> s;
int main() {
    int n;
    string home;
    cin >> n >> home;
    for (int i = 0; i < n; i++) {
        string a, b;
        cin >> a >> b;
        if (b == home)
            s.erase(a);
        else
            s.insert(b);
    }
    cout << (s.size() ? "contest" : "home");
    return 0;
}
