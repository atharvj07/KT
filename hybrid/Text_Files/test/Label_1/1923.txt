#include <algorithm>
#include <iostream>
typedef long long ll;
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  ll a[n], sum[n + 1];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sum[0] = 0;
  for (int i = 0; i < n; i++) sum[i + 1] = sum[i] + a[i];
  ll ans = -1, b = 1;
  for (int i = 0; i <= n - 2 * k; i++) {
    a[i] = sum[i + k] - sum[i];
    if (ans == -1 || a[i] > ans) ans = a[i], b = i + 1;
  }
  cout << b << " " << b + k << endl;
}
