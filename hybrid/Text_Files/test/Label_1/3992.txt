#include <bits/stdc++.h>
using namespace std;
int main() {
    long long n;
    cin >> n;
    vector<int> ans;
    for (long long i = 30; i >= 0; i--) {
        if ((1 << i) & n) {
            ans.push_back(1 << i);
            n -= 1 << i;
        }
    }
    if (n) ans.push_back(n);
    cout << ans.size() << endl;
    for (long long i : ans) cout << i << " ";
    cout << endl;
    return 0;
}
