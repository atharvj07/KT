import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[]]
for i in range(n + 1):
    g.append([])

for _ in range(m):
    a, b = map(int, input().split())
    g[a].append(b)

vis = [False] * (n + 1)
vs = []

def dfs(x):
    if vis[x]:
        return
    vs.append(x)
    vis[x] = True
    for i in g[x]:
        dfs(i)

for i in range(1, n + 1):
    if vis[i] == False:
        dfs(i)

ans = []
vs.sort()
for v in vs:
    ans.append(str(v))

print(" ".join(ans))
