import queue
a,c=map(int,input().split())
b=list(map(int,input().split()))
d=[-1]*a
d[0]=0
q=queue.PriorityQueue()
q.put((0,0))
for i in range(a):
    j=q.get()
    if j[1]+b[i] <=c:
        d[i]=j[0]
        q.put((j[0],j[1]+b[i]))
    else:
        if q.empty() or -1==j[0]:
            break
        else:
            d[i]=j[0]+1
            q.put((j[0]+1,j[1]+b[i]-c))
print(*d)
