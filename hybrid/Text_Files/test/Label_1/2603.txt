#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const ll mod = 1e9 + 7;

#define vi(x) vector<x>
#define pb push_back
#define all(x) x.begin(), x.end()
#define rep(i, n) for (int i = 0; i < n; ++i)

int main() {
  int n;
  cin >> n;
  vi(int) a(n);
  rep(i, n) { cin >> a[i]; }
  ll sum_even = 0, sum_odd = 0;
  rep(i, n) {
    if (i % 2 == 0)
      sum_even += a[i];
    if (i + 1 < n && i % 2 == 0)
      sum_even -= a[i + 1];
    if (i % 2 == 1)
      sum_odd += a[i];
    if (i + 1 < n && i % 2 == 1)
      sum_odd -= a[i + 1];
  }
  cout << max(sum_even, sum_odd) << "\n";
  return 0;
}
