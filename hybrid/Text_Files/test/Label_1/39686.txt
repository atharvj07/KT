from collections import defaultdict

n = int(input())
ss = [input() for _ in range(n)]
m = int(input())
ops = [list(map(int, input().split())) for _ in range(m)]

MX = 100
for op in ops:
    a, b = op
    a -= 1
    b -= 1
    sa, sb = ss[a], ss[b]
    d = defaultdict(int)
    for i in range(len(sa)):
        for j in range(i+1, len(sa)+1):
            d[sa[i:j]] += 1
    for i in range(len(sb)):
        for j in range(i+1, len(sb)+1):
            d[sb[i:j]] += 1
    ans = 0
    for x in d:
        if d[x] == n:
            ans = max(ans, len(x))
    print(ans)
