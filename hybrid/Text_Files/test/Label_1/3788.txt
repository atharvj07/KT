#include <bits/stdc++.h>
using namespace std;
#define LL long long
int main() {
  LL n, k;
  cin >> n >> k;
  if (n % 2 == 1 || k == 1)
    cout << "No" << endl;
  else {
    vector<int> v;
    n >>= 1;
    for (int i = 0;; i++) {
      if (n % 2)
        v.push_back((i << 1) | 1);
      else
        v.push_back(i << 1);
      n >>= 1;
      if (!n)
        break;
    }
    if (v.size() <= k) {
      reverse(v.begin(), v.end());
      while (v.size() < k) {
        v.push_back(0);
      }
      cout << "Yes" << endl;
      for (int i : v)
        cout << i << " ";
      cout << endl;
    } else
      cout << "No" << endl;
  }
  return 0;
}
