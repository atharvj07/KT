#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 1;
char s[MAXN];
bool palin[MAXN];
int main()
{
  cin >> s + 1;
  for (int i = 1; i <= strlen(s + 1); i++)
  {
    if ((s[i] == 'A' && s[i + 1] == 'H') || (s[i] == 'H' && s[i + 1] == 'A') ||
        (s[i] == 'E' && s[i + 1] == 'H') || (s[i] == 'H' && s[i + 1] == 'E') ||
        (s[i] == 'I' && s[i + 1] == 'I') || (s[i] == 'O' && s[i + 1] == 'O') ||
        (s[i] == 'W' && s[i + 1] == 'W') || (s[i] == 'X' && s[i + 1] == 'X') ||
        (s[i] == 'Y' && s[i + 1] == 'Y') || (s[i] == '1' && s[i + 1] == '1') ||
        (s[i] == '8' && s[i + 1] == '8'))
    {
      i++;
      palin[i - 2] = true;
      palin[i] = true;
    }
  }

  bool ok = true;
  for (int i = 1; i <= strlen(s + 1) / 2; i++)
    ok &= (palin[i] && palin[strlen(s + 1) - i + 1]);
  if (ok)
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
