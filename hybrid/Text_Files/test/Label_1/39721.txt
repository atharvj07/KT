from collections import deque

q = int(input())
books = deque()
answers = []
for i in range(q):
    query = input().split()
    if query[0] == 'L':
        books.appendleft(int(query[1]))
    elif query[0] == 'R':
        books.append(int(query[1]))
    else:
        id = int(query[1])
        l = 0
        r = 0
        while books and books[0] != id:
            l += 1
            books.popleft()
        while books and books[-1] != id:
            r += 1
            books.pop()
        answers.append(min(l, r))
print(*answers, sep='\n')
