import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        int a = input.nextInt();
        int b = input.nextInt();
        int v = input.nextInt();
        System.out.println((int) Math.ceil( (double)(a) / ((k-1) * v - 1) ) + (b!=0 ? 1 : 0) );
    }
}
