import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int m = sc.nextInt();
		ArrayList<ArrayList<Integer>> g = new ArrayList<>();
		for (int i = 0; i < n; i++) {
			g.add(new ArrayList<>());
		}
		int[] vis = new int[n];
		for (int i = 0; i < m; i++) {
			int a = sc.nextInt();
			int b = sc.nextInt();
			g.get(a).add(b);
		}
		for (int i = 0; i < n; i++) {
			if (dfs(i, -1, g, vis)) {
				System.out.println(1);
				return;
			}
		}
		System.out.println(0);
	}
	static boolean dfs(int u, int p, ArrayList<ArrayList<Integer>> g, int[] vis) {
		if (vis[u] == 1) {
			return true;
		}
		vis[u] = 1;
		for (int v : g.get(u)) {
			if (v != p) {
				if (dfs(v, u, g, vis)) {
					return true;
				}
			}
		}
		vis[u] = 2;
		return false;
	}
}
