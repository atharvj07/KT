import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] d = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            d[i] = input.nextInt();
        }
        int a = input.nextInt() - 1;
        int b = input.nextInt() - 1;
        int ans = 0;
        for (int i = a; i < b; i++) {
            ans += d[i];
        }
        System.out.println(ans);
    }
}
