import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int[] rgb = new int[3];
    for (int i = 0; i < 3; i++) {
      rgb[i] = in.nextInt();
    }
    int[] sorted = new int[3];
    sorted[0] = rgb[0];
    sorted[1] = rgb[1];
    sorted[2] = rgb[2];
    Arrays.sort(sorted);
    int fights = sorted[2] - sorted[0] + sorted[2] - sorted[1];
    if (fights > sorted[0] + sorted[1])
      System.out.println(-1);
    else
      System.out.println(fights);
  }
}
