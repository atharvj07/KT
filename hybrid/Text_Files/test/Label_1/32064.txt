import java.util.Scanner;
import java.util.TreeSet;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        TreeSet<Integer> cafes = new TreeSet<>();
        for (int i = 0; i < n; i++) {
            int cafe = in.nextInt();
            cafes.add(cafe);
        }
        System.out.println(cafes.first());
    }
}
