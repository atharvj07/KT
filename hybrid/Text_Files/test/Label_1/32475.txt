import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] primes = Arrays.stream(in.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int k = Integer.parseInt(in.readLine());
        long answer = 1;
        for (int i = 0; i < k && answer <= 1e18; i++) {
            boolean valid = true;
            for (int prime : primes) {
                while (answer % prime == 0) answer /= prime;
            }
            answer++;
            for (int prime : primes) {
                valid &= answer % prime != 0;
            }
            if (valid) i++;
        }
        System.out.println(answer - 1);
    }
}
