import java.util.*;

class Main {
    static final int MAX_N = 500;
    static long[][] s = new long[MAX_N][MAX_N];
    static int[][] mark = new int[MAX_N][MAX_N];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                s[i][j] = sc.nextLong();
            }
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (mark[i][j] == 0) ans = Math.max(ans, dfs(i, j, n, m));
            }
        }
        System.out.println(ans);
    }

    static int[][] dir = { { 0, 1 }, { 1, 0 }, { 0, -1 }, { -1, 0 } };

    static long dfs(int x, int y, int n, int m) {
        long ans = s[x][y];
        mark[x][y] = 1;
        for (int k = 0, p = 0; k < 4; k++) {
            int dx = x + dir[k][0], dy = y + dir[k][1];
            while (dx >= 0 && dx < n && dy >= 0 && dy < m && mark[dx][dy] == 0) {
                ans += s[dx][dy];
                mark[dx][dy] = 1;
                p++;
                dx += dir[k][0];
                dy += dir[k][1];
            }
            if (p == 0) continue;
            dx -= dir[k][0];
            dy -= dir[k][1];
            p--;
            for (int l = (k + 1) % 4; l != k; l = (l + 1) % 4) {
                int curx = dx + dir[l][0], cury = dy + dir[l][1];
                if (curx < 0 || curx >= n || cury < 0 || cury >= m || mark[curx][cury] != 0) continue;
                ans = Math.max(ans, s[curx][cury] + dfs(curx, cury, n, m));
            }
        }
        return ans;
    }
}
