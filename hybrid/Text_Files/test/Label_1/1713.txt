#include <bits/stdc++.h>
#define ll long long
using namespace std;
int a[200005], b[200005], pt[200005];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++)
        b[i] = a[i];
    sort(b, b + n);
    pt[b[n / 2 - 1]] = 1;
    int j = (n % 2) ? n / 2 : n / 2 - 1;
    for (int i = j - 1; i >= 0; i--) { pt[b[i]] = -1; }
    for (int i = j + 1; i < n; i++) { pt[b[i]] = 2; }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (pt[a[i]] == 1) {
            ans += (i) * (n - i - 1);
        } else if (pt[a[i]] == 2) {
            ans += (i) * (n - i - 1);
        }
    }
    cout << ans << endl;
}
