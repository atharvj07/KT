import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        for (int i = 0; i < n; i++) {
            int x1 = input.nextInt(), y1 = input.nextInt(), a1 = input.nextInt(), b1 = input.nextInt();
            int x2 = input.nextInt(), y2 = input.nextInt(), a2 = input.nextInt(), b2 = input.nextInt();
            int x3 = input.nextInt(), y3 = input.nextInt(), a3 = input.nextInt(), b3 = input.nextInt();
            int x4 = input.nextInt(), y4 = input.nextInt(), a4 = input.nextInt(), b4 = input.nextInt();
            long dx1 = a1 - x1, dy1 = b1 - y1, dx2 = a2 - x2, dy2 = b2 - y2, dx3 = a3 - x3, dy3 = b3 - y3, dx4 = a4 - x4, dy4 = b4 - y4;
            if (dx1 * dy2 == dx2 * dy1 && dx2 * dy3 == dx3 * dy2 && dx3 * dy4 == dx4 * dy3) {
                System.out.println(-1);
            } else {
                int ans = 0;
                if (dx1 * dy2 != dx2 * dy1) {
                    ans++;
                    dx1 = -dy1;
                    dy1 = dx1;
                }
                if (dx2 * dy3 != dx3 * dy2) {
                    ans++;
                    dx2 = -dy2;
                    dy2 = dx2;
                }
                if (dx3 * dy4 != dx4 * dy3) {
                    ans++;
                    dx3 = -dy3;
                    dy3 = dx3;
                }
                System.out.println(ans);
            }
        }
    }
}
