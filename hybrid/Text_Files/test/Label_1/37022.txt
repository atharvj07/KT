import sys

def solve(n, packages):
    if packages[0][1] > 0:
        return "NO"
    packages.sort()
    path = []
    x, y = 0, 0
    for package in packages:
        x_dist = package[0] - x
        y_dist = package[1] - y
        path += ["R" * x_dist] + ["U" * y_dist]
        x, y = package[0], package[1]
    return "".join(path)

t = int(input())
for _ in range(t):
    n = int(input())
    packages = [list(map(int, input().split())) for _ in range(n)]
    result = solve(n, packages)
    print("YES" if result else "NO")
    if result:
        print(result)
