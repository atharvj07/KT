import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Person[] people = new Person[n];
        for (int i = 0; i < n; i++) {
            people[i] = new Person(sc.next(), sc.nextInt());
        }
        boolean possible = true;
        int[] heights = new int[n];
        int[] usedHeights = new int[1000001];
        int[] cnt = new int[n];
        for (Person p : people) {
            if (p.taller == 0) {
                // set the height of the person with 0 taller as 1
                heights[p.index] = 1;
                usedHeights[1]++;
                cnt[p.index]++;
            }
        }
        for (int i = 0; i < n; i++) {
            // find the smallest unused height for the person
            for (int j = 1; j <= 1000000; j++) {
                if (usedHeights[j] < cnt[i]) {
                    heights[i] = j;
                    usedHeights[j]++;
                    cnt[i]--;
                    break;
                }
            }
            if (usedHeights[heights[i]] > cnt[i]) {
                possible = false;
                break;
            }
        }
        if (!possible) {
            System.out.println(-1);
            return;
        }
        for (Person p : people) {
            System.out.println(p.name + " " + heights[p.index]);
        }
    }

    static class Person {
        String name;
        int taller;
        int index;

        Person(String name, int taller) {
            this.name = name;
            this.taller = taller;
        }

        public void setIndex(int index) {
            this.index = index;
        }
    }
}
