import java.util.Scanner;

public class Main {
    static int n, x, y, min = Integer.MAX_VALUE, idx;
    static int[] X;
    static int[] Y;
    static StringBuilder sb;
    static boolean check;

    static void dfs(int cur, String str) {
        if (check)
            return;
        if (cur == n) {
            if (str.length() < min) {
                min = str.length();
                idx = cur;
                sb = new StringBuilder(str);
            } else if (str.length() == min) {
                if (str.compareTo(sb.toString()) < 0) {
                    min = str.length();
                    idx = cur;
                    sb = new StringBuilder(str);
                }
            }
            check = true;
            return;
        }
        dfs(cur + 1, str + "U");
        dfs(cur + 1, str + "R");
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            sb = new StringBuilder();
            n = sc.nextInt();
            X = new int[n];
            Y = new int[n];
            for (int i = 0; i < n; i++) {
                X[i] = sc.nextInt();
                Y[i] = sc.nextInt();
            }
            min = Integer.MAX_VALUE;
            idx = -1;
            check = false;
            dfs(0, "");
            if (check)
                System.out.println("YES\n" + sb.toString());
            else
                System.out.println("NO");
        }
        sc.close();
    }
}
