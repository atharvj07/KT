#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int min_diff = 2e9;
    int cnt = 0;
    for (int i = 1; i < n; i++) {
        int diff = abs(a[i] - a[i - 1]);
        if (diff < min_diff) {
            min_diff = diff;
            cnt = 1;
        } else if (diff == min_diff) {
            cnt++;
        }
    }
    cout << min_diff << " " << cnt << endl;
    return 0;
}
