#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(1 << n);
  for (int i = 2; i < (1 << n); i++) { cin >> a[i]; }

  int ans = 0;
  for (int i = (1 << (n - 1)); i < (1 << n); i++) {
    if (a[i] != a[i ^ (1 << (n - 1))]) {
      ans += min(a[i], a[i ^ (1 << (n - 1))]);
      a[i] = a[i ^ (1 << (n - 1))] =
          max(a[i], a[i ^ (1 << (n - 1))]);
    }
  }
  cout << ans << endl;
  return 0;
}
