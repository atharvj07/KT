import java.util.*;

public class Main {
    static int[] arr;
    static int[][] dp;
    static int[] t;
    static String a, b;
    static int n, m, ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        t = new int[4];
        int i, j, k;
        for (i = 0; i < 4; i++) {
            t[i] = sc.nextInt();
        }
        a = sc.nextLine();
        while (a.isEmpty()) {
            a = sc.nextLine();
        }
        b = sc.nextLine();
        while (b.isEmpty()) {
            b = sc.nextLine();
        }
        n = a.length();
        m = b.length();
        arr = new int[n + 1];
        for (i = 1; i <= n; i++) {
            arr[i] = a.charAt(i - 1) - 'a';
        }
        dp = new int[n + 1][m + 1];
        ans = Integer.MAX_VALUE;
        for (i = n; i >= 0; i--) {
            for (j = m; j >= 0; j--) {
                dp[i][j] = Integer.MAX_VALUE;
                if (i == n && j == m) {
                    dp[i][j] = 0;
                } else if (i == n) {
                    dp[i][j] = Math.min(dp[i][j], dp[i][j + 1] + t[1]);
                } else if (j == m) {
                    dp[i][j] = Math.min(dp[i][j], dp[i + 1][j] + t[0]);
                } else if (arr[i + 1] == b.charAt(j) - 'a') {
                    dp[i][j] = Math.min(dp[i][j], dp[i + 1][j]);
                } else {
                    dp[i][j] = Math.min(dp[i][j], dp[i + 1][j] + t[0]);
                    dp[i][j] = Math.min(dp[i][j], dp[i][j + 1] + t[1]);
                    dp[i][j] = Math.min(dp[i][j], dp[i + 1][j + 1] + t[2]);
                    if (i + 1 < n && arr[i + 1] == arr[i + 2] && i + 2 < n && arr[i + 2] == b.charAt(j) - 'a') {
                        dp[i][j] = Math.min(dp[i][j], dp[i + 3][j]);
                    }
                    if (j + 1 < m && b.charAt(j) - 'a' == b.charAt(j + 1) - 'a' && i + 1 < n && arr[i + 1] == b.charAt(j + 1) - 'a') {
                        dp[i][j] = Math.min(dp[i][j], dp[i + 1][j + 2]);
                    }
                    if (i + 1 < n && j + 1 < m && arr[i + 1] == b.charAt(j) - 'a' && arr[i + 1] == b.charAt(j + 1) - 'a') {
                        dp[i][j] = Math.min(dp[i][j], dp[i + 1][j + 1] + t[3]);
                    }
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
