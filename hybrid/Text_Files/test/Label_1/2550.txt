#include <bits/stdc++.h>
using namespace std;
const int mx = 3e5 + 5;
const long long mod = 1000000007LL;
int main() {
  string s;
  cin >> s;
  int n = s.length();
  long long pw10[n + 5];
  pw10[0] = 1;
  for (int i = 1; i <= n; i++) pw10[i] = (pw10[i - 1] * 10) % 4;
  vector<long long> hsh[4];
  for (int i = 0; i < 4; i++) hsh[i].resize(n + 5, 0);
  for (int i = n - 1; i >= 0; i--) {
    hsh[0][i] = (hsh[0][i + 1] * 10 + (s[i] - '0')) % 4;
    hsh[1][i] = (hsh[1][i + 1] * 10 + (s[i] - '0')) % 4;
    hsh[2][i] = (hsh[2][i + 1] * 10 + (s[i] - '0')) % 4;
    hsh[3][i] = (hsh[3][i + 1] * 10 + (s[i] - '0')) % 4;
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (hsh[0][i] == 0) ans++;
    if (hsh[1][i] == 0) ans++;
    if (hsh[2][i] == 0) ans++;
    if (hsh[3][i] == 0) ans++;
    if (i + 1 < n) {
      if (hsh[0][i + 1] == hsh[0][i] * 10 % 4) ans++;
      if (hsh[1][i + 1] == (hsh[1][i] * 10 + (s[i] - '0')) % 4) ans++;
      if (hsh[2][i + 1] == (hsh[2][i] * 10 + (s[i] - '0')) % 4) ans++;
      if (hsh[3][i + 1] == (hsh[3][i] * 10 + (s[i] - '0')) % 4) ans++;
      for (int j = i + 2; j < n; j++) {
        if (hsh[0][j] == (hsh[0][i + 1] * 10 + (s[i] - '0')) % 4) ans++;
        if (hsh[1][j] == (hsh[1][i + 1] * pw10[j - i - 1] + (s[i] - '0')) % 4) ans++;
        if (hsh[2][j] == (hsh[2][i + 1] * pw10[j - i - 1] + (s[i] - '0')) % 4) ans++;
        if (hsh[3][j] == (hsh[3][i + 1] * pw10[j - i - 1] + (s[i] - '0')) % 4) ans++;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
