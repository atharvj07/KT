import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        boolean[] stable = new boolean[n];
        int minSteps = -1;
        int[] stableSeq = null;

        for (int steps = 0; minSteps < 0; steps++) {
            boolean changed = false;
            int[] b = new int[n];
            for (int i = 1; i < n - 1; i++) {
                b[i] = median(a[i - 1], a[i], a[i + 1]);
                if (b[i] != a[i]) {
                    changed = true;
                }
            }
            if (!changed) {
                minSteps = steps;
                stableSeq = b;
            } else {
                a = b;
            }
        }

        if (minSteps >= 0) {
            System.out.println(minSteps);
            for (int i = 0; i < n; i++) {
                System.out.print(stableSeq[i] + " ");
            }
        } else {
            System.out.println(1);
        }
    }

    private static int median(int a, int b, int c) {
        if (a > b) {
            if (b > c) {
                return b;
            } else if (a > c) {
                return c;
            } else {
                return a;
            }
        } else {
            if (a > c) {
                return a;
            } else if (b > c) {
                return c;
            } else {
                return b;
            }
        }
    }
}
