import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        int[] deg = new int[n + 1];
        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = input.nextInt(), b = input.nextInt();
            edges.add(new int[]{a, b});
            deg[a]++; deg[b]++;
        }
        int s1 = input.nextInt(), t1 = input.nextInt(), l1 = input.nextInt();
        int s2 = input.nextInt(), t2 = input.nextInt(), l2 = input.nextInt();
        int[] dis1 = dijkstra(s1, n, edges, deg);
        if (dis1[t1] > l1) {
            System.out.println(-1);
            return;
        }
        int[] dis2 = dijkstra(s2, n, edges, deg);
        if (dis2[t2] > l2) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        for (int[] edge : edges) {
            int a = edge[0], b = edge[1];
            int[] dis = dijkstra(s1, n, edges, deg);
            if (dis[t1] > l1 || dis[t2] > l2) {
                ans++;
                deg[a]--; deg[b]--;
            }
        }
        System.out.println(ans);
    }
    public static int[] dijkstra(int s, int n, List<int[]> edges, int[] deg) {
        int[] dis = new int[n + 1];
        Arrays.fill(dis, Integer.MAX_VALUE);
        dis[s] = 0;
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[1] - b[1]);
        pq.offer(new int[]{s, 0});
        while (!pq.isEmpty()) {
            int[] cur = pq.poll();
            int u = cur[0], d = cur[1];
            if (d > dis[u]) continue;
            for (int[] edge : edges) {
                int a = edge[0], b = edge[1];
                int w = 1;
                if (deg[a] == 1 || deg[b] == 1) w = 0;
                if (a == u) {
                    if (dis[b] > d + w) {
                        dis[b] = d + w;
                        pq.offer(new int[]{b, d + w});
                    }
                } else if (b == u) {
                    if (dis[a] > d + w) {
                        dis[a] = d + w;
                        pq.offer(new int[]{a, d + w});
                    }
                }
            }
        }
        return dis;
    }
}
