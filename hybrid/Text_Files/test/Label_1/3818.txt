#include <bits/stdc++.h>
using namespace std;

int32_t main() {
  int k;
  cin >> k;
  if (k % 3 == 0) {
    for (int i = 0; i < k / 3; i++) {
      cout << "aeiou";
    }
  } else if (k % 4 == 0) {
    for (int i = 0; i < k / 4; i++) {
      cout << "aeiou";
    }
    cout << "a";
  } else if (k % 5 == 0) {
    for (int i = 0; i < k / 5; i++) {
      cout << "aioeu";
    }
  } else if (k % 6 == 0) {
    for (int i = 0; i < k / 6; i++) {
      cout << "aioeu";
    }
    cout << "a";
  } else {
    cout << -1;
  }
  return 0;
}
