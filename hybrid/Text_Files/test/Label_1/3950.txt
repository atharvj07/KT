#include <bits/stdc++.h>
 
using namespace std;
 
typedef long long ll;
const int MAXN = 1e5 + 10;
map<ll, map<ll, ll>> mp;
vector<pair<ll, ll>> ops;
int main() {
    ll n;
    cin >> n;
    while(n--) {
        ll t, x;
        int op;
        cin >> op >> t >> x;
        ops.push_back({t, op});
        if(op == 1)  mp[t][x]++;
        if(op == 2)  mp[t][x]--;
        if(op == 3)  cout << mp[t][x] << '\n';
    }
    return 0;
}
