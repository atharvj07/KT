for _ in range(int(input())):
    n, l, r, s = map(int, input().split())
    if n * (n + 1) // 2 < s or s % (r - l + 1):
        print(-1)
        continue
    arr = [i + 1 for i in range(l - 1)]
    arr.append((s - (r - l + 1) * (arr[l - 2] or 0)) // (r - l + 1))
    for i in range(l + 1, r + 1):
        arr.append(arr[i - 2] + 1)
    arr.extend([i + 1 for i in range(r, n)])
    print(*arr)
