import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		long n = sc.nextLong(), x = sc.nextLong();
		if (x < n * 9) {
			System.out.println(-1);
			return;
		}
		for (long b = x; b >= n; b--) {
			long d = (long) Math.pow(b, n) - 1;
			if (d % (b - 1) == 0) {
				System.out.println(b);
				return;
			}
		}
		System.out.println(-1);
	}
}
