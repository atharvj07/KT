import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws java.lang.Exception {
        java.util.Scanner input = new java.util.Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 1; i < n; i++) {
            ans += a[i] - a[i - 1];
        }
        System.out.println(ans);
    }
}
