import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong();
        long b = in.nextLong();
        long s = in.nextLong();
        if (Math.abs(a) + Math.abs(b) > s || (Math.abs(a) + Math.abs(b) - s) % 2 != 0) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
        }
        in.close();
    }
}
