import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[][] c = new int[3][3];
    for (int i = 0; i < 3; i++) {
      for (int j = 0; j < 3; j++) {
        c[i][j] = sc.nextInt();
      }
    }
    sc.close();

    int[] a = new int[3], b = new int[3];
    for (int i = 0; i < 3; i++) {
      a[i] = c[i][0];
      b[i] = c[0][i];
    }

    boolean ok = true;
    for (int i = 0; i < 3; i++) {
      for (int j = 0; j < 3; j++) {
        ok &= c[i][j] == a[i] + b[j];
      }
    }

    System.out.println(ok ? "Yes" : "No");
  }
}
