import java.util.Scanner;

public class GregAndArray {

    private static int[] performQuery(int[] a, int[] operation, int[] query) {
        for (int i = operation[0] - 1; i <= operation[1] - 1; i++) {
            a[i] += operation[2];
        }
        return a;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[][] operations = new int[m][3];
        for (int i = 0; i < m; i++) {
            operations[i][0] = scanner.nextInt();
            operations[i][1] = scanner.nextInt();
            operations[i][2] = scanner.nextInt();
        }

        int[][] queries = new int[k][2];
        for (int i = 0; i < k; i++) {
            queries[i][0] = scanner.nextInt();
            queries[i][1] = scanner.nextInt();
        }

        for (int i = 0; i < k; i++) {
            for (int j = queries[i][0] - 1; j <= queries[i][1] - 1; j++) {
                a = performQuery(a, operations[j], queries[i]);
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
