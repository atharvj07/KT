import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    long k = scan.nextLong();
    long a = scan.nextLong();
    long b = scan.nextLong();
    long put = a / k;
    long upon = b / k;
    if (upon < put) {
      System.out.println(0);
    } else {
      System.out.println(upon - put + 1);
    }
  }
}
