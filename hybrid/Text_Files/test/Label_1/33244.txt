import java.util.*;
import java.io.*;
class Main {
    long mod = 998244353;
    long power(long x, long y) {
        long res = 1;
        x %= mod;
        while (y > 0) {
            if((y & 1) == 1)
                res = (res * x) % mod;
            x = (x * x) % mod;
            y >>= 1;
        }
        return res;
    }
    void solve() {
        int n, k;
        long ans;
        Scanner sc = new Scanner(System.in);
        String[] s = sc.nextLine().split(" ");
        n = Integer.parseInt(s[0]);
        k = Integer.parseInt(s[1]);
        if(2 * k > n * (n - 1))
            ans = 0;
        else if((n - 1) % 2 == k % 2)
            ans = power(2, n) - power(2, n - 1 - k);
        else
            ans = 0;
        ans %= mod;
        ans += mod;
        ans %= mod;
        System.out.println(ans);
    }
    public static void main(String[] args) {
        Main obj = new Main();
        obj.solve();
    }
}
