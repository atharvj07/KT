import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long maxScore = 0;
        for (int i = 0; i + k - 1 < n; i++) {
            long score = 0;
            for (int j = i; j < i + k; j++) {
                score += Math.abs(a[j]);
            }
            maxScore = Math.max(maxScore, score);
        }
        System.out.println(maxScore);
    }
}
