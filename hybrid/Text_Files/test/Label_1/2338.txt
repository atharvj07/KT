#include <algorithm>
#include <cstdio>
#include <vector>
using namespace std;

typedef long long ll;

const int MAXN = 50000 + 5;
int n, p[MAXN][3], ans[MAXN / 2][2];
vector<int> g[MAXN];

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%d%d%d", &p[i][0], &p[i][1], &p[i][2]);
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            bool ok = 1;
            for (int k = 0; k < 3; k++) {
                ok &= p[i][k] > p[j][k] || p[i][k] < p[j][k];
            }
            if (ok) {
                g[i].push_back(j);
                g[j].push_back(i);
            }
        }
    }
    vector<int> vis(n, 0);
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        if (vis[i]) continue;
        vector<int> need;
        int sz = g[i].size();
        for (int j = 0; j < sz; j++) {
            if (!vis[g[i][j]]) need.push_back(g[i][j]);
        }
        sort(need.begin(), need.end());
        need.resize(unique(need.begin(), need.end()) - need.begin());
        for (int j = need.size() - 1; j >= 0; j--) {
            vis[need[j]] = 1;
            ans[cnt][0] = i;
            ans[cnt][1] = need[j];
            cnt++;
        }
    }
    for (int i = 0; i < n / 2; i++) {
        printf("%d %d\n", ans[i][0] + 1, ans[i][1] + 1);
    }
    return 0;
}
