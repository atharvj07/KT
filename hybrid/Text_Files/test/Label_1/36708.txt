import sys

def solve():
    for line in sys.stdin:
        arr = list(map(int, line.split()))
        if len(arr) == 1:
            if arr[0] == 0:
                break
            else:
                res = ['YES']
                n = arr[0]
                a = list(map(int, sys.stdin.readline().split()))
                k = int(sys.stdin.readline())
                b = list(map(int, sys.stdin.readline().split()))
                i = 0
                j = 0
                while i < n:
                    if a[i] == b[j]:
                        i += 1
                        j += 1
                    else:
                        if i == 0 or a[i] > a[i - 1]:
                            res.append(str(i + 1) + ' R')
                        else:
                            res.append(str(i + 1) + ' L')
                        a[i] += a[i - 1]
                        a.pop(i - 1)
                        n -= 1
                print('\n'.join(res))
        else:
            continue

solve()
