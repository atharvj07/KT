import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt(), b = sc.nextInt();
        int n = sc.nextInt();
        if (n == 0) {
            System.out.print(0);
            return;
        }
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = sc.nextInt();
            arr[i][1] = sc.nextInt();
        }
        Arrays.sort(arr, Comparator.comparingInt(x -> x[0]));
        for (int[] r : arr) {
            if (Math.max(a, r[0]) < Math.min(b, r[1])) {
                System.out.print(1);
                return;
            }
        }
        System.out.print(0);
    }
}
