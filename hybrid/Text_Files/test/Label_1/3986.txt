#include <iostream>
#include <cmath>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        unsigned long long n;
        cin >> n;
        long long ans = (long long)sqrt(n) + (long long)cbrt(n) - (long long)pow(cbrt(n), 2);
        cout << ans << endl;
    }
    return 0;
}
