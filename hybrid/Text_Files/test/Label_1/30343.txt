import java.io.*;
import java.util.*;

public class Main {
    static class Creature {
        int[] characteristics;

        Creature(int[] characteristics) {
            this.characteristics = characteristics;
        }
    }

    static class Query {
        int type;
        int a;
        int b;

        Query(int type, int a, int b) {
            this.type = type;
            this.a = a;
            this.b = b;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));

        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int k = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());

        Creature[] creatures = new Creature[k + q];

        for (int i = 0; i < k; i++) {
            st = new StringTokenizer(br.readLine());
            int[] characteristics = new int[n];
            for (int j = 0; j < n; j++) {
                characteristics[j] = Integer.parseInt(st.nextToken());
            }
            creatures[i] = new Creature(characteristics);
        }

        List<Query> queries = new ArrayList<>();

        for (int i = 0; i < q; i++) {
            st = new StringTokenizer(br.readLine());
            int type = Integer.parseInt(st.nextToken());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            queries.add(new Query(type, a, b));
        }

        int creatureCount = k;

        for (Query query : queries) {
            if (query.type == 1) {
                int[] characteristics = new int[n];
                for (int i = 0; i < n; i++) {
                    characteristics[i] = Math.max(creatures[query.a].characteristics[i], creatures[query.b].characteristics[i]);
                }
                creatures[creatureCount++] = new Creature(characteristics);
            } else if (query.type == 2) {
                int[] characteristics = new int[n];
                for (int i = 0; i < n; i++) {
                    characteristics[i] = Math.min(creatures[query.a].characteristics[i], creatures[query.b].characteristics[i]);
                }
                creatures[creatureCount++] = new Creature(characteristics);
            } else {
                bw.write(creatures[query.a].characteristics[query.b] + "\n");
            }
        }

        bw.flush();
        bw.close();
        br.close();
    }
}
