import java.util.*;

public class Main {
  static ArrayList<Integer>[] g;
  static int n, k, ans = 0;
  static int[] visited = new int[100001];
  static Queue<Integer> bfs = new LinkedList<>();

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    int q = sc.nextInt();
    k = sc.nextInt();
    g = new ArrayList[n + 1];
    for (int i = 0; i <= n; i++) {
      g[i] = new ArrayList<Integer>();
    }
    while (q-- > 0) {
      int cmd = sc.nextInt();
      if (cmd == 1) {
        int a = sc.nextInt();
        int b = sc.nextInt();
        g[a].add(b);
        g[b].add(a);
      } else if (cmd == 2) {
        visited = new int[100001];
        int v = sc.nextInt();
        bfs.add(v);
        visited[v] = 1;
        ans = 1;
        while (!bfs.isEmpty()) {
          int a = bfs.poll();
          for (int i = 0; i < g[a].size(); i++) {
            int b = g[a].get(i);
            if (visited[b] == 0) {
              bfs.add(b);
              visited[b] = 1;
              ans++;
            }
          }
        }
        System.out.println(ans);
      } else {
        bfs.clear();
      }
    }
  }
}
