#include <bits/stdc++.h>
using namespace std;
int p[1000005],b[1000005];
int main(){
    int n;cin>>n;
    for(int i=1;i<=n;i++){cin>>b[i];p[b[i]]=i;}
    long long ans=0,M=1;
    for(int i=2;i<=n;i++){
        if(b[M]<b[i]){ans+=b[M];M=i;}
    }
    cout<<(ans%=(int)(1e9+7))<<endl;
    return 0;
}
