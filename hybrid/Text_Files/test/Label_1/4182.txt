#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod=1e9+7;
int main(){
    ll n,ans=0;
    cin>>n;
    vector<ll>v(n+1);
    for(int i=1;i<=n;i++) cin>>v[i];
    ll sum=0,cnt=0;
    for(int i=1;i<=n;i++){
        if(v[i]>v[i-1]) sum+=v[i-1],cnt++;
        ans+=sum*cnt;
        ans%=mod;
    }
    cout<<ans<<'\n';
    return 0;
}
