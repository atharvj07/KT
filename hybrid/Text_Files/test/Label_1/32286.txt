import java.util.Arrays;
import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int citizens = scanner.nextInt();
        int[] initialBalance = new int[citizens];
        for (int i = 0; i < citizens; i++) {
            initialBalance[i] = scanner.nextInt();
        }
        int events = scanner.nextInt();
        for (int i = 0; i < events; i++) {
            int type = scanner.nextInt();
            int x = scanner.nextInt();
            if (type == 1) {
                initialBalance[x - 1] = x;
            } else {
                for (int j = 0; j < citizens; j++) {
                    if (initialBalance[j] < x) {
                        initialBalance[j] = x;
                    }
                }
            }
        }
        System.out.println(Arrays.toString(initialBalance).replace("[", "").replace("]", "").replace(",", ""));
    }
}
