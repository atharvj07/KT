import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1000000000000000000L;

    static long[][] dp;
    static int[] x;

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            x = new int[n];
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int i = 0; i < n; i++) {
                x[i] = Integer.parseInt(st.nextToken());
            }
            dp = new long[n][100000];
            for (int i = 0; i < n; i++) {
                Arrays.fill(dp[i], -1);
            }

            for (int i = 0; i < n; i++) {
                System.out.println(solve(i, 0));
            }
        }
    }

    static long solve(int k, int sum) {
        if (k == 0) {
            return sum == 0 ? 1 : 0;
        }
        if (dp[k][sum] >= 0) {
            return dp[k][sum];
        }
        long ans = 0;
        for (int i = 0; i < x.length; i++) {
            ans += solve(k - 1, (sum + x[i]) % 100000);
            ans %= MOD;
        }
        return dp[k][sum] = ans;
    }
}
