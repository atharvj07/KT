import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int M = scanner.nextInt();
        int X = scanner.nextInt();

        int[][] costs = new int[N][M];
        int[] items = new int[N];
        for (int i = 0; i < N; i++) {
            costs[i][0] = scanner.nextInt();
            items[i] = scanner.nextInt();
            for (int j = 1; j < M; j++) {
                costs[i][j] = scanner.nextInt();
            }
        }

        int minCost = -1;
        for (int i = 0; i < (1 << N); i++) {
            int totalItems = 0;
            int totalCost = 0;
            for (int j = 0; j < N; j++) {
                if (((i >> j) & 1) == 1) {
                    totalItems += items[j];
                    totalCost += costs[j][0];
                    for (int k = 1; k < M; k++) {
                        if (costs[j][k] < costs[j][0]) {
                            totalItems += k;
                            totalCost += costs[j][k];
                            break;
                        }
                    }
                }
            }
            if (totalItems >= X && (minCost == -1 || totalCost < minCost)) {
                minCost = totalCost;
            }
        }

        System.out.println(minCost);
    }
}
