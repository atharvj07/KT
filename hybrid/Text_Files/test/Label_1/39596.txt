def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

s = sorted(map(int, input().split()))
k = len(s)
for i in range(k):
    for j in range(i + 1):
        g = s[i]
        for k in range(i, j, -1):
            g = gcd(g, s[k])
        if g > 1:
            print(-1)
            exit()
for i in range(1, k):
    if s[i] % s[i - 1] == 0:
        print(-1)
        exit()
print(k)
print(*s)
