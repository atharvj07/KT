#include <vector>
#include <iostream>
#include <algorithm>

int main() {
  int n; std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }
  std::vector<int> b(n);

  int cnt = 0;
  bool f = true;
  while (f) {
    f = false;
    for (int i = 1; i < n - 1; i++) {
      b[i] = (a[i - 1] + a[i] + a[i + 1]) / 2;
      if (b[i] != a[i]) f = true;
    }
    b[0] = a[0], b[n - 1] = a[n - 1];
    a = b;
    cnt++;
  }
  if (f) std::cout << -1 << std::endl;
  else {
    std::cout << cnt - 1 << std::endl;
    for (int i = 0; i < n; i++) std::cout << a[i] << ' ';
    std::cout << std::endl;
  }
  return 0;
}
