import sys

h = int(input())
a = [int(x) for x in input().split()]
c = [1]
for i in range(1, h+1):
    for j in range(c[-1], c[-1]+a[i-1]):
        c.append(c[-1]+1)
    c[-1] += 1
if a[0] == 1:
    print('perfect')
else:
    print('ambiguous')
    print(' '.join(str(i) for i in c))
    print(' '.join(str(i+1 if i > 0 else 0) for i in c[:-1]))
