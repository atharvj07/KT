import java.util.*;
import java.lang.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][][] f = new int[n + 1][6][3];
        long[][] p = new long[n + 1][11];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= 5; j++) {
                for (int k = 1; k <= 2; k++) {
                    f[i][j][k] = sc.nextInt();
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= 10; j++) {
                p[i][j] = sc.nextInt();
            }
        }
        long[][][] dp = new long[n + 1][6][11];
        dp[0][0][0] = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= 5; j++) {
                for (int k = 0; k <= 10; k++) {
                    for (int l = 0; l <= 2; l++) {
                        if (f[i][j][l] == 1 && (j == 0 || f[i][j - 1][l] == 1)) {
                            dp[i][j][k] = Math.max(dp[i][j][k], dp[i - 1][j][k]);
                            dp[i][j][k] = Math.max(dp[i][j][k], dp[i - 1][j - 1][k] + p[i][k]);
                        }
                    }
                }
            }
        }
        long ans = 0;
        for (int j = 0; j <= 5; j++) {
            for (int k = 0; k <= 10; k++) {
                ans = Math.max(ans, dp[n][j][k]);
            }
        }
        System.out.println(ans);
    }
}
