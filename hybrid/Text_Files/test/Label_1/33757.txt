import java.util.*;

public class Main {
    public static int[] sieve(int n) {
        int[] primes = new int[n + 1];
        Arrays.fill(primes, 1);
        primes[0] = primes[1] = 0;
        for (int i = 2; i * i <= n; i++) {
            if (primes[i] == 1)
                for (int j = i * i; j <= n; j += i) {
                    primes[j] = 0;
                }
        }
        return primes;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n], pf = new int[1 << 19], primes = sieve(1000000);
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            int cnt = 0;
            for (int j = 2; j * j <= a[i]; j++) {
                if (a[i] % j == 0) {
                    cnt++;
                    while (a[i] % j == 0) a[i] /= j;
                }
            }
            if (a[i] > 1) cnt++;
            pf[cnt]++;
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int l = sc.nextInt() - 2, r = sc.nextInt() - 2;
            long ans = 0;
            for (int j = 0; j <= 18; j++) {
                ans += (long) pf[j] * (r - l + 1 - (r - l + 1 - pf[j]));
            }
            System.out.println(ans);
        }
    }
}
