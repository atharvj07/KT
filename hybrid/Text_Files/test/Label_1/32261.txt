import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        long[] sum = new long[n];
        long[] xor = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
            if (i > 0) {
                sum[i] = sum[i - 1] + a[i];
            } else {
                sum[i] = a[i];
            }
        }
        xor[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            xor[i] = xor[i + 1] ^ a[i];
        }
        long ans = 0;
        for (int i = 1; i < n - 1; i++) {
            if ((sum[i - 1] ^ xor[i + 1]) == a[i]) {
                ans += n - i - 1;
            }
        }
        System.out.println(ans);
    }
}
