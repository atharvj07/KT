import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int R = sc.nextInt();
            int C = sc.nextInt();
            if (R == 0 && C == 0) {
                break;
            }
            int[][] A = new int[R][C];

            for (int i = 0; i < R; i++) {
                for (int j = 0; j < C; j++) {
                    A[i][j] = sc.nextInt();
                }
            }
            int num = 0;

            for (int i = 0; i < R; i++) {
                for (int j = 0; j < C; j++) {
                    if (A[i][j] == 1) {
                        continue;
                    }
                    num++;
                    for (int k = i + 1; k < R; k++) {
                        if (A[k][j] == 0) {
                            num++;
                        } else {
                            break;
                        }
                    }
                    for (int k = j + 1; k < C; k++) {
                        if (A[i][k] == 0) {
                            num++;
                        } else {
                            break;
                        }
                    }
                }
            }

            System.out.println(num);
        }
    }
}
