import java.util.Scanner;

public class Main {

    private static final long MOD = 1000000007;
    private static long[][] dp;  // dp[l][r] = count of teams with players in range [l, r]
    private static int canBePlayer[];  // canBePlayer[i] = 1 if soldier i can be a player

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] skills = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            skills[i] = in.nextInt();
        }
        int m = in.nextInt();
        canBePlayer = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            canBePlayer[i] = 1;
        }
        dp = new long[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            dp[i][i] = 1;
        }
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (skills[j] == skills[i]) {
                    dp[i][j] = (dp[i][j - 1] * 2) % MOD;
                    if (i + 1 < j) {
                        dp[i][j] = (dp[i][j] + dp[i + 1][j - 1]) % MOD;
                    }
                }
            }
        }
        for (int i = 0; i < m; i++) {
            int t = in.nextInt();
            int x = in.nextInt();
            if (t == 1) {
                canBePlayer[x] = 0;
            } else {
                canBePlayer[x] = 1;
            }
            long ans = 0;
            for (int j = 1; j <= n; j++) {
                if (canBePlayer[j] == 1) {
                    ans = (ans + dp[j][n]) % MOD;
                }
            }
            System.out.println(ans);
        }
    }
}
