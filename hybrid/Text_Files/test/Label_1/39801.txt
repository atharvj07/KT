rooms = [input() for _ in range(6)]
max_conv = 0
max_idx = (0, 0)
for i in range(6):
    for j in range(3):
        if rooms[i][2 * j] == '.' and rooms[i][2 * j + 1] == '.':
            room_conv = (6 - i) * 3 - abs(j - 1)
            if room_conv > max_conv:
                max_conv = room_conv
                max_idx = (i, j)
rooms[max_idx[0]] = rooms[max_idx[0]][:2 * max_idx[1]] + 'P' + rooms[max_idx[0]][2 * max_idx[1] + 1:]
for room in rooms:
    print(room)
