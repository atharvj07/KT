#include<bits/stdc++.h>
using namespace std;
#define ll long long int
int main(){
    string s;
    cin>>s;
    ll mod64 = 0;
    ll power = 1;
    for(ll i=s.length()-1; i>=0; i--){
        int digit = s[i]-48;
        mod64 = (mod64 + digit*power)%64;
        power = (power*2)%64;
    }
    if(mod64==0) cout<<"yes"<<endl;
    else cout<<"no"<<endl;
    return 0;
}
