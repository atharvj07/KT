import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            char[] arr = s.toCharArray();
            for (int i = 0; i < arr.length; i++) {
                if (i == arr.length - 1) {
                    arr[i] = 'a';
                } else {
                    arr[i] = 'z';
                }
            }
            System.out.println(new String(arr));
        }
    }
}
