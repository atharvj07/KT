import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = in.nextInt();
    }
    int sum = 0;
    for (int i = 0; i < n; i++) {
      sum += arr[i];
    }
    if (sum == 0) {
      System.out.println("NO");
    } else {
      System.out.println("YES");
      int cnt = 0;
      int l = 1;
      for (int i = 0; i < n; i++) {
        sum -= arr[i];
        if (sum == 0) {
          System.out.println(l + " " + (i + 1));
          l = i + 2;
          cnt++;
          sum = arr[i + 1];
        }
      }
      System.out.println(cnt);
    }
  }
}
