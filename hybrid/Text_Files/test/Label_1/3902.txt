#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;
const int mod = 998244353;

ll po (ll a, ll p) {
	if (p == 0) return 1LL;
	if (p == 1) return a;
	ll t = po (a, p / 2);
	if (p % 2 == 0) return (t * t) % mod;
	else return (t * t % mod) * a % mod;
}

int main() {
	ll n, a, b, c, d, x, y; cin >> n;
	vector < pair <int, int> > vec;
	map <int, int> mp;
	for (int i = 0; i < n; i++) {
		cin >> a;
		if (a != -1) vec.push_back({a, i + 1});
	}
	a = vec.size();
	if (a == 0) {cout << 1 << endl; return 0;}
	for (int i = 0; i < a - 1; i++) {
		b = vec[i].second; c = vec[i + 1].second;
		if (vec[i].first < vec[i + 1].first) {
			mp[b - 1]++; mp[c + 1]--;
		}
	}
	ll ans = 1LL;
	for (auto i: mp) {
		x = i.first; y = i.second;
		ans = (ans * po (200LL, y)) % mod;
	}
	cout << ans << endl;
	return 0;
}
