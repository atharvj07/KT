import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class GoodString {

    private static class Substring {
        private final String value;
        private int count;

        public Substring(String value) {
            this.value = value;
        }

        public String getValue() {
            return value;
        }

        public int getCount() {
            return count;
        }

        public void incrementCount() {
            count++;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // consume newline

        List<String> strings = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            strings.add(scanner.nextLine());
        }

        String goodString = findGoodString(strings);
        System.out.println(goodString);
    }

    private static String findGoodString(List<String> strings) {
        Map<String, Substring> substrings = new HashMap<>();

        // Count occurrences of all substrings in all strings
        for (String string : strings) {
            for (int i = 0; i < string.length(); i++) {
                for (int j = i + 1; j <= string.length(); j++) {
                    String substring = string.substring(i, j);
                    substrings.putIfAbsent(substring, new Substring(substring));
                    substrings.get(substring).incrementCount();
                }
            }
        }

        // Find the maximum occurrence count
        int maxCount = 0;
        for (Substring substring : substrings.values()) {
            maxCount = Math.max(maxCount, substring.getCount());
        }

        // Find all good substrings with maximum occurrence count
        List<Substring> goodSubstrings = new ArrayList<>();
        for (Substring substring : substrings.values()) {
            if (substring.getCount() == maxCount) {
                goodSubstrings.add(substring);
            }
        }

        // Sort good substrings lexicographically
        goodSubstrings.sort((s1, s2) -> s1.getValue().compareTo(s2.getValue()));

        // Build and return the good string
        StringBuilder goodStringBuilder = new StringBuilder();
        for (Substring substring : goodSubstrings) {
            goodStringBuilder.append(substring.getValue());
        }

        String goodString = goodStringBuilder.toString();
        return goodString.isEmpty() ? "NO" : goodString;
    }
}
