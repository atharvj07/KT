import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long k = sc.nextLong();
            int[] h = new int[n];
            for (int i = 0; i < n; i++) {
                h[i] = sc.nextInt();
            }
            long ans = -1;
            for (int i = 0; i < n; i++) {
                if (k <= 0) {
                    ans = i;
                    break;
                }
                if (i == n - 1) {
                    k--;
                } else if (h[i] >= h[i + 1]) {
                    k--;
                } else {
                    h[i + 1]--;
                }
            }
            System.out.println(ans);
        }
    }
}
