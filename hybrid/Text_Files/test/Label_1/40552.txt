import sys

n=int(input())
a=[]
for i in range(n):
    a.append([int(x) for x in input().split()])
s=0
for i in range(n):
    s+=a[i][i]
for i in range(n):
    if sum(a[i])!=s or sum([a[j][i] for j in range(n)])!=s:
        print(-1)
        sys.exit()
for i in range(n):
    if a[i].count(0)!=1:
        print(-1)
        sys.exit()
for i in range(n):
    for j in range(n):
        if a[i][j]==0:
            a[i][j]=s-sum(a[i])-sum([a[k][j] for k in range(n)])
            print(a[i][j])
            sys.exit()
