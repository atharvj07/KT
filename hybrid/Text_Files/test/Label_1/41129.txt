import sys
input = sys.stdin.readline
n, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
pq = []
for _ in range(q):
    l, r = map(int, input().split())
    pq.append((l-1, r-1))
pq.sort()
p, c = [], []
for l, r in pq:
    x = 0
    for i in range(l, r+1):
        if a[i] < b[i]: x+=1
        else: x-=1
    if x==0: p.append(l)
    elif x==2: c.append(l)
    else: p.append(-1)
for x in p+c: print(x)
