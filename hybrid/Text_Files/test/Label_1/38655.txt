import math

for _ in range(int(input())):
    n, x = map(int, input().split())
    a = [int(i) for i in input().split()]
    b = [x//i for i in a]
    c = []
    for i in range(len(a)):
        if b[i] > 0:
            c.append((a[i], b[i]))
    c.sort()
    ans = math.inf
    for i in range(len(c)):
        if x % c[i][0] == 0:
            ans = min(ans, x // c[i][0])
        else:
            ans = min(ans, c[i][1] + 1)
    print(ans)
