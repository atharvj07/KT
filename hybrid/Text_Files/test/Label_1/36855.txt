n = int(input())
left_trees = []
right_trees = []
for _ in range(n):
    x, a = map(int, input().split())
    if x < 0:
        left_trees.append((x, a))
    else:
        right_trees.append((x, a))

left_trees.sort(reverse=True)
right_trees.sort()

max_apples = 0
for i in range(min(len(left_trees), len(right_trees)) + 1):
    left_apples = sum([a for _, a in left_trees[:i]])
    right_apples = sum([a for _, a in right_trees[:i]])
    if i < len(left_trees):
        max_apples = max(max_apples, left_apples + right_apples)
    if i < len(right_trees):
        max_apples = max(max_apples, left_apples + right_apples + right_trees[i][1])

print(max_apples)
