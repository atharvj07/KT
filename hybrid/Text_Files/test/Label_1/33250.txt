import java.util.*;
import java.io.*;
public class Main {
    static final int mod = 1000000007;
    public static void main(String[] args) {
        Scanner cin=new Scanner(System.in);
        int w=cin.nextInt(), n=cin.nextInt(), m=cin.nextInt();
        long[] a = new long[n];
        for(int i=0; i<n; i++) a[i]=cin.nextInt();
        int[] cnt = new int[m];
        for(int i=0; i<m; i++){
            String gate=cin.next();
            for(int j=0; j<n; j++){
                for(int k=j+1; k<n; k++){
                    long mx=a[j], mi=a[k];
                    if(a[j]>a[k]) {mx=a[k]; mi=a[j];}
                    long x=1L<<w-1, ans=0L;
                    for(int l=0; l<w; l++){
                        long bitj=(mx&x)!=0?1:0;
                        long bitk=(mi&x)!=0?1:0;
                        char c=gate.charAt(l);
                        if(c=='a'){
                            ans=ans*2+((bitj&bitk)==0?1:0);
                        }
                        else if(c=='o'){
                            ans=ans*2+((bitj|bitk)!=0?1:0);
                        }
                        else if(c=='x'){
                            ans=ans*2+((bitj!=bitk)?1:0);
                        }
                        else if(c=='X'){
                            ans=ans*2+((bitj!=bitk)?0:1);
                        }
                        else if(c=='A'){
                            ans=ans*2+((bitj&bitk)!=0?0:1);
                        }
                        else if(c=='O'){
                            ans=ans*2+((bitj|bitk)!=0?0:1);
                        }
                        mx<<=1; mi<<=1;
                    }
                    if(ans==0) cnt[i]++;
                }
            }
            System.out.println(cnt[i]);
        }
    }
}
