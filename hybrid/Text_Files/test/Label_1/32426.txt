import java.util.*;
import java.io.*;

public class Main {
    static int n, m;
    static char[][] graph;
    static int[] path;
    static int[] dp;
    static int[] prev;
    static int[][] parent;
    static int[][] count;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        graph = new char[n][n];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < n; j++) {
                graph[i][j] = s.charAt(j);
            }
        }
        m = sc.nextInt();
        path = new int[m];
        for (int i = 0; i < m; i++) {
            path[i] = sc.nextInt() - 1;
        }
        dp = new int[m];
        prev = new int[m];
        parent = new int[m][50];
        count = new int[m][50];
        int best = -1;
        int bestPos = -1;
        for (int i = 0; i < m; i++) {
            dp[i] = 1;
            parent[i][0] = path[i];
            count[i][0] = 1;
            for (int j = 0; j < i; j++) {
                if (graph[path[j]][path[i]] == '1' && dp[j] + 1 > dp[i]) {
                    dp[i] = dp[j] + 1;
                    prev[i] = j;
                    parent[i][0] = parent[j][0];
                    count[i][0] = count[j][0];
                    for (int k = 0; k < count[j][0]; k++) {
                        parent[i][k + 1] = parent[j][k];
                        count[i][k + 1] = count[j][k];
                    }
                    count[i][0]++;
                    parent[i][count[i][0] - 1] = path[i];
                    count[i][count[i][0] - 1]++;
                }
            }
            if (dp[i] > best) {
                best = dp[i];
                bestPos = i;
            }
        }
        System.out.println(best);
        int[] res = new int[best];
        int pos = bestPos;
        for (int i = best - 1; i >= 0; i--) {
            res[i] = parent[pos][count[pos][0] - 1];
            count[pos][0]--;
            pos = prev[pos];
        }
        for (int i = 0; i < best; i++) {
            System.out.print((res[i] + 1) + " ");
        }
        System.out.println();
    }
}
