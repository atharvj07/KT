import sys

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
# Get common algorithm knowledge for each pair of students
msk = [[0 for j in range(n)] for i in range(n)]
for i in range(n):
    for j in range(n):
        msk[i][j] = (a[i] ^ a[j])
# Check if a pair of students can work together peacefully
can_work = [[True for j in range(n)] for i in range(n)]
for i in range(n):
    for j in range(n):
        if i == j:
            continue
        for k in range(60):
            if (msk[i][j] >> k) & 1:
                can_work[i][j] = False
                can_work[j][i] = False
# Find a subset of students who can work together peacefully with maximum sum of b
mx_sum = 0
for i in range(n):
    for j in range(i+1, n):
        if can_work[i][j]:
            mx_sum = max(mx_sum, b[i] + b[j])
print(mx_sum)
