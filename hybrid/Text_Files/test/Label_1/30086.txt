import java.util.*;

public class Main {
    static int[] arr;
    static int[] ans;
    static int target;
    static boolean[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNext()) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) break;

            target = m;
            arr = new int[n];
            ans = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = sc.nextInt() + sc.nextInt();
            }
            dp = new boolean[n + 5][target + 5];
            Arrays.fill(dp[n], true);

            for (int i = ans.length - 1; i >= 0; i--) {
                for (int j = 0; j <= target; j++) {
                    dp[i][j] = dp[i + 1][j];
                    if (j + arr[i] <= target) dp[i][j] |= dp[i + 1][j + arr[i]];
                }
            }

            int min = n;
            for (int i = 0; i < n; i++) {
                if (dp[i + 1][0] == true) {
                    min = i;
                    break;
                }
            }

            if (min == n) {
                System.out.println("Impossible");
                continue;
            }

            for (int i = 0, j = 0; i < n; i++) {
                if (j + arr[i] <= target && dp[i + 1][j + arr[i]] == true) {
                    ans[min++] = i;
                    j += arr[i];
                }
            }
            System.out.println(min);
        }
    }
}
