import sys
s=sys.stdin.readline().strip()
q=int(sys.stdin.readline())
d={}
ans=[]
for i in range(len(s)):
    if not s[i] in d:
        d[s[i]]=0
    d[s[i]]+=1
for i in range(q):
    t=sys.stdin.readline().split()
    if t[0]=='1':
        d[s[int(t[1])-1]]-=1
        s=s[:int(t[1])-1]+t[2]+s[int(t[1]):]
        if d[s[int(t[1])-1]]==0:
            del d[s[int(t[1])-1]]
        else:
            d[s[int(t[1])-1]]+=1
    if t[0]=='2':
        ans.append(len(d))
for i in ans:
    print(i)
