import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean[] visited = new boolean[n];
        int[] b = new int[n];
        int m = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                visited[i] = true;
                int cnt = 1;
                int p = a[i];
                for (int j = i + 1; j < n; j++) {
                    if (a[j] == p) {
                        visited[j] = true;
                        cnt++;
                    }
                }
                b[m++] = cnt;
            }
        }
        if (m <= 2) {
            System.out.println("cslnb");
        } else {
            System.out.println("sjfnb");
        }
    }
}
