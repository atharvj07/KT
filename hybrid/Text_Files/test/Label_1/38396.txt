import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int,input().split()))
q = int(input())
b = [0] * n
c = [0] * n

for i in range(n):
    c[i] = a[i]

ans = max(c)

for i in range(q):
    l, r, x = map(int, input().split())
    for j in range(l-1, r):
        a[j] += x
    for j in range(l-1, r):
        if j > 0 and a[j] - c[j-1] > ans:
            ans = a[j] - c[j-1]
            b[j] = a[j] - ans
            c[j] = ans
        else:
            b[j] = a[j] - c[j]
    print(ans)
