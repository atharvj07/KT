#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<vector<int>> ranges(m, vector<int>(2));
  for (int i = 0; i < m; i++) cin >> ranges[i][0] >> ranges[i][1];
  vector<vector<int>> in_ranges(n);
  for (int i = 0; i < m; i++) {
    for (int j = ranges[i][0] - 1; j < ranges[i][1]; j++) in_ranges[j].push_back(i);
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int sum = 0;
    for (int j : in_ranges[i]) sum += a[i];
    ans = max(ans, sum);
  }
  cout << ans << endl;
}
