import java.util.*;
import java.io.BufferedReader;
import java.io.InputStreamReader;

class Main {
    static long[] a;
    static HashMap<Long, List<Integer>> sumIndexMap;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        sumIndexMap = new HashMap<Long, List<Integer>>();
        a = new long[n];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(s[i]);
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (!sumIndexMap.containsKey(a[i] + a[j])) {
                    sumIndexMap.put(a[i] + a[j], new ArrayList<Integer>());
                }
                sumIndexMap.get(a[i] + a[j]).add(i);
                sumIndexMap.get(a[i] + a[j]).add(j);
            }
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long sum = a[i] + a[j];
                if (sumIndexMap.containsKey(sum)) {
                    List<Integer> indices = sumIndexMap.get(sum);
                    for (int k = 0; k < indices.size(); k++) {
                        int z = indices.get(k);
                        if (z != i && z != j) {
                            int w = 0;
                            for (int p = 0; p < indices.size(); p++) {
                                if (indices.get(p) != z && indices.get(p) != i && indices.get(p) != j) {
                                    w = indices.get(p);
                                    break;
                                }
                            }
                            System.out.println("YES");
                            System.out.println((i + 1) + " " + (j + 1) + " " + (z + 1) + " " + (w + 1));
                            return;
                        }
                    }
                }
            }
        }

        System.out.println("NO");
    }
}
