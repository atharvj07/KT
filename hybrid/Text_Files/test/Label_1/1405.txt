#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        long long sum = 0;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            sum += a[i];
        }
        if (sum % 2 || *max_element(begin(a), end(a)) * 2 > sum) cout << "NO" << endl; else cout << "YES" << endl;
    }
    return 0;
}
