import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long[] towers = new long[1000000];
        towers[0] = 0;
        long n = scanner.nextLong();
        long m = scanner.nextLong();
        if (n == 0) {
            System.out.println(Math.max(m * 3, 3L));
        } else if (m == 0) {
            System.out.println(Math.max(n * 2, 2L));
        } else {
            long res = 0, p = 0;
            while (res < n + m) {
                p++;
                res += (p + 1) / 2;
            }
            long a = 2 * p - res + n, b = 3 * p - res + m;
            System.out.println(Math.max(a + b * 3, b + a * 2));
        }
    }
}
