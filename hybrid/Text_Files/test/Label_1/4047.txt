#include<bits/stdc++.h>
const double eps=1e-6;
int main(){
    int n;scanf("%d",&n);double p[3][n+1];
    for(int i=0;i<3;++i){
        for(int j=1;j<=n;++j){
            scanf("%lf",&p[i][j]);
        }
    }
    for(int i=1;i<=n;++i){
        for(int j=i+1;j<=n;++j){
            double t=p[0][j];p[0][j]=p[0][i];p[0][i]=t;
            t=p[1][j];p[1][j]=p[1][i];p[1][i]=t;
            t=p[2][j];p[2][j]=p[2][i];p[2][i]=t;
        }
    }
    double ans1[n+1],ans2[n+1];
    for(int i=1;i<=n;++i){
        ans1[i]=0;
        for(int j=i;j<=n;++j){
            ans1[i]+=p[0][j]+p[2][j];
        }
        ans2[i]=0;
        for(int j=1;j<=i;++j){
            ans2[i]+=p[1][j];
        }
    }
    for(int i=1;i<=n;++i){
        printf("%.8lf ",ans1[i]);
    }puts("");
    for(int i=1;i<=n;++i){
        printf("%.8lf ",ans2[i]);
    }puts("");
    return 0;
}
