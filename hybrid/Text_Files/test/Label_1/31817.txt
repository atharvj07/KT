import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger[] arr = new BigInteger[3];
        for (int i = 0; i < arr.length; i++)
            arr[i] = in.nextBigInteger();
        BigInteger sum = new BigInteger("0");
        for (BigInteger x : arr)
            sum = sum.add(x);
        if (sum.compareTo(BigInteger.ONE) == 0)
            System.out.println(0);
        else if (arr[0].compareTo(sum.add(BigInteger.ONE).divide(BigInteger.valueOf(2))) == -1 ||
                arr[1].compareTo(sum.add(BigInteger.ONE).divide(BigInteger.valueOf(2))) == -1 ||
                arr[2].compareTo(sum.add(BigInteger.ONE).divide(BigInteger.valueOf(2))) == -1)
            System.out.println(-1);
        else
            System.out.println(sum.subtract(BigInteger.ONE).divide(BigInteger.valueOf(2)));
    }
}
