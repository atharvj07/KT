#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, x; cin >> n;
  vector<int> v;
  while (n--) {
    cin >> x;
    v.push_back(x);
  }
  vector<int> now = {-1, -1}, nxt(2, -1);
  now[0] = 0;
  for (int i = 0; i < v.size(); i++) {
    for (int j = 0; j < 2; j++) {
      nxt[(j * 10 + v[i]) % 90] = max(nxt[(j * 10 + v[i]) % 90], now[j] + 1);
    }
    now = nxt;
  }
  cout << ((now[0] == -1) ? -1 : pow(10, now[0]));
}
