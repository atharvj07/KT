import java.util.*;
import java.io.*;

public class Main {
    static long n, q, a[];

    static void solve() {
        try (Scanner in = new Scanner(System.in)) {
            n = in.nextLong();
            q = in.nextLong();
            a = new long[(int) n];
            for (int i = 0; i < n; i++)
                a[i] = -1;
            long last = 0;
            for (int i = 0; i < q; i++) {
                long p = in.nextLong();
                if (p == 0) {
                    long j = in.nextLong();
                    long k = in.nextLong();
                    long cnt = 0;
                    for (long j1 = j - 1; j1 < k; j1++)
                        if (a[(int) j1] >= last)
                            cnt++;
                    System.out.println(cnt);
                } else {
                    a[(int) p - 1] = last + 1;
                    a[(int) p - 1] %= 1000000000;
                    a[(int) p - 1] += p;
                    last++;
                }
            }
        } catch (Exception e) {
        }
    }

    public static void main(String[] args) {
        solve();
    }
}
