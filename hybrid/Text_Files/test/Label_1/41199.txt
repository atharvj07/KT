MOD = 998244353

def mod_pow(x:int, p:int, mod:int=MOD) -> int:
    if p == 0: return 1
    if p == 1: return x
    val = mod_pow(x, int(p/2))
    val = (val * val) % mod
    if p%2 != 0:
        val = (val * x) % mod
    return val % mod

def get_inv(x:int, mod:int=MOD) -> int:
    return mod_pow(x, mod-2)

def ncr(n:int, r:int, mod:int=MOD) -> int:
    if n < r: return 0
    p = 1
    k = 1
    for i in range(1,r+1):
        p = (p * (n-i+1)) % mod
        k = (k * i) % mod
    return (p * get_inv(k)) % mod

def add(x:int, y:int) -> int:
    return ((x+y) % MOD)

def sub(x:int, y:int) -> int:
    return ((x-y+MOD) % MOD)

n,k = map(int, input().split())
a = [int(x) for x in input().split()]
ans = 0
for i in range(n):
    pos = (i-1+n)%n
    if a[pos] != a[i]:
        ans = add(ans, ncr(k,k-1))
    else:
        ans = add(ans, sub(ncr(k,k),1))
print(ans)
