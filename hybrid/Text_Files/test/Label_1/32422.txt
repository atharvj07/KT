import java.util.Scanner;

public class Main {
    static int mod = 1000000007;
    static long[][] fact;

    static long pow(long a, long b) {
        if (b == 0) return 1;
        if (b == 1) return a;
        if (b % 2 == 0) {
            long val = pow(a, b / 2);
            return (val * val) % mod;
        }
        return (a * pow(a, b - 1)) % mod;
    }

    static long solve(int n, long[] x) {
        fact = new long[n + 2][];
        for (int i = 0; i <= n + 1; i++) {
            fact[i] = new long[i + 2];
            fact[i][0] = 1;
            fact[i][1] = 1;
            for (int j = 2; j <= i + 1; j++) {
                fact[i][j] = (fact[i][j - 1] * j) % mod;
            }
        }
        long sum = 0;
        for (int i = 2; i <= n; i++) {
            sum = (sum + ((x[i] - x[i - 1]) * ((i - 1) * fact[n - 1][n - i + 1])) % mod) % mod;
        }
        return sum;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] x = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextLong();
        }
        System.out.println(solve(n, x));
    }
}
