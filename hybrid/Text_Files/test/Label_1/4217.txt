#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=1e5+5;
#define rep(i,a,b) for(int i=a;i<=b;++i)
int n,k;
ll c[maxn],p[maxn];
vector<pair<ll,int>> t;
int main(){
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>k;++k;
    rep(i,1,n){
        cin>>p[i];
        t.emplace_back(p[i],i);
    }
    rep(i,1,n) cin>>c[i];
    sort(t.begin(),t.end());
    int l=0,r=-1;
    ll ans=0,tmp=0;
    rep(i,0,n-1){
        while(l<=r&&t[i].first!=t[l].first) tmp-=c[t[l++].second];
        while(r+1<=n-1&&t[r+1].first<=t[i].first) tmp+=c[t[++r].second];
        tmp+=c[t[i].second];
        ans=max(ans,tmp);
        if(r-l+1<=k){
            ans=max(ans,tmp);
        }
    }
    rep(i,1,n) cout<<ans<<" ";
    return 0;
}
