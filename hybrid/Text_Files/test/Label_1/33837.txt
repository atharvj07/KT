import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int n = sc.nextInt();
        int[] ax = new int[k];
        int[] ay = new int[k];
        for (int i = 0; i < k; i++) {
            ax[i] = sc.nextInt();
            ay[i] = sc.nextInt();
        }
        int[] mx = new int[n];
        int[] my = new int[n];
        for (int i = 0; i < n; i++) {
            mx[i] = sc.nextInt();
            my[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                int dx = mx[i] - ax[j];
                int dy = my[i] - ay[j];
                boolean ok = true;
                for (int l = 0; l < k; l++) {
                    if (l == j) {
                        continue;
                    }
                    if (dx * (ay[l] - ay[j]) == dy * (ax[l] - ax[j])) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    ans++;
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
