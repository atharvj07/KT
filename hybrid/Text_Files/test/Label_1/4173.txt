#include <bits/stdc++.h>
using namespace std;
const int N = 55, mod = 1000000007;
int n, x[N], y[N], f[N][N], ord[N], ans, m;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> x[i];
    cin >> m;
    for (int i = 1; i <= n; i++) y[i] = x[i];
    sort(y + 1, y + n + 1);
    for (int i = 1; i <= n; i++) {
        ord[i] = y[i] == x[1] ? n : n - 1;
        if (x[i] == x[1]) {
            f[0][i] = 0;
            continue;
        }
        for (int j = 1; j < i; j++) {
            if (x[j] != x[1]) continue;
            f[j][i] = (1LL * f[j - 1][i] + 1) % mod;
        }
        f[i][i] = f[i - 1][i];
        for (int j = i; j < n; j++) {
            if (x[j] != x[1]) continue;
            int pos = upper_bound(y + i, y + n + 1, m - x[j] - x[1]) - y - 1;
            f[i][i] = (f[i][i] + f[j][pos]) % mod;
        }
        f[i][i] = (f[i][i] + 1) % mod;
    }
    for (int i = 1; i <= n; i++) ans = (ans + f[i][ord[i]]) % mod;
    cout << fixed << setprecision(9) << (long double)ans / n << endl;
}
