import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
import java.lang.Math;

public class Main {
  static final double EPS = 1e-9;
  static int n;
  static Point[] points;

  static class Point {
    double x, y;
    Point(double x, double y) { this.x = x; this.y = y; }
    double dist(Point p) { return Math.hypot(x - p.x, y - p.y); }
    Point unitVector() { double d = dist(new Point(0, 0)); return new Point(x / d, y / d); }
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    n = Integer.parseInt(br.readLine());
    points = new Point[n];

    for (int i = 0; i < n; i++) {
      StringTokenizer st = new StringTokenizer(br.readLine());
      int x = Integer.parseInt(st.nextToken());
      int y = Integer.parseInt(st.nextToken());
      points[i] = new Point(x, y);
    }

    double low = 0, high = 1e9;
    while (high - low > EPS) {
      double mid = (low + high) / 2;
      if (canDeform(mid)) low = mid;
      else high = mid;
    }

    System.out.println(low);
  }

  static boolean canDeform(double d) {
    Point[] deformed = new Point[n];
    for (int i = 0; i < n; i++) {
      Point v = points[(i + 1) % n].unitVector();
      Point p = points[i];
      double a = d * (points[(i + 2) % n].dist(p) / v.dist(p));
      deformed[i] = new Point(p.x + a * v.x, p.y + a * v.y);
    }

    for (int i = 0; i < n; i++) {
      int j = (i + 1) % n;
      int k = (i + 2) % n;
      double x = (deformed[j].x - deformed[i].x) * (deformed[k].y - deformed[j].y);
      double y = (deformed[j].y - deformed[i].y) * (deformed[k].x - deformed[j].x);
      if (x - y < EPS) return false;
    }

    return true;
  }
}
