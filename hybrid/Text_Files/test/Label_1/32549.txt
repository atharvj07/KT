import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) p[i] = i;
        for (int i = 0; i < q; i++) {
            int t = sc.nextInt();
            if (t == 1) {
                int x = sc.nextInt();
                for (int j = 1; j <= n; j++) {
                    p[j] = p[j] + x;
                    if (p[j] > n) p[j] -= n;
                    else if (p[j] < 1) p[j] += n;
                }
            } else {
                for (int j = 1; j <= n; j += 2) {
                    int tmp = p[j];
                    p[j] = p[j + 1];
                    p[j + 1] = tmp;
                }
            }
        }
        for (int i = 1; i <= n; i++) System.out.print(p[i] + " ");
    }
}
