hashtags=[]
for _ in range(int(input())):
    hashtags.append(input())
dp=[hashtags[0]]
for i in range(1, len(hashtags)):
    for j in range(len(dp)):
        if hashtags[i].startswith(dp[j]):
            dp[j]=max(dp[j], hashtags[i])
            break
    else: dp.append(hashtags[i])
print(*dp, sep='\n')
