#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    vector<int> v(n+1), id(n+1);
    for(int i=1; i<=n; i++){
        int x;
        cin>>x;
        v[x] = i;
        id[i] = x;
    }
    while(m--){
        int l, r, x;
        cin>>l>>r>>x;
        if(v[x] >= l && v[x] <= r){
            cout<<"No\n";
            swap(id[ v[x] ], id[ l ]);
            swap(v[ id[ v[x] ] ], v[ id[ l ] ]);
        }else{
            cout<<"Yes\n";
        }
    }
    return 0;
}
