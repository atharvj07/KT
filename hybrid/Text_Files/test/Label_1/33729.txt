import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int sum = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            sum += a[i];
        }
        int mx = 0;
        for (int i = 0; i < n; i++) {
            mx = Math.max(mx, a[i]);
        }
        int ans;
        if (sum % 2 == 1) {
            if (mx % 2 == 0) {
                ans = 0;
            } else {
                ans = 1;
            }
        } else {
            if (mx % 2 == 1) {
                ans = 0;
            } else {
                ans = 1;
            }
        }
        if (ans == 0) {
            System.out.println("BitLGM");
        } else {
            System.out.println("BitAryo");
        }
    }
}
