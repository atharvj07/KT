import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<String> strings = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            strings.add(sc.next());
        }
        String t = sc.next();
        strings.add(t);
        Collections.sort(strings);
        for (int i = 0; i < strings.size(); i++) {
            if (t.equals(strings.get(i))) {
                System.out.print(i + 1 + " ");
            }
        }
        System.out.println();
    }
}
