import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String line;
        while ((line = in.readLine()) != null && !line.equals("0 0")) {
            String[] parts = line.split("\\s+");
            int n = Integer.parseInt(parts[0]);
            int k = Integer.parseInt(parts[1]);
            printDigits(n, k);
        }
    }

    public static void printDigits(int n, int k) {
        StringBuilder sb = new StringBuilder();
        int current = n;
        int count = 0;
        while (count < k) {
            int numDigits = 0;
            int num = current;
            while (num > 0) {
                num /= 10;
                numDigits++;
            }
            if (k - count <= numDigits) {
                int start = numDigits - (k - count);
                int power = (int) Math.pow(10, start);
                int digit = current / power;
                digit %= 10;
                sb.append(digit);
                count++;
                break;
            } else {
                count += numDigits;
                current += numDigits;
            }
        }
        System.out.println(sb.toString());
    }
}
