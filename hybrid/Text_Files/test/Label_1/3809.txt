#include <iostream>
#include <vector>

int main() {
    long long n, m;
    std::cin >> n >> m;
    std::vector<long long> a(n);
    for (long long i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    bool found = false;
    long long sum = 0;
    for (long long i = 0; i < n; i++) {
        sum += a[i];
        sum %= m;
        if (sum == 0) {
            found = true;
            break;
        }
        for (long long j = 0; j < i; j++) {
            sum -= a[j];
            sum %= m;
            if (sum == 0) {
                found = true;
                break;
            }
        }
    }
    std::cout << (found ? "YES" : "NO") << std::endl;
    return 0;
}
