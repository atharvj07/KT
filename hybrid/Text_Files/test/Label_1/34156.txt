import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(reader.readLine());
        while (t-- > 0) {
            int n = Integer.parseInt(reader.readLine());
            int[] a = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            int[] b = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
            System.out.println(valid(a, b) ? "YES" : "NO");
        }
    }

    private static boolean valid(int[] a, int[] b) {
        int n = a.length;
        int[] cnt = new int[3];
        for (int i : a) {
            cnt[i + 1]++;
        }

        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                if (b[i] == 0) {
                    continue;
                }
                if (a[i] == 0) {
                    return false;
                }
                if (a[i] == 1) {
                    if (cnt[0] > 0) {
                        cnt[0]--;
                    } else {
                        return false;
                    }
                }
                if (a[i] == -1) {
                    if (cnt[2] > 0) {
                        cnt[2]--;
                    } else {
                        return false;
                    }
                }
            }
        }
        return true;
    }
}
