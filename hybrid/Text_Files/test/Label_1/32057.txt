import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), a = sc.nextInt(), b = 1;
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
            if (c[i] == a || (b == -1 && c[i] != a)) continue;
            if (c[i] == b) {
                b = 1;
                continue;
            }
            if (b == 1) b = c[i];
        }
        System.out.println(b);
    }
}
