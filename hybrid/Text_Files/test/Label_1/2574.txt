#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;

const int N = 5e5 + 10;

struct Info{
    int l, r, idx;
}a[N];
bool cmp(Info a, Info b) {
    return a.l < b.l;
}
int n, m, ans;
char s[N];
stack<Info>st;

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cin >> n;
    for(int i = 1; i <= n; i++) {
        cin >> s + 1;
        int len = strlen(s + 1);
        int cnt = 0;
        for(int i = 1; i <= len; i++) {
            if(s[i] == '(') cnt++;
            else cnt--;
            if(cnt < 0) break;
        }
        if(cnt != 0) {
            a[++m].l = 1e9;
            continue;
        }
        int l = 0;
        for(int i = 1; i <= len; i++) {
            if(s[i] == '(') l++;
            else l--;
            a[++m] = (Info){l, len - l, i};
        }
    }
    sort(a + 1, a + m + 1, cmp);
    for(int i = 1; i <= m; i++) {
        if(st.empty() || st.top().l != a[i].l || st.top().r != a[i].r) {
            st.push(a[i]);
        } else {
            st.pop();
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
