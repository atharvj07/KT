#include<bits/stdc++.h>
using namespace std;
#define vi vector<int>
const int mx=5500;
vi skl[mx];
int main(){
    int n,w=0;
    cin>>n;
    for(int i=0,x;i<n;i++){
        cin>>x;
        skl[x].push_back(i+1);
    }
    while(skl[1].size()&&skl[2].size()&&skl[3].size()){
        cout<<skl[1].back()<<" "<<skl[2].back()<<" "<<skl[3].back()<<endl;
        skl[1].pop_back();skl[2].pop_back();skl[3].pop_back();
        ++w;
    }
    cout<<w<<endl;
    return 0;
}
