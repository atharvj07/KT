#include <vector>
#include <cstdio>
#include <algorithm>
int main() {
  int n, i, j; scanf("%d", &n);
  std::vector<int> a(n + 2), b(n + 2);
  for (i = 1; i <= n; i++) scanf("%d", &a[i]);
  int round = 0; do {
    b[0] = a[0], b[n + 1] = a[n + 1];
    for (i = 1; i <= n; i++) {
      b[i] = a[i - 1] + a[i] + a[i + 1] >= 2;
    }
    std::swap(a, b); j = 0;
    for (i = 1; i <= n; i++) j += (a[i] != a[i + 1]);
  } while (j); printf("%d\n", round);
  for (i = 1; i <= n; i++) printf("%d ", a[i]); return 0;
}
