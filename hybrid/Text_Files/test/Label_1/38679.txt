n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = list(map(int, input().split()))
dp = [[0 for i in range(4)] for j in range(n + 1)]
dp[1][3] = a[0]
for i in range(2, n + 1):
    dp[i][2] = max(dp[i - 1][0] + a[i - 1], dp[i - 1][2] + c[i - 1])
    dp[i][3] = max(dp[i - 1][0] + b[i - 1], dp[i - 1][1] + a[i - 1], dp[i - 1][2] + b[i - 1], dp[i - 1][3] + c[i - 1])
    dp[i][1] = dp[i][3]
print(dp[n][3])
