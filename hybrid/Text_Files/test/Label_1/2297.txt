#include <iostream>
using namespace std;
int main() {
  int n, winM = 0, winC = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int m, c;
    cin >> m >> c;
    if (m > c) winM++;
    else if (c > m) winC++;
  }
  if (winM > winC)
    cout << "Mishka" << endl;
  else if (winC > winM)
    cout << "Chris" << endl;
  else
    cout << "Friendship is magic!^^" << endl;
}
