#include<bits/stdc++.h>
using namespace std;

typedef long long int ll;
int main()
{
    ll n,m;
    int p;
    cin>>n>>m;
    ll h[16],mi[16];
    memset(h,0,sizeof(h));
    memset(mi,0,sizeof(mi));
    ll c=0;
    ll temp,x,y;
    p=0;
    while(n)
    {
        temp=n%7;
        h[p++]=temp;
        n/=7;
    }
    p=0;
    while(m)
    {
        temp=m%7;
        mi[p++]=temp;
        m/=7;
    }
    ll g=0;
    x=h[0];
    y=mi[0];
    for(ll i=1;i<p;i++)
    {
        if(h[i]!=x)
        {
            g=g+(7-i);
            x=h[i];
        }
        if(mi[i]!=y)
        {
            g=g+(7-i);
            y=mi[i];
        }
    }
    cout<<g<<endl;
}
