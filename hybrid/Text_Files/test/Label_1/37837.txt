import sys

def dist_sq(p1, p2):
    return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2

n = int(input())
ant = [list(map(int, input().split())) for _ in range(n)]
m = int(input())
q = [list(map(int, input().split())) for _ in range(m)]

for distances in q:
    dist = sorted({d**0.5 for d in distances})

    cnt = 0
    for x1, y1 in ant:
        ok = True
        for i in range(n):
            if dist[i]**2 != dist_sq((x1, y1), ant[i]):
                ok = False
        if ok:
            cnt += 1
            print(x1, y1, end=' ')
    print(cnt)
