import numpy as np

n = int(input())
a = list(map(int, input().split()))
a = np.array(a)
min = np.sum(a)
for i in a:
    for j in a:
        if i % j == 0 and j != i:
            a1 = np.copy(a)
            a1[np.where(a1 == i)] = a1[np.where(a1 == i)] / (i/j)
            a1[np.where(a1 == j)] = a1[np.where(a1 == j)] * (i/j)
            s = np.sum(a1)
            if s < min:
                min = s
print(min)
