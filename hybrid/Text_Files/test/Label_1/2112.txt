#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
ll p[100005], q[100005];
int main() {
    ll n, m; cin >> n >> m;
    vector<ll> a(n+1), b(n+1);
    for(ll i = 0; i < n; i++) cin >> a[i];
    for(ll i = 0; i < n; i++) cin >> b[i];
    p[n] = q[n] = 1;
    for(ll i = n-1; i >= 0; i--) {
        if(!a[i]) p[i] = p[i+1] * (m-1) % mod;
        else p[i] = p[i+1];
        if(!b[i]) q[i] = q[i+1] * (m-1) % mod;
        if(a[i]) q[i] = q[i+1] * m % mod;
    }
    ll inv = 0, x = 1000000009, y = 1;
    while(y) {
        ll r = x%y;
        x = y;
        y = r;
        inv++;
    }
    inv--;
    ll ans = (p[0] * q[0] % mod * inv % mod) % mod;
    cout << ans << endl;
    return 0;
}
