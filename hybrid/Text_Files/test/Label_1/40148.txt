n, k = map(int, input().split())
a = list(map(int, input().split()))

candies = 0
day = 0

for i in range(n):
    candies += a[i]
    if candies >= k:
        print(i+1)
        break
    elif candies > 8:
        k -= 8
        candies -= 8
    else:
        k -= candies
        candies = 0
    if i == n-1:
        if candies >= k:
            print(n)
        else:
            print(-1)
