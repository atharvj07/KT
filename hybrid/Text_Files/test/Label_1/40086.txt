n, k = map(int, input().split())
questions = list(map(int, input().split()))

# Track cells asked by Bob
asked = set()
# Track adjacent cells asked by Bob
adjacent_asked = set()

for x in questions:
    asked.add(x)
    if x > 1:
        adjacent_asked.add(x-1)
    if x < n:
        adjacent_asked.add(x+1)

# Total possible scenarios without restrictions
total_scenarios = n*3 - 2

# Subtract scenarios where Alice cannot stay or move to an adjacent cell
for x in asked:
    total_scenarios -= 1
    if x in adjacent_asked:
        total_scenarios -= 1

# Correct scenarios for cells at the ends of the line
if 1 in asked:
    total_scenarios -= 1
if n in asked:
    total_scenarios -= 1

print(total_scenarios)
