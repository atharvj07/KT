import java.util.*;

public class Main {
    static int[] color;
    static int[] dist;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        color = new int[n + 1];
        dist = new int[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        dfs(1, -1, 0);
        while (q-- > 0) {
            int type = sc.nextInt();
            int node = sc.nextInt();
            if (type == 1) {
                if (color[node] == 0) {
                    dfs(node, -1, 0);
                }
            } else {
                System.out.println(dist[node]);
            }
        }
    }

    static void dfs(int node, int parent, int d) {
        color[node] = 1;
        dist[node] = d;
        for (int u : adj[node]) {
            if (u != parent) {
                dfs(u, node, d + 1);
            }
        }
    }
}
