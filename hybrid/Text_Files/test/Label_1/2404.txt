#include<bits/stdc++.h>
using namespace std;
double piness(const string &s) {
  int n = s.size(), ans = 0;
  for (int i = 0; i < n; ++i) {
    int cnt = 0;
    for (int j = i; j < n; ++j) {
      cnt += (s[j] == 'A' || s[j] == 'E' || s[j] == 'I' || s[j] == 'O' || s[j] == 'U' || s[j] == 'Y');
      ans += cnt;
    }
  }
  return ans * 1.0 / n / n;
}
int main() {
  string s;
  cin >> s;
  cout << fixed << setprecision(8) << piness(s) << endl;
}
