#include<bits/stdc++.h>
using namespace std;
int n;
int main(){
    cin>>n;
    vector<int> v(n+1);
    for(int i=1;i<=n;++i) cin>>v[i];
    int sum=0;
    for(int i=1;i<=n;++i){
        if(sum==0 && v[i]!=0){
            cout<<"YES\n";
            int j=i;
            while(j<=n && v[j]*sum<=0) j++;
            if(j>n) j=n;
            cout<<1<<'\n';
            cout<<i<<' '<<j<<'\n';
            i=j;
        }
        sum+=v[i];
    }
    if(sum!=0) cout<<"NO\n";
    return 0;
}
