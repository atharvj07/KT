#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int fre = 0, id;
    for (int i = 0; i < n; i++) {
      if (a[i] != a[0]) fre++, id = i;
    }
    if (fre == 1)
      cout << id + 1 << endl;
    else
      cout << 1 << endl;
  }
  return 0;
}
