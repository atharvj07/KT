import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] start = scanner.nextLine().split(":");
        String[] end = scanner.nextLine().split(":");
        int h = (Integer.parseInt(start[0]) + Integer.parseInt(end[0])) / 2;
        int m = (Integer.parseInt(start[1]) + Integer.parseInt(end[1])) / 2;
        if (m % 2 != 0) {
            m++;
            h++;
        }
        System.out.printf("%02d:%02d", h, m);
    }
}
