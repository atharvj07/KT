for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    if sum(a) % 2 == 0 or k % 2 == 0 and sum(a) % 2 == 1 or k % 2 == 1 and sum(a) % 2 == 0 or k > n:
        print('NO')
    else:
        print('YES')
        ans = [1]
        for i in range(2, n + 1):
            if (sum(a[i - 1:]) + sum(a[:i - 1])) % 2 == 1:
                ans += [i]
        ans += [n]
        print(*ans)
