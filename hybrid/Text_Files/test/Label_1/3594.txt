#include<bits/stdc++.h>
using namespace std;
const int mod=998244353;
int main() { ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
    string a,l,r; cin>>a>>l>>r;
    int x=a.size(); int y=l.size(); int z=r.size(); reverse(a.begin(),a.end());
    while(a.size()<max(y,z)) a.push_back('0'); reverse(a.begin(),a.end());
    reverse(l.begin(),l.end());
    while(l.size()<max(x,z)) l.push_back('0'); reverse(l.begin(),l.end());
    reverse(r.begin(),r.end());
    while(r.size()<max(x,y)) r.push_back('0'); reverse(r.begin(),r.end());
    int dp[x+5][y+5][z+5][2][2]; memset(dp,0,sizeof(dp));
    dp[x][y][z][0][0]=1;
    for(int i=x-1;i>=0;i--){
        for(int j=y-1;j>=0;j--){
            for(int k=z-1;k>=0;k--){
                for(int x=0;x<2;x++){
                    for(int y=0;y<2;y++){
                        dp[i][j][k][x][y]=(1LL*dp[i][j][k][x][y]+1LL*dp[i+1][j][k][x|(a[i]==l[j])][y])%mod;
                        if(a[i]>l[j]) continue;
                        dp[i][j][k][x][y]=(1LL*dp[i][j][k][x][y]+1LL*dp[i+1][j+1][k][x|(a[i]==l[j])][y|(a[i]==r[k])])%mod;
                    }
                }
            }
        }
    }
    cout<<dp[0][0][0][0][0]<<endl;
}
