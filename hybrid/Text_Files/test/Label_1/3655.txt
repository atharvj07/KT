#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  bool possible = false;
  int start = -1, end = -1;
  for (int i = 1; i < n; i++) {
    if (a[i] < a[i - 1]) {
      if (!possible) {
        possible = true;
        start = i - 1;
      }
      end = i;
    }
  }

  if (possible) {
    cout << "yes" << endl;
    cout << start + 1 << " " << end + 1 << endl;
  } else {
    cout << "no" << endl;
  }

  return 0;
}
