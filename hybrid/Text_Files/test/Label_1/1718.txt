#include <bits/stdc++.h>
using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        long long a, b, m;
        cin >> a >> b >> m;
        if (b < a || (b - a) % m != 0)
            cout << -1 << endl;
        else {
            int k = (b - a) / m + 1;
            cout << k << " ";
            for (int i = 1; i <= k; i++) { cout << a + (i - 1) * m << " "; }
            cout << endl;
        }
    }
    return 0;
}
