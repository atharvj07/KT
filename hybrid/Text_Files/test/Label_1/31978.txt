import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        boolean speedLimit = false, overtakeAllowed = true;
        int speed = sc.nextInt(), ans = 0;
        while (--n > 0) {
            int t = sc.nextInt();
            if (t == 1) {
                speed = sc.nextInt();
            } else if (t == 2) {
                if (!overtakeAllowed) {
                    ans++;
                }
            } else if (t == 3) {
                speed = sc.nextInt();
                speedLimit = true;
            } else if (t == 4) {
                overtakeAllowed = true;
            } else if (t == 5) {
                speedLimit = false;
            } else if (t == 6) {
                overtakeAllowed = false;
            }
            if (speedLimit && speed > sc.nextInt()) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
