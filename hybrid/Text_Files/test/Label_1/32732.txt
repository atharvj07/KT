
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int d = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        for (int i = 0; i < n; ) {
            int j = i + 1;
            while (j < n && a[j] - a[i] <= d) {
                j++;
            }
            int sz = j - i;
            if (sz >= k) {
                ans++;
                i = j;
            } else {
                break;
            }
        }
        if (ans > 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
