import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
m = int(input())

dp = [[0] * (n + 1) for _ in range(n + 1)]
for i in range(n - 1, -1, -1):
    for j in range(n):
        if a[i] > dp[i + 1][j]:
            dp[i][j] = a[i]
        dp[i][j] = max(dp[i + 1][j], dp[i][j], dp[i + 1][j + 1])

for i in range(m):
    k, p = map(int, input().split())
    print(dp[n - k + p - 1][n - k])
