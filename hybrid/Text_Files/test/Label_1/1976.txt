#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5, M = 4e5 + 5;
int n, q, a[N], w[N], l[M], r[M], u[M], lst[N];
bool vis[N];

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1, j; i <= q; i++) {
        cin >> j >> l[i] >> r[i];
        if (j < u[l[i]]) continue;
        u[l[i]] = j;
        w[l[i]] = j;
        u[r[i]] = j;
        w[r[i]] = j;
    }
    for (int i = 1; i <= n; i++)
        if (a[i] == 0)
            vis[i] = 1;
    set<int> Q;
    for (int i = 1; i <= n; i++) {
        if (!vis[i])
            Q.erase(w[i]);
        else
            Q.insert(w[i]);
        if (a[i] == 0) {
            if (Q.empty())
                return cout << "NO", 0;
            a[i] = *Q.begin();
        } else if (w[i] != a[i] || lst[a[i]])
            return cout << "NO", 0;
        lst[a[i]] = i;
    }
    cout << "YES\n";
    for (int i = 1; i <= n; i++) cout << a[i] << ' ';
    cout << '\n';
}
