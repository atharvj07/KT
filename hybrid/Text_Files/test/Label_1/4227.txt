#include <iostream>
using namespace std;

int b[200001];
int main(int argc, char const *argv[])
{
    int n, k;
    cin >> n >> k;
    int a[n];
    for (int i = 0; i < n; ++i)
    {
        cin >> a[i];
        b[a[i]] = i;
    }
    bool team = 1;
    while (n)
    {
        int mx = 0, pos;
        for (int i = 1; i <= n; ++i)
        {
            if (!b[i])
                continue;
            if (b[i] > mx)
            {
                mx = b[i];
                pos = i;
            }
        }
        int l = max(1, mx - k), r = min(n, mx + k);
        for (int i = l; i <= r; ++i)
            b[a[i - 1]] = 0;
        n -= r - l + 1;
        for (int i = l; i <= r; ++i)
            cout << (team ? '1' : '2');
        team ^= 1;
    }
    return 0;
}
