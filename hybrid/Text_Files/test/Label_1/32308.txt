import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            long n = in.nextLong();
            int k = in.nextInt();
            if (n % 2 != k % 2 || n < k) {
                System.out.println("NO");
            } else {
                System.out.println("YES");
                long ans = n / k;
                if (k % 2 == 0) {
                    if (ans % 2 == 1) ans++;
                }
                for (int i = 0; i < k; i++)
                    System.out.print(ans + " ");
                System.out.println();
            }
        }
    }
}
