#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, b, a;
    cin >> n >> b >> a;
    vector<int> s(n);
    for (int i = 0; i < n; i++) { cin >> s[i]; }
    vector<int> dp(n + 1, -1e9);
    dp[0] = 0;
    for (int i = 0; i < n; i++) {
        if (dp[i] >= 0) {
            dp[i + 1] = max(dp[i + 1], dp[i]);
            if (b) { dp[i + 1] = max(dp[i + 1], dp[i] + s[i] + 1); }
        }
        if (dp[i] >= 0 && a) { dp[i + 1] = max(dp[i + 1], dp[i] + 1); }
    }
    cout << dp[n] << endl;
    return 0;
}
