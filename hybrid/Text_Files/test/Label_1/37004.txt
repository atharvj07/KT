hh, mm = map(int, input().split())
H, D, C, N = map(int, input().split())

if hh >= 20:
    cost = C * 0.8
else:
    time_to_discount = (19 - hh) * 60 + (60 - mm)
    H += time_to_discount * D
    cost = C * 0.8

buns_needed = (H + N - 1) // N
total_cost = buns_needed * cost

print(f"{total_cost:.4f}")
