#include <iostream>
#include <string>

int main() {
  std::string s;
  std::cin >> s;
  int n = s.length();
  int ans = 0, cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'A' || s[i] == 'E' || s[i] == 'I' || s[i] == 'O' || s[i] == 'U' || s[i] == 'Y') {
      cnt++;
    } else {
      ans = std::max(ans, cnt);
      cnt = 0;
    }
  }
  ans = std::max(ans, cnt);
  std::cout << ans << std::endl;
  return 0;
}
