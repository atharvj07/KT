import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class HanhsTree {

    private static final int MOD = 998244353;

    private static class Node {
        int id;
        List<Node> children;

        Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int q = scanner.nextInt();

        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }

        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            nodes[u].children.add(nodes[v]);
            nodes[v].children.add(nodes[u]);
        }

        long[] subtreeSize = new long[n + 1];
        long[] sumValue = new long[n + 1];
        dfs(nodes[1], null, subtreeSize, sumValue);

        for (int i = 0; i < q; i++) {
            int type = scanner.nextInt();
            int v = scanner.nextInt();

            if (type == 1) {
                int d = scanner.nextInt();
                update(nodes[1], null, v, d, subtreeSize, sumValue);
            } else {
                long expectedValue = (sumValue[v] * inv(subtreeSize[v])) % MOD;
                System.out.println(expectedValue);
            }
        }
    }

    private static void dfs(Node node, Node parent, long[] subtreeSize, long[] sumValue) {
        subtreeSize[node.id] = 1;
        sumValue[node.id] = 0;

        for (Node child : node.children) {
            if (child != parent) {
                dfs(child, node, subtreeSize, sumValue);
                subtreeSize[node.id] += subtreeSize[child.id];
                sumValue[node.id] += (sumValue[child.id] + subtreeSize[child.id]) % MOD;
            }
        }
    }

    private static void update(Node node, Node parent, int v, int d, long[] subtreeSize, long[] sumValue) {
        if (node.id == v) {
            sumValue[node.id] += d;
        }

        for (Node child : node.children) {
            if (child != parent) {
                update(child, node, v, d, subtreeSize, sumValue);
            }
        }
    }

    private static long inv(long x) {
        return pow(x, MOD - 2);
    }

    private static long pow(long x, long n) {
        long result = 1;
        while (n > 0) {
            if ((n & 1) == 1) {
                result = (result * x) % MOD;
            }
            x = (x * x) % MOD;
            n >>= 1;
        }
        return result;
    }
}
