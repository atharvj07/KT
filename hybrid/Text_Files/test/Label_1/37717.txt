import sys
from collections import deque

n, k = map(int, input().split())
arr = []
for _ in range(n):
    arr.append(tuple(map(int, sys.stdin.readline().split())))
arr.sort(key=lambda x: x[0])

ans = mxs = 0
dq = deque()
for i in range(n):
    l, r, a = arr[i]
    while dq and dq[0][1] < l:
        mxs -= dq.popleft()[0]
    mxs = max(mxs, a)
    ans += mxs
    if mxs <= k and r - l <= k - mxs:
        mxs += a
        continue
    dq.append((k - mxs, r))
    mxs = k
if mxs <= k:
    ans += mxs
else:
    ans = -1
print(ans)
