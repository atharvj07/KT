n, x = [int(i) for i in input().split()]
candies = []
for i in range(n):
    t, h, m = [int(i) for i in input().split()]
    if h <= x:
        candies.append((h, m))
candies.sort(key=lambda c: c[0])
ans = 0
h = x
while candies:
    a = -1
    b = -1
    for i in range(len(candies)):
        x, y = candies[i]
        if x > h:
            break
        if t == 0 and a == -1:
            a = i
        elif t == 1 and b == -1:
            b = i
    if a != -1:
        h += candies[a][1]
        candies.pop(a)
        ans += 1
    elif b != -1:
        h += candies[b][1]
        candies.pop(b)
        ans += 1
    else:
        break
print(ans)
