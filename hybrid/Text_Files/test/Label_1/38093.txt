def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def is_coprime_array(arr):
    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):
            if gcd(arr[i], arr[j]) != 1:
                return False
    return True

n = int(input())
a = list(map(int, input().split()))

b = a.copy()

for i in range(n):
    while not is_coprime_array(b[:i+1]):
        b[i] += 1

print(*b)
