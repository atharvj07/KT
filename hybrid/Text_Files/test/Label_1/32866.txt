import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        Map<String, String> map = new HashMap<>();
        map.put("rock", "p");
        map.put("paper", "s");
        map.put("scissors", "r");
        String[] gestures = new String[3];
        gestures[0] = in.nextLine();
        gestures[1] = in.nextLine();
        gestures[2] = in.nextLine();
        int count = 0;
        for(int i = 0; i < 3; i++) {
            if(gestures[i].equals(gestures[(i+1)%3])) continue;
            count += map.get(gestures[i]).equals(gestures[(i+1)%3]) ? 1 : -1;
        }
        System.out.println(count == 2 ? "?" : count == 1 ? gestures[0].substring(0, 1).toUpperCase() : gestures[1].substring(0, 1).toUpperCase());
    }
}
