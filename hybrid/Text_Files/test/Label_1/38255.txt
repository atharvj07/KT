t=int(input())
for i in range(t):
	n,d=map(int,input().split())
	a=list(map(int,input().split()))
	p=0
	for j in range(1,n):
		if a[p]<a[j]:
			p=j
	for j in range(d):
		if p!=0 and a[p-1]<a[p]:
			a[p-1]+=1
			a[p]-=1
			p-=1
		elif p!=n-1 and a[p+1]<a[p]:
			a[p+1]+=1
			a[p]-=1
			p+=1
	print(a[0])
