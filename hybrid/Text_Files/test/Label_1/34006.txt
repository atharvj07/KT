import java.util.*;

public class Main {
    static final long INF = Long.MAX_VALUE / 2;
    static int V, E;
    static long[][] d;
    static List<Edge>[] edges;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        V = sc.nextInt();
        E = sc.nextInt();
        edges = new List[V];
        for (int i = 0; i < V; i++) {
            edges[i] = new ArrayList<>();
        }
        for (int i = 0; i < E; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            long w = sc.nextLong();
            edges[u].add(new Edge(v, w));
        }
        d = new long[V][V];
        for (int i = 0; i < V; i++) {
            Arrays.fill(d[i], INF);
        }

        // initialize
        d[0][0] = 0;

        // run Bellman Ford
        for (int i = 1; i < V; i++) {
            for (int j = 0; j < V; j++) {
                for (Edge edge : edges[j]) {
                    if (d[j][i] + edge.w < d[edge.v][i]) {
                        d[edge.v][i] = d[j][i] + edge.w;
                    }
                }
            }
        }

        // check negative cycle
        boolean hasNegativeCycle = false;
        for (int j = 0; j < V; j++) {
            for (Edge edge : edges[j]) {
                if (d[j][V - 1] + edge.w < d[edge.v][V - 1]) {
                    hasNegativeCycle = true;
                }
            }
        }

        if (hasNegativeCycle) {
            System.out.println("NEGATIVE CYCLE");
            return;
        }

        // output
        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                if (d[i][j] == INF) {
                    System.out.print("INF ");
                } else {
                    System.out.print(d[i][j] + " ");
                }
            }
            System.out.println();
        }
    }

    static class Edge {
        int v;
        long w;

        public Edge(int v, long w) {
            this.v = v;
            this.w = w;
        }
    }
}
