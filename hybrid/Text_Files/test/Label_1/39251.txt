MOD = 10**9 + 7

def solve(m, updates):
    forbidden = {"0011", "0101", "1110", "1111"}
    dp = [0] * (m+1)  # dp[i] stores the number of sequences ending at i
    s = ""
    for i in range(m):
        s += updates[i]
        dp[i+1] = dp[i] + 1  # current character can always form a letter
        for l in range(2, 5):  # check for substrings of length 2 to 4
            if i+1 >= l and s[i+1-l:i+1] not in forbidden:
                dp[i+1] += dp[i+1-l]
            dp[i+1] %= MOD
        print(dp[i+1])

m = int(input().strip())
updates = [input().strip() for _ in range(m)]
solve(m, updates)
