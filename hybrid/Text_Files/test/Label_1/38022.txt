from collections import Counter
from math import gcd

def max_k_beautiful_necklace(n, k, s):
    counts = Counter(s)
    ans = 1
    for g in range(1, n+1):
        if k % g == 0:
            cycle = k // g
            temp = sum(min(v, cycle) for v in counts.values())
            ans = max(ans, g * (temp // g))
    return ans

t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    s = input().strip()
    print(max_k_beautiful_necklace(n, k, s))
