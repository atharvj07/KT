import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        long a = scanner.nextLong();
        long c = scanner.nextLong();

        long b = 0;
        int numBits = 0;

        long aOrig = a;
        long cOrig = c;

        while (a > 0 || c > 0) {
            long bBit = (a % 10 + c % 10) % 3;
            a /= 10;
            c /= 10;
            if (bBit > 0) {
                b = b * 10 + bBit;
                numBits++;
            }
        }

        for (int i = 0; i < numBits; i++) {
            long aBit = aOrig % 10;
            long cBit = cOrig % 10;
            if (aBit != cBit) {
                b = b * 10 + 1;
            }
            aOrig /= 10;
            cOrig /= 10;
        }

        System.out.println(b);
    }
}
