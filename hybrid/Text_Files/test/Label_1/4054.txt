#include <iostream>
#include <vector>
using namespace std;
#define ll long long

ll p, k;

vector<ll> solve(ll n) {
  vector<ll> v(n);
  v[n - 1] = 1;
  for (ll i = n - 2; i >= 0; i--) {
    v[i] = p / k;
    p -= v[i] * k;
  }
  return v;
}

int main() {
  cin >> p >> k;
  if (p % k == 0) {
    cout << -1 << endl;
  } else {
    vector<ll> v = solve(k - 1);
    cout << v.size() << endl;
    for (auto x : v) cout << x << " ";
    cout << endl;
  }
  return 0;
}
