#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long
void solve() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n + 4);
  map<int, int> mp;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    mp[a[i]]++;
  }
  int mn = (n + k - 1) / k;
  int ans = mn;
  for (auto i : mp) {
    int occ = i.second;
    occ /= k;
    if (occ > 0) ans = min(ans, mn - occ);
  }
  cout << ans << endl;
}
int32_t main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
