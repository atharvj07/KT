#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

ll n, ans = 0, base;

int main() {
  cin >> n;
  base = n * (n + 1) / 2;
  ll rem = base % 10;
  if (rem == 0)
    ans = n * (n - 1) / 2;
  else if (rem == 9)
    ans = n * (n - 1) / 2 + (n % 2) * (n - 1);
  cout << ans << endl;
  return 0;
}
