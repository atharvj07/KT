#include<bits/stdc++.h>
using namespace std;
const int M = 1e5+5;
int p[M],ans[M];
int main(){
  int n,i,j,x,f=0;
  cin >> n;
  for(i=0;i<=n;++i){
    ans[i] = i;
  }
  ans[0] = 1;
  for(i=0;i<n;++i){
    if(ans[i]==n){
      swap(ans[i],ans[n]);
      for(j=i+1;j<n;++j){
        ans[j] = n-j+i+1;
      }
      f=1;
      break;
    }
  }
  if(f){
    cout << "YES" << endl;
    for(i=0;i<n;++i){
      cout << ans[i] << endl;
    }
  }
  else{
    cout << "NO" << endl;
  }
  return 0;
}
