import sys

for _ in range(int(sys.stdin.readline())):
    n, k = map(int, sys.stdin.readline().split())
    STR = str(n)
    DIGITS = set(STR)
    if len(DIGITS) <= k:
        print(n)
        continue
    MAX = '9' * (len(STR) + 1)
    while len(set(MAX)) > k or (len(set(MAX)) == k and MAX <= STR):
        MAX = str(int(MAX) + 1)
    print(MAX)
