p, x = map(int, input().split())
for last_digit in range(1, 10):
    carry = 0
    number = [0] * p
    number[-1] = last_digit
    for i in range(p-1, 0, -1):
        digit = (number[i] * x + carry) % 10
        carry = (number[i] * x + carry) // 10
        number[i-1] = digit
    if carry == last_digit * x // 10 and (last_digit * x + carry) % 10 == number[0]:
        print(''.join(map(str, number)))
        break
else:
    print("Impossible")
