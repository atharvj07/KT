import java.util.*;
import java.io.*;

public class Main {
    static List<Integer> divisors = new ArrayList<>();
    
    static void generateDivisors(int limit) {
        divisors.add(1);
        for (int i = 2; i <= limit; i++) {
            if (isPrime(i)) {
                divisors.add(i);
            } else {
                for (int j = 0; j < divisors.size(); j++) {
                    if (i % divisors.get(j) == 0) {
                        divisors.add(i);
                        break;
                    }
                }
            }
        }
    }
    
    static boolean isPrime(int n) {
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
    
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] primes = new int[n];
        String[] line = in.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            primes[i] = Integer.parseInt(line[i]);
        }
        int k = Integer.parseInt(in.readLine());
        generateDivisors(primes[n - 1] * primes[n - 1]);
        int ans = 1;
        for (int i = 1; i < k; i++) {
            ans++;
            while (!isValid(ans, primes)) {
                ans++;
            }
        }
        System.out.println(ans);
    }
    
    static boolean isValid(int n, int[] primes) {
        for (int i = 0; i < primes.length; i++) {
            if (n % primes[i] != 0) {
                return false;
            }
        }
        return true;
    }
}
