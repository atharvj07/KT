import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong(), x = in.nextLong(), min = 0;
        long[] a = new long[(int) n];
        for (long i = 0; i < n; i++) {
            a[(int) i] = in.nextLong();
        }
        Arrays.sort(a);
        for (long i = 0; i < n; i++) {
            if (x > 1) {
                min += a[(int) i] * x;
            } else {
                min += a[(int) i];
            }
            x--;
        }
        System.out.println(min);
    }
}
