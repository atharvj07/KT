#include <iostream>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    int mat[n][m];
    bool valid[n][m];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        cin >> mat[i][j];
        valid[i][j] = (mat[i][j] == 0);
      }
    }
    bool win = false;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (!valid[i][j]) continue;
        int r = 0, c = 0;
        for (int k = 0; k < n; k++) { r += valid[k][j]; }
        for (int k = 0; k < m; k++) { c += valid[i][k]; }
        win |= (r == 1 || c == 1);
      }
    }
    cout << (win ? "Ashish" : "Vivek") << endl;
  }
  return 0;
}
