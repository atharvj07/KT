n, k = map(int, input().split())
a = sorted([int(x) for x in input().split()])
if k >= n - 1:
    print(0, n - 1)
    for i in range(n - 1):
        print(i + 1, i + 2)
else:
    print(a[n - 1] - a[0], k)
    for i in range(k):
        a[0] += 1
        a[n - 1] -= 1
    for i in range(n - 1):
        print(i + 1, i + 2)
