#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

#define sz(x) (ll)x.size()
#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()

const ll inf = 1e18;
const ll mod = 1e9 + 7;
const ll N = 2e5 + 5;

ll n, m, a[N], F[N], G[N];

void push(ll l, ll r) {
  if (~F[l]) {
    if (F[l] && F[l] == G[l]) G[l] = 0LL;
    else if (F[l] && G[l] == 0LL) G[l] = F[l];
    else if (!F[l] && G[l]) F[l] = G[l];
    else if (!F[l] && !G[l]) F[l] = 0LL;
  }
  while (l <= r) {
    if (F[l] == 0LL) {
      G[l] = G[l + 1];
      F[l] = F[l + 1];
    } else if (G[l] == 0LL) {
      G[l] = G[l + 1];
      F[l] = F[l + 1];
    }
    l++;
  }
}

void modify(ll p, ll val, ll l, ll r, ll cur) {
  if (l == r) {
    F[cur] = val;
    G[cur] = val;
    return;
  }
  ll mid = (l + r) / 2;
  push(cur, cur);
  if (p <= mid) modify(p, val, l, mid, 2 * cur);
  else modify(p, val, mid + 1, r, 2 * cur + 1);
  F[cur] = F[2 * cur] | F[2 * cur + 1];
  G[cur] = G[2 * cur] ^ G[2 * cur + 1];
}

ll get(ll cur, ll l, ll r) {
  if (l == r) return F[cur];
  push(cur, cur);
  ll mid = (l + r) / 2;
  return get(2 * cur, l, mid) | get(2 * cur + 1, mid + 1, r);
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cout.tie(NULL);
  memset(F, -1, sizeof F);
  cin >> n >> m;
  for (ll i = 1; i <= (1 << n); i++) { cin >> a[i]; }
  for (ll i = 1; i <= (1 << n); i++) modify(i, a[i], 1, (1 << n), 1);

  while (m--) {
    ll p, val;
    cin >> p >> val;
    modify(p, val, 1, (1 << n), 1);
    cout << get(1, 1, (1 << n)) << '\n';
  }
  return 0;
}
