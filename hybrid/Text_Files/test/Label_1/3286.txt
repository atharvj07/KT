#include <bits/stdc++.h>
#define N 100005

using namespace std;

int a[N], L[N], R[N], ans;
vector<int> G[N];
bool vis[N];

void dfs(int u) {
    vis[u] = 1;
    for (int v : G[u]) {
        if (vis[v]) continue;
        dfs(v);
        L[u] = min(L[u], L[v]);
        R[u] = max(R[u], R[v]);
    }
}

int solve(int n) {
    memset(vis, 0, sizeof(vis));
    memset(L, 0x3f, sizeof(L));
    for (int i = 1; i <= n; i++) {
        if (a[i] > i) G[i].push_back(a[i] - i);
        else G[i].push_back(a[i] + n - i);
    }
    for (int i = 1; i <= n; i++) {
        if (vis[i]) continue;
        L[i] = R[i] = i;
        dfs(i);
        if (L[i] == 1 && R[i] == n) ans++;
    }
    return ans;
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    cout << solve(n) << endl;
}
