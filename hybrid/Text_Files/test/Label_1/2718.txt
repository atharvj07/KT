#include <algorithm>
#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> t(n);
    for (int i = 0; i < n; i++) { std::cin >> t[i]; }
    std::vector<int> c(n);
    c[0] = 20;
    for (int i = 1; i < n; i++) {
        int s = t[i] - t[i - 1] - 1;
        c[i] = c[i - 1] + std::min(20, std::min(50, 120 - s * 20));
    }
    std::cout << c[n - 1] << std::endl;
    return 0;
}
