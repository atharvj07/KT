import sys

for _ in range(int(sys.stdin.readline())):
    s = list(sys.stdin.readline().rstrip())
    A, B, ans = 0, 0, 0
    for ch in s:
        if ch == 'a':
            A += 1
        else:
            B += 1
    if A < B:
        ans += B - A
        for i in range(len(s)):
            if s[i] == 'b':
                s[i] = 'a'
                ans -= 1
                if A == B:
                    break
    else:
        ans += A - B
        for i in range(len(s)):
            if s[i] == 'a':
                s[i] = 'b'
                ans -= 1
                if A == B:
                    break
    print(''.join(s))
