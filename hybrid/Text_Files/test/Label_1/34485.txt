import java.util.*;

class Main {
    static long[] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), d = sc.nextInt(), s = sc.nextInt();
        int[] c = new int[n], f = new int[n], l = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
            f[i] = sc.nextInt();
            l[i] = sc.nextInt();
        }
        dp = new long[s + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = s; j >= f[i]; j--) {
                if (dp[j - f[i]] != -1) {
                    dp[j] = Math.max(dp[j], dp[j - f[i]] + c[i]);
                }
            }
        }
        long ans = 0, fuel = s + 1;
        for (int i = 0; i <= s; i++) {
            if (dp[i] == -1) continue;
            for (int j = 0; j < n; j++) {
                if (c[j] == 0 && l[j] >= d) ans = Math.max(ans, dp[i] + 1);
            }
            if (dp[i] + n >= ans) fuel = Math.min(fuel, i);
        }
        System.out.println(ans + " " + fuel);
    }
}
