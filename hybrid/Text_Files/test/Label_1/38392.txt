from collections import Counter

n = int(input())
a = [int(x) for x in input().split()]
a.sort()
cnt = Counter(a)
mx = 0
ans = 0
for x in cnt:
    if cnt[x] > mx:
        mx = cnt[x]
        ans = x
cur = [x] * mx
for x in a:
    if x == ans:
        if len(cur) == 1:
            ans ^= cur[0]
            cur.pop()
        else:
            ans ^= cur.pop()
    else:
        ans ^= x
print(ans)
