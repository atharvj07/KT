#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

ll gcd(ll a, ll b) {
  return b == 0 ? a : gcd(b, a % b);
}

int main() {
  ll n;
  cin >> n;
  map<ll, ll> xs, ys;
  while (n--) {
    ll x, y;
    cin >> x >> y;
    xs[x]++;
    ys[y]++;
  }
  pair<ll, ll> ax, ay;
  for (auto& x : xs) {
    if (x.second > 1) {
      ax = x;
      break;
    }
  }
  for (auto& y : ys) {
    if (y.second > 1) {
      ay = y;
      break;
    }
  }
  if (xs.size() + ys.size() - (ax.second > 0 && ay.second > 0) == n) {
    cout << "YES" << endl;
  } else
    cout << "NO" << endl;
  return 0;
}
