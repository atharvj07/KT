#include<bits/stdc++.h>
using namespace std;

int n;
int t[91];

int main(){
    cin>>n;
    for(int i=1; i<=n; i++)
        cin>>t[i];
    int ans=0;
    bool flag=false;
    for(int i=1; i<=n; i++){
        if(t[i]-ans>14)
            ans=t[i];
        else{
            flag=true;
            break;
        }
    }
    if(flag)
        cout<<ans<<'\n';
    else
        cout<<90<<'\n';
    return 0;
}
