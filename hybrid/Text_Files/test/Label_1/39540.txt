n = int(input())
a = list(map(int, input().split()))
m = int(input())
b = []
for i in range(m):
    x, y = map(int, input().split())
    b.append((x, y))
j, ans = 0, []
a.sort()
for i, z in b:
    while j < i:
        j += 1
        while j < n and a[j] == a[j - 1]: j += 1
    ans.append(a[j - z + 1])
print(*ans, sep = '\n')
