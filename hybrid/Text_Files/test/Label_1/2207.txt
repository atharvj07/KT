#include <bits/stdc++.h>
using namespace std;

const int maxn = 110;

int n, m, hh;
int a[maxn], b[maxn];
bool top[maxn][maxn];
int h[maxn][maxn];

int main() {
  cin >> n >> m >> hh;
  for (int i = 1; i <= m; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> top[i][j];
      hh -= top[i][j];
    }
  }
  for (int i = 1; i <= m; i++) {
    h[1][i] = a[i];
    for (int j = 2; j <= n; j++) {
      h[j][i] = top[j][i] ? h[j][i - 1] - 1 : h[j][i - 1];
    }
  }
  for (int i = 2; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      h[i][j] -= b[i];
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cout << h[i][j] << " ";
    }
    puts("");
  }
  return 0;
}
