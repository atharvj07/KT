#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> nums(n * n);
  for (int i = 0; i < n * n; i++) {
    cin >> nums[i];
  }
  vector<int> a(n, 1);
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      if (nums[i * n + j] == nums[j * n + i]) {
        a[i] = nums[i * n + j];
        break;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = i; j < n; j++) {
      nums[i * n + j] /= a[i];
      nums[j * n + i] /= a[i];
    }
  }
  vector<int> b(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (nums[i * n + j] == nums[j * n + i]) {
        b[i] = nums[i * n + j];
        break;
      }
    }
  }
  for (int i = 0; i < n; i++) {
    a[i] *= b[i];
  }
  for (int num : a) {
    cout << num << ' ';
  }
  cout << endl;
  return 0;
}
