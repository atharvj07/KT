#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  vector<long long> sum(2000, 0);
  cin >> s;
  sum[0] = (s[0] == '0');
  for (int i = 1; i < s.size(); i++) sum[i] = sum[i - 1] + (s[i] == '0');
  string t = "";
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '0' || (i > 0 && sum[i - 1] == sum[i])) t += '0';
    else t += '1';
  }
  cout << t << endl;
  return 0;
}
