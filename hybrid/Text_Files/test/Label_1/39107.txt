for i in range(int(input())):
    n,k=list(map(int,input().split()))
    s=input()
    mn=[0]*n
    mn[k-1]=0
    for i in range(k-2,-1,-1):
        if(s[i]==s[i+1]):
            mn[i]=mn[i+1]
        else:
            mn[i]=mn[i+1]+1
    ans=mn[0]
    for i in range(n-k):
        if(s[i]==s[i+k]):
            ans=min(ans,mn[i+1])
        else:
            ans=min(ans,mn[i+1]+1)
    print(ans)
