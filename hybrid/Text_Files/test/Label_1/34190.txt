import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] heights = new int[n];
        for (int i = 0; i < n; i++) {
            heights[i] = sc.nextInt();
        }
        int max = 0;
        int width = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0 || heights[i] != heights[i - 1]) {
                width = 0;
            }
            width++;
            max += Math.min(heights[i], width) / 2;
        }
        System.out.println(max);
    }
}
