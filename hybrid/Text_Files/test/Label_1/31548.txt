import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        long k = Long.parseLong(br.readLine());

        for (long i = 1; ; i++) {
            long n = i * 7;
            if (n % k == 0) {
                System.out.println(i);
                break;
            }
        }
    }
}
