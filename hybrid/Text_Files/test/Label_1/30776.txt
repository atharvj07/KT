import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

public class RookAndSpells {
    static class Spell {
        int x1, x2, y;

        public Spell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Integer> verticalSpells = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            verticalSpells.add(scanner.nextInt());
        }

        List<Spell> horizontalSpells = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            horizontalSpells.add(new Spell(scanner.nextInt(), scanner.nextInt(), scanner.nextInt()));
        }

        horizontalSpells.sort(Comparator.comparingInt(a -> a.y));

        int minSpellsToRemove = 0;
        int currentY = 1;
        for (int spell : verticalSpells) {
            if (spell > currentY) {
                minSpellsToRemove++;
                currentY = spell + 1;
            }
        }

        for (Spell spell : horizontalSpells) {
            if (spell.y > currentY) {
                minSpellsToRemove++;
                currentY = spell.y + 1;
            }
        }

        System.out.println(minSpellsToRemove);
    }
}
