#include <bits/stdc++.h>
using namespace std;
const int M = 1e4 + 10;
int p[M], p1[M], p2[M];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> p[i]; }
    while (m--) {
        int l, r, x;
        cin >> l >> r >> x;
        for (int i = l; i <= r; i++) { p1[i] = p[i]; }
        for (int i = l; i < x; i++) { p2[i + 1] = p1[i]; }
        for (int i = x; i <= r; i++) { p2[x] = p1[i]; }
        for (int i = l; i <= r; i++) { p[i] = p2[i]; }
        cout << (p[x] == p1[x] ? "Yes\n" : "No\n");
    }
    return 0;
}
