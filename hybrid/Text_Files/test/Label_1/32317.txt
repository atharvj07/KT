import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            int ans = 0;
            while (b >= 2 && c >= 2) {
                ans += 3;
                b -= 2;
                c -= 2;
            }
            while (a >= 1 && b >= 2) {
                ans += 3;
                a -= 1;
                b -= 2;
            }
            System.out.println(ans);
        }
    }
}
