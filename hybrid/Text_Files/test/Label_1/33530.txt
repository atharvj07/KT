import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] sum = new int[n];
        sum[0] = a[0];
        for (int i = 1; i < n; i++) {
            sum[i] = sum[i - 1] + a[i];
        }
        int[] xor = new int[n];
        xor[0] = a[0];
        for (int i = 1; i < n; i++) {
            xor[i] = xor[i - 1] ^ a[i];
        }
        long ans = 0;
        for (int i = 1; i < n - 1; i++) {
            int l = i - 1, r = i + 1;
            while (l >= 0 && r < n) {
                if ((xor[i] ^ xor[l]) == (sum[r] - sum[i] + a[i])) {
                    ans++;
                    l--;
                    r++;
                } else {
                    break;
                }
            }
        }
        System.out.println(ans);
    }
}
