#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll t, a, b, g;
string s;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr), cout.tie(nullptr);
  cin >> t >> a >> b >> g >> s;
  ll c = 0;
  for (ll i = 1; i <= g; i++) {
    if (s[i - 1] == 'h') {
      if (i == 1 || s[i - 2] != 'h') (c += b), (t -= b);
    }
    t--, c++;
    if (t < 0) break;
  }
  for (ll i = g; i >= 1; i--) {
    if (s[i - 1] == 'h') {
      if (i == g || s[i] != 'h') (c += b), (t -= b);
    }
    if (t < 0) break;
    t--, c++;
  }
  cout << c - 1 << endl;
  return 0;
}
