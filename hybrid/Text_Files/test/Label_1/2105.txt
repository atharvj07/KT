#include <bits/stdc++.h>
using namespace std;

int n, a[100005];
int main() {
    cin >> n;
    if(n % 2) cout << -1 << endl;
    else {
        for(int i = 1; i <= n; i++) a[i] = i;
        for(int i = 2; i <= n; i += 2) swap(a[i], a[n - i + 1]);
        for(int i = 1; i <= n; i++) cout << a[i] << " ";
    }
}
