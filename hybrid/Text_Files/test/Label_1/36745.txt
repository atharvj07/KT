import sys, itertools

n = int(sys.stdin.readline())
a = []
for _ in range(n):
    a.append(list(map(int, sys.stdin.readline().split())))

# get max points for each problem
max_pts = [500] * 5
for x, y in [(0, 1), (1, 2), (2, 4), (3, 8), (4, 16)]:
    for p in range(5):
        for i in range(n):
            if a[i][p] != -1 and a[i][p] > y:
                max_pts[p] = x

# get score for each participant
score = [0] * n
for p in range(5):
    for i in range(n):
        if a[i][p] != -1:
            score[i] += max_pts[p] * (250 - a[i][p]) / 250

# find the minimum number of accounts needed to beat Petya
diff = score[1] - score[0]
if diff <= 0:
    print(-1)
    sys.exit()

accounts = 0
for p in range(5):
    if max_pts[p] * (250 - a[0][p]) / 250 + diff <= 0:
        accounts += 1
        max_pts[p] = 0
        for i in range(n):
            if a[i][p] != -1:
                score[i] -= max_pts[p] * (250 - a[i][p]) / 250

print(accounts)
