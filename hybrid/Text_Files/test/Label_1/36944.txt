for _ in range(int(input())):
    n, k = map(int, input().split())
    s = input()
    i, j, ans = 0, 0, 1000000
    mn = [1000000] * 3
    for j in range(n):
        mn[0 if s[j] == 'R' else 1 if s[j] == 'G' else 2] = min(mn[0 if s[j] == 'R' else 1 if s[j] == 'G' else 2], j)
    for j in range(n):
        if (k - j % k) % 3 == 0:
            ans = min(ans, j - mn[0])
        elif (k - j % k) % 3 == 1:
            ans = min(ans, j - mn[1])
        elif (k - j % k) % 3 == 2:
            ans = min(ans, j - mn[2])
    print(ans)
