import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int q = Integer.parseInt(reader.readLine());
        for (int i = 0; i < q; i++) {
            int n = Integer.parseInt(reader.readLine());
            String s = reader.readLine();
            String t = reader.readLine();
            int[] sIndices = new int[26];
            int[] tIndices = new int[26];
            Arrays.fill(sIndices, -1);
            Arrays.fill(tIndices, -1);
            int ans = 0;
            for (int j = 0; j < n; j++) {
                int sIndex = s.charAt(j) - 'a';
                int tIndex = t.charAt(j) - 'a';
                if (sIndices[sIndex] == -1) {
                    sIndices[sIndex] = j;
                }
                if (tIndices[tIndex] == -1) {
                    tIndices[tIndex] = j;
                }
                ans += Math.abs(sIndices[sIndex] - tIndices[tIndex]);
                sIndices[sIndex]++;
                tIndices[tIndex]--;
            }
            for (int j = 0; j < 26; j++) {
                if (sIndices[j] != tIndices[j]) {
                    ans = -1;
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
