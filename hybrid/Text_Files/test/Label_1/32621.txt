import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        char[] colors = { 'R', 'O', 'Y', 'G', 'B', 'I', 'V' };
        char[] eggs = new char[n];
        int[] cnt = new int[7];
        for (int i = 0; i < n; i++) {
            eggs[i] = colors[i%7];
            cnt[i%7]++;
        }
        for (int i = 0; i < n; i++) {
            if (cnt[i%7] < 2) {
                continue;
            }
            for (int j = i + 1; j < n; j++) {
                if (eggs[(j - 1) % n] == eggs[(j + 1) % n]) {
                    eggs[j] = colors[(j + 1) % 7];
                    cnt[(j + 1) % 7]++;
                    break;
                }
            }
        }
        System.out.println(new String(eggs));
    }
}
