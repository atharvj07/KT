import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long minx = Long.MAX_VALUE, maxx = Long.MIN_VALUE, miny = Long.MAX_VALUE, maxy = Long.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            long ax = sc.nextInt(), bx = sc.nextInt(), cy = sc.nextInt();
            minx = Math.min(minx, ax);
            maxx = Math.max(maxx, bx);
            miny = Math.min(miny, cy);
            maxy = Math.max(maxy, cy);
        }
        for (int i = 0; i < m; i++) {
            long cx = sc.nextInt(), cy = sc.nextInt(), dx = sc.nextInt();
            minx = Math.min(minx, cx);
            maxx = Math.max(maxx, cx);
            miny = Math.min(miny, cy);
            maxy = Math.max(maxy, dx);
        }
        if (minx <= 0 && maxx >= 0 && miny <= 0 && maxy >= 0) {
            System.out.println("INF");
            return;
        }
        long ans = (maxx - minx) * (maxy - miny);
        System.out.println(ans);
    }
}
