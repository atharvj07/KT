import sys
input = sys.stdin.readline

n, m = map(int, input().split())
g = [[] for i in range(n+1)]
for i in range(m):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)

ans = 0

def dfs(u, p, dep):
    global ans
    for v in g[u]:
        if v == p: continue
        if dep & 1: ans += 1
        dfs(v, u, dep+1)

dfs(1, 1, 0)
print(ans//2)
