#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  string t;
  t += s[0];
  for (int i = 1; i < n; i++) {
    if (t.back() == s[i] && t.size() >= 2 && t[t.size() - 2] == s[i]) {
      continue;
    }
    if (s[i] != s[i - 1] || i == n - 1 || s[i] != s[i + 1]) {
      t += s[i];
    }
  }
  cout << t;
  return 0;
}
