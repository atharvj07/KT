import java.util.Scanner;

public class Main {
    private static long mod = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        long ans = (fastpow(2, n % (m - 1)) - 1 + mod) % mod;
        System.out.println(ans);
    }

    private static long fastpow(long base, long exp) {
        if (exp == 0) {
            return 1;
        }
        if (exp == 1) {
            return base;
        }
        long halfpow = fastpow(base, exp / 2);
        long fullpow = (halfpow * halfpow) % mod;
        if (exp % 2 == 1) {
            fullpow = (fullpow * base) % mod;
        }
        return fullpow;
    }
}
