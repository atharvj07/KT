from sys import stdin
n, m = map(int, stdin.readline().split())
field = [stdin.readline().strip() for i in range(n)]
for i in range(n):
    for j in range(m):
        if field[i][j] == '*':
            field[i][j] = '.'
            comp = [[i, j]]
            while comp:
                ni, nj = comp.pop()
                for di, dj in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    if 0 <= ni + di < n and 0 <= nj + dj < m and field[ni + di][nj + dj] == '.':
                        comp.append([ni + di, nj + dj])
                        field[ni + di][nj + dj] = str((int(field[ni + di][nj + dj]) + 1) % 10)
            field[i][j] = '*'
print(''.join(''.join(line) for line in field))
