import sys
input = sys.stdin.readline

n, m = map(int, input().split())

x = [0] * n
for i in range(n):
    s, xi = map(float, input().split())
    x[i] = (s, xi)

x.sort(key = lambda p: p[1])

cur = 0
ans = 0
for s, xi in x:

    if s != cur + 1:
        ans += 1
    else:
        cur += 1

print(ans)
