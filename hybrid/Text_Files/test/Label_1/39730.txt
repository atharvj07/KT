from collections import defaultdict, deque

def solve(n, k, edges):
    if n <= 2:
        return max(0, n - k * 2)

    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)

    leaves = deque([node for node in adj if len(adj[node]) == 1])
    depth = {leaf: 1 for leaf in leaves}

    while leaves:
        leaf = leaves.popleft()
        if depth[leaf] > k:
            break
        for neighbor in adj[leaf]:
            adj[neighbor].remove(leaf)
            if len(adj[neighbor]) == 1:
                leaves.append(neighbor)
                depth[neighbor] = depth[leaf] + 1
        n -= 1

    return n

t = int(input().strip())
for _ in range(t):
    input()
    n, k = map(int, input().strip().split())
    edges = [list(map(int, input().strip().split())) for _ in range(n - 1)]
    print(solve(n, k, edges))
