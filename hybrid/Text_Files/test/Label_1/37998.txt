import sys

mod = 1000000007
n = int(sys.stdin.readline())
edges = [int(x) for x in sys.stdin.readline().split()]
colors = [int(x) for x in sys.stdin.readline().split()]

adj = [[] for _ in range(n)]
for i in range(n - 1):
    adj[edges[i]].append(i + 1)
    adj[i + 1].append(edges[i])

cnt = [0] * n

def dfs(u, p):
    cnt[u] = colors[u]
    for v in adj[u]:
        if v == p:
            continue
        dfs(v, u)
        cnt[u] += cnt[v]
    return cnt[u]

dfs(0, -1)

ans = 0
for i in range(n):
    if cnt[i] == 1:
        ans += 1

print(ans % mod)
