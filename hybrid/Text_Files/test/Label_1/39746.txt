from collections import Counter

n = int(input())
words = [input() for _ in range(n)]
wcnt = Counter(words)
ans = 0
pairs = []
for w1 in words:
    for w2 in words:
        if w1 != w2 and Counter(w1)[-1] == Counter(w2)[-1]:
            pairs.append((w1, w2))
            ans += wcnt[w1] * wcnt[w2]
print(ans, *pairs, sep='\n')
