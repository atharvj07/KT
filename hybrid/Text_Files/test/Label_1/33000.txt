import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        String t = scanner.next();
        if (!s.startsWith(t.substring(0, Math.min(t.length(), s.length())))) {
            System.out.println(-1);
            return;
        }
        int k = 0;
        while (!s.equals(t)) {
            int idx = s.indexOf(t.substring(0, t.length() - 1));
            if (idx == -1) {
                idx = 0;
            }
            s = s.substring(idx + 1) + s.charAt(idx);
            k++;
        }
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(s.length() + " ");
        }
    }
}
