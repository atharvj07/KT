import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] g = new int[n];
        long sumA = 0, sumG = 0;
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            g[i] = sc.nextInt();
            sumA += a[i];
            sumG += g[i];
        }

        if (Math.abs(sumA - sumG) > 500) {
            System.out.println("-1");
            return;
        }

        if (sumA > sumG) {
            for (int i = 0; i < n; i++) {
                if (a[i] > g[i]) {
                    System.out.print("A");
                } else {
                    System.out.print("G");
                }
            }
        } else {
            for (int i = 0; i < n; i++) {
                if (a[i] < g[i]) {
                    System.out.print("A");
                } else {
                    System.out.print("G");
                }
            }
        }
    }
}
