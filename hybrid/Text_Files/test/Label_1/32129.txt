import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static final int MOD = (int) (1e9 + 7);

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Map<Integer, List<Segment>> xToSegments = new HashMap<>();
        Map<Integer, List<Segment>> yToSegments = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int x1 = Integer.parseInt(line[0]);
            int y1 = Integer.parseInt(line[1]);
            int x2 = Integer.parseInt(line[2]);
            int y2 = Integer.parseInt(line[3]);
            Segment segment = new Segment(x1, y1, x2, y2);
            xToSegments.computeIfAbsent(x1, k -> new ArrayList<>()).add(segment);
            xToSegments.computeIfAbsent(x2, k -> new ArrayList<>()).add(segment);
            yToSegments.computeIfAbsent(y1, k -> new ArrayList<>()).add(segment);
            yToSegments.computeIfAbsent(y2, k -> new ArrayList<>()).add(segment);
        }
        long ans = 0;
        for (int x : xToSegments.keySet()) {
            List<Segment> segments = xToSegments.get(x);
            for (Segment segment1 : segments) {
                for (Segment segment2 : segments) {
                    if (segment1.isVertical() || segment2.isVertical()) {
                        continue;
                    }
                    int y1 = segment1.y1;
                    int y2 = segment1.y2;
                    int y3 = segment2.y1;
                    int y4 = segment2.y2;
                    long count = 0;
                    if (y1 < y3 && y2 < y3) {
                        count += yToSegments.getOrDefault(y3, new ArrayList<>()).stream()
                                .filter(s -> s.isVertical()).count();
                    }
                    if (y1 < y4 && y2 < y4) {
                        count += yToSegments.getOrDefault(y4, new ArrayList<>()).stream()
                                .filter(s -> s.isVertical()).count();
                    }
                    ans += count * count % MOD;
                    ans %= MOD;
                }
            }
        }
        System.out.println(ans);
    }

    private static class Segment {
        int x1, y1, x2, y2;

        public Segment(int x1, int y1, int x2, int y2) {
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
        }

        public boolean isVertical() {
            return x1 == x2;
        }
    }
}
