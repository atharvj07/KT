import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            char[] s = scanner.next().toCharArray();
            int ans = 0;
            int n = s.length;
            for (int i = 0; i < n; i++) {
                if (s[i] == '?') {
                    int j = i;
                    while (j < n && s[j] == '?') {
                        j++;
                    }
                    if (j == n) {
                        ans += (j - i) * (j - i + 1) / 2;
                    } else {
                        if (s[j] == '0' || s[j] == '1') {
                            ans += (j - i) * (j - i + 1) / 2;
                        } else {
                            ans += (j - i);
                        }
                        i = j;
                    }
                } else {
                    int j = i + 1;
                    while (j < n && s[j] != '?') {
                        j++;
                    }
                    ans += j - i;
                    i = j - 1;
                }
            }
            System.out.println(ans);
        }
    }
}
