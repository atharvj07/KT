#include <bits/stdc++.h>
using namespace std;
#define int long long

vector<int> a, b;

int32_t main() {
  int n;
  cin >> n;
  a.resize(n);  
  b.resize(n);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    a[i] = x % 2;
    b[i] = x % 2;
    ans ^= a[i];
  }
  if (ans == 1) {
    cout << "No";
  } else {
    cout << "Yes";
  }
}
