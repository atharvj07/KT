#include<bits/stdc++.h>
using namespace std;

constexpr int maxn = 1e5 + 5;
vector<int> adj[maxn];
bool vis[maxn];

bool dfs(int u,int p)
{
    vis[u] = true;
    bool ret = false;
    for (auto &v : adj[u])
    {
        if (v == p) continue;
        if (!vis[v])
        {
            ret |= dfs(v,u);
            if (ret) return true;
        }
        else
        {
            return true;
        }
    }
    return ret;
}

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n-1; i++)
    {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    bool ok = dfs(1,1);
    if (!ok) {
        cout << "No\n";
        return 0;
    }
    cout << "Yes\n";
    cout << n-1 << endl;
    for (int i = 0; i < n-1; i++)
    {
        cout << adj[i+1][0] << ' ' << adj[i+1][1] << endl;
    }
    return 0;
}
