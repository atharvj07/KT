import math

n = int(input())
a = list(map(int, input().split()))

if max(a) + 1 < n:
    print("NO")
else:
    m = n - 1
    d = math.ceil(math.log2(n))
    print("YES", d)
    print(m)
    for i in range(2, n + 1):
        for j in range(1, i):
            if a[i - 1] > 0 and a[j - 1] > 0:
                print(i, j)
                a[i - 1] -= 1
                a[j - 1] -= 1
                m -= 1
                if not m:
                    break
