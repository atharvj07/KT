#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll ord[110],a[110],b[110];
vector<pair<ll,ll>>v;
int main(){
	ll n,m;
	cin>>n>>m;
	for(ll i=1;i<=n;i++)cin>>a[i],b[i]=a[i];
	for(ll i=1;i<=m;i++){
		ll x,y;
		cin>>x>>y;
		v.push_back({x,y});
		ord[x]++;
		ord[y]++;
	}
	ll ans=0,d=0;
	while(d<m){
		ll pos=-1,val=1e9;
		for(ll i=1;i<=n;i++){
			if(ord[i]%2==1 && a[i]!=1 && val>a[i])pos=i,val=a[i];
		}
		if(pos==-1)break;
		for(auto j:v){
			if(j.first==pos || j.second==pos){
				ll x,y;
				x=a[j.first];
				y=a[j.second];
				ll g=__gcd(x,y);
				if(g!=1)a[j.first]/=g,a[j.second]/=g;
				if(a[j.first]==1 || a[j.second]==1)d++;
			}
		}
		ans++;
	}
	cout<<ans<<endl;
}
