#include <bits/stdc++.h>

using namespace std;

vector<int> p;
vector<bool> was;
int val;

int main() {
  int n, m;
  cin >> n >> m;
  p.resize(n + 1);
  was.resize(n + 1);
  for (int i = 1; i <= n; ++i) cin >> p[i];
  while (m--) {
    int l, r, x;
    cin >> l >> r >> x;
    if (!was[x]) {
      val = p[x];
      was[p[x]] = true;
    }
    reverse(p.begin() + l, p.begin() + r + 1);
    if (p[x] != val)
      cout << "No" << endl;
    else
      cout << "Yes" << endl;
  }
  return 0;
}
