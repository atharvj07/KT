#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define MOD 998244353

int main() {
    string s;
    cin >> s;
    int l, r;
    cin >> l >> r;

    int m = s.length();
    vector<vector<ll>> dp(m + 1, vector<ll>(r + 1));
    dp[m][0] = 1;
    for (int i = m - 1; i >= 0; i--) {
        for (int j = 0; j <= r; j++) {
            if(s[i]=='0' && j==0) continue;
            dp[i][j] = (dp[i + 1][j] + (j >= l ? dp[i + 1][j - int(s[i] - '0')] : 0)) % MOD;
        }
    }
    cout << dp[0][r] << "\n";
    return 0;
}
