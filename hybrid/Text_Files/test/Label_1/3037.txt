#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int MAXN = 1e5;

ll a[MAXN], pre[MAXN], suf[MAXN];

int main() {
    ll n, ans = LLONG_MIN;
    cin >> n;
    for (ll i = 0; i < n; i++) { cin >> a[i]; }
    for (ll i = 1; i < n; i++) { pre[i] = abs(a[i] - a[i - 1]); }
    for (ll i = n - 2; i >= 0; i--) { suf[i] = abs(a[i + 1] - a[i]); }
    for (ll i = 0; i < n; i++) { ans = max(ans, pre[i] + suf[i]); }
    cout << ans << endl;
    return 0;
}
