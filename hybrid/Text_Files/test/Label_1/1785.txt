#include <iostream>
#include <utility>
#include <vector>
#include <map>

using namespace std;

int main()
{
	map<pair<int, char>, int> tiles;
	string tile;
	for(int i = 0; i < 3; i++)
	{
		cin >> tile;
		int suit = tile[1];
		int num = tile[0] - '0';
		if(tiles.count(make_pair(num, suit)) == 0)
			tiles.insert(make_pair(make_pair(num, suit), 1));
		else
			tiles[make_pair(num, suit)]++;
	}
	int ans = 0;
	for(auto tile : tiles)
	{
		if(tile.second >= 3)
			ans -= 2;
		else if(tile.second == 2)
			ans++;
	}
	ans = max(0, ans);
	cout << ans << endl;
	return 0; 
}
