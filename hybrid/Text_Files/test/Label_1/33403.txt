import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int k = sc.nextInt();
        int t = sc.nextInt();

        long mod = 1000000007;
        long[][] dp = new long[2 * k + 1][t + 1];
        dp[k][0] = 1;
        for (int i = 1; i <= t; i++) {
            for (int j = -k; j <= k; j++) {
                for (int l = -k; l <= k; l++) {
                    if (a + j <= b + l) continue;
                    dp[k + j][i] = (dp[k + j][i] + dp[k + l][i - 1]) % mod;
                }
            }
        }
        System.out.println(dp[k][t]);
    }
}
