#include <bits/stdc++.h>
using namespace std;
const int MAX = 100005;
bool open[MAX];
int main() {
  int n, pairs = 0;
  string s;
  cin >> n;
  for (int i = 0; i < n; i++) {
    int pos = 0;
    cin >> s;
    int l = s.length();
    for (int j = 0; j < l; j++) {
      if (s[j] == '(')
        pos++;
      else
        pos--;
      if (pos < 0) break;
    }
    if (pos == 0) {
      open[i] = 1;
      pairs++;
    }
  }
  for (int i = 0; i < n; i++) {
    if (open[i]) continue;
    for (int j = i + 1; j < n; j++) {
      if (!open[j]) continue;
      string t = s + s;
      int pos = 0;
      for (int l = i; l < j; l++) {
        if (t[l] == '(')
          pos++;
        else
          pos--;
        if (pos < 0) break;
      }
      if (pos == 0) { pairs++; }
    }
  }
  cout << pairs;
  return 0;
}
