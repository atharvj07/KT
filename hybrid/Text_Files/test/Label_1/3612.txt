#include <bits/stdc++.h>
#define MOD 1000000007

using namespace std;

long long dp[1000000];

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> bombs;
  for (int i = 0; i < n; i++) {
    if (s[i] == '*') bombs.push_back(i);
  }
  dp[0] = 1;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j < bombs.size(); j++) {
      if (i - bombs[j] <= 2) {
        dp[i] = 0;
        break;
      }
    }
    if (!dp[i]) continue;
    for (int j = 1; j <= 2; j++) {
      if (i - j >= 0) dp[i] = (dp[i] + dp[i - j]) % MOD;
    }
  }
  cout << dp[n] << endl;
  return 0;
}
