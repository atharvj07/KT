t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    s = input().strip()

    # Counting deletions needed from both ends until a matching pair is found
    left_del = right_del = 0
    for char in s:
        if char == '>':
            break
        left_del += 1
    for char in reversed(s):
        if char == '<':
            break
        right_del += 1
    
    # The minimum number of deletions needed is the smaller count of '<' from the start or '>' from the end
    print(min(left_del, right_del))
