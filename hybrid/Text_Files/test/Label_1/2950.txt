#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
using ll = long long;
using vi = vector<int>;
void solve() {
  int n;
  cin >> n;
  vi v(n);
  for (auto& i : v) { cin >> i; }
  sort(v.begin(), v.end());
  ll ans = 0;
  for (int i = 0; i < n - 2; ++i) {
    int l = i + 1, r = i + 2;
    while (l < n && r < n) {
      if (v[r] - v[l] <= 2) {
        ans += n - r;
        ++l;
      } else {
        ++r;
      }
    }
  }
  cout << ans << '\n';
}
int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
}
