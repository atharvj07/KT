for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    ans = []
    for i in a:
        if not ans:
            ans.append(i)
        else:
            x = ans[-1]
            if x % i == 0:
                ans.append(i)
            else:
                y = i // x
                if y > x % i:
                    ans.append(x % i)
                    ans.append(y)
                else:
                    ans.append(y)
                    ans.append(x % i)
    print(*ans)
