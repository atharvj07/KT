#include <bits/stdc++.h>
using namespace std;
#define long long long long
const int MAXN = 1e5 + 1, MAXQ = 1e5 + 1;
int n, a[MAXN], t[MAXN], x[MAXN];
vector<pair<int, int>> v[MAXQ];
map<pair<int, int>, int> cnt;
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i] >> t[i] >> x[i];
        v[t[i]].push_back({a[i], x[i]});
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < (int)v[i].size(); j++) {
            auto p = v[i][j];
            if (p.first == 1) {
                cnt[{p.second, i}]++;
            }
            if (p.first == 2) {
                cnt[{p.second, i}]--;
            }
            if (p.first == 3) {
                cout << cnt[{p.second, i}] << '\n';
            }
        }
    }
    return 0;
}
