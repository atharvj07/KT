import java.util.*;

public class Main {
    static final long MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] xs = new long[n];
        long[] ys = new long[n];
        for (int i = 0; i < n; i++) {
            xs[i] = scanner.nextLong();
            ys[i] = scanner.nextLong();
        }
        scanner.close();
        
        long[][] dp = new long[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        
        long ans = 0;
        for (int i = 3; i <= n; i++) {
            ans = Math.max(ans, f(0, i - 1, xs, ys, dp));
        }
        
        System.out.println(ans);
    }
    
    static long f(int l, int r, long[] xs, long[] ys, long[][] dp) {
        if (l == r) {
            return 0;
        }
        if (dp[l][r] != -1) {
            return dp[l][r];
        }
        
        long ans = 0;
        for (int i = l + 1; i < r; i++) {
            long dx1 = xs[i] - xs[l];
            long dy1 = ys[i] - ys[l];
            long dx2 = xs[r] - xs[l];
            long dy2 = ys[r] - ys[l];
            
            long cross = dx1 * dy2 - dx2 * dy1;
            if (cross > 0 || (cross == 0 && (dx1 * dx2 + dy1 * dy2) <= 0)) {
                ans = Math.max(ans, f(l, i, xs, ys, dp) + f(i, r, xs, ys, dp));
            }
        }
        
        ans += Math.abs(xs[l] - xs[r]) + Math.abs(ys[l] - ys[r]);
        
        return dp[l][r] = ans;
    }
}
