import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, x = map(int, input().split())
    a = list(map(int, input().split()))
    ans = sum(a)
    for i in range(n):
        if a[i] % x == 0:
            ans += (a[i] // x) * x
        else:
            break
    print(ans)
