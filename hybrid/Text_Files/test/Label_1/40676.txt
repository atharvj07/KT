import sys
input=sys.stdin.readline
n, m, s, t=map(int,input().split())
l=[[] for i in range(n+1)]
for i in range(m):
	a, b=map(int,input().split())
	l[a].append(b)
	l[b].append(a)
vis=[0]*(n+1)
def dfs(a, d):
	global ans
	vis[a]=1
	if a==t:
		ans=d-1
		return
	for b in l[a]:
		if vis[b]==0:
			dfs(b, d+1)
ans=0
dfs(s, 0)
vis=[0]*(n+1)
dfs(t, 0)
ans*=2
print(n*(n-1)//2-m-ans)
