import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int A = in.nextInt();
		int B = in.nextInt();
		int C = in.nextInt();
		int D = in.nextInt();
		long res = 0;
		for (long i = 2; i <= n; i++) {
			long sum = A * i * i * i + B * i * i + C * i + D;
			long k = 1;
			long j = 2;
			while (k <= n) {
				res = (res + sum * k) % 4294967296L;
				k *= j++;
			}
		}
		System.out.println(res % 4294967296L);
	}
}
