import java.util.*;

class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] p = new int[n];
		for (int i = 0; i < n; i++) {
			p[i] = sc.nextInt() - 1;
		}
		int[] f = new int[n];
		for (int i = 0; i < n; i++) {
			f[i] = sc.nextInt();
		}
		boolean[] vis = new boolean[n];
		for (int i = 0; i < n; i++) {
			if (!vis[i]) {
				List<Integer> cycle = new ArrayList<>();
				int j = i;
				while (!vis[j]) {
					vis[j] = true;
					cycle.add(j);
					j = p[j];
				}
				if (cycle.size() > 1) {
					boolean flag = false;
					for (int k = 0; k < cycle.size(); k++) {
						if (f[cycle.get(k)] == cycle.size()) {
							flag = true;
						}
					}
					if (!flag) {
						System.out.println("NO");
						return;
					}
				}
			}
		}
		System.out.println("YES");
	}
}
