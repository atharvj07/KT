import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String fyodor = scanner.nextLine();
        String matroskin = scanner.nextLine();
        String sharic = scanner.nextLine();
        if (fyodor.equals(matroskin) && matroskin.equals(sharic)) {
            System.out.println("?");
        } else if (fyodor.equals("rock") && matroskin.equals("scissors") && sharic.equals("scissors")) {
            System.out.println("F");
        } else if (matroskin.equals("rock") && fyodor.equals("scissors") && sharic.equals("scissors")) {
            System.out.println("M");
        } else if (sharic.equals("rock") && matroskin.equals("scissors") && fyodor.equals("scissors")) {
            System.out.println("S");
        } else if (fyodor.equals("paper") && matroskin.equals("rock") && sharic.equals("rock")) {
            System.out.println("F");
        } else if (matroskin.equals("paper") && fyodor.equals("rock") && sharic.equals("rock")) {
            System.out.println("M");
        } else if (sharic.equals("paper") && matroskin.equals("rock") && fyodor.equals("rock")) {
            System.out.println("S");
        } else {
            System.out.println("?");
        }
    }
}
