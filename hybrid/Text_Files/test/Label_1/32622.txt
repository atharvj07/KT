import java.util.*;
import java.io.*;

class Main {
    static char[][] map;
    static boolean[][] bcolor;
    static boolean[][] wcolor;
    static int[] dy = {1,-1,0,0};
    static int[] dx = {0,0,1,-1};
    static int n,m;

    static void dfs_b(int y, int x) {
        if (y < 0 || y >= n || x < 0 || x >= m) return;
        if (map[y][x] == 'B') return;
        if (bcolor[y][x]) return;
        bcolor[y][x] = true;
        dfs_b(y+1,x);
        dfs_b(y-1,x);
        dfs_b(y,x+1);
        dfs_b(y,x-1);
    }

    static void dfs_w(int y, int x) {
        if (y < 0 || y >= n || x < 0 || x >= m) return;
        if (map[y][x] == 'W') return;
        if (wcolor[y][x]) return;
        wcolor[y][x] = true;
        dfs_w(y+1,x);
        dfs_w(y-1,x);
        dfs_w(y,x+1);
        dfs_w(y,x-1);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        while(true) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            m = Integer.parseInt(st.nextToken());
            n = Integer.parseInt(st.nextToken());
            if (m == 0 && n == 0) break;
            map = new char[n][m];
            bcolor = new boolean[n][m];
            wcolor = new boolean[n][m];
            for(int i=0; i<n; i++) {
                char[] chs = br.readLine().toCharArray();
                for(int j=0; j<m; j++) {
                    map[i][j] = chs[j];
                }
            }

            int bcnt = 0;
            int wcnt = 0;
            for(int i=0; i<n; i++) {
                for(int j=0; j<m; j++) {
                    if (map[i][j] == 'B') {
                        bcnt++;
                        continue;
                    }
                    if (!bcolor[i][j] && map[i][j] != 'W') {
                        dfs_b(i,j);
                    }
                    if (!wcolor[i][j] && map[i][j] != 'B') {
                        dfs_w(i,j);
                    }
                    if (bcolor[i][j] && !wcolor[i][j]) bcnt++;
                    if (!bcolor[i][j] && wcolor[i][j]) wcnt++;
                }
            }
            bw.write(bcnt + " " + wcnt);
            bw.newLine();
        }
        bw.flush();
        bw.close();
    }
}
