import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[] s = new long[n];
        long[] d = new long[n];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            s[i] = Long.parseLong(line[0]);
            d[i] = Long.parseLong(line[1]);
        }
        long[] start = new long[n];
        long[] end = new long[n];
        long t = 1;
        for (int i = 0; i < n; i++) {
            if (t >= s[i]) {
                start[i] = t;
                t += d[i];
                end[i] = start[i] + d[i] - 1;
            } else {
                start[i] = s[i];
                t = s[i] + d[i];
                end[i] = start[i] + d[i] - 1;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.println(start[i] + " " + end[i]);
        }
    }
}
