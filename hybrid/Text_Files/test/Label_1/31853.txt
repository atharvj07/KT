import java.util.Scanner;

public class Main {
  static final int MOD = 1000000007;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), k = in.nextInt();
    in.close();

    int[][][] dp = new int[n + 1][k + 1][2]; // dp[i][j][0] == no need to return back, dp[i][j][1] == need
    dp[1][0][0] = dp[1][0][1] = dp[1][1][0] = 1;
    for (int i = 2; i <= n; i++) {
      for (int j = 0; j <= k; j++) {
        if (j == 0) {
          dp[i][j][0] = (dp[i - 1][j][0] + dp[i - 1][j][1]) % MOD;
          dp[i][j][1] = (dp[i - 1][j][0] + dp[i][j][0]) % MOD;
        } else {
          dp[i][j][0] = (dp[i - 1][j - 1][0] + dp[i - 1][j - 1][1]) % MOD;
          dp[i][j][1] = (dp[i - 1][j - 1][0] + dp[i][j][0]) % MOD;
        }
      }
    }
    System.out.println(dp[n][k][0]);
  }
}
