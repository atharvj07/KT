#include<bits/stdc++.h>
using namespace std;
int main(){
    int n, a, x, b, y;
    cin >> n >> a >> x >> b >> y;
    bool swap = false;
    if (abs(a - x) > abs(b - y)) {
        swap = true;
        a ^= b ^= a ^= b;
        x ^= y ^= x ^= y;
    }
    if (abs(a - x) < abs(b - y))
        cout << "NO";
    else if (a != x && b != y && ((x < a && y > b) || (y < b && x > a)))
        cout << "NO";
    else
        cout << "YES";
}
