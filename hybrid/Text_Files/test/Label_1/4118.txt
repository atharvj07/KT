#include <iostream>
#include <cstring>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;
int main() {
    ll n; cin >> n;
    string s, t; cin >> s >> t;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] != t[i]) {
            if (s[i] == '0') {
                ans = -1;
                break;
            }
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
