n = input()
ans = 'YES'
for i, s in enumerate(n):
    if i != len(n) - 1 and s != 'n' and s != 'a' and s != 'e' and s != 'i' and s != 'o' and s != 'u':
        if n[i + 1] != 'a' and n[i + 1] != 'e' and n[i + 1] != 'i' and n[i + 1] != 'o' and n[i + 1] != 'u':
            ans = 'NO'
            break
if n[-1] != 'a' and n[-1] != 'e' and n[-1] != 'i' and n[-1] != 'o' and n[-1] != 'u':
    ans = 'NO'
print(ans)
