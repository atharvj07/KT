def is_divisible_by_60(number):
    # A number is divisible by 60 if and only if it is divisible by 3 and by 20.
    # To be divisible by 3, the sum of its digits must be divisible by 3.
    # To be divisible by 20, it must end in a 0 and be even.
    # Since we can rearrange digits, we just need at least one '0' and at least two even digits (including the '0').
    # Also, the sum of digits must be divisible by 3.
    digit_count = [0] * 10
    digit_sum = 0

    for digit in number:
        digit_count[int(digit)] += 1
        digit_sum += int(digit)

    has_zero = digit_count[0] > 0
    even_digits = sum(digit_count[2*i] for i in range(5))

    return has_zero and even_digits >= 2 and digit_sum % 3 == 0

n = int(input().strip())

for _ in range(n):
    y = input().strip()
    if is_divisible_by_60(y):
        print("red")
    else:
        print("cyan")
