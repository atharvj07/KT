#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
int n, a[105];
int b[105], e[105], c[105], d[105];
bool cha[3], chk[3];
ll ans;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    if (n <= 4 || (n == 5 && a[1] == a[5])) {
        ans = 4;
        cout << "Finite" << endl << ans << endl;
        return 0;
    }
    for (int i = 1; i <= n; i++) {
        if (a[i] == 1) {
            cha[0] = 1;
            if (!chk[0]) {
                chk[0] = 1;
                d[0]++;
            }
            b[i] = e[i] = 0;
            if (c[i - 1]) ans += c[i - 1];
            if (d[i - 1]) ans += d[i - 1];
            c[i]++;
        } else if (a[i] == 2) {
            cha[1] = 1;
            if (!chk[1]) {
                chk[1] = 1;
                c[0]++;
            }
            b[i] = 0;
            e[i] = 1;
            if (d[i - 1]) ans += d[i - 1];
            d[i]++;
        } else if (a[i] == 3) {
            cha[2] = 1;
            if (!chk[2]) {
                chk[2] = 1;
                b[0]++;
            }
            e[i] = 0;
            b[i] = 1;
            if (c[i - 1]) ans += c[i - 1];
            if (d[i - 1]) ans += d[i - 1];
            d[i]++;
        }
        if (chk[0] && chk[1] && chk[2] && cha[0] && cha[1] && cha[2]) ans++;
    }
    if (ans == -1) ans = 0;
    cout << (ans >= LLONG_MAX ? "Infinite" : "Finite") << endl;
    if (ans < LLONG_MAX) cout << ans << endl;
    return 0;
}
