t = int(input())
for _ in range(t):
    n = int(input())
    p = list(map(int, input().split()))
    m = int(input())
    q = list(map(int, input().split()))
    p_count = {}
    q_count = {}
    for item in p:
        if item % 2 == 0:
            p_count[0] = p_count.get(0, 0) + 1
        else:
            p_count[1] = p_count.get(1, 0) + 1
    for item in q:
        if item % 2 == 0:
            q_count[0] = q_count.get(0, 0) + 1
        else:
            q_count[1] = q_count.get(1, 0) + 1
    result = p_count.get(0, 0) * q_count.get(0, 0) + p_count.get(1, 0) * q_count.get(1, 0)
    print(result)
