#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MAXN = 200;

int matrix[MAXN][MAXN];
vector<int> rows, cols;

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> matrix[i][j];
        }
    }
    for (int i = 0; i < n; i++) {
        rows.push_back(i);
    }
    for (int j = 0; j < m; j++) {
        cols.push_back(j);
    }
    bool can_sort = true;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (matrix[i][j] != (i < j)) {
                can_sort = false;
            }
        }
    }
    if (!can_sort) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) {
        cout << (rows[i] % 2);
    }
    cout << endl;
    for (int j = 0; j < m; j++) {
        cout << (cols[j] % 2);
    }
    cout << endl;
    return 0;
}
