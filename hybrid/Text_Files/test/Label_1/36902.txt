import math

# Fermat's Little Theorem
def modpow(base, exp, mod):
    return pow(base, exp, mod)

# Binomial Coefficient
def binom(n, k, mod):
    res = math.factorial(n) // ((math.factorial(k) * math.factorial(n - k)) % mod)
    return res % mod

n, k = map(int, input().split())
s = list(map(int, input().split()))
ans = 1
for i in range(n - 1):
    x = s[i + 1] - s[i] - 1
    if x > 0:
        ans = ans * binom(n - i - 1, x, 998244353) % 998244353
    ans = ans * k % 998244353
    k -= 1
print(ans)
