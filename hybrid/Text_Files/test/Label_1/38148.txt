n = int(input())
a = list(map(int, input().split()))
k = int(input())
b = list(map(int, input().split()))
if a == b:
    print("YES")
    exit()
if b[0] < a[0]:
    print("NO")
    exit()
for i in range(1, k):
    if b[i] < a[i] or b[i] > b[i - 1] + a[i]:
        print("NO")
        exit()
print("YES")
for i in range(n - k):
    if b[0] > a[i + 1]:
        print(1, "R")
        b[0] += a[i + 1]
        a[i + 1] = 0
    elif b[k - 1] > a[i]:
        print(k, "L")
        b[k - 1] += a[i]
        a[i] = 0
    else:
        print("NO")
        exit()
