import java.util.Scanner;

public class ParenthesisSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        scanner.nextLine();
        String s = scanner.nextLine();

        StringBuilder result = new StringBuilder();
        int balance = 0;

        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == '(') {
                balance++;
            } else if (c == ')') {
                balance--;
            }

            if (c == '?' && balance > 0) {
                result.append('(');
                balance++;
            } else if (c == '?' && balance == 0) {
                result.append(')');
                balance--;
            } else if (c == '?') {
                result.append('(');
            } else {
                result.append(c);
            }
        }

        if (balance != 0) {
            System.out.println(":(");
        } else {
            System.out.println(result);
        }
    }
}
