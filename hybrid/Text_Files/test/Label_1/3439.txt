#include <bits/stdc++.h>
using namespace std;

void solve() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  string s[n + 1];
  s[0] = "a";
  for (int i = 0; i < n; i++) {
    string ch = s[i].substr(0, a[i]);
    s[i + 1] = ch + (char)('a' + (i + 1) % 26);
  }
  for (int i = 0; i <= n; i++) {
    cout << s[i] << endl;
  }
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}
