import sys

n, m = (int(x) for x in sys.stdin.readline().split())
t = [int(x) for x in sys.stdin.readline().split()]
adj = [[] for _ in range(n)]
for _ in range(m):
    a, b = (int(x) - 1 for x in sys.stdin.readline().split())
    adj[b].append(a)
ans = []
vis = [False for _ in range(n)]
def visit(i):
    if vis[i]:
        return
    vis[i] = True
    ans.append(i)
    for j in adj[i]:
        visit(j)
for i in range(n):
    if not vis[i]:
        visit(i)
print(len(ans) // (sum(1 for x in adj if x) or 1))
