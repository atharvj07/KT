#include <iostream>
#include <vector>
#include <limits.h>

using namespace std;

int min_distance(vector<int>& arr) {
    int min_val = INT_MAX;
    for (int i = 0; i < arr.size(); i++) {
        min_val = min(min_val, arr[i]);
    }

    int prev = -1;
    int min_dist = INT_MAX;
    for (int i = 0; i < arr.size(); i++) {
        if (arr[i] == min_val) {
            if (prev != -1) {
                min_dist = min(min_dist, i - prev);
            }
            prev = i;
        }
    }

    return min_dist;
}

int main() {
    int n;
    cin >> n;

    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }

    cout << min_distance(arr) << endl;

    return 0;
}
