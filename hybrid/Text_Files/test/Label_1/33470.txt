import java.util.Scanner;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();

        int[][] dp = new int[s.length() + 1][4];
        dp[0][0] = 1;
        dp[0][1] = 3;
        for (int i = 1; i <= s.length(); i++) {
            for (int j = 0; j < 4; j++) {
                dp[i][j] = dp[i - 1][j];
                if (s.charAt(i - 1) == '(') {
                    if (j == 0) dp[i][j] += 2;
                    else if (j == 1) dp[i][j] += 1;
                } else if (s.charAt(i - 1) == '[') {
                    if (j == 0) dp[i][j] += 2;
                    else if (j == 2) dp[i][j] += 1;
                } else if (s.charAt(i - 1) == ')') {
                    if (j == 1) dp[i][j] += 2;
                    else if (j == 0) dp[i][j] += 1;
                } else if (s.charAt(i - 1) == ']') {
                    if (j == 2) dp[i][j] += 2;
                    else if (j == 3) dp[i][j] += 1;
                }
            }
        }

        int max = 0;
        int[][] ind = new int[s.length() + 1][4];
        for (int i = 0; i < 4; i++) {
            if (dp[s.length()][i] > max) {
                max = dp[s.length()][i];
                ind[s.length()][i] = 1;
            }
        }
        for (int i = s.length() - 1; i >= 0; i--) {
            for (int j = 0; j < 4; j++) {
                if (dp[i][j] == max - 1 && ind[i + 1][j] == 1) {
                    ind[i][j] = 1;
                    max--;
                }
            }
        }

        StringBuilder sb = new StringBuilder();
        int i = 0;
        int j = 2;
        while (i < s.length()) {
            if (ind[i + 1][j] == 1) {
                sb.append(s.charAt(i));
                if (s.charAt(i) == '[' || s.charAt(i) == '(') j--;
                else if (s.charAt(i) == ']' || s.charAt(i) == ')') j++;
                i++;
            } else {
                i++;
            }
        }

        System.out.println((dp[s.length()][2] + 1) / 2);
        System.out.println(sb.toString());
    }
}
