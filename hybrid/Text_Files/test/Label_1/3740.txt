#include <algorithm>
#include <iostream>
using namespace std;
typedef long long ll;
const int MAXN = 2e5 + 5;
ll n, q, a[MAXN], sum[MAXN], ans;
int main() {
  cin >> n >> q;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    sum[i] = sum[i - 1] + a[i];
  }
  sort(a + 1, a + n + 1, greater<ll>());
  for (int i = 1; i <= q; i++) {
    ll l, r;
    cin >> l >> r;
    ans += sum[r] - sum[l - 1];
  }
  cout << ans << endl;
}
