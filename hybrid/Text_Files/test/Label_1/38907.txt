import sys
input = sys.stdin.readline

for _ in range(int(input())):
    t = input().strip()
    cur = []
    last = []
    vis = [False] * len(t)
    for i in range(len(t)):
        if t[i] == '0':
            cur.append('0')
            last.append('0')
        else:
            if last:
                cur.extend(last)
            cur.append('1')
            last.append('1')
        vis[i] = True
    for i in range(len(t)):
        if not vis[i]:
            cur.append('0')
        vis[i] = True
    for i in range(len(t)):
        if not vis[i]:
            cur[-1] = t[i]
        vis[i] = True
    print(''.join(cur))
