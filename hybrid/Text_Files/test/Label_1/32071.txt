import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static final int MOD = 998244353;
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] primes = new int[n];
        for (int i = 0; i < n; i++) {
            primes[i] = scanner.nextInt();
        }

        long[][] dp = new long[n + 1][2]; // dp[i, j] stores the number of good subsets of S that include the first i primes and have the last prime included or not (j = 0 or 1)
        dp[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= 1; j++) {
                dp[i][j] = (dp[i - 1][j] * ((j == 0) ? 1 : primes[i - 1]));
                dp[i][j] %= MOD;
                dp[i][j] += (dp[i - 1][j == 0 ? 1 : 0] * (j == 1 ? 1 : primes[i - 1]));
                dp[i][j] %= MOD;
            }
        }

        System.out.println(dp[n][0]);
    }
}
