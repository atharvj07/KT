#include<bits/stdc++.h>
using namespace std;

int main() {
    long long n, sum = 0, ans = 0, cur = 0;
    cin >> n;
    vector<long long> v(n);
    for (long long &x : v) {
        cin >> x;
        sum += x;
    }
    sort(v.begin(), v.end());
    sum /= n;
    for (int i = 0; i < n; i++) {
        ans += abs(sum - cur);
        cur += v[i];
    }
    cout << ans << endl;
    return 0;
}
