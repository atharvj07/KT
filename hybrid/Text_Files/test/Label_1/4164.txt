#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, m;
  cin >> n >> m;

  if (m == 0) {
    cout << "YES" << endl;
    return 0;
  }

  set<long long> dirty;
  for (long long i = 0; i < m; i++) {
    long long d;
    cin >> d;
    dirty.insert(d);
  }

  if (dirty.count(1) || dirty.count(n)) {
    cout << "NO" << endl;
    return 0;
  }

  long long cur = 1;
  while (cur + 2 <= n) {
    long long jump = -1;
    if (!dirty.count(cur + 1)) {
      jump = 1;
    } else if (!dirty.count(cur + 2)) {
      jump = 2;
    }

    if (jump == -1) {
      cout << "NO" << endl;
      return 0;
    }

    cur += jump;
  }

  cout << "YES" << endl;

  return 0;
}
