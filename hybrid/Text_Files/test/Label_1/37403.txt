k = int(input())
loops = [0, 1, 0, 0, 1, 0, 1, 0, 2, 1]

if k > 18 * 2:  # Maximum loops that can be achieved within the constraint of n <= 10^18
    print(-1)
else:
    result = ''
    while k > 0:
        for i in range(9, -1, -1):
            if loops[i] <= k:
                result += str(i)
                k -= loops[i]
                break
    print(result)
