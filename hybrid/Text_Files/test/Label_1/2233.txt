#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int r = 0, b = 0, g = 0;
  for (auto x : s) {
    r += (x == 'R');
    b += (x == 'B');
    g += (x == 'G');
  }
  vector<char> c;
  if (r % 2) c.push_back('R');
  if (b % 2) c.push_back('B');
  if (g % 2) c.push_back('G');
  sort(begin(c), end(c));
  for (auto x : c) { cout << x; }
  cout << endl;
  return 0;
}
