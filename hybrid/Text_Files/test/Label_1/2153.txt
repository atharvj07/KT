#include <bits/stdc++.h>
using namespace std;
int a[100010];
int main() {
  long long n, k;
  cin >> n >> k;
  for (int i = 0; i < 100010; i++) a[i] = -1;
  a[63] = a[62] = n / 2;
  n -= n / 2;
  int sum = 0;
  int pos = 0;
  for (int i = 61; i >= 0; i--) {
    if (sum + (1LL << i) <= n) sum += 1LL << i, a[i] = 1, pos = i;
    if (sum == n) break;
  }
  if (sum != n) puts("No");
  else {
    puts("Yes");
    for (int i = 63; i >= 0; i--) cout << a[i] << " ";
  }
  return 0;
}
