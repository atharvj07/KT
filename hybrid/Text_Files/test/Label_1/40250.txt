n, k = map(int, input().split())
time = 0
correct_chain_length = 0

for _ in range(k):
    chain = list(map(int, input().split()))[1:]  # Skip the length of the chain

    if chain[0] == 1:
        # Find the length of the correct starting chain
        for i in range(len(chain)):
            if i + 1 < len(chain) and chain[i] + 1 == chain[i + 1]:
                correct_chain_length += 1
            else:
                break
        # Add operations for disassembling the rest of the chain, if it's not in perfect order
        time += len(chain) - 1 - correct_chain_length
    else:
        # Entire chain needs to be disassembled
        time += len(chain) - 1

# Calculate final assembly time
time += n - 1 - correct_chain_length

print(time)
