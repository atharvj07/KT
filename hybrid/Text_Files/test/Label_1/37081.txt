def min_unattractiveness(n, a, b, h):
    dp = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    dp[0][0] = 0

    for i in range(1, n + 1):
        for j in range(i + 1):
            if a >= h[i - 1]:
                dp[i][j] = min(dp[i][j], dp[i - 1][j] + (h[i - 1] if j < i - 1 else 0))
            if b >= h[i - 1]:
                dp[i][j] = min(dp[i][j], dp[i - 1][j - 1] + (h[i - 1] if j > 0 else 0))
            a -= h[i - 1] * (j == i)
            b -= h[i - 1] * (j == 0)

    ans = min(dp[n])
    return -1 if ans == float('inf') else ans

n = int(input())
a, b = map(int, input().split())
h = list(map(int, input().split()))

print(min_unattractiveness(n, a, b, h))
