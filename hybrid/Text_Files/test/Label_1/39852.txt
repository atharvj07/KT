for t in range(int(input())):
    n,a,b = map(int,input().split())
    if (n/2)%2: print(-1)
    else:
        p = [a]
        for i in range(n,a,-1):
            if i!=b: p.append(i)
        p.append(b)
        for i in range(n//2-1,0,-1):
            if i!=a and i!=b: p.append(i)
        print(*p)
