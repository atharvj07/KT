import sys

inp=sys.stdin.readlines()
n=int(inp[0])
x=list(map(int,inp[1].split()))
x.sort()
ans=[x[0]]
lst=[x[0]]
for i in range(1,n):
    if x[i]==lst[-1]:
        continue
    if lst and (x[i]-lst[-1])&((x[i]-lst[-1])-1)==0:
        lst.append(x[i])
    else:
        lst=[x[i]]
    if len(lst)>len(ans):
        ans=lst
print(len(ans))
print(*ans)
