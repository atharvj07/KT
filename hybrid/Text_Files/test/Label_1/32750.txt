import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int k = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        Arrays.sort(p);
        int sum = 0;
        for (int i = n - 1; i >= 0 && k > 0; i--) {
            sum += p[i];
            k--;
            i -= m - 1;
        }
        System.out.println(sum);
    }
}
