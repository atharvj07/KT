import java.util.Scanner;

public class Main {
    static int mod = 1000000007;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        in.close();
        long res = 1L * n * (n + 1) % mod * m % mod * (m + 1) % mod * 2 % mod;
        System.out.println(res);
    }
}
