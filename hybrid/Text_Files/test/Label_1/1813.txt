#include <cstdio>
#define u64 long long

u64 sum(u64 n) {
    u64 ans = n;
    for (u64 l = 2, r; l <= n; l = r + 1) {
        r = n / (n / l);
        ans += (n / l) * (r - l + 1) / 2;
    }
    return ans;
}

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        u64 n;
        scanf("%lld", &n);
        printf("%lld\n", sum(n));
    }
    return 0;
}
