#include <bits/stdc++.h>
using namespace std;
int n, v[1010];
long long dp[1010][10][2];
long long f(int pos, int sum, int lead) {
  if (pos == n) {
    if (sum % 90 == 0) return 0;
    else return -1e9;
  }
  long long &ans = dp[pos][sum][lead];
  if (ans != -1) return ans;
  ans = f(pos + 1, sum + v[pos], lead);
  if (lead == 0 && v[pos] == 0) ans = max(ans, f(pos + 1, 0, 1));
  if (!lead) ans = max(ans, f(pos + 1, sum, 1) * 10 + v[pos]);
  return ans;
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> v[i];
  memset(dp, -1, sizeof dp);
  cout << f(0, 0, 0) << endl;
}
