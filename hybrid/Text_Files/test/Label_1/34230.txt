import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long a = sc.nextInt();
            long b = sc.nextInt();
            long n = sc.nextInt();
            if (n == 0) {
                System.out.println(a);
            } else if (n == 1) {
                System.out.println(b);
            } else {
                for (long i = 2; i < n; i++) {
                    a ^= b;
                    b ^= a;
                }
                System.out.println(a ^ b);
            }
        }
    }
}
