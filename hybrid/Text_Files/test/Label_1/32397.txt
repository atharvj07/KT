import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        int t = in.nextInt();
        int[] q = new int[t];
        for (int i = 0; i < t; i++) {
            q[i] = in.nextInt() - 1;
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = p[i];
        }
        for (int i = 0; i < t; i++) {
            x[q[i]] = -p[q[i]];
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (x[i] != 0) {
                for (int j = i + 1; j < n; j++) {
                    if (x[j] == x[i]) {
                        x[j] = 0;
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (x[i] != 0) {
                ok = false;
            }
        }
        if (ok) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                System.out.print(x[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }
}
