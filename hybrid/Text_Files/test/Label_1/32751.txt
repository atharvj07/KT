import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        char[] chars = scanner.nextLine().toCharArray();
        char[] ans = new char[chars.length * 2];
        int ansIdx = 0;
        int bal = 0;
        for (char c : chars) {
            if (c == '[') {
                bal++;
                if (bal > ansIdx) {
                    ansIdx = bal;
                    ans = Arrays.copyOf(chars, chars.length + ansIdx);
                }
            } else {
                bal--;
            }
        }
        System.out.println(ansIdx);
        System.out.println(new String(Arrays.copyOf(ans, chars.length + ansIdx)));
    }
}
