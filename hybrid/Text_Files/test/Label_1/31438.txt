import java.util.*;

public class Main {
    static int n, k;
    static double[] minR, maxR, minC, maxC;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        minR = new double[n + 1];
        minC = new double[n + 1];
        maxR = new double[n + 1];
        maxC = new double[n + 1];
        for (int i = 1; i <= n; i++) {
            double x = sc.nextInt(), y = sc.nextInt();
            minR[i] = minR[i - 1];
            minC[i] = minC[i - 1];
            maxR[i] = maxR[i - 1];
            maxC[i] = maxC[i - 1];
            minR[i] = Math.min(minR[i], x);
            minC[i] = Math.min(minC[i], y);
            maxR[i] = Math.max(maxR[i], x);
            maxC[i] = Math.max(maxC[i], y);
        }
        int[] ans = new int[k];
        for (int i = 1; i <= k; i++) {
            double midR = (minR[n] + maxR[n]) / 2;
            double midC = (minC[n] + maxC[n]) / 2;
            int pt = -1;
            double best = Double.MAX_VALUE;
            for (int j = 1; j <= n; j++) {
                double dist = Math.hypot(midR - minR[j], midC - minC[j]) + Math.hypot(maxR[j] - midR, maxC[j] - midC);
                if (dist < best) {
                    best = dist;
                    pt = j;
                }
            }
            maxR[pt] = minR[pt] = maxC[pt] = minC[pt] = -1e10;
            ans[i - 1] = pt;
        }
        for (int u : ans) {
            System.out.print(u + " ");
        }
    }
}
