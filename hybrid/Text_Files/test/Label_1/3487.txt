#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (n == 2)
        cout << char(3 - (s[0] - 'B') - (s[1] - 'B') + 'B');
    else {
        int b = 0, r = 0, g = 0;
        for (char c : s) {
            if (c == 'B')
                b++;
            else if (c == 'R')
                r++;
            else
                g++;
        }
        cout << string((b + 1) / 2, 'B') + string(r % 2 + (g + 1) / 2, 'R');
    }
    return 0;
}
