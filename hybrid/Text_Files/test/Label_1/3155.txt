#include <bits/stdc++.h>
using namespace std;

long long p[5001], dp[5001][5001];
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) { cin >> p[i]; }
  for (int j = 1; j <= m; j++) {
    for (int i = j; i <= n; i++) { dp[i][j] = dp[i - 1][j]; }
  }
  for (int j = 1; j <= m; j++) {
    for (int i = j + 1; i <= n; i++) {
      dp[i][j] = max(dp[i][j], dp[i - 1][j - 1] + p[i]);
    }
  }
  for (int i = 2; i <= k; i++) {
    for (int j = n; j >= 1; j--) {
      for (int l = j + m; l <= n; l++) {
        dp[l][j] = max(dp[l][j], dp[l][j + 1] + dp[j + m - 1][j]);
      }
    }
  }
  cout << dp[n][1] << endl;
  return 0;
}
