#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,t;
    cin>>n>>t;
    string s1,s2;
    cin>>s1>>s2;
    if(n==1){
        if(t==0){
            cout<<s1;
        }
        else{
            cout<<-1;
        }
    }
    else{
        for(int i=0;i<n;i++){
            if(s1[i]==s2[i]) cout<<s1[i];
            else{
                if(t>1){
                    if(s1[i]==s2[i]){
                        if(s1[i]=='z') cout<<'a';
                        else
                            cout<<s1[i]+1;
                        t--;
                    }
                    else{
                        cout<<s1[i];
                    }
                }
                else{
                    if(s1[i]!=s2[i]){
                        if(s1[i]!='z') cout<<s1[i]+1;
                        else cout<<'a';
                        t--;
                    }
                    else
                        cout<<s1[i];
                }
            }
        }
    }
    cout<<endl;
    return 0;
}
