import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int ans = 0;
        int cnt = 0;
        for (char c : s.toCharArray()) {
            if (c == 'A' || c == 'C' || c == 'G' || c == 'T') {
                cnt++;
            } else {
                cnt = 0;
            }
            ans = Math.max(ans, cnt);
        }
        System.out.println(ans);
    }
}
