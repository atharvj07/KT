#include<bits/stdc++.h>
using namespace std;

int main() {
    int n,i,x,ans,curr=0;
    cin>>n;
    vector<int>a(n);
    for(i=0; i<n; i++) {
        cin>>a[i];
    }
    while(true) {
        ans=++curr;
        x=a[ans-1];
        a[ans-1]=0;
        curr=curr%n;
        x--;
        if(x>0) {
            a[curr]=x;
        }
        if(x==0) {
            break;
        }
    }
    cout<<ans<<endl;
    return 0;
}
