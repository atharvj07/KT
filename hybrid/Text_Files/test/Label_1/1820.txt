#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=998244353;
int main(){
    ll n,k;cin>>n>>k;
    ll pref[n+2]={0},suff[n+2]={0},a[n];
    for(ll i=0;i<n;i++){cin>>a[i];if(a[i]!=-1)pref[i+1]=1;}
    for(ll i=n-1;i>=0;i--){suff[i]=suff[i+1];if(a[i]!=-1)suff[i]=1;}
    for(ll i=0;i<n;i++){
        pref[i+1]+=pref[i];pref[i+1]%=mod;
        suff[i]=suff[i+1];suff[i]%=mod;
    }
    ll ans=0;
    for(ll i=0;i<n;i++){
        if(a[i]!=-1)continue;
        ans=(ans+pref[i]*suff[i+1])%mod;
    }
    cout<<ans%mod;
    return 0;
}
