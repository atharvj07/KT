import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] grid = new int[n][m];
        int[][] dp = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                grid[i][j] = scanner.nextInt();
                dp[i][j] = grid[i][j];
            }
        }

        for (int i = 1; i < n; i++) {
            dp[i][0] += dp[i - 1][0];
        }
        for (int j = 1; j < m; j++) {
            dp[0][j] += dp[0][j - 1];
        }

        for (int i = 1; i < n; i++) {
            for (int j = 1; j < m; j++) {
                dp[i][j] = dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1] + grid[i][j];
            }
        }

        int max = 0;
        for (int i = 0; i + 2 < n; i++) {
            for (int j = 0; j + 2 < m; j++) {
                int sum = dp[i + 2][j + 2] - dp[i][j + 2] - dp[i + 2][j] + dp[i][j];
                max = Math.max(max, sum);
            }
        }

        System.out.println(max);
    }
}
