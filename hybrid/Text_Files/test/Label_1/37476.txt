n = int(input())
s = list(input())

count = {'0': s.count('0'), '1': s.count('1'), '2': s.count('2')}
avg = n // 3

# Replace excess characters from left to right
for i in range(3):
    char = str(i)
    for j in range(n):
        if count[char] > avg and s[j] != char:
            if count[s[j]] < avg:
                count[char] -= 1
                count[s[j]] += 1
                s[j] = char

# Replace excess characters from right to left
for i in range(2, -1, -1):
    char = str(i)
    for j in range(n - 1, -1, -1):
        if count[char] > avg and s[j] != char:
            if count[s[j]] < avg:
                count[char] -= 1
                count[s[j]] += 1
                s[j] = char

print(''.join(s))
