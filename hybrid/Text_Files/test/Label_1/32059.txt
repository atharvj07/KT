import java.util.Scanner;

public class SnitchCatcher {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] points = new int[n + 1][3];
        for (int i = 0; i <= n; i++) {
            points[i][0] = scanner.nextInt();
            points[i][1] = scanner.nextInt();
            points[i][2] = scanner.nextInt();
        }
        int vp = scanner.nextInt();
        int vs = scanner.nextInt();
        int Px = scanner.nextInt();
        int Py = scanner.nextInt();
        int Pz = scanner.nextInt();

        double time = 0;
        int currentIndex = 0;
        while (currentIndex < n) {
            double distanceToNextPoint = distance(points[currentIndex], points[currentIndex + 1]);
            double timeToNextPoint = distanceToNextPoint / vs;
            double distanceToHarry = distance(points[currentIndex], new int[]{Px, Py, Pz});
            double timeToHarry = distanceToHarry / vp;
            if (timeToHarry <= timeToNextPoint) {
                // Harry can catch the snitch before it reaches the next point
                System.out.println("YES");
                System.out.println(time + timeToHarry);
                System.out.println(points[currentIndex][0] + (points[currentIndex + 1][0] - points[currentIndex][0]) * timeToHarry / timeToNextPoint);
                System.out.println(points[currentIndex][1] + (points[currentIndex + 1][1] - points[currentIndex][1]) * timeToHarry / timeToNextPoint);
                System.out.println(points[currentIndex][2] + (points[currentIndex + 1][2] - points[currentIndex][2]) * timeToHarry / timeToNextPoint);
                return;
            } else {
                // Harry cannot catch the snitch before it reaches the next point
                time += timeToNextPoint;
                currentIndex++;
            }
        }

        // Harry cannot catch the snitch during its moving along the polyline
        System.out.println("NO");
    }

    private static double distance(int[] point1, int[] point2) {
        return Math.sqrt(Math.pow(point2[0] - point1[0], 2) + Math.pow(point2[1] - point1[1], 2) + Math.pow(point2[2] - point1[2], 2));
    }
}
