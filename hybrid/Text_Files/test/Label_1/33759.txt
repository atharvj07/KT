import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int m = sc.nextInt();
        int n = sc.nextInt();

        Map<Integer, Integer> table = new HashMap<>();
        int dummyCalls = 0;
        for (int i = 0; i < n; i++) {
            String operation = sc.next();
            if (operation.charAt(0) == '+') {
                int id = sc.nextInt();
                int hash = sc.nextInt();
                dummyCalls += add(table, id, hash, m, h);
            } else {
                int id = sc.nextInt();
                table.remove(id);
            }
        }

        System.out.println(dummyCalls);
    }

    private static int add(Map<Integer, Integer> table, int id, int hash, int m, int h) {
        int dummyCalls = 0;
        int cell = hash;
        while (table.containsKey(cell)) {
            dummyCalls++;
            cell = (cell + m) % h;
        }
        table.put(cell, id);
        return dummyCalls;
    }
}
