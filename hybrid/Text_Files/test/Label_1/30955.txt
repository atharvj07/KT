import java.util.*;
import java.io.*;
import java.lang.*;

public class Main {
    static int n, k, l;
    static int[] x;
    static int[] a;
    static List<int[]> ans;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        k = scanner.nextInt();
        l = scanner.nextInt();

        x = new int[k];
        for (int i = 0; i < k; i++) {
            x[i] = scanner.nextInt();
        }

        a = new int[l];
        for (int i = 0; i < l; i++) {
            a[i] = scanner.nextInt();
        }

        ans = new ArrayList<>();
        Arrays.sort(x);
        Arrays.sort(a);

        int j = 0;
        for (int i = 0; i < k; i++) {
            int[] tmp = new int[2];
            while (j < l && x[i] > a[j]) {
                j++;
            }

            if (j == l || x[i] != a[j]) {
                System.out.println(-1);
                return;
            }

            int start = j;
            while (j + 1 < l && x[i] == a[j + 1]) {
                j++;
            }

            tmp[0] = start;
            tmp[1] = j;
            ans.add(tmp);
            j++;
        }

        int sz = ans.size();
        int ops = 0;
        for (int[] t : ans) {
            int op = t[1] - t[0] + 1;
            if (op % 2 == 1) {
                ops++;
            }
        }

        System.out.println(ops);
    }
}
