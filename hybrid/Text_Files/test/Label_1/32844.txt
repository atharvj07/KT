import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        // Find the largest element in the array.
        int max = a[n - 1];

        // Create a map to store the exponents of 2 in the prairie partitions of the elements.
        Map<Integer, Integer> exponents = new HashMap<>();

        // Find the exponents of 2 in the prairie partitions of the elements.
        for (int i = 0; i < n; i++) {
            int exponent = 0;
            while (a[i] % 2 == 0) {
                a[i] /= 2;
                exponent++;
            }
            exponents.put(a[i], exponent);
        }

        // Find the number of distinct elements in the array.
        Set<Integer> distinctElements = new HashSet<>();
        for (int i = 0; i < n; i++) {
            distinctElements.add(a[i]);
        }

        // Find the greatest common divisor of the exponents of 2 in the prairie partitions of the elements.
        int gcd = 0;
        for (int exponent : exponents.values()) {
            gcd = gcd(gcd, exponent);
        }

        // Find the possible number of elements in the original sequence.
        List<Integer> possibleNumbers = new ArrayList<>();
        for (int i = 1; i <= max; i++) {
            if (gcd % i == 0 && distinctElements.contains(i)) {
                possibleNumbers.add(i);
            }
        }

        // Sort the possible number of elements in the original sequence.
        Collections.sort(possibleNumbers);

        // Print the possible number of elements in the original sequence.
        if (possibleNumbers.isEmpty()) {
            System.out.println(-1);
        } else {
            for (int possibleNumber : possibleNumbers) {
                System.out.println(possibleNumber);
            }
        }
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
