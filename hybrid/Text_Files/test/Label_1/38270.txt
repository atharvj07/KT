from collections import Counter

def construct_palindromic_matrix(n, arr):
    count = Counter(arr)
    matrix = [[0]*n for _ in range(n)]
    ones = []
    twos = []
    for num, freq in count.items():
        if freq % 4 == 0:
            for _ in range(freq // 4):
                twos.append(num)
        elif freq % 2 == 0:
            twos.append(num)
            for _ in range(freq // 4):
                twos.append(num)
        elif freq % 2 == 1:
            ones.append(num)
            for _ in range(freq // 4):
                twos.append(num)
    
    if len(ones) > 1 or len(ones) == 1 and n % 2 == 0:
        return "NO"
    
    if n % 2 == 1 and len(ones) != 1:
        return "NO"
    
    if n % 2 == 1:
        matrix[n//2][n//2] = ones[0]
    
    i = 0
    j = 0
    
    for num in twos:
        if i < n // 2:
            matrix[i][j] = num
            matrix[n-i-1][j] = num
            matrix[i][n-j-1] = num
            matrix[n-i-1][n-j-1] = num
            j += 1
            if j >= n // 2:
                i += 1
                j = 0
    
    return "\n".join(["YES"] + [" ".join(map(str, row)) for row in matrix])

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(construct_palindromic_matrix(n, arr))
