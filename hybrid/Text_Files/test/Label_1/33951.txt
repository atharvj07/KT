import java.util.Scanner;

class Main {

    private static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        // dp[i] = the number of ways to form a string of length i
        // dp[i][0] = the number of ways to form a string of length i that ends with 'A'
        // dp[i][1] = the number of ways to form a string of length i that ends with 'B'
        // dp[i][2] = the number of ways to form a string of length i that ends with 'C'
        long[][] dp = new long[n + 1][3];

        // Base cases
        dp[0][0] = 1;
        dp[0][1] = 1;
        dp[0][2] = 1;

        for (int i = 1; i <= n; i++) {
            // If the string ends with 'A', it can be formed by adding 'B' or 'C' to the end of a string of length i-1
            dp[i][0] = (dp[i-1][1] + dp[i-1][2]) % MOD;

            // If the string ends with 'B', it can be formed by adding 'A' or 'C' to the end of a string of length i-1
            dp[i][1] = (dp[i-1][0] + dp[i-1][2]) % MOD;

            // If the string ends with 'C', it can be formed by adding 'A' or 'B' to the end of a string of length i-1
            dp[i][2] = (dp[i-1][0] + dp[i-1][1]) % MOD;
        }

        // The answer is the number of ways to form a string of length n
        System.out.println(dp[n][0]);
    }
}
