import sys

input = sys.stdin.readline

n, m = map(int, input().split())

G = [[] for i in range(m + 1)]

for i in range(m):
    a, b, c = map(int, input().split())
    G[c].append((a, b))

q = int(input())

for i in range(q):
    u, v = map(int, input().split())
    ans = 0
    
    for k in G:
        if (u in k) and (v in k):
            ans += 1
    print(ans)
