import java.util.Scanner;

public class MinInclusionSegment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] count = new int[100001];
        int distinct = 0, l = 0, r = 0, minLen = Integer.MAX_VALUE;

        for (r = 0; r < n; r++) {
            if (count[a[r]]++ == 0) {
                distinct++;
            }
            while (distinct > k) {
                if (--count[a[l]] == 0) {
                    distinct--;
                }
                l++;
            }
            if (distinct == k && r - l + 1 < minLen) {
                minLen = r - l + 1;
                l = l;
            }
        }

        if (minLen == Integer.MAX_VALUE) {
            System.out.println("-1 -1");
        } else {
            System.out.println(l + " " + (l + minLen - 1));
        }
    }
}
