import sys

t = int(input())
for _ in range(t):
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]
    ans = 'NO'
    for i in range(5):
        for j in range(i + 1, 5):
            f = True
            for k in range(n):
                f &= a[k][i] | a[k][j]
            if f:
                ans = 'YES'
                break
    print(ans)
