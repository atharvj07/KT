import java.util.Scanner;

public class JumpStyle {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            int n = scanner.nextInt();
            if (n == 0) {
                break;
            }

            int[][] grid = new int[n][n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    grid[i][j] = scanner.nextInt();
                }
            }

            int count = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    boolean[][] visited = new boolean[n][n];
                    if (dfs(grid, visited, i, j, i, j)) {
                        count++;
                    }
                }
            }

            System.out.println(count);
        }

        scanner.close();
    }

    private static boolean dfs(int[][] grid, boolean[][] visited, int x, int y, int startX, int startY) {
        if (visited[x][y]) {
            return true;
        }

        visited[x][y] = true;

        int nextX = grid[x][y] % grid.length;
        int nextY = grid[x][y] / grid.length;

        if (nextX == startX && nextY == startY) {
            return true;
        }

        return dfs(grid, visited, nextX, nextY, startX, startY);
    }
}
