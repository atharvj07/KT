from collections import deque, defaultdict

def solve(n, k, p, special_pairs):
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    for i in range(1, n + 1):
        if p[i - 1] != 0:
            graph[p[i - 1]].append(i)
            in_degree[i] += 1

    for x, y in special_pairs:
        graph[x].append(y)
        in_degree[y] += 1

    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)

    order = []
    while queue:
        if len(queue) > 1:
            return [0]
        node = queue.popleft()
        order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if len(order) != n:
        return [0]
    return order

# Reading input
n, k = map(int, input().split())
p = list(map(int, input().split()))
special_pairs = [tuple(map(int, input().split())) for _ in range(k)]

# Solving the problem
result = solve(n, k, p, special_pairs)

# Printing the output
print(*result)
