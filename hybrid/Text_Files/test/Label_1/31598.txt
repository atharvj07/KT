import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }

    int cnt = 0;
    int[] b = new int[n];
    boolean stable = false;
    while (!stable) {
      stable = true;
      for (int i = 1; i < n - 1; i++) {
        b[i] = (a[i - 1] + a[i] + a[i + 1]) / 2;
        if (b[i] != a[i]) {
          stable = false;
        }
      }
      if (!stable) {
        b[0] = a[0];
        b[n - 1] = a[n - 1];
        a = b;
        cnt++;
      }
    }

    if (stable) {
      System.out.println(cnt);
      for (int i = 0; i < n; i++) {
        System.out.print(b[i] + " ");
      }
      System.out.println();
    } else {
      System.out.println(1);
    }
  }
}
