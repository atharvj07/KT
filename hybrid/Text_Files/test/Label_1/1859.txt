#include <iostream>
#include <bits/stdc++.h>
using namespace std;

bool valid(string s) {
    while (s.length() > 0 && s.back() == '0')
        s.pop_back();
    if (s.length() == 0)
        return false;
    reverse(s.begin(), s.end());
    int cnt2 = 0, cnt3 = 0;
    for (int i = 0; i < s.length(); i++) {
        if (i % 2 == 0)
            cnt2 += (s[i] == '1');
        else
            cnt3 += (s[i] == '1');
    }
    cnt2 %= 4;
    cnt3 %= 4;
    return (cnt2 == 0 && cnt3 == 0);
}

int main() {
    string s;
    cin >> s;
    cout << (valid(s) ? "yes" : "no") << endl;
    return 0;
}
