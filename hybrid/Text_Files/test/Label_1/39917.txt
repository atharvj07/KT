from sys import stdin
input = stdin.readline

for i in range(int(input())):
    n, m, a, b = map(int, input().split())
    b -= 2
    b = max(0, b) if a < b else b
    s = list(map(int, input().split()))
    s.sort()
    
    ans = 0
    l = a - 1
    r = l + 1
    for t in s:
        if t <= b:
            ans += 1
        if b:
            b -= 1
        while l and s[l - 1] <= t:
            l -= 1
            r += 1
        if r == n: break
    print(ans)
