#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<bool> used(26, false);
  for (int i = 0; i < n; i++) {
    if (s[i] >= 'a' && s[i] <= 'z') {
      used[s[i] - 'a'] = true;
    } else {
      used[s[i] - 'A'] = true;
    }
  }
  bool isPangram = true;
  for (int i = 0; i < 26; i++) {
    if (!used[i]) {
      isPangram = false;
      break;
    }
  }
  if (isPangram) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
