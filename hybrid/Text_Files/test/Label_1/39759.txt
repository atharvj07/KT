for _ in range(int(input())):
    a, b, k = map(int, input().split())
    pa = list(map(int, input().split()))
    pb = list(map(int, input().split()))
    p = set()
    for i in range(k):
        p.add((pa[i], pb[i]))
    if len(p) < 4:
        print(0)
    else:
        print(len(p) * (len(p) - 1) // 2 - k)
