n, m = [int(x) for x in input().split()]
s = [input() for i in range(n)]
r = [['.'] * m for i in range(n)]
for i in range(n):
    for j in range(m):
        if s[i][j] == '*':
            t = [['.'] * m for i in range(n)]
            t[i][j] = 1
            q = [(i, j)]
            while q:
                x, y = q.pop(0)
                if x > 0 and t[x - 1][y] == '.':
                    t[x - 1][y] = t[x][y] + 1
                    q.append((x - 1, y))
                if x < n - 1 and t[x + 1][y] == '.':
                    t[x + 1][y] = t[x][y] + 1
                    q.append((x + 1, y))
                if y > 0 and t[x][y - 1] == '.':
                    t[x][y - 1] = t[x][y] + 1
                    q.append((x, y - 1))
                if y < m - 1 and t[x][y + 1] == '.':
                    t[x][y + 1] = t[x][y] + 1
                    q.append((x, y + 1))
            for k in range(n):
                for l in range(m):
                    if t[k][l] != '.':
                        r[k][l] = str(t[k][l] % 10)
print(''.join([''.join(x) for x in r]))
