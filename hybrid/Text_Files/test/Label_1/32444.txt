import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            long a = sc.nextLong();
            long max_gcd = 0;
            for (long b = 1; b < a; b++) {
                max_gcd = Math.max(max_gcd, gcd(a, a ^ b & a));
            }
            System.out.println(max_gcd);
        }
    }

    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
