import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static int[] arr;
    static int[] count;
    static int[] maxRight;
    static int[] maxLeft;

    static boolean check(int l, int r) {
        return (count[r] - (l > 0 ? count[l - 1] : 0)) % 2 == 0;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input = br.readLine().split(" ");
        int n = Integer.parseInt(input[0]);
        int q = Integer.parseInt(input[1]);
        arr = new int[n];
        count = new int[n];
        maxRight = new int[n];
        maxLeft = new int[n];
        String[] arrStr = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            arr[i] = Integer.parseInt(arrStr[i]);
            if (i > 0) {
                maxLeft[i] = arr[i] > arr[maxLeft[i - 1]] ? i : maxLeft[i - 1];
            } else {
                maxLeft[i] = i;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (i < n - 1) {
                maxRight[i] = arr[i] > arr[maxRight[i + 1]] ? i : maxRight[i + 1];
            } else {
                maxRight[i] = i;
            }
        }
        for (int i = 0; i < n; i++) {
            int right = maxRight[i];
            count[i] = count[right] + 1;
            if (i > 0) {
                count[i] -= count[i - 1];
            }
        }
        for (int i = 0; i < q; i++) {
            input = br.readLine().split(" ");
            int l = Integer.parseInt(input[0]) - 1;
            int r = Integer.parseInt(input[1]) - 1;
            int left = l + 1;
            if (left <= r && maxLeft[r] >= l) {
                while (left <= r && maxLeft[r] >= l) {
                    left = maxLeft[r] + 1;
                    r = maxRight[r];
                }
                if (check(l, r)) {
                    System.out.println("Shi");
                } else {
                    System.out.println("Fou");
                }
            } else {
                System.out.println("Fou");
            }
        }
    }
}
