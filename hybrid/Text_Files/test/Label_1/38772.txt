n, m, q = map(int, input().split())
grid = [[0] * m for i in range(n)]
ops = []
for i in range(q):
    t, *p = map(int, input().split())
    ops.append((t, p))
for t, p in ops:
    if t == 1:
        grid[p[0] - 1] = grid[p[0] - 1][1:] + grid[p[0] - 1][:1]
    elif t == 2:
        for j in range(n):
            grid[j][p[0] - 1] = grid[j][1:] + grid[j][:1]
    else:
        grid[p[0] - 1][p[1] - 1] = p[2]
for row in grid:
    print(' '.join(map(str, row)))
