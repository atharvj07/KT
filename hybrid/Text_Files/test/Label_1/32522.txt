import java.util.PriorityQueue;
import java.util.Scanner;
import java.util.Set;
import java.util.TreeSet;

class Main {

    static class Competitor implements Comparable<Competitor> {

        int s, r;
        double ratio;

        public Competitor(int s, int r) {
            this.s = s;
            this.r = r;
            this.ratio = (double) s / (double) r;
        }

        @Override
        public int compareTo(Competitor o) {
            return (int) ((ratio - o.ratio) * 100000);
        }

    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Set<Integer> possibleWinners = new TreeSet<>();
        PriorityQueue<Competitor> competitors = new PriorityQueue<>();
        for (int i = 1; i <= n; i++) {
            int s = scanner.nextInt();
            int r = scanner.nextInt();
            competitors.offer(new Competitor(s, r));
        }
        double ratio = -1;
        while (!competitors.isEmpty()) {
            Competitor competitor = competitors.poll();
            if (competitor.ratio != ratio) {
                ratio = competitor.ratio;
                possibleWinners.clear();
            }
            possibleWinners.add(competitor.s);
        }
        for (Integer winner : possibleWinners)
            System.out.print(winner + " ");
        System.out.println();
    }
}
