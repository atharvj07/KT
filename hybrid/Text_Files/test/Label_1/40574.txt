k, n = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Calculate possible initial scores
possible_initial_scores = set()
for i in range(k):
    possible_initial_scores.add(b[0] - sum(a[:i+1]))

# Verify each possible initial score
valid_initial_scores = set()
for initial_score in possible_initial_scores:
    current_scores = {initial_score}
    for points in a:
        current_scores = {score + points for score in current_scores}
        if not any(score in b for score in current_scores):
            break
    else:
        valid_initial_scores.add(initial_score)

print(len(valid_initial_scores))
