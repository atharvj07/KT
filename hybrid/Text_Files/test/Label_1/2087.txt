#include <bits/stdc++.h>

using namespace std;
#define ll long long

ll solve(vector<ll> &a, ll n) {
  int l = 0, r = 0, ans = 0, cnt = 0;
  map<ll, int> mp;
  while (r < n) {
    mp[a[r]]++;
    cnt += mp[a[r] - (a[r] & (-a[r]))];
    cnt -= mp[a[r]] > 1;
    if (!cnt)
      ans = max(ans, r - l + 1), ++r;
    else
      mp[a[l++]]--;
  }
  return n - ans;
}

int main() {
  ll n;
  cin >> n;
  vector<ll> a(n);
  for (ll i = 0; i < n; i++) cin >> a[i];
  cout << solve(a, n) << endl;
  return 0;
}
