import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong(), b = in.nextLong();
        if (a > b || (b - a) % 3 != 0) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            int ans = (int) ((b - a) / 3);
            System.out.println(ans + 1);
            for (int i = 0; i < ans; i++) {
                if (a % 2 == 0) {
                    a /= 2;
                } else {
                    a = a * 10 + 1;
                }
                System.out.print(a + " ");
            }
            System.out.println(b);
        }
    }
}
