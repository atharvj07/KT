import java.util.Arrays;
import java.util.Scanner;

public class InnovativeOlympiadInIrrigation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();
        int[] previousHosts = new int[n];
        for (int i = 0; i < n; i++) {
            previousHosts[i] = scanner.nextInt() - 1;
        }
        int[] hostCounts = new int[m];
        Arrays.fill(hostCounts, Integer.MAX_VALUE);
        for (int i = n - 1; i >= 0; i--) {
            hostCounts[previousHosts[i]] = i;
        }
        for (int i = 0; i < q; i++) {
            long year = scanner.nextLong();
            int index = (int) ((year - n - 1) % m);
            int minHostYear = Integer.MAX_VALUE;
            int minHostIndex = -1;
            for (int j = 0; j < m; j++) {
                int hostYear = hostCounts[(index + j) % m];
                if (hostYear < minHostYear) {
                    minHostYear = hostYear;
                    minHostIndex = (index + j) % m;
                }
            }
            System.out.println(minHostIndex + 1);
        }
    }
}
