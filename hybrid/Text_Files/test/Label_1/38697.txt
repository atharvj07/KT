from collections import defaultdict
from sys import stdin


# read input
n = int(stdin.readline())
queries = []
for _ in range(n):
    queries.append(stdin.readline().strip())

# process input
hostnames = defaultdict(list)
for query in queries:
    hostname, path = query[7:].split('/', 1)
    hostnames[hostname].append(path)

# find groups of hostnames
groups = []
for hostname, paths in hostnames.items():
    if len(paths) < 2:
        continue
    group = set([hostname])
    for other_hostname, other_paths in hostnames.items():
        if hostname != other_hostname and set(paths) == set(other_paths):
            group.add(other_hostname)
    if len(group) > 1:
        groups.append(group)

# print output
print(len(groups))
for group in groups:
    print(' '.join(sorted(group)))
