import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n, s;
        while (in.hasNext()) {
            n = in.nextInt();
            s = in.nextInt();
            System.out.println(count(n, s));
        }
    }

    static int count(int n, int s) {
        if (s < 0 || s > 9 * n) return 0;
        int[] dp = new int[s + 1];
        dp[0] = 1;
        for (int i = 1; i <= s; i++) {
            for (int j = 1; j <= Math.min(9, i); j++) {
                dp[i] += dp[i - j];
            }
        }
        return dp[s];
    }
}
