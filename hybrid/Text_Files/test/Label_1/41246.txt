def can_transform(s, t):
    vowels = 'aeiou'
    if len(s) != len(t):
        return "No"
    for i in range(len(s)):
        if (s[i] in vowels and t[i] not in vowels) or (s[i] not in vowels and t[i] in vowels):
            return "No"
    return "Yes"

s = input()
t = input()
print(can_transform(s, t))
