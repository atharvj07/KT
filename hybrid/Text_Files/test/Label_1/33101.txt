import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int c = sc.nextInt();
        int v0 = sc.nextInt();
        int v1 = sc.nextInt();
        int a = sc.nextInt();
        int l = sc.nextInt();
        if (v0 + l >= c)
            System.out.println(1);
        else {
            int v = v0;
            int d = 2;
            while (v + l < c) {
                v += a;
                if (v > v1)
                    v = v1;
                d++;
            }
            System.out.println(d);
        }
        sc.close();
    }
}
