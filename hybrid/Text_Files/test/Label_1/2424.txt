#include <vector>
#include <string>
#include <iostream>

int main() {
  std::string s;
  std::cin >> s;
  int ans = 1e9, cnt = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == 'A' || s[i] == 'E' || s[i] == 'I' || s[i] == 'O' || s[i] == 'U' || s[i] == 'Y') {
      cnt = 0;
    } else {
      cnt++;
    }
    ans = std::min(ans, cnt);
  }
  ans++;
  std::cout << ans << std::endl;
  return 0;
}
