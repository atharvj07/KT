n = int(input().strip())
arr = list(map(int, input().strip().split()))

sequences = []
used = [False] * n

while not all(used):
    sequence = []
    last_num = 0
    for i in range(n):
        if not used[i] and arr[i] > last_num:
            sequence.append(arr[i])
            used[i] = True
            last_num = arr[i]
    sequences.append(sequence)

for seq in sequences:
    print(" ".join(map(str, seq)))
