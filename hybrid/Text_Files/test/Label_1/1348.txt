#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    if (s.find("2020") != string::npos)
      cout << "YES" << endl;
    else if (s.find("202") != string::npos && s.find("20") != string::npos)
      cout << "YES" << endl;
    else if (s.find("20") != string::npos && s.find("2020") == string::npos)
      cout << "YES" << endl;
    else if (s.find("020") != string::npos && s.find("20") != string::npos)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
