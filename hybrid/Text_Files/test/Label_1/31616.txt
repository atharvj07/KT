import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] points = new int[n][2];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            points[i][0] = Integer.parseInt(line[0]);
            points[i][1] = Integer.parseInt(line[1]);
        }
        int[] result = solve(points, n);
        for (int i : result) {
            System.out.print(i + " ");
        }
    }

    private static int[] solve(int[][] points, int n) {
        int[] result = new int[3];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (isPositiveArea(points[i], points[j], points[k])) {
                        result[0] = i + 1;
                        result[1] = j + 1;
                        result[2] = k + 1;
                        return result;
                    }
                }
            }
        }
        return result;
    }

    private static boolean isPositiveArea(int[] a, int[] b, int[] c) {
        return (a[0] * (b[1] - c[1]) + b[0] * (c[1] - a[1]) + c[0] * (a[1] - b[1])) > 0;
    }
}
