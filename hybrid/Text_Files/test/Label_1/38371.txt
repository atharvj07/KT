n, bx = map(int, input().split())
x = tuple(map(int, input().split()))[::-1]
m, by = map(int, input().split())
y = tuple(map(int, input().split()))[::-1]

x_10 = 0
for i in range(n):
    x_10 += x[i] * (bx ** i)
y_10 = 0
for i in range(m):
    y_10 += y[i] * (by ** i)

if x_10 < y_10:
    print('<')
elif x_10 > y_10:
    print('>')
else:
    print('=')
