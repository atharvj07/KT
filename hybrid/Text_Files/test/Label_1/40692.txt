def solve():
    mod = 1000000007
    h, w = map(int, input().split())
    r = list(map(int, input().split()))
    c = list(map(int, input().split()))
    dp = [[0] * (w + 1) for i in range(h + 1)]
    dp[0][0] = 1
    for i in range(h):
        for j in range(w):
            dp[i + 1][j] = (dp[i + 1][j] + dp[i][j] * (max(r[i] - j, 0))) % mod
            dp[i][j + 1] = (dp[i][j + 1] + dp[i][j] * (max(c[j] - i, 0))) % mod
    return dp[h][w]
print(solve())
