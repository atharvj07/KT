#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
typedef long long ll;
struct Beacon { ll pos, pow; };
int n;
Beacon beacons[N];
ll ans;
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> beacons[i].pos >> beacons[i].pow; }
    sort(beacons, beacons + n,
         [](Beacon a, Beacon b) { return a.pos < b.pos; });
    ll maxR = -1;
    for (int i = n - 1; i >= 0; i--) {
        if (maxR < beacons[i].pos - beacons[i].pow) {
            maxR = beacons[i].pos + beacons[i].pow;
            ans++;
        }
    }
    cout << ans << "\n";
    return 0;
}
