#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, k, a[101] = {};
  cin >> n >> k;
  string s;
  cin >> s;
  for (int i = 0; i < n; ++i) {
    ++a[s[i]];
    if (a[s[i]] > 2) {
      cout << "NO" << endl;
      return 0;
    }
  }
  if (n <= k)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
