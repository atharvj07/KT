import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long s = scanner.nextLong();
        long x = scanner.nextLong();

        if ((s & x) != 0) {
            System.out.println(0);
        } else {
            long a = (s + x) / 2;
            long b = (s - x) / 2;

            if (a < 0 || b < 0) {
                System.out.println(0);
            } else if (a + b != s || (a ^ b) != x) {
                System.out.println(0);
            } else {
                System.out.println(1);
            }
        }
    }
}
