import java.util.Scanner;

public class StudyTimeChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            int targetTime = scanner.nextInt();
            if (targetTime == 0) {
                break;
            }

            int studyCount = scanner.nextInt();
            int totalStudyTime = 0;

            for (int i = 0; i < studyCount; i++) {
                int startTime = scanner.nextInt();
                int endTime = scanner.nextInt();
                totalStudyTime += endTime - startTime;
            }

            if (totalStudyTime >= targetTime) {
                System.out.println("OK");
            } else {
                int shortageTime = targetTime - totalStudyTime;
                System.out.println(shortageTime);
            }
        }

        scanner.close();
    }
}
