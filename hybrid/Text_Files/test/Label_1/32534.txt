import java.util.HashMap;
import java.util.Scanner;

class Main {

  private static final String[] NOTES = {"C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "B", "H"};
  private static final HashMap<String, Integer> NOTE_NUMBERS = new HashMap<>();

  static {
    for (int i = 0; i < NOTES.length; i++) NOTE_NUMBERS.put(NOTES[i], i);
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    String[] notes = scanner.nextLine().split(" ");
    int[] noteNumbers = new int[notes.length];
    for (int i = 0; i < notes.length; i++) noteNumbers[i] = NOTE_NUMBERS.get(notes[i]);
    int diff1 = noteNumbers[1] - noteNumbers[0];
    int diff2 = noteNumbers[2] - noteNumbers[1];
    if (diff1 == 4 && diff2 == 3) System.out.println("major");
    else if (diff1 == 3 && diff2 == 4) System.out.println("minor");
    else System.out.println("strange");
  }
}
