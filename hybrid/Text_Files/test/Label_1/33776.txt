import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n], g = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            g[i] = in.nextInt();
        }
        if (Math.abs(Arrays.stream(a).sum() - Arrays.stream(g).sum()) > 500) {
            System.out.println("-1");
        } else {
            int[] ans = new int[n];
            for (int i = 0; i < n; i++) {
                if (a[i] < g[i]) {
                    ans[i] = 0;
                } else {
                    ans[i] = 1;
                }
            }
            StringBuilder sb = new StringBuilder();
            for (int p : ans) {
                if (p == 0) {
                    sb.append('A');
                } else {
                    sb.append('G');
                }
            }
            System.out.println(sb.toString());
        }
    }
}
