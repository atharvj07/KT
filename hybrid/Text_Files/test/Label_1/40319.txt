n,k=map(int,input().split())
b=list(map(int,input().split()))
c=input()
o=[i for i in range(n) if c[i]=='O']
r=[i for i in range(n) if c[i]=='R']
w=[i for i in range(n) if c[i]=='W']
if k%2==0 and len(r)%2==1:print(-1);exit()
if k<=len(o):print(sum(sorted(b[i] for i in o)[-k:]));exit()
if k<=len(o)+len(r):print(sum(sorted(b[i] for i in o)[-k:]+sorted(b[i] for i in r)[-k+len(o):]));exit()
if k<=len(o)+len(r)+len(w):print(sum(sorted(b[i] for i in o)[-k:]+sorted(b[i] for i in r)[-k+len(o):]+sorted(b[i] for i in w)[:k-len(o)-len(r)]));exit()
print(-1)
