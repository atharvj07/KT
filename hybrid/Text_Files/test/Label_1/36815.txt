n, m = map(int, input().split())
a = list(map(int, input().split()))

total_turns = sum(abs(a[i + 1] - a[i]) for i in range(m - 1))

if m == 1:
    print(0)
else:
    savings = {}
    for i in range(m - 1):
        x, y = sorted([a[i], a[i + 1]])
        if x != y:
            savings[(x, y)] = savings.get((x, y), 0) + abs(x - y) - 1

    max_saving = max(savings.values(), default=0)
    print(total_turns - max_saving)
