#include <bits/stdc++.h>
using namespace std;
const int MAXN = 100;

int n, ans, dp[MAXN][5][5], val[MAXN];
vector<pair<int, int>> block[MAXN];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n;
  for (int i = 0; i < n; ++i) {
    int x, y, z;
    cin >> x >> y >> z;
    block[i] = {{x, y}, {z, y}};
  }
  memset(dp, -1, sizeof(dp));
  for (int i = n - 1; i >= 0; --i) {
    for (int cl = 0; cl < 5; ++cl) {
      for (int cr = 0; cr < 5; ++cr) {
        if (dp[i + 1][cl][cr] == -1) continue;
        ans = max(ans, val[i] + dp[i + 1][cl][cr]);
        for (int j = 0; j < 2; ++j) {
          int nl = block[i][j].first;
          int nr = block[i][j].second;
          if (!(cl == nl && cr == nr)) {
            dp[i][cl][nr] = max(dp[i][cl][nr], val[i] + dp[i + 1][cl][cr]);
            dp[i][nl][cr] = max(dp[i][nl][cr], val[i] + dp[i + 1][cl][cr]);
          }
        }
      }
    }
  }
  cout << ans << '\n';
}
