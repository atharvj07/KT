#include <vector>
#include <iostream>

template <typename T>
void Deduplicate(std::vector<T>& v)
{
    std::vector<T> rsv;
    std::vector<T> rsvx;
    std::vector<bool> isUsed(1000);

    for (int i = v.size() - 1; i >= 0; i--)
    {
        if (!isUsed[v[i]])
        {
            rsvx.push_back(v[i]);
            isUsed[v[i]] = true;
        }
    }

    for (int i = rsvx.size() - 1; i >= 0; i--)
    {
        rsv.push_back(rsvx[i]);
    }

    v.clear();

    for (int i = 0; i < rsv.size(); i++)
    {
        v.push_back(rsv[i]);
    }
}

int main()
{
    std::vector<int> v;
    int n;

    std::cin >> n;

    for (int i = 0; i < n; i++)
    {
        int tmp;

        std::cin >> tmp;
        v.push_back(tmp);
    }

    Deduplicate(v);

    std::cout << v.size() << std::endl;

    for (int i = 0; i < v.size(); i++)
    {
        std::cout << v[i] << " ";
    }

    return 0;
}
