import sys

n, p = map(int, sys.stdin.readline().split())
agree = [set() for _ in range(n+1)]
for _ in range(n):
    x, y = map(int, sys.stdin.readline().split())
    agree[x].add(y)
    agree[y].add(x)

ans = 0
for i in range(1, n+1):
    for j in agree[i]:
        if i != j and len(agree[i] | agree[j]) >= p:
            ans += 1
print(ans)
