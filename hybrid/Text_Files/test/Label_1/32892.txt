import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long k = sc.nextLong();
        long p = n * m / k;
        if (p * k != n * m)
            System.out.println("NO");
        else {
            System.out.println("YES");
            System.out.println(1 + " " + 0);
            System.out.println(2 + " " + (p - 1));
            System.out.println(n + " " + (p - 1));
        }
    }
}
