import java.util.*;

public class Main {

    static class Point {
        double x, y;
    }

    static double dist(Point a, Point b) {
        return Math.sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y));
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            points[i] = new Point();
            points[i].x = sc.nextDouble();
            points[i].y = sc.nextDouble();
        }
        Arrays.sort(points, (a, b) -> {
            if (a.x == b.x) {
                return Double.compare(a.y, b.y);
            }
            return Double.compare(a.x, b.x);
        });
        double[][] dp = new double[n][k + 1];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], Integer.MAX_VALUE);
        }
        dp[0][0] = 0;
        for (int i = 1; i < n; i++) {
            dp[i][0] = dp[i - 1][0] + dist(points[i - 1], points[i]);
            for (int j = 1; j <= k; j++) {
                for (int l = 0; l < i; l++) {
                    dp[i][j] = Math.min(dp[i][j], dp[l][j - 1] + dist(points[l], points[i]));
                }
            }
        }
        int[] ans = new int[k];
        int i = n - 1, j = k;
        while (i >= 0 && j > 0) {
            boolean found = false;
            for (int l = 0; l < i; l++) {
                if (dp[i][j] == dp[l][j - 1] + dist(points[l], points[i])) {
                    ans[j - 1] = i + 1;
                    i = l;
                    j--;
                    found = true;
                    break;
                }
            }
            if (!found) {
                i--;
            }
        }
        for (int a : ans) {
            System.out.print(a + " ");
        }
        System.out.println();
    }
}
