from collections import defaultdict

def dfs(node, parent, graph, zero_edge_passed):
    count_0, count_1 = 0, 1  # count_1 includes the node itself
    for child, edge_type in graph[node]:
        if child == parent:
            continue
        if edge_type == 0:
            if not zero_edge_passed:
                c0, c1 = dfs(child, node, graph, True)
                count_0 += c0 + c1
            # If a 0-edge has already been passed, don't count paths through another 0-edge
        else:
            c0, c1 = dfs(child, node, graph, zero_edge_passed)
            count_0 += c0
            count_1 += c1
    return count_0, count_1

def count_valid_pairs(n, edges):
    graph = defaultdict(list)
    for x, y, c in edges:
        graph[x].append((y, c))
        graph[y].append((x, c))
    
    count_0, count_1 = dfs(1, -1, graph, False)
    
    # Total pairs = pairs within 1-edges + pairs with 0-edges before any 1-edge
    total_pairs = count_1 * (count_1 - 1) // 2 + count_0 * (count_0 + count_1)
    return total_pairs

n = int(input().strip())
edges = [list(map(int, input().strip().split())) for _ in range(n-1)]
print(count_valid_pairs(n, edges))
