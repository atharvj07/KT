n, L, a = map(int, input().split())
t = [0] * n
l = [0] * n
for i in range(n):
    t[i], l[i] = map(int, input().split())
i = 0
ans = 0
while i < n:
    if t[i] >= L:
        break
    if l[i] == a:
        i += 1
        ans += 1
    else:
        if t[i] + a <= L:
            temp = (l[i] - 1) // a
            ans += temp
            i += temp
            t[i + 1] = t[i] + a + 1
        i += 1
print(ans)
