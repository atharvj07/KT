import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        String a = in.next();
        in.close();

        StringBuilder b = new StringBuilder();
        b.append(a.charAt(0));
        int l = 1;
        for (int i = 1; i < n; i++) {
            int d = a.charAt(i) - '0';
            if ((l < k) && (d != 0)) {
                if (l == 0) {
                    b.deleteCharAt(0);
                }
                b.append(d);
                l++;
            } else if (d != 0) {
                b.append(d);
            }
        }
        while (l < k) {
            b.append('0');
            l++;
        }
        while (b.length() < n) {
            b.insert(0, '0');
        }
        System.out.println(b);
    }
}
