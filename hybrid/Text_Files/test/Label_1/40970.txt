def find_permutation(n, l, r, s):
    # Calculate the minimum and maximum possible sum for the given range
    min_sum = sum(range(1, r - l + 2))
    max_sum = sum(range(n - r + l, n + 1))
    
    if not (min_sum <= s <= max_sum):
        return -1  # It's impossible to find a permutation with the given sum
    
    # Initialize an array to store the permutation
    permutation = [0] * n
    # Fill the range from l to r with the smallest possible values initially
    current_sum = 0
    for i in range(l - 1, r):
        permutation[i] = i - l + 2
        current_sum += i - l + 2
    
    # Adjust the values in the range [l, r] to reach the target sum s
    i = r - 1
    while current_sum < s and i >= l - 1:
        diff = min(n - permutation[i], s - current_sum)
        current_sum += diff
        permutation[i] += diff
        i -= 1
    
    # If the current_sum is still not equal to s, it's impossible
    if current_sum != s:
        return -1
    
    # Fill in the remaining numbers
    remaining_numbers = set(range(1, n + 1)) - set(permutation)
    j = 0
    for i in range(n):
        if permutation[i] == 0:
            permutation[i] = remaining_numbers.pop()
    
    return permutation

t = int(input())
for _ in range(t):
    n, l, r, s = map(int, input().split())
    result = find_permutation(n, l, r, s)
    if result == -1:
        print(-1)
    else:
        print(" ".join(map(str, result)))
