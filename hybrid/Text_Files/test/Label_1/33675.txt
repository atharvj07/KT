import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h1 = sc.nextInt(), h2 = sc.nextInt(), n = sc.nextInt();
        int[] v = new int[n], a = new int[n], b = new int[n], h = new int[n];
        char[] c = new char[n];
        for (int i = 0; i < n; i++) {
            v[i] = sc.nextInt();
            String line = sc.next();
            c[i] = line.charAt(0);
            h[i] = c[i] == 'F' ? h1 : h2;
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        long ans = dfs(a, b, h, c, v, h1, h2, 0, 0, 0);
        System.out.println(ans);
    }
    static long dfs(int[] a, int[] b, int[] h, char[] c, int[] v, int h1, int h2, int x1, int y1, int score) {
        if (x1 == 105 && y1 == h1) return score;
        long ans = 0;
        for (int i = 0; i < a.length; i++) {
            if (y1 == h[i] && a[i] <= x1 && x1 <= b[i]) {
                ans = Math.max(ans, dfs(a, b, h, c, v, h1, h2, b[i], h[i] == h1 ? h2 : h1, score + v[i]));
            }
        }
        if (y1 == h1) ans = Math.max(ans, dfs(a, b, h, c, v, h1, h2, x1 + 1, h1, score));
        if (y1 == h2) ans = Math.max(ans, dfs(a, b, h, c, v, h1, h2, x1 + 1, h2, score));
        return ans;
    }
}
