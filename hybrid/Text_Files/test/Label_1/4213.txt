#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = 0;
    int cur = 0;
    for (int i = 0; i < n; i++) {
        cur += a[i];
        if (cur >= (i + 1) * 100) {
            ans++;
        }
    }
    cout << ans << endl;
}
