#include <bits/stdc++.h>

using namespace std;

const int N = 1e5 + 10;

int n, k;
int a[N], b[N], fr[N];

int main() {
    cin >> n >> k;
    for(int i = 1; i <= n; ++i) cin >> a[i];
    for(int i = 1; i <= n; ++i) cin >> b[i];
    for(int i = 1; i <= n; ++i) ++fr[a[i]];
    int ans = 0, id; bool flag = false;
    for(int i = 1; i <= k; ++i) {
        if(fr[i] == 0) {
            ans += b[1];
            flag = true;
            id = a[1];
            a[1] = i;
            break;
        }
    }
    if(!flag) {
        cout << 0 << endl;
        return 0;
    }
    --fr[id];
    for(int i = 2; i <= n; ++i) {
        if(a[i] != id) continue;
        if(fr[a[i]] == 0) {
            ans += b[i];
            a[i] = id;
            --fr[id];
        }
    }
    cout << ans << endl;
    return 0;
}
