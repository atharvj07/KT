#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;
const ll INF = 1e18;
const int N = 2e5 + 10;

ll x[N], y[N];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n;
    cin >> n;
    vector<pair<ll, ll>> p(n);
    for (int i = 0; i < n; i++) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        p[i] = {a, c};
        x[i * 2] = a;
        x[i * 2 + 1] = c;
        y[i * 2] = b;
        y[i * 2 + 1] = d;
    }
    sort(p.begin(), p.end());
    sort(x, x + 2 * n);
    sort(y, y + 2 * n);
    vector<int> cntx(2 * n), cnty(2 * n);
    for (int i = 0; i < n; i++) {
        int idx1 = lower_bound(x, x + 2 * n, p[i].first) - x;
        int idx2 = lower_bound(x, x + 2 * n, p[i].second) - x;
        cntx[idx1]++;
        cntx[idx2]--;
    }
    for (int i = 1; i < 2 * n; i++) {
        cntx[i] += cntx[i - 1];
    }
    ll ansx = -1;
    for (int i = 0; i < 2 * n; i++) {
        if (cntx[i] == n - 1) {
            ansx = x[i];
            break;
        }
    }
    for (int i = 0; i < n; i++) {
        int idx1 = lower_bound(y, y + 2 * n, p[i].first) - y;
        int idx2 = lower_bound(y, y + 2 * n, p[i].second) - y;
        cnty[idx1]++;
        cnty[idx2]--;
    }
    for (int i = 1; i < 2 * n; i++) {
        cnty[i] += cnty[i - 1];
    }
    ll ansy = -1;
    for (int i = 0; i < 2 * n; i++) {
        if (cnty[i] == n - 1) {
            ansy = y[i];
            break;
        }
    }
    cout << ansx << ' ' << ansy << '\n';
}
