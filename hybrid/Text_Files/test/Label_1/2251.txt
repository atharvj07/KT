#include <cstdio>
#include <algorithm>
using namespace std;
struct ca {char c,a;} a[101];
int main() {
  int n;
  scanf("%d",&n);
  for(int i=0; i<n; i++) scanf(" %c%c",&a[i].c,&a[i].a);
  sort(a,a+n,[](ca c, ca d){return c.c<d.c||c.c==d.c&&c.a<d.a;});
  int ans=0;
  for(int i=0,cnt=0,prv=-1; i<n; cnt=0,++i) {
    if(a[i].c==prv) cnt++;
    else prv=a[i].c, ans++;
    if(cnt>=1) ans--;
  }
  sort(a,a+n,[](ca c, ca d){return c.a<d.a||c.a==d.a&&c.c<d.c;});
  for(int i=0,cnt=0,prv=-1; i<n; cnt=0,++i) {
    if(a[i].a==prv) cnt++;
    else prv=a[i].a, ans++;
    if(cnt>=1) ans--;
  }
  printf("%d\n",ans);
  return 0;
}
