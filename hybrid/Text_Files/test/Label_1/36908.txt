import sys
input=sys.stdin.readline
write=sys.stdout.write
n,m=map(int,input().split())
w=[int(x) for x in input().split()]
f=[[]for x in range(n+1)]
for x in range(m):
  a,b=map(int,input().split())
  f[a].append(b)
  f[b].append(a)
vis=[0 for x in range(m+1)]
o=[]
def dfs(x):
  if vis[x]!=0: return
  vis[x]=1
  for y in f[x]:
    dfs(y)
  o.insert(0,x)
for x in range(1,n+1):
  if vis[x]==0:
    dfs(x)
if len(o)==m:
  print("ALIVE")
  write(' '.join([str(x) for x in o]))
else: print("DEAD")
