import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = sc.nextInt();
      if (i > 0) {
        if (arr[i] - arr[i - 1] > 15) {
          System.out.println(arr[i - 1]);
          return;
        }
      }
    }
    System.out.println(90);
  }
}
