import java.util.*;
import java.io.*;

class Main {

    static final int MAXN = 1000000;
    static char[] g1 = new char[MAXN], g2 = new char[MAXN];

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine());
        String s = br.readLine();
        g1 = s.toCharArray();
        s = br.readLine();
        g2 = s.toCharArray();

        int x1 = 0, y1 = 0, x2 = 0, y2 = 0;
        for (int i = 0; i < n - 1; ++i) {
            switch (g1[i]) {
                case 'N':
                    y1++;
                    break;
                case 'S':
                    y1--;
                    break;
                case 'E':
                    x1++;
                    break;
                case 'W':
                    x1--;
            }
            switch (g2[i]) {
                case 'N':
                    y2++;
                    break;
                case 'S':
                    y2--;
                    break;
                case 'E':
                    x2++;
                    break;
                case 'W':
                    x2--;
            }
            if (x1 != x2 || y1 != y2) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
