#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define N 1005
string ans;
bool vis[N][N];
char s[N];
int main() {
	int n;
	cin >> n;
	bool ok = 0;
	for(int i = 0; i < N; i++) {
		for(int j = 1; j < N; j++) {
			if(i * j > n) break;
			if(i * j == n) {
				memset(vis, 0, sizeof vis);
				memset(s, 0, sizeof s);
				bool ok2 = 1;
				for(int k = 0; k < i; k++) {
					for(int l = 0; l < j; l++) {
						if(k == 0) {
							s[l] = l + 97;
							vis[k][l] = 1;
						} else if(l == 0) {
							int pos = -1;
							for(int a = 97; a <= 122; a++) {
								if(!vis[k - 1][l]) {
									s[l] = a;
									pos = a;
									break;
								}
							}
							if(pos == -1) {
								ok2 = 0;
							}
							vis[k][l] = 1;
						} else {
							int pos = -1;
							for(int a = 97; a <= 122; a++) {
								if(!vis[k][l - 1] && !vis[k - 1][l]) {
									s[l] = a;
									pos = a;
									break;
								}
							}
							if(pos == -1) {
								ok2 = 0;
							}
							vis[k][l] = 1;
						}
					}
				}
				for(int k = 0; k < i; k++) {
					int cnt = 0;
					for(int l = 0; l < j; l++) {
						if(s[l] != 'a' && s[l] != 'e' && s[l] != 'i' && s[l] != 'o' && s[l] != 'u') {
							cnt++;
						}
					}
					if(cnt > 0) {
						ok2 = 0;
					}
				}
				for(int k = 0; k < j; k++) {
					int cnt = 0;
					for(int l = 0; l < i; l++) {
						if(s[k] != 'a' && s[k] != 'e' && s[k] != 'i' && s[k] != 'o' && s[k] != 'u') {
							cnt++;
						}
					}
					if(cnt > 0) {
						ok2 = 0;
					}
				}
				if(ok2) {
					for(int k = 0; k < i; k++) {
						for(int l = 0; l < j; l++) {
							ans += s[l];
						}
					}
					ok = 1;
					break;
				}
			}
		}
	}
	if(ok) {
		cout << ans << "\n";
	} else {
		cout << "-1\n";
	}
	return 0;
}
