from collections import defaultdict
a = list(map(int, input().split()))
s = input()
fre = defaultdict(int)
pref = [0] * len(s)
for i in range(len(s)):
    pref[i] = a[ord(s[i]) - ord('a')]
    if i: pref[i] += pref[i - 1]
ans = 0
for i in range(len(s)):
    for j in range(i + 1, len(s)):
        if s[i] == s[j]:
            if (i == 0 or j == len(s) - 1) or (pref[j - 1] - (pref[i - 1] if i > 0 else 0)) == 0:
                ans += 1
print(ans)
