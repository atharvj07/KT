import java.util.*;
import java.io.*;

public class Main {
    static class Elem implements Comparable<Elem> {
        int xor, id;

        Elem(int xor, int id) {
            this.xor = xor;
            this.id = id;
        }

        @Override
        public int compareTo(Elem e) {
            return Integer.compare(xor, e.xor);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] a = new int[n];
        Elem[] b = new Elem[n];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
            b[i] = new Elem(a[i], i);
        }
        Arrays.sort(b);
        int[] ans = new int[1 << 20];
        int lim = 0;
        for (int i = 1, l = 0, r = n - 1; i < (1 << 20); l++, r--) {
            if (l < n && r >= 0 && b[l].xor == b[r].xor) {
                ans[i] = b[l].id;
                ans[i ^ b[l].xor] = b[r].id;
                lim = i;
            } else
                break;
        }
        int x = 0;
        while ((1 << x) <= lim)
            x++;
        x--;
        System.out.println(x);
        for (int i = 0; i < (1 << x); i++)
            System.out.print(ans[i] + " ");
        System.out.println();
    }
}
