import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class SonyaAndMultiset {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Integer> multiset = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String operation = scanner.next();
            if (operation.equals("+")) {
                multiset.add(scanner.nextInt());
            } else if (operation.equals("-")) {
                multiset.remove(Integer.valueOf(scanner.nextInt()));
            } else if (operation.equals("?")) {
                String pattern = scanner.next();
                int count = 0;
                for (int num : multiset) {
                    if (matchesPattern(num, pattern)) {
                        count++;
                    }
                }
                System.out.println(count);
            }
        }
    }

    private static boolean matchesPattern(int num, String pattern) {
        String binaryString = Integer.toBinaryString(num);
        int patternLength = pattern.length();
        int binaryStringLength = binaryString.length();
        if (binaryStringLength < patternLength) {
            binaryString = String.format("%0" + (patternLength - binaryStringLength) + "d", 0) + binaryString;
        }
        for (int i = 0; i < patternLength; i++) {
            if (pattern.charAt(i) == '1' && binaryString.charAt(i) != '1') {
                return false;
            }
        }
        return true;
    }
}
