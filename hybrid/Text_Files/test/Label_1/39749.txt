from collections import deque
def solve():
    n,m=map(int,input().split())
    graph=[[] for i in range(n+1)]
    indeg=[0]*(n+1)
    for i in range(m):
        a,b=map(int,input().split())
        graph[a].append(b)
        indeg[b]+=1
    q=deque([i for i in range(1,n+1) if indeg[i]==0])
    ans=[]
    while q:
        node=q.popleft()
        ans.append(node)
        for adj in graph[node]:
            indeg[adj]-=1
            if indeg[adj]==0:
                q.append(adj)
    if len(ans)==n:
        print(' '.join(map(str,ans)))
    else:
        print(-1)
solve()
