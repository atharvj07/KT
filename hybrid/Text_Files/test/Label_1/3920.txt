#include <bits/stdc++.h>
using namespace std;
int main() {
    int q;
    cin >> q;
    while (q--) {
        int n;
        cin >> n;
        vector<int> p(n);
        for (int &x : p) { cin >> x; }
        int c1 = 0, c2 = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] == (i + 1) % n + 1) { c1++; }
            if (p[i] == (i - 1 + n) % n + 1) { c2++; }
        }
        cout << (c1 == n || c2 == n ? "YES" : "NO") << endl;
    }
    return 0;
}
