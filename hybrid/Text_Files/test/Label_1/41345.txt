def find_possibilities(n, m, d, affected_settlements, paths):
    graph = [[] for _ in range(n + 1)]
    for a, b in paths:
        graph[a].append(b)
        graph[b].append(a)
    visited = [False] * (n + 1)
    distance = [0] * (n + 1)
    queue = [affected_settlements[0]]
    visited[affected_settlements[0]] = True
    while queue:
        current = queue.pop(0)
        for neighbor in graph[current]:
            if not visited[neighbor] and distance[current] < d:
                distance[neighbor] = distance[current] + 1
                if distance[neighbor] <= d:
                    queue.append(neighbor)
                    visited[neighbor] = True
    possibilities = 0
    for i in range(1, n + 1):
        if visited[i] and i not in affected_settlements:
            possibilities += 1
    return possibilities

n, m, d = map(int, input().split())
affected_settlements = list(map(int, input().split()))
paths = [list(map(int, input().split())) for _ in range(n - 1)]
result = find_possibilities(n, m, d, affected_settlements, paths)
print(result)
