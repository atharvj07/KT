import java.util.HashSet;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String num = in.next();
        HashSet<String> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (fingerMovements(num.substring(0, i + 1)).equals(fingerMovements(num.substring(j, j + 1)))
                        || fingerMovements(num.substring(0, i + 1)).equals(fingerMovements(num.substring(j, j + 1))))
                    set.add(num.substring(0, i + 1) + num.substring(j, j + 1));
            }
        }
        if (set.size() > 0)
            System.out.println("NO");
        else
            System.out.println("YES");
    }

    public static String fingerMovements(String num) {
        StringBuilder sb = new StringBuilder();
        for (char c : num.toCharArray()) {
            if (c == '0')
                sb.append("000");
            else if (c == '1')
                sb.append("1001");
            else if (c == '2')
                sb.append("2001");
            else if (c == '3')
                sb.append("3000");
            else if (c == '4')
                sb.append("4000");
            else if (c == '5')
                sb.append("5000");
            else if (c == '6')
                sb.append("6000");
            else if (c == '7')
                sb.append("7000");
            else if (c == '8')
                sb.append("8000");
            else
                sb.append("9000");
        }
        return sb.toString();
    }
}
