import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int r = input.nextInt();
        int n = input.nextInt();
        boolean[][] reclaimed = new boolean[r + 1][3];
        for (int i = 0; i < n; i++) {
            int ri = input.nextInt();
            int ci = input.nextInt();
            reclaimed[ri][ci] = true;
            reclaimed[ri][3 - ci] = true;
        }
        boolean win = false;
        for (int i = r; i >= 1; i--) {
            if (!reclaimed[i][1] || !reclaimed[i][2]) {
                win = !win;
            }
        }
        System.out.println(win ? "WIN" : "LOSE");
    }
}
