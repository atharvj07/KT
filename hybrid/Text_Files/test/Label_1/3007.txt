#include <bits/stdc++.h>
using namespace std;
using ll = unsigned long long;
ll const mod = 998244353;
struct trio {
  int a, b, c;
  trio() : a(0), b(0), c(0) {}
  trio(int x, int y, int z) : a(x), b(y), c(z) {}
  trio operator+(const trio &rhs) const {
    return trio((a + rhs.a) % mod, (b + rhs.b) % mod, (c + rhs.c) % mod);
  }
  trio operator%(const int &rhs) const { return trio(a % rhs, b % rhs, c % rhs); }
};
ll power(ll x, ll y) {
  ll ret = 1;
  while (y) {
    if (y & 1) ret = ret * x % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ret;
}
int main() {
  int n;
  cin >> n;
  trio ans;
  for (int i = 0; i < n; ++i) {
    char a, b;
    cin >> a >> b;
    a == b ? ++ans.c : a == '?' ? ++ans.b : ++ans.a;
  }
  cout << ((ans.c % mod) + (ans.b % mod) * 2 % mod) * (ans.b % mod) % mod * power(2, ans.b % mod) % mod
       << '\n';
  return 0;
}
