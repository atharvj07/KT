import java.util.*;

class Main {
    public static long dist[][];
    public static boolean vis[];

    public static long dist(long d[][], int u, int v) {
        LinkedList<Integer> q = new LinkedList<Integer>();
        q.add(u);
        vis[u] = true;
        while (!q.isEmpty()) {
            u = q.remove(0);
            for (int v1 = 0; v1 < d.length; v1++) {
                if (vis[v1] == false && d[u][v1] > 0) {
                    if (v1 == v) return d[u][v];
                    vis[v1] = true;
                    q.add(v1);
                }
            }
        }
        return -1;
    }

    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        dist = new long[n][n];
        vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dist[i][j] = sc.nextLong();
            }
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            long c = sc.nextLong();
            long oldDist = dist(dist, u, v);
            long newDist = Math.min(oldDist, c);
            if (oldDist != newDist) {
                for (int j = 0; j < n; j++) {
                    dist[j][u] = dist[u][j] = Math.min(dist[j][u], newDist + dist[j][v]);
                }
            }
            System.out.println(sum(dist));
        }
    }

    public static long sum(long d[][]) {
        long sum = 0;
        for (int i = 0; i < d.length; i++) {
            for (int j = i + 1; j < d.length; j++) {
                sum += d[i][j];
            }
        }
        return sum;
    }
}
