MOD = 10**9 + 7

def solve(n, k, weights):
    fifty = weights.count(50)
    hundred = n - fifty
    dp = [[-1] * (hundred + 1) for _ in range(fifty + 1)]
    dp[0][0] = [0, 1]

    for i in range(fifty + 1):
        for j in range(hundred + 1):
            if dp[i][j] == -1:
                continue
            for x in range(2):
                for y in range(2):
                    if x + 2*y <= k // 50 and i >= x and j >= y and (x + y > 0):
                        ni, nj = i - x + (1 - x), j - y + (1 - y)
                        if dp[ni][nj] == -1:
                            dp[ni][nj] = [dp[i][j][0] + 1, 0]
                        if dp[ni][nj][0] == dp[i][j][0] + 1:
                            dp[ni][nj][1] += dp[i][j][1]
                            dp[ni][nj][1] %= MOD

    if dp[fifty][hundred] == -1:
        return "-1\n0"
    else:
        return f"{dp[fifty][hundred][0]}\n{dp[fifty][hundred][1]}"

n, k = map(int, input().split())
weights = list(map(int, input().split()))
print(solve(n, k, weights))
