def can_transform(S, T, queries):
    def encode(s):
        count = [0, 0, 0]  # Count of A, B, C
        for char in s:
            count[ord(char) - ord('A')] += 1
        # Transform count to (count of B and C is odd, sum of all counts, count of A is odd)
        return (count[1] % 2, sum(count), count[0] % 2)

    S_encoded = encode(S)
    T_encoded = encode(T)
    
    results = []
    for a, b, c, d in queries:
        # Check if transformation is possible based on encoded values
        if S_encoded == T_encoded:
            results.append('1')
        else:
            s_bc_odd, s_sum, s_a_odd = S_encoded
            t_bc_odd, t_sum, t_a_odd = T_encoded
            if s_bc_odd == t_bc_odd and ((s_sum % 2 == t_sum % 2) or s_a_odd):
                results.append('1')
            else:
                results.append('0')
    return ''.join(results)

# Reading input
S = input().strip()
T = input().strip()
Q = int(input().strip())
queries = [tuple(map(int, input().strip().split())) for _ in range(Q)]

# Processing queries
result = can_transform(S, T, queries)
print(result)
