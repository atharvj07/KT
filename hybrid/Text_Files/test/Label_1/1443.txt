#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pi;
#define F first
#define S second
const int MM = 2e3 + 5;
int t[MM][MM], dp[MM][MM], p[MM], n;
string v;
int main() {
    cin >> v;
    n = v.size();
    for (int i = n - 1; i >= 0; i--) {
        for (int j = i; j < n; j++) {
            if (v[i] <= v[j])
                dp[i][j] = dp[i + 1][j] + 1;
            else
                dp[i][j] = dp[i + 1][j];
            t[i][j] = v[i] < v[j] ? i : t[i + 1][j];
            t[i][j] = j < t[i][j] ? j : t[i][j];
        }
    }
    for (int i = 0; i <= n; i++) p[i] = -1;
    for (int i = n - 1; i >= 0; i--) {
        if (t[i][n - 1] == i) {
            v[i] = '0', p[i] = n;
            break;
        }
    }
    int mx = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (dp[i][n - 1] == dp[i + 1][n - 1] && t[i][n - 1] == i)
            mx = i;
    }
    for (int j = mx; j < n; j++) {
        p[t[j][n - 1]] = j;
        v[t[j][n - 1]] = '0';
    }
    cout << v << "\n";
    return 0;
}
