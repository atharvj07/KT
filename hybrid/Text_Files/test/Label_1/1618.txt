#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long int n, k;
        cin >> n >> k;
        long long int equal = n / k, extra = n % k;
        if ((k & 1 || equal < 2) && extra <= k / 2)
            cout << n - n % (k + 1) << endl;
        else
            cout << n / 2 << endl;
    }
}
