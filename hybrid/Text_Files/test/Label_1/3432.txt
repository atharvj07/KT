#include <cstdio>
#include <vector>

int main() {
    int n;
    scanf("%d", &n);
    std::vector<int> p(n);
    for (int i = 0; i < n; ++i) {
        scanf("%d", &p[i]);
        --p[i];
    }
    for (int i = 0; i < n; ++i) {
        while (p[p[i]] != p[i]) {
            std::swap(p[i], p[p[i]]);
        }
        ++p[i];
    }
    for (int i = 0; i < n; ++i) {
        printf("%d ", p[i]);
    }
    return 0;
}
