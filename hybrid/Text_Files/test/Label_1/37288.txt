import sys

input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    adj = [[] for _ in range(n+1)]
    deg = [0] * (n+1)
    for _ in range(n-1):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
    
    # Mark the leaves and buds
    leaves = []
    buds = []
    for i in range(2, n+1):
        if deg[i] == 1:
            leaves.append(i)
        elif deg[i] > 2 and all(deg[j] == 1 for j in adj[i]):
            buds.append(i)
    
    # Process the buds
    while buds:
        bud = buds.pop()
        for leaf in adj[bud]:
            deg[leaf] = 0
            leaves.remove(leaf)
        deg[bud] = 0
        leaves.remove(bud)
        for v in adj[bud]:
            deg[v] -= 1
            if deg[v] == 1:
                leaves.append(v)
            elif deg[v] > 2 and all(deg[j] == 1 for j in adj[v]):
                buds.append(v)
    
    print(len(leaves))
