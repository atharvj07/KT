#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> books(n);
  for (int i = 0; i < n; i++) {
    cin >> books[i];
  }
  vector<int> count(m + 1);
  for (int book : books) {
    count[book]++;
  }
  long long ans = 0;
  for (int i = 1; i <= m; i++) {
    for (int j = i + 1; j <= m; j++) {
      ans += (long long)count[i] * count[j];
    }
  }
  cout << ans << endl;
  return 0;
}
