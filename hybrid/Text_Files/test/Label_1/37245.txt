import sys
input = sys.stdin.readline

def solve():
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    if d == 0:
        return max(a)
    b = sorted(filter(lambda x:x>0, a))
    l = len(b)
    if l == 0:
        return 0
    if l == 1:
        return b[0]
    if l % 2 == 0:
        ans = sum(b) + min(b) * (d % 2)
        return ans
    mx = b.pop()
    ans = sum(b) + min(b) * (d % 2) + mx * (d % 2 == 0)
    return ans

for _ in range(int(input())):
    print(solve())
