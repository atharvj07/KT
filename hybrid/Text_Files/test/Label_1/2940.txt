#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

const int MAX = 1e5+10;
long long int dp[MAX];
vector<int> pos[MAX];
int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int x;
        cin >> x;
        pos[x].push_back(i);
    }
    for (int i = 1; i <= n; i++) {
        if (pos[i].size() > 0) dp[i] = 0;
        else dp[i] = INT64_MAX;
    }
    for (int i = 2; i <= n; i++) {
        long long int dis = 0, next = dp[i-1];
        for (int j = 0; j < pos[i].size(); j++) {
            dis = dis + pos[i][j] - j;
            next = min(next, dp[i-1] + dis);
        }
        dp[i] = next;
    }
    cout << dp[n] << endl;
    return 0;
}
