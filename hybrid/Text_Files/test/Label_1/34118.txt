import java.util.*;

public class Main {

    private static class Edge {
        public int to;
        public long weight;

        public Edge(int to, long weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        long xor = 0;
        List<List<Edge>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int from = scanner.nextInt() - 1;
            int to = scanner.nextInt() - 1;
            long weight = scanner.nextInt();
            graph.get(from).add(new Edge(to, weight));
            graph.get(to).add(new Edge(from, weight));
            xor ^= weight;
        }
        if (xor != 0) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (visited[i]) {
                continue;
            }
            long sum = 0;
            dfs(i, -1, sum, graph, visited, ans);
        }
        System.out.println(ans);
    }

    private static void dfs(int node, int parent, long sum, List<List<Edge>> graph, boolean[] visited, long ans) {
        visited[node] = true;
        for (Edge edge : graph.get(node)) {
            if (edge.to == parent) {
                continue;
            }
            if (!visited[edge.to]) {
                dfs(edge.to, node, sum ^ edge.weight, graph, visited, ans);
            } else {
                ans = Math.min(ans, (sum ^ edge.weight));
            }
        }
    }
}
