#include <bits/stdc++.h>
#define endl "\n"

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
      cin >> v[i];
    }
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int l = i + 1, r = n - 1;
      while (l < r) {
        if (v[r] - v[i] <= 2) {
          ans += r - l;
          l++;
        } else {
          r--;
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
