import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static final long MOD = 1000000007L;
    static long[] fact;

    public static void main(String[] args) {
        fact = new long[1001];
        fact[0] = 1;
        for (int i = 1; i <= 1000; i++) {
            fact[i] = fact[i - 1] * i % MOD;
        }

        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] r = new int[h + 1];
        int[] c = new int[w + 1];
        for (int i = 1; i <= h; i++) {
            r[i] = sc.nextInt();
        }
        for (int i = 1; i <= w; i++) {
            c[i] = sc.nextInt();
        }

        long[][] dp = new long[h + 1][w + 1];
        dp[h][w] = 1;
        for (int i = h; i >= 1; i--) {
            for (int j = w; j >= 1; j--) {
                if (r[i] >= j && c[j] >= i) {
                    dp[i][j] = (dp[i + 1][j] + dp[i][j + 1]) % MOD;
                }
            }
        }

        for (int i = h - 1; i >= 1; i--) {
            for (int j = w - 1; j >= 1; j--) {
                if (r[i] >= j && c[j] >= i) {
                    dp[i][j] = (dp[i + 1][j] + dp[i][j + 1] - dp[i + 1][j + 1] + MOD) % MOD;
                }
            }
        }

        long ans = dp[1][1];
        for (int i = 1; i <= h; i++) {
            ans = ans * fact[r[i]] % MOD;
        }
        for (int i = 1; i <= w; i++) {
            ans = ans * fact[c[i]] % MOD;
        }

        System.out.println(ans);
    }
}
