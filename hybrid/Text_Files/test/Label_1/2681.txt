#include <iostream>
#include <vector>
#include <algorithm>
#define int long long
using namespace std;
const int mod = 1e9 + 7;

int32_t main() {
  int n, m; cin >> n >> m;
  vector<int> a(n); for (auto& x : a) cin >> x;
  if (n == 2) { cout << (a[1] - a[0]) * m << endl; return 0; }
  if (n == 3) {
    vector<int> b(m); for (auto& x : b) cin >> x; sort(b.begin(), b.end());
    cout << 2 * (a.back() - *b.begin()) << endl;
    return 0;
  }
  vector<int> b(m); for (auto& x : b) cin >> x;
  for (int i = 0; i < m; i += b[i] - (i ? b[i - 1] : 0)) {
    if (i + m - 1 - (i ? b[i - 1] : 0) < n) {
      cout << (a[i + m - 1 - (i ? b[i - 1] : 0)] - (i ? a[i - 1] : 0)) << endl;
      return 0;
    }
  }
  cout << -1 << endl;
}
