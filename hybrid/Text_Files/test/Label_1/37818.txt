import sys

n = int(input())
a = list(map(int, input().split()))
k = int(input())
b = list(map(int, input().split()))

if frozenset(a) != frozenset(b):
    print("NO")
    sys.exit()

ans = []
a.reverse()
b.reverse()

while a:
    if a[-1] == b[-1]:
        a.pop()
        b.pop()
    elif a[-1] < b[-1]:
        ans.append(len(a) + " R")
        b[-1] -= a[-1]
        a.pop()
    else:
        ans.append(len(a) + " L")
        b[-1] -= a[-1]
        a[-1] = b[-1]

ans.reverse()

print("YES")
print('\n'.join(ans))
