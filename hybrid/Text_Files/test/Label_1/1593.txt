#include <iostream>
#include <vector>

using namespace std;

int solve(int n, int x, int y, vector<int> &v) {
  int cnt = 0;
  for (int i = x - 1, j = y - 1; i >= 0; i--, j--) {
    if (v[i] != (j >= 0 ? v[j] : 0)) cnt++;
  }
  return cnt;
}

int main() {
  int n, x, y;
  cin >> n >> x >> y;
  vector<int> v(n);
  cin >> v[0];
  for (int i = 1; i < n; i++) {
    char c;
    cin >> c;
    v[i] = c - '0';
  }
  cout << solve(n, x, y, v) << "\n";
  return 0;
}
