#include <iostream>
#include <vector>
using namespace std;

int main() {
  unsigned int n, l, x, y, p;
  cin >> n >> l >> x >> y;
  vector<unsigned int> a(n + 2);
  a[0] = 0;
  a[n + 1] = l;
  for (unsigned int i = 1; i < n + 1; i++) cin >> a[i];
  bool xFound = false;
  bool yFound = false;
  vector<unsigned int> ans;
  for (unsigned int i = 0; i < n + 1; i++) {
    if (a[i + 1] - a[i] == x) {
      xFound = true;
    }
    if (a[i + 1] - a[i] == y) {
      yFound = true;
    }
  }
  if (xFound && yFound)
    cout << "0" << endl;
  else if (xFound) {
    cout << "1" << endl;
    cout << y << endl;
  } else if (yFound) {
    cout << "1" << endl;
    cout << x << endl;
  } else {
    ans.push_back(x);
    p = y - x;
    while (p <= l) {
      ans.push_back(p);
      p += y;
    }
    cout << ans.size() << endl;
    for (unsigned int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
    cout << endl;
  }
  return 0;
}
