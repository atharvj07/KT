import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[] firstFly = new int[3];
        int[] secondFly = new int[3];
        for (int i = 0; i < 3; i++) {
            firstFly[i] = scanner.nextInt();
            secondFly[i] = scanner.nextInt();
        }
        int sum = 0;
        for (int i = 0; i < 3; i++) {
            sum += Math.abs(firstFly[i] - secondFly[i]);
        }
        System.out.println((sum == 1 || sum == 2) ? "YES" : "NO");
    }
}
