import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class TreePath {

    static class Node {
        int value;
        List<Node> children = new ArrayList<>();

        public Node(int value) {
            this.value = value;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());

        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }

        for (int i = 0; i < n - 1; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int u = Integer.parseInt(tokenizer.nextToken());
            int v = Integer.parseInt(tokenizer.nextToken());
            nodes[u].children.add(nodes[v]);
            nodes[v].children.add(nodes[u]);
        }

        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int k = Integer.parseInt(tokenizer.nextToken());
            int[] vertices = new int[k];
            for (int j = 0; j < k; j++) {
                vertices[j] = Integer.parseInt(tokenizer.nextToken());
            }
            System.out.println(solve(nodes[1], vertices) ? "YES" : "NO");
        }
    }

    static boolean solve(Node root, int[] vertices) {
        boolean[] visited = new boolean[vertices.length];
        return dfs(root, vertices, visited, 0);
    }

    static boolean dfs(Node node, int[] vertices, boolean[] visited, int index) {
        if (index == vertices.length) {
            return true;
        }

        if (node.value == vertices[index]) {
            visited[index] = true;
            return dfs(node, vertices, visited, index + 1);
        }

        for (Node child : node.children) {
            if (dfs(child, vertices, visited, index)) {
                return true;
            }
        }

        for (int i = 0; i < vertices.length; i++) {
            if (!visited[i] && Math.abs(node.value - vertices[i]) == 1) {
                visited[i] = true;
                if (dfs(node, vertices, visited, index + 1)) {
                    return true;
                }
                visited[i] = false;
            }
        }

        return false;
    }
}
