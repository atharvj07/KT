#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int P = 1e9 + 7;

int dp[100001][17];
int a[100001];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  dp[n + 1][1] = 1;
  for (int i = n; i >= 1; i--) {
    for (int j = 1; j <= 16; j++) {
      dp[i][j] = dp[i + 1][j];
      if (a[i] % j == 0) (dp[i][j] += dp[i + 1][j / (a[i] / j)]) %= P;
    }
  }
  cout << dp[1][1] << '\n';
  return 0;
}
