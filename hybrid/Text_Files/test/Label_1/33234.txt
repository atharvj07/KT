import java.util.Scanner;

public class DominoCarpet {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        // Read the carpet layout
        char[][] carpet = new char[n][m];
        for (int i = 0; i < n; i++) {
            carpet[i] = scanner.nextLine().toCharArray();
        }

        // Calculate the number of ways to assemble the carpet
        int[][][] dp = new int[n][m][2]; // dp[i][j][k] = number of ways to assemble the carpet up to (i, j) with the last domino placed vertically if k = 0, horizontally if k = 1
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (carpet[i][j] == '.') {
                    continue;
                }

                // If this is the first domino, place it vertically or horizontally
                if (i == 0 && j == 0) {
                    dp[i][j][0] = 1;
                    dp[i][j][1] = 1;
                } else {
                    // If the previous domino was placed vertically
                    if (i > 0 && carpet[i - 1][j] == 'O') {
                        dp[i][j][0] = (dp[i - 1][j][0] + dp[i - 1][j][1]) % MOD;
                    }

                    // If the previous domino was placed horizontally
                    if (j > 0 && carpet[i][j - 1] == 'O') {
                        dp[i][j][1] = (dp[i][j - 1][0] + dp[i][j - 1][1]) % MOD;
                    }

                    // If this domino is placed vertically
                    if (i > 0 && carpet[i - 1][j] == '.') {
                        dp[i][j][0] = (dp[i][j][0] + dp[i - 1][j][0]) % MOD;
                    }

                    // If this domino is placed horizontally
                    if (j > 0 && carpet[i][j - 1] == '.') {
                        dp[i][j][1] = (dp[i][j][1] + dp[i][j - 1][1]) % MOD;
                    }
                }
            }
        }

        // Print the result
        System.out.println((dp[n - 1][m - 1][0] + dp[n - 1][m - 1][1]) % MOD);
    }
}
