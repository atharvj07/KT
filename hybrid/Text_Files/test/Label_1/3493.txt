#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    int a[2 * n], b[2 * n];
    for (int i = 0; i < 2 * n; i++) {
        a[i] = i + 1;
        b[i] = i + 1;
    }
    for (int i = 1; i < n; i++) {
        swap(a[i], a[i + n - 1]);
    }
    int l = 0, r = 2 * n - 1;
    while (k > 0) {
        if (b[l] < b[r]) {
            swap(b[l], b[r]);
            l++;
            k--;
        } else {
            swap(b[r - 1], b[r]);
            r--;
            k--;
        }
    }
    for (int i = 0; i < 2 * n; i++) {
        cout << b[i] << " ";
    }
    cout << "\n";
    return 0;
}
