import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }
            int ans = 0;
            for (int i = 1; i < n; i++) {
                if (a[i] < a[i - 1]) {
                    ans = Math.max(ans, (a[i - 1] - a[i] + 1) / 2);
                }
            }
            System.out.println(ans);
        }
    }
}
