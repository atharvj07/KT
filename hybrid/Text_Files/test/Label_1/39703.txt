a = list(map(int, input().split()))
b = list(map(int, input().split()))
if sum(a) != sum(b):
    print("No")
    exit()
dk = dict()
for i in range(len(a)):
    x = b[i] - a[i]
    if x < 0 or x > 1:
        print("No")
        exit()
    if not x in dk:
        dk[x] = 0
    dk[x] += 1
v = dk.values()
if len(v) == 0:
    print("Yes")
elif min(v) == max(v):
    print("Yes")
else:
    print("No")
