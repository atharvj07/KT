#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
    int n;
    scanf("%d", &n);
    std::vector<std::pair<int, int>> v1, v2;
    for (int i = 0; i < n; ++i) {
        int a, b;
        scanf("%d %d", &a, &b);
        v1.emplace_back(a, b);
    }
    for (int i = 0; i < n; ++i) {
        int a, b;
        scanf("%d %d", &a, &b);
        v2.emplace_back(a, b);
    }
    std::vector<bool> ans1(n, false), ans2(n, false);
    for (int k = 0; 2 * k <= n; ++k) {
        int rest = n - 2 * k;
        std::vector<int> a, b;
        for (int i = 0; i < n; ++i) {
            a.push_back(v1[i].second);
            b.push_back(v2[i].second);
        }
        std::sort(a.begin(), a.end());
        std::sort(b.begin(), b.end());
        for (int i = 0; i < k; ++i) {
            ans1[std::lower_bound(a.begin(), a.end(), v1[i].second) - a.begin()] = true;
            ans2[std::lower_bound(b.begin(), b.end(), v2[i].second) - b.begin()] = true;
        }
        for (int i = 0; i < rest; ++i) {
            ans1[std::lower_bound(a.begin(), a.end(), v1[n - rest + i].second) - a.begin()] = true;
            ans2[std::lower_bound(b.begin(), b.end(), v2[n - rest + i].second) - b.begin()] = true;
        }
    }
    for (auto c : ans1) {
        putchar(c + '0');
    }
    putchar('\n');
    for (auto c : ans2) {
        putchar(c + '0');
    }
    putchar('\n');
    return 0;
}
