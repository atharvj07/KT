from collections import Counter

n = int(input())
a = list(map(int, input().split()))
d = Counter(a)

ans_min = 0
ans_max = 0
L = []

for i in d:
    c = d[i]
    if c > 1:
        ans_min += 1
    if c > 2:
        ans_max += int((c - 1) / 3)
    elif c == 2:
        L.append(i)

ans_min += len(d) - len(L)
ans_max += len(d)
print(ans_min, ans_max)
