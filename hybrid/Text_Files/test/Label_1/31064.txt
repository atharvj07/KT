import java.util.*;
import java.io.*;

public class Main {
    private static final int MOD = 1000000007;
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        int n = Integer.parseInt(br.readLine());
        int[] p = new int[n + 1];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) p[i] = Integer.parseInt(st.nextToken());
        int[] degree = new int[n + 1];
        for (int i = 1; i <= n; i++) if (p[i] != 0) { degree[p[i]]++; degree[i]++; }
        if (degree[1] % 2 == 1) { bw.write("NO\n"); bw.close(); return; }
        bw.write("YES\n");
        List<Integer> ans = new ArrayList<>();
        int[] degreeCopy = new int[n + 1];
        System.arraycopy(degree, 0, degreeCopy, 0, degree.length);
        LinkedList<Integer> queue = new LinkedList<>();
        queue.offer(1); degreeCopy[1] = 0;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            ans.add(cur);
            for (int i = 1; i <= n; i++) {
                if (p[i] == cur) {
                    if (--degreeCopy[i] % 2 == 0) {
                        queue.offer(i); degreeCopy[i] = 0;
                    }
                }
            }
        }
        for (int a : ans) bw.write(a + "\n");
        bw.close();
    }
}
