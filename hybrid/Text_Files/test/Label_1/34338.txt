import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        String s = input.next();

        int[] freq = new int[26];
        for (char c : s.toCharArray()) {
            freq[c - 'a']++;
        }

        int len = s.length() / k;
        for (int i = 0; i < 26; i++) {
            if (freq[i] % len != 0) {
                System.out.println("-1");
                return;
            }
        }

        char[] result = new char[s.length()];
        int idx = 0;
        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < freq[i] / k; j++) {
                for (int l = 0; l < len; l++) {
                    result[idx++] = (char) (i + 'a');
                }
            }
        }

        System.out.println(new String(result));
    }
}
