#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, b, a, s[200001], dp[200001][2];
  cin >> n >> b >> a;
  for (int i = 1; i <= n; ++i) {
    cin >> s[i];
  }
  dp[1][0] = (s[1]) ? b + 1 : b;
  dp[1][1] = (s[1]) ? 1 : 0;
  for (int i = 2; i <= n; ++i) {
    dp[i][0] = max({dp[i - 1][0] + (s[i]), dp[i - 1][1] + (s[i] ? 1 : 0), 0});
    dp[i][1] = max(dp[i - 1][0], dp[i - 1][1] + 1);
  }
  cout << max(dp[n][0] + (s[n]), dp[n][1] + (s[n] ? 1 : 0)) << endl;
}
