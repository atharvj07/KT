import java.util.Scanner;
import java.io.PrintWriter;

class Main {

    private static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = in.nextInt();
        int m = in.nextInt();

        int[] blockedRow = new int[n];
        int[] leftCol = new int[n];
        int[] rightCol = new int[n];
        for (int i = 0; i < n; i++) {
            blockedRow[i] = in.nextInt();
            leftCol[i] = in.nextInt();
            rightCol[i] = in.nextInt();
        }

        long[][] dp = new long[4][m + 1];
        dp[2][1] = 1L;
        for (int j = 1; j < m; j++) {
            for (int i = 0; i < 4; i++) {
                long inc = 0L;
                if (i > 1) {
                    inc += dp[i - 1][j + 1];
                    inc %= MOD;
                }
                inc += dp[i][j + 1];
                inc %= MOD;
                if (i < 3) {
                    inc += dp[i + 1][j + 1];
                    inc %= MOD;
                }

                dp[i][j + 1] = inc;
            }

            for (int k = 0; k < n; k++) {
                if (blockedRow[k] == 2 && leftCol[k] <= j && j <= rightCol[k]) {
                    dp[2][j + 1] = 0L;
                } else if (blockedRow[k] == 3 && leftCol[k] <= j) {
                    dp[3][j + 1] = 0L;
                }
            }
        }

        out.println(dp[2][m]);
        out.close();
    }
}
