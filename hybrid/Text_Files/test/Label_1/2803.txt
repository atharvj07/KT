#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, m, ans, a[N], p[N], s[N];
int main() {
    int t;
    cin >> t;
    while (t--) {
        ans = INT_MAX;
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> a[i];
        cin >> m;
        for (int i = 1; i <= m; i++) {
            cin >> p[i] >> s[i];
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                if (p[i] >= a[j]) cnt++;
                if (cnt == s[i]) break;
            }
            if (cnt == n) ans = min(ans, (n + s[i] - 1) / s[i]);
        }
        if (ans == INT_MAX)
            ans = -1;
        cout << ans << endl;
    }
    return 0;
}
