import java.util.*;

public class MinimalSegment {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int l = 1, r = n;
        int minLength = n + 1;
        int[] distinct = new int[100001];

        while (l <= r) {
            int m = l + (r - l) / 2;
            int count = 0;

            // Reset distinct array for each segment
            Arrays.fill(distinct, 0);

            for (int i = m - 1; i >= l - 1; i--) {
                if (distinct[a[i]] == 0) {
                    distinct[a[i]] = 1;
                    count++;
                }
            }

            if (count >= k) {
                if (m - l + 1 < minLength) {
                    minLength = m - l + 1;
                    r = m - 1;
                }
            } else {
                l = m + 1;
            }
        }

        if (minLength > n) {
            System.out.println("-1 -1");
        } else {
            System.out.println((n - minLength + 1) + " " + n);
        }
    }
}
