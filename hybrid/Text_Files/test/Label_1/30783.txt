import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Arrays;
import java.util.BitSet;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] adj = new int[n][n];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            for (int j = 0; j < n; j++) {
                adj[i][j] = Integer.parseInt(line[j]);
            }
        }
        long[] dp = new long[1 << n];
        boolean[][] connect = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            BitSet vis = new BitSet();
            vis.set(i);
            dfs(i, adj, connect, vis);
        }
        dp[0] = 1;
        for (int mask = 1; mask < 1 << n; mask++) {
            int last = -1;
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    if (last != -1 && !connect[last][i]) {
                        dp[mask] = 0;
                        break;
                    }
                    dp[mask] = dp[mask] * dp[mask ^ (1 << i)] % MOD;
                    last = i;
                }
            }
        }
        System.out.println(dp[(1 << n) - 1]);
    }

    private static void dfs(int u, int[][] adj, boolean[][] connect, BitSet vis) {
        for (int v = 0; v < adj.length; v++) {
            if (adj[u][v] == 1 && !vis.get(v)) {
                connect[u][v] = true;
                connect[v][u] = true;
                vis.set(v);
                dfs(v, adj, connect, vis);
            }
        }
    }
}
