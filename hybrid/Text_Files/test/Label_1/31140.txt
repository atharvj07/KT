import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++)
            arr[i] = in.nextLong();
        for (int i = 1; i < n; i++)
            arr[i] += arr[i - 1];
        long max = arr[n - 1];
        for (int i = 0; i < n - 1; i++)
            max = Math.max(max, arr[n - 1] - arr[i]);
        System.out.println(max);
    }
}
