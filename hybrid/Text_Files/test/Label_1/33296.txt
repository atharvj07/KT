import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        int b = 0;
        while (true) {
            b++;
            if ((a + b) % 10 == 8 || (a + b) / 10 % 10 == 8 || (a + b) / 100 % 10 == 8 || (a + b) / 1000 % 10 == 8 || (a + b) / 10000 % 10 == 8 || (a + b) / 100000 % 10 == 8 || (a + b) / 1000000 % 10 == 8 || (a + b) / 10000000 % 10 == 8 || (a + b) / 100000000 % 10 == 8) {
                break;
            }
        }
        System.out.println(b);
    }
}
