import math
n = int(input())
colors = [0] * (n + 1)
primes = [True] * (n + 1)
for i in range(2, math.isqrt(n) + 1):
    if primes[i]:
        for j in range(i * i, n + 1, i):
            primes[j] = False
            if colors[j] == 0:
                colors[j] = i
cnt = 0
for i in range(2, n + 1):
    if primes[i]:
        cnt += 1
for i in range(2, n + 1):
    if colors[i] == 0:
        colors[i] = cnt + 1
print(cnt + 1)
print(*colors[2:])
