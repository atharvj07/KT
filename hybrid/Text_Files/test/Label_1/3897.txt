#include <bits/stdc++.h>
#include <string>

using namespace std;
string ans, a, b;

int main() {
  cin >> a >> b;
  int n = a.size(), m = b.size(), i = 0, j = 0, k, cnt = 0;
  while (i < n) {
    if (a[i] == b[j]) {
      cnt++;
      j++;
    }
    if (j == m) {
      while (k <= i) { ans += a[k++]; }
      j = 0;
      cnt = 0;
    } else
      i++;
  }
  while (k <= i) { ans += a[k++]; }
  if (cnt != 0) cout << ans << endl;
  else
    cout << b << endl;
}
