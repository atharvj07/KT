import sys

n = int(input())
a = list(reversed(list(map(int, list(input())))))
b = list(reversed(list(map(int, list(input())))))
c = [10**5]
op = []
for i in range(1, n):
    if a[i] == b[i]:
        continue
    if a[i] == b[i] - 1:
        c.append(c[-1] + 1)
        op.append((i, 1))
        a[i] = b[i]
    elif a[i] == b[i] + 1:
        c.append(c[-1] + 1)
        op.append((i, -1))
        a[i] = b[i]
    else:
        c.append(10**5)
        break
if c[-1] == 10**5:
    print(-1)
    sys.exit()
print(c[-1])
for j in reversed(op):
    print(*j)
    if len(op) == 10**5:
        break
