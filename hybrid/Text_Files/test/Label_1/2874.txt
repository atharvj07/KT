#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m;
    cin >> n;
    long long a[n], b[n], ta = 0, tb = 0;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        ta += a[i];
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> b[i];
        tb += b[i];
    }
    long long ans = abs(ta - tb), a1 = 0, b1 = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (abs(ta + b[j] - a[i] - tb) < ans) {
                ans = abs(ta + b[j] - a[i] - tb);
                a1 = a[i];
                b1 = b[j];
            }
        }
    }
    cout << ans << endl;
    if (a1 != 0 && b1 != 0) {
        cout << 1 << endl;
        cout << a1 << " " << b1 << endl;
    } else {
        cout << 0 << endl;
    }
    return 0;
}
