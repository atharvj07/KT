import sys
input = sys.stdin.readline
n=int(input())
A=list(map(int,input().split()))
A.sort()
L=[]
for i in range(1,len(A)):
    if A[i]-A[i-1]>1: L.append(A[i]-1)
    elif A[i]-A[i-1]<1: break
if not L: L.append(1e9)
d=L[0]+1
if L[0]%d>0: L[0]=L[0]-L[0]%d
maxd=L[0]
mind=d
for v in L:
    if v%d>0: d=v-v%d
    mind=min(mind,v-v%d)
    maxd=max(maxd,v-v%d)
if d==1e9: print(-1)
else:
    print(int((maxd-mind)/d+1))
    for i in range(int(mind/d),int(maxd/d)+1):
        print(int(i*d),end=" ")
    print()
