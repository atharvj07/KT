import java.util.Arrays;
import java.util.PriorityQueue;
import java.util.Scanner;

class Main {
  static class Task implements Comparable<Task> {
    int time, n, d;

    public Task(int t, int n, int d) {
      this.time = t;
      this.n = n;
      this.d = d;
    }

    public int compareTo(Task t) {
      return this.time - t.time;
    }
  }

  static class FreeServer implements Comparable<FreeServer> {
    int id, end;

    public FreeServer(int id, int end) {
      this.id = id;
      this.end = end;
    }

    public int compareTo(FreeServer s) {
      return this.end - s.end;
    }
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();
    PriorityQueue<FreeServer> pq = new PriorityQueue<>();
    for (int i = 1; i <= n; i++) {
      pq.offer(new FreeServer(i, 0));
    }
    Task[] tasks = new Task[q];
    for (int i = 0; i < q; i++) {
      tasks[i] = new Task(sc.nextInt(), sc.nextInt(), sc.nextInt());
    }
    Arrays.sort(tasks);
    for (Task task : tasks) {
      while (!pq.isEmpty() && pq.peek().end <= task.time) {
        pq.poll();
      }
      if (pq.size() >= task.n) {
        int sum = 0;
        for (int i = 0; i < task.n; i++) {
          FreeServer s = pq.poll();
          sum += s.id;
          s.end = task.time + task.d;
          pq.offer(s);
        }
        System.out.println(sum);
      } else {
        System.out.println(-1);
      }
    }
  }
}
