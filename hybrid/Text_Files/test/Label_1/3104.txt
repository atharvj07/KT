#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 3e5 + 5;
vector<ll> a, b, c;
int main() {
  int n, m;
  cin >> n;
  a.resize(n);
  for (auto &x : a) cin >> x;
  cin >> m;
  b.resize(m);
  for (auto &x : b) cin >> x;
  for (auto x : a) c.push_back(x);
  for (auto x : b) c.push_back(x);
  sort(c.begin(), c.end());
  if (*max_element(a.begin(), a.end()) != *max_element(b.begin(), b.end())) {
    cout << -1;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    ll sum = 0;
    while (i < n && a[i] == c[i]) sum += a[i++];
    while (i < n && a[i] != c[i]) i++;
    if (i < n) a[i] = sum;
  }
  for (int i = 0; i < m; i++) {
    ll sum = 0;
    while (i < m && b[i] == c[i]) sum += b[i++];
    while (i < m && b[i] != c[i]) i++;
    if (i < m) b[i] = sum;
  }
  n = a.size();
  m = b.size();
  if (n != m) {
    cout << -1;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (a[i] != b[i]) {
      cout << -1;
      return 0;
    }
  }
  cout << n;
}
