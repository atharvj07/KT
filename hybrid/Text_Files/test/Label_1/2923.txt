#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    vector<ll> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    int cnt1 = 0, cnt2 = 0;
    for (int i = 0; i < n; ++i) {
        if (a[i] > 0) cnt1++;
        if (a[i] < 0) cnt2++;
    }
    if (cnt1 > cnt2)
        cout << "first";
    else if (cnt1 < cnt2)
        cout << "second";
    else {
        bool flag = 0;
        for (int i = 0; i < n; ++i) {
            if (a[i] > 0) {
                if (!flag) {
                    flag = 1;
                    continue;
                }
                if (a[i] <= a[i - 1]) { cout << "second"; return 0; }
            }
            if (a[i] < 0) {
                if (!flag) {
                    flag = 1;
                    continue;
                }
                if (-a[i] >= -a[i - 1]) { cout << "second"; return 0; }
            }
        }
        cout << "first";
    }
    return 0;
}
