import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] arr = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int m = Integer.parseInt(br.readLine());
        long[] tree = new long[n * 4];
        long[] lazy = new long[n * 4];
        build(arr, 1, 0, n - 1, tree);
        for (int i = 0; i < m; i++) {
            String[] line = br.readLine().split(" ");
            int op = Integer.parseInt(line[0]);
            if (op == 1) {
                long res = query(Integer.parseInt(line[1]) - 1, Integer.parseInt(line[2]) - 1, 1, 0, n - 1, tree, lazy);
                System.out.println(res);
            } else {
                update(Integer.parseInt(line[1]) - 1, Integer.parseInt(line[2]) - 1, Integer.parseInt(line[3]), 1, 0, n - 1, tree, lazy);
            }
        }
    }

    private static void build(int[] arr, int node, int start, int end, long[] tree) {
        if (start == end) {
            tree[node] = arr[start];
            return;
        }
        int mid = (start + end) / 2;
        build(arr, node * 2, start, mid, tree);
        build(arr, node * 2 + 1, mid + 1, end, tree);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    private static void update(int l, int r, int val, int node, int start, int end, long[] tree, long[] lazy) {
        if (lazy[node] != 0) {
            tree[node] ^= (end - start + 1) * lazy[node];
            if (start != end) {
                lazy[node * 2] ^= lazy[node];
                lazy[node * 2 + 1] ^= lazy[node];
            }
            lazy[node] = 0;
        }
        if (l > end || r < start) {
            return;
        }
        if (l <= start && r >= end) {
            tree[node] ^= (end - start + 1) * val;
            if (start != end) {
                lazy[node * 2] ^= val;
                lazy[node * 2 + 1] ^= val;
            }
            return;
        }
        int mid = (start + end) / 2;
        update(l, r, val, node * 2, start, mid, tree, lazy);
        update(l, r, val, node * 2 + 1, mid + 1, end, tree, lazy);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    private static long query(int l, int r, int node, int start, int end, long[] tree, long[] lazy) {
        if (lazy[node] != 0) {
            tree[node] ^= (end - start + 1) * lazy[node];
            if (start != end) {
                lazy[node * 2] ^= lazy[node];
                lazy[node * 2 + 1] ^= lazy[node];
            }
            lazy[node] = 0;
        }
        if (l > end || r < start) {
            return 0;
        }
        if (l <= start && r >= end) {
            return tree[node];
        }
        int mid = (start + end) / 2;
        long left = query(l, r, node * 2, start, mid, tree, lazy);
        long right = query(l, r, node * 2 + 1, mid + 1, end, tree, lazy);
        return left + right;
    }
}
