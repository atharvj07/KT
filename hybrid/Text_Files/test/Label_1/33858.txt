import java.util.Scanner;

public class BinaryHeapProperties {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int heapSize = scanner.nextInt();
        int[] heap = new int[heapSize + 1];
        for (int i = 1; i <= heapSize; i++) {
            heap[i] = scanner.nextInt();
        }

        for (int i = 1; i <= heapSize; i++) {
            int parentIndex = i / 2;
            int leftIndex = i * 2;
            int rightIndex = i * 2 + 1;

            System.out.printf("node %d: key = %d, ", i, heap[i]);
            if (parentIndex > 0) {
                System.out.printf("parent key = %d, ", heap[parentIndex]);
            }
            if (leftIndex <= heapSize) {
                System.out.printf("left key = %d, ", heap[leftIndex]);
            }
            if (rightIndex <= heapSize) {
                System.out.printf("right key = %d, ", heap[rightIndex]);
            }
            System.out.println();
        }
    }
}
