import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), x = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = scanner.nextInt();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long t = Math.max(1, x - i);
            ans += a[i] * t;
        }
        System.out.println(ans);
    }
}
