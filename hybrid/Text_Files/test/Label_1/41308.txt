import math
n, k = map(int, input().split())
a = sorted(int(x) for x in input().split())
mn, mx = math.inf, 0
for i in range(len(a)):
    mn = min(mn, a[i])
    mx = max(mx, a[i])
mn -= mn % k
mx = math.ceil(mx / k) * k
b = [(i, []) for i in range(mn, mx + k, k)]
for x in a:
    for i in range(len(b)):
        if b[i][0] <= x < b[i][0] + k:
            b[i][1].append(x)
for i in b: i[1].sort()
for x in a:
    for i in b:
        if x in i[1]:
            print(i[0], end=" ")
