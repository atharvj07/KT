#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1000000007;
const ll MAXN = 1e5 + 10;

ll num[MAXN];
ll vis[MAXN];
vector<ll> vec[MAXN];
vector<ll> v;

ll dfs(ll u, ll c) {
    if (vis[u] == 1 || num[u] != c) {
        return 0;
    }
    vis[u] = 1;
    ll ans = 0;
    for (ll j = 0; j < vec[u].size(); j++) {
        ll p = vec[u][j];
        if (num[p] == c) {
            ans++;
        }
    }
    return ans;
}

int main() {
    ll n, m;
    cin >> n >> m;
    for (ll i = 1; i <= n; i++) {
        ll x;
        cin >> x;
        num[i] = x;
    }
    for (ll i = 1; i <= m; i++) {
        ll x, y;
        cin >> x >> y;
        vec[x].push_back(y);
        vec[y].push_back(x);
    }
    for (ll i = 1; i <= n; i++) {
        if (vis[i] == 0) {
            ll mx = 0;
            for (ll j = 0; j < vec[i].size(); j++) {
                ll p = vec[i][j];
                mx = max(mx, dfs(p, num[i]));
            }
            v.push_back(mx);
        }
    }
    ll ans = 0;
    ll mn = 1e9;
    for (ll i = 0; i < v.size(); i++) {
        if (v[i] > ans) {
            ans = v[i];
            mn = num[i + 1];
        }
        else if (v[i] == ans) {
            mn = min(mn, num[i + 1]);
        }
    }
    cout << mn << endl;
    return 0;
}
