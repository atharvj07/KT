#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for(int i=a;i<b;++i)
const int maxn = 2e5+1;
int n, m, indeg[maxn], d[maxn];
bool used[maxn];
vector<int> g[maxn];
int main(){
    ios_base::sync_with_stdio(false), cin.tie(nullptr);
    cin >> n >> m;
    rep(i,0,m){
        int u, v;
        cin >> u >> v;
        --u, --v;
        g[u].push_back(v);
        g[v].push_back(u);
        ++indeg[u], ++indeg[v];
    }
    int mx = 0;
    rep(u,0,n){
        mx = max(mx, indeg[u]);
    }
    set<pair<int,int>> st;
    rep(u,0,n){
        if(indeg[u]==mx){
            st.insert({indeg[u], u});
        }
    }
    int lst = 0;
    while(st.size()){
        auto top = *st.rbegin();
        st.erase(top);
        if(d[top.second]==mx) continue;
        cout << ++lst << ' ' << top.second+1 << '\n';
        for(auto v: g[top.second]){
            --indeg[v];
            ++d[top.second];
            ++d[v];
            if(indeg[v]==mx){
                st.insert({indeg[v], v});
            }
        }
    }
}
