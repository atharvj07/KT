import java.util.*;

class Main {
  static int n;
  static long[] a;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextLong();
    }
    Arrays.sort(a);
    long ans = 0;
    int j = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] >= a[j]) {
        j++;
        ans++;
      }
    }
    System.out.println(ans);
  }
}
