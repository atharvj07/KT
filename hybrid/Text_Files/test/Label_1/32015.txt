import java.util.*;

public class KuroAndShiro {

    static class Road {
        int u, v, w;

        Road(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }

    static class Plan {
        int t, x;

        Plan(int t, int x) {
            this.t = t;
            this.x = x;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();

        List<Road> roads = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            int w = sc.nextInt();
            roads.add(new Road(u, v, w));
        }

        List<Plan> plans = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            int t = sc.nextInt() - 1;
            int x = sc.nextInt();
            plans.add(new Plan(t, x));
        }

        for (Plan plan : plans) {
            int roadIndex = plan.t;
            int newPrice = plan.x;
            roads.get(roadIndex).w = newPrice;

            int[] dist = new int[n];
            Arrays.fill(dist, Integer.MAX_VALUE);
            dist[0] = 0;
            PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[1] - b[1]);
            pq.offer(new int[]{0, 0});

            while (!pq.isEmpty()) {
                int[] curr = pq.poll();
                int node = curr[0];
                int cost = curr[1];

                if (cost > dist[node]) {
                    continue;
                }

                for (Road road : roads) {
                    if (road.u == node) {
                        int nextNode = road.v;
                        int nextCost = cost + road.w;
                        if (nextCost < dist[nextNode]) {
                            dist[nextNode] = nextCost;
                            pq.offer(new int[]{nextNode, nextCost});
                        }
                    } else if (road.v == node) {
                        int nextNode = road.u;
                        int nextCost = cost + road.w;
                        if (nextCost < dist[nextNode]) {
                            dist[nextNode] = nextCost;
                            pq.offer(new int[]{nextNode, nextCost});
                        }
                    }
                }
            }

            System.out.println(dist[n - 1]);
            roads.get(roadIndex).w = plan.x; // Restore original price
        }
    }
}
