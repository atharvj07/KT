import sys

T = int(input())

for _ in range(T):
    s = input()
    arr = []
    last = 0

    for i in range(len(s)):
        if i == 0: last = int(s[0])
        else: last = last + int(s[i]) - int(s[i - 1])
        arr.append(last)

    ans = list(s)

    for i in range(len(ans)):
        for j in range(i + 1, len(ans)):
            if arr[j] - (arr[i - 1] if i > 0 else 0) == 0:
                x = list(s[i:j + 1])
                x.reverse()

                for k in range(i, j + 1):
                    if x[k - i] == '0': ans[k] = '1'
                    else: ans[k] = '0'

    print(''.join(ans))
