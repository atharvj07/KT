import sys
n, q = map(int, sys.stdin.readline().split())
arr = [n]
print_arr = []

for i in range(q):
    query = sys.stdin.readline().split()
    op = int(query[0])
    if op == 2:
        l, r = map(int, query[1:3])
        print_arr.append(arr[l] - arr[r])
    else:
        p = int(query[1])
        arr.append(p)
        arr[-2] -= p
print(*print_arr, sep='\n')
