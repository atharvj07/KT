#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n + 1), b(n + 1);
    for (int i = 1; i <= n; ++i) {
        std::cin >> a[i];
    }
    int k = -1;
    for (int i = 1; i <= n; ++i) {
        std::cin >> b[i];
        if (a[i] == 0) k = i;
    }
    std::vector<int> vis(n + 1);
    vis[k] = 1;
    bool ok = true;
    int j = k;
    for (int i = 1; i <= n; ++i) {
        int nk = (j + a[j]) % n + 1;
        if (vis[nk]) {
            ok = false;
            break;
        }
        j = nk;
        vis[j] = true;
    }
    std::cout << (ok ? "YES" : "NO") << std::endl;
    return 0;
}
