#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; ++i) cin >> a[i];
  sort(a.rbegin(), a.rend());
  long long sum = accumulate(a.begin() + 1, a.end(), 0LL);
  int m;
  cin >> m;
  vector<int> q(m);
  for (int i = 0; i < m; ++i) {
    cin >> q[i];
    cout << sum << ' ';
    sum -= q[i] * a[q[i] - 1];
  }
  cout << '\n';
  return 0;
}
