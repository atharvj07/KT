#include <bits/stdc++.h>
using namespace std;
#define ll long long
void solve() {
    ll n, d = 0, x = 0, y = 0, f = 0;
    cin >> n;
    vector<vector<ll>> g(n + 1);
    vector<ll> mark(n + 1);
    for (ll i = 1; i < n; i++) {
        ll u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
        if (g[u].size() > 2 || g[v].size() > 2) {
            f++;
        }
    }
    if (f > 0) {
        cout << "NO" << endl;
        return;
    }
    for (ll i = 1; i <= n; i++) {
        if (!mark[i]) {
            d++;
            if (g[i].size() == 1) {
                x++;
            } else if (g[i].size() == 2) {
                y++;
            }
        }
        queue<ll> qu;
        qu.push(i);
        mark[i] = 1;
        while (!qu.empty()) {
            ll gop = qu.front();
            qu.pop();
            for (ll j = 0; j < g[gop].size(); j++) {
                if (!mark[g[gop][j]]) {
                    mark[g[gop][j]] = 1;
                    qu.push(g[gop][j]);
                }
            }
        }
    }
    if (d != 1 || x % 2 != 0) {
        cout << "NO" << endl;
        return;
    }
    cout << "YES" << endl;
    for (ll i = 1; i <= n; i++) {
        if (g[i].size() == 1) {
            cout << x << " " << y << endl;
            x += 2;
        } else {
            cout << x << " " << (!y) << endl;
            y++;
        }
    }
}
int main() {
    solve();
    return 0;
}
