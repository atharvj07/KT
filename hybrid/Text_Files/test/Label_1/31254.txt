import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String a = sc.next();
        String b = sc.next();
        int[] cnta = new int[26];
        int[] cntb = new int[26];
        for (int i = 0; i < a.length(); i++) {
            cnta[a.charAt(i) - 'a']++;
        }
        for (int i = 0; i < b.length(); i++) {
            cntb[b.charAt(i) - 'a']++;
        }
        for (int i = 0; i < 26; i++) {
            if (cnta[i] != cntb[i]) {
                System.out.println("No");
                return;
            }
        }
        System.out.println("Yes");
    }
}
