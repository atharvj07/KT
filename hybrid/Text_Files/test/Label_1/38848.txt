import sys
input = sys.stdin.readline
INF = 10**10

def dfs1(u, p):
    dp[u][0][0] = dp[u][1][0] = 0
    for v, w in g[u]:
        if v == p: continue
        dfs1(v, u)
        dp[u][0][0] += max(dp[v][0][0] + (w == 0), dp[v][1][0])
        dp[u][0][1] = max(dp[v][0][0] + (w == 0), dp[v][1][0] + (w == 1))
        dp[u][1][0] += max(dp[v][0][1], dp[v][1][1] + (w == 0))
        dp[u][1][1] = max(dp[v][0][1] + (w == 1), dp[v][1][1] + (w == 1))

def dfs2(u, p):
    for v, w in g[u]:
        if v == p: continue
        dp[v][0][0] = max(dp[u][0][0], dp[u][1][0])
        dp[v][1][1] = max(dp[u][1][1], dp[u][0][1] + (w == 1))
        dp[v][0][1] = dp[v][1][0] = max(dp[u][0][0], dp[u][1][0], dp[u][0][1], dp[u][1][1] + (w == 0))
        dfs2(v, u)

n = int(input())
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    x, y, z = map(int, input().split())
    g[x].append((y, z))
    g[y].append((x, z))

dp = [[[-INF] * 2 for _ in range(2)] for _ in range(n+1)]
dfs1(1, 1)
dfs2(1, 1)
ans = 0
for i in range(1, n+1):
    for j in range(2):
        for k in range(2):
            ans = max(ans, dp[i][j][k])

print(ans)
