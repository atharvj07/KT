import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int x = s.nextInt();
        int left = x;
        int distressed = 0;
        for (int i = 0; i < n; i++) {
            char c = s.next().charAt(0);
            int d = s.nextInt();
            if (c == '+') {
                left += d;
            } else {
                if (d <= left) {
                    left -= d;
                } else {
                    distressed++;
                }
            }
        }
        System.out.println(left + " " + distressed);
    }
}
