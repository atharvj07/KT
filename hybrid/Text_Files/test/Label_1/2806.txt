#include<bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
typedef long long ll;

ll f[105], g[105], p[105];

int main() {
    ll n, ans = 0; cin >> n;
    for(ll i = 1, x; i <= (1 << n) - 1; ++i) {
        cin >> x; g[i] = f[i - 1] + x; f[i] = f[(i >> 1) - 1] + f[i - 1] + x;
    }
    for(ll i = 1; i <= (1 << n) - 1; ++i) {
        ans += p[f[i]] + p[g[i]]; p[f[i]]++, p[g[i]]++;
    }
    cout << ans << endl;
    return 0;
}
