import sys
input = sys.stdin.readline
def dfs(i, p):
    if i == n:
        ans.append(q[:])
        return
    if p[i] != -1:
        dfs(i + 1, p)
    for j in range(len(q)):
        if p[i] == -1 and not in_q[q[j]]:
            q[i] = q[j]
            in_q[q[i]] = True
            p[i] = p[q[j]]
            dfs(i + 1, p)
            in_q[q[i]] = False
            q[i] = 0
            p[i] = -1
n, k = map(int, input().split())
p = list(map(int, input().split()))
e = [[] for i in range(n)]
for i in range(k):
    x, y = map(int, input().split())
    x, y = x - 1, y - 1
    e[x].append(y)
ans = []
in_q = [False] * (n + 1)
q = [0] * n
for i in range(n):
    if p[i] == 0:
        q[0] = i + 1
        in_q[i + 1] = True
        dfs(1, p)
        in_q[i + 1] = False
if len(ans) == 0:
    print(0)
else:
    print(*ans[0])
