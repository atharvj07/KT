#include <iostream>
using namespace std;

const int MAX_N = 50;
int dp[MAX_N][MAX_N], n, s, k;

int main() {
    cin >> n >> s >> k;
    int a[MAX_N], b[MAX_N];
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) { cin >> b[i]; }
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= k; j++) { dp[i][j] = 1e9; }
    }
    dp[s][0] = 0;
    for (int i = s + 1; i <= n; i++) {
        for (int j = 1; j <= k; j++) {
            if (b[i - 1] != b[i] && a[i] > a[i - 1]) {
                dp[i][j] = min(dp[i][j], dp[i - 1][j - a[i]]);
            }
            dp[i][j] = min(dp[i][j], dp[i - 1][j] + 1);
        }
    }
    int ans = 1e9;
    for (int j = k; j <= k; j++) { ans = min(ans, dp[n][j]); }
    if (ans == 1e9) {
        cout << "-1" << endl;
    } else {
        cout << ans << endl;
    }
    return 0;
}
