#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  while (k--) {
    long long m = LLONG_MAX;
    for (int i = 0; i < n; i++) {
      if (a[i] != 0) m = min(m, a[i]);
    }
    if (m == LLONG_MAX) {
      cout << 0 << '\n';
    } else {
      cout << m << '\n';
      for (int i = 0; i < n; i++) {
        if (a[i] != 0) a[i] -= m;
      }
    }
  }
  return 0;
}
