import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

class Main {
    public static void main(String[] args) throws IOException {
        final int MOD = 1009;
        java.math.BigInteger[] pow2 = new java.math.BigInteger[100005];
        pow2[0] = java.math.BigInteger.ONE;
        for (int i = 1; i < 100005; i++) {
            pow2[i] = (pow2[i-1].multiply(java.math.BigInteger.valueOf(2))).mod(java.math.BigInteger.valueOf(MOD));
        }

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = reader.readLine().split(" ");
        int n = Integer.parseInt(parts[0]);
        int m = Integer.parseInt(parts[1]);
        int k = Integer.parseInt(parts[2]);

        int[] colors = new int[n];

        String[] colorsLine = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            colors[i] = Integer.parseInt(colorsLine[i]);
        }

        int[] counts = new int[m];
        for (int color : colors) {
            counts[color-1]++;
        }

        java.math.BigInteger result = java.math.BigInteger.ONE;
        for (int count : counts) {
            if (count > 0) {
                result = result.multiply(pow2[count]).mod(java.math.BigInteger.valueOf(MOD));
            }
        }
        result = result.subtract(java.math.BigInteger.ONE).mod(java.math.BigInteger.valueOf(MOD));
        if (result.compareTo(java.math.BigInteger.ZERO) < 0) {
            result = result.add(java.math.BigInteger.valueOf(MOD));
        }

        System.out.println(result);
    }
}
