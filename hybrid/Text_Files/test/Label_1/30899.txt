import java.util.*;

class Main {
    private static Map<Long, Long> cache = new HashMap<>();

    private static long getRotatableNumber(int n, int b) {
        long res = 0;
        for (int i = 0; i < n; i++) {
            res = res * b + 1;
        }
        return res;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), x = in.nextInt();
        in.close();
        long ans = -1;
        for (int i = x - 1; i >= 2; i--) {
            long res = getRotatableNumber(n, i);
            if (res < x && cache.putIfAbsent(res, res) == null && isRotatableNumber(res, i, n)) {
                ans = i;
                break;
            }
        }
        System.out.println(ans);
    }

    private static boolean isRotatableNumber(long num, int b, int n) {
        for (int i = 2; i <= n; i++) {
            num = num * b / 10 + num % 10;
            if (cache.containsKey(num))
                return false;
            cache.put(num, num);
        }
        return true;
    }
}
