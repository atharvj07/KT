#include<bits/stdc++.h>
using namespace std;
map<string,string>mp;
int main(){
    int n;
    cin>>n;
    for(int i=0;i<n;i++){
        string s,p;
        cin>>s;
        int c=0;
        for(int j=0;j<s.size();j++){
            if(j==0&&s[j]>='0'&&s[j]<='9')c=1;
            if(s[j]=='=')break;
            p.push_back(s[j]);
        }
        s=s.substr(s.find('=')+1);
        if(c==0)mp[p]=s;
    }
    cout<<mp.size()<<endl;
    for(map<string,string>::iterator it=mp.begin();it!=mp.end();it++)cout<<it->first<<"="<<it->second<<endl;
}
