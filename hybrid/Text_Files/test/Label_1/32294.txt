import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Auction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt(); // Number of participants and bids
        List<Bid> bids = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            bids.add(new Bid(scanner.nextInt(), scanner.nextInt()));
        }

        int q = scanner.nextInt(); // Number of questions
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt(); // Number of absent participants
            int[] absentParticipants = new int[k];
            for (int j = 0; j < k; j++) {
                absentParticipants[j] = scanner.nextInt();
            }
            Bid winner = findWinner(bids, absentParticipants);
            if (winner != null) {
                System.out.println(winner.participant + " " + winner.bid);
            } else {
                System.out.println("0 0");
            }
        }
    }

    private static Bid findWinner(List<Bid> bids, int[] absentParticipants) {
        boolean[] isAbsent = new boolean[bids.size() + 1];
        for (int participant : absentParticipants) {
            isAbsent[participant] = true;
        }

        Bid winner = null;
        for (Bid bid : bids) {
            if (!isAbsent[bid.participant] && (winner == null || bid.bid > winner.bid)) {
                winner = bid;
            }
        }
        return winner;
    }

    private static class Bid {
        int participant;
        int bid;

        public Bid(int participant, int bid) {
            this.participant = participant;
            this.bid = bid;
        }
    }
}
