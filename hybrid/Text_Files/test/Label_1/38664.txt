def count_good_subarrays():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = input()
        prefix_sums = {0: 1}
        current_sum = 0
        good_subarrays = 0
        for i, value in enumerate(a, 1):
            current_sum += int(value)
            target_sum = current_sum - i
            if target_sum in prefix_sums:
                good_subarrays += prefix_sums[target_sum]
            if current_sum in prefix_sums:
                prefix_sums[current_sum] += 1
            else:
                prefix_sums[current_sum] = 1
        print(good_subarrays)

count_good_subarrays()
