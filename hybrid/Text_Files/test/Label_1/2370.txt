#include <iostream>
#include <algorithm>
using namespace std;
long long n, k, ans, pos;

int main() {
    cin >> n;
    while (n--) {
        cin >> k;
        pos = (k + 9) / 10;
        ans = (pos * (pos - 1)) / 2;
        k -= ans * 10 - 10;
        ans += k + 1;
        ans %= 10;
        cout << ans << "\n";
    }
    return 0;
}
