import sys
import re

reserved = list(map(str, sys.stdin.readline().split()))
words = [''] + list(map(str, sys.stdin.readline().split()))
tokens = re.compile(r'(#\S*|\d+|[^#\d\s]+|,)')
output = []
counter = 0
for line in sys.stdin.readlines():
    line = line.strip()
    for token in tokens.findall(line):
        if token in reserved:
            output.append(token)
        elif token in words:
            output.append(words[words.index(token)])
        else:
            counter += 1
            output.append(words[0] * (token not in words))
            words.append(token)
print(' '.join(output)[:-1] if not counter else ' '.join(output[:-1]))
