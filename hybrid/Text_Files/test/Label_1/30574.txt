import java.util.Scanner;

public class MinimumPossibleMaximumElement {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();

            // Find the maximum multiple of k less than n
            int maxMultiple = (n / k) * k;

            // Calculate the maximum element in the array
            int maximumElement = (maxMultiple + k - 1) / n;

            System.out.println(maximumElement);
        }
    }
}
