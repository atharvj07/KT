#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> freq(n + 1, 0);
    for (int i = 0; i < n; ++i) {
      int x;
      cin >> x;
      freq[x]++;
    }
    long long ans = 0;
    for (int i = 1; i <= n; ++i) {
      ans += 1LL * freq[i] * (freq[i] - 1) * (freq[i] - 2) / 6;
      ans += 1LL * freq[i] * freq[i + 1] * freq[i + 2];
      ans += 1LL * freq[i] * freq[i - 1] * freq[i - 2];
    }
    cout << ans << endl;
  }
  return 0;
}
