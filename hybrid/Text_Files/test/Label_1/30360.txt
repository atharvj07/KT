import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        int k = scanner.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = scanner.nextLong();
        }
        Arrays.sort(a);
        long maxCount = n / a[k - 1];
        System.out.println(k + " " + maxCount);
    }
}
