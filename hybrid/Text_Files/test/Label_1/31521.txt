import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n;
    while ((n = sc.nextInt()) > 0) {
      int a[] = new int[n];
      for (int i = 0; i < n; i++)
        a[i] = sc.nextInt();
      Arrays.sort(a);
      System.out.println(a[1] - a[0]);
    }
  }
}
