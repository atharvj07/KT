import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = sc.nextInt();
            roads[i][1] = sc.nextInt();
        }

        boolean[][] connected = new boolean[4][4];
        for (int[] road : roads) {
            connected[road[0] - 1][road[1] - 1] = true;
            connected[road[1] - 1][road[0] - 1] = true;
        }

        for (int i = 0; i < 4; i++) {
            int degree = 0;
            for (int j = 0; j < 4; j++) {
                if (connected[i][j]) {
                    degree++;
                }
            }
            if (degree != 2) {
                System.out.println("NO");
                return;
            }
        }

        System.out.println("YES");
    }
}
