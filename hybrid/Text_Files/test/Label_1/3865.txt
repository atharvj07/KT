#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<string> g(n);
  for (string &s : g)
    cin >> s;
  bool ok = false;
  for (int i = 0; i + 2 < n && !ok; i++)
    for (int j = 0; j + 2 < m && !ok; j++)
      ok = g[i][j] == g[i + 1][j] && g[i][j] == g[i][j + 1] &&
            g[i + 1][j + 1] == g[i][j] && g[i + 2][j] == g[i][j] &&
            g[i][j + 2] == g[i][j];
  if (!ok)
    for (int i = 0; i + 2 < n && !ok; i++)
      for (int j = 0; j + 2 < m && !ok; j++)
        ok = g[i][j] == g[i + 1][j] && g[i][j] == g[i][j + 1] &&
              g[i + 1][j + 1] == g[i][j] && g[i + 2][j] == g[i][j] &&
              g[i + 2][j + 1] == g[i][j] && g[i + 2][j + 2] == g[i][j];
  if (ok)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
}
