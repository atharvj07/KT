#include <bits/stdc++.h>
using namespace std;
#define pb push_back
const int maxn = 101, INF = 1e9 + 7;
vector<int> adj[maxn];
int t[maxn], dp[maxn][maxn];
bool vis[maxn][maxn];

int dfs(int node, int rem) {
    if (rem < 0 || node == 1) return -INF;
    if (vis[node][rem]) return dp[node][rem];
    vis[node][rem] = true;
    int &ret = dp[node][rem];
    ret = 0;
    for (int u : adj[node]) {
        if (rem >= t[u]) {
            ret = max(ret, 1 + dfs(u, rem - t[u] - 10));
        }
        ret = max(ret, 1 + dfs(u, rem - 5));
    }
    return ret;
}

int main() {
    int n, d, ans = -1;
    while (~scanf("%d %d", &n, &d)) {
        for (int i = 0; i < maxn; i++) adj[i].clear();
        memset(t, 0, sizeof t);
        memset(dp, 0, sizeof dp);
        memset(vis, false, sizeof vis);
        for (int i = 2; i <= n; i++) adj[1].pb(i);
        for (int i = 1; i <= n; i++) scanf("%d", &t[i]);
        ans = dfs(1, d);
        cout << (ans < 0 ? -1 : ans) << endl;
    }
    return 0;
}
