#include <iostream>
#include <cstdio>

using namespace std;

int main() {
    int n, x;
    cin >> n;
    int amt = 0;
    for (int i = 0; i < n; i++) {
        cin >> x;
        amt += x - 25;
        if (amt < 0) {
            cout << "NO" << endl;
            return 0;
        }
    }
    cout << "YES" << endl;
    return 0;
}
