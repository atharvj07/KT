import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        long sx = in.nextLong();
        long sy = in.nextLong();
        long fx = in.nextLong();
        long fy = in.nextLong();
        Set<Long> setx = new HashSet<>(), sety = new HashSet<>();
        for (long i = 0; i < m; i++) {
            long x = in.nextLong();
            long y = in.nextLong();
            setx.add(x);
            sety.add(y);
        }
        long ans = Math.abs(fx - sx) + Math.abs(fy - sy);
        if (setx.contains(sx) || sety.contains(sy)) ans = 1;
        if (setx.contains(fx) || sety.contains(fy)) ans = 2;
        System.out.println(ans);
    }
}
