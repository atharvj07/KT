import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String animals = sc.next();
        int h = 0, t = 0, misplaced = 0;
        for (char animal : animals.toCharArray()) {
            if (animal == 'H') {
                h++;
                if (misplaced > 0) misplaced--;
            } else {
                t++;
                misplaced++;
            }
        }
        System.out.println(misplaced);
    }
}
