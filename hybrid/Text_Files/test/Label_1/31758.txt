import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n + 1];
        int[] w = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        TreeMap<Integer, Integer> tm = new TreeMap<>();
        for (int i = 1; i <= n; i++) {
            tm.put(a[i], w[i]);
        }
        while (q-- > 0) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            if (x > 0) {
                int sum = 0;
                int l = x;
                int r = y;
                Map.Entry<Integer, Integer> floor = tm.floorEntry(l);
                Map.Entry<Integer, Integer> ceiling = tm.ceilingEntry(r);
                if (floor == null || floor.getKey() > l) {
                    floor = tm.lowerEntry(l);
                }
                if (ceiling == null || ceiling.getKey() < r) {
                    ceiling = tm.higherEntry(r);
                }
                if (floor != null && ceiling != null) {
                    sum += ceiling.getValue() * (ceiling.getKey() - r);
                    sum += floor.getValue() * (l - floor.getKey());
                    if (floor.getKey() != ceiling.getKey()) {
                        sum += (ceiling.getKey() - floor.getKey() - 1) * Math.min(floor.getValue(), ceiling.getValue());
                    }
                }
                System.out.println(sum % 1000000007);
            } else {
                int id = -x;
                int nw = y;
                tm.put(a[id], nw);
            }
        }
    }
}
