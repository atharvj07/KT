from math import gcd

def simplifyFraction(p, q):
    common_factor = gcd(p, q)
    return p // common_factor, q // common_factor

def isFinite(p, q, b):
    p, q = simplifyFraction(p, q)
    while True:
        g = gcd(q, b)
        if g == 1:
            break
        q //= g
    return q == 1

n = int(input())
for _ in range(n):
    p, q, b = map(int, input().split())
    print("Finite" if isFinite(p, q, b) else "Infinite")
