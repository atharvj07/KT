def calculate_maximum_strength(n, heights):
    answer = [0] * n
    stack = []
    for i in range(n):
        while stack and heights[stack[-1]] >= heights[i]:
            stack.pop()
        if stack:
            left = stack[-1]
        else:
            left = -1
        width = i - left
        answer[width-1] = max(answer[width-1], heights[i])
        stack.append(i)

    for i in range(n - 2, -1, -1):
        answer[i] = max(answer[i], answer[i + 1])

    return answer

n = int(input().strip())
heights = list(map(int, input().strip().split()))
print(*calculate_maximum_strength(n, heights))
