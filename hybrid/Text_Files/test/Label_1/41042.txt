for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    s = sum(a)
    k = s / n
    ans = 0
    l, r = [], []
    for v in a:
        if v < k:
            l += [v]
        elif v > k:
            r += [v]
    for v in l:
        lo = -1; hi = len(r) - 1
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if r[mid] * 2 <= s - 2 * v:
                lo = mid
            else:
                hi = mid - 1
        if lo != -1 and r[lo] * 2 == s - 2 * v:
            ans += len(l) - lo
    print(ans)
