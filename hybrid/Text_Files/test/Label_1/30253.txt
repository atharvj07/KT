import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), cnt = 0;
        char[][] a = new char[n][m], b = new char[n][m];
        for (int i = 0; i < n; i++) a[i] = in.next().toCharArray();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int[][] f = new int[26][5];
                f[a[i][j] - 'A'][0] = 1;
                f[a[i][j] - 'A'][1]++;
                f[a[i][j] - 'A'][2]++;
                for (int x = i; x < n; x++) {
                    for (int y = j; y < m; y++) {
                        f[a[x][y] - 'A'][0] = 1;
                        f[a[x][y] - 'A'][1]++;
                    }
                }
                for (int x = i; x < n; x++) {
                    for (int y = j; y < m; y++) {
                        f[a[x][y] - 'A'][2]++;
                        if (f[a[x][y] - 'A'][0] > 2 || f[a[x][y] - 'A'][1] > 2 || f[a[x][y] - 'A'][2] > 2 || f[a[x][y] - 'A'][3] > 2) {
                            if (f[a[x][y] - 'A'][3] < 2) b[x][y] = a[x][y];
                            else if (f[a[x][y] - 'A'][4] < 2) b[x][y] = (char) (a[x][y] + 1);
                            else if (f[a[x][y] - 'A'][5] < 2) b[x][y] = (char) (a[x][y] - 1);
                            else b[x][y] = (char) (a[x][y] % 4 + 'A');
                            cnt++;
                        } else b[x][y] = a[x][y];
                        f[a[x][y] - 'A'][0] = 0;
                        f[a[x][y] - 'A'][3]++;
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) System.out.print(b[i][j]);
            System.out.println();
        }
    }
}
