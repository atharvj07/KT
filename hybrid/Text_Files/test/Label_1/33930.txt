import java.util.Scanner;

public class Main {
    static int n, c;
    static int[] a, b;
    static int[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            n = sc.nextInt();
            c = sc.nextInt();
            if (n == 0 && c == 0) {
                break;
            }
            a = new int[n];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 16; j++) {
                    a[i] |= (sc.nextInt() << j);
                }
            }
            b = new int[c];
            for (int i = 0; i < c; i++) {
                for (int j = 0; j < 16; j++) {
                    b[i] |= (sc.nextInt() << j);
                }
            }
            dp = new int[n + 1][1 << 16];
            for (int i = 0; i < n + 1; i++) {
                for (int j = 0; j < (1 << 16); j++) {
                    dp[i][j] = -1;
                }
            }
            System.out.println(dfs(0, 0));
        }
    }

    static int dfs(int i, int s) {
        if (i == n) {
            return 0;
        }
        if (dp[i][s] != -1) {
            return dp[i][s];
        }
        int ret = 0;
        for (int j = 0; j < c; j++) {
            int ns = (s | b[j]) & a[i];
            ret = Math.max(ret, Integer.bitCount(ns) + dfs(i + 1, ns));
        }
        ret = Math.max(ret, dfs(i + 1, s));
        dp[i][s] = ret;
        return ret;
    }
}
