from sys import stdin

n, k = map(int, stdin.readline().split())
q = list(map(int, stdin.readline().split()))

l = [0] * n
for x in q:
    l[x - 1] = 1
ans = 0
for i in range(n):
    if not l[i]:
        for j in [i + 1, i - 1]:
            if 0 <= j < n and not l[j]:
                ans += 2
    else:
        if i == 0 or i == n - 1:
            ans += 1
        else:
            if l[i - 1] == 0 and l[i + 1] == 0:
                ans += 2
            elif l[i - 1] == 0 or l[i + 1] == 0:
                ans += 1
print(ans)
