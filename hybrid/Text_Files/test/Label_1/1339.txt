#include <bits/stdc++.h>
using namespace std;

int n, m, ans[200000];
bool mark[200000];
vector<int> v[200000];

int main() {
  scanf("%d %d", &n, &m);
  for (int i = 1; i <= m; ++i) {
    int x, y, z;
    scanf("%d %d", &x, &y);
    if (x > y) {
      swap(x, y);
    }
    if (mark[x * n + y]) {
      puts("NO");
      return 0;
    }
    mark[x * n + y] = true;
    v[x].push_back(y);
    v[y].push_back(x);
  }
  for (int i = 1; i <= n; ++i) {
    sort(v[i].begin(), v[i].end());
    ans[i] = i;
  }
  int cnt = 0;
  for (int i = 1; i <= n; ++i) {
    for (auto j : v[i]) {
      if (ans[i] == ans[j]) {
        ++cnt;
        if (cnt >= 2) {
          puts("YES");
          for (int i = 1; i <= n; ++i) {
            printf("%d ", ans[i]);
          }
          puts("");
          for (int i = 1; i <= n; ++i) {
            printf("%d ", i);
          }
          puts("");
          return 0;
        }
        ans[j] = ans[i] + 1;
      }
    }
  }
  puts("NO");
  return 0;
}
