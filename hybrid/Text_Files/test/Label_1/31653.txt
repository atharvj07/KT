import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long c = sc.nextInt();
        long[] a = new long[n - 1];
        long[] b = new long[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            b[i] = sc.nextInt();
        }
        long[] t = new long[n];
        t[0] = 0;
        for (int i = 1; i < n; i++) {
            if (a[i - 1] < b[i - 1] + c) {
                t[i] = t[i - 1] + a[i - 1];
            } else {
                t[i] = t[i - 1] + b[i - 1] + c;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(t[i] + " ");
        }
        System.out.println();
    }
}
