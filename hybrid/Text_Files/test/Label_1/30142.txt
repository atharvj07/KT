import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int s = sc.nextInt();
            int ans = 0;
            while (s >= 10) {
                int x = s / 10 * 10;
                ans += x;
                s -= x;
                s += x / 10;
            }
            ans += s;
            System.out.println(ans);
        }
    }
}
