#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int ans = -1;
    if (a == b) ans = 0;
    else {
      int cnt = 0;
      for (int i = 0; i < n; i++) {
        cnt += (a[i] == b[i]);
      }
      if (cnt == n - 1) ans = 1;
    }
    cout << ans << endl;
  }
  return 0;
}
