#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, w, a;
    cin >> n >> w;
    vector<int> v(n);
    for (int i = 0; i < n; ++i) {
        cin >> v[i];
    }
    unordered_map<int, int> seen;
    int ans = 0;
    for (int i = 0; i < w; ++i) {
        int cur = i;
        bool valid = true;
        for (int j = 0; j < n; ++j) {
            cur += v[j];
            if (cur < 0 || cur > w) {
                valid = false;
                break;
            }
            if (seen[cur] > 0) {
                valid = false;
                break;
            }
            seen[cur] += 1;
        }
        if (valid) ans++;
        seen.clear();
    }
    cout << ans << endl;
    return 0;
}
