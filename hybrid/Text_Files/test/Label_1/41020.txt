#!/bin/python3

import sys

def solve(n, m, s, queries):
    palindromes = [[-1] * n for i in range(n)]

    def is_palindrome(i, j):
        if i > j: return True
        if palindromes[i][j] != -1: return palindromes[i][j]
        if i == j: palindromes[i][j] = True
        elif s[i] == s[j]: palindromes[i][j] = is_palindrome(i + 1, j - 1)
        else: palindromes[i][j] = False
        return palindromes[i][j]

    ans = []
    for l, r in queries:
        l -= 1; r -= 1
        if is_palindrome(l, r): ans.append(0)
        elif s[l] == s[r]: ans.append(1)
        else: ans.append(2)
    return ans

n, m = map(int, input().split())
s = input()
queries = [list(map(int, input().split())) for i in range(m)]
print(*solve(n, m, s, queries))
