#include <bits/stdc++.h>
#define int long long
using namespace std;
const int maxn = 1e5 + 5;
vector<int> G[maxn];
int32_t main() {
    int n, k;
    cin >> n >> k;
    vector<int> d(n);
    for (int &x : d) { cin >> x; }
    vector<bool> vis(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        if (vis[i]) continue;
        vis[i] = 1;
        queue<pair<int, int>> q;
        q.push({i, 0});
        while (!q.empty()) {
            int u = q.front().first, dist = q.front().second;
            q.pop();
            if (dist > k) continue;
            for (int j = 1; j <= n; j++) {
                if (j == u || vis[j]) continue;
                if (dist + 1 == d[j]) {
                    G[u].push_back(j);
                    G[j].push_back(u);
                    vis[j] = 1;
                    q.push({j, dist + 1});
                }
            }
        }
    }
    int m = 0;
    for (int i = 1; i <= n; i++) { m += G[i].size(); }
    if (m / 2 != n - 1) {
        cout << -1 << endl;
        return 0;
    }
    cout << m / 2 << "\n";
    for (int i = 1; i <= n; i++) {
        for (int j : G[i]) {
            if (i < j) { cout << i << " " << j << "\n"; }
        }
    }
}
