#include <iostream>
#include <cstring>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    long long x;
    cin >> x;
    int digit[11] = {0};
    int t = 0;
    while (x > 0) {
      t = x % 10;
      x /= 10;
      digit[t]++;
    }
    int flag = 1;
    for (int j = 1; j <= k; j++) {
      if (digit[j] == 0) {
        flag = 0;
        break;
      }
    }
    ans += flag;
  }
  cout << ans << endl;
  return 0;
}
