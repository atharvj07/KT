t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    numbers = list(map(int, input().strip().split()))
    numbers.sort()
    ans = max(
        numbers[0] * numbers[1] * numbers[2] * numbers[3] * numbers[4],
        numbers[-1] * numbers[-2] * numbers[-3] * numbers[-4] * numbers[-5],
        numbers[0] * numbers[1] * numbers[2] * numbers[3] * numbers[-1],
        numbers[0] * numbers[1] * numbers[-1] * numbers[-2] * numbers[-3],
        numbers[0] * numbers[1] * numbers[2] * numbers[-2] * numbers[-1]
    )
    print(ans)
