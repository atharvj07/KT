n, T = map(int, input().split())
a = list(map(int, input().split()))

total_candies = 0
while True:
    spend = 0
    candies_this_round = 0
    for price in a:
        if T >= price:
            T -= price
            spend += price
            candies_this_round += 1
    if candies_this_round == 0:
        break
    rounds = T // spend
    T -= rounds * spend
    total_candies += rounds * candies_this_round
    if rounds == 0:
        total_candies += candies_this_round
        break

print(total_candies)
