#include <bits/stdc++.h>

const int MXN = 1e5 + 5;
int n, m;
int a[MXN], ans[MXN], l[MXN], r[MXN], mex[MXN];

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(0);
    std::cin >> n >> m;
    for (int i = 0; i < m; ++i) {
        std::cin >> l[i] >> r[i];
        mex[r[i]] = std::max(mex[r[i]], l[i]);
    }
    int last = -1, cur = 0;
    for (int i = 0; i < MXN; ++i) {
        if (mex[i] != 0) {
            while (last + 1 < i) {
                if (last != -1)
                    ans[last] = cur;
                cur++;
                last++;
            }
        }
    }
    std::cout << cur << '\n';
    for (int i = 0; i < n; ++i) {
        std::cout << ans[i] << ' ';
    }
    return 0;
}
