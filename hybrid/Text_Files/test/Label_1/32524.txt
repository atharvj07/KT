import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] >= 2) {
                ans += a[i] / 2;
                a[i] %= 2;
            }
            if (a[i] == 1 && i + 1 < n && a[i + 1] >= 1) {
                ans++;
                a[i]--;
                a[i + 1]--;
            }
        }
        System.out.println(ans);
    }
}
