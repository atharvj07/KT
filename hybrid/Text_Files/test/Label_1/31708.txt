import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt(), k = sc.nextInt();
            int[] h = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                h[i] = sc.nextInt();
            }
            int pos = 1;
            for (int i = 1; i < k; i++) {
                if (pos == n) {
                    pos = -1;
                    break;
                }
                if (h[pos] >= h[pos + 1]) {
                    pos++;
                } else {
                    h[pos]++;
                }
            }
            System.out.println(pos);
        }
    }
}
