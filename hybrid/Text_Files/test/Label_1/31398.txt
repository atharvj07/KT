import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    while (t-- > 0) {
      int x = sc.nextInt();
      int p = sc.nextInt();
      int k = sc.nextInt();
      int count = 0;
      for (int i = x + 1; ; i++) {
        if (gcd(i, p) == 1) {
          count++;
        }
        if (count == k) {
          System.out.println(i);
          break;
        }
      }
    }
  }

  static int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
  }
}
