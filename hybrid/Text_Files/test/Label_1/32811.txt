import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int m = input.nextInt();
    int t = input.nextInt();
    int r = input.nextInt();
    int[] visits = new int[m];
    for (int i = 0; i < m; i++) {
      visits[i] = input.nextInt();
    }
    int min = -1;
    for (int i = 0; i < m; i++) {
      int curMin = (int) Math.ceil((double) (visits[i] - 1) / (double) t);
      if (curMin >= r) {
        if (min == -1) {
          min = curMin;
        } else {
          min = Math.min(min, curMin);
        }
      }
    }
    System.out.println(min);
  }
}
