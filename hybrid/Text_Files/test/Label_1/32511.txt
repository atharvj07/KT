import java.util.*;

public class Main {

    static int[] a, b;
    static List<Integer>[] g;
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        g = new List[n];
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            g[u].add(v);
            g[v].add(u);
        }
        int[] ans = solve();
        if (ans == null) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int i : ans) {
                System.out.println(i);
            }
        }
    }

    public static int[] solve() {
        int[] flow = new int[n];
        for (int i = 0; i < n; i++) {
            flow[i] = b[i] - a[i];
        }
        int[] deg = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j : g[i]) {
                deg[i]++;
                deg[j]++;
            }
        }
        int sum = 0;
        for (int i : flow) {
            sum += i;
        }
        if (sum % 2 != 0) {
            return null;
        }
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j : g[i]) {
                int d = (deg[i] + deg[j]) % 2 == 0 ? 1 : -1;
                int val = (flow[i] - flow[j]) / 2 * d;
                if (flow[i] + val >= 0 && flow[j] - val >= 0) {
                    ans[i] += val;
                    ans[j] -= val;
                    flow[i] += val;
                    flow[j] -= val;
                }
            }
        }
        boolean ok = true;
        for (int i : flow) {
            ok &= i == 0;
        }
        return ok ? ans : null;
    }
}
