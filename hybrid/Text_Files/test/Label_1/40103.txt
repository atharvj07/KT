import sys

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    x = int(input())
    sa = [0] * n
    sa[0] = a[0]
    for i in range(1, n):
        sa[i] = sa[i - 1] + a[i]
    ans = 0
    l = 0
    for r in range(n):
        while l < r and sa[r] - (sa[l - 1] if l > 0 else 0) < x * (r - l + 1):
            l += 1
        ans = max(ans, r - l + 1)
    print(ans)
