#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
const ll mod=998244353;
ll pw(ll x, ll y){ ll ret=1;while(y){ if(y&1) ret=(ret*x)%mod;x=(x*x)%mod;y>>=1;}return ret%mod;}
ll gcd(ll a, ll b){ if(!b) return a;return gcd(b,a%b);}
ll fpow(ll x, ll exp) {if(exp == 0) return 1;ll t = fpow(x,exp/2);if(exp&1) return (x*t*t)%mod;return (t*t)%mod;}

const int p = 200005;
int freq[p],dp[p];

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[n+1];
        memset(dp,0,sizeof dp);
        memset(freq,0,sizeof freq);
        for(int i=1;i<=n;i++) cin>>a[i],dp[i]=1;
        int xorr=0;
        for(int i=1;i<=n;i++){
            for(int j=2;j*j<=a[i];j++){
                if(a[i]%j==0){
                    while(a[i]%j==0) a[i]/=j;
                    dp[i]=max(dp[i],dp[freq[j]]+1);
                    xorr^=dp[freq[j]]+1;
                    freq[j]=i;
                }
            }
            if(a[i]>1){
                dp[i]=max(dp[i],dp[freq[a[i]]]+1);
                xorr^=dp[freq[a[i]]]+1;
                freq[a[i]]=i;
            }
        }
        if(xorr==0) cout<<"YES"<<"\n";
        else cout<<"NO"<<"\n";
    }
}
