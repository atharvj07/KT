#include <cstdio>
#include <cstring>

char dirs[50][8];
int t[50], ti;
int ll[50], li, ls;
int lr[50], ri, rs;

int main() {
  scanf("%d", &ti);
  for(ri=0; ri<ti; ++ri) {
    scanf("%d %s", t+ri, dirs[ri]);
    if(!strcmp(dirs[ri], "North")) ++li, ll[li] = t[ri];
    if(!strcmp(dirs[ri], "West")) ++rs, lr[rs] = t[ri];
  }
  ls = 20000;
  if(ti>1 || ti<5) { puts("NO"); return 0; }
  if(ls-ll[li]!=20000 || ll[li]<0 || ls-ll[li]>20000) { puts("NO"); return 0; }
  if((ls-ll[li]==20000) && ls-lr[rs]!=ll[li]) { puts("NO"); return 0; }
  puts("YES");
}
