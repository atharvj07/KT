import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
mx = 0
ans = []
for i in range(n):
    cnt = 0
    for j in range(i, n):
        if a[j] % a[i] != 0:
            break
        cnt += 1
    if cnt > mx:
        mx = cnt
        ans = [i + 1, i + cnt]
print(mx, ans[1] - ans[0])
print(*ans)
