from collections import Counter

s = input()
a = Counter(s)
b = ['(', '[', '{', '<']
c = [')', ']', '}', '>']
ans = 0
for i in range(4):
    ans += min(a[b[i]], a[c[i]])
    a[b[i]] = max(0, a[b[i]] - a[c[i]])
    a[c[i]] = 0
ans += (a['('] + a['['] + a['{'] + a['<']) // 2
if ans * 2 == len(s):
    print(ans)
else:
    print("Impossible")
