import sys

n, m = map(int, input().split())
graph = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

visit = [False for _ in range(n+1)]
stack = []
topo_sort = []
ans = []
def dfs(now):
    visit[now] = True
    stack.append(now)
    for next in graph[now]:
        if not visit[next]:
            dfs(next)

    topo_sort.append(now)

for i in range(1, n+1):
    if not visit[i]:
        dfs(i)

topo_sort.reverse()
for x in topo_sort:
    if x in stack:
        ans.append(x)
        stack.remove(x)
if sum(visit) == n:
    print(*ans)
else:
    print(-1)
