import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] board = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                board[i][j] = sc.nextInt();
            }
        }
        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int rowSum = 0;
                int colSum = 0;
                for (int k = 0; k < n; k++) {
                    rowSum += board[i][k];
                    colSum += board[k][j];
                }
                if (colSum > rowSum) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
