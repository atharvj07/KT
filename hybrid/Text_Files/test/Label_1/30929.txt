import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int[] arr = new int[m];
        for (int i = 0; i < m; i++) {
            arr[i] = sc.nextInt();
        }
        int[] dp = new int[m];
        int max = 0;
        for (int i = 0; i < m; i++) {
            dp[i] = 0;
            for (int j = i + 1; j < m; j++) {
                if (arr[j] > arr[i]) {
                    dp[i] = Math.max(dp[i], dp[j]);
                }
            }
            dp[i] += 1;
            max = Math.max(max, dp[i]);
        }
        System.out.println(max);
        int[] res = new int[max];
        int pos = 0;
        for (int i = m - 1; i >= 0 && pos < max; i--) {
            if (dp[i] == max) {
                res[pos++] = arr[i];
                max--;
            }
        }
        for (int i = res.length - 1; i >= 0; i--) {
            System.out.print(res[i] + " ");
        }
    }
}
