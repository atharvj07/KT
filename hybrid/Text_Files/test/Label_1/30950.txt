import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int[] a = new int[n];
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = in.nextInt();
            }
            for (int i = 0; i < n; i++) {
                b[i] = in.nextInt();
            }
            boolean valid = true;
            for (int i = 0; i < n; i++) {
                if (b[i] - a[i] < 0 || (b[i] - a[i]) % 2 == 1) {
                    valid = false;
                    break;
                }
            }
            System.out.println(valid ? "YES" : "NO");
        }
        in.close();
    }
}
