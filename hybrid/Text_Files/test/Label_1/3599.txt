#include<bits/stdc++.h>
using namespace std;
int main()
{
	string s;
	cin >> s;
	int cnt[256]={0};
	int odd=0;
	for(int i=0;i<s.length();i++)
	{
		cnt[s[i]]++;
		if(cnt[s[i]]%2)odd++;
		else odd--;
	}
	if(odd>1)
	{
		cout << "Impossible\n";
		return 0;
	}
	string ans1="",ans2="";
	for(int i=0;i<256;i++)
	{
		if(cnt[i]%2)
		{
			int add=(cnt[i]-1)/2;
			while(add--)ans1+=(char)i,ans2=(char)i+ans2;
		}
		else
		{
			int add=cnt[i]/2;
			while(add--)ans1+=(char)i,ans2=(char)i+ans2;
		}
	}
	cout << ans1+ans2 << '\n';
}
