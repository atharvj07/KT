#include<bits/stdc++.h>
#define pb push_back
using namespace std;typedef long long ll;
vector<ll>v,v1;
main()
{
    ll n;cin>>n;
    v.pb(0);
    for(ll i=1;i<=n;i++){
        ll x;cin>>x;
        v.pb(x);
    }
    sort(v.begin(),v.end());
    ll cnt=0;
    for(ll i=1;i<n;i++)
    {
        if(v[i]!=v[i-1]) cnt++;
        else if(i>1 and v[i]==v[i-2]){cout<<"NO";return 0;}
    }
    cnt++;
    if(cnt>2){cout<<"NO";return 0;}
    for(ll i=1;i<=n;i++){
        if(i==1||v[i]==v[i-1]) v1.pb(1);
        else v1.pb(0);
    }
    cout<<"YES"<<endl;
    for(ll i=0;i<n;i++) cout<<v1[i];
}
