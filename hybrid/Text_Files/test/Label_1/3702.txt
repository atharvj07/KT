#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0, ans = 0;
  vector<int> pref(n), suf(n);
  for (int i = 0; i < n; i++) {
    if (s[i] == '0')
      cnt--;
    else
      cnt++;
    pref[i] = cnt;
  }
  cnt = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == '0')
      cnt--;
    else
      cnt++;
    suf[i] = cnt;
  }
  for (int i = 0; i < n; i++) {
    int l = i - 1, r = i + 1;
    while (l >= 0 && suf[l] >= pref[i])
      l--;
    while (r < n && suf[r] >= pref[i])
      r++;
    ans = max(ans, r - l - 1);
  }
  cout << ans << endl;
}
