import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class SegmentTree {
  int tree[];
  long lazy[];
  int N;

  public SegmentTree(int n) {
    N = n;
    tree = new int[4 * N];
    lazy = new long[4 * N];
  }

  void build(int node, int s, int e, int a[]) {
    if (s == e) {
      tree[node] = a[s];
      return;
    }
    int mid = (s + e) / 2;
    build(2 * node, s, mid, a);
    build(2 * node + 1, mid + 1, e, a);
    tree[node] = tree[2 * node] + tree[2 * node + 1];
  }

  void update(int node, int s, int e, int l, int r, long val) {
    if (lazy[node] != 0) {
      tree[node] ^= (e - s + 1) * lazy[node];
      if (s != e) {
        lazy[2 * node] ^= lazy[node];
        lazy[2 * node + 1] ^= lazy[node];
      }
      lazy[node] = 0;
    }
    if (s > r || e < l)
      return;
    if (s >= l && e <= r) {
      tree[node] ^= (e - s + 1) * val;
      if (s != e) {
        lazy[2 * node] ^= val;
        lazy[2 * node + 1] ^= val;
      }
      return;
    }
    int mid = (s + e) / 2;
    update(2 * node, s, mid, l, r, val);
    update(2 * node + 1, mid + 1, e, l, r, val);
    tree[node] = tree[2 * node] + tree[2 * node + 1];
  }

  long query(int node, int s, int e, int l, int r) {
    if (lazy[node] != 0) {
      tree[node] ^= (e - s + 1) * lazy[node];
      if (s != e) {
        lazy[2 * node] ^= lazy[node];
        lazy[2 * node + 1] ^= lazy[node];
      }
      lazy[node] = 0;
    }
    if (s > r || e < l)
      return 0;
    if (s >= l && e <= r)
      return tree[node];
    int mid = (s + e) / 2;
    long p1 = query(2 * node, s, mid, l, r);
    long p2 = query(2 * node + 1, mid + 1, e, l, r);
    return p1 + p2;
  }
}

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++)
      a[i] = sc.nextInt();
    SegmentTree tree = new SegmentTree(n);
    tree.build(1, 0, n - 1, a);
    int m = sc.nextInt();
    List<String> ans = new ArrayList<>();
    while (m-- > 0) {
      int type = sc.nextInt();
      if (type == 1) {
        int l = sc.nextInt() - 1;
        int r = sc.nextInt() - 1;
        long sum = tree.query(1, 0, n - 1, l, r);
        ans.add(sum + "");
      } else {
        int l = sc.nextInt() - 1;
        int r = sc.nextInt() - 1;
        int val = sc.nextInt();
        tree.update(1, 0, n - 1, l, r, val);
      }
    }
    for (String an : ans)
      System.out.println(an);
  }
}
