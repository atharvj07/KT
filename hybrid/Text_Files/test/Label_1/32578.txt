import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        try {
            StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            int n = Integer.parseInt(tokenizer.nextToken());
            int k = Integer.parseInt(tokenizer.nextToken());

            Map<Integer, List<Integer>> graph = new HashMap<>();

            for (int i = 0; i < n; i++) {
                graph.put(i + 1, new ArrayList<>());
            }

            for (int i = 0; i < n - 1; i++) {
                tokenizer = new StringTokenizer(reader.readLine());
                int a = Integer.parseInt(tokenizer.nextToken());
                int b = Integer.parseInt(tokenizer.nextToken());

                graph.get(a).add(b);
                graph.get(b).add(a);
            }

            if (!isEulerian(graph, n, k)) {
                System.out.println("No");
            } else {
                System.out.println("Yes");

                int[] colors = new int[n];

                eulerianPath(graph, 1, 1, colors, new boolean[n + 1]);

                for (int color : colors) {
                    System.out.print(color + " ");
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

    private static boolean isEulerian(Map<Integer, List<Integer>> graph, int n, int k) {
        int oddDegreeCount = 0;

        for (List<Integer> neighbors : graph.values()) {
            if (neighbors.size() % 2 != 0) {
                oddDegreeCount++;
            }
        }

        return oddDegreeCount == 0 || (oddDegreeCount == 2 && k % 2 == 0);
    }

    private static void eulerianPath(Map<Integer, List<Integer>> graph, int current, int color, int[] colors, boolean[] visited) {
        colors[current - 1] = color;
        visited[current] = true;

        List<Integer> neighbors = graph.get(current);

        for (int next : neighbors) {
            if (!visited[next]) {
                eulerianPath(graph, next, color % graph.size() + 1, colors, visited);
            }
        }
    }
}
