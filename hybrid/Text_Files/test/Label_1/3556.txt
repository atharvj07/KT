#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;

  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  sort(a.begin(), a.end());

  int min_diff = a[1] - a[0];
  for (int i = 2; i < n; i++) { min_diff = min(min_diff, a[i] - a[i - 1]); }

  int num_pairs = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] - a[i - 1] == min_diff) { num_pairs++; }
  }

  cout << min_diff << " " << num_pairs << endl;

  return 0;
}
