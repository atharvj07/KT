#include<bits/stdc++.h>
using namespace std;
#define rep(i,a,n) for(int i=a;i<n;i++)
#define per(i,a,n) for(int i=n-1;i>=a;i--)
typedef long long ll;
const ll mod=1000000007;
const int N=(int)1e6+10;
void solve(){
    int n,m;
    cin>>n>>m;
    vector<vector<int>>a(n,vector<int>(m));
    vector<int>r(n),c(m);
    rep(i,0,n){
        rep(j,0,m){
            cin>>a[i][j];
            if(a[i][j])r[i]^=1,c[j]^=1;
        }
    }
    rep(i,0,n){
        rep(j,0,m){
            if(a[i][j]!=((r[i]^c[j])&1))return void(cout<<"NO");
        }
    }
    cout<<"YES\n";
    rep(i,0,n)cout<<(r[i]&1);
    cout<<'\n';
    rep(j,0,m)cout<<(c[j]&1);cout<<'\n';
    return;
}
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    solve();
    return 0;
}
