#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ll n, k, d;
    cin >> n >> k >> d;
    vector<ll> a(n);
    for (auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    ll mx_num = 0;
    for (auto x : a) {
        mx_num = (mx_num * k + x) % k;
    }
    cout << ((mx_num % k == (k - d % k) % k) ? "1\n0" : "0") << endl;
}
