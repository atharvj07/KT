q = int(input().strip())
for _ in range(q):
    n, k = map(int, input().strip().split())
    s = list(input().strip())
    i = 0
    while k > 0 and i < n:
        if s[i] == '0':
            j = i
            while j > 0 and k > 0 and s[j-1] == '1':
                s[j], s[j-1] = s[j-1], s[j]
                k -= 1
                j -= 1
        i += 1
    print("".join(s))
