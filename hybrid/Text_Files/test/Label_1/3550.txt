#include <iostream>
#include <vector>
#include <queue>
#include <utility>

struct Edge { int to, w; };

std::vector<int> dijkstra(int n, int src, const std::vector<std::vector<Edge>>& g) {
    std::vector<int> dist(n + 1, 1e9);
    dist[src] = 0;
    std::priority_queue<std::pair<int, int>> pq;
    pq.push({ 0, src });
    while (!pq.empty()) {
        auto [w, u] = pq.top(); pq.pop();
        if (w > dist[u]) continue;
        for (const auto& e : g[u]) {
            int v = e.to, w2 = e.w;
            if (dist[v] > dist[u] + w2) {
                dist[v] = dist[u] + w2;
                pq.push({ dist[v], v });
            }
        }
    }
    return dist;
}

int main() {
    int t; std::cin >> t;
    while (t--) {
        int n; std::cin >> n;
        std::vector<std::vector<Edge>> g(n + 1);
        for (int i = 0; i < n - 1; i++) {
            int u, v; std::cin >> u >> v;
            g[u].push_back({ v, 1 });
            g[v].push_back({ u, 1 });
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            auto dist = dijkstra(n, i, g);
            int mx = 0;
            for (int j = 1; j <= n; j++) {
                if (j == i) continue;
                mx = std::max(mx, dist[j]);
            }
            ans = std::max(ans, mx);
        }
        std::cout << ans << "\n";
    }
    return 0;
}
