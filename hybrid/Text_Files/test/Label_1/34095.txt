import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[100000];
        dp[0] = 1;
        for (int i = 0; i < n; i++) {
            int[] ndp = new int[100000];
            for (int j = 0; j < 100000; j++) {
                ndp[(j + a[i]) % 100000] += dp[j];
                ndp[(j + a[i]) % 100000] %= (int) Math.pow(2, 58);
            }
            dp = ndp;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(dp[i]);
        }
    }
}
