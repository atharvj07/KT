#include <bits/stdc++.h>

using namespace std;
const int MAXN = 1005;
int r, c, vis[MAXN][MAXN];
char grid[MAXN][MAXN];

inline void dfs(int i, int j) {
  if (i < 0 || j < 0 || i >= r || j >= c || vis[i][j] || grid[i][j] == '*')
    return;
  vis[i][j] = 1;
  dfs(i + 1, j);
  dfs(i - 1, j);
  dfs(i, j + 1);
  dfs(i, j - 1);
}

int main() {
  cin >> r >> c;
  for (int i = 0; i < r; i++) { cin >> grid[i]; }
  bool ok = 0;
  for (int i = 0; i < r; i++) {
    for (int j = 0; j < c; j++) {
      if (grid[i][j] == '.') {
        memset(vis, 0, sizeof vis);
        dfs(i, j);
        ok |= (vis[0][0] && vis[r - 1][0] && vis[0][c - 1] && vis[r - 1][c - 1]);
      }
    }
  }
  if (ok)
    cout << "YES\n";
  else
    cout << "NO\n";
  return 0;
}
