import java.util.*;

public class Main {
    static class Node {
        char ch;
        int cnt;
        HashMap<Character, Node> nxt;
        public Node(char ch) { this.ch = ch; cnt = 0; nxt = new HashMap<>(); }
    }

    static class Trie {
        Node root;
        public Trie() { root = new Node(Character.MIN_VALUE); }

        void add(String s) { add(s, 0); }

        void add(String s, int idx) {
            if (idx == s.length()) { root.cnt++; return; }
            Node cur = root;
            while (idx < s.length()) {
                char ch = s.charAt(idx++);
                if (!cur.nxt.containsKey(ch)) cur.nxt.put(ch, new Node(ch));
                cur = cur.nxt.get(ch);
            }
            cur.cnt++;
        }

        boolean isGood(String s) {
            return isGood(s, 0, root);
        }

        boolean isGood(String s, int idx, Node cur) {
            if (idx == s.length()) return true;
            char ch = s.charAt(idx++);
            Node nxt = cur.nxt.get(ch);
            return nxt != null && nxt.cnt == cur.cnt && isGood(s, idx, nxt);
        }
    }

    static String sol(String[] ar) {
        Trie trie = new Trie();
        for (String s : ar) trie.add(s);
        String ans = "NO";
        for (String s : ar)
            if (trie.isGood(s))
                if (ans.isEmpty() || s.length() < ans.length() || (s.length() == ans.length() && s.compareTo(ans) < 0))
                    ans = s;
        return ans;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(); input.nextLine();
        String[] ar = new String[n];
        for (int i = 0; i < n; i++)
            ar[i] = input.nextLine();
        System.out.println(sol(ar));
    }
}
