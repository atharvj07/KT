import math
t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    b = [1] * n
    for j in range(n):
        for k in range(j, n):
            b[k] *= a[j]
    perfect_square = False
    for j in b:
        c = int(math.sqrt(j))
        if c * c == j:
            perfect_square = True
            break
    if not perfect_square:
        print("YES")
    else:
        print("NO")
