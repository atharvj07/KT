#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    int n;
    cin>>n;
    vector<pair<int,int> > v;
    for(int i=0;i<2*n;i++){
        int x;
        cin>>x;
        v.push_back({x,0});
    }
    sort(v.begin(),v.end());
    int mx=0,mn=1e9;
    for(int i=0;i<2*n;i++){
        if(v[i].second==0)
            v[i].second=1,mx=max(mx,v[i].first);
        else
            mn=min(mn,v[i].first);
    }
    cout<<abs(mn-mx)<<endl;
    return 0;
}
