import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            pq.add(sc.nextLong());
        }
        long totalCost = 0;
        while (pq.size() > 1) {
            long s1 = pq.poll();
            long s2 = pq.poll();
            totalCost += s1 + s2;
            pq.add(s1 + s2);
        }
        System.out.println(totalCost);
    }
}
