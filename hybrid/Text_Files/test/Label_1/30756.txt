import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();

            int[] savings = new int[n];
            for (int j = 0; j < n; j++) {
                savings[j] = scanner.nextInt();
            }

            int maxWealthyPeople = calculateMaxWealthyPeople(savings, x);
            System.out.println(maxWealthyPeople);
        }
    }

    private static int calculateMaxWealthyPeople(int[] savings, int x) {
        long totalSavings = 0;
        for (int saving : savings) {
            totalSavings += saving;
        }

        if (totalSavings < (long) x * savings.length) {
            return 0;
        } else {
            return (int) Math.min(totalSavings / x, savings.length);
        }
    }
}
