import java.util.Scanner;

public class DiverseStrings {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        for (int i = 0; i < n; i++) {
            String str = scanner.nextLine();
            if (isDiverse(str)) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }
    }

    private static boolean isDiverse(String str) {
        boolean[] seen = new boolean[26]; // Use a boolean array to track seen letters

        for (char c : str.toCharArray()) {
            int index = c - 'a';
            if (seen[index]) {
                return false; // Duplicate letter found
            }
            seen[index] = true;

            // Check if the current letter is adjacent to the previous letter
            if (index > 0 && !seen[index - 1]) {
                return false; // Non-consecutive letters
            }
        }

        return true; // All letters are distinct and consecutive
    }
}
