import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    long n = scanner.nextLong();
    long m = scanner.nextLong();
    long k = scanner.nextInt();
    long[] arr = new long[(int) m];
    for (int i = 0; i < m; i++) {
      arr[i] = scanner.nextInt();
    }
    System.out.println(solve(n, m, k, arr));
  }

  static long solve(long n, long m, long k, long[] arr) {
    Arrays.sort(arr);
    long l = 0;
    long r = (n + k - 1) / k;
    long mid = 0;
    long ans = r;
    while (l <= r) {
      mid = (l + r) / 2;
      long cnt = mid;
      for (long x : arr) {
        long p = (x - 1) / k;
        if (p >= mid) {
          cnt--;
        }
      }
      if (cnt >= 1) {
        ans = mid;
        r = mid - 1;
      } else {
        l = mid + 1;
      }
    }
    return ans;
  }
}
