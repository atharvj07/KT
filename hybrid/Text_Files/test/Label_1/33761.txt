import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        int k = sc.nextInt();
        char[] a = sc.next().toCharArray();
        Arrays.sort(a);
        int i = n - 1;
        int[] pos = new int[n];
        int[] len = new int[n];
        pos[n - 1] = 0;
        len[n - 1] = l;
        for (int j = n - 2; j >= 0; j--) {
            if (j + 1 < k) {
                len[j] = 1;
            } else if (j + 1 == k) {
                len[j] = Math.max(1, l - len[j + 1]);
            } else {
                len[j] = l - len[j + 1] - 1;
            }
            pos[j] = pos[j + 1] + len[j + 1];
        }
        for (int j = 0; j < n; j++) {
            for (int ind = 0; ind < len[j]; ind++) {
                System.out.print(a[pos[j] + ind]);
            }
            System.out.println();
        }
    }
}
