#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
vector<int> divisor(ll n) {
    vector<int> ans;
    for (ll i = 1; i * i <= n; ++i) {
        if (n % i == 0) {
            ans.push_back(i);
            if (n / i != i) { ans.push_back(n / i); }
        }
    }
    return ans;
}
pair<ll, ll> solve(ll n) {
    auto divs = divisor(n);
    sort(divs.begin(), divs.end());
    if (divs.size() % 2 == 0) {
        ll cur = divs[divs.size() / 2 - 1];
        ll prv = divs[divs.size() / 2 - 2];
        return {prv, cur};
    } else {
        ll cur = divs[(divs.size() - 1) / 2];
        return {cur, cur};
    }
}
int main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        auto ans = solve(n);
        cout << ans.second << endl;
    }
}
