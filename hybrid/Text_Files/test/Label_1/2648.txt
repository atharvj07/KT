#include <iostream>
#include <unordered_map>
#include <vector>

using namespace std;

int main() {
  unordered_map<char, int> suits;
  vector<int> numbers(10);
  char s[3];
  for (int i = 0; i < 3; ++i) {
    cin >> s;
    ++suits[s[1]];
    ++numbers[s[0] - '0'];
  }
  int ans = 0;
  for (auto &p : suits) {
    if (p.second == 1) ans += 2;
  }
  for (int i = 1; i < 10; ++i) {
    if (numbers[i] == 1) ans += 2;
  }
  cout << ans << endl;
}
