#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, k, x;
    cin >> n >> m;
    vector<bool> rick(n + 1), morty(n + 1);
    bool f = false;
    while (m--) {
        cin >> k;
        while (k--) {
            cin >> x;
            x = abs(x);
            if (x > n)
                continue;
            if (x <= 0 || (x <= n && rick[x] && morty[x]))
                continue;
            if (x > 0)
                morty[x] = true;
            else
                rick[abs(x)] = true;
            f = true;
        }
        if (!f) return cout << "NO", 0;
        f = false;
    }
    cout << "YES";
}
