import java.util.*;
import java.io.*;

public class Main {
    static int[] color, dist;
    static boolean[] vis;
    static List<Integer>[] adj;
    static Queue<Integer> q;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] in = br.readLine().split(" ");
        int n = Integer.parseInt(in[0]), m = Integer.parseInt(in[1]);
        color = new int[n + 1];
        dist = new int[n + 1];
        vis = new boolean[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            in = br.readLine().split(" ");
            int u = Integer.parseInt(in[0]), v = Integer.parseInt(in[1]);
            adj[u].add(v);
            adj[v].add(u);
        }
        bfs(1);
        while (m-- > 0) {
            in = br.readLine().split(" ");
            int t = Integer.parseInt(in[0]), v = Integer.parseInt(in[1]);
            if (t == 1) {
                color[v] = 1;
                bfs(v);
            } else {
                System.out.println(dist[v]);
            }
        }
    }

    static void bfs(int s) {
        q = new LinkedList<>();
        q.add(s);
        vis[s] = true;
        dist[s] = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            for (int v : adj[u]) {
                if (!vis[v]) {
                    q.add(v);
                    vis[v] = true;
                    if (color[u] == 0) {
                        color[v] = 1;
                        dist[v] = dist[u] + 1;
                    } else {
                        color[v] = 0;
                        dist[v] = dist[u] - 1;
                    }
                }
            }
        }
    }
}
