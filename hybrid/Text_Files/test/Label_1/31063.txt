import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), x = in.nextInt();
        long max_b = 0;
        for (int b = 2; b < x; b++) {
            if (check(n, b)) {
                max_b = b;
            }
        }
        System.out.println(max_b > 0 ? max_b : -1);
    }

    static boolean check(int n, int b) {
        long num = 0;
        for (int i = n - 1; i >= 0; i--) {
            num *= b;
            num += 1;
        }
        for (int i = 2; i <= n; i++) {
            num *= b;
            String str = String.valueOf(num);
            if (str.length() > n) {
                return false;
            }
            for (int j = 0; j < n - str.length(); j++) {
                str = "0" + str;
            }
            StringBuilder sb = new StringBuilder();
            for (int j = n - i; j < n; j++) {
                sb.append(str.charAt(j));
            }
            for (int j = 0; j < n - i; j++) {
                sb.append(str.charAt(j));
            }
            if (!str.equals(sb.toString())) {
                return false;
            }
        }
        return true;
    }
}
