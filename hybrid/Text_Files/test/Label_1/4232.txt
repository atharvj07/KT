#include <bits/stdc++.h>
using namespace std;
typedef array<int,2> element;
bool operator < (const element& l, const element& r) {return l[1] < r[1];}
int main() {
    int n, m; cin >> n;
    vector<element> a(n); for (auto&e:a) cin >> e[0] >> e[1];
    cin >> m;
    vector<element> b(m); for (auto&e:b) cin >> e[0] >> e[1];
    sort(a.begin(), a.end());
    sort(b.begin(), b.end());
    element x = {0, 0LL}, y={0,0LL};
    long long ans = 0;
    for (int i = n-1; i >= 0; i--) {
        if (a[i][1] <= x[1] && a[i][1] <= y[1]) continue;
        x = a[i];
        ans += a[i][1];
    }
    for (int i = m-1; i >= 0; i--) {
        if (b[i][1] <= x[1] && b[i][1] <= y[1]) continue;
        y = b[i];
        ans += b[i][1];
    }
    cout << ans << endl;
}
