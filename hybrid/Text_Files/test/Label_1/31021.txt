import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int[] r = new int[n];
        for (int i = 0; i < n; i++)
            if (a[a[i] - 1] == 0 || a[a[i] - 1] - 1 > i) r[i] += a[i] - 1;
            else if (a[i] > a[a[i] - 1]) r[i] += a[i] - a[a[i] - 1] - 1;
        for (int i = 1; i < n; i++)
            r[i] = Math.min(r[i], r[i - 1]);
        for (int x : r)
            System.out.print(x + " ");
        System.out.println();
    }
}
