const int mod = 998244353;
ll binpow(ll a, ll b) {
    ll res = 1;
    while (b > 0) {
        if (b & 1)
            res *= a, res %= mod;
        a *= a, a %= mod;
        b >>= 1;
    }
    return res;
}
int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    for (ll i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] == -1) { a[i] = k + 1; }
    }
    ll dp[n + 1], pref[n + 1];
    pref[0] = a[0];
    for (ll i = 1; i <= n; i++) {
        dp[i] = dp[i - 1] * k;
        dp[i] %= mod;
        pref[i] = pref[i - 1] + a[i];
        pref[i] %= (k + 1);
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
        for (ll j = i + 1; j < n; j++) {
            if (((pref[j] - pref[i] + (k + 1)) % (k + 1)) == 0) {
                ans += dp[i] * dp[n - j - 1];
                ans %= mod;
            }
        }
    }
    cout << ans << "\n";
    return 0;
}
