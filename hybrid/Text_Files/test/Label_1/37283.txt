from collections import deque

n, q = map(int, input().split())
a = deque(map(int, input().split()))
q = deque(map(int, [input() for _ in range(q)]))

while q:
    m = q.popleft()
    if m <= len(a):
        l1, l2 = a[m - 1], a[-m] if len(a) > m else 0
        if l1 > l2:
            a.rotate(m)
        else:
            a.rotate(-m)
        print(a[0], a[-1])
