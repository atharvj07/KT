def main():
    n, k = map(int, input().split())
    a = []
    for i in range(n):
        a.append(int(input(), 2))
    if sum(a) + k <= n * 7:
        ans = []
        for i in range(n):
            tmp = a[i] | (1 << 7) - 1
            if tmp < 256:
                ans.append('%o' % tmp)
            else:
                ans.append('7')
        print(''.join(ans))
    else:
        print(-1)

main()
