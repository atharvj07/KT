from collections import defaultdict

def solve_dinner_problem(n, m, w, friends):
    food_count = defaultdict(int)
    for i in range(n):
        food_count[i + 1] = w[i]
    
    graph = defaultdict(list)
    degrees = [0] * (m + 1)
    
    for idx, (x, y) in enumerate(friends, 1):
        graph[x].append(idx)
        graph[y].append(idx)
        degrees[idx] = 2
    
    order = []
    queue = [i for i in range(1, m + 1) if degrees[i] == 1]
    
    while queue:
        friend = queue.pop()
        order.append(friend)
        for food in friends[friend - 1]:
            if food_count[food] > 0:
                food_count[food] -= 1
                break
        else:
            return "DEAD", []
        
        for neighbor in graph[food]:
            if degrees[neighbor] > 0:
                degrees[neighbor] -= 1
                if degrees[neighbor] == 1:
                    queue.append(neighbor)
    
    if len(order) != m:
        return "DEAD", []
    
    return "ALIVE", order[::-1]

n, m = map(int, input().split())
w = list(map(int, input().split()))
friends = [tuple(map(int, input().split())) for _ in range(m)]

status, order = solve_dinner_problem(n, m, w, friends)
print(status)
if status == "ALIVE":
    print(*order)
