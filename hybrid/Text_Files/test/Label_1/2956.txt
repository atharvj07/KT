#include <bits/stdc++.h>

using namespace std;

const int MM = 2e2 + 5;
int n, m;
vector<int> row(MM), col(MM);

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            int x;
            cin >> x;
            row[i] += x;
            col[j] += x;
        }
    }
    int mx = 0;
    for (int i = 1; i <= n; i++) { mx = max(mx, row[i] + col[i]); }
    if ((m - mx) % 2) {
        cout << "NO\n";
    } else {
        cout << "YES\n";
        for (int i = 1; i <= n; i++) {
            if (row[i] + col[i] != mx) row[i] = 1;
            cout << (row[i] ? '1' : '0');
        }
        cout << '\n';
        for (int j = 1; j <= m; j++) {
            cout << (col[j] == mx / 2 ? '1' : '0');
        }
        cout << '\n';
    }
}
