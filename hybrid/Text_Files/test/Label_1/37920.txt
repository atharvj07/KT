import sys
n = int(input())
L = list(map(int, input().split()))
i = 0
while i < n - 1:
   j = i + 1
   while j < n and L[j] >= L[j - 1]:
      j += 1
   if j == n:
      break
   k = j + 1
   while k < n and L[k] <= L[j - 1]:
      k += 1
   if k == n:
      break
   L[i], L[j - 1] = L[j - 1], L[i]
   L[j], L[k - 1] = L[k - 1], L[j]
   print(i + 1, k)
   i += 1
