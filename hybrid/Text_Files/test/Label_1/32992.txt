import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int q = sc.nextInt();
        int r = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int max = -1000000000;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int sum = p * a[i] + q * a[j];
                for (int k = j; k < n; k++) {
                    sum += r * a[k];
                    max = Math.max(max, sum);
                }
            }
        }

        System.out.println(max);
    }
}
