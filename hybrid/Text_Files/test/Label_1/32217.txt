import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        int[] a = new int[n];
        String s = sc.nextLine();
        for (int i = 0; i < n; i++) {
            a[i] = Character.getNumericValue(s.charAt(i));
        }
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = a[i];
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < n - i; j++) {
                x[j] = Math.abs(x[j] - x[j + 1]);
            }
        }
        System.out.println(x[0]);
    }
}
