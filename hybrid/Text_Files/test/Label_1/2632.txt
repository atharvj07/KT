#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> ii;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n, m;
        cin >> n >> m;

        int ans = 0;
        int nega = 0;
        for (int i = 0; i < n; ++i) {
            int curnega = 0;
            for (int j = 0; j < m; ++j) {
                int x;
                cin >> x;
                ans += abs(x);
                curnega += (x < 0);
            }
            nega += (curnega % 2 == 1);
        }

        if (nega % 2 == 1)
            ans -= 2 * min(nega, n * m - nega);

        cout << ans << endl;
    }

    return 0;
}
