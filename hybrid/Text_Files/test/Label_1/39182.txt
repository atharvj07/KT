def count_m(n, t):
    if t == 1:
        return n
    if t <= 2 or t % 2 == 1:
        return 0
    ans, cnt = 0, 0
    while t % 2 == 0:
        t //= 2
        cnt += 1
    ans = 2 ** cnt
    if ans <= n + 1:
        return min(n - ans + 2, ans // 2)
    return 0

n, t = map(int, input().split())
print(count_m(n, t))
