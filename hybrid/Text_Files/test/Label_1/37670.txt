n,m,k=map(int,input().split())
a=[int(i) for i in input().split()]
b=[int(i) for i in input().split()]
ans=0
for i in range(n):
    aa=pa=0
    for j in range(m):
        if a[i]==0:aa=pa=0
        else:
            if b[j]==0:pa=0
            else:
                pa+=1
            aa+=pa
        if aa==k:ans+=1
print(ans)
