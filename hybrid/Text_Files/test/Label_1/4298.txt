#include <bits/stdc++.h>
using namespace std;

const int MN = 1e5 + 5;

int main() {
    int n, k, a[MN], b[MN]; cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> a[i];
    for (int i = 0; i < n; ++i) cin >> b[i];
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; ++i) v[i] = {b[i], a[i]};
    sort(v.begin(), v.end());
    int ans = 0;
    int freqs[k + 5]; memset(freqs, 0, sizeof(freqs));
    bool flag = true;
    for (auto i : v) {
        if (freqs[i.second]++ == 0 && flag) ans += i.first;
        else if (flag) flag = false;
    }
    cout << ans << "\n";
    return 0;
}
