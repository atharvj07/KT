def solve(rebus):
    nums = [int(x) for x in rebus.split() if x.isdigit()]
    ops = [x for x in rebus.split() if not x.isdigit()]
    if len(nums) != len(ops) + 1:
        return "Impossible"
    n = nums[0]
    for i in range(1, len(nums)):
        if ops[i - 1] == "+":
            n += nums[i]
        else:
            n -= nums[i]
    if n == 0:
        return "Possible\n" + rebus.replace("?", "1")
    else:
        return "Impossible"

print(solve(input()))
