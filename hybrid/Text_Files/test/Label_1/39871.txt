def modpow(x: int, y: int, m: int):
    ret = 1

    while(y > 0):
        if(y % 2 == 0):
            x = x * x
            x %= m
            y /= 2
        else:
            ret = ret * x
            ret %= m
            y -= 1

    return ret

q = int(input())
mod = 1000000007
dp = {}

for i in range(q):
    n, f = map(int, input().split())
    if((n, f) in dp):
        print(dp[(n, f)])
    else:
        x = f - 1
        y = n - f
        num = modpow(x, y, mod)
        den = 1
        for i in range(1, f):
            den = den * i
            den %= mod
        den = modpow(den, mod - 2, mod)
        ans = num * den
        ans %= mod
        dp[(n, f)] = ans
        print(ans)
