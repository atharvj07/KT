import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        List<Integer>[] adj = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            adj[x].add(y);
            adj[y].add(x);
        }
        int[] deg = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            deg[i] = adj[i].size();
        }
        int[] max = new int[m + 1];
        for (int i = 1; i <= m; i++) {
            PriorityQueue<Integer> pq = new PriorityQueue<>((a, b) -> deg[b] - deg[a]);
            boolean[] vis = new boolean[n + 1];
            int cnt = 0;
            for (int j = 1; j <= n; j++) {
                if (!vis[j] && deg[j] >= k) {
                    pq.add(j);
                }
            }
            while (!pq.isEmpty()) {
                int u = pq.poll();
                if (vis[u]) {
                    continue;
                }
                cnt++;
                vis[u] = true;
                for (int v : adj[u]) {
                    deg[v]--;
                    if (!vis[v] && deg[v] >= k) {
                        pq.add(v);
                    }
                }
            }
            max[i] = cnt;
        }
        for (int i = 1; i <= m; i++) {
            System.out.println(max[i]);
        }
    }
}
