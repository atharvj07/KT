#include <bits/stdc++.h>
using namespace std;
int n,a[5001],ans;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    bool p;
    while(1){
        p=false;
        for(int i=1;i<=n;i++){
            if(a[i]>0){
                p=true;
                ans++;
                for(int j=i;j<=n;j++){
                    a[j]=a[j]-1;
                    if(a[j]<0) break;
                }                
                break;
            }
        }
        if(!p) break;
    }
    cout<<ans<<endl;
}
