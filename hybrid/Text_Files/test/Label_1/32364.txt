import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] arrivalTimes = new int[n];

        for (int i = 0; i < n; i++) {
            int hours = scanner.nextInt();
            int minutes = scanner.nextInt();
            arrivalTimes[i] = hours * 60 + minutes;
        }

        Arrays.sort(arrivalTimes);

        int maxConcurrentCustomers = 0;
        int currentNumberOfCashes = 0;

        for (int arrivalTime : arrivalTimes) {
            if (currentNumberOfCashes < arrivalTime) {
                currentNumberOfCashes++;
            }
            maxConcurrentCustomers = Math.max(maxConcurrentCustomers, currentNumberOfCashes);
        }

        System.out.println(maxConcurrentCustomers);
    }
}
