import java.util.*;
import java.io.*;

class Main {
  private static int dfs(int[] a, Map<Integer, List<int[]>> graph, int[] dp, int node) {
    if (dp[node] != -1)
      return dp[node];
    dp[node] = a[node];
    for (int[] edge : graph.getOrDefault(node, List.of()))
      dp[node] = Math.max(dp[node], dfs(a, graph, dp, edge[0]) + edge[1] + a[node]);
    return dp[node];
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt(), k = in.nextInt();
    int[] a = new int[n + 1];
    Map<Integer, List<int[]>> graph = new HashMap<>();
    for (int i = 1; i <= n; i++) {
      a[i] = in.nextInt();
    }
    while (k-- > 0) {
      int x = in.nextInt(), y = in.nextInt(), c = in.nextInt();
      graph.computeIfAbsent(x, l -> new ArrayList<>()).add(new int[] { y, c });
    }
    int[] dp = new int[n + 1];
    Arrays.fill(dp, -1);
    int max = 0;
    for (int i = 1; i <= n; i++)
      max = Math.max(max, dfs(a, graph, dp, i));
    System.out.println(max);
  }
}
