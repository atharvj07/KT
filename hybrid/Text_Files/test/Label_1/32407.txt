import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        ArrayList<Pair> trees = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            trees.add(new Pair(input.nextInt(), input.nextInt()));
        }
        Collections.sort(trees, (a, b) -> a.x - b.x);
        int count = 0;
        int left = Integer.MIN_VALUE, right = Integer.MAX_VALUE;
        for (Pair tree : trees) {
            if (tree.x - tree.h >= left && tree.x + tree.h <= right) {
                count++;
                left = Math.max(left, tree.x - tree.h);
                right = Math.min(right, tree.x + tree.h);
            }
        }
        System.out.println(count);
    }

    static class Pair {
        int x, h;

        Pair(int x, int h) {
            this.x = x;
            this.h = h;
        }
    }
}
