#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x, ans = 0;
        cin >> n >> x;
        vector<int> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        sort(a.begin(), a.end());
        for (int i = 0, j = n - 1; i < j;) {
            if (a[i] * (j - i + 1) >= x) {
                j--, ans++;
            } else {
                i++;
            }
        }
        cout << ans << endl;
    }
}
