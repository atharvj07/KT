import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.nextLine();
    String t = in.nextLine();

    int[] freq = new int[26];
    for (char c : t.toCharArray()) {
      freq[c - 'a']++;
    }

    for (char c : s.toCharArray()) {
      if (freq[c - 'a'] == 0) {
        System.out.println("No");
        return;
      }
      freq[c - 'a']--;
    }

    System.out.println("Yes");
  }
}
