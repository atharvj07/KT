#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int odd = 0, even = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      if (a[i] % 2 == 0)
        even++;
      else
        odd++;
    }
    if (even == 0 || odd == 0)
      cout << 0 << "\n";
    else if (even % 2 == 0 || odd % 2 == 0)
      cout << 1 << "\n";
    else
      cout << "-1\n";
  }
  return 0;
}
