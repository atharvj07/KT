import java.util.Scanner;
import java.util.HashMap;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    HashMap<Long, Integer> map = new HashMap<>();
    for (int i = 0; i < n; i++) {
      long si = sc.nextLong();
      long root = (long)Math.cbrt(si);
      if (root * root * root == si) {
        map.remove(root);
      } else {
        map.put(si, map.getOrDefault(si, 0) + 1);
      }
    }
    int max = 0;
    for (long si : map.keySet()) {
      max += map.get(si) > map.getOrDefault(si * si, 0) ? map.get(si) : map.get(si * si);
    }
    System.out.println(max);
  }
}
