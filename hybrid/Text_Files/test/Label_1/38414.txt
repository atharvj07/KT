import sys

n, t, k = map(int, input().split())
a = list(map(int, input().split()))

if sum(a) + k < n - 1 or sum(a) > n - 1 or (t == 1 and n > 3):
    print(-1)
    sys.exit()

ans = []
dist = [0 for i in range(n + 1)]
for i in range(1, t + 1):
    for j in range(a[i - 1]):
        ans.append((1, i + 1))
        dist[i + 1] += 1
dist[1] = 0
for i in range(2, n + 1):
    if dist[i] == 0 and k > 0:
        ans.append((1, i))
        k -= 1

for i in range(2, n + 1):
    if dist[i] == 1:
        for j in range(2, n + 1):
            if dist[j] == 0:
                ans.append((i, j))
                dist[j] += 1
                break

print(n)
for i in ans:
    print(*i)
