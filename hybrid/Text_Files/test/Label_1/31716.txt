import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class GeoRepainting {

    private static final int[][] DIRECTIONS = {{-1, 0}, {0, 1}, {1, 0}, {0, -1}};
    private static final int TRANSPARENT_COLOR = 0;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] map = new int[n][m];
        int[][] symbols = new int[n][m];

        // Read map and symbols
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                map[i][j] = scanner.nextInt();
                symbols[i][j] = scanner.nextInt();
            }
        }

        // Get starting point
        int x = scanner.nextInt() - 1;
        int y = scanner.nextInt() - 1;

        // Start simulation
        int repaintings = repaint(map, symbols, x, y);

        // Print result
        System.out.println(repaintings);
    }

    private static int repaint(int[][] map, int[][] symbols, int x, int y) {
        Queue<Integer> queue = new LinkedList<>();
        queue.offer(symbols[x][y]);
        symbols[x][y] = -1; // Remove starting symbol

        int repaintings = 0;

        while (!queue.isEmpty()) {
            int symbolColor = queue.poll();
            int cellColor = map[x][y];

            if (cellColor != TRANSPARENT_COLOR && cellColor != symbolColor) {
                repaintings += repaint(map, symbols, x, y, cellColor);
            }

            for (int[] direction : DIRECTIONS) {
                int newX = x + direction[0];
                int newY = y + direction[1];

                if (newX >= 0 && newX < map.length && newY >= 0 && newY < map[0].length) {
                    if (symbols[newX][newY] == symbolColor) {
                        queue.offer(symbolColor);
                        symbols[newX][newY] = -1; // Remove symbol
                    }
                }
            }
        }

        return repaintings;
    }

    private static int repaint(int[][] map, int[][] symbols, int x, int y, int targetColor) {
        int repaintings = 0;
        int spiralIndex = 1;

        while (true) {
            int spiralX = x + spiralIndex * (spiralIndex % 2 == 0 ? 1 : -1);
            int spiralY = y + spiralIndex * (spiralIndex % 2 == 0 ? -1 : 1);

            if (spiralX >= 0 && spiralX < map.length && spiralY >= 0 && spiralY < map[0].length) {
                if (map[spiralX][spiralY] == targetColor) {
                    map[spiralX][spiralY] = symbols[x][y];
                    repaintings++;
                }
            } else {
                break;
            }

            spiralIndex++;
        }

        return repaintings;
    }
}
