#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, a, x, b, y;
  cin >> n >> a >> x >> b >> y;
  bool v = true;
  while (v) {
    if (a == b) {
      cout << "YES" << endl;
      return 0;
    }
    a = a < x ? a + 1 : 1;
    b = b > y ? b - 1 : n;
    v = (a != x && b != y);
  }
  cout << "NO" << endl;
}
