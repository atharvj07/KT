import java.util.Scanner;

public class TreeProblem {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        System.out.println(isTreePossible(n, c) ? "YES" : "NO");
    }

    private static boolean isTreePossible(int n, int[] c) {
        int[] d = new int[n + 1];
        for (int x : c) {
            d[x]++;
        }

        for (int i = 2; i <= n; i++) {
            if (d[i] == 0) {
                return false;
            }
            d[i - 1] -= d[i];
        }

        return true;
    }
}
