import java.util.Scanner;

public class Main {
    static Scanner scan = new Scanner(System.in);

    public static void main(String[] args) {
        int r = scan.nextInt();
        int g = scan.nextInt();
        int b = scan.nextInt();
        int n = scan.nextInt();
        System.out.println(solve(r, g, b, n));
    }

    static long solve(int r, int g, int b, int n) {
        long ans = 0;
        for (int i = 0; i <= n / r; i++) {
            for (int j = 0; j <= (n - r * i) / g; j++) {
                int k = (n - r * i - g * j) / b;
                if (k >= 0)
                    ans++;
            }
        }
        return ans;
    }
}
