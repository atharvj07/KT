#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define ull unsigned long long
#define pb push_back
#define pf push_front
#define mp make_pair
#define F first
#define S second
#define mod 1000000007
#define inf 1000000000000000007ll
#define all(c) c.begin(),c.end()
#define sz(a) a.size()
#define nl "\n"

int n;
int a[100005];
int freq[10];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        freq[a[i]]++;
    }
    int mx = 0, sum = 0;
    for (int i = 1; i < 9; i++) mx = max(mx, freq[i]);
    for (int i = 1; i < 9; i++) {
        if (freq[i] != 0) sum++;
    }
    if (sum < 3) cout << n << nl;
    else cout << mx * 2 << nl;
    return 0;
}
