import java.util.Scanner;

public class Main {
    static int[] color;
    static int[] subtree;
    static int[][] graph;
    static int n;
    static boolean[] vis;
    static int mod = 1000000007;

    static void dfs(int u, int p) {
        vis[u] = true;
        subtree[u] = 1;
        for (int v : graph[u]) {
            if (v == p)
                continue;
            if (!vis[v]) {
                dfs(v, u);
                subtree[u] += subtree[v];
            }
        }
    }

    static int solve(int u, int p) {
        int ans = 1000000000;
        int white = 0;
        int black = 0;
        int gray = 0;
        for (int v : graph[u]) {
            if (v == p)
                continue;
            if (!vis[v]) {
                int res = solve(v, u);
                if (res == -1)
                    return -1;
                ans = Math.min(ans, res);
            } else {
                if (color[v] == 1)
                    white++;
                else if (color[v] == 2)
                    black++;
                else
                    gray++;
            }
        }

        if (color[u] == 0) {
            gray++;
            if (white > 0)
                return -1;
            else if (black > 0)
                return 1;
            else
                return 0;
        }

        if (color[u] == 1) {
            white++;
            ans = Math.min(ans, gray + black);
            if (black > 0)
                return -1;
            else if (gray == 0)
                return 1;
            else
                return 2;
        }

        if (color[u] == 2) {
            black++;
            ans = Math.min(ans, gray + white);
            if (white > 0)
                return -1;
            else if (gray == 0)
                return 1;
            else
                return 2;
        }

        ans = Math.min(ans, white + black + gray);
        if (ans == 1000000000)
            return -1;
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            n = sc.nextInt();
            color = new int[n + 1];
            graph = new int[n + 1][];
            for (int i = 0; i <= n; i++)
                graph[i] = new int[0];
            subtree = new int[n + 1];
            vis = new boolean[n + 1];
            for (int i = 1; i <= n; i++)
                color[i] = sc.nextInt();
            for (int i = 1; i < n; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                graph[u] = extend(graph[u], v);
                graph[v] = extend(graph[v], u);
            }
            dfs(1, 1);
            int ans = solve(1, 1);
            if (ans == -1)
                System.out.println(-1);
            else
                System.out.println(ans);
        }
    }

    static int[] extend(int[] a, int x) {
        int[] ret = new int[a.length + 1];
        for (int i = 0; i < a.length; i++)
            ret[i] = a[i];
        ret[a.length] = x;
        return ret;
    }
}
