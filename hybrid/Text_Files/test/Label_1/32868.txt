import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        int cnt = 0;
        for (char c : s.toCharArray()) {
            if (c == 'N') cnt++;
            else cnt = 0;
            if (cnt >= k) break;
        }
        System.out.println(cnt >= k ? "YES" : "NO");
    }
}
