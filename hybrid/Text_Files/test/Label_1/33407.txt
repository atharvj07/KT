import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt();
        int[] c = new int[n];
        long sum = 0L;
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
            sum += (long) c[i] * x;
            x = Math.max(x - 1, 1);
        }
        System.out.println(sum);
    }
}
