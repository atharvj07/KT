n, *a = map(int, input().split())
ans = 1
mx = [0]*9
mn = [9]*9
for i in a:
    mx[i] = mx[i] + 1
    mn[i] = min(mn[i], mx[i])
mn[0] = 9
cnt = 0
for i in range(1, 9):
    if mx[i] - mn[i] <= 1:
        cnt = cnt + 1
if cnt == 8:
    ans = n
else:
    for i in range(1, 9):
        if mx[i] == mn[i]:
            ans = ans + 1
        else:
            ans = ans + mx[i] - mn[i]
print(ans)
