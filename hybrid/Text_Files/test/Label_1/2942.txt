#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
const int maxn = 3e5 + 10;
int n, a[maxn], b[65];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 61; j++) {
            if ((a[i] >> j) & 1) { b[j]++; }
        }
    }
    long long ans = 0;
    for (int j = 0; j < 61; j++) {
        ans += 1ll * b[j] * (b[j] - 1) / 2;
    }
    cout << ans << "\n";
}
