#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;
  std::vector<long long> t(n);
  for (int i = 0; i < n; i++) {
    std::cin >> t[i];
  }
  std::sort(t.begin(), t.end());
  long long sum = 0;
  int res = 0;
  for (int i = 0; i < n; i++) {
    sum += t[i];
    if (sum <= t[i]) {
      res++;
    } else {
      break;
    }
  }
  std::cout << res << std::endl;
  return 0;
}
