def solve(s):
    count_0 = count_1 = 0
    for char in s:
        if char == '0':
            count_0 += 1
        else:
            count_1 += 1
    if count_0 == count_1:
        return ''.join(sorted(s))
    elif count_0 > count_1:
        return '0' * count_0 + '1' * count_1
    else:
        return '1' * count_1 + '0' * count_0

T = int(input().strip())
for _ in range(T):
    s = input().strip()
    print(solve(s))
