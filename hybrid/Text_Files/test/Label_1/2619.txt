#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1e5+5;
ll fact[N], ifact[N];
ll pw(ll a, ll b, ll mod){
    if(b==0)return 1;
    if(b==1)return a;
    ll val=pw(a,b/2,mod);
    val*=val;
    val%=mod;
    if(b%2)val*=a,val%=mod;
    return val;
}
ll inv(ll x, ll mod){
    return pw(x,mod-2,mod);
}
void preprocess(ll n, ll mod){
    fact[0] = 1;
    for(int i = 1; i <= n; i++){
        fact[i] = (fact[i-1] * i) % mod;
    }
    ifact[n] = inv(fact[n], mod);
    for(int i = n-1; i >= 0; i--){
        ifact[i] = (ifact[i+1] * (i+1)) % mod;
    }
}
ll ncr(ll n, ll r, ll mod){
    return (fact[n] * ((ifact[n-r] * ifact[r]) % mod)) % mod;
}
ll comb(ll n, ll k, ll mod){
    return (fact[n] * ((ifact[n-k] * ifact[k]) % mod)) % mod;
}
ll catalan(ll n, ll mod){
    return comb(2*n, n, mod)/((n+1)%mod);
}
int32_t main(){
    ll n;
    cin>>n;
    if(n%2==1){
        cout<<"NO\n";
        return 0;
    }
    preprocess(n, n);
    cout<<"YES\n";
    for(ll i = 1; i <= n; i++){
        for(ll j = 1; j <= n; j++){
            if(i*catalan(j,n)%n==j-1){
                cout<<i<<"\n";
                break;
            }
        }
    }
    return 0;
}
