#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int sumOdd = 0, sumEven = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] % 2 == 0) { sumEven += a[i]; } else { sumOdd += a[i]; }
    }
    cout << min(sumEven, sumOdd) << endl;
    return 0;
}
