n=int(input())
a=list(map(int,input().split()))
b=a.copy()
for i in range(1,n+1):
    cnt=a.count(i)
    if cnt==0:
        b[i-1]=i
    else:
        while cnt>0:
            if i in a:
                a.remove(i)
            i+=1
            cnt-=1
        b[i-1]=i
if min(b)>=0:
    print(*b)
else:
    print(-1)
