import sys, math
readline = sys.stdin.readline

n, m = map(int,readline().split())
e = [[] for _ in range(n+1)]
for _ in range(m):
  x, y = map(int,readline().split())
  e[x].append(y)
  e[y].append(x)
ans = [-1]*(n+1)

same = [False]*(n+1)
ne = []
for i in range(1,n+1):
  if not same[i]:
    t = [i]
    same[i] = True
    for j in e[i]:
      if not same[j]:
        same[j] = True
        t.append(j)
    ne.append(t)

if len(ne)!=3:
  print(-1)
  sys.exit()
for k in range(3):
  for i in ne[k]:
    ans[i] = k+1

print(*ans[1:])
