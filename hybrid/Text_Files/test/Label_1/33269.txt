import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        List<Integer> a = new ArrayList<>();
        List<Integer> w = new ArrayList<>();
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a.add(Integer.parseInt(st.nextToken()));
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            w.add(Integer.parseInt(st.nextToken()));
        }
        SegmentTree tree = new SegmentTree(a, w, n);
        
        while (q-- > 0) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            if (x < 0) {
                tree.update(-x, y);
            } else {
                System.out.println(tree.query(x, y));
            }
        }
    }

    private static class SegmentTree {
        private int[] tree;
        private int[] weights;
        private int size;

        public SegmentTree(List<Integer> a, List<Integer> w, int n) {
            size = n;
            tree = new int[4 * n];
            weights = new int[4 * n];
            build(a, w, 1, 0, n - 1);
        }

        private void build(List<Integer> a, List<Integer> w, int node, int start, int end) {
            if (start == end) {
                tree[node] = a.get(start);
                weights[node] = w.get(start);
                return;
            }
            int mid = (start + end) / 2;
            build(a, w, 2 * node, start, mid);
            build(a, w, 2 * node + 1, mid + 1, end);
            tree[node] = tree[2 * node];
            weights[node] = weights[2 * node];
        }

        public int query(int l, int r) {
            return query(1, 0, size - 1, l, r);
        }

        private int query(int node, int start, int end, int l, int r) {
            if (r < start || end < l) {
                return 0;
            }
            if (l <= start && end <= r) {
                return tree[node];
            }
            int mid = (start + end) / 2;
            int left = query(2 * node, start, mid, l, r);
            int right = query(2 * node + 1, mid + 1, end, l, r);
            return Math.abs(tree[node] - left) * weights[node] + query(2 * node + 1, mid + 1, end, l, r);
        }

        public void update(int id, int val) {
            update(1, 0, size - 1, id, val);
        }

        private void update(int node, int start, int end, int id, int val) {
            if (id < start || end < id) {
                return;
            }
            if (start == end) {
                weights[node] = val;
                return;
            }
            int mid = (start + end) / 2;
            update(2 * node, start, mid, id, val);
            update(2 * node + 1, mid + 1, end, id, val);
        }
    }
}
