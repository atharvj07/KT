#include<bits/stdc++.h>
using namespace std;
#define int long long
#define pb push_back
int32_t main(){
int t;cin>>t;
while(t--){
int n;cin>>n;
string s;cin>>s;
int mp[26]={0};
for(int i=0;i<n;i++)
mp[s[i]-'a']++;
int odd=0;
vector<int>v;
for(int i=0;i<26;i++){
if(mp[i]%2)
odd++;
if(mp[i]>0)
v.pb(mp[i]);
}
if(odd>1)
cout<<"-1"<<endl;
else if(odd==1){
int sum=0;
for(int val:v)
sum+=val-1;
cout<<sum<<endl;
}
else{
int sum=0;
for(int val:v)
sum+=val-2;
cout<<sum<<endl;
}
}
return 0;
}
