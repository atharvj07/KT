#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        vector<long long> a(n);
        vector<pair<int, int>> b;
        long long sum = 0;
        for (auto &ai : a) { cin >> ai; }
        for (int i = 0; i < n - 1; ++i) {
            if (a[i] > a[i + 1]) b.emplace_back(i, i + 1);
            sum += abs(a[i] - a[i + 1]);
        }
        sum += a[n - 1];
        sum &= 1;
        if (b.size() > 1 || (sum && b.size() > 0))
            cout << "NO" << endl;
        else
            cout << "YES" << endl;
    }
    return 0;
}
