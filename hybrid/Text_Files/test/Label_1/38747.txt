n = int(input())
hashtags = [input()[1:] for _ in range(n)]

for i in range(n - 2, -1, -1):
    # Compare current hashtag with the next one
    if hashtags[i] > hashtags[i + 1]:
        # Find the minimum length to trim
        for j in range(min(len(hashtags[i]), len(hashtags[i + 1]))):
            if hashtags[i][j] != hashtags[i + 1][j]:
                hashtags[i] = hashtags[i][:j]
                break
        else:
            if len(hashtags[i]) > len(hashtags[i + 1]):
                hashtags[i] = hashtags[i][:len(hashtags[i + 1])]

# Print the resulting hashtags
for hashtag in hashtags:
    print(f'#{hashtag}')
