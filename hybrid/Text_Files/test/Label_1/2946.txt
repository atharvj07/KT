#include <vector>
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, a, b;
        cin >> n >> a >> b;
        vector<int> p(n);
        int p1 = n, p2 = 1;
        bool found = false;
        for (int i = 0; i < n; ++i) {
            if (i % 2 == 0) {
                if (p1 == a && !found) {
                    p[i] = a;
                    found = true;
                    ++p2;
                } else {
                    p[i] = p2++;
                }
            } else {
                p[i] = --p1;
            }
        }
        if (p[n / 2 - 1] != a || p[n / 2] != b) {
            for (int i = 0; i < n; ++i) {
                if (p[i] == b) {
                    swap(p[i], p[n / 2]);
                    break;
                }
            }
        }
        if (p[n / 2 - 1] == a && p[n / 2] == b) {
            for (int i = 0; i < n; ++i) {
                cout << p[i] << ' ';
            }
            cout << endl;
        } else {
            cout << -1 << endl;
        }
    }
    return 0;
}
