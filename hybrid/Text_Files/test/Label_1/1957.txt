#include <iostream>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        long long x, y, a, b;
        std::cin >> x >> y >> a >> b;
        if ((y - x) % (a + b) != 0) {
            std::cout << -1 << '\n';
        } else {
            std::cout << (y - x) / (a + b) << '\n';
        }
    }
    return 0;
}
