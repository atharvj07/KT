import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt();
        long[] c = new long[n];
        for (int i = 0; i < n; i++) c[i] = sc.nextInt();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += Math.max(1, x);
            x = Math.max(1, x - 1);
            x -= c[i] - 1;
        }
        System.out.println(ans);
    }
}
