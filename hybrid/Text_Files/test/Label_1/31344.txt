import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a1 = in.nextInt();
        int a2 = in.nextInt();
        int a3 = in.nextInt();
        int a4 = in.nextInt();
        if (a1 > a3 + a2 || a2 > a3 + a4 || a3 > a1 + a2 || a4 > a1 + a2) {
            System.out.println("-1");
        } else {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < a1 - a3 - a4; i++) {
                sb.append("4");
            }
            for (int i = 0; i < a2 - a3 - a4; i++) {
                sb.append("7");
            }
            while (a1-- > 0) sb.append("4");
            while (a2-- > 0) sb.append("7");
            while (a3-- > 0) sb.append("47");
            while (a4-- > 0) sb.append("74");
            System.out.println(sb);
        }
    }
}
