import sys

t = int(input())
for i in range(t):
    n, m = map(int, input().split())
    deg = [0] * (n + 1)
    for j in range(m):
        u, v = map(int, input().split())
        deg[u] += 1
        deg[v] -= 1
    ans = 0
    for j in range(1, n + 1):
        if deg[j] == 0:
            ans += 1
    print(ans)
    for j in range(1, n + 1):
        for k in range(1, n + 1):
            if j != k and deg[j] > 0 and deg[k] < 0:
                print(j, k)
                deg[j] -= 1
                deg[k] += 1
