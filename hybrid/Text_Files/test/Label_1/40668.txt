import math

def main():
    a, b, c = map(int, input().split())
    x1, y1, x2, y2 = map(int, input().split())
    if a == 0 and b == 0:
        print(math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2))
        return
    if a == 0:
        if (y2 - y1) * b <= c:
            print(abs(y2 - y1))
            return
        else:
            print(math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2))
            return
    if b == 0:
        if (x2 - x1) * a <= c:
            print(abs(x2 - x1))
            return
        else:
            print(math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2))
            return
    
    # calculate the intersection point of the diagonal and the line connecting A and B
    x = (c + b * y1 - a * x1) / (a * b)
    y = (c + a * x1 - b * y1) / (a * b)
    
    # calculate the distance from A to the intersection point
    dist1 = math.sqrt((x - x1) ** 2 + (y - y1) ** 2)
    
    # calculate the distance from B to the intersection point
    dist2 = math.sqrt((x - x2) ** 2 + (y - y2) ** 2)
    
    # print the minimum of the two distances
    print(min(dist1 + dist2, abs(x2 - x1) + abs(y2 - y1)))

main()
