import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] values = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        System.out.println(solve(n, values) ? "Yes" : "No");
    }

    public static boolean solve(int n, int[] values) {
        List<Integer> left = new ArrayList<>(), right = new ArrayList<>();
        for (int i = 1; i < n; i++) {
            if (values[i] % values[i - 1] == 0) {
                return false;
            } else if (values[i] % values[i - 1] > 1) {
                left.add(values[i - 1]);
                right.add(values[i]);
            } else if (values[i - 1] % values[i] > 1) {
                left.add(values[i]);
                right.add(values[i - 1]);
            } else {
                return false;
            }
        }
        return true;
    }
}
