#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin>>n;
    long long a[n];
    for(int i=0; i<n; i++) {
        cin>>a[i];
    }
    for(int i=0; i<n; i++) {
        if(i) cout<<(a[i]-a[i-1])<<" ";
        else cout<<a[i+1]-a[i]<<" ";
        if(i==n-1) cout<<a[n-1]-a[n-2];
        else cout<<a[n-1]-a[i];
        cout<<endl;
    }
}
