from queue import PriorityQueue
t=int(input())
for tc in range(t):
    n=int(input())
    adj=[[] for i in range(2001)]
    for i in range(n):
        x,y=map(int,input().split())
        adj[x].append(y)
        adj[y].append(x)
    vis=[False]*2001
    q=PriorityQueue()
    ans=[]
    q.put((0,0,''))
    while not q.empty():
        d,nd,path=q.get()
        if vis[nd]==True: continue
        vis[nd]=True
        ans=path
        for nx in adj[nd]:
            if vis[nx]==False:
                q.put((d+1,nx,path+'U' if nd<nx else 'R'))
    if max(vis)!=True: print("NO")
    else: print("YES\n"+ans)
