#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int score = 0;
    int idx = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] > a[idx]) {
        idx = i;
        score = 0;
      }
      score += a[idx];
      idx += a[idx];
      idx %= n;
    }
    cout << score << endl;
  }
  return 0;
}
