import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = "ROYGBIV";
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = i % 7;
        }
        if (n % 2 == 0) {
            for (int i = 0; i < n / 2; i++) {
                int temp = a[i];
                a[i] = a[n - i - 1];
                a[n - i - 1] = temp;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(s.charAt(a[i]));
        }
        System.out.println();
    }
}
