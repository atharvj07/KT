from collections import deque

for _ in range(int(input())):
    s = deque(input())
    t = deque()
    while s:
        a = s.popleft()
        if len(t) >= 1 and a == t[-1]:
            continue
        t.append(a)
    if len(t) % 2 == 1:
        t[-1] = '0' if t[-1] == '1' else '1'
    print(''.join(t))
