t = int(input())
a = list(map(int, input().split()))

a.sort()
if a[0] != 0 or a.count(a[1]) != 4:
    print(-1)
else:
    n = a.count(1)
    m = (t-1) // n
    if n * m + 1 != t or any(a.count(i) != 2*(n-i)+2*(m-i) for i in range(1, min(n, m)+1)):
        print(-1)
    else:
        x = n // 2 + 1
        y = (a[1:].index(2) + 1) // 2 + 1
        print(n, m)
        print(x, y)
