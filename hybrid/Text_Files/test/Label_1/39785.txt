w, m, k = map(int, input().split())

sequence_length = 0

while w > 0:
    current_length = len(str(m))
    next_threshold = 10**current_length
    numbers_in_range = next_threshold - m
    cost = numbers_in_range * current_length * k
    
    if w >= cost:
        sequence_length += numbers_in_range
        w -= cost
        m = next_threshold
    else:
        additional_numbers = w // (current_length * k)
        sequence_length += additional_numbers
        break

print(sequence_length)
