import sys
import math
def solve():
    n = int(input())
    a = sorted([int(x) for x in input().split()])
    ans = 0
    for i in range(n):
        for j in range(i + 1, n):
            if math.gcd(a[i], 2 * a[j]) > 1:
                ans += 1
    return ans
for _ in range(int(input())):
    print(solve())
