#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll maxn = 2e5 + 10;
const ll mod = 1e9 + 7;
ll n, a[maxn];
vector<ll> adj[maxn];
ll up[maxn], down[maxn];
void dfs1(ll u, ll p) {
    up[u] = a[u];
    for (auto v : adj[u]) {
        if (v == p) continue;
        dfs1(v, u);
        up[u] = __gcd(up[u], up[v]);
    }
}
void dfs2(ll u, ll p) {
    for (auto v : adj[u]) {
        if (v == p) continue;
        if (up[v] % up[u] == 0) {
            down[v] = up[u];
        } else
            down[v] = __gcd(a[v], down[u]);
        dfs2(v, u);
    }
}
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n;
    for (size_t i = 1; i <= n; ++i) { cin >> a[i]; }
    for (ll i = 1; i < n; ++i) {
        ll x, y; cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    dfs1(1, 1);
    down[1] = a[1];
    dfs2(1, 1);
    for (ll i = 1; i <= n; ++i) { cout << max(down[i], up[i]) << " "; }
    return 0;
}
