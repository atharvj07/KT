import java.util.Scanner;

public class Zebra {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] photo = new int[n];
        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        int count = 0;
        int color = photo[0];
        for (int i = 1; i < n; i++) {
            if (photo[i] != color) {
                count++;
                color = photo[i];
            }
        }

        if (count % 2 == 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
