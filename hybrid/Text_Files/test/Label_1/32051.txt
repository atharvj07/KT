import java.util.*;
class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String dir = sc.next();
        int[] len = new int[n];
        for(int i=0; i<n; i++) len[i] = sc.nextInt();
        int cur = 0;
        while(cur >= 0 && cur < n){
            if(dir.charAt(cur) == '>'){
                cur += len[cur];
            }else{
                cur -= len[cur];
            }
        }
        if(cur < 0 || cur >= n) System.out.println("FINITE");
        else System.out.println("INFINITE");
    }
}
