k = int(input())

# Initializing the bus layout
bus = ['+------------------------+',
       '|#.#.#.#.#.#.#.#.#.#.#.|D|)',
       '|#.#.#.#.#.#.#.#.#.#.#.|.|',
       '|#.......................|',
       '|#.#.#.#.#.#.#.#.#.#.#.|.|)',
       '+------------------------+']

# Filling the seats based on the participants
for i in range(1, k+1):
    if i <= 4:  # Filling the last row first
        bus[3] = bus[3][:i*2] + 'O' + bus[3][i*2+1:]
    elif i <= 7:  # Filling the first seat of each row
        bus[i-3] = bus[i-3][:1] + 'O' + bus[i-3][2:]
    else:  # Filling the remaining seats
        row = (i - 8) % 3 + 1
        col = (i - 8) // 3 + 3
        bus[row] = bus[row][:col*2] + 'O' + bus[row][col*2+1:]

# Printing the final bus layout
for line in bus:
    print(line)
