#include <bits/stdc++.h>
using namespace std;
const int mod = (int)1e9 + 7, mx = 2005;
int dp[mx][mx], p[mx][mx], fail[mx][mx];
vector<int>v, ans;
bool ok[mx];
int n, gg;
bool can(int l, int r) {
    if(l > r)return 1;
    int &ret = dp[l][r];
    if(~ret)return ret;
    if(v[l] == l + 1) {
        ret = can(l + 2, r);
        if(ret)p[l][r] = 2;
        else fail[l][r] = 1;
        return ret;
    }
    ret = 0;
    int pos = -1;
    for(int i = l + 1; i <= r; i++) {
        if(v[i] == l + 1) {
            pos = i;
            break;
        }
    }
    if(pos != -1) {
        if(can(pos + 1, r)) {
            if(!ok[pos])ans.push_back(pos - l);
            ok[l] = 1;
            ret = 1;
            p[l][r] = pos - l + 1;
        }
    }
    int lft = l + 1;
    while(lft <= r) {
        if(v[lft] == l + 1) {
            continue;
        }
        int rgt = r;
        while(rgt >= lft) {
            if(v[rgt] == l + 1) {
                if(can(lft, rgt - 1) && can(rgt + 1, r) && ok[lft] && ok[rgt]) {
                    ret = 1;
                    p[l][r] = rgt - l + 1;
                    ok[l] = 1;
                    return ret;
                }
            }
            rgt--;
        }
        lft++;
    }
    fail[l][r] = 1;
    return ret;
}
void print(int l, int r) {
    if(l > r)return;
    if(fail[l][r]) {
        gg = 1;
        return;
    }
    if(p[l][r]) {
        if(p[l][r] % 2 == 1)ans.push_back(p[l][r] - 1);
        print(l + p[l][r], r);
    }
    else for(int i = l + 1; i <= r; i++) {
        if(v[i] == l + 1) {
            if(p[i + 1][r]) {
                if(p[i + 1][r] % 2 == 1)ans.push_back(p[i + 1][r] - 1);
                print(i + 1 + p[i + 1][r], r);
            }
        }
    }
}
int main() {
    int t, cs = 1;
    cin >> t;
    while(t--) {
        ans.clear();
        v.clear();
        gg = 0;
        memset(dp, -1, sizeof dp);
        memset(p, 0, sizeof p);
        memset(fail, 0, sizeof fail);
        memset(ok, 0, sizeof ok);
        cin >> n; v.push_back(0);
        for(int i = 1; i <= n; i++) {
            int x; cin >> x;
            v.push_back(x);
        }
        if(can(1, n)) {
            print(1, n);
            if(!gg) {
                cout << ans.size() << endl;
                for(int x : ans)cout << x << " ";
                cout << endl;
            }
            else cout << -1 << endl;
        }
        else cout << -1 << endl;
    }
    return 0;
}
