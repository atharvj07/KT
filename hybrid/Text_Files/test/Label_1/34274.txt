import java.util.Scanner;

public class HedgehogPresents {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int c = scanner.nextInt();
        int[] holidays = new int[c];
        for (int i = 0; i < c; i++) {
            holidays[i] = scanner.nextInt();
        }

        int presents = 0;
        int lastPresentDay = 0;
        for (int i = 1; i <= n; i++) {
            if (isHoliday(holidays, i) || i - lastPresentDay >= k) {
                presents++;
                lastPresentDay = i;
            }
        }

        System.out.println(presents);
    }

    private static boolean isHoliday(int[] holidays, int day) {
        for (int holiday : holidays) {
            if (holiday == day) {
                return true;
            }
        }
        return false;
    }
}
