import java.util.*;
import java.math.*;

public class Main {
    final static long M = 1000000007;

    static long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    static long pow(long a, long b) {
        if (b == 0) {
            return 1;
        }
        long t = pow(a, b / 2);
        t = t * t % M;
        if (b % 2 == 1) {
            t = t * a % M;
        }
        return t;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt();
        long ans = 0;
        for (int i = 2; i <= n; i++) {
            ans += pow((i - 1) * pow(i, M - 2) % M, m);
        }
        ans = ans % M;
        long t1 = pow(n, n * (long) m % (M - 1)), t2 = pow(n * pow(n - 1, n - 1) % M, M - 2);
        ans = ans * t1 % M;
        ans = ans * t2 % M;
        System.out.println(ans);
    }
}
