import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Point[] p = new Point[n];
        for (int i = 0; i < n; i++)
            p[i] = new Point(sc.nextInt(), sc.nextInt());
        System.out.println(solve(p));
    }

    private static int solve(Point[] p) {
        int ans = 0;
        for (int i = 0; i < p.length; i++) {
            for (int j = i + 1; j < p.length; j++) {
                for (int k = 0; k < p.length; k++) {
                    if (i == k || j == k)
                        continue;
                    if (isMiddle(p[i], p[j], p[k]))
                        ans++;
                }
            }
        }
        return ans;
    }

    private static boolean isMiddle(Point p1, Point p2, Point p3) {
        int dx1 = p2.x - p1.x;
        int dy1 = p2.y - p1.y;
        int dx2 = p3.x - p2.x;
        int dy2 = p3.y - p2.y;
        return dx1 == dx2 && dy1 == dy2;
    }

    private static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
