import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<BigInteger> a = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            a.add(in.nextBigInteger());
        }
        Collections.sort(a);
        List<BigInteger> b = new ArrayList<>();
        for (BigInteger x : a) {
            if (b.isEmpty() || !b.get(b.size() - 1).equals(x)) {
                b.add(x);
            }
        }
        Set<Integer> ans = new HashSet<>();
        for (BigInteger x : b) {
            int m = 0;
            BigInteger t = x;
            while (!t.equals(BigInteger.ZERO)) {
                m++;
                t = t.divide(BigInteger.TWO);
            }
            if (m > 0) {
                ans.add(m);
            }
        }
        List<Integer> res = new ArrayList<>();
        for (int x : ans) {
            res.add(x);
        }
        Collections.sort(res);
        if (res.isEmpty()) {
            System.out.println("-1");
        } else {
            for (int x : res) {
                System.out.println(x);
            }
        }
    }
}
