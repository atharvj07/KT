import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[k], b = new int[n * m - k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n * m - k; i++) {
            b[i] = sc.nextInt();
        }
        boolean ok = true;
        int[] aa = new int[n * m];
        for (int i = 0; i < n * m; i++) {
            aa[i] = n + m - 1;
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n * m; j++) {
                if (a[i] >= aa[j]) {
                    aa[j] = n + m - Math.max(j % m + 1, (n - j / m - 1) + 1) - 1;
                    break;
                }
            }
        }
        for (int i = 0; i < n * m - k; i++) {
            for (int j = 0; j < n * m; j++) {
                if (b[i] >= aa[j]) {
                    aa[j] = n + m - Math.max(j % m + 1, (n - j / m - 1) + 1) - 1;
                    break;
                }
            }
        }
        for (int i = 0; i < n * m; i++) {
            if (aa[i] < n + m - 1) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
