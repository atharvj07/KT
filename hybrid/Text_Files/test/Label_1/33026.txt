import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n * n];
        for (int i = 0; i < n * n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        boolean possible = true;
        for (int i = 0; i < n / 2; i++) {
            if (a[i] != a[n * n - i - 1]) {
                possible = false;
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n / 2; j++) {
                if (a[i * n + j] != a[i * n + n - j - 1]) {
                    possible = false;
                }
            }
        }
        if (possible) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    System.out.print(a[i * n + j] + " ");
                }
                System.out.println();
            }
        } else {
            System.out.println("NO");
        }
    }
}
