for _ in range(int(input())):
    a = [int(i) for i in input().split()]
    b = []
    for j in range(0, len(a), 2):
        if (j + 1 < len(a) and a[j] + a[j + 1] == 1) or j == len(a) - 1:
            b.append(a[j])
            continue
        k = j + 1
        while k < len(a) and a[k] == a[j]:
            k += 1
        if k == len(a):
            for i in range(j, len(a), 2):
                b.append(a[i])
            break
        else:
            b.append(a[j])
    print(len(b))
    print(*b)
