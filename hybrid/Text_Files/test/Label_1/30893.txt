import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int m = Integer.parseInt(line1[1]);
        int[][] relations = new int[m][3];
        for (int i = 0; i < m; i++) {
            String[] line2 = br.readLine().split(" ");
            relations[i][0] = Integer.parseInt(line2[0]);
            relations[i][1] = Integer.parseInt(line2[1]);
            relations[i][2] = Integer.parseInt(line2[2]);
        }

        System.out.println(isConsistent(n, m, relations));
    }

    public static String isConsistent(int n, int m, int[][] relations) {
        // Construct a graph to represent the relations
        Map<Integer, Set<Relation>> graph = new HashMap<>();
        for (int[] relation : relations) {
            int start = relation[0];
            int end = relation[1];
            int weight = relation[2];
            if (!graph.containsKey(start)) {
                graph.put(start, new HashSet<>());
            }
            graph.get(start).add(new Relation(end, weight));
        }

        // Perform topological sort to check if the graph is acyclic
        int[] inDegrees = new int[n + 1];
        for (int[] relation : relations) {
            inDegrees[relation[1]]++;
        }
        boolean[] visited = new boolean[n + 1];
        int count = 0;
        Set<Integer> topologicalOrder = new HashSet<>();
        for (int i = 1; i <= n; i++) {
            if (inDegrees[i] == 0) {
                topologicalOrder.add(i);
            }
        }
        while (!topologicalOrder.isEmpty()) {
            int node = topologicalOrder.iterator().next();
            topologicalOrder.remove(node);
            visited[node] = true;
            count++;
            if (graph.containsKey(node)) {
                for (Relation relation : graph.get(node)) {
                    int neighbor = relation.getNeighbor();
                    inDegrees[neighbor]--;
                    if (inDegrees[neighbor] == 0) {
                        topologicalOrder.add(neighbor);
                    }
                }
            }
        }

        if (count == n) {
            return "Yes";
        } else {
            return "No";
        }
    }

    public static class Relation {
        private int neighbor;
        private int weight;

        public Relation(int neighbor, int weight) {
            this.neighbor = neighbor;
            this.weight = weight;
        }

        public int getNeighbor() {
            return neighbor;
        }

        public int getWeight() {
            return weight;
        }
    }
}
