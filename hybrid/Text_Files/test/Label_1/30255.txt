import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }
            int[] p = new int[m];
            for (int i = 0; i < m; i++) {
                p[i] = sc.nextInt();
            }
            int[] cnt = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                boolean ok = true;
                for (int j = 0; j < m; j++) {
                    if (i % p[j] == 0) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    cnt[i]++;
                }
            }
            double ans = 0;
            for (int i = 1; i <= n; i++) {
                ans += cnt[i] * (double) i;
            }
            System.out.println(ans / n);
        }
        sc.close();
    }
}
