#include<bits/stdc++.h>
using namespace std;
const int N = 100005, OO = 0x3f3f3f3f;

struct node{
	int sum, lz;
}tree[N * 4];

void prop(int node,int s,int e){
	if(!tree[node].lz) return;
	tree[node].sum += (e - s + 1) * tree[node].lz;
	if(s != e){
		tree[node * 2].lz += tree[node].lz;
		tree[node * 2 + 1].lz += tree[node].lz;
	}
	tree[node].lz = 0;
}

void update(int node,int s,int e,int l,int r,int val){
	prop(node,s,e);
	if(s > r || e < l) return;
	if(s >= l && e <= r){
		tree[node].lz += val;
		prop(node,s,e);
		return;
	}
	int mid = (s + e) / 2;
	update(node * 2,s,mid,l,r,val);
	update(node * 2 + 1,mid + 1,e,l,r,val);
	tree[node].sum = tree[node * 2].sum + tree[node * 2 + 1].sum;
}

int get(int node,int s,int e,int l,int r){
	prop(node,s,e);
	if(s > r || e < l) return 0;
	if(s >= l && e <= r) return tree[node].sum;
	int mid = (s + e) / 2;
	return get(node * 2,s,mid,l,r) + get(node * 2 + 1,mid + 1,e,l,r);
}

int n,m,a[N],l[N],r[N];
int main(){
	scanf("%d%d",&n,&m);
	for(int i = 0 ; i < m ; ++i){
		scanf("%d%d",l + i,r + i);
		update(1,1,n,l[i],r[i],1);
	}
	for(int i = 0 ; i < n ; ++i){
		if(!get(1,1,n,i + 1,i + 1)){
			printf("%d\n",i + 1);
			for(int j = 0 ; j < n ; ++j)
				printf("%d ",j < i);
		}
		else update(1,1,n,i + 1,i + 1,1);
	}
	return 0;
}
