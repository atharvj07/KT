#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<char> field(n);
    for (int i = 0; i < n; i++) {
        cin >> field[i];
    }
    vector<int> dist(n, -1);
    queue<int> q;
    for (int i = 0; i < n; i++) {
        if (field[i] == 'P') {
            q.push(i);
            dist[i] = 0;
        }
    }
    while (!q.empty()) {
        int curr = q.front();
        q.pop();
        if (field[curr] == '*') {
            cout << dist[curr] << endl;
            return 0;
        }
        if (curr > 0 && dist[curr - 1] == -1) {
            dist[curr - 1] = dist[curr] + 1;
            q.push(curr - 1);
        }
        if (curr < n - 1 && dist[curr + 1] == -1) {
            dist[curr + 1] = dist[curr] + 1;
            q.push(curr + 1);
        }
    }
    cout << -1 << endl;
    return 0;
}
