n, m, q = map(int, input().split())
operations = [list(map(int, input().split())) for _ in range(q)]

# Initial matrix with zeros
matrix = [[0]*m for _ in range(n)]

# Counters for shifts
row_shifts = [0]*n
col_shifts = [0]*m

for operation in reversed(operations):
    if operation[0] == 1:
        row_shifts[operation[1]-1] -= 1
    elif operation[0] == 2:
        col_shifts[operation[1]-1] -= 1
    else:
        t, r, c, x = operation
        r, c = r-1, c-1
        # Adjust the position according to the shifts
        r = (r + row_shifts[r]) % n
        c = (c + col_shifts[c]) % m
        matrix[r][c] = x

for row in matrix:
    print(*row)
