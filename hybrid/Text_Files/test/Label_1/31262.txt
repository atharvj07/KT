import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int A = sc.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        int B = -1;
        int cntA = 0;
        for (int i = 0; i < n; i++) {
            if (c[i] == A) {
                cntA++;
            } else if (B == -1 || c[i] == B || cntA < i) {
                B = c[i];
            }
        }
        System.out.println(B);
    }
}
