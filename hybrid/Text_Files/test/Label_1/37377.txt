n = int(input())
parts = [list(map(int, input().split())) for _ in range(n)]

# Calculate max possible road width for each part from left to right
for i in range(1, n):
    parts[i][0] = min(parts[i][0] + parts[i][1], parts[i-1][0] + 1)

# Calculate max possible road width for each part from right to left
for i in range(n-2, -1, -1):
    parts[i][0] = min(parts[i][0], parts[i+1][0] - 1)

# Check if it's possible to achieve the goal
for i in range(1, n):
    if abs(parts[i][0] - parts[i-1][0]) > 1:
        print(-1)
        break
else:
    total_lawn_destroyed = sum(s + g - parts[i][0] for i, (s, g) in enumerate(parts))
    print(total_lawn_destroyed)
    print(' '.join(str(part[0]) for part in parts))
