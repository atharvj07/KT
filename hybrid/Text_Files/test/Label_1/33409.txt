import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] d = new int[N];
        for (int i = 0; i < N; i++) {
            d[i] = sc.nextInt();
        }

        int count = 0;
        for (int k = 1; k <= 100000; k++) {
            int arc = 0;
            int abc = 0;
            for (int i = 0; i < N; i++) {
                if (d[i] >= k) {
                    arc++;
                } else {
                    abc++;
                }
            }
            if (arc == abc) {
                count++;
            }
        }

        System.out.println(count);
    }
}
