import java.util.*;

public class Main {
    static int n, m, k;
    static char[][] g, ans;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        g = new char[n][m];
        ans = new char[n][m];
        for (int i = 0; i < n; i++) {
            g[i] = sc.next().toCharArray();
            Arrays.fill(ans[i], '.');
        }
        solve(0, 0);
        for (char[] cc : ans) System.out.println(new String(cc));
    }

    static boolean solve(int x, int y) {
        if (k == 0) return true;
        for (int i = x; i < n; i++) {
            for (int j = (i == x ? y : 0); j < m; j++) {
                if (g[i][j] == '.') {
                    for (int d = 0; d < 4; d++) {
                        int nx = i + dx[d], ny = j + dy[d];
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && g[nx][ny] == '.') {
                            ans[i][j] = 'X';
                            g[i][j] = '#';
                            if (solve(i, j)) return true;
                            ans[i][j] = '.';
                            g[i][j] = '.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }
}
