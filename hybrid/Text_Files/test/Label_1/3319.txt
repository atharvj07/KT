#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    int max_el = *max_element(a.begin(), a.end());
    int min_el = *min_element(a.begin(), a.end());
    if (min_el >= 0) {
        cout << 0 << endl;
        return 0;
    }
    int ans;
    if (max_el > abs(min_el)) {
        ans = abs(min_el) + 1;
    } else {
        ans = abs(max_el);
    }
    cout << ans << endl;
    return 0;
}
