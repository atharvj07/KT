import java.util.*;

class Main {
    static int[] dr = {0, 1, 0, -1};
    static int[] dc = {1, 0, -1, 0};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int r = in.nextInt();
        int n = in.nextInt();
        int[][] a = new int[r + 1][3];

        for (int i = 0; i < n; i++) {
            int r1 = in.nextInt();
            int c1 = in.nextInt();
            a[r1][c1] = 1;
            a[r1 - 1][3 - c1] = 1;
            a[r1][3 - c1] = 1;
            a[r1 + 1][3 - c1] = 1;
        }
        boolean ok = false;
        for (int i = 1; i <= r; i++) {
            for (int j = 1; j <= 2; j++) {
                if (a[i][j] == 0) {
                    if (n % 2 == 0) {
                        ok = false;
                    } else {
                        ok = true;
                    }
                    break;
                }
            }
        }
        System.out.println(ok ? "WIN" : "LOSE");
    }
}
