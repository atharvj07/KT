#include <iostream>
#include <vector>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int l, r;
    std::cin >> l >> r;
    long long ans = 0;
    while (l < r) {
      l++;
      std::string sl = std::to_string(l), sr = std::to_string(r);
      int n = sl.size(), m = sr.size();
      if (n > m) ans += n;
      else ans += m;
    }
    std::cout << ans << std::endl;
  }
  return 0;
}
