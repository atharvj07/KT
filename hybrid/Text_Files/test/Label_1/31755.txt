import java.io.*;
import java.util.*;

public class Main {
    static int[] p;
    static int[] g;
    static int[] s;
    static int[] d;
    static ArrayList<ArrayList<Integer>> al;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        p = new int[n + 1];
        g = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            String[] s = br.readLine().split(" ");
            p[i] = Integer.parseInt(s[0]);
            g[i] = s[1].charAt(0) - 'a';
        }
        int k = Integer.parseInt(br.readLine());
        s = new int[k + 1];
        String[] s1 = br.readLine().split(" ");
        for (int i = 1; i <= k; i++) s[i] = Integer.parseInt(s1[i - 1]);
        d = new int[n + 1];
        al = new ArrayList<>();
        for (int i = 0; i <= n; i++) al.add(new ArrayList<>());
        for (int i = 1; i <= n; i++) al.get(p[i]).add(i);
        int l = 0;
        for (int i = 1; i <= n; i++) {
            if (d[i] == 0) {
                l++;
                Queue<Integer> q = new LinkedList<>();
                q.add(i);
                d[i] = 1;
                while (!q.isEmpty()) {
                    int v = q.poll();
                    for (int u : al.get(v)) {
                        d[u] = d[v] + 1;
                        q.add(u);
                    }
                }
            }
        }
        for (int i = 1; i <= k; i++) System.out.print((s[i] != 0 ? d[s[i]] - 1 : 0) + " ");
        System.out.println();
    }
}
