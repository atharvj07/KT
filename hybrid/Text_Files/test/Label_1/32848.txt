import java.util.*;

public class Main {
    static long[] mod = new long[]{1000000007L};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        int m = sc.nextInt();
        long[] ans = new long[m];
        long[] cnt = new long[n + 1];
        long[] pref = new long[n + 1];
        int[] L = new int[n + 1], R = new int[n + 1];
        int[] able = new int[n + 1];
        Arrays.fill(able, 1);
        long c = 0;
        for (int i = 0; i < n; i++) if (a[i] == a[(i + 1) % n]) c++;
        for (int i = 0; i < n; i++) pref[i + 1] = pref[i] + (a[i] == a[(i + 1) % n] ? c : 0);
        for (int i = 0; i < n; i++) {
            if (able[i] != 0) {
                for (int j = i; j < n; j++) {
                    if (a[j] != a[i]) break;
                    R[j] = i;
                }
                for (int j = i - 1; j >= 0; j--) {
                    if (a[j] != a[i]) break;
                    L[j] = i;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            int t = sc.nextInt(), x = sc.nextInt() - 1;
            able[x] ^= 1;
            if (able[x] != 0) {
                for (int j = x; j < n; j++) {
                    if (a[j] != a[x]) break;
                    R[j] = x;
                }
                for (int j = x - 1; j >= 0; j--) {
                    if (a[j] != a[x]) break;
                    L[j] = x;
                }
            } else {
                for (int j = x; j < n; j++) {
                    if (a[j] != a[x]) break;
                    if (R[j] == x) {
                        for (int k = j + 1; k < n; k++) {
                            if (a[k] != a[x]) break;
                            R[k] = L[k];
                        }
                    }
                }
                for (int j = x - 1; j >= 0; j--) {
                    if (a[j] != a[x]) break;
                    if (L[j] == x) {
                        for (int k = j - 1; k >= 0; k--) {
                            if (a[k] != a[x]) break;
                            L[k] = R[k];
                        }
                    }
                }
            }
            c = pref[n] - pref[R[x] + 1] - pref[L[x]] + pref[R[L[x]] + 1];
            ans[i] = c * (c - 1) % mod[0] * (c - 2) % mod[0];
            System.out.println(ans[i]);
        }
    }
}
