#include <bits/stdc++.h>
using namespace std;
#define MOD 1000000007
#define ll long long
typedef pair<ll, ll> pi;

const ll C = 6;
ll n, k, m;
vector<pi> a;
map<ll, ll> fp[C];

ll f[61][2001][C];
ll g[61][2001][C];

void pre() {
  for (ll i = 0; i < k; i++) {
    for (ll j = 0; j < m; j++) {
      for (ll c = 0; c < C; c++) {
        if (fp[c].find(a[j].first >> i & 1) != fp[c].end()) {
          f[i][j][c] = fp[c][a[j].first >> i & 1];
          if (c == a[j].second) {
            g[i][j][c] = 1;
          }
        } else {
          f[i][j][c] = -1;
        }
      }
    }
  }
}

bool valid(ll i, ll j, ll c) {
  if (j > 0 && f[i][j - 1][c] == -1) return false;
  if (i > 0 && f[i - 1][j][(c + 3) % 6] == -1) return false;
  if (i > 0 && f[i - 1][j + (1 << i)][(c + 2) % 6] == -1) return false;
  return true;
}

ll dp[61][2001][C];

ll solve(ll i, ll j, ll c) {
  if (i == k) {
    return 1;
  }

  auto &ans = dp[i][j][c];
  if (~ans) return ans;
  if (!valid(i, j, c)) {
    return ans = 0;
  }

  ll op = 0;
  if (g[i][j][c]) {
    op = solve(i + 1, j, c);
  } else {
    if (f[i][j][c] != -1) {
      op = solve(i + 1, j, c);
      if ((c + 3) % 6 != f[i][j][c]) op += solve(i + 1, j, (c + 3) % 6);
      if ((c + 2) % 6 != f[i][j][c]) op += solve(i + 1, j + (1 << i), (c + 2) % 6);
      op %= MOD;
    } else {
      op = solve(i + 1, j, (c + 3) % 6) + solve(i + 1, j + (1 << i), (c + 2) % 6);
      op %= MOD;
    }
  }
  return ans = op;
}

int main() {
  cin >> k >> m;
  a.resize(m);
  for (ll i = 0; i < m; i++) {
    ll v;
    string x;
    cin >> v >> x;
    if (x == "white") {
      a[i] = {v, 0};
      fp[0][v] = 1;
    } else if (x == "yellow") {
      a[i] = {v, 1};
      fp[1][v] = 1;
    } else if (x == "green") {
      a[i] = {v, 2};
      fp[2][v] = 1;
    } else if (x == "blue") {
      a[i] = {v, 3};
      fp[3][v] = 1;
    } else if (x == "red") {
      a[i] = {v, 4};
      fp[4][v] = 1;
    } else if (x == "orange") {
      a[i] = {v, 5};
      fp[5][v] = 1;
    }
  }
  pre();
  memset(dp, -1, sizeof dp);
  cout << solve(0, 0, 0) << endl;
  return 0;
}
