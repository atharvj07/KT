#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  map<int, int> m;
  while (n--) {
    int x;
    cin >> x;
    m[x]++;
  }
  int ans = m.begin()->second;
  for (int i = 0; i < m.size(); i++) {
    ans = max(ans, m.begin()->second + i * (int)m.size());
    m.erase(m.begin());
  }
  cout << ans << endl;
  return 0;
}
