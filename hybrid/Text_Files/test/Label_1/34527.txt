import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        scanner.close();

        long ans = a[0];
        for (int i = 1; i < n; i++) {
            ans = Math.max(ans, a[i]);
            if (ans < 0) {
                ans = a[i];
            } else {
                ans -= a[i];
            }
        }

        System.out.println(ans);
    }
}
