import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);

        int n = stdin.nextInt();

        String colors = "ROYGBIV";

        if (n % 7 == 0) {
            System.out.println(colors.repeat(n / 7));
        } else {
            StringBuilder sb = new StringBuilder(colors.substring(0, n % 7));
            colors = colors.substring(n % 7);

            while (sb.length() < n) {
                sb.append(colors, 0, Math.min(n - sb.length(), colors.length()));
            }

            System.out.println(sb);
        }
    }
}
