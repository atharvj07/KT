#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> c(n);
    for (int& i: c) cin >> i;
    sort(c.begin(), c.end());
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (c[i] > 2LL * c[i - 1]) ans++;
    }
    cout << ans << '\n';
    return 0;
}
