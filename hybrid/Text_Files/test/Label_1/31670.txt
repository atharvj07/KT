import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] strs = br.readLine().trim().split("\\s+");
        Map<Integer, List<Integer>> map = new HashMap<>();
        int n = Integer.parseInt(strs[0]);
        int m = Integer.parseInt(strs[1]);
        int k = Integer.parseInt(strs[2]);
        int dummyCalls = 0;
        for (int i = 0; i < k; i++) {
            String[] ss = br.readLine().trim().split("\\s+");
            if (ss[0].equals("+")) {
                int hash = Integer.parseInt(ss[2]);
                List<Integer> list = map.getOrDefault(hash, new ArrayList<>());
                int currDummies = Collections.binarySearch(list, i);
                if (currDummies < 0) currDummies = -currDummies - 1;
                list.add(currDummies, i);
                dummyCalls += currDummies;
                map.put(hash, list);
            }
            if (ss[0].equals("-")) {
                int id = Integer.parseInt(ss[1]);
                for (Map.Entry<Integer, List<Integer>> entry : map.entrySet()) {
                    List<Integer> list = entry.getValue();
                    int idx = Collections.binarySearch(list, id);
                    if (idx >= 0) {
                        list.remove(idx);
                        for (int j = idx; j < list.size(); j++) {
                            list.set(j, list.get(j) - 1);
                        }
                    }
                }
            }
        }
        System.out.println(dummyCalls);
    }
}
