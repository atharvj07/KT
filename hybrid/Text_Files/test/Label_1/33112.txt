import java.util.Arrays;
import java.util.Scanner;

public class AssignValues {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }

        int[] a = new int[n];
        Arrays.fill(a, -1);

        // Check if the graph is weakly connected
        boolean[] visited = new boolean[n];
        dfs(p, 0, visited);
        for (boolean b : visited) {
            if (!b) {
                System.out.println("IMPOSSIBLE");
                return;
            }
        }

        // Assign values to the vertices
        for (int i = 0; i < n; i++) {
            int j = p[i] - 1;
            if (a[j] == -1) {
                a[j] = 0;
            }
            while (a[i] == -1) {
                if (a[j] == n - 1) {
                    System.out.println("IMPOSSIBLE");
                    return;
                }
                a[i] = a[j] + 1;
                j = p[i] - 1;
            }
        }

        System.out.println("POSSIBLE");
    }

    private static void dfs(int[] p, int i, boolean[] visited) {
        visited[i] = true;
        int j = p[i] - 1;
        if (!visited[j]) {
            dfs(p, j, visited);
        }
    }
}
