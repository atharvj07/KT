#include <iostream>
using namespace std;
int main() {
    int n;
    cin >> n;
    int a[2*n-1], sum = 0, neg = 0;
    for (int i = 0; i < 2*n-1; i++) {
        cin >> a[i];
        if (a[i] < 0) neg++;
        sum += abs(a[i]);
    }
    if (neg % 2 == 1) sum -= 2*abs(a[2*n-2]);
    cout << sum << endl;
    return 0;
}
