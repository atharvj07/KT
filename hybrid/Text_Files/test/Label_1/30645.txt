import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    public static final long MOD = 1000000007;
    public static int n, q;
    public static long[] a, w;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        q = Integer.parseInt(st.nextToken());
        a = new long[n];
        w = new long[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(st.nextToken());
        }
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            w[i] = Long.parseLong(st.nextToken());
        }

        SegmentTree stree = new SegmentTree(n);
        while (q-- > 0) {
            st = new StringTokenizer(br.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            if (x > 0) {
                System.out.println(stree.query(x - 1, y - 1));
            } else {
                stree.update(y - 1, y);
            }
        }
    }

    static class SegmentTree {
        int[] st;
        long[] lazy;

        public SegmentTree(int n) {
            st = new int[4 * n + 3];
            lazy = new long[4 * n + 3];
        }

        public void update(int l, int r, long val, int node, int sl, int sr) {
            if (l > sr || r < sl) {
                return;
            }
            if (l <= sl && r >= sr) {
                lazy[node] += val;
                return;
            }
            int m = (sl + sr) / 2;
            update(l, r, val, 2 * node + 1, sl, m);
            update(l, r, val, 2 * node + 2, m + 1, sr);
        }

        public void update(int ind, long val) {
            update(ind, ind, val, 0, 0, n - 1);
        }

        public long query(int l, int r) {
            return query(l, r, 0, 0, n - 1);
        }

        public long query(int l, int r, int node, int sl, int sr) {
            propagate(node, sl, sr);
            if (l > sr || r < sl) {
                return 0;
            }
            if (l <= sl && r >= sr) {
                return st[node];
            }
            int m = (sl + sr) / 2;
            long left = query(l, r, 2 * node + 1, sl, m);
            long right = query(l, r, 2 * node + 2, m + 1, sr);
            return left + right;
        }

        private void propagate(int node, int sl, int sr) {
            st[node] += lazy[node] * (sr - sl + 1);
            if (sl != sr) {
                lazy[2 * node + 1] += lazy[node];
                lazy[2 * node + 2] += lazy[node];
            }
            lazy[node] = 0;
        }
    }
}
