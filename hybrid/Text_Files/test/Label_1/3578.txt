#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 1e5 + 5;

int n;
ll a[N];
int id[1LL<<20], deg[1LL<<20];
bool vis[1LL<<20];
queue<int> q;

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
    if (id[a[i]] == 0) id[a[i]] = i, vis[a[i]] = false, deg[a[i]] = 0;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      if ((a[i] & a[j]) > 0) {
        deg[a[i]]++;
        deg[a[j]]++;
        q.push(a[i]);
        q.push(a[j]);
      }
    }
  }
  while (!q.empty()) {
    int u = q.front(); q.pop();
    if (vis[u]) continue;
    vis[u] = true;
    int v = 0;
    for (int bit = 0; bit < 20; bit++) {
      if (((1LL<<bit) & u) > 0) {
        v |= (1LL<<bit);
      }
    }
    int pi = id[v];
    if (u != v) {
      cout << 4 << endl;
      return 0;
    }
    if (deg[u] % 2 == 1) {
      cout << 3 << endl;
      return 0;
    }
    if (!vis[v ^ a[pi]]) {
      q.push(v ^ a[pi]);
    }
  }
  cout << -1 << endl;
  return 0;
}
