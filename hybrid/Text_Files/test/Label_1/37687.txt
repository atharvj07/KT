from collections import deque
n, q = map(int, input().split())
a = deque(map(int, input().split()))
ans = []
for i in range(q):
    m = int(input()) % (n - i)
    ans.append('{} {}'.format(a[m], a[m + 1]))
    if a[m] > a[m + 1]:
        a.rotate(-m - 1)
    else:
        a.rotate(-m)
        a.reverse()
print(*ans, sep='\n')
