#include <iostream>
#include <vector>

using namespace std;
int main() {
  int n, k, x;
  while (cin >> n >> k) {
    vector<int> v;
    for (int i = 1; i <= n; i++) v.push_back(i);
    int pos = 0;
    for (int i = 0; i < k; i++) {
      cin >> x;
      pos = (pos + x) % n;
      cout << v[pos] << ' ';
      v.erase(v.begin() + pos);
    }
    cout << endl;
  }
  return 0;
}
