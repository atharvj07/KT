#include<bits/stdc++.h>
using namespace std;

int main() {
  int n, m; cin >> n >> m;
  vector<int> att, def, c;
  for (int i = 0; i < n; i++) {
    string pos; int st;
    cin >> pos >> st;
    if (pos == "ATK") att.push_back(st);
    else def.push_back(st);
  }
  for (int i = 0; i < m; i++) {
    int x; cin >> x;
    c.push_back(x);
  }
  sort(att.begin(), att.end(), greater<int>());
  sort(def.begin(), def.end(), greater<int>());
  sort(c.begin(), c.end(), greater<int>());
  int ans = 0;
  int j = 0;
  for (int i = 0; i < c.size(); i++) {
    if (j == att.size()) {
      if (def.size() > 0 && def[0] < c[i]) {
        ans += c[i] - def[0];
        def.erase(def.begin());
      }
    } else {
      if (att[j] >= c[i]) {
        ans += c[i];
        j++;
      } else {
        if (def.size() > 0 && def[0] < c[i]) {
          ans += c[i] - def[0];
          def.erase(def.begin());
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
