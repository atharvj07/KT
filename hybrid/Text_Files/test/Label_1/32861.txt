import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        int p = sc.nextInt();
        for (int i = 0; i < p; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt();
            long ans = 0;
            for (int j = 0; j < n; j++) {
                if ((j - a) % b == 0) {
                    ans += w[j];
                }
            }
            System.out.println(ans);
        }
    }
}
