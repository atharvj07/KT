#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        unordered_map<int,int> fr;
        while(n--) {
            ll x;
            cin >> x;
            fr[x&x^x]++;
        }
        ll ans = 0;
        for(auto it:fr) {
            ll x = it.second;
            ans += x*(x-1)/2;
        }
        cout << ans << endl;
    }
    return 0;
}
