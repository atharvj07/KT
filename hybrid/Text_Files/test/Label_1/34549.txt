import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        int ans = 0;
        int last = -1;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                ans++;
                last = i;
            }
        }
        if (last != -1) {
            ans++;
        }
        System.out.println(ans);
    }
}
