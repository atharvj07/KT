import sys
for _ in range(int(input())):
    n = int(input())
    s = input()
    t = input()
    for i in range(n):
        if s[i] == t[i]:
            continue
        j = t.find(s[i], i)
        if j == -1: break
        t = t[:j] + s[i] + t[j + 1:]
        break
    print('Yes' if s == t else 'No')
