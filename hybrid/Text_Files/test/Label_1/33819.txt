import java.util.*;

public class Main {
    private static int n, m, k, s;
    private static int[][] a;
    private static int[] q;
    private static int[][] dp;
    private static Map<Integer, List<int[]>> map = new HashMap<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt(); m = sc.nextInt(); k = sc.nextInt(); s = sc.nextInt();
        a = new int[n][m];
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        q = new int[s];
        for(int i = 0; i < s; i++) {
            q[i] = sc.nextInt();
        }
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < m; j++) {
                if(!map.containsKey(a[i][j])) map.put(a[i][j], new ArrayList<>());
                map.get(a[i][j]).add(new int[]{i, j});
            }
        }
        dp = new int[s][k + 1];
        for(int i = 0; i < s; i++) Arrays.fill(dp[i], Integer.MAX_VALUE);
        int ans = 0;
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < m; j++) {
                if(a[i][j] == q[0]) {
                    dp[0][a[i][j]] = 0;
                    ans = Math.max(ans, 0);
                }
            }
        }
        for(int i = 1; i < s; i++) {
            for(int j = 0; j < n; j++) {
                for(int l = 0; l < m; l++) {
                    if(a[j][l] == q[i]) {
                        for(int t = 1; t <= k; t++) {
                            if(dp[i - 1][t] != Integer.MAX_VALUE) {
                                dp[i][a[j][l]] = Math.min(dp[i][a[j][l]], dp[i - 1][t] + Math.abs(j - map.get(t).get(map.get(t).size() - 1)[0]) + Math.abs(l - map.get(t).get(map.get(t).size() - 1)[1]));
                            }
                        }
                        ans = Math.max(ans, dp[i][a[j][l]]);
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
