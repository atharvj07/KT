import bisect
n, t = map(int, input().split())
a = []
c = []
p = []
for i in range(n):
    ai, ti = map(int, input().split())
    a.append(ai)
    c.append(ti)
    p.append(i + 1)
a.sort()
c.sort()
p.sort(key=lambda x: c[x - 1])
s = 0
ans = []
for x in p:
    i = bisect.bisect_right(a, s)
    if i == len(a):
        break
    s += 1
    ans.append(x)
print(s, len(ans))
print(*ans)
