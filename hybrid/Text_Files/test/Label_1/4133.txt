#include <iostream>
#include <algorithm>
using namespace std;

const int MAXN = int(2e3);

int n, a[MAXN], odd[MAXN], even[MAXN];

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) cin >> a[i];

    for (int i = 0; i < n; ++i)
        if (a[i] % 2) ++odd[a[i] % 2];
        else ++even[a[i] % 2];

    int ans = 0;
    bool ok = true;
    for (int j = 0; j < 2; ++j) {
        for (int i = 0; i <= 1000000; ++i) {
            if ((j && even[j] - 1 > 0) || (!j && odd[j] - 1 > 0)) {
                ok = false;
                break;
            }
            else {
                if (!j) ans += i * odd[j];
                else ans += i * even[j];
            }
            if (j) --even[j];
            else --odd[j];
        }
    }

    if (ok) ans = 0;

    cout << ans << endl;

    return 0;
}
