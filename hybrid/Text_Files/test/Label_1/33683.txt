import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        int n = in.nextInt();
        int[][] a = new int[n][2];
        for (int i = 0; i < n; i++) {
            a[i][0] = in.nextInt();
            a[i][1] = in.nextInt();
        }
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            int ans = 0;
            for (int j = 0; j < n; j++) {
                if (a[j][0] <= x && x <= a[j][1] && a[j][1] <= y) {
                    ans = y;
                    break;
                }
                if (a[j][0] <= x && y <= a[j][1]) {
                    ans = y;
                    break;
                }
                if (x <= a[j][0] && a[j][0] <= y) {
                    ans = a[j][1];
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
