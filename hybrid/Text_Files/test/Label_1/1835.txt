#include <bits/stdc++.h>
using namespace std;
#define f first
#define s second
typedef pair<int, int> pi;
const int N = 1e5+1;
int n, m, u, v, par[N], ans;
vector<pi> edges;

int find(int x) {return par[x]==x?x:par[x]=find(par[x]);}

int main() {
    cin >> n >> m;
    for (int i=0; i<=n; i++) par[i]=i;
    while (m--) {
        cin >> u >> v;
        edges.push_back({u, v});
    }
    for (auto &edge: edges) {
        u=find(edge.f), v=find(edge.s);
        if (u!=v) par[u]=v, ans++;
    }
    cout << ans << endl;
}
