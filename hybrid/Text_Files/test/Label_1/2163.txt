#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 1;
int64_t f[N];
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  for (int i = n - 1; i > 0; i--) {
    f[i] = f[i + 1];
    if (a[i] == i) continue;
    f[i] = min(f[i], f[a[i]] + (i ^ a[i] ^ 1));
  }
  for (int i = 1; i <= n; i++) cout << f[i] << ' ';
  return 0;
}
