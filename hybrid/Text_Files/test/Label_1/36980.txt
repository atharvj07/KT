from collections import deque
from sys import stdin

input = stdin.readline

for _ in range(int(input())):
    input()
    n, m = map(int, input().split())
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        a, b = map(int, input().split())
        g[a].append(b)

    vis = [0] * (n + 1)
    q = deque()
    q.append((1, 0))
    vis[1] = 1
    d = [0] * (n + 1)
    ans = []
    while q:
        v, dist = q.popleft()
        if dist:
            ans.append(dist)
        for v1 in g[v]:
            if not vis[v1]:
                q.append((v1, dist + 1))
                d[v1] = dist + 1
                vis[v1] = 1
    ans += [min(d[1:])] * (n - len(ans))
    for a in ans:
        print(a, end=' ')
    print()
