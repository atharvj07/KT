#include <iostream>
#include <map>
#include <set>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  map<int, int> sell, buy;
  for (int i = 0; i < n; i++) {
    char d;
    int p, q;
    cin >> d >> p >> q;
    if (d == 'B')
      buy[p] += q;
    else
      sell[p] += q;
  }
  set<int> s1, s2;
  for (auto x : sell) s1.insert(-x.first);
  for (auto x : buy) s2.insert(-x.first);
  while (s1.size() > s) s1.erase(s1.begin());
  while (s2.size() > s) s2.erase(s2.begin());
  for (auto x : s1) cout << "S " << -x << " " << sell[-x] << endl;
  for (auto x : s2) cout << "B " << -x << " " << buy[-x] << endl;
  return 0;
}
