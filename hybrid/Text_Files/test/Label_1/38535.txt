a, b, w, x, c = map(int, input().split())

d = max(0, (c - a + a - 1) // a)
low, high = 0, c

while low < high:
    mid = (low + high) // 2
    ca = c - mid
    ba = b - (mid * x) % w
    aa = a - (mid * x) // w
    if ba < 0:
        aa -= 1
        ba += w
    if ca <= aa:
        high = mid
    else:
        low = mid + 1

print(low)
