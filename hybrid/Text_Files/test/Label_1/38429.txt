import sys
read = sys.stdin.readline
n, m = map(int, read().split())
e = {}
for _ in range(m):
    a, b, c = map(int, read().split())
    if (a, b) not in e: e[(a, b)] = []
    e[(a, b)].append(c)
q = int(read())
for _ in range(q):
    u, v = map(int, read().split())
    ans = 0
    for a, b in [(u, v), (v, u)]:
        for c in e.get((a, b), []):
            if not ans: ans = set(e.get((u, v), [])+e.get((v, u), []))
            ans &= set(e.get((a, b), []))
    print(len(ans))
