#include <iostream>
#include <map>
#include <vector>

const int INF = 2e9;

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int& ai : a) std::cin >> ai;

  std::map<int, int> ink;
  for (int i = 0; i < n; i++) ink[i + 1] = a[i];
  for (int day = 1; ; day++) {
    int pen = (day - 1) % n + 1;
    ink[pen]--;
    if (ink[pen] == 0) {
      std::cout << pen << std::endl;
      return 0;
    }
  }
}
