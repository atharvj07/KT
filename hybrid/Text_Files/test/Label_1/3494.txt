#include <iostream>
#include <unordered_set>
#include <algorithm>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    if (k > n) {
        cout << 0 << endl;
        return 0;
    }
    unordered_set<int> s;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        s.insert(x);
    }
    int ans = s.size() * k - n;
    cout << ans << endl;
    return 0;
}
