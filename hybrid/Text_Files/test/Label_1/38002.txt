import sys
n = int(input())
a = list(map(int,input().split()))
a.sort()
if n==1:
    print(-1)
    sys.exit()
for i in range(n):
    if i and 2*(a[i]-a[i-1])<a[i-1]-a[i-2]:
        print(0)
        sys.exit()
mn,mx=a[1]-a[0],a[-1]-a[-2]
for i in range(1,n-1):
    mn=min(mn,a[i+1]-a[i])
    mx=max(mx,a[i+1]-a[i])
if mn>=mx: print(-1)
else:
    print(1)
    print(a[1]-mn)
