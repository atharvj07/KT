import java.util.*;

public class Main {
    static int n, m;
    static boolean[][] graph;
    static int[] wishes;
    static int[] res;

    static void dfs(int u, int p) {
        if (u != p) {
            res[u] = p;
        }
        for (int v = 1; v <= n; v++) {
            if (graph[u][v] && v != p) {
                dfs(v, u);
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        graph = new boolean[n + 1][n + 1];
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            graph[u][v] = true;
        }
        wishes = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            wishes[i] = sc.nextInt();
        }
        res = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (res[i] == 0) {
                dfs(i, i);
            }
        }
        int k = 0;
        for (int i = 1; i <= n; i++) {
            if (res[i] != i) {
                k++;
            }
        }
        System.out.println(k);
        for (int i = 1; i <= n; i++) {
            if (res[i] != i) {
                System.out.println(wishes[i]);
            }
        }
    }
}
