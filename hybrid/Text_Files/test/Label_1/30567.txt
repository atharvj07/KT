import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int h = sc.nextInt();
    int w = sc.nextInt();
    int[] r = new int[h];
    int[] c = new int[w];
    for (int i = 0; i < h; i++) r[i] = sc.nextInt();
    for (int i = 0; i < w; i++) c[i] = sc.nextInt();
    sc.close();

    long[][] dp = new long[h + 1][w + 1];
    dp[0][0] = 1;
    for (int i = 0; i < h; i++)
      for (int j = 0; j < w; j++) {
        if (r[i] >= j + 1 && c[j] >= i + 1) dp[i + 1][j + 1] = (dp[i + 1][j + 1] + dp[i][j]) % 1000000007;
        if (r[i] >= j && c[j] >= i + 1) dp[i][j + 1] = (dp[i][j + 1] + dp[i][j]) % 1000000007;
      }

    System.out.println(dp[h][w]);
  }
}
