n = int(input())
p = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
m = int(input())
c = [int(x) for x in input().split()]
for j in c:
    ans = -1
    for i in range(n):
        if j == a[i] or j == b[i]:
            if ans == -1 or ans > p[i]:
                ans = p[i]
    print(ans)
