#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,n) for(int i=a;i<n;++i)
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(0);cout.tie(0);
    int n;cin>>n;
    vector<int> a(n);
    rep(i,0,n)cin>>a[i];
    sort(a.begin(),a.end());
    int ans=0,ev=-1,od=-1;
    rep(i,0,n){
    	if(a[i]%2==0){
    		if(ev==-1)ev=i;
    		else{
    			od=i;
    			break;
    		}
    	}
    }
    if(od==-1)od=n;
    for(int i=0;i<n;++i){
    	if(i%2==0){
    		if(ev<od)	ans+=a[ev++];
    		else ans+=a[od++];
    	}
    	else{
    		if(od<ev)	ans+=a[od++];
    		else ans+=a[ev++];
    	}
    }
    cout<<ans<<endl;
    return 0;
}
