import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
    private static int n;
    private static int[] deg;
    private static List<Integer>[] adj;
    private static int[] p;
    private static List<Operation> res;
    private static boolean ok = false;

    private static class Operation {
        int a, b, c;

        Operation(int a, int b, int c) {
            this.a = a;
            this.b = b;
            this.c = c;
        }
    }

    private static boolean dfs(int v, int par) {
        boolean isLeaf = true;
        for (int u : adj[v]) {
            if (u != par) {
                isLeaf = false;
                if (!dfs(u, v)) {
                    return false;
                }
                p[u] = v;
            }
        }
        if (isLeaf) {
            deg[par]++;
        }
        return true;
    }

    private static void add(int v, int par, int x) {
        res.add(new Operation(v, par, x));
        int a = v;
        int b = par;
        while (a != b) {
            if (deg[a] % 2 == 0) {
                int tmp = a;
                a = b;
                b = tmp;
            }
            res.add(new Operation(a, b, x / 2));
            deg[a]--;
            deg[b]--;
            a = p[a];
        }
    }

    private static boolean solve(int[] val) {
        res = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (deg[i] % 2 != 0) {
                return false;
            }
        }
        int st = -1;
        int maxdeg = 0;
        for (int i = 0; i < n; i++) {
            if (deg[i] > maxdeg) {
                maxdeg = deg[i];
                st = i;
            }
        }
        for (int i = 0; i < n; i++) {
            if (i != st && deg[i] % 2 != 0) {
                int lca = -1;
                int a = i;
                int b = st;
                while (a != b) {
                    if (deg[a] % 2 == 0) {
                        int tmp = a;
                        a = b;
                        b = tmp;
                    }
                    a = p[a];
                    b = p[b];
                    lca = a;
                }
                add(i, lca, val[i] / 2);
                add(st, lca, (val[st] - val[i]) / 2);
            }
        }
        return true;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        deg = new int[n];
        adj = new List[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            int c = Integer.parseInt(st.nextToken());
            adj[a].add(b);
            adj[b].add(a);
            deg[a]++;
            deg[b]++;
        }
        p = new int[n];
        dfs(0, 0);
        int[] val = new int[n];
        for (int i = 0; i < n; i++) {
            val[i] = adj[i].get(0).intValue() % 2;
        }
        ok = solve(val);
        if (ok) {
            System.out.println("YES");
            System.out.println(res.size());
            for (Operation op : res) {
                System.out.println((op.a + 1) + " " + (op.b + 1) + " " + op.c);
            }
        } else {
            System.out.println("NO");
        }
    }
}
