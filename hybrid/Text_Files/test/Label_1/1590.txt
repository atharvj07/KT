#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        string s;
        cin >> n >> s;
        int cnt[26], odd = 0;
        memset(cnt, 0, sizeof cnt);
        for (char c : s) cnt[c - 'a']++;
        for (int i = 0; i < 26; ++i) odd += cnt[i] % 2;
        if (n % 2 == 0)
            cout << (odd > 0 ? -1 : 0) << endl;
        else
            cout << (odd <= 1 ? 0 : -1) << endl;
    }
    return 0;
}
