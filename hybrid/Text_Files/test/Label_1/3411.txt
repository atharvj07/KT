#include <iostream>
#include <vector>

struct pt {
    long long x, y;
};

int main() {
    int test_cases;
    std::cin >> test_cases;
    while (test_cases--) {
        int n;
        std::cin >> n;
        std::vector<long long> v(n);
        for (int i = 0; i < n; i++) std::cin >> v[i];
        int m;
        std::cin >> m;
        std::vector<long long> w(m);
        for (int i = 0; i < m; i++) std::cin >> w[i];
        pt ans;
        long long dist = 0;
        for (long long i = 0; i < n; i++) {
            for (long long j = 0; j < m; j++) {
                if (v[i] - w[j] >= 0) {
                    if ((v[i] - w[j]) % 2 == 1) {
                        long long val = (v[i] - w[j] - 1) / 2;
                        if (val >= 0 && val % 2 == 0) {
                            ans = {val, val};
                            dist = (v[i] - w[j]) / 2;
                        }
                    } else {
                        long long val = (v[i] - w[j]) / 2;
                        if (val >= 0 && val % 2 == 0) {
                            ans = {val, val};
                            dist = (v[i] - w[j]) / 2;
                        }
                    }
                } else {
                    long long val = (v[i] - w[j] + 1) / -2;
                    if (val >= 0 && val % 2 == 0) {
                        ans = {-val, val};
                        dist = -(v[i] - w[j]) / 2;
                    }
                }
                if (dist) break;
            }
            if (dist) break;
        }
        long long cnt = 0;
        for (long long i = 0; i < n; i++) {
            if (ans.x + ans.y <= v[i] && -ans.x + ans.y >= w[0]) cnt++;
        }
        std::cout << cnt << '\n';
    }
    return 0;
}
