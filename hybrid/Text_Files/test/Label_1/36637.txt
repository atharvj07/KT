for _ in range(int(input())):
    n, m = map(int, input().split())
    a = sorted(map(int, input().split()))
    i = 0
    ans = 0
    while n:
        if i == m or n < a[i]:
            ans = -1
            break
        if n >= a[i]:
            n -= a[i]
            ans += 1
            if n >= a[i]:
                i += 1
    print(ans)
