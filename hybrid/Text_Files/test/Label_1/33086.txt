import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] m = new long[n];
        long total = 0;
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
            total += m[i];
        }
        Arrays.sort(m);
        long avg = total / n;
        long ans = 0;
        for (long mi : m) {
            if (mi > avg) {
                ans += mi - avg;
            }
        }
        System.out.println(ans);
    }
}
