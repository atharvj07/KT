#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int cnt = 0;
        for (int i = 0; i + 1 < n; ++i) {
            cnt += s[i] > s[i + 1];
        }
        cout << cnt + 1 << "\n";
    }
    return 0;
}
