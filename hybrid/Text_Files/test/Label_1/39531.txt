n, k = map(int, input().split())
a = [int(x) for x in input().split()]
q = int(input())
for i in range(q):
  x = int(input())
  ans = -1
  for j in range(len(a)):
    if x >= a[j]:
      b = x // a[j]
      c = x % a[j]
      if c == 0:
        ans = min(ans, b) if ans != -1 else b
      if len(a) > j + 1:
        d = c // a[j + 1]
        e = c % a[j + 1]
        if e == 0:
          ans = min(ans, b + d) if ans != -1 else b + d
  print(ans if ans != -1 and ans <= k else -1)
