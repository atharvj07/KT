import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] groups = new int[n];
        for (int i = 0; i < n; i++) {
            groups[i] = sc.nextInt();
        }
        Arrays.sort(groups);
        int taxis = 0;
        int curr = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (curr + groups[i] <= 4) {
                curr += groups[i];
            } else {
                taxis++;
                curr = groups[i];
            }
        }
        System.out.println(taxis);
    }
}
