n, k = map(int, input().split())
a = list(map(int, input().split()))

actions = []
for i in range(1, n):
    desired_height = a[i-1] + k
    if a[i] < desired_height:
        actions.append(f"+ {i+1} {desired_height - a[i]}")
        a[i] = desired_height
    elif a[i] > desired_height:
        actions.append(f"- {i+1} {a[i] - desired_height}")
        a[i] = desired_height

print(len(actions))
for action in actions:
    print(action)
