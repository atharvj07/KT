#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<long long> a(n);
        for (int i = 0; i < n; i++) { cin >> a[i]; }
        long long g = a[0];
        for (int i = 1; i < n; i++) { g = __gcd(g, a[i]); }
        int ans = 0;
        for (auto x : a) { ans += x % g != 0; }
        if (ans % 2 == 0) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}
