#include <iostream>
#include <algorithm>
using namespace std;

const int MAXN = 1e5 + 5;

int n, m, x[MAXN], y[MAXN], a[MAXN], b[MAXN];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i] >> x[i];
    cin >> m;
    for (int i = 1; i <= m; i++) cin >> b[i] >> y[i];
    sort(a + 1, a + n + 1);
    sort(b + 1, b + m + 1);
    sort(x + 1, x + n + 1, greater<int>());
    sort(y + 1, y + m + 1, greater<int>());
    long long ans = 0;
    int i = 1, j = 1;
    while (i <= n && j <= m) {
        if (a[i] != b[j]) {
            ans += x[i] + y[j];
            i++, j++;
        } else
            i++;
    }
    cout << ans << endl;
    return 0;
}
