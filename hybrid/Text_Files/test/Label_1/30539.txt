import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = in.nextLong();
        long[] t = new long[n];
        for (int i = 0; i < n; i++) t[i] = in.nextLong();
        Arrays.sort(a);
        Arrays.sort(t);
        long res = 0;
        for (int i = 0; i < n - 1; i++)
            if (a[i] == a[i + 1])
                res += t[i] * (a[i] - (i > 0 ? a[i - 1] : 0));
        System.out.println(res);
    }
}
