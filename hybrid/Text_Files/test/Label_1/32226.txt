import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int R = sc.nextInt();
        int G = sc.nextInt();
        int B = sc.nextInt();
        int N = sc.nextInt();
        long ans = 0;
        for (int i = 0; i <= N / R; i++) {
            for (int j = 0; j <= N / G; j++) {
                int k = (N - i * R - j * G) / B;
                if (i * R + j * G + k * B == N) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
