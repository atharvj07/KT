import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int L = scanner.nextInt();
        int[] flavors = new int[N];
        for (int i = 0; i < N; i++) {
            flavors[i] = L + i;
        }
        Arrays.sort(flavors); // sort flavors to find the middle
        int sum = 0;
        for (int i = 0; i < N; i++) {
            sum += flavors[i];
        }
        int middle = N % 2 == 0 ? (flavors[N / 2 - 1] + flavors[N / 2]) / 2 : flavors[N / 2];
        System.out.println(sum - middle);
    }
}
