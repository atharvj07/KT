#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;
  std::string s;
  std::cin >> s;
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (s[i] == 'F') {
      ++cnt;
    } else {
      --cnt;
    }
  }
  if (cnt > 0) {
    std::cout << "YES" << std::endl;
  } else {
    std::cout << "NO" << std::endl;
  }
  return 0;
}
