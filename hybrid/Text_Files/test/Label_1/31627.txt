import java.util.*;

class Main {
    static int[][][] m;
    static int n, k, x, y, z;
    static boolean[][][] vis;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        k = in.nextInt();
        m = new int[n + 2][k + 2][k + 2];
        vis = new boolean[n + 2][k + 2][k + 2];
        for (int i = 1; i <= n; i++) {
            String[] a = new String[k];
            in.nextLine();
            for (int j = 0; j < k; j++) {
                a[j] = in.nextLine();
                for (int l = 0; l < k; l++) {
                    m[i][j + 1][l + 1] = Integer.parseInt(a[j].charAt(l) + "");
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= k; j++) {
                for (int l = 1; l <= k; l++) {
                    if (!vis[i][j][l] && dfs(i, j, l)) ans++;
                }
            }
        }
        System.out.println(ans);
    }

    static boolean dfs(int i, int j, int l) {
        if (vis[i][j][l]) return false;
        vis[i][j][l] = true;
        boolean f = true;
        if (m[i][j][l] == 0) return false;
        if (i + 1 <= n && m[i + 1][j][l] > 0) f &= dfs(i + 1, j, l);
        if (j + 1 <= k && m[i][j + 1][l] > 0) f &= dfs(i, j + 1, l);
        if (l + 1 <= k && m[i][j][l + 1] > 0) f &= dfs(i, j, l + 1);
        return f;
    }
}
