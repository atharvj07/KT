#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
const int N = 5e5 + 5;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    string s;
    cin >> s;
    int n = s.size();
    vector<int> p(n + 1);
    for (int i = 1; i <= n; i++) {
        p[i] = p[i - 1];
        if (s[i - 1] == 'A' || s[i - 1] == 'E' || s[i - 1] == 'I' || s[i - 1] == 'O' || s[i - 1] == 'U' || s[i - 1] == 'Y') p[i] += 1;
    }
    double ans = 0.0;
    for (int i = 0; i < n; i++) {
        ans += 1.0 * p[i + 1] * (n - i) / n;
    }
    cout << fixed << setprecision(8) << ans << '\n';
    return 0;
}
