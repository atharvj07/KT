n = int(input())
if n == 1:
    print('No')
else:
    print('Yes')
    if n % 2 == 0:
        print(n // 2 - 1, *[i for i in range(2, n + 1, 2)])
        print(n // 2, 1, *[i for i in range(3, n + 1, 2)])
    else:
        print(n // 2, *[i for i in range(2, n + 1, 2)])
        print(n // 2 + 1, 1, *[i for i in range(3, n + 1, 2)])
