n = int(input())
a = list(map(int, input().split()))
if 0 in a or sum(a) < n - 1:
    print(-1)
else:
    ind = [i for i in range(n) if a[i] == 0]
    adj = [[] for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            if a[i] > 0 and a[j] > 0 and (i not in ind or j not in ind):
                if len(adj[i]) < a[i]:
                    adj[i].append(j)
                    adj[j].append(i)
    vis = [0] * n

    def dfs(u):
        vis[u] = 1
        for v in adj[u]:
            if vis[v] == 0:
                dfs(v)

    dfs(0)
    if all(vis):
        cnt = 0
        for i in range(n):
            for j in adj[i]:
                if vis[j] == 1:
                    print(i + 1, j + 1)
                    cnt += 1
        print(cnt)
    else:
        print(-1)
