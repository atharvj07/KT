#include<bits/stdc++.h>
using namespace std;

int n,a[105],f[105];
long long dp[105][105];

long long solve(int pos, int sum){
    if(pos==n+1){
        if(sum%2){
            return 1;
        }
        return 0;
    }
    if(dp[pos][sum]!=-1)return dp[pos][sum];

    long long ans=0;
    for(int i=0;i<=a[pos];i++){
        if(sum+i<=100)
            ans+=solve(pos+1,sum+i);
    }
    return dp[pos][sum]=ans;
}

main(){
    int i;
    cin>>n;
    for(i=1;i<=n;i++)cin>>a[i];
    memset(dp,-1,sizeof dp);
    cout<<solve(1,f[1]);
}
