#include <bits/stdc++.h>
using namespace std;

const int MAXN = 110, MAXM = 110, MAXQ = 10010;
int n, m, q;
vector<int> a[MAXN];
vector<bool> v[MAXN];

int main() {
  cin >> n >> m >> q;
  for (int i = 1; i <= n; i++) {
    a[i].resize(m);
    v[i].resize(m);
  }
  while (q--) {
    int op, x, y, z;
    cin >> op;
    if (op <= 2)
      cin >> x;
    else
      cin >> x >> y >> z;
    if (op <= 2) {
      if (op == 1)
        reverse(v[x].begin(), v[x].end());
      if (op == 2)
        for (int i = 1; i <= n; i++) {
          if (v[i][x])
            a[i][x] = z;
          v[i][x] = 1 - v[i][x];
        }
    } else
      a[x][y] = z;
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cout << a[i][j] << ' ';
    }
    cout << '\n';
  }
}
