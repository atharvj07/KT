import sys

n = int(sys.stdin.readline().strip())
points = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]
vectors = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]

# calculate the centroid of the points
centroid = [sum(x) / n for x in zip(*points)]

# calculate the sum of the vectors
sum_vectors = [sum(x) for x in zip(*vectors)]

# calculate the treasure coordinates by adding the centroid and the sum of the vectors
treasure = [c + s for c, s in zip(centroid, sum_vectors)]

print(*treasure)
