import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();

        String ans = "";
        for (int i = 0; i < 32; i++) {
            if ((w & (1 << i)) != 0) {
                if (ans.length() != 0) {
                    ans += "0";
                }
                if (i % 2 == 1) {
                    ans += "-";
                } else {
                    ans += "+";
                }
            }
        }
        System.out.println(ans);
    }
}
