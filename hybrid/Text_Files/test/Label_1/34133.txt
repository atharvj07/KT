import java.io.*;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        Graph graph = new Graph(n, m);
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken()) - 1;
            int v = Integer.parseInt(st.nextToken()) - 1;
            graph.addEdge(u, v);
        }
        st = new StringTokenizer(br.readLine());
        int[] parity = new int[n];
        for (int i = 0; i < n; i++) {
            parity[i] = Integer.parseInt(st.nextToken());
        }
        if (!graph.checkPath(parity)) {
            System.out.println(-1);
            return;
        }
        System.out.println(graph.path.size());
        for (int node : graph.path) {
            System.out.print((node + 1) + " ");
        }
    }

    static class Graph {
        int n;
        int m;
        ArrayList<Integer>[] adjList;
        boolean[] visited;
        ArrayList<Integer> path;

        Graph(int n, int m) {
            this.n = n;
            this.m = m;
            adjList = new ArrayList[n];
            for (int i = 0; i < n; i++) {
                adjList[i] = new ArrayList<>();
            }
            visited = new boolean[n];
            path = new ArrayList<>();
        }

        void addEdge(int u, int v) {
            adjList[u].add(v);
            adjList[v].add(u);
        }

        boolean checkPath(int[] parity) {
            for (int i = 0; i < n; i++) {
                if (parity[i] == 0 && adjList[i].size() % 2 == 1) {
                    return false;
                } else if (parity[i] == 1 && adjList[i].size() % 2 == 0) {
                    return false;
                }
            }
            return dfs(0, -1, parity);
        }

        boolean dfs(int node, int parent, int[] parity) {
            if (path.size() == n) {
                return true;
            }
            visited[node] = true;
            path.add(node);
            for (int next : adjList[node]) {
                if (!visited[next]) {
                    if (dfs(next, node, parity)) {
                        return true;
                    }
                } else if (parent != next) {
                    return false;
                }
            }
            path.remove(path.size() - 1);
            visited[node] = false;
            return false;
        }
    }
}
