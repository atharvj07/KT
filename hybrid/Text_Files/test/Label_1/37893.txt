n, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
b = []
for i in range(m):
    b.append(m * (i + 1))
for i in a:
    for j in b:
        if i < j:
            break
        if j % i == 0:
            b.remove(j)
b.sort()
for k in range(1, n + 1):
    print(a[k - 1] * b[k - 1])
