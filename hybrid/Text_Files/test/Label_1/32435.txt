import java.io.*;
import java.util.*;

public class Main {  
  static int n, m;
  static Long x, k, y;

  static class Node {
    long power;
    int pos;

    Node(long power, int pos) {
      this.power = power;
      this.pos = pos;
    }
  }

  public static void main(String[] args) throws IOException {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    m = in.nextInt();
    x = in.nextLong();
    k = in.nextLong();
    y = in.nextLong();
    //sort by power
    Node[] powersA = new Node[n];
    for (int i = 0; i < n; i++) {
      powersA[i] = new Node(in.nextLong(), i + 1);
    }
    Arrays.sort(powersA, (a, b) -> Long.compare(a.power, b.power));

    Node[] powersB = new Node[m];
    for (int i = 0; i < m; i++) {
      powersB[i] = new Node(in.nextLong(), i + 1);
    }
    Arrays.sort(powersB, (a, b) -> Long.compare(a.power, b.power));

    long ans = 0;
    if (!Arrays.equals(powersA, powersB)) {
      System.out.println(-1);
    } else {
      for (int i = 0; i < n; i++) {
        if (powersA[i].pos == powersB[i].pos) {
          continue;
        }
        int right = powersB[i].pos;
        int left = powersA[i].pos;
        while (right > left) {
          //fireballs
          if (right - left <= k) {
            ans += (right - left) * x;
            powersA[i].pos = right;
            break;
          } else {
            ans += x * k;
            left += k;
          }
          //berserk
          if (right > left + 1) {
            ans += y;
            left++;
          } else {
            break;
          }
        }
      }
      System.out.println(ans);
    }
  }
}
