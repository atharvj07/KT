n=int(input())
a=list(map(int,input().split()))
l=[0]*n
r=[0]*n
for i in range(n):
    l[a[i]-1]=i
    r[a[i]-1]=i
for i in range(1,n):
    if a[i]>a[i-1]:
        r[a[i]-1]=max(r[a[i]-1],r[a[i-1]-1])
for i in range(n-2,-1,-1):
    if a[i]>a[i+1]:
        l[a[i]-1]=min(l[a[i]-1],l[a[i+1]-1])
for i in range(n):
    print(*a[l[i]:r[i]+1])
