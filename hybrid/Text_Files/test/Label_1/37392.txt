import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort(reverse = True)
# prefix sum
presum = [0]
for _a in a:
    presum.append(presum[-1]+_a)

ans = [0 for i in range(n+1)]
for k in range(1,n+1):
    l, r = 0, m*k-1
    cur = 0
    if r >= n:
        r = n-1
    for j in range(m):
        cur += a[l+j]*(1+j)
    ans[k] = cur
    while r < n and l<m*k-1:
        l += 1
        r += 1
        cur -= (l-1)*a[l-1]
        cur += r*a[r]
        ans[k] = min(ans[k],cur)
for _ans in ans[1:]:
    print(_ans,end=' ')
