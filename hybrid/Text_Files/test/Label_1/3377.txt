#include <iostream>
using namespace std;
int main() {
    long long n, k;
    cin >> n >> k;
    if (k == 1) {
        cout << 1 << endl;
        return 0;
    }
    int cnt = 0;
    for (int i = 1;; i++) {
        cnt += i / k * (i % k);
        if (cnt == n) { cout << i << endl; return 0; }
        if (cnt + i / k * (i % k) > n) { cout << i - 1 << endl; return 0; }
    }
}
