from sys import stdin
from collections import defaultdict
read=lambda:stdin.readline().strip()
n=int(read())
dpx,dpy=defaultdict(list),defaultdict(list)
for x,y in zip(map(int,read().split()),map(int,read().split())):
    dpx[x].append(y)
    dpy[y].append(x)
ans=0
for x in dpx:
    dpx[x].sort()
    l=len(dpx[x])
    for i in range(l):
        for j in range(i+1,l):
            ans+=len(dpy[dpx[x][i]])*len(dpy[dpx[x][j]])
print(ans)
