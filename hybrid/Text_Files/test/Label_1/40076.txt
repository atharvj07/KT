import sys

def main():
    input = sys.stdin.readline
    q = int(input())
    for _ in range(q):
        h, n = map(int, input().split())
        p = list(map(int, input().split()))
        if h <= 3:
            print(0)
            continue
        ans = 0
        for i in range(n):
            if p[i] <= 2:
                break
            if p[i] - p[i - 1] > 2:
                ans += 1
        print(ans)

main()
