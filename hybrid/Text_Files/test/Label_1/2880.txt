#include <iostream>
#include <vector>
using namespace std;

const int dr[] = {0, 0, 1, -1, -1, 1, -1, 1};
const int dc[] = {1, -1, 0, 0, 1, -1, -1, 1};

bool is_ok(int n, vector<string>& b, vector<pair<int, int>>& v) {
  if (v.size() == 0) return 0;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      if (b[i][j] == 'o') {
        bool ok = 0;
        for (auto p : v) {
          int ii = i + p.first, jj = j + p.second;
          if (ii >= 0 && ii < n && jj >= 0 && jj < n && b[ii][jj] == 'x')
            ok = 1;
        }
        if (!ok) return 0;
      }
  return 1;
}

int main() {
  int n;
  cin >> n;
  vector<string> b(n);
  for (auto& x : b) cin >> x;
  vector<pair<int, int>> v;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < n; j++)
      if (b[i][j] == '.') {
        vector<pair<int, int>> tmp;
        for (int k = 0; k < 8; k++) {
          int ii = i + dr[k], jj = j + dc[k];
          if (ii < 0 || ii >= n || jj < 0 || jj >= n) continue;
          if (b[ii][jj] == 'x') tmp.push_back({dr[k], dc[k]});
        }
        if (is_ok(n, b, tmp)) {
          v = tmp;
        }
      }
  cout << (is_ok(n, b, v) ? "YES" : "NO") << '\n';
  if (v.size()) {
    for (int k = 0; k < n; k++) {
      for (int i = -n + 1; i <= n - 1; i++, cout << '\n')
        for (int j = -n + 1; j <= n - 1; j++) {
          bool ok = 0;
          for (auto p : v)
            if (i == p.first && j == p.second) ok = 1;
          cout << (ok ? "x" : ".");
        }
    }
  }
  return 0;
}
