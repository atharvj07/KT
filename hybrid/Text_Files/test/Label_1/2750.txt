#include <iostream>
#include <unordered_map>

using namespace std;

const int MXN = 1e5 + 5;
const int INF = 1e9 + 7;

int main() {
    unordered_map<int, pair<int, int>> mp;
    int n;
    cin >> n;
    int time = INF;
    while (n--) {
        int type, t, x;
        cin >> type >> t >> x;
        time = min(time, t);
        if (type == 1)
            mp[x].second++;
        if (type == 2)
            mp[x].second--;
        if (type == 3)
            cout << (t >= mp[x].first ? mp[x].second : 0) << endl;
        mp[x].first = min(mp[x].first, t);
    }
    return 0;
}
