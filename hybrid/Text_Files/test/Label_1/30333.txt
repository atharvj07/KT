import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in=new Scanner(System.in);
        int n=in.nextInt(),ans=n;
        HashMap<Character,Integer>c=new HashMap<>();
        HashMap<Character,Integer>v=new HashMap<>();
        for(int i=0;i<n;i++){
            String s=in.next();
            c.put(s.charAt(0),c.getOrDefault(s.charAt(0),0)+1);
            v.put(s.charAt(1),v.getOrDefault(s.charAt(1),0)+1);
        }
        ans=Math.min(ans,Math.max(c.size(),v.size()));
        System.out.println(ans);
    }
}
