import java.util.*;

public class MostFrequentSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        Set<String> strings = new HashSet<>();
        for (int i = 0; i < n; i++) {
            strings.add(scanner.nextLine());
        }

        String result = findGoodString(strings);
        System.out.println(result);
    }

    private static String findGoodString(Set<String> strings) {
        // Build a trie to store all substrings of all strings in the set.
        Trie trie = new Trie();
        for (String s : strings) {
            for (int i = 0; i < s.length(); i++) {
                for (int j = i + 1; j <= s.length(); j++) {
                    trie.insert(s.substring(i, j));
                }
            }
        }

        // Find the most frequent substring in the trie.
        String mostFrequentSubstring = trie.findMostFrequentSubstring();

        // Check if all strings in the set are the most frequent substrings.
        boolean isGoodString = true;
        for (String s : strings) {
            if (!s.contains(mostFrequentSubstring)) {
                isGoodString = false;
                break;
            }
        }

        // If there is a good string, return it. Otherwise, return "NO".
        return isGoodString ? mostFrequentSubstring : "NO";
    }

    private static class Trie {
        private TrieNode root;

        public Trie() {
            root = new TrieNode();
        }

        public void insert(String word) {
            TrieNode current = root;
            for (char c : word.toCharArray()) {
                TrieNode child = current.children.get(c);
                if (child == null) {
                    child = new TrieNode();
                    current.children.put(c, child);
                }
                current = child;
                current.count++;
            }
            current.isWord = true;
        }

        public String findMostFrequentSubstring() {
            return findMostFrequentSubstringHelper(root, "");
        }

        private String findMostFrequentSubstringHelper(TrieNode node, String current) {
            if (node == null) {
                return "";
            }

            if (node.isWord) {
                String candidate = current + node.c;
                if (node.count > root.count) {
                    return candidate;
                } else if (node.count == root.count && candidate.compareTo(findMostFrequentSubstringHelper(root, "")) < 0) {
                    return candidate;
                }
            }

            String result = "";
            for (char c : node.children.keySet()) {
                String candidate = findMostFrequentSubstringHelper(node.children.get(c), current + c);
                if (candidate.length() > result.length()) {
                    result = candidate;
                } else if (candidate.length() == result.length() && candidate.compareTo(result) < 0) {
                    result = candidate;
                }
            }

            return result;
        }
    }

    private static class TrieNode {
        private char c;
        private int count;
        private boolean isWord;
        private Map<Character, TrieNode> children;

        public TrieNode() {
            count = 0;
            isWord = false;
            children = new HashMap<>();
        }
    }
}
