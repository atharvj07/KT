import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int c = in.nextInt();
        int t = 0;
        while (a * (t + 1) > b * t && t <= c) {
            t++;
        }
        System.out.println(t);
    }
}
