t = int(input())
for _ in range(t):
    n = int(input())
    candies = list(map(int, input().split()))
    oranges = list(map(int, input().split()))
    
    min_candies = min(candies)
    min_oranges = min(oranges)
    
    moves = 0
    for i in range(n):
        diff_candies = candies[i] - min_candies
        diff_oranges = oranges[i] - min_oranges
        
        moves += max(diff_candies, diff_oranges)
    
    print(moves)
