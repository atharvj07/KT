#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

typedef long long ll;
typedef vector<int> vi;
typedef pair<int,int> pi;
typedef vector<pi> vpi;

#define pb push_back
#define all(x) (x).begin(),(x).end()
#define sz(x) (int)(x).size()

#define FOR(i,a,b) for (auto i = (a); i < (b); ++i)
#define ROF(i,a,b) for (auto i = (b)-1; i >= (a); --i)
#define trav(a,x) for (auto& a: x)

const int MX = 1e5+5;
vi adj[MX];
int n;

bool dfs(int v, int p, int x, vi& w) {
    w[v] = x;
    trav(u,adj[v]) {
        if (u==p) continue;
        if (w[u]==0 && dfs(u,v,x^1,w)) return true;
        else if (w[u] == w[v]) return true;
    }
    return false;
}

void solve() {
    cin >> n;
    FOR(i,0,n-1) {
        int a,b; cin >> a >> b;
        adj[a].pb(b); adj[b].pb(a);
    }

    int worst = 0; bool no_sol = false;
    vi w(n);
    FOR(i,1,n+1) if (w[i]==0) {
        if (dfs(i,i,0,w)) {
            no_sol = true;
            break;
        }
        worst++;
    }

    int best = 0;
    trav(x,w) if (x) best++;

    if (no_sol) cout << -1 << endl;
    else cout << best << ' ' << worst << endl;
}

int main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    solve();
    return 0;
}
