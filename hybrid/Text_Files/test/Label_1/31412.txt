import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        boolean odd = true;
        for (int i = 0; i < n; i++) {
            if (odd) {
                if (s.charAt(i) == 'L') {
                    System.out.println("No");
                    return;
                }
            } else {
                if (s.charAt(i) == 'R') {
                    System.out.println("No");
                    return;
                }
            }
            odd = !odd;
        }
        System.out.println("Yes");
    }
}
