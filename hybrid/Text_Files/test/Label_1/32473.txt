import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
  private static class GasStation {
    int type;
    int position;

    public GasStation(int type, int position) {
      this.type = type;
      this.position = position;
    }
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] line = br.readLine().split(" ");
    int e = Integer.parseInt(line[0]);
    int s = Integer.parseInt(line[1]);
    int n = Integer.parseInt(line[2]);
    int m = Integer.parseInt(line[3]);

    List<GasStation> gasStations = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      line = br.readLine().split(" ");
      gasStations.add(new GasStation(Integer.parseInt(line[0]), Integer.parseInt(line[1])));
    }

    line = br.readLine().split(" ");
    int[] startPositions = new int[line.length];
    for (int i = 0; i < line.length; i++) {
      startPositions[i] = Integer.parseInt(line[i]);
    }

    for (int startPosition : startPositions) {
      int[][] dp = new int[2][s + 1];
      for (int i = 0; i < 2; i++) {
        Arrays.fill(dp[i], Integer.MAX_VALUE);
      }
      dp[0][0] = 0;
      dp[1][s] = 0;

      for (GasStation gasStation : gasStations) {
        if (gasStation.position >= startPosition) {
          break;
        }
        for (int j = 0; j < s; j++) {
          if (dp[0][j] != Integer.MAX_VALUE || dp[1][j] != Integer.MAX_VALUE) {
            int diff = startPosition - gasStation.position;
            if (diff <= j) {
              dp[0][j - diff] = Math.min(dp[0][j - diff], dp[0][j]);
              dp[1][j - diff + gasStation.type] = Math.min(dp[1][j - diff + gasStation.type], dp[0][j]);
            }
          }
        }
      }

      for (int j = 0; j < s; j++) {
        if (dp[0][j] != Integer.MAX_VALUE) {
          dp[0][j + 1] = Math.min(dp[0][j + 1], dp[0][j] + 1);
        }
        if (dp[1][j] != Integer.MAX_VALUE) {
          dp[1][j + 1] = Math.min(dp[1][j + 1], dp[1][j] + 1);
        }
      }

      boolean canReach = false;
      int minRegular92 = Integer.MAX_VALUE;
      int minPremium95 = Integer.MAX_VALUE;
      for (int j = 0; j <= s; j++) {
        if (dp[0][j] != Integer.MAX_VALUE && startPosition + j >= e) {
          canReach = true;
          if (dp[0][j] < minRegular92 || (dp[0][j] == minRegular92 && dp[1][j] < minPremium95)) {
            minRegular92 = dp[0][j];
            minPremium95 = dp[1][j];
          }
        }
        if (dp[1][j] != Integer.MAX_VALUE && startPosition + j >= e) {
          canReach = true;
          if (dp[1][j] < minPremium95 || (dp[1][j] == minPremium95 && dp[0][j] < minRegular92)) {
            minRegular92 = dp[0][j];
            minPremium95 = dp[1][j];
          }
        }
      }

      if (!canReach) {
        System.out.println("-1 -1");
      } else {
        System.out.println(minRegular92 + " " + minPremium95);
      }
    }
  }
}
