from collections import defaultdict
import sys
input = sys.stdin.readline

n, m = map(int, input().split())
s = input().strip()
graph = defaultdict(list)
for _ in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)

dp = [[[-1] * 3 for _ in range(n+1)] for _ in range(2)]

def dfs(node, par, state):
    if dp[state][node][par] != -1:
        return dp[state][node][par]

    ans = 0
    for nxt in graph[node]:
        if nxt != par:
            if s[node-1] == s[nxt-1]:
                ans = max(ans, dfs(nxt, node, 0))
            else:
                ans = max(ans, dfs(nxt, node, 1))
    if state == 0:
        ans = max(ans, dfs(node, par, 1))

    dp[state][node][par] = ans + 1
    return dp[state][node][par]

res = 0
for i in range(1, n+1):
    res = max(res, dfs(i, -1, 0))
    res = max(res, dfs(i, -1, 1))

if res > n:
    print(-1)
else:
    print(res)
