#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, d;
  cin >> n >> d;
  
  vector<int> t(n);
  for (int i = 0; i < n; ++i) {
    cin >> t[i];
  }
  
  int total_time = 0;
  int jokes = 0;
  for (int i = 0; i < n; ++i) {
    if (total_time + t[i] + 10 <= d) {
      total_time += t[i] + 10;
    } else {
      break;
    }
    jokes++;
  }
  
  if (total_time != d) {
    cout << -1 << endl;
  } else {
    cout << jokes << endl;
  }
  
  return 0;
}
