#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> b(n);
    for (int i = 0; i < n; ++i) cin >> b[i];
    long long sum = accumulate(b.begin(), b.end(), 0LL);
    if (sum % n != 0) {
      cout << "NO" << endl;
      continue;
    }
    long long x = sum / n;
    vector<bool> vis(n, false);
    vector<long long> a(n);
    for (int i = 0; i < n; ++i) {
      if (!vis[i]) {
        long long cur = b[i];
        a[i] = cur;
        vis[i] = true;
        int j = (i + 1) % n;
        while (cur < x) {
          if (!vis[j]) {
            a[j] = x - cur;
            cur = x;
            vis[j] = true;
          }
          j = (j + 1) % n;
        }
        if (cur > x) {
          cout << "NO" << endl;
          goto outer;
        }
      }
    }
    cout << "YES" << endl;
    for (int i = 0; i < n; ++i) cout << a[i] << " ";
    cout << endl;
outer:;
  }
  return 0;
}
