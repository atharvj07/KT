import math
n=int(input())
x=[0]*n
y=[0]*n
for i in range(n):
    a,b=map(int,input().split())
    x[i]=a;y[i]=abs(b)
x.sort();y.sort()
if len(x)==1:
    if n==1:print(y[0]/2)
    else:print(-1)
else:
    if y[0]==y[-1]:print(-1)
    else:
        d=y[-1]-y[0]
        if n==2:print(d/2)
        else:
            ans=(x[n//2+1]-x[n//2]+y[-1]-y[0])/4
            print(ans)
