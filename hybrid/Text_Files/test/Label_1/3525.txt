#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin >> s;
    int sz = s.size();
    int op = 0, cl = 0;
    for(char ch: s){
        if(ch == '(') op++;
        else if(ch == ')') cl++;
    }
    vector<int> ans;
    int rem = op - cl;
    for(char ch: s){
        if(ch == '#'){
            if(rem > 0){
                ans.push_back(rem);
                rem--;
            }
            else{
                cout << -1;
                return 0;
            }
        }
        else if(ch == '(') rem++;
        else rem--;
        if(rem < 0){
            cout << -1;
            return 0;
        }
    }
    for(int i: ans) cout << i << '\n';
    return 0;
}
