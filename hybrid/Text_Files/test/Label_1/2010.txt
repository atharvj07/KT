#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

struct person {
    long long money;
    int idx;
    bool operator<(const person &b) const { return money < b.money; }
};
int main() {
    int n;
    cin >> n;
    vector<person> peo(n);
    for (int i = 0; i < n; ++i) {
        long long x;
        cin >> x;
        peo[i] = {x, i};
    }
    sort(peo.begin(), peo.end());
    for (int i = 1; i < n; ++i) {
        if (peo[i].money <= peo[i - 1].money) {
            cout << ":(" << endl;
            return 0;
        }
        if (peo[i].idx < peo[i - 1].idx) {
            peo[i].money -= peo[i - 1].money + 1;
            if (peo[i].money <= peo[i - 1].money) {
                cout << ":(" << endl;
                return 0;
            }
        }
    }
    long long sum = 0;
    for (int i = 0; i < n; ++i) {
        peo[i].money += sum;
        sum = peo[i].money;
        cout << peo[i].money << ' ';
    }
    cout << endl;
    return 0;
}
