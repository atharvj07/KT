import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();

        List<Integer>[] adj = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            adj[a].add(b);
        }

        boolean[] visited = new boolean[n + 1];
        int count = 0;

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                int[] cnt = new int[2];
                dfs(i, adj, visited, cnt);
                count += cnt[0] * (cnt[1] + 1);
            }
        }

        System.out.println(count % 1000000007);
    }

    static void dfs(int u, List<Integer>[] adj, boolean[] visited, int[] cnt) {
        visited[u] = true;
        cnt[0]++;
        boolean flag = false;
        for (int v : adj[u]) {
            if (!visited[v]) {
                dfs(v, adj, visited, cnt);
                flag = true;
            }
        }
        if (!flag) {
            cnt[1]++;
        }
    }
}
