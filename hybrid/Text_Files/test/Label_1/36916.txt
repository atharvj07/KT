import sys

n = int(sys.stdin.readline())
seg = sorted([(int(x), int(y)) for x, y in [e.split() for e in sys.stdin.readlines()]])

for j in range(1, len(seg)):
    l, r = seg[j]
    for i in range(j):
        L, R = seg[i]
        if L <= l and r <= R:
            print(j + 1, i + 1)
            exit()
print(-1, -1)
