#include <iostream>
#include <map>
#include <vector>
#include <queue>
#include <set>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;

        vector<vector<int>> adj(n + 1);
        map<pair<int, int>, int> e;

        for (int i = 0; i < n - 1; i++) {
            int a, b;
            cin >> a >> b;
            adj[a].push_back(b);
            adj[b].push_back(a);
            e[{a, b}] = e[{b, a}] = i + 1;
        }

        int s = 1;
        vector<int> d(n + 1, -1), p(n + 1, -1);
        set<pair<int, int>> q;
        q.insert({0, s});
        d[s] = 0;

        while (!q.empty()) {
            int v = q.begin()->second;
            q.erase(q.begin());

            for (int u : adj[v]) {
                if (d[u] == -1) {
                    d[u] = d[v] + 1;
                    p[u] = e[{v, u}];
                    q.insert({d[u], u});
                }
            }
        }

        int k = -1;
        int v = n;
        while (v != s) {
            if (d[v] > k) k = d[v];
            v = p[v];
        }

        cout << k << '\n';
    }
    return 0;
}
