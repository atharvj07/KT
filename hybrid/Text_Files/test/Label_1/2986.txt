#include <iostream>
#include <vector>
#include <set>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    set<int> s;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        s.insert(a[i]);
    }
    int x = 0;
    while ((1 << x) < (int) s.size()) {
        x++;
    }
    x--;
    vector<int> perm;
    int cur = 0;
    for (int i = 0; i < (1 << x); i++) {
        int nx = cur;
        for (int j = 0; j < x; j++) {
            if ((i >> j) & 1) {
                nx ^= (1 << j);
            }
        }
        perm.push_back(nx);
        cur = nx;
    }
    cout << x << endl;
    for (int i = 0; i < (1 << x); i++) {
        cout << perm[i] << " ";
    }
    cout << endl;
    return 0;
}
