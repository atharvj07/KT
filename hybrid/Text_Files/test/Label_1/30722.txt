import java.io.*;
import java.util.*;

public class Main {
	static int[] color;
	static ArrayList<Integer>[] gr;
	static boolean ok;

	public static void dfs(int v, int par, int cnt) {
		if (ok == false) {
			return;
		}
		color[v] = cnt;
		for (Integer u : gr[v]) {
			if (color[u] != 0) {
				if (color[u] != 3 - cnt) {
					ok = false;
					return;
				}
				continue;
			}
			dfs(u, v, 3 - cnt);
		}
	}

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int m = in.nextInt();
		gr = new ArrayList[n + 1];
		color = new int[n + 1];
		ok = true;
		int[] x = new int[n + 1];
		for (int i = 0; i <= n; i++) {
			gr[i] = new ArrayList<Integer>();
		}
		for (int i = 1; i <= m; i++) {
			int a = in.nextInt();
			int b = in.nextInt();
			gr[a].add(b);
			gr[b].add(a);
		}
		for (int i = 1; i <= n; i++) {
			x[i] = in.nextInt();
		}
		for (int i = 1; i <= n; i++) {
			if (color[i] == 0) {
				dfs(i, 0, 1);
			}
		}
		if (ok == false) {
			System.out.println(-1);
			return;
		}
		int ans = 0;
		ArrayList<Integer> res = new ArrayList<Integer>();
		for (int i = 1; i <= n; i++) {
			if (x[i] == 0 && color[i] == 2) {
				ans += 2;
				res.add(i);
				res.add(i);
			} else if (x[i] == 1 && color[i] == 2) {
				ans += 1;
				res.add(i);
			} else if (x[i] == 1 && color[i] == 1) {
				ans += 1;
				res.add(i);
			}
		}
		System.out.println(ans);
		for (Integer v : res) {
			System.out.print(v + " ");
		}
		System.out.println();
		in.close();
	}
}
