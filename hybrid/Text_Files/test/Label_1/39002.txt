from collections import defaultdict

n = int(input())

obelisks = []
for _ in range(n):
    obelisks.append(tuple(map(int,input().split())))

clues = []
for _ in range(n):
    clues.append(tuple(map(int,input().split())))

d = defaultdict(list)
for i in range(n):
    d[obelisks[i]].append(clues[i])

ans = 0, 0
for obelisk in d:
    for clue in d[obelisk]:
        x, y = clue[0]+obelisk[0], clue[1]+obelisk[1]
        if ans[0] == 0 and ans[1] == 0:
            ans = x, y
        if ans[0] != x or ans[1] != y:
            ans = -1, -1
            break

print(*ans)
