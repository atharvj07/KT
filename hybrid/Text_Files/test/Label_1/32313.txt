import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] seg = new int[4 * n];
        init(a, seg, 1, 0, n - 1);
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int t = sc.nextInt();
            int x = sc.nextInt() - 1;
            if (t == 1) {
                update(seg, 1, 0, n - 1, x, Integer.MIN_VALUE);
            } else {
                update(seg, 1, 0, n - 1, x, Integer.MAX_VALUE);
            }
            System.out.println(query(seg, 1, 0, n - 1, 1, n));
        }
    }

    static void init(int[] a, int[] seg, int node, int start, int end) {
        if (start == end) {
            seg[node] = a[start];
            return;
        }
        int mid = (start + end) / 2;
        init(a, seg, node * 2, start, mid);
        init(a, seg, node * 2 + 1, mid + 1, end);
        seg[node] = Math.max(seg[node * 2], seg[node * 2 + 1]);
    }

    static void update(int[] seg, int node, int start, int end, int idx, int val) {
        if (start == end && start == idx) {
            seg[node] = val;
            return;
        }
        int mid = (start + end) / 2;
        if (idx <= mid) {
            update(seg, node * 2, start, mid, idx, val);
        } else {
            update(seg, node * 2 + 1, mid + 1, end, idx, val);
        }
        seg[node] = Math.max(seg[node * 2], seg[node * 2 + 1]);
    }

    static int query(int[] seg, int node, int start, int end, int l, int r) {
        if (start > r || end < l) {
            return 0;
        }
        if (start >= l && end <= r) {
            return seg[node];
        }
        int mid = (start + end) / 2;
        return Math.max(query(seg, node * 2, start, mid, l, r), query(seg, node * 2 + 1, mid + 1, end, l, r));
    }
}
