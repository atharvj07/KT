from math import gcd

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    if len(set(a)) == 1:
        print(0)
        continue
    x = -1
    for i in range(1, n):
        g = gcd(a[i], a[i - 1])
        if x == -1:
            x = g
        elif x != g:
            print(2)
            break
    else:
        print(1)
