import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] freq = new int[1000001];
        for (int i = 0; i < n; i++) {
            freq[a[i]]++;
        }
        int ans = 0;
        for (int i = 1; i <= 1000000; i++) {
            if (freq[i] > i) {
                ans += freq[i] - i;
            }
        }
        System.out.println(ans);
    }
}
