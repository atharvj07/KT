#include<bits/stdc++.h>

using namespace std;

#define max 300001

int main()
{
    int n,m,i,op[max],cl[max];
    string s;
    cin>>n>>m;
    cin>>s;
    for(i=1;i<=n;i++)cl[i]=0;
    for(i=1;i<=n-1;i++)
    {
        if(s[i]==s[i+1] && s[i]=='.')
        {
            cl[i]=1;
            op[i+1]=op[i]+1;
        }
        else op[i+1]=op[i];
    }
    for(i=1;i<=m;i++)
    {
        int x;
        char c;
        cin>>x>>c;
        if(s[x]=='.')
        {
            if(cl[x]==1)
            {
                if(s[x-1]==c && c=='.')
                {
                    cout<<op[x-2]<<endl;
                    op[x]=op[x-2];
                }
                else cout<<op[x-1]<<endl;
            }
            else cout<<op[x]<<endl;
        }
        else
        {
            if(x+1<=n && s[x+1]=='.')
            {
                if(cl[x+1]==1)
                {
                    if(s[x]==c && c=='.')
                    {
                        cout<<op[x]-1<<endl;
                        op[x]=op[x]-1;
                    }
                    else cout<<op[x]<<endl;
                }
                else cout<<op[x]<<endl;
            }
            else cout<<op[x]<<endl;
        }
        s[x]=c;
        cl[x]=0;
        if(x-1>=1)
        {
            if(s[x]==s[x-1] && s[x]=='.')cl[x]=1;
        }
        if(x+1<=n)
        {
            if(s[x]==s[x+1] && s[x]=='.')cl[x+1]=1;
        }
    }
    return 0;
}
