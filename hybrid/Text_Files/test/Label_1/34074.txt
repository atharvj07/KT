import java.util.Arrays;
import java.util.Scanner;

class Main {
  public static void main(String... args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextInt();
    Arrays.sort(a);
    int ans = 0;
    int c = 0;
    for (int i = 0; i < n; i++) {
      c += a[i];
      if (c >= 3) {
        c -= 3;
        ans++;
      }
    }
    System.out.println(ans);
  }
}
