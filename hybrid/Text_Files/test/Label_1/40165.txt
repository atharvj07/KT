n, k = map(int, input().split())
s = input()
a = list(map(int, s))
a.insert(0, 0)
dp = [10 ** 9] * (n + 1)
dp[n] = 0
for i in range(n - 1, -1, -1):
    if a[i]:
        for j in range(i, i + k + 1):
            dp[i] = min(dp[i], dp[max(i, j + 1)] + j)
    else:
        dp[i] = dp[i + 1] + i
print(dp[0])
