import java.util.Scanner;

public class BinaryHeapProperties {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int H = scanner.nextInt();
        int[] A = new int[H + 1];
        for (int i = 1; i <= H; i++) {
            A[i] = scanner.nextInt();
        }

        for (int i = 1; i <= H; i++) {
            System.out.printf("node %d: key = %d, ", i, A[i]);
            if (i > 1) {
                System.out.printf("parent key = %d, ", A[i / 2]);
            }
            if (2 * i <= H) {
                System.out.printf("left key = %d, ", A[2 * i]);
            }
            if (2 * i + 1 <= H) {
                System.out.printf("right key = %d, ", A[2 * i + 1]);
            }
            System.out.println();
        }
    }
}
