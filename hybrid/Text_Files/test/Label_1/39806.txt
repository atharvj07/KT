t = int(input())
for _ in range(t):
    n, k, l = map(int, input().split())
    d = list(map(int, input().split()))
    safe = True
    wait_time = 0

    for i in range(n):
        if d[i] + k > l:
            max_wait = d[i] + k - l
            if max_wait > k:
                safe = False
                break
            wait_time = max(wait_time, 2 * k - max_wait)
        else:
            if wait_time > 0:
                if d[i] + k - (wait_time % (2 * k)) <= l:
                    wait_time = max(0, wait_time - 1)
                else:
                    safe = False
                    break
            else:
                wait_time = 0

    if safe:
        print("Yes")
    else:
        print("No")
