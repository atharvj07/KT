import java.util.Arrays;
import java.util.Scanner;

class Main {
    private static final Scanner scan = new Scanner(System.in);

    private static int XOR(int a, int b) {
        return (a | b) & ~(a & b);
    }

    private static String game(int[] arr) {
        int n = arr.length;
        int p1 = 0;
        for (int num : arr) {
            p1 = XOR(p1, num);
        }

        if (n % 2 == 0) {
            return n > 1 || p1 == 0 ? "WIN" : "LOSE";
        } else {
            return (p1 == 0) ? "DRAW" : "WIN";
        }
    }

    public static void main(String[] args) {
        int q = scan.nextInt();
        while (q-- > 0) {
            int n = scan.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) {
                arr[i] = scan.nextInt();
            }
            System.out.println(game(arr));
        }
    }
}
