#include <iostream>
using namespace std;
int main() {
  string s;
  cin >> s;
  int a = 0, b = 0, ans = 0;
  for (char c : s) {
    if (c == 'a') {
      if (b > 0) ans += 2;
      a++;
    } else if (c == 'b') {
      if (a > 0) ans++, a--;
      b++, ans++;
    }
  }
  cout << ans << '\n';
  return 0;
}
