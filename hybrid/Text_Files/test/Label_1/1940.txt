#include <iostream>
using namespace std;

int Max(int a,int b){
    return a>b?a:b;
}
int main(){
    int m[5],w[5],h,hu;
    cin>>m[0]>>m[1]>>m[2]>>m[3]>>m[4];
    cin>>w[0]>>w[1]>>w[2]>>w[3]>>w[4];
    cin>>h>>hu;
    cout<<500*Max(0.3,1-(m[0]/250.0))-(50*w[0]) +
        1000*Max(0.3,1-(m[1]/250.0))-(50*w[1]) +
        1500*Max(0.3,1-(m[2]/250.0))-(50*w[2]) +
        2000*Max(0.3,1-(m[3]/250.0))-(50*w[3]) +
        2500*Max(0.3,1-(m[4]/250.0))-(50*w[4]) +
        (100*h)-(50*hu);
}
