n, r, avg = map(int, input().split())
a = []
b = []
for _ in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)
a.sort()
b.sort()
ans = 0
cur = sum(a)
for i in range(n):
    if cur / n < avg:
        need = avg * n - cur
        if b[i] <= need:
            ans += b[i]
            cur += b[i]
        else:
            ans += need
            cur += need
            break
print(ans)
