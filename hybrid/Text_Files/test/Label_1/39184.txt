t = int(input())
for _ in range(t):
    n, a, b = map(int, input().split())
    if a == 1:
        print("Yes" if (n-1) % b == 0 else "No")
    else:
        x = 1
        while x <= n:
            if (n - x) % b == 0:
                print("Yes")
                break
            x *= a
        else:
            print("No")
