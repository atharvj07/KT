#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int MAX = 1e5 + 5;
const int dx[] = {1, -1, 0, 0}, dy[] = {0, 0, 1, -1};
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  ll x1, y1, x2, y2, n;
  cin >> x1 >> y1 >> x2 >> y2 >> n;
  string s;
  cin >> s;
  s += s;
  ll dX = x2 - x1, dY = y2 - y1;
  x1 = 0, y1 = 0;
  x2 = abs(dX), y2 = abs(dY);
  bool ok = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == s[0]) {
      if ((dX > 0) ^ ((s[i] == 'R') ^ (s[i] == 'L'))) {
        if (dx[s[i] - 'L'] * dX > 0 && dy[s[i] - 'L'] * dY > 0) {
          ll timex = dX / dx[s[i] - 'L'], timey = dY / dy[s[i] - 'L'];
          if (timex == timey) {
            cout << timex;
            return 0;
          }
        }
      }
      if ((dX < 0) ^ ((s[i] == 'R') ^ (s[i] == 'L'))) {
        if (dx[s[i] - 'L'] * dX < 0 && dy[s[i] - 'L'] * dY > 0) {
          ll timex = dX / dx[s[i] - 'L'], timey = dY / dy[s[i] - 'L'];
          if (timex == timey) {
            cout << timex;
            return 0;
          }
        }
      }
      if ((dY > 0) ^ ((s[i] == 'U') ^ (s[i] == 'D'))) {
        if (dx[s[i] - 'L'] * dX > 0 && dy[s[i] - 'L'] * dY > 0) {
          ll timex = dX / dx[s[i] - 'L'], timey = dY / dy[s[i] - 'L'];
          if (timex == timey) {
            cout << timex;
            return 0;
          }
        }
      }
      if ((dY < 0) ^ ((s[i] == 'U') ^ (s[i] == 'D'))) {
        if (dx[s[i] - 'L'] * dX > 0 && dy[s[i] - 'L'] * dY < 0) {
          ll timex = dX / dx[s[i] - 'L'], timey = dY / dy[s[i] - 'L'];
          if (timex == timey) {
            cout << timex;
            return 0;
          }
        }
      }
    }
    x1 += dx[s[i] - 'L'];
    y1 += dy[s[i] - 'L'];
  }
  cout << -1;
  return 0;
}
