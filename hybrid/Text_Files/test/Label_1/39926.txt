def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        if sum(a) & 1:
            print("NO")
        else:
            print("YES")
            occur = []
            for i in range(n):
                if a[i] == 1:
                    occur += [i]
            ans = []
            for i in range(0, len(occur) - 1, 3):
                ans += [occur[i + 1] - 1, occur[i + 2] - occur[i + 1]]
            print(len(ans))
            print(*ans)

solve()
