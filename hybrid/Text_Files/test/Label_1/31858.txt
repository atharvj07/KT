import java.util.*;
class Main {
    static int longestSegment(int n, int k, int[] a) {
        int[] cnt = new int[k + 1];
        int ans = 0, cur = 0;
        for (int i = 0, j = 0; i < n; i++) {
            while (j < n && cur < k) {
                if (cnt[a[j]]++ == 0) cur++;
                j++;
            }
            ans = Math.max(ans, j - i);
            if (--cnt[a[i]] == 0) cur--;
        }
        return ans;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        System.out.println(longestSegment(n, k, a));
    }
}
