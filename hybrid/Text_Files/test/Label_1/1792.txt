#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    bool odd = 0, even = 0;
    while (n--) {
      int a;
      cin >> a;
      odd ^= (a & 1);
      even ^= (a % 4 == 2);
    }
    (x & 1 ? odd : even && x > 1) ? cout << "Yes\n" : cout << "No\n";
  }
  return 0;
}
