#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    for (int i = 0; i < k; i++)
        for (int j = i % 2; j < n; j += 2) a[j] ^= x;
    cout << *max_element(a.begin(), a.end()) << ' ' << *min_element(a.begin(), a.end()) << endl;
    return 0;
}
