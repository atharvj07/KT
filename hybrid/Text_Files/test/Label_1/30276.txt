import java.util.Scanner;

public class PhoenixAndBerries {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int totalRed = 0;
        int totalBlue = 0;
        int maxBaskets = 0;

        for (int i = 0; i < n; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();

            totalRed += a;
            totalBlue += b;

            maxBaskets = Math.max(maxBaskets, Math.min(a, k));
            maxBaskets = Math.max(maxBaskets, Math.min(b, k));
        }

        maxBaskets = Math.max(maxBaskets, totalRed / k);
        maxBaskets = Math.max(maxBaskets, totalBlue / k);

        System.out.println(maxBaskets);
    }
}
