MOD = 10**9 + 7

def sum_of_max_distances(n, coordinates):
    coordinates.sort()
    total_sum = 0
    for i in range(n):
        total_sum += (coordinates[i] * (2**i - 2**(n-i-1))) % MOD
        total_sum %= MOD
    return total_sum

n = int(input().strip())
coordinates = list(map(int, input().strip().split()))

print(sum_of_max_distances(n, coordinates))
