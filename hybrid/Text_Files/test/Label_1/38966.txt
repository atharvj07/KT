n = int(input())
a = input()
b = input()

def min_operations(a, b, n):
    diff = [int(b[i]) - int(a[i]) for i in range(n)]
    
    moves = []
    for i in range(n - 1):
        if diff[i] != 0:
            moves.append((i + 1, diff[i]))
            if i + 1 < n:
                diff[i + 1] -= diff[i]
            diff[i] = 0
    
    if diff[-1] != 0:
        print(-1)
    else:
        print(len(moves))
        for move in moves[:min(100000, len(moves))]:
            print(move[0], move[1] // abs(move[1]))

min_operations(a, b, n)
