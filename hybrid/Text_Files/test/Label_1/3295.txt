#include <bits/stdc++.h>
#define ll long long
#define pii pair<ll, ll>
#define vi vector<ll>
#define pb push_back
using namespace std;
const int MX = 1e6 + 1;
const ll INF = 1e18;
int main() {
	ios_base::sync_with_stdio(false); cin.tie(nullptr);
	ll t; cin >> t;
	while (t--) {
		ll x, y, c[6];
		cin >> x >> y;
		for (ll i = 0; i < 6; i++) cin >> c[i];
		ll fx = abs(x) + abs(y);
		ll fy = y - x;
		ll ans = INF;
		for (ll k = 0; k <= fx; k++) {
			ll p1 = (fx - k) / 2, p2 = (fy - k) / 2;
			if (p1 < 0 || p2 < 0 || (fx - k) % 2 || (fy - k) % 2) continue;
			ans = min(ans, p1 * c[2] + p2 * c[4] + k * c[3]);
			if (x <= 0) ans = min(ans, p1 * c[1] + p2 * c[5] + k * c[3]);
			if (y <= 0) ans = min(ans, p1 * c[1] + p2 * c[4] + k * c[3]);
			if (x <= 0 && y <= 0) ans = min(ans, p1 * c[1] + p2 * c[5] + k * c[3]);
		}
		cout << ans << "\n";
	}
}
