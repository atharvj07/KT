#include <cstdio>
#include <algorithm>

int n, a[300010], b[300010], f[300010];

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d", a + i);
    for (int i = 1; i <= n; i++) scanf("%d", b + i);
    for (int i = 1; i <= n; i++) f[i] = std::max(f[i - 1] + a[i], f[i - 2] + b[i]);
    printf("%d", f[n]);
}
