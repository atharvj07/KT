def min_program(n, lines):
    var_map = {}
    for line in lines:
        lvalue, rvalue = line.split('=')
        if len(rvalue.split()) > 1:
            arg1, op, arg2 = rvalue.split()
            rvalue = var_map.get(arg1, arg1) + op + var_map.get(arg2, arg2)
        var_map[lvalue] = rvalue

    res_line = None
    for line in lines[::-1]:
        if line.startswith('res='):
            res_line = line
            break

    if res_line:
        lvalue, rvalue = res_line.split('=')
        if len(rvalue.split()) > 1:
            arg1, op, arg2 = rvalue.split()
            new_rvalue = var_map.get(arg1, arg1) + op + var_map.get(arg2, arg2)
            return ["2", "aaaa=" + var_map[arg1], "res=aaaa^aaaa" if op == '^' else "res=aaaa" + op + "aaaa"]
        else:
            return ["1", "res=" + var_map.get(rvalue, rvalue)]
    else:
        return ["0"]

n = int(input().strip())
lines = [input().strip() for _ in range(n)]
result = min_program(n, lines)
print('\n'.join(result))
