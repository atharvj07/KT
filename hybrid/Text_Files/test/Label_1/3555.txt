#include <bits/stdc++.h>
using namespace std;
bool check(int n, vector<int> arr, int k)
{
    int t = arr.size() - k;
    vector<int> arr2(arr.end() - t, arr.end());
    for (int j = 0; j < n; j++)
        if (arr[j] > arr2[k - 1])
            return false;
    return true;
}
int main()
{
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    string ans1 = "", ans2 = "";
    int ans = 0;
    for (int i = 0; i <= n / 2; i++)
    {
        if (check(n, a, i) && check(n, b, i))
            ans = i;
    }
    for (int i = 0; i < n; i++)
    {
        ans1 += (i < ans) ? '1' : '0';
        ans2 += (i < ans) ? '1' : '0';
    }
    cout << ans1 << endl;
    cout << ans2 << endl;
}
