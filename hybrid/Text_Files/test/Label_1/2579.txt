#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, x, y;
    cin >> n >> x >> y;
    string s;
    cin >> s;
    vector<int> dp(n + 1, 1e9);
    dp[n] = 0;
    for (int i = n - 1; i >= 0; i--) {
        dp[i] = dp[i + 1];
        int mask = 0;
        for (int j = i; j < i + x; j++) {
            mask = mask * 2 + (s[j] - '0');
        }
        int cur = mask % (1 << x - y);
        int need = (1 << x - y) - 1 - cur;
        if ((s[i] - '0') != (need & (1 << (x - y - 1)))) {
            dp[i] = min(dp[i], dp[i + 1] + 1);
        }
    }
    cout << dp[0] << endl;
    return 0;
}
