import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
        }
        Arrays.sort(m);
        int ans = 0;
        while (m[n - 1] - m[0] > 1) {
            m[n - 1]--;
            m[0]++;
            ans++;
            Arrays.sort(m);
        }
        System.out.println(ans);
    }
}
