n, k = map(int, input().split())
s = list(input())
r = [i for i in range(n) if s[i] == "1"]

cost = 0
pos = 0
while pos < n:
    rpos = pos
    while rpos < len(r) and r[rpos] <= pos + k:
        rpos += 1
    if rpos == len(r):
        cost += n - pos
        break
    else:
        cost += r[rpos] - pos
        pos = r[rpos] + k + 1

print(cost)
