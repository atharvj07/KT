import java.io.*;
class Main {
    static int[][] poly = new int[3000][3000];
    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int N = Integer.parseInt(in.readLine());
        String[] s = in.readLine().split(" ");
        int[] a = new int[N], p = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = Integer.parseInt(s[i]);
        }
        for (int i = 0; i < N; i++) {
            p[i] = i;
        }
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                for (int k = 0; k < N; k++) {
                    p[k] = p[k] * (p[j] - p[i]) % N;
                }
            }
        }
        for (int i = 0; i < N; i++) {
            p[i] = fastpow(p[i], -1, N);
        }
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                poly[i][j] = fastpow(i, j, N) * p[i] % N;
            }
        }
        for (int j = 0; j < N; j++) {
            for (int i = 0; i < N; i++) {
                for (int k = 0; k < N; k++) {
                    poly[j][i] = (poly[j][i] + poly[j][k] * poly[k][i] % N) % N;
                }
            }
        }
        for (int i = 0; i < N; i++) {
            int x = a[i] * p[i] % N;
            for (int j = 0; j < N; j++) {
                x = (x - poly[i][j] * p[i] % N + N) % N;
            }
            System.out.print(x + " ");
        }
    }
    static int fastpow(int a, long pow, int m) {
        int p = a;
        long n = pow;
        int ret = 1;
        while (n > 0) {
            if ((n & 1) == 1) ret = ret * p % m;
            p = p * p % m;
            n >>>= 1;
        }
        return ret;
    }
}
