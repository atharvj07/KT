t = int(input())
for _ in range(t):
    n = int(input())
    a = list(input())
    pref = [0] * n
    pref[0] = int(a[0])
    for i in range(1, n):
        pref[i] = pref[i - 1] + int(a[i])
    ans = 0
    for i in range(n):
        for j in range(i + 1):
            if i == 0:
                s = pref[i]
            else:
                s = pref[i] - pref[j - 1]
            if s == i - j + 1:
                ans += 1
    print(ans)
