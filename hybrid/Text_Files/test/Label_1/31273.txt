import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur < a[i]) {
                ans++;
                cur = i;
            }
        }
        System.out.println(ans);
    }
}
