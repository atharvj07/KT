n, k = map(int, input().split())
a = list(map(int, input().split()))
t = list(map(int, input().split()))
ans = sum([i * j for i, j in zip(a, t)])
for i in range(n - k + 1):
    ans = max(ans, sum([j for j in a[i:i + k]]) + sum([i * j for i, j in zip(a[:i], t[:i])]) + sum([i * j for i, j in zip(a[i + k:], t[i + k:])]))
print(ans)
