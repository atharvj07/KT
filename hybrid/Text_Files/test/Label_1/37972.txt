def max_score_after_move(holes):
    max_score = 0
    for i in range(14):
        if holes[i] > 0:
            holes_copy = holes[:]
            stones = holes_copy[i]
            holes_copy[i] = 0
            distribute, extra = divmod(stones, 14)
            for j in range(14):
                holes_copy[j] += distribute
            for j in range(1, extra + 1):
                holes_copy[(i + j) % 14] += 1
            score = sum(stone for stone in holes_copy if stone % 2 == 0)
            max_score = max(max_score, score)
    return max_score

holes = list(map(int, input().split()))
print(max_score_after_move(holes))
