import java.util.*;
import java.math.*;
import java.io.*;
class Main{
    static int mod = 1000000007;

    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        char[] ar = in.next().toCharArray();
        int n = ar.length;
        long[][] dp = new long[n][3];
        if(ar[0] == '?' || ar[0] == '0')dp[0][0] = 1;
        for(int i = 1; i < n; i++){
            if(ar[i] == '*'){
                dp[i][0] = dp[i - 1][0];
                dp[i][1] = dp[i-1][1];
                dp[i][2] = dp[i-1][2];
            }else if(ar[i] == '0'){
                dp[i][0] = (dp[i - 1][0] + dp[i - 1][2]) % mod;
                dp[i][1] = (dp[i - 1][0]) % mod;
                dp[i][2] = 0;
            }else if(ar[i] == '1'){
                dp[i][0] = (dp[i - 1][0]) % mod;
                dp[i][1] = (dp[i - 1][0] + dp[i - 1][1] + dp[i - 1][2]) % mod;
                dp[i][2] = 0;
            }else{
                dp[i][0] = (dp[i - 1][0]) % mod;
                dp[i][1] = (dp[i - 1][0] + dp[i - 1][1]) % mod;
                dp[i][2] = (dp[i - 1][1] + dp[i - 1][2]) % mod;
            }
        }
        System.out.println(dp[n - 1][0]);
    }
}
