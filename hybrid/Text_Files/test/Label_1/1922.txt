#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  int cnt1 = 0;
  string ans = "";
  for (int i = 0; i < n; i++) {
    ans += '0';
  }
  for (int i = 0; i < m; i++) {
    if (ans.size() < 2 || ans[ans.size() - 2] != '1') {
      ans += '1';
      cnt1++;
    } else {
      if (cnt1 >= 2) {
        ans += '01';
        cnt1 = 1;
      } else {
        cout << "-1" << endl;
        return 0;
      }
    }
  }
  cout << ans << endl;
  return 0;
}
