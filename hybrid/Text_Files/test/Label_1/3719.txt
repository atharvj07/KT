#include <bits/stdc++.h>
using namespace std;
const int M = 3e5 + 5;

int n, k, d, p[M], par[M];
vector<pair<int, int>> adj[M];

int find(int x) { return par[x] == x ? x : par[x] = find(par[x]); }

int add(int u, int v) {
  if (find(u) == find(v)) return 0;
  par[find(u)] = find(v);
  return 1;
}

bool bfs(int x, int dist) {
  queue<pair<int, int>> q;
  q.push({x, 0});
  while (q.size()) {
    auto [u, dis] = q.front();
    q.pop();
    if (dis > dist) break;
    if (find(u) != find(x)) return 0;
    for (auto [v, w] : adj[u]) {
      if (dis + w <= dist) q.push({v, dis + w});
    }
  }
  return 1;
}

int main() {
  cin >> n >> k >> d;
  for (int i = 1; i <= k; i++) { cin >> p[i]; }

  for (int i = 1; i <= n; i++) par[i] = i;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back({v, 1});
    adj[v].push_back({u, 1});
    add(u, v);
  }

  vector<int> ans;
  for (int i = 1; i <= n; i++) {
    for (auto [v, w] : adj[i]) {
      if (find(i) == find(v) && !bfs(v, d)) ans.push_back(i);
    }
  }

  cout << ans.size() << "\n";
  for (auto it : ans) cout << it << " ";
}
