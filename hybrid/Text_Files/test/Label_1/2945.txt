#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
int main()
{
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    vector<pair<int, int>> ans;
    for (int i = 0; i < n; i++)
    {
        if (a[i] == b[i])
            continue;
        for (int j = i + 1; j < n; j++)
        {
            if (a[j] == b[i])
            {
                ans.push_back({j, i});
                a[j] = a[i];
                a[i] = b[i];
                break;
            }
        }
    }
    cout << ans.size() << endl;
    for (auto x : ans)
        cout << x.first + 1 << " " << x.second + 1 << endl;
    return 0;
}
