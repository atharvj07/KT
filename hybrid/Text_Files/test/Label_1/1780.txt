#include <bits/stdc++.h>

using namespace std;

int n, m;
vector<set<int>> adj;

int main() {
  cin >> n >> m;
  adj.resize(n + 1);
  for (int i = 0; i < m; i++) {
    int x;
    cin >> x;
    for (int j = 0; j < x; j++) {
      int y;
      cin >> y;
      adj[abs(y)].insert(y > 0 ? y : -y);
    }
  }
  bool flag = 0;
  for (int i = 1; i <= n; i++) {
    if (adj[i].size() == adj[-i].size() && adj[i].size() == 1) {
      flag = 1;
    }
  }
  cout << (flag ? "YES" : "NO") << endl;
}
