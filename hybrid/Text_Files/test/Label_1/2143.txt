#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, s;
    cin >> n >> m >> s;
    vector<vector<int>> adj(n + 1);
    while (m--) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
    }
    vector<bool> vis(n + 1, false);
    queue<int> q;
    q.push(s);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto v : adj[u]) {
            q.push(v);
        }
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) {
            ans++;
            if (ans > 1) break;
        }
    }
    cout << ans;
    return 0;
}
