#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MAXN = 3e5 + 5;

int a[MAXN], n, k;

int main() {
    cin >> n >> k;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a + k, a + n);
    sort(a, a + k);
    ll ans = 0;
    for (int i = 0; i < n - k; i++) ans += abs(a[i] - a[i + k]);
    cout << ans << endl;
    return 0;
}
