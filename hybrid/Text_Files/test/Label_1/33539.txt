import java.util.*;
import java.io.*;
import java.nio.charset.StandardCharsets;

public class Main {
    static int t, n;
    static int[] a, b;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in, StandardCharsets.UTF_8);
        t = sc.nextInt();
        while (t-- > 0) {
            n = sc.nextInt();
            a = new int[n];
            b = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
            }

            boolean yes = true;
            for (int i = 0; i < n; i++) {
                if (b[i] % 3 != a[i]) {
                    yes = false;
                    break;
                }
            }
            System.out.println(yes ? "YES" : "NO");
        }
    }
}
