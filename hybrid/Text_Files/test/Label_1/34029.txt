import java.util.Scanner;

public class Main {
    private static int[] a;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int d = in.nextInt();
        int[] c = new int[m];
        a = new int[n + 2];
        for (int i = 0; i < m; i++) {
            c[i] = in.nextInt();
        }
        int[] p = new int[m];
        int[] s = new int[m];
        int len = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < c[i]; j++) {
                if (a[len + j] == 0) s[i]++;
            }
            len += c[i];
            p[i] = len;
        }
        int cur = 0;
        for (int i = 0; i < m; i++) {
            if (s[i] == 0) {
                cur += c[i];
                continue;
            }
            int l = p[i];
            int r = p[i] + c[i] - 1;
            if (cur + d < l) {
                System.out.println("NO");
                return;
            }
            while (cur <= r) {
                a[cur] = i + 1;
                cur++;
            }
        }
        System.out.println("YES");
        for (int i = 1; i < n + 1; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
