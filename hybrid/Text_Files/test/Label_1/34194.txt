import java.util.*;

public class Main {
    private static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }

        public double dist(Point other) {
            return Math.sqrt((x - other.x) * (x - other.x) + (y - other.y) * (y - other.y));
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        List<Point> points = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            points.add(new Point(sc.nextInt(), sc.nextInt()));
        }
        double[][] dists = new double[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                dists[i][j] = dists[j][i] = points.get(i).dist(points.get(j));
            }
        }
        int[] ans = new int[k];
        boolean[] deleted = new boolean[n];
        for (int i = 0; i < k; i++) {
            int best = -1;
            double maxDist = -1;
            for (int j = 0; j < n; j++) {
                if (!deleted[j]) {
                    double curMaxDist = -1;
                    for (int l = 0; l < n; l++) {
                        if (l != j && !deleted[l]) {
                            curMaxDist = Math.max(curMaxDist, dists[j][l]);
                        }
                    }
                    if (curMaxDist > maxDist) {
                        maxDist = curMaxDist;
                        best = j;
                    }
                }
            }
            ans[i] = best + 1;
            deleted[best] = true;
        }
        for (int i = 0; i < k; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
