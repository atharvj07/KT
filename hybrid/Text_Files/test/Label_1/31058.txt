import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        Arrays.sort(a);
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] <= 5) {
                count++;
            }
        }
        System.out.println(count);
    }
}
