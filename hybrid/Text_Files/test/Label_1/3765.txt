#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, p;
    cin >> n >> p;
    vector<int> a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
    }
    vector<int> c(n + 1), d(n + 1);
    for (int i = 1; i <= n; i++) {
        c[a[i]]++;
        c[b[i]]++;
        d[i]++;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (c[i] + c[j] - d[i] - d[j] >= p) {
                ans++;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
