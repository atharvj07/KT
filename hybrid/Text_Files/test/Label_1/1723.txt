#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

string s,tmp;
int n,l,d,c;
int main() {
	getline(cin,s);
	n=s.size(); l=s[0]-48;
	for (int i=0; i<n; i++){
		if (s[i]==','){
			c++;
			if (c>d) d=c;
		}
	}
	cout<<d<<endl;
	for (int i=1; i<=d; i++){
		tmp=""; c=0;
		for (int j=0; j<n; j++){
			if (s[j]==','){
				if (c==i-1){
					if (tmp.size()>0){
						if (c>0) cout<<" ";
						cout<<tmp;
					}
					tmp="";
				}
				c++;
			}
			else tmp=tmp+s[j];
		}
		cout<<endl;
	}
	return 0;
}
