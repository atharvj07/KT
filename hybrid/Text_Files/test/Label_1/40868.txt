for _ in range(int(input())):
  n, m = map(int, input().split())
  grid = []
  for _ in range(n):
    grid.append(input().strip())
  ans = 0
  for i in range(n):
    for j in range(m):
      if grid[i][j] == '.':
        continue
      h = 1
      while i + h - 1 < n and j - h + 1 >= 0 and j + h - 1 < m:
        cnt = True
        for x in range(h):
          if grid[i+x][j-h+1+x] != '*':
            cnt = False
            break
        if cnt:
          ans += 1
          h += 1
        else:
          break
  print(ans)
