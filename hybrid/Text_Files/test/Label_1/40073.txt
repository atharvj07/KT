from sys import stdin

n, k = map(int, stdin.readline().split())
xs = list(map(int, stdin.readline().split()))
L = [1] * n
R = [n] * n
for i in range(k):
    x = xs[i]
    L[x - 1] = max(L[x - 1], i + 1)
    R[x - 1] = min(R[x - 1], i + 1)
MOD = 10 ** 9 + 7
ans = 0
for i in range(n):
    if L[i] <= R[i]:
        ans += 2
    else:
        ans += 1
    ans %= MOD
print(ans)
