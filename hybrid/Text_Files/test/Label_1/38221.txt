import bisect
for _ in range(int(input())):
    n = int(input())
    a = []
    b = []
    for _ in range(n):
        l, r = map(int, input().split())
        a.extend([l, r])
        b.append([l, r])
    a.sort()
    ans = 0
    for i in range(n):
        ind = bisect.bisect_right(a, b[i][1])
        ans = max(ans, ind - i + n - ind - 1)
    print(ans)
