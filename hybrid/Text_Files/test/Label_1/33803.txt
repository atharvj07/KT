import java.util.Scanner;

public class Cinema {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Number of cars
        int k = scanner.nextInt(); // Number of gas stations
        int s = scanner.nextInt(); // Length of the road
        int t = scanner.nextInt(); // Time in which the film starts

        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        int minPrice = -1;
        for (Car car : cars) {
            if (canReachCinema(car, gasStations, s, t)) {
                minPrice = Math.min(minPrice, car.price);
            }
        }

        System.out.println(minPrice);
    }

    static boolean canReachCinema(Car car, int[] gasStations, int s, int t) {
        int time = 0;
        int fuel = car.tankCapacity;
        for (int i = 0; i < gasStations.length; i++) {
            int distance = gasStations[i];
            if (fuel < distance) {
                return false;
            }
            time += distance / 2;
            fuel -= distance;
            if (i < gasStations.length - 1) {
                fuel = car.tankCapacity;
            }
        }
        int remainingDistance = s - gasStations[gasStations.length - 1];
        if (fuel < remainingDistance) {
            return false;
        }
        time += remainingDistance / 2;
        return time <= t;
    }

    static class Car {
        int price;
        int tankCapacity;

        Car(int price, int tankCapacity) {
            this.price = price;
            this.tankCapacity = tankCapacity;
        }
    }
}
