import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) break;
            int[] a = new int[n], b = new int[n], w = new int[n];
            for (int i = 0; i < n; i++) {
                w[i] = sc.nextInt();
                a[i] = sc.nextInt();
                b[i] = sc.nextInt();
            }
            int[] c = new int[232];
            for (int i = 0; i < n; i++) for (int j = a[i]; j < b[i]; j++) c[j] += w[i];
            boolean ok = true;
            for (int i = 0; i < 232; i++) if (c[i] > 150) ok = false;
            System.out.println(ok ? "OK" : "NG");
        }
    }
}
