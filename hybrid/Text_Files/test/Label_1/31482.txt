import java.util.Scanner;

public class BerryBaskets {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[][] berries = new int[n][2];
        for (int i = 0; i < n; i++) {
            berries[i][0] = scanner.nextInt(); // Red berries
            berries[i][1] = scanner.nextInt(); // Blue berries
        }

        int maxBaskets = 0;
        for (int i = 0; i < n; i++) {
            maxBaskets += Math.min(berries[i][0], k) + Math.min(berries[i][1], k); // Add berries from each shrub
        }

        // Check for color-based baskets
        int redBerries = 0, blueBerries = 0;
        for (int i = 0; i < n; i++) {
            redBerries += berries[i][0];
            blueBerries += berries[i][1];
        }
        maxBaskets = Math.max(maxBaskets, redBerries / k + blueBerries / k); // Add color-based baskets

        System.out.println(maxBaskets);
    }
}
