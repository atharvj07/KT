import java.util.*;

public class Main {
    static int n, m, rowCnt, colCnt;
    static int[] row, col;
    static boolean[][] grid;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        row = new int[n + 1];
        col = new int[m + 1];
        grid = new boolean[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            char[] line = in.next().toCharArray();
            for (int j = 1; j <= m; j++) {
                grid[i][j] = line[j - 1] == '#';
                if (grid[i][j]) {
                    row[i]++;
                    col[j]++;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            if (row[i] == 0 || col[i] == 0) {
                System.out.println("No");
                return;
            }
        }
        int rowCnt = 0, colCnt = 0;
        for (int i = 1; i <= n; i++) {
            if (row[i] > 0) rowCnt++;
            if (col[i] > 0) colCnt++;
            if (row[i] + col[i] > m) {
                System.out.println("No");
                return;
            }
        }
        if (rowCnt == n && colCnt == m) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
