import sys
import heapq
input = sys.stdin.readline

for t in range(int(input())):
    k = int(input())
    g = [[] for _ in range(k*2+1)]
    for x in range(2*k-1):
        a, b, c = map(int, input().split())
        g[a].append((c,b))
        g[b].append((c,a))
    vis, ans = [False]*(2*k+1), []
    def dfs(p, t):
        vis[p] = True
        ans.append(t)
        for x, y in g[p]:
            if not vis[y]:
                dfs(y, t+x)
    for i in range(1, 2*k+1):
        if not vis[i]:
            dfs(i, 0)
            break
    for i in range(1, 2*k+1):
        if not vis[i]:
            dfs(i, 0)
            break
    ans.sort()
    print(ans[k-1], ans[2*k-k-1])
