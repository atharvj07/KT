#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> p(n);
  for (int i = 0; i < n; i++) { cin >> p[i].first >> p[i].second; }
  sort(p.begin(), p.end());
  int ans = 1;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < n - 1 && (long long)(p[j + 1].second - p[i].second) * (p[j + 1].first - p[i].first) >
                           (long long)(p[j].second - p[i].second) * (p[j + 1].first - p[i].first)) {
      j++;
    }
    if (j < n - 1 && (long long)(p[j + 1].second - p[i].second) * (p[j + 1].first - p[i].first) ==
                       (long long)(p[j].second - p[i].second) * (p[j + 1].first - p[i].first)) {
      ans = 0;
      break;
    }
  }
  for (int i = 0, j = 0; i < n; i++) {
    while (j < n - 1 && (long long)(p[j + 1].second - p[i].second) * (p[j].first - p[i].first) <
                           (long long)(p[j].second - p[i].second) * (p[j + 1].first - p[i].first)) {
      j++;
    }
    if (j < n - 1 && (long long)(p[j + 1].second - p[i].second) * (p[j].first - p[i].first) ==
                       (long long)(p[j].second - p[i].second) * (p[j + 1].first - p[i].first)) {
      ans = 0;
      break;
    }
  }
  cout << (ans ? "YES" : "NO") << endl;
}
