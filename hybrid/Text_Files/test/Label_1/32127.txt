import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);

        int N = Integer.parseInt(in.readLine());
        int[] P = new int[N * N + 1];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 1; i <= N * N; i++) {
            P[i] = Integer.parseInt(st.nextToken());
        }

        int[][] row = new int[N][N + 1];
        int[][] col = new int[N][N + 1];

        int[] rPos = new int[N * N + 1];
        int[] cPos = new int[N * N + 1];

        for (int i = 1; i <= N * N; i++) {
            int r = (P[i] - 1) / N;
            int c = (P[i] - 1) % N;

            row[r][c] = i;
            col[c][r] = i;

            rPos[i] = r;
            cPos[i] = c;
        }

        int[][][] dp = new int[N][N][5]; // 0: up, 1: down, 2: left, 3: right, 4: none

        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                dp[i][j][0] = Integer.MAX_VALUE;
                dp[i][j][1] = Integer.MAX_VALUE;
                dp[i][j][2] = Integer.MAX_VALUE;
                dp[i][j][3] = Integer.MAX_VALUE;
                dp[i][j][4] = 0;
            }
        }

        dp[0][0][4] = 0;

        for (int i = 0; i < N; i++) {
            for (int j = 0; j < N; j++) {
                for (int d = 0; d < 4; d++) {
                    int ni = i + (d == 0 ? -1 : (d == 1 ? 1 : 0));
                    int nj = j + (d == 2 ? -1 : (d == 3 ? 1 : 0));

                    if (ni >= 0 && ni < N && nj >= 0 && nj < N && row[ni][nj] < row[i][j]) {
                        dp[i][j][d] = Math.min(dp[i][j][d], dp[ni][nj][4]);
                    }
                }

                dp[i][j][4] += col[j][i] < row[i][j] ? 1 : 0;
            }
        }

        int ans = dp[N - 1][N - 1][4];

        out.println(ans);
        out.flush();
    }
}
