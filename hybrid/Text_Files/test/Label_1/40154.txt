import sys

n = int(input())
gcd = list(map(int, input().split()))
g = [[0]*n for _ in range(n)]
idx = 0

for i in range(n):
    for j in range(n):
        g[i][j] = gcd[idx]
        idx += 1

a = [0]*n

for i in range(n):
    for j in range(n):
        if i == j:
            a[i] = g[i][j]
            continue
        if a[i] == 0:
            a[i] = g[i][j] // g[j][j]
        if a[j] == 0:
            a[j] = g[i][j] // g[i][i]

print(*a)
