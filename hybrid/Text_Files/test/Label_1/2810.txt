#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, w;
  cin >> n >> w;
  vector<vector<int>> a(n);
  for (int i = 0; i < n; i++) {
    int l;
    cin >> l;
    a[i].resize(l);
    for (int j = 0; j < l; j++) { cin >> a[i][j]; }
  }
  vector<int> b(w);
  for (int j = 0; j < w; j++) {
    for (int i = 0; i < n; i++) {
      int l = a[i].size();
      if (j < l) b[j] = max(b[j], a[i][j]);
      else if (j - l + 1 >= 0) b[j] = max(b[j], 0);
    }
  }
  for (int j = 1; j < w; j++) {
    b[j] = max(b[j - 1], b[j]);
  }
  for (int j = 0; j < w; j++) { cout << b[j] << " "; }
  cout << endl;
  return 0;
}
