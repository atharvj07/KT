#include<bits/stdc++.h>
using namespace std;
int n,a[105];
bool f[110][110][4];
char s[110];
int main()
{
	cin>>n;
	cin>>s+1;
	f[0][0][0]=f[0][0][1]=f[0][0][2]=f[0][0][3]=1;
	for(int i=1;i<=n;i++)
		for(int j=0;j<i;j++)
			for(int k=0;k<=3;k++)
				if(f[i-1][j][k])
				{
					if(s[i]!='?'&&k==3-s[i]-'C') continue;
					f[i][j][k]=f[i-1][j][k];
					if(k!=3-s[i]-'C') f[i][j+1][3-s[i]-'C']=1;
					if(j) f[i][j-1][k]=1;
				}
	if(f[n][0][0]||f[n][0][1]||f[n][0][2]) puts("Yes");
	else puts("No");
	return 0;
}
