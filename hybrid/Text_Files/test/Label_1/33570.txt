import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class JohnnyBlogs {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int m = Integer.parseInt(tokenizer.nextToken());

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int a = Integer.parseInt(tokenizer.nextToken());
            int b = Integer.parseInt(tokenizer.nextToken());
            graph.get(a).add(b);
            graph.get(b).add(a);
        }

        tokenizer = new StringTokenizer(reader.readLine());
        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = Integer.parseInt(tokenizer.nextToken());
        }

        boolean[] visited = new boolean[n + 1];
        int[] order = new int[n];
        int current = 1;
        int index = 0;

        while (index < n) {
            if (!visited[current]) {
                visited[current] = true;
                Set<Integer> availableTopics = new HashSet<>();
                for (int neighbor : graph.get(current)) {
                    if (visited[neighbor]) {
                        availableTopics.add(desiredTopics[neighbor]);
                    }
                }
                for (int i = 1; i <= n; i++) {
                    if (!availableTopics.contains(i)) {
                        order[index++] = current;
                        desiredTopics[current] = i;
                        break;
                    }
                }
            }
            current = (current % n) + 1;
        }

        if (index == n) {
            for (int i = 0; i < n; i++) {
                System.out.print(order[i] + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
