import java.util.Arrays;
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Main {

    private static Set<String> FORBIDDEN_PAIRS = new HashSet<>();

    public static void main(String[] args) {
        StringBuilder str = new StringBuilder();
        Scanner scanner = new Scanner(System.in);
        str.append(scanner.next());
        int k = scanner.nextInt();
        for (int i = 0; i < k; i++) {
            FORBIDDEN_PAIRS.add(scanner.next());
        }
        System.out.println(removeLetters(str));
    }

    private static int removeLetters(StringBuilder str) {
        int n = str.length();
        int[] dp = new int[n + 1];
        int[] lastSeen = new int[26];
        Arrays.fill(lastSeen, -1);
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            char c = str.charAt(i - 1);
            dp[i] = dp[i - 1] + 1;
            if (lastSeen[c - 'a'] != -1) {
                dp[i] = Math.min(dp[i], dp[lastSeen[c - 'a']] + 1);
            }
            lastSeen[c - 'a'] = i - 1;
            int j = i - 1;
            while (j >= 0 && FORBIDDEN_PAIRS.contains(str.substring(j, i))) {
                dp[i] = Math.min(dp[i], dp[j] + i - j - 1);
                j--;
            }
        }
        return dp[n];
    }
}
