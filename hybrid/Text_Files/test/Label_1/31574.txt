import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] hands = new int[n];
        int[] frustrations = new int[n];
        for (int i = 0; i < n; i++) {
            hands[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            frustrations[i] = sc.nextInt();
        }
        int[][] dp = new int[n][2];
        dp[0][0] = 0;
        dp[0][1] = frustrations[0];
        for (int i = 1; i < n; i++) {
            dp[i][0] = Math.min(dp[i - 1][0], dp[i - 1][1]);
            dp[i][1] = Math.min(dp[i - 1][0] + frustrations[i], dp[i - 1][1] + frustrations[(i - 1 + n) % n]);
        }
        System.out.println(Math.min(dp[n - 1][0], dp[n - 1][1]));
    }
}
