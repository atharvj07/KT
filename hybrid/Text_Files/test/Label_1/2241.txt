#include <bits/stdc++.h>
#define ll long long
using namespace std;
unordered_map<ll, vector<ll>> g;
ll vis[5010];
void dfs(ll v) {
  if (vis[v] == 1) return;
  vis[v] = 1;
  for (ll x : g[v]) {
    if (vis[x] == 0) dfs(x);
  }
}
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
  ll t, n, x, ans = 0;
  cin >> t;
  while (t--) {
    ans = 0;
    memset(vis, 0, sizeof vis);
    g.clear();
    cin >> n;
    ll arr[n];
    for (ll i = 0; i < n; i++) cin >> arr[i];
    bool flag = 0;
    for (ll i = 0; i < n; i++) {
      for (ll j = i + 1; j < n; j++) {
        if (arr[j] == arr[i]) {
          g[i + 1].push_back(j + 1);
          g[j + 1].push_back(i + 1);
          ans++;
        }
      }
    }
    for (ll i = 0; i < n; i++) {
      if (vis[i + 1] == 0) {
        dfs(i + 1);
        ans++;
      }
    }
    if (ans == n - 1) {
      cout << "YES" << endl;
      for (ll i = 0; i < n; i++) {
        for (ll j = i + 1; j < n; j++) {
          if (arr[j] != arr[i] && g[i + 1].size()) {
            cout << i + 1 << " " << j + 1 << endl;
            g[i + 1].pop_back();
            break;
          }
        }
      }
    } else
      cout << "NO" << endl;
  }
  return 0;
}
