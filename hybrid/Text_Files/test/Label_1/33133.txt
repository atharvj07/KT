import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int l = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }

        Arrays.sort(a);
        int maxArea = 0;
        int currArea = 0;
        int cnt = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] >= l) {
                cnt++;
                currArea += a[i];
            } else {
                if (cnt > 0) {
                    maxArea = Math.max(maxArea, currArea);
                    currArea = 0;
                    cnt = 0;
                }
            }
        }
        if (cnt > 0) {
            maxArea = Math.max(maxArea, currArea);
        }
        System.out.println(maxArea);
    }
}
