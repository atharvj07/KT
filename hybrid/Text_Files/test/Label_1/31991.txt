import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int q = sc.nextInt();

        int[] days = new int[n + 1];
        for (int i = 0; i < q; i++) {
            if (sc.nextInt() == 1) {
                days[sc.nextInt()] += sc.nextInt();
            } else {
                int p = sc.nextInt();
                long sum = 0;
                long repairs = 0;
                for (int j = p; j <= p + k - 1; j++) {
                    repairs += b;
                }
                for (int j = 1; j <= n; j++) {
                    if (j >= p && j <= p + k - 1) {
                        sum += Math.min(repairs, days[j]);
                    } else {
                        sum += Math.min(a, days[j]);
                    }
                }
                System.out.println(sum);
            }
        }
    }
}
