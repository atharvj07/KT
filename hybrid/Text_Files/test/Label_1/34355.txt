import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        System.out.println(n / 2);
        for (int i = 2; i <= n / 2; i++) {
            if (n % i == 0) {
                System.out.println(i + " " + n);
            }
        }
    }
}
