#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }

  std::vector<int> b(n);
  b[0] = 0;
  for (int i = 1; i < n; i++) {
    if (a[i] == 0) {
      std::cout << -1 << std::endl;
      return 0;
    }
    b[i] = *std::max_element(a.begin(), a.begin() + a[i]);
    a[b[i]]++;
  }
  for (int i = 0; i < n; i++) {
    std::cout << b[i] << " ";
  }
  std::cout << std::endl;
  return 0;
}
