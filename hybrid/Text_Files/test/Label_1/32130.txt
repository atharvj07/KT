import java.util.Scanner;
import java.text.DecimalFormat;

class Main {

    private static final double EPS = 1e-9;

    private static class Glass {
        double capacity;
        double water;

        public Glass(double capacity, double water) {
            this.capacity = capacity;
            this.water = water;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Glass[] glasses = new Glass[n];
        for (int i = 0; i < n; i++) {
            double capacity = scanner.nextDouble();
            double water = scanner.nextDouble();
            glasses[i] = new Glass(capacity, water);
        }

        double[] maxWater = new double[n + 1];
        for (int k = 1; k <= n; k++) {
            double[][] dp = new double[n][k + 1];
            for (int i = 0; i < n; i++) {
                dp[i][1] = glasses[i].water;
            }
            for (int j = 2; j <= k; j++) {
                for (int i = 0; i < n; i++) {
                    for (int l = 0; l < n; l++) {
                        if (i == l) {
                            continue;
                        }
                        double water1 = dp[i][j - 1] - Math.min(glasses[i].water, glasses[l].capacity - glasses[l].water) / 2;
                        double water2 = dp[l][j - 1] + Math.min(glasses[i].water, glasses[l].capacity - glasses[l].water) / 2;
                        if (water1 > 0) {
                            dp[i][j] = Math.max(dp[i][j], water1);
                        }
                        if (water2 > 0) {
                            dp[l][j] = Math.max(dp[l][j], water2);
                        }
                    }
                }
            }
            double max = 0;
            for (int i = 0; i < n; i++) {
                max = Math.max(max, dp[i][k]);
            }
            maxWater[k] = max;
        }

        DecimalFormat df = new DecimalFormat("0.0000000000");
        for (int i = 1; i <= n; i++) {
            System.out.println(df.format(maxWater[i]));
        }
    }
}
