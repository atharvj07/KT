import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] bulbs = new int[n];
        for (int i = 0; i < n; i++) {
            bulbs[i] = scanner.nextInt();
        }
        Arrays.sort(bulbs);
        int complexity = 0;
        for (int i = 0; i < n - 1; i++) {
            if (bulbs[i] % 2 != bulbs[i + 1] % 2) {
                complexity++;
            }
        }
        System.out.println(complexity);
    }
}
