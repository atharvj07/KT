#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> v(n);
  for (int& x : v) cin >> x;
  map<int, int> cnt;
  for (int x : v) cnt[x]++;
  int ans = min(cnt[1], min(cnt[2], cnt[3]));
  if (ans == 0)
    cout << 0 << endl;
  else {
    cout << ans << endl;
    vector<array<int, 3>> sol;
    for (int i = 0; i < ans; i++) {
      array<int, 3> cur;
      for (int j = 1; j <= 3; j++) {
        while (cnt[j] > 0) {
          auto it = find(v.begin(), v.end(), j);
          if (it != v.end()) {
            cur[j - 1] = it - v.begin() + 1;
            cnt[j]--;
            v.erase(it);
            break;
          }
        }
      }
      sol.push_back(cur);
    }
    for (auto& t : sol) {
      for (int x: t) cout << x << ' ';
      cout << endl;
    }
  }
}
