from collections import Counter
from math import sqrt

n=int(input())
a=list(map(int,input().split()))
b=[0]*n
for i in range(n):b[i]=int(sqrt(a[i]))==sqrt(a[i])
c=Counter(a)
for i in range(n):
    for j in range(i+1):
        if i==j:
            if b[i]:print(1,end=' ')
            else:print(0,end=' ')
        else:
            if b[i] and b[j] and c[a[i]]==1 and c[a[j]]==1:
                print(0,end=' ')
            else:print(1,end=' ')
