#include<bits/stdc++.h>
using namespace std;
int n, m, k;
const int maxn = 300005;
int d[maxn], p[maxn], rk[maxn];
typedef pair<int, int> ii;
vector<ii> edges;
vector<int> edg;
int get(int x)
{
    if (x == p[x])
        return x;
    return p[x] = get(p[x]);
}
void uni(int x, int y)
{
    x = get(x);
    y = get(y);
    if (rk[x] < rk[y])
        p[x] = y;
    else
    {
        p[y] = x;
        if (rk[x] == rk[y])
            rk[x]++;
    }
}
int main()
{
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++)
        p[i] = i, rk[i] = 0;
    for (int i = 0; i < m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        edges.push_back({w, i});
        edg.push_back(i);
    }
    sort(edges.begin(), edges.end());
    int cnt = k;
    for (auto x : edges)
    {
        int u, v, w;
        w = x.first;
        v = x.second;
        u = edg[v];
        u = get(u);
        v = get(v);
        if (u != v)
        {
            cnt--;
            uni(u, v);
            if (!cnt)
                break;
        }
    }
    int e = k - cnt;
    cout << e << endl;
    for (int i = 0; i < e; i++)
    {
        cout << edg[i] + 1 << " ";
    }
}
