import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long x = sc.nextLong();
        long b = 1;
        for (long i = 2; i < x; i++) {
            if (check(n, i)) {
                b = i;
            }
        }
        if (b == 1) {
            System.out.println(-1);
        } else {
            System.out.println(b);
        }
    }

    private static boolean check(long n, long b) {
        long num = 0;
        long tmp = n;
        while (tmp > 0) {
            num = num * b + tmp % b;
            tmp /= b;
        }
        for (long i = 2; i <= n; i++) {
            tmp = num;
            long t = 1;
            while (t < i) {
                t *= b;
            }
            while (tmp > 0) {
                long last = tmp % b;
                tmp /= b;
                num = num / b + last * t;
            }
            if (num != n) {
                return false;
            }
        }
        return true;
    }
}
