#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<unsigned long long> v(n);
  for (int i = 0; i < n; ++i) std::cin >> v[i];
  unsigned long long c = v[0] | v[n - 1];
  for (int i = 1; i < n - 1; ++i) c |= v[i];
  unsigned long long res = 0;
  for (int i = 0; i < n - 1; ++i) {
    res += (c > v[i]);
  }
  std::cout << res << std::endl;
}
