#include <bits/stdc++.h>
using namespace std;
int cnt[26];
int main() {
  string s;
  cin >> s;
  for (int i = 0; i < s.length(); i++) {
    if (s[i] != '?') cnt[s[i] - 'A']++;
  }
  int c = 26 - *max_element(cnt, cnt + 26);
  if (s.length() < c) {
    cout << "-1";
    return 0;
  }
  int f = 0;
  for (int i = 0; i < 26; i++) {
    f += (cnt[i] == 0);
  }
  for (int i = 0; i < s.length(); i++) {
    if (s[i] == '?') {
      for (int j = 0; j < 26; j++) {
        if (cnt[j] == 0) {
          s[i] = j + 'A';
          cnt[j]++;
          break;
        }
      }
    }
  }
  if (f == 0) {
    for (int i = 0; i < s.length(); i++) {
      if (s[i] == '?') {
        for (int j = 0; j < 26; j++) {
          if (cnt[j] > 0) {
            s[i] = j + 'A';
            cnt[j]++;
          } else {
            break;
          }
        }
      }
    }
  }
  cout << s;
  return 0;
}
