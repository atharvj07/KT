import java.util.Scanner;

public class IahubinaAndATree {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        boolean possible = checkTree(c);
        System.out.println(possible ? "YES" : "NO");
    }

    private static boolean checkTree(int[] c) {
        int sum = 0;
        for (int value : c) {
            sum += value;
        }
        return sum == c.length - 1;
    }
}
