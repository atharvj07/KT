import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextLong();
        }
        int k = (n + 1) / 2 - 1;
        for (int i = 0; i < n; i++) {
            System.out.print((i < k || a[i] == a[k]) ? 1 : 0);
        }
        System.out.println();
        for (int i = 0; i < n; i++) {
            System.out.print((i < k || b[i] == b[k]) ? 1 : 0);
        }
        sc.close();
    }
}
