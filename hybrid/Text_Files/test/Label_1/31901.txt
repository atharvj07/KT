import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int x = input.nextInt();
        long ans = -1;
        for (int b = x - 1; b >= 2; b--) {
            long pow = 1;
            boolean ok = true;
            for (int i = 1; i <= n; i++) {
                if (pow < (long) Long.MAX_VALUE / b) {
                    pow *= b;
                } else {
                    ok = false;
                    break;
                }
                if (pow < Long.MAX_VALUE - 1) {
                    pow++;
                } else {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                ans = b;
                break;
            }
        }
        System.out.println(ans);
    }
}
