import java.util.*;
import java.io.*;

public class Main {

    private static int solve(int n, String s) {
        int[] black = new int[n];
        int[] red = new int[n];
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'b') {
                black[i]++;
            } else if (s.charAt(i) == 'r') {
                red[i]++;
            }
        }
        int ans = Math.min(black[0], red[0]);
        for (int i = 1; i < n; i++) {
            black[i] += black[i - 1];
            red[i] += red[i - 1];
            ans = Math.min(ans, black[i] + red[i - 1]);
            ans = Math.min(ans, red[i] + black[i - 1]);
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        System.out.println(solve(n, s));
    }
}
