#include <algorithm>
#include <iostream>

int main() {
  long long a, b, s, t;
  std::cin >> a >> b;
  s = (a < b) ? a : b;
  a -= s;
  t = (a < b) ? a : b;
  std::cout << s << '\n';
  for (long long i = 1; i <= s; i++) std::cout << i << ' ';
  std::cout << '\n';
  std::cout << t << '\n';
  for (long long i = 1; i <= t; i++) std::cout << i << ' ';
  return 0;
}
