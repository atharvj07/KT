t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    flowers = []
    for _ in range(m):
        a, b = map(int, input().split())
        flowers.append((a, b, a - b))
    flowers.sort(key=lambda x: (-x[2], -x[0]))
    total_happiness = 0
    for a, b, _ in flowers:
        if n > 1:
            total_happiness += a + (n - 1) * b
            n = 0
        elif n == 1:
            total_happiness += a
            n -= 1
        if n == 0:
            break
    print(total_happiness)
