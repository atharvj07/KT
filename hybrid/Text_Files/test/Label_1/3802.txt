#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        int ans = n / k;
        int rem = n % k;
        if (rem > k / 2) {
            ans++;
        }
        cout << ans << endl;
    }
    return 0;
}
