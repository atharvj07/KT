#include <bits/stdc++.h>
using namespace std;
#define ll long long
int n,x,y,ans,maxx,maxy,mnx,mny;
vector<pair<int,int>>a;
int main(){
    cin>>n;
    for(int i=0;i<n;i++){
        cin>>x>>y;a.push_back({x,y});
    }
    sort(a.begin(),a.end());
    maxx=a[n-1].first;
    maxy=a[n-1].second;
    mnx=a[0].first;
    mny=a[0].second;
    if(mnx==maxx)ans++;
    if(mny==maxy)ans++;
    cout<<ans<<endl;
}
