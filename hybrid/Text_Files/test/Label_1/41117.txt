n = int(input().strip())
s = input().strip()

MOD = 998244353

# Count continuous characters from the start
start_count = 1
for i in range(1, n):
    if s[i] == s[0]:
        start_count += 1
    else:
        break

# Count continuous characters from the end
end_count = 1
for i in range(n - 2, -1, -1):
    if s[i] == s[-1]:
        end_count += 1
    else:
        break

# If the first and the last character are the same, we have two cases:
# 1. Removing any substring starting from the beginning up to the point where characters start changing.
# 2. Removing any substring from the end up to the point where characters start changing.
# Since the characters at both ends are the same, these operations are independent and can be combined.
# Thus, the total number of ways is (start_count + 1) * (end_count + 1).
# If the characters are different, the answer is simply start_count + end_count + 1.
if s[0] == s[-1]:
    print((start_count + 1) * (end_count + 1) % MOD)
else:
    print((start_count + end_count + 1) % MOD)
