import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][][] dp = new int[n + 1][11][2];
        int[][] profit = new int[n + 1][11];
        int[][] time = new int[n + 1][10];
        for (int i = 1; i <= n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int j = 1; j <= 10; j++) {
                time[i][j - 1] = Integer.parseInt(st.nextToken());
            }
        }
        for (int i = 1; i <= n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int j = 0; j <= 10; j++) {
                profit[i][j] = Integer.parseInt(st.nextToken());
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= 10; j++) {
                for (int k = 0; k < 2; k++) {
                    dp[i][j][k] = Math.max(dp[i - 1][j][k], dp[i - 1][j - 1][k] + profit[i][time[i][j - 1] + k]);
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= 10; i++) {
            ans = Math.max(ans, dp[n][i][1]);
        }
        System.out.println(ans);
    }
}
