import sys
input = sys.stdin.readline

def cardgame(arr):
    arr.sort(key=lambda x: x[0], reverse=True)
    csum, jsum = 0, 0
    i, j = 0, len(arr)-1
    while i <= j:
        if arr[i][1] > arr[j][1]:
            csum += arr[i][1]
            i += 1
        else:
            csum += arr[j][-1]
            j -= 1
        if i <= j:
            jsum += arr[j][1]
            j -= 1
        if i <= j:
            csum += arr[i][1]
            i += 1
    return csum, jsum

n = int(input())
arr = []
for i in range(n):
    s, *l = map(int, input().split())
    arr.append([s, max(l)] + l)
print(*cardgame(arr))
