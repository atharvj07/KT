import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        int n = scanner.nextInt();
        if (n < 3 || m > (n - 1) * 2) {
            System.out.println(-1);
        } else {
            int ans = 0;
            while (m > 0) {
                if (m >= n - 1) {
                    ans += 2;
                    m -= n - 1;
                } else {
                    ans += 1;
                    m -= 1;
                }
            }
            System.out.println(ans);
        }
    }
}
