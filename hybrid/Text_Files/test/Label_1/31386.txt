import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), d = in.nextInt(), ans = 0, j = 0;
        int[] a = new int[n];
        for(int i = 0; i < n; i++){
            a[i] = in.nextInt();
            while(a[i] - a[j] > d)j++;
            ans += i-j;
        }
        System.out.println(ans);
    }
}
