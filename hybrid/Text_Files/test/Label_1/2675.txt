#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;

ll pw(ll a, ll b) {
    ll ans = 1;
    while(b) {
        if(b % 2) ans = ans * a % mod;
        a = a * a % mod;
        b /= 2;
    }
    return ans;
}

ll inv(ll a) { return pw(a, mod - 2); }

const int N = 1 << 19;
ll dp[N][7];

int main() {
    int n, k; cin >> k >> n;
    if(n + 1 > (1 << k)) return cout << 0 << endl, 0;
    map<int, int> mp;
    for(int i = 0; i < n; i++) {
        int v; string s;
        cin >> v >> s;
        if(s == "white") mp[v] = 1;
        else if(s == "yellow") mp[v] = 2;
        else if(s == "green") mp[v] = 3;
        else if(s == "blue") mp[v] = 4;
        else if(s == "red") mp[v] = 5;
        else mp[v] = 6;
    }
    dp[1][0] = 1;
    for(int i = 2; i < N; i++) {
        if(i % 2 == 1) continue;
        int p = i / 2;
        if(mp[i] != 0) {
            dp[i][mp[i]] = (dp[p][mp[i]] + dp[p][0]) % mod;
            continue;
        }
        for(int j = 0; j <= 6; j++) {
            if(j == 1) {
                dp[i][j] = dp[p][0];
                if(mp[p] == 2) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
            else if(j == 2) {
                dp[i][j] = dp[p][0];
                if(mp[p] == 1) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
            else if(j == 3) {
                dp[i][j] = (dp[p][0] + dp[p][4]) % mod;
                if(mp[p] == 4) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
            else if(j == 4) {
                dp[i][j] = (dp[p][0] + dp[p][3]) % mod;
                if(mp[p] == 3) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
            else if(j == 5) {
                dp[i][j] = (dp[p][0] + dp[p][6]) % mod;
                if(mp[p] == 6) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
            else {
                dp[i][j] = (dp[p][0] + dp[p][5]) % mod;
                if(mp[p] == 5) {
                    dp[i][j] -= dp[p][mp[p]];
                    if(dp[i][j] < 0) dp[i][j] += mod;
                }
            }
        }
    }
    ll ans = 0;
    for(int i = 0; i <= 6; i++) ans = (ans + dp[1 << k][i]) % mod;
    cout << ans << endl;
    return 0;
}
