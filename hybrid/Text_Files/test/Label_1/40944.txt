n, k = map(int, input().split())

# Adjust n to account for the initial pipe with water
n -= 1

# Calculate the total number of pipes that can be achieved with the available splitters
total_pipes = k * (k - 1) // 2

# Check if it's possible to build the pipeline
if n > total_pipes:
    print(-1)
else:
    # Binary search to find the minimum number of splitters required
    low, high = 1, k
    while low < high:
        mid = (low + high) // 2
        if mid * (mid + 1) // 2 >= n + 1:
            high = mid
        else:
            low = mid + 1
    print(k - low + 1)
