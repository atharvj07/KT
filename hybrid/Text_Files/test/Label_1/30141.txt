import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        int k = in.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) a[i] = in.nextInt();
        long num = n / a[0];
        int type = 0;
        for (int i = 1; i < k; i++) {
            long buy = n / a[i];
            if (buy > num) {
                type = i;
                num = buy;
            }
        }
        System.out.println(type + 1 + " " + num);
    }
}
