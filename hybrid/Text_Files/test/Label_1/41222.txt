import sys

input = sys.stdin.readline

s = list(input().strip())
sl = len(s)
i, j = 0, sl - 1
nc = 0
while i <= j:
  if s[i] == s[j]:
    i += 1
    j -= 1
  else:
    nc += 1
    j -= 1
t = s.copy()
t.reverse()
k = 0
for i in range(sl):
  if s[i] != t[i]:
    k = i
    break
if nc & 1:
  nc += 1
  c = ord(s[k + nc // 2])
  if s[k + nc // 2 - 1] < s[k + nc // 2 + 1]:
    s[k + nc // 2 - 1] = chr(c + 1)
  else:
    s[k + nc // 2 + 1] = chr(c - 1)
s = s[0:k + nc] + s[k::-1]
print("".join(s))
