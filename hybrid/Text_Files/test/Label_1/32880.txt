import java.util.*;

public class ArrayOperations {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[][] operations = new int[m][3];
        for (int i = 0; i < m; i++) {
            operations[i][0] = scanner.nextInt() - 1;
            operations[i][1] = scanner.nextInt() - 1;
            operations[i][2] = scanner.nextInt();
        }

        int[][] queries = new int[k][2];
        for (int i = 0; i < k; i++) {
            queries[i][0] = scanner.nextInt() - 1;
            queries[i][1] = scanner.nextInt() - 1;
        }

        // Build a difference array based on the operations
        int[] diff = new int[m + 1];
        for (int i = 0; i < m; i++) {
            int l = operations[i][0];
            int r = operations[i][1];
            int d = operations[i][2];
            diff[l] += d;
            diff[r + 1] -= d;
        }

        // Apply the difference array to the original array
        for (int i = 1; i < m + 1; i++) {
            diff[i] += diff[i - 1];
        }

        // Apply the queries to the modified array
        for (int i = 0; i < k; i++) {
            int x = queries[i][0];
            int y = queries[i][1];
            for (int j = x; j <= y; j++) {
                a[j] += diff[j];
            }
        }

        // Print the final array
        for (int i : a) {
            System.out.print(i + " ");
        }
    }
}
