s, x1, x2 = map(int, input().split())
t1, t2 = map(int, input().split())
p, d = map(int, input().split())

# Time for Igor to walk directly
time_walk = abs(x2 - x1) * t2

# Time for the tram to catch Igor and then reach x2
if d == 1 and x1 >= p and x1 <= x2:
    # Tram catches Igor directly on its way to x2
    time_tram = (x2 - p) * t1
elif d == -1 and x1 <= p and x1 >= x2:
    # Tram catches Igor directly on its way to x2
    time_tram = (p - x2) * t1
else:
    # Tram needs to turn around first
    if d == 1:
        if x1 < x2:
            # Tram goes to s, then to 0, then to x2
            time_tram = (s - p + s + x2) * t1
        else:
            # Tram goes to s, turns around, then goes to x1, and then to x2
            time_tram = (s - p + s + abs(x2 - x1)) * t1
    else:
        if x1 > x2:
            # Tram goes to 0, then to s, then to x2
            time_tram = (p + s + x2) * t1
        else:
            # Tram goes to 0, turns around, then goes to x1, and then to x2
            time_tram = (p + s + abs(x2 - x1)) * t1

# Minimum time is the least of walking or taking the tram
print(min(time_walk, time_tram))
