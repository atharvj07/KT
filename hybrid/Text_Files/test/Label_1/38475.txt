n, k = map(int, input().split())
d = list(map(int, input().split()))

remainder_counts = [0] * k

for candy in d:
    remainder_counts[candy % k] += 1

count = min(remainder_counts[0], 1)  # Handle the special case where candies % k == 0

for i in range(1, (k + 1) // 2):
    count += min(remainder_counts[i], remainder_counts[k - i])

if k % 2 == 0:
    count += min(remainder_counts[k // 2], 1)

print(count * 2)
