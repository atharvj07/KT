#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
int color[MAXN];
int deg[MAXN];
int ansColor;
int ansDeg = 0;
queue<int> q;

int main() {
  int n, m;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) {
    cin >> color[i];
  }
  for (int i = 0; i < m; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  for (int i = 1; i <= n; i++) {
    for (auto u : adj[i]) {
      if (color[u] != color[i]) {
        deg[color[i]]++;
      }
    }
    if (deg[color[i]] > ansDeg) {
      ansDeg = deg[color[i]];
      ansColor = color[i];
    }
  }
  cout << ansColor << endl;
  return 0;
}
