#include <bits/stdc++.h>
using namespace std;

int main()
{
    int t, n;
    cin >> t;
    while (t--)
    {
        cin >> n;
        vector<long long> h(n);
        for (int i = 0; i < n; i++) cin >> h[i];
        sort(h.begin(), h.end());
        bool ok = true;
        for (int i = 1; i < n; i++) ok &= h[i] > h[i - 1];
        cout << (ok ? "YES\n" : "NO\n");
    }
}
