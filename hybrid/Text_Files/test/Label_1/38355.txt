m, n = map(int, input().split())
B = [list(map(int, input().split())) for _ in range(m)]

A = [[1]*n for _ in range(m)]
for i in range(m):
    for j in range(n):
        if B[i][j] == 0:
            for k in range(m):
                A[k][j] = 0
            for l in range(n):
                A[i][l] = 0

possible = True
for i in range(m):
    for j in range(n):
        if B[i][j] == 1:
            row = col = False
            for k in range(m):
                row |= A[k][j]
            for l in range(n):
                col |= A[i][l]
            if not row or not col:
                possible = False
                break

if possible:
    print("YES")
    for row in A:
        print(*row)
else:
    print("NO")
