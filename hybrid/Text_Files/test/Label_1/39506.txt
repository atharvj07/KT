n = int(input())
teams = [list(map(int, input().split())) for _ in range(n)]

home_away_conflicts = [0] * n

color_home_count = {}

for i in range(n):
    home_color = teams[i][0]
    if home_color in color_home_count:
        color_home_count[home_color] += 1
    else:
        color_home_count[home_color] = 1

for i in range(n):
    away_color = teams[i][1]
    if away_color in color_home_count:
        home_away_conflicts[i] += color_home_count[away_color]

for i in range(n):
    print(f"{n - 1 + home_away_conflicts[i]} {n - 1 - home_away_conflicts[i]}")
