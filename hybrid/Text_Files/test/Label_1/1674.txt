#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define M 45
int p[M], cnt;
vector<ll> ans;
void precalc() {
  ll val = 3;
  while (1) {
    ans.push_back(val);
    p[cnt++] = val;
    val *= 3;
    if (val > (ll)1e18) break;
  }
}
ll get(ll n) {
  int op = 0;
  for (int i = 0; i < cnt; i++) {
    while (n >= p[i]) {
      n -= p[i];
      op++;
    }
    if (!n) {
      if (op & 1) {
        op++;
        n = p[i];
      }
      break;
    }
  }
  if (!n) return ans[op];
  return -1;
}
int main() {
  precalc();
  int q;
  cin >> q;
  while (q--) {
    ll n;
    cin >> n;
    ll sol = get(n - 1);
    if (sol >= 0)
      sol++;
    else
      sol = n;
    cout << sol << endl;
  }
  return 0;
}
