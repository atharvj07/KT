import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] w = new long[n];
        for (int i = 0; i < n; i++) {
            w[i] = 1 << in.nextInt();
        }
        long sum = 0;
        int steps = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (sum + w[i] <= sum) {
                steps++;
                sum = w[i];
            } else {
                sum += w[i];
            }
        }
        System.out.println(steps);
    }
}
