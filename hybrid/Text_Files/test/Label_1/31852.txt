import java.util.*;

public class Main {

    public static final String[] HANDS = {"Rock", "Fire", "Scissors", "Snake", "Human", "Tree", "Wolf", "Sponge", "Paper", "Air", "Water", "Dragon", "Devil", "Lightning", "Gun"};
    public static final int[] DEFEAT_TABLE = {7, 0, 6, 2, 5, 3, 4, 1};

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int N;
        while ((N = input.nextInt()) != 0) {
            Set<Integer> winners = new HashSet<>();
            Set<Integer> losers = new HashSet<>();
            for (int i = 0; i < N; i++) {
                int hand = Arrays.asList(HANDS).indexOf(input.next());
                for (int j = 1; j <= 7; j++) {
                    int defeated = (hand + j) % 15;
                    if (!losers.contains(defeated)) {
                        winners.add(hand);
                    }
                }
                for (int j = 8; j <= 14; j++) {
                    int defeated = (hand + j) % 15;
                    if (losers.contains(defeated)) {
                        winners.remove(hand);
                        break;
                    }
                }
                losers.add(hand);
            }
            if (winners.isEmpty()) {
                System.out.println("Draw");
            } else {
                System.out.println(HANDS[winners.iterator().next()]);
            }
        }
    }
}
