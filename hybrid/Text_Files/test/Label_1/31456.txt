import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner console = new Scanner(System.in);
        int n = console.nextInt(), d = console.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = console.nextInt();
        }
        Arrays.sort(arr);
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (arr[i + 1] - arr[i] > d) {
                ans++;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
