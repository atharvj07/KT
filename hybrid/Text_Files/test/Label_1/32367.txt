import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), ans = 0;
        long sum = 0, a = 0;
        for (int i = 0; i < n; i++) {
            long ai = sc.nextLong() * 2;
            sum += ai;
            if (a > 0) ans += (ai - a) / 2;
            a = ai;
        }
        ans += (2 * a - (sum + 1)) / 2;
        System.out.println(ans);
    }
}
