#include <bits/stdc++.h>
using namespace std;
#define int long long
vector<set<int>> c, r;
int32_t main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < n + m; i++) {
        c.push_back(set<int>());
        r.push_back(set<int>());
    }
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int x;
            cin >> x;
            x--;
            r[i].insert(x / m);
            c[x / m].insert(i);
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (r[i].size() > 1 or c[i].size() > 1) {
            ans = -1;
            break;
        }
        if (*r[i].begin() != i) {
            int shift = i - *r[i].begin();
            for (int j = 0; j < shift; j++) {
                for (int x : c[i]) {
                    r[x].erase(i);
                    r[x].insert((x + shift) % n);
                }
            }
            ans += shift;
        }
    }
    cout << ans << "\n";
}
