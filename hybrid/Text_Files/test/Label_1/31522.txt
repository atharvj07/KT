import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int s = sc.nextInt();
        if ((Math.abs(a) + Math.abs(b)) % 2 != s % 2 || (Math.abs(a) + Math.abs(b)) > s) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
        }
    }
}
