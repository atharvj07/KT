MOD = 10**9 + 7

def count_strings(s):
    dp = [0] * (len(s) + 1)
    dp[0] = 1
    for i in range(1, len(s) + 1):
        dp[i] = dp[i-1]
        if s[i-1] == 'u' and i > 1 and s[i-2] == 'u':
            dp[i] = (dp[i] + dp[i-2]) % MOD
        elif s[i-1] == 'n' and i > 1 and s[i-2] == 'n':
            dp[i] = (dp[i] + dp[i-2]) % MOD
    return dp[-1]

s = input().strip()
print(count_strings(s))
