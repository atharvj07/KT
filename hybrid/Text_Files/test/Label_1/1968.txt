#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ll;
const int mod = 1e9 + 7;
ll fastpow(ll x, ll y) {
  ll ret = 1;
  while (y) {
    if (y & 1) ret = ret * x % mod;
    x = x * x % mod;
    y >>= 1;
  }
  return ret;
}
int main() {
  string s;
  cin >> s;
  int n = s.size(), L = 0, R = 0;
  for (char c : s) {
    if (c == '(') ++L;
    else ++R;
  }
  if (L != R) {
    cout << 0 << endl;
    return 0;
  }
  cout << fastpow(2, L) - 1 << endl;
}
