#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    vector<int> a(n), ans(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int prv = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            if (prv == 0) {
                ans[i] = 1;
                prv = 1;
            } else {
                ans[i] = prv + 1;
                prv++;
            }
        } else {
            prv = a[i];
            ans[i] = prv;
        }
    }
    set<int> s(ans.begin(), ans.end());
    if (s.size() != q) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for (auto x : ans) { cout << x << " "; }
    return 0;
}
