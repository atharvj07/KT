#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    vector<long long> ans;
    
    while (t--) {
        long long s, a, b, c;
        cin >> s >> a >> b >> c;
        long long x = s / c;
        ans.push_back(((x + a - 1) / a) * b + x);
    }

    for (auto x : ans) {
        cout << x << endl;
    }

    return 0;
}
