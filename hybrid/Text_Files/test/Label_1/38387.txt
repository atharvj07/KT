import sys

n = int(input())
arr = []
for _ in range(n):
    arr.append(list(map(int, input().split())))

ans = []
tot = 0
lsum = rsum = arr[0][0]
for i in range(1, n):
    lsum += arr[i][1] - min(arr[i - 1][1], arr[i][0] - 1)
    rsum += min(arr[i][0], arr[i][1] - 1)
    if min(arr[i][1], arr[i - 1][1]) < max(arr[i][0], arr[i][1] - 1):
        ans.append(-1)
        tot = -1
        break
    ans.append(max(arr[i][0], arr[i][1] - 1))
    tot += arr[i][1] - ans[i] + 1

if tot == -1:
    print(tot)
else:
    print(tot)
    ans.insert(0, arr[0][0])
    print(*ans)
