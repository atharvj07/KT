#include <iostream>
using namespace std;
#define ll long long
int main() {
  ll n, m;
  cin >> n >> m;
  ll a[n], sum = 0;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int i = 0; i < n; i++) {
    sum += a[i];
    sum %= m;
    if (sum == 0) {
      cout << "YES" << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    sum = a[i];
    for (int j = i + 1; j < n; j++) {
      sum += a[j];
      sum %= m;
      if (sum == 0) {
        cout << "YES" << endl;
        return 0;
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
