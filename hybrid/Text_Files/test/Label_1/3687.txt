#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  for (int i = 0; i < m; i++) {
    std::cin >> b[i];
  }
  std::sort(a.begin(), a.end());
  std::sort(b.begin(), b.end());
  int ans = n;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < m && b[j] < a[i]) {
      j++;
    }
    if (j < m && b[j] == a[i]) {
      ans--;
      j++;
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
