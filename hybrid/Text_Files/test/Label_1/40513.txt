import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
c = list(map(int, input().split()))
edges = [[] for _ in range(n+1)]
for _ in range(m):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)

ok = True
for i in c:
    for j in c:
        if i != j and i in edges[j]:
            ok = False

if ok: print(n-m-1)
else: print(n-m)
