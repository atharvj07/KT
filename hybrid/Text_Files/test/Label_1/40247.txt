from collections import deque
n=int(input())
a=list(map(int,input().split()))+[0]
adj=[[] for _ in range(n+1)]
for i in range(1,n+1):
 for j in range(i+1,n+1):
  if a[i] and a[j]:
   adj[i]+=j,
   adj[j]+=i,
vis=[0]*(n+1)
vis[1]=1
q=deque([1])
while q:
  u=q.popleft()
  for v in adj[u]:
   if not vis[v]:
    print(u,v)
    q.append(v)
    vis[v]=1
if sum(vis)<n:print(-1)
