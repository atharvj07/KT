from collections import deque
import sys

n = int(input())
is_leaf = [True] * (n + 1)
adj = [[]]
for i in range(n): adj.append([])
for i in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)
    is_leaf[a] = is_leaf[b] = False

# BFS to find two leafs
que = deque([1])
d = [-1] * (n + 1)
d[1] = 0
can_reach = False
while que:
    now = que.popleft()
    if is_leaf[now]:
        if can_reach: break
        can_reach = True
        # find the other leaf
        for i in adj[now]:
            if d[i] == -1:
                d[i] = d[now] + 1
                que.append(i)
    else:
        for i in adj[now]:
            if d[i] == -1:
                d[i] = d[now] + 1
                que.append(i)

if not can_reach:
    print("NO")
    sys.exit(0)

# find the path
path = [now]
while now != 1:
    for i in adj[now]:
        if d[i] == d[now] - 1:
            now = i
            path.append(now)
            break

# check if there's an edge on the path
flag = False
for i in range(len(path) - 1):
    if adj[path[i]][0] == path[i + 1]:
        adj[path[i]].remove(path[i + 1])
        flag = True
    if adj[path[i]][1] == path[i + 1]:
        adj[path[i]].remove(path[i + 1])
        flag = True
    if flag: break

if not flag:
    print("NO")
else:
    print("YES")
