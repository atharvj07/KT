import java.util.*;

class Main {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Interval[] ns = new Interval[n];
        Interval[] ew = new Interval[m];
        for(int i=0; i<n; i++){
            ns[i] = new Interval(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        for(int i=0; i<m; i++){
            ew[i] = new Interval(sc.nextInt(), sc.nextInt(), sc.nextInt());
        }
        int maxNorth=0;
        int minNorth=0;
        int maxEast=0;
        int minEast=0;

        for(Interval i:ns){
            maxNorth=Math.max(maxNorth, i.max);
        }

        for(Interval i:ew){
            maxEast=Math.max(maxEast, i.max);
        }

        for(Interval i:ns){
            minNorth=Math.min(minNorth, i.min);
        }

        for(Interval i:ew){
            minEast=Math.min(minEast, i.min);
        }

        if(minEast>=maxEast || minNorth>=maxNorth){
            System.out.println("INF");
            return;
        }
        long res=(long) (maxNorth-minNorth)*(long)(maxEast-minEast);
        System.out.println(res);
    }
}

class Interval {
    int min, max, c;
    public Interval(int m, int M, int C){
        min=m;
        max=M;
        c=C;
    }
}
