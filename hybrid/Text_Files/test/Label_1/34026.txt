import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }

            char[][] grid = new char[n][n];
            for (int i = 0; i < n; i++) {
                String line = sc.next();
                for (int j = 0; j < n; j++) {
                    grid[i][j] = line.charAt(j);
                }
            }

            int maxSide = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (grid[i][j] == '*') {
                        continue;
                    }

                    int side = 1;
                    while (i + side < n && j + side < n) {
                        boolean valid = true;
                        for (int k = i; k < i + side; k++) {
                            for (int l = j; l < j + side; l++) {
                                if (grid[k][l] == '*') {
                                    valid = false;
                                    break;
                                }
                            }
                            if (!valid) {
                                break;
                            }
                        }

                        if (valid) {
                            side++;
                        } else {
                            break;
                        }
                    }

                    maxSide = Math.max(maxSide, side - 1);
                }
            }

            System.out.println(maxSide);
        }
    }
}
