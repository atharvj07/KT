import java.util.*;

public class Main {
    private static class Freq {
        private String s;
        private int freq;
        public Freq(String s) {
            this.s = s;
            freq = s.length();
        }
        private void incrementFreq() { freq++; }
        private boolean isMostFrequent() { return freq == s.length(); }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int count = in.nextInt();
        Map<Character, Freq> map = new HashMap<>();
        List<String> input = new ArrayList<>();
        for (int i = 0; i < count; i++) {
            String s = in.next();
            input.add(s);
            for (char c : s.toCharArray()) {
                if (!map.containsKey(c)) {
                    map.put(c, new Freq(s));
                } else {
                    Freq freq = map.get(c);
                    freq.incrementFreq();
                    freq.s = (freq.isMostFrequent() ? s : freq.s);
                }
            }
        }
        StringBuilder result = new StringBuilder();
        for (Freq freq : map.values()) {
            if (freq.isMostFrequent()) {
                result.append(freq.s);
            } else {
                result.setLength(0);
                System.out.println("NO");
                break;
            }
        }
        if (result.length() > 0) {
            String sorted = result.toString();
            for (String s : input) {
                if (result.toString().equals(s)) {
                    sorted = s;
                    break;
                }
            }
            System.out.println(sorted);
        }
    }
}
