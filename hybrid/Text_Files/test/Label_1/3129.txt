#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::string> grid(n);
  for (int i = 0; i < n; i++) {
    std::cin >> grid[i];
  }
  bool has_wall = false;
  int x, y;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '*') {
        if (has_wall) {
          std::cout << "NO" << std::endl;
          return 0;
        }
        x = i + 1;
        y = j + 1;
        has_wall = true;
      }
    }
  }
  if (!has_wall) {
    std::cout << "NO" << std::endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (grid[i][y - 1] == '*') {
      std::cout << "NO" << std::endl;
      return 0;
    }
  }
  for (int j = 0; j < m; j++) {
    if (grid[x - 1][j] == '*') {
      std::cout << "NO" << std::endl;
      return 0;
    }
  }
  std::cout << "YES" << std::endl << x << " " << y << std::endl;
  return 0;
}
