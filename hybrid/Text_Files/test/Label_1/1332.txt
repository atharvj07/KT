#include <bits/stdc++.h>
using namespace std;

string solve() {
  int n, l, r;
  cin >> n >> l >> r;
  unordered_map<int, int> a;
  for (int x, i = 0; i < n; ++i) { cin >> x, ++a[x], a[x + n] += n; }
  for (int x, i = 0; i < n; ++i) { cin >> x, --a[x], ++a[x - n]; }
  for (int i = l - 1; i < r; ++i) if (a[i] && a[i + n]) return "LIE";
  return "TRUTH";
}
int main() { cout << solve() << endl; }
