import sys

n = int(input())
x = [int(s) for s in sys.stdin.readline().split(' ')]
y = [int(s) for s in sys.stdin.readline().split(' ')]

def sq_dist(a, b):
    return (a[0] - b[0])**2 + (a[1] - b[1])**2

mn_dist = min(map(sq_dist, zip(x, y), [(0, 0)] * n))
mn_x, mn_y = 0, 0

for i in range(n):
    if sq_dist((x[i], y[i]), (0, 0)) == mn_dist:
        mn_x, mn_y = x[i], y[i]
        break

if mn_y != 0:
    print(-1)
elif len(set(x)) == 1:
    r = max(abs(xi) for xi in x)
    print(r)
else:
    mx = max(abs(xi) for xi in x)
    r = mx + 0.5
    print(r)
