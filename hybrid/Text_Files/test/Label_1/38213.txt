t,_,q = map(int,input().split())
friends = sorted(zip(map(int,input().split()),range(1,t+1)),reverse=True)
d = {}
for ti,i in friends:
    d[i] = ti
msk = 0
for ti,i in friends[:min(t,6)]:
    msk |= 1<<i
for _ in range(q):
    ti,i = map(int,input().split())    
    if ti==1:
        msk |= 1<<i
    else:
        print('YES'if msk & (1<<i) else 'NO')
