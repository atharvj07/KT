#include <bits/stdc++.h>
using namespace std;
#define INF 1000000000
int main() {
    int n;
    cin >> n;
    vector<int> d(n);
    for (auto &v : d) cin >> v;
    int s, t;
    cin >> s >> t;
    int ans = min(abs(s - t), n - abs(s - t));
    for (int i = 1; i < n; ++i) {
        ans = min(ans, d[i] + min(abs(s - i) + abs(i - t), n - abs(s - i) + n - abs(i - t)));
    }
    cout << ans << endl;
    return 0;
}
