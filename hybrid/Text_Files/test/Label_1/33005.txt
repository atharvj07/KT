import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    List<Card> cards = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      String card = scanner.next();
      cards.add(new Card(card.substring(0, 1), card.substring(1)));
    }
    System.out.println(isPossible(cards) ? "YES" : "NO");
  }

  private static boolean isPossible(List<Card> cards) {
    if (cards.size() == 1) {
      return true;
    }
    Map<Card, Integer> count = new HashMap<>();
    for (Card card : cards) {
      count.put(card, count.getOrDefault(card, 0) + 1);
    }
    List<Card> uniqueCards = new ArrayList<>(count.keySet());
    Collections.sort(uniqueCards, Comparator.comparingInt(count::get).reversed());
    for (Card card : uniqueCards) {
      if (count.get(card) % 2 == 1) {
        return false;
      }
      int countOfPairs = count.get(card) / 2;
      for (int i = 0; i < countOfPairs; i++) {
        count.put(card, count.get(card) - 2);
        count.put(card, count.getOrDefault(card, 0) + 1);
      }
    }
    return true;
  }

  private static class Card {
    private String value;
    private String suit;

    public Card(String value, String suit) {
      this.value = value;
      this.suit = suit;
    }

    @Override
    public boolean equals(Object o) {
      if (this == o) {
        return true;
      }
      if (o == null || getClass() != o.getClass()) {
        return false;
      }
      Card card = (Card) o;
      return Objects.equals(value, card.value) && Objects.equals(suit, card.suit);
    }

    @Override
    public int hashCode() {
      return Objects.hash(value, suit);
    }
  }
}
