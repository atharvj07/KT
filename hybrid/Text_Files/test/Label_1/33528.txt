import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] p = new int[n], c = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            c[i] = sc.next().charAt(0) - 'a' + 1;
        }
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt() - 1;
        }

        int[] dp = new int[n];
        for (int i = 0; i < n; i++) {
            dp[i] = 1;
        }

        for (int i = 1; i < n; i++) {
            dp[i] = Math.min(dp[p[i]] + 1, dp[i - 1] + c[i - 1]);
        }

        for (int i = 0; i < k; i++) {
            System.out.println(dp[a[i]]);
        }
    }
}
