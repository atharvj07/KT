import java.util.Scanner;

public class CircleColoring {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        String s = scanner.next();

        long[][] dp = new long[n][m]; // dp[i][j] = number of ways to paint the circle with i arcs painted red and j moves
        dp[0][0] = 1; // base case: no arcs painted red, no moves made
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (dp[i][j] > 0) {
                    // paint the next arc red
                    int nextI = (i + 1) % n;
                    int nextJ = (j + 1) % m;
                    dp[nextI][nextJ] = (dp[nextI][nextJ] + dp[i][j]) % MOD;

                    // paint the next arc blue
                    nextI = (i - 1 + n) % n; // handle negative indices
                    dp[nextI][nextJ] = (dp[nextI][nextJ] + dp[i][j]) % MOD;
                }
            }
        }

        long result = 0;
        for (int i = 0; i < n; i++) {
            if (dp[i][m - 1] > 0) {
                // check if the circle generates the string s from this starting point
                boolean valid = true;
                int startIndex = i;
                for (int j = 0; j < m; j++) {
                    if (s.charAt(j) == 'R') {
                        startIndex = (startIndex + 1) % n;
                    } else {
                        startIndex = (startIndex - 1 + n) % n;
                    }
                    if (startIndex != i) {
                        valid = false;
                        break;
                    }
                }
                if (valid) {
                    result = (result + dp[i][m - 1]) % MOD;
                }
            }
        }

        System.out.println(result);
    }
}
