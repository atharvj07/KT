import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int k = sc.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int maxScore = 0;
        int sum = 0;
        for (int i = 0; i < k - 1; i++) {
            sum += a[i];
        }
        for (int i = k - 1; i < n; i++) {
            sum += a[i];
            maxScore = Math.max(maxScore, sum);
            sum -= a[i - k + 1];
        }

        System.out.println(maxScore);
    }
}
