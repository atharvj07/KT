import queue
n, k, d = map(int, input().split())
s = list(map(int, input().split()))
g = [[] for _ in range(n+1)]
for _ in range(n-1):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)
vis = [False]*(n+1)
ans = []
q = queue.Queue()
for i in range(1, n+1):
    if i in s:
        vis[i] = True
    else:
        q.put((i, -1))
while not q.empty():
    u, p = q.get()
    if len(ans) == (n-1)-k:
        break
    ans.append(p)
    for v in g[u]:
        if vis[v]:
            continue
        vis[v] = True
        q.put((v, p))
print(len(ans))
print(*ans)
