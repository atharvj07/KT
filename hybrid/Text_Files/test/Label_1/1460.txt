#include <iostream>
#include <vector>
#include <set>
using namespace std;
#define MAXN 200000
vector<vector<int>> G(MAXN+1);
int subtree_size[MAXN+1], depth[MAXN+1], ancestors[MAXN+1];
vector<int> path;
int calculate_depth(int current_node, int parent_node) {
    depth[current_node]=depth[parent_node]+1;
    subtree_size[current_node]=1;
    ancestors[current_node]=parent_node;
    for(int j:G[current_node]){
        if(j!=parent_node){
            subtree_size[current_node]+=calculate_depth(j,current_node);
        }
    }
    return subtree_size[current_node];
}
int find_path(int current_node, int destination_node, set<int>& visited_nodes){
    visited_nodes.insert(current_node);
    int sz=0;
    if(current_node==destination_node){
        sz=0;
    }
    else{
        for(int j:G[current_node]){
            if(!visited_nodes.count(j)){
                sz=1+find_path(j, destination_node, visited_nodes);
                if(sz>=0){
                    path.push_back(current_node);
                    return sz;
                }
            }
        }
    }
    return -1;
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, x; cin>>n>>x;
    for(int i=0;i<n-1;++i){
        int a, b; cin>>a>>b;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    calculate_depth(1, 1);
    int first_move=0;
    set<int> visited_nodes;
    int sz=0;
    path.push_back(x);
    sz=find_path(x, 1, visited_nodes);
    if(depth[x]>=depth[1]-sz)
        first_move=depth[1]-sz-1;
    cout<<first_move+depth[x]<<'\n';
    return 0;
}
