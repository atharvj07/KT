#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    map<int, int> freq;
    vector<int> cards[n];
    for (int i = 0; i < n; ++i) {
        int m;
        cin >> m;
        for (int j = 0; j < m; ++j) {
            int a;
            cin >> a;
            freq[a]++;
            cards[i].push_back(a);
        }
    }
    for (int i = 0; i < n; ++i) {
        bool canWin = true;
        for (int j = 0; j < cards[i].size(); ++j) {
            if (freq[cards[i][j]] > 1) {
                canWin = false;
                break;
            }
        }
        cout << (canWin ? "YES" : "NO") << endl;
    }
    return 0;
}
