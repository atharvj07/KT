#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 998244353;

int calc(ll l, ll r) {
  int ans = 0;
  for (ll i = max(l, 1LL); i <= r; i *= 10) {
    if (i <= 9)
      ans++;
    else
      break;
  }
  return ans;
}

int main() {
  string a;
  ll l, r;
  cin >> a >> l >> r;
  int n = a.length();
  reverse(a.begin(), a.end());

  vector<vector<ll>> dp(n + 1, vector<ll>(r + 1));
  dp[n][0] = 1;
  for (int i = n - 1; i >= 0; i--) {
    for (int sm = 0; sm <= r; sm++) {
      dp[i][sm] = dp[i + 1][sm];
      int d = a[i] - '0';
      if (sm + d <= r)
        dp[i][sm] = (dp[i][sm] + dp[i + 1][sm + d]) % mod;
      if (i == 0 && sm >= l)
        dp[i][sm] = (dp[i][sm] + 1) % mod;
    }
  }
  for (int i = max(l, 1LL); i <= r; i *= 10) {
    if (i <= 9)
      dp[0][0]++;
    else
      break;
  }
  dp[0][0] %= mod;
  cout << dp[0][0] << '\n';
  return 0;
}
