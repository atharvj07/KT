#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100000;

int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; ++i) {
    cin >> a[i];
  }
  vector<int> pref(n + 1);
  for (int i = 1; i <= n; ++i) {
    pref[i] = pref[i - 1] + a[i];
  }
  int ans = 0x3f3f3f3f;
  for (int i = 1; i <= n; ++i) {
    for (int j = 1; j <= n; ++j) {
      if (i == j) continue;
      int dist = (i - j) * (i - j);
      int sum = pref[max(i, j)] - pref[min(i, j) - 1];
      int g = sum * sum;
      ans = min(ans, dist + g);
    }
  }
  cout << ans << endl;
  return 0;
}
