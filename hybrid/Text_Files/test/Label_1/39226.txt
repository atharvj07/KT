from typing import List, Any

n = int(input())
g = [int(i) for i in input().split()]
a = [1 for i in range(n)]
for i in range(n):
    for j in range(i):
        k = n * i + j
        for ik in range(n):
            if g[k] == g[n * ik + i]:
                a[i] = max(a[i], a[ik])
            if g[k] == g[n * j + ik]:
                a[j] = max(a[j], a[ik])
for i in a:
    print(i, end=" ")
