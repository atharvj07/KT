#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string t;
  cin >> t;
  assert(n % 2 == 0);
  int sum[2] = {0, 0};
  int q = 0;
  for (int i = 0; i < n; i++) {
    if (t[i] != '?') {
      sum[i < n / 2] += t[i] - '0';
    } else
      q++;
  }
  if (q == 0)
    cout << (sum[0] == sum[1] ? "Bicarp" : "Monocarp");
  else if (q == 2)
    cout << "Bicarp";
  else if (q > 2)
    cout << "Monocarp";
  else if (q == 1) {
    int l = -1, r = -1;
    for (int i = 0; i < n; i++) {
      if (t[i] == '?') {
        if (l == -1)
          l = i;
        else
          r = i;
      }
    }
    if (l > n / 2 || r < n / 2)
      cout << "Monocarp";
    else
      cout << "Bicarp";
  }
}
