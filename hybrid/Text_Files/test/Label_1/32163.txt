import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (sc.hasNextInt()) {
            int n = sc.nextInt();
            List<Integer> weights = new ArrayList<>();
            for (int i = 0; i < 10; i++) {
                if ((n & (1 << i)) != 0) {
                    weights.add(1 << i);
                }
            }
            Collections.sort(weights);
            for (int weight : weights) {
                System.out.print(weight + " ");
            }
            System.out.println();
        }

        sc.close();
    }
}
