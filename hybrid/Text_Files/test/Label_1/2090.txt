
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> deg(n);
    queue<int> q;
    for (int i = 0; i < n - 1; ++i) {
        int u, v; cin >> u >> v; --u; --v;
        ++deg[u]; ++deg[v];
        if (deg[u] == 3) q.push(u);
        if (deg[v] == 3) q.push(v);
    }
    int ans = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ans++;
        deg[u] = 0;
        for (int v = 0; v < n; ++v) {
            if (u != v && deg[v] > 1) {
                if (--deg[v] == 2) q.push(v);
            }
        }
    }
    cout << ans << endl;
}
