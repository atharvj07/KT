def is_s_palindrome(s):
    mirror_letters = {'b': 'd', 'd': 'b', 'p': 'q', 'q': 'p', 'o': 'o', 'x': 'x', 'w': 'w', 'v': 'v', 'A': 'A', 'H': 'H', 'I': 'I', 'M': 'M', 'O': 'O', 'T': 'T', 'U': 'U', 'V': 'V', 'W': 'W', 'X': 'X', 'Y': 'Y'}
    n = len(s)
    for i in range(n // 2 + n % 2):
        if s[i] in mirror_letters and s[n - 1 - i] == mirror_letters[s[i]]:
            continue
        else:
            return "NIE"
    return "TAK"

s = input().strip()
print(is_s_palindrome(s))
