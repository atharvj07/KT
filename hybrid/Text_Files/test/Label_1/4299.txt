#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1000000007;
ll power(ll a, ll b) {
    if (b == 0) return 1;
    if (b == 1) return a;
    ll ret = power(a, b / 2);
    if (b % 2 == 0) return (ret * ret) % mod;
    else return ((ret * ret) % mod * a) % mod;
}
int main() {
    ll x, y;
    cin >> x >> y;
    if (y % x != 0) { cout << 0 << endl; return 0; }
    ll ans = power(2, y / x - 1);
    cout << ans << endl;
    return 0;
}
