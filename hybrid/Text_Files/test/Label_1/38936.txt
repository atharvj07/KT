n = int(input().strip())
q = list(map(int, input().strip().split()))

p = [0] * n
p[0] = 1
for i in range(1, n):
    p[i] = p[i-1] + q[i-1]

min_p = min(p)

if min_p <= 0:
    shift = 1 - min_p
    p = [x + shift for x in p]
else:
    shift = 0

if len(set(p)) == n and max(p) == n:
    print(*p)
else:
    print(-1)
