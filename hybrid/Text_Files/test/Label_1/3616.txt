#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int MAXN = 1e5;
int n, a[MAXN];
int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    if (n == 2) {
        if (a[0] != a[1])
            cout << 1 << endl;
        else
            cout << -1 << endl;
        return 0;
    }
    int cnt = 0;
    for (int i = 1; i < n; i++) {
        if (a[i - 1] != a[i]) cnt++;
        if (cnt > 1) {
            cout << -1 << endl;
            return 0;
        }
    }
    if (cnt == 0) cout << -1 << endl;
    else
        cout << cnt - 1 << endl;
    return 0;
}
