#include <iostream>
#include <vector>
#include <unordered_map>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;

  unordered_map<char, int> cnt;
  for (char c : s) {
    cnt[c]++;
  }

  vector<char> possible;
  for (auto& p : cnt) {
    if (p.second % 2 == 1) {
      possible.push_back(p.first);
    }
  }

  if (possible.size() == 1) {
    cout << possible[0];
  } else if (possible.size() == 2) {
    cout << possible[0] << possible[1];
  } else {
    cout << "BGR";
  }
  return 0;
}
