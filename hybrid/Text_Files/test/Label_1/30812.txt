import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int l = sc.nextInt();
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            if (a > b) {
                int tmp = a;
                a = b;
                b = tmp;
            }
            int ans = (x[b - 1] - x[a - 1] - 1) / l + 1;
            System.out.println(ans);
        }
    }
}
