import java.util.*;

public class Main {

    // MAX_WEIGHT is the maximum weight that the bridge can withstand
    private static final int MAX_WEIGHT = 150;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            // Read the number of passersby
            int n = scanner.nextInt();

            // If n is 0, then we have reached the end of the input
            if (n == 0) {
                break;
            }

            // Initialize the time intervals and weights of the passersby
            int[] startTimes = new int[n];
            int[] endTimes = new int[n];
            int[] weights = new int[n];

            // Read the time intervals and weights of the passersby
            for (int i = 0; i < n; i++) {
                weights[i] = scanner.nextInt();
                startTimes[i] = scanner.nextInt();
                endTimes[i] = scanner.nextInt();
            }

            // Check if the bridge will break
            if (willBridgeBreak(startTimes, endTimes, weights)) {
                System.out.println("NG");
            } else {
                System.out.println("OK");
            }
        }

        scanner.close();
    }

    // Check if the bridge will break
    private static boolean willBridgeBreak(int[] startTimes, int[] endTimes, int[] weights) {
        // Initialize the current weight on the bridge
        int currentWeight = 0;

        // Sort the time intervals by start time
        int[] sortedStartTimes = startTimes.clone();
        Arrays.sort(sortedStartTimes);

        // Iterate over the sorted start times
        for (int i = 0; i < sortedStartTimes.length; i++) {
            // Get the index of the passerby with the current start time
            int index = Arrays.binarySearch(startTimes, sortedStartTimes[i]);

            // Add the weight of the passerby to the current weight on the bridge
            currentWeight += weights[index];

            // If the current weight on the bridge exceeds the maximum weight, then the bridge will break
            if (currentWeight > MAX_WEIGHT) {
                return true;
            }

            // Iterate over the end times of the passersby who are currently on the bridge
            for (int j = index + 1; j < endTimes.length; j++) {
                // If the end time of the passerby is less than or equal to the start time of the current passerby, then the passerby is no longer on the bridge
                if (endTimes[j] <= sortedStartTimes[i]) {
                    // Subtract the weight of the passerby from the current weight on the bridge
                    currentWeight -= weights[j];
                }
            }
        }

        // If the bridge has not broken, then it will not break
        return false;
    }
}
