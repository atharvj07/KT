#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5+5;
long long l[maxn], r[maxn], c[maxn], dp[maxn];
int main() {
    int n, x;
    cin >> n >> x;
    for (int i = 1; i <= n; ++i) cin >> l[i] >> r[i] >> c[i];
    sort(l+1,l+n+1);
    sort(r+1,r+n+1);
    memset(dp, -1, sizeof dp);
    dp[0] = 0;
    for (int i = 1; i <= n; ++i) {
        dp[i] = *upper_bound(dp, dp + i, 0ll + x - (r[i] - l[i] + 1)) + c[i];
    }
    cout << (dp[n] >= 0 ? dp[n] : -1) << endl;
    return 0;
}
