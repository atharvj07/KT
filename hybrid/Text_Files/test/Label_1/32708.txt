import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();

        // Check if Hikuhashi will never become Kaiden
        if (a <= b) {
            System.out.println(-1);
            return;
        }

        // Calculate the number of contests before Hikuhashi becomes Kaiden
        long contests = (k - a) / (a - b) + 1;
        // Check if the number of contests is odd
        if (contests % 2 == 0) {
            contests++;
        }

        System.out.println(contests);
    }
}
