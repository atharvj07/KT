#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pii;
const ll linf = 4e18;
int main() {
  ll n;
  cin >> n;
  vector<pii> res;
  vector<ll> a(n);
  for (ll i = 0; i < n; ++i) { cin >> a[i]; }
  ll mn = *min_element(a.begin(), a.end());
  ll mx = *max_element(a.begin(), a.end());
  if (mn == mx) {
    cout << 0 << '\n';
    return 0;
  }
  for (ll i = 0; i < n; ++i) {
    if (a[i] < mx) { res.emplace_back(1, i + 1); }
    if (a[i] > mn) { res.emplace_back(2, i + 1); }
    a[i] = mx;
  }
  cout << res.size() << '\n';
  for (auto &el : res) { cout << el.first << ' ' << el.second << '\n'; }
  return 0;
}
