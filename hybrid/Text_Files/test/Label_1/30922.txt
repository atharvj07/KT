import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt(), bx = scanner.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++)
            x[i] = scanner.nextInt();

        int m = scanner.nextInt(), by = scanner.nextInt();
        int[] y = new int[m];
        for (int i = 0; i < m; i++)
            y[i] = scanner.nextInt();

        int x_sum = 0;
        for (int i = 0; i < n; i++)
            x_sum += x[i] * Math.pow(bx, n - i - 1);

        int y_sum = 0;
        for (int i = 0; i < m; i++)
            y_sum += y[i] * Math.pow(by, m - i - 1);

        if (x_sum < y_sum)
            System.out.print('<');
        else if (x_sum > y_sum)
            System.out.print('>');
        else
            System.out.print('=');
    }
}
