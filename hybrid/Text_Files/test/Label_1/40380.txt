n = int(input())
arr = list(map(int, input().split()))

# Create a dictionary to hold the maximum length of subsequence ending with the number as the key
dp = {}
# Store the previous element in the longest subsequence that ends with the current element
prev = {}

# The length and last element of the longest subsequence
max_len = 0
last_element = None

for i, num in enumerate(arr):
    # If num-1 exists in dp, it means there is a subsequence ending with num-1
    if num - 1 in dp:
        dp[num] = dp[num - 1] + 1
        prev[num] = num - 1
    else:
        dp[num] = 1
        prev[num] = None
    
    # Update max_len and last_element if this subsequence is the longest so far
    if dp[num] > max_len:
        max_len = dp[num]
        last_element = num

# Reconstruct the longest subsequence
sequence = []
current = last_element
while current is not None:
    sequence.append(current)
    current = prev[current]

# Find the indices of the elements in the original array
indices = []
for num in reversed(sequence):
    indices.append(arr.index(num) + 1)
    arr[arr.index(num)] = None  # Mark as visited

print(max_len)
print(" ".join(map(str, indices)))
