#include <iostream>
#include <vector>
using namespace std;
int abs(int x){
	if(x<0) return -x;
	return x;
}
int main(){
	int a, b;
	cin >> a >> b;
	vector<int> v(7);

	for(auto &item : v){
		cin >> item;
	}
	int ans[3]={0};
	for(int i=0; i<6; i++){
		int d=abs(v[i+1]-a);
		int dd=abs(v[i+1]-b);
		if(d<dd) ans[0]++;
		else if(d==dd) ans[1]++;
		else ans[2]++;
	}
	for(int i=0; i<3; i++){
		cout << ans[i];
		if(i<2) cout << " ";
	}
	cout << "\n";
	return 0;
}
