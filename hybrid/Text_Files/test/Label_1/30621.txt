import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean[] win = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                win[a[i]] = !win[a[i]];
            }
        }
        if (win[0]) {
            System.out.println("cslnb");
        } else {
            System.out.println("sjfnb");
        }
    }
}
