import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int max = 0;
        for (int i = 0; i < n; i++) {
            int vol = 0;
            for (int j = 0; j < s.length(); j++) {
                if (Character.isUpperCase(s.charAt(j))) {
                    vol++;
                }
            }
            max = Math.max(max, vol);
        }
        System.out.println(max);
    }
}
