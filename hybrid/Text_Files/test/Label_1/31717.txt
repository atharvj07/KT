import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (sc.hasNextLine()) {
            String line = sc.nextLine();
            int r1 = 0, r2 = 0;
            char op = 0;
            for (char c : line.toCharArray()) {
                if (c >= '0' && c <= '9') {
                    r2 = r2 * 10 + c - '0';
                } else if (c == '+') {
                    r1 = op != 0 ? doOp(r1, r2, op) : r2;
                    r2 = 0; op = c;
                } else if (c == '-') {
                    r1 = op != 0 ? doOp(r1, r2, op) : r2;
                    r2 = 0; op = c;
                } else if (c == '*') {
                    r1 = op != 0 ? doOp(r1, r2, op) : r2;
                    r2 = 0; op = c;
                } else if (c == '=') {
                    r1 = op != 0 ? doOp(r1, r2, op) : r2;
                    if (r1 < 0 || r1 > 9999) System.out.println("E"); 
                    else System.out.println(r1);
                    op = 0; r1 = r2 = 0;
                }
            }
        }
        sc.close();
    }

    private static int doOp(int r1, int r2, char op) {
        if (op == '+') return r1 + r2;
        else if (op == '-') return r1 - r2;
        else return r1 * r2;
    }
}
