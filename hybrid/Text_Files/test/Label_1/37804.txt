n, k = map(int, input().split())
a = list(map(int, input().split()))
q = int(input())
for x in map(int, input().split()):
    cur = []
    for i in a[::-1]:
        while x >= i:
            x -= i
            cur.append(i)
            if len(cur) > k:
                cur = []
                break
    print(len(cur)) if cur else print(-1)
