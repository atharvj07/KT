#include <bits/stdc++.h>

using namespace std;

const int MX = 1e9;

int n, m, x, y;
long long dist[1005][1005];
bool vis[1005][1005];
pair<int, int> volcanoes[1000005];
long long dx[] = {1, 0};
long long dy[] = {0, 1};

int main() {
  cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    cin >> x >> y;
    volcanoes[i] = {x, y};
  }
  for (int i = 0; i <= n; i++) {
    for (int j = 0; j <= n; j++) { dist[i][j] = MX; }
  }
  dist[1][1] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (!vis[i][j]) {
        vis[i][j] = true;
        for (int k = 0; k < 2; k++) {
          int nx = i + dx[k], ny = j + dy[k];
          if (nx <= n && ny <= n) {
            bool fl = false;
            for (int a = 1; a <= m; a++) {
              if (volcanoes[a].first == nx && volcanoes[a].second == ny) fl = true;
            }
            if (!fl && dist[nx][ny] > dist[i][j] + 1) { dist[nx][ny] = dist[i][j] + 1; }
          }
        }
      }
    }
  }
  if (dist[n][n] == MX)
    cout << -1 << endl;
  else
    cout << dist[n][n] << endl;
  return 0;
}
