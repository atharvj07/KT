#include <iostream>
#include <vector>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<int> p(n + 1);
    for (int i = 1; i <= n; i++) {
      cin >> p[i];
    }
    bool clockwise = true, counterclockwise = true;
    for (int i = 1; i < n; i++) {
      clockwise &= p[i + 1] == p[i] + 1;
      counterclockwise &= p[i + 1] == p[i] - 1;
    }
    clockwise &= p[1] == p[n] + 1;
    counterclockwise &= p[1] == p[n] - 1;
    if (clockwise || counterclockwise)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
