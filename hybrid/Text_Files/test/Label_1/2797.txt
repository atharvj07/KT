#include <bits/stdc++.h>

#define ll long long

const int mod = 998244353;

int main() {
  ll n;
  std::cin >> n;
  std::vector<ll> a(n);
  for (ll &x : a) { std::cin >> x; }
  ll p = 0;
  for (ll i = 0; i < n; i++) { p += a[i] * (1LL << (n - 1 - i)); }
  std::cout << (p << 1) % mod << '\n';
  return 0;
}
