import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong(), b = in.nextLong();
        long ans = 1;
        for (long i = 2; i <= Math.min(a, 12); i++) {
            if (a % i == 0) {
                int cnt = 0;
                while (a % i == 0) {
                    a /= i;
                    cnt++;
                }
                ans *= Math.pow(i, Math.min(cnt, b));
            }
        }
        for (long i = 2; i <= Math.min(b, 12); i++) {
            if (b % i == 0) {
                int cnt = 0;
                while (b % i == 0) {
                    b /= i;
                    cnt++;
                }
                ans *= Math.pow(i, Math.min(cnt, a));
            }
        }
        System.out.println(ans);
    }
}
