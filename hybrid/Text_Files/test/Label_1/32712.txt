import java.util.*;

public class Main {
    public static void main(String[] args) {
        int n;
        int[] a;
        Integer[] b;
        Map<Integer, Set<Integer>> f;
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        b = new Integer[n];
        for (int i = 0; i < n; i++) {
            b[i] = a[i];
        }
        Arrays.sort(b);
        f = new HashMap<>();
        for (int i = 0; i < n; i++) {
            Set<Integer> g = new HashSet<>();
            for (int j = 2; j <= Math.sqrt(a[i]); j++) {
                if (a[i] % j == 0) {
                    while (a[i] % j == 0) {
                        a[i] /= j;
                    }
                    g.add(j);
                }
            }
            if (a[i] > 1) {
                g.add(a[i]);
            }
            f.put(b[i], g);
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (!Collections.disjoint(f.get(b[i]), f.get(b[j]))) {
                    System.out.println(1);
                    return;
                }
            }
        }
        System.out.println(0);
    }
}
