import java.util.*;

class Main {
    static final int MAX = 200000;
    static ArrayList<Integer>[] mapX = new ArrayList[MAX];
    static ArrayList<Integer>[] mapY = new ArrayList[MAX];

    static int[] x;
    static int[] y;
    static int n, k;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();

        x = new int[n + 1];
        y = new int[n + 1];

        for (int i = 0; i < MAX; i++) {
            mapX[i] = new ArrayList<>();
            mapY[i] = new ArrayList<>();
        }

        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
            mapX[x[i] + 100000].add(i);
            mapY[y[i] + 100000].add(i);
        }

        for (int i = 1; i <= k; i++) {
            int xx = sc.nextInt();
            int yy = sc.nextInt();
            mapX[xx + 100000].add(-i);
            mapY[yy + 100000].add(-i);
        }

        System.out.println(bfs());
    }

    static int bfs() {
        Queue<Integer> q = new LinkedList<>();
        boolean[] visit = new boolean[n + 1];
        q.add(0);

        int time = 0;
        while (!q.isEmpty()) {
            int size = q.size();
            for (int s = 0; s < size; s++) {
                int now = q.poll();
                if (now > 0 && !visit[now]) {
                    visit[now] = true;
                    q.add(x[now] + y[now] + 100000);
                } else if (now < 0) {
                    q.addAll(mapX[-now + 100000]);
                    q.addAll(mapY[-now + 100000]);
                }
            }
            time++;
        }

        return time - 1;
    }
}
