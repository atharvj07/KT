import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;

public class Main {
  public static void main(String[] args) throws IOException {
    BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

    int n = Integer.parseInt(reader.readLine());
    Map<String, List<Integer>> piles = new HashMap<>();
    piles.put("S", new LinkedList<>());
    piles.put("D", new LinkedList<>());
    piles.put("H", new LinkedList<>());
    piles.put("C", new LinkedList<>());

    String[] cards = reader.readLine().split(" ");
    for (int i = 0; i < n; i++) {
      String card = cards[i];
      piles.get(card.substring(1)).add(i + 1);
    }

    for (List<Integer> pile : piles.values()) {
      if (pile.isEmpty()) {
        System.out.println("NO");
        return;
      }
    }

    while (piles.size() > 1) {
      int maxPile = 0;
      String maxSuit = "";
      for (String suit : piles.keySet()) {
        int pile = piles.get(suit).get(piles.get(suit).size() - 1);
        if (pile > maxPile) {
          maxPile = pile;
          maxSuit = suit;
        }
      }

      int secondMaxPile = 0;
      String secondMaxSuit = "";
      for (String suit : piles.keySet()) {
        if (!suit.equals(maxSuit)) {
          int pile = piles.get(suit).get(piles.get(suit).size() - 1);
          if (pile > secondMaxPile) {
            secondMaxPile = pile;
            secondMaxSuit = suit;
          }
        }
      }

      if (maxPile - secondMaxPile > 2) {
        System.out.println("NO");
        return;
      }

      List<Integer> maxPileList = piles.get(maxSuit);
      List<Integer> secondMaxPileList = piles.get(secondMaxSuit);

      int topCard = maxPileList.remove(maxPileList.size() - 1);
      int secondTopCard = secondMaxPileList.remove(secondMaxPileList.size() - 1);
      if (topCard - secondTopCard != 1) {
        System.out.println("NO");
        return;
      }

      secondMaxPileList.add(topCard);

      if (maxPileList.isEmpty()) {
        piles.remove(maxSuit);
      }
    }

    System.out.println("YES");
  }
}
