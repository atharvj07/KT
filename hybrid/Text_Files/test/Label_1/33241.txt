import java.io.IOException;
import java.util.ArrayList;
import java.util.Scanner;

public class Main {
  static int n, k;
  static ArrayList<ArrayList<Integer>> adj;
  static int[] depth;
  static int[] vis;
  static int maxDepth;
  public static void main(String[] args) throws IOException {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    k = in.nextInt();
    depth = new int[n + 1];
    vis = new int[n + 1];
    adj = new ArrayList<>(n + 1);
    for (int i = 0; i <= n; i++) {
      adj.add(new ArrayList<>());
    }
    for (int i = 2; i <= n; i++) {
      int p = in.nextInt();
      adj.get(i).add(p);
      adj.get(p).add(i);
    }
    dfs1(1, 1, 0);
    dfs2(1, 1);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (depth[i] == maxDepth && vis[i] <= k) {
        ans++;
      }
    }
    System.out.println(ans);
  }

  static void dfs1(int cur, int p, int dep) {
    depth[cur] = dep;
    for (int u : adj.get(cur)) {
      if (u == p) {
        continue;
      }
      dfs1(u, cur, dep + 1);
    }
  }

  static void dfs2(int cur, int p) {
    for (int u : adj.get(cur)) {
      if (u == p) {
        continue;
      }
      if (depth[u] > maxDepth) {
        maxDepth = depth[u];
        dfs2(u, cur);
        vis[u]++;
        continue;
      }
      if (depth[u] == maxDepth) {
        vis[u]++;
        dfs2(u, cur);
      }
    }
  }
}
