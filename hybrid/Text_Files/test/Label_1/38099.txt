import sys

h, m = map(int, sys.stdin.readline().split())
H, D, C, N = map(int, sys.stdin.readline().split())

if H - D * (24 * 60 + m - h * 60) <= 0:
    print(0)
    sys.exit()

time = (H + D - 1) // D
if time < 12 * 60:
    cost = ((time - h * 60 - m + 1 + D - 1) // D + 1) * C
else:
    cost = ((time - h * 60 - m + 1 + D - 1) // D + 1) * C * 0.8
    
print("{:.3f}".format(cost))
