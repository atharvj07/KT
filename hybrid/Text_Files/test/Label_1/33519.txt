import java.util.Scanner;

public class RoadTrip {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = sc.nextInt();
            roads[i][1] = sc.nextInt();
        }

        // Check if any two roads connect the same pair of towns.
        for (int i = 0; i < 3; i++) {
            for (int j = i + 1; j < 3; j++) {
                if ((roads[i][0] == roads[j][0] && roads[i][1] == roads[j][1]) ||
                        (roads[i][0] == roads[j][1] && roads[i][1] == roads[j][0])) {
                    System.out.println("NO");
                    return;
                }
            }
        }

        // Check if any town is unreachable from any other town.
        boolean[] visited = new boolean[5];
        visited[1] = true;
        int count = 1;
        while (count < 4) {
            for (int i = 0; i < 3; i++) {
                if (visited[roads[i][0]] && !visited[roads[i][1]]) {
                    visited[roads[i][1]] = true;
                    count++;
                } else if (visited[roads[i][1]] && !visited[roads[i][0]]) {
                    visited[roads[i][0]] = true;
                    count++;
                }
            }
        }

        // If all towns are reachable and no two roads connect the same pair of towns, then it is possible to visit all the towns by traversing each of the roads exactly once.
        System.out.println(count == 4 ? "YES" : "NO");
    }
}
