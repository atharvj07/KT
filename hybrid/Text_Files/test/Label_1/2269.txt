#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 998244353;
int a[300001], b[300001], c[300001], d[300001];
int32_t main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        c[a[i]]++;
        d[b[i]]++;
    }
    int ans = 1;
    vector<int> mul;
    for (int i = 1; i <= n; i++) {
        mul.push_back(c[i] * d[i] % mod);
        ans = ans * mul.back() % mod;
    }
    cout << ans << endl;
    return 0;
}
