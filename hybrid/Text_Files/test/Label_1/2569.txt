#include <bits/stdc++.h>

using namespace std;

const int N = 5e5 + 5;
const long long INF = 1LL * 1e9 * 1e9;
typedef long long ll;

struct Line {
  ll k, m, p; // y = kx + m, p = (x * x + y * y)
};

ll cross(const Line& a, const Line& b) {
  return a.k * b.m - a.m * b.k;
}

struct Circle {
  ll x, y, r; // (x - X)^2 + (y - Y)^2 <= r^2
};

bool is_on_the_same_line(const Circle& a, const Circle& b) {
  return (a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y) == (a.r + b.r) * (a.r + b.r);
}

vector<Line> lines;

bool is_not_in_check(const vector<Circle>& points, const ll& x, const ll& y) {
  const ll d = x * x + y * y;
  for (const auto& point : points) {
    if (point.x == x && point.y == y) {
      return false;
    }
    if (is_on_the_same_line({x, y, 0}, point)) {
      return false;
    }
    const Line new_line = {2 * (point.x - x), 2 * (point.y - y), point.x * point.x + point.y * point.y + d - point.r * point.r};
    for (const auto& line : lines) {
      if (cross(new_line, line) == 0) {
        return false;
      }
    }
    lines.push_back(new_line);
  }
  return true;
}

int main() {
  ll n;
  ll x_king, y_king;
  cin >> n;
  cin >> x_king >> y_king;
  vector<Circle> black_pieces(n);
  for (auto& point : black_pieces) {
    char c;
    ll x, y;
    cin >> c >> x >> y;
    if (c == 'B') {
      point = {x, y, INF};
    } else if (c == 'R') {
      point = {x, INF, y};
    } else {
      point = {x, y, INF};
    }
  }
  cout << (is_not_in_check(black_pieces, x_king, y_king) ? "NO" : "YES");
  return 0;
}
