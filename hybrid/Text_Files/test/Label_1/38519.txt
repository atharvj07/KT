from sys import stdin
input = stdin.readline
n, s, m, k = map(int, input().split())
a = list(map(int, input().split()))
seg = [[] for _ in range(n+1)]
for _ in range(s):
    l, r = map(int, input().split())
    seg[l].append(r)
ans = []
cur = 0
for i in range(1, n+1):
    for j in seg[i]:
        if len(seg[j]) >= m-cur:
            ans = sorted([a[x-1] for x in range(i, j+1)])
if len(ans) >= k:
    print(ans[k-1])
else:
    print(-1)
