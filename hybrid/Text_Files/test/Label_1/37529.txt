n = int(input())
chemForces = {}
for _ in range(n):
    a, x = map(int, input().split())
    chemForces[a] = x

m = int(input())
topChemist = {}
for _ in range(m):
    b, y = map(int, input().split())
    topChemist[b] = y

total_income = 0
for element in chemForces:
    if element in topChemist:
        total_income += max(chemForces[element], topChemist[element])
    else:
        total_income += chemForces[element]

for element in topChemist:
    if element not in chemForces:
        total_income += topChemist[element]

print(total_income)
