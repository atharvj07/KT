import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long h = sc.nextLong(), m = sc.nextLong(), k = sc.nextLong();
        long h1 = sc.nextLong(), m1 = sc.nextLong();
        long h2 = sc.nextLong(), m2 = sc.nextLong();

        long ans = 0;
        long min = Math.max(m1, m2);
        long max = Math.min(m1 + m - 1, m2 + m - 1);
        if (min <= max) ans += Math.max(0, max - min + 1 - k);
        min = Math.max(h1, h2);
        max = Math.min(h1 + h - 1, h2 + h - 1);
        if (min <= max) ans += Math.max(0, max - min + 1 - k) * m;
        System.out.println(ans);
    }
}
