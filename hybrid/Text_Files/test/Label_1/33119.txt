import java.util.Scanner;

public class BerryPicking {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int maxBaskets = 0;
        for (int i = 0; i < n; i++) {
            int redBerries = scanner.nextInt();
            int blueBerries = scanner.nextInt();

            maxBaskets += Math.max(redBerries / k, blueBerries / k);
            maxBaskets += Math.max((redBerries + blueBerries) / k, 1); // Consider baskets with mixed berries
        }

        System.out.println(maxBaskets);
    }
}
