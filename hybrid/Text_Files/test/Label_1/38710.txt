n = int(input().strip())
cities = [tuple(map(int, input().split())) for _ in range(n)]

total_cost = 0
max_ai_plus_ci = cities[0][0] + cities[0][1]

for i in range(1, n):
    ai, ci = cities[i]
    total_cost += max(0, ai - max_ai_plus_ci)
    max_ai_plus_ci = max(max_ai_plus_ci, ai + ci)

total_cost += cities[0][1] + sum(ci for _, ci in cities[1:])

print(total_cost)
