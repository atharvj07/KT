#include <bits/stdc++.h>
using namespace std;
#define rep(i,n) for(int i=0;i<(int)(n);++i)
int N,M,C,H;
const int dx[4]={1,0,-1,0},dy[4]={0,1,0,-1};
struct XY{int x,y;} p[100005],q[100005];
bool inrange(int x,int y){ return 0<=x && x<N && 0<=y && y<M; }
int main() {
    cin >> N >> M >> C >> H;
    rep(i,C) cin >> p[i].x >> p[i].y;
    rep(i,H) cin >> q[i].x >> q[i].y;
    int ans=1<<30, a;
    rep(i,H) {
        int t=0;
        rep(j,C) t=max(t, abs(p[j].x-q[i].x)+abs(p[j].y-q[i].y) );
        if(t<ans) ans=t, a=i+1;
    }
    cout << ans << endl;
    cout << a << endl;
    return 0;
}
