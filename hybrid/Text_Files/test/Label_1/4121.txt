#include <bits/stdc++.h>
using std::cin, std::cout;
typedef std::pair<long long, long long> pll;
#define x first
#define y second
int main() {
    long long n, ans = 0, last;
    cin >> n;
    std::map<long long, std::set<long long>> sx, sy;
    for (long long i = 0, x, y; i < n; i++) {
        cin >> x >> y;
        sx[x].insert(y);
        sy[y].insert(x);
    }
    for (auto &[_, ys] : sx) {
        last = *ys.rbegin();
        ys.erase(ys.begin());
        ans += ys.size() > 0;
        for (auto &yt : ys) {
            sy[yt].erase(last);
        }
    }
    for (auto &[_, xs] : sy) {
        last = *xs.rbegin();
        xs.erase(xs.begin());
        ans += xs.size() > 0;
        for (auto &xt : xs) { sx[xt].erase(last); }
    }
    cout << ans << '\n';
}
