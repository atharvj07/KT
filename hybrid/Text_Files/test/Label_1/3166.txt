#include <bits/stdc++.h>
using namespace std;

int main() {
    string tiles[3];
    for (int i = 0; i < 3; i++) { cin >> tiles[i]; }
    sort(tiles, tiles + 3);
    int cnt = 0;
    if (tiles[0][1] == tiles[1][1] && tiles[1][1] == tiles[2][1])
        cnt = 2;
    else if (tiles[0][0] == tiles[1][0] - 1 && tiles[1][0] == tiles[2][0] - 1 &&
             tiles[0][1] == tiles[1][1] && tiles[1][1] == tiles[2][1])
        cnt = 1;
    else if (tiles[0][1] == tiles[1][1]) {
        if (tiles[1][0] == tiles[2][0] - 1)
            cnt = 1;
        else
            cnt = 2;
    } else if (tiles[0][0] == tiles[1][0] - 1 && tiles[1][0] == tiles[2][0] - 1) {
        if (tiles[0][1] == tiles[1][1] || tiles[1][1] == tiles[2][1])
            cnt = 1;
        else
            cnt = 2;
    } else
        cnt = 3;
    cout << cnt << endl;
    return 0;
}
