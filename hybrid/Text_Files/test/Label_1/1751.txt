#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, r, a[32], b[32];
  cin >> n >> m >> r;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= m; i++) cin >> b[i];
  sort(a + 1, a + 1 + n);
  sort(b + 1, b + 1 + m);
  int ans = r;
  int cnt = 0;
  for (int i = n; i >= 1; i--) {
    if (ans < a[i]) break;
    int x = min((ans - a[i]) / b[1], cnt);
    ans -= x * b[1];
    ans += x * a[i];
    cnt -= x;
  }
  for (int i = 1; i <= m; i++) {
    if (ans < b[i]) break;
    int x = min((ans - b[i]) / a[n], cnt);
    ans -= x * a[n];
    ans += x * b[i];
    cnt += x;
  }
  cout << ans << endl;
}
