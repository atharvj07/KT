n = int(input())
pairs = [list(map(int, input().split())) for _ in range(n-1)]

# Use a dictionary to keep track of the merges
groups = {}
for x, y in pairs:
    if x not in groups and y not in groups:
        groups[x] = [x, y]
        groups[y] = groups[x]
    elif x in groups and y not in groups:
        groups[x].append(y)
        groups[y] = groups[x]
    elif y in groups and x not in groups:
        groups[y].append(x)
        groups[x] = groups[y]
    else:
        if groups[x] == groups[y]:
            continue
        if len(groups[x]) < len(groups[y]):
            x, y = y, x
        groups[x].extend(groups[y])
        for k in groups[y]:
            groups[k] = groups[x]

# Find the group with all elements
for group in groups.values():
    if len(group) == n:
        print(*group)
        break
