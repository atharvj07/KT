import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();

        while (t-- > 0) {
            long x = input.nextLong();
            long y = input.nextLong();
            long n = input.nextLong();

            System.out.println(Math.min(x - 1, (n - y) / x) * x + y);
        }

        input.close();
    }
}
