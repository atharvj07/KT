n, m = map(int, input().split())
a = list(map(int, input().split()))
s = [[int(x) for x in input().split()] for _ in range(m)]
dp = [[[-1] * 3 for _ in range(m + 1)] for _ in range(n + 1)]
def dfs(l, r, pre):
    if l > r:
        return 0
    if dp[l][r][pre] != -1:
        return dp[l][r][pre]
    dp[l][r][pre] = dfs(l + 1, r, pre)
    for i in range(m):
        if s[i][0] >= l and s[i][1] <= r:
            if s[i][0] != s[i][1]:
                if pre == 0:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, 1) + dfs(s[i][1] + 1, r, 1) + a[s[i][0] - 1] * (r - s[i][0] + 1) + a[s[i][1]] * (s[i][1] - l + 1))
                elif pre == 1:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, 1) + dfs(s[i][1] + 1, r, 1) + a[s[i][0] - 1] * (r - s[i][0] + 1) - a[s[i][1]] * (s[i][1] - l + 1))
                elif pre == 2:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, 2) + dfs(s[i][1] + 1, r, 2) - a[s[i][0] - 1] * (r - s[i][0] + 1) - a[s[i][1]] * (s[i][1] - l + 1))
            else:
                if pre == 0:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, pre) + dfs(s[i][1] + 1, r, pre) + a[s[i][0] - 1] * (r - s[i][0] + 2))
                elif pre == 1:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, pre) + dfs(s[i][1] + 1, r, pre) + a[s[i][0] - 1] * (r - s[i][0] + 2))
                elif pre == 2:
                    dp[l][r][pre] = max(dp[l][r][pre], dfs(l, s[i][0] - 1, pre) + dfs(s[i][1] + 1, r, pre) - a[s[i][0] - 1] * (r - s[i][0] + 2))
    return dp[l][r][pre]
print(dfs(1, n, 0))
