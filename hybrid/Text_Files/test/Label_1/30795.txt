import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int m = sc.nextInt();
        int k = sc.nextInt();
        long[] p = new long[m];
        for (int i = 0; i < m; i++) {
            p[i] = sc.nextLong();
        }

        int pages = (int) Math.ceil((double) n / k);
        int ops = 0;
        int page = 0;
        for (int i = 0; i < m;) {
            if (p[i] <= (page + 1) * k) {
                ops++;
                page++;
                while (i < m && p[i] <= (page + 1) * k) {
                    i++;
                }
            } else {
                page++;
                if (page >= pages) {
                    break;
                }
            }
        }
        System.out.println(ops);
    }
}
