import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long a = scanner.nextLong();
        long b = scanner.nextLong();
        long gcd = 1;
        for (long i = 2; i <= Math.min(a, b); i++) {
            if (a % i == 0 && b % i == 0) {
                gcd = i;
                while (a % i == 0) {
                    a /= i;
                }
                while (b % i == 0) {
                    b /= i;
                }
            }
        }
        System.out.println(gcd);
    }
}
