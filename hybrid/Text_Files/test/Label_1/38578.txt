from collections import defaultdict
import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

pa = defaultdict(list)
for i in range(n):
    pa[a[i]].append(i)

gb = defaultdict(list)
for k in pa.keys():
    if len(pa[k]) > 1:
        x = min(pa[k])
        y = min(pa[k], key=lambda y: y != x)
        gb[x].append(y)
        gb[y].append(x)

fre = defaultdict(int)
for k in pa.keys():
    fre[k] = len(pa[k])

ans = 0
for k in fre.keys():
    if fre[k] > 2:
        ans += (fre[k] - 2)
    elif fre[k] == 2:
        for i in pa[k]:
            if len(gb[i]) >= 1:
                ans -= 1
                break

print(ans)
