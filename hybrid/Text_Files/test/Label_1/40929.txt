def is_possible(rebus):
    """
    Checks if a rebus has a solution.

    Args:
    rebus: The rebus string.

    Returns:
    True if the rebus has a solution, False otherwise.
    """
    # Split the rebus into tokens.
    tokens = rebus.split()

    # Check if the rebus has an even number of tokens.
    if len(tokens) % 2 == 0:
        return False

    # Check if the first and last tokens are not numbers.
    if not tokens[0].isdigit() or not tokens[-1].isdigit():
        return False

    # Check if the middle token is not an arithmetic operation.
    if not tokens[1] in "+-":
        return False

    # Check if the remaining tokens are numbers.
    for token in tokens[2:]:
        if not token.isdigit():
            return False

    return True



def solve_rebus(rebus):
    """
    Solves a rebus.

    Args:
    rebus: The rebus string.

    Returns:
    The solved rebus string, or None if the rebus has no solution.
    """
    # Check if the rebus is possible.
    if not is_possible(rebus):
        return None

    # Split the rebus into tokens.
    tokens = rebus.split()

    # Convert the first and last tokens to integers.
    n1 = int(tokens[0])
    n2 = int(tokens[-1])

    # Check if the middle token is a '+' operation.
    if tokens[1] == "+":
        # Try all possible values for the remaining tokens.
        for i in range(1, n2):
            for j in range(1, n2):
                if i + j - n1 == n2:
                    # Return the solved rebus string.
                    return " ".join(
                        map(str, [i, j, n1, tokens[1], n2])
                    )

    # Check if the middle token is a '-' operation.
    elif tokens[1] == "-":
        # Try all possible values for the remaining tokens.
        for i in range(1, n2):
            for j in range(1, n2):
                if j - i == n2:
                    # Return the solved rebus string.
                    return " ".join(
                        map(str, [i, j, n1, tokens[1], n2])
                    )

    # No solution found.
    return None



rebus = input()

# Check if the rebus has a solution.
if is_possible(rebus):
    # Solve the rebus.
    solution = solve_rebus(rebus)

    # Print the solution.
    if solution is not None:
        print("Possible")
        print(solution)
    else:
        print("Impossible")
else:
    # Print that the rebus has no solution.
    print("Impossible")
