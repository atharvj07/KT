#include<bits/stdc++.h>
using namespace std;
int main(){
    int n, k;
    cin>>n>>k;
    int ans = 0;
    while(n--){
        int x;
        cin>>x;
        int f=1;
        while(x){
            if(x%10>k){
                f=0;
                break;
            }
            x/=10;
        }
        if(f) ans++;
    }
    cout<<ans<<"\n";
    return 0;
}
