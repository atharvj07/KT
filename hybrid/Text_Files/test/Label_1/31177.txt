import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        int j = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] >= 2) {
                ans++;
                a[i] -= 2;
                j = i + 1;
            } else {
                break;
            }
        }
        for (int i = j; i < n; i++) {
            if (a[i] >= 1) {
                ans++;
                a[i] -= 1;
            } else {
                break;
            }
        }
        System.out.println(ans);
    }
}
