#include<bits/stdc++.h>
using namespace std;

int main(){
    int n,tot=0,now,ans=0,cnt=0;
    cin>>n;
    while(n--){
        cin>>now;tot+=now;
        int x=0;
        while(now--){
            int a,b;cin>>a>>b;
            if(a<=3){
                x++;
                tot--;
                if((x-cnt)%10==0)b*=2;
                ans+=b;
            }
        }
        cnt=x;
    }
    cout<<ans<<endl;
    return 0;
}
