#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int n, m, deg[N], vis[N];
vector<int> adj[N];
int main() {
    cin >> n >> m;
    while (m--) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y), adj[y].push_back(x);
        deg[x]++, deg[y]++;
    }
    queue<int> q;
    q.push(1);
    vis[1] = 1;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (auto &v : adj[u]) {
            if (!vis[v]) {
                q.push(v);
                vis[v] = 1;
                deg[u]--, deg[v]--;
                cout << u << ' ' << v << endl;
            }
            if (!deg[u] and !deg[v]) {
                break;
            }
        }
    }
    return 0;
}
