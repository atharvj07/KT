import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        input.nextLine();
        String str = input.nextLine();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = str.charAt(i) - '0';
        }
        for (int i = 1; i < n; i++) {
            int[] newArr = new int[n - i];
            for (int j = 0; j < n - i; j++) {
                newArr[j] = Math.abs(arr[j] - arr[j + 1]);
            }
            arr = newArr;
        }
        System.out.println(arr[0]);
    }
}
