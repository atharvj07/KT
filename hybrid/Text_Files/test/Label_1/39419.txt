import heapq

for _ in range(int(input())):
    n, r = map(int, input().split())
    x = sorted(map(int, input().split()))
    q = []
    ans = 0
    for i in range(n):
        if x[i] - r <= 0:
            ans += 1
            continue
        heapq.heappush(q, (x[i] - r - 1 // r, x[i] + r + 1 // r + 1))
    while q:
        l, r = heapq.heappop(q)
        if len(q) == 0 or l > r:
            ans += 1
        else:
            ans += 2
    print(ans)
