import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int l = sc.nextInt();
            if (l == 0) {
                break;
            }
            int[] m = new int[12];
            int[] n = new int[12];
            for (int i = 0; i < 12; i++) {
                m[i] = sc.nextInt();
                n[i] = sc.nextInt();
            }
            int s = 0;
            int c = 0;
            for (int i = 0; i < 12; i++) {
                s += m[i] - n[i];
                c++;
                if (s >= l) {
                    System.out.println(c);
                    break;
                }
            }
            if (s < l) {
                System.out.println("NA");
            }
        }
    }
}
