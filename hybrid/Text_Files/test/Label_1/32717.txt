import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.PriorityQueue;

public class Main {

	private static long MOD = (long) (Math.pow(10, 9) + 7);

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int n = Integer.parseInt(br.readLine());

		Map<Long, Integer> cnt = new HashMap<>();
		PriorityQueue<Long> pq = new PriorityQueue<>();
		long res = 0;
		while (n-- > 0) {
			String[] parts = br.readLine().split(" ");
			if (parts[0].equals("add")) {
				long x = Long.parseLong(parts[1]);
				cnt.merge(x, 1, Integer::sum);
				pq.add(x);
				res += cnt.get(x) - 1;
				res %= MOD;
			} else if (parts[0].equals("del")) {
				long x = Long.parseLong(parts[1]);
				if (cnt.get(x) > 1) {
					cnt.computeIfPresent(x, (k, v) -> v - 1);
				} else {
					cnt.remove(x);
				}
				pq.remove(x);
				res -= cnt.getOrDefault(x, 0) + 1;
				res %= MOD;
			} else {
				// sum
				while (pq.size() % 5 != 0) {
					pq.add(Long.MAX_VALUE);
				}

				int[] idx = new int[pq.size() / 5];
				long[][] medians = new long[pq.size() / 5][2];
				long[] arr = new long[5];
				int pos = 0;
				while (!pq.isEmpty()) {
					for (int i = 0; i < 5; i++) {
						arr[i] = pq.remove();
					}
					Arrays.sort(arr);
					medians[pos][0] = arr[2];
					medians[pos][1] = 0;
					for (int i = 1; i < 5; i++) {
						if (arr[i] != arr[i - 1]) {
							medians[pos][1]++;
						}
					}
					idx[pos] = pos;
					pos++;
				}

				long sum = 0;
				for (int i = idx.length - 1; i >= 0; i--) {
					long curr = medians[i][0];
					long sumFreq = medians[i][1];
					long freq = idx.length - i;
					sum += ((curr % MOD) * ((sumFreq % MOD) * (freq % MOD))) % MOD;
					sum %= MOD;
				}

				System.out.println(sum);
			}
		}
	}
}
