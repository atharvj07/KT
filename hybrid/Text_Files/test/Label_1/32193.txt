import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]), m = Integer.parseInt(line[1]);
        ArrayList<int[]>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int a = Integer.parseInt(line[0]), b = Integer.parseInt(line[1]), c = Integer.parseInt(line[2]);
            adj[a].add(new int[]{b, c});
            adj[b].add(new int[]{a, c});
        }
        int[] ans = solve(n, m, adj);
        if (ans == null) {
            System.out.println("Impossible");
        } else {
            System.out.println(ans[0]);
            for (int i = 1; i <= ans[0]; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        }
    }

    private static int[] solve(int n, int m, ArrayList<int[]>[] adj) {
        int[] vis = new int[n + 1];
        Queue<Integer> q = new LinkedList<>();
        int[] ans = new int[n + 1];
        int cnt = 0, day = 0;
        for (int i = 1; i <= n; i++) {
            if (vis[i] != 0) continue;
            q.add(i);
            cnt = 0;
            day++;
            while (!q.isEmpty()) {
                int cur = q.poll();
                if (vis[cur] == day) continue;
                vis[cur] = day;
                cnt++;
                for (int[] edge : adj[cur]) {
                    int v = edge[0], w = edge[1];
                    if (vis[v] == day) {
                        if (vis[v] != w) return null;
                    } else {
                        if (w == 0) q.add(v);
                    }
                }
            }
            ans[day] = cnt;
        }
        ans[0] = day;
        return ans;
    }
}
