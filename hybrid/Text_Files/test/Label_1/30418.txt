import java.util.*;

class Main {
    static final long INF = (long)1e18;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] xy = new long[n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2; j++) {
                xy[i][j] = sc.nextLong();
            }
        }
        PriorityQueue<Edge> pq = new PriorityQueue<>(Comparator.comparingLong(x -> x.cost));
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long cost = Math.min(Math.abs(xy[i][0] - xy[j][0]), Math.abs(xy[i][1] - xy[j][1]));
                pq.offer(new Edge(i, j, cost));
            }
        }
        long ans = 0;
        int[] uf = new int[n];
        for (int i = 0; i < n; i++) {
            uf[i] = i;
        }
        while (!pq.isEmpty()) {
            Edge edge = pq.poll();
            if (find(uf, edge.from) != find(uf, edge.to)) {
                ans += edge.cost;
                unite(uf, edge.from, edge.to);
            }
        }
        System.out.println(ans);
    }

    static int find(int[] uf, int x) {
        if (uf[x] == x) {
            return x;
        } else {
            return uf[x] = find(uf, uf[x]);
        }
    }

    static void unite(int[] uf, int x, int y) {
        x = find(uf, x);
        y = find(uf, y);
        uf[x] = y;
    }

    static class Edge {
        int from;
        int to;
        long cost;

        Edge(int from, int to, long cost) {
            this.from = from;
            this.to = to;
            this.cost = cost;
        }
    }
}
