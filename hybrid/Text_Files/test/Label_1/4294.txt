#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;

  unordered_map<string, int> a, b;
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    a[s]++;
  }
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    b[s]++;
  }

  int ans = 0;
  for (auto p : a) {
    int cnt = p.second - b[p.first];
    if (cnt > 0) ans += cnt;
  }
  cout << ans << endl;

  return 0;
}
