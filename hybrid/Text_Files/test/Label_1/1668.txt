#include <bits/stdc++.h>
using namespace std;
const int base=7;
int n,m,best;
int main(){
    cin>>n>>m;
    for(int i=1;i<n;++i)
        for(int j=1;j<m;++j){
            set<int> s;
            int tmp=i;
            while(tmp>0){
                if(s.count(tmp%base)) break;
                s.insert(tmp%base);
                tmp/=base;
            }
            tmp=j;
            while(tmp>0){
                if(s.count(tmp%base)) break;
                s.insert(tmp%base);
                tmp/=base;
            }
            if(s.size()==(int)s.max_size()) ++best;
        }
    cout<<best<<'\n';
    return 0;
}
