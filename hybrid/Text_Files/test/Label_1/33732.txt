import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        long[] pref = new long[n];
        pref[0] = a[0];
        for (int i = 1; i < n; i++) {
            pref[i] = pref[i - 1] ^ a[i];
        }
        long count = 0;
        for (int i = 2; i < n - 1; i++) {
            long sum = 0;
            for (int j = i - 1; j >= 1; j--) {
                sum += a[j];
                if (sum == (pref[i] ^ pref[j - 1])) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
