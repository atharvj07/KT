#include <bits/stdc++.h>
#define int long long
using namespace std ;
int32_t main()
{
    string s ;
    cin >> s ;
    int n = s.size();
    vector<int> pfx (n) ;
    pfx[0] = (s[0] == 'A' || s[0] == 'E' || s[0] == 'I' || s[0] == 'O' || s[0] == 'U' || s[0] == 'Y') ;
    for (int i = 1 ; i < n ; i ++)
    {
        pfx[i] = pfx[i - 1] + (s[i] == 'A' || s[i] == 'E' || s[i] == 'I' || s[i] == 'O' || s[i] == 'U' || s[i] == 'Y') ;
    }
    double ans = 0.0 ;
    for (int i = 0 ; i < n ; i ++)
    {
        ans += (1.0l * pfx[i]) / (n - i) ;
    }
    cout << fixed << setprecision(8) << ans << endl;
}
