#include <iostream>
#include <queue>
#include <vector>

using namespace std;

int main() {
  int k, n1, n2, n3, t1, t2, t3;
  cin >> k >> n1 >> n2 >> n3 >> t1 >> t2 >> t3;

  vector<int> washing(n1, 0);
  vector<int> drying(n2, 0);
  vector<int> folding(n3, 0);

  priority_queue<int, vector<int>, greater<int>> next_free_washing;
  priority_queue<int, vector<int>, greater<int>> next_free_drying;
  priority_queue<int, vector<int>, greater<int>> next_free_folding;

  for (int i = 0; i < n1; i++) next_free_washing.push(0);
  for (int i = 0; i < n2; i++) next_free_drying.push(0);
  for (int i = 0; i < n3; i++) next_free_folding.push(0);

  for (int i = 0; i < k; i++) {
    int start_washing = next_free_washing.top();
    next_free_washing.pop();
    washing[i] = start_washing + t1;
    next_free_washing.push(washing[i]);

    int start_drying = next_free_drying.top();
    next_free_drying.pop();
    drying[i] = max(start_drying, washing[i]) + t2;
    next_free_drying.push(drying[i]);

    int start_folding = next_free_folding.top();
    next_free_folding.pop();
    folding[i] = max(start_folding, drying[i]) + t3;
    next_free_folding.push(folding[i]);
  }

  cout << folding[k - 1] << endl;

  return 0;
}
