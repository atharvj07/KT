for t in range(int(input())):
    s=input()
    l=[]
    for i in range(26):
        a=[]
        for j in range(len(s)):
            if ord(s[j])-97==i:
                a.append(j)
        if len(a)>1:l.append(a)
    l.sort(key=lambda x:len(x),reverse=True)
    for a in l:
        for i in range(len(a)-1,0,-1):
            s=s[:a[i]]+s[a[i]+1:]
    print(s)
