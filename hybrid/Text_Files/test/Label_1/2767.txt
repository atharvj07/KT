#include<bits/stdc++.h>
using namespace std;
using ll = long long;
int main(){
    ll t; cin >> t;
    while(t--){
        ll x, y; cin >> x >> y;
        if((x==y)) cout << "YES\n";
        else if(x&1){
            if((y-x)&1) cout << "NO\n";
            else cout << "YES\n";
        }
        else{
            if(((x-y)&1)||(y%3==0)) cout << "YES\n";
            else cout << "NO\n";
        }
    }
}
