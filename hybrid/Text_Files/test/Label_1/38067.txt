import sys
input = sys.stdin.readline

n = int(input())
a, b = map(int, input().split())
h = list(map(int, input().split()))
if sum(h) > a + b:
    sys.stdout.write("-1")
else:
    h.sort()
    ans = 0
    i = 0
    while i < n and a > 0:
        ans += h[i]
        a -= h[i]
        i += 1
    while i < n and b > 0:
        ans += h[i]
        b -= h[i]
        i += 1
    sys.stdout.write(str(ans))
