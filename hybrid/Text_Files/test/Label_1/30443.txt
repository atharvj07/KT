import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long[] sum = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            int x = in.nextInt();
            sum[i] = (sum[i - 1] + x) % m;
        }
        boolean flag = false;
        for (int i = 1; i <= n; i++) {
            if (sum[i] == 0 || sum[i] == sum[i - 1]) {
                flag = true;
                break;
            }
        }
        if (flag) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
