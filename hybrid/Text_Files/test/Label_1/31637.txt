import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] p = new int[6];
        for(int i = 0; i < p.length; i++)
            p[i] = sc.nextInt();
        int[][] a = { { p[0] - p[2], p[1] - p[3] }, { p[2] - p[4], p[3] - p[5] } };
        int b = a[0][0] * a[1][1] - a[0][1] * a[1][0];
        System.out.println(b == 0 ? "Yes" : "No");
    }
}
