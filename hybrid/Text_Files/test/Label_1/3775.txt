#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> ra(n), ba(n);
  int ca = 0, cb = 0;
  for (int i = 0; i < n; i++) {
    cin >> ra[i];
    ca += ra[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> ba[i];
    cb += ba[i];
  }
  if (ca <= cb) {
    cout << -1 << endl;
    return 0;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (ra[i] == 0) ans = max(ans, ba[i]);
  }
  cout << ++ans << endl;
}
