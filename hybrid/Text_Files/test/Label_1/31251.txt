import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long x = sc.nextLong();
        long y = sc.nextLong();
        int p = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long sum = 0;
        for (int i = 0; i < n - 1; i++) {
            long diff = a[i + 1] - a[i];
            long newMushrooms = (diff * (x + y)) % p;
            sum = (sum + newMushrooms) % p;
        }
        System.out.println(sum);
    }
}
