import java.util.Scanner;

public class Main {
    static int n, m;
    static int[][] a;
    static int[] c;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n][m];
        c = new int[m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
                c[j] += a[i][j];
            }
        }
        int ans = n * m;
        for (int i = 0; i < m; i++) {
            ans = Math.min(ans, n * m - 2 * c[i]);
        }
        System.out.println(ans);
    }
}
