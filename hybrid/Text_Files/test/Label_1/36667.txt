n=int(input())
l=input()
r=[]
for i in range(10):
    r.append(0)
for i in l:
    if(i=='L'):
        for j in range(10):
            if(r[j]==0):
                r[j]=1
                break
    elif(i=='R'):
        for j in range(9,-1,-1):
            if(r[j]==0):
                r[j]=1
                break
    else:
        r[int(i)]=0
for i in r:
    print(i,end="")
print()
