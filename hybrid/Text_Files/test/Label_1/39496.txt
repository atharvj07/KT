n, k = map(int, input().split())
a = [input() for _ in range(n)]
ans = n + n
for x in range(1, n - k + 1):
    for y in range(1, n - k + 1):
        x2 = x + k - 1
        y2 = y + k - 1
        full = 1
        for i in range(x, x2 + 1):
            for j in range(y, y2 + 1):
                if a[i][j] == 'B':
                    full = 0
                    break
            if not full:
                break
        if full:
            _x = []
            _y = []
            for i in range(x - 1, x2):
                if a[i][y - 1] == 'B':
                    _x.append(i)
            for j in range(y - 1, y2):
                if a[x - 1][j] == 'B':
                    _y.append(j)
            ans = max(ans, len(_x) + 1, len(_y) + 1)
print(ans)
