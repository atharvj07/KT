#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int N = 3e5 + 10;

int n, a[N];

int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    vector<int> v;
    for (int i = 2; i <= n; i++) {
      if (a[i] > a[i - 1])
        v.push_back(i);
    }
    if (v.size() % 2) {
      cout << "NO\n";
    } else {
      cout << "YES\n";
    }
  }
  return 0;
}
