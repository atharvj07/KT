import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt(), b = sc.nextInt();
        int g = 1;
        for (int i = 1; i <= Math.min(a, b); i++) {
            g = gcd(g, i);
        }
        System.out.println(g);
    }

    public static int gcd(int a, int b) {
        return b == 0 ? a : gcd(b, a % b);
    }
}
