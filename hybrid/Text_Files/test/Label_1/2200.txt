#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }
  int sum = 0;
  for (int i = 0; i < n; i++) { sum += w[i]; }
  if (sum % 2 == 0) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
}
