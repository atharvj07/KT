#include <iostream>
int main() {
    int n, k = 0;
    std::cin >> n;
    bool b = false, a[n + 2];
    for (int i = 1; i <= n; i++) {
        char x;
        std::cin >> x;
        a[i] = x - '0';
        if (b && !a[i] && a[i - 1] && a[i + 1]) k++;
        if (a[i]) b = true;
    }
    std::cout << k << '\n';
}
