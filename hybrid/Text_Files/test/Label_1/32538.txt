import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Set<String> set = new HashSet<String>();
        for (int i = 0; i < n; i++) {
            String s = br.readLine();
            set.add(s);
        }
        String ans = getGoodString(set);
        System.out.println(ans);
    }

    static String getGoodString(Set<String> set) {
        int maxFreq = 0;
        for (String s : set) {
            maxFreq = Math.max(maxFreq, getFreq(s, set));
        }
        for (String s : set) {
            if (getFreq(s, set) == maxFreq) {
                return s;
            }
        }
        return "NO";
    }

    static int getFreq(String s, Set<String> set) {
        int freq = 0;
        for (String t : set) {
            int i = 0, j = 0;
            while (i < t.length() && j < s.length()) {
                if (t.charAt(i) == s.charAt(j)) {
                    j++;
                }
                i++;
            }
            if (j == s.length()) {
                freq++;
            }
        }
        return freq;
    }
}
