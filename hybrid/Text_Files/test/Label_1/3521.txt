#include <iostream>
#include <vector>

const int MOD = 1e9 + 7;
const int MAXN = 2e5 + 5;

int n;
char s[MAXN];

int dp[MAXN][3];

inline int add(int a, int b) {
    return ((a + b) % MOD + MOD) % MOD;
}

int solve() {
    dp[n][0] = 1;
    dp[n][1] = dp[n][2] = 0;

    for (int i = n - 1; i >= 0; --i) {
        if (s[i] != '?') {
            dp[i][s[i] - 'a'] = dp[i + 1][s[i] - 'a'];
        } else {
            dp[i][0] = add(dp[i][0], dp[i + 1][0]);
            dp[i][0] = add(dp[i][0], dp[i + 1][1]);
            dp[i][0] = add(dp[i][0], dp[i + 1][2]);

            dp[i][1] = add(dp[i][1], dp[i + 1][0]);
            dp[i][1] = add(dp[i][1], dp[i + 1][1]);
            dp[i][1] = add(dp[i][1], dp[i + 1][2]);

            dp[i][2] = add(dp[i][2], dp[i + 1][0]);
            dp[i][2] = add(dp[i][2], dp[i + 1][1]);
            dp[i][2] = add(dp[i][2], dp[i + 1][2]);
        }
    }

    return dp[0][0];
}

int main() {
    std::cin >> n;
    std::cin >> s;
    std::cout << solve() << "\n";
    return 0;
}
