#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int a[maxn], b[maxn], back[maxn], ans[maxn];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> b[i]; }
  for (int i = 1; i <= n; i++) { back[b[i]] = i; }
  int cur = 1;
  for (int i = 1; i <= n; i++) {
    if (a[i] == cur) {
      ans[cur] += i - 1 - back[a[i]];
      if (cur + 1 <= n) ans[cur + 1] += i - 1 - back[a[i] + 1];
      if (cur - 1 >= 1) ans[cur - 1] += i - 1 - back[a[i] - 1];
      cur++;
    }
  }
  for (int i = 1; i <= n; i++) { cout << ans[i] << " "; }
  cout << endl;
}
