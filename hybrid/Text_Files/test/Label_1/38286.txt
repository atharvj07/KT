def is_visible(x1, y1, x2, y2, x3, y3, x4, y4, x5, y5, x6, y6):
    # Calculate the maximum covered area in x and y axis by both black sheets
    cover_left = max(x3, x5)
    cover_right = min(x4, x6)
    cover_bottom = max(y3, y5)
    cover_top = min(y4, y6)

    # Check if any side of the white sheet is outside the covered area
    if x1 < cover_left or x2 > cover_right or y1 < cover_bottom or y2 > cover_top:
        return "YES"
    # Check if any corner of the white sheet is outside the area of both black sheets
    if (x1 < x3 or x1 > x4 or y1 < y3 or y1 > y4) and (x1 < x5 or x1 > x6 or y1 < y5 or y1 > y6):
        return "YES"
    if (x2 < x3 or x2 > x4 or y2 < y3 or y2 > y4) and (x2 < x5 or x2 > x6 or y2 < y5 or y2 > y6):
        return "YES"
    if (x1 < x3 or x1 > x4 or y2 < y3 or y2 > y4) and (x1 < x5 or x1 > x6 or y2 < y5 or y2 > y6):
        return "YES"
    if (x2 < x3 or x2 > x4 or y1 < y3 or y1 > y4) and (x2 < x5 or x2 > x6 or y1 < y5 or y1 > y6):
        return "YES"

    # If none of the above, check if the white sheet is fully inside any of the black sheets
    if (x1 >= x3 and x2 <= x4 and y1 >= y3 and y2 <= y4) or (x1 >= x5 and x2 <= x6 and y1 >= y5 and y2 <= y6):
        return "NO"
    return "YES"

x1, y1, x2, y2 = map(int, input().split())
x3, y3, x4, y4 = map(int, input().split())
x5, y5, x6, y6 = map(int, input().split())

print(is_visible(x1, y1, x2, y2, x3, y3, x4, y4, x5, y5, x6, y6))
