#include <bits/stdc++.h>
using namespace std;
long long a[105], s[105], ans[105];
int main() {
  long long n, k, p = 0, x = 1, id = 1;
  cin >> n >> k;
  for (long long i = 0; i < k; i++) cin >> a[i];
  while (x < n) {
    s[p] = id;
    for (long long i = 0; i * n + id < n; i++) {
      s[i * n + id] = id + (i + 1) * s[(i + 1) * n + id - 1] % n;
    }
    id = (id + a[p] % n) % n + 1;
    for (long long i = 0; i < n; i++) ans[p + i] = s[i];
    p += n;
    x++;
  }
  for (long long i = 0; i < k; i++) { cout << ans[i] << " "; }
  cout << endl;
  return 0;
}
