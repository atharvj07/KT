#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  std::vector<int> v(5);
  for (auto &e : v) std::cin >> e;
  std::sort(v.begin(), v.end());
  if(v[0]==v[1]&&v[2]==v[3]&&v[3]==v[4]){
    std::cout << v[0]+v[0]+v[1]+v[1]+v[2] << '\n';
  }else if(v[0]==v[1]&&v[2]==v[3]){
    std::cout << v[0]+v[1]+v[1]+v[3]+v[4] << '\n';
  }else if(v[0]==v[1]){
    std::cout << v[0]+v[1]+v[2]+v[3]+v[3] << '\n';
  }else if(v[1]==v[2]){
    std::cout << v[0]+v[1]+v[2]+v[2]+v[4] << '\n';
  }else if(v[2]==v[3]){
    std::cout << v[0]+v[1]+v[2]+v[3]+v[3] << '\n';
  }else if(v[3]==v[4]){
    std::cout << v[0]+v[1]+v[2]+v[3]+v[4] << '\n';
  }else{
    std::cout << v[0]+v[1]+v[1]+v[2]+v[2] << '\n';
  }
}
