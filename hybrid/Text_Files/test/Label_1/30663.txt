import java.util.Scanner;

public class Main {

   public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);

      int n = sc.nextInt();
      long[] arr = new long[n];
      for (int i = 0; i < n; i++) arr[i] = sc.nextLong();

      long sum = 0;
      for (int i = 0; i < n; i++) sum += arr[i];
      sc.close();
      int ans = 0;
      long A = 0;

      for (int i = 0; i < n; i++) {
         if (arr[i] == arr[0]) continue;
         sum -= arr[i];
         ans++;
         if (arr[i] > arr[0]) A = arr[i - 1];
         if (arr[i] < arr[0]) A = arr[i];
      }
      if (A == 0) A = arr[n - 1];
      System.out.println(sum + A + " " + ans);
      for (int i = 0; i < n; i++) {
         if (arr[i] == arr[0] || arr[i] == A) continue;
         System.out.print(i + 1 + " ");
      }
   }
}
