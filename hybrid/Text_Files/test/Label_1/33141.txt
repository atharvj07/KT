import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] nums = new int[n];
    for (int i = 0; i < n; i++) {
      nums[i] = in.nextInt();
    }
    int[] dp = new int[n];
    int[] path = new int[n];
    for (int i = 0; i < n; i++) {
      dp[i] = 1;
      path[i] = -1;
      for (int j = 0; j < i; j++) {
        if (nums[j] < nums[i] && dp[j] + 1 > dp[i]) {
          dp[i] = dp[j] + 1;
          path[i] = j;
        }
      }
    }
    int max = 0;
    int maxIdx = -1;
    for (int i = 0; i < n; i++) {
      if (dp[i] > max) {
        max = dp[i];
        maxIdx = i;
      }
    }
    System.out.println(max);
    List<Integer> res = new ArrayList<>();
    while (maxIdx != -1) {
      res.add(nums[maxIdx]);
      maxIdx = path[maxIdx];
    }
    Collections.reverse(res);
    for (int num : res) {
      System.out.print(num + " ");
    }
  }
}
