import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }
            int[][] files = new int[n][2];
            for (int i = 0; i < n; i++) {
                files[i][0] = sc.nextInt();
                files[i][1] = sc.nextInt();
            }
            Arrays.sort(files, Comparator.comparingInt(a -> a[1]));
            int ans = 0;
            int used = 0;
            for (int i = 0; i < n; i++) {
                if (used + files[i][0] <= m) {
                    used += files[i][0];
                } else if (used + files[i][1] <= m) {
                    used = files[i][1];
                    ans++;
                } else {
                    ans = -1;
                    break;
                }
            }
            System.out.println(ans == -1 ? "Impossible" : ans);
        }
    }
}
