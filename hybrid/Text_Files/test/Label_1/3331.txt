#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, k, x, sum = 0;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sum += abs(a[i]);
    }
    if (k & 1) {
        if (a[0] > 0) {
            sum -= 2 * a[0];
        } else {
            for (int i = n - 1; i >= 0; i--) {
                if (a[i] < 0) {
                    sum += 2 * a[i];
                    break;
                }
            }
        }
    }
    if (k >= 2) {
        for (int i = 0; i < n; i++) {
            if (a[i] > 0) {
                sum -= 2 * a[i];
                k -= 2;
            } else {
                break;
            }
        }
    }
    cout << sum << endl;
    return 0;
}
