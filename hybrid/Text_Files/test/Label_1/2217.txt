#include <iostream>
#include <string>

using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  int grid[r][c];
  for (int i = 0; i < r; i++) {
    string s;
    cin >> s;
    for (int j = 0; j < c; j++) { grid[i][j] = (s[j] == '.'); }
  }
  int ans = 0;
  for (int i = 0; i < r; i++) {
    bool empty = true;
    for (int j = 0; j < c; j++) { empty &= grid[i][j]; }
    ans += empty;
  }
  for (int j = 0; j < c; j++) {
    bool empty = true;
    for (int i = 0; i < r; i++) { empty &= grid[i][j]; }
    ans += empty;
  }
  cout << ans << endl;
  return 0;
}
