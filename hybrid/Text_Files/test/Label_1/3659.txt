#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define ll long long
int main() {
  string a[9];
  int n = 8;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  bool winA = 0, winB = 0;
  for (int i = 0; i < n; i++) {
    if (a[0][i] == 'W') winA = 1;
    if (a[n - 1][i] == 'B') winB = 1;
  }
  if (winA)
    cout << 'A' << endl;
  else if (winB)
    cout << 'B' << endl;
  else if (winA == false && winB == false) {
    int cntA = 0, cntB = 0;
    for (int j = 0; j < n; j++) {
      for (int i = 0; i < n; i++) {
        if (a[i][j] == 'B') cntB++;
        if (a[i][j] == 'W') cntA++;
      }
    }
    (cntA > cntB) ? cout << 'A' << endl : cout << 'B' << endl;
  }
}
