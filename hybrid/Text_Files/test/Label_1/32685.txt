import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        String s = sc.next();
        int[] cnt = new int[26];
        for(char c : s.toCharArray())
            cnt[c-'a']++;
        int n = s.length();
        if(n % k != 0) {
            System.out.println("-1");
            return;
        }
        StringBuilder ans = new StringBuilder();
        int x = n / k;
        for(int i = 0; i < 26; i++) {
            if(cnt[i] > 0 && cnt[i] % x != 0) {
                System.out.println("-1");
                return;
            }
            while(cnt[i] > 0) {
                for(int j = 0; j < x; j++)
                    ans.append((char)(i+'a'));
                cnt[i] -= x;
            }
        }
        System.out.println(ans.toString());
    }
}
