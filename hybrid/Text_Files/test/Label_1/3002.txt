#include <iostream>
using namespace std;

const int MAXN = 100005;
int n, k, ans, a[MAXN];
bool check[MAXN];

int main() {
  cin >> n >> k;
  for (int i = 1; i <= k; i++) {
    cin >> a[i];
    check[a[i]] = true;
  }
  for (int i = 1; i <= n; i++) ans += (check[i] == 0 && check[i + 1] == 0);
  cout << ans << endl;
  return 0;
}
