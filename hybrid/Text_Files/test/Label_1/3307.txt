#include<bits/stdc++.h>
using namespace std;
typedef long long llong;

int main(){
    string s;
    cin >> s;
    int n = s.size();
    vector<llong>dp(n,0);
    dp[n-1] = (s[n-1]-'0')%4;
    for(int i=n-2;i>=0;i--){
        dp[i] = ((s[i]-'0')%4) + (10*dp[i+1])%4;
        dp[i]%=4;
    }
    llong sum = 0;
    for(int i=0;i<n;i++){
        sum += dp[i];
    }
    cout << sum << endl;
    return 0;
}
