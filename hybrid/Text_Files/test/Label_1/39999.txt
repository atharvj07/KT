import sys
input = sys.stdin.readline

def gcd(x: int, y: int) -> int:
    if x < y:
        x, y = y, x
    if y == 0:
        return x
    return gcd(y, x % y)

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    mn = min(a)
    ans = 'YES'
    for i in range(n-1):
        if a[i] > a[i+1] and gcd(a[i], a[i+1]) != mn:
            ans = 'NO'
    print(ans)
