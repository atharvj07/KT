#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int qmark = 0, a = 0, b = 0, c = 0, ans = 0;
    for (auto t : s) {
        if (t == '?')
            qmark++;
        if (t == 'a') {
            a = (a + c) % mod;
            c = 0;
        }
        if (t == 'b') {
            b = (b + a) % mod;
            a = 0;
        }
        if (t == 'c') c = (c + b) % mod;
    }
    ans = 1LL * qmark * (qmark - 1) % mod * (qmark - 2) % mod + 1LL * qmark * a % mod + 1LL * b * (qmark) % mod + 1LL * c % mod;
    ans = ans % mod;
    cout << ans;
    return 0;
}
