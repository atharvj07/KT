import sys
input = sys.stdin.readline

n, m = map(int, input().split())
roads = [set() for _ in range(n+1)]

for _ in range(m):
    a, b = map(int, input().split())
    roads[a].add(b)
    roads[b].add(a)

vis = [False] * (n+1)
cnt = 0

for i in range(1, n+1):
    if vis[i]: continue
    
    cnt += 1
    vis[i] = True
    
    q = [i]
    while q:
        nxt = q.pop()
        for j in roads[nxt]:
            if not vis[j]:
                vis[j] = True
                q.append(j)

print(cnt-1)
