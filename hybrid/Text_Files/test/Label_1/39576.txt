import sys
input = sys.stdin.readline

n = int(input())
s = [[int(i) for i in input().split()] for j in range(n)]
s.sort(key = lambda x: x[0])

ans = [-1, -1]
for i in range(n):
    for j in range(i+1, n):
        if s[i][0] >= s[j][0] and s[i][1] <= s[j][1]:
            ans = [i+1, j+1]
            break
print(*ans)
