#include<bits/stdc++.h>
using namespace std;
#define mod 998244353
#define ll long long
const int maxn=2e5+10;
int32_t main(){
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n,k;cin>>n>>k;
    vector<int>a(n+1),suff(n+1),pref(n+1),pw(n+1);
    pw[0]=1;
    for(int i=1;i<=n;i++)pw[i]=2*pw[i-1]%mod;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        if(a[i]==-1)a[i]=k+1;
        suff[i]=suff[i-1]+(a[i]==a[i-1]);
        pref[i]=pref[i-1]+(a[i]==a[i+1]);
    }
    int ans=0;
    for(int i=1;i<=n;i++){
        if(a[i]==a[i-1]&&a[i]==a[i+1]){
            int len=2*suff[i-1]-1;
            ans+=pw[len]-1;ans%=mod;
        }
        if(a[i]!=a[i-1]||a[i]!=a[i+1]){
            int len1=pref[i]-1,len2=suff[i]-1;
            ans+=pw[len1]+pw[len2]-1;
            ans%=mod;
        }
    }
    cout<<ans<<endl;
}
