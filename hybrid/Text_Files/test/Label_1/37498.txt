import sys
input = sys.stdin.readline

n =int(input())
ans = 0
for i in range(n):
    cards = []
    k = int(input())
    for j in range(k):
        c,d = map(int,input().split())
        cards.append([c,d])
    cards.sort()
    cur = 0
    cnt = 0
    for c,d in cards:
        if cur+c <= 3:
            cur+=c
            if cnt%10==9: d*=2
            ans+=d
            cnt+=1
    cur = 0
    cnt = 0
    for c,d in cards[::-1]:
        if cur+c <= 3:
            cur+=c
            if cnt%10==9: d*=2
            ans+=d
            cnt+=1
print(ans)
