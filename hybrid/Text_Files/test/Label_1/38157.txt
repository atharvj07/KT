def process_orders():
    n, m, k = map(int, input().split())
    items = list(map(int, input().split()))
    total_time = 0
    for _ in range(n):
        order = list(map(int, input().split()))
        for item in order:
            pos = items.index(item) + 1
            total_time += pos
            items.remove(item)
            items.insert(0, item)
    return total_time

print(process_orders())
