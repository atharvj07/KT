n, a, b = map(int, input().split())
min_area = 6 * n

if a * b >= min_area:
    print(f"{a * b}\n{a} {b}")
else:
    enlarged = False
    for new_a in range(a, min_area + 1):
        new_b = min_area // new_a
        if new_a * new_b >= min_area and new_b >= b:
            print(f"{new_a * new_b}\n{new_a} {new_b}")
            enlarged = True
            break
        if (min_area % new_a) and new_a * (new_b + 1) >= min_area and (new_b + 1) >= b:
            print(f"{new_a * (new_b + 1)}\n{new_a} {new_b + 1}")
            enlarged = True
            break
    if not enlarged:
        new_b = min_area // a
        if min_area % a:
            new_b += 1
        print(f"{a * new_b}\n{a} {new_b}")
