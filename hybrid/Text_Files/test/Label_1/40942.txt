import sys

input = sys.stdin.readline
n, k = map(int, input().split())

segs = []
for _ in range(n):
  l, r = map(int, input().split())
  segs.append([l, r])

segs.sort()

ans = 0
for i in range(n):
  l1, r1 = segs[i]
  for j in range(i + 1, n):
    l2, r2 = segs[j]
    if l2 > r1:
      break
    ans += max(0, min(r1, r2) - max(l1, l2) + 1)

print(ans % 1000000007)
