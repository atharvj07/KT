#include <bits/stdc++.h>

using namespace std;
using ll = long long ;
const int LIM = 2e5 + 5;
#define psb push_back

void solve() {
    int n, k, l;
    cin >> n >> k >> l;
    vector<ll> vp(n), v;
    for (int i = 0; i < n; i++) cin >> vp[i];
    ll p[2 * k];
    for (int i = 0; i < k; i++) p[i] = i, p[k + i] = k - 1 - i;
    for (int t = 1; t < 2 * k; t++) {
        v.clear();
        for (int j = 0; j < n; j++) v.psb(p[t] + vp[j]);
        if (*max_element(v.begin(), v.end()) > l) {
            cout << "No\n";
            return;
        }
    }
    cout << "Yes\n";
}
int main() {
    int t;
    cin >> t;
    while (t--) solve();
}
