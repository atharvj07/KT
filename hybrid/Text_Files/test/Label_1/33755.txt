import java.util.*;
import java.lang.*;

class Main {
	static class Car implements Comparable<Car> {
		int price, capacity;

		Car(int price, int capacity) {
			this.price = price;
			this.capacity = capacity;
		}

		@Override
		public int compareTo(Car o) {
			return this.price - o.price;
		}
	}

	public static void main(String[] args) throws Exception {
		Scanner scanner = new Scanner(System.in);
		int n = scanner.nextInt(), k = scanner.nextInt(), s = scanner.nextInt(), t = scanner.nextInt();
		Car[] cars = new Car[n];
		for (int i = 0; i < n; i++) {
			cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
		}
		int[] gasStations = new int[k];
		for (int i = 0; i < k; i++) {
			gasStations[i] = scanner.nextInt();
		}

		Arrays.sort(cars);

		int minPrice = -1;
		for (Car car : cars) {
			if (canReachCinema(car, gasStations, s, t)) {
				minPrice = car.price;
				break;
			}
		}

		System.out.println(minPrice);
	}

	static boolean canReachCinema(Car car, int[] gasStations, int s, int t) {
		int time = 0, fuel = car.capacity;
		int gasStationIndex = 0;
		for (int i = 0; i < s;) {
			if (gasStationIndex < gasStations.length && i == gasStations[gasStationIndex]) {
				fuel = car.capacity;
				gasStationIndex++;
			}

			int distanceToNextGasStation = gasStationIndex < gasStations.length ? gasStations[gasStationIndex] - i : s - i;
			int timeToNextGasStation = distanceToNextGasStation;

			if (fuel < distanceToNextGasStation) {
				return false;
			}

			int timeInNormalMode = (int) Math.ceil((double) distanceToNextGasStation / 2);
			int timeInAcceleratedMode = distanceToNextGasStation;

			if (timeInNormalMode + time <= t) {
				time += timeInNormalMode;
				fuel -= distanceToNextGasStation;
				i += distanceToNextGasStation;
			} else if (timeInAcceleratedMode + time <= t) {
				time += timeInAcceleratedMode;
				fuel -= distanceToNextGasStation * 2;
				i += distanceToNextGasStation;
			} else {
				return false;
			}
		}

		return true;
	}
}
