import sys

n = int(sys.stdin.readline())
a, b = map(int, sys.stdin.readline().split())
h = list(map(int, sys.stdin.readline().split()))

dp = [[0] * 3 for _ in range(n + 1)]
dp[0][0] = dp[0][1] = dp[0][2] = -1

for i in range(1, n + 1):
    if dp[i - 1][1] != -1:
        if a >= h[i - 1]:
            dp[i][0] = dp[i - 1][1] + 2
        if b >= h[i - 1]:
            dp[i][0] = min(dp[i][0] or sys.maxsize, dp[i - 1][1] + 1)
        dp[i][1] = dp[i - 1][1] + 1
    if dp[i - 1][2] != -1:
        if a >= h[i - 1]:
            dp[i][2] = dp[i - 1][2] + 2
        if b >= h[i - 1]:
            dp[i][2] = min(dp[i][2] or sys.maxsize, dp[i - 1][2] + 1)
        dp[i][2] = min(dp[i][2], dp[i - 1][1] + 1)

ans = min(dp[n])

if ans == -1:
    print(-1)
else:
    print(ans - 1)
