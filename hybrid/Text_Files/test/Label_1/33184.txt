import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++)
            m[i] = scanner.nextInt();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (i > m[i])
                sum += i - m[i];
        }
        System.out.println(sum);
    }
}
