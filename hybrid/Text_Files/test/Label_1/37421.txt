n=int(input())
a=[*map(int,input().split())]
if max(a)!=2:
    print(0)
else:
    k=0
    res=[]
    for i in range(n):
        if a[i]==2 or i==n-1:
            if i>0:
                res.append((k,i+1))
            k=0
        else:
            k+=1
    if len(res)==0 or res[0][0]!=1:
        print(0)
    else:
        res.sort()
        res.reverse()
        print(len(res))
        for x,y in res:
            print(x,y)
