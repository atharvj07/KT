import java.util.Scanner;

class Main {
    static final int MAX_N = 16;
    static final int MAX_M = 9;
    static final double EPS = 1e-8;

    static int n;
    static char[] v = new char[MAX_M];
    static double[][] p = new double[MAX_N][MAX_M];
    static double[][] d = new double[MAX_N][MAX_M];

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            n = input.nextInt();
            if (n == 0) {
                break;
            }
            v[0] = input.next().charAt(0);
            v[1] = input.next().charAt(0);
            v[2] = input.next().charAt(0);
            solve();
        }
    }

    static void solve() {
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j < 5; j++) {
                p[i][j] = 0.0;
            }
        }
        p[n][getIdx(v[0])] = 1.0;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 5; j++) {
                if (p[i][j] > 0.0) {
                    int[] dx = {1, -1, 0, 0};
                    int[] dy = {0, 0, 1, -1};
                    for (int k = 0; k < 4; k++) {
                        int nx = j + dx[k];
                        int ny = v[nx] - 'A';
                        if (nx >= 0 && nx < 5 && ny < 5) {
                            p[i][nx] += p[i][j] / 4.0;
                        }
                    }
                }
            }
        }
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j < 5; j++) {
                d[i][j] = 0.0;
            }
        }
        d[n][getIdx(v[1])] = 1.0;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 5; j++) {
                if (d[i][j] > 0.0) {
                    int[] dx = {1, -1, 0, 0};
                    int[] dy = {0, 0, 1, -1};
                    for (int k = 0; k < 4; k++) {
                        int nx = j + dx[k];
                        int ny = v[nx] - 'A';
                        if (nx >= 0 && nx < 5 && ny < 5) {
                            d[i][nx] += d[i][j] / 4.0;
                        }
                    }
                }
            }
        }
        double ans = 0.0;
        for (int j = 0; j < 5; j++) {
            if (j != getIdx(v[2])) {
                ans += p[0][j] * d[0][j];
            }
        }
        if (Math.abs(ans) < EPS) {
            ans = 0.0;
        }
        System.out.printf("%.8f\n", ans);
    }

    static int getIdx(char s) {
        return s - 'A';
    }
}
