import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n + 1];
        for (int i = 0; i < n; i++) {
            b[a[i]]++;
        }
        int c = 0;
        for (int i = 0; i <= n; i++) {
            c += b[i] & 1;
        }
        if (c != 0 && c != 2) {
            System.out.println(-1);
        } else {
            for (int i = n; i >= 0; i--) {
                if (b[i] != 0) {
                    System.out.print(i + " ");
                }
            }
        }
    }
}
