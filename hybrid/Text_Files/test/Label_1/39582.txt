from math import ceil
for t in range(int(input())):
    n,m,x=map(int,input().split())
    h=list(map(int,input().split()))
    h.sort()
    if x<h[0]*(m-1):print('NO')
    else:
        print('YES');res=[];j=0
        for i in range(m):
            s=0
            while j<n and s+h[j]<=x*(i+1):s+=h[j];j+=1
            for k in range(j-1):res+=[i+1]
        res+=[m]
        print(*res)
