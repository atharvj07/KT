import sys

ri = lambda: int(sys.stdin.readline())
rl = lambda: list(map(int, sys.stdin.readline().split()))

for i in range(ri()):
    n = ri()
    a = rl()
    h = 1
    for j in range(n):
        if j > 0 and a[j] and a[j - 1]:
            h += 5
        elif a[j]:
            h += 1
        else:
            h -= 2
            if h <= 0:
                break
    sys.stdout.write(str(h if h > 0 else -1) + "\n")
