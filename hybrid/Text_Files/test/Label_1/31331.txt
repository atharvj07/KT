import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nk = br.readLine().split(" ");
        int n = Integer.parseInt(nk[0]);
        int k = Integer.parseInt(nk[1]);
        int[][] dp = new int[n][n / 2 + 10];
        dp[0][n / 2 + k] = 1;
        for (int i = 0; i < n - 1; ++i)
            for (int j = 0; j <= n + k; j++)
                if (j <= k && (n / 2 + k) != (n / 2 + j))
                    dp[i + 1][(n / 2 + k - j) % (n + k)] = (int) ((dp[i + 1][(n / 2 + k - j) % (n + k)] + (long) dp[i][j] * (j + 1) % 998244353) % 998244353);
                else if (j <= n && j > k)
                    dp[i + 1][n + k - j] = (int) ((dp[i + 1][n + k - j] + (long) dp[i][j] * (j - k) % 998244353) % 998244353);
                else if (j > n)
                    dp[i + 1][n + k - j] = (int) ((dp[i + 1][n + k - j] + (long) dp[i][j] * (n - j) % 998244353) % 998244353);
        System.out.println(dp[n - 1][0]);
    }
}
