import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        if (n == 1) {
            System.out.println(1);
        } else if (m == 1 || m == n) {
            System.out.println(2);
        } else {
            System.out.println(m);
        }
    }
}
