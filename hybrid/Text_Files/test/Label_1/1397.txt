#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n;
  vector<pair<int, int>> chapters(n);
  for (int i = 0; i < n; i++) {
    cin >> chapters[i].first >> chapters[i].second;
  }
  cin >> k;
  int last = 0;
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (k > chapters[i].first && k <= chapters[i].second) {
      ans++;
    } else if (last < chapters[i].first && last >= chapters[i - 1].first) {
      ans++;
    }
    last = chapters[i].second;
  }
  cout << ans << endl;
  return 0;
}
