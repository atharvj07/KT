import java.util.Scanner;

public class StudyTimeChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            int targetTime = scanner.nextInt();
            if (targetTime == 0) {
                break;
            }

            int numStudies = scanner.nextInt();
            int[] studyStartTimes = new int[numStudies];
            int[] studyEndTimes = new int[numStudies];

            for (int i = 0; i < numStudies; i++) {
                studyStartTimes[i] = scanner.nextInt();
                studyEndTimes[i] = scanner.nextInt();
            }

            int totalStudyTime = 0;
            for (int i = 0; i < numStudies; i++) {
                totalStudyTime += studyEndTimes[i] - studyStartTimes[i];
            }

            if (totalStudyTime >= targetTime) {
                System.out.println("OK");
            } else {
                int shortageTime = targetTime - totalStudyTime;
                System.out.println(shortageTime);
            }
        }

        scanner.close();
    }
}
