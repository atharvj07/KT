#include <algorithm>
#include <iostream>
#include <string>
using namespace std;
string maxNumber(string a, string b) {
  if (a == b) return a;
  int na = a.size();
  int nb = b.size();
  if (na < nb) return b;
  string ans = "";
  for (int i = 0; i < na; i++) {
    char Max = '0' - 1;
    for (int j = i; j < na; j++) {
      if (a[j] <= b[j] && a[j] > Max) { Max = a[j]; }
    }
    if (Max == '0' - 1) {
      int j = i;
      while (j < na && a[j] == '0') j++;
      if (j < na) { Max = a[j]; }
    }
    if (Max == b[i]) {
      for (int j = i; j < na; j++) {
        if (a[j] < b[j] && a[j] > Max) { Max = a[j]; }
      }
    }
    if (Max == '0' - 1) return "0";
    ans += Max;
  }
  return ans;
}
int main() {
  string a, b;
  cin >> a >> b;
  cout << maxNumber(a, b);
}
