n, H = map(int, input().split())

# If the total sand packs can form a perfect pyramid of height H or less
if n <= (H * (H + 1)) // 2:
    # Binary search to find the height at which the sand packs can form a perfect pyramid
    low, high = 1, H
    while low < high:
        mid = (low + high) // 2
        if (mid * (mid + 1)) // 2 < n:
            low = mid + 1
        else:
            high = mid
    print(low)
else:
    # Adjust n to account for the maximum pyramid we can form with height H
    n -= (H * (H + 1)) // 2
    # Calculate the additional rows we need beyond the perfect pyramid of height H
    # Each row beyond height H adds 2 more sand packs than the previous row
    # Solve the quadratic equation to find the number of additional rows
    additional_rows = (-1 + (1 + 8 * n) ** 0.5) / 2
    # The total spots occupied is the height of the pyramid plus the additional rows
    # We take the ceiling of additional_rows since we can't have a fraction of a row
    total_spots = H + int(additional_rows) if additional_rows.is_integer() else H + int(additional_rows) + 1
    print(total_spots)
