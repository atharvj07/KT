MOD = 10**9 + 7

def factorial(n):
    result = 1
    for i in range(2, n + 1):
        result = (result * i) % MOD
    return result

def comb(n, k):
    return factorial(n) // (factorial(k) * factorial(n - k))

def solve(n, x, pos):
    left_count = pos
    right_count = n - pos - 1
    smaller_needed = x - 1
    larger_needed = n - x

    if left_count > smaller_needed or right_count > larger_needed:
        return 0

    ans = comb(smaller_needed, left_count) * comb(larger_needed, right_count)
    ans *= factorial(left_count) * factorial(right_count)
    ans %= MOD
    ans *= factorial(n - left_count - right_count - 1)
    ans %= MOD

    return ans

n, x, pos = map(int, input().split())
print(solve(n, x, pos))
