n = int(input())
a = [0] * (2 * n - 2)
b = [0] * (2 * n - 2)
for i in range(2 * n - 2):
    s = input()
    l = len(s)
    if s == s[::-1]:
        a[l-1] = 1
    else:
        b[l-1] = 1
ans = ''
for i in range(2 * n - 2):
    if a[i] and b[i]:
        ans += 'X'
    elif a[i]:
        ans += 'P'
    else:
        ans += 'S'
print(ans)
