#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  if (k > n || k > 26) {
    cout << -1 << endl;
    return 0;
  }
  vector<int> stages(26, -1);
  int j = 0;
  for (int i = 0; i < n; i++) {
    int idx = s[i] - 'a';
    if (stages[idx] == -1) stages[idx] = j++;
  }
  int ans = 0;
  for (int i = 0; i < k; i++) {
    int minIdx = -1;
    for (int j = 0; j < 26; j++) {
      if (stages[j] != -1 && (minIdx == -1 || stages[minIdx] > stages[j])) minIdx = j;
    }
    if (minIdx == -1) {
      cout << -1 << endl;
      return 0;
    }
    ans += minIdx + 1;
    stages[minIdx] = -1;
  }
  cout << ans << endl;
  return 0;
}
