q = int(input())
shelf = []
pos = {}
for _ in range(q):
    query = input().split()
    op, id = query[0], int(query[1])
    if op == 'L':
        shelf.insert(0, id)
        pos[id] = len(shelf) - 1
    elif op == 'R':
        shelf.append(id)
        pos[id] = len(shelf) - 1
    else:  # query type '?'
        index = pos[id]
        print(min(index, len(shelf) - 1 - index))
