def factorial_mod(n, m):
    result = 1
    for i in range(1, n + 1):
        result = (result * i) % m
    return result

def solve(n, m):
    if n == 1:
        return 1
    dp = [0] * (n + 1)
    dp[1], dp[2] = 1, 2
    for i in range(3, n + 1):
        dp[i] = (i * dp[i - 1] + (i - 1) * dp[i - 2]) % m
    return (factorial_mod(n, m) * dp[n]) % m

n, m = map(int, input().split())
print(solve(n, m))
