n = int(input())
a = list(map(int, input().split()))

a_with_index = [(a[i], i + 1) for i in range(n)]
a_with_index.sort(reverse=True)

total_shots = 0
order = []
for i in range(n):
    total_shots += a_with_index[i][0] * i + 1
    order.append(str(a_with_index[i][1]))

print(total_shots)
print(" ".join(order))
