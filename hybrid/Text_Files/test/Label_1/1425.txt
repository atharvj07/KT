#include <algorithm>
#include <iostream>
#include <string>
using namespace std;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    string s;
    cin >> n >> s;
    string t = s;
    sort(t.begin(), t.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] != t[i]) { ans++; }
    }
    cout << ans << endl;
  }
  return 0;
}
