#include <bits/stdc++.h>
using namespace std;

#define PI 3.14159265
#define pb push_back
typedef long long ll;

int main() {
    int t;
    cin >> t;
    while (t--) {
        ll l, r;
        cin >> l >> r;
        int dig = 0;
        while (l < r) {
            l++;
            int temp = l;
            while (temp > 0) {
                temp /= 10;
                dig++;
            }
        }
        cout << dig << endl;
    }
    return 0;
}
