#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int k, n, m;
    cin >> k >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    vector<int> c(k + n + m, -1);
    int ca = 0, cb = 0;
    for (int i = 0; i < k + n + m; i++) {
      if (b[cb] <= a[ca] || cb == m) {
        c[i] = a[ca];
        ca++;
      } else {
        c[i] = b[cb];
        cb++;
      }
      if (a[ca] == c[i]) ca++;
      if (b[cb] == c[i]) cb++;
    }
    if (ca == n && cb == m) {
      for (int i = 0; i < k + n + m; i++) {
        cout << c[i] << ' ';
      }
      cout << '\n';
    } else {
      cout << -1 << '\n';
    }
  }
  return 0;
}
