import sys

def solve():
    n, k = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    visited = [False] * (n + 1)
    count = 0

    def dfs(node, parent):
        nonlocal count
        count += 1
        visited[node] = True
        for neighbor in adj[node]:
            if not visited[neighbor] and neighbor != parent:
                dfs(neighbor, node)

    dfs(1, 1)
    for i in range(k):
        count //= 2
    return count

for _ in range(int(sys.stdin.readline())):
    sys.stdin.readline()
    print(solve())
