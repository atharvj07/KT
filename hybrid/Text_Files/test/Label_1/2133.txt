#include <bits/stdc++.h>
using namespace std;
string a, b;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    cin >> a >> b;
    int x = 0;
    for (int i = 0; i < n; ++i)
      x += (a[i] - b[i]);
    if (!x || x == -2 || x == 2 * n)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
