#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, m, h;
    cin >> n >> m >> h;
    int maxl[m], maxh[n], ok[105][105];
    memset(ok, 0, sizeof ok);
    for(int i = 1; i <= m; i++) cin >> maxl[i];
    for(int i = 1; i <= n; i++) cin >> maxh[i];
    for(int i = 1; i <= n; i++) {
        string s;
        cin >> s;
        for(int j = 1; j <= m; j++) ok[i][j] = (s[j - 1] == '1');
    }
    vector<vector<int>> ans(n, vector<int>(m));
    for(int i = 1; i <= m; i++) {
        for(int j = 1; j <= h; j++) {
            for(int k = 1; k <= n; k++) {
                if(j > maxl[i] || j > maxh[k] || !ok[k][i]) {
                    ans[k - 1][i - 1] = j - 1;
                    break;
                }
            }
        }
    }
    for(auto x : ans) {
        for(auto y : x) cout << y << " ";
        cout << "\n";
    }
    return 0;
}
