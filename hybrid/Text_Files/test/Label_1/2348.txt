#include <bits/stdc++.h>
using namespace std;
const int maxn = 70;
const long long inf = 1e18;
int n, m, p, a, b, par[maxn];
long long d[maxn];
struct edge {
  int u, v, w;
  bool operator<(const edge &rhs) const { return w < rhs.w; }
} es[205];
int find(int x) { return par[x] == x ? x : par[x] = find(par[x]); }
bool merge(int x, int y) {
  x = find(x), y = find(y);
  if (x == y) return 0;
  par[x] = y;
  return 1;
}
void dij(int st, int t) {
  for (int i = 0; i <= n; i++) d[i] = inf;
  d[st] = 0;
  set<pair<long long, int>> q;
  q.insert({0, st});
  while (q.size()) {
    int v = q.begin()->second;
    q.erase(q.begin());
    if (v == t) break;
    for (int i = 0; i < m; i++) {
      int u = es[i].u, w = es[i].w;
      if (v == u) swap(u, v);
      if (v == st || merge(u, v)) {
        if (d[v] + w < d[u]) d[u] = d[v] + w, q.insert({d[u], u});
      }
    }
  }
}
int main() {
  cin >> n >> m >> a >> b;
  for (int i = 0; i < m; i++) {
    cin >> es[i].u >> es[i].v >> es[i].w;
    if (es[i].w == b) es[i].w = a + b;
  }
  sort(es, es + m);
  for (int i = 1; i <= n; i++) par[i] = i;
  for (int i = 0; i < m; i++) {
    if (merge(es[i].u, es[i].v)) {
      cin >> p;
      dij(1, p);
      for (int j = 1; j <= n; j++) cout << d[j] << " ";
      cout << endl;
      break;
    }
  }
  return 0;
}
