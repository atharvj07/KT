#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100 + 10;
int n, a[MAXN];

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a, a + n, [](int x, int y) { return x - y > y - x; });
        for (int i = 0; i < n; i++) cout << a[i] << " ";
        cout << "\n";
    }
    return 0;
}
