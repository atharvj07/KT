#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int &i : v) cin >> i;
    string ans = "";
    ans += '+';
    int tot = v[0];
    for (int i = 1; i < n; i++) {
        int need = max(2 * v[i] - tot, 0);
        if (need > v[i]) {
            cout << "-1\n";
            return 0;
        }
        tot += v[i] - need;
        ans += (need ? '-' : '+');
    }
    if (tot > 0) {
        cout << "-1\n";
        return 0;
    }
    cout << ans << endl;
    return 0;
}
