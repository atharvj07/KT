import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] w = new int[n];
            int[] a = new int[n];
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                w[i] = sc.nextInt();
                a[i] = sc.nextInt();
                b[i] = sc.nextInt();
            }
            int[] weight = new int[232];
            for (int i = 0; i < n; i++) {
                for (int j = a[i]; j < b[i]; j++) {
                    weight[j] += w[i];
                }
            }
            boolean broken = false;
            for (int i = 0; i < weight.length; i++) {
                if (weight[i] > 150) {
                    broken = true;
                    break;
                }
            }
            System.out.println(broken ? "NG" : "OK");
        }
        sc.close();
    }
}
