public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x=sc.nextInt();
        B a=new B();
        B b=a.clone();

class B implements Cloneable{
    int[] a;
    
    public B(){
        a=new int[10];
    }
    
    public B clone(){
        B b=null;
        try{
            b=(B)super.clone();
        }catch(Exception e){}
        return b;
    }
}
