s, x1, x2 = map(int, input().split())
t1, t2 = map(int, input().split())
p, d = map(int, input().split())
ans = 1e9
if x2 < x1:
    x1, x2 = x2, x1
if d == 1:
    if x1 <= p <= x2:
        ans = min(ans, ((x2 - p) + (p - x1)) * t1 + (x2 - x1) * t2)
    else:
        ans = min(ans, (s - p) * t1 + (x2 - x1) * t2)
        ans = min(ans, (p + s - x2) * t1 + (x2 - x1) * t2)
else:
    if x1 <= p <= x2:
        ans = min(ans, (p - x1) * t1 + (x2 - p) * t2)
    else:
        ans = min(ans, p * t1 + (x2 - x1) * t2)
        ans = min(ans, (s - p + x2) * t1 + (x2 - x1) * t2)
print(ans)
