import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        long l = 1;
        long r = 1000000000;
        while (l < r) {
            long mid = (l + r) / 2;
            int cnt = 0;
            int j = 0;
            for (int i = 0; i < n; i++) {
                if (s[i] > mid) {
                    break;
                }
                if (j + s[i] > mid) {
                    j = 0;
                    cnt++;
                }
                j += s[i];
            }
            if (cnt + 1 <= k) {
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        System.out.println(l);
    }
}
