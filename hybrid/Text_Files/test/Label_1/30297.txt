import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int newA = sc.nextInt();
    int newB = sc.nextInt();
    int n = sc.nextInt();
    boolean overlap = false;
    for (int i = 0; i < n; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      if (overlap) continue;
      if (a < newB && newA < b) overlap = true;
    }
    System.out.println(overlap ? "1" : "0");
  }
}
