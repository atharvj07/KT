#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m, min, max;
  cin >> n >> m >> min >> max;
  set<int> temps;
  bool poss = 1;
  for (int i = 0; i < m; i++) {
    int t;
    cin >> t;
    temps.insert(t);
    poss &= *temps.begin() >= min && *temps.rbegin() <= max;
  }
  if (temps.size() != n || !poss)
    cout << "Incorrect";
  else
    cout << "Correct";
}
