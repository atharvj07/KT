for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split())) + [float('inf')]
    l = [0] * (n + 1)
    for i in range(n):
        if a[i] > a[i + 1]:
            l[i] = 1
    for i in range(2, n + 1):
        if l[i - 2] > 0 and l[i] > 0:
            print('NO')
            break
    else:
        print('YES')
