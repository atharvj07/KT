import java.util.Scanner;

public class Main {
    static int n, m, k, d[][][], visited[][][];

    static boolean valid(int x, int y, int z) {
        return x >= 1 && x <= n && y >= 1 && y <= m && z >= 1 && z <= k && d[x][y][z] == 1;
    }

    static void dfs1(int x, int y, int z) {
        visited[x][y][z] = 1;
        if (valid(x + 1, y, z) && visited[x + 1][y][z] == 0) {
            dfs1(x + 1, y, z);
        }
        if (valid(x, y + 1, z) && visited[x][y + 1][z] == 0) {
            dfs1(x, y + 1, z);
        }
        if (valid(x, y, z + 1) && visited[x][y][z + 1] == 0) {
            dfs1(x, y, z + 1);
        }
    }

    static void dfs2(int x, int y, int z) {
        visited[x][y][z] = 2;
        if (valid(x + 1, y, z) && visited[x + 1][y][z] == 0) {
            dfs2(x + 1, y, z);
        }
        if (valid(x, y + 1, z) && visited[x][y + 1][z] == 0) {
            dfs2(x, y + 1, z);
        }
        if (valid(x, y, z + 1) && visited[x][y][z + 1] == 0) {
            dfs2(x, y, z + 1);
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        d = new int[n + 1][m + 1][k + 1];
        visited = new int[n + 1][m + 1][k + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                String s = in.next();
                for (int l = 1; l <= k; l++) {
                    d[i][j][l] = s.charAt(l - 1) == '1' ? 1 : 0;
                }
            }
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                for (int l = 1; l <= k; l++) {
                    if (d[i][j][l] == 0) {
                        continue;
                    }
                    if (visited[i][j][l] == 0) {
                        dfs1(i, j, l);
                    }
                    visited[i][j][l] = 0;
                    if (visited[i][j][l] == 0) {
                        dfs2(i, j, l);
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
