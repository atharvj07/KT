#include<bits/stdc++.h>
using namespace std;
int main() {
  string s;
  while (cin>>s) {
    int n = s.size();
    vector<vector<unordered_map<char,bool>>>dp(n+1,vector<unordered_map<char,bool>>(3));
    dp[n][0]['0'] = dp[n][0]['1'] = dp[n][2]['0'] = dp[n][2]['1'] = true;

    for(int i=n-1;i>=0;i--) {
      for(int prv=0;prv<3;prv++) {
        for(auto&[key,val]:dp[i+1][prv]) {
          dp[i][prv][key] = val;
          dp[i][prv][s[i]] = true;
          if(prv == 0) dp[i][1][s[i]] = true;
          if(prv == 0 || prv == 1) dp[i][2][s[i]] = true;
        }
      }
    }

    string t;
    int cur = 0;
    for(int i=0;i<n;i++) {
      for(int nxt=2;nxt>=0;nxt--) {
        if(dp[i+1][nxt][s[i]]) {
          t.push_back(s[i]);
          cur = nxt;
          break;
        }
      }
    }
    cout<<t<<endl;
  }
  return 0;
}
