n, k = map(int, input().split())
x = list(map(int, input().split()))
a = int(input())
c = list(map(int, input().split()))
dp = [0] + [1e18] * n
for i in range(1, n+1):
    dp[i] = min(dp[i], dp[i-1] + c[i-1])
    for j in range(i, n+1):
        if k + a * (j-i) < x[j-1]:
            break
        dp[j] = min(dp[j], dp[i-1] + c[i-1])
print(dp[n] if dp[n] != 1e18 else -1)
