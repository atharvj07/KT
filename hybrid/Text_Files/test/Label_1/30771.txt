import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] c = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                c[i][j] = sc.nextInt();
            }
        }
        Set<Integer> s1 = new HashSet<>();
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 2; j++) {
                s1.add(c[i][j] - c[i][j+1]);
            }
        }
        Set<Integer> s2 = new HashSet<>();
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < 3; j++) {
                s2.add(c[i][j] - c[i+1][j]);
            }
        }
        System.out.println(s1.size() == 1 && s2.size() == 1 ? "Yes" : "No");
    }
}
