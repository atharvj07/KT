#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  if (a[0] == 3 || a.back() == 3) {
    cout << "Infinite" << endl;
  } else {
    cout << "Finite" << endl;
    cout << (n - 1) * 4 << endl;
  }
  return 0;
}
