a, b = input().split()
a = int(a)
b = int(b)
if b > a:
    a, b = b, a
ans = 0
while a != b:
    if a % 3 == b % 3:
        a = a // 3
        b = b // 3
        ans += 1
    elif a % 2 == b % 2:
        a = a // 2
        b = b // 2
        ans += 1
    else:
        ans = -1
        break
print(ans)
