import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        int d = sc.nextInt();
        char[][] color = new char[H][W];
        boolean[][][] used = new boolean[H][W][4];
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                for (int k = 0; k < 4; k++) {
                    used[i][j][k] = false;
                }
            }
        }
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                for (int k = 0; k < 4; k++) {
                    if (used[i][j][k]) continue;
                    for (int ii = 0; ii < H; ii++) {
                        for (int jj = 0; jj < W; jj++) {
                            if (i != ii || j != jj) {
                                if (Math.abs(i - ii) + Math.abs(j - jj) == d) {
                                    used[ii][jj][k] = true;
                                }
                            }
                        }
                    }
                    color[i][j] = "RYGB".charAt(k);
                    break;
                }
            }
        }
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                System.out.print(color[i][j]);
            }
            System.out.println();
        }
    }
}
