from itertools import product

def min_cost_to_remember(n, m, strings, cost):
    # Initialize memoization table with None values
    dp = [None] * (1 << n)
    dp[0] = 0

    def get_cost(mask, pos):
        # Calculate the cost to make strings at pos unique
        freq = {}
        for i in range(n):
            if mask & (1 << i):
                if strings[i][pos] not in freq:
                    freq[strings[i][pos]] = 0
                freq[strings[i][pos]] += 1
        total_cost = sum(cost[i][pos] for i in range(n) if mask & (1 << i))
        return min(total_cost - cost[i][pos] for i in range(n) if mask & (1 << i) and freq[strings[i][pos]] > 1)

    def solve(mask):
        # Solve using memoization
        if dp[mask] is not None:
            return dp[mask]
        dp[mask] = float('inf')
        for pos in range(m):
            new_mask = mask
            for i in range(n):
                if mask & (1 << i) and all((mask & (1 << j)) == 0 or strings[i][pos] != strings[j][pos] for j in range(n)):
                    new_mask |= (1 << i)
            if new_mask != mask:
                dp[mask] = min(dp[mask], get_cost(mask, pos) + solve(new_mask))
        return dp[mask]

    # Minimize the cost for all strings to be unique in some position
    answer = solve((1 << n) - 1)
    return 0 if answer == float('inf') else answer

# Sample Input
n, m = map(int, input().split())
strings = [input().strip() for _ in range(n)]
cost = [list(map(int, input().split())) for _ in range(n)]

# Print the minimum sum to make the multiset easy to remember
print(min_cost_to_remember(n, m, strings, cost))
