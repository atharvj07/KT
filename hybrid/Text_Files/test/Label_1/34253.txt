import java.util.Scanner;

public class PetyasSwaps {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }

        int minSwaps = Integer.MAX_VALUE;
        int ways = 0;

        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                int[] copy = permutation.clone();
                swap(copy, i, j);
                int swaps = countSwaps(copy);

                if (swaps < minSwaps) {
                    minSwaps = swaps;
                    ways = 1;
                } else if (swaps == minSwaps) {
                    ways++;
                }
            }
        }

        System.out.println(minSwaps + " " + ways);
    }

    static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    static int countSwaps(int[] arr) {
        int swaps = 0;
        for (int i = 1; i < arr.length; i++) {
            int j = i;
            while (j > 0 && arr[j] < arr[j - 1]) {
                swap(arr, j, j - 1);
                swaps++;
                j--;
            }
        }
        return swaps;
    }
}
