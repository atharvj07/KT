#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<bool> rooms(10, false);
  string events;
  cin >> events;

  for (char event : events) {
    if (event == 'L') {
      for (int i = 0; i < 10; i++) {
        if (!rooms[i]) {
          rooms[i] = true;
          break;
        }
      }
    } else if (event == 'R') {
      for (int i = 9; i >= 0; i--) {
        if (!rooms[i]) {
          rooms[i] = true;
          break;
        }
      }
    } else {
      int room = event - '0';
      rooms[room] = false;
    }
  }

  for (bool room : rooms) cout << (room ? '1' : '0');
  cout << '\n';

  return 0;
}
