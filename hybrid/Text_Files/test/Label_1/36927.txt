n, c, d = map(int, input().split())
f = []
for _ in range(n):
    b, p, t = input().split()
    f.append((int(b), int(p), t))
f.sort(key=lambda x: x[1])

ans = 0
for b1, p1, t1 in f:
    for b2, p2, t2 in f:
        if t1 == t2: continue
        if t1 == 'C':
            if p1 + p2 <= c: ans = max(ans, b1 + b2)
        else:
            if p1 + p2 <= d: ans = max(ans, b1 + b2)
print(ans)
