import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        if (m % n == 0) {
            System.out.println(m);
            return;
        }
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        int[] c = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = Math.min(a[i], b[i]);
            d[i] = Math.max(a[i], b[i]) - c[i];
        }
        int x = m / n;
        int y = m % n;
        int ans = 0;
        if (y == 0) {
            for (int i = 0; i < n; i++) {
                ans += x * c[i];
            }
        } else {
            for (int i = 0; i < y; i++) {
                ans += x * c[i] + d[i];
            }
            for (int i = y; i < n; i++) {
                ans += x * c[i];
            }
        }
        System.out.println(ans);
    }
}
