import sys
input = sys.stdin.readline
n, k = map(int, input().split())
a = list(map(int, input().split()))
if sum(a) <= k:
    print(1, 1)
elif sum(a) > 2*k:
    print(-1, 0)
else:
    if 200 <= k:
        print(n//4, (1<<(n//4))%1000000007)
    else:
        print((n+1)//3, (2<<(n+1//3-1))%1000000007)
