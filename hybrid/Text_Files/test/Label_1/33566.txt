import java.util.Arrays;
import java.util.Scanner;

class Main {
    static int MOD = 1000000007;

    static long[] pow2 = new long[61];
    static {
        pow2[0] = 1;
        for (int i = 1; i < 61; i++) {
            pow2[i] = (pow2[i - 1] << 1) % MOD;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        int even = 0, odd = 0;
        for (int ai : a) {
            if (ai % 2 == 0) {
                even++;
            } else {
                odd++;
            }
        }

        long ans = 0;
        for (int i = 2; i <= m; i++) {
            long cur = 0;
            if ((i & 1) == 0) {
                cur = (pow2[n - even] - 1 + MOD) % MOD;
                cur *= pow2[even];
                cur %= MOD;
            }
            cur += (pow2[n - odd] - 1 + MOD) % MOD;
            cur %= MOD;
            ans += cur;
            ans %= MOD;
        }
        System.out.println(ans);
    }
}
