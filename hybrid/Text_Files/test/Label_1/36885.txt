n, a, b, k = map(int, input().split())
h = list(map(int, input().split()))

def calculate_hits(hp, a, b):
    total_damage = a + b
    cycles = (hp + total_damage - 1) // total_damage
    return max(0, cycles - 1)

hits_needed = [calculate_hits(hp, a, b) for hp in h]
hits_needed.sort()

points = 0
for hits in hits_needed:
    if k >= hits:
        k -= hits
        points += 1
    else:
        break

print(points)
