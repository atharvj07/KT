#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, a[55], pos[55], p;
vector<int> per;
double ans;
int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) cin >> a[i];
    cin >> p;
    per.resize(n);
    for (int i = 0; i < n; ++i) per[i] = i + 1;
    int num = 1;
    do {
        int sum = 0;
        bool flag = 1;
        for (int i = 0; i < n; ++i) {
            sum += a[per[i] - 1];
            if (sum > p) {
                flag = 0;
                break;
            }
        }
        if (flag) ++num;
    } while (next_permutation(per.begin(), per.end()));
    ans = (double)num / (double)(n);
    cout << fixed << setprecision(9) << ans << endl;
    return 0;
}
