#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int a, b, p;
    string s;
    cin >> a >> b >> p >> s;
    int n = s.size();
    vector<bool> can(n + 1, false);
    can[n] = true;
    if (a + b <= p) {
      can[n - 1] = true;
    }
    for (int i = n - 2; i >= 0; i--) {
      if (s[i] == 'A') {
        if (a <= p) {
          can[i] = true;
        }
        if (i + 1 < n && can[i + 1]) {
          can[i] = true;
        }
      } else {
        if (b <= p) {
          can[i] = true;
        }
        if (i + 1 < n && can[i + 1]) {
          can[i] = true;
        }
      }
    }
    int ans = -1;
    for (int i = 1; i <= n; i++) {
      if (can[i]) {
        ans = i;
        break;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
