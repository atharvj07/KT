from sys import stdin
input = stdin.readline
mod = 10**9+7

def dfs1(u, p, bc):
    bc[u]=xc[u]
    for v in adj[u]:
        if v==p: continue
        dfs1(v, u, bc)
        bc[u]=gcd(bc[u], bc[v])

def dfs2(u, p, bc, ac):
    ac[u]+=bc[u]
    for v in adj[u]:
        if v==p: continue
        dfs2(v, u, bc, ac)
        ac[u]+=ac[v]

n = int(input())
xc = list(map(int, input().split()))
adj = [[] for _ in range(n+1)]
for _ in range(n-1):
    a,b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

bc = [0]*n
dfs1(1,1,bc)
ac = [0]*n
dfs2(1,1,bc, ac)
print(sum(ac)%mod)
