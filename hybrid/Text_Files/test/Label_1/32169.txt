import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            if (a[i] % 3 == 0) {
                flag = false;
            }
        }
        if (flag) {
            System.out.println("BitAryo");
        } else {
            System.out.println("BitLGM");
        }
    }
}
