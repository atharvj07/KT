#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  set<int> s;
  for (int i = 0; i < n; i++) cin >> a[i], s.insert(a[i]);
  n = s.size();
  vector<int> v(s.begin(), s.end());
  if (n == 1)
    cout << v[0];
  else
    cout << v[n - 2];
  return 0;
}
