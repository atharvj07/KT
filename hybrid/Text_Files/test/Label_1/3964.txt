#include <iostream>
#include <cmath>
using namespace std;
int main()
{
    long long l, r, k;
    cin >> l >> r >> k;
    long long pow = 1, cnt = 0;
    while (true)
    {
        pow *= k;
        if (pow > r)
            break;
        if (pow >= l)
        {
            cout << pow << " ";
            cnt++;
        }
    }
    if (cnt == 0)
        cout << "-1";
    return 0;
}
