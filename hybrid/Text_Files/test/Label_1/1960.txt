#include <bits/stdc++.h>
using namespace std;
int n, m, p, dp[105][105][55], pre[105][105][55];
string s[105], ans;
int f(int l, int r, int lf) {
  if (l > r)
    return 0;
  if (dp[l][r][lf] != -1)
    return dp[l][r][lf];
  if (l == r)
    dp[l][r][lf] = 1;
  else if (r - l == 1)
    dp[l][r][lf] = (s[l][lf] == s[r][m - lf - 1]);
  else {
    dp[l][r][lf] = f(l + 1, r, lf + 1);
    if (s[l][lf] == s[r][m - lf - 1])
      dp[l][r][lf] = max(dp[l][r][lf], 2 + f(l + 1, r - 1, lf + 1));
    for (int i = l; i < r; i++)
      if (s[i][lf] == s[r][m - lf - 1])
        dp[l][r][lf] = max(dp[l][r][lf], 1 + f(i + 1, r, lf + 1) + f(l, i, lf));
  }
  return dp[l][r][lf];
}
void g(int l, int r, int lf) {
  if (l > r)
    return;
  if (l == r)
    ans += s[l][lf];
  else if (r - l == 1) {
    if (s[l][lf] == s[r][m - lf - 1])
      ans += s[l][lf];
  } else {
    if (dp[l][r][lf] == f(l + 1, r, lf + 1))
      g(l + 1, r, lf + 1);
    else if (dp[l][r][lf] == 2 + f(l + 1, r - 1, lf + 1) &&
             s[l][lf] == s[r][m - lf - 1]) {
      ans += s[l][lf];
      g(l + 1, r - 1, lf + 1);
      ans += s[r][m - lf - 1];
    } else {
      for (int i = l; i < r; i++)
        if (s[i][lf] == s[r][m - lf - 1] &&
            dp[l][r][lf] == 1 + f(i + 1, r, lf + 1) + f(l, i, lf)) {
          g(i + 1, r, lf + 1);
          g(l, i, lf);
          break;
        }
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(0);
  cout.tie(0);
  cin >> n >> m;
  for (int i = 0; i < n; i++) {
    cin >> s[i];
  }
  memset(dp, -1, sizeof dp);
  p = f(0, n - 1, 0);
  g(0, n - 1, 0);
  cout << p << "\n";
  if (p)
    cout << ans << "\n";
  return 0;
}
