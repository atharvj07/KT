#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll mod = 1e9 + 7;

ll add(ll a, ll b) { return (a + b) % mod; }
ll mul(ll a, ll b) { return (a * b) % mod; }

ll pw[100005];

int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;

  vector<int> bit(n + 2, 0);
  for (int i = 0; i < n; i++) {
    if (s[i] == '0') bit[i + 1] = -1;
    else bit[i + 1] = 1;
  }

  for (int i = 2; i <= n; i++) bit[i] += bit[i - 1];
  for (int i = 1; i <= n; i++) bit[i] += bit[i - 1];

  pw[0] = 1;
  for (int i = 1; i <= n; i++) { pw[i] = mul(pw[i - 1], 2); }

  while (q--) {
    int l, r;
    cin >> l >> r;
    ll ans = pow(2, max(0, bit[r] - (l > 1 ? bit[l - 1] : 0)));
    ans = mul(ans, pw[max(0, r - l + 1 - (bit[r] - (l > 1 ? bit[l - 1] : 0)))]);
    cout << ans << endl;
  }

  return 0;
}
