import java.util.*;
import java.io.*;

class Main {
    static int[] indegree;
    static List<Integer>[] graph;
    static boolean[] onCoprocessor;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        line = br.readLine().split(" ");
        onCoprocessor = new boolean[n];
        for (int i = 0; i < n; i++) {
            onCoprocessor[i] = line[i].equals("1");
        }
        indegree = new int[n];
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            graph[a].add(b);
            indegree[b]++;
        }
        int ans = 0;
        Queue<Integer> queue = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (indegree[i] == 0 && !onCoprocessor[i]) {
                queue.offer(i);
            }
        }
        while (!queue.isEmpty()) {
            int size = queue.size();
            ans++;
            for (int i = 0; i < size; i++) {
                int curr = queue.poll();
                for (int next : graph[curr]) {
                    indegree[next]--;
                    if (indegree[next] == 0 && !onCoprocessor[next]) {
                        queue.offer(next);
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            if (indegree[i] > 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
