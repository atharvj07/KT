def gcd(x, y):
    return gcd(y, x % y) if y else x

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort(key=lambda x: -x)
    ans = 0
    for i in range(n):
        for j in range(i + 1, n):
            if gcd(a[i], 2 * a[j]) > 1:
                ans += 1
    print(ans)
