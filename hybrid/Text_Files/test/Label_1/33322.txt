import java.util.*;

class Main {
    static final int MOD = 1000000007;

    static int n, p;
    static int[] c;
    static int[][][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        p = sc.nextInt();
        c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextInt();
        }
        dp = new int[n + 1][2][2]; // dp[i][j][k] = the number of ways to color the first i pieces and put
        // arrows so that the score has the parity of k,
        // and the last piece is colored j

        // initialize the base case
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < 2; j++) {
                dp[n][i][j] = 1;
            }
        }

        // calculate the dp table in reverse order
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 2; j++) {
                for (int k = 0; k < 2; k++) {
                    // if the current piece is unpainted, we can color it either 0 or 1
                    if (c[i] == -1) {
                        dp[i][j][k] = (dp[i + 1][0][k] + dp[i + 1][1][k]) % MOD;
                    }
                    // if the current piece is painted with the same color as the last piece, we can only
                    // put an arrow from the last piece to the current piece
                    else if (c[i] == j) {
                        dp[i][j][k] = (dp[i + 1][j][k] + dp[i + 1][j ^ 1][k ^ 1]) % MOD;
                    }
                    // if the current piece is painted with a different color than the last piece, we can
                    // put an arrow from the current piece to the last piece
                    else {
                        dp[i][j][k] = dp[i + 1][j ^ 1][k ^ 1] % MOD;
                    }
                }
            }
        }

        // print the answer
        System.out.println(dp[0][0][p]);
    }
}
