#include <bits/stdc++.h>
using namespace std;
typedef long long LL;
const int kN = 200005, kB = 400005;
LL n, a[kN], b[kB], ans;
int m, fa[kB];
inline LL Find(int x) { return x == fa[x] ? fa[x] : fa[x] = Find(fa[x]); }
inline void Union(int x, int y) { fa[Find(x)] = Find(y); }
inline LL Get(int l, int r) { return r < 2 ? a[1] : a[r] - a[l - 1]; }
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> m;
  for (int i = 1; i <= m; i++) cin >> b[i];
  for (int i = 1; i <= m; i++) {
    if (b[i] == b[i - 1])
      continue;
    Union(b[i], b[i - 1]);
  }
  int l = 0, r = 0, las = 0;
  for (int i = 1; i <= m; i++) {
    if (Find(b[i]) == Find(b[las]))
      continue;
    ans += Get(l, r);
    l = r = b[las] = Find(b[i]);
    if (i < m && b[i] == b[i + 1])
      r = Find(b[i]);
  }
  ans += Get(l, r);
  cout << ans << endl;
}
