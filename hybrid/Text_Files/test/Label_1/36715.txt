s=input();ans=0;tmp=-1;num=[0]*len(s)
for i in range(len(s)):
    num[i]=int(s[i])
    tmp=(tmp*10+num[i])%3
    if tmp==0:ans+=1
for i in range(len(s)-1):
    for j in range(i+1,len(s)):
        tmp=(num[i]*(10**(j-i))-tmp)%3
        if num[j]==0:ans+=tmp==0
        else:ans+=tmp==num[j]%3
print(ans)
