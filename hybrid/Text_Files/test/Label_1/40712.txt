n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

sa = [a[0]]
sb = [b[0]]
ans = 0

for i in range(1, n):
    sa.append(max(sa[i - 1], a[i]))
    sb.append(min(sb[i - 1], b[i]))

for i in range(n):
    for j in range(i, n):
        if sa[j] == sb[i]:
            ans += 1

print(ans)
