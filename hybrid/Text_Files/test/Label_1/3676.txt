#include <bits/stdc++.h>
using namespace std;

int main() {
    long n, k = LLONG_MAX;
    cin >> n;
    vector<long> arr(n);
    for (long i = 0; i < n; i++) { cin >> arr[i]; }
    sort(arr.begin(), arr.end());
    for (long i = 1; i < n; i++) {
        if (arr[i] - arr[i - 1] < k) { k = arr[i] - arr[i - 1]; }
    }
    cout << k << endl;
    return 0;
}
