import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    mn = min(a)
    ok = True
    for i in range(n):
        for j in range(i + 1, n):
            if gcd(a[i], a[j]) == mn:
                ok = False
                break
    print("YES" if ok else "NO")
