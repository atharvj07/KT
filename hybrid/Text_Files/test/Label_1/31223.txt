import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] dp = new int[10001];
        dp[0] = 1;
        for (int i = 1; i <= 9; i++)
            for (int j = i; j <= 10000; j++)
                dp[j] += dp[j - i];
        while (sc.hasNext()) {
            int n = sc.nextInt(), s = sc.nextInt();
            System.out.println(dp[s]);
        }
    }
}
