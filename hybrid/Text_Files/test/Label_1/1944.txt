#include <iostream>
using namespace std;

int main() {
    string s;
    int k;
    cin >> s >> k;
    int n = s.length();
    int ret = 0;
    for (int i = 0; i < n; i++) {
        int j = i + n;
        while (j < n + k && s[i] == s[j]) j++;
        ret = max(ret, j - i);
    }
    cout << ret << endl;
    return 0;
}
