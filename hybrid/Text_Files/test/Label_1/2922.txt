#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=2e5+5;
int n,m;
ll a[maxn], sum[maxn];
bool check(int l, int r){
    ll mn=sum[r]-sum[l-1];
    if(mn%2==1){
        mn=(mn+1)/2;
        return a[l+mn-1]<=m;
    }
    else{
        mn/=2;
        return a[l+mn-1]<=m&&a[l+mn]<=m;
    }
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin>>n>>m;
    sum[0]=0;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        sum[i]=sum[i-1]+a[i];
    }
    ll ans=0;
    for(int i=1;i<=n;i++){
        int r=n;
        int l=i;
        while(l<=r){
            int mid=(l+r)/2;
            if(check(i,mid)){
                ans+=r-mid+1;
                l=mid+1;
            }
            else{
                r=mid-1;
            }
        }
    }
    cout<<ans<<'\n';
    return 0;
}
