#include <bits/stdc++.h>
using namespace std;
int inv[1501], a[1501];
int32_t main() {
  int n, m, l, r;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i], inv[a[i]] = i;
  cin >> m;
  while (m--) {
    cin >> l >> r;
    int odd = 0;
    for (int i = l; i < r; i++) {
      odd += inv[a[i + 1]] < inv[a[i]];
      swap(inv[a[i]], inv[a[i + 1]]);
      swap(a[i], a[i + 1]);
    }
    cout << (odd & 1 ? "odd" : "even") << '\n';
  }
}
