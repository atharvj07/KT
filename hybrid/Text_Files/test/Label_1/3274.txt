#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n, r;
        cin >> n >> r;
        vector<int> monsters(n);
        for (int i = 0; i < n; i++) { cin >> monsters[i]; }
        sort(monsters.begin(), monsters.end());
        int ans = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (monsters[i] <= 0) {
                ans++;
            } else {
                ans += (monsters[i] + r - 1) / r;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
