#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
#define int long long
int32_t main() {
    int n, m, r;
    cin >> n >> m >> r;
    vector<int> buy(n), sell(m);
    for (int& x : buy) cin >> x;
    for (int& x : sell) cin >> x;
    sort(buy.begin(), buy.end());
    sort(sell.begin(), sell.end(), greater<int>());
    for (int i = 0; i < n; i++) {
        if (r >= buy[i]) {
            r -= buy[i];
            r += sell[i];
        } else
            break;
    }
    cout << r << endl;
    return 0;
}
