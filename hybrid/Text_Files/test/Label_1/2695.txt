#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=3e5+10;
int n,m,d,p[maxn],sz[maxn],vis[maxn];
vector<int>adj[maxn];
int dfs(int s){
    vis[s]=1;
    sz[s]=1;
    for(auto u:adj[s])if(!vis[u])sz[s]+=dfs(u),sz[s]=min(sz[s],n-sz[s]);
    return sz[s];
}
int main(){
    cin>>n>>m>>d;
    for(int i=0;i<m;i++){
        cin>>p[i];
        vis[p[i]]=1;
    }
    for(int i=0;i<n-1;i++){
        int u,v;cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int sum=0;
    for(int i=1;i<=n;i++)if(!vis[i]&&dfs(i)>=d)sum++;
    cout<<sum<<endl;
    for(int i=1;i<=n;i++)if(!vis[i]&&dfs(i)>=d)cout<<i<<" ";
}
