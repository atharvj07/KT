import java.io.*;
import java.util.*;

public class Main {
    static final String NO = "NO";
    static final String[] DIRS = {"U", "L", "D", "R"};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), l = sc.nextInt();
        List<Point> data = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            long t = sc.nextLong();
            int x = sc.nextInt(), y = sc.nextInt();
            data.add(new Point(t, x, y));
        }

        Map<Long, Integer> freq = new HashMap<>();
        for (Point p : data) {
            freq.merge(p.t, 1, (a, b) -> a + b);
        }

        long curr_x = 0, curr_y = 0, last_t = 0;
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<Long, Integer> entry : freq.entrySet()) {
            long curr_t = entry.getKey(), ct = entry.getValue();
            while (curr_t - last_t >= l) {
                last_t += l;
                int dir = (int) ((curr_y - curr_x) % 4);
                if (dir < 0) {
                    dir += 4;
                }
                sb.append(DIRS[dir]);
            }
            long dx = (curr_t - last_t) % l, dy = (curr_t - last_t) / l;
            int dir = 0;
            if (curr_y == curr_x + dy) {
                dir = 1;
            } else if (curr_y == curr_x + dy * 2) {
                dir = 2;
            } else if (curr_y == curr_x + dy * 2 + dx) {
                dir = 3;
            }
            for (int i = 0; i < ct; i++) {
                sb.append(DIRS[dir]);
            }
            curr_x += dx;
            curr_y += dy * ct;
            last_t = curr_t;
        }
        String ans = sb.toString();
        if (ans.length() != l * (freq.size() - 1)) {
            ans = NO;
        }
        System.out.println(ans);
    }

    static class Point {
        long t;
        int x, y;

        Point(long t, int x, int y) {
            this.t = t;
            this.x = x;
            this.y = y;
        }
    }
}
