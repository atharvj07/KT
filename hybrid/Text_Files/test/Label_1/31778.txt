import java.util.Scanner;

public class DiverseStrings {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        for (int i = 0; i < n; i++) {
            String string = scanner.nextLine();

            if (isDiverse(string)) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }

        scanner.close();
    }

    private static boolean isDiverse(String string) {
        boolean[] seen = new boolean[26]; // One boolean flag for each letter of the alphabet

        for (char c : string.toCharArray()) {
            int index = c - 'a'; // Convert the character to its index in the alphabet

            if (seen[index]) {
                return false; // Duplicate letter found
            }

            seen[index] = true;

            // Check if the current letter is adjacent to the previous one
            if (index > 0 && !seen[index - 1]) {
                return false;
            }
        }

        return true; // All letters are distinct and consecutive
    }
}
