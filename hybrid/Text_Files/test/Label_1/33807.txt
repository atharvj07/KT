import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int[] p = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            p[i] = Integer.parseInt(st.nextToken()) - 1;
        }
        List<int[]> pairs = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            pairs.add(new int[] { Integer.parseInt(st.nextToken()) - 1, Integer.parseInt(st.nextToken()) - 1 });
        }
        int[] inv = new int[n];
        for (int i = 0; i < n; i++) {
            inv[p[i]] = i;
        }
        int[] cycle = new int[n];
        int[] next = new int[n];
        int[] deg = new int[n];
        for (int[] pair : pairs) {
            int a = pair[0], b = pair[1];
            if (a == b) {
                continue;
            }
            cycle[a] = cycle[b] = 1;
            if (inv[a] < inv[b]) {
                next[inv[a]] = inv[b];
            } else {
                next[inv[b]] = inv[a];
            }
            deg[inv[a]]++;
            deg[inv[b]]++;
        }
        Queue<Integer> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            if (deg[i] == 1) {
                q.add(i);
            }
        }
        while (!q.isEmpty()) {
            int u = q.poll();
            if (cycle[u] == 1) {
                int v = next[u];
                deg[v]--;
                if (deg[v] == 1) {
                    q.add(v);
                }
            }
            p[u] = p[inv[u]];
            inv[p[u]] = u;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            sb.append(p[i] + 1).append(" ");
        }
        System.out.println(sb);
    }
}
