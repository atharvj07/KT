#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int cnt = 0, ans = -1;
    for (int i = 0; i < n; i++) {
      if (a[0] != a[i]) {
        ans = i + 1;
        cnt++;
      }
    }
    if (cnt == 0) ans = -1;
    if (cnt > 1) ans = -1;
    cout << ans << endl;
  }
  return 0;
}
