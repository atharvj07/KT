#include <bits/stdc++.h>
using namespace std;
const int64_t mod = 998244353;
int main() {
  int64_t n, k, l = 0, r = 0, ans = 1;
  cin >> n >> k;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    cin >> v[i];
    if (v[i] == -1) { r++; } else { r = 0; }
  }
  for (int i = n - 1; i >= 0; i--) {
    if (v[i] == -1) { l++; } else { l = 0; }
    if (l + r > 1) { ans = 0; }
  }
  if (ans == 0) {
    cout << 0 << '\n';
    return 0;
  }
  for (int i = 0; i < n; i++) {
    if (v[i] != -1) { continue; }
    l = 0, r = 0;
    for (int j = i - 1; j >= 0; j--) {
      if (v[j] == -1) { l++; } else { l = 0; }
    }
    for (int j = i + 1; j < n; j++) {
      if (v[j] == -1) { r++; } else { r = 0; }
    }
    int64_t tl = min(l, r);
    ans = (1LL * ans * (k - 1)) % mod;
    if (l == r && r == tl) { ans = (1LL * ans * (k - 2)) % mod; }
  }
  cout << ans << '\n';
}
