import sys
input = sys.stdin.readline

n = int(input())
intervals = []
ans = []
for _ in range(n):
    q, l, r = map(int, input().split())
    if q == 1:
        i = len(intervals)
        intervals.append((l, r))
        if i > 0 and intervals[i - 1][1] >= l:
            ans.append('YES')
        else:
            ans.append('NO')
    else:
        print('YES' if l != r and intervals[l - 1][0] < intervals[r - 1][0] < intervals[l - 1][1] else 'NO')
