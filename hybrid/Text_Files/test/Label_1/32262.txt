import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int cnt = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur + s[i] <= 4) {
                cur += s[i];
            } else {
                cur = s[i];
                cnt++;
            }
        }
        if (cur > 0) {
            cnt++;
        }
        System.out.println(cnt);
    }
}
