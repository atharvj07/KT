#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  int n; cin >> n;
  unordered_map<ll, unordered_set<ll>> x, y;
  bool ok = false;
  ll s1 = 0, s2 = 0;
  while (n--) {
    ll a, b;
    cin >> a >> b;
    if (x.count(a) && x[a].count(b)) {
      ok = true;
      continue;
    }
    if (x.count(a)) {
      s1 += x[a].size();
    }
    if (y.count(b)) {
      s2 += y[b].size();
    }
    x[a].insert(b);
    y[b].insert(a);
  }
  cout << ((x.size() + y.size() <= s1 + s2 + ok) ? "YES" : "NO") << endl;
}
