import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        int ans = 0;
        int[] p = new int[1 << 20];
        boolean[] exist = new boolean[1 << 20];
        for (int i = 0; i < (1 << 20); i++) {
            for (int j = 0; j < n; j++) {
                if (exist[i ^ s[j]]) {
                    p[i] = i ^ s[j];
                    exist[i] = true;
                    break;
                }
            }
            if (exist[i]) {
                ans = Math.max(ans, Integer.bitCount(i));
            }
        }
        System.out.println(ans);
        int[] res = new int[1 << ans];
        int t = (1 << ans) - 1;
        for (int i = (1 << ans) - 1; i >= 0; i--) {
            res[i] = t;
            t = p[t];
        }
        for (int i : res) {
            System.out.print(i + " ");
        }
        scanner.close();
    }
}
