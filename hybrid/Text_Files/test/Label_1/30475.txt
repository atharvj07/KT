import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();

    // Create a map to store the frequency of each element in the array.
    Map<Integer, Integer> frequencyMap = new HashMap<>();

    // Initialize the frequency of each element to 0.
    for (int i = 1; i <= n; i++) {
      frequencyMap.put(i, 0);
    }

    // Process each query.
    for (int i = 0; i < q; i++) {
      int l = sc.nextInt();
      int r = sc.nextInt();
      int x = sc.nextInt();

      // Increment the frequency of each element in the range [l, r] by x.
      for (int j = l; j <= r; j++) {
        frequencyMap.put(j, frequencyMap.get(j) + x);
      }
    }

    // Create a list to store the possible maximum values.
    List<Integer> possibleMaxValues = new ArrayList<>();

    // Iterate over the frequency map and add the elements with maximum frequency to the list.
    int maxFrequency = 0;
    for (int element : frequencyMap.keySet()) {
      int frequency = frequencyMap.get(element);
      if (frequency > maxFrequency) {
        maxFrequency = frequency;
      }
    }

    for (int element : frequencyMap.keySet()) {
      int frequency = frequencyMap.get(element);
      if (frequency == maxFrequency) {
        possibleMaxValues.add(element);
      }
    }

    // Print the number of possible maximum values.
    System.out.println(possibleMaxValues.size());

    // Print the possible maximum values in increasing order.
    for (int possibleMaxValue : possibleMaxValues) {
      System.out.print(possibleMaxValue + " ");
    }
    System.out.println();
  }
}
