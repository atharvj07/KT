import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int k = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    int l = 1;
    int r = 1;
    int ans = 1;
    int last = -1;
    for (l = 0; l < n; l++) {
      if (a[l] == last) {
        continue;
      }
      if (last != -1) {
        ans = Math.max(ans, r - l);
      }
      r = l + 1;
      last = a[l];
    }
    if (last != -1) {
      ans = Math.max(ans, r - l);
    }
    System.out.println(ans);
  }
}
