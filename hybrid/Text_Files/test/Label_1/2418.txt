#include <iostream>
using namespace std;
const int mod = 1e9 + 7;

int power(int a, int b) {
    if (b == 0)
        return 1;
    if (b == 1)
        return a;
    if (b % 2 == 0)
        return power((a * 1LL * a) % mod, b / 2) % mod;
    else
        return (a * 1LL * power(a, b - 1)) % mod;
}

int main() {
    int x, y;
    cin >> x >> y;
    if (y % x != 0) {
        cout << "0\n";
        return 0;
    }
    cout << power(2, (y / x) - 1) << "\n";
    return 0;
}
