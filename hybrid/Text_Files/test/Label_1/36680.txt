n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
for i in range(n):
    for j in range(n):
        if a[i][j] != 1:
            found = False
            for k in range(n):
                if a[i][k] != 1 and a[i][j] % a[i][k] == 0:
                    for l in range(n):
                        if a[l][j] != 1 and a[i][j] % a[l][j] == 0:
                            found = True
                            break
                if found:
                    break
            if not found:
                print('No')
                exit()
print('Yes')
