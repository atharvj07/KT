import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int N = Integer.parseInt(br.readLine());
        List<Block> blocks = new ArrayList<>();
        for (int i = 0; i < N; i++) {
            String[] line = br.readLine().split(" ");
            int w = Integer.parseInt(line[0]);
            int s = Integer.parseInt(line[1]);
            int v = Integer.parseInt(line[2]);
            blocks.add(new Block(w, s, v));
        }

        long[][] dp = new long[N + 1][10001];
        for (int i = 0; i <= N; i++) {
            for (int j = 0; j <= 10000; j++) {
                dp[i][j] = -1;
            }
        }

        dp[0][0] = 0;
        for (int i = 0; i < N; i++) {
            for (int j = 0; j <= 10000; j++) {
                if (dp[i][j] == -1) {
                    continue;
                }

                dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j]);

                if (j + blocks.get(i).w <= 10000 && blocks.get(i).s >= j + blocks.get(i).w) {
                    dp[i + 1][j + blocks.get(i).w] = Math.max(dp[i + 1][j + blocks.get(i).w], dp[i][j] + blocks.get(i).v);
                }
            }
        }

        long max = 0;
        for (int i = 0; i <= 10000; i++) {
            max = Math.max(max, dp[N][i]);
        }

        System.out.println(max);
    }

    static class Block {
        int w;
        int s;
        int v;

        public Block(int w, int s, int v) {
            this.w = w;
            this.s = s;
            this.v = v;
        }
    }
}
