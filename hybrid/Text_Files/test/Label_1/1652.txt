#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n + 1);
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  vector<int> vis(n + 1, 0);
  int cur = 1, day = 0;
  while (cur <= n) {
    if (!vis[cur]) {
      day++;
      int nxt = n + 1;
      while (cur <= n) {
        vis[cur] = 1;
        nxt = min(nxt, a[cur]);
        cur++;
      }
      cur = nxt;
    } else {
      cur++;
    }
  }
  cout << day << endl;
  return 0;
}
