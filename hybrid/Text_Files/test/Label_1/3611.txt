#include <bits/stdc++.h>
#define LL long long
using namespace std;

const int N = 20, mod = 1e9 + 7;
int n, m, q, cy, id; string s[N], t[N], name;

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < m; i++) cin >> t[i];
    cin >> q; while (q--) {
        int y; cin >> y; cy = (y - 1) % (n * m); id = cy / m;
        name = s[id] + t[(cy + m) % m]; cout << name << endl;
    }
    return 0;
}
