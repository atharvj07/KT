import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n + 1];
    int[] b = new int[n + 1];
    for (int i = 1; i <= n; i++) {
      a[i] = sc.nextInt();
      b[a[i]] = i;
    }
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (a[i] != i) {
        ans++;
        int j = b[i];
        int tmp = a[i];
        a[i] = a[j];
        a[j] = tmp;
        b[a[i]] = i;
        b[a[j]] = j;
      }
    }
    System.out.println(ans);
  }
}
