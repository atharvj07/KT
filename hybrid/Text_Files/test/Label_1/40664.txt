q = int(input())
for _ in range(q):
    n = int(input())
    candies = list(map(int, input().split()))
    counts = {}
    for candy in candies:
        if candy in counts:
            counts[candy] += 1
        else:
            counts[candy] = 1
    counts = sorted(counts.values(), reverse=True)
    max_gift = counts[0]
    for i in range(1, len(counts)):
        if counts[i] >= counts[i-1]:
            counts[i] = max(0, counts[i-1]-1)
        max_gift += counts[i]
    print(max_gift)
