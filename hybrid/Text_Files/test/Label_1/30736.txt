import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long m = sc.nextLong();
        long a = 0;
        if (m <= n / 2) {
            a = m + 1;
        } else {
            a = n - m;
        }
        System.out.println(a);
    }
}
