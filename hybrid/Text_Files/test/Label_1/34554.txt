import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] stamina1 = new int[k];
        for (int i = 0; i < k; i++) {
            stamina1[i] = scanner.nextInt();
        }
        int[] stamina2 = new int[n * m - k];
        for (int i = 0; i < n * m - k; i++) {
            stamina2[i] = scanner.nextInt();
        }
        Arrays.sort(stamina1);
        Arrays.sort(stamina2);
        boolean possible = true;
        for (int i = 0; i < k && possible; i++) {
            possible &= stamina1[i] >= n + m - 1;
        }
        for (int i = 0; i < n * m - k && possible; i++) {
            possible &= stamina2[i] >= n + m - 1;
        }
        System.out.println(possible ? "YES" : "NO");
    }
}
