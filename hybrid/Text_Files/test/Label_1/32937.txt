import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt();
        int W = sc.nextInt();
        int d = sc.nextInt();
        char[][] grid = new char[H][W];
        int[] dx = {0, 1, 0, -1};
        int[] dy = {1, 0, -1, 0};
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (i % 2 == 0) {
                    for (int k = 0; k < d; k++) {
                        if (j + k * dx[0] >= 0 && j + k * dx[0] < W && i + k * dy[0] >= 0 && i + k * dy[0] < H) {
                            grid[i + k * dy[0]][j + k * dx[0]] = 'R';
                        }
                    }
                } else {
                    for (int k = 0; k < d; k++) {
                        if (j + k * dx[1] >= 0 && j + k * dx[1] < W && i + k * dy[1] >= 0 && i + k * dy[1] < H) {
                            grid[i + k * dy[1]][j + k * dx[1]] = 'R';
                        }
                    }
                }
            }
        }
        int base = W;
        if (W > base) {
            base = W;
        }
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (grid[i][j] == 'R') {
                    continue;
                }
                int num = (i + j) % base;
                if (num == 0 || num == base / 2) {
                    grid[i][j] = 'Y';
                } else {
                    grid[i][j] = 'G';
                }
            }
        }
        for (char[] chars : grid) {
            for (char c : chars) {
                System.out.print(c);
            }
            System.out.println();
        }
    }
}
