import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        int b = input.nextInt();
        int s = input.nextInt();
        int x = Math.abs(a);
        int y = Math.abs(b);
        int distance = x + y;
        if (s < distance || s % 2 != distance % 2) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
        }
    }
}
