#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
int main()
{
    int n; cin >> n;
    string s; cin >> s;
    vector<int> a(26), dp(n+1, 0), msf(n+1, 0);
    for(int i = 0; i < 26; i++) cin >> a[i];
    dp[0] = 1;
    for(int i = 0; i < n; i++)
    {
        msf[i+1] = msf[i];
        for(int j = i, cnt = 0; j >= 0; j--)
        {
            cnt += (s[j] == 'a' + i);
            if(cnt <= a[i])
            {
                (dp[i+1] += dp[j]) %= mod;
                msf[i+1] = max(msf[i+1], i-j+1);
            }
            else break;
        }
    }
    cout << dp[n] << endl;
    cout << msf[n] << endl;
    for(int i = 0; i < n; i++)
    {
        if(dp[i] > 0)
        {
            cout << i+1 << endl;
            break;
        }
    }
}
