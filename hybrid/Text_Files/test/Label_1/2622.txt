#include <bits/stdc++.h>
using namespace std;
int main() {
  string s, t;
  cin >> s >> t;
  int cnt[2] = {0, 0};
  for (char c : s) cnt[c - '0']++;
  int cnt_match = 0;
  int l = 0, r = 0;
  while (r < s.size()) {
    cnt[s[r] - '0']--;
    cnt_match += cnt[t[r % t.size()] - '0'];
    while (cnt[s[l] - '0'] < cnt[t[l % t.size()] - '0']) {
      cnt[s[l] - '0']++;
      cnt_match--;
      l++;
    }
    r++;
    if (cnt_match == t.size()) {
      s.replace(l, r - l, t);
      l += t.size() - 1;
      r = l + 1;
      cnt_match = 0;
    }
  }
  cout << s << endl;
  return 0;
}
