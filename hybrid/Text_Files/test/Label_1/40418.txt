h, w = map(int, input().split())
arr = [['.' for j in range(w)] for i in range(h)]
for i in range(h):
    arr[i] = list(input())
c = 0
for i in range(h):
    for j in range(w):
        if arr[i][j] == '*':
            if i == 0 or i == h - 1 or j == 0 or j == w - 1:
                c += 1
            else:
                if arr[i - 1][j] == '.' or arr[i + 1][j] == '.' or arr[i][j - 1] == '.' or arr[i][j + 1] == '.':
                    c += 1
if c == 1:
    print('YES')
else:
    print('NO')
