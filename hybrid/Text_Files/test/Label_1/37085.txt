def lca(a,b):
    while b!=0:
        a,b = b,a%b
    return a

def solve(a):
    global tot,res,flag
    n=sum(a)
    if n==1:
        res+=1
        return 0
    l=len(a)
    d=lca(2,n-1)
    if d>l: return -1
    if a[d]==1:
        flag=1
        return 0
    dp=[[0]*2 for _ in range(n+1)]
    dl=[[0]*2 for _ in range(n+1)]
    dr=[[0]*2 for _ in range(n+1)]
    dp[0][0]=1;
    dl[0][0]=1;
    dr[0][0]=1;
    cnt=0
    for i in range(d,l):
        cnt+=a[i]
        if cnt>n-cnt:
            if n%2==1:
                dp[d][0]+=1
                dp[d][1]+=1
                dl[d][1]+=cnt-1
                dr[d][1]+=n-cnt-1
            else:
                dl[d][0]+=cnt-1
                dr[d][0]+=n-cnt-1
        else:
            if n%2==1:
                dl[d][0]+=cnt-1
                dr[d][0]+=n-cnt-1
            else:
                dp[d][0]+=1
                dp[d][1]+=1
                dl[d][1]+=cnt-1
                dr[d][1]+=n-cnt-1
    dp[d][0]+=1;
    dl[d][0]+=cnt;
    dr[d][0]+=n-cnt;
    dl[d][1]+=cnt-1;
    dr[d][1]+=n-cnt-1;
    res+=dp[d][0];
    res+=dp[d][1];
    if res>1:
        if dp[d][0]>1 or dp[d][1]>1:
            if dp[d][0]>1:
                dl[d][0]-=1
            else:
                dr[d][1]-=1
        dl[d][0]+=1
        dl[d][1]+=1
        if d-1 > 0:
            dl[d][0]+=dl[d-1][0]
            dl[d][1]+=dl[d-1][1]
        tot.append(dl[d])
        dr[d][0]+=1
        dr[d][1]+=1
        if d-1 > 0:
            dr[d][0]+=dr[d-1][0]
            dr[d][1]+=dr[d-1][1]
        tot.append(dr[d])
    else:
        dl[d][0]+=dl[d-1][0]
        dl[d][1]+=dl[d-1][1]
        tot.append(dl[d])

        dr[d][0]+=dr[d-1][0]
        dr[d][1]+=dr[d-1][1]
        tot.append(dr[d])
    return 0

res=0
tot=[]
flag=0
h=int(input())
a=list(map(int,input().split()))
solve(a)
if res>1:
    print("ambiguous")
    print(*tot[0])
    print(*tot[1])
else:
    print("perfect")
