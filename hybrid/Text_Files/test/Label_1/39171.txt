import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n,t = map(int,input().split())
    l1,r1 = map(int,input().split())
    l2,r2 = map(int,input().split())
    cur = r1-l1+r2-l2
    if cur>=t:
        print(0)
    else:
        ans = (t-cur+1)//2
        if (t-cur+1)%2!=0:
            ans+=1
        print(ans)
