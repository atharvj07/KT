import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.next();
        String t = input.next();
        if(s.compareTo(t) <= 0) {
            System.out.println("-1");
            return;
        }
        int[] count = new int[26];
        for(char c: s.toCharArray()) {
            count[c - 'a']++;
        }
        StringBuilder sb = new StringBuilder();
        for(int i = 0; i < 26; i++) {
            while(count[i] > 0) {
                count[i]--;
                if(i + 'a' > t.charAt(0)) {
                    sb.append((char)(i + 'a'));
                }
                else if(sb.length() == 0) {
                    sb.append((char)(i + 'a'));
                }
            }
        }
        for(int i = 0; i < 26; i++) {
            while(count[i] > 0) {
                count[i]--;
                sb.append((char)(i + 'a'));
            }
        }
        System.out.println(sb.toString());
    }
}
