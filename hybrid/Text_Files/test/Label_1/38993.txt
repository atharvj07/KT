import sys

def solve():
    n = int(sys.stdin.readline())
    a = [list(map(int, sys.stdin.readline().split())) for j in range(3)]
    b = [list(map(int, sys.stdin.readline().split())) for i in range(n - 1)]
    d = {j: [] for j in range(1, n + 1)}
    for i, j in b:
        d[i].append(j)
        d[j].append(i)
    vis = [False] * (n + 1)
    res = 0
    ans = []
    def dfs(u, p, t):
        nonlocal res, ans
        res += a[t - 1][u - 1]
        ans.append(t)
        vis[u] = True
        for v in d[u]:
            if p != v and not vis[v]:
                for j in range(1, 4):
                    if j != t and j != p:
                        dfs(v, u, j)
                        return

    if not dfs(1, 0, 1):
        print(-1)
    else:
        print(res)
        print(*ans)

solve()
