#include<bits/stdc++.h>
using namespace std;
int n,a[101],b[101],ans;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=n;i++)cin>>b[i];
    for(int i=1;i<=n;i++){
        if(a[i]>b[i]){
            ans=max(ans,2);
            continue;
        }
        if(a[i]<b[i])return puts("-1"),0;
        if(!a[i]&&!b[i])continue;
        ans=max(ans,3);
    }
    cout<<ans<<endl;
}
