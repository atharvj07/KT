from collections import defaultdict, deque
n, k, d = map(int, input().split())
ps = list(map(int, input().split()))
g = defaultdict(list)
for u, v in [map(int, input().split()) for _ in range(n-1)]:
    g[u].append(v)
    g[v].append(u)
vis = [False] * (n+1)
shut = set()
for i in ps: vis[i] = True
q = deque([i for i in ps])
while q:
    u = q.popleft()
    for v in g[u]:
        if not vis[v]:
            if len(shut) == n-k-1: break
            if len(q) == n-k-1: break
            vis[v] = True
            shut.add(u*n+v if u>v else v*n+u)
            q.append(v)
print(len(shut))
print(*shut)
