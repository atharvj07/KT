import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int a = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }
        scanner.close();
        int b = -1;
        for (int i = 0; i < n; i++) {
            if (c[i] != a && (b == -1 || c[i] < b)) {
                b = c[i];
            }
        }
        System.out.println(b);
    }
}
