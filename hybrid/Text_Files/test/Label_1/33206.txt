import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        int min = n;
        for (int i = 0; i < m; i++) {
            if (n - a[i] < 0) {
                System.out.println(-1);
                return;
            }
            n -= a[i];
            min = Math.min(min, n);
        }
        System.out.println(min);
    }
}
