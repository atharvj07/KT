import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    static long INF = (long) 1e15;
    static void dijkstra(int n, int m, int[][] g, Pair[] s, long[] d) {
        PriorityQueue<Pair> q = new PriorityQueue<>((a, b) -> (int) (a.second - b.second));
        Arrays.fill(d, INF);
        for (Pair p : s) q.add(p);
        d[0] = 0;
        while (!q.isEmpty()) {
            Pair p = q.poll();
            if (p.second > d[p.first]) continue;
            for (int i : g[p.first]) {
                if (d[i] > d[p.first] + 1) {
                    d[i] = d[p.first] + 1;
                    q.add(new Pair(i, d[i]));
                }
            }
        }
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        Pair[] s = new Pair[n + m];
        long[] d = new long[n + m];
        for (int i = 0; i < n; i++) {
            s[i] = new Pair(i, in.nextInt());
            s[i].second += in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            s[n + i] = new Pair(n + i, in.nextInt());
            s[n + i].second += in.nextInt();
        }
        int[][] g = new int[n + m][];
        for (int i = 0; i < n + m; i++) {
            g[i] = new int[n + m];
            for (int j = 0; j < n + m; j++) {
                if (i == j) continue;
                long dx = (long) s[i].first - s[j].first;
                long dy = (long) s[i].second - s[j].second;
                if (dx >= 0 && dy >= 0 && dx * dx + dy * dy <= 4) {
                    g[i][j] = 1;
                }
            }
        }
        dijkstra(n + m, 1, g, s, d);
        System.out.println(d[0]);
    }
    static class Pair {
        int first;
        long second;
        Pair(int a, long b) {
            first = a;
            second = b;
        }
    }
}
