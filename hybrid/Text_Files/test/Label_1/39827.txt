import math
import sys

def min_hours(n, heights):
    dp = [[float('inf')] * (n + 1) for _ in range(n + 1)]
    dp[0][0] = 0
    
    for i in range(1, n + 1):
        dp[i][0] = 0
        for j in range(1, min(i + 1, (n // 2) + 1)):
            dp[i][j] = dp[i - 1][j]
            if i > 1:
                cost = max(0, heights[i - 2] - heights[i - 1] + 1)
                dp[i][j] = min(dp[i][j], dp[i - 2][j - 1] + cost)
            if i > 2:
                cost = max(0, min(heights[i - 2], heights[i - 3] - 1) - heights[i - 1] + 1)
                dp[i][j] = min(dp[i][j], dp[i - 3][j - 1] + cost)

    res = [None] * ((n + 1) // 2)
    for j in range(1, (n + 1) // 2 + 1):
        res[j - 1] = min(dp[i][j] for i in range(n + 1))
    return res

n = int(input().strip())
heights = list(map(int, input().strip().split()))
print(*min_hours(n, heights))
