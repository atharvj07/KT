import java.io.*;
import java.util.*;

class Main {
  final static long MOD = 1000000007L;

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), m = in.nextInt();
    LinkedList<Integer>[] edges = new LinkedList[n + 1];
    for (int i = 1; i <= n; i++) { edges[i] = new LinkedList<>(); }
    int[] deg = new int[n + 1];
    for (int i = 0; i < m; i++) {
      int u = in.nextInt();
      int v = in.nextInt();
      edges[u].add(v);
      deg[u] += 1;
    }
    boolean[] vis = new boolean[n + 1];
    int count = 0;
    for (int i = 1; i <= n; i++) {
      if (vis[i]) { continue; }
      count += dfs(i, vis, edges, deg);
    }
    System.out.println(count);
  }

  static int dfs(int i, boolean[] vis, LinkedList<Integer>[] edges, int[] deg) {
    if (vis[i]) { return 0; }
    vis[i] = true;
    int result = 1;
    for (int j : edges[i]) { deg[j] -= 1; }
    for (int j : edges[i]) {
      if (deg[j] == 0) { result += dfs(j, vis, edges, deg); }
    }
    result %= MOD;
    return result;
  }
}
