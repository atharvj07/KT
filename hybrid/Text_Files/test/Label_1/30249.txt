import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] c1 = new int[n + 1];
        int[] c2 = new int[n + 1];
        int[] c3 = new int[n + 1];
        for (int i = 1; i <= n; i++) c1[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) c2[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) c3[i] = sc.nextInt();
        List<Integer>[] graph = new List[n + 1];
        for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            graph[u].add(v);
            graph[v].add(u);
        }
        int[] col = new int[n + 1];
        if (!dfs(1, 0, graph, c1, c2, c3, col)) {
            System.out.println(-1);
            return;
        }
        int cost = 0;
        for (int i = 1; i <= n; i++) cost += Math.min(c1[i], Math.min(c2[i], c3[i]));
        System.out.println(cost);
        for (int i = 1; i <= n; i++) System.out.print(col[i] + " ");
    }

    static boolean dfs(int u, int p, List<Integer>[] graph, int[] c1, int[] c2, int[] c3, int[] col) {
        col[u] = 1;
        int[] cnt = new int[4];
        cnt[col[u]] = 1;
        for (int v : graph[u]) {
            if (v == p) continue;
            if (!dfs(v, u, graph, c1, c2, c3, col)) return false;
            cnt[col[v]]++;
        }
        if (cnt[1] > 1 && cnt[2] > 1 && cnt[3] > 1) return false;
        if (cnt[1] == 0 || cnt[2] == 0 || cnt[3] == 0) return false;
        int minCost = c1[u];
        if (c2[u] < minCost) {
            minCost = c2[u];
            col[u] = 2;
        }
        if (c3[u] < minCost) col[u] = 3;
        return true;
    }
}
