import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int s = sc.nextInt();
        int t = sc.nextInt();
        int u = sc.nextInt();
        int n = sc.nextInt();
        int o = sc.nextInt();
        int d = sc.nextInt();

        int[] e = new int[t];
        int[] c = new int[t];
        for (int i = 0; i < t; i++) {
            e[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }

        int[] h = new int[n];
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
        }

        int[][] dp = new int[d + 1][s + 1];
        for (int i = 0; i < dp.length; i++) {
            Arrays.fill(dp[i], -1);
        }
        dp[0][s] = 0;

        for (int day = 1; day <= d; day++) {
            for (int power = 0; power <= s; power++) {
                if (dp[day - 1][power] == -1) continue;

                // donut
                for (int i = 0; i < n; i++) {
                    if (power >= a[i]) {
                        dp[day][power - a[i]] = Math.max(dp[day][power - a[i]], dp[day - 1][power] + h[i]);
                    }
                }

                // training
                for (int i = 0; i < t; i++) {
                    int powerRequired = e[i];
                    int caloryConsumed = c[i];

                    if (power >= powerRequired && (power - powerRequired) + o < s) {
                        dp[day][(power - powerRequired) + o] = Math.max(dp[day][(power - powerRequired) + o], dp[day - 1][power]);
                    }

                    if (u > 1) {
                        for (int j = i + 1; j < t; j++) {
                            powerRequired = e[i] + e[j];
                            caloryConsumed = c[i] + c[j];

                            if (power >= powerRequired && (power - powerRequired) + o < s) {
                                dp[day][(power - powerRequired) + o] = Math.max(dp[day][(power - powerRequired) + o], dp[day - 1][power] + h[i] + h[j]);
                            }
                        }
                    }
                }
            }
        }

        int maxHappiness = 0;
        for (int i = 0; i <= s; i++) {
            maxHappiness = Math.max(maxHappiness, dp[d][i]);
        }

        if (maxHappiness == 0) {
            System.out.println(-1);
        } else {
            System.out.println(maxHappiness);
        }
    }
}
