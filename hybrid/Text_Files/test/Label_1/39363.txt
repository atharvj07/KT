t = int(input())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    sorted_a = sorted(a)
    positions = {val: i for i, val in enumerate(a)}
    sorted_positions = [positions[sorted_a[i]] for i in range(n)]
    count = 1
    for i in range(1, n):
        if sorted_positions[i] != sorted_positions[i-1] + 1:
            count += 1
    if count <= k:
        print("Yes")
    else:
        print("No")
