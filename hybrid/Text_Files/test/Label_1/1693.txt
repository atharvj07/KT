#include <bits/stdc++.h>
#include <vector>
using namespace std;
typedef long long ll;
const ll MOD = 1e9 + 7;

ll n, x;
vector<tuple<ll, ll, ll>> v;

int main() {
    cin >> n >> x;
    v.resize(n);
    for (ll i = 0; i < n; i++) {
        ll l, r, c;
        cin >> l >> r >> c;
        v[i] = make_tuple(l, r, c);
    }
    sort(v.begin(), v.end());
    ll ans = -1;
    for (ll i = 0; i < n; i++) {
        ll l1, r1, c1;
        tie(l1, r1, c1) = v[i];
        for (ll j = i + 1; j < n; j++) {
            ll l2, r2, c2;
            tie(l2, r2, c2) = v[j];
            if (r1 < l2 || r2 < l1) {
                if (l1 + r2 - l2 - r1 == x) {
                    if (ans == -1) ans = c1 + c2;
                    else ans = min(ans, c1 + c2);
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
