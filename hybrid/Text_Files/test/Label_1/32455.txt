import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[][] a = new char[4][4];
        for (int i = 0; i < 4; i++) {
            a[i] = in.nextLine().toCharArray();
        }
        boolean win = false;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (a[i][j] == '.') {
                    a[i][j] = 'x';
                    if (check(a)) {
                        win = true;
                    }
                    a[i][j] = '.';
                }
            }
        }
        System.out.println(win ? "YES" : "NO");
    }

    static boolean check(char[][] a) {
        for (int i = 0; i < 4; i++) {
            if (a[i][0] == 'x' && a[i][1] == 'x' && a[i][2] == 'x' || a[i][1] == 'x' && a[i][2] == 'x' && a[i][3] == 'x') {
                return true;
            }
        }
        for (int i = 0; i < 4; i++) {
            if (a[0][i] == 'x' && a[1][i] == 'x' && a[2][i] == 'x' || a[1][i] == 'x' && a[2][i] == 'x' && a[3][i] == 'x') {
                return true;
            }
        }
        if (a[0][0] == 'x' && a[1][1] == 'x' && a[2][2] == 'x' || a[1][1] == 'x' && a[2][2] == 'x' && a[3][3] == 'x') {
            return true;
        }
        if (a[0][3] == 'x' && a[1][2] == 'x' && a[2][1] == 'x' || a[1][2] == 'x' && a[2][1] == 'x' && a[3][0] == 'x') {
            return true;
        }
        return false;
    }
}
