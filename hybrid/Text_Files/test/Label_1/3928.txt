#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100+2;

int a[MAXN], ans[MAXN];

int main() {
    int n, m, x, y;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> m;

    while (m--) {
        cin >> x >> y;
        a[x] -= y;
        ans[x-1] += y;
        ans[x+1] += y;
    }

    for (int i = 1; i <= n; i++) {
        a[i] += ans[i];
        cout << a[i] << endl;
    }

    return 0;
}
