typedef long double ld;

int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<ld, ld>> a(n);
    for (int i = 0; i < n; i++) {
        int w, c;
        cin >> w >> c;
        a[i] = {double(w) / double(c), double(c)};
    }
    sort(a.rbegin(), a.rend());
    ld ans = 0;
    for (auto p : a) {
        if (m >= p.first) {
            ans += p.second;
            m -= p.first;
        } else {
            ans += m * p.second / p.first;
            m = 0;
            break;
        }
    }
    cout << ans;
    return 0;
}
