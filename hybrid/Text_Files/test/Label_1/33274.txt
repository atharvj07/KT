import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt(), m = s.nextInt();
        int[][] e = new int[n][];
        for (int i = 0; i < n; i++) {
            e[i] = new int[s.nextInt()];
            for (int j = 0; j < e[i].length; j++) e[i][j] = s.nextInt();
        }
        int[] l = new int[m + 1];
        int ans = 0;
        for (int i = 0; i < n; i++) for (int j : e[i]) l[j]++;
        for (int i = 1; i <= m; i++) if (l[i] == 0) ans++;
        System.out.println(ans);
    }
}
