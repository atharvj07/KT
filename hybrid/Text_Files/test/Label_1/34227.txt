import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        boolean found = false;
        for (int i = 0; i < n - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                int[] inc = new int[i + 1];
                for (int j = 0; j <= i; j++) {
                    inc[j] = arr[j];
                }
                int[] dec = new int[n - i - 1];
                for (int j = i + 1; j < n; j++) {
                    dec[j - i - 1] = arr[j];
                }
                System.out.println("YES");
                System.out.println(i + 1);
                for (int j = 0; j <= i; j++) {
                    System.out.print(inc[j] + " ");
                }
                System.out.println();
                System.out.println(n - i - 1);
                for (int j = n - i - 2; j >= 0; j--) {
                    System.out.print(dec[j] + " ");
                }
                System.out.println();
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("NO");
        }
    }
}
