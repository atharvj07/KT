def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

t = int(input())
for _ in range(t):
    n = int(input())
    if n == 1:
        print("Bob")
    elif n == 2 or is_prime(n):
        print("Alice")
    elif n % 2 == 0:
        print("Alice")
    else:
        if is_prime(n-2):
            print("Bob")
        else:
            print("Alice")
