import decimal
for _ in range(int(input())):
    p, q, b = map(int, input().split())
    decimal.getcontext().prec = 100000
    d = decimal.Decimal(p) / decimal.Decimal(q)
    res = str(d)
    if '.' not in res:
        res += '.0'
    if res.count('0', res.find('.')) == len(res) - res.find('.') - 1:
        print('Finite')
    else:
        print('Infinite')
