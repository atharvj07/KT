#include <bits/stdc++.h>

#define endl '\n'
#define ll long long

using namespace std;

int32_t main() {
    ll t;
    cin >> t;
    while (t--) {
        ll n;
        cin >> n;
        ll arr[n];
        for (ll i = 0; i < n; ++i) {
            cin >> arr[i];
        }
        ll ans = 0;
        for (ll i = 1; i < n; ++i) {
            if (arr[i] % arr[i - 1]) {
                ans++;
            }
        }
        if (arr[n - 1] % arr[0]) {
            ans++;
        }
        cout << ans << endl;
    }
}
