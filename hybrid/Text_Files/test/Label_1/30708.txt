import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        dp[0] = 1;
        int max = 1;
        int end = 0;
        for (int i = 1; i < n; i++) {
            dp[i] = 1;
            for (int j = 0; j < i; j++) {
                if (a[j] < a[i] && dp[i] < dp[j] + 1) {
                    dp[i] = dp[j] + 1;
                    if (dp[i] > max) {
                        max = dp[i];
                        end = i;
                    }
                }
            }
        }
        System.out.println(max);
        int[] ans = new int[max];
        int j = end;
        for (int i = max - 1; i >= 0; i--) {
            ans[i] = a[j];
            j--;
        }
        for (int i = 0; i < max; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
