import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class JohnnyBlogs {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Integer>[] graph = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            graph[a].add(b);
            graph[b].add(a);
        }

        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = scanner.nextInt();
        }

        int[] order = new int[n + 1];
        boolean[] visited = new boolean[n + 1];
        Queue<Integer> queue = new LinkedList<>();
        queue.add(1);
        visited[1] = true;
        order[1] = 1;

        for (int i = 2; i <= n; i++) {
            order[i] = -1;
        }

        while (!queue.isEmpty()) {
            int currentBlog = queue.poll();
            for (int neighbor : graph[currentBlog]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    order[neighbor] = desiredTopics[currentBlog];
                    queue.add(neighbor);
                } else if (order[neighbor] != -1 && order[neighbor] != desiredTopics[currentBlog]) {
                    System.out.println(-1);
                    return;
                }
            }
        }

        for (int i = 1; i <= n; i++) {
            if (order[i] == -1) {
                System.out.println(-1);
                return;
            }
        }

        for (int i = 1; i <= n; i++) {
            System.out.print(order[i] + " ");
        }
    }
}
