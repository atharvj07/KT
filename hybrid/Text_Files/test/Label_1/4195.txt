#include <iostream>
#include <string>

using namespace std;

int main() {
    string s;
    cin >> s;
    int ans = 1;
    for (int i = 0; i + 2 < s.length(); i += 2)
        if (s[i + 2] == 'A' || s[i + 2] == 'E' || s[i + 2] == 'I' || s[i + 2] == 'O' || s[i + 2] == 'U' || s[i + 2] == 'Y')
            ans = max(ans, i + 3);
    cout << ans << endl;
    return 0;
}
