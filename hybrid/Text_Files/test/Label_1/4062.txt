#include <iostream>
#include <vector>

using namespace std;

int main() {
  string good;
  cin >> good;
  string pattern;
  cin >> pattern;
  int n;
  cin >> n;
  vector<string> queries(n);
  for (int i = 0; i < n; ++i) { cin >> queries[i]; }

  for (const string& query : queries) {
    bool match = true;
    int q = 0, p = 0;
    int asterisk = -1;
    while (q < query.size() && p < pattern.size()) {
      if (pattern[p] == '*') {
        asterisk = p;
        ++p;
        while (p < pattern.size() && pattern[p] == '*') { ++p; }
      } else if (pattern[p] == '?' || pattern[p] == query[q]) {
        ++p;
        ++q;
      } else if (asterisk != -1) {
        q = query.find(pattern[p], q);
        if (q == string::npos) {
          match = false;
          break;
        }
        ++q;
      } else {
        match = false;
        break;
      }
    }
    if (q < query.size() || p < pattern.size() && asterisk == -1) {
      match = false;
    }
    cout << (match ? "YES" : "NO") << endl;
  }
  return 0;
}
