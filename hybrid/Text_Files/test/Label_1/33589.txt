import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            String[] s = br.readLine().split(" ");
            BigInteger k = new BigInteger(s[0]);
            BigInteger l = new BigInteger(s[1]);
            BigInteger r = new BigInteger(s[2]);
            BigInteger p = new BigInteger(s[3]);
            BigInteger lcm = BigInteger.ONE;
            for (BigInteger i = l; i.compareTo(r.add(BigInteger.ONE)) <= 0; i = i.add(BigInteger.ONE)) {
                lcm = lcm.multiply(k.pow(i.intValue()).add(BigInteger.ONE));
            }
            System.out.println(lcm.mod(p));
        }
    }
}
