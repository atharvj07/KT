import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        int xor = 0;
        for (int x : a) {
            xor ^= x;
        }
        for (int y : b) {
            xor ^= y;
        }
        if (xor == 0) {
            System.out.println("Koyomi");
        } else {
            System.out.println("Karen");
        }
    }
}
