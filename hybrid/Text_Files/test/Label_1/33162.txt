import java.util.*;
import java.math.*;

class Main {
  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    int N = cin.nextInt();
    BigInteger[] t = new BigInteger[N];
    BigInteger[] a = new BigInteger[N];

    BigInteger gcd;
    for (int i = 0; i < N; i++) {
      t[i] = BigInteger.valueOf(cin.nextInt());
      a[i] = BigInteger.valueOf(cin.nextInt());
      gcd = t[i].gcd(a[i]);
      t[i] = t[i].divide(gcd);
      a[i] = a[i].divide(gcd);
    }

    BigInteger ans = BigInteger.ONE;
    for (int i = 0; i < N; i++) ans = ans.multiply(t[i]).divide(ans.gcd(t[i]));
    for (int i = 0; i < N; i++) ans = ans.multiply(a[i]).divide(ans.gcd(a[i]));
    System.out.println(ans);
  }
}
