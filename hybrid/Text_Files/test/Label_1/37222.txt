n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
dp = [[[10**18] * 3 for _ in range(k+1)] for _ in range(n+1)]
dp[0][0][0] = 0
for i in range(n):
    for j in range(k+1):
        dp[i+1][j][0] = min(dp[i+1][j][0], dp[i][j][0])
        dp[i+1][j][1] = min(dp[i+1][j][1], dp[i][j][0]+a[i])
        dp[i+1][j][2] = min(dp[i+1][j][2], dp[i][j][1]+b[i])
        if j+1 <= k:
            dp[i+1][j+1][0] = min(dp[i+1][j+1][0], dp[i][j][2])
print(dp[n][k][0])
