#include <bits/stdc++.h>
using namespace std;
int a[200000], b[200000], f[200000], s[200000]; //f=increase, s=decrease count
int main() {
  int t, n, l, r, ans, add, minus, val;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    l = max_element(a, a + n) - a;
    r = max_element(b, b + n) - b;
    add = 0;
    minus = 0;
    val = 0;
    ans = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (a[i] >= b[i]) {
        val = b[i];
        f[i] = add;
        s[i] = minus;
        add++;
      } else {
        val = a[i];
        f[i] = add;
        s[i] = minus;
        minus++;
      }
    }
    for (int i = 0; i < n; i++) {
      if (l == i) ans += f[i];
      if (r == i) ans += s[i];
      if (l == i && r == i) ans--;
    }
    ans += abs(val - b[r]);
    cout << ans << "\n";
  }
  return 0;
}
