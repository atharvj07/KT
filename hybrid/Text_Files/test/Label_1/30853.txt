import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class TaxiFee {

    static class Edge {
        int u, v, w;

        public Edge(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }

    static class Query {
        int t, x;

        public Query(int t, int x) {
            this.t = t;
            this.x = x;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int q = scanner.nextInt();

        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            int w = scanner.nextInt();
            edges.add(new Edge(u, v, w));
        }

        List<Query> queries = new ArrayList<>();
        for (int i = 0; i < q; i++) {
            int t = scanner.nextInt() - 1;
            int x = scanner.nextInt();
            queries.add(new Query(t, x));
        }

        int[] distances = dijkstra(n, edges);

        for (Query query : queries) {
            int t = query.t;
            int x = query.x;

            edges.get(t).w = x;
            int[] newDistances = dijkstra(n, edges);
            System.out.println(newDistances[n - 1]);
            edges.get(t).w = x; // Restore original weight
        }
    }

    private static int[] dijkstra(int n, List<Edge> edges) {
        int[] distances = new int[n];
        for (int i = 0; i < n; i++) {
            distances[i] = Integer.MAX_VALUE;
        }
        distances[0] = 0;

        boolean[] visited = new boolean[n];

        for (int i = 0; i < n; i++) {
            int u = -1;
            for (int j = 0; j < n; j++) {
                if (!visited[j] && (u == -1 || distances[j] < distances[u])) {
                    u = j;
                }
            }

            visited[u] = true;

            for (Edge edge : edges) {
                if (edge.u == u) {
                    distances[edge.v] = Math.min(distances[edge.v], distances[edge.u] + edge.w);
                } else if (edge.v == u) {
                    distances[edge.u] = Math.min(distances[edge.u], distances[edge.v] + edge.w);
                }
            }
        }

        return distances;
    }
}
