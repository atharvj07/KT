import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int r = in.nextInt();
        int n = in.nextInt();
        int[] cells = new int[n];
        for (int i = 0; i < n; i++) {
            cells[i] = in.nextInt() * 3 + in.nextInt();
        }
        in.close();
        if ((r & 1) == (n & 1)) {
            System.out.println("WIN");
        } else {
            boolean[] vis = new boolean[r * 3];
            for (int c : cells) {
                vis[c - 1] = true;
                vis[c] = true;
                vis[c + 1] = true;
            }
            boolean turn = false;
            for (int i = 1; i <= r; i++) {
                for (int j = 1; j <= 2; j++) {
                    int c = i * 3 + j;
                    if (!vis[c] && !vis[c - 1] && !vis[c + 1]) {
                        vis[c - 1] = true;
                        vis[c] = true;
                        vis[c + 1] = true;
                        turn = !turn;
                    }
                }
            }
            if (turn) {
                System.out.println("WIN");
            } else {
                System.out.println("LOSE");
            }
        }
    }
}
