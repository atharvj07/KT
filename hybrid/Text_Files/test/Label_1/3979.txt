#include <cstdio>
#include <vector>
#include <queue>

using namespace std;

const int N = 1e5 + 5;

int n, m, c[N], deg[N], ans, dq;
vector<int> e[N];
queue<int> q;

int main() {
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; i++) {
    scanf("%d", &c[i]);
    deg[c[i]]++;
  }
  for (int i = 1; i <= m; i++) {
    int a, b;
    scanf("%d%d", &a, &b);
    e[a].push_back(b);
    e[b].push_back(a);
  }
  for (int i = 1; i <= n; i++) {
    if (deg[c[i]] > 1 && deg[c[i]] > deg[ans]) {
      ans = c[i];
      dq = deg[c[i]];
    }
  }
  for (int i = 1; i <= n; i++) {
    if (dq == deg[c[i]] && c[i] < ans) {
      ans = c[i];
    }
  }
  for (int i = 1; i <= n; i++) {
    if (c[i] == ans) {
      for (int j : e[i]) {
        if (c[j] != c[i]) q.push(c[j]);
      }
      break;
    }
  }
  while (!q.empty()) {
    int x = q.front(); q.pop();
    if (deg[x] > 1 && deg[x] > deg[ans]) {
      ans = x;
      dq = deg[x];
    }
  }
  printf("%d\n", ans);
  return 0;
}
