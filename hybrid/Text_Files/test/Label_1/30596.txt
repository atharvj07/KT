import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] > 4) {
                ans += 1;
                s[i] -= 4;
            }
            if (s[i] > 0) {
                ans += 1;
            }
        }
        System.out.println(ans);
    }
}
