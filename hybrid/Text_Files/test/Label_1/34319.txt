import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] perm = new int[n];
        for (int i = 0; i < n; i++) {
            perm[i] = in.nextInt();
        }

        int[] dp = new int[n];
        Map<Integer, Integer> map = new HashMap<>();
        map.put(perm[0], 0);
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            int num = perm[i];
            dp[i] = dp[map.get(num - 1)] + 1;
            map.put(num, i);
        }

        for (int i = 0; i < n; i++) {
            System.out.print(dp[i] + " ");
        }
        System.out.println();
    }
}
