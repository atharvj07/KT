#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
#define fi first
#define se second
#define pb push_back

ll n, ans;
vector<ll> a;

int main(){
	cin>>n;
	a.resize(n+1);
	for(int i=1; i<=n; i++)
		cin>>a[i];
	for(int i=n; i>=2; i--)
		if(a[i]==1){
			if(a[i-1]==2)
				ans++;
			else
				ans+=2;
		}
		else if(a[i]==2){
			if(a[i-1]==3)
				ans++;
			else
				ans+=2;
		}
	if(ans)	cout<<"Finite\n"<<ans<<endl;
	else	cout<<"Infinite"<<endl;
    return 0;
}
