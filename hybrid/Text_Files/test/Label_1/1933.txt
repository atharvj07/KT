#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const int maxN = 1e5 + 5;
int n, x;
int cnt[maxN];
ll ans;

int main() {
    cin >> n >> x;
    for(int i = 0, a; i < n; i++) {
        cin >> a;
        ans += cnt[a ^ x];
        cnt[a]++;
    }
    cout << ans << "\n";
    return 0;
}
