import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        int ans = 0;
        int last = 0;
        for (int i = 0; i < n; i++) {
            if (t[i] - last >= 15) {
                ans += t[i] - last;
            }
            last = t[i];
        }
        ans += 90 - last;
        System.out.println(ans);
    }
}
