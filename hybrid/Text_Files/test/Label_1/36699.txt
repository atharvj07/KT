import sys
input = sys.stdin.readline

n,m = map(int,input().split())
G = [[] for _ in range(n+1)]
for _ in range(m):
    a,b = map(int,input().split())
    G[a].append(b)
    G[b].append(a)

ok = True
for u in range(1,n):
    for v in G[u]:
        if u > v: continue
        for w in G[v]:
            if u < w < v:
                ok = False

if ok: print(0)
else: print(1)
