#include <iostream>
#include <vector>

void solve() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) { std::cin >> a[i]; }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        int need = std::max(k, a[i] + (i > 0 ? 1 : 0));
        if (a[i] < need) {
            ans += need - a[i];
            a[i] = need;
        } else if (i > 0 && a[i - 1] < k) {
            int extra = std::max(0, need - a[i - 1]);
            a[i] += extra;
            a[i - 1] += extra;
            ans += extra;
        }
    }
    std::cout << ans << '\n';
    for (int i = 0; i < n; i++) { std::cout << a[i] << ' '; }
}

int main() { std::cout.sync_with_stdio(false);solve();return 0; }
