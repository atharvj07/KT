from operator import mul

def solve():
    import sys
    input = sys.stdin.readline
    mod = 10**9 + 7

    def get_sum(arr):
        length = len(arr)
        if length == 1:
            return arr[0]
        max_num = arr[0]
        min_num = arr[0]
        for i in range(1, length):
            if arr[i] > max_num:
                max_num = arr[i]
            if arr[i] < min_num:
                min_num = arr[i]
        return (max_num - min_num) * length % mod
    
    n = int(input())
    arr = list(map(int, input().split()))
    arr.sort()
    print(get_sum(arr))

solve()
