import numpy as np

n, a, b = map(int, input().split())
if n > a * b:
    print(-1)
else:
    assignment = np.zeros((a, b), dtype=np.int)
    for i in range(0, a):
        for j in range(0, b):
            if n > 0:
                if j % 2 == i % 2:
                    assignment[i][j] = n
                    n -= 1

    for i in range(0, a):
        print(*assignment[i])
