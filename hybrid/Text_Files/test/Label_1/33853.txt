import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int i = sc.nextInt();
        int j = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int[] dx = {a, a, -a, -a};
        int[] dy = {b, -b, b, -b};
        int[][] dis = new int[n + 1][m + 1];
        for (int[] row : dis) {
            for (int i1 = 0; i1 < m + 1; i1++) {
                row[i1] = -1;
            }
        }
        Queue<int[]> q = new LinkedList<>();
        q.add(new int[]{i, j});
        boolean impossible = true;
        int ans = 0;
        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int x = cur[0];
            int y = cur[1];
            dis[x][y] = ans;
            for (int k = 0; k < 4; k++) {
                int nx = x + dx[k];
                int ny = y + dy[k];
                if (nx >= 1 && nx <= n && ny >= 1 && ny <= m && dis[nx][ny] == -1) {
                    if (nx == 1 || nx == n || ny == 1 || ny == m) {
                        impossible = false;
                        q.clear();
                        break;
                    }
                    q.add(new int[]{nx, ny});
                }
            }
            ans++;
        }
        if (impossible) {
            System.out.println("Poor Inna and pony!");
        } else {
            System.out.println(ans - 1);
        }
    }
}
