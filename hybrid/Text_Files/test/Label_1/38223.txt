from collections import Counter
def solve():
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        c = Counter(a)
        ans = []
        col = 1
        for i in range(1, n + 1):
            if c[i] > k - len(ans):
                ans.append(0)
            else:
                ans.append(col)
                col += 1
                if col > k:
                    col = 1
        print(*ans)

solve()
