#include <cstdio>
const int maxn=11;
char a[maxn][maxn];
int n,m,ans,ok;
inline int mx(int x,int y){return x>y?x:y;}
int main(){
    int i,j;
    while(scanf("%d%d",&n,&m)!=EOF){
        ok=1, ans=0;
        for(i=1;i<=n;i++){scanf("%s",a[i]+1);for(j=1;j<=m;j++)if(a[i][j]=='S')ok=0;}
        for(i=1;i<=n;i++){for(j=1;j<=m;j++)if(a[i][j]=='.') ans=mx(ans,n);}
        for(i=1;i<=m;i++){for(j=1;j<=n;j++)if(a[j][i]=='.') ans=mx(ans,m);}
        if(ok) ans=-1;
        printf("%d\n",ans);
    }
    return 0;
}
