#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    ll n, a, b, p, q, ans = 0;
    cin >> n >> a >> b >> p >> q;
    ll lcm_ab = (a * b) / __gcd(a, b);
    ans += ((n / a + n / b - n / lcm_ab) * p);
    ans += ((n / lcm_ab) * min(p, q));
    cout << ans << endl;
    return 0;
}
