
import sys

EPS = 1e-6

def cross(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

def vec(a, b):
    return [b[i] - a[i] for i in range(2)]

def inside_polygon(polygon, point):
    cnt = 0
    for i in range(len(polygon)):
        j = (i + 1) % len(polygon)
        if (polygon[i][1] <= point[1] < polygon[j][1] or
            polygon[j][1] <= point[1] < polygon[i][1]):
            if cross(polygon[i], polygon[j], point) > 0:
                cnt ^= 1
    return cnt != 0

speed, width, bus_speed, ped_speed = map(int, sys.stdin.readline().split())
bus = [list(map(int, sys.stdin.readline().split())) for _ in range(speed)]
time = width / ped_speed
sx, sy = bus[0]
for x, y in bus[1:]:
    max_time = (sx - x) / (bus_speed - ped_speed)
    if inside_polygon(bus, [sx + max_time * bus_speed, sy + max_time * ped_speed]):
        time = min(time, max_time + EPS)
    sx, sy = x, y
print("%.9f" % time)
