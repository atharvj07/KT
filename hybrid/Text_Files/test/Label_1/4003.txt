#include <iostream>
#include <vector>
#include <algorithm>
#define ll long long
using namespace std;
int main() {
  int n, k; cin >> n >> k;
  vector<ll> a(n), b(k);
  for (ll &x : a) cin >> x;
  for (ll &x : b) cin >> x;
  sort(b.rbegin(), b.rend());
  int t = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) {
      if (!t) t = b[0], b.erase(b.begin());
      else if (a[i - 1] > t) t = b[0], b.erase(b.begin());
      a[i] = t;
    }
  }
  vector<ll> c = a;
  sort(c.begin(), c.end());
  for (int i = 0; i < n; i++) if (c[i] != a[i]) {
    cout << "Yes\n";
    return 0;
  }
  cout << "No\n";
  return 0;
}
