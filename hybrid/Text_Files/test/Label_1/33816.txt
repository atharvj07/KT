import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[][] lr = new int[n][2];
    for (int i = 0; i < n; i++) {
      lr[i][0] = sc.nextInt();
      lr[i][1] = sc.nextInt();
    }
    sc.close();

    Arrays.sort(lr, (a, b) -> a[0] - b[0]);
    int ans = 0;
    int rMax = 0;
    for (int i = 0; i < n; i++) {
      if (lr[i][0] >= rMax) ans++;
      rMax = Math.max(rMax, lr[i][1]);
    }

    System.out.println(ans);
  }
}
