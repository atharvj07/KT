import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int x = input.nextInt();
        int sum = 0;
        int distressed = 0;
        for (int i = 0; i < n; i++) {
            char op = input.next().charAt(0);
            int d = input.nextInt();
            if (op == '+') {
                sum += d;
            } else {
                if (sum >= d) {
                    sum -= d;
                } else {
                    distressed++;
                }
            }
        }
        System.out.println(sum + " " + distressed);
    }
}
