import sys
input = sys.stdin.readline

n = int(input())
li = list(map(int, input().split()))
mod = 10**9+7

g = []

for i in range(n+1):
    g.append([])    

vis = [False]*(n+1)
res = 0

for i in range(1, n+1):
    if not vis[i]:
        cnt = 0
        q = [i]
        g[i].append(i)
        vis[i] = True
        while q:
            v = q.pop()
            cnt += 1
            for j in g[v]:
                if j==i:
                    res += pow(2, cnt-1, mod)
                if not vis[j]:
                    q.append(j)
                    g[j].append(i)
                    vis[j] = True
print(res)
