#include <iostream>
#include <string>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    string a, b;
    cin >> n >> k >> a >> b;
    bool possible = true;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) {
        if (b[i] != a[i] + 1 || (k == 1 && a[i] != 'z')) {
          possible = false;
          break;
        }
        k--;
      }
    }
    cout << (possible ? "YES\n" : "NO\n");
  }
  return 0;
}
