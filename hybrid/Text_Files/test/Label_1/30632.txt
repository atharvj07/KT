import java.util.Scanner;
import java.util.Set;
import java.util.TreeSet;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();

        Set<Integer> remainingUtensils = new TreeSet<>();
        for (int i = 0; i < n; i++) {
            remainingUtensils.add(input.nextInt());
        }

        System.out.println(Math.max(0, k * remainingUtensils.size() - remainingUtensils.stream().reduce(Integer::max).orElse(0)));
    }
}
