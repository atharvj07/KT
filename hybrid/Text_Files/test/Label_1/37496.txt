n, m = map(int,input().split())
prv = [0 for i in range (n+1)]
tmp = [0 for i in range (n+1)]
for i in range (m):
	tmp1, tmp2, tmp3 = map(int,input().split())
	prv[tmp2] = m + 1
	for j in range (tmp1, tmp2):
		tmp[j] = i+1
		prv[j+tmp3+1] = i+1
prv1 = []
tmp1 = 0
for i in range (1, n+1):
	if tmp1 == tmp[i]:
		continue
	tmp1 = tmp[i]
	prv1.append(tmp[i])
if len(set(prv)) == n+1:
	for i in range (1, n+1):
		print(prv[i], end=' ')
else:
	print(-1)
