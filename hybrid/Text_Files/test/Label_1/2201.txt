#include <iostream>
using namespace std;

long long gcd(long long a, long long b) {
    if (b == 0) return a;
    return gcd(b, a % b);
}

int main() {
    long long v;
    cin >> v;
    for (long long i = 1; i < v; i++) {
        if (gcd(v, i) == 1) {
            cout << v - i << endl;
            return 0;
        }
    }
    cout << 1 << endl;
    return 0;
}
