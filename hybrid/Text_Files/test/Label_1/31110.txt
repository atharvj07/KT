import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        int t = sc.nextInt();
        Set<Integer> q = new HashSet<>();
        for (int i = 0; i < t; i++) {
            q.add(sc.nextInt());
        }
        boolean[] used = new boolean[n];
        int[] x = new int[n];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (q.contains(i + 1)) {
                x[i] = -p[i];
                cnt++;
                used[i] = true;
            }
        }
        if (cnt == t) {
            System.out.println("YES");
            for (int i = 0; i < n; i++) {
                if (!used[i]) {
                    x[i] = p[i];
                }
            }
            for (int i : x) {
                System.out.print(i + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }
}
