#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll mod = 1e9 + 7;

int main() {
  ll n1, n2, n3;
  cin >> n1 >> n2 >> n3;
  vector<ll> v1(n1), v2(n2), v3(n3);
  for (int i = 0; i < n1; i++) cin >> v1[i];
  for (int i = 0; i < n2; i++) cin >> v2[i];
  for (int i = 0; i < n3; i++) cin >> v3[i];
  ll ans = 0;
  sort(v1.rbegin(), v1.rend());
  sort(v2.rbegin(), v2.rend());
  sort(v3.rbegin(), v3.rend());
  while (!v2.empty() && !v3.empty()) {
    if (v2.back() < v3.back()) {
      ans += v2.back();
      v2.pop_back();
    } else {
      ans += v3.back();
      v3.pop_back();
    }
  }
  while (!v1.empty() && !v2.empty()) {
    if (v1.back() < v2.back()) {
      ans += v1.back();
      v1.pop_back();
    } else {
      ans += v2.back();
      v2.pop_back();
    }
  }
  while (!v1.empty() && !v3.empty()) {
    if (v1.back() < v3.back()) {
      ans += v1.back();
      v1.pop_back();
    } else {
      ans += v3.back();
      v3.pop_back();
    }
  }
  cout << ans << "\n";
  return 0;
}
