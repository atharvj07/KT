import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Map<Integer, Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new HashMap<>());
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            Map<Integer, Integer> map = graph.get(u);
            if (!map.containsKey(v)) {
                map.put(v, 0);
            }
            map.put(v, Math.max(map.get(v), w));
        }
        int start = sc.nextInt();
        long[] dp = new long[n + 1];
        Arrays.fill(dp, -1);
        System.out.println(solve(graph, start, dp));
    }

    public static long solve(List<Map<Integer, Integer>> graph, int start, long[] dp) {
        if (dp[start] != -1) {
            return dp[start];
        }
        long max = 0;
        Map<Integer, Integer> map = graph.get(start);
        for (int v : map.keySet()) {
            max = Math.max(max, map.get(v) + solve(graph, v, dp));
        }
        return dp[start] = max;
    }
}
