import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int[] cnt = new int[1000001];
        for (int i = 0; i < N; i++) {
            cnt[a[i]]++;
            if (cnt[a[i]] > a[i]) {
                ans += cnt[a[i]] - a[i];
                cnt[a[i]] = a[i];
            }
        }
        System.out.println(ans);
    }
}
