import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt(), t = in.nextInt(), r = in.nextInt();
        int[] w = new int[m + 1];
        for (int i = 1; i <= m; i++) w[i] = in.nextInt();

        int ans = 0;
        int cnt = 0;
        for (int i = 1; i <= m; i++) {
            if (w[i] - 1 - cnt > 0) ans += (w[i] - 1 - cnt + t - 1) / t;
            cnt = Math.max(w[i], cnt + t);
        }

        if (cnt + t - 1 >= w[m] + 1) System.out.println(ans);
        else System.out.println(-1);
    }
}
