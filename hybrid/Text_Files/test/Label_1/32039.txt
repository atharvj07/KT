import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Auction {

    private static class Bid implements Comparable<Bid> {
        int participant;
        int value;

        public Bid(int participant, int value) {
            this.participant = participant;
            this.value = value;
        }

        @Override
        public int compareTo(Bid other) {
            return Integer.compare(value, other.value);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<Bid> bids = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            bids.add(new Bid(scanner.nextInt(), scanner.nextInt()));
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            List<Integer> absent = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                absent.add(scanner.nextInt());
            }
            Bid winnerBid = findWinner(bids, absent);
            if (winnerBid != null) {
                System.out.println(winnerBid.participant + " " + winnerBid.value);
            } else {
                System.out.println("0 0");
            }
        }
    }

    private static Bid findWinner(List<Bid> bids, List<Integer> absent) {
        Collections.sort(bids);
        int lastParticipant = 0;
        for (int i = 0; i < bids.size(); i++) {
            Bid bid = bids.get(i);
            if (!absent.contains(bid.participant) && bid.participant != lastParticipant) {
                lastParticipant = bid.participant;
                return bid;
            }
        }
        return null;
    }
}
