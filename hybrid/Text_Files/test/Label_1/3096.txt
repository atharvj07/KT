#include <iostream>
#include <vector>
#include <queue>
using namespace std;

#define INF 1e9

struct edge {
  int to, w;
  edge(int a, int b): to(a), w(b) {}
};

vector<vector<edge>> g;
int n, m, a, b;
int ans[71];

int dijkstra(int st, int ed) {
  int d[n+1];
  for (int i = 1; i <= n; i++) d[i] = INF;
  priority_queue<pair<int,int>> que;
  que.push(make_pair(0, st));
  d[st] = 0;
  while (!que.empty()) {
    int cost = -que.top().first;
    int now = que.top().second;
    que.pop();
    if (cost > d[now]) continue;
    if (now == ed) return cost;
    for (edge e: g[now]) {
      int next = e.to;
      int ncost = cost + e.w;
      if (d[next] > ncost) {
        d[next] = ncost;
        que.push(make_pair(-ncost, next));
      }
    }
  }
  return -1;
}

void dfs(int now, int pre, int c) {
  for (edge e: g[now]) {
    int next = e.to;
    int nc = c + e.w;
    if (next == pre) continue;
    if (ans[next] > nc) dfs(next, now, nc);
  }
}

int main() {
  cin >> n >> m >> a >> b;
  g.resize(n+1);
  while (m--) {
    int u, v, w;
    cin >> u >> v >> w;
    g[u].push_back(edge(v, w));
    g[v].push_back(edge(u, w));
  }
  for (int i = 2; i <= n; i++) {
    ans[i] = dijkstra(1, i);
  }
  dfs(1, 1, 0);
  for (int i = 1; i <= n; i++) cout << ans[i] << " ";
  cout << '\n';
  return 0;
}
