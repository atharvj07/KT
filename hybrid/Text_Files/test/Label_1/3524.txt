#include <iostream>

const int MOD = 1e9 + 7;

int popcnt(int x) {
    int cnt = 0;
    while (x) {
        cnt += x % 2;
        x /= 2;
    }
    return cnt;
}

int32_t main() {
    int n;
    std::cin >> n;
    int x = 0;
    for (int i = 0; i < n; i++) {
        int d;
        std::cin >> d;
        x = x * 2 + d;
    }
    int res = 0;
    for (int i = 0; i < n; i++) {
        int pow = 1 << i;
        int inv = x ^ pow;
        if (inv > pow) {
            res = (res + 1ll * popcnt(x & pow) * popcnt(inv - pow)) % MOD;
        }
    }
    std::cout << res << std::endl;
}
