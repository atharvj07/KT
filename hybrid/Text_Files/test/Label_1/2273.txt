#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i], b[i] = a[i];
    sort(b.begin(), b.end());
    int cnt = 0;
    for (int i = 0; i < n; i++)
        if (a[i] != b[i]) {
            for (int j = i + 1; j < n; j++)
                if (a[j] == b[i]) {
                    swap(a[i], a[j]);
                    cout << i << ' ' << j << endl;
                    cnt++;
                    break;
                }
        }
    cout << cnt << "\n";
    return 0;
}
