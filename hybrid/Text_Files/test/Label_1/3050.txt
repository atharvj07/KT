#include <iostream>
#include <climits>

int main() {
  long long n, k;
  std::cin >> n >> k;
  long long p = 10, pp = p;
  while (k != 0) {
    if (k % 2 == 0) {
      p *= p;
      k /= 2;
    }
    else {
      pp *= p;
      k--;
    }
  }
  long long cnt = 0, ans = LLONG_MAX;
  while (n > 0) {
    long long t = n % p;
    n /= p;
    if (t == 0) continue;
    cnt++;
    ans = std::min(ans, cnt + (pp - t) / 10);
  }
  if (ans == LLONG_MAX)
    ans = 0;
  std::cout << ans << std::endl;
  return 0;
}
