import java.util.Scanner;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int r = in.nextInt();
        int n = in.nextInt();
        HashSet<Integer> reclaimed = new HashSet<>();
        for (int i = 0; i < n; i++) {
            int ri = in.nextInt();
            int ci = in.nextInt();
            reclaimed.add(ri * 3 + ci);
            reclaimed.add((ri - 1) * 3 + (3 - ci));
            reclaimed.add(ri * 3 + (3 - ci));
            reclaimed.add((ri + 1) * 3 + (3 - ci));
        }
        boolean win = true;
        for (int i = 1; i <= r; i++) {
            for (int j = 1; j <= 2; j++) {
                if (!reclaimed.contains(i * 3 + j)) {
                    reclaimed.add(i * 3 + j);
                    reclaimed.add((i - 1) * 3 + (3 - j));
                    reclaimed.add(i * 3 + (3 - j));
                    reclaimed.add((i + 1) * 3 + (3 - j));
                    win = !win;
                }
            }
        }
        System.out.println(win ? "WIN" : "LOSE");
    }
}
