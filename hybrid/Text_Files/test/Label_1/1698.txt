#include<iostream>
#include<string>

using namespace std;

string password;
bool flag[4] = {false, false, false, false};

int main() {
    getline(cin, password);

    for (int i = 0; i < password.size(); i++) {
        if (password[i] >= 'a' && password[i] <= 'z') {
            flag[0] = true;
        } else if (password[i] >= 'A' && password[i] <= 'Z') {
            flag[1] = true;
        } else if (password[i] >= '0' && password[i] <= '9')  {
            flag[2] = true;
        }
    }
    if (flag[0] == true && flag[1] == true && flag[2] == true && password.size() >= 5) {
        cout << "Correct" << endl;
    } else {
        cout << "Too weak" << endl;
    }

    return 0;
}
