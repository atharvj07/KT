import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();

    Point[] vertices = new Point[n];
    for (int i = 0; i < n; i++) {
      vertices[i] = new Point(scanner.nextInt(), scanner.nextInt());
    }

    ConvexPolygon polygon = new ConvexPolygon(vertices);

    double d = polygon.getMaxExpansionDistance();
    System.out.println(d);
  }

  static class Point {
    int x, y;

    public Point(int x, int y) {
      this.x = x;
      this.y = y;
    }
  }

  static class ConvexPolygon {
    Point[] vertices;
    int n;

    public ConvexPolygon(Point[] vertices) {
      this.vertices = vertices;
      this.n = vertices.length;
    }

    // Returns the maximum distance by which the polygon can be expanded without intersecting itself.
    double getMaxExpansionDistance() {
      double maxDistance = 0;
      for (int i = 0; i < n; i++) {
        Point p1 = vertices[i], p2 = vertices[(i + 1) % n], p3 = vertices[(i + 2) % n];

        // Calculate the area of the triangle formed by p1, p2, and p3.
        double area = Math.abs(p1.x * (p2.y - p3.y) + p2.x * (p3.y - p1.y) + p3.x * (p1.y - p2.y)) / 2;

        // Calculate the distance from p2 to the line through p1 and p3.
        double distance = area / (Math.hypot(p2.x - p1.x, p2.y - p1.y) + Math.hypot(p2.x - p3.x, p2.y - p3.y));

        // Update the maximum distance.
        maxDistance = Math.max(maxDistance, distance);
      }

      return maxDistance;
    }
  }
}
