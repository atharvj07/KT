#include <iostream>
#include <numeric>
#include <vector>
using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cout << (long long)((double)accumulate(a.begin(), a.end(), 0LL) / n + 0.5) << endl;
  }
  return 0;
}
