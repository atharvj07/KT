import sys
input = sys.stdin.readline

n, m = map(int, input().split())
p = list(map(int, input().split()))
c = list(map(int, input().split()))
d = int(input())

for i in range(d):
    l = list(map(int, input().split()))
    for j in l:
        p[j - 1] = -1

    cl = {}
    for i in range(n):
        if p[i] != -1:
            if c[i] not in cl:
                cl[c[i]] = [p[i]]
            else:
                cl[c[i]].append(p[i])

    ans = []
    for i in cl.values():
        i.sort()
        j = 0
        while j < len(i):
            if j == 0 or i[j] != i[j - 1]:
                ans.append(i[j])
                j += 1
            else:
                j += 1

    ans.sort()
    if len(ans) == 0:
        print(0)
    else:
        print(ans[0])
