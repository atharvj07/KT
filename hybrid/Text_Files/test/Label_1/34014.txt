import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] count = new int[4][2];
        while (sc.hasNext()) {
            double left = sc.nextDouble();
            double right = sc.nextDouble();
            if (1.1 <= left) count[0][0]++;
            else if (0.6 <= left) count[1][0]++;
            else if (0.2 <= left) count[2][0]++;
            else count[3][0]++;
            if (1.1 <= right) count[0][1]++;
            else if (0.6 <= right) count[1][1]++;
            else if (0.2 <= right) count[2][1]++;
            else count[3][1]++;
        }
        for (int i = 0; i < 4; i++) {
            System.out.println(count[i][0] + " " + count[i][1]);
        }
    }
}
