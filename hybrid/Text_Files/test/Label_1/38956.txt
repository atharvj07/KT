import sys

n, h, m, k = map(int, input().split())
t = []
for i in range(n):
    t.append(tuple(map(int, input().split())))

t.sort()
ans = []
l = 0
r = 0
for i in range(1, m, 2):
    while l < n and t[l][0] * m + t[l][1] <= i - k:
        l += 1
    while r < n and t[r][0] * m + t[r][1] <= i:
        r += 1
    if l >= n:
        continue
    if m * t[l][0] + t[l][1] >= i:
        ans = [t[l][0], t[l][1]]
        break
    if l < r:
        if t[r][1] >= k:
            ans = [t[r][0], t[r][1] - k]
            break
        else:
            ans = [t[l][0], t[l][1]]
            break

print(r - l, ans[1])
for i in range(l, r):
    print(t[i][0], t[i][1])
