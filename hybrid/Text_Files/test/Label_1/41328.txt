import math

def scalar_product(x, y):
    return sum(x[i] * y[i] for i in range(5))

def vector_length(x):
    return math.sqrt(scalar_product(x, x))

def is_acute_angle(a, b, c):
    ab = [b[i] - a[i] for i in range(5)]
    ac = [c[i] - a[i] for i in range(5)]
    cos_angle = scalar_product(ab, ac) / (vector_length(ab) * vector_length(ac))
    return cos_angle > 0

points = [list(map(int, input().split())) for _ in range(n)]
good_points = []

for i in range(n):
    is_good = True
    for j in range(n):
        for k in range(j + 1, n):

                    is_good = False
                    break
        if not is_good:
            break
    if is_good:
        good_points.append(i + 1)
