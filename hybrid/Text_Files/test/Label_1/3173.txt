#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    map<int, int> m;
    for (int i = 0; i < n; i++) {
        int a;
        cin >> a;
        m[a]++;
    }
    int ans = INT_MAX;
    for (auto &p : m) {
        ans = min(ans, p.first ^ (++p.second) / 2);
    }
    cout << ans << endl;
}
