import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] x = new int[n], y = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = s.nextInt();
            y[i] = s.nextInt();
        }
        Arrays.sort(x);
        boolean[] g = new boolean[n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (g[i]) continue;
            int miny = y[i], maxy = y[i];
            for (int j = i + 1; j < n; j++) {
                if (g[j]) continue;
                if (x[j] == x[i]) continue;
                g[i] = g[j] = true;
                miny = Math.min(miny, y[j]);
                maxy = Math.max(maxy, y[j]);
                int a = x[i], b = x[j];
                int c = -a * a - b * b;
                if (c < miny || c > maxy) ans++;
            }
        }
        System.out.println(ans);
    }
}
