import java.util.*;
import java.io.*;

public class Main {
    static final int MOD = 1000000007;

    static int pow(long base, int exp) {
        if (exp == 0) return 1;
        if (exp == 1) return (int)base;
        if (exp % 2 == 0) {
            int mid = pow(base, exp / 2);
            return (int)((long)mid * (long)mid % MOD);
        }
        return (int)((long)pow(base, exp - 1) * (long)base % MOD);
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        long m = Long.parseLong(br.readLine());
        int ans = 0;
        for (long i = 4; i <= m; i += 7) {
            int cnt = 0;
            while (i > 0) {
                if (i % 10 == 4 || i % 10 == 7) {
                    cnt++;
                }
                i /= 10;
            }
            ans += (pow(6, cnt) - pow(5, cnt) + MOD) % MOD;
            ans %= MOD;
        }
        System.out.println(ans);
    }
}
