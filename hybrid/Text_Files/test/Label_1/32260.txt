import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<Integer> l = new ArrayList<>();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            int a = in.nextInt();
            l.add(a);
            arr[i] = a;
        }
        Collections.sort(l);
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == l.get(i)) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
