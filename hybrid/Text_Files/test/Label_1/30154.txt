import java.math.BigInteger;
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    private static int n, m, k;
    private static List<Integer>[] adj;
    private static int[] vis;
    private static int[][] memo;
    private static final int MOD = 1000000000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        vis = new int[n + 1];
        memo = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                memo[i][j] = -1;
            }
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[u].add(v);
            adj[v].add(u);
        }
        System.out.println(solve(1, m));
    }

    private static int solve(int u, int t) {
        if (t == 0) {
            return 1;
        }
        if (memo[u][t] != -1) {
            return memo[u][t];
        }
        int ans = 0;
        vis[u] = 1;
        for (int v : adj[u]) {
            if (vis[v] == 0) {
                ans = (ans + solve(v, t - 1)) % MOD;
            }
        }
        vis[u] = 0;
        return memo[u][t] = ans;
    }
}
