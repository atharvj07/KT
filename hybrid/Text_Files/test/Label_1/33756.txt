import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int k = sc.nextInt();
        Set<Integer> favs = new HashSet<>();
        for (int i = 0; i < k; i++) {
            favs.add(sc.nextInt());
        }
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            String title = sc.next();
            int d = sc.nextInt();
            int numFavs = 0;
            for (int j = 0; j < d; j++) {
                int id = sc.nextInt();
                if (favs.contains(id)) {
                    numFavs++;
                }
            }
            if (numFavs == k) {
                System.out.println(0);
            } else if (numFavs == 0) {
                System.out.println(1);
            } else {
                System.out.println(2);
            }
        }
    }
}
