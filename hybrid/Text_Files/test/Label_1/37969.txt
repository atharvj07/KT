def max_citizens_caught(n, p, a):
    children = [[] for _ in range(n)]
    for i in range(1, n):
        children[p[i-1]-1].append(i)

    def dfs(node):
        max_catch = a[node]
        sums = []
        for child in children[node]:
            child_catch = dfs(child)
            sums.append(child_catch)
        sums.sort(reverse=True)
        for i, s in enumerate(sums, 1):
            max_catch = max(max_catch, s + i)
        return max_catch

    return dfs(0)

n = int(input().strip())
p = list(map(int, input().strip().split()))
a = list(map(int, input().strip().split()))

print(max_citizens_caught(n, p, a))
