#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, p;
  cin >> n >> p;
  if (p == 1) {
    cout << n - 1 << endl;
    return 0;
  }
  for (int i = 1; i < n; i++) {
    double prob = double(i + 1) * (i + 2) / double(n * (n - 1));
    if (prob >= p) {
      cout << i << endl;
      return 0;
    }
  }
  return 0;
}
