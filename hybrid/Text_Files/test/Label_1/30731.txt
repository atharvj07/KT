import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] x = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextLong();
        }
        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            int j = (int) (Math.random() * (n - i - 1)) + 1;
            ans += x[j] - x[j - 1];
            x[j] = x[j - 1];
        }
        ans %= 1000000007;
        ans *= fact(n - 1);
        ans %= 1000000007;
        System.out.println(ans);
    }
    static long fact(int n) {
        long ans = 1;
        for (int i = 1; i <= n; i++) {
            ans *= i;
            ans %= 1000000007;
        }
        return ans;
    }
}
