n, k, m, a = map(int, input().split())
votes = [0] * n
for g in map(int, input().split()):
    votes[g - 1] += 1

sorted_votes = sorted((v, i) for i, v in enumerate(votes))
guaranteed_seats = [False] * n
remaining_votes = m - a
results = [0] * n

# Determine guaranteed seats
seats_filled = 0
for v, i in reversed(sorted_votes):
    if seats_filled < k and v + remaining_votes > sorted_votes[k - 1][0]:
        guaranteed_seats[i] = True
        seats_filled += 1
    else:
        break

# Determine possible outcomes for each candidate
for i, (v, _) in enumerate(sorted_votes):
    if guaranteed_seats[i]:
        results[i] = 1
    elif v + remaining_votes >= sorted_votes[k - 1][0] or seats_filled < k:
        results[i] = 2
    else:
        results[i] = 3

print(' '.join(map(str, results)))
