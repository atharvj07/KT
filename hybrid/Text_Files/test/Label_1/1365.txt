#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> p(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
  }
  vector<int> f(n + 1, 1e9);
  f[0] = 0;
  for (int k = 1; k <= n; k++) {
    for (int i = 0; i <= n - k; i++) {
      int cost = 0;
      if (p[i] == k) {
        cost--;
      }
      if (p[i + k - 1] == k) {
        cost++;
      }
      f[k] = min(f[k], f[k - 1] + cost);
    }
  }
  for (int k = 1; k <= n; k++) {
    cout << f[k] << " ";
  }
  cout << endl;
  return 0;
}
