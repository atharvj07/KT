import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] songs = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                songs[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                int max = -100000000;
                for (int k = 0; k <= j; k++) {
                    int sum = dp[i - 1][j - k] + songs[i - 1][0] * k + songs[i - 1][1] * (j - k) + songs[i - 1][2];
                    max = Math.max(max, sum);
                }
                dp[i][j] = max;
            }
        }
        System.out.println(dp[n][m]);
    }
}
