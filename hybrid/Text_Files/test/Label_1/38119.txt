import sys

rd = sys.stdin.readline
w = sys.stdout.write

m, n = map(int, rd().split())
a = [int(rd().replace('0', '1').replace('1', '0'), 2) for i in range(n)]
for p in range(2 << m):
    t = p
    for i in a:
        if i & p != i:
            t = 0
            break
    for i in range(m):
        if (1 << i) & t and (1 << i + 1) & t:
            t = 0
            break
    for i in a:
        if i & t != i:
            t = 0
            break
    if 1 << m - 1 & t:
        t = 0
    if t:
        print((t + 1) % (10 ** 9 + 7))
        break
