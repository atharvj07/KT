n = int(input())
p = sorted(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
m = int(input())
c = list(map(int, input().split()))

s = []
for i, j in enumerate(zip(a, b)):
    temp = (p[i], j)
    s.append(temp)

s.sort()

for x in c:
    l = 4
    r = len(s) - 1
    pos = -1
    while l <= r:
        mid = (l + r) // 2
        if s[mid][1][0] == x or s[mid][1][1] == x:
            pos = mid
            r = mid - 1
        else:
            l = mid + 1
    if pos == -1:
        print(-1)
    else:
        print(s[pos][0])
