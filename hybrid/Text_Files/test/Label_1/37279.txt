def solve_string_partition(n, p, q, s):
    for i in range(n//p + 1):
        if (n - i*p) % q == 0:
            print(i + (n - i*p) // q)
            for j in range(i):
                print(s[j*p:(j+1)*p])
            for j in range(i*p, n, q):
                print(s[j:j+q])
            return
    print(-1)

n, p, q = map(int, input().split())
s = input()
solve_string_partition(n, p, q, s)
