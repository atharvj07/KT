import sys
input=sys.stdin.readline

n=int(input())
a=[list(map(int,input().split())) for _ in range(n)]

ans=[]
for l0,r0 in a:
    for l,r in a:
        if l<=l0<=r<=r0 and l0!=l:
            ans=[a.index([l,r])+1,a.index([l0,r0])+1]
            break
    if ans:break
print(*ans if ans else [-1]*2)
