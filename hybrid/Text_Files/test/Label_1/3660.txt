#include<bits/stdc++.h>
#define fastio ios_base::sync_with_stdio(false)
#define double long double
using namespace std;
double f[55][55][55];
bool vis[55][55][55];
double solve(int pos,int guest,int p){
    if(pos==guest){
        return 0;
    }
    if(vis[pos][guest][p]==true){
        return f[pos][guest][p];
    }
    vis[pos][guest][p]=true;
    double ans=0;
    for(int i=1;i<=min(p,50);i++){
        if(pos+i<=guest){
            ans+=((i+solve(pos+i,guest,p-i))/(double)guest);
        }
    }
    return f[pos][guest][p]=ans;
}
int main(){
    int n,p;
    cin>>n>>p;
    int a[n+1];
    for(int i=1;i<=n;i++)cin>>a[i];
    double ans=solve(0,n,p);
    cout<<fixed<<setprecision(10)<<ans<<endl;
    return 0;
}
