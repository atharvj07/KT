#include <bits/stdc++.h>
#define repd(i,a,b) for(int i=a;i>=b;i--)
using namespace std;

typedef long long ll;
const ll md=1e9+7;
const int M=5e6+5;

void solve(){
    int n,m;
    cin>>n>>m;
    string s[n+1],ans="";
    map<string,int>mp;

    for(int i=1;i<=n;i++) cin>>s[i];

    for(int i=1;i<=n;i++){
        string t=s[i];
        reverse(t.begin(),t.end());
        int cnt=0;
        while(!t.empty() && mp[t]==i){
            t.pop_back();
            cnt++;
        }
        if(!t.empty()) ans+=s[i].substr(0,cnt);
        reverse(s[i].begin(),s[i].end());
        while(!s[i].empty() && mp[s[i]]==i){
            s[i].pop_back();
            cnt++;
        }
        if(!s[i].empty()) ans+=s[i].substr(0,cnt);
        mp[t]=mp[s[i]]=i;
    }

    reverse(ans.begin(),ans.end());
    cout<<ans.size()<<endl;
    if(ans.size()) cout<<ans<<endl;
}

int main(){
    solve();
    return 0;
}
