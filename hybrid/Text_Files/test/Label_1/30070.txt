import java.io.*;
import java.util.*;

public class Main {
  static final long mod = 1000000007;

  static long pow(long a, long b) {
    if (b == 0) return 1;
    long x = pow(a, b / 2);
    x = (x * x) % mod;
    if (b % 2 == 1) x = (x * a) % mod;
    return x;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(in.readLine());
    int h = Integer.parseInt(st.nextToken());
    int w = Integer.parseInt(st.nextToken());
    int[] r = new int[h];
    st = new StringTokenizer(in.readLine());
    for (int i = 0; i < h; i++) r[i] = Integer.parseInt(st.nextToken());
    int[] c = new int[w];
    st = new StringTokenizer(in.readLine());
    for (int i = 0; i < w; i++) c[i] = Integer.parseInt(st.nextToken());
    long[] fact = new long[h + w + 2];
    fact[0] = 1;
    for (int i = 1; i <= h + w + 1; i++) fact[i] = (fact[i - 1] * i) % mod;
    long[] invfact = new long[h + w + 2];
    for (int i = 0; i <= h + w + 1; i++) invfact[i] = pow(fact[i], mod - 2);
    boolean ok = true;
    for (int i = 0; i < h; i++) {
      for (int j = 0; j < w; j++) {
        if (r[i] + c[j] != h + w - 1) ok = false;
      }
    }
    if (!ok) {
      System.out.println(0);
      return;
    }
    long ans = 1;
    for (int i = 0; i < h; i++) {
      ans = (ans * invfact[r[i]]) % mod;
      ans = (ans * invfact[h + w - 1 - c[r[i]]]) % mod;
    }
    ans = (ans * fact[h + w - 1]) % mod;
    System.out.println(ans);
  }
}
