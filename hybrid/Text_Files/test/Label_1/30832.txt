import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String line;
        while ((line = in.readLine()) != null && !line.equals("0")) {
            int n = Integer.parseInt(line);
            String[] hands = new String[n];
            for (int i = 0; i < n; i++) {
                hands[i] = in.readLine();
            }
            String winner = getWinner(hands);
            System.out.println(winner);
        }
    }

    public static String getWinner(String[] hands) {
        int[] count = new int[15];
        for (String hand : hands) {
            switch (hand) {
                case "Rock":
                    count[0]++;
                    break;
                case "Fire":
                    count[1]++;
                    break;
                case "Scissors":
                    count[2]++;
                    break;
                case "Snake":
                    count[3]++;
                    break;
                case "Human":
                    count[4]++;
                    break;
                case "Tree":
                    count[5]++;
                    break;
                case "Wolf":
                    count[6]++;
                    break;
                case "Sponge":
                    count[7]++;
                    break;
                case "Paper":
                    count[8]++;
                    break;
                case "Air":
                    count[9]++;
                    break;
                case "Water":
                    count[10]++;
                    break;
                case "Dragon":
                    count[11]++;
                    break;
                case "Devil":
                    count[12]++;
                    break;
                case "Lightning":
                    count[13]++;
                    break;
                case "Gun":
                    count[14]++;
                    break;
            }
        }
        int maxCount = 0;
        int winnerIndex = -1;
        for (int i = 0; i < 15; i++) {
            if (count[i] > maxCount) {
                maxCount = count[i];
                winnerIndex = i;
            }
        }
        if (maxCount == 0) {
            return "Draw";
        } else {
            switch (winnerIndex) {
                case 0:
                    return "Rock";
                case 1:
                    return "Fire";
                case 2:
                    return "Scissors";
                case 3:
                    return "Snake";
                case 4:
                    return "Human";
                case 5:
                    return "Tree";
                case 6:
                    return "Wolf";
                case 7:
                    return "Sponge";
                case 8:
                    return "Paper";
                case 9:
                    return "Air";
                case 10:
                    return "Water";
                case 11:
                    return "Dragon";
                case 12:
                    return "Devil";
                case 13:
                    return "Lightning";
                case 14:
                    return "Gun";
            }
        }
        return null;
    }
}
