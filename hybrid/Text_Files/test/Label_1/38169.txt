n, k = map(int, input().split())
s = input()
letters = set(input().split())
ans = 0
for i in range(n):
    for j in range(i, n):
        ok = True
        for c in s[i:j + 1]:
            if c not in letters:
                ok = False
                break
        if ok:
            ans += 1
print(ans)
