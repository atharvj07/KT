import sys
t = 0
s = input()
st = [0] * 3
x = 0
g = [[-1] * 3 for i in range(3)]
d = [0] * 3
for i in range(int(s)):
    u = input()
    if u[-1] == 'B':
        t += 1
        st[1] += 1
        st[2] += 1
        d[2] += int(u[:-1]) - x
        x = int(u[:-1])
    elif u[-1] == 'R':
        t += 1
        st[0] += 1
        st[2] += 1
        d[2] += int(u[:-1]) - x
        x = int(u[:-1])
    else:
        st[2] += 1
        d[2] += int(u[:-1]) - x
        x = int(u[:-1])
        g[st[0]][st[1]] += 1
        g[st[1]][st[0]] += 1
        g[st[2]][st[1]] += 1
        g[st[1]][st[2]] += 1
        g[st[2]][st[0]] += 1
        g[st[0]][st[2]] += 1
        if g[st[0]][st[1]] > d[0]:
            d[0] = g[st[0]][st[1]]
        if g[st[1]][st[0]] > d[1]:
            d[1] = g[st[1]][st[0]]
        if g[st[2]][st[0]] > d[1]:
            d[1] = g[st[2]][st[0]]
        if g[st[0]][st[2]] > d[1]:
            d[1] = g[st[0]][st[2]]
        if g[st[2]][st[1]] > d[0]:
            d[0] = g[st[2]][st[1]]
        if g[st[1]][st[2]] > d[0]:
            d[0] = g[st[1]][st[2]]
print(d[0] + d[1] + d[2])
