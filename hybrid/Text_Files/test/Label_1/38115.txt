from collections import deque

n, q = map(int, input().split())
deq = deque(map(int, input().split()))
for i in range(q):
    m = int(input())
    for j in range(m - 1):
        a, b = deq.popleft(), deq.popleft()
        if a > b:
            deq.appendleft(b)
            deq.append(a)
        else:
            deq.appendleft(a)
            deq.append(b)
    print(*deq[:2])
