#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
int main() {
    int n, q;
    cin >> n >> q;
    string s;
    cin >> s;
    vector<int> can(n);
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') can[i] = 1;
    }
    vector<ll> pref(n + 1);
    for (int i = 1; i <= n; i++) {
        pref[i] = pref[i - 1] + can[i - 1];
    }
    while (q--) {
        int l, r;
        cin >> l >> r;
        int cnt = r - l + 1;
        ll ans = pref[r] - pref[l - 1];
        ans += ((cnt - 1LL) * (cnt)) / 2;
        ans %= mod;
        cout << ans << '\n';
    }
    return 0;
}
