import java.util.*;

// Define a class to represent a pair of integers
class Pair implements Comparable<Pair> {
    int a, b;

    public Pair(int a, int b) {
        this.a = a;
        this.b = b;
    }

    public int compareTo(Pair other) {
        if (a == other.a) {
            return b - other.b;
        } else {
            return a - other.a;
        }
    }

    public boolean equals(Object obj) {
        if (obj instanceof Pair) {
            Pair other = (Pair) obj;
            return a == other.a && b == other.b;
        } else {
            return false;
        }
    }

    public int hashCode() {
        return Objects.hash(a, b);
    }
}

// Define a class to represent a graph
class Graph {
    int n;
    List<List<Integer>> adj;

    public Graph(int n) {
        this.n = n;
        adj = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }
    }

    public void addEdge(int u, int v) {
        adj.get(u).add(v);
    }

    public boolean isConnected() {
        // Perform a depth-first search to check if the graph is connected
        Stack<Integer> stack = new Stack<>();
        Set<Integer> visited = new HashSet<>();
        stack.push(0);
        while (!stack.isEmpty()) {
            int u = stack.pop();
            visited.add(u);
            for (int v : adj.get(u)) {
                if (!visited.contains(v)) {
                    stack.push(v);
                }
            }
        }
        return visited.size() == n;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        // Find the prime factors of each number
        Map<Integer, Set<Integer>> factors = new HashMap<>();
        for (int i = 0; i < n; i++) {
            Set<Integer> pf = new HashSet<>();
            for (int j = 2; j * j <= a[i]; j++) {
                while (a[i] % j == 0) {
                    pf.add(j);
                    a[i] /= j;
                }
            }
            if (a[i] > 1) {
                pf.add(a[i]);
            }
            factors.put(i, pf);
        }

        // Construct a graph where each node represents a number and each edge represents a common prime factor
        Graph g = new Graph(n);
        for (int i = 0; i < n; i++) {
            Set<Integer> pf1 = factors.get(i);
            for (int j = i + 1; j < n; j++) {
                Set<Integer> pf2 = factors.get(j);
                if (!Collections.disjoint(pf1, pf2)) {
                    g.addEdge(i, j);
                }
            }
        }

        // Check if the graph is connected
        boolean connected = g.isConnected();

        // Output the result
        if (connected) {
            System.out.println(1);
        } else {
            System.out.println(0);
        }
    }
}
