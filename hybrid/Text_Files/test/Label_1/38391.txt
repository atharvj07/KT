n,k=map(int,input().split())
a=sorted(map(int,input().split()))
for _ in range(int(input())):
    x=int(input())
    t=0
    j=len(a)-1
    ans=-1
    for i in range(k):
        while j>=0 and a[j]>x: j-=1
        if j<0: break
        q=x//a[j]
        if q==0: break
        t+=q
        x=x-q*a[j]
        if x==0: ans=t
    print(ans)
