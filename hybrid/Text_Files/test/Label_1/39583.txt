import re
s = input()
n = int(re.findall(r" = .*$", s)[0][3:])
s = re.sub(r'[? =]', '', s)
arr = [1] * len(s)
l = n
r = 1
for i in range(len(s)):
    if s[i] in '+-':
        r = l + 1 if s[i] == '+' else r - 1
    else:
        arr[i] = r
        l = r
if sum(arr) == n:
    print("Possible")
    print(' '.join(map(str, arr)))
else:
    print("Impossible")
