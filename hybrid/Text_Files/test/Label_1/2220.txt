#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }
  sort(w.begin(), w.end());
  string s;
  cin >> s;
  set<int> pairs_occupied, singletons_occupied;
  vector<int> ans(2 * n);
  for (int i = 0; i < 2 * n; i++) {
    int p = s[i] - '0';
    if (p == 0) {  // introvert
      int pos = -1;
      for (int j = 0; j < n; j++) {
        if (pairs_occupied.count(j) == 0) {
          ans[i] = w[j];
          pairs_occupied.insert(j);
          pos = j;
          break;
        }
      }
      assert(pos != -1);
    } else {  // extrovert
      int pos = -1;
      for (int j = n - 1; j >= 0; j--) {
        if (singletons_occupied.count(j) == 0) {
          ans[i] = w[j];
          singletons_occupied.insert(j);
          pos = j;
          break;
        }
      }
      assert(pos != -1);
    }
  }
  for (auto x : ans) cout << x << " ";
  cout << endl;
}
