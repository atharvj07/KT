import java.util.*;
import java.math.*;
class Main {
    public static int solve(int m) {
        int mod = 1000000007;
        long n = (long) m * m;
        BigInteger nn = BigInteger.valueOf(n).modPow(BigInteger.valueOf(m - 7), BigInteger.valueOf(mod));
        BigInteger ans = BigInteger.ZERO.modPow(BigInteger.valueOf(m - 1), BigInteger.valueOf(mod));
        ans = ans.multiply(nn).mod(BigInteger.valueOf(mod));
        return ans.intValue();
    }
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt();
        System.out.println(solve(m));
    }
}
