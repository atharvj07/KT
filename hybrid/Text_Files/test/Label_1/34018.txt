import java.util.*;
import java.io.*;

public class Main {
    static int[] par, dep;
    static ArrayList<Integer>[] adj;
    static ArrayList<Edge> edg;
    static ArrayList<String> ans;

    static void dfs(int u, int p) {
        par[u] = p;
        for (int v : adj[u]) {
            if (v == p) continue;
            dep[v] = dep[u] + 1;
            dfs(v, u);
        }
    }

    static long[] sum;
    static int get(int a, int b) {
        return (int) ((sum[b] - sum[a] + mod) % mod);
    }

    static final long mod = (long) (1e9 + 7);

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        adj = new ArrayList[n + 1];
        par = new int[n + 1];
        dep = new int[n + 1];
        sum = new long[n + 1];
        edg = new ArrayList<>();
        for (int i = 0; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            int c = Integer.parseInt(st.nextToken());
            adj[a].add(b);
            adj[b].add(a);
            edg.add(new Edge(a, b, c));
        }
        dfs(1, 1);
        for (Edge e : edg) {
            int a = e.a;
            int b = e.b;
            if (dep[a] < dep[b]) {
                a ^= b;
                b ^= a;
                a ^= b;
            }
            sum[a] += e.w;
            sum[b] += e.w;
        }
        for (int i = 1; i <= n; i++) sum[i] %= mod;
        for (int i = 2; i <= n; i++) {
            if (get(0, i) != 0) {
                System.out.println("NO");
                return;
            }
        }
        ans = new ArrayList<>();
        for (Edge e : edg) {
            int a = e.a;
            int b = e.b;
            if (dep[a] < dep[b]) {
                a ^= b;
                b ^= a;
                a ^= b;
            }
            if (get(0, a) - get(0, b) - e.w != 0) {
                long diff = get(0, a) - get(0, b) - e.w;
                diff = (diff % mod + mod) % mod;
                ans.add(a + " " + b + " " + diff);
            }
        }
        System.out.println("YES");
        System.out.println(ans.size());
        for (String x : ans) System.out.println(x);
    }

    static class Edge {
        int a, b, w;

        public Edge(int a, int b, int w) {
            this.a = a;
            this.b = b;
            this.w = w;
        }
    }
}
