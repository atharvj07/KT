def find_sequence(n):
    # Since we need to form 1337, we can use the formula of combinations:
    # n = (x+2)*(x+1)/2, where x is the number of 3's between 1 and 3 of 1337.
    # Rearranging the formula gives us x^2 + 3x - 2n = 0, which is a quadratic equation.
    # We solve it to find x.
    
    # Coefficients for the quadratic equation ax^2 + bx + c = 0
    a = 1
    b = 3
    c = -2*n

    # Using the quadratic formula to find x: x = (-b + sqrt(b^2 - 4ac)) / 2a
    x = int(((-b + (b**2 - 4*a*c)**0.5) / (2*a)))

    # Constructing the sequence
    # Adding 1 at the beginning
    sequence = '1'

    # Adding x number of 3's
    sequence += '3' * x

    # Adding 37 at the end to complete 1337
    sequence += '37'

    # The remaining 3's are added after the initial 1 to minimize the sequence length
    # while still achieving exactly n subsequences of 1337.
    # Any extra 3's needed to reach n subsequences can be added here. 
    # However, based on the formula and approach, x is chosen to precisely achieve n subsequences,
    # so there's no need for additional characters beyond what's calculated.

    return sequence

# Reading number of queries
t = int(input())

# Processing each query
for _ in range(t):
    n = int(input())
    print(find_sequence(n))
