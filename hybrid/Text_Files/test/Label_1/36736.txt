from collections import defaultdict
n=int(input())
a=list(map(int,input().split()))
d=[defaultdict(lambda:n)for _ in range(n+2)]
t=0
for i in range(n):
    for j in range(0,i):
        d[i][a[j]]+=1
    x=a[i]
    for j in range(n):
        d[i+1][a[j]]=min(d[i+1][a[j]],d[i][x]+(j>i))
        x=a[j]
    print(d[i+1][min(d[i+1].keys())])
