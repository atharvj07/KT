#include <bits/stdc++.h>

using namespace std;

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0), cout.tie(0);
  int n, k, ans = 0;
  cin >> n >> k;
  vector<int> vec(n);
  for (auto &x : vec) cin >> x;
  sort(vec.rbegin(), vec.rend());
  for (int i = 0; i < n;) {
    int sum = 0;
    for (int j = i; j < min(n, i + 2); j++) {
      sum += vec[j];
    }
    if (sum <= k) {
      ans++;
      i += 2;
    } else {
      ans++;
      i++;
    }
  }
  cout << ans;
  return 0;
}
