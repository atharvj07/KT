n,m=[int(z)for z in input().split()]
grid=[]
for j in range(n):
    grid.append([cell=='#'for cell in input()])
xr=[0]*n;xc=[0]*m
for i in range(n):
    for j in range(m):
        if grid[i][j]:
            xr[i]+=1
            xc[j]+=1
row=all(x>0for x in xr)
col=all(x>0for x in xc)
if row and col:
    print("Yes")
else:
    print("No")
