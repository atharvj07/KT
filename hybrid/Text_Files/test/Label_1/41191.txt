n, m = map(int, input().split())
s = input()
t = input()

min_changes = n
best_pos = []

for i in range(m - n + 1):
    changes = 0
    positions = []
    for j in range(n):
        if s[j] != t[i + j]:
            changes += 1
            positions.append(j + 1)
    if changes < min_changes:
        min_changes = changes
        best_pos = positions

print(min_changes)
print(*best_pos)
