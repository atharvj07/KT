import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] c = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                c[i][j] = sc.nextInt();
            }
        }
        int[] a = new int[3];
        int[] b = new int[3];
        for (int i = 0; i < 3; i++) {
            a[i] = c[i][0];
            for (int j = 1; j < 3; j++) {
                a[i] -= c[i][j];
            }
        }
        for (int j = 0; j < 3; j++) {
            b[j] = c[0][j];
            for (int i = 1; i < 3; i++) {
                b[j] -= c[i][j];
            }
        }
        boolean ok = true;
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                if (c[i][j] != a[i] + b[j]) {
                    ok = false;
                }
            }
        }
        System.out.println(ok ? "Yes" : "No");
    }
}
