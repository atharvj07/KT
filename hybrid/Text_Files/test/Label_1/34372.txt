import java.util.Arrays;
import java.util.Scanner;

public class VasyaAndCarRental {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int s = scanner.nextInt();
        int t = scanner.nextInt();
        int[] cars = new int[n];
        for (int i = 0; i < n; i++) {
            cars[i] = scanner.nextInt();
            cars[i] += scanner.nextInt() * 2; // Add cost of fuel to car price
        }
        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }
        Arrays.sort(gasStations); // Sort gas stations for efficient calculation

        int minPrice = -1;
        for (int carPrice : cars) {
            int time = 0;
            int distance = 0;
            int fuel = 0;
            int stationIndex = 0;
            while (distance < s && time <= t) {
                int nextStation = stationIndex < k ? gasStations[stationIndex] : s;
                int distanceToNextStation = nextStation - distance;
                int timeToNextStation = distanceToNextStation; // Normal mode
                if (fuel >= distanceToNextStation) {
                    timeToNextStation = distanceToNextStation / 2; // Accelerated mode if fuel allows
                } else {
                    time += distanceToNextStation - fuel; // Normal mode for remaining distance
                    fuel = 0;
                }
                time += timeToNextStation;
                distance = nextStation;
                fuel += s - distance; // Fill tank at gas station or end of road
                stationIndex++;
            }
            if (distance >= s && time <= t) {
                minPrice = Math.min(minPrice, carPrice);
            }
        }
        System.out.println(minPrice);
    }
}
