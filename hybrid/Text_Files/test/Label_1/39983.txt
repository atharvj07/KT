mod = 10**9 + 7
def mat_pow(a, n):
    if n == 0: return 1
    elif n == 1: return a
    else:
        if n % 2 == 0:
            return mat_pow((a * a) % mod, n // 2)
        else:
            return (mat_pow((a * a) % mod, n // 2) * a) % mod
n, m = map(int, input().split())
print(mat_pow(2, n * m))
