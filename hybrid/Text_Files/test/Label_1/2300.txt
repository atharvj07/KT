#include <iostream>
#include <vector>
#include <numeric>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<long long> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    long long t = accumulate(v.begin(), v.end(), 0LL) - v[0];
    if (7 * t % v[0] == 0) cout << (t / v[0] + 1) % n;
    else cout << (t / v[0] + 2) % n;
}
