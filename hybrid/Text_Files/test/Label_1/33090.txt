import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        // Read the tic-tac-toe position
        char[][] board = new char[4][4];
        for (int i = 0; i < 4; i++) {
            String line = in.nextLine();
            for (int j = 0; j < 4; j++) {
                board[i][j] = line.charAt(j);
            }
        }

        // Check if Ilya can win by making a single turn
        boolean canWin = false;
        for (int i = 0; i < 4; i++) {
            for (int j = 0; j < 4; j++) {
                if (board[i][j] == '.') {
                    // Place an X in the empty cell
                    board[i][j] = 'x';

                    // Check if Ilya has won
                    if (hasWon(board, 'x')) {
                        canWin = true;
                        break;
                    }

                    // Remove the X from the empty cell
                    board[i][j] = '.';
                }
            }
        }

        // Print the result
        if (canWin) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    // Check if a player has won
    private static boolean hasWon(char[][] board, char player) {
        // Check rows
        for (int i = 0; i < 4; i++) {
            if (board[i][0] == player && board[i][1] == player && board[i][2] == player && board[i][3] == player) {
                return true;
            }
        }

        // Check columns
        for (int j = 0; j < 4; j++) {
            if (board[0][j] == player && board[1][j] == player && board[2][j] == player && board[3][j] == player) {
                return true;
            }
        }

        // Check diagonals
        if (board[0][0] == player && board[1][1] == player && board[2][2] == player && board[3][3] == player) {
            return true;
        }
        if (board[3][0] == player && board[2][1] == player && board[1][2] == player && board[0][3] == player) {
            return true;
        }

        // No winner
        return false;
    }
}
