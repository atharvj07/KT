for _t in range(int(input())):
    n, m, a, b = map(int, input().split())
    g = [[] for _ in range(n+1)]
    for _ in range(m):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)
    q = [a]
    vis = [False] * (n+1)
    vis[a] = True
    while q:
        p = q.pop(0)
        for adj in g[p]:
            if not vis[adj]:
                vis[adj] = True
                q.append(adj)
                g[adj].remove(p)
            else:
                g[p].remove(adj)
    ans = 0
    for c in g[b]:
        if not vis[c]:
            ans += 1
    print(ans)
