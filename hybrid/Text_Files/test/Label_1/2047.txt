#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        long long x, y;
        std::cin >> x >> y;
        long long ans = 0;
        for (long long i = 1; i <= x; i++) {
            for (long long j = 1; j <= y; j++) {
                if (i / j == i % j) ans++;
            }
        }
        std::cout << ans << "\n";
    }
    return 0;
}
