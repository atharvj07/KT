#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  vector<vector<int>> g(n);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    g[u].push_back(v);
  }
  vector<int> d(n, -1);
  d[p[n - 1] - 1] = 0;
  for (int i = n - 1; i >= 0; i--) {
    int cur = p[i] - 1;
    for (int j : g[cur]) {
      if (d[j] == -1) continue;
      d[cur] = d[j] + 1;
      break;
    }
  }
  cout << d[0] << endl;
  return 0;
}
