#include<bits/stdc++.h>
using namespace std;

int main() {
    string s; getline(cin, s);
    int n[3] = {};
    
    for(int i = 0; i < s.size(); i++) {
        switch(s[i]) {
            case '1': n[0]++; break;
            case '2': n[1]++; break;
            case '3': n[2]++; break;
        }
    }
    
    cout << string(n[0], '1') << '+' << string(n[1], '2') << '+' << string(n[2], '3');
}
