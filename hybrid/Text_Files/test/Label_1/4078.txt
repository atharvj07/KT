#include <vector>
#include <algorithm>
#include <bits/stdc++.h>
using namespace std;

int main() {
    long long n, m, l = 0, r = 0, ans = 0;
    priority_queue<long long> L;
    priority_queue<long long, vector<long long>, greater<long long>> R;
    cin >> n >> m;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    L.push(a[0]);
    for (int i = 1; i < n; i++) {
        int x = L.top();
        if (a[i] <= x)
            L.push(a[i]);
        else
            R.push(a[i]);
        while (L.size() < R.size()) {
            L.push(R.top());
            R.pop();
        }
        while (L.size() > R.size() + 1) {
            R.push(L.top());
            L.pop();
        }
        if (L.size() == R.size()) {
            if (L.top() == R.top()) {
                if (a[i] < L.top())
                    L.push(a[i]);
                else
                    R.push(a[i]);
            } else {
                if (a[i] < L.top())
                    R.push(L.top()), L.pop(), L.push(a[i]);
                else
                    L.push(R.top()), R.pop(), R.push(a[i]);
            }
        }
        if (L.top() == m) ans += i - l + 1;
        while (L.top() > m) {
            l++;
            x = L.top();
            if (x == a[l]) {
                L.pop();
                while (L.top() > m) {
                    L.pop();
                    l++;
                }
                break;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
