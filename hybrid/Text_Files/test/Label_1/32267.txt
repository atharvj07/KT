import java.util.*;

public class Main {
	static final long M = 1000000007;
	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int D = in.nextInt();
		int[] p = new int[n - 1];
		for (int i = 0; i < n - 1; i++) {
			p[i] = in.nextInt() - 1;
		}
		int root = n - 1;
		long[] dp = new long[n];
		Arrays.fill(dp, 0);
		dp[root] = 1;
		for (int i = n - 2; i >= 0; i--) {
			if (dp[i] == 0)
				continue;
			for (int j = 1; j <= D; j++) {
				if (j > dp[p[i]])
					break;
				dp[p[i]] = (dp[p[i]] + dp[i]) % M;
			}
		}
		System.out.print(dp[root]);
	}
}
