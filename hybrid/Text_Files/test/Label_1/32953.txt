import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int left = x;
        int sad = 0;
        for (int i = 0; i < n; i++) {
            char t = sc.next().charAt(0);
            int d = sc.nextInt();
            if (t == '+') {
                left += d;
            } else {
                if (left >= d) {
                    left -= d;
                } else {
                    sad++;
                }
            }
        }
        System.out.println(left + " " + sad);
    }
}
