import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        double xA = Double.parseDouble(line1[0]);
        double yA = Double.parseDouble(line1[1]);
        double xB = Double.parseDouble(line1[2]);
        double yB = Double.parseDouble(line1[3]);
        int n = Integer.parseInt(br.readLine());
        List<Tank> tanks = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] line2 = br.readLine().split(" ");
            double x = Double.parseDouble(line2[0]);
            double y = Double.parseDouble(line2[1]);
            double ai = Double.parseDouble(line2[2]);
            double wi = Double.parseDouble(line2[3]);
            tanks.add(new Tank(x, y, ai, wi));
        }
        int k = Integer.parseInt(br.readLine());

        double l = 0;
        double r = 1e9;
        while (r - l > 1e-9) {
            double m = (l + r) / 2;
            if (canReach(xA, yA, xB, yB, tanks, k, m)) {
                r = m;
            } else {
                l = m;
            }
        }

        System.out.printf("%.4f", r);
    }

    private static boolean canReach(double xA, double yA, double xB, double yB, List<Tank> tanks, int k, double v) {
        for (Tank tank : tanks) {
            double x = tank.x;
            double y = tank.y;
            double ai = tank.ai;
            double wi = tank.wi;
            double dx = xB - xA;
            double dy = yB - yA;
            double t = (dx * (x - xA) + dy * (y - yA)) / (dx * dx + dy * dy);
            if (t < 0 || t > 1) {
                continue;
            }
            double px = x + t * dx;
            double py = y + t * dy;
            double dist = Math.sqrt((px - x) * (px - x) + (py - y) * (py - y));
            double time = dist / v;
            double angle = Math.atan2(py - y, px - x);
            if (Math.abs(angle - ai) <= time * wi) {
                k--;
            }
            if (k < 0) {
                return false;
            }
        }
        return true;
    }

    private static class Tank {
        double x;
        double y;
        double ai;
        double wi;

        public Tank(double x, double y, double ai, double wi) {
            this.x = x;
            this.y = y;
            this.ai = ai;
            this.wi = wi;
        }
    }
}
