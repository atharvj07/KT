#include<bits/stdc++.h>
using namespace std;
const int MM = 2e5+5;
vector<int>adj[MM];
bool vis[MM];
int dis[MM];
void bfs(int s){
	memset(dis,0,sizeof(dis));
	memset(vis,0,sizeof(vis));
	int mx = 0, idx = 0;
	queue<int>q;
	q.push(s);
	vis[s]=1;
	while(!q.empty()){
		int u = q.front();
		q.pop();
		for(int v: adj[u]){
			if(!vis[v]){
				q.push(v);
				vis[v]=1;
				dis[v]=dis[u]+1;
				if(mx<dis[v]){
					mx=dis[v];
					idx=v;
				}
			}
		}
	}
	mx=0;
	memset(vis,0,sizeof(vis));
	q.push(idx);
	vis[idx]=1;
	while(!q.empty()){
		int u = q.front();
		q.pop();
		for(int v: adj[u]){
			if(!vis[v]){
				q.push(v);
				vis[v]=1;
				dis[v]=dis[u]+1;
				if(mx<dis[v]){
					mx=dis[v];
				}
			}
		}
	}
	cout << mx <<'\n';
}
int main(){
	int n,x; cin>>n>>x;
	for(int i=1;i<n;i++){
		int a,b; cin>>a>>b;
		adj[a].push_back(b);
		adj[b].push_back(a);
	}
	bfs(x);
}
