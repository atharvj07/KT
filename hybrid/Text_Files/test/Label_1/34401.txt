import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Queue<Integer> q = new LinkedList<>();
        int n = sc.nextInt();
        int[][] a = new int[n + 1][2];
        for (int i = 1; i <= n; i++) {
            a[i][0] = sc.nextInt();
            a[i][1] = sc.nextInt();
        }
        int c = 0;
        for (int i = 1; i <= n; i++) {
            while (!q.isEmpty() && a[i][1] >= a[q.peek()][1]) q.poll();
            if (q.isEmpty()) c++;
            q.add(i);
        }
        for (int i : q) System.out.print(i + " ");
    }
}
