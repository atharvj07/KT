import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int[] res = new int[n];
        int m = 0;
        for (int i = 0; i < n; i++) {
            if (m == 0 || res[m - 1] + 1 >= a[i]) {
                res[m++] = a[i];
            }
        }
        System.out.println(m);
        for (int i = 0; i < m; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
