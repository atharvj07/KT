#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> t(n);
  unordered_map<int, vector<int>> m;
  for (int i = 0; i < n; i++) {
    cin >> t[i];
    m[t[i]].push_back(i + 1);
  }

  int w = min(m[1].size(), min(m[2].size(), m[3].size()));
  cout << w << "\n";
  for (int i = 0; i < w; i++) {
    cout << m[1][i] << " " << m[2][i] << " " << m[3][i] << "\n";
  }

  return 0;
}
