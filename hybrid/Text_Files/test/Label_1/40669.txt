t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    p = list(map(int, input().strip().split()))
    min_pos, max_pos = {}, {}
    for i, num in enumerate(p):
        min_pos[num] = min(min_pos.get(num, float('inf')), i)
        max_pos[num] = max(max_pos.get(num, -float('inf')), i)
    
    result = ['0'] * n
    current_min, current_max = float('inf'), -float('inf')
    for i in range(1, n + 1):
        current_min = min(current_min, min_pos[i])
        current_max = max(current_max, max_pos[i])
        if current_max - current_min == i - 1:
            result[i-1] = '1'
    print(''.join(result))
