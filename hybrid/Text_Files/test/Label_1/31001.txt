import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt(), k = scanner.nextInt();
        Set<Integer> favoriteActors = new HashSet<>();
        for (int i = 0; i < k; i++) {
            favoriteActors.add(scanner.nextInt());
        }
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            String title = scanner.next();
            int actors = scanner.nextInt();
            int favoriteCount = 0;
            for (int j = 0; j < actors; j++) {
                int actor = scanner.nextInt();
                if (actor != 0 && favoriteActors.contains(actor)) {
                    favoriteCount++;
                }
            }
            if (favoriteCount == k) {
                System.out.println(0);
            } else if (favoriteCount == 0) {
                System.out.println(1);
            } else {
                System.out.println(2);
            }
        }
    }
}
