for _ in range(int(input())):
    s = list(input())
    mn, i, lst = '9' * len(s), 0, -1
    for j in range(len(s)):
        if s[j] < s[i]:
            mn = ''.join(s)
        if int(s[j]) & 1 != int(s[i]) & 1:
            lst = i
        i = j
    if lst > -1:
        s[i], s[lst] = s[lst], s[i]
        mn = min(mn, ''.join(s))
    print(mn)
