import sys,math
n=int(input())
a=list(map(int,input().split()))
d=[{} for _ in range(n)]
for i in range(n):
    for j in range(n):
        if i==j:continue
        if a[i]%a[j]==0:d[i][a[j]]=1
        elif a[j]%a[i]==0:d[j][a[i]]=1
mx=0
for i in range(n):
    now=0
    vis=[False]*n
    q=[i]
    vis[i]=True
    while q:
        x=q.pop()
        now+=1
        for j in d[x]:
            if not vis[j]:
                q.append(j)
                vis[j]=True
    mx=max(mx,now)
print(mx)
