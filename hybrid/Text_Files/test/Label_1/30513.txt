import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class Main {

    static class Node {
        int id;
        int weight;
        boolean isLeaf;
        Set<Node> children = new HashSet<>();

        Node(int id, int weight, boolean isLeaf) {
            this.id = id;
            this.weight = weight;
            this.isLeaf = isLeaf;
        }

        void addChild(Node child) {
            children.add(child);
        }
    }

    static class Tree {
        Map<Integer, Node> nodes = new HashMap<>();

        void addNode(int id, int weight, boolean isLeaf) {
            nodes.put(id, new Node(id, weight, isLeaf));
        }

        void addEdge(int from, int to) {
            nodes.get(from).addChild(nodes.get(to));
            nodes.get(to).addChild(nodes.get(from));
        }

        long getMinimumApplesToRemove() {
            long totalWeight = 0;
            for (Node node : nodes.values()) {
                totalWeight += node.weight;
            }
            long minWeight = totalWeight / nodes.size();
            long extraWeight = 0;
            for (Node node : nodes.values()) {
                extraWeight += Math.abs(node.weight - minWeight);
            }
            return extraWeight / 2;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        List<Node> nodes = new ArrayList<>();
        String[] line1 = reader.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        String[] line2 = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            int weight = Integer.parseInt(line2[i]);
            boolean isLeaf = (weight == 0);
            Node node = new Node(i + 1, weight, isLeaf);
            nodes.add(node);
        }
        Tree tree = new Tree();
        for (Node node : nodes) tree.addNode(node.id, node.weight, node.isLeaf);
        for (int i = 1; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            int from = Integer.parseInt(line[0]);
            int to = Integer.parseInt(line[1]);
            tree.addEdge(from, to);
        }
        System.out.println(tree.getMinimumApplesToRemove());
    }
}
