import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] board = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                board[i][j] = in.nextInt();
            }
        }

        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int sumRow = 0;
                int sumCol = 0;
                for (int k = 0; k < n; k++) {
                    sumRow += board[i][k];
                    sumCol += board[k][j];
                }

                if (sumCol > sumRow) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }
}
