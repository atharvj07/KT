import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int x = sc.nextInt(), k = sc.nextInt(), y = sc.nextInt();
        int[] a = new int[n], b = new int[m];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        for (int i = 0; i < m; i++) b[i] = sc.nextInt();
        long ans = 0, cnt = 0;
        Map<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            if (map.containsKey(a[i])) continue;
            int j = i;
            while (j < n && a[j] == a[i]) {
                j++;
            }
            if (j - i >= k) {
                ans += x * (j - i) / k;
                for (int l = i; l < j; l += k) {
                    map.put(a[l], 1);
                }
                i = j;
            } else cnt++;
        }
        if (cnt <= 1) {
            System.out.println(ans);
            sc.close();
            return;
        }
        int[] gaps = new int[(int) cnt - 1];
        cnt = 0;
        for (int i = 0; i < n; i++) {
            if (map.containsKey(a[i])) continue;
            int j = i + 1;
            while (j < n && a[j] != a[i]) {
                j++;
            }
            if (j - i > 1) {
                gaps[(int) cnt++] = j - i - 1;
            }
            i = j;
        }
        Arrays.sort(gaps);
        if (cnt > 1) {
            System.out.println(-1);
            sc.close();
            return;
        } else if (cnt == 1) {
            if (gaps[0] >= k) {
                ans += x * (gaps[0]) / k;
                System.out.println(ans);
                sc.close();
                return;
            }
        }
        System.out.println(-1);
        sc.close();
    }
}
