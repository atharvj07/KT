def min_operations_to_beautiful(s, queries):
    # Function to calculate the cost of making a substring beautiful
    def calculate_cost(substring):
        cost = 0
        for i in range(len(substring) - 2):
            # Check for palindrome substrings of length 3
            if substring[i] == substring[i+2]:
                cost += 1
                # Change the middle character to avoid palindrome
                substring = substring[:i+1] + ('a' if substring[i+1] != 'a' else 'b') + substring[i+2:]
        # Check every two consecutive characters, no need to change characters already checked in triples
        for i in range(len(substring) - 1):
            if substring[i] == substring[i+1]:
                cost += 1
                # Change next character to avoid palindrome
                if i + 2 < len(substring):
                    substring = substring[:i+1] + ('a' if substring[i+1] != 'a' else 'b') + substring[i+2:]
                else:
                    substring = substring[:i+1] + ('a' if substring[i+1] != 'a' else 'b')
        return cost
    
    results = []
    for l, r in queries:
        # Adjust indices for Python's 0-based indexing
        substring = s[l-1:r]
        cost = calculate_cost(substring)
        results.append(cost)
    return results

# Read input
n, m = map(int, input().split())
s = input()
queries = [tuple(map(int, input().split())) for _ in range(m)]

# Solve
results = min_operations_to_beautiful(s, queries)

# Print output
for result in results:
    print(result)
