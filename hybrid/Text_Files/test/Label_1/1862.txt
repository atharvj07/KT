#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
int main(){
    ll n,k;
    cin>>n>>k;
    set<ll> s;
    for(ll i=0;i<k;i++)
    {
        ll x;
        cin>>x;
        s.insert(x);
    }
    queue<pair<ll,ll>> q;
    ll in[n+1]={};
    for(ll i=1;i<=n;i++){
        ll x;
        cin>>x;
        in[i]=x;
        if(!x)q.push({i,0});
    }
    ll ans=0;
    vector<ll> v;
    while(!q.empty()){
        ll x=q.front().first;
        ll y=q.front().second;
        q.pop();
        if(s.count(x))
        {
            y++;
            if(y==in[x])
            {
                ans++;
                v.push_back(x);
                for(ll i=1;i<=n;i++){
                    if(in[i]>0)in[i]--;
                    if(!in[i])q.push({i,0});
                }
            }
        }
    }
    if(ans<k){
        cout<<"-1"<<endl;
        return 0;
    }
    cout<<ans<<endl;
    for(ll x:v)cout<<x<<" ";
    cout<<endl;
}
