#include <iostream>
#include <vector>
using namespace std;

const int maxn = 105;
int n, m, h;
long long a[maxn], b[maxn], ans[maxn][maxn];
vector<long long> v;

void dfs(int now, long long sum) {
  if (now == n) {
    v.push_back(sum);
    return;
  }
  dfs(now + 1, sum * 2);
  dfs(now + 1, sum * 2 + 1);
}

int main() {
  cin >> n >> m >> h;
  for (int i = 1; i <= m; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      char ch;
      cin >> ch;
      ans[i][j] = ch - '0';
    }
  }
  dfs(0, 0);
  for (int i = n; i >= 1; i--) {
    for (int j = 1; j <= m; j++) {
      long long k = lower_bound(v.begin(), v.end(), b[i]) - v.begin();
      ans[i][j] = v[k] ^ a[j] ^ (v[k] >> (j - 1));
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cout << ans[i][j];
    }
    cout << endl;
  }
  return 0;
}
