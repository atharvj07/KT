import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextLong();
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n; i++)
            ans += a[i] * (i - (n - i - 1));
        System.out.println(ans);
    }
}
