import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] c = sc.next().toCharArray();
        int cntH = 0, cntT = 0;
        for (char ch : c) {
            if (ch == 'H') {
                cntH++;
            } else {
                cntT++;
            }
        }
        System.out.println(Math.min(cntH, cntT) - 1);
    }
}
