from collections import defaultdict
import sys

n = int(input())
a = list(map(int, input().split()))

mp = defaultdict(list)

for x in a:
    k = 0
    while (1 << k) <= x:
        k += 1
    for i in range(0, k):
        if x & (1 << i):
            mp[x].append(1 << i)

ans = []

for i in range(len(a)):
    for j in range(i + 1, len(a)):
        uni = set()
        for x in mp[a[i]]:
            for y in mp[a[j]]:
                uni.add(x + y)
        if len(uni) == 0:
            continue
        ans.append(len(uni))

for x in sorted(ans):
    print(x)
    
if len(ans) == 0:
    print(-1)
