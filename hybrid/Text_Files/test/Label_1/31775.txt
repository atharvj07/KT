import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int ans = -1;
        for (int i = 2; i < x; i++) {
            int pw = 1;
            for (int j = 0; j < n; j++) {
                pw *= i;
                if (pw > 999999999) {
                    break;
                }
            }
            if (pw % i == 0) {
                ans = i;
            }
        }
        System.out.println(ans);
        sc.close();
    }
}
