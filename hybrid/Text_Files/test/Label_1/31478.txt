import java.util.Arrays;
import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int x = sc.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = sc.nextInt();
		}
		Arrays.sort(a);
		long ans = a[0];
		for (int i = 1; i < n; i++) {
			ans = Math.min(ans + a[i], ans + x);
		}
		System.out.println(ans);
	}
}
