import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
        }
        StringBuilder sb = new StringBuilder();
        sb.append("+");
        for (int i = 1; i < n; i++) {
            if (arr[i] <= arr[i - 1]) {
                sb.append("+");
            } else {
                sb.append("-");
            }
        }
        System.out.println(sb);
    }
}
