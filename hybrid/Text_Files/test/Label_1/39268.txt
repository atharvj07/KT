from math import acos, sqrt

n = int(input())
good = []

for point in range(1, n + 1):
    x, y, z, u, v = map(int, input().split())
    bad = False
    for j in range(1, point):
        x1, y1, z1, u1, v1 = map(int, input().split())
        for k in range(point + 1, n + 1):
            x2, y2, z2, u2, v2 = map(int, input().split())
            dot = x * x1 + y * y1 + z * z1 + u * u1 + v * v1
            cross1 = x * x2 + y * y2 + z * z2 + u * u2 + v * v2
            cross2 = x1 * x2 + y1 * y2 + z1 * z2 + u1 * u2 + v1 * v2
            a = sqrt(x * x + y * y + z * z + u * u + v * v)
            b = sqrt(x1 * x1 + y1 * y1 + z1 * z1 + u1 * u1 + v1 * v1)
            c = sqrt(x2 * x2 + y2 * y2 + z2 * z2 + u2 * u2 + v2 * v2)
            if acos((dot / (a * b))) * 180 / 3.14159265 < 90:
                bad = True
            if acos((cross1 / (a * c))) * 180 / 3.14159265 < 90:
                bad = True
            if acos((cross2 / (b * c))) * 180 / 3.14159265 < 90:
                bad = True
    if not bad:
        good.append(point)

print(len(good))
print(*good, sep='\n')
