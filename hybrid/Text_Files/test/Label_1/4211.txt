#include <iostream>
#include <vector>
#include <string>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    if (m > n + 1) {
        cout << -1;
        return 0;
    }
    string s;
    for (int i = 0; i < m; i++) {
        s += "10";
    }
    for (int i = 0; i < n - m; i++) {
        s += "0";
    }
    cout << s;
    return 0;
}
