from collections import deque
from sys import stdin

n, k = map(int, stdin.readline().split())
p = list(map(int, stdin.readline().split()))
g = [[] for i in range(n + 1)]
for _ in range(k):
    x, y = map(int, stdin.readline().split())
    g[x].append(y)
in_deg = [0] * (n + 1)
for i in range(1, n + 1):
    if p[i - 1]:
        in_deg[i] = 1
        g[p[i - 1]].append(i)
q = deque()
for i in range(1, n + 1):
    if not in_deg[i]:
        q.append(i)
ans = []
while q:
    u = q.popleft()
    ans.append(u)
    for v in g[u]:
        in_deg[v] -= 1
        if not in_deg[v]:
            q.append(v)
if len(ans) != n:
    print(0)
else:
    print(*ans)
