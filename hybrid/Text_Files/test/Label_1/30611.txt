import java.util.Scanner;

public class BerlandStateUniversity {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int k = scanner.nextInt();
            int[] days = new int[7];
            for (int j = 0; j < 7; j++) {
                days[j] = scanner.nextInt();
            }

            int result = solve(k, days);
            System.out.println(result);
        }
    }

    private static int solve(int k, int[] days) {
        int minLength = Integer.MAX_VALUE;

        for (int startDay = 0; startDay < 7; startDay++) {
            int daysStudied = 0;
            int currentLength = 0;

            for (int day = startDay; daysStudied < k; day = (day + 1) % 7) {
                if (days[day] == 1) {
                    daysStudied++;
                }
                currentLength++;
            }

            minLength = Math.min(minLength, currentLength);
        }

        return minLength;
    }
}
