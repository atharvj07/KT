import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        String[] s = new String[n];
        s[0] = sc.nextLine();
        int ans = 0;
        for (int i = 1; i < n; i++) {
            s[i] = sc.nextLine();
            if (!s[i].equals(s[i - 1])) {
                ans = -1;
                break;
            }
        }
        if (ans != -1) {
            for (int i = 0; i < s[0].length(); i++) {
                char c = s[0].charAt(i);
                boolean same = true;
                for (int j = 1; j < n; j++) {
                    if (s[j].charAt(i) != c) {
                        same = false;
                    }
                }
                if (!same) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
