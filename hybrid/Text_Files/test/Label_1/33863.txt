import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int cnt = 0;
        int last = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                if (last == 0) {
                    cnt++;
                    last = 1;
                }
            } else {
                last = 0;
            }
        }
        System.out.println(cnt);
    }
}
