import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int minLuckyNumber = -1;
        int sum = 0;
        int digit = 4;
        while (sum < n) {
            if (digit > 7) {
                digit = 4;
                sum++;
            }
            sum += digit;
            minLuckyNumber = minLuckyNumber * 10 + digit;
            digit += 3;
        }
        if (sum > n) {
            minLuckyNumber = -1;
        }
        System.out.println(minLuckyNumber);
    }
}
