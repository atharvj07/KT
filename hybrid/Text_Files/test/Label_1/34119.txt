import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        Arrays.sort(a);

        long sum = 0;
        for (int i = 0; i < n; i++) {
            if (k == 0) {
                break;
            }
            int numCarrots = Math.min(k, a[i]);
            sum += (long) numCarrots * numCarrots;
            k -= numCarrots;
        }

        System.out.println(sum);
    }
}
