import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        ArrayList<Integer> ls = new ArrayList<>(100000);

        int n = cin.nextInt();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            int x = cin.nextInt();
            if (!ls.contains(x)) {
                ls.add(x);
            }
            sum += ls.size();
        }

        System.out.println(sum);
    }
}
