import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        int[] dp = new int[402];
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] cost = new int[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            cost[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            w[i] = in.nextInt();
        }

        for (int i = 1; i <= n; i++) {
            dp[i] = 0;
            for (int j = 1; j <= i; j++) {
                if (j == 1) {
                    dp[i] = Math.max(dp[i], cost[i - 1]);
                } else if (j == i) {
                    dp[i] = Math.max(dp[i], dp[j - 1]);
                } else if (w[j - 1] + 1 == w[j] && 2 * w[j] - w[j - 1] - w[j + 1] >= 0) {
                    dp[i] = Math.max(dp[i], dp[j - 1] + cost[i - 1]);
                }
            }
        }
        System.out.println(dp[n]);
    }
}
