import sys
n = int(input())
o=[]
c=[]
for i in range(n):
	x, y = map(int, sys.stdin.readline().split())
	o.append((x,y))
for i in range(n):
	x, y = map(int, sys.stdin.readline().split())
	c.append((x,y))
for i in range(n):
	for j in range(n):
		if o[i][0]+c[j][0]==o[(i+1)%n][0]+c[(j+1)%n][0] and o[i][1]+c[j][1]==o[(i+1)%n][1]+c[(j+1)%n][1]:
			print(o[i][0]+c[j][0], o[i][1]+c[j][1])
			sys.exit()
