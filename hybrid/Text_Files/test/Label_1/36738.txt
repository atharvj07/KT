for t in range(int(input())):
    n,s=map(int,input().split())
    a=list(map(int,input().split()))
    pref=[0]
    for i in a:
        pref.append(i+pref[-1])
    ans=0
    l,r=0,0
    while l<n:
        r=max(r,l)
        while r<n and pref[r+1]-pref[l]<=s:
            r+=1
        ans=max(ans,r-l)
        l+=1
    print(ans)
