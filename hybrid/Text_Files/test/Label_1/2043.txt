#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
const int MX = 2e5 + 5;
int a[MX];
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int odd = 0;
        for (int i = 1; i <= n; ++i) {
            cin >> a[i];
            odd += a[i] % 2;
        }
        if (odd <= 1 || n % 2 == odd)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
