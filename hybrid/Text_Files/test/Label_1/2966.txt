#include <bits/stdc++.h>
using namespace std;
const int N = (int)2e5 + 5, MAXC = 26;
int32_t main()
{
    string good, pattern;
    cin >> good >> pattern;
    int n;
    cin >> n;
    unordered_set<char> bad;
    for (int i = 0; i < MAXC; i++)
        if (good.find((char)(i + 'a')) == string::npos)
            bad.insert((char)(i + 'a'));
    for (int i = 0; i < n; i++)
    {
        string s;
        cin >> s;
        int j = 0, l1 = (int)s.size(), l2 = (int)pattern.size();
        int wildcard = -1;
        for (int k = 0; k < l2; k++)
        {
            if (pattern[k] == '*')
            {
                if (wildcard != -1)
                    return cout << "NO\n", 0;
                wildcard = k;
            }
            else if (pattern[k] == '?' || pattern[k] == s[j])
                j++;
            else if (bad.find(s[j]) != bad.end())
                return cout << "NO\n", 0;
        }
        if (j == l1 && (wildcard == -1 || j >= wildcard))
            cout << "YES\n";
        else
            cout << "NO\n";
    }
    return 0;
}
