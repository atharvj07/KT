import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long sum = 0;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            a[i] *= 2;
            sum += a[i];
            if (sum < 2 * a[i]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
