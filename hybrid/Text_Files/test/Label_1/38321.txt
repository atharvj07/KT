def calculate_min_bitcoins():
    HP_Y, ATK_Y, DEF_Y = map(int, input().split())
    HP_M, ATK_M, DEF_M = map(int, input().split())
    h, a, d = map(int, input().split())

    min_bitcoins = float('inf')
    for added_atk in range(101):
        for added_def in range(101):
            real_atk_Y = max(0, ATK_Y + added_atk - DEF_M)
            real_atk_M = max(0, ATK_M - DEF_Y - added_def)
            if real_atk_Y <= 0:
                continue
            necessary_HP = 0 if real_atk_M == 0 else ((HP_M + real_atk_Y - 1) // real_atk_Y) * real_atk_M - HP_Y + 1
            necessary_HP = max(0, necessary_HP)
            cost = necessary_HP * h + added_atk * a + added_def * d
            min_bitcoins = min(min_bitcoins, cost)
            
    return min_bitcoins

print(calculate_min_bitcoins())
