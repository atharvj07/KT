import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = sc.nextInt();
            roads[i][1] = sc.nextInt();
        }

        boolean[][] graph = new boolean[5][5];
        for (int i = 0; i < 3; i++) {
            graph[roads[i][0]][roads[i][1]] = true;
            graph[roads[i][1]][roads[i][0]] = true;
        }

        for (int i = 1; i <= 4; i++) {
            if (degree(graph, i) != 2) {
                System.out.println("NO");
                return;
            }
        }

        System.out.println("YES");
    }

    static int degree(boolean[][] graph, int node) {
        int degree = 0;
        for (int i = 1; i <= 4; i++) {
            if (graph[node][i]) {
                degree++;
            }
        }
        return degree;
    }
}
