import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine());
        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine());
            StringTokenizer st = new StringTokenizer(br.readLine());
            Map<Integer, Integer> freqA = new HashMap<>();
            for (int i = 0; i < n; i++) {
                int a = Integer.parseInt(st.nextToken());
                freqA.put(a, freqA.getOrDefault(a, 0) + 1);
            }
            st = new StringTokenizer(br.readLine());
            Map<Integer, Integer> freqB = new HashMap<>();
            for (int i = 0; i < n; i++) {
                int b = Integer.parseInt(st.nextToken());
                freqB.put(b, freqB.getOrDefault(b, 0) + 1);
            }
            boolean isPossible = true;
            for (Map.Entry<Integer, Integer> entry : freqB.entrySet()) {
                int b = entry.getKey();
                int countB = entry.getValue();
                if ((b % 2 == 0 && countB % 2 != freqA.getOrDefault(0, 0)) ||
                        (b % 2 == 1 && (countB % 2 != freqA.getOrDefault(1, 0) + freqA.getOrDefault(-1, 0)))) {
                    isPossible = false;
                    break;
                }
            }
            System.out.println(isPossible ? "YES" : "NO");
        }
    }
}
