import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int w = scan.nextInt();
        StringBuilder sb = new StringBuilder();
        int m = 3;
        while (w % 3 == 0) {
            sb.append("0");
            w /= 3;
            m *= 3;
        }
        while (w > 0) {
            if (w % 3 == 1) {
                sb.append("+");
                w -= 1;
            } else {
                sb.append("-");
                w -= 2;
            }
            w /= 3;
            m /= 3;
        }
        System.out.println(sb.reverse().toString());
    }
}
