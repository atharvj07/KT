#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
    long long n, x, sum = 0, ans = 0;
    cin >> n >> x;
    vector<long long> d(n);
    for (long long i = 0; i < n; i++) {
        cin >> d[i];
        sum += d[i];
    }
    for (long long i = 0; i < n; i++) {
        sum -= d[i];
        if (x > d[i]) {
            ans += d[i] * (i + 1);
        } else {
            ans += x * (i + 1);
            break;
        }
        x -= d[i];
    }
    ans += sum;
    cout << ans << endl;
    return 0;
}
