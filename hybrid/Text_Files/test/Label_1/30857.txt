import java.util.*;
import java.io.*;
class Main {
  static final int MOD = (int)1e9+7;
  static long fastpow(long a, long p) {
    if (p == 0) return 1L;
    if (p == 1) return a;
    long res = fastpow(a, p>>>1);
    res = (res*res) % MOD;
    if ((p&1) == 1) res = (res*a) % MOD;
    return res;
  }

  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    int n = cin.nextInt(), m = cin.nextInt();
    int[] a = new int[n+1], b = new int[m+1], t = new int[100001], freqA = new int[100001], freqB = new int[100001];
    for (int i = 1; i <= n; ++i) a[i] = cin.nextInt();
    for (int i = 1; i <= m; ++i) b[i] = cin.nextInt();
    for (int i = n; i >= 1; --i) {
      t[a[i]] = t[a[i]]*2 + 1;
    }
    for (int i = m; i >= 1; --i) {
      ++freqB[b[i]];
    }
    long ans = 0;
    for (int i = 1; i <= 100000; ++i) {
      long x = fastpow(t[i] + 1, freqB[i]) - 1;
      x = (x + MOD) % MOD;
      ans = (ans + x) % MOD;
    }
    System.out.println(ans);
  }
}
