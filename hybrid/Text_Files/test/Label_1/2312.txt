#include <bits/stdc++.h>
using namespace std;
const int maxn = 5005;
int n, m, k, a[maxn], dp[maxn][maxn];
int main() {
  cin >> n >> m >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n - m + 1; i++) {
    dp[i][0] = a[i];
  }
  for (int j = 1; j <= k; j++) {
    for (int i = 1; i <= n - m + 1; i++) {
      dp[i][j] = max(dp[i][j - 1], dp[i + 1][j - 1]);
      if (i + m - 1 <= n) dp[i][j] = max(dp[i][j], dp[i + 1][j - 1] + a[i]);
    }
  }
  int ans = 0;
  for (int i = 1; i <= k; i++) {
    ans = max(ans, dp[i][k]);
  }
  cout << ans << endl;
  return 0;
}
