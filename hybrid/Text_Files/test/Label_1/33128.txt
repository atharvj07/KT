import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++)
            arr[i] = sc.nextLong();
        long sum = 0;
        for (int i = 0; i < n - 1; i++)
            sum += Math.min(arr[i], arr[i + 1]);
        System.out.println(sum);
    }
}
