
import sys
input = sys.stdin.readline

n,m = map(int,input().split())
a = list(map(int,input().split()))
g = [[]for i in range(n + 1)]
for i in range(m):
    x,y = map(int,input().split())
    g[x].append(y)
    g[y].append(x)

mnm = [0] * (n + 1)
mnm[1] = a[0]
ans = 0

def dfs(x,pre):
    res = a[x - 1]
    for y in g[x]:
        if y == pre:
            continue
        dfs(y,x)
        res = min(res,mnm[y])
    mnm[x] = res
    
dfs(1,1)

for i in range(1,n + 1):
    for j in g[i]:
        if mnm[i] > mnm[j]:
            ans += mnm[j]
        else:
            ans += mnm[i]

print(ans/(n*(n - 1)))
