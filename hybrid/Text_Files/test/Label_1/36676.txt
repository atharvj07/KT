n, m = map(int, input().split())

# Impossible cases
if m < n-1 or m > 2*(n+1):
    print(-1)
else:
    # Calculate the number of extra 1's
    extra_ones = max(0, m - n - 1)
    
    result = ""
    while n or m:
        # If we have extra ones, start with "11"
        if extra_ones > 0 and m > 0:
            result += "11"
            m -= 2
            extra_ones -= 1
        # If no extra ones but still 1's left, add "1"
        elif m > 0:
            result += "1"
            m -= 1
        # If we still have 0's, add "0"
        if n > 0:
            result += "0"
            n -= 1
    
    print(result)
