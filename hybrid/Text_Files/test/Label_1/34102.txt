import java.util.*;
import java.math.BigInteger;

public class Main {
    static BigInteger MOD = BigInteger.valueOf(1000000007);

    static class Edge implements Comparable<Edge> {
        public int to;
        public BigInteger weight;

        public Edge(int to, BigInteger weight) {
            this.to = to;
            this.weight = weight;
        }

        @Override
        public int compareTo(Edge other) {
            return weight.compareTo(other.weight);
        }
    }

    static List<List<Edge>> adj;
    static boolean[] visited;
    static BigInteger[] dist;
    static int[] prev;

    static void dijkstra(int s) {
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(s, BigInteger.ZERO));
        dist[s] = BigInteger.ZERO;

        while (!pq.isEmpty()) {
            Edge edge = pq.poll();
            int u = edge.to;

            if (visited[u]) continue;
            visited[u] = true;

            for (Edge e : adj.get(u)) {
                int v = e.to;

                if (dist[v].compareTo(dist[u].add(e.weight)) > 0) {
                    dist[v] = dist[u].add(e.weight);
                    prev[v] = u;
                    pq.add(new Edge(v, dist[v]));
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        adj = new ArrayList<>(n + 1);
        visited = new boolean[n + 1];
        dist = new BigInteger[n + 1];
        prev = new int[n + 1];

        for (int i = 0; i <= n; i++) adj.add(new ArrayList<>());
        for (int i = 0; i <= n; i++) dist[i] = BigInteger.valueOf(Long.MAX_VALUE);
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt(), v = sc.nextInt(), w = sc.nextInt();
            adj.get(u).add(new Edge(v, BigInteger.valueOf(1 << w)));
            adj.get(v).add(new Edge(u, BigInteger.valueOf(1 << w)));
        }

        int s = sc.nextInt(), t = sc.nextInt();
        dijkstra(s);

        if (dist[t].compareTo(BigInteger.valueOf(Long.MAX_VALUE)) == 0) {
            System.out.println(-1);
        } else {
            System.out.println(dist[t].mod(MOD));
            int cur = t;
            List<Integer> path = new ArrayList<>();
            while (cur != s) {
                path.add(cur);
                cur = prev[cur];
            }
            path.add(s);
            Collections.reverse(path);
            System.out.println(path.size());
            for (int i : path) System.out.print(i + " ");
            System.out.println();
        }
    }
}
