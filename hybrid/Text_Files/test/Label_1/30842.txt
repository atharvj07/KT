import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int w = sc.nextInt();
    int k = sc.nextInt();
    long[] a = new long[n];
    long[] t = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextLong();
    }
    for (int i = 0; i < n; i++) {
      t[i] = sc.nextLong();
    }

    long[][] dp = new long[n + 5][w + 5];
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= w; j++) {
        dp[i][j] = -1;
      }
    }
    dp[0][0] = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 0; j <= w; j++) {
        if (j > 0) {
          dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
        }
        if (j != w) {
          long tt = Math.max(t[i - 1] / 2, 1);
          dp[i][j] = Math.max(dp[i][j], dp[i - 1][j + 1] + a[i - 1] * tt);
        }
      }
    }
    long ans = 0;
    for (int i = 0; i <= w; i++) {
      ans = Math.max(ans, dp[n][i]);
    }

    System.out.println(ans);
  }
}
