import java.util.*;
import java.io.*;

class Main {
    static int n;
    static Card[] a;
    static boolean can;
    static void rec(int l, int r) {
        if (l == r) {
            can = true;
            return;
        }
        for (int i = l + 1; i <= r; i++) {
            if (a[l].can(a[i])) {
                Card c = a[i];
                a[i] = a[l];
                a[l] = c;
                rec(l + 1, r);
                c = a[i];
                a[i] = a[l];
                a[l] = c;
            }
        }
    }

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        n = cin.nextInt();
        a = new Card[n];
        for (int i = 0; i < n; i++) {
            a[i] = new Card(cin.next());
        }
        rec(0, n - 1);
        System.out.println(can ? "YES" : "NO");
    }

    static class Card {
        char v, s;
        Card(String s) {
            v = s.charAt(0);
            this.s = s.charAt(1);
        }
        boolean can(Card o) {
            return this.v == o.v || this.s == o.s;
        }
    }
}
