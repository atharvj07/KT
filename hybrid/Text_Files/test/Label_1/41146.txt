from collections import defaultdict

n=int(input());c=[int(x) for x in input().split()];e=defaultdict(list)
for i in range(n-1): a,b=map(int,input().split());e[a].append(b);e[b].append(a)

vis=[False for x in range(n+1)]
def dfs(i,prv,ans,same):
    vis[i]=True
    for x in e[i]:
        if not vis[x]:
            if c[i-1]!=c[x-1] and same==0: same=-1
            if c[i-1]==c[x-1] and same==-1: same=0
            if c[i-1]!=c[x-1] and same==1: same=-1
            if c[i-1]==c[x-1] and same==1: same=1
            ans=dfs(x,i,ans,same)
    if same==-1: ans+=1
    if same==0 and prv!=0 and not (c[i-1]==c[prv-1]): ans+=1
    return ans

print(dfs(1,0,0,0))
