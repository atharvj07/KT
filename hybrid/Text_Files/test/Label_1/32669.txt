import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        SpeedLimit speedLimit = null;
        boolean allowOvertake = true;
        List<Integer> unnoticedSigns = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int type = Integer.parseInt(line[0]);
            int speed = 0;
            if (type == 1 || type == 3) {
                speed = Integer.parseInt(line[1]);
            }
            switch (type) {
                case 1:
                    speedLimit = new SpeedLimit(speed);
                    break;
                case 2:
                    if (!allowOvertake) {
                        unnoticedSigns.add(6);
                    }
                    break;
                case 3:
                    speedLimit = new SpeedLimit(speed);
                    break;
                case 4:
                    allowOvertake = true;
                    break;
                case 5:
                    speedLimit = null;
                    break;
                case 6:
                    allowOvertake = false;
                    break;
            }
        }
        System.out.println(unnoticedSigns.size());
    }

    static class SpeedLimit {
        int speed;

        SpeedLimit(int speed) {
            this.speed = speed;
        }
    }
}
