#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

const int maxn = 3e5 + 5;
const ll inf = 1e18;

ll f[maxn][3];
ll a[maxn][2], b[maxn][2];

void solve() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i + 1][0] >> a[i + 1][1]; }
    for (int i = 0; i < n; i++) { cin >> b[i + 1][0] >> b[i + 1][1]; }
    f[1][1] = a[1][0];
    f[1][2] = b[1][0];
    for (int i = 2; i <= n; i++) {
        f[i][0] = inf;
        f[i][1] = max(f[i - 1][0], f[i - 1][2]) + a[i][0];
        f[i][2] = max(f[i - 1][0], f[i - 1][2]) + b[i][0];
        f[i][1] = max(f[i][1], max(f[i - 1][1], f[i - 1][2]) + a[i][1]);
        f[i][2] = max(f[i][2], max(f[i - 1][1], f[i - 1][2]) + b[i][1]);
    }
    cout << max(f[n][1], f[n][2]) << '\n';
}

int main() {
    solve();
    return 0;
}
