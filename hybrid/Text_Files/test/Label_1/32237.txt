import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x = in.nextInt();
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            String s = in.next();
            int d = in.nextInt();
            if (s.equals("+")) {
                x += d;
            } else {
                if (x < d) {
                    cnt++;
                } else {
                    x -= d;
                }
            }
        }
        System.out.println(x + " " + cnt);
    }
}
