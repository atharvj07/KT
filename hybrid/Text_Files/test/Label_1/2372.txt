#include <bits/stdc++.h>

using namespace std;

void dfs(int u, int p, vector<vector<int>>& g, vector<int>& dist, vector<bool>& vis) {
    vis[u] = true;
    for (int v : g[u]) {
        if (!vis[v]) {
            dist[v] = dist[u] + 1;
            dfs(v, u, g, dist, vis);
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    vector<int> dist(n + 1, 0);
    vector<bool> vis(n + 1, false);
    dfs(1, 0, g, dist, vis);
    int mx = -1;
    int idx = -1;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > mx) {
            mx = dist[i];
            idx = i;
        }
    }
    fill(vis.begin(), vis.end(), false);
    fill(dist.begin(), dist.end(), 0);
    dfs(idx, 0, g, dist, vis);
    mx = -1;
    for (int i = 1; i <= n; i++) {
        if (dist[i] > mx) {
            mx = dist[i];
            idx = i;
        }
    }
    cout << idx << "\n" << mx;
    return 0;
}
