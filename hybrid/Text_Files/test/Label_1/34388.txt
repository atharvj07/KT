import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        boolean has100 = false;
        int last100 = -1;
        int first100 = -1;
        int maxLength = 0;
        int firstMax = -1;
        int lastMax = -1;
        for (int i = 0; i < s.length(); i++) {
            for (int j = i + 1; j < s.length(); j++) {
                if (s.charAt(i) == s.charAt(j)) {
                    int length = j - i + 1;
                    if (length == 100) {
                        has100 = true;
                        if (last100 < i) {
                            last100 = j;
                            first100 = i;
                        }
                    } else if (length > maxLength) {
                        maxLength = length;
                        firstMax = i;
                        lastMax = j;
                    }
                }
            }
        }
        if (has100) {
            for (int i = first100; i <= last100; i++) {
                System.out.print(s.charAt(i));
            }
        } else {
            for (int i = firstMax; i <= lastMax; i++) {
                System.out.print(s.charAt(i));
            }
        }
    }
}
