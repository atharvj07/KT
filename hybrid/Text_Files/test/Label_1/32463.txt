import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int[] cnt = new int[3];
        for (int i = 0; i < s.length(); i++) {
            cnt[s.charAt(i) - 'A']++;
        }
        int cnt2 = -1;
        for (int i = 0; i < 3; i++) {
            if (cnt2 == -1) {
                cnt2 = cnt[i];
            } else if (cnt2 != cnt[i]) {
                System.out.println("No");
                return;
            }
        }
        System.out.println("Yes");
    }
}
