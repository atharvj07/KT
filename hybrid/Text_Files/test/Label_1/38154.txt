import sys

n, a, b = map(int, sys.stdin.readline().strip().split())
if n > a * b:
    print(-1)
    exit()
ans = [[0 for _ in range(b)] for _ in range(a)]
for i in range(1, n + 1):
    if (i % 2 == 0):
        ans[i // b - 1][(i % b) - 1] = i
    else:
        ans[(i + 1) // b - 1][(i % b) - 1] = i
for row in ans:
    print(*row)
