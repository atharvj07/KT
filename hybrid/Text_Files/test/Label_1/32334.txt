import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int x = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        long ans = 0;
        for (int i = 1; i < N; i++) {
            if (a[i] < a[i - 1]) {
                ans += a[i];
            } else {
                ans += ((a[i] - a[i - 1]) / 2);
            }
        }
        ans += x;
        System.out.println(ans);
    }
}
