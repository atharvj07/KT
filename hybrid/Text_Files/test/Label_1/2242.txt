#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int cnt = 0;
    for (int i = 1; i < n; i++) {
        int t = 2 - abs(a[i] - a[i - 1]);
        if (t == 1) {
            cnt += 2;
        } else if (t == 2) {
            cnt += 4;
        } else {
            cnt++;
        }
    }
    if (cnt == 100) {
        cout << "Finite\n" << cnt;
    } else {
        cout << "Infinite";
    }
    return 0;
}
