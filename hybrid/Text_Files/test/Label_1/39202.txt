n = int(input())
a = [int(x) for x in input().split()]
b = [int(x) for x in input().split()]
ans = [0]*n
astack = []
for i in range(n-1,-1,-1):
    if a[i] != b[i]:
        for j in range(len(astack)-1,-1,-1):
            if astack[j] == b[i]:
                ans[i] = len(astack)-j
                break
    astack.append(a[i])
print(*ans)
