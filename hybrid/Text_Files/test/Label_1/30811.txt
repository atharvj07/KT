import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int[] h = new int[n];
    for (int i = 0; i < n; i++) {
      h[i] = sc.nextInt();
    }

    int[][] dp = new int[n][k + 1];
    for (int i = 0; i < k + 1; i++) {
      dp[0][i] = 0;
    }
    for (int i = 1; i < n; i++) {
      for (int j = 1; j <= k && i - j >= 0; j++) {
        dp[i][j] = dp[i - j][j] + Math.abs(h[i] - h[i - j]);
        if (j > 1) {
          dp[i][j] = Math.min(dp[i][j], dp[i - j + 1][j - 1] + Math.abs(h[i] - h[i - j + 1]));
        }
      }
    }

    int min = Integer.MAX_VALUE;
    for (int i = 1; i <= k; i++) {
      min = Math.min(min, dp[n - 1][i]);
    }

    System.out.println(min);
  }
}
