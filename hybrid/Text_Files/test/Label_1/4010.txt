#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n + 2), sum(n + 2, 0ll);
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) sum[i] = sum[i - 1] + a[i];
    int ans = 1e9;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == j) continue;
            int tmp = (i - j) * (i - j) + (sum[max(i, j)] - sum[min(i, j)]) * (sum[max(i, j)] - sum[min(i, j)]);
            ans = min(ans, tmp);
        }
    }
    cout << ans << endl;
}
