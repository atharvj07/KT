n = int(input().strip())
a = list(map(int, input().strip().split()))
b = list(map(int, input().strip().split()))

diff = [a[i] - b[i] for i in range(n)]
diff.sort()

count = 0
left = 0
right = n - 1

while left < right:
    if diff[left] + diff[right] > 0:
        count += right - left
        right -= 1
    else:
        left += 1

print(count)
