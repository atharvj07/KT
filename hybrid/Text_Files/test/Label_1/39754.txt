import sys

n, w = map(int, input().split())
a = list(map(int, input().split()))
a.sort(reverse=True)

ans = 0

i = 0
while i < n and a[i] >= 2 and w >= 2 * a[i]:
    w -= 2 * a[i]
    ans += a[i]
    i += 1

if w > 0:
    if i < n and a[i] >= 1 and w >= a[i]:
        w -= a[i]
        ans += a[i]
    elif i < 2 * n and a[i] >= 1 and w >= 2 * a[i]:
        w -= 2 * a[i]
        ans += a[i]

print(ans)
