import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int[] f = new int[100001];
        for (int i = 1; i <= 100000; i++) {
            for (int j = 0; j < n; j++) {
                f[i] += i % a[j];
            }
        }

        int max = 0;
        for (int i = 1; i <= 100000; i++) {
            max = Math.max(max, f[i]);
        }

        System.out.println(max);
    }
}
