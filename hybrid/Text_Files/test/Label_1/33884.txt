import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean[] increasing = new boolean[n];
        boolean[] decreasing = new boolean[n];
        int incCnt = 0, decCnt = 0;
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                increasing[i] = true;
            } else if (a[i] > a[i - 1]) {
                increasing[i] = true;
                incCnt++;
            } else {
                decreasing[i] = true;
                decCnt++;
            }
        }
        if (incCnt == n || decCnt == n) {
            System.out.println("YES");
            System.out.println(incCnt);
            for (int i = 0; i < n; i++) {
                if (increasing[i]) {
                    System.out.print(a[i] + " ");
                }
            }
            System.out.println();
            System.out.println(decCnt);
            for (int i = 0; i < n; i++) {
                if (decreasing[i]) {
                    System.out.print(a[i] + " ");
                }
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
        in.close();
    }
}
