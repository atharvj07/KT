n = int(input())
a = list(map(int, input().split()))
a.sort()
ans = sum(a)
for i, ai in enumerate(a[: -1]):
    for j in range(i + 1, len(a)):
        for x in range(1, ai + 1):
            if ai % x == 0 and a[j] % x == 0:
                if ai // x + a[j] * x < ans:
                    ans = ai // x + a[j] * x
print(ans)
