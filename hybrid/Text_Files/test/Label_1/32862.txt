import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int t = sc.nextInt();
        String s = sc.next();
        int ans = 0;
        int cur = 1;
        int tt = 0;
        char[] cs = s.toCharArray();
        boolean[] seen = new boolean[n + 1];
        while (tt + a <= t) {
            if (seen[cur]) {
                tt += a;
                continue;
            }
            if (cs[cur - 1] == 'h') {
                tt += b;
            }
            tt += a;
            seen[cur] = true;
            ans++;
            if (cur == n) {
                cur = 1;
            } else {
                cur++;
            }
        }
        System.out.println(ans);
    }
}
