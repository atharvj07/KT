import sys

for t in range(int(input())):
    n = int(input())
    p = list(map(int, input().split()))
    L = []
    a = []
    for i in p:
        while len(a) > 0 and a[-1] > i:
            L.append(a.pop())
        a.append(i)
    while len(a) > 0:
        L.append(a.pop())
    s = []
    k = 0
    for x in L:
        if not s or s[-1] != x:
            s.append(x)
            k += 1
    print(k)
    print(*s)
