#include <iostream>
#include <vector>
using namespace std;
int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) {
      cin >> b[i];
    }
    a[0] = b[0];
    a[n - 1] = b[n - 1];
    int j = 0, k = n - 1;
    for (int i = 1; i < n - 1; i++) {
      if (i % 2 == 0)
        a[j++] = b[i];
      else
        a[k--] = b[i];
    }
    for (int i = 0; i < n; i++) {
      cout << a[i] << " ";
    }
    cout << "\n";
  }
}
