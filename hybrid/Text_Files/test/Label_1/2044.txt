#include<bits/stdc++.h>
using namespace std;
int main(){
  long long int n,q,a[100000],t,x,k,i,px,ps;
  cin >> n >> q;
  for(i=0;i<n;i++) cin >> a[i];
  while(q--){
    cin >> t >> x;
    if(t==1) a[x-1]=1-a[x-1];
    else{
      sort(a,a+n,greater<long long int>());
      if(x<=n) cout << a[x-1] << endl;
      else cout << -1 << endl;
    }
  }
}
