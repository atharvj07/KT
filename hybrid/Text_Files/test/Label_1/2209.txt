#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
typedef long long ll;

struct dt {
    ll x, i;
    bool operator<(const dt &p) const { return x < p.x; }
};
dt a[200005];

int main() {
    ll n, z;
    cin >> n >> z;
    vector<ll> v;
    for (ll i = 0; i < n; i++) { cin >> a[i].x, a[i].i = i; }
    sort(a, a + n);
    for (ll i = 0; i < n; i++) {
        auto it = lower_bound(a, a + n, (dt){a[i].x + z, -1LL});
        auto it2 = upper_bound(a, a + n, (dt){a[i].x, -1LL});
        ll cnt = 0;
        if (it != a) cnt = it->i - it2->i;
        if (it2 != a) cnt -= (it2->i == a[i].i);
        if (cnt > 0) v.push_back(cnt);
    }
    if (v.empty())
        cout << 0 << endl;
    else
        cout << *max_element(v.begin(), v.end()) << endl;
}
