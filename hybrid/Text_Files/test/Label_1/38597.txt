import sys
input = sys.stdin.readline

for _ in range(int(input())):
    input()
    n,m = map(int,input().split())
    adj = [[]]
    for _ in range(n):
        adj.append([])
    dist = [10**9 for i in range(n+1)]
    dist[1] = 0
    for i in range(m):
        a,b = map(int,input().split())
        adj[a].append(b)
    q = [1]
    while q:
        u = q.pop(0)
        for v in adj[u]:
            if dist[v] > dist[u]+1:
                dist[v] = dist[u]+1
                q.append(v)
    for i in dist[1:]:
        print(i if i!=10**9 else -1,end=' ')
    print()
