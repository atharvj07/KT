#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int h = 1;
    bool dead = false;
    for (int i = 0; i < n; i++) {
      if (a[i] == 1) {
        if (!dead) {
          if (a[i - 1] == 1)
            h += 5;
          else
            h += 1;
        }
      } else {
        if (!dead && a[i - 1] == 0) { dead = true; }
      }
    }
    if (dead)
      cout << -1 << endl;
    else
      cout << h << endl;
  }
  return 0;
}
