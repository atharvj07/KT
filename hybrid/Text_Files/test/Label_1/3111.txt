#include<iostream>
#include<string>
#include<algorithm>
#include<climits>
using namespace std;
int main() {
    int n, t; cin >> n >> t;
    string s; cin >> s;
    int pos = s.find('.') + 1;
    int first = 1;
    for (int i = pos; i < min(n, pos + t + 1); i++) {
        if (s[i] > '4') {
            if (first) {
                if (s[pos] != '9') {
                    s[pos]++;
                    while (i != pos) {
                        s[i--] = '0';
                    }
                    if (i) break;
                } else {
                    pos++;
                    first = 0;
                    i = pos - 1;
                }
            }
        } else {
            while (i != pos) {
                s[i--] = '0';
            }
            if (i) break;
        }
    }
    while (s.back() == '0' && s.size() > pos + 1) {
        s.pop_back();
    }
    if (pos == n) s += ".0";
    cout << s << endl;
}
