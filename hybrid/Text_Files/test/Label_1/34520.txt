import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] s = new String[n];
        for (int i = 0; i < n; i++) s[i] = sc.next();
        int len = s[0].length();
        for (String ss : s) if (ss.length() != len) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        for (int i = 0; i < len; i++) {
            int[] cnt = new int[26];
            cnt[s[0].charAt(i) - 'a'] = 1;
            for (int j = 1; j < n; j++) {
                cnt[s[j].charAt((i + j) % len) - 'a']++;
            }
            int mx = Arrays.stream(cnt).max().getAsInt();
            ans += n - mx;
        }
        System.out.println(ans);
    }
}
