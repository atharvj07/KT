import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[][] adj = new int[n + 1][n + 1];
    for (int i = 0; i < m; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      adj[a][b] = adj[b][a] = 1;
    }
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        for (int k = j + 1; k <= n; k++) {
          if (adj[i][j] == 1 && adj[j][k] == 1 && adj[i][k] == 0) {
            System.out.println("NO");
            return;
          }
        }
      }
    }
    System.out.println("YES");
  }
}
