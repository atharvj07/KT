#include <iostream>
#include <string>

struct data {
  int l, r;
};

const int maxn = 1e6 + 10;
data st[4 * maxn];
bool lz[4 * maxn];
bool was[maxn];
int a[maxn];

int n, x;
char s[maxn];
std::string t;
bool fl = 1;

void build(int id, int l, int r) {
  st[id].l = st[id].r = 0;
  lz[id] = 0;
  if (l == r) { return; }
  int mid = (l + r) >> 1;
  build(id << 1, l, mid);
  build(id << 1 | 1, mid + 1, r);
}

void push(int id, int l, int r) {
  if (lz[id]) {
    st[id].l = st[id].r = 0;
    if (l < r) {
      lz[id << 1] = lz[id << 1 | 1] = 1;
    }
    lz[id] = 0;
  }
}

void update(int id, int l, int r, int pos) {
  if (pos < l || r < pos) { return; }
  if (l == r) {
    if (s[l] == '(') { st[id].l++; } else { st[id].r++; }
    return;
  }
  push(id, l, r);
  int mid = (l + r) >> 1;
  update(id << 1, l, mid, pos);
  update(id << 1 | 1, mid + 1, r, pos);
  st[id].l = st[id << 1].l + st[id << 1 | 1].l;
  st[id].r = st[id << 1].r + st[id << 1 | 1].r;
}

void range(int id, int l, int r, int lq, int rq) {
  if (rq < l || r < lq) { return; }
  push(id, l, r);
  if (lq <= l && r <= rq) {
    if (l == r) {
      a[x] = st[id].l + st[id].r;
    }
    st[id].l = st[id].r = 0;
    was[x] = 1;
    return;
  }
  int mid = (l + r) >> 1;
  range(id << 1, l, mid, lq, rq);
  range(id << 1 | 1, mid + 1, r, lq, rq);
  st[id].l = st[id << 1].l + st[id << 1 | 1].l;
  st[id].r = st[id << 1].r + st[id << 1 | 1].r;
}

bool check() {
  if (t.empty()) return 1;
  if (t[0] == '(') {
    int balance = 0;
    for (char c : t) {
      balance += c == '(' ? 1 : -1;
      if (balance < 0) return 0;
    }
    return balance == 0;
  }
  return 0;
}

int main() {
  std::ios_base::sync_with_stdio(0);
  std::cin >> n >> s;
  build(1, 1, n);
  for (int i = 1; i <= n; i++) {
    if (s[i] == 'L') {
      if (x > 0) x--;
    } else if (s[i] == 'R') {
      if (x < n) x++;
    } else {
      update(1, 1, n, x);
      if (s[i] == '(') t += s[i];
      if (s[i] == ')') {
        if (t.back() == '(') t.pop_back(); else continue;
      }
      if (!check()) {
        t.clear();
        fl = 0;
        std::cout << -1 << " ";
      } else {
        x++;
        range(1, 1, n, 1, x - 1);
        int ans = 0;
        for (int j = 1; j <= x; j++) {
          if (!was[j]) continue;
          ans += a[j] > 0;
          was[j] = 0;
        }
        std::cout << ans << " ";
      }
    }
  }
  return 0;
}
