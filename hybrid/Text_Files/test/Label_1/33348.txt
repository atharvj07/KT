import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    boolean[] g = new boolean[a[n - 1] + 1];
    ArrayList<ArrayList<Integer>> f = new ArrayList<>();
    for (int i = 0; i <= a[n - 1]; i++) {
      f.add(new ArrayList<>());
    }
    for (int i = 2; i <= a[n - 1]; i++) {
      if (!g[i]) {
        f.get(i).add(i);
        for (int j = i + i; j <= a[n - 1]; j += i) {
          f.get(j).add(i);
          g[j] = true;
        }
      }
    }
    int c = 0;
    for (int i = 0; i < n - 1; i++) {
      if (Collections.disjoint(f.get(a[i]), f.get(a[i + 1]))) c++;
    }
    System.out.println(c == 0 ? 1 : 0);
  }
}
