#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll solve(ll n, ll a[], ll m, ll p[], ll s[]) {
  vector<pair<ll, ll>> v;
  for (ll i = 0; i < m; i++) v.push_back({p[i], s[i]});
  sort(a, a + n);
  sort(v.begin(), v.end());
  ll ans = 0, j = 0;
  for (ll i = 0; i < n; i++) {
    while (j < m && v[j].first < a[i]) j++;
    if (j == m) return -1;
    ans++;
    if (--v[j].second == 0) j++;
  }
  return ans;
}

int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, m;
    cin >> n;
    ll a[n];
    for (ll i = 0; i < n; i++) cin >> a[i];
    cin >> m;
    ll p[m], s[m];
    for (ll i = 0; i < m; i++) cin >> p[i] >> s[i];
    cout << solve(n, a, m, p, s) << endl;
  }
  return 0;
}
