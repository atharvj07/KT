#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, x, y;
    cin >> n >> x >> y;
    string a;
    cin >> a;
    int ans = 0;
    for (int i = x - y - 1; i >= 0; i--) {
        if (a[i] != a[i + y]) {
            ans++;
        }
    }
    cout << ans << endl;
    return 0;
}
