import sys

n = int(input())
a = list(map(int, input().split()))

dp = [0] * (n + 1)
for i in range(n):
    dp[i + 1] = dp[i] + 1
    if i + 1 < n:
        for j in range(i + 1):
            if a[i] % a[j] == 1 and dp[j + 1] + 1 > dp[i + 1]:
                dp[i + 1] = dp[j + 1] + 1

print(max(dp))
