import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long d = 7;
        long cnt = 0;
        while (d % k != 0) {
            d = (d * 10 + 7) % k;
            cnt++;
        }
        if (cnt == 0) {
            System.out.println(-1);
        } else {
            System.out.println(cnt + 1);
        }
    }
}
