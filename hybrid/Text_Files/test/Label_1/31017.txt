import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-->0) {
            int n = sc.nextInt();
            char[] s = sc.next().toCharArray();
            int l = 0, r = n - 1, ans = 0;
            while (l <= r) {
                if (s[l] != s[r]) {
                    ans++;
                    if (s[l] == '<') l++;
                    else r--;
                } else {
                    l++;
                    r--;
                }
            }
            System.out.println(ans);
        }
    }
}
