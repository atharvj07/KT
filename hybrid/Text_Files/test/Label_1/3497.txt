#include <cstdio>
#include <vector>
#include <cstring>
#include <algorithm>
using namespace std;

int deg[100001];
vector< int > adj[100001];
int path[100001][2];

int main() {
  int n;
  scanf("%d", &n);

  memset(deg, 0, sizeof deg);
  for (int a, b, i = 1; i < n; i++) {
    scanf("%d%d", &a, &b);
    deg[a]++;
    deg[b]++;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }

  int st = -1;
  for (int u = 1; u <= n; u++) {
    if (deg[u] > 2) {
      puts("No");
      return 0;
    } else if (deg[u] == 1) {
      if (st != -1) {
        puts("No");
        return 0;
      }
      st = u;
    }
  }

  puts("Yes");
  int cnt = 0;
  int p = 1;
  while (deg[st]) {
    path[p][0] = st;
    path[p][1] = adj[st][0];
    deg[st]--;
    deg[adj[st][0]]--;
    st = adj[st][0];
    if (!deg[st])
      cnt++;
    p++;
  }
  printf("%d\n", cnt);
  for (int i = 1; i <= cnt; i++)
    printf("%d %d\n", path[i][0], path[i][1]);
  return 0;
}
