import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] t = new int[n];
        String[] line = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            t[i] = Integer.parseInt(line[i]);
        }

        // Count the number of occurrences of each color in each interval
        int[][] count = new int[n + 1][n + 1];
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                count[t[i]][j - i + 1]++;
            }
        }

        // Find the dominant color for each interval
        int[] dominant = new int[n + 1];
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int maxCount = 0;
                int dominantColor = 0;
                for (int k = 1; k <= n; k++) {
                    if (count[k][j - i + 1] > maxCount) {
                        maxCount = count[k][j - i + 1];
                        dominantColor = k;
                    }
                }
                dominant[i + 1]++;
            }
        }

        // Print the number of intervals where each color is dominant
        for (int i = 1; i <= n; i++) {
            System.out.println(dominant[i]);
        }
    }
}
