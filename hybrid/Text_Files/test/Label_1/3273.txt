#include <algorithm>
#include <iostream>
#include <map>
#include <vector>
using namespace std;
map<int, long long> ms, mb;
int main() {
  int n, s;
  cin >> n >> s;
  for (int i = 0; i < n; i++) {
    char c;
    int x, y;
    cin >> c >> x >> y;
    if (c == 'B')
      mb[x] += y;
    else
      ms[x] += y;
  }
  vector<int> a, b;
  for (pair<int, long long> p : ms) a.push_back(p.first);
  for (pair<int, long long> p : mb) b.push_back(p.first);
  sort(a.begin(), a.end());
  sort(b.rbegin(), b.rend());
  for (int i = 0; i < min(s, int(a.size())); i++)
    cout << "S " << a[i] << ' ' << ms[a[i]] << '\n';
  for (int i = 0; i < min(s, int(b.size())); i++)
    cout << "B " << b[i] << ' ' << mb[b[i]] << '\n';
  return 0;
}
