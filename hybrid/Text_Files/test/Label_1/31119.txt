import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextLong();
    long sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];

    long ans = sum;
    long cut = 0;
    boolean[] del = new boolean[n];
    for (int i = 1; i < n - 1; i++) {
      if (a[i] < 0) {
        ans -= a[i];
        del[i] = true;
        cut++;
      }
    }
    if (a[0] != a[n - 1]) {
      if (a[0] < a[n - 1]) {
        ans -= a[0];
        del[0] = true;
        cut++;
      } else {
        ans -= a[n - 1];
        del[n - 1] = true;
        cut++;
      }
    }
    System.out.println(ans + " " + cut);
    for (int i = 0; i < n; i++) {
      if (del[i]) System.out.print((i + 1) + " ");
    }
    System.out.println();
  }
}
