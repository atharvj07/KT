t = int(input())
for _ in range(t):
    n = int(input())
    days = list(map(int, input().split()))
    height = 1
    consecutive_no_water = 0
    for i in range(n):
        if days[i] == 0:
            consecutive_no_water += 1
            if consecutive_no_water == 2:
                height = -1
                break
        else:
            if i > 0 and days[i-1] == 1:
                height += 5
            else:
                height += 1
            consecutive_no_water = 0
    print(height)
