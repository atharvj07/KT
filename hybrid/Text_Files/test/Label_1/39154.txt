n = int(input())
a = list(map(int, input().split()))

# Preprocess to find the minimum value in each subarray
min_prefix = [0] * n
min_suffix = [0] * n

min_prefix[0] = a[0]
for i in range(1, n):
    min_prefix[i] = min(min_prefix[i-1], a[i])

min_suffix[n-1] = a[n-1]
for i in range(n-2, -1, -1):
    min_suffix[i] = min(min_suffix[i+1], a[i])

# Function to check if all elements in the subarray are divisible by the minimum element
def check(l, r, min_val):
    for i in range(l, r+1):
        if a[i] % min_val != 0:
            return False
    return True

# Main logic to find the required pairs
max_length = 0
start_indices = []

for i in range(n):
    l, r = 0, i
    while l < r:
        mid = (l + r) // 2
        if check(mid, i, min(min_prefix[mid], min_suffix[i])):
            r = mid
        else:
            l = mid + 1
    length = i - r
    if length > max_length:
        max_length = length
        start_indices = [r + 1]
    elif length == max_length:
        start_indices.append(r + 1)

print(len(start_indices), max_length)
print(' '.join(map(str, start_indices)))
