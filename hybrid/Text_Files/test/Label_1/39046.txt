n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

answer = float('inf')
for mask in range(512):
    valid = True
    for ai in a:
        found = False
        for bj in b:
            if (ai & bj) | mask == mask:
                found = True
                break
        if not found:
            valid = False
            break
    if valid:
        answer = min(answer, mask)
print(answer)
