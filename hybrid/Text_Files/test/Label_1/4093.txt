#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;

    vector<long long> a(k);
    for (int i = 0; i < k; i++) {
        cin >> a[i];
    }

    vector<bool> alive(n + 1, true);
    int leader = 1;

    for (int i = 0; i < k; i++) {
        int cnt = 0;
        while (cnt < a[i]) {
            leader = (leader % n) + 1;
            if (alive[leader]) {
                cnt++;
            }
        }
        cout << leader << " ";
        alive[leader] = false;
    }

    cout << endl;
    return 0;
}
