#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

#define ll long long

const int maxN = 2e5 + 1;

pair<ll, ll> arr[maxN];

int main() {
  ll n, x;  
  cin >> n >> x;

  for (ll i = 1; i <= n; i++) {
    cin >> arr[i].first >> arr[i].second;
    arr[i].second -= arr[i].first - 1;
  }

  sort(arr + 1, arr + n + 1);  
  ll ans = -1;

  for (ll i = 1; i <= n; i++) {
    ll l = i + 1, r = n, need = x - arr[i].second;

    while (l <= r) {
      ll mid = (l + r) / 2;

      if (arr[mid].first >= arr[i].second) {
        r = mid - 1;
      } else {
        if (arr[mid].second >= need) {
          ll cost = arr[i].first + arr[mid].first;
          if (ans == -1) ans = cost;
          ans = min(ans, cost);
        }
        l = mid + 1;
      }
    }
  }

  cout << ans << endl;

  return 0;
}
