import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = s.charAt(i) - '0';
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        if (sum % 2 == 1) {
            System.out.println("NO");
            return;
        }
        int left = 0;
        int right = sum / 2;
        for (int i = 0; i < n; i++) {
            left += a[i];
            right -= a[i];
            if (left == right) {
                System.out.println("YES");
                return;
            }
        }
        System.out.println("NO");
    }
}
