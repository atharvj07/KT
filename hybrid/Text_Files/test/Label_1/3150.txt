#include<bits/stdc++.h>
using namespace std;
int n, m, a[300001], b[300001];
set<int> s;

int main() {
    cin >> n >> m;
    for (int i = 1; i <= m; i++) cin >> a[i] >> b[i];
    s.insert(a[1]);
    s.insert(b[1]);
    for (int i = 2; i <= m; i++) {
        if (s.find(a[i]) != s.end() || s.find(b[i]) != s.end()) continue;
        s.insert(a[i]);
        s.insert(b[i]);
    }
    if (s.size() == n) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
