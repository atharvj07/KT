import sys
from collections import deque
from collections.abc import Iterable

n, m = map(int, input().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

q = deque()
in_degree = [0] * (n + 1)
for i in range(1, n + 1):
    if not adj[i]:
        q.append(i)
    else:
        for j in adj[i]:
            in_degree[j] += 1

ans = []
while q:
    if len(q) > 1:
        ans = [-1]
        break

    u = q.popleft()
    ans.append(u)
    for v in adj[u]:
        in_degree[v] -= 1
        if in_degree[v] == 0:
            q.append(v)

if ans != [-1]:
    print(*ans)
else:
    print(ans[0])
