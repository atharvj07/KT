#include <cstdio>
int main() {
  int n;
  scanf("%d", &n);
  long long a[n], b[n];
  for (int i = 0; i < n; ++i) scanf("%lld", &a[i]);
  for (int i = 0; i < n; ++i) scanf("%lld", &b[i]);
  for (int i = 1; i < n - 1; ++i)
    if (a[i] + b[i + 1] + b[i - 1] - (b[i] << 1)) {
      puts("No");
      return 0;
    }
  puts("Yes");
}
