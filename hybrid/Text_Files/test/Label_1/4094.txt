#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int n, a[N], c[N], f;
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        c[a[i]]++;
    }
    for (int i = 1; i <= 10; i++) f += (c[i] % 2);
    while (a[n] == a[n - 1]) n--;
    if (n % 2 == f % 2)
        cout << n + 1 << endl;
    else
        cout << n << endl;
}
