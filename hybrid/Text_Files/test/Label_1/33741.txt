import java.util.*;
import java.io.*;

public class Main {
    static int[] d;
    static int[] p;
    static int n, m;
    static ArrayList<ArrayList<Integer>> edges;
    static ArrayList<Integer> ans;

    public static void dfs(int v, int par) {
        p[v] = par;
        for (int u : edges.get(v)) {
            if (u == par) {
                continue;
            }
            d[u] = d[v] + 1;
            dfs(u, v);
        }
    }

    public static int lca(int v, int u) {
        if (d[v] < d[u]) {
            int tmp = v;
            v = u;
            u = tmp;
        }
        while (d[v] != d[u]) {
            v = p[v];
        }
        while (v != u) {
            v = p[v];
            u = p[u];
        }
        return v;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] parts = br.readLine().split(" ");
        n = Integer.parseInt(parts[0]);
        m = Integer.parseInt(parts[1]);
        d = new int[n + 1];
        p = new int[n + 1];
        edges = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            edges.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            parts = br.readLine().split(" ");
            int a = Integer.parseInt(parts[0]);
            int b = Integer.parseInt(parts[1]);
            edges.get(a).add(b);
            edges.get(b).add(a);
        }
        parts = br.readLine().split(" ");
        int s1 = Integer.parseInt(parts[0]);
        int t1 = Integer.parseInt(parts[1]);
        int l1 = Integer.parseInt(parts[2]);
        parts = br.readLine().split(" ");
        int s2 = Integer.parseInt(parts[0]);
        int t2 = Integer.parseInt(parts[1]);
        int l2 = Integer.parseInt(parts[2]);
        dfs(1, 1);
        int LCA = lca(s1, t1);
        int len1 = d[s1] + d[t1] - 2 * d[LCA];
        LCA = lca(s2, t2);
        int len2 = d[s2] + d[t2] - 2 * d[LCA];
        if (len1 > l1 || len2 > l2) {
            System.out.println(-1);
            return;
        }
        ans = new ArrayList<>();
        int v = s1;
        while (v != t1) {
            for (int u : edges.get(v)) {
                if (u == p[v]) {
                    continue;
                }
                if (d[u] + d[t1] - 2 * d[lca(u, t1)] == len1) {
                    ans.add(u);
                    v = u;
                    break;
                }
            }
        }
        v = s2;
        while (v != t2) {
            for (int u : edges.get(v)) {
                if (u == p[v]) {
                    continue;
                }
                if (d[u] + d[t2] - 2 * d[lca(u, t2)] == len2) {
                    ans.add(u);
                    v = u;
                    break;
                }
            }
        }
        System.out.println(ans.size());
        for (int u : ans) {
            System.out.println(v + " " + u);
            v = u;
        }
    }
}
