n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
ans = [0 for i in range(n)]
l = [0 for i in range(n+1)]

for i in a:
    l[i] = 1

for i in range(n-1, -1, -1):
    if l[b[i]]:
        ans[i] = 1
        for j in range(b[i], n+1):
            l[j] = 0

for i in ans:
    print(i)
