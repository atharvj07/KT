#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> beacons(n);
    for (int i = 0; i < n; i++) {
        cin >> beacons[i].first >> beacons[i].second;
    }
    sort(beacons.begin(), beacons.end(), [](const pair<int, int>& a, const pair<int, int>& b) {
        return a.second < b.second;
    });
    int ans = 0;
    int last = -1;
    for (const auto& beacon : beacons) {
        if (beacon.first > last) {
            ans++;
            last = beacon.first + beacon.second - 1;
        }
    }
    cout << ans << endl;
    return 0;
}
