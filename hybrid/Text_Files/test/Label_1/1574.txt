#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  if (s == t)
    cout << 0;
  else {
    int cnt = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] != t[i]) {
        if (s[i] == '0' && t[i] == '1')
          cnt++;
        else {
          cout << -1;
          return 0;
        }
      }
    }
    cout << cnt;
  }
}
