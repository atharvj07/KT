#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

const int MOD = 1e9 + 7;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  vector<vector<int>> dp(n + 1, vector<int>(k + 1, MOD));
  dp[0][0] = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      dp[i][j] = dp[i - 1][j];
      if (j > 0) dp[i][j] = min(dp[i][j], dp[i - 1][j - 1] + a[i - 1] + b[i - 1]);
    }
  }
  cout << dp[n][k] << endl;
  return 0;
}
