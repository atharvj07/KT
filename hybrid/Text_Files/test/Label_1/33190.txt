import java.util.Scanner;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[][] dp = new long[n + 1][m + 1]; // dp[i][j] = number of ways to paint a cave on the first i rows and first j columns
        dp[1][0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                dp[i][j] = dp[i - 1][j] + dp[i][j - 1]; // number of ways to paint a cave without the current row or column
                if (j >= 2) {
                    dp[i][j] -= dp[i][j - 2]; // subtract the number of ways to paint a cave that has two black cells in the current row
                }
                dp[i][j] %= MOD;
            }
        }
        System.out.println(dp[n][m]);
    }
}
