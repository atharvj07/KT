import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    private static int n, m, s;
    private static long[] mushrooms;
    private static boolean[] visited;
    private static long maxMushrooms;

    private static void dfs(int node) {
        if (visited[node]) {
            return;
        }
        visited[node] = true;
        for (int i = 0; i < n; i++) {
            if (mushrooms[i] > 0) {
                maxMushrooms += mushrooms[i];
                mushrooms[i]--;
            }
        }
        for (int i = 0; i < n; i++) {
            if (mushrooms[i] > 0) {
                dfs(i);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        n = Integer.parseInt(line1[0]);
        m = Integer.parseInt(line1[1]);
        s = Integer.parseInt(br.readLine());
        mushrooms = new long[n];
        visited = new boolean[n];
        for (int i = 0; i < m; i++) {
            String[] line2 = br.readLine().split(" ");
            int from = Integer.parseInt(line2[0]) - 1;
            int to = Integer.parseInt(line2[1]) - 1;
            long w = Long.parseLong(line2[2]);
            mushrooms[from] += w;
            mushrooms[to] += w;
        }
        dfs(s - 1);
        System.out.println(maxMushrooms);
    }
}
