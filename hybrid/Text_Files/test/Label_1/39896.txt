import sys
input = sys.stdin.readline

for i in range(int(input())):
    n, p, k = map(int, input().split())
    a = [0] + list(map(int, input().split()))
    a.sort()
    i, ans = 0, 0
    while i < n and a[i] <= p:
        if i + k - 1 < n and p - a[i] >= a[i + k - 1]:
            p -= a[i]
            ans += k
            i += k
        else:
            p -= a[i]
            ans += 1
            i += 1
    print(ans)
