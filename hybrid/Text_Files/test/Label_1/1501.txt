#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    int mn = *a.begin(), mx = *a.rbegin();
    for (int i = 0; i < k and a.size() > 1; i++) {
        if ((mn ^ x) < mn) mn ^= x, mx ^= x;
        a.erase(a.begin(), a.begin() + 1);
        if ((mn ^ x) < mn) mn ^= x, mx ^= x;
        if (!a.empty()) a.erase(a.begin());
    }
    cout << mx << ' ' << mn << endl;
}
