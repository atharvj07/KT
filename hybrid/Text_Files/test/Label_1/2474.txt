#include <bits/stdc++.h>
using namespace std;
int n, t;
string s;
int main() {
  cin >> n >> t >> s;
  bool dot = false, need = false;
  for (int i = 0; i < n; i++) {
    if (s[i] == '.') {
      dot = true;
      continue;
    }
    if (!dot) {
      continue;
    }
    if (t <= 0) {
      break;
    }
    if (s[i] == '9') {
      need = true;
    } else if (s[i] >= '5' && (!need || i <= n - 1)) {
      s[i]++;
      t--;
      break;
    } else {
      break;
    }
  }
  if (need) {
    for (int i = n - 1; i >= 0; i--) {
      if (s[i] == '.') {
        break;
      } else {
        s[i] = '0';
      }
    }
    for (int i = n - 1; i >= 0; i--) {
      if (s[i] == '0') {
        s.pop_back();
      } else {
        s[i]++;
        break;
      }
    }
  }
  cout << s << endl;
}
