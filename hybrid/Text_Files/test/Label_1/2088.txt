#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> a(n);
    vector<pair<int, int>> b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first;
        cin >> a[i].second;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i].first;
        cin >> b[i].second;
    }
    string s1 = "";
    string s2 = "";
    for (int k = 0; k <= n / 2; k++) {
        int x = k;
        int y = n - x - 1;
        for (int i = 0; i < n; i++) {
            if (i < x || i >= x && a[i].second <= b[y - (i - x)].second)
                s1 += "1";
            else
                s1 += "0";
        }
        for (int i = 0; i < n; i++) {
            if (i < x || i >= x && b[i].second <= a[y - (i - x)].second)
                s2 += "1";
            else
                s2 += "0";
        }
    }
    cout << s1 << "\n";
    cout << s2;
    return 0;
}
