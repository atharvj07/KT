import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class MostFrequentSubstring {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = scanner.nextLine();
        }

        String result = findGoodString(strings);
        System.out.println(result);
    }

    private static String findGoodString(String[] strings) {
        Map<String, Integer> substringCount = new HashMap<>();
        for (String string : strings) {
            for (int i = 0; i < string.length(); i++) {
                for (int j = i + 1; j <= string.length(); j++) {
                    String substring = string.substring(i, j);
                    substringCount.put(substring, substringCount.getOrDefault(substring, 0) + 1);
                }
            }
        }

        int maxCount = 0;
        for (int count : substringCount.values()) {
            maxCount = Math.max(maxCount, count);
        }

        String result = null;
        for (String string : strings) {
            boolean isGood = true;
            for (int i = 0; i < string.length(); i++) {
                for (int j = i + 1; j <= string.length(); j++) {
                    String substring = string.substring(i, j);
                    if (substringCount.get(substring) != maxCount) {
                        isGood = false;
                        break;
                    }
                }
            }

            if (isGood && (result == null || string.compareTo(result) < 0)) {
                result = string;
            }
        }

        return result == null ? "NO" : result;
    }
}
