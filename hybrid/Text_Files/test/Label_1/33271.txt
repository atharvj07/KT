import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int input = scanner.nextInt();
        int output;
        switch (input) {
            case 1:
            case 3:
            case 5:
                output = 1;
                break;
            case 2:
            case 4:
                output = 2;
                break;
            default:
                output = 3;
                break;
        }
        System.out.println(output);
    }
}
