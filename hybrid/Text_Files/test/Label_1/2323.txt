#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n, k;
int isok;
vector<ll> ans;
void dfs(int pos, ll sum) {
  if (pos == k) {
    if (sum == n) {
      isok = 1;
      reverse(ans.begin(), ans.end());
      for (int i = 0; i < ans.size(); i++) cout << ans[i] << " ";
      cout << "\n";
    }
    return;
  }
  for (ll i = 31; i >= 0; i--) {
    if (sum + (1 << i) <= n) {
      ans.push_back(i);
      dfs(pos + 1, sum + (1 << i));
      ans.pop_back();
    }
  }
}
int main() {
  cin >> n >> k;
  dfs(0, 0);
  if (!isok) cout << "No\n";
  return 0;
}
