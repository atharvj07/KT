import sys
n, m = map(int, input().split())
a = []
for i in range(n):
    a.append(list(map(int, input().split())))
b = []
for i in range(m):
    c = 0
    for j in range(n):
        c ^= a[j][i]
    b.append(c)
if sum(b) == 0:
    sys.stdout.write("NIE")
else:
    sys.stdout.write("TAK\n")
    for i in range(m):
        if b[i] != 0:
            for j in range(n):
                if a[j][i] ^ b[i] == 0:
                    sys.stdout.write("{} ".format(j + 1))
                    break
