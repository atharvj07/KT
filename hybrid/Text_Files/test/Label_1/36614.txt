from math import gcd
for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    idx = []
    for i in range(1, n):
        if a[i] < a[i - 1]:
            idx.append(i)
    if len(idx) <= 2:
        print('YES')
    elif idx[1] - idx[0] == 1:
        print('YES')
    else:
        print('NO')
