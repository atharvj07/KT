import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        PriorityQueue<Long> pq = new PriorityQueue<>();
        for (long i : a) {
            pq.add(i);
        }
        long ans = 0;
        while (pq.size() > 1) {
            long x = pq.poll();
            long y = pq.poll();
            ans += x + y;
            pq.add(x + y);
        }
        System.out.println(ans);
    }
}
