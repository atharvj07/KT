import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        for (int i = n; i <= m; i += n) {
            for (int j = i; j >= i - n + 1; j--) {
                System.out.print(j + " ");
            }
        }
        for (int i = n - 1; i >= 1; i--) {
            for (int j = i; j <= i + n - 1; j++) {
                System.out.print(j + " ");
            }
        }
        System.out.println();
    }
}
