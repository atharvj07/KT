import numpy as np

n = int(input())
a = np.array([int(x) for x in input().split()])
if n % 2 == 1:
    if (a[::2] == a[::-2]).all():
        print('YES')
        print('\n'.join(' '.join(list(map(str, x))) for x in np.split(a, n)))
    else:
        print('NO')
else:
    a1 = a[a % 100 > n]
    a2 = a[a % 100 <= n]
    if len(a1) != len(a2):
        print('NO')
    else:
        a = np.concatenate((a1, a2))
        if len(set(np.ravel(a))) != len(a):
            print('NO')
        else:
            print('YES')
            nr = n // 2
            nc = n
            first = np.zeros(shape=(nr, nc), dtype=int)
            last = np.zeros(shape=(nr, nc), dtype=int)
            for i in range(0, n):
                for j in range(0, n):
                    if i < nr:
                        first[i][j] = a[i * n + j]
                    if i >= nr:
                        last[i - nr][j] = a[i * n + j]
            for i in range(0, nr):
                for j in range(0, n):
                    if first[i][j] == last[i][j]:
                        print(' '.join(list(map(str, first[i]))))
                        print(' '.join(list(map(str, last[i]))))
                    else:
                        print(' '.join(list(map(str, first[i]))))
                        print(' '.join(list(map(str, last[i][::-1]))))
