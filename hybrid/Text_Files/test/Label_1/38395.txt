from sys import stdin

t = int(stdin.readline())

for _ in range(t):
    n, m = map(int, stdin.readline().split())
    a = [int(x) for x in stdin.readline().split()]
    b = [int(x) for x in stdin.readline().split()]
    if sum(a) < m:
        print(-1)
    else:
        a.sort()
        b.sort()
        ans = 0
        i = 0
        while i < n:
            if a[i] > m:
                break
            m -= a[i]
            ans += b[i]
            i += 1
        print(ans)
