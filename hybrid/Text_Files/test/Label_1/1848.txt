#include <bits/stdc++.h>

using namespace std;

const int inf = (int)2e9;

int n, k;
vector<int> a;

vector<int> solve() {
  vector<int> cnt(k, 0), ans(n);
  for (int x : a) {
    if (++cnt[x - 1] > 1) return {};
  }
  int need = (n + k - 1) / k, cur = 0, last = -inf;
  for (int i = 0; i < n; i++) {
    if (++cur > need) cur = 1, last = i;
    if (i == last)
      ans[i] = a[i];
    else
      ans[i] = i - last + 1;
  }
  return ans;
}

int main() {
  cin >> n >> k;
  a.resize(n);
  for (int &x : a) cin >> x;
  auto res = solve();
  if (res.empty())
    cout << "NO" << endl;
  else {
    cout << "YES" << endl;
    for (int x : res) cout << x << " ";
    cout << endl;
  }
}
