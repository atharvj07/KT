import java.util.BitSet;
import java.util.Scanner;

public class Main {
    private static final int MAX_N = 10;
    private static final int MAX_K = 64;
    private static final int MAX_Q = 200000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        BitSet[] masks = new BitSet[n];
        for (int i = 0; i < n; i++) {
            masks[i] = new BitSet();
            int k = sc.nextInt();
            for (int j = 0; j < k; j++) {
                masks[i].set(sc.nextInt());
            }
        }

        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            int m = sc.nextInt();
            switch (type) {
                case 0:
                    System.out.println(test(m, masks));
                    break;
                case 1:
                    set(m, masks);
                    break;
                case 2:
                    clear(m, masks);
                    break;
                case 3:
                    flip(m, masks);
                    break;
                case 4:
                    System.out.println(all(m, masks));
                    break;
                case 5:
                    System.out.println(any(m, masks));
                    break;
                case 6:
                    System.out.println(none(m, masks));
                    break;
                case 7:
                    System.out.println(count(m, masks));
                    break;
                case 8:
                    System.out.println(val(m, masks));
                    break;
            }
        }
    }

    private static int test(int i, BitSet[] masks) {
        for (BitSet mask : masks) {
            if (mask.get(i)) {
                return 1;
            }
        }
        return 0;
    }

    private static void set(int m, BitSet[] masks) {
        masks[m].set(0, 64, true);
    }

    private static void clear(int m, BitSet[] masks) {
        masks[m].clear(0, 64);
    }

    private static void flip(int m, BitSet[] masks) {
        masks[m].flip(0, 64);
    }

    private static int all(int m, BitSet[] masks) {
        return masks[m].cardinality() == 64 ? 1 : 0;
    }

    private static int any(int m, BitSet[] masks) {
        return masks[m].cardinality() > 0 ? 1 : 0;
    }

    private static int none(int m, BitSet[] masks) {
        return masks[m].cardinality() == 0 ? 1 : 0;
    }

    private static int count(int m, BitSet[] masks) {
        return masks[m].cardinality();
    }

    private static long val(int m, BitSet[] masks) {
        long val = 0;
        for (int i = 0; i < 64; i++) {
            if (masks[m].get(i)) {
                val |= 1L << i;
            }
        }
        return val;
    }
}
