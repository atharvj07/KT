#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

int main() {
    string a, b;
    cin >> a >> b;
    map<char, int> ma, mb;
    for (char c : a) ma[c]++;
    for (char c : b) mb[c]++;
    int ans = 0;
    for (auto [c, x] : ma) if (!mb.count(c)) ans += x;
    for (auto [c, x] : mb) if (!ma.count(c)) ans += x;
    if (ans == 0) cout << -1;
    else cout << ans;
    return 0;
}
