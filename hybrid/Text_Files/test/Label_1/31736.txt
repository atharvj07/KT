import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] arr = new int[m];
        int si = 0;
        int ei = m - 1;
        for (int i = 1; i <= 2 * n; i++) {
            for (int j = 0; j < 2; j++) {
                if (si > ei) {
                    break;
                }
                arr[si++] = i;
                arr[ei--] = i;
            }
        }
        for (int i : arr) {
            System.out.print(i + " ");
        }
    }
}
