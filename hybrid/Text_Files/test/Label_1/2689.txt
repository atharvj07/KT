#include <bits/stdc++.h>
using namespace std;
using ll=long long;
int main() {
  int n; cin>>n; ll a[n];
  for(int i=0; i<n; i++) cin>>a[i];
  vector<int> g[n];  // graph represent
  for(int i=0; i<n; i++) for(int j=0; j<n; j++) {
    if(i==j) continue;
    if(a[i]&a[j]) g[i].push_back(j);
  }
  vector<int> vis(n,0);
  vector<int> from(n,0);
  vector<int> len(n,0);
  queue<int> q;
  int ans=INT_MAX;
  for(int st=0; st<n; st++) {
    if(vis[st]) continue;
    q.push(st); vis[st]=1; len[st]=1;
    while(!q.empty()) {
      int x=q.front(); q.pop();
      for(int i:g[x]) {
        if(i==from[x]) continue;
        if(vis[i]==0) {
          from[i]=x; len[i]=len[x]+1; q.push(i);
          vis[i]=1;
        } else {
          ans=min(ans, len[x]+len[i]+1);
        }
      }
    }
  }
  if(ans==INT_MAX) ans=-1;
  cout<<ans<<'\n';
}
