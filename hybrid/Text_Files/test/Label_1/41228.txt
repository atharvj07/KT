a_1, b_1 = map(int, input().split())
a_2, b_2 = map(int, input().split())
a_3, b_3 = map(int, input().split())

# Check if paintings can be placed side by side or one above the other
if ((a_2 + a_3 <= a_1 and max(b_2, b_3) <= b_1) or
    (a_2 + a_3 <= b_1 and max(b_2, b_3) <= a_1) or
    (b_2 + b_3 <= b_1 and max(a_2, a_3) <= a_1) or
    (b_2 + b_3 <= a_1 and max(a_2, a_3) <= b_1)):
    print("YES")
else:
    print("NO")
