#include <iostream>
#include <vector>
using namespace std;

int n;
vector<string> board;
vector<vector<int>> ans;
vector<vector<int>> moves;
vector<vector<int>> sums;

bool inRange(int x, int y) {
    return 0 <= x && x < n && 0 <= y && y < n;
}

bool valid(int x, int y) {
    for (auto& move : moves) {
        int nx = x + move[0], ny = y + move[1];
        if (inRange(nx, ny) && board[nx][ny] == 'x') return false;
    }
    return true;
}

bool dfs(int x, int y) {
    if (x == n) {
        ans = sums;
        return true;
    }
    if (board[x][y] != '.') return false;
    for (auto& move : moves) {
        int nx = x + move[0], ny = y + move[1];
        if (!inRange(nx, ny) || board[nx][ny] == 'x' || !valid(nx, ny)) continue;
        sums[nx][ny] = 1;
        if (dfs(x + 1, y)) return true;
        sums[nx][ny] = 0;
    }
    return false;
}

int main() {
    cin >> n;
    board.resize(n);
    for (int i = 0; i < n; ++i) {
        cin >> board[i];
    }
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (board[i][j] == 'o') {
                moves.clear();
                for (int dx = -2; dx <= 2; ++dx) {
                    for (int dy = -2; dy <= 2; ++dy) {
                        if (dx != 0 || dy != 0) {
                            moves.push_back({dx, dy});
                        }
                    }
                }
                sums.assign(n, vector<int>(n, 0));
                sums[i][j] = 1;
                if (dfs(i + 1, 0)) {
                    cout << "YES\n";
                    for (int x = 0; x < n; ++x) {
                        for (int y = 0; y < n; ++y) {
                            cout << (sums[x][y] == 1 ? 'x' : '.');
                        }
                        cout << endl;
                    }
                    return 0;
                }
            }
        }
    }
    cout << "NO\n";
    return 0;
}
