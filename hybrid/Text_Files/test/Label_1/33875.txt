import java.util.Scanner;

public class StringValue {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String S = sc.next();
        int K = sc.nextInt();

        int N = S.length();

        // dp[i][j] := the longest common subsequence length of S[0..i) and S[j..N-1-j)
        int[][] dp = new int[N + 1][N + 1];

        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                if (S.charAt(i - 1) == S.charAt(N - j)) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                } else {
                    dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
                }
            }
        }

        int maxLen = dp[N][N];

        // Iterate over all possible positions to change characters
        for (int i = 0; i < N; i++) {
            for (char c = 'a'; c <= 'z'; c++) {
                if (S.charAt(i) == c) continue;

                // Calculate the new length
                int newLen = 0;
                if (S.charAt(i) == S.charAt(N - 1 - i)) {
                    newLen = Math.max(dp[N][N] - 1, dp[i][N - i - 1] + 1);
                } else {
                    newLen = dp[i][N - i - 1] + 1;
                }

                if (newLen + K >= maxLen) {
                    maxLen = newLen + K;
                }
            }
        }

        System.out.println(maxLen);
    }
}
