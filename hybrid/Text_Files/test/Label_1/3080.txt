#include <bits/stdc++.h>
using namespace std;
char b[9][9];
bool visw[9][9], visb[9][9];
int dx[] = {0, 0, -1, 1}, dy[] = {-1, 1, 0, 0};
int main() {
  for (int i = 1; i <= 8; i++)
    for (int j = 1; j <= 8; j++) { cin >> b[i][j]; }
  queue<pair<int, int>> qw, qb;
  for (int i = 8; i >= 1; i--)
    for (int j = 1; j <= 8; j++)
      if (b[i][j] == 'W') {
        qw.push({i, j});
        visw[i][j] = 1;
      }
  for (int i = 1; i <= 8; i++)
    for (int j = 1; j <= 8; j++)
      if (b[i][j] == 'B') {
        qb.push({i, j});
        visb[i][j] = 1;
      }
  while (!qw.empty() && !qb.empty()) {
    int sz = qw.size();
    while (sz--) {
      int x = qw.front().first, y = qw.front().second;
      qw.pop();
      if (x == 1) { cout << "A" << endl; return 0; }
      for (int i = 0; i < 4; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (nx < 1 || nx > 8 || ny < 1 || ny > 8 || visw[nx][ny] ||
            b[nx][ny] == 'B')
          continue;
        visw[nx][ny] = 1;
        qw.push({nx, ny});
      }
    }
    sz = qb.size();
    while (sz--) {
      int x = qb.front().first, y = qb.front().second;
      qb.pop();
      if (x == 8) { cout << "B" << endl; return 0; }
      for (int i = 0; i < 4; i++) {
        int nx = x + dx[i], ny = y + dy[i];
        if (nx < 1 || nx > 8 || ny < 1 || ny > 8 || visb[nx][ny] ||
            b[nx][ny] == 'W')
          continue;
        visb[nx][ny] = 1;
        qb.push({nx, ny});
      }
    }
  }
  return 0;
}
