#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<char>> grid(n, vector<char>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> grid[i][j];
    }
  }

  bool possible = false;
  int x, y;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if (grid[i][j] == '*') {
        bool row_ok = true;
        bool col_ok = true;
        for (int k = 0; k < n; k++) {
          if (grid[k][j] != '*') {
            row_ok = false;
          }
        }
        for (int k = 0; k < m; k++) {
          if (grid[i][k] != '*') {
            col_ok = false;
          }
        }
        if (row_ok || col_ok) {
          possible = true;
          x = i + 1;
          y = j + 1;
        }
      }
    }
  }

  if (possible) {
    cout << "YES" << endl;
    cout << x << " " << y << endl;
  } else {
    cout << "NO" << endl;
  }

  return 0;
}
