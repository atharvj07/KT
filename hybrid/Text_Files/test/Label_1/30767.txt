import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
  static final int MODULO = 1000000007;
  static final int INF = 987654321;
  static int n, m, k;
  static List<int[]> cuts;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());
    k = Integer.parseInt(st.nextToken());
    cuts = new ArrayList<>();
    for (int i = 0; i < k; i++) {
      st = new StringTokenizer(br.readLine());
      cuts.add(new int[] {
        Integer.parseInt(st.nextToken()),
        Integer.parseInt(st.nextToken()),
        Integer.parseInt(st.nextToken()),
        Integer.parseInt(st.nextToken())
      });
    }

    int result = solve();
    if (result == 1) {
      System.out.println("FIRST");
      System.out.println(getFirstMove());
    } else {
      System.out.println("SECOND");
    }
  }

  public static int solve() {
    int[][][] dp = new int[n + 1][m + 1][4];
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          dp[i][j][k] = -INF;
        }
      }
    }

    dp[0][0][0] = 0;
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          if (dp[i][j][k] >= 0) {
            // Horizontal cut
            int nextK = (k + 1) % 4;
            if (j + 1 <= m) {
              dp[i][j + 1][nextK] = Math.max(dp[i][j + 1][nextK], dp[i][j][k] + getScore(i, j, i, j + 1, nextK));
            }
            // Vertical cut
            nextK = (k + 3) % 4;
            if (i + 1 <= n) {
              dp[i + 1][j][nextK] = Math.max(dp[i + 1][j][nextK], dp[i][j][k] + getScore(i, j, i + 1, j, nextK));
            }
          }
        }
      }
    }

    int result = -INF;
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          result = Math.max(result, dp[i][j][k]);
        }
      }
    }
    if (result < 0) {
      result = 0;
    }
    return result % 2;
  }

  public static int getScore(int x1, int y1, int x2, int y2, int dir) {
    if (dir == 0) {
      // Horizontal cut
      int left = y1;
      int right = y2 - 1;
      for (int[] cut : cuts) {
        if (x1 <= cut[2] && cut[0] <= x2 && cut[1] <= y1 && y1 < cut[3]) {
          left = Math.max(left, cut[3]);
        }
        if (x1 <= cut[2] && cut[0] <= x2 && cut[1] <= y2 && y2 < cut[3]) {
          right = Math.min(right, cut[1] - 1);
        }
      }
      return right - left + 1;
    } else {
      // Vertical cut
      int top = x1;
      int bottom = x2 - 1;
      for (int[] cut : cuts) {
        if (y1 <= cut[3] && cut[1] <= y2 && cut[0] <= x1 && x1 < cut[2]) {
          top = Math.max(top, cut[2]);
        }
        if (y1 <= cut[3] && cut[1] <= y2 && cut[0] <= x2 && x2 < cut[2]) {
          bottom = Math.min(bottom, cut[0] - 1);
        }
      }
      return bottom - top + 1;
    }
  }

  public static String getFirstMove() {
    int[][][] dp = new int[n + 1][m + 1][4];
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          dp[i][j][k] = -INF;
        }
      }
    }

    dp[0][0][0] = 0;
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          if (dp[i][j][k] >= 0) {
            // Horizontal cut
            int nextK = (k + 1) % 4;
            if (j + 1 <= m) {
              dp[i][j + 1][nextK] = Math.max(dp[i][j + 1][nextK], dp[i][j][k] + getScore(i, j, i, j + 1, nextK));
            }
            // Vertical cut
            nextK = (k + 3) % 4;
            if (i + 1 <= n) {
              dp[i + 1][j][nextK] = Math.max(dp[i + 1][j][nextK], dp[i][j][k] + getScore(i, j, i + 1, j, nextK));
            }
          }
        }
      }
    }

    int result = -INF;
    int x1 = -1;
    int y1 = -1;
    int x2 = -1;
    int y2 = -1;
    int dir = -1;
    for (int i = 0; i <= n; i++) {
      for (int j = 0; j <= m; j++) {
        for (int k = 0; k < 4; k++) {
          if (dp[i][j][k] > result) {
            result = dp[i][j][k];
            x1 = i;
            y1 = j;
            if (k == 0 || k == 2) {
              x2 = i;
              y2 = j + 1;
            } else {
              x2 = i + 1;
              y2 = j;
            }
            dir = k;
          }
        }
      }
    }

    return x1 + " " + y1 + " " + x2 + " " + y2;
  }
}
