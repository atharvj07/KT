import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), l = in.nextInt();
        int[] x = new int[k], a = new int[l];
        for (int i = 0; i < k; i++) x[i] = in.nextInt();
        for (int i = 0; i < l; i++) a[i] = in.nextInt();

        Set<Integer> set = new HashSet<>();
        for (int i : x) set.add(i);

        int ans = 0, cur = -1, max = -1;
        for (int i = 0; i < l; i++) {
            if (!set.contains(a[i])) {
                if (max != -1) {
                    ans += Math.min(max - cur, a[i] - max - 1);
                }
                cur = a[i];
                max = -1;
            } else {
                if (max == -1) max = a[i];
            }
            if (i == l - 1 && max != -1) {
                ans += Math.min(max - cur, n - max - 1);
            }
        }
        System.out.println(ans == 0 ? -1 : ans);
    }
}
