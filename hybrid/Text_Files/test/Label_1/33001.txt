import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);

        int n = cin.nextInt(), k = cin.nextInt();
        String s = cin.next();

        char[] t = s.toCharArray();
        int changed = 0;

        for (int i = 0; i < n; i++) {
            if (t[i] > '0') {
                t[i] = '0';
                changed++;
                if (changed == k + 1) break;
            }
        }
        if (changed <= k) t[n - changed] = '0';

        System.out.println(new String(t));
    }
}
