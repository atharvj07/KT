#include <bits/stdc++.h>

using namespace std;

const int MOD = 1e9 + 7;
const int MX = 1e5 + 5;

long long fastpow(long long b, long long p) {
    if (!p)
        return 1LL;
    if (p % 2 == 0) {
        long long x = fastpow(b, p / 2);
        return x * x % MOD;
    }
    return b * fastpow(b, p - 1) % MOD;
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n + 1), b(n + 1);
    vector<int> cnt(m + 1), fact(m + 1);
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) { cin >> b[i]; }
    fact[0] = 1;
    for (int i = 1; i <= m; i++) { fact[i] = 1LL * i * fact[i - 1] % MOD; }
    for (int i = 1; i <= m; i++) {
        if (a[i] == 0) { cnt[a[i]]++; }
        if (b[i] == 0) { cnt[b[i]]++; }
    }
    long long ans = 1LL;
    for (int i = 0; i <= m; i++) {
        ans = 1LL * ans * fastpow(cnt[i], cnt[i]) % MOD;
    }
    cnt[0] = 1;
    long long tot = 1LL;
    for (int i = 0; i <= m; i++) {
        tot = 1LL * tot * cnt[i] % MOD;
    }
    ans = 1LL * ans * fastpow(tot, MOD - 2) % MOD;
    cout << ans << endl;
}
