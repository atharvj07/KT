import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        System.out.println(pw(2, n / 2));
    }

    static long pw(long a, long b) {
        if (b == 0)
            return 1;
        if (b == 1)
            return a;
        long half = pw(a, b / 2);
        long ans = (half * half) % 998244353;
        if (b % 2 == 1)
            ans = (ans * a) % 998244353;
        return ans;
    }
}
