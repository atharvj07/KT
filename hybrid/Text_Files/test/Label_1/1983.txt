#include <bits/stdc++.h>

#define rep(i,a,b) for(int i=a;i<=b;i++)
#define newline printf("\n")
#define chk(a,mask) (((mask)&(1<<(a)))!=0)
using namespace std;

int main()
{
    int n;
    cin>>n;
    int ans[n+5];
    int G[n+5][n+5];
    rep(i,1,n)
    {
        rep(j,1,n)
        {
            cin>>G[i][j];
        }
    }
    ans[1]=G[1][2];
    rep(i,2,n)
    {
        ans[i]=G[1][i]/ans[1];
    }
    rep(i,1,n) cout<<ans[i]<<" ";
    newline;
    return 0;
}
