#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s, t;
    cin >> s >> t;
    sort(s.begin(), s.end());
    for (int i = 0; i < s.size(); i++) {
      if (t[0] == s[i]) {
        if (i < s.size() - 1) swap(s[i], s[i + 1]);
        break;
      }
    }
    int p1 = 0, p2 = 0;
    while (p1 < s.size() && p2 < t.size()) {
      if (s[p1] == t[p2]) p2++;
      p1++;
    }
    if (p2 < t.size()) swap(s[0], s[1]);
    cout << s << '\n';
  }
  return 0;
}
