#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, mx = 0, cur = 0;
  cin >> n;
  vector<int> a(n);
  for (int &x : a) cin >> x;
  for (int x : a) cur = x ? cur + 1 : 0, mx = max(mx, cur);
  for (int x : a) cur = x ? cur + 1 : 0, mx = max(mx, cur);
  cout << mx << '\n';
}
