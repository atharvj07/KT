import java.util.Scanner;
import java.util.ArrayList;

public class Main {

  static int n, l, m, w;
  static int[][] quests;
  static boolean[][] visited;
  static int[] path;
  static int maxSum;

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    n = scanner.nextInt();
    quests = new int[n][3];
    visited = new boolean[n][3];
    path = new int[n];
    maxSum = Integer.MIN_VALUE;

    for (int i = 0; i < n; i++) {
      l = scanner.nextInt();
      m = scanner.nextInt();
      w = scanner.nextInt();
      quests[i] = new int[]{l, m, w};
    }

    dfs(0, 0, 0, 0);

    if (maxSum == Integer.MIN_VALUE) {
      System.out.println("Impossible");
    } else {
      for (int i = 0; i < n; i++) {
        System.out.println(path[i] == 0 ? "LM" : path[i] == 1 ? "MW" : "LW");
      }
    }
  }

  public static void dfs(int currentQuest, int currentSum, int companion1, int companion2) {
    if (currentQuest == n) {
      if (currentSum > maxSum) {
        maxSum = currentSum;
        for (int i = 0; i < n; i++) {
          path[i] = companion1 * 2 + companion2;
        }
      }
      return;
    }

    for (int i = 0; i < 3; i++) {
      if (visited[currentQuest][i]) {
        continue;
      }

      visited[currentQuest][i] = true;

      int newSum = currentSum + quests[currentQuest][i];
      int newCompanion1 = companion1, newCompanion2 = companion2;

      if (i == 0) {
        newCompanion1 = 1;
      } else if (i == 1) {
        newCompanion2 = 1;
      }

      dfs(currentQuest + 1, newSum, newCompanion1, newCompanion2);
      visited[currentQuest][i] = false;
    }
  }
}
