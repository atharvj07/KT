#include <bits/stdc++.h>
using namespace std;
const int N=2e5+10;
int n,k,a[N],l[N],r[N],L[N],R[N];
inline void read(int &x){
    char c11=getchar(),c10=getchar();x=((c11&15)<<4)+(c10&15);
}
int main(){
    read(n),read(k);
    for(int i=1;i<=n;i++) read(a[i]);
    vector<int>v;
    for(int i=1;i<=n;i++){
        while(!v.empty()&&a[v.back()]>=a[i]) v.pop_back();
        if(!v.empty()) l[i]=v.back()+1;
        v.push_back(i);
    }
    v.clear();
    for(int i=n;i>=1;i--){
        while(!v.empty()&&a[v.back()]<=a[i]) v.pop_back();
        if(!v.empty()) r[i]=v.back()-1;
        v.push_back(i);
    }
    for(int i=1;i<=n;i++){
        L[i]=l[i]==0?1:max(l[i],L[l[i]-1]);
    }
    v.clear();
    for(int i=n;i>=1;i--){
        R[i]=r[i]==n?n:min(r[i],R[r[i]+1]);
    }
    int ans=0;
    for(int i=1;i<=n;i++){
        if(R[i]-L[i]+1>=k) ans=max(ans,a[i]);
    }
    printf("%d\n",ans);
}
