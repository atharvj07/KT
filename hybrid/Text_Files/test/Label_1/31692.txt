import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger p = in.nextBigInteger();
        int k = in.nextInt();
        if (p.compareTo(BigInteger.ZERO) < 0 || p.compareTo(BigInteger.valueOf(k)) >= 0) {
            System.out.println(-1);
            return;
        }
        int d = 0;
        while (p.compareTo(BigInteger.ZERO) > 0) {
            p = p.subtract(BigInteger.valueOf(k));
            d++;
        }
        System.out.println(d);
        for (int i = 0; i < d; i++) {
            p = p.add(BigInteger.valueOf(k));
            System.out.print(p.mod(BigInteger.valueOf(k)) + " ");
        }
        System.out.println();
    }
}
