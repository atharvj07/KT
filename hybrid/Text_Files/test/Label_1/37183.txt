n, a = map(int, input().split())

best_diff = float('inf')
best_v1 = 1
best_v2 = 2
best_v3 = 3

for v2 in range(2, n):
    angle = 180 * (v2 - 1) / n
    diff = abs(angle - a)
    if diff < best_diff:
        best_diff = diff
        best_v2 = 1
        best_v1 = v2
        best_v3 = v2 + 1

print(best_v1, best_v2, best_v3)
