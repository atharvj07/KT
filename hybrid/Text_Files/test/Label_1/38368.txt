import sys
input = sys.stdin.readline
n, m = [int(i) for i in input().split()]
a = [[int(i) for i in input().split()] for i in range(n)]
ans = 99999999999999999999999999999999999999999999999999999999999999999
for i in range(n):
    for j in range(m):
        if a[i][j] == 1:
            for k in range(n):
                for l in range(m):
                    if a[k][l] == 0:
                        ans = min(ans, abs(k - i) + abs(l - j) + 1)
print(ans)
