import java.util.Arrays;
import java.util.Scanner;
import java.util.TreeMap;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] r = new int[m];
        for (int i = 0; i < m; i++) {
            int li = sc.nextInt();
            int ri = sc.nextInt();
            r[i] = ri;
        }
        Arrays.sort(r);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            int left = -1;
            int right = m;
            while (left < right - 1) {
                int mid = (left + right) / 2;
                if (r[mid] <= y) {
                    left = mid;
                } else {
                    right = mid;
                }
            }
            if (left == -1) {
                System.out.println(y);
            } else if (r[left] >= x) {
                System.out.println(y);
            } else {
                System.out.println(r[left]);
            }
        }
    }
}
