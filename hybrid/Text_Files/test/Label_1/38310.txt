from collections import defaultdict

n = int(input())
a = [int(x) for x in input().split()]
b = sorted(a)
p = [0] * n
pinv = [0] * n
for i, x in enumerate(a):
    p[b.index(x)] = i
    pinv[i] = b.index(x)
d = defaultdict(int)
for x in a:
    d[x] += 1
f = [0] * n
cnt = 0
for i in p:
    if i + 1 in d and d[i + 1] > 0:
        f[pinv[i + 1]] += a[i]
        d[i + 1] -= 1
    else:
        cnt += a[i]
ans = cnt
for x in f:
    ans += x
print(ans % (10 ** 9 + 7))
