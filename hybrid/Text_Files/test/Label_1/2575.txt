#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    string m;
    cin >> m;
    int ans = 0;
    for (int x = 0; x < (1 << n); x++) {
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if ((x & (1 << i)) != 0) {
                sum += a[i];
            }
        }
        if (sum > ans) {
            ans = sum;
        }
    }
    cout << ans << endl;
    return 0;
}
