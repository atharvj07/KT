import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] x = new int[n];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) x[i] = Integer.parseInt(st.nextToken());
        List<Integer> z = new ArrayList<>();
        Map<Integer, Integer> cnt = new HashMap<>();
        for (int i : x) {
            int y = 0, j = 1;
            for (; y + j <= i; y+=j, j<<=1) cnt.put(j, cnt.getOrDefault(j, 0) + 1);
            cnt.put(i - y, cnt.getOrDefault(i - y, 0) + 1);
        }
        int[] k = new int[cnt.keySet().size()];
        int i = 0;
        for (int num : cnt.keySet()) k[i++] = num;
        Arrays.sort(k);
        label:
        for (i = 1; i <= n; i++) {
            int[] count = new int[cnt.keySet().size()];
            for (int key : cnt.keySet()) {
                int j = Arrays.binarySearch(k, key);
                if (j < 0) continue label;
                int p = count[j], q = i - p;
                if ((j == 0 || k[j - 1] < q) && q >= 0 && (j == k.length - 1 || q < k[j + 1])) count[j] = p + 1;
                else continue label;
            }
            z.add(i);
        }
        if (z.size() == 0) z.add(-1);
        for (int v : z) System.out.print(v + " ");
        System.out.println();
    }
}
