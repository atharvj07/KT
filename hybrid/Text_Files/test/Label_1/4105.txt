#include <bits/stdc++.h>

using namespace std;

const int M = 5005;
// n : number of nodes, m : number of one-way edges
int n, m, s, ans;
int to[M << 1], nxt[M << 1], head[M], w[M << 1], cnt;
int u[M], v[M];
bool vis[M];

inline void add(int a, int b, int c) {
    to[++cnt] = b;
    nxt[cnt] = head[a];
    head[a] = cnt;
    w[cnt] = c;
}

void dfs(int u) {
    vis[u] = true;
    for (int p = head[u]; p; p = nxt[p]) {
        int v = to[p];
        if (!vis[v]) dfs(v);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m >> s;

    for (int i = 1; i <= m; ++i) cin >> u[i] >> v[i];

    for (int i = 1; i <= m; ++i) {
        u[i] ^= v[i], v[i] ^= u[i], u[i] ^= v[i];
        add(u[i], v[i], 1);
    }

    dfs(s);

    for (int i = 1; i <= n; ++i) {
        if (!vis[i]) {
            u[++m] = s, v[m] = i;
            add(u[m], v[m], 1);
            ++ans;
        }
    }

    cout << ans << '\n';

    return 0;
}
