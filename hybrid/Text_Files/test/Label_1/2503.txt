#include <iostream>
#include <vector>
#include <map>
using namespace std;

int main() {
    string s; getline(cin, s);
    int n, pos = 0; cin >> n, s += ' ';
    map<int, int> m;
    for (char c : s) {
        if (c == '?') {
            m[pos] = n--;
            pos++;
        } else if (c == '+') pos += 2;
        else if (c == '-') pos++;
    }
    vector<int> ans(pos);
    int l = 0, r = pos - 1;
    for (auto it : m) {
        ans[it.first] = (it.second + 1) / 2;
        if (s[it.first - 1] == '+') l += ans[it.first];
        else r -= ans[it.first];
    }
    if (l == r) {
        cout << "Possible\n";
        for (int x : ans) cout << x << ' '; cout << '\n';
    } else cout << "Impossible\n";
}
