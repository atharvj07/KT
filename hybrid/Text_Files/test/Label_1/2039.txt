#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    cin >> k;
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (v[i].first > k || v[i].second < k) ans++;
    }
    cout << ans << endl;
    return 0;
}
