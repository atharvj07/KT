import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), x = input.nextInt();
        long sum = 0, time = x;
        for (int i = 0; i < n; i++) {
            long num = input.nextInt();
            sum += time * num;
            time = Math.max(1, time - 1);
        }
        System.out.println(sum);
    }
}
