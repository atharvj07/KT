import sys

def read_line(): return list(map(int, sys.stdin.readline().split()))

n, m, q = read_line()

grid = [read_line() for _ in range(n)]

for _ in range(q):
    i, j = read_line()
    grid[i-1][j-1] ^= 1

    ans = 0
    for x in range(n):
        cnt = 0
        for y in range(m):
            cnt += grid[x][y]
            ans = max(ans, cnt)
            if grid[x][y] == 0:
                cnt = 0

    print(ans)
