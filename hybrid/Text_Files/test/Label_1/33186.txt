class Main {

    public static void main(String[] args) {
        try (var input = new java.util.Scanner(System.in)) {
            var a = input.nextInt();
            var b = input.nextInt();
            if (a % 2 == 0 || b % 2 == 0) {
                System.out.println("Even");
            } else {
                System.out.println("Odd");
            }
        }
    }
}
