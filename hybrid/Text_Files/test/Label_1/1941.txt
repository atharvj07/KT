#include <bits/stdc++.h>
using namespace std;
#define endl "\n"
void solve() {
    int n, k; cin >> n >> k;
    deque<int> dq;
    set<int> active;
    vector<int> ans;
    for (int i = 0; i < n; i++) {
        int id; cin >> id;
        if (active.count(id)) {
            dq.erase(find(dq.begin(), dq.end(), id));
        } else if (active.size() == k) {
            active.erase(dq.back());
            dq.pop_back();
        }
        dq.push_front(id);
        active.insert(id);
    }
    cout << dq.size() << endl;
    for (const int& x : dq) cout << x << ' ';
    cout << endl;
}
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr); cout.tie(nullptr);
    int t; cin >> t;
    while (t--) solve();
    return 0;
}
