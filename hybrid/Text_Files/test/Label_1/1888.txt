#include <algorithm>
#include <iostream>
using namespace std;

int main() {
  int a, b, n = 0, m = 0;
  cin >> a >> b;
  for (int i = 1; a >= i; i++) {
    n++;
    a -= i;
  }
  for (int i = 1; b >= i; i++) {
    m++;
    b -= i;
  }
  cout << n << endl;
  for (int i = 1; i <= n; i++) { cout << i << " "; }
  cout << endl;
  cout << m << endl;
  for (int i = 1; i <= m; i++) { cout << i << " "; }
  cout << endl;
}
