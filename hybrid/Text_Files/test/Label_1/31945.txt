import java.util.Scanner;

public class RoadTrip {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a1 = scanner.nextInt();
        int b1 = scanner.nextInt();
        int a2 = scanner.nextInt();
        int b2 = scanner.nextInt();
        int a3 = scanner.nextInt();
        int b3 = scanner.nextInt();

        boolean[][] connected = new boolean[5][5];
        connected[a1][b1] = true;
        connected[b1][a1] = true;
        connected[a2][b2] = true;
        connected[b2][a2] = true;
        connected[a3][b3] = true;
        connected[b3][a3] = true;

        boolean[] visited = new boolean[5];
        visited[1] = true;

        int count = 1;
        dfs(connected, visited, 1, count);

        if (count == 4) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    static void dfs(boolean[][] connected, boolean[] visited, int node, int count) {
        for (int i = 1; i < 5; i++) {
            if (connected[node][i] && !visited[i]) {
                visited[i] = true;
                dfs(connected, visited, i, count + 1);
                visited[i] = false;
            }
        }
    }
}
