#include<bits/stdc++.h>
using namespace std;
int n,m,k,a[105][105][105];
int dx[]={0,0,0,1,1,-1,-1,0};
int dy[]={0,0,-1,0,1,0,1,0};
int dz[]={1,-1,0,0,0,0,0,0};
bool vis[105][105][105];
void f(int x,int y,int z)
{
	if(vis[x][y][z])return ;
	vis[x][y][z]=1;
	for(int d=0;d<8;++d)
	{
		int nx=x+dx[d],ny=y+dy[d],nz=z+dz[d];
		if(nx>0 && nx<=n && ny>0 && ny<=m && nz>0 && nz<=k && a[nx][ny][nz])
		f(nx,ny,nz);
	}
}
int main()
{
	cin>>n>>m>>k;
	for(int i=1;i<=n;++i)
	{
		for(int j=1;j<=m;++j)
		{
			for(int l=1;l<=k;++l)
			{
				char c;
				cin>>c;
				a[i][j][l]=(c=='1');
			}
		}
	}
	int ans=0;
	for(int i=1;i<=n;++i)
	{
		for(int j=1;j<=m;++j)
		{
			for(int l=1;l<=k;++l)
			{
				if(a[i][j][l])
				{
					memset(vis,0,sizeof(vis));
					f(i,j,l);
					bool f=0;
					for(int d=0;d<8;++d)
					{
						int nx=i+dx[d],ny=j+dy[d],nz=l+dz[d];
						if(nx>0 && nx<=n && ny>0 && ny<=m && nz>0 && nz<=k && a[nx][ny][nz])
						{
							if(!vis[nx][ny][nz])f=1;
						}
					}
					if(f)++ans;
				}
			}
		}
	}
	cout<<ans<<endl;
	return 0;
}
