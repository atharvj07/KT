#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> d(n);
    vector<int> x(n);
    for (int i = 0; i < n; i++) {
        char q;
        cin >> q;
        d[i] = (q == 'R' ? 1 : -1);
    }
    for (int i = 0; i < n; i++) {
        cin >> x[i];
    }
    long long ans = -1;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if ((x[j] - x[i] + d[i] - d[j]) % 2 != 0) {
                long long t = (x[i] - x[j]) / 2;
                if (ans == -1 || t < ans) {
                    ans = t;
                }
            }
        }
    }
    cout << ans << endl;
    return 0;
}
