#include <iostream>
#include <vector>
#include <algorithm>
int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (auto&& ai : a) std::cin >> ai;
  std::sort(a.begin(), a.end());
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    a[i] ^= ans;
    ans = std::max(ans, (long long)a[i]);
  }
  std::cout << ans << "\n";
  return 0;
}
