import java.io.*;
import java.util.*;

public class Main {
	static int mod = 998244353;
	static long[] fact = new long[300005];
	static long[] inv = new long[300005];

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int k = Integer.parseInt(st.nextToken());
		int[] col = new int[n];
		st = new StringTokenizer(br.readLine());
		for (int i = 0; i < n; i++)
			col[i] = Integer.parseInt(st.nextToken());
		Map<Integer, Integer> mp = new HashMap<>();
		for (int x : col)
			mp.put(x, mp.getOrDefault(x, 0) + 1);
		if (mp.size() == 1) {
			System.out.println(n - 1);
			return;
		}
		int[] deg = new int[n];
		int a = -1;
		int b = -1;
		for (int i = 0; i < n - 1; i++) {
			st = new StringTokenizer(br.readLine());
			int x = Integer.parseInt(st.nextToken()) - 1;
			int y = Integer.parseInt(st.nextToken()) - 1;
			if (x > y) {
				int tmp = x;
				x = y;
				y = tmp;
			}
			if (a == -1 && col[x] == col[y])
				continue;
			if (a != -1 && (col[x] != col[y] || (col[x] == col[y] && deg[x] > 0 && deg[y] > 0))) {
				b = i;
				break;
			}
			deg[x]++;
			deg[y]++;
			a = i;
		}
		long ans = (fact[b] * inv[a] % mod) * inv[b - a] % mod;
		ans = (ans * k % mod) * (k - 1) % mod;
		System.out.println(ans);
	}

	static void pre() {
		fact[0] = 1;
		for (int i = 1; i < 300005; i++)
			fact[i] = (fact[i - 1] * i) % mod;
		for (int i = 300004; i >= 1; i--)
			inv[i] = power(fact[i], mod - 2);
	}

	static long power(long base, long p) {
		if (p == 0)
			return 1;
		if (p == 1)
			return base;
		if (p % 2 == 1)
			return (base * power(base, p - 1)) % mod;
		long val = power(base, p / 2);
		return (val * val) % mod;
	}
}
