import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Auction {

    static class Bid {
        int participantId;
        int bidSize;

        public Bid(int participantId, int bidSize) {
            this.participantId = participantId;
            this.bidSize = bidSize;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Bid[] bids = new Bid[n];
        for (int i = 0; i < n; i++) {
            bids[i] = new Bid(scanner.nextInt(), scanner.nextInt());
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            int[] excludedParticipants = new int[k];
            for (int j = 0; j < k; j++) {
                excludedParticipants[j] = scanner.nextInt();
            }
            int winner = findWinner(bids, excludedParticipants);
            if (winner == -1) {
                System.out.println("0 0");
            } else {
                System.out.println(winner + " " + bids[winner].bidSize);
            }
        }
    }

    private static int findWinner(Bid[] bids, int[] excludedParticipants) {
        List<Bid> remainingBids = new ArrayList<>();
        for (Bid bid : bids) {
            if (!contains(excludedParticipants, bid.participantId)) {
                remainingBids.add(bid);
            }
        }
        if (remainingBids.isEmpty()) {
            return -1;
        }
        int winner = -1;
        int winningBidSize = Integer.MIN_VALUE;
        for (Bid bid : remainingBids) {
            if (bid.bidSize > winningBidSize) {
                winner = bid.participantId;
                winningBidSize = bid.bidSize;
            }
        }
        return winner;
    }

    private static boolean contains(int[] array, int value) {
        return Arrays.stream(array).anyMatch(i -> i == value);
    }
}
