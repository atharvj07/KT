#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MAXN = 300000;

int n, r;
int x[MAXN], y[MAXN];

int main() {
  cin >> n >> r;
  for (int i = 0; i < n; ++i) cin >> x[i] >> y[i];

  int ans = 0;
  for (int i = 0; i < n; ++i) {
    int cnt = 0;
    for (int j = 0; j < n; ++j) {
      if (abs(x[i] - x[j]) + abs(y[i] - y[j]) <= r) ++cnt;
    }
    ans = max(ans, cnt);
  }
  cout << ans << "\n";
  return 0;
}
