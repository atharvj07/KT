#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  if (n == 1) {
    cout << "Alice" << endl;
    return 0;
  }
  if (is_sorted(a.begin(), a.end())) {
    cout << "Bob" << endl;
    return 0;
  }
  if (a[0] > a[1]) {
    cout << "Bob" << endl;
    return 0;
  }
  if (a[n - 1] < a[n - 2]) {
    cout << "Bob" << endl;
    return 0;
  }
  cout << "Alice" << endl;
  return 0;
}
