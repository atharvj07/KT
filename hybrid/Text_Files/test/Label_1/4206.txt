#include <iostream>
using namespace std;
int main() {
  int n, ans = 0, l[300001] = {0}, r[300001] = {0};
  cin >> n;
  for (int i, a; i < n; i++) {
    cin >> a;
    l[a] = ((l[a - 1]) ? l[a - 1] : i);
    r[a] = i;
  }
  for (int i = 1; i <= n; i++) ans += r[i] >= l[i] + (i - 1);
  cout << ans;
}
