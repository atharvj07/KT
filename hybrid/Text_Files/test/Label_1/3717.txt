#include <iostream>
#include <vector>

using namespace std;

typedef long long ll;

const int MAXN = 2e5 + 5;

vector<int> adj[MAXN];
int depth[MAXN], father[MAXN];

void dfs(int v, int p) {
  father[v] = p;
  depth[v] = depth[p] + 1;
  for (int u : adj[v]) {
    if (u == p) continue;
    dfs(u, v);
  }
}

int main() {
  int n, x;
  cin >> n >> x;
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  dfs(1, 1);
  int ans = 0;
  for (int v = 1; v <= n; v++) {
    if (v == x) continue;
    ans = max(ans, depth[v] + depth[x] - 2);
  }
  cout << ans << endl;
  return 0;
}
