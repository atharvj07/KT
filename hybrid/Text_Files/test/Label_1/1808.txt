#include <string>
#include <vector>
#include <iostream>

int main() {
    // Get the input string and number of queries
    std::string s;
    std::cin >> s;
    int m;
    std::cin >> m;

    // Process the queries
    std::vector<int> answers(m);
    for (int i = 0; i < m; i++) {
        int l, r;
        std::cin >> l >> r;
        answers[i] = s[l] < s[r];
    }

    // Output the answers
    for (int i = 0; i < m; i++) {
        std::cout << answers[i] << '\n';
    }

    return 0;
}
