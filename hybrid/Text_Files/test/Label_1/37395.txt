n = int(input())
a = list(map(int, input().split()))

if a[0] == 0:
    print(-1)
else:
    messages = []
    informed = [False] * n
    informed[0] = True
    for i in range(n):
        if a[i] > 0 and informed[i]:
            for j in range(a[i]):
                if i + j + 1 < n and not informed[i + j + 1]:
                    informed[i + j + 1] = True
                    messages.append((i + 1, i + j + 2))
                    a[i] -= 1
                else:
                    break
    if all(informed):
        print(len(messages))
        for message in messages:
            print(message[0], message[1])
    else:
        print(-1)
