from collections import deque
n,k=map(int,input().split())
adj=[[] for _ in range(n+1)]
in_degree=[0]*(n+1)
for _ in range(k):
    m,*a=map(int,input().split())
    for j in range(1,m):
        adj[a[j-1]].append(a[j])
        in_degree[a[j]]+=1
q=deque([x for x in range(1,n+1) if in_degree[x]==0])
ans=0
while q:
    size=len(q)
    for _ in range(size):
        i=q.popleft()
        ans+=1
        for j in adj[i]:
            in_degree[j]-=1
            if in_degree[j]==0:
                q.append(j)
print(ans)
