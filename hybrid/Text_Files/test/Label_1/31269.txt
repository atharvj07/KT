import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long L = sc.nextLong();
        long N = sc.nextLong();
        long[] X = new long[((int)N)];
        for (long i = 0; i < N; i++) {
            X[((int)i)] = sc.nextLong();
        }

        // clockwise
        long ans = (L - 1) * 2 + X[((int)(N - 1))];
        for (long i = 0; i < N - 1; i++) {
            ans += (Math.min(X[((int)(i + 1))] - X[((int)i)], L - (X[((int)(i + 1))] - X[((int)i)])));
        }

        // counter-clockwise
        long ans2 = (L - 1) * 2 + L - X[0];
        for (long i = 0; i < N - 1; i++) {
            ans2 += (Math.min(X[((int)(i + 1))] - X[((int)i)], L - (X[((int)(i + 1))] - X[((int)i)])));
        }

        System.out.println(Math.max(ans, ans2));
    }
}
