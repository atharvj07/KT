def max_bishwocks(top, bottom):
    n = len(top)
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        dp[i] = dp[i - 1]
        if i > 1 and top[i - 1] == '0' and bottom[i - 1] == '0' and bottom[i - 2] == '0':
            dp[i] = max(dp[i], dp[i - 2] + 1)
        if i > 1 and top[i - 2] == '0' and top[i - 1] == '0' and bottom[i - 1] == '0':
            dp[i] = max(dp[i], dp[i - 2] + 1)
    return dp[n]

top = input().strip()
bottom = input().strip()

print(max_bishwocks(top, bottom))
