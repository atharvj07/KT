import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
ops = [list(map(int, input().split())) for i in range(m)]
queries = [list(map(int, input().split())) for i in range(k)]
for x, y in queries:
    for l, r, d in ops[x-1:y]:
        a[l-1] += d
        if r < n:
            a[r] -= d
for i in a:
    print(i, end=' ')
