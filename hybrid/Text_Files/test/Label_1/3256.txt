#include<bits/stdc++.h>
using namespace std;
int n, k, a[5001], b[5001];
double ans;
int main() {
    cin >> n >> k;
    for(int i = 1; i <= n; i++)
        cin >> a[i];
    b[0] = 0;
    for(int i = 1; i <= n; i++)
        b[i] = b[i - 1] + a[i];
    for(int i = k; i <= n; i++)
        ans = max(ans, 1.0 * (b[i] - b[i - k]) / k);
    cout << ans;
    return 0;
}
