#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;
    string s,t;
    cin>>s>>t;
    bool v[n+5][m+5];
    memset(v,false,sizeof v);
    for(int i=0;i<s.length();i++){
        for(int j=0;j<t.length();j++){
            if((s[i]=='>' && t[j]=='^') || (s[i]=='<' && t[j]=='v')){
                v[i+1][j+1]=true;
            }
            else if((t[j]=='v' && s[i]=='^') || (t[j]=='^' && s[i]=='v')){
                v[i+1][j+1]=true;
            }
        }
    }
    bool vis[n+5][m+5];
    memset(vis,false,sizeof vis);
    queue<pair<int, int> > q;
    q.push({1,1});
    vis[1][1]=true;
    bool able=true;
    while(!q.empty()){
        int x,y;
        x=q.front().first;
        y=q.front().second;
        q.pop();
        for(int i=max(x-1,1);i<=min(x+1,n);i++){
            for(int j=max(y-1,1);j<=min(y+1,m);j++){
                if(v[i][j] && !vis[i][j]){
                    vis[i][j]=true;
                    q.push({i,j});
                }
            }
        }
        if(!v[x][y]){
            able=false;
            break;
        }
    }
    if(able){
        cout<<"YES"<<'\n';
    }
    else{
        cout<<"NO"<<'\n';
    }
    return 0;
}
