import java.util.*;

class Main {
    public static void main(String[] args) {
        var sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        var map = new HashMap<Integer, List<Integer>>();
        for (int i = 0; i < n; i++) {
            map.put(i + 1, new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            map.get(a).add(b);
            map.get(b).add(a);
        }
        var visited = new boolean[n + 1];
        var queue = new LinkedList<Integer>();
        queue.add(1);
        visited[1] = true;
        int count = 0;
        while (!queue.isEmpty()) {
            var size = queue.size();
            boolean flag = false;
            for (int i = 0; i < size; i++) {
                var v = queue.poll();
                for (var next : map.get(v)) {
                    if (!visited[next]) {
                        queue.add(next);
                        visited[next] = true;
                        if (map.get(v).size() % 2 != map.get(next).size() % 2) {
                            flag = true;
                        }
                    }
                }
            }
            if (flag) {
                count++;
            }
        }
        for (boolean v : visited) {
            if (!v) {
                System.out.println(-1);
                return;
            }
        }
        System.out.println(count);
    }
}
