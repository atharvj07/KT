import sys
input = sys.stdin.readline

def solve():
    n, m = map(int, input().split())
    dist = [list(map(int, input().split())) for _ in range(n)]
    ans = []
    for d in dist:
        d.sort()
        ans.append(d)

    for a in ans:
        print(*a)

for _ in range(int(input())):
    solve()
