import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int n = sc.nextInt();
        int m = sc.nextInt();
        boolean[][] a = new boolean[n + 1][m + 1];
        for (int i = 0; i < d; i++) {
            int x1 = sc.nextInt();
            int y1 = sc.nextInt();
            int x2 = sc.nextInt();
            int y2 = sc.nextInt();
            if (x1 == x2) {
                for (int j = Math.min(y1, y2); j <= Math.max(y1, y2); j++) {
                    a[x1][j] = true;
                }
            } else if (y1 == y2) {
                for (int j = Math.min(x1, x2); j <= Math.max(x1, x2); j++) {
                    a[j][y1] = true;
                }
            }
        }
        int cntl = sc.nextInt();
        int cntr = sc.nextInt();
        int cntt = sc.nextInt();
        int cntb = sc.nextInt();
        int ans = -1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (a[i][j]) {
                    int l = cntl;
                    int r = cntr;
                    int t = cntt;
                    int b = cntb;
                    for (int k = i - 1; k >= 1; k--) {
                        if (a[k][j]) {
                            l--;
                        }
                    }
                    for (int k = i + 1; k <= n; k++) {
                        if (a[k][j]) {
                            r--;
                        }
                    }
                    for (int k = j - 1; k >= 1; k--) {
                        if (a[i][k]) {
                            t--;
                        }
                    }
                    for (int k = j + 1; k <= m; k++) {
                        if (a[i][k]) {
                            b--;
                        }
                    }
                    if (l == cntl && r == cntr && t == cntt && b == cntb) {
                        ans = i * m + j;
                        break;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
