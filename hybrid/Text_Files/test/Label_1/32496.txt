import java.util.*;
public class Main {
    public static int n, d, mod;
    public static long an;
    public static long ans = 0;
    public static long p = 0;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        d = in.nextInt();
        mod = in.nextInt();
        ans = pow(2,mod-2);
        an = n - 2;
        for (int i = 0; i < n / 2 - 1; i++) {
            p = p + i;
            if (p > mod) p = p % mod;
        }
        an = an * p;
        if (an > mod) an = an % mod;
        ans = ans * an;
        if (ans > mod) ans = ans % mod;
        System.out.println(ans);
    }
    public static long pow(long a, long m) {
        long res = 1;
        while (m > 0) {
            if ((m & 1) == 1) {
                res = res * a;
                if (res > mod) res = res % mod;
            }
            a = a * a;
            if (a > mod) a = a % mod;
            m = m >> 1;
        }
        return res;
    }
}
