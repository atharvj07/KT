import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        StringBuilder s = new StringBuilder(sc.next());
        StringBuilder t = new StringBuilder(sc.next());
        int ops = 6100;
        boolean possible = true;
        while (ops-- > 0) {
            for (int i = 1; i <= n && possible; i++) {
                if (t.subSequence(0, i).equals(s.subSequence(n - i, n))) {
                    StringBuilder tmp = new StringBuilder(s.substring(0, n - i));
                    tmp.reverse();
                    tmp.append(s.substring(n - i));
                    s = tmp;
                    System.out.print(i + " ");
                    possible = false;
                    break;
                }
            }
            s.append(s.charAt(0));
            s.deleteCharAt(0);
            if (s.toString().equals(t.toString())) {
                System.out.println("\n" + ops);
                return;
            }
        }
        System.out.println("-1");
    }
}
