#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int a, b;
  cin >> a >> b;
  vector<int> v;
  while (a > 0) {
    v.push_back(a % 10);
    a /= 10;
  }
  sort(v.rbegin(), v.rend());
  int cur = 0, idx = 0;
  while (cur < b && idx < v.size()) {
    if (v[idx] == 0 && cur > 0) {
      idx++;
      continue;
    }
    cur = cur * 10 + v[idx++];
  }
  if (cur > b) {
    while (v.back() == 0) {
      v.pop_back();
      idx--;
    }
    sort(v.rbegin(), v.rend());
    cur = 0;
    while (cur < b && idx >= 0) {
      cur = cur * 10 + v[idx--];
    }
  }
  reverse(v.begin(), v.end());
  for (int i = 0; i < v.size(); i++) {
    cout << v[i];
  }
  return 0;
}
