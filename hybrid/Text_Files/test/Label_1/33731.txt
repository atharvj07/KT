import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int a = scanner.nextInt();
            int maxGcd = 0;
            for (int b = 1; b < a; b++) {
                int gcd = gcd(a ^ b, a & b);
                if (gcd > maxGcd) {
                    maxGcd = gcd;
                }
            }
            System.out.println(maxGcd);
        }
    }

    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        } else {
            return gcd(b, a % b);
        }
    }
}
