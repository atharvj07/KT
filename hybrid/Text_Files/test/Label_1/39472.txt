INF = int(1e9)
N, M = map(int, input().split())
isCp = list(map(int, input().split()))
graph = [[] for i in range(N)]
inDeg = [0 for i in range(N)]
for _ in range(M):
    a, b = map(int, input().split())
    graph[a].append(b)
    inDeg[b] += 1

Q = []
for i in range(N):
    if inDeg[i] == 0:
        Q.append(i)
        isCp[i] = INF
res = 0
cnt = 0
while Q:
    cur = Q.pop(0)
    for nxt in graph[cur]:
        inDeg[nxt] -= 1
        if inDeg[nxt] == 0:
            Q.append(nxt)
            isCp[nxt] = isCp[cur]
    if isCp[cur] != INF:
        cnt += 1
    else:
        cnt = 1
        res += 1
    isCp[cur] = min(isCp[cur], cnt)
print(res)
