n, m = map(int, input().split())
x = list(map(int, input().split()))
x.sort()

# Initialize variables to store the result and the positions of humans
res = 0
human_positions = []

# Use two pointers to place humans on both sides of each tree
left = x[0] - 1
right = x[-1] + 1
left_index = 0
right_index = n - 1

while m > 0:
    if m > 0:
        # Place a human to the left if the distance is less or equal than to the right
        if left_index == 0 or (abs(x[left_index] - left) <= abs(x[right_index] - right)):
            res += abs(x[left_index] - left)
            human_positions.append(left)
            left -= 1
            m -= 1
    if m > 0:
        # Place a human to the right
        res += abs(x[right_index] - right)
        human_positions.append(right)
        right += 1
        m -= 1

# Print the result
print(res)
print(" ".join(map(str, human_positions)))
