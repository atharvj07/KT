import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++) a[i] = in.nextInt();
        int ans = 0;
        for(int i=0; i<n; i++) {
            if(i + a[i] < n) ans++;
            else ans += 2;
        }
        System.out.println(ans);
    }
}
