#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n - 1; i++) {
    if (a[i] == a[i + 1]) a[i] = -1;
  }
  int res = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] != -1) res++;
  }
  cout << res << endl;
  for (int i = 0; i < n; i++) {
    if (a[i] != -1) cout << a[i] << " ";
  }
  cout << endl;
  return 0;
}
