import java.util.Scanner;

public class ZebraPhoto {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] photo = new int[n];
        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        boolean isZebra = true;
        int color = photo[0];
        int stripeWidth = 0;
        for (int i = 0; i < n; i++) {
            if (photo[i] != color) {
                if (stripeWidth == 0) {
                    stripeWidth = i;
                } else if (i - stripeWidth != stripeWidth) {
                    isZebra = false;
                    break;
                }
                color = photo[i];
            }
        }

        if (isZebra && stripeWidth != 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
