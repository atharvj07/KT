import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long[] track = new long[n + 1];
        for (int i = 0; i < m; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            int w = in.nextInt();
            track[x] += w;
            track[y] -= w;
        }
        long ans = 0;
        long cur = 0;
        int s = in.nextInt();
        for (int i = 1; i <= n; i++) {
            cur += track[i];
            if (i != s) {
                if (cur > 0) {
                    ans += (cur + 1) * cur / 2;
                }
            }
        }
        System.out.println(ans);
    }
}
