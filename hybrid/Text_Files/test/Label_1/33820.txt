import java.util.Scanner;
import java.math.BigInteger;

class Main {
    private static BigInteger mod;
    private static BigInteger[][] matrix;
    private static int n;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        mod = BigInteger.valueOf(1000000007);
        n = in.nextInt();
        matrix = new BigInteger[n][n];
        in.nextLine();
        for (int i = 0; i < n; i++) {
            String[] line = in.nextLine().split(" ");
            for (int j = 0; j < n; j++) {
                matrix[i][j] = BigInteger.valueOf(Integer.parseInt(line[j]));
            }
        }

        BigInteger[][] permutationMatrix = permute();
        BigInteger determinant = determinant(permutationMatrix);
        BigInteger answer = determinant.mod(mod);

        if (answer.compareTo(BigInteger.ZERO) < 0) {
            answer = answer.add(mod);
        }
        System.out.println(answer);
    }

    private static BigInteger[][] permute() {
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    BigInteger tmp = matrix[i][k];
                    matrix[i][k] = matrix[j][k];
                    matrix[j][k] = tmp;
                }
                BigInteger[] tmpRow = matrix[i];
                matrix[i] = matrix[j];
                matrix[j] = tmpRow;
            }
        }
        return matrix;
    }

    private static BigInteger determinant(BigInteger[][] matrix) {
        BigInteger answer = BigInteger.ONE;
        for (int i = 0; i < n; i++) {
            if (matrix[i][i].compareTo(BigInteger.ZERO) == 0) {
                return BigInteger.ZERO;
            }
            answer = answer.multiply(matrix[i][i]);
            for (int j = i + 1; j < n; j++) {
                BigInteger factor = matrix[j][i].divide(matrix[i][i]);
                for (int k = 0; k < n; k++) {
                    matrix[j][k] = matrix[j][k].subtract(factor.multiply(matrix[i][k]));
                }
            }
        }
        return answer;
    }
}
