#include <iostream>
#include <vector>

using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> ans(n);
    for (int i = 0; i < n; i++) {
        int l = i, r = i;
        while (l >= 0 || r < n) {
            if (l >= 0 && a[l] == 0) {
                ans[i] = i - l;
                break;
            }
            if (r < n && a[r] == 0) {
                ans[i] = r - i;
                break;
            }
            l--, r++;
        }
    }
    for (auto x : ans) { cout << x << " "; }
    return 0;
}
