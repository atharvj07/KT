n = int(input())
x = list(map(int, input().split()))
if len(x) != n ** 2:
    print('NO')
else:
    a = [[] for _ in range(n)]
    for i in range(n):
        for j in range(n): a[i].append(x[i * n + j])
    b = [[] for _ in range(n)]
    for i in range(n):
        for j in range(n): b[j].append(x[i * n + j])
    if a == a[::-1] or b == b[::-1]: print('YES', *map(lambda r: ' '.join(map(str, r)), a if a == a[::-1] else b))
    else: print('NO')
