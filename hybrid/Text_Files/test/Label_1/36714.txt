
def bin_search(lst, x):
    left = 0
    right = len(lst)
    while left < right:
        mid = (left + right) // 2
        if lst[mid] <= x:
            left = mid + 1
        else:
            right = mid
    if left > 0 and lst[left - 1] == x:
        return True
    else:
        return False

def perm(n, x, pos):
    mod = int(1e9) + 7
    pos = pos - 1
    ans = [i for i in range(1, n + 1)]
    ans.remove(x)
    ans.insert(pos, x)
    return ans

def main():
    n, x, pos = map(int, input().split())
    ans = perm(n, x, pos)
    ans = bin_search(ans, x)
    if ans:
        print(0)
    else:
        print(1)

main()
