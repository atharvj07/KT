import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        int ans = -1;
        if (n % 2 == 0 && m % 2 == 0 && m <= n) {
            ans = m / n + (m % n == 0 ? 0 : 1);
        } else if (m % 2 == 1 && n % 2 == 1 && m <= n) {
            ans = m / n + (m % n == 0 ? 0 : 1) + 1;
        } else if (m <= n) {
            ans = m / n + (m % n == 0 ? 0 : 1) + 2;
        }
        System.out.println(ans);
    }
}
