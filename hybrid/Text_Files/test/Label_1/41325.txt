from sys import stdin
from collections import defaultdict
input=stdin.readline

n=int(input())
x=[int(i) for i in input().split()]
y=[int(i) for i in input().split()]
d=defaultdict(list)
for i in range(n):
    d[x[i]].append(y[i])
ans=0
for i in d:
    d[i].sort()
    for j in range(len(d[i])):
        for k in range(j+1,len(d[i])):
            ans+=len(d[i])-k-1
print(ans)
