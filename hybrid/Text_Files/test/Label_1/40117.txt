import sys
input = sys.stdin.readline
n = int(input())
a = sorted([(int(x), int(y)) for x, y in [input().split() for i in range(n)]], key=lambda e: e[1], reverse=True)
m = int(input())
b = sorted([(int(x), int(y)) for x, y in [input().split() for i in range(m)]], key=lambda e: e[1], reverse=True)
c = 0
while a and b:
    if a[0][0] != b[0][0]:
        c += a[0][1] + b[0][1]
        a.pop(0)
        b.pop(0)
    else:
        if a[0][1] > b[0][1]:
            a.pop(0)
        else:
            b.pop(0)
print(c)
