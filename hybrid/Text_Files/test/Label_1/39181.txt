import sys

input = sys.stdin.readline

n, m, k, q = map(int, input().split())
safe = [0] * m
treasures = []
for _ in range(k):
    x, y = map(int, input().split())
    treasures.append((x, y))
    safe[y - 1] = 1
safe_columns = [int(x) for x in input().split()]

treasures.sort()
mn = 2 * 10 ** 9
now = 1
for x, y in treasures:
    if now == x:
        mn = min(mn, y)
    else:
        if safe[y - 1]:
            mn = min(mn, x + y - now - 1)
        else:
            lb = 0
            rb = len(safe_columns) - 1
            idx = -1
            while lb <= rb:
                mid = (lb + rb) // 2
                if safe_columns[mid] <= y:
                    idx = mid
                    lb = mid + 1
                else:
                    rb = mid - 1
            if idx == -1:
                mn = min(mn, x + y - now)
            else:
                mn = min(mn, x - now + safe_columns[idx] - y + 1)
        now = x
print(mn)
