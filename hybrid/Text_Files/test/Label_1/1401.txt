#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,l; cin>>n>>l;
	vector<int> a(n),b(n);
	for(int i=0; i<n; i++)	cin>>a[i];
	for(int i=0; i<n; i++)	cin>>b[i];
	sort(a.begin(), a.end());
	sort(b.begin(), b.end());

	for(int i=0; i<n; i++)	a[i]%=l;
	sort(a.begin(), a.end());

	if(a==b)	cout<<"YES\n";
	else	cout<<"NO\n";
	return 0;
}
