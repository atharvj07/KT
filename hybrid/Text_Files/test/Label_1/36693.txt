n = int(input())
a = [int(x) for x in input().split()]
up = False
down = False
for i in range(1, n):
    if a[i] == a[i - 1]:
        continue
    if not up:
        if a[i] > a[i - 1]:
            up = True
        else:
            print("NO")
            break
    elif not down:
        if a[i] < a[i - 1]:
            down = True
        else:
            print("NO")
            break
    else:
        if a[i] > a[i - 1]:
            print("NO")
            break
else:
    print("YES")
