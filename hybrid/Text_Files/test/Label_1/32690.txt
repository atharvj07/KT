import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {

    static boolean[][][] a;
    static boolean[][][] can;
    static int[][][] from;
    static int n, m, k;
    static List<Integer> list;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        n = scan.nextInt();
        m = scan.nextInt();
        k = scan.nextInt();
        a = new boolean[n + 1][m + 1][k + 1];
        can = new boolean[n + 1][m + 1][k + 1];
        from = new int[n + 1][m + 1][k + 1];
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) {
                String s = scan.next();
                for (int l = 1; l <= k; l++)
                    a[i][j][l] = s.charAt(l - 1) == '1';
            }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                for (int l = 1; l <= k; l++) {
                    if (a[i][j][l]) {
                        if (a[i - 1][j][l])
                            can[i][j][l] = true;
                        if (a[i][j - 1][l])
                            can[i][j][l] = true;
                        if (a[i][j][l - 1])
                            can[i][j][l] = true;
                    }
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                for (int l = 1; l <= k; l++) {
                    if (can[i][j][l]) {
                        if (!can[i - 1][j][l] && a[i - 1][j][l])
                            from[i][j][l] = 1;
                        else if (!can[i][j - 1][l] && a[i][j - 1][l])
                            from[i][j][l] = 2;
                        else if (!can[i][j][l - 1] && a[i][j][l - 1])
                            from[i][j][l] = 3;
                    }
                }
            }
        }
        list = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                for (int l = 1; l <= k; l++) {
                    if (can[i][j][l])
                        dfs(i, j, l);
                }
            }
        }
        System.out.println(list.size());
    }

    static void dfs(int i, int j, int l) {
        if (!a[i][j][l])
            return;
        if (list.contains(i * 100000 + j * 1000 + l))
            return;
        list.add(i * 100000 + j * 1000 + l);
        int ii = from[i][j][l];
        if (ii == 1)
            dfs(i - 1, j, l);
        if (ii == 2)
            dfs(i, j - 1, l);
        if (ii == 3)
            dfs(i, j, l - 1);
    }
}
