import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x = new int[n], y = new int[n], z = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
            y[i] = in.nextInt();
            z[i] = in.nextInt();
        }
        boolean eq = true;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                eq &= (x[i] == -x[j] && y[i] == -y[j] && z[i] == -z[j]);
            }
        }
        System.out.println(eq ? "YES" : "NO");
    }
}
