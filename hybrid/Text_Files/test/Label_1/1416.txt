#include <iostream>
#include <string>

using namespace std;

int main() {
  string password;
  cin >> password;
  bool hasUpper = false;
  bool hasLower = false;
  bool hasDigit = false;
  for (int i = 0; i < password.size(); i++) {
    if (isupper(password[i])) hasUpper = true;
    if (islower(password[i])) hasLower = true;
    if (isdigit(password[i])) hasDigit = true;
  }
  if (password.size() >= 5 && hasUpper && hasLower && hasDigit) {
    cout << "Correct" << endl;
  } else {
    cout << "Too weak" << endl;
  }
  return 0;
}
