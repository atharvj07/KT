#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int& x : a) {
        cin >> x;
    }
    int ans = 0;
    for (int i = 0; i < n && k > 0; i++) {
        if (a[i] < 0) {
            ans -= a[i];
            a[i] = -a[i];
            k--;
        }
    }
    sort(a.begin(), a.end());
    for (int i = 0; i < n && k > 0; i++) {
        ans -= a[i];
        a[i] = -a[i];
        k--;
    }
    for (int x : a) {
        ans += x;
    }
    cout << ans << endl;
    return 0;
}
