import sys
def checkCoprime(a,b):
    while(b):
        a,b = b,a%b
    return a == 1
n = int(input())
lst = []
for val in input().split():
    lst.append(int(val))
ans = []
for x in lst:
    if x >= 2:
        ans.append(x)
    else:
        ans.append(2)
        break
for i in range(len(lst)):
    if lst[i] >= 2:
        continue
    ans[i] = 3
    flag = False
    for j in range(i):
        if checkCoprime(ans[i], ans[j]) and ans[i] != lst[i]:
            flag = True
            ans[i] += 1
            break
    if not flag:
        while(True):
            if checkCoprime(ans[i], ans[i-1]) and ans[i] != lst[i]:
                break
            ans[i] += 1
sys.stdout.write(' '.join(map(str, ans)))
