import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        int d = scanner.nextInt();
        char[][] ans = new char[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                int dist = Math.abs(i - (h - 1) / 2) + Math.abs(j - (w - 1) / 2);
                if (dist % d == 0) {
                    ans[i][j] = 'R';
                } else if (dist % d == 1) {
                    ans[i][j] = 'Y';
                } else if (dist % d == 2) {
                    ans[i][j] = 'G';
                } else {
                    ans[i][j] = 'B';
                }
            }
        }
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                System.out.print(ans[i][j]);
            }
            System.out.println();
        }
    }
}
