#include <cstdio>
using namespace std;

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int n, k;
        char s[105];
        scanf("%d%d", &n, &k);
        scanf("%s", s);
        if (n % (2 * k + 1) == 0) {
            puts("YES");
        } else {
            puts("NO");
        }
    }
    return 0;
}
