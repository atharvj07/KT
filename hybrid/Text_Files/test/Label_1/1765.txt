#include <iostream>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int a[n], ans = 0, left = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = n - 1; i >= 0; i--) {
    if (left + a[i] <= m) {
      left += a[i];
      ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
