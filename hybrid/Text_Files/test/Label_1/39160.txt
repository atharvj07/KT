n = int(input())
grid = [[0] * 101 for i in range(101)]
for i in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    for x in range(x1, x2 + 1):
        for y in range(y1, y2 + 1):
            grid[x][y] += 1
ans = 0
for i in range(101):
    for j in range(101):
        ans += grid[i][j]
print(ans)
