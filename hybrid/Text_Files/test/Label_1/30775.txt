import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int[] fly1 = new int[3];
        for (int i = 0; i < 3; i++) {
            fly1[i] = scan.nextInt();
        }
        int[] fly2 = new int[3];
        for (int i = 0; i < 3; i++) {
            fly2[i] = scan.nextInt();
        }
        boolean same = true;
        for (int i = 0; i < 3; i++) {
            same &= fly1[i] == fly2[i];
        }
        System.out.println(same ? "YES" : "NO");
        scan.close();
    }
}
