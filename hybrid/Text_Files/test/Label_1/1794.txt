#include <bits/stdc++.h>
using namespace std;

int main() {
    int n; cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) cin >> v[i].second, v[i].first = v[i].second < 0 ? -1 : 1;
    sort(v.rbegin(), v.rend());
    for (auto [sign, val] : v) cout << sign * val << " ";
    cout << endl;
}
