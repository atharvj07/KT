import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), l = sc.nextInt();
        int[] x = new int[k];
        int[] a = new int[l];
        for (int i = 0; i < k; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 0; i < l; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(x);
        Arrays.sort(a);

        int ans = 0, j = 0, cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (j < k && i == x[j]) {
                j++;
                cnt++;
            } else {
                if (cnt > 0) {
                    ans += (cnt + 1) / 2;
                    cnt = 0;
                }
                while (j < k && i > x[j]) {
                    j++;
                }
            }
        }
        if (cnt > 0) {
            ans += (cnt + 1) / 2;
        }

        System.out.println(ans == 0 ? -1 : ans);
    }
}
