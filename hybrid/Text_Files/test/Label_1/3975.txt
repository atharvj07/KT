#include <bits/stdc++.h>

using namespace std;

typedef long long ll;
const int P = 1e9 + 7;

inline ll power(ll base, ll p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  ll ans = power(1ll * base * base % P, p / 2);
  if (p & 1) ans = 1ll * ans * base % P;
  return ans;
}

inline ll mod_inverse(ll x) {
  return power(x, P - 2);
}

ll gcd(ll a, ll b) {
  if (b == 0) return a;
  return gcd(b, a % b);
}

int divisor(ll n) {
  int cnt = 0;
  for (ll i = 1; i * i <= n; ++i) {
    if (n % i == 0) {
      if (i * i == n) {
        ++cnt;
      } else {
        cnt += 2;
      }
    }
  }
  return cnt;
}

int32_t main() {
  ll n, k;
  cin >> n >> k;
  cout << (1ll * divisor(n) * mod_inverse(k + 1) % P + P) % P << "\n";
  return 0;
}
