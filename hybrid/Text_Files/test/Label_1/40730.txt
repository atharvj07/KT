n=int(input())
a=[int(x) for x in input().split()]
a=[(n,x) for x in a]
a.sort(key=lambda x:x[1],reverse=True)
ans=[0]*int((n+1)/2)
for i in range(int((n+1)/2)):
    d=a[i][0]
    for j in range(d,0,-1):
        if a[i][1]>=j:
            ans[i]+=a[i][1]-j
            for k in range(i+1,int((n+1)/2)):
                a[k]=(a[k][0]-1,(a[k][1]-(a[i][1]-j))%j if a[k][1]<j else a[k][1]%j)
            break
for x in ans:
    print(x,end=" ")
