#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    int ar[n],sq[n];
    for(int i=0;i<n;i++){ cin>>ar[i]; sq[ar[i]-1]=i+1;}
    for(int i=0;i<n;i++){ sq[i]=sq[sq[i]-1];}
    for(int i=0;i<n;i++){
        if(i+1!=sq[i]) {cout<<-1; return 0;}
    }
    for(int i=0;i<n;i++) cout<<sq[i]<<" ";
    return 0;
}
