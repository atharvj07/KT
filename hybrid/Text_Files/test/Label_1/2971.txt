#include <bits/stdc++.h>
using namespace std;
const int maxn=2210;
long long n,k,a[maxn],b[maxn],f[maxn][maxn][2];
int main()
{
    cin>>n>>k;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++) cin>>b[i];
    for(int i=n;i>=1;i--)
    {
        for(int j=k;j>=0;j--)
        {
            for(int p=0;p<=1;p++)
            {
                f[i][j][p]=1e18;
                if(j>0) f[i][j][p]=min(f[i][j][p],f[i][j-1][p]);
                if(i==n || p==1)
                {
                f[i][j][p]=min(f[i][j][p],f[i+1][j][1]+a[i]);
                f[i][j][p]=min(f[i][j][p],f[i+1][j][0]+b[i]);
                }
            }
        }
    }
    cout<<f[1][k][0]<<endl;
    return 0;
}
