import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long d = sc.nextLong(), k = sc.nextLong();
            if (k <= d / 2) System.out.println("Utkarsh");
            else System.out.println("Ashish");
        }
    }
}
