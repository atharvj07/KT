import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] count = new int[4][2];
        while (scanner.hasNextLine()) {
            String[] lr = scanner.nextLine().split(" ");
            double l = Double.parseDouble(lr[0]);
            double r = Double.parseDouble(lr[1]);
            count[getJudgment(l)][0]++;
            count[getJudgment(r)][1]++;
        }
        for (int i = 0; i < 4; i++) {
            System.out.println(count[i][0] + " " + count[i][1]);
        }
    }
    private static int getJudgment(double v) {
        if (v >= 1.1) return 0;
        else if (v >= 0.6) return 1;
        else if (v >= 0.2) return 2;
        else return 3;
    }
}
