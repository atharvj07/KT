#include <bits/stdc++.h>
using namespace std;
#define int long long
const int maxn = 2e5 + 5;
vector<int> adj[maxn];
int a[maxn], mx[maxn], dp[maxn];
void dfs(int u, int p) {
    mx[u] = a[u];
    for (auto v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        mx[u] = __gcd(mx[u], mx[v]);
    }
}
void dfs2(int u, int p) {
    dp[u] = mx[u];
    for (auto v : adj[u]) {
        if (v == p) continue;
        int gcd = __gcd(mx[u], mx[v]);
        if (gcd == mx[u])
            dp[v] = max(dp[v], mx[v]);
        else
            dp[v] = max(dp[v], __gcd(mx[v], dp[u]));
        dfs2(v, u);
    }
}
signed main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, -1);
    dfs2(1, -1);
    for (int i = 1; i <= n; i++) { cout << dp[i] << " "; }
    cout << endl;
}
