import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] input1 = br.readLine().split(" ");
        int n = Integer.parseInt(input1[0]);
        long x = Long.parseLong(input1[1]);
        long y = Long.parseLong(input1[2]);
        long p = Long.parseLong(input1[3]);
        
        String[] input2 = br.readLine().split(" ");
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = Long.parseLong(input2[i]);
        }

        long total = 0;
        for (long ai : a) {
            total += ai;
            total %= p;
        }

        long factor = pow(2, x, p);
        total = (total * factor) % p;
        factor = pow(2, y, p);
        total = (total * factor) % p;
        System.out.println(total);
    }

    public static long pow(long base, long exp, long mod) {
        if (exp == 0) return 1;
        if (exp == 1) return base;
        long halfpow = pow(base, exp / 2, mod);
        long halfpow2 = (halfpow * halfpow) % mod;
        if (exp % 2 == 0) {
            return halfpow2;
        } else {
            return (halfpow2 * base) % mod;
        }
    }
}
