import sys

n, t, k = map(int, input().split())
if k == n - 1 or t < k:
    print(-1)
    sys.exit()
a = [int(x) for x in input().split()]
if sum(a) != n - 1:
    print(-1)
    sys.exit()
adj = [[] for _ in range(n + 1)]
for i in range(0, t - k):
    y = a[i]
    for j in range(1, y + 1):
        adj[j].append(y + j)
        adj[y + j].append(j)
for i in range(t - k, t):
    y = a[i]
    for j in range(1, y + 1):
        adj[j].append(t + j)
        adj[t + j].append(j)
print(n)
for i in range(2, n + 1):
    for j in adj[i]:
        if i < j:
            print(i, j)
            break
