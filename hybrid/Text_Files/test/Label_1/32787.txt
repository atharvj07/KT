import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;
import java.util.HashMap;
import java.util.Map;
import java.util.PriorityQueue;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int k = Integer.parseInt(line1[1]);
        int m = Integer.parseInt(line1[2]);
        int a = Integer.parseInt(line1[3]);

        String[] line2 = br.readLine().split(" ");
        int[] votes = new int[n + 1];
        Map<Integer, Integer> lastVote = new HashMap<>();
        PriorityQueue<Candidate> candidates = new PriorityQueue<>(new Comparator<Candidate>() {
            @Override
            public int compare(Candidate c1, Candidate c2) {
                if (c1.votes == c2.votes) {
                    return lastVote.get(c2.id) - lastVote.get(c1.id);
                } else {
                    return c2.votes - c1.votes;
                }
            }
        });

        for (int i = 0; i < a; i++) {
            int g = Integer.parseInt(line2[i]);
            votes[g]++;
            lastVote.put(g, i);
            candidates.offer(new Candidate(g, votes[g]));
        }

        while (!candidates.isEmpty()) {
            Candidate c = candidates.poll();
            if (c.votes >= k) {
                System.out.print("1 ");
            } else if (c.votes + (m - a) >= k) {
                System.out.print("2 ");
            } else {
                System.out.print("3 ");
            }
        }

        System.out.println();
    }

    static class Candidate {
        int id;
        int votes;

        public Candidate(int id, int votes) {
            this.id = id;
            this.votes = votes;
        }
    }
}
