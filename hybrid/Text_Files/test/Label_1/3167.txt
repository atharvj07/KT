#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main(){
    ll n,k,a,ans=0;
    cin>>n>>k;
    vector<ll>v;
    for(ll i=0;i<n;i++){
        cin>>a;
        v.push_back(a);
    }
    sort(v.begin(),v.end());
    for(ll i=0;i<(n+1)/2;i++){
        ans+=ceil((double)(v[i]+v[n-i-1])/k);
    }
    cout<<ans<<endl;
    return 0;
}
