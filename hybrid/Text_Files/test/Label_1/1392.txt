#include <bits/stdc++.h>
using namespace std;
int d[31];
int s;
vector<int> e[31];
vector<pair<int, int>> ans;
void dfs(int v, int p){
    ans.push_back({d[v], s});
    for(int u : e[v]){
        if(u == p) continue;
        d[u] = d[v] + (s == 0 ? 1 : -1);
        s ^= 1;
        dfs(u, v);
    }
}
int main() {
    int n;
    cin >> n;
    for(int i = 1, u, v; i < n; i++){
        cin >> u >> v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    int root;
    for(int i = 1; i <= n; i++){
        if(e[i].size() > 2){
            cout << "NO" << endl;
            return 0;
        }
        else if(e[i].size() == 1){
            root = i;
        }
    }
    dfs(root, 0);
    cout << "YES" << endl;
    for(auto x : ans){
        cout << x.first << " " << x.second << "\n";
    }
    return 0;
}
