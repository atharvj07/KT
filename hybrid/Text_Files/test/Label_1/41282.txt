n=int(input())
name=[['']*2 for _ in range(n)]
for i in range(n):
 a,b=input().split()
 name[i][0],name[i][1]=a,b
 permutation=list(map(int,input().split()))
 name.sort(key=lambda x:(x[permutation[0]-1],x[0]))
 for i in range(1,n):
  if name[permutation[i]-1][0]!=name[permutation[i-1]-1][0]:continue
  if name[permutation[i]-1][1]<name[permutation[i-1]-1][1]:continue
  print('NO');break
 else:print('YES')
