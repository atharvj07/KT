for _ in range(int(input())):
    n, a, b = map(int, input().split())
    if n == 2:
        print(b, a)
        continue
    if n%2 != 0 or a == b or a > b:
        print(-1)
        continue
    l = [a]
    for i in range(n, 0, -1):
        if i != b and i != a:
            l.append(i)
    l.append(b)
    print(*l)
