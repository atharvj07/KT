#include <bits/stdc++.h>
using namespace std;

const int MX = 1e5;
int a[MX], par[MX], sz[MX], deg[MX], dis[MX], vis[MX], fl = 0;
vector<int> adj[MX];

int find(int u) {
    if(par[u] == u)
        return u;
    return par[u] = find(par[u]);
}

void unite(int u, int v) {
    u = find(u);
    v = find(v);
    if(u != v) {
        if(sz[u] < sz[v]) {
            par[u] = v;
            sz[v] += sz[u];
        }
        else {
            par[v] = u;
            sz[u] += sz[v];
        }
    }
}

int main() {
    int n;
    cin >> n;
    for(int i = 0; i < n; i++)
        cin >> a[i];
    for(int i = 0; i < n; i++) {
        par[i] = i;
        sz[i] = 1;
        for(int j = i + 1; j < n; j++) {
            if(a[i] & a[j]) {
                unite(i, j);
                adj[i].push_back(j);
                adj[j].push_back(i);
                deg[i]++; deg[j]++;
            }
        }
    }
    int st = -1, en = -1;
    for(int i = 0; i < n; i++) {
        if(st == -1) {
            st = i;
            dis[st] = 0;
            vis[st] = 1;
            queue<int> q;
            q.push(st);
            while(!q.empty()) {
                int u = q.front();
                q.pop();
                for(int x : adj[u]) {
                    if(vis[x] == 0) {
                        vis[x] = 1;
                        q.push(x);
                        dis[x] = dis[u] + 1;
                        if(par[x] == par[st]) {
                            en = x;
                            fl = 1;
                            break;
                        }
                    }
                }
                if(fl)    break;
            }
            memset(vis, 0, sizeof vis);
        }
        if(fl) {
            dis[st] = 0;
            vis[st] = 1;
            queue<int> q;
            q.push(st);
            int mi = dis[en];
            while(!q.empty()) {
                int u = q.front();
                q.pop();
                for(int x : adj[u]) {
                    if(vis[x] == 0) {
                        vis[x] = 1;
                        q.push(x);
                        mi = min(mi, dis[x] + dis[u] + 1);
                    }
                }
            }
            cout << mi << "\n";
            return 0;
        }
    }
    cout << -1 << "\n";    
    return 0;
}
