N,K,Q = map(int,input().split())
L = [int(i) for i in input().split()]
K -= 1
A = []
for i in range(Q):
    T,ID = map(int,input().split())
    if T == 1:
        L[ID-1] = True
A = [i for i,x in enumerate(L) if x]
A = sorted([(-L[i],i) for i in A])
for i in range(min(K,len(A))):
    print('YES')
for i in range(min(K,len(A)),len(A)):
    print('NO')
