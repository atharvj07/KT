import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String[] strs = new String[n];
    for (int i = 0; i < n; i++) strs[i] = sc.next();
    int l = strs[0].length();
    int min = Integer.MAX_VALUE;
    for (int i = 0; i < l; i++) {
      String cur = strs[0];
      int cnt = 0;
      for (int j = 1; j < n; j++) {
        String next = cur.substring(1) + cur.charAt(0);
        if (!strs[j].equals(next)) {
          cnt = -1;
          break;
        }
        cnt++;
        cur = next;
      }
      if (cnt >= 0) min = Math.min(min, cnt);
    }
    System.out.println(min == Integer.MAX_VALUE ? -1 : min);
  }
}
