#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
typedef __uint128_t lll;

#define int ll
#define all(x) x.begin(), x.end()
#define pb push_back

const int mod = 1e9 + 7;

void euclid(int a, int b, int &g, int &x, int &y) {
  if (b == 0) {
    g = a, x = 1, y = 0;
    return;
  }
  euclid(b, a % b, g, y, x);
  y -= (a / b) * x;
}

int modinv(int a, int m) {
  int g, x, y;
  euclid(a, m, g, x, y);
  return (x + m) % m;
}

int32_t main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  int mx_a = *max_element(all(a)), mx_b = *max_element(all(b));
  if (mx_a > mx_b) {
    cout << 0;
    return 0;
  }
  int ans = 1;
  for (int i = 0; i < n; i++) {
    if (a[i] == b[i]) continue;
    if (!a[i])
      ans = (ans * (m - 1)) % mod;
    else if (!b[i])
      ans = (ans * m) % mod;
    else
      ans = 0;
  }
  int g, x, y;
  euclid(ans, mod, g, x, y);
  cout << (x + mod) % mod << '\n';
  return 0;
}
