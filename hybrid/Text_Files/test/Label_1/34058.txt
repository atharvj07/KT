import java.util.*;

public class Main {

    public static void main(String args[]) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n * n];
        for (int i = 0; i < n * n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        if (n % 2 == 1) {
            int[][] ans = new int[n][n];
            int mid = n * n / 2;
            int c = 0, r = n / 2, x = 0, y = n / 2;
            while (c < n * n) {
                ans[x][y] = a[c];
                c++;
                x++;
                y--;
                if (x == n) {
                    x = 0;
                    r--;
                }
                if (y == -1) {
                    y = n - 1;
                    r++;
                }
            }
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    System.out.print(ans[i][j] + " ");
                }
                System.out.println();
            }
        } else if (n % 4 == 0) {
            int[][] ans = new int[n][n];
            int mid = n * n / 2;
            int c = 0, r = n / 2, x = 0, y = n / 2 - 1;
            while (c < mid) {
                ans[x][y] = a[c];
                c++;
                x++;
                y--;
                if (x == n) {
                    x = 0;
                    r--;
                }
                if (y == -1) {
                    y = n - 1;
                    r++;
                }
            }
            c = mid;
            x = n / 2;
            y = n / 2;
            while (c < n * n) {
                ans[x][y] = a[c];
                c++;
                x--;
                y++;
                if (x == -1) {
                    x = n - 1;
                    r--;
                }
                if (y == n) {
                    y = 0;
                    r++;
                }
            }
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    System.out.print(ans[i][j] + " ");
                }
                System.out.println();
            }
        } else {
            System.out.println("NO");
        }
    }
}
