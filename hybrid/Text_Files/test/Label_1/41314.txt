n = int(input())
a = list(map(int, input().split()))
b = sorted(a)
if a == b:
    print("yes\n1 1")
    exit()
l = -1
r = -1
for i in range(n):
    if a[i] != b[i]:
        if l == -1:
            l = i
        r = i
if b[l-1] > a[r] or b[r+1] < a[l]:
    print("no")
    exit()
print("yes")
print(l+1, r+1)
