#include <iostream>
#include <vector>
#include <queue>

#define INF 1e9
#define ll long long
#define vi vector<int>
#define pii pair<int,int>
#define mp make_pair

using namespace std;

const int N = 2e5+10;
int n, a[N], p[N];

int main() {
    cin >> n;
    for (int i = 0; i < n; ++i) cin >> a[n-1-i];
    
    int mn = a[n-1];
    p[n-1] = mn;
    
    for (int i = n-2; i >= 0; --i) {
        mn = min(mn, a[i]);
        p[i] = mn;
    }

    for (int i = n-2; i >= 0; --i) {
        if (a[i] < p[i+1]) {
            cout << ":(";
            return 0;
        }
    }

    for (int i = 0; i < n; ++i) cout << p[i] << ' ';
    return 0;
}
