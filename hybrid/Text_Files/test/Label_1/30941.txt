import java.util.Scanner;

public class Main {
    static long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), k = in.nextInt();
        System.out.println((longMath.pow(k, (long) n) * longMath.pow(k, (long) m)) % MOD);
    }

    static class longMath {
        private longMath() {
        }

        static long pow(long base, long p) {      // Binary Exponentiation
            if (p == 0) {
                return 1;
            }
            if (p == 1) {
                return base;
            }
            if ((p & 1) == 0) {
                long val = pow(base, p >> 1);
                return (val * val) % MOD;
            }
            return (base * pow(base, p - 1)) % MOD;
        }
    }
}
