import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = br.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);
        int[][] edges = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(edges[i], -1);
        }
        for (int i = 0; i < m; i++) {
            String[] edge = br.readLine().split(" ");
            int u = Integer.parseInt(edge[0]) - 1, v = Integer.parseInt(edge[1]) - 1, p = Integer.parseInt(edge[2]);
            edges[u][v] = p;
            edges[v][u] = p;
        }
        String[] st = br.readLine().split(" ");
        int s = Integer.parseInt(st[0]) - 1, t = Integer.parseInt(st[1]) - 1;
        Queue<Integer> q = new LinkedList<>();
        BigInteger[] dis = new BigInteger[n];
        for (int i = 0; i < n; i++) dis[i] = BigInteger.valueOf(Long.MAX_VALUE);
        dis[s] = BigInteger.ZERO;
        boolean[] visited = new boolean[n];
        q.add(s);
        BigInteger res = BigInteger.ZERO;
        while (!q.isEmpty()) {
            int u = q.poll();
            if (u == t) {
                res = dis[u];
                break;
            }
            visited[u] = true;
            for (int v = 0; v < n; v++) {
                if (edges[u][v] != -1 && !visited[v]) {
                    BigInteger d = BigInteger.valueOf(edges[u][v] * 2);
                    if (dis[u].add(d).compareTo(dis[v]) < 0) {
                        dis[v] = dis[u].add(d);
                        q.add(v);
                    }
                }
            }
        }
        if (res.equals(BigInteger.ZERO)) {
            System.out.println(-1);
        } else {
            System.out.println(res.mod(BigInteger.valueOf(1000000007)));
            int path[] = new int[n];
            int p = -1, cnt = 0;
            for (int i = 0; i < n; i++) {
                if (dis[i].equals(res)) {
                    p = i;
                    break;
                }
            }
            while (p != -1) {
                path[cnt++] = p + 1;
                for (int i = 0; i < n; i++) {
                    if (edges[p][i] != -1 && dis[i].equals(dis[p].subtract(BigInteger.valueOf(edges[p][i] * 2)))) {
                        p = i;
                        break;
                    }
                }
            }
            System.out.println(cnt);
            for (int i = cnt - 1; i >= 0; i--) {
                System.out.print(path[i] + " ");
            }
            System.out.println();
        }
    }
}
