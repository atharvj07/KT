#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int a[26] = {0}, cnt = 0;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] != '?') a[s[i] - 'A']++;
  }
  for (int i = 0; i < 26; i++) {
    if (a[i]) cnt++;
  }
  if (cnt < 26) {
    cout << -1;
    return 0;
  }
  string ans;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] == '?') {
      for (int j = 0; j < 26; j++) {
        if (!a[j]) {
          a[j]++;
          ans += char(j + 'A');
          break;
        }
      }
    } else {
      ans += s[i];
    }
  }
  cout << ans;
}
