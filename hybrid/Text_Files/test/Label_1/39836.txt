# Read initial string
s = input().strip()

# Read number of queries
n = int(input())

# Process each query
for _ in range(n):
    query = input().split('->')
    d, t = query[0], query[1]
    s = s.replace(d, t)

# Convert the final string to a number and find the remainder
result = int(s) % 1000000007 if s else 0

# Output the result
print(result)
