#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            cin >> a[i][j];
        }
    }
    vector<int> b(n * m);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            b[a[i][j] - 1] = i * m + j;
        }
    }
    int ans = 0;
    for (int i = 0; i < n * m; ++i) {
        if (b[i] == i) continue;
        int j = i;
        while (b[j] != i) {
            j = b[j];
            ++ans;
        }
    }
    cout << ans << endl;
    return 0;
}
