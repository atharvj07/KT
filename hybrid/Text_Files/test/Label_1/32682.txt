import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        int k = in.nextInt();
        long[] a = new long[k];
        for (int i = 0; i < k; i++) {
            a[i] = in.nextLong();
        }
        Arrays.sort(a);
        int cnt = 0;
        for (int i = k - 1; i >= 0; i--) {
            if (n % a[i] == 0) {
                cnt = (int) (n / a[i]);
                break;
            }
        }
        System.out.println((Arrays.binarySearch(a, n % a[k - 1]) + 1) + " " + cnt);
    }
}
