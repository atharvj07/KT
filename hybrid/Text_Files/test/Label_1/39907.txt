import sys
input = sys.stdin.readline
n = int(input())
tags = [input() for _ in range(n)]
mn = min(tags)
for i, t in enumerate(tags):
    if t == mn:
        print(t)
        continue
    j = 0
    while j < len(mn) and j < len(t) and t[j] == mn[j]:
        j += 1
    tags[i] = t[:j] + "#"
tags.sort()
for t in tags:
    print(t)
