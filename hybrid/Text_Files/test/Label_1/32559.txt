import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();

        int[] pos = new int[n + 1];
        int curPos = 1;
        for (int i = 1; i <= n; i++) {
            pos[i] = i;
        }

        for (int i = 0; i < q; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                int x = sc.nextInt();
                curPos = (curPos + x + n) % n + 1;
            } else {
                for (int j = 1; j <= n; j += 2) {
                    int temp = pos[j];
                    pos[j] = pos[j + 1];
                    pos[j + 1] = temp;
                }
            }
        }

        for (int i = 1; i <= n; i++) {
            System.out.print(pos[(curPos + i - 1) % n + 1] + " ");
        }
    }
}
