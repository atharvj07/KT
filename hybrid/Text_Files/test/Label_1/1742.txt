#include <cstdio>
#include <numeric>
using namespace std;

int n, k;

int main() {
    scanf(" %d%d", &n, &k);
    long long sum[n + 1];
    partial_sum(sum, sum + n + 1, sum);
    long long ans = 0;
    int st, en;
    for (int i = 1; i + k - 1 < n; i++) {
        long long cur = sum[i + k + k - 2] - sum[i + k - 1];
        if (cur > ans) {
            ans = cur;
            st = i, en = i + k;
        }
    }
    printf("%d %d", st, en);
    return 0;
}
