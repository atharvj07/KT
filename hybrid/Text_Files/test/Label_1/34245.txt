import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] w = new String[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.next();
        }
        int l = n;
        for (int i = 0; i < n - 1; i++) {
            if (w[i].charAt(w[i].length() - 1) == w[i + 1].charAt(0)) {
                l--;
            }
        }
        System.out.println(l);
    }
}
