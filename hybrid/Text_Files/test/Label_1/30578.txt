import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] c = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            c[i] = sc.nextInt();
        }
        ArrayList<Integer>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        boolean[] vis = new boolean[n + 1];
        int[] sum = new int[n + 1];
        dfs(1, 0, adj, vis, sum, c);
        if (sum[1] != 0) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int i = 1; i <= m; i++) {
                System.out.println(sum[i]);
            }
        }
    }

    public static void dfs(int v, int p, ArrayList<Integer>[] adj, boolean[] vis, int[] sum, int[] c) {
        vis[v] = true;
        for (int u : adj[v]) {
            if (!vis[u]) {
                dfs(u, v, adj, vis, sum, c);
                sum[p] += sum[u];
            }
        }
        sum[v] = c[v] - sum[p];
    }
}
