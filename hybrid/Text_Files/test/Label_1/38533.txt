from typing import List

n = int(input())
a = [int(x) for x in input().split()]
ans = [0] * (n // 2 + 1)
for i in range(1, n // 2 + 1):
    l = []
    for j in range(n):
        if a[j] > 0:
            l.append(a[j])
        a[j] -= 1
    l.sort(reverse=True)
    for j in range(i - 1, len(l)):
        ans[i] += l[j]
for x in ans:
    print(x, end=' ')
