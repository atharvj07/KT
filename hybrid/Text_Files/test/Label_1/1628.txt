#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) { cin >> a[i]; }
    bool odd = false;
    for (int i = 0; i < n; ++i) {
      if (a[i] % 2 == 1) {
        odd = true;
        break;
      }
    }
    if (x % 2 == 1) {
      cout << "Yes" << endl;
    } else if (odd && x > 1) {
      cout << "Yes" << endl;
    } else {
      cout << "No" << endl;
    }
  }
}
