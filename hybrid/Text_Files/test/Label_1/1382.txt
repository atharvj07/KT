#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string b;
    cin >> b;
    int n = b.size();
    vector<char> a{'0'};
    for (int i = 0; i < n; i += 2) {
      a.push_back(b[i]);
      a.push_back(b[i + 1]);
    }
    a.pop_back();
    for (int i = 1; i < a.size(); i++) { cout << a[i]; }
    cout << endl;
  }
  return 0;
}
