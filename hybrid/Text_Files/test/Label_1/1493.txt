#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while(t--){
        string s; cin >> s;
        for(int i = 0; i < s.size(); i++){
            if(i+1 == s.size() || s[i] > s[i+1]){
                s[i] = 'a';
            } else {
                s[i] = s[i+1];
            }
        }
        cout << s << endl;
    }
    return 0;
}
