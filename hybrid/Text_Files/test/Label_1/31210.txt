import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = scanner.nextLine();
        }
        int[][] costs = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                costs[i][j] = scanner.nextInt();
            }
        }
        System.out.println(solve(strings, costs));
    }

    public static int solve(String[] strings, int[][] costs) {
        int n = strings.length;
        int m = strings[0].length();
        Map<Character, Integer>[][] lastPositions = new Map[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                lastPositions[i][j] = new HashMap<>();
            }
        }
        boolean easyToRemember = true;
        for (int i = 0; i < m; i++) {
            Map<Character, Integer> lastPositionInColumn = new HashMap<>();
            for (int j = 0; j < n; j++) {
                char c = strings[j].charAt(i);
                if (lastPositionInColumn.containsKey(c)) {
                    easyToRemember = false;
                    break;
                }
                lastPositionInColumn.put(c, j);
            }
        }
        int answer = 0;
        if (!easyToRemember) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    char c = strings[i].charAt(j);
                    if (lastPositions[i][j].containsKey(c)) {
                        int lastPosition = lastPositions[i][j].get(c);
                        answer += Math.min(costs[i][j], costs[lastPosition][j]);
                        lastPositions[lastPosition][j].remove(c);
                    } else {
                        answer += costs[i][j];
                    }
                    lastPositions[i][j].put(c, i);
                }
            }
        }
        return answer;
    }
}
