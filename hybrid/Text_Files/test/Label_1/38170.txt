import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edges = [[] for i in range(n+1)]
for _ in range(m):
    a, b, c = map(int, input().split())
    edges[a].append([b, c])
    edges[b].append([a, c])

q = int(input())
for _ in range(q):
    u, v = map(int, input().split())
    vis = [False for i in range(n+1)]
    col = set()
    que = [(u, v)]
    vis[u] = True
    while que:
        p, q = que.pop(0)
        for k in edges[q]:
            if not vis[k[0]]:
                vis[k[0]] = True
                que.append((q, k[0]))
                if k[0] == p:
                    col.add(k[1])
    print(len(col))
