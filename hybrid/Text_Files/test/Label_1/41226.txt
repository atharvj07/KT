import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    i1, i2 = [], []
    for i in range(n):
        if b[i]==2: i1.append((a[i], i))
        else: i2.append((a[i], i))
    i1.sort()
    i2.sort()
    ans = 0
    for x, i in i1:
        if m<=0: break
        ans+=b[i]
        m-=x
    for x, i in i2:
        if m<=0: break
        ans+=b[i]
        m-=x
    if m>0: print(-1)
    else: print(ans)
