n,m=map(int,input().split())
a=[0 for i in range (n+1)]
b=[0 for i in range (m+1)]
for i in range (n):
 for j in range (m):
  if input()[j]=="#":
   a[i]+=1
   b[j]+=1
flag=1
for i in range (n):
 if a[i]==0 or a[i]%2==1:
  flag=0
for i in range (m):
 if b[i]==0 or b[i]%2==1:
  flag=0
if flag==0:
 print("No")
else:
 print("Yes")
