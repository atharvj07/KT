import java.util.*;
public class Main {
    static String[] s;
    static int l;
    public static void main(String[] args) {
        Scanner in=new Scanner(System.in);
        int n=in.nextInt();
        s=new String[n];
        for(int i=0;i<n;i++) {
            s[i]=in.next();
            l=Math.max(l,s[i].length());
        }
        for(int i=0;i<l;i++) {
            char ch=s[0].charAt(i);
            for(int j=0;j<n;j++) {
                if(ch!=s[j].charAt(i)) {
                    ch='?';
                    break;
                }
            }
            if(ch=='?') {
                System.out.println(-1);
                return;
            }
        }
        int ans=0;
        for(int i=0;i<l;i++) {
            for(int j=0;j<n;j++) {
                ans=(ans+i)%l;
                if(s[j].charAt(ans)!=s[0].charAt(i)) {
                    ans=-1;
                    break;
                }
            }
            if(ans==-1) break;
        }
        System.out.println(ans);
    }
}
