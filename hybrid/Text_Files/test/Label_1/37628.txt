def f(s):
    L="".join(s.split(":")[0][2:-1])
    R="".join(s.split(":")[1][1:-1])
    if L[0]!="_" or R[0]!="_":return"false"
    for x in L+R:
        if x not in "+->_<[]":return"false"
    p=0
    for x in L+R:
        if x in "->_<":p+=1;
        if x in "<_":p-=1;
        if p<0:return"false"
    v=[0 for x in range(len(L)+len(R)+1)]
    i=0
    p=len(L)+len(R)
    for c in L+R:
        if c=="+":v[p]+=1
        elif c=="-":v[p]-=1
        elif c==">":p-=1
        elif c=="<":p+=1
        elif c=="_":p=i
        i+=1
    for x in v:
        if x>9 or x<0:return"false"
    return"".join([str(x)for x in v])
for s in[input()]:
    print(f(s))
