#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, cnt;
  char c;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cnt = 0;
    for (int j = 0; j < n; j++) {
      cin >> c;
      if (c == 'o') {
        if (i == 0 or j == 0)
          cnt++;
        else if (i == n - 1 or j == n - 1)
          cnt++;
        else
          cnt += 4;
      }
    }
    if (cnt % 2) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
