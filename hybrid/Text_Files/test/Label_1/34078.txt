import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) a[i] = input.nextInt();
        for (int i = 0; i < n; i++) b[i] = input.nextInt();
        boolean[] vis = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (vis[a[i]]) {
                System.out.println("NO");
                return;
            }
            vis[a[i]] = true;
        }
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                for (int j = i + 1; j < n; j++) {
                    if (a[j] == b[i] && b[j] == a[i]) {
                        int d = Math.abs(i - j);
                        if (d == b[i]) {
                            a[i] = b[i];
                            a[j] = b[j];
                            break;
                        }
                    }
                }
            }
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
