from collections import deque

def is_valid(g):
    open = 0
    for c in g:
        if c in '{[(': open += 1
        elif c in ')}]': open -= 1
        if open < 0: return False
    return open == 0

def get_vars(g):
    ans = []
    for c in g:
        if c in '0123456789': ans += [int(c)]
    return ans

def simulate(g, vars):
    p = deque()
    for var in vars: p.append(var)
    cur = 0
    i = 0
    while i < len(g):
        c = g[i]
        if c in '{[(':
            p.append(cur)
            cur = 0
        elif c in ')}]':
            cur = p.pop()
        elif c == '+':
            cur += p.pop()
        elif c == '-':
            cur -= p.pop()
        elif c == '*':
            cur *= p.pop()
        elif c == '/':
            cur //= p.pop()
        elif c == '>':
            p.append(cur)
        elif c == '<':
            cur = p.pop()
        i += 1
    return cur

def solve(g):
    all_vars = get_vars(g)
    for vars in all_vars:
        if simulate(g, vars) == 0:
            return ''.join(map(str, vars))
    return "false"

g = input()
if not is_valid(g):
    print('false')
else:
    print(solve(g))
