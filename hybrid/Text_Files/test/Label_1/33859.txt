import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;

public class Main {

  static class Edge {
    int v1, v2;

    Edge(int v1, int v2) {
      this.v1 = v1;
      this.v2 = v2;
    }

    Edge() {

    }
  }

  static class Tree {

    HashMap<Integer, List<Integer>> adj;
    HashMap<Integer, Integer> color;
    ArrayList<Edge> edges;

    Tree() {
      this.adj = new HashMap<>();
      this.color = new HashMap<>();
      this.edges = new ArrayList<>();
    }

    void addEdge(int u, int v) {
      if (!adj.containsKey(u)) {
        adj.put(u, new ArrayList<>());
      }
      if (!adj.containsKey(v)) {
        adj.put(v, new ArrayList<>());
      }
      adj.get(u).add(v);
      adj.get(v).add(u);
      edges.add(new Edge(u, v));
    }

    void addVertex(int u, int color) {
      this.adj.put(u, new ArrayList<>());
      this.color.put(u, color);
    }
  }

  public static void dfs(Tree tree, int node, HashSet<Integer> visited, int color,
      ArrayList<Integer> sameComponents) {

    visited.add(node);
    sameComponents.add(node);
    for (int n : tree.adj.getOrDefault(node, new ArrayList<>())) {
      if (!visited.contains(n)) {
        if (tree.color.getOrDefault(n, -1) == color) {
          dfs(tree, n, visited, color, sameComponents);
        }
      }
    }
  }

  public static void main(String[] args) throws IOException {
    try (BufferedReader bf = new BufferedReader(new InputStreamReader(System.in))) {
      int t = Integer.parseInt(bf.readLine());
      StringBuilder sb = new StringBuilder();

      for (int i = 0; i < t; i++) {
        int n = Integer.parseInt(bf.readLine());
        int[] arr = new int[n];
        String[] arrStr = bf.readLine().trim().split("\\s+");

        for (int j = 0; j < n; j++) {
          arr[j] = Integer.parseInt(arrStr[j]);
        }

        Tree tree = new Tree();

        for (int j = 0; j < n; j++) {
          tree.addVertex(j + 1, arr[j]);
        }

        for (int j = 0; j < n - 1; j++) {
          String[] uv = bf.readLine().trim().split("\\s+");
          int u = Integer.parseInt(uv[0]), v = Integer.parseInt(uv[1]);
          tree.addEdge(u, v);
        }

        int minOps = n;

        for (int j = 0; j < n; j++) {
          if (tree.color.getOrDefault(j + 1, -1) == -1) {
            continue;
          }

          HashSet<Integer> visited = new HashSet<>();
          ArrayList<Integer> sameComponents = new ArrayList<>();

          dfs(tree, j + 1, visited, tree.color.getOrDefault(j + 1, -1), sameComponents);

          int white = 0, black = 0, grey = 0;

          for (int k : sameComponents) {
            if (tree.color.getOrDefault(k, -1) == 0) {
              grey++;
            } else if (tree.color.getOrDefault(k, -1) == 1) {
              white++;
            } else if (tree.color.getOrDefault(k, -1) == 2) {
              black++;
            }
          }

          minOps = Math.min(minOps, n - black - white - grey);

          for (int k : sameComponents) {
            tree.color.put(k, -1);
          }
        }

        sb.append(minOps).append("\n");
      }

      System.out.print(sb.toString());
    }
  }
}
