n = int(input().strip())
commands = input().strip()

def count_loops(commands):
    count = 0
    for i in range(len(commands)):
        x, y = 0, 0
        for j in range(i, len(commands)):
            if commands[j] == 'U':
                y += 1
            elif commands[j] == 'D':
                y -= 1
            elif commands[j] == 'R':
                x += 1
            elif commands[j] == 'L':
                x -= 1
            
            if x == 0 and y == 0:
                count += 1
    return count

print(count_loops(commands))
