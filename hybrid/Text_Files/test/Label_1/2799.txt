#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        vector<int> open;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '(')
                open.push_back(i);
            else {
                if (open.empty())
                    ans++;
                else
                    open.pop_back();
            }
        }
        cout << ans << endl;
    }
    return 0;
}
