#include <iostream>
#include <vector>
using namespace std;

int f(const vector<long long>& a, int n) {
  for (int i = 2; i < 2e9; i++) {
    bool ok = true;
    for (int j = 0; j < n; j++) {
      if (j > 0 && a[j] % i == a[j - 1] % i) {
        ok = false;
        break;
      }
    }
    if (ok) return i;
  }
  return 0;
}

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cout << f(a, n) << '\n';
  }
}
