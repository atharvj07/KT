n, m, k = map(int, input().split())
a = list(map(int, input().split()))
a[m - 1] = -1
ans = 1e9
j = 1
for i in range(0, n):
    if a[i] == 0:
        ans = min(ans, abs(j - m) * 10)
    else:
        if a[i] <= k:
            ans = min(ans, abs(j - m) * 10)
    if a[i] > 0:
        while j < n and a[j] > 0:
            j += 1
    j += 1
print(ans)
