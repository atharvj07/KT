#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N = 2e5 + 5;
int32_t main() {
    int t, n;
    cin >> t;
    while(t--) {
        cin >> n;
        int arr[N], r[N], c[N];
        for(int i = 0; i < n; i++) { cin >> r[i]; }
        for(int i = 0; i < n; i++) { cin >> c[i]; }
        for(int i = 0; i < n; i++) { arr[i] = r[i] + c[i]; }
        int dp[n + 1];
        memset(dp, -1, sizeof dp);
        dp[0] = 0;
        for(int i = 0; i < n; i++) {
            for(int j = 0; j <= i; j++) {
                if(dp[j] != -1) {
                    if(arr[i] % 2 == 0) {
                        dp[i + 1] = max(dp[i + 1], dp[j] + (arr[i] / 2 - c[i]));
                    } else {
                        dp[i + 1] = max(dp[i + 1], dp[j] + c[i] - arr[i] / 2);
                    }
                }
            }
        }
        int ans = 0;
        for(int i = 0; i <= n; i++) {
            ans = max(ans, dp[i]);
        }
        cout << ans << "\n";
    }
}
