#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MAXN = 2e5 + 5;
const ll inf = 1e18;
struct car {
    ll cost, fuel;
};
struct gas {
    ll pos;
    bool operator<(gas a) const { return pos < a.pos; }
};
car cars[MAXN];
gas gas_stations[MAXN];
int main() {
    ll n, k, s, t;
    cin >> n >> k >> s >> t;
    for (int i = 1; i <= n; i++) { cin >> cars[i].cost >> cars[i].fuel; }
    for (int i = 1; i <= k; i++) { cin >> gas_stations[i].pos; }
    sort(gas_stations + 1, gas_stations + k + 1);
    ll ans = -1;
    ll l = 1, r = n;
    while (l <= r) {
        ll mid = (l + r) / 2;
        ll time[s + 1] = {0};
        ll fuel[s + 1] = {0};
        fuel[0] = cars[mid].fuel;
        bool flag = true;
        for (ll i = 1; i <= k; i++) {
            ll dist = gas_stations[i].pos;
            ll time_needed = fuel[i - 1] / 2 + (dist - fuel[i - 1]) / 1;
            if (fuel[i - 1] + 1 >= dist) {
                time_needed = fuel[i - 1] / 2;
            }
            if (time_needed <= t - gas_stations[i].pos) {
                time[gas_stations[i].pos] = time_needed;
                fuel[gas_stations[i].pos] = fuel[i - 1] + 2;
            } else {
                flag = false;
                break;
            }
        }
        ll dist = s - gas_stations[k].pos;
        ll time_needed = fuel[k] / 2 + (dist - fuel[k]) / 1;
        if (fuel[k] + 1 >= dist) {
            time_needed = fuel[k] / 2;
        }
        if (time_needed <= t - gas_stations[k].pos) {
            time[s] = time_needed;
            fuel[s] = fuel[k] + 2;
        } else {
            flag = false;
        }
        if (flag && time[s] <= t) {
            ans = cars[mid].cost;
            r = mid - 1;
        } else
            l = mid + 1;
    }
    cout << ans << endl;
}
