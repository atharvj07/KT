#include <bits/stdc++.h>
using namespace std;
int n, k, a[111], b[111], c;
int main() {
  ios_base::sync_with_stdio(0), cin.tie(0);
  cin >> n >> k;
  for (int i = 0; i < n; i++) cin >> a[i], b[i] = a[i];
  sort(b, b + n);
  int l = 0, r = n - 1, mx = b[r] - b[l];
  while (k && mx > 0) {
    if (b[l] == a[l] && b[r] == a[r]) {
      b[l]++, a[r]--, c++, k--;
    } else if (b[l] == a[l]) {
      b[r]++, a[r]--, c++, k--;
    } else if (b[r] == a[r]) {
      b[l]--, a[l]++, c++, k--;
    } else if (a[l] > 0) {
      a[l]--, a[r]++, c++, k--;
    } else {
      a[r]--, a[l]++, c++, k--;
    }
    sort(b, b + n);
    mx = b[r] - b[l];
    if (mx == 0) break;
  }
  cout << mx << ' ' << c << '\n';
  for (int i = 0; i < c; i++) {
    for (int j = 0; j < n; j++) {
      if (b[r] > a[j]) {
        b[r]--, a[j]++;
        cout << j + 1 << ' ' << r + 1 << '\n';
        break;
      }
    }
    sort(b, b + n);
  }
}
