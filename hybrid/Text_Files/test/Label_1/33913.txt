import java.util.*;

public class Solution {
    private static final int MAX_N = 200000;
    private static final int MAX_M = 200000;

    private static int[] parent = new int[MAX_N + 1];
    private static int[] depth = new int[MAX_N + 1];
    private static List<Integer>[] adjList = new List[MAX_N + 1];

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        for (int i = 0; i <= n; i++) {
            adjList[i] = new ArrayList<>();
        }

        for (int i = 1; i < n; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            adjList[u].add(v);
            adjList[v].add(u);
        }

        dfs(1, 0);

        for (int i = 0; i < m; i++) {
            int k = scanner.nextInt();
            int[] vertices = new int[k];
            for (int j = 0; j < k; j++) {
                vertices[j] = scanner.nextInt();
            }

            if (isValid(vertices)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }

    private static void dfs(int node, int parentNode) {
        parent[node] = parentNode;
        depth[node] = depth[parentNode] + 1;

        for (int child : adjList[node]) {
            if (child != parentNode) {
                dfs(child, node);
            }
        }
    }

    private static boolean isValid(int[] vertices) {
        int rootDepth = depth[1];
        Set<Integer> pathVertices = new HashSet<>();
        Set<Integer> adjacentVertices = new HashSet<>();

        for (int vertex : vertices) {
            if (depth[vertex] == rootDepth) {
                pathVertices.add(vertex);
            } else if (depth[vertex] == rootDepth + 1) {
                adjacentVertices.add(parent[vertex]);
            } else {
                return false;
            }
        }

        return pathVertices.size() + adjacentVertices.size() == vertices.length;
    }
}
