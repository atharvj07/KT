#include <bits/stdc++.h>
using namespace std;

#define endl "\n"

typedef long long ll;
typedef long double ld;
typedef pair<ll, ll> pi;
typedef pair<ll, ll> pl;

const ll N = 1e5 + 10;
vector<ll> adj[N];
ll n, odd;

void dfs(ll u, ll p) {
    for (ll v : adj[u]) {
        if (v == p)
            continue;
        dfs(v, u);
        odd += (adj[v].size() % 2);
    }
    odd += (adj[u].size() % 2);
}

int main() {
    cin >> n;
    for (ll i = 0; i < n - 1; i++) {
        ll u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1);
    if (odd > 2)
        cout << -1 << endl;
    else
        cout << n - 1 - (odd / 2) << endl;
    return 0;
}
