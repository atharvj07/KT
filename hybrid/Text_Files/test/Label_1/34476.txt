import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int[] p = new int[(n + k - 1) / k];
        for (int i = 0; i < n; i++)
            p[i / k] += a[i];
        Arrays.sort(p);
        int ans = 1;
        for (int i = 1; i < (n + k - 1) / k; i++)
            ans += p[i - 1];
        System.out.println(ans);
    }
}
