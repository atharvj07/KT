import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        Arrays.sort(a);
        long l = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > n - 1 - i) {
                System.out.println(-1);
                return;
            }
            l += a[i];
        }
        if (l < n - 1) {
            System.out.println(-1);
            return;
        }
        for (int i = 0; i <= n - 1; i++) {
            if (l - i < n - 1 - i) {
                System.out.println(-1);
                return;
            }
            System.out.print(i + " ");
        }
    }
}
