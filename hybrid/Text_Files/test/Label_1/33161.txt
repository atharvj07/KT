import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            b[i] = a[i];
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            int min = b[i];
            int index = i;
            for (int j = i + 1; j < n; j++) {
                if (b[j] < min) {
                    min = b[j];
                    index = j;
                }
            }
            if (min != b[i]) {
                ans++;
            }
            int t = b[i];
            b[i] = b[index];
            b[index] = t;
        }
        System.out.println(ans);
    }
}
