#include <vector>
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<bool>> knows(n + 1, vector<bool>(m + 1, false));
    int ans = n * m;
    for (int i = 1; i <= n; i++) {
        int k;
        cin >> k;
        for (int j = 1; j <= k; j++) {
            int a;
            cin >> a;
            knows[i][a] = true;
        }
    }
    for (int a = 1; a <= m; a++) {
        vector<bool> vis(n + 1, false);
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            if (!knows[i][a]) continue;
            if (vis[i]) continue;
            cnt++;
            vector<int> q = {i};
            vis[i] = true;
            while (!q.empty()) {
                int now = q.back();
                q.pop_back();
                for (int j = 1; j <= m; j++) {
                    if (!knows[now][j]) continue;
                    if (vis[j]) continue;
                    vis[j] = true;
                    q.push_back(j);
                }
            }
        }
        ans = min(ans, n - cnt);
    }
    cout << ans << endl;
    return 0;
}
