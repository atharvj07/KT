def solve(n, a):
    bit_count = [0] * 30
    for x in a:
        for bit in range(30):
            if x & (1 << bit):
                bit_count[bit] += 1
    divisors = []
    for gcd in range(1, n+1):
        if all(count % gcd == 0 for count in bit_count):
            divisors.append(gcd)
    return divisors

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    result = solve(n, a)
    print(*result)
