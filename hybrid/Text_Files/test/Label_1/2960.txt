#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int N = 2e5 + 5;

int n, m;
ll a[N], b[N];

int getInv(int l, int r) {
  int cnt[n + 5] = {};
  ll ans = 0;
  for (int i = l; i <= r; i++) cnt[a[i]]++;
  for (int i = l; i <= r; i++) {
    for (int j = 1; j < a[i]; j++) ans += cnt[j];
    cnt[a[i]]--;
  }
  return ans % 2;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; b[a[i]] = i; }
  cin >> m;
  for (int i = 1, l, r; i <= m; i++) {
    cin >> l >> r;
    if (l > r) swap(l, r);
    reverse(a + l, a + r + 1);
    if (getInv(l, r)) cout << "odd" << endl; else cout << "even" << endl;
  }
  return 0;
}
