n = int(input())
points = [list(map(int, input().split())) for i in range(n)]
adj = [[] for i in range(n)]
bad = [False for i in range(n)]
for i in range(n):
    for j in range(i+1,n):
        x,y = points[i], points[j]
        if x[0]*y[0]+x[1]*y[1]+x[2]*y[2]+x[3]*y[3]+x[4]*y[4] < 0:
            adj[i].append(j)
            adj[j].append(i)
            bad[i] = True
            bad[j] = True
good = [str(i+1) for i in range(n) if not bad[i]]
print(len(good))
print("\n".join(good))
