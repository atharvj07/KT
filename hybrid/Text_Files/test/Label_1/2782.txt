#include<bits/stdc++.h>
using namespace std;
int main() {
  int n; cin >> n;
  vector<int> a(n);
  for (auto &ai : a) cin >> ai;
  int m; cin >> m;
  vector<int> b(m);
  for (auto &bi : b) cin >> bi;
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  int sa = accumulate(a.begin(), a.end(), 0LL);
  int sb = accumulate(b.begin(), b.end(), 0LL);
  int ta = 0, tb = 0, ans = abs(sa - sb), c = 0;
  for (int i = 0; i < n; i++) {
    ta += a[i];
    tb -= b[i];
    int v = abs(sa - ta - sb + tb);
    if (v < ans) {
      ans = v;
      c = 1;
      sa = ta, sb = tb;
    }
  }
  int ta2 = 0, tb2 = 0;
  for (int i = 0; i < m; i++) {
    ta2 -= a[i];
    tb2 += b[i];
    int v = abs(sa - ta2 - sb + tb2);
    if (v < ans) {
      ans = v;
      c = 2;
      sa = ta2, sb = tb2;
    }
  }
  cout << ans << '\n' << c << '\n';
  if (c == 1) {
    for (int i = 0; i < n; i++) {
      int v = abs(sa - a[i] - sb + b[i]);
      if (abs(sa - a[i] - sb + b[i]) == ans) {
        cout << i + 1 << ' ' << i + 1 << endl;
        break;
      }
    }
  }
  if (c == 2) {
    for (int i = 0; i < n; i++) {
      int v = abs(sa - a[i] - sb + b[i]);
      if (abs(sa - a[i] - sb + b[i]) == ans) {
        cout << i + 1 << ' ' << i + 1 << endl;
        break;
      }
    }
  }
  return 0;
}
