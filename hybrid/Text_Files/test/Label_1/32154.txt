import java.util.*;
import java.io.*;

public class Main {
  static int n, x, y;
  static String s;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    x = Integer.parseInt(st.nextToken());
    y = Integer.parseInt(st.nextToken());
    s = br.readLine();

    System.out.println(solve());
  }

  static int solve() {
    int cur = 0;
    for (int i = 0; i < n; i++) {
      int pos = (n - i) % x;
      if (s.charAt(i) == '0') {
        if (((cur >> pos) & 1) == 1) cur++;
      } else {
        if (((cur >> pos) & 1) == 0) cur++;
      }
    }
    int ans = Integer.MAX_VALUE;
    for (int i = n - 1; i >= 0; i--) {
      int pos = (n - i) % x;
      int bit = Integer.parseInt(s.substring(i, i + 1));
      int prev = (cur >> pos) & 1;
      if (bit == 1) {
        if (((cur >> pos) & 1) == 1) cur++;
        for (int j = i - 1; j >= 0; j--) {
          pos = (n - j) % x;
          bit = Integer.parseInt(s.substring(j, j + 1));
          if (bit == 1) {
            if (((cur >> pos) & 1) == 0) cur++;
          }
        }
        ans = Math.min(ans, cur - 1);
        cur = ((cur >> pos) << pos) | ((1 - prev) << pos);
      }
    }
    return ans;
  }
}
