#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> v(n);
  for (int &x : v) cin >> x;
  vector<double> pf(n + 1);
  for (int i = 0; i < n; i++) pf[i + 1] = pf[i] + v[i];
  double ans = (double)(pf[k] - pf[0]) / k;
  for (int i = k; i < n; i++) {
    ans = max(ans, (double)(pf[i + 1] - pf[i - k + 1]) / k);
  }
  cout << ans << '\n';
  return 0;
}
