import java.io.*;

class Main {

    public static void main(String[] args) {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        try {
            String[] line = br.readLine().split(" ", 3);
            int n = Integer.parseInt(line[0]);
            int m = Integer.parseInt(line[1]);
            int k = Integer.parseInt(line[2]);
            char[][] matrix = new char[n + 2][m + 2];

            for (int i = 1; i <= n; i++) {
                String[] strs = br.readLine().split("", m);
                for (int j = 1; j <= m; j++) {
                    matrix[i][j] = strs[j - 1].charAt(0);
                }
            }

            long[][] prefixHorizonal = new long[n + 2][m + 2];
            long[][] prefixVertical = new long[n + 2][m + 2];

            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= m; j++) {
                    if (matrix[i][j] == '1') {
                        prefixHorizonal[i][j] = prefixHorizonal[i][j - 1] + 1;
                        prefixVertical[i][j] = prefixVertical[i - 1][j] + 1;
                    }
                }
            }

            long ans = 0;
            for (int i1 = 1; i1 <= n; i1++) {
                for (int j1 = 1; j1 <= m; j1++) {
                    for (int i2 = i1; i2 <= n; i2++) {
                        for (int j2 = j1; j2 <= m; j2++) {
                            long sum = prefixVertical[i2][j2] - prefixVertical[i2][j1 - 1] - prefixVertical[i1 - 1][j2] + prefixVertical[i1 - 1][j1 - 1];
                            sum += prefixHorizonal[i2][j2] - prefixHorizonal[i2][j1 - 1] - prefixHorizonal[i1 - 1][j2] + prefixHorizonal[i1 - 1][j1 - 1];
                            if (sum >= k) ans++;
                        }
                    }
                }
            }

            System.out.println(ans);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

}
