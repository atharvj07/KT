#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (auto &x : a) cin >> x;
  sort(a.begin(), a.end());
  long long ans = a.back() - a.front();
  k = min(k, n - 1);
  ans = min(ans, (a.back() - a[k]) + (a[n - 1 - k] - a.front()));
  cout << ans << endl;
  return 0;
}
