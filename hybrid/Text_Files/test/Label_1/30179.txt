import java.util.*;

class Main {
  private static int n, p;
  private static int[][] a;
  private static char[] s;

  private static int solve() {
    Queue<Integer> q = new LinkedList<>();
    int[] deg = new int[p];
    for (int i = 0; i < p; i++) {
      for (int j = 0; j < p; j++) {
        if (a[i][j] > 0) {
          deg[i]++;
        }
      }
    }
    for (int i = 0; i < p; i++) {
      if (deg[i] == 1) {
        q.add(i);
      }
    }
    while (!q.isEmpty()) {
      int u = q.poll();
      int cnt = 0;
      for (int j = 0; j < p; j++) {
        if (a[u][j] > 0) {
          deg[j]--;
          if (deg[j] == 1) {
            q.add(j);
          }
          cnt++;
        }
      }
      for (int i = 0; i < n; i++) {
        if (s[i] == u + 'a') {
          n -= cnt;
          for (int j = i + 1; j < n; j++) {
            s[j - cnt] = s[j];
          }
          i--;
        }
      }
    }
    return n;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    n = in.nextInt();
    p = in.nextInt();
    a = new int[p][p];
    s = new String(in.next()).toCharArray();
    for (int i = 0; i < p; i++) {
      for (int j = 0; j < p; j++) {
        a[i][j] = in.nextInt();
      }
    }
    System.out.println(solve());
  }
}
