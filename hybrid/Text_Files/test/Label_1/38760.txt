t = int(input())
for _ in range(t):
    a, b, x, y, n = map(int, input().split())
    def min_product(a, b, x, y, n):
        option1 = max(x, a - n) * max(y, b - (n - (a - max(x, a - n))))
        option2 = max(y, b - n) * max(x, a - (n - (b - max(y, b - n))))
        return min(option1, option2)
    print(min_product(a, b, x, y, n))
