import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        int b = -1;
        int cnta = 0;
        boolean found = false;
        for (int i = 0; i < n && !found; i++) {
            if (c[i] != a) {
                if (cnta == 0) {
                    b = c[i];
                }
                cnta--;
            } else {
                cnta++;
            }
            if (cnta < 0) {
                found = true;
            }
        }
        System.out.println(b);
    }
}
