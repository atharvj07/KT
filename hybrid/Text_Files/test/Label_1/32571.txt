import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int d = sc.nextInt();
        int n = sc.nextInt(), m = sc.nextInt();
        int[] lx = new int[d + 1], rx = new int[d + 1], uy = new int[d + 1], dy = new int[d + 1];
        for (int i = 1; i <= d; i++) {
            int x1 = sc.nextInt(), y1 = sc.nextInt(), x2 = sc.nextInt(), y2 = sc.nextInt();
            if (x1 == x2) {
                lx[i] = rx[i] = x1;
                uy[i] = Math.min(y1, y2);
                dy[i] = Math.max(y1, y2);
            } else {
                uy[i] = dy[i] = y1;
                lx[i] = Math.min(x1, x2);
                rx[i] = Math.max(x1, x2);
            }
        }
        int[] l = new int[d + 1], r = new int[d + 1], u = new int[d + 1], dwn = new int[d + 1];
        for (int i = 1; i <= d; i++) {
            l[i] = lx[i] - 1;
            r[i] = n - rx[i];
            u[i] = uy[i] - 1;
            dwn[i] = m - dy[i];
        }
        int cntl = sc.nextInt(), cntr = sc.nextInt(), cntt = sc.nextInt(), cntb = sc.nextInt();
        int ans = -1;
        for (int i = 1; i <= d; i++) {
            if (l[i] == cntl && r[i] == cntr && u[i] == cntt && dwn[i] == cntb) {
                ans = i;
                break;
            }
        }
        System.out.println(ans);
    }
}
