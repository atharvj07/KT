import bisect
n,h,m,k=map(int,input().split())
time=[]
for i in range(n):
    x,y=map(int,input().split())
    time.append((x*m+y))
time.sort()
ans=h*2
ansT=0
L=0
for i in time:
    if i<h*m-k:
        L+=1
        ans-=1
    else:
        ansT=h*m-i
        break
if ans<0:ans=0
print(ans,ansT)
time=time[L:]
for i in time:
    if i<=h*m-k:
        print(i//m,i%m)
