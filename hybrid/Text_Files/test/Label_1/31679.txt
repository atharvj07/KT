import java.util.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++)
      arr[i] = in.nextInt();

    int ans = Integer.MAX_VALUE;
    for (int i = 1; i < n - 1; i++)
      ans = Math.min(ans, arr[i + 1] - arr[i - 1]);

    System.out.println(ans);
  }
}
