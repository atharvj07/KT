import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {
  static class Request implements Comparable<Request> {
    int start;
    int duration;

    public Request(int start, int duration) {
      this.start = start;
      this.duration = duration;
    }

    @Override
    public int compareTo(Request other) {
      if (start == other.start) {
        return duration - other.duration;
      }
      return start - other.start;
    }
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());

    Request[] requests = new Request[n];
    for (int i = 0; i < n; i++) {
      StringTokenizer st = new StringTokenizer(br.readLine());
      int start = Integer.parseInt(st.nextToken());
      int duration = Integer.parseInt(st.nextToken());
      requests[i] = new Request(start, duration);
    }

    Arrays.sort(requests);

    long lastFinish = 0;
    for (Request request : requests) {
      if (request.start < lastFinish) {
        request.start = (int) lastFinish;
      }
      lastFinish = request.start + request.duration - 1;
      System.out.println(request.start + " " + lastFinish);
    }
  }
}
