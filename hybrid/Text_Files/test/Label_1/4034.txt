#include <iostream>
#include <vector>
using namespace std;
int64_t H(vector<int64_t> &v) {
    // prefix sum of #occurrences
    vector<int64_t> s(v.size() + 1, 0);
    for (int i = 0; i < v.size(); ++i) s[i + 1] = s[i] + (v[i] != 0);
    // weight of a single element
    int64_t w = 0;
    for (int i = 0; i < v.size(); ++i) w += s[i] * (s[i + 1] - s[i]);
    // weight for ranges with 2 elements
    for (int i = 0; i < v.size(); ++i) {
        int64_t x = s[i + 1] - s[i];
        w += x * (x - 1);
    }
    // weight for range with > 2 elements
    w -= v.size();
    return w;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int64_t> v(n);
        for (int i = 0; i < n; ++i) cin >> v[i];
        cout << H(v) << endl;
    }
}
