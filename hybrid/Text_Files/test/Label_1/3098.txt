#include<bits/stdc++.h>
typedef long long ll;
using namespace std;
int main() {
    ll y, k, n;
    cin >> y >> k >> n;
    if ((n - y) % k != 0) cout << -1 << endl;
    else {
        for (ll x = max(0LL, (n - y) / k); x * k + y <= n; x++) {
            cout << x * k + y << ' ';
        }
        cout << endl;
    }
}
