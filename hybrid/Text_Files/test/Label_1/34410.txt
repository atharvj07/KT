import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(reader.readLine());
            int[][] time = new int[n][2];
            int i, t, size = 0, ans = 0;
            for (i = 0; i < n; i++) {
                String s = reader.readLine();
                time[i][0] = Integer.parseInt(s.split(" ")[0]);
                time[i][1] = Integer.parseInt(s.split(" ")[1]);
                if (i > 0 && time[i][0] == time[i - 1][0] && time[i][1] == time[i - 1][1]) size--;
            }
            Arrays.sort(time, (a, b) -> {
                if (a[0] == b[0]) return a[1] - b[1];
                return a[0] - b[0];
            });

            for (i = t = size = 0; i < n; i++) {
                if (i > 0 && time[i][0] == time[i - 1][0] && time[i][1] == time[i - 1][1]) continue;
                if (t == time[i][0] && size == 0) size++;
                if (size > 0) {
                    size--;
                    t++;
                }
                if (size == 0) {
                    size++;
                    t = time[i][0];
                }
                ans = Math.max(ans, size);
            }
            System.out.println(ans);
        } catch (IOException e) {
        }
    }
}
