def find_centroids(n, edges):
    graph = [[] for _ in range(n)]
    for u, v in edges:
        graph[u-1].append(v-1)
        graph[v-1].append(u-1)

    size = [1] * n

    def dfs(v, p):
        for u in graph[v]:
            if u == p: continue
            dfs(u, v)
            size[v] += size[u]

    def find_centroid(v, p, n):
        for u in graph[v]:
            if u != p and size[u] > n // 2:
                return find_centroid(u, v, n)
        return v

    centroid = find_centroid(0, -1, n)
    if size[centroid] * 2 == n:
        for u in graph[centroid]:
            if size[u] == n // 2:
                return (centroid+1, u+1), (centroid+1, u+1)

    p = -1
    for u in graph[centroid]:
        if p == -1 or size[u] > size[p]:
            p = u

    cut = (centroid+1, p+1)
    add = (centroid+1, p+1)

    return cut, add

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    edges = [tuple(map(int, input().strip().split())) for _ in range(n-1)]
    cut, add = find_centroids(n, edges)
    print(f'{cut[0]} {cut[1]}')
    print(f'{add[0]} {add[1]}')
