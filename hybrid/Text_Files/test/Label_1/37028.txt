def solve_sequence_problem(n, a):
    left, right = 0, n - 1
    last_taken = 0
    result = ''
    while left <= right:
        if a[left] < a[right]:
            if a[left] > last_taken:
                last_taken = a[left]
                left += 1
                result += 'L'
            elif a[right] > last_taken:
                last_taken = a[right]
                right -= 1
                result += 'R'
            else:
                break
        else:
            if a[right] > last_taken:
                last_taken = a[right]
                right -= 1
                result += 'R'
            elif a[left] > last_taken:
                last_taken = a[left]
                left += 1
                result += 'L'
            else:
                break
    print(len(result))
    print(result)

n = int(input().strip())
a = list(map(int, input().strip().split()))
solve_sequence_problem(n, a)
