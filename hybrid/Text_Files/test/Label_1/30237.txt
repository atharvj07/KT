import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] b = new int[n];
        int k = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                b[k++] = a[i];
            }
        }
        if (k == 0 || b[k - 1] != b[0]) {
            System.out.println(-1);
            return;
        }
        int m = b[0];
        int[] c = new int[m + 1];
        int[] d = new int[m + 1];
        for (int i = 0; i < k; i++) {
            c[b[i]]++;
            d[b[i]] = Math.max(d[b[i]], i - c[b[i]] + 1);
        }
        int max = 0;
        for (int i = 1; i <= m; i++) {
            max = Math.max(max, d[i]);
        }
        int[] e = new int[n];
        int p = 0;
        for (int i = 1; i <= m; i++) {
            if (d[i] == max) {
                for (int j = 0; j < max; j++) {
                    e[p++] = i;
                }
            }
        }
        int[] f = new int[n];
        int r = 0;
        int s = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                f[i] = a[i];
                r++;
            } else {
                f[i] = e[s++];
                r++;
            }
            if (r >= max) {
                r = 0;
                s = 0;
            }
        }
        System.out.println(max);
        System.out.println(Arrays.toString(f).replaceAll("[\\[\\], ]", ""));
    }
}
