#include<bits/stdc++.h>
using namespace std;
const int N = 3e5 + 5;
int n, m, s, u[N], v[N], type[N];
int col[N];
vector<int> G[N];
queue<int> Q;
int main(){
    cin >> n >> m >> s;
    for(int i = 0 ; i < m ; i++){
        cin >> type[i] >> u[i] >> v[i];
        if(type[i] == 2){
            G[u[i]].push_back(v[i]);
            G[v[i]].push_back(u[i]);
        }
    }
    col[s] = 1;
    Q.push(s);
    int reach1 = 1;
    string plan1 = "";
    while(Q.size()){
        int cur = Q.front();
        Q.pop();
        for(int adj : G[cur]){
            if(!col[adj]){
                col[adj] = col[cur] + 1;
                reach1++;
                Q.push(adj);
            }
        }
    }
    plan1 = string(m, '+');
    for(int i = 0 ; i < m ; i++){
        if(type[i] != 2) continue;
        else{
            if(col[v[i]] > col[u[i]])
                plan1[i] = '-';
        }
    }
    memset(col, 0, sizeof col);
    Q.push(s);
    int reach2 = 1;
    string plan2 = "";
    while(Q.size()){
        int cur = Q.front();
        Q.pop();
        for(int adj : G[cur]){
            if(!col[adj]){
                col[adj] = col[cur] + 1;
                reach2++;
                Q.push(adj);
            }
        }
    }
    plan2 = string(m, '+');
    for(int i = 0 ; i < m ; i++){
        if(type[i] != 2) continue;
        else{
            if(col[u[i]] > col[v[i]])
                plan2[i] = '-';
        }
    }
    cout << reach1 << '\n' << plan1 << '\n'; 
    cout << reach2 << '\n' << plan2 << '\n';
}
