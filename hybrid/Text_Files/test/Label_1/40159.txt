def max_game_time(a_1, a_2):
    minutes = 0
    while a_1 > 0 and a_2 > 0:
        if a_1 == 1 and a_2 == 1:
            break
        if a_1 < a_2:
            a_1 += 1
            a_2 -= 2
        else:
            a_2 += 1
            a_1 -= 2
        minutes += 1
    return minutes

a_1, a_2 = map(int, input().split())
print(max_game_time(a_1, a_2))
