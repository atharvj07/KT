import java.util.*;
import java.math.*;
public class Main{
    public static void main(String[] args){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int d = in.nextInt();
        long[] arr = new long[n];
        long[] ans = new long[n];
        for(int i=0; i<n; i++){
            arr[i] = in.nextLong() + in.nextLong();
        }
        Arrays.sort(arr);
        for(int i=0; i<n-1; i++){
            if(arr[i+1] - arr[i] <= d){
                ans[i+1] = ans[i];
            }else{
                ans[i+1] = ans[i]+1;
            }
        }
        long max = 0;
        long cur = 0;
        for(int i=n-1; i>=0; i--){
            cur += i;
            max = Math.max(max, cur+ans[i]);
        }
        System.out.println(max);
    }
}
