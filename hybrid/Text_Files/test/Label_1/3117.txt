#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> dp(n + 1, n + 1);
    dp[n] = 0;
    for (int i = n - 1; i >= 1; i--) {
        if (s[i] == '1') {
            dp[i] = min(dp[i], dp[i + k + 1] + i);
        }
        dp[i] = min(dp[i], dp[i + 1] + i);
    }
    cout << dp[1] << endl;
    return 0;
}
