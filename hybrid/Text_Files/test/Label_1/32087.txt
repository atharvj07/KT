import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), h = in.nextInt(), m = in.nextInt(), k = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; ++i) {
            arr[i] = in.nextInt() * m + in.nextInt();
        }

        int t = 0, cnt = 0;
        Arrays.sort(arr);

        for (int x : arr) {
            if (x - t < k) {
                ++cnt;
            } else {
                t = x - k;
            }
        }
        System.out.println(cnt + " " + t);
        for (int i = 0; i < n; ++i) {
            if (arr[i] >= t && arr[i] - t < k) {
                System.out.print((arr[i] - t) / m + " " + arr[i] % m + "\n");
            }
        }
    }
}
