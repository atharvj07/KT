import sys

t = int(input())

for _ in range(t):
    n, m = map(int, input().split())
    g = [['*' for j in range(n)] for i in range(n)]

    for i in range(n):
        s = input()
        for j in range(n):
            g[i][j] = s[j]

    if m % 2 == 1:
        print("NO")
        continue

    m //= 2
    odd = []
    for i in range(n):
        for j in range(n):
            if g[i][j] != '*' and g[i][j] != g[j][i]:
                odd.append((i, j))

    if len(odd) % 2 == 1:
        print("NO")
        continue

    path = []
    used = set()
    q = [(i, i) for i in range(n) if len(g[i]) % 2 == 1]
    while q and len(path) < m:
        i, j = q.pop()
        if i == j and len(used) % 2 == 0:
            path.append(i)
        elif len(used) % 2 == 1 and (i, j) in used:
            path.append(i)
        else:
            if len(g[i]) % 2 == 1:
                path.append(i)
            for k in range(n):
                if k != j and g[j][k] == g[k][j] and (j, k) not in used:
                    q.append((j, k))
            used.add((i, j))
    if len(path) == m:
        print("YES")
        print(*path, path[-2])
    else:
        print("NO")
