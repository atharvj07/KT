import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextLong();
    }
    long[] pre = new long[n + 1];
    for (int i = 0; i < n; i++) {
      pre[i + 1] = pre[i] ^ a[i];
    }
    long ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 3; j <= n; j++) {
        if ((pre[j - 1] ^ pre[i]) ==
            a[i] + a[j - 1]) {
          ans++;
        }
      }
    }
    System.out.println(ans);
  }
}
