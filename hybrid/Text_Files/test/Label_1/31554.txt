import java.util.Scanner;

public class IahubinaTree {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        if (isTreePossible(c)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isTreePossible(int[] c) {
        int sum = 0;
        for (int value : c) {
            sum += value;
        }

        if (sum != c.length) {
            return false;
        }

        int[] degree = new int[c.length];
        for (int i = 0; i < c.length; i++) {
            degree[i] = c[i] - 1;
        }

        int rootCount = 0;
        for (int i = 0; i < c.length; i++) {
            if (degree[i] == 0) {
                rootCount++;
            }
        }

        if (rootCount != 1) {
            return false;
        }

        for (int i = 0; i < c.length; i++) {
            if (degree[i] > 1 && c[i] == 1) {
                return false;
            }
        }

        return true;
    }
}
