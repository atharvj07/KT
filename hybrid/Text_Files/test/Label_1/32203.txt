import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    static String next() {
        try {
            return br.readLine();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }
    static void print(Object o) {
        pw.print(o);
        pw.flush();
    }
    static void println(Object o) {
        pw.println(o);
        pw.flush();
    }

    static BufferedReader br;
    static PrintWriter pw;
    static StringTokenizer st;
    static long[] Com;
    static int[] Fac;
    static long[] InvFac;
    static int Mod =998244353;

    public static void main(String[] args) {
        br = new BufferedReader(new InputStreamReader(System.in), 32768);
        pw = new PrintWriter(new OutputStreamWriter(System.out));

        String s = next();
        String t = next();
        int[] cntS = new int[26];
        int[] cntT = new int[26];
        for (char c: s.toCharArray()) ++cntS[c - 'a'];
        for (char c: t.toCharArray()) ++cntT[c - 'a'];
        
        StringBuilder ans = new StringBuilder();
        boolean done = false;
        for (char c = 'a'; c <= 'z'; c++) {
            int add = cntS[c - 'a'] - cntT[c - 'a'];
            for (int i = 0; i < add; i++) ans.append(c);
            if (!done && cntT[c - 'a'] > 0) {
                ans.append(c);
                done = true;
            }
        }
        for (char c = 'a'; c <= 'z'; c++) {
            for (int i = cntT[c - 'a']; i < cntS[c - 'a']; i++) ans.append(c);
        }
        if (ans.length() == 0 || ans.toString().compareTo(t) <= 0) {
            println("-1");
        } else {
            println(ans.toString());
        }
    }
}
