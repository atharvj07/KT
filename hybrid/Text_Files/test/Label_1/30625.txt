import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr1 = new long[n];
        long[] arr2 = new long[n];
        long xor = 0;
        for (int i = 0; i < n; i++) {
            arr1[i] = sc.nextLong();
            xor ^= arr1[i];
        }
        for (int i = 0; i < n; i++) {
            arr2[i] = sc.nextLong();
            xor ^= arr2[i];
        }
        System.out.println(xor == 0 ? "Koyomi" : "Karen");
    }
}
