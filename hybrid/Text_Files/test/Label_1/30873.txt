import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        Map<Integer, Set<Integer>> g = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            g.put(i, new HashSet<>());
        }
        Map<Integer, Integer> edge = new HashMap<>();
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int w = Integer.parseInt(st.nextToken());
            edge.put(u*n + v, w);
            edge.put(v*n + u, w);
            g.get(u).add(v);
            g.get(v).add(u);
        }

        Integer[] a = edge.values().toArray(new Integer[edge.size()]);
        Arrays.sort(a);
        for (int i = 0; i < a.length - 1; i++) {
            if (a[i] == a[i + 1]) {
                System.out.println("NO");
                return;
            }
        }
        List<Integer[]> ans = new ArrayList<>();
        int[][] par = new int[n + 1][2];
        boolean[] vis = new boolean[n + 1];
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        while (!q.isEmpty()) {
            int v = q.poll();
            for (int u : g.get(v)) {
                if (!vis[u]) {
                    vis[u] = true;
                    int w = edge.get(u*n + v);
                    if (w % 2 != 0) {
                        System.out.println("NO");
                        return;
                    }
                    par[u][0] = v;
                    par[u][1] = w / 2;
                    q.add(u);
                }
            }
        }

        int cnt = 0;
        for (int i = 2; i <= n; i++) {
            if (par[i][1] != 0) {
                cnt++;
                ans.add(new Integer[]{i, par[i][0], -par[i][1]});
            }
        }
        ans.add(new Integer[]{1, 3, -a[cnt] + a[cnt - 1]});
        System.out.println("YES");
        System.out.println(ans.size());
        for (Integer[] x : ans) {
            System.out.println(x[0] + " " + x[1] + " " + x[2]);
        }
    }
}
