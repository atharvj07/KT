#include <iostream>
#include <vector>
using namespace std;

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> p(n), q(n);
  for (int i = 0; i < n; i++) {
    cin >> p[i];
    p[i]--;
  }
  for (int i = 0; i < n; i++) { q[p[i]] = i; }
  for (int i = 0; i < n; i++) { cout << q[i] + 1 << " "; }
  cout << "\n";
  return 0;
}
