import java.util.*;
import java.math.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), h = sc.nextInt();
        int[] u = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            u[i] = sc.nextInt();
        }
        ArrayList<Integer>[] cli = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            cli[i] = new ArrayList<>();
        }
        for (int i = 1; i <= m; i++) {
            int c1 = sc.nextInt(), c2 = sc.nextInt();
            cli[c1].add(i);
            cli[c2].add(i);
        }
        ArrayList<Integer> dc = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (cli[i].size() > 1) {
                dc.add(i);
            }
        }
        int sz = dc.size();
        System.out.println(sz);
        for (int i = 0; i < sz; i++) {
            System.out.print(dc.get(i) + " ");
        }
        System.out.println();
    }
}
