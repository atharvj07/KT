t = int(input())
for _ in range(t):
    n, s = map(int, input().split())
    a = list(map(int, input().split()))
    total_time, max_time, skip_index = 0, 0, 0
    for i in range(n):
        total_time += a[i]
        if a[i] > max_time:
            max_time = a[i]
            skip_index = i + 1
        if total_time > s:
            print(skip_index)
            break
    else:
        print(0)
