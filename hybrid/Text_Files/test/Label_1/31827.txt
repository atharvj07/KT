import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] p;
    static int[] sz;

    static void init() {
        p = new int[n];
        sz = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = i;
            sz[i] = 1;
        }
    }

    static int find(int x) {
        if (x != p[x]) {
            p[x] = find(p[x]);
        }
        return p[x];
    }

    static void merge(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) return;
        if (sz[x] < sz[y]) {
            p[x] = y;
            sz[y] += sz[x];
        } else {
            p[y] = x;
            sz[x] += sz[y];
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        init();
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            merge(a, b);
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (p[i] == i) cnt++;
        }
        if (cnt == 1) System.out.println(0);
        else {
            System.out.println(cnt - 1);
            for (int i = 0; i < n; i++) {
                if (p[i] == i) {
                    for (int j = i + 1; j < n; j++) {
                        if (find(j) != find(i)) {
                            System.out.println((i + 1) + " " + (j + 1) + " " + (j + 1) + " " + (i + 1));
                            merge(i, j);
                            break;
                        }
                    }
                }
            }
        }
    }
}
