import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt(), d = scan.nextInt(), x = scan.nextInt();
        int[][] prices = new int[d][n];
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < n; j++) {
                prices[i][j] = scan.nextInt();
            }
        }

        int[][] dp = new int[d][2];
        dp[0][0] = x;
        dp[0][1] = -Integer.MAX_VALUE;
        for (int i = 1; i < d; i++) {
            for (int j = 0; j < n; j++) {
                dp[i][0] = Math.max(dp[i][0], dp[i - 1][1] + prices[i][j]);
                dp[i][1] = Math.max(dp[i][1], dp[i - 1][0] - prices[i][j]);
            }
        }
        System.out.println(dp[d - 1][0]);
    }
}
