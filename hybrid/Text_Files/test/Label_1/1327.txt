#include <bits/stdc++.h>

using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  string s;
  cin >> s;
  int n = s.size();
  int open = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '(')
      open++;
    else if (s[i] == ')')
      open--;
    else if (open > 0) {
      open--;
      s[i] = ')';
    } else {
      cout << "-1";
      return 0;
    }
  }
  if (open == 0) {
    for (int i = 0; i < n; i++) {
      if (s[i] == '#')
        s[i] = ')';
    }
    cout << s;
  } else
    cout << "-1";
  return 0;
}
