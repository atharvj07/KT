#include <bits/stdc++.h>
#define MOD 1000000007
using namespace std;
typedef long long LL;

const int MX = 36;
LL f[MX][MX][MX];

void precalc() {
    memset(f, 0, sizeof f);
    f[0][0][0] = 1;
    for (int i = 0; i < MX - 1; i++) {
        for (int j = 0; j < MX; j++) {
            for (int k = 0; k < MX; k++) {
                if (f[i][j][k] == 0) continue;

                // equal
                f[i + 1][j][k] += f[i][j][k];
                f[i + 1][j][k] %= MOD;

                // up
                if (j + i < MX - 1) {
                    f[i + 1][j + i + 1][k] += f[i][j][k];
                    f[i + 1][j + i + 1][k] %= MOD;
                }
                // down
                if (k - i >= 0) {
                    f[i + 1][j][k - i] += f[i][j][k];
                    f[i + 1][j][k - i] %= MOD;
                }
                // up + equal
                if (j + i < MX - 1) {
                    f[i + 1][j + i + 1][k] += f[i][j][k];
                    f[i + 1][j + i + 1][k] %= MOD;
                }
                // down + equal
                if (k - i >= 0) {
                    f[i + 1][j][k - i] += f[i][j][k];
                    f[i + 1][j][k - i] %= MOD;
                }
            }
        }
    }
}

int32_t main() {
    int n, k;
    cin >> n >> k;

    precalc();

    LL ans = f[2 * n][0][0];
    while (k--) {
        int a, b;
        char c;
        cin >> a >> c >> b;
        if (c == '=') ans = (ans - f[2 * n][a - 1][b - 1] + MOD) % MOD;
        else if (c == '<') ans = (ans - f[2 * n][b - 1][a - 1] - f[2 * n][a - 1][b - 1] + MOD) % MOD;
        else if (c == '>') ans = (ans - f[2 * n][a - 1][b - 1] + MOD) % MOD;
        else if (c == '<=') ans = (ans - f[2 * n][a - 1][b - 1] - f[2 * n][b - 1][a - 1] + MOD + MOD) % MOD;
        else if (c == '>=') ans = (ans - f[2 * n][a - 1][b - 1] + MOD + MOD) % MOD;
    }

    cout << ans << endl;
    return 0;
}
