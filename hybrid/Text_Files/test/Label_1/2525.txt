#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  string s; cin >> s;
  int n = s.size(), ans = 0, cnt[n] = {};
  for (int i = n - 2; i >= 0; i--) {
    cnt[i] = (s[i] == s[i + 1] ? cnt[i + 1] : 0) + 1;
    ans = max(ans, cnt[i]);
  }
  cout << ans << endl;
}
