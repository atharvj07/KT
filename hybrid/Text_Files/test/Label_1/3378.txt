#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m, q;
    cin >> n >> m;
    vector<vector<int>> g(n);
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    cin >> q;
    vector<int> ans(q + 1);
    vector<int> deg(n);
    vector<int> sal(n);
    for (int i = 0; i < n; i++) sal[i] = i;
    for (int i = 1; i <= q; i++) {
        int v;
        cin >> v;
        v--;
        ans[i] = ans[i - 1];
        for (int u : g[v]) {
            ans[i] -= (sal[u] < sal[v] && deg[u] > 0);
        }
        for (int u : g[v]) {
            if (sal[u] < sal[v]) deg[u]--;
        }
        deg[v] = g[v].size();
        sal[v] = n + i - 1;
        for (int u : g[v]) {
            if (sal[u] < sal[v]) deg[u]++;
        }
    }
    for (int i = 0; i <= q; i++) {
        cout << ans[i] << endl;
    }
    return 0;
}
