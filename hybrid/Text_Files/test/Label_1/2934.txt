#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    while (k--) {
        int mn = 1e9;
        bool all0 = true;
        for (int x : a) {
            if (x) mn = min(mn, x), all0 = false;
        }
        if (all0) cout << "0\n";
        else {
            cout << mn << "\n";
            for (int i = 0; i < n; i++) {
                if (a[i]) a[i] -= mn;
            }
        }
    }
}
