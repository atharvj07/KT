n, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

# Calculate prefix sums for the difference between b and a.
diff = [b[i] - a[i] for i in range(n)]
prefix_sum = [0] * (n + 1)
for i in range(1, n + 1):
    prefix_sum[i] = prefix_sum[i - 1] + diff[i - 1]

for _ in range(q):
    l, r = map(int, input().split())
    l -= 1
    # Check if the total sum of the segment is even.
    if (prefix_sum[r] - prefix_sum[l]) % 2 != 0:
        print(-1)
        continue
    
    # Check if the total sum of the segment is zero.
    if prefix_sum[r] != prefix_sum[l]:
        print(-1)
        continue
    
    # Calculate the minimal number of operations.
    odd = 0
    even = 0
    for i in range(l, r):
        if diff[i] % 2 == 0:
            even += diff[i]
        else:
            odd += abs(diff[i])
    
    # The minimal number of operations is the max between the absolute sum of odd differences and even differences divided by 2.
    print(max(abs(odd), abs(even)) // 2)
