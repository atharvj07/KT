#include<bits/stdc++.h>
using namespace std;
int main() {
    string s;
    cin >> s;
    int k;
    cin >> k;
    int n = s.size();
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (s[i] == s[j]) {
                ans = max(ans, j - i + 1);
            }
        }
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + ans; j < n + k; j++) {
            if (s[i] == s[j]) {
                ans = max(ans, j - i + 1);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
