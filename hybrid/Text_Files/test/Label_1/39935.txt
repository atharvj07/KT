import sys

input = sys.stdin.readline

n, m, k = map(int, input().split())
G = [[] for _ in range(n+1)]
C = set(map(int, input().split()))

for _ in range(m):
    x, y = map(int, input().split())
    G[x].append(y)
    G[y].append(x)

def dfs(x: int, f: int, d: int) -> int:
    visited[x] = True
    dp[x][0] = 1
    dp[x][1] = 0
    for y in G[x]:
        if visited[y]: continue
        if y == f: continue
        if y in C:
            dp[x][0] += dfs(y, f, not d)
            dp[x][1] += dfs(y, f, d)
        else:
            dp[x][0] += dfs(y, f, d)
            dp[x][1] += max(dfs(y, f, d), dfs(y, f, not d))
    return dp[x][d]

visited = [False] * (n+1)
dp = [[0, 0] for _ in range(n+1)]
dfs(1, 1, 0)
print(dp[1][1])
