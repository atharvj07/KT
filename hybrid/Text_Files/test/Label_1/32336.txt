import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            p[i] = sc.nextInt();
        }
        ArrayList<int[]> pairs = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            pairs.add(new int[]{sc.nextInt(), sc.nextInt()});
        }
        for (int[] pair : pairs) {
            int temp = p[pair[0]];
            p[pair[0]] = p[pair[1]];
            p[pair[1]] = temp;
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(p[i] + " ");
        }
    }
}
