import java.util.Scanner;

public class Nim {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int x = scanner.nextInt();
        double[] p = new double[x + 1];
        for (int i = 0; i <= x; i++) {
            p[i] = scanner.nextDouble();
        }

        double firstPlayerWinProbability = 0;
        for (int i = 1; i <= x; i++) {
            firstPlayerWinProbability += p[i] * Math.pow(1 - p[0], n - 1);
        }

        System.out.println(firstPlayerWinProbability);
    }
}
