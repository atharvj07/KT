#include <bits/stdc++.h>
using namespace std;
int main() {
    int t, n, m, k;
    cin >> t;
    while (t--) {
        cin >> n >> m >> k;
        int x = n / m;
        int y = n - x * m;
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < y; j++) {
                cout << x + 1 << " ";
                for (int idx = j * (x + 1); idx < (j + 1) * (x + 1); idx++)
                    cout << idx + 1 << " ";
                cout << endl;
            }
            for (int j = 0; j < m - y; j++) {
                cout << x << " ";
                for (int idx = y * (x + 1) + j * x;
                     idx < y * (x + 1) + (j + 1) * x; idx++)
                    cout << idx + 1 << " ";
                cout << endl;
            }
        }
        cout << endl;
    }
}
