import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    try (Scanner in = new Scanner(System.in)) {
      int h = in.nextInt();
      int m = in.nextInt();
      int n = in.nextInt();

      int[] table = new int[h];
      int result = 0;

      while (n-- > 0) {
        String op = in.next();
        int id = in.nextInt();

        if (op.equals("+")) {
          int hash = in.nextInt();
          int j = 0;
          while (table[(hash + j * m) % h] != 0) {
            j++;
          }
          table[(hash + j * m) % h] = id;
          result += j;
        } else {
          int j = 0;
          while (table[(id + j * m) % h] != id) {
            j++;
          }
          table[(id + j * m) % h] = 0;
        }
      }

      System.out.println(result);
    }
  }
}
