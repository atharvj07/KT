n, q = map(int, input().split())
s = [0] * q
st = []
for i in range(q):
    t, x = map(int, input().split())
    s[i] = (t, x)
    if t == 2:
        st.append(x)
st.sort()
cur = n
for t, x in s:
    if t == 1:
        cur -= x
    else:
        l, r = st.index(x), st.index(x) + 1
        print(cur - st[l] - n + st[r])
