#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pb push_back
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  int mx = a[n - 1] - a[0];
  if (mx == 0) {
    cout << 0 << " " << 0;
    return 0;
  }
  int oper = 0;
  for (int i = 1; i < n; i++) {
    int add = min(k, mx - (a[n - i - 1] - a[0]));
    oper += add;
    k -= add;
    if (k == 0) break;
  }
  cout << mx - oper << " " << oper << endl;
  for (int i = n - 1; i >= 1 && oper; i--) {
    int add = min(oper, mx - (a[n - i] - a[0]));
    cout << n << " " << n - i << endl;
    oper -= add;
    a[n - i] += add;
  }
  return 0;
}
