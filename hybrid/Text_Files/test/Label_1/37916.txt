s = input()
L = [0] * 4
R = [0] * 4
for c in s:
    if   c=='<': L[0]+=1
    elif c=='{': L[1]+=1
    elif c=='[': L[2]+=1
    elif c=='(': L[3]+=1
    elif c=='>': R[0]+=1
    elif c=='}': R[1]+=1
    elif c==']': R[2]+=1
    elif c==')': R[3]+=1
    else: print('Impossible');exit(0)
ans = max(0,max(R)-min(L))
for i in range(4): ans+=abs(L[i]-R[i])//2
print(ans)
