#include<bits/stdc++.h>
using namespace std;
int n,m,x,y,cx,cy,vis[502][502];
char a[502][502];
bool valid(int x,int y){return x>=1 && x<=n && y>=1 && y<=m;}
bool bfs(){
    memset(vis,0,sizeof(vis));
    queue<pair<int,int>>q;
    q.push({cx,cy}); vis[cx][cy]=1;
    while(!q.empty()){
        x=q.front().first; y=q.front().second; q.pop();
        if(x==cx && y==cy) break;
        for(int dx=-1;dx<=1;dx++){
            for(int dy=-1;dy<=1;dy++){
                if(dx==0 && dy==0) continue;
                int nx=x+dx, ny=y+dy;
                if(valid(nx,ny) && vis[nx][ny]==0){
                    if(a[nx][ny]=='.'){
                        a[nx][ny]='X'; q.push({nx,ny});
                        vis[nx][ny]=1;
                    }else{
                        cx=nx; cy=ny; q.push({nx,ny});
                        vis[nx][ny]=1;
                    }
                }
            }
        }
    }
    return vis[cx][cy]==1;
}
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++){
            cin>>a[i][j];
            if(a[i][j]=='X'){
                cx=i; cy=j;
            }
        }
    cin>>x>>y;
    cx=x; cy=y;
    cout<<( bfs()? "YES" : "NO" )<<endl;
    return 0;
}
