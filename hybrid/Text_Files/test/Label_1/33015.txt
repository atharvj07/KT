import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    private static final long MOD = 1000000007;
    private static long[][] dp;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        long[][] polygon = new long[n][2];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            polygon[i][0] = Long.parseLong(line[0]);
            polygon[i][1] = Long.parseLong(line[1]);
        }
        long sum = 0;
        dp = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 3; j < n; j++) {
                sum = (sum + solve(i, j, polygon)) % MOD;
            }
        }
        System.out.println(sum);
    }

    private static long solve(int i, int j, long[][] polygon) {
        if (j - i < 3) {
            return 0;
        }
        if (dp[i][j] != 0) {
            return dp[i][j];
        }
        long area = getArea(polygon, i, j);
        long maxArea = 0;
        for (int k = i + 1; k < j; k++) {
            maxArea = Math.max(maxArea, getArea(polygon, i, k) + getArea(polygon, k, j));
        }
        dp[i][j] = (area - maxArea) * 2;
        return dp[i][j];
    }

    private static long getArea(long[][] polygon, int i, int j) {
        long x1 = polygon[i][0];
        long y1 = polygon[i][1];
        long x2 = polygon[j][0];
        long y2 = polygon[j][1];
        return Math.abs(x1 * y2 - x2 * y1);
    }
}
