import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt();
        int n = in.nextInt();
        if (m % (n - 1) == 0) {
            System.out.println(m / (n - 1));
        } else {
            System.out.println(-1);
        }
    }
}
