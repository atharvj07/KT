import java.util.*;
import java.io.*;

public class Main {
    private static int n, m;
    private static int[][] adj;
    private static int[] vis;

    private static boolean dfs(int u, int cnt1, int cnt2, int p) {
        if (cnt2 > 2) return false;
        if (cnt1 == m - 2 && cnt2 == 2) return true;
        vis[u] = 1;
        for (int v : adj[u]) {
            if (v == p) continue;
            if (vis[v] == 0 && dfs(v, cnt1 + 1, cnt2 + (adj[u][0] != adj[v][0] ? 1 : 0), u)) return true;
        }
        vis[u] = -1;
        return false;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] in = br.readLine().split(" ");
        n = Integer.parseInt(in[0]);
        m = Integer.parseInt(in[1]);
        adj = new int[n + 1][2];
        vis = new int[n + 1];
        for (int i = 1; i <= m; i++) {
            in = br.readLine().split(" ");
            int u = Integer.parseInt(in[0]);
            int v = Integer.parseInt(in[1]);
            adj[u][0] = v;
            adj[v][0] = u;
            adj[u][1]++;
            adj[v][1]++;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (vis[i] == 0 && dfs(i, 0, 0, 0)) ans++;
        }
        System.out.println(ans);
    }
}
