def max_candies(n, x, candies):
    candies.sort(key=lambda c: (-c[2], c[1]))
    res = 0
    for first_type in range(2):
        jump_height = x
        eaten = 0
        type_mask = [False] * n
        current_type = first_type
        while True:
            found = False
            for i in range(n):
                if candies[i][0] == current_type and not type_mask[i] and candies[i][1] <= jump_height:
                    jump_height += candies[i][2]
                    type_mask[i] = True
                    found = True
                    eaten += 1
                    break
            if not found:
                break
            current_type = 1 - current_type
        res = max(res, eaten)
    return res

n, x = map(int, input().split())
candies = [list(map(int, input().split())) for _ in range(n)]
print(max_candies(n, x, candies))
