n=int(input())
for i in range(n):
    p,q,b=map(int,input().split())
    temp=p/q
    if b!=10:
        temp=str(temp)
        if '.' not in temp:
            print("Finite")
        else:
            temp=temp[temp.index('.')+1:]
            print("Finite" if len(set(temp))==1 else "Infinite")
    else:
        print("Infinite" if q%2==0 and p%2==1 else "Finite")
