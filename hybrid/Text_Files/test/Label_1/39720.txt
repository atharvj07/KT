def is_perfect_square(n):
    return int(n**0.5)**2 == n

def solve(n, even_numbers):
    result = [0] * n
    prev_square = 0
    for i in range(1, n, 2):
        x = even_numbers[i // 2]
        target = prev_square + x
        left, right = 0, target
        found = False
        while left <= right:
            mid = (left + right) // 2
            if mid**2 >= target and is_perfect_square(mid**2 - target):
                result[i] = x
                result[i-1] = mid**2 - target - prev_square
                prev_square = mid**2
                found = True
                break
            elif mid**2 < target:
                left = mid + 1
            else:
                right = mid - 1
        if not found:
            return "No"
    return "Yes\n" + " ".join(map(str, result))

n = int(input().strip())
even_numbers = list(map(int, input().split()))
print(solve(n, even_numbers))
