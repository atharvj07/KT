#include<bits/stdc++.h>
using namespace std;
int n, m, ans;
unordered_map<int, int> pos;
vector<int> adj[300005];
bool vis[300005];
int dfs(int u) {
    vis[u] = true;
    if (u == n) {
        ans++;
        return 1;
    }
    int l = 0;
    for (auto v : adj[u]) {
        if (!vis[v]) {
            l += dfs(v);
        }
    }
    return l;
}
int main() {
    cin >> n >> m;
    for (int i = 1, u; i <= n; i++) {
        cin >> u;
        pos[u] = i;
    }
    for (int i = 0, a, b; i < m; i++) {
        cin >> a >> b;
        if (pos[a] + 1 == pos[b]) {
            adj[pos[a]].push_back(pos[b]);
        }
        if (pos[b] + 1 == pos[a]) {
            adj[pos[b]].push_back(pos[a]);
        }
    }
    dfs(1);
    cout << ans;
    return 0;
}
