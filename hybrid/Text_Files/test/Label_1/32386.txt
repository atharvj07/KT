import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Time[] times = new Time[n];
        for (int i = 0; i < n; i++) {
            int h = sc.nextInt();
            int m = sc.nextInt();
            times[i] = new Time(h, m);
        }
        Arrays.sort(times);
        int cnt = 0, mx = 0;
        for (Time t : times) {
            if (cnt < t.h) cnt = t.h;
            cnt++;
            mx = Math.max(mx, cnt);
            if (cnt > 0 && t.m > 0) cnt--;
        }
        System.out.println(mx);
    }

    static class Time implements Comparable<Time> {
        int h, m;

        public Time(int h, int m) {
            this.h = h;
            this.m = m;
        }

        @Override
        public int compareTo(Time other) {
            if (this.h == other.h) return this.m - other.m;
            return this.h - other.h;
        }
    }
}
