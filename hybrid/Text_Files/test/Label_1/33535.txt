import java.util.*;
import java.io.*;

class Main {
    private static class Edge {
        int from;
        int to;
        Edge(int a, int b) { from = a; to = b; }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < n - 1; ++i) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            Edge e = new Edge(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
            edges.add(e);
        }
        Kruskal kruskal = new Kruskal(n, edges);
        System.out.println(kruskal.spanningTree().size() - n + 1);
        for (Edge e : kruskal.spanningTree()) {
            System.out.println(e.from + " " + e.to);
        }
    }

    private static class Kruskal {
        private int n;
        private int[] parent;
        private int[] rank;
        private List<Edge> edges;

        Kruskal(int n, List<Edge> edges) {
            this.n = n;
            this.edges = edges;
            this.parent = new int[n + 1];
            this.rank = new int[n + 1];
            for (int i = 0; i <= n; ++i) {
                parent[i] = i;
                rank[i] = 0;
            }
            Collections.sort(this.edges, new Comparator<Edge>() {
                @Override
                public int compare(Edge a, Edge b) {
                    return a.from - b.from;
                }
            });
        }

        public List<Edge> spanningTree() {
            List<Edge> tree = new ArrayList<>();
            for (Edge e : edges) {
                if (find(e.from) == find(e.to)) continue;
                union(e.from, e.to);
                tree.add(e);
            }
            return tree;
        }

        private int find(int x) {
            if (parent[x] == x) return x;
            else return parent[x] = find(parent[x]);
        }

        private void union(int x, int y) {
            x = find(x);
            y = find(y);
            if (rank[x] > rank[y]) parent[y] = x;
            else if (rank[x] < rank[y]) parent[x] = y;
            else {
                parent[y] = x;
                rank[x]++;
            }
        }
    }
}
