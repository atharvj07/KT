#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, t1 = 0, t2 = 0;
  cin >> n;
  vector<int> vec(n);
  for (int i = 0; i < n; i++) {
    cin >> vec[i];
    t1 += (vec[i] == 100);
    t2 += (vec[i] == 200);
  }
  if (t1 == t2 || (t1 & 1 && t2 & 1 && t1 - 1 == t2))
    cout << "YES";
  else
    cout << "NO";
  return 0;
}
