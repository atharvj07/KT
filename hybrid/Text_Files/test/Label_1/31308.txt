import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        try {
            BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
            int t = Integer.parseInt(in.readLine());
            for (int i = 0; i < t; i++) {
                int n = Integer.parseInt(in.readLine());
                int[] a = new int[2 * n];
                StringTokenizer st = new StringTokenizer(in.readLine());
                for (int j = 0; j < 2 * n; j++) {
                    a[j] = Integer.parseInt(st.nextToken());
                }
                Arrays.sort(a);
                int[] b = new int[n - 1];
                int[] indices = new int[2 * n - 2];
                int idx = 0;
                for (int j = 0; j < n - 1; j++) {
                    b[j] = a[2 * j] + a[2 * j + 1];
                    indices[idx++] = 2 * j + 1;
                    indices[idx++] = 2 * j + 2;
                }
                for (int j = 0; j < n - 1; j++) {
                    System.out.println(indices[2 * j] + " " + indices[2 * j + 1]);
                }
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
