from functools import reduce
mod = 1000000007
def mul_inv(a, b):
    b0 = b
    x0, x1 = 0, 1
    if b == 1: return 1
    while a > 1:
        q = a // b
        a, b = b, a % b
        x0, x1 = x1, x0 - q * x1
    if x1 < 0: x1 += b0
    return x1

n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
ans_a, ans_b = 0, 1
for i in range(n - 1, -1, -1):
    cnt_a, cnt_b = 0, 0
    for j in range(i + 1, n):
        if a[j] != 0: cnt_a += (a[j] > b[j])
        if b[j] != 0: cnt_b += (b[j] > a[j])
    if a[i] == 0: ans_a += cnt_a * ans_b % mod
    if b[i] == 0: ans_b += cnt_b * ans_a % mod
print(ans_a * mul_inv(ans_b, mod) % mod)
