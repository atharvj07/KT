import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long r = sc.nextInt();
        long g = sc.nextInt();
        long b = sc.nextInt();
        long n = sc.nextInt();
        long ans = 0;
        for (long i = 0; i <= n / r; i++) {
            for (long j = 0; j <= (n - i * r) / g; j++) {
                long k = (n - i * r - j * g) / b;
                if (i * r + j * g + k * b == n) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
