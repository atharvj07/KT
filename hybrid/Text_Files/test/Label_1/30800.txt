import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int r = input.nextInt();
        double R = r * Math.sin(Math.PI / n) / (1 - Math.sin(Math.PI / n));
        System.out.printf("%.9f", R);
    }
}
