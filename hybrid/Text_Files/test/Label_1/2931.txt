#include <bits/stdc++.h>
using namespace std;
map<pair<char, char>, int> mp;
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    string s, t;
    cin >> s >> t;
    if (s.size() != t.size()) {
      cout << "NO\n";
      continue;
    }
    bool ok = true;
    for (int j = 0; j < s.size(); j++) {
      if (s[j] < t[j]) {
        ok = false;
      }
      mp[{s[j], t[j]}]++;
      if (mp[{s[j], t[j]}] > 1) {
        ok = false;
      }
    }
    if (ok) {
      cout << "YES\n";
    } else {
      cout << "NO\n";
    }
    mp.clear();
  }
}
