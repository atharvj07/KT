import java.util.*;
import java.io.*;

public class Main {
    private static long[] dp, myF;
    private static long[] answers, powers;
    private static int MODULUS = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());

        int length = Integer.parseInt(tokenizer.nextToken());

        tokenizer = new StringTokenizer(reader.readLine());
        int[] c = new int[length];
        for (int i = 0; i < length; i++) {
            c[i] = Integer.parseInt(tokenizer.nextToken());
        }

        tokenizer = new StringTokenizer(reader.readLine());
        int[] b = new int[length - 1];
        for (int i = 0; i < length - 1; i++) {
            b[i] = Integer.parseInt(tokenizer.nextToken());
        }

        tokenizer = new StringTokenizer(reader.readLine());
        int numOfQueries = Integer.parseInt(tokenizer.nextToken());

        tokenizer = new StringTokenizer(reader.readLine());
        int[] queries = new int[numOfQueries];
        for (int i = 0; i < numOfQueries; i++) {
            queries[i] = Integer.parseInt(tokenizer.nextToken());
        }

        dp = new long[length];
        Arrays.fill(dp, -1);
        myF = new long[length];
        answers = new long[numOfQueries];

        powers = new long[100001];
        powers[0] = 1;
        for (int i = 1; i < 100001; i++) {
            powers[i] = (powers[i - 1] * 2) % MODULUS;
        }

        for (int i = 0; i < numOfQueries; i++) {
            answers[i] = calculateAnswers(0, queries[i], c, b, length);
        }

        StringBuilder builder = new StringBuilder();
        for (long answer : answers) {
            builder.append(answer).append(' ');
        }
        System.out.println(builder.toString());
    }

    private static long calculateAnswers(int startIndex, long x, int[] c, int[] b, int length) {
        if (startIndex == length - 1) {
            if (myF[startIndex] >= x) {
                return 1;
            }
            return 0;
        }

        if (dp[startIndex] != -1) {
            return dp[startIndex];
        }

        long currentAnswer = 0;

        long lowerBound = (long) Math.floor((double) (myF[startIndex] - x) / 2);
        long upperBound = (long) Math.floor((double) (c[startIndex] - myF[startIndex]) / 2);

        for (long i = lowerBound; i <= upperBound; i++) {
            myF[startIndex + 1] = myF[startIndex] + 2 * i - b[startIndex];
            currentAnswer = (currentAnswer + calculateAnswers(startIndex + 1, x, c, b, length)) % MODULUS;
            myF[startIndex + 1] = myF[startIndex] - 2 * i + b[startIndex];
        }

        dp[startIndex] = currentAnswer % MODULUS;
        return dp[startIndex];
    }
}
