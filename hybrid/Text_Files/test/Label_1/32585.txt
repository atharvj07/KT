import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        in.nextLine();
        boolean isBlackWhite = true;
        for (int i = 0; i < n; i++) {
            String line = in.nextLine();
            for (int j = 0; j < m; j++) {
                char color = line.charAt(j);
                if (color == 'C' || color == 'M' || color == 'Y') {
                    isBlackWhite = false;
                    break;
                }
            }
            if (!isBlackWhite) {
                break;
            }
        }
        System.out.println(isBlackWhite ? "#Black&White" : "#Color");
    }
}
