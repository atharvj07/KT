import java.util.*;
import java.io.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt(), cnt = 0;
        Set<Integer> s = new HashSet<>();
        for (int i = 0; i < n; i++) {
            int t = in.nextInt();
            s.clear();
            while (t > 0) {
                if (t % 10 > k) {
                    t = -1;
                    break;
                }
                s.add(t % 10);
                t /= 10;
            }
            if (t > 0 && s.size() == k + 1) cnt++;
        }
        System.out.println(cnt);
    }
}
