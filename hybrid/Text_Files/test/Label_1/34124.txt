import java.util.Scanner;
import java.util.List;
import java.util.ArrayList;
import java.util.Collections;

public class Main {
        public static void main(String[] args) {
                Scanner sc = new Scanner(System.in);

                int n = sc.nextInt();
                List<Rectangle> rectangles = new ArrayList<>();

                for (int i = 0; i < n; i++) {
                        int x1 = sc.nextInt();
                        int y1 = sc.nextInt();
                        int x2 = sc.nextInt();
                        int y2 = sc.nextInt();
                        rectangles.add(new Rectangle(x1, y1, x2, y2));
                }

                Collections.sort(rectangles, (a, b) -> a.x1 - b.x1);

                boolean isSquare = true;
                for (int i = 0; i < n - 1; i++) {
                        Rectangle r1 = rectangles.get(i);
                        Rectangle r2 = rectangles.get(i + 1);

                        if (r1.y2 != r2.y1 || r1.x2 != r2.x1) {
                                isSquare = false;
                        }
                }

                System.out.println(isSquare ? "YES" : "NO");
        }

        static class Rectangle {
                int x1, y1, x2, y2;

                Rectangle(int x1, int y1, int x2, int y2) {
                        this.x1 = x1;
                        this.y1 = y1;
                        this.x2 = x2;
                        this.y2 = y2;
                }
        }
}
