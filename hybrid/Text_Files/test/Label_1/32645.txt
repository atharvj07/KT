import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            w[i] = in.nextInt();
        }
        int[] dp = new int[n];
        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if ((a[j] == 0 && a[i] == 1) || (a[j] == 1 && a[i] == 0)) {
                    dp[i] = Math.min(dp[i], dp[j] + w[i]);
                } else {
                    dp[i] = Math.min(dp[i], dp[j]);
                }
            }
        }
        System.out.println(dp[n - 1]);
    }
}
