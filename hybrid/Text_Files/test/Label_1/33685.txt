import java.util.*;

class Main {
    static int n, k;
    static int[][] d, roads;
    static boolean[] used;
    static int[] par;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        d = new int[n][n];
        roads = new int[n][n];
        used = new boolean[n];
        par = new int[n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                d[i][j] = in.nextInt();
                d[j][i] = d[i][j];
            }
        }
        k = in.nextInt();
        for (int i = 0; i < k; i++) {
            int a = in.nextInt() - 1;
            int b = in.nextInt() - 1;
            int c = in.nextInt();
            roads[a][b] = c;
            roads[b][a] = c;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += d[0][i];
        }
        int[] answ = new int[k];
        answ[0] = (int) ans;
        for (int i = 1; i < k; i++) {
            dfs(0);
            int v = -1;
            for (int j = 0; j < n; j++) {
                if (!used[j] && (v == -1 || par[j] < par[v])) {
                    v = j;
                }
            }
            for (int j = 0; j < n; j++) {
                if (!used[j]) {
                    d[v][j] = d[j][v] = Math.min(d[v][j], d[j][par[j]] + roads[par[j]][v]);
                    d[j][par[j]] = d[par[j]][j] = Math.min(d[j][par[j]], d[j][v] + roads[par[j]][v]);
                }
            }
            ans += d[0][v];
            answ[i] = (int) ans;
        }
        for (int i = 0; i < k; i++) {
            System.out.print(answ[i] + " ");
        }
        System.out.println();
    }

    static void dfs(int v) {
        used[v] = true;
        for (int i = 0; i < n; i++) {
            if (!used[i] && d[v][i] != 0 && (par[i] == -1 || d[v][i] + d[v][par[i]] < d[par[i]][i])) {
                par[i] = v;
                dfs(i);
            }
        }
    }
}
