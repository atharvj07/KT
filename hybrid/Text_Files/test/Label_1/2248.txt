#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int min_ops = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] != a[0]) {
            min_ops += min(a[i], 2 * a[0] - a[i]);
            a[i] = a[0];
        }
    }
    cout << min_ops << endl;
    return 0;
}
