import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] ans = new int[m + 1];

        int id = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= 2; j++) {
                if (id == m) {
                    break;
                }
                ans[++id] = (i - 1) * 2 + j;
            }
        }

        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= 2; j++) {
                if (id == m) {
                    break;
                }
                ans[++id] = (i - 1) * 2 + j;
            }
        }

        for (int i = 1; i <= m; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
