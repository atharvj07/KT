import java.util.*;

public class Main {
    static final int INF = (int) 1e9;
    static final int MAXN = 100;

    static int n, k;
    static int[][] graph = new int[MAXN][MAXN];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                graph[i][j] = sc.nextInt();
            }
        }

        int[][] dp = new int[1 << n][k + 1];
        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j <= k; j++) {
                dp[i][j] = INF;
            }
        }

        dp[1][0] = 0;

        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j < k; j++) {
                if (dp[i][j] == INF) continue;

                for (int v = 1; v <= n; v++) {
                    if ((i & (1 << (v - 1))) == 0) {
                        int bitmask = i | (1 << (v - 1));
                        dp[bitmask][j + 1] = Math.min(dp[bitmask][j + 1], dp[i][j] + graph[j % n + 1][v]);
                    }
                }
            }
        }

        int ans = INF;
        for (int i = 0; i < (1 << n); i++) {
            if ((i & (1 << (n - 1))) != 0) {
                ans = Math.min(ans, dp[i][k]);
            }
        }

        if (ans == INF) ans = -1;

        System.out.println(ans);
    }
}
