import sys
input = sys.stdin.readline

def main():
    n = int(input())
    nums = list(map(int, input().split()))
    nums.sort()
    nums.append(2 * nums[n - 1] - nums[n - 2])
    nums = list(set(nums))
    nums.sort()
    if len(nums) == n + 1:
        print(0)
    elif len(nums) == n:
        print(1)
        print(nums[-1] - 1)
    else:
        print(-1)

main()
