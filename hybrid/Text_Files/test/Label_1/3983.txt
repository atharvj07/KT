#include <bits/stdc++.h>
using namespace std;

int n, k, vis[100005];

int main() {
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    vis[i] = 0;
  }
  int p = 1;
  for (int i = 1; i <= n; i++) {
    if (vis[i] == 0 && __gcd(i, n - k + 1) > 1) {
      cout << i << " ";
      vis[i] = 1;
      k--;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (vis[i] == 0) {
      cout << i << " ";
    }
  }
}
