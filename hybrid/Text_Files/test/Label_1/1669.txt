#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n, k;
    cin >> n >> k;
    vector< vector<int> >v(k);
    set<int>st;
    int a[n];
    for(int i = 0; i < n; i++)
    {
        cin >> a[i];
        st.insert(a[i]);
    }
    if(st.size() > k || k > n)
    {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    for(int i = 0; i < n; i++)
    {
        int x = (i % k) + 1;
        v[x - 1].push_back(a[i]);
    }
    for(int i = 0; i < k; i++)
    {
        for(int j = 0; j < v[i].size(); j++)
        {
            cout << v[i][j] << " ";
        }
    }
    cout << endl;
    return 0;
}
