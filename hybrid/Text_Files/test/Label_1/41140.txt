import sys

def cross(p, q):
    a, b = p
    c, d = q
    return a * d - b * c

def vec(A, B):
    return [B[0] - A[0], B[1] - A[1]]

def dist(A, B):
    a, b = A
    c, d = B
    return (a - c) * (a - c) + (b - d) * (b - d)

def check(A, B, C, D, target):
    return abs(cross(vec(A, B), vec(A, C))) == 0.5 * abs(cross(vec(A, B), vec(A, D))) * (dist(C, D) + target) / dist(C, D)

def solve(ant, dis):
    n = len(ant)
    ans = []
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                for sign in [-1, 1]:
                    if check(ant[i], ant[j], ant[k], [0, sign * dis[0]], dis[1] - dis[0]):
                        x = (dist(ant[j], ant[k]) * dis[0] + cross(vec(ant[i], ant[j]), vec(ant[i], ant[k])) * sign * dis[1]) / cross(vec(ant[i], ant[j]), vec(ant[i], ant[k]))
                        y = (dist(ant[j], ant[k]) - x * cross(vec(ant[j], ant[k]), vec(ant[i], ant[j])) / cross(vec(ant[i], ant[j]), vec(ant[i], ant[k]))) / sign
                        if 0 <= x <= 1e8 and 0 <= y <= 1e8:
                            ans.append([x, y])
    return ans

n = int(input())
ant = []
for _ in range(n):
    ant.append(list(map(int, input().split())))
m = int(input())
for _ in range(m):
    dis = [int(i) for i in input().split()]
    ans = solve(ant, dis)
    print(len(ans))
    for x, y in sorted(ans):
        print(x, y)
