T = int(input().strip())
for _ in range(T):
    n, x = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    a.sort(reverse=True)
    total = 0
    for i in range(n):
        total += a[i]
        if total < x * (i + 1):
            print(i)
            break
    else:
        print(n)
