MOD = 1000000007

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(h, w, r, c):
    grid = [[0 for _ in range(w)] for _ in range(h)]
    for i in range(h):
        for j in range(r[i]):
            grid[i][j] = 1
        if r[i] < w:
            grid[i][r[i]] = -1
    for j in range(w):
        for i in range(c[j]):
            if grid[i][j] == -1:
                return 0
            grid[i][j] = 1
        if c[j] < h:
            if grid[c[j]][j] == 1:
                return 0
            grid[c[j]][j] = -1
    free_cells = sum(cell == 0 for row in grid for cell in row)
    return power(2, free_cells, MOD)

h, w = map(int, input().split())
r = list(map(int, input().split()))
c = list(map(int, input().split()))

print(solve(h, w, r, c))
