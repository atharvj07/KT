n, r = map(int, input().split())
a = []
b = []
for i in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)

dp = [[-1]*(r+1) for i in range(n+1)]

def solve(i, rat):
    if i == n:
        return 0
    if dp[i][rat] != -1:
        return dp[i][rat]
    op1 = 0
    if rat >= a[i]:
        op1 = 1 + solve(i+1, rat+b[i])

    op2 = solve(i+1, rat)

    dp[i][rat] = max(op1,op2)
    return dp[i][rat]

print(solve(0, r))
