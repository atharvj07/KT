import java.util.*;

public class Main {
    public static class Pair {
        int min, max;

        public Pair(int min, int max) {
            this.min = min;
            this.max = max;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n];

        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }

        Pair[] ans = new Pair[n];

        for (int i = 0; i < n; i++) {
            int min = Integer.MAX_VALUE;
            int max = Integer.MIN_VALUE;

            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }

                int dist = Math.abs(x[i] - x[j]);
                min = Math.min(min, dist);
                max = Math.max(max, dist);
            }

            ans[i] = new Pair(min, max);
        }

        for (Pair p : ans) {
            System.out.println(p.min + " " + p.max);
        }
    }
}
