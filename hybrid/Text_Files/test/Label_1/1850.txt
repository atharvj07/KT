#include <iostream>
#include <vector>

using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  vector<string> wall(r);
  for (string& row : wall) { cin >> row; }
  int segments = 0;
  for (int i = 0; i < c; i++) {
    bool in_segment = false;
    for (int j = 0; j < r; j++) {
      if (wall[j][i] == 'B') {
        if (!in_segment) {
          segments++;
          in_segment = true;
        }
      } else {
        in_segment = false;
      }
    }
  }
  cout << segments << endl;
  return 0;
}
