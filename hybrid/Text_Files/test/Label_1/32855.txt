import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String[] pos = in.nextLine().split(" ");
        int[][] board = new int[8][8];
        for (String p : pos) {
            int x = p.charAt(0) - 'a';
            int y = p.charAt(1) - '1';
            board[x][y] = p.charAt(0) == 'a' ? 1 : -1;
        }
        boolean checkmate = true;
        for (int i = 0; i < 8; i++) {
            if (board[i][pos[3].charAt(1) - '1'] * board[pos[2].charAt(0) - 'a'][i] > 0) checkmate = false;
        }
        System.out.println(checkmate ? "CHECKMATE" : "OTHER");
    }
}
