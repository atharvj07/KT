#include<bits/stdc++.h>

using namespace std;

#define pb push_back
#define ll long long

const ll N=2e5+5;
const ll LG=log2(N);

ll n,k,a[N],ans[N][LG+1],lg2[N];

void pre(){
    lg2[0]=-1;
    for(ll i=1;i<N;++i){
        if((i&(i+1))==0)++lg2[i];
        else lg2[i]=lg2[i&(i-1)]+1;
    }
    for(ll i=0;i<N;++i)for(ll j=0;j<=LG;++j){
        if(j==0)ans[i][j]=a[i];
        else ans[i][j]=ans[ans[i][j-1]][j-1];
    }
}

ll get_kth(ll i,ll k){
    for(ll j=LG;j>=0;--j)if(((k>>j)&1))i=ans[i][j];
    return i;
}

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(0);cout.tie(0);
    cin>>n>>k;
    for(ll i=0;i<n;++i){
        cin>>a[i];
    }
    pre();
    ll now=0;
    ll cnt=0;
    ll ans=0;
    for(ll l=0;l<n;++l){
        while(now<k){
            cnt=get_kth(l,now+1);
            if(cnt==now){
                if(now+1>a[l] && (now-1)*2<a[l]){
                    now++;
                    ans+=a[l]-(now+1)*2;
                }
                else if(now+1<a[l] && (now-1)*2>=a[l]){
                    now++;
                    ans+=(now+1)*2-a[l];
                }
                else break;
            }
            else{
                now++;
                ans+=a[l]-now*2;
            }
        }
        if(now>=k)break;
    }
    cout<<ans;
    return 0;
}
