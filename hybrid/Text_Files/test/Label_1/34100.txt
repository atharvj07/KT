import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] max = new int[n];
        max[0] = 0;
        for (int i = 1; i < n; i++) {
            max[i] = Math.max(max[i - 1], a[i - 1]);
        }
        int[] max2 = new int[n];
        max2[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            max2[i] = Math.max(max2[i + 1], a[i + 1]);
        }
        for (int i = 0; i < n; i++) {
            System.out.println(Math.max(max[i], max2[i]));
        }
    }
}
