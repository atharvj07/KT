import sys

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [int(i) for i in input().split()]
    b = [(a[0]-i,i) for i in a]
    b.sort()
    ans = m-1
    cur = 3
    for x,y in b:
        cur = max(x+1,cur)
        if cur<=y:
            ans -= 1
        else:
            break
    print(max(0,ans))
