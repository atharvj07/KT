#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int& x : a) cin >> x;
    vector<int> l(n), r(n);  // Distance to nearest zero on the left/right
    l[0] = a[0] == 0 ? 0 : INT32_MAX;
    for (int i = 1; i < n; i++) {
        if (a[i] == 0) l[i] = 0;
        else l[i] = l[i - 1] + 1;
    }
    r[n - 1] = a[n - 1] == 0 ? 0 : INT32_MAX;
    for (int i = n - 2; i >= 0; i--) {
        if (a[i] == 0) r[i] = 0;
        else r[i] = r[i + 1] + 1;
    }
    for (int i = 0; i < n; i++) cout << min(l[i], r[i]) << " ";
    cout << endl;
    return 0;
}
