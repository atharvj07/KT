#include <iostream>
#include <string>

using namespace std;

bool solve() {
    int n, k;
    cin >> n >> k;
    if (2 * k + 1 > n) return false;
    string s;
    cin >> s;
    for (int i = k; i >= 0; i--) {
        if (s.substr(0, i) == string(s.rbegin(), s.rbegin() + i)) {
            s = s.substr(i);
            k -= i;
            if (k <= 0) return true;
        }
    }
    return false;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cout << (solve() ? "YES" : "NO") << '\n';
    }
    return 0;
}
