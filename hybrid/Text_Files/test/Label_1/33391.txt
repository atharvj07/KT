import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        int r = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = scanner.nextInt();
        int[] b = new int[n];
        for (int i = 0; i < n; i++) b[i] = scanner.nextInt();
        if (a[l - 1] == b[l - 1] && a[r - 1] == b[r - 1]) {
            boolean[] used = new boolean[n + 1];
            for (int i = l; i <= r; i++) used[a[i - 1]] = true;
            int cnt = 0;
            for (int i = l; i <= r; i++) {
                if (!used[b[i - 1]]) cnt++;
                used[b[i - 1]] = true;
            }
            if (cnt <= 1) System.out.println("TRUTH");
            else System.out.println("LIE");
        } else System.out.println("LIE");
        scanner.close();
    }
}
