q = int(input())
for _ in range(q):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    
    odd_positions = [i for i, x in enumerate(a, start=1) if x % 2 == 1]
    
    if len(odd_positions) < k or (len(odd_positions) - k) % 2 == 1:
        print("NO")
    else:
        print("YES")
        print(*odd_positions[:k-1], n)
