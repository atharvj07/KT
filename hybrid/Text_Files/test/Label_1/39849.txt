mod = 10**9 + 7
s = list(input().strip())
n = len(s)
dp = [[[0 for k in range(3)] for j in range(n + 1)] for i in range(2)]
dp[0][n][0] = dp[1][n][0] = 1
for i in range(n - 1, -1, -1):
    for j in range(n):
        for k in range(3):
            dp[0][j][k] = dp[0][j + 1][k]
            if s[j] == 'a' and k < 2:
                dp[0][j][k] += dp[0][j + 1][k + 1]
            if s[j] == 'b' and k > 0:
                dp[0][j][k] += dp[0][j + 1][k - 1]
            dp[1][j][k] = dp[0][j][k]
            if s[j] == 'a':
                dp[1][j][k] += dp[1][j + 1][k]
            if k > 0 and s[j] == 'b':
                dp[1][j][k] += dp[1][j + 1][k - 1]
            dp[0][j][k] %= mod
            dp[1][j][k] %= mod
print(dp[1][0][2])
