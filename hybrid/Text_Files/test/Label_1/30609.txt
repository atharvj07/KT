import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int cnt = 0, ans = x;
        while (n-- > 0) {
            char op = sc.next().charAt(0);
            int val = sc.nextInt();
            if (op == '+') {
                ans += val;
            } else {
                if (ans >= val) {
                    ans -= val;
                } else {
                    cnt++;
                }
            }
        }
        sc.close();
        System.out.println(ans + " " + cnt);
    }
}
