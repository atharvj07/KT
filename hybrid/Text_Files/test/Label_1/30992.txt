import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int m = scanner.nextInt(); // number of animals
        int n = scanner.nextInt(); // boat capacity

        // It is impossible to transport all the animals so that no one got upset, and all the goats survived
        if (m > n && m % 2 == 0) {
            System.out.println(-1);
            return;
        }

        // Calculate the minimum number of crossings required
        int crossings = 0;

        // Transport the goats first
        crossings += Math.ceil(m / 2.0);

        // Transport the remaining animals
        crossings += Math.ceil((double) (m - Math.ceil(m / 2.0)) / n);

        System.out.println(crossings);
    }
}
