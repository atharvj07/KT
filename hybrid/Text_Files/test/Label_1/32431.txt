import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(reader.readLine());
        for (int i = 0; i < t; i++) {
            String[] line = reader.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            int q = Integer.parseInt(line[2]);
            while (q-- > 0) {
                line = reader.readLine().split(" ");
                BigInteger l = new BigInteger(line[0]);
                BigInteger r = new BigInteger(line[1]);
                BigInteger count = r.subtract(l).add(BigInteger.ONE);
                if (a == b) {
                    count = BigInteger.ZERO;
                } else if (a == 1) {
                    count = count.subtract(r.divide(BigInteger.valueOf(b)).add(BigInteger.ONE));
                } else if (b == 1) {
                    count = count.subtract(r.divide(BigInteger.valueOf(a)).add(BigInteger.ONE));
                } else {
                    BigInteger step = BigInteger.valueOf(Math.max(a, b)).multiply(BigInteger.valueOf(Math.min(a, b)));
                    count = count.subtract(r.divide(step).subtract(l.divide(step)).add(BigInteger.ONE));
                }
                System.out.print(count + " ");
            }
            System.out.println();
        }
    }
}
