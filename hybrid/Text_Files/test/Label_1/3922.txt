#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1 << 17, MAXQ = 1e5, MAXBIT = 30;
int n, m, a[MAXN], v[MAXN];
bool f[MAXN][MAXBIT];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n >> m;
  int lim = 1 << n;
  for (int i = 0; i < lim; i++) cin >> a[i];
  for (int i = 0; i < lim; i++) f[i][0] = a[i] & 1;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < lim; j++) {
      if ((j >> i) % 2)
        f[j][i + 1] = f[j][i] ^ f[j ^ (1 << i)][i];
      else
        f[j][i + 1] = f[j][i] | f[j ^ (1 << i)][i];
    }
  }
  for (int i = 0; i < n; i++) {
    for (int j = lim - 1; j >= 0; j--) {
      if (i % 2 == 0)
        v[j] = v[j | (1 << i)] | f[j][i];
      else
        v[j] = v[j | (1 << i)] ^ f[j][i];
    }
  }
  while (m--) {
    int p, b;
    cin >> p >> b;
    p--;
    int g = 0;
    for (int i = 0; i < MAXBIT; i++) {
      if (b & (1 << i)) g |= 1 << i;
    }
    for (int i = lim - 1; i >= 0; i--) {
      if ((i >> p) % 2 == 0)
        v[i] = v[i | (1 << p)] | f[i][p];
      else
        v[i] = v[i | (1 << p)] ^ f[i][p];
      if (v[i] >= g) f[i][p] = 1;
    }
    cout << v[0] << '\n';
  }
}
