import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt(), n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] gates = new int[m][w];
        for (int i = 0; i < m; i++) {
            String gate = sc.next();
            for (int j = 0; j < w; j++) {
                char op = gate.charAt(j);
                switch (op) {
                    case 'A': gates[i][j] = 0; break;
                    case 'O': gates[i][j] = 1; break;
                    case 'X': gates[i][j] = 2; break;
                    case 'a': gates[i][j] = 3; break;
                    case 'o': gates[i][j] = 4; break;
                    case 'x': gates[i][j] = 5; break;
                }
            }
        }
        sc.close();
        for (int i = 0; i < m; i++) {
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    cnt += evaluate(gates[i], a[j], a[k]);
                }
            }
            System.out.println(cnt);
        }
    }
    public static int evaluate(int[] gate, int a, int b) {
        int[] x = new int[gate.length];
        int[] y = new int[gate.length];
        for (int i = 0; i < gate.length; i++) {
            x[i] = (a >> i) & 1;
            y[i] = (b >> i) & 1;
            switch (gate[i]) {
                case 0: x[i] = x[i] & y[i]; break;
                case 1: x[i] = x[i] | y[i]; break;
                case 2: x[i] = x[i] ^ y[i]; break;
                case 3: x[i] = 1 - (x[i] & y[i]); break;
                case 4: x[i] = 1 - (x[i] | y[i]); break;
                case 5: x[i] = 1 - (x[i] ^ y[i]); break;
            }
        }
        int ans = 0;
        for (int i = 0; i < gate.length; i++) {
            ans = ans | (x[i] << i);
        }
        return ans == 0 ? 1 : 0;
    }
}
