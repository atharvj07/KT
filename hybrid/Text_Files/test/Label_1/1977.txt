#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    getline(cin,s);
    int l=s.length();
    bool vis[26];
    int c=0;
    memset(vis, 0, sizeof vis);
    for(int i=1;i<l-1;i++)
    {
        if(s[i]>='a' && s[i]<='z')
        {
            int pos=s[i]-'a';
            if(!vis[pos])
            {
                c++;
                vis[pos]=1;
            }
        }
    }
    cout<<c<<"\n";
    return 0;
}
