import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 4) {
                ans++;
            } else if (arr[i] == 3) {
                if (ans % 2 == 1) {
                    ans++;
                }
                ans++;
            } else if (arr[i] == 2) {
                if (ans % 2 == 0) {
                    ans++;
                }
                ans++;
            } else {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
