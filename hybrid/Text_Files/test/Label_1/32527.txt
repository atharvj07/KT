import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        char[] s = scanner.nextLine().toCharArray();
        int count = 0;
        for (int i = 0; i < s.length - 1; i++) {
            if (s[i] != s[i + 1]) {
                count++;
                i++;
            }
        }
        System.out.println(count);
    }
}
