import java.util.*;

public class Main {
    private static int n, m;
    private static int[] parent, wants;
    private static List<List<Integer>> adj;

    private static boolean dfs(int u, int p) {
        parent[u] = p;
        for (int v : adj.get(u)) {
            if (v != p && !dfs(v, u)) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        parent = new int[n + 1];
        wants = new int[n + 1];
        adj = new ArrayList<>(Collections.nCopies(n + 1, new ArrayList<>()));

        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj.get(u).add(v);
        }

        for (int i = 1; i <= n; i++) {
            wants[i] = sc.nextInt();
        }

        boolean cycle = !dfs(1, 0);

        if (cycle) {
            System.out.println(-1);
            return;
        }

        List<Integer> result = new ArrayList<>();
        boolean[] visited = new boolean[n + 1];

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                int v = i;
                while (!visited[v]) {
                    visited[v] = true;
                    int w = wants[v];
                    if (v != w && !adj.get(w).contains(v)) {
                        System.out.println(-1);
                        return;
                    }
                    v = parent[w];
                }
                result.add(i);
            }
        }

        System.out.println(result.size());
        for (int v : result) {
            System.out.println(v);
        }
    }
}
