#include <bits/stdc++.h>
using namespace std;
#define ll long long
int n, p;
ll sum;
vector<int> a;
double backtrack(int id, int cur) {
  if (id == n) {
    if (cur <= p) return 1;
    return 0;
  }
  return (backtrack(id + 1, cur + a[id]) + backtrack(id + 1, cur)) / 2;
}
int main() {
  cin >> n;
  a = vector<int>(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  cin >> p;
  cout << fixed << setprecision(9) << backtrack(0, 0) << "\n";
}
