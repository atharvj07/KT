import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static class Airplane {
        int x, y;
        char direction;

        public Airplane(int x, int y, char direction) {
            this.x = x;
            this.y = y;
            this.direction = direction;
        }
    }

    static class Pair {
        Airplane a1, a2;
        int time;

        public Pair(Airplane a1, Airplane a2, int time) {
            this.a1 = a1;
            this.a2 = a2;
            this.time = time;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        Queue<Pair> queue = new LinkedList<>();
        Airplane[] airplanes = new Airplane[n];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            char direction = sc.next().charAt(0);

            airplanes[i] = new Airplane(x, y, direction);
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                Airplane a1 = airplanes[i];
                Airplane a2 = airplanes[j];

                if (willCollide(a1, a2)) {
                    int time = getTimeToCollision(a1, a2);
                    queue.add(new Pair(a1, a2, time));
                }
            }
        }

        if (queue.isEmpty())
            System.out.println("SAFE");
        else {
            Pair firstCollision = queue.poll();
            int x1 = firstCollision.a1.x;
            int y1 = firstCollision.a1.y;
            int x2 = firstCollision.a2.x;
            int y2 = firstCollision.a2.y;
            int time = firstCollision.time;
            System.out.println(time);
        }
    }

    static boolean willCollide(Airplane a1, Airplane a2) {
        if (a1.direction == a2.direction)
            return false;

        if (a1.direction == 'U' && a2.direction == 'D') {
            return a1.x == a2.x;
        } else if (a1.direction == 'R' && a2.direction == 'L') {
            return a1.y == a2.y;
        } else if (a1.direction == 'D' && a2.direction == 'U') {
            return a1.x == a2.x;
        } else if (a1.direction == 'L' && a2.direction == 'R') {
            return a1.y == a2.y;
        }

        return false;
    }

    static int getTimeToCollision(Airplane a1, Airplane a2) {
        if (a1.direction == 'U' && a2.direction == 'D') {
            return (a2.y - a1.y) / 10;
        } else if (a1.direction == 'R' && a2.direction == 'L') {
            return (a2.x - a1.x) / 10;
        } else if (a1.direction == 'D' && a2.direction == 'U') {
            return (a1.y - a2.y) / 10;
        } else if (a1.direction == 'L' && a2.direction == 'R') {
            return (a1.x - a2.x) / 10;
        }

        return -1;
    }
}
