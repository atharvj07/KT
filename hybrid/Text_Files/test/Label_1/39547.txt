MOD = 10**9 + 7

def count_RSBS(s):
    n = len(s)
    if n % 2 != 0:
        return 0

    # Precompute factorials and their modular inverses for efficient binomial coefficient calculation
    fact = [1] * (n + 1)
    inv_fact = [1] * (n + 1)
    for i in range(2, n + 1):
        fact[i] = (fact[i-1] * i) % MOD
    inv_fact[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n - 1, 1, -1):
        inv_fact[i] = (inv_fact[i + 1] * (i + 1)) % MOD

    def binomial(n, k):
        if k < 0 or k > n:
            return 0
        return fact[n] * inv_fact[k] * inv_fact[n - k] % MOD

    left_brackets = 0
    answer = 0
    for bracket in s:
        if bracket == '(':
            left_brackets += 1
        else:
            if left_brackets > 0:
                answer = (answer + binomial(left_brackets + n//2 - 1 - (n//2 - left_brackets), n//2 - 1)) % MOD
            left_brackets = max(left_brackets - 1, 0)

    return answer

# Read input
s = input().strip()

# Compute and print the answer
print(count_RSBS(s))
