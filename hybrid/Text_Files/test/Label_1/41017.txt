from collections import deque

t = int(input())

for _ in range(t):
    n = int(input())
    users = deque(input().split())
    m = int(input())
    messages = []
    for _ in range(m):
        message = input().split(":")
        messages.append(message)
    if len(messages) < n:
        print("Impossible")
    else:
        users.rotate(-1)
        for message in messages:
            if message[0] == "?":
                message[0] = users[0]
            users.rotate(-1)
        for message in messages:
            print(":".join(message))
