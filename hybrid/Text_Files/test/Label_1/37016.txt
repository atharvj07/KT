import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    x = any(a[i] ^ a[i + 1] ^ a[i + 2] for i in range(n - 2))
    if x:
        print('NO')
    else:
        ans = [i for i in range(1, n - 2) if a[i - 1] ^ a[i] ^ a[i + 1]]
        print('YES')
        print(len(ans))
        print(*ans)
