#include <iostream>
#include <string>
#include <vector>
#define MOD 1000000007

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> dp(n + 1);
  dp[0] = dp[1] = 1;
  int open = 0;
  for (int i = 2; i <= n; i++) {
    open += (s[i - 1] == '(');
    if (s[i - 1] == ')' && open > 0) {
      dp[i] = (2 * dp[i - 1]) % MOD;
      open--;
    }
  }
  cout << dp[n] << endl;
  return 0;
}
