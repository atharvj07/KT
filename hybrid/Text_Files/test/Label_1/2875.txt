#include <bits/stdc++.h>
#define ll long long
using namespace std;

const ll maxn = 2e5 + 10;
vector<ll>adj[maxn];
ll dist[maxn], ans, n, x;

void dfs(ll s, ll p, ll d) {
    dist[s] = d;
    for(auto i : adj[s]) {
        if(i != p) {
            dfs(i, s, d + 1);
        }
    }
}

int main() {
    cin >> n >> x;
    for(int i = 1; i < n; i++) {
        ll u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1, 0);
    dfs(x, x, 0);
    for(int i = 1; i <= n; i++) {
        ans = max(ans, dist[i]);
    }
    cout << ans << endl;
    return 0;
}
