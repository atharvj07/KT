#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int MX = 1e6 + 5;
int arr[MX], psum[MX];
ll dp[MX];

ll get_unique(int l, int r) {
    return psum[r] - (l != 1 ? psum[l - 1] : 0);
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> arr[i]; }

    sort(arr + 1, arr + n + 1);
    int cnt = 1;
    psum[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (arr[i] != arr[i - 1]) {
            cnt++;
        }
        psum[i] = cnt;
    }

    double ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (arr[i] != arr[j]) {
                ans += (1.0 * (j - i) / ((double)n * (n - 1)));
            }
        }
        dp[i] = get_unique(1, i);
    }

    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (arr[i] != arr[j]) {
                ans += 1.0 * (dp[j] - dp[i - 1]) * ((double)(j - i) / (n * (n - 1)));
            }
        }
    }

    printf("%.6lf\n", ans);
    return 0;
}
