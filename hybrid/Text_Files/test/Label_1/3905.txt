#include<bits/stdc++.h>
using namespace std;
#define int long long
const int mod=1e9+7;
int dp[1001];
int solve(int n){
	if(n==0) return 1;
	if(dp[n]!=-1) return dp[n];
	int ans=0;
	for(int i=1;i<=n;i+=2) ans=(ans+solve(n-i))%mod;
	for(int i=1;i<=n;i+=5) ans=(ans+solve(n-i))%mod;
	for(int i=1;i<=n;i+=10) ans=(ans+solve(n-i))%mod;
	for(int i=1;i<=n;i+=50) ans=(ans+solve(n-i))%mod;
	dp[n]=ans;
	return ans;
}
signed main()
{
	int n;
	memset(dp,-1,sizeof dp);
	cin>>n;
	cout<<(solve(n)+mod)%mod<<endl;
}
