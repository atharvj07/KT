#include <bits/stdc++.h>
#define pb push_back
#define vi vector<int>
#define vd vector<double>
#define mk make_pair
#define F first
#define S second
#define ll long long
using namespace std;

int main() {
  int n, x, sm = 0, ans = 0;
  cin >> n;
  vi v;
  while (n--) cin >> x, v.pb(x), sm += x;
  sort(v.begin(), v.end());
  int val = sm / v.size();
  for (auto i : v) ans += abs(val - i);
  ans /= 2;
  cout << ans << endl;
  return 0;
}
