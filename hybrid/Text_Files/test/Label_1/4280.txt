#include <bits/stdc++.h>
using namespace std;
#define pb push_back
#define ff first
#define ss second
#define ll long long
#define pii pair<int,int>
#define vi vector<int>
const int N = 1e5+10;
int n, k, a[N], ans = 0;
int main() {
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 1; i <= n; i++) {
        if (k <= 0) break;
        if (a[i] >= 0) continue;
        k--;
        ans -= a[i] * 2;
        a[i] *= -1;
    }
    for (int i = 1; i <= n; i++) ans += a[i];
    cout << ans << "\n";
    return 0;
}
