#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    string s;
    cin >> n >> s;
    vector<int> freq(26, 0);
    for (int i = 0; i < n; i++) { freq[s[i] - 'a']++; }
    int odd = 0;
    for (int i = 0; i < 26; i++) { if (freq[i] % 2 == 1) odd++; }
    if (odd > 1) {
      cout << -1 << endl;
    } else {
      int ans = 0;
      for (int i = 0; i < 26; i++) { ans += freq[i] % 2; }
      cout << ans / 2 << endl;
    }
  }
  return 0;
}
