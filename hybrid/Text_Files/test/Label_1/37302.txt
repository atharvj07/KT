a, b, c = map(int, input().split())
n = int(input())
x = list(map(int, input().split()))
x.sort()
if b not in x and c not in x:
    ans = n
elif b in x and c not in x:
    ans = sum([i == b for i in x])
elif b not in x and c in x:
    ans = sum([i == c for i in x])
else:
    idx_b = x.index(b)
    idx_c = x.index(c)
    ans = sum([i < b for i in x]) + sum([i > c for i in x])
    if idx_b + 1 == idx_c:
        ans -= 1
print(ans)
