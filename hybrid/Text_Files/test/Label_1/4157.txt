#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll n;
vector<pair<ll, ll>>coord;
int main(){
	cin>>n;
	ll x, y;
	for(ll i=0; i<2*n; i++){
		cin>>x>>y;
		coord.push_back({x, y});
	}
	sort(coord.begin(), coord.end());
	ll xmin=LLONG_MAX, xmax=0, ymin=LLONG_MAX, ymax=0;
	for(auto a: coord){
		xmin=min(xmin, a.first);
		xmax=max(xmax, a.first);
		ymin=min(ymin, a.second);
		ymax=max(ymax, a.second);
	}
	if(xmin==xmax || ymin==ymax){
		cout<<0<<endl;
	}else{
		cout<<(xmax-xmin)*(ymax-ymin)<<endl;
	}
	return 0;
}
