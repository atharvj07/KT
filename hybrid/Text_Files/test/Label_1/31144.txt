import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        Map<String, String> teamToName = new LinkedHashMap<>();
        Map<String, String> nameTaken = new LinkedHashMap<>();
        for (int i = 0; i < n; i++) {
            String[] input = br.readLine().split(" ");
            String team = input[0];
            String name = input[1];

            if (!teamToName.containsKey(team)) {
                teamToName.put(team, name);
            }

            String shortName1 = team.substring(0, 3);
            String shortName2 = team.substring(0, 2) + name.substring(0, 1);
            if (!nameTaken.containsKey(shortName1) && !nameTaken.containsKey(shortName2)) {
                teamToName.put(team, shortName1);
                nameTaken.put(shortName1, team);
            } else if (!nameTaken.containsKey(shortName2)) {
                teamToName.put(team, shortName2);
                nameTaken.put(shortName2, team);
            } else {
                System.out.println("NO");
                return;
            }
        }

        System.out.println("YES");
        for (String team : teamToName.keySet()) {
            System.out.println(teamToName.get(team));
        }
    }
}
