import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long m = sc.nextInt();
        Map<Long, Long> count = new HashMap<>();
        List<Pair> edges = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            long w = sc.nextInt();
            edges.add(new Pair(u, v, w));
            count.put(w, count.getOrDefault(w, 0L) + 1);
        }
        long gcd = 0;
        for (long x : count.keySet()) {
            gcd = GCD(gcd, x);
        }
        long ans = 0;
        for (long x : count.keySet()) {
            if (x % gcd == 0 && m % gcd == 0) {
                ans += count.get(x) * count.getOrDefault(m / gcd, 0L);
            }
        }
        for (Pair edge : edges) {
            long w = edge.w;
            if (w % gcd != 0 || m % gcd != 0) {
                continue;
            }
            ans -= count.getOrDefault(w / gcd, 0L) - 1;
        }
        System.out.println(ans / 2);
    }

    public static long GCD(long a, long b) {
        if (b == 0) {
            return a;
        }
        return GCD(b, a % b);
    }

    private static class Pair {
        int u;
        int v;
        long w;

        public Pair(int u, int v, long w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }
}
