import java.util.*;

class Main {
    static int mod = 1000000007;
    static int p = 31;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        String t = in.nextLine();
        int n = s.length();
        int m = t.length();
        long[] h = new long[n];
        long[] pwr = new long[n];
        pwr[0] = 1;
        for (int i = 1; i < n; i++) {
            pwr[i] = (pwr[i - 1] * p) % mod;
        }
        h[0] = s.charAt(0);
        for (int i = 1; i < n; i++) {
            h[i] = (h[i - 1] * p + s.charAt(i)) % mod;
        }
        long[] ht = new long[m];
        ht[0] = t.charAt(0);
        for (int i = 1; i < m; i++) {
            ht[i] = (ht[i - 1] * p + t.charAt(i)) % mod;
        }
        long tHash = ht[m - 1];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            long curHash = 0;
            for (int j = i; j < n; j++) {
                if (j - i + 1 > m) {
                    curHash = (curHash - (h[j] - (i > 0 ? h[i - 1] : 0)) + mod) % mod;
                } else {
                    curHash = (curHash * p + s.charAt(j)) % mod;
                }
                if (j - i + 1 == m && curHash == tHash) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
