import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int minCount = arr[n - 1] - arr[0];
        int maxCount = arr[n - 1] - arr[k - 1];
        System.out.println(Math.min(minCount, maxCount));
    }
}
