import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] cnt = new int[10];
        for (char c : s.toCharArray())
            cnt[c - '0']++;
        if (cnt[8] == 0)
            System.out.println(0);
        else
            System.out.println(Math.min(cnt[8], Math.min(cnt[0], Math.min(cnt[1], cnt[2] / 3))));
    }
}
