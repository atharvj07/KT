import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int x = sc.nextInt();
        int[][] p = new int[d + 1][n + 1];
        for (int i = 1; i <= d; i++) {
            for (int j = 1; j <= n; j++) {
                p[i][j] = sc.nextInt();
            }
        }
        int[][] dp = new int[d + 1][x + 1];
        for (int i = 1; i <= d; i++) {
            for (int j = 0; j <= x; j++) {
                for (int k = 1; k <= n; k++) {
                    if (j >= p[i][k]) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - p[i][k]] + p[i][k]);
                    }
                }
                dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
            }
        }
        System.out.println(dp[d][x]);
    }
}
