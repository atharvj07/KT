import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long ans = (long) Math.pow(10, n) - (long) Math.pow(9, n) - (long) Math.pow(9, n - 1);
        ans %= 1000000007;
        System.out.println(ans);
    }
}
