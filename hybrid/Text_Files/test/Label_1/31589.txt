import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scan.nextInt();
        }
        int day = 0;
        int[] visited = new int[n];
        for (int i = 0; i < n; i++) {
            if (visited[i] == 0) {
                int cur = i;
                while (visited[cur] == 0) {
                    visited[cur] = 1;
                    cur = a[cur] - 1;
                }
                day++;
            }
        }
        System.out.println(day);
    }
}
