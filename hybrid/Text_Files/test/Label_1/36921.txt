from collections import deque

def solve(t, data):
    results = []
    for i in range(t):
        n, m, edges = data[i]
        graph = [[] for _ in range(n)]
        reverse_graph = [[] for _ in range(n)]
        in_degree = [0] * n
        for u, v in edges:
            graph[u-1].append(v-1)
            reverse_graph[v-1].append(u-1)
            in_degree[v-1] += 1

        # Step 1: Find shortest distances from capital using BFS
        distances = [-1] * n
        distances[0] = 0
        queue = deque([0])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if distances[neighbor] == -1:
                    distances[neighbor] = distances[node] + 1
                    queue.append(neighbor)

        # Step 2: Propagate minimum distances backwards
        queue = deque([i for i in range(n) if in_degree[i] == 0])
        min_distances = distances[:]
        while queue:
            node = queue.popleft()
            for neighbor in reverse_graph[node]:
                if min_distances[node] < min_distances[neighbor]:
                    min_distances[neighbor] = min_distances[node]
                    queue.append(neighbor)

        results.append(min_distances)
    return results


# Reading and parsing input
t = int(input().strip())
data = []
for _ in range(t):
    input()  # Skip empty line
    n, m = map(int, input().strip().split())
    edges = [tuple(map(int, input().strip().split())) for _ in range(m)]
    data.append((n, m, edges))

# Solve each test case
results = solve(t, data)

# Print results
for result in results:
    print(' '.join(map(str, result)))
