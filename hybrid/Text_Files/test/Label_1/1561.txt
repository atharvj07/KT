#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> instr(n + 1, vector<int>(m + 1));
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      cin >> instr[i][j];
    }
  }
  vector<bool> locked(k + 1, false);
  vector<int> time(n + 1, 0);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (locked[instr[i][j]]) {
        time[i] = j;
        break;
      }
      if (instr[i][j]) { locked[instr[i][j]] = true; }
    }
  }
  for (int i = 1; i <= n; i++) { cout << time[i] << endl; }
  return 0;
}
