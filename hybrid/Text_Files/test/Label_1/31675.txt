import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int n = in.nextInt();
        int m = in.nextInt();
        int k = in.nextInt();
        int s = in.nextInt();

        int[][] aij = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                aij[i][j] = in.nextInt();
            }
        }

        int[] qi = new int[s];
        for (int i = 0; i < s; i++) {
            qi[i] = in.nextInt();
        }

        int[][] dp = new int[s][k + 1];
        for (int i = 0; i < s; i++) {
            Arrays.fill(dp[i], -1);
        }
        dp[0][qi[0]] = 0;
        for (int i = 1; i < s; i++) {
            for (int j = 1; j <= k; j++) {
                if (dp[i - 1][j] != -1) {
                    for (int x = 0; x < n; x++) {
                        for (int y = 0; y < m; y++) {
                            if (aij[x][y] == j) {
                                dp[i][aij[x][y]] = Math.max(dp[i][aij[x][y]], dp[i - 1][j] + Math.abs(x - (i)) + Math.abs(y - (i)));
                            }
                        }
                    }
                }
            }
        }

        int ans = 0;
        for (int i = 1; i <= k; i++) {
            ans = Math.max(ans, dp[s - 1][i]);
        }

        System.out.println(ans);
    }
}
