from sys import stdin
input = stdin.readline

n, m, q = map(int, input().split())

grid = []
for _ in range(n):
  grid.append(list(map(int, input().split())))

for _ in range(q):
  x, y = map(int, input().split())
  grid[x-1][y-1] ^= 1
  
  ans = 0
  for row in grid:
    cnt = 0
    for cell in row:
      cnt = cnt + cell if cell else 0
      ans = max(ans, cnt)
  print(ans)
