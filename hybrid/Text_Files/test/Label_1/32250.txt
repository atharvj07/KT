import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        System.out.println(countCamels(n, t));
    }

    private static int countCamels(int n, int t) {
        if (n < 3 || t < 1 || n < t * 2 + 1) {
            return 0;
        } else if (t == 1) {
            return countCamels(n - 2);
        } else {
            return countCamels(n - 2) * (n - t * 2);
        }
    }

    private static int countCamels(int n) {
        if (n < 3) {
            return 0;
        } else if (n == 3) {
            return 1;
        } else {
            return countCamels(n - 1) + countCamels(n - 2);
        }
    }
}
