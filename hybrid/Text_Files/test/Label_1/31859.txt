import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[] s = sc.next().toCharArray();
        int q = sc.nextInt();
        int[] ans = new int[q];
        for (int i = 0; i < q; i++) {
            int m = sc.nextInt();
            char c = sc.next().charAt(0);
            int[] cnt = new int[26];
            for (int j = 0; j < m; j++) {
                cnt[s[j] - 'a']++;
            }
            int mx = 0;
            for (int j = 0; j < 26; j++) {
                if (j == c - 'a') {
                    cnt[j] = m;
                }
                mx = Math.max(mx, cnt[j]);
            }
            ans[i] = mx;
        }
        for (int i = 0; i < q; i++) {
            System.out.println(ans[i]);
        }
    }
}
