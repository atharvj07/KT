import sys
from sys import stdin
input = stdin.readline
n, m, q = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
ans = [[0] * m for _ in range(n)]
for i in range(n):
    for j in range(1, m):
        if a[i][j] == a[i][j - 1]:
            ans[i][j] = ans[i][j - 1] + 1
for j in range(m):
    for i in range(1, n):
        if a[i][j] == a[i - 1][j]:
            ans[i][j] = max(ans[i][j], ans[i - 1][j] + 1)
for _ in range(q):
    x, y = map(int, input().split())
    a[x - 1][y - 1] ^= 1
    ans[x - 1][y - 1] = 0
    for i in range(1, m):
        if a[x - 1][y - 1] == a[x - 1][y - 1 - i]:
            ans[x - 1][y - 1] = max(ans[x - 1][y - 1], ans[x - 1][y - 1 - i] + 1)
    for i in range(1, n):
        if a[x - 1][y - 1] == a[x - 1 - i][y - 1]:
            ans[x - 1][y - 1] = max(ans[x - 1][y - 1], ans[x - 1 - i][y - 1] + 1)
    print(max(map(max, ans)))
