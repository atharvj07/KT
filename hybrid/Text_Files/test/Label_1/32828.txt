import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Vertex[] g = new Vertex[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new Vertex();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].adj.add(b);
            g[b].adj.add(a);
        }
        int s1 = sc.nextInt();
        int t1 = sc.nextInt();
        int l1 = sc.nextInt();
        int s2 = sc.nextInt();
        int t2 = sc.nextInt();
        int l2 = sc.nextInt();
        int ans = -1;
        if (bfs(g, s1, t1, l1) && bfs(g, s2, t2, l2)) {
            ans = m - dfs(g, 1, -1, s1, t1, l1, s2, t2, l2);
        }
        System.out.println(ans);
    }

    static boolean bfs(Vertex[] g, int s, int t, int l) {
        Queue<Integer> q = new LinkedList<>();
        q.add(s);
        g[s].d = 0;
        while (!q.isEmpty()) {
            int u = q.poll();
            if (u == t) {
                return true;
            }
            for (int v : g[u].adj) {
                if (g[v].d == -1) {
                    g[v].d = g[u].d + 1;
                    q.add(v);
                }
            }
        }
        return false;
    }

    static int dfs(Vertex[] g, int u, int p, int s1, int t1, int l1, int s2, int t2, int l2) {
        int ans = 0;
        for (int v : g[u].adj) {
            if (v != p) {
                if (u == s1 && v == t1 && g[v].d > l1) {
                    ans++;
                } else if (u == s2 && v == t2 && g[v].d > l2) {
                    ans++;
                } else {
                    ans += dfs(g, v, u, s1, t1, l1, s2, t2, l2);
                }
            }
        }
        return ans;
    }
}

class Vertex {
    List<Integer> adj = new ArrayList<>();
    int d = -1;
}
