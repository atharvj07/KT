import sys
input = sys.stdin.readline

for t in range(int(input())):
    n, k = map(int, input().split())
    s = list(input().strip())
    last_zero = -1
    for i in range(n):
        if s[i] == '1':
            last_zero = i
    for i in range(k):
        if 0 <= last_zero + 1 < n and s[last_zero] == '1' and s[last_zero + 1] == '0':
            s[last_zero], s[last_zero + 1] = '0', '1'
            last_zero += 1
        else:
            break
    print("".join(s))
