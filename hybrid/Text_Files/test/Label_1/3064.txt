#include <iostream>

using namespace std;

int main() {
  int a[2];
  cin >> a[0] >> a[1];
  while (a[0] > 0 && a[1] > 0) {
    if (a[0] > a[1]) {
      a[1] += a[0] % 2;
      if (a[0] > 1)
        a[0] -= 2;
      else
        a[0] = 0;
    } else {
      a[0] += a[1] % 2;
      if (a[1] > 1)
        a[1] -= 2;
      else
        a[1] = 0;
    }
  }
  cout << min((a[0] + 1) / 2, (a[1] + 1) / 2);
}
