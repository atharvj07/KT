import java.util.*;

class Main {
    static int[] a, b;
    static int n, m;
    static int x, k, y;
    static long[] p;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        x = in.nextInt();
        k = in.nextInt();
        y = in.nextInt();

        p = new long[n];
        a = new int[n];
        b = new int[m];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }

        HashMap<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            map.put(a[i], i);
        }

        for (int i = 0; i < m; i++) {
            int j = map.get(b[i]);
            if (i == 0) {
                p[j] = 0;
            } else if (j > map.get(b[i - 1])) {
                p[j] = p[j - 1] + x;
            } else {
                p[j] = p[j - 1] - x;
            }
        }

        long ans = Long.MAX_VALUE;
        long sum = 0;
        for (int i = 0; i < n; i++) {
            if (i + k < n) {
                sum = Math.min(sum + y, p[i] - p[i + k]);
            }
            ans = Math.min(ans, p[i] + sum);
        }

        if(ans == Long.MAX_VALUE){
            System.out.println(-1);
        }else{
            System.out.println(ans);
        }

    }
}
