import java.util.Scanner;

public class DiverseStringChecker {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        for (int i = 0; i < n; i++) {
            String string = scanner.nextLine();
            if (isDiverse(string)) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }
    }

    private static boolean isDiverse(String string) {
        boolean[] seen = new boolean[26]; // Use a boolean array to track seen letters
        int prevPosition = -1; // Keep track of the previous letter's position

        for (char c : string.toCharArray()) {
            int position = c - 'a';
            if (seen[position] || (prevPosition != -1 && Math.abs(position - prevPosition) > 1)) {
                return false; // Duplicate letter or non-consecutive letters
            }
            seen[position] = true;
            prevPosition = position;
        }

        return true; // All letters are distinct and consecutive
    }
}
