import queue

n=int(input())
a=list(map(int,input().split()))
f=a.index(-1)
a[f]=0
q=queue.PriorityQueue()
for i in range(n):
    q.put((a[i],i))
ans=0
while q.qsize()>1:
    c1=q.get()
    c2=q.get()
    if c1[0]>0 and c2[0]>0:
        ans+=min(c1[0],c2[0])
        c1=(c1[0]-min(c1[0],c2[0]),c1[1])
        c2=(c2[0]-min(c1[0],c2[0]),c2[1])
    if c2[0]==0:
        if f==c2[1]:
            f=c1[1]
        q.put(c1)
    else:
        q.put(c2)
print(ans)
