def solve_brackets(n, segments):
    dp = [[False] * (2 * n + 1) for _ in range(n + 1)]
    parent = [[None] * (2 * n + 1) for _ in range(n + 1)]
    dp[0][0] = True

    for i in range(1, n + 1):
        l, r = segments[i - 1]
        for j in range(2 * n):
            if dp[i - 1][j]:
                for k in range(l, r + 1):
                    if j + k + 1 <= 2 * n:
                        dp[i][j + k + 1] = True
                        parent[i][j + k + 1] = (j, k + 1)

    if not dp[n][2 * n]:
        return "IMPOSSIBLE"

    seq = [0] * (2 * n)
    balance = 2 * n
    for i in range(n, 0, -1):
        prev_balance, diff = parent[i][balance]
        balance = prev_balance
        seq[prev_balance] = '('
        seq[prev_balance + diff] = ')'

    return ''.join(seq)

n = int(input().strip())
segments = [list(map(int, input().strip().split())) for _ in range(n)]
print(solve_brackets(n, segments))
