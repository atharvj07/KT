#include <iostream>
using namespace std;
int main() {
  int n, s, t, a[101], b[101], mini;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> s >> t;
  if (s > t) {
    int tmp = t;
    t = s;
    s = tmp;
  }
  b[s] = 0;
  for (int i = s + 1; i <= n; i++) b[i] = b[i - 1] + a[i - 1];
  for (int i = 1; i < s; i++) b[i] = b[n] + b[i];
  mini = b[t] - b[s - 1];
  for (int i = s; i + 1 <= t; i++)
    if (b[i + 1] - b[i] < mini) mini = b[i + 1] - b[i];
  cout << mini << endl;
  return 0;
}
