#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll a[300005], dp[300005][20], f[65];
int main() {
    int n;
    cin >> n;
    int msb = 0;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        for (int j = 0; j < 64; j++) {
            if (a[i] & (1LL << j)) f[j]++;
            if (f[j] > 0) msb = max(msb, j);
        }
        for (int j = 0; j <= msb; j++) { dp[i][j] = dp[i - 1][j]; }
        dp[i][msb] ^= 1;
    }
    ll ans = 0;
    int now = 0;
    for (int i = n; i >= 1; i--) {
        for (int j = msb; j >= 0; j--) {
            int tmp = now ^ j;
            if (!((tmp & 1) == j)) tmp++;
            if (f[j] > 0 && dp[i - 1][tmp]) {
                ans += f[j];
                break;
            }
        }
        for (int j = 0; j < 64; j++) {
            if (a[i] & (1LL << j)) f[j]--;
            if (f[j] == 0) msb--;
        }
        now ^= msb;
    }
    cout << ans << endl;
    return 0;
}
