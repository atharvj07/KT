#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, l;
  cin >> n >> l;
  vector<int> k(n), s(n);
  for (int &i : k) cin >> i;
  for (int &i : s) cin >> i;
  int ks = k[0], ss = s[0];
  bool ans = true;
  for (int i = 0; i < n; ++i) {
    if (k[i] - ks == s[i] - ss) {
      continue;
    } else if (l - k[i] + ks == s[i] - ss) {
      ks = k[i];
    } else {
      ans = false;
      break;
    }
  }
  cout << (ans ? "YES" : "NO") << endl;
  return 0;
}
