import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int a = sc.nextInt(), b = sc.nextInt(), c = sc.nextInt();
            int ans = 0;
            while (a > 0 && b > 1) {
                a--;
                b -= 2;
                ans += 3;
            }
            while (b > 0 && c > 1) {
                b--;
                c -= 2;
                ans += 3;
            }
            System.out.println(ans);
        }
    }
}
