import java.util.*;

public class Main {
    static int[] a, b;
    static int[][] adj;

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) a[i] = input.nextInt();
        for (int i = 0; i < n; i++) b[i] = input.nextInt();
        adj = new int[n][n];
        for (int i = 0; i < m; i++) {
            int u = input.nextInt() - 1, v = input.nextInt() - 1;
            adj[u][v] = adj[v][u] = 1;
        }
        boolean[] vis = new boolean[n];
        int[] ans = new int[n];
        for (int i = 0; i < n; i++) {
            if (vis[i]) continue;
            ans[i] = b[i];
            if (!dfs(i, -1, vis, ans, b[i])) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(adj[i][j] * ans[j] + " ");
            }
            System.out.println();
        }
    }

    static boolean dfs(int u, int p, boolean[] vis, int[] ans, int sum) {
        vis[u] = true;
        for (int v = 0; v < adj[u].length; v++) {
            if (v != p && adj[u][v] == 1) {
                ans[v] = b[v] - ans[u];
                if (ans[v] < 0 || ans[u] + ans[v] != b[u] || !dfs(v, u, vis, ans, sum - ans[v]))
                    return false;
            }
        }
        return true;
    }
}
