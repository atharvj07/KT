import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String[] s = new String[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.next();
        }
        String t = sc.next();
        Set<Integer> ans = new TreeSet<>();
        for (int i = 0; i <= n; i++) {
            boolean ok = true;
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                int res = s[j].compareTo(t);
                if (res < 0) {
                    ok = false;
                    break;
                } else if (res == 0) {
                    if (s[j].length() == t.length()) {
                        ok = false;
                        break;
                    }
                }
            }
            if (ok) {
                ans.add(i + 1);
            }
        }
        for (int i : ans) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}
