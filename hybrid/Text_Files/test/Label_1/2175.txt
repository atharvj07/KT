#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  int ans = 0, cur = 1;
  for (int i = 1; i < n; i++) {
    if (a[i] != a[i - 1])
      cur++;
    else {
      ans = max(ans, cur);
      cur = 1;
    }
  }
  ans = max(ans, cur);
  cout << ans << endl;
}
