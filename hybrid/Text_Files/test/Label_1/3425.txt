#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll day = 1440;
const ll mod = 1e9 + 7;
const ll mx = 1e6 + 10;
ll a[mx];

int main() {
    ll n; cin >> n;
    a[0] = 20;
    a[1] = 50;
    a[2] = 120;
    for (ll i = 3; i < mx; i++) {
        a[i] = min({20 + a[i - 1], 50 + a[i - 90], 120 + a[i - day]});
    }

    ll time, ans = 0;
    for (ll i = 0; i < n; i++) {
        cin >> time;
        ans = a[time];
        cout << ans << '\n';
    }
    return 0;
}
