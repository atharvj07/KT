import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();
        for (int i = 0; i < T; i++) {
            String s = sc.next();
            int[] count = new int[26];
            int n = s.length();
            for (int j = 0; j < n; j++) {
                count[s.charAt(j) - 'a']++;
            }
            long ans = 0;
            for (int j = 0; j < 26; j++) {
                ans += (long) count[j] * (count[j] + 1) / 2;
            }
            System.out.println(ans);
        }
    }
}
