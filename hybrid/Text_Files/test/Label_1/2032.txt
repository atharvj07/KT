#include <iostream>
#include <cmath>

using namespace std;

int main() {
  long long n;
  cin >> n;
  for (long long k = 1; k <= n; k++) {
    long long cnt = 0, rem = n;
    for (;;) {
      cnt += k;
      rem -= k;
      if (rem < 0) break;
      rem -= rem / 10;
      if (rem < 10) break;
    }
    if (cnt * 2 >= n) {
      cout << k;
      return 0;
    }
  }
  cout << -1;
  return 0;
}
