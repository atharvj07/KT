import java.util.*;
import java.lang.*;

class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int x = sc.nextInt();
            long[] a = new long[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }

            long sum = 0;
            for (long ai : a) {
                sum += ai;
            }

            long wealthy = 0;
            if (sum >= x * n) {
                wealthy = n;
            } else {
                wealthy = sum / x;
            }

            System.out.println(wealthy);
        }
    }
}
