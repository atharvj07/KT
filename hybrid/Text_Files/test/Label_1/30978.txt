import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'o') cnt++;
            else {
                cnt++;
                break;
            }
        }
        System.out.println(cnt);
    }
}
