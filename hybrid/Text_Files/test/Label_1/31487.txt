import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long m = in.nextLong();
        long k = in.nextLong();
        int[][] grid = new int[(int) (n + 4)][(int) (m + 4)];
        for (int i = 0; i < k; i++) {
            int x1 = in.nextInt();
            int y1 = in.nextInt();
            int x2 = in.nextInt();
            int y2 = in.nextInt();
            if (x1 == x2) {
                grid[x1 + 1][y1 + 1] ^= 1;
                grid[x2 + 1][y2 + 1] ^= 1;
            } else {
                grid[x1 + 1][y1 + 1] ^= 2;
                grid[x2 + 1][y2 + 1] ^= 2;
            }
        }
        int ans = 0;
        for (int i = 0; i <= n + 1; i++) {
            for (int j = 0; j <= m + 1; j++) {
                ans ^= grid[i][j];
            }
        }
        if (ans == 0) {
            System.out.println("SECOND");
            return;
        }
        System.out.println("FIRST");
        int x = 0, y = 0, t = 0;
        for (int i = 0; i <= n + 1; i++) {
            for (int j = 0; j <= m + 1; j++) {
                if (grid[i][j] != 0) {
                    if (grid[i][j] == 1) {
                        x = i;
                        y = j;
                        t = 1;
                    } else {
                        x = j;
                        y = i;
                        t = 2;
                    }
                }
            }
        }
        if (t == 1) {
            System.out.println((x - 1) + " " + (y - 1) + " " + (x - 1) + " " + y);
        } else {
            System.out.println((x - 1) + " " + (y - 1) + " " + x + " " + (y - 1));
        }
    }
}
