import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long prod = 1;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            prod *= x;
        }
        System.out.println(prod);
    }
}
