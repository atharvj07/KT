from collections import defaultdict

def dfs(graph, node, target, color, visited):
    if node == target:
        return True
    visited.add(node)
    for neighbour in graph[node]:
        if neighbour not in visited and graph[node][neighbour] == color and dfs(graph, neighbour, target, color, visited):
            return True
    return False

n, m = map(int, input().split())
graph = defaultdict(lambda: defaultdict(int))

for _ in range(m):
    a, b, c = map(int, input().split())
    graph[a][b] = c
    graph[b][a] = c

q = int(input())
for _ in range(q):
    u, v = map(int, input().split())
    colors = set()
    for a in graph:
        for b in graph[a]:
            if dfs(graph, u, v, graph[a][b], set()):
                colors.add(graph[a][b])
    print(len(colors))
