#include<bits/stdc++.h>
using namespace std;
#define MX 100005

int main(){
  int t,px,py,f;
  char s[100005];
  cin>>t;
  while(t--){
    cin>>px>>py>>s;
    int x=0,y=0;
    f=1;
    int n=strlen(s);
    for(int i=0;i<n;++i){
      if(s[i]=='U') y++;
      else if(s[i]=='D') y--;
      else if(s[i]=='L') x--;
      else if(s[i]=='R') x++;
      if(x==px&&y==py) {
        f=0;
        break;
      }
    }
    if(f) cout<<"NO"<<endl;
    else cout<<"YES"<<endl;
  }
  return 0;
}
