import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static long[] a;
    static long[] st;
    static long[] lazy;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        int m = (int) (Math.log(n) / Math.log(2)) + 1;
        st = new long[n * 4];
        lazy = new long[n * 4];
        build(1, 0, n - 1);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            System.out.println(query(1, 0, n - 1, l, r));
        }
    }

    static void build(int node, int start, int end) {
        if (start == end) {
            st[node] = a[start];
        } else {
            int mid = (start + end) / 2;
            build(2 * node, start, mid);
            build(2 * node + 1, mid + 1, end);
            st[node] = st[2 * node] ^ st[2 * node + 1];
        }
    }

    static long query(int node, int start, int end, int l, int r) {
        if (lazy[node] != 0) {
            st[node] ^= lazy[node];
            if (start != end) {
                lazy[2 * node] ^= lazy[node];
                lazy[2 * node + 1] ^= lazy[node];
            }
            lazy[node] = 0;
        }
        if (start > r || end < l) {
            return 0;
        }
        if (start >= l && end <= r) {
            return st[node];
        }
        int mid = (start + end) / 2;
        long p1 = query(2 * node, start, mid, l, r);
        long p2 = query(2 * node + 1, mid + 1, end, l, r);
        return p1 ^ p2;
    }
}
