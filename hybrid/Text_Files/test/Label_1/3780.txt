#include <bits/stdc++.h>
using namespace std;
#define int long long

const int MOD = 1e9+7;

int n, a, b, k, f;
unordered_map<string,unordered_map<string,int>> mp;
int dfs(string u){
    if(k==0) return a;
    for(auto &p: mp[u]){
        if(p.second==1){
            k--, p.second=0;
            return b+dfs(p.first);
        }
    }
    return a;
}

int32_t main(){
    cin >> n >> a >> b >> k >> f;
    a*=n, b*=n, f*=k;
    for(int i=0;i<n;i++){
        string u, v;
        cin >> u >> v;
        mp[u][v]++, mp[v][u]++;
    }
    int ans = min(dfs("BerBank"), min(f+b*(n-1), a));
    cout << ans << endl;
}
