import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            int[] p = new int[m];
            for (int i = 0; i < m; i++) {
                p[i] = sc.nextInt() - 1;
            }
            boolean[] vis = new boolean[n];
            for (int x : p) {
                vis[x] = true;
                vis[x + 1] = true;
            }
            int[] b = a.clone();
            for (int i = 1; i < n; i++) {
                if (b[i] < b[i - 1]) {
                    if (!vis[i - 1]) {
                        System.out.println("NO");
                        return;
                    }
                    for (int j = i; j > 0; j--) {
                        if (b[j] < b[j - 1]) {
                            int tmp = b[j - 1];
                            b[j - 1] = b[j];
                            b[j] = tmp;
                        } else {
                            break;
                        }
                    }
                }
            }
            System.out.println("YES");
        }
    }
}
