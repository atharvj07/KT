import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int d = sc.nextInt();
            if (d == 0) break;
            int hd = sc.nextInt();
            int md = sc.nextInt();
            int a = sc.nextInt();
            int ha = sc.nextInt();
            int ma = sc.nextInt();
            int dist = Math.abs(d - a);
            int fare = fare_table[dist];
            if (fare == 0) {
                System.out.println(0);
                continue;
            }
            if ((17 <= hd && hd <= 19 && 30 <= md) || (17 <= ha && ha <= 19 && 30 <= ma)) {
                fare = (fare + 24) / 50 * 50;
            }
            System.out.println(fare);
        }
        sc.close();
    }

    static int[] fare_table = {0, 300, 450, 600, 800, 1000, 1300};
}
