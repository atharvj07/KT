import sys

def peaks_and_valleys(n, a, b):
    if (a + b + 1) % 2 or a + b > n - 2:
        return -1

    arr = [0] * n
    for i in range(a + b):
        if i % 2 == 0:
            arr[i // 2 + 1] = i + 2
        else:
            arr[n - i // 2 - 1] = i + 2

    idx = 1
    for i in range(n):
        if not arr[i]:
            arr[i] = idx
            idx += 1

    return ' '.join(map(str, arr))

for _ in range(int(input())):
    n, a, b = map(int, input().split())
    print(peaks_and_valleys(n, a, b))
