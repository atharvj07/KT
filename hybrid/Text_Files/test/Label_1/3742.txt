#include <bits/stdc++.h>
typedef long long ll;
using namespace std;

const int MAXN = 1e5 + 10;
int cnt[MAXN];
int main(){
    string a, b;
    cin >> a >> b;
    for(char cur : a){
        cnt[cur - 'a']++;
    }
    int ans = 0;
    for(char cur : b){
        int tmp = cur - 'a';
        ans += (cnt[tmp] == 0);
    }
    if(ans) cout << ans;
    else cout << -1;
    return 0;
}
