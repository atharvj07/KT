n = int(input());s = input()
a = [int(i) for i in input().split()]
dp=[[0 for k in range(n+1)] for l in range(n+1)]
for i in range(n):
 dp[i][i+1]=a[0]
for l in range(2,n+1):
  for i in range(n):
    j=i+l-1
    if j>=n: continue
    for k in range(i,j+1):
      dp[i][j+1]=max(dp[i][j+1],dp[i][k]+dp[k+1][j+1]+a[l-1])
print(dp[0][n])
