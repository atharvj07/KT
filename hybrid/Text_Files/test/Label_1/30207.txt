import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String n = sc.nextLine();
        int len = n.length();
        int m = 0;
        ArrayList<String> ans = new ArrayList<>();
        for (int i = len - 1; i >= 0; i--) {
            if (n.charAt(i) == '0') {
                continue;
            }
            int p = len - 1 - i;
            if (p > m) {
                m = p;
            }
            if (n.charAt(i) == '1') {
                ans.add("+2^" + p);
            } else {
                ans.add("-2^" + p);
            }
        }
        System.out.println(m);
        for (String s : ans) {
            System.out.println(s);
        }
    }
}
