import java.util.*;

public class Main {
    static long[] masks;
    static long full;

    static void initMasks(int n, long[][] bm) {
        full = (1L << n) - 1;
        masks = new long[n];
        for (int i = 0; i < n; i++) {
            masks[i] = 0;
            for (int j = 0; j < bm[i][0]; j++) {
                masks[i] |= (1L << bm[i][j + 1]);
            }
        }
    }

    static void test(long state, int i) {
        System.out.println((state >> i) % 2);
    }

    static void set(long state, int m) {
        System.out.println(state | masks[m]);
    }

    static void clear(long state, int m) {
        System.out.println(state & ~masks[m]);
    }

    static void flip(long state, int m) {
        System.out.println(state ^ masks[m]);
    }

    static void all(long state, int m) {
        System.out.println((state & masks[m]) == masks[m] ? 1 : 0);
    }

    static void any(long state, int m) {
        System.out.println((state & masks[m]) != 0 ? 1 : 0);
    }

    static void none(long state, int m) {
        System.out.println((state & masks[m]) == 0 ? 1 : 0);
    }

    static void count(long state, int m) {
        System.out.println(Long.bitCount(state & masks[m]));
    }

    static void val(long state, int m) {
        System.out.println(state & masks[m]);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] bm = new long[n][n + 1];
        for (int i = 0; i < n; i++) {
            bm[i][0] = sc.nextInt();
            for (int j = 1; j <= bm[i][0]; j++) {
                bm[i][j] = sc.nextInt();
            }
        }
        initMasks(n, bm);
        int q = sc.nextInt();
        long state = 0;
        for (int i = 0; i < q; i++) {
            int code = sc.nextInt();
            int m = sc.nextInt();
            switch (code) {
                case 0:
                    test(state, m);
                    break;
                case 1:
                    set(state, m);
                    break;
                case 2:
                    clear(state, m);
                    break;
                case 3:
                    flip(state, m);
                    break;
                case 4:
                    all(state, m);
                    break;
                case 5:
                    any(state, m);
                    break;
                case 6:
                    none(state, m);
                    break;
                case 7:
                    count(state, m);
                    break;
                case 8:
                    val(state, m);
                    break;
            }
        }
    }
}
