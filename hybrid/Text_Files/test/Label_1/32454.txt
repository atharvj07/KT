import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt() - 1;
        int b = sc.nextInt() - 1;
        String s = sc.next();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = s.charAt(i) - '0';
        }
        int[] dp = new int[n];
        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[a] = 0;
        for (int i = a + 1; i < n; i++) {
            if (arr[i] == arr[i - 1]) {
                dp[i] = dp[i - 1];
            } else {
                dp[i] = dp[i - 1] + 1;
            }
        }
        for (int i = b - 1; i >= 0; i--) {
            if (arr[i] == arr[i + 1]) {
                dp[i] = Math.min(dp[i], dp[i + 1]);
            } else {
                dp[i] = Math.min(dp[i], dp[i + 1] + 1);
            }
        }
        System.out.println(dp[b]);
    }
}
