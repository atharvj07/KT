import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class BobsGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            graph.get(a).add(b);
            graph.get(b).add(a);
        }
        System.out.println(solve(graph, 1, -1));
    }

    static int solve(List<List<Integer>> graph, int node, int parent) {
        int size = 1;
        int maxSubtreeSize = 0;
        for (int child : graph.get(node)) {
            if (child != parent) {
                int subtreeSize = solve(graph, child, node);
                size += subtreeSize;
                maxSubtreeSize = Math.max(maxSubtreeSize, subtreeSize);
            }
        }
        maxSubtreeSize = Math.max(maxSubtreeSize, graph.size() - size);
        return maxSubtreeSize;
    }
}
