#include<iostream>
#include<string>
using namespace std;

int main() {
    string s;
    getline(cin, s);
    bool zero = false, one = false, two = false, three = false;
    for (int i = 0; i < s.size(); i++) {
        if (s[i] >= 'A' && s[i] <= 'Z') zero = true;
        if (s[i] >= '0' && s[i] <= '9') one = true;
        if (s[i] >= 'a' && s[i] <= 'z') two = true;
        if (s[i] == '!' || s[i] == '?' || s[i] == '.' || s[i] == ',' || s[i] == '_') three = true;
    }
    if (s.size() >= 5 && zero && one && two && three) cout << "Correct" << endl;
    else cout << "Too weak" << endl;
    return 0;
}
