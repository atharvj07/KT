import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int t = scanner.nextInt();
        int res = 0;
        if (t == 1) {
            for (int i = 1; i <= n; i++) {
                for (int j = i; j <= n; j++) {
                    if (i != j) {
                        res++;
                    }
                }
            }
        }
        System.out.println(res);
    }
}
