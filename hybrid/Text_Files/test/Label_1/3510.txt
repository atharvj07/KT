#include<iostream>
#include<cmath>
using namespace std;
typedef long long ll;
int main()
{
    ll n,i,a[105],cnt=0,l,r,flag=0;
    cin>>n;
    for(i=1;i<=n;i++)
        cin>>a[i];
    for(i=2;i<=n;i++)
    {
        if(a[i]==a[i-1])
        {
            flag=1;
            break;
        }
        if(a[i]==a[i+1] && i+1<=n)
        {
            flag=1;
            break;
        }
        if(a[i]==1){
            l=0;
            r=sqrt(2)*(i-1);
        }
        if(a[i]==2)
        {
            l=2*(i-1);
            r=l+sqrt(2);
        }
        if(a[i]==3)
        {
            l=2*(i-1)+sqrt(2);
            r=l+2;
        }
        cnt+=((ll)ceil(r-(l+0.000000000000001))+1);
    }
    if(flag==1)
        cout<<"Infinite\n";
    else
    {
        cout<<"Finite\n";
        cout<<cnt<<endl;
    }
    return 0;
}
