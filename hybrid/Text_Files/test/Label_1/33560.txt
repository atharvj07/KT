import java.util.*;

public class Main {
    public static final int MAX_N = 100;
    public static final int MAX_M = 200;

    static int n, m;
    static int[] a = new int[MAX_N];
    static int[] b = new int[MAX_N];
    static boolean[][] adj = new boolean[MAX_N][MAX_N];
    static int[][] ans = new int[MAX_N][MAX_N];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            adj[u][v] = adj[v][u] = true;
        }
        sc.close();

        if (!isPossible()) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }

    static boolean isPossible() {
        // Find the number of soldiers that should move from each city to each other city
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    ans[i][j] = b[i] - a[i];
                } else if (adj[i][j]) {
                    ans[i][j] = Math.min(b[j], a[i] - b[i]);
                    a[i] -= ans[i][j];
                    b[j] += ans[i][j];
                }
            }
        }

        // Check if the number of soldiers in each city is equal to the target number
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                return false;
            }
        }

        return true;
    }
}
