import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Arrays;
import java.util.StringTokenizer;

class Main {
    static final int mod = (int) 1e9 + 7;
    static int[] a;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        a = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        SegmentTree stree = new SegmentTree(n, a);
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int t = Integer.parseInt(st.nextToken());
            int l = Integer.parseInt(st.nextToken()) - 1;
            int r = Integer.parseInt(st.nextToken());
            if (t == 1) {
                int x = Integer.parseInt(st.nextToken());
                stree.update(l, r, x);
            } else {
                System.out.println(stree.query(l, r));
            }
        }
    }

    static class SegmentTree {
        int[] tree;
        int n;

        public SegmentTree(int n, int[] a) {
            this.n = n;
            tree = new int[4 * n];
            build(1, 0, n - 1, a);
        }

        void build(int node, int start, int end, int[] a) {
            if (start == end) {
                tree[node] = a[start];
            } else {
                int mid = (start + end) / 2;
                build(2 * node, start, mid, a);
                build(2 * node + 1, mid + 1, end, a);
                tree[node] = tree[2 * node] + tree[2 * node + 1];
            }
        }

        void update(int start, int end, int x) {
            update(1, 0, n - 1, start, end, x);
        }

        void update(int node, int start, int end, int l, int r, int x) {
            if (start > r || end < l) {
                return;
            }
            if (start >= l && end <= r) {
                tree[node] += x;
                return;
            }
            int mid = (start + end) / 2;
            update(2 * node, start, mid, l, r, x);
            update(2 * node + 1, mid + 1, end, l, r, x);
            tree[node] = (tree[2 * node] + tree[2 * node + 1]) % mod;
        }

        int query(int start, int end) {
            return query(1, 0, n - 1, start, end);
        }

        int query(int node, int start, int end, int l, int r) {
            if (start > r || end < l) {
                return 0;
            }
            if (start >= l && end <= r) {
                return tree[node];
            }
            int mid = (start + end) / 2;
            int left = query(2 * node, start, mid, l, r);
            int right = query(2 * node + 1, mid + 1, end, l, r);
            return (left + right) % mod;
        }
    }

    static long[] fib = new long[100001];

    static {
        fib[1] = 1L;
        fib[2] = 1L;
        for (int i = 3; i < 100001; i++) {
            fib[i] = (fib[i - 1] + fib[i - 2]) % mod;
        }
    }
}
