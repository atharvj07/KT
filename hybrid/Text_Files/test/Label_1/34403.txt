import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] x, y;
    static HashSet<Integer> set;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(reader.readLine());
        x = new int[n];
        y = new int[n];
        set = new HashSet<>();

        String[] line = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            x[i] = Integer.parseInt(line[i]);
            set.add(x[i]);
        }

        line = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            y[i] = Integer.parseInt(line[i]);
            set.add(y[i]);
        }

        int count = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (set.contains(x[i] ^ y[j])) {
                    count++;
                }
            }
        }

        if (count % 2 == 0) {
            System.out.println("Karen");
        } else {
            System.out.println("Koyomi");
        }
    }
}
