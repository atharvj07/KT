import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        int[][] queries = new int[q][3];
        for (int i = 0; i < q; i++) {
            for (int j = 0; j < 3; j++) {
                queries[i][j] = sc.nextInt();
            }
        }
        int[] counts = Arrays.stream(queries).mapToInt(query -> count(a, query[0], query[1], query[2])).toArray();
        for (int count : counts) {
            System.out.println(count);
        }
    }

    private static int count(int[] a, int b, int e, int k) {
        int count = 0;
        for (int i = b; i < e; i++) {
            if (a[i] == k) {
                count++;
            }
        }
        return count;
    }
}
