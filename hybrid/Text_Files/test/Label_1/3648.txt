#include <bits/stdc++.h>
using namespace std;
#define ll unsigned long long
ll get(ll a, ll b) {
  ll c = a & b;
  ll cc = a;
  while (c) { cc = cc >> 1, c = c >> 2; }
  return cc;
}
int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    vector<ll> a(n);
    for (ll &x : a) cin >> x;
    ll ans = 0;
    for (ll i = 0; i < n; i++) {
      for (ll j = i + 1; j < n; j++) {
        if (get(a[i], a[j]) >= get(a[i] ^ a[j], a[j])) ans++;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
