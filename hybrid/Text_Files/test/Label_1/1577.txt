#include <cstdio>
#include <vector>
using namespace std;
const int maxn = 1e5 + 10;
vector<int> ans, g[maxn];
int n, deg[maxn], p[maxn], dis[maxn], vis[maxn];
bool win[maxn];
void dfs(int u) {
  if (vis[u]) return;
  vis[u] = 1;
  for (auto v : g[u]) {
    dfs(v);
    if (win[v]) deg[u]++;
  }
  if (deg[u] % 2) win[u] = 0;
}
int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; i++) scanf("%d", p + i), g[p[i]].push_back(i);
  dfs(1);
  for (int i = 1; i <= n; i++) {
    if (!win[p[i]] && dis[p[i]] == 1) ans.push_back(i);
    for (auto j : g[p[i]]) dis[j] = dis[p[i]] + 1;
  }
  printf("%d\n", ans.size());
  for (auto x : ans) printf("%d ", x);
  return 0;
}
