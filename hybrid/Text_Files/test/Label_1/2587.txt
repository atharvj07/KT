#include<bits/stdc++.h>
using namespace std;
long long a[200005],b[200005];
bool v[200005]={0};
int main()
{
	long long n,ans=0;
	cin>>n;
	for(int i=1;i<=n;i++)
	{
		cin>>a[i];
		b[i]=a[i];
	}
	sort(b+1,b+n+1);
	for(long long i=1;i<=n;i++)
	{
		long long now=0;
		for(long long j=i+1;j<=n;j++)
		{
			now|=a[j];
			if(now>b[i])
				ans++;
		}
	}
	cout<<ans<<endl;
}
