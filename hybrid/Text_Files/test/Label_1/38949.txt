import sys
input=sys.stdin.readline
n,m=map(int,input().split())
g=[[] for i in range(n+1)]
for _ in range(m):
    a,b=map(int,input().split())
    g[a].append(b)
flag=[0 for i in range(11000)]
def dfs(v):
    flag[v]=1
    for i in g[v]:
        if(flag[i]==0):
            dfs(i)
cnt=0
for i in range(1,n+1):
    if(flag[i]==0):
        dfs(i)
        cnt+=1
print(cnt)
