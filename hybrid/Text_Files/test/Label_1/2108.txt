#include <iostream>
#include <vector>

using namespace std;

int main() {
  bool ans = false;
  string s;
  vector<bool> vis(101, false);
  cin >> s;
  int sz = s.size();
  for (int i = 0; i < sz; i++) {
    if (!vis[i]) {
      int j = i, cnt = 0;
      while (j < sz && !vis[j]) {
        vis[j] = true;
        cnt += (s[j] - '0');
        j += 6;
      }
      if (cnt % 2 == 0) {
        ans = true;
        break;
      }
    }
  }
  cout << (ans ? "yes" : "no") << endl;
  return 0;
}
