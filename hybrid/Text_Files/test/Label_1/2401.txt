#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        vector<vector<int>> ans;
        for (int i = 0; i < k; i++) {
            vector<int> cur;
            int per_table = (n % m + k - 1) / k;
            for (int j = 0; j < m; j++) {
                int player_cnt = n / m + (per_table >= 2);
                cur.emplace_back(player_cnt);
                for (int l = 1; l <= player_cnt; l++) {
                    cur.emplace_back(((j * per_table + l - 1) % n) + 1);
                }
            }
            ans.emplace_back(cur);
        }
        for (const auto &v : ans) {
            for (int i = 0; i < m; i++) {
                cout << v[i];
                for (int j = i * v[i] + 1; j <= i * v[i] + v[i]; j++) {
                    cout << ' ' << v[j];
                }
                cout << '\n';
            }
            cout << '\n';
        }
    }
}
