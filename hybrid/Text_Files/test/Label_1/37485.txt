import math

def is_perfect_square(n):
    return math.isqrt(n) ** 2 == n

def solve(n, a):
    for i in a:
        if not is_perfect_square(i):
            return "YES"
    return "NO"

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(solve(n, a))
