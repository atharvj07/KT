#include <iostream>
#include <vector>
#include <queue>

using namespace std;

struct Node {
    vector<int> adj;
    int dist;
    bool visited;
};

int main() {
    int n, k;
    cin >> n >> k;

    vector<Node> nodes(n + 1);
    vector<int> universities(2 * k);
    for (int i = 0; i < 2 * k; i++) {
        cin >> universities[i];
    }

    for (int i = 0; i < n - 1; i++) {
        int x, y;
        cin >> x >> y;
        nodes[x].adj.push_back(y);
        nodes[y].adj.push_back(x);
    }

    int max_sum = 0;
    for (int i = 0; i < 2 * k; i++) {
        for (int j = i + 1; j < 2 * k; j++) {
            queue<int> q;
            q.push(universities[i]);
            nodes[universities[i]].visited = true;
            nodes[universities[i]].dist = 0;

            int max_dist = 0;
            while (!q.empty()) {
                int curr = q.front();
                q.pop();

                max_dist = max(max_dist, nodes[curr].dist);

                for (int neighbor : nodes[curr].adj) {
                    if (!nodes[neighbor].visited) {
                        nodes[neighbor].dist = nodes[curr].dist + 1;
                        nodes[neighbor].visited = true;
                        q.push(neighbor);
                    }
                }
            }

            max_sum = max(max_sum, max_dist);

            for (int i = 1; i <= n; i++) {
                nodes[i].visited = false;
                nodes[i].dist = 0;
            }
        }
    }

    cout << max_sum << endl;

    return 0;
}
