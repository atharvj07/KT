import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();

        int speedLimit = 0;
        boolean canOvertake = true;
        int signsIgnored = 0;

        while (n-- > 0) {
            int type = scanner.nextInt();

            switch (type) {
                case 1:
                    speedLimit = 0;
                    canOvertake = true;
                    break;
                case 2:
                    if (!canOvertake) signsIgnored++;
                    break;
                case 3:
                    speedLimit = scanner.nextInt();
                    break;
                case 4:
                    canOvertake = true;
                    break;
                case 5:
                    speedLimit = 0;
                    break;
                case 6:
                    canOvertake = false;
                    break;
            }
        }

        System.out.println(signsIgnored);
    }
}
