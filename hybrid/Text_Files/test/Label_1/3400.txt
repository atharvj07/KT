#include <bits/stdc++.h>
using namespace std;
const int N=2e5+5;
#define ll long long
map<ll,vector<pair<int,int>>>mp;
vector<pair<ll,ll>>v;
int32_t main() {
    ios_base::sync_with_stdio(false), cin.tie(NULL), cout.tie(NULL);
    int n,m;
    cin>>n>>m;
    for(int i=1; i<=n; i++){
        for(int j=1; j<=m; j++){
            int x;
            cin>>x;
            mp[x].push_back({i,j});
        }
    }
    for(auto x:mp){
        v.push_back({x.second[0].second,x.second[0].first});
    }
    sort(v.begin(),v.end());
    int mov=0;
    for(int i=0; i<v.size(); i++){
        int p=v[i].first,q=v[i].second;
        if(p!=i+1){
            mov++;
            p=i+1;
            for(auto x:mp[q]){
                if(x.second==p){
                    swap(q,x.second);
                }
                v[x.second-1].first=p;
            }
        }
    }
    cout<<mov<<endl;
    return 0;
}
