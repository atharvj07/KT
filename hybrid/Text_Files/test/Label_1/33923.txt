import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {

    private static class Edge {
        public int to;
        public int weight;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    private static class Node {
        public int numEdges;
        public List<Edge> edges;

        public Node() {
            numEdges = 0;
            edges = new ArrayList<>();
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node();
        }

        int[] weights = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            int weight = Integer.parseInt(st.nextToken());
            weights[i] = weight;
            nodes[u].numEdges++;
            nodes[v].numEdges++;
            nodes[u].edges.add(new Edge(v, weight));
            nodes[v].edges.add(new Edge(u, weight));
        }

        Arrays.sort(weights);

        if (weights[0] != 2 || weights[n - 2] % 2 != 0) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            int m = n - 2;
            System.out.println(m);
            for (int i = 1; i <= n; i++) {
                if (nodes[i].numEdges == 1) {
                    for (Edge edge : nodes[i].edges) {
                        int weight = edge.weight;
                        int j = Arrays.binarySearch(weights, weight);
                        int k = Arrays.binarySearch(weights, weight + 2);
                        if (k >= 0) {
                            System.out.println(i + " " + edge.to + " " + (weights[k] - weight));
                            m--;
                        }
                    }
                }
            }

            for (int i = 1; i <= n; i++) {
                if (nodes[i].numEdges == 1) {
                    for (Edge edge : nodes[i].edges) {
                        int weight = edge.weight;
                        int j = Arrays.binarySearch(weights, weight);
                        int k = Arrays.binarySearch(weights, weight + 4);
                        if (k >= 0 && m > 0) {
                            System.out.println(i + " " + edge.to + " " + (weights[k] - weight));
                            m--;
                        }
                    }
                }
            }
        }
    }
}
