#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    string s;
    cin >> n >> k;
    cin >> s;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == '*') {
        ans += (i % k == 0 || (i + 1) % k == 0);
      }
    }
    cout << ans << endl;
  }
  return 0;
}
