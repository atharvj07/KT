#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int q; cin >> q; while (q--) {
        int n; cin >> n;
        int a[n]; for (int i = 0; i < n; i++) cin >> a[i];
        sort(a, a + n);
        long long sum = 0;
        for (int i = 1; i < n; i++) sum += a[i] - a[0];
        cout << a[0] + sum / n << endl;
    }
    return 0;
}
