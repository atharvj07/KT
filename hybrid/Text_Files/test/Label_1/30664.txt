import java.util.*;

public class Main {

    static class Edge {
        int src;
        int dest;

        public Edge(int src, int dest) {
            this.src = src;
            this.dest = dest;
        }
    }

    static class Graph {
        int V;
        List<Edge>[] adjList;

        public Graph(int V) {
            this.V = V;
            adjList = new List[V];
            for (int i = 0; i < V; i++) {
                adjList[i] = new ArrayList<>();
            }
        }

        public void addEdge(int src, int dest) {
            adjList[src].add(new Edge(src, dest));
            adjList[dest].add(new Edge(dest, src));
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int V = sc.nextInt();
        int E = sc.nextInt();
        int[] A = new int[V];
        for (int i = 0; i < V; i++) {
            A[i] = sc.nextInt();
        }
        int[] B = new int[V];
        for (int i = 0; i < V; i++) {
            B[i] = sc.nextInt();
        }

        Graph graph = new Graph(V);
        for (int i = 0; i < E; i++) {
            int src = sc.nextInt() - 1;
            int dest = sc.nextInt() - 1;
            graph.addEdge(src, dest);
        }

        boolean possible = true;
        for (int i = 0; i < V; i++) {
            int sum = 0;
            for (Edge edge : graph.adjList[i]) {
                sum += B[edge.dest];
            }
            if (sum > A[i]) {
                possible = false;
            }
        }

        if (!possible) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        int[][] ans = new int[V][V];
        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                if (i == j) {
                    ans[i][j] = B[i];
                } else {
                    for (Edge edge : graph.adjList[i]) {
                        if (edge.dest == j) {
                            ans[i][j] = B[j];
                            break;
                        }
                    }
                }
            }
        }

        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }
}
