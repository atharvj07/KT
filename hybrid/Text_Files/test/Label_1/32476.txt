import java.util.*;

public class Main {
  static final long mod = ((long)1e9)+7;
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    final int n = sc.nextInt();
    long[] a = new long[n];
    for (int i=0; i<n; i++) {
      a[i]=sc.nextLong();
    }
    Arrays.sort(a);
    long ans = a[n-1];
    for (int i=n-2; i>=0; i--) {
      if (ans < 0 && a[i]<0) {
        ans = Math.max(ans, a[i]);
        continue;
      }
      ans = (ans-a[i]%mod)%mod;
    }
    ans = (ans+mod)%mod;
    System.out.println(ans);
  }
}
