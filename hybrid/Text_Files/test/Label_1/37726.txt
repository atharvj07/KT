from collections import Counter
n = int(input())
a = list(map(int, input().split()))
ca = Counter(a)
ans = [0] * (n+1)
for i in range(1, n+1):
    for j in range(i, n+1):
        b = [ca[k] for k in a[i-1:j]]
        mx = b.count(max(b))
        if mx == 1 or max(b) * mx > sum(b) - max(b):
            ans[i] += 1
for i in range(1, n+1):
    print(ans[i], end=' ')
