#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    if (n == 2)
      cout << "YES" << endl;
    else {
      int mx = 0;
      for (int i = 0; i < n; i++) mx = max(mx, int(a[i]));
      long long sm = 0;
      for (int i = 0; i < n; i++) sm += a[i];
      if (sm % 2 || 2 * mx > sm)
        cout << "NO" << endl;
      else
        cout << "YES" << endl;
    }
  }
}
