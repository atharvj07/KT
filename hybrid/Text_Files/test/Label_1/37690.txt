import collections

n, k, m, a = map(int, input().split())
votes = collections.defaultdict(lambda: [])
for i in range(1, a + 1):
    votes[int(input())].append(i)
    
candidates = list(votes.keys())
candidates.sort(key=lambda x: (-len(votes[x]), -min(votes[x])))

for x in candidates[:k]:
    print(1)
for x in candidates[k:]:
    if len(votes[x]) > 0:
        print(2)
    else:
        print(3)
