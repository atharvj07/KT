import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextLong();
    Arrays.sort(a);
    long ans = 0l;
    for (int i = 0; i < n; i++) ans += a[i] * i;
    ans -= a[0] * (n - 1);
    System.out.println(ans);
  }
}
