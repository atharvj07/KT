import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String str = sc.next();
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (str.charAt(i) == 'I') {
                boolean allin = true;
                for (int j = 0; j < n; j++) {
                    if (j != i && str.charAt(j) != 'A' && str.charAt(j) != 'F') {
                        allin = false;
                        break;
                    }
                }
                if (allin) {
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
