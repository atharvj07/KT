t = int(input().strip())
for _ in range(t):
    n, p, k = map(int, input().strip().split())
    prices = list(map(int, input().strip().split()))
    prices.sort()
    prefix_sum = [0] * (n + 1)
    for i in range(1, n + 1):
        prefix_sum[i] = prefix_sum[i - 1] + prices[i - 1]
    max_goods = 0
    for i in range(k):
        total_cost = 0
        goods_count = 0
        if i <= n:
            total_cost += prefix_sum[i]
            if total_cost <= p:
                goods_count += i
                for j in range(i + k - 1, n, k):
                    if total_cost + prices[j] <= p:
                        total_cost += prices[j]
                        goods_count += k
                    else:
                        break
        max_goods = max(max_goods, goods_count)
    print(max_goods)
