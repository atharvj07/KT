#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll gcd(ll a,ll b){
    if(!a) return b;
    if(!b) return a;
    return gcd(b,a%b);
}
int main() {
    ll t,w,b; cin >> t >> w >> b;
    ll p = (t/w)*(w+b) - (t%w==0?0:w);
    ll q = t+b;
    ll g = gcd(p,q);
    cout << p/g << '/' << q/g << '\n';
}
