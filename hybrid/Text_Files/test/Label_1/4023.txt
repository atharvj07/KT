#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<bool> vis(n + 1, false);
  vis[1] = true;
  for (int i = 1; i <= n; i++) {
    if (s[i] == '.') {
      for (int j = i + 1; j <= min(n, i + k); j++) {
        vis[j] = true;
      }
    }
  }
  if (vis[n])
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
