#include <algorithm>
#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 2;

int main() {
    long long n, ans = 0, power = 1;
    cin >> n;
    string s = to_string(n);
    reverse(s.begin(), s.end());
    for (int i = 0; i < s.size(); i++) {
        if (s[i] <= '4') {
            ans += (s[i] - '0') * power;
        } else {
            ans += (9 - (s[i] - '0')) * power;
        }
        power *= 10;
    }
    cout << ans << "\n";
    return 0;
}
