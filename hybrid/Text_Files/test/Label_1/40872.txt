def lcp(a, b):
    i = 0
    while i < min(len(a), len(b)) and a[i] == b[i]:
        i += 1
    return i

n = int(input())
a = [input() for _ in range(n)]
b = [input() for _ in range(n)]
sa = sorted(list(range(n)), key = lambda x: a[x])
sb = sorted(list(range(n)), key = lambda x: b[x])
ans = 0
for i in range(n):
    ans += lcp(a[sa[i]], b[sb[i]])
print(ans)
for i in range(n):
    print(sa[i] + 1, sb[i] + 1)
