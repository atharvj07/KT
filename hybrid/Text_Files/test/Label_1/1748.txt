#include <bits/stdc++.h>

using namespace std;

const int M = 101, INF = 1e9;

int dp[M][M][M][M], pv[M][M], hv[M][M];

int main() {
    int n, m;
    cin >> n >> m;
    string s[n];
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++) pv[i][j] = pv[i - 1][j] + (s[i - 1][j - 1] == 'W');
    for (int j = 1; j <= m; j++)
        for (int i = 1; i <= n; i++) hv[i][j] = hv[i][j - 1] + (s[i - 1][j - 1] == 'W');
    for (int i1 = 1; i1 <= n; i1++)
        for (int j1 = 1; j1 <= m; j1++)
            for (int i2 = i1; i2 <= n; i2++)
                for (int j2 = j1; j2 <= m; j2++) {
                    if (!(i1 - 1) || !(j1 - 1)) dp[i1][j1][i2][j2] = pv[i2][j2];
                    else if (!j1) dp[i1][j1][i2][j2] = dp[i1 - 1][j1][i2 - 1][j2] + pv[i2][j2] - pv[i1 - 1][j2];
                    else dp[i1][j1][i2][j2] = dp[i1][j1 - 1][i2][j2 - 1] + pv[i2][j2] - pv[i1 - 1][j2] + hv[i2][j1 - 1];
                    if (i1 > 1 && j1 > 1) dp[i1][j1][i2][j2] = min(dp[i1][j1][i2][j2], dp[i1 - 1][j1 - 1][i2][j2] + pv[i2][j2] - pv[i1 - 1][j2] - pv[i2][j1 - 1] + pv[i1 - 1][j1 - 1]);
                }
    cout << dp[n][m][n][m] << '\n';
    return 0;
}
