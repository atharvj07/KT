#include<bits/stdc++.h>
#define mp make_pair
#define ll long long
#define pb push_back
using namespace std;
pair<int, int> a[101];
int main() {
  int n, k, l = 0, r = 0, ans = 2e9, ans2 = 0;
  cin >> n >> k;
  for (int i = 0;i < n;i++) {
    cin >> a[i].first;
    a[i].second = i;
  }
  sort(a, a + n);
  for (int i = 0;i < n;i++) {
    while (r < n - 1 && a[r + 1].first - a[i].first <= k)
      r++;
    if (a[r].first - a[i].first <= k) {
      if (a[r].first - a[i].first < ans)
        ans = a[r].first - a[i].first, ans2 = r - i;
      else if (a[r].first - a[i].first == ans)
        ans2 = min(ans2, r - i);
    }
  }
  cout << ans << " " << ans2 << endl;
  while (ans2--) {
    cout << a[r].second + 1 << " " << a[l].second + 1 << endl;
    l++, r--;
  }
  return 0;
}
