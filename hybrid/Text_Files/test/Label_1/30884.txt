import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int count = 0;
        for (int i = 0; i < n; i++) {
            String num = sc.next();
            boolean isKGood = true;
            for (int j = 0; j <= k && isKGood; j++) {
                isKGood &= num.contains(j + "");
            }
            count += isKGood ? 1 : 0;
        }
        System.out.println(count);
    }
}
