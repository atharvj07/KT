import sys
t = int(input())
for cas in range(t):
    n, m = map(int, input().split())
    g = [[] for _ in range(n+1)]
    inDeg = [0]*(n+1)
    for _ in range(m):
        u, v = map(int, input().split())
        g[u].append(v); inDeg[v]+=1
    need = []
    for i in range(1,n+1):
        if inDeg[i]==0:
            need.append(i)
    cnt = 0; ans = []
    while need:
        cur = need.pop()
        cnt+=1
        for nxt in g[cur]:
            inDeg[nxt]-=1
            if inDeg[nxt]==0:
                need.append(nxt)
                ans.append((cur,nxt))
    print(cnt)
    for u, v in ans:
        print(u,v)
