#include<iostream>
#include<vector>
#include<algorithm>
using std::cin;
using std::cout;
using std::endl;
using std::vector;
int main() {
	int t;
	cin >> t;
	while (t--) {
		int n;
		cin >> n;
		vector<int> b(n + 1);
		for (int i = 1; i <= n; ++i)
			cin >> b[i];
		bool flag = true;
		for (int i = n; i >= 2; --i) {
			if (b[i] < b[i - 1]) {
				flag = false;
				break;
			}
			b[i] -= b[i - 1];
		}
		if (b[1] == 0 || !flag) {
			cout << "NO" << endl;
			continue;
		}
		cout << "YES" << endl;
		for (int i = 1; i <= n; ++i)
			cout << b[i] << " ";
		cout << endl;
	}
}
