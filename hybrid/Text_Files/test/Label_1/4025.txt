#include <iostream>
#include <vector>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        std::string s;
        std::cin >> s;
        std::vector<int> cnt(3); // count of A, B, C
        for (char c : s)
            ++cnt[c - 'A'];
        bool can_erase = true;
        while (can_erase) {
            can_erase = false;
            if (cnt[0] > 0 && cnt[1] > 0) {
                --cnt[0];
                --cnt[1];
                can_erase = true;
            }
            if (cnt[1] > 0 && cnt[2] > 0) {
                --cnt[1];
                --cnt[2];
                can_erase = true;
            }
        }
        if (cnt[0] == 0 && cnt[1] == 0 && cnt[2] == 0)
            std::cout << "YES" << std::endl;
        else
            std::cout << "NO" << std::endl;
    }
}
