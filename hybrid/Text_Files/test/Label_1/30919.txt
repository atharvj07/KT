import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] c = new long[n], t = new long[n];
        for (int i = 0; i < n; i++) {
            c[i] = sc.nextLong();
        }
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextLong();
        }
        System.out.println(Arrays.equals(c, t) ? "Yes" : "No");
    }
}
