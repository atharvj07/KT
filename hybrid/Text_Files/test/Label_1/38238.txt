import sys

readline = sys.stdin.readline
write = sys.stdout.write

s, correct = list(), list()
for x in range(int(readline())):
    q = readline().rstrip()
    if q[0] == q[-1] == '(':
        correct.append(q)
    else:
        s.append(q)
ans = 0
for x in correct:
    if '(' in s and ')' in s:
        s.remove('(')
        s.remove(')')
        ans += 1
print(ans)
