#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

const int MAX_N = 200000;
const int MAX_K = 100000;
int dist[MAX_N];
vector<int> adj[MAX_N];
vector<int> sorted;
int n, k;
int main() {
  cin >> n >> k;
  for (int i = 0; i < 2 * k; i++) {
    int u;
    cin >> u;
    sorted.push_back(u);
  }
  sort(sorted.begin(), sorted.end());
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  dist[sorted[0]] = 1;
  vector<int> q = {sorted[0]};
  while (!q.empty()) {
    int a = q.back();
    q.pop_back();
    for (int u : adj[a]) {
      if (dist[u] == 0) {
        dist[u] = dist[a] + 1;
        q.push_back(u);
      }
    }
  }
  int ans = 0;
  for (int i = 0; i < 2 * k; i++) {
    int u = sorted[i];
    ans += dist[u];
  }
  cout << ans << endl;
  return 0;
}
