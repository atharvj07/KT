#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<bool> f(10);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    int x;
    cin >> x;
    f.assign(10, false);
    while (x) {
      f[x % 10] = true;
      x /= 10;
    }
    bool good = true;
    for (int i = 0; i <= k; i++) {
      if (!f[i]) {
        good = false;
        break;
      }
    }
    if (good) ans++;
  }
  cout << ans << endl;
  return 0;
}
