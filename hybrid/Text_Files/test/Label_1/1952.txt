#include<bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    stack<char> st;
    int l = 0;
    for (char c:s) {
        if (c=='(' || c=='[') {
            st.push(c);
        }
        else if (c==')') {
            if (st.empty() || st.top()!='(')
                l = -1;
            else
                st.pop();
        }
        else if (c==']') {
            if (st.empty() || st.top()!='[')
                l = -1;
            else
                st.pop();
        }
        if (st.empty())
            l = 0;
        cout << l << ' ';
    }
    cout << '\n';
    return 0;
}
