#include <bits/stdc++.h>

using namespace std;

int main() {
  string home, away;
  int n;
  cin >> home >> away >> n;
  map<int, int> home_cards, away_cards;
  for (int i = 0; i < n; i++) {
    int t;
    string team, player, card;
    cin >> t >> team >> player >> card;
    if (team == home) {
      if (card == "y") {
        home_cards[stoi(player)] += 1;
        if (home_cards[stoi(player)] == 2) {
          cout << home << " " << player << " " << t << endl;
        }
      } else {
        cout << home << " " << player << " " << t << endl;
      }
    } else {
      if (card == "y") {
        away_cards[stoi(player)] += 1;
        if (away_cards[stoi(player)] == 2) {
          cout << away << " " << player << " " << t << endl;
        }
      } else {
        cout << away << " " << player << " " << t << endl;
      }
    }
  }
  return 0;
}
