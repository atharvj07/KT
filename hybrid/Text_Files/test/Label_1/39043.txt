n, m = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
pref = [0] * (n + 1)
for i in range(n):
    pref[i + 1] = pref[i] + a[i]
for i in range(1, n + 1):
    l, r = i, n
    mn = pref[n]
    while l <= r:
        mid = (l + r) // 2
        if mid - i > m:
            r = mid - 1
        else:
            mn = min(mn, pref[mid] - pref[i - 1])
            l = mid + 1
    print(mn, end=' ')
