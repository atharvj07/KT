#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    
    vector<int> arr(n);
    for (int &a : arr) cin >> a;
    
    vector<int> freq(8);
    for (int a : arr) freq[a]++;
    
    if (freq[1] < (n / 3) || freq[2] < (n / 3) || freq[4] < (n / 3)) {
        cout << -1 << endl;
        return 0;
    }
    
    for (int i = 0; i < (n / 3); i++) {
        cout << 1 << ' ' << 2 << ' ' << 4 << ' ';
        freq[1]--;
        freq[2]--;
        freq[4]--;
    }
    cout << endl;
    
    return 0;
}
