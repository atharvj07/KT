#include <bits/stdc++.h>
using namespace std;
const int maxn = 50010;
int n, a[maxn], x, ans;
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    cin >> x;
    ans = 0;
    long long sum = 0;
    bool flag = false;
    for (int i = 0; i < n; i++) {
      sum += a[i];
      if (sum < (1LL) * x) {
        ans++;
        flag = true;
      } else {
        while (sum >= (1LL) * x && i + 1 < n) {
          ans++;
          sum -= a[++i];
        }
        if (sum < (1LL) * x) {
          flag = true;
          ans++;
        }
      }
    }
    if (!flag) ans++;
    cout << ans << "\n";
  }
  return 0;
}
