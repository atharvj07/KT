n, m = map(int, input().split())
a = [(a, b) for a, b in (map(int, input().split()) for _ in range(m))]
ok = False
for x in range(1, n + 1):
    for y in range(x + 1, n + 1):
        for (ax, ay) in a:
            if ax == x or ax == y or ay == x or ay == y:
                continue
            break
        else:
            ok = True
            break
    if ok:
        break
print('YES' if ok else 'NO')
