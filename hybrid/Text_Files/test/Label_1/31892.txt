import java.util.Scanner;
import java.lang.Math;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        double[][] points = new double[n][2];
        for (int i = 0; i < n; i++) {
            points[i][0] = scanner.nextDouble();
            points[i][1] = scanner.nextDouble();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 2; j < n; j++) {
                double area1 = 0;
                double area2 = 0;
                for (int k = 0; k < n; k++) {
                    double a = points[(i + k + 1) % n][0] - points[i][0];
                    double b = points[(i + k + 1) % n][1] - points[i][1];
                    double c = points[j][0] - points[i][0];
                    double d = points[j][1] - points[i][1];
                    area1 += (a * d - b * c) / 2.0;
                    a = points[(j + k + 1) % n][0] - points[j][0];
                    b = points[(j + k + 1) % n][1] - points[j][1];
                    c = points[i][0] - points[j][0];
                    d = points[i][1] - points[j][1];
                    area2 += (a * d - b * c) / 2.0;
                }
                area1 = Math.abs(area1);
                area2 = Math.abs(area2);
                sum += 2 * Math.abs(area1 - area2);
            }
        }
        System.out.println(sum % (1000 * 1000 * 1000 + 7));
    }
}
