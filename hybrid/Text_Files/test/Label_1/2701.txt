#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll MAXN = 1e5 + 5;
ll n, a[MAXN], p[MAXN], r, l, b[MAXN];
int main() {
  cin >> n >> l >> r;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> p[i];
  // calculate number of elements in c that are smaller than each number in a
  for (int i = 0; i < n; i++) {
    if (p[i] == 1)
      b[i] = l - 1;
    else if (p[i] == n)
      b[i] = r + 1;
    else
      b[i] = l + p[i] - 2;
  }
  for (int i = 0; i < n; i++) {
    if (p[i] != p[i - 1]) { cout << "-1" << endl; return 0; }
  }
  for (int i = 0; i < n; i++) {
    cout << b[i] + a[i] << " ";
  }
  cout << endl;
  return 0;
}
