#include <iostream>
#include <vector>
using namespace std;
int main() {
    int x, y, a, b;
    cin >> x >> y >> a >> b;
    vector<pair<int, int>> ans;
    for (int i = a; i <= x; ++i) {
        for (int j = b; j <= y; ++j) {
            if (i > j) {
                ans.push_back({i, j});
            }
        }
    }
    cout << ans.size() << endl;
    for (auto v : ans) {
        cout << v.first << ' ' << v.second << endl;
    }
}
