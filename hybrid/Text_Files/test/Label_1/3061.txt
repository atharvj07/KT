#include <iostream>
#include <vector>
#include <map>
using namespace std;

bool check(int x0, int y0, vector<pair<int, pair<int, int>>> b)
{
    map<pair<int, int>, bool> vis;
    for (auto& e : b)
    {
        int type = e.first, x = e.second.first, y = e.second.second;
        if (!vis[make_pair(x, y)])
        {
            vis[make_pair(x, y)] = true;
            if ((type == 'B' && (x - x0 == y - y0 || x - x0 == y0 - y)) || (type == 'R' && (x == x0 || y == y0)) || (type == 'Q'))
                return true;
        }
    }
    return false;
}

int main()
{
    int n;
    cin >> n;
    int x0, y0;
    cin >> x0 >> y0;
    vector<pair<int, pair<int, int>>> b;
    while (n--)
    {
        char c;
        int x, y;
        cin >> c >> x >> y;
        b.push_back(make_pair(c, make_pair(x, y)));
    }
    cout << (check(x0, y0, b) ? "YES" : "NO") << endl;
    return 0;
}
