#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
ll mpow(ll a, ll b) {
    if (b == 0) return 1;
    if (b == 1) return a;
    return 1LL * mpow(1LL * a * a % mod, b / 2) % mod * mpow(a, b % 2) % mod;
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        ll pw = mpow(2, k);
        cout << mpow(pw + 1, n) - 1 << "\n";
    }
    return 0;
}
