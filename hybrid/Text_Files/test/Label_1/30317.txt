import java.util.*;

public class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        System.out.println(countWorlds(n, m));
    }

    private static int countWorlds(int n, int m) {
        if (m < 2 || n < m - 1) {
            return 0;
        }
        long[][] dp = new long[n + 1][m + 1];
        dp[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                // Don't add the new vertex.
                dp[i][j] = dp[i - 1][j];
                // Add the new vertex and connect it to an existing edge.
                for (int k = 2; k <= j; k += 2) {
                    dp[i][j] = (dp[i][j] + dp[i - 1][j - k]) % MOD;
                }
            }
        }
        return (int) dp[n][m];
    }
}
