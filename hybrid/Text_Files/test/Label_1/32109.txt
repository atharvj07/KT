import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        String input = scanner.nextLine();
        scanner.close();

        int numDistinctDigits = 0;
        int[] digitCount = new int[10];

        for (int i = 0; i < input.length(); i++) {
            int digit = Integer.parseInt(input.substring(i, i + 1));
            digitCount[digit]++;
            if (digitCount[digit] == 1) {
                numDistinctDigits++;
            }
        }

        long numVariants = 1;
        for (int i = 0; i < 10; i++) {
            numVariants *= factorial(digitCount[i]);
        }

        numVariants /= factorial(numDistinctDigits);

        System.out.println(numVariants);
    }

    private static long factorial(int n) {
        long result = 1;
        for (int i = 2; i <= n; i++) {
            result *= i;
        }
        return result;
    }
}
