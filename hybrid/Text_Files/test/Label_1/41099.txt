import sys
input = sys.stdin.readline

n, p = map(int, input().split())
adj = [[] for _ in range(n+1)]
for i in range(n):
    x, y = map(int, input().split())
    adj[x].append(y)
    adj[y].append(x)

ans = 0
for i in range(1, n+1):
    for j in range(i+1, n+1):
        if i in adj[j]:
            continue
        cnt = 0
        for k in adj[i]:
            if k != j and k in adj[j]:
                cnt += 1
        if cnt >= p-1:
            ans += 1
print(ans)
