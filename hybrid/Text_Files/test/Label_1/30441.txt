import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] v = new int[n];
        for (int i = 0; i < n; i++)
            v[i] = in.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++)
            w[i] = in.nextInt();
        in.close();
        int[] dp = new int[n];
        for (int i = 0; i < n; i++) {
            dp[i] = v[i];
            for (int j = 0; j < i; j++) {
                if (dp[j] + v[i] > dp[i] && w[i] - w[j] <= 1 && 2 * w[i] - w[j] - w[j + 1] >= 0)
                    dp[i] = dp[j] + v[i];
            }
        }
        System.out.println(Arrays.stream(dp).max().getAsInt());
    }
}
