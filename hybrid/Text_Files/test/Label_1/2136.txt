#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s, sc;
  cin >> sc;
  for (int d = n; d >= 1; --d) {
    if (n % d == 0) {
      for (int j = d - 1; j >= 0; --j) s += sc[j];
      sc = s;
      s = "";
    }
  }
  cout << sc << endl;
  return 0;
}
