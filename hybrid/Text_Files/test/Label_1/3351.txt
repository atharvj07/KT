#include <vector>
#include <algorithm>
#include <iostream>

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> v(n);
    for (int i = 0; i < n; i++) std::cin >> v[i];
    std::sort(v.begin(), v.end());
    int sum = 0;
    for (int i = 0; i < n - k; i++) sum += v[i + k] - v[i];
    std::cout << sum << '\n';
}
