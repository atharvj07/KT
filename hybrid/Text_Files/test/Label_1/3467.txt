#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int n;
bool vis[1000001];
int f[1000001];
typedef long long ll;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  cin >> n;
  int l, r;
  cin >> l >> r;
  f[0] = 1;
  for (int i = 1; i <= 1000000; i++) {
    f[i] = (f[i - 1] + f[i - 1]) % mod;
  }
  for (int i = 1; i <= n; i++) {
    vis[n % i] = 1;
    n /= i;
  }
  ll ans = 0;
  for (int i = l; i <= r; i++) {
    if (vis[i]) { ans = (ans + f[i - 1]) % mod; }
  }
  cout << ans << '\n';
}
