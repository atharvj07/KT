def solve():
    mod = "RGBRGBRGB"

    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        s = input()
        i = k - 1
        j = (n - k) % len(mod)
        ans = 0
        cnt = {}
        for c in s[j:] + s[:j]:
            if c == mod[i]:
                cnt[c] = cnt.get(c, 0) + 1
                i -= 1
                if i < 0:
                    i = len(mod) - 1
                    ans = cnt.get(mod[i], 0)
            else:
                cnt[c] = cnt.get(c, 0) + 1
                ans = min(ans + 1, cnt.get(mod[i], 0))
        print(ans)

solve()
