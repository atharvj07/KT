#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<char> v(n);
  for (int i = 0; i < n; i++) { cin >> v[i]; }
  int d = 0, r = 0, turn = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (v[i] == 'D') {
      if (turn % 2 == 0) d++;
    } else {
      if (turn % 2 != 0) r++;
    }
    if (turn % 2 == 0 && d > r) {
      v[i] = '-';
    } else if (turn % 2 != 0 && r > d) {
      v[i] = '-';
    }
    turn++;
  }
  if (d > r) {
    cout << 'D';
  } else {
    cout << 'R';
  }
  return 0;
}
