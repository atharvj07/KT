import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int d = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (sum + a[i] <= k) {
                sum += a[i];
                d++;
            } else {
                break;
            }
        }
        System.out.println(d);
    }
}
