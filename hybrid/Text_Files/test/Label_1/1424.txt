#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    if(n < 8) cout << -1 << endl;
    else
    {
        int gcd = 210;
        for(int i = 1;;i++)
        {
            if(gcd%n == 0)
            {
                cout << gcd << endl;
                break;
            }
            gcd+=210;
        }
    }
    return 0;
}
