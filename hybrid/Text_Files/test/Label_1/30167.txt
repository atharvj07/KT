import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        int c = Integer.parseInt(st.nextToken());
        int[] x = new int[n];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            x[i] = Integer.parseInt(st.nextToken());
        }
        double[] p = new double[n - 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n - 1; i++) {
            p[i] = Integer.parseInt(st.nextToken()) * 1.0 / 100;
        }
        int[][] intervals = new int[m][2];
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            intervals[i][0] = Integer.parseInt(st.nextToken()) - 1;
            intervals[i][1] = Integer.parseInt(st.nextToken()) - 1;
        }
        double[][] dp = new double[n][2];
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 0; j < 2; j++) {
                double curFine = 0.0;
                if (j == 1) {
                    curFine = c * p[i];
                }
                double ifInspect = dp[i + 1][1] + curFine;
                double ifNotInspect = dp[i + 1][0];
                for (int[] interval : intervals) {
                    if (interval[0] <= i && i < interval[1]) {
                        ifNotInspect += x[interval[1]] - x[i];
                    }
                }
                dp[i][j] = Math.max(ifInspect, ifNotInspect) / 2.0;
            }
        }
        System.out.printf("%.9f", dp[0][0]);
    }
}
