import java.util.Scanner;

public class Main {
  static final int MOD = 1 << 32;
  static Scanner sc;
  static long n, a, b, c, d, ans;

  public static void main(String[] args) {
    sc = new Scanner(System.in);
    input();
    solve();
    output();
  }

  static void input() {
    n = sc.nextLong();
    a = sc.nextInt();
    b = sc.nextInt();
    c = sc.nextInt();
    d = sc.nextInt();
  }

  static void solve() {
    long now = 0, tmp = 0, sq = 0, cube = 0;
    long step = n / 4;

    tmp = (a * n + step * (2 * a + b * 3)) % MOD * (n + step) % MOD;
    ans = (ans + tmp) % MOD;

    sq = (b * n + step * (2 * b + c * 3)) % MOD * (n + step) % MOD;
    cube = (c * n + step * (2 * c + d * 3)) % MOD * (n + step) % MOD;
    tmp = (sq + cube) % MOD * (n / 2) % MOD;
    ans = (ans + tmp) % MOD;

    now = ((cube + d * n) % MOD * (n + 1)) % MOD;
    tmp = now * step % MOD;
    ans = (ans + tmp) % MOD;

    tmp = (now + (n - step) * (cube + d * n) % MOD) % MOD * (n - step) % MOD;
    tmp = tmp * step % MOD;
    ans = (ans + tmp) % MOD;

    ans = ans * (n / 4) % MOD;
    n %= 4;

    if (n == 0)
      return;
    now = 0;
    for (int i = 1; i <= n; i++) {
      long exlog_i =
          (a * i * i * i % MOD + b * i * i % MOD + c * i % MOD + d % MOD) % MOD * i % MOD;
      now = (now + exlog_i) % MOD;
    }
    ans = (ans + now) % MOD;
  }

  static void output() {
    System.out.printf("%d\n", ans);
  }
}
