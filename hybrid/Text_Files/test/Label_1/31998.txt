import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt();
        int[] arr = new int[n];
        String s = sc.next();
        for (int i = 0; i < n; i++) {
            arr[i] = s.charAt(i) - '0';
        }
        int ans = Integer.MAX_VALUE;
        if (arr[a - 1] == arr[b - 1]) {
            ans = 0;
        } else {
            for (int i = 0; i < n; i++) {
                if (arr[i] == arr[a - 1]) {
                    ans = Math.min(ans, Math.abs(i - (b - 1)));
                } else if (arr[i] == arr[b - 1]) {
                    ans = Math.min(ans, Math.abs(i - (a - 1)));
                }
            }
        }
        System.out.println(ans);
    }
}
