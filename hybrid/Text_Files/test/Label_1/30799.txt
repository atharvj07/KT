import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int l = i, r = i + 2, xor = a[l], sum = 0;
            while (r <= n) {
                sum += a[r - 1];
                if (xor == sum) {
                    ans++;
                }
                xor ^= a[r - 1];
                r++;
            }
        }
        System.out.println(ans);
    }
}
