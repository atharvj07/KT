#include <bits/stdc++.h>
using namespace std;

int main() {
  int m, n;
  cin >> m >> n;
  vector<vector<int>> t(m, vector<int>(n));
  for (int i = 0; i < m; i++)
    for (int j = 0; j < n; j++) cin >> t[i][j];
  vector<int> ans(m);
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
      if (j > 0) ans[i] += t[i][j - 1];
      ans[i] += t[i][j];
    }
  }
  for (int i = 0; i < m; i++) cout << ans[i] << " ";
  return 0;
}
