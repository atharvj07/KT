import java.util.Arrays;
import java.util.Scanner;

class Main {
    private static final int NOT_VISITED = -1;
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), m = scanner.nextInt(), ans = 0;
        int[] degree = new int[n + 1];
        int[][] adj = new int[n + 1][];

        while (m-- != 0) {
            int u = scanner.nextInt(), v = scanner.nextInt();
            degree[u]++;
            degree[v]++;
            if (adj[u] == null) {
                adj[u] = new int[] { v };
            } else {
                adj[u] = Arrays.copyOf(adj[u], adj[u].length + 1);
                adj[u][adj[u].length - 1] = v;
            }
            if (adj[v] == null) {
                adj[v] = new int[] { u };
            } else {
                adj[v] = Arrays.copyOf(adj[v], adj[v].length + 1);
                adj[v][adj[v].length - 1] = u;
            }
        }

        int[] counter = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            if (degree[i] >= 2 && degree[i] % 2 == 0) {
                Arrays.fill(counter, 0);
                int odd = 0, start = -1;
                for (int v : adj[i]) {
                    if (++counter[v] == 1) odd++;
                    if (odd > 2) break;
                    start = v;
                }
                if (odd == 2 && dfs(start, i, counter, degree, adj)) {
                    ans++;
                }
            }
        }

        System.out.println(ans);
    }

    private static boolean dfs(int u, int p, int[] counter, int[] degree, int[][] adj) {
        if (counter[u] != 1) return false;
        if (degree[u] == 2 && u != p) return true;
        for (int v : adj[u]) {
            if (v == p) continue;
            if (counter[v]++ == 1) {
                if (!dfs(v, u, counter, degree, adj)) {
                    return false;
                }
            }
        }
        return counter[u] == degree[u];
    }
}
