#include <bits/stdc++.h>
using namespace std;
#define nl "\n"
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n), sorted(a);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    sort(sorted.rbegin(), sorted.rend());
    bool impossible = false;
    for (int i = 0; i < n; i++) {
      if (a[i] != sorted[i]) {
        if (i % 2 == (n % 2)) {
          impossible = true;
        }
      }
    }
    if (impossible) {
      cout << -1 << nl;
      continue;
    }
    int m = (n - 1) / 2;
    cout << m << nl;
    for (int i = 3; i <= n; i += 2) {
      cout << i << " ";
    }
    cout << nl;
  }
  return 0;
}
