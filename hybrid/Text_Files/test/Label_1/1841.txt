#include <iostream>

using namespace std;

const int N = 100005;
pair<int, int> a[N];
int n, p, q;

int main() {
    cin >> n >> p >> q;
    for (int i = 1; i <= n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    double l = 0.0, r = min(p, q) * 1.0 / min(a[1].first, a[1].second), mid;
    for (int iter = 1; iter <= 200; iter++) {
        mid = (l + r) / 2.0;
        double x = 0.0, y = 0.0;
        for (int i = 1; i <= n; i++) {
            if (a[i].first * mid >= a[i].second * mid) {
                if (x + a[i].first * mid >= p) {
                    if (x + a[i].first * mid - p > 1e-9) {
                        break;
                    }
                    y += (q - y) / (a[i].second * mid + 1e-9);
                    break;
                } else {
                    x += a[i].first * mid;
                    y += a[i].second * mid;
                }
            } else {
                if (y + a[i].second * mid >= q) {
                    if (y + a[i].second * mid - q > 1e-9) {
                        break;
                    }
                    x += (p - x) / (a[i].first * mid + 1e-9);
                    break;
                } else {
                    x += a[i].first * mid;
                    y += a[i].second * mid;
                }
            }
        }
        if (x >= p && y >= q) {
            r = mid;
        } else {
            l = mid;
        }
    }
    cout << r;
}
