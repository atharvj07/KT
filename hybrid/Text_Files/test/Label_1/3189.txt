#include<bits/stdc++.h>
using namespace std;
int main(){
    string s, res="";
    int n;
    cin>>n>>s;
    for(int i=0; i<n;){
        if(i+2<n && s[i]=='o' && s[i+1]=='g' && s[i+2]=='o'){
            res+="***";
            i+=3;
        }
        else{
            res+=s[i++];
        }
    }
    cout<<res<<endl;
    return 0;
}
