#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
  int nx, bx;
  cin >> nx >> bx;
  vector<int> x(nx);
  for (int& xi : x) cin >> xi;

  int ny, by;
  cin >> ny >> by;
  vector<int> y(ny);
  for (int& yi : y) cin >> yi;

  long long X = 0, Y = 0;
  for (int i = nx - 1, p = 1; i >= 0; i--, p *= bx) {
    X += (long long)x[i] * p;
  }
  for (int i = ny - 1, p = 1; i >= 0; i--, p *= by) {
    Y += (long long)y[i] * p;
  }

  if (X < Y) cout << '<';
  else if (X > Y) cout << '>';
  else cout << '=';
}
