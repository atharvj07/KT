from collections import defaultdict

def build_tree(n, edges):
    tree = defaultdict(list)
    for edge in edges:
        a, b = edge
        tree[a].append(b)
        tree[b].append(a)
    return tree

def dfs(tree, node, parent, depth, depths):
    depths[node] = depth
    for child in tree[node]:
        if child != parent:
            dfs(tree, child, node, depth + 1, depths)

def calculate_expectation(n, edges):
    tree = build_tree(n, edges)
    depths = [0] * (n + 1)
    dfs(tree, 1, -1, 1, depths)
    expectation = sum(1 / depths[i] for i in range(1, n + 1))
    return expectation

n = int(input())
edges = [list(map(int, input().split())) for _ in range(n - 1)]
print(f'{calculate_expectation(n, edges):.20f}')
