#include <bits/stdc++.h>
using namespace std;
#define ll long long int
int main() {
    ll n;
    cin >> n;
    vector<ll> v(n + 1);
    for (ll i = n; i >= 1; i--) {
        ll x;
        cin >> x;
        v[x] = i;
    }
    ll mx = 0;
    ll pos = 0;
    for (ll i = 1; i <= n; i++) {
        if (v[i] > mx) {
            mx = v[i];
            pos = i;
        }
    }
    cout << pos;
}
