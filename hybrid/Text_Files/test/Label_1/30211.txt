import java.util.Scanner;

public class ArrayOperations {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[][] operations = new int[m][3];
        for (int i = 0; i < m; i++) {
            operations[i][0] = scanner.nextInt() - 1;
            operations[i][1] = scanner.nextInt() - 1;
            operations[i][2] = scanner.nextInt();
        }

        int[][] queries = new int[k][2];
        for (int i = 0; i < k; i++) {
            queries[i][0] = scanner.nextInt() - 1;
            queries[i][1] = scanner.nextInt() - 1;
        }

        for (int[] query : queries) {
            int xi = query[0];
            int yi = query[1];
            for (int i = xi; i <= yi; i++) {
                int li = operations[i][0];
                int ri = operations[i][1];
                int di = operations[i][2];
                for (int j = li; j <= ri; j++) {
                    a[j] += di;
                }
            }
        }

        for (int i : a) {
            System.out.print(i + " ");
        }
    }
}
