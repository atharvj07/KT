import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
# prefix gcd
p = [0] * n
p[0] = a[0]
for i in range(1, n):
    p[i] = a[i]
    p[i] = p[i] // __gcd(p[i], p[i - 1])
# suffix gcd
s = [0] * n
s[n - 1] = a[n - 1]
for i in range(n - 2, -1, -1):
    s[i] = a[i]
    s[i] = s[i] // __gcd(s[i], s[i + 1])
# find maximum pair interval
max_int = 0
min_l = 0
for i in range(0, n):
    l, r = i, i
    while l >= 0 and r < n and p[l] == s[r]:
        if r - l + 1 > max_int:
            max_int = r - l + 1
            min_l = l
        l -= 1
        r += 1
print(max_int, min_l)
for i in range(min_l, min_l + max_int):
    print(i + 1, end=" ")
