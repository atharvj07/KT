import java.util.*;

public class Main {
    private static final Set<Character> values = new HashSet<>(Arrays.asList('2', '3', '4', '5', '6', '7', '8', '9', 'T', 'J', 'Q', 'K', 'A'));
    private static final Set<Character> suits = new HashSet<>(Arrays.asList('S', 'D', 'H', 'C'));

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String[] cards = sc.nextLine().split(" ");

        Map<String, Integer> cardCount = new HashMap<>();
        for (String card : cards) {
            cardCount.put(card, cardCount.getOrDefault(card, 0) + 1);
        }

        List<String> uniqueCards = new ArrayList<>(cardCount.keySet());
        Collections.sort(uniqueCards, (a, b) -> cardCount.get(b) - cardCount.get(a));

        boolean possible = true;
        while (!uniqueCards.isEmpty()) {
            String topCard = uniqueCards.get(0);
            uniqueCards.remove(0);

            if (cardCount.get(topCard) > 1) {
                cardCount.put(topCard, cardCount.get(topCard) - 2);
            } else {
                char topCardValue = topCard.charAt(0);
                char topCardSuit = topCard.charAt(1);

                boolean foundMatch = false;
                for (int i = 1; i < uniqueCards.size(); i++) {
                    String otherCard = uniqueCards.get(i);
                    char otherCardValue = otherCard.charAt(0);
                    char otherCardSuit = otherCard.charAt(1);

                    if (topCardValue == otherCardValue || topCardSuit == otherCardSuit) {
                        cardCount.put(otherCard, cardCount.get(otherCard) - 1);
                        foundMatch = true;
                        break;
                    }
                }

                if (!foundMatch) {
                    possible = false;
                    break;
                }
            }
        }

        System.out.println(possible ? "YES" : "NO");
    }
}
