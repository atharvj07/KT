from collections import defaultdict

n = int(input())
color = list(map(int, input().split()))
par = list(range(n+1))
sz = [1] * (n+1)
def find(x):
    if par[x] != x:
        par[x] = find(par[x])
    return par[x]
def join(a, b):
    a = find(a)
    b = find(b)
    if a != b:
        if sz[a] > sz[b]:
            par[b] = a
            sz[a] += sz[b]
        else:
            par[a] = b
            sz[b] += sz[a]
graph = defaultdict(list)
for i in range(n-1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)
cnt = [0, 0]
def dfs(u, p):
    cnt[color[u-1]] += 1
    for v in graph[u]:
        if v != p:
            dfs(v, u)
            join(u, v)
dfs(1, 1)
mn = min(cnt)
print(n - 2*mn)
