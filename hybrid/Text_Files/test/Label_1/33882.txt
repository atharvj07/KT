import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int tests = input.nextInt();
        while(tests-- > 0) {
            int n = input.nextInt();
            if(n == 1) System.out.println(1);
            else System.out.println(n * 2 - 1);
        }
    }
}
