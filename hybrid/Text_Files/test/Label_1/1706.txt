#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n * n);
    for (int i = 0; i < n * n; i++) cin >> a[i];

    map<int, vector<int>> g;
    for (int i = 0; i < n * n; i++) g[a[i]].push_back(i);

    vector<int> ans(n, -1);
    for (auto &p : g) {
        if (p.second.size() == n) {
            for (int i = 0; i < n; i++) ans[i] = p.first;
        }
    }

    for (int i = 0; i < n * n; i++) {
        int x = i / n, y = i % n;
        if (ans[x] != -1 && ans[y] != -1) continue;
        if (ans[x] == -1) ans[x] = a[i] / ans[y];
        else ans[y] = a[i] / ans[x];
    }

    for (auto &x : ans) cout << x << " ";
    cout << endl;

    return 0;
}
