import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        java.util.Scanner stdin = new java.util.Scanner(System.in);
        while (true) {
            int n = stdin.nextInt();
            if (n == 0) break;
            int[] a = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = stdin.nextInt();
            }
            Arrays.sort(a);
            System.out.println(Math.abs(a[1] - a[0]));
        }
    }
}
