import sys
input = sys.stdin.readline
n = int(input())
a = list(map(int, input().split()))
dp = [1] * n
ans = [0] * n
for i in range(1, n):
    for j in range(i):
        if a[i] == a[j] + 1:
            dp[i] = max(dp[i], dp[j] + 1)
            ans[i] = j
k = max(dp)
print(k)
lst = []
while k > 0:
    k -= 1
    lst += [a[ans[k]]]
print(*lst)
