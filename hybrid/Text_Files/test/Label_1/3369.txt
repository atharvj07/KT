#include <iostream>
#include <vector>

int main() {
  long long m;
  std::cin >> m;
  long long num = 0, vol = 0;
  for (long long i = 1; i * i * i <= m; i++) {
    const long long v = i * i * i;
    if (vol + v <= m) {
      vol += v;
      num += 1;
    } else
      break;
  }
  std::cout << num << " " << vol << std::endl;
  return 0;
}
