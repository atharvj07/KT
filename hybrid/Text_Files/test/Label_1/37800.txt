from bisect import bisect_right
n, q = map(int, input().split())
a = [int(x) for x in input().split()]
aa = sorted(a, reverse=True)
for t, x in [map(int, input().split()) for _ in range(q)]:
    if t == 1:
        a[x - 1] ^= 1
        aa[bisect_right(aa, a[x - 1])] = a[x - 1]
    else:
        print(aa[x - 1])
