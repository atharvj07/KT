import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws IOException{
        List<Pair> list = new ArrayList<>();
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            list.add(new Pair(Integer.parseInt(s[0]), Integer.parseInt(s[1]), s[2]));
        }
        double min = Double.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double time = cal(list.get(i), list.get(j));
                if (time != -1) {
                    min = Math.min(min, time);
                }
            }
        }
        if (min != Double.MAX_VALUE) {
            System.out.println(min);
        } else {
            System.out.println("SAFE");
        }
    }
    public static double cal(Pair p1, Pair p2) {
        if (p1.dir.equals(p2.dir)) {
            return -1;
        }
        double x = p1.x - p2.x;
        double y = p1.y - p2.y;
        if (p1.dir.equals("U") && p2.dir.equals("D")) {
            if (x == 0) {
                return y / 0.2;
            }
        } else if (p1.dir.equals("D") && p2.dir.equals("U")) {
            if (x == 0) {
                return y / 0.2;
            }
        } else if (p1.dir.equals("L") && p2.dir.equals("R")) {
            if (y == 0) {
                return x / 0.2;
            }
        } else if (p1.dir.equals("R") && p2.dir.equals("L")) {
            if (y == 0) {
                return x / 0.2;
            }
        }
        return -1;
    }
    static class Pair {
        int x, y;
        String dir;
        Pair(int x, int y, String dir) {
            this.x = x;
            this.y = y;
            this.dir = dir;
        }
    }
}
