#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, a[52];
    cin >> n >> k;
    for (int i = 0; i < n; ++i) cin >> a[i];
    sort(a, a + n);
    int ans = 0;
    for (int i = 0; i < n - k;) {
        int j = i;
        while (a[j] == a[i]) j++;
        if (j == i) {
            ans++;
            a[i] = (a[i] + 1) / 2;
            sort(a, a + n);
        } else
            i = j;
    }
    cout << ans << endl;
}
