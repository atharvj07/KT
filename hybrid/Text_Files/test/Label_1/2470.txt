#include<bits/stdc++.h>
using namespace std;
int main() {
    int t; cin >> t;
    while (t--) {
        int n, m; cin >> n >> m;
        int grid[n][m];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                cin >> grid[i][j];
                cnt += grid[i][j];
            }
        }
        cout << (((cnt + (n % 2 == m % 2)) % 2) ? "Vivek" : "Ashish") << endl;
    }
    return 0;
}
