import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = scanner.nextInt();
        }
        int mini, maxi;
        for (int i = 0; i < n; i++) {
            mini = Math.abs(x[i] - x[(i - 1 + n) % n]);
            maxi = Math.abs(x[i] - x[(i + 1) % n]);
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                mini = Math.min(mini, Math.abs(x[i] - x[j]));
                maxi = Math.max(maxi, Math.abs(x[i] - x[j]));
            }
            System.out.println(mini + " " + maxi);
        }
    }
}
