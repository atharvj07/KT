#include <bits/stdc++.h>
using namespace std;

const int dx[] = {0, 1}, dy[] = {1, 0};
const int INF = 1e9;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<bool>> vis(n + 1, vector<bool>(n + 1, 0));
    vector<pair<int, int>> v(m);
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        vis[x][y] = 1;
        v[i] = {x, y};
    }
    queue<pair<int, int>> q;
    q.push({1, 1});
    vector<vector<int>> dist(n + 1, vector<int>(n + 1, INF));
    dist[1][1] = 0;
    while (q.size()) {
        pair<int, int> u = q.front();
        q.pop();
        for (int i = 0; i < 2; ++i) {
            int x = u.first + dx[i], y = u.second + dy[i];
            if (1 <= x && x <= n && 1 <= y && y <= n && !vis[x][y] && dist[u.first][u.second] + 1 < dist[x][y]) {
                dist[x][y] = dist[u.first][u.second] + 1;
                q.push({x, y});
            }
        }
    }
    if (dist[n][n] < INF) cout << dist[n][n] << endl;
    else cout << -1 << endl;
    return 0;
}
