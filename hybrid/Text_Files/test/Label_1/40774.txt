n=int(input())
adj=[[] for i in range(n+1)]
for i in range(n):
    a,b=map(int,input().split())
    if a: adj[a].append(i+1)
    if b: adj[i+1].append(b)
q=[1]
used=[0]*(n+1)
while q:
    x=q.pop()
    print(x,end=' ')
    used[x]=1
    for y in adj[x]:
        if not used[y]:
            q.append(y)
