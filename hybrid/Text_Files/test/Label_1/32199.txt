import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        if (sum % 2 == 0) {
            for (int i = 0; i <= sum / 2; i++) {
                System.out.print(i + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
