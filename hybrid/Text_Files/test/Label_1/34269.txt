import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        int cnt = 0;
        for (long x: a) {
            if ((x & 1) == 1) {
                cnt++;
            }
        }
        long ans = 0;
        if (cnt == 0) {
            ans = a[n - 1];
        } else {
            ans = a[n - 1] + 1;
        }
        System.out.println(ans);
    }
}
