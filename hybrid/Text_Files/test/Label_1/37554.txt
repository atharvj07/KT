import sys, math

input = sys.stdin.readline

n, m = map(int, input().split())
x = list(map(int, input().split()))
p = list(map(int, input().split()))

for j in range(m):
    l = math.ceil((x[0] - 1) / p[j])
    if all((i - 1 + p[j] * l) % p[j] == 0 for i in x):
        print("YES")
        print(x[0] - 1 + p[j] * l, j + 1)
        exit()

print("NO")
