import sys

def fast_pow(base, p):
    if p == 0:
        return 1

    if p == 1:
        return base

    val = fast_pow(base, int(p / 2))
    val = (val * val) % (10 ** 9 + 7)

    if p % 2 != 0:
        val = (val * base) % (10 ** 9 + 7)

    return val % (10 ** 9 + 7)


def count_maximum_vasya_distance_strings(s: str) -> int:
    MOD = 10 ** 9 + 7
    n = len(s)
    cnt = [0] * 4
    for i in s:
        if i == 'A':
            cnt[0] += 1
        elif i == 'C':
            cnt[1] += 1
        elif i == 'G':
            cnt[2] += 1
        else:
            cnt[3] += 1

    ans = 1
    for i in range(4):
        ans = (ans * fast_pow(i + 1, cnt[i])) % MOD

    return ans


s = sys.stdin.readline().strip()
print(count_maximum_vasya_distance_strings(s))
