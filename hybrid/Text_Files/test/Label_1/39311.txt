n = int(input())
grid = [list(map(int,input().split())) for i in range(n)]
flag = True
for i in range(n):
    for j in range(n):
        if grid[i][j] == 1: continue
        check = False
        for ii in range(n):
            if grid[ii][j] != 1 and ii != i:
                temp = grid[i][j] - grid[ii][j]
                if temp > 0 and temp in grid[i]:
                    check = True
                    break
        if check == False:
            for jj in range(n):
                if grid[i][jj] != 1 and jj != j:
                    temp = grid[i][j] - grid[i][jj]
                    if temp > 0 and temp in grid:
                        check = True
                        break
        if check == False:
            flag = False
            break
    if flag == False: break
if flag:
    print("Yes")
else:
    print("No")
