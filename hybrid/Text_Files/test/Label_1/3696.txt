#include <bits/stdc++.h>

using namespace std;

const int MAXN = 55;
const int INF = 0x3f3f3f3f;

int n, k;
char grid[MAXN][MAXN];
pair<int, int> car[MAXN][MAXN];

int main() {
    cin >> n >> k;
    for (int i = 1; i <= 4; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> grid[i][j];
            if (grid[i][j] != '0') {
                car[grid[i][j]][i] = {j, 1};
            }
        }
    }
    for (int i = 1; i <= k; i++) {
        int u = car[i][1].first, v = car[i][1].second, u2 = car[i][2].first, v2 = car[i][2].second;
        if (v2 == 1 && u2 == u) {
            continue;
        }
        if (u == u2) {
            cout << "-1" << endl;
            return 0;
        }
        car[grid[v2][u2]][v2] = {u2, v2};
        car[grid[v][u]][v] = {INF, INF};
        grid[v][u] = '0';
        grid[v2][u2] = grid[v][u];
        cout << i << " " << v2 << " " << u2 << endl;
        u = u2, v = v2;
    }
    cout << "0" << endl;
    return 0;
}
