from collections import deque
n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
q = deque([(i, j) for i in range(1, n + 1) for j in range(1, m + 1) if a[i - 1][j - 1]])
ans = 0
while q:
    ans += 1
    i, j = q.popleft()
    for x in range(i, n + 1):
        for y in range(1, m + 1):
            if a[x - 1][y - 1] == 1 and (x < i or y > j):
                q.append((x, y))
print(ans)
