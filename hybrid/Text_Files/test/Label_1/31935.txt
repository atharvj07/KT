import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int queries = input.nextInt();
        for (int i = 0; i < queries; i++) {
            int x = input.nextInt();
            int p = input.nextInt();
            int k = input.nextInt();
            int y = x + 1;
            while (true) {
                if (gcd(p, y) == 1) {
                    k--;
                    if (k == 0) {
                        System.out.println(y);
                        break;
                    }
                }
                y++;
            }
        }
        input.close();
    }
    private static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
