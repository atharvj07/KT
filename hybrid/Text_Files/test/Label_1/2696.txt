#include <bits/stdc++.h>

using namespace std;

int n, m, q;
vector<vector<int>> grid;
vector<int> pref1, suf1;

int query(int x) {
  if (pref1[x] == x) return x;
  return pref1[x] = query(pref1[x]);
}

void unite(int x, int y) {
  x = query(x), y = query(y);
  pref1[x] = y;
}

int main() {
  cin >> n >> m >> q;
  grid.resize(n, vector<int>(m));
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> grid[i][j];
    }
  }
  pref1.resize(n * m);
  iota(pref1.begin(), pref1.end(), 0);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      if(i > 0 && grid[i][j] == grid[i-1][j]) unite(i*m + j, (i-1)*m + j);
      if(j > 0 && grid[i][j] == grid[i][j-1]) unite(i*m + j, i*m + j-1);
    }
  }
  suf1.resize(n * m);
  iota(suf1.rbegin(), suf1.rend(), 0);
  for (int i = n - 1; i >= 0; i--) {
    for (int j = m - 1; j >= 0; j--) {
      if(i < n-1 && grid[i][j] == grid[i+1][j]) unite(i*m + j, (i+1)*m + j);
      if(j < m-1 && grid[i][j] == grid[i][j+1]) unite(i*m + j, i*m + j+1);
    }
  }
  while (q--) {
    int x, y;
    cin >> x >> y;
    x--, y--;
    int p = query(x * m + y);
    int lo = 0, hi = min(n, m);
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (query((x - mid + n) % n * m + y) == p) lo = mid;
      else hi = mid - 1;
    }
    lo = 0, hi = min(n, m);
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (query((x + mid) % n * m + y) == p) lo = mid;
      else hi = mid - 1;
    }
    lo = 0, hi = min(n, m);
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (query(x * m + (y - mid + m) % m) == p) lo = mid;
      else hi = mid - 1;
    }
    lo = 0, hi = min(n, m);
    while (lo < hi) {
      int mid = (lo + hi + 1) / 2;
      if (query(x * m + (y + mid) % m) == p) lo = mid;
      else hi = mid - 1;
    }
    cout << lo << '\n';
  }
  return 0;
}
