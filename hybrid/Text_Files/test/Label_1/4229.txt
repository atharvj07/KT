#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;

int main() {
	ll n, m, a;
	cin >> n >> m;
	bool b[m];
	memset(b, false, sizeof(b));
	b[0] = true;
	b[(m - 1) % m] = true;
	ll sum = 0;
	for (ll i = 0; i < n; i++) {
		cin >> a;
		sum = (sum + a) % m;
		b[sum] = true;
		if (b[0]) {
			cout << "YES" << endl;
			return 0;
		}
	}
	cout << "NO" << endl;
	return 0;
}
