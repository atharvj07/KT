from math import ceil

for _ in range(int(input())):
    a,b,x,y,n = map(int,input().split())
    ans = a * b
    if n == 0:
        pass
    elif n == 1:
        if a-b > x-y:
            ans -= b-y
        else:
            ans -= a-x
    else:
        if a-x <= y-b:
            ans -= (a-x)*(n//2)
            rem = n % 2
            if rem:
                ans -= min(a-x,y-b)
        else:
            ans -= (y-b)*(n//2)
            rem = n % 2
            if rem:
                ans -= min(a-x,y-b)
    print(ans)
