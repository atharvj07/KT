s = input().strip()
x, y = map(int, input().split())

# Count occurrences of 0s and 1s, and question marks
count_0 = count_1 = count_question = 0
for char in s:
    if char == '0':
        count_0 += 1
    elif char == '1':
        count_1 += 1
    else:
        count_question += 1

# Calculate initial angry comments without considering question marks
angry_comments = x*count_0*count_1 + y*count_0*count_1

# Calculate minimum angry comments by considering each question mark replacement scenario
min_angry_comments = float('inf')
for replace_0 in range(count_question + 1):
    replace_1 = count_question - replace_0
    current_angry_comments = x * (count_1 + replace_1) * count_0 + y * count_1 * (count_0 + replace_0)
    min_angry_comments = min(min_angry_comments, current_angry_comments)

print(min_angry_comments)
