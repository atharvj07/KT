import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int t = scanner.nextInt();
            if (t == 0) break;
            int[][] data = new int[12][2];
            for (int i = 0; i < 12; i++) {
                data[i][0] = scanner.nextInt();
                data[i][1] = scanner.nextInt();
            }
            int sum = 0;
            int cnt = 0;
            for (int[] d : data) {
                sum += d[0] - d[1];
                cnt++;
                if (sum >= t) break;
            }
            if (sum < t) cnt = -1;
            System.out.println(cnt);
        }
        scanner.close();
    }
}
