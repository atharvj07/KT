import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.next();
        int k = input.nextInt();
        Map<Character, Character> map = new HashMap<>();
        for (int i = 0; i < k; i++) {
            String str = input.next();
            map.put(str.charAt(0), str.charAt(1));
            map.put(str.charAt(1), str.charAt(0));
        }
        int n = s.length();
        int ans = 0;
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (vis[i])
                continue;
            char ch = s.charAt(i);
            if (map.containsKey(ch)) {
                char next = map.get(ch);
                int j = i + 1;
                while (j < n && s.charAt(j) != next) {
                    vis[j] = true;
                    ans++;
                    j++;
                }
                if (j == n)
                    vis[i] = true;
                else
                    vis[j] = true;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
