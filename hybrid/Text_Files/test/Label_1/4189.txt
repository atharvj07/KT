#include<bits/stdc++.h>
using namespace std;
bool ok(string s) {
  int n = s.size();
  if(n <= 2) return false;
  for(int i = 1; i < n - 1; i++) {
    if(s[i] != s[i + 1] && s[i] == s[i - 1]) return true;
  }
  return false;
}
int main() {
  int n;
  string s;
  cin >> n >> s;
  cout << (ok("?" + s) || ok(s + "?") ? "Yes" : "No") << endl;
  return 0;
}
