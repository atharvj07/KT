#include <iostream>
#include <vector>

using namespace std;

int main() {
  int q;
  cin >> q;
  while (q--) {
    long long a, b, c;
    cin >> a >> b >> c;
    long long sum = a + b + c;
    if (sum % 2 == 1) {
      cout << (sum - 1) / 2 << endl;
    } else {
      cout << sum / 2 << endl;
    }
  }
  return 0;
}
