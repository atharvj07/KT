#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  if (s == t) {
    cout << 0 << '\n';
    return 0;
  }
  int cnt = 0;
  vector<int> ans;
  for (int i = 0; i < n; ++i) {
    if (s[i] != t[i]) {
      int j = i + 1;
      while (j < n && s[j] != t[i]) ++j;
      if (j == n) {
        cout << -1 << '\n';
        return 0;
      }
      ans.push_back(j);
      swap(s[i], s[j]);
      ++cnt;
    }
  }
  cout << cnt << '\n';
  for (int x : ans) cout << x << ' ';
  return 0;
}
