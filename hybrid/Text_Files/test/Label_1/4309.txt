#include <bits/stdc++.h>
#define int int64_t
const int mod = 1e9 + 7;
using namespace std;
int m, a[200001], b[200001], ans;
char s[200001];
int32_t main() {
  cin >> m;
  cin >> s + 1;
  for (int i = 1; i <= m; i++) {
    if (i > 2) {
      (a[i] = a[i - 3] * 3) %= mod;
    }
    if (s[i] == '?') {
      (a[i] += a[i - 1] * 3 + 1) %= mod;
      (b[i] += b[i - 1] + a[i - 1] + 1) %= mod;
    } else {
      if (s[i] == 'a') (a[i] += a[i - 1]) %= mod;
      if (s[i] == 'b') (b[i] += b[i - 1] + a[i - 1]) %= mod;
      if (s[i] == 'c') (b[i] += b[i - 1]) %= mod;
    }
  }
  cout << b[m] << '\n';
}
