import heapq

n, m, c_l, c_e, v = map(int,input().split())
l = [0]+list(map(int,input().split()))+[m+1]
e = [0]+list(map(int,input().split()))+[m+1]
q = int(input())
for i in range(q):
    x1, y1, x2, y2 = map(int,input().split())
    if y1==y2:
        print(abs(x2-x1))
    else:
        h = [(0,x1,y1)]
        dist = [[10**9 for j in range(m+1)] for i in range(n+1)]
        dist[x1][y1] = 0
        while h:
            t, x, y = heapq.heappop(h)
            if (x,y)==(x2,y2):
                print(t)
                break
            for dx, dy in ((-1,0),(1,0),(0,-1),(0,1)):
                if 1<=x+dx<=n and 1<=y+dy<=m and dist[x+dx][y+dy]>t+1:
                    dist[x+dx][y+dy]=t+1
                    heapq.heappush(h,(t+1,x+dx,y+dy))
            if x>1 and y not in l and dist[x-1][y]>t+1:
                dist[x-1][y]=t+1
                heapq.heappush(h,(t+1,x-1,y))
            if x<n and y not in l and dist[x+1][y]>t+1:
                dist[x+1][y]=t+1
                heapq.heappush(h,(t+1,x+1,y))
            for i in range(1,v+1):
                if x-i>0 and y not in l and dist[x-i][y]>t+1:
                    dist[x-i][y]=t+1
                    heapq.heappush(h,(t+1,x-i,y))
                if x+i<=n and y not in l and dist[x+i][y]>t+1:
                    dist[x+i][y]=t+1
                    heapq.heappush(h,(t+1,x+i,y))
            for i in range(len(e)):
                if e[i]<y and dist[x][e[i]+1]>t+1:
                    dist[x][e[i]+1]=t+1
                    heapq.heappush(h,(t+1,x,e[i]+1))
                if e[i]>y and dist[x][e[i]-1]>t+1:
                    dist[x][e[i]-1]=t+1
                    heapq.heappush(h,(t+1,x,e[i]-1))
