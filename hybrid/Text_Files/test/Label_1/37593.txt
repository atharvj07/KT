def lcp(a, b):
    i = 0
    while i < min(len(a), len(b)) and a[i] == b[i]:
        i += 1
    return i

n = int(input())
students = [input() for _ in range(n)]
pseudonyms = [input() for _ in range(n)]

# Calculate lcp for each possible pair
lcp_values = []
for i, student in enumerate(students):
    for j, pseudonym in enumerate(pseudonyms):
        lcp_values.append((lcp(student, pseudonym), i, j))

lcp_values.sort(reverse=True)

matched_students = set()
matched_pseudonyms = set()
matches = []
total_quality = 0

for value, student_index, pseudonym_index in lcp_values:
    if student_index not in matched_students and pseudonym_index not in matched_pseudonyms:
        matches.append((student_index + 1, pseudonym_index + 1))
        matched_students.add(student_index)
        matched_pseudonyms.add(pseudonym_index)
        total_quality += value

print(total_quality)
for match in matches:
    print(*match)
