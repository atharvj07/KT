import java.math.BigInteger;
import java.io.*;

class Main {
    private static BigInteger pow10 = BigInteger.valueOf(10);
    private static boolean isZigzag(BigInteger num) { // num is greater than 9
        boolean inc = num.remainder(pow10).compareTo(BigInteger.ZERO) < 0;
        BigInteger preDig = num.remainder(pow10);
        num = num.divide(pow10);
        while (num.compareTo(BigInteger.ZERO) > 0) {
            BigInteger dig = num.remainder(pow10);
            if (inc && dig.compareTo(preDig) >= 0) return false;
            if (!inc && dig.compareTo(preDig) <= 0) return false;
            inc = !inc;
            preDig = dig;
            num = num.divide(pow10);
        }
        return true;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BigInteger A = new BigInteger(br.readLine());
        BigInteger B = new BigInteger(br.readLine());
        BigInteger M = new BigInteger(br.readLine());

        BigInteger cnt = BigInteger.ZERO;
        for (BigInteger i = A; i.compareTo(B) <= 0; i = i.add(M)) {
            if (i.compareTo(BigInteger.valueOf(9)) < 0) continue;
            cnt = cnt.add(BigInteger.valueOf(isZigzag(i) ? 1 : 0));
        }
        System.out.println(cnt.remainder(BigInteger.valueOf(10000)));
    }
}
