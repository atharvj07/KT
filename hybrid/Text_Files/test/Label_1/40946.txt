import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
dp = [1] * n

for i in range(1, n):
    if a[i] == a[i - 1] + 1:
        dp[i] = dp[i - 1] + 1

print(max(dp))
for i in range(n):
    if dp[i] == max(dp):
        for j in range(i - dp[i] + 1, i + 1):
            print(j + 1, end=" ")
        print()
        break
