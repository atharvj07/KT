#include <iostream>
typedef long long ll;

int main() {
    ll l1, r1, l2, r2, k;
    std::cin >> l1 >> r1 >> l2 >> r2 >> k;
    if ((l1 <= k && k <= r1) && (l1 <= l2 && l2 <= r1)) { l1 = l2; r2 = r1; }
    if ((l2 <= k && k <= r2) && (l2 <= l1 && l1 <= r2)) { l2 = l1; r1 = r2; }
    if (l1 > r2 || l2 > r1) std::cout << 0;
    else std::cout << std::max(0LL, r1 - l2);
}
