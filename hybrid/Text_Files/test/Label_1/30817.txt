import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), k = in.nextInt();
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int a = in.nextInt(), b = in.nextInt();
            map.computeIfAbsent(a, z -> new ArrayList<>()).add(b);
            map.computeIfAbsent(b, z -> new ArrayList<>()).add(a);
        }
        int ans = 0;
        boolean[] visited = new boolean[n + 1];
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        visited[1] = true;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int next : map.keySet()) {
                if (!visited[next] && map.get(cur).contains(next)) {
                    ans++;
                    visited[next] = true;
                    q.add(next);
                }
            }
        }
        System.out.println(ans == n - 1 ? (long) Math.pow(2, n - 1 - m) % k : 0);
    }
}
