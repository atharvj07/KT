import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int length(int[] p, int[] s, int rows, int cols, int[][] dp) {
        for (int i = 1; i <= rows; i++) {
            dp[i][0] = 0;
        }
        for (int i = 1; i <= cols; i++) {
            dp[0][i] = 0;
        }
        for (int i = 1; i <= rows; i++) {
            for (int j = 1; j <= cols; j++) {
                dp[i][j] = (p[i - 1] == s[j - 1]) ? dp[i - 1][j - 1] + 1 : Math.max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
        return dp[rows][cols];
    }

    static int[] construct(int[] p, int[] s, int rows, int cols, int[][] dp) {
        int[] ans = new int[length(p, s, rows, cols, dp)];
        int i = rows;
        int j = cols;
        int k = length(p, s, rows, cols, dp) - 1;
        while (i != 0 && j != 0) {
            if (p[i - 1] == s[j - 1]) {
                ans[k--] = p[i - 1];
                i--;
                j--;
            } else {
                if (dp[i - 1][j] > dp[i][j - 1]) {
                    i--;
                } else {
                    j--;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) p[i] = scanner.nextInt();
        int m = scanner.nextInt();
        int[] s = new int[m];
        for (int i = 0; i < m; i++) s[i] = scanner.nextInt();
        int[][] dp = new int[n + 1][m + 1];
        int[] ans = construct(p, s, n, m, dp);
        System.out.println(ans.length);
        System.out.println(Arrays.toString(ans).replace("[", "").replace("]", "").replace(",", ""));
    }
}
