#include <bits/stdc++.h>
using namespace std;
const int N = 35;
int n, a, b, k; vector<int> v;
long long dp[N][N][N][2];

long long solve(int l, int r, int na, bool br) {
    if (l > r) return 0;
    long long &ret = dp[l][r][na][br];
    if (ret != -1) return ret;
    ret = 1e18;
    if (br == 0) {
        ret = min(ret, solve(l + 1, r, na, 1) + b * na * (r - l + 1));
        ret = min(ret, solve(l, r - 1, na, 1) + b * na * (r - l + 1));
        int nl = l, nr = r, ans = 0;
        while (nl <= nr) {
            int mid = (nl + nr) / 2;
            if (binary_search(v.begin(), v.end(), mid)) ++ans, nl = mid + 1;
            else nr = mid - 1;
        }
        ret = min(ret, solve(nl, nr, na - ans, 0) + a);
        if ((r - l + 1) / 2 >= 1) {
            int m = (l + r) / 2;
            ret = min(ret, solve(l, m, na, 0) + solve(m + 1, r, na, 0));
        }
    } else {
        ret = 1e18;
    }
    return ret;
}

int main() {
    cin >> n >> k >> a >> b;
    memset(dp, -1, sizeof(dp));
    for (int i = 0; i < k; ++i) {
        int x; cin >> x;
        v.push_back(x);
    }
    sort(v.begin(), v.end());
    cout << solve(1, (1 << n), k, 0) << "\n";
}
