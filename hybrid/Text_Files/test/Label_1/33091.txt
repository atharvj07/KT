import java.io.*;
import java.util.Scanner;

public class Main {
    static int a[];
    static int N;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        a = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            a[i] = sc.nextInt();
        }

        int ans = 0;
        a[0] = Integer.MAX_VALUE;
        for (int i = 1; i <= N; i++) {
            if (a[i] < a[ans]) {
                ans = i;
            }
        }

        System.out.println(ans);
    }
}
