import queue

n, m = map(int, input().split())
edges = [[] for _ in range(n + 1)]
for i in range(m):
    u, v = map(int, input().split())
    edges[u].append(v)

vis = [False] * (n + 1)
order = []
q = queue.Queue()
q.put(1)
vis[1] = True
while not q.empty():
    u = q.get()
    order.append(u)
    for v in edges[u]:
        if not vis[v]:
            vis[v] = True
            q.put(v)

if len(order) == n:
    print(*order)
else:
    print(-1)
