import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edge = [[] for _ in range(n+1)]
color = {}
q = []
for i in range(m):
  a, b, c = map(int, input().split())
  color[(a, b, c)] = 1
  color[(b, a, c)] = 1
  edge[a].append(b)
  edge[b].append(a)
for i in range(int(input())):
  q.append(map(int, input().split()))
for u, v in q:
  vis = [0] * (n+1)
  vis[u] = 1
  stk = [u]
  cnt = 0
  while stk:
    t = stk.pop()
    for a in edge[t]:
      if a == v:
        for i in color:
          if i[0] == t and i[1] == a:
            cnt += 1
        continue
      if vis[a] == 0:
        vis[a] = 1
        stk.append(a)
  print(cnt)
