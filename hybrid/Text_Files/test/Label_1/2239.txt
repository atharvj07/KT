#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> a(n);
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      a[i] = -1;
      cnt++;
    } else {
      a[i] = s[i] - '0';
    }
  }
  if (cnt % 2 == 0) {
    cout << "Bicarp" << endl;
  } else {
    cout << "Monocarp" << endl;
  }
  return 0;
}
