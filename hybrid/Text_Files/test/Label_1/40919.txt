import sys
input = sys.stdin.readline

a, b = map(int, input().split())
if a == b:
    print(0)
elif a == 1 or b == 1:
    print(-1)
else:
    cnt = 0
    while a % 2 == 0 and b % 2 == 0:
        a //= 2
        b //= 2
        cnt += 1
    while a % 3 == 0 and b % 3 == 0:
        a //= 3
        b //= 3
        cnt += 1
    while a % 5 == 0 and b % 5 == 0:
        a //= 5
        b //= 5
        cnt += 1
    if a == b:
        print(cnt)
    else:
        print(-1)
