#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  if (n < 10) {
    cout << -1 << endl;
    return 0;
  }
  string ans = "";
  while (n > 0) {
    if (n % 2 == 0) {
      ans = '0' + ans;
      n /= 2;
    } else if (n % 3 == 0) {
      ans = '3' + ans;
      n /= 3;
    } else if (n % 5 == 0) {
      ans = '5' + ans;
      n /= 5;
    } else if (n % 7 == 0) {
      ans = '7' + ans;
      n /= 7;
    } else {
      cout << -1 << endl;
      return 0;
    }
  }
  cout << ans << endl;
  return 0;
}
