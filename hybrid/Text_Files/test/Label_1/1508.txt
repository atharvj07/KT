#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int t;
    cin >> t;
    while (t--) {
        ll n, k;
        cin >> n >> k;
        ll e = n / k;
        ll ans;
        if (n % k <= k / 2)
            ans = e;
        else
            ans = e - 1;
        cout << ans << endl;
    }
}
