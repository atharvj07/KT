import java.util.*;

class Main {
    static final long INF = (long) 1e18;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), k = sc.nextInt();
        long x = sc.nextLong(), y = sc.nextLong();
        int[] a = new int[n + 1], b = new int[m + 1];
        HashMap<Integer, Integer> p = new HashMap<>();
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            p.put(a[i], i);
        }
        for (int i = 0; i < m; i++)
            b[i] = sc.nextInt();
        if (n == m) {
            for (int i = 0; i < n; i++)
                if (a[i] != b[i]) {
                    System.out.println(-1);
                    return;
                }
            System.out.println(0);
            return;
        }
        boolean flag = true;
        long ans = 0;
        for (int i = 0; i + k <= n; i++) {
            if (!p.containsKey(b[0])) {
                flag = false;
                break;
            }
            if (p.containsKey(b[0]) && p.get(b[0]) >= i && p.get(b[0]) <= i + k - 1) {
                ans += y;
                for (int j = 0; j < k - 1; j++) {
                    p.remove(a[i + j]);
                    a[i + j] = a[i + j + 1];
                }
                i += k - 1;
            } else {
                if (p.get(b[0]) < i) {
                    flag = false;
                    break;
                }
                ans += x;
                a[i + k - 1] = a[i];
                for (int j = i; j < i + k - 1; j++)
                    p.remove(a[j]);
            }
        }
        if (flag) {
            for (int i = 0; i < m; i++)
                if (!p.containsKey(b[i]))
                    flag = false;
            if (!flag) {
                System.out.println(-1);
                return;
            }
        }
        for (int i = 0, j = 0; i < n; i++) {
            if (a[i] == b[j]) {
                j++;
                if (j == m)
                    break;
            } else
                ans += y;
        }
        System.out.println(ans);
    }
}
