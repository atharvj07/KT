#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<bool> canVote(n, true);
  for (int i = 0; i < n; i++) {
    if (!canVote[i]) continue;
    for (int j = i + 1; j < n; j++) {
      if (s[i] != s[j]) canVote[j] = false;
    }
  }
  cout << (canVote[0] ? s[0] : s[1]) << endl;
  return 0;
}
