import sys

graph = [[] for _ in range(100001)]
visited = [False] * 100001

def dfs(node, prev):
    visited[node] = True
    for neighbor in graph[node]:
        if not visited[neighbor] and neighbor != prev:
            dfs(neighbor, node)
        elif neighbor == prev:
            return True
    return False

n = int(input())
b = list(map(int, input().split()))
a = list(map(int, input().split()))

for _ in range(n - 1):
    x, k = map(int, input().split())
    graph[k].append(x)
    graph[x].append(k)

for i in range(1, n + 1):
    if b[i - 1] < a[i - 1]:
        if dfs(i, -1):
            print("YES")
            sys.exit()
    b[i - 1] -= a[i - 1]

print("NO")
