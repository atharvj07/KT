import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int max = 0;
        int min = 200;
        int money = 1000;
        int num = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] < min) {
                min = a[i];
                num = money / min;
                money -= num * min;
            } else if (a[i] > max) {
                max = a[i];
                money += num * max;
            }
        }
        System.out.println(money);
    }
}
