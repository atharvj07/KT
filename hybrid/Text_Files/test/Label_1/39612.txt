n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
s = [x[0] for x in a]
a = [x[1:] for x in a]

ciel, jiro = 0, 0
p = 0
while sum(s) > 0:
    if p == 0:
        while p < len(s) and s[p] == 0:
            p += 1
        if p == len(s):
            break
        ciel += a[p][0]
        s[p] -= 1
        if s[p] == 0:
            p += 1
    else:
        while p > 0 and s[p] == 0:
            p -= 1
        if p <= 0:
            break
        jiro += a[p][-1]
        s[p] -= 1
        if s[p] == 0:
            p -= 1

print(ciel, jiro)
