import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (scanner.hasNext()) {
            int n = scanner.nextInt();
            int s = scanner.nextInt();

            // Create a 2D array to store the number of combinations for each state
            int[][] dp = new int[n + 1][s + 1];

            // Initialize the first row and column of the dp array to 0
            for (int i = 0; i <= n; i++) {
                dp[i][0] = 0;
            }
            for (int j = 0; j <= s; j++) {
                dp[0][j] = 0;
            }

            // Fill in the dp array
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= s; j++) {
                    // If the current number is less than or equal to the remaining sum, then we can include it in the combination
                    if (i <= j) {
                        // The number of combinations for the current state is the sum of the number of combinations for the previous state without the current number and the number of combinations for the previous state with the current number
                        dp[i][j] = dp[i - 1][j] + dp[i - 1][j - i];
                    } else {
                        // If the current number is greater than the remaining sum, then we cannot include it in the combination
                        dp[i][j] = dp[i - 1][j];
                    }
                }
            }

            // Print the number of combinations for the given n and s
            System.out.println(dp[n][s]);
        }
    }
}
