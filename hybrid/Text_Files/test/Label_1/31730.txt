import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    static long MOD = 1000000007;
    static long[] pow2;
    static long[][] fact;
    static long[][] revFact;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        pow2 = new long[n + 2];
        fact = new long[n + 2][n + 2];
        revFact = new long[n + 2][n + 2];
        for (int i = 0; i <= n + 1; i++) {
            if (i == 0) {
                pow2[i] = 1;
                fact[i][0] = 1;
                revFact[i][0] = 1;
            } else {
                pow2[i] = (pow2[i - 1] * 2) % MOD;
                fact[i][0] = (fact[i - 1][0] * i) % MOD;
                revFact[i][0] = modInv(fact[i][0], MOD);
            }
        }
        for (int i = 1; i <= n + 1; i++) {
            for (int j = 1; j <= n + 1; j++) {
                fact[i][j] = (fact[i - 1][j] * i) % MOD;
            }
        }
        for (int i = 1; i <= n + 1; i++) {
            for (int j = 1; j <= n + 1; j++) {
                revFact[i][j] = (revFact[i - 1][j] * i) % MOD;
            }
        }
        long ans = ((pow2[n] * fact[n][n]) % MOD + MOD) % MOD;
        System.out.println(ans);
    }

    private static long modInv(long a, long m) {
        long x = 0, y = 1;
        while (a > 1) {
            long q = a / m;
            long t = m;
            m = a % m;
            a = t;
            t = y;
            y = x - q * y;
            x = t;
        }
        if (x < 0) {
            x += m;
        }
        return x;
    }
}
