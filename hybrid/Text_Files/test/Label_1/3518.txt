#include <iostream>
#include <vector>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> dp(n + 1);
  dp[n] = 0;
  for (int i = n - 1; i >= 0; i--) {
    if (i + 2 < n && s[i] == s[i + 1] && s[i] == s[i + 2]) {
      dp[i] = dp[i + 3];
    } else if (i + 1 < n && s[i] == s[i + 1]) {
      dp[i] = dp[i + 2];
    } else {
      dp[i] = dp[i + 1] + 1;
    }
  }
  string t;
  for (int i = 0; i < n; i++) {
    if (i + 2 < n && s[i] == s[i + 1] && s[i] == s[i + 2]) {
      i += 2;
    } else if (i + 1 < n && s[i] == s[i + 1]) {
      i++;
    } else {
      t += s[i];
    }
  }
  cout << t << endl;
  return 0;
}
