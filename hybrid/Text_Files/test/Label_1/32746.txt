import java.util.*;

public class Main {

    static int b;
    static int n;
    static int[][] dp;

    static int solve(int idx, int rem, boolean leadingZero) {
        if (idx == n)
            return rem == 0 ? 1 : 0;
        if (dp[idx][rem] != -1)
            return dp[idx][rem];
        int ans = 0;
        int start = leadingZero ? 1 : 0;
        for (int i = start; i < b; i++)
            if ((rem * 10 + i) % b == 0)
                ans += solve(idx + 1, i, leadingZero && i == 0);
        return dp[idx][rem] = ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int x = sc.nextInt();
        dp = new int[n + 1][b + 1];
        for (int i = 0; i <= n; i++)
            Arrays.fill(dp[i], -1);
        for (b = 2; b < x && solve(0, 0, true) == 0; b++) ;
        System.out.println(b == x ? -1 : b);
    }
}
