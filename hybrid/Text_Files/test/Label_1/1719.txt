#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  sort(p.begin(), p.end());
  vector<int> ans(n);
  int j = 0;
  for (int i = 0; i < n; i++) {
    while (p[i] - p[j] > k) j++;
    ans[i] = p[j];
  }
  for (int i = 0; i < n; i++) cout << ans[i] << ' ';
  cout << endl;
  return 0;
}
