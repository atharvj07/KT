#include <iostream>
#include <string>

int main() {
  int n;
  std::cin >> n;
  std::string s;
  std::cin >> s;
  for (int i = 0; i < n - 1; ++i) {
    if (s[i] != '?' && s[i] == s[i + 1]) {
      std::cout << "No" << std::endl;
      return 0;
    }
  }
  int cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (s[i] == '?') ++cnt;
  }
  std::cout << (cnt > 1 ? "Yes" : "No") << std::endl;
  return 0;
}
