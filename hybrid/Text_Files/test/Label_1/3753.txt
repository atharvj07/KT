#include <iostream>
#include <string>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string b;
        cin >> b;
        int n = b.length();
        for (int i = 0; i < n - 1; i++) {
            cout << b[i];
        }
        cout << endl;
    }
    return 0;
}
