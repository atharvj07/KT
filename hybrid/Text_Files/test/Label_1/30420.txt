import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt(), n = sc.nextInt();
        if ((m + 1) / n % 2 == 0 || 2 * n - 1 < m) {
            System.out.println(-1);
        } else {
            System.out.println((m + 1) / n + (m % n == 0 ? 0 : 1));
        }
    }
}
