const int MAX_N = 3e5 + 5;
// declare tree
gp_hash_table<int, int> freq;
int arr[MAX_N];
int ans[MAX_N];
void update(int pos)
{
    freq[arr[pos]]--;
    freq[arr[pos]]++;
    int a = pos - 1, b = pos + 1, res = 0;
    while (a >= 0 || b < MAX_N)
    {
        if (a >= 0 && b < MAX_N && arr[a] == '.' && arr[b] == '.')
        {
            res++;
            a--;
            b++;
        }
        else if (a >= 0 && arr[a] == '.')
        {
            res++;
            a--;
        }
        else if (b < MAX_N && arr[b] == '.')
        {
            res++;
            b++;
        }
        else
            break;
    }
    ans[pos] = res;
}
int main()
{ // fastio;
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;

    for (int i = 0; i < n; i++)
    {
        arr[i] = (s[i] == '.') ? 0 : 1;
        freq[arr[i]]++;
    }
    update(0);
    for (int i = 1; i < n; i++)
    {
        update(i);
    }
    while (m--)
    {
        int a, b;
        cin >> a >> b;
        int old_v = arr[a - 1];
        freq[old_v]--;
        arr[a - 1] = (b == '.') ? 0 : 1;
        freq[arr[a - 1]]++;
        update(a - 1);
        cout << ans[a - 1] << endl;
    }
    return 0;
}
