import java.util.Arrays;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }

        Arrays.sort(a);

        int index = -1;
        long max = 0;
        for (int i = 0; i < k; i++) {
            long total = n / a[i];
            if (total > max) {
                max = total;
                index = i;
            }
        }
        System.out.println((index + 1) + " " + max);
    }
}
