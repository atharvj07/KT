import sys

def solve():
    t = int(input())

    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))

        stack = []
        ok = True
        for i in range(n):
            if not stack:
                stack.append((a[i], i))
            elif stack[-1][0] > a[i]:
                ok = False
                break
            elif stack[-1][1] % 2 == (i % 2):
                stack.append((a[i], i))
            else:
                stack.pop()
                stack.append((a[i], i))

        print("YES" if ok and not stack else "NO")


solve()
