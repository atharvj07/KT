import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;

public class Main {
    private static final long MAX_LENGTH = 2500000000L;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[][] points = new int[n][2];
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            points[i][0] = Integer.parseInt(line[0]);
            points[i][1] = Integer.parseInt(line[1]);
        }

        List<Integer> path = new ArrayList<>();
        List<Integer> visited = new ArrayList<>();
        long length = 0;
        int start = 0;
        while (visited.size() < n) {
            if (!visited.isEmpty()) {
                start = visited.get(visited.size() - 1);
            }
            int next = -1;
            long minDist = Long.MAX_VALUE;
            for (int i = 0; i < n; i++) {
                if (visited.contains(i)) {
                    continue;
                }
                long dist = getDistance(points[start][0], points[start][1], points[i][0], points[i][1]);
                if (dist < minDist) {
                    minDist = dist;
                    next = i;
                }
            }
            visited.add(next);
            path.add(next + 1);
            length += minDist;
            if (length > MAX_LENGTH) {
                break;
            }
        }

        if (length <= MAX_LENGTH) {
            for (int p : path) {
                System.out.print(p + " ");
            }
            System.out.println();
        } else {
            System.out.println(-1);
        }
    }

    private static long getDistance(int x1, int y1, int x2, int y2) {
        return Math.abs(x1 - x2) + Math.abs(y1 - y2);
    }
}
