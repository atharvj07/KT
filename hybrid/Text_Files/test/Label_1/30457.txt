import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;
import java.util.Stack;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        sc.nextLine(); // consume end of line

        Map<String, String> types = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            String[] parts = line.split(" ");
            if (parts[0].equals("typedef")) {
                types.put(parts[2], parts[1]);
            } else if (parts[0].equals("typeof")) {
                String type = types.getOrDefault(parts[1], "errtype");
                // Process pointer operations
                type = type.replaceAll("\\*+$", ""); // remove trailing asterisks
                type = type.replaceAll("&+$", ""); // remove trailing ampersands
                // Process typedef operations
                type = type.replaceFirst("void?", ""); // ignore void prefixes
                // Add necessary number of asterisks
                // The number of asterisks removed in the previous steps is equal to the number of ampersands removed
                int ampersandCount = type.length() - type.replaceAll("&", "").length();
                for (int j = 0; j < ampersandCount; j++) {
                    type = "*" + type;
                }
                System.out.println(type);
            }
        }

        sc.close();
    }
}
