#include <bits/stdc++.h>
using namespace std;
const int mod=1e9+7,maxn=2e5+10;
int add(int x,int y){return (x+y)%mod;}
int sub(int x,int y){return (x-y+mod)%mod;}
int mul(int x,int y){return (int)((1ll*x*y)%mod);}
int32_t main(){
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int n,k;cin>>n>>k;vector<int>a(n);
    for(int &x:a)cin>>x;
    int ans=0,cur=0;
    unordered_map<int,int>mp;
    mp[0]=1;
    for(int i=0;i<n;i++){
        int suf=0;
        for(int j=i+1;j<n;j++){
            suf=(suf+a[j]%k)%k;
            if(suf==0)ans++;
        }
        int prv=0;
        for(int j=i-1;j>=0;j--){
            prv=(prv*10)%k;
            prv=(prv+a[j]%k)%k;
            ans+=mp[(k-prv)%k];
        }
        cur=(cur+a[i]%k)%k;
        mp[cur]++;
    }
    cout<<ans<<endl;
}
