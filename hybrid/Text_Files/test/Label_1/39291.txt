n = int(input())
cities = [input().split() for _ in range(n)]
cities = [(int(x), c) for x, c in cities]

total_cost = 0
last_b = last_r = last_p = None
min_gap_b = min_gap_r = float('inf')

for x, c in cities:
    if c == 'P':
        if last_b is not None:
            total_cost += x - last_b
            last_b = None
            min_gap_b = float('inf')
        if last_r is not None:
            total_cost += x - last_r
            last_r = None
            min_gap_r = float('inf')
        last_p = x
    elif c == 'B':
        if last_b is not None:
            total_cost += x - last_b
        elif last_p is not None:
            min_gap_b = min(min_gap_b, x - last_p)
        last_b = x
    elif c == 'R':
        if last_r is not None:
            total_cost += x - last_r
        elif last_p is not None:
            min_gap_r = min(min_gap_r, x - last_p)
        last_r = x

if last_b is not None and last_p is not None:
    total_cost += min_gap_b
if last_r is not None and last_p is not None:
    total_cost += min_gap_r

print(total_cost)
