import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        TreeSet<Long> s = new TreeSet<>();
        while(q-->0){
            char t = sc.next().charAt(0);
            long x = sc.nextLong();
            switch(t){
                case '+': s.add(x);break;
                case '-': s.remove(x);break;
            }
            Long[] arr = s.toArray(new Long[0]);
            long lst=0;
            int c = 0;
            for(long y:arr){
                if(y<=2*lst)c++;
                lst = y;
            }
            System.out.println(c);
        }
    }
}
