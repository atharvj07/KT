#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> w(n);
        for (int i = 0; i < n; i++) { cin >> w[i]; }
        sort(w.begin(), w.end());
        bool ok = true;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += w[i];
            if (sum == x) {
                ok = false;
                break;
            }
        }
        if (ok) {
            cout << "YES ";
            for (int i : w) { cout << i << " "; }
        } else {
            cout << "NO";
        }
        cout << endl;
    }
    return 0;
}
