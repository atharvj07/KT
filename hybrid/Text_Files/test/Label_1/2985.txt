#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int M = 998244353;
const int N = 1e5 + 10;

int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    char s[N][4], t[N][4];
    

    for (int i = 0; i < n; i++) cin >> s[i];

    for (int i = 0; i < n; i++) {
        t[i][0] = s[i][0];
        t[i][1] = s[i][1] == '?' ? 'B' : s[i][1];
        t[i][2] = s[i][1] == '?' ? 'W' : s[i][0];
        t[i][3] = s[i][0];
    }

    ll ans = 1;

    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = i; j < n; j++) {
            for (int k = 0; k < 4; k++) {
                t[j][(k + 1) % 4] == s[j][(k + 1) % 4] ? cnt++ : 0;
            }
        }
        if (!cnt) ans = (ans * 2) % M;
    }

    cout << ans << "\n";
}
