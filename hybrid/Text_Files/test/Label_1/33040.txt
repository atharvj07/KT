import java.util.*;

class Main {
    static List<Integer>[] graph;
    static int[] ancestors, a;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        graph = new ArrayList[n + 1];
        ancestors = new int[n + 1];
        a = new int[n + 1];
        for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int p = sc.nextInt(), q = sc.nextInt();
            graph[p].add(q);
        }
        for (int i = 1; i <= n; i++) a[i] = sc.nextInt();
        for (int i = 1; i <= n; i++) dfs1(i, -1);
        for (int i = 1; i <= n; i++) dfs2(i, -1, 0);
        boolean ok = true;
        for (int i = 1; i <= n; i++) if (ancestors[a[i]] == 0) ok = false;
        if (!ok) System.out.println(-1);
        else {
            int i = 0, ans = 0;
            List<Integer> b = new ArrayList<>();
            while (i != -1) {
                i = ancestors[i];
                if (i != 0) {
                    ans++;
                    b.add(i);
                }
            }
            System.out.println(ans);
            Collections.reverse(b);
            for (int x : b) System.out.println(x);
        }
    }

    static void dfs1(int u, int p) {
        ancestors[u] = p;
        for (int v : graph[u]) if (v != p) dfs1(v, u);
    }

    static void dfs2(int u, int p, int last) {
        ancestors[u] = last;
        for (int v : graph[u]) if (v != p) dfs2(v, u, a[v]);
    }
}
