import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            int T = scanner.nextInt();
            int H = scanner.nextInt();
            int S = scanner.nextInt();
            if (T == -1 && H == -1 && S == -1) {
                break;
            }
            int remainingTime = 120 * 60 - T * 60 * 60 - H * 60 - S;
            int standardRecordingHours = remainingTime / 60 / 60;
            int standardRecordingMinutes = (remainingTime / 60) % 60;
            int standardRecordingSeconds = remainingTime % 60;
            int tripleRecordingHours = standardRecordingHours * 3;
            int tripleRecordingMinutes = standardRecordingMinutes * 3;
            int tripleRecordingSeconds = standardRecordingSeconds * 3;
            System.out.println(String.format("%02d:%02d:%02d", standardRecordingHours, standardRecordingMinutes, standardRecordingSeconds));
            System.out.println(String.format("%02d:%02d:%02d", tripleRecordingHours, tripleRecordingMinutes, tripleRecordingSeconds));
        }
        scanner.close();
    }
}
