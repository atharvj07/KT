#include <iostream>
#include <map>

int main() {
  std::string home, away;
  std::map<std::string, std::map<int, int>> yellows;
  std::cin >> home >> away;
  int n;
  std::cin >> n;
  while (n--) {
    int t;
    char c;
    int m;
    char type;
    std::cin >> t >> c >> m >> type;
    if (type == 'y')
      yellows[c == 'h' ? home : away][m] |= 1 << (t / 2);
    else
      std::cout << (c == 'h' ? home : away) << ' ' << m << ' ' << t << '\n';
  }
  for (const auto &[team, players] : yellows) {
    for (auto &[p, y] : players) {
      if (y == 3) std::cout << team << ' ' << p << ' ' << 2 * __builtin_ctz(y) << '\n';
    }
  }
  return 0;
}
