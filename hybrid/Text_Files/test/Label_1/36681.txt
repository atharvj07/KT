for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    i, j = 0, 0
    l = []
    while i < n and j < m:
        if a[i] == b[j]:
            l.append(a[i])
            i += 1
            j += 1
        elif a[i] > b[j]:
            j += 1
        else:
            i += 1
    if l:
        print("YES")
        print(len(l), *l)
    else:
        print("NO")
