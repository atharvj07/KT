import java.util.*;

class Main {
    static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] pow2 = new int[1000001];
        pow2[0] = 1;
        for (int i = 1; i < 1000001; i++) pow2[i] = pow2[i - 1] * 2 % MOD;

        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        long amax = 0;
        for (int aa : a) amax = Math.max(amax, aa);
        long[] b = new long[n + 1];

        for (int j = 1; j <= n; j++) {
            long sum = 0;
            for (int i = 0; i < n; i++) {
                sum += pow2[n - j] * (amax % MOD) % MOD;
                amax = Math.min(amax, a[i]);
            }
            b[j] = sum % MOD;
            amax = 0;
            for (int aa : a) amax = Math.max(amax, aa);
        }

        long gcd = 0;
        for (long bb : b) gcd = gcd(gcd, bb);
        System.out.println(gcd);
    }

    static long gcd(long a, long b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
}
