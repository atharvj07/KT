import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class RookPath {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Number of vertical spells
        int m = scanner.nextInt(); // Number of horizontal spells

        List<Integer> verticalSpells = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            verticalSpells.add(scanner.nextInt() - 1); // Subtract 1 to adjust for 0-based indexing
        }

        List<Spell> horizontalSpells = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x1 = scanner.nextInt() - 1;
            int x2 = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            horizontalSpells.add(new Spell(x1, x2, y));
        }

        int result = solve(verticalSpells, horizontalSpells);
        System.out.println(result);
    }

    private static int solve(List<Integer> verticalSpells, List<Spell> horizontalSpells) {
        // Sort vertical spells in ascending order
        verticalSpells.sort(null);

        // Sort horizontal spells by y-coordinate in ascending order, then by x1-coordinate in ascending order
        horizontalSpells.sort((spell1, spell2) -> {
            if (spell1.y != spell2.y) {
                return Integer.compare(spell1.y, spell2.y);
            } else {
                return Integer.compare(spell1.x1, spell2.x2);
            }
        });

        // Initialize variables
        int spellsToRemove = 0;
        int currentColumn = 0; // Current column of the rook
        int currentRow = 0; // Current row of the rook
        int maxRow = 999999999; // Target row (10^9 - 1)

        // Iterate over horizontal spells
        for (Spell spell : horizontalSpells) {
            // If the rook is below the current spell, ignore it
            if (currentRow < spell.y) {
                continue;
            }

            // If the current spell blocks the rook's path, remove it
            if (spell.x1 <= currentColumn && spell.x2 >= currentColumn) {
                spellsToRemove++;
            } else {
                // If the rook can move to the right without hitting a vertical spell, do so
                while (currentColumn < spell.x1 && verticalSpells.size() > 0 && verticalSpells.get(0) <= currentColumn) {
                    currentColumn = verticalSpells.remove(0) + 1;
                }

                // If the rook can move up to the current spell's row without hitting a horizontal spell, do so
                while (currentRow < spell.y && horizontalSpells.size() > 0 && horizontalSpells.get(0).y <= currentRow) {
                    horizontalSpells.remove(0);
                }
            }

            // If the rook has reached the target row, stop iterating
            if (currentRow == maxRow) {
                break;
            }
        }

        return spellsToRemove;
    }

    private static class Spell {
        int x1, x2, y;

        public Spell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }
    }
}
