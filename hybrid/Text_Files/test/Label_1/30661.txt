import java.util.Arrays;
import java.util.Scanner;

public class VasyaAndTheCarRental {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int s = scanner.nextInt();
        int t = scanner.nextInt();

        int[][] cars = new int[n][2];
        for (int i = 0; i < n; i++) {
            cars[i][0] = scanner.nextInt();
            cars[i][1] = scanner.nextInt();
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        Arrays.sort(gasStations);

        int minCost = -1;
        for (int[] car : cars) {
            int cost = car[0];
            int fuelCapacity = car[1];
            int time = 0;
            int currentFuel = fuelCapacity;

            for (int gasStation : gasStations) {
                int distanceToStation = gasStation - time;
                int timeToStation = Math.min(distanceToStation, currentFuel);
                time += timeToStation;
                currentFuel -= timeToStation;

                if (currentFuel == 0) {
                    currentFuel = fuelCapacity;
                    time += 2;
                }
            }

            int distanceToCinema = s - time;
            int timeToCinema = Math.min(distanceToCinema, currentFuel);
            time += timeToCinema;

            if (time <= t && (minCost == -1 || cost < minCost)) {
                minCost = cost;
            }
        }

        System.out.println(minCost);
    }
}
