#include <iostream>
#include <vector>
using namespace std;

const unsigned long long UNSIGNED_MAX = 1<<30;
int main() {
  unsigned long long n;
  cin >> n;
  vector<unsigned long long> v(n);
  unsigned long long ones = 0;
  unsigned long long zeroes = 0;
  for (unsigned long long i = 0; i < n; i++) {
    cin >> v[i];
    if (v[i] == 0) {
      zeroes++;
    } else {
      ones++;
    }
  }
  unsigned long long result = 0;
  if (zeroes % 2 == 1) {
    result ^= UNSIGNED_MAX;
  }
  cout << result << endl;

  return 0;
}
