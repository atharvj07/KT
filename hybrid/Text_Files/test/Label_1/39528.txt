def check_respectable(points):
    x_coords = sorted(set([p[0] for p in points]))
    y_coords = sorted(set([p[1] for p in points]))
    
    if len(x_coords) != 3 or len(y_coords) != 3:
        return "ugly"
    
    expected_points = [(x, y) for x in x_coords for y in y_coords if (x, y) != (x_coords[1], y_coords[1])]
    
    if sorted(points) == sorted(expected_points):
        return "respectable"
    else:
        return "ugly"

points = [tuple(map(int, input().split())) for _ in range(8)]
print(check_respectable(points))
