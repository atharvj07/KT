n, k = map(int, input().split())
s = input()

# Find the minimum character in the prefix of s that can reach length k
min_prefix = s[0]
for i in range(1, min(n, k)):
    if s[i] < min_prefix:
        min_prefix = s[i]

# Construct the result based on the minimum character found
result = s
while len(result) < k:
    result += result
result = result[:k]

# Replace all characters that are not the minimum with the minimum character
result = ''.join([min_prefix if c > min_prefix else c for c in result])

print(result)
