def solve():
    n, k = map(int, input().strip().split())
    arr = list(input().strip())
    for i in range(k):
        for j in range(1, n + 1):
            cnt_w = (arr[(j - 2) % n] == 'W') + (arr[j - 1] == 'W') + (arr[j] == 'W')
            cnt_b = (arr[(j - 2) % n] == 'B') + (arr[j - 1] == 'B') + (arr[j] == 'B')
            if cnt_w > cnt_b:
                arr[j - 1] = 'W'
            else:
                arr[j - 1] = 'B'
    return ''.join(arr)

print(solve())
