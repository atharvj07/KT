import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), d = sc.nextInt(), S = sc.nextInt();
        int[][] robots = new int[n][3];
        for (int i = 0; i < n; i++) {
            robots[i][0] = sc.nextInt();
            robots[i][1] = sc.nextInt();
            robots[i][2] = sc.nextInt();
        }
        int maxRobots = 0, minFuel = S + 1;
        for (int i = 0; i < n; i++) {
            if (robots[i][2] >= d) {
                int robotsTaken = 1;
                int fuelNeeded = robots[i][1];
                int currentRobot = i;
                while (robotsTaken <= robots[currentRobot][0] && currentRobot != -1) {
                    fuelNeeded += robots[currentRobot][1];
                    if (fuelNeeded > S) {
                        break;
                    }
                    currentRobot = getRobotWithMaxCapacity(robots, currentRobot);
                    robotsTaken++;
                }
                if (robotsTaken > maxRobots) {
                    maxRobots = robotsTaken;
                    minFuel = fuelNeeded;
                } else if (robotsTaken == maxRobots) {
                    minFuel = Math.min(minFuel, fuelNeeded);
                }
            }
        }
        System.out.println(maxRobots + " " + minFuel);
    }

    public static int getRobotWithMaxCapacity(int[][] robots, int currentRobot) {
        int maxCapacity = -1, nextRobot = -1;
        for (int i = 0; i < robots.length; i++) {
            if (i == currentRobot || robots[i][0] == 0) {
                continue;
            }
            if (robots[i][0] > maxCapacity) {
                maxCapacity = robots[i][0];
                nextRobot = i;
            }
        }
        return nextRobot;
    }
}
