#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, r;
  cin >> n >> r;
  vector<pair<int, int>> points(n);
  for (int i = 0; i < n; i++) {
    cin >> points[i].first >> points[i].second;
  }
  sort(points.begin(), points.end());
  int ans = 0;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < n && points[j].first - points[i].first <= r) { j++; }
    while (j < n && points[j].second - points[i].second <= r) { j++; }
    ans = max(ans, j - i);
  }
  cout << ans << endl;
  return 0;
}
