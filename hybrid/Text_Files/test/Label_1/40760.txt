def solve_path(n, path):
    dx = {0, 1, -1}
    dy = {1, -1, 0}

    x, y = 0, 0
    for i in range(1, n):
        px, py = divmod(path[i-1]-1, y) if y else (0, 0)
        cx, cy = divmod(path[i]-1, y) if y else (0, path[i]-1)
        if path[i] <= path[i-1]:
            y = max(y, path[i-1] - path[i] + 1)
        if (cx-px, cy-py) not in zip(dx, dy):
            print("NO")
            return
    if y == 0: y = path[-1]
    x = (path[-1]-1) // y + 1
    print("YES")
    print(x, y)

n = int(input().strip())
path = list(map(int, input().strip().split()))
solve_path(n, path)
