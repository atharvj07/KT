import sys

input = sys.stdin.readline
n = int(input())
adjList = [[] for i in range(n+1)]
for i in range(n-1):
    a, b, c = map(int, input().split())
    adjList[a].append([b, c])
    adjList[b].append([a, c])

visited = [False] * (n+1)
val = [-1] * (n+1)

def dfs(u, p):
    visited[u] = True
    if len(adjList[u]) == 1 and u != 1:
        val[u] = 0
        return
    for v, w in adjList[u]:
        if v != p:
            dfs(v, u)
            if w == 0:
                val[u] = (val[u] or val[v])
            else:
                if val[u] == 0:
                    val[u] = val[v]
                elif val[v] == 0:
                    val[u] = 1
                else:
                    val[u] = 2

dfs(1, 1)
ans = 0

for i in range(1, n+1):
    if i == 1:
        continue
    if val[i] == 0:
        ans += n - 1
    elif val[i] == 1:
        ans += n - 2
    else:
        ans += n - 3

print(ans)
