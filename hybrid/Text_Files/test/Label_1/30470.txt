import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        String[] positions = s.nextLine().split(" ");

        char[][] board = new char[8][8];

        // Initialize the board with '.'
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                board[i][j] = '.';
            }
        }

        // Place the pieces on the board
        for (String position : positions) {
            char file = position.charAt(0);
            int rank = position.charAt(1) - '0';
            board[rank - 1][file - 'a'] = position.charAt(0);
        }

        // Check if the white rooks can checkmate the black king
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                if (board[i][j] == 'R') {
                    // Check horizontal
                    for (int k = j + 1; k < 8; k++) {
                        if (board[i][k] == '.') {
                            continue;
                        } else if (board[i][k] == 'K') {
                            System.out.println("CHECKMATE");
                            return;
                        } else {
                            break;
                        }
                    }
                    for (int k = j - 1; k >= 0; k--) {
                        if (board[i][k] == '.') {
                            continue;
                        } else if (board[i][k] == 'K') {
                            System.out.println("CHECKMATE");
                            return;
                        } else {
                            break;
                        }
                    }

                    // Check vertical
                    for (int k = i + 1; k < 8; k++) {
                        if (board[k][j] == '.') {
                            continue;
                        } else if (board[k][j] == 'K') {
                            System.out.println("CHECKMATE");
                            return;
                        } else {
                            break;
                        }
                    }
                    for (int k = i - 1; k >= 0; k--) {
                        if (board[k][j] == '.') {
                            continue;
                        } else if (board[k][j] == 'K') {
                            System.out.println("CHECKMATE");
                            return;
                        } else {
                            break;
                        }
                    }
                }
            }
        }

        // Otherwise, print "OTHER"
        System.out.println("OTHER");
    }
}
