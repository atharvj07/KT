import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt(), k = in.nextInt();
        List<Integer> f = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            f.add(in.nextInt());
        }
        Map<Integer, Integer> count = new HashMap<>();
        for (int i : f) {
            count.merge(i, 1, Integer::sum);
        }
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            String s = in.next();
            int d = in.nextInt();
            int c = 0;
            boolean zero = false;
            for (int j = 0; j < d; j++) {
                int e = in.nextInt();
                if (e != 0) {
                    if (count.containsKey(e)) {
                        c++;
                    }
                } else {
                    zero = true;
                }
            }
            if (zero) {
                System.out.println(2);
            } else if (c == 0) {
                System.out.println(1);
            } else if (c == f.size()) {
                System.out.println(0);
            } else {
                System.out.println(2);
            }
        }
    }
}
