n, k = map(int, input().split())
s = list(map(int, input().split()))

MOD = 998244353

# Calculate the number of consecutive suffixes where the next starts at a position
# greater than the current. These positions need distinct characters.
diff = [0] * (n - 1)
for i in range(n - 1):
    if s[i] > s[i + 1]:
        diff[i] = 1

# Calculate the total number of distinct characters needed, which is the sum of diffs + 1
# for each position where a distinct character is needed.
total_diffs = sum(diff) + 1

# If the total number of distinct characters needed is greater than the alphabet size,
# it's impossible to form a string.
if total_diffs > k:
    print(0)
else:
    # result holds the total number of ways to form the string
    result = 1
    for i in range(n - 1):
        if diff[i] == 0:
            result = (result * k) % MOD
    print(result)
