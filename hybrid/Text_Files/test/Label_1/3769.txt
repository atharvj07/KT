#include<bits/stdc++.h>
#define int long long
#define mod 998244353
using namespace std;
const int N = 1e6 + 5;
int f[N], ans[N], pw[N];
int32_t main() {
	pw[0] = 1;
	for(int i = 1; i < N; i++) pw[i] = (pw[i - 1] * 2) % mod;
	f[0] = f[1] = 1;
	f[2] = 3;
	for(int i = 3; i < N; i++) f[i] = (f[i - 1] + f[i - 2]) % mod;
	ans[0] = ans[1] = ans[2] = ans[3] = 0;
	for(int i = 4; i < N; i++) {
		ans[i] = (f[i] - 3 * ans[i - 1] + 2 * ans[i - 2] + mod) % mod;
	}
	int n; cin >> n;
	cout << ans[n] % mod << endl;
	return 0;
}
