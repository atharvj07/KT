t = int(input())
for _ in range(t):
    n = int(input())
    p = list(map(int, input().split()))
    
    max_medals = n // 2
    while max_medals > 0 and p[max_medals] == p[max_medals - 1]:
        max_medals -= 1
    
    g = 1
    while g < max_medals and p[g] == p[g - 1]:
        g += 1
    
    s = g + 1
    while s < max_medals and (p[s] == p[s - 1] or s - g <= g):
        s += 1
    
    b = max_medals - s
    
    if g < s - g and g < b:
        print(g, s - g, b)
    else:
        print(0, 0, 0)
