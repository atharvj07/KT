import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int,input().split()))
p = [0]*64
for x in a:
    i = 0
    while x>0:
        if x%2: p[i]+=1
        i+=1
        x//=2
mn = min(x for x in p if x>0)
mx = max(p)
if min(a)!=1 or mn==mx: print(-1)
else:
    ans = []
    for i in range(mn,mx+1):
        if all(x%i==0 for x in p): ans.append(i)
    print(*ans)
