import sys

def solve():
    a = int(sys.stdin.readline().strip())
    l = int(sys.stdin.readline().strip())
    r = int(sys.stdin.readline().strip())
    if a == 0:
        return 2
    if l == 0:
        l = 1
    if r == 0:
        r = 9
    return (r - l + 1) ** (len(str(a)) - 1) % 998244353

print(solve())
