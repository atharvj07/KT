import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong(), k = in.nextLong(), ans = 0, box = 0;
        long[] a = new long[100000];
        for (int i = 0; i < k; i++) {
            a[i] = in.nextLong();
            long cur = (n + a[i] - 1) / a[i];
            if (ans < cur) {
                ans = cur;
                box = i + 1;
            }
        }
        System.out.println(box + " " + ans);
    }
}
