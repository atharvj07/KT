import java.util.*;
import java.io.*;

public class Main {
    private static final int MAX_COORDINATE = 106;
    private static final long MAX_LENGTH = 25L * 1000000000L;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());

        List<Point> points = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            points.add(new Point(x, y));
        }

        long[][] distances = new long[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                distances[i][j] = calculateDistance(points.get(i), points.get(j));
            }
        }

        int[] permutation = findHamiltonianPath(distances);
        for (int i = 0; i < n; i++) {
            System.out.print((permutation[i] + 1) + " ");
        }
        System.out.println();
    }

    private static int[] findHamiltonianPath(long[][] distances) {
        int n = distances.length;

        // Initialize the dp table
        long[][] dp = new long[n][1 << n];
        for (int i = 0; i < n; i++) {
            for (int mask = 0; mask < (1 << n); mask++) {
                dp[i][mask] = Long.MAX_VALUE;
            }
        }

        // Set the base case
        dp[0][1 << 0] = 0;

        // Populate the dp table
        for (int mask = 1; mask < (1 << n); mask++) {
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    for (int j = 0; j < n; j++) {
                        if ((mask & (1 << j)) != 0 && i != j) {
                            dp[i][mask] = Math.min(dp[i][mask], dp[j][mask ^ (1 << i)] + distances[j][i]);
                        }
                    }
                }
            }
        }

        // Find the shortest Hamiltonian path
        int[] permutation = new int[n];
        int mask = (1 << n) - 1;
        int current = 0;
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < n; j++) {
                if ((mask & (1 << j)) != 0 && dp[j][mask] + distances[j][current] == dp[current][mask]) {
                    permutation[i] = j;
                    mask ^= (1 << j);
                    current = j;
                    break;
                }
            }
        }

        return permutation;
    }

    private static long calculateDistance(Point p1, Point p2) {
        return Math.abs(p1.x - p2.x) + Math.abs(p1.y - p2.y);
    }

    private static class Point {
        int x;
        int y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
