import java.util.Arrays;
import java.util.Scanner;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] rooms = new int[n][3];
        for (int i = 0; i < n; i++) {
            rooms[i][0] = scanner.nextInt();
            rooms[i][1] = scanner.nextInt();
            rooms[i][2] = scanner.nextInt();
        }

        int m = scanner.nextInt();
        int[][] wallpaper = new int[m][3];
        for (int i = 0; i < m; i++) {
            wallpaper[i][0] = scanner.nextInt();
            wallpaper[i][1] = scanner.nextInt();
            wallpaper[i][2] = scanner.nextInt();
        }

        Arrays.sort(rooms, (a, b) -> a[0] * a[1] * a[2] - b[0] * b[1] * b[2]);
        Arrays.sort(wallpaper, (a, b) -> a[2] - b[2]);

        long ans = 0;
        for (int[] room : rooms) {
            int area = room[0] * room[1] * room[2];
            for (int[] paper : wallpaper) {
                int count = (area + paper[0] - 1) / paper[0];
                long cost = (long) count * paper[2];
                if (cost < area) {
                    ans += cost;
                    break;
                }
            }
        }

        System.out.println(ans);
    }
}
