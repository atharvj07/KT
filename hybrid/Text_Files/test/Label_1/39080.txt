MOD = 10**9 + 7

def mod_pow(base, exponent, modulus):
    result = 1
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus
        exponent = exponent // 2
        base = (base * base) % modulus
    return result

def solve(p, k):
    if k == 0:
        return mod_pow(p, p - 1, MOD)
    elif k == 1:
        return mod_pow(p, p, MOD)
    else:
        cycle_length = 1
        value = k
        while value != 1:
            value = (value * k) % p
            cycle_length += 1
        return mod_pow(p, (p - 1) // cycle_length, MOD)

p, k = map(int, input().split())
print(solve(p, k))
