#include <iostream>
#include <vector>
using namespace std;
vector<vector<int>> g;
int ans, cnt, vis[5005];
void dfs(int now) {
  vis[now] = true;
  cnt++;
  for (int i : g[now]) {
    if (!vis[i]) dfs(i);
  }
}
int main() {
  int n, m, s;
  cin >> n >> m >> s;
  g.resize(n + 5);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    g[u].push_back(v);
  }
  dfs(s);
  cout << (n != cnt ? n - cnt : 0) << endl;
  return 0;
}
