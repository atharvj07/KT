#include <bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
int main() {
  int t;
  cin >> t;
  while (t--) {
    ull n, sum = 0, ans = 0;
    cin >> n;
    vector<ull> a(n);
    for (auto &x : a) cin >> x, sum += x;
    sort(a.begin(), a.end());
    for (auto &x : a) {
      if (sum - x <= 2 * x)
        break;
      else
        ans += (sum - 2 * x) / (n - 1), sum -= ans * (n - 1);
    }
    cout << ans << "\n";
  }
}
