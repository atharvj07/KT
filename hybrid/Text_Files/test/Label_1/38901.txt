import sys
import math

a, b, c = map(int, sys.stdin.readline().split())
x1, y1, x2, y2 = map(int, sys.stdin.readline().split())

if a == 0 and b == 0:
    print(math.sqrt((x2 - x1)**2 + (y2 - y1)**2))
elif a == 0 or b == 0:
    print(abs(x2 - x1) + abs(y2 - y1))
else:
    d1 = abs(a * x1 + b * y1 + c) / math.sqrt(a**2 + b**2)
    d2 = abs(a * x2 + b * y2 + c) / math.sqrt(a**2 + b**2)
    print(math.sqrt((x2 - x1)**2 + (y2 - y1)**2) - d1 - d2)
