import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        long sum = 0;
        for (int i = 1; i < n - 1; i++) {
            sum += Math.min(arr[i], Math.min(arr[i - 1], arr[i + 1]));
        }

        System.out.println(sum);
    }
}
