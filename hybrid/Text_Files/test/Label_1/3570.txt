#include <iostream>
#include <vector>
#include <limits>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int& ai : a) cin >> ai;
  vector<int> k(m);
  for (int& ki : k) cin >> ki;
  vector<int> l(m), r(m);
  int b = -1, e = -1;
  for (int i = 0; i < n; i++) {
    int j = a[i] - 1;
    r[j] = i;
    if (l[j] == 0) l[j] = i;
  }
  for (int i = 0; i < m; i++) {
    if (l[i] > r[i] || k[i] == 0) {
      cout << -1;
      return 0;
    }
    if (l[i] <= b && r[i] >= e) continue;
    int al = l[i], ar = r[i];
    while (al > b || ar < e) {
      if (al > b) al--, b++;
      if (ar < e) ar++, e--;
    }
    if (e - b + 1 < k[i]) {
      cout << -1;
      return 0;
    }
  }
  cout << e - b + 1 - n;
  return 0;
}
