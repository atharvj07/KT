#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main(){
    int n, m, r; cin >> n >> m >> r;
    vector<int> s(n), b(m);
    for(int & i: s) cin >> i;
    for(int & i: b) cin >> i;
    sort(s.begin(), s.end()); sort(b.rbegin(), b.rend());
    for(int i = 0; s[i] <= r && i < n; ++i, r -= s[i]);
    for(int i = 0; r >= b[i] && i < m; ++i) r += b[i] - s[r];
    cout << r << endl;
}
