#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) { cin >> h[i]; }
    bool increasing = true;
    for (int i = 1; i < n; i++) {
      if (h[i] < h[i - 1]) { increasing = false; }
    }
    if (increasing) {
      cout << "YES" << endl;
    } else {
      bool possible = false;
      for (int i = 1; i < n; i++) {
        if (h[i] > 0 && h[i - 1] == 0) {
          possible = true;
          break;
        }
      }
      if (possible) {
        cout << "YES" << endl;
      } else {
        cout << "NO" << endl;
      }
    }
  }
}
