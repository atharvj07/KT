#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> f(101, 0);
  for (int i = 0; i < n; i++) f[a[i]]++;
  bool ok = false;
  for (int i = 1; i <= 100; i++) {
    if (f[i] % 2 == 0) {
      f[i] /= 2;
      if (ok) {
        cout << "NO" << endl;
        return 0;
      }
      ok = true;
    } else {
      if (f[i] == 1) {
        ok = true;
      } else {
        cout << "NO" << endl;
        return 0;
      }
    }
  }
  if (!ok) {
    cout << "NO" << endl;
    return 0;
  }
  int x, y;
  for (int i = 1; i <= 100; i++) {
    if (f[i]) { x = i; } else { y = i; }
  }
  cout << "YES" << endl;
  cout << x << " " << y << endl;
  return 0;
}
