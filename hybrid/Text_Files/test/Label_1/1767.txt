#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 4;
int n, a[maxn], dp[maxn][2];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  string s;
  cin >> s;
  s = "0" + s;
  int m = s.size();
  dp[m][0] = dp[m][1] = 0;
  for (int i = m - 1; i >= 1; i--) {
    dp[i][0] = dp[i + 1][0] + a[i - 1] * (s[i] - '0');
    dp[i][1] = max(dp[i + 1][0], dp[i + 1][1]) + a[i - 1] * (s[i] - '0');
  }
  cout << max(dp[1][0], dp[1][1]);
}
