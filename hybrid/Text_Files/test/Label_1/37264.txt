n = int(input())
a = list(map(int, input().split()))
dp = [[[-1] * 3 for _ in range(2)] for _ in range(n + 1)]
def solve(i, sum, rem):
    if i == n:
        if sum % 9 != 0 or rem % 10 != 0:
            return -1
        return sum
    
    if dp[i][sum % 2][rem % 10] != -1:
        return dp[i][sum % 2][rem % 10]
    
    res = -1
    if a[i] != 0:
        res = solve(i + 1, sum + a[i], (rem * 10 + a[i]) % 10)
    if a[i] != 5:
        res = max(res, solve(i + 1, sum, rem))
    
    dp[i][sum % 2][rem % 10] = res
    return res
print(solve(0, 0, 0))
