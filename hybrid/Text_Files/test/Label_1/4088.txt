#include<iostream>
using namespace std;
int main(){
    int n, arr[1005], ans[1005], t = 0;
    cin >> n;
    for(int i = 0; i < n; i++) cin >> arr[i];
    for(int i = 0; i < n; i++){
        for(int j = arr[i]; j; j--) ans[t++] = j;
    }
    cout << t << endl;
    for(int i = 0; i < t; i++) cout << ans[i] << ' ';
    return 0;
}
