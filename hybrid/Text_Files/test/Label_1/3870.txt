#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  long long ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] != 1) {
      if (a[i] == -1)
        ans++;
      else
        ans += abs(a[i] - 1) + 1;
    }
  }
  cout << ans << endl;
  return 0;
}
