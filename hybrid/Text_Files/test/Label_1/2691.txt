#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1e9 + 5;
int dx[] = {0, 1}, dy[] = {1, 0};
struct node {
    int x, y, t;
    node(int a, int b, int c) : x(a), y(b), t(c) {}
};
int main() {
    int n, m;
    cin >> n >> m;
    vector<pair<int, int>> vul(m);
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        vul[i] = {x, y};
    }
    queue<node> q;
    q.push(node(1, 1, 0));
    int dist[n + 5][n + 5];
    memset(dist, -1, sizeof(dist));
    dist[1][1] = 0;
    while (!q.empty()) {
        int x = q.front().x, y = q.front().y, t = q.front().t;
        q.pop();
        for (int i = 0; i < 2; i++) {
            int nx = x + dx[i], ny = y + dy[i], nt = t + 1;
            if (nx > n || ny > n) continue;
            bool ok = 1;
            for (auto p : vul) ok &= (nx != p.first || ny != p.second);
            if (ok && dist[nx][ny] == -1) {
                dist[nx][ny] = nt;
                q.push(node(nx, ny, nt));
            }
        }
    }
    cout << dist[n][n] << endl;
    return 0;
}
