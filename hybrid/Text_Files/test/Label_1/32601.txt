import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int d = sc.nextInt();
            int k = sc.nextInt();
            if (k * k > d * d) {
                System.out.println("Utkarsh");
            } else {
                System.out.println("Ashish");
            }
        }
    }
}
