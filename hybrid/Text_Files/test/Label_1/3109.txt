#include <bits/stdc++.h>
using namespace std;
char a[205];
int main() {
  int n;
  cin >> n;
  cin >> a;
  int t = n / 2;
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] == 'X') t--;
    if (t < 0) {
      ans = i + 1;
      break;
    }
  }
  cout << ans << endl;
  t = n / 2;
  for (int i = 0; i < n; ++i) {
    if (a[i] == 'X') {
      t--;
      if (t < 0) {
        a[i] = 'x';
      }
    }
  }
  cout << a << endl;
  return 0;
}
