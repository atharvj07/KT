import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> map = new TreeMap<>();
        for (int i = 0; i < n; i++) {
            String name = sc.next();
            int type = sc.nextInt();
            map.put(name, type);
        }

        int e = 0;
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            if (entry.getValue() == 1) e++;
        }

        System.out.println(e);
        int count = 1;
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            if (entry.getValue() == 1) {
                System.out.println("move " + entry.getKey() + " " + count);
                count++;
            }
        }

        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            if (entry.getValue() == 0) {
                System.out.println("move " + entry.getKey() + " " + count);
                count++;
            }
        }
    }
}
