#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const int maxn = 2e5 + 10;
int n, k;
int a[maxn];
ll dp[maxn][2]; // dp[i][0] = end at i, dp[i][1] = start at i

int main() {
  cin >> n >> k;
  for (int i = 1; i <= k; ++i) {
    cin >> a[i];
  }
  dp[a[1]][1] = 1;
  dp[a[1] - 1][0] = 1;
  ll ans = 0;
  for (int i = 2; i <= k; ++i) {
    dp[a[i]][1] = dp[a[i] - 1][0];
    dp[a[i] + 1][0] = dp[a[i]][1];
  }
  for (int i = 1; i <= n; ++i) {
    ans += dp[i][0] * dp[i][1];
  }
  cout << ans << endl;
  return 0;
}
