import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] d = new int[n - 1];
        for (int i = 0; i < d.length; i++) {
            d[i] = sc.nextInt();
        }
        int a = sc.nextInt();
        int b = sc.nextInt();
        int ans = 0;
        for (int i = a - 1; i < b - 1; i++) {
            ans += d[i];
        }
        System.out.println(ans);
    }
}
