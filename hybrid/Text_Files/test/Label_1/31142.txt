import java.io.*;
import java.util.*;

public class Main {
    public static long MOD = 1000000007;
    public static long[] x;
    public static long[] y;
    public static int n;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        x = new long[n];
        y = new long[n];
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            x[i] = Long.parseLong(s[0]);
            y[i] = Long.parseLong(s[1]);
        }
        long perimeter = 0;
        for (int i = 0; i < n; i++) {
            perimeter += Math.sqrt((x[(i + 1) % n] - x[i]) * (x[(i + 1) % n] - x[i]) + (y[(i + 1) % n] - y[i]) * (y[(i + 1) % n] - y[i]));
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 2; j < n; j++) {
                long a = orient(i, j, (i + 1) % n);
                long b = orient(i, j, (j + 1) % n);
                if (a == 0 && b == 0) continue;
                if (a == 0 && b != 0) sum++;
            }
        }
        System.out.println(sum * perimeter % MOD);
    }

    public static long orient(int p, int q, int r) {
        return (x[q] - x[p]) * (y[r] - y[q]) - (x[r] - x[q]) * (y[q] - y[p]);
    }
}
