#include <iostream>

using namespace std;

int main() {
  string rebus;
  cin >> rebus;
  int n;
  cin >> n;
  int values[100], pos = 0, neg = 0, op = 0;
  for (int i = 0; i < rebus.size(); i++) {
    if (rebus[i] == '?') values[pos++] = i;
    else if (rebus[i] == '+') op++;
    else if (rebus[i] == '-') op--, neg++;
  }
  if (pos != 2 * op + 1 || neg > op) {
    cout << "Impossible" << endl;
    return 0;
  }
  for (int i = 0; i < n; i++) {
    int v1 = i + 1, v2 = n - i;
    if (v1 != v2 || op == 0 && neg == 0) {
      values[--pos] = v1;
      values[--pos] = v2;
      cout << "Possible" << endl;
      string ans = rebus;
      for (int j = 0; j < pos; j++) {
        ans[values[j]] = (char)('0' + v1);
        v1++;
      }
      cout << ans << endl;
      return 0;
    }
  }
  cout << "Impossible" << endl;
  return 0;
}
