#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int mod=1e9+7;
const int mxn=3e5+5;
vector<ll>adj[mxn];
ll sub[mxn],sum;
ll n,m,u,v;
bool isspine;
void dfs1(ll node,ll par){
    sub[node]=1;
    for(ll child:adj[node]){
        if(child==par) continue;
        dfs1(child,node);
        sub[node]+=sub[child];
    }
}
void dfs2(ll node,ll par,ll sz){
    for(ll child:adj[node]){
        if(child==par) continue;
        if(2*sub[child]>sz){
            isspine=true;
            return;
        }
        dfs2(child,node,sz);
    }
}
vector<pair<ll,ll>>vec;
ll x=1,y=1;
void dfs3(ll node,ll par){
    vec.push_back({x,y});
    for(ll child:adj[node]){
        if(child==par) continue;
        if(x<y) x++;
        else y++;
        dfs3(child,node);
    }
}
int main(){
    cin>>n;
    for(ll i=1;i<n;i++) {
       cin>>u>>v;
       adj[u].push_back(v);
       adj[v].push_back(u);
    }
    dfs1(1,1);
    dfs2(1,1,n);
    if(isspine) cout<<"NO\n";
    else{
        cout<<"YES\n";
        dfs3(1,1);
        for(auto pp:vec) cout<<pp.first<<" "<<pp.second<<endl;
    }
    return 0;
}
