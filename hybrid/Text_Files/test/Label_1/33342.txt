import java.util.*;
class Main {
    static final long mod = 1000000007;
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		String s = sc.next();
		long ans = 1;
		char[] sa = s.toCharArray();
		int n = s.length();
		int a = 0;
		for (int i = 0; i < n; i++) {
			if (sa[i] == 'a') {
				a++;
			} else {
				ans = (ans * a) % mod;
				a = 0;
			}
		}
		ans = (ans * a) % mod;
		ans = (ans * a) % mod;
		System.out.println(ans);
	}
}
