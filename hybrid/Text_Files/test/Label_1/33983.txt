import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        int[] arr = new int[n];
        int max = 0;
        int cost = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            max = Math.max(max, arr[i]);
        }

        for (int i = 0; i < n; i++) {
            if (arr[i] > 1) {
                cost += y;
            } else if (arr[i] == 1) {
                cost += Math.min(x, y);
            }
        }

        System.out.println(cost);
    }
}
