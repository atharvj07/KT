#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < (n); ++i)
#define repu(i, a, b) for(int i = (a); i <= (b); ++i)

typedef long long ll;
typedef pair<int, int> pint;
typedef vector<pint> pintvec;
typedef vector<int> intvec;
typedef vector<ll> llvec;

const int mod = 1e9 + 7;
const double eps = 1e-9;
const ll inf = 1LL << 60;

int n;
int mat[180][180];

void solve() {
    rep(i, n) rep(j, n) {
        double avg = (mat[i][j] + mat[j][i]) / 2.0;
        mat[i][j] = round(avg);
        mat[j][i] = round(mat[i][j] - avg);
    }
    rep(i, n) {
        rep(j, n) cout << fixed << setprecision(9) << mat[i][j] << " ";
        cout << endl;
    }
    rep(i, n) {
        rep(j, n) cout << fixed << setprecision(9) << mat[j][i] << " ";
        cout << endl;
    }
}

int main() {
    cin >> n;
    rep(i, n) rep(j, n) cin >> mat[i][j];
    solve();
}
