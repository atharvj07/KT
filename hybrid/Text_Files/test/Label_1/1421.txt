#include <bits/stdc++.h>
using namespace std;

const int N = 1005;

int a[N], n, m;
long long ans;

struct edge {
    int u, v, w;
    int vis;
    bool operator<(const edge &e) const {
        return w < e.w;
    }
} e[N * N];

int f[N];
int find(int x) {
    if (x != f[x]) f[x] = find(f[x]);
    return f[x];
}

void init() {
    for (int i = 1; i <= n; i++) f[i] = i;
    sort(e + 1, e + 1 + m);
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= m; i++) {
        cin >> e[i].u >> e[i].v >> e[i].w;
    }
    init();
    for (int i = 1; i <= m; i++) {
        int u = e[i].u, v = e[i].v, w = e[i].w;
        int fx = find(u), fy = find(v);
        if (fx != fy) {
            ans += w;
            f[fx] = fy;
            ans -= a[u];
            ans -= a[v];
        }
    }
    cout << ans << endl;
    return 0;
}
