import java.util.*;

public class Main {
    static List<Task> tasks;
    static int n, q;

    static class Task implements Comparable<Task> {
        int time, ki, di, ans;

        Task(int time, int ki, int di) {
            this.time = time;
            this.ki = ki;
            this.di = di;
        }

        @Override
        public int compareTo(Task other) {
            return this.time - other.time;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        q = sc.nextInt();
        tasks = new ArrayList<>();
        for (int i = 0; i < q; i++)
            tasks.add(new Task(sc.nextInt(), sc.nextInt(), sc.nextInt()));

        Collections.sort(tasks);
        List<Integer> busy = new ArrayList<>();
        for (Task task : tasks) {
            int sum = 0;
            while (busy.size() < task.ki && !busy.isEmpty()) {
                int cur = busy.get(0);
                if (cur + task.di - 1 >= task.time)
                    break;
                busy.remove(0);
                sum += cur;
            }
            for (int i = 0; i < task.ki; i++) {
                if (busy.isEmpty() || busy.get(0) + task.di - 1 >= task.time) {
                    busy.add(task.time + task.di);
                    sum += (i + 1);
                }
            }
            task.ans = sum;
        }
        for (Task task : tasks)
            System.out.println(task.ans == 0 ? -1 : task.ans);
    }
}
