import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur + a[i] >= k) {
                break;
            }
            cur += a[i];
            ans++;
        }
        System.out.println(ans);
    }
}
