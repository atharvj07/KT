t = int(input())
for _ in range(t):
    x0, n = map(int, input().split())
    if n == 0:
        print(x0)
    else:
        remainder = n % 4
        if remainder == 0:
            print(x0)
        elif remainder == 1:
            if x0 % 2 == 0:
                print(x0 - n)
            else:
                print(x0 + n)
        elif remainder == 2:
            if x0 % 2 == 0:
                print(x0 + 1)
            else:
                print(x0 - 1)
        else:
            if x0 % 2 == 0:
                print(x0 + n + 1)
            else:
                print(x0 - (n + 1))
