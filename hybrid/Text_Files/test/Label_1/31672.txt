import java.util.Scanner;

public class BerryBaskets {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int maxBaskets = 0;
        for (int i = 0; i < n; i++) {
            int red = scanner.nextInt();
            int blue = scanner.nextInt();

            // Calculate baskets for same color berries
            maxBaskets += Math.min(red / k, blue / k);

            // Calculate baskets for same shrub berries
            maxBaskets += Math.min(red + blue, k) / k;
        }

        System.out.println(maxBaskets);
    }
}
