import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt(),
        k=sc.nextInt();
        long[] a=new long[n],pre=new long[n];
        for(int i=0;i<n;a[i]=sc.nextLong(),i++);
        for(int i=1;i<n;i++)
            pre[i]=pre[i-1]+a[i];
        long cur=Long.MAX_VALUE,ans=0;
        for(int i=0;i<=n-k;i++)
            if(cur>pre[i]+pre[n-1]-pre[i+k-1]){
                cur=pre[i]+pre[n-1]-pre[i+k-1];
                ans=i+1;
            }
        System.out.println(ans);
    }
}
