n = int(input())
total_damage = 0
cards_played = 0

for _ in range(n):
    k = int(input())
    cards = [list(map(int, input().split())) for _ in range(k)]
    cards.sort(key=lambda x: (-x[1], x[0]))

    dp = [0] * 4
    for c, d in cards:
        for j in range(3, c - 1, -1):
            dp[j] = max(dp[j], dp[j - c] + d)

    best_turn_damage = max(dp)
    if cards_played % 10 == 9:
        best_turn_damage *= 2
    total_damage += best_turn_damage
    cards_played += 1

print(total_damage)
