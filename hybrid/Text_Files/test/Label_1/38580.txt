from collections import defaultdict, deque

def topological_sort(n, edges):
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    for v, u in edges:
        graph[u].append(v)
        in_degree[v] += 1

    q = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            q.append(i)

    order = []
    while q:
        u = q.pop()
        order.append(u)
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)

    return order[::-1]

def solve(n, m):
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    order = topological_sort(n, edges)
    label = [0] * n
    for i, v in enumerate(order):
        label[v-1] = n - i
    return label

n, m = map(int, input().split())
print(*solve(n, m))
