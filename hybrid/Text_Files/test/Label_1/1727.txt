#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  long long n, a, ans = 0;
  cin >> n;
  for (long long i = 0; i < n; i++) {
    cin >> a;
    ans += (a % 2);
  }
  if (ans % 2 == 0)
    ans += n % 2;
  else
    ans += (n % 2) - 1, ans += n % 2 > 0;
  cout << ans << endl;
  return 0;
}
