#include <vector>
#include <queue>
#include <iostream>
using namespace std;
#define MAX_N 200005

int N, ans = 2E9;
int subTreeSz[MAX_N], cen;
int dist[MAX_N];
vector<int> adj[MAX_N];

void dfs1(int u, int p) {
    subTreeSz[u] = 1;
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs1(v, u);
        subTreeSz[u] += subTreeSz[v];
    }
}
void dfs2(int u, int p, int sum) {
    ans = min(ans, sum);
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs2(v, u, sum + N - 2 * subTreeSz[v]);
    }
}
int findCen(int u, int p) {
    dfs1(u, p);

    queue<int> Q;
    Q.push(u);

    while (!Q.empty()) {
        int cur = Q.front();
        Q.pop();
        bool isCen = true;
        for (int v : adj[cur]) {
            if (v == p) continue;
            if (subTreeSz[v] > N / 2) isCen = false;
            Q.push(v);
        }
        if (isCen) {
            cen = cur;
            return cur;
        }
    }

    return -1;
}
void bfs(int root) {
    queue<int> Q;
    Q.push(root);
    dist[root] = 0;

    while (!Q.empty()) {
        int cur = Q.front();
        Q.pop();
        for (int v : adj[cur]) {
            if (dist[v] != -1) continue;
            dist[v] = dist[cur] + 1;
            Q.push(v);
        }
    }
}
void solve() {
    cin >> N;
    for (int i = 0; i < N - 1; i++) {
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    findCen(1, 1);
    bfs(cen);

    int mxDist = -1;
    for (int i = 1; i <= N; i++)
        mxDist = max(mxDist, dist[i]);

    cout << mxDist << "\n";
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    solve();

    return 0;
}
