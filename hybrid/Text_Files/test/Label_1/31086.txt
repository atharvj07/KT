import java.util.Scanner;
class Main {
    private static int[] seq;
    private static int[] result;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        seq = new int[n];
        result = new int[n];
        String line = sc.nextLine();
        String[] strs = line.split(" ");
        for (int i = 0; i < n; i++)
            seq[i] = Integer.valueOf(strs[i]);
        int cnt = 0;
        boolean flag = false;
        while (cnt < n) {
            cnt++;
            int[] pre = new int[n];
            int[] succ = new int[n];
            for (int i = 0; i < n; i++) {
                int prev = (i - 1 + n) % n;
                int next = (i + 1) % n;
                if (i == 0)
                    pre[i] = 0;
                else
                    pre[i] = seq[prev];
                if (i == n - 1)
                    succ[i] = 0;
                else
                    succ[i] = seq[next];
            }
            for (int i = 0; i < n; i++) {
                int p = pre[i];
                int s = succ[i];
                int[] x = new int[] {p, seq[i], s};
                result[i] = median(x);
            }
            for (int i = 0; i < n; i++) {
                if (seq[i] != result[i]) {
                    flag = true;
                    break;
                }
            }
            if (!flag)
                break;
            seq = result;
        }
        if (flag)
            System.out.println(1);
        else
            System.out.printf("%d\n", cnt);
        for (int x : result)
            System.out.print(x + " ");
        System.out.println();
    }
    private static int median(int[] arr) {
        int[] clone = arr.clone();
        java.util.Arrays.sort(clone);
        return clone[1];
    }
}
