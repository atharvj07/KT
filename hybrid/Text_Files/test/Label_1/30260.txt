import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Get the number of songs and the number of psyllium sticks
        int n = sc.nextInt();
        int m = sc.nextInt();

        // Initialize the maximum satisfaction level to 0
        int maxSatisfaction = 0;

        // For each song
        for (int i = 0; i < n; i++) {
            // Get the satisfaction levels for shaking level 2 and level 1 psyllium sticks, and for not shaking any psyllium sticks
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();

            // If the satisfaction level for shaking a level 2 psyllium stick is greater than the maximum satisfaction level, update the maximum satisfaction level
            if (a > maxSatisfaction) {
                maxSatisfaction = a;
            }

            // If the satisfaction level for shaking a level 1 psyllium stick is greater than the maximum satisfaction level, update the maximum satisfaction level
            if (b > maxSatisfaction) {
                maxSatisfaction = b;
            }

            // If the satisfaction level for not shaking any psyllium sticks is greater than the maximum satisfaction level, update the maximum satisfaction level
            if (c > maxSatisfaction) {
                maxSatisfaction = c;
            }

            // If the number of psyllium sticks is greater than 0, update the maximum satisfaction level to include the satisfaction level for shaking a level 2 psyllium stick
            if (m > 0) {
                maxSatisfaction += a;
                m--;
            }

            // If the number of psyllium sticks is greater than 0, update the maximum satisfaction level to include the satisfaction level for shaking a level 1 psyllium stick
            if (m > 0) {
                maxSatisfaction += b;
                m--;
            }
        }

        // Output the maximum satisfaction level
        System.out.println(maxSatisfaction);
    }
}
