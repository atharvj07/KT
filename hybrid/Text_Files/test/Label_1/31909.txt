import java.util.*;

public class Main {
    static class Edge {
        int to;
        int idx;

        Edge(int to, int idx) {
            this.to = to;
            this.idx = idx;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        List<Edge>[] g = new List[N + 1];
        for (int i = 1; i <= N; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < M; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            g[a].add(new Edge(b, i + 1));
            g[b].add(new Edge(a, i + 1));
        }
        boolean[] visited = new boolean[M + 1];
        int[] color = new int[N + 1];
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        color[1] = 1;
        int ans = 0;
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (Edge e : g[cur]) {
                if (!visited[e.idx]) {
                    visited[e.idx] = true;
                    if (color[cur] == color[e.to]) {
                        ans++;
                    }
                    color[e.to] = 3 - color[cur];
                    q.add(e.to);
                }
            }
        }
        for (int i = 1; i <= N; i++) {
            if (color[i] == 0) {
                System.out.println(-1);
                return;
            }
        }
        System.out.println(ans);
    }
}
