from collections import Counter
s = input()
cnt = Counter(s)
odd_cnt = 0
odd = ''
for i in cnt:
    if cnt[i] % 2 == 1:
        odd_cnt += 1
        odd = i
ans = ''
for i in cnt:
    if i == odd and cnt[odd] > 1:
        ans += i * (cnt[odd] - 1)
        cnt[odd] -= cnt[odd] - 1
        odd_cnt -= 1
    if cnt[i] > 0:
        ans += i * (cnt[i] // 2)
        cnt[i] -= cnt[i] // 2
if odd_cnt > 0:
    ans += odd
for i in cnt:
    if i == odd and cnt[odd] > 1:
        ans += i * (cnt[odd] - 1)
        cnt[odd] -= cnt[odd] - 1
        odd_cnt -= 1
    if cnt[i] > 0:
        ans += i * (cnt[i] // 2)
        cnt[i] -= cnt[i] // 2
print(ans)
