import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for(int i=0; i<n; i++) a[i] = in.nextInt();
        Arrays.sort(a);
        int l = 0, r = n-1, ans = 1;
        while(l<r){
            if(a[r]-a[l]>1){
                r--;
            }
            else{
                ans = Math.max(ans, r-l+1);
                l++;
            }
        }
        System.out.println(ans);
        l = 0; r = n-1;
        while(l<r){
            if(a[r]-a[l]>1){
                r--;
            }
            else{
                if(ans == r-l+1){
                    for(int i=l; i<=r; i++) System.out.print(a[i]+" ");
                    return;
                }
                l++;
            }
        }
    }
}
