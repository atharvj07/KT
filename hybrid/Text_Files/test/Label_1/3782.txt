#include <iostream>
#include <vector>

std::vector<int> mex(const std::vector<int>& a) {
  int n = a.size();
  std::vector<int> b(n + 1, 0);
  std::vector<int> mex(n);
  for (int i = n - 1; i >= 0; --i) {
    b[a[i]] = 1;
    for (int j = 0; j <= n; ++j) {
      if (!b[j]) {
        mex[i] = j;
        break;
      }
    }
  }
  return mex;
}

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  auto b = mex(a);
  if (b[0] != 0) {
    std::cout << -1;
  } else {
    for (const auto& x : b) {
      std::cout << x << ' ';
    }
  }
  std::cout << std::endl;
  return 0;
}
