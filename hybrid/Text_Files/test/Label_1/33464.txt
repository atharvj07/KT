import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        if (n < 4) {
            System.out.println(-1);
        } else if (n % 4 == 0) {
            System.out.println(4.0 * n / 4);
        } else if (n % 7 == 0) {
            System.out.println(7.0 * n / 7);
        } else if (n % 4 > n % 7) {
            System.out.println(4 + (n - 4) / 7 * 7);
        } else {
            System.out.println(7 + (n - 7) / 4 * 4);
        }
    }
}
