#include <bits/stdc++.h>
using namespace std;

const int START = 1;
const int MAXN = 100;
string s[MAXN];
int dp[MAXN][MAXN];
int main() {
    int n, m;
    cin >> n;
    for (int i = 0; i < n; ++i) {
        cin >> s[i];
        dp[i][i] = 1;
    }
    for (int k = 1; k < MAXN; ++k) {
        for (int i = 0; i <= n - k; ++i) {
            int j = i + k - 1;
            if (s[i] == s[j]) {
                dp[i][j] = dp[i][j - 1];
            } else {
                dp[i][j] = 0;
            }
        }
    }
    cin >> m;
    for (int i = 0; i < m; ++i) {
        int a, b;
        cin >> a >> b;
        --a, --b;
        int low = 0, high = MAXN - 1, mid, ans = 0;
        while (low <= high) {
            mid = (low + high) >> 1;
            if (dp[a][a + mid - 1] && dp[b][b + mid - 1]) {
                ans = mid;
                low = mid + 1;
            } else
                high = mid - 1;
        }
        cout << ans << endl;
    }
    return 0;
}
