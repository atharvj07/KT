#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> absurdity(n);
  for (int i = 0; i < n; i++) cin >> absurdity[i];
  long long absurditySum = 0;
  for (int i = 0; i < k; i++) absurditySum += absurdity[i];
  long long maxSum = absurditySum;
  int maxA = 0;
  int maxB = k;
  for (int i = 1; i < n - k + 1; i++) {
    absurditySum = absurditySum - absurdity[i - 1] + absurdity[i + k - 1];
    if (absurditySum > maxSum) {
      maxSum = absurditySum;
      maxA = i;
      maxB = i + k;
    }
  }
  cout << maxA << " " << maxB << endl;
  return 0;
}
