#include <cstdio>
#include <vector>
#include <cstring>

using ll = long long;
constexpr ll mod = 1000000007;

int main() {
  std::vector<int> pw(300005, 1);
  for (int i = 1; i < 300005; i++) pw[i] = 10 * pw[i - 1] % mod;
  char s[300005];
  scanf("%s", s);
  ll dp[300005];
  dp[0] = (s[0] - '0') % 4 == 0;
  for (int i = 1; s[i]; i++) {
    dp[i] = dp[i - 1] * 10 % mod;
    dp[i] = (dp[i] + (s[i] - '0') * pw[(i + 1) % 4]) % mod;
  }
  printf("%lld", dp[strlen(s) - 1]);
  return 0;
}
