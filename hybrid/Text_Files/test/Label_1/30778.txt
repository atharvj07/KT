import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        boolean[] b = new boolean[n + 1];
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt();
            b[x] = !b[x];
            int ans = 0;
            for (int j = 1; j <= n; j++) {
                if (b[j]) {
                    for (int k = j + 1; k <= n; k++) {
                        if (b[k]) {
                            ans += gcd(a[j], a[k]);
                        }
                    }
                }
            }
            System.out.println(ans);
        }
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        } else {
            return gcd(b, a % b);
        }
    }
}
