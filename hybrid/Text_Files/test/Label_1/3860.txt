#include <bits/stdc++.h>
using namespace std;
int a[100005];
bool ok;
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 2; i <= n; i++) {
    if (a[i] ^ a[i - 1]) {
      if (ok) {
        cout << "-1\n";
        return 0;
      }
      ok = 1;
    }
  }
  if (!ok) {
    cout << "-1\n";
    return 0;
  }
  int ans = 0;
  for (int i = 2; i <= n; i++) {
    if (a[i] ^ a[i - 1]) { ans += 2; }
  }
  cout << ans / 2 << endl;
  return 0;
}
