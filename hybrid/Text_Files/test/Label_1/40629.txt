from collections import deque

def bfs(start, graph, d, police_stations):
    queue = deque([(start, 0)])
    visited = set()
    roads_to_keep = set()
    
    while queue:
        current, dist = queue.popleft()
        if current in visited:
            continue
        visited.add(current)
        if dist > d:
            break
        for next_city, road_index in graph[current]:
            if next_city not in visited:
                if dist + 1 <= d:
                    roads_to_keep.add(road_index)
                    queue.append((next_city, dist + 1))
                if next_city in police_stations:
                    queue.append((next_city, 0))
    return roads_to_keep

def max_roads_to_shut_down(n, k, d, police_stations, roads):
    graph = {i: [] for i in range(1, n + 1)}
    for index, (u, v) in enumerate(roads, 1):
        graph[u].append((v, index))
        graph[v].append((u, index))
    
    all_roads = set(range(1, n))
    roads_to_keep = set()
    for station in police_stations:
        roads_to_keep.update(bfs(station, graph, d, set(police_stations)))
    
    roads_to_shut_down = all_roads - roads_to_keep
    print(len(roads_to_shut_down))
    print(*sorted(roads_to_shut_down))

n, k, d = map(int, input().split())
police_stations = list(map(int, input().split()))
roads = [tuple(map(int, input().split())) for _ in range(n - 1)]

max_roads_to_shut_down(n, k, d, police_stations, roads)
