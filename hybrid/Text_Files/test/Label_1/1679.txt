#include <bits/stdc++.h>
using namespace std;
char s[9][9];
int main() {
  for (int i = 1; i <= 8; ++i) { cin >> s[i] + 1; }
  int a = 0, b = 0;
  for (int i = 1; i <= 8; ++i) {
    for (int j = 1; j <= 8; ++j) {
      if (s[i][j] == 'B') {
        ++b;
        if (s[i][j] == 'B' && i == 7) return cout << "B", 0;
      } else if (s[i][j] == 'W') {
        ++a;
        if (i == 2) return cout << "A", 0;
      }
    }
  }
  if (b > a)
    return cout << "B", 0;
  cout << "A";
  return 0;
}
