n, c, d = map(int, input().split())
f = []
for _ in range(n):
    b, p, t = input().split()
    if t == 'C':
        f.append((int(b), int(p), False))
    else:
        f.append((int(b), int(p), True))
f.sort(key=lambda x: x[0], reverse=True)
cur_c, cur_d = c, d
ans = 0
for b, p, t in f:
    if t:
        if cur_d >= p:
            ans += b
            cur_d -= p
    else:
        if cur_c >= p:
            ans += b
            cur_c -= p
print(ans)
