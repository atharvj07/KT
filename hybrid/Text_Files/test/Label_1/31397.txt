import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Rook {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Number of vertical spells
        int m = scanner.nextInt(); // Number of horizontal spells

        List<Integer> verticalSpells = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            verticalSpells.add(scanner.nextInt());
        }

        List<HorizontalSpell> horizontalSpells = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x1 = scanner.nextInt();
            int x2 = scanner.nextInt();
            int y = scanner.nextInt();
            horizontalSpells.add(new HorizontalSpell(x1, x2, y));
        }

        // Sort horizontal spells by y coordinate in ascending order
        Collections.sort(horizontalSpells);

        int minSpellsToRemove = 0;
        int currentRow = 1;

        // Iterate through horizontal spells
        for (HorizontalSpell spell : horizontalSpells) {
            // If the spell is above the current row, ignore it
            if (spell.y > currentRow) {
                continue;
            }

            // Find the first vertical spell that intersects the current horizontal spell
            int firstIntersectingSpell = findFirstIntersectingSpell(verticalSpells, spell.x1, spell.x2);

            // If there is an intersecting spell, remove it and update the current row
            if (firstIntersectingSpell != -1) {
                minSpellsToRemove++;
                verticalSpells.remove(firstIntersectingSpell);
                currentRow = spell.y + 1;
            }
        }

        System.out.println(minSpellsToRemove);
    }

    // Find the first vertical spell that intersects the given horizontal spell
    private static int findFirstIntersectingSpell(List<Integer> verticalSpells, int x1, int x2) {
        int left = 0;
        int right = verticalSpells.size() - 1;

        while (left <= right) {
            int mid = (left + right) / 2;
            int spellX = verticalSpells.get(mid);

            if (spellX >= x1 && spellX <= x2) {
                return mid;
            } else if (spellX < x1) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }

        return -1;
    }

    // Horizontal spell class
    private static class HorizontalSpell implements Comparable<HorizontalSpell> {
        int x1;
        int x2;
        int y;

        public HorizontalSpell(int x1, int x2, int y) {
            this.x1 = x1;
            this.x2 = x2;
            this.y = y;
        }

        @Override
        public int compareTo(HorizontalSpell other) {
            return Integer.compare(y, other.y);
        }
    }
}
