#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (long long i = 0; i < n; ++i) cin >> a[i];
  partial_sum(a.begin(), a.end(), a.begin());
  long long mx = a[k - 1], ansL = 0, ansR = k;
  for (long long i = k; i < n; ++i) {
    if (mx < a[i] - a[i - k]) {
      mx = a[i] - a[i - k];
      ansL = i - k + 1;
      ansR = i + 1;
    }
  }
  cout << ansL << ' ' << ansR << endl;
}
