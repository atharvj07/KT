import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int s = sc.nextInt();
        if (s % 2 == (Math.abs(a) + Math.abs(b)) % 2) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
