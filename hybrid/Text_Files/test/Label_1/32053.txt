import java.util.Scanner;

public class TreeProblem {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        if (isTreePossible(n, c)) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isTreePossible(int n, int[] c) {
        // Check if the sum of subtree sizes is equal to n
        int sum = 0;
        for (int value : c) {
            sum += value;
        }
        if (sum != n) {
            return false;
        }

        // Check if it's possible to build a valid tree with the given subtree sizes
        int[] degree = new int[n];
        for (int i = 0; i < n; i++) {
            degree[c[i] - 1]++;
        }

        for (int i = 0; i < n; i++) {
            if (degree[i] > 1 && i > 0) {
                return false;
            }
        }

        return true;
    }
}
