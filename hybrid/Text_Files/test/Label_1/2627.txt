#include <iostream>
#include <cstring>
using namespace std;

int main() {
    int t;
    cin >> t;
    while(t--) {
        long long l, r;
        cin >> l >> r;
        l = min(l, r - l);
        string s = to_string(l);
        int n = s.size();
        int ans = n;
        for(int i = n - 2; i >= 0; i--) {
            if(s[i] == s[i + 1] && s[i] == '9') {
                ans--;
                continue;
            }
            else if(s[i] == s[i + 1] && s[i] < '9') {
                ans += n - i - 1;
                break;
            }
            else {
                ans += n - i - 1;
                break;
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
