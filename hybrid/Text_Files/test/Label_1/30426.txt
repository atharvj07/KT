import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();

        int[] dp = new int[n];
        dp[0] = 1;
        int mx = 1;
        for (int i = 1; i < n; i++) {
            dp[i] = 1;
            for (int j = 0; j < i; j++) {
                if (a[i] > a[j]) dp[i] = Math.max(dp[i], dp[j] + 1);
            }
            mx = Math.max(mx, dp[i]);
        }

        System.out.println(mx);
        int[] c = new int[mx];
        int l = mx - 1;
        for (int i = n - 1; i >= 0; i--) {
            if (dp[i] == mx) {
                c[l--] = a[i];
                mx--;
            }
        }

        for (int i = 0; i < c.length; i++) System.out.print(c[i] + " ");
    }
}
