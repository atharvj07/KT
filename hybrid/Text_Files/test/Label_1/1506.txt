#include <bits/stdc++.h>
using namespace std;
const int mod = 1e9 + 7;
using ll = long long;

ll power(ll x, ll y) {
    ll res = 1;
    while (y > 0) {
        if (y % 2 == 1) res = (res * x) % mod;
        x = (x * x) % mod;
        y = y / 2;
    }
    return res;
}

vector<ll> factorial;

void preprocess(ll n) {
    factorial.push_back(1);
    for (ll i = 1; i <= n; i++) {
        factorial.push_back(i * factorial[i - 1]);
    }
}

ll ncr(ll n, ll r) {
    if (n < 0 || r < 0 || n < r) return (ll)0;
    ll numerator = factorial[n];
    ll denominator = (factorial[r] * factorial[n - r]) % mod;
    ll res = (numerator * power(denominator, mod - 2)) % mod;
    return res;
}

int main() {
    ll n, p;
    cin >> n >> p;
    preprocess(n);
    vector<pair<ll, ll>> edge;
    for (ll i = 0; i < n; i++) {
        ll x, y;
        cin >> x >> y;
        edge.push_back({x, y});
    }
    unordered_map<ll, vector<pair<ll, ll>>> graph;
    for (ll i = 0; i < n; i++) {
        graph[edge[i].first].push_back({edge[i].second, i});
        graph[edge[i].second].push_back({edge[i].first, i});
    }
    ll res = 0;
    for (ll i = 1; i <= n; i++) {
        vector<pair<ll, ll>> adj = graph[i];
        ll count = 0;
        for (pair<ll, ll> u : adj) {
            ll x = u.first;
            count += (graph[x].size() >= p - 1);
        }
        if (count >= p - 1) { res += (graph[i].size() * count) % mod; }
    }
    res %= mod;
    cout << res << endl;
    return 0;
}
