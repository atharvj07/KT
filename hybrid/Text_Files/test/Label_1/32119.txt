import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int p = input.nextInt();
    String s = input.next();
    int[][] adj = new int[p][p];
    for (int i = 0; i < p; i++) {
      for (int j = 0; j < p; j++) {
        adj[i][j] = input.nextInt();
      }
    }
    int ans = Integer.MAX_VALUE;
    for (int i = 0; i < (1 << p); i++) {
      boolean valid = true;
      int cnt = 0;
      for (int j = 0; j < n; j++) {
        int curr = s.charAt(j) - 'a';
        if ((i & (1 << curr)) == 0) {
          cnt++;
          if (j > 0 && (i & (1 << (s.charAt(j - 1) - 'a'))) == 0) {
            valid = false;
            break;
          }
        }
      }
      if (valid) {
        ans = Math.min(ans, cnt);
      }
    }
    System.out.println(ans);
  }
}
