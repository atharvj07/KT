import sys
input = sys.stdin.readline

n = int(input())
a10 = []
a01 = []
a00 = []
a11 = []
for i in range(n):
    s, a = input().split()
    a = int(a)
    if s == "10": a10.append(a)
    elif s == "01": a01.append(a)
    elif s == "00": a00.append(a)
    elif s == "11": a11.append(a)
a10.sort(reverse=True)
a01.sort(reverse=True)
a00.sort(reverse=True)
a11.sort(reverse=True)
ans = 0
t01 = 0
t10 = 0
if len(a10) + len(a11) >= (n+1)//2:
    for x in a10:
        if t10 < (n+1)//2: ans += x; t10 += 1
        else: break
    for x in a11:
        if t10 < (n+1)//2: ans += x; t10 += 1
        else: break
if len(a01) + len(a11) >= (n+1)//2:
    t01 = 0
    for x in a01:
        if t01 < (n+1)//2: ans += x; t01 += 1
        else: break
    for x in a11:
        if t01 < (n+1)//2: ans += x; t01 += 1
        else: break
if t10 < (n+1)//2 or t01 < (n+1)//2: ans = 0
t = t01 + t10
if t > (n+1)//2:
    for x in a00:
        if t > (n+1)//2: ans += x; t -= 1
        else: break
print(ans)
