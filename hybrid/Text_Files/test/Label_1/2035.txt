#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int a[27] = {0};
  for (int i = 0; i < n; i++) a[s[i] - 'A']++;
  int ans = 0;
  for (int i = 0; i < 26; i++) {
    ans += a[i] > 1;
  }
  if (ans > k)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
