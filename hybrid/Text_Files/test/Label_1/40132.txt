from collections import defaultdict
from heapq import heapify, heappush, heappop

n, m = map(int, input().split())
s = input()
graph = defaultdict(list)
dist = [[-1,-1] for _ in range(26)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    graph[u].append(v)
for i in range(26):
    dist[i][0] = n
st = []
dist[ord(s[0])-ord('a')][0] = 0
heappush(st, [0, 0, s[0]])
ans = -1
while st:
    d, u, p = heappop(st)
    if d > ans:
        ans = d
    if u == n - 1:
        print(ans)
        exit()
    for v in graph[u]:
        if dist[ord(s[v])-ord('a')][0] == -1:
            dist[ord(s[v])-ord('a')][0] = d + 1
            heappush(st, [d + 1, v, s[v]])
print(-1)
