#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N = 1e6 + 5;
int n, arr[N], id[N], ans_id[N];

struct Node {
    int l, r, idx;
    bool operator<(const Node &b) const {
        return l < b.l;
    }
};
vector<Node> v;

void solve() {
    cin >> n;
    v.clear();
    for (int i = 1; i <= n; i++) {
        cin >> arr[i], id[arr[i]] = i;
        v.push_back({arr[i], arr[i], i});
    }
    sort(v.begin(), v.end());
    int last = -1;
    for (Node x : v) {
        if (x.l == last) continue;
        last = x.l;
        int l = x.idx, r = x.idx;
        while (r < n && v[r].l == v[r + 1].l && (id[v[r + 1].r] != l)) r++;
        if (r >= n) continue;
        ans_id[l] = v[r + 1].r;
        ans_id[v[r + 1].r] = l;
        last = v[r + 1].r;
    }
    for (int i = 1; i <= n; i++) {
        if (!ans_id[i]) {
            if (ans_id[arr[i]]) {
                ans_id[i] = ans_id[arr[i]];
            } else {
                ans_id[i] = arr[i];
            }
        }
    }
    vector<int> ans;
    for (int i = 1; i <= n; i++) {
        if (ans_id[i] != i) ans.push_back(i);
    }
    for (int x : ans) cout << arr[x] << ' ';
    cout << '\n';
}
int main() {
    ios::sync_with_stdio(false), cin.tie(0), cout.tie(0);
    int t;
    cin >> t;
    while (t--) solve();
    return 0;
}
