import sys

n, k = list(map(int, sys.stdin.readline().split()))
data = [list(map(int, sys.stdin.readline().strip())) for i in range(n)]
score = [data[i][0] * (1 << 6) + data[i][1] * (1 << 5) + data[i][2] * (1 << 4) + data[i][3] * (1 << 3) + data[i][4] * (1 << 2) + data[i][5] * (1 << 1) + data[i][6] for i in range(n)]
tot = sum(score)
if (tot + k) & 1:
    print(-1)
else:
    score.sort(reverse=True)
    mn = 0
    k_ = k
    i = 0
    while k_ and i < n:
        if score[i] & 1:
            mn += 1
            score[i] -= 1
            k_ -= 1
        i += 1
    score.sort(reverse=True)
    ans = 0
    for s in score:
        if k_ == 0:
            break
        for i in range(7):
            if s & (1 << i) and k_:
                s -= (1 << i)
                k_ -= 1
        s = min(9, s)
        ans = ans * 10 + s
    if k_ != 0:
        print(-1)
    else:
        for j in range(mn):
            ans *= 10
        print(ans)
