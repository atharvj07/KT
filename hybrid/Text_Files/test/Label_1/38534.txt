import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    a = list(map(int, input().split()))
    cnt0 = a.count(0)
    cnt1 = a.count(1)
    cnt2 = a.count(2)
    
    if cnt0 % 2 == 0:
        ans = cnt0 + cnt2 + min(cnt1, cnt2 + 1)
    else:
        ans = cnt0 + cnt2 + min(cnt1 - 1, cnt2)
    
    print(ans)

t = int(input())
for _ in range(t):
    solve()
