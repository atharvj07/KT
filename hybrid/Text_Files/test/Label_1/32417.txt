import java.util.*;
import java.io.*;

public class Main {

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt(), k = in.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = in.nextInt();
    }
    Arrays.sort(arr);
    int ans = 0;
    int cur = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (cur + arr[i] >= k) {
        break;
      }
      cur += arr[i];
      ans++;
    }
    System.out.println(ans + 1);
  }
}
