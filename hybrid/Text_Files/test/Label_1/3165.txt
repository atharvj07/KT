#include <bits/stdc++.h>

#define ll long long

using namespace std;

const ll mod = 1e9 + 7;

ll power(ll a, ll b)
{
    if (b == 0) return 1;
    if (b == 1) return a;
    if (b % 2 == 1) return (a * power(a, b - 1)) % mod;
    ll val = power(a, b / 2);
    val %= mod;
    return (val * val) % mod;
}

ll inv(ll a)
{ return power(a, mod - 2); }

int main()
{
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n);
    unordered_map<ll, ll> freq, psum;
    for (ll i = 0; i < n; i++)
    {
        cin >> a[i];
        freq[a[i]]++;
        psum[a[i]]++;
    }
    ll ans = 0;
    for (ll i = 0; i < n; i++)
    {
        ll cur = 0;
        for (ll j = 0; j < n; j++)
        {
            cur = (1ll * cur * 10 + a[j]) % k;
            if (cur % k == 0)
            {
                if (i == j) ans += freq[a[i]] - 1;
                else ans += freq[a[j]];
            }
        }
        freq[a[i]]--;
        psum[a[i]]--;
        for (auto it : psum)
        {
            ll x = (1ll * a[i] * it.first) % k;
            if (x % k == 0)
            {
                ans -= 1ll * freq[a[i]] * it.second;
                freq[a[i]] -= it.second;
            }
        }
    }
    cout << ans << endl;
    return 0;
}
