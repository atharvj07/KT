#include<bits/stdc++.h>
#include<vector>
using namespace std;

int n;
char a[55][55];
int dx[] = {0, 1, 0, -1, -1, -1, 1, 1};
int dy[] = {1, 0, -1, 0, 1, -1, -1, 1};
bool vis[55][55], b[55][55];

void dfs(int x, int y) {
    vis[x][y] = 1;
    for(int k = 0; k < 8; k++) {
        int nx = x + dx[k];
        int ny = y + dy[k];
        if(nx >= 1 && nx <= n && ny >= 1 && ny <= n) {
            if(a[nx][ny] == 'x') b[x][y] = 1;
            else if(a[nx][ny] == 'o' && !vis[nx][ny]) dfs(nx, ny);
        }
    }
}

int main() {
    cin >> n;
    for(int i = 1; i <= n; i++) cin >> a[i] + 1;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++) {
            if(a[i][j] == 'o' && !vis[i][j]) {
                dfs(i, j);
                break;
            }
        }
    }
    bool ok = 1;
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= n; j++) {
            if(a[i][j] == 'x' && !b[i][j]) ok = 0;
        }
    }
    cout << (ok ? "YES" : "NO") << endl;
    if(ok) {
        for(int i = -n; i <= n; i++) {
            for(int j = -n; j <= n; j++) {
                for(int k = 0; k < 8; k++) {
                    int nx = i + dx[k];
                    int ny = j + dy[k];
                    if(nx * nx + ny * ny <= n * n && nx >= -n && nx <= n && ny >= -n && ny <= n) {
                        cout << (b[i][j] ? 'x' : '.');
                    } else cout << '.';
                }
                cout << endl;
            }
        }
    }
}
