#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (pair<int, int> &p : v) {
    cin >> p.first >> p.second;
  }
  int ans = n;
  sort(v.begin(), v.end());
  for (int i = 1; i < n; ++i) {
    if (v[i].first != v[i - 1].first) ans--;
  }
  sort(v.begin(), v.end(), [](const pair<int, int> &lhs, const pair<int, int> &rhs) { return lhs.second < rhs.second; });
  for (int i = 1; i < n; ++i) {
    if (v[i].second != v[i - 1].second) ans--;
  }
  cout << ans << endl;
}
