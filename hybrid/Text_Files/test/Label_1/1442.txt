#include <iostream>
#include <vector>
using namespace std;
#define ll long long
const int N = (int)2e5 + 7;
vector<int> g[N];
int n, m;
void dfs(int u, int p) {
    for (int v : g[u]) {
        if (v == p) continue;
        dfs(v, u);
        g[u].pop_back();
        g[v].pop_back();
    }
}
int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) g[i].clear();
        for (int i = 0; i < m; i++) {
            int u, v;
            cin >> u >> v;
            g[u].push_back(v);
            g[v].push_back(u);
        }
        int k = (4 * n) / 7;
        cout << k << '\n';
        for (int i = 1; i <= n; i++) {
            if (i > n - k) {
                cout << i << ' ';
                continue;
            }
            int sz = g[i].size();
            if (sz == 0) continue;
            if (sz == 1) {
                dfs(i, g[i][0]);
                k--;
            }
        }
        cout << '\n';
    }
    return 0;
}
