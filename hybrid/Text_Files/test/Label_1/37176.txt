n = int(input())
a = list(map(int, input().split()))
if n == 1:
    print("YES")
elif n == 2:
    print("NO")
else:
    inc = True
    dec = False
    for i in range(1, n):
        if a[i] == a[i - 1]:
            continue
        if a[i] > a[i - 1]:
            if dec:
                print("NO")
                exit()
        else:
            if not inc:
                print("NO")
                exit()
            inc = False
            dec = True
    print("YES")
