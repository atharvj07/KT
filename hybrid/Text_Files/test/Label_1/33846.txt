import java.util.*;
public class Main {
    static double PI = Math.acos(-1);
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        double[] r = new double[n];
        for (int i = 0; i < n; i++) {
            r[i] = sc.nextDouble();
        }
        Arrays.sort(r);
        double sum = 0;
        for (int i = 0; i < n; i++) {
            sum += r[i] * r[i] * PI;
            if (i + 1 < n) {
                sum -= r[i + 1] * r[i + 1] * PI;
            }
        }
        System.out.println(sum);
    }
}
