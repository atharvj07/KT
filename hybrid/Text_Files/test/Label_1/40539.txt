import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]
edges.sort(reverse=True, key=lambda x: x[0] != x[1])
tree = set()
deg = [0] * (n + 1)
for a, b in edges:
    if a == b: continue
    if (a, b) in tree or (b, a) in tree: continue
    tree.add((a, b))
    deg[a] += 1
    deg[b] += 1
mx = max(deg)
for a, b in edges:
    if deg[a] == mx or deg[b] == mx:
        print(*sorted([a, b]))
