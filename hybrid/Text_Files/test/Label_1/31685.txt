import java.util.Scanner;

public class Main {
	static long[] getBeautifulArray(long[] a) {
		int n = a.length;
		long[] b = new long[n];
		b[0] = 1;
		b[n - 1] = a[n - 1];
		for (int i = 1; i < n - 1; i++) {
			long gcd = GCD(a[i - 1], a[i]);
			long lcm = LCM(gcd, a[i + 1]);
			b[i] = (a[i] / gcd) * lcm;
		}
		return b;
	}

	static long GCD(long a, long b) {
		if (a == 0)
			return b;
		return GCD(b % a, a);
	}

	static long LCM(long a, long b) {
		return a * b / GCD(a, b);
	}

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while (t-- > 0) {
			int n = sc.nextInt();
			long[] a = new long[n];
			for (int i = 0; i < n; i++) {
				a[i] = sc.nextLong();
			}
			long[] b = getBeautifulArray(a);
			for (int i = 0; i < n; i++) {
				System.out.print(b[i] + " ");
			}
			System.out.println();
		}
		sc.close();
	}
}
