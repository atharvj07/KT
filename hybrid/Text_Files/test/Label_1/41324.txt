import sys

def solve():
    n,m=map(int,input().split())
    a=[list(map(int,input().split())) for i in range(n)]
    b=[0]*n+[0]*m

    for i in range(n):
        for j in range(m):
            if a[i][j]<b[i]+b[j]:
                return -1
            else:
                b[i]+=a[i][j]-b[i]
                b[j]+=a[i][j]-b[j]
    c=sum(b[n:n+m])
    b=b[0:n]
    k=0
    while max(b)-min(b)>c and k<1000:
        k+=1
        mi=min(b)
        ma=max(b)
        for i in range(n):
            if b[i]==ma:
                for j in range(m):
                    a[i][j]-=ma-mi
                    b[j]+=ma-mi
    if max(b)-min(b)>c:
        return -1
    print(k)
    for i in range(n):
        if max(a[i])>0:
            print("row "+str(i+1))
            for j in range(m):
                a[i][j]=0
    for i in range(m):
        if max(list(zip(*a))[i])>0:
            print("col "+str(i+1))
            for j in range(n):
                a[j][i]=0

solve()
