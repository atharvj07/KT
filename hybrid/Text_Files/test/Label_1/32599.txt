import java.util.Arrays;
import java.util.Scanner;

class Main {
    private static final int MOD = 1_000_000_007;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] points = new int[n][2];
        for (int i = 0; i < n; i++) {
            points[i][0] = sc.nextInt();
            points[i][1] = sc.nextInt();
        }
        ConvexHull hull = new ConvexHull(Arrays.copyOf(points, n));
        int[] f = new int[n+1];
        f[3] = hull.perimeter();
        for (int i = 4; i <= n; i++) {
            f[i] = (int)((f[i-1] + hull.triangulation()) % MOD);
        }
        for (int i = 3; i <= n; i++) {
            System.out.println(f[i]);
        }
    }
    
    private static class ConvexHull {
        private final int n;
        private final int[][] points;
        
        public ConvexHull(int[][] points) {
            this.n = points.length;
            this.points = points;
            Arrays.sort(this.points, (a, b) -> {
                if (a[0] != b[0]) {
                    return Integer.compare(a[0], b[0]);
                } else {
                    return Integer.compare(a[1], b[1]);
                }
            });
        }
        
        public int perimeter() {
            int perimeter = 0;
            for (int i = 0; i < n; i++) {
                int dx = Math.abs(points[i][0] - points[(i+1)%n][0]);
                int dy = Math.abs(points[i][1] - points[(i+1)%n][1]);
                perimeter += dx + dy;
            }
            return perimeter;
        }
        
        public int triangulation() {
            int triangulation = 0;
            for (int i = 0; i < n; i++) {
                for (int j = i+2; j < n; j++) {
                    int dx1 = points[j][0] - points[i][0];
                    int dy1 = points[j][1] - points[i][1];
                    int dx2 = points[(i+1)%n][0] - points[i][0];
                    int dy2 = points[(i+1)%n][1] - points[i][1];
                    int cross = dx1 * dy2 - dx2 * dy1;
                    if (cross > 0) {
                        int dx3 = points[(j+1)%n][0] - points[j][0];
                        int dy3 = points[(j+1)%n][1] - points[j][1];
                        int cross2 = dx2 * dy3 - dx3 * dy2;
                        if (cross2 > 0) {
                            triangulation += 2 * (Math.abs(dx1) + Math.abs(dy1));
                        }
                    }
                }
            }
            return triangulation;
        }
    }
}
