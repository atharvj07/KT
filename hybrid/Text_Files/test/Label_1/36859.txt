def calculate_paths(n, k):
    left, right = 1, n
    while left < right:
        mid = (left + right + 1) // 2
        total, power_of_two = 0, 1
        while mid // power_of_two > 0:
            total += n // (mid // power_of_two)
            power_of_two *= 2
        if total >= k:
            left = mid
        else:
            right = mid - 1
    return left

n, k = map(int, input().split())
print(calculate_paths(n, k))
