from sys import stdin

n = int(stdin.readline())
s = [int(x) for x in stdin.readline().split()]

cur_max = s[0]
second_max = s[1] if s[0] != s[1] else -1
for x in s:
    if x > cur_max:
        second_max = cur_max
        cur_max = x
    elif x != cur_max and (second_max == -1 or x > second_max):
        second_max = x

print(cur_max ^ second_max)
