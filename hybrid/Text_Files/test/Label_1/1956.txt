#include<bits/stdc++.h>
using namespace std;
#define fast  ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define endl "\n"
#define int long long
int32_t main()
{
    fast;
    int n;
    cin >> n;

    while (n--)
    {
        string s, t;
        cin >> s >> t;
        int freq[26] = {0};

        for (char c : s)
            freq[c - 'a']++;

        bool flag = true;
        for (char c : t)
            if (freq[c - 'a'] == 0)
            {
                flag = false;
                break;
            }

        cout << (flag ? "YES" : "NO") << endl;
    }
    return 0;
}
