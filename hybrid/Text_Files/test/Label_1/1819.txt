#include <iostream>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int c1, c2, c3, a1, a2, a3, a4, a5;
    cin >> c1 >> c2 >> c3 >> a1 >> a2 >> a3 >> a4 >> a5;
    if (c1 + c2 + c3 < a1 + a2 + a3 + a4 + a5) {
      cout << "NO" << endl;
    } else {
      if (c1 >= a1 + a4) {
        a4 = 0;
      } else {
        a4 -= c1 - a1;
      }
      if (c2 >= a2 + a5) {
        a5 = 0;
      } else {
        a5 -= c2 - a2;
      }
      if (c3 >= a3 + a4 + a5) {
        cout << "YES" << endl;
      } else {
        cout << "NO" << endl;
      }
    }
  }
  return 0;
}
