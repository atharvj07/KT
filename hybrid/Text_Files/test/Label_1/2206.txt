#include <bits/stdc++.h>
#define ll long long
#pragma GCC optimize ("O3")
#pragma GCC target ("sse4")

using namespace std;

const int maxn = 1e6 + 10;
vector<int> divs[maxn];

int solve(int n, int a[]) {
	for (int i = 1; i < maxn; i++) {
		for (int j = i; j < maxn; j += i) {
			divs[j].push_back(i);
		}
	}
	int ans = 1e9;
	for (int i = 0; i < n; i++) {
		if (a[i] == 0) {
			continue;
		}
		for (int j = 0; j < divs[a[i]].size(); j++) {
			int g = divs[a[i]][j];
			int sum = 0;
			for (int k = 0; k < n; k++) {
				sum += (a[k] + g - 1) / g;
			}
			ans = min(ans, sum - 1);
		}
	}
	if (ans == 1e9) {
		ans = -1;
	}
	return ans;
}

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int n;
	cin >> n;
	int a[n];
	for (int i = 0; i < n; i++) {
		cin >> a[i];
	}
	cout << solve(n, a) << endl;
	return 0;
}
