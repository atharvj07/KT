from collections import Counter
from functools import reduce

for _ in range(int(input())):
    n = int(input())
    L = [int(x) for x in input().split()]
    mn = min(L)
    mx = max(L)
    L[-1] += mn
    L[-1] %= 2**30
    ok = []
    c, l = Counter(L).most_common()[0]
    for i in range(1, n + 1):
        if l % i == mn and c >= i:
            ok.append(i)
    ok = sorted(ok)
    print(*ok)
