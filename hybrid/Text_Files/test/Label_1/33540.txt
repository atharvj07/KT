import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[] white = {in.nextInt(), in.nextInt(), in.nextInt(), in.nextInt()};
        int[] black1 = {in.nextInt(), in.nextInt(), in.nextInt(), in.nextInt()};
        int[] black2 = {in.nextInt(), in.nextInt(), in.nextInt(), in.nextInt()};

        // Check if any corners of the white sheet are outside both black sheets
        if (white[0] < black1[0] && white[0] < black2[0] ||  // Left
          white[1] < black1[1] && white[1] < black2[1] ||  // Bottom
          white[2] > black1[2] && white[2] > black2[2] ||  // Right
          white[3] > black1[3] && white[3] > black2[3]) {  // Top
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
