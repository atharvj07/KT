from itertools import combinations
MOD = 1000000007

def f(l, r):
    return max(0, r - l + 1)

n, k = map(int, input().split())
segments = [tuple(map(int, input().split())) for _ in range(n)]

answer = 0
for combo in combinations(segments, k):
    l_max = max(seg[0] for seg in combo)
    r_min = min(seg[1] for seg in combo)
    answer = (answer + f(l_max, r_min)) % MOD

print(answer)
