#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, ans, c = 0;
  cin >> n;
  map<int, int> mp;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    mp[x]++;
    if (mp[x] > 1) c++;
  }
  if (c > 0)
    cout << n << endl;
  else {
    for (int i = 1; i <= n; i++) {
      if (mp[i] == 0) {
        cout << i << endl;
        break;
      }
    }
  }
  return 0;
}
