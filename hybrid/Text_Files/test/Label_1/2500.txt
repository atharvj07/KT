#include <iostream>
#include <vector>
#include <map>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::string> words(n);
  std::map<std::string, int> cnt;
  for (int i = 0; i < n; ++i) {
    std::cin >> words[i];
    ++cnt[words[i]];
  }

  std::string s;
  std::cin >> s;

  int cnt_l = 0, cnt_r = 0;
  for (int i = 0; i < s.size(); ++i) {
    if (s[i] == '<') ++cnt_l;
    else if (s[i] == '3') ++cnt_r;
  }

  if (cnt_l != cnt_r) {
    std::cout << "no" << std::endl;
    return 0;
  }

  int j = 0;
  for (int i = 0; i < s.size() - 4; ++i) {
    if (s[i] == '<' && s[i + 1] == '3' && s[i + 2] == words[j][0] &&
        s[i + 3] == words[j][1] && s[i + 4] == '3') {
      if (--cnt[words[j]] < 0) {
        std::cout << "no" << std::endl;
        return 0;
      }
      i += 3;
      ++j;
    }
  }

  std::cout << "yes" << std::endl;
  return 0;
}
