#include <bits/stdc++.h>
#define ll long long

using namespace std;

const int MOD = 1e9 + 7;
int n;
string s;
ll dp[(int)2e5 + 1];

ll solve(int i) {
  if (i == n) { return 1; }
  if (dp[i] > -1) { return dp[i]; }
  ll ret = 0;
  if (s[i] == '(') {
    ret = solve(i + 1);
    ret %= MOD;
    ret += (s[i + 1] == ')');
    ret %= MOD;
  } else {
    ret = (solve(i + 1) + MOD - solve(i + 2)) % MOD;
  }
  return dp[i] = ret;
}

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> s;
  n = s.length();
  fill(dp, dp + n + 1, -1);
  cout << solve(0);
  return 0;
}
