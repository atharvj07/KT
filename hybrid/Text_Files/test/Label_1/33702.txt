import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int R = sc.nextInt();
        int G = sc.nextInt();
        int B = sc.nextInt();
        int N = sc.nextInt();
        long ans = 0;
        for (int r = 0; r <= N / R; r++) {
            for (int g = 0; g <= (N - r * R) / G; g++) {
                int b = (N - r * R - g * G) / B;
                if (r * R + g * G + b * B == N) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
