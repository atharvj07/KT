#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  int neg = 0, pos = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] < 0) neg++;
    else if (a[i] > 0) pos++;
  }
  if (neg % 2 == 1) {
    for (int i = 0; i < n; i++) {
      if (a[i] < 0) a[i] = -a[i] - 1;
    }
  }
  for (int i = 0; i < n; i++) {
    cout << a[i] << " ";
  }
  cout << endl;
  return 0;
}
