import java.io.*;
import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n + 1];
        HashMap<Integer, Integer> h = new HashMap<>();
        int max_or = 0;
        for (int i = 0; i < m; i++) {
            int tp = sc.nextInt();
            if (tp == 1) {
                int l = sc.nextInt(), r = sc.nextInt(), x = sc.nextInt();
                h.put(r, Math.max(h.getOrDefault(r, 0), x));
            } else {
                int k = sc.nextInt(), d = sc.nextInt();
                a[k] = d;
                max_or |= d;
            }
        }
        for (int r = n; r > 0; r--) {
            a[r] = Math.max(a[r], h.getOrDefault(r, 0));
            max_or |= a[r];
        }
        if (max_or > 1e9) System.out.println("NO");
        else {
            System.out.println("YES");
            for (int i = 1; i <= n; i++) System.out.printf("%d ", a[i]);
            System.out.println();
        }
    }
}
