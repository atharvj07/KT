from collections import Counter

s = input()
n = len(s)
dp = [['' for _ in range(n)] for _ in range(n)]
for i in range(n):
    dp[i][i] = s[i]
for l in range(1, n):
    for i in range(n - l):
        j = i + l
        if s[i] == s[j]:
            dp[i][j] = s[i] + dp[i + 1][j - 1] + s[j]
        else:
            dp[i][j] = max(dp[i][j - 1], dp[i + 1][j])
secret = dp[0][n - 1]
cnt = Counter(secret)
print(max(cnt.values()))
