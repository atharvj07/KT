#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

void solve() {
  ll n, k;
  cin >> n >> k;
  ll ans = 1, cur = n;
  for (ll p = 2; p * p <= n; ++p) {
    ll cnt = 0;
    while (cur % p == 0)
      cur /= p, cnt++;
    ans = ans * (ll)(k + cnt + 1) % (1000000007);
    ans = ans * (ll)pow(p, k % (cnt + 1)) % (1000000007);
  }
  if (cur > 1)
    ans = ans * (ll)(k + 1) % (1000000007);
  cout << ans << "\n";
}

int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  solve();
  return 0;
}
