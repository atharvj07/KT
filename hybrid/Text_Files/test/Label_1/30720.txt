import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = s.nextInt();

        int d = 1;
        for (int i = 1; i < a.length; i++) {
            if (a[i] > a[i - 1]) d++;
        }

        System.out.println(d);
        for (int i = 0; i < d; i++) System.out.print(a[i] + " ");
    }
}
