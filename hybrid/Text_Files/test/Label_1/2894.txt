#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int l = 0, r = 0, q = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] != '?') {
      if (i < n / 2)
        l += s[i] - '0';
      else
        r += s[i] - '0';
    } else
      q++;
  }
  if (q == 0 || (q % 2 == 0 && l == r))
    cout << "Bicarp" << endl;
  else
    cout << "Monocarp" << endl;
  return 0;
}
