import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < k; i++) {
            int[] s = new int[n + 1];
            for (int j = 1; j <= n; j++) {
                s[j] = (s[j - 1] + a[j]) % 1000000007;
            }
            a = s;
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
