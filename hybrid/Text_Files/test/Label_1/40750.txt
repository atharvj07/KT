h = int(input())
a = list(map(int, input().split()))

# Check if the given sequence can form more than one tree
perfect = True
for i in range(1, h):
    if a[i] > 1 and a[i + 1] > 1:
        perfect = False
        break

if perfect:
    print("perfect")
else:
    print("ambiguous")
    parents = [0]  # Root
    current_parent = 0
    for level in range(1, h + 1):
        for _ in range(a[level]):
            parents.append(current_parent + 1)
        current_parent += a[level - 1]
    print(*parents)

    # Creating the second tree with a slight difference to ensure they are not isomorphic
    parents = [0]  # Root
    current_parent = 0
    for level in range(1, h + 1):
        next_parent = current_parent
        if level < h and a[level] > 1 and a[level + 1] > 1:
            next_parent += 1
        for _ in range(a[level]):
            parents.append(current_parent + 1)
            current_parent = next_parent
    print(*parents)
