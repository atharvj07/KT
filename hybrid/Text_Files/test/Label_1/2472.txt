#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
  int n, L;
  cin >> n >> L;
  vector<int> p1(n), p2(n);
  for (int i = 0; i < n; i++) cin >> p1[i];
  for (int i = 0; i < n; i++) cin >> p2[i];
  int shifting = p1[0] - p2[0];
  for (int i = 0; i < n; i++) {
    if (p1[i] - shifting != p2[i]) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
  return 0;
}
