import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] deg = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int a = in.nextInt(), b = in.nextInt();
            deg[a]++; deg[b]++;
        }
        boolean ok = true;
        for (int i = 1; i <= n; i++) {
            ok &= deg[i] >= 1 && deg[i] <= 2;
        }
        System.out.println(ok ? "Yes" : "No");
    }
}
