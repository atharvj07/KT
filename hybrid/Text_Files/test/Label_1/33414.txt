import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

public class Main {

  private static final long MOD = 1000000007;
  private static long[][] fib = new long[100001][2];

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nm = br.readLine().split(" ");
    int n = Integer.parseInt(nm[0]), m = Integer.parseInt(nm[1]);
    String[] a = br.readLine().split(" ");
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = Integer.parseInt(a[i]);
    }
    fib[1][0] = fib[1][1] = fib[2][0] = 1;
    for (int i = 3; i < 100001; i++) {
      fib[i][0] = (fib[i - 1][0] + fib[i - 2][0]) % MOD;
      fib[i][1] = (fib[i - 1][1] + fib[i - 2][1]) % MOD;
    }
    long[] pre = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      pre[i] = pre[i - 1] + arr[i - 1];
    }
    for (int i = 0; i < m; i++) {
      String[] query = br.readLine().split(" ");
      int type = Integer.parseInt(query[0]), l = Integer.parseInt(query[1]), r = Integer.parseInt(query[2]);
      if (type == 1) {
        int x = Integer.parseInt(query[3]);
        pre[r] += (long) x;
        pre[l - 1] -= (long) x;
      } else {
        long sum = (pre[r] - pre[l - 1] + MOD) % MOD;
        long ans = (fib[(int) sum][1] + MOD) % MOD;
        System.out.println(ans);
      }
    }
  }
}
