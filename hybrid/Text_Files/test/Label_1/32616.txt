import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[][] operations = new int[m][3];
        for (int i = 0; i < m; i++) {
            operations[i][0] = scanner.nextInt() - 1;
            operations[i][1] = scanner.nextInt() - 1;
            operations[i][2] = scanner.nextInt();
        }

        int[][] queries = new int[k][2];
        for (int i = 0; i < k; i++) {
            queries[i][0] = scanner.nextInt() - 1;
            queries[i][1] = scanner.nextInt() - 1;
        }

        int[] frequency = new int[m];
        for (int i = 0; i < k; i++) {
            for (int j = queries[i][0]; j <= queries[i][1]; j++) {
                frequency[j]++;
            }
        }

        for (int i = 0; i < m; i++) {
            a[operations[i][0]] += operations[i][2] * frequency[i];
            if (operations[i][1] + 1 < n) {
                a[operations[i][1] + 1] -= operations[i][2] * frequency[i];
            }
        }

        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            a[i] = (int) sum;
        }

        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
