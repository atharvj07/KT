#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }
  std::sort(a.begin(), a.end());
  int ans = 0;
  for (int i = 0; i < n; i += k) {
    if (i + k - 1 >= n || a[i] != a[i + k - 1]) {
      ++ans;
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
