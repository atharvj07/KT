#include <bits/stdc++.h>
using namespace std;

int main() {
  int n; cin >> n;
  long long ans = 0;
  bool second = false;
  while (n--) {
    int c, d; cin >> c >> d;
    if (d == 1) {
      if (!second) ans += c;
      else second = false;
    }
    else {
      if (ans + c >= 1900) { ans = 1900; second = true; }
      else ans += c;
    }
  }
  if (ans == 1900) ans = -1;
  if (ans == -1) cout << "Infinity\n";
  else if (ans == 0) cout << "Impossible\n";
  else cout << ans << "\n";
}
