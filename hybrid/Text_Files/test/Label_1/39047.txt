import heapq

def min_penalty(n, balls):
    if n == 1:
        return 0
    heapq.heapify(balls)
    penalty = 0
    while len(balls) > 1:
        if len(balls) % 2 == 0:
            balls.append(heapq.heappop(balls) + heapq.heappop(balls))
        else:
            balls.append(heapq.heappop(balls) + heapq.heappop(balls) + heapq.heappop(balls))
        penalty += balls[-1]
        heapq.heapify(balls)
    return penalty

n = int(input())
balls = list(map(int, input().split()))
print(min_penalty(n, balls))
