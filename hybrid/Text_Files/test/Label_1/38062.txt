from functools import reduce
from math import gcd

n = int(input())
edges = [[] for i in range(n+1)]
for i in range(n-1):
    a, b = map(int, input().split())
    edges[a]+=[b]
    edges[b]+=[a]

dp = [[0] * 3 for i in range(n+1)]
dp[1] = [1,0,0]

def dfs(x, p, g):
    for y in edges[x]:
        if y!=p:
            dfs(y,x,gcd(g,x))
            dp[x][0] += dp[y][0]
            dp[x][1] += dp[y][1]
            dp[x][2] += dp[y][2]
            dp[x][2] += dp[y][0] * dp[x][1]
            dp[x][1] += dp[y][0] * dp[x][0]
    dp[x][2] += 1
    dp[x][1] -= 1

dfs(1,1,1)
for x in range(1, n+1):
    g = gcd(dp[x][0], dp[x][2])
    print(dp[x][0]//g / dp[x][2]//g)
