#include <iostream>
#include <vector>
using namespace std;
#define INF 0x3f3f3f3f
const vector<int> nums = {2, 3, 5, 7};
int main() {
    int n;
    cin >> n;
    if (n <= 3) {
        cout << -1 << endl;
        return 0;
    }
    int res = INF;
    int tmp = n;
    vector<int> cnt(4, 0);
    for (int num : nums) {
        while (tmp % num == 0) {
            cnt[num - 2]++;
            tmp /= num;
        }
        if (tmp == 1) {
            int ans = 1;
            for (int i = 0; i < 4; i++) {
                ans *= nums[i];
                ans /= cnt[i] + 1;
            }
            res = min(res, ans);
        }
        tmp = n;
    }
    cout << (res == INF ? -1 : res) << endl;
}
