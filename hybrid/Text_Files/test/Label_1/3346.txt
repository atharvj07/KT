#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  int n, x;
  cin >> n;
  vector<int> v(n);
  for (int i = 0; i < n; i++) {
    string s;
    cin >> s;
    int base = 1;
    for (int j = s.size() - 1; j >= 0; j--) {
      if (s[j] != '?') v[i] += (s[j] - '0') * base;
      base *= 10;
    }
  }
  sort(v.begin(), v.end());
  bool flag = false;
  for (int i = 0; i < n; i++) {
    if (v[i] == v[i + 1]) {
      flag = true;
      break;
    }
  }
  if (flag)
    cout << "NO";
  else {
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) cout << v[i] << endl;
  }
  return 0;
}
