#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
const int maxn = 100005;
int x[maxn];
ll pre[maxn];
int main() {
  int n, q;
  cin >> n >> q;
  string s;
  cin >> s;
  for (int i = 1; i <= n; i++) { x[i] = s[i - 1] - '0'; }
  for (int i = 1; i <= n; i++) { pre[i] = pre[i - 1] + x[i]; }
  while (q--) {
    int l, r;
    cin >> l >> r;
    cout << (pre[r] - pre[l - 1] + (ll)x[l] * (pre[r] - pre[l - 1])) % mod
         << '\n';
  }
  return 0;
}
