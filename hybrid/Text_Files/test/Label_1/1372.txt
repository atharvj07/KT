#include <bits/stdc++.h>

int main() {
    int n;
    std::cin >> n;
    std::vector<bool> v(n), h(n);
    for (int i = 0; i < n * n; ++i) {
        int x, y;
        std::cin >> x >> y;
        --x, --y;
        if (!v[y] || !h[x]) {
            v[y] = h[x] = 1;
            std::cout << i + 1 << ' ';
        }
    }
    std::cout << std::endl;
    return 0;
}
