import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static int[] a;
    static int mex(int[] b) {
        boolean[] c = new boolean[300];
        for (int x : b) c[x] = true;
        int ret = 0;
        while (c[ret]) ret++;
        return ret;
    }
    static int nimber(int[] b) {
        int[] c = new int[300];
        for (int x : b) c[x] ^= b.length - x;
        int ret = c[0];
        for (int x = 1; x < 300; x++) ret ^= c[x];
        return ret;
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int nim = nimber(a);
        if (nim == 0) System.out.println("BitAryo");
        else System.out.println("BitLGM");
    }
}
