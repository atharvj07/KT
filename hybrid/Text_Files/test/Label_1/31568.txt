import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
  static void swap(int mat[][], int i, int j, int p, int q){
      int temp = mat[i][j];
      mat[i][j] = mat[p][q];
      mat[p][q] = temp;
  }
  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int[] nm = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
    int n = nm[0], m = nm[1], cnt=0;
    int[][] mat = new int[n][m];
    ArrayList<Integer>[] ar = new ArrayList[n+m+1];
    for (int i = 1; i <= n + m; i++) ar[i] = new ArrayList<>();
    for (int i = 0; i < n; i++) {
      int[] temp = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
      for (int j = 0; j < m; j++) {
        mat[i][j] = temp[j];
        ar[temp[j]].add(i * m + j);
      }
    }
    int[][] matcopy = new int[n][m], mattemp = new int[n][m];

    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++)
        matcopy[i][j] = mat[i][j];

    for (int i = 0; i < n; i++) {
      ArrayList<Integer> a = ar[mat[i][0]];
      int id = -1;
      for (int j = 1; j < m; j++) {
        if (ar[mat[i][j]].get(0) > ar[mat[i][j - 1]].get(0))
          id = j;
        else break;
      }
      if (id != -1) {
        for (int j = id; j < m; j++) {
          swap(mat, i, j, i, m - j - 1 + id);
        }
      }
    }

    for (int j = 0; j < m; j++) {
      ArrayList<Integer> a = ar[mat[0][j]];
      int id = -1;
      for (int i = 1; i < n; i++) {
        if (ar[mat[i][j]].get(0) > ar[mat[i - 1][j]].get(0))
          id = i;
        else break;
      }
      if (id != -1) {
        for (int i = id; i < n; i++) {
          swap(mat, i, j, n - i - 1 + id, j);
        }
      }
    }
    boolean bit = true;
    for (int i = 0; i < n; i++) {
      for (int j = 1; j < m; j++) {
        if (mat[i][j] < mat[i][j - 1]) {
          bit = false;
          break;
        }
      }
    }
    for (int j = 0; j < m; j++) {
      for (int i = 1; i < n; i++) {
        if (mat[i][j] < mat[i - 1][j]) {
          bit = false;
          break;
        }
      }
    }
    if (!bit) {
      System.out.println(-1);
      return;
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        System.out.print(mat[i][j] + " ");
      }
      System.out.println();
    }
  }
}
