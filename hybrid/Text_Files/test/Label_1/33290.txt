import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int r = scanner.nextInt();
        int n = scanner.nextInt();
        boolean[][] grid = new boolean[r + 1][3];
        for (int i = 0; i < n; i++) {
            int ri = scanner.nextInt();
            int ci = scanner.nextInt();
            grid[ri][ci] = true;
            grid[ri][3 - ci] = true;
            if (ri > 1) grid[ri - 1][3 - ci] = true;
            if (ri < r) grid[ri + 1][3 - ci] = true;
        }
        boolean turn = true;
        for (int i = 1; i <= r; i++) {
            for (int j = 1; j <= 2; j++) {
                if (!grid[i][j]) {
                    grid[i][j] = true;
                    grid[i][3 - j] = true;
                    if (i > 1) grid[i - 1][3 - j] = true;
                    if (i < r) grid[i + 1][3 - j] = true;
                    turn = !turn;
                    break;
                }
            }
        }
        System.out.println(turn ? "WIN" : "LOSE");
    }
}
