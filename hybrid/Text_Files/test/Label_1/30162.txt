import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt();
        }

        boolean possible = true;
        List<Integer>[] graph = new List[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new ArrayList<>();
        }
        for (int i = 0; i < n; i++) {
            graph[p[i] - 1].add(i);
        }

        int[] assigned = new int[n];
        for (int i = 0; i < n; i++) {
            if (assigned[i] == 0) {
                possible &= dfs(graph, i, assigned);
            }
        }

        System.out.println(possible ? "POSSIBLE" : "IMPOSSIBLE");
    }

    private static boolean dfs(List<Integer>[] graph, int v, int[] assigned) {
        for (int u : graph[v]) {
            if (assigned[u] == 0) {
                for (int i = 0; i < assigned[v]; i++) {
                    if (!graph[u].contains(i)) {
                        assigned[u] = i;
                        if (!dfs(graph, u, assigned)) {
                            return false;
                        }
                        break;
                    }
                }
                if (assigned[u] == 0) {
                    return false;
                }
            } else {
                if (assigned[u] >= assigned[v]) {
                    return false;
                }
            }
        }
        return true;
    }
}
