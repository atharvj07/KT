#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  int cnt[n + 1];
  for (int i = 0; i <= n; i++) cnt[i] = 0;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    cnt[a[i]] = 1;
  }
  int b[n];
  int j = 0;
  for (int i = 0; i < n; i++) {
    if (j > a[i]) {
      cout << -1;
      return 0;
    }
    while (cnt[j]) j++;
    b[i] = j;
    cnt[j] = 1;
  }
  for (int i = 0; i < n; i++) cout << b[i] << " ";
  return 0;
}
