import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        String s = sc.next();
        char[] arr = s.toCharArray();
        for (int i = 0; i < n && k > 0; i++) {
            if (arr[i] > '0') {
                arr[i] = '0';
                k--;
            }
        }
        if (k > 0) {
            while (k > 0 && n > 0) {
                arr[n - 1] = '0';
                n--;
                k--;
            }
        }
        System.out.println(new String(arr));
    }
}
