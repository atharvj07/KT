#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    int q;
    cin >> q;
    for (int i = 0; i < q; i++) {
        ll n;
        cin >> n;
        n--;
        ll ans = 1;
        while (1) {
            if (n == 0) break;
            if (n % 3 == 2) ans *= 3;
            n /= 3;
        }
        cout << ans << "\n";
    }
}
