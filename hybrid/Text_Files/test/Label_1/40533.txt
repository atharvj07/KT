def solve(n, vectors):
    px, py = 0, 0
    for x, y in vectors:
        px += x
        py += y
    signs = []
    for x, y in vectors:
        if (px * x + py * y) > 0:
            signs.append(1)
        else:
            signs.append(-1)
            px -= 2*x  # Adjusting the total path if we choose the negative direction
            py -= 2*y
    return signs

n = int(input())
vectors = [tuple(map(int, input().split())) for _ in range(n)]
signs = solve(n, vectors)
print(' '.join(map(str, signs)))
