import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
            if (sum >= 2e9) {
                System.out.println(i);
                return;
            }
            sum *= 2;
        }
        System.out.println(n);
    }
}
