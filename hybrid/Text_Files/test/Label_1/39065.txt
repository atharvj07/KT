def solve():
    t = int(input())
    for _ in range(t):
        n, l, r, s = map(int, input().split())
        if (r - l + 1) * (r - l) // 2 < s or s > n * (n + 1) // 2:
            print(-1)
            continue
        p = list(range(l - 1)) + [i for i in range(l, r + 1, 2)] + [i for i in range(r, l - 1, -2)]
        for i in range(max(0, (l + r) // 2 - len(p)), n - len(p)):
            p.append(i + 1)
        if sum(p[l - 1: r]) == s:
            print(*p)
        else:
            print(-1)

solve()
