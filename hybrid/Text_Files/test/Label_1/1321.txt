#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int &ai : a) { cin >> ai; }

  sort(a.begin(), a.end());

  int min_cost = 1e9;
  int ans_t = 0;
  for (int t = 1; t <= 100; t++) {
    int cost = 0;
    for (int ai : a) { cost += min(abs(ai - t), abs(ai - t - 1)); }
    if (cost < min_cost) {
      min_cost = cost;
      ans_t = t;
    }
  }

  cout << ans_t << " " << min_cost << endl;

  return 0;
}
