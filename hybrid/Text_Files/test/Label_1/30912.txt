import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        boolean ok = true;
        for (int i = 0; i < 26; i++) {
            if (cnt[i] >= 3) {
                ok = false;
            }
            if (cnt[i] == 2) {
                if (!isP(cnt[i])) {
                    ok = false;
                }
            }
        }
        if (!ok) {
            System.out.println("NO");
            return;
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 2; i <= s.length(); i++) {
            if (!isP(i)) {
                continue;
            }
            for (int j = 0; j < 26; j++) {
                if (cnt[j] == 0) {
                    continue;
                }
                sb.append((char) (j + 'a'));
                cnt[j]--;
                i--;
            }
        }
        for (int i = 0; i < 26; i++) {
            if (cnt[i] == 1) {
                sb.append((char) (i + 'a'));
            }
        }
        System.out.println("YES");
        System.out.println(sb);
    }

    public static boolean isP(int x) {
        if (x == 1) {
            return false;
        }
        for (int i = 2; i * i <= x; i++) {
            if (x % i == 0) {
                return false;
            }
        }
        return true;
    }
}
