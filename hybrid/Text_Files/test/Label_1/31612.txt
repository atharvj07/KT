import java.util.*;
import java.io.*;

class Main {
    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static char[][] map;
    static int[][] dist;
    static Queue<Point> q;
    static Set<Character> types;
    static int n, m, k;
    static final int[] dx = { -1, 0, 1, 0 };
    static final int[] dy = { 0, 1, 0, -1 };

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        k = Integer.parseInt(st.nextToken());
        map = new char[n][m];
        dist = new int[n][m];
        q = new LinkedList<Point>();
        types = new HashSet<Character>();
        for (int i = 0; i < n; i++) {
            String line = br.readLine();
            for (int j = 0; j < m; j++) {
                map[i][j] = line.charAt(j);
                if (map[i][j] != 'S' && map[i][j] != 'T') {
                    types.add(map[i][j]);
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                dist[i][j] = Integer.MAX_VALUE;
            }
        }
        q.offer(new Point(0, 0));
        dist[0][0] = 0;
        while (!q.isEmpty()) {
            Point p = q.poll();
            for (int i = 0; i < 4; i++) {
                int nx = p.x + dx[i];
                int ny = p.y + dy[i];
                if (nx < 0 || nx >= n || ny < 0 || ny >= m || map[nx][ny] == 'T') {
                    continue;
                }
                int nd = dist[p.x][p.y] + 1;
                if (map[nx][ny] != 'S' && map[nx][ny] != 'T') {
                    if (types.size() == k && types.contains(map[nx][ny])) {
                        continue;
                    }
                    types.add(map[nx][ny]);
                }
                if (nd < dist[nx][ny]) {
                    dist[nx][ny] = nd;
                    q.offer(new Point(nx, ny));
                }
            }
        }
        String path = "";
        if (dist[n - 1][m - 1] == Integer.MAX_VALUE) {
            System.out.println("-1");
        } else {
            int x = n - 1, y = m - 1;
            while (x > 0 || y > 0) {
                for (int i = 0; i < 4; i++) {
                    int nx = x + dx[i];
                    int ny = y + dy[i];
                    if (nx < 0 || nx >= n || ny < 0 || ny >= m) {
                        continue;
                    }
                    if (dist[nx][ny] + 1 == dist[x][y]) {
                        if (map[nx][ny] != 'S' && map[nx][ny] != 'T') {
                            path = map[nx][ny] + path;
                        }
                        x = nx;
                        y = ny;
                        break;
                    }
                }
            }
            System.out.println(path);
        }
    }
}
