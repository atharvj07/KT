import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        long ans = 1;
        int n = new Scanner(System.in).nextInt();
        int[] f = new int[n + 2];
        f[0] = 1;
        for (int i = 1; i <= n; i++) {
            f[i] = f[i - 1] * i % 1000000007;
        }
        int e = f[n] * f[n] % 1000000007;
        for (int a = 1; a < n; a++) {
            long b = f[a] * f[a] % 1000000007 * f[n - a] % 1000000007 * 2 % 1000000007;
            ans += b;
            ans %= 1000000007;
        }
        ans = ans * e % 1000000007;
        System.out.println(ans);
    }
}
