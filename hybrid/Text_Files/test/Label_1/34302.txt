import java.util.*;

class Main {
    public static void main(String[] args) {
        long MOD = 1000000007;
        Scanner scan = new Scanner(System.in);
        int N = scan.nextInt();
        int M = scan.nextInt();
        int[] A = new int[N];
        int[] B = new int[M];
        for (int i = 0; i < N; i++) {
            A[i] = scan.nextInt();
        }
        for (int i = 0; i < M; i++) {
            B[i] = scan.nextInt();
        }
        long ans = 1;
        for (int i = 0; i < N; i++) {
            int ind = 0;
            while (ind < M && B[ind] < A[i]) {
                ind++;
            }
            if (ind < M && B[ind] == A[i]) {
                ans = (ans * 2) % MOD;
            } else {
                ans = (ans * 3) % MOD;
            }
        }
        System.out.println(ans);
    }
}
