#include <bits/stdc++.h>
using namespace std;
const int mxN = 2e5;
int p[mxN], c[mxN];
int bs(int a, int l, int r){
    int m, ret = -1;
    while(l <= r){
        m = (l + r) >> 1;
        if(p[m] == a) ret = m, r = m - 1;
        else if(p[m] < a) l = m + 1;
        else r = m - 1;
    }
    return ret;
}
int32_t main(){
    int n, m, l, r, ans = 0;
    cin >> n >> m;
    m--;
    for(int i = 0; i < n; i++){ cin >> p[i]; c[i] = p[i]; }
    sort(p, p + n);
    for(int i = 0; i < n; i++){ l = bs(c[i], i, n - 1); if(l == -1) continue; if(p[l] == m) ans += max(0, l - i - 1) * max(0, n - l - 1); }
    cout << ans << endl;
    return 0;
}
