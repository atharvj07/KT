#include<iostream>
#include<vector>
#include<queue>
using namespace std;

int main(){
    int n;
    cin>>n;
    string s;
    cin>>s;
    int p=0;
    int len=0;
    vector<int> v;
    for(int i=0;i<n;i++){
        if(s[i]=='.'){
            len++;
        }
        else if(s[i]=='*'){
            len++;
        }
        else if(s[i]=='P'){
            v.push_back(p);
        }
        p++;
    }
    int ans=0;
    for(int i=0;i<v.size();i++){
        int x=v[i];
        int y=p-1-v[i];
        ans+=min(x,y);
    }
    ans+=len-1;
    cout<<ans<<endl;
    return 0;
}
