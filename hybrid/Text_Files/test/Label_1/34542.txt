import java.util.Scanner;

public class TestGenerator {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s1 = scanner.nextLine();
        String s2 = scanner.nextLine();
        String s3 = scanner.nextLine();

        // Find the minimum length of the string containing s1, s2, and s3 as substrings.
        int minLength = s1.length() + s2.length() + s3.length();

        // Check if any two substrings overlap.
        if (s1.contains(s2) || s1.contains(s3) || s2.contains(s3)) {
            minLength--;
        }

        // Check if all three substrings overlap.
        if (s1.contains(s2) && s1.contains(s3) && s2.contains(s3)) {
            minLength--;
        }

        // Print the minimum length of the string.
        System.out.println(minLength);
    }
}
