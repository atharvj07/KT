#include <bits/stdc++.h>
using namespace std;

const int MAXN = 18;
const int MAX_QUERIES = 1e5+1;

int n, m;
unsigned int a[MAXN], DP[MAXN][MAX_QUERIES];

int main() {
    cin >> n >> m;
    for (int i = 0; i < (1<<n); i++) cin >> a[i];
    for (int i = 1; i <= m; i++) {
        unsigned int x, y;
        cin >> x >> y;
        a[x-1] = y;
        unsigned int value = 0;
        for (int j = 0; j < (1<<n); j+=2) value ^= a[j] | a[j+1];
        DP[n][i] = value;
        for (int j = n-1; j >= 1; j--)
            for (int k = 0; k < (1<<(j)); k++)
                DP[j][i] ^= DP[j+1][i] ^ (a[k]|a[k+(1<<(j))]);
        cout << DP[1][i] << endl;
    }
}
