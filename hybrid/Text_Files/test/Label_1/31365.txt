import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BigInteger[] lr = new BigInteger[2];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < 2; i++)
            lr[i] = new BigInteger(s[i]);
        BigInteger n = lr[1].subtract(lr[0]).add(BigInteger.ONE);
        System.out.println(n.subtract(n.divide(BigInteger.TEN).add(n.mod(BigInteger.TEN).equals(BigInteger.ZERO) ? BigInteger.ONE : BigInteger.ZERO)));
    }
}
