#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    long long max_element = 0;

    cin >> n;

    vector<long long> a(n, 0);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        max_element = max(max_element, a[i]);
    }

    long long mex = 0;

    while (mex <= max_element) mex++;

    cout << mex << endl;
}
