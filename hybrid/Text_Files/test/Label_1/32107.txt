import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            if (n == 0 && k == 0) {
                break;
            }
            StringBuilder sb = new StringBuilder();
            for (int i = n; i < n + k; i++) {
                sb.append(i);
            }
            System.out.println(sb);
        }
        sc.close();
    }
}
