t = int(input().strip())
for _ in range(t):
    input()
    m, n = map(int, input().strip().split())
    joy_matrix = [list(map(int, input().strip().split())) for _ in range(m)]
    
    # Transpose the joy matrix to make it easier to work with
    joy_matrix_t = list(zip(*joy_matrix))
    
    # For each friend, find the maximum joy he can get from all shops
    max_joy_per_friend = [max(joy) for joy in joy_matrix_t]
    
    # Since Vlad can visit at most n-1 shops, 
    # we need to find the n-1 highest joys among friends and ensure the lowest of these is maximized
    # This is because, even if one friend gets a lower joy than the others, it should be as high as possible
    # Sort the max joys in ascending to easily find the smallest joy after excluding the lowest one if n > m
    max_joy_per_friend.sort()
    
    # If there are more friends than shops, one shop's item must be taken for granted
    # hence, we consider all shops. Otherwise, consider n-1 max joys (which includes all due to sorting above)
    if n > m:
        print(max_joy_per_friend[0])
    else:
        print(max_joy_per_friend[-n])
