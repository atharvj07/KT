import sys
input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split())) if k else []
b = {i - 1 for i in b}
need_change = []
for i in range(1, n):
    if a[i] < a[i - 1]: need_change.append(i - 1)
if len(need_change) == 0: print(0)
elif any(i in b for i in need_change): print(-1)
else: print(len(need_change))
