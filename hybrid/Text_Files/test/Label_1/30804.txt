import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    private static final long M = (long)Math.pow(10, 9) + 7;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        long[][] m = new long[n][n];
        for (int i = 0; i < n; i++) {
            String[] line = sc.nextLine().split(" ");
            for (int j = 0; j < n; j++)
                m[i][j] = Long.parseLong(line[j]);
        }
        long d = det(m, n);
        long inv = inv(d);
        System.out.println(d * inv % M);
    }

    private static long det(long[][] m, int n) {
        if (n == 1) return m[0][0];
        if (n == 2) return m[0][0] * m[1][1] - m[0][1] * m[1][0];
        long res = 0;
        for (int i = 0; i < n; i++) {
            long d = det(subMatrix(m, i, 0, n), n - 1);
            res += ((i % 2 == 0) ? 1 : -1) * m[i][0] * d;
        }
        return res;
    }

    private static long[][] subMatrix(long[][] m, int exRow, int exCol, int n) {
        long[][] sm = new long[n - 1][n - 1];
        for (int i = 0, ii = 0; i < n; i++) {
            if (i == exRow) continue;
            for (int j = 0, jj = 0; j < n; j++) {
                if (j == exCol) continue;
                sm[ii][jj++] = m[i][j];
            }
            ii++;
        }
        return sm;
    }

    private static long inv(long a) {
        BigInteger b = BigInteger.valueOf(a);
        return b.modPow(BigInteger.valueOf(M - 2), BigInteger.valueOf(M)).longValue();
    }
}
