#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
        }
    }
    vector<int> b;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            b.push_back(a[i][j]);
        }
    }
    sort(b.begin(), b.end());
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] != b[ans]) ans++;
        }
    }
    if (ans > n + m) {
        cout << "NO" << endl;
        return 0;
    }
    cout << "YES" << endl;
    string r(n, '0'), c(m, '0');
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < m; j++) {
            if (a[i][j] != b[ans]) cnt++;
        }
        if (cnt > m / 2) {
            r[i] = '1';
            for (int j = 0; j < m; j++) {
                a[i][j] = 1 - a[i][j];
            }
        }
        ans += m;
    }
    for (int j = 0; j < m; j++) {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (a[i][j] != b[ans]) cnt++;
        }
        if (cnt > n / 2) {
            c[j] = '1';
            for (int i = 0; i < n; i++) {
                a[i][j] = 1 - a[i][j];
            }
        }
        ans++;
    }
    cout << r << endl;
    cout << c << endl;
    return 0;
}
