import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int H = sc.nextInt();
        int W = sc.nextInt();
        int d = sc.nextInt();

        char[][] c = new char[H][W];
        boolean[][] used = new boolean[H][W];

        int[] dx = { -1, 0, 1, 0 };
        int[] dy = { 0, -1, 0, 1 };

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                c[i][j] = 'R';
                used[i][j] = false;
            }
        }

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (!used[i][j]) {
                    for (int k = 0; k < 4; k++) {
                        int nx = i + dx[k];
                        int ny = j + dy[k];
                        if (0 <= nx && nx < H && 0 <= ny && ny < W) {
                            used[nx][ny] = true;
                            if (d <= Math.abs(i - nx) + Math.abs(j - ny)) {
                                c[nx][ny] = c[i][j];
                            }
                        }
                    }
                }
            }
        }

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (c[i][j] == 'R') {
                    c[i][j] = 'Y';
                } else if (c[i][j] == 'Y') {
                    c[i][j] = 'G';
                } else if (c[i][j] == 'G') {
                    c[i][j] = 'B';
                }
            }
        }

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                System.out.print(c[i][j]);
            }
            System.out.println();
        }
    }
}
