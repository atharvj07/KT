n = int(input())
c = [[] for i in range(n)]
x = []
y = []
for i in range(n):
	c[i] = [int(j) for j in input().split()]
	assert c[i][0] == len(c[i]) - 1
	if len(c[i]) % 2 == 0:
		x += c[i][1:]
	else:
		y += c[i][1:]
x.sort(reverse = True)
y.sort()
sumx = 0
sumy = 0
for a, b in zip(x, y):
	sumx += a
	sumy += b
for a in x[len(y):]:
	sumx += a
for a in y[len(x):]:
	sumy += a
print(sumx, sumy)
