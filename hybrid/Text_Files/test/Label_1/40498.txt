from collections import defaultdict

n = int(input())
d = defaultdict(set)
for i in range(n):
    _, host, *path = input().split('/')
    if not path: path = ['']
    d[host].add('/'.join(path))

print(len(grp := {f'{k} #{i}' for i, k in enumerate(list(d)) if len(d[k])>1}))
for k in grp: print(*[k.split('#')[0] for k in grp if '#' and k.split('#')[1]==k.split('#')[1]])
