#include <iostream>
using namespace std;
int main() {
  long long n;
  cin >> n;
  for (long long k = (n + 5) / 6; k <= n / 2; k++) {
    long long rem = n;
    long long vasya = 0;
    while (rem > 0) {
      if (rem >= k) {
        rem -= k;
        vasya += k;
      } else {
        vasya += rem;
        rem = 0;
      }
      long long petya = (rem + 9) / 10;
      rem -= petya;
    }
    if (vasya >= n / 2) {
      cout << k << endl;
      return 0;
    }
  }
  return 0;
}
