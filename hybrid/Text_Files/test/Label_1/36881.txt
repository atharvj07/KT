import sys

a = sys.stdin.readline().strip()
b = sys.stdin.readline().strip()

if len(a) != len(b):
    print("NO")
else:
    for i in range(len(a)):
        if a[i] == b[i]:
            continue
        elif a[i] == '0' and b[i] == '1':
            a = a[:i] + '1' + a[i+1:]
        else:
            print("NO")
            break
    else:
        print("YES")
