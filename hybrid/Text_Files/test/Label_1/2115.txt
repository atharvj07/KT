#include<bits/stdc++.h>
using namespace std;
#define ll long long
struct seg{
	ll b,e;
	seg(ll b,ll e){this->b=b,this->e=e;}
	bool operator<(const seg& o){
		return this->e < o.e;
	}
};
int main(){
	int t;
	cin>>t;
	while(t--){
		ll n,k;
		cin>>n>>k;
		ll a1,a2;
		cin>>a1>>a2;
		ll b1,b2;
		cin>>b1>>b2;
		vector<seg> seg1,seg2;
		seg1.push_back(seg(a1,a2));
		seg2.push_back(seg(b1,b2));
		for(ll i=0;i<n-1;++i){
			seg1.push_back(seg1.back());
			seg2.push_back(seg2.back());
		}
		ll ans=0;
		sort(seg1.begin(),seg1.end());
		sort(seg2.begin(),seg2.end());
		while(k){
			if(seg1.empty() || seg2.empty())break;
			if(seg1.back().e > seg2.back().e){
				auto tmp = seg1.back();
				if(tmp.b >= seg2.back().b && tmp.b<=seg2.back().e){
					ll r=tmp.b-seg2.back().b;
					k-=r;
					seg2.back().b+=r;
				}
				else{
					ll r=seg2.back().b-tmp.b;
					k-=r;
					seg1.back().b+=r;
				}
				++ans;
			}
			else{
				auto tmp = seg2.back();
				if(tmp.b >= seg1.back().b && tmp.b<=seg1.back().e){
					ll r=tmp.b-seg1.back().b;
					k-=r;
					seg1.back().b+=r;
				}
				else{
					ll r=seg1.back().b-tmp.b;
					k-=r;
					seg2.back().b+=r;
				}
				++ans;
			}
			seg1.pop_back(),seg2.pop_back();
		}
		cout<<ans<<endl;
	}
	return 0;
}
