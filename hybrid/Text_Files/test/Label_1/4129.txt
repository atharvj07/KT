#include <bits/stdc++.h>
using namespace std;
#define int long long
const int N = 5005, mod = 1e9 + 7;
int32_t main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (auto &x : a) cin >> x;
    vector<int> dp(n + 1, LLONG_MIN / 2);
    dp[n] = 0;
    for (int i = n - 1; ~i; i--) {
        dp[i] = dp[i + 1];
        if (i + k <= n) {
            int sum = 0;
            for (int j = i; j < i + k; j++) sum += a[j];
            if (x >= k) {
                dp[i] = max(dp[i], dp[i + k] + sum);
            }
            if (x > 0) {
                dp[i] = max(dp[i], dp[i + 1] + a[i]);
            }
        }
    }
    cout << (dp[0] == LLONG_MIN / 2 ? -1 : dp[0]);
    return 0;
}
