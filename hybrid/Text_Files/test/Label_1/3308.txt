#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, a[100005], dp[100005][3];
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    dp[n][0] = dp[n][2] = 1;
    dp[n][1] = 0;
    for (int i = n - 1; i >= 1; i--) {
        if (a[i] == -1) {
            dp[i][0] = (dp[i + 1][0] + dp[i + 1][1]) % mod;
            dp[i][1] = (dp[i + 1][1] + dp[i + 1][2]) % mod;
            dp[i][2] = dp[i + 1][2];
        } else {
            if (a[i] > a[i + 1]) {
                dp[i][0] = dp[i][2] = 0;
                dp[i][1] = dp[i + 1][0];
            } else if (a[i] == a[i + 1]) {
                dp[i][0] = dp[i + 1][0];
                dp[i][1] = (dp[i + 1][0] + dp[i + 1][1]) % mod;
                dp[i][2] = dp[i + 1][2];
            } else {
                dp[i][0] = dp[i][1] = 0;
                dp[i][2] = dp[i + 1][0];
            }
        }
    }
    cout << dp[1][0];
}
