import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] l = new int[m];
        for (int i = 0; i < m; i++) {
            l[i] = scanner.nextInt();
        }
        int[] p = new int[m];
        Arrays.fill(p, -1);
        boolean[] colored = new boolean[n];
        int lastColor = 0;
        for (int i = 0; i < m; i++) {
            if (l[i] == n) {
                if (colored[0]) {
                    System.out.println("-1");
                    return;
                } else {
                    p[i] = 1;
                    for (int j = 0; j < n; j++) {
                        colored[j] = true;
                    }
                    lastColor = i + 1;
                }
            } else {
                for (int j = 0; j < n; j++) {
                    if (!colored[j] && j + l[i] <= n) {
                        p[i] = j + 1;
                        for (int k = j; k < j + l[i]; k++) {
                            colored[k] = true;
                        }
                        lastColor = i + 1;
                        break;
                    }
                }
                if (p[i] == -1) {
                    System.out.println("-1");
                    return;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            if (p[i] == -1) {
                for (int j = 0; j < n; j++) {
                    if (!colored[j]) {
                        p[i] = j + 1;
                    }
                }
            }
        }
        if (lastColor == m) {
            for (int i = 0; i < m; i++) {
                for (int j = 0; j < n; j++) {
                    if (!colored[j] && j + l[i] <= n) {
                        p[i] = j + 1;
                        for (int k = j; k < j + l[i]; k++) {
                            colored[k] = true;
                        }
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.print(p[i] + " ");
        }
    }
}
