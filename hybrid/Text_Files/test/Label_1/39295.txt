import sys
input = sys.stdin.readline

n, m, k = map(int, input().split())
g = [[] for i in range(n+1)]
for i in map(int, input().split()):
    g[i].append(i)
for i in range(m):
    a, b = map(int, input().split())
    g[a].append(b)
    g[b].append(a)
ans = 0
for i in range(1, n+1):
    if len(g[i])==1:
        ans += 1
        for j in g[i]:
            g[j].remove(i)
print(ans)
