import java.util.Arrays;
import java.util.Scanner;

public class Main {
    private static long[] masks;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = in.nextInt();
            masks[i] = 0;
            for (int j = 0; j < k; j++) {
                int b = in.nextInt();
                masks[i] |= (1L << b);
            }
        }
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            int op = in.nextInt();
            long m = 0;
            if (op != 0) {
                m = masks[in.nextInt()];
            }
            long val = 0;
            switch (op) {
                case 0:
                    System.out.println((val & (1L << in.nextInt())) == 0 ? 0 : 1);
                    break;
                case 1:
                    val |= m;
                    System.out.println(val);
                    break;
                case 2:
                    val &= ~m;
                    System.out.println(val);
                    break;
                case 3:
                    val ^= m;
                    System.out.println(val);
                    break;
                case 4:
                    System.out.println((val & m) == m ? 1 : 0);
                    break;
                case 5:
                    System.out.println((val & m) != 0 ? 1 : 0);
                    break;
                case 6:
                    System.out.println((val & m) == 0 ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(val & m));
                    break;
                case 8:
                    System.out.println(val & m);
                    break;
            }
        }
    }
}
