import bisect
import math

n = int(input())
cake = []
for _ in range(n):
    r, h = map(int, input().split())
    cake.append((r**2 * h, r**2 * math.pi))

cake.sort(key=lambda x: x[0])
dp = []
for _, s in cake:
    i = bisect.bisect_left(dp, s)
    if i < len(dp):
        dp[i] = s
    else:
        dp.append(s)
print(dp[-1])
