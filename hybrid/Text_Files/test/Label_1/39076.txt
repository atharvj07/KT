import sys
input = sys.stdin.readline

n, m = map(int, input().split())
t = list(map(int, input().split()))
a = [[] for i in range(m)]
for i in range(n):
    a[t[i] - 1].append(i + 1)
q = [list(map(int, input().split())) for i in range(m - 1)]
ans = [0] * m
for i in range(m - 1):
    ai, bi = q[i]
    ai, bi = ai - 1, bi - 1
    if not a[ai] or not a[bi]:
        continue
    if a[ai][-1] < a[bi][-1]:
        ans[i + 1] = ans[i] + 1
        continue
    lai, lbi = len(a[ai]), len(a[bi])
    j, k = lai - 1, lbi - 1
    while j >= 0 and k >= 0 and a[ai][j] >= a[bi][k]:
        if a[ai][j] > a[bi][k]:
            ans[i + 1] = ans[i] + 1
        j -= 1
        k -= 1
    a[bi].extend(a[ai])
    del a[ai]
for i in ans:
    print(i)
