import java.util.*;
class Main {
    public static void main(String[] args){
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] mass = new int[n];
        for(int i = 0;i < n;++i){
            mass[i] = input.nextInt();
        }
        int p = input.nextInt();
        for(int i = 0;i < p;++i){
            int a = input.nextInt() - 1;
            int b = input.nextInt() - 1;
            int ans = 0;
            for(int j = a;(j + b) < n;++j){
                ans += mass[j];
                j += b;
            }
            System.out.println(ans);
        }
    }
}
