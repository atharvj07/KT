import java.util.*;
import java.io.*;
import java.nio.file.StandardOpenOption;

class Main {
    static void solve(long[] arr, int n, int k) {
        if (n < k) {
            System.out.println("Incorrect sequence");
            return;
        }

        long[] sums = new long[n - k + 1];
        for (int i = 0; i < k; i++) {
            sums[0] += arr[i];
        }

        for (int i = 1; i < n - k + 1; i++) {
            sums[i] = sums[i - 1] + arr[i + k - 1] - arr[i - 1];
        }

        for (int i = 1; i < n - k + 1; i++) {
            if (sums[i] <= sums[i - 1]) {
                System.out.println("Incorrect sequence");
                return;
            }
        }

        long[] minAbsSumArr = new long[n];
        long minAbsSum = Long.MAX_VALUE;

        for (int i = 0; i <= 1000000000; i++) {
            long absSum = 0;
            int quesMarkCnt = 0;
            for (int j = 0; j < n; j++) {
                if (arr[j] == -1) {
                    quesMarkCnt++;
                } else {
                    absSum += Math.abs(arr[j]);
                }
            }

            if (quesMarkCnt == 0) {
                if (absSum < minAbsSum) {
                    minAbsSum = absSum;
                    minAbsSumArr = Arrays.copyOf(arr, n);
                }
                break;
            }

            int low = i - k + 1;
            int high = i;
            long segSum = sums[0] - i;

            for (int j = 1; j < n - k + 1; j++) {
                segSum += sums[j] - i;
                if (segSum < sums[j - 1]) {
                    segSum -= i;
                    while (segSum < sums[j - 1]) {
                        segSum += low;
                        low++;
                    }

                    if (low > high) {
                        break;
                    }

                    minAbsSumArr[j - quesMarkCnt] = low;
                    segSum -= low;
                    low++;
                }
            }

            if (low <= high) {
                for (int j = n - k + 1 - quesMarkCnt; j < n; j++) {
                    minAbsSumArr[j] = low;
                    low++;
                }

                if (absSum < minAbsSum) {
                    minAbsSum = absSum;
                    arr = Arrays.copyOf(minAbsSumArr, n);
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (arr[i] == -1) {
                arr[i] = minAbsSumArr[i];
            }

            System.out.print(arr[i] + " ");
        }

        System.out.println();
    }

    public static void main(String[] args) throws IOException {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();
        long[] arr = new long[n];

        for (int i = 0; i < n; i++) {
            String num = scanner.next();
            if (num.equals("?")) {
                arr[i] = -1;
            } else {
                arr[i] = Long.parseLong(num);
            }
        }

        solve(arr, n, k);
    }
}
