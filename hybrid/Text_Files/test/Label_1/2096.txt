#include <bits/stdc++.h>
using namespace std;
#define ll long long
bool cmp(pair<ll,ll> a,pair<ll,ll> b){
    return a.first<b.first;
}
ll n;
pair<ll,ll> arr[100005];
int main(){
    cin>>n;
    for(ll i=0;i<n;i++){
        cin>>arr[i].first>>arr[i].second;
    }
    sort(arr,arr+n);
    ll ans=0,cnt=0;
    for(ll i=1;i<n;i++){
        if(arr[i].first==arr[i-1].first){
            cnt++;
        }
        else{
            ans++;
            cnt=0;
        }
    }
    sort(arr,arr+n,cmp);
    ll ans2=0,cnt2=0;
    for(ll i=1;i<n;i++){
        if(arr[i].second==arr[i-1].second){
            cnt2++;
        }
        else{
            ans2++;
            cnt2=0;
        }
    }
    cout<<min(ans,ans2)<<"\n";
    return 0;
}
