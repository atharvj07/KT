import java.util.Scanner;
import java.math.BigInteger;

public class Main {
  static final int MOD = 1000000007;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[][] ps = new int[n][n];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        ps[i][j] = sc.nextInt();
      }
    }

    BigInteger p = BigInteger.ONE;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < i; j++) {
        if (ps[i][j] == 0) {
          continue;
        }
        // (100 - p[i][j]) / 100
        // = 1 - p[i][j] / 100
        // = 1 - (p[i][j] / 100 * p[j][i] / 100) / (100 - p[j][i])
        // = 1 - q[i][j]
        int q_ij = 10000 - ps[i][j] * ps[j][i];
        int sign = j % 2 == i % 2 ? -1 : 1;
        p = p.multiply(BigInteger.valueOf(q_ij).modPow(BigInteger.valueOf(n - 1), BigInteger.valueOf(MOD))).mod(BigInteger.valueOf(MOD));
      }
    }

    BigInteger q = BigInteger.ONE;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < i; j++) {
        q = q.multiply(BigInteger.valueOf(100 - ps[i][j])).mod(BigInteger.valueOf(MOD));
      }
    }

    System.out.println(p.multiply(q.modPow(BigInteger.valueOf(MOD - 2), BigInteger.valueOf(MOD))).mod(BigInteger.valueOf(MOD)));
  }
}
