        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt(), k = sc.nextInt(), fav[] = new int[k];
        for (int i = 0; i < k; i++) fav[i] = sc.nextInt();
        int n = sc.nextInt();

            for (int i = 0; i < actors; i++)
