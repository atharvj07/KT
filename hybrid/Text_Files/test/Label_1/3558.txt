#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define ll long long

int main() {
  int n;
  cin >> n;
  rep(_, n) {
    string s, t;
    cin >> s >> t;
    int m = s.length(), n = t.length();
    bool yes = true;
    int now = 0;
    rep(i, n) {
      if (s[now] == t[i]) now++;
      if (now == m) break;
    }
    yes &= now == m;
    m = t.length(), n = s.length();
    now = 0;
    rep(i, n) {
      if (s[now] == t[i]) now++;
      if (now == m) break;
    }
    yes &= now == m;
    cout << (yes ? "YES" : "NO") << endl;
  }
  return 0;
}
