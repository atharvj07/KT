#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353

ll binpow(ll b, ll p) {
  if (p == 0) return 1;
  if (p == 1) return b % mod;
  if (p % 2 == 1)
    return (b * 1ll * binpow(b, p - 1)) % mod;
  else {
    ll x = binpow(b, p / 2) % mod;
    return (x * 1ll * x) % mod;
  }
}

ll sum[1000001];

int main() {
  ll n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    ll x;
    cin >> x;
    sum[i] = sum[i - 1] + x;
  }
  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    ll val = binpow(2, n - i);
    val = val * 1ll * (sum[i - 1] + (sum[n] - sum[i]));
    val %= mod;
    ans += val;
    ans %= mod;
  }
  cout << ans << endl;
  return 0;
}
