#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        int ans = (1LL << (n - 1)) % mod;
        if (k == 0) ans = 1;
        cout << ans << "\n";
    }
    return 0;
}
