import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] time = sc.nextLine().split(":");
        String hours = time[0];
        String minutes = time[1];
        int maxHourValue = 0;
        int maxMinuteValue = 0;
        for (char c : hours.toCharArray()) {
            if (c >= '0' && c <= '9') {
                maxHourValue = Math.max(maxHourValue, c - '0');
            } else if (c >= 'A' && c <= 'Z') {
                maxHourValue = Math.max(maxHourValue, c - 'A' + 10);
            }
        }
        for (char c : minutes.toCharArray()) {
            if (c >= '0' && c <= '9') {
                maxMinuteValue = Math.max(maxMinuteValue, c - '0');
            } else if (c >= 'A' && c <= 'Z') {
                maxMinuteValue = Math.max(maxMinuteValue, c - 'A' + 10);
            }
        }
        maxMinuteValue += 1;
        if (maxHourValue == 0 || maxMinuteValue == 0) {
            System.out.println("0");
            return;
        }
        int minRadix = Math.max(maxHourValue, maxMinuteValue);
        if (minRadix <= 1) {
            System.out.println("0");
            return;
        }
        int maxRadix = 36;
        StringBuilder sb = new StringBuilder();
        for (int i = minRadix; i <= maxRadix; i++) {
            if (i != minRadix) {
                sb.append(' ');
            }
            sb.append(i);
        }
        if (minRadix > maxRadix) {
            sb.delete(0, sb.length());
            sb.append("-1");
        }
        System.out.println(sb);
    }
}
