#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ll n, a, b, w;
    cin >> n >> w;
    unordered_map<ll, ll> m;
    for (int i = 0; i < n; i++) {
        cin >> a;
        m[a]++;
    }
    vector<ll> cnt(w + 1, 0);
    for (auto &pr : m) {
        cnt[0] = 1;
        for (int i = 0; i <= w; i++) {
            cnt[i + pr.first] += cnt[i];
            cnt[i + pr.first] %= 1000000007;
        }
    }
    cout << cnt[w];
}
