import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    long x = in.nextLong();
    int dist = 0;
    for (int i = 0; i < n; i++) {
      char c = in.next().charAt(0);
      long d = in.nextLong();
      if (c == '+') {
        x += d;
      } else {
        if (x >= d) {
          x -= d;
        } else {
          dist++;
        }
      }
    }
    System.out.println(x + " " + dist);
  }
}
