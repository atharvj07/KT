import java.util.*;

public class Main {

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);

    int n = scanner.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = scanner.nextLong();
    }

    int q = scanner.nextInt();
    Event[] events = new Event[q];
    for (int i = 0; i < q; i++) {
      int type = scanner.nextInt();
      long x = scanner.nextLong();
      events[i] = new Event(type, x);
    }

    long[] result = solve(a, events);
    for (long value : result) {
      System.out.print(value + " ");
    }
  }

  private static long[] solve(long[] a, Event[] events) {
    int n = a.length;
    long[] result = new long[n];
    System.arraycopy(a, 0, result, 0, n);

    // Sort events by type
    Arrays.sort(events, Comparator.comparingInt(Event::getType));

    // Process events
    for (Event event : events) {
      int type = event.getType();
      long x = event.getX();

      if (type == 1) {
        // Receipt event
        result[event.getP() - 1] = x;
      } else {
        // Payout event
        for (int i = 0; i < n; i++) {
          if (result[i] < x) {
            result[i] = x;
          }
        }
      }
    }

    return result;
  }

  private static class Event {
    private int type;
    private long x;

    public Event(int type, long x) {
      this.type = type;
      this.x = x;
    }

    public int getType() {
      return type;
    }

    public long getX() {
      return x;
    }

    public int getP() {
      return type == 1 ? (int) x : 0;
    }
  }
}
