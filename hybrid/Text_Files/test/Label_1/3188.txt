#include <bitset>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> cnt(k);
  bitset<100001> bs;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    cnt[a - 1]++;
    bs |= (1ULL << a);
  }
  int ans = k;
  int min_cnt = n;
  for (int i = 0; i < k; i++) {
    int cur = n - cnt[i];
    if (bs.test(i + 1)) cur--;
    if (min_cnt > cur) {
      min_cnt = cur;
      ans = i + 1;
    }
  }
  cout << ans << "\n";
  return 0;
}
