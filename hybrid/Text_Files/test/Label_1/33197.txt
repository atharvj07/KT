import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        LIS(a, n);
    }

    static void LIS(int[] arr, int n) {
        int[] lis = new int[n];
        int[] ld = new int[n];
        Arrays.fill(lis, 1);
        Arrays.fill(ld, 1);
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (arr[i] > arr[j] && lis[i] < lis[j] + 1) {
                    lis[i] = lis[j] + 1;
                    ld[i] = j;
                }
            }
        }
        int len = 0;
        int idx = 0;
        for (int i = 0; i < n; i++) {
            if (lis[i] > len) {
                len = lis[i];
                idx = i;
            }
        }
        System.out.println(len);
        Stack<Integer> st = new Stack<>();
        while (true) {
            st.push(arr[idx]);
            if (idx == 0) break;
            idx = ld[idx];
        }
        while (!st.isEmpty()) {
            System.out.print(st.pop() + " ");
        }
    }
}
