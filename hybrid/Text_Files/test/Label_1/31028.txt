import java.util.Scanner;

public class Main {
  final static long INF = Long.MAX_VALUE / 2;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int V = sc.nextInt();
    int E = sc.nextInt();

    long[][] d = new long[V][V];

    for (int i = 0; i < V; i++) {
      for (int j = 0; j < V; j++) {
        d[i][j] = (i == j) ? 0 : INF;
      }
    }

    for (int i = 0; i < E; i++) {
      int s = sc.nextInt();
      int t = sc.nextInt();
      long w = sc.nextLong();
      d[s][t] = w;
    }

    for (int k = 0; k < V; k++) {
      for (int i = 0; i < V; i++) {
        for (int j = 0; j < V; j++) {
          if (d[i][k] == INF || d[k][j] == INF) continue;
          if (d[i][j] > d[i][k] + d[k][j]) {
            d[i][j] = d[i][k] + d[k][j];
          }
        }
      }
    }

    for (int i = 0; i < V; i++) {
      if (d[i][i] < 0) {
        System.out.println("NEGATIVE CYCLE");
        return;
      }
    }

    for (int i = 0; i < V; i++) {
      for (int j = 0; j < V; j++) {
        System.out.print((d[i][j] == INF) ? "INF" : d[i][j]);
        System.out.print(' ');
      }
      System.out.println();
    }
  }
}
