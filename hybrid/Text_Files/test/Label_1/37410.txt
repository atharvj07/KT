import math

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        b = list(map(int, input().split()))
        if any(math.gcd(x, y) > 1 for x, y in zip(b, b[1:])) or n % 2 != sum(b) % 2:
            print("NO")
            continue
        a = [0] * n
        a[0] = b[0] // n
        for i in range(1, n):
            a[i] = (b[i] - a[i - 1] * (n - i)) // i
        print("YES")
        print(*a)

solve()
