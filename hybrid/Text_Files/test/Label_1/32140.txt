import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        int[] minPos = new int[n + 1];
        int[] maxPos = new int[n + 1];
        Arrays.fill(minPos, n + 1);
        int curPos = 0;
        for (int i = 0; i < m; i++) {
            if (curPos == 0) {
                minPos[a[i]] = 1;
                maxPos[a[i]] = 1;
            } else {
                minPos[a[i]] = Math.min(minPos[a[i]], curPos + 1);
                maxPos[a[i]] = curPos + 1;
            }
            curPos++;
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(minPos[i] + " " + maxPos[i]);
        }
    }
}
