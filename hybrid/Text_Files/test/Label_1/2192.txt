#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m, r;
  cin >> n >> m >> r;
  vector<int> s(n), b(m);
  for (int i = 0; i < n; i++) cin >> s[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(s.begin(), s.end());
  sort(b.begin(), b.end());
  for (int i = 0; i < n && i < m && s[i] < b[i]; i++) r += b[i] - s[i];
  cout << r << endl;
}
