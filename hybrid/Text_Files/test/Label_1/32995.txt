import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    long n = in.nextLong();
    int k = in.nextInt();
    long[] hats = new long[k];
    for (int i = 0; i < k; ++i) { hats[i] = in.nextLong(); }
    Arrays.sort(hats);
    long ansCnt = 0;
    long ansVal = -1;
    for (long hat : hats) {
      if (n % hat == 0) {
        long cnt = n / hat;
        if (cnt > ansCnt) {
          ansCnt = cnt;
          ansVal = hat;
        }
      }
    }
    System.out.println(ansVal + " " + ansCnt);
  }
}
