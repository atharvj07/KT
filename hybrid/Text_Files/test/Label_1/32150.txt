import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    static class Edge {
        int a, b, w;

        Edge(int a, int b, int w) {
            this.a = a;
            this.b = b;
            this.w = w;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++)
            a[i] = Integer.parseInt(s[i]);
        List<Edge>[] adj = new ArrayList[n];
        for (int i = 0; i < n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            s = br.readLine().split(" ");
            int x = Integer.parseInt(s[0]) - 1;
            int y = Integer.parseInt(s[1]) - 1;
            adj[x].add(new Edge(x, y, 1));
            adj[y].add(new Edge(y, x, 1));
        }
        int min = Integer.MAX_VALUE, id = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] < min) {
                min = a[i];
                id = i;
            }
        }
        long ans = 0;
        int[] deg = new int[n], dis = new int[n];
        dfs(id, -1, adj, a, deg, dis, 0);
        for (int i = 0; i < n; i++) {
            ans += (long) deg[i] * a[i];
            for (Edge e : adj[i]) {
                if (e.b == id)
                    continue;
                ans += (long) (Math.log(dis[e.b]) / Math.log(2) + 1) * Math.min(a[i], a[e.b]);
            }
        }
        System.out.println(ans);
    }

    static void dfs(int u, int p, List<Edge>[] adj, int[] a, int[] deg, int[] dis, int d) {
        dis[u] = d;
        deg[u] = 1;
        for (Edge e : adj[u]) {
            if (e.b == p)
                continue;
            dfs(e.b, u, adj, a, deg, dis, d + 1);
            deg[u] += deg[e.b];
        }
    }
}
