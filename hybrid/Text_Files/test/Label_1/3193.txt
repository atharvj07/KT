#include <iostream>
using namespace std;

int main() {
    int n, M = 0, C = 0, m, c;
    cin >> n;
    while (n--) {
        cin >> m >> c;
        if (m > c) M++;
        else if (c > m) C++;
    }
    if (M > C) cout << "Mishka";
    else if (C > M) cout << "Chris";
    else cout << "Friendship is magic!^^";
    return 0;
}
