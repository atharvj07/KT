def find_petya_seat():
    classroom = [input() for _ in range(6)]
    for i in range(6):
        row = classroom[i]
        if '.' in row:
            for j in range(8):
                if row[j] == '.':
                    classroom[i] = row[:j] + 'P' + row[j+1:]
                    return "\n".join(classroom)

print(find_petya_seat())
