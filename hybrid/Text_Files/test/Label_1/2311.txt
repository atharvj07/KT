#include <bits/stdc++.h>
using namespace std;

int a[100001], b[100001];

int main() {
    int n; cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i], b[i] = a[i]; }
    sort(b + 1, b + 1 + n);
    int ans = 0, l = -1, r = -1;
    for (int i = 1; i <= n; i++) {
        if (a[i] == b[1]) l = i;
        if (a[i] == b[n]) r = i;
    }
    if (b[n] <= 2 * b[1]) {
        cout << 0 << endl;
        return 0;
    }
    if (b[2] > 2 * b[1]) {
        ans += l - 1;
        a[l] = b[2];
    }
    if (b[n] > 2 * b[n - 1]) {
        ans += n - r;
        a[r] = b[n - 1];
    }
    sort(a + 1, a + 1 + n);
    if (a[n] <= 2 * a[1]) {
        cout << ans << endl;
        return 0;
    }
    cout << -1 << endl;
}
