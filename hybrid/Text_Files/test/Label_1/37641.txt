import heapq
from collections import defaultdict

INF = 1<<60
n, m, k = map(int, input().split())
adj = defaultdict(lambda: defaultdict(int))
for _ in range(m):
    x, y, w = map(int, input().split())
    adj[x][y] = adj[y][x] = w
if k:
    storages = list(map(int, input().split()))
else:
    storages = []

dist = defaultdict(lambda: [INF]*n)
q = [(0,i) for i in range(1,n+1) if i not in storages]
while q:
    d, v = heapq.heappop(q)
    if v in dist and d<=dist[v][0]:
        for u in adj[v]:
            if u not in dist or d+adj[v][u]<dist[u][0]:
                heapq.heappush(q, (d+adj[v][u], u))
                dist[u][0] = d+adj[v][u]

ans = INF
for i in range(1,n+1):
    if i not in storages:
        ans = min(ans,min([dist[i][u] for u in storages if dist[i][u]<INF]))
print(ans if ans!=INF else -1)
