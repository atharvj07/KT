import sys

n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
m = int(input())
b = [list(map(int, input().split())) for _ in range(m)]
a.sort(key=lambda x: x[1], reverse=True)
b.sort(key=lambda x: x[1], reverse=True)
ans = 0
i = 0
j = 0
while i < n and j < m:
    if a[i][0] != b[j][0]:
        ans += a[i][1]
        i += 1
    else:
        j += 1
while i < n:
    ans += a[i][1]
    i += 1
while j < m:
    ans += b[j][1]
    j += 1
print(ans)
