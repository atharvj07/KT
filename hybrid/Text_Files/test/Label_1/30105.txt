import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] cities = new int[n];

        StringTokenizer tokenizer = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            cities[i] = Integer.parseInt(tokenizer.nextToken());
        }

        int[] min = new int[n];
        int[] max = new int[n];

        for (int i = 0; i < n; i++) {
            if (i == 0) {
                min[i] = cities[1] - cities[0];
                max[i] = cities[n - 1] - cities[0];
            } else if (i == n - 1) {
                min[i] = cities[n - 1] - cities[n - 2];
                max[i] = cities[n - 1] - cities[0];
            } else {
                min[i] = Math.min(cities[i + 1] - cities[i], cities[i] - cities[i - 1]);
                max[i] = Math.max(cities[n - 1] - cities[i], cities[i] - cities[0]);
            }
        }

        for (int i = 0; i < n; i++) {
            System.out.println(min[i] + " " + max[i]);
        }
    }
}
