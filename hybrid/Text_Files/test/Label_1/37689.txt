for _ in range(int(input())):
    n = int(input())
    k = 0
    ans = []
    while n > 0:
        if n % 10 != 0:
            ans.append(n)
            n -= 1
            k += 1
        else:
            x = n // 10
            n -= x * 10
            ans.append(x * 10)
            k += 1
    print(k)
    print(*ans)
