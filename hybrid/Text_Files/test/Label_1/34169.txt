import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {
    static class Edge implements Comparable<Edge> {
        int u, v, w;

        public Edge(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }

        @Override
        public int compareTo(Edge o) {
            return this.w - o.w;
        }
    }

    static int n, m;
    static List<Edge> edges = new ArrayList<>();
    static Map<Integer, Integer> mst = new HashMap<>();
    static int[] p = new int[100001];

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        for (int i = 1; i <= n; i++) {
            p[i] = i;
        }
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int w = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            edges.add(new Edge(i + 1, i + 2, w));
            if (b == 1) {
                mst.put(i + 1, i + 2);
            }
        }
        if (!check()) {
            System.out.println(-1);
            return;
        }
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        for (Edge edge : edges) {
            pq.add(edge);
        }
        int cnt = 0;
        while (!pq.isEmpty()) {
            Edge edge = pq.poll();
            int u = edge.u, v = edge.v;
            if (find(u) == find(v)) {
                continue;
            }
            if (mst.containsKey(u) && mst.get(u) == v) {
                p[find(u)] = find(v);
            } else {
                cnt++;
            }
        }
        if (cnt != n - 1) {
            System.out.println(-1);
            return;
        }
        for (int i = 1; i <= m; i++) {
            int u = edges.get(i - 1).u, v = edges.get(i - 1).v;
            System.out.println(u + " " + v);
        }
    }

    static int find(int x) {
        if (p[x] != x) {
            p[x] = find(p[x]);
        }
        return p[x];
    }

    static boolean check() {
        for (int u : mst.keySet()) {
            if (find(u) != find(mst.get(u))) {
                return false;
            }
        }
        return true;
    }
}
