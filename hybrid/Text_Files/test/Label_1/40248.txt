from collections import defaultdict, deque
n = int(input()); adj = defaultdict(list); q = deque(); ct = n; num = 0
for i in range(n-1): u, v = map(int, input().split()); adj[u].append(v); adj[v].append(u)
vis = [False] * (n+1); vis[1] = True; q.append(1)
while q:
    x = q.popleft()
    if len(adj[x]) == 1: num += 1
    for y in adj[x]:
        if not vis[y]:
            vis[y] = True; q.append(y)
print(num)
