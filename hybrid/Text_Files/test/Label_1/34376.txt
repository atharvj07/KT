import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        int b = input.nextInt();
        int c = input.nextInt();
        boolean can = false;
        for (int i = 0; i <= c/a; i++) {
            int rem = c - i*a;
            if (rem % b == 0) {
                can = true;
                break;
            }
        }
        System.out.println(can ? "Yes" : "No");
    }
}
