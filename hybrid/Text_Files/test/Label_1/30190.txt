import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long res = 1;
        for (int i = 0; i < n; i++) {
            long a = sc.nextLong();
            if (a != 0) {
                res *= a;
            } else {
                res = 0;
                break;
            }
        }
        System.out.println(res);
    }
}
