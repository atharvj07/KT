#include <iostream>
#include <vector>

using namespace std;

struct Task {
  int t, q;
};

int main() {
  int n, T;
  cin >> n >> T;
  vector<Task> tasks(n);
  for (int i = 0; i < n; i++) {
    cin >> tasks[i].t >> tasks[i].q;
  }
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (tasks[i].t <= T) {
      ans = max(ans, tasks[i].q);
      for (int j = 0; j < n; j++) {
        if (tasks[j].t + tasks[i].t <= T) {
          ans = max(ans, tasks[i].q + tasks[j].q);
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
