#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> b(n);
    for (long long &x : b) cin >> x;
    vector<long long> a(n);
    if (b[0] % n != 0) {
      cout << "NO" << endl;
      continue;
    }
    a[0] = b[0] / n;
    for (int i = 1; i < n; ++i) a[i] = b[i] - a[i - 1] - a[0];
    if (any_of(begin(a), end(a), [&](long long x) { return x <= 0; })) {
      cout << "NO" << endl;
      continue;
    }
    cout << "YES" << endl;
    for (int i = 0; i < n; ++i) cout << a[i] << " ";
    cout << endl;
  }
  return 0;
}
