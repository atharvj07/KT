#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int a[N];

int main() {
    int n, s;
    cin >> n >> s;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int mid = n / 2;
    int ans = 0;
    if (s < a[mid]) {
        for (int i = 0; i < mid; i++) ans += a[mid] - a[i];
    } else {
        for (int i = mid + 1; i < n; i++) ans += a[i] - a[mid];
    }
    cout << ans << endl;
}
