import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        long k = scanner.nextLong();
        long a = scanner.nextLong();
        long b = scanner.nextLong();

        long ans = (b / k) - ((a - 1) / k);

        System.out.println(ans);
    }
}
