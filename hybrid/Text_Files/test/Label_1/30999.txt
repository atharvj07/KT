import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }

            int[] factors = new int[m];
            for (int i = 0; i < m; i++) {
                factors[i] = sc.nextInt();
            }

            int lcm = 1;
            for (int factor : factors) {
                lcm = lcm(lcm, factor);
            }

            if (lcm > n) {
                System.out.println(0.0);
            } else {
                double expectedIncome = (double) n / 2;
                System.out.println(expectedIncome);
            }
        }
        sc.close();
    }

    private static int lcm(int a, int b) {
        return a * b / gcd(a, b);
    }

    private static int gcd(int a, int b) {
        if (a % b == 0) {
            return b;
        } else {
            return gcd(b, a % b);
        }
    }
}
