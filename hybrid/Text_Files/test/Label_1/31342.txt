import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] masks = new int[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            int mask = 0;
            for (int j = 0; j < k; j++) {
                int b = sc.nextInt();
                mask |= (1 << b);
            }
            masks[i] = mask;
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            switch (op) {
                case 0:
                    System.out.println(test(masks, m));
                    break;
                case 1:
                    set(masks, m);
                    break;
                case 2:
                    clear(masks, m);
                    break;
                case 3:
                    flip(masks, m);
                    break;
                case 4:
                    System.out.println(all(masks, m));
                    break;
                case 5:
                    System.out.println(any(masks, m));
                    break;
                case 6:
                    System.out.println(none(masks, m));
                    break;
                case 7:
                    System.out.println(count(masks, m));
                    break;
                case 8:
                    System.out.println(val(masks, m));
                    break;
            }
        }
    }

    static int test(int[] masks, int i) {
        return (masks[0] & (1 << i)) != 0 ? 1 : 0;
    }

    static void set(int[] masks, int i) {
        masks[0] |= (1 << i);
    }

    static void clear(int[] masks, int i) {
        masks[0] &= ~(1 << i);
    }

    static void flip(int[] masks, int i) {
        masks[0] ^= (1 << i);
    }

    static int all(int[] masks, int i) {
        return (masks[0] & masks[i]) == masks[i] ? 1 : 0;
    }

    static int any(int[] masks, int i) {
        return (masks[0] & masks[i]) != 0 ? 1 : 0;
    }

    static int none(int[] masks, int i) {
        return (masks[0] & masks[i]) == 0 ? 1 : 0;
    }

    static int count(int[] masks, int i) {
        int count = 0;
        for (int j = 0; j < 64; j++) {
            if ((masks[0] & masks[i] & (1 << j)) != 0) {
                count++;
            }
        }
        return count;
    }

    static int val(int[] masks, int i) {
        int val = 0;
        for (int j = 0; j < 64; j++) {
            if ((masks[0] & masks[i] & (1 << j)) != 0) {
                val += (1 << j);
            }
        }
        return val;
    }
}
