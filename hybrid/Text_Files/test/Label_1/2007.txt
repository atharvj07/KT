#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  
  int freq[2] = {0};
  for (int i = 0; i < n; i++) {
    freq[s[i] - '0']++;
  }
  
  int odd = 0;
  for (int i = 0; i < 2; i++) {
    odd += freq[i] % 2;
  }
  
  if (odd <= 1) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
