from collections import Counter

def solve():
    n, m, x = map(int, input().split())
    h = list(map(int, input().split()))
    c = Counter(h)
    if m > len(c):
        return 'NO'
    sh = sorted(h)
    i = 0
    j = min(m - 1, n - 1)
    ans = []
    while j >= 0:
        if sh[j] + x < sh[i]:
            j -= 1
            continue
        if c[sh[j]] == 0:
            j -= 1
            continue
        while sh[i] + x < sh[j]:
            i += 1
        if c[sh[i]] == 0:
            i += 1
            continue
        for _ in range(c[sh[i]]):
            ans.append(j + 1)
        c[sh[i]] -= 1
        j -= 1
    for i in range(n):
        if c[sh[i]] > 0:
            ans.append(m)
        c[sh[i]] -= 1
    return 'YES\n' + ' '.join(map(str, ans))

t = int(input())
for i in range(t):
    print(solve())
