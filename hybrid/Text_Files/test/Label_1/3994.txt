#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int lis = 0;
    for (int i = 0; i < n; i++) {
      int c = 1;
      for (int j = i + 1; j < n; j++) {
        if (a[j] > a[j - 1]) { c++; }
      }
      lis = max(lis, c);
      if (lis > 1) { break; }
    }
    if (lis > 1) { cout << "YES" << endl; } else { cout << "NO" << endl; }
  }
  return 0;
}
