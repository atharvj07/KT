import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n=sc.nextInt(), m=sc.nextInt(), q=sc.nextInt();
        int[][] arr = new int[n+1][m+1];
        for(int i=1; i <= n; i++) {
            for(int j=1; j <= m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        while(q-- > 0) {
            int a1 = sc.nextInt();
            int b1 = sc.nextInt();
            int c1 = sc.nextInt();
            int d1 = sc.nextInt();
            int h = sc.nextInt();
            int w = sc.nextInt();
            int a2 = c1, b2 = d1, c2 = a1, d2 = b1;
            for(int i = 0; i < h; i++) {
                for(int j = 0; j < w; j++) {
                    int tmp = arr[a1+i][b1+j];
                    arr[a1+i][b1+j] = arr[a2+i][b2+j];
                    arr[a2+i][b2+j] = tmp;
                }
            }
        }
        for(int i=1; i <= n; i++) {
            for(int j=1; j <= m; j++) {
                System.out.print(arr[i][j] + " ");
            }
            System.out.println();
        }
    }
}
