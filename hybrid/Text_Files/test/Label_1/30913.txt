import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int k = in.nextInt();
        for (int i = 0; i < k; i++) {
            b[i] = in.nextInt();
        }
        int[] pos = new int[5000001];
        Arrays.fill(pos, -1);
        for (int i = 0; i < n; i++) {
            pos[a[i]] = i;
        }
        boolean[] used = new boolean[n];
        StringBuilder sb = new StringBuilder();
        if (solve(pos, a, b, used, 0, sb)) {
            System.out.println("YES");
            System.out.println(sb);
        } else {
            System.out.println("NO");
        }
    }

    private static boolean solve(int[] pos, int[] a, int[] b, boolean[] used, int cur, StringBuilder sb) {
        if (cur == b.length) {
            return true;
        }
        for (int i = 0; i < a.length; i++) {
            if (used[i] || pos[b[cur]] == -1) {
                continue;
            }
            int j = pos[b[cur]];
            if (j > i && !used[j - 1] && a[j - 1] > a[i]) {
                used[j - 1] = true;
                sb.append((j - 1 + 1) + " L\n");
                if (solve(pos, a, b, used, cur + 1, sb)) {
                    return true;
                }
                used[j - 1] = false;
                sb.deleteCharAt(sb.length() - 1);
                sb.deleteCharAt(sb.length() - 1);
                sb.deleteCharAt(sb.length() - 1);
            }
            if (j < i && !used[j + 1] && a[j + 1] > a[i]) {
                used[j + 1] = true;
                sb.append((j + 1 + 1) + " R\n");
                if (solve(pos, a, b, used, cur + 1, sb)) {
                    return true;
                }
                used[j + 1] = false;
                sb.deleteCharAt(sb.length() - 1);
                sb.deleteCharAt(sb.length() - 1);
                sb.deleteCharAt(sb.length() - 1);
            }
        }
        return false;
    }
}
