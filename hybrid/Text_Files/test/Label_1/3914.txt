#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Block {
    int c1, val, c2;
};

int main() {
    int n;
    cin >> n;
    vector<Block> blocks(n);
    for (int i = 0; i < n; i++) {
        cin >> blocks[i].c1 >> blocks[i].val >> blocks[i].c2;
    }
    vector<vector<int>> dp(n, vector<int>(5, -1));
    for (int i = 0; i < n; i++) {
        dp[i][blocks[i].c1] = blocks[i].val;
        dp[i][blocks[i].c2] = blocks[i].val;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 1; j <= 4; j++) {
            if (dp[i][j] == -1) continue;
            for (int k = 1; k <= 4; k++) {
                if (i == 0 || j == k) {
                    dp[i][j] = max(dp[i][j], blocks[i].val);
                } else {
                    dp[i][j] = max(dp[i][j], dp[i - 1][k] + blocks[i].val);
                }
            }
            ans = max(ans, dp[i][j]);
        }
    }
    cout << ans << '\n';
}
