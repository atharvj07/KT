#include <bits/stdc++.h>
typedef long long LL;
const int MAXN = 1e5 + 5;
LL a[MAXN];
LL gcd(LL x, LL y) { return y ? gcd(y, x % y) : x; }
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::cin >> n;
    LL mn = 1e18;
    for (int i = 1; i <= n; i++) {
      std::cin >> a[i];
      mn = std::min(mn, a[i]);
    }
    bool flag = true;
    for (int i = 2; i <= n; i++) {
      if (gcd(a[i - 1], a[i]) != mn) {
        flag = false;
        break;
      }
    }
    if (flag)
      std::cout << "YES\n";
    else
      std::cout << "NO\n";
  }
}
