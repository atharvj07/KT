#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<int> boring;
  bool watching = true;
  for (int i = 0; i < n; i++) {
    int t;
    cin >> t;
    if (watching) {
      if ((i > 13 && boring[boring.size() - 1] == t - 1) ||
          boring.size() == 14) {
        cout << t - 1 << endl;
        watching = false;
      } else
        boring.push_back(t);
    }
  }
  if (watching) cout << 90 << endl;
  return 0;
}
