import math
from collections import Counter

def dist(p1, p2):
    return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2

n = int(input())
ants = [tuple(map(int,input().split())) for i in range(n)]

for q in range(int(input())):
    dsq = list(map(int,input().split()))
    dsq.sort()
    cnt = Counter(dsq)
    res = []

    for i in range(n):
        for j in range(i+1,n):
            d = dist(ants[i],ants[j])
            if d in cnt and cnt[d]%2==1:
                cnt[d] -= 1
                x = math.sqrt(dsq[0]+d-dsq[i])
                y = math.sqrt(dsq[0]+d-dsq[j])
                res.append((round(x),round(y)))
    print(len(res))
    res.sort()
    print(*res)
