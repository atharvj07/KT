import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int t = sc.nextInt();
        int r = sc.nextInt();
        if (r > t)
            System.out.println(-1);
        else {
            int ans = 0;
            int[] a = new int[m];
            for (int i = 0; i < m; i++)
                a[i] = sc.nextInt();
            for (int i = 0; i < m; i++) {
                if (i == 0 || a[i] - ans > t)
                    ans += r;
                else if (a[i] - ans <= t)
                    ans--;
            }
            System.out.println(ans);
        }
    }
}
