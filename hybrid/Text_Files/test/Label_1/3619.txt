#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

const int MAX = 1 << 16;

int fre[MAX];
int main() {
  int n, x;
  cin >> n >> x;
  for (int i = 0; i < n; i++) {
    int a;
    cin >> a;
    fre[a]++;
  }
  ll ans = 0;
  for (int i = 0; i < MAX; i++) {
    if (i == x)
      ans += ll(fre[i]) * fre[i] - 1;
    else
      ans += ll(fre[i]) * fre[x ^ i];
  }
  ans >>= 1;
  cout << ans << endl;
}
