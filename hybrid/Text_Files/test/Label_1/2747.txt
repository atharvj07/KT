#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) { std::cin >> a[i]; }

  int l = 0, r = n - 1, ans = 0;
  while (l <= r) {
    if (a[l] <= k || a[r] <= k) {
      if (a[l] <= a[r]) {
        l++;
      } else {
        r--;
      }
      ans++;
    } else {
      break;
    }
  }

  std::cout << ans << std::endl;

  return 0;
}
