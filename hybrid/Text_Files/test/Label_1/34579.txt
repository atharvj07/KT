import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int A = scanner.nextInt();
        int B = scanner.nextInt();

        int minPowerStrips = 1;
        int availableSockets = A;

        while (availableSockets < B) {
            availableSockets += A - 1;
            minPowerStrips++;
        }

        System.out.println(minPowerStrips);
    }
}
