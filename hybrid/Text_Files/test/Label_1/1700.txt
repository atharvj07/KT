#include <bits/stdc++.h>
using namespace std;
const int N = 5005;
int n, k, x, a[N], sum[N], dp[N][N], pre[N][N];

int main() {
	cin >> n >> k >> x;
	for (int i = 1; i <= n; i++) cin >> a[i], sum[i] = sum[i - 1] + a[i];
	for (int i = 1; i <= n; i++) {
		if (i >= k) dp[i][1] = sum[i] - sum[i - k];
		for(int j = 2; j <= x; j++){
			for (int l = i; l >= max(i - k + 1, j); l--) {
				if (dp[i][j] < dp[l-1][j-1] + sum[i] - sum[l - 1]) {
					dp[i][j] = dp[l - 1][j - 1] + sum[i] - sum[l - 1];
					pre[i][j] = l;
				}
			}
		}
	}
	if (dp[n][x] == 0) cout << -1 << endl;
	else {
		vector<int> b;
		int i = n, j = x;
		while (i >= 1 && j >= 1) {
			b.push_back(pre[i][j]);
			i = pre[i][j] - 1;
			j--;
		}
		cout << dp[n][x] << endl;
	}
}
