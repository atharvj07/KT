#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n;
  cin >> n;
  vector<long long int> b(n);
  long long int sum = 0;
  for (int i = 0; i < n; i++) { cin >> b[i]; }
  for (auto a : b) sum += a;
  long long int avg = sum / n;
  long long int ans = 0;
  for (auto a : b) ans += abs(avg - a);
  ans /= 2;
  cout << ans << endl;
  return 0;
}
