import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            masks[i] = 0L;
            for (int j = 0; j < k; j++) {
                int b = sc.nextInt();
                masks[i] |= (1L << b);
            }
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            switch (op) {
                case 0:
                    System.out.println(((masks[m] >> i) & 1L) == 1 ? 1 : 0);
                    break;
                case 1:
                    masks[m] |= masks[0];
                    break;
                case 2:
                    masks[m] &= ~masks[0];
                    break;
                case 3:
                    masks[m] ^= masks[0];
                    break;
                case 4:
                    System.out.println((masks[m] & masks[0]) == masks[0] ? 1 : 0);
                    break;
                case 5:
                    System.out.println((masks[m] & masks[0]) != 0 ? 1 : 0);
                    break;
                case 6:
                    System.out.println((masks[m] & masks[0]) == 0 ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(masks[m] & masks[0]));
                    break;
                case 8:
                    System.out.println(masks[m] & masks[0]);
                    break;
            }
        }
    }
}
