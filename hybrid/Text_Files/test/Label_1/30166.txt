import java.util.*;

public class Main {
    private static List<MemoryBlock> memory = new ArrayList<>();
    private static int idGenerator = 0;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        int m = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            String command = scanner.next();
            switch (command) {
                case "alloc":
                    int n = scanner.nextInt();
                    int id = alloc(n);
                    if (id != -1) {
                        System.out.println(id);
                    } else {
                        System.out.println("NULL");
                    }
                    break;
                case "erase":
                    int id2 = scanner.nextInt();
                    if (erase(id2)) {
                        System.out.println("ILLEGAL_ERASE_ARGUMENT");
                    }
                    break;
                case "defragment":
                    defragment();
                    break;
            }
        }
    }

    private static int alloc(int n) {
        int start = -1;
        int end = -1;
        for (int i = 0; i < memory.size(); i++) {
            if (!memory.get(i).isUsed()) {
                if (start == -1) {
                    start = i;
                }
                end = i;
            }
        }
        if (end - start + 1 >= n) {
            for (int i = start; i <= end; i++) {
                memory.get(i).setUsed(true);
            }
            idGenerator++;
            return idGenerator;
        }
        return -1;
    }

    private static boolean erase(int id) {
        if (id <= 0 || id > idGenerator) {
            return true;
        }
        for (int i = 0; i < memory.size(); i++) {
            if (memory.get(i).isUsed() && memory.get(i).getId() == id) {
                memory.get(i).setUsed(false);
                return false;
            }
        }
        return true;
    }

    private static void defragment() {
        List<MemoryBlock> newMemory = new ArrayList<>();
        for (MemoryBlock block : memory) {
            if (block.isUsed()) {
                newMemory.add(block);
            }
        }
        memory = newMemory;
    }

    private static class MemoryBlock {
        private boolean used;
        private int id;

        public MemoryBlock(int id) {
            this.id = id;
        }

        public boolean isUsed() {
            return used;
        }

        public void setUsed(boolean used) {
            this.used = used;
        }

        public int getId() {
            return id;
        }
    }
}
