#include <iostream>
#include <utility>
#include <vector>
#define ll long long
#define MX 100005
using namespace std;
 
ll pow10[MX];
const ll MD = 1000000007LL;
 
int main() {
    int n;
    cin >> n;
    vector<ll> v(n);
    ll ans = 1;
    for (auto &x : v) {
        cin >> x;
        while (x % 10 == 0) {
            x /= 10;
        }
        ans = (ans * x) % MD;
    }
    cout << ans << endl;
    return 0;
}
