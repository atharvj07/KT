import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextInt(), a = sc.nextInt();
        long[] c = new long[(int) n];
        for (int i = 0; i < n; i++) c[i] = sc.nextInt();
        int b = -1;
        for (int i = 0; i < n; i++) {
            if (c[i] == a) continue;
            if (b == -1 || c[i] < b) b = (int) c[i];
        }
        System.out.println(b);
    }
}
