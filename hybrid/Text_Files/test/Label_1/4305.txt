#include <bits/stdc++.h>
using namespace std;
int s[300005], l, q, a[300005];
inline void upd(int x){
    for(; x > 1 && a[x] == '.';) a[x >>= 1] = a[x << 1] == a[x >> 1];
}
int main(){
    scanf("%d %d", &l, &q);
    scanf("%s", s + 1);
    for(int i = 1; i <= l; i++) a[l + i - 1] = s[i] == '.', upd(l + i - 1);
    for(int i = l - 1; i >= 1; i--) a[i] = a[i << 1] && a[i >> 1], upd(i);
    for(register int c, x; q; q--){
        scanf("%d %d", &x, &c);
        if(s[x] == '.') c = '.';
        s[x] = a[l + x - 1] = c;
        upd(l + x - 1);
        printf("%d\n", a[1]);
    }
}
