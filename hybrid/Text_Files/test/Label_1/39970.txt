
import sys
from math import acos, sqrt

def dot(a, b):
    return sum(x*y for x, y in zip(a, b))

def norm(a):
    return sqrt(dot(a, a))

def angle(a, b):
    return acos(dot(a, b) / (norm(a)*norm(b)))

n = int(input())
a = [(int(x) for x in input().split()) for _ in range(n)]
b = [0] * n
for i in range(1, n):
    for j in range(i + 1, n + 1):
        if angle(a[i - 1], a[j - 1]) < acos(0):
            b[i] += 1
            b[j] += 1
print(sum(x == 0 for x in b))
print(*[i+1 for i in range(n) if b[i] == 0])
