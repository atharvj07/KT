import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            long a = scanner.nextLong();
            long mask = 0;
            while (a != 0) {
                mask |= a;
                a >>= 1;
            }
            System.out.println(mask);
        }
    }
}
