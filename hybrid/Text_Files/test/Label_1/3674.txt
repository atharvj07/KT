#include<bits/stdc++.h>
using namespace std;
#define INF 2000000000
int n, k, m, c;
int a[5][55];
int x[25], y[25];
int main() {
    cin >> n >> k;
    for (int i = 1; i <= 4; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> a[i][j];
            if (a[i][j] && !x[a[i][j]]) {
                x[a[i][j]] = i;
                y[a[i][j]] = j;
            }
        }
    }
    for (int i = 1; i <= 4; i += 3) {
        for (int j = 1; j <= n; j++) {
            for (int h = 1; h <= k; h++) {
                if (a[i][j] == h) {
                    if (j != y[h]) { m += abs(j - y[h]); a[i][j] = 0; a[i][y[h]] = h; break; }
                    if (x[h] != i) { m += abs(x[h] - i); a[x[h]][y[h]] = 0; a[i][y[h]] = h; x[h] = i; break; }
                }
            }
        }
    }
    for (int i = 2; i <= 4; i += 3) {
        for (int j = 1; j <= n; j++) {
            for (int h = 1; h <= k; h++) {
                if (a[i][j] == h) {
                    if (j != y[h]) { m += abs(j - y[h]); a[i][j] = 0; a[i][y[h]] = h; break; }
                    if (x[h] != i) { m += abs(x[h] - i); a[x[h]][y[h]] = 0; a[i][y[h]] = h; x[h] = i; break; }
                    if (c) { m += abs(c - i); a[c][j] = 0; a[i][j] = h; c = 0; break; }
                    else { c = i; break; }
                }
            }
        }
    }
    if (m <= 20000) { cout << m << endl; for (int i = 1; i <= 4; i++) { for (int j = 1; j <= n; j++) { for (int h = 1; h <= k; h++) { if (a[i][j] == h) cout << h << " " << i << " " << j << endl; } } } } else cout << -1 << endl;
}
