import sys
input = sys.stdin.readline

n = int(input())
b = list(map(int, input().split()))
a = sorted(b)
d = a[-1] - a[0]
if d == 0:
    print(0)
else:
    l = (min(b) - a[0] + d) // d
    r = (a[-1] - max(b) + d) // d
    if a[l] == a[0] + l * d and a[n - 1 - r] == a[-1] - r * d:
        print(l + r)
    else:
        print(-1)
