#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define pii pair<int, int>
int n, par[5001], sz[5001];
vector<pii> edges;
bool g[5001][5001];

int findpar(int u) {
  if (par[u] == u) return u;
  return par[u] = findpar(par[u]);
}

bool merge(int a, int b) {
  a = findpar(a);
  b = findpar(b);
  if (a == b) return false;
  if (sz[a] < sz[b]) swap(a, b);
  par[b] = a;
  sz[a] += sz[b];
  return true;
}

void solve() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    par[i] = i;
    sz[i] = 1;
  }
  memset(g, 0, sizeof(g));
  map<ll, vector<int>> mp;
  for (int i = 1; i <= n; i++) {
    ll x;
    cin >> x;
    mp[x].push_back(i);
  }
  for (auto x : mp) {
    vector<int> v = x.second;
    for (int i = 0; i < v.size(); i++) {
      for (int j = i + 1; j < v.size(); j++) {
        if (merge(v[i], v[j])) {
          edges.push_back({v[i], v[j]});
          g[v[i]][v[j]] = g[v[j]][v[i]] = 1;
        }
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      if (g[i][j] == 0 && findpar(i) != findpar(j)) {
        edges.push_back({i, j});
        merge(i, j);
      }
    }
  }
  int sz = edges.size();
  if (sz == n - 1) {
    cout << "YES\n";
    for (int i = 0; i < sz; i++) cout << edges[i].first << " " << edges[i].second
                                     << "\n";
  } else
    cout << "NO\n";
}

int main() {
  int t;
  cin >> t;
  while (t--) solve();
  return 0;
}
