#include <bits/stdc++.h>
using namespace std;
long long n, m, p = 1, ans = -1e18;
int a[100009], s[100009];
void get(int x) {
  long long ans = 0;
  for (int i = n - 1; i >= 0; i--) {
    ans += 1ll * a[i] * (x & p);
    x >>= 1;
  }
  ::ans = max(::ans, ans);
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) {
    cin >> s[i];
    m += 1ll * s[i] * p;
    p *= 2;
  }
  for (long long x = 0; x <= m; x++) get(x);
  cout << ans;
}
