import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int p = sc.nextInt();
        int q = sc.nextInt();
        int r = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        long ans = Long.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                long sum = 0;
                for (int k = j; k < n; k++) {
                    sum = Math.max(sum, 1L * p * a[i] + 1L * q * a[j] + 1L * r * a[k]);
                    ans = Math.max(sum, ans);
                }
            }
        }
        System.out.println(ans);
    }
}
