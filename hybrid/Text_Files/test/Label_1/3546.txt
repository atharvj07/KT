#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    int n, d;
    cin >> n >> d;
    int arr[n], dp[n][d + 1];
    for (int i = 0; i < n; i++) cin >> arr[i];
    for (int t = d; t >= 0; t--) {
        dp[n - 1][t] = (arr[n - 1] <= t) && (t - arr[n - 1] >= 10);
    }
    for (int i = n - 2; i >= 0; i--) {
        for (int t = d; t >= 0; t--) {
            int ans = 0;
            if (arr[i] + 10 <= t) {
                ans = max(dp[i + 1][t - (arr[i] + 10)] + 1, ans);
            }
            ans = max(dp[i + 1][t], ans);
            dp[i][t] = ans;
        }
    }
    cout << (dp[0][d] != 0 ? dp[0][d] : -1) << endl;
}
