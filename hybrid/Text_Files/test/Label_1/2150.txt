#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 1e5 + 5;
vector<int> adj[MAXN];
ll sum[MAXN];
bool vis[MAXN];

void dfs(int u, int p, ll w) {
  vis[u] = true;
  sum[u] = w;
  for (auto v : adj[u]) {
    if (v != p) {
      dfs(v, u, w + 1);
    }
  }
}

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  dfs(1, 1, 0);
  bool ok = true;
  for (int i = 1; i <= n; i++) {
    ok &= (sum[i] % 2 == 0);
  }
  if (ok) {
    cout << "YES" << endl;
  } else {
    cout << "NO" << endl;
  }
  return 0;
}
