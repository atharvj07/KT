#include <bits/stdc++.h>
using namespace std;

int main() {
    string s;
    cin >> s;
    int n = s.size();
    int ans = 0, cnt = 0;
    char cur = 'w';
    for (int i = 0; i < n; i++) {
        if (s[i] == cur) {
            cnt++;
        } else {
            ans = max(ans, cnt);
            cnt = 1;
            cur = s[i];
        }
    }
    ans = max(ans, cnt);
    cout << ans << endl;
    return 0;
}
