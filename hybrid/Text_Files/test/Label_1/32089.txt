import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long x = sc.nextLong();
        long b = 1;
        for (long i = 2; i < x; i++) {
            long num = 0;
            long cur = i;
            for (int j = 0; j < n; j++) {
                num = num * i + cur % i;
                cur /= i;
            }
            if (num % i == 0) {
                b = i;
            }
        }
        if (b == 1) {
            System.out.println(-1);
        } else {
            System.out.println(b);
        }
    }
}
