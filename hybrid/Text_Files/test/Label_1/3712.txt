#include <bits/stdc++.h>
#define ll long long

using namespace std;

int main() {
  ll n, l, r;
  char dir;
  cin >> n;
  ll lmax = 1e9, rmin = 1e9;
  for (int i = 0; i < n; i++) {
    cin >> dir;
    cin >> l;
    if (dir == 'L') {
      lmax = min(lmax, l);
    } else {
      rmin = min(rmin, l);
    }
  }
  if (lmax < rmin)
    cout << lmax;
  else
    cout << -1;
}
