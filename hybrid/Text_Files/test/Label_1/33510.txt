import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        if (!check(a)) {
            System.out.println(-1);
            return;
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            int mex = 0;
            for (int j = 0; j < i; j++) {
                if (b[j] == mex) {
                    mex++;
                }
            }
            b[i] = mex;
        }
        for (int i = 0; i < n; i++) {
            System.out.print(b[i] + " ");
        }
    }

    private static boolean check(int[] a) {
        int max = 0;
        for (int i : a) {
            max = Math.max(max, i);
        }
        int[] cnt = new int[max + 1];
        for (int i : a) {
            cnt[i]++;
        }
        for (int i = 0; i <= max; i++) {
            if (cnt[i] > 1) {
                return false;
            }
        }
        return true;
    }
}
