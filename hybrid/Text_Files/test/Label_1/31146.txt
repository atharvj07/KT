import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String time = scanner.nextLine();
        boolean[] isDigit = new boolean[128];
        for (char c : time.toCharArray()) {
            isDigit[c] = true;
        }
        int maxDigit = 0;
        for (char c : time.toCharArray()) {
            if (c >= 'A' && c <= 'Z') {
                maxDigit = Math.max(maxDigit, c - 'A' + 10);
            } else {
                maxDigit = Math.max(maxDigit, c - '0');
            }
        }
        if (maxDigit == 0) {
            System.out.println(0);
            return;
        }
        int ans = 0;
        for (int i = maxDigit + 1; i <= 36; i++) {
            boolean flag = true;
            for (char c : time.toCharArray()) {
                if (c >= 'A' && c <= 'Z') {
                    if (!isDigit[c - 'A' + 10]) {
                        flag = false;
                        break;
                    }
                } else {
                    if (!isDigit[c - '0']) {
                        flag = false;
                        break;
                    }
                }
            }
            if (flag) {
                if (ans == 0) {
                    ans = i;
                } else {
                    ans = -1;
                    break;
                }
            }
        }
        System.out.println(ans);
        scanner.close();
    }
}
