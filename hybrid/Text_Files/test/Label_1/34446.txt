import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int cnt = 0;
        for (int i = 1, j = 1; i <= n; i++) {
            int num = i, check = i;
            boolean flag = false;
            while (num > 0) {
                int digit = num % 10;
                if (digit != 4 && digit != 7) {
                    flag = true;
                    break;
                }
                num /= 10;
            }
            while (check > 0) {
                int digit = check % 10;
                if (digit == 4 || digit == 7) {
                    j++;
                }
                check /= 10;
            }
            if (!flag && j == k) {
                cnt++;
            }
        }
        System.out.println(cnt);
    }
}
