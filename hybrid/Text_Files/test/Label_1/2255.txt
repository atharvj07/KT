#include <bits/stdc++.h>

using namespace std;

int main() {
  string home, away;
  int n;
  cin >> home >> away >> n;

  map<pair<string, int>, int> mymap; // [team, id] -> last yellow card
  while (n--) {
    int t, m;
    char h, c;
    cin >> t >> h >> m >> c;
    if (c == 'y') mymap[{string(1, h), m}] = t;
    else
      cout << (h == 'h' ? home : away) << ' ' << m << ' ' << t << '\n';
  }
  return 0;
}
