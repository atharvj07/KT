import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long m = sc.nextLong();

        long x = 1;
        long s = 1;
        while (s <= m) {
            x++;
            s += x * x * x;
        }
        s -= x * x * x;
        x--;
        System.out.println(x * x * x - s + 1 + " " + s);
    }
}
