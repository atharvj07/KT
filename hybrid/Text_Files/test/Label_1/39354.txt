T = int(input().strip())
for _ in range(T):
    A, B, C, D = map(int, input().split())
    
    total_toys = A + B + C + D
    odd_counts = sum(1 for x in [A, B, C, D] if x % 2 != 0)

    # Conditions for a box to potentially be the special box
    can_be_first = (A % 2 != 0) or (B % 2 != 0 and D % 2 != 0)
    can_be_second = ((B % 2 != 0 and not (D % 2 != 0 and A == 0)) or
                     (odd_counts == 0 and total_toys > 1) or
                     (odd_counts == 2 and B == 0 and A != 0 and D != 0))
    can_be_third = False  # Cannot be the special box due to multiplication properties
    can_be_fourth = (D % 2 != 0) or (B % 2 != 0 and A % 2 != 0)

    print(" ".join(["Ya" if can else "Tidak" for can in [can_be_first, can_be_second, can_be_third, can_be_fourth]]))
