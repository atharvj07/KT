import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] t = new long[n];
        for (int i = 0; i < n; i++) {
            t[i] = scanner.nextInt();
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (t[i] == -t[j]) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
