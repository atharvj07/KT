#include <cstdio>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n, x;
    vector<int> a;
    scanf("%d", &n);
    for (int i = 0; i < 2 * n; i++) {
        scanf("%d", &x);
        a.push_back(x);
    }
    sort(a.begin(), a.end());
    long long res = 1LL * (a[n] - a[n-1]) * (a[2*n-1] - a[2*n-2]);
    printf("%lld\n", res);
    return 0;
}
