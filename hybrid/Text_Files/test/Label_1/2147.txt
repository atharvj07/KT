#include <bits/stdc++.h>
using namespace std;
#define int long long int
typedef pair<int, int> pi;
#define endl '\n'
int32_t main() {
  int n, d, k;
  cin >> n >> d >> k;
  if (k > n - 1 || d > 2 * (n - 1) || (n - 1) % 2 == d % 2) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  vector<int> adj[n + 1];
  int node = 3;
  for (int i = 2; i <= n; i++) {
    if (i != node) {
      adj[i].push_back(1);
      adj[1].push_back(i);
    } else {
      for (int j = 1; j <= k - 1; ++j) {
        if (node == n) {
          node = 1;
        }
        adj[node].push_back(i);
        adj[i].push_back(node);
        node++;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    for (int x : adj[i]) {
      if (x > i) {
        cout << i << " " << x << endl;
      }
    }
  }
}
