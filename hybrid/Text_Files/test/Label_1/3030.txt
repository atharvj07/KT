#include <bits/stdc++.h>
#define int long long
#define mod 1000000007
using namespace std;
int c[2005];
int powx[61];
int f[61][2005][7];
int32_t main() {
    powx[0] = 1;
    for (int i = 1; i <= 60; i++) { powx[i] = powx[i - 1] * 2; }
    int k, n;
    cin >> k >> n;
    for (int i = 1; i <= n; i++) {
        int x;
        string s;
        cin >> x >> s;
        if (s == "white")
            c[x] = 1;
        else if (s == "yellow")
            c[x] = 2;
        else if (s == "green")
            c[x] = 3;
        else if (s == "orange")
            c[x] = 4;
        else if (s == "red")
            c[x] = 5;
        else if (s == "blue")
            c[x] = 6;
    }
    f[0][1][0] = 1;
    for (int i = 1; i <= k; i++) {
        for (int j = 1; j <= powx[i] - 1; j++) {
            for (int col = 0; col <= 6; col++) {
                if (col != c[j] && col != c[2 * j] && col != c[2 * j + 1])
                    f[i][j][col] = (f[i - 1][2 * j][col] + f[i - 1][2 * j + 1][col]) % mod;
                if (!c[j]) {
                    if (col != c[2 * j] && col != c[2 * j + 1])
                        f[i][j][col] = (f[i][j][col] + f[i - 1][2 * j][col] * 5 % mod + f[i - 1][2 * j + 1][col] * 5 % mod) % mod;
                }
                if (!c[2 * j]) {
                    if (col != c[j] && col != c[2 * j + 1])
                        f[i][j][col] = (f[i][j][col] + f[i - 1][j][col] * 5 % mod + f[i - 1][2 * j + 1][col] * 5 % mod) % mod;
                }
                if (!c[2 * j + 1]) {
                    if (col != c[j] && col != c[2 * j])
                        f[i][j][col] = (f[i][j][col] + f[i - 1][j][col] * 5 % mod + f[i - 1][2 * j][col] * 5 % mod) % mod;
                }
            }
        }
    }
    cout << f[k][1][0] << endl;
    return 0;
}
