import sys

n, k = map(int, sys.stdin.readline().split())
temp = list(map(int, sys.stdin.readline().split()))
sumt = [0] * n
sumt[0] = temp[0]
for i in range(1, n):
  sumt[i] = sumt[i - 1] + temp[i]
maxi = 0
for i in range(n - k + 1):
  if i == 0:
    avg = sumt[i + k - 1] / k
  else:
    avg = (sumt[i + k - 1] - sumt[i - 1]) / k
  maxi = max(maxi, avg)
print(maxi)
