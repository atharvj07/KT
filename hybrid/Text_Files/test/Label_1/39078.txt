n, m = map(int, input().split())
blanket = [input() for _ in range(n)]

def count_flags(n, m, blanket):
    stripes = [[0] * m for _ in range(n)]
    
    # Count consecutive rows with the same color
    for i in range(n):
        for j in range(m):
            if i == 0:
                stripes[i][j] = 1
            else:
                if blanket[i][j] == blanket[i-1][j]:
                    stripes[i][j] = stripes[i-1][j] + 1
                else:
                    stripes[i][j] = 1
    
    flags = 0
    # Iterate through each row that can be the start of the middle stripe
    for i in range(1, n-1):
        cnt = 0
        for j in range(m):
            if i + stripes[i][j] < n and stripes[i][j] == stripes[i-1][j] and stripes[i][j] == stripes[i+stripes[i][j]][j]:
                cnt += 1
                flags += cnt
            else:
                cnt = 0
                
    return flags

print(count_flags(n, m, blanket))
