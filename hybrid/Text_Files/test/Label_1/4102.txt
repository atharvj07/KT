#include <iostream>
#include <string>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<std::pair<std::string, std::pair<int, int>>> arr(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> arr[i].first >> arr[i].second.first >> arr[i].second.second;
  }
  std::cout << (n > 0 && arr[0].second.first >= 2400 && arr[0].second.second > arr[0].second.first ? "YES"
                                                                                          : "NO")
            << std::endl;
  return 0;
}
