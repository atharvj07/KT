n, m = map(int, input().split())
a = list(map(int, input().split()))

parent = list(range(n))
rank = [0] * n

def find(x):
    if parent[x] != x:
        parent[x] = find(parent[x])
    return parent[x]

def union(x, y):
    rootX = find(x)
    rootY = find(y)
    if rootX != rootY:
        if rank[rootX] < rank[rootY]:
            parent[rootX] = rootY
        elif rank[rootX] > rank[rootY]:
            parent[rootY] = rootX
        else:
            parent[rootY] = rootX
            rank[rootX] += 1

for _ in range(m):
    x, y = map(int, input().split())
    union(x-1, y-1)

group_min = [float('inf')] * n
for i in range(n):
    root = find(i)
    group_min[root] = min(group_min[root], a[i])

sum_min = sum(group_min[i] for i in range(n) if find(i) == i)
result = sum_min * 2 / (n - 1)

print(f'{result:.6f}')
