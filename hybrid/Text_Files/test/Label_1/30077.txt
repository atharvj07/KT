import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] arr = sc.nextLine().split("");
        int n = Integer.parseInt(arr[0]) + Integer.parseInt(arr[1]) + Integer.parseInt(arr[2]);
        int m = Integer.parseInt(arr[3]) + Integer.parseInt(arr[4]) + Integer.parseInt(arr[5]);
        System.out.println(Math.abs(n - m));
    }
}
