import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        if (n == 1) {
            if (a[0] % 2 == 0) {
                System.out.println("BitLGM");
            } else {
                System.out.println("BitAryo");
            }
        } else {
            if (a[0] % 2 == 0) {
                System.out.println("BitLGM");
            } else {
                int min = a[0];
                for (int i = 1; i < n; i++) {
                    if (a[i] < min) {
                        min = a[i];
                    }
                }
                if (min % 2 == 1) {
                    System.out.println("BitLGM");
                } else {
                    System.out.println("BitAryo");
                }
            }
        }
    }
}
