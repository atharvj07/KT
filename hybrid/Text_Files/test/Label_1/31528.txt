import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double xa = sc.nextDouble(), ya = sc.nextDouble(), xb = sc.nextDouble(), yb = sc.nextDouble();
        int n = sc.nextInt();
        double[] x = new double[n], y = new double[n], a = new double[n], w = new double[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextDouble();
            y[i] = sc.nextDouble();
            a[i] = sc.nextDouble();
            w[i] = sc.nextDouble();
        }
        int k = sc.nextInt();
        double ub = 1e9, lb = 0;
        for (int t = 0; t < 100; t++) {
            double mid = (ub + lb) / 2;
            if (check(xa, ya, xb, yb, x, y, a, w, mid, k)) ub = mid;
            else lb = mid;
        }
        System.out.printf("%.4f", ub);
    }

    private static boolean check(double xa, double ya, double xb, double yb, double[] x, double[] y, double[] a, double[] w, double mid, int k) {
        double t = ((xb - xa) * (xb - xa) + (yb - ya) * (yb - ya)) / (mid * mid);
        for (int i = 0; i < x.length; i++) {
            double dx = xa + (xb - xa) * t - x[i], dy = ya + (yb - ya) * t - y[i];
            double angle = Math.atan2(dy, dx);
            if (angle < 0) angle += 2 * Math.PI;
            if (a[i] <= angle && angle <= a[i] + w[i] * t) k--;
        }
        return k >= 0;
    }
}
