#include <bits/stdc++.h>
using namespace std;
const int N = 50005;
long long x[N], a[N];
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    cin >> x[n];
    for (int i = n - 1; i >= 0; i--) x[i] += x[i + 1];
    long long ans = 0, mx = 0;
    for (int i = 0; i < n; i++) {
      if (i > 0) x[i] -= a[i - 1];
      if (a[i] > 0)
        mx = max(mx, a[i]);
      else
        mx = max(mx, a[i] + x[i + 1]);
      if (mx >= i * x[i])
        ans++;
      else
        break;
    }
    cout << ans << endl;
  }
}
