import re
from collections import defaultdict

for _ in range(int(input())):
    n = int(input())
    users = input().split()
    m = int(input())
    user_messages = defaultdict(list)
    for _ in range(m):
        a, b = input().split(':')
        if a != '?':
            user_messages[a].append(b)

    is_possible = True
    for user, messages in user_messages.items():
        for message in messages:
            for other_user in users:
                if user != other_user and re.search(other_user, message):
                    is_possible = False
                    break
        if not is_possible:
            break

    if is_possible:
        for user, messages in user_messages.items():
            for message in messages:
                print(user + ':' + message)
    else:
        print("Impossible")
