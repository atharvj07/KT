#include <bits/stdc++.h>
 
using namespace std;
 
const int MAXN = 2e5;
 
int main() {
  int n, s;
  cin >> n >> s;
  vector<int> v(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = 0; i < (n + 1) / 2; i++) {
    ans += abs(v[i] - s);
  }
  cout << ans << endl;
  return 0;
}
