import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long ans = 0;
        for (long i = 1;; i++) {
            ans = ans + ((7*i)%k);
            if ((7*i)%k == 0){
                System.out.println(i);
                break;
            }
        }
    }
}
