import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] d = new int[n - 1];
        for (int i = 0; i < n - 1; i++) d[i] = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int years = 0;
        for (int i = a - 1; i < b - 1; i++) years += d[i];
        System.out.println(years);
    }
}
