#include <bits/stdc++.h>
using namespace std;
int n, s, k, r[55], pos[3], cur = -1, ans = 1e9;
char c[55];
int main() {
    cin >> n >> s >> k;
    for (int i = 1; i <= n; i++) {
        cin >> r[i];
    }
    cin >> c;
    for (int i = 1; i <= n; i++) {
        if (!pos[c[i] - 'R'])
            pos[c[i] - 'R'] = i;
    }
    for (int i = 1; i <= n; i++) {
        if (c[i] != c[cur])
            ans += r[cur] < r[i];
        (cur != -1 && c[i] != c[cur]) ? ans++ : 0;
        cur = i;
        k -= min(k, r[i]);
        if (k == 0) {
            cout << ans << endl;
            return 0;
        }
    }
    cout << -1 << endl;
    return 0;
}
