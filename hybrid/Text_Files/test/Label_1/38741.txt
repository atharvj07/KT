from collections import deque

n, k = map(int, input().split())
a = deque(range(n,0,-1))
b = deque()
aa = list(map(int, input().split()))
for i in aa:
    while i != a[0]:
        b.append(a.popleft())
    a.popleft()
    b.append(i)
if len(a) > len(b): print(-1)
else:
    b.extend(a)
    ans = []
    for i in aa:
        b.remove(i)
        ans.append(i)
        ans.append(b.popleft())
    print(*ans)
