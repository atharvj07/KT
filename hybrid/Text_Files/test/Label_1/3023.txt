#include <iostream>
#include <vector>

using namespace std;

bool solve(const vector<int>& a, vector<int>& res) {
  int l = 0, r = a.size() - 1;
  while (l < a.size() && r >= 0) {
    if (res[l]) {
      while (l < a.size() && a[l] >= a[l + 1]) l++;
      if (l == a.size()) l--;
    }
    if (!res[r]) {
      while (r >= 0 && a[r] <= a[r - 1]) r--;
      if (r == -1) r++;
    }
    if (l >= r) break;
    if (l > 0 && r < a.size() - 1 && a[l] > a[r]) return false;
    if (l == 0 || res[l]) l++;
    if (r == a.size() - 1 || !res[r]) r--;
  }
  return true;
}

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> res(n, 1);
  if (!solve(a, res)) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    for (int i = 0; i < n; i++) cout << res[i] << ' ';
    cout << endl;
  }
  return 0;
}
