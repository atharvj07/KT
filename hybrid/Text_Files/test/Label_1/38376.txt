n = int(input())
p = list(map(int, input().split()))

hardness = [1]  # Initial hardness when all coins are out of circulation.
active_coins = set()

for pos in p:
    active_coins.add(pos)
    max_pos = max(active_coins)
    hardness.append(max(1, max_pos - len(active_coins) + 1))

print(*hardness)
