#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  int n, l, r;
  cin >> n >> l >> r;
  unordered_map<int, int> m;
  for (int i = 1; i <= n; i++) {
    int x;
    cin >> x;
    m[x] = i;
  }
  bool ok = true;
  for (int i = l; i <= r; i++) {
    int x;
    cin >> x;
    if (m[x] < l || m[x] > r) { ok = false; }
  }
  cout << (ok ? "TRUTH" : "LIE") << endl;
  return 0;
}
