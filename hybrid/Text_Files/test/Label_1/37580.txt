from bisect import bisect_right

for _ in range(int(input())):
    N, K = map(int, input().split())
    L1, R1 = map(int, input().split())
    L2, R2 = map(int, input().split())
    lst1 = [L1 + R1] * N
    lst2 = [L2 + R2] * N
    diff = K
    res = 0
    lst1.sort()
    for l2 in range(min(L1, L2), max(R1, R2) + 1):
        if diff <= 0:
            break
        idx1 = bisect_right(lst1, l2)
        idx2 = N - idx1
        if l2 % 2 == 0:
            l = min(diff // idx1, R1 - L1 + 1)
            lst1[:idx1] = [l2 + i * 2 for i in range(l)]
            diff -= l * idx1
            res += l
        else:
            r = min(diff // idx2, R2 - L2 + 1)
            lst2[idx2:] = [l2 + i * 2 for i in range(r)]
            diff -= r * idx2
            res += r
    print(res)
