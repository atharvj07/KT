import java.io.*;

public class Main {
    static int[] p, q, d;
    static long sum;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s;
        int n, m, u, v;
        while (true) {
            s = br.readLine().split(" ");
            n = Integer.parseInt(s[0]);
            m = Integer.parseInt(s[1]);
            if (n == 0) break;
            init(n);
            for (int i = 0; i < n - 1; i++) {
                u = Integer.parseInt(br.readLine());
                p[u] = i + 2;
            }
            for (int i = 0; i < m; i++) {
                s = br.readLine().split(" ");
                v = Integer.parseInt(s[1]);
                if (s[0].equals("M")) q[v] = 1;
                else sum += d[v] = query(v);
            }
            System.out.println(sum);
        }
    }

    public static void init(int n) {
        p = new int[n + 1];
        q = new int[n + 1];
        d = new int[n + 1];
        for (int i = 1; i <= n; i++) d[i] = 1;
    }

    public static int query(int v) {
        if (q[v] > 0) return 0;
        if (p[v] == 0) return v;
        else return d[v] += query(p[v]);
    }
}
