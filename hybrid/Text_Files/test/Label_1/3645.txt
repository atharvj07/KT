#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;

    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    int ans = 0;
    int i = 0, j = n - 1;
    while (i <= j) {
        if (a[i] <= k) ++i, ++ans;
        else if (a[j] <= k) --j, ++ans;
        else break;
    }

    cout << ans << '\n';
    return 0;
}
