n = int(input())
p = list(map(int, input().split()))
ans = [-1 for i in range(n+1)]
vis = [False for i in range(n+1)]
for i in range(1, n+1):
    if vis[i]: continue
    j = i
    while not vis[j]:
        vis[j] = True
        j = p[j-1]
    ans[i] = j
for i in range(1, n+1): print(ans[i], end = ' ')
