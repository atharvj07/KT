for _ in range(int(input())):
    n = int(input())
    b = list(map(int, input().split()))
    # Iterate over all towns finding the GCD of their durations.
    g = b[0]
    for i in range(1, n):
        g = gcd(g, b[i])
    # Use the GCD to find the original durations.
    ans = []
    for i in range(n):
        ans.append(b[i] // g)
    # Print the answer or "NO" if it's not possible.
    if len(set(ans)) == n:
        print("YES")
        print(*ans)
    else:
        print("NO")
