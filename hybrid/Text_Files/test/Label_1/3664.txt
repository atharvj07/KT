#include<iostream>
using namespace std;
int main(){
    string p;
    int up=0,low=0,dig=0,sp=0;
    cin>>p;
    for(int i=0;i<p.length();i++){
        if(isupper(p[i]))
            up=1;
        if(islower(p[i]))
            low=1;
        if(isdigit(p[i]))
            dig=1;
        if(p[i]=='.'|p[i]=='!'|p[i]=='?'|p[i]==','|p[i]=='_')
            sp=1;
    }
    if(up==0|low==0|dig==0|sp==0|p.length()<5)
        cout<<"Too weak";
    else cout<<"Correct";
    return 0;
}
