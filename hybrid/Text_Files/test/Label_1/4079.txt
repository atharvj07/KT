#include <bits/stdc++.h>

using namespace std;

int main() {
  long long n, x1 = 0, x2 = 0, mx1 = LLONG_MIN, mx2 = LLONG_MIN, last1 = LLONG_MIN,
               last2 = LLONG_MIN;
  cin >> n;
  while (n--) {
    int a;
    cin >> a;
    if (a > 0) {
      x1 += a;
      if (a > mx1) mx1 = a;
      last1 = a;
    } else {
      x2 -= a;
      if (-a > mx2) mx2 = -a;
      last2 = -a;
    }
  }
  if (x1 > x2)
    cout << "first";
  else if (x1 == x2) {
    if (mx1 > mx2)
      cout << "first";
    else if (mx1 == mx2) {
      if (last1 > last2)
        cout << "first";
      else
        cout << "second";
    } else
      cout << "second";
  } else
    cout << "second";
  return 0;
}
