import java.util.*;
import java.io.*;

public class JohnnyBlogs {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(System.out));

        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            int a = Integer.parseInt(line[0]);
            int b = Integer.parseInt(line[1]);
            graph.get(a).add(b);
            graph.get(b).add(a);
        }

        line = reader.readLine().split(" ");
        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = Integer.parseInt(line[i - 1]);
        }

        int[] order = new int[n + 1];
        boolean[] visited = new boolean[n + 1];
        if (!solve(1, graph, desiredTopics, order, visited)) {
            writer.write("-1");
        } else {
            for (int i = 1; i <= n; i++) {
                writer.write(order[i] + " ");
            }
        }
        writer.close();
    }

    static boolean solve(int blog, List<List<Integer>> graph, int[] desiredTopics, int[] order, boolean[] visited) {
        if (visited[blog]) {
            return false;
        }
        visited[blog] = true;

        Set<Integer> usedTopics = new HashSet<>();
        for (int neighbor : graph.get(blog)) {
            if (visited[neighbor]) {
                usedTopics.add(desiredTopics[neighbor]);
            }
        }

        int desiredTopic = desiredTopics[blog];
        while (usedTopics.contains(desiredTopic)) {
            desiredTopic++;
            if (desiredTopic > graph.size() - 1) {
                return false;
            }
        }

        order[blog] = desiredTopic;
        for (int neighbor : graph.get(blog)) {
            if (!solve(neighbor, graph, desiredTopics, order, visited)) {
                return false;
            }
        }
        return true;
    }
}
