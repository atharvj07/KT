import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] masks = new long[n];
        for (int i = 0; i < n; i++) {
            masks[i] = 0L;
            int k = sc.nextInt();
            for (int j = 0; j < k; j++) {
                int b = sc.nextInt();
                masks[i] |= (1L << b);
            }
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            long m = 0L;
            if (op > 0) {
                m = masks[sc.nextInt()];
            }
            switch (op) {
                case 0:
                    System.out.println(test(m, sc.nextInt()));
                    break;
                case 1:
                    set(m);
                    break;
                case 2:
                    clear(m);
                    break;
                case 3:
                    flip(m);
                    break;
                case 4:
                    System.out.println(all(m));
                    break;
                case 5:
                    System.out.println(any(m));
                    break;
                case 6:
                    System.out.println(none(m));
                    break;
                case 7:
                    System.out.println(count(m));
                    break;
                case 8:
                    System.out.println(val(m));
                    break;
            }
        }
    }

    private static int test(long m, int i) {
        return (m & (1L << i)) != 0 ? 1 : 0;
    }

    private static void set(long m) {
        state |= m;
    }

    private static void clear(long m) {
        state &= ~m;
    }

    private static void flip(long m) {
        state ^= m;
    }

    private static int all(long m) {
        return (state & m) == m ? 1 : 0;
    }

    private static int any(long m) {
        return (state & m) != 0 ? 1 : 0;
    }

    private static int none(long m) {
        return (state & m) == 0 ? 1 : 0;
    }

    private static int count(long m) {
        int cnt = 0;
        while (m != 0) {
            if ((m & 1) != 0) {
                cnt++;
            }
            m >>= 1;
        }
        return cnt;
    }

    private static long val(long m) {
        return state & m;
    }

    private static long state = 0L;
}
