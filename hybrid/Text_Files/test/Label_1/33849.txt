import java.io.*;
import java.util.*;

class Main {

    static int MOD = 1000000007;
    static int N;
    static int[] a, b, par;
    static ArrayList<Integer>[] adj;

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        N = Integer.parseInt(br.readLine());
        a = new int[N - 1];
        b = new int[N - 1];
        par = new int[N + 1];
        adj = new ArrayList[N + 1];
        for (int i = 0; i <= N; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < N - 1; i++) {
            String[] line = br.readLine().split(" ");
            a[i] = Integer.parseInt(line[0]);
            b[i] = Integer.parseInt(line[1]);
            adj[a[i]].add(b[i]);
            adj[b[i]].add(a[i]);
        }
        dfs(1, 0);
        long ans = (dfs2(1, -1, 0) + dfs2(1, -1, 1)) % MOD;
        System.out.println(ans);
    }

    static void dfs(int node, int p) {
        par[node] = p;
        for (int u : adj[node]) {
            if (u != p) dfs(u, node);
        }
    }

    static long dfs2(int node, int p, int dir) {
        long ans = 1;
        if (dir == 1) {
            for (int u : adj[node]) {
                if (u != p) ans = (ans * dfs2(u, node, 0)) % MOD;
            }
        } else {
            for (int u : adj[node]) {
                if (u != p) ans = (ans * dfs2(u, node, 1)) % MOD;
            }
        }
        return ans;
    }
}
