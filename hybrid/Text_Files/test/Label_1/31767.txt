import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] rows = new int[n];
        int[] cols = new int[n];
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                int num = sc.nextInt();
                rows[i]+=num;
                cols[j]+=num;
            }
        }
        int count=0;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                if(cols[j]>rows[i]){
                    count++;
                }
            }
        }
        System.out.println(count);
    }
}
