from sys import stdin
input=stdin.readline

n,m=map(int,input().split())
s=input().strip()
l=[-1]*n; r=[-1]*n
st=[0]
for i in range(n):
    if i>0 and s[i]==s[i-1]:
        l[i]=l[i-1]
    if s[i]==s[n-i-1]:
        r[n-i-1]=r[n-i-2]
    if r[n-i-1]==-1 or l[i]==-1:
        st.append(i)
for i in reversed(range(n-1)):
    l[i]=max(l[i],l[i+1])
    r[i]=max(r[i],r[i+1])
for i in range(1,len(st)):
    lstl=l[st[i-1]-1]; lstr=r[st[i-1]+1]
    if lstl!=-1 and lstr!=-1 and lstl<=lstr:
        l[st[i]-1]=max(l[st[i]-1],lstr)
for i in range(m):
    a,b=map(int,input().split()); a-=1; b-=1
    if l[a]==-1 or l[a]>b: print(0)
    else: print(1)
