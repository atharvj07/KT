import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] c = new long[n], s = new long[n], f = new long[n];
        for (int i = 0; i < n - 1; i++) {
            c[i] = sc.nextInt();
            s[i] = sc.nextInt();
            f[i] = sc.nextInt();
        }
        long[] ans = new long[n];
        ans[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            ans[i] = ans[i + 1] + c[i];
            s[i] = (s[i] + ans[i]) % f[i];
            ans[i] += s[i];
        }
        for (long x : ans) {
            System.out.println(x);
        }
    }
}
