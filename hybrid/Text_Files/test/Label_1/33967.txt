import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[] l = new int[m];
        for (int i = 0; i < m; i++) l[i] = in.nextInt();

        // find the smallest remaining unpainted range
        int[] r = new int[n + 1];
        Arrays.fill(r, n + 1);
        for (int i = 0; i < m; i++) {
            for (int j = l[i]; j <= n; j++) {
                r[j] = Math.min(r[j], i);
            }
        }

        int ans = 0;
        int[] p = new int[m];
        for (int i = 1; i <= n; i++) {
            if (r[i] == n + 1) {
                ans = -1;
                break;
            }
            if (r[i] == ans) {
                ans++;
            }
            p[r[i]] = i;
        }

        System.out.println(ans == -1 ? "-1" : Arrays.toString(p).substring(1, Arrays.toString(p).length() - 1));
    }
}
