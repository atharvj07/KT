from sys import stdin as si

n,m=map(int,si.readline().split())
g=[list(si.readline().strip())for i in range(n)]
ans=0
for i in range(n):
    for j in range(m):
        if g[i][j]=='.':continue
        chk=[[0]*m for i in range(n)]
        for k in range(i,-1,-1):
            if g[k][j]!='*':chk[k][j]=1
            else:break
        for k in range(i,n):
            if g[k][j]!='*':chk[k][j]=1
            else:break
        for k in range(j,-1,-1):
            if g[i][k]!='*':chk[i][k]=1
            else:break
        for k in range(j,m):
            if g[i][k]!='*':chk[i][k]=1
            else:break
        for x in range(n):
            for y in range(m):
                if chk[x][y]==0:
                    ans+=1
                    print(x+1,y+1,min(i-x,n-1-x,j-y,m-1-y))
print(ans)
