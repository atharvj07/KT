from collections import Counter
from math import sqrt

def is_perfect_square(n):
    sq = int(sqrt(n))
    return sq * sq == n

n = int(input())
a = list(map(int, input().split()))
cnt = Counter(a)

ans = []
for i in range(1, n + 1):
    cur = []
    for j in range(n):
        if cnt[a[j]] >= i:
            cur.append(j)
    if len(cur) >= i:
        ans.append(len(cur) - i + 1)
    else:
        ans.append(0)

print(*ans)
