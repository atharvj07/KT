for _ in range(int(input())):
    n = int(input())
    s = input()
    cnt = {}
    for c in s:
        if c in cnt:
            cnt[c] ^= 1
        else:
            cnt[c] = 1
    odd = 0
    for c in cnt:
        if cnt[c]:
            odd += 1
    if odd <= 1:
        print(0)
    elif odd == 3:
        print(2)
    else:
        print(-1)
