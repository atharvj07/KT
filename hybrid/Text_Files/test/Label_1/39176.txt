n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

for i in range(n):
    if a[i] != b[i]:
        missing = set(range(1, n + 1)) - set(a) - set(b)
        if len(missing) == 1:
            a[i] = missing.pop()
        else:
            a[i] = b[i] if a.count(b[i]) < a.count(a[i]) else a[i]
        break

print(*a)
