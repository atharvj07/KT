import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int q = sc.nextInt();
        while (q-- > 0) {
            int b = sc.nextInt(), e = sc.nextInt(), k = sc.nextInt();
            int cnt = 0;
            for (int j = b; j < e; j++) {
                if (a[j] == k) cnt++;
            }
            System.out.println(cnt);
        }
    }
}
