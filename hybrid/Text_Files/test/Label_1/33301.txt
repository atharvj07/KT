import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long product = 1;
    for (int i = 0; i < n; i++) {
      long a = sc.nextLong();
      if (a == 0) {
        product = 0;
        break;
      }
      if (!isBeautiful(a)) {
        continue;
      }
      product *= a;
    }
    System.out.println(product);
  }

  private static boolean isBeautiful(long a) {
    int count = 0;
    while (a > 0) {
      if (a % 10 == 1) {
        count++;
      }
      a /= 10;
    }
    return count <= 1;
  }
}
