t = int(input())
for _ in range(t):
    n, a, b = map(int, input().split())
    if a > n//2 or b <= n//2:
        print(-1)
        continue
    left = list(range(1, n+1))
    left.remove(a)
    left.remove(b)
    left_half = [a] + sorted(left[:n//2-1])
    right_half = sorted(left[n//2-1:], reverse=True) + [b]
    print(*(left_half + right_half))
