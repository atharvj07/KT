from collections import Counter
from math import prod
from itertools import combinations
MOD = 10**9 + 7

def prime_factors(n):
    i = 2
    factors = []
    while i * i <= n:
        if n % i:
            i += 1
        else:
            n //= i
            factors.append(i)
    if n > 1:
        factors.append(n)
    return factors

def is_square(factors):
    return all(count % 2 == 0 for count in Counter(factors).values())

def solve(n, arr):
    prime_counts = [Counter(prime_factors(x)) for x in arr]
    dp = [0] * (1 << len(prime_counts))
    dp[0] = 1

    for x in arr:
        factors = prime_factors(x)
        if is_square(factors):
            for i in range(len(dp)):
                dp[i] = (dp[i] * 2) % MOD
    
    return (dp[0] - 1) % MOD

n = int(input().strip())
arr = list(map(int, input().strip().split()))
print(solve(n, arr))
