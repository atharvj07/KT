#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int main() {
	long long int t,w,b,pw=1,pb=1,count=0;
	cin>>t>>w>>b;
	while(pw<=t) {
		while(pb<=t) {
			if(pw==pb) {
				count++;
			}
			pb=pb+b;
		}
		pb=1;
		pw=pw+w;
	}
	long long int ans=count;
	long long int total=t/w+1;
	ans=(ans*total)/t;
	cout<<ans<<"/"<<total<<endl;
	return 0;
}
