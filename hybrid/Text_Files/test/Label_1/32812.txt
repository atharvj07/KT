import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt();
        int res1 = x, res2 = 0;
        while(n-->0) {
            char op = sc.next().charAt(0);
            int d = sc.nextInt();
            if(op=='+') res1 += d;
            else if(res1 - d >= 0) res1 -= d;
            else res2++;
        }
        System.out.println(res1 + " " + res2);
    }
}
