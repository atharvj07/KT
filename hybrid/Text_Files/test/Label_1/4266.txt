#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int &i : a) cin >> i;
  vector<int> c(k + 2);
  for (int i = 0; i < n; i++) c[a[i]]++;
  int mn = n, j = 0;
  for (int i = 1; i <= k; i++) {
    if (c[i] < mn) {
      mn = c[i];
      j = i;
    }
  }
  cout << j;
  return 0;
}
