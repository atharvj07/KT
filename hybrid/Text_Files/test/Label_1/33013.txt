import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();

        while (t-- > 0) {
            long x = input.nextLong();

            if (x % 2 == 1) {
                System.out.println("NO");
                continue;
            }

            x /= 2;

            long b = (long) Math.cbrt(x);
            long a = x - b * b * b;

            if (a * a * a + b * b * b == x) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
