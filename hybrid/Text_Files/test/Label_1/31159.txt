import java.util.*;
import java.lang.*;
import java.io.*;
import java.text.DecimalFormat;

class Main {
    static double eps = 1e-9;
    static boolean eq(long a, long b) {
        return Math.abs(a - b) < eps;
    }
    public static void main(String[] args) throws java.lang.Exception {
        Scanner in = new Scanner(System.in);
        int n, H;
        n = in.nextInt();
        H = in.nextInt();
        int[][] a = new int[n][2];
        for (int i = 0; i < n; i++) {
            a[i][0] = in.nextInt();
            a[i][1] = in.nextInt();
        }
        in.close();
        double ans = 0;
        int l = -1, r = -1;
        for (int i = 0; i < n - 1; i++) {
            int dy = a[i + 1][1] - a[i][1];
            int dx = a[i + 1][0] - a[i][0];
            int c = dy * a[i][0] - dx * a[i][1];
            int ll = -1, rr = -1;
            if (dx > 0) {
                if ((ll = a[i][0] - dy * H / dx) >= 0) {
                    ll = (int) ll;
                    if (ll >= a[i][0] && ll <= a[i + 1][0]) {
                        if (l == -1 || ll > l) {
                            l = ll;
                            r = i;
                        }
                    }
                }
                if ((rr = a[i + 1][0] - dy * H / dx) >= 0) {
                    rr = (int) rr;
                    if (rr >= a[i][0] && rr <= a[i + 1][0]) {
                        if (r == -1 || rr < r) {
                            l = rr;
                            r = i + 1;
                        }
                    }
                }
            } else if (dx < 0) {
                if ((ll = a[i][0] + dy * H / dx) >= 0) {
                    ll = (int) ll;
                    if (ll >= a[i][0] && ll <= a[i + 1][0]) {
                        if (l == -1 || ll > l) {
                            l = ll;
                            r = i;
                        }
                    }
                }
                if ((rr = a[i + 1][0] + dy * H / dx) >= 0) {
                    rr = (int) rr;
                    if (rr >= a[i][0] && rr <= a[i + 1][0]) {
                        if (r == -1 || rr < r) {
                            l = rr;
                            r = i + 1;
                        }
                    }
                }
            }
        }
        if (l == -1 || l > a[n - 1][0]) {
            DecimalFormat df = new DecimalFormat("0.0000000000");
            System.out.print(df.format(0.0000000000));
            return;
        }
        while (l > a[r][0]) {
            long ny = a[r + 1][1] + (long) ((a[r + 1][0] - l) * 1.0 * (a[r + 1][1] - a[r][1])) / (a[r + 1][0] - a[r][0]);
            if (ny > a[r][1] + H) {
                ans += a[r + 1][0] - l;
                l = a[r + 1][0];
                r++;
            } else {
                break;
            }
        }
        if (r == n - 1) {
            DecimalFormat df = new DecimalFormat("0.0000000000");
            System.out.print(df.format(ans));
            return;
        }
        int b = r;
        while (l < a[n - 1][0]) {
            if (eq(l, a[b][0])) {
                ans += a[b + 1][0] - a[b][0];
                b++;
            }
            if (b == n - 1) {
                break;
            }
            long ny = a[b + 1][1] + (long) ((a[b + 1][0] - l) * 1.0 * (a[b + 1][1] - a[b][1])) / (a[b + 1][0] - a[b][0]);
            if (ny > a[b][1] + H) {
                ans += a[b + 1][0] - l;
                l = a[b + 1][0];
                if (b + 1 < n - 1) {
                    b++;
                }
            } else {
                break;
            }
        }
        DecimalFormat df = new DecimalFormat("0.0000000000");
        System.out.print(df.format(ans));
    }
}
