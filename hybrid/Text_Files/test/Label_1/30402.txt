import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

class Main {

  static Integer[] swap(Integer[] arr, int i, int j) {
    Integer tmp = arr[i];
    arr[i] = arr[j];
    arr[j] = tmp;
    return arr;
  }

  static Integer[] nextPermutation(Integer[] arr) {
    int n = arr.length;
    int idx = n - 2;

    while (idx >= 0 && arr[idx] >= arr[idx + 1]) idx--;

    if (idx >= 0) {
      int j = n - 1;
      while (arr[idx] >= arr[j]) j--;
      arr = swap(arr, idx, j);
    }

    idx++;
    int end = n - 1;
    while (idx < end) {
      arr = swap(arr, idx, end);
      idx++;
      end--;
    }

    return arr;
  }

  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());

    int n = Integer.parseInt(st.nextToken());
    int m = Integer.parseInt(st.nextToken());

    int[][] a = new int[n][m];
    int[][] pos = new int[n + m][2];

    for (int i = 0; i < n; i++) {
      st = new StringTokenizer(br.readLine());
      for (int j = 0; j < m; j++) {
        a[i][j] = Integer.parseInt(st.nextToken());
        pos[a[i][j]][0] = i;
        pos[a[i][j]][1] = j;
      }
    }
    br.close();

    boolean isPossible = true;
    for (int[] v : a) {
      Integer[] x = new Integer[v.length];
      for (int i = 0; i < v.length; i++) x[i] = i;

      Integer[] w = nextPermutation(x);
      while (!w[0].equals(0)) {
        boolean f = true;
        for (int i = 0; i < v.length - 1; i++) {
          int i1 = pos[v[w[i]]][0];
          int i2 = pos[v[w[i + 1]]][0];
          f &= i1 <= i2;
        }
        if (!f) {
          isPossible = false;
          break;
        }
        w = nextPermutation(w);
      }
      if (!isPossible) break;
    }

    for (int[] v : a.clone()) {
      Integer[] x = new Integer[v.length];
      for (int i = 0; i < v.length; i++) x[i] = i;

      Integer[] w = nextPermutation(x);
      while (!w[0].equals(0)) {
        boolean f = true;
        for (int i = 0; i < v.length - 1; i++) {
          int j1 = pos[v[w[i]]][1];
          int j2 = pos[v[w[i + 1]]][1];
          f &= j1 <= j2;
        }
        if (!f) {
          isPossible = false;
          break;
        }
        w = nextPermutation(w);
      }
      if (!isPossible) break;
    }

    if (!isPossible) System.out.println(-1);
    else {
      Integer[] perm = new Integer[n * m];
      for (int i = 0; i < n * m; i++) perm[i] = i;

      for (int[] v : a) {
        Integer[] x = new Integer[v.length];
        for (int i = 0; i < v.length; i++) x[i] = i;

        Integer[] w = nextPermutation(x);
        while (!w[0].equals(0)) {
          boolean f = true;
          for (int i = 0; i < v.length - 1; i++) {
            int i1 = pos[v[w[i]]][0];
            int i2 = pos[v[w[i + 1]]][0];
            f &= i1 <= i2;
          }
          if (f) {
            for (int i = 0; i < v.length; i++) {
              pos[v[i]][0] = pos[perm[w[i]]][0];
              pos[v[i]][1] = pos[perm[w[i]]][1];
            }
          }
          w = nextPermutation(w);
        }
        for (int i = 0; i < v.length; i++) perm[pos[v[i]][0] * m + pos[v[i]][1]] = i;
      }

      for (int[] v : a) {
        Integer[] x = new Integer[v.length];
        for (int i = 0; i < v.length; i++) x[i] = i;

        Integer[] w = nextPermutation(x);
        while (!w[0].equals(0)) {
          boolean f = true;
          for (int i = 0; i < v.length - 1; i++) {
            int j1 = pos[v[w[i]]][1];
            int j2 = pos[v[w[i + 1]]][1];
            f &= j1 <= j2;
          }
          if (f) {
            for (int i = 0; i < v.length; i++) {
              pos[v[i]][0] = pos[perm[pos[v[i]][0] * m + w[i]]][0];
              pos[v[i]][1] = pos[perm[pos[v[i]][0] * m + w[i]]][1];
            }
          }
          w = nextPermutation(w);
        }
      }

      for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
          System.out.print(perm[i * m + j] + 1 + " ");
        }
        System.out.println();
      }
    }
  }
}
