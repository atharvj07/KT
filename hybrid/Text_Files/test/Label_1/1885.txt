#include <bits/stdc++.h>

using namespace std;

int32_t main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int add = 0;
    for (int i = k; i < n; i++) {
        if (s[i] < s[i - k]) {
            add = 1;
            break;
        }
    }
    int m = n;
    if (add == 1) m++;
    cout << m << endl;
    if (add == 1) {
        int i = n - 1;
        while (i >= 0 && s[i] == '9') i--;
        while (i >= k) {
            s[i] = s[i - k];
            i--;
        }
        int j = i - 1;
        while (k != 0) {
            s[j] = '0' + (s[j] - '0' + 1) % 10;
            j--;
            k--;
        }
        while (s[0] == '0' && m > n) s.erase(s.begin()), m--;
    }
    cout << s << endl;
}
