#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<int> a(n);
  for (int& el : a) {
    std::cin >> el;
  }
  int ans = 0;
  int diff = -1, cnt = 0;
  for (int i = 0; i < n; ++i) {
    if (a[i] == -1) {
      ++cnt;
      continue;
    }
    if (diff == -1) {
      diff = a[i] - a[i + 1];
      ++cnt;
    } else if (diff != a[i] - a[i + 1]) {
      ans += cnt / 2 + (cnt & 1);
      cnt = 1;
      diff = a[i] - a[i + 1];
    } else {
      ++cnt;
    }
  }
  ans += cnt / 2 + (cnt & 1);
  std::cout << ans << '\n';
}
