#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  for (int i = 0; i < n; i++) {
    cin >> b[i];
  }
  vector<int> pos_a(n + 1), pos_b(n + 1);
  for (int i = 0; i < n; i++) {
    pos_a[a[i]] = i;
    pos_b[b[i]] = i;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    ans += (pos_a[i] == pos_b[i]);
  }
  cout << ans << endl;
  return 0;
}
