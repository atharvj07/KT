import java.util.Scanner;

public class DominoCarpet {
    private static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine(); // Consume the newline

        char[][] carpet = new char[n * 3][m * 3];
        for (int i = 0; i < n * 3; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m * 3; j++) {
                carpet[i][j] = line.charAt(j);
            }
        }

        long[][][] dp = new long[n][m][2]; // dp[i][j][0] = horizontal, dp[i][j][1] = vertical
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (i == 0 && j == 0) {
                    dp[i][j][0] = 1;
                    dp[i][j][1] = 1;
                } else if (i == 0) {
                    if (carpet[i * 3][j * 3 + 2] == '.') {
                        dp[i][j][0] = dp[i][j - 1][0];
                    }
                    if (carpet[i * 3][j * 3] != '.') {
                        dp[i][j][1] = dp[i][j - 1][1];
                    }
                } else if (j == 0) {
                    if (carpet[i * 3][j * 3 + 1] == '.') {
                        dp[i][j][1] = dp[i - 1][j][1];
                    }
                    if (carpet[i * 3 + 2][j * 3] != '.') {
                        dp[i][j][0] = dp[i - 1][j][0];
                    }
                } else {
                    if (carpet[i * 3][j * 3 + 1] == '.') {
                        dp[i][j][1] = (dp[i - 1][j][1] + dp[i][j - 1][1]) % MOD;
                    }
                    if (carpet[i * 3][j * 3 + 2] == '.') {
                        dp[i][j][0] = (dp[i - 1][j][0] + dp[i][j - 1][0]) % MOD;
                    }
                    if (carpet[i * 3 + 2][j * 3] != '.') {
                        dp[i][j][0] = (dp[i][j][0] + dp[i - 1][j][1]) % MOD;
                    }
                    if (carpet[i * 3][j * 3] != '.') {
                        dp[i][j][1] = (dp[i][j][1] + dp[i][j - 1][0]) % MOD;
                    }
                }
            }
        }

        System.out.println(dp[n - 1][m - 1][0] + dp[n - 1][m - 1][1]);
    }
}
