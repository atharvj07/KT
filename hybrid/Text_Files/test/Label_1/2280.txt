#include<bits/stdc++.h>
using namespace std;
const int N=105;
char s[N];
bool isok[3][N];
int main()
{
	int n,i,j,k,l,ans=0;
	cin>>n>>s;
	for(i=0;i<n;i++)
	{
		if(i)
		{
			for(j=0;j<3;j++) isok[j][i]=isok[j][i-1];
		}
		if(s[i]=='?'||s[i]=='C') isok[0][i]=1;
		if(s[i]=='?'||s[i]=='M') isok[1][i]=1;
		if(s[i]=='?'||s[i]=='Y') isok[2][i]=1;
		if(s[i]!='?') isok[0][i]=isok[1][i]=isok[2][i]=0;
	}
	for(i=0;i<3;i++)
	{
		ans|=isok[i][n-1];
	}
	cout<<(ans==3?"Yes":"No")<<endl;
	return 0;
}
