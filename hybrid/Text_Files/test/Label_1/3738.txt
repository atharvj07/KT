#include<bits/stdc++.h>
using namespace std;
int main() {
  long long n, t;
  cin >> n >> t;
  string a, b;
  cin >> a >> b;
  map<char, long long> freq;
  for(char c: a) freq[c]++;
  vector<pair<long long, char>> v;
  for(auto &p: freq) v.push_back({p.second, p.first});
  sort(v.rbegin(), v.rend());
  if(v.size() < t) {
    cout << "-1" << endl;
    return 0;
  }
  string ans = b;
  long long cnt = 0;
  for(auto &p: v) {
    if(cnt < t) {
      for(long long i = 0; i < p.first; i++) {
        ans[i] = p.second;
        cnt++;
      }
    }
    else break;
  }
  if(cnt == t) cout << ans << endl;
  else cout << "-1" << endl;
  return 0;
}
