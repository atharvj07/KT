import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] dp = new int[n + 1];
        int[] prev = new int[n + 1];
        Arrays.fill(dp, 1);
        int max = 1;
        int r = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (Math.abs(a[i] - a[j]) <= 1 && dp[j] + 1 > dp[i]) {
                    dp[i] = dp[j] + 1;
                    prev[i] = j;
                }
            }
            if (dp[i] > max) {
                max = dp[i];
                r = i;
            }
        }
        int[] res = new int[max];
        for (int i = max - 1; i >= 0; i--) {
            res[i] = a[r];
            r = prev[r];
        }
        System.out.println(max);
        for (int i = 0; i < max; i++) {
            System.out.print(res[i] + " ");
        }
    }
}
