#include <iostream>
#include <cstring>

using namespace std;

string min_typos(string s) {
    int n = s.length();
    string res = s;

    int l = 0, r = n - 1, has_tri = 0, has_quad = 0;
    while (l <= r) {
        bool flag = false;

        if (s[l] == s[l + 1] && s[l] == s[l + 2]) {
            flag = true;
            has_tri = 1;
        } else if (l < r && s[l] == s[l + 1] && s[l + 1] == s[l + 2] && s[l + 2] == s[l + 3]) {
            flag = true;
            has_quad = 1;
        }

        if (!flag) {
            res[l] = s[l], res[r] = s[r], ++l, --r;
        } else if (has_tri && !has_quad) {
            res[l + 2] = s[l + 2], ++l;
        } else {
            res[l + 1] = s[l + 1], res[l + 2] = s[l + 2], res[l + 3] = s[l + 3], l += 2, r -= 2;
        }
    }

    return res.substr(0, l);
}

int main() {
    string s;
    cin >> s;
    cout << min_typos(s);
    return 0;
}
