#include <iostream>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    int n, k;
    cin >> n >> k;

    string s;
    cin >> s;

    for (int i = 1; i < n;) {
        if (s[i] == '#') {
            i = max(i + 1, i + k);
            if (i > n) {
                cout << "NO" << '\n';
                return 0;
            }
        } else {
            i++;
        }
    }

    cout << "YES" << '\n';

    return 0;
}
