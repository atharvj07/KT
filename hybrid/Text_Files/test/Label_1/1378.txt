#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> a(n);
  for (auto &x : a) { std::cin >> x; }
  std::sort(a.begin(), a.end());
  int ans = 0;
  for (int l = 0; l < n; l++) {
    int r = std::lower_bound(a.begin() + l, a.end(), m) - a.begin() - 1;
    if (r < l || (r - l) % 2 == 0) continue;
    int med = a[(l + r) / 2];
    if (med == m) ans++;
  }
  std::cout << ans << std::endl;
  return 0;
}
