#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 1e5 + 5;
ll f[N], g[N];
int main() {
    ll n, k, m;
    cin >> n >> k >> m;
    vector<ll> a(n + 5);
    for (ll i = 1; i <= n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    for (ll i = n; i >= 1; i--) {
        if (i == n)
            f[i] = a[i];
        else
            f[i] = max(f[i + 1], a[i] + g[i + 1]);
        g[i] = g[i + 1] + max(0ll, min(a[i] - f[i + 1], k));
    }
    ll ans = 0;
    for (ll p = 0, i = 1; i <= n && p <= m; i++) {
        ans = max(ans, f[i] + min(m - p, g[i]));
        p += min(m - p, g[i]);
    }
    cout << ((long double)ans / (long double)n) << endl;
    return 0;
}
