#include<bits/stdc++.h>
using namespace std;
#define ll long long
#define FOR(i, a, b) for(int i = a; i <= b; i++)
bool canDivide(ll n, ll a[]){
    sort(a, a + n);
    ll total = 0, onebefore = 0;
    FOR(i, 0, n - 1){
        total += abs(a[i]);
        if (a[i] < 0) onebefore++;
    }
    if (onebefore == 0 || onebefore == n) return true;
    else if((total % n) == 0 && total / n >= a[onebefore]) return true;
    else return false;
}
int main(){
    ll t;
    cin >> t;
    while(t--){
        ll n;
        cin >> n;
        ll a[n];
        FOR(i, 0, n - 1){
            cin >> a[i];
        }
        ll s1, s2, sum = 0;
        s1 = s2 = n / 2;
        sort(a, a + n);
        FOR(i, 0, n / 2 - 1) sum += a[i];
        sum += (ll)ceil(a[n / 2 - 1] / 2.0);
        s1 = s1 - (ll)ceil(a[n / 2 - 1] / 2.0);
        FOR(i, n / 2, n - 1) sum += a[i];
        if (sum % s1 == 0){
            if (canDivide(n, a)) cout << sum / s1 << "\n";
            else cout << -1 << "\n";
        }
        else if (sum % s2 == 0){
            if (canDivide(n, a)) cout << sum / s2 << "\n";
            else cout << -1 << "\n";
        }
        else cout << -1 << "\n";
    }
    return 0;
}
