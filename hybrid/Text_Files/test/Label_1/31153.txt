import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String[] paths = new String[2];
        paths[0] = sc.nextLine();
        paths[1] = sc.nextLine();
        boolean[][] dp = new boolean[2][n];
        dp[0][0] = true;
        dp[1][0] = true;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < 2; j++) {
                char c = paths[j].charAt(i - 1);
                if (dp[j][i - 1] && (c == 'N' || c == 'W')) {
                    dp[j][i] = true;
                }
                if (dp[j][i - 1] && (c == 'E' || c == 'S')) {
                    dp[j ^ 1][i] = true;
                }
                if (dp[j ^ 1][i - 1] && (c == 'W' || c == 'S')) {
                    dp[j][i] = true;
                }
                if (dp[j ^ 1][i - 1] && (c == 'N' || c == 'E')) {
                    dp[j ^ 1][i] = true;
                }
            }
        }
        if (dp[0][n - 1] && dp[1][n - 1]) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
