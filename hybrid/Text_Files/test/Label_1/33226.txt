import java.util.Scanner;

public class StoneGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] piles = new int[n];
            for (int j = 0; j < n; j++) {
                piles[j] = scanner.nextInt();
            }

            System.out.println(determineWinner(piles) ? "T" : "HL");
        }
    }

    private static boolean determineWinner(int[] piles) {
        int totalStones = 0;
        for (int pile : piles) {
            totalStones += pile;
        }

        return totalStones % 2 == 1; // T wins if the total number of stones is odd
    }
}
