import java.util.*;

public class Main {
    static final long INF = 1L << 60;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] ans = new long[m];

        UF uf = new UF(n);
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            long w = sc.nextLong();
            pq.add(new Edge(a, b, w));
        }

        for (int i = 0; i < m; i++) {
            Edge e = pq.poll();
            int a = e.a;
            int b = e.b;
            long w = e.w;
            if (uf.find(a) != uf.find(b)) {
                uf.union(a, b);
                if (uf.deg[uf.find(a)] % 2 == 0 && uf.deg[uf.find(b)] % 2 == 0) {
                    ans[i] = -1;
                } else {
                    ans[i] = w;
                }
            } else {
                ans[i] = -1;
            }
        }

        for (long a : ans) {
            System.out.println(a);
        }
    }

    static class UF {
        int[] par;
        int[] deg;

        UF(int n) {
            par = new int[n];
            deg = new int[n];
            for (int i = 0; i < n; i++) {
                par[i] = i;
                deg[i] = 1;
            }
        }

        int find(int x) {
            if (par[x] == x) {
                return x;
            } else {
                return par[x] = find(par[x]);
            }
        }

        void union(int a, int b) {
            a = find(a);
            b = find(b);
            if (a != b) {
                par[b] = a;
                deg[a] += deg[b];
                deg[b] = deg[a];
            }
        }
    }

    static class Edge implements Comparable<Edge> {
        int a, b;
        long w;

        Edge(int a, int b, long w) {
            this.a = a;
            this.b = b;
            this.w = w;
        }

        @Override
        public int compareTo(Edge o) {
            return Long.compare(w, o.w);
        }
    }
}
