from heapq import *
n,m=map(int,input().split())
E=[[]for _ in range(n+1)]
for _ in range(m):a,b=map(int,input().split());E[a]+=[b];E[b]+=[a]
s,b,k,h=map(int,input().split())
S=[];B=[];P=[float('inf')]*(s+1)
for _ in range(s):x,a,f=map(int,input().split());S+=[(a,f,x)];P[x]=0
for _ in range(b):x,d=map(int,input().split());B+=[(d,x)]
S.sort();B.sort()
ans=0;D=[]
for a,f,x in S:
    heapify(D)
    while D and D[0]<=f:heappop(D)
    for d,y in B:
        if d>a:break
        if P[y]>f:heappush(D,P[y])
        if not D:break
        P[y]=min(P[y],f)
for x in P:
    if x==float('inf'):continue
    if x>0:
        ans+=k
    else:
        ans-=h
print(-ans)
