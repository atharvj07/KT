#include<cstdio>
#include<algorithm>

bool board[5][5];
void print(void){
    for(int i=1;i<=4;++i)
        for(int j=1;j<=4;++j)
            putchar(board[i][j]?'#':'.');
    putchar('\n');
}
int main(void){
    char c;
    for(int i=1;i<=4;++i)
        for(int j=1;j<=4;++j){
            scanf(" %c",&c);
            board[i][j]=c=='#';
        }
    int cnt1=0,cnt2=0,cnt3=0,cnt4=0;
    for(int i=1;i<=4;++i)
        for(int j=1;j<=4;++j){
            if(board[i][j]==board[i][j+1] && board[i][j+1]==board[i+1][j+1] && board[i+1][j+1]==board[i+1][j])
                cnt1++;
            if(board[i][j]==board[i+1][j] && board[i+1][j]==board[i+1][j+1] && board[i+1][j+1]==board[i][j+1])
                cnt2++;
            if(board[i][j]==board[i][j+1] && board[i][j+1]==board[i+1][j] && board[i+1][j]==board[i+1][j+1])
                cnt3++;
            if(board[i][j+1]==board[i+1][j] && board[i+1][j]==board[i+1][j+1] && board[i+1][j+1]==board[i][j])
                cnt4++;
        }
    if(cnt1 || cnt2 || cnt3 || cnt4 || board[1][1]==board[1][2] && board[1][2]==board[2][1] && board[2][1]==board[2][2])
        puts("YES");
    else
        puts("NO");
    return 0;
}
