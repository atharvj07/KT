import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long count = 0;
        for (long i = 1; i <= n; i++) {
            boolean divisible = true;
            for (long j = 2; j <= 10; j++) {
                if (i % j != 0) {
                    divisible = false;
                    break;
                }
            }
            if (divisible) {
                count++;
            }
        }
        System.out.println(count);
    }
}
