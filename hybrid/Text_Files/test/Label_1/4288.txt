#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;
typedef long long ll;
const int mod = 1e9 + 7;
int f(ll n) {
    if (n % 2 == 0) return (n + 2) / 2;
    return n + 1;
}
int g(ll n) {
    int sum = 0;
    for (ll i = 1; i * i <= n; i++) {
        if (n % i == 0) {
            sum = (sum + f(n / i)) % mod;
            if (n / i != i) sum = (sum + f(i)) % mod;
        }
    }
    return sum;
}
int main() {
    ll n, k; cin >> n >> k;
    unordered_map<ll, vector<int>> memo;
    vector<int> &v = memo[n];
    if (k == 1) return cout << f(g(n)) << endl, 0;
    if (k & 1) {
        if (v.size() < k) { v.push_back(g(f(n))); }
        return cout << v[k - 1] << endl, 0;
    } else {
        if (v.size() < k) { v.push_back(f(g(n))); }
        return cout << v[k - 1] << endl, 0;
    }
}
