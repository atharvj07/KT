from collections import defaultdict
n,k=map(int,input().split())
c=list(map(int,input().split()))
mp=defaultdict(int)
mp[0]=1
for i in c:
    for j in list(mp):
        mp[j+i]=1
mn=k
for i in c:
    if k-i>=0:
        mn=min(mn,k-i)
ans=[]
for i in mp:
    if i>=mn:
        ans.append(i)
print(len(ans))
print(*ans)
