import sys
def area(points):
    x1, y1, x2, y2 = sys.maxsize, sys.maxsize, -sys.maxsize, -sys.maxsize
    for px, py in points:
        x1 = min(x1, px)
        y1 = min(y1, py)
        x2 = max(x2, px)
        y2 = max(y2, py)
    if x1 == x2 or y1 == y2: return -1
    return (x2 - x1) * (y2 - y1)

n = int(input())
points = []
for _ in range(n):
    x, y = map(int, input().split())
    points.append((x, y))
print(area(points))
