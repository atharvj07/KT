#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main() {
	int n, x, f = 0, s = 0, mx = 0, mn = 9e8;
	vector<int> a, b;
	cin >> n;
	for (int i = 0; i < n; i++) {
		cin >> x;
		if(x > 0) a.push_back(x);
		else if(x < 0) b.push_back(-x);
		f += (x > 0 ? x : 0);
		s += (x < 0 ? x : 0);
	}
	sort(a.begin(), a.end());
	sort(b.begin(), b.end());
	if(f > s) cout << "first";
	else if(f < s) cout << "second";
	else {
		if(a.size() > b.size()) cout << "first";
		else if(a.size() < b.size()) cout << "second";
		else {
			for (int i = 0; i < a.size(); i++) {
				if(a[i] > b[i]) {cout << "first"; return 0;}
				if(a[i] < b[i]) {cout << "second"; return 0;}
			}
		}
	}
	return 0;
}
