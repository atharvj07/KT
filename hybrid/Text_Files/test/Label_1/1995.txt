#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int need = 0;
    for (int i = 0; i < n; i++) {
        if (need) need--;
        if (i && a[i] + a[i - 1] < k)
            need = k - (a[i] + a[i - 1]);
        b[i] = max(a[i], need);
        need = b[i] - a[i];
    }
    cout << need << '\n';
    for (int i = 0; i < n; i++) cout << b[i] << ' ';
    cout << '\n';
    return 0;
}
