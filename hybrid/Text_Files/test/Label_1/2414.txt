#include <iostream>
using namespace std;
int grid[501][501], mr[501][501], ans;
int main() {
  int n, m, q, val;
  cin >> n >> m >> q;
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) {
      cin >> val;
      grid[i][j] = val;
      mr[i][j] = (val == 0);
    }
  for (int i = 1; i <= n; i++)
    for (int j = 2; j <= m; j++) mr[i][j] += mr[i][j - 1];
  for (int i = 1; i <= m; i++)
    for (int j = 2; j <= n; j++) mr[j][i] += mr[j - 1][i];
  for (int k = 1; k <= q; k++) {
    int i, j;
    cin >> i >> j;
    if (grid[i][j] == 0) {
      ans = max(ans, mr[i][j] + mr[i][j - 1]);
    } else {
      ans = max(ans, mr[i - 1][j] + mr[i][j] + (mr[i][j - 1] - mr[i - 1][j - 1]));
    }
    grid[i][j] ^= 1;
    cout << ans << endl;
  }
}
