h,w=list(map(int, input().split()))
pic=[input() for i in range(h)]
c=s=e=n=0
for i in range(w):
    cc=0
    for j in range(h):
        if pic[j][i]=="*":
            if cc==0:
                if j-1>=0 and pic[j-1][i]=="*":
                    c+=1
            else:
                s=1
        else: cc+=1
    if s==0 and cc==1:
        e+=1
for i in range(h):
    cc=0
    for j in range(w):
        if pic[i][j]=="*":
            if cc==0:
                if j-1>=0 and pic[i][j-1]=="*":
                    c+=1
            else:
                s=1
        else: cc+=1
    if s==0 and cc==1:
        n+=1
if s==0 and c==1 and e==1 and n==1:
    print("YES")
else:
    print("NO")
