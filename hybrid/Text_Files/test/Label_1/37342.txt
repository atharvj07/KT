import difflib

n = int(input())
a = [input() for _ in range(n)]
b = [input() for _ in range(n)]
res = []
sum = 0
for i in range(n):
    t = -1
    for j in range(n):
        if b[j] not in res:
            c = difflib.get_close_matches(a[i], [b[j]])
            if len(c) and (len(c[0]) > t or len(c[0]) == t and c[0] < b[res[i - 1]]):
                t = len(c[0])
                res.append(j)
for i in range(n):
    sum += t
    print(i + 1, res[i] + 1)
print(sum)
