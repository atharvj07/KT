#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  vector<int> cnt(3);
  for (char c : s) cnt[c - 'B']++;
  vector<char> ans;
  for (int i = 0; i < 3; i++) {
    if (cnt[i] % 2 == 1) ans.push_back('B' + i);
  }
  for (char c : ans) cout << c;
  cout << endl;
  return 0;
}
