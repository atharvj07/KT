import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        StringBuilder sb = new StringBuilder();
        while (w > 0) {
            if (w % 3 == 1) {
                sb.append("+");
                w -= 4;
            } else if (w % 3 == 2) {
                sb.append("0");
                w -= 2;
            } else {
                sb.append("-");
                w -= 1;
            }
        }
        while (sb.charAt(0) == '0') {
            sb.deleteCharAt(0);
        }
        System.out.println(sb.reverse());
    }
}
