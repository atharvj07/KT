#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n;) {
    int j = i + 1;
    while (j < n && s[j] == s[i]) {
      j++;
    }
    if (j - i > 1) {
      ans++;
    }
    i = j;
  }
  cout << ans << endl;
  return 0;
}
