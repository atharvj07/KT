import sys

inp = sys.stdin.readlines()
n, k = list(map(int,inp[0].split()))
arr = [list(map(int,x.split())) for x in inp[1:]]
need, c, ans = 0, k, 0
for l, r, a in arr:
    need = max(0, need + a - k)
    re = (need + k - 1) // k
    ans += re
    c -= re*k
    c = max(0, c)
    need = max(0, a - c)
print(ans*k + need if need+c<=k else -1)
