import java.util.*;
public class Main {
    static int k, mx = 0;
    static List<Integer>[] g;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); k = sc.nextInt();
        g = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) g[i] = new ArrayList<>();
        for (int i = 2; i <= n; i++) {
            int u = sc.nextInt();
            g[i].add(u); g[u].add(i);
        }
        dfs(1, -1, 0);
        System.out.println(mx);
    }

    static void dfs(int v, int p, int dist) {
        boolean leaf = true;
        for (int u: g[v]) {
            if (u != p) {
                leaf = false;
                dfs(u, v, dist + 1);
            }
        }
        if (leaf) {
            mx = Math.max(mx, Math.min(dist, k) + 1);
        }
    }
}
