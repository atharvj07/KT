n = int(input())
session_ids = list(map(int, input().split()))

session_counts = {}
error = False
for session_id in session_ids:
    if session_id != 0:
        if session_id in session_counts:
            session_counts[session_id] += 1
            if session_counts[session_id] > 2:
                error = True
                break
        else:
            session_counts[session_id] = 1

if error or any(count == 1 for count in session_counts.values()):
    print(-1)
else:
    print(len([count for count in session_counts.values() if count == 2]) // 2)
