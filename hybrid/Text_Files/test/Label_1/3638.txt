#include <iostream>
#include <string>

using namespace std;

int main() {
  int n, t;
  cin >> n >> t;
  string s1, s2;
  cin >> s1 >> s2;
  if (t % 2 != 0 || t > n) {
    cout << "-1" << endl;
    return 0;
  }
  string s3 = s1;
  for (int i = 0; i < n; i++) {
    if (s1[i] != s2[i]) {
      if (t == 0) {
        s3[i] = s1[i];
      } else {
        s3[i] = 'a' + 'z' - s1[i];
        t -= 2;
      }
    }
  }
  for (int i = 0; i < n && t > 0; i++) {
    if (s3[i] == s1[i] && s3[i] == s2[i]) {
      s3[i] = 'a' + 'z' - s3[i];
      t -= 2;
    }
  }
  cout << s3 << endl;
  return 0;
}
