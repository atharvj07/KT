import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        line = br.readLine().split(" ");
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(line[i]);
        }
        line = br.readLine().split(" ");
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = Integer.parseInt(line[i]);
        }
        boolean[] hasA = new boolean[10];
        boolean[] hasB = new boolean[10];
        for (int i = 0; i < n; i++) {
            hasA[a[i]] = true;
        }
        for (int i = 0; i < m; i++) {
            hasB[b[i]] = true;
        }
        int ans = 1;
        while (true) {
            boolean hasDigitA = false;
            boolean hasDigitB = false;
            int temp = ans;
            while (temp > 0) {
                if (hasA[temp % 10]) {
                    hasDigitA = true;
                }
                if (hasB[temp % 10]) {
                    hasDigitB = true;
                }
                temp /= 10;
            }
            if (hasDigitA && hasDigitB) {
                break;
            }
            ans++;
        }
        System.out.println(ans);
    }
}
