#include <bits/stdc++.h>
using namespace std;
pair<int, int> p[5];
int ans = 0;
int main() {
  int t, n, m, x, y;
  cin >> t;
  while (t--) {
    ans = 0;
    cin >> n >> m >> x >> y;
    p[1] = {x, y};
    p[2] = {x, m + 1 - y};
    p[3] = {n + 1 - x, y};
    p[4] = {n + 1 - x, m + 1 - y};
    for (int i = 1; i <= 4; ++i) {
      for (int j = i + 1; j <= 4; ++j) {
        int len = max(abs(p[i].first - p[j].first), abs(p[i].second - p[j].second));
        len = len + len - 1;
        ans = max(ans, len);
      }
    }
    int len = max(n - 1, m - 1);
    len = len + len;
    ans = max(ans, len);
    cout << 1 << ' ' << 1 << ' ' << n << ' ' << m << '\n';
  }
}
