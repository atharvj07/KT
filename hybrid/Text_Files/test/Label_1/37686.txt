n, m = map(int, input().split())
species = {}
events = []
for i in range(n):
    s, x = map(float, input().split())
    if s not in species:
        species[s] = [x]
    else:
        species[s].append(x)
    events.append((x, s, i))

events.sort()
curr = [events[0][1]]
rep = 0
for x, s, i in events:
    if s not in curr:
        rep += 1
    curr.append(s)
    if len(curr) == m:
        curr.pop(0)
print(rep)
