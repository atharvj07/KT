t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    diff_counts = {}
    for i in range(n):
        diff = a[i] - i
        if diff in diff_counts:
            diff_counts[diff] += 1
        else:
            diff_counts[diff] = 1
    result = 0
    for count in diff_counts.values():
        result += (count * (count - 1)) // 2
    print(result)
