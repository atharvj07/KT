#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, have = 0;
  cin >> n;
  while (n--) {
    cin >> a;
    if (have >= a && have % 25 >= a)
      have -= a;
    else if (have >= 25) {
      have = a - have;
      have += 25;
    } else
      return cout << "NO", 0;
  }
  return cout << "YES", 0;
}
