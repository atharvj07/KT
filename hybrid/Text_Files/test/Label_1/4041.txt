#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,t;
	cin>>n>>t;
	string s;
	cin>>s;
	bool f=false; int p=0;
	for(int i=s.size()-1;i>=0;i--){
		if(s[i]=='.'){
			p=i;
			continue;
		}
		if(!f and s[i]<'5') continue;
		f=true;
		if(s[i]=='9'){
			while(i>=0 and s[i]=='9') i--;
			if(i>=0){
				s[i]+=1;
				while(i+1<s.size()) s[i+1]='0',i++;
			}
			else{
				s="1";
				for(int j=1;j<n;j++) s+='0';
			}
		}
		else{
			s[i]+=1;
			if(p!=0 and (i-p)<=t) s.erase(p+i-p-t+1,s.size());
			else if(p==0 and i<=t) s=s.substr(0,i+1);
			break;
		}
	}
	if(p==0 and f==false) cout<<s;
	else{
		while(s.size()>1 and s[s.size()-1]=='0') s.erase(s.size()-1,1);
		if(s[s.size()-1]=='.') s.erase(s.size()-1,1);
		cout<<s;
	}
}
