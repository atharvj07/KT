import sys

n, m = map(int, input().split())
a = [input() for _ in range(n)]
ans = []
for i, l in enumerate(a):
    for j, c in enumerate(l):
        if c != '.':
            if j > 0 and a[i][j - 1] == c:
                ans.append((i + 1, j - 1, int(c)))
            elif i > 0 and a[i - 1][j] == c:
                ans.append((i + 1 - int(c), j, int(c)))
            else:
                ans.append((i + 1, j, int(c)))
print(len(ans))
for x, y, s in ans:
    print(x, y, s)
