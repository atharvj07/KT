#include <bits/stdc++.h>
using namespace std;
vector<int> v[1001];
int main() {
  int n, m, a, b, visited[1001] = {};
  cin >> n >> m;
  while (m--) {
    cin >> a >> b;
    v[a].push_back(b);
    v[b].push_back(a);
  }
  for (int i = 1; i <= n; i++) {
    if (!visited[i]) {
      cout << v[i].size() << endl;
      for (int j : v[i]) { cout << i << ' ' << j << endl; }
      for (int j : v[i]) {
        for (int k : v[j]) {
          visited[k] = 1;
          if (k != j) cout << j << ' ' << k << endl;
        }
      }
    }
  }
  return 0;
}
