import java.util.Scanner;

public class ZebraPhoto {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] photo = new int[n];
        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        boolean isZebra = true;
        int currentColor = photo[0];
        int stripeWidth = 0;
        for (int i = 1; i < n; i++) {
            if (photo[i] == currentColor) {
                stripeWidth++;
            } else {
                if (stripeWidth == 0) {
                    isZebra = false;
                    break;
                }
                currentColor = photo[i];
                stripeWidth = 1;
            }
        }

        if (stripeWidth == 0) {
            isZebra = false;
        }

        System.out.println(isZebra ? "YES" : "NO");
    }
}
