import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        int[] a = new int[n];
        StringTokenizer st = new StringTokenizer(in.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }
        int[] b = new int[1 << (n + 1)];
        for (int i = 0; i < (1 << n); i++) {
            int x = 0;
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) != 0) {
                    x ^= a[j];
                }
            }
            b[i] = x;
        }
        for (int i = (1 << n) - 1; i > 0; i--) {
            for (int j = 0; j < i; j++) {
                if (b[j] > b[j + 1]) {
                    int t = b[j];
                    b[j] = b[j + 1];
                    b[j + 1] = t;
                }
            }
        }
        int x = 0;
        for (int i = 0; i < (1 << n); i++) {
            if (b[i] == b[i + 1]) {
                x = Math.max(x, Integer.bitCount(i));
            }
        }
        System.out.println(x);
        for (int i = 0; i < (1 << x); i++) {
            int y = 0;
            for (int j = 0; j < x; j++) {
                if ((i & (1 << j)) != 0) {
                    y ^= a[j];
                }
            }
            System.out.print(y + " ");
        }
        System.out.println();
    }
}
