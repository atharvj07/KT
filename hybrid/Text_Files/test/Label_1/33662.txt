import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        for (int i = 0; i < t; i++) {
            int n = input.nextInt();
            int k = input.nextInt();
            int m = 0;
            List<Integer> ans = new ArrayList<>();
            for (int j = 1; j <= n; j++) {
                if (j == k) {
                    continue;
                }
                m++;
                ans.add(j);
            }
            System.out.println(m);
            for (int j = 0; j < m; j++) {
                System.out.print(ans.get(j) + " ");
            }
            System.out.println();
        }
    }
}
