#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int n;
    cin >> n;
    ll a[30];
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            continue;
        }
        ll t = 0;
        for (int j = 0; j <= i; j++) {
            if (a[j] != 0) {
                a[j]--, t++;
            }
        }
        if (t) {
            ans += t;
        }
    }
    cout << ans << endl;
    return 0;
}
