import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[][] points = new int[4 * n][4];
        for (int i = 0; i < 4 * n; i++) {
            for (int j = 0; j < 4; j++) {
                points[i][j] = scanner.nextInt();
            }
        }

        for (int i = 0; i < n; i++) {
            int[][] p = new int[4][2];
            for (int j = 0; j < 4; j++) {
                p[j][0] = points[4 * i + j][0];
                p[j][1] = points[4 * i + j][1];
            }

            int[] dx = {0, 1, 0, -1};
            int[] dy = {1, 0, -1, 0};

            int ans = -1;
            for (int j = 0; j < 4; j++) {
                for (int k = 0; k < 4; k++) {
                    boolean ok = true;
                    for (int l = 0; l < 4; l++) {
                        int x = p[(j + l) % 4][0] - dx[k] * points[4 * i + l][2] + dy[k] * points[4 * i + l][3];
                        int y = p[(j + l) % 4][1] - dx[k] * points[4 * i + l][3] - dy[k] * points[4 * i + l][2];
                        if (x != p[(j + l + 1) % 4][0] || y != p[(j + l + 1) % 4][1]) {
                            ok = false;
                            break;
                        }
                    }
                    if (ok) {
                        ans = j;
                        break;
                    }
                }
                if (ans != -1) {
                    break;
                }
            }

            System.out.println(ans);
        }
    }
}
