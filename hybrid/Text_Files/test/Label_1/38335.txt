from collections import Counter
for _ in range(int(input())):
    a = Counter(map(int, input().split()))
    check = 0
    for i in [2048, 1024, 512, 256, 128, 64, 32, 16, 8, 4, 2, 1]:
        while a[i]:
            if a[i] % 2:
                check = 1
                break
            a[i // 2] += a[i] // 2
            a[i] = 0
    print(['NO', 'YES'][check])
