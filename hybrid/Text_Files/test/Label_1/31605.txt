import java.util.Scanner;
import java.util.HashMap;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        HashMap<Integer, Integer> x = new HashMap<Integer, Integer>();
        HashMap<Integer, Integer> y = new HashMap<Integer, Integer>();
        for (int i = 0; i < n; i++) {
            int xi = sc.nextInt();
            int yi = sc.nextInt();
            String ui = sc.next();
            int di = 0;
            if (ui.equals("U")) di = 1;
            else if (ui.equals("R")) di = 2;
            else if (ui.equals("D")) di = 3;
            else if (ui.equals("L")) di = 4;
            if (x.containsKey(xi * 1000000 + di)) {
                System.out.println(x.get(xi * 1000000 + di));
                return;
            }
            if (y.containsKey(yi * 1000000 + di)) {
                System.out.println(y.get(yi * 1000000 + di));
                return;
            }
            x.put(xi * 1000000 + di, i + 1);
            y.put(yi * 1000000 + di, i + 1);
        }
        System.out.println("SAFE");
    }
}
