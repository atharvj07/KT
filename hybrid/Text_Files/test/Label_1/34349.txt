import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int s = in.nextInt();
        in.close();
        System.out.println((Math.abs(a) + Math.abs(b) <= s && s % 2 == (a + b) % 2) ? "Yes" : "No");
    }
}
