#include <bits/stdc++.h>
using namespace std;
int n, m, cnt, a[200005];
vector<int> v;
int main() {
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) {
    v.push_back(a[i]);
    while (v.size() >= 3 && v[v.size() - 2] > v[v.size() - 3] &&
           v[v.size() - 3] > v[v.size() - 1]) {
      v.erase(v.begin() + v.size() - 2);
    }
    if (v.size() > 1 && v[v.size() - 1] == m) cnt++;
  }
  cout << cnt << endl;
  return 0;
}
