import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long h = sc.nextLong();
        long m = sc.nextLong();
        int k = sc.nextInt();
        long h1 = sc.nextLong();
        long m1 = sc.nextLong();
        long h2 = sc.nextLong();
        long m2 = sc.nextLong();
        long ans = (h - 1 - h1) * m + m1;
        if (h1 > 0) ans += (h2 - h1) * m;
        if (h2 == 0) ans--;
        ans /= k;
        System.out.println(ans);
    }
}
