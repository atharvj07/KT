for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    MX = max(a)
    ans = []
    for k in range(1, n + 1):
        msk = 0
        for j in range(0, n, k):
            msk |= a[j]
        if msk == 0:
            ans.append(k)
            if k == MX:
                break
    print(*ans)
