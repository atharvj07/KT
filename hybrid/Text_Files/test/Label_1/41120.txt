import heapq
for t in range(int(input())):
    n, m = map(int,input().split())
    a = [[int(i) for i in input().split()] for i in range(n)]
    for i in range(n):
        a[i].sort()
    b = [[] for i in range(n)]
    for i in range(n):
        for j in range(m):
            heapq.heappush(b[i],a[i][j])
    for i in range(n):
        for j in range(m):
            print(heapq.heappop(b[i]),end=' ')
        print()
