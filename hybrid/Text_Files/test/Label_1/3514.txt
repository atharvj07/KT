#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    int x;
    cin >> s >> x;
    vector<int> w(s.length(), -1);
    for (int i = 0; i < s.length(); i++) {
      if (s[i] == '1' && i >= x) w[i - x] = 1;
      if (s[i] == '1' && i + x < s.length()) w[i + x] = 1;
    }
    bool found = false;
    for (int i = 0; i < w.size(); i++) {
      if (w[i] == 1) {
        found = true;
        for (int j = 0; j < s.length(); j++) {
          if (j >= x && w[j - x] == -1) w[j - x] = 0;
          if (j + x < s.length() && w[j + x] == -1) w[j + x] = 0;
        }
        break;
      }
    }
    if (!found) {
      cout << "-1" << endl;
      continue;
    }
    for (int i = 0; i < w.size(); i++) {
      if (w[i] == -1) w[i] = 0;
    }
    for (int i = 0; i < w.size(); i++) {
      cout << w[i];
    }
    cout << endl;
  }
  return 0;
}
