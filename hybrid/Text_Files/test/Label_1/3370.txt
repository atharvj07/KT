#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> b(n);
  for (int &x : b) cin >> x;
  sort(b.begin(), b.end());

  if (n < 3) {
    cout << 0 << "\n";
    return 0;
  }
  int ans = INT_MAX;
  for (int i = 1; i < n - 1; i++) {
    int x = b[i] - b[i - 1], y = b[i + 1] - b[i];
    if (x != y) {
      int change = abs(x - y) + 1;
      ans = min(ans, change);
    }
  }
  cout << (ans == INT_MAX ? -1 : ans) << "\n";
}
