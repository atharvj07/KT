#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) { cin >> p[i]; }
    int q = 0;
    for (int i = 0; i < n; i++) {
        int x = p[i];
        for (int j = 0; j < n; j++) { x ^= (i % (j + 1)); }
        q ^= x;
    }
    cout << q << endl;
    return 0;
}
