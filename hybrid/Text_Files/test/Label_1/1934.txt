#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> h(2 * n);
    for (int i = 0; i < 2 * n; i++) h[i] = i + 1;
    long long ans = 1;
    for (int i = 0; i < k; i++) {
        int x, y;
        char s;
        cin >> x >> s >> y;
        x--, y--;
        if (h[x] < h[y] && (s == '>' || s == '>=')) continue;
        if (h[x] > h[y] && (s == '<' || s == '<=')) continue;
        if (h[x] == h[y] && (s != '=')) continue;
        ans *= (s != '=' ? 2 : 1);
    }
    cout << ans << endl;
    return 0;
}
