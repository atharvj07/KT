#include<iostream>
#include<map>
using namespace std;

int main(){
    int n; scanf("%d", &n);
    map<string,int> cnt;
    for(int i=0;i<n;++i){
        string s; cin >> s;
        cnt[s]++;
    }
    int ans=0;
    for(map<string, int>::iterator it=cnt.begin();it!=cnt.end();it++){
        int x=it->second;
        x/=2; ans+=x; it->second-=2*x;
    }
    for(map<string, int>::iterator it=cnt.begin();it!=cnt.end();it++){
        int x=it->second;
        int y=cnt["("+it->first+")]"];
        x=min(x,y); y-=x; ans+=x; it->second-=x; cnt["("+it->first+")]"]-=y;
    }
    printf("%d\n", ans);
}
