import sys

n, m, d = map(int, sys.stdin.readline().split())
a = sorted(list(map(int, sys.stdin.readline().split())))

ans = []
day = 1
cur = -sys.maxsize
for x in a:
    if x - cur >= d:
        ans.append(day)
        cur = x
    else:
        ans.append(day+1)
        day += 1
        ans.append(day)
        cur = x
print(len(set(ans)))
print(*ans)
