import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long ans = 0;
        while (n > 0) {
            long digit = n % 10;
            ans += digit * digit;
            n /= 10;
            ans *= 10;
        }
        ans /= 10;
        System.out.println(ans);
    }
}
