import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        if (t == 1) System.out.println(3);
        else if (t == 2 || t == 3) System.out.println(1);
        else System.out.println(2);
    }
}
