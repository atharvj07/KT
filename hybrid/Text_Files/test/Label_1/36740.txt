import sys

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a


def check(arr):
    n = len(arr)
    for i in range(1, n):
        if gcd(arr[i], arr[i - 1]) <= 1:
            return 'No'
    return 'Yes'


a = list(map(int, sys.stdin.readline().split()))
n = a[0]
a = a[1:]
a.sort()
print(check(a))
