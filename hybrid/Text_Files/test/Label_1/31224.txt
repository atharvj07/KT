import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n + 1];
        long sum = 0;
        for (int i = 1; i <= n; ++i) {
            a[i] = sc.nextLong();
            sum += a[i];
        }
        if (sum % 3 != 0) {
            System.out.println(0);
            return;
        }
        long subsum = sum / 3;
        long pref[] = new long[n + 1];
        long prefCnt[] = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            pref[i] = pref[i - 1] + a[i];
        }
        for (int i = 1; i <= n; i++) {
            if (pref[i] == subsum) {
                prefCnt[i]++;
            }
            prefCnt[i] += prefCnt[i - 1];
        }
        long suf[] = new long[n + 1];
        long sufCnt[] = new long[n + 1];
        for (int i = n; i >= 1; i--) {
            suf[i] = suf[i + 1] + a[i];
        }
        for (int i = n; i >= 1; i--) {
            if (suf[i] == subsum) {
                sufCnt[i]++;
            }
            sufCnt[i] += sufCnt[i + 1];
        }
        long ans = 0;
        for (int i = 1; i + 1 < n; i++) {
            if (pref[i] == subsum) {
                ans += sufCnt[i + 1];
            }
        }
        System.out.println(ans);
    }
}
