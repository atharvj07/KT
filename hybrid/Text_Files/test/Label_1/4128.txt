#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (n); ++i)
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
const int MOD = 1e9 + 7;
const int INF = 1e9;
const ll LINF = 1e18;

struct Edge {
    int to;
    ll w;
};

vector<vector<Edge>> G;
vector<ll> ans;

void dfs(int u, int p, ll w) {
    for (Edge e : G[u]) {
        if (e.to != p) {
            ans[e.to - 1] = w + e.w;
            dfs(e.to, u, w + e.w);
        }
    }
}

int main() {
    int n;
    cin >> n;
    G.resize(n + 1);
    rep(i, n - 1) {
        int u, v;
        cin >> u >> v;
        G[u].push_back({v, 0});
        G[v].push_back({u, 0});
    }
    ans.resize(n);
    dfs(1, 1, 0);
    bool ok = true;
    rep(i, n) {
        if (ans[i] != 0) ok = false;
    }
    if (ok) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
