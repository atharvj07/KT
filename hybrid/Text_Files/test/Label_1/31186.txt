import java.util.Scanner;
import java.util.ArrayList;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] cards = new int[n];
            for (int i = 0; i < n; i++) {
                cards[i] = sc.nextInt();
            }
            if (solve(cards)) {
                for (int i = 0; i < n; i++) {
                    System.out.print(cards[i] + " ");
                }
                System.out.println();
            } else {
                System.out.println("No");
            }
        }
    }

    public static boolean solve(int[] cards) {
        int n = cards.length;
        int[] sum = new int[14];
        int[] count = new int[14];
        for (int i = 0; i < n; i++) {
            sum[cards[i]] += cards[i];
            count[cards[i]]++;
        }
        ArrayList<Integer> played = new ArrayList<Integer>();
        return solve(cards, sum, count, played);
    }

    public static boolean solve(int[] cards, int[] sum, int[] count, ArrayList<Integer> played) {
        if (played.size() == cards.length) {
            return true;
        }
        for (int i = 1; i <= 13; i++) {
            if (count[i] == 0) {
                continue;
            }
            boolean valid = true;
            for (int j : played) {
                if (sum[i] % j != 0) {
                    valid = false;
                    break;
                }
            }
            if (!valid) {
                continue;
            }
            played.add(i);
            count[i]--;
            if (solve(cards, sum, count, played)) {
                return true;
            }
            played.remove(played.size() - 1);
            count[i]++;
        }
        return false;
    }
}
