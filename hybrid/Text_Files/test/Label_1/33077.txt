import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int k = in.nextInt();
        int[] s1 = new int[k];
        for (int i = 0; i < k; i++) {
            s1[i] = in.nextInt();
        }
        int l = in.nextInt();
        int[] s2 = new int[l];
        for (int i = 0; i < l; i++) {
            s2[i] = in.nextInt();
        }
        Arrays.sort(s1);
        Arrays.sort(s2);
        boolean ok = true;
        for (int i = 0; i < k; i++) {
            if (s1[i] < n - i - 1) {
                ok = false;
            }
        }
        for (int i = 0; i < l; i++) {
            if (s2[i] < m - i - 1) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
