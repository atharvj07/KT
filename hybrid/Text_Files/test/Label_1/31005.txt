import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }
        int[] weight = new int[n];
        for (int i = 0; i < n - 1; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int c = sc.nextInt();
            int gcd = GCD(m, c);
            parent[findRoot(parent, b)] = findRoot(parent, a);
            weight[findRoot(parent, a)] += gcd;
            weight[findRoot(parent, b)] += gcd;
        }
        int ans = 0;
        for (int i : weight) {
            if (i % m == 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }

    private static int GCD(int a, int b) {
        if (a == 0) {
            return b;
        }
        return GCD(b % a, a);
    }

    private static int findRoot(int[] parent, int i) {
        if (parent[i] == i) {
            return i;
        }
        return parent[i] = findRoot(parent, parent[i]);
    }
}
