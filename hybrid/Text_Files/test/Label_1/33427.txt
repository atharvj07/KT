import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int n = s.length();
        int ans = 0;
        int i = 0, j = 0, cnt = 0;
        while(j < n) {
            if(s.charAt(j)=='A' || s.charAt(j)=='C' || s.charAt(j)=='G' || s.charAt(j)=='T') cnt++;
            else cnt = 0;
            if(cnt>ans) ans = cnt;
            j++;
        }
        System.out.println(ans);
    }
}
