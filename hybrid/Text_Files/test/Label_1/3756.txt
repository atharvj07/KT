#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m;
    cin >> n >> m;
    vector<string> a(n);
    for (auto &x : a) cin >> x;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
      for (int j = 0; j < m / 2; ++j) {
        ans += (a[i][j] != a[i][m - 1 - j]);
      }
    }
    cout << ans << "\n";
  }
}
