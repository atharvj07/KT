#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, m;
    cin >> n >> m;
    int a[m + 1], dp[n + 1];
    for (int i = 1; i <= m; i++) cin >> a[i];
    memset(dp, 0x3f, (n + 1) * sizeof(dp[0]));
    dp[1] = 0;
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (a[i] == j) dp[j] = min(dp[j], dp[j - 1]);
            dp[j] = min(dp[j], dp[(j + n - 1) % n] + 1);
        }
    }
    cout << dp[a[m]] << endl;
    return 0;
}
