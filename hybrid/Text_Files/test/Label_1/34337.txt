import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class BytelandLanguage {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read the vowel and consonant information
        String alphabet = scanner.nextLine();
        boolean[] isVowel = new boolean[alphabet.length()];
        for (int i = 0; i < alphabet.length(); i++) {
            isVowel[i] = alphabet.charAt(i) == 'V';
        }

        // Read the number of letters and rules
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        // Read the rules
        List<Rule> rules = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int pos1 = scanner.nextInt() - 1;
            char type1 = scanner.next().charAt(0);
            int pos2 = scanner.nextInt() - 1;
            char type2 = scanner.next().charAt(0);
            rules.add(new Rule(pos1, type1, pos2, type2));
        }

        // Read the string
        String s = scanner.next();

        // Find the lexicographically smallest word
        char[] result = findSmallestWord(n, isVowel, rules, s.toCharArray());

        // Print the result
        if (result == null) {
            System.out.println("-1");
        } else {
            System.out.println(new String(result));
        }
    }

    // Rule class to represent a rule of the language
    static class Rule {
        int pos1;
        char type1;
        int pos2;
        char type2;

        Rule(int pos1, char type1, int pos2, char type2) {
            this.pos1 = pos1;
            this.type1 = type1;
            this.pos2 = pos2;
            this.type2 = type2;
        }
    }

    // Find the lexicographically smallest word that is not smaller than the given string
    static char[] findSmallestWord(int n, boolean[] isVowel, List<Rule> rules, char[] s) {
        char[] result = new char[n];
        Arrays.fill(result, 'a');

        // Try to find a valid word by incrementing the characters one by one
        for (int i = 0; i < n; i++) {
            char c = result[i];
            while (c < 'a' + isVowel.length) {
                // Check if the current word is valid
                if (isValidWord(n, isVowel, rules, result)) {
                    return result;
                }

                // Check if the current character can be incremented
                if (i == n - 1 || c < s[i]) {
                    c++;
                } else {
                    break;
                }
            }

            // If the current character cannot be incremented, reset it to 'a' and move to the next character
            result[i] = 'a';
        }

        // If no valid word is found, return null
        return null;
    }

    // Check if the given word is valid according to the rules
    static boolean isValidWord(int n, boolean[] isVowel, List<Rule> rules, char[] word) {
        for (Rule rule : rules) {
            if ((isVowel[word[rule.pos1] - 'a'] == (rule.type1 == 'V')) &&
                (isVowel[word[rule.pos2] - 'a'] != (rule.type2 == 'V'))) {
                return false;
            }
        }
        return true;
    }
}
