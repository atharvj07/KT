import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] dp = new int[n];
        Arrays.fill(dp, 1);
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {
                if (a[i] > a[j]) {
                    dp[i] = Math.max(dp[i], dp[j] + 1);
                }
            }
        }
        int max = 0;
        int idx = 0;
        for (int i = 0; i < n; i++) {
            if (dp[i] > max) {
                max = dp[i];
                idx = i;
            }
        }
        System.out.println(max);
        int[] ans = new int[max];
        for (int i = max - 1; i >= 0; i--) {
            while (dp[idx] != i + 1) {
                idx--;
            }
            ans[i] = a[idx];
            idx--;
        }
        for (int i = 0; i < max; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
