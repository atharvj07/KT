#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Card
{
    string pos;
    int str;
};

int main()
{
    int n, m;
    cin >> n >> m;
    vector<Card> jiro(n);
    for (int i = 0; i < n; i++)
    {
        cin >> jiro[i].pos >> jiro[i].str;
    }
    vector<int> ciel(m);
    for (int i = 0; i < m; i++)
    {
        cin >> ciel[i];
    }
    sort(ciel.begin(), ciel.end());
    sort(jiro.begin(), jiro.end(), [](const Card &a, const Card &b) {
        return a.str > b.str;
    });
    int damage = 0, j = 0;
    for (int i = 0; i < m; i++)
    {
        while (j < n && jiro[j].str > ciel[i])
        {
            j++;
        }
        if (j < n)
        {
            if (jiro[j].pos == "ATK")
            {
                damage += ciel[i] - jiro[j].str;
            }
            j++;
        }
        else
        {
            damage += ciel[i];
            break;
        }
    }
    cout << damage << endl;
    return 0;
}
