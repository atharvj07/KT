import sys

input = sys.stdin.readline

t = int(input())
arr = list(map(int, input().split()))

if t % 2 == 0 and all(arr[i] % 2 == 0 for i in range(t)):
    n = t // 2
    m = n + (0 in arr)
    print(n, m)
    j = arr.index(0) if 0 in arr else -1
    print(j // m + 1, j % m + 1)
else:
    print(-1)
