import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int q = in.nextInt();
        int[] arr = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            arr[i] = i;
        }
        int[] temp = new int[n + 1];
        for (int i = 0; i < q; i++) {
            int type = in.nextInt();
            if (type == 1) {
                int x = in.nextInt();
                if (x > 0) {
                    for (int j = 1; j <= n; j++) {
                        temp[j] = arr[(j + x) % n == 0 ? n : (j + x) % n];
                    }
                } else {
                    for (int j = 1; j <= n; j++) {
                        temp[j] = arr[(j + x + n) % n == 0 ? n : (j + x + n) % n];
                    }
                }
                arr = temp.clone();
            } else {
                for (int j = 1; j <= n; j++) {
                    if (j % 2 == 0) {
                        temp[j] = arr[(j + 1) / 2];
                    } else {
                        temp[j] = arr[n / 2 + (j + 1) / 2];
                    }
                }
                arr = temp.clone();
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(arr[i] + " ");
        }
        in.close();
    }
}
