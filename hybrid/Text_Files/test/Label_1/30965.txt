import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();
        int[][] arr = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < q; i++) {
            int a1 = sc.nextInt() - 1;
            int b1 = sc.nextInt() - 1;
            int c1 = sc.nextInt() - 1;
            int d1 = sc.nextInt() - 1;
            int h = sc.nextInt();
            int w = sc.nextInt();
            for (int j = 0; j < h; j++) {
                for (int k = 0; k < w; k++) {
                    int temp = arr[a1 + j][b1 + k];
                    arr[a1 + j][b1 + k] = arr[c1 + j][d1 + k];
                    arr[c1 + j][d1 + k] = temp;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(arr[i][j] + " ");
            }
            System.out.println();
        }
    }
}
