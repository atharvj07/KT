import sys
a, b = map(int, input().split())
s = input()
loc = [0, 0]
for c in s:
    if (c == 'U' and loc[1] < b) or (c == 'D' and loc[1] > 0) or (c == 'L' and loc[0] > a) or (c == 'R' and loc[0] < 0):
        if c == 'U': loc[1] += 1
        elif c == 'D': loc[1] -= 1
        elif c == 'L': loc[0] -= 1
        elif c == 'R': loc[0] += 1
        if loc == [a, b]:
            print('Yes')
            sys.exit()
print('No')
