#include <iostream>
using namespace std;
int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    if (n % 3 == 0)
      cout << n / 3 << " " << n / 3 << endl;
    else if ((n - 1) % 3 == 0)
      cout << 1 << " " << (n - 1) / 3 << endl;
    else
      cout << 2 << " " << (n - 2) / 3 << endl;
  }
}
