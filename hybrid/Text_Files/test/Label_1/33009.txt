import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;

public class Main {
    private static int h;
    private static int[] table;
    private static int dummyCalls = 0;

    public static void main(String[] args) throws IOException {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
            String[] meta = reader.readLine().split(" ");
            h = Integer.parseInt(meta[0]);
            int m = Integer.parseInt(meta[1]);
            int n = Integer.parseInt(meta[2]);
            table = new int[h];

            for (int i = 0; i < n; i++) {
                String[] instruction = reader.readLine().split(" ");
                if (instruction[0].equals("+")) {
                    int id = Integer.parseInt(instruction[1]);
                    int hash = Integer.parseInt(instruction[2]);
                    add(id, hash, m);
                } else {
                    int id = Integer.parseInt(instruction[1]);
                    remove(id);
                }
            }
            System.out.println(dummyCalls);
        }
    }

    private static void add(int id, int hash, int m) {
        int i = 0;
        int cell = (hash + i * m) % h;
        while (table[cell] != 0) {
            i++;
            dummyCalls++;
            cell = (hash + i * m) % h;
        }
        table[cell] = id;
    }

    private static void remove(int id) {
        boolean deleted = false;
        for (int i = 0; i < h; i++) {
            if (table[i] == id) {
                table[i] = 0;
                deleted = true;
            }
            if (deleted) {
                break;
            }
        }
    }
}
