from collections import defaultdict

n = int(input())
d = defaultdict(list)
for _ in range(n):
    s = input()
    c = 0
    v = ''
    for x in s:
        if x in 'aeiou':
            c += 1
            v = x
    d[(c, v)].append(s)

for k in d:
    d[k].sort()

ans = 0
for k in d:
    l = d[k]
    for i in range(0, len(l), 2):
        if i + 1 < len(l):
            ans += 1
            print(l[i], l[i + 1])

print(ans)
