from collections import defaultdict, deque
import sys
input = sys.stdin.readline

n = int(input())
s = deque()
d = defaultdict(list)
cnt = defaultdict(int)
for _ in range(n):
    t = input().rstrip()
    cnt[t] += 1
    d[len(t)].append(t)
for i in sorted(d):
    for j in d[i]:
        while cnt[j]:
            if not s or j != s[0]:
                s.appendleft(j)
                cnt[j] -= 1
            else:
                s.popleft()
print(len(s) // 2)
