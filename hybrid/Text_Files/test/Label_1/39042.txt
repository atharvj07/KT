def solve():
    n, a, b, k = map(int, input().split())
    arr = list(map(int, input().split()))
    arr.sort()
    i, ans = n-1, 0
    for _ in range(k):
        if i < 0 or arr[i] <= 0:
            break
        ans += arr[i] <= b
        i -= 1
    i, j = n-1, 0
    while i >= 0:
        if j >= n or arr[j] <= 0:
            break
        if arr[i] <= a:
            ans += 1
            i -= 1
        elif arr[j] <= b:
            j += 1
        else:
            i -= 1
    return ans

print(solve())
