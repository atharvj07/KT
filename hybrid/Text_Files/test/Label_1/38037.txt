n, r, avg = map(int, input().split())
exams = []

required_sum = avg * n
current_sum = 0

for _ in range(n):
    a, b = map(int, input().split())
    exams.append((a, b))
    current_sum += a

exams.sort(key=lambda x: x[1])

essays = 0
for a, b in exams:
    if current_sum >= required_sum:
        break
    max_possible = min(r - a, required_sum - current_sum)
    essays += max_possible * b
    current_sum += max_possible

print(essays)
