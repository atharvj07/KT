#include<bits/stdc++.h>
using namespace std;
#define int long long
main(){
    int a,b,p,x,ans=0,modi=1;
    cin>>a>>b>>p>>x;
    int bb=b;
    for(int i=2;i*i<=p;++i){
        if(p%i==0){modi*=i,p/=i;while(p%i==0)p/=i;}
    }
    if(p>1)modi*=p;
    for(int i=1;i<=x/modi;++i){
        b=bb;
        int aa=a;
        int tmp=i;
        while(tmp){
            if(tmp%2)b=b*aa%modi;
            tmp/=2;
            aa=aa*aa%modi;
        }
        if(b==0)++ans;
    }
    cout<<ans<<endl;
}
