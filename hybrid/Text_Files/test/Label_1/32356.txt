import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s = in.nextLine();
        int[] a = new int[3];
        for (char c : s.toCharArray()) {
            if (c == '+')
                continue;
            int x = c - '0';
            ++a[x - 1];
        }
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < a[i]; j++) {
                System.out.print(i + 1);
                if (j < a[i] - 1)
                    System.out.print('+');
            }
        }
        System.out.println();
    }
}
