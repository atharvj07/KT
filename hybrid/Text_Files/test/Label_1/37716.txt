def min_operations(a, b):
    def factorize(x):
        cnt = {2: 0, 3: 0, 5: 0}
        for f in cnt.keys():
            while x % f == 0:
                x //= f
                cnt[f] += 1
        return cnt, x
    
    cnt_a, rem_a = factorize(a)
    cnt_b, rem_b = factorize(b)
    
    if rem_a != rem_b:
        return -1
    
    return abs(cnt_a[2] - cnt_b[2]) + abs(cnt_a[3] - cnt_b[3]) + abs(cnt_a[5] - cnt_b[5])

a, b = map(int, input().split())
print(min_operations(a, b))
