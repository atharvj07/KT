n, q = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))

prefix_sum = [0] * (n+1)
for i in range(1, n+1):
    prefix_sum[i] = prefix_sum[i-1] + a[i-1]

total_arrows = 0
for arrows in k:
    total_arrows += arrows
    if total_arrows >= prefix_sum[-1]:
        total_arrows = 0
        print(n)
    else:
        low, high = 0, n
        while low < high:
            mid = (low + high) // 2
            if prefix_sum[mid] < total_arrows:
                low = mid + 1
            else:
                high = mid
        print(n - low)
