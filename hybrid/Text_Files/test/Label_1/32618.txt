import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (sc.hasNext()) {
            int n = sc.nextInt(), s = sc.nextInt();
            int[][] dp = new int[n + 1][s + 1];
            dp[0][0] = 1;

            for (int i = 1; i <= n; i++) {
                for (int j = 0; j <= s; j++) {
                    dp[i][j] = dp[i - 1][j];
                    if (j >= i) dp[i][j] += dp[i - 1][j - i];
                }
            }

            System.out.println(dp[n][s]);
        }
    }
}
