import sys

mod = 10 ** 9 + 7  # mod

n = int(sys.stdin.readline())
a = [int(i) for i in sys.stdin.readline().split()]


def fact(n):
    ans = 1
    for i in range(1, n + 1):
        ans = (ans * i) % mod
    return ans


def pow(x, p):
    if p == 0:
        return 1
    if p == 1:
        return x
    if p % 2 == 0:
        t = pow(x, p // 2) % mod
        return (t * t) % mod
    else:
        t = pow(x, p - 1) % mod
        return (t * x) % mod


ans = 0
for i in range(0, n):
    cnt = 0
    for j in range(i, n):
        cnt += (a[j] == a[i])
    ans += pow(2, cnt - 1) % mod
ans -= 1
ans = (ans + mod) % mod
print(ans)
