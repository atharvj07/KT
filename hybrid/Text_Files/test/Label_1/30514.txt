import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        int b = input.nextInt();
        int s = input.nextInt();
        input.close();
        if (Math.abs(a) + Math.abs(b) > s || (Math.abs(a) + Math.abs(b) - s) % 2 != 0) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
        }
    }
}
