#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
#include <set>
using namespace std;
typedef long long ll;

int main() {
    ll n; cin >> n;
    string a, b; cin >> a >> b;
    vector<ll> pre(n, -1), suf(n, -1);
    for(ll i = 0; i < n; ++i) {
        if(a[i] == '1') {
            if(i == 0) pre[i] = 0;
            else pre[i] = pre[i - 1] + 1;
        }
        if(b[i] == '1') {
            if(i == 0) suf[i] = n - 1;
            else suf[i] = suf[i - 1] - 1;
        }
    }
    ll ans = n;
    for(ll i = 0; i < n; ++i) {
        if(pre[i] == -1 || suf[i] == -1) continue;
        ans = min(ans, pre[i] + n - suf[i]);
    }
    if(ans == n) ans = -1;
    cout << ans << endl;
}
