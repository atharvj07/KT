#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(2 * m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 2 * m; j++) {
            cin >> a[i][j];
        }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][2 * j] == 1 || a[i][2 * j + 1] == 1) {
                ans++;
            }
        }
    }
    cout << ans << "\n";
    return 0;
}
