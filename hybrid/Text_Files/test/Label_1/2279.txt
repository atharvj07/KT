#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    string a;
    cin>>a;
    int cnt = 0;
    char b[n/2+1],c[n-n/2+1];
    for(int i=0;i<n/2;i++){
        if(a[i]=='?'){
            b[i]='0';
            cnt++;
        }
        else{
            b[i]=a[i];
        }
    }
    for(int i=n/2;i<n;i++){
        if(a[i]=='?'){
            c[i-n/2]='0';
            cnt++;
        }
        else{
            c[i-n/2]=a[i];
        }
    }
    int sum1=0,sum2=0;
    for(int i=0;i<n/2;i++){
        sum1+=(b[i]-'0');
    }
    for(int i=0;i<n-n/2;i++){
        sum2+=(c[i]-'0');
    }
    if(cnt%2==0){
        if(sum1==sum2){
            cout<<"Bicarp"<<endl;
        }
        else{
            cout<<"Monocarp"<<endl;
        }
    }
    else{
        if(sum1==sum2){
            cout<<"Monocarp"<<endl;
        }
        else{
            cout<<"Bicarp"<<endl;
        }
    }
    return 0;
}
