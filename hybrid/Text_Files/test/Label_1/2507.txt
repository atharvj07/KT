#include <bits/stdc++.h>

using namespace std;

const int MX = 1e5 + 5;

int n, k;
set<int> need;
vector<int> adj[MX], ts;
int indeg[MX];

void topo() {
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        if (!indeg[i]) q.push(i);
    }
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        ts.push_back(u);
        for (int v : adj[u]) {
            if (--indeg[v] == 0) {
                q.push(v);
            }
        }
    }
}

int main() {
    cin >> n >> k;
    for (int i = 0; i < k; ++i) {
        int x;
        cin >> x;
        need.insert(x);
    }
    for (int i = 1; i <= n; ++i) {
        int m;
        cin >> m;
        while (m--) {
            int x;
            cin >> x;
            adj[x].push_back(i);
            ++indeg[i];
        }
    }
    topo();
    if (ts.size() < n) {
        return puts("-1"), 0;
    }
    int ans = 0;
    for (int x : ts) {
        if (need.count(x)) {
            ++ans;
            need.erase(x);
        }
    }
    cout << ans << endl;
    for (int x : ts) {
        if (need.count(x)) {
            cout << x << ' ';
        }
    }
    return 0;
}
