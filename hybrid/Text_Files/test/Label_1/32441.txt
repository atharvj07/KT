import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int k = input.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = input.nextInt();
    }
    int ans = k / a[0];
    for (int i = 1; i < n; i++) {
      ans = Math.min(ans, (k + a[i] - 1) / a[i]);
    }
    System.out.println(ans);
  }
}
