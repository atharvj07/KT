def main():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    pos = {}
    for i in range(k):
        pos[a[i]] = i
    p = [0 for i in range(n)]
    q = []
    for i in range(n, 0, -1):
        if i in pos:
            p[pos[i]] = i
            continue
        if len(q) > 0 and q[-1] > i:
            return -1
        q.append(i)
    p[k:n] = q[::-1]
    print(*p)

main()
