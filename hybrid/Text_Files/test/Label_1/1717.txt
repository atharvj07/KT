#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
using ll = long long;

int32_t main() {
  ll n, u;
  cin >> n >> u;
  vector<ll> e(n);
  for (ll i = 0; i < n; i++) { cin >> e[i]; }
  double ans = -1;
  for (ll i = 0; i < n; i++) {
    for (ll j = i + 1; j < n; j++) {
      if (e[j] - e[i] > u) { break; }
      ll k = lower_bound(e.begin() + j + 1, e.end(), e[j] + e[j] - e[i]) - e.begin();
      if (k < n && e[k] - e[i] <= u) {
        ans = max(ans, double(e[k] - e[j]) / double(e[k] - e[i]));
      }
    }
  }
  cout << fixed << setprecision(9) << ans << endl;
  return 0;
}
