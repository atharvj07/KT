n, m = map(int, input().split())
discs = list(map(int, input().split()))

# Initialize towers
towers = [[] for _ in range(m + 1)]
for i, tower_index in enumerate(discs, start=1):
    towers[tower_index].append(i)

# Function to calculate difficulty
def calculate_difficulty(towers):
    difficulty = 0
    for tower in towers:
        for i in range(len(tower) - 1):
            if tower[i] + 1 != tower[i + 1]:
                difficulty += 1
    return difficulty

# Initial difficulty
print(calculate_difficulty(towers))

# Processing queries
for _ in range(m - 1):
    a, b = map(int, input().split())

    # Merge towers a and b
    if len(towers[a]) < len(towers[b]):
        a, b = b, a
    towers[a].extend(towers[b])
    towers[b] = []

    # Recalculate difficulty
    print(calculate_difficulty(towers))
