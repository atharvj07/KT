import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Point[] p = new Point[n];
        for (int i = 0; i < n; i++) {
            p[i] = new Point(sc.nextInt(), sc.nextInt(), i + 1);
        }
        Arrays.sort(p, (a, b) -> a.x - b.x);
        for (int i = 0; i < n - 2; i++) {
            for (int j = i + 1; j < n - 1; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (isTriangle(p[i], p[j], p[k])) {
                        System.out.println(p[i].ind + " " + p[j].ind + " " + p[k].ind);
                        return;
                    }
                }
            }
        }
    }

    static boolean isTriangle(Point p1, Point p2, Point p3) {
        int x1 = p1.x, y1 = p1.y;
        int x2 = p2.x, y2 = p2.y;
        int x3 = p3.x, y3 = p3.y;
        int s1 = (x2 - x1) * (y3 - y1) - (x3 - x1) * (y2 - y1);
        int s2 = (x3 - x2) * (y1 - y2) - (x1 - x2) * (y3 - y2);
        return s1 * s2 > 0;
    }

    static class Point {
        int x, y, ind;

        Point(int x, int y, int ind) {
            this.x = x;
            this.y = y;
            this.ind = ind;
        }
    }
}
