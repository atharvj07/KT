#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll INF=1e18+1;

int main() {
    ll n,k; cin>>n>>k;
    vector<ll> p(n),c(n);
    for (ll i=0; i<n; i++) cin>>p[i];
    for (ll i=0; i<n; i++) cin>>c[i];
    sort(p.begin(),p.end());
    vector<pair<ll,ll>> ranges;
    for (ll i=0; i<n; i++) {
        ll pos=distance(p.begin(),lower_bound(p.begin(),p.end(),p[i]+1));
        ranges.push_back({max(0ll,pos-(k+1)),pos-1});
    }
    vector<ll> pr(n),nx(n);
    for (ll i=0; i<n; i++) {
        if (i==0) pr[i]=-1;
        else pr[i]=ranges[i-1].second;
        if (i==n-1) nx[i]=n;
        else nx[i]=ranges[i+1].first;
    }
    vector<ll> dp(n,0);
    for (ll i=n-1; i>=0; i--) {
        dp[i]=c[i];
        if (i==0) continue;
        dp[i]=max(dp[i],dp[nx[i]]);
        if (pr[i]!=-1) dp[i]=max(dp[i],c[i]+dp[pr[i]]);
    }
    for (ll i=0; i<n; i++) cout<<dp[i]<<' '; cout<<endl;
}
