n = int(input())
a = []
b = []
for _ in range(n):
    l,r = map(int,input().split())
    a.append(l)
    b.append(r)
l,r = sorted(zip(a,b))
l2,r2 = zip(*sorted(zip(b,a)))
i = 0
j = 0
flag = 0
while i < n:
    while j < n and l2[j] <= l[i]:
        j += 1
    if j < n and r2[j] >= r[i]:
        print(i+1,j+1)
        flag = 1
        break
    i += 1
if not flag:
    print(-1,-1)
