a1,b1,a2,b2,l,r=map(int,input().split());ans=0
#solve b1-b2
if b1>b2:
    b1,b2=b2,b1;a1,a2=a2,a1
if a1*l+b1<=a2*l+b2:
    ans=l
else:
    ans=((a2*l+b2)-(b1-b2))//a1+l
    if ans!=((a2*l+b2)-(b1-b2))//a1:ans+=1
if a1*r+b1<=a2*r+b2:
    ans=r-ans+1
else:
    ans=(((a2*r+b2)-(b1-b2))//a1)-((a2*l+b2)-(b1-b2))//a1
    if ans!=(((a2*r+b2)-(b1-b2))//a1)-((a2*l+b2)-(b1-b2))//a1:ans+=1
print(ans)
