import sys

n, m = map(int, sys.stdin.readline().split())
e = list(map(int, sys.stdin.readline().split()))
d = [0] * n
for _ in range(m):
    u, v = map(int, sys.stdin.readline().split())
    d[v] += 1
q = []
for i in range(n):
    if d[i] == 0:
        q.append(i)
ans = 0
while q:
    nq = []
    need = []
    for i in q:
        if e[i] == 1:
            need.append(i)
        for j in range(n):
            if d[j] != 0 and d[j] - 1 == 0:
                nq.append(j)
            d[j] -= int(e[i] == e[j])
    if need:
        ans += 1
    q = nq
print(ans)
