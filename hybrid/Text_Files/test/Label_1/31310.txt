import java.util.*;

public class Main {
    static final int INF = (int) 2e9;
    static int[] dx = {-1, 0, 1, 0};
    static int[] dy = {0, 1, 0, -1};

    static class Point {
        int x, y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static int dist(Point a, Point b) {
        return Math.abs(a.x - b.x) + Math.abs(a.y - b.y);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            points[i] = new Point(in.nextInt(), in.nextInt());
        }
        int[][] dp = new int[n][1 << n];
        int[] path = new int[n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], INF);
            dp[i][1 << i] = 0;
        }
        for (int mask = 0; mask < (1 << n); mask++) {
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    for (int j = 0; j < n; j++) {
                        if (i == j || (mask & (1 << j)) != 0) {
                            continue;
                        }
                        dp[j][mask | (1 << j)] = Math.min(dp[j][mask | (1 << j)], dp[i][mask] + dist(points[i], points[j]));
                    }
                }
            }
        }
        int mask = (1 << n) - 1;
        int cur = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n; j++) {
                if (j == cur || (mask & (1 << j)) == 0) {
                    continue;
                }
                if (dp[j][mask] == dp[cur][mask] + dist(points[cur], points[j])) {
                    cur = j;
                    path[i] = j + 1;
                    mask ^= 1 << j;
                    break;
                }
            }
        }
        path[n - 1] = 1;
        for (int i = 0; i < n; i++) {
            System.out.print(path[i] + " ");
        }
        System.out.println();
    }
}
