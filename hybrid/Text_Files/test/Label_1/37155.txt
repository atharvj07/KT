import sys
from collections import defaultdict
from heapq import heappop, heappush
n = int(input())
c = []
for i in range(3):
    c.append([int(x) for x in input().split()])
e = defaultdict(list)
for l in sys.stdin:
    v1, v2 = map(int, l.split())
    e[v1].append(v2)
    e[v2].append(v1)
vis = [0] * (n + 1)
ans = []
def dfs(v, pr, col):
    ans.append((v, col))
    vis[v] = 1
    for u in e[v]:
        if vis[u] == 0: dfs(u, v, (col + 1) % 3)
        elif u != pr and col == (vis[u] + 1) % 3: return False
    return True
flag = False
min_cost = 1e18
for i in range(1, n + 1):
    if vis[i] == 0:
        ans = []
        if dfs(i, -1, 0) == True:
            flag = True
            res = 0
            for v, col in ans: res += c[col][v - 1]
            if res < min_cost:
                min_cost = res
                res_ans = ans
if flag:
    print(min_cost)
    for v, col in res_ans: print(col + 1, end=' ')
else:
    print(-1)
