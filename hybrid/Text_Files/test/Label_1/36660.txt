from bisect import bisect_left, insort_left

n = int(input())
lst = [0] + list(map(int, input().split()))
m = int(input())
ans = [0] * m
for j, (k, pos) in enumerate(map(lambda x: map(int, x.split()), [input() for _ in range(m)])):
    ind = bisect_left(lst, lst[-1] - (lst[k] - lst[pos]))
    if ind >= k: ans[j] = lst[ind + pos - k]
print(*ans, sep="\n")
