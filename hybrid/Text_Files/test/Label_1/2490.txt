#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e5+5;
int n, m, f[MAXN], a[MAXN];
int main(){
    cin >> n >> m;
    string s;
    cin >> s;
    for(int i = 0; i < n; i ++){
        if(s[i]=='.') a[i] = 1;
    }
    for(int i = 0; i < n; i ++){
        if(a[i]){
            if(a[i-1]){
                f[i] = f[i-1];
            }else f[i] = f[i-1]+1;
        }else f[i] = f[i-1];
    }
    for(int i = 0; i < m; i ++){
        int x;
        char c;
        cin >> x >> c;
        x--;
        if(s[x]==c){
            cout << f[n-1] << '\n';
            continue;
        }
        s[x] = c;
        if(c=='.') a[x] = 1;
        else a[x] = 0;
        if(a[x-1]){
            if(a[x]){
                f[x] = f[x-1];
            }else f[x] = f[x-1]+1;
        }else f[x] = f[x-1];
        cout << f[n-1] << '\n';
    }
    return 0;
}
