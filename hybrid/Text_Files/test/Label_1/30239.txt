import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {
    static boolean[] isPlayed;
    static List<Integer> ans;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] cards = new int[n];
            isPlayed = new boolean[14];
            ans = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                cards[i] = sc.nextInt();
            }
            solve(cards, 0, 0);
            if (ans.size() == n) {
                for (int i = 0; i < n; i++) {
                    System.out.print(ans.get(i) + " ");
                }
                System.out.println();
            } else {
                System.out.println("No");
            }
        }
    }

    static void solve(int[] cards, int sum, int pos) {
        if (pos == cards.length) {
            return;
        }
        for (int i = 0; i < cards.length; i++) {
            if (isPlayed[cards[i]]) {
                continue;
            }
            if (pos == 0 || sum % cards[i] == 0) {
                ans.add(cards[i]);
                isPlayed[cards[i]] = true;
                solve(cards, sum + cards[i], pos + 1);
                isPlayed[cards[i]] = false;
                ans.remove(ans.size() - 1);
            }
        }
    }
}
