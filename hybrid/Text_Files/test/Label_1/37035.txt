_ = int(input())
i = input()
h, m = map(int, i.split(':'))
if h < (12 if _ == 12 else 24) and int(m / 10) == 0:
    print("%02d:0%d" % (h, m))
elif h == 24:
    print("00:00")
else:
    if _:
        h = h if h < 12 else 1 if h == 12 else (h % 12) + 1
    m = int(m / 10) * 10 + m % 10
    print("%02d:%02d" % (h, m))
