def min_operations_to_good_string(s):
    operations = 0
    i = 0
    while i < len(s) - 2:
        if s[i:i+3] == "010" or s[i:i+3] == "101":
            operations += 1
            i += 3
        else:
            i += 1
    return operations

t = int(input().strip())
for _ in range(t):
    s = input().strip()
    print(min_operations_to_good_string(s))
