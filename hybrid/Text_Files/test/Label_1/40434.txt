n = int(input())
s = input()
k = 0
cs = sum(1 for i in s if i == '1')
z = 0
ans = []
for i in range(n):
    z += (s[i] == '0')
    ones = cs - (s[i] == '1')
    if ones == z:
        if k > 0:
            ans.append(s[i - k : i])
        k = i + 1
if k > 0:
    ans.append(s[k:])
print(len(ans))
print(*ans)
