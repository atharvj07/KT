n = int(input())
a = list(map(int, input().split()))
if a == sorted(a):
    print("no")
elif a[1:] == sorted(a[1:]):
    print("yes\n1", n)
else:
    for i in range(1, n - 1):
        if a[i - 1] > a[i] or a[i] > a[i + 1]:
            print("yes\n", i, i + 1)
            break
    else:
        print("no")
