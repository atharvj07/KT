#include <bits/stdc++.h>
#define N 1000001
using namespace std;
typedef long long ll;

int n;
ll a[N], f[N];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    sort(a + 1, a + n + 1);
    ll tot = 0;
    for (int i = 1; i <= n; i++) {
        if (a[i] != a[i - 1]) f[i] = f[i - 1] + 1;
        else f[i] = f[i - 1];
        tot += f[i];
    }
    double ans = tot * 2.0 / n / n;
    printf("%.6f\n", ans);
    return 0;
}
