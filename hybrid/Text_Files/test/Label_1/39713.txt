n = int(input())
h = list(map(int, input().split()))

max_height_right = 0
result = []

for i in range(n-1, -1, -1):
    if h[i] > max_height_right:
        result.append(0)
    else:
        result.append(max_height_right - h[i] + 1)
    max_height_right = max(max_height_right, h[i])

print(*result[::-1])
