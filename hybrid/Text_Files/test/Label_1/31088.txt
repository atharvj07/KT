import java.util.*;

class Main {
    static class Edge {
        int u, v, w;

        public Edge(int u, int v, int w) {
            this.u = u;
            this.v = v;
            this.w = w;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] g = new int[n + 1][n + 1];
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            g[u][v] = g[v][u] = w;
            edges.add(new Edge(u, v, w));
        }
        int xor = 0;
        for (Edge edge : edges) {
            xor ^= edge.w;
        }
        int ans = 0;
        if (xor == 0) {
            for (Edge edge : edges) {
                ans += edge.w;
            }
        } else {
            for (Edge edge : edges) {
                ans += Math.min(edge.w, xor);
            }
        }
        System.out.println(ans);
    }
}
