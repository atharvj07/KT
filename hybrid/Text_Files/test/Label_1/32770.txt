import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        Arrays.sort(a);
        int ans = 0;
        int i = 0;
        int cur = 0;
        while (cur < k) {
            if (i < n && a[i] <= k - cur) {
                cur += a[i];
                ++i;
            }
            ++ans;
        }
        System.out.println(ans);
    }
}
