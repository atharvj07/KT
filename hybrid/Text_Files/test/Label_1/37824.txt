def find_max_blocks(m):
    def max_blocks_and_volume(m):
        blocks, volume, a = 0, 0, int(m**(1/3))
        while a > 0:
            a_cubed = a**3
            if volume + a_cubed <= m:
                volume += a_cubed
                blocks += 1
            else:
                a -= 1
                continue
            a = int(((m - volume) ** (1/3)))
        return blocks, volume

    return max_blocks_and_volume(m)

m = int(input())
print(*find_max_blocks(m))
