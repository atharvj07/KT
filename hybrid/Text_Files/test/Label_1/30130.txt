import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong(), k = in.nextLong();
        int[] a = new int[(int)k];
        for (int i = 0; i < k; i++) {
            a[i] = in.nextInt();
        }
        int type = 1, boxes = 0;
        for (int i = 0; i < k; i++) {
            if (a[i] > a[type - 1]) {
                type = i + 1;
                boxes = 0;
            }
            boxes += (n / a[i]);
            n %= a[i];
        }
        System.out.println(type + " " + boxes);
    }
}
