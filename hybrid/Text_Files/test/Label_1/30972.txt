import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Get input
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        char[][] grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = input.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = line.charAt(j);
            }
        }

        // Find the minimum number of changes required to make the table nice
        int minChanges = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int numChanges = 0;
                // Check the 2x2 square around the current cell
                for (int ii = i; ii < i + 2 && ii < n; ii++) {
                    for (int jj = j; jj < j + 2 && jj < m; jj++) {
                        if (ii == i && jj == j) continue;
                        if (grid[ii][jj] == grid[i][j]) numChanges++;
                    }
                }
                // Update the minimum number of changes required
                minChanges = Math.min(minChanges, numChanges);
            }
        }

        // Make the minimum number of changes to make the table nice
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int numChanges = 0;
                // Check the 2x2 square around the current cell
                for (int ii = i; ii < i + 2 && ii < n; ii++) {
                    for (int jj = j; jj < j + 2 && jj < m; jj++) {
                        if (ii == i && jj == j) continue;
                        if (grid[ii][jj] == grid[i][j]) numChanges++;
                    }
                }
                // Change the current cell if necessary
                if (numChanges > 0) {
                    Set<Character> set = new HashSet<>();
                    set.add(grid[i][j]);
                    for (int ii = i; ii < i + 2 && ii < n; ii++) {
                        for (int jj = j; jj < j + 2 && jj < m; jj++) {
                            if (ii == i && jj == j) continue;
                            set.add(grid[ii][jj]);
                        }
                    }
                    char[] chars = new char[]{'A', 'G', 'C', 'T'};
                    for (char c : chars) {
                        if (!set.contains(c)) {
                            grid[i][j] = c;
                            break;
                        }
                    }
                }
            }
        }

        // Print the updated table
        for (char[] row : grid) {
            for (char c : row) {
                System.out.print(c);
            }
            System.out.println();
        }
    }
}
