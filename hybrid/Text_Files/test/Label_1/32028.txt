import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        sc.nextLine();
        String s = sc.nextLine();
        char[] digits = s.toCharArray();
        for (int i = 0; i < n && k > 0; i++) {
            if (i > 0 && digits[i] == '0' && digits[i - 1] != '0') continue;
            if (i == 0 && digits[i] != '1') {
                digits[i] = '1';
                k--;
                continue;
            }
            for (char d = '0'; d <= '9' && k > 0; d++) {
                if (d < digits[i]) {
                    digits[i] = d;
                    k--;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(digits[i]);
        }
    }
}
