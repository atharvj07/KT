#include <bits/stdc++.h>
using namespace std;
int n, m, ans;
string s[105], pal;
int main() {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = n - 1; i >= 0; i--) {
        int l = 0, r = n - 1, p = 1;
        for (int j = 0; j < n; j++)
            if (i == j) continue;
            else if (s[i] + s[j] == s[j] + s[i]) continue;
            else {
                p = 0;
                break;
            }
        while (l < r) {
            if (s[l] != s[r]) p = 0;
            l++, r--;
        }
        if (p) {
            pal = s[i];
            ans = m;
            break;
        }
    }
    for (int i = 0; i < n; i++) {
        int l = 0, r = n - 1, p = 1;
        for (int j = 0; j < n; j++)
            if (i == j) continue;
            else if (s[i] + s[j] + s[i] == s[i] + s[i] + s[j]) continue;
            else {
                p = 0;
                break;
            }
        while (l < r) {
            if (s[l] != s[r]) p = 0;
            l++, r--;
        }
        if (p) {
            pal = s[i] + s[i];
            ans = m + m;
            break;
        }
    }
    for (int a = 0; a < n; a++) {
        for (int b = 0; b < n; b++) {
            if (a == b) continue;
            for (int c = 0; c < n; c++) {
                if (a == c || b == c) continue;
                int l = 0, r = n - 1, p = 1;
                for (int j = 0; j < n; j++)
                    if (a == j || b == j || c == j) continue;
                    else if (s[a] + s[b] + s[c] + s[j] == s[j] + s[c] + s[b] + s[a])
                        continue;
                    else {
                        p = 0;
                        break;
                    }
                while (l < r) {
                    if (s[l] != s[r]) p = 0;
                    l++, r--;
                }
                if (p) {
                    pal = s[a] + s[b] + s[c];
                    ans = m + m + m;
                    break;
                }
            }
        }
    }
    cout << ans << endl;
    cout << pal << endl;
    return 0;
}
