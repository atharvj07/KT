from collections import deque

n = int(input())
a = tuple(map(int, input().split()))
if sum(a) < 2 * n - 2:
    print('NO')
else:
    print('YES', (n-1)//2)
    print(n-1)
    q = deque(range(1, n+1))
    while q:
        i = q.popleft()
        j = q.popleft()
        if a[i-1] > 0 and a[j-1] > 0:
            print(i, j)
            a[i-1] -= 1
            a[j-1] -= 1
            if a[i-1] > 0: q.append(i)
            if a[j-1] > 0: q.append(j)
