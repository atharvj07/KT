import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            String c = sc.next();
            int n = s.length();
            int m = c.length();
            if (n > m) {
                System.out.println("---");
                continue;
            }
            int[] pos = new int[26];
            for (int i = 0; i < n; i++) {
                pos[s.charAt(i) - 'A'] = i;
            }
            char[] ans = s.toCharArray();
            boolean found = false;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < c.charAt(i) - 'A'; j++) {
                    if (pos[j] > i) {
                        ans[i] = (char) (j + 'A');
                        ans[pos[j]] = s.charAt(i);
                        found = true;
                        break;
                    }
                }
                if (found) {
                    break;
                }
            }
            if (found) {
                System.out.println(new String(ans));
            } else {
                System.out.println("---");
            }
        }
    }
}
