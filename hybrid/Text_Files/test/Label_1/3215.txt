#include <bits/stdc++.h>
using namespace std;

int main() {
  int r, c;
  cin >> r >> c;
  vector<string> b(r);
  for (auto &x : b) cin >> x;
  int ans = 0;
  vector<bool> vis(c);
  for (int i = 0; i < c; i++) {
    if (vis[i]) continue;
    ans++;
    queue<int> q;
    q.push(i);
    vis[i] = 1;
    while (!q.empty()) {
      int u = q.front();
      q.pop();
      for (int j = 0; j < r; j++) {
        if (b[j][u] == 'B') {
          if (u > 0 && !vis[u - 1]) { vis[u - 1] = 1; q.push(u - 1); }
          if (u + 1 < c && !vis[u + 1]) { vis[u + 1] = 1; q.push(u + 1); }
          break;
        }
      }
    }
  }
  cout << ans << endl;
  return 0;
}
