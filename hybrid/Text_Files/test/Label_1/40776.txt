n = int(input())

house_price = 1234567
car_price = 123456
computer_price = 1234

for a in range(0, n // house_price + 1):
    for b in range(0, (n - a * house_price) // car_price + 1):
        c = (n - a * house_price - b * car_price) // computer_price
        if a * house_price + b * car_price + c * computer_price == n:
            print("YES")
            exit()
print("NO")
