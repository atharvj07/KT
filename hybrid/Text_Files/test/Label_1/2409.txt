#include <bits/stdc++.h>
using namespace std;
const int SZ = 2e5 + 5;
#define rep(i,a,b) for(int i = a; i < b; i++)
int a[SZ], n, l, x, y;
int main() {
    cin >> n >> l >> x >> y;
    int need[2];
    need[0] = (x ? 1 : 0);
    need[1] = (y > x ? 1 : 0);
    rep(i,0,n) cin >> a[i];
    rep(i,0,n-1) {
        int d = a[i+1] - a[i];
        if (d == x || d == y) need[0] = need[1] = 0;
    }
    cout << need[0] + need[1] << endl;
    if (need[0]) cout << x << ' ';
    if (need[1]) cout << y << ' ';
    return 0;
}
