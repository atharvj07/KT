#include <bits/stdc++.h>
using namespace std;

const int maxn = 1e2 + 5;
int p[maxn];

int main() {
    int tt;
    cin >> tt;
    while (tt--) {
        int n;
        cin >> n;
        vector<int> a(n), l(n);
        for (int& i : a) cin >> i;
        for (int& i : l) cin >> i;

        vector<int> ord;
        for (int i = 0; i < n; ++i)
            if (!l[i]) ord.push_back(i);
        sort(ord.begin(), ord.end(),
             [&](int x, int y) { return a[x] > a[y]; });
        vector<int> b(n);
        for (int i = 0, j = 0; i < n; ++i) {
            if (l[i])
                b[i] = a[i];
            else
                b[i] = a[ord[j++]];
        }
        bool ok = 1;
        p[0] = b[0];
        for (int i = 1; i < n; ++i) {
            p[i] = p[i - 1] + b[i];
            if (p[i] < 0) {
                ok = 0;
                break;
            }
        }
        if (!ok) {
            int mx = 0;
            for (int i = 0; i < n; ++i) {
                if (p[i] < 0) mx = i;
            }
            int nmx = -1;
            for (int i = 0; i < n; ++i) {
                if (p[i] < 0) nmx = max(nmx, i);
            }
            int r1 = 0, r2 = 0;
            for (int i = 0; i < mx; ++i) {
                if (b[i] < 0) r1 = i;
            }
            for (int i = mx + 1; i < n; ++i) {
                if (b[i] > 0) r2 = i;
            }
            if (r1 > 0 && r2 > 0) swap(b[r1], b[r2]);
        }
        for (int i : b) cout << i << ' ';
        cout << '\n';
    }
}
