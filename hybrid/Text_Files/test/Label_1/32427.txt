import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();

        boolean[] used = new boolean[n * m + 1];
        for (int i = 0; i < k; i++) {
            int s = sc.nextInt();
            if (s > Math.max(n, m))
                System.out.println("NO");
            else
                used[s] = true;
        }

        int l = sc.nextInt();
        for (int i = 0; i < l; i++) {
            int s = sc.nextInt();
            if (s > Math.max(n, m))
                System.out.println("NO");
            else
                used[s] = true;
        }

        boolean ok = true;
        for (int i = 1; i < Math.max(n, m); i++) {
            if (!used[i]) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
