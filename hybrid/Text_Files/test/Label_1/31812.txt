import java.util.Scanner;

public class VasyaAndTheCinema {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(); // Number of cars
        int k = scanner.nextInt(); // Number of gas stations
        int s = scanner.nextInt(); // Length of the road
        int t = scanner.nextInt(); // Time in which the film starts

        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        int minPrice = -1;
        for (Car car : cars) {
            if (canReachCinema(car, gasStations, s, t)) {
                minPrice = Math.min(minPrice, car.price);
            }
        }

        System.out.println(minPrice);
    }

    static boolean canReachCinema(Car car, int[] gasStations, int s, int t) {
        int time = 0;
        int fuel = car.capacity;
        for (int i = 0; i < gasStations.length; i++) {
            int distance = gasStations[i];
            time += Math.min(distance, fuel) * 2;
            fuel -= Math.min(distance, fuel);
            if (fuel == 0) {
                return false;
            }
            fuel = car.capacity;
        }
        int distance = s - gasStations[gasStations.length - 1];
        time += Math.min(distance, fuel) * 2;
        return time <= t;
    }

    static class Car {
        int price;
        int capacity;

        public Car(int price, int capacity) {
            this.price = price;
            this.capacity = capacity;
        }
    }
}
