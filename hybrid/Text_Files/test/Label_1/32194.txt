import java.io.*;
import java.math.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = in.nextInt();
        int ans = 0;
        Line[] lines = new Line[n];
        for (int i = 0; i < n; i++) {
            int x1 = in.nextInt();
            int y1 = in.nextInt();
            int x2 = in.nextInt();
            int y2 = in.nextInt();
            lines[i] = new Line(x1, y1, x2, y2);
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (lines[i].isHorizontal() && lines[j].isVertical() && lines[i].intersects(lines[j])) {
                    for (int k = j + 1; k < n; k++) {
                        for (int l = k + 1; l < n; l++) {
                            if (lines[k].isHorizontal() && lines[l].isVertical() && lines[k].intersects(lines[j]) && lines[k].intersects(lines[l])) {
                                ans++;
                            }
                        }
                    }
                }
            }
        }
        out.println(ans);
        out.close();
    }
}

class Line {
    int x1, y1, x2, y2;
    Line(int x1, int y1, int x2, int y2) {
        this.x1 = x1;
        this.y1 = y1;
        this.x2 = x2;
        this.y2 = y2;
    }
    boolean isHorizontal() {
        return y1 == y2;
    }
    boolean isVertical() {
        return x1 == x2;
    }
    boolean intersects(Line other) {
        if (this.isHorizontal() && other.isVertical()) {
            return other.x1 <= this.x1 && this.x1 <= other.x2 && other.y1 <= this.y1 && this.y1 <= other.y2;
        } else if (this.isVertical() && other.isHorizontal()) {
            return this.x1 <= other.x1 && other.x1 <= this.x2 && other.y1 <= this.y1 && this.y1 <= other.y2;
        }
        return false;
    }
}
