import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static int[] ia(String[] s) {
        int[] ret = new int[s.length];
        for (int i = 0; i < s.length; i++) ret[i] = Integer.parseInt(s[i]);
        return ret;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = ia(br.readLine().split(" "));
        int k = Integer.parseInt(br.readLine());
        int[] b = ia(br.readLine().split(" "));
        Map<Integer, List<Integer>> h = new HashMap<>();
        for (int i = 0; i < k; i++) {
            if (!h.containsKey(b[i])) h.put(b[i], new ArrayList<>());
            h.get(b[i]).add(i);
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (!h.containsKey(a[i])) {
                ok = false;
                break;
            }
            List<Integer> p = h.get(a[i]);
            if (p.size() == 0) {
                ok = false;
                break;
            }
            p.remove(0);
        }
        if (ok) {
            List<String> l = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                if (a[i] >= a[i + 1]) l.add((i + 1) + " R");
                else l.add((i + 1) + " L");
            }
            System.out.println("YES");
            for (String s : l) System.out.println(s);
        } else {
            System.out.println("NO");
        }
    }
}
