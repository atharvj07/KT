#include <bits/stdc++.h>
using namespace std;

struct edge {
  int u, v;
};

const int N = 2e5;
int n, u, v, d[N], sz[N], C;
bool vis[N];
vector<int> adj[N];
vector<edge> E;

void dfs(int u, int p) {
  sz[u] = 1;
  for (int v : adj[u]) {
    if (v == p) continue;
    dfs(v, u);
    sz[u] += sz[v];
    d[u] = max(d[u], sz[v]);
  }
  d[u] = max(d[u], C - sz[u]);
}

int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL);
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    E.push_back({u, v});
  }
  C = n;
  int ans = n - 1;
  for (int i = 1; i <= n; i++) {
    dfs(i, i);
    ans = min(ans, d[i]);
  }
  cout << ans << endl;
}
