import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] count = new int[4][2];
        while (sc.hasNext()) {
            double l = sc.nextDouble();
            double r = sc.nextDouble();
            if (l >= 1.1) {
                count[0][0]++;
            } else if (l >= 0.6) {
                count[1][0]++;
            } else if (l >= 0.2) {
                count[2][0]++;
            } else {
                count[3][0]++;
            }
            if (r >= 1.1) {
                count[0][1]++;
            } else if (r >= 0.6) {
                count[1][1]++;
            } else if (r >= 0.2) {
                count[2][1]++;
            } else {
                count[3][1]++;
            }
        }
        for (int i = 0; i < 4; i++) {
            System.out.println(String.format("%d %d", count[i][0], count[i][1]));
        }
    }
}
