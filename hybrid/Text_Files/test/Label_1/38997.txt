from collections import Counter

n, k, m, a = map(int, input().split())
g = list(map(int, input().split()))

cnt = Counter(g)
f = [False for _ in range(n+1)]
for i in g:
    f[i] = True
    
for i in range(1, n+1):
    if i not in cnt or cnt[i]<k or (cnt[i]==k and not f[i]):
        print(3)
    elif cnt[i]>=k and f[i]:
        print(1)
    else:
        print(2)
