import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    int min = 0;
    int max = n - 1;
    int ans = 0;
    while (a[max] - a[min] > 0 && (n - max - 1) + (min) < k) {
      if (a[max] - a[max - 1] < a[min + 1] - a[min]) {
        a[max] -= 1;
        ans += 1;
        max -= 1;
      } else {
        a[min] += 1;
        ans += 1;
        min += 1;
      }
    }
    System.out.println(ans);
  }
}
