import java.util.*;

public class Main {
    static String s;

    static void solve(int n, int l, int k) {
        String[] words = new String[n];
        int p = 0;
        for (int i = 0; i < n; i++) {
            StringBuilder sb = new StringBuilder();
            int[] cnt = new int[26];
            for (int j = 0; j < l; j++) {
                if (i == k - 1)
                    cnt[s.charAt(p) - 'a']++;
                sb.append(s.charAt(p++));
            }
            if (i != k - 1)
                Arrays.fill(cnt, 0);
            words[i] = sb.toString();
            for (int j = 1; j < 26; j++)
                while (cnt[j] > 0) {
                    words[i] += (char) ('a' + j);
                    cnt[j]--;
                }
        }
        Arrays.sort(words);
        for (String word : words) {
            System.out.println(word);
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), l = in.nextInt(), k = in.nextInt();
        s = in.next();
        solve(n, l, k);
    }
}
