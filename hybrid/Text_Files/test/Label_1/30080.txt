import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x1 = new int[n];
        int[] y1 = new int[n];
        int[] x2 = new int[n];
        int[] y2 = new int[n];
        for (int i = 0; i < n; i++) {
            x1[i] = in.nextInt();
            y1[i] = in.nextInt();
            x2[i] = in.nextInt();
            y2[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (x1[i] == x2[i]) {
                for (int j = 0; j < n; j++) {
                    if (y1[j] == y2[j] && x1[j] != x2[j]) {
                        for (int k = 0; k < n; k++) {
                            if (x1[k] == x2[k] && y1[k] != y2[k] && x1[k] == x1[i] && y1[k] <= y1[i] && y2[k] >= y1[i]) {
                                for (int l = 0; l < n; l++) {
                                    if (y1[l] == y2[l] && x1[l] != x2[l] && y1[l] == y1[j] && x1[l] <= x1[j] && x2[l] >= x1[j]) {
                                        ans++;
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
