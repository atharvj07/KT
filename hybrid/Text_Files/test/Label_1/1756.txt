#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    string t;
    cin >> t;
    bool erase = 0, can = 1;
    int x = n / 2, sum1 = 0, sum2 = 0;
    for (int i = 0; i < x; i++) {
        if (t[i] == '?') {
            erase = 1;
            sum1 += n + i % 10;
        } else
            sum1 += t[i] - '0';
    }
    for (int i = x; i < n; i++) {
        if (t[i] == '?') {
            erase = 1;
            sum2 += n + i % 10;
        } else
            sum2 += t[i] - '0';
    }
    if (erase == 0) {
        if (sum1 == sum2)
            cout << "Bicarp" << endl;
        else
            cout << "Monocarp" << endl;
    } else {
        if (sum1 == sum2)
            cout << "Monocarp" << endl;
        else
            cout << "Bicarp" << endl;
    }
    return 0;
}
