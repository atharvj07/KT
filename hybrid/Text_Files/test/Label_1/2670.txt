#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> groups(m);
  for (int i = 0; i < m; i++) {
    int k;
    cin >> k;
    groups[i].resize(k);
    for (int j = 0; j < k; j++) {
      cin >> groups[i][j];
      if (groups[i][j] < 0) groups[i][j] = -groups[i][j] + n;
    }
  }
  for (auto& group : groups) {
    for (int i = 0; i < group.size(); i++) {
      for (int j = i + 1; j < group.size(); j++) {
        if (group[i] == group[j]) {
          cout << "YES" << endl;
          return 0;
        }
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
