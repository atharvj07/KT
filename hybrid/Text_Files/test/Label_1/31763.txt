import java.util.*;

public class Main {

    static double EPS = 1e-9;
    static double INF = 1e9;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int H = sc.nextInt();
        Point[] points = new Point[n];
        for (int i = 0; i < n; i++) {
            points[i] = new Point(sc.nextInt(), sc.nextInt());
        }
        sc.close();
        double ans = 0;
        for (int i = 0; i + 1 < n; i++) {
            double dist = points[i].dist(points[i + 1]);
            if (is_visible(points[i], points[i + 1], H, points[n - 1])) {
                ans += dist;
            } else {
                Point p = get_intersection(points[i], points[i + 1], points[n - 1], H);
                double d1 = points[i].dist(p);
                double d2 = p.dist(points[i + 1]);
                ans += d1 + d2;
            }
        }
        System.out.println(ans);
    }

    static Point get_intersection(Point p1, Point p2, Point p3, double h) {
        double px = p1.x;
        double py = p1.y;
        double vx = p2.x - p1.x;
        double vy = p2.y - p1.y;
        double y = p3.y + h;
        double x = (y - py) / vy * vx + px;
        return new Point(x, y);
    }

    static boolean is_visible(Point p1, Point p2, double H, Point p3) {
        if (Math.abs(p1.x - p2.x) < EPS) {
            return p1.y <= p3.y + H && p2.y <= p3.y + H;
        }
        double a = (p2.y - p1.y) / (p2.x - p1.x);
        double b = p1.y - a * p1.x;
        double y = a * p3.x + b;
        return y <= p3.y + H;
    }

    static class Point {
        double x, y;

        Point(double x, double y) {
            this.x = x;
            this.y = y;
        }

        double dist(Point p) {
            return Math.sqrt((x - p.x) * (x - p.x) + (y - p.y) * (y - p.y));
        }
    }
}
