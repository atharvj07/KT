def min_removals_to_make_sequence_good(n, arr):
    def dfs(arr, bit=30):
        if len(arr) <= 1 or bit < 0:
            return 0
        ones = [a for a in arr if a & (1 << bit)]
        zeros = [a for a in arr if not a & (1 << bit)]
        if not ones:
            return dfs(zeros, bit - 1)
        if not zeros:
            return dfs(ones, bit - 1)
        return 1 + min(dfs(ones, bit - 1), dfs(zeros, bit - 1))

    return dfs(arr)

n = int(input())
arr = list(map(int, input().split()))
print(min_removals_to_make_sequence_good(n, arr))
