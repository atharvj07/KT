#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n;
  std::cin >> n;
  std::vector<long long> b(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> b[i];
  }
  if (n < 3) {
    std::cout << 0 << std::endl;
    return 0;
  }
  std::sort(b.begin(), b.end());
  long long ans = 1e18;
  ans = std::min(ans, (b[n - 1] - b[1] + 1) / 2);
  ans = std::min(ans, (b[n - 2] - b[0] + 1) / 2);
  if (ans > 1e9) {
    std::cout << -1 << std::endl;
  } else {
    std::cout << ans << std::endl;
  }
  return 0;
}
