import sys

n, m = map(int, input().split())
g = [list(input()) for _ in range(n)]
ans = []

for x in range(n):
    for y in range(m):
        if g[x][y] == '*':
            sz = 1
            while x-sz >= 0 and x+sz < n and y-sz >= 0 and y+sz < m and \
                  g[x-sz][y] == g[x+sz][y] == g[x][y-sz] == g[x][y+sz] == '*':
                sz += 1
            ans.append((x, y, sz-1))
            for i in range(x-sz+1, x+sz):
                for j in range(y-sz+1, y+sz):
                    g[i][j] = '.'

print(len(ans))
for x, y, s in ans:
    print(x+1, y+1, s)
