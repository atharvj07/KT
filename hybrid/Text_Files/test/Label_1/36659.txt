import sys

n, k = [int(x) for x in input().split()]
chains = [[] for _ in range(k)]
for i in range(k):
    m, *nums = [int(x) for x in input().split()]
    chains[i] = nums

ans = 0
stack = []
for chain in chains:
    while chain:
        x = chain.pop()
        if stack and stack[-1] == x - 1:
            stack.pop()
            ans += 1
        else:
            stack.append(x)
ans += len(stack)
print(ans)
