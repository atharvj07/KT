import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int p[] = new int[n];
        int f[] = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            f[i] = in.nextInt();
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            boolean found = false;
            for (int j = 0; j < n; j++) {
                if (p[j] == i + 1 && f[j] == f[i] && Math.abs(i - j) % f[j] == 0) {
                    found = true;
                    swap(p, i, j);
                    break;
                }
            }
            if (!found) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }

    public static void swap(int[] p, int i, int j) {
        int temp = p[i];
        p[i] = p[j];
        p[j] = temp;
    }
}
