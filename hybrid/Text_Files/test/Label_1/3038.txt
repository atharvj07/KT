#include <algorithm>
#include <iostream>
#include <map>
#include <vector>
using namespace std;

int main() {
    int n, s;
    cin >> n >> s;
    map<int, int> sellOrders, buyOrders;
    while (n--) {
        char d;
        int p, q;
        cin >> d >> p >> q;
        if (d == 'B')
            buyOrders[p] += q;
        else
            sellOrders[p] += q;
    }
    vector<pair<int, int>> sellResult, buyResult;
    for (const auto& [price, volume] : sellOrders) {
        if (sellResult.size() == s) break;
        sellResult.push_back({price, volume});
    }
    reverse(sellResult.begin(), sellResult.end());
    for (const auto& [price, volume] : buyOrders) {
        if (buyResult.size() == s) break;
        buyResult.push_back({price, volume});
    }
    reverse(buyResult.begin(), buyResult.end());
    for (const auto& [price, volume] : sellResult) cout << "S " << price << " " << volume << '\n';
    for (const auto& [price, volume] : buyResult) cout << "B " << price << " " << volume << '\n';
    return 0;
}
