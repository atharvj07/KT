#include <bits/stdc++.h>
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
#define ll long long
#define ff first
#define ss second
#define endl "\n"
#define pb push_back

ll check(vector<ll>&mon,vector<pair<ll,ll>>&heroes)
{
    ll sz=mon.size();
    ll sz2=heroes.size();
    sort(begin(mon),end(mon));
    sort(begin(heroes),end(heroes));
    ll j=0,res=0;
    for(ll i=0;i<sz;i++)
    {
        
        if(mon[i]<=heroes[j].ff)
        {
            heroes[j].ss--;
            if(heroes[j].ss==0)
            {
                res++,j++;
                if(j==sz2)
                {
                    return res;
                }
            }
        }
        else
        {
            res=-1;
            break;
        }
    }
    return res;
}
int main() {
    ll t;
    cin>>t;
    while(t--)
    {
        ll n;
        cin>>n;
        vector<ll>mon(n);
        for(ll i=0;i<n;i++)
        cin>>mon[i];
        ll m;
        cin>>m;
        vector<pair<ll,ll>>heroes(m);
        for(ll i=0;i<m;i++)
        {
            cin>>heroes[i].ff>>heroes[i].ss;
        }
        cout<<check(mon,heroes)<<endl;
    }
    return 0;
}
