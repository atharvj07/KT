s=input()
flag=0
for i in s:
    if i=='z':
        flag=1
    if i=='a':
        for j in s[s.index(i):]:
            if j=='z':
                flag=1
            if j!=chr(ord(j)+1):
                flag=1
if flag:
    print(-1)
else:
    ans=[]
    ans.append(min(s))
    s=s[s.index(min(s))+1:]
    while s:
        if len(ans)>=26:
            break
        mi=s[0]
        ind=0
        for i in range(len(s)):
            if s[i]<mi:
                mi=s[i]
                ind=i
        if s[ind]==chr(ord(ans[-1])+1):
            ans.append(s[ind])
            s=s[ind+1:]
        else:
            s=s[ind:]
    if len(ans)==26:
        print(''.join(ans))
    else:
        print(-1)
