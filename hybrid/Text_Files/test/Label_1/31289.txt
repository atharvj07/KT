import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long m = sc.nextLong();
    sc.close();
    long volume = 1;
    long count = 0;
    while (volume * volume * volume <= m) {
      count++;
      volume++;
    }
    System.out.println(count + " " + ((volume - 1) * (volume - 1) * (volume - 1)));
  }
}
