import java.util.*;

public class Main {
    static int[] pars;
    static int[] ran;
    static ArrayList<int[]> fri;
    static ArrayList<int[]> dis;

    static int find(int x) {
        if (pars[x] == x)
            return x;
        return pars[x] = find(pars[x]);
    }

    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            if (ran[x] > ran[y])
                pars[y] = x;
            else {
                pars[x] = y;
                if (ran[x] == ran[y])
                    ran[y]++;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        pars = new int[n + 1];
        ran = new int[n + 1];
        fri = new ArrayList<>();
        dis = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            pars[i] = i;
            ran[i] = 0;
        }
        for (int i = 0; i < k; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            fri.add(new int[] { u, v });
            union(u, v);
        }
        int m = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            dis.add(new int[] { u, v });
        }
        boolean[] vis = new boolean[n + 1];
        int max = 0;

        for (int i = 0; i < n; i++) {
            if (vis[i])
                continue;
            int comp = find(i);
            int cnt = 0;
            boolean ok = true;
            for (int j = 0; j < n; j++) {
                if (find(j) == comp) {
                    for (int[] d : dis) {
                        if (d[0] == j) {
                            if (find(d[1]) == comp)
                                ok = false;
                        }
                        if (d[1] == j) {
                            if (find(d[0]) == comp)
                                ok = false;
                        }
                    }
                    cnt++;
                }
            }
            if (ok)
                max = Math.max(max, cnt);
            for (int j = 0; j < n; j++) {
                if (find(j) == comp)
                    vis[j] = true;
            }
        }
        System.out.println(max);
    }
}
