n = int(input())
prices = {('A',): float('inf'), ('B',): float('inf'), ('C',): float('inf'),
          ('A', 'B'): float('inf'), ('A', 'C'): float('inf'), ('B', 'C'): float('inf'),
          ('A', 'B', 'C'): float('inf')}

for _ in range(n):
    c, vitamins = input().split()
    c = int(c)
    vitamins = tuple(sorted(vitamins))
    prices[vitamins] = min(prices[vitamins], c)

min_cost = float('inf')
for a in [('A',), ('A', 'B'), ('A', 'C'), ('A', 'B', 'C')]:
    for b in [('B',), ('A', 'B'), ('B', 'C'), ('A', 'B', 'C')]:
        for c in [('C',), ('A', 'C'), ('B', 'C'), ('A', 'B', 'C')]:
            if len(set(a + b + c)) == 3:
                min_cost = min(min_cost, prices[a] + prices[b] + prices[c])

print(min_cost if min_cost != float('inf') else -1)
