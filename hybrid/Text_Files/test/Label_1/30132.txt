import java.util.*;

class Node {
    int num;
    int weight;
    Node parent;
    List<Node> children;

    Node(int num) {
        this.num = num;
        this.children = new ArrayList<>();
    }
}

class Main {
    static Node root;
    static int minApplesToRemove;

    private static int dfs(Node node) {
        node.weight = node.num;
        for (Node child : node.children) {
            node.weight += dfs(child);
        }
        if (node != root) {
            int diff = Math.abs(node.weight - (root.weight - node.weight));
            minApplesToRemove = Math.min(minApplesToRemove, diff);
        }
        return node.weight;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] apples = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            apples[i] = input.nextInt();
        }
        Map<Integer, Node> nodes = new HashMap<>();
        for (int i = 1; i <= n; i++) {
            nodes.put(i, new Node(apples[i]));
        }
        for (int i = 1; i < n; i++) {
            int parent = input.nextInt();
            int child = input.nextInt();
            nodes.get(parent).children.add(nodes.get(child));
            nodes.get(child).parent = nodes.get(parent);
        }
        root = nodes.get(1);
        minApplesToRemove = Integer.MAX_VALUE;
        dfs(root);
        System.out.println(minApplesToRemove);
    }
}
