from collections import defaultdict, deque
import queue
n,m=map(int,input().split())
s=[]
d=[]
c=[]
for i in range(m):
    p,q,r=map(int,input().split())
    s.append(p)
    d.append(q)
    c.append(r)
a=[0]*n
days=max(d)
g=defaultdict(list)
for i in range(m):
    g[s[i]].append((i,1))
    g[d[i]].append((i,0))
q=deque([(0,0)])
while q:
    day,mask=q.popleft()
    for p,tp in g[day]:
        if tp==1:
            if mask&(1<<p)==0:
                q.append((day+c[p],mask|(1<<p)))
        else:
            if mask&(1<<p)!=0:
                q.append((day+1,mask^(1<<p)))
if mask==(1<<m)-1:
    for i in range(days):
        ans=[0]*n
        for p,tp in g[i]:
            if tp==1:
                if mask&(1<<p)!=0:
                    ans[i]=p+1
            else:
                if mask&(1<<p)==0:
                    ans[i]=p+1
        a.extend(ans)
    print(*a)
else: print(-1)
