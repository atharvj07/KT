#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, m, tmp, ans = 0;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.rbegin(), a.rend());
    for (int i = 0; i < n; i++) {
        if (m > 0) {
            tmp = min(m, a[i]);
            m -= tmp;
            ans++;
        } else {
            break;
        }
    }
    cout << ans << endl;
    return 0;
}
