import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];                                  // remaining amount of soda
        int[] b = new int[n];                                  // bottle volume
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        long sum = 0;
        for (int i = 0; i < n; i++) {                           // calculate total amount of soda
            sum += a[i];
        }
        int k = 0;
        long t = 0;
        for (int i = 0; i < n; i++) {
            if (sum > 0) {
                k++;
                long need = b[i] - a[i];
                t += need;
                sum -= need;
            }
        }
        System.out.println(k + " " + t);
    }
}
