import java.util.Scanner;

public class MinimalByInclusionSegment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int l = 1, r = 1;
        int minL = -1, minR = -1;
        int distinct = 1;

        for (; r <= n; r++) {
            if (a[r - 1] == a[l - 1]) {
                continue;
            }

            distinct++;

            while (distinct > k && l <= r) {
                if (a[l - 1] == a[r - 1]) {
                    distinct--;
                }
                l++;
            }

            if (distinct == k && (minL == -1 || r - l + 1 < minR - minL + 1)) {
                minL = l;
                minR = r;
            }
        }

        System.out.println(minL + " " + minR);
    }
}
