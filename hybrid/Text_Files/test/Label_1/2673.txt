#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cntD = 0, cntR = 0;
    for (char c : s) cntD += (c == 'D');
    cntR = n - cntD;
    while (cntD != 1 && cntR != 1) {
        if (cntD > cntR) {
            cntD--;
            cntR -= 2;
        } else {
            cntD -= 2;
            cntR--;
        }
    }
    cout << (cntD == 1 ? 'D' : 'R');
    return 0;
}
