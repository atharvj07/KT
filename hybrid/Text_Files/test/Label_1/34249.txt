import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] ratios = new long[n][2];
        for (int i = 0; i < n; i++) {
            ratios[i][0] = sc.nextLong();
            ratios[i][1] = sc.nextLong();
        }
        long lcm = lcm(ratios[0][0], ratios[0][1]);
        for (int i = 1; i < n; i++) {
            lcm = lcm(lcm, lcm(ratios[i][0], ratios[i][1]));
        }
        System.out.println(lcm);
    }

    private static long lcm(long a, long b) {
        return a * b / gcd(a, b);
    }

    private static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
