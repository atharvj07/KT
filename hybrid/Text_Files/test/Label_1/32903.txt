import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    static int[] par;
    static int[] sz;
    static int n, m, k;
    static ArrayList<int[]> q;

    static int find(int x) {
        if (x != par[x]) {
            par[x] = find(par[x]);
        }
        return par[x];
    }

    static void unite(int x, int y) {
        x = find(x);
        y = find(y);
        if (sz[x] > sz[y]) {
            par[y] = x;
            sz[x] += sz[y];
        } else {
            par[x] = y;
            sz[y] += sz[x];
        }
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        par = new int[n + 1];
        sz = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            par[i] = i;
            sz[i] = 1;
        }
        q = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            unite(x, y);
            q.add(new int[]{x, y});
        }
        for (int i = m - 1; i >= 0; i--) {
            int[] u = q.get(i);
            int x = find(u[0]);
            int y = find(u[1]);
            par[x] = par[y];
            sz[y] -= sz[x];
        }
        for (int i = 1; i <= m; i++) {
            int mx = 0;
            for (int j = 1; j <= n; j++) {
                if (find(j) == find(1)) {
                    mx = Math.max(mx, sz[j]);
                }
            }
            if (mx >= k) {
                System.out.println(mx);
            } else {
                System.out.println(0);
            }
        }
    }
}
