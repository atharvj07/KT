import sys, math
input = sys.stdin.readline

for _ in range(int(input())):
    n = int(input())
    graph = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().split())
        graph[u].append(v)
        graph[v].append(u)
    q = [1]
    dist = [-1]*(n+1)
    dist[1] = 0
    while q:
        x = q.pop(0)
        for y in graph[x]:
            if dist[y] == -1:
                dist[y] = dist[x]+1
                q.append(y)
    max_dist = max(dist[1:])
    bad = []
    for i in range(2, n+1):
        if dist[i] == max_dist:
            bad.append(i)
    if len(bad) == 1:
        print(max_dist+1)
    else:
        print(max_dist)
