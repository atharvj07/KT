import sys
import copy

sys.stdin = open('input.txt', 'r')
board = [input() for _ in range(8)]

for _ in range(8):
    newBoard = copy.deepcopy(board)
    for i in range(8):
        newBoard[i] = newBoard[i][1:] + newBoard[i][0]

    if newBoard[0][0] == 'W':
        if all(newBoard[i][0] != newBoard[i - 1][0] for i in range(1, 8)) and \
           all(newBoard[i][j] != newBoard[i][j - 1] for i in range(8) for j in range(1, 8)):
            print('YES')
            break
    board = newBoard

else:
    print('NO')
