t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    min_val, max_val = float('inf'), -float('inf')
    max_gap = 0
    for i in range(n):
        if a[i] == -1:
            if i > 0 and a[i-1] != -1:
                min_val = min(min_val, a[i-1])
                max_val = max(max_val, a[i-1])
            if i < n-1 and a[i+1] != -1:
                min_val = min(min_val, a[i+1])
                max_val = max(max_val, a[i+1])
        elif i > 0 and a[i-1] != -1:
            max_gap = max(max_gap, abs(a[i] - a[i-1]))
    
    k = (min_val + max_val) // 2
    max_gap = max(max_gap, max_val - k, k - min_val)
    print(max_gap, k)
