from collections import Counter

n = int(input())
p = list(map(int, input().split()))
x = list(map(int, input().split()))
adj = [[] for _ in range(n)]

for i in range(1, n):
    adj[p[i - 1]].append(i)

cnt = Counter()
for i in range(n):
    if x[i]:
        cnt[i] = 1

def dfs(u, fup, to_del):
    if len(to_del) == 2:
        return 1
    ans = 0
    for v in adj[u]:
        if v == fup:
            continue
        ans += dfs(v, u, to_del + [u])
    return ans

print(dfs(0, 0, []))
