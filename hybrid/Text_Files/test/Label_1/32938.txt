import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        int[][] buses = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                buses[i][j] = in.nextInt();
            }
        }
        int[][] citizens = new int[m][3];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                citizens[i][j] = in.nextInt();
            }
        }
        int[] ans = new int[m];
        Arrays.fill(ans, -1);
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (buses[j][0] <= citizens[i][0] && citizens[i][1] <= buses[j][1] && citizens[i][2] <= buses[j][2]) {
                    ans[i] = j + 1;
                    break;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            System.out.print(ans[i] + " ");
        }
        System.out.println();
    }
}
