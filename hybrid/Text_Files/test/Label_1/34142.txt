import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner cin=new Scanner(System.in);
        int n=cin.nextInt();
        int[]d=new int[n-1];
        for(int i=0;i<n-1;i++){
            d[i]=cin.nextInt();
        }
        int a=cin.nextInt();
        int b=cin.nextInt();
        int ans=0;
        for(int i=a-1;i<b-1;i++){
            ans+=d[i];
        }
        System.out.println(ans);
    }
}
