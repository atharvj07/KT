#include <iostream>
#include <vector>
#include <set>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;

  vector<set<int>> adj(n + 1);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].insert(v);
    adj[v].insert(u);
  }

  int ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j : adj[i]) {
      if (j > i) {
        for (int k : adj[j]) {
          if (k > j && !adj[i].count(k)) {
            cout << i << " " << j << endl;
            cout << j << " " << k << endl;
            adj[i].insert(k);
            adj[k].insert(i);
            ans += 2;
          }
        }
      }
    }
  }

  cout << ans << endl;

  return 0;
}
