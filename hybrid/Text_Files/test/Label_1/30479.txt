import java.util.*;
import java.io.*;

public class SonyaAndMultiset {

    static class Multiset {
        private HashMap<Long, Integer> counts;

        public Multiset() {
            counts = new HashMap<>();
        }

        public void add(long value) {
            counts.put(value, counts.getOrDefault(value, 0) + 1);
        }

        public void remove(long value) {
            int count = counts.get(value);
            if (count == 1) {
                counts.remove(value);
            } else {
                counts.put(value, count - 1);
            }
        }

        public long countMatching(String pattern) {
            long mask = 1L << (pattern.length() - 1);
            long result = 0;
            for (long value : counts.keySet()) {
                boolean matches = true;
                for (int i = 0; i < pattern.length(); i++) {
                    if (pattern.charAt(i) == '1' && (value & mask) == 0) {
                        matches = false;
                        break;
                    }
                    mask >>= 1;
                }
                if (matches) {
                    result += counts.get(value);
                }
            }
            return result;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(reader.readLine());

        Multiset multiset = new Multiset();
        for (int i = 0; i < t; i++) {
            String[] parts = reader.readLine().split(" ");
            char type = parts[0].charAt(0);
            if (type == '+') {
                long value = Long.parseLong(parts[1]);
                multiset.add(value);
            } else if (type == '-') {
                long value = Long.parseLong(parts[1]);
                multiset.remove(value);
            } else if (type == '?') {
                String pattern = parts[1];
                long count = multiset.countMatching(pattern);
                System.out.println(count);
            }
        }
    }
}
