import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String n = scanner.next();
        scanner.close();
        int[] digits = new int[10];
        for (char c : n.toCharArray()) {
            digits[c - '0']++;
        }
        int count = 0;
        for (int i = 1; i <= 9; i++) {
            if (digits[i] > 0) {
                digits[i]--;
                count += (digits[0] + 1) * factorial(digits[0]);
                for (int j = 1; j <= 9; j++) {
                    count += digits[j] * factorial(digits[0] + digits[j]);
                }
                digits[i]++;
            }
        }
        System.out.println(count);
    }

    public static int factorial(int n) {
        if (n == 0) {
            return 1;
        } else {
            return n * factorial(n - 1);
        }
    }
}
