import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<String, Integer> ai = new HashMap<>();
        List<String> names = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String name = sc.next();
            names.add(name);
            ai.put(name, sc.nextInt());
        }
        if (!check(ai)) {
            System.out.println(-1);
            return;
        }
        int[] hi = new int[n];
        int[] used = new int[n + 1];
        for (String name : names) {
            int free = -1;
            for (int i = 1; i <= n; i++) {
                if (used[i] == 0) {
                    if (free < 0 || used[hi[free]] < used[i]) {
                        free = i;
                    }
                }
            }
            hi[names.indexOf(name)] = free;
            used[free] = ai.get(name) + 1;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(names.get(i) + " " + hi[i]);
        }
    }

    public static boolean check(Map<String, Integer> ai) {
        int sum = 0;
        for (Integer i : ai.values()) {
            sum += i;
        }
        return sum == ai.size() - 1;
    }
}
