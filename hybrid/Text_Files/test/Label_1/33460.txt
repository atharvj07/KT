import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] a) {
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            long n=sc.nextLong();
            long k=sc.nextInt();
            if(n%2==1 && k%2==0 || n%2==0 && k%2==1 || k>n){
                System.out.println("NO");
            }
            else{
                long x=(long)Math.ceil((double)n/k);
                if(x%2!=n%2)x++;
                System.out.print("YES ");
                for(long i=0;i<k;i++){
                    System.out.print(x+" ");
                    n-=x;
                }
                System.out.println();
            }
        }
    }
}
