import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int a = input.nextInt();
        input.close();
        int b = 10 - a % 10;
        while ((a + b) % 8 != 0) {
            b += 10;
        }
        System.out.println(b);
    }
}
