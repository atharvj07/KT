import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int flag = 0x0;
        for (int i = 0; i < n; i++) {
            flag |= sc.nextInt();
        }
        int ans = 0;
        for (int j = 0; j < 32; j++) {
            if ((flag & (0x1 << j)) > 0) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
