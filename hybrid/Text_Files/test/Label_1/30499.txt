import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = in.nextLong();
        }
        List<Long> bList = new ArrayList<>();
        for (long val : b) {
            bList.add(val);
        }
        Collections.sort(bList);
        long xor = bList.get(0);
        boolean possible = true;
        for (int i = 1; i < n; i++) {
            if (xor > bList.get(i)) {
                possible = false;
                break;
            }
            xor ^= bList.get(i);
        }
        if (!possible) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            for (long val : bList) {
                System.out.print(val + " ");
            }
            System.out.println();
        }
    }
}
