#include <bits/stdc++.h>
using namespace std;

int main(){
    int t; cin >> t;
    while(t--){
        int n; cin >> n;
        vector <int> a(n), b(n), pf(1000002, 0);
        for(int i = 0; i < n; i++){
            cin >> b[i];
            a[i] = (1 << b[i]);
        }
        for(int i = 0; i < n; i++){
            pf[a[i]]++;
        }
        int ans = 0;
        for(int i = 1; i <= 1000001; i++){
            if(pf[i] > 1){
                ans = 1;
            }
        }
        if(ans == 1){
            cout << "YES" << endl;
        }
        else{
            cout << "NO" << endl;
        }
    }
    return 0;
}
