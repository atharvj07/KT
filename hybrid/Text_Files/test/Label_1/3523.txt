#include <iostream>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<vector<int>> adj(n + 1);
  for (int i = 1; i < n; i++) {
    int a, b;
    cin >> a >> b;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  vector<int> deg(n + 1, 0);
  for (int i = 1; i <= n; i++) deg[i] = adj[i].size();
  int odd = 0;
  for (int i = 1; i <= n; i++) odd += deg[i] % 2;
  if (odd > 2) {
    cout << -1 << "\n";
    return 0;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) ans += deg[i] % 2;
  ans /= 2;
  cout << ans << "\n";
  return 0;
}
