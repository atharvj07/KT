import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), b = in.nextInt();
        in.close();
        int x = 1, y = 0, x1 = 0, y1 = 1, x2, y2;
        while (b > 0) {
            int q = a / b;
            x2 = x - q * x1;
            x = x1;
            x1 = x2;
            y2 = y - q * y1;
            y = y1;
            y1 = y2;
            a = b;
            b = a % b;
        }
        System.out.println(x + " " + y);
    }
}
