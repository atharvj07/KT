#include <bits/stdc++.h>
using namespace std;
int main(){
    int n,l,r;cin>>n>>l>>r;
    vector<int>a(n),b(n);
    for(int i=0;i<n;i++)cin>>a[i];
    for(int i=0;i<n;i++)cin>>b[i];
    map<int,int>ma;
    for(int i=0;i<n;i++)ma[a[i]]++,ma[b[i]]--;
    int cnt=0;
    for(auto it:ma){
        if(it.second){cnt=1;break;}
    }
    if(cnt){cout<<"LIE";return 0;}
    for(int i=0;i<n;i++){
        if(a[i]!=b[i]){
            if(i<l||i>r){cout<<"LIE";return 0;}
        }
    }
    cout<<"TRUTH";
}
