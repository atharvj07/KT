import sys

def solve(n, lst):
    d = {}
    for s in lst:
        s = s.lstrip('0')
        if len(s) < len(str(n)) or len(s) > len(str(n)): return False
        if s in d: return False
        if s[0] == '?': return False
        d[s] = True

    lkp = {}
    for s in lst:
        s = s.lstrip('0')
        t = ''
        for c in s:
            if c == '?': t += '0'
            else: t += c
        lkp[t] = s

    ans = []
    prev = 0
    s = ''.join(['0'] * (len(str(n))))
    for i in range(n):
        while lkp[s] != lst[i]:
            s = str(int(s) + 1)
            s = ''.join(['0'] * (len(str(n)) - len(s))) + s
        if int(lkp[s]) <= prev: return False
        prev = int(lkp[s])
        ans.append(lkp[s])
        s = str(int(s) + 1)

    for i in range(n):
        print(ans[i])

    return True

n = int(input())
lst = [input() for _ in range(n)]

if solve(n, lst):
    print('YES')
else:
    print('NO')
