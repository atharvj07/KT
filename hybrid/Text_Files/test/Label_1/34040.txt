import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        int q = sc.nextInt();
        int[] ans = new int[q + 1];
        SegmentTree st = new SegmentTree(a);
        ans[0] = st.query(0, n - 1) / 2;
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            int x = sc.nextInt();
            st.update(l, r, x);
            ans[i + 1] = st.query(0, n - 1) / 2;
        }
        for (int val : ans) System.out.println(val);
    }

    static class SegmentTree {
        int[] arr;
        int[] tree;

        SegmentTree(int[] arr) {
            this.arr = arr;
            tree = new int[arr.length * 4];
            build(0, arr.length - 1, 1);
        }

        void build(int start, int end, int index) {
            if (start == end) {
                tree[index] = arr[start];
                return;
            }
            int mid = (start + end) / 2;
            build(start, mid, index * 2);
            build(mid + 1, end, index * 2 + 1);
            tree[index] = Math.max(tree[index * 2], tree[index * 2 + 1]);
        }

        void update(int start, int end, int val) {
            update(0, arr.length - 1, start, end, val, 1);
        }

        void update(int treeStart, int treeEnd, int start, int end, int val, int index) {
            if (treeStart > end || treeEnd < start) return;
            if (treeStart >= start && treeEnd <= end) {
                tree[index] += val;
                return;
            }
            int mid = (treeStart + treeEnd) / 2;
            update(treeStart, mid, start, end, val, index * 2);
            update(mid + 1, treeEnd, start, end, val, index * 2 + 1);
            tree[index] = Math.max(tree[index * 2], tree[index * 2 + 1]);
        }

        int query(int start, int end) {
            return query(0, arr.length - 1, start, end, 1);
        }

        int query(int treeStart, int treeEnd, int start, int end, int index) {
            if (treeStart > end || treeEnd < start) return Integer.MIN_VALUE;
            if (treeStart >= start && treeEnd <= end) return tree[index];
            int mid = (treeStart + treeEnd) / 2;
            int leftQuery = query(treeStart, mid, start, end, index * 2);
            int rightQuery = query(mid + 1, treeEnd, start, end, index * 2 + 1);
            return Math.max(leftQuery, rightQuery);
        }
    }
}
