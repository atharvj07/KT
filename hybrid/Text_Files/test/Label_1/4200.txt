#include <cstdio>
#include <vector>
#include <queue>
using namespace std;
vector<int>g[110];
int a[110][110],n,m,d[110][110];
int dx[]={1,-1,0,0};
int dy[]={0,0,1,-1};
queue<int>q;
int main()
{
    scanf("%d%d",&n,&m);
    int player=0,mx=0;
    for (int i=1; i<=n; i++)
    {
        for (int j=1; j<=m; j++)
        {
            a[i][j]=++player;
            if (i==1 || i==n || j==1 || j==m)
            {
                q.push(a[i][j]),d[i][j]=1;
            }
        }
    }
    int ans=0;
    while (!q.empty())
    {
        int now=q.front(),x=now/(m+1),y=now%(m+1);
        q.pop();
        g[now].push_back(x);
        g[now].push_back(y);
        ans++;
        for (int i=0; i<4; i++)
        {
            int u=x+dx[i],v=y+dy[i];
            if (u>=1 && v>=1 && u<=n && v<=m && d[u][v]==0)
            {
                d[u][v]=1;
                q.push(a[u][v]);
            }
        }
    }
    if (ans%2==0) printf("Malvika\n");
    else printf("Akshat\n");
}
