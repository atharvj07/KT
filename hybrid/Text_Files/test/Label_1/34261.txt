import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int x = sc.nextInt();
        int[] a = new int[n + 1];

        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }

        Set<Integer> visited = new HashSet<>();
        Set<Integer> possiblePositions = new HashSet<>();

        for (int i = 1; i <= n; i++) {
            if (visited.contains(i)) {
                continue;
            }

            int curr = i;
            while (a[curr] != 0) {
                visited.add(curr);
                curr = a[curr];
            }

            possiblePositions.add(curr);
        }

        List<Integer> sortedPositions = new ArrayList<>(possiblePositions);
        Collections.sort(sortedPositions);

        for (int position : sortedPositions) {
            if (position == x) {
                continue;
            }

            System.out.println(position);
        }
    }
}
