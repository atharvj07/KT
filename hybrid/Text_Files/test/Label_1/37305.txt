def game_outcome(n, k, cards):
    if k == 1:
        return "once again"
    unique_cards = set(cards)
    if len(unique_cards) == 1:
        return "tokitsukaze"
    if k > n // 2:
        return "tokitsukaze"
    if cards[:k] == '0'*k or cards[:k] == '1'*k:
        if cards[k] != cards[0]:
            return "tokitsukaze"
    if cards[-k:] == '0'*k or cards[-k:] == '1'*k:
        if cards[-k-1] != cards[-1]:
            return "tokitsukaze"
    first_one = cards.find('1')
    last_one = cards.rfind('1')
    first_zero = cards.find('0')
    last_zero = cards.rfind('0')
    if (last_one - first_one < k) or (last_zero - first_zero < k):
        return "tokitsukaze"
    if (last_one - first_one + 1 <= k) and (last_zero - first_zero + 1 <= k):
        return "quailty"
    
    return "once again"

n, k = map(int, input().strip().split())
cards = input().strip()
print(game_outcome(n, k, cards))
