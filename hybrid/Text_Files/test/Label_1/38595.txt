def main():
    t=int(input())
    for _ in range(t):
        n,a,b=map(int, input().split())
        if (n/2)%2!=0 or a==b:
            print(-1)
            continue
        ans=[]
        ans.extend(range(a+1,b+1))
        ans.extend(range(1,a+1))
        ans.extend(range(b+1,n+1))
        print(*ans)

main()
