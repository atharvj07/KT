#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++)
    cin >> a[i];
  sort(a.begin(), a.end());
  int ans = 0;
  for (int l = 0; l < n; l++) {
    int r = l;
    while (r < n && a[r] <= m)
      r++;
    ans += max(0, r - l - 1);
  }
  cout << ans << endl;
  return 0;
}
