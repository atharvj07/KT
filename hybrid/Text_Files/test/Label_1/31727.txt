import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        int k = sc.nextInt();
        int t = sc.nextInt();
        BigInteger[] powK = new BigInteger[t + 1];
        powK[0] = powK[1] = BigInteger.ONE;
        for (int i = 2; i <= t; i++) {
            powK[i] = powK[i - 1].multiply(BigInteger.valueOf(2 * k + 1));
        }
        BigInteger ans = powK[t].add(powK[t].negate()).multiply(powK[t].subtract(BigInteger.ONE)).divide(BigInteger.valueOf(2)).mod(BigInteger.valueOf(1000000007));
        // There is a mistake in the formula in the statement, it should be ((2*k + 1)^t + (-2*k - 1)^t) * ((2*k + 1)^t - 1) / 2 % 1000000007
        System.out.println(ans);
    }
}
