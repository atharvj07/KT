import sys
input = sys.stdin.readline

n,m,c_l,c_e,v = list(map(int,input().split(' ')))
l=[0]+list(map(int,input().split(' ')))
e=[0]+list(map(int,input().split(' ')))
q = int(input())
for i in range(q):
    x1,y1,x2,y2 = list(map(int,input().split(' ')))
    ans = abs(x1-x2)
    if x1==x2:
        ans += abs(y1-y2)
    else:
        ans += min(abs(y1-l[c_l]),abs(y1-e[c_e]))
        ans += min(abs(y2-l[c_l]),abs(y2-e[c_e]))
        if c_e>0:
            ans = min(ans,abs(x1-x2)+1)
            ans = min(ans,abs(x1-x2)//v+1)
        if c_l>0:
            ans = min(ans,abs(x1-x2)+1)
            ans = min(ans,abs(x1-x2)//v+abs(y1-l[c_l]))
            ans = min(ans,abs(x1-x2)//v+abs(y2-l[c_l]))
    print(ans)
