#include <algorithm>
#include <vector>
#include <string>
#include <stack>
#include <iostream>

int main() {
    std::string s, t, u;
    std::cin >> s;
    std::stack<char> st;
    for (int i = (int)s.size() - 1; i >= 0; i--) {
        if (st.empty() || st.top() <= s[i]) {
            st.push(s[i]);
        }
        else {
            while (!st.empty() && st.top() > s[i]) {
                u += st.top();
                st.pop();
            }
            st.push(s[i]);
        }
    }
    while (!st.empty()) {
        u += st.top();
        st.pop();
    }
    std::cout << u << std::endl;
    return 0;
}
