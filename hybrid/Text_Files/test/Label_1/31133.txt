import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), m = sc.nextInt(), a = sc.nextInt();
        int[] g = new int[a];
        for (int i = 0; i < a; i++) g[i] = sc.nextInt();
        
        Arrays.sort(g);
        int[] cnt = new int[n + 1];
        for (int i = a - 1; i >= 0; i--) cnt[g[i]]++;
        
        int[] ans = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            if (cnt[i] >= k) ans[i] = 1;
            else if (cnt[i] + m - a >= k) ans[i] = 2;
            else ans[i] = 3;
        }
        
        for (int i = 1; i <= n; i++) System.out.print(ans[i] + " ");
    }
}
