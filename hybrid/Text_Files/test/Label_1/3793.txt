#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    long long ans = 0;
    long long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];
        ans = max(ans, min(sum, a[(i + 1) % n] + 1));
    }
    cout << ans << endl;
    return 0;
}
