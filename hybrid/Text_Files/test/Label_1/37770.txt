def max_permutation(a, b):
    a_str, b_str = str(a), str(b)
    if len(a_str) < len(b_str):
        return ''.join(sorted(a_str, reverse=True))
    a_sorted = sorted(a_str)
    for i in range(len(a_str)):
        for j in range(len(a_sorted)-1, -1, -1):
            if a_sorted[j] > b_str[i]:
                continue
            temp = a_sorted[j]
            if i == 0 and temp == '0':
                continue
            rest = a_sorted[:j] + a_sorted[j+1:]
            if all(b_str[i + k + 1] >= max(rest[k:]) for k in range(len(rest))):
                return a_str[:i] + temp + ''.join(sorted(rest, reverse=True))
    return ''.join(a_sorted)

a = int(input().strip())
b = int(input().strip())
print(max_permutation(a, b))
