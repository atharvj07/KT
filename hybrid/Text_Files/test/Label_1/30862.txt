import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] c = new int[n - 1];
        int[] s = new int[n - 1];
        int[] f = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            c[i] = sc.nextInt();
            s[i] = sc.nextInt();
            f[i] = sc.nextInt();
        }
        int[] ans = new int[n];
        ans[n - 1] = 0;
        for (int i = n - 2; i >= 0; i--) {
            int min = (s[i] + f[i] - 1) / f[i] * f[i] + c[i];
            ans[i] = ans[i + 1] + min;
        }
        for (int i = 0; i < n; i++) {
            System.out.println(ans[i]);
        }
    }
}
