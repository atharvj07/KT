#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<ll> a(n);
        for (auto &e : a) cin >> e;
        sort(a.rbegin(), a.rend());
        ll sum = 0;
        for (int i = 1; i < n; i++) sum += a[i] - a[0];
        cout << sum << '\n';
    }
    return 0;
}
