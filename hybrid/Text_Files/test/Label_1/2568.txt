#include <bits/stdc++.h>
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
using namespace std;
using namespace __gnu_pbds;
typedef tree<int, null_type, less<int>, rb_tree_tag, tree_order_statistics_node_update> indexed_set;
int vis[26];
string invalid="-1";
int main() {
	string s;
	cin >> s;
	int n=s.size();
	for(int i=0; i<n; ++i) {
		if(s[i]!='?')	vis[s[i]-'A']++;
	}
	int cnt=0;
	for(int i=0; i<26; ++i)	if(vis[i]>0)	cnt++;
	if(cnt<26 || n<26)	cout << invalid << endl;
	else {
		for(int i=0; i<n; ++i) {
			if(s[i]=='?') {
				bool ok=true;
				for(int j=0; j<26; ++j) {
					if(vis[j]==0) {
						s[i]='A'+j;
						vis[j]++;
						ok=false;
						break;
					}
				}
				if(ok) {
					s[i]='A';
					vis['A'-'A']++;
				}
			}
		}
		cout << s << endl;
	}
	return 0;
}
