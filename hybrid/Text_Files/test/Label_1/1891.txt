#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi vector<int>
#define pb push_back
#define ff first
#define ss second
#define pii pair<int,int>
const int N=2e5+5;
vi gr[N];
int par[N],sub[N],dp[2][N];
bool vis[N];
void dfs1(int u,int p){
	sub[u]=1;
	for(auto v:gr[u]){
		if(v!=p){
			dfs1(v,u);
			sub[u]+=sub[v];
		}
	}
}
void dfs2(int u,int p){
	for(auto v:gr[u]){
		if(v!=p){
			dp[0][v]=dp[0][u]+1;
			dp[1][v]=dp[1][u]+sub[u]-sub[v];
			dfs2(v,u);
		}
	}
}
int main() {
	int n,m;
	cin>>n>>m;
	for(int i=0;i<n-1;++i){
		int u,v;
		cin>>u>>v;
		gr[u].pb(v);
		gr[v].pb(u);
	}
	dfs1(1,1);
	dfs2(1,1);
	while(m--){
		int k;
		cin>>k;
		vi v(k);
		int flg=0;
		for(int i=0;i<k;++i){
			cin>>v[i];
			if(v[i]==1) flg=1;
		}
		if(flg){
			cout<<"YES\n";
		}
		else{
			int ans=0;
			int x,mn=INT_MAX;
			for(int j=0;j<k;++j){
				if(dp[1][v[j]]<mn){
					mn=dp[1][v[j]];
					x=v[j];
				}
			}
			for(int j=0;j<k;++j){
				if(dp[0][v[j]]<=mn || v[j]==x) ans++;
			}
			if(ans==k) cout<<"YES\n";
			else cout<<"NO\n";
		}
	}
    return 0;
}
