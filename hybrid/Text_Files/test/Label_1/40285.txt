from collections import Counter
s=list(input())
t=sorted(s);a=Counter(s)
x=[t[i] for i in range(len(t)) if a[t[i]]%2==1]
y=[t[i] for i in range(len(t)) if a[t[i]]%2==0]
if len(x)==0:print(''.join(s));exit()
for j in range(int((len(y)+1)/2),len(y)):t[len(t)-1-j]=y[j]
t[len(t)//2]=x[0]
for i in range(1,len(x)):t[len(t)//2-i]=t[len(t)-1-i]=x[i]
print(''.join(t))
