#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define N 100010
int a[N];
vector<int> p[N];
ll ans = 0;
typedef pair<int, int> pii;
pii e[N];
int cnt[N];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    ans = n - 1;
    for (int i = 1; i <= m; i++) { cin >> a[i]; }
    for (int i = 1; i < m; i++) {
        int u = a[i], v = a[i + 1];
        if (u > v) swap(u, v);
        pii now = {u, v};
        cnt[v]++;
        if (e[v].first == 0) ans++;
        if (cnt[v] == 1)
            e[v] = now;
        else {
            if (e[v].second == 0) {
                e[v].second = now.first;
            } else
                e[v].second = now.first < e[v].second
                                   ? now.first
                                   : e[v].second;
        }
    }
    for (int i = 1; i <= n; i++) {
        if (e[i].first) {
            int u = e[i].first, v = e[i].second;
            ans -= v != 0;
        }
    }
    cout << ans;
    return 0;
}
