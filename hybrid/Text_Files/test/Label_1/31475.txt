import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        int n = sc.nextInt();
        int[] a = new int[n];
        for(int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        
        int k = 0;
        for(int i = 0; i < n; i++) {
            k += Math.abs(a[i]);
        }
        
        int[] b = new int[k];
        for(int i = k - 1, j = 0; i >= 0; i--) {
            int p = (int)Math.pow(2, j);
            while(Math.abs(k) >= p) {
                if(k >= 0) {
                    b[i] = p;
                } else {
                    b[i] = -p;
                }
                k -= b[i];
            }
            j++;
        }
        
        System.out.println(k);
        for(int i = 0; i < k; i++) {
            System.out.print(b[i] + " ");
        }
    }
}
