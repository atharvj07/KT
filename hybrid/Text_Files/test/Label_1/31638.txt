import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int d = scanner.nextInt();
        int[] m = new int[n];
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = scanner.nextInt();
            s[i] = scanner.nextInt();
        }
        scanner.close();
        Arrays.sort(m);
        Arrays.sort(s);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (i < n - 1 && m[i + 1] - m[i] < d) {
                continue;
            }
            ans += s[i];
        }
        System.out.println(ans);
    }
}
