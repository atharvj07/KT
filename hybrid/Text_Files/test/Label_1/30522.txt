import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n;
    List<Integer>[] xlist, ylist;
    Map<Long, Integer> xpoint, ypoint;

    void solve() throws Exception {
        n = Integer.parseInt(br.readLine());

        xlist = new List[10001];
        ylist = new List[10001];
        for (int i = 0; i < 10001; i++) {
            xlist[i] = new ArrayList<>();
            ylist[i] = new ArrayList<>();
        }
        xpoint = new HashMap<>();
        ypoint = new HashMap<>();

        for (int i = 0; i < n; i++) {
            String[] t = br.readLine().split(" ");
            int x1 = Integer.parseInt(t[0]), y1 = Integer.parseInt(t[1]), x2 = Integer.parseInt(t[2]), y2 = Integer.parseInt(t[3]);
            int x, y;
            if (x1 == x2) {
                x = x1;
                y = y1 >= y2 ? y1 : y2;
            } else {
                x = x1 >= x2 ? x1 : x2;
                y = y1;
            }
            xlist[x].add(y);
            ylist[y].add(x);
            long point = x * 1000000000L + y;
            int cnt = xpoint.containsKey(point) ? xpoint.get(point) : 0;
            xpoint.put(point, ++cnt);
            point = y * 1000000000L + x;
            cnt = ypoint.containsKey(point) ? ypoint.get(point) : 0;
            ypoint.put(point, ++cnt);
        }

        long ans = 0;
        for (int i = 0; i < 10001; i++) {
            if (xlist[i].isEmpty())
                continue;
            for (int a : xlist[i]) {
                ans += (long)xlist[i].size() * (long)ylist[a].size() - (long)xpoint.get(i * 1000000000L + a) - (long)ypoint.get(a * 1000000000L + i);
            }
        }
        System.out.println(ans / 4);
    }

    public static void main(String[] args) throws Exception {
        new Main().solve();
    }
}
