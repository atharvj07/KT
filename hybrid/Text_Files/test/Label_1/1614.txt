#include <bits/stdc++.h>
using namespace std;
#define f first
#define s second
typedef pair <int, int> pi;
typedef vector <pi> vp;
const int mxn = 1001;
int n, m;
char g[mxn][mxn];

bool ch(int x, int y) {
    return (x < 0 || x >= n || y < 0 || y >= m || g[x][y] == '#');
}

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            cin >> g[i][j];
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            if (!ch(i, j) && !ch(i + 1, j) && !ch(i, j + 1) && !ch(i + 1, j + 1)) {
                cout << "YES" << endl;
                return 0;
            }
            if (!ch(i, j) && !ch(i, j + 1) && !ch(i + 1, j + 1) && !ch(i + 1, j + 2)) {
                cout << "YES" << endl;
                return 0;
            }
        }
    }
    cout << "NO" << endl;
}
