for _ in range(int(input())):
    n = int(input())
    a = [int(i) for i in input().split()]
    b, c, d, m = 0, 1 << 30, 0, 0
    for i in a:
        c = min(c, i)
        m = max(m, i)
        d += i
    while b < c:
            b += c
    if d <= b * 2:
        print(' '.join([str(b)] * n))
    else:
        b += c
        x = (d - b * 2) // n
        y = (d - b * 2) % n + 1
        if x >= 0:
            t = []
            for i in range(y):
                t.append(str(b + x + 1))
            for i in range(y, n):
                t.append(str(b + x))
            print(' '.join(t))
        else:
            print(' '.join([str(i) for i in a]))
