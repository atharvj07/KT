from sys import stdin

x1,y1,x2,y2,x3,y3 = map(int, stdin.readline().split())
s=max(x1,y1,x2,y2,x3,y3)
for n in range(s,101):
    if n%x1==n%y1 and n%x2==n%y2 and n%x3==n%y3:
        print(n)
        for i in range(n//x1):
            print("A"*x1)
        for i in range(n//x2):
            print("B"*x2)
        for i in range(n//x3):
            print("C"*x3)
        break
else:
    print(-1)
