import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    private static List<Segment> segments;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        segments = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split("\\s+");
            int x = Integer.parseInt(line[0]);
            int h = Integer.parseInt(line[1]);
            segments.add(new Segment(x - h, x + h, h, false));
        }
        segments.sort((s1, s2) -> s1.left - s2.left);
        int ans = 0;
        for (Segment s : segments) {
            if (!s.used) {
                s.used = true;
                ans++;
                for (Segment other : segments) {
                    if (other.used) {
                        continue;
                    }
                    if (other.left >= s.left && other.left <= s.right) {
                        other.used = true;
                    }
                    if (other.right >= s.left && other.right <= s.right) {
                        other.used = true;
                    }
                }
            }
        }
        System.out.println(ans);
    }

    private static class Segment {
        int left, right, h;
        boolean used;

        Segment(int left, int right, int h, boolean used) {
            this.left = left;
            this.right = right;
            this.h = h;
            this.used = used;
        }
    }
}
