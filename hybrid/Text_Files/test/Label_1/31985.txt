import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        if (canSwap(s)) {
            System.out.println("YES");
            System.out.println(color(s));
        } else {
            System.out.println("NO");
        }
    }

    private static boolean canSwap(String s) {
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        return canSwap(cnt);
    }

    private static boolean canSwap(int[] cnt) {
        int sum = 0;
        for (int i = 0; i < 26; i++) {
            sum += cnt[i];
            if (cnt[i] > sum) {
                return false;
            }
        }
        return true;
    }

    private static String color(String s) {
        StringBuilder sb = new StringBuilder();
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        for (char c : s.toCharArray()) {
            int idx = c - 'a';
            if (cnt[idx] == 1) {
                sb.append('0');
            } else {
                sb.append('1');
                cnt[idx]--;
            }
        }
        return sb.toString();
    }
}
