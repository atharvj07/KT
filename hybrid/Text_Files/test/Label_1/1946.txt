#include <bits/stdc++.h>
#include <array>
using namespace std;
array<int, 100 + 5> ar;
int main() {
  int t;
  cin >> t;
  auto check_death = [&ar](int l) {
    for (int i = 1; i < l; i++) {
      if (ar[i] + ar[i - 1] == 0) return true;
    }
    return false;
  };
  while (t--) {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> ar[i];
    if (check_death(n))
      cout << -1 << endl;
    else {
      int sum = 1;
      for (int i = 1; i <= n; i++) {
        if (ar[i]) {
          sum = ar[i - 1] ? sum + 5 : sum + 1;
        }
      }
      cout << sum << endl;
    }
  }
  return 0;
}
