#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int sit = 0;
    for (char x : s) sit += (x == 'x');
    if (sit == n / 2) {
        return cout << "0\n" << s, 0;
    }
    int ans = abs(sit - n / 2) * 2;
    sit = n - sit;
    while (sit != n / 2) {
        if (s[0] == 'x') {
            s[0] = 'X';
            sit--;
            ans += 2;
        } else if (s.back() == 'X') {
            s.back() = 'x';
            sit++;
            ans += 2;
        } else {
            if (sit < n / 2) {
                s[0] = 'X';
                sit++;
                ans += 2;
            } else {
                s.back() = 'x';
                sit--;
                ans += 2;
            }
        }
    }
    cout << ans << '\n' << s << '\n';
    return 0;
}
