#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

struct Project {
  long long required, change;
};

int main() {
  int n, r;
  cin >> n >> r;
  vector<Project> projects(n);
  for (int i = 0; i < n; i++) cin >> projects[i].required >> projects[i].change;
  sort(projects.begin(), projects.end(), [](const Project& p1, const Project& p2) { return p1.required < p2.required; });
  int ans = 0;
  for (const auto& project : projects) {
    if (r < project.required) break;
    r += project.change;
    ans++;
  }
  cout << ans << endl;
  return 0;
}
