#include <bits/stdc++.h>
using namespace std;
int main() {
  string a;
  cin >> a;
  string b;
  cin >> b;
  int n;
  cin >> n;
  while (n--) {
    string c;
    cin >> c;
    int i = 0, j = 0, k = 0;
    bool flag = 0;
    while (i < c.size() && j < b.size() && !flag) {
      if (b[j] == '*') {
        k = j++;
        while (j < b.size() && b[j] != '*') j++;
      } else if (b[j] == '?' || b[j] == c[i])
        j++;
      else if (k != 0)
        j = k;
      else
        flag = 1;
      i++;
    }
    if (i == c.size() && j == b.size())
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
}
