import sys
inp=sys.stdin
int1=lambda x:int(x)-1
n,m,c,e,v=map(int,inp.readline().split())
l=[int1(x) for x in inp.readline().split()]
e=[int1(x) for x in inp.readline().split()]
q=int(inp.readline())
for i in range(q):
    p1,p2,p3,p4=map(int1,[x for x in inp.readline().split()])
    if p1==p3:print(abs(p2-p4))
    elif l and p2<=l[0]<=p4:print(abs(p1-p3)+1)
    elif e and p2<=e[0]<=p4:print(abs(p1-p3)+2 if v>=abs(p1-p3) else abs(p1-p3)+1)
    else:print(abs(p1-p3)+min(abs(p2-l[0]),abs(p2-e[0]))+1)
