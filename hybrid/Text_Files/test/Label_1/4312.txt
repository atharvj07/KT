#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pi;
#define fi first
#define se second

int main() {
    int n, x;
    cin >> n >> x;
    vector<pi> v(n);
    for (int i = 0; i < n; i++) cin >> v[i].fi >> v[i].se;
    sort(v.begin(), v.end());
    int ans = 2e9;
    // Try splitting x into two
    for (int i = 0, j = 0; i < n; i++) {
        while (j < n && v[j].fi <= v[i].se) j++;
        while (j < n && v[i].se + v[j].se <= x) {
            ans = min(ans, v[i].fi + v[j].fi);
            j++;
        }
    }
    if (ans == 2e9)
        cout << -1;
    else
        cout << ans;
    return 0;
}
