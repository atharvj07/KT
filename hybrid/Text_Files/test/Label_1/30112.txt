import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    try (Scanner in = new Scanner(System.in)) {
      int n = in.nextInt();
      int m = in.nextInt();
      int x = in.nextInt();
      System.out.println(x == 1 ? (n - 1) * (m - 1) : (n - 1) * 2 + (m - 1) * 2);
    }
  }
}
