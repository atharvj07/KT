import java.util.*;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt(), c = in.nextInt();
        int len = Math.max(Integer.toString(a, 3).length(), Integer.toString(c, 3).length());
        String a3 = String.format("%" + len + "s", Integer.toString(a, 3)).replace(' ', '0');
        String c3 = String.format("%" + len + "s", Integer.toString(c, 3)).replace(' ', '0');

        int b = 0;
        for (int i = 0; i < len; ++i) {
            int da = a3.charAt(i) - '0', dc = c3.charAt(i) - '0';
            b = b * 3 + (dc - da + 3) % 3;
        }
        System.out.println(b);
    }
}
