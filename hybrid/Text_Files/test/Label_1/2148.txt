#include <bits/stdc++.h>
#define ll long long
using namespace std;
const ll mod = 998244353;

int main() {
  int n;
  cin >> n;
  vector<ll> a(2 * n);
  for (auto &i : a) cin >> i;
  sort(a.begin(), a.end());
  ll ans = 0;
  for (int i = 0; i < n; i++) {
    ans = (ans + a[i] - a[n + i]) % mod;
    ans = (ans + mod) % mod;
  }
  cout << ans << endl;
}
