import java.util.Arrays;
import java.util.Scanner;

public class VasyaAndTheCarRental {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int s = scanner.nextInt();
        int t = scanner.nextInt();

        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        Arrays.sort(gasStations);

        int minPrice = -1;
        for (Car car : cars) {
            if (canReachCinema(car, gasStations, s, t)) {
                minPrice = Math.min(minPrice == -1 ? car.price : minPrice, car.price);
            }
        }

        System.out.println(minPrice);
    }

    static boolean canReachCinema(Car car, int[] gasStations, int s, int t) {
        int time = 0;
        int fuel = car.capacity;
        int currentStation = 0;

        for (int i = 0; i < s; i++) {
            if (i == gasStations[currentStation]) {
                fuel = car.capacity;
                currentStation++;
            }

            time += (car.speed == 1 ? 2 : 1);
            fuel -= (car.speed == 1 ? 1 : 2);

            if (time > t) {
                return false;
            }

            if (fuel == 0) {
                return false;
            }
        }

        return true;
    }

    static class Car {
        int price;
        int capacity;
        int speed;

        Car(int price, int capacity) {
            this.price = price;
            this.capacity = capacity;
            this.speed = 1; // default to normal speed
        }
    }
}
