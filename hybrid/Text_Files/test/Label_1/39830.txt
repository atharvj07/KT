t = input().strip()
sequence = input().strip()

digit_map = {'2': '5', '5': '2', '6': '9', '9': '6'}
digit_counts = {str(i): 0 for i in range(10)}

for digit in sequence:
    if digit in digit_map:
        digit_counts[digit_map[digit]] += 1
    digit_counts[digit] += 1

min_count = float('inf')
for digit in t:
    if digit in digit_map:
        count = digit_counts[digit] + digit_counts[digit_map[digit]]
        min_count = min(min_count, count // 2)
    else:
        min_count = min(min_count, digit_counts[digit])

print(min_count)
