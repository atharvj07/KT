s, g = [], []
for i in range(int(input())):
    a, b = list(map(int, input().split()))
    s.append(a), g.append(b)
print(-1 if any(abs((s[i - 1] + g[i - 1]) - (s[i] + min(g[i], g[i - 1]))) > 1 for i in range(1, len(s))) else sum([min(g[i], g[i - 1]) for i in range(1, len(s))]))
print(*[(s[i] + min(g[i], g[i - 1])) for i in range(len(s))])
