import java.util.Arrays;

class Main {
    static final long mod = 998244353;
    public static void main(String[] args) {
        FastReader sc = new FastReader();
        int n = sc.nextInt();
        int k = sc.nextInt();
        long fact[] = new long[n + 5];
        fact[0] = 1;
        for (int i = 1; i <= n; i++) {
            fact[i] = (fact[i - 1] * i) % mod;
        }
        long ans = ((n * (n - 1)) / 2) - k;
        long two = 1;
        for (int i = 0; i < k; i++) {
            two = (two * 2) % mod;
        }
        ans = (ans % mod + mod) % mod;
        ans = (ans * two) % mod;
        long p = 1;
        for (int i = 0; i < n; i++) {
            p = (p * 2) % mod;
        }
        ans = (ans * p) % mod;
        System.out.println(ans);
    }

    static class FastReader {
        byte[] buf = new byte[2048];
        int index, total;
        StringBuilder sbuf = new StringBuilder();

        int scan() {
            if (index >= total) {
                index = 0;
                try {
                    total = System.in.read(buf);
                } catch (Exception e) {
                }
                if (total <= 0) {
                    return -1;
                }
            }
            return buf[index++];
        }

        String next() {
            while (scan() <= 32)
                ;
            sbuf.setLength(0);
            while (index < total) {
                byte c = buf[index++];
                if (c > 32) {
                    sbuf.append((char) c);
                } else {
                    break;
                }
            }
            return sbuf.toString();
        }

        int nextInt() {
            int sign = 1;
            while (scan() <= 32)
                ;
            if (buf[index - 1] == '-') {
                sign = -1;
                index++;
            }
            int ret = 0;
            while (index < total) {
                byte c = buf[index++];
                if (c <= 32) {
                    break;
                }
                ret = ret * 10 + c - '0';
            }
            return ret * sign;
        }

        long nextLong() {
            int sign = 1;
            while (scan() <= 32)
                ;
            if (buf[index - 1] == '-') {
                sign = -1;
                index++;
            }
            long ret = 0;
            while (index < total) {
                byte c = buf[index++];
                if (c <= 32) {
                    break;
                }
                ret = ret * 10 + c - '0';
            }
            return ret * sign;
        }

        double nextDouble() {
            double ret = 0, div = 1;
            int sign = 1;
            while (scan() <= 32)
                ;
            if (buf[index - 1] == '-') {
                sign = -1;
                index++;
            }
            while (index < total) {
                byte c = buf[index++];
                if (c > 57) {
                    break;
                }
                ret = ret * 10 + c - '0';
            }
            if (buf[index - 1] == '.') {
                while (index < total) {
                    byte c = buf[index++];
                    if (c <= 57) {
                        ret += (c - '0') / (div *= 10);
                    } else {
                        break;
                    }
                }
            }
            return ret * sign;
        }

        char nextChar() {
            while (scan() <= 32)
                ;
            return (char) buf[index++];
        }
    }
}
