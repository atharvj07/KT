#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5 + 5;

int n, m;
long long arr[MAXN];
vector<pair<int, int>> ops[MAXN];

int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin >> n >> m;
  for (int i = 1; i <= n; ++i) { cin >> arr[i]; }
  for (int i = 0; i < m; ++i) {
    int t;
    cin >> t;
    if (t == 1) {
      int v, x;
      cin >> v >> x;
      ops[v].push_back({1, x});
    } else if (t == 2) {
      int y;
      cin >> y;
      for (int i = 1; i <= n; ++i) { ops[i].push_back({2, y}); }
    } else {
      int q;
      cin >> q;
      ops[q].push_back({3, i});
    }
  }
  for (int i = 1; i <= n; ++i) {
    long long val = arr[i];
    for (pair<int, int> &op : ops[i]) {
      if (op.first == 1)
        val = op.second;
      else if (op.first == 2)
        val += op.second;
      else
        cout << val << '\n';
    }
  }
  return 0;
}
