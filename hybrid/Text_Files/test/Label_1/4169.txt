#include <bits/stdc++.h>
#define ll long long int
using namespace std;
int main() {
  ll t;
  cin >> t;
  while (t--) {
    ll n, w;
    ll ans(1);
    cin >> n >> w;
    vector<ll> v;
    for (ll i = 0; i < n; i++) {
      ll tmp;
      cin >> tmp;
      v.push_back(tmp);
    }
    sort(v.begin(), v.end());
    ll tmp = 0;
    for (ll i = n - 1; i >= 0; i--) {
      tmp += v[i];
      if (w < tmp) {
        tmp = v[i];
        ans++;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
