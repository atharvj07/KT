import java.util.Scanner;
import java.util.Arrays;

public class Main {
    static int t, n, s;
    static int[] ans = new int[10005];

    static void dfs(int pos, int sum) {
        if (pos == n + 1) {
            ans[sum]++;
            return;
        }

        for (int i = 0; i <= 9; i++) {
            if (sum + pos * i <= s) {
                dfs(pos + 1, sum + pos * i);
            }
        }
    }

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        while (cin.hasNext()) {
            n = cin.nextInt();
            s = cin.nextInt();
            Arrays.fill(ans, 0);
            dfs(1, 0);
            System.out.println(ans[s]);
        }
    }
}
