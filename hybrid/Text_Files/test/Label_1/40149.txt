def assign_seats(n, a, b):
    if n > a * b:
        return -1
    # Initialize an empty assembly hall
    hall = [[0 for _ in range(b)] for _ in range(a)]
    odd = 1  # Starting index for Democrats
    even = 2  # Starting index for Republicans
    for i in range(a):
        for j in range(b):
            if (i+j) % 2 == 0 and odd <= n:  # Check for odd indices and if any Democrats left
                hall[i][j] = odd
                odd += 2
            elif (i+j) % 2 != 0 and even <= n:  # Check for even indices and if any Republicans left
                hall[i][j] = even
                even += 2
    return hall

n, a, b = map(int, input().split())
result = assign_seats(n, a, b)
if result == -1:
    print(-1)
else:
    for row in result:
        print(*row)
