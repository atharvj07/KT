import sys
input = sys.stdin.readline

t = [list(map(int, input().split())) for i in range(3)]
n = int(input())

dp = [[[-1] * 4 for j in range(4)] for k in range(n + 1)]

def solve(pos, prv, cnt):
    if cnt == n: return 0
    if dp[cnt][pos][prv] != -1: return dp[cnt][pos][prv]

    ret = 1e9
    for i in range(3):
        if i != prv:
            ret = min(ret, solve(i, pos, cnt + 1) + t[pos][i])
            ret = min(ret, solve(prv, i, cnt + 1) + t[prv][i])

    dp[cnt][pos][prv] = ret
    return ret

print(solve(0, 0, 0))
