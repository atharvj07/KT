#include<bits/stdc++.h>
using namespace std;
typedef long long ll;const ll N=2e5+1;ll a[N],dp[N],dp2[N],p=1e9+7;
int main(){ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);ll n,ans=0;cin>>n;for(ll i=1;i<=n;i++)cin>>a[i];
    dp[n+1]=dp2[n+1]=p;for(ll i=n;i>=1;i--){dp[i]=((a[i]|dp[i+1])+p)%p;dp2[i]=dp2[i+1]|a[i];}
    for(ll i=1;i<n;i++)ans=(ans+(dp[i]|dp2[i+1])%p)%p;cout<<ans<<endl;return 0;}
