#include <bits/stdc++.h>
using namespace std;

const int N = 22;
char ns[N], nt[N];
string s[N], t[N];
int main() {
	int n, m, q;
	cin >> n >> m;
	for (int i = 0; i < n; i++) {
		cin >> ns[i];
		s[i] = ns[i];
	}
	for (int i = 0; i < m; i++) {
		cin >> nt[i];
		t[i] = nt[i];
	}
	cin >> q;
	while (q--) {
		int x;
		cin >> x;
		cout << s[(x - 1) % n] << t[(x - 1) % m] << endl;
	}
}
