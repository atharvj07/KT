#include <iostream>
using namespace std;

typedef long long ll;

const int mod = 1e9 + 7;
ll qpow(ll a, ll b) {
	ll ret = 1;
	while (b) {
		if (b & 1) ret = ret * a % mod;
		a = a * a % mod;
		b >>= 1;
	}
	return ret;
}
ll gcd(ll a, ll b) {
	if (!b) return a;
	return gcd(b, a % b);
}
int main() {
	ll n, a1, a2;
	int m;
	cin >> n >> m;
	cin >> a1;
	cin >> a2;
	ll up = 0, down = 1;
	for (ll i = n - 1; i >= 0; --i) {
		ll a1_now = (a1 % 10), a2_now = (a2 % 10);
		a1 /= 10;
		a2 /= 10;
		up += a1_now * qpow(m, i) % mod;
		down += a2_now * qpow(m, i) % mod;
		up %= mod;
		down %= mod;
	}
	ll divisor = gcd(up, down);
	up /= divisor;
	down /= divisor;
	cout << up * qpow(down, mod - 2) % mod << endl;
	return 0;
}
