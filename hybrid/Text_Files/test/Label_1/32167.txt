import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int lastDiff = -1;
        int maxLength = 0;
        int currLength = 0;
        for (int i = 0; i < n; i++) {
            if (lastDiff == -1 || arr[i] != arr[lastDiff]) {
                lastDiff = i;
                currLength++;
            } else {
                currLength = 1;
                lastDiff = i;
            }
            maxLength = Math.max(maxLength, currLength);
        }
        System.out.println(maxLength);
    }
}
