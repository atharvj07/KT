#include <algorithm>
#include <cstdio>
#include <iostream>
#include <vector>

using namespace std;

string S, T;

int main() {
  cin >> S;
  for (int i = 0; i < S.size(); i++) {
    while (T.size() > 0 && T.back() > S[i])
      T.pop_back();
    T += S[i];
    printf("%d ", T.size());
    if (T.size() > 10)
      cout << T.substr(0, 5) << "..." << T.substr(T.size() - 2, 2) << endl;
    else
      cout << T << endl;
  }
  return 0;
}
