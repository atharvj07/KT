import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] times = {sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt()};
        int[] wrongs = {sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt()};
        int hs = sc.nextInt();
        int hu = sc.nextInt();
        int score = 0;
        int[] probs = {500, 1000, 1500, 2000, 2500};
        for (int i = 0; i < 5; i++) {
            score += probs[i] * (1.0 - (double) times[i] / 119.0) - wrongs[i] * 250.0;
        }
        score += hs * 100 - hu * 50;
        System.out.println(score);
    }
}
