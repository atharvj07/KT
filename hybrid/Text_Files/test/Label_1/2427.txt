#include <bits/stdc++.h>
using namespace std;

const int maxn = 1 << 19;
bool vis[maxn];
vector<int> primes;

void gen_primes() {
    for (int i = 2; i < maxn; ++i) {
        if (!vis[i]) {
            primes.push_back(i);
            for (int j = i + i; j < maxn; j += i) {
                vis[j] = true;
            }
        }
    }
}

int main() {
    gen_primes();
    int n;
    cin >> n;
    if (n == 3) {
        cout << 1 << endl;
        cout << 3 << endl;
    } else {
        int idx = 0;
        while (idx < primes.size() && primes[idx] * 3 <= n) { idx++; }
        cout << 2 + (n % 2) << endl;
        while (idx >= 0) {
            while (n >= primes[idx]) {
                n -= primes[idx];
                cout << primes[idx] << ' ';
            }
            idx--;
        }
        cout << endl;
    }
    return 0;
}
