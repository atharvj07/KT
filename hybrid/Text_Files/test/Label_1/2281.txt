#include <iostream>
#include <vector>
#include <map>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> ans(n, 0);
    map<char, int> mp;
    int color = 0;
    int bal = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            bal++;
            if (!mp[s[i]]) {
                mp[s[i]] = ++color;
            }
            ans[i] = max(ans[i], mp[s[i]]);
        } else if (s[i] == ')') {
            bal--;
            if (!mp[s[i]]) {
                mp[s[i]] = ++color;
            }
            ans[i] = max(ans[i], mp[s[i]]);
        } else {
            if (!mp[s[i]]) {
                mp[s[i]] = color;
            }
            ans[i] = mp[s[i]];
        }
        if (bal < 0) {
            ans[i] = -1;
        }
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
