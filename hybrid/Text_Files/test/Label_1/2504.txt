#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;
const int MOD = 1e9 + 7;

vector<int> prime;
vector<bool> is_prime;
int mu[250001];
int sum_mu[250001];

void sieve(int n) {
    is_prime.resize(n + 1, true);
    is_prime[0] = is_prime[1] = false;
    mu[1] = 1;
    for (int p = 2; p <= n; p++) {
        if (is_prime[p]) {
            prime.push_back(p);
            mu[p] = -1;
        }
        for (int i = 0; i < prime.size() && prime[i] * p <= n; i++) {
            is_prime[prime[i] * p] = false;
            if (p % prime[i] == 0) {
                mu[prime[i] * p] = 0;
                break;
            } else {
                mu[prime[i] * p] = -mu[p];
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        sum_mu[i] = sum_mu[i - 1] + mu[i];
    }
}

int sum_mob(int l, int r) {
    return sum_mu[r] - sum_mu[l - 1];
}

int main() {
    int n, m;
    cin >> n >> m;
    sieve(n);
    ll ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            ans += 1LL * sum_mob(i, j) * (j - i + 1) % m;
            ans %= m;
        }
    }
    cout << ans << endl;
    return 0;
}
