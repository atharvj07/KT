n = int(input())
points = [tuple(map(int, input().split())) for _ in range(n + 1)]

dangerous_turns = 0

for i in range(1, n):
    x0, y0 = points[i - 1]
    x1, y1 = points[i]
    x2, y2 = points[i + 1]

    if (x1 == x0 and x2 == x1) or (y1 == y0 and y2 == y1):
        continue
    if (x0 < x1 and y1 < y2) or (y0 < y1 and x1 > x2) or (x0 > x1 and y1 > y2) or (y0 > y1 and x1 < x2):
        dangerous_turns += 1

print(dangerous_turns)
