def solve():
    s = input()
    t = input()
    q = int(input())
    a = [0] * q
    b = [0] * q
    c = [0] * q
    d = [0] * q
    for i in range(q):
        a[i], b[i], c[i], d[i] = map(int, input().split())
    for i in range(q):
        if len(s[a[i] - 1:b[i]]) != len(t[c[i] - 1:d[i]]):
            print(0)
        else:
            x = s[a[i] - 1:b[i]]
            y = t[c[i] - 1:d[i]]
            if s[a[i] - 1:b[i]] == t[c[i] - 1:d[i]]:
                print(1)
            elif x.count('A') < y.count('A'):
                print(0)
            elif x.count('B') < y.count('B'):
                print(0)
            elif x.count('C') < y.count('C'):
                print(0)
            else:
                print(1)


solve()
