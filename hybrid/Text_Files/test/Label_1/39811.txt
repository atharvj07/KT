import sys
input = sys.stdin.readline

n = int(input())
adj = [[] for i in range(n + 1)]
for i in range(n - 1):
    a, b = map(int, input().split())
    adj[a].append(b)
    adj[b].append(a)

ans = [0] * (n + 1)
vis = [0] * (n + 1)


def dfs(node, par):
    sum = 1
    for i in adj[node]:
        if i != par:
            dfs(i, node)
            sum += ans[i]
    ans[node] = sum
    return sum


s = dfs(1, 1)
for i in range(1, n + 1):
    print(ans[i] / s)
