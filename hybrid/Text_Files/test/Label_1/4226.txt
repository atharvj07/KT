#include <cstdio>
#include <vector>
#include <algorithm>

const long long BASE = 60;
const long long PER_TRIP = 20;
const long long PER_90MIN = 50;
const long long PER_DAY = 120;
const long long PER_DAY_MIN = 24 * BASE;

int main() {
    long long n;
    scanf("%lld", &n);

    std::vector<long long> times(n);
    for (long long i = 0; i < n; ++i) {
        scanf("%lld", &times[i]);
    }

    long long totalCost = 0, prevCost = 0;
    for (long long i = 0; i < n; ++i) {
        long long minCost = totalCost + PER_TRIP;

        // check if 90 min ticket is cheaper
        long long tripEnd = times[i] + BASE - 1;
        long long ticketStart = times[i] - BASE + 1;
        long long ticketCost = PER_90MIN;
        if (ticketStart >= 0) {
            long long tripsCovered = std::upper_bound(times.begin(), times.end(), tripEnd) - times.begin();
            tripsCovered -= std::upper_bound(times.begin(), times.end(), ticketStart - 1) - times.begin();
            ticketCost -= tripsCovered * PER_TRIP;
            if (ticketCost < minCost) {
                minCost = ticketCost;
            }
        }

        // check if day ticket is cheaper
        tripEnd = times[i] + PER_DAY_MIN - 1;
        ticketStart = times[i] - PER_DAY_MIN + 1;
        ticketCost = PER_DAY;
        if (ticketStart >= 0) {
            long long tripsCovered = std::upper_bound(times.begin(), times.end(), tripEnd) - times.begin();
            tripsCovered -= std::upper_bound(times.begin(), times.end(), ticketStart - 1) - times.begin();
            ticketCost -= tripsCovered * PER_TRIP;
            if (ticketCost < minCost) {
                minCost = ticketCost;
            }
        }

        prevCost = totalCost;
        totalCost = minCost;
        printf("%lld\n", totalCost - prevCost);
    }

    return 0;
}
