import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        List<List<int[]>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt();
            int v = scanner.nextInt();
            int c = scanner.nextInt();
            graph.get(u).add(new int[]{v, c});
            graph.get(v).add(new int[]{u, c});
        }
        int[] maxSum = new int[n];
        dfs(0, -1, graph, maxSum);
        System.out.println(maxSum[0]);
    }

    public static void dfs(int u, int p, List<List<int[]>> graph, int[] maxSum) {
        for (int[] edge : graph.get(u)) {
            int v = edge[0];
            int c = edge[1];
            if (v != p) {
                dfs(v, u, graph, maxSum);
                maxSum[u] = Math.max(maxSum[u], maxSum[v] + c);
            }
        }
    }
}
