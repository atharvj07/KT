#include <cstdio>

long long a, b, m, r;

int main() {
	int q;
	scanf("%d", &q);
	while (q--) {
		scanf("%lld %lld %lld", &a, &b, &m);
		for (r = 1; r < m && a + r * (r + 1) / 2 < b; ++r)
			;
		if (b == a + r * (r + 1) / 2)
			printf("%d %lld", r + 1, a + r * (r + 1) / 2);
		else
			printf("-1");
		putchar('\n');
	}
}
