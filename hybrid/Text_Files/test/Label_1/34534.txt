import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    final static long MOD = (long)1e18;

    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = in.nextInt();
        }
        long[][] pref = new long[n + 1][100000];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = 0; j < 100000; j++) {
                pref[i][j] = pref[i + 1][j];
                if ((j + x[i]) < 100000) {
                    pref[i][j] = (pref[i][j] + pref[i + 1][j + x[i]]) % MOD;
                }
            }
        }
        long[] ans = new long[n];
        Arrays.fill(ans, 1);
        for (int i = n - 2; i >= 0; i--) {
            for (int j = 0; j < 100000; j++) {
                if ((j + x[i + 1]) < 100000) {
                    ans[i] = (ans[i] + pref[i + 1][j + x[i + 1]]) % MOD;
                }
            }
        }
        for (long i : ans) {
            System.out.println(i);
        }
    }
}
