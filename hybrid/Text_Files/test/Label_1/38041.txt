import sys

n, m, q = map(int, input().split())
a = [[0]*m for i in range(n)]
for i in range(q):
	t, *b = map(int, input().split())
	if t == 1:
		a[b[0]-1] = a[b[0]-1][-len(b[0])%m:] + a[b[0]-1][:-len(b[0])%m]
	elif t == 2:
		for j in range(n):
			a[j][b[0]-1] = a[j][(b[0]-2+j)%m]
	else:
		a[b[0]-1][b[1]-1] = b[2]

for i in a:
	print(*i)
