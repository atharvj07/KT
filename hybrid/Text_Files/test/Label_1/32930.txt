import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) a[i] = in.nextInt();
    int q = in.nextInt();
    TreeSet<Integer> set = new TreeSet<>();
    int[] ans = new int[n];
    Arrays.fill(ans, -1);
    for (int i = 0; i < q; i++) {
      int type = in.nextInt();
      int x = in.nextInt();
      if (type == 1) {
        if (ans[x - 1] == -1) ans[x - 1] = a[x - 1];
        else ans[x - 1] += x;
        Integer ceil = set.ceiling(x);
        if (ceil != null) set.remove(ceil);
      } else {
        set.add(x);
        while (!set.isEmpty() && set.first() <= x) set.remove(set.first());
        for (int j = 0; j < n; j++) {
          if (ans[j] == -1) ans[j] = a[j];
          if (ans[j] < x) ans[j] = x;
        }
      }
    }
    for (int i : ans) System.out.print(i + " ");
  }
}
