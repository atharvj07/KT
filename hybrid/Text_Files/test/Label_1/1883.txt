#include <bits/stdc++.h>
using namespace std;
const int kN = 1e5 + 10;
int main() {
  int n;
  cin >> n;
  vector<int> permutation(n + 1);
  if (n % 2 == 1) {
    cout << "-1" << endl;
  } else {
    for (int i = 1; i <= n / 2; i++) {
      permutation[i] = 2 * i;
      permutation[n - i + 1] = 2 * i - 1;
    }
    for (int i = 1; i <= n; i++) { cout << permutation[i] << " "; }
    cout << endl;
  }
  return 0;
}
