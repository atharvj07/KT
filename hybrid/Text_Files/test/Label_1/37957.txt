m = int(input())
S = set([int(x) for x in input().split()])
Cand = list(range(1,4001))
for s in S:
    t = s
    while t > 1 and t not in Cand:
        t = t//2
    if t > 1:
        Cand.remove(t)
for a1 in Cand:
    L = []
    g = a1
    L.append(g)
    for i in range(1,4000):
        if g not in S:
            break
        g += a1
        L.append(g)
    if set(L) == S:
        print(len(L))
        print(*L)
        exit()
print(-1)
