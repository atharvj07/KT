import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String dir = sc.nextLine();
        int[] dis = new int[n];
        for (int i = 0; i < n; i++) dis[i] = sc.nextInt();
        boolean fin = false;
        for (int i = 0; i < n; i++) {
            if (dir.charAt(i) == '>' && i + dis[i] >= n) {
                fin = true;
                break;
            }
            if (dir.charAt(i) == '<' && i - dis[i] < 0) {
                fin = true;
                break;
            }
        }
        System.out.println(fin ? "FINITE" : "INFINITE");
    }
}
