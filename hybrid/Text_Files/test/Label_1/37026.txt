n, m, d = map(int, input().split())
p = set(map(int, input().split()))
g = [[] for _ in range(n+1)]
for _ in range(n-1): a, b = map(int, input().split()); g[a].append(b); g[b].append(a)
q = [[[[] for _ in range(2)] for _ in range(d+1)] for _ in range(n+1)]
q[1][0][0].append([1])
for i in range(1, n+1):
    for j in range(d+1):
        for l in q[i][j][0]:
            for k in g[i]:
                if not k in l: q[k][j][0].append(l+[k])
        for l in q[i][j][1]:
            for k in g[i]:
                if k in l: q[k][j+1][1].append(l+[k])
                if not k in l: q[k][j+1][0].append(l+[k])
ans = 0
for i in range(1, n+1):
    if bool(q[i][d][0]) or bool(q[i][d][1]): ans += 1
    for j in range(1, d+1):
        if bool(q[i][j][0]) and bool(q[i][d-j][1]): ans += 1
print(ans if ans else 0)
