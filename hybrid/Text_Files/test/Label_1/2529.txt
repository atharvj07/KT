#include <bits/stdc++.h>
using namespace std;

int a[200000 + 1];
int main() {
  int n, k;
  cin >> n >> k;
  for (int i = 1; i <= n; i++) cin >> a[i];
  priority_queue<int, vector<int>, greater<int>> min_heap;
  priority_queue<int> max_heap;
  for (int i = 1; i <= k; i++) min_heap.push(a[i]);
  int ans = min_heap.top();
  for (int i = k + 1; i <= n; i++) {
    if (min_heap.top() < a[i]) {
      max_heap.push(min_heap.top());
      min_heap.pop();
      min_heap.push(a[i]);
    } else {
      max_heap.push(a[i]);
    }
    while (min_heap.size() > max_heap.size() + 1) {
      max_heap.push(min_heap.top());
      min_heap.pop();
    }
    while (min_heap.size() < max_heap.size()) {
      min_heap.push(max_heap.top());
      max_heap.pop();
    }
    ans = max(ans, min_heap.top());
  }
  cout << ans << endl;
}
