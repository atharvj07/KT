#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 5;
int m[maxn];
int main() {
  int n, k;
  cin >> n >> k;
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < k; j++) {
      int x;
      cin >> x;
      m[x] = i + 1;
    }
  }
  int ans = 0;
  for (int i = 1; i <= n * k; i++) ans += m[i] != (i - 1) / k + 1;
  cout << ans;
  return 0;
}
