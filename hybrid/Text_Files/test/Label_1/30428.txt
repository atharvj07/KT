import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String s = input.nextLine();
        
        Map<String, Integer> count = new HashMap<>();
        
        for (int i = 0; i <= s.length() - 10; i++) {
            String sub = s.substring(i, i + 10);
            
            if (sub.charAt(2) == '-' && sub.charAt(5) == '-' 
                    && Character.isDigit(sub.charAt(0)) && Character.isDigit(sub.charAt(1)) 
                    && Character.isDigit(sub.charAt(3)) && Character.isDigit(sub.charAt(4)) 
                    && Character.isDigit(sub.charAt(6)) && Character.isDigit(sub.charAt(7)) 
                    && Character.isDigit(sub.charAt(8)) && Character.isDigit(sub.charAt(9))) {
                
                int month = Integer.parseInt(sub.substring(3, 5));
                if (month > 0 && month <= 12) {
                    int day = Integer.parseInt(sub.substring(0, 2));
                    if (day > 0) {
                        int year = Integer.parseInt(sub.substring(6, 10));
                        if (year >= 2013 && year <= 2015) {
                            int num = 0;
                            if (month == 2) {
                                num = 28;
                            } else if (month == 4 || month == 6 || month == 9 || month == 11) {
                                num = 30;
                            } else {
                                num = 31;
                            }
                            
                            if (day <= num) {
                                count.put(sub, count.getOrDefault(sub, 0) + 1);
                            }
                        }
                    }
                }
            }
        }
        
        int maxCount = 0;
        String answer = "";
        
        for (String date : count.keySet()) {
            if (count.get(date) > maxCount) {
                maxCount = count.get(date);
                answer = date;
            }
        }
        
        System.out.println(answer);
    }
}
