#include <bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
  ll n, x, y, a;
  cin >> n >> x >> y;
  string s;
  cin >> s;
  if (x - y > 1) {
    cout << 0 << endl;
  } else if (x - y == 1) {
    ll ans = 0;
    for (ll i = y; i < n; i++) {
      if (s[i] == '0') {
        ans++;
      }
    }
    cout << ans << endl;
  } else {
    if (s[n - 1] == '0' || s[n - 2] != '0') {
      cout << 0 << endl;
    } else {
      ll ans = 0;
      for (ll i = 0; i < n - 2; i++) {
        if (s[i] == '0') {
          ans++;
        }
      }
      cout << ans << endl;
    }
  }
  return 0;
}
