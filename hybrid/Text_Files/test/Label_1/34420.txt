import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt(), k = scanner.nextInt();
        int[] pebbles = new int[n];
        for (int i = 0; i < n; i++) {
            pebbles[i] = scanner.nextInt();
        }
        Arrays.sort(pebbles);
        int days = 0, pockets = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (pockets + pebbles[i] <= k) {
                pockets += pebbles[i];
            } else {
                days++;
                pockets = pebbles[i];
            }
        }
        System.out.println(days);
    }
}
