#include <bits/stdc++.h>
#define LL long long
using namespace std;

int main() {
    LL n, k, s, t;
    cin >> n >> k >> s >> t;
    vector<pair<LL, LL>> cars(n);
    for (auto &car : cars) cin >> car.first >> car.second;
    vector<LL> gas_stations(k);
    for (auto &station : gas_stations) cin >> station;
    sort(gas_stations.begin(), gas_stations.end());
    LL ans = LLONG_MAX;
    for (auto &car : cars) {
        LL time = 0, fuel = car.second, price = car.first;
        LL idx = 0;
        while (time + 2 <= t) {
            LL dist = idx < k ? gas_stations[idx] - time : s - time;
            if (fuel >= dist) {
                time += dist;
                fuel -= dist;
            } else {
                time += fuel;
                price += fuel * 2;
                fuel = car.second;
            }
            if (idx < k) idx++;
        }
        if (time <= t) {
            ans = min(ans, price);
        }
    }
    if (ans == LLONG_MAX) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    return 0;
}
