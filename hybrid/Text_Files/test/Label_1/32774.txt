import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            long k = sc.nextLong();
            k--;
            long start = (k + 1) * (k + 2) / 2;
            long num = (start - 1) / 9;
            long rem = start - 1 - (9 * num);
            System.out.println((rem + 1) % 10);
        }
    }
}
