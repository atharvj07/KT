#include<bits/stdc++.h>
#define ll long long int
using namespace std;

int main()
{
    ll n,k,i,j,cnt=0;
    cin>>n>>k;
    vector<ll> vec(n);
    for(i=0;i<n;i++){
        cin>>vec[i];
    }
    for(i=0;i<k;i++){
        ll sum=0,flag=1;
        for(j=0;j<n;j++){
            if((vec[j]%k)==i){
                sum+=vec[j]/k;
                flag=0;
            }
        }
        if(!flag){
            if(sum%k==i){
                cnt++;
            }
        }
    }
    cout<<cnt<<endl;
    for(i=0;i<k;i++){
        ll sum=0,flag=1;
        for(j=0;j<n;j++){
            if((vec[j]%k)==i){
                sum+=vec[j]/k;
                flag=0;
            }
        }
        if(!flag){
            if(sum%k==i){
                cout<<i<<" ";
            }
        }
    }
    return 0;
}
