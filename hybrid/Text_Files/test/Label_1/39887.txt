for _ in range(int(input())):
    n = int(input())
    ar = list(map(int,input().split()))
    mn = sum(ar) - 2 * min(ar)
    mx = sum(ar) - 2 * max(ar)
    if (sum(ar) - mn) % (2 * (n - 2)) == 0 and (sum(ar) - mx) % (2 * (n - 2)) == 0:
        print(((sum(ar) - mn) // (2 * (n - 2))) * ((sum(ar) - mx) // (2 * (n - 2))))
    else: print(0)
