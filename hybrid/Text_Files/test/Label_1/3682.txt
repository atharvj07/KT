#include <bits/stdc++.h>

using namespace std;

int n, m, h, a[110], b[110], t[110][110];
vector<vector<int>> ans;

int main() {
  cin >> n >> m >> h;
  for (int i = 0; i < m; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  for (int x = 0; x < n; x++)
    for (int y = 0; y < m; y++) cin >> t[x][y];
  for (int i = 0; i < n; i++) {
    vector<int> tmp;
    for (int j = 0; j < m; j++) {
      tmp.push_back(0);
      for (int k = 0; k < a[j] && k < b[i]; k++) {
        if (t[i][j] == 0) {
          tmp[j] = k;
          break;
        }
        tmp[j] = k + 1;
        t[i][j] = 0;
      }
    }
    ans.push_back(tmp);
  }
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) cout << ans[i][j] << " ";
    cout << endl;
  }
  return 0;
}
