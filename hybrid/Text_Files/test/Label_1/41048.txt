import sys
def num(s):
    try:
        return int(s)
    except:
        return s

t=num(input())
for i in range(t):
    n=num(input())
    l=input().split()
    m=num(input())
    d={}
    for j in l:
        d[j]=""
    for j in range(m):
        ln=input().split(":")
        if ln[0]=='?':
            done=False
            for p in l:
                if not ln[1].__contains__(p) and not done and d[p]=="":
                    print(p+":"+ln[1])
                    d[p]='Y'
                    done=True
            if not done:
                print("Impossible")
                break;
        else:
            print(ln[0]+":"+ln[1])
