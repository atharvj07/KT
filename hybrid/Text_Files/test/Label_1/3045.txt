#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(2 * n);
    for (int i = 0; i < 2 * n; ++i) {
        std::cin >> a[i];
    }
    std::sort(a.begin(), a.end());
    std::cout << (a[n - 1] - a[0]) * (a[2 * n - 1] - a[n]) << std::endl;
    return 0;
}
