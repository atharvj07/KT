#include<bits/stdc++.h>
#include<vector>
using namespace std;
#define pb push_back
#define mp make_pair
#define endl '\n'
typedef long long int ll;
typedef pair<ll,ll>pi;

ll T,N,M;
vector<ll>G[100005],dp[100005],ans;
ll sum=0,tot=0,vis[100005];

void dfs(ll v,ll p,ll psum)
{
    dp[v].pb(0);
    dp[v].pb(1);
    vis[v]=1;
    for(auto u:G[v])
        if(!vis[u])
        {
            dfs(u,v,psum+1);
            for(auto i:dp[u])
                dp[v].pb(i+1);
        }
    sum+=psum;
}

void dfs1(ll v,ll p,ll psum)
{
    ll mx1=-1,mx2=-1;
    for(auto i:dp[v])
    {
        if(i>mx1) mx1=i;
    }
    for(auto i:dp[v])
    {
        if(i>0 && i!=mx1) mx2=max(mx2,i);
    }
    if(mx2==-1) mx2=0;
    if(psum>1) mx2=max(mx2,psum);
    tot=max(tot,psum+mx1+mx2-1);
    for(auto u:G[v])
        if(u!=p){
            vector<ll>temp;
            for(auto i:dp[v])
                if(i!=mx1)
                    temp.pb(i);
            if(mx1==dp[v][0])
                temp.pb(psum);
            dp[u]=temp;
            dfs1(u,v,psum+1);
        }
}

int main()
{
    cin>>N;
    tot=0,sum=0;
    for(ll i=1; i<N; i++){
        ll a,b;
        cin>>a>>b;
        G[a].pb(b);
        G[b].pb(a);
    }
    dfs(1,0,0);
    dfs1(1,0,0);
    if(sum%2) cout<<-1<<endl;
    else{
        if(tot>=sum) cout<<sum/2<<endl;
        else cout<<tot<<endl;
    }
}
