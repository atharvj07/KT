#include <iostream>
#include <vector>

using namespace std;

struct Edge {
    long long v, u, w;
};

int main() {
    long long n, m;
    cin >> n >> m;
    vector<long long> a(n);
    for (long long i = 0; i < n; i++) {
        cin >> a[i];
    }
    vector<Edge> e(m);
    for (long long i = 0; i < m; i++) {
        cin >> e[i].v >> e[i].u >> e[i].w;
        e[i].v--, e[i].u--;
    }
    long long ans = 0;
    for (const Edge &edge : e) {
        ans += edge.w - a[edge.v] - a[edge.u];
    }
    cout << ans << endl;
    return 0;
}
