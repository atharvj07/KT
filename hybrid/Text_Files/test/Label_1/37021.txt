from sys import stdin, stdout
n, m = map(int, stdin.readline().split())
grid = [list(stdin.readline().rstrip()) for i in range(n)]
visited = [[0] * m for i in range(n)]
def dfs(i, j, st):
    if i < 0 or i >= n or j < 0 or j >= m or visited[i][j] or grid[i][j] == '#' or grid[i][j] != st:
        return
    visited[i][j] = 1
    dfs(i, j + 1, st)
    dfs(i, j - 1, st)
    dfs(i + 1, j, st)
    dfs(i - 1, j, st)
flag = 0
for i in range(n):
    for j in range(m):
        if not visited[i][j] and grid[i][j] != '.':
            dfs(i, j, grid[i][j])
            flag += 1
if flag == 3:
    print(0)
else:
    print(-1)
