#include <iostream>
#include <string>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int ans = 0;
    int pre[n + 1];
    pre[0] = 0;
    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i] + (s[i] - '0');
    }
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j <= n; j++) {
            if (pre[j] - pre[i] == (j - i) / 2) {
                ans = max(ans, j - i);
            }
        }
    }
    cout << ans << endl;
    return 0;
}
