import sys

n = int(input())
s = input()
adj = [0] * 10
for i in range(n):
    adj[int(s[i])] |= 1 << i

vis = [False] * 10
for i in range(10):
    if vis[i]:
        continue
    q = [i]
    vis[i] = True
    while q:
        u = q.pop()
        for j in range(n):
            if adj[u] & (1 << j) and not vis[int(s[j])]:
                q.append(int(s[j]))
                vis[int(s[j])] = True
if vis.count(True) == 10:
    print("YES")
else:
    print("NO")
