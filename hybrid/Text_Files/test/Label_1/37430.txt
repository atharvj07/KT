n, m = map(int, input().split())
G = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    G[u].append(v)
    G[v].append(u)
vis = [False] * (n+1)
used = [False] * (n+1)
order = []
def dfs(node):
    vis[node] = True
    for u in G[node]:
        if not vis[u]:
            dfs(u)
    order.append(node)
for i in range(1, n+1):
    if not vis[i]:
        dfs(i)
if len(order) == n:
    print(*order[::-1])
else:
    print(-1)
