#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N = 2005, eps = 1e-6;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  ll n;
  cin >> n;
  vector<double> pb1(n), pb2(n);
  for (int i = 0; i < n; i++) { cin >> pb1[i]; }
  for (int i = 0; i < n; i++) { cin >> pb2[i]; }
  vector<double> pa(n), pb(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i >= j) {
        pa[i] += pb1[i] * pb2[j];
        pb[j] += pb1[i] * pb2[j];
      } else {
        pb[i] += pb1[i] * pb2[j];
        pa[j] += pb1[i] * pb2[j];
      }
    }
  }
  for (int i = 0; i < n; i++) {
    cout << fixed << setprecision(8) << pa[i] << ' ';
  }
  cout << '\n';
  for (int i = 0; i < n; i++) {
    cout << fixed << setprecision(8) << pb[i] << ' ';
  }
  cout << '\n';
  return 0;
}
