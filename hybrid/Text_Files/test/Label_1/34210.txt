import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long k = sc.nextLong();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        sc.close();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > k) {
                ans = Math.max(ans, (k + a[i] - 1) / a[i]);
            } else {
                ans = Math.max(ans, 1);
            }
        }
        System.out.println(ans);
    }
}
