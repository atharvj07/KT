#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
const int N = 1e5+20;
ld a[N], b[N];
ld ansa[N], ansb[N];
int n;
ld eps = 1e-6;
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    cin >> b[i];
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (i >= j) {
        ansa[i] += a[i] * b[j] / 1.0;
      }
      if (i <= j) {
        ansb[i] += b[i] * a[j] / 1.0;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << fixed << setprecision(8) << ansa[i] << " ";
  }
  cout << endl;
  for (int i = 1; i <= n; i++) {
    cout << fixed << setprecision(8) << ansb[i] << " ";
  }
  cout << endl;
  return 0;
}
