import sys

t = int(sys.stdin.readline())

for _ in range(t):
    n = int(sys.stdin.readline())
    a = list(range(1, n + 1))
    op = []
    for i in range(n - 2, 0, -1):
        if a[i] == 1:
            continue
        op.append([i + 1, a[i] + 1])
        a[i] = 1
    print(len(op))
    for x, y in op:
        print(x, y)
