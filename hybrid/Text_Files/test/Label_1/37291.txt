import sys
input = sys.stdin.readline

n, m = map(int, input().split())
d = list(map(int, input().split()))
e = []
for i in range(m):
    u, v = map(lambda x: int(x) - 1, input().split())
    e.append((u, v))

odd = []
even = []
for i in range(n):
    if d[i] == 0:
        if len(e[i]) % 2 == 1:
            odd.append(i)
        else:
            even.append(i)
    elif d[i] == 1:
        if len(e[i]) % 2 == 0:
            odd.append(i)
        else:
            even.append(i)
    elif d[i] == -1:
        if len(e[i]) % 2 == 1:
            continue
        else:
            odd.append(i)
    else:
        print(-1)
        sys.exit()

if len(odd) % 2 == 1:
    print(-1)
    sys.exit()

ans = []
for i in range(len(odd)):
    if len(e[odd[i]]) > 1:
        ans.append(i + 1)
    else:
        even.append(odd[i])

for i in range(len(even)):
    if len(e[even[i]]) > 1:
        ans.append(i + 1)

print(len(ans))
for i in ans:
    print(i)
