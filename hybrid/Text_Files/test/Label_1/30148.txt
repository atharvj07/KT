import java.util.Scanner;

public class TownTraversal {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = scanner.nextInt();
            roads[i][1] = scanner.nextInt();
        }

        boolean[][] adjacencyMatrix = new boolean[5][5];
        for (int i = 0; i < 3; i++) {
            adjacencyMatrix[roads[i][0]][roads[i][1]] = true;
            adjacencyMatrix[roads[i][1]][roads[i][0]] = true;
        }

        boolean[] visited = new boolean[5];
        boolean isPossible = depthFirstSearch(adjacencyMatrix, visited, 1);

        System.out.println(isPossible ? "YES" : "NO");
    }

    private static boolean depthFirstSearch(boolean[][] adjacencyMatrix, boolean[] visited, int currentTown) {
        if (visited[currentTown]) {
            return false;
        }

        visited[currentTown] = true;

        for (int nextTown = 1; nextTown < 5; nextTown++) {
            if (adjacencyMatrix[currentTown][nextTown] && !depthFirstSearch(adjacencyMatrix, visited, nextTown)) {
                return false;
            }
        }

        return true;
    }
}
