#include<bits/stdc++.h>
using namespace std;
int main(){
   int t; cin >> t;
   while(t--){
      int n; cin >> n;
      vector<int> h(n);
      for(int &x:h) cin >> x;
      int j=0;
      for(int &x:h){
         if(x) j++;
         x=j;
      }
      cout<<((is_sorted(h.begin(), h.end()))?"YES\n":"NO\n");
   }
}
