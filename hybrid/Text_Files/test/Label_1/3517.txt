#include<bits/stdc++.h>
using namespace std;
#define mp make_pair
int r, c, n, k, vio[11], ans = 0;
int main(){
	cin >> r >> c >> n >> k;
	for(int i = 1; i <= n; i++){
		int x, y;
		cin >> x >> y;
		vio[x*(r+1)+y]++;
	}
	for(int i = 1; i <= r; i++){
		int sum = 0;
		for(int j = 1; j <= c; j++){
			sum += vio[i*(r+1)+j];
			if(sum >= k) ans += c - j + 1;
		}
	}
	for(int j = 1; j <= c; j++){
		int sum = 0;
		for(int i = 1; i <= r; i++){
			sum += vio[i*(r+1)+j];
			if(sum >= k) ans += r - i + 1;
		}
	}
	cout << ans << endl;
	return 0;
}
