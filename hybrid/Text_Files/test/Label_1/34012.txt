import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    char[] c = in.next().toCharArray();
    int[] f = new int[n];
    Arrays.fill(f, -1);
    int res = 1;
    for (int i = 0; i < n; i++) {
      if (f[i] != -1) {
        res += f[i];
        continue;
      }
      char cc = c[i];
      int len = 1;
      for (int j = i + 1; j < n && c[j] == cc; j++) {
        len++;
      }
      res += len;
      for (int j = i; j < i + len; j++) {
        f[j] = len;
      }
      i += len - 1;
    }
    System.out.println(res % (int) (1e9 + 7));
  }
}
