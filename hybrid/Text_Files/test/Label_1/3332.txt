#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int q;
    cin >> q;
    while (q--) {
        int n, r;
        cin >> n >> r;
        int pos[n];
        for (int i = 0; i < n; i++) {
            cin >> pos[i];
        }
        sort(pos, pos + n);
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (cur < pos[i]) {
                cur = pos[i] + r;
                ans++;
            }
        }
        cout << ans << endl;
    }
    return 0;
}
