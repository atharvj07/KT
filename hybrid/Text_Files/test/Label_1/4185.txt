#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  string k, s, t;
  cin >> k >> s >> t;
  int cnt1 = 0, cnt2 = 0, cnt3 = 0;
  for (int i = 0; i < k.size(); i++) {
    if (k[i] == s[i]) cnt1++;
    if (k[i] == t[i]) cnt2++;
    if (s[i] == t[i]) cnt3++;
  }
  if ((k.size() + n - 2 * cnt1) % 2 && (k.size() + n - 2 * cnt2) % 2 && (k.size() + n - 2 * cnt3) % 2) {
    cout << "Kuro";
  } else if ((k.size() + n - 2 * cnt1) % 2 && (k.size() + n - 2 * cnt3) % 2 == 0) {
    cout << "Shiro";
  } else if ((k.size() + n - 2 * cnt2) % 2 && (k.size() + n - 2 * cnt1) % 2 == 0) {
    cout << "Katie";
  } else if ((k.size() + n - 2 * cnt3) % 2 && (k.size() + n - 2 * cnt2) % 2 == 0) {
    cout << "Kuro";
  } else {
    cout << "Draw";
  }
  return 0;
}
