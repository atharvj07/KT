import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int x1 = in.nextInt();
        int x2 = in.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        if (x1 + x2 > Arrays.stream(c).sum()) {
            System.out.println("No");
        } else {
            List<Integer> s1 = new ArrayList<>();
            List<Integer> s2 = new ArrayList<>();
            int used = 0;
            for (int i = 0; i < n; i++) {
                if (used + c[i] <= x1) {
                    s1.add(i + 1);
                    used += c[i];
                } else if (used + c[i] <= x1 + x2) {
                    s2.add(i + 1);
                    used += c[i];
                } else {
                    break;
                }
            }
            if (used == x1 + x2) {
                System.out.println("Yes");
                System.out.println(s1.size() + " " + s2.size());
                for (int i : s1) {
                    System.out.print(i + " ");
                }
                System.out.println();
                for (int i : s2) {
                    System.out.print(i + " ");
                }
                System.out.println();
            } else {
                System.out.println("No");
            }
        }
    }
}
