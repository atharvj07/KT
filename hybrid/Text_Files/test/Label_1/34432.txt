import java.util.Scanner;

public class ValueAssignment {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = scanner.nextInt() - 1;
        }

        // Check for cycles
        boolean[] visited = new boolean[n];
        boolean[] stack = new boolean[n];
        boolean hasCycle = false;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                hasCycle = dfs(i, p, visited, stack);
                if (hasCycle) {
                    break;
                }
            }
        }

        if (hasCycle) {
            System.out.println("IMPOSSIBLE");
            return;
        }

        // Assign values to vertices
        int[] values = new int[n];
        for (int i = 0; i < n; i++) {
            values[i] = i;
        }

        // Check for conflicts
        for (int i = 0; i < n; i++) {
            int j = p[i];
            if (values[i] == values[j]) {
                System.out.println("IMPOSSIBLE");
                return;
            }
        }

        // Check for missing values
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < values[i]; j++) {
                boolean found = false;
                for (int k = 0; k < n; k++) {
                    if (p[k] == i && values[k] == j) {
                        found = true;
                        break;
                    }
                }
                if (!found) {
                    System.out.println("IMPOSSIBLE");
                    return;
                }
            }
        }

        System.out.println("POSSIBLE");
    }

    private static boolean dfs(int i, int[] p, boolean[] visited, boolean[] stack) {
        visited[i] = true;
        stack[i] = true;
        int j = p[i];
        if (!visited[j]) {
            if (dfs(j, p, visited, stack)) {
                return true;
            }
        } else if (stack[j]) {
            return true;
        }
        stack[i] = false;
        return false;
    }
}
