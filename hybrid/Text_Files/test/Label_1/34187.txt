import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Party {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int w = scanner.nextInt();

        int[] weights = new int[n];
        int[] beauties = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            beauties[i] = scanner.nextInt();
        }

        List<List<Integer>> friends = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            friends.add(Arrays.asList(x, y));
        }

        System.out.println(solve(n, m, w, weights, beauties, friends));
    }

    private static int solve(int n, int m, int w, int[] weights, int[] beauties, List<List<Integer>> friends) {
        int[] dp = new int[w + 1];
        for (int i = 0; i < n; i++) {
            int weight = weights[i];
            int beauty = beauties[i];
            for (int j = w; j >= weight; j--) {
                dp[j] = Math.max(dp[j], dp[j - weight] + beauty);
            }
        }

        for (List<Integer> friend : friends) {
            int x = friend.get(0);
            int y = friend.get(1);
            if (weights[x] + weights[y] <= w) {
                for (int j = w; j >= weights[x] + weights[y]; j--) {
                    dp[j] = Math.max(dp[j], dp[j - weights[x] - weights[y]] + beauties[x] + beauties[y]);
                }
            }
        }

        return dp[w];
    }
}
