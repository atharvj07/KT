import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int m = sc.nextInt();
        int n = sc.nextInt();
        int[] table = new int[h];
        int dummyCalls = 0;
        for (int i = 0; i < n; i++) {
            char op = sc.next().charAt(0);
            if (op == '+') {
                int id = sc.nextInt();
                int hash = sc.nextInt();
                int idx = id % h;
                while (table[idx] > 0) {
                    idx = (idx + m) % h;
                    dummyCalls++;
                }
                table[idx] = id;
            } else {
                table[sc.nextInt() % h] = 0;
            }
        }
        System.out.println(dummyCalls);
    }
}
