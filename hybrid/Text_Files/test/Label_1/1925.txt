#include <iostream>
#include <vector>
using namespace std;
int n, q, op;
vector<int> a, v;
int main() {
  cin >> n >> q;
  a.resize(n);
  for (auto &x : a) cin >> x;
  while (q--) {
    cin >> op;
    if (op == 1) {
      int i, val;
      cin >> i >> val;
      --i;
      a[i] = val;
    } else if (op == 2) {
      int val;
      cin >> val;
      for (auto &x : a) x += val;
    } else {
      int i;
      cin >> i;
      --i;
      cout << a[i] << '\n';
    }
  }
}
