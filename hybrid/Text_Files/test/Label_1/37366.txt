n=int(input())
b=list(map(int,input().split()))
c=0
for i in range(1,n-1):
 if (b[i]-b[i-1])!=(b[i+1]-b[i]): c=1
if c==0: print(0)
else:
 a=[]
 for i in range(1,n-1):
  if (b[i]-b[i-1])!=(b[i+1]-b[i]):
   a.append((i,b[i-1],b[i],b[i+1]))
 ans=10000000
 for x in a:
  for y in a:
   if x==y: continue
   ans=min(ans,abs(x[1]-y[3])+abs(x[2]-y[2]))
   ans=min(ans,abs(x[1]-y[2])+abs(x[2]-y[3]))
 print(ans)
