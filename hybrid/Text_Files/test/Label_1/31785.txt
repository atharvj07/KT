import java.util.Scanner;

public class Main {
    private static final int MOD = 1000000007;

    private static int n;
    private static int[][] p;
    private static int[][] g;
    private static int[] x;
    private static int[] y;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        p = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                p[i][j] = sc.nextInt();
            }
        }
        g = new int[2 * n][2 * n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                g[i][n + j] = p[i][j];
            }
        }
        x = new int[2 * n];
        y = new int[2 * n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans = add(ans, dfs(i));
        }
        System.out.println(ans);
    }

    private static int dfs(int u) {
        if (x[u] != -1) {
            return 0;
        }
        x[u] = y[u] = 1;
        int ans = 1;
        for (int v = 0; v < 2 * n; v++) {
            if (g[u][v] == 0) {
                continue;
            }
            if (y[v] == 1) {
                continue;
            }
            if (dfs(v) == 0) {
                return 0;
            }
            x[u] = add(x[u], x[v]);
            y[u] = add(y[u], y[v]);
            ans = mul(ans, x[v]);
            ans = mul(ans, mul(x[u], y[u]));
        }
        return ans;
    }

    private static int add(int a, int b) {
        return (a + b) % MOD;
    }

    private static int mul(int a, int b) {
        return (int) ((long) a * b % MOD);
    }
}
