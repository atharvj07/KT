#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    map<int, int> freq;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        freq[a[i]]++;
    }
    int petya, vasya;
    bool fair = false;
    for (auto &kv : freq) {
        if (kv.second % 2 == 1) {
            cout << "NO" << endl;
            return 0;
        }
        if (kv.second >= 2) {
            if (!fair) {
                petya = kv.first;
                fair = true;
            } else {
                vasya = kv.first;
            }
        }
    }
    if (fair) {
        cout << "YES" << endl;
        cout << petya << " " << vasya << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
