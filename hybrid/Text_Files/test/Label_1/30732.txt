import java.util.Scanner;

public class DominoCarpet {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine(); // Consume newline

        int[][] carpet = new int[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.nextLine();
            for (int j = 0; j < m; j++) {
                carpet[i][j] = (line.charAt(3 * j + 1) == 'O') ? 1 : 0;
            }
        }

        long[][][] dp = new long[n][m][4]; // dp[i][j][k]: number of ways to fill the carpet up to (i, j) with state k
        // state 0: no horizontal domino on the left, 1: horizontal domino on the left, 2: no vertical domino above, 3: vertical domino above
        dp[0][0][0] = 1; // Base case: empty carpet

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (carpet[i][j] == 0) {
                    // If the square is empty, we can place a horizontal or vertical domino
                    if (j > 0) {
                        dp[i][j][0] = (dp[i][j][0] + dp[i][j - 1][1]) % MOD; // Place horizontal domino
                        dp[i][j][2] = (dp[i][j][2] + dp[i][j - 1][3]) % MOD; // Place vertical domino
                    }
                    if (i > 0) {
                        dp[i][j][1] = (dp[i][j][1] + dp[i - 1][j][2]) % MOD; // Place horizontal domino
                        dp[i][j][3] = (dp[i][j][3] + dp[i - 1][j][0]) % MOD; // Place vertical domino
                    }
                } else {
                    // If the square is occupied, we can only place a domino that fits
                    if (j > 0 && carpet[i][j - 1] == 0) {
                        dp[i][j][0] = (dp[i][j][0] + dp[i][j - 1][1]) % MOD; // Place horizontal domino
                    }
                    if (i > 0 && carpet[i - 1][j] == 0) {
                        dp[i][j][2] = (dp[i][j][2] + dp[i - 1][j][0]) % MOD; // Place vertical domino
                    }
                }
            }
        }

        long result = 0;
        for (int k = 0; k < 4; k++) {
            result = (result + dp[n - 1][m - 1][k]) % MOD;
        }

        System.out.println(result);
    }
}
