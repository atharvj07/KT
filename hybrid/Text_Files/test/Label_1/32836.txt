import java.util.*;
class Main {
    public static void main(String[] args) {       
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long[] a = new long[n];
        long[] dp = new long[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        dp[1] = a[0];
        for (int i = 2; i <= n; i++) {
            dp[i] = Long.MAX_VALUE;
            for (int j = (i - k <= 0 ? 1 : i - k); j < i; j++) {
                dp[i] = Math.min(dp[i], dp[j] + a[i - 1]);
            }
        }
        int idx = 1;
        long min = Long.MAX_VALUE;
        for (int i = 1; i <= n; i++) {
            if (dp[i] < min) {
                min = dp[i];
                idx = i;
            }
        }
        System.out.println(idx);
    }
}
