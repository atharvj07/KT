MOD = 998244353

def modinv(a, m = MOD):
    return pow(a, m - 2, m)

def modfact(n, m = MOD):
    ans = 1
    for i in range(1, n + 1):
        ans = (ans * i) % m
    return ans

def solve(n, a):
    sum_a = sum(a)
    max_a = max(a)
    if max_a > (sum_a + 1) // 2:
        return 0
    if sum_a % 2 == 0:
        return (modfact(n) * modfact(n)) % MOD * modinv(modfact(2 * n)) % MOD
    else:
        return modfact(n) * modinv(modfact(2 * n - 1)) % MOD * modfact(n - 1) % MOD

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(solve(n, a))
