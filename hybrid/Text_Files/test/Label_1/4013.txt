#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n, m, ans = 0;
  std::cin >> n >> m;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) {
    std::cin >> a[i];
  }
  std::sort(a.rbegin(), a.rend());
  for (int i = 0; i < n; i++) {
    if (m <= 0) break;
    m -= a[i];
    ans++;
  }
  std::cout << ans << std::endl;
  return 0;
}
