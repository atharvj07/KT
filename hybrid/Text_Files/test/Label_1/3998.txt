#include<bits/stdc++.h>
using namespace std;
#define ll long long

inline int sum(ll x) {
    int ans=0;
    while(x>0) {
        ans+=(x%10);
        x/=10;
    }
    return ans;
}

int main()
{
    ll num;
    cin>>num;
    num++;
    ll ten=num/10;
    ll one=num%10;
    cout<<(sum(ten)*(one!=0)+(one!=0))<<endl;
    return 0;
}
