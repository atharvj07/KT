n = int(input())
a = list(map(int, input().split()))

def bubble_sort_operations(arr):
    operations = []
    for i in range(len(arr) - 1):
        for j in range(0, len(arr) - i - 1, 2):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                operations.append((j + 1, j + 2))
    return operations

def adjust_operations(operations):
    adjusted = []
    for op in operations:
        if not adjusted or adjusted[-1][1] != op[0]:
            adjusted.append(op)
        else:
            adjusted[-1] = (adjusted[-1][0], op[1])
    return adjusted

operations = bubble_sort_operations(a)
adjusted_operations = adjust_operations(operations)

for op in adjusted_operations:
    print(op[0], op[1])
