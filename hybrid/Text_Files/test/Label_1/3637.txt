#include <vector>
#include <cstdio>
#include <algorithm>
#define ll long long

using namespace std;

int main() {
    ll n, p, a, b;
    scanf("%lld %lld", &n, &p);
    vector<ll> v;
    double ans = 0;
    for (int i = 0; i < n; i++) {
        scanf("%lld %lld", &a, &b);
        if (a > p) v.push_back(b * 1.0 / (a - p));
    }
    if (v.empty()) {
        printf("-1\n");
        return 0;
    }
    sort(v.begin(), v.end());
    printf("%.9f\n", v[0]);
    return 0;
}
