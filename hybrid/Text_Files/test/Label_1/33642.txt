import java.util.Arrays;

public class Main {
    private static long[] masks;
    private static long state;

    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int n = sc.nextInt();
        masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            long mask = 0L;
            for (int j = 0; j < k; j++) {
                int pos = sc.nextInt();
                mask |= 1L << pos;
            }
            masks[i] = mask;
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            int m = sc.nextInt();
            switch (op) {
                case 1:
                    set(m);
                    break;
                case 2:
                    clear(m);
                    break;
                case 3:
                    flip(m);
                    break;
                case 4:
                    System.out.println(all(m));
                    break;
                case 5:
                    System.out.println(any(m));
                    break;
                case 6:
                    System.out.println(none(m));
                    break;
                case 7:
                    System.out.println(count(m));
                    break;
                case 8:
                    System.out.println(val(m));
                    break;
                case 0:
                    System.out.println(test(m));
                    break;
            }
        }
    }

    private static int test(int mask) {
        return (state & masks[mask]) != 0L ? 1 : 0;
    }

    private static void set(int mask) {
        state |= masks[mask];
    }

    private static void clear(int mask) {
        state &= ~masks[mask];
    }

    private static void flip(int mask) {
        state ^= masks[mask];
    }

    private static int all(int mask) {
        return (state & masks[mask]) == masks[mask] ? 1 : 0;
    }

    private static int any(int mask) {
        return (state & masks[mask]) != 0L ? 1 : 0;
    }

    private static int none(int mask) {
        return (state & masks[mask]) == 0L ? 1 : 0;
    }

    private static int count(int mask) {
        return Long.bitCount(state & masks[mask]);
    }

    private static long val(int mask) {
        return state & masks[mask];
    }
}
