n, m = [int(i) for i in input().split()]
field = [input() for _ in range(n)]
ans = []

for x in range(n):
    for y in range(m):
        if field[x][y] == '*':
            vis = [[False] * m for i in range(n)]
            cnt = 0

            def dfs(i, j):
                if i < 0 or j < 0 or i >= n or j >= m:
                    return
                if not vis[i][j] and field[i][j] == '.':
                    vis[i][j] = True
                    cnt += 1
                    dfs(i - 1, j)
                    dfs(i + 1, j)
                    dfs(i, j - 1)
                    dfs(i, j + 1)

            dfs(x, y)
            ans.append(str(cnt % 10))
        else:
            ans.append('.')
    ans.append('\n')
print(''.join(ans).rstrip())
