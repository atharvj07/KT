from collections import deque

def bfs(adj, start, n):
    distance = [-1] * (n + 1)
    queue = deque([start])
    distance[start] = 0
    while queue:
        current = queue.popleft()
        for neighbor in adj[current]:
            if distance[neighbor] == -1:
                distance[neighbor] = distance[current] + 1
                queue.append(neighbor)
    return distance

def solve(n, m, d, affected, paths):
    adj = [[] for _ in range(n + 1)]
    for a, b in paths:
        adj[a].append(b)
        adj[b].append(a)

    # Find the farthest affected settlement from any affected settlement
    distance_from_first_affected = bfs(adj, affected[0], n)
    farthest_affected = affected[0]
    max_dist = -1
    for settlement in affected:
        if distance_from_first_affected[settlement] > max_dist:
            max_dist = distance_from_first_affected[settlement]
            farthest_affected = settlement

    # Find the farthest settlement from the farthest affected settlement
    distance_from_farthest_affected = bfs(adj, farthest_affected, n)
    second_farthest_affected = farthest_affected
    max_dist = -1
    for i in range(1, n + 1):
        if distance_from_farthest_affected[i] > max_dist:
            max_dist = distance_from_farthest_affected[i]
            second_farthest_affected = i

    # Find settlements within d distance from the center of the two farthest affected settlements
    distance_from_second_farthest = bfs(adj, second_farthest_affected, n)
    center = (distance_from_farthest_affected[second_farthest_affected] + 1) // 2
    possible_locations = 0
    for i in range(1, n + 1):
        if distance_from_farthest_affected[i] <= d and distance_from_second_farthest[i] <= d:
            possible_locations += 1

    return possible_locations

n, m, d = map(int, input().split())
affected = list(map(int, input().split()))
paths = [tuple(map(int, input().split())) for _ in range(n - 1)]
print(solve(n, m, d, affected, paths))
