import sys
input=lambda : sys.stdin.readline()

def solve():
    n=int(input())
    l=list(map(int,input().split()))
    l.sort()
    flag=1
    for i in range(1,2*n-1):
        if l[i]==((l[i-1]+l[i+1])/2):
            flag=0
            break
    if flag==0:
        print(-1)
    else:
        for i in range(0,2*n-1,2):
            print(l[i],l[i+1])
            
t=int(input())
while t>0:
    solve()
    t-=1
