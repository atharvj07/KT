import sys
mod = 10**9 + 7
def solve(n:int,e:list) -> int:
    adj = [[] for i in range(n+1)]
    for i in e:
        adj[i[0]].append(i[1])
        adj[i[1]].append(i[0])
    dp = [[-1 for i in range(2)] for j in range(n+1)]
    def dfs(node:int,prv:int,cas:int)->int:
        if dp[node][cas]!=-1:
            return dp[node][cas]
        ans = 1
        for nbr in adj[node]:
            if nbr!=prv:
                if cas==0:
                    ans = ans * (1+dfs(nbr,node,1)) % mod
                    ans = ans * (1+dfs(nbr,node,0)) % mod
                else:
                    ans = ans * (dfs(nbr,node,0)) % mod
        dp[node][cas] = ans
        return ans
    return dfs(1,1,0)
n = int(input())
e = []
for i in range(n-1):
    x,y = map(int,input().split())
    e.append([x,y])
print(solve(n,e))
