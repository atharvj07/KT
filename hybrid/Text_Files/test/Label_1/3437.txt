#include <iostream>
using namespace std;

int canPlace(int a1, int b1, int a2, int b2, int a3, int b3) {
  if (a2 > a1 || b2 > b1) return 0;
  if (a3 > a1 || b3 > b1) return 0;
  return 1;
}
int main() {
  int a1, b1, a2, b2, a3, b3;
  cin >> a1 >> b1;
  cin >> a2 >> b2;
  cin >> a3 >> b3;
  cout << (canPlace(a1, b1, a2, b2, a3, b3) || canPlace(a1, b1, a2, b2, b3, a3) ||
                 canPlace(a1, b1, b2, a2, a3, b3) || canPlace(a1, b1, b2, a2, b3, a3)
               ? "YES"
               : "NO")
      << endl;
}
