t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    p = list(map(int, input().strip().split()))
    peaks = [p[0]]
    for i in range(1, n-1):
        if (p[i] > p[i-1] and p[i] > p[i+1]) or (p[i] < p[i-1] and p[i] < p[i+1]):
            peaks.append(p[i])
    peaks.append(p[-1])
    print(len(peaks))
    print(" ".join(map(str, peaks)))
