import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Auction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[][] bids = new int[n][2];
        for (int i = 0; i < n; i++) {
            bids[i][0] = scanner.nextInt() - 1;
            bids[i][1] = scanner.nextInt();
        }

        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            int[] absent = new int[k];
            for (int j = 0; j < k; j++) {
                absent[j] = scanner.nextInt() - 1;
            }

            int winner = findWinner(bids, absent);
            if (winner != -1) {
                System.out.println(winner + 1 + " " + bids[winner][1]);
            } else {
                System.out.println("0 0");
            }
        }
    }

    private static int findWinner(int[][] bids, int[] absent) {
        List<Integer> remaining = new ArrayList<>();
        for (int i = 0; i < bids.length; i++) {
            if (!contains(absent, bids[i][0])) {
                remaining.add(i);
            }
        }

        int winner = -1;
        int maxBid = -1;
        for (int i : remaining) {
            if (bids[i][1] > maxBid) {
                winner = i;
                maxBid = bids[i][1];
            }
        }

        return winner;
    }

    private static boolean contains(int[] arr, int value) {
        for (int i : arr) {
            if (i == value) {
                return true;
            }
        }
        return false;
    }
}
