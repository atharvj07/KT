from collections import defaultdict, deque

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve():
    n = int(input())
    x = list(map(int, input().split()))
    g = defaultdict(list)
    for _ in range(n - 1):
        a, b = map(int, input().split())
        g[a].append(b)
        g[b].append(a)
    sum = 0
    q = deque()
    q.append((1, 0))
    while q:
        u, p = q.popleft()
        sum += x[u - 1]
        for v in g[u]:
            if v != p:
                x[v - 1] = gcd(x[u - 1], x[v - 1])
                q.append((v, u))
    return sum

print(solve())
