#include <cstdio>

int main() {
  int t; scanf("%d", &t);
  while(t--) {
    int r, g, b; scanf("%d %d %d", &r, &g, &b);
    if(r == 0 || g == 0 || b == 0) { printf("No\n"); continue; }
    if(r == 1 && g == 1 && b == 1) { printf("No\n"); continue; }
    if(r == 1 && g <= 1 && b <= 1) { printf("No\n"); continue; }
    if(g == 1 && r <= 1 && b <= 1) { printf("No\n"); continue; }
    if(b == 1 && r <= 1 && g <= 1) { printf("No\n"); continue; }
    printf("Yes\n");
  }
  return 0;
}
