import java.util.*;
import java.io.*;

class Main {
    static int[] dist;
    static ArrayList<Edge>[] edge;
    static boolean[] visited;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        dist = new int[n];
        edge = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            edge[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            int w = Integer.parseInt(st.nextToken());

            edge[a].add(new Edge(b, w));
        }

        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            visited = new boolean[n];
            Arrays.fill(dist, Integer.MAX_VALUE);
            dist[i] = 0;

            if (dfs(i, i, 0)) {
                ans = Math.min(ans, dist[i]);
            }
        }

        if (ans == Integer.MAX_VALUE) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }

    static boolean dfs(int from, int start, int cnt) {
        if (cnt == edge.length - 1) {
            for (Edge e : edge[from]) {
                if (e.to == start) {
                    dist[from] = Math.min(dist[from], dist[start] + e.w);
                    return true;
                }
            }
        }

        for (Edge e : edge[from]) {
            if (!visited[e.to]) {
                visited[e.to] = true;
                dist[e.to] = dist[from] + e.w;
                if (dfs(e.to, start, cnt + 1)) {
                    return true;
                }
                visited[e.to] = false;
            }
        }

        return false;
    }

    static class Edge {
        int to, w;

        public Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }
}
