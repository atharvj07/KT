#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<long long int> dp(n);
    dp[0] = a[0];
    long long int ans = dp[0];
    for (int i = 1; i < n; i++) {
      dp[i] = max(dp[i - 1] + a[i], a[i]);
      ans = max(dp[i], ans);
    }
    cout << ans << endl;
  }
}
