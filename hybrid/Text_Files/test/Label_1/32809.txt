import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        int n = s.length();
        int c = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'o') {
                c++;
            }
        }
        if (c == 0 || c == n) {
            System.out.println("YES");
        } else if ((n - c) % (c - 1) == 0) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
