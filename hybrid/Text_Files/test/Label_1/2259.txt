#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> m(n);
  for (int i = 0; i < n; i++) {
    cin >> m[i];
  }
  vector<long long> ans(n);
  ans[0] = m[0];
  for (int i = 1; i < n; i++) {
    ans[i] = min(m[i], ans[i - 1] + 1);
  }
  for (int i = n - 2; i >= 0; i--) {
    ans[i] = min(ans[i], ans[i + 1] + 1);
  }
  for (int i = 0; i < n; i++) {
    cout << ans[i] << " ";
  }
  cout << endl;
  return 0;
}
