import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[][] dp = new int[n][n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            dp[i][0] = Integer.MAX_VALUE / 2;
        }
        dp[0][0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= i; j++) {
                if (dp[i][j] < Integer.MAX_VALUE / 2) {
                    if (j + 1 < n && a[j + 1] <= i) {
                        dp[j + 1][j + 1] = Math.min(dp[j + 1][j + 1], dp[i][j]);
                    }
                    if (j - 1 >= 0 && a[j - 1] <= i) {
                        dp[j - 1][j] = Math.min(dp[j - 1][j], dp[i][j] + 1);
                    }
                }
            }
        }
        int ans = Integer.MAX_VALUE / 2;
        for (int i = n - 1; i >= 0; i--) {
            if (dp[i][n - 1] < Integer.MAX_VALUE / 2) {
                ans = Math.min(ans, dp[i][n - 1]);
            }
        }
        System.out.println(ans);
    }
}
