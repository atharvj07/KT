#include <bits/stdc++.h>
using namespace std;
int main(int argc, char const *argv[])
{
    int n, room[100000], f = 10;
    string s;
    cin >> n >> s;
    for (int i = 0; i < 10; i++)
        room[i] = 0;
    for (int i = 0; i < n; i++)
    {
        if (s[i] == 'L')
        {
            for (int j = 0; j < 10; j++)
            {
                if (!room[j])
                {
                    room[j] = 1;
                    break;
                }
            }
        }
        else if (s[i] == 'R')
        {
            for (int j = 9; j >= 0; j--)
            {
                if (!room[j])
                {
                    room[j] = 1;
                    break;
                }
            }
        }
        else if (s[i] == '0')
            room[0] = 0;
        else if (s[i] == '1')
            room[1] = 0;
        else if (s[i] == '2')
            room[2] = 0;
        else if (s[i] == '3')
            room[3] = 0;
        else if (s[i] == '4')
            room[4] = 0;
        else if (s[i] == '5')
            room[5] = 0;
        else if (s[i] == '6')
            room[6] = 0;
        else if (s[i] == '7')
            room[7] = 0;
        else if (s[i] == '8')
            room[8] = 0;
        else if (s[i] == '9')
            room[9] = 0;
    }
    for (int i = 0; i < 10; i++)
        cout << room[i];
    cout << endl;
    return 0;
}
