import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] stars = new int[n][2];
        for (int i = 0; i < n; i++) {
            stars[i][0] = sc.nextInt();
            stars[i][1] = sc.nextInt();
        }

        int[] res = new int[3];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    int ax = stars[i][0], ay = stars[i][1];
                    int bx = stars[j][0], by = stars[j][1];
                    int cx = stars[k][0], cy = stars[k][1];
                    int area = (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by));
                    if (area == 0) continue;
                    boolean out = true;
                    for (int l = 0; l < n; l++) {
                        int lx = stars[l][0], ly = stars[l][1];
                        if (l == i || l == j || l == k) continue;
                        int area1 = (ax * (by - ly) + bx * (ly - ay) + lx * (ay - by));
                        int area2 = (bx * (cy - ly) + cx * (ly - by) + lx * (by - cy));
                        int area3 = (cx * (ay - ly) + ax * (ly - cy) + lx * (cy - ay));
                        if (area1 * area2 * area3 <= 0) {
                            out = false;
                            break;
                        }
                    }
                    if (out) {
                        res[0] = i + 1;
                        res[1] = j + 1;
                        res[2] = k + 1;
                        System.out.println(res[0] + " " + res[1] + " " + res[2]);
                        return;
                    }
                }
            }
        }
    }
}
