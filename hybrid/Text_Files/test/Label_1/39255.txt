from sys import stdin
input = stdin.readline

n, m, k, q = map(int, input().split())
b = sorted(map(int, input().split()))
t = []
ans = 1 << 60
for i in range(k): t.append(tuple(map(int, input().split())))
for i in range(k):
    r, c = t[i]
    if c in b: ans = min(ans, abs(n - r) + c)
    else: ans = min(ans, abs(n - r) + m + c)
print(ans)
