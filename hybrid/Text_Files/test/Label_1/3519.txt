#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    int l = 0, r = 0;
    for (int x : a) {
      if (x > 3)
        r++;
      else if (x == 3)
        l++;
    }
    cout << min(l / 2, r) << "\n";
  }
  return 0;
}
