#include <bits/stdc++.h>
using namespace std;
int main() {
  long long t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    vector<int> cnt(26, 0);
    long long dif = 0;
    for (char c : s) cnt[c - 'a']++;
    for (int i = 0; i < 26; i++) {
      if (cnt[i] > 1) { dif += cnt[i] - 1; }
    }
    cout << dif << endl;
  }
  return 0;
}
