#include <iostream>

const int mod = 1e9 + 7;

int main() {
    int p, k;
    std::cin >> p >> k;
    if (p < k) {
        std::cout << 1 << '\n' << p << '\n';
        return 0;
    }
    p = (p + k - 1) / k;
    int d = (63 - __builtin_clzll(p)) / 3 + 3;
    std::cout << d << '\n';
    for (int i = 0; i < d - 1; i++) {
        std::cout << p % k << ' ';
        p /= k;
    }
    std::cout << p % k << '\n';
}
