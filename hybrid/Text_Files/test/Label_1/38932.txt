import sys

n = int(sys.stdin.readline())
b = [int(x) for x in sys.stdin.readline().split()]
a = [int(x) for x in sys.stdin.readline().split()]
for i in range(n - 1):
    x, k = map(int, sys.stdin.readline().split())
    b[x - 1] += b[i] * k
    b[i] = 0

ans = 'YES'
for i in range(n):
    if b[i] < a[i]:
        ans = 'NO'
        break

print(ans)
