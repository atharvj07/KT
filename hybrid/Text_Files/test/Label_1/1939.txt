#include <bits/stdc++.h>
using namespace std;
#define eps 1e-6
int main() {
    double w, h, a;
    cin >> w >> h >> a;
    a = a * M_PI / 180;
    double b = sqrt(h * h + w * w);
    double ans = w * h + 0.5 * b * b * (1 - cos(a));
    cout << fixed << setprecision(9) << ans << endl;
    return 0;
}
