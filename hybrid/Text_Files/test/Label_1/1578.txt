#include <bits/stdc++.h>
using namespace std;
const int MAXN = 15;
char cake[MAXN][MAXN];
int dp[MAXN][MAXN][2][2];
int r, c;

int calc(int x, int y, int ateRow, int ateCol){
  if(x == r || y == c){
    return 0;
  }
  if(dp[x][y][ateRow][ateCol] != -1){
    return dp[x][y][ateRow][ateCol];
  }
  int ret = 0;
  if(!ateRow && cake[x][y] == '.'){
    ret = calc(x + 1, y, true, ateCol) + c - y;;
  }
  if(!ateCol && cake[x][y] == '.'){
    ret = max(ret, calc(x, y + 1, ateRow, true) + r - x);
  }
  ret = max(ret, calc(x + 1, y, ateRow, ateCol));
  ret = max(ret, calc(x, y + 1, ateRow, ateCol));
  return dp[x][y][ateRow][ateCol] = ret;
}

int main(){
  scanf("%d%d", &r, &c);
  for(int i=0; i<r; ++i){
    scanf("%s", cake[i]);
  }
  memset(dp, -1, sizeof dp);
  printf("%d\n", calc(0, 0, 0, 0));
  return 0;
}
