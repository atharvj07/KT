#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, x, m;
    cin >> n >> x >> m;
    vector<pair<long long, long long>> ops(m);
    for (auto &[l, r] : ops) { cin >> l >> r; }
    int ans = (x > 1 && x <= n && (n - x <= m)) + (x == 1 && (m % 2 == 0));
    for (auto &[l, r] : ops) {
      if (l == x) ans++;
      if (r == x) ans++;
      if (l < x && x < r) ans += 2;
    }
    cout << ans << endl;
  }
  return 0;
}
