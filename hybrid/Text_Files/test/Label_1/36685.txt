from collections import defaultdict
import sys

a = defaultdict(int)
n = int(input())
t = list(map(int, input().split()))
ans = {}
for i in range(n):
    for j in range(i, n):
        b = t[i:j+1]
        g = 0
        for x in b:
            g |= (1<<(x%3))
        a[g]+=1
for i in range(2):
    for j in range(2):
        for k in range(2):
            ans[i*4+j*2+k] = a[i*4+j*2+k]
for key in ans:
    print(ans[key], end=' ')
