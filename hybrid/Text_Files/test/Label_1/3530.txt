#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> v(n);
    for (int i = 0; i < n; i++) { cin >> v[i]; }
    sort(v.begin(), v.end());
    int ans = 0, j = 0;
    for (int i = n - 1; i >= 0; i--) {
      if (v[i] * (i - j + 1) >= x) {
        ans++;
        j = i;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
