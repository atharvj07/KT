import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int[] r = new int[h];
        int[] c = new int[w];
        for (int i = 0; i < h; i++) {
            r[i] = sc.nextInt();
        }
        for (int j = 0; j < w; j++) {
            c[j] = sc.nextInt();
        }
        long ans = 1;
        for (int i = 0; i < h; i++) {
            ans = (ans * pow(2, r[i] - c[0])) % 1000000007;
        }
        for (int j = 1; j < w; j++) {
            ans = (ans * pow(2, c[j] - c[0])) % 1000000007;
        }
        System.out.println(ans);
    }

    public static long pow(long a, long b) {
        long ans = 1;
        while (b > 0) {
            if ((b & 1) == 1) {
                ans = (ans * a) % 1000000007;
            }
            a = (a * a) % 1000000007;
            b >>= 1;
        }
        return ans;
    }
}
