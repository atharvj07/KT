import java.util.*;

class Main {
    int mod = 1000000007;
    int n;
    int[] p, r;
    long res = Long.MAX_VALUE;
    int[][] a;
    int ans;

    int find(int u) {
        if (p[u] == u)
            return u;
        return p[u] = find(p[u]);
    }

    void unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b)
            return;
        if (r[a] < r[b]) {
            p[a] = b;
            r[b] += r[a];
        } else {
            p[b] = a;
            r[a] += r[b];
        }
    }

    void solve() {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n][2];
        p = new int[n];
        r = new int[n];
        for (int i = 0; i < n; i++) {
            a[i][0] = sc.nextInt();
            a[i][1] = sc.nextInt();
            p[i] = i;
            r[i] = 1;
        }
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                long dist = Math.abs(a[i][0] - a[j][0]) + Math.abs(a[i][1] - a[j][1]);
                if (res > dist) {
                    res = dist;
                    ans = 0;
                }
                if (res == dist) {
                    ans++;
                }
            }
        }
        ans %= mod;
        System.out.println(res);
        System.out.println(ans);
    }

    public static void main(String[] args) {
        new Main().solve();
    }
}
