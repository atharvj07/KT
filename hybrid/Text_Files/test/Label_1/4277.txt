#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long x,y,z=0,a[3005];
    cin>>x>>y;
    for(int i=0; i<y; i++)
    {
        cin>>a[i];
    }
    for(int i=0; i<y; i++)
    {
        if(a[i]==1 || a[i]==x)
        z=1;
    }
    if(z==0)
    cout<<"YES"<<endl;
    else
    cout<<"NO"<<endl;
    return 0;
}
