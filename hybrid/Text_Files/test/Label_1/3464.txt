#include <iostream>
using namespace std;

long long solve(int N)
{
    long long p, mx = 0, ans = 0;
    cin >> p;
    while (--N)
    {
        cin >> mx;
        ans += mx - p;
        p = mx;
    }
    return ans;
}

int main()
{
    int N;
    cin >> N;
    cout << solve(N) << endl;
    return 0;
}
