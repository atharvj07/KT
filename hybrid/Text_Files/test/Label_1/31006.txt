import java.util.HashMap;
import java.util.Scanner;

class Main {
   public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);
      int n = Integer.parseInt(sc.nextLine());
      String s1 = sc.nextLine();
      String s2 = sc.nextLine();

      HashMap<String, Integer> map = new HashMap<>();
      map.put("", 0);
      int cur1 = 0;
      int cur2 = 0;
      boolean ans = true;
      for (int i = 0; i < n - 1; i++) {
         // update current positions
         if (s1.charAt(i) == 'N') cur1++;
         else if (s1.charAt(i) == 'S') cur1--;
         else if (s1.charAt(i) == 'E') cur1++;
         else cur1--;
         if (s2.charAt(i) == 'N') cur2++;
         else if (s2.charAt(i) == 'S') cur2--;
         else if (s2.charAt(i) == 'E') cur2++;
         else cur2--;

         // check if possible
         if (cur1 != cur2 && !map.containsKey(cur1 + "," + cur2)) {
            ans = false;
            break;
         }
         map.put(cur1 + "," + cur2, map.getOrDefault(cur1 + "," + cur2, 0) + 1);
      }
      System.out.println(ans ? "YES" : "NO");
   }
}
