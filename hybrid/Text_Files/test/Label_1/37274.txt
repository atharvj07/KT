def clr(a, b, c):
    if sum('W' in x for x in (a, b, c)) > 1:
        return 'W'
    return 'B'

n, k = map(int, input().split())
colors = list(input())
for _ in range(k):
    next_colors = [clr(colors[i - 1], colors[i], colors[i + 1]) for i in range(n)]
    colors = next_colors
print(''.join(colors))
