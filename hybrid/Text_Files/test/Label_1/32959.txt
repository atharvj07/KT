import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[n + 2];
        for (int i = 0; i < n; i++) {
            a[i + 1] = s.charAt(i) - '0';
        }
        for (int i = 1; i < n; i++) {
            for (int j = 1; j <= n - i; j++) {
                a[j] = Math.abs(a[j] - a[j + 1]);
            }
        }
        sc.close();
        System.out.println(a[1]);
    }
}
