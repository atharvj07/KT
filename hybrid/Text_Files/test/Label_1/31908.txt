import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        long A = sc.nextLong();
        long B = sc.nextLong();
        long C = sc.nextLong();
        long D = sc.nextLong();
        boolean ok = false;
        if (Math.abs(B - A) <= D) ok = true;
        else if (D < C) ok = false;
        else if (C <= D && Math.abs(B - A) % 2 == D % 2) ok = true;
        System.out.println(ok ? "YES" : "NO");
    }
}
