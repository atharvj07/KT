import java.util.Scanner;

public class Main {
    static int n, k;
    static char[] s;
    static int[][] dp;
    static final int mod = (int) 1e9 + 7;

    static int solve(int pos, int cnk) {
        if (pos == n)
            return cnk == k ? 1 : 0;
        if (dp[pos][cnk] != -1)
            return dp[pos][cnk];
        long ans = 0;
        for (char ch = 'a'; ch <= 'z'; ch++) {
            if (ch > s[pos])
                ans += solve(pos + 1, cnk + 1);
            else
                ans += solve(pos + 1, cnk);
        }
        ans %= mod;
        return dp[pos][cnk] = (int) ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        s = sc.next().toCharArray();
        dp = new int[n + 5][k + 5];
        for (int i = 0; i <= n; i++)
            for (int j = 0; j <= k; j++)
                dp[i][j] = -1;
        System.out.println(solve(0, 0));
    }
}
