import numpy as np

n = int(input())
a = list(map(int, input().split()))
a.sort()

if n % 2 == 0:
    m = n // 2
else:
    m = n // 2 + 1

b = np.array([a[i:i + m] for i in range(0, len(a), m)])
b[:, -1:] = np.flip(b[:, -1:], axis=0)

if np.array_equal(b, np.flip(b, axis=0)) and np.array_equal(b, np.flip(b, axis=1)):
    print("YES")
    for row in b:
        print(*row)
else:
    print("NO")
