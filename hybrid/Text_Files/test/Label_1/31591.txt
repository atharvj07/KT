import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        long a = input.nextLong(), b = input.nextLong();
        long gcd = b;
        while (a % b != 0) {
            gcd = a % b;
            a = b;
            b = gcd;
        }
        System.out.println(gcd);
    }
}
