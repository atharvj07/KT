import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        if (k >= n) {
            System.out.println(0);
            return;
        }
        int[] f = new int[n + 1];
        f[1] = 1;
        for (int i = 2; i <= n; i++) {
            if (i <= k) {
                f[i] = (int) ((f[i - 1] * (n - i + 1)) % 1000000007);
            } else {
                f[i] = (int) ((f[i - 1] * 2) % 1000000007);
            }
        }
        System.out.println(f[n]);
    }
}
