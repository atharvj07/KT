import heapq

n,q=map(int,input().split())
a=list(map(int,input().split()))
hq=[]
for i in range(n):
    if a[i]==0: heapq.heappush(hq,-i)

for i in range(q):
    t,x=map(int,input().split())
    if t==1:
        if a[x-1]==0: heapq.heappush(hq,-x)
        else: a[x-1]=0; heapq.heappush(hq,-x)
    else:
        for j in range(x-1):
            y=heapq.heappop(hq)
            if y>0: break
            a[-y]=1
        print(-heapq.heappop(hq))
