MOD = 10**9 + 7

def count_sequences(s):
    count_a = s.count('a')
    if count_a == 0: return 0
    sequences = 1
    count_b = 0
    for char in s:
        if char == 'a':
            sequences = sequences * (count_b + 1) % MOD
            count_b = 0
        elif char == 'b':
            count_b += 1
    return sequences - 1

s = input().strip()
print(count_sequences(s))
