import sys
for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    last = 0
    ans = []
    for i in reversed(range(n)):
        if a[i] >= last:
            ans.append(a[i])
            last = a[i]
        else:
            ans = a[:i + 1] + ans
            last = a[i + 1]
    print(*ans)
