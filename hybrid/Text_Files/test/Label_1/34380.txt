import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            long n = scanner.nextLong();
            System.out.println(n % 2 == 0 ? 0 : 1);
        }
    }
}
