#include <iostream>
#define ll long long

using namespace std;

int main(){
	ll n,p;
	cin>>n>>p;
	ll a[n+2],b[n+2];
	ll pwsum=0,mnpw=1e10;
	for(ll i=1;i<=n;i++){
		cin>>a[i]>>b[i];
		pwsum+=b[i];
		mnpw=min(mnpw,b[i]);
	}
	if(pwsum*1.0/p>=mnpw){
		pwsum+=p;
		ll ans=pwsum*1.0/mnpw;
		cout<<ans;
	} else cout<<-1;
	return 0;
}
