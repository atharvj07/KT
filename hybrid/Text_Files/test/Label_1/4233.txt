#include<bits/stdc++.h>
using namespace std;
#define N 123456
typedef pair<int,int>ii;
vector<int>e[N];
int vis[N]={0};
int lv[N]={0};
int par[N]={0};
int ans=0;
int mn=1e9;
void dfs(int u,int p,int l)
{
    vis[u]=1;
    par[u]=p;
    lv[u]=l;
    for(int i=0;i<e[u].size();i++)
    {
        int v=e[u][i];
        if(!vis[v]) dfs(v,u,l+1);
    }
}
int fnd(int u)
{
    vector<int>ans_;
    while(u!=0)
    {
        ans_.push_back(u);
        u=par[u];
    }
    reverse(ans_.begin(),ans_.end());
    ans=ans_[0];
    mn=ans_.size();
    for(int i=1;i<ans_.size();i++)
    {
        if(ans_.size()-i<mn)
        {
            ans=ans_[i];
            mn=ans_.size()-i;
        }
    }
    return ans;
}
int main()
{
    int n,m;
    cin>>n>>m;
    int u,v;
    for(int i=1;i<n;i++)
    {
        cin>>u>>v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    dfs(1,0,1);
    int x;
    for(int i=0;i<m;i++)
    {
        cin>>x;
        fnd(x);
    }
    cout<<ans<<endl;
    cout<<mn-1<<endl;
    return 0;
}
