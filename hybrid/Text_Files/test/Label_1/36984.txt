def min_rotations_to_print_name(name):
    rotations = 0
    position = 'a'
    
    for char in name:
        distance = abs(ord(char) - ord(position))
        rotations += min(distance, 26 - distance)
        position = char
    
    return rotations

# Read the exhibit name from input
exhibit_name = input().strip()

# Calculate and print the minimum rotations
print(min_rotations_to_print_name(exhibit_name))
