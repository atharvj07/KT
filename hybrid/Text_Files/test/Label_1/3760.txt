#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<vector<int>> a(n, vector<int>(m, 0));
  vector<vector<int>> b(n, vector<int>(m, 0));
  vector<int> v(n, 0);
  vector<int> t(n, -1);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < m; j++) {
      cin >> a[i][j];
      if (a[i][j] > 0) b[i][j] = 1;
    }
  }
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
      if (b[j][i] == 1) {
        if (v[j] == 0) {
          v[j] = 1;
          t[j] = i + 1;
        }
      }
    }
  }
  for (int i = 0; i < n; i++) {
    cout << t[i] << "\n";
  }
  return 0;
}
