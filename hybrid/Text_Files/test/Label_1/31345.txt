import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] t = new long[n];
        long[] h = new long[n];
        for (int i = 0; i < n; i++) t[i] = scanner.nextInt();
        for (int i = 0; i < n; i++) h[i] = scanner.nextInt();

        List<Integer>[] adj = new List[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            adj[u].add(v);
            adj[v].add(u);
        }
        long[][] dp = new long[n][2];
        for (int i = 0; i < n; i++) Arrays.fill(dp[i], -1);

        long answer = dfs(0, -1, t, h, adj, dp);
        System.out.println(answer);
    }

    static long dfs(int u, int p, long[] t, long[] h, List<Integer>[] adj, long[][] dp) {
        if (dp[u][0] != -1) return dp[u][0];

        long min = Long.MAX_VALUE;
        long maxh = h[u];
        long sum = t[u];
        for (int v : adj[u]) {
            if (v == p) continue;
            long tmp = dfs(v, u, t, h, adj, dp);
            if (h[v] >= maxh) {
                sum += tmp;
            } else {
                min = Math.min(min, sum + tmp);
                sum = tmp;
            }
            maxh = Math.max(maxh, h[v]);
        }
        min = Math.min(min, sum);
        dp[u][0] = min;
        dp[u][1] = sum;

        return dp[u][0];
    }
}
