n, k = map(int, input().split());
a = list(map(int, input().split()));
b = list(map(int, input().split()));
ma = 0;
for i in range(n):
    for j in range(n):
        if i != j:
            ta = a[i] + a[j];
            tb = b[i] + b[j];
            if ta / tb == k: ma = max(ma, ta);
print(-1 if ma == 0 else ma)
