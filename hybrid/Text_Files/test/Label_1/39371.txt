n=int(input())
a=[list(map(int,input().split())) for i in range(n)]
a.sort(key=lambda x:x[1])
m,i,j=-1,0,0
for k in range(1,n):
    if i==j:
        if a[k][0]>=a[i][0] and a[k][1]<=a[i][1]:
            j=k
            m=i
    elif i!=j:
        if a[k][0]>=a[j][0] and a[k][1]<=a[j][1]:
            m=j
            i=k
            j=k
print(*(m+1,i+1)) if m>-1 else print(-1,-1)
