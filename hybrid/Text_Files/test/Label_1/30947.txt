import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), ans = 0;
        String a = in.next();
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a.charAt(i) - 48;
        }
        for (int i = 1; i < n; i++) {
            int sum1 = 0, sum2 = 0;
            for (int j = 0; j < i; j++) {
                sum1 += a.charAt(j) - 48;
            }
            for (int j = i; j < n; j++) {
                sum2 += a.charAt(j) - 48;
            }
            if (sum1 == sum2) {
                ans = 1;
                break;
            }
        }
        System.out.println(ans == 1 ? "YES" : "NO");
        in.close();
    }
}
