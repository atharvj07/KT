from bisect import bisect_right
n, k = [int(x) for x in input().split()]
a = sorted([int(x) for x in input().split()])
ans1 = n
if 2*a[0] > k:
    ans1 = -1
else:
    b = a[::-1]
    for i in range(1, n):
        j = bisect_right(b, k-a[i-1])-1
        if j<0:
            ans1 = -1
            break
        ans1+=1
        del b[j]
ans2 = 1
if ans1!=-1:
    ans2 = pow(2, ans1-1, 1000000007)
print(ans1)
print(ans2)
