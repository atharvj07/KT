#include <bits/stdc++.h>

using namespace std;

int n, m, cnt[1505];
vector<int> a;
bool vis[1505];

long long inversions() {
    long long inv = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j < i; j++) {
            if (a[i] < a[j]) {
                inv += (vis[i] == vis[j]);
            }
        }
    }
    return inv;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    cin >> n;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    cin >> m;
    while (m--) {
        int l, r;
        cin >> l >> r;
        reverse(a.begin() + l, a.begin() + r + 1);
        for (int i = l; i <= r; i++) vis[i] = !vis[i];
        cout << ((inversions() & 1) ? "odd" : "even") << endl;
    }

    return 0;
}
