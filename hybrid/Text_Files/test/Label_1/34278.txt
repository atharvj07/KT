import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int ans = 1;
        while (ans % k != 0) {
            ans *= 10;
            ans += 7;
        }
        System.out.println(ans);
    }
}
