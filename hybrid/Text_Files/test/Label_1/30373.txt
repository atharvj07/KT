import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            int cnt = 0;
            while (n >= 10) {
                int max = 0;
                for (int j = 1; j <= n / 2; j++) {
                    int a = n / j;
                    int b = n % j;
                    if (a >= 10 && b >= 10) {
                        max = Math.max(max, a * b);
                    }
                }
                if (max == 0) {
                    cnt = -1;
                    break;
                }
                n = max;
                cnt++;
            }
            System.out.println(cnt);
        }
    }
}
