def solve(n, a):
    if sum(a) % 3 != 0:
        return "NO"
    ones = [i for i, x in enumerate(a) if x == 1]
    if not ones:
        return "YES\n0\n"
    if len(ones) % 3 != 0:
        return "NO"
    third = len(ones) // 3
    gap1 = ones[third] - ones[0]
    gap2 = ones[2*third] - ones[third]
    if gap1 != gap2:
        return "NO"
    return "YES\n2\n{} {}\n".format(ones[0]+1, ones[third]+1)

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(solve(n, a))
