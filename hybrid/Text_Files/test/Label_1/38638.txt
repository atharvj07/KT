import sys

n, d, k = map(int, input().split())
if n<=1 or d>n-1 or (n-d+1)%2 and d%2==0 or (n-1)%2==0 and d%2: sys.stdout.write('NO')
else:
    sys.stdout.write('YES\n')
    if n-d+1>0:
        for i in range(2,d+1): sys.stdout.write(str(i)+' '+str(i-1)+'\n')
    for i in range(d+2,n): sys.stdout.write(str(i)+' '+str(1)+'\n')
