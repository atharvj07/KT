#include <iostream>
#include <vector>
using namespace std;

int gcd(int a, int b) {
  while (b) {
    a %= b;
    swap(a, b);
  }
  return a;
}

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<pair<int, int>> pairs(m);
  for (int i = 0; i < m; i++) {
    cin >> pairs[i].first >> pairs[i].second;
    pairs[i].first--;
    pairs[i].second--;
  }
  int ans = 0;
  for (auto &p : pairs) {
    ans += gcd(a[p.first], a[p.second]) > 1;
  }
  cout << ans << '\n';
  return 0;
}
