#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int sum = 0;
  for (int i = 0; i < n; i++) {
    sum += a[i];
    if (sum % m == 0) {
      cout << "YES" << endl;
      return 0;
    }
    for (int j = i - 1; j >= 0 && sum > m; j--) {
      sum -= a[j];
      if (sum % m == 0) {
        cout << "YES" << endl;
        return 0;
      }
    }
  }
  cout << "NO" << endl;
  return 0;
}
