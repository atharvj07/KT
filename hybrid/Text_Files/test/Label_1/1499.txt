#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    sort(a.begin(), a.end());
    if (x >= n - k + 1) {
        int sum = 0;
        for (int i = n - k + 1; i < n; i++) { sum += a[i]; }
        cout << sum << endl;
    } else {
        cout << -1 << endl;
    }
    return 0;
}
