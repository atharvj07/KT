#include <iostream>
using namespace std ;
int main() {
    int a, b, c, d ;
    cin >> a >> b >> c >> d ;
    if ( (b-d)%c==0 && (b-d)%c>=0 ) {
        cout << b + (b-d)/c * a ;
        return 0 ;
    }
    if ( (b-d)%a==0 && (b-d)%a>=0 ) {
        cout << b + (b-d)/a * c ;
        return 0 ;
    }
    cout << -1 ;
    return 0 ;
}
