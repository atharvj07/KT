import java.util.*;

public class Main {
    public static long MAX_SIZE = 100000;
    public static int[] suffixMax = new int[(int)MAX_SIZE];
    public static HashSet<Integer> ans = new HashSet<>();
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int mx = 0;
        for (int i = 0; i < q; i++) {
            int start = sc.nextInt();
            int end = sc.nextInt();
            int add = sc.nextInt();
            mx = Math.max(mx, add);
            suffixMax[start - 1] = Math.max(suffixMax[start - 1], add);
            suffixMax[end] = Math.max(suffixMax[end], add);
        }
        for (int i = n - 1; i >= 0; i--) {
            suffixMax[i] = Math.max(suffixMax[i], suffixMax[i + 1]);
            ans.add(suffixMax[i]);
        }
        System.out.println(ans.size());
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
