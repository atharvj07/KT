#include <bits/stdc++.h>

using namespace std;

#define endl '\n'

#define ll long long
#define IOS ios_base::sync_with_stdio(false), cin.tie(nullptr), cout.tie(nullptr);

int main() {
  IOS;
  int n, a, b;
  cin >> n >> a >> b;
  string s;
  cin >> s;
  vector<int> v;
  for (int i = 0; i < n; i++) {
    if (s[i] - '0')
      v.push_back(i + 1);
  }
  int ans;
  if (find(v.begin(), v.end(), a) != v.end() &&
      find(v.begin(), v.end(), b) != v.end())
    ans = abs(a - b);
  else if (find(v.begin(), v.end(), a) != v.end()) {
    int pos = lower_bound(v.begin(), v.end(), a) - v.begin();
    if (abs(v[pos] - a) <= abs(v[pos] - b))
      ans = abs(v[pos] - a);
    else
      ans = abs(v[pos - 1] - a);
  } else {
    int pos = upper_bound(v.begin(), v.end(), b) - v.begin();
    if (abs(v[pos - 1] - b) <= abs(v[pos - 1] - a))
      ans = abs(v[pos - 1] - b);
    else
      ans = abs(v[pos - 2] - b);
  }
  cout << ans << endl;
  return 0;
}
