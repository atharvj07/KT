import sys

input = sys.stdin.readline

n = int(input())
a00 = []
a10 = []
a01 = []
a11 = []
for _ in range(n):
    s, a = input().split()
    if s == "00":
        a00.append(int(a))
    elif s == "10":
        a10.append(int(a))
    elif s == "01":
        a01.append(int(a))
    elif s == "11":
        a11.append(int(a))

a00.sort()
a10.sort()
a01.sort()
a11.sort()

ans = 0
cnt = 0

for a in a01 + a00:
    cnt += 1
    ans += a
    if cnt >= (n + 1) // 2:
        break

for a in a11 + a10:
    if cnt >= (n + 1) // 2:
        break
    cnt += 1
    ans += a

if cnt < (n + 1) // 2:
    ans = 0

print(ans)
