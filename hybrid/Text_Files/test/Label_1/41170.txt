n, k = map(int, input().split())
a = list(map(int, input().split()))
ans = []
for i in range(n - 1):
    if a[i] + 1 - a[i + 1] != k:
        ans.append(f'+ {i + 2} {k - (a[i] + 1 - a[i + 1])}')
    if a[i] + 1 - a[i + 1] != k:
        ans.append(f'- {i + 2} {k - (a[i] + 1 - a[i + 1])}')
print(len(ans))
for ans in ans:
    print(ans)
