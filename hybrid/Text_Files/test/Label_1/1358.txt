#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);

        for (int i = 0; i < n; i++) cin >> a[i];

        vector<int> dp(n, 0);
        dp[0] = a[0];

        for (int i = 1; i < n; i++) {
            dp[i] = max(dp[i - 1] + a[i], a[i]);
        }

        int ans = 0;
        for (int i = 0; i < n; i++) ans = max(ans, dp[i]);

        cout << ans << endl;
    }

    return 0;
}
