import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String x = sc.next();
        int n = x.length();
        int sCnt = 0;
        int tCnt = 0;
        int stCnt = 0;
        for (int i = 0; i < n; i++) {
            if (x.charAt(i) == 'S') {
                sCnt++;
            }
            if (x.charAt(i) == 'T') {
                tCnt++;
            }
            if (i < n - 1 && x.charAt(i) == 'S' && x.charAt(i + 1) == 'T') {
                stCnt++;
            }
        }
        if (sCnt == tCnt) {
            System.out.println(n - 2 * stCnt);
        } else {
            System.out.println(n);
        }
    }
}
