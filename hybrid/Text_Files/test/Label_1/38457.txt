from sys import stdin
input = stdin.readline

n, k = map(int, input().split())
c = sorted(map(int, input().split()))
ans = set()
for i in range(1 << n):
    sum = 0
    for j in range(n):
        if i >> j & 1: sum += c[j]
    if sum == k:
        for j in range(1 << n):
            s = 0
            for l in range(n):
                if j >> l & 1: s += c[l]
            if s <= k: ans.add(s)
ans = list(ans)
ans.sort()
print(len(ans))
print(*ans)
