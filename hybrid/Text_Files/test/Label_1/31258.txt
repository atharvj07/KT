import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n + 1], c = new int[n + 1];
        for (int i = 0; i < n; i++) {
            p[i + 1] = sc.nextInt();
            c[i + 1] = sc.next().charAt(0);
        }
        int k = sc.nextInt();
        int[] a = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
        }
        int[] ans = new int[k];
        for (int i = 0; i < k; i++) {
            int u = a[i], cur = 0;
            while (u != 0) {
                cur++;
                u = p[u];
            }
            ans[i] = cur;
        }
        for (int i : ans) {
            System.out.print(i + " ");
        }
    }
}
