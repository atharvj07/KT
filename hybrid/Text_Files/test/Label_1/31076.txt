import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        Arrays.sort(arr);
        int sum = 0;
        for (int i = 0; i < n - 1; i++) {
            if (arr[i] < 0) {
                sum -= arr[i];
            } else {
                break;
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (arr[i] > 0) {
                sum += arr[i];
            } else {
                break;
            }
        }
        System.out.println(sum);
    }
}
