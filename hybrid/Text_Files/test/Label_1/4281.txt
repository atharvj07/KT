#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, a[200005], res[200005];
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    if (n == 1) {
        cout << "YES" << endl;
        cout << "0" << endl;
        return 0;
    }
    if (a[0] > a[1]) {
        res[0] = 1;
        res[1] = 0;
        for (int i = 2; i < n; i++) {
            if (a[i] < a[i - 1])
                res[i] = 1;
            else
                res[i] = 0;
        }
    } else {
        res[0] = 0;
        res[1] = 1;
        for (int i = 2; i < n; i++) {
            if (a[i] > a[i - 1])
                res[i] = 1;
            else
                res[i] = 0;
        }
    }
    bool ok = true;
    for (int i = 1; i < n; i++) {
        if (res[i] == 0 && res[i - 1] == 0 && a[i] < a[i - 1]) {
            ok = false;
            break;
        }
        if (res[i] == 1 && res[i - 1] == 1 && a[i] > a[i - 1]) {
            ok = false;
            break;
        }
    }
    if (ok) {
        cout << "YES" << endl;
        for (int i = 0; i < n; i++) cout << res[i] << " ";
        cout << endl;
    } else {
        cout << "NO" << endl;
    }
    return 0;
}
