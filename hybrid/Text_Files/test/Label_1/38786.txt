def gcd(x, y):
    while y:
        x, y = y, x % y
    return x

def lcm(x, y):
    return x * y // gcd(x, y)

def solve(a, b):
    if a > b:
        a, b = b, a
    diff = b - a
    if diff == 0:
        return 0
    min_lcm = float('inf')
    best_k = 0
    for i in range(1, int(diff**0.5) + 1):
        if diff % i == 0:
            k1 = i - a % i
            k2 = diff // i - a % (diff // i)
            lcm1 = lcm(a + k1, b + k1)
            lcm2 = lcm(a + k2, b + k2)
            if lcm1 < min_lcm:
                min_lcm = lcm1
                best_k = k1
            if lcm2 < min_lcm:
                min_lcm = lcm2
                best_k = k2
    return best_k

a, b = map(int, input().split())
print(solve(a, b))
