import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
c = [[j for j in range(60) if 1<<(j) & i] for i in a]

msk = [0 for i in range(n)]
for i in range(n):
    flag = True
    for j in c[i]:
        if len([k for k in range(n) if j in c[k] and b[k] > b[i]]):
            flag = False
    if flag:
        msk[i] = 1

ans = 0
for i in range(n):
    if msk[i]:
        ans = max(ans, b[i])
        for j in range(i+1, n):
            if msk[j] and not (msk[j] & msk[i]):
                ans = max(ans, b[i]+b[j])

print(ans)
