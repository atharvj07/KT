#include <iostream>
#include <queue>
using namespace std;
const int mx=20;
int dx[]={-1,0,1},dy[]={0,1,0};
int main()
{
    int a[mx][mx], vis[mx][mx], dist[mx][mx];
    for(int i=1; i<=8; i++)
    {
        for(int j=1; j<=8; j++)
        {
            char c;
            cin>>c;
            if(c=='.') a[i][j]=0;
            else if(c=='W') a[i][j]=2;
            else if(c=='B') a[i][j]=1;
        }
    }
    queue<pair<int, int> > q;
    for(int i=1; i<=8; i++)
    {
        q.push({8,i});
        dist[8][i]=0;
    }
    while(!q.empty())
    {
        int x=q.front().first, y=q.front().second;
        q.pop();
        for(int i=0; i<3; i++)
        {
            int p=x+dx[i], r=y+dy[i];
            if(p>=1 && p<=8 && r>=1 && r<=8 && !a[p][r] && (vis[p][r]==0 || vis[p][r]>vis[x][y]))
            {
                vis[p][r]=1+vis[x][y];
                dist[p][r]=dist[x][y]+1;
                q.push({p,r});
            }
        }
    }
    int mn=1e9;
    for(int i=1; i<=8; i++)
    {
        if(vis[1][i]) mn=min(mn,dist[1][i]);
    }
    for(int i=1; i<=8; i++)
    {
        q.push({1,i});
        dist[1][i]=0;
    }
    while(!q.empty())
    {
        int x=q.front().first, y=q.front().second;
        q.pop();
        for(int i=0; i<3; i++)
        {
            int p=x+dx[i], r=y+dy[i];
            if(p>=1 && p<=8 && r>=1 && r<=8 && !a[p][r] && (vis[p][r]==0 || vis[p][r]>vis[x][y]))
            {
                vis[p][r]=1+vis[x][y];
                dist[p][r]=dist[x][y]+1;
                q.push({p,r});
            }
        }
    }
    int mx=0;
    for(int i=1; i<=8; i++)
    {
        if(vis[8][i]) mx=max(mx,dist[8][i]);
    }
    if(mn==1e9) mn=0;
    if(mx==0) mx=0;
    if(mn<=mx) cout<<"A\n";
    else cout<<"B\n";
}
