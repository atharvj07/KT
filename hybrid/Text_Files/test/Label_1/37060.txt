import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
d = {}
for x in a:
    if x not in d:
        d[x] = []
    d[x].append(a.index(x))
mx = 0
ans = []
for x in d:
    t = d[x]
    for i in range(len(t)):
        for j in range(i+1, len(t)):
            if t[j] - t[i] + 1 > mx:
                mx = t[j] - t[i] + 1
                ans = [t[i]+1]
            elif t[j] - t[i] + 1 == mx:
                ans.append(t[i]+1)
print(len(ans), mx)
print(*ans)
