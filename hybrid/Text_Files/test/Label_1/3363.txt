#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int l = 0, r = n - 1, ans = 0;
    while (l <= r) {
        if (a[l] <= k) {
            l++, ans++;
        } else if (a[r] <= k)
            r--, ans++;
        else
            break;
    }
    cout << ans << endl;
}
