import java.util.Scanner;

public class StoneGame {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] piles = new int[n];

            for (int j = 0; j < n; j++) {
                piles[j] = scanner.nextInt();
            }

            boolean tWins = canWin(piles);
            System.out.println(tWins ? "T" : "HL");
        }
    }

    private static boolean canWin(int[] piles) {
        int sum = 0;
        for (int pile : piles) {
            sum += pile;
        }
        return sum % 2 == 1;
    }
}
