import java.util.Arrays;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    long n = scanner.nextLong();
    long k = scanner.nextLong();
    if (k > permCount(n)) {
      System.out.println(-1);
      return;
    }
    int[] perm = perm(n, k);
    int sum = 0;
    for (int i = 0; i < n; i++) {
      if (i == perm[i] - 1 && hasLuckyDigits(i + 1) && hasLuckyDigits(perm[i])) {
        sum++;
      }
    }
    System.out.println(sum);
  }

  static boolean hasLuckyDigits(long n) {
    return Arrays.stream(String.valueOf(n).split("")).allMatch(s -> s.equals("4") || s.equals("7"));
  }

  static int[] perm(long n, long k) {
    int[] p = new int[(int) n];
    for (int i = 0; i < n; i++) {
      p[i] = i + 1;
    }
    for (long i = 0; i < k - 1; ++i) {
      nextPermutation(p);
    }
    return p;
  }

  static long permCount(long n) {
    long perm = 1;
    for (int i = 2; i <= n; ++i) {
      perm *= i;
    }
    return perm;
  }

  static void nextPermutation(int[] p) {
    for (int i = p.length - 1; i > 0; --i) {
      if (p[i] > p[i - 1]) {
        int min = i;
        for (int j = i + 1; j < p.length; ++j) {
          if (p[j] > p[i - 1] && p[j] < p[min]) {
            min = j;
          }
        }
        int temp = p[i - 1];
        p[i - 1] = p[min];
        p[min] = temp;
        Arrays.sort(p, i, p.length);
        return;
      }
    }
    Arrays.sort(p);
  }

  static Scanner scanner = new Scanner(System.in);
}
