import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int v = sc.nextInt();
        int e = sc.nextInt();
        Edge[] edges = new Edge[e];
        for (int i = 0; i < e; i++) {
            int s = sc.nextInt();
            int t = sc.nextInt();
            int w = sc.nextInt();
            edges[i] = new Edge(s, t, w);
        }
        int ans = ShortestHamiltonianCycle.find(v, e, edges);
        System.out.println(ans);
    }
}

class Edge {
    int s;
    int t;
    int w;

    public Edge(int s, int t, int w) {
        this.s = s;
        this.t = t;
        this.w = w;
    }
}

class ShortestHamiltonianCycle {
    static final long INF = (long) 1e18;

    public static int find(int v, int e, Edge[] edges) {
        // dp[i][(1<<v)-1] = the shortest distance to reach vertex i and visit all vertices exactly once
        long[][] dp = new long[v][1 << v];
        for (int i = 0; i < v; i++) {
            for (int j = 0; j < 1 << v; j++) {
                dp[i][j] = INF;
            }
        }
        dp[0][1] = 0;
        for (int i = 0; i < (1 << v); i++) {
            for (int j = 0; j < v; j++) {
                if (dp[j][i] == INF) {
                    continue;
                }
                for (Edge edge : edges) {
                    if ((i & (1 << edge.s)) != 0 && (i & (1 << edge.t)) == 0) {
                        dp[edge.t][i | (1 << edge.t)] = Math.min(dp[edge.t][i | (1 << edge.t)], dp[edge.s][i] + edge.w);
                    }
                }
            }
        }
        int ans = (int) (dp[0][(1 << v) - 1] == INF ? -1 : dp[0][(1 << v) - 1]);
        return ans;
    }
}
