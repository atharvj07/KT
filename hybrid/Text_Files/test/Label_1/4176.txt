#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long> v(n);
    for (int i = 0; i < n; ++i) cin >> v[i];
    long long ans = 0, bal = 0;
    for (int i = 0; i < n; ++i) {
        long long buy = max(bal, -v[i]);
        long long sell = min(bal, v[i]);
        bal = buy + sell;
        ans += bal;
    }
    cout << ans;
}
