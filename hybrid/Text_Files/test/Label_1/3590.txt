#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  vector<string> suffixes = {"po", "desu", "masu", "mnida"};
  vector<string> languages = {"FILIPINO", "JAPANESE", "JAPANESE", "KOREAN"};
  while (t--) {
    string s;
    cin >> s;
    for (int i = 0; i < suffixes.size(); i++) {
      if (s.substr(s.size() - suffixes[i].size()) == suffixes[i]) {
        cout << languages[i] << endl;
        break;
      }
    }
  }
  return 0;
}
