#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> pii;

int main() {
    long long a,b; cin>>a>>b;
    string s; cin>>s;
    int dx=0,dy=0;
    pii p;
    for(char c : s){
        if(c=='U') dy++;
        if(c=='D') dy--;
        if(c=='L') dx--;
        if(c=='R') dx++;
        if(dx==a&&dy==b) p={dx,dy};
    }
    if(p==pii(a,b)) cout<<"Yes\n";
    else cout<<"No\n";
}
