import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n;
        while ((n = sc.nextInt()) != 0) {
            int[] c = new int[7];
            for (int i = 0; i < n; i++) {
                int a = sc.nextInt();
                if (a < 10) c[0]++;
                else if (a < 20) c[1]++;
                else if (a < 30) c[2]++;
                else if (a < 40) c[3]++;
                else if (a < 50) c[4]++;
                else if (a < 60) c[5]++;
                else c[6]++;
            }
            for (int i = 0; i < 7; i++) {
                System.out.println(c[i]);
            }
        }
    }
}
