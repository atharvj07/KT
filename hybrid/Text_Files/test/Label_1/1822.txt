#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m, q;
  cin >> n >> m >> q;
  vector<vector<int>> mat(n, vector<int>(m));
  for (int i = 0; i < q; i++) {
    int t;
    cin >> t;
    if (t == 3) {
      int r, c, x;
      cin >> r >> c >> x;
      mat[r - 1][c - 1] = x;
    } else {
      int p;
      cin >> p;
      --p;
      if (t == 1) {
        for (auto &row : mat) rotate(row.rbegin(), row.rbegin() + 1, row.rend());
      } else {
        for (int i = 0; i < n; i++) rotate(mat[i].begin() + p, mat[i].begin() + p + 1, mat[i].end());
      }
    }
  }
  for (auto &row : mat) {
    for (auto el : row) cout << el << " ";
    cout << "\n";
  }
}
