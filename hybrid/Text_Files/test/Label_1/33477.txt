import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] edges = new int[m][4];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 4; j++) {
                edges[i][j] = sc.nextInt();
            }
        }
        int[][] limits = new int[n + 1][2];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(limits[i], -1);
        }
        boolean[] visited = new boolean[n + 1];
        int[] path = new int[n + 1];
        long loyalty = 0;
        int[] ans = dfs(1, n, 0, edges, limits, visited, path);
        if (ans[0] == -1) {
            System.out.println("Nice work, Dima!");
        } else {
            System.out.println(ans[0]);
        }
    }

    public static int[] dfs(int u, int n, int loyalty, int[][] edges, int[][] limits, boolean[] visited, int[] path) {
        if (u == n) {
            return new int[]{loyalty, u};
        }
        visited[u] = true;
        int[] ans = new int[]{-1, -1};
        for (int[] edge : edges) {
            int v = edge[1], l = edge[2], r = edge[3];
            if (v != u || !visited[v]) {
                limits[v][0] = Math.max(limits[v][0], l);
                limits[v][1] = Math.min(limits[v][1], r);
            }
        }
        for (int i = 1; i <= n; i++) {
            if (visited[i]) continue;
            for (int[] edge : edges) {
                int v = edge[1], l = edge[2], r = edge[3];
                if (v == i && limits[i][0] <= l && limits[i][1] >= l) {
                    path[u] = l;
                    visited[i] = true;
                    ans = dfs(i, n, loyalty + 1, edges, limits, visited, path);
                    visited[i] = false;
                    if (ans[0] != -1) break;
                }
            }
            if (ans[0] != -1) break;
        }
        return ans;
    }
}
