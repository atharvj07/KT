import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {

  static boolean[] notPrime;

  public static void main(String[] args) throws IOException {

    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    StringTokenizer st = new StringTokenizer(br.readLine());
    int[] arr = new int[n];
    notPrime = new boolean[n * 2 + 2];
    List<Integer> primes = new ArrayList<>();
    boolean[] arrCnt = new boolean[n * 2 + 2];
    for (int i = 0; i < n; i++) {
      arr[i] = Integer.parseInt(st.nextToken());
      arrCnt[arr[i]] = true;
    }
    for (int i = 2; i < arrCnt.length; i++) {
      if (!arrCnt[i]) notPrime[i] = true;
      else {
        for (int j = i * 2; j < arrCnt.length; j += i) {
          notPrime[j] = true;
        }
      }
    }
    for (int i = 2; i < arrCnt.length; i++) {
      if (!notPrime[i]) primes.add(i);
    }
    List<Integer> ans = new ArrayList<>();
    for (Integer p : primes) {
      int k = 0;
      while (k < n && arrCnt[p]) {
        if (arr[k] == p) {
          arrCnt[arr[k]] = false;
          ans.add(arr[k]);
          k++;
        } else {
          break;
        }
      }
    }
    for (int i = 0; i < n; i++) {
      if (arrCnt[arr[i]]) {
        ans.add(arr[i]);
      }
    }
    ans.forEach(a -> System.out.print(a + " "));
  }
}
