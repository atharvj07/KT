#include <bits/stdc++.h>
typedef long long ll;
using namespace std;
const int MN = 1e5 + 5, inf = 1e9;
ll n, k, a[MN], b[MN];
int main() {
    cin >> n >> k;
    vector<ll> f(k + 1, inf);
    for (int i = 1; i <= n; ++i) cin >> a[i];
    for (int i = 1; i <= n; ++i) cin >> b[i];
    f[0] = 0;
    for (int i = 1; i <= n; ++i) {
        for (int j = a[i]; j <= k; ++j) f[j] = min(f[j], f[j - a[i]] + b[i]);
    }
    cout << f[k] << endl;
}
