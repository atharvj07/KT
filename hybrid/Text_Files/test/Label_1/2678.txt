#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    ll n, k;
    cin >> n >> k;
    vector<ll> a(n), b(n);
    for (auto &x : a) cin >> x;
    for (auto &x : b) cin >> x;
    vector<ll> dpa(n + 1), dpb(n + 1);
    dpa[n] = dpb[n] = LONG_LONG_MAX;
    for (int i = n - 1; i >= 0; --i) {
        dpa[i] = min(dpa[i + 1], a[i]);
        dpb[i] = min(dpb[i + 1], b[i]);
    }
    ll ans = LONG_LONG_MAX;
    for (int i = 0; i < n; ++i) {
        if (k == 0) break;
        ans = min(ans, dpa[i] + dpb[i + 1] * k);
        --k;
    }
    cout << ans << endl;
    return 0;
}
