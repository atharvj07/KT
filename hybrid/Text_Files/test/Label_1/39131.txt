import sys

n = int(input())
cities = []
for _ in range(n):
    x, c = input().split()
    cities.append((int(x), c))

byte, disputed, berland = [], [], []
for x, c in cities:
    if c == 'B':
        byte.append(x)
    elif c == 'P':
        disputed.append(x)
    else:
        berland.append(x)

ans = 0
if len(byte) > 1:
    for i in range(len(byte) - 1):
        ans += byte[i + 1] - byte[i]
if len(berland) > 1:
    for i in range(len(berland) - 1):
        ans += berland[i + 1] - berland[i]
if len(byte) > 0 and len(disputed) > 0 and disputed[0] > byte[-1]:
    ans += disputed[0] - byte[-1]
if len(berland) > 0 and len(disputed) > 0 and disputed[-1] < berland[0]:
    ans += berland[0] - disputed[-1]

print(ans)
