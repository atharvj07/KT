#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 2205;

ll dp[MAXN][MAXN];

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, k;
  cin >> n >> k;
  vector<ll> a(n), b(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < n; i++) cin >> b[i];
  memset(dp, 0x3f, sizeof(dp));
  dp[n][0] = 0;
  for (int i = n - 1; i >= 0; i--) {
    for (int j = 0; j <= k; j++) {
      dp[i][j] = dp[i + 1][j];
      dp[i][j] = min(dp[i][j], dp[i + 1][j + 1] + a[i]);
      if (j > 0) dp[i][j] = min(dp[i][j], dp[i + 1][j - 1] + b[i]);
    }
  }
  cout << dp[0][k] << '\n';
  return 0;
}
