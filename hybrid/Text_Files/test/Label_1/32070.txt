import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n], b = new int[n], c = Arrays.copyOf(a, n);
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        int steps = 0;
        while (true) {
            b[0] = c[0];
            b[n - 1] = c[n - 1];
            for (int i = 1; i < n - 1; i++) b[i] = c[i] == c[i - 1] ? c[i - 1] : c[i + 1];
            steps++;
            if (Arrays.equals(b, c)) break;
            c = Arrays.copyOf(b, n);
        }
        if (steps == 0) System.out.println(1);
        else {
            System.out.println(steps);
            for (int i = 0; i < n; i++) System.out.print(b[i] + " ");
        }
    }
}
