#define fast ios_base::sync_with_stdio(false);cin.tie(nullptr);cout.tie(nullptr);
vi p;
vi ps(maxn, 0);
vi a;
vi b;
int main() {
    fast;
    int n, l, r;
    cin >> n >> l >> r;
    a.resize(n);
    p.resize(n);
    for (int i = 0;i < n;i++)cin >> a[i];
    for (int i = 0;i < n;i++)cin >> p[i];
    ps[p[0]]++;
    for (int i = 1;i < n;i++) {
        ps[p[i]]++;
        if (ps[p[i]] > 1) {
            cout << "-1" << endl;
            return 0;
        }
    }
    for (int i = n;i >= 0;i--) {
        int x = l;
        int y = r;
        for (int j = 0;j < n;j++) {
            if (ps[j] == 1) {
                if (x + a[j] <= r) {
                    x = x + a[j];
                    b.pb(x);
                    ps[j] = 2;
                    break;
                }
            }
        }
        if (i == 0) {
            cout << "-1" << endl;
            return 0;
        }
        for (int j = 0;j < n;j++) {
            if (ps[j] == 1) {
                if (y + a[j] <= r) {

                    b.pb(y);
                    break;
                }
            }
        }
        if (i == 0) {
            cout << "-1" << endl;
            return 0;
        }
    }
