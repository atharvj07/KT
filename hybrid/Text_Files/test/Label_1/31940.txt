import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        int ans = -1;
        for (int i = 7; i <= 1000000; i += 7) {
            if (i % k == 0) {
                ans = i;
                break;
            }
        }
        System.out.println(ans);
    }
}
