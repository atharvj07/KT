import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            long s = input.nextLong();
            long ans = 0;
            while (s >= 10) {
                long x = s / 10 * 10;
                ans += x;
                s -= x;
                s += x / 10;
            }
            ans += s;
            System.out.println(ans);
        }
    }
}
