#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int t;
    std::cin >> t;

    while (t--) {
        int n;
        std::cin >> n;

        std::vector<int> a(n);
        for (int i = 0; i < n; ++i) {
            std::cin >> a[i];
        }

        std::sort(a.begin(), a.end());
        int idx = 1;
        while (a[idx] == a[0]) {
            ++idx;
        }

        std::cout << idx + 1 << std::endl;
    }

    return 0;
}
