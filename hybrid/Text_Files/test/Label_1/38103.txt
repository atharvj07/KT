n, a, b = map(int, input().split())
t = [int(x) for x in input().split()]
denied = 0
occupied = 0
for x in t:
    if x == 1:
        if a > 0:
            a -= 1
        elif b > 0:
            b -= 1
            occupied += 1
        else:
            if occupied > 0:
                occupied -= 1
            else:
                denied += 1
    else:
        if b > 0:
            b -= 1
        else:
            denied += 2
print(denied)
