#include <bits/stdc++.h>
using namespace std;
int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int r, c, n, k;
    cin >> r >> c >> n >> k;
    int x[n], y[n];
    for (int i = 0; i < n; i++) { cin >> x[i] >> y[i]; }
    sort(x, x + n);
    sort(y, y + n);
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            int xl = x[i], xr = x[j], yl = y[i], yr = y[j];
            if (k <= (xr - xl + 1) * (yr - yl + 1)) ans++;
        }
    }
    cout << ans << '\n';
    return 0;
}
