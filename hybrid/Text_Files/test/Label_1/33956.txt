import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int[] l = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
        }

        Arrays.sort(l);

        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    if (l[i] + l[j] + l[k] >= a + b + c) {
                        int mp = 0;
                        if (l[i] >= c) {
                            mp += (l[i] - c);
                        } else {
                            mp += (c - l[i] + 10);
                            l[i] = c;
                        }
                        if (l[j] >= b) {
                            mp += (l[j] - b);
                        } else {
                            mp += (b - l[j] + 10);
                            l[j] = b;
                        }
                        if (l[k] >= a) {
                            mp += (l[k] - a);
                        } else {
                            mp += (a - l[k] + 10);
                            l[k] = a;
                        }
                        ans = Math.min(ans, mp);
                    }
                }
            }
        }

        System.out.println(ans);
    }
}
