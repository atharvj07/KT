#include <bits/stdc++.h>
using namespace std;

const int mod = 1e9 + 7;

int32_t main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    cnt += (s[i] == '?');
  }
  int a[cnt + 1] = {0};
  int b[cnt + 1] = {0};
  int c[cnt + 1] = {0};
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == '?') {
      a[cnt] = (2 * a[cnt] + b[cnt]) % mod;
      b[cnt] = (2 * b[cnt] + c[cnt]) % mod;
      c[cnt] = (2 * c[cnt] + 1) % mod;
      cnt--;
    } else {
      if (s[i] == 'a') a[cnt]++;
      if (s[i] == 'b') b[cnt]++;
      if (s[i] == 'c') c[cnt]++;
    }
  }
  int ans = 0;
  for (int i = 0; i <= cnt; i++) {
    ans = (ans + 1LL * a[i] * b[i] % mod * c[i]) % mod;
  }
  cout << ans << "\n";
}
