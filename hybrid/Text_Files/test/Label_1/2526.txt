#include <cstdio>
#include <string>
int main() {
  int a, ta, b, tb;
  scanf("%d %d %d %d", &a, &ta, &b, &tb);
  int h, m;
  scanf("%d:%d", &h, &m);
  h = 60 * h + m;
  int t = h % a;
  int z = 1; /* source side */
  t += ta;
  while (t < h) t += a;
  if (t == h) z = 0;
  h += tb;
  while (t < h) { t += b, z++; }
  h -= tb;
  while (h < t) { h += a, z++; }
  printf("%d\n", z);
  return 0;
}
