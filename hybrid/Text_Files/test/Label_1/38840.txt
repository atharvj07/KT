t = int(input())
for _ in range(t):
    a, b, c, m = map(int, input().split())
    letters = sorted([a, b, c])
    
    max_pairs = sum([x - 1 for x in letters])
    min_pairs = letters[2] - 1 - (letters[0] + letters[1])
    
    if min_pairs < 0:
        min_pairs = 0
    
    if min_pairs <= m <= max_pairs:
        print("YES")
    else:
        print("NO")
