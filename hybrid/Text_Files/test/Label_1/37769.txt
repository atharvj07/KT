def checkreturnable(n:int, s:str) -> int:
    """return the number of returnable rooms in the snake exhibition"""
    ret = 0
    turn = [0]*n
    for i in range(n):
        if s[i] == '-': continue
        elif s[i] == '<': turn[i^1] = 1
        else: turn[(i+1)%n] = 1
    for i in range(n):
        if not turn[i]: ret+=1
    return ret

for _ in range(int(input())):
    n = int(input())
    s = input()
    print(checkreturnable(n, s))
