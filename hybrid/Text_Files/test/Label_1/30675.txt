import java.util.Scanner;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Stack<Integer> speedLimits = new Stack<>();
        Stack<Boolean> canOvertake = new Stack<>();
        int speedLimit = 0;
        boolean canOvertakeNow = true;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int type = in.nextInt();
            switch (type) {
                case 1:
                    int newSpeed = in.nextInt();
                    if (canOvertakeNow && newSpeed > speedLimit) {
                        ans++;
                    }
                    break;
                case 2:
                    if (!canOvertakeNow) {
                        ans++;
                    }
                    break;
                case 3:
                    speedLimit = in.nextInt();
                    speedLimits.push(speedLimit);
                    canOvertake.push(canOvertakeNow);
                    canOvertakeNow = false;
                    break;
                case 4:
                    canOvertakeNow = true;
                    while (!canOvertake.empty()) {
                        canOvertakeNow |= canOvertake.pop();
                    }
                    break;
                case 5:
                    speedLimit = 0;
                    while (!speedLimits.empty()) {
                        speedLimit = Math.max(speedLimit, speedLimits.pop());
                    }
                    canOvertakeNow = true;
                    while (!canOvertake.empty()) {
                        canOvertakeNow |= canOvertake.pop();
                    }
                    break;
                case 6:
                    canOvertakeNow = false;
                    canOvertake.push(canOvertakeNow);
                    break;
            }
        }
        System.out.println(ans);
    }
}
