import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        boolean isColored = false;
        for (int i = 0; i < n; i++) {
            String line = input.nextLine();
            for (char c : line.toCharArray()) {
                if (c != 'W' && c != 'B' && c != 'G') {
                    isColored = true;
                    break;
                }
            }
            if (isColored) {
                break;
            }
        }
        System.out.println(isColored ? "#Color" : "#Black&White");
    }
}
