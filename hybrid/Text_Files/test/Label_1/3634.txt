#include <iostream>
#include <cmath>

long long gcd(long long a, long long b) {
    if (a == 0) {
        return b;
    }
    return gcd(b % a, a);
}

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        long long a, b, c;
        std::cin >> a >> b >> c;
        long long g = gcd(a, gcd(b, c));
        long long ans = 0;
        for (long long i = 1; i <= g; i++) {
            if (a % i == 0 && b % i == 0 && c % i == 0) {
                ans++;
            }
        }
        std::cout << ans << std::endl;
    }
    return 0;
}
