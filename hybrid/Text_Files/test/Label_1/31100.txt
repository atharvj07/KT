import java.util.*;

public class Main {
    static class Query {
        int l, r, x;

        Query(int l, int r, int x) {
            this.l = l;
            this.r = r;
            this.x = x;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        List<Query> queries = new ArrayList<>();
        int[] ans = new int[n];
        boolean hasContradiction = false;
        int or = 0;
        for (int i = 0; i < m; i++) {
            int type = sc.nextInt();
            if (type == 1) {
                int l = sc.nextInt() - 1;
                int r = sc.nextInt() - 1;
                int x = sc.nextInt();
                queries.add(new Query(l, r, x));
            } else {
                int k = sc.nextInt() - 1;
                int d = sc.nextInt();
                if (ans[k] != 0 && ans[k] != d) hasContradiction = true;
                ans[k] = d;
                or |= d;
            }
        }
        if (hasContradiction) {
            System.out.println("NO");
            return;
        }
        for (Query q : queries) {
            int mx = 0;
            for (int i = q.l; i <= q.r; i++) {
                mx = Math.max(mx, ans[i]);
            }
            if (mx != q.x) hasContradiction = true;
        }
        if (hasContradiction) {
            System.out.println("NO");
            return;
        }
        for (int i = 0; i < n; i++) {
            if (ans[i] == 0) ans[i] = or;
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
