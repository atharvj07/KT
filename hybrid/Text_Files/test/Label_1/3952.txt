#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> ans;
  while (n > 0) {
    int cur = 0;
    while ((1 << cur) <= n) cur++;
    cur--;
    ans.push_back((1 << cur));
    n -= (1 << cur);
  }
  cout << ans.size() << endl;
  for (auto i : ans) cout << i << " ";
  cout << endl;
  return 0;
}
