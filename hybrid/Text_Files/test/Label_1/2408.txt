#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll a, b, c, d;
int main() {
  cin >> a >> b >> c >> d;
  if (a > b) swap(a, b);
  if (a > c) swap(a, c);
  if (b > c) swap(b, c);
  ll need = max(d - (c - b), d - (b - a));
  if (need % 2 == 0)
    need /= 2;
  else
    need = need / 2 + 1;
  cout << need << endl;
}
