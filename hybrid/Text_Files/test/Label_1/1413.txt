#include<bits/stdc++.h>
using namespace std;
long long int a[300001],f[300001][30],dp[300001][2][30],ans,ck,n,x;
int main()
{
    cin>>n;
    for(int i=1; i<=n; i++)
    {
        cin>>a[i];
    }
    for(int i=1; i<=n; i++)
    {
        x=a[i];
        ck=0;
        for(int j=0; j<30; j++)
        {
            if(x%2==0)
                f[i][j]=ck;
            else
                ck=j+1;
            x=x/2;
        }
    }
    for(int i=2; i<=n; i++)
    {
        for(int j=0; j<30; j++)
        {
            dp[i][0][j]=max(dp[i-1][0][j],dp[i-1][1][j]);
            if(f[i][j]==0)
                dp[i][1][j]=dp[i-1][1][j]+1;
            else
                dp[i][1][j]=max(dp[i-1][0][f[i][j]-1],dp[i-1][1][f[i][j]-1]);
        }
        ans+=dp[i][0][0];
    }
    cout<<ans;
    return 0;
}
