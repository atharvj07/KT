from collections import Counter

n, m = map(int, input().split())
s = input()

c = Counter(s)
if c['a'] % 2 == c['b'] % 2 == c['c'] % 2 == 1:
    print(1)
    for i in range(m):
        l, r = map(int, input().split())
        print((r - l + 1) % 2)
else:
    print(0)
    for i in range(m):
        l, r = map(int, input().split())
        print(0)
