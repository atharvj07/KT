import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int r = input.nextInt();
        int g = input.nextInt();
        int b = input.nextInt();
        int n = input.nextInt();
        int ans = 0;
        for (int i = 0; i <= n / r; i++) {
            for (int j = 0; j <= (n - r * i) / g; j++) {
                int k = (n - r * i - g * j) / b;
                if (r * i + g * j + b * k == n) ans++;
            }
        }
        System.out.println(ans);
    }
}
