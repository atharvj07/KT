n, k = map(int, input().split())
a = list(map(int, input().split()))
b = set(map(int, input().split())) if k else set()

def is_possible(a, b):
    for i in range(1, len(b)):
        if a[b[i]-1] <= a[b[i-1]-1]:
            return False
    return True

if k > 0:
    b = sorted(b)
    if not is_possible(a, b):
        print(-1)
        exit()

operations = 0
last_fixed = 0 if 1 not in b else a[0]
for i in range(1, n):
    if i+1 in b:
        if a[i] <= last_fixed:
            print(-1)
            exit()
        last_fixed = a[i]
    else:
        if a[i] <= last_fixed:
            operations += 1
            last_fixed += 1
        else:
            last_fixed = a[i]

print(operations)
