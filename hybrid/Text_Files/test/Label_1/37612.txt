import sys
input = sys.stdin.readline

n,m=map(int,input().split())
s=input().rstrip()
G=[[] for _ in range(n+1)]
for _ in range(m):
    x,y=map(int,input().split())
    G[x].append(y)

cnt=[[0]*26 for _ in range(n+1)]
for i in range(n):cnt[i+1][ord(s[i])-ord('a')]=1

vis=[0]*(n+1)
ans=0
que=[(1,1)]
while que:
    v,c=que.pop()
    ans=max(ans,c)
    if v==n:continue
    vis[v]=1
    for u in G[v]:
        if not vis[u]:
            cnt[u]=[max(cnt[u][i],cnt[v][i]) for i in range(26)]
            cnt[u][ord(s[u-1])-ord('a')]+=1
            que.append((u,c+1))

print(ans if ans>0 else -1)
