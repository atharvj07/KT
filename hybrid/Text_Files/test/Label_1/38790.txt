n = int(input())
books = []
for _ in range(n):
    thickness, width = map(int, input().split())
    books.append((thickness, width))

books.sort(key=lambda book: book[1], reverse=True)

vertical_thickness = 0
horizontal_width = 0
for thickness, width in books:
    if horizontal_width + width <= vertical_thickness:
        horizontal_width += width
    else:
        vertical_thickness += thickness
        horizontal_width = width

print(vertical_thickness)
