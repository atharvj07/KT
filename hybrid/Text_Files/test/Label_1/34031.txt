import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int c = in.nextInt();
        int[] a = new int[n - 1];
        int[] b = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            b[i] = in.nextInt();
        }
        int[] t = new int[n];
        t[0] = 0;
        for (int i = 1; i < n; i++) {
            t[i] = Math.min(t[i - 1] + a[i - 1], c + t[i - 1] + b[i - 1]);
        }
        for (int i = 0; i < n; i++) {
            System.out.print(t[i] + " ");
        }
        System.out.println();
    }
}
