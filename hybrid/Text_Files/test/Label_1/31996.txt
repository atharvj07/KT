import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if (n == 0) {
            System.out.println(-1);
        } else if (n <= 4) {
            System.out.println(4);
        } else if (n <= 7) {
            System.out.println(7);
        } else {
            int ans = n / 4;
            if (4 * ans == n) {
                for (int i = 0; i < ans; i++) {
                    System.out.print(4);
                }
            } else {
                for (int i = 0; i < ans - 1; i++) {
                    System.out.print(4);
                }
                System.out.print(7);
            }
        }
    }
}
