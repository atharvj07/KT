def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

n = [int(i) for i in input().split()]
d1 = n[1] - n[0]
d2 = int((n[2] - n[1]) / (n[1] - n[0]))
if d1 == 0 and all(i % n[0] == 0 for i in n):
    print(n[0] * int(sum(n) / n[0]))
elif gcd(int((n[2] - n[1]) / d2), int((n[3] - n[2]) / d2)) > 1:
    print(42)
elif all(n[i] == n[0] + d1 * i for i in range(1, 4)):
    print(n[3] + d1)
elif all(n[i] / n[0] ** i == d2 for i in range(1, 4)):
    print(int(n[3] * d2))
else:
    print(42)
