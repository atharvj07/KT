import sys
import math
def check(num, k):
    cnt = [0] * 10
    while num:
        cnt[num % 10] += 1
        num //= 10
    if cnt[0] + cnt[4] + cnt[6] + cnt[9] == k:
        return True
    if cnt[2] + cnt[3] + cnt[5] + cnt[6] + cnt[8] == k:
        return True
    if cnt[1] + cnt[2] + cnt[3] + cnt[5] + cnt[6] + cnt[7] + cnt[8] == k:
        return True
    if cnt[0] + cnt[1] + cnt[2] + cnt[4] + cnt[5] + cnt[6] + cnt[8] + cnt[9] == k:
        return True
    return False
def solve(k):
    if k <= 9:
        if k == 0: return 10
        return k
    for i in range(10):
        for j in range(10):
            if i + j != k: 
                continue
            for x in range(1, 10):
                if check(x, i) and check(x**2, j):
                    return x
    return -1
k = int(input())
ans = solve(k)
print(ans)
