import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int v = sc.nextInt();
        if(m != n - 1 || v == 1){
            System.out.println(-1);
            return;
        }
        Set<Integer> adj[] = new HashSet[n + 1];
        for(int i = 1; i < n + 1; i++){
            adj[i] = new HashSet<>();
        }
        for(int i = 1; i < n + 1; i++){
            if(i != v){
                adj[i].add(v);
                adj[v].add(i);
            }
        }
        for(int i = 1; i < n + 1; i++){
            for(int j = i + 1; j < n + 1; j++){
                if(i != v && j != v){
                    System.out.println(i + " " + j);
                }
            }
        }
    }
}
