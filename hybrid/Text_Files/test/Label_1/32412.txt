import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Scanner;
import java.util.stream.Stream;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
        int[] row = Stream.of(input.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] col = Stream.of(input.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] diag = Stream.of(input.nextLine().split(" ")).mapToInt(Integer::parseInt).toArray();

        int topLeft = (row[1] - row[0] + col[0] - col[1] + diag[0] - diag[1]) / 2;
        int topRight = row[1] - topLeft;
        int bottomLeft = col[1] - topLeft;
        int bottomRight = row[0] + col[0] - topLeft - bottomLeft;

        if (topLeft < 1 || topLeft > 9 || topRight < 1 || topRight > 9 || bottomLeft < 1 || bottomLeft > 9 || bottomRight < 1 || bottomRight > 9) {
            System.out.println(-1);
        } else {
            System.out.println(topLeft + " " + topRight);
            System.out.println(bottomLeft + " " + bottomRight);
        }
    }
}
