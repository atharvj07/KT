import sys

N = int(input())
LRUD = input()

x, y, dx, dy = 0, 0, {'L': -1, 'R': 1, 'U': 0, 'D': 0}[LRUD[0]], {'L': 0, 'R': 0, 'U': 1, 'D': -1}[LRUD[0]]
ans = 1

for m in LRUD[1:]:
    nx, ny = x + {'L': -1, 'R': 1, 'U': 0, 'D': 0}[m], y + {'L': 0, 'R': 0, 'U': 1, 'D': -1}[m]
    if (nx - dx) * (y - ny) != (x - dx) * (ny - dy):
        ans += 1
    x, y, dx, dy = nx, ny, dx, dy

print(ans)
