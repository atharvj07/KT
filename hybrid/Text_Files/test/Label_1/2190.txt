#include <cstdio>
#include <queue>
#include <vector>
using namespace std;

#define int long long
#define REP(i, n) for (int i = 0; i < (int)(n); ++i)
#define REP3(i, m, n) for (int i = (m); i < (int)(n); ++i)
#define REPR(i, n) for (int i = (int)(n)-1; i >= 0; --i)
#define REPR3(i, m, n) for (int i = (int)(n)-1; i >= (int)(m); --i)
#define ALL(v) (v).begin(), v.end()
#define CLR(v) v.clear()
#define SZ(v) (int)(v.size())
#define FI(a, b) fixed << setprecision(b) << a

const int MAX_N = 50, MAX_K = 100;
const int dx[] = {1, -1, 0, 0};
const int dy[] = {0, 0, 1, -1};
const int MAX_MOVES = 20000LL;

int n, k;
int grid[4][MAX_N], dest[4][MAX_N];
int prev_r[MAX_K], prev_c[MAX_K], in_dest[MAX_K];

void solve() {
  REP3(i, 1, k + 1) { dest[0][i] = dest[3][i] = -1; }
  REP(i, 4) {
    REP(j, n) {
      int x = grid[i][j];
      if (x != 0) {
        if (i < 2)
          dest[0][x] = i;
        else
          dest[3][x] = i;
      }
    }
  }

  queue<int> q;
  REP3(i, 1, k + 1) {
    prev_r[i] = prev_c[i] = -1;
    if (grid[1][i] == i || grid[2][i] == i) { q.push(i); }
  }
  while (!q.empty()) {
    int x = q.front();
    q.pop();
    int r = (x - 1) / n + 1;
    int c = (x - 1) % n + 1;
    REP(dir, 4) {
      int rr = r + dx[dir], cc = c + dy[dir];
      if (rr < 1 || rr > 4 || cc < 1 || cc > n) continue;
      if (grid[rr][cc] != 0) continue;
      if (dest[rr][grid[r][c]] == rr) {
        prev_r[grid[r][c]] = r, prev_c[grid[r][c]] = c;
        q.push(grid[r][c]);
        in_dest[grid[r][c]] = 1;
      }
    }
  }

  int cnt = 0;
  REP3(i, 1, k + 1) {
    if (!in_dest[i]) {
      printf("-1\n");
      return;
    }
    while (prev_r[i] != -1) {
      cnt++;
      if (cnt > MAX_MOVES) {
        printf("-1\n");
        return;
      }
      grid[prev_r[i]][prev_c[i]] = 0;
      grid[prev_r[i] + 1][prev_c[i]] = i;
      printf("%lld %lld %lld\n", i, prev_r[i] + 1, prev_c[i]);
      prev_r[i] = prev_c[i] = -1;
    }
  }
  printf("%lld\n", cnt);
}

signed main() {
  scanf("%lld %lld", &n, &k);
  REP(i, 4) {
    REP(j, n) { scanf("%lld", &grid[i][j]); }
  }
  solve();
  return 0;
}
