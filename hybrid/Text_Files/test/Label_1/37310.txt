import sys
input = sys.stdin.readline

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def dfs(u, p):
    ans = 0
    for v in graph[u]:
        if v == p: continue
        ans += dfs(v, u)
        ans = (ans + gcd(beauty[u], beauty[v])) % mod
    return ans

mod = int(1e9 + 7)
n = int(input())
beauty = list(map(int, input().split()))
graph = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

print(dfs(1, 0))
