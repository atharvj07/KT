#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  vector<int> cnt(26, 0);
  for (int i = 0; i < n; i++) { cnt[s[i] - 'a']++; }
  vector<bool> odd(26, false);
  int oddCnt = 0;
  int midIdx = -1;
  char c;
  for (int i = 0; i < 26; i++) {
    if (cnt[i] % 2 == 1) {
      oddCnt++;
      odd[i] = true;
      if (oddCnt > 1) {
        cout << "-1" << endl;
        return 0;
      }
      midIdx = i;
      c = 'a' + i;
    }
  }
  vector<char> res;
  if (oddCnt == 0) {
    midIdx = 0;
    c = 'a';
  }
  int L = 0, R = n - 1;
  while (L < R) {
    for (int i = 0; i < 26; i++) {
      if (cnt[i] > 0) {
        if (L < R) { res.push_back('a' + i); }
        if (L < R) { res.push_back('a' + i); }
        cnt[i] -= 2;
      }
    }
    L++;
    R--;
  }
  if (L == R) {
    if (cnt[midIdx] > 0) { res.push_back(c); }
  }
  reverse(res.begin(), res.end());
  for (int i = 0; i < n; i++) { cout << res[i]; }
  cout << endl;
  return 0;
}
