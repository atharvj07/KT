#include <iostream>
#include <vector>
using namespace std;

int main() {
  vector<int> m(5), w(5);
  cin >> m[0] >> m[1] >> m[2] >> m[3] >> m[4];
  cin >> w[0] >> w[1] >> w[2] >> w[3] >> w[4];
  int hs, hu;
  cin >> hs >> hu;
  long long ans = 0;
  for (int i = 0; i < 5; i++) {
    int pt = 500 + 500 * i;
    ans += max(0.3 * pt, (1.0 - m[i] / 250.0) * pt - 50 * w[i]);
  }
  ans += hs * 100 - hu * 50;
  cout << ans << "\n";
  return 0;
}
