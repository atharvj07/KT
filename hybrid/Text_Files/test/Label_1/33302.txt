import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        boolean stable = false;
        int count = 0;
        int[] b = new int[n];
        while (!stable) {
            stable = true;
            count++;
            for (int i = 1; i < n - 1; i++) {
                int m = (a[i - 1] + a[i] + a[i + 1]) / 2;
                if (b[i] != m) {
                    stable = false;
                    b[i] = m;
                }
            }
            System.arraycopy(b, 0, a, 0, n);
        }
        System.out.println(count);
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
        System.out.println();
    }
}
