import sys

def solve():
    input = sys.stdin.readline
    n, k = [int(x) for x in input().split()]
    events = []
    for _ in range(n):
        l, r, a = [int(x) for x in input().split()]
        events.append((l, a))
        events.append((r + 1, -a))
    events.sort()

    ans, cur = 0, 0
    for t, a in events:
        if a > 0:
            ans += (a + k - 1) // k
        else:
            a = -a
            cur = max(0, cur - a)
    if cur:
        ans += (cur + k - 1) // k
    print(ans if ans < 2e9 else -1)

solve()
