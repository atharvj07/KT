import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), I = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = in.nextLong();
        Arrays.sort(a);
        long L = a[0], R = a[a.length-1];
        long tot = 0;
        for (long x: a) {
            if (x < L) tot += L - x;
            else if (x > R) tot += x - R;
        }
        long ans = 0;
        if (tot > I*8) {
            ans = (tot+4*I-1) / (4*I);
            R = L + 4 * ans;
        }
        for (long x: a) {
            if (x < L) ans += L - x;
            else if (x > R) ans += x - R;
        }
        System.out.println(ans);
    }
}
