import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int d = scanner.nextInt();
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int ans = 0;
        int cur = 1;
        for (int i = 0; i < n; i++) {
            int days = a[i];
            while (cur > days) {
                cur -= days;
            }
            while (cur < days) {
                cur++;
                ans++;
            }
        }
        System.out.println(ans);
    }
}
