import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        final int mod = 1000000007;
        int h = sc.nextInt(), w = sc.nextInt();
        int[] r = new int[h], c = new int[w];
        for (int i = 0; i < h; i++) r[i] = sc.nextInt();
        for (int i = 0; i < w; i++) c[i] = sc.nextInt();
        long[][] dp = new long[h + 1][w + 1];
        dp[0][0] = 1;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                dp[i + 1][j] = (dp[i + 1][j] + dp[i][j] * Math.max(0, c[j] - r[i])) % mod;
                dp[i][j + 1] = (dp[i][j + 1] + dp[i][j] * Math.max(0, r[i] - c[j])) % mod;
            }
        }
        System.out.println(dp[h][w]);
    }
}
