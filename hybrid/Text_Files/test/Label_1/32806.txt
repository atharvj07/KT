import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean possible = true;
        for (int i = 1; i < n; i++) {
            if (gcd(a[i], a[i - 1]) == 1) {
                possible = false;
                break;
            }
        }
        if (possible) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }

    private static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
