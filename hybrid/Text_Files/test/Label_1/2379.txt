#include <bits/stdc++.h>
using namespace std;

const int N = 5001;
const int MOD = 1000000007;

int main() {
    int n; cin >> n;

    vector<int> a = vector<int>(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    vector<vector<int>> dp(n + 1, vector<int>(n + 1, MOD));

    // dp[i][j] = minimum number of operations to delete all elements from the subarray a[i : j]

    // dp[i][i] = 0 for all i
    for (int i = 0; i <= n; i++) dp[i][i] = 0;

    // dp[i][j] = min(dp[i][j], dp[i][k] + dp[k + 1][j]) for all i < k < j
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j <= n; j++) {
            for (int k = i; k < j; k++) {
                dp[i][j] = min(dp[i][j], dp[i][k] + dp[k + 1][j]);
            }
        }
    }

    // dp[0][n] = minimum number of operations to delete all elements from the array
    cout << dp[0][n] << endl;

    return 0;
}
