import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] a = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextLong();
    }
    long[] f = new long[n];
    long[] g = new long[n];
    Arrays.fill(f, Long.MAX_VALUE);
    for (int i = 0; i < n; i++) {
      g[i] = a[i];
    }
    for (int i = 1; i < n; i++) {
      for (int j = i; j < n; j++) {
        g[j] = Math.min(g[j], f[j - i] + a[j]);
      }
      for (int j = 0; j < i; j++) {
        f[j] = g[j];
      }
    }
    long ans = f[n - 1];
    for (int i = 0; i < n; i++) {
      ans = gcd(ans, g[i]);
    }
    System.out.println(ans % 998244353);
  }

  private static long gcd(long a, long b) {
    if (b == 0) {
      return a;
    }
    return gcd(b, a % b);
  }
}
