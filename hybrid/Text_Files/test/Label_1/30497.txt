import java.util.Scanner;

public class JumpStyle {

    static int N;
    static int[][] grid;
    static boolean[][] visited;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            N = scanner.nextInt();
            if (N == 0) {
                break;
            }

            grid = new int[N][N];
            visited = new boolean[N][N];

            for (int i = 0; i < N; i++) {
                for (int j = 0; j < N; j++) {
                    grid[i][j] = scanner.nextInt();
                }
            }

            int count = 0;
            for (int i = 0; i < N; i++) {
                for (int j = 0; j < N; j++) {
                    if (!visited[i][j]) {
                        dfs(i, j);
                        count++;
                    }
                }
            }

            System.out.println(count);
        }

        scanner.close();
    }

    static void dfs(int x, int y) {
        visited[x][y] = true;

        int nextX = grid[x][y] % N;
        int nextY = grid[x][y] / N;

        if (!visited[nextX][nextY]) {
            dfs(nextX, nextY);
        }
    }
}
