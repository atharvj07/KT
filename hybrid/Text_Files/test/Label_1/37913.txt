from sys import stdin as si
n,m=map(int,si.readline().split())
a=[int(x) for x in si.readline().split()]
b=[int(x) for x in si.readline().split()]
for i in range(m):
    t,x=map(int,si.readline().split())
    if t==1:
        y,k=map(int,si.readline().split())
        for j in range(k):b[y+j]=a[x+j]
    else:print(b[x-1])
