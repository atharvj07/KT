import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        int[][] f = new int[n][n];
        int[][] d = new int[n][n];
        for (int i = 0; i < k; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            f[x][y] = 1;
            f[y][x] = 1;
        }
        int m = in.nextInt();
        for (int i = 0; i < m; i++) {
            int x = in.nextInt() - 1;
            int y = in.nextInt() - 1;
            d[x][y] = 1;
            d[y][x] = 1;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            Queue<Integer> q = new LinkedList<>();
            q.add(i);
            int cnt = 0;
            while (!q.isEmpty()) {
                int x = q.poll();
                if (a[x] == 1) {
                    continue;
                }
                a[x] = 1;
                ++cnt;
                for (int j = 0; j < n; j++) {
                    if (f[x][j] == 1 && d[x][j] == 0) {
                        q.add(j);
                    }
                }
            }
            for (int j = 0; j < n; j++) {
                b[j] = 0;
            }
            ans = Math.max(ans, cnt);
        }
        for (int i = 0; i < n; i++) {
            a[i] = 0;
            b[i] = 0;
        }
        for (int i = 0; i < n; i++) {
            Queue<Integer> q = new LinkedList<>();
            q.add(i);
            int cnt = 0;
            while (!q.isEmpty()) {
                int x = q.poll();
                if (b[x] == 1) {
                    continue;
                }
                b[x] = 1;
                ++cnt;
                for (int j = 0; j < n; j++) {
                    if (f[x][j] == 1 && d[x][j] == 0) {
                        q.add(j);
                    }
                }
            }
            for (int j = 0; j < n; j++) {
                a[j] = 0;
            }
            ans = Math.max(ans, cnt);
        }
        System.out.println(ans);
    }
}
