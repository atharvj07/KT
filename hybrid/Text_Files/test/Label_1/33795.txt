import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int[] a = new int[n];
            for (int j = 0; j < n; j++) {
                a[j] = scanner.nextInt();
            }
            int res = 0;
            for (int j = 1; j < n; j++) {
                if (a[j] < a[j - 1]) {
                    res = Math.max(res, 32 - Integer.numberOfLeadingZeros(a[j - 1] - a[j]));
                }
            }
            System.out.println(res);
        }
    }
}
