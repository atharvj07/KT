n, k = map(int, input().split())
a = list(map(int, input().split()))

indices = sorted(range(n), key=lambda i: a[i], reverse=True)
max_indices = sorted(indices[:k])

diffs = [max_indices[0] + 1] + [max_indices[i] - max_indices[i-1] for i in range(1, k)]
diffs[-1] += n - max_indices[-1] - 1

print(sum(a[i] for i in indices[:k]))
print(*diffs)
