import sys
input = sys.stdin.readline

n, s, m, k = map(int, input().split())
a = list(map(int, input().split()))
se = [[] for i in range(n)]
for i in range(s):
    l, r = map(int, input().split())
    l -= 1
    r -= 1
    se[l].append((r, i, 0))
    se[r].append((l, i, 1))
for i in se:
    i.sort()
cnt = [0] * s
ans = {}
cur = []
for j in range(n):
    for x, y, z in se[j]:
        cnt[y] += 1 - 2 * z
        if cnt[y] != 0:
            cur.append(a[j])
    if len(cur) >= k:
        cur.sort()
        ans[j] = cur[k - 1]
    cur = []
if len(ans) >= m:
    ans = sorted(list(ans.values()))
    print(ans[m - 1])
else:
    print(-1)
