#include <bits/stdc++.h>
#define FOR(i, a, b, c) for(int i = a ; i < b ; i += c)
using namespace std ;
int main() {
    int n ; cin >> n ;
    vector<int> p(n) ;
    FOR(i, 0, n, 1) {
        cin >> p[i] ; p[i]-- ;
    }
    vector<int> ans(n + 1) ;
    ans[0] = 1 ;
    FOR(i, 0, n, 1) {
        ans[i + 1] = ans[i] ;
        if(p[i] > i) ans[i + 1]++ ;
    }
    FOR(i, 0, n + 1, 1) cout << ans[i] << " " ;
    return 0 ;
}
