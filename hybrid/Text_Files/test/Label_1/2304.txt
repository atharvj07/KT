#include <iostream>
#include <deque>
#include <vector>
#include <algorithm>
using namespace std;

const int MAXN = 100001;

int n, q, m, a[MAXN], pos[MAXN], ans, c[MAXN];
vector<pair<int, int>> adj[MAXN];
deque<int> dq, tmp;

void add_edge(int x, int y)
{
    adj[x].push_back({y, a[y]});
}

void bfs()
{
    while (!dq.empty()) dq.pop_front();
    for (int i = 1; i <= n; i++)
    {
        c[a[i]] = 0;
        if (a[i] == 0) dq.push_back(i);
    }
    while (!dq.empty())
    {
        int v = dq.front();
        dq.pop_front();
        if (c[a[v]] != 0) continue;
        c[a[v]] = 1;
        for (pair<int, int> u : adj[v])
        {
            if (u.second == a[v]) dq.push_back(u.first);
        }
    }
}

void update()
{
    ans = 0;
    while (!dq.empty()) dq.pop_front();
    for (int i = 1; i <= n; i++)
    {
        if (a[i] == 0) dq.push_back(i);
    }
    while (!dq.empty())
    {
        int v = dq.front();
        dq.pop_front();
        for (pair<int, int> u : adj[v])
        {
            if (a[u.first] == u.second)
            {
                if (c[u.second] != c[a[v]])
                {
                    c[u.second] = c[a[v]];
                    dq.push_back(u.first);
                }
                else if (a[u.first] != a[v]) ans++;
            }
        }
    }
}

int main()
{
    ios_base::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= n; i++) { a[i] = i; pos[i] = i; }
    for (int i = 1; i <= m; i++)
    {
        int x, y;
        cin >> x >> y;
        add_edge(x, y);
        add_edge(y, x);
    }
    bfs();
    cout << ans;
    cin >> q;
    for (int i = 1; i <= q; i++)
    {
        int x;
        cin >> x;
        a[x] = n + i;
        pos[n + i] = x;
        pos[a[x]] = 0;
        a[pos[x]] = 0;
        pos[x] = n + i;
        update();
        cout << ' ' << ans;
    }
    cout << '\n';
    return 0;
}
