import sys
input = sys.stdin.readline

n = int(input())
names = [list(input().split()) for _ in range(n)]
p = list(map(int, input().split()))

first = [names[p[i] - 1][0] for i in range(n)]
last = [names[p[i] - 1][1] for i in range(n)]

if sorted(first) == sorted(last):
    print("NO")
else:
    print("YES")
