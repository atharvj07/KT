#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
    string n;
    cin >> n;
    int sum = 0;
    for (auto c : n) {
        sum += c - '0';
    }
    int ans = sum;
    for (int i = 0; i < n.size(); i++) {
        int num = n[i] - '0';
        if (num == 9) continue;
        int a = num + 1;
        int b = sum - num - 1;
        ans = max(ans, a + b);
    }
    cout << ans << endl;
}
