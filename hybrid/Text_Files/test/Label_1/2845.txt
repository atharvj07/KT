#include<bits/stdc++.h>
using namespace std;
int main(){
    string s;
    cin>>s;
    int open=0,ques=0;
    for(auto c:s){
        if(c=='('){++open;}
        else if(c=='#'){++ques;}
        else if(open){--open;}
        else{cout<<-1<<endl;return 0;}
    }
    if(ques!=open){cout<<-1<<endl;return 0;}
    for(auto &c:s){
        if(c=='#'){
            if(open){--open;++c;}
            else c='(';
        }
    }
    for(auto c:s)cout<<(c=='#'?open:c-'(')<<endl;
    return 0;
}
