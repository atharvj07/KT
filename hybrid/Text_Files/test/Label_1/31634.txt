import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class TheseusAndMinotaur {

    private static final char WALL = '*';
    private static final int[][] DIRECTIONS = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        char[][] labyrinth = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = scanner.next();
            for (int j = 0; j < m; j++) {
                labyrinth[i][j] = row.charAt(j);
            }
        }

        int startX = scanner.nextInt() - 1;
        int startY = scanner.nextInt() - 1;
        int endX = scanner.nextInt() - 1;
        int endY = scanner.nextInt() - 1;

        int minMinutes = findMinMinutes(labyrinth, startX, startY, endX, endY);
        System.out.println(minMinutes);
    }

    private static int findMinMinutes(char[][] labyrinth, int startX, int startY, int endX, int endY) {
        int n = labyrinth.length;
        int m = labyrinth[0].length;

        Queue<State> queue = new LinkedList<>();
        queue.add(new State(startX, startY, 0));

        boolean[][] visited = new boolean[n][m];
        visited[startX][startY] = true;

        while (!queue.isEmpty()) {
            State state = queue.poll();
            int currentX = state.x;
            int currentY = state.y;
            int minutes = state.minutes;

            if (currentX == endX && currentY == endY) {
                return minutes;
            }

            for (int[] direction : DIRECTIONS) {
                int nextX = currentX + direction[0];
                int nextY = currentY + direction[1];

                if (isValid(labyrinth, nextX, nextY) && !visited[nextX][nextY] && canPass(labyrinth, currentX, currentY, nextX, nextY)) {
                    visited[nextX][nextY] = true;
                    queue.add(new State(nextX, nextY, minutes + 1));
                }
            }

            // Rotate the block
            if (labyrinth[currentX][currentY] != WALL) {
                rotateBlock(labyrinth, currentX, currentY);
                for (int[] direction : DIRECTIONS) {
                    int nextX = currentX + direction[0];
                    int nextY = currentY + direction[1];

                    if (isValid(labyrinth, nextX, nextY) && !visited[nextX][nextY] && canPass(labyrinth, currentX, currentY, nextX, nextY)) {
                        visited[nextX][nextY] = true;
                        queue.add(new State(nextX, nextY, minutes + 1));
                    }
                }
                // Rotate the block back
                rotateBlock(labyrinth, currentX, currentY);
            }
        }

        return -1;
    }

    private static boolean isValid(char[][] labyrinth, int x, int y) {
        int n = labyrinth.length;
        int m = labyrinth[0].length;
        return x >= 0 && x < n && y >= 0 && y < m;
    }

    private static boolean canPass(char[][] labyrinth, int currentX, int currentY, int nextX, int nextY) {
        char currentBlock = labyrinth[currentX][currentY];
        char nextBlock = labyrinth[nextX][nextY];

        if (currentBlock == WALL || nextBlock == WALL) {
            return false;
        }

        if (currentX == nextX) {
            return (currentBlock == '-' || currentBlock == 'L' || currentBlock == 'R') && (nextBlock == '-' || nextBlock == 'L' || nextBlock == 'R');
        } else {
            return (currentBlock == '|' || currentBlock == 'U' || currentBlock == 'D') && (nextBlock == '|' || nextBlock == 'U' || nextBlock == 'D');
        }
    }

    private static void rotateBlock(char[][] labyrinth, int x, int y) {
        char currentBlock = labyrinth[x][y];
        switch (currentBlock) {
            case '+':
                labyrinth[x][y] = '|';
                break;
            case '-':
                labyrinth[x][y] = '-';
                break;
            case '|':
                labyrinth[x][y] = '+';
                break;
            case '^':
                labyrinth[x][y] = '>';
                break;
            case '>':
                labyrinth[x][y] = 'v';
                break;
            case '<':
                labyrinth[x][y] = '^';
                break;
            case 'v':
                labyrinth[x][y] = '<';
                break;
            case 'L':
                labyrinth[x][y] = 'U';
                break;
            case 'R':
                labyrinth[x][y] = 'D';
                break;
            case 'U':
                labyrinth[x][y] = 'R';
                break;
            case 'D':
                labyrinth[x][y] = 'L';
                break;
        }
    }

    private static class State {
        int x;
        int y;
        int minutes;

        public State(int x, int y, int minutes) {
            this.x = x;
            this.y = y;
            this.minutes = minutes;
        }
    }
}
