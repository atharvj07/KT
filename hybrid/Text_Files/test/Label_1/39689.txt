def find_possible_lengths(n, nums):
    counts = {}
    for num in nums:
        counts[num] = counts.get(num, 0) + 1
    
    max_pow = max(nums).bit_length()
    pows = [2**i for i in range(max_pow + 1)]
    
    def check(k):
        temp_counts = counts.copy()
        for pow_val in reversed(pows[:k+1]):
            if pow_val not in temp_counts:
                return False
            temp_counts[pow_val] -= 1
            if temp_counts[pow_val] == 0:
                del temp_counts[pow_val]
            pow_half = pow_val // 2
            while pow_half > 0:
                if pow_half not in temp_counts or temp_counts[pow_half] == 0:
                    return False
                temp_counts[pow_half] -= 1
                if temp_counts[pow_half] == 0:
                    del temp_counts[pow_half]
                pow_half //= 2
        return sum(temp_counts.values()) == 0
    
    possible_lengths = []
    for k in range(max_pow + 1):
        if check(k):
            possible_lengths.append(k + 1)
    if not possible_lengths:
        print(-1)
    else:
        print(*possible_lengths)

n = int(input().strip())
nums = list(map(int, input().strip().split()))
find_possible_lengths(n, nums)
