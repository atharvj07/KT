n, m = map(int, input().split())
a = [input() for _ in range(n)]
ans = 0
for i in range(n):
    for j in range(m):
        for k in range(i + 1, n):
            if a[i][j] != a[k][j]:
                for l in range(k + 1, n):
                    if a[i][j] != a[l][j] and a[k][j] != a[l][j]:
                        ans += 1
print(ans)
