import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        long ans = Long.MIN_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                long sum = 0;
                for (int k = j; k < n; k++) {
                    sum += a[i] + a[j] + a[k];
                    ans = Math.max(ans, sum);
                }
            }
        }
        System.out.println(ans);
    }
}
