import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            char[] s = input.next().toCharArray();
            int count = 0;
            int q = 0;
            for (char c : s) {
                if (c == '0' || c == '1') {
                    if (c == '1') {
                        q++;
                    }
                    count += q;
                } else {
                    q = 0;
                }
            }
            System.out.println(count);
        }
    }
}
