n = int(input())
a = list(map(int, input().split()))
f = [1] * n
for i in range(n):
    for j in range(i):
        if a[j] > a[i] and f[j] + 1 > f[i]:
            f[i] = f[j] + 1
mx = max(f)
for i in range(n):
    if f[i] == mx:
        l = i + 1
        break
for i in range(n-1,-1,-1):
    if f[i] == mx:
        r = i + 1
        break
if l == r:
    pass
else:
    print(l, r)
    for i in range(l, r + 1, 2):
        a[i - 1], a[i] = a[i], a[i - 1]
    mx -= 1
    for i in range(n):
