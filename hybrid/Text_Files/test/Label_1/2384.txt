#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k; cin >> n >> k;
    string a; cin >> a;
    int arr[26] = {};
    int cnt = 0;
    for (char x : a) {
        if (arr[x - 'A'] == 0) cnt++;
        arr[x - 'A']++;
        if (cnt > k) {
            puts("YES");
            return 0;
        }
    }
    puts("NO");
}
