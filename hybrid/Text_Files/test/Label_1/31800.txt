import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] difficulties = new int[n];
        for (int i = 0; i < n; i++) {
            difficulties[i] = scanner.nextInt();
        }

        // Sort the difficulties in ascending order.
        Arrays.sort(difficulties);

        int count = 0;

        // Find the number of K values that make the number of problems for ARCs and the number of problems for ABCs the same.
        for (int k = difficulties[0]; k <= difficulties[n - 1]; k++) {
            int numArcs = 0;
            int numAbcs = 0;
            for (int d : difficulties) {
                if (d >= k) {
                    numArcs++;
                } else {
                    numAbcs++;
                }
            }
            if (numArcs == numAbcs) {
                count++;
            }
        }

        // Print the number of K values.
        System.out.println(count);
    }
}
