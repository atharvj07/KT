import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        long min = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            arr[i] = in.nextInt();
            min = Math.min(min, arr[i]);
        }

        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == min) {
                ans += n - 1 - i;
            }
        }
        System.out.println(ans);
    }
}
