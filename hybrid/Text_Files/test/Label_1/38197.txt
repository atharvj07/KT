import sys
from typing import List

def solve(n: int, arr: List[int]) -> int:
    arr.sort()
    return min([abs(max(arr[:idx+1]) - min(arr[idx+1:])) for idx in range(n-1)])

t = int(input())
for _ in range(t):
    n = int(input())
    arr = [int(x) for x in input().split()]
    print(solve(n, arr))
