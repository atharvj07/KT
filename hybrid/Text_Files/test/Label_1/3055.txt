#include <bits/stdc++.h>
using namespace std;
int main(){
    int n, m;
    cin >> n >> m;
    if(m > (n * (n - 1) / 2)){
        cout << "NO" << endl;
        return 0;
    }
    map<int, vector<int>> c;
    int a, b;
    for(int i = 0; i < m; i++){
        cin >> a >> b;
        if(a > b)
            swap(a, b);
        c[a].push_back(b);
    }
    int ord = 2;
    int p = 0;
    vector<int> ans(n), ans2(n);
    for(auto i: c){
        if(i.second.size() > 7){
            cout << "NO" << endl;
            return 0;
        }
        for(int j: i.second){
            ans[p] = ord;
            ans2[j] = ord;
            p++;
            ord++;
        }
    }
    if(n - 1 - p > ord){
        cout << "NO" << endl;
        return 0;
    }
    for(int i = p; i < n; i++){
        ans[i] = ord;
        ans2[i] = ord + 1;
        ord++;
    }
    cout << "YES" << endl;
    for(int i: ans)
        cout << i << " ";
    cout << endl;
    for(int i: ans2)
        cout << i << " ";
    cout << endl;
}
