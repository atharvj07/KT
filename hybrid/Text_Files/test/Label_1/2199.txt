#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    cin.tie(0);
    cin.sync_with_stdio(0);
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string a, b;
        cin >> a >> b;
        bool ok = 1;
        for (int i = 0; i < n; ++i) {
            if (i >= k - 1 && a.substr(i - k + 1, k) == a.substr(i - k + 1, k - 1) + string(k - 1, a[i])) {
                ok &= (b[i] == a[i] + 1);
            } else {
                ok &= (a[i] == b[i]);
            }
        }
        cout << (ok ? "YES" : "NO") << '\n';
    }
    return 0;
}
