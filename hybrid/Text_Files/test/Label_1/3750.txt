#include <iostream>
#include <string>
using namespace std;
int main() {
  string s;
  cin >> s;
  int ans = 0;
  int cur = 0;
  for (int i = 0; i < s.size(); i++) {
    if (i > 0 && s[i] == s[i - 1])
      ++cur;
    else
      cur = 1;
    ans = max(ans, cur);
  }
  cout << ans << endl;
  return 0;
}
