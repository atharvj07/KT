#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, s;
  cin >> n >> s;
  vector<pair<int, int>> buy, sell;
  for (int i = 0; i < n; ++i) {
    char d;
    int p, q;
    cin >> d >> p >> q;
    if (d == 'B') buy.push_back({p, q});
    else sell.push_back({p, q});
  }
  sort(buy.begin(), buy.end(), greater<pair<int, int>>());
  sort(sell.begin(), sell.end());
  for (int i = 0; i < min(s, (int)buy.size()); ++i) cout << "B " << buy[i].first << " " << buy[i].second << '\n';
  for (int i = 0; i < min(s, (int)sell.size()); ++i) cout << "S " << sell[i].first << " " << sell[i].second << '\n';
  return 0;
}
