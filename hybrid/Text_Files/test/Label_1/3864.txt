#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(n), b(m);
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(b.begin(), b.end());
  int ans = n;
  for (int i = 0, j = 0; i < n; i++) {
    while (j < m && b[j] < a[i]) j++;
    if (j < m && b[j] == a[i]) ans--;
  }
  cout << ans << endl;
  return 0;
}
