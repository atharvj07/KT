import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int max = 0;
        for (int i = 0; i < 100000; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                sum += i % a[j];
            }
            max = Math.max(max, sum);
        }
        System.out.println(max);
    }
}
