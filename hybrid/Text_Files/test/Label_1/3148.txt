#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int maxn = 1e5 + 5;
int n, x;
int main() {
    cin >> n >> x;
    int a[n];
    unordered_map<int, int> m;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        m[x ^ a[i]]++;
    }
    ll ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += m[a[i]];
        if ((x ^ a[i]) == a[i]) --ans;
    }
    cout << (ans / 2) << "\n";
    return 0;
}
