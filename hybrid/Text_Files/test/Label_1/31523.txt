import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String s = in.next();
        int[] cnt = new int[26];
        for (char c : s.toCharArray()) {
            cnt[c - 'a']++;
        }
        int odd = 0;
        for (int i = 0; i < 26; i++) {
            if (cnt[i] % 2 == 1) {
                odd++;
            }
        }
        if (odd > 1) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < 26; i++) {
                if (cnt[i] % 2 == 1) {
                    for (int j = 0; j < cnt[i] / 2; j++) {
                        sb.append('0');
                    }
                    sb.append('1');
                    for (int j = 0; j < cnt[i] / 2; j++) {
                        sb.append('0');
                    }
                } else {
                    for (int j = 0; j < cnt[i] / 2; j++) {
                        sb.append('0');
                    }
                    for (int j = 0; j < cnt[i] / 2; j++) {
                        sb.append('1');
                    }
                }
            }
            System.out.println(sb.toString());
        }
    }
}
