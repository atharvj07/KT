import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int c = in.nextInt();
        int v0 = in.nextInt();
        int v1 = in.nextInt();
        int a = in.nextInt();
        int l = in.nextInt();
        int day = 1;
        int pages = v0;
        while (pages < c) {
            day++;
            pages += Math.min(v0 + a * (day - 1), v1) - l;
        }
        System.out.println(day);
    }
}
