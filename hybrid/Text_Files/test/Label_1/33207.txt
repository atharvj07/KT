import java.util.*;
import java.io.*;

public class Main {
    private static final int INF = Integer.MAX_VALUE / 2;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int testCase = 0;
        while (true) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());
            if (n == 0 && m == 0) {
                break;
            }

            testCase++;
            int[][] adj = new int[m + 1][m + 1];
            for (int i = 0; i < m + 1; i++) {
                for (int j = 0; j < m + 1; j++) {
                    adj[i][j] = INF;
                }
            }

            for (int i = 0; i < n; i++) {
                st = new StringTokenizer(br.readLine());
                int a = Integer.parseInt(st.nextToken());
                int b = Integer.parseInt(st.nextToken());
                int c = Integer.parseInt(st.nextToken());
                int d = Integer.parseInt(st.nextToken());
                adj[a][b] = c;
                adj[b][a] = c;
                adj[a][m] = Math.min(adj[a][m], d);
                adj[b][m] = Math.min(adj[b][m], d);
            }

            int[] dist = new int[m + 1];
            Arrays.fill(dist, INF);
            dist[m] = 0;

            dijkstra(adj, dist, m);

            int q = Integer.parseInt(br.readLine());
            for (int i = 0; i < q; i++) {
                st = new StringTokenizer(br.readLine());
                int p = Integer.parseInt(st.nextToken());
                int r = Integer.parseInt(st.nextToken());
                int v = Integer.parseInt(st.nextToken());
                if (v == 0) {
                    System.out.println(dist[p] + dist[r]);
                } else {
                    System.out.println(Math.min(dist[p], dist[r]));
                }
            }
        }
    }

    private static void dijkstra(int[][] adj, int[] dist, int v) {
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(v, 0));

        while (!pq.isEmpty()) {
            Edge e = pq.poll();
            int current = e.v;
            int cost = e.w;
            if (dist[current] < cost) {
                continue;
            }

            for (int next = 1; next <= v; next++) {
                if (adj[current][next] != INF) {
                    int newCost = cost + adj[current][next];
                    if (newCost < dist[next]) {
                        dist[next] = newCost;
                        pq.add(new Edge(next, newCost));
                    }
                }
            }
        }
    }

    private static class Edge implements Comparable<Edge> {
        int v;
        int w;

        public Edge(int v, int w) {
            this.v = v;
            this.w = w;
        }

        @Override
        public int compareTo(Edge other) {
            return Integer.compare(this.w, other.w);
        }
    }
}
