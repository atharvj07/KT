import sys
input = sys.stdin.readline

n = int(input())
a = [int(x) for x in input().split()]
gcd = [a[0]]
for i in a[1:]:
    gcd.append(max(gcd[-1], i))
ans = []
for i in range(n):
    for j in range(i + 1, n):
        if gcd[j] % a[i] == 0:
            ans.append((i + 1, j + 1))
print(len(ans), ans[0][1] - ans[0][0])
for x in ans:
    print(x[0])
