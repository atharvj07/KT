#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;

        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }

        int height = 1;
        bool alive = true;
        for (int i = 0; i < n; i++) {
            if (a[i] == 1) {
                if (i > 0 && a[i - 1] == 1) {
                    height += 5;
                } else {
                    height += 1;
                }
            } else {
                if (i > 0 && a[i - 1] == 0) {
                    alive = false;
                    break;
                }
            }
        }

        cout << (alive ? height : -1) << endl;
    }

    return 0;
}
