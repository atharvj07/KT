import java.util.Scanner;

public class Main {
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        char[][] matrix = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                matrix[i][j] = s.charAt(j);
            }
        }
        int[][] dp = new int[n][m]; //dp[i][j] stores the number of all possible beautiful submatrixes with matrix[i][j] as the top-left corner
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (i == 0 || j == 0) { //if we are at the top row or the leftmost column, we can only have a single beautiful submatrix
                    dp[i][j] = 1;
                } else if (matrix[i][j] == matrix[i - 1][j] && matrix[i][j] == matrix[i][j - 1] && matrix[i][j] == matrix[i - 1][j - 1]) { //if the current cell is equal to its top, left and top-left neighboring cells, we can add the previous beautiful submatrixes and form new beautiful submatrixes
                    dp[i][j] = dp[i - 1][j] + dp[i][j - 1] + dp[i - 1][j - 1];
                } else { //if the current cell is not equal to its top, left and top-left neighboring cells, we cannot form any beautiful submatrixes with this cell as the top-left corner
                    dp[i][j] = 0;
                }
            }
        }
        System.out.println(dp[n - 1][m - 1]);
    }
}
