import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve():
    n = int(input())
    a = [int(x) for x in input().split()]
    for i in range(n):
        for j in range(n):
            if i == j: continue
            if gcd(a[i], a[j]) == 1:
                return "No"
    return "Yes"

print(solve())
