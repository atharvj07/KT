import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        int[][] min = new int[n + 1][m + 1];
        int[][] max = new int[n + 1][m + 1];
        for (int i = 0; i <= n; i++) {
            Arrays.fill(min[i], n + 1);
            Arrays.fill(max[i], 0);
        }
        for (int i = 0; i < m; i++) {
            a[i] = scanner.nextInt();
        }
        for (int i = 0; i < m; i++) {
            min[a[i]][i + 1] = Math.min(min[a[i]][i + 1], i + 1);
            max[a[i]][i + 1] = Math.max(max[a[i]][i + 1], i + 1);
            for (int j = a[i] + 1; j <= n; j++) {
                min[j][i + 1] = Math.min(min[j][i + 1], min[j][i] + 1);
                max[j][i + 1] = Math.max(max[j][i + 1], max[j][i] + 1);
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(min[i][m] + " " + max[i][m]);
        }
    }
}
