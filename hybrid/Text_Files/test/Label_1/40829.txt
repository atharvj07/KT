from collections import deque, defaultdict
n, m = map(int, input().split())
graph = defaultdict(list)
for i in range(m):
    a, b = map(int, input().split())
    graph[a].append(b)
dist = [0] * (n+1)
q = deque([n])
while q:
    u = q.popleft()
    for v in graph[u]:
        if dist[v] == 0:
            dist[v] = dist[u] + 1
            q.append(v)
print(*sorted(range(1, n+1), key=lambda x: dist[x]))
