#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < n; i++)
using namespace std;
const int maxn = 1e5 + 5;
int a[maxn], r[maxn], deg[maxn];
vector<int> adj[maxn];

int main() {
  int n;
  cin >> n;
  rep(i, n) { cin >> a[i]; r[a[i] - 1] = i; }
  rep(i, n - 1) {
    int u = r[i], v = r[i + 1];
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg[u]++;
    deg[v]++;
  }
  int ans = 0;
  queue<int> q;
  rep(i, n) {
    if (deg[i] == 1) q.push(i);
  }
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    ans++;
    for (auto v : adj[u]) {
      deg[v]--;
      if (deg[v] == 1) q.push(v);
    }
  }
  cout << ans << endl;
  return 0;
}
