def can_build_reserve(n, lairs):
    above = any(y > 0 for _, y in lairs)
    below = any(y < 0 for _, y in lairs)
    return not (above and below)

def find_min_radius(n, lairs):
    if not can_build_reserve(n, lairs):
        return -1
    left = 0
    right = 10**14
    for _ in range(100):  # Binary search iterations
        mid = (left + right) / 2
        can_make = False
        min_x = -10**14
        max_x = 10**14
        for x, y in lairs:
            if 2 * mid < abs(y):
                d = (abs(y) * abs(y) / (4 * mid)) - mid
                r = (y * y) / (4 * mid)
                if y > 0:
                    min_x = max(min_x, x - r)
                    max_x = min(max_x, x + r)
                else:
                    d = -d
                    min_x = max(min_x, x + d - r)
                    max_x = min(max_x, x + d + r)
            if min_x > max_x:
                can_make = False
                break
            else:
                can_make = True
        if can_make:
            right = mid
        else:
            left = mid
    return right / 2

n = int(input())
lairs = [tuple(map(int, input().split())) for _ in range(n)]
answer = find_min_radius(n, lairs)
print(f"{answer:.6f}" if answer != -1 else -1)
