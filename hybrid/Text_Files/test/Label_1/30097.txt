import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();
    for (int tt = 0; tt < t; tt++) {
      int n = sc.nextInt();
      int[][] p = new int[n][2];
      for (int i = 0; i < n; i++) {
        p[i][0] = sc.nextInt();
        p[i][1] = sc.nextInt();
      }
      if (!valid(p, n)) {
        System.out.println("NO");
      } else {
        System.out.println("YES");
        int[][] ord = new int[n][2];
        for (int i = 0; i < n; i++) {
          ord[i][0] = p[i][0] + p[i][1];
          ord[i][1] = p[i][0] - p[i][1];
        }
        Arrays.sort(ord, (a, b) -> Integer.compare(a[0], b[0]));
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
          int x = ord[i][0] == 0 ? 1 : (ord[i][0] < 0 ? -1 : 1);
          int y = ord[i][1] == 0 ? 1 : (ord[i][1] < 0 ? -1 : 1);
          while (x > 0) {
            sb.append("R");
            x--;
          }
          while (y > 0) {
            sb.append("U");
            y--;
          }
        }
        System.out.println(sb);
      }
    }
  }

  public static boolean valid(int[][] p, int n) {
    int[] r = new int[1001];
    int[] c = new int[1001];
    for (int i = 0; i < n; i++) {
      if (p[i][0] == 0 && p[i][1] == 0) {
        return false;
      }
      if (r[p[i][0]] == 1 || c[p[i][1]] == 1) {
        return false;
      }
      r[p[i][0]] = 1;
      c[p[i][1]] = 1;
    }
    return true;
  }
}
