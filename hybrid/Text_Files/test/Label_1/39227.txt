s=input()
a=[0]*len(s)
b=[0]*len(s)
k=1
t=0
for i in range(0,len(s)):
    if i!=0 and s[i-1]==s[i]:
        a[i]=a[i-1]
    else:
        a[i]=k
        k+=1
k=1
for i in range(len(s)-1,-1,-1):
    if i!=len(s)-1 and s[i+1]==s[i]:
        b[i]=b[i+1]
    else:
        b[i]=k
        k+=1
for i in range(0,len(s)):
    if a[i]!=b[i]:
        t+=1
print(t)
