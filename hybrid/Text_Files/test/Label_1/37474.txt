n = int(input())
i = 0
a = []
while i < n:
    a.append(list(input()))
    a[i][0] = chr(ord(a[i][0]) - 32)
    i += 1
a.sort()
i = 0
while i < n:
    a[i][0] = chr(ord(a[i][0]) + 32)
    k = 1
    j = i + 1
    while k < len(a[i]) and j < n:
        if a[i][k] < a[j][k]:
            k += 1
        else:
            a[j] = a[j][:k]
            j += 1
    i += 1
i = 0
while i < n:
    print(''.join(a[i]))
    i += 1
