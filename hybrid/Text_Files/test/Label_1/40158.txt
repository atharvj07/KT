import sys

n, m = map(int, input().split())
d = list(map(int, input().split()))
g = [[] for i in range(n)]
for i in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    g[u].append(v)
    g[v].append(u)
    if d[u] == d[v] == -1:
        print(-1)
        sys.exit()
vis = [False for i in range(n)]
ans = []
def dfs(v, p, deg):
    vis[v] = True
    ans.append((deg, v))
    for u in g[v]:
        if u == p:
            continue
        if d[u] == d[v] == 0 and deg % 2 == 1:
            print(-1)
            sys.exit()
        if d[u] == -1:
            dfs(u, v, deg)
        else:
            dfs(u, v, deg + 1 - d[u])
dfs(0, -1, 0)
print(len(ans))
for deg, v in ans:
    print(v + 1)
