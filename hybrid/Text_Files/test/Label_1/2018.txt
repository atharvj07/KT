#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, x, m, y;
    cin >> n >> x >> m >> y;
    vector<int> a(n), b(m);
    for (int i =0 ; i< n ; ++i) cin >> a[i];
    for (int i =0 ; i< m ; ++i) cin >> b[i];
    long long Xa = 0, Ya = 0, pxa = 1, pya = 1;
    for (int i = 0; i < a.size(); ++i) {
        Xa = Xa + a[i] * pxa;
        pxa *= x;
    }
    for (int i = 0; i < b.size(); ++i) {
        Ya = Ya + b[i] * pya;
        pya *= y;
    }
    if (Xa == Ya) cout << "=";
    else if (Xa < Ya) cout << "<";
    else cout << ">";
    return 0;
}
