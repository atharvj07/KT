import java.util.*;

public class Main {

    static int n, d, x;
    static int[][] p;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        d = sc.nextInt();
        x = sc.nextInt();
        p = new int[d + 1][n + 1];
        for (int i = 1; i <= d; i++) {
            for (int j = 1; j <= n; j++) {
                p[i][j] = sc.nextInt();
            }
        }
        System.out.println(solve());
    }

    static int solve() {
        int[][] dp = new int[d + 1][x + 1];
        for (int i = 1; i <= d; i++) {
            for (int j = 0; j <= x; j++) {
                for (int k = 1; k <= n; k++) {
                    if (j >= p[i][k]) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - p[i][k]] + p[i][k]);
                    }
                }
                dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
            }
        }
        return dp[d][x];
    }
}
