import sys
sys.setrecursionlimit(100000)
n,m=map(int,input().split())
edge = [[]]
for i in range(n): edge.append([])
for i in range(m):
 a,b=map(int,input().split())
 edge[a].append(b)
 edge[b].append(a)
color = [-1]*n
ans=-1
def dfs(now,col):
   global ans
   if color[now]!=-1:
      ans = -1
      return
   color[now] = col
   for nxt in edge[now]:
      if color[nxt]==-1: dfs(nxt, (col+1)%3)
      elif color[nxt]==color[now]:
         ans = -1
         return
dfs(1,0)
if ans==-1: print(ans)
else:
   print(*[color[i]+1 for i in range(1,n+1)])
