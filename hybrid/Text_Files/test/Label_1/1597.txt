#include<bits/stdc++.h>
using namespace std;
const int N=105;
int n,a[N];
int main(){
    int t;
    cin>>t;
    while(t--){
        cin>>n;
        int d=0;
        for(int i=0;i<n;++i){
            cin>>a[i];
            a[i]=max(i+1,a[i]);
            d=max(d,a[i]);
        }
        cout<<d-n<<'\n';
    }
}
