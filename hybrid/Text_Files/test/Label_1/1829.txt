#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        map<int, int> m;
        for (int i = 0, x; i < n; i++) {
            cin >> x;
            m[x]++;
        }
        int ans = 0, num = 0;
        for (auto x : m) {
            ans += (x.second % 2 == 1);
            if (x.second % 2 == 1) num = x.first;
        }
        if (m.begin()->first > 1) {
            ans++;
            num = m.begin()->first;
        }
        cout << ans << '\n';
        for (auto x : m) {
            if (x.second % 2 == 1 || x.first == num) {
                cout << x.first << ' ';
            }
        }
        cout << '\n';
    }
    return 0;
}
