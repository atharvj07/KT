from typing import List, Tuple
import sys

def dfs(vert: int, p: int, e: int, d: List[int], gr: dict, used: set, ans: list) -> bool:
    used.add(vert)
    ans.append(e)
    for v in gr[vert]:
        if v == p:
            continue
        if v in used:
            if (d[vert] + d[v]) % 2 != (e % 2):
                return False
            continue
        if dfs(v, vert, e + 1, d, gr, used, ans):
            return True
        ans.pop()
    return False

n, m = map(int, input().split())
d = list(map(int, input().split()))
gr = {i: [] for i in range(1, n + 1)}
for j in range(m):
    u, v = map(int, input().split())
    gr[u].append(v)
    gr[v].append(u)

ans = []
used = set()
if dfs(1, -1, 0, d, gr, used, ans):
    print(len(ans))
    for res in ans:
        print(res)
else:
    print(-1)
