#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, x, ans = 0, mx = 0;
    vector<int> v(100001, 0);
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> x;
        v[x]++;
        mx = max(mx, x);
    }
    for (int i = 1; i <= mx; i++) {
        if (v[i] && !(--v[ans + 1])) ans++;
    }
    cout << ans << endl;
    return 0;
}
