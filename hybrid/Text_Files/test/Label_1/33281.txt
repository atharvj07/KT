import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[][] rooms = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                rooms[i][j] = scanner.nextInt();
            }
        }
        int m = scanner.nextInt();
        int[][] wallpapers = new int[m][3];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                wallpapers[i][j] = scanner.nextInt();
            }
        }
        int[][] dp = new int[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], Integer.MAX_VALUE);
        }
        int result = Integer.MAX_VALUE;
        for (int i = 0; i < m; i++) {
            int min = Integer.MAX_VALUE;
            for (int j = 0; j < n; j++) {
                int perimeter = 2 * (rooms[j][0] + rooms[j][1]);
                int rolls = (int) Math.ceil((double) perimeter / wallpapers[i][1]);
                min = Math.min(min, rolls * wallpapers[i][2]);
            }
            result = Math.min(result, min);
        }
        System.out.println(result);
    }
}
