import java.util.Scanner;

class Main {
    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        // dp[i][j][0/1]: the number of ways to assign numbers to the first i houses
        // such that the penguin can/cannot walk to house 1 from house i, and the last
        // number assigned was j
        long[][][] dp = new long[n + 1][n + 1][2];
        dp[0][0][0] = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                // If the penguin can walk to house 1 from house i-1, it can
                // walk to house 1 from house i if j <= k, or if j > k and the
                // last number assigned was <= k
                dp[i][j][0] = (dp[i - 1][j][0] + dp[i - 1][j][1]) % MOD;
                if (j <= k || dp[i - 1][j][1] > 0)
                    dp[i][j][0] = (dp[i][j][0] + dp[i - 1][j - 1][0]) % MOD;

                // If the penguin cannot walk to house 1 from house i-1, it
                // cannot walk to house 1 from house i if j <= k
                dp[i][j][1] = dp[i - 1][j][1];
                if (j > k)
                    dp[i][j][1] = (dp[i][j][1] + dp[i - 1][j - 1][1]) % MOD;
            }
        }

        System.out.println(dp[n][1][0]);
    }
}
