import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long p = sc.nextLong();
        int w = sc.nextInt();
        int d = sc.nextInt();

        if (n * w < p || p % d != 0) {
            System.out.println(-1);
        } else {
            long x = (p / d + n - 1) / n;
            long y = (p - x * w) / d;
            long z = n - x - y;
            System.out.println(x + " " + y + " " + z);
        }
    }
}
