import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = scanner.nextInt();
        }

        int count = 0;
        int left = 0;
        int right = n - 1;
        while (left <= right) {
            int mid = (left + right) / 2;
            int problemsForArcs = 0;
            for (int i = mid; i < n; i++) {
                if (d[i] >= d[mid]) {
                    problemsForArcs++;
                } else {
                    break;
                }
            }
            int problemsForAbcs = n - mid - problemsForArcs;
            if (problemsForArcs == problemsForAbcs) {
                count++;
                left = mid + 1;
            } else if (problemsForArcs > problemsForAbcs) {
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }

        System.out.println(count);
    }
}
