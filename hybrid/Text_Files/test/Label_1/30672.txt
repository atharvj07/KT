import java.util.Scanner;

public class MinimalInclusionSegment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int[] count = new int[100001];
        int distinct = 0, l = 0, r = 0, minLength = Integer.MAX_VALUE;

        for (int i = 0; i < n; i++) {
            if (count[a[i]]++ == 0) {
                distinct++;
            }

            while (distinct > k) {
                if (--count[a[l]] == 0) {
                    distinct--;
                }
                l++;
            }

            if (distinct == k && i - l + 1 < minLength) {
                r = i;
                minLength = i - l + 1;
            }
        }

        System.out.println(l + " " + r);
    }
}
