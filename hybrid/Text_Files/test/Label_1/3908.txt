#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int M=200+5,N=1e9+5;
int id[M],a[M],n,k,ans,ansid[M];
bool vis[N];
set<int>s;
int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin>>n>>k;
    for(int i=1; i<=n; i++)cin>>id[i];
    for(int i=1; i<=n; i++){
        if(vis[id[i]]==0)
        {
            if(s.size()==k){
                vis[a[k]]=0;
                s.erase(a[k]);
                a[k]=id[i];
            }
            else{
                s.insert(id[i]);
                a[++ans]=id[i];
            }
            vis[id[i]]=1;
        }
    }
    cout<<ans<<'\n';
    for(int i=1; i<=ans; i++)cout<<a[i]<<" ";
    return 0;
}
