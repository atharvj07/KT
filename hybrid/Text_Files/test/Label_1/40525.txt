def LCS(s1, s2):
    m, n = len(s1), len(s2)
    dp = [["" for _ in range(n+1)] for _ in range(m+1)]

    for i in range(m+1):
        for j in range(n+1):
            if i == 0 or j == 0:
                dp[i][j] = ""
            elif s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1] + s1[i-1]
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1], key=len)

    return dp[m][n]

def remove_virus(lcs, virus):
    while virus in lcs:
        idx = lcs.index(virus)
        lcs = lcs[:idx] + lcs[idx+len(virus):]
    return lcs

def longest_common_subsequence_without_virus(s1, s2, virus):
    lcs = LCS(s1, s2)
    result = remove_virus(lcs, virus)
    return result if result else "0"

s1 = input().strip()
s2 = input().strip()
virus = input().strip()

print(longest_common_subsequence_without_virus(s1, s2, virus))
