def is_nice(a, n, m):
    for i in range(n):
        for j in range(m):
            if a[i][j] != a[i][m - 1 - j]:
                return False
    for j in range(m):
        for i in range(n):
            if a[i][j] != a[n - 1 - i][j]:
                return False
    return True

def nice(a, n, m):
    ans = 0
    for i in range(n):
        for j in range(m):
            if a[i][j] < a[i][m - 1 - j]:
                ans += a[i][m - 1 - j] - a[i][j]
            else:
                ans += a[i][j] - a[i][m - 1 - j]
    for j in range(m):
        for i in range(n):
            if a[i][j] < a[n - 1 - i][j]:
                ans += a[n - 1 - i][j] - a[i][j]
            else:
                ans += a[i][j] - a[n - 1 - i][j]
    return ans

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = [list(map(int, input().split())) for i in range(n)]
    if is_nice(a, n, m):
        print(0)
    else:
        print(nice(a, n, m))
