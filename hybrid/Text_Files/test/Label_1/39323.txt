import sys

n, m = map(int, input().split())
day = [0] * (n + 1)
for _ in range(m):
    s, d, c = map(int, input().split())
    for j in range(max(s, d - c), d):
        if day[j] > 0:
            print(-1)
            sys.exit()
        day[j] = 1
for j in range(1, n + 1):
    print(day[j] + (j in [s[0] for s in sorted([list(map(int, input().split())) for _ in range(m)], key = lambda x: x[0])]))
