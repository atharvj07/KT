import sys
n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
dp = [[sys.maxsize] * (k+1) for _ in range(n+1)]
dp[0][0] = 0
for i in range(1, n+1):
    for j in range(k+1):
        dp[i][j] = min(dp[i-1][j], dp[i-1][j-1] + a[i-1] + b[i-1])
print(dp[n][k])
