#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
  ll n, k;
  cin >> n >> k;
  if (k & 1) {
    cout << -1 << endl;
    return 0;
  } 
  vector<ll> ans(n + 1);
  ans[1] = ans[2] = 1;
  ll s = 3, t = 2;
  for (ll i = 3; i <= n; ++i) {
    if (i & 1) {
      if (k) {
        ans[i] = t;
        --k;
      } else
        ans[i] = i;
    } else {
      ans[i] = s;
      ++s;
    }
  }
  for (ll i = 1; i <= n; ++i) {
    cout << ans[i] << ' ';
  }
  cout << endl;
  return 0;
}
