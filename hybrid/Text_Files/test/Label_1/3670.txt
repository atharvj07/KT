#include <iostream>
using namespace std;
int a[100001], n, k;
int main() {
  cin >> n >> k;
  if (((n + k - 1) / k) * k != n) {
    cout << "NO";
    return 0;
  }
  int p = (n + k - 1) / k;
  cout << "YES" << endl;
  for (int i = 1; i < k; i++) { a[i] = p, p -= p / 2; }
  a[k] = p;
  for (int i = 1; i <= k; i++) cout << a[i] << ' ';
}
