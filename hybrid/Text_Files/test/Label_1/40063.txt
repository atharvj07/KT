from collections import deque

n, m = map(int, input().split())
a = [list(input()) for _ in range(n)]
vis = [[[False for _ in range(4)] for _ in range(m)] for _ in range(n)]
dist = [[[10**9 for _ in range(4)] for _ in range(m)] for _ in range(n)]
q = deque()
for i in range(n):
    for j in range(m):
        if a[i][j] != '.':
            q.append((i, j, int(a[i][j]), 0))
            vis[i][j][int(a[i][j])] = True
            dist[i][j][int(a[i][j])] = 0
deltas = [(0, -1), (0, 1), (1, 0), (-1, 0)]
while q:
    x, y, s, d = q.popleft()
    if s != 1 and dist[x][y][1] > d + 1:
        q.append((x, y, 1, d + 1))
        dist[x][y][1] = d + 1
        vis[x][y][1] = True
    if s != 2 and dist[x][y][2] > d + 1:
        q.append((x, y, 2, d + 1))
        dist[x][y][2] = d + 1
        vis[x][y][2] = True
    if s != 3 and dist[x][y][3] > d + 1:
        q.append((x, y, 3, d + 1))
        dist[x][y][3] = d + 1
        vis[x][y][3] = True
    for dx, dy in deltas:
        nx, ny = x + dx, y + dy
        if 0 <= nx < n and 0 <= ny < m and a[nx][ny] != '#' and not vis[nx][ny][s]:
            q.append((nx, ny, s, d + 1))
            vis[nx][ny][s] = True
            dist[nx][ny][s] = d + 1
ans = 10**9
for i in range(n):
    for j in range(m):
        if a[i][j] == '.' and (vis[i][j][1] or vis[i][j][2] or vis[i][j][3]):
            ans = min(ans, dist[i][j][1] + dist[i][j][2] + dist[i][j][3])
if ans == 10**9:
    ans = -1
print(ans)
