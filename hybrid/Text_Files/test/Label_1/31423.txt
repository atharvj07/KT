import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;
public class Main {
  // Finds the length of the longest strictly increasing
  // sequence of integers in an array such that no two
  // adjacent elements are coprime.
  public static int solve(int[] arr) {
    // Store the length of the longest strictly increasing
    // sequence ending at each index.
    int[] dp = new int[arr.length];

    // Initialize the dp array with 1s.
    Arrays.fill(dp, 1);

    // Iterate over the array.
    for (int i = 1; i < arr.length; i++) {
      // Iterate over the elements before i.
      for (int j = 0; j < i; j++) {
        // If arr[i] is greater than arr[j] and they
        // are not coprime, then update the length of
        // the longest strictly increasing sequence ending
        // at i.
        if (arr[i] > arr[j] && !gcd(arr[i], arr[j])) {
          dp[i] = Math.max(dp[i], dp[j] + 1);
        }
      }
    }

    // Return the maximum length of the longest strictly
    // increasing sequence ending at any index.
    return IntStream.of(dp).max().getAsInt();
  }

  // Finds the greatest common divisor of two integers
  // using the Euclidean algorithm.
  public static boolean gcd(int a, int b) {
    if (a == 0) {
      return b == 1;
    }
    if (b == 0) {
      return a == 1;
    }
    if (a > b) {
      return gcd(b, a % b);
    }
    return gcd(a, b % a);
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int[] arr = new int[n];
    for (int i = 0; i < n; i++) {
      arr[i] = scanner.nextInt();
    }
    System.out.println(solve(arr));
  }
}
