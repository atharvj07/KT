import java.util.*;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int k = sc.nextInt();
		String s = sc.next();
		char[] a = s.toCharArray();
		for (int i = 0; i < n && k > 0; i++) {
			if (a[i] > '0') {
				a[i] = '0';
				k--;
			}
		}
		System.out.println(new String(a));
	}
}
