#include <iostream>
using namespace std;
int main() {
  long long x1, y1, x2, y2;
  cin >> x1 >> y1 >> x2 >> y2;
  long long x3, y3, x4, y4, x5, y5, x6, y6;
  cin >> x3 >> y3 >> x4 >> y4 >> x5 >> y5 >> x6 >> y6;
  bool flag = false;
  bool b1 = false, b2 = false;
  if (((x1 >= x3 && x1 <= x4) || (x1 <= x3 && x2 >= x3)) && ((y1 >= y3 && y1 <= y4) || (y1 <= y3 && y2 >= y3)))
    b1 = true;
  if (((x1 >= x5 && x1 <= x6) || (x1 <= x5 && x2 >= x5)) && ((y1 >= y5 && y1 <= y6) || (y1 <= y5 && y2 >= y5)))
    b2 = true;
  if (!b1 || !b2)
    flag = true;
  cout << (flag ? "YES" : "NO") << endl;
  return 0;
}
