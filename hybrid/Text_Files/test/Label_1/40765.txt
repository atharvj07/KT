n = int(input())
a = list(map(int, input().split()))
ans = 0
mn = [0] * n
mx = [0] * n
mn[0] = a[0]
mx[n-1] = a[n-1]
for i in range(1, n):
    mn[i] = min(mn[i-1], a[i])
for i in range(n-2, -1, -1):
    mx[i] = max(mx[i+1], a[i])
for i in range(n):
    if a[i] >= 0 or mn[i] > 0 or mx[i] < 0:
        ans += 1
print(ans)
