import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        long[] dp = new long[n];
        for (int i = 0; i < k; i++) {
            dp[i] = arr[i];
        }
        for (int i = k; i < n; i++) {
            dp[i] = arr[i] + dp[i - k];
        }
        long min = dp[n - 1];
        int ans = 1;
        for (int i = n - k; i >= 0; i--) {
            if (dp[i] < min) {
                min = dp[i];
                ans = i + 1;
            }
        }
        System.out.println(ans);
    }
}
