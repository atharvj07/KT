import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = cin.nextInt();
        }
        if (n == 1 || a[0] == 0 || a[n - 1] == 0) {
            System.out.println("BitLGM");
        } else {
            System.out.println("BitAryo");
        }
    }
}
