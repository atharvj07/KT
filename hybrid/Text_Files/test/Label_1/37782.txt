import sys
sys.setrecursionlimit(10**9)
n,k = map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
a.sort(key=lambda x:x[1])
ans=0
t=0
for l,r,c in a:
    if l<=t:
        if t<=r:
            c-=(t-l)
            if c<=0:continue
            t=max(t,l+c-1)
            ans+=(t-l+k-1)//k*k
        else:continue
    else:
        t=l
        ans+=c
ans+=k-1
print(ans if t<=r else -1)
