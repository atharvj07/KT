n, k = map(int, input().split())
coins = list(map(int, input().split()))

dp = [False] * (k+1)
dp[0] = True

for coin in coins:
    for i in range(k, coin-1, -1):
        if dp[i-coin]:
            dp[i] = True

reachable = [i for i, x in enumerate(dp) if x]

print(len(reachable))
print(*reachable)
