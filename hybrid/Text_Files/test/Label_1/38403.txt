n, m = map(int,input().split())
g = [list(map(int,input().split())) for _ in range(n)]
for i in range(n):
 for j in range(m):
  if(g[i][j] == 0):
   flag = 1
   for k in range(n):
    if(g[k][j] != 0):
     flag = 0
   if(flag == 1):
    for k in range(m):
     g[i][k]+=1
    print(n-i)
    print(*[f"row {i+1}" for _ in range(n-i)])
    print(*[f"col {j+1}" for _ in range(m)])
    exit(0)

print(-1)
