import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] c = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                c[i][j] = sc.nextInt();
            }
        }
        int[] a = new int[3];
        int[] b = new int[3];
        for (int i = 0; i < 3; i++) {
            a[i] = c[0][i];
        }
        for (int j = 0; j < 3; j++) {
            b[j] = c[j][0];
        }
        boolean valid = true;
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                valid &= (c[i][j] == a[i] + b[j]);
            }
        }
        System.out.println(valid ? "Yes" : "No");
    }
}
