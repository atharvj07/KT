def fn(n,a,k,b):
    # i: index in a
    # j: index in b
    i=0
    j=0
    ans=[]
    while i<n and j<k:
      if a[i]==b[j]:
        i+=1
        j+=1
        continue

      if a[i]>b[j]:
        ans.append(str(i+1)+" R")
        a[i]+=a[i+1]
        del a[i+1]
        n-=1
      else:
        return "NO"

    if i<n:
      return "NO"

    return "YES\n"+"\n".join(ans)

n=int(input())
a=list(map(int,input().split()))
k=int(input())
b=list(map(int,input().split()))
print(fn(n,a,k,b))
