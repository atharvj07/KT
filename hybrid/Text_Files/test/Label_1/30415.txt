import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] a = new int[N];
        for (int i = 0; i < N; i++) a[i] = sc.nextInt();
        int sum = 0;
        for (int x : a) sum += x;
        int ave = sum / N;
        int diff = 1000, t = 0;
        for (int i = 0; i < N; i++) {
            if (diff > Math.abs(a[i] - ave)) {
                diff = Math.abs(a[i] - ave);
                t = i;
            }
        }
        System.out.println(t);
    }
}
