#include <bits/stdc++.h>
using namespace std;
int main() {
    int T;
    cin >> T;
    while (T--) {
        int n;
        cin >> n;
        int a[n + 5];
        for (int i = 1; i <= n; i++) cin >> a[i];
        int cnt0 = 0, cnt1 = 0;
        for (int i = 1; i <= n; i++) cnt0 += !a[i];
        for (int i = 1; i <= n; i++) cnt1 += a[i] != 0;
        if ((cnt0 == n - 1 and cnt1 == 1) or (cnt0 == n and cnt1 == 0))
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
