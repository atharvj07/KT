import java.util.Scanner;

public class Main {
    private static final int MOD = 1000000007;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String[] names = new String[n];
        for (int i = 0; i < n; i++) {
            names[i] = sc.next();
        }
        long[][] dp = new long[n][m + 1];
        dp[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                // Don't swap
                dp[i][j + 1] = (dp[i][j + 1] + dp[i][j]) % MOD;
                // Swap with the next name
                if (i + 1 < n) {
                    for (int k = 1; k <= Math.min(m - j, m); k++) {
                        if (names[i].substring(j, j + k).equals(names[i + 1].substring(0, k))) {
                            dp[i][j + 1] = (dp[i][j + 1] + dp[i + 1][k]) % MOD;
                        }
                    }
                }
            }
        }
        System.out.println(dp[0][m]);
    }
}
