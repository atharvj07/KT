import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        boolean[][] can = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (Math.abs(i - j) == d[i]) {
                    can[i][j] = true;
                }
            }
        }
        for (int k = 0; k < n; k++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    can[i][j] |= can[i][k] && can[k][j];
                }
            }
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (!can[i][p[i] - 1]) {
                ok = false;
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
