n = int(input())
a = list(map(int, input().split()))
inc = [0] * n
dec = [0] * n
if a[0] >= a[1]:
    inc[0] = a[0] - a[1] + 1
for i in range(1, n - 1):
    if a[i] >= a[i + 1]:
        inc[i] = a[i] - a[i + 1] + 1
    if a[i] <= a[i - 1]:
        dec[i] = a[i - 1] - a[i] + 1
if a[n - 1] <= a[n - 2]:
    dec[n - 1] = a[n - 2] - a[n - 1] + 1
print(max(max(inc), max(dec)))
