#include <iostream>

int main() {
    std::string s;
    std::cin >> s;
    int k;
    std::cin >> k;
    int ans = 0;
    for (int i = 0; i < s.size(); i++) {
        int j = i;
        while (j + k < s.size() && s[j] == s[j + k]) { j += k; }
        ans = std::max(ans, j - i);
    }
    std::cout << ans << std::endl;
    return 0;
}
