import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int k = 0;
        int t = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        for (int i = 0; i < n; i++) {
            if (sum <= b[i]) {
                k++;
                t += a[i];
                sum -= a[i];
            } else {
                t += b[i] - a[i];
                sum -= b[i] - a[i];
            }
        }
        System.out.println(k + " " + t);
    }
}
