#include<bits/stdc++.h>
using namespace std;
int a[105],b[105];
int main(){
    int n,k;cin>>n>>k;
    for(int i=1;i<=n;i++)cin>>a[i];
    for(int i=1;i<=k;i++)cin>>b[i];
    sort(b+1,b+k+1);
    bool f=0;
    for(int i=1;i<=n;i++){
        if(a[i]==0){
            if(i!=1&&b[1]<=a[i-1]){f=1;break;}
            if(b[k]>=a[i+1]){f=1;break;}
            for(int j=2;j<k;j++){
                if(b[j]<=a[i-1]||b[j]>=a[i+1]){f=1;break;}
            }
        }
    }
    if(f)cout<<"No";
    else cout<<"Yes";
    return 0;
}
