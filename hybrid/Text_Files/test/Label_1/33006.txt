import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int N = scanner.nextInt();
        int[] a = new int[N];
        double sum = 0;
        for (int i = 0; i < N; i++) {
            a[i] = scanner.nextInt();
            sum += a[i];
        }
        double avg = sum / N;
        double minDist = 1000000000;
        int t = -1;
        for (int i = 0; i < N; i++) {
            double dist = Math.abs(a[i] - avg);
            if (dist < minDist) {
                minDist = dist;
                t = i;
            }
        }
        System.out.println(t);
    }
}
