#include<iostream>
#include<string>
using namespace std;
int main() {
  int n, t;
  cin >> n >> t;
  string s1, s2;
  cin >> s1 >> s2;
  string s3 = s1;
  for (int i = 0; i < n; i++) {
    if (t > 0 && s1[i] != s2[i]) {
      if (s1[i] == s3[i]) {
        s3[i] = s2[i];
      } else {
        s3[i] = s1[i];
      }
      t--;
    }
  }
  if (t == 0) {
    cout << s3 << endl;
  } else {
    cout << -1 << endl;
  }
  return 0;
}
