n, k = [int(x) for x in input().split()]
a = [int(x) for x in input().split()]
ans = []
for i in range(n - 1):
    diff = a[i] + 1 - a[i + 1]
    if diff != k:
        if diff < k:
            ans.append("+ {} {}".format(i + 1, k - diff))
        else:
            ans.append("- {} {}".format(i + 2, diff - k))
print(len(ans))
for x in ans:
    print(x)
