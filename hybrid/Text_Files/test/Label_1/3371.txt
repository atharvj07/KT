#include <iostream>
#include <string>

using namespace std;

int main() {
  string s;
  cin >> s;
  int cnt[26];
  fill(cnt, cnt + 26, 0);

  for (char c : s) {
    if (c != '?') cnt[c - 'A']++;
  }

  bool ok = true;
  int qCnt = 0;

  for (int i = 0; i < 26; i++) {
    if (cnt[i] == 0) {
      if (qCnt > 0) {
        ok = false;
        break;
      }

      qCnt++;
    }
  }

  if (!ok) {
    cout << -1 << endl;
    return 0;
  }

  string ans;
  int aCnt = 0;
  int bCnt = 0;

  for (char c : s) {
    if (c == '?') {
      if (aCnt < 26) {
        ans += char('A' + (aCnt % 26));
        cnt[aCnt++ % 26]++;
      } else {
        ans += char('A' + (bCnt % 26));
        cnt[bCnt++ % 26]++;
      }
    } else {
      ans += c;
    }
  }

  cout << ans << endl;
}
