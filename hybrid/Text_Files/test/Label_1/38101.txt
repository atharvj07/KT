def find_path(h, n):
    path = 0
    total_nodes = 2**h - 1
    left = 1
    right = 2**h
    while h > 0:
        mid = (left + right) // 2
        if n > mid:
            if n <= right:
                path += 2**(h-1) + (2**(h-1) - 1)
                left = mid + 1
        else:
            if n >= left:
                path += 1
                right = mid
        h -= 1
    return path

h, n = map(int, input().split())
print(find_path(h, n))
