#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long int lli;

const lli MXN = 2e5+5;
vector<lli> a,b,psuma,psumb;

int main(){
    lli n,m,q,k;
    cin>>n>>m>>q;
    a.resize(n+1);
    b.resize(m+1);
    psuma.resize(MXN);
    psumb.resize(MXN);
    for(lli i=1; i<=n; i++){
        cin>>a[i];
        psuma[i] = psuma[i-1] + a[i];
    }
    for(lli i=1; i<=m; i++){
        cin>>b[i];
        psumb[i] = psumb[i-1] + b[i];
    }
    while(q--){
        cin>>k;
        lli ans = 0;
        ans = max(ans,psuma[n]);
        for(lli i=1,j=m; i<=n; i++){
            int l = i, r = m, xl = -1;
            while(l<=r){
                lli mid = (l+r)/2;
                if(a[i]+k>=b[mid]) xl = mid,l = mid+1;
                else r = mid-1;
            }
            if(xl>=0) ans = max(ans,psuma[i-1] + psumb[xl]);
        }
        cout<<ans<<endl;
    }
    return 0;
}
