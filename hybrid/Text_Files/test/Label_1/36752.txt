t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().strip().split())
    a = list(map(int, input().strip().split()))
    w = list(map(int, input().strip().split()))

    a.sort(reverse=True)
    w.sort()

    happiness = 0
    i = 0
    for wi in w:
        if wi == 1:
            happiness += 2 * a[i]
            i += 1
        else:
            happiness += a[i] + a[i + wi - 1]
            i += wi

    print(happiness)
