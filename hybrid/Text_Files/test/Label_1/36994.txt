import math
n = int(input())
cakes = []
for i in range(n):
    r, h = map(int, input().split())
    cakes.append((r, h))
cakes.sort(key=lambda x: x[0] * x[1], reverse=True)
dp = [0] * (n + 1)
dp[0] = 0
for i in range(1, n + 1):
    r, h = cakes[i - 1]
    dp[i] = math.pi * r ** 2 * h
    for j in range(i):
        if r > cakes[j][0]:
            dp[i] = max(dp[i], dp[j] + math.pi * r ** 2 * h)
print(max(dp))
