import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.Queue;

public class GeoRepaintings {
    private static final int TRANSPARENT_COLOR = 0;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] dimensions = reader.readLine().split(" ");
        int n = Integer.parseInt(dimensions[0]);
        int m = Integer.parseInt(dimensions[1]);

        int[][] geoPanels = new int[n][m];
        for (int i = 0; i < n; i++) {
            geoPanels[i] = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        }

        int[][] geoSymbols = new int[n][m];
        for (int i = 0; i < n; i++) {
            geoSymbols[i] = Arrays.stream(reader.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        }

        String[] coordinates = reader.readLine().split(" ");
        int x = Integer.parseInt(coordinates[0]) - 1;
        int y = Integer.parseInt(coordinates[1]) - 1;

        int repaintings = solve(geoPanels, geoSymbols, x, y);
        System.out.println(repaintings);
    }

    private static int solve(int[][] geoPanels, int[][] geoSymbols, int startX, int startY) {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(geoSymbols[startX][startY]);
        geoSymbols[startX][startY] = -1; // Mark as removed

        int repaintings = 0;
        while (!queue.isEmpty()) {
            int currentSymbol = queue.poll();

            if (currentSymbol != TRANSPARENT_COLOR) {
                for (int i = 1; ; i++) {
                    for (int j = -i; j <= i; j++) {
                        for (int k = -i; k <= i; k++) {
                            if (Math.abs(j) + Math.abs(k) == i) {
                                int nextX = startX + j;
                                int nextY = startY + k;
                                if (isValid(geoPanels, nextX, nextY) && geoPanels[nextX][nextY] == currentSymbol) {
                                    repaintings++;
                                    geoPanels[nextX][nextY] = TRANSPARENT_COLOR;

                                    if (geoSymbols[nextX][nextY] != -1) {
                                        queue.add(geoSymbols[nextX][nextY]);
                                        geoSymbols[nextX][nextY] = -1; // Mark as removed
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        return repaintings;
    }

    private static boolean isValid(int[][] grid, int x, int y) {
        return x >= 0 && x < grid.length && y >= 0 && y < grid[0].length;
    }
}
