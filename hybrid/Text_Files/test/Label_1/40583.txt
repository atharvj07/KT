n, a, b = map(int, input().split())
t = list(map(int, input().split()))
one, two = 0, 0
ans = 0
for x in t:
    if x == 1:
        if one == 0:
            if two > 0:
                two -= 1
            else:
                if b > 0:
                    b -= 1
                else:
                    ans += 1
        else:
            one -= 1
    else:
        if two == 0:
            if b > 0:
                b -= 1
            else:
                ans += 2
        else:
            two -= 1
print(ans)
