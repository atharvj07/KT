import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long k = sc.nextLong();
    long[] a = new long[n + 1];
    long max = 0;
    for (int i = 1; i <= n; i++) {
      a[i] = sc.nextLong();
      max = Math.max(max, a[i]);
    }
    sc.close();

    long d = (k + max - 1) / max;
    System.out.println(d);
  }
}
