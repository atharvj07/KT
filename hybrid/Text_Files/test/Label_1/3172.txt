#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
int32_t main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    long long ans = 0;
    cin >> n;
    vector<long long> a(n);
    for (auto& x : a) {
        cin >> x;
        x %= MOD;
        if (abs(x) % 2 != 0) {
            x += MOD / 2;
        }
        x %= MOD;
        ans += min(x, MOD - x);
    }
    cout << ans << "\n";
}
