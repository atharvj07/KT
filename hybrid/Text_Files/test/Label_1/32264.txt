import java.util.Scanner;

class Main {
    private static final int MOD = 1_000_000_007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int m = in.nextInt();
        int ans = 0;
        for (int i = 2; i <= m; i++) {
            int l = i % 10, r = i / 10;
            if (l == 4 || l == 7 || r == 4 || r == 7) {
                ans += m - i + 1;
                ans %= MOD;
            }
        }
        System.out.println(ans);
    }
}
