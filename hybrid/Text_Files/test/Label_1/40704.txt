s = input()
n = len(s)
ans = 0
cur = ''
t =[0]*n
for i in range(n):
    if s[i] == '0':
        t[i] = -1
    else:
        t[i] = 1
for i in range(n):
    if i == 0 or t[i] != t[i-1]:
        ans += 1
print(ans)
ans = []
cur = ''
for i in range(n):
    cur += s[i]
    if i == n-1 or t[i] != t[i+1]:
        ans.append(cur)
        cur = ''
print(*ans)
