#include <bits/stdc++.h>
#define ll long long
using namespace std;

bool solve(ll a, ll b, ll m) {
    if (m == 1) {
        if ((b - a) % 2 == 0 && (b - a) / 2 <= 50) return 1;
        else return 0;
    }
    if ((b - a) % (m + 1) == 0 && (b - a) / (m + 1) <= 50) return 1;
    else return 0;
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    ll q;
    cin >> q;
    while (q--) {
        ll a, b, m;
        cin >> a >> b >> m;
        if (!solve(a, b, m)) {
            cout << -1 << '\n';
            continue;
        }
        ll num = (b - a) / (m + 1);
        if (m == 1) num = (b - a) / 2;
        cout << num + 1 << ' ';
        for (ll i = 0; i < num; i++) {
            cout << a + (m + 1) * i << ' ';
        }
        cout << b << '\n';
    }
    return 0;
}
