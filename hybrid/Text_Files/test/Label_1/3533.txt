#include <bits/stdc++.h>
using namespace std;
int sum[200005], n, m;
int main() {
  cin >> n >> m;
  int l = 0, r = 0, now = 0;
  for (int i = 1; i <= n; i++) cin >> sum[i], sum[i] += sum[i - 1], r += sum[i];
  while (m--) {
    int x;
    cin >> x;
    int id = upper_bound(sum + 1, sum + n + 1, x) - sum;
    now = max(now, x - sum[id - 1]);
    cout << id << ' ' << now << endl;
    now++;
  }
}
