import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int[] b = new int[m];
        for (int i = 0; i < m; i++) {
            b[i] = in.nextInt();
        }
        Arrays.sort(a);
        Arrays.sort(b);
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            if (in.nextInt() == 1) {
                a[in.nextInt() - 1] = in.nextInt();
                Arrays.sort(a);
            } else {
                b[in.nextInt() - 1] = in.nextInt();
                Arrays.sort(b);
            }
            int j = m - 1;
            int ans = -1;
            for (int i1 = n - 1; i1 >= 0; i1--) {
                while (j >= 0 && a[i1] > b[j]) {
                    j--;
                }
                if (j >= 0) {
                    ans = a[i1];
                    break;
                }
            }
            System.out.println(ans);
        }
    }
}
