#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> b(n);
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<int> a(n);
    int l = 0, r = n - 1, i = 0;
    while (l <= r) {
      if (l == r) {
        a[i] = b[l];
        break;
      } else if (b[l] < b[r]) {
        a[i] = b[l];
        l++;
        i++;
      } else {
        a[i] = b[r];
        r--;
        i++;
      }
    }
    for (int i = 0; i < n; i++) cout << a[i] << ' ';
    cout << '\n';
  }
  return 0;
}
