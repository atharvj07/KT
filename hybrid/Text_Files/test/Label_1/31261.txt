import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        final Scanner scanner = new Scanner(System.in);
        long n = scanner.nextLong();
        long m = scanner.nextLong();
        Map<Long, Long> cnt = new HashMap<>();
        cnt.put(0L, 1L);
        long prefix = 0;
        for (long i = 0; i < n; i++) {
            prefix += scanner.nextLong();
            prefix %= m;
            cnt.put(prefix, cnt.getOrDefault(prefix, 0L) + 1L);
        }

        long ans = 0;
        for (Map.Entry<Long, Long> entry : cnt.entrySet()) {
            long key = entry.getKey();
            long val = entry.getValue();
            ans += val * (val - 1) / 2;
            if (key == 0L) {
                ans += val;
            } else if (m - key == key) {
                ans += val * (val - 1) / 2;
            } else {
                ans += val * cnt.getOrDefault(m - key, 0L);
            }
        }
        System.out.println(ans == 0L ? "NO" : "YES");

        scanner.close();
    }
}
