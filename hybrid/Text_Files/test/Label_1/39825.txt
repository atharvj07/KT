for _ in range(int(input())):
    n = int(input())
    a = list(map(int, input().split()))
    ans = 0
    s = 0
    for x in a:
        s += x % 3
    ans += s // 3
    if s % 3 == 0:
        ans += 1
    elif s % 3 == 1:
        ans += max(0, s // 3 - 1)
    elif s % 3 == 2:
        ans += max(0, s // 3 - 2)
    print(ans)
