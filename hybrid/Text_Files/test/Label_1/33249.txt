import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int l = sc.nextInt();
        boolean[] panel = new boolean[n + 1];
        for (int i = 0; i < k; i++)
            panel[sc.nextInt()] = true;

        int[] a = new int[l];
        for (int i = 0; i < l; i++)
            a[i] = sc.nextInt();

        boolean flag = true;
        int ans = 0;
        for (int y : a) {
            boolean flip = false;
            for (int j = 0; j < y; j++) {
                flip ^= panel[j + 1];
            }
            if (!flip)
                continue;
            if (flag)
                ans++;
            flag = flip;
        }
        System.out.println(flag ? ans : -1);
    }
}
