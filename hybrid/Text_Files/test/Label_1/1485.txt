#include <bits/stdc++.h>
using namespace std;
int a[105],d[105];
int main() {
    int n, m;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> m;
    while (m--) {
        int x, y;
        cin >> x >> y;
        d[x - 1] += a[x] - y;
        d[x + 1] += y - 1;
        a[x] = 0;
    }
    for (int i = 1; i <= n; i++) a[i] += d[i - 1];
    for (int i = 1; i <= n; i++) cout << a[i] << '\n';
}
