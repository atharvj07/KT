import java.io.*;
import java.util.*;

class Main {
  static int n, m, k, ans;
  static char[][][] map;
  static boolean[][][] vis;

  static void dfs(int x, int y, int z) {
    if (vis[x][y][z]) return;
    vis[x][y][z] = true;
    if (x > 0 && map[x - 1][y][z] == '1') dfs(x - 1, y, z);
    if (y > 0 && map[x][y - 1][z] == '1') dfs(x, y - 1, z);
    if (z > 0 && map[x][y][z - 1] == '1') dfs(x, y, z - 1);
  }

  static boolean can(int x, int y, int z) {
    dfs(x, y, z);
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        for (int l = 0; l < k; l++) {
          if (vis[i][j][l] && map[i][j][l] == '1') {
            if (i < n - 1 && map[i + 1][j][l] == '1') return true;
            if (j < m - 1 && map[i][j + 1][l] == '1') return true;
            if (l < k - 1 && map[i][j][l + 1] == '1') return true;
          }
        }
      }
    }
    return false;
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    m = sc.nextInt();
    k = sc.nextInt();
    map = new char[n][m][k];
    vis = new boolean[n][m][k];
    for (int i = 0; i < n; i++) {
      sc.nextLine();
      for (int j = 0; j < m; j++) {
        String str = sc.nextLine();
        for (int l = 0; l < k; l++) {
          map[i][j][l] = str.charAt(l);
        }
      }
    }
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        for (int l = 0; l < k; l++) {
          if (map[i][j][l] == '1') {
            dfs(i, j, l);
            ans += can(i, j, l) ? 1 : 0;
          }
        }
      }
    }
    System.out.println(ans);
  }
}
