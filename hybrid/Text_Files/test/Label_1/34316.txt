import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextLong();
        long ans = 0;
        for (int i = 0; i < n; i++) {
            ans += Math.max(0, a[i] - Math.ceil(-(a[i] + 1) / 2));
        }
        System.out.println(ans);
    }
}
