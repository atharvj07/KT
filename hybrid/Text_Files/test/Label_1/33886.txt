import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in .nextLong();
        long b = in .nextLong();
        long x1 = in .nextLong();
        long y1 = in .nextLong();
        long x2 = in .nextLong();
        long y2 = in .nextLong();
        in .close();
        long dx = Math.abs(x2 - x1);
        long dy = Math.abs(y2 - y1);
        long badl = Math.max(0, Math.min(dx, dy) - 1);
        long badh = (dx + dy - (dx % 2 + dy % 2)) / 2;
        if(a == b){
            System.out.println(badh);
        }else if((x1 + y1 + x2 + y2) % 2 == 0) {
            System.out.println(badl);
        }else {
            long ans = Math.min(2 * badl + badh, badh + 2 * badl + 1);
            System.out.println(ans);
        }
    }
}
