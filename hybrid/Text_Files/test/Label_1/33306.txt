import java.util.Scanner;

public class Main {
    static int mod = 1000000007;
    static int[][][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();
        String t = sc.nextLine();
        dp = new int[s.length() + 1][t.length() + 1][s.length() + 1];
        System.out.println(solve(0, 0, 0, s, t));
    }

    public static int solve(int i, int j, int l, String s, String t) {
        if (i == s.length()) {
            return j == t.length() ? 1 : 0;
        }
        if (dp[i][j][l] != 0) {
            return dp[i][j][l];
        }
        if (s.charAt(i) == t.charAt(j)) {
            dp[i][j][l] = (solve(i + 1, j + 1, l + 1, s, t) + solve(i + 1, j, l, s, t)) % mod;
        } else {
            if (l > 0) {
                dp[i][j][l] = solve(i + 1, j, l - 1, s, t);
            } else {
                dp[i][j][l] = solve(i + 1, j, l, s, t);
            }
        }
        return dp[i][j][l];
    }
}
