#include <cstdio>
#include <algorithm>
#include <vector>

using namespace std;

int t[100005];

int main() {
  int n, p;
  scanf("%d%d", &n, &p);
  for (int i = 0; i < n; i++) {
    scanf("%d", &t[i]);
  }
  vector<pair<int, int> > b;
  for (int i = 0; i < n; i++) {
    b.push_back(make_pair(t[i], i + 1));
  }
  sort(b.begin(), b.end());
  int ans[100005];
  int occ = 1, cur = b[0].first;
  for (int i = 0; i < n; i++) {
    ans[b[i].second] = cur;
    while (occ < b[i].second && i + 1 < n && b[i].first == b[i + 1].first) {
      i++;
      occ++;
    }
    cur += p;
  }
  for (int i = 0; i < n; i++) {
    printf("%d ", ans[i + 1]);
  }
  printf("\n");
}
