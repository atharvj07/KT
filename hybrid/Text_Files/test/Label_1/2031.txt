#include <iostream>
#include <map>

using namespace std;

int main() {
    int n, a, t, x;
    map<pair<int, int>, int> ms;
    cin >> n;
    while (n--) {
        cin >> a >> t >> x;
        if (a == 1)
            ms[{t, x}]++;
        if (a == 2) {
            if (ms.count({t, x})) {
                ms[{t, x}]--;
                if (ms[{t, x}] == 0)
                    ms.erase({t, x});
            }
        }
        if (a == 3)
            cout << (ms.count({t, x}) ? ms[{t, x}] : 0) << endl;
    }
    return 0;
}
