import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long m = sc.nextLong();
        long x = 0, cnt = 0;
        for (long i = 1; i <= m; i++) {
            if (x + i * i * i > m)
                break;
            x += i * i * i;
            cnt++;
        }
        System.out.println(cnt + " " + x);
    }
}
