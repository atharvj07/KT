import sys

a=[0]*10
for s in input():
    if s>='0'and s<='9':a[int(s)]=0
    elif s=='L':
        for i in range(10):
            if not a[i]:
                a[i]=1;break
    else:
        for i in range(9,-1,-1):
            if not a[i]:
                a[i]=1;break
print(''.join(map(str,a)))
