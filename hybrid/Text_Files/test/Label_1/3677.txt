#include <bits/stdc++.h>
#define MX 200001
using namespace std;
int x[MX], mx = INT_MIN;
int main() {
  int n, ans = 0;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> x[i];
    mx = max(abs(x[i]), mx);
  }
  for (int i = 0; i < n; i++) {
    if (x[i] != 0) ans += (abs(x[i]) + mx - 1) / mx;
  }
  cout << ans << "\n";
}
