#include <iostream>
#include <vector>
#include <limits>

using namespace std;

int main() {
  int n, u;
  cin >> n >> u;

  vector<long long> energies(n);
  for (int i = 0; i < n; i++) {
    cin >> energies[i];
  }

  double max_efficiency = -1;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      for (int k = j + 1; k < n; k++) {
        if (energies[k] - energies[i] <= u) {
          double efficiency = (double)(energies[k] - energies[j]) / (energies[k] - energies[i]);
          max_efficiency = max(max_efficiency, efficiency);
        }
      }
    }
  }

  if (max_efficiency == -1) {
    cout << -1 << endl;
  } else {
    cout << max_efficiency << endl;
  }

  return 0;
}
