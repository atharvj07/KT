import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int m = input.nextInt();
        int n = input.nextInt();

        if ((n & 1) == 1 && m > n || (m + n) % 2 == 1) {
            System.out.println(-1);
            return;
        }

        int steps = (m + n) / n;
        if (steps % 2 == 0) {
            steps++;
        }

        System.out.println(steps / 2);
    }
}
