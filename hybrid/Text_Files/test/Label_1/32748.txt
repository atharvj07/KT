import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.StringTokenizer;

class Node {
    int index;
    int apples;
    List<Node> children;

    Node(int index, int apples) {
        this.index = index;
        this.apples = apples;
        this.children = new ArrayList<>();
    }
}

class Main {
    static int n;
    static boolean[] visited;
    static List<Node> nodes;
    static long[] subtreeWeights;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        StringTokenizer st = new StringTokenizer(br.readLine());
        nodes = new ArrayList<>(n);
        subtreeWeights = new long[n];
        for (int i = 0; i < n; i++) {
            nodes.add(new Node(i, Integer.parseInt(st.nextToken())));
        }
        for (int i = 0; i < n - 1; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken()) - 1;
            int b = Integer.parseInt(st.nextToken()) - 1;
            nodes.get(a).children.add(nodes.get(b));
            nodes.get(b).children.add(nodes.get(a));
        }
        visited = new boolean[n];
        dfs(0, -1);
        long total = 0;
        for (long weight : subtreeWeights) {
            total += weight;
        }
        long minWeight = Long.MAX_VALUE;
        for (long weight : subtreeWeights) {
            minWeight = Math.min(minWeight, weight);
        }
        System.out.println(total - n * minWeight);
    }

    private static long dfs(int node, int parent) {
        if (visited[node]) {
            return 0;
        }
        visited[node] = true;
        long sum = nodes.get(node).apples;
        for (Node child : nodes.get(node).children) {
            if (child.index != parent) {
                sum += dfs(child.index, node);
            }
        }
        subtreeWeights[node] = sum;
        return sum;
    }
}
