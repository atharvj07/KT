def check_obfuscation(s):
    used = set()
    i = 0
    a = 'a'
    while i < len(s):
        if s[i] not in used:
            used.add(s[i])
            j = i + 1
            while j < len(s):
                if s[j] == s[i]:
                    s = s[:j] + a + s[j + 1:]
                j += 1
            a = chr(ord(a) + 1)
        i += 1
    return s == a * len(used)

s = input()
print("YES" if check_obfuscation(s) else "NO")
