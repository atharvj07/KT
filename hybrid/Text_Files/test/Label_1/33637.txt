import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String[] names = new String[n];
        for (int i = 0; i < n; i++) {
            names[i] = sc.nextLine();
        }
        Map<String, int[]> map = new HashMap<>();
        for (String name : names) {
            map.put(name, new int[]{0, 0, 0});
        }
        for (int i = 0; i < n * (n - 1) / 2; i++) {
            String[] line = sc.nextLine().split(" ");
            String[] teams = line[0].split("-");
            int[] goals = Arrays.stream(line[1].split(":")).mapToInt(Integer::parseInt).toArray();
            if (goals[0] > goals[1]) {
                map.get(teams[0])[0] += 3;
                map.get(teams[1])[0] += 0;
                map.get(teams[0])[1] += goals[0] - goals[1];
                map.get(teams[1])[1] += goals[1] - goals[0];
                map.get(teams[0])[2] += goals[0];
                map.get(teams[1])[2] += goals[1];
            } else if (goals[0] < goals[1]) {
                map.get(teams[0])[0] += 0;
                map.get(teams[1])[0] += 3;
                map.get(teams[0])[1] += goals[0] - goals[1];
                map.get(teams[1])[1] += goals[1] - goals[0];
                map.get(teams[0])[2] += goals[0];
                map.get(teams[1])[2] += goals[1];
            } else {
                map.get(teams[0])[0] += 1;
                map.get(teams[1])[0] += 1;
                map.get(teams[0])[1] += goals[0] - goals[1];
                map.get(teams[1])[1] += goals[1] - goals[0];
                map.get(teams[0])[2] += goals[0];
                map.get(teams[1])[2] += goals[1];
            }
        }
        map.entrySet()
                .stream()
                .sorted((a, b) -> {
                    int cmp = Integer.compare(b.getValue()[0], a.getValue()[0]);
                    if (cmp != 0) {
                        return cmp;
                    }
                    cmp = Integer.compare(b.getValue()[1], a.getValue()[1]);
                    if (cmp != 0) {
                        return cmp;
                    }
                    cmp = Integer.compare(b.getValue()[2], a.getValue()[2]);
                    if (cmp != 0) {
                        return cmp;
                    }
                    return a.getKey().compareTo(b.getKey());
                })
                .limit(n / 2)
                .map(Map.Entry::getKey)
                .forEach(System.out::println);
    }
}
