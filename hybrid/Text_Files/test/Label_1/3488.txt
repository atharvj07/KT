#include <bits/stdc++.h>
using namespace std;
const int N = 105;

int n, m, a[N], dp[N][N], b[N];
bool ok[N][N];

int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; ++i)
        cin >> a[i];
    for (int i = 1; i <= m; ++i)
    {
        int l, r;
        cin >> l >> r;
        b[++b[0]] = l, b[++b[0]] = r;
        ok[l][r] = true;
    }
    for (int l = b[0]; l >= 1; --l)
    {
        for (int r = 1; r <= b[0]; ++r)
        {
            if (l > r)
                continue ;
            for (int i = b[l]; i <= b[l + 1]; ++i)
            {
                dp[l][r] = max({dp[l][r], dp[l][i] + dp[i + 1][r] + a[i], dp[l][i - 1] + dp[i][r] + a[i]});
            }
        }
    }
    cout << dp[1][b[0]] << "\n";
}
