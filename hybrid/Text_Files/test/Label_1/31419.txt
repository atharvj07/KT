import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int m = s.nextInt();
        int k = s.nextInt();

        System.out.println(power(k, (long) n * m) % 1000000007);
    }

    private static long power(int x, long y) {
        if (y == 0)
            return 1;
        if (y == 1)
            return x % 1000000007;
        long res = power(x, y / 2);
        res = (res * res) % 1000000007;
        if (y % 2 == 1)
            res = (res * x) % 1000000007;
        return res;
    }
}
