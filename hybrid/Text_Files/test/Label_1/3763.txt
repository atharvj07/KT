#include<bits/stdc++.h>
using namespace std;
const int mod=998244353;
typedef long long ll;
ll arr[100009], dp[100009][3];
ll f(int idx, int last, int type) {
    if(idx>arr[0]) return 1;
    ll &ans=dp[idx][type];
    if(~ans) return ans;
    ans=0;
    if(type==0) {
        for(ll i=last; i<=200; i++) {
            if(i<arr[idx] or (idx>=2 and i<arr[idx-1])) continue;
            ans=(ans+f(idx+1, i, 0))%mod;
        }
    }
    else if(type==1) {
        for(ll i=last; i<=arr[idx]; i++) {
            if(i<arr[idx] or (idx>=2 and i<arr[idx-1])) continue;
            ans=(ans+f(idx+1, i, 0))%mod;
        }
    }
    else {
        ans=(ans+(f(idx+1, arr[idx], 1)+f(idx+1, arr[idx], 2))%mod)%mod;
    }
    return ans;
}
int main() {
    ll n;
    cin>>n;
    for(int i=1; i<=n; i++) {
        cin>>arr[i];
        if(arr[i]==-1) arr[i]=201;
    }
    memset(dp, -1, sizeof(dp));
    cout<<f(1, 0, 0)%mod<<endl;
    return 0;
}
