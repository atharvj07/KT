import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long k = sc.nextLong();
        long a = sc.nextLong();
        long b = sc.nextLong();
        long r = 0;
        for (long n = 1;; n++) {
            if (n % 2 == 1) r += a;
            else r -= b;
            if (r >= k) {
                System.out.println(n);
                return;
            }
        }
    }
}
