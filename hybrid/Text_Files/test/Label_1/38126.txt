def largest_pile(s):
    max_pile = 1
    for char in set(s):
        i, temp_max, count = 0, 1, 0
        while i < len(s):
            if s[i] == char:
                count += 1
                j = i + 1
                while j < len(s):
                    if s[j] == char:
                        count += 1
                        temp_max = max(temp_max, min(count, (j - i)))
                    j += 1
                count = 0
            i += 1
        max_pile = max(max_pile, temp_max)
    return max_pile

# Read input
s = input().strip()
# Print output
print(largest_pile(s))
