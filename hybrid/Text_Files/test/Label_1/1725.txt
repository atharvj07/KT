#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    int x;
    cin >> s >> x;
    vector<bool> res(s.size(), false);
    bool ok = true;
    for (int i = 0; i < s.size(); i++) {
      if (s[i] == '1') {
        if (i - x >= 0 && res[i - x]) res[i] = true;
        else if (i + x < s.size() && res[i + x]) res[i] = true;
        else if (i - x >= 0 && s[i - x] == '1') res[i] = true;
        else if (i + x < s.size() && s[i + x] == '1') res[i] = true;
        else ok = false;
      }
    }
    if (!ok) cout << -1 << endl;
    else {
      for (int i = 0; i < s.size(); i++) {
        if (i - x >= 0 && res[i - x]) cout << 1;
        else if (i + x < s.size() && res[i + x]) cout << 1;
        else cout << 0;
      }
      cout << endl;
    }
  }
  return 0;
}
