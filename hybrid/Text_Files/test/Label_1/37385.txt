import math
a, b, c = map(int, input().split())
x1, y1, x2, y2 = map(int, input().split())

dist = math.hypot(x2 - x1, y2 - y1)
if (a * x1 + b * y1 + c) * (a * x2 + b * y2 + c) > 0:
    dist = min(dist, abs(a * x1 + b * y1 + c) / math.sqrt(a * a + b * b))
    dist = min(dist, abs(a * x2 + b * y2 + c) / math.sqrt(a * a + b * b))
print(dist)
