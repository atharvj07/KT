import sys

def solve():
    t = int(input().strip())
    for _ in range(t):
        n = int(input().strip())
        users = set(input().strip().split())
        m = int(input().strip())
        messages = []
        for _ in range(m):
            message = input().strip()
            if message[0] != '?':
                user, text = message.split(':', 1)
                assert user in users
                messages.append((user, text))
            else:
                text = message[1:]
                messages.append((None, text))
        if solve_chat(users, messages):
            for user, text in messages:
                print('{}:{}'.format(user if user else '?', text))
        else:
            print('Impossible')

def solve_chat(users, messages):
    n = len(users)
    users = list(users)
    for i in range(len(messages)):
        user, text = messages[i]
        if user:
            continue
        if i == 0:
            messages[i] = (users[0], text)
            continue
        if i == len(messages) - 1:
            messages[i] = (users[n-1], text)
            continue
        for j in range(n):
            if text.count(users[j]) == 0:
                messages[i] = (users[j], text)
                break
        if messages[i][0] is None:
            return False
    return True

solve()
