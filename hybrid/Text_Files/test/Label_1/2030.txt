#include <algorithm>
#include <cmath>
#include <iostream>

using ll = long long;

ll lcm(ll a, ll b) {
    return (a / std::__gcd(a, b)) * b;
}

int main() {
    ll n, a, b, p, q;
    std::cin >> n >> a >> b >> p >> q;
    std::cout << (lcm(a, b) / a * p + lcm(a, b) / b * q) * (n / lcm(a, b)) +
                     (n % lcm(a, b) / a) * p + (n % lcm(a, b) / b) * q
              << '\n';
}
