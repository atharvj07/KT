#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int n, m, q;
  cin >> n >> m >> q;
  vector<ll> a(n), b(m);
  for (auto &x : a) cin >> x;
  for (auto &x : b) cin >> x;
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  while (q--) {
    ll k;
    cin >> k;
    ll ans = 0;
    ll j = m - 1;
    for (int i = 0; i < n; i++) {
      while (j >= 0 && b[j] > a[i] + k) j--;
      if (j < 0) break;
      ans += b[j];
      j--;
    }
    cout << ans << '\n';
  }
  return 0;
}
