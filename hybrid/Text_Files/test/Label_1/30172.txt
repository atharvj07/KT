import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    long k = sc.nextLong();
    int pos = (int) Math.ceil(Math.log10(k) / Math.log10(7));
    if (7 % k != 0) {
      System.out.println(-1);
    } else {
      System.out.println(pos);
    }
  }
}
