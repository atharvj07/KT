#include <bits/stdc++.h>
#define ll long long
const int mod = 1e9 + 7;
using namespace std;
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n;
  cin >> n;
  vector<int> a(n), f(n + 1, 0), cnt(1 << n, 0);
  for (auto &x : a) cin >> x;
  f[1] = 1;
  for (int i = 0; i < n; i++) {
    int t = a[i];
    map<int, int> freq;
    for (int j = 2; j * j <= t; j++) {
      while (t % j == 0) {
        freq[j] += 1;
        t = t / j;
      }
    }
    if (t > 1) freq[t] = 1;
    for (auto &[x, y] : freq) {
      for (int p = 0; p <= i; p++)
        if (f[p]) {
          f[p + y] = 1;
          cnt[p + y] = (cnt[p + y] + cnt[p]) % mod;
        }
    }
    cnt[1] += 1;
  }
  int ans = 0;
  for (int i = 1; i <= n; i++) ans = (ans + f[i]) % mod;
  cout << ans << "\n";
  return 0;
}
