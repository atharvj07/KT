import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int m = Integer.parseInt(line1[1]);
        List<int[]> light = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            String[] line2 = br.readLine().split(" ");
            int k = Integer.parseInt(line2[0]);
            int[] switches = new int[k];
            for (int j = 0; j < k; j++) {
                switches[j] = Integer.parseInt(line2[j + 1]);
            }
            light.add(switches);
        }
        int[] p = new int[m];
        String[] line3 = br.readLine().split(" ");
        for (int i = 0; i < m; i++) {
            p[i] = Integer.parseInt(line3[i]);
        }
        int[][] dp = new int[1 << n][m + 1];
        dp[0][0] = 1;
        for (int i = 0; i < (1 << n); i++) {
            for (int j = 0; j < m; j++) {
                if (dp[i][j] == 0) continue;
                int[] switches = light.get(j);
                int ones = 0;
                for (int k = 0; k < switches.length; k++) {
                    if ((i & (1 << (switches[k] - 1))) != 0) ones++;
                }
                if (ones % 2 == p[j]) {
                    dp[i][j + 1] = dp[i][j];
                }
                if (ones % 2 != p[j]) {
                    dp[i | (1 << (switches[0] - 1))][j + 1] += dp[i][j];
                }
            }
        }
        System.out.println(dp[(1 << n) - 1][m]);
    }
}
