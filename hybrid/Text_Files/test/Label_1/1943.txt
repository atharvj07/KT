#include <cstdio>
#include <algorithm>

typedef long long i64;

const int MAXN = 5005;

i64 a[MAXN], f[MAXN][MAXN], dp[MAXN];
int n, k, x;

int main() {
    scanf("%d%d%d", &n, &k, &x);
    for (int i = 1; i <= n; ++i) scanf("%I64d", &a[i]);
    for (int i = n; i >= 1; --i) {
        for (int j = 1; j <= n - i + 1; ++j) {
            f[i][j] = std::max({f[i + 1][j], f[i + 1][j + 1], dp[j + k - 1] + a[i]});
        }
    }
    for (int j = 1; j <= n; ++j) {
        dp[j] = std::max(dp[j - 1], f[1][j - k + 1]);
    }
    if (dp[n] < x * a[1]) {
        puts("-1");
    } else {
        printf("%I64d", dp[n]);
    }
    return 0;
}
