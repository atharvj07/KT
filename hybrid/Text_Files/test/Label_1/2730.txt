#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
  int k;
  cin >> k;

  if (k <= 9) { cout << "-1\n"; return 0; }

  string s;

  for (int i = 2; i * i <= k; i++) {
    if (k % i != 0) continue;
    vector<vector<int>> g(i, vector<int>(k / i));
    map<int, int> cnt;

    for (int j = i - 1; j >= 0; j--) {
      for (int l = k / i - 1; l >= 0; l--) {
        g[j][l] = cnt['a'];
        for (char ch : "aeiou") cnt[ch]++;
      }
    }

    for (int j = 0; j < i; j++) {
      for (int l = 0; l < k / i; l++) {
        s += 'a' + g[j][l] % 5;
        if (l % 2 == 1) g[j][l]--;
        else g[j][l]++;
        cnt['a' + g[j][l] % 5]++;
      }
    }

    break;
  }

  if (s.empty()) cout << "-1\n";
  else cout << s << "\n";

  return 0;
}
