n = int(input())
piles = [list(map(int, input().split())) for _ in range(n)]
sum1, sum2 = 0, 0
piles.sort(reverse=True, key=lambda p: p[1]-p[-1])
while piles:
    p = piles.pop(0)
    if p[1] > p[-1]:
        sum1 += p[1]
        p[1] -= 1
    else:
        sum2 += p[-1]
        p[-1] -= 1
    if p[1] > p[-1]:
        piles.append(p)
print(sum1, sum2)
