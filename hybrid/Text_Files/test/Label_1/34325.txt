import java.io.*;
import java.util.*;

public class Main {

    static class Pipe {
        int from, to, diameter;

        public Pipe(int from, int to, int diameter) {
            this.from = from;
            this.to = to;
            this.diameter = diameter;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        int p = Integer.parseInt(st.nextToken());

        List<Pipe>[] pipes = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            pipes[i] = new ArrayList<>();
        }

        for (int i = 0; i < p; i++) {
            st = new StringTokenizer(in.readLine());
            int from = Integer.parseInt(st.nextToken());
            int to = Integer.parseInt(st.nextToken());
            int diameter = Integer.parseInt(st.nextToken());

            pipes[from].add(new Pipe(from, to, diameter));
            pipes[to].add(new Pipe(to, from, diameter));
        }

        Queue<Integer> tanks = new LinkedList<>();
        Queue<Integer> taps = new LinkedList<>();
        boolean[] visited = new boolean[n + 1];

        for (int i = 1; i <= n; i++) {
            if (pipes[i].size() > 1) {
                visited[i] = true;
            } else if (pipes[i].size() == 1) {
                if (pipes[i].get(0).from == i) {
                    tanks.add(i);
                    visited[i] = true;
                } else {
                    taps.add(i);
                    visited[i] = true;
                }
            }
        }

        Queue<Pair<Integer, Integer>> queue = new LinkedList<>();

        for (int tank : tanks) {
            queue.add(new Pair<>(tank, tank));
        }

        int t = 0;

        while (!queue.isEmpty()) {
            Pair<Integer, Integer> current = queue.remove();
            int tank = current.getKey();
            int tap = current.getValue();

            if (visited[tap]) {
                t++;

                int maxDiameter = Integer.MAX_VALUE;

                for (Pipe pipe : pipes[tank]) {
                    if (pipe.to != tap) {
                        maxDiameter = Math.min(maxDiameter, pipe.diameter);
                    }
                }

                System.out.println(tank + " " + tap + " " + maxDiameter);
                visited[tank] = false;
            } else {
                for (Pipe pipe : pipes[tap]) {
                    if (pipe.to != tank) {
                        queue.add(new Pair<>(tank, pipe.to));
                    }
                }
            }
        }

        if (t == 0) {
            System.out.println(0);
        }
    }

    static class Pair<T, U> {

        private final T key;
        private final U value;

        public Pair(T key, U value) {
            this.key = key;
            this.value = value;
        }

        public T getKey() {
            return key;
        }

        public U getValue() {
            return value;
        }
    }
}
