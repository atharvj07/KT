import java.util.Scanner;

public class EqualizeArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        // Find the maximum element in the array.
        int max = a[0];
        for (int i = 1; i < n; i++) {
            max = Math.max(max, a[i]);
        }

        // Find the minimum number of operations required to make all elements equal to the maximum element.
        int operations = 0;
        for (int i = 0; i < n; i++) {
            operations += (max - a[i]);
        }

        // If the number of elements that need to be equal is less than the number of operations required to make all elements equal to the maximum element,
        // then we can simply make all elements equal to the minimum element.
        if (n - k < operations) {
            operations = 0;
            for (int i = 0; i < n; i++) {
                operations += a[i];
            }
        }

        System.out.println(operations);
    }
}
