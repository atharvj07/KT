import java.util.*;

class Main {
    static int[][] roads;
    static int[] armies, soldiers;

    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt(), m = cin.nextInt();
        roads = new int[n][n];
        armies = new int[n];
        soldiers = new int[n];
        for (int i = 0; i < n; i++) {
            armies[i] = cin.nextInt();
            soldiers[i] = cin.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int u = cin.nextInt() - 1, v = cin.nextInt() - 1;
            roads[u][v] = roads[v][u] = 1;
        }
        if (!canBe())
            System.out.println("NO");
        else {
            System.out.println("YES");
            for (int[] row : roads) {
                for (int val : row)
                    System.out.print(val + " ");
                System.out.println();
            }
        }
    }

    static boolean canBe() {
        int[][] map = new int[202][202];
        for (int i = 0; i < 202; i++)
            for (int j = 0; j < 202; j++)
                map[i][j] = -1;
        for (int i = 0; i < 202; i++)
            map[i][i] = 0;
        for (int k = 0; k < armies.length; k++)
            for (int i = 0; i < armies.length; i++)
                for (int j = 0; j < armies.length; j++)
                    if (map[i][k] != -1 && map[k][j] != -1)
                        map[i][j] = map[i][k] + map[k][j];
        int total = 0;
        for (int x : armies)
            total += x;
        if (total != soldiers.length)
            return false;
        for (int i = 0; i < soldiers.length; i++) {
            if (soldiers[i] > armies[i])
                return false;
            int sum = 0;
            for (int j = 0; j < armies.length; j++)
                if (map[i][j] != -1) {
                    if (armies[i] - soldiers[i] > armies[j] - soldiers[j]) {
                        sum += armies[j] - soldiers[j];
                        roads[i][j] = armies[j] - soldiers[j];
                    } else {
                        sum += armies[i] - soldiers[i];
                        roads[i][j] = armies[i] - soldiers[i];
                        break;
                    }
                }
            if (sum < soldiers[i] - armies[i])
                return false;
        }
        return true;
    }
}
