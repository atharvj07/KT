import java.util.Scanner;

public class Valerie {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        int[][] teeth = new int[n][2];
        for (int i = 0; i < n; i++) {
            teeth[i][0] = scanner.nextInt();
            teeth[i][1] = scanner.nextInt();
        }

        int maxCrucians = 0;
        for (int i = 0; i < m; i++) {
            int rowViability = 0;
            for (int j = 0; j < n; j++) {
                if (teeth[j][0] == i + 1) {
                    rowViability += teeth[j][1];
                }
            }
            maxCrucians = Math.max(maxCrucians, Math.min(rowViability, k));
        }

        System.out.println(maxCrucians);
    }
}
