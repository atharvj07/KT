import java.util.*;
class Main {
    static final String SJFNB = "sjfnb", CSLNB = "cslnb";
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int[] a = new int[n];
        TreeSet<Integer> set = new TreeSet<>();
        for (int i = 0; i < n; i++) {
            a[i] = cin.nextInt();
            set.add(a[i]);
        }
        cin.close();
        System.out.println(set.size() == 1 ? SJFNB : CSLNB);
    }
}
