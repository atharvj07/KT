#include <bits/stdc++.h>

using namespace std;

const int N = 2e5 + 5;
vector<int> a(N), pos(N);
vector<vector<int>> occ(N);
int f[N];

int main() {
  int n;
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> a[i];
    pos[a[i]] = i;
  }
  for (int i = 0; i < n; ++i) occ[a[i]].push_back(i);

  for (int k = 1; k <= n; ++k) {
    int m = occ[k].size();
    if (m == 0) {
      f[k] = -1;
    } else {
      f[k] = n + 1;
      for (int i = 0; i < m; ++i) {
        if (i < m - 1) {
          f[k] = min(f[k], occ[k][i + 1] - occ[k][i] - 1);
        }
        if (i > 0) {
          f[k] = min(f[k], occ[k][i] - occ[k][i - 1] - 1);
        }
        if (i > 0 && i < m - 1) {
          f[k] = min(f[k], occ[k][i + 1] - occ[k][i - 1] - 2);
        }
        if (k > 1) {
          if (occ[k - 1].empty()) {
            f[k] = min(f[k], pos[k - 1] - occ[k][i] + 1);
          } else if (occ[k - 1][0] < occ[k][i]) {
            f[k] = min(f[k], occ[k - 1][0] - occ[k][i] + 1);
          }
        }
        if (k < n) {
          if (occ[k + 1].empty()) {
            f[k] = min(f[k], occ[k][i] - pos[k + 1] + 1);
          } else if (occ[k + 1].back() > occ[k][i]) {
            f[k] = min(f[k], occ[k][i] - occ[k + 1].back() + 1);
          }
        }
      }
      f[k] = min(f[k], pos[k] + n - occ[k].back());
      f[k] = min(f[k], n - 1 - pos[k] + occ[k][0]);
    }
    cout << f[k] << " ";
  }
  cout << endl;

  return 0;
}
