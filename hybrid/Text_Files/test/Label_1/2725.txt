#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 998244353;
int main() {
    ll n;
    cin >> n;
    ll ans = 0;
    if (n % 2 == 0) {
        n /= 2;
        ans = (n * ((n + 1) % mod)) % mod;
        ans = (ans * ((2 * n + 1) % mod)) % mod;
    }
    cout << ans;
    return 0;
}
