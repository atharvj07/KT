t = int(input())
for _ in range(t):
    s = input()
    pos = s.find('a')
    if pos == -1:
        print("NO")
        continue
    l, r = pos, pos
    ok = True
    for c in range(1, len(s)):
        char = chr(ord('a') + c)
        if l > 0 and s[l-1] == char:
            l -= 1
        elif r < len(s)-1 and s[r+1] == char:
            r += 1
        else:
            ok = False
            break
    print("YES" if ok else "NO")
