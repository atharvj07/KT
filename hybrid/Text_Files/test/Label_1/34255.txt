import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String s = br.readLine();
        int q = Integer.parseInt(br.readLine());

        int[][] dp = new int[n + 1][26]; // dp[i][c] = length of the longest segment of colour c ending at index i
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 26; j++) {
                dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j]);
            }
            dp[i + 1][s.charAt(i) - 'a'] = Math.max(dp[i + 1][s.charAt(i) - 'a'], i + 1);
        }

        for (int i = 0; i < q; i++) {
            String[] tokens = br.readLine().split(" ");
            int m = Integer.parseInt(tokens[0]);
            char c = tokens[1].charAt(0);

            int ans = 0;
            for (int j = 0; j < 26; j++) {
                if (dp[n][j] > m) {
                    ans = Math.max(ans, j == c - 'a' ? n - dp[n][j] + dp[n - m][j] : n - dp[n][j] + dp[n - m][c - 'a']);
                } else {
                    ans = Math.max(ans, n);
                }
            }

            System.out.println(ans);
        }
    }
}
