import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int k = 0;
        int t = 0;
        int sum = 0;
        for (int i = 0; i < n; i++) {
            sum += a[i];
        }
        Arrays.sort(a);
        Arrays.sort(b);
        for (int i = n - 1; i >= 0; i--) {
            if (sum <= b[i]) {
                k = i + 1;
                break;
            }
            sum -= a[i];
            t += a[i];
        }
        System.out.println(k + " " + t);
    }
}
