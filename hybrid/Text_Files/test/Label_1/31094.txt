import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.StringTokenizer;

public class WhoWillWin {

    static class Bid implements Comparable<Bid> {
        int participant;
        int size;

        public Bid(int participant, int size) {
            this.participant = participant;
            this.size = size;
        }

        @Override
        public int compareTo(Bid other) {
            return Integer.compare(this.size, other.size);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());

        List<Bid> bids = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int participant = Integer.parseInt(tokenizer.nextToken());
            int size = Integer.parseInt(tokenizer.nextToken());
            bids.add(new Bid(participant, size));
        }

        int q = Integer.parseInt(reader.readLine());
        for (int i = 0; i < q; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int k = Integer.parseInt(tokenizer.nextToken());
            List<Integer> absentParticipants = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                absentParticipants.add(Integer.parseInt(tokenizer.nextToken()));
            }

            Bid winner = findWinner(bids, absentParticipants);
            System.out.println(winner == null ? "0 0" : winner.participant + " " + winner.size);
        }
    }

    private static Bid findWinner(List<Bid> bids, List<Integer> absentParticipants) {
        Collections.sort(bids);
        Bid currentWinner = null;
        for (Bid bid : bids) {
            if (!absentParticipants.contains(bid.participant)) {
                if (currentWinner == null || bid.size > currentWinner.size) {
                    currentWinner = bid;
                }
            }
        }
        return currentWinner;
    }
}
