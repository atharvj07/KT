import sys
n=int(input())
b,vis,lis=[],[],[]
for x in map(int,input().split()):
  b.append(x)
  vis.append(0)
for x in b:
  lis.append(0)
i,j=0,0
while i<n:
  if vis[i]==1:
    i+=1
    continue
  j=i
  sum=0
  while j<n:
    if vis[j]==0:
      sum^=b[j]
    j+=1
  if sum not in lis:
    print("No")
    sys.exit()
  else:
    lis.remove(sum)
    ind=-1
    for k in range(n):
      if vis[k]==0 and sum^b[k] in lis:
        ind=k
        break
    lis.remove(sum^b[ind])
    for k in range(i,ind+1):
      vis[k]=1
      print(b[k],end=' ')
