def min_cost(n, k, a, b):
    costs = sorted([(a[i] + b[i], b[i], i) for i in range(n)])
    dp = [[float('inf')]*(k+1) for _ in range(n+1)]
    dp[0][0] = 0
    for i in range(1, n+1):
        for j in range(k+1):
            dp[i][j] = dp[i-1][j]
            if j > 0:
                dp[i][j] = min(dp[i][j], dp[i-1][j-1] + costs[i-1][0])
                if costs[i-1][2] < i-1:
                    dp[i][j] = min(dp[i][j], dp[i-1][j-1] + costs[i-1][1])
    return dp[n][k]

n, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
print(min_cost(n, k, a, b))
