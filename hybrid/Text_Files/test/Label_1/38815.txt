import sys
import math

def gcd(a,b):
    while(b):
        a, b = b, a % b
    return a

def solve():
    n = int(input())
    x = [int(i) for i in input().split()]
    x.sort()
    diff = []
    for i in range(1,n):
        diff.append(x[i]-x[i-1])
    g = diff[0]
    for i in diff:
        g = gcd(g,i)
    ans = []
    for i in x:
        if i%g==0:
            ans.append(i)
    print(len(ans))
    print(*ans)

solve()
