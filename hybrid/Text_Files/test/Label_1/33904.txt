import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int sum = 0;
        for (int ai : a) {
            sum += ai;
        }
        if (sum % 2 == 0) {
            System.out.println("BitAryo");
        } else {
            System.out.println("BitLGM");
        }
    }
}
