import sys

input = sys.stdin.readline

n = int(input())
A = []
B = []

for _ in range(n):
	s, a = input().split()
	if s=='00': pass
	elif s=='10': A.append(int(a))
	elif s=='01': B.append(int(a))
	else: A.append(int(a)); B.append(int(a))

A.sort(reverse=True)
B.sort(reverse=True)

i, j, ans = 0, 0, 0
while i<len(A) and j<len(B) and 2*(i+j+1)>=n:
	if A[i]>=B[j]:
		ans += A[i]
		i += 1
	else:
		ans += B[j]
		j += 1

if 2*(i+j)>=n: print(ans)
else: print(0)
