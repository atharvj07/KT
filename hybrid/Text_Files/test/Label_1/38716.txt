from sys import stdin
input = stdin.readline

def solve():
    s = input()
    n = len(s)
    a = [0] * n
    b = [0] * n
    if s[n-1] != '0':
        b[n-1] = 1
    for i in range(n-2, -1, -1):
        if (int(s[i] + s[i+1]) % 4 == 0):
            b[i] = b[i+2] + 1
        b[i] += b[i+1]
    for i in range(n-1, -1, -1):
        if (int(s[i]) % 4 == 0):
            a[i] = a[i+1] + 1
        a[i] += b[i]
    return a[0]
print(solve())
