MOD = 998244353

def count_partitions(a, l, r):
    n = len(a)
    dp = [0] * (n + 1)
    dp[0] = 1
    for i in range(1, n + 1):
        for j in range(i, 0, -1):
            if a[j-1] == '0':
                continue
            num = int(a[j-1:i])
            if l <= num <= r:
                dp[i] = (dp[i] + dp[j-1]) % MOD
            elif num > r:
                break
    return dp[n]

a = input().strip()
l = int(input().strip())
r = int(input().strip())

print(count_partitions(a, l, r))
