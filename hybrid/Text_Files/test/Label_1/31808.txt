import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Queue<int[]> tanks = new LinkedList<>(), send = new LinkedList<>();
        int n = scanner.nextInt();
        for (int i = 0; i < n; i++) {
            tanks.add(new int[]{scanner.nextInt(), i + 1});
        }
        int time = 0;
        while (!tanks.isEmpty() || !send.isEmpty()) {
            if (!send.isEmpty()) {
                int[] item = send.poll();
                time++;
            }
            while (!tanks.isEmpty() && tanks.peek()[0] >= tanks.peek()[1]) {
                send.add(tanks.poll());
            }
            if (send.isEmpty() && !tanks.isEmpty()) {
                send.add(tanks.poll());
                time++;
            }
        }
        System.out.println(time);
    }
}
