#include <iostream>
using namespace std;
const int N = 100005;
int a[N], b[N];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  b[n] = a[n];
  for (int i = n - 1; i >= 1; i--) b[i] = a[i] + b[i + 1];
  for (int i = 1; i <= n; i++) cout << b[i] << " ";
  cout << '\n';
}
