import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Auction {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] participants = new int[n];
        int[] bids = new int[n];
        for (int i = 0; i < n; i++) {
            participants[i] = scanner.nextInt() - 1;
            bids[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int k = scanner.nextInt();
            List<Integer> excluded = new ArrayList<>();
            for (int j = 0; j < k; j++) {
                excluded.add(scanner.nextInt() - 1);
            }
            int winner = findWinner(participants, bids, excluded);
            if (winner == -1) {
                System.out.println("0 0");
            } else {
                System.out.println((winner + 1) + " " + bids[winner]);
            }
        }
    }

    private static int findWinner(int[] participants, int[] bids, List<Integer> excluded) {
        int n = participants.length;
        int[] activeParticipants = new int[n];
        for (int i = 0; i < n; i++) {
            if (!excluded.contains(participants[i])) {
                activeParticipants[participants[i]]++;
            }
        }
        int winner = -1;
        for (int i = 0; i < n; i++) {
            if (activeParticipants[participants[i]] > 0) {
                if (winner == -1 || bids[i] > bids[winner]) {
                    winner = i;
                }
                activeParticipants[participants[i]]--;
            }
        }
        return winner;
    }
}
