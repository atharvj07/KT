s = input()
t = input()
if len(s) != len(t):
    print("No")
else:
    s = s.lower()
    t = t.lower()
    vowels = ["a", "e", "i", "o", "u"]
    can = True
    for i in range(len(s)):
        if s[i] != t[i]:
            if s[i] in vowels and t[i] in vowels:
                continue
            elif s[i] not in vowels and t[i] not in vowels:
                continue
            else:
                can = False
                break
    if can:
        print("Yes")
    else:
        print("No")
