import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long a = in.nextLong(), b = in.nextLong();
        long x1 = in.nextLong(), y1 = in.nextLong();
        long x2 = in.nextLong(), y2 = in.nextLong();
        System.out.print(Math.min(isBad(x1, y1, a, b) + isBad(x2, y2, a, b), isBad(x1, y2, a, b) + isBad(x2, y1, a, b)));
    }
    public static long isBad(long x, long y, long a, long b) {
        return (Math.abs(x + y) % (2 * a) == 0 || Math.abs(x - y) % (2 * b) == 0) ? 1 : 0;
    }
}
