import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            int[] a = new int[n];
            int[] b = new int[n];
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
            }
            for (int i = 0; i < n; i++) {
                b[i] = sc.nextInt();
            }
            int[][] dp = new int[n][2];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < 2; j++) {
                    dp[i][j] = -1;
                }
            }
            int ans = Math.min(dfs(0, 0, a, b, dp), dfs(0, 1, a, b, dp));
            if (ans == -1) {
                System.out.println("NA");
            } else {
                System.out.println(ans);
            }
        }
    }

    public static int dfs(int floor, int building, int[] a, int[] b, int[][] dp) {
        if (floor == a.length - 1) {
            if (a[floor] == 0) {
                return 1;
            } else if (a[floor] == 1) {
                return 0;
            } else {
                return -1;
            }
        }
        if (dp[floor][building] != -1) {
            return dp[floor][building];
        }
        int ans = -1;
        if (building == 0) {
            if (a[floor] == 0) {
                ans = 1 + dfs(floor + 1, 0, a, b, dp);
            } else if (a[floor] == 1) {
                ans = dfs(floor + 2, 0, a, b, dp);
            } else {
                if (floor + 1 < a.length && b[floor + 1] != 2) {
                    ans = 1 + dfs(floor + 1, 1, a, b, dp);
                }
                if (floor + 2 < a.length) {
                    ans = Math.min(ans, 1 + dfs(floor + 2, 1, a, b, dp));
                }
            }
        } else {
            if (b[floor] == 0) {
                ans = 1 + dfs(floor + 1, 1, a, b, dp);
            } else if (b[floor] == 1) {
                ans = dfs(floor + 2, 1, a, b, dp);
            } else {
                if (floor + 1 < b.length && a[floor + 1] != 2) {
                    ans = 1 + dfs(floor + 1, 0, a, b, dp);
                }
                if (floor + 2 < b.length) {
                    ans = Math.min(ans, 1 + dfs(floor + 2, 0, a, b, dp));
                }
            }
        }
        dp[floor][building] = ans;
        return ans;
    }
}
