#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, x;
  cin >> n >> k >> x;
  vector<int> v(n);
  for (int i = 0; i < n; i++) cin >> v[i];
  sort(v.begin(), v.end());
  for (int i = 0, j = n - 1; i < j && k > 0; i++, j--, k--) {
    v[i] ^= x;
    v[j] ^= x;
  }
  cout << *max_element(v.begin(), v.end()) << ' '
       << *min_element(v.begin(), v.end()) << endl;
  return 0;
}
