import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        Arrays.sort(a);
        int[] cnt = new int[1000001];
        for (long ai : a) {
            for (long j = 1; j <= 1000000; j *= 2) {
                while (ai >= j) {
                    cnt[(int) j]++;
                    ai -= j;
                }
            }
            cnt[(int) ai]++;
        }
        int cur = 0, ans = 0;
        for (int i = 1000000; i >= 1; i--) {
            if (cur == n) break;
            ans += cnt[i];
            if (ans >= cur + 1) {
                System.out.print(cur + 1 + " ");
                cur++;
            }
        }
        if (cur < n) System.out.print(-1);
        System.out.println();
    }
}
