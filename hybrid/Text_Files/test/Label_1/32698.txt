import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int k = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int v = in.nextInt();

        int boxes = (a + v - 1) / v; // the minimum number of boxes needed
        if (k < boxes) boxes = (a + k - 1) / k; // the minimum number of boxes, taking into account the maximum number of sections
        if (b < boxes - 1) boxes = -1; // if we don't have enough divisors, it's impossible

        System.out.println(boxes);
    }
}
