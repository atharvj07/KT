#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s = to_string(n);
  if (s.length() == 1 || (k == 1 && s[0] == '0')) {
    cout << 0 << endl;
    return 0;
  }
  int rem = (int)pow(10, k) - n % (int)pow(10, k);
  if (rem > 0 && rem <= s.length()) {
    cout << rem << endl;
  } else if (n == 0)
    cout << 0 << endl;
  else {
    int ans = INT_MAX;
    int l = s.length();
    for (int i = 0; i < l; i++) {
      if ((n - (s[i] - '0') * (int)pow(10, (l - 1 - i))) % (int)pow(10, k) == 0)
        ans = min(ans, l - 1 - i);
    }
    if (n % 10 == 0)
      ans = min(ans, (int)s.length() - 1);
    if (ans == INT_MAX)
      ans = s.length();
    cout << ans << endl;
  }
  return 0;
}
