#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
void solve() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    vector<bool> vis(n, false);
    for (ll i = 0; i < n; i++) cin >> a[i];
    vector<ll> t;
    for (ll i = 0; i < n;) {
        if (a[i] != -1) {
            ll j = i, c = 1;
            while (j + 1 < n && a[j + 1] != -1 && a[j + 1] == a[j] + c) c++, j++;
            i = j + 1;
            t.push_back(c);
        } else {
            t.push_back(0);
            i++;
        }
    }
    ll sum = 0;
    for (auto x : t) sum += x;
    cout << sum << "\n";
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    solve();
    return 0;
}
