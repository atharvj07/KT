from sys import stdin
from copy import deepcopy
from math import gcd

def preprocess(a):
    min_val = 1e9
    for x in a:
        min_val = min(min_val, x)
    for i in range(0, len(a)):
        if a[i] % min_val != 0:
            return None
    return min_val

for _ in range(int(stdin.readline())):
    n = int(stdin.readline())
    a = list(map(int, stdin.readline().split()))
    min_val = preprocess(a)
    if min_val is None:
        print("NO")
    else:
        b = deepcopy(a)
        b.sort()
        flag = True
        for i in range(0, len(a)):
            if a[i] != b[i] and a[i] % min_val != 0:
                flag = False
                break
        if flag:
            print("YES")
        else:
            print("NO")
