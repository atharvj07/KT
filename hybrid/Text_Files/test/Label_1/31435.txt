import java.util.*;

class Main {
    private static long[] masks;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        masks = new long[n];
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            masks[i] = 0L;
            for (int j = 0; j < k; j++) {
                int pos = sc.nextInt();
                masks[i] |= 1L << pos;
            }
        }

        int q = sc.nextInt();
        while (q-- > 0) {
            int type = sc.nextInt();
            long m = 0L;
            if (type != 0) {
                m = masks[sc.nextInt()];
            }

            switch (type) {
                case 0:
                    System.out.println((sc.nextInt() & 1L));
                    break;
                case 1:
                    System.out.println(Long.bitCount(m));
                    break;
                case 2:
                    System.out.println(Long.bitCount(63L & ~m));
                    break;
                case 3:
                    System.out.println(Long.bitCount(63L ^ m));
                    break;
                case 4:
                    System.out.println((m == 63L) ? 1 : 0);
                    break;
                case 5:
                    System.out.println((m != 0L) ? 1 : 0);
                    break;
                case 6:
                    System.out.println((m == 0L) ? 1 : 0);
                    break;
                case 7:
                    System.out.println(Long.bitCount(m));
                    break;
                case 8:
                    System.out.println(m);
                    break;
            }
        }
    }
}
