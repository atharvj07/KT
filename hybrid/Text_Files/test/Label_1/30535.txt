import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        for (int i = 0; i < t; i++) {
            long s = in.nextLong();
            long ans = 0;
            while (s > 0) {
                long x = Math.min(s, 10);
                ans += x;
                s -= x;
                s += x / 10;
            }
            System.out.println(ans);
        }
    }
}
