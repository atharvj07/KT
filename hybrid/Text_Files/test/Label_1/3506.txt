#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main() {
	int t;
	cin >> t;
	while (t--) {
		map<int, int>a, b;
		int n;
		cin >> n;
		int c = 0;
		for (int i = 1; i <= n; i++) {
			int x, y;
			cin >> x >> y;
			a[x]++;
			b[y]++;
			if (a[x] > 1 || b[y] > 1)
				c = 1;
		}
		if (c)
			cout << n * (n - 1) * (n - 2) / 6 << "\n";
		else
			cout << n * (n - 1) * (n - 2) / 6 + (n - 1) * (n - 2) / 2 << "\n";
	}
}
