#include <bits/stdc++.h>

using namespace std;

typedef long long int ll;
typedef long double ld;
#define pi pair<ll,ll>
#define pb push_back
#define ump unordered_map
#define all(x) x.begin(),x.end()
#define imax numeric_limits<ll>::max()
#define imin numeric_limits<ll>::min()
#define FASTIO ios_base::sync_with_stdio(false);cin.tie(NULL);

const int NN = 1e5+1;

ll na,nb,m;
ll a[NN], b[NN];

int main() {
    FASTIO;
    cin >> na >> m;
    for(ll i=1;i<=na;++i) cin >> a[i];
    for(ll i=1;i<=na;++i) cin >> b[i];
    while(m--){
        ll t; cin >> t;
        if(t==1){
            ll s,d,k; cin >> s >> d >> k;
            for(ll i=0;i<k;++i){
                b[d+i] = a[s+i];
            }
        }
        else{
            ll q; cin >> q;
            cout << b[q] << "\n";
        }
    }
    return 0;
}
