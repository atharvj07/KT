import java.util.Arrays;
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int[] a=new int[5];
        for(int i=0;i<5;i++) {
            a[i]=sc.nextInt();
        }
        Arrays.sort(a);
        int sum=0;
        for(int x:a) {
            if(sum%10!=0) sum+=(10-sum%10);
            sum+=x;
        }
        System.out.println(sum);
    }
}
