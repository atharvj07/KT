n = int(input().strip())
a = list(map(int, input().strip().split()))

last_seen = {}
sum_unique = 0

for i, val in enumerate(a):
    if val in last_seen:
        sum_unique += (last_seen[val] + 1) * (n - i)
    else:
        sum_unique += (i + 1) * (n - i)
    last_seen[val] = i

total_combinations = n * (n + 1) // 2
expected_value = sum_unique / total_combinations

print(f"{expected_value:.6f}")
