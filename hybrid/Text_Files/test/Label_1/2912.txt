#include <bits/stdc++.h>
using namespace std;
#define endl '\n'

int a[100001];
int dp[100001];

int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  dp[1] = abs(a[1] - a[2]);
  for (int i = 2; i < n; i++) {
    dp[i] = dp[i - 1] + abs(a[i] - a[i + 1]);
  }
  long long ans = 0;
  for (int i = 2; i <= n; i++) {
    ans = max(ans, 1LL * (dp[i - 1] - dp[i - 2]) + abs(a[i] - a[i - 1]));
  }
  cout << ans << endl;
  return 0;
}
