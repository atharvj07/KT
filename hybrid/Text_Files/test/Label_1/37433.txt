def find_max_beauty_numbers(n, arr):
    max_beauty = -1
    best_subset = []

    for bits in range(32, -1, -1):
        subset = [num for num in arr if num & (1 << bits)]
        if len(subset) > len(best_subset):
            max_beauty = bits
            best_subset = subset

    if max_beauty == -1:
        print(1)
        print(arr[0])
    else:
        print(len(best_subset))
        print(' '.join(map(str, best_subset)))

n = int(input().strip())
arr = list(map(int, input().strip().split()))
find_max_beauty_numbers(n, arr)
