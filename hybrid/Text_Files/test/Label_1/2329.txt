#include <cstdio>
#include <cmath>
typedef long long ll;
const int mod=1e9+9;
ll pow(ll a,ll b){
    if(b==0) return 1;
    ll ans=pow(a,b/2);
    ans*=ans; ans%=mod;
    if(b&1) ans*=a, ans%=mod;
    return ans;
}
ll w[15];
bool check(ll bv){
    for(ll i=0;i<=10;i++) if((bv>>i)&1){
        ll tmp=w[i]; tmp*=tmp; tmp%=mod;
        if(tmp==bv) return true;
    }
    return false;
}
int main(){
    ll a,b,p,x; scanf("%lld%lld%lld%lld",&a,&b,&p,&x);
    a%=p; b%=p;
    for(ll i=0;i<=10;i++){
        w[i]=pow(a,i);
    }
    ll ans=0; x%=mod;
    if(b==1){
        for(ll i=0;i<=10;i++) if((x>>i)&1) ans++;
    }
    else{
        ll invb=pow(b,p-2);
        for(ll i=0;i<=10;i++) if((x*invb%mod>>i)&1){
            ll bv=x*invb%mod; bv=bv>>i;
            if(check(bv)) ans++;
        }
    }
    printf("%lld\n",ans);
    return 0;
}
