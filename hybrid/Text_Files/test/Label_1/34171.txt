import java.util.Arrays;
import java.util.Scanner;

class Main {
    static int n, m, k;
    static int[] a;
    static int[][] dp;
    static int[][] rule;
    static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        rule = new int[k][3];
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < 3; j++) {
                rule[i][j] = sc.nextInt();
            }
        }
        dp = new int[1 << n][m + 1];
        for (int i = 0; i < (1 << n); i++) {
            Arrays.fill(dp[i], -1);
        }
        System.out.println(rec(0, 0));
    }

    static int rec(int mask, int cnt) {
        if (cnt == m) {
            return 0;
        }
        if (dp[mask][cnt] != -1) {
            return dp[mask][cnt];
        }
        dp[mask][cnt] = 0;
        for (int i = 0; i < n; i++) {
            if ((mask & (1 << i)) == 0) {
                int temp = a[i];
                for (int j = 0; j < k; j++) {
                    if ((mask & (1 << (rule[j][0] - 1))) != 0 && (mask & (1 << (rule[j][1] - 1))) == 0) {
                        temp += rule[j][2];
                    }
                }
                dp[mask][cnt] = Math.max(dp[mask][cnt], temp + rec(mask | (1 << i), cnt + 1));
            }
        }
        return dp[mask][cnt] % MOD;
    }
}
