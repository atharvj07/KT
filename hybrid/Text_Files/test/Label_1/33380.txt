import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    static class Pipe {
        int a, b, d;

        Pipe(int a, int b, int d) {
            this.a = a;
            this.b = b;
            this.d = d;
        }
    }

    static int n, m;
    static List<Integer>[] adj;
    static int[] indeg, outdeg;
    static Pipe[] pipes;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        adj = new List[n + 1];
        indeg = new int[n + 1];
        outdeg = new int[n + 1];
        pipes = new Pipe[m + 1];
        for (int i = 1; i <= n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int d = sc.nextInt();
            pipes[i] = new Pipe(a, b, d);
            adj[a].add(b);
            indeg[b]++;
            outdeg[a]++;
        }
        int t = 0;
        Queue<Integer> q = new LinkedList<>();
        for (int i = 1; i <= n; i++) {
            if (outdeg[i] > 0 && indeg[i] == 0) {
                q.add(i);
                t++;
            }
        }
        List<Integer> ans = new ArrayList<>();
        int[] tanks = new int[n + 1];
        while (!q.isEmpty()) {
            int u = q.poll();
            if (tanks[u] == 0) {
                List<Integer> cur = new ArrayList<>();
                cur.add(u);
                while (!cur.isEmpty()) {
                    int v = cur.remove(0);
                    if (tanks[v] != 0)
                        continue;
                    tanks[v] = u;
                    for (int i : adj[v]) {
                        if (tanks[i] == 0) {
                            cur.add(i);
                            ans.add(u);
                            ans.add(i);
                            ans.add(pipes[adj[v].indexOf(i) + 1].d);
                        }
                    }
                }
            }
        }
        System.out.println(t);
        for (int i = 0; i < ans.size(); i += 3)
            System.out.println(ans.get(i) + " " + ans.get(i + 1) + " " + ans.get(i + 2));
    }
}
