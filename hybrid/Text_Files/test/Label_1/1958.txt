#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;
ll f[500050],sz[500050];
ll n,ans1,ans2; vector<ll> g[500050];
void dfs1(ll v,ll p){
    sz[v]=1;
    for(auto i:g[v]){
        if(i!=p){
            dfs1(i,v);
            sz[v]+=sz[i];
        }
    }
    if(sz[v]%2==1){
        f[v]=1;
        ans1=1;
    }
}
void dfs2(ll v,ll p){
    ans2=max(ans2,f[v]);
    for(auto i:g[v]){
        if(i!=p){
            f[i]=(f[v]^1);
            dfs2(i,v);
        }
    }
}
int main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cin>>n;
    for(ll i=0;i<n-1;i++){
        ll a,b; cin>>a>>b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    dfs1(1,1);
    dfs2(1,1);
    cout<<ans1<<' '<<ans2<<'\n';
    return 0;
}
