import java.util.*;

public class Main {
    static int[] cards, conq;  // cards, conquerable cards

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int n = input.nextInt();
            if (n == 0)
                break;
            cards = new int[n];
            conq = new int[n];
            for (int i = 0; i < n; i++) {
                cards[i] = input.nextInt();
                conq[i] = cards[i];
            }
            if (solve(0))  // if there is a winning strategy
                for (int i = 0; i < n; i++)
                    System.out.print(conq[i] + " ");
            else
                System.out.println("No");
        }
    }

    static boolean solve(int idx) {
        if (idx == cards.length)  // if all cards are played
            return true;
        int best_conq = -1;
        for (int i = 0; i < idx; i++) {
            if (cards[idx] % conq[i] == 0) {
                int tmp = conq[i];
                conq[i] = conq[idx];
                if (solve(idx + 1))
                    return true;
                conq[i] = tmp;
                if (best_conq == -1 || conq[i] > best_conq)
                    best_conq = conq[i];
            }
        }
        if (best_conq != -1) {
            conq[idx] = best_conq;
            return solve(idx + 1);
        }
        return false;
    }
}
