#include <vector>
#include <string>
#include <iostream>

using namespace std;

struct Direction {
    int dist;
    string dir;
};

int main() {
    int n;
    cin >> n;
    Direction dirs[n];

    for (int i = 0; i < n; i++) {
        cin >> dirs[i].dist >> dirs[i].dir;
    }

    bool isNorthPole = true;
    bool isSouthPole = false;
    for (int i = 0; i < n; i++) {
        if (isNorthPole && dirs[i].dir != "South") {
            cout << "NO" << endl;
            return 0;
        }
        if (isSouthPole && dirs[i].dir != "North") {
            cout << "NO" << endl;
            return 0;
        }
        if (dirs[i].dir == "West" || dirs[i].dir == "East") {
            continue;
        }

        if (dirs[i].dir == "South") {
            isNorthPole = false;
            isSouthPole = true;
        }
        if (dirs[i].dir == "North") {
            isNorthPole = true;
            isSouthPole = false;
        }
    }

    if (isNorthPole) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }

    return 0;
}
