n=int(input())
a,b,c,d=[int(x) for x in input().split()]
a1,b1,c1,d1=[int(x) for x in input().split()]
a2,b2,c2,d2=[int(x) for x in input().split()]
a3,b3,c3,d3=[int(x) for x in input().split()]
if n>=a+b and (n-a-b)%2==0: print(1,a,(n-a-b)//2)
elif n>=a+d and (n-a-d)%2==0: print(1,a,(n-a-d)//2)
elif n>=b+c and (n-b-c)%2==0: print(1,(n-b-c)//2,c)
elif n>=b+d and (n-b-d)%2==0: print(1,(n-b-d)//2,d)
elif n>=a1+b1 and (n-a1-b1)%2==0: print(2,a1,(n-a1-b1)//2)
elif n>=a1+d1 and (n-a1-d1)%2==0: print(2,a1,(n-a1-d1)//2)
elif n>=b1+c1 and (n-b1-c1)%2==0: print(2,(n-b1-c1)//2,c1)
elif n>=b1+d1 and (n-b1-d1)%2==0: print(2,(n-b1-d1)//2,d1)
elif n>=a2+b2 and (n-a2-b2)%2==0: print(3,a2,(n-a2-b2)//2)
elif n>=a2+d2 and (n-a2-d2)%2==0: print(3,a2,(n-a2-d2)//2)
elif n>=b2+c2 and (n-b2-c2)%2==0: print(3,(n-b2-c2)//2,c2)
elif n>=b2+d2 and (n-b2-d2)%2==0: print(3,(n-b2-d2)//2,d2)
elif n>=a3+b3 and (n-a3-b3)%2==0: print(4,a3,(n-a3-b3)//2)
elif n>=a3+d3 and (n-a3-d3)%2==0: print(4,a3,(n-a3-d3)//2)
elif n>=b3+c3 and (n-b3-c3)%2==0: print(4,(n-b3-c3)//2,c3)
elif n>=b3+d3 and (n-b3-d3)%2==0: print(4,(n-b3-d3)//2,d3)
else: print(-1)
