#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    string s;
    cin >> s;
    vector<int> v(n + 1, -1);
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '.') {
            v[i] = 0;
        } else {
            v[i] = s[i] - '0';
        }
    }
    int p = -1;
    for (int i = 0; i < n; i++) {
        if (p == -1 && v[i] != 0) {
            p = i;
        }
        v[i + 1] = max(v[i + 1], v[i]);
    }
    int ans = 0;
    for (int i = 0; i < n && t > 0; i++) {
        if (i == p) {
            if (t >= 1) {
                ans = max(ans, v[i + 1] + 1);
                t--;
            }
        } else {
            if (t >= 1 && v[i + 1] > v[i]) {
                ans = max(ans, v[i + 1]);
                t--;
            }
        }
    }
    cout << ans;
    return 0;
}
