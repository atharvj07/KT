#include <bits/stdc++.h>
using namespace std;

int n, t[31], dp[31][1000005];

int solve(int pos, int tot) {
  if (pos == n) return tot;
  if (dp[pos][tot] != -1) return dp[pos][tot];
  int ans;
  if (pos % 2 == 0) {
    ans = max(solve(pos + 1, tot + t[pos]), solve(pos + 1, tot + t[pos] - 1));
  } else {
    ans = max(solve(pos + 1, tot + t[pos] - 1), solve(pos + 1, tot + t[pos]));
  }
  return dp[pos][tot] = ans;
}

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> t[i];
  memset(dp, -1, sizeof(dp));
  cout << solve(0, 0) << endl;
  return 0;
}
