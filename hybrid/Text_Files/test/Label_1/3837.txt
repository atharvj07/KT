#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 5005;
ll sum[MAXN];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
        int x;
        cin >> x;
        sum[i] = sum[i - 1] + x;
    }
    double ans = 0;
    for (int i = k; i <= n; ++i) {
        ans = max(ans, (double)(sum[i] - sum[i - k]) / k);
    }
    cout << ans << endl;
    return 0;
}
