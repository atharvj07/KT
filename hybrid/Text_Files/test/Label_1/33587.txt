import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long h = sc.nextLong();
        long m = sc.nextLong();
        long k = sc.nextLong();
        long h1 = sc.nextLong();
        long m1 = sc.nextLong();
        long h2 = sc.nextLong();
        long m2 = sc.nextLong();
        sc.close();
        if (h == 24) h--;
        if (m == 60) m--;
        long hh1 = (h1 + h - 1) / h;
        long mm1 = (m1 + m - 1) / m;
        long hh2 = (h2 + h - 1) / h;
        long mm2 = (m2 + m - 1) / m;
        long ans = hh2 * mm2 - hh1 * mm1;
        if (h1 == 0 && m1 == 0) ans++;
        if (hh2 == 1 && mm2 == 1) ans--;
        if (hh1 * mm1 > hh2 * mm2) ans--;
        System.out.println(ans);
    }
}
