import java.util.LinkedList;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        LinkedList<Integer> possibleIndex = new LinkedList<>();

        int n = scanner.nextInt();
        scanner.nextLine();

        String t = scanner.nextLine();
        String[] s = new String[n + 1];

        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextLine();
        }
        s[n] = t;

        int m = t.length();
        for (int i = 0; i < n; i++) {
            if (s[i].length() != m) {
                continue;
            }
            boolean same = true;
            for (int j = 0; j < m; j++) {
                if (t.charAt(j) == '?' || s[i].charAt(j) == '?') {
                    continue;
                }
                if (t.charAt(j) != s[i].charAt(j)) {
                    same = false;
                    break;
                }
            }
            if (same) {
                possibleIndex.add(i + 1);
            }
        }

        for (int i = 0; i < n; i++) {
            if (s[i].length() >= m) {
                continue;
            }
            boolean smaller = true;
            for (int j = 0; j < s[i].length(); j++) {
                if (t.charAt(j) == '?' || s[i].charAt(j) == '?') {
                    continue;
                }
                if (t.charAt(j) < s[i].charAt(j)) {
                    smaller = false;
                    break;
                }
            }
            if (smaller) {
                possibleIndex.add(i + 1);
            }
        }

        for (int i = 0; i < n; i++) {
            if (s[i].length() < m) {
                continue;
            }
            boolean larger = true;
            for (int j = 0; j < m; j++) {
                if (t.charAt(j) == '?' || s[i].charAt(j) == '?') {
                    continue;
                }
                if (t.charAt(j) > s[i].charAt(j)) {
                    larger = false;
                    break;
                }
            }
            if (larger) {
                possibleIndex.add(i + 1);
            }
        }

        possibleIndex.add(n + 1);

        possibleIndex.sort(null);

        for (int i = 0; i < possibleIndex.size(); i++) {
            System.out.print(possibleIndex.get(i) + " ");
        }
        System.out.println();
    }
}
