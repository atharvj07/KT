from sys import stdin

m, n = [int(i) for i in stdin.readline().strip().split()]
b = [[int(x) for x in input().strip().split()] for _ in range(m)]

ok = True
for i in range(m):
    for j in range(n):
        if b[i][j] == 0:
            continue
        for ii in range(m):
            if b[i][j] <= b[ii][j]:
                ok = False
                break
        for jj in range(n):
            if b[i][j] <= b[i][jj]:
                ok = False
                break
        break

print(["NO", "YES"][ok])
if ok:
    a = [[0] * n for _ in range(m)]
    for i in range(m):
        for j in range(n):
            for ii in range(m):
                if b[i][j] < b[ii][j]:
                    a[i][j] = 1
                    break
            for jj in range(n):
                if b[i][j] < b[i][jj]:
                    a[i][j] = 1
                    break
    for aa in a:
        print(' '.join(map(str, aa)))
