import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[][] competitors = new int[n][2];
        for (int i = 0; i < n; i++) {
            competitors[i][0] = input.nextInt();
            competitors[i][1] = input.nextInt();
        }
        List<Integer> possibleWinners = new ArrayList<>();
        Arrays.sort(competitors, (a, b) -> {
            if (a[0] != b[0]) {
                return Integer.compare(a[0], b[0]);
            } else {
                return Integer.compare(a[1], b[1]);
            }
        });
        double minSI = Double.MAX_VALUE;
        int lastSI = -1;
        for (int i = 0; i < n; i++) {
            if (competitors[i][0] != lastSI) {
                possibleWinners.add(i + 1);
                minSI = competitors[i][0];
                lastSI = competitors[i][0];
            } else {
                if (competitors[i][1] <= minSI) {
                    possibleWinners.add(i + 1);
                }
            }
        }
        Collections.sort(possibleWinners);
        for (int i : possibleWinners) {
            System.out.print(i + " ");
        }
    }
}
