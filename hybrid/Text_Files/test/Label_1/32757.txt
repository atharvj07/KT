import java.util.Scanner;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);

    String[] notes = input.nextLine().split(" ");
    int[] distances = new int[2];

    Arrays.sort(notes);

    for (int i = 1; i < notes.length; i++) {
      distances[i - 1] = notes[i].charAt(0) - notes[i - 1].charAt(0);

      if (distances[i - 1] < 0) distances[i - 1] += 12;
    }

    System.out.println(distances[0] == 4 && distances[1] == 3 ? "major" : distances[0] == 3 && distances[1] == 4 ? "minor" : "strange");
  }
}
