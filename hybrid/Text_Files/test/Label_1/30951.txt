import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long l = sc.nextLong();
        long n = sc.nextInt();
        long[] x = new long[(int) n];
        long max = -1;
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextLong();
            max = Math.max(max, x[i]);
        }
        long distance = 0;
        long clockwise = 0;
        for (int i = 0; i < n; i++) {
            clockwise += l - x[i];
        }
        distance += 2 * clockwise;
        distance += Math.min(l - max, clockwise);
        System.out.println(distance);
    }
}
