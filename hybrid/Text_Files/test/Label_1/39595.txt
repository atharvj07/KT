n = int(input())
a = list(map(int, input().split()))
dp = [-1] * (n + 1)
dp[1] = 0
for i in range(2, n + 1):
    if i % 2 == 0 and a[i // 2] > 0:
        dp[i] = dp[i // 2] + 1
    if i % 3 == 0 and a[i // 3] > 0:
        dp[i] = min(dp[i], dp[i // 3] + 1) if dp[i] != -1 else dp[i // 3] + 1
    if a[i - 1] > 0:
        dp[i] = min(dp[i], dp[i - 1] + 1) if dp[i] != -1 else dp[i - 1] + 1
print(dp[n])
