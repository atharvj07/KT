#include <bits/stdc++.h>
#define rep(i, a, b) for (int i = a; i <= b; ++i)
using namespace std;
int n, x, a[50005];
int main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    rep(i, 1, n) cin >> a[i];
    cin >> x;
    int ans = 0, l = 1, sum = 0;
    rep(r, 1, n) {
      sum += a[r];
      while (sum < x * (r - l + 1) && l < r) sum -= a[l++];
      if (sum >= x) ans++;
    }
    cout << ans << endl;
  }
}
