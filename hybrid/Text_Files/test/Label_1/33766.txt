import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Queue;

class Main {

    private static final int[] dx = {-1, 0, 1, 0};
    private static final int[] dy = {0, 1, 0, -1};
    private static char[][] grid;
    private static int n, m, k;
    private static int[][] dist;
    private static Queue<Integer> queue;
    private static String path;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int[] input = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        n = input[0];
        m = input[1];
        k = input[2];
        grid = new char[n][m];
        int sx = 0, sy = 0, ex = 0, ey = 0;
        for (int i = 0; i < n; i++) {
            grid[i] = br.readLine().toCharArray();
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 'S') {
                    sx = i;
                    sy = j;
                } else if (grid[i][j] == 'T') {
                    ex = i;
                    ey = j;
                }
            }
        }
        dist = new int[n][m];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }
        dist[sx][sy] = 0;
        queue = new ArrayDeque<>();
        queue.offer(sx * m + sy);
        boolean found = false;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            int x = cur / m;
            int y = cur % m;
            if (x == ex && y == ey) {
                found = true;
                break;
            }
            for (int i = 0; i < 4; i++) {
                int nx = x + dx[i];
                int ny = y + dy[i];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m && dist[nx][ny] == -1 && grid[nx][ny] != '0') {
                    Map<Character, Integer> cnt = new HashMap<>();
                    for (int j = 0; j < 4; j++) {
                        int nnx = nx + dx[j];
                        int nny = ny + dy[j];
                        if (nnx >= 0 && nnx < n && nny >= 0 && nny < m && grid[nnx][nny] != '0') {
                            cnt.put(grid[nnx][nny], cnt.getOrDefault(grid[nnx][nny], 0) + 1);
                        }
                    }
                    boolean ok = true;
                    for (Map.Entry<Character, Integer> entry : cnt.entrySet()) {
                        if (entry.getValue() > k) {
                            ok = false;
                            break;
                        }
                    }
                    if (ok) {
                        queue.offer(nx * m + ny);
                        dist[nx][ny] = dist[x][y] + 1;
                        if (i == 0 || i == 1) {
                            path += grid[nx][ny];
                        } else if (i == 2 || i == 3) {
                            path = grid[nx][ny] + path;
                        }
                    }
                }
            }
        }
        if (!found) {
            System.out.println(-1);
        } else {
            System.out.println(path);
        }
    }
}
