def max_beauty_permutation(n):
    # Initialize an empty list for the permutation
    permutation = []
    
    # Find the highest power of 2 less than or equal to n
    highest_power_of_2 = 1
    while highest_power_of_2 <= n:
        highest_power_of_2 *= 2
    highest_power_of_2 //= 2
    
    # If n is already a power of 2 minus 1, start the permutation from 0 to n
    if highest_power_of_2 * 2 - 1 == n:
        permutation = list(range(n + 1))
    else:
        # Exclude the highest power of 2 and start from the next number
        exclude = highest_power_of_2
        for i in range(exclude + 1, n + 1):
            permutation.append(i)
        permutation.append(exclude)
        for i in range(exclude):
            permutation.append(i)
    
    # Calculate the beauty of the permutation
    beauty = sum(i ^ permutation[i] for i in range(len(permutation)))
    
    return beauty, permutation

n = int(input())
beauty, permutation = max_beauty_permutation(n)
print(beauty)
print(*permutation)
