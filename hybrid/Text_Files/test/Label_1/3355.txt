#include <bits/stdc++.h>
using namespace std;

const int MAXN = 3e5 + 5;
const int offset = 62;
const long long mod = 1e9 + 7;
inline long long add(long long x, long long y) { return (x + y) % mod; }
inline long long mul(long long x, long long y) { return (x * y) % mod; }

int q, bin[MAXN], sum[MAXN][offset];
vector<int> pos[MAXN];
long long ans[MAXN];

int main() {
  cin >> q;
  for (int i = 0, x; i < q; i++) {
    cin >> x;
    pos[x].push_back(i);
  }
  for (int b = 0; b < offset; b++) {
    for (int i = 0; i < q; i++) {
      if (i) sum[i][b] = sum[i - 1][b];
      sum[i][b] += (bin[q - 1 - i] ^ (bin[q - 1 - i] >> b)) & 1;
    }
  }
  for (int i = q - 1; i >= 0; i--) {
    for (int b = 0; b < offset; b++) {
      if (i) bin[i - 1] |= bin[i] << b;
      if (!pos[i].empty()) {
        for (int p : pos[i]) {
          int low = p, high = q - 1;
          while (low < high) {
            int mid = (low + high) >> 1;
            if (sum[mid][b] - (i ? sum[i - 1][b] : 0) <= 1)
              low = mid + 1;
            else
              high = mid;
          }
          ans[p] = add(ans[p], mul((low - p + 1) % mod, (q - high) % mod));
        }
      }
    }
  }
  for (int i = 0; i < q; i++) cout << ans[i] << " ";
  cout << endl;
  return 0;
}
