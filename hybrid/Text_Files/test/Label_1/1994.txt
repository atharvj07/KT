#include<bits/stdc++.h>
#include<algorithm>
using namespace std;
#define int long long
bool cmp(pair<int,int>a, pair<int,int>b){
    if(a.first==b.first)
    return a.second<b.second;
    return a.first<b.first;
}
signed main(){
    int n;
    cin>>n;
    vector<pair<int,int> > v;
    map<int,vector<int> > mp;
    for(int i=0;i<n;++i){
        int x,y;
        cin>>x>>y;
        v.push_back({x,y});
        mp[x].push_back(y);
    }
    sort(v.begin(),v.end(),cmp);
    int l1=0,l2=0;
    for(int i=0;i<n;++i){
        if(i==0){
            l1=v[i].first+v[i].second;
            l2=v[i].first-v[i].second;
            continue;
        }
        if(l1==v[i].first+v[i].second || l1==v[i].first-v[i].second){
            continue;
        }
        else{
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
    return 0;
}
