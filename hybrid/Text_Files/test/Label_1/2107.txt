#include<bits/stdc++.h>
using namespace std;
int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin>>t;
    while(t--){
        long long n,m;
        cin>>n;
        vector<long long>v(n);
        for(auto &x:v)cin>>x;
        cin>>m;
        vector<long long>w(m);
        for(auto &x:w)cin>>x;
        long long ans=0;
        sort(w.begin(),w.end());
        for(long long i=0;i<n;i++){
            if(binary_search(w.begin(),w.end(),-v[i])){
                ans++;
            }
        }
        cout<<ans<<endl;
    }
    return 0;
}
