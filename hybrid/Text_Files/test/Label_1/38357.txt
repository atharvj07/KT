import bisect

n, m = map(int, input().split())
events = list(map(int, input().split()))
intervals = list(map(int, input().split()))
intervals.sort()

for event in events:
    idx = bisect.bisect_left(intervals, event)
    if idx < m and event % intervals[idx] == 0:
        print('YES')
        print(event, idx + 1)
        break
else:
    print('NO')
