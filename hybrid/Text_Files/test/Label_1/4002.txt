#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;
typedef long long ll;
const int N = 1e5 + 5;
int n, m, d[N], ans;
int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= m; i++) {
        int x, y;
        scanf("%d%d", &x, &y);
        d[x]++; d[y]++;
    }
    for (int i = 1; i <= n; i++) {
        if (d[i] % 2) ans++;
    }
    printf("%d", ans / 2);
    return 0;
}
