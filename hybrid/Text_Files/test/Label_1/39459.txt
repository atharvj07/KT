import collections

n, k = map(int, input().split())
q = collections.deque()
a = []
b = list(map(int, input().split()))
for x in b:
    if x in q:
        q.remove(x)
        q.append(x)
    else:
        if len(q) == k:
            a.pop()
        q.append(x)
    a = [x for x in q]
print(len(a))
print(*a)
