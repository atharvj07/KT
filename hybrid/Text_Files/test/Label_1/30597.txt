import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[] blocked = new int[m];
        for (int i = 0; i < m; i++) {
            blocked[i] = sc.nextInt();
        }

        long[] cost = new long[k + 1];
        for (int i = 1; i <= k; i++) {
            cost[i] = sc.nextInt();
        }

        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= Math.min(i, k); j++) {
                int start = Math.max(0, i - j);
                if (Arrays.binarySearch(blocked, start) >= 0) {
                    continue;
                }

                if (dp[start] != -1) {
                    dp[i] = Math.min(dp[i] == -1 ? Integer.MAX_VALUE : dp[i], dp[start] + (int) cost[j]);
                }
            }
        }

        System.out.println(dp[n] == -1 ? -1 : dp[n]);
    }
}
