from sys import stdin
s=stdin.readline().strip()
x,y=map(int,stdin.readline().split())
ans=0
Stack=['']
for char in s:
    if char in '01':Stack.append(char)
    else:
        if not Stack[-1]:Stack.append('0')
        if Stack[-1]==char and len(Stack)>1:Stack.pop()
        else:Stack.append(char)
for i in range(1,len(Stack)):
    if Stack[i-1]=='0' and Stack[i]=='1':ans+=x
    if Stack[i-1]=='1' and Stack[i]=='0':ans+=y
print(ans)
