n = int(input())
a = list(map(int, input().split()))
ans = []
for i in range(n):
    for j in range(i, n):
        flag = True
        for k in range(i, j + 1):
            if a[k] % a[i] != 0:
                flag = False
                break
        if flag:
            ans.append((i, j))
ans = set(ans)
print(len(ans), max(ans, key=lambda x: x[1] - x[0])[1] - max(ans, key=lambda x: x[1] - x[0])[0])
for i in sorted(ans, key=lambda x: x[0]):
    print(i[0] + 1, end=' ')
