#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        long long x;
        cin >> x;
        bool ok = x;
        for (int i = 1; i < n; i++) {
            cin >> x;
            ok &= x == 0;
            if (!ok) break;
        }
        if (!ok) cout << "YES\n";
        else cout << "NO\n";
    }
    return 0;
}
