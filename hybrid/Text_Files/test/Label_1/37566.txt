mo = 998244353
x = list(input())[::-1]
l = list(input())[::-1]
r = list(input())[::-1]
n = len(x)
m = len(l)
dp = [[[-1] * 5 for _ in range(m+1)] for _ in range(n+1)]
def solve(i, j, rem):
    if i == n:
        return (rem == 0)
    if dp[i][j][rem] != -1:
        return dp[i][j][rem]
    res = 0
    if rem > 0:
        for d in r:
            if j < m and d >= l[j]:
                res += solve(i+1, j+1, rem-int(d))
    if rem > 0 or x[i] != '0':
        res += solve(i+1, j, rem+int(x[i]))
    res %= mo
    dp[i][j][rem] = res
    return res
print(solve(0, 0, 0))
