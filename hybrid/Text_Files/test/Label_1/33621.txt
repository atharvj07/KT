import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        String t = sc.next();
        if (s.equals(t)) {
            System.out.println(0);
            return;
        }
        int[] next = new int[n];
        for (int i = 1; i < n; i++) {
            int j = next[i - 1];
            while (j > 0 && s.charAt(j) != s.charAt(i)) {
                j = next[j - 1];
            }
            if (s.charAt(j) == s.charAt(i)) {
                j++;
            }
            next[i] = j;
        }
        int i = 0;
        int j = 0;
        int k = 0;
        int[][] ans = new int[10000][2];
        while (i < n) {
            if (j == n) {
                if (k % 2 == 0) {
                    ans[k][0] = i;
                    ans[k][1] = n;
                } else {
                    ans[k][0] = n - i;
                    ans[k][1] = n;
                }
                k++;
                j = 0;
            }
            if (s.charAt(i) == t.charAt(j)) {
                i++;
                j++;
            } else if (j == 0) {
                i++;
            } else {
                j = next[j - 1];
            }
        }
        if (j == n) {
            System.out.println(k);
            for (int l = 0; l < k; l++) {
                System.out.print(ans[l][0] + " " + ans[l][1] + " ");
            }
        } else {
            System.out.println(-1);
        }
    }
}
