import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int k = scan.nextInt();
        List<Integer> a = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            a.add(scan.nextInt());
        }
        Collections.sort(a);
        int l = 0, r = n - 1, ans = 0;
        while (l < r) {
            if (a.get(r) - a.get(l) <= 5) {
                ans += 2;
                l++;
                r--;
            } else {
                r--;
            }
        }
        if (l == r) ans++;
        System.out.println(ans);
    }
}
