#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1000000007;
const ll N = 1e9 + 5;
ll dist(ll x1, ll y1, ll x2, ll y2) {
    return abs(x1 - x2) + abs(y1 - y2);
}
int main() {
    ll n, m, c, h, x, y;
    cin >> n >> m;
    cin >> c;
    vector<pair<ll, ll>> points;
    for (ll i = 0; i < c; i++) {
        cin >> x >> y;
        points.push_back({x, y});
    }
    cin >> h;
    vector<pair<ll, ll>> restaurants;
    for (ll i = 0; i < h; i++) {
        cin >> x >> y;
        restaurants.push_back({x, y});
    }
    ll ans = LLONG_MAX;
    ll ind = 0;
    for (ll i = 0; i < h; i++) {
        ll temp = 0;
        for (ll j = 0; j < c; j++) {
            temp = max(temp, dist(restaurants[i].first, restaurants[i].second, points[j].first, points[j].second));
        }
        if (temp < ans) {
            ans = temp;
            ind = i;
        }
    }
    cout << ans << endl;
    cout << ind + 1 << endl;
    return 0;
}
