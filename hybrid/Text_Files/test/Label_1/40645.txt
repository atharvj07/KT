a, b, c = list(map(int, input().split()))
n = int(input())
x = list(map(int, input().split()))
ans = 0
cur = a
while cur != b and cur != c:
    if cur in x:
        ans += 1
        x.remove(cur)
    if cur != a:
        if x:
            if x[0] > cur:
                cur = x[0]
            else:
                break
        else:
            break
    if cur != c:
        if x:
            if x[-1] < cur:
                cur = x[-1]
            else:
                break
        else:
            break
print(ans)
