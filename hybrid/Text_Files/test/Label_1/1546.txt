#include <iostream>
#include <algorithm>
using namespace std;
int main() {
    long long n,a[100005],s=0;
    cin>>n;
    for(int i=0;i<n;i++) cin>>a[i];
    sort(a,a+n);
    for(int i=n-1;i>=0;i--) {
        if(a[i]%2==1) s+=a[i];
        else break;
    }
    for(int i=0;i<n;i++) {
        if(a[i]%2==0) s+=a[i];
    }
    cout<<s<<endl;
    return 0;
}
