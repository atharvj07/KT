import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);

    for (int n; (n = in.nextInt()) != 0;) {
      int[] arr = new int[n];
      for (int i = 0; i < n; i++) {
        arr[i] = in.nextInt();
      }
      Arrays.sort(arr);
      System.out.println(arr[1] - arr[0]);
    }
  }
}
