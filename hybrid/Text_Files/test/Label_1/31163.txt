import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    static int N = 200;
    static int[][] g = new int[N][N];
    static int[] a = new int[N];
    static int[] b = new int[N];
    static int[][] ans = new int[N][N];

    public static void main(String[] args) throws java.lang.Exception {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = in.nextInt();
        }
        for (int i = 0; i < m; i++) {
            int p = in.nextInt() - 1;
            int q = in.nextInt() - 1;
            g[p][q] = 1;
            g[q][p] = 1;
        }
        if (!dfs(0, -1)) {
            System.out.println("NO");
            return;
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }

    public static boolean dfs(int v, int p) {
        if (a[v] != b[v]) {
            return false;
        }
        for (int i = 0; i < N; i++) {
            if (g[v][i] == 1 && i != p) {
                if (a[i] + a[v] < b[i] || a[i] + a[v] > b[i] + b[v]) {
                    return false;
                }
                if (a[i] + a[v] == b[i]) {
                    ans[v][i] = a[v];
                    ans[i][v] = 0;
                } else {
                    ans[v][i] = b[i];
                    ans[i][v] = b[v] - a[i];
                }
                a[v] -= ans[v][i];
                a[i] += ans[v][i];
                if (!dfs(i, v)) {
                    return false;
                }
            }
        }
        return true;
    }
}
