import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int hl = input.nextInt(), hr = input.nextInt(), n = input.nextInt();
        Mirror[] mirrors = new Mirror[n];
        for (int i = 0; i < n; i++) {
            mirrors[i] = new Mirror(input.nextInt(), input.next().charAt(0), input.nextInt(), input.nextInt());
        }
        System.out.println(shootLaser(mirrors, hl, hr));
    }
    static class Mirror {
        int score, x1, x2;
        char dir;
        Mirror(int score, char dir, int x1, int x2) {
            this.score = score;
            this.dir = dir;
            this.x1 = x1;
            this.x2 = x2;
        }
        boolean touches(int x, int y) {
            return dir == 'F' ? x >= x1 && x <= x2 && y == 0 : x >= x1 && x <= x2 && y == 100;
        }
    }
    static int shootLaser(Mirror[] mirrors, int hl, int hr) {
        int[][] memo = new int[2][10001];
        return Math.max(shootLaser(mirrors, hl, hr, 0, memo), shootLaser(mirrors, hr, hl, 1, memo));
    }
    static int shootLaser(Mirror[] mirrors, int hl, int hr, int dir, int[][] memo) {
        if (memo[dir][hl] != 0) return memo[dir][hl];
        int score = 0;
        for (Mirror m : mirrors) {
            if (m.touches(hl, dir)) score = Math.max(score, m.score + shootLaser(mirrors, m.dir == 'F' ? hr : hl, m.dir == 'F' ? hl : hr, m.dir == 'T' ? 0 : 1, memo));
        }
        memo[dir][hl] = score;
        return score;
    }
}
