#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n), c(n);
  for (int &x : a) { cin >> x; }
  for (int &x : b) { cin >> x; }
  for (int i = 0; i < n; i++) { c[a[i] - 1] = b[i]; }
  int ans = 0;
  for (int i = 0; i < n; i++) { ans += (a[i] == c[i]); }
  cout << ans << endl;
  return 0;
}
