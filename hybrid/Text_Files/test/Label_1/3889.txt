#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) cin >> v[i];
    int cnt = 0;
    for (int i = 0; i < n; i++) {
        int mn = i;
        for (int j = i + 1; j < n; j++) {
            if (v[j] < v[mn]) mn = j;
        }
        if (mn != i) {
            swap(v[mn], v[i]);
            cnt++;
        }
    }
    cout << cnt << "\n";
    for (int i = 0; i < n; i++) {
        int mn = i;
        for (int j = i + 1; j < n; j++) {
            if (v[j] < v[mn]) mn = j;
        }
        if (mn != i) {
            cout << i << " " << mn << "\n";
        }
    }
}
