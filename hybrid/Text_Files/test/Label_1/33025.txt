import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[2 * n];
        for (int i = 0; i < 2 * n; i++) {
            a[i] = sc.nextInt();
        }
        int[] loc = new int[n + 1];
        for (int i = 0; i < 2 * n; i++) {
            loc[a[i]] = i;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (loc[i] % 2 == loc[i + 1] % 2) {
                ans++;
            }
        }
        System.out.println(ans / 2);
    }
}
