#include <bits/stdc++.h>
#define ll long long
using namespace std;
const int mod = 1e9 + 7;

ll fastpow(ll a, int n) {
    ll ans = 1;
    while (n > 0) {
        if (n % 2 == 1) { ans = (ans * a) % mod; }
        a = (a * a) % mod;
        n /= 2;
    }
    return ans;
}
const int N = 100005;
vector<int> adj[N];
int col[N]; // 1: love, -1: hate
bool vis[N];

bool dfs(int u, int c) {
    if (vis[u]) { return col[u] == c; }
    vis[u] = true;
    col[u] = c;
    for (int v : adj[u]) {
        if (!dfs(v, -c)) { return false; }
    }
    return true;
}

int main() {
    int n, m;
    cin >> n >> m;
    int love_cnt = 0;
    while (m--) {
        int a, b, c;
        cin >> a >> b >> c;
        if (c) { adj[a].push_back(b); adj[b].push_back(a); love_cnt++; }
    }
    ll ans = fastpow(2, love_cnt);
    for (int i = 1; i <= n; i++) {
        if (vis[i]) { continue; }
        if (!dfs(i, 1)) { ans = 0; break; }
    }
    cout << ans << '\n';
    return 0;
}
