a, b, h, w, n = map(int, input().split())
l = [int(i) for i in input().split()]
l.sort()
ans = -1
for i in range(n):
    if a <= w and b <= h:
        ans = i
        break
    if a <= h and b <= w:
        a, b = b, a
    if l[i] >= 2:
        h *= l[i]
if ans == -1:
    print(-1)
else:
    print(ans)
