#include<bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n,p[N],ans;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++)cin>>p[i];
    ans=n;
    for(int i=1;i<=n;i++)
        for(int j=1;j<=n;j++)
            if(i!=j){
                int x=i,y=j;
                bool a=0,b=0;
                while(x!=y){
                    if(p[x]<p[y]){
                        if(x==1 or p[x-1]>=p[x])a=1;
                        break;
                    }
                    else{
                        if(x==n or p[x+1]<=p[x])b=1;
                        x++;
                    }
                }
                if(a or b)ans--;
            }
    cout<<ans<<endl;
    return 0;
}
