n, m = map(int, input().split())
grid = ['.'*m]*n
for i in range(n):
    grid[i] = input()
for i in range(n):
    for j in range(m):
        if grid[i][j] == '.':
            if any(grid[k][j] == '#' for k in range(n)) or any(grid[i][k] == '#' for k in range(m)):
                print('No')
                exit()
print('Yes')
