import java.util.ArrayList;
import java.util.Arrays;
class Main {
    static Tree[] trees;

    static class Tree implements Comparable<Tree> {
        int pos;
        int height;

        public Tree(int p, int h) {
            pos = p;
            height = h;
        }

        public int compareTo(Tree other) {
            return pos - other.pos;
        }
    }

    public static void main(String[] args) throws Exception {
        java.io.BufferedReader r = new java.io.BufferedReader(new java.io.InputStreamReader(System.in));
        int n = Integer.parseInt(r.readLine());
        trees = new Tree[n];
        for (int i = 0; i < n; i++) {
            String[] line = r.readLine().split(" ");
            trees[i] = new Tree(Integer.parseInt(line[0]), Integer.parseInt(line[1]));
        }
        Arrays.sort(trees);

        int ans = 0;
        int lastR = Integer.MIN_VALUE;
        int lastL = Integer.MAX_VALUE;
        for (Tree t : trees) {
            boolean canR = t.pos + t.height > lastR;
            boolean canL = t.pos - t.height < lastL;
            if (canR || canL) {
                ans++;
                if (canR)
                    lastR = t.pos + t.height;
                if (canL)
                    lastL = t.pos - t.height;
            }
        }
        System.out.println(ans);
    }
}
