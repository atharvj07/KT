def can_win(n, k, x, adj):
    degree = [0] * (n+1)
    for u, v in adj:
        degree[u] += 1
        degree[v] += 1
    lst = [i for i in range(1, n+1) if degree[i] == 1 and i not in x]
    return len(lst) > 0

for i in range(int(input())):
    input()
    n, k = map(int, input().split())
    x = list(map(int, input().split()))
    lst = []
    for j in range(n-1):
        u, v = map(int, input().split())
        lst.append((u, v))
    print("YES" if can_win(n, k, x, lst) else "NO")
