import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt(), g = sc.nextInt(), b = sc.nextInt();
        if (r == g && g == b) {
            System.out.println(0);
        } else if (r == g || g == b || r == b) {
            System.out.println(1);
        } else {
            System.out.println(2);
        }
    }
}
