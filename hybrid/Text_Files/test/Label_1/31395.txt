import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        char[][] grid = new char[n][n];
        for (int i = 0; i < n; i++) {
            grid[i] = sc.next().toCharArray();
        }
        int[][] dp = new int[n][n]; // dp[i][j] = minimum cost to paint all cells in white from (i, j) to (n-1, n-1)
        dp[n-1][n-1] = (grid[n-1][n-1] == '#' ? 1 : 0);
        for (int i = n-2; i >= 0; i--) {
            dp[i][n-1] = (grid[i][n-1] == '#' ? 1 : 0) + dp[i+1][n-1];
        }
        for (int j = n-2; j >= 0; j--) {
            dp[n-1][j] = (grid[n-1][j] == '#' ? 1 : 0) + dp[n-1][j+1];
        }
        for (int i = n-2; i >= 0; i--) {
            for (int j = n-2; j >= 0; j--) {
                if (grid[i][j] == '#') {
                    dp[i][j] = 1 + Math.min(dp[i+1][j], dp[i][j+1]);
                } else {
                    dp[i][j] = Math.min(dp[i+1][j], dp[i][j+1]);
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
