import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.util.Arrays;

class Main {
    static final int MOD = (int) 1e9;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String[] data = in.readLine().split(" ");
        int n = Integer.parseInt(data[0]);
        long[] p = new long[4];
        for (int i = 0; i < 4; i++) {
            p[i] = Long.parseLong(data[i + 1]);
        }

        // calculate exlog_f(i) for i = 1 to n
        long[][] exlog = new long[n + 1][4];
        exlog[0][0] = 0;
        exlog[0][1] = -1;
        exlog[0][2] = -1;
        exlog[0][3] = -1;
        for (int i = 1; i <= n; i++) {
            exlog[i][0] = 0;
            exlog[i][1] = (exlog[i][1] + p[1] % MOD) % MOD;
            exlog[i][2] = (exlog[i][2] + p[2] * i % MOD) % MOD;
            exlog[i][3] = (exlog[i][3] + p[3] * i * i % MOD) % MOD;
        }

        // calculate prefix sum of exlog_f(i)
        long[][] prefixSum = new long[n + 1][4];
        for (int i = 0; i < 4; i++) {
            prefixSum[0][i] = 0;
            for (int j = 1; j <= n; j++) {
                prefixSum[j][i] = (prefixSum[j - 1][i] + exlog[j][i]) % MOD;
            }
        }

        // calculate the sum of prefix sum of exlog_f(i) for i = 1 to n
        long result = 0;
        long[] coef = new long[4];
        coef[0] = 1;
        for (int i = 1; i <= n; i++) {
            long[] tmp = Arrays.copyOf(coef, 4);
            for (int j = 0; j < 4; j++) {
                tmp[j] = (tmp[j] * i) % MOD;
            }
            coef = tmp;
            for (int j = 0; j < 4; j++) {
                result = (result + coef[j] * prefixSum[i][j] % MOD) % MOD;
            }
        }

        // print the result
        System.out.println(result);
    }
}
