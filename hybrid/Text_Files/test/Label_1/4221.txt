#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n; cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    int ans = 1;
    for (int i = 1; i < n; i++) if (h[i] < h[i - 1]) ans++;
    cout << ans << endl;
    return 0;
}
