import java.util.*;

public class Main {
  static int[] parent;
  static int[] boss;
  static int[] level;
  static int[] projectManager;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int q = sc.nextInt();
    parent = new int[n + 1];
    boss = new int[n + 1];
    level = new int[n + 1];
    projectManager = new int[q];
    Arrays.fill(boss, -1);
    for (int i = 2; i <= n; i++) {
      int p = sc.nextInt();
      parent[i] = p;
      if (boss[p] == -1) {
        boss[p] = p;
      }
      boss[i] = boss[p];
    }
    for (int i = 1; i <= n; i++) {
      if (boss[i] == -1) {
        boss[i] = i;
      }
    }
    for (int i = n; i >= 1; i--) {
      if (level[parent[i]] + 1 > level[i]) {
        level[i] = level[parent[i]] + 1;
      }
    }
    for (int i = 0; i < q; i++) {
      int l = sc.nextInt();
      int r = sc.nextInt();
      projectManager[i] = -1;
      int maxLevel = -1;
      for (int j = l; j <= r; j++) {
        if (level[boss[j]] > maxLevel) {
          maxLevel = level[boss[j]];
          projectManager[i] = j;
        } else if (level[boss[j]] == maxLevel) {
          if (j < projectManager[i]) {
            projectManager[i] = j;
          }
        }
      }
      System.out.println(projectManager[i] + " " + maxLevel);
    }
  }
}
