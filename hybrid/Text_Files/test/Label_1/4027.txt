#include <iostream>
#include <vector>
using namespace std;
const int N = 1e5 + 5;
int a[N], b[N], c[N], l[N];
int main() {
  ios_base::sync_with_stdio(false), cin.tie(NULL);
  int n, m, x, q, k, t;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  for (int i = 1; i <= n; i++) { cin >> b[i]; }
  for (int i = 1; i <= m; i++) {
    cin >> t >> x;
    if (t == 1) {
      cin >> q >> k;
      c[x + k] -= q;
      c[x] += q;
    } else {
      l[++l[0]] = x;
    }
  }
  for (int i = 1; i <= n; i++) {
    c[i] += c[i - 1];
    b[i] += c[i];
  }
  for (int i = 1; i <= l[0]; i++) { cout << b[l[i]] << endl; }
}
