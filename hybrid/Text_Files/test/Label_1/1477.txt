#include <bits/stdc++.h>
#define REP(i,n) for(int i=0;i<(n);i++)
#define FOR(i,a,b) for(int i=(a);i<(b);i++)
using namespace std;

int main(){
  int n,m; cin>>n>>m;
  vector<vector<char>> f(n,vector<char>(m));
  REP(i,n){
    string s; cin>>s;
    REP(j,m) f.at(i).at(j)=s.at(j);
  }
  vector<vector<vector<int>>> p(n,vector<vector<int>>(m,vector<int>(2,0)));
  REP(i,n){
    REP(j,m){
      if(i==0 && j==0){
        p.at(i).at(j).at(0)=f.at(i).at(j)=='W';
        p.at(i).at(j).at(1)=f.at(i).at(j)=='B';
      } else if(i==0){
        p.at(i).at(j).at(0)=p.at(i).at(j-1).at(0)+((f.at(i).at(j)=='W')?1:0);
        p.at(i).at(j).at(1)=p.at(i).at(j-1).at(1)+((f.at(i).at(j)=='B')?1:0);
      } else if(j==0){
        p.at(i).at(j).at(0)=p.at(i-1).at(j).at(0)+((f.at(i).at(j)=='W')?1:0);
        p.at(i).at(j).at(1)=p.at(i-1).at(j).at(1)+((f.at(i).at(j)=='B')?1:0);
      } else {
        p.at(i).at(j).at(0)=p.at(i-1).at(j).at(0)+p.at(i).at(j-1).at(0)-p.at(i-1).at(j-1).at(0)+((f.at(i).at(j)=='W')?1:0);
        p.at(i).at(j).at(1)=p.at(i-1).at(j).at(1)+p.at(i).at(j-1).at(1)-p.at(i-1).at(j-1).at(1)+((f.at(i).at(j)=='B')?1:0);
      }
    }
  }
  int ans=n*m;
  REP(i,n){
    REP(j,m){
      int w=0,b=0;
      FOR(ii,i,n){
        FOR(jj,j,m){
          int tmpw=p.at(ii).at(jj).at(0);
          int tmpb=p.at(ii).at(jj).at(1);
          if(i>0) {tmpw-=p.at(i-1).at(jj).at(0); tmpb-=p.at(i-1).at(jj).at(1);}
          if(j>0) {tmpw-=p.at(ii).at(j-1).at(0); tmpb-=p.at(ii).at(j-1).at(1);}
          if(i>0 && j>0) {tmpw+=p.at(i-1).at(j-1).at(0); tmpb+=p.at(i-1).at(j-1).at(1);}
          w+=tmpw;
          b+=tmpb;
        }
      }
      ans=min(ans,i*j+2);
      ans=min(ans,(i+1)*(j+1)+1);
      ans=min(ans,(n-i)*(m-j)+1);
      ans=min(ans,(n-i+1)*(m-j+1));
      ans=min(ans,i*(m-j)+j+1);
      ans=min(ans,(i+1)*(m-j+1)+1);
      ans=min(ans,(n-i)*j+m-j+1);
      ans=min(ans,(n-i+1)*j+m-j+2);
      ans=min(ans,i*j+j+1);
      ans=min(ans,(i+1)*j+1);
      ans=min(ans,(n-i)*j+n-i+1);
      ans=min(ans,(n-i+1)*j+n-i+2);
      ans=min(ans,j+1);
      ans=min(ans,2);
      ans=min(ans,m-j);
      ans=min(ans,m-j+1);
      ans=min(ans,i*j+(n-i)*(m-j)+1);
      ans=min(ans,(i+1)*j+(n-i-1)*(m-j)+2);
      ans=min(ans,(n-i)*j+(m-j)*(i+1)+1);
      ans=min(ans,(n-i)*(j+1)+(m-j-1)*(i+1)+1);
      ans=min(ans,(i+1)*(j+1));
      ans=min(ans,(n-i+1)*(m-j+1));
      ans=min(ans,(n-i)*(m-j+1));
      ans=min(ans,(m-j)*(i+1));
    }
  }
  cout<<ans<<endl;
  return 0;
}
