from collections import deque
n = int(input())
a = deque(map(int, input().split()))
k = int(input())
b = list(map(int, input().split()))
if set(b) < set(a):
    print('NO')
else:
    eat = []
    for i in range(n-k):
        if a[0] < a[1]:
            a.rotate(-1)
            eat.append('R')
        else:
            eat.append('L')
            a.popleft()
    for j, w in enumerate(a):
        if b[j] != w:
            print('NO')
            exit()
    print('YES')
    for e in eat:
        print(len(a), e)
