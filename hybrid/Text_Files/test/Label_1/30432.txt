import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();
        long[] ts = new long[n];
        long[] xs = new long[n];
        long[] ys = new long[n];
        for (int i = 0; i < n; i++) {
            ts[i] = scanner.nextLong();
            xs[i] = scanner.nextLong();
            ys[i] = scanner.nextLong();
        }
        scanner.close();
        long[] ds = new long[n];
        for (int i = 0; i < n - 1; i++) {
            ds[i] = ts[i + 1] - ts[i];
        }
        long[] dxds = new long[n - 1];
        long[] dyds = new long[n - 1];
        for (int i = 0; i < n - 1; i++) {
            dxds[i] = xs[i + 1] - xs[i];
            dyds[i] = ys[i + 1] - ys[i];
        }
        boolean[][] can = new boolean[l][4];
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < l; j++) {
                if (Math.abs(dxds[i] + (dyds[i] < 0 ? -1 : 1) * ds[i]) % 2 == 0) {
                    can[j][0] = true;
                }
                if (Math.abs(dxds[i] + (dyds[i] > 0 ? -1 : 1) * ds[i]) % 2 == 0) {
                    can[j][1] = true;
                }
                if (Math.abs(dyds[i] + (dxds[i] < 0 ? -1 : 1) * ds[i]) % 2 == 0) {
                    can[j][2] = true;
                }
                if (Math.abs(dyds[i] + (dxds[i] > 0 ? -1 : 1) * ds[i]) % 2 == 0) {
                    can[j][3] = true;
                }
            }
        }
        boolean[] ans = new boolean[l];
        for (int i = 0; i < l; i++) {
            ans[i] = true;
            for (int j = 0; j < 4; j++) {
                ans[i] &= can[i][j];
            }
        }
        boolean ok = false;
        for (int i = 0; i < l; i++) {
            if (ans[i]) {
                ok = true;
                for (int j = 0; j < l; j++) {
                    System.out.print((i == j ? "R" : "L"));
                }
                System.out.println();
                break;
            }
        }
        if (!ok) {
            System.out.println("NO");
        }
    }
}
