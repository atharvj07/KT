import java.util.Scanner;

public class Main {
    private static int S;
    private static int T;
    private static int U;
    private static int N;
    private static int O;
    private static int D;
    private static int[] e;
    private static int[] c;
    private static int[] h;
    private static int[] a;
    private static long[][] dp;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        S = sc.nextInt();
        T = sc.nextInt();
        U = sc.nextInt();
        N = sc.nextInt();
        O = sc.nextInt();
        D = sc.nextInt();
        e = new int[T];
        c = new int[T];
        h = new int[N];
        a = new int[N];
        for (int i = 0; i < T; i++) {
            e[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        for (int i = 0; i < N; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
        }
        dp = new long[D + 1][S + 1];
        for (int i = 0; i <= D; i++) {
            for (int j = 0; j <= S; j++) {
                dp[i][j] = -1;
            }
        }
        long ans = solve(0, S);
        if (ans == -1) {
            System.out.println(-1);
        } else {
            System.out.println(ans);
        }
    }

    private static long solve(int day, int strength) {
        if (day == D) {
            return 0;
        }
        if (dp[day][strength] != -1) {
            return dp[day][strength];
        }
        long max = 0;
        for (int i = 0; i < T; i++) {
            if (strength - e[i] >= 0) {
                int remain = strength - e[i];
                max = Math.max(max, solve(day + 1, remain + Math.min(O, S)) + getHappiness(i, remain + Math.min(O, S)));
            }
        }
        dp[day][strength] = max;
        return dp[day][strength];
    }

    private static long getHappiness(int index, int remain) {
        long max = 0;
        for (int i = 0; i < N; i++) {
            if (remain - a[i] >= 0) {
                max = Math.max(max, h[i] + getHappiness(index, remain - a[i]));
            }
        }
        return max;
    }
}
