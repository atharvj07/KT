#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    int minDiff = a[1] - a[0];
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (a[i] - a[i - 1] == minDiff) ans++;
    }
    cout << minDiff << ' ' << ans << endl;
    return 0;
}
