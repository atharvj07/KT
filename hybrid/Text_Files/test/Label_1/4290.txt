#include <iostream>

#define MOD 998244353
#define ll long long

using namespace std;

int main() {
  int n;
  cin >> n;

  ll a[n+1];
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }

  ll dp[n+1];
  dp[n] = 1;

  for (int i = n-1; i >= 1; i--) {
    dp[i] = (dp[i+1] + dp[i+1]*a[i] % MOD) % MOD;
  }

  ll ans = 0;
  for (int i = 1; i <= n; i++) {
    ans = (ans + dp[i]) % MOD;
  }

  ans = (ans * (1LL << (n - 1))) % MOD;
  cout << ans << endl;

  return 0;
}
