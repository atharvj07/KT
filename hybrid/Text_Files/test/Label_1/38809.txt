import math
import sys

sys.setrecursionlimit(1500000)

def get_max_explosions(n, m, a, b, s):
    dp = [[-1] * (m + 1) for _ in range(n + 1)]

    def rec(pos, time, rem):
        if pos < 1 or pos > n or pos == b:
            return -math.inf

        if time >= 1e9:
            return 0

        if dp[pos][rem] != -1:
            return dp[pos][rem]

        dp[pos][rem] = 0

        # Drop a firecracker
        if rem > 0:
            dp[pos][rem] = max(dp[pos][rem], rec(pos, time + 1, rem - 1) + 1)

        # Move left
        dp[pos][rem] = max(dp[pos][rem], rec(pos - 1, time + 1, rem))

        # Move right
        dp[pos][rem] = max(dp[pos][rem], rec(pos + 1, time + 1, rem))

        # Stay in place
        if time + s[rem - 1] <= 1e9:
            dp[pos][rem] = max(dp[pos][rem], rec(pos, time + s[rem - 1], rem))

        return dp[pos][rem]

    return rec(a, 0, m)

t = int(input())

for _ in range(t):
    n, m, a, b = map(int, input().split())
    s = list(map(int, input().split()))
    print(get_max_explosions(n, m, a, b, s))
