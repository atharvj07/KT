import sys
input = sys.stdin.readline

q = int(input())
arr = []
for i in range(q):
    cmd, *args = map(int, input().split())
    if cmd == 1:
        arr.append(args[0])
    else:
        x, y = args
        for i in range(len(arr)):
            if arr[i] == x:
                arr[i] = y
print(*arr)
