#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, x;
  cin >> n >> k >> x;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  sort(a.begin(), a.end());
  for (int i = 0; i < n; i += 2) {
    if (k & 1) { a[i] ^= x; }
    k >>= 1;
  }
  cout << *max_element(a.begin(), a.end()) << ' '
       << *min_element(a.begin(), a.end()) << '\n';
  return 0;
}
