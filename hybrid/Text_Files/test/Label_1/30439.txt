import java.util.*;
import java.io.*;

public class Main {
  static final long MAX = 1000000000000L;
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int k = in.nextInt();
    int n = in.nextInt();
    long[] xa = new long[k];
    long[] ya = new long[k];
    for(int i = 0; i < k; i++) {
      xa[i] = in.nextInt();
      ya[i] = in.nextInt();
    }
    long[] xb = new long[n];
    long[] yb = new long[n];
    for(int i = 0; i < n; i++) {
      xb[i] = in.nextInt();
      yb[i] = in.nextInt();
    }
    int ans = 0;
    for(int i = 0; i < k; i++) {
      int ni = 0;
      for(int j = 0; j < n; j++) {
        long dx = xb[j] - xa[i];
        long dy = yb[j] - ya[i];
        long d = dx*dx + dy*dy;
        if(d <= MAX) {
          ni++;
        }
      }
      ans = Math.max(ans, ni);
    }
    for(int i = 0; i < n; i++) {
      int ni = 0;
      for(int j = 0; j < k; j++) {
        long dx = xb[i] - xa[j];
        long dy = yb[i] - ya[j];
        long d = dx*dx + dy*dy;
        if(d <= MAX) {
          ni++;
        }
      }
      ans = Math.max(ans, ni);
    }
    System.out.println(ans);
  }
}
