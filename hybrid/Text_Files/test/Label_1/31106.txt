import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) throws java.lang.Exception {
    Scanner in = new Scanner(System.in);
    int t = in.nextInt();
    while (t-- > 0) {
      int n = in.nextInt();
      long k = in.nextLong();
      int[] h = new int[n];
      for (int i = 0; i < n; i++)
        h[i] = in.nextInt();

      int i = 0;
      while (k > 0 && i + 1 < n) {
        if (h[i + 1] <= h[i]) {
          k--;
          i++;
          h[i]++;
        } else
          i++;
      }
      System.out.println(k == 0 ? i + 1 : -1);
    }
  }
}
