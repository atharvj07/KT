import java.util.*;
import java.lang.Math;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    int[] w = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    for (int i = 0; i < n; i++) {
      w[i] = sc.nextInt();
    }
    int[][] dp = new int[n][2];
    Arrays.fill(dp[0], 0);
    for (int i = 1; i < n; i++) {
      dp[i][0] = Integer.MAX_VALUE;
      dp[i][1] = Integer.MAX_VALUE;
    }
    for (int i = 2; i < n; i++) {
      if (a[i - 1] == 0) {
        dp[i][0] = Math.min(dp[i - 1][0], dp[i - 1][1] + w[i - 1]);
      } else {
        dp[i][0] = Math.min(dp[i - 1][0] + w[i - 1], dp[i - 1][1]);
      }
      if (a[i - 2] == 0) {
        dp[i][1] = Math.min(dp[i - 2][0], dp[i - 2][1] + w[i - 2]);
      } else {
        dp[i][1] = Math.min(dp[i - 2][0] + w[i - 2], dp[i - 2][1]);
      }
    }
    int ans = Math.min(dp[n - 1][0], dp[n - 1][1]);
    System.out.println(ans);
  }
}
