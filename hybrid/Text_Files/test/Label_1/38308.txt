from collections import deque
n,m=map(int,input().split())
g=[list(input())for i in range(n)]
vis=[[False for j in range(m)]for i in range(n)]
dx=[1,-1,0,0]
dy=[0,0,1,-1]
def bfs(x,y,p):
    q=deque()
    q.append((x,y))
    vis[x][y]=True
    while q:
        x,y=q.popleft()
        for i in range(4):
            nx,ny=x+dx[i],y+dy[i]
            if nx<0 or nx>=n or ny<0 or ny>=m:continue
            if vis[nx][ny]:continue
            if g[nx][ny]=='#' or g[nx][ny]==p:continue
            if g[nx][ny]=='.':cost[nx][ny]+=1
            vis[nx][ny]=True
            q.append((nx,ny))
cost=[[0 for j in range(m)]for i in range(n)]
res=0
for i in range(n):
    for j in range(m):
        if g[i][j]!='#' and not vis[i][j]:
            bfs(i,j,g[i][j])
            res+=max(cost[i][j],0)
print(res)
