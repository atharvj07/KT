import java.util.Scanner;

public class PetyaAndPermutation {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }

        int minSwaps = Integer.MAX_VALUE;
        int ways = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                int[] tempPermutation = permutation.clone();
                swap(tempPermutation, i, j);
                int swaps = countSwaps(tempPermutation);
                if (swaps < minSwaps) {
                    minSwaps = swaps;
                    ways = 1;
                } else if (swaps == minSwaps) {
                    ways++;
                }
            }
        }

        System.out.println(minSwaps + " " + ways);
    }

    private static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    private static int countSwaps(int[] permutation) {
        int swaps = 0;
        for (int i = 1; i < permutation.length; i++) {
            int j = i;
            while (j > 0 && permutation[j] < permutation[j - 1]) {
                swap(permutation, j, j - 1);
                j--;
                swaps++;
            }
        }
        return swaps;
    }
}
