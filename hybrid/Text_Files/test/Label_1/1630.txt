#include <bits/stdc++.h>
using namespace std;

const int N = 1e6;
int n, k;
int g[5][55], pos[55][2];

int main() {
    cin >> n >> k;
    for (int t = 1; t <= 2; t++) for (int i = 1; i <= n; i++) {
        int x; cin >> x;
        if (x) pos[x][t] = i;
    }
    for (int t = 3; t <= 4; t++) for (int i = 1; i <= n; i++) {
        int x; cin >> x;
        if (x) pos[x][t] = i;
    }
    for (int i = 1; i <= k; i++) {
        swap(pos[i][2], pos[i][3]);
        if (pos[i][2] > pos[i][3])
            swap(pos[i][2], pos[i][3]);
    }
    for (int i = 1; i <= k; i++) {
        for (int j = pos[i][2]; j <= pos[i][3]; j++)
            g[2][j] = g[3][j] = i;
    }
    int ans = 0;
    for (int i = 1; i <= k; i++) {
        int t = (pos[i][2] + pos[i][3] + 1) % 2 + 2;
        for (int j = pos[i][2] + 1; j < pos[i][3]; j++) {
            g[t][j] = 0;
        }
        if (g[t][pos[i][2]] != i) {
            swap(g[t][pos[i][2]], g[t][pos[i][3]]);
            ans += 2;
        }
    }
    if (ans > N) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
        for (int t = 1; t <= 4; t++) for (int i = 1; i <= n; i++)
            if (g[t][i])
                cout << g[t][i] << ' ' << t << ' ' << i << endl;
    }
    return 0;
}
