import sys

input = sys.stdin.readline

n = int(input())
color = list(map(int, input().split()))
graph = [[] for _ in range(n+1)]
for _ in range(n-1):
    a, b = map(int, input().split())
    graph[a].append(b)
    graph[b].append(a)

# DFS to calculate the number of black and white nodes in each subtree
black = [0] * (n+1)
white = [0] * (n+1)
def dfs(node, par):
    if color[node-1] == 1:
        black[node] = 1
    else:
        white[node] = 1
    for adj in graph[node]:
        if adj != par:
            dfs(adj, node)
            black[node] += black[adj]
            white[node] += white[adj]

dfs(1, 1)

# Calculate the minimum number of operations to make the entire tree the same color
ans = min(black[1], white[1])
print(ans)
