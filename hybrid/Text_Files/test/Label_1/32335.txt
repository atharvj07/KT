import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int[][] roads = new int[3][2];
        for (int i = 0; i < 3; i++) {
            roads[i][0] = scanner.nextInt();
            roads[i][1] = scanner.nextInt();
        }

        boolean[][] connected = new boolean[5][5];
        for (int[] road : roads) {
            connected[road[0]][road[1]] = true;
            connected[road[1]][road[0]] = true;
        }

        boolean[] visited = new boolean[5];
        visited[1] = true;

        int count = 1;
        for (int i = 1; i <= 4; i++) {
            if (!visited[i]) {
                if (!dfs(connected, visited, i, count + 1)) {
                    System.out.println("NO");
                    return;
                }
            }
            count++;
        }

        System.out.println("YES");
    }

    private static boolean dfs(boolean[][] connected, boolean[] visited, int start, int count) {
        if (count == 4) {
            return true;
        }

        for (int i = 1; i <= 4; i++) {
            if (connected[start][i] && !visited[i]) {
                visited[i] = true;
                if (dfs(connected, visited, i, count + 1)) {
                    return true;
                }
                visited[i] = false;
            }
        }

        return false;
    }
}
