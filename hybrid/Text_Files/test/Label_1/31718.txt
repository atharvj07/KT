import java.util.Scanner;

public class Main {
    private static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long result = 1;
        for (long i = 1; i <= n; i++) {
            result = (result * i) % MOD;
        }
        System.out.println(result);
    }
}
