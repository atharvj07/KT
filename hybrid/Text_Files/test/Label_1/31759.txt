import java.util.Scanner;

public class DominoCarpet {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        char[][] carpet = new char[n * 3][m * 3];
        for (int i = 0; i < n * 3; i++) {
            String line = scanner.next();
            for (int j = 0; j < m * 3; j++) {
                carpet[i][j] = line.charAt(j);
            }
        }

        long[][][] dp = new long[n + 1][m + 1][2]; // dp[i][j][k] = number of ways to cover the first i rows and j columns with k dominoes placed vertically (k = 0) or horizontally (k = 1)
        dp[0][0][0] = 1;
        dp[0][0][1] = 1;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (carpet[i * 3][j * 3] == 'O') {
                    dp[i + 1][j][1] = (dp[i + 1][j][1] + dp[i][j][0] + dp[i][j][1]) % MOD;
                    dp[i][j + 1][0] = (dp[i][j + 1][0] + dp[i][j][0] + dp[i][j][1]) % MOD;
                } else {
                    dp[i + 1][j][1] = (dp[i + 1][j][1] + dp[i][j][0]) % MOD;
                    dp[i][j + 1][0] = (dp[i][j + 1][0] + dp[i][j][1]) % MOD;
                }
            }
        }

        long result = dp[n][m][0] + dp[n][m][1];
        System.out.println(result % MOD);
    }
}
