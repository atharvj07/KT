import sys
input = sys.stdin.readline

n,m=map(int,input().split())
edges=[list(map(int,input().split())) for i in range(m)]
s,b,k,h=map(int,input().split())
spaceships=[list(map(int,input().split())) for i in range(s)]
bases=[list(map(int,input().split())) for i in range(b)]

graph=[[] for i in range(n+1)]
for u,v in edges:
    graph[u].append(v)
    graph[v].append(u)

ans=1e18
for mask in range(1<<b):
    taken=[False for i in range(b)]
    cost=0
    for i in range(b):
        if (mask>>i)&1:
            cost+=h
        else:
            taken[i]=True
    
    taken_spaceships=[False for i in range(s)]
    for i in range(b):
        if taken[i]:
            dist=[-1 for i in range(n+1)]
            q=[(bases[i][0],0)]
            while q:
                v,d=q.pop(0)
                if dist[v]!=-1: continue
                dist[v]=d
                for u in graph[v]:
                    if dist[u]==-1:
                        q.append((u,d+1))
            
            for j in range(s):
                if not taken_spaceships[j] and spaceships[j][0]!=bases[i][0] and spaceships[j][1]>=bases[i][1] and spaceships[j][2]>=dist[spaceships[j][0]]:
                    taken_spaceships[j]=True
                    cost+=k
                    break
    
    ans=min(ans,cost)

print(ans)
