import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt(), t = sc.nextInt(), r = sc.nextInt();
        int[] wi = new int[m];
        for (int i = 0; i < m; i++) wi[i] = sc.nextInt();
        int ans = 0, cnt = 0, cur = 0;
        for (int i = 0; i < m; i++) {
            if (wi[i] - cur < t) cnt++;
            else {
                ans += cnt / r;
                cnt = 1;
            }
            cur = wi[i];
        }
        ans += cnt / r;
        System.out.println(ans > 0 ? ans : -1);
    }
}
