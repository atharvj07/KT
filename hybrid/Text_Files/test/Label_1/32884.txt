import java.util.Deque;
import java.util.ArrayDeque;
import java.util.Scanner;
import java.util.Set;
import java.util.HashSet;

class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        Deque<Integer> latest = new ArrayDeque<>();
        Set<Integer> active = new HashSet<>();

        for (int i = 0; i < n; i++) {
            int id = in.nextInt();

            if (active.contains(id)) {
                latest.remove(id);
                latest.addFirst(id);
            } else if (latest.size() < k) {
                latest.addFirst(id);
                active.add(id);
            } else {
                int remove = latest.removeLast();
                active.remove(remove);

                latest.addFirst(id);
                active.add(id);
            }
        }

        System.out.println(latest.size());
        for (int id : latest)
            System.out.print(id + " ");
        System.out.println();
    }
}
