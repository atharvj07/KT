#include <bits/stdc++.h>
using namespace std;
#define ll long long

const ll mod = 1e9 + 7;
int main() {
    ll n, m;
    cin >> n >> m;
    vector<ll> a(n), cnt(m + 1);
    for (auto &it : a) { cin >> it, cnt[it]++; }
    ll ans = 0;
    for (int i = 1; i <= m; i++) ans += cnt[i] * (cnt[i] - 1) / 2;
    ans *= m;
    cout << ans << endl;
    return 0;
}
