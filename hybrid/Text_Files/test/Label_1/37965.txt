from math import gcd

def can_make_non_decreasing(n, arr):
    min_element = min(arr)
    for i in range(n):
        if arr[i] % min_element != 0 and arr[i] != sorted(arr)[i]:
            return "NO"
    return "YES"

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    arr = list(map(int, input().strip().split()))
    print(can_make_non_decreasing(n, arr))
