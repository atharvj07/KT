def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

n = int(input().strip())
colors = [0] * n
color = 1

for i in range(2, n + 2):
    if is_prime(i):
        colors[i-2] = color
        color += 1

max_color = max(colors)
for i in range(n):
    if colors[i] == 0:
        colors[i] = max_color

print(max_color)
print(' '.join(map(str, colors)))
