#include <bits/stdc++.h>

const int N = 1e6 + 10;

int n;
int a[N];
int ans;
int vis[N];
int GCD(int x, int y) { return y ? GCD(y, x%y) : x; }

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; ++i) {
        scanf("%d", a + i);
        if (a[i]) ans = GCD(ans, a[i]);
    }
    if (!ans) {
        puts("-1");
        return 0;
    }
    bool found = false;
    for (int i = 0; i < n; ++i) {
        if (a[i] % ans == 0) {
            vis[i] = 1;
            found = true;
            break;
        }
    }
    if (!found) {
        puts("-1");
        return 0;
    }
    for (int i = 0; i < n; ++i) {
        if (!vis[i] && a[i]) {
            printf("-1\n");
            return 0;
        }
        if (i && !vis[i - 1] && a[i - 1]) {
            vis[i] = 1;
            ++ans;
        }
    }
    printf("%d\n", ans - 1);
    return 0;
}
