n = int(input())
A = list(map(int, input().split()))
L = [1] * n
R = [n] * n
st = []
for i in range(n):
    while st and A[st[-1]-1] >= A[i]:
        st.pop()
    if st:
        L[i] = st[-1]
    st.append(i+1)
st = []
for i in range(n-1,-1,-1):
    while st and A[st[-1]-1] > A[i]:
        st.pop()
    if st:
        R[i] = st[-1]
    st.append(i+1)
ans = []
for i in range(n):
    ans.append(max(x for x in A[L[i]-1:R[i]]))
print(*ans)
