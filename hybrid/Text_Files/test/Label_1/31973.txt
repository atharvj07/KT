import java.util.*;

public class Main {
    static int n, m, k;
    static int[] deg;
    static int[] used;
    static int[] edge;
    static ArrayList<Integer>[] graph;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = (n + m) / 2;
        deg = new int[n + 1];
        used = new int[m + 1];
        edge = new int[m + 1];
        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; ++i) {
            graph[i] = new ArrayList<Integer>();
        }
        for (int i = 1; i <= m; ++i) {
            int u = in.nextInt();
            int v = in.nextInt();
            deg[u] += 1;
            deg[v] += 1;
            graph[u].add(i);
            graph[v].add(i);
        }
        int cnt = 0;
        for (int i = 1; i <= n; ++i) {
            cnt += deg[i] / 2;
        }
        if (cnt < k) {
            System.out.println(-1);
        } else {
            k = cnt;
            System.out.println(k);
            for (int i = 1; i <= n; ++i) {
                for (int id : graph[i]) {
                    if (used[id] == 0 && k > 0) {
                        System.out.println(id + " " + edge[id]);
                        k -= 1;
                        used[id] = used[edge[id]] = 1;
                    }
                }
            }
        }
    }
}
