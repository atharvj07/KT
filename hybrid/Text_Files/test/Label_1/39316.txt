def get_A(B):
    m, n = len(B), len(B[0])
    A = [['0' for _ in range(n)] for _ in range(m)]

    for i in range(m):
        for j in range(n):
            if B[i][j] == '1':
                A[i][j] = '1'
                for k in range(i):
                    A[k][j] = '1'
                for k in range(j):
                    A[i][k] = '1'

    return A

m, n = map(int, input().split())
B = [input().split() for _ in range(m)]
A = get_A(B)
print('NO' if any('1' in row and '0' in row for row in A) else 'YES')
if 'YES':
    for row in A:
        print(' '.join(row))
