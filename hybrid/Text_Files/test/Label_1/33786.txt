import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);

        int[] w1 = new int[4], b1 = new int[4], b2 = new int[4];

        for (int i = 0; i < 4; i++) {
            w1[i] = in.nextInt();
        }
        for (int i = 0; i < 4; i++) {
            b1[i] = in.nextInt();
        }
        for (int i = 0; i < 4; i++) {
            b2[i] = in.nextInt();
        }

        if ((w1[2] < b1[0] || w1[0] > b1[2] || w1[3] < b1[1] || w1[1] > b1[3]) &&
                (w1[2] < b2[0] || w1[0] > b2[2] || w1[3] < b2[1] || w1[1] > b2[3])) {
            System.out.println("YES");
        } else {
            System.out.println("NO");
        }
    }
}
