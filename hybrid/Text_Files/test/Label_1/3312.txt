#include<iostream>
#include<vector>
using namespace std;

const int MAXN = 105;
const int MAXM = 105;
int n, m, a[MAXN], b[MAXN][MAXN];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    cin >> m;
    while (m--) {
        int x, y; cin >> x >> y;
        for (int i = y; i < a[x]; i++) {
            b[x-1][i] += 1;
            b[x+1][i+1] += 1;
        }
        a[x] = y;
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= a[i]; j++) a[i] += b[i][j];
        cout << a[i] << '\n';
    }
}
