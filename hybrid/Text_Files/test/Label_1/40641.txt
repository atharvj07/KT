import sys
input = sys.stdin.readline

n, s, m, k = map(int, input().split())
a = list(map(int, input().split()))
seg = []
for _ in range(s):
    l, r = map(int, input().split())
    seg.append((l - 1, r - 1))

mn = -1
for mask in range(1 << s):
    cnt = 0
    cur = []
    for i in range(s):
        if mask >> i & 1:
            l, r = seg[i]
            cnt += r - l + 1
            for j in range(l, r + 1):
                cur.append(a[j])
    cur.sort()
    if cnt >= k and (mn == -1 or cur[k - 1] < mn):
        mn = cur[k - 1]

print(mn)
