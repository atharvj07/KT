from collections import defaultdict, deque
from sys import stdin

n, s = map(int, stdin.readline().split())
edges = defaultdict(list)
d = [0] * n

for _ in range(n - 1):
    a, b = map(lambda x: int(x) - 1, stdin.readline().split())
    edges[a].append(b)
    edges[b].append(a)

ans = 0

q = deque()
q.append((0, -1))

while q:
    v, p = q.popleft()
    for u in edges[v]:
        if u == p:
            continue

        q.append((u, v))
        d[u] = d[v] + 1

x = d.index(max(d))
ans = d[x]

q = deque()
q.append((x, -1))

while q:
    v, p = q.popleft()
    for u in edges[v]:
        if u == p:
            continue

        q.append((u, v))
        d[u] = d[v] + 1

y = d.index(max(d))
ans += d[y]

print(s / ans)
