import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static int n, m;
    static int[] a = new int[300005], ans = new int[300005];
    static int[][] op = new int[300005][3];

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        int mx = 0;
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int t = Integer.parseInt(line[0]);
            if (t == 1) {
                op[i][0] = t;
                op[i][1] = Integer.parseInt(line[1]);
                op[i][2] = Integer.parseInt(line[2]);
                a[Integer.parseInt(line[1]) - 1] = Math.max(a[Integer.parseInt(line[1]) - 1], Integer.parseInt(line[3]));
                mx = Math.max(mx, a[Integer.parseInt(line[1]) - 1]);
            } else {
                op[i][0] = t;
                op[i][1] = Integer.parseInt(line[1]) - 1;
                op[i][2] = Integer.parseInt(line[2]);
            }
        }
        if (a[0] != mx) {
            System.out.println("NO");
            return;
        }
        int cur = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > cur) {
                cur = a[i];
                ans[i] = (1 << 30) - 1;
            }
        }
        for (int i = 0; i < m; i++) {
            if (op[i][0] == 1) {
                int l = op[i][1], r = op[i][2];
                for (int j = l; j <= r; j++) {
                    ans[j] = (1 << 30) - 1;
                }
            } else {
                int idx = op[i][1], val = op[i][2];
                ans[idx] = val;
            }
        }
        System.out.println("YES");
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
