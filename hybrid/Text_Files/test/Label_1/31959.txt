import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), w = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n], t = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        int[][] dp = new int[n][w + 2];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }
        System.out.println(solve(0, w, k, a, t, dp));
    }

    private static int solve(int i, int w, int k, int[] a, int[] t, int[][] dp) {
        if (i == a.length) {
            return 0;
        }
        if (dp[i][w] != -1) {
            return dp[i][w];
        }
        int ans = 0;
        if (w > 0) {
            ans = Math.max(ans, solve(i + 1, w - 1, k, a, t, dp) + a[i]);
        }
        if (k >= t[i] / 2) {
            ans = Math.max(ans, solve(i + 1, w, k - t[i], a, t, dp) + a[i]);
        }
        ans = Math.max(ans, solve(i + 1, w, k, a, t, dp));
        return dp[i][w] = ans;
    }
}
