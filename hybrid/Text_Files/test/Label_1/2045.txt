#include <iostream>
#include <string>
using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.length();
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    for (int j = i + 1; j < n; ++j) {
      bool is_palindrome = true;
      for (int k = 0; k < (j - i + 1) / 2; ++k) {
        if (s[i + k] != s[j - k]) {
          is_palindrome = false;
          break;
        }
      }
      if (!is_palindrome) {
        ans = max(ans, j - i + 1);
      }
    }
  }
  cout << ans << endl;
  return 0;
}
