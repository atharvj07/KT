import sys

m = 1000000007

x = list(map(int, sys.stdin.readline().split()))
l = [0] * (x[0] + 1)

for i in list(map(int, sys.stdin.readline().split())):
    l[i] = 1

pref = []
for i in range(x[0] + 1):
    if l[i]:
        pref.append(i)

sum = 0
for i in range(len(pref)):
    for j in range(i + 1, len(pref)):
        sum += (pref[j] - pref[i])
        sum %= m

print(sum)
