import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class HarmonySequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> key = findKey(a);

        for (int i : key) {
            System.out.print(i + " ");
        }
    }

    static List<Integer> findKey(int[] a) {
        int n = a.length;
        int[][] dp = new int[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], Integer.MAX_VALUE);
        }
        for (int i = 0; i < n; i++) {
            dp[i][i] = a[i];
        }

        for (int len = 2; len <= n; len++) {
            for (int i = 0; i <= n - len; i++) {
                int j = i + len - 1;
                for (int k = i; k < j; k++) {
                    if (gcd(dp[i][k], dp[k + 1][j]) == 1) {
                        dp[i][j] = Math.min(dp[i][j], dp[i][k] + dp[k + 1][j]);
                    }
                }
            }
        }

        return reconstructKey(dp, a);
    }

    static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }

    static List<Integer> reconstructKey(int[][] dp, int[] a) {
        int n = a.length;
        List<Integer> key = new ArrayList<>();
        int i = 0;
        int j = n - 1;
        while (i <= j) {
            if (i == j) {
                key.add(a[i]);
                break;
            }
            int minSum = Integer.MAX_VALUE;
            int minK = -1;
            for (int k = i; k < j; k++) {
                if (gcd(dp[i][k], dp[k + 1][j]) == 1 && dp[i][k] + dp[k + 1][j] < minSum) {
                    minSum = dp[i][k] + dp[k + 1][j];
                    minK = k;
                }
            }
            key.add(dp[i][minK]);
            i = minK + 1;
        }
        return key;
    }
}
