#include <vector>
#include <iostream>
#include <algorithm>

const unsigned long long MOD = 998244353;

int main() {
  int n;
  std::cin >> n;

  int first_max = 1, second_max = 1;
  int first_min = n, second_min = n;
  for (int i = 0; i < n; i++) {
    int a, b;
    std::cin >> a >> b;
    first_max = std::max(first_max, a);
    first_min = std::min(first_min, a);
    second_max = std::max(second_max, b);
    second_min = std::min(second_min, b);
  }

  if (first_max == first_min || second_max == second_min) {
    std::cout << 0 << std::endl;
    return 0;
  }

  if (first_min > second_max || first_max < second_min) {
    std::cout << n << std::endl;
    return 0;
  }

  unsigned long long ans = (first_max - second_min) % MOD;
  ans *= (second_max - first_min) % MOD;
  ans %= MOD;
  std::cout << ans << std::endl;

  return 0;
}
