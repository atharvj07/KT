# Function to check if it is possible to find an
# array b such that MEX(b[1..i]) = a[i]
def isPossible(a, n):
    # Initialize the frequency array
    freq = [0] * (n + 1)

    # Traverse the array a and update the frequency array
    for i in range(0, n):
        freq[a[i]] += 1

    # Initialize the MEX
    mex = 0

    # Traverse the array b and update the MEX
    for i in range(0, n):
        # If the current MEX is not present
        # in the frequency array
        if freq[mex] == 0:
            return mex
        # If the current MEX is present in the
        # frequency array
        else:
            freq[mex] -= 1
            mex += 1

    # If all the elements of the frequency array
    # are zero then return -1
    return -1

# Function to find an array b such that MEX(b[1..i]) = a[i]
def findArray(a, n):
    # Initialize the array b
    b = [0] * n

    # Initialize the frequency array
    freq = [0] * (n + 1)

    # Traverse the array a and update the frequency array
    for i in range(0, n):
        freq[a[i]] += 1

    # Initialize the MEX
    mex = 0

    # Traverse the array b and update the MEX
    for i in range(0, n):
        # If the current MEX is not present
        # in the frequency array
        if freq[mex] == 0:
            b[i] = mex
            freq[mex] += 1
        # If the current MEX is present in the
        # frequency array
        else:
            freq[mex] -= 1
            mex += 1
            b[i] = mex
            freq[mex] += 1

    # Return the array b
    return b

# Driver code
n = int(input())
a = list(map(int, input().split()))

# Check if it is possible to find an array b such that MEX(b[1..i]) = a[i]
if isPossible(a, n) == -1:
    print(-1)
else:
    # Find an array b such that MEX(b[1..i]) = a[i]
    b = findArray(a, n)
    # Print the array b
    print(*b)
