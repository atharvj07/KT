import java.util.*;
import java.lang.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        long a = scanner.nextLong();
        boolean found = false;
        long ans = -1;
        long prefix = a > 0 ? 0 : 1;
        while (!found) {
            if (a + prefix > 0) {
                long x = a + prefix;
                if (x % 10 == 8 || x / 10 == 0) {
                    ans = prefix;
                    found = true;
                }
            }
            prefix++;
        }
        System.out.println(ans);
    }
}
