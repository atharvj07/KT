import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int ans = 0;
        char cur = s.charAt(0);
        for (int i = 1; i < n; i++) {
            if (s.charAt(i) == cur) ans++;
            cur = cur == 'r' ? 'b' : 'r';
        }
        System.out.println(ans);
    }
}
