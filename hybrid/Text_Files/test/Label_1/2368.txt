#include<bits/stdc++.h>
using namespace std;
#define LL long long
int main(){
    LL x1,y1,x2,y2;
    cin>>x1>>y1>>x2>>y2;
    LL n;
    string s;
    cin>>n>>s;
    LL ans=1e9,dx,dy;
    for(int i=0;i<n;i++){
        dx=dy=0;
        for(int j=0;j<n;j++){
            char ch=s[(i+j)%n];
            if(ch=='U') dy++;
            else if(ch=='D') dy--;
            else if(ch=='L') dx--;
            else if(ch=='R') dx++;
        }
        LL cost=abs(x2-x1-dx)+abs(y2-y1-dy);
        if(cost>0) ans=min(ans,cost);
    }
    if(ans==1e9) cout<<-1<<endl;
    else cout<<ans<<endl;
    return 0;
}
