def count_subrectangles(arr):
    count = 0
    current_length = 0
    for i in arr:
        if i == 1:
            current_length += 1
            count += current_length
        else:
            current_length = 0
    return count

n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

a_subrectangles = count_subrectangles(a)
b_subrectangles = count_subrectangles(b)

total_subrectangles = 0
for i in range(1, k+1):
    if k % i == 0:
        if i <= n and k // i <= m:
            total_subrectangles += (a_subrectangles - (i-1) * (n-i+1)) * (b_subrectangles - (k//i-1) * (m-k//i+1))

print(total_subrectangles)
