from collections import defaultdict

n = int(input())
color = defaultdict(int)
for _ in range(n):
    x, y = map(int, input().split())
    color[x] ^= 1
    color[y] ^= 2

for x in color:
    if color[x] == 1:
        print(n-1, 1)
    elif color[x] == 2:
        print(n-1, 0)
    else:
        print(n, 0)
