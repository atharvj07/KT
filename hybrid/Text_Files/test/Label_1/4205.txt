#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> z_fun(string const& s) {
    int n = s.size();
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; ++i) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) ++z[i];
        if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;
    }
    return z;
}

int main() {
    string s; cin >> s;
    int n = s.size();
    vector<int> z = z_fun(s + s);
    vector<int> ans(n + 1);
    for (int i = n; i < 2 * n; ++i) ans[z[i]]++;
    for (int i = n - 1; i >= 0; --i) ans[i] += ans[i + 1];
    int last = 0;
    for (int i = n - 1; i >= 0; --i) {
        cout << ans[i] - last << " ";
        string t;
        for (int j = i, cnt = ans[i] - last; j < i + ans[i] && cnt > 0; ++j)
            if (j == i + ans[i] - 1 || j + z[j] > i + ans[i] - 1) --cnt, t += s[j];
        if (t.size() > 10) {
            for (int j = 0; j < 5; ++j) cout << t[j];
            cout << "...";
            for (int j = t.size() - 2; j < t.size(); ++j) cout << t[j];
        } else cout << t;
        cout << "\n";
        last = ans[i];
    }
    return 0;
}
