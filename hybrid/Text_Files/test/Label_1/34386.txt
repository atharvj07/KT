import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[][] rooms = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                rooms[i][j] = input.nextInt();
            }
        }
        int m = input.nextInt();
        int[][] wallpapers = new int[m][3];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < 3; j++) {
                wallpapers[i][j] = input.nextInt();
            }
        }
        int minCost = Integer.MAX_VALUE;
        for (int i = 0; i < m; i++) {
            int cost = 0;
            for (int[] room : rooms) {
                int length = room[0] * room[1];
                int width = room[2] * room[1];
                int numRollsLength = (length + wallpapers[i][0] - 1) / wallpapers[i][0];
                int numRollsWidth = (width + wallpapers[i][1] - 1) / wallpapers[i][1];
                cost += numRollsLength * numRollsWidth * wallpapers[i][2];
            }
            minCost = Math.min(minCost, cost);
        }
        System.out.println(minCost);
    }
}
