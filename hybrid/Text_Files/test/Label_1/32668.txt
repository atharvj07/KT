import java.util.*;
import java.io.*;

public class Main {

    private static int n;
    private static long[] px, py;
    private static double eps = 1e-9;

    public static double getManhattanDistance(int i, int j) {
        return Math.abs(px[i] - px[j]) + Math.abs(py[i] - py[j]);
    }

    public static double getDanielDistance(int i, int j) {
        return Math.sqrt((px[i] - px[j]) * (px[i] - px[j]) + (py[i] - py[j]) * (py[i] - py[j]));
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] l1 = br.readLine().split(" ");
        n = Integer.parseInt(l1[0]);
        px = new long[n];
        py = new long[n];
        for (int i = 0; i < n; i++) {
            String[] l2 = br.readLine().split(" ");
            px[i] = Long.parseLong(l2[0]);
            py[i] = Long.parseLong(l2[1]);
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (Math.abs(getManhattanDistance(i, j) - getDanielDistance(i, j)) < eps) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
