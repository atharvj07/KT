import java.util.Scanner;

public class Main {

    private static int n, s;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (scanner.hasNext()) {
            n = scanner.nextInt();
            s = scanner.nextInt();
            System.out.println(calc());
        }
        scanner.close();
    }

    private static int calc() {
        int[] dp = new int[s + 1];
        dp[0] = 1;
        for (int i = 1; i <= 9; i++) {
            for (int j = n * i; j <= s; j++) {
                dp[j] += dp[j - n * i];
            }
        }
        return dp[s];
    }
}
