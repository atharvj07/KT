import java.util.*;
class Main{
    static int n,m,k;
    static char[][]a;
    static boolean[][]vis;
    static int[]dx={-1,0,1,0};
    static int[]dy={0,1,0,-1};
    static int ans=0;
    public static void dfs(int x,int y){
        vis[x][y]=true;
        for(int i=0;i<4;++i){
            if(x+dx[i]>=1&&x+dx[i]<=n&&y+dy[i]>=1&&y+dy[i]<=m){
                if(a[x+dx[i]][y+dy[i]]=='1'&&!vis[x+dx[i]][y+dy[i]]){
                    dfs(x+dx[i],y+dy[i]);
                }
            }
        }
    }
    public static void main(String[]args){
        Scanner in=new Scanner(System.in);
        n=in.nextInt();
        m=in.nextInt();
        k=in.nextInt();
        a=new char[n+1][m+1];
        vis=new boolean[n+1][m+1];
        int cnt=0;
        for(int i=1;i<=n;++i){
            String s=in.next();
            for(int j=1;j<=m;++j)a[i][j]=s.charAt(j-1);
        }
        for(int i=1;i<=n;++i){
            for(int j=1;j<=m;++j){
                if(a[i][j]=='1'&&!vis[i][j]){
                    dfs(i,j);
                    cnt++;
                }
            }
        }
        if(cnt<k)System.out.println(0);
        else{
            ans=0;
            for(int i=1;i<=n;++i){
                for(int j=1;j<=m;++j){
                    if(a[i][j]=='1'&&!vis[i][j]){
                        ans++;
                        dfs(i,j);
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
