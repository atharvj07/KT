import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();
        int a = input.nextInt();
        int[] colors = new int[n];
        int best = -1;

        for (int i = 0; i < n; i++) {
            int c = input.nextInt();
            colors[i] = c;

            if (c == a) {
                best = -1;
            } else if (c != a && (best == -1 || c < best)) {
                best = c;
            }
        }

        System.out.println(best);
    }
}
