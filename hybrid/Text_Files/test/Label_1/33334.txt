import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        try {
            int n = Integer.parseInt(reader.readLine());
            Map<String, String> map = new HashMap<>();
            List<String> list = new ArrayList<>();
            for (int i = 1; i <= n; i++) {
                String[] parts = reader.readLine().split(" ");
                int type = Integer.parseInt(parts[0]);
                String suffix = "";
                if (parts.length > 1) {
                    suffix = parts[1];
                }
                if (type == 1) {
                    map.put(String.valueOf(i), suffix);
                    list.add(suffix);
                } else {
                    String name = map.get(String.valueOf(type));
                    map.put(String.valueOf(i), name + suffix);
                    list.add(name + suffix);
                }
            }
            int m = Integer.parseInt(reader.readLine());
            for (int i = 0; i < m; i++) {
                String[] parts = reader.readLine().split(" ");
                int song = Integer.parseInt(parts[0]);
                String pattern = parts[1];
                String name = map.get(String.valueOf(song));
                int count = 0;
                int index = name.indexOf(pattern);
                while (index >= 0) {
                    count++;
                    index = name.indexOf(pattern, index + 1);
                }
                System.out.println(count);
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
