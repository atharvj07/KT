import sys

n = int(input())
a = list(int(i) for i in input().split())

mod=998244353

dp=[[0]*201 for i in range(n+1)]

dp[n][0]=1

for i in range(n-1,-1,-1):
    for prv in range(201):
        for x in range(1,201):
            if(x<=prv or (a[i]!=-1 and a[i]!=x)):
                continue
            dp[i][x] = (dp[i][x] + dp[i+1][prv])%mod;

print(dp[0][0])
