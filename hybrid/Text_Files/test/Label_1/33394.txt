import java.io.*;
import java.util.*;

class Main {
    static final long MOD = 1000000007;
    static HashMap<Integer, ArrayList<Integer>> m = new HashMap<>();
    static boolean[] v = new boolean[1000];

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String line = in.readLine();
        int n = Integer.parseInt(line.split(" ")[0]);
        int k = Integer.parseInt(line.split(" ")[1]);
        for (int i = 0; i < k; i++) {
            line = in.readLine();
            int s = Integer.parseInt(line.split(" ")[0]) - 1;
            int d = Integer.parseInt(line.split(" ")[1]) - 1;
            if (!m.containsKey(s)) m.put(s, new ArrayList<>());
            m.get(s).add(d);
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (v[i]) continue;
            Stack<Integer> s = new Stack<>();
            Set<Integer> b = new HashSet<>();
            s.push(i);
            v[i] = true;
            while (!s.empty()) {
                int c = s.pop();
                b.add(c);
                if (m.containsKey(c)) {
                    ArrayList<Integer> l = m.get(c);
                    for (int j : l) {
                        if (!v[j]) {
                            v[j] = true;
                            s.push(j);
                        }
                    }
                }
            }
            long cn = 1;
            for (int j : b) {
                cn = cn * 2 % MOD;
            }
            ans = (ans + cn) % MOD;
        }
        System.out.println(ans);
    }
}
