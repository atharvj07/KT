import java.util.*;
import java.math.BigDecimal;
class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        while(sc.hasNextInt()){
            int n = sc.nextInt();
            if(n == 0) break;
            Map<Integer, BigDecimal> map = new HashMap<>();
            for(int i = 0; i < n; i++){
                int id = sc.nextInt();
                long p = sc.nextInt();
                long q = sc.nextInt();
                BigDecimal price = new BigDecimal(p);
                BigDecimal amount = new BigDecimal(q);
                map.put(id, map.getOrDefault(id, BigDecimal.ZERO).add(price.multiply(amount)));
            }
            boolean flag = false;
            for(Map.Entry<Integer, BigDecimal> e: map.entrySet()){
                if(e.getValue().compareTo(new BigDecimal("1000000")) >= 0){
                    System.out.println(e.getKey());
                    flag = true;
                }
            }
            if(!flag) System.out.println("NA");
        }
    }
}
