import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        int m = 1;
        for (int i = 0; i < n; i++) {
            m = Math.max(m, a[i] + i);
        }

        System.out.println(m);
    }
}
