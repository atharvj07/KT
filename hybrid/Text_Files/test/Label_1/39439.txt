n, c, d = map(int, input().split())
coins = []
diamonds = []

for _ in range(n):
    b, p, m = input().split()
    b, p = int(b), int(p)
    if m == 'C':
        if p <= c:
            coins.append((b, p))
    else:
        if p <= d:
            diamonds.append((b, p))

coins.sort(reverse=True)
diamonds.sort(reverse=True)

max_beauty = 0

if len(coins) > 1:
    max_beauty = max(max_beauty, coins[0][0] + coins[1][0])
if len(diamonds) > 1:
    max_beauty = max(max_beauty, diamonds[0][0] + diamonds[1][0])
if coins and diamonds:
    max_beauty = max(max_beauty, coins[0][0] + diamonds[0][0])

print(max_beauty)
