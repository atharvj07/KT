import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        BigInteger p = sc.nextBigInteger();
        int k = sc.nextInt();
        if (p.mod(BigInteger.valueOf(k)).equals(BigInteger.ZERO)) {
            System.out.println(-1);
            return;
        }
        int d = p.toString(k).length() - 1;
        BigInteger[] a = new BigInteger[d + 1];
        a[d] = p.mod(BigInteger.valueOf(k));
        p = p.divide(BigInteger.valueOf(k));
        for (int i = d - 1; i >= 0; i--) {
            a[i] = p.mod(BigInteger.valueOf(k)).add(BigInteger.valueOf(1));
            p = p.divide(BigInteger.valueOf(k));
        }
        System.out.println(d + 1);
        for (BigInteger ai : a) {
            System.out.print(ai + " ");
        }
        System.out.println();
    }
}
