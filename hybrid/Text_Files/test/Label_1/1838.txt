#include <bits/stdc++.h>
using namespace std;
int main() {
    string s;
    while (cin >> s) {
        stack<pair<char, int>> st;
        int ans = 0;
        for (char c : s) {
            if (!st.empty() && st.top().first == c)
                st.top().second++;
            else
                st.push({c, 1});
            ans = max(ans, st.top().second);
        }
        cout << ans << endl;
    }
}
