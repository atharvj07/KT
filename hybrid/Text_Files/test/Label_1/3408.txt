#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int x = 0, p = 1;
      for (int j = i; j < n; j++) {
        x += (s[j] - '0') * p;
        p *= 2;
        if (j - i + 1 == x) ans++;
      }
    }
    cout << ans << '\n';
  }
  return 0;
}
