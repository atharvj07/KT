#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int mx = 0, cur = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] == 1)
            cur++;
        else
            cur = 0;
        mx = max(mx, cur);
    }
    cur = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (a[i] == 1)
            cur++;
        else
            cur = 0;
        mx = max(mx, cur);
    }
    cout << mx << endl;
    return 0;
}
