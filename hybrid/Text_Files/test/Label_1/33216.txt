import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Set;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] NM = br.readLine().split(" ");
        int N = Integer.parseInt(NM[0]), M = Integer.parseInt(NM[1]);
        Set<String> relations = new HashSet<>();
        for (int i = 0; i < M; i++) {
            String[] relation = br.readLine().split(" ");
            relations.add(relation[0] + " " + relation[1] + " " + relation[2]);
        }

        int[] maxDiff = new int[N + 1];
        Arrays.fill(maxDiff, Integer.MIN_VALUE);
        for (String relation : relations) {
            String[] parts = relation.split(" ");
            int a = Integer.parseInt(parts[0]), b = Integer.parseInt(parts[1]), diff = Integer.parseInt(parts[2]);
            maxDiff[b] = Math.max(maxDiff[b], diff);
            if (maxDiff[a] != Integer.MIN_VALUE && maxDiff[a] + diff != maxDiff[b]) {
                System.out.println("No");
                return;
            }
        }

        System.out.println("Yes");
    }
}
