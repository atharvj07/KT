def f(n):
    v,c,g=[],n[1:],[1+n.count('>', i) - n.count('<', i) for i,j in enumerate(n) if i > 0 and j in '<>']
    for i in range(len(g)):
        if g[i] > 0:
            u = g[i:g.index(max(g[i:])) + 1]
            t = [0] * len(c)
            for j in range(len(u)): t[c.index('>', i)] = u[j]
            if all(map(lambda x: x <= 9, t)): v.extend(t)
    return ''.join(map(str, v)) if v else 'false'

import sys
sys.stdout.write(f(sys.stdin.read()))
