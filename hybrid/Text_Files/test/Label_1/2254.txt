#include <iostream>
#include <algorithm>
using namespace std;
int wire[110];
int main() {
    int n, m, x, y;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> wire[i];
    }
    cin >> m;
    for (int i = 0; i < m; i++) {
        cin >> x >> y;
        wire[x] -= y;
        wire[x - 1] += y / 2;
        wire[x + 1] += y - y / 2;
    }
    for (int i = 1; i <= n; i++) {
        cout << wire[i] << endl;
    }
    return 0;
}
