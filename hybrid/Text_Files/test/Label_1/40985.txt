m, n = map(int, input().split())
B = [list(map(int, input().split())) for _ in range(m)]
for i in range(m):
    for j in range(n):
        if i == 0 and j > 0 and B[i][j] != B[i][j - 1]:
            print('NO')
            exit()
        if j == 0 and i > 0 and B[i][j] != B[i - 1][j]:
            print('NO')
            exit()
        if i > 0 and j > 0:
            B[i][j] = (B[i][j] != B[i][j - 1]) or (B[i][j] != B[i - 1][j])
print('YES')
for b in B:
    print(*b)
