import java.util.Scanner;

public class Solution {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextLong();
        }

        int count = 0;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                long xor = 0;
                for (int k = i; k <= j; k++) {
                    xor ^= a[k];
                }
                if (xor == 0) {
                    count++;
                }
            }
        }

        System.out.println(count);
    }
}
