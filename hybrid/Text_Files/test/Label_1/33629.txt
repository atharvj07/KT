import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int l = scanner.nextInt();

        // Calculate the sum of flavors of all apples
        int sum = (n + 1) * n / 2 + l * (n - 1);

        // Calculate the flavor of the apple pie made of the remaining N-1 apples
        int target = sum / n;

        // Find the apple to eat so that the absolute difference between the flavors of the two apple pies 
        // is minimized
        int minDiff = Integer.MAX_VALUE;
        int ans = -1;
        for (int i = 1; i <= n; i++) {
            int diff = Math.abs(target - (l + i - 1));
            if (diff < minDiff) {
                minDiff = diff;
                ans = i;
            }
        }

        // Output the flavor of the apple pie made of the remaining N-1 apples
        System.out.println(sum - (l + ans - 1));
    }
}
