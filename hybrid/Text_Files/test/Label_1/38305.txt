from functools import lru_cache

@lru_cache(maxsize=None)
def is_pretty(s, l, r):
    if l > r:
        return True
    if s[l] != s[r]:
        return False
    if s[l] != '?' and s[l] != '(':
        return False
    if s[r] != '?' and s[r] != ')':
        return False
    return is_pretty(s, l+1, r-1)

s = input()
n = len(s)
ans = 0
for i in range(n):
    for j in range(i, n):
        if is_pretty(s, i, j):
            ans += 1
print(ans)
