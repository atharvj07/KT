import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        long ans = 1;
        long mod = 1000000007;

        for (int i = 1; i <= n; i++) {
            ans = (ans * i) % mod;
        }

        System.out.println(ans);
    }
}
