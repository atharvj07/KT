#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> w(n);
  for (int i = 0; i < n; i++) { cin >> w[i]; }

  int days = 0;
  while (n > 0) {
    int cur = 0;
    for (int i = 0; i < n; i++) {
      if (cur + w[i] <= k) {
        cur += w[i];
        w[i] = 0;
        n--;
      } else {
        w[i] -= k - cur;
        cur = k;
      }
    }
    days++;
  }
  cout << days << endl;
  return 0;
}
