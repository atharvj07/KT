import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int t = input.nextInt();
        while (t-- > 0) {
            int a = input.nextInt();
            int b = input.nextInt();
            int n = input.nextInt();
            if (n == 0) {
                System.out.println(a);
            } else if (n == 1) {
                System.out.println(b);
            } else {
                System.out.println(a ^ b);
            }
        }
    }
}
