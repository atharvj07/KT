import java.util.*;

public class Main {
  static class Pair implements Comparable<Pair> {
    long x;
    int pos;
    Pair(long x, int pos) {
      this.x = x;
      this.pos = pos;
    }
    public int compareTo(Pair o) {
      return Long.compare(x, o.x);
    }
  }
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    long[] b = new long[n];
    for (int i = 0; i < n; i++) {
      b[i] = in.nextLong();
    }
    Pair[] p = new Pair[n];
    for (int i = 0; i < n; i++) {
      p[i] = new Pair(b[i], i);
    }
    Comparator<Pair> cmp = Comparator.comparing(o -> o.x);
    Arrays.sort(p, cmp);
    long x = 0;
    for (int i = 0; i < n; i++) {
      if (p[i].pos == i && p[i].x != x) {
        x = p[i].x;
      } else if (p[i].x == x) {
        System.out.println("No");
        return;
      }
    }
    System.out.println("Yes");
    long[] ans = new long[n];
    for (int i = 0; i < n; i++) {
      ans[p[i].pos] = b[i];
    }
    for (int i = 0; i < n; i++) {
      System.out.print(ans[i] + " ");
    }
    System.out.println();
  }
}
