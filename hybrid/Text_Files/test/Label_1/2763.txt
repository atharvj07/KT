#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vi = vector<ll>;
int32_t main() {
  ll n, k;
  cin >> n >> k;
  vi d(n);
  for (auto &x : d) cin >> x;
  ll m = 0;
  set<ll> adj[n];
  for (ll i = 0; i < n; i++) {
    if (d[i] > k) {
      cout << -1 << '\n';
      return 0;
    }
    for (ll j = i + 1; j < n; j++) {
      if (d[i] + d[j] <= k) {
        adj[i].insert(j);
        adj[j].insert(i);
        m++;
      }
    }
  }
  cout << m << '\n';
  for (ll i = 0; i < n; i++)
    for (auto u : adj[i]) cout << i + 1 << ' ' << u + 1 << '\n';
}
