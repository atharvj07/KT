import java.util.*;
 class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] p = new int[n + 1];
        for (int i = 2; i <= n; i++) {
            p[i] = input.nextInt();
        }
        boolean[] v = new boolean[n + 1];
        boolean f = false;
        for (int i = 2; i <= n; i++) {
            if (v[i]) {
                continue;
            }
            List<Integer> ls = new ArrayList<>();
            int x = i;
            while (!v[x]) {
                v[x] = true;
                ls.add(x);
                x = p[x];
            }
            if (ls.size() > 2) {
                f = true;
            }
        }
        if (f) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
        }
    }
}
