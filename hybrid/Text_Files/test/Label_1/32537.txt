import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();
            int[] a = new int[n];

            for (int i = 0; i < n; i++) {
                a[i] = scanner.nextInt();
            }

            System.out.println(solve(n, x, a));
        }
    }

    static int solve(int n, int x, int[] a) {
        long totalSum = 0;
        for (int value : a) {
            totalSum += value;
        }

        if (totalSum < (long) n * x) {
            return 0; // Not enough money to make everyone wealthy
        } else {
            return n; // Everyone can be made wealthy by redistributing equally
        }
    }
}
