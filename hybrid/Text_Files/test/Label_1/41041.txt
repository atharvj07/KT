n=int(input())
a=list(map(int, input().split()))
freq=dict()
ansl,ansr=-1,n
for i in range(n):
    if a[i] not in freq:
        freq[a[i]]=[i,i]
    else:
        freq[a[i]][1]=i
mx=0
for i in freq:
    if freq[i][1]-freq[i][0]+1>mx:
        mx=freq[i][1]-freq[i][0]+1
        ansl,ansr=freq[i][0]+1,freq[i][1]+1
print(ansl,ansr)
