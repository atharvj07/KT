#include <iostream>
#include <string>
#include <algorithm>

int main() {
  int t;
  std::cin >> t;
  while (t--) {
    std::string keyboard, s;
    std::cin >> keyboard >> s;
    int ans = 0;
    int pos = 0;
    for (auto c : s) {
      ans += std::abs(c - keyboard[pos]);
      pos = c - 'a';
    }
    std::cout << ans << std::endl;
  }
}
