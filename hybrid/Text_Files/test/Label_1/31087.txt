import java.util.Scanner;
import java.util.Stack;

public class Main {
  public static void main(String[] args) {
    Stack<Integer> ans = new Stack<>();
    Scanner sc = new Scanner(System.in);
    int[] r = {sc.nextInt(), sc.nextInt()};
    int[] c = {sc.nextInt(), sc.nextInt()};
    int[] d = {sc.nextInt(), sc.nextInt()};
    int m = 1;
    for (int i = 0; i < 4; i++) {
      if (r[0] != r[1]) {
        ans.push(r[m]);
        r[m] = 0;
        m = m == 0 ? 1 : 0;
      } else
        break;
    }
    m = 1;
    for (int i = 0; i < 4; i++) {
      if (c[0] != c[1]) {
        ans.push(c[m]);
        c[m] = 0;
        m = m == 0 ? 1 : 0;
      } else
        break;
    }
    m = 1;
    for (int i = 0; i < 4; i++) {
      if (d[0] != d[1]) {
        ans.push(d[m]);
        d[m] = 0;
        m = m == 0 ? 1 : 0;
      } else
        break;
    }
    ans.push(r[0] == 0 ? 5 : r[0]);
    ans.push(c[0] == 0 ? 5 : c[0]);
    ans.push(d[0] == 0 ? 5 : d[0]);
    if (ans.size() == 4) {
      while (!ans.isEmpty()) System.out.print(ans.pop() + " ");
    } else
      System.out.println("-1");
    sc.close();
  }
}
