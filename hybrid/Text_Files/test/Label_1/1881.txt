#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> w(n);
        for (int i = 0; i < n; i++) {
            cin >> w[i];
        }
        sort(w.begin(), w.end());
        bool exploded = false;
        int sum = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (sum + w[i] == x) {
                exploded = true;
                break;
            } else if (sum + w[i] > x) {
                break;
            }
            sum += w[i];
        }
        if (exploded) {
            cout << "NO" << endl;
        } else {
            cout << "YES" << endl;
            for (int i = n - 1; i >= 0; i--) {
                cout << w[i] << " ";
            }
            cout << endl;
        }
    }
    return 0;
}
