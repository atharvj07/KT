import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int R = scanner.nextInt();
        int G = scanner.nextInt();
        int B = scanner.nextInt();
        int N = scanner.nextInt();
        long[][][] dp = new long[R + 1][G + 1][B + 1];
        dp[0][0][0] = 1;
        for (int i = 0; i <= R; i++) {
            for (int j = 0; j <= G; j++) {
                for (int k = 0; k <= B; k++) {
                    if (i > 0) {
                        dp[i][j][k] += dp[i - 1][j][k];
                    }
                    if (j > 0) {
                        dp[i][j][k] += dp[i][j - 1][k];
                    }
                    if (k > 0) {
                        dp[i][j][k] += dp[i][j][k - 1];
                    }
                }
            }
        }
        long answer = 0;
        for (int i = 0; i <= R; i++) {
            for (int j = 0; j <= G; j++) {
                for (int k = 0; k <= B; k++) {
                    if (i * R + j * G + k * B == N) {
                        answer += dp[i][j][k];
                    }
                }
            }
        }
        System.out.println(answer);
    }
}
