import sys
for t in range(int(input())):
    n = int(input())
    users = {user for user in sorted(input().split())}
    m = int(input())
    ans = []
    sender = None
    for line in sys.stdin:
        user, text = line.strip().split(':', 1)
        if user == '?':
            for u in users:
                if u not in text and not u.lower() in text:
                    sender = u; break
        else:
            sender = user
        ans.append(':'.join((sender, text)))
    if len({sender for sender in ans}) != n:
        print("Impossible")
    else:
        print('\n'.join(ans))
