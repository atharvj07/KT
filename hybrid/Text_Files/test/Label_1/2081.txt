#include <cstdio>
#include <vector>

const int MAXN = 1e6 + 6;

int uf[MAXN], sz[MAXN];
char a[MAXN];
int dp[MAXN];

int find(int x) { return uf[x] == x ? x : uf[x] = find(uf[x]); }

void join(int a, int b) {
  if ((a = find(a)) == (b = find(b))) return;
  if (sz[a] < sz[b]) std::swap(a, b);
  sz[a] += sz[b];
  uf[b] = a;
}

int main() {
  int n, m, q;
  scanf("%d%d", &n, &m);
  for (int i = 1; i <= n; ++i) {
    scanf(" %s", a + 1);
    for (int j = 1; j <= m; ++j) {
      int c = a[j] == '.' ? 0 : 1;
      dp[j] = dp[j - 1] + c;
      uf[j] = j;
      sz[j] = 1;
      if (c == 0) {
        if (i > 1) join(dp[j], dp[j - 1]);
        if (j > 1) join(dp[j], dp[j] - 1);
      }
    }
  }
  scanf("%d", &q);
  while (q--) {
    int l, r;
    scanf("%d%d", &l, &r);
    int cur = dp[r] - (l == 1 ? 0 : dp[l - 1]);
    if (sz[cur] == r - l + 1)
      printf("YES\n");
    else
      printf("NO\n");
  }
}
