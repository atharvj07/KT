import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int c = in.nextInt();
        int cnt = 0;
        for (int i = 0; i <= n; i += 2) {
            for (int j = 0; i + j <= n; j++) {
                for (int k = 0; i + j + 2 * k <= n; k++) {
                    if (i <= 2 * a && j <= 2 * b && k <= 2 * c) {
                        cnt++;
                    }
                }
            }
        }
        System.out.println(cnt);
    }
}
