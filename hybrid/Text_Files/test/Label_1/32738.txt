import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger n = new BigInteger(in.next());
        BigInteger p = new BigInteger(in.next());
        long w = in.nextLong();
        long d = in.nextLong();
        if (p.compareTo(n.multiply(new BigInteger(String.valueOf(w)))) > 0 || n.multiply(new BigInteger(String.valueOf(d))).compareTo(p) > 0) {
            System.out.println(-1);
            return;
        }
        BigInteger x = BigInteger.ZERO;
        BigInteger y = p.divideAndRemainder(new BigInteger(String.valueOf(w)))[1].divideAndRemainder(new BigInteger(String.valueOf(w - d)))[0];
        BigInteger z = n.subtract(x).subtract(y);
        System.out.println(x + " " + y + " " + z);
    }
}
