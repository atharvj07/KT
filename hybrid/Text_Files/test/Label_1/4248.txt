#include<bits/stdc++.h>
using namespace std;
int main() {
    long long x;
    cin >> x;
    while (x--) {
        long long a, b;
        cin >> a >> b;
        if (a == b) {
            cout << 0 << endl;
            continue;
        }
        if (a > b) {
            long long dif = a - b;
            if (dif % 2 == 1) {
                cout << 2 << endl;
            } else {
                cout << 1 << endl;
            }
            continue;
        }
        long long dif = b - a;
        if (dif % 2 == 0) {
            cout << dif / 2 << endl;
        } else {
            cout << dif / 2 + 1 << endl;
        }
    }
    return 0;
}
