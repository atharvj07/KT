#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<long long, long long>> a(n), b(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    for (int i = 0; i < n; i++) {
        cin >> b[i].first >> b[i].second;
    }
    vector<int> ans1(n), ans2(n);
    for (int i = 0; i < n; i++) {
        if (i < n / 2) {
            ans1[i] = 1;
            ans2[i] = 1;
        } else {
            ans1[i] = (a[i].first < a[n / 2].first);
            ans2[i] = (b[i].first < b[n / 2].first);
        }
    }
    for (int i = 0; i < n; i++) {
        cout << ans1[i];
    }
    cout << endl;
    for (int i = 0; i < n; i++) {
        cout << ans2[i];
    }
    cout << endl;
    return 0;
}
