from collections import defaultdict
from math import sqrt

def isPerfectSquare(n): 
    sr = int(sqrt(n)) 
    return (sr * sr == n)

n = int(input())
a = list(map(int, input().split()))

last = [0] * (n+1)
cnt = defaultdict(int)
for i in range(n):
    cnt[a[i]] += 1
    last[i+1] = last[i]
    if isPerfectSquare(a[i]):
        last[i+1] += 1

lst = []
for v in cnt.values():
    lst.append(v*(v+1)//2)

ans = [0]*n
for i in range(1, n+1):
    for j in range(i):
        if isPerfectSquare(a[i-1]*a[j]):
            ans[i-j-1] += 1

for i in range(n):
    for j in lst:
        ans[i] += j * last[i+1] * max(0, last[n] - last[i+1])

for i in ans:
    print(i, end=' ')
