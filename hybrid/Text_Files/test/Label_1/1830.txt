#include<bits/stdc++.h>
using namespace std;
#define mod 1000000007
long long power(long long x, long long y){
  if (y==0) return 1;
  long long p = power(x, y/2);
  p = (p * p) % mod;
  if (y%2) p = (p * x) % mod;
  return p;
}
int main(){
  long long n, m, ans=0, cnt[11]={0};
  cin >> n >> m;
  for(long long i=0; i<n; i++){
    long long x;
    cin >> x;
    cnt[x]++;
  }
  for(long long i=1; i<=m; i++){
    for(long long j=i+1; j<=m; j++){
      ans = (ans + (cnt[i] * cnt[j])) % mod;
    }
  }
  cout << ans << endl;
}
