import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();
        int[] w = new int[m];
        for (int i = 0; i < m; i++)
            w[i] = sc.nextInt();
        Arrays.sort(w);
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            System.out.println((l <= r && r < m) ? w[r] : -1);
        }
    }
}
