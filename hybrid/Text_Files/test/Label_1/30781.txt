import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int cur = 0, dir = 1, ans = 0;
        for (int i = 0; i < n; i++) {
            if (cur >= a[i]) {
                cur -= a[i];
            } else {
                ans++;
                cur = n - 1 - a[i];
                dir *= -1;
            }
            cur++;
        }
        System.out.println(ans);
    }
}
