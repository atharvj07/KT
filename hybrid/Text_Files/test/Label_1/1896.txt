#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  string sn = to_string(n);
  int m = sn.size();
  string s = "";
  for (int i = 0; i < m; i++) {
    if (sn[i] == '9') {
      s += '0';
    } else {
      s += sn[i];
      break;
    }
  }
  if (s.empty()) s = "0";
  long long last = stoll(s);
  m = s.size();
  long long ans = 1;
  for (int i = 1; i < m; i++) {
    if (n / (long long)pow(10, i) != 0) ans++;
  }
  m--;
  ans += last;
  if (last > 0) ans += 1;
  cout << ans << endl;
  return 0;
}
