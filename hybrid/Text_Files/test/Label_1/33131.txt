import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        Arrays.sort(a);
        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                sum += a[i];
            }
        }
        if (sum % 2 == 0) {
            System.out.println("cslnb");
        } else {
            System.out.println("sjfnb");
        }
    }
}
