import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int p = 0, c = 0;
            boolean valid = true;
            for (int i = 0; i < n; i++) {
                int a = sc.nextInt(), b = sc.nextInt();
                if (a < p || b < c) valid = false;
                if (a == p && b > c) valid = false;
                p = a;
                c = b;
            }
            System.out.println(valid ? "YES" : "NO");
        }
    }
}
