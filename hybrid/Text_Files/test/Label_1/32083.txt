import java.util.*;
import java.math.*;

public class Main {
    static final int MOD = 1000000007;
    static long[][] f;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        f = new long[n][1 << n];
        System.out.println(solve(n, 0, 0, false));
    }

    static long solve(int n, int mask, int sum, boolean last) {
        if (mask == (1 << n) - 1) {
            return sum == 0 ? 1 : 0;
        }
        if (f[sum][mask] != 0) {
            return f[sum][mask];
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if ((mask & (1 << i)) == 0) {
                ans += solve(n, mask | (1 << i), (sum + (i + 1)) % n, last || i + 1 == n);
                ans %= MOD;
            }
        }
        if (last && sum == 0) {
            ans++;
        }
        return f[sum][mask] = ans;
    }
}
