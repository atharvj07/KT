import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        long q = Long.parseLong(br.readLine());
        List<Long> eels = new ArrayList<>();
        while (q-- > 0) {
            String[] line = br.readLine().split(" ");
            if (line[0].equals("+")) {
                eels.add(Long.parseLong(line[1]));
            } else {
                eels.remove(Long.parseLong(line[1]));
            }
            Collections.sort(eels);
            long danger = 0;
            for (int i = 0; i < eels.size() - 1; i++) {
                if (eels.get(i + 1) <= eels.get(i) * 2) {
                    danger++;
                }
            }
            System.out.println(danger);
        }
    }
}
