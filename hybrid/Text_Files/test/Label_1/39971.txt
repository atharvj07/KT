s = input()
t = 'abcdefghijklmnopqrstuvwxyz'
ti, si = 0, 0
res = list(s)
while ti < len(t) and si < len(s):
    if t[ti] == s[si]:
        ti += 1
        si += 1
    elif t[ti] > s[si]:
        res[si] = t[ti]
        ti += 1
        si += 1
    else:
        si += 1
if ti != len(t):
    print(-1)
else:
    print(''.join(res))
