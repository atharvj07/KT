import java.util.Arrays;
import java.util.Collections;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int m = in.nextInt();
        int[] days = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int l = in.nextInt();
            int r = in.nextInt();
            int c = in.nextInt();
            int p = in.nextInt();
            for (int j = l; j <= r; j++) {
                days[j] = Math.max(days[j], Math.min(c, k - days[j - 1]));
            }
        }
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            if (days[i] < k) {
                ans += (long) (k - days[i]) * days[i];
            }
        }
        System.out.println(ans);
    }
}
