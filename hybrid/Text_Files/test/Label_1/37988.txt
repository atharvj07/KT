def can_light_garlands(k1, k2, k3):
    k = sorted([k1, k2, k3])
    if k[0] == 1 or (k[0] == 2 and k[1] == 2) or (k[0] == 2 and k[1] == 4 and k[2] == 4) or (k[0] == 3 and k[1] == 3 and k[2] == 3):
        return "YES"
    return "NO"

k1, k2, k3 = map(int, input().split())
print(can_light_garlands(k1, k2, k3))
