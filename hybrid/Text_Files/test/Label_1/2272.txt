#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> r;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') r.push_back(i);
    }
    int m = r.size(), ans = n;
    for (int i = 0; i < m; i++) {
        int l = max(0, r[i] - k), h = min(n - 1, r[i] + k);
        ans = min(ans, r[i] + (l ? n - h - 1 : 0));
    }
    cout << ans << "\n";
    return 0;
}
