n = int(input())
a = list(map(int, input().split()))

if sum(a) < 2 * (n - 1) or max(a) == 1:
    print("NO")
else:
    print("YES", min(2, n - 1))
    connections = []
    vertices = [(degree, i) for i, degree in enumerate(a, start=1)]
    vertices.sort(reverse=True)

    # Connect highest degree vertices in a line to maximize diameter
    for i in range(len(vertices) - 1):
        connections.append((vertices[i][1], vertices[i + 1][1]))
        vertices[i] = (vertices[i][0] - 1, vertices[i][1])
        vertices[i + 1] = (vertices[i + 1][0] - 1, vertices[i + 1][1])

    # Connect remaining vertices to the line to maintain connectivity without reducing diameter
    line_endpoints = [vertices[0][1], vertices[1][1]]
    for degree, vertex in vertices[2:]:
        if degree > 0:
            for endpoint in line_endpoints:
                if degree > 0:
                    connections.append((vertex, endpoint))
                    degree -= 1
                else:
                    break

    print(len(connections))
    for connection in connections:
        print(connection[0], connection[1])
