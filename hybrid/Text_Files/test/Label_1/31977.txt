import java.util.*;

public class Main {
    static long[] dis;
    static ArrayList<Edge>[] adj;
    static boolean[] vis;
    static int n;

    static class Edge {
        int to;
        long w;

        public Edge(int to, long w) {
            this.to = to;
            this.w = w;
        }
    }

    static void dijkstra(int s) {
        for (int i = 0; i < n; i++) {
            dis[i] = Long.MAX_VALUE / 2;
        }
        PriorityQueue<Edge> pq = new PriorityQueue<>((a, b) -> (int) (a.w - b.w));
        pq.add(new Edge(s, 0));
        dis[s] = 0;
        while (!pq.isEmpty()) {
            Edge cur = pq.poll();
            if (vis[cur.to]) continue;
            vis[cur.to] = true;
            for (Edge e : adj[cur.to]) {
                if (dis[e.to] > dis[cur.to] + e.w) {
                    dis[e.to] = dis[cur.to] + e.w;
                    pq.add(new Edge(e.to, dis[e.to]));
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int m = sc.nextInt();
        adj = new ArrayList[n];
        for (int i = 0; i < n; i++) adj[i] = new ArrayList<>();
        dis = new long[n];
        vis = new boolean[n];
        while (m-- > 0) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            long w = sc.nextInt();
            adj[x].add(new Edge(y, w));
        }
        int s = sc.nextInt() - 1;
        dijkstra(s);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            long cur = dis[i];
            ans += (cur + 1) * cur / 2;
        }
        System.out.println(ans);
    }
}
