#include <iostream>
#include <vector>

using namespace std;

int main() {
  string s;
  cin >> s;
  int m;
  cin >> m;
  vector<int> ans(m);
  int depth = 0;
  for (int i = 0; i < s.size(); i++) {
    depth += s[i] == '(' ? 1 : -1;
    if (depth == 0)
      ans[0]++;
  }
  for (int i = 1; i < m; i++) {
    int l, r;
    cin >> l >> r;
    l--, r--;
    depth += s[l] == '(' ? -1 : 1;
    depth += s[r] == ')' ? -1 : 1;
    if (depth == 0)
      ans[i] = ans[i - 1] + 2;
    else
      ans[i] = ans[i - 1];
  }
  for (int i : ans)
    cout << i << endl;
  return 0;
}
