R, C = map(int, input().split())
wall = [input() for _ in range(R)]

segments = 0
previous_was_empty = True

for c in range(C):
    column_has_brick = any(wall[r][c] == 'B' for r in range(R))
    if column_has_brick and previous_was_empty:
        segments += 1
    previous_was_empty = not column_has_brick

print(segments)
