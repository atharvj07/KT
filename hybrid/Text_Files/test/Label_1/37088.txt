def read_book(times):
    for _ in range(times):
        n = int(input())
        chapters = []
        for _ in range(n):
            chapters.append(list(map(int, input().split()))[1:])
        read_times = 0
        understood = [False] * n
        while True:
            progress = False
            for i, prereqs in enumerate(chapters):
                if not understood[i] and all(understood[j - 1] for j in prereqs):
                    understood[i] = True
                    progress = True
            if progress:
                read_times += 1
            else:
                break
        if all(understood):
            print(read_times)
        else:
            print(-1)

t = int(input())
read_book(t)
