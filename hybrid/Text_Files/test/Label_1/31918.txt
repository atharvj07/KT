import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] weight = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            weight[i] = input.nextInt();
        }
        int p = input.nextInt();
        for (int i = 0; i < p; i++) {
            int a = input.nextInt();
            int b = input.nextInt();
            int sum = 0;
            for (int j = a; j <= n && j < a + b * (n / b); j += b) {
                sum += weight[j];
            }
            System.out.println(sum);
        }
    }
}
