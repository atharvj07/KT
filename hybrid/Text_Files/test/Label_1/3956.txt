#include <iostream>
#include <vector>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, m, k;
        cin >> n >> m >> k;
        int cur = 0;
        int rem = n % m;
        for (int i = 0; i < k; ++i) {
            for (int j = 0; j < m; ++j) {
                if (rem > 0) {
                    cout << (n / m + 1) << " ";
                    for (int l = 1; l <= (n / m + 1); ++l) {
                        cout << cur + l << " ";
                    }
                    cout << "\n";
                    cur += (n / m + 1);
                    --rem;
                } else {
                    cout << (n / m) << " ";
                    for (int l = 1; l <= (n / m); ++l) {
                        cout << cur + l << " ";
                    }
                    cout << "\n";
                    cur += (n / m);
                }
            }
        }
    }
    return 0;
}
