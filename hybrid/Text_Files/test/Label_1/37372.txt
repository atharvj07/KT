n,k=map(int,input().split())

a=[]
for i in range(n):
    a.append([0]*n)

for i in range(0,n):
    for j in range(0,n):
        if i==j:
            a[i][j]=n*n-k*n+i+1
        elif j==(k-1):
            a[i][j]=n*n-(k-1)*n+i+1
        elif j>i:
            a[i][j]=a[i][j-1]+1
        else:
            a[i][j]=a[i-1][j-1]+1

print(sum(x[k-1] for x in a))
for x in a:
    print(*x)
