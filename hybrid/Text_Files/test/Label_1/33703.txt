import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            if (n == 1) {
                System.out.println(1);
            } else {
                System.out.println(n + (n - 1) * (n - 2) / 2);
            }
        }
        scanner.close();
    }
}
