import java.util.*;
import java.io.*;
 
public class Main {
    static long res;
    static int MOD = 1000000007;
    static int[] dis;
    static int c;
    static long[] resarr;
    static int[] arr;
    static int[] sub;
    static int[] tad;
    static ArrayList<Integer>[] tree;
   
    public static int dfs(int curr,int p)
    {
        int sum = 0;
        for(int x:tree[curr])
        {
            if(x == p) continue;
            sum += dfs(x,curr);
        }
        tad[curr] = sum;
        sub[curr] = 1;
        resarr[curr] = 1;
        dis[curr] = sum;
        return sum + 1;
    }
   
    public static void solve()
    {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        tree = new ArrayList[n+4];
        for(int i=1;i<=n;i++)
        {
            tree[i] = new ArrayList<>();
        }
        dis = new int[n+4];
        arr = new int[n+4];
        resarr = new long[n+4];
        tad = new int[n+4];
        sub = new int[n+4];
        for(int i=1;i<n;i++)
        {
            int x = sc.nextInt();
            int y = sc.nextInt();
            tree[x].add(y);
            tree[y].add(x);
        }
        dfs(1,1);
        c = (int)1e9 + 7;
        long sum = arr[1];
        dfs2(1,-1,1,0,k,sum,0);
        System.out.print(sum%c);
    }
   
    public static void dfs2(int curr,int p,int currsub,int curn,int lim,long sum,int ct)
    {
        if(ct == lim)
        {
            resarr[curr] = ((resarr[curr]%c)*(tad[1]))%c;
            sum = sum + resarr[curr];
            sum = sum%c;
            return;
        }
        sum = sum + ((Long.valueOf(resarr[curr])%c)*(Long.valueOf(sub[1])%c))%c;
        sum = sum%c;
        for(int x:tree[curr])
        {
            if(x == p) continue;
            dfs2(x,curr,currsub + tad[x] + 1,curn + sub[x],lim,sum,ct + 1);
        }
    }
   
    public static void main(String[] args) {
        solve();
    }
}
