#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<long long> b(n);
        for (auto &x : b) {
            cin >> x;
            x = 1ll << x;
        }
        vector<bool> vis(n), vis2(n, true);
        function<void(int, long long)> dfs = [&](int u, long long sum) {
            if (u == n) {
                vis2[sum] = false;
                return;
            }
            dfs(u + 1, sum);
            dfs(u + 1, sum + b[u]);
            if (!vis[u]) vis2[sum] = false;
        };
        dfs(0, 0);
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                long long sum = 0;
                for (int k = i; k <= j; k++) sum += b[k], vis[k] = true;
                if (vis2[sum]) {
                    cout << "YES" << endl;
                    return 0;
                }
                for (int k = i; k <= j; k++) vis[k] = false;
            }
        }
        cout << "NO" << endl;
    }
}
