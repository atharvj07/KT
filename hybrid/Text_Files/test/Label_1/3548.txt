#include<bits/stdc++.h>
using namespace std;
int x,y,ans=1e9;
int main()
{
    int n,a,b,sum1=0,sum2=0;
    cin>>n;
    for(int i=0;i<n;i++)
    {
        cin>>a>>b;
        sum1+=a;
        sum2+=b;
    }
    if((sum1%2==0&&sum2%2==0)||(sum1%2==1&&sum2%2==1))cout<<0;
    else
    {
        for(int i=0;i<n;i++)
        {
            cin>>x>>y;
            if((sum1-x+y)%2==0&&(sum2-y+x)%2==0)
            ans=min(ans,1);
        }
        if(ans==1e9)cout<<-1;
        else cout<<ans;
    }
}
