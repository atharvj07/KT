import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int p = in.nextInt();
        int[] a = new int[p];
        for (int i = 0; i < p; i++) {
            a[i] = in.nextInt();
        }
        int[] b = new int[p];
        for (int i = 0; i < p; i++) {
            for (int j = 0; j < p; j++) {
                if ((i * j) % p == a[j]) {
                    b[i] = j;
                    break;
                }
            }
        }
        for (int i = 0; i < p - 1; i++) {
            System.out.print(b[i] + " ");
        }
        System.out.println(b[p - 1]);
    }
}
