import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = in.nextInt();
            arr[i][1] = in.nextInt();
        }
        Arrays.sort(arr, Comparator.comparingInt(a -> a[0] * 60 + a[1]));
        int ans = 0;
        int cur = 0;
        for (int i = 0; i < n; i++) {
            int t = arr[i][0] * 60 + arr[i][1];
            if (cur <= t) {
                ans++;
                cur++;
            }
        }
        System.out.println(ans);
    }
}
