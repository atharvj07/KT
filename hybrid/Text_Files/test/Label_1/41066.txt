from collections import defaultdict

n, m, s, t = map(int, input().split())
edge = defaultdict(lambda: [])
for _ in range(m):
    a, b = map(int, input().split())
    edge[a].append(b)
    edge[b].append(a)
dist = [float('inf')] * (n + 1)
q = [s]
dist[s] = 0
while q:
    v = q.pop(0)
    for u in edge[v]:
        if dist[v] + 1 < dist[u]:
            dist[u] = dist[v] + 1
            q.append(u)
ans = 0
for i in range(1, n + 1):
    for j in range(i + 1, n + 1):
        if dist[s] == dist[i] + 1 + dist[j] and dist[i] + 1 + dist[j] == dist[t]:
            ans += 1
print(ans)
