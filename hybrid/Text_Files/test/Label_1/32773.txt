import java.util.*;
import java.math.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] cnt = new int[1000001];
        for (int x : a) {
            for (int i = 2; i * i <= x; i++) {
                while (x % i == 0) {
                    x /= i;
                    cnt[i]++;
                }
            }
            if (x > 1) {
                cnt[x]++;
            }
        }
        for (int i = 2; i < 1000001; i++) {
            if (cnt[i] >= n) {
                System.out.println(1);
                return;
            }
        }
        System.out.println(0);
    }
}
