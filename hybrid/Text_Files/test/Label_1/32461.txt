import java.util.*;
import java.io.*;

public class Main {

    static int[] solve(int n, int[] x) {
        assert n > 1 && n % 2 == 0;
        Arrays.sort(x);
        int[] b = new int[n];
        int m = (n + 1) / 2;
        for (int i = 0; i < n; i++) {
            int[] a = new int[n - 1];
            int cnt = 0;
            for (int j = 0; j < n; j++) {
                if (i != j) {
                    a[cnt++] = x[j];
                }
            }
            b[i] = a[m - 1];
        }
        return b;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] s = br.readLine().split(" ");
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = Integer.parseInt(s[i]);
        }
        int[] b = solve(n, x);
        for (int i = 0; i < n; i++) {
            System.out.println(b[i]);
        }
    }
}
