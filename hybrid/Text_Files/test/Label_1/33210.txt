import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] a;
    static int[] ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(br.readLine());
        a = new int[n];
        ans = new int[n];

        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(st.nextToken());
        }

        //exists[i] = true if i exists in a
        boolean[] exists = new boolean[n + 1];
        int mex = 0;
        
        for (int i = 0; i < n; i++) {
            if (mex < a[i]) {
                ans[i] = mex;
                exists[mex] = true;
                mex++;
            } else {
                if (!exists[a[i]]) {
                    ans[i] = a[i];
                } else {
                    while (exists[mex]) {
                        mex++;
                    }
                    ans[i] = mex;
                    exists[mex] = true;
                    mex++;
                }
            }
        }

        for (int i = 0; i < n; i++) {
            if (a[i] < i) {
                ans[i] = a[i];
            }
        }

        if(a[n-1] < n){
            System.out.println(-1);
            return;
        }

        StringBuilder sb = new StringBuilder();
        for (int e : ans) {
            sb.append(e + " ");
        }
        System.out.println(sb);
    }
}
