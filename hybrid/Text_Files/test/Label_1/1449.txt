#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    int evenCount = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      evenCount += (a[i] % 2 == 0);
    }
    if (evenCount == 0 || evenCount == n) {
      cout << 0 << endl;
    } else if (evenCount % 2 == 0) {
      cout << 1 << endl;
    } else {
      cout << 2 << endl;
    }
  }
  return 0;
}
