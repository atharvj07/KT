namespace fastIO{
    template<class T> struct Reader{
        T *buf; int bpos, bsize;
        Reader(const string& s, int _bsize = 8192){
            bpos = 0, bsize = _bsize;
            buf = new T[bsize]; stringstream ss(s);
            ss >> buf[0];
        }
        Reader(const char* s, int _bsize = 8192){
            bpos = 0, bsize = _bsize;
            buf = new T[bsize]; stringstream ss(s);
            ss >> buf[0];
        }
        void grow(const T& c = 0, int _bsize = 0){
            if (c == 0) {
                if (_bsize == 0) _bsize = bsize * 2;
                else _bsize = max(1, _bsize);
                buf = (T*)realloc(buf, sizeof(T) * _bsize);
            }
            else {
                if (_bsize == 0) _bsize = buf - c;
                else _bsize = max(1, _bsize);
                buf = (T*)realloc(buf, sizeof(T) * (_bsize + bpos));
                memset(buf + _bsize, 0, sizeof(T) * (bpos - _bsize));
            }
            bsize = _bsize;
        }
        T read(){
            if (bpos == bsize){
                grow();
                stringstream ss("");
                ss >> buf[0];
                bpos = 0;
            }
            return buf[bpos++];
        }
        T operator()(int p){
            return buf[p % bsize];
        }
        T& operator[](int p){
            return buf[p];
        }
        int pos(){
            return bpos;
        }
        ~Reader(){
            delete[] buf;
        }
    };
}

const int MAXN = 200007;

struct dsu{
    vi p,sz;
    dsu(int n){ p.resize(n), iota(all(p),0), sz.assign(n,1); }
    int find(int x){ return x == p[x] ? x : p[x] = find(p[x]); }
    bool unite(int x, int y){
        x = find(x), y = find(y);
        if(x == y) return false;
        if(sz[x] < sz[y]) swap(x,y);
        p[y] = x, sz[x] += sz[y];
        return true;
    }
};

struct edge { int a,b,w; };

int main(){
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    int n,k; cin >> n >> k;
    vi u(2*k); for(int &x : u) cin >> x;

    vector<edge> edges;
    for(int i=0; i<n-1; ++i){
        int a,b; cin >> a >> b;
        edges.pb({a,b,1});
    }

    sort(all(edges), [](const edge &a, const edge &b){ return a.w < b.w; });

    dsu d(n+1);
    int mx = 0;
    for(edge &e : edges){
        if(d.unite(e.a, e.b)){
            int sz = d.sz[d.find(e.a)];
            mx = max(mx, sz * (n - sz));
        }
    }

    cout << mx << endl;

    return 0;
}
