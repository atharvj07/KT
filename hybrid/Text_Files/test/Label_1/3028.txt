#include <bits/stdc++.h>
using namespace std;
const int INF = 1e9;
int n, a, b, h[205], f[205][205][2];
int main() {
  cin >> n >> a >> b;
  for (int i = 1; i <= n; i++) cin >> h[i];
  for (int i = n; i >= 1; i--)
    for (int j = 0; j <= n; j++)
      for (int p = 0; p < 2; p++) {
        f[i][j][p] = INF;
        for (int k = 0; k <= j; k++) {
          if (p == 0 && h[i] <= a - k || p == 1 && h[i] <= b - j + k) {
            f[i][j][p] = min(f[i][j][p], f[i + 1][j - k][0]);
            if (j) f[i][j][p] = min(f[i][j][p], f[i + 1][j - k][1] + j - k);
          }
        }
      }
  cout << (f[1][a][0] > a ? -1 : f[1][a][0]) << '\n';
  return 0;
}
