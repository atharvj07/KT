#include <iostream>

using namespace std;

int main() {
  long long n, k, ans = 0, last = 1, ten = 1;
  cin >> n >> k;
  while (n) {
    if (last < ten) { last = ten; }
    if (n % 10 != 0) { last = n % 10; }
    if (last % ten == 0) {
      for (int i = ten; i < 10 && i <= last; i++) {
        last = ten;
        for (long long j = n; j; j /= 10) {
          if (j % 10 == i) { last = i; }
        }
        ten *= 10;
      }
    }
    if (last % ten != 0) { ans++; }
    for (int i = 0; i < k; i++) { last /= 10; }
    n /= 10;
  }
  cout << ans << endl;
}
