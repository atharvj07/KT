import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line = br.readLine().split(" ");
        int n = Integer.parseInt(line[0]), k = Integer.parseInt(line[1]);
        if (n > 109 || k > 109) {
            System.out.println("-1");
            return;
        }
        if (n == 1) {
            System.out.println("1");
            return;
        }

        List<Integer> lucky = new ArrayList<>();
        lucky.add(4);
        lucky.add(7);
        lucky.add(44);
        lucky.add(47);
        lucky.add(74);
        lucky.add(77);
        lucky.add(444);
        lucky.add(447);
        lucky.add(474);
        lucky.add(477);
        lucky.add(744);
        lucky.add(747);
        lucky.add(774);
        lucky.add(777);

        int count = 0;
        List<Integer> perm = new ArrayList<>();
        for (int i = 0; i < n; i++) perm.add(i + 1);
        while (--k >= 0) {
            for (int j = n - 1; j >= 0; j--) {
                if (lucky.contains(j + 1) && lucky.contains(perm.get(j))) {
                    count++;
                }
                if (perm.get(j) < n) {
                    int pos = j;
                    for (int l = j + 1; l < n; l++) {
                        if (perm.get(l) > perm.get(j)) pos = l;
                    }
                    int tmp = perm.get(pos);
                    perm.set(pos, perm.get(j));
                    perm.set(j, tmp);
                    break;
                }
            }
        }
        System.out.println(count);
    }
}
