import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        long[] sum = new long[n + 1];
        long[] cnt = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + a[i - 1];
            cnt[i] = cnt[i - 1] + 1;
        }
        long[] b = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            b[i] = sum[n] - sum[n - i] + a[n - i] * (i - cnt[n - i]);
        }
        long ans = 0;
        long mod = 998244353;
        for (int i = 1; i <= n; i++) {
            ans = gcd(ans, b[i]);
        }
        System.out.println(ans % mod);
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
