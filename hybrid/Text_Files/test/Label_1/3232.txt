#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        string s, c;
        cin >> s >> c;
        bool p = false;
        for (int i = 0; i < s.size(); i++) {
            for (int j = i + 1; j < s.size(); j++) {
                if (s[i] > s[j]) {
                    swap(s[i], s[j]);
                    if (s < c) {
                        p = true;
                        cout << s << endl;
                    }
                    swap(s[i], s[j]);
                }
            }
        }
        if (!p) cout << "---" << endl;
    }
}
