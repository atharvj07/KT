import sys
input = sys.stdin.readline

n,m = map(int,input().split())
p = list(map(int,input().split()))
c = list(map(int,input().split()))
d = int(input())
l = [[] for _ in range(m)]
for i in range(n):l[c[i]-1].append(p[i])
for i in range(m):l[i].sort()

for k in map(int,input().split()):
    l[c[k-1]-1].remove(p[k-1])
    while not l[c[k-1]-1]:c[k-1]+=1
    print(next((i for i in range(5000) if i not in l[c[k-1]-1]),0))
