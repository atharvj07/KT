#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    while (k--) {
        int mn = *min_element(a.begin(), a.end());
        if (mn > 0) {
            cout << mn << endl;
            for (int &x : a) if (x > 0) x -= mn;
        } else {
            cout << 0 << endl;
        }
    }
}
