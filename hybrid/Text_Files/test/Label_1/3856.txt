#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll mod=1e9+7;
int main() {
    ios::sync_with_stdio(0), cin.tie(0);
    int n, m;
    cin >> n >> m;
    vector <bool> v(n + 1, 1);
    int cnt = n * (n - 1) / 2;
    for(int i=0;i<m;++i) {
        int a, b, k;
        cin >> a >> b >> k;
        v[a] = v[b] = 0;
        cnt -= k;
    }
    ll ans = 1;
    for(int i=1;i<=n;++i) {
        if(v[i]) {
            ans *= 2;
            ans %= mod;
        }
    }
    ans = (ll) pow(ans, cnt) % mod;
    cout << ans;
    return 0;
}
