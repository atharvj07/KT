#include <bits/stdc++.h>
using namespace std;

#define FOR(i,a,b) for(int i=(a);i<(b);i++)
#define REP(i,n) FOR(i,0,n)

typedef long long ll;

const ll MOD = 1e9+7;
const int MAXN = 1e6+5;

ll p[MAXN], phi[MAXN], dp[MAXN][2];
bool seen[MAXN];

void sieve() {
  phi[1] = 1;
  FOR(i,2,MAXN) {
    if (!seen[i]) {
      p[++p[0]] = i;
      phi[i] = i-1;
    }
    FOR(j,1,p[0]+1) {
      int x = i * p[j];
      if (x >= MAXN) break;
      seen[x] = 1;
      if (i % p[j] == 0) {
        phi[x] = phi[i] * p[j];
        break;
      }
      else phi[x] = phi[i] * (p[j]-1);
    }
  }
}

ll modpow(ll x, ll y) {
  ll ans = 1;
  x %= MOD;
  while (y > 0) {
    if (y % 2 == 1) ans = ans * x % MOD;
    x = x * x % MOD;
    y /= 2;
  }
  return ans;
}

ll solve(ll x, ll k) {
  if (x == 1) return dp[1][k] = 1;
  if (dp[x][k] != -1) return dp[x][k];

  ll ans = (k+1) * 1ll * modpow(x,k+1) % MOD;
  for (ll cur = 2; cur*cur <= x; ++cur) {
    if (x % cur == 0) {
      ans = (ans - cur * 1ll * solve(x/cur, k)) % MOD;
      ans = (ans + MOD) % MOD;
      if (x / cur != cur) {
        ans = (ans - x/cur * 1ll * solve(cur, k)) % MOD;
        ans = (ans + MOD) % MOD;
      }
    }
  }
  ans = ans * phi[x]%MOD;
  ans = ans * modpow(x, MOD-2) % MOD;
  return dp[x][k] = ans;
}

int main() {
  sieve();
  ll n, k;
  cin >> n >> k;
  memset(dp, -1, sizeof dp);
  cout << solve(n, k) << endl;
  return 0;
}
