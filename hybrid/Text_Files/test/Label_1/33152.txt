import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int[] cnt = new int[3];
        for (char c : s.toCharArray()) {
            cnt[c - 'A']++;
        }
        if (cnt[0] == cnt[1] && cnt[1] == cnt[2]) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
