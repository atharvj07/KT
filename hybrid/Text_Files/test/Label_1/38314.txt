def is_possible(n, a):
    b = sorted(a)
    for i in range(n):
        if a[i] != b[i]:
            if i == n-1 or a[i] != a[i+1] or b[i] != b[i+1]:
                return "NO"
            else:
                i += 1
    return "YES"

t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    print(is_possible(n, a))
