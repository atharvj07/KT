#include <iostream>
#include <vector>
#include <algorithm>

int main() {
  int n;
  std::cin >> n;

  std::vector<int> a(n);
  for (auto &i : a) { std::cin >> i; }

  std::sort(a.begin(), a.end());
  int j = 0;
  for (int i = 0; i < n; ++i) {
    if (i != 0 && a[i] == a[i - 1]) continue;
    a[j++] = a[i];
  }
  std::cout << j << std::endl;
  for (int i = 0; i < j; ++i) { std::cout << a[i] << ' '; }
  std::cout << std::endl;

  return 0;
}
