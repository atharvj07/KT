import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] x1, y1, x2, y2;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        x1 = new int[n]; y1 = new int[n]; x2 = new int[n]; y2 = new int[n];
        for (int i = 0; i < n; i++) {
            x1[i] = in.nextInt(); y1[i] = in.nextInt();
            x2[i] = in.nextInt(); y2[i] = in.nextInt();
        }
        long ans = 0;
        for (int hv = 0; hv < 2; hv++) {
            for (int h = 0; h < n; h++) {
                for (int v = 0; v < n; v++) {
                    if (hv == 0) {
                        if (x1[h] != x2[h]) continue;
                        if (y1[v] != y2[v]) continue;
                    } else {
                        if (x1[h] != x2[h]) continue;
                        if (y1[v] != y2[v]) continue;
                        int tx = x1[h]; x1[h] = x1[v]; x1[v] = tx;
                        int ty = y1[h]; y1[h] = y1[v]; y1[v] = ty;
                        tx = x2[h]; x2[h] = x2[v]; x2[v] = tx;
                        ty = y2[h]; y2[h] = y2[v]; y2[v] = ty;
                    }
                    for (int hv2 = 0; hv2 < 2; hv2++) {
                        for (int ha = 0; ha < n; ha++) {
                            for (int va = 0; va < n; va++) {
                                if (hv2 == 0) {
                                    if (x1[ha] != x2[ha]) continue;
                                    if (y1[va] != y2[va]) continue;
                                } else {
                                    if (x1[ha] != x2[ha]) continue;
                                    if (y1[va] != y2[va]) continue;
                                    int tx = x1[ha]; x1[ha] = x1[va]; x1[va] = tx;
                                    int ty = y1[ha]; y1[ha] = y1[va]; y1[va] = ty;
                                    tx = x2[ha]; x2[ha] = x2[va]; x2[va] = tx;
                                    ty = y2[ha]; y2[ha] = y2[va]; y2[va] = ty;
                                }
                                if (h != ha && h != va && v != ha && v != va && (x1[h] <= x1[ha] && x2[ha] <= x2[h]) && (y1[v] <= y1[va] && y2[va] <= y2[v])) ans++;
                            }
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
