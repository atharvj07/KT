import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        String ss = s.nextLine();
        int n = ss.length();
        long hq[] = new long[n + 2], tq[] = new long[n + 2];
        long p = 31;
        long m = 1000000009;
        long[][] t = new long[n + 2][n + 2];
        for (int i = 0; i < n; i++) {
            hq[i + 1] = (hq[i] * p + ss.charAt(i)) % m;
            tq[i + 1] = (tq[i] * p + ss.charAt(n - 1 - i)) % m;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (ss.charAt(i) == ss.charAt(n - 1 - j))
                    t[i + 1][j + 1] = t[i][j] + 1;
                else
                    t[i + 1][j + 1] = Math.max(t[i][j + 1], t[i + 1][j]);
            }
        }
        long h = 0, q = 0, mx = 0;
        int i1 = 0, i2 = 0;
        for (int i = 0; i < n - 100; i++) {
            for (int j = 100; j <= n - i; j++) {
                if (t[i][j - 100] >= 50) {
                    h = hq[i + j - 50] - hq[i] * p % m + m;
                    h %= m;
                    q = tq[n - 1 - i] - tq[n - 1 - i - j + 50] * p % m + m;
                    q %= m;
                    if (h == q) {
                        if (j > mx) {
                            mx = j;
                            i1 = i;
                            i2 = j;
                        }
                    }
                }
            }
        }
        if (mx == 0) {
            for (int i = 0; i < n - 1; i++) {
                for (int j = i + 1; j < n; j++) {
                    if (ss.charAt(i) == ss.charAt(j)) {
                        if (j - i + 1 > mx) {
                            mx = j - i + 1;
                            i1 = i;
                            i2 = j;
                        }
                    }
                }
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int i = i1; i <= i1 + i2 - 1 - i1; i++) {
            sb.append(ss.charAt(i));
        }
        System.out.println(sb);
    }
}
