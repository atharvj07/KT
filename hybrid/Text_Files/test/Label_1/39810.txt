from sys import stdin

s = input()
L = len(s)
dp = [[-1] * (L + 1) for i in range(L + 1)]

def pretty(l, r):
    if dp[l][r] != -1: return dp[l][r]
    if l == r: dp[l][r] = (s[l] != '?')
    else:
        if s[l] == s[r] == '?':
            dp[l][r] = pretty(l + 1, r - 1) + 1
        elif s[l] == '?':
            dp[l][r] = pretty(l + 1, r)
            if s[r] != ')': dp[l][r] += pretty(l + 1, r - 1)
        elif s[r] == '?':
            dp[l][r] = pretty(l, r - 1)
            if s[l] != '(': dp[l][r] += pretty(l + 1, r - 1)
        else:
            if s[l] == '(' and s[r] == ')': dp[l][r] = pretty(l + 1, r - 1) + 1
            else: dp[l][r] = 0
    return dp[l][r]

print(pretty(0, L - 1))
