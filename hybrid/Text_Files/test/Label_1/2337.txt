#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e3 + 5, mod = 1e9 + 7;
int dp[maxn][maxn];
int f[maxn], last[maxn];
string s;
int main() {
    cin >> s;
    int n = s.size();
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            dp[i][j] = 1;
            if (i > 0) dp[i][j] = max(dp[i][j], dp[i - 1][j]);
            if (j > 0) dp[i][j] = max(dp[i][j], dp[i][j - 1]);
            if (i != j && s[i] == s[j]) {
                if (i - last[s[i]] >= 2)
                    dp[i][j] = max(dp[i][j], dp[i - last[s[i]] - 1][j - 1] + 1);
            }
            ans = max(ans, dp[i][j]);
        }
        last[s[i]] = i;
    }
    cout << ans << '\n';
    return 0;
}
