import sys

readline = sys.stdin.readline
write = sys.stdout.write

n, k = list(map(int, readline().split()))
a = list(map(int, readline().split()))
if k > 0:
    b = set(list(map(int, readline().split())))
    b = [x-1 for x in b]
else:
    b = []

ans = 0
for i in range(n-1):
    if i in b:
        continue
    if a[i] >= a[i+1]:
        ans += a[i+1] - a[i]
        a[i+1] = a[i] + 1
write(str(ans) + '\n')
