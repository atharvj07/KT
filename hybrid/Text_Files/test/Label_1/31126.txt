import java.util.Scanner;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();

        int[] x = new int[n];
        for (int i = 0; i < n; i++) x[i] = in.nextInt();

        int[] y = new int[n];
        for (int i = 0; i < n; i++) y[i] = in.nextInt();

        int[] set = new int[2 * n];
        System.arraycopy(x, 0, set, 0, n);
        System.arraycopy(y, 0, set, n, n);
        Arrays.sort(set);

        Map<Integer, Integer> cnt = new HashMap<>();
        for (int number : set) {
            cnt.put(number, cnt.getOrDefault(number, 0) + 1);
        }

        int result = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                result += cnt.getOrDefault(x[i] ^ y[j], 0);
            }
        }

        System.out.println(result % 2 == 0 ? "Karen" : "Koyomi");
    }
}
