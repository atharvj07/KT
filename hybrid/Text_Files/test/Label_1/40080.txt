import sys

s=input()
#store s's suffix sum, index i's value means sum of A, B & C in s[:i]
ss=[0]*len(s)
ss[0]=ord(s[0])-64
for i in range(1, len(s)):
    ss[i]=ss[i-1]+ord(s[i])-64

t=input()
#store t's prefix sum, index i's value means sum of A, B & C in t[:i+1]
tt=[0]*len(t)
tt[0]=ord(t[0])-64
for i in range(1, len(t)):
    tt[i]=tt[i-1]+ord(t[i])-64

q=int(input())

for l in range(q):
    a, b, c, d=map(int, input().split())
    print(1 if tt[d-1]-((tt[c-2]) if c>1 else 0)==ss[b-1]-((ss[a-2]) if a>1 else 0) else 0)
