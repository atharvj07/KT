#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<long long int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    long long int sum = 0;
    for (int i = 0; i < n; i++) {
        if (sum + a[i] >= (i + 1)) {
            sum += a[i] - (i + 1);
        } else {
            cout << i + 1 << endl;
            return 0;
        }
    }
    cout << n << endl;
    return 0;
}
