import java.util.Scanner;

public class Main {
    static int n, m, k;
    static int[] p, r;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        p = new int[n + 1];
        r = new int[n + 1];
        for (int i = 0; i < n + 1; i++) {
            p[i] = i;
            r[i] = 1;
        }
        for (int i = 0; i < m; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            union(x, y);
        }
        long ans = 1;
        for (int i = 1; i <= n; i++) {
            if (p[i] == i) {
                ans *= r[i] * r[i] % k - 1;
                ans %= k;
            }
        }
        System.out.println(ans);
    }

    static int root(int x) {
        while (p[x] != x) {
            x = p[x];
        }
        return x;
    }

    static void union(int x, int y) {
        x = root(x);
        y = root(y);
        if (x != y) {
            if (r[x] > r[y]) {
                p[y] = x;
            } else {
                p[x] = y;
                if (r[x] == r[y]) {
                    r[y]++;
                }
            }
        }
    }
}
