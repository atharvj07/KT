#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    vector<vector<char>> a(n, vector<char>(m));
    for (int i = 0; i < n; i++)
      for (int j = 0; j < m; j++) cin >> a[i][j];
    bool ok = 1;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] == '*') {
          bool good = 0;
          for (int d = k; d <= min(n - i, m - j); d++) {
            bool bad = 0;
            for (int x = i - d; x <= i + d; x++)
              bad |= x >= 0 && x < n && j + i - x < m &&
                   j + i - x >= 0 && a[x][j + i - x] != '*';
            if (!bad) good = 1;
          }
          ok &= good;
        }
      }
    }
    cout << (ok ? "YES" : "NO") << '\n';
  }
}
