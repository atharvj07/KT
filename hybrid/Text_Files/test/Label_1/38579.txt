n, a, b, k = map(int, input().split())
s = list(input())
block = (a ** (k - 1), b ** (k - 1), 1)
mod = 10 ** 9 + 9
ans = 0
p0, p1 = 1, 1
for i in range(k - 1, n):
    ans = (ans + (p0 * s[i - (k - 1)]) % mod * (p1 * (a ** (n - i)) % mod) % mod * (b ** i) % mod) % mod
    p0, p1 = (p0 * p0) % mod, (p1 * p1) % mod
print(ans)
