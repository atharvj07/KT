#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n + 1);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  vector<pair<int, int>> edges;
  vector<bool> vis(n + 1);
  function<void(int)> dfs = [&](int u) {
    vis[u] = true;
    for (int v : adj[u]) {
      if (!vis[v]) {
        edges.push_back({u, v});
        dfs(v);
      }
    }
  };
  dfs(1);
  for (auto e : edges) { cout << e.first << " " << e.second << "\n"; }
  return 0;
}
