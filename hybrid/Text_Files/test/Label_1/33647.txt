import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] m = new long[n];
        for (int i = 0; i < n; i++) {
            m[i] = in.nextInt();
        }
        Arrays.sort(m);
        long ans = 0;
        while (m[n - 1] - m[0] > 1) {
            m[0]++;
            m[n - 1]--;
            Arrays.sort(m);
            ans++;
        }
        System.out.println(ans);
    }
}
