#include <bits/stdc++.h>
using namespace std;
map<int, int> mp;
int main() {
  string s;
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> s;
  int n = s.length();
  int ans[n];
  if (s[0] == ')') {
    ans[0] = 0;
  } else {
    ans[0] = 1;
    mp[0] = 1;
  }
  for (int i = 1; i < n; i++) {
    if (s[i] == '(') {
      if (mp[i - 1] != 0) {
        ans[i] = ans[i - 1] + 2;
        mp[i] = 1;
      } else {
        ans[i] = 0;
      }
    } else {
      if (mp[i - 1] != 0) {
        ans[i] = ans[i - 1] - 1;
      } else {
        ans[i] = 0;
      }
    }
  }
  int q;
  cin >> q;
  for (int i = 0; i < q; i++) {
    int l, r;
    cin >> l >> r;
    --l, --r;
    if (l == 0) {
      cout << ans[r] << '\n';
    } else {
      cout << ans[r] - ans[l - 1] << '\n';
    }
  }
}
