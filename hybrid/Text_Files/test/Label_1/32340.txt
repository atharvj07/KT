import java.util.Scanner;

public class Main {
    static boolean[][] grid;
    static int n, m;
    static final int[] dx = {0, 0, 1, -1};
    static final int[] dy = {1, -1, 0, 0};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        grid = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String s = in.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = s.charAt(j) == 'P';
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] && isSafe(i, j)) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }

    static boolean isSafe(int i, int j) {
        for (int k = 0; k < 4; k++) {
            int nx = i + dx[k];
            int ny = j + dy[k];
            if (0 <= nx && nx < n && 0 <= ny && ny < m && grid[nx][ny]) {
                return false;
            }
        }
        return true;
    }
}
