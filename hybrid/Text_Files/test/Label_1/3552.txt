#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;

int main() {
  int n;
  cin >> n;
  vector<ll> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  ll ans = 0, buy = p[0] - 1;
  for (int i = 1; i < n; i++) {
    if (p[i] < buy) buy = p[i];
    else ans += p[i] - buy, buy = p[i];
  }
  cout << ans << '\n';
}
