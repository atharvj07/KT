n,m=map(int,input().split())
a=[list(map(int,input().split())) for i in range(n)]
ans=1e9
for i in range(1,n):
    for j in range(1,m):
        if a[i][j]==1:
            ans=min(ans,i+j-1)
            for x in range(1,n):
                for y in range(1,m):
                    mnx=min(x,i)
                    mny=min(y,j)
                    mnxx=max(x,i)
                    mnyy=max(y,j)
                    if x==i and y==j:continue
                    if mnx<=x<=mnxx and mny<=y<=mnyy:
                        ans=min(ans,mnx+mny-1+mnxx+mnyy-x-y)
print(ans)
