from heapq import heappop, heappush

def min_extensions(a, b, h, w, n, extensions):
    extensions.sort(reverse=True)
    queue = [(-max(h, w), -min(h, w), 0)]
    visited = {(h, w), (w, h)}
    
    while queue:
        max_side, min_side, used = heappop(queue)
        max_side, min_side = -max_side, -min_side
        
        if max_side >= a and min_side >= b:
            return used
        
        if used < n:
            for i in range(used, n):
                new_max = max_side * extensions[i]
                if (new_max, min_side) not in visited:
                    heappush(queue, (-new_max, -min_side, used + 1))
                    visited.add((new_max, min_side))
                new_min = min_side * extensions[i]
                if (max_side, new_min) not in visited:
                    heappush(queue, (-max_side, -new_min, used + 1))
                    visited.add((max_side, new_min))
                break
    
    return -1

a, b, h, w, n = map(int, input().split())
extensions = list(map(int, input().split()))

print(min_extensions(a, b, h, w, n, extensions))
