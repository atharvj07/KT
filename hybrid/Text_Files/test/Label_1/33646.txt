import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextLong();
        }
        long[] a = new long[n];
        a[0] = b[0];
        for (int i = 1; i < n; i++) {
            a[i] = a[i - 1] ^ b[i];
        }
        if (!isIncreasing(a)) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            for (int i = 0; i < n; i++) {
                System.out.print(b[i] + " ");
            }
            System.out.println();
        }
    }

    private static boolean isIncreasing(long[] a) {
        for (int i = 1; i < a.length; i++) {
            if (a[i] <= a[i - 1]) {
                return false;
            }
        }
        return true;
    }
}
