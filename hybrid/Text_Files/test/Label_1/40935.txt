from collections import defaultdict

def find_duplicate_websites(n, queries):
    sites = defaultdict(set)
    for query in queries:
        parts = query[7:].split("/", 1)  # Remove 'http://' and split by the first '/'
        hostname = parts[0]
        path = parts[1] if len(parts) > 1 else ""
        sites[hostname].add(path)
    
    # Reverse mapping from sorted paths to hostnames
    paths_to_hosts = defaultdict(set)
    for host, paths in sites.items():
        paths_to_hosts[frozenset(paths)].add(host)
    
    # Filter out unique sites and prepare output
    duplicates = [hosts for hosts in paths_to_hosts.values() if len(hosts) > 1]
    return duplicates

# Input handling
n = int(input().strip())
queries = [input().strip() for _ in range(n)]

# Process and output results
duplicates = find_duplicate_websites(n, queries)
print(len(duplicates))
for group in duplicates:
    print(" ".join(sorted(group)))
