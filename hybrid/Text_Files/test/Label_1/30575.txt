import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.PriorityQueue;
import java.util.StringTokenizer;

public class Main {
    static class Node implements Comparable<Node> {
        int id, fatigue;
        long height;

        public Node(int id, int fatigue, long height) {
            this.id = id;
            this.fatigue = fatigue;
            this.height = height;
        }

        @Override
        public int compareTo(Node o) {
            return fatigue - o.fatigue;
        }
    }

    static int[] inDegree;
    static List<List<Node>> adjList;
    static int n;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(reader.readLine());
        int[] fatigue = new int[n];
        long[] height = new long[n];
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        for (int i = 0; i < n; i++) {
            fatigue[i] = Integer.parseInt(tokenizer.nextToken());
        }
        tokenizer = new StringTokenizer(reader.readLine());
        for (int i = 0; i < n; i++) {
            height[i] = Integer.parseInt(tokenizer.nextToken());
        }
        inDegree = new int[n];
        adjList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int u = Integer.parseInt(tokenizer.nextToken()) - 1;
            int v = Integer.parseInt(tokenizer.nextToken()) - 1;
            adjList.get(u).add(new Node(v, fatigue[v], height[v]));
            inDegree[v]++;
        }
        long ans = 0;
        PriorityQueue<Node> pq = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            if (inDegree[i] == 0) {
                pq.add(new Node(i, fatigue[i], height[i]));
            }
        }
        while (!pq.isEmpty()) {
            Node curr = pq.poll();
            ans += curr.fatigue;
            for (Node next : adjList.get(curr.id)) {
                if (--inDegree[next.id] == 0) {
                    pq.add(next);
                }
            }
        }
        System.out.println(ans);
    }
}
