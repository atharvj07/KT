#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(m);
    for (int i = 0; i < m; i++) cin >> a[i];
    int cur = 1, ans = 0;
    for (int i = 0; i < m; i++) {
        int dif = (a[i] - cur + n) % n;
        ans += dif;
        cur = a[i];
    }
    cout << ans << endl;
    return 0;
}
