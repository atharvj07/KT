import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int max = 0;
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            max = Math.max(max, a[i]);
        }
        in.close();
        if (max == 0) {
            System.out.println(0);
            return;
        }
        int ans = 0;
        while ((1<<ans) < max) {
            ans++;
        }
        System.out.println(ans);
    }
}
