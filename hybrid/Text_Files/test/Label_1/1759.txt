#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  vector<int> cnt(9, 0);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    cnt[a[i]]++;
  }
  vector<int> dp(9, 0);
  for (int i = 1; i < 9; i++) {
    dp[i] += min(cnt[i], cnt[i - 1] + 1);
  }
  for (int i = 7; i > 0; i--) {
    dp[i] += min(cnt[i], cnt[i + 1] + 1);
  }
  int ans = 0;
  for (int i = 1; i < 9; i++) {
    ans = max(ans, dp[i]);
  }
  cout << ans << endl;
  return 0;
}
