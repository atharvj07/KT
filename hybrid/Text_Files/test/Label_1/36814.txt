n, k = map(int, input().split())
beauties = list(map(int, input().split()))
colors = input()

# Group roses by color and sort each group by beauty in descending order
grouped_beauties = {}
for i in range(n):
    color = colors[i]
    if color in grouped_beauties:
        grouped_beauties[color].append(beauties[i])
    else:
        grouped_beauties[color] = [beauties[i]]
for color in grouped_beauties:
    grouped_beauties[color].sort(reverse=True)

# Compute the prefix sums for each color to facilitate the calculation of the total beauty
for color in grouped_beauties:
    prefix_sum = 0
    for i in range(len(grouped_beauties[color])):
        prefix_sum += grouped_beauties[color][i]
        grouped_beauties[color][i] = prefix_sum

# Try to make a bouquet
max_beauty = -1
for color in 'WR':
    if color in grouped_beauties and len(grouped_beauties[color]) >= k:
        max_beauty = max(max_beauty, grouped_beauties[color][k-1])
if 'O' in grouped_beauties:
    for i in range(min(k, len(grouped_beauties['O']))):
        total_orange_beauty = grouped_beauties['O'][i]
        remaining = k - (i + 1)
        for color in 'WR':
            if remaining > 0 and color in grouped_beauties and len(grouped_beauties[color]) >= remaining:
                max_beauty = max(max_beauty, total_orange_beauty + grouped_beauties[color][remaining-1])
print(max_beauty)
