n=int(input())
a,b=map(int,input().split())
h=list(map(int,input().split()))
h.sort(reverse=1)
su=0
for i in h:
    if b==0:
        break
    else:
        su+=i
        b-=i
for i in h:
    if a==0:
        break
    else:
        su+=i
        a-=i
if b==0 and a==0:
    print(su)
else:
    print(-1)
