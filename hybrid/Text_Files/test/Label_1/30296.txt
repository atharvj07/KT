import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    static int mod = 998244353;
    static long[][][] dp;
    static int[] a, b;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        dp = new long[n + 1][100001][2];
        a = new int[n];
        b = new int[n];
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            a[i] = Integer.parseInt(s[0]);
            b[i] = Integer.parseInt(s[1]);
        }
        System.out.println(solve(1, 0, 1));
    }

    public static long solve(int x, int last, int dir) {
        if (x == a.length + 1) {
            return 1;
        }
        if (dp[x][last][dir] != -1) {
            return dp[x][last][dir];
        }
        long ans = 0;
        if (dir == 1) {
            if (a[x - 1] - last <= 5) {
                ans += solve(x + 1, last + 1, 1);
            }
            if (a[x - 1] - last <= 5 && last > 0) {
                ans += solve(x + 1, last - 1, 0);
            }
        } else {
            if (last - b[x - 1] <= 5) {
                ans += solve(x + 1, last - 1, 0);
            }
            if (last - b[x - 1] <= 5 && a[x] - last > 0) {
                ans += solve(x + 1, last + 1, 1);
            }
        }
        ans %= mod;
        return dp[x][last][dir] = ans;
    }
}
