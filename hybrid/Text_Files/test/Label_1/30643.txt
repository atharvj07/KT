import java.util.*;

public class Main {
    static int[] par;
    static int[] dep;
    static int[] rank;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        List<Integer>[] g = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            g[i] = new ArrayList<>();
        }
        for (int i = 0; i < n - 1; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            g[a].add(b);
            g[b].add(a);
        }
        int m = in.nextInt();
        int[] ans = new int[m];
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            par = new int[n + 1];
            dep = new int[n + 1];
            rank = new int[n + 1];
            for (int j = 1; j <= n; j++) {
                par[j] = j;
                dep[j] = 1;
                rank[j] = 0;
            }
            int da = dfs1(a, -1, g);
            int db = dfs1(b, -1, g);
            int lca = dfs2(da, db, g);
            ans[i] = dep[da] + dep[db] - 2 * dep[lca] + 1;
        }
        for (int i = 0; i < m; i++) {
            System.out.println(ans[i]);
        }
    }

    static int dfs1(int u, int p, List<Integer>[] g) {
        for (int v : g[u]) {
            if (v == p) {
                continue;
            }
            dep[v] = dep[u] + 1;
            dfs1(v, u, g);
        }
        return u;
    }

    static int dfs2(int u, int p, List<Integer>[] g) {
        int ans = u;
        for (int v : g[u]) {
            if (v == p) {
                continue;
            }
            if (dep[v] > dep[ans]) {
                ans = v;
            }
            dfs2(v, u, g);
        }
        return ans;
    }
}
