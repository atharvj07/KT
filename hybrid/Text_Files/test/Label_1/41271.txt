def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    return a * b // gcd(a, b)

def solve(a1, b1, a2, b2, L, R):
    if b1 > b2:
        a1, a2 = a2, a1
        b1, b2 = b2, b1
    g = gcd(a1, a2)
    if (b2 - b1) % g != 0:
        return 0
    a = lcm(a1, a2)
    b = b1 + (b2 - b1) // g * a1
    left = (L - b + a - 1) // a
    right = (R - b) // a
    return max(0, right - left + 1)

a1, b1, a2, b2, L, R = map(int, input().split())
print(solve(a1, b1, a2, b2, L, R))
