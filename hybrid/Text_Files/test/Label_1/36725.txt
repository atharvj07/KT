import queue
n = int(input())
G = [[] for i in range(n+1)]
E = [0 for i in range(n+1)]
for _ in range(n-1):
    a, b, c = map(int, input().split())
    G[a].append((b, c))
    G[b].append((a, c))
    E[a] +=1
    E[b] +=1

X = [int(input()) for i in range(n)]

q = queue.LifoQueue()
Lv = [0 for i in range(n+1)]

for i in range(1, n+1):
    if E[i] == 1: q.put(i)

while not q.empty():
    x = q.get()
    Lv[x] = 0
    for u, w in G[x]:
        if Lv[u] < Lv[x] + w:
            Lv[u] = Lv[x] + w
            E[u] -= 1
            if E[u] == 1: q.put(u)

s = 0
for i in range(1, n+1):
    for u, w in G[i]:
        s = max(s, Lv[i] + Lv[u] - w)

print(s)
