import java.util.Scanner;
import java.util.Arrays;

public class Main {
    static int[] d;
    static int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        d = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            d[i] = sc.nextInt();
            d[n] += d[i];
        }
        Arrays.sort(d, 1, n + 1);
        int lb = 0, ub = n;
        while (lb < ub) {
            int m = (lb + ub) / 2;
            boolean ok = true;
            for (int j = 1; j <= n; j++) {
                if (d[m] - d[j] > m - j) {
                    ok = false;
                }
            }
            if (ok) {
                ub = m;
            } else {
                lb = m + 1;
            }
        }
        int res = n - ub;
        if (res < 0) {
            System.out.println(-1);
        } else {
            for (int i = 0; i <= res; i++) {
                System.out.print(i + " ");
            }
        }
    }
}
