#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

double get_max_efficiency(vector<int>& states, int n, int u) {
  if (u < states[n - 1] - states[0]) return -1.0;
  double max_efficiency = 0.0;
  for (int i = 0; i < n - 2; i++) {
    int j = i + 1;
    int k = i + 2;
    while (k < n && states[k] - states[i] <= u) {
      double efficiency = static_cast<double>(states[k] - states[j]) / (states[k] - states[i]);
      if (efficiency > max_efficiency) {
        max_efficiency = efficiency;
      }
      k++;
    }
  }
  return max_efficiency;
}

int main() {
  int n, u;
  cin >> n >> u;
  vector<int> states(n);
  for (int i = 0; i < n; i++) {
    cin >> states[i];
  }

  double max_efficiency = get_max_efficiency(states, n, u);
  if (max_efficiency < 0) {
    cout << -1.0 << endl;
  } else {
    printf("%.9f\n", max_efficiency);
  }

  return 0;
}
