#include <bits/stdc++.h>
using namespace std;
const int N = 100000;
int n, h;
vector<int> a(N);
void solve() {
  cin >> n >> h;
  for (int i = 0; i < n; i++) cin >> a[i];
  vector<int> b(n);
  sort(a.begin(), a.begin() + n);
  // There are only two possible partitions:
  // 1. First subsequence contains even number of elements
  // 2. First subsequence contains odd number of elements
  // If the number of negative elements is odd, then the first case is always better.
  // Otherwise, calculate and compare the possible goodness of two cases.
  int neg = 0;
  for (int i = 0; i < n; i++) neg += a[i] < 0;
  if (neg % 2 == 1) {
    for (int i = 0; i < n; i++) b[i] = 1;
  } else {
    int sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; i++) sum1 += max(a[i], 0);
    for (int i = 0; i < n; i++) {
      sum1 -= min(a[i], 0);
      sum2 += max(a[i] + h, 0);
    }
    if (sum1 < sum2)
      for (int i = 0; i < n; i++) b[i] = 1;
  }
  int goodness = 0;
  for (int i = 0; i < n; i++) {
    for (int j = i + 1; j < n; j++) {
      if (b[i] == b[j])
        goodness = max(goodness, a[i] + a[j]);
      else
        goodness = max(goodness, a[i] + a[j] + h);
    }
  }
  int mn = *min_element(a.begin(), a.begin() + n);
  int mx = *max_element(a.begin(), a.begin() + n);
  goodness = max(goodness, mx + mx + h);
  goodness -= max(mn + mn + h, mx + mn);
  cout << goodness << "\n";
  for (int i = 0; i < n; i++) cout << b[i] << " ";
}
int main() {
  solve();
  return 0;
}
