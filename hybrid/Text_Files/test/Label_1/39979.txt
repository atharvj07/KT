def next_palindrome_time(h, m):
    while True:
        m += 1
        if m == 60:
            m = 0
            h += 1
        if h == 24:
            h = 0
        if h // 10 == m % 10 and h % 10 == m // 10:
            return h, m

time = input().split(":")
hh, mm = int(time[0]), int(time[1])

n_hh, n_mm = next_palindrome_time(hh, mm)
minutes = (n_hh - hh) * 60 + (n_mm - mm)
if minutes < 0:
    minutes += 24 * 60
print(minutes)
