import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt(), ans = -1;
        char[][] v = new char[n][m];
        for (int i = 0; i < n; i++) {
            v[i] = in.next().toCharArray();
            for (int j = 0; j < m; j++)
                if (v[i][j] == 'B') {
                    if (ans == -1)
                        ans = i + j;
                    else
                        ans = Math.min(ans, i + j);
                }
        }
        for (int l = 0; ans != -1 && l < ans / 2; l++) {
            int cnt = 0, nr = 0, c = 0;
            for (int i = l, j = l; i < n - l && j < m - l; i++, j++) {
                if (v[i][j] == 'B') {
                    cnt++;
                    c = j;
                    nr = i;
                }
            }
            if (cnt != 0 && cnt != ans - l * 2)
                ans = -1;
            for (int i = l, j = c + 1; i < nr && j < m - l; i++, j++)
                if (v[i][j] == 'B')
                    ans = -1;
            for (int i = nr + 1, j = c; i < n - l && j < m - l; i++, j++)
                if (v[i][j] == 'B')
                    ans = -1;
        }
        if (ans >= 0)
            ans /= 2;
        if (ans == 0)
            ans = 1;
        System.out.println(ans);
    }
}
