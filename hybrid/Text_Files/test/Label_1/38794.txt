for i in range(int(input())):
    n, k = map(int, input().split())
    s = list(str(n))
    t = {}
    for i in s: t[i] = 1
    if len(t) <= k: print(n)
    else:
        while len(t) > k:
            s.remove(max(s))
            t = {}
            for i in s: t[i] = 1
        if int(''.join(s)) < n: print(int(''.join(s)) + 10 ** (len(s) - 1))
        else: print(int(''.join(s)))
