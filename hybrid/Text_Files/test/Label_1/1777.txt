#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  sort(a.begin(), a.end());
  cout << a[k - 1] << endl;
  for (int i = 0; i < k; i++) {
    cout << lower_bound(a.begin(), a.end(), a[i]) - a.begin() + 1 << ' ';
  }
  cout << endl;
  return 0;
}
