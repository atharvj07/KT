import sys
for _ in range(int(input())):
    n = int(input())
    d = {}
    for _ in range(n):
        s = input()
        for c in s:
            if c in d:
                d[c] += 1
            else:
                d[c] = 1
    if not d:
        print(0)
    else:
        mx = max(d.values())
        sm = sum(d.values())
        if mx * 2 > sm:
            print(n)
        else:
            print(0)
