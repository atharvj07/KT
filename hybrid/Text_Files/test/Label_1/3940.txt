#include <bits/stdc++.h>
using namespace std;

int main() {
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        unordered_map<long long, bool> mp;
        long long sum = 0, mx = 0;
        for (int i = 0; i < n; i++) {
            long long x;
            cin >> x;
            mp[x] = 1;
            sum += x;
            mx = max(mx, x);
        }
        int ans = 0;
        for (auto it : mp) {
            long long x = it.first;
            if (x < mx && sum - x > mx) continue;
            ans++;
        }
        cout << ans << endl;
        for (auto it : mp) {
            long long x = it.first;
            if (x < mx && sum - x > mx) continue;
            cout << it.first << " ";
        }
        cout << endl;
    }
    return 0;
}
