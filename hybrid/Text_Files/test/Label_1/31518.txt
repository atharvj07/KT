import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[][] songs = new int[n][3];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 3; j++) {
                songs[i][j] = sc.nextInt();
            }
        }
        int max = 0;
        for (int i = 0; i < (1 << n); i++) {
            int[] cnt = new int[3];
            int k = i, cur = 0;
            while (k > 0) {
                if ((k & 1) == 1) {
                    for (int j = 0; j < 3; j++) {
                        cnt[j] += songs[cur][j];
                    }
                }
                k >>= 1;
                cur++;
            }
            if (cnt[0] + cnt[1] + cnt[2] <= m) {
                max = Math.max(max, cnt[1] + 2 * cnt[2]);
            }
        }
        System.out.println(max);
    }
}
