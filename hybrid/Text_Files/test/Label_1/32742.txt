import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x1 = sc.nextInt(), x2 = sc.nextInt(), x3 = sc.nextInt(), x4 = sc.nextInt(), x5 = sc.nextInt(), x6 = sc.nextInt();
        int y1 = sc.nextInt(), y2 = sc.nextInt(), y3 = sc.nextInt(), y4 = sc.nextInt(), y5 = sc.nextInt(), y6 = sc.nextInt();
        long ans = ((x4 - x3 + 1) * (y4 - y3 + 1)) % 1000000007;
        ans *= ((x6 - x5 + 1) * (y6 - y5 + 1)) % 1000000007;
        ans %= 1000000007;
        ans *= ((x2 - x1 + 1) * (y2 - y1 + 1) - ans) % 1000000007;
        ans %= 1000000007;
        System.out.println(ans);
    }
}
