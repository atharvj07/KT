#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;
const int MAXN = 25;
typedef pair<int, int> pi;
bool grid[MAXN][MAXN];
int h, w, mark[MAXN][MAXN];
const int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
bool valid(int r, int c) {
	return 0 <= r && r < h && 0 <= c && c < w && mark[r][c] == -1;
}
bool dfs(int r, int c, int flg) {
	mark[r][c] = flg;
	for (int k = 0; k < 4; ++k) {
		int nr = r + dr[k], nc = c + dc[k];
		if (!valid(nr, nc)) continue;
		if (flg == 0 || flg == 2) {
			if (grid[r][c] == (k == 0 || k == 1)) {
				if (dfs(nr, nc, flg)) return true;
			}
		} else {
			if (grid[r][c] == (k == 2 || k == 3)) {
				if (dfs(nr, nc, flg)) return true;
			}
		}
	}
	return false;
}
int main() {
	cin >> h >> w;
	string s, t;
	cin >> s >> t;
	for (int i = 0; i < h; ++i) {
		for (int j = 0; j < w; ++j) {
			mark[i][j] = -1;
		}
	}
	for (int i = 0; i < h; ++i) {
		for (int j = 0; j < w; ++j) {
			if (i > 0 && s[i - 1] == '<') grid[i][j] = 1;
			else if (i + 1 < h && s[i] == '>') grid[i][j] = 0;
			else if (j > 0 && t[j - 1] == '^') grid[i][j] = 2;
			else if (j + 1 < w && t[j] == 'v') grid[i][j] = 3;
		}
	}
	if (dfs(0, 0, 0) && dfs(0, w - 1, 1) && dfs(h - 1, 0, 2) && dfs(h - 1, w - 1, 3)) cout << "YES\n";
	else cout << "NO\n";
}
