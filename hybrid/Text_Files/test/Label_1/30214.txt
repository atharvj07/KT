import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class Main {
  public static void main(String[] args) throws Exception {
    try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
      int n = Integer.parseInt(br.readLine());
      Map<Long, Map<Long, Integer>> distToCnt = new HashMap<>();
      Map<Long, Set<Long>> allDists = new HashMap<>();

      for (int i = 0; i < n; ++i) {
        String[] line = br.readLine().split(" ");
        long x = Long.parseLong(line[0]);
        long y = Long.parseLong(line[1]);

        Map<Long, Integer> dists = distToCnt.getOrDefault(x, new HashMap<>());
        dists.put(y, dists.getOrDefault(y, 0) + 1);
        distToCnt.put(x, dists);

        Set<Long> d = allDists.getOrDefault(x, new HashSet<>());
        d.add(y);
        allDists.put(x, d);
      }

      long ans = 0;
      for (Set<Long> dists : allDists.values()) {
        for (long d : dists) {
          ans += dists.stream().filter(dt -> dt != d && d * d <= dt * dt).count();
        }
      }

      for (Map<Long, Integer> d : distToCnt.values()) {
        for (int cnt : d.values()) {
          ans += cnt * (cnt - 1) / 2;
        }
      }

      System.out.println(ans);
    }
  }
}
