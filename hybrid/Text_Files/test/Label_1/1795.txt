#include <iostream>
#include <vector>
using namespace std;
const int maxn = 1e5 + 100;
vector<int> adj[maxn];
int sz[maxn], dist[maxn];
int n;
void dfs(int u, int p) {
	sz[u] = 1;
	dist[u] = dist[p] ^ 1;
	for (auto v : adj[u]) {
		if (v != p) {
			dfs(v, u);
			sz[u] += sz[v];
		}
	}
}
int check(int u, int p, int cp) {
	int odd = 0, even = 0;
	for (auto v : adj[u]) {
		if (v != p) {
			if (dist[v] == dist[u]) {
				even += check(v, u, cp);
			} else {
				odd += check(v, u, cp);
			}
		}
	}
	if ((odd + even) % 2 == 1) {
		return -1;
	} else if (sz[u] % 2 == 1) {
		if (cp % 2 == 1) {
			return -1;
		} else {
			return (odd + even) / 2 + 1;
		}
	} else {
		return (odd + even) / 2;
	}
}
int main() {
	cin >> n;
	for (int i = 0; i < n - 1; i++) {
		int u, v;
		cin >> u >> v;
		adj[u].push_back(v);
		adj[v].push_back(u);
	}
	dfs(1, 1);
	int ans = check(1, 1, sz[1]);
	cout << ans << "\n";
	return 0;
}
