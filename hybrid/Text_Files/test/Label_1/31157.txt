import java.util.Scanner;

public class TwoDimensionalKingdom {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int x = scanner.nextInt();
        int y = scanner.nextInt();

        int[] soldiers = new int[n];
        for (int i = 0; i < n; i++) {
            soldiers[i] = scanner.nextInt();
        }

        int[] vests = new int[m];
        for (int i = 0; i < m; i++) {
            vests[i] = scanner.nextInt();
        }

        int k = 0;
        int[] vestIndex = new int[m];

        for (int i = 0; i < n; i++) {
            int left = 0;
            int right = m - 1;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (vests[mid] >= soldiers[i] - x && vests[mid] <= soldiers[i] + y) {
                    if (vestIndex[mid] == 0) {
                        k++;
                        vestIndex[mid] = 1;
                        break;
                    } else {
                        left = mid + 1;
                    }
                } else if (vests[mid] < soldiers[i] - x) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
        }

        System.out.println(k);

        for (int i = 0; i < m; i++) {
            if (vestIndex[i] == 1) {
                for (int j = 0; j < n; j++) {
                    if (soldiers[j] >= vests[i] - x && soldiers[j] <= vests[i] + y) {
                        System.out.println((j + 1) + " " + (i + 1));
                        break;
                    }
                }
            }
        }
    }
}
