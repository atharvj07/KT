import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int R = sc.nextInt();
        int G = sc.nextInt();
        int B = sc.nextInt();
        int N = sc.nextInt();
        BigInteger a = BigInteger.valueOf(N / R);
        BigInteger b = BigInteger.valueOf(N / G);
        BigInteger c = BigInteger.valueOf(N / B);
        BigInteger ans = a.multiply(b).multiply(c);
        System.out.println(ans);
    }
}
