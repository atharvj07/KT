#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n;
  cin >> n;
  string s, t;
  cin >> s >> t;
  vector<pair<int, int>> ans;
  for (int i = 0; i < n; i++) {
    if (s[i] != t[i]) {
      int j = i + 1;
      while (j < n && s[j] != t[i]) j++;
      if (j == n) {
        cout << -1 << endl;
        return 0;
      }
      ans.push_back({j, j - 1});
      swap(s[j], s[j - 1]);
    }
  }
  cout << ans.size() << endl;
  for (auto &p : ans) {
    cout << p.first << " ";
  }
  cout << endl;
  return 0;
}
