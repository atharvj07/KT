import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] inc = new int[n];
        int[] dec = new int[n];
        int incCnt = 0;
        int decCnt = 0;
        for (int i = 0; i < n; i++) {
            if (incCnt > 0 && a[i] < inc[incCnt - 1]) {
                System.out.println("NO");
                return;
            }
            if (decCnt > 0 && a[i] > dec[decCnt - 1]) {
                System.out.println("NO");
                return;
            }
            if (incCnt == 0 || a[i] > inc[incCnt - 1]) {
                inc[incCnt++] = a[i];
            } else {
                dec[decCnt++] = a[i];
            }
        }
        System.out.println("YES");
        System.out.println(incCnt);
        for (int i = 0; i < incCnt; i++) {
            System.out.print(inc[i] + " ");
        }
        System.out.println();
        System.out.println(decCnt);
        for (int i = 0; i < decCnt; i++) {
            System.out.print(dec[i] + " ");
        }
        System.out.println();
    }
}
