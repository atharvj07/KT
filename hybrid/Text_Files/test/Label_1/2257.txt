#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n);
    for (int i = 0; i < n - 1; i++) {
        int a, b;
        cin >> a >> b;
        a--, b--;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    vector<int> seq(n);
    for (int i = 0; i < n; i++) {
        cin >> seq[i];
        seq[i]--;
    }
    vector<int> vis(n, 0);
    queue<int> q;
    q.push(0);
    vis[0] = 1;
    int idx = 0;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (u != seq[idx++]) {
            cout << "No" << endl;
            return 0;
        }
        for (auto v : adj[u]) {
            if (!vis[v]) {
                q.push(v);
                vis[v] = 1;
            }
        }
    }
    cout << "Yes" << endl;
    return 0;
}
