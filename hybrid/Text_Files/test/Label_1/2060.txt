#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> t(n);
  for (int i = 0; i < n; i++) {
    cin >> t[i];
  }

  int ans = 0;
  int cur = 1;
  for (int i = 0; i < n; i++) {
    ans += cur;
    cur = cur * 2 + t[i] * cur;
  }
  cout << ans << endl;
  return 0;
}
