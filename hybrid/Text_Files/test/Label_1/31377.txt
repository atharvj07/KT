import java.util.Arrays;
import java.util.Scanner;

class Main {
    static boolean[][] grid;
    static boolean[] rows, cols;
    static int n, m;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        sc.nextLine();
        grid = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < m; j++) {
                if (line.charAt(j) == '#') grid[i][j] = true;
            }
        }
        rows = new boolean[n];
        cols = new boolean[m];
        System.out.println(check(0) ? "Yes" : "No");
    }

    static boolean check(int cnt) {
        if (cnt == n * m) return true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j]) {
                    if (!rows[i] && !cols[j]) {
                        rows[i] = cols[j] = true;
                        if (check(cnt + 1)) return true;
                        rows[i] = cols[j] = false;
                    } else return false;
                }
            }
        }
        return false;
    }
}
