def find_possible_eating_order(n, a, k, b):
    actions = []
    j = 0
    for i in range(k):
        if j >= n:
            return "NO"
        current_weight = a[j]
        j += 1
        eat_actions = []
        while j < n and current_weight < b[i]:
            if a[j] < b[i] - current_weight:
                eat_actions.append('R')
                current_weight += a[j]
                j += 1
            else:
                break
        if current_weight != b[i]:
            return "NO"
        actions.extend(reversed(eat_actions))
    if j != n:
        return "NO"
    return actions

n = int(input().strip())
a = list(map(int, input().strip().split()))
k = int(input().strip())
b = list(map(int, input().strip().split()))

result = find_possible_eating_order(n, a, k, b)
if result == "NO":
    print("NO")
else:
    print("YES")
    current_index = 1
    for action in result:
        print(f"{current_index} {action}")
        if action == 'L':
            current_index -= 1
        else:
            current_index += 1
