import sys
n, k = map(int, sys.stdin.readline().split())
a = list(map(int, sys.stdin.readline().split()))
total = sum(a)
if total % k != 0:
    print('No')
else:
    print('Yes')
    x = total // k
    b = []
    for i in a:
        if i > x:
            print('No')
            exit()
        b.append(i)
    for i in range(k - 1):
        print(x, end=' ')
        for j in range(n):
            if b[j] <= x:
                b[j] -= x
                print(j + 1, end=' ')
            else:
                break
    print()
