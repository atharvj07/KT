import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String[] cards = br.readLine().split(" ");
        boolean[][] equ = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            equ[i][i] = true;
            char[] f = cards[i].toCharArray();
            for (int j = i + 1; j < n; j++) {
                char[] s = cards[j].toCharArray();
                equ[i][j] = f[0] == s[0] || f[1] == s[1];
            }
        }
        for (int k = n - 1; k >= 0; k--) {
            for (int i = 0; i < n - 1; i++) {
                for (int j = i + 1; j < n; j++) {
                    equ[i][j] |= equ[i][k] && equ[k][j];
                }
            }
        }
        System.out.println(equ[0][n - 1] ? "YES" : "NO");
    }
}
