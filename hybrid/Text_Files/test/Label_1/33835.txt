import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        int r = sc.nextInt();
        int[] a = new int[n + 1];
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            b[i] = sc.nextInt();
        }
        boolean flag = true;
        for (int i = l; i <= r; i++) {
            if (a[i] != b[i]) {
                flag = false;
            }
        }
        if (flag) {
            System.out.println("TRUTH");
        } else {
            System.out.println("LIE");
        }
    }
}
