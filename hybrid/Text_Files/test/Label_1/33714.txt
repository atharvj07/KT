import java.util.Scanner;

class Main {
  static long[] depth;
  static int[] root;

  static long mod = 1000000007;
  static int n, k;
  static boolean[] visited;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    k = sc.nextInt();

    depth = new long[n + 1];
    visited = new boolean[n + 1];
    root = new int[n + 1];

    for (int i = 0; i <= n; i++) root[i] = i;

    for (int i = 0; i < n - 1; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      depth[b] = depth[a] + 1;
      root[find(b)] = find(a);
    }
    long total = powMod(2, k);
    for (int i = 1; i <= n; i++) {
      if (visited[i]) continue;
      long subtreeDepth = findSubtreeDepth(i);
      total = (total + powMod(subtreeDepth, k) - 1 + mod) % mod;
    }

    System.out.println(total);
  }

  static long powMod(long base, long exp) {
    if (exp == 0) return 1;
    if (exp == 1) return base;
    long p = powMod(base, exp / 2);
    p = (p * p) % mod;
    if (exp % 2 == 1) p = (p * base) % mod;
    return p;
  }

  static long findSubtreeDepth(int node) {
    if (visited[node]) return 0;
    visited[node] = true;

    long subtreeDepth = 0;
    for (int i = 1; i <= n; i++) {
      if (find(i) == node) subtreeDepth = Math.max(subtreeDepth, depth[i]);
    }

    return subtreeDepth;
  }

  static int find(int node) {
    if (node == root[node]) return node;
    return root[node] = find(root[node]);
  }
}
