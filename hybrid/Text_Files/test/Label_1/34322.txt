import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        while (t-- > 0) {
            int s = scanner.nextInt();
            int ans = 0;
            while (s >= 10) {
                int x = s / 10 * 10;
                ans += x;
                s -= x;
                s += x / 10;
            }
            ans += s;
            System.out.println(ans);
        }
    }
}
