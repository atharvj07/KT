def find_index(n):
    digits = list(str(n))
    index = 0
    length = len(digits)
    
    for i in range(1, length):
        index += 2**i
    
    for i in range(length):
        if digits[i] == '7':
            index += 2**(length-i-1)
    
    return index + 1

n = int(input().strip())
print(find_index(n))
