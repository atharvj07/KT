#include <iostream>
#include <utility>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        std::cin >> arr[i];
    }

    std::vector<std::pair<int, int>> res;
    int l = 0, r = n - 1, ans = 0;
    while (l <= r) {
        if (arr[l] < arr[r]) {
            if (ans % 2 == 0) {
                res.emplace_back(1, arr[l++]);
            } else {
                res.emplace_back(2, arr[r--]);
            }
        } else {
            if (ans % 2 == 0) {
                res.emplace_back(2, arr[r--]);
            } else {
                res.emplace_back(1, arr[l++]);
            }
        }
        ans++;
    }
    std::cout << (ans % 2 == 0 ? "Alice" : "Bob") << std::endl;

    return 0;
}
