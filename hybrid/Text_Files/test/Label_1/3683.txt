#include <iostream>
#include <algorithm>
#include <vector>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;

    vector<vector<int>> table(n, vector<int>(n));
    int sum = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            table[i][j] = i * n + j + 1;
            if (j == k - 1)
                sum += table[i][j];
        }
    }

    cout << sum << endl;

    for (auto& row : table) {
        for (auto& num : row) {
            cout << num << " ";
        }
        cout << endl;
    }
    return 0;
}
