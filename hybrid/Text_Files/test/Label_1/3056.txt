#include <bits/stdc++.h>
using namespace std;
#define all(a) a.begin(), a.end()
int main() {
  int n, x, y;
  cin >> n;
  vector<vector<int>> G(n + 1);
  for (int i = 0; i < n - 1; ++i) {
    cin >> x >> y;
    G[x].push_back(y);
    G[y].push_back(x);
  }
  vector<int> a(n + 1);
  for (int i = 1; i <= n; ++i) { cin >> a[i]; }
  vector<int> vis(n + 1);
  vis[1] = 1;
  queue<int> Q;
  Q.push(1);
  int idx = 1;
  while (!Q.empty()) {
    int v = Q.front();
    Q.pop();
    if (a[idx++] != v) {
      cout << "No" << endl;
      return 0;
    }
    for (auto &u : G[v])
      if (!vis[u]) {
        Q.push(u);
        vis[u] = 1;
      }
  }
  cout << "Yes" << endl;
  return 0;
}
