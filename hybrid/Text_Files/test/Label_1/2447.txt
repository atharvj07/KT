#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn=2e5+5;
int a[maxn],L,R;
ll ans,pre[maxn];
int main(){
    int n; cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i],a[i+n]=a[i];
    for(int i=1;i<=2*n;i++) pre[i]=pre[i-1]+abs(a[i]-a[i+1]);
    for(int i=1;i<=n;i++){
        int l=i-1,r=i;
        while(r<2*n&&a[r+1]-a[r+2]>0) r++;
        while(l>0&&a[l]-a[l-1]>=a[r+1]-a[r+2]) l--;
        if(pre[r+1]-pre[l]>ans) ans=pre[r+1]-pre[l],L=l,R=r;
    }
    cout<<ans<<'\n';
    return 0;
}
