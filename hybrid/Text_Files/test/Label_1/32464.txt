import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] l = new int[n + 1];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            l[a[i]] = i;
        }
        
        int[] r = new int[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            r[a[i]] = i;
        }
        
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans = Math.max(ans, Math.max(l[i], r[i]));
        }
        System.out.println(ans);
    }
}
