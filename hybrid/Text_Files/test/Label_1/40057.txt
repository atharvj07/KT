import sys
input = sys.stdin.readline

n, q = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
segs = []
for i in range(q):
    segs.append(list(map(int, input().split())))

diffs = [b[i] - a[i] for i in range(n)]
L, R = zip(*segs)

def is_possible(l, r, diff):
    l -= 1
    r -= 1
    if sum(diff[l:r+1]) == 0 and all(x % 2 == 0 for x in diff[l:r+1]):
        return (r - l + 2) // 2
    return -1

for l, r in segs:
    print(is_possible(l, r, diffs))
