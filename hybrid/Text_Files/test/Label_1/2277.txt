#include <iostream>
using namespace std;
int main() {
    long long a, b, w, x, c;
    cin >> a >> b >> w >> x >> c;
    long long ans = 0;
    while (c > a) {
        ans++;
        if (b >= x) {
            b -= x;
        } else {
            a--;
            b = w - (x - b);
        }
        c--;
    }
    cout << ans << endl;
    return 0;
}
