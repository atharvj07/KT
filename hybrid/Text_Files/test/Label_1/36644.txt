for _ in range(int(input())):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    odd = False
    ans = []
    for x in a:
        odd ^= (x % 2 == 1)
    if odd:
        if k % 2 == 0:
            print("NO")
            continue
        ans.append(n)
        k -= 1
    if k == 1:
        print("YES")
        print(n)
        continue
    if k % 2 == 0 or k > n - odd:
        print("NO")
        continue
    k -= 1
    st = 0
    for i in range(n):
        if (a[i] % 2 == odd):
            if i > st:
                ans.append(i)
            st = i
            odd ^= True
            k -= 1
            if k == 0:
                break
    ans.append(n)
    print("YES")
    print(*ans)
