#include<bits/stdc++.h>
using namespace std;
#define pb push_back
#define x first
#define y second
int main() {
    int t; cin >> t;
    while(t--) {
        int n, x; cin >> n >> x;
        vector<int> w(n), ord;
        for(auto &e : w) cin >> e;
        sort(w.begin(), w.end());
        for(int i = n - 1; i >= 0; i--) {
            ord.pb(w[i]);
            int sum = w[i];
            for(int j = i - 1; j >= 0; j--) {
                sum += w[j];
                if(sum == x) {
                    cout << "NO" << endl;
                    goto skip;
                }
                ord.pb(w[j]);
            }
        }
        cout << "YES";
        for(auto e : ord) cout << " " << e;
        cout << endl;
        skip:;
    }
}
