import sys

n = int(input())
a = []
b = []
for i in range(n):
	x, y = map(int, input().split())
	if y < 0:
		x, y = -x, -y
	a.append(x)
	b.append(y)

def ccw(a, b, c):
	return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])

def intersect(a, b, c, d):
	return ccw(a, b, c) * ccw(a, b, d) <= 0 and ccw(c, d, a) * ccw(c, d, b) <= 0

if len(set(a)) == 1:
	print(-1)
else:
	l, r = 0.0, 1e9
	while r - l > 1e-6:
		m = (l + r) / 2
		k = []
		for i in range(n):
			if b[i] > m:
				k.append([a[i], b[i] - m])
			else:
				k.append([a[i], b[i] + m])
		flg = False
		for i in range(n):
			if flg:
				break
			for j in range(i + 1, n):
				if intersect(k[i], k[(i + 1) % n], k[j], k[(j + 1) % n]):
					flg = True
					break
		if flg:
			r = m
		else:
			l = m
	print(l)
