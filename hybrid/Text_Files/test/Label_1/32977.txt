import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] di = new int[n - 1];
        for (int i = 0; i < n - 1; i++) {
            di[i] = in.nextInt();
        }
        int a = in.nextInt();
        int b = in.nextInt();
        int total = 0;
        for (int i = a - 1; i < b - 1; i++) {
            total += di[i];
        }
        System.out.println(total);
    }
}
