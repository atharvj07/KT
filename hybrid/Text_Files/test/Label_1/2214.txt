#include <bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin >> n;
    vector<int> a(n);
    for(int i = 0;i < n;i++){
        cin >> a[i];
    }
    sort(a.begin(), a.end());
    int ans = INT_MAX;
    int t = -1;
    for(int j = 1;j <= 100;j++){
        int sum = 0;
        for(int k = 0;k < n;k++){
            sum += min(abs(j - a[k]), abs(j - 1 - a[k]));
        }
        if(ans > sum){
            ans = sum;
            t = j;
        }
    }
    cout << t << " " << ans << endl;
    return 0;
}
