for _ in range(int(input())):
    n = int(input())
    a = list(map(int,input().split()))
    g = [[] for i in range(n)]
    f = {}
    mx = 0
    for i in range(n):
        if not a[i] in f:
            f[a[i]] = []
        f[a[i]].append(i)
    for i in f:
        f[i].sort()
        for j in range(len(f[i])-1):
            mx = max(mx, f[i][j+1]-f[i][j]-1)
    print(mx)
