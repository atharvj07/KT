#include <bits/stdc++.h>
using namespace std;
int n;
int pre[100005];
int main() {
    cin >> n;
    string s;
    cin >> s;
    for (int i = 0; i < n; ++i) {
        int tmp = (s[i] == '0' ? 1 : -1);
        pre[i + 1] = pre[i] + tmp;
    }
    int mx = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= i; ++j) {
            int tmp = pre[i + 1] - pre[j];
            if (!tmp) mx = max(mx, i - j + 1);
        }
    }
    cout << mx << endl;
}
