l1, r1, l2, r2, k = map(int, input().split())
ans = 0
if l2 <= k:
    if l1 <= l2 <= r1 <= k:
        ans += k - l1
    elif l1 <= k <= r1 <= r2:
        ans += r1 - l1
elif l2 <= r2 <= k:
    if l1 <= l2 <= r1 <= r2:
        ans += r2 - l1
    elif l1 <= k <= r1 <= r2:
        ans += r1 - k
elif k <= l1:
    if l2 <= l1 <= r2 <= r1:
        ans += r2 - l1
    elif l2 <= r1 <= r2 <= l1:
        ans += r1 - l2
elif k <= r1:
    if l2 <= l1 <= r2 <= r1:
        ans += r1 - l1
    elif l2 <= k <= r2 <= r1:
        ans += r2 - k
print(ans)
