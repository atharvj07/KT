import sys
input = sys.stdin.readline

s = []
k = [[] for _ in range(12)]
for _ in range(int(input())):
    t = input().strip()
    s.append(t)
    x = t.count('a') + t.count('e') + t.count('i') + t.count('o') + t.count('u')
    if len(k[x]) < 5:
        k[x].append(t)

ans = []
memo = {}
for i in range(1, 12):
    memo[i] = {}
    for x in range(len(k[i])):
        for y in range(len(k[i])):
            if k[i][x][-1] == k[i][y][-1] and k[i][x] != k[i][y]:
                if (k[i][x], k[i][y]) not in memo[i]: memo[i][k[i][x], k[i][y]] = []
                memo[i][k[i][x], k[i][y]].append((k[i][x], k[i][y]))

for i in range(1, 12):
    for j in range(1, 12):
        if i == j:
            for a in memo[i]:
                for x in memo[i][a]:
                    ans.append((x[0], x[1]))
        else:
            for a in memo[i]:
                for x in memo[i][a]:
                    for b in memo[j]:
                        for y in memo[j][b]:
                            if x[1] == y[0]:
                                ans.append((x[0], y[1]))

print(len(ans))
for x in ans:
    print(x[0], x[1])
