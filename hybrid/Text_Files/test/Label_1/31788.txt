import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] heights = new int[n];
        String[] input = reader.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            heights[i] = Integer.parseInt(input[i]);
        }

        List<Integer> result = new ArrayList<>();
        int minHeight = heights[0];
        int minIndex = 0;
        int maxLength = 1;
        for (int i = 1; i < n; i++) {
            if (heights[i] < minHeight) {
                minHeight = heights[i];
                minIndex = i;
            }
        }

        int left = minIndex - 1;
        int right = minIndex + 1;
        while (left >= 0 || right < n) {
            if (left >= 0 && heights[left] <= minHeight + 1) {
                result.add(heights[left]);
                minHeight = heights[left];
                maxLength++;
                left--;
            } else if (right < n && heights[right] <= minHeight + 1) {
                result.add(heights[right]);
                minHeight = heights[right];
                maxLength++;
                right++;
            } else {
                break;
            }
        }

        Collections.reverse(result);
        result.add(heights[minIndex]);
        System.out.println(maxLength);
        for (int height : result) {
            System.out.print(height + " ");
        }
        System.out.println();
    }
}
