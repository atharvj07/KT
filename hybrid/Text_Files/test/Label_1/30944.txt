import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        if (n == 2 && a[0] == a[1]) {
            System.out.println("BitLGM");
        } else if (a[0] + a[1] < a[2]) {
            System.out.println("BitLGM");
        } else {
            System.out.println("BitAryo");
        }
    }
}
