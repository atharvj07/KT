for _ in range(int(input())):
    a, b, k = map(int, input().split())
    arr1 = list(map(int, input().split()))
    arr2 = list(map(int, input().split()))
    cnt = 0
    for i in range(k):
        for j in range(i + 1, k):
            if arr1[i] != arr1[j] and arr2[i] != arr2[j]:
                cnt += 1
    print(cnt)
