#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int a = 0, b = 0, c = 0;
    for (char x : s) {
      if (x == 'A') a++;
      if (x == 'B') b++;
      if (x == 'C') c++;
    }
    if ((a + b + c) % 2 == 0)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
}
