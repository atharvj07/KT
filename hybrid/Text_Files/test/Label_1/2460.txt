#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &i : a) cin >> i;
    map<int, vector<int>> m;
    for (int i = 0; i < n; i++) m[a[i]].push_back(i);
    int num = 0;
    for (auto &i : m) {
      num = max(num, (int)i.second.size());
    }
    if (num > n / 2) {
      cout << "NO\n";
      continue;
    }
    cout << "YES\n";
    for (auto &i : m) {
      vector<int> &v = i.second;
      for (int j = 1; j < (int)v.size(); j++) cout << v[j - 1] + 1 << " " << v[j] + 1 << "\n";
    }
    for (int i = 0; i < m.size() - 1; i++) {
      cout << *m.begin()->second.begin() + 1 << " " << *next(m.begin())->second.begin() + 1 << "\n";
      m.erase(m.begin());
    }
  }
  return 0;
}
