import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int[] t = new int[4];
    for (int i = 0; i < 4; i++) {
      t[i] = sc.nextInt();
    }
    String a = sc.nextLine().strip();
    while (a.isEmpty()) a = sc.nextLine().strip();
    String b = sc.nextLine().strip();
    while (b.isEmpty()) b = sc.nextLine().strip();
    int n = a.length();
    int[][] dp = new int[n + 1][n + 1];
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= n; j++) {
        if (a.charAt(i - 1) == b.charAt(j - 1))
          dp[i][j] = dp[i - 1][j - 1];
        else
          dp[i][j] = Math.min(Math.min(dp[i - 1][j] + t[1], dp[i][j - 1] + t[0]), Math.min(dp[i - 1][j - 1] + t[2], dp[i][j - 1] + t[3]));
      }
    }
    System.out.println(dp[n][n]);
  }
}
