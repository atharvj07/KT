#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }

  int min_val = *min_element(a.begin(), a.end());
  int ans = n;
  for (int i = 0; i < n; i++) {
    if (a[i] == min_val) {
      for (int j = i + 1; j < n; j++) {
        if (a[j] == min_val) { ans = min(ans, j - i); }
      }
    }
  }
  cout << ans << endl;
}
