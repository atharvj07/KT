#include <bits/stdc++.h>
using namespace std;
const int N = 1e5+10;
int n,m,a,b,vis[N],ans;
vector<int>v[N];
void dfs(int node){
    if(vis[node])
        return;
    vis[node]=1;
    for(int i=0;i<v[node].size();i++){
        int u = v[node][i];
        if(vis[u]==0)
            dfs(u);
    }
}
int main()
{
    cin>>n>>m;
    for(int i=0;i<m;i++){
        cin>>a>>b;
        v[a].push_back(b);
        v[b].push_back(a);
    }
    for(int i=1;i<=n;i++){
        if(vis[i]==0){
            dfs(i);
            ans++;
        }
    }
    cout<<ans<<endl;
    return 0;
}
