import java.util.*;

public class Main {
    static final int MAX_N = 20;
    static final int MAX_T = 10;
    static int[][][][][] dp = new int[MAX_N + 1][MAX_T + 1][4][4][4];

    public static int solve(int n, int t, int v1, int v2, int v3) {
        if (n == 1) return 1;
        if (dp[n][t][v1][v2][v3] != -1) return dp[n][t][v1][v2][v3];
        int ans = 0;
        for (int i = 1; i <= 4; ++i) {
            if (i > v1 && i > v2 && i > v3) {
                if (v2 < v3) {
                    if (v1 == 0 || t == 1 && i == v1 + 1) continue;
                    if (t > 1 && i != v1 + 1) continue;
                    ans += solve(n - 1, t - 1, v1, i, v3);
                } else {
                    if (v1 == 0 || t == 1 && i == v1 + 1) continue;
                    if (t > 1 && i == v1 + 1) continue;
                    ans += solve(n - 1, t - 1, v1, v3, i);
                }
            }
            if (i < v1 && i < v2 && i < v3) {
                if (v2 > v3) {
                    if (v1 == 4 || t == 1 && i == v1 - 1) continue;
                    if (t > 1 && i != v1 - 1) continue;
                    ans += solve(n - 1, t + 1, i, v1, v3);
                } else {
                    if (v1 == 4 || t == 1 && i == v1 - 1) continue;
                    if (t > 1 && i == v1 - 1) continue;
                    ans += solve(n - 1, t + 1, i, v3, v1);
                }
            }
        }
        return dp[n][t][v1][v2][v3] = ans;
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int t = in.nextInt();
        for (int i = 0; i <= n; ++i) {
            for (int j = 0; j <= t; ++j) {
                for (int k = 0; k < 4; ++k) {
                    for (int l = 0; l < 4; ++l) {
                        for (int m = 0; m < 4; ++m) {
                            dp[i][j][k][l][m] = -1;
                        }
                    }
                }
            }
        }
        System.out.println(solve(n, t, 0, 0, 0));
    }
}
