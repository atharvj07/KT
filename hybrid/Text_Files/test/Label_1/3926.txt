#include <bits/stdc++.h>
using namespace std;
int n, m, k, a[105][105][105], d[7], ans;
int main() {
    cin >> n >> m >> k;
    for (int l = 1; l <= n; l++) {
        cin.ignore();
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= k; j++) { cin >> a[l][i][j]; }
        }
    }
    for (int l = 1; l <= n; l++) {
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= k; j++) {
                if (!a[l][i][j]) {
                    d[1] = d[2] = d[3] = 0;
                    if (i > 1) d[1] = a[l][i - 1][j];
                    if (j < k) d[2] = a[l][i][j + 1];
                    if (l < n) d[3] = a[l + 1][i][j];
                    int p = d[1] + d[2] + d[3];
                    if (p == 1) ans++;
                }
            }
        }
    }
    cout << ans << endl;
}
