import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
m = int(input())

dp = [[[-1]*(n+1) for _ in range(n+1)] for _ in range(m+1)]
dp[0][0][0] = 0

def getpos(k, l, r, p, x):
    if dp[k][l][r] != -1:
        return dp[k][l][r]
    dp[k][l][r] = x
    if l == r:
        return a[l-1]
    if k == p:
        return dp[k][l+1][r]
    op1 = getpos(k, l+1, r, p, x)
    op2 = getpos(k-p, l+1, r, p, x+a[l])
    if op1 > op2:
        dp[k][l][r] = op1
    else:
        dp[k][l][r] = op2
    return dp[k][l][r]

for i in range(1,m+1):
    k,p = map(int, input().split())
    print(getpos(k,1,n,p,0))
