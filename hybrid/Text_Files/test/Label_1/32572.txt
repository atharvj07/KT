import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int M = sc.nextInt();
        int R = sc.nextInt();
        int Q = sc.nextInt();
        List<int[]> rules = new ArrayList<>();
        for (int i = 0; i < M; i++) {
            int[] rule = new int[3];
            rule[0] = sc.nextInt();
            rule[1] = sc.nextInt();
            rule[2] = sc.nextInt();
            rules.add(rule);
        }
        int[] reward = new int[Q];
        int[] x = new int[Q];
        int[] y = new int[Q];
        int[] z = new int[Q];
        for (int i = 0; i < Q; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
            z[i] = sc.nextInt();
            reward[i] = x[i] % R == y[i] % R ? z[i] : 0;
        }
        int[][] dp = new int[Q + 1][N + 1];
        for (int i = 1; i <= Q; i++) {
            for (int j = 0; j <= N; j++) {
                dp[i][j] = dp[i - 1][j];
                for (int[] rule : rules) {
                    if (j == rule[0]) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][rule[1]] + reward[i - 1] - rule[2]);
                    }
                }
            }
        }
        System.out.println(dp[Q][0]);
    }
}
