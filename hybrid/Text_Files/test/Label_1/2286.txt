#include <bits/stdc++.h>

typedef long long ll;
typedef pair<int, int> pi;
#define FOR(i, a, b) for (int i = (a); i < (b); i++)
#define F0R(i, a) for (int i = 0; i < (a); i++)
#define F0Rd(i, a) for (int i = (a) - 1; i >= 0; i--)
#define trav(a, x) for (auto& a : x)

void setIO(string name) {
  freopen((name + ".in").c_str(), "r", stdin);
  freopen((name + ".out").c_str(), "w", stdout);
}

const int MOD = 1e9 + 7;
const int INF = 1e9;
const ll LINF = 1e18;

int n;
ll a[100001], psa[100001];

int main() {
  // setIO("name");
  cin >> n;
  FOR(i, 0, n) { cin >> a[i]; }
  F0R(i, n) psa[i + 1] = psa[i] + a[i];
  ll ans = LLONG_MAX;
  FOR(i, 0, n) {
    ll suml = psa[min(n - 1, i + 1)] - psa[i - 1];
    ll sumr = psa[max(0, n - 1 - i)] - psa[n - i - 1];
    ans = min(ans, i * i + suml * suml + sumr * sumr);
  }
  cout << ans << endl;
  return 0;
}
