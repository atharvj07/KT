import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int m = in.nextInt();
    int[] w = new int[m];
    for (int i = 0; i < m; i++) {
      w[i] = in.nextInt();
    }
    Arrays.sort(w);
    int max = 0;
    int[] d = new int[n + 1];
    for (int i = 0; i < m; i++) {
      int cw = w[i];
      for (int j = 1; j <= n; j++) {
        if (d[j] == 0) continue;
        d[j] = cw;
        max = Math.max(max, d[j]);
      }
      int lw = i > 0 ? w[i - 1] : 0;
      for (int j = 1; j <= n; j++) {
        d[j] = Math.max(d[j], lw);
      }
    }
    System.out.println(max);
  }
}
