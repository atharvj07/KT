import java.util.Scanner;
import java.math.BigInteger;

public class Main {
    static final long MOD = 1000000007;
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
            b[i] = in.nextInt();
        }
        long[][] dp = new long[n + 1][2];
        dp[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                if (a[i] * a[j] + b[i] * b[j] == 0) {
                    dp[i + 1][1] += dp[i][0];
                    dp[i + 1][0] += dp[i][1];
                } else {
                    dp[i + 1][0] += dp[i][0];
                    dp[i + 1][1] += dp[i][1];
                }
            }
            dp[i + 1][0] %= MOD;
            dp[i + 1][1] %= MOD;
        }
        System.out.println(dp[n][0]);
    }
}
