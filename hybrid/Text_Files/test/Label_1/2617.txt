#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> d(n);
  for (int i = 0; i < n; i++) cin >> d[i];
  if (k + 1 < n) {
    cout << -1 << endl;
    return 0;
  }
  vector<pair<int, int>> ans;
  for (int i = 1; i <= n; i++) {
    for (int j = i + 1; j <= n; j++) {
      if (d[j - 1] == d[i - 1] + 1) {
        ans.emplace_back(i, j);
        k--;
        if (k < 0) {
          cout << -1 << endl;
          return 0;
        }
      }
    }
  }
  cout << ans.size() << endl;
  for (auto &e : ans) cout << e.first << " " << e.second << endl;
}
