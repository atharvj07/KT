import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] data = new int[4][2];
        while (scanner.hasNext()) {
            double l = scanner.nextDouble();
            double r = scanner.nextDouble();
            int[] e = getJudgment(l, r);
            data[e[0]][0]++;
            data[e[1]][1]++;
        }
        scanner.close();
        for (int i = 0; i < data.length; i++) {
            System.out.println(data[i][0] + " " + data[i][1]);
        }
    }

    public static int[] getJudgment(double l, double r) {
        int[] e = new int[2];
        if (l >= 1.1) {
            e[0] = 0;
        } else if (l >= 0.6) {
            e[0] = 1;
        } else if (l >= 0.2) {
            e[0] = 2;
        } else {
            e[0] = 3;
        }
        if (r >= 1.1) {
            e[1] = 0;
        } else if (r >= 0.6) {
            e[1] = 1;
        } else if (r >= 0.2) {
            e[1] = 2;
        } else {
            e[1] = 3;
        }
        return e;
    }
}
