#include<bits/stdc++.h>
using namespace std;
#define pii pair<int,int>
#define pb push_back
#define fr first
#define se second
typedef long long ll;

const int maxn = 1e6 + 10;
int a[maxn];
ll p[maxn];
const double eps = 1e-4;
const double mod = 1e9 + 7;

int32_t main()
{
    int n;
    cin >> n;
    for(int i = 1;i <= n;i++)
        cin >> a[i];
    for(int i = 1;i <= n;i++)
        p[i] = p[i - 1] + a[i];
    double ans = 0;
    for(int i = 1;i <= n;i++)
    {
        for(int j = i + 1;j <= n;j++)
        {
            int now = p[j] - p[i - 1];
            if(now > j - i + 1)
                now = j - i + 1;
            if(i == j)
                now = 1;
            ans += (double)now / (j - i + 1.0);
        }
    }
    cout << fixed << setprecision(6) << ans << endl;
    return 0;
}
