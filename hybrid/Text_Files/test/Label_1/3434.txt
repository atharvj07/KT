#include <iostream>

int main() {
  int a1, b1, a2, b2, a3, b3;
  std::cin >> a1 >> b1 >> a2 >> b2 >> a3 >> b3;
  bool hor = (a2 <= a1 && b2 <= b1) || (a2 <= b1 && b2 <= a1);
  bool ver = (a3 <= a1 && b3 <= b1) || (a3 <= b1 && b3 <= a1);
  std::cout << ((hor || ver) ? "YES" : "NO") << "\n";
  return 0;
}
