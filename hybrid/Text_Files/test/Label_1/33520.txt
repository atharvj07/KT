import java.util.ArrayDeque;
import java.util.Queue;
import java.util.Scanner;

public class GeoPanel {

    private static final int TRANSPARENT = 0;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine();

        int[][] panels = new int[n][m];
        int[][] symbols = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                panels[i][j] = scanner.nextInt();
            }
            scanner.nextLine();
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                symbols[i][j] = scanner.nextInt();
            }
            scanner.nextLine();
        }

        int x = scanner.nextInt() - 1;
        int y = scanner.nextInt() - 1;

        System.out.println(repaint(panels, symbols, x, y));
    }

    private static int repaint(int[][] panels, int[][] symbols, int x, int y) {
        int repaintCount = 0;
        Queue<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{x, y});

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int currentX = current[0];
            int currentY = current[1];
            int currentColor = panels[currentX][currentY];

            if (currentColor != TRANSPARENT && currentColor != symbols[currentX][currentY]) {
                repaintCount++;
                for (int i = 1; ; i++) {
                    for (int j = 0; j < i; j++) {
                        int nextX = currentX + i;
                        int nextY = currentY + j;
                        if (isValid(panels, symbols, nextX, nextY) && panels[nextX][nextY] == currentColor) {
                            panels[nextX][nextY] = symbols[currentX][currentY];
                            if (symbols[nextX][nextY] != TRANSPARENT) {
                                queue.offer(new int[]{nextX, nextY});
                                symbols[nextX][nextY] = TRANSPARENT;
                            }
                        }

                        nextX = currentX - i;
                        nextY = currentY - j;
                        if (isValid(panels, symbols, nextX, nextY) && panels[nextX][nextY] == currentColor) {
                            panels[nextX][nextY] = symbols[currentX][currentY];
                            if (symbols[nextX][nextY] != TRANSPARENT) {
                                queue.offer(new int[]{nextX, nextY});
                                symbols[nextX][nextY] = TRANSPARENT;
                            }
                        }

                        nextX = currentX + j;
                        nextY = currentY - i;
                        if (isValid(panels, symbols, nextX, nextY) && panels[nextX][nextY] == currentColor) {
                            panels[nextX][nextY] = symbols[currentX][currentY];
                            if (symbols[nextX][nextY] != TRANSPARENT) {
                                queue.offer(new int[]{nextX, nextY});
                                symbols[nextX][nextY] = TRANSPARENT;
                            }
                        }

                        nextX = currentX - j;
                        nextY = currentY + i;
                        if (isValid(panels, symbols, nextX, nextY) && panels[nextX][nextY] == currentColor) {
                            panels[nextX][nextY] = symbols[currentX][currentY];
                            if (symbols[nextX][nextY] != TRANSPARENT) {
                                queue.offer(new int[]{nextX, nextY});
                                symbols[nextX][nextY] = TRANSPARENT;
                            }
                        }
                    }
                }
            }
        }

        return repaintCount;
    }

    private static boolean isValid(int[][] panels, int[][] symbols, int x, int y) {
        return x >= 0 && x < panels.length && y >= 0 && y < panels[0].length;
    }
}
