from collections import deque
from math import gcd

def sign(x):
    if x > 0:
        return 1
    if x < 0:
        return -1
    return 0

def solve():
    x1, y1 = map(int, input().split())
    x2, y2 = map(int, input().split())
    n = int(input())
    d = []; m = []; dx = []; dy = []
    for i in range(n):
        a, b, c = map(int, input().split())
        d.append(a * y1 - b * x1 - c)
        m.append(abs(gcd(a, b)))
        dx.append(sign(a))
        dy.append(sign(b))
    d.append(a * y2 - b * x2 - c)
    m.append(abs(gcd(a, b)))
    dx.append(sign(a))
    dy.append(sign(b))
    vis = [False] * (len(d))
    q = deque([(x1, y1)])
    ans = 0
    while q:
        x, y = q.popleft()
        if x == x2 and y == y2:
            ans -= 1
            break
        for i in range(len(d)):
            if vis[i]:
                continue
            if m[i] != 0:
                if (d[i] - dx[i] * x - dy[i] * y) % m[i] == 0:
                    q.append((x + dx[i], y + dy[i]))
                    vis[i] = True
        ans += 1
    print(ans)

solve()
