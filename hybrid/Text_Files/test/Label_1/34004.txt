import java.util.Collections;
import java.util.Scanner;
import java.util.Vector;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        Vector<Integer> v = new Vector<>();
        for (int i = 0; i < n; i++) {
            int h = input.nextInt();
            int m = input.nextInt();
            v.add(h * 60 + m);
        }
        Collections.sort(v);
        int ans = 0, cnt = 0;
        for (int x : v) {
            if (x >= ans) {
                cnt++;
            } else {
                cnt--;
            }
            ans++;
            if (cnt > ans) {
                ans = cnt;
            }
        }
        System.out.println(ans);
    }
}
