#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll n,k,pow2[61]={},x;
bool f;
int main(){
    pow2[0]=1;
    for(int i=1;i<=60;++i)pow2[i]=pow2[i-1]*2;
    cin>>n>>k;
    if(n==1){
        for(int i=0;i<k;++i)cout<<"-1 ";
        return 0;
    }
    for(int i=0;i<=60;++i){
        if(pow2[i]>n)continue;
        if((n-pow2[i])%pow2[i+1]==0){
            x=(n-pow2[i])/pow2[i+1];
            f=true;
            break;
        }
    }
    if(!f){cout<<"No";return 0;}
    cout<<"Yes\n";
    for(int i=0;i<k-1;++i)cout<<x<<" ";
    cout<<x+x<<endl;
    return 0;
}
