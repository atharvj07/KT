from sys import stdin

n = int(stdin.readline())
a = list(map(int, stdin.readline().split()))
a = [201 if i == -1 else i for i in a]
mod = 998244353
dp = [[[0] * 3 for _ in range(202)] for _ in range(n + 1)]
dp[n][a[n-1] - 1][0] = 1
for i in range(n - 1, -1, -1):
    if a[i] == 201:
        for lft in range(a[i + 1]):
            for pr in range(3):
                dp[i][lft][pr] += dp[i + 1][lft][pr] + dp[i + 1][a[i + 1]][1] + dp[i + 1][max(lft, a[i + 1])][2]
    else:
        for lft in range(a[i]):
            dp[i][lft][0] = ((dp[i][lft][0] + dp[i + 1][lft][0]) % mod + dp[i + 1][a[i]][1]) % mod
        dp[i][a[i] - 1][1] = ((dp[i][a[i] - 1][1] + dp[i + 1][a[i]][0]) % mod + dp[i + 1][a[i]][1]) % mod
        dp[i][a[i]][2] = ((dp[i][a[i]][2] + dp[i + 1][a[i]][0]) % mod + dp[i + 1][a[i]][2]) % mod
print(dp[0][0][0])
