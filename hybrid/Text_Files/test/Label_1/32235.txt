import java.util.PriorityQueue;
import java.util.Scanner;

public class SharkValerie {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int k = scanner.nextInt();

        PriorityQueue<Tooth> teeth = new PriorityQueue<>();
        for (int i = 0; i < n; i++) {
            int row = scanner.nextInt();
            int viability = scanner.nextInt();
            teeth.offer(new Tooth(row, viability));
        }

        int eaten = 0;
        while (!teeth.isEmpty() && eaten < k) {
            Tooth tooth = teeth.poll();
            int remainingViability = tooth.viability - 1;
            if (remainingViability > 0) {
                teeth.offer(new Tooth(tooth.row, remainingViability));
            }
            eaten++;
        }

        System.out.println(eaten);
    }

    private static class Tooth implements Comparable<Tooth> {
        int row;
        int viability;

        public Tooth(int row, int viability) {
            this.row = row;
            this.viability = viability;
        }

        @Override
        public int compareTo(Tooth other) {
            return Integer.compare(this.viability, other.viability);
        }
    }
}
