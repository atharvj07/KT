import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String fyodor = input.nextLine();
        String matroskin = input.nextLine();
        String sharic = input.nextLine();
        if (fyodor.equals(matroskin) && matroskin.equals(sharic)) {
            System.out.println("?");
        } else if (fyodor.equals("paper") && matroskin.equals("scissors") && sharic.equals("rock")) {
            System.out.println("F");
        } else if (matroskin.equals("paper") && sharic.equals("scissors") && fyodor.equals("rock")) {
            System.out.println("M");
        } else if (sharic.equals("paper") && fyodor.equals("scissors") && matroskin.equals("rock")) {
            System.out.println("S");
        } else {
            System.out.println("?");
        }
    }
}
