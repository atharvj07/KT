import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        line = reader.readLine().split(" ");
        int sx = Integer.parseInt(line[0]);
        int sy = Integer.parseInt(line[1]);
        int fx = Integer.parseInt(line[2]);
        int fy = Integer.parseInt(line[3]);
        List<Location> locations = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            line = reader.readLine().split(" ");
            locations.add(new Location(Integer.parseInt(line[0]), Integer.parseInt(line[1])));
        }
        System.out.println(findMinTime(n, m, sx, sy, fx, fy, locations));
    }

    public static int findMinTime(int n, int m, int sx, int sy, int fx, int fy, List<Location> locations) {
        int[][] dp = new int[n + 1][n + 1];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                dp[i][j] = Integer.MAX_VALUE;
            }
        }
        dp[sx][sy] = 0;
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                if (dp[i][j] != Integer.MAX_VALUE) {
                    if (i + 1 <= n) {
                        dp[i + 1][j] = min(dp[i + 1][j], dp[i][j] + 1);
                    }
                    if (j + 1 <= n) {
                        dp[i][j + 1] = min(dp[i][j + 1], dp[i][j] + 1);
                    }
                    if (i - 1 >= 0) {
                        dp[i - 1][j] = min(dp[i - 1][j], dp[i][j] + 1);
                    }
                    if (j - 1 >= 0) {
                        dp[i][j - 1] = min(dp[i][j - 1], dp[i][j] + 1);
                    }
                    for (Location location : locations) {
                        if (location.x == i || location.y == j) {
                            dp[location.x][location.y] = min(dp[location.x][location.y], dp[i][j]);
                        }
                    }
                }
            }
        }
        if (dp[fx][fy] == Integer.MAX_VALUE) {
            return -1;
        } else {
            return dp[fx][fy];
        }
    }

    public static int min(int a, int b) {
        return a < b ? a : b;
    }

    public static class Location {

        public int x;
        public int y;

        public Location(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
