#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int maxn = 1e5 + 5;

int n, a[maxn], sum[maxn], f[maxn];

int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i <= n; i++) { sum[i] = sum[i - 1] + abs(a[i] - a[i - 1]); }
    for (int i = 1; i <= n; i++) {
        f[i] = 1e9;
        for (int j = i + 1; j <= n; j++) {
            f[i] = min(f[i], (i - j) * (i - j) + (sum[j] - sum[i - 1]) * (sum[j] - sum[i - 1]));
        }
    }
    cout << *min_element(f + 1, f + n + 1) << endl;
    return 0;
}
