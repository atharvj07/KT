def get_digit(ch):
    if ch>='0' and ch<='9':
        return ord(ch)-ord('0')
    return ord(ch)-ord('A')+10

def get_integer(arr, b):
    res = 0
    for i in range(len(arr)-1,-1,-1):
        res = res*b+get_digit(arr[i])
    return res

n, bx = input().split()
n = int(n)
arr = input().split()
x = get_integer(arr,int(bx))

m, by = input().split()
m = int(m)
arr = input().split()
y = get_integer(arr,int(by))

if x==y: print('=')
elif x<y: print('<')
else: print('>')
