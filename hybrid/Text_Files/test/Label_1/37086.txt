n = int(input())
b = list(map(int, input().split()))
if n == 1 or n == 2:
    print(0)
else:
    ans = 0
    for i in range(1, n):
        if b[i] != b[i - 1]:
            if b[i] - b[i - 1] > 1:
                ans = -1
                break
            else:
                ans += 1
    print(ans)
