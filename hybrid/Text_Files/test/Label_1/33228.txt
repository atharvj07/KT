import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.Queue;

public class GeoPanel {

    private static final int EMPTY = -1;

    private static int[][] map;
    private static int[][] symbols;
    private static int n, m;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        n = Integer.parseInt(line[0]);
        m = Integer.parseInt(line[1]);
        map = new int[n][m];
        symbols = new int[n][m];
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                map[i][j] = Integer.parseInt(line[j]);
            }
        }
        for (int i = 0; i < n; i++) {
            line = reader.readLine().split(" ");
            for (int j = 0; j < m; j++) {
                symbols[i][j] = Integer.parseInt(line[j]);
            }
        }
        line = reader.readLine().split(" ");
        int x = Integer.parseInt(line[0]) - 1;
        int y = Integer.parseInt(line[1]) - 1;
        int color = symbols[x][y];
        symbols[x][y] = EMPTY;
        Queue<Integer> queue = new ArrayDeque<>();
        queue.add(color);
        int repaints = 0;
        while (!queue.isEmpty()) {
            int currentColor = queue.poll();
            if (currentColor == EMPTY) {
                continue;
            }
            for (int i = x - 1; i <= x + 1; i++) {
                for (int j = y - 1; j <= y + 1; j++) {
                    if (i >= 0 && i < n && j >= 0 && j < m && map[i][j] == currentColor && symbols[i][j] != EMPTY) {
                        queue.add(symbols[i][j]);
                        symbols[i][j] = EMPTY;
                        repaints++;
                    }
                }
            }
        }
        System.out.println(repaints);
    }
}
