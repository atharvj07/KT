import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt(), d = 0, ans = 0;
        int[] p = new int[n+1], d1 = new int[n+1];
        ArrayList<Integer>[] a = new ArrayList[n+1];
        for(int i = 1; i <= n; i++) a[i] = new ArrayList();
        for(int i = 2; i <= n; i++) {
            p[i] = sc.nextInt();
            a[p[i]].add(i);
        }
        Queue<Integer> q = new LinkedList();
        q.offer(1);
        while(!q.isEmpty()) {
            int u = q.poll();
            d = d1[u] + 1;
            if(a[u].size() == 0) {
                ans = Math.max(ans, d-1);
            } else if(a[u].size() == 1) {
                ans = Math.max(ans, d);
            } else {
                for(int v: a[u]) {
                    d1[v] = d;
                    q.offer(v);
                }
            }
        }
        ans += k+1;
        System.out.println(ans);
    }
}
