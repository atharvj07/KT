#include <iostream>
#include <vector>

using namespace std;

template <typename T>
struct SegmentTree {
  size_t sz;
  vector<T> tree;

  SegmentTree(size_t n, const T& init = 0) : sz(n), tree(2 * n, init) {}

  void set(size_t i, T val) {
    tree[sz + i] = val;
  }

  void build() {
    for (int i = sz - 1; i > 0; --i) {
      tree[i] = tree[2 * i] | tree[2 * i + 1];
    }
  }

  void update(size_t ind, T val) {
    tree[sz + ind] = val;
    for (size_t i = (sz + ind) / 2; i > 0; i /= 2) {
      tree[i] = tree[2 * i] | tree[2 * i + 1];
    }
  }

  T get(size_t start, size_t end) {
    T res = 0;
    for (start += sz, end += sz; start <= end; start /= 2, end /= 2) {
      if (start % 2) res |= tree[start++];
      if (end % 2 == 0) res |= tree[end--];
    }
    return res;
  }
};

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, q;
  cin >> n >> q;
  vector<int> a(1 << n, 0);
  for (int i = 0; i < (1 << n); ++i) cin >> a[i];

  SegmentTree<int> tree(1 << n);
  for (int i = 0; i < (1 << n); ++i) {
    tree.set(i, a[i]);
  }
  tree.build();

  while (q--) {
    int p, b;
    cin >> p >> b;
    tree.update(p - 1, b);
    cout << tree.get(0, (1 << n) - 1) << endl;
  }
}
