#include <bits/stdc++.h>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long int n, k;
    cin >> n >> k;
    long long int a = k % n;
    if (a == 0)
      a = n;
    if (a < ((n + 1) / 2)) {
      cout << n - a + 1 << endl;
    } else {
      if ((n + 1) % 2 == 0)
        a = a - (n + 1) / 2 + 1;
      else
        a = a - (n + 1) / 2;
      cout << a + 1 << endl;
    }
  }
  return 0;
}
