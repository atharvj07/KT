import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    static final long MODULO = 1000000007L;
    static int n, k;
    static long[][] dp;
    static List<int[]>[] adj;

    public static void main(String[] args) {
        try (var br = new BufferedReader(new InputStreamReader(System.in))) {
            String[] line = br.readLine().split(" ");
            n = Integer.parseInt(line[0]);
            k = Integer.parseInt(line[1]);

            adj = new List[n + 1];
            for (int i = 0; i <= n; i++)
                adj[i] = new ArrayList<>();

            for (int i = 0; i < n - 1; i++) {
                line = br.readLine().split(" ");
                int u = Integer.parseInt(line[0]);
                int v = Integer.parseInt(line[1]);
                int color = Integer.parseInt(line[2]);

                adj[u].add(new int[]{v, color});
                adj[v].add(new int[]{u, color});
            }

            dp = new long[n + 1][k + 1];
            System.out.println(memo(1, 0, -1));
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    static long memo(int node, int seq, int parent) {
        if (seq == k)
            return 1L;

        if (dp[node][seq] != 0)
            return dp[node][seq];

        long ans = 0;
        for (int[] to : adj[node]) {
            if (to[0] == parent)
                continue;

            ans = (ans + memo(to[0], seq + 1, node) * (to[1] + 1)) % MODULO;
        }

        return dp[node][seq] = ans;
    }
}
