#include <bits/stdc++.h>

using namespace std;
using ll = long long;
#define MOD 1000000007

int main() {
  int n, m;
  cin >> n >> m;
  unordered_map<int, vector<pair<int, int>>> adj;
  while (m--) {
    int u, v, w;
    cin >> u >> v >> w;
    adj[u].emplace_back(v, w);
    adj[v].emplace_back(u, w);
  }
  ll ans = 1;
  for (int i = 1; i <= n; i++) {
    unordered_set<int> love, hate;
    for (auto &[u, w] : adj[i]) {
      if (w == 0)
        hate.insert(u);
      else
        love.insert(u);
    }
    ans = ans * (love.size() + hate.size()) % MOD;
  }
  cout << ans << endl;
  return 0;
}
