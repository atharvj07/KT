#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> p(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    int c = 0;
    for (int i = 0; i < n; i++) {
        if (p[i] <= i + 1) c++;
    }
    vector<int> ans(n + 2);
    ans[0] = c;
    for (int i = 0; i < n; i++) {
        if (c && p[i] > i + 1) c--;
        else if (!c && p[i] <= i + 1) c++;
        ans[i + 1] = c;
    }
    for (int i = 0; i <= n; i++) cout << ans[i] << " ";
    return 0;
}
