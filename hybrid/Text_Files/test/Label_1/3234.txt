#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<long long> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  while (k--) {
    long long min = *min_element(a.begin(), a.end());
    if (min == 0) break;
    cout << min << "\n";
    for (int i = 0; i < n; i++) a[i] -= min;
  }
  return 0;
}
