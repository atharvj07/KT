import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int K = sc.nextInt();
        int i = 0;
        while (true) {
            i++;
            if (i % K == 0) {
                break;
            }
            i *= 10;
            i += 7;
            i %= K;
        }
        System.out.println(i);
    }
}
