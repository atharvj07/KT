import sys

n = int(input())
a = list(map(int, input().split()))
ans = cnt = 0
b = [0] * n
lst = -1
for i in range(n):
    if i == 0 or i == n - 1:
        continue
    if a[i] < min(a[i - 1], a[i + 1]):
        cnt += 1
        b[i] = 1
        lst = i
ans = max(ans, cnt)
for i in range(n):
    if b[i]:
        b[i] = lst
    lst = lst - 1 if lst > 0 else lst
print(ans)
print(" ".join(map(str, b)))
