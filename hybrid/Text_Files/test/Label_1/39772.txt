for _ in range(int(input())):
    n,a = int(input()),list(map(int,input().split()))
    drench = [False for i in range(n)]
    cur = 0
    for i in range(n):
        cur += a[i]
        drench[max(0,i-cur+1):] = [True] * min(n-max(0,i-cur+1),cur)
    print(' '.join(['1' if x else '0' for x in drench]))
