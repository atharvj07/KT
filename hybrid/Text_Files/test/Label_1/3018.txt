#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+5;
#define ll long long
int n, m;
pair<string,int> jiro[N];
int ciel[N];
ll dp[N][2];
ll fun(int i, int flag){
	if(i == n+1) return 0;
	if(dp[i][flag]!=-1) return dp[i][flag];
	ll ans = 0;
	if(flag == 0){
		if(jiro[i].first == "ATK"){
			if(ciel[i] >= jiro[i].second){
			ll v = fun(i+1,flag);
			if(v != -1) ans = jiro[i].second - ciel[i] + v;
			}
			ll v = fun(i+1,1);
			if(v != -1) ans = max(ans, v);
		}
		else{
			if(ciel[i] > jiro[i].second){
				ll v = fun(i+1,flag);
				if(v != -1) ans = v;
			}
			ll v = fun(i+1,1);
			if(v != -1) ans = max(ans, v);
		}
	}
	else{
		ans = fun(i+1,0);
		if(jiro[i].first == "ATK"){
			if(ciel[i] >= jiro[i].second){
			ll v = fun(i+1,1);
			if(v != -1) ans = max(ans, jiro[i].second - ciel[i] + v);
			}
		}
		else{
			if(ciel[i] > jiro[i].second){
				ll v = fun(i+1,1);
				if(v != -1) ans = max(ans, v);
			}
		}
	}
	return dp[i][flag] = ans;
}
int main(){
	memset(dp,-1,sizeof dp);
	cin >> n >> m;
	for(int i = 1; i <= n; i++){
		string s;
		cin >> s >> jiro[i].second;
		jiro[i].first = s;
	}
	for(int i = 1; i <= m; i++){
		cin >> ciel[i];
	}
	cout << fun(1,0) << "\n";
	return 0;
}
