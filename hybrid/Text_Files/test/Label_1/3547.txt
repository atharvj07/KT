#include <iostream>
using namespace std;
int main() {
    int t;
    cin >> t;
    for (int tt = 0; tt < t; tt++) {
        long long x, y;
        cin >> x >> y;
        long long ans = 0;
        for (long long b = 2; b <= y; b++) {
            long long a = b * ((x / b) + 1);
            if (a <= x) ans += (x - a) / b + 1;
        }
        cout << ans << endl;
    }
    return 0;
}
