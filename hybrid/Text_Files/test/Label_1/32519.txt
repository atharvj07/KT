import java.util.Scanner;

class Main {

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();

    int[][] points = new int[n][2];
    for (int i = 0; i < n; i++) {
      points[i][0] = scanner.nextInt();
      points[i][1] = scanner.nextInt();
    }

    int[] result = findTriangle(n, points);
    System.out.println(result[0] + " " + result[1] + " " + result[2]);
    scanner.close();
  }

  private static int[] findTriangle(int n, int[][] points) {
    // Calculate the area of the triangle formed by three points using cross product.
    int area = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        for (int k = j + 1; k < n; k++) {
          area = Math.abs(points[i][0] * (points[j][1] - points[k][1])
              + points[j][0] * (points[k][1] - points[i][1])
              + points[k][0] * (points[i][1] - points[j][1]));

          // Check if the area is positive and all other points are outside the triangle.
          if (area > 0) {
            boolean isOutside = true;
            for (int l = 0; l < n; l++) {
              if (l == i || l == j || l == k) {
                continue;
              }

              int x1 = points[l][0] - points[i][0];
              int y1 = points[l][1] - points[i][1];
              int x2 = points[j][0] - points[i][0];
              int y2 = points[j][1] - points[i][1];
              int x3 = points[k][0] - points[i][0];
              int y3 = points[k][1] - points[i][1];

              int dot1 = x1 * x2 + y1 * y2;
              int dot2 = x1 * x3 + y1 * y3;
              int dot3 = x2 * x3 + y2 * y3;

              if (dot1 >= 0 && dot2 >= 0 && dot3 >= 0) {
                isOutside = false;
                break;
              }
            }

            if (isOutside) {
              return new int[]{i + 1, j + 1, k + 1};
            }
          }
        }
      }
    }

    return new int[]{-1, -1, -1};
  }
}
