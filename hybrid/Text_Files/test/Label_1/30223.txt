import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class Main {
    private static final long MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        long[][] pairs = new long[n][2];
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            pairs[i][0] = Long.parseLong(line[0]);
            pairs[i][1] = Long.parseLong(line[1]);
        }

        // Sort the pairs by their first element
        Arrays.sort(pairs, (a, b) -> Long.compare(a[0], b[0]));

        // Create a map to store the frequency of each first element
        Map<Long, Integer> frequencyMap = new HashMap<>();
        for (long[] pair : pairs) {
            frequencyMap.put(pair[0], frequencyMap.getOrDefault(pair[0], 0) + 1);
        }

        // Create an array to store the cumulative frequency of each first element
        int[] cumulativeFrequency = new int[n];
        cumulativeFrequency[0] = frequencyMap.get(pairs[0][0]);
        for (int i = 1; i < n; i++) {
            cumulativeFrequency[i] = cumulativeFrequency[i - 1] + frequencyMap.get(pairs[i][0]);
        }

        // Create an array to store the number of ways to choose a set of sardines with a given first element
        long[] waysToChoose = new long[n];
        waysToChoose[0] = 1;
        for (int i = 1; i < n; i++) {
            // If the first element of the current pair is the same as the first element of the previous pair,
            // then the number of ways to choose a set of sardines with the current first element is equal to
            // the number of ways to choose a set of sardines with the previous first element
            if (pairs[i][0] == pairs[i - 1][0]) {
                waysToChoose[i] = waysToChoose[i - 1];
            } else {
                // Otherwise, the number of ways to choose a set of sardines with the current first element is equal to
                // the number of ways to choose a set of sardines with the previous first element plus the number of
                // ways to choose a set of sardines with the current first element and all of the sardines with its
                // first element
                waysToChoose[i] = (waysToChoose[i - 1] + cumulativeFrequency[i]) % MOD;
            }
        }

        // Output the number of ways to choose a set of sardines modulo 1000000007
        System.out.println(waysToChoose[n - 1]);
    }
}
