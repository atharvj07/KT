#include <iostream>
#include <vector>
#include <cmath>

using namespace std;

int gcd(int a, int b) {
    if (b == 0)
        return a;
    return gcd(b, a % b);
}

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++)
        cin >> a[i];
    vector<pair<int, int>> goodPairs(m);
    for (int i = 0; i < m; i++) {
        cin >> goodPairs[i].first >> goodPairs[i].second;
        goodPairs[i].first--;
        goodPairs[i].second--;
    }
    int ans = 0;
    for (auto p : goodPairs) {
        int g = gcd(a[p.first], a[p.second]);
        if (g > 1) {
            ans++;
            a[p.first] /= g;
            a[p.second] /= g;
        }
    }
    cout << ans << endl;
    return 0;
}
