import java.util.Scanner;
import java.util.Arrays;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n];
    long[] pre = new long[n];
    long[] suf = new long[n];
    for (int i = 0; i < n; i++) {
      a[i] = in.nextInt();
    }
    pre[0] = a[0];
    suf[n-1] = a[n-1];
    for (int i = 1; i < n; i++) {
      pre[i] = Math.max(pre[i-1], a[i]);
    }
    for (int i = n-2; i >= 0; i--) {
      suf[i] = Math.max(suf[i+1], a[i]);
    }
    long ans = 0;
    for (int i = 0; i < n; i++) {
      ans = Math.max(ans, pre[i] + suf[i] - a[i]);
    }
    System.out.println(ans);
  }
}
