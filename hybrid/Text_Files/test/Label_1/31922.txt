import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int n = sc.nextInt();
            int count = 0;
            for (int j = 1; j <= n; j++) {
                for (int k = 0; k <= n; k++) {
                    if (gcd(k, j) == 1) {
                        count++;
                    }
                }
            }
            System.out.println(count);
        }
    }

    public static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
