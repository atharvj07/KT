#include <bits/stdc++.h>
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define rrep(i, n) for (int i = (int)(n) - 1; i >= 0; i--)
using namespace std;
template <class T>
using V = vector<T>;
using vi = V<int>;
using vvi = V<vi>;
using vvvi = V<vvi>;
using VP = V<pair<int, int>>;
using VVP = V<VP>;
using ll = long long;
using P = pair<int, int>;
using S = string;
using vb = V<bool>;
using vvb = V<vb>;
using vvvb = V<vvb>;
using Vl = V<ll>;
using Vvl = V<Vl>;
using Vvvl = V<Vvl>;
const int INF = 1e9 + 7;
const int MOD = 1e9 + 7;
#define MAX_N (int)2021
int n, a[MAX_N];
bool dfs(int len, vi& r, int& t) {
  bool ok = 1;
  rep(i, len) ok &= a[i] == i + 1;
  rep(i, len / 2) ok &= a[i] >= a[len - 1 - i];
  if (len <= 3) return ok;
  rep(i, len - 3) {
    int j = len - 1 - i;
    if (a[i] < a[j] && a[i + 1] < a[j - 1]) {
      r.push_back(len);
      swap(a[i], a[j]);
      swap(a[i + 1], a[j - 1]);
      t++;
      if (dfs(len - 2, r, t)) return 1;
      t--;
      swap(a[i], a[j]);
      swap(a[i + 1], a[j - 1]);
      r.pop_back();
    }
  }
  if (ok) return 1;
  return dfs(len - 1, r, t);
}
int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int T;
  cin >> T;
  rep(t, T) {
    cin >> n;
    rep(i, n) cin >> a[i];
    vi r;
    int c = 0;
    if (!dfs(n, r, c)) c = -1;
    if (c != -1) c = min(c, (5 * n) / 2);
    cout << c;
    rep(i, r.size()) cout << ' ' << r[i];
    cout << '\n';
  }
  return 0;
}
