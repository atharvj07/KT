import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] x = new int[n + 1];
        int[] y = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            x[i] = sc.nextInt();
        }
        for (int i = 1; i <= n; i++) {
            y[i] = sc.nextInt();
        }
        Arrays.sort(x, 1, n + 1);
        Arrays.sort(y, 1, n + 1);
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            ans ^= x[i] ^ y[i];
        }
        if (ans == 0) {
            System.out.println("Koyomi");
        } else {
            System.out.println("Karen");
        }
    }
}
