import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] ropes = new int[m];
        TreeMap<Integer, Integer> map = new TreeMap<>();
        for (int i = 0; i < m; i++) {
            ropes[i] = in.nextInt();
        }
        Arrays.sort(ropes);
        for (int i = 0; i < m; i++) {
            int r = in.nextInt();
            map.put(ropes[i], r);
        }
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            int x = in.nextInt();
            int y = in.nextInt();
            int ans = -1;
            for (Map.Entry<Integer, Integer> e : map.entrySet()) {
                if (e.getKey() >= x && e.getValue() <= y) {
                    ans = e.getValue();
                }
            }
            System.out.println(ans);
        }
    }
}
