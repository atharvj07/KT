#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<long long int> p(n), c(n);
    for (int i = 0; i < n; i++) cin >> p[i];
    for (int i = 0; i < n; i++) cin >> c[i];
    vector<pair<long long int, long long int>> coins(n);
    for (int i = 0; i < n; i++) coins[i] = {p[i], c[i]};
    sort(coins.begin(), coins.end());
    vector<long long int> dp(n + 1, 0);
    for (int i = k; i >= 0; i--) {
        for (int j = 0; j < n; j++) {
            dp[i] = max(dp[i], dp[i + 1]);
            if (j + i >= n) break;
            dp[i] = max(dp[i], dp[i + 1] + coins[j + i].second);
        }
    }
    cout << dp[0] << endl;
}
