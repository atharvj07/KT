#include <iostream>
#include <vector>
using namespace std;

void chk_max(long long int &x) {
  if (x > 1e18) x = 1e18;
}

int main() {
  int n;
  cin >> n;
  vector<long long int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<pair<int, pair<int, int>>> ans;
  for (int i = 0; i < n - 1; i++) {
    if (a[i] > a[i + 1]) {
      ans.push_back({2, {i + 1, i + 2}});
      chk_max(a[i]);
      chk_max(a[i + 1]);
    } else if (a[i] < a[i + 1]) {
      ans.push_back({1, {i + 1, i + 2}});
      chk_max(a[i]);
      chk_max(a[i + 1]);
    }
  }
  cout << ans.size() << "\n";
  for (int i = 0; i < ans.size(); i++) {
    cout << ans[i].first << " " << ans[i].second.first << " "
         << ans[i].second.second << "\n";
  }
  return 0;
}
