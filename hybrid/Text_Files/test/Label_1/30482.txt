import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);
        int k = Integer.parseInt(line[2]);
        char[][] map = new char[n][m];
        for (int i = 0; i < n; i++) {
            String row = reader.readLine();
            map[i] = row.toCharArray();
        }

        List<Constellation> constellations = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == '*') {
                    Constellation constellation = new Constellation(i, j);
                    if (isValidConstellation(constellation, map)) {
                        constellations.add(constellation);
                    }
                }
            }
        }

        if (constellations.size() < k) {
            System.out.println(-1);
        } else {
            Constellation constellation = constellations.get(k - 1);
            System.out.println(constellation.center.i + " " + constellation.center.j);
            System.out.println(constellation.up.i + " " + constellation.up.j);
            System.out.println(constellation.down.i + " " + constellation.down.j);
            System.out.println(constellation.left.i + " " + constellation.left.j);
            System.out.println(constellation.right.i + " " + constellation.right.j);
        }
    }

    private static boolean isValidConstellation(Constellation constellation, char[][] map) {
        return isValidUp(constellation, map) && isValidDown(constellation, map) && isValidLeft(constellation, map) && isValidRight(constellation, map);
    }

    private static boolean isValidUp(Constellation constellation, char[][] map) {
        int i = constellation.center.i - 1;
        while (i >= 0) {
            if (map[i][constellation.center.j] == '*') {
                constellation.up = new Point(i, constellation.center.j);
                return true;
            }
            i--;
        }
        return false;
    }

    private static boolean isValidDown(Constellation constellation, char[][] map) {
        int i = constellation.center.i + 1;
        while (i < map.length) {
            if (map[i][constellation.center.j] == '*') {
                constellation.down = new Point(i, constellation.center.j);
                return true;
            }
            i++;
        }
        return false;
    }

    private static boolean isValidLeft(Constellation constellation, char[][] map) {
        int j = constellation.center.j - 1;
        while (j >= 0) {
            if (map[constellation.center.i][j] == '*') {
                constellation.left = new Point(constellation.center.i, j);
                return true;
            }
            j--;
        }
        return false;
    }

    private static boolean isValidRight(Constellation constellation, char[][] map) {
        int j = constellation.center.j + 1;
        while (j < map[0].length) {
            if (map[constellation.center.i][j] == '*') {
                constellation.right = new Point(constellation.center.i, j);
                return true;
            }
            j++;
        }
        return false;
    }

    private static class Point {
        int i;
        int j;

        public Point(int i, int j) {
            this.i = i;
            this.j = j;
        }
    }

    private static class Constellation {
        Point center;
        Point up;
        Point down;
        Point left;
        Point right;

        public Constellation(int i, int j) {
            this.center = new Point(i, j);
        }

        @Override
        public String toString() {
            return "Constellation{" +
                    "center=" + center +
                    ", up=" + up +
                    ", down=" + down +
                    ", left=" + left +
                    ", right=" + right +
                    '}';
        }
    }
}
