import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            long l = sc.nextLong();
            long t = sc.nextLong();
            long m = sc.nextLong();
            long r = (t * m + l - 1) / b;
            if (r * b + a - (l - 1) > t * m) {
                r--;
            }
            if (r < 0 || l + r * b - a > t * m) {
                System.out.println(-1);
            } else {
                System.out.println(r);
            }
        }
    }
}
