#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int r, g, b;
    cin >> r >> g >> b;
    if (r % 2 == 0 && g % 2 == 0 && b % 2 == 0)
      cout << "Yes" << endl;
    else if (r % 2 == 1 && g % 2 == 1 && b % 2 == 1)
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
  return 0;
}
