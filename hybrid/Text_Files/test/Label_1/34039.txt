import java.util.*;

class Main {

    private static final int MAX_N = 30000;
    private static final int MAX_K = 200;

    private static class Edge {
        public int a;
        public int b;

        public Edge(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }

    private static int n;
    private static int k;
    private static int[][] distances;
    private static int[] parent;
    private static List<Edge> edges = new ArrayList<>();

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        k = scanner.nextInt();
        distances = new int[k][n];
        parent = new int[n];
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                distances[i][j] = scanner.nextInt();
            }
        }

        if (!isTreePossible()) {
            System.out.println(-1);
            return;
        }

        for (Edge edge : edges) {
            System.out.println(edge.a + " " + edge.b);
        }
    }

    private static boolean isTreePossible() {
        for (int i = 0; i < k; i++) {
            int root = -1;
            for (int j = 0; j < n; j++) {
                if (distances[i][j] == 0) {
                    root = j;
                    break;
                }
            }

            if (root == -1) {
                return false;
            }

            for (int j = 0; j < n; j++) {
                if (distances[i][j] == -1 || i == j) {
                    continue;
                }

                if (parent[j] != 0) {
                    return false;
                }
                parent[j] = root;
                edges.add(new Edge(root + 1, j + 1));
            }
        }

        return true;
    }
}
