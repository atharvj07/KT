import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int speed = 0;
        int ignoreCnt = 0;
        int last = 0;
        for (int i = 0; i < n; i++) {
            int type = sc.nextInt();
            if (type == 3) {
                if (speed > sc.nextInt()) {
                    ignoreCnt++;
                }
                speed = sc.nextInt();
            } else if (type == 1) {
                speed = sc.nextInt();
            } else if (type == 6 && last != 6) {
                ignoreCnt++;
            }
            last = type;
        }
        System.out.println(ignoreCnt);
    }
}
