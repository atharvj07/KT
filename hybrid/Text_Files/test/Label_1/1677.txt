#include <iostream>
#include <vector>

int main() {
  long long int n;
  std::cin >> n;

  std::vector<long long int> arr(n);
  for (long long int i = 0; i < n; i++) std::cin >> arr[i];

  long long int ans = 0;
  for (long long int i = 0; i < n - 1; i++) {
    if (arr[i] >= arr[i + 1]) {
      ans += arr[i] - arr[i + 1] + 1;
      arr[i + 1] = arr[i] + 1;
    }
  }

  std::cout << ans << std::endl;
  return 0;
}
