from collections import Counter
n = int(input())
b = [int(_) for _ in input().split()]
c = Counter(b)
k = list(c.keys())
if len(k) == 1:
    print(0)
elif len(k) <= 2:
    print(1)
else:
    k.sort()
    x = k[0]
    y = k[-1]
    if y - x > 1:
        print(-1)
    else:
        if c[k[0]] > c[k[-1]]:
            print(c[k[-1]])
        else:
            print(c[k[0]])
