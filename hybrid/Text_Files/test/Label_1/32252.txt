import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n - 1; i++) {
            a[i] = sc.nextInt();
        }

        long ans = 0;
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < a[i]; j++) {
                ans += j - i;
            }
        }

        System.out.println(ans);
    }
}
