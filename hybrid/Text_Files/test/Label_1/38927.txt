def solve():
    t = int(input())
    for _ in range(t):
        n, l, r, s = map(int, input().split())
        if (r - l + 1) * (l + r) // 2 != s:
            print(-1)
            continue
        a = [i for i in range(l, r + 1)]
        sum1 = (a[0] + a[-1]) * (len(a) // 2)
        for i in range(1, len(a) // 2):
            sum1 -= a[i] * 2
        ans = [0] * n
        for i in range(l - 1):
            ans[i] = i + 1
        ans[l - 1] = sum1
        k = 0
        for i in range(r + 1, n):
            ans[i] = a[k]
            k += 1
        ans[r] = s - sum1
        print(*ans)

solve()
