import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] pos = sc.nextLine().split(" ");
        sc.close();

        boolean[][] board = new boolean[8][8];
        for (String p : pos) {
            board[p.charAt(0) - 'a'][p.charAt(1) - '1'] = true;
        }

        int[] dx = {-1, 1, 0, 0};
        int[] dy = {0, 0, -1, 1};
        boolean[][] check = new boolean[8][8];
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                if (board[i][j]) {
                    if (board[i][j] && (pos[2].equals(i + "" + j) || pos[3].equals(i + "" + j))) {
                        System.out.println("OTHER");
                        return;
                    }
                    for (int k = 0; k < 4; k++) {
                        int x = i, y = j;
                        while (x >= 0 && x < 8 && y >= 0 && y < 8) {
                            if (board[x][y] && (pos[2].equals(x + "" + y) || pos[3].equals(x + "" + y))) {
                                System.out.println("OTHER");
                                return;
                            }
                            x += dx[k];
                            y += dy[k];
                        }
                    }
                }
            }
        }

        System.out.println("CHECKMATE");
    }
}
