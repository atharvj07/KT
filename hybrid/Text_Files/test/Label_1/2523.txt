#include <bits/stdc++.h>
using namespace std;

const int maxn = 105;

int dp[maxn][maxn];
int col[maxn];
int val[maxn];
int rev[maxn];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; ++i) {
        cin >> col[i] >> val[i] >> rev[i];
        dp[i][i] = val[i];
    }
    for (int i = n; i >= 1; --i) {
        for (int j = i + 1; j <= n; ++j) {
            if (col[i] == rev[j]) {
                dp[i][j] = max(dp[i][j - 1], dp[i + 1][j]);
                dp[i][j] = max(dp[i][j], dp[i + 1][j - 1] + val[i] + val[j]);
            }
            dp[i][j] = max(dp[i][j], max(dp[i][j - 1], dp[i + 1][j]));
        }
    }
    cout << dp[1][n] << '\n';
    return 0;
}
