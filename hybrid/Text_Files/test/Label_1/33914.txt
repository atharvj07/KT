import java.util.*;

public class Main {

    private static final int MAX_GROUP_COUNT = 100;
    private static final int MAX_MEMBER_COUNT_PER_GROUP = 10;

    private static int n;
    private static String[] groupNames;
    private static Set<String>[] groupMemberships;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        while (true) {
            n = scanner.nextInt();
            if (n == 0) {
                break;
            }
            groupNames = new String[n];
            groupMemberships = new Set[n];
            for (int i = 0; i < n; i++) {
                groupMemberships[i] = new HashSet<>();
                String line = scanner.next();
                String[] tokens = line.split(":");
                groupNames[i] = tokens[0];
                for (String member : tokens[1].split(",")) {
                    groupMemberships[i].add(member);
                }
            }
            int numEmployees = countEmployees(0);
            System.out.println(numEmployees);
        }
        scanner.close();
    }

    private static int countEmployees(int groupIndex) {
        Set<String> visitedGroups = new HashSet<>();
        return countEmployees(groupIndex, visitedGroups);
    }

    private static int countEmployees(int groupIndex, Set<String> visitedGroups) {
        if (!visitedGroups.add(groupNames[groupIndex])) {
            return 0;
        }
        int numEmployees = 0;
        for (String member : groupMemberships[groupIndex]) {
            if (!isEmployee(member)) {
                numEmployees += countEmployees(getIndexOfGroup(member), visitedGroups);
            } else {
                numEmployees++;
            }
        }
        return numEmployees;
    }

    private static boolean isEmployee(String member) {
        for (int i = 0; i < n; i++) {
            if (groupNames[i].equals(member)) {
                return false;
            }
        }
        return true;
    }

    private static int getIndexOfGroup(String groupName) {
        for (int i = 0; i < n; i++) {
            if (groupNames[i].equals(groupName)) {
                return i;
            }
        }
        return -1;
    }
}
