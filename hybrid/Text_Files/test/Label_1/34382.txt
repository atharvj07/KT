import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        Arrays.sort(d);
        int max = 0;
        for (int i = n - 1; i >= 0; i--) {
            int cal = c + d[i];
            int cost = a + b;
            double calPerCost = (double) cal / cost;
            if (calPerCost > max) {
                max = (int) calPerCost;
            }
        }
        System.out.println(max);
    }
}
