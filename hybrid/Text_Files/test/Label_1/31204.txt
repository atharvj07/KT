import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int k = input.nextInt();
        String s = input.nextLine();
        char[] ch = s.toCharArray();
        int cnt = 0;
        for (char c : ch) {
            if (c == 'Y') {
                cnt = 0;
            } else if (c == 'N') {
                cnt++;
                if (cnt >= k) {
                    System.out.println("YES");
                    return;
                }
            }
        }
        System.out.println("NO");
    }
}
