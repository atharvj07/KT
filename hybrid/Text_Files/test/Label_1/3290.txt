#include <vector>
#include <iostream>
#include <algorithm>

int main() {
  int n; std::cin >> n;
  std::vector<long long> a(n);
  for (int i = 0; i < n; ++i) std::cin >> a[i];
  std::sort(a.begin(), a.end());
  long long ans = a[n - 1];
  long long cnt = 1;
  for (int i = n - 2; i >= 0; --i) {
    ans += std::min(a[i], cnt);
    ++cnt;
  }
  std::cout << ans << std::endl;
}
