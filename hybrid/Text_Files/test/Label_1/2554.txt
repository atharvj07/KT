#include <iostream>

int main() {
    long long x1, y1, x2, y2, x3, y3, x4, y4, x5, y5, x6, y6;
    std::cin >> x1 >> y1 >> x2 >> y2;
    std::cin >> x3 >> y3 >> x4 >> y4;
    std::cin >> x5 >> y5 >> x6 >> y6;

    if ((x5 <= x1 && x6 >= x2) || (y5 <= y1 && y6 >= y2)) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    if ((x3 <= x1 && x4 >= x2) || (y3 <= y1 && y4 >= y2)) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    if ((x3 <= x1 && x4 >= x2) && (x5 <= x1 && x6 >= x2)) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    if ((x3 <= x1 && x4 >= x2) && (y5 <= y1 && y6 >= y2)) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    if ((x5 <= x1 && x6 >= x2) && (y3 <= y1 && y4 >= y2)) {
        std::cout << "NO" << std::endl;
        return 0;
    }
    std::cout << "YES" << std::endl;
    return 0;
}
