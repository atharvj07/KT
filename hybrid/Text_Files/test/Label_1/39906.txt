n = int(input())
a = list(input())
a.sort()
t = n//3
x=0
y=0
for i in range(n):
    if a[i]=='0': x+=1
    elif a[i]=='1': y+=1
if x>t:
    for i in range(n):
        if a[i]=='0' and y<t:
            a[i]='1'
            y+=1
elif y>t:
    for i in range(n):
        if a[i]=='1' and x<t:
            a[i]='0'
            x+=1
print(''.join(a))
