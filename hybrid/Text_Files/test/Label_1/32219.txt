import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), g = 0;
        for (int a[], i = 0; i < n; i++) {
            a = new int[n];
            for (int j = 0; j < n; j++) {
                if (i != j) a[j] = GCD(sc.nextInt(), a[j]);
            }
            g = GCD(g, a[0]);
        }
        System.out.println(g);
        sc.close();
    }
    public static int GCD(int p, int q) {
        return p == 0 ? q : GCD(q % p, p);
    }
}
