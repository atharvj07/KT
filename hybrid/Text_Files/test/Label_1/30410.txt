import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int m = sc.nextInt();
        int a = sc.nextInt();
        int[] g = new int[a];
        for (int i = 0; i < a; i++) {
            g[i] = sc.nextInt();
        }
        int[] b = new int[n + 1];
        int[] c = new int[n + 1];
        for (int i = 0; i < a; i++) {
            b[g[i]]++;
            c[g[i]] = i;
        }
        Arrays.sort(b);
        int[] r = new int[n + 1];
        int j = 0;
        for (int i = n; i >= 1; i--) {
            if (j + b[i] >= k) {
                for (int i1 = 1; i1 <= n; i1++) {
                    if (b[i1] == b[i]) {
                        if (b[i] > 0) {
                            r[i1] = 1;
                        }
                    }
                }
            } else {
                for (int i1 = 1; i1 <= n; i1++) {
                    if (b[i1] == b[i]) {
                        if (c[i1] < a - k) {
                            r[i1] = 3;
                        } else {
                            r[i1] = 2;
                        }
                    }
                }
            }
            j += b[i];
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(r[i] + " ");
        }
        sc.close();
    }
}
