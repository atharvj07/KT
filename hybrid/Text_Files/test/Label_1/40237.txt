n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

pref_a = [0]*n
pref_b = [0]*m
for i in range(n): pref_a[i] = pref_a[i-1] + a[i]
for i in range(m): pref_b[i] = pref_b[i-1] + b[i]

ans = 0
for i in range(n):
    for j in range(m):
        if i!=0: s1 = pref_a[i-1]
        else: s1 = 0
        if j!=0: s2 = pref_b[j-1]
        else: s2 = 0
        s = pref_a[i] - s1
        t = pref_b[j] - s2
        if s*t==k: ans += 1
        
print(ans)
