from collections import Counter
n=int(input())
a=list(map(int,input().split()))
f=[Counter() for i in range(n+1)]
ans=0
for x in a:
    f[n]-=Counter({x:-1})
    f[n-1][x]-=1
    for i in range(n-1,0,-1):
        f[i]=f[i]-f[i+1]
        if f[i][a[i-1]]>f[i+1][a[i-1]]:
            ans+=f[i][a[i-1]]
print(ans)
