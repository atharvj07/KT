#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 2e5;

int n, q;
vector<int> a;

int main() {
  cin >> n >> q;
  a.resize(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  vector<vector<int>> ans(q);
  vector<int> b(n + 1);
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) {
      continue;
    }
    if (b[a[i]] == 0) {
      b[a[i]] = i + 1;
    } else {
      ans[a[i] - 1].push_back(b[a[i]]);
      ans[a[i] - 1].push_back(i + 1);
      b[a[i]] = i + 1;
    }
  }
  for (int i = 0; i < n; i++) {
    if (a[i] != 0) {
      continue;
    }
    for (int j = 1; j <= q; j++) {
      if (b[j] != 0 && j != a[i - 1] && j != a[i + 1]) {
        ans[j - 1].push_back(b[j]);
        ans[j - 1].push_back(i + 1);
        a[i] = j;
        b[j] = i + 1;
        break;
      }
    }
  }
  cout << "YES" << endl;
  for (int i = 0; i < n; i++) {
    cout << a[i] << " ";
  }
  cout << endl;
  return 0;
}
