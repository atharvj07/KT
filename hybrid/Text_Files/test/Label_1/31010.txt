import java.util.*;
import java.io.*;

class Main {
    static int[] xs, ys, zs;
    static int[] px, py, pz;

    static boolean solve() {
        for (int i = 0; i < 8; i++) {
            for (int j = i + 1; j < 8; j++) {
                for (int k = j + 1; k < 8; k++) {
                    if (xs[i] == xs[j] && xs[j] == xs[k] &&
                        ys[i] == ys[j] && ys[j] == ys[k] &&
                        zs[i] == zs[j] && zs[j] == zs[k])
                        return false;
                }
            }
        }

        for (int i = 0; i < 8; i++) {
            Map<Integer, Integer> mpX = new HashMap<>();
            mpX.put(xs[i], mpX.getOrDefault(xs[i], 0) + 1);
            Map<Integer, Integer> mpY = new HashMap<>();
            mpY.put(ys[i], mpY.getOrDefault(ys[i], 0) + 1);
            Map<Integer, Integer> mpZ = new HashMap<>();
            mpZ.put(zs[i], mpZ.getOrDefault(zs[i], 0) + 1);

            if (mpX.size() != 2 || mpY.size() != 2 || mpZ.size() != 2)
                return false;
        }

        return true;
    }

    static void restore() {
        List<int[]> perm = new ArrayList<>();
        perm.add(new int[]{px[0], py[0], pz[0]});
        for (int i = 1; i < 8; i++) {
            int[] p = new int[]{px[i], py[i], pz[i]};
            int[] ref = perm.get(0);
            for (int j = 0; j < 3; j++) {
                for (int k = 0; k < 3; k++) {
                    for (int l = 0; l < 3; l++) {
                        int[] q = new int[]{
                            ref[j],
                            ref[(k + 1) % 3],
                            ref[(l + 1) % 3]
                        };

                        if (p[0] == q[0] && p[1] == q[1] && p[2] == q[2])
                            perm.add(p);
                    }
                }
            }
        }
        Collections.sort(perm, (x, y) -> {
            int[] a = x, b = y;
            if (a[0] != b[0]) return a[0] - b[0];
            if (a[1] != b[1]) return a[1] - b[1];
            return a[2] - b[2];
        });
        for (int[] xyzs : perm)
            System.out.println(Arrays.toString(xyzs).replace("[", "").replace("]", ""));
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;
        xs = new int[8];
        ys = new int[8];
        zs = new int[8];
        for (int i = 0; i < 8; i++) {
            st = new StringTokenizer(br.readLine());
            xs[i] = Integer.parseInt(st.nextToken());
            ys[i] = Integer.parseInt(st.nextToken());
            zs[i] = Integer.parseInt(st.nextToken());
        }

        px = xs.clone();
        py = ys.clone();
        pz = zs.clone();

        Arrays.sort(xs);
        Arrays.sort(ys);
        Arrays.sort(zs);

        if (solve())
            restore();
        else
            System.out.println("NO");
    }
}
