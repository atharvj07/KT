import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextLong();
        }

        long gcd = a[0];
        for (int i = 1; i < n; i++) {
            gcd = gcd(gcd, a[i]);
        }

        long f = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] % gcd == 0) {
                f++;
            }
        }

        if (f >= n / 2) {
            System.out.println(gcd);
        } else {
            System.out.println(0);
        }
    }

    public static long gcd(long a, long b) {
        if (b == 0) {
            return a;
        } else {
            return gcd(b, a % b);
        }
    }
}
