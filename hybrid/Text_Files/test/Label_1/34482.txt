import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = sc.nextInt();
        int[][] g = new int[n][n];
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1;
            int v = sc.nextInt() - 1;
            g[u][v] = g[v][u] = 1;
        }
        int minw = Integer.MAX_VALUE;
        int minidx = -1;
        for (int i = 0; i < n; i++) {
            if (a[i] < minw) {
                minw = a[i];
                minidx = i;
            }
        }
        long ans = 0;
        int[] deg = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == minidx)
                continue;
            int d = 0;
            for (int j = 0; j < n; j++)
                d += g[i][j];
            ans += (long) (d * a[i]);
            deg[i] = d;
        }
        int[] dist = new int[n];
        Arrays.fill(dist, 1000000000);
        dist[minidx] = 0;
        dfs(minidx, -1, g, dist);
        for (int i = 0; i < n; i++) {
            if (i == minidx)
                continue;
            int d = dist[i];
            if (d == 1000000000)
                d = 0;
            ans += (long) ((long) Math.ceil(Math.log(d) / Math.log(2)) * Math.min(a[i], minw));
        }
        System.out.println(ans);
    }

    static void dfs(int node, int p, int[][] g, int[] dist) {
        for (int i = 0; i < g.length; i++) {
            if (i == node || i == p || g[node][i] == 0)
                continue;
            dist[i] = dist[node] + 1;
            dfs(i, node, g, dist);
        }
    }
}
