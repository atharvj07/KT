import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] d = new int[n][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < 2; j++) {
                d[i][j] = sc.nextInt();
            }
        }
        sc.close();
        boolean ans = false;
        for (int i = 0; i < n - 2; i++) {
            ans |= d[i][0] == d[i][1] && d[i + 1][0] == d[i + 1][1] && d[i + 2][0] == d[i + 2][1];
        }
        System.out.println(ans ? "Yes" : "No");
    }
}
