import math

def find_ab(d):
    disc = d*d - 4*d
    if disc < 0:
        return ("N",)
    a = (d + math.sqrt(disc)) / 2
    b = (d - math.sqrt(disc)) / 2
    return ("Y", a, b)

t = int(input().strip())
for _ in range(t):
    d = int(input().strip())
    result = find_ab(d)
    if result[0] == "N":
        print("N")
    else:
        print(f"Y {result[1]:.9f} {result[2]:.9f}")
