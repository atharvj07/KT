import java.util.*;

public class Main {
    static int[] parent, marked;
    static int sum = 0;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int q = sc.nextInt();
            if (n == 0 && q == 0) break;
            parent = new int[n + 1];
            marked = new int[n + 1];
            for (int i = 2; i <= n; i++) {
                parent[i] = sc.nextInt();
            }
            marked[1] = 1;
            for (int i = 0; i < q; i++) {
                String op = sc.next();
                int v = sc.nextInt();
                if (op.equals("M")) {
                    marked[v] = 1;
                } else if (op.equals("Q")) {
                    int j = v;
                    while (j != 0 && marked[j] == 0) {
                        j = parent[j];
                    }
                    sum += j;
                }
            }
            System.out.println(sum);
            sum = 0;
        }
        sc.close();
    }
}
