#include<bits/stdc++.h>
using namespace std;
#define f(i,a,b,c) for(int i=a;i<b;i+=c)

int main(){
  int n,b,a; cin>>n>>b>>a;
  int s[n+1];
  f(i,0,n,1)cin>>s[i];
  int ans=0,x=0;
  while(true){
    if(x==n)break;
    if(s[x+1]){
      if(b){x++;b--;a++;ans++;}
      else if(a){x++;a--;ans++;}
      else break;
    }else{
      if(a){x++;a--;ans++;}
      else break;
    }
  }
  cout<<ans;
  return 0;
}
