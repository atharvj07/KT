#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    int q;
    std::cin >> q;
    while (q--) {
        int n, r;
        std::cin >> n >> r;
        std::vector<int> x(n);
        for (int &xi : x) { std::cin >> xi; }
        std::sort(x.begin(), x.end());
        int missiles = 0;
        int right = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (x[i] - right > r) {
                missiles++;
                right -= r;
            }
            right += r;
        }
        std::cout << missiles << "\n";
    }
    return 0;
}
