#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, t;
    cin >> n >> t;
    vector<pair<int, int>> tasks;
    for (int i = 0; i < n; i++) {
        int ti, qi;
        cin >> ti >> qi;
        tasks.push_back(make_pair(ti, qi));
    }
    sort(tasks.begin(), tasks.end());
    int ans = 0;
    int time = 0;
    for (int i = 0; i < n; i++) {
        if (time + tasks[i].first <= t) {
            ans += tasks[i].second;
            time += tasks[i].first;
        }
    }
    cout << ans << endl;
    return 0;
}
