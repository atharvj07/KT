#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while(t--)
    {
        string s, key;
        cin >> key >> s;
        int ans = 0;
        for(int i = 1; i < s.size(); i++)
        {
            ans += abs(find(key.begin(), key.end(), s[i - 1]) - find(key.begin(), key.end(), s[i]));
        }
        cout << ans << endl;
    }
}
