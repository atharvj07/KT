#include <bits/stdc++.h>
using namespace std;

int main() {
  long long n, m, k;
  cin >> n >> m >> k;
  long long l = 0, r = (n + 1LL) * (m + 1LL), mid, ans;
  while (l <= r) {
    mid = (l + r) / 2;
    long long cnt = 0;
    for (long long j = 1; j <= m; j++) {
      cnt += min(mid / j, n);
    }
    if (cnt >= k) {
      r = mid - 1;
      ans = mid;
    } else {
      l = mid + 1;
    }
  }
  cout << ans << "\n";
  return 0;
}
