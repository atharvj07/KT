import java.util.*;

class Main {
  final static int INF = 1000000000;
  static int[] minDist;
  static boolean[] visited;
  static int totalCost = 0;

  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int n = s.nextInt();
    int k = s.nextInt();
    minDist = new int[n + 1];
    Arrays.fill(minDist, INF);
    visited = new boolean[n + 1];
    visited[1] = true;

    for (int i = 1; i <= n; i++) {
      minDist[i] = s.nextInt();
    }

    for (int i = 2; i <= n; i++) {
      if (i != 1 && !visited[i] && minDist[i] != 0) {
        totalCost += minDist[i];
        visited[i] = true;
        for (int j = 1; j <= n; j++) {
          if (j != 1 && !visited[j] && minDist[j] != 0) {
            minDist[j] = Math.min(minDist[j], minDist[i] + s.nextInt());
          } else {
            s.nextInt();
          }
        }
      } else {
        for (int j = 1; j <= n; j++) {
          s.nextInt();
        }
      }
    }

    System.out.println(totalCost);
  }
}
