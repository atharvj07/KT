#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string s[n];
    for (int i = 0; i < n; i++) cin >> s[i];
    sort(s, s + n);
    string ans;
    int l = 0, r = n - 1;
    while (l <= r) {
        if (s[l] == s[r]) {
            ans += s[l];
            l++, r--;
        } else if (s[l].back() == s[r].front()) {
            ans += s[l++].back();
        } else if (s[l].front() == s[r].back()) {
            ans = s[r--].back() + ans;
        } else {
            break;
        }
    }
    cout << ans.size() << endl;
    if(ans.size()) cout << ans << endl;
    return 0;
}
