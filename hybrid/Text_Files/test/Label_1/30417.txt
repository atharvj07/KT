import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                vis[i] = true;
                if (a[i] == 0) {
                    for (int j = i - 1; j >= 0; j--) {
                        if (!vis[j]) {
                            ans++;
                            vis[j] = true;
                        }
                    }
                } else {
                    for (int j = i + 1; j < n; j++) {
                        if (!vis[j]) {
                            ans++;
                            vis[j] = true;
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
