import sys

n = int(sys.stdin.readline())
arr = list(map(int, sys.stdin.readline().split()))
mod = 1000000007

# Initialize dp array to store number of ways to recover permutation
dp = [[0 for _ in range(n + 1)] for _ in range(n + 1)]

# Base case: For n = 1, there is only one way to recover the permutation
dp[1][1] = 1

# Iterate over the values of n and i
for i in range(2, n + 1):
    for j in range(1, n + 1):
        # Check if the value at index j is -1
        if arr[j - 1] == -1:
            # If it is, then the value at index j can be placed in any valid position
            dp[i][j] = (dp[i - 1][j - 1] + dp[i - 1][j + 1]) % mod
        # If the value at index j is not -1, then it must be placed in the correct position
        else:
            dp[i][j] = dp[i - 1][arr[j - 1]] % mod

# Output the number of ways to recover the permutation
print(dp[n][1])
