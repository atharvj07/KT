import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[n], b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] dp = new int[m + 1];
        Arrays.fill(dp, -1);
        dp[0] = 0;
        for (int i = 0; i < n; i++) {
            for (int j = m; j >= 0; j--) {
                if (dp[j] != -1 && a[i] <= j && j <= b[i]) dp[b[i]] = Math.max(dp[b[i]], dp[j] + 1);
            }
        }
        System.out.println(dp[m] == -1 ? "NO" : "YES");
    }
}
