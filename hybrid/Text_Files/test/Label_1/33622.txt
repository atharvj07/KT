import java.util.ArrayList;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        ArrayList<Integer> ans = new ArrayList<>();
        for (int i = 30; i >= 0; i--) {
            if ((1 << i) <= n) {
                ans.add(1 << i);
                n -= 1 << i;
            }
        }
        System.out.println(ans.size());
        for (int x : ans) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
