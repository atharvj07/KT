import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        while (true) {
            long n = in.nextInt();
            long k = in.nextInt();
            if (n == 0 && k == 0) {
                break;
            }
            long m = n + k - 1;
            long r = 1;
            while (m > 9 * r) {
                m -= 9 * r;
                r *= 10;
            }
            long num = (n - 1) / 9;
            long dig = (n - 1) % 9;
            num += (m + 1) / r;
            if (dig != 0) {
                num--;
            }
            while (r > 1) {
                r /= 10;
                if (num / r % 10 != 0) {
                    num++;
                }
            }
            long ans = num % 10;
            num /= 10;
            for (long i = 0; i < k - 1; i++) {
                ans = ans * 10 + num % 10;
                num /= 10;
            }
            System.out.println(ans);
        }
    }
}
