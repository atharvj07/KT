import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();

        List<int[]> edges = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            edges.add(new int[] {sc.nextInt() - 1, sc.nextInt() - 1, sc.nextInt()});
        }

        boolean isConsistent = checkConsistency(n, edges);
        System.out.println(isConsistent ? "Yes" : "No");
    }

    public static boolean checkConsistency(int n, List<int[]> edges) {
        // Create a graph representation.
        Map<Integer, List<int[]>> graph = new HashMap<>();
        for (int[] edge : edges) {
            int u = edge[0], v = edge[1], w = edge[2];
            graph.computeIfAbsent(u, k -> new ArrayList<>()).add(new int[]{v, w});
            graph.computeIfAbsent(v, k -> new ArrayList<>()).add(new int[]{u, -w});
        }

        // Perform a topological sort.
        int[] visited = new int[n];
        Stack<Integer> stack = new Stack<>();
        for (int i = 0; i < n; i++) {
            if (visited[i] == 0) {
                if (!dfs(i, graph, visited, stack)) {
                    return false;
                }
            }
        }

        // Check the topological order.
        int[] order = new int[n];
        int idx = 0;
        while (!stack.isEmpty()) {
            order[idx++] = stack.pop();
        }

        for (int[] edge : edges) {
            int u = edge[0], v = edge[1], w = edge[2];
            if (order[u] > order[v]) {
                return false;
            }
        }

        return true;
    }

    public static boolean dfs(int u, Map<Integer, List<int[]>> graph, int[] visited, Stack<Integer> stack) {
        visited[u] = 1;
        for (int[] edge : graph.getOrDefault(u, new ArrayList<>())) {
            int v = edge[0], w = edge[1];
            if (visited[v] == 0) {
                if (!dfs(v, graph, visited, stack)) {
                    return false;
                }
            } else if (visited[v] == 1) {
                return false;
            }
        }

        visited[u] = 2;
        stack.push(u);
        return true;
    }
}
