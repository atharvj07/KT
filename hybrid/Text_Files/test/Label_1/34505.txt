import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        if ((n & 1) == 1) {
            System.out.println(-1);
        } else {
            n = n / 2;
            for (int i = 0; i < n / 4; i++) {
                System.out.print(4);
            }
            for (int i = n / 4; i < n; i++) {
                System.out.print(7);
            }
            System.out.println();
        }
    }
}
