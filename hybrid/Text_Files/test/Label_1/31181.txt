import java.io.*;
import java.util.*;
import java.math.*;

public class Main {
    static final long mod = 1000000007;
    static ArrayList<Integer>[] adj;
    static long modPow(long a, long b) {
        if (b == 0) return 1;
        long tmp = modPow(a, b / 2) % mod;
        tmp = tmp * tmp % mod;
        if (b % 2 == 1) tmp = tmp * a % mod;
        return tmp;
    }
    static long modInv(long a) {
        return modPow(a, mod - 2) % mod;
    }
    static long[][] matPow(long[][] a, long b) {
        long[][] c = new long[a.length][a[0].length];
        while (b > 0) {
            if (b % 2 == 1) {
                c = matMul(c, a);
            }
            a = matMul(a, a);
            b /= 2;
        }
        return c;
    }
    static long[][] matMul(long[][] a, long[][] b) {
        if (a[0].length != b.length) throw new IllegalArgumentException("The number of columns of a must be equal to the number of rows of b");
        long[][] c = new long[a.length][b[0].length];
        for (int i = 0; i < a.length; ++i)
            for (int j = 0; j < b[0].length; ++j)
                for (int k = 0; k < b.length; ++k)
                    c[i][j] = (c[i][j] + a[i][k] * b[k][j]) % mod;
        return c;
    }
    static void dfs(int u, int p, int[][] a, long[] f, long[] g) {
        f[u] = 1;
        g[u] = 1;
        for (int v : adj[u]) {
            if (v == p) continue;
            dfs(v, u, a, f, g);
            f[u] = f[u] * f[v] % mod;
            long[][] c = {{f[u], a[u][v]}, {f[u], 0}};
            long[][] d = matPow(c, f[v]);
            long tmp = (d[0][1] + g[v]) % mod;
            tmp = tmp * modInv(d[0][0]) % mod;
            g[u] = g[u] * tmp % mod;
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(in.readLine());
        adj = new ArrayList[n];
        for (int i = 0; i < n; ++i) adj[i] = new ArrayList<>();
        int[][] a = new int[n][n];
        for (int i = 0; i < n; ++i) {
            StringTokenizer st = new StringTokenizer(in.readLine());
            for (int j = 0; j < n; ++j) {
                a[i][j] = Integer.parseInt(st.nextToken());
                if (a[i][j] == 1) adj[i].add(j);
            }
        }
        long[] f = new long[n];
        long[] g = new long[n];
        dfs(0, -1, a, f, g);
        System.out.println(g[0]);
    }
}
