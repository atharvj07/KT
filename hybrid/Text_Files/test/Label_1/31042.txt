import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int[] y = new int[m];
        for (int i = 0; i < m; i++) {
            y[i] = sc.nextInt();
        }
        long ans = 1;
        long mod = 1000000007;
        for (int i = 0; i < n; i++) {
            int left = 0;
            int right = m - 1;
            while (left <= right) {
                int mid = (left + right) / 2;
                if (y[mid] < x[i]) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
            int index = left;
            if (index == m || y[index] != x[i]) {
                index--;
            }
            ans = ans * (index + 1) % mod;
        }
        System.out.println(ans);
    }
}
