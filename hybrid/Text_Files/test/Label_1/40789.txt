n, t = map(int, input().split())
ans = 0
mn = 1e9
for i in range(n):
    s, d = map(int, input().split())
    if t % d == 0:
        if s <= t:
            if s < mn:
                ans = i + 1
                mn = s
    else:
        nxt = (t // d + 1) * d
        if nxt - s < mn:
            ans = i + 1
            mn = nxt - s
print(ans)
