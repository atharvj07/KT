import math

n, m, k, s = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
gadgets = [tuple(map(int, input().split())) for _ in range(m)]
costs = [math.inf] * m
for i in range(n):
    for j in range(m):
        if gadgets[j][0] == 1:
            costs[j] = min(costs[j], gadgets[j][1] * a[i])
        else:
            costs[j] = min(costs[j], gadgets[j][1] * b[i])
costs.sort()
ans = []
for i in range(min(k, m)):
    if s < costs[i]:
        ans = []
        break
    s -= costs[i]
    ans.append(i + 1)
if not ans:
    print(-1)
else:
    print(len(ans))
    for i in ans:
        print(i, costs[i - 1])
