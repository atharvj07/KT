import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

class Main {
    private static final int ILLEGAL = -1;

    public static void main(String[] args) throws IOException {
        BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(input.readLine());
        int operations = Integer.parseInt(st.nextToken());
        int memorySize = Integer.parseInt(st.nextToken());
        List<Integer> blocks = new ArrayList<>();
        for (int i = 0; i < operations; i++) {
            st = new StringTokenizer(input.readLine());
            String command = st.nextToken();
            int number = Integer.parseInt(st.nextToken());
            if ("alloc".equals(command)) {
                int id = allocate(memorySize, blocks, number);
                if (id == ILLEGAL) {
                    System.out.println("NULL");
                } else {
                    System.out.println(id);
                }
            } else if ("erase".equals(command)) {
                boolean success = erase(blocks, number);
                if (!success) {
                    System.out.println("ILLEGAL_ERASE_ARGUMENT");
                }
            } else {
                defragment(blocks);
            }
        }
    }

    private static int allocate(int memorySize, List<Integer> blocks, int size) {
        int start = 0;
        int end = blocks.size();
        while (start < end) {
            int mid = (start + end) / 2;
            if (blocks.get(mid) == -1) {
                int next = mid + 1;
                while (next < blocks.size() && blocks.get(next) == -1) {
                    next++;
                }
                if (next - mid >= size) {
                    for (int i = mid; i < mid + size; i++) {
                        blocks.set(i, 1);
                    }
                    return mid + 1;
                } else {
                    start = mid + 1;
                }
            } else {
                end = mid;
            }
        }
        if (memorySize >= blocks.size() + size) {
            for (int i = blocks.size(); i < blocks.size() + size; i++) {
                blocks.add(1);
            }
            return blocks.size() - size + 1;
        }
        return ILLEGAL;
    }

    private static boolean erase(List<Integer> blocks, int id) {
        if (id <= 0 || id > blocks.size()) {
            return false;
        }
        int end = id - 1;
        while (end < blocks.size() && blocks.get(end) == -1) {
            end++;
        }
        int start = id - 1;
        while (start >= 0 && blocks.get(start) == -1) {
            start--;
        }
        if (start != id - 1 || end != id - 1) {
            return false;
        }
        for (int i = start; i <= end; i++) {
            blocks.set(i, -1);
        }
        return true;
    }

    private static void defragment(List<Integer> blocks) {
        List<Integer> newBlocks = new ArrayList<>();
        for (int block : blocks) {
            if (block != -1) {
                newBlocks.add(block);
            }
        }
        blocks.clear();
        blocks.addAll(newBlocks);
    }
}
