#include <vector>
#include <iostream>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> p(n);
  for (int i = 0; i < n; i++) cin >> p[i];
  int ans = 0;
  for (int l = 0; l < n; l++) {
    int sum = p[l];
    int cnt = 1;
    for (int r = l + 1; r < n; r++) {
      sum += p[r];
      cnt++;
      int med = (cnt % 2 ? p[(l + r) / 2] : (p[(l + r) / 2] + p[(l + r) / 2 - 1]) / 2);
      if (med == m) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
