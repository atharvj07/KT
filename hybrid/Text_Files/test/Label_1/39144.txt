import sys
mod = 998244353
n = int(input())
a = list(map(int, input().split()))
f = [0] * 201
for i in a:
    if i != -1:
        f[i] = 1
for i in range(1, 201):
    f[i] = (f[i] + f[i - 1]) % mod
n1 = 0
ans = 1
for i in range(n):
    if a[i] == -1:
        if n1 != 0:
            ans = (ans * f[a[i - 1] - n1]) % mod
        n1 = 0
    else:
        n1 += 1
print((ans * f[n1]) % mod)
