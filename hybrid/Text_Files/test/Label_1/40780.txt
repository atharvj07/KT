import math

def max_cake_volume(n, cakes):
    for i in range(n):
        cakes[i].append(math.pi * cakes[i][0]**2 * cakes[i][1])
    cakes.sort(key=lambda x: x[2])
    
    dp = [0] * n
    dp[0] = cakes[0][2]
    
    for i in range(1, n):
        dp[i] = cakes[i][2]
        for j in range(i):
            if cakes[i][2] > cakes[j][2]:
                dp[i] = max(dp[i], dp[j] + cakes[i][2])
                
    return max(dp)

n = int(input())
cakes = [list(map(int, input().split())) for _ in range(n)]
print('{:.9f}'.format(max_cake_volume(n, cakes)))
