MOD = 998244353

def ways_to_restore(n, a):
    dp = [[[0 for _ in range(201)] for _ in range(2)] for _ in range(n+1)]
    dp[0][0][0] = 1

    for i in range(1, n+1):
        for j in range(201):
            for k in range(2):
                if a[i-1] != -1 and a[i-1] != j:
                    continue
                dp[i][0][j] += dp[i-1][k][j]
                dp[i][0][j] %= MOD
                if j > 0:
                    dp[i][1][j] += dp[i-1][k][j-1]
                    dp[i][1][j] %= MOD

    return sum(dp[n][k][j] for j in range(201) for k in range(2)) % MOD

n = int(input())
a = list(map(int, input().split()))

print(ways_to_restore(n, a))
