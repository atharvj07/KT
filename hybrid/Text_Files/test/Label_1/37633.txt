n=int(input())
a=list(map(int,input().split()))
ans=[[0,0]*4 for i in range(n)]
for i in range(n):
	ans[i][0]=a[i]
	ans[i][1]=ans[i][0]-(i+1)
	ans[i][2]=ans[i][0]//ans[i][1]
	ans[i][3]=ans[i][0]-ans[i][1]*ans[i][2]
last=1000000000
for i in range(n-1,-1,-1):
	if ans[i][1]<1 or ans[i][2]<1 or ans[i][3]<1 or last<ans[i][1]: last=1000000000; break
	last=ans[i][1]
if last==1000000000: print('NO')
else: print('YES\n'+' '.join(map(str,ans[0][1:3])))
