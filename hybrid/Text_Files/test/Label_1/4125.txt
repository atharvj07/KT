#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,ans=0;
    int pre[100005],suf[100005];
    string s;
    cin>>n>>s;
    memset(pre,0,sizeof(pre));
    memset(suf,0,sizeof(suf));
    if(s[0]=='0')pre[0]=1,ans=1;
    for(int i=1;i<n;i++){
        pre[i]=pre[i-1];
        if(s[i]=='0'){
            pre[i]++;
            ans=max(ans,pre[i]);
        }
    }
    if(s[n-1]=='1')suf[n-1]=1,ans=max(ans,1);
    for(int i=n-2;i>=0;i--){
        suf[i]=suf[i+1];
        if(s[i]=='1'){
            suf[i]++;
            ans=max(ans,suf[i]);
        }
    }
    for(int i=0;i<n-1;i++){
        if(pre[i]==suf[i+1])ans=max(ans,pre[i]+suf[i+1]);
    }
    cout<<ans<<endl;
}
