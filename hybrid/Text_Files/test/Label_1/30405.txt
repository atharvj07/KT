import java.util.LinkedList;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        LinkedList<Integer> l = new LinkedList<>();
        for (int i = 0; i < k - 1; i++) {
            while (!l.isEmpty() && a[l.peekLast()] > a[i]) {
                l.pollLast();
            }
            l.offer(i);
        }

        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        for (int i = k - 1; i < n; i++) {
            while (!l.isEmpty() && a[l.peekLast()] > a[i]) {
                l.pollLast();
            }
            l.offer(i);
            int x = a[l.peekFirst()];
            if (i >= q * k) {
                x = a[l.pollFirst()];
            }
            min = Math.min(x, min);
            max = Math.max(a[i], max);
        }

        System.out.println(max - min);
    }
}
