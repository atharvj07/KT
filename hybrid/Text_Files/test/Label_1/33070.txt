import java.util.*;

class Main {

    static int n, m;
    static int[] a;
    static int[][] g;
    static boolean[] visited;

    static int dfs(int v) {
        visited[v] = true;
        int res = a[v];
        for (int to : g[v]) {
            if (!visited[to]) {
                res += dfs(to);
            }
        }
        return res;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        a = new int[n];
        g = new int[n][];
        visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            g[i] = new int[0];
        }
        for (int i = 0; i < m; i++) {
            int from = sc.nextInt();
            int to = sc.nextInt();
            g[from] = Arrays.copyOf(g[from], g[from].length + 1);
            g[from][g[from].length - 1] = to;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                ans += dfs(i) > 0 ? 1 : 0;
            }
        }
        System.out.println(ans);
    }
}
