def solve_sequence(n, seq):
    for i in range(n):
        seq[i] = list(seq[i])

    def fill_increasing(prev, curr):
        for i in range(len(curr)):
            if curr[i] == '?':
                if i == 0:
                    if len(curr) > len(prev) or prev == []:
                        curr[i] = '1'
                    else:
                        curr[i] = str(max(int(prev[i]) + 1, 1)) if len(prev) == len(curr) and i == len(curr) - 1 else '0'
                else:
                    curr[i] = '0' if len(prev) == len(curr) and int("".join(curr[:i])) == int("".join(prev[:i])) else '0'
            if len(prev) == len(curr) and "".join(curr) <= "".join(prev):
                if i < len(curr) - 1 and curr[i + 1] == '?':
                    curr[i + 1] = '0'
                elif i == len(curr) - 1:
                    return False
        return True

    prev = []
    for i in range(n):
        if not fill_increasing(prev, seq[i]):
            return "NO"
        prev = seq[i]

    return ["YES"] + ["".join(s) for s in seq]

n = int(input().strip())
seq = [input().strip() for _ in range(n)]
result = solve_sequence(n, seq)
for line in result:
    print(line)
