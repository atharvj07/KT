#include <iostream>
#include <vector>
using namespace std;
const int N = 2e5 + 5;
int n, a[N], sub[N], res[N], mx[N], sz[N];
vector<int> adj[N];
void dfs(int u, int p) {
    sub[u] = a[u];
    sz[u] = 1;
    for (int v : adj[u]) {
        if (v == p) continue;
        dfs(v, u);
        sub[u] += sub[v];
        sz[u] += sz[v];
        mx[u] = max(mx[u], sub[v]);
    }
    res[u] = sub[u] - max(mx[u], sz[u] - sub[u]);
}
int main() {
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1);
    for (int i = 1; i <= n; i++) { cout << res[i] << " "; }
    cout << endl;
}
