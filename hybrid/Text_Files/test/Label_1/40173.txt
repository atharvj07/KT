def solve():
    n = int(input())
    d, g = {}, []
    for i in range(n):
        s = input()
        host = s[7:].split('/')[0]
        if host not in d:
            d[host] = 1
        else:
            d[host] += 1
    for host, cnt in d.items():
        if cnt >= 2:
            g += [list(d.keys())[list(d.values()).index(cnt)] + ' ' + host]
    print(len(g))
    for x in g:
        print(' '.join(sorted(x.split(' '))))

solve()
