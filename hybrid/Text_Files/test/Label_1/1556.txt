#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  int n, m, q;
  cin >> n >> m >> q;
  vector<int> a(n), b(m);
  for (int &x : a) cin >> x;
  for (int &x : b) cin >> x;
  sort(a.begin(), a.end());
  sort(b.begin(), b.end());
  while (q--) {
    int k;
    cin >> k;
    int ans = 0;
    int j = m - 1;
    for (int i = 0; i < n; ++i) {
      while (j >= 0 && (a[i] + k < b[j])) --j;
      if (j >= 0) ans += b[j];
    }
    cout << ans << '\n';
  }
}
