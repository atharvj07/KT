import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        double max = 0;
        for (int i = 0; i <= n; i++) {
            double cal = c + (i * b);
            double price = a + (i * b);
            double tmp = cal / price;
            if (max < tmp) {
                max = tmp;
            }
        }
        System.out.println((int) Math.floor(max));
    }
}
