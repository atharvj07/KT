import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        int[] cnt = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                cnt[i]++;
                if (i != j) {
                    cnt[t[j - 1]]--;
                }
            }
        }
        for (int i = 2; i <= n; i++) {
            cnt[i] += cnt[i - 1];
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(cnt[i] + " ");
        }
        System.out.println();
    }
}
