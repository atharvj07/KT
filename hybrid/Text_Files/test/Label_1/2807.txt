#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int>a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        sort(a.begin(), a.end());
        int mini = a[0], maxi = a.back(), ans = 2;
        for (int i = 1; i < n - 1; i++) {
            if (a[i] == mini) {
                ans = min(ans, i + 1);
            } else if (a[i] == maxi) {
                ans = min(ans, n - i);
            }
        }
        cout << ans << endl;
    }
    return 0;
}
