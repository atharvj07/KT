mod=int(1e9)+7
def pow(base,ex):
    if ex==0: return 1
    if ex==1: return base
    if ex%2==0:
        temp=pow(base,ex//2)
        return (temp*temp)%mod
    else:
        return (base*pow(base,ex-1))%mod

n=int(input())
a=[int(i) for i in input().split()]
ans=[0]*n
M=1
for i in range(2,n+1):
    if a[M]<a[i]:
        ans[M]+=a[M]
        ans[0]+=a[M]
        ans[i]+=a[M]
        M=i
sm=0
for i in range(n):
    sm+=ans[i]
print(sm%mod*pow(n,mod-2))
