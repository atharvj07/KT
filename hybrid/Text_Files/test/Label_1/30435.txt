import java.util.*;

public class Main {
    static Set<Integer> tanks = new HashSet<>();
    static Set<Integer> taps = new HashSet<>();
    static int n, p;
    static int[] d;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        n = scan.nextInt();
        p = scan.nextInt();
        d = new int[n + 1];
        Pipe[] pipes = new Pipe[p + 1];
        for (int i = 1; i <= p; i++) {
            pipes[i] = new Pipe(scan.nextInt(), scan.nextInt(), scan.nextInt());
        }
        for (int i = 1; i <= p; i++) {
            int a = pipes[i].a, b = pipes[i].b;
            if (!tanks.contains(a)) tanks.add(a);
            if (!taps.contains(b)) taps.add(b);
        }
        for (int i = 1; i <= p; i++) {
            int a = pipes[i].a, b = pipes[i].b, c = pipes[i].d;
            if (taps.contains(a)) {
                d[a] = Math.max(d[a], c);
            }
            if (tanks.contains(b)) {
                d[b] = Math.max(d[b], c);
            }
        }
        System.out.println(tanks.size());
        ArrayList<Pair> ans = new ArrayList<>();
        for (int tank : tanks) {
            if (d[tank] != 0) ans.add(new Pair(tank, d[tank]));
        }
        ans.sort((a, b) -> a.a - b.a);
        for (Pair pair : ans) {
            System.out.println(pair.a + " " + pair.d);
        }
    }

    static class Pair {
        int a, d;

        Pair(int a, int d) {
            this.a = a;
            this.d = d;
        }
    }

    static class Pipe {
        int a, b, d;

        Pipe(int a, int b, int d) {
            this.a = a;
            this.b = b;
            this.d = d;
        }
    }
}
