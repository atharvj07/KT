#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    vector<int> sorted(a);
    sort(sorted.begin(), sorted.end());
    int l = -1, r = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] != sorted[i]) {
            if (l == -1) l = i;
            else r = i;
        }
    }
    if (l == -1) cout << "no" << endl;
    else {
        cout << "yes" << endl;
        cout << l + 1 << " " << r + 1 << endl;
    }
    return 0;
}
