import sys
n = int(sys.stdin.readline()[:-1])
s = sys.stdin.readline()[:-1]
bin_s = [0 for i in range(n)]
for i in range(n):
    if s[i] == 'A':
        bin_s[i] = 0
    elif s[i] == 'C':
        bin_s[i] = 1
    elif s[i] == 'G':
        bin_s[i] = 2
    else:
        bin_s[i] = 3
bin_s += bin_s
pos = [0 for i in range(n)]
cnt = [0 for i in range(4)]
for i in range(n):
    cnt[bin_s[i]] += 1
    for j in range(4):
        pos[i] += cnt[j] * (j != bin_s[i])
max_pos = max(pos)
ans = 0
for i in range(n):
    if pos[i] == max_pos:
        ans += 1
print(ans % (10**9 + 7))
