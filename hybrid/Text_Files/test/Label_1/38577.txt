n, p = map(int, input().split())
buyers = [input() for _ in range(n)]

apples = 1
total_money = 0

for buyer in reversed(buyers):
    if buyer == "halfplus":
        apples = apples * 2 + 1
        total_money += (apples // 2) * p
        apples -= 1  # Grandma gives a half apple as a gift
    else:
        apples = apples * 2
        total_money += (apples // 2) * p

print(total_money)
