s=input()
if s[0]=='1' or s[-1]=='0' or '00' in s or '11' in s:
    print(-1)
else:
    z=[]
    t=0
    for i,c in enumerate(s):
        if c=='1':
            t+=1
        else:
            z.append((i+1,t))
            t=0
    print(len(z))
    for a,b in z:
        print(b,end=' ')
        for j in range(a,a+b):
            print(j,end=' ')
        print()
