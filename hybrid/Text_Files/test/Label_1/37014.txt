from collections import defaultdict
s = input()
ans = 0
a = defaultdict(int)
for c in s:
    if c in '([{<':
        a[c] += 1
    else:
        a[c[:-1]] -= 1
        if a[c[:-1]] < 0:
            ans += 1
            a[c[:-1]] = 0
if any(x != 0 for x in a.values()):
    print("Impossible")
else:
    print(ans)
