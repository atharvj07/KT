#include <bits/stdc++.h>
using namespace std;
int l[1000000];
int r[1000000];
int main()
{
    int k;
    cin>>k;
    string s;
    cin.ignore();
    getline(cin,s);
    int n=(int)s.size();
    vector<char> a;
    int cnt=0;
    a.push_back('a');
    for (int i=0;i<n;i++)
    {
        if (s[i]!=' ' && s[i]!='-')
        {
            a.push_back(s[i]);
            cnt++;
        }
        else
        {
            r[cnt]=i;
            cnt=0;
        }
    }
    r[cnt]=n;
    int m=(int)a.size();
    l[0]=1;
    for (int i=1;i<m;i++)
    {
        l[i]=l[i-1]+1;
        if (a[i]=='-') l[i]--;
    }
    // for (int i=0;i<m;i++) cout<<l[i]<<" ";
    // cout<<"\n";
    // for (int i=0;i<m;i++) cout<<r[i]<<" ";
    int ans=0;
    int cur=0;
    for (int i=0;i<m;i++)
    {
        if (cur+l[i]>k)
        {
            ans=max(ans,cur);
            cur=0;
        }
        cur+=l[i];
        if (a[i]=='-') cur--;
    }
    ans=max(ans,cur);
    cout<<ans<<"\n";
}
