#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<int> a(m);
  for (int i = 0; i < m; i++) cin >> a[i];

  sort(a.begin(), a.end());

  int ans = 0;
  for (int i = 0; i < m; i++) {
    if (i == 0) ans += a[i];
    else ans += min(n - (a[i] - 1) + a[i] - 1, (a[i] - a[i - 1]));
  }

  cout << ans << endl;

  return 0;
}
