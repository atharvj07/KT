#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        string a, b;
        cin >> a >> b;
        bool ok = true;
        for (int i = 0; i < n; ++i) {
            if (a[i] < b[i] || (a[i] == b[i] && i + k - 1 < n && b[i] == b[i + k - 1])) {
                continue;
            } else {
                ok = false;
                break;
            }
        }
        if (ok)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
