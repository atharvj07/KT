#include<bits/stdc++.h>
#define rep(i,a,n) for(int i = a; i < n; i++)
#define endl "\n"
using namespace std;
const int MOD = 1e9 + 7;
typedef long long ll;

bool check(pair<ll, ll> a, pair<ll, ll> b) {
  if (a.first == b.first) return true;
  ll c = (b.second - a.second) * (a.first - 0);
  ll d = (a.second - 0) * (b.first - a.first);
  if (c < 0 && d > 0) return true;
  if (c > 0 && d < 0) return true;
  return false;
}

int main() {
  int t; cin >> t;
  while (t--) {
    int n; cin >> n;
    vector<pair<ll, ll>> p(n), q(n);
    map<pair<ll, ll>, int> mp;
    rep(i, 0, n) {
      int a, b, c, d; cin >> a >> b >> c >> d; p[i] = {a, b}; q[i] = {c, d}; mp[{c, d}]++;
    }
    ll ans = 0;
    rep(i, 0, n) {
      for (auto j : mp) {
        if (check(p[i], q[i]) == check(p[i], j.first)) {
          ans += j.second;
        }
      }
      ans -= mp[q[i]];
    }
    cout << ans / 2 << endl;
  }
  return 0;
}
