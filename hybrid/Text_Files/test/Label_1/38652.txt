s = input().strip()
n = len(s)
count = [0]*26
for i in s:
    count[ord(i) - ord('a')] += 1
odd_count = 0
odd_char = ''
for i in range(26):
    if count[i] % 2 != 0:
        odd_count += 1
        odd_char = chr(i + ord('a'))
if odd_count > 1:
    for i in range(26):
        if count[i] % 2 != 0:
            count[i] -= 1
            break
res = ['']*n
start = 0
end = n - 1
for i in range(26):
    while count[i] > 0:
        res[start] = chr(i + ord('a'))
        res[end] = chr(i + ord('a'))
        start += 1
        end -= 1
        count[i] -= 2
if odd_count == 1:
    res[start] = odd_char
print(''.join(res))
