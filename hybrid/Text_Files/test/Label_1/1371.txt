#include <iostream>
#include <vector>
#include <queue>

using namespace std;

const int INF = 2e9;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < n - 1; i++) {
      int u, v;
      cin >> u >> v;
      adj[u].push_back(v);
      adj[v].push_back(u);
    }
    vector<int> dist(n + 1, INF);
    queue<int> q;
    q.push(1);
    dist[1] = 0;
    while (!q.empty()) {
      int u = q.front();
      q.pop();
      for (int v : adj[u]) {
        if (dist[v] == INF) {
          dist[v] = dist[u] + 1;
          q.push(v);
        }
      }
    }
    int ans = 0;
    for (int i = 2; i <= n; i++) {
      ans = max(ans, dist[i]);
    }
    cout << ans << endl;
  }
  return 0;
}
