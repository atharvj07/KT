def decode(s):
    return max(int(''.join(map(str, reversed([not c=='0' for c in s]))), 2))

n, k = map(int, input().split())
s = [input() for _ in range(n)]
used = 0
ans = []
for i in range(n-1, -1, -1):
    t = s[i]
    j = 0
    while j < 7:
        if t[j] == '0':
            if used < k:
                t = t[:j] + '1' + t[j+1:]
                used += 1
                break
        j += 1
    ans.append(t)

if used < k:
    print(-1)
else:
    print(''.join(map(str, map(decode, ans[::-1]))))
