from collections import deque

for _ in range(int(input())):
    n, k = map(int, input().split())
    a = deque(input())
    l, r = 0, n - 1

    while l < r and k > 0:
        if a[l] > a[r]:
            a[l], a[r] = a[r], a[l]
            k -= 1
        l += 1
        r -= 1

    print(''.join(a))
