import java.util.*;

class Main {
    static final long MOD = 1000000007;
    static int N, M;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        int[] S = new int[N];
        int[] T = new int[M];
        for (int i = 0; i < N; i++) S[i] = sc.nextInt();
        for (int i = 0; i < M; i++) T[i] = sc.nextInt();
        int[][] freqS = new int[N + 1][100001];
        for (int i = N - 1; i >= 0; i--) {
            for (int j = 0; j < 100001; j++) freqS[i][j] = freqS[i + 1][j];
            freqS[i][S[i]]++;
        }
        int[][] freqT = new int[M + 1][100001];
        for (int i = M - 1; i >= 0; i--) {
            for (int j = 0; j < 100001; j++) freqT[i][j] = freqT[i + 1][j];
            freqT[i][T[i]]++;
        }
        long ans = 0;
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < 100001; j++) {
                ans += (long) freqS[i + 1][j] * freqT[0][j];
            }
        }
        ans %= MOD;
        System.out.println(ans);
    }
}
