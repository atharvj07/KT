n = int(input())
a = list(map(int, input().split()))

distinct_values = set()
prefix_or = 0
for i in range(n):
    new_prefix_or = a[i]
    new_values = {new_prefix_or}
    for prev_or in distinct_values:
        new_prefix_or = prev_or | a[i]
        new_values.add(new_prefix_or)
    distinct_values = new_values
    prefix_or |= a[i]
    distinct_values.add(prefix_or)

print(len(distinct_values))
