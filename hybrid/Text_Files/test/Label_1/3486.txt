#include <bits/stdc++.h>
#define int long long
using namespace std ;
const int N = 3e5+5;
vector<pair<int,int>> g[N];
int w[N],dp[N];
int dfs(int u,int p){
    dp[u]=w[u];
    for(auto v : g[u]){
        if(v.first==p) continue;
        dp[u] = max(dp[u],dfs(v.first,u) + w[u] - v.second);
    }
    return dp[u];
}
int32_t main(){
    int n; cin >> n;
    for(int i=1;i<=n;i++) cin >> w[i];
    for(int i=1;i<n;i++){
        int u,v,c; cin >> u >> v >> c;
        g[u].push_back({v,c});
        g[v].push_back({u,c});
    }
    cout << dfs(1,1) << '\n';
    return 0;
}
