#include <bits/stdc++.h>
using namespace std;
int a[100000], pre[100000];
int main() {
  int t, n, x, mx = 0;
  cin >> t;
  while (t--) {
    cin >> n >> x;
    a[0] = 0;
    pre[0] = 0;
    mx = 0;
    for (int i = 1; i <= n; i++) {
      cin >> a[i];
      pre[i] = (pre[i - 1] + a[i]) % x;
    }
    map<int, int> mp;
    for (int i = 0; i <= n; i++) {
      mx = max(mx, i - mp[pre[i]]);
      mp[pre[i]] = i;
    }
    if (!mx)
      mx = -1;
    cout << mx << endl;
  }
  return 0;
}
