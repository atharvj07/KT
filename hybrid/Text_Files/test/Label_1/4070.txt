#include <iostream>
using namespace std;

int main()
{
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    int cur = 1, ans = 0;
    for (int i = 1; i < n; i++)
    {
        if (s[i] == s[i - 1])
            cur++;
        else
            cur = 1;
        if (cur == k)
        {
            ans++;
            cur = 1;
        }
    }
    cout << ans << "\n";
    return 0;
}
