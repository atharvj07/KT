import sys

t = int(input())
x = list(map(int, input().split()))
r = 10**6 - sum(x)
l = []
a = []
x.sort(reverse = True)
for i in x:
    l.append(i - 1)
l.sort()
a.append(r - l[-1])
for i in l[:-1]:
    if i + a[-1] <= r:
        a[-1] = i + a[-1]
    else:
        a.append(i)
print(len(a))
print(*a)
