#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  string s; cin >> s;
  long long int l = 0, r = 0, ans = 0;
  for(int i = 0; i < s.size(); i++) {
    if(s[i] == 'b') r++;
    else {
      ans += min(l, r);
      l = r;
      r = 0;
    }
  }
  ans += min(l, r);
  cout << ans % 1000000007 << endl;
  return 0;
}
