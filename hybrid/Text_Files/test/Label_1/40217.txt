import sys
input = sys.stdin.readline
mod = 1000000007

def bi(a,k):
	l,r=1,len(a)-1
	while l<=r:
		m=(l+r)//2
		if a[m]==k:
			if m==len(a)-1: return -1
			return m
		elif a[m]>k:
			r=m-1
		else: l=m+1
	return -1

ans = 0
n = int(input())
a = list(map(int,input().split()))
b = sorted(a)
for i in range(n):
	ans=(ans+a[i]*(bi(b,a[i])+1))%mod
print(ans)
