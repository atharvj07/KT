#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n);
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  vector<int> dir(m);
  for (int i = 0; i < n; i++) {
    if (adj[i].size() > 2) {
      cout << "NO" << endl;
      return 0;
    }
    if (adj[i].size() == 1) {
      dir[adj[i][0]] = 1;
    }
  }
  cout << "YES" << endl;
  for (int d : dir) {
    cout << d;
  }
  cout << endl;
}
