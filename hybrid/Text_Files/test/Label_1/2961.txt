#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> c(n), a(n);
  for (int i = 0; i < n; i++) { cin >> c[i]; }
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> vis(n);
  int ans = 0;
  for (int i = 0; i < n; i++) {
    if (vis[i] == 1) continue;
    int cur = i, cost = 0;
    while (vis[cur] == 0) {
      vis[cur] = 1;
      cost = max(cost, c[cur]);
      cur = a[cur] - 1;
    }
    ans += cost;
  }
  cout << ans << endl;
  return 0;
}
