import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    long[] ar = new long[in.nextInt()];
    long d = in.nextInt();
    for (int i = 0; i < ar.length; i++) {
      ar[i] = in.nextInt();
    }
    long c = 0;
    for (int i = 0; i < ar.length; i++) {
      int l = i + 1, r = ar.length - 1;
      int j = i;
      while (l <= r) {
        int m = (l + r) / 2;
        if (ar[m] - ar[i] <= d) {
          l = m + 1;
          j = m;
        } else {
          r = m - 1;
        }
      }
      c += j - i;
    }
    System.out.println(c);
  }
}
