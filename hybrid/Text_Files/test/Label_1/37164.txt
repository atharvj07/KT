from collections import defaultdict, Counter
s = input()
n = int(input())
rules = defaultdict(list)
for i in range(n):
    p, l, r = input().split()
    l = int(l)
    r = int(r)
    rules[p].append((l, r))
cnt = 0
for i in range(len(s)):
    for j in range(i + 1, len(s) + 1):
        t = s[i:j]
        is_good = True
        for p in rules:
            if t not in rules[p]:
                is_good = False
            else:
                c = Counter([k for l, r in rules[p] for k in range(l, r + 1)])[len(t)]
                if c < rules[p][0][0] or c > rules[p][-1][1]:
                    is_good = False
            if not is_good:
                break
        if is_good:
            cnt += 1
print(cnt)
