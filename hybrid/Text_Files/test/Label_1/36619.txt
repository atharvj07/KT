def segments_count(a, b):
    segments = [6, 2, 5, 5, 4, 5, 6, 3, 7, 6]  # 0-9 segment counts
    total_segments = 0
    for number in range(a, b + 1):
        while number:
            total_segments += segments[number % 10]
            number //= 10
    return total_segments

a, b = map(int, input().split())
print(segments_count(a, b))
