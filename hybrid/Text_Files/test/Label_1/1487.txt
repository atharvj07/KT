#include <bits/stdc++.h>
#define fi first
#define se second
#define mp make_pair
#define pb push_back
const long long INF = 1e9 + 7;

using namespace std;

typedef long long ll;
typedef pair<ll, ll> pi;
typedef vector<ll> vi;

struct node {
    ll t, q;
    node* l, * r;
    node(ll t, ll q, node* l = nullptr, node* r = nullptr): t(t), q(q), l(l), r(r) {}
};

ll n, T;

ll f(node* u) {
    if (!u) {
        return 0;
    }
    if (!u -> l && !u -> r) {
        return u -> q;
    }
    ll ans = 0;
    ans = max(ans, u -> q);
    if (T >= u -> t) {
        if (u -> l) {
            ans = max(ans, u -> q + f(u -> l));
        }
        if (u -> r) {
            ans = max(ans, u -> q + f(u -> r));
        }
    }
    return ans;
}

int main() {
    cin >> n >> T;
    node* head = nullptr;
    for (ll i = 0; i < n; i++) {
        ll a, b;
        cin >> a >> b;
        if (i) {
            head = new node(a, b, head, nullptr);
        } else {
            head = new node(a, b);
        }
    }
    cout << f(head) << "\n";
}
