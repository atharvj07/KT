def generate_words():
    alphabet = "abcdefghijklmnopqrstuvwxyz"
    for length in range(1, 6):  # Assuming max length of 5 for the word to be generated
        for i in range(26**length):
            word = ""
            temp = i
            for _ in range(length):
                word = alphabet[temp % 26] + word
                temp //= 26
            yield word

def minify_code(n, reserved_tokens, m, source):
    reserved = set(reserved_tokens.split())
    word_map = {}
    new_word_generator = generate_words()

    tokens = []
    for _ in range(m):
        line = source.pop(0)
        line = line.split("#")[0]  # Remove comments
        i = 0
        while i < len(line):
            if line[i] == ' ':
                i += 1
                continue
            j = i
            while j < len(line) and line[j] != ' ':
                j += 1
            token = line[i:j]
            if token.isdigit() or token in reserved or not token[0].isalpha() or not token[0].islower():
                tokens.append(token)
            else:
                if token not in word_map:
                    word_map[token] = next(new_word_generator)
                tokens.append(word_map[token])
            i = j

    return ''.join(token if token in reserved or not token.isalpha() or not token[0].islower() else ' ' + token for token in tokens).strip()

n = int(input().strip())
reserved_tokens = input().strip()
m = int(input().strip())
source = [input().strip() for _ in range(m)]

print(minify_code(n, reserved_tokens, m, source))
