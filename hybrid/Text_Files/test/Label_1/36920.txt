q = int(input().strip())
for _ in range(q):
    h, n = map(int, input().split())
    platforms = list(map(int, input().split()))
    platforms.append(0)  # Adding ground level for simplification
    crystals_needed = 0
    i = 0
    while i < n:
        if platforms[i] - platforms[i + 1] > 1:
            crystals_needed += 1
            i += 1  # Skip next platform as it's now out (due to crystal use)
        else:
            i += 2  # Move to the next potential platform to jump
    print(crystals_needed)
