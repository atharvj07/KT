import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), w = sc.nextInt(), k = sc.nextInt();
        int[] a = new int[n], t = new int[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        for (int i = 0; i < n; i++) t[i] = sc.nextInt();
        int[][] dp = new int[n][w + 1];
        for (int i = 0; i < n; i++) for (int j = 0; j <= w; j++) dp[i][j] = -1;
        int ans = solve(0, 0, a, t, dp, k, w);
        System.out.println(ans);
    }

    private static int solve(int idx, int cnt, int[] a, int[] t, int[][] dp, int k, int w) {
        if (idx == a.length || cnt > w) return 0;
        if (dp[idx][cnt] != -1) return dp[idx][cnt];
        int op1 = 0, op2 = 0;
        if (t[idx] <= k) op1 = a[idx] + solve(idx + 1, cnt, a, t, dp, k - t[idx], w);
        if (t[idx] / 2 <= k) op2 = solve(idx + 1, cnt + 1, a, t, dp, k, w);
        dp[idx][cnt] = Math.max(op1, op2);
        return dp[idx][cnt];
    }
}
