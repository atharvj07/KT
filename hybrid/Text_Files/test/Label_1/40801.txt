import sys
input = sys.stdin.readline

n = int(input())
l = [list(map(float, input().split())) for i in range(n)]
EPS = 1e-9
def dot(a, b):
    return a[0]*b[0] + a[1]*b[1]
def cross(a, b):
    return a[0]*b[1] - a[1]*b[0]
def norm(v):
    return (v[0]*v[0] + v[1]*v[1])**0.5

ans = -1
for i in range(n):
    x1, y1 = l[i]
    for j in range(i+1, n):
        x2, y2 = l[j]
        v1 = [x2-x1, y2-y1]
        d1 = norm(v1)
        mv = v1
        mv[0], mv[1] = -mv[1], mv[0]
        mv = [mv[0]/d1, mv[1]/d1]
        ok = True
        for k in range(n):
            x3, y3 = l[k]
            d = abs(cross(v1, [x3-x1, y3-y1])/d1)
            if d < EPS:
                ok = False
                break
            elif d > 1+EPS:
                ok = False
                break
            else:
                if dot(mv, [x3-x1, y3-y1]) < -EPS:
                    ok = False
                    break
        if ok:
            if d1*0.5 < ans or ans == -1:
                ans = d1*0.5

print(ans)
