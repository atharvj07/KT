#include <bits/stdc++.h>
using namespace std;

int main() {
  int t, n;
  cin >> t;
  while (t--) {
    cin >> n;
    vector<int> v(n);
    map<int, int> cnt;
    for (int &x : v) {
      cin >> x;
      cnt[x]++;
    }
    long long sum = 0;
    int mx = 0;
    sort(v.begin(), v.end());
    for (auto e : v) {
      sum += e;
      mx = max(mx, cnt[e]);
    }
    if (mx > n / 2) cout << -1 << endl;
    else {
      cout << (abs(sum) + (n / 2 - mx)) / (n / 2) << endl;
    }
  }
  return 0;
}
