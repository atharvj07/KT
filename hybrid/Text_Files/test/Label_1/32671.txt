import java.util.Scanner;

class Main {
    static int[] medianFilter(int[] a) {
        int n = a.length;
        int[] b = new int[n];

        // Initialize the first and last elements
        b[0] = a[0];
        b[n - 1] = a[n - 1];

        // Iterate over the remaining elements
        for (int i = 1; i < n - 1; i++) {
            // Compute the median of the three adjacent elements
            int med = (a[i - 1] + a[i] + a[i + 1]) / 3;

            // Assign the median to the corresponding element in the new array
            b[i] = med;
        }

        return b;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Read the length of the sequence
        int n = scanner.nextInt();

        // Read the sequence of zeros and ones
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        // Initialize the number of applications of the median filter
        int applications = 0;

        // Keep applying the median filter until the sequence becomes stable
        while (true) {
            // Apply the median filter
            int[] b = medianFilter(a);

            // Check if the sequence is stable
            boolean stable = true;
            for (int i = 0; i < n; i++) {
                if (a[i] != b[i]) {
                    stable = false;
                    break;
                }
            }

            // If the sequence is stable, break the loop
            if (stable) {
                break;
            }

            // Increment the number of applications
            applications++;

            // Update the sequence
            a = b;
        }

        // Print the result
        if (applications == 0) {
            System.out.println(1);
        } else {
            System.out.println(applications);
            for (int i = 0; i < n; i++) {
                System.out.print(a[i] + " ");
            }
            System.out.println();
        }
    }
}
