import java.util.*;
import java.io.*;

class Main
{
    public static void main(String[] args)
    {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++)
            a[i] = cin.nextInt();
        int[] ans1 = get_ans(a), ans2 = new int[n];
        for (int i = 0; i < n; i++)
            ans2[i] = 3 - ans1[i];
        for (int i = 0; i < n; i++)
            System.out.print(ans1[i] + " ");
        System.out.println("");
    }

    static int[] get_ans(int[] p)
    {
        int[] ans = new int[p.length];
        int q = 0;
        for (int i = 0; i < p.length; i++)
            if (p[i] == 1)
                ans[q++] = 1;
        for (int i = 0; i < p.length; i++)
            if (p[i] == 2)
                ans[q++] = 2;
        return ans;
    }
}
