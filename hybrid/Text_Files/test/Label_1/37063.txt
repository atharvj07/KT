n = int(input())
support_none = []
support_alice = []
support_bob = []
support_both = []

for _ in range(n):
    s, influence = input().split()
    influence = int(influence)
    if s == "00":
        support_none.append(influence)
    elif s == "10":
        support_alice.append(influence)
    elif s == "01":
        support_bob.append(influence)
    elif s == "11":
        support_both.append(influence)

support_alice.sort(reverse=True)
support_bob.sort(reverse=True)
support_none.sort(reverse=True)
support_both.sort(reverse=True)

total_influence = sum(support_both)
min_supporters = len(support_both)

alice_extra = len(support_both)
bob_extra = len(support_both)

alice_index = 0
bob_index = 0

while alice_index < len(support_alice) and bob_index < len(support_bob):
    if alice_extra >= bob_extra:
        total_influence += support_bob[bob_index]
        bob_index += 1
        min_supporters += 1
        bob_extra += 1
    else:
        total_influence += support_alice[alice_index]
        alice_index += 1
        min_supporters += 1
        alice_extra += 1

while alice_index < len(support_alice):
    total_influence += support_alice[alice_index]
    alice_index += 1
    min_supporters += 1
    alice_extra += 1

while bob_index < len(support_bob):
    total_influence += support_bob[bob_index]
    bob_index += 1
    min_supporters += 1
    bob_extra += 1

if alice_extra * 2 >= min_supporters and bob_extra * 2 >= min_supporters:
    for influence in support_none:
        if alice_extra * 2 > min_supporters and bob_extra * 2 > min_supporters:
            break
        total_influence += influence
        min_supporters += 1
    print(total_influence)
else:
    print(0)
