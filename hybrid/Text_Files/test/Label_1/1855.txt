#include <iostream>
#include <unordered_map>
using namespace std;

int main() {
  int n;
  string home;
  unordered_map<string, int> map;
  cin >> n >> home;
  while (n--) {
    string a, b;
    cin >> a >> b;
    map[a]++;
    map[b]--;
  }
  for (auto p : map) {
    if (p.second && p.first != home) {
      cout << "contest";
      return 0;
    }
  }
  cout << "home";
}
