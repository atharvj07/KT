#include <cstdio>
#include <vector>

int n;
std::vector<int> adj[100005];
bool used[100005];
int mx[100005];
int sz[100005];

void dfs(int x, int p) {
  sz[x] = 1;
  used[x] = 1;
  for (auto v : adj[x]) {
    if (used[v])
      continue;
    dfs(v, x);
    sz[x] += sz[v];
  }
}

int dfs1(int x, int p, int l) {
  mx[x] = l;
  used[x] = 1;
  for (auto v : adj[x]) {
    if (used[v])
      continue;
    mx[x] = dfs1(v, x, l + sz[v]);
  }
  return mx[x];
}

int dfs2(int x, int p, int l) {
  mx[x] = l;
  used[x] = 1;
  for (auto v : adj[x]) {
    if (used[v])
      continue;
    if (mx[x] == l + sz[v])
      mx[x] = dfs2(v, x, l);
    else
      mx[x] = dfs2(v, x, l + sz[v]);
  }
  return mx[x];
}

int main() {
  scanf("%d", &n);
  for (int i = 0; i < n - 1; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    adj[x].push_back(y);
    adj[y].push_back(x);
  }
  dfs(1, 1);
  used[1] = 0;
  dfs1(1, 1, 0);
  for (int i = 1; i <= n; ++i)
    used[i] = 0;
  int m = 0;
  for (int i = 1; i <= n; ++i)
    if (mx[i] > m)
      m = mx[i];
  dfs2(1, 1, 0);
  int M = 0;
  for (int i = 1; i <= n; ++i)
    if (mx[i] > M)
      M = mx[i];
  M = (M + 1) / 2;
  printf("%d %d\n", m == M ? m : m + 1, M);
  return 0;
}
