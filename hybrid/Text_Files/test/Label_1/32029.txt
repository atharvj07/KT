import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int k = input.nextInt();
        int a = input.nextInt();
        int b = input.nextInt();
        int v = input.nextInt();

        int minBoxes = 0;
        int[] box = new int[k];
        int i = 0;
        while (a > 0) {
            minBoxes++;
            while (i < k && a > 0) {
                while (b > 0 && box[i] + v <= v * k) {
                    b--;
                    box[i] += v;
                    a--;
                }
                i++;
            }
            i = 0;
            Arrays.fill(box, 0);
        }
        System.out.println(minBoxes);
    }
}
