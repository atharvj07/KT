#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;cin>>t;
    while(t--){
        string s;cin>>s;
        int a=0,b=0,c=0;
        for(char x:s){
            if(x=='A')a++;
            else if(x=='B')b++;
            else c++;
        }
        if(a%2==0 && b%2==0 && c%2==0)cout<<"YES"<<endl;
        else if(a==0 || (a%2!=0 && b%2!=0))cout<<"NO"<<endl;
        else if(b==0 || (a%2!=0 && c%2!=0))cout<<"NO"<<endl;
        else if(c==0 || (b%2!=0 && c%2!=0))cout<<"NO"<<endl;
        else cout<<"YES"<<endl;
    }
}
