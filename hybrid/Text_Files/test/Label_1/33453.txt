import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        long[] dp = new long[n];
        for (int i = 0; i < k; i++) {
            dp[i] = a[i];
        }
        for (int i = k; i < n; i++) {
            dp[i] = a[i] + dp[i - k];
        }
        long min = Long.MAX_VALUE;
        int ans = 1;
        for (int i = 0; i < k; i++) {
            if (dp[i] < min) {
                min = dp[i];
                ans = i + 1;
            }
        }
        System.out.println(ans);
    }
}
