from sys import stdin

n, m = map(int, stdin.readline().split())
grid = [[0]*m for _ in range(n)]
goals = [[0]*m for _ in range(n)]
for i in range(n):
    goals[i] = list(map(int, stdin.readline().split()))
for i in range(n):
    for j in range(m):
        grid[i][j] = goals[i][j] - (i+j)%2

ok = True
for i in range(n):
    for j in range(m):
        if grid[i][j] > 0:
            ok = False
if not ok:
    print(-1)
    exit()
ans = []
for i in range(n):
    for j in range(m):
        if grid[i][j] < 0:
            ans.append('row %d' % (i+1))
            for k in range(m):
                grid[k][j] += 1
for j in range(m):
    for i in range(n):
        if grid[i][j] < 0:
            ans.append('col %d' % (j+1))
            for k in range(n):
                grid[k][j] += 1
print(len(ans))
for step in ans:
    print(step)
