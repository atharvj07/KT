import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int num = n - 1;
        int den = 1;
        for (int i = n; i > 1; i--) {
            num *= i;
        }
        for (int i = 1; i < n; i++) {
            den *= i;
        }
        System.out.println(num + "/" + den);
    }
}
