import java.util.ArrayList;
import java.util.Scanner;

public class Main {

	static int[] prefCnts = new int[200001];
	static int[] prodRates = new int[200001];
	static int[] queries = new int[200001];
	static ArrayList<Integer> queryDays = new ArrayList<Integer>();

	public static void main(String[] args) {
		Scanner in = new Scanner(System.in);
		int n = in.nextInt();
		int k = in.nextInt();
		int a = in.nextInt();
		int b = in.nextInt();
		int q = in.nextInt();
		in.nextLine();
		for (int i = 0; i < q; i++) {
			String line = in.nextLine();
			String[] parts = line.split(" ");
			if (parts[0].equals("1")) {
				int day = Integer.parseInt(parts[1]);
				int cnt = Integer.parseInt(parts[2]);
				prefCnts[day] += cnt;
			} else {
				int day = Integer.parseInt(parts[1]);
				queryDays.add(day);
				queries[day] = i + 1;
			}
		}

		for (int i = 1; i <= n; i++) {
			prodRates[i] = (i <= k) ? b : a;
		}

		int sum = 0;
		for (int i = 1; i <= n; i++) {
			sum += prefCnts[i - 1];
			prefCnts[i] = Math.min(sum, prodRates[i]);
			sum -= prodRates[i];
		}

		for (int day : queryDays) {
			int ans = prefCnts[day + k - 1];
			if (day > 1) {
				ans -= prefCnts[day - 1];
			}
			System.out.println(ans);
		}
	}
}
