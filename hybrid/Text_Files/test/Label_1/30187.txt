import java.util.*;

public class Main {
    static int n, k;
    static List<int[]> dwarfs;
    static List<int[]> stations;
    static int[][] dist;
    static int[] px = {-1, 1, 0, 0};
    static int[] py = {0, 0, -1, 1};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        k = sc.nextInt();
        dwarfs = new ArrayList<>();
        stations = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            dwarfs.add(new int[]{x, y});
        }
        for (int i = 0; i < k; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            stations.add(new int[]{x, y});
        }

        dist = new int[n][k];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dist[i], -1);
        }

        int ans = Integer.MAX_VALUE;
        for (int[] station : stations) {
            ans = Math.min(ans, bfs(station));
        }

        System.out.println(ans);
    }

    static int bfs(int[] station) {
        Queue<int[]> q = new LinkedList<>();
        for (int i = 0; i < n; i++) {
            dist[i][0] = 0;
            q.add(new int[]{i, 0});
        }

        while (!q.isEmpty()) {
            int[] cur = q.poll();
            int id = cur[0];
            int prev = cur[1];
            for (int i = 0; i < 4; i++) {
                int x = dwarfs.get(id)[0] + px[i];
                int y = dwarfs.get(id)[1] + py[i];
                if (x == station[0] && y == station[1]) {
                    return prev + 1;
                }
                for (int j = 0; j < k; j++) {
                    if (x == stations.get(j)[0] && y == stations.get(j)[1] && dist[id][j] == -1) {
                        dist[id][j] = prev + 1;
                        q.add(new int[]{id, j});
                    }
                }
            }
        }

        return Integer.MAX_VALUE;
    }
}
