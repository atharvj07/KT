import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int gcd = 1;
        for (int i = 1; i <= Math.min(a, b); i++) {
            gcd *= i;
        }
        System.out.println(gcd);
    }
}
