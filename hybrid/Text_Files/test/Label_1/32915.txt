import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] p = new int[n];
        int[] f = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = in.nextInt() - 1;
            f[i] = in.nextInt() - 1;
        }
        boolean[] vis = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (vis[i]) continue;
            int cnt = 0;
            int cur = i;
            while (!vis[cur]) {
                vis[cur] = true;
                cnt++;
                cur = p[cur];
            }
            if (cnt % 2 != f[i] % 2) System.out.println("NO");
            return;
        }
        System.out.println("YES");
    }
}
