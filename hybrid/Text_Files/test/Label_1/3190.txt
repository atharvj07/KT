#include<bits/stdc++.h>
#define REP(i,a,b) for(int i=a;i<=b;++i)
using namespace std;
const int maxn=1e6+7;
int n,k;
long long p[maxn],c[maxn];
bool vis[maxn];
vector<int> G[maxn];
int main()
{
    cin>>n>>k;
    REP(i,1,n) cin>>p[i];
    REP(i,1,n) cin>>c[i];
    sort(p+1,p+n+1);
    REP(i,1,n)
    {
        int now=i;
        while(now>1&&p[now]==p[now-1]) now--;
        G[now].push_back(i);
    }
    long long ans[maxn]={};
    for(int i=1;i<=n;++i) ans[i]+=c[i];
    long long dp[maxn]={};
    for(int i=n;i>=1;--i)
    {
        for(int j=0;j<G[i].size();++j)
        {
            int u=G[i][j];
            if(dp[u]==0) continue;
            if(k>=1)
            {
                dp[i]=max(dp[i],dp[u]+c[i]+ans[u+1]);
                k--;
            }
        }
        ans[i]=max(ans[i],dp[i]);
    }
    REP(i,1,n) cout<<ans[i]<<' ';
    return 0;
}
