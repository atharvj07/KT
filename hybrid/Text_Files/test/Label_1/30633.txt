import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int t = sc.nextInt();
        int[] dp = new int[n + 1];
        dp[2] = t;
        if (n == 2) {
            System.out.println(t);
            return;
        }
        for (int i = 3; i <= n; i++) {
            for (int j = 2; j <= i - 1; j++) {
                for (int k = 1; k <= 4; k++) {
                    if (j == 2 || dp[j - 1] > k) {
                        if (j == i - 1 || dp[j + 1] < k) {
                            dp[i] += dp[j];
                            dp[i] %= 1000000007;
                        }
                    }
                }
            }
        }
        System.out.println(dp[n]);
    }
}
