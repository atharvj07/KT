def factorial_mod(n, mod):
    if n == 0:
        return 1
    else:
        return n * factorial_mod(n - 1, mod) % mod

def solve(n):
    mod = 998244353
    if n == 1:
        return 1
    fact_n = factorial_mod(n, mod)
    ans = (fact_n * (fact_n + 1) // 2) % mod
    for i in range(1, n):
        ans = (ans + (factorial_mod(i, mod) * (factorial_mod(n - i, mod) - 1)) % mod) % mod
    return ans

n = int(input())
print(solve(n))
