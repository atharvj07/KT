import java.util.Scanner;

public class Main {
    private static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long m = in.nextLong();
        long ans = 0;
        long pw7 = 1;
        for (long i = 1; i <= m; i++) {
            long lucky = 0;
            long pwi = pw7;
            long cur = i;
            while (cur > 0) {
                if (cur % 10 == 4 || cur % 10 == 7) {
                    lucky += pwi % MOD;
                }
                cur /= 10;
                pwi *= pw7;
                pwi %= MOD;
            }
            long sum = (7 * lucky % MOD - lucky) % MOD;
            if (sum >= 0) ans += sum;
            else ans += sum + MOD;
            pw7 *= 7;
            pw7 %= MOD;
        }
        System.out.println(ans % MOD);
    }
}
