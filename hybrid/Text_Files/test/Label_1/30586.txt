import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            t[a[i] - 1]++;
        }
        int ans = 0, cur = 0;
        for (int i = 0; i < n; i++) {
            t[i] += cur;
            cur = t[i];
            if (t[i] >= i + 1) ans++;
        }
        System.out.println(ans);
    }
}
