#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &ai : a) { cin >> ai; }
    int ans = 0, idx = 0;
    bool vis[n + 1];
    fill(vis, vis + n + 1, false);
    while (idx < n) {
        int cur = idx;
        while (!vis[cur + 1]) {
            vis[cur + 1] = true;
            cur = a[cur] - 1;
        }
        idx = cur + 1;
        ans++;
    }
    cout << ans;
    return 0;
}
