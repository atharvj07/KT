import java.util.*;
import java.io.*;
import java.lang.*;

public class Main {
    static int n, m;
    static int[][] arr;
    static boolean[][] visited;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        arr = new int[n + 1][n + 1];
        visited = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                arr[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < m; i++) {
            int o = sc.nextInt();
            if (o == 0) {
                int r = sc.nextInt();
                int c = sc.nextInt();
                int size = sc.nextInt();
                int angle = sc.nextInt();
                rotate(r, c, size, angle);
            } else if (o == 1) {
                int r = sc.nextInt();
                int c = sc.nextInt();
                int size = sc.nextInt();
                reverse(r, c, size);
            } else if (o == 2) {
                int r = sc.nextInt();
                leftShift(r);
            } else if (o == 3) {
                int r = sc.nextInt();
                rightShift(r);
            } else {
                int r = sc.nextInt();
                int c = sc.nextInt();
                islandReversal(r, c);
            }
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                System.out.print(arr[i][j] + " ");
            }
            System.out.println();
        }
    }

    static void rotate(int r, int c, int size, int angle) {
        int[][] temp = new int[size + 1][size + 1];
        for (int i = r; i < r + size; i++) {
            for (int j = c; j < c + size; j++) {
                temp[i - r][j - c] = arr[i][j];
            }
        }
        if (angle == 90) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    arr[r + j][c + size - 1 - i] = temp[i][j];
                }
            }
        } else if (angle == 180) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    arr[r + size - 1 - i][c + size - 1 - j] = temp[i][j];
                }
            }
        } else if (angle == 270) {
            for (int i = 0; i < size; i++) {
                for (int j = 0; j < size; j++) {
                    arr[r + size - 1 - j][c + i] = temp[i][j];
                }
            }
        }
    }

    static void reverse(int r, int c, int size) {
        for (int i = r; i < r + size; i++) {
            for (int j = c; j < c + size; j++) {
                arr[i][j] = 1 - arr[i][j];
            }
        }
    }

    static void leftShift(int r) {
        int[] temp = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            temp[i] = arr[r][i];
        }
        for (int i = 2; i <= n; i++) {
            arr[r][i - 1] = temp[i];
        }
        arr[r][n] = temp[1];
    }

    static void rightShift(int r) {
        int[] temp = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            temp[i] = arr[r][i];
        }
        for (int i = n - 1; i >= 1; i--) {
            arr[r][i + 1] = temp[i];
        }
        arr[r][1] = temp[n];
    }

    static void islandReversal(int r, int c) {
        if (visited[r][c]) {
            return;
        }
        visited[r][c] = true;
        if (arr[r][c] == 0) {
            return;
        }
        arr[r][c] = 0;
        if (r > 1 && arr[r - 1][c] == 1) {
            islandReversal(r - 1, c);
        }
        if (r < n && arr[r + 1][c] == 1) {
            islandReversal(r + 1, c);
        }
        if (c > 1 && arr[r][c - 1] == 1) {
            islandReversal(r, c - 1);
        }
        if (c < n && arr[r][c + 1] == 1) {
            islandReversal(r, c + 1);
        }
    }
}
