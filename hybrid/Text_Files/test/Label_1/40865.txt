from sys import stdin
from collections import Counter

mod = 10**9 + 7

for _ in range(int(stdin.readline())):
    n, k = map(int, stdin.readline().split())
    a = Counter(stdin.readline().split())
    lst = sorted([*a.keys()])[::-1]
    ans = 1
    for i in range(k):
        ans *= a[lst[i]]
        ans %= mod
    print(ans)
