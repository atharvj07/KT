import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            b[i] = scanner.nextLong();
        }
        for (int i = 1; i < n; i++) {
            b[i] ^= b[i - 1];
        }
        Arrays.sort(b);
        boolean valid = true;
        for (int i = 1; i < n; i++) {
            if (b[i] <= b[i - 1]) {
                valid = false;
            }
        }
        if (valid) {
            System.out.println("Yes");
            for (long v : b) {
                System.out.print(v + " ");
            }
            System.out.println();
        } else {
            System.out.println("No");
        }
    }
}
