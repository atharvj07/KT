#include <bits/stdc++.h>
#define modulo 1000000007
using namespace std;
typedef long long ll;
ll powMod(ll a, ll b){
    ll ans = 1;
    while(b>0){
        if(b%2){
            ans = (ans*a)%modulo;
        }
        a=(a*a)%modulo;
        b/=2;
    }
    return ans;
}
int main(){
    ll n, k; cin>>n>>k;
    ll ans = powMod(k, n)- powMod(k-1, n);
    ans%=modulo;
    ans+= modulo;
    ans %=modulo;
    cout<<ans<<'\n';
    return 0;
}
