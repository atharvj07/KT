import sys
input=sys.stdin.readline
n=int(input())
a=list(map(int,input().split()))
e=[[] for i in range(n+1)]
for i in range(n-1):
    u,v=map(int,input().split())
    e[u].append(v)
    e[v].append(u)
vis=[0]*(n+1)
ans=[0,0]
def dfs(u,p,s):
    vis[u]=1
    ans[not s]+=a[u-1]
    for v in e[u]:
        if vis[v]==0:
            dfs(v,u,s^1)
dfs(1,0,0)
if ans[0]==ans[1]:print(ans[0])
else: print("Impossible")
