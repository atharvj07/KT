#include <bits/stdc++.h>
using namespace std;
const int N = 3e5 + 5;
int32_t main() {
    int n, m;
    cin >> n >> m;
    bool vis[n + 5] = {false};
    int a, b;
    for (int i = 0; i < m; i++) {
        cin >> a >> b;
        vis[a] = true;
        vis[b] = true;
    }
    int cnt = 0;
    for (int i = 1; i <= n; i++) {
        cnt += vis[i];
    }
    if (cnt >= 2)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
}
