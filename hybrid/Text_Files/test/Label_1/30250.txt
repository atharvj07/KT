import java.util.*;

public class Main {
    public static void main(String[] agrs) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int w = sc.nextInt();
        int d = sc.nextInt();
        char[][] c = new char[h][w];
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                c[i][j] = (i + j) % d == 0 ? 'R' : (i + j) % d == 1 ? 'Y' : (i + j) % d == 2 ? 'G' : 'B';
            }
        }
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                System.out.print(c[i][j]);
            }
            System.out.println();
        }
    }
}
