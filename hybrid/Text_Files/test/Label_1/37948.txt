n = int(input())

def is_fibonacci(n):
    fib1, fib2 = 1, 1
    while fib1 <= n:
        if fib1 == n:
            return True
        fib1, fib2 = fib2, fib1 + fib2
    return False

name = ""
for i in range(1, n+1):
    if is_fibonacci(i):
        name += "O"
    else:
        name += "o"

print(name)
