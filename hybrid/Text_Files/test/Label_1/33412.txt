import java.util.Scanner;

public class Main {
    static int[][] dp;
    static int S, T, U, N, O, D;
    static int[] e, c, h, a;
    static boolean flag = true;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        S = sc.nextInt();
        T = sc.nextInt();
        U = sc.nextInt();
        N = sc.nextInt();
        O = sc.nextInt();
        D = sc.nextInt();
        e = new int[T + 1];
        c = new int[T + 1];
        h = new int[N + 1];
        a = new int[N + 1];
        for (int i = 1; i <= T; i++) {
            e[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        for (int i = 1; i <= N; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
        }
        dp = new int[D + 1][S + 1];
        for (int i = 1; i <= D; i++) {
            for (int j = 0; j <= S; j++) {
                dp[i][j] = -1;
            }
        }
        System.out.println(dfs(1, S));
        sc.close();
    }

    static int dfs(int day, int power) {
        if (day == D + 1)
            return 0;
        if (dp[day][power] != -1)
            return dp[day][power];
        dp[day][power] = 0;
        int max_happiness = 0;
        boolean canTrain = false;
        for (int i = 1; i <= T; i++) {
            if (power >= e[i]) {
                canTrain = true;
                int next_power = power - e[i];
                if (next_power + O >= S)
                    next_power = S;
                else
                    next_power += O;
                max_happiness = Math.max(max_happiness, dfs(day + 1, next_power) + h[1]);
                for (int j = 2; j <= N; j++) {
                    if (c[i] >= a[j])
                        max_happiness = Math.max(max_happiness, dfs(day + 1, next_power) + h[j]);
                }
            }
        }
        if (!canTrain)
            flag = false;
        dp[day][power] = max_happiness;
        return dp[day][power];
    }
}
