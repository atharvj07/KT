import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt();
        int[][] a = new int[n][m];
        int[] xr = new int[n], xc = new int[m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                a[i][j] = input.nextInt();
                xr[i] = Math.max(xr[i], a[i][j]);
                xc[j] = Math.max(xc[j], a[i][j]);
            }
        }
        int[][] ans = new int[n][m];
        for (int i = 0; i < n; i++) {
            int[] row = new int[m];
            int[] cnt = new int[n * m + 1];
            for (int j = 0; j < m; j++) {
                row[j] = a[i][j];
                cnt[a[i][j]]++;
            }
            for (int j = m - 1; j >= 0; j--) {
                for (int val = xr[i]; val >= 1; val--) {
                    if (cnt[val] > 0) {
                        ans[i][j] = val;
                        cnt[val]--;
                        break;
                    }
                }
            }
        }
        for (int j = 0; j < m; j++) {
            int[] col = new int[n];
            int[] cnt = new int[n * m + 1];
            for (int i = 0; i < n; i++) {
                col[i] = a[i][j];
                cnt[a[i][j]]++;
            }
            for (int i = n - 1; i >= 0; i--) {
                for (int val = xc[j]; val >= 1; val--) {
                    if (cnt[val] > 0) {
                        ans[i][j] = val;
                        cnt[val]--;
                        break;
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(ans[i][j] + " ");
            }
            System.out.println();
        }
    }
}
