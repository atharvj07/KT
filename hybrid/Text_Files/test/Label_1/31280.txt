import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] p = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            d[i] = sc.nextInt();
        }
        boolean ok = true;
        for (int i = 0; i < n; i++) {
            if (p[i] != i + 1) {
                int j = i;
                while (j > 0 && p[j - 1] > p[j]) {
                    if (d[j] != p[j] - p[j - 1]) {
                        ok = false;
                        break;
                    }
                    int tmp = p[j];
                    p[j] = p[j - 1];
                    p[j - 1] = tmp;
                    j--;
                }
                if (!ok) {
                    break;
                }
            }
        }
        System.out.println(ok ? "YES" : "NO");
    }
}
