import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int x = input.nextInt();
        int iceCreams = x;
        int distressedKids = 0;
        while (n-- > 0) {
            String sign = input.next();
            int di = input.nextInt();
            if (sign.equals("+")) {
                iceCreams += di;
            } else if (iceCreams >= di) {
                iceCreams -= di;
            } else {
                ++distressedKids;
            }
        }
        System.out.println(iceCreams + " " + distressedKids);
    }
}
