#include <bits/stdc++.h>
using namespace std;
int a[13845],n,b[13845];
int main(){
  cin>>n;
  for(int i=0;i<n;i++)cin>>a[i];
  int p=0,ne=0,sum=0;
  for(int i=0;i<n;i++){
    if(a[i]>0)p+=a[i];
    if(a[i]<0)ne-=a[i];
    sum+=a[i];
  }
  assert(sum==0);
  int x=(p-ne)/2;
  for(int i=0;i<n;i++){
    if(a[i]>0)b[i]=(a[i]+x+1)/2;
    if(a[i]<0)b[i]=-(abs(a[i])-x+1)/2;
    if(a[i]==0)b[i]=0;
  }
  for(int i=0;i<n;i++)cout<<b[i]<<endl;
  return 0;
}
