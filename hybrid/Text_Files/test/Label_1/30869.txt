import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] speedLimit = new int[n];
    int[] overtakeAllowed = new int[n];
    int speed = 0;
    int overtake = 1;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int t = sc.nextInt();
      if (t == 1) {
        speed = sc.nextInt();
        if (speed > speedLimit[i]) speed = speedLimit[i];
      } else if (t == 2) {
        if (overtake == 0) ans++;
      } else if (t == 3) {
        speedLimit[i] = sc.nextInt();
        if (speed > speedLimit[i]) speed = speedLimit[i];
      } else if (t == 4) {
        overtake = 1;
      } else if (t == 5) {
        speedLimit[i] = 0;
      } else if (t == 6) {
        overtake = 0;
      }
    }
    System.out.println(ans);
  }
}
