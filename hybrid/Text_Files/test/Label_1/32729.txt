import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] pos = new int[m + 1];
        Arrays.fill(pos, -1);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int t = sc.nextInt();
            double x = sc.nextDouble();
            if (pos[t] != -1) {
                if (x < pos[t]) {
                    pos[t] = -2; // cannot satisfy
                } else {
                    ans++;
                }
            }
            pos[t] = (int) x;
        }
        int bad = 0;
        for (int i : pos) {
            bad += i == -2 ? 1 : 0;
        }
        System.out.println(ans + bad);
    }
}
