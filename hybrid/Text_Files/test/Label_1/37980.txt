from collections import Counter

n, m = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
if Counter(a) != Counter(b):
    print(-1)
    exit(0)
l = 0
r = 0
ans = n
count = Counter()
while r < n:
    count[a[r]] += 1
    while all(count[i] >= b[i] for i in range(1, m + 1)):
        ans = min(ans, r - l + 1)
        count[a[l]] -= 1
        l += 1
    r += 1
if ans != n:
    print(ans)
else:
    print(-1)
