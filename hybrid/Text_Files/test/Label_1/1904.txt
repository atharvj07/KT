#include <iostream>
#include <limits>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    long long s, a, b, c;
    cin >> s >> a >> b >> c;
    long long ans = 0;
    while (s >= c * a) {
      s -= c * a;
      ans += a + b;
    }
    ans += s / c;
    cout << ans << endl;
  }
  return 0;
}
