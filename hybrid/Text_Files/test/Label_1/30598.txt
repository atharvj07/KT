import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int m = Integer.parseInt(br.readLine());
        int[] heights = new int[n + 1];
        List<int[]> ropes = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int[] rope = Arrays.stream(br.readLine().split(" "))
                    .mapToInt(Integer::parseInt)
                    .toArray();
            heights[rope[0]] = rope[1];
            ropes.add(rope);
        }
        int q = Integer.parseInt(br.readLine());
        for (int i = 0; i < q; i++) {
            int[] query = Arrays.stream(br.readLine().split(" "))
                    .mapToInt(Integer::parseInt)
                    .toArray();
            System.out.println(solve(heights, ropes, query[0], query[1]));
        }
    }

    private static int solve(int[] heights, List<int[]> ropes, int x, int y) {
        int max = 0;
        for (int[] rope : ropes) {
            if (rope[0] >= x && rope[1] <= y && rope[1] > max) {
                max = rope[1];
            }
        }
        return max;
    }
}
