import java.util.Scanner;

public class VestAssignment {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt(); // Number of soldiers
        int m = scanner.nextInt(); // Number of vests
        int x = scanner.nextInt(); // Size difference tolerance (left)
        int y = scanner.nextInt(); // Size difference tolerance (right)

        int[] desiredSizes = new int[n];
        for (int i = 0; i < n; i++) {
            desiredSizes[i] = scanner.nextInt();
        }

        int[] vestSizes = new int[m];
        for (int i = 0; i < m; i++) {
            vestSizes[i] = scanner.nextInt();
        }

        int k = 0; // Number of soldiers equipped
        int[] soldierIndices = new int[n];
        int[] vestIndices = new int[m];

        int j = 0; // Index for vests
        for (int i = 0; i < n; i++) {
            while (j < m && vestSizes[j] < desiredSizes[i] - x) {
                j++;
            }
            while (j < m && vestSizes[j] > desiredSizes[i] + y) {
                j++;
            }
            if (j < m) {
                soldierIndices[k] = i;
                vestIndices[k] = j;
                k++;
                j++;
            }
        }

        System.out.println(k);

        for (int i = 0; i < k; i++) {
            System.out.println((soldierIndices[i] + 1) + " " + (vestIndices[i] + 1));
        }
    }
}
