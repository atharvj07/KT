import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, r = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(reverse=True)
    cur = 0
    cnt = 0
    for x in a:
        cur += x
        if cur >= r:
            cur -= r
            cnt += 1
    print(cnt + (cur > 0))
