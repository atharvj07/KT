#include <bits/stdc++.h>

using namespace std;
using ll = long long;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    priority_queue<ll> a;
    for (int i = 0; i < n; i++) {
      ll x;
      cin >> x;
      a.push(x);
    }
    ll ans = 0;
    while (k--) {
      ll x = a.top();
      a.pop();
      ll y = a.top();
      a.pop();
      ans += x / y;
    }
    while (!a.empty()) {
      ans += a.top();
      a.pop();
    }
    cout << ans << endl;
  }
}
