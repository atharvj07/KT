import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt(), k = cin.nextInt(), a[] = new int[n], b[] = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = cin.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = cin.nextInt();
        }

        int sum = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] > b[i]) {
                sum += a[i] - b[i];
            }
        }
        if (k < sum) {
            System.out.println(-1);
            return;
        }
        int ans = 0;
        Arrays.sort(a);
        for (int i = 0; i < n; i++) {
            if (a[i] > b[i]) {
                int need = a[i] - b[i];
                if (k >= need) {
                    k -= need;
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
