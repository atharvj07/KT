t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    S = sum(a)
    b = [1] * n  # Initialize beautiful array b with 1s
    diff_sum = sum([abs(a[i] - b[i]) for i in range(n)])  # Calculate the initial sum of differences
    # We try to maximize elements in b while keeping the condition 2*diff_sum <= S
    for i in range(n):
        for j in range(30, 0, -1):  # Check powers of 2 from 2^30 downwards, as 2^30 is close to 10^9
            if 2 * (diff_sum - abs(a[i] - b[i]) + abs(a[i] - (2 ** j))) <= S:
                b[i] = 2 ** j
                diff_sum = diff_sum - abs(a[i] - b[i]) + abs(a[i] - (2 ** j))  # Update diff_sum
                break
    print(*b)
