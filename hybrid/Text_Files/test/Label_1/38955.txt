n, a = map(int, input().split())
x = list(map(int, input().split()))

x.sort()

if n == 1:
    print(0)
else:
    # Calculate the distance if starting from the first checkpoint to the last but one or
    # from the second checkpoint to the last. Choose the minimum.
    min_distance = min(abs(a - x[0]) + abs(x[n - 2] - x[0]), abs(a - x[1]) + abs(x[n - 1] - x[1]))
    for i in range(1, n - 1):
        # Choose between starting from the current checkpoint to the last, or to the first
        # then from the second to the last checkpoint. Update the minimum distance accordingly.
        distance = min(abs(a - x[i]) + abs(x[n - 1] - x[i]), abs(a - x[i + 1]) + abs(x[i] - x[i + 1]))
        min_distance = min(min_distance, distance)
    print(min_distance)
