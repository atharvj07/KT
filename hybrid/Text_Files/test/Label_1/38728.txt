n = int(input())
a = [int(x) for x in input().split()]
assert n == len(a)
a.insert(0, 0)
L = [0] * (n+1)
R = [0] * (n+1)
L[1] = a[1]
R[1] = a[1]
for i in range(2, n+1):
    L[i] = L[i-1] + a[i]
    R[i] = R[i-1] + a[n+1-i]
ans = -1
for x in range(1, n+1):
    tmp = R[x-1] + L[n] - L[x]
    if ans == -1 or tmp < ans:
        ans = tmp
print(ans)
