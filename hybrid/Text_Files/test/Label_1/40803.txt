for _ in range(int(input())):
    a = list(map(int, input().split()))
    if a == sorted(a):
        print(0)
    else:
        ans = 0
        x = 10**9
        for l in range(len(a) - 1):
            if a[l] <= a[l + 1]:
                x = a[l + 1]
            else:
                ans += 1
                x = a[l]
        print(ans)
