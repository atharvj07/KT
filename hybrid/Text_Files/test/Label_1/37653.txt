for _ in range(int(input())):
    n, m, a, b = map(int, input().split())
    adj = [[] for i in range(n + 1)]
    for _ in range(m):
        x, y = map(int, input().split())
        adj[x].append(y)
        adj[y].append(x)
    vis = [False] * (n + 1)
    s = [(a, False)]
    ans = 0
    disa = [-1] * (n + 1)
    disb = [-1] * (n + 1)
    while s:
        x, f = s.pop(0)
        if vis[x]: continue
        vis[x] = True
        disa[x] = disa[f] + 1 if f else 0
        disb[x] = disb[f] + 1 if f else 0
        if disa[x] > 0 and disb[x] > 0: ans += len(adj[x]) - 1
        for y in adj[x]:
            if vis[y]: continue
            s.append((y, f or x == a))
            s.append((y, f or x == b))
    print(ans)
