#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5;
int a[N], dep[N], vis[N], sz[N];
vector<int> g[N];
bool valid() {
    queue<int> q;
    q.push(1);
    vis[1] = 1;
    int i = 0;
    while (!q.empty()) {
        int u = q.front();
        if (u != a[++i]) return 0;
        q.pop();
        for (auto v : g[u]) {
            if (!vis[v]) {
                vis[v] = 1;
                q.push(v);
            }
        }
    }
    return 1;
}
int main() {
    int n;
    cin >> n;
    for (int i = 0;i < n - 1;i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    for (int i = 1;i <= n;i++) {
        cin >> a[i];
    }
    if (valid()) cout << "Yes" << endl;
    else cout << "No" << endl;
    return 0;
}
