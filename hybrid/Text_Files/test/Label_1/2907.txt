#include <iostream>
#include <vector>

int main() {
  int n, w;
  std::cin >> n >> w;
  std::vector<int> a(n);
  for (int i = 0; i < n; i++) std::cin >> a[i];
  int ans = 0;
  for (int i = 0; i <= w; i++) {
    bool flag = true;
    int now = i;
    for (int j = 0; j < n; j++) {
      now += a[j];
      if (now < 0 || now > w) flag = false;
    }
    if (flag) ans++;
  }
  std::cout << ans << std::endl;
  return 0;
}
