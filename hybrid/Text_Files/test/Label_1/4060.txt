#include <iostream>
#include <string>

int main() {
  std::string s;
  std::cin >> s;

  std::string t;
  for (size_t i = 0; i < s.size(); ++i) {
    if (i + 2 < s.size() && s[i] == s[i + 1] && s[i] == s[i + 2]) {
      continue;
    } else if (i + 3 < s.size() && s[i] == s[i + 1] && s[i + 2] == s[i + 3]) {
      continue;
    } else {
      t += s[i];
    }
  }

  std::cout << t << std::endl;
  return 0;
}
