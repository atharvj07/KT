def find_shortest_original_title(n, titles):
    alphabet = set(chr(i) for i in range(97, 123))
    used = set()
    for title in titles:
        for c in title:
            if c in alphabet:
                used.add(c)
    available = alphabet - used
    if available:
        return min(available)
    for length in range(1, 22):
        for i in range(97, 123):
            for j in range(length):
                candidate = chr(i) * j + chr(i + 1) * (length - j)
                if all(candidate not in title for title in titles):
                    return candidate
                if i == 122:
                    break
            if i == 122:
                break

n = int(input().strip())
titles = [input().strip() for _ in range(n)]
print(find_shortest_original_title(n, titles))
