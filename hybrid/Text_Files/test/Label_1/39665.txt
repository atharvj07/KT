def dfs(tree, node, parent, depth, depths, paths):
    depths[node] = depth
    for child, weight in tree[node]:
        if child != parent:
            paths[child] = paths[node] + weight
            dfs(tree, child, node, depth+1, depths, paths)

def solve(k, roads):
    tree = [[] for _ in range(2*k+1)]
    for a, b, t in roads:
        tree[a].append((b, t))
        tree[b].append((a, t))
    
    # Depth and path length from root for each node
    depths = [0] * (2*k + 1)
    paths = [0] * (2*k + 1)
    
    # Run DFS from an arbitrary node (1) to find depths and path lengths
    dfs(tree, 1, -1, 0, depths, paths)
    
    # Find the node that is the furthest from node 1
    furthest_node = max(range(1, 2*k+1), key=lambda x: paths[x])
    
    # Run DFS again from the furthest node to find the true depths and path lengths
    depths = [0] * (2*k + 1)
    paths = [0] * (2*k + 1)
    dfs(tree, furthest_node, -1, 0, depths, paths)
    
    # Diameter of the tree
    B = max(paths)
    
    # Number of nodes that lie on the longest path
    G = sum(1 for x in depths if x % 2 == 0)
    
    return G, B

t = int(input().strip())
for _ in range(t):
    k = int(input().strip())
    roads = [list(map(int, input().strip().split())) for _ in range(2*k - 1)]
    G, B = solve(k, roads)
    print(G, B)
