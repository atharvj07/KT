def solve(d):
    sorted_d = sorted(d)
    t = [''] * len(d)
    pointer = 0
    for i in range(len(d)):
        if d[i] > sorted_d[pointer]:
            t[i] = '2'
        elif d[i] < sorted_d[pointer]:
            t[i] = '1'
        else:  # d[i] == sorted_d[pointer]
            if pointer + 1 < len(d) and sorted_d[pointer] == sorted_d[pointer + 1]:
                t[i] = '1'
            else:
                t[i] = '2'
            pointer += 1
    # Verify if the solution is correct
    ones = [d[i] for i in range(len(d)) if t[i] == '1']
    twos = [d[i] for i in range(len(d)) if t[i] == '2']
    if ones + twos == sorted_d:
        return ''.join(t)
    else:
        return '-'

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    d = list(map(int, list(input().strip())))
    print(solve(d))
