import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int tests = scanner.nextInt();

        for (int t = 0; t < tests; t++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();
            long[] savings = new long[n];
            for (int i = 0; i < n; i++) {
                savings[i] = scanner.nextInt();
            }

            System.out.println(calculateMaxWealthy(savings, x));
        }
        scanner.close();
    }

    private static long calculateMaxWealthy(long[] savings, int wealthThreshold) {
        long totalSavings = 0;
        for (long saving : savings) {
            totalSavings += saving;
        }

        if (totalSavings < wealthThreshold) {
            return 0;
        }

        return totalSavings / wealthThreshold;
    }
}
