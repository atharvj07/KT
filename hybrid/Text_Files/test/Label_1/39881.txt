n=int(input())
k=list(map(int,input().split()))
k.sort()
j=0
p=[]
q=[]
for i in list(map(int,input().split())):
    while j<len(k) and k[j]<i:
        j+=1
    if j<len(k) and k[j]==i:
        p.append(i)
        j+=1
    else:
        q.append(i)
x=0
for i in p:
    x^=i
for i in q:
    x^=i
print(['Koyomi','Karen'][x==0])
