import java.util.*;
class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    boolean[][] g = new boolean[n + 1][n + 1];
    for (int i = 0; i < m; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      g[a][b] = true;
      g[b][a] = true;
    }
    int k = sc.nextInt();
    boolean ok = true;
    for (int i = 0; i < k; i++) {
      int a = sc.nextInt();
      int b = sc.nextInt();
      if (!g[a][b]) {
        ok = false;
        break;
      }
    }
    System.out.println(ok ? "Yes" : "No");
    if (ok) {
      for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
          if (g[i][j]) System.out.println(i + " " + j);
        }
      }
    }
  }
}
