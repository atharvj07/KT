#include <iostream>
#include <unordered_map>
using namespace std;

typedef long long ll;
#define pb push_back
const int MAX = 1e9;

int main() {
  int t;
  cin >> t;
  unordered_map<ll, int> mp;
  while (t--) {
    int n;
    ll sum = 0;
    cin >> n;
    for (int i = 0; i < n; i++) {
      int x;
      cin >> x;
      ll val = 1LL << x;  // 2^x
      sum += val;
      if (sum == 0 || mp[sum]) {
        cout << "YES" << endl;
        return 0;
      }
      mp[sum]++;
    }
    cout << "NO" << endl;
    mp.clear();
  }
  return 0;
}
