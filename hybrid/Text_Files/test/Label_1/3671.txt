#include <bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 1e9 + 7;
vector<int> v;
vector<pair<int, int>> st;
int32_t main() {
    int n;
    cin >> n;
    v.resize(n + 1);
    for (int i = 1; i <= n; i++) { cin >> v[i]; }
    int ans = 0, cur = 0;
    for (int j = 1; j <= n; j++) {
        while (!st.empty() && st.back().second <= v[j]) {
            cur -= st.back().first * st.back().second;
            st.pop_back();
        }
        st.push_back({j, v[j]});
        cur += st.back().first * st.back().second;
        int mult = (j - st.back().first + 1) % mod;
        mult *= ((n - j) % mod);
        mult %= mod;
        ans += cur * mult;
        ans %= mod;
    }
    cout << ans << endl;
    return 0;
}
