g=[[int(x) for x in input().split()] for _ in range(5)]
ans=0
for x in range(1,6):
    for y in range(1,6):
        if g[x-1][y-1]==0:continue
        if x==y:continue
        k=False
        for i in range(1,6):
            if i!=x and g[x-1][i-1]>=g[x-1][y-1]:k=True
        if k==False:
            for i in range(1,6):
                if i!=y and g[y-1][i-1]>=g[x-1][y-1]:k=True
        if k==False:
            ans+=2*g[x-1][y-1]
print(ans)
