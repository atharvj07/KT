import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Queue<int[]> q = new PriorityQueue<>((a, b) -> a[0] == b[0] ? a[1] - b[1] : a[0] - b[0]);
        for (int i = 0; i < n; i++) {
            q.add(new int[]{in.nextInt(), in.nextInt()});
        }
        int ans = 0;
        int[] cur = q.poll();
        while (!q.isEmpty()) {
            int[] next = q.poll();
            if (cur[1] > next[0] || cur[1] + 1 == next[0] && cur[1] + 1 == next[1]) {
                ans++;
            }
            cur = next;
        }
        System.out.println(ans);
    }
}
