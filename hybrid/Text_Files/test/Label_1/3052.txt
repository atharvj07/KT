#include <iostream>

int main() {
    int n, rounds = 0;
    std::cin >> n;
    for (int i = 0, a, b; i < n; i++) {
        std::cin >> a >> b;
        if (a > b) rounds++;
    }
    if (rounds > n / 2) std::cout << "Mishka";
    else if (rounds == n / 2) std::cout << "Friendship is magic!^^";
    else std::cout << "Chris";
    return 0;
}
