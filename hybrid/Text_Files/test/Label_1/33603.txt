import java.io.*;
import java.util.*;

public class BlogNetwork {

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());

        List<Integer>[] graph = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            graph[a].add(b);
            graph[b].add(a);
        }

        st = new StringTokenizer(br.readLine());
        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = Integer.parseInt(st.nextToken());
        }

        int[] order = new int[n];
        boolean[] visited = new boolean[n + 1];
        boolean possible = true;

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                possible = dfs(i, graph, desiredTopics, order, visited, 0);
                if (!possible) {
                    break;
                }
            }
        }

        if (!possible) {
            System.out.println(-1);
        } else {
            for (int i = 0; i < n; i++) {
                System.out.print(order[i] + " ");
            }
        }
    }

    static boolean dfs(int node, List<Integer>[] graph, int[] desiredTopics, int[] order, boolean[] visited, int index) {
        visited[node] = true;
        order[index] = node;

        Set<Integer> availableTopics = new HashSet<>();
        for (int neighbor : graph[node]) {
            if (visited[neighbor]) {
                availableTopics.add(desiredTopics[neighbor]);
            }
        }

        for (int i = 1; i <= desiredTopics.length; i++) {
            if (!availableTopics.contains(i)) {
                if (desiredTopics[node] != i) {
                    return false;
                }
                break;
            }
        }

        for (int neighbor : graph[node]) {
            if (!visited[neighbor]) {
                if (!dfs(neighbor, graph, desiredTopics, order, visited, index + 1)) {
                    return false;
                }
            }
        }

        return true;
    }
}
