import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), h = input.nextInt(), m = input.nextInt(), k = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt() * 60 + input.nextInt();
        }
        Arrays.sort(a);
        int l = 0, r = m / 2, ansL = 0, ansR = 0;
        while (l <= r) {
            int mid = (l + r) / 2, cnt = 0;
            for (int i = 0; i < n; i++) {
                if (a[i] >= mid && a[i] <= mid + k) {
                    cnt++;
                }
            }
            if (cnt <= 1) {
                ansL = mid;
                ansR = cnt;
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        }
        System.out.println(ansR + " " + ansL);
        for (int i = 0; i < n; i++) {
            if (a[i] >= ansL && a[i] <= ansL + k) {
                System.out.println(i+1);
            }
        }
    }
}
