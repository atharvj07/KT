#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long p, k;
    cin >> p >> k;
    if (p % k == 0) {
        cout << "-1" << endl;
        return 0;
    }
    vector<long long> q;
    p %= k * k;
    for (int i = 0; i < k; i++) {
        q.push_back(p % k);
        p /= k;
    }
    long long d = q.size() - 1;
    while (q[d] == 0) d--;
    cout << d + 1 << endl;
    for (int i = 0; i <= d; i++) {
        cout << q[i] << ' ';
    }
}
