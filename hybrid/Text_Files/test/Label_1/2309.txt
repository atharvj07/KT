#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n, m, ans, a, l = 0, r = 0, sum = 0;
  cin >> n >> m;
  vector<int> v(n);
  for (long long i = 0; i < n; i++) {
    cin >> v[i];
    sum += v[i];
  }
  vector<long long> b(m);
  for (long long i = 0; i < m; i++) {
    cin >> b[i];
  }
  for (long long i = 0; i < m; i++) {
    r = upper_bound(v.begin(), v.end(), b[i] - l - 1) - v.begin();
    cout << r + 1 << " " << b[i] - l << endl;
    l += v[r];
  }
}
