#include <bits/stdc++.h>
using namespace std;
const int maxN = 3e5 + 5;
vector<int> adj[maxN], adjr[maxN], order;
int inDeg[maxN], vis[maxN];
bool cycle;
void dfs(int x, int sign) {
    vis[x] = sign;
    for (int u : adj[x]) {
        if (vis[u] == 0) {
            dfs(u, sign);
            if (cycle) return;
        } else if (vis[u] == sign) {
            cycle = true;
            return;
        }
    }
}
void dfsT(int x, int sign) {
    vis[x] = sign;
    order.emplace_back(x);
    for (int u : adjr[x]) {
        if (vis[u] == 0) {
            dfsT(u, sign);
            if (cycle) return;
        }
    }
}
int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; ++i) {
        int u;
        cin >> u;
        if (u) adj[u].emplace_back(i), adjr[i].emplace_back(u);
    }
    for (int i = 1; i <= n; ++i) if (inDeg[i] == 0) dfs(i, 1);
    for (int i = 1; i <= n; ++i) if (vis[i] == 0) dfs(i, -1);
    if (cycle) {
        cout << 0;
        return 0;
    }
    for (int i = 1; i <= n; ++i) vis[i] = 0;
    for (int i = 1; i <= n; ++i) {
        int x, y;
        cin >> x >> y;
        adj[x].emplace_back(y);
        inDeg[y]++;
    }
    for (int i = 1; i <= n; ++i) {
        if (inDeg[i] == 0) dfsT(i, 1);
        if (cycle) {
            cout << 0;
            return 0;
        }
    }
    reverse(order.begin(), order.end());
    for (int u : order) cout << u << ' ';
}
