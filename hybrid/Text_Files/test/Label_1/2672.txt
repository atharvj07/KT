#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, a, ans = 0, f[10] = {};
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a, f[a]++;
  if (n % 2) {
    for (int i = 1; i <= 8; i++) {
      if (f[i] > n / 2) { ans = f[i]; }
    }
  } else {
    for (int i = 1; i <= 8; i++) {
      if (f[i] > n / 2 - 1) { ans = n / 2 - 1; }
    }
  }
  cout << ans << endl;
  return 0;
}
