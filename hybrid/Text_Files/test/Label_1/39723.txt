for t in range(int(input())):
    _ = int(input())
    a = list(map(int, input().split()))
    x = int(input())
    inc, exc = [True] * len(a), [False] * len(a)
    for i in range(1, len(a)):
        if a[i - 1] + a[i] >= x: inc[i], exc[i] = True, True
        exc[i] |= exc[i - 1]
    for i in range(len(a) - 2, -1, -1):
        if a[i] + a[i + 1] >= x: inc[i], exc[i] = True, True
        inc[i] |= inc[i + 1]
    ans = 0
    for i in range(len(a)):
        if inc[i]: ans += 1
        elif exc[i]: ans += 1
        else: ans += a[i] >= x
    print(ans)
