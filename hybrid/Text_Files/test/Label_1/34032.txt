import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), x = sc.nextInt(), dp[] = new int[n + 1];
        long ans = 0;
        int arr[] = new int[n + 1];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        dp[0] = 0;
        for (int i = 1; i <= n; i++) {
            if (dp[i - 1] + arr[i - 1] * Math.max(1, x - i + 1) < dp[i]) {
                dp[i] = dp[i - 1] + arr[i - 1] * Math.max(1, x - i + 1);
            }
            ans = dp[i];
        }
        System.out.println(ans);
    }
}
