#include <bits/stdc++.h>

using namespace std;
const long long mod = 1e9 + 7;
#define f first
#define s second
#define mp make_pair
#define pb push_back

long long n, a, b, t;
string p;
bool f1[500010];
long long ans;

int main() {
    cin >> n >> a >> b >> t;
    cin >> p;
    if (t < a) {
        return cout << "0", 0;
    }
    t -= a;
    f1[0] = false;
    for (long long i = 0; i < n; i++) {
        f1[i + 1] = (p[i] == 'h');
    }
    f1[n + 1] = false;
    for (long long i = 2; i <= n; i++) {
        if (!f1[i] && !f1[i - 1]) {
            ans++;
            t -= b;
            if (t < 0) {
                break;
            }
        } else if (!f1[i] || !f1[i - 1]) {
            ans++;
            t -= a;
            if (t < 0) {
                break;
            }
        }
    }
    cout << ans << endl;
}
