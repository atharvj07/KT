import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), bx = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        int m = sc.nextInt(), by = sc.nextInt();
        int[] y = new int[m];
        for (int i = 0; i < m; i++) {
            y[i] = sc.nextInt();
        }
        long X = 0, Y = 0;
        for (int i = 0; i < n; i++) {
            X *= bx;
            X += x[i];
        }
        for (int i = 0; i < m; i++) {
            Y *= by;
            Y += y[i];
        }
        System.out.println(X < Y ? '<' : X > Y ? '>' : '=');
    }
}
