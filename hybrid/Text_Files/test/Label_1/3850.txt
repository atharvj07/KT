#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

ll dp[300005];

int main() {
  string s;
  cin >> s;
  dp[s.length()] = 0;
  dp[s.length() - 1] = s[s.length() - 1] - '0';
  for (int i = s.length() - 2; i >= 0; i--) {
    int cur = (s[i] - '0') * pow(10, s.length() - i - 1);
    dp[i] = dp[i + 1] + cur;
    dp[i] %= 4;
  }
  ll ans = 0;
  for (int i = 0; i < s.length(); i++) {
    if (dp[i] == 0 || dp[i] == 3) ans += i + 1;
  }
  cout << ans << "\n";
  return 0;
}
