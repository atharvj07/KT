import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {
    private static class Point {
        private int x1, x2, x3, x4, x5;

        public Point(int x1, int x2, int x3, int x4, int x5) {
            this.x1 = x1;
            this.x2 = x2;
            this.x3 = x3;
            this.x4 = x4;
            this.x5 = x5;
        }

        public long dotProduct(Point other) {
            return (long) x1 * other.x1 + (long) x2 * other.x2 + (long) x3 * other.x3 + (long) x4 * other.x4 + (long) x5 * other.x5;
        }

        public double magnitude() {
            return Math.sqrt((long) x1 * x1 + (long) x2 * x2 + (long) x3 * x3 + (long) x4 * x4 + (long) x5 * x5);
        }
    }

    private static int n;
    private static List<Point> points;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        points = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            String[] line = br.readLine().split(" ");
            int x1 = Integer.parseInt(line[0]);
            int x2 = Integer.parseInt(line[1]);
            int x3 = Integer.parseInt(line[2]);
            int x4 = Integer.parseInt(line[3]);
            int x5 = Integer.parseInt(line[4]);
            points.add(new Point(x1, x2, x3, x4, x5));
        }

        int numGood = 0;
        List<Integer> goodIndices = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            Point a = points.get(i);
            boolean isGood = true;
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                Point b = points.get(j);
                for (int k = j + 1; k < n; k++) {
                    if (k == i) continue;
                    Point c = points.get(k);
                    double angle = calculateAngle(a, b, c);
                    if (angle < Math.PI / 2) {
                        isGood = false;
                        break;
                    }
                }
            }
            if (isGood) {
                numGood++;
                goodIndices.add(i + 1);
            }
        }

        System.out.println(numGood);
        for (int index : goodIndices) {
            System.out.println(index);
        }
    }

    private static double calculateAngle(Point a, Point b, Point c) {
        long dotProductAB = a.dotProduct(b);
        long dotProductAC = a.dotProduct(c);
        double magnitudeAB = a.magnitude();
        double magnitudeAC = a.magnitude();
        return Math.acos(dotProductAB / (magnitudeAB * magnitudeAC));
    }
}
