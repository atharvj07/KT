from collections import Counter

def solve():
    n, m = map(int, input().split())
    foods = list(map(int, input().split()))
    favs = [list(map(int, input().split())) for _ in range(m)]

    eaten = Counter()
    for f1, f2 in favs:
        eaten[f1] = 1
        eaten[f2] = 1

    ans = []
    for f in range(1, n+1):
        if eaten[f] == 0:
            ans.append(f)
    for f1, f2 in favs:
        if foods[f1-1] > 0:
            if f1 not in ans:
                ans.append(f1)
            foods[f1-1] -= 1
        elif foods[f2-1] > 0:
            if f2 not in ans:
                ans.append(f2)
            foods[f2-1] -= 1
        else:
            return 'DEAD'

    return 'ALIVE\n' + ' '.join(map(str, ans))

print(solve())
