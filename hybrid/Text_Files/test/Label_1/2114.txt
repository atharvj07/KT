#include<bits/stdc++.h>
using namespace std;
#define ll long long
ll sum,mx;

bool check(ll x){
    ll c=0;
    while(x){
        if(x%2)c++;
        x/=2;
    }
    return c==1;
}

int main()
{
    cin>>sum>>mx;
    vector<ll>ans;

    for(ll i=1; sum&&i<=mx; i++){
        if(sum>i-1&&check(i)){
            ans.push_back(i);
            sum-=i-1;
        }
    }

    if(sum)cout<<-1<<endl;
    else{
        cout<<ans.size()<<endl;
        for(auto x:ans)cout<<x<<' ';
        cout<<endl;
    }

    return 0;
}
