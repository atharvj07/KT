import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String str = scanner.nextLine();
        int l = 0, r = 0, ans = 0, lb = 0;
        for (int i = 0; i < str.length(); i++) {
            if (str.charAt(i) == '(') l++;
            if (str.charAt(i) == '[') lb++;
            if (str.charAt(i) == ')' && l > 0) {
                l--; r++;
                if (r > ans && lb > 0) { ans = r; }
            }
            if (str.charAt(i) == ']' && l > 0 && lb > 0) {
                lb--; l--; r++;
                if (r > ans) { ans = r; }
            }
        }
        System.out.println(ans);
        for (int i = 0; i < ans; i++) { System.out.print('('); }
        for (int i = 0; i < ans; i++) { System.out.print('['); }
        for (int i = 0; i < ans; i++) { System.out.print(']'); System.out.print(')'); }
        System.out.println();
    }
}
