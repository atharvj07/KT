#include <iostream>
#include <string>
using namespace std;

int n;
string s;

int gcd(int a, int b){return b? gcd(b, a%b): a;}

int main(){
    cin>>n>>s;
    int sum=0, cnt0=0;
    for(char c: s) if(c=='0') cnt0++;
    for(char c: s) sum+=c-'0';
    if(sum==0){cout<<"YES"<<endl; return 0;}
    if(cnt0==n){cout<<"NO"<<endl; return 0;}
    if(sum%n!=0 || gcd(sum,n)>1) {cout<<"NO"<<endl; return 0;}
    cout<<"YES"<<endl;
}
