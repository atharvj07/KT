import java.math.BigInteger;
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N = sc.nextInt();
		String A = sc.next();
		BigInteger[][] dp = new BigInteger[N][N + 1];
		for (int i = 0; i < N; i++) {
			for (int j = 0; j < N + 1; j++) {
				dp[i][j] = BigInteger.ZERO;
			}
		}
		dp[0][0] = BigInteger.valueOf(Integer.parseInt(String.valueOf(A.charAt(0))));
		for (int i = 1; i < N; i++) {
			for (int j = 0; j <= i; j++) {
				if (j == 0) {
					dp[i][j] = dp[i - 1][j].subtract(dp[i - 1][j + 1]);
				} else if (j == i) {
					dp[i][j] = dp[i - 1][j - 1].subtract(dp[i - 1][j]);
				} else {
					dp[i][j] = dp[i - 1][j - 1].subtract(dp[i - 1][j]).subtract(dp[i - 1][j]).subtract(dp[i - 1][j + 1]);
				}
			}
		}
		System.out.println(dp[N - 1][0]);
	}
}
