import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            if (n == 0 && m == 0) {
                break;
            }
            Edge[] edges = new Edge[n];
            for (int i = 0; i < n; i++) {
                int a = sc.nextInt();
                int b = sc.nextInt();
                int cost = sc.nextInt();
                int time = sc.nextInt();
                edges[i] = new Edge(a, b, cost, time);
            }
            int k = sc.nextInt();
            for (int i = 0; i < k; i++) {
                int p = sc.nextInt();
                int q = sc.nextInt();
                int r = sc.nextInt();
                int[][] dp = new int[m + 1][2];
                for (int j = 0; j <= m; j++) {
                    Arrays.fill(dp[j], Integer.MAX_VALUE);
                }
                dp[p][r] = 0;
                for (int j = 0; j <= m; j++) {
                    for (Edge edge : edges) {
                        if (dp[edge.a][r] != Integer.MAX_VALUE && dp[edge.b][r] > dp[edge.a][r] + (r == 0 ? edge.cost : edge.time)) {
                            dp[edge.b][r] = dp[edge.a][r] + (r == 0 ? edge.cost : edge.time);
                        }
                        if (dp[edge.b][r] != Integer.MAX_VALUE && dp[edge.a][r] > dp[edge.b][r] + (r == 0 ? edge.cost : edge.time)) {
                            dp[edge.a][r] = dp[edge.b][r] + (r == 0 ? edge.cost : edge.time);
                        }
                    }
                }
                System.out.println(dp[q][r]);
            }
        }
    }

    static class Edge {
        int a;
        int b;
        int cost;
        int time;

        public Edge(int a, int b, int cost, int time) {
            this.a = a;
            this.b = b;
            this.cost = cost;
            this.time = time;
        }
    }
}
