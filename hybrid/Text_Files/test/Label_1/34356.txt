import java.io.*;
import java.util.*;
import java.math.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        boolean[] isPrime = new boolean[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            isPrime[i] = a[i] == 1;
        }

        for (int i = 1; i < n; i++) {
            if (!isPrime[i - 1]) {
                if (isPrime[i]) {
                    isPrime[i] = false;
                    isPrime[i - 1] = true;
                }
            } else {
                isPrime[i] = true;
            }
        }

        List<Integer> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            list.add(a[i]);
        }
        Collections.sort(list);

        for (int i = 0; i < n; i++) {
            System.out.print(list.get(i) + " ");
        }
        System.out.println();
    }
}
