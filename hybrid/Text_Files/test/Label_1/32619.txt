import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), a = sc.nextInt(), b = sc.nextInt(), c = sc.nextInt();
        int[] l = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
        }
        Arrays.sort(l);
        int ans = Integer.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                for (int k = j; k < n; k++) {
                    int la = l[i], lb = l[j], lc = l[k];
                    if (la == a && lb == b && lc == c) {
                        ans = Math.min(ans, 0);
                        continue;
                    }
                    int cost = 0;
                    if (la < a) {
                        cost += (a - la);
                    } else if (la > a) {
                        cost += ((la - a) + 1) / 2;
                    }
                    if (lb < b) {
                        cost += (b - lb);
                    } else if (lb > b) {
                        cost += ((lb - b) + 1) / 2;
                    }
                    if (lc < c) {
                        cost += (c - lc);
                    } else if (lc > c) {
                        cost += ((lc - c) + 1) / 2;
                    }
                    if (la + lb >= a && lb + lc >= b && la + lc >= c) {
                        cost += 10;
                    } else if (la + lb >= a && lb + lc >= c) {
                        cost += 5;
                    } else if (la + lc >= c && lb + lc >= b) {
                        cost += 5;
                    } else if (la + lb >= b && la + lc >= c) {
                        cost += 5;
                    }
                    ans = Math.min(ans, cost);
                }
            }
        }
        System.out.println(ans);
    }
}
