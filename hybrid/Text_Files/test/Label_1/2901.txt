#include <iostream>
#include <vector>
#include <unordered_set>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<unordered_set<int>> h(n + 1), v(n + 1);
  for (int i = 0; i < n * n; i++) {
    int x, y;
    cin >> x >> y;
    if (!h[x].count(y)) h[x].insert(y);
    if (!v[y].count(x)) v[y].insert(x);
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (!h[i].count(j) && !v[j].count(i)) {
        cout << i << ' ';
        break;
      }
    }
  }
}
