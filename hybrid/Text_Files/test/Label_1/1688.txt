#include <iostream>
#include <vector>
 
using namespace std;
 
const int MAXN = 2e5 + 5;
const int MAXLOG = 20;
 
struct path
{
    int vert, len;
};
 
vector<int> adj[MAXN], tmp[MAXN];
vector<path> paths[MAXN];
bool mark[MAXN];
int dep[MAXN], par[MAXN][MAXLOG], tin[MAXN], tout[MAXN], timer = 0;
 
void dfs(int v, int p, int d)
{
    dep[v] = d;
    par[v][0] = p;
    tin[v] = ++timer;
    for (int u : adj[v])
        if (u != p)
            dfs(u, v, d + 1);
    tout[v] = ++timer;
}
 
bool ancestor(int x, int y)
{
    return tin[x] <= tin[y] && tout[x] >= tout[y];
}
 
int lca(int x, int y)
{
    if (ancestor(x, y))
        return x;
    if (ancestor(y, x))
        return y;
    for (int j = MAXLOG - 1; j >= 0; j--)
        if (!ancestor(par[x][j], y))
            x = par[x][j];
    return par[x][0];
}
 
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    int n, m;
    cin >> n >> m;
    for (int i = 1; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    dfs(1, 1, 0);
    for (int j = 1; j < MAXLOG; j++)
        for (int i = 1; i <= n; i++)
            par[i][j] = par[par[i][j - 1]][j - 1];
    for (int q = 0; q < m; q++)
    {
        int k;
        cin >> k;
        for (int i = 0; i < k; i++)
        {
            int v;
            cin >> v;
            mark[v] = true;
        }
        bool ok = false;
        for (int i = 0; i < k; i++)
        {
            int v;
            cin >> v;
            tmp[dep[v]].push_back(v);
            paths[dep[v]].push_back({v, tmp[dep[v]].size() - 1});
        }
        for (int i = n; i >= 0; i--)
        {
            if (tmp[i].size() > 0)
            {
                for (int v : tmp[i])
                    mark[v] = false;
                for (int j = 0; j < tmp[i].size(); j++)
                {
                    int cur = tmp[i][j];
                    int p = paths[i][j].vert;
                    int s = paths[i][j].len;
                    int h;
                    if (j == 0)
                        h = dep[p];
                    else
                        h = dep[lca(p, tmp[i][j - 1])];
                    if (h == i && mark[p])
                        ok = true;
                    if (h < i)
                        ok = true;
                    paths[h].push_back({cur, s});
                }
                tmp[i].clear();
            }
        }
        if (ok)
            cout << "YES" << '\n';
        else
            cout << "NO" << '\n';
    }
}
