import sys

n, m, k, s = map(int, input().split())
dol_rate = list(map(int, input().split()))
gbp_rate = list(map(int, input().split()))
gadgets = []
for i in range(m):
    t, c = map(int, input().split())
    if t == 1:
        gadgets.append([c, dol_rate])
    else:
        gadgets.append([c, gbp_rate])
gadgets.sort(key=lambda x: x[0])
total, ans = 0, []
for i in range(m):
    c, rate = gadgets[i]
    day = (s - total + c - 1) // c
    if day > n: continue
    total += c * day
    s -= rate[day - 1] * day
    ans.append([i + 1, day])
    if total >= k * c:
        break
if total < k * c:
    print(-1)
else:
    print(len(ans))
    [print(*x) for x in ans]
