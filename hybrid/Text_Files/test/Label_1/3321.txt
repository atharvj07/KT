#include <algorithm>
#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }
  double ans = 0;
  for (int i = 0; i + k - 1 < n; ++i) {
    double sum = 0;
    for (int j = i; j < i + k; ++j) { sum += a[j]; }
    ans = std::max(ans, sum / k);
  }
  std::cout.precision(12);
  std::cout << ans << std::endl;
  return 0;
}
