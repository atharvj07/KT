#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 1;

int n, h, a[N], b[N], ans;
bool vis[N];

int main() {
  cin >> n >> h;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  sort(a + 1, a + n + 1);
  if (a[1] > h) {
    ans = 0;
    for (int i = 1; i <= n; i++) { b[i] = 1; }
  } else {
    ans = h;
    int j = n;
    for (int i = 1; i < j; i++) {
      if (a[i] + a[j] <= h) {
        b[i] = 1, b[j] = 2;
        j--;
      } else {
        b[i] = 1;
      }
    }
    if (j == 1) { ans = 0; }
  }
  cout << ans << '\n';
  for (int i = 1; i <= n; i++) { cout << b[i] << " "; }
  return 0;
}
