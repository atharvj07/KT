from sys import stdin

for _ in range(int(stdin.readline())):
    n = int(stdin.readline())
    a = [int(x) for x in stdin.readline().strip()]
    pref = [0] * n
    pref[0] = 1 if a[0] == 0 else 0
    for i in range(1, n):
        pref[i] = pref[i - 1]
        if a[i] == 0:
            pref[i] += 1
    cnt = 0
    for i in range(n):
        cnt += pref[i]
        for j in range(i + 1, n):
            if pref[j] - pref[i] == j - i + 1:
                cnt += 1
    print(cnt)
