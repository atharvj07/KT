import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class Main {

    private static boolean isPrime(int num) {
        if (num < 2) {
            return false;
        }
        for (int i = 2; i * i <= num; i++) {
            if (num % i == 0) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        int n;
        try (Scanner scanner = new Scanner(System.in)) {
            n = scanner.nextInt();
            int[] nums = new int[n];
            for (int i = 0; i < n; i++) {
                nums[i] = scanner.nextInt();
            }
            int[] minPrimeFactors = new int[n];
            Map<Integer, List<Integer>> samePrimeFactors = new HashMap<>();
            for (int i = 0; i < n; i++) {
                int primeFactor = 2;
                while (nums[i] > 1) {
                    if (nums[i] % primeFactor == 0) {
                        nums[i] /= primeFactor;
                        minPrimeFactors[i] = primeFactor;
                        if (!samePrimeFactors.containsKey(primeFactor)) {
                            samePrimeFactors.put(primeFactor, new ArrayList<>());
                        }
                        samePrimeFactors.get(primeFactor).add(i);
                    } else {
                        primeFactor++;
                    }
                }
            }
            int minPrime = Arrays.stream(minPrimeFactors).min().getAsInt();
            List<Integer> minPrimeIndices = samePrimeFactors.get(minPrime);
            for (int left = 0, right = minPrimeIndices.size() - 1; left < right; left++, right--) {
                int temp = nums[minPrimeIndices.get(left)];
                nums[minPrimeIndices.get(left)] = nums[minPrimeIndices.get(right)];
                nums[minPrimeIndices.get(right)] = temp;
            }
            for (int i = 1; i < n; i++) {
                if (nums[i] < nums[i - 1]) {
                    System.out.println(0);
                    return;
                }
            }
            System.out.println(1);
        }
    }
}
