import java.util.Scanner;

public class EqualNumbers {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int max = 0;
        for (int x : a) {
            max = Math.max(max, x);
        }

        int operations = 0;
        while (true) {
            int count = 0;
            for (int x : a) {
                if (x == max) {
                    count++;
                }
            }

            if (count >= k) {
                break;
            }

            operations++;
            for (int i = 0; i < n; i++) {
                if (a[i] < max) {
                    a[i] = (a[i] + 1) / 2;
                }
            }
        }

        System.out.println(operations);
    }
}
