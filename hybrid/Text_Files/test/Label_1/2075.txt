#include <bits/stdc++.h>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<int, int>> v(n);
  for (int i = 0; i < n; ++i) {
    int x;
    cin >> x;
    v[i] = {x, i};
  }
  vector<int> b(n);
  for (int i = 0; i < n; ++i) {
    cin >> b[i];
    b[i]--;
  }
  sort(v.rbegin(), v.rend());
  vector<int> order;
  int ans = 0;
  for (auto [x, i] : v) {
    order.push_back(i + 1);
    ans += x;
    if (b[i] != -1) { v[b[i]].first += x; }
  }
  cout << ans << endl;
  for (auto x : order) cout << x << ' ';
  cout << endl;
  return 0;
}
