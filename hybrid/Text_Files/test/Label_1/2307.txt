#include<bits/stdc++.h>
using namespace std;
int main(){
 int n;
 cin>>n;
 vector<int> v(n);
 map<int, int> mp;
 for(int i=0;i<n;i++){
  cin>>v[i];
  mp[v[i]]++;
 }
 sort(v.begin(), v.end());
 int ans=-1;
 for(int i=0;i<n;i++){
  int cnt=0, sum=0;
  for(int j=i;j<n;j++){
   if((mp[v[j]]+cnt)<=2){
    cnt+=mp[v[j]];
    sum++;
   }else{
    ans=v[j-1];
    break;
   }
  }
 }
 cout<<ans<<"\n";
 return 0;
}
