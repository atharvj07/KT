from sys import stdin
s, m = (int(x) for x in stdin.readline().split())
a = list(stdin.readline().strip())
for x, c in (stdin.readline().split() for _ in range(m)):
    a[int(x) - 1] = c
    i = a.index('.') if '.' in a else len(a)
    c = a[i:]
    a[i:] = [c[i + j - 1] for j in range(1, len(c)) if j % 2 == 0]
    print(i // 2 + (i % 2 and c[i] != '.'))
