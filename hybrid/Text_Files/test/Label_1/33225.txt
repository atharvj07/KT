import java.util.Scanner;

public class Main {
    private static final int MOD = 998244353;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long k = sc.nextLong();
        long res = power(n - k + 1, k, MOD);
        res = (res - power(n - k, k, MOD) + MOD) % MOD;
        System.out.println(res);
    }

    private static long power(long base, long exp, long mod) {
        if (exp == 0)
            return 1;
        if (exp == 1)
            return base;
        long val = power(base, exp / 2, mod);
        val = (val * val) % mod;
        if (exp % 2 == 1)
            val = (val * base) % mod;
        return val;
    }
}
