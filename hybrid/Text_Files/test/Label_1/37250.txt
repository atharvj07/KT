n = int(input())
g = [[] for i in range(n + 1)]
for i in range(n - 1):
    x, y = map(int, input().split())
    g[x].append(y)
    g[y].append(x)
q = [1]
v = [False for i in range(n + 1)]
ans = 0
while len(q) > 0:
    x = q.pop(0)
    if v[x] == True:
        continue
    v[x] = True
    deg = 0
    for y in g[x]:
        if v[y] == False:
            deg += 1
    if deg == 1:
        ans += 1
    for y in g[x]:
        if v[y] == False:
            q.append(y)
print(ans)
