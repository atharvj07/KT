#include <iostream>
#include <set>
#include <vector>
using namespace std;
typedef long long ll;
#define int ll
const int INF = 1e15;

const int maxn = 2e5 + 5;
vector<int> adj[maxn];
vector<int> val(maxn);
vector<int> up(maxn, -1);
vector<int> gcdSub(maxn);

int gcd(int a, int b) {
  if (a == 0)
    return b;
  else
    return gcd(b % a, a);
}

void dfs(int u, int p) {
  int ans = val[u];
  for (int v : adj[u]) {
    if (v == p)
      continue;
    dfs(v, u);
    ans = gcd(ans, gcdSub[v]);
  }
  gcdSub[u] = ans;
}

void dfs2(int u, int p, int ans) {
  up[u] = ans;
  for (int v : adj[u]) {
    if (v == p)
      continue;
    if (ans == -1)
      dfs2(v, u, gcd(val[u], gcdSub[v]));
    else
      dfs2(v, u, gcd(ans, gcdSub[v]));
  }
}

int32_t main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> val[i - 1];
  for (int i = 0; i < n - 1; i++) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    adj[a].push_back(b);
    adj[b].push_back(a);
  }
  dfs(0, -1);
  dfs2(0, -1, gcd(val[0], gcdSub[0]));
  for (int i = 0; i < n; i++) cout << up[i] << " ";
  cout << endl;
  return 0;
}
