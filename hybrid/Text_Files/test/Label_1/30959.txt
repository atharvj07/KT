import java.util.Arrays;
import java.util.Scanner;

public class BeautifulArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] coupons = new int[m];
        int[] cost = new int[m];
        for (int i = 0; i < m; i++) {
            coupons[i] = scanner.nextInt();
            cost[i] = scanner.nextInt();
        }

        Arrays.sort(coupons); // Sort coupons to optimize search

        int maxPayment = 0;
        int[] lastIndexes = new int[m]; // Track last index for each coupon
        for (int i = 0; i < n; i++) {
            int left = 0, right = m - 1;
            int maxCouponIndex = -1; // Find the largest coupon no greater than current number
            while (left <= right) {
                int mid = left + (right - left) / 2;
                if (coupons[mid] <= i) {
                    maxCouponIndex = mid;
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }

            if (maxCouponIndex != -1) { // Use current number if coupon found
                maxPayment += cost[maxCouponIndex];
                lastIndexes[maxCouponIndex] = i; // Update last index
            } else { // Check previous coupons for potential pairs
                for (int j = maxCouponIndex - 1; j >= 0; j--) {
                    if (lastIndexes[j] == i - 1) { // Pair found: use current number with previous coupon
                        maxPayment += cost[j];
                        break;
                    }
                }
            }
        }

        System.out.println(maxPayment);
    }
}
