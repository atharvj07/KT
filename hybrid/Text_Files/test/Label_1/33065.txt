import java.util.*;
import java.io.*;

public class Main {
    static int n, m, roads[], par[], count[];

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        roads = new int[n + 1];
        par = new int[n + 1];
        count = new int[n + 1];
        Arrays.fill(par, -1);
        for (int i = 1; i < n; i++) {
            int u = sc.nextInt(), v = sc.nextInt(), typ = sc.nextInt();
            if (typ == 1) {
                if (findPar(v) == findPar(u)) {
                    roads[i] = 1;
                } else {
                    par[findPar(u)] = findPar(v);
                    roads[i] = 0;
                }
            }
        }
        m = sc.nextInt();
        int id[] = new int[m];
        for (int i = 0; i < m; i++) id[i] = sc.nextInt();
        for (int i = 0; i < m; i++) {
            int idx = id[i];
            if (roads[idx] == 0) {
                count[findPar(roads[idx])] = 0;
            }
            roads[idx] ^= 1;
            int p1 = findPar(roads[idx]);
            if (roads[idx] == 0) count[p1]++;
            System.out.println(count[p1] * 2);
        }
    }

    static int findPar(int node) {
        if (par[node] == -1) return node;
        return par[node] = findPar(par[node]);
    }
}
