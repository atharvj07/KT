#include <bits/stdc++.h>

using namespace std;

const int N = 3e5 + 5;

vector<pair<long long, int>> adj[N];
vector<long long> dis(N, LLONG_MAX);
vector<int> parent(N);
vector<pair<long long, long long>> ans(N);
long long best = LLONG_MIN;

void dfs(int now, long long gas) {
  ans[now] = {gas, 0};
  for (auto [w, v] : adj[now]) {
    if (v == parent[now]) continue;
    parent[v] = now, dfs(v, min(gas + w, LLONG_MAX));
  }
}

void dfs2(int now, long long gas) {
  ans[now].second = gas;
  best = max(best, gas);
  for (auto [w, v] : adj[now]) {
    if (v == parent[now]) continue;
    dfs2(v, min(gas + w, dis[v]));
  }
}

int main() {
  int n;
  cin >> n;
  vector<long long> w(n);
  for (long long &x : w) cin >> x;
  for (int i = 1; i < n; ++i) {
    int u, v, c;
    cin >> u >> v >> c;
    adj[u].push_back({c, v});
    adj[v].push_back({c, u});
  }
  dis[1] = w[1];
  dfs(1, w[1]);
  long long prev = dis[1];
  for (int i = 1; i <= n; ++i) dis[i] = LLONG_MAX;
  dis[1] = w[1] + ans[1].second;
  dfs2(1, w[1] + ans[1].second);
  cout << best;
  return 0;
}
