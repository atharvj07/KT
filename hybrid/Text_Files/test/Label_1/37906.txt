from math import gcd

def steps_to_equal_gcd(n, a):
    def array_to_gcd(a):
        return [gcd(a[i], a[(i + 1) % n]) for i in range(n)]
        
    def all_equal(a):
        return all(x == a[0] for x in a)
    
    if all_equal(a):
        return 0
    
    step = 0
    while not all_equal(a):
        a = array_to_gcd(a)
        step += 1
        if step > 1:  # Since after the first step, if not all elements are equal, then it will never become equal
            return -1
    return step

t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    print(steps_to_equal_gcd(n, a))
