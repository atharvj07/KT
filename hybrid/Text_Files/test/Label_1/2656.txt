#include <bits/stdc++.h>

using namespace std;

void solve() {
    int N, X;
    cin >> N >> X;
    vector<vector<int>> adj(N + 1);
    for (int u, v, i = 1; i < N; i++) {
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int Alice = 0;
    int Bob = 0;

    function<void(int, int)> dfs = [&](int u, int p) {
        int mx = 0;
        for (int v : adj[u]) {
            if (v != p) {
                dfs(v, u);
                mx = max({mx, Bob});
                Bob = max({Alice, Bob + 1});
            }
        }

        Alice = mx + 1;
    };

    dfs(X, -1);
    cout << Bob;
}

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    solve();
    return 0;
}
