import java.util.*;
import java.lang.*;
import java.io.*;

class Main {
    public static void main(String[] args) throws java.lang.Exception {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = input.nextInt();
        }
        int ans = 0;
        for (int i = 1; i < n; ++i) {
            ans = Math.max(ans, a[i] - a[i - 1]);
        }
        int last = ans;
        for (int i = 1; i < n - 1; ++i) {
            ans = Math.min(ans, Math.max(a[i + 1] - a[i - 1], Math.max(a[i + 1] - a[i], a[i] - a[i - 1])));
        }
        last = Math.min(last, ans);
        System.out.println(last);
    }
}
