import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int left = 0, right = 0;
        for (int i = 0; i < n; i++) {
            left += scanner.nextInt();
            right += scanner.nextInt();
        }
        System.out.println(Math.max(left, right));
    }
}
