#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<pair<char, int>> v;
  for (int i = 0; i < n; i++) {
    char c;
    int d;
    cin >> c >> d;
    v.push_back(make_pair(c, d));
  }
  sort(v.begin(), v.end());
  int c = 0, d = 0;
  for (int i = 0; i < n; i++) {
    if (v[i].first == v[i + 1].first) c++;
  }
  for (int i = 0; i < n; i++) {
    if (v[i].second == v[i + 1].second) d++;
  }
  cout << max(c, d) << endl;
  return 0;
}
