import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        List<long[]> edges = new ArrayList<>();
        int[] deg = new int[n + 1];
        for (int i = 0; i < m; i++) {
            int u = in.nextInt(), v = in.nextInt(), w = in.nextInt();
            edges.add(new long[]{u, v, w});
            deg[v]++;
        }
        Queue<Integer> q = new LinkedList<>(), qq = new LinkedList<>();
        q.add(in.nextInt());
        long[] dp = new long[n + 1];
        IntStream.range(1, n + 1).forEach(i -> dp[i] = Long.MAX_VALUE);
        while (!q.isEmpty()) {
            int u = q.poll();
            for (long[] e : edges) {
                if (e[1] == u) {
                    int v = (int) e[0];
                    long w = e[2];
                    if (dp[u] + w < dp[v]) {
                        dp[v] = dp[u] + w;
                        if (--deg[v] == 0) qq.add(v);
                    }
                }
            }
            if (q.isEmpty()) q.addAll(qq);
            qq.clear();
        }
        long ans = Arrays.stream(dp).max().orElse(0);
        System.out.println(ans == Long.MAX_VALUE ? 0 : ans);
    }
}
