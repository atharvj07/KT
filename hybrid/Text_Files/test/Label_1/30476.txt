import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long n = in.nextLong();
        long a = in.nextLong();
        long b = in.nextLong();
        long c = in.nextLong();
        long d = in.nextLong();
        long ans = 0L;
        if (d > 0) {
            ans += (d * (n * (n + 1) / 2)) % (1L << 32);
        }
        long a2 = a * 2;
        long a3 = a * 3;
        if (c > 0) {
            ans += (c * (a * n * (n + 1) * (2 * n + 1) / 6 + a2 * n * (n + 1) / 4 + a3 * (n + 1) / 4)) % (1L << 32);
        }
        long b2 = b * 2;
        long b3 = b * 3;
        if (b > 0) {
            ans += (b * (a2 * n * (n + 1) / 4 + a3 * (n + 1) / 2 + b * n * (n + 1) * (2 * n + 1) / 6 + b2 * n * (n + 1) / 4 + b3 * (n + 1) / 4)) % (1L << 32);
        }
        if (a > 0) {
            ans += (a * (a3 * (n + 1) / 4 + b3 * (n + 1) / 2 + c * n * (n + 1) * (2 * n + 1) / 6 + d * n * (n + 1) / 4)) % (1L << 32);
        }
        System.out.println(ans % (1L << 32));
    }
}
