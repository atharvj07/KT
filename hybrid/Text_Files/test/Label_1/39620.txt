mod=998244353
a=int(input());l=int(input());r=int(input());sb=0
for i in str(a):
    sb+=i=='0'
a=str(a);ans=1
for i in range(len(a)):
    if int(a[i])>r:
        ans=0;break
    if i+1<len(a) and int(a[i])==0 and int(a[i+1])!=0:
        ans=0;break
    if int(a[i])==0:
        ans=ans*sb%mod
    if int(a[i])>=l:
        ans=ans*10%mod
ans=ans%mod
print(ans)
