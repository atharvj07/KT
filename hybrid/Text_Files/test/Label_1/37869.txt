n, k = map(int, input().split())
grid = [input() for _ in range(n)]

row_black = [0] * n
col_black = [0] * n

for i in range(n):
    for j in range(n):
        if grid[i][j] == 'B':
            row_black[i] += 1
            col_black[j] += 1

initial_white_rows = sum(1 for row in row_black if row == 0)
initial_white_cols = sum(1 for col in col_black if col == 0)

max_white_lines = initial_white_rows + initial_white_cols

for i in range(n - k + 1):
    for j in range(n - k + 1):
        erased_row_black = row_black[:]
        erased_col_black = col_black[:]

        for x in range(k):
            for y in range(k):
                if grid[i + x][j + y] == 'B':
                    erased_row_black[i + x] -= 1
                    erased_col_black[j + y] -= 1

        current_white_rows = sum(1 for row in erased_row_black if row == 0)
        current_white_cols = sum(1 for col in erased_col_black if col == 0)
        max_white_lines = max(max_white_lines, current_white_rows + current_white_cols)

print(max_white_lines)
