t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    total = sum(a)
    max_segment = current_segment = a[0]
    for i in range(1, n):
        current_segment = max(a[i], current_segment + a[i])
        max_segment = max(max_segment, current_segment)
    if max_segment >= total:
        print("NO")
        continue
    max_segment = current_segment = a[-1]
    for i in range(n-2, -1, -1):
        current_segment = max(a[i], current_segment + a[i])
        max_segment = max(max_segment, current_segment)
    if max_segment >= total:
        print("NO")
    else:
        print("YES")
