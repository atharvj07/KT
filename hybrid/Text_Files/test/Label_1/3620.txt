#include <bits/stdc++.h>
using namespace std;
#undef int
#define int long long
using ll = int;

const int mod = 998244353;

int add(int a, int b) { return (a + b) % mod; }
int modpow(int base, int p) {
  if (p == 0) return 1;
  if (p == 1) return base;
  int half = modpow(base, p / 2);
  half = add(half, half);
  if (p % 2 == 1) return add(half, modpow(base, 1));
  return half;
}

int n, ans = 1;
char domino[100005][2];

int32_t main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> domino[i];
  for (int i = 0; i < n; i++) {
    if (domino[i][0] == domino[i][1] and domino[i][0] != '?') {
      cout << 0 << endl;
      return 0;
    }
  }
  for (int i = 0; i < n; i++) {
    if (domino[i][1] == '?') continue;
    if (i == 0 or domino[i - 1][0] != domino[i][1]) continue;
    ans = ans * 2;
    ans %= mod;
  }
  for (int i = 0; i < n; i++) {
    if (domino[i][0] == '?') continue;
    if (i == 0 or domino[i - 1][1] != domino[i][0]) continue;
    ans = ans * 2;
    ans %= mod;
  }
  cout << ans << endl;
}
