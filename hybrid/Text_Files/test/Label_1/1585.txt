#include<bits/stdc++.h>
#define ll long long
using namespace std;
int main()
{
    string s;
    cin>>s;
    int c1=0,c2=0,ans;
    for(int i=0;i<s.size();i++)
    {
        if(s[i]=='(') c1++;
        else if(s[i]==')') c2++;
    }
    if(c1!=c2) {cout<<"-1"<<endl; return 0;}
    stack<int> st;
    for(int i=0;i<s.size();i++)
    {
        if(s[i]=='(') st.push(i);
        else if(s[i]==')')
        {
            if(st.empty()) {cout<<"-1"<<endl; return 0;}
            st.pop();
        }
        else
        {
            ans=i;
            int x=0;
            while(!st.empty())
            {
                x++;
                st.pop();
            }
            cout<<x<<endl;
        }
    }
    return 0;
}
