import java.util.*;

class Main {
    static int n, m, k;
    static char[][] a;
    static int[][] d;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        k = in.nextInt();
        a = new char[n][];
        d = new int[n][m];
        for (int i = 0; i < n; i++) {
            a[i] = in.next().toCharArray();
        }
        for (int i = 0; i < n; i++) {
            Arrays.fill(d[i], -1);
        }
        Queue<State> q = new LinkedList<>();
        q.add(new State(0, 0, ""));
        while (!q.isEmpty()) {
            State v = q.poll();
            if (v.x == n - 1 && v.y == m - 1) {
                System.out.println(v.path.substring(1, v.path.length() - 1));
                return;
            }
            for (int i = -1; i <= 1; i++) {
                for (int j = -1; j <= 1; j++) {
                    if (i == 0 && j == 0) continue;
                    int x = v.x + i, y = v.y + j;
                    if (x >= 0 && x < n && y >= 0 && y < m && d[x][y] == -1 && (a[x][y] == '.' || a[x][y] == v.last)) {
                        char type = a[x][y];
                        if (type != '.' && type != v.last) {
                            if (v.typeCount >= k) continue;
                            type = 'a';
                            for (char t = 'a'; t <= 'z'; t++) {
                                if (t != v.last && v.path.indexOf(t) == -1) {
                                    type = t;
                                    break;
                                }
                            }
                        }
                        d[x][y] = d[v.x][v.y] + 1;
                        q.add(new State(x, y, v.path + type));
                    }
                }
            }
        }
        System.out.println(-1);
    }

    static class State {
        int x, y;
        String path;
        int typeCount;
        char last;

        State(int x, int y, String path) {
            this.x = x;
            this.y = y;
            this.path = path;
            last = path.length() > 0 ? path.charAt(path.length() - 1) : '.';
            if (last != '.' && last != 'S' && last != 'T') {
                typeCount = path.replaceAll("[^" + last + "]", "").length();
            } else {
                typeCount = 0;
            }
        }
    }
}
