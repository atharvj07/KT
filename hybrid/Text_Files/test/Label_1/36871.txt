from collections import defaultdict

n = int(input())
poles = []
for _ in range(n):
    x, y = map(int, input().split())
    poles.append((x, y))

lines = defaultdict(list)
for i in range(n):
    x1, y1 = poles[i]
    for j in range(i+1, n):
        x2, y2 = poles[j]
        slope = (y2-y1)/(x2-x1) if x2!=x1 else float('inf')
        lines[slope].append((i, j))

count = 0
for slope in lines:
    pairs = lines[slope]
    count += len(pairs)*(len(pairs)-1)//2

print(count)
