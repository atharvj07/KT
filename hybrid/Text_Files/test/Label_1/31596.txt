import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] s = sc.next().toCharArray();
        int n = s.length;

        int k = 0;
        for (int i = 0; i < n; ++i) {
            if (s[i] == 'S' && (i == 0 || s[i - 1] != 'S')) { ++k; }
        }

        System.out.println(n - 2 * k);
    }
}
