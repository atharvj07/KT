import java.util.*;

public class Main {
  static long[] a, b, dp;
  static int n;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt();
    a = new long[n + 1];
    b = new long[n + 1];
    dp = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      a[i] = sc.nextLong();
    }
    for (int i = 1; i <= n; i++) {
      b[i] = sc.nextLong();
    }
    dp[1] = a[1];
    for (int i = 2; i <= n; i++) {
      dp[i] = dp[i - 1] + a[i];
    }
    long ans = dp[n];
    long max = 0;
    for (int i = n - 1; i >= 1; i--) {
      max = Math.max(max, dp[i]);
      ans += b[i] + max;
    }
    System.out.println(ans);
  }
}
