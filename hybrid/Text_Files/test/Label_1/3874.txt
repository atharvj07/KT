#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod=1e9+7;
const int mx=2e5+25;
#define ff first
#define ss second
#define pb push_back
#define pf push_front
#define mp make_pair
#define all(v) (v.begin(),v.end())
ll pwmd(ll b, ll p){ if(!p)return 1;ll x=pwmd(b,p/2);x=(x*x)%mod;if(p%2){ x=(x*b)%mod; }return x;}
ll gcd(ll a,ll b){ if(!b)return a;return gcd(b,a%b);}
int n,k;
int a[mx];
struct Node{
    int idx,data;
};
vector<Node>v;
bool cmp(Node& a,Node& b){
    return a.data>b.data;
}
int main(){
    cin>>n>>k;
    for(int i=1;i<=n;++i){
        cin>>a[i];
        v.push_back({i,a[i]});
    }
    sort(v.begin(),v.end(),cmp);
    int mn=0,mx=0;
    for(int i=0;i<n;++i)mn+=v[i].data;
    for(int i=n-1;i>=0;--i)mx+=v[i].data;
    int dif=mx-mn;
    vector<pair<int,int>>ans;
    if(dif>k){
        cout<<dif-k<<" "<<k<<endl;
        for(int i=0;i<k;++i){
            ans.pb({v[i].idx,v[i+1].idx});
            v[i+1].data+=v[i].data;
        }
    }
    else{
        cout<<"0 "<<dif<<endl;
        for(int i=0;i<dif;++i){
            ans.pb({v[i].idx,v[i+1].idx});
            v[i+1].data+=v[i].data;
        }
    }
    for(auto x:ans){
        cout<<x.ff<<" "<<x.ss<<endl;
    }
    return 0;
}
