import java.util.*;
import java.io.*;

class Main {
    static class Edge {
        int to, w;

        public Edge(int to, int w) {
            this.to = to;
            this.w = w;
        }
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] c1 = new int[n], c2 = new int[n], c3 = new int[n];
        for (int i = 0; i < n; i++) {
            c1[i] = sc.nextInt();
            c2[i] = sc.nextInt();
            c3[i] = sc.nextInt();
        }
        List<Edge>[] adj = new List[n];
        for (int i = 0; i < n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt() - 1, v = sc.nextInt() - 1;
            adj[u].add(new Edge(v, 1));
            adj[v].add(new Edge(u, 1));
        }
        int[] color = new int[n];
        if (!dfs(0, -1, 1, color, adj, c1, c2, c3))
            System.out.println(-1);
        else {
            int cost = 0;
            for (int i = 0; i < n; i++)
                cost += (color[i] == 1 ? c1[i] : color[i] == 2 ? c2[i] : c3[i]);
            System.out.println(cost);
            for (int i = 0; i < n; i++)
                System.out.print(color[i] + " ");
        }
        sc.close();
    }

    static boolean dfs(int u, int p, int col, int[] color, List<Edge>[] adj, int[] c1, int[] c2, int[] c3) {
        color[u] = col;
        for (Edge e : adj[u]) {
            int v = e.to;
            if (v == p)
                continue;
            if (color[v] == col)
                return false;
            if (color[v] != 0) {
                if (col == color[v])
                    return false;
            } else {
                if (!dfs(v, u, col == 1 ? 2 : col == 2 ? 3 : 1, color, adj, c1, c2, c3))
                    return false;
            }
        }
        return true;
    }
}
