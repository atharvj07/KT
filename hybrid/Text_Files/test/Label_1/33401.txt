import java.util.*;

public class Main {
    static long[] tree;
    static long[] sums;
    static int[] parent;
    static int numNodes;
    static long sum;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            numNodes = sc.nextInt();
            int numQueries = sc.nextInt();
            if (numNodes == 0 && numQueries == 0) {
                break;
            }
            tree = new long[numNodes + 1];
            sums = new long[numNodes + 1];
            parent = new int[numNodes + 1];
            sum = 0;
            for (int i = 2; i <= numNodes; i++) {
                parent[i] = sc.nextInt();
            }
            mark(1);
            for (int i = 0; i < numQueries; i++) {
                String op = sc.next();
                int node = sc.nextInt();
                if (op.equals("Q")) {
                    sum += query(node);
                } else {
                    mark(node);
                }
            }
            System.out.println(sum);
        }
    }

    static void mark(int node) {
        tree[node] = 1;
        update(node, 1);
    }

    static long query(int node) {
        long res = sums[node - 1];
        while (node != 1) {
            node = parent[node];
            res += sums[node - 1];
        }
        return res;
    }

    static void update(int node, long val) {
        while (node <= numNodes) {
            sums[node - 1] += val;
            node += node & -node;
        }
    }
}
