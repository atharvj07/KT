import sys
from collections import defaultdict

input_list = sys.stdin.readlines()
n,m = map(int,input_list[0].split())
graph=defaultdict(list)
for i in range(1,m+1):
  x,y,c = map(int,input_list[i].split())
  graph[x].append((y,c))
  graph[y].append((x,c))
q = input_list[m+1]
for i in range(m+2,m+2+int(q)):
  u,v = map(int,input_list[i].split())
  visited=set()
  color=[]
  visited.add(u)
  stack=[(u,c)for c in graph[u]]
  while stack:
    node,c = stack.pop()
    if node==v: color.append(c)
    if node not in visited:
      visited.add(node)
      stack.extend([(x,c) for x,c in graph[node] if x not in visited])
  print(len(set(color)))
