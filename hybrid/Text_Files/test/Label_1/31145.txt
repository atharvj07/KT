import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class TreeDecomposition {

    static class Edge {
        int u, v;

        public Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Path {
        int u, v;

        public Path(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            edges.add(new Edge(u, v));
        }

        List<Path> paths = decompose(edges);
        if (paths == null) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            System.out.println(paths.size());
            for (Path path : paths) {
                System.out.println(path.u + " " + path.v);
            }
        }
    }

    static List<Path> decompose(List<Edge> edges) {
        int n = edges.size() + 1;
        List<List<Integer>> adjList = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adjList.add(new ArrayList<>());
        }
        for (Edge edge : edges) {
            adjList.get(edge.u).add(edge.v);
            adjList.get(edge.v).add(edge.u);
        }

        List<Path> paths = new ArrayList<>();
        for (int i = 1; i <= n; i++) {
            if (adjList.get(i).size() == 1) {
                int neighbor = adjList.get(i).get(0);
                paths.add(new Path(i, neighbor));
                adjList.get(i).clear();
                adjList.get(neighbor).remove((Integer) i);
            }
        }

        while (true) {
            boolean foundPath = false;
            for (int i = 1; i <= n; i++) {
                if (adjList.get(i).size() == 2) {
                    int neighbor1 = adjList.get(i).get(0);
                    int neighbor2 = adjList.get(i).get(1);
                    paths.add(new Path(neighbor1, neighbor2));
                    adjList.get(i).clear();
                    adjList.get(neighbor1).remove((Integer) i);
                    adjList.get(neighbor2).remove((Integer) i);
                    foundPath = true;
                    break;
                }
            }
            if (!foundPath) {
                break;
            }
        }

        for (int i = 1; i <= n; i++) {
            if (!adjList.get(i).isEmpty()) {
                return null;
            }
        }

        return paths;
    }
}
