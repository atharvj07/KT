import java.util.*;
class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int N = sc.nextInt();
		int X = sc.nextInt() - 1;
		int Y = sc.nextInt() - 1;
		long[] dist = new long[N];
		for (int i = 0; i < N; i++) {
			dist[i] = Long.MAX_VALUE;
		}
		Queue<Integer> q = new LinkedList<>();
		dist[X] = 0L;
		q.add(X);
		while (!q.isEmpty()) {
			int v = q.poll();
			for (int i = 0; i < 3; i++) {
				int nv = 0;
				if (i == 0) {
					nv = v - 1;
				} else if (i == 1) {
					nv = v + 1;
				} else if (i == 2) {
					nv = Y;
				}
				if ( nv >= 0 && nv < N && dist[nv] > dist[v] + 1) {
					dist[nv] = dist[v] + 1;
					q.add(nv);
				}
			}
		}
		for (int k = 1; k < N; k++) {
			int ans = 0;
			for (int i = 0; i < N; i++) {
				for (int j = i + 1; j < N; j++) {
					if (dist[i] + dist[j] == k) {
						ans++;
					}
				}
			}
			System.out.println(ans);
		}
	}
}
