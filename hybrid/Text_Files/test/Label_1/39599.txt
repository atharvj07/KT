import sys
input = sys.stdin.readline

m, n = map(int, input().split())
times = [list(map(int, input().split())) for i in range(m)]

for t in times:
    t.insert(0, 0)

t = [0] * (n + 1)
for i in range(1, n + 1):
    for j in range(1, m + 1):
        t[i] = max(t[i], t[i - 1] + times[j - 1][i])
    print(t[i])
