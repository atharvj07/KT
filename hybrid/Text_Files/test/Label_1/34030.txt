import java.util.*;
import java.io.*;

class Main {
  static boolean ok;
  static int[] a;
  static int n;

  static void swap(int[] arr, int i, int j) {
    int t = arr[i];
    arr[i] = arr[j];
    arr[j] = t;
  }

  static void sort(int[] arr, int l, int r) {
    if (l < 0 || l >= r)
      return;
    int m = (l + r) / 2;
    swap(arr, m, r);
    int i = l;
    for (int j = l; j <= r; j++) {
      if (arr[j] < arr[r]) {
        swap(arr, i, j);
        i++;
      }
    }
    swap(arr, i, r);
    sort(arr, l, i - 1);
    sort(arr, i + 1, r);
  }

  static void reverse(int[] arr, int l, int r) {
    while (l < r) {
      swap(arr, l++, r--);
    }
  }

  static boolean isStable() {
    for (int i = 1; i < n - 1; i++) {
      if (a[i] < a[i - 1] || a[i] > a[i + 1]) {
        return false;
      }
    }
    return true;
  }

  static int[] getMedianFiltered() {
    int[] ret = new int[n];
    for (int i = 1; i < n - 1; i++) {
      ret[i] = a[i - 1] + a[i] + a[i + 1];
      ret[i] /= 2;
    }
    return ret;
  }

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    n = Integer.parseInt(br.readLine());
    String[] input = br.readLine().split(" ");
    a = new int[n];
    ok = true;
    for (int i = 0; i < n; i++) {
      a[i] = Integer.parseInt(input[i]);
      if (a[i] != 0 && a[i] != 1) {
        ok = false;
      }
    }
    if (!ok) {
      System.out.println(1);
      return;
    }
    int cnt = 0;
    while (!isStable()) {
      a = getMedianFiltered();
      cnt++;
    }
    System.out.println(cnt);
    for (int i = 0; i < n; i++) {
      System.out.print(a[i] + " ");
    }
    System.out.println();
  }
}
