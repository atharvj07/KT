#include <iostream>
#include <vector>

int main() {
  int n, p;
  std::cin >> n >> p;
  std::vector<std::pair<int, int>> a(n);
  for (int i = 0; i < n; ++i) {
    int x, y;
    std::cin >> x >> y;
    a[i] = std::make_pair(x, y);
  }
  std::vector<int> b(n + 1);
  for (const auto &ai : a) {
    ++b[ai.first];
    ++b[ai.second];
    --b[n + 1];
  }
  for (int i = n - 1; i > 0; --i) {
    b[i] += b[i + 1];
  }
  int ans = 0;
  for (int i = 1; i <= n; ++i) {
    if (b[i] >= p) {
      ans += i - 1;
    }
  }
  std::cout << ans << std::endl;
  return 0;
}
