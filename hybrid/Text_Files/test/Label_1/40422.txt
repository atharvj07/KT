n, m = map(int, input().split())
a = list(map(int, input().split()))
e = [[] for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    e[u].append(v)
    e[v].append(u)
vis = [False] * (n+1)
p = [0] * (n+1)
def dfs(v, fr):
    vis[v] = True
    for u in e[v]:
        if u == fr: continue
        if not vis[u]:
            p[u] = v
            dfs(u, v)
dfs(1, 0)
ans = 0
for i in range(2, n+1):
    ans += min(a[p[i]-1], a[i-1])
print(ans / (n-1))
