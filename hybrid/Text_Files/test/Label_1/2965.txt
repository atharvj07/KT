#include <cstdio>
#include <vector>
#include <algorithm>

int main() {
    int n, m;
    scanf("%d%d", &n, &m);
    std::vector<int> a(m);
    for (int i = 0; i < m; ++i) {
        scanf("%d", &a[i]);
    }
    std::sort(a.begin(), a.end());
    long long ans = 0;
    for (int i = 1; i < m; ++i) {
        ans += std::abs(a[i] - a[i - 1]);
    }
    for (int i = 2; i < m; ++i) {
        if (a[i] == a[i - 1]) continue;
        long long diff = std::abs(a[i] - a[i - 1]);
        long long diff1 = std::abs(a[i] - a[i - 2]);
        long long diff2 = std::abs(a[i - 1] - a[i - 2]);
        ans = std::min(ans, diff1 + diff2 - diff);
    }
    printf("%lld\n", ans);
    return 0;
}
