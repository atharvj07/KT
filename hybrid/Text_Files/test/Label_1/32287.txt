import java.util.Scanner;

public class DominoCarpet {

    private static final int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        char[][] carpet = new char[n * 3][m * 3];
        for (int i = 0; i < n * 3; i++) {
            carpet[i] = scanner.nextLine().toCharArray();
        }

        long[][][] dp = new long[n][m][4]; // dp[i][j][k] = number of ways to fill the sub-carpet from (0, 0) to (i, j) with the last domino placed in state k
        // k = 0: horizontal, k = 1: vertical, k = 2: horizontal (special case), k = 3: vertical (special case)

        // Base case: fill the first square
        if (carpet[0][0] == 'O') {
            dp[0][0][0] = 1;
            dp[0][0][1] = 1;
        } else {
            dp[0][0][2] = 1;
            dp[0][0][3] = 1;
        }

        // Fill the rest of the carpet
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k < 4; k++) {
                    if (i > 0) {
                        // Place a horizontal domino on top
                        if (k == 0 || k == 2) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3 - 1][j * 3] == 'O') {
                                dp[i][j][0] = (dp[i][j][0] + dp[i - 1][j][k]) % MOD;
                            }
                        }
                        // Place a vertical domino on the left
                        if (k == 1 || k == 3) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3][j * 3 - 1] == 'O') {
                                dp[i][j][1] = (dp[i][j][1] + dp[i - 1][j][k]) % MOD;
                            }
                        }
                    }
                    if (j > 0) {
                        // Place a horizontal domino on the left
                        if (k == 0 || k == 2) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3][j * 3 - 1] == 'O') {
                                dp[i][j][0] = (dp[i][j][0] + dp[i][j - 1][k]) % MOD;
                            }
                        }
                        // Place a vertical domino on top
                        if (k == 1 || k == 3) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3 - 1][j * 3] == 'O') {
                                dp[i][j][1] = (dp[i][j][1] + dp[i][j - 1][k]) % MOD;
                            }
                        }
                    }
                    // Special cases for the first row and column
                    if (i == 0 && j == 0) {
                        continue;
                    }
                    if (i == 0) {
                        // Place a horizontal domino on the left (special case)
                        if (k == 2) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3][j * 3 - 1] == 'O') {
                                dp[i][j][2] = (dp[i][j][2] + dp[i][j - 1][k]) % MOD;
                            }
                        }
                    }
                    if (j == 0) {
                        // Place a vertical domino on top (special case)
                        if (k == 3) {
                            if (carpet[i * 3][j * 3] == 'O' && carpet[i * 3 - 1][j * 3] == 'O') {
                                dp[i][j][3] = (dp[i][j][3] + dp[i - 1][j][k]) % MOD;
                            }
                        }
                    }
                }
            }
        }

        long result = 0;
        for (int k = 0; k < 4; k++) {
            result = (result + dp[n - 1][m - 1][k]) % MOD;
        }

        System.out.println(result);
    }
}
