for _ in range(int(input())):
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]
    for i in range(5):
        for j in range(i + 1, 5):
            flag = True
            for day in a:
                if day[i] + day[j] == 0:
                    flag = False
                    break
            if flag:
                print("YES")
                exit()
    print("NO")
