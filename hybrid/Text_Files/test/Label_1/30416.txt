import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int q = in.nextInt();
        for (int i = 0; i < q; i++) {
            int n = in.nextInt();
            String s = in.next();
            String t = in.next();
            int[] a = new int[26];
            int[] b = new int[26];
            for (int j = 0; j < n; j++) {
                a[s.charAt(j) - 'a']++;
                b[t.charAt(j) - 'a']++;
            }
            boolean ok = true;
            for (int j = 0; j < 26; j++) {
                if (a[j] != b[j]) {
                    ok = false;
                    break;
                }
            }
            if (!ok) {
                System.out.println(-1);
                continue;
            }
            int ans = 0;
            int[] pos1 = new int[26];
            int[] pos2 = new int[26];
            Arrays.fill(pos1, -1);
            Arrays.fill(pos2, -1);
            for (int j = 0; j < n; j++) {
                if (s.charAt(j) != t.charAt(j)) {
                    if (pos1[s.charAt(j) - 'a'] == -1) {
                        pos1[s.charAt(j) - 'a'] = j;
                    } else {
                        ans += Math.min(j - pos1[s.charAt(j) - 'a'], n - j + pos1[s.charAt(j) - 'a']);
                        pos1[s.charAt(j) - 'a'] = j;
                    }
                    if (pos2[t.charAt(j) - 'a'] == -1) {
                        pos2[t.charAt(j) - 'a'] = j;
                    } else {
                        ans += Math.min(j - pos2[t.charAt(j) - 'a'], n - j + pos2[t.charAt(j) - 'a']);
                        pos2[t.charAt(j) - 'a'] = j;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
