import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = 0;
        while (n > 0) {
            int p = 1;
            while (p <= n) p *= 2;
            p /= 2;
            n -= p;
            k++;
        }
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print("1 ");
        }
        System.out.println();
    }
}
