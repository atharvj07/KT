import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] perm = new int[n];
        int[] fav = new int[n];
        for (int i = 0; i < n; i++) {
            perm[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            fav[i] = in.nextInt();
        }
        boolean[] vis = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (vis[perm[i]]) {
                System.out.println("NO");
                return;
            }
            vis[perm[i]] = true;
        }
        int[] cnt = new int[n + 1];
        for (int i = 0; i < n; i++) {
            cnt[fav[i]]++;
        }
        for (int i = 1; i <= n; i++) {
            if (cnt[i] % 2 != 0) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
