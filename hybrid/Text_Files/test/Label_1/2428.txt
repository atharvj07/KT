#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  vector<int> can(n);
  for (int i = 0; i < n; i++) { can[i] = s[i] == '1'; }
  vector<int> dp(n, -1);
  function<int(int)> get = [&](int i) -> int {
    if (i >= n) return 0;
    if (dp[i] != -1) return dp[i];
    dp[i] = get(i + 1) + i;
    if (can[i]) dp[i] = min(dp[i], get(i + k + 1) + i);
    return dp[i];
  };
  cout << get(0) << endl;
}
