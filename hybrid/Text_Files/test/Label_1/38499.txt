import sys
n = int(input())
p = list(map(float,input().split()))
dp=[[[0. for j in range(2)] for k in range(2)] for i in range(n+1)]
dp[0][0][0] = 1.
for i in range(1,n+1):
  for j in range(2):
    for k in range(2):
      dp[i][j][k] = 1.-(1-dp[i-1][j][k])*(1-p[i-1])
      if j: dp[i][j][k] = max(dp[i][j][k],1.-(1-dp[i-1][j-1][k])*p[i-1])
      if not k and dp[i][j][k]>0.: dp[i][j][1]=1.
print(dp[n][1][1])
