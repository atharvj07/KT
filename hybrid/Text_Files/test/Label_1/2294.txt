#include <bits/stdc++.h>

using namespace std;

const int maxn = 35;

int n, k, a[maxn], b[1 << maxn];
long long dp[1 << maxn], A, B;

int main() {
    cin >> n >> k >> A >> B;
    for (int i = 0; i < k; i++) {
        cin >> a[i];
        b[a[i]] = 1;
    }
    int bas = 1 << n;
    for (int i = 0; i < bas; i++) { dp[i] = 1e18; }
    dp[0] = 0;
    for (int i = 1; i < bas; i++) {
        for (int j = i; j; j = (j - 1) & i) {
            dp[i] = min(dp[i], dp[i ^ j] + (j ? B * b[j] * __builtin_popcount(j) : A));
            dp[i] = min(dp[i], dp[i ^ j] + (j ? B * b[j ^ i] * __builtin_popcount(j ^ i) : A));
        }
    }
    cout << dp[bas - 1] << endl;
    return 0;
}
