#include<iostream>
#include <algorithm>
#include <numeric>
using namespace std;
int main(){
    int n; cin>>n;
    int a[n]; for(int &i:a) cin>>i;
    sort(a,a+n,greater<int>());
    bool ok=false;
    string ans;
    for(int i:a){
        if(((ans.size()%2==0 and i==0) or (ans.empty() and i==0))) continue;
        ok=true;
        ans+=to_string(i);
    }
    if(ans.empty()) cout<<-1;
    else if(ok){
        int sum=0;
        for(auto i:ans) sum+=i-'0';
        if(sum%9==0) cout<<ans;
        else cout<<-1;
    }
    else cout<<0;
    return 0;
}
