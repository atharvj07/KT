import java.util.Scanner;

public class MinimalTestLength {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        String s1 = scanner.nextLine();
        String s2 = scanner.nextLine();
        String s3 = scanner.nextLine();

        int s1_length = s1.length();
        int s2_length = s2.length();
        int s3_length = s3.length();

        // Find the longest common substring of s1 and s2
        int longestCommonSubstrLength = findLongestCommonSubstr(s1, s2);

        // Find the longest common substring of s1, s2 and s3
        int longestCommonSubstrLength3 = findLongestCommonSubstr(s1, s2, s3);

        // Add the lengths of the substrings and the longest common substring(s) to find the minimal test length
        int minimalTestLength = s1_length + s2_length + s3_length - 2 * longestCommonSubstrLength - longestCommonSubstrLength3;

        // Print the minimal test length
        System.out.println(minimalTestLength);
    }

    // Method to find the longest common substring of two strings
    static int findLongestCommonSubstr(String s1, String s2) {
        int[][] dp = new int[s1.length() + 1][s2.length() + 1];
        int maxLen = 0;

        for (int i = 1; i <= s1.length(); i++) {
            for (int j = 1; j <= s2.length(); j++) {
                if (s1.charAt(i - 1) == s2.charAt(j - 1)) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                    maxLen = Math.max(maxLen, dp[i][j]);
                } else {
                    dp[i][j] = 0;
                }
            }
        }

        return maxLen;
    }

    // Method to find the longest common substring of three strings
    static int findLongestCommonSubstr(String s1, String s2, String s3) {
        int[][][] dp = new int[s1.length() + 1][s2.length() + 1][s3.length() + 1];
        int maxLen = 0;

        for (int i = 1; i <= s1.length(); i++) {
            for (int j = 1; j <= s2.length(); j++) {
                for (int k = 1; k <= s3.length(); k++) {
                    if (s1.charAt(i - 1) == s2.charAt(j - 1) && s1.charAt(i - 1) == s3.charAt(k - 1)) {
                        dp[i][j][k] = dp[i - 1][j - 1][k - 1] + 1;
                        maxLen = Math.max(maxLen, dp[i][j][k]);
                    } else {
                        dp[i][j][k] = 0;
                    }
                }
            }
        }

        return maxLen;
    }
}
