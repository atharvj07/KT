#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> m(n);
    for (int& mi : m) {
        cin >> mi;
    }
    vector<int> a(n);

    for (int i = 0; i < n; ++i) {
        if (i == 0) {
            a[i] = m[i];
        } else if (i == n - 1) {
            a[i] = min(m[i], m[i - 1]);
        } else {
            a[i] = min({m[i], m[i - 1], m[i + 1]});
        }
    }

    for (int ai : a) {
        cout << ai << " ";
    }
    cout << "\n";

    return 0;
}
