#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  vector<int> a(n);
  for (int i = 0; i < n; i++) cin >> a[i];
  double ans = 0;
  for (int i = 0; i < n - k + 1; i++) {
    double sum = 0;
    for (int j = i; j < i + k; j++) sum += a[j];
    ans = max(ans, sum / k);
  }
  cout << ans << endl;
  return 0;
}
