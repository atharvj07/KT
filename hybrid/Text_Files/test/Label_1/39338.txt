from heapq import heapify, heappush, heappop
n, q = map(int, input().split())
a = list(map(int, input().split()))
heapify(a)
for t, x in (map(int, input().split()) for _ in range(q)):
    if t == 1:
        a[a.index(1 - a[x - 1])] = 1 - a[x - 1]
        heapify(a)
    else:
        for _ in range(x - 1):
            heappop(a)
        print(heappop(a))
        heapify(a)
