def restore_expression(s):
    for i in range(1, len(s)):
        for j in range(i + 1, len(s)):
            a, b, c = s[:i], s[i:j], s[j:]
            if int(a) + int(b) == int(c) and not (a.startswith('0') and len(a) > 1) and not (b.startswith('0') and len(b) > 1) and not (c.startswith('0') and len(c) > 1):
                return a + '+' + b + '=' + c

s = input().strip()
print(restore_expression(s))
