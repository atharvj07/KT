import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Map<Integer, Integer> leafCount = new HashMap<>(n);
        List<Integer> even = new ArrayList<>();
        int[][] edges = new int[n - 1][3];
        for (int i = 0; i < n - 1; i++) {
            edges[i][0] = sc.nextInt();
            edges[i][1] = sc.nextInt();
            edges[i][2] = sc.nextInt();
            leafCount.compute(edges[i][0], (k, v) -> v == null ? 1 : v + 1);
            leafCount.compute(edges[i][1], (k, v) -> v == null ? 1 : v + 1);
            if (edges[i][2] % 2 == 0) even.add(edges[i][2]);
        }
        Collections.sort(even);
        int size = leafCount.size();
        for (Map.Entry<Integer, Integer> entry : leafCount.entrySet()) {
            if (entry.getValue() == 1) size--;
        }
        if (size != even.size()) {
            System.out.println("NO");
            return;
        }
        int idx = 0;
        System.out.println("YES");
        System.out.println(n - 1);
        for (int[] edge : edges) {
            if (leafCount.get(edge[0]) == 1 && leafCount.get(edge[1]) == 1) {
                int a = edge[0], b = edge[1], x = even.get(idx++);
                System.out.println(a + " " + b + " " + x);
            }
        }
    }
}
