#include <vector>
#include <iostream>
int main() {
  int n;
  std::cin >> n;
  std::vector<int> v(n);
  std::vector<int> w;
  for (int i = 0; i < n; ++i) std::cin >> v[i];
  int ans = 0;
  while (true) {
    bool flag = w == v;
    w.clear();
    if (flag) break;
    ++ans;
    w.push_back(v[0]);
    for (int i = 1; i < n - 1; ++i)
      w.push_back((v[i - 1] + v[i] + v[i + 1] + 1) / 2);
    w.push_back(v[n - 1]);
    v = w;
  }
  std::cout << ans << std::endl;
  for (int i = 0; i < n; ++i) std::cout << v[i] << ' ';
}
