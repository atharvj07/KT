import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] p = new int[n];
        int[] d = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            d[i] = input.nextInt();
        }
        boolean[] vis = new boolean[n + 1];
        for (int i = 0; i < n; i++) {
            if (!vis[p[i]]) {
                int j = i;
                int cycleLen = 0;
                while (!vis[p[j]]) {
                    vis[p[j]] = true;
                    j = p[j] - 1;
                    cycleLen++;
                }
                if (cycleLen % d[i] != 0) {
                    System.out.println("NO");
                    return;
                }
            }
        }
        System.out.println("YES");
    }
}
