import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] b = new int[n];
        int[] i = new int[n];
        int[] r = new int[n];
        for (int j = 0; j < n; j++) {
            b[j] = sc.nextInt();
        }
        for (int j = 0; j < n; j++) {
            i[j] = sc.nextInt();
        }
        for (int j = 0; j < n; j++) {
            r[j] = sc.nextInt();
        }
        int ans = 0;
        for (int j = 0; j < n; j++) {
            boolean flag = false;
            for (int k = 0; k < n; k++) {
                if (j == k) {
                    continue;
                }
                if (b[j] < b[k] && i[j] < i[k] && r[j] < r[k]) {
                    flag = true;
                    break;
                }
            }
            if (flag) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
