#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string s, t;
    cin >> s >> t;
    if (s == t)
        cout << "0\n";
    else {
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] != t[i]) {
                if (cnt)
                    cout << "-1\n";
                cnt = 1;
                int j = i;
                while (j < n && t[j] != s[i]) j++;
                if (j == n)
                    cout << "-1\n";
                else
                    s[j] = s[i], i = j - 1;
            }
        }
        cout << "1\n";
    }
}
