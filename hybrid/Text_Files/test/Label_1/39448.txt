import sys
n = int(input())
a = list(map(int, input().split()))
ans = []
b = sorted(range(n), key=lambda x: a[x])
for i in range(n):
    if a[i] != a[b[i]]:
        j = i + 1
        while j < n and a[j] == a[i]:
            j += 1
        ans.append([i + 1, j])
        a[i + 1: j + 1] = reversed(a[i + 1: j + 1])
        for k in range(i + 1, j):
            b[k], b[k + 1] = b[k + 1], b[k]
print(*ans, sep='\n')
