import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    a = [int(i) for i in input().split()]
    b = []
    if sum(a) % 3 != 0:
        print('NO')
    else:
        print('YES')
        cnt = 0
        for i in range(n - 2):
            if a[i] == a[i + 1] == a[i + 2]:
                continue
            b.append(i + 1)
            cnt += 1
            a[i:i + 3] = [a[i] ^ a[i + 1] ^ a[i + 2]] * 3
        print(cnt)
        print(*b)
