import java.util.Scanner;

public class Main {

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    while (in.hasNextLine()) {
      calculate(in.nextLine());
    }
  }

  private static void calculate(String input) {
    int r1 = 0, r2 = 0, r3 = 0;
    for (char c : input.toCharArray()) {
      if (c >= '0' && c <= '9') {
        r2 = r2 * 10 + (c - '0');
      } else {
        switch (c) {
          case '+':
            r1 = calc(r1, r2, r3);
            r2 = 0;
            r3 = 1;
            break;
          case '-':
            r1 = calc(r1, r2, r3);
            r2 = 0;
            r3 = 2;
            break;
          case '*':
            r1 = calc(r1, r2, r3);
            r2 = 0;
            r3 = 3;
            break;
          case '=':
            r1 = calc(r1, r2, r3);
            r3 = 0;
            if (r1 < 0 || r1 > 9999) {
              System.out.println("E");
            } else {
              System.out.println(r1);
            }
            break;
        }
      }
    }
  }

  private static int calc(int r1, int r2, int r3) {
    switch (r3) {
      case 1:
        return r1 + r2;
      case 2:
        return r1 - r2;
      case 3:
        return r1 * r2;
      default:
        return r2;
    }
  }
}
