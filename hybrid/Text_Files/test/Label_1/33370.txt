import java.util.*;

class Main {

    static int[] solve(int[] a) {
        int n = a.length;
        int[] ans = new int[n];

        for (int i = 0; i < n; i++) {
            int ind = 0;
            for (int j = 1; j < 32; j++) {
                if ((a[i] & (1 << j)) > 0)
                    ind = j;
            }
            if (ind > 0) ans[i] = (1 << (ind - 1));
        }

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int x : solve(a)) System.out.println(x);
    }
}
