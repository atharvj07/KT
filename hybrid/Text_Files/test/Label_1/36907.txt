def count_bits(num):
    cnt = 0
    while num > 0:
        cnt += num % 2
        num //= 2
    return cnt

def get_median(n, m, a):
    a.sort()
    l = 0
    r = (1<<m) -1
    for i in range(m):
        cnt_l = 0
        for j in range(n):
            if a[j][i] == '0':
               cnt_l = cnt_l + 1
        cnt_r = n-cnt_l
        if cnt_l <= (n+1)//2 and cnt_r <= (n+1)//2:
            l = l + (1<<i)
        elif cnt_l > (n+1)//2:
            r = r - (1<<i)
    return bin(l)[2:].zfill(m)

t = int(input())
for i in range(t):
    n, m = map(int, input().split())
    a = []
    for j in range(n):
        x = input()
        a.append(x)
    print(get_median(n, m, a))
