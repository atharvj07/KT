#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> c(n), t(n);
  for (int i = 0; i < n; i++) cin >> c[i];
  for (int i = 0; i < n; i++) cin >> t[i];
  for (int i = 1; i + 1 < n; i++)
    if (t[i] != 2LL * t[i] - c[i - 1] - c[i + 1]) {
      cout << "No" << endl;
      return 0;
    }
  cout << "Yes" << endl;
}
