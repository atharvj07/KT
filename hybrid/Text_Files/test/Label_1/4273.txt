#include <bits/stdc++.h>
#define ll long long
using namespace std;

string s, t, u;

int main() {
    cin >> s;
    while (s.length()) {
        if (t.empty() || u.empty()) {
            t += s[0];
            s.erase(s.begin());
            continue;
        }
        if (u.back() <= t.back()) {
            t += s[0];
            s.erase(s.begin());
        } else {
            u += t.back();
            t.pop_back();
        }
    }
    while (t.length()) {
        if (t.back() >= u.back()) {
            u += t.back();
            t.pop_back();
        } else {
            break;
        }
    }
    cout << u << endl;
    return 0;
}
