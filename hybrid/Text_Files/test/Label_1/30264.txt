import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int h = sc.nextInt();
      int m = sc.nextInt();
      int s = sc.nextInt();
      if (h == -1 && m == -1 && s == -1) {
        break;
      }
      int max = 120 * 60;
      int used = (h * 60 + m) * 60 + s;
      int remain = max - used;
      int remain3 = remain * 3;
      int remainH = remain / 3600;
      int remainM = (remain % 3600) / 60;
      int remainS = (remain % 3600) % 60;
      int remain3H = remain3 / 3600;
      int remain3M = (remain3 % 3600) / 60;
      int remain3S = (remain3 % 3600) % 60;
      System.out.println(String.format("%02d:%02d:%02d", remainH, remainM, remainS));
      System.out.println(String.format("%02d:%02d:%02d", remain3H, remain3M, remain3S));
    }
  }
}
