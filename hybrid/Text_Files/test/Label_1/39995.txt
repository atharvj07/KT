import sys,math
from collections import Counter
input=sys.stdin.readline

n,m=map(int,input().split())
a=[]
for i in range(n):
    a.append(input().strip())
b=[]
for i in range(n):
    b.append(list(map(int,input().strip().split())))
ans=0
for j in range(m):
    cnt=Counter()
    for i in range(n):
        cnt[a[i][j]]+=1
    f=0
    for k in cnt:
        if cnt[k]==1:
            f=1
            break
    if f==0:
        for i in range(n):
            ans+=min(b[i])
print(ans)
