import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String[] pos = in.nextLine().split(" ");
        System.out.println(checkPos(pos[0], pos[1], pos[2], pos[3]) ? "CHECKMATE" : "OTHER");
    }

    public static boolean checkPos(String r1, String r2, String king1, String king2) {
        int[][] board = new int[8][8];
        board[r1.charAt(0) - 'a'][r1.charAt(1) - '1'] = 1;
        board[r2.charAt(0) - 'a'][r2.charAt(1) - '1'] = 1;
        board[king1.charAt(0) - 'a'][king1.charAt(1) - '1'] = 2;
        board[king2.charAt(0) - 'a'][king2.charAt(1) - '1'] = 3;

        for (int i = 0; i < 8; i++) {
            if (board[i][king2.charAt(1) - '1'] == 1) return true;
            if (board[king2.charAt(0) - 'a'][i] == 1) return true;
        }
        for (int i = king2.charAt(0) - 'a', j = king2.charAt(1) - '1'; i >= 0 && j >= 0; i--, j--) {
            if (board[i][j] == 1) return true;
        }
        for (int i = king2.charAt(0) - 'a', j = king2.charAt(1) - '1'; i >= 0 && j < 8; i--, j++) {
            if (board[i][j] == 1) return true;
        }

        return false;
    }
}
