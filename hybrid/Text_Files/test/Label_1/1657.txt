#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
long long x, y, a[6];
int main() {
  long long t, ans;
  cin >> t;
  while (t--) {
    cin >> x >> y;
    for (int i = 0; i < 6; i++) cin >> a[i];
    if (abs(x) >= abs(y)) {
      x -= y;
      if (x >= 0)
        ans = (abs(x) - abs(y) + abs(y)) * a[2] + abs(y) * a[0];
      else
        ans = (abs(x) - abs(y) + abs(y)) * a[3] + abs(y) * a[1];
    } else {
      y -= x;
      if (y >= 0)
        ans = (abs(y) - abs(x) + abs(x)) * a[0] + abs(x) * a[5];
      else
        ans = (abs(y) - abs(x) + abs(x)) * a[1] + abs(x) * a[4];
    }
    cout << ans << endl;
  }
  return 0;
}
