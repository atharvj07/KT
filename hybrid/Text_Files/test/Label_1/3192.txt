#include <bits/stdc++.h>
using namespace std;
const int maxn = 5005;
int n, k, a[maxn], c[maxn];
bool vis[maxn];
vector<int> same[maxn];
int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    same[a[i]].push_back(i);
  }
  int cnt = 0;
  for (int i = 1; i <= n; i++) {
    if (vis[a[i]]) continue;
    vis[a[i]] = true;
    if (k - cnt < (int)same[a[i]].size()) {
      cout << "NO" << endl;
      return false;
    }
    cnt++;
  }
  cout << "YES" << endl;
  int cur = 1;
  for (int i = 1; i <= 5000; i++) {
    if (!vis[i]) continue;
    for (int x : same[i]) {
      c[x] = cur;
      cur++;
    }
    cur = (cur + 1) % k;
  }
  for (int i = 1; i <= n; i++) {
    cout << c[i] << " ";
  }
  cout << endl;
  return 0;
}
