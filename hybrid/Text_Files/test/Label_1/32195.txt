import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
    static final long MOD = 1000000007L;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] x = new int[6], y = new int[6];
        for (int i = 0; i < 6; i++) x[i] = sc.nextInt();
        for (int i = 0; i < 6; i++) y[i] = sc.nextInt();
        // cx[i][j] is the number of ways to move from (x[i],y[i]) to (x[j],y[j])
        long[][] cx = new long[6][6], cy = new long[6][6];
        for (int i = 0; i < 6; i++) for (int j = i+1; j < 6; j++) {
            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    if (dx == 0 && dy == 0) continue;
                    if (dx != 0 && dy != 0) continue;
                    if (!(x[i]+dx >= x[j] && x[i]+dx <= x[j] && y[i]+dy >= y[j] && y[i]+dy <= y[j])) continue;
                    cx[i][j] = (cx[i][j] + 1) % MOD;
                    cy[i][j] = (cy[i][j] + 1) % MOD;
                }
            }
        }
        // dp[mask] is the number of ways to visit the subset of nodes represented by the mask
        long[] dp = new long[1 << 6];
        dp[1 << 0] = 1;
        for (int mask = 1; mask < 1 << 6; mask++) {
            for (int i = 0; i < 6; i++) {
                for (int j = i+1; j < 6; j++) {
                    // move from i to j if j is in the subset and i is not
                    if ((mask & (1 << j)) != 0 && (mask & (1 << i)) == 0) {
                        dp[mask] = (dp[mask] + (dp[mask ^ (1 << i)] * cx[i][j] % MOD * cy[i][j] % MOD)) % MOD;
                    }
                }
            }
        }
        System.out.println(dp[(1 << 6) - 1]);
    }
}
