import sys
mod = 998244353

def add(a, b):
  return (a + b) % mod

def sub(a, b):
  return (a - b) % mod

def mul(a, b):
  return (a * b) % mod

n, m, l, r = map(int, sys.stdin.readline().split())
ans = 1
if n > 1:
  ans = mul(ans, sub(r + 1, l))
if m > 1:
  ans = mul(ans, sub(r + 1, l))
print(ans)
