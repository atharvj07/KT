n, m, k = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
pref = [[0] * m for _ in range(n)]
for j in range(m):
    if b[j] == 0: continue
    for i in range(n):
        pref[i][j] = pref[i - 1][j] + a[i]
ans = 0
for i1 in range(1, n - k + 2):
    for j1 in range(1, m - k + 2):
        if pref[i1 - 1][j1 - 1] == 0:
            continue
        i2 = i1 + k - 1
        j2 = j1 + k - 1
        if pref[i1 - 1][j2] == 0 or pref[i2][j2] == 0 or pref[i2][j1 - 1] == 0 or pref[i1 - 1][j2] != pref[i1 - 1][j1 - 1] or pref[i2][j2] != pref[i2][j1 - 1] or pref[i2][j2] != pref[i1 - 1][j2]:
            continue
        ans += 1
print(ans)
