import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Minotaur {

    static class Block {
        int x;
        int y;
        int dist;
        char type;

        Block(int x, int y, int dist, char type) {
            this.x = x;
            this.y = y;
            this.dist = dist;
            this.type = type;
        }
    }

    static int n, m;
    static char[][] labyrinth;
    static int xT, yT, xM, yM;

    static int[] dx = {-1, 1, 0, 0};
    static int[] dy = {0, 0, -1, 1};

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        labyrinth = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                labyrinth[i][j] = line.charAt(j);
            }
        }
        xT = scanner.nextInt() - 1;
        yT = scanner.nextInt() - 1;
        xM = scanner.nextInt() - 1;
        yM = scanner.nextInt() - 1;

        int result = bfs();
        System.out.println(result == Integer.MAX_VALUE ? -1 : result);
    }

    static int bfs() {
        Queue<Block> queue = new LinkedList<>();
        queue.add(new Block(xT, yT, 0, labyrinth[xT][yT]));
        boolean[][] visited = new boolean[n][m];
        visited[xT][yT] = true;

        while (!queue.isEmpty()) {
            Block current = queue.poll();
            if (current.x == xM && current.y == yM) {
                return current.dist;
            }

            for (int i = 0; i < 4; i++) {
                int nx = current.x + dx[i];
                int ny = current.y + dy[i];
                if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny]) {
                    char nextType = labyrinth[nx][ny];
                    if (isAccessible(current.type, nextType)) {
                        visited[nx][ny] = true;
                        queue.add(new Block(nx, ny, current.dist + 1, nextType));
                    } else if (isRotatable(current.type) && isRotatable(nextType)) {
                        visited[nx][ny] = true;
                        queue.add(new Block(nx, ny, current.dist + 2, nextType)); // +1 for rotation
                    }
                }
            }
        }

        return Integer.MAX_VALUE;
    }

    static boolean isAccessible(char from, char to) {
        switch (from) {
            case '+':
                return true;
            case '-':
                return to == '-' || to == 'L' || to == 'R';
            case '|':
                return to == '|' || to == 'U' || to == 'D';
            case '^':
                return to == '^' || to == 'U';
            case '>':
                return to == '>' || to == 'R';
            case '<':
                return to == '<' || to == 'L';
            case 'v':
                return to == 'v' || to == 'D';
            case 'L':
                return to != '<';
            case 'R':
                return to != '>';
            case 'U':
                return to != '^';
            case 'D':
                return to != 'v';
        }
        return false;
    }

    static boolean isRotatable(char type) {
        return type == '+' || type == '-' || type == '|';
    }
}
