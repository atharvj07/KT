import sys
input = sys.stdin.readline

for t in range(int(input())):
    n = int(input())
    adj = [[] for _ in range(n+1)]
    for _ in range(n-1):
        a, b = map(int, input().split())
        adj[a].append(b)
        adj[b].append(a)
    
    vis = [False]*(n+1)
    
    def dfs(node, par):
        leaves = 0
        buds = []
        for i in adj[node]:
            if not vis[i]:
                vis[i] = True
                if len(adj[i]) == 1:
                    leaves += 1
                else:
                    x, y = dfs(i, node)
                    if y == 0:
                        buds.append(i)
                    leaves += x
        if len(buds) > 0 and leaves > 0:
            buds.append(node)
        return leaves, buds

    leaves, buds = dfs(1, 0)
    print(leaves)
