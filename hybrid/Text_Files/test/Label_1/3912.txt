#include <vector>
#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> color(n + 1);
    for (int i = 1; i <= n; ++i) cin >> color[i];
    vector<vector<int>> adj(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int ans_color = 1;
    int max_neighbors = 0;
    for (int i = 1; i <= n; ++i) {
        int curr_neighbors = 0;
        for (auto u : adj[i]) {
            if (color[i] != color[u]) ++curr_neighbors;
        }
        if (curr_neighbors > max_neighbors) {
            max_neighbors = curr_neighbors;
            ans_color = color[i];
        } else if (curr_neighbors == max_neighbors) {
            ans_color = min(ans_color, color[i]);
        }
    }
    cout << ans_color << '\n';
    return 0;
}
