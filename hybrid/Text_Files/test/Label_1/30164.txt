class Main {
    long mod;
    long[] pow, inv, fact, invfact;
    int n;

    public static void main(String[] args) {
        new Main().run();
    }

    void run() {
        java.util.Scanner in = new java.util.Scanner(System.in);
        n = in.nextInt();
        int d = in.nextInt();
        mod = in.nextInt();
        int r = (n - d + 1) % 2;
        if (n == d) r++;
        if (n % d != 0 || r == 0) System.out.println(0);
        else {
            init();
            System.out.println(mul(d - 1, binpow(2, ((n + d - r) / d - 1) / 2, mod)));
        }
    }

    long binpow(long x, long p, long m) {
        x %= m;
        p %= (m - 1);
        long ret = 1;
        while (p != 0) {
            if ((p & 1) != 0) ret = ret * x % m;
            x = x * x % m;
            p /= 2;
        }
        return ret;
    }

    long mul(long l, long r) {
        return (l % mod) * (r % mod) % mod;
    }

    long add(long l, long r) {
        return (l % mod) + (r % mod) % mod;
    }

    long sub(long l, long r) {
        return (l % mod) - (r % mod) + mod;
    }

    long inv(long v) {
        return binpow(v, mod - 2, mod);
    }

    void init() {
        int ms = n + 5;
        pow = new long[ms];
        inv = new long[ms];
        fact = new long[ms];
        invfact = new long[ms];
        for (int i = 0; i < ms; i++) {
            if (i == 0) pow[i] = 1;
            else pow[i] = mul(pow[i - 1], 2);
            inv[i] = inv(pow[i]);
            if (i == 0) fact[i] = 1;
            else fact[i] = mul(fact[i - 1], i);
            invfact[i] = inv(fact[i]);
        }
    }
}
