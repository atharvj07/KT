#include <bits/stdc++.h>
#define endl "\n"
using ll = long long;
using namespace std;
typedef pair<ll, ll> pi;

ll const maxn = 2e5 + 10;
ll n;
vector<vector<pi>>v;
priority_queue<pi>pq;

int main() {
	cin >> n;
	v.resize(n);
	ll tot = 0;
	for (ll i = 0; i < n; i++) {
		ll k; cin >> k;
		for (ll j = 0; j < k; j++) {
			ll x, y; cin >> x >> y;
			tot += x;
			v[i].push_back({ x,y });
		}
	}
	sort(v.begin(), v.end());
	ll sum = 0, cnt = 10;
	for (auto x : v) {
		for (auto y : x) {
			if (sum + y.first <= 3) {
				sum += y.first;
				pq.push(y);
				if (cnt == 0) {
					pq.push({ 0,y.second * 2 });
				}
			}
			else {
				while (!pq.empty()) {
					sum -= pq.top().first;
					pq.pop();
					if (sum + y.first <= 3) {
						sum += y.first;
						pq.push(y);
						if (cnt == 0) {
							pq.push({ 0,y.second * 2 });
						}
						break;
					}
				}
			}
			cnt--;
			if (cnt == 0) cnt = 10;
		}
	}
	ll ans = 0;
	while (!pq.empty()) {
		ans += pq.top().second;
		pq.pop();
	}
	cout << ans << endl;
}
