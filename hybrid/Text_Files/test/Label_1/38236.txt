from sys import stdin

n, k = map(int, stdin.readline().split())
w = list(map(int, stdin.readline().split()))
a = int(stdin.readline())
c = list(map(int, stdin.readline().split()))

ans = -1
for i in range(n):
    if k >= w[i]:
        k += a
    elif k + a >= w[i]:
        ans = min(ans, c[i]) if ans != -1 else c[i]
    else:
        break
print(ans)
