#include <iostream>
#include <cmath>
using namespace std;

int main() {
    long long n;
    cin >> n;
    long long ans = -1;
    for (long long i = 1; i * i <= n; ++i) {
        long long tmp = sqrt(n - i * i);
        if (tmp * tmp == n - i * i) {
            if (ans == -1) ans = 2;
            ans = min(ans, 2LL);
        }
    }
    cout << ans << endl;
    return 0;
}
