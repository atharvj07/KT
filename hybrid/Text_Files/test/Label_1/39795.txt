n = int(input().strip())
cities = list(map(int, input().strip().split()))

cities.sort()

min_distance = float('inf')
pairs = 0

for i in range(1, n):
    distance = cities[i] - cities[i-1]
    if distance < min_distance:
        min_distance = distance
        pairs = 1
    elif distance == min_distance:
        pairs += 1

print(min_distance, pairs)
