import java.util.*;

public class Main {

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    String[] strings = new String[n];
    for (int i = 0; i < n; i++) {
      strings[i] = sc.next();
    }
    int minMoves = Integer.MAX_VALUE;
    for (int i = 0; i < strings[0].length(); i++) {
      int moves = 0;
      for (int j = 1; j < n; j++) {
        int idx = strings[j].indexOf(strings[0].charAt(i));
        if (idx == -1) {
          moves = Integer.MAX_VALUE;
          break;
        }
        moves += idx;
      }
      if (moves < minMoves) {
        minMoves = moves;
      }
    }
    if (minMoves == Integer.MAX_VALUE) {
      System.out.println(-1);
    } else {
      System.out.println(minMoves);
    }
  }
}
