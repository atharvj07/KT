import sys

def solve():
    line1 = sys.stdin.readline().split()
    k, n = map(int, line1)

    line2 = sys.stdin.readline().split()
    a = list(map(int, line2))

    line3 = sys.stdin.readline().split()
    b = list(map(int, line3))

    prefix = [0] * (k + 1)
    for i in range(k):
        prefix[i + 1] = prefix[i] + a[i]

    b.sort()
    count = 0
    for i in range(n):
        if b[i] >= prefix[n - 1]:
            count += 1

    print(count)

solve()
