import java.util.*;

class Main {
    static final int MAX_N = 10010;
    static String[] s = new String[MAX_N];
    static String t;
    static int n;

    static boolean check(int pos) {
        return (s[pos].substring(0, t.length()).compareTo(t) <= 0);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            s[i] = sc.next();
        }
        t = sc.next();
        int lo = 0;
        int hi = n;
        while (lo + 1 < hi) {
            int mid = (lo + hi) / 2;
            if (check(mid)) lo = mid;
            else hi = mid;
        }
        List<Integer> ans = new ArrayList<>();
        for (int i = lo; i < n; i++) {
            if (check(i)) ans.add(i + 1);
            else break;
        }
        Collections.sort(ans);
        for (int r : ans) System.out.print(r + " ");
        System.out.println();
    }
}
