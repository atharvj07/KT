def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def transform_sequence(n):
    result = []
    for i in range(1, n // 2 + 1):
        result.append(2 * i)
    if n % 2 == 1:
        result.append(n)
    return result

n = int(input().strip())
result = transform_sequence(n)
print(*result)
