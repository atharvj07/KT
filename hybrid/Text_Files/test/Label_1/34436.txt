import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class BeautifulArray {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Coupon[] coupons = new Coupon[m];
        for (int i = 0; i < m; i++) {
            int qi = scanner.nextInt();
            int wi = scanner.nextInt();
            coupons[i] = new Coupon(qi, wi);
        }

        Arrays.sort(coupons);

        List<Integer> array = new ArrayList<>();
        long totalCost = 0;
        for (Coupon coupon : coupons) {
            if (!array.contains(coupon.qi)) {
                array.add(coupon.qi);
                totalCost += coupon.wi;
            }
        }

        System.out.println(totalCost);
    }

    static class Coupon implements Comparable<Coupon> {
        int qi;
        int wi;

        public Coupon(int qi, int wi) {
            this.qi = qi;
            this.wi = wi;
        }

        @Override
        public int compareTo(Coupon o) {
            return Integer.compare(o.wi, wi);
        }
    }
}
