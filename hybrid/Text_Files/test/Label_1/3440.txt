#include <bits/stdc++.h>

using namespace std;
const int N = 1e5 + 5;
int n, a[N];
int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n, greater<int>());
    int ans = 0;
    if (a[1] >= 2) ans = 1;
    if (a[0] >= 3) ans = 2;
    if (n >= 4 && a[1] >= 3 && a[2] >= 2) ans = 2;
    if (n >= 4 && a[1] >= 2 && a[2] >= 2 && a[3] >= 2) ans = 3;
    cout << ans << '\n';
  }
  return 0;
}
