import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int m = scan.nextInt();
        int[] l = new int[m];
        for (int i = 0; i < m; i++)
            l[i] = scan.nextInt() - 1;
        int[] p = new int[m];
        boolean[] used = new boolean[n];
        int[] col = new int[n];
        for (int i = m - 1; i >= 0; i--) {
            for (int j = 0; j + l[i] - 1 < n; j++)
                if (!used[j]) {
                    p[i] = j + 1;
                    used[j] = true;
                    for (int k = j; k < j + l[i]; k++)
                        col[k] = i + 1;
                    break;
                }
        }
        for (int i = 0; i < n; i++)
            if (!used[i]) {
                System.out.println(-1);
                return;
            }
        for (int i = 0; i < m; i++)
            System.out.print(p[i] + " ");
        System.out.println();
    }
}
