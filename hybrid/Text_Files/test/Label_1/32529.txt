import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.List;

public class Main {
    private static final int MODULO = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        String field = br.readLine();
        solve(field, n);
    }

    private static void solve(String field, int n) {
        List<Character> possibilities = Arrays.asList('?', '0', '1', '2', '*');
        long[] dp = new long[n + 1];
        dp[0] = 1;
        for (int i = 1; i <= n; i++) {
            if (field.charAt(i - 1) == '*') continue;
            for (char possibility : possibilities) {
                dp[i] = (dp[i] + dp[i - 1]) % MODULO;
                if (possibility == field.charAt(i - 1)) break;
            }
        }
        System.out.println(dp[n]);
    }
}
