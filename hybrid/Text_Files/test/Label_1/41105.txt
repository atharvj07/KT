n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

counter = {}
for ai in a:
    counter[ai] = counter.get(ai, 0) + 1

calm_students = [i for i in range(n) if counter[a[i]] > 1]
max_sum = sum(b[i] for i in calm_students)

for i in range(n):
    if counter[a[i]] == 1:
        for j in calm_students:
            if a[i] & a[j] == a[i]:
                max_sum += b[i]
                break

print(max_sum)
