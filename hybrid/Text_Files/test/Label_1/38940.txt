from collections import defaultdict, deque

n = int(input())
a = list(map(int, input().split()))
edges = defaultdict(list)
for _ in range(n - 1):
    u, v = map(int, input().split())
    edges[u].append(v)
    edges[v].append(u)

vis = [False] * (n + 1)
p = [-1] * (n + 1)

def dfs(u):
    vis[u] = True
    mx = [a[u - 1]]
    for v in edges[u]:
        if not vis[v]:
            p[v] = u
            mx.append(dfs(v))
    return max(mx)

mx = dfs(1)
if mx > 0:
    print(mx)
else:
    print('Impossible')
