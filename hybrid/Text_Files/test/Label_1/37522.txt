t = int(input())
for _ in range(t):
    n, m, x = map(int, input().split())
    heights = list(map(int, input().split()))
    towers = [0] * m
    assignments = [0] * n
    for i in range(n):
        min_tower = towers.index(min(towers))
        towers[min_tower] += heights[i]
        assignments[i] = min_tower + 1
    if max(towers) - min(towers) > x:
        print("NO")
    else:
        print("YES")
        print(*assignments)
