import java.util.Arrays;
import java.util.Scanner;

public class Main {
    final static long LIM = 4_500_000_000L;
    static int n;
    static int[] x, y, p;
    static long[] d;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        x = new int[n];
        y = new int[n];
        p = new int[n];
        d = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        sc.close();

        precomp();
        Arrays.fill(p, -1);
        p[0] = 0;
        dfs(0);
        StringBuilder sb = new StringBuilder();
        sb.append(p[0] + 1);
        for (int i = 1; i < n; i++) sb.append(" ").append(p[i] + 1);
        System.out.println(sb);
    }

    static void precomp() {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                d[j] = Math.abs(x[i] - x[j]) + Math.abs(y[i] - y[j]);
            }
            Arrays.sort(d);
        }
    }

    static boolean dfs(int u) {
        long sum = 0;
        for (int i = 1; i < n; i++) {
            if (p[i] != -1) sum += d[p[i]];
        }
        if (sum > LIM) return false;
        if (u == n - 1) return true;
        for (int i = 0; i < n; i++) {
            if (p[i] != -1) continue;
            p[u + 1] = i;
            if (dfs(u + 1)) return true;
        }
        return false;
    }
}
