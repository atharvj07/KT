import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        StringBuilder sb = new StringBuilder();
        while (w > 0) {
            if (w % 3 == 1) {
                sb.append("+");
                w -= 4;
            } else if (w % 3 == 2) {
                sb.append("-");
                w -= 2;
            } else {
                sb.append("0");
                w -= 1;
            }
            w /= 3;
        }
        System.out.println(sb.reverse());
    }
}
