#include <iostream>
#include <cstring>

int main() {
  int n;
  std::cin >> n;
  char s[101];
  std::cin >> s;
  for (int i = 0; i < n; i++) {
    if (s[i] == 'g' && (i == 0 || s[i - 1] == 'o')) {
      int j = i + 1;
      while (j < n && s[j] == 'o') j++;
      while (j < n && s[j] == 'g') {
        j++;
      }
      if (j - i >= 2) {
        for (int k = i; k < i + 3; k++) std::cout << '*';
        i = j - 1;
        continue;
      }
    }
    std::cout << s[i];
  }
  std::cout << std::endl;
  return 0;
}
