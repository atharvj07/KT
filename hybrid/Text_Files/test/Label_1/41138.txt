from collections import deque, defaultdict

s = input()
cnt = defaultdict(int)
ans = 0
stack = deque()

for c in s:
    if c in "<[{(":
        stack.append((c, cnt[c]))
        cnt[c] += 1
    else:
        if not stack:
            ans += 1
        else:
            top = stack.pop()
            if (c == ')' and top[0] != '(') or (c == ']' and top[0] != '[') or (c == '}' and top[0] != '{') or (c == '>' and top[0] != '<'):
                ans += 1
            else:
                if top[1] < cnt[c]:
                    ans += 1
print('Impossible' if stack else ans)
