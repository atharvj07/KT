import sys
input=sys.stdin.readline

n,m,s,t=map(int,input().split())
d=[99]*(n+1)
d[s]=0
adj=[[]]
for i in range(n+1):adj.append([])
for _ in range(m):
    a,b=map(int,input().split())
    adj[a].append(b)
    adj[b].append(a)
q=[]
q.append(s)
while q:
    x=q.pop(0)
    for y in adj[x]:
        if d[y]==99:
            d[y]=d[x]+1
            q.append(y)
c=0
for x in adj[s]:
    if d[t]>d[x]:c+=n-1-len(adj[x])
for x in adj[t]:
    if d[s]>d[x]:c+=n-1-len(adj[x])
print(c)
