R, C = map(int, input().split())
pasture = [list(input()) for _ in range(R)]

def is_protectable():
    for i in range(R):
        for j in range(C):
            if pasture[i][j] == 'S':
                for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
                    nx, ny = i + dx, j + dy
                    if 0 <= nx < R and 0 <= ny < C:
                        if pasture[nx][ny] == 'W':
                            return "No"
                        if pasture[nx][ny] == '.':
                            pasture[nx][ny] = 'D'
    return "Yes"

result = is_protectable()
print(result)
if result == "Yes":
    for row in pasture:
        print(''.join(row))
