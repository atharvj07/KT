import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        int[] cnta = new int[1000001];
        int[] cntb = new int[1000001];
        for (int i = 0; i < n; i++) {
            cnta[a[i]]++;
            cntb[b[i]]++;
        }
        int ans = n;
        for (int i = 0; i < 1000001; i++) {
            int sum = cnta[i] + cntb[i];
            ans = Math.min(ans, n - sum);
        }
        if (ans == n) {
            ans = -1;
        }
        System.out.println(ans);
    }
}
