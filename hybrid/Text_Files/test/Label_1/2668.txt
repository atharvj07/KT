#include <bits/stdc++.h>
using namespace std;

int n, k, a, b, l, av, ans = 1e9;
int arr[31], fr[31][100001];

void dfs(int pos, int sum) {
  if (pos > n) return;
  if (pos == n) {
    ans = min(ans, sum);
    return;
  }
  int sum2 = sum;
  for (int i = 1; i <= l; i++) {
    if (!fr[pos + 1][i]) {
      fr[pos + 1][i] = 1;
      sum2 += b * av * l;
      dfs(pos + 1, sum2);
      sum2 -= b * av * l;
      fr[pos + 1][i] = 0;
      dfs(pos + 1, sum);
      break;
    }
  }
}

int main() {
  cin >> n >> k >> a >> b;
  l = (1 << n);
  for (int i = 0; i < k; i++) {
    int x;
    cin >> x;
    arr[++av] = x;
  }
  dfs(0, 0);
  cout << ans << '\n';
  return 0;
}
