import sys
input = sys.stdin.readline

mod = 998244353

n = int(input())
a = [int(x) for x in input().split()]

dp = [0] * (n + 1)
dp[n] = 1
for i in range(n - 1, -1, -1):
  if a[i] != -1:
    dp[i] = dp[i + 1]
    continue
  low, high = 1, 200
  if a[i + 1] != -1:
    low = a[i + 1]
  if a[i - 1] != -1:
    high = a[i - 1]
  dp[i] = (dp[i + 1] * (high - low + 1)) % mod

print(dp[0])
