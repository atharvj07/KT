import java.util.*;

class Main {
  private static int solve(int n, int[] a) {
    int ans = 0;
    int[] prefix = new int[n + 1];
    for (int i = 1; i <= n; i++) {
      prefix[i] = prefix[i - 1] + a[i];
    }
    int minPrefix = 0;
    for (int i = 1; i <= n; i++) {
      if (prefix[i] < minPrefix) {
        minPrefix = prefix[i];
      }
    }
    for (int i = 1; i <= n; i++) {
      ans += prefix[i] - minPrefix;
    }
    return ans / 2;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int[] a = new int[n + 1];
    for (int i = 1; i <= n; i++) {
      a[i] = in.nextInt();
    }
    System.out.println(solve(n, a));
  }
}
