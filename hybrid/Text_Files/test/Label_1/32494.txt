import java.util.Scanner;

public class Xorangles {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Precompute the number of non-degenerate triangles for each side length
        int[] triangles = new int[n + 1];
        for (int a = 1; a <= n; a++) {
            for (int b = a; b <= n; b++) {
                for (int c = b; c <= n; c++) {
                    if (isTriangle(a, b, c)) {
                        triangles[a]++;
                    }
                }
            }
        }

        int count = 0;
        for (int a = 1; a <= n; a++) {
            for (int b = a; b <= n; b++) {
                int c = a ^ b; // Calculate c using xor
                if (c > n || c < b) {
                    continue;
                }
                count += triangles[a] * triangles[b] * triangles[c];
            }
        }

        System.out.println(count);
    }

    // Check if the given side lengths form a non-degenerate triangle
    private static boolean isTriangle(int a, int b, int c) {
        return a + b > c && b + c > a && c + a > b;
    }
}
