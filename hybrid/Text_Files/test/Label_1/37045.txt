for _ in range(int(input())):
    n, s = map(int, input().split())
    a = list(map(int, input().split()))
    b = [0] * (n + 1)
    for i in range(n):
        if i > 0:
            b[i] = b[i - 1]
        b[i] += a[i]
    ans = 0
    cur = 0
    for i in range(n):
        idx = i
        while idx < n and cur + a[idx] <= s:
            cur += a[idx]
            idx += 1
        ans = max(ans, idx - i + 1)
        cur -= a[i]
    print(ans)
