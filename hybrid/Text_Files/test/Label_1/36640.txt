from collections import Counter

n, k = map(int, input().split())
a = list(map(int, input().split()))
q = int(input())
for i in range(q):
  x = int(input())
  cnt = Counter()
  for ai in a:
    if x < ai:
      break
    while x >= ai and cnt[ai] < k:
      x -= ai
      cnt[ai] += 1
  if x == 0:
    print(sum(cnt.values()))
  else:
    print(-1)
