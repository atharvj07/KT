import java.io.IOException;
import java.util.*;

public class Main {

    static List<Integer>[] forward, backward;
    static boolean[] visited;

    static ArrayList<String> convertToGraph(int n, int m, int[] x, int[] y) {
        forward = new ArrayList[n + 1];
        backward = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            forward[i] = new ArrayList<>();
            backward[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int a = x[i], b = y[i];
            forward[a].add(b);
            backward[b].add(a);
        }
        return null;
    }

    static boolean can(int[] x, int[] y, int k) {
        visited = new boolean[x.length];
        for (int i = 0; i < k; i++) {
            if (x[i] == y[i]) continue;
            if (!dfs(x[i], y[i], true)) return false;
            if (!dfs(y[i], x[i], false)) return false;
        }
        return true;
    }

    static boolean dfs(int cur, int des, boolean direction) {
        if (cur == des) return true;
        visited[cur] = true;
        if (direction) {
            for (int next : forward[cur]) {
                if (!visited[next] && dfs(next, des, direction)) return true;
            }
        } else {
            for (int next : backward[cur]) {
                if (!visited[next] && dfs(next, des, direction)) return true;
            }
        }
        return false;
    }

    public static void main(String[] args) throws IOException {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] x = new int[m], y = new int[m];
        for (int i = 0; i < m; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }
        int k = sc.nextInt();
        int[] a = new int[k], b = new int[k];
        for (int i = 0; i < k; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        convertToGraph(n, m, x, y);
        if (!can(a, b, k)) {
            System.out.println("No");
            return;
        }
        System.out.println("Yes");
        for (int i = 1; i <= n; i++) {
            List<Integer> adj;
            if (forward[i].size() == 0) adj = backward[i];
            else adj = forward[i];
            for (int next : adj) {
                System.out.println(i + " " + next);
            }
        }
    }
}
