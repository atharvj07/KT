n, k, q = map(int, input().split())
t = list(map(int, input().split()))

online = set()
displayed = []

for _ in range(q):
    type_i, id_i = map(int, input().split())
    if type_i == 1:
        online.add(id_i)
        displayed.append((t[id_i-1], id_i))
        displayed.sort(reverse=True)
        if len(displayed) > k:
            displayed.pop()
    else:
        if id_i in online and any(d[1] == id_i for d in displayed[:k]):
            print("YES")
        else:
            print("NO")
