#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int open = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == ')') {
                if (open > 0) {
                    open--;
                } else {
                    open++;
                }
            } else {
                open++;
            }
        }
        cout << open << endl;
    }
}
