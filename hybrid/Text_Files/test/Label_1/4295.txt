#include<vector>
#include<unordered_map>
#include<iostream>
using namespace std;
int main(){
    int n;
    cin>>n;
    vector<int> p(n);
    vector<int> q(n);
    unordered_map<int,int> pos;
    for(int i=0;i<n;i++){
        cin>>p[i];
        p[i]--;
        pos[p[i]]=i;
    }
    for(int i=0;i<n;i++){
        q[i] = pos[p[i]];
    }
    for(int i=0;i<n;i++){
        cout<<q[i]+1<<" ";
    }
}
