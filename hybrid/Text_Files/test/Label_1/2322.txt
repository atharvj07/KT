#include <bits/stdc++.h>
#define endl "\n"
using namespace std;
typedef long long ll;
const int mod = 998244353;
ll p[150005], q[150005];
ll ans = 0;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    ll n, x;
    cin >> n;
    for (ll i = 0; i < 2 * n; i++) { cin >> x; if (i < n) p[i] = x; else q[i - n] = x; }
    sort(p, p + n);
    sort(q, q + n, greater<ll>());
    for (ll i = 0; i < n; i++) {
        ans = (ans + abs(p[i] - q[i])) % mod;
    }
    cout << ans << endl;
    return 0;
}
