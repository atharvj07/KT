import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] rooms = new int[n][3];
        for (int i = 0; i < n; i++) {
            rooms[i][0] = sc.nextInt();
            rooms[i][1] = sc.nextInt();
            rooms[i][2] = sc.nextInt();
        }
        int m = sc.nextInt();
        int[][] wallpapers = new int[m][3];
        for (int i = 0; i < m; i++) {
            wallpapers[i][0] = sc.nextInt();
            wallpapers[i][1] = sc.nextInt();
            wallpapers[i][2] = sc.nextInt();
        }
        int[][][] dp = new int[n][m][2];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                Arrays.fill(dp[i][j], Integer.MAX_VALUE);
            }
        }
        for (int j = 0; j < m; j++) {
            int perimeter = (rooms[0][0] + rooms[0][1]) * 2 + rooms[0][2];
            int cost = perimeter / wallpapers[j][0] * wallpapers[j][2];
            if (cost % wallpapers[j][2] > 0) cost += wallpapers[j][2];
            cost += wallpapers[j][2];
            dp[0][j][0] = cost;
            dp[0][j][1] = cost;
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = 0; k < m; k++) {
                    int perimeter = (rooms[i][0] + rooms[i][1]) * 2 + rooms[i][2];
                    int cost = perimeter / wallpapers[j][0] * wallpapers[j][2];
                    if (cost % wallpapers[j][2] > 0) cost += wallpapers[j][2];
                    cost += wallpapers[j][2];
                    dp[i][j][0] = Math.min(dp[i][j][0], dp[i - 1][k][0] + cost);
                    dp[i][j][1] = Math.min(dp[i][j][1], dp[i - 1][k][1] + cost);
                    cost = perimeter / wallpapers[j][1] * wallpapers[j][2];
                    if (cost % wallpapers[j][2] > 0) cost += wallpapers[j][2];
                    cost += wallpapers[j][2];
                    dp[i][j][0] = Math.min(dp[i][j][0], dp[i - 1][k][0] + cost);
                    dp[i][j][1] = Math.min(dp[i][j][1], dp[i - 1][k][1] + cost);
                }
            }
        }
        int ans = Integer.MAX_VALUE;
        for (int j = 0; j < m; j++) {
            ans = Math.min(ans, dp[n - 1][j][1]);
        }
        System.out.println(ans);
    }
}
