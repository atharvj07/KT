import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
    private static final int MAX_X = 30;
    private static final int MOD = 1000000007;
    private static int n;
    private static int m;
    private static int s;
    private static int t;
    private static long[][] paths;
    private static ArrayList<Integer>[] edges;
    private static int[] parent;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));

        String[] firstLine = in.readLine().split(" ");
        n = Integer.parseInt(firstLine[0]);
        m = Integer.parseInt(firstLine[1]);
        paths = new long[n + 1][MAX_X + 1];
        for (long[] path : paths) {
            Arrays.fill(path, -1);
        }
        edges = new ArrayList[n + 1];
        for (int i = 0; i < n + 1; i++) {
            edges[i] = new ArrayList<>();
        }
        parent = new int[n + 1];
        Arrays.fill(parent, -1);
        for (int i = 0; i < m; i++) {
            String[] line = in.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            int x = Integer.parseInt(line[2]);
            edges[u].add(v);
            edges[v].add(u);
            paths[u][x] = 1 << x;
            paths[v][x] = 1 << x;
        }
        String[] lastLine = in.readLine().split(" ");
        s = Integer.parseInt(lastLine[0]);
        t = Integer.parseInt(lastLine[1]);
        long result = dijkstra(s);
        if (result == -1) {
            System.out.println(-1);
        } else {
            long remainder = result % MOD;
            System.out.println(remainder);
            int count = 0;
            int current = t;
            while (current != s) {
                count++;
                current = parent[current];
            }
            System.out.println(count + 1);
            current = t;
            List<Integer> path = new ArrayList<>();
            while (current != s) {
                path.add(current);
                current = parent[current];
            }
            path.add(s);
            for (int i = path.size() - 1; i >= 0; i--) {
                System.out.print(path.get(i) + " ");
            }
            System.out.println();
        }
    }

    private static long dijkstra(int source) {
        long[] distance = new long[n + 1];
        Arrays.fill(distance, Long.MAX_VALUE / 2);
        distance[source] = 0;
        for (int i = 0; i < MAX_X; i++) {
            boolean updated = false;
            for (int j = 1; j <= n; j++) {
                if (distance[j] != Long.MAX_VALUE / 2) {
                    for (int neighbor : edges[j]) {
                        for (int x = 0; x <= MAX_X; x++) {
                            if (paths[j][x] != -1 && paths[neighbor][x] != -1) {
                                long newDistance = distance[j] + paths[j][x] + paths[neighbor][x];
                                if (newDistance < distance[neighbor]) {
                                    distance[neighbor] = newDistance;
                                    updated = true;
                                    parent[neighbor] = j;
                                }
                            }
                        }
                    }
                }
            }
            if (!updated) {
                break;
            }
        }
        return distance[t] == Long.MAX_VALUE / 2 ? -1 : distance[t];
    }
}
