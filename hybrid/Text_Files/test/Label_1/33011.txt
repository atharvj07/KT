import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        long[] w = new long[4];
        long[] b1 = new long[4];
        long[] b2 = new long[4];
        for (int i = 0; i < 4; ++i)
            w[i] = in.nextLong();
        for (int i = 0; i < 4; ++i)
            b1[i] = in.nextLong();
        for (int i = 0; i < 4; ++i)
            b2[i] = in.nextLong();
        boolean ans = w[0] < b1[2] || w[1] < b1[3] || w[2] > b1[0] || w[3] > b1[1];
        ans |= w[0] < b2[2] || w[1] < b2[3] || w[2] > b2[0] || w[3] > b2[1];
        System.out.println(ans ? "YES" : "NO");
    }
}
