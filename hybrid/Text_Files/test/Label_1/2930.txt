#include <iostream>

using namespace std;

int n;
int a[100001];

int main() {
  cin >> n;
  if (n % 2 == 0) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  a[1] = n;
  int tmp = 2;
  for (int i = 2; i <= n; i++) {
    if (i % 2 == 0) {
      a[i] = tmp;
      tmp++;
    } else {
      a[i] = n - tmp + 1;
      tmp++;
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << a[i] << endl;
  }
  return 0;
}
