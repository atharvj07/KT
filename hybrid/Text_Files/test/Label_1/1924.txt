#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll gcd(ll a, ll b){
    if(b==0){
        return a;
    }else{
        return gcd(b, a%b);
    }
}

int main(){
    ll x, y;
    cin>>x>>y;
    if(gcd(x, y)!=1)
        cout<<"Impossible"<<endl;
    else{
        ll g;
        g = y/gcd(x, y);
        string s;
        for(int i=0; i<g; i++){
            s+="B";
        }
        for(int i=0; i<x/g; i++){
            s+="A";
        }
        cout<<s<<endl;
    }
    return 0;
}
