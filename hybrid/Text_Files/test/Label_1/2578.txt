#include <bits/stdc++.h>
using namespace std;

int main() {
  string a, b;
  cin >> a >> b;
  reverse(a.begin(), a.end());
  reverse(b.begin(), b.end());
  int n = max(a.size(), b.size());
  for (int i = 0; i < n; i++) {
    if (i < a.size()) {
      if (a[i] > b[i]) return cout << ">", 0;
      if (a[i] < b[i]) return cout << "<", 0;
    } else if (i < b.size())
      return cout << "<", 0;
  }
  return cout << "=", 0;
}
