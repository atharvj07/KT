import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            long n = sc.nextLong();
            long m = sc.nextLong();
            if (n == 0 && m == 0) {
                break;
            }
            long[] nums = new long[(int) (n + 1)];
            for (int i = 0; i < m; i++) {
                long num = sc.nextLong();
                for (long j = num; j <= n; j += num) {
                    nums[(int) j] = -1;
                }
            }
            double sum = 0;
            for (int i = 1; i <= n; i++) {
                if (nums[i] > 0) {
                    sum += i;
                }
            }
            System.out.println(sum / n);
        }
        sc.close();
    }
}
