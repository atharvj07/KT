#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> arr(n);
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    long long sum = 0;
    int neg = 0;
    for (int i = 0; i < n; i++) {
        sum += abs(arr[i]);
        if (arr[i] < 0) neg++;
    }
    if (neg <= k)
        sum += 2 * abs(arr[0]);
    else if (k % 2 == 1)
        sum += 2 * abs(arr[0]);
    else {
        if (neg == n) {
            sum -= 2 * abs(arr[n - 1]);
        } else {
            sum += 2 * abs(arr[0]);
        }
    }
    cout << sum;
    return 0;
}
