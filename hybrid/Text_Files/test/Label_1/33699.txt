import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

class Main {

    static long n, m, s;
    static List<Edge>[] adj;
    static long[][] dp;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Long.parseLong(st.nextToken());
        m = Long.parseLong(st.nextToken());
        adj = new List[(int) (n + 1)];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        s = Long.parseLong(br.readLine());

        for (long i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            long from = Long.parseLong(st.nextToken());
            long to = Long.parseLong(st.nextToken());
            long weight = Long.parseLong(st.nextToken());
            adj[(int) from].add(new Edge(to, weight));
        }

        dp = new long[(int) (n + 1)][(int) (m + 1)];
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= m; j++) {
                dp[i][j] = -1;
            }
        }

        long ans = dfs(s, 0);
        System.out.println(ans);
    }

    static long dfs(long u, long cnt) {
        if (dp[(int) u][(int) cnt] != -1) {
            return dp[(int) u][(int) cnt];
        }

        long ans = 0;
        for (Edge edge : adj[(int) u]) {
            long v = edge.to;
            long w = edge.weight;
            if (w > cnt) {
                ans = Math.max(ans, dfs(v, cnt + 1) + w - cnt);
            }
        }

        dp[(int) u][(int) cnt] = ans;
        return ans;
    }

    static class Edge {
        long to;
        long weight;

        Edge(long to, long weight) {
            this.to = to;
            this.weight = weight;
        }
    }
}
