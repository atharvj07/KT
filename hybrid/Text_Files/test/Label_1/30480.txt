import java.util.Arrays;
import java.util.Scanner;

class Main {
    public static void main(String[] x) throws Exception {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextLong();
            b[i] = in.nextLong();
        }
        in.close();
        Arrays.sort(a);
        Arrays.sort(b);
        long res = n;
        for (int i = 0, j = n - 1; i < n && j >= 0; i++, j--) {
            if (a[i] != b[j]) {
                res = i;
                break;
            }
        }
        for (int i = 0, j = n - 1; i < n && j >= 0; i++, j--) {
            if (a[j] != b[i]) {
                res = Math.min(res, i);
                break;
            }
        }
        if (res == n) {
            System.out.println(-1);
        } else {
            System.out.println(res);
        }
    }
}
