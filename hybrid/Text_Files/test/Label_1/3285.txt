#include<bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int nA, nB, k, m;
    cin >> nA >> nB >> k >> m;
    if (nA < k || nB < m) {
        cout << "NO" << endl;
        return 0;
    }
    int i = 0, j = 0;
    while (i < k && j < m) {
        int a, b;
        cin >> a >> b;
        if (a >= b) {
            cout << "NO" << endl;
            return 0;
        }
        i++;
        j++;
    }
    cout << "YES" << endl;
    return 0;
}
