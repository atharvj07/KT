#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int main() {
  int n, m, a[2001], b[2001];
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  cin >> m;
  for (int i = 0; i < m; i++) cin >> b[i];
  sort(a, a + n);
  sort(b, b + m);
  long long int sa = 0, sb = 0;
  for (int i = 0; i < n; i++) sa += a[i];
  for (int i = 0; i < m; i++) sb += b[i];
  long long int ans = abs(sa - sb);
  pair<int, pair<int, int>> sw[3];
  if (n <= m) {
    for (int i = 0; i <= 1; i++) {
      for (int j = i; j <= 1; j++) {
        if (sa + b[j] - a[i] >= sb) {
          long long int v = sa + b[j] - a[i] - sb;
          if (v < ans || (ans == v && i + j < sw[0].first + sw[0].second.second)) {
            ans = v;
            sw[0] = {i + j, {i, j}};
          }
        }
        if (sb + a[i] - b[j] >= sa) {
          long long int v = sb + a[i] - b[j] - sa;
          if (v < ans || (ans == v && i + j < sw[0].first + sw[0].second.second)) {
            ans = v;
            sw[0] = {i + j, {j, i}};
          }
        }
      }
      sa += a[i];
      sb -= b[i];
    }
  } else {
    for (int i = 0; i <= 1; i++) {
      for (int j = i; j <= 1; j++) {
        if (sb + a[j] - b[i] >= sa) {
          long long int v = sb + a[j] - b[i] - sa;
          if (v < ans || (ans == v && i + j < sw[0].first + sw[0].second.second)) {
            ans = v;
            sw[0] = {i + j, {i, j}};
          }
        }
        if (sa + b[i] - a[j] >= sb) {
          long long int v = sa + b[i] - a[j] - sb;
          if (v < ans || (ans == v && i + j < sw[0].first + sw[0].second.second)) {
            ans = v;
            sw[0] = {i + j, {j, i}};
          }
        }
      }
      sa -= a[i];
      sb += b[i];
    }
  }
  cout << ans << endl;
  if (!ans)
    cout << 0 << endl;
  else if (sw[0].first == 1)
    cout << 1 << endl << sw[0].second.first + 1 << " " << sw[0].second.second + 1 << endl;
  else {
    for (int i = 1; i < 3; i++) {
      if (sw[i].first > 0) {
        cout << 2 << endl;
        cout << sw[i - 1].second.first + 1 << " " << sw[i - 1].second.second + 1 << endl;
        cout << sw[i].second.first + 1 << " " << sw[i].second.second + 1 << endl;
        break;
      }
    }
  }
}
