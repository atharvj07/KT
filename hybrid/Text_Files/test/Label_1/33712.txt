import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int[] minutes = new int[n];
        for (int i = 0; i < n; i++) {
            minutes[i] = scanner.nextInt();
        }

        int lastInterestingMinute = 0;
        int minutesWatched = 0;
        for (int i = 0; i < n; i++) {
            if (minutes[i] - lastInterestingMinute >= 15) {
                break;
            }
            lastInterestingMinute = minutes[i];
            minutesWatched++;
        }

        System.out.println(minutesWatched);
    }
}
