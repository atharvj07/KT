import functools

n, m = map(int, input().split())
a = [list(map(int, input().split())) for _ in range(n)]
or_list = functools.reduce(lambda x, y: [x[i] | y[i] for i in range(m)], a)
if not any(or_list):
    print('NIE')
else:
    print('TAK')
    print(*[i + 1 for i, o in enumerate(or_list) if o != 0])
