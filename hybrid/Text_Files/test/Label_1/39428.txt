def solve_rebus(rebus):
    parts = rebus.split()
    n = int(parts[-1])
    min_val, max_val = 0, 0
    sign = 1
    for part in parts[:-2]:
        if part == "+":
            sign = 1
        elif part == "-":
            sign = -1
        else:
            min_val += sign * 1
            max_val += sign * n
    if min_val <= n <= max_val:
        print("Possible")
        diff = n - min_val
        res = []
        sign = 1
        for part in parts[:-2]:
            if part in "+-":
                res.append(part)
                if part == '+':
                    sign = 1
                else:
                    sign = -1
            else:
                add = min(n-1, diff) if sign == 1 else 0
                res.append(str(1 + add))
                diff -= add
        res.append("=")
        res.append(str(n))
        print(" ".join(res))
    else:
        print("Impossible")

rebus = input()
solve_rebus(rebus)
