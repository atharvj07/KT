import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), x = input.nextInt(), y = input.nextInt();
        String num = input.next();
        int ans = 0;
        for (int i = y; i < x; i++) {
            if (num.charAt(i) != '0') {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
