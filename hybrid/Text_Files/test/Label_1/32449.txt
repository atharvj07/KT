import java.util.*;

public class Main {
    static class Point implements Comparable<Point> {
        int x, y;
        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
        @Override
        public int compareTo(Point that) {
            if (this.x != that.x) {
                return this.x - that.x;
            } else {
                return this.y - that.y;
            }
        }
    }
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();
        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            Point[] points = new Point[n];
            for (int j = 0; j < n; j++) {
                int x = scanner.nextInt();
                int y = scanner.nextInt();
                points[j] = new Point(x, y);
            }
            Arrays.sort(points);
            int max_x = points[n - 1].x;
            int max_y = points[n - 1].y;
            boolean[][] visited = new boolean[max_x + 1][max_y + 1];
            for (Point point : points) {
                if (visited[point.x][point.y]) {
                    System.out.println("NO");
                    continue;
                }
                StringBuilder path = new StringBuilder();
                if (!dfs(point, path, visited)) {
                    System.out.println("NO");
                    continue;
                }
                System.out.println("YES");
                System.out.println(path);
            }
        }
    }
    private static boolean dfs(Point point, StringBuilder path, boolean[][] visited) {
        if (point.x == 0 && point.y == 0) {
            return true;
        }
        visited[point.x][point.y] = true;
        if (point.x > 0 && !visited[point.x - 1][point.y]) {
            path.append('R');
            if (dfs(new Point(point.x - 1, point.y), path, visited)) {
                return true;
            }
            path.setLength(path.length() - 1);
        }
        if (point.y > 0 && !visited[point.x][point.y - 1]) {
            path.append('U');
            if (dfs(new Point(point.x, point.y - 1), path, visited)) {
                return true;
            }
            path.setLength(path.length() - 1);
        }
        return false;
    }
}
