#include <vector>
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];
    vector<int> ans = {1};
    int pos = -1, j = 0, cnt = 0;
    for (int i = 0; i < n; ++i) {
        if (pos != -1 && a[i] == pos) {
            cnt++;
        } else if (pos == -1 && a[i] != i + 1) {
            pos = i + 1;
            cnt = 1;
        }
        if (pos == i + 1) j += cnt;
        ans.push_back(j + 1);
    }
    for (int x : ans) cout << x << " ";
    cout << endl;
}
