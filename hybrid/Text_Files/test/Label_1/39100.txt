md = 1000000009

def binpow(x, p):
    if p == 0:
        return 1
    if p == 1:
        return x
    val = binpow(x, int(p / 2))
    val = (val * val) % md
    if p % 2 != 0:
        val = (val * x) % md
    return val % md

n, m = map(int, input().split())
print((binpow(2, n * m) - binpow(2, n - 1)) % md)
