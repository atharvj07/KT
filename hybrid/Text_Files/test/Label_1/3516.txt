#include <iostream>
#include <vector>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, x;
    cin >> n >> x;
    vector<int> a(n);
    int odd = 0, sum = 0;
    for (int &i : a) {
      cin >> i;
      sum += i;
      odd += (i & 1);
    }
    if (odd == 0)
      cout << "No" << endl;
    else if (odd >= x)
      cout << "Yes" << endl;
    else if (sum & 1)
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
  return 0;
}
