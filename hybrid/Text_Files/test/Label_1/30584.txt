import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        // dp[i] stores the maximum possible value of the last slime after considering the first i slimes
        long[] dp = new long[n + 1];
        dp[0] = 0;
        dp[1] = a[0];
        for (int i = 2; i <= n; i++) {
            dp[i] = Math.max(dp[i - 1] + a[i - 1], dp[i - 2] + a[i - 1]);
        }
        System.out.println(dp[n]);
    }
}
