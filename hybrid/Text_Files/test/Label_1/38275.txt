s=input()
t=input()
if s==t: 
    print(s)
    exit()
n=len(s)
x=[ord(i) for i in s]
y=[ord(i) for i in t]
c=0
for i in range(n):
    c+=(x[i]==y[i])
if c==0: ans='1'*n
else: ans=''
for i in range(n):
    if x[i]==y[i]:
        ans+=('1' if x[i]=='0' else '0')
    else: 
        ans+=('0' if c==n/2 else '1')
print(ans)
