import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), ans = 0, cur = 0;
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt(), y = 2 * x;
            if (y > cur) ans += (y - cur) >> 1;
            cur = y;
        }
        System.out.println(ans);
    }
}
