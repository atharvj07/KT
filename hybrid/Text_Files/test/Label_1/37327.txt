n, m = map(int, input().split())
grid = [list(input()) for _ in range(n)]
ans = "Yes"
for i in range(n):
    for j in range(m):
        if grid[i][j] == '#' and (i != 0 and grid[i-1][j] == '#') and (j != 0 and grid[i][j-1] == '#'):
            ans = "No"
            break
    if ans == "No":
        break
print(ans)
