import sys
n = int(input())
a = list(map(int, input().split()))
q = int(input())
ans = [0] * q
g = [0] * (n + 1)
for i in a:
    for j in range(i, n + 1, i):
        g[j] += 1
for i in range(q):
    x = int(input())
    for j in range(1, n + 1):
        ans[i] += 1 if g[j] == x else 0
for i in ans:
    print(i)
