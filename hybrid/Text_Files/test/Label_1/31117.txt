import java.util.*;

public class DancingParty {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int w = scanner.nextInt();

        int[] weights = new int[n];
        int[] beauties = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = scanner.nextInt();
            beauties[i] = scanner.nextInt();
        }

        List<List<Integer>> friends = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            friends.add(Arrays.asList(x, y));
        }

        System.out.println(getMaxBeauty(weights, beauties, friends, w));
    }

    static int getMaxBeauty(int[] weights, int[] beauties, List<List<Integer>> friends, int maxWeight) {
        int n = weights.length;
        int[][] dp = new int[n + 1][maxWeight + 1];

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= maxWeight; j++) {
                dp[i][j] = dp[i - 1][j];
                for (List<Integer> friend : friends) {
                    if (friend.get(0) == i - 1 && weights[friend.get(1)] <= j) {
                        dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - weights[friend.get(1)]] + beauties[friend.get(1)]);
                    }
                }
            }
        }

        return dp[n][maxWeight];
    }
}
