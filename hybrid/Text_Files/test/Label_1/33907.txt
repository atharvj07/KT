import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] matrix = new int[n][n];
        for (int i = 0; i < n; i++) {
            String line = sc.next();
            for (int j = 0; j < n; j++) {
                if (line.charAt(j) == '#') {
                    matrix[i][j] = 1;
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (matrix[i][j] == 1) {
                    int maxRow = 0;
                    int maxCol = 0;
                    for (int k = i; k < n; k++) {
                        if (matrix[k][j] == 0) {
                            break;
                        }
                        maxRow = Math.max(maxRow, k - i + 1);
                    }
                    for (int k = j; k < n; k++) {
                        if (matrix[i][k] == 0) {
                            break;
                        }
                        maxCol = Math.max(maxCol, k - j + 1);
                    }
                    ans += Math.max(maxRow, maxCol);
                    for (int k = i; k < i + maxRow; k++) {
                        for (int l = j; l < j + maxCol; l++) {
                            matrix[k][l] = 0;
                        }
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
