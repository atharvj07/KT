import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Set<String> s = new HashSet<String>();
        for (int i = 0; i < n; i++) {
            s.add(in.next());
        }
        int minLen = Integer.MAX_VALUE;
        String minStr = null;
        for (String str : s) {
            if (isGood(s, str)) {
                if (str.length() < minLen) {
                    minLen = str.length();
                    minStr = str;
                } else if (str.length() == minLen) {
                    minStr = minStr.compareTo(str) < 0 ? minStr : str;
                }
            }
        }
        if (minStr != null) {
            System.out.println(minStr);
        } else {
            System.out.println("NO");
        }
    }

    private static boolean isGood(Set<String> s, String str) {
        for (String sub : s) {
            int cnt = 0;
            for (int i = 0; i < str.length(); i++) {
                if (str.startsWith(sub, i)) {
                    cnt++;
                }
            }
            boolean found = false;
            for (String str2 : s) {
                if (str.equals(str2)) {
                    continue;
                }
                int cnt2 = 0;
                for (int i = 0; i < str2.length(); i++) {
                    if (str2.startsWith(sub, i)) {
                        cnt2++;
                    }
                }
                if (cnt2 > cnt) {
                    found = true;
                    break;
                }
            }
            if (found) {
                return false;
            }
        }
        return true;
    }
}
