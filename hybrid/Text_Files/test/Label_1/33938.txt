import java.util.*;

public class Main {
    private static List<Integer>[] adj;
    private static int n, k;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        int m = sc.nextInt();
        k = sc.nextInt();
        adj = new List[n + 1];
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            adj[a].add(b);
            adj[b].add(a);
        }
        System.out.println(dfs(1, -1, 0));
    }

    private static int dfs(int u, int p, int depth) {
        int ans = 0;
        if (depth == 0) {
            ans = 1;
        }
        for (int v : adj[u]) {
            if (v == p) {
                continue;
            }
            ans = (ans + dfs(v, u, depth + 1)) % k;
        }
        return ans;
    }
}
