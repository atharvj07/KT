import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] pos = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            pos[i] = i;
        }
        while (q-- > 0) {
            int type = sc.nextInt();
            if (type == 1) {
                int x = sc.nextInt();
                for (int i = 1; i <= n; i++) {
                    pos[i] = (pos[i] + x) % n + 1;
                }
            } else {
                for (int i = 1; i <= n; i += 2) {
                    int temp = pos[i];
                    pos[i] = pos[i + 1];
                    pos[i + 1] = temp;
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.print(pos[i] + " ");
        }
    }
}
