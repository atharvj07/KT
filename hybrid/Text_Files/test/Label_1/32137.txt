import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
  public static void main(String[] args) throws Exception {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    Map<Integer, Integer> count = new HashMap<>();
    List<Integer> lst = new ArrayList<>();
    int l = 0;
    for (String s : br.readLine().split(" ")) {
      int x = Integer.parseInt(s);
      lst.add(x);
      if (!count.containsKey(x)) {
        count.put(x, l++);
      }
    }
    while (!lst.isEmpty()) {
      int min = Integer.MAX_VALUE, minp = -1;
      Map<Integer, Integer> pcount = new HashMap<>();
      for (int i = 0, j = lst.size() - 1; i < j && i >= 0 && j < lst.size(); i++, j--) {
        if (lst.get(i).equals(lst.get(j))) {
          int len = j - i + 1;
          if (len < min || (min == len && pcount.getOrDefault(lst.get(i), 0) > count.getOrDefault(lst.get(i), 0))) {
            min = len;
            minp = i;
          }
          pcount.put(lst.get(i), pcount.getOrDefault(lst.get(i), 0) + 1);
        }
      }
      if (minp > -1) {
        for (int i = 0; i < minp; i++) {
          count.remove(lst.get(i));
          lst.remove(i);
          i--;
        }
      } else {
        break;
      }
    }
    System.out.println(lst.size());
    for (int x : lst) {
      System.out.print(x + " ");
    }
    System.out.println();
  }
}
