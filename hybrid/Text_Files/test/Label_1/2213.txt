#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        sort(a.begin(), a.end());
        int ans = 0;
        for (int i = 0; i < k; i++) {
            ans += a[i] / a[i + 1];
        }
        for (int i = k; i < n; i++) {
            ans += a[i];
        }
        cout << ans << endl;
    }
    return 0;
}
