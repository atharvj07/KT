import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int[] m = new int[5], w = new int[5], score = {500, 1000, 1500, 2000, 2500};
        int hs, hu, sum = 0;
        String[] s = scan.nextLine().split(" ");
        for (int i = 0; i < 5; i++) {
            m[i] = Integer.parseInt(s[i]);
            sum += score[i];
        }
        s = scan.nextLine().split(" ");
        for (int i = 0; i < 5; i++) {
            w[i] = Integer.parseInt(s[i]);
            sum -= (w[i] * score[i] / 250.0);
        }
        s = scan.nextLine().split(" ");
        hs = Integer.parseInt(s[0]);
        hu = Integer.parseInt(s[1]);
        sum += hs * 100 - hu * 50;
        System.out.println((int)sum);
    }
}
