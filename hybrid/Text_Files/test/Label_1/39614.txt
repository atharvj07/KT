import sys
import heapq

n, m, k = map(int, input().split())
adj = [[] for i in range(n + 1)]
for i in range(m):
    a, b, w = map(int, input().split())
    adj[a].append((b, w))
    adj[b].append((a, w))
if k == 0:
    print(-1)
    sys.exit()
storages = list(map(int, input().split()))
dist = [[1e9] * (k + 1) for i in range(n + 1)]
for storage in storages:
    q = [(0, storage)]
    dist[storage][storage] = 0
    while q:
        d, node = heapq.heappop(q)
        for neighbor, weight in adj[node]:
            nd = d + weight
            if nd < dist[neighbor][storage]:
                heapq.heappush(q, (nd, neighbor))
                dist[neighbor][storage] = nd
ans = 1e9
for i in range(1, n + 1):
    if i not in storages:
        for storage in storages:
            ans = min(ans, dist[i][storage])
print(-1 if ans == 1e9 else ans)
