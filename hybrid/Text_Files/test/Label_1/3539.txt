#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    int cnt[26] = {0};
    for (char c : s) {
      cnt[c - 'a']++;
    }
    int odd = 0;
    for (int i = 0; i < 26; i++) {
      if (cnt[i] & 1) {
        odd++;
      }
    }
    if (odd > 1) {
      cout << -1 << endl;
    } else {
      int ans = 0;
      for (int i = 0; i < 26; i++) {
        if (cnt[i] & 1) {
          ans += cnt[i] - 1;
        }
      }
      cout << ans << endl;
    }
  }
  return 0;
}
