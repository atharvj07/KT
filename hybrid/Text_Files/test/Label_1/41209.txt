MOD = 1000000007

def dfs(vertex, parent):
    ways = 1
    black_subtrees = 0
    for child in tree[vertex]:
        if child != parent:
            subtree_ways, is_black = dfs(child, vertex)
            ways = ways * subtree_ways % MOD
            if is_black:
                black_subtrees += 1
                ways = ways * 2 % MOD
    if colors[vertex] == 1:
        if black_subtrees:
            ways = ways * pow(2, black_subtrees, MOD) % MOD
            ways = ways * pow(2, MOD-2, MOD) % MOD
        return ways, True
    else:
        if black_subtrees > 1:
            return ways, True
        else:
            return ways, False

n = int(input().strip())
parents = list(map(int, input().strip().split()))
colors = list(map(int, input().strip().split()))

tree = [[] for _ in range(n)]
for i, parent in enumerate(parents):
    tree[i + 1].append(parent)
    tree[parent].append(i + 1)

result, _ = dfs(0, -1)
print(result)
