#include <bits/stdc++.h>
using namespace std;
int main() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    vector<int> x(a), y(b);
    for (int i = 0; i < a; i++) cin >> x[i];
    for (int i = 0; i < b; i++) cin >> y[i];
    cout << ((c <= a && d <= b && x[c - 1] < y[d - 1]) ? "YES\n" : "NO\n");
    return 0;
}
