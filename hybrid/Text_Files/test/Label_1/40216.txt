n = int(input())
a = list(map(int, input().split()))
mx = 0
for i in range(n):
    mx = max(mx, a[i])
cnt = 0
for i in range(n):
    if a[i] == mx:
        cnt += 1
l = -1
r = -1
for i in range(n):
    if a[i] == mx:
        if l == -1:
            l = i
        r = i
f = 1
for i in range(n):
    if a[i] != mx:
        if i - l < r - i:
            l = i
            f = 0
        if r - i < i - l:
            r = i
            f = 0
if f:
    print(l + 1, r + 1)
else:
    print(l + 1, r)
