import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int d = sc.nextInt();
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[][] lr = new int[d][2];
    int[][] tb = new int[d][2];
    for (int i = 0; i < d; i++) {
      int x1 = sc.nextInt() - 1;
      int y1 = sc.nextInt() - 1;
      int x2 = sc.nextInt() - 1;
      int y2 = sc.nextInt() - 1;
      lr[i][0] = x1;
      lr[i][1] = x2;
      tb[i][0] = y1;
      tb[i][1] = y2;
    }
    int cntl = sc.nextInt();
    int cntr = sc.nextInt();
    int cntt = sc.nextInt();
    int cntb = sc.nextInt();
    for (int i = 0; i < d; i++) {
      int left = 0;
      int right = 0;
      int top = 0;
      int bottom = 0;
      for (int j = 0; j < d; j++) {
        if (i == j)
          continue;
        if (lr[j][0] <= lr[i][0] && lr[j][1] >= lr[i][0])
          left++;
        if (lr[j][0] <= lr[i][1] && lr[j][1] >= lr[i][1])
          right++;
        if (tb[j][0] <= tb[i][0] && tb[j][1] >= tb[i][0])
          top++;
        if (tb[j][0] <= tb[i][1] && tb[j][1] >= tb[i][1])
          bottom++;
      }
      if (left == cntl && right == cntr && top == cntt && bottom == cntb) {
        System.out.println(i + 1);
        return;
      }
    }
    System.out.println(-1);
  }
}
