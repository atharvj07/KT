import sys, bisect

def solve():
    n = int(input())
    a = [list(input()) for _ in range(n)]
    for i in range(n):
        for j in range(len(a[i])):
            if a[i][j] != '?':
                a[i][j] = int(a[i][j])
    b = [0] * n
    for i in range(n):
        b[i] = len(a[i]) - a[i].count('?')
    if b[0] > 0:
        c = a[0].copy()
        for i in range(1, n):
            if b[i] > 0:
                if a[i][b[i] - 1] != c[-1]:
                    return 'NO'
                c.extend(a[i][b[i]:])
            else:
                j = bisect.bisect_right(c, int(''.join(a[i])))
                if j == len(c):
                    c.append(int(''.join(a[i])))
                else:
                    c[j] = int(''.join(a[i]))
        return 'YES\n' + '\n'.join(''.join(map(str, x)).lstrip('0') for x in c)
    else:
        return 'NO'

print(solve())
