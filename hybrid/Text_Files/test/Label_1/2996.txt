#include <iostream>
typedef long long ll;
ll fastexp(ll b, ll p) {ll r = 1;while (p > 0) {if (p & 1) r = r * b % 1000000007;b = b * b % 1000000007;p >>= 1;}return r;}
int main() {
  ll n, k;
  std::cin >> n >> k;
  ll r = (fastexp(2, k) - 1) * fastexp(2, k - 1) % 1000000007;
  r = (r + 1000000007) % 1000000007;
  r = (r * fastexp(n, k)) % 1000000007;
  std::cout << r << std::endl;
}
