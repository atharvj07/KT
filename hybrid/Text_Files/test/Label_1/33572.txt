import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int k = sc.nextInt();
    int d = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    Arrays.sort(a);
    boolean ok = true;
    for (int i = 0; i < n - 1; i++) {
      if (a[i + 1] - a[i] > d) {
        ok = false;
      }
    }
    if (ok) {
      System.out.println("YES");
    } else {
      System.out.println("NO");
    }
  }
}
