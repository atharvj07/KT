#include <iostream>
#include <vector>
using namespace std;
typedef long long LL;
const int MAXN = 100005;
vector<int> a;
int n, k;

int main() {
    cin >> n >> k;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    LL ans = 0;
    int neg = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) {
            if (k > 0) {
                k--;
                a[i] = -a[i];
                neg++;
            }
        } else {
            break;
        }
    }
    if (neg % 2 == 1 && k > 0) {
        a[neg - 1] = -a[neg - 1];
    }
    for (int i = 0; i < n; i++) {
        ans += a[i];
    }
    cout << ans << endl;
    return 0;
}
