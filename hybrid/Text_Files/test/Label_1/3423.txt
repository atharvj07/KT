#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long n, x, y;
        cin >> n >> x >> y;
        long long min_place = max(x, y);
        long long max_place = n - min(x, y) + 1;
        cout << min_place << " " << max_place << endl;
    }
    return 0;
}
