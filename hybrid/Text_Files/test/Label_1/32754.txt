import java.util.*;

public class Main {
    static int n;
    static ArrayList<Integer>[] v;
    static boolean[][] visit;
    static int[] ans, idx;
    static int idx1, idx2;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        v = new ArrayList[n + 1];
        visit = new boolean[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            v[i] = new ArrayList<>();
        }
        ans = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            ans[i] = sc.nextInt();
        }
        idx = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            idx[ans[i]] = i;
        }
        int anss = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i + 1; j <= n; j++) {
                if (idx[i] > idx[j]) {
                    anss++;
                }
            }
        }
        System.out.println(anss);
    }
}
