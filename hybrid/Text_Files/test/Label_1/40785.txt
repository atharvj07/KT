import sys

input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
if n < 3:
    print("Yes")
    exit()
for i in range(1, n - 1):
    if a[i] % a[i - 1] == 0 or a[i] % a[i + 1] == 0:
        print("No")
        exit()
a.sort()
b = []
for i in range(0, n, 2):
    b.append(a[i])
for i in range(1, n, 2):
    b.append(a[i])
print("Yes")
