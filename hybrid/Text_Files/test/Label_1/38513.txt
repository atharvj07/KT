import sys
input = sys.stdin.readline

def main():
    n, k = map(int,input().split())
    a = sorted(int(i) for i in input().split())
    ans = []
    i = 0
    while i < n:
        ans.append(a[i])
        j = i + 1
        while j < n and a[j] <= a[i] + k:
            j += 1
        i = j
    print(*ans)

main()
