#include <iostream>
#include <vector>

using namespace std;

int main() {
    long long c, hr, hb, wr, wb;
    cin >> c >> hr >> hb >> wr >> wb;
    long long ans = 0;
    if (hr / wr > hb / wb) {
        ans += min(c / wr, hb / wb);
        c -= min(c / wr, hb / wb) * wr;
    } else {
        ans += min(c / wb, hr / wr);
        c -= min(c / wb, hr / wr) * wb;
    }
    if (c > 0) {
        if (hr / wr > hb / wb) {
            ans += min(c / wr, hb / wb);
        } else {
            ans += min(c / wb, hr / wr);
        }
    }
    cout << ans * hr + (ans - min(c / wr, hb / wb)) * hb << endl;
    return 0;
}
