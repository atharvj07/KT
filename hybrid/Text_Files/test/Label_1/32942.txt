import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; ) {
            int j = i;
            int sum = 0;
            while (j < n && sum + s[j] <= 4) {
                sum += s[j++];
            }
            ans++;
            i = j;
        }
        System.out.println(ans);
    }
}
