def who_wins(board):
    min_white = 8  # Initialize with the maximum possible distance + 1
    min_black = 8  # Initialize with the maximum possible distance + 1
    for r in range(8):
        for c in range(8):
            if board[r][c] == 'W':
                for k in range(r, -1, -1):  # Check if the path upwards is clear
                    if board[k][c] == 'B':
                        break
                else:  # No break, path is clear
                    min_white = min(min_white, r)
            elif board[r][c] == 'B':
                for k in range(r, 8):  # Check if the path downwards is clear
                    if board[k][c] == 'W':
                        break
                else:  # No break, path is clear
                    min_black = min(min_black, 7-r)
    return 'A' if min_white <= min_black else 'B'

board = [input() for _ in range(8)]
print(who_wins(board))
