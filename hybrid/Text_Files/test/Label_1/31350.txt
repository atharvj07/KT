import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static int[] p;
    static int[] deg;
    static List<Integer>[] adj;
    static boolean[] vis;
    static List<Integer> ord;

    static void dfs1(int u) {
        vis[u] = true;
        for (int v : adj[u]) {
            if (!vis[v]) {
                dfs1(v);
                deg[u] += deg[v];
            }
        }
    }

    static boolean dfs2(int u, int par, boolean f) {
        vis[u] = true;
        ord.add(u);

        if (f) {
            deg[u]--;
        }

        int cnt = 0;
        for (int v : adj[u]) {
            if (!vis[v]) {
                if (!dfs2(v, u, deg[u] % 2 == 0)) {
                    return false;
                }
                cnt++;
            }
        }

        if (par != -1 && cnt == 0) {
            return false;
        }

        return true;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader r = new BufferedReader(new InputStreamReader(System.in));

        n = Integer.parseInt(r.readLine());
        p = new int[n + 1];
        deg = new int[n + 1];
        adj = new List[n + 1];
        vis = new boolean[n + 1];
        ord = new ArrayList<>();

        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }

        StringTokenizer st = new StringTokenizer(r.readLine());
        for (int i = 1; i <= n; i++) {
            p[i] = Integer.parseInt(st.nextToken());
            if (p[i] != 0) {
                adj[i].add(p[i]);
                adj[p[i]].add(i);
                deg[i]++;
                deg[p[i]]++;
            }
        }

        if (n % 2 == 1) {
            System.out.println("NO");
            return;
        }

        dfs1(1);

        Arrays.fill(vis, false);

        if (!dfs2(1, -1, false)) {
            System.out.println("NO");
            return;
        }

        System.out.println("YES");
        for (int u : ord) {
            System.out.println(u);
        }
    }
}
