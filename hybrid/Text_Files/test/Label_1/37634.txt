import sys

n, k = map(int, sys.stdin.readline().split())
screen = []
for _ in range(n):
    screen.append(list(sys.stdin.readline().strip()))
max_lines = 0
for r in range(n-k+1):
    for c in range(n-k+1):
        count = 0
        for i in range(r, r+k):
            for j in range(c, c+k):
                if screen[i][j] == 'B':
                    count += 1
        max_lines = max(max_lines, count + r + c - int(count > 0))
print(max_lines)
