#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<vector<int>> adj(n);
  for (int i = 0; i < m; ++i) {
    int u, v;
    cin >> u >> v;
    --u, --v;
    adj[u].push_back(v);
  }
  int ans = 0;
  vector<int> vis(n);
  for (int i = 0; i < n; ++i) {
    if (vis[i] == 0) {
      ++ans;
      int cnt = 0;
      vector<int> q = {i};
      while (!q.empty()) {
        int u = q.back();
        q.pop_back();
        if (vis[u] == 1) continue;
        vis[u] = 1;
        ++cnt;
        for (int v : adj[u]) {
          if (vis[v] == 0) q.push_back(v);
        }
      }
      if (cnt == 1) --ans;
    }
  }
  cout << ans << endl;
  return 0;
}
