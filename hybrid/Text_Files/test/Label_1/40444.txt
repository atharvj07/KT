from collections import Counter

n, q = map(int, input().split())
a = list(map(int, input().split()))

c = Counter(a)
if len(c) == 1:
    print('NO')
else:
    print('YES')
    for i in range(n):
        if a[i] == 0:
            a[i] = max(c.keys())
            c[max(c.keys())] -= 1
        elif a[i] == max(c.keys()):
            a[i] = max(j for j in c.keys() if c[j] > 0)
            c[max(j for j in c.keys() if c[j] > 0)] -= 1
    print(*a)
