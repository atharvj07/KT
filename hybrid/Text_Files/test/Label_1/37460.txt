for _ in range(int(input())):
    n = int(input())
    h = []
    w = []
    for i in range(n):
        hi, wi = map(int, input().split())
        if hi > wi:
            h.append(hi)
            w.append(wi)
        else:
            h.append(wi)
            w.append(hi)
    
    ans = []
    for i in range(n):
        found = False
        for j in range(n):
            if i == j:
                continue
            if h[i] < h[j] and w[i] < w[j]:
                ans.append(j + 1)
                found = True
                break
            if w[i] < h[j] and h[i] < w[j]:
                ans.append(j + 1)
                found = True
                break
        if not found:
            ans.append(-1)
    print(' '.join(map(str, ans)))
