#include<bits/stdc++.h>
using namespace std;
int main(){
   int n,p;
   cin>>n>>p;
   long long t[n];
   for(int i=0;i<n;i++)cin>>t[i];
   deque<int> q;
   for(int i=0;i<n;i++){
      //t[q.front()]<t[i]: the passenger sitting in the front of the queue has already left
      while(q.size()&&t[q.front()]<=t[i])q.pop_front();
      //if there's a passenger standing in front of me, I will wait
      if(q.size())cout<<q.front()+p<<" ";
      else cout<<t[i]<<" ";
      q.push_front(i);
   }
   return 0;
}
