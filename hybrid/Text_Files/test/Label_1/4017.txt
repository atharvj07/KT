#include <bits/stdc++.h>
using namespace std;
void solve(int l, int r, string s) {
  if (l + r == 0) {
    cout << s << endl;
  } else if (l != 0) {
    solve(l - 1, r, s + '(');
  } else {
    solve(l, r - 1, s + ')');
  }
}
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    solve(n, n, "");
  }
}
