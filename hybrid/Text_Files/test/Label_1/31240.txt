import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        while (t-- > 0) {
            int n = in.nextInt();
            int count = 0;
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (i != n) {
                        count++;
                    } else {
                        if (j == 1) {
                            count++;
                        }
                    }
                }
            }
            System.out.println(count);
        }
    }
}
