for _ in range(int(input())):
    n,l,r,s=map(int,input().split())
    if not (s+r-l)&1:
        a=[-1]*n
        b=[]
        for i in range(r-l+1):
            if i&1:
                b.append(s//2)
                s-=s//2
            else:
                b.append(s//2+1)
                s-=s//2+1
        for i in range(l-1):
            a[i]=i+1
        for i in range(r-l+1):
            a[l+i-1]=n-i
        a.extend(b)
        print(*a)
    else: print(-1)
