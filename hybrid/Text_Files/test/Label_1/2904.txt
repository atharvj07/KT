#include <iostream>
#include <vector>
using namespace std;
int n,a[200000],b[200000],vis[200000],prv[200000];
vector<int> e[200000];
bool solve(int pos){
for (int u:e[pos]){
if (!vis[u]){
vis[u]=true;
prv[u]=pos;
if (!solve(u)){
return false;
}
}else if (prv[u]!=pos){
return false;
}
}
return true;
}
int main(){
cin>>n;
for (int i=0;i<n;i++){
int x;
cin>>x;
a[i]=x;
}
for (int i=0;i<n;i++){
int x;
cin>>x;
b[i]=x;
}
for (int i=0;i<n-1;i++){
e[i].push_back(i+1);
e[i+1].push_back(i);
}
e[0].push_back(n-1);
e[n-1].push_back(0);
int ans=1;
for (int i=0;i<n;i++){
if (a[i]!=b[i]){
ans=0;
break;
}
}
if (!ans){
cout<<"NO"<<endl;
}else{
for (int i=0;i<n;i++){
if (!vis[i]){
if (!solve(i)){
cout<<"NO"<<endl;
return 0;
}
}
}
cout<<"YES"<<endl;
}
return 0;
}
