import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] w = new int[n];
        for (int i = 0; i < n; i++) {
            w[i] = sc.nextInt();
        }
        int p = sc.nextInt();
        for (int i = 0; i < p; i++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt();
            long ans = 0;
            for (int j = a; j < n; j += b) {
                ans += w[j];
            }
            System.out.println(ans);
        }
    }
}
