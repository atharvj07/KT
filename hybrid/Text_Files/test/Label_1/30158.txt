import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long a = sc.nextLong();
        long b = sc.nextLong();
        long m = sc.nextLong();

        //initialize zigzag counter
        long ans = 0;

        if (a % m == 0) ans++;

        long bound = (b / m) + (a / m);
        long increase = 1;

        for (long i = a / m + 1; i <= bound; i++) {
            if ((i - 1) % 2 == 0) {
                ans += increase;
            } else {
                ans -= increase;
            }

            increase++;
        }

        System.out.println(ans % 10000);
    }
}
