#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main() {
    ll n;
    cin>>n;
    vector<ll> a(n);
    for(int i=0; i<n; i++) cin>>a[i];
    ll ans=0,cur=0;
    sort(a.begin(),a.end());
    for(int i=0; i<n; i++) {
        if(a[i]>=cur) {
            ans++;
            cur++;
        }
    }
    cout<<ans<<'\n';
}
