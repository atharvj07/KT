#include <iostream>
#include <string>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int n = s.size();
    int eq = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == s[(i + 1) % n]) eq++;
    }
    cout << n - eq << endl;
  }
  return 0;
}
