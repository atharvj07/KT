import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            long k = sc.nextLong(), l = sc.nextLong(), r = sc.nextLong(), p = sc.nextLong();
            long ans = 0;
            for (long j = l; j <= r; j++) {
                ans += gcd(k * j * j + j + 1, p);
            }
            System.out.println(ans % p);
        }
    }

    static long gcd(long a, long b) {
        if (a < b) {
            return gcd(b, a);
        }
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
