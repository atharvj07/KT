import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String num = sc.next();
        
        Set<String> set = new HashSet<>();
        set.add(num);
        
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                String tmp = "";
                for (int k = 0; k < n; k++) {
                    if (k == i) {
                        tmp += num.charAt(j);
                    } else if (k == j) {
                        tmp += num.charAt(i);
                    } else {
                        tmp += num.charAt(k);
                    }
                }
                set.add(tmp);
            }
        }
        
        System.out.println(set.size() == 1 ? "YES" : "NO");
    }
}
