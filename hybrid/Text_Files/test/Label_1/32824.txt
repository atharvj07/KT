import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] h = new int[n];
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
            m[i] = sc.nextInt();
        }
        Arrays.sort(h);
        Arrays.sort(m);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (h[i] < h[n - 1]) {
                ans++;
            } else if (m[i] < m[n - 1]) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
