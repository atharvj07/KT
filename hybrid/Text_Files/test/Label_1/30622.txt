import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String s = sc.next();
    int k = 1;
    char prev = s.charAt(0);
    for (int i = 1; i < s.length(); i++) {
      if (s.charAt(i) != prev) {
        k++;
        prev = s.charAt(i);
      }
    }
    System.out.println(k);
  }
}
