from collections import defaultdict

n, m = map(int, input().split())
p = list(map(int, input().split()))
c = list(map(int, input().split()))
d = int(input())
l = [int(input()) for i in range(d)]
d = defaultdict(list)

for i in range(n):
    d[c[i]].append(p[i])

for x in d.values():
    x.sort()

for i in l:
    d[c[i - 1]].remove(p[i - 1])
    if len(d[c[i - 1]]) == 0:
        print(0)
        continue

    mx = 0
    for j in d.values():
        if len(j) == 0:
            continue

        cur = j[0]
        while cur in j:
            cur += 1

        mx = max(mx, cur - 1)

    print(mx)
