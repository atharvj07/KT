#include<bits/stdc++.h>
#define pb push_back
using namespace std;
int main() {
  int n, m, min, max;
  cin >> n >> m >> min >> max;
  set<int> s;
  for (int i = 1; i <= m; i++) {
    int x;
    cin >> x;
    s.insert(x);
  }
  if (s.count(min) && s.count(max) && (int)s.size() >= m)
    cout << "Correct" << endl;
  else
    cout << "Incorrect" << endl;
  return 0;
}
