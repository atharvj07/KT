import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner in = new Scanner(new BufferedReader(new InputStreamReader(System.in)));
		int n = in.nextInt(), i = in.nextInt();
		int[] a = new int[n];
		for (int j = 0; j < n; j++)
			a[j] = in.nextInt();
		in.close();
		Arrays.sort(a);
		int l = a[0], r = a[n - 1], mid;
		int min = n / 2 + 1;
		while (l < r) {
			mid = (l + r) / 2;
			int lsum = 0, rsum = 0;
			for (int j = 0; j < n; j++) {
				if (a[j] < mid)
					lsum++;
				else if (a[j] > mid)
					rsum++;
			}
			lsum += rsum / 2;
			rsum /= 2;
			if (8L * (n - lsum - rsum) <= i) {
				r = mid;
				min = Math.min(min, lsum + rsum);
			} else
				l = mid + 1;
		}
		System.out.print(min);
	}
}
