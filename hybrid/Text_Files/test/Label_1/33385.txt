import java.util.Scanner;

public class HedgehogsPresents {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int k = scanner.nextInt();

        int c = scanner.nextInt();
        int[] holidays = new int[c];
        for (int i = 0; i < c; i++) {
            holidays[i] = scanner.nextInt();
        }

        int presents = 0;
        int lastPresentDay = 0;

        for (int day = 1; day <= n; day++) {
            if (isHoliday(day, holidays)) {
                presents++;
                lastPresentDay = day;
            } else if (day - lastPresentDay >= k) {
                presents++;
                lastPresentDay = day;
            }
        }

        System.out.println(presents);
    }

    private static boolean isHoliday(int day, int[] holidays) {
        for (int holiday : holidays) {
            if (holiday == day) {
                return true;
            }
        }
        return false;
    }
}
