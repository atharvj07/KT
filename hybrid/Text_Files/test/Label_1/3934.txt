#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int cnt = 0;
  int ans = 0;
  for (int i = 1; i < n - 1; i++) {
    if (s[i] == '0') cnt++;
    if (cnt == k - 1 && s[i] == '1') {
      ans = max(ans, i - 1);
      cnt = 0;
    }
  }
  cout << ans << endl;
  return 0;
}
