import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int w = sc.nextInt();
        String ans = "";
        while (w > 0) {
            if (w % 3 == 1) {
                ans += "+";
                w -= 4;
            } else if (w % 3 == 2) {
                ans += "0";
                w -= 2;
            } else {
                ans += "-";
                w -= 1;
            }
            w /= 3;
        }
        System.out.println(ans);
    }
}
