import sys
class UnionFind:
    def __init__(self):
        self.par =[0]*200010
        self.siz =[0]*200010
        for i in range(200010):
            self.par[i]=i
    def find(self , u):
        if self.par[u] != u:
            self.par[u] = self.find(self.par[u])
        return self.par[u]
    def merge(self , a , b):
        ar = self.find(a)
        br = self.find(b)
        if ar != br:
            if self.siz[ar]<self.siz[br]:
                ar , br = br , ar
            self.par[br] = ar
            self.siz[ar] += self.siz[br]

n = int(input());
arr = list(map(int , input().split()))
graph = [[] for i in range(n+1)]
for i in range(n-1):
    a , b = map(int , sys.stdin.readline().split())
    graph[a].append(b)
    graph[b].append(a)
uf = UnionFind()
for i in range(n):
    uf.siz[i]=1
for i in range(n):
    if arr[i]:
        for v in graph[i+1]:
            if not arr[v-1]:
                uf.merge(i , v-1)
mx = 0
for i in range(n):
    mx = max(mx , uf.siz[uf.find(i)])
print(n-mx)
