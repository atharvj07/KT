for _ in range(int(input())):
    n = int(input())
    if n < 6 or n % 2 == 1:
        print("NO")
    else:
        print("YES")
        for i in range(2, n // 3):
            if n % i == 0:
                j = n // i
                if j % 2 == 1:
                    j -= 1
                if i % 2 == 1:
                    i -= 1
                print(i // 2, j // 2, n // i // j)
                break
