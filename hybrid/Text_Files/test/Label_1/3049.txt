#include <vector>
#include <cstdio>
#include <algorithm>
using namespace std;

typedef long long ll;
#define min(a,b) ((a)<(b)?(a):(b))
#define max(a,b) ((a)>(b)?(a):(b))
#define pb push_back

vector<int> a, c, p;

int main() {
    int n, l, r;
    scanf("%d %d %d", &n, &l, &r);
    a.resize(n);
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    c.resize(n);
    p.resize(n);
    for (int i = 0; i < n; i++)
        scanf("%d", &p[i]);
    for (int i = 0; i < n; i++) {
        for (int j = l; j <= r; j++) {
            if (p[i] <= 0) {
                printf("-1\n");
                return 0;
            }
            if (p[i] == 1) {
                c[i] = r;
                p[i] = n;
                break;
            }
            if (p[i] == n) {
                c[i] = l;
                break;
            }
            if (j > a[i]) {
                if (p[i] == n - i) {
                    c[i] = j;
                    p[i] -= n - i - 1;
                    break;
                }
                if (p[i] >= n - i + 1) {
                    c[i] = j;
                    p[i] -= n - i;
                }
            }
        }
        if (p[i] != 0) {
            printf("-1\n");
            return 0;
        }
    }
    for (int i = 0; i < n; i++)
        printf("%lld ", a[i] + c[i]);
    printf("\n");
    return 0;
}
