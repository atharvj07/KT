import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int k = scanner.nextInt();
    long[] power = new long[n + 1];
    for (int i = 1; i <= n; i++) {
      power[i] = scanner.nextInt();
    }
    long minPower = Long.MAX_VALUE;
    int minStart = -1;
    for (int i = 1; i <= n; i++) {
      long sumPower = 0;
      for (int j = i; j <= n; j += k) {
        sumPower += power[j];
      }
      if (sumPower < minPower) {
        minPower = sumPower;
        minStart = i;
      }
    }
    System.out.println(minStart);
  }
}
