#include<bits/stdc++.h>
using namespace std;

int main()
{
    int n, x, a[200005], ans[200005];
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> a[i];
    }
    set <int> s;
    for(int i = n - 1; i >= 0; i--)
    {
        s.insert(a[i]);
        auto it = s.upper_bound(a[i]);
        if(it == s.begin())
        {
            ans[i] = -1;
        }
        else
        {
            it--;
            ans[i] = *it;
        }
    }
    for(int i = 0; i < n; i++)
    {
        if(ans[i] == -1)
        {
            cout << ":(" << endl;
            return 0;
        }
    }
    for(int i = 0; i < n; i++)
    {
        cout << ans[i] << " ";
    }
    return 0;
}
