#include <bits/stdc++.h>
using namespace std;
int main(){
    int t; cin >> t;
    while(t--){
        string s, c;
        cin >> s >> c;
        if(s == c){
            cout << "---\n";
            continue;
        }
        bool f = false;
        for(int i = 0; i < s.size(); i++){
            for(int j = 0; j < s.size(); j++){
                if(i == j) continue;
                string str = s;
                swap(str[i], str[j]);
                if(str < c){
                    cout << str << "\n";
                    f = true;
                    break;
                }
            }
            if(f) break;
        }
        if(!f) cout << "---\n";
    }
}
