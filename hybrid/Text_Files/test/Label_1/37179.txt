n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
ans = []
st = []
for i in range(n-1,-1,-1):
    if b[i] not in st:
        st.append(b[i])
        ans.append(0)
    else:
        ind = st.index(b[i])
        ans.append(i-ind)
        for j in range(ind+1):
            st.pop()
ans.reverse()
print(*ans)
