#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5;

int main() {
  int n; cin >> n;
  long long arr[MAXN + 7];
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  long long mn = *min_element(arr, arr + n);
  long long mx = *max_element(arr, arr + n);
  if ((mx - mn) % (n - 1) != 0) {
    cout << -1 << endl;
  } else {
    long long d = (mx - mn) / (n - 1);
    long long ans = 0;
    for (int i = 1; i < n; i++) {
      if (arr[i] != arr[0] + d * i) ans++;
    }
    cout << ans << endl;
  }
  return 0;
}
