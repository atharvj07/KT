#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> m, a;

pair<int, vector<int>> solve(int l, int r) {
  if (l > r) { return {0, {}}; }
  int mx = 0, mi = 1e9;
  for (int i = l; i <= r; i++) {
    mx = max(mx, m[i]);
    mi = min(mi, m[i]);
  }
  int best = -1;
  vector<int> cur;
  for (int x = mi; x <= mx; x++) {
    vector<int> ret = {x};
    for (int i = l + 1; i <= r; i++) { ret.push_back(max({a[i - 1], x, 0})); }
    pair<int, vector<int>> res = solve(l, l + (int)ret.size() - 2);
    res.second.insert(res.second.begin(), ret.begin(), ret.end());
    if (res.first + x > best) {
      best = res.first + x;
      cur = res.second;
    }
  }
  return {best, cur};
}

int main() {
  cin >> n;
  m.resize(n), a.resize(n);
  for (int &x : m) { cin >> x; }
  pair<int, vector<int>> res = solve(0, n - 1);
  for (int x : res.second) { cout << x << " "; }
  cout << "\n";
  return 0;
}
