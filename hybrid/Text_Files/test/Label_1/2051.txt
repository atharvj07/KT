#include <bits/stdc++.h>
#define LL long long
using namespace std;
const int mod = 998244353;
LL powmod(LL x, LL p) {
    LL ans = 1;
    for (; p; p >>= 1) {
        if (p & 1) ans = ans * x % mod;
        x = x * x % mod;
    }
    return ans;
}
int main() {
    cin.tie(nullptr)->sync_with_stdio(false);
    int n;
    cin >> n;
    vector<int> a(2 * n);
    for (int& x : a) cin >> x;
    sort(a.begin(), a.end());
    int sum1 = 0, sum2 = 0;
    for (int i = 0; i < n; ++i) {
        sum1 += a[i];
        sum2 += a[2 * n - 1 - i];
    }
    cout << (1LL * (sum1 - sum2) * (sum1 - sum2) % mod * powmod(2, mod - 2)) % mod << "\n";
}
