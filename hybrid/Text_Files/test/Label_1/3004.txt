#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::string s;
  std::cin >> s;
  std::vector<bool> dp(n + 1, false);
  dp[1] = true;
  for (int i = 1; i <= n; i++) {
    if (!dp[i]) continue;
    for (int j = 1; j <= k; j++) {
      if (i + j <= n && s[i + j - 1] != '#') dp[i + j] = true;
    }
  }
  std::cout << (dp[n] ? "YES" : "NO") << std::endl;
  return 0;
}
