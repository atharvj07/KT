s=input()
t=input()
if s==t:
    print("Yes")
    exit(0)
if len(s)!=len(t):
    print("No")
    exit(0)
vowels=["A","E","I","O","U","a","e","i","o","u"]
cnt1=0
cnt2=0
for i in range(len(s)):
    if s[i] in vowels:
        cnt1+=1
        if s[i]!=t[i]:
            cnt2+=1
    else:
        if s[i]!=t[i]:
            cnt2+=1
if cnt1==cnt2:
    print("Yes")
else:
    print("No")
