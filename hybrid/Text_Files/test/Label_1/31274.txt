import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;

class Main {

    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            int cntnl = 0, cnt = 0;
            int speed = 0;
            boolean canOvertake = true;
            ArrayList<Integer> speedLimit = new ArrayList<>();
            for (int i = 0; i < n; i++) {
                String[] s = br.readLine().split(" ");
                int type = Integer.parseInt(s[0]);
                switch (type) {
                    case 1:
                        speed = Integer.parseInt(s[1]);
                        break;
                    case 2:
                        if (canOvertake) {
                            cnt++;
                        }
                        break;
                    case 3:
                        speedLimit.add(Integer.parseInt(s[1]));
                        if (Integer.parseInt(s[1]) < speed) {
                            cntnl++;
                            cnt++;
                        }
                        break;
                    case 4:
                        canOvertake = true;
                        break;
                    case 5:
                        speedLimit.clear();
                        break;
                    case 6:
                        canOvertake = false;
                        break;
                }
            }
            int res = cnt + cntnl;
            if (res > speedLimit.size()) {
                res = speedLimit.size();
            }
            System.out.println(res);
        }
    }
}
