import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] x = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            x[i] = sc.nextInt();
        }
        int L = sc.nextInt();
        int Q = sc.nextInt();
        for (int i = 0; i < Q; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int dist = x[b] - x[a];
            if (dist % L == 0) {
                System.out.println(dist / L);
            } else {
                System.out.println(dist / L + 1);
            }
        }
    }
}
