#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, L;
    cin >> n >> L;
    vector<int> k(n), s(n);
    for (int& x : k) cin >> x;
    for (int& x : s) cin >> x;
    for (int i = 0; i < n; i++)
        if (k[(i + 1) % n] - k[i] != s[(i + 1) % n] - s[i])
            return cout << "NO", 0;
    cout << "YES";
}
