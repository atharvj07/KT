import java.util.Scanner;

public class Main {
    private static final int MOD = 1000000007;
    private static long[][] dp;
    private static int[] r, c;
    private static int h, w;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        h = sc.nextInt();
        w = sc.nextInt();

        r = new int[h + 1];
        c = new int[w + 1];
        for (int i = 1; i <= h; i++) {
            r[i] = sc.nextInt();
        }
        for (int i = 1; i <= w; i++) {
            c[i] = sc.nextInt();
        }

        dp = new long[h + 1][w + 1];
        dp[0][0] = 1;
        for (int i = 1; i <= h; i++) {
            for (int j = 1; j <= w; j++) {
                dp[i][j] = dp[i - 1][j] * c[j] % MOD;
                if (r[i] > 0) {
                    dp[i][j] = (dp[i][j] + dp[i][j - 1] * r[i] % MOD) % MOD;
                }
            }
        }

        System.out.println(dp[h][w]);
    }
}
