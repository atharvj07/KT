import sys
input = sys.stdin.readline

n = int(input())
s = input()
x, y, ans = 0, 0, 1
for i in s:
    if i == 'L':
        x -= 1
        ans = max(ans, -x, -y)
    elif i == 'R':
        x += 1
        ans = max(ans, x, y)
    elif i == 'U':
        y += 1
        ans = max(ans, x, -y)
    else:
        y -= 1
        ans = max(ans, x, y)
print(ans)
