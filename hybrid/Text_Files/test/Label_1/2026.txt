#include <bits/stdc++.h>
using namespace std;
const int mxn = 1e5;
int main() {
  int n;
  cin >> n;
  set<int> s;
  vector<int> a(n);
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    s.insert(a[i]);
  }
  for (int i = 0; i < n; i++) {
    int ans = 0;
    while (s.count(ans)) ans++;
    cout << ans << " ";
    s.insert(ans);
  }
  return 0;
}
