#include <bits/stdc++.h>
using namespace std;
#define ll long long int
ll p[30];
ll msbPos(int n)
{
    for(ll i=29;i>=0;i--)
        if(p[i]) return i;
}
int main()
{
    ll n,x;
    cin>>n;
    set<ll> st;
    for(ll i=0; i<n; i++)
    {
        cin>>x;
        st.insert(x);
        while(x)
        {
            ll pos=msbPos(x);
            p[pos]++;
            x-=(1<<pos);
        }
    }
    memset(p,0,sizeof(p));
    ll mx=0;
    for(ll i=29; i>=0; i--)
    {
        if(p[i]%2==0)
        {
            p[i]/=2;
            mx=i;
        }
        else
        {
            p[i]=p[i]/2+1;
            mx=i;
            break;
        }
    }
    ll temp=mx;
    ll arr[1<<(mx+1)];
    arr[0]=0;
    ll pos=0,val=0;
    while(temp>=0)
    {
        if(p[temp])
        {
            val+=1<<temp;
            p[temp]--;
        }
        else if(val)
        {
            arr[++pos]=val;
            val=0;
        }
        temp--;
    }
    if(val) arr[++pos]=val;
    cout<<mx<<endl;
    for(ll i=0; i<(1<<mx); i++)
    {
        ll x=arr[i];
        while(x)
        {
            ll pos=msbPos(x);
            cout<<i<<' ';
            x-=(1<<pos);
        }
        cout<<i<<' ';
    }
    cout<<endl;
    return 0;
}
