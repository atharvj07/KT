from bisect import *

n, r = map(int, input().split())

a = []
b = []
for i in range(n):
    ai, bi = map(int, input().split())
    a.append(ai)
    b.append(bi)

a.sort()
b.sort(reverse=True)

j = 0
ans = 0
for i in b:
    while j < n and r + sum(b[j : i + 1]) >= a[i]:
        r += b[j]
        j += 1
    r -= i
    ans += 1

print(ans)
