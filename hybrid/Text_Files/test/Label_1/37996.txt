from math import gcd
from collections import Counter

def restore_array(n, gcd_table):
    gcd_table.sort(reverse=True)
    array_a = [gcd_table[0]]
    for _ in range(1, n):
        new_element = gcd_table[1]
        for j in range(len(array_a)):
            gc = gcd(new_element, array_a[j])
            gcd_table.remove(gc)
            gcd_table.remove(gc)
        array_a.append(new_element)
    return array_a

n = int(input())
gcd_table = list(map(int, input().split()))
result = restore_array(n, gcd_table)
print(*result)
