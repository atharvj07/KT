N, K = map(int, input().split())
box_types = list(map(int, input().split()))

min_leftover = N
best_type = 0
best_count = 0

for i, box_capacity in enumerate(box_types):
    num_boxes = N // box_capacity
    leftover = N - num_boxes * box_capacity
    if leftover < min_leftover:
        min_leftover = leftover
        best_type = i + 1
        best_count = num_boxes

print(best_type, best_count)
