from collections import deque

def bfs(graph, start):
    dist = [-1] * len(graph)
    dist[start] = 0
    queue = deque([start])
    while queue:
        u = queue.popleft()
        for v, w in graph[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + w
                queue.append(v)
    return dist

def solve(n, edges, restrictions):
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u - 1].append((v - 1, w))
        graph[v - 1].append((u - 1, w))
    
    all_pairs = []
    for i in range(n):
        dist = bfs(graph, i)
        for j in range(i + 1, n):
            all_pairs.append((dist[j], i, j))
    all_pairs.sort(reverse=True)
    
    p = [-1] * n
    for d, i, j in all_pairs:
        if restrictions[i] > 0 and restrictions[j] > 0:
            p[i], p[j] = j, i
            restrictions[i] -= 1
            restrictions[j] -= 1
    
    if p.count(-1) == n:
        return 0
    return min([w for u, v, w in edges if p[u - 1] == v - 1 or p[v - 1] == u - 1], default=0)

n = int(input())
edges = [tuple(map(int, input().split())) for _ in range(n - 1)]
restrictions = [int(input()) for _ in range(n)]
print(solve(n, edges, restrictions))
