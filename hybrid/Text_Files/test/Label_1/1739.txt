#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define vi vector<int>
#define pb push_back
#define mp make_pair
#define f first
#define s second
#define pii pair<int, int>
#define pll pair<ll, ll>
#define MAXN 100005
#define INF (ll)2e18

int n;
string s[MAXN];

void solve() {
    cin >> n;
    for (ll i = 0; i < n; i++) {
        cin >> s[i];
    }
    vi ans;
    for (ll i = 0; i < n; i++) {
        ll x = 0, p = 1;
        for (ll j = s[i].size() - 1; j >= 0; j--) {
            if (s[i][j] == '?') {
                x += p;
                p *= 10;
            }
            else {
                x += (s[i][j] - '0') * p;
                p *= 10;
            }
        }
        ans.pb(x);
    }
    if (ans.size() != unique(ans.begin(), ans.end()) - ans.begin()) {
        cout << "NO" << endl;
        return;
    }
    cout << "YES" << endl;
    for (auto i : ans) cout << i << endl;
}
int main() {
    solve();
    return 0;
}
