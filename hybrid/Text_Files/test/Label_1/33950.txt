import java.util.*;
import java.lang.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        int[] pos = new int[n + 1];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
            pos[p[i]] = i;
        }
        List<Integer>[] adj = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            adj[pos[u]].add(pos[v]);
            adj[pos[v]].add(pos[u]);
        }
        int[] dist = new int[n];
        Queue<Integer> q = new LinkedList<>();
        q.add(n - 1);
        while (!q.isEmpty()) {
            int cur = q.poll();
            for (int next : adj[cur]) {
                if (dist[next] == 0) {
                    dist[next] = dist[cur] + 1;
                    q.add(next);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans = Math.max(ans, dist[i]);
        }
        System.out.println(ans);
    }
}
