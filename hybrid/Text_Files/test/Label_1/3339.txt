#include <iostream>
#include <cmath>
using namespace std;
const int mod = 1000000007;
int main(){
    int n;
    string s;
    cin >> n >> s;
    int sum = 0;
    for(int i=0;i<n;++i) sum += (s[i] - '0');
    int ans = sum;
    for(int i=3;i*i<=ans;++i){
        if(ans % i == 0){
            while(ans % i == 0) ans /= i;
            if(i % 2 == 0) i += 1;
        }
    }
    if(ans > 1) cout << "NO\n";
    else cout << "YES\n";
    return 0;
}
