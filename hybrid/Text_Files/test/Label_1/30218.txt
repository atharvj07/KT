import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        for (int i = 0; i < t; i++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int k = sc.nextInt();
            int[] aList = new int[k];
            int[] bList = new int[k];
            for (int j = 0; j < k; j++) {
                aList[j] = sc.nextInt();
            }
            for (int j = 0; j < k; j++) {
                bList[j] = sc.nextInt();
            }
            int ans = 0;
            for (int j = 0; j < k; j++) {
                for (int l = j + 1; l < k; l++) {
                    if (aList[j] != aList[l] && bList[j] != bList[l]) {
                        ans++;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
