#include <iostream>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) {
    cin >> a[i];
  }
  bool ok = true;
  int l = -1, r = -1;
  for (int i = 1; i < n; i++) {
    if (a[i] < a[i - 1]) {
      if (l == -1) {
        l = i - 1;
      }
      r = i;
      ok = false;
    }
  }
  if (ok) {
    cout << "yes" << endl;
  } else {
    if (l == 0 || r == n - 1) {
      cout << "no" << endl;
    } else {
      cout << "yes" << endl;
      cout << l + 1 << " " << r + 1 << endl;
    }
  }
  return 0;
}
