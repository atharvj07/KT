#include <bits/stdc++.h>
using namespace std;
int main() {
    int n; cin >> n;
    vector<int> a(n), b(n);
    for (auto &i : a) { cin >> i; b.push_back(-i - 1); }
    for (int i = 0; i < n; i++) {
        if (a[i] < 0) swap(a[i], b[i]);
    }
    sort(a.begin(), a.end(), greater<>());
    for (auto &i : a) cout << i << ' '; cout << endl;
}
