import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] p = new int[n];
        for (int i = 0; i < n; i++) {
            p[i] = sc.nextInt();
        }
        List<Pair> pairs = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            pairs.add(new Pair(sc.nextInt() - 1, sc.nextInt() - 1));
        }
        int[] max = new int[n];
        for (int i = 0; i < n; i++) {
            max[i] = p[i];
        }
        for (Pair pair : pairs) {
            int a = pair.a;
            int b = pair.b;
            if (p[a] > p[b]) {
                int tmp = p[a];
                p[a] = p[b];
                p[b] = tmp;
            }
            if (p[a] > max[a]) {
                max[a] = p[a];
            }
            if (p[b] > max[b]) {
                max[b] = p[b];
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(max[i] + " ");
        }
    }

    static class Pair {
        int a;
        int b;

        public Pair(int a, int b) {
            this.a = a;
            this.b = b;
        }
    }
}
