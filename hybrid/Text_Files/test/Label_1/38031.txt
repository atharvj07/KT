cnt = [[0] * 101 for i in range(101)]
n = int(input())
for i in range(n):
    x1, y1, x2, y2 = map(int, input().split())
    cnt[y1][x1] += 1
    cnt[y1][x2 + 1] -= 1
    cnt[y2 + 1][x1] -= 1
    cnt[y2 + 1][x2 + 1] += 1
sum = 0
for i in range(1, 101):
    for j in range(1, 101):
        cnt[i][j] += cnt[i - 1][j] + cnt[i][j - 1] - cnt[i - 1][j - 1]
        sum += cnt[i][j]
print(sum)
