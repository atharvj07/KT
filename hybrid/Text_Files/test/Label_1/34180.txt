import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int k = in.nextInt();
        int n = in.nextInt();
        int[] ax = new int[k];
        int[] ay = new int[k];
        int[] mx = new int[n];
        int[] my = new int[n];
        for (int i = 0; i < k; i++) {
            ax[i] = in.nextInt();
            ay[i] = in.nextInt();
        }
        for (int i = 0; i < n; i++) {
            mx[i] = in.nextInt();
            my[i] = in.nextInt();
        }
        int cnt = 0;
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                if ((mx[j] - ax[i]) * (my[j] - ay[i]) < 0) {
                    cnt++;
                    break;
                }
            }
        }
        System.out.println(cnt);
    }

}
