import sys
n, b, k, x = map(int,sys.stdin.readline().split())
a = list(map(int,sys.stdin.readline().split()))
f = [0]*100
ans = 1
for i in range(n):
	f[a[i]]+=1
	if f[a[i]] == 1: b-=1
ans = (ans * b) % (10**9+7)
s = 0
for i in range(100):
	if i == k: s+=f[i]-1
	s%=x
	ans=(ans * f[i]) % (10**9+7)
if b == 1 and s == 0: ans+=1
ans%=10**9+7
print(ans)
