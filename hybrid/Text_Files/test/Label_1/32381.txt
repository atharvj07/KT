import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int m = input.nextInt();
        input.close();
        long ans = 0;
        for (int i = n; i >= m; i--) {
            ans = (ans + C(i, m)) % 1000000007;
        }
        System.out.println(ans);
    }

    private static long C(int n, int m) {
        long ans = 1;
        for (int i = 1; i <= m; i++) {
            ans = ans * (n - i + 1) / i;
        }
        return ans;
    }
}
