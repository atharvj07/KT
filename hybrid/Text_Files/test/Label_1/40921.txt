n, M = map(int, input().split())
t = list(map(int, input().split()))

prefix_sum = [0]
for time in t:
    prefix_sum.append(prefix_sum[-1] + time)

for i in range(1, n + 1):
    ans = 0
    total_time = prefix_sum[i]
    if total_time > M:
        diff = total_time - M
        for j in range(i - 1, 0, -1):
            if diff <= 0:
                break
            diff -= t[j]
            ans += 1
    print(ans, end=" ")
