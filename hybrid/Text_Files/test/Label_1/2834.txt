#include <bits/stdc++.h>

using namespace std;

void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  int ans_a = a[0], ans_b = 0, now_a = 0, now_b = 0, ans = (a[0] == 1 ? 2 : 1);
  for (int i = 1; i < n; i++) {
    ans++;
    if (now_a < now_b) {
      now_a += a[i];
      ans_a += a[i];
    } else {
      now_b += a[n - i - 1];
      ans_b += a[n - i - 1];
    }
  }
  cout << ans << ' ' << ans_a << ' ' << ans_b << '\n';
}

int main() {
  int t;
  cin >> t;
  while (t--) { solve(); }
  return 0;
}
