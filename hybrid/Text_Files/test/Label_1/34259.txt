import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int[][] points = new int[8][3];
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 3; j++) {
                points[i][j] = in.nextInt();
            }
        }

        int[] min = new int[3];
        for (int i = 0; i < 3; i++) {
            min[i] = 1000000000;
            for (int j = 0; j < 8; j++) {
                min[i] = Math.min(min[i], points[j][i]);
            }
        }

        int[] max = new int[3];
        for (int i = 0; i < 3; i++) {
            max[i] = -1000000000;
            for (int j = 0; j < 8; j++) {
                max[i] = Math.max(max[i], points[j][i]);
            }
        }

        if (max[0] - min[0] != max[1] - min[1] || max[0] - min[0] != max[2] - min[2]) {
            System.out.println("NO");
        } else {
            int side = max[0] - min[0];
            int[][] newPoints = new int[8][3];
            for (int i = 0; i < 8; i++) {
                for (int j = 0; j < 3; j++) {
                    newPoints[i][j] = points[i][j] - min[j];
                }
            }
            int[][] cubes = new int[6][4];
            cubes[0][0] = 0;
            cubes[0][1] = 0;
            cubes[0][2] = 0;
            cubes[0][3] = 1;
            cubes[1][0] = 0;
            cubes[1][1] = side;
            cubes[1][2] = 0;
            cubes[1][3] = 1;
            cubes[2][0] = side;
            cubes[2][1] = side;
            cubes[2][2] = 0;
            cubes[2][3] = 1;
            cubes[3][0] = side;
            cubes[3][1] = 0;
            cubes[3][2] = side;
            cubes[3][3] = 1;
            cubes[4][0] = 0;
            cubes[4][1] = 0;
            cubes[4][2] = side;
            cubes[4][3] = 1;
            cubes[5][0] = 0;
            cubes[5][1] = side;
            cubes[5][2] = side;
            cubes[5][3] = 1;
            boolean found = false;
            for (int i = 0; i < 6; i++) {
                boolean match = true;
                for (int j = 0; j < 8; j++) {
                    boolean found2 = false;
                    for (int k = 0; k < 4; k++) {
                        if (newPoints[j][0] == cubes[i][k * 2] && newPoints[j][1] == cubes[i][k * 2 + 1] && newPoints[j][2] == cubes[i][k * 2 + 2]) {
                            found2 = true;
                            break;
                        }
                    }
                    if (!found2) {
                        match = false;
                        break;
                    }
                }
                if (match) {
                    System.out.println("YES");
                    for (int j = 0; j < 8; j++) {
                        for (int k = 0; k < 4; k++) {
                            if (newPoints[j][0] == cubes[i][k * 2] && newPoints[j][1] == cubes[i][k * 2 + 1] && newPoints[j][2] == cubes[i][k * 2 + 2]) {
                                System.out.println(min[0] + cubes[i][k * 2] + " " + min[1] + cubes[i][k * 2 + 1] + " " + min[2] + cubes[i][k * 2 + 2]);
                                break;
                            }
                        }
                    }
                    found = true;
                    break;
                }
            }
            if (!found) {
                System.out.println("NO");
            }
        }
    }
}
