MOD = 10**9 + 7

def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def solve(n, m):
    if n > m:
        n, m = m, n
    
    fib = [0] * (n + 1)
    fib[1] = 2
    if n >= 2: 
        fib[2] = 4
    for i in range(3, n + 1):
        fib[i] = (fib[i - 1] + fib[i - 2]) % MOD
    
    total = (fib[n] + fib[m]) % MOD
    return (total - 2) % MOD

n, m = map(int, input().split())
print(solve(n, m))
