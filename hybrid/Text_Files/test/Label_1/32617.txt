import java.util.Arrays;
import java.util.PriorityQueue;
import java.util.Scanner;

public class VasyaAndTheCinema {

    static class Car implements Comparable<Car> {
        int price;
        int fuelCapacity;

        Car(int price, int fuelCapacity) {
            this.price = price;
            this.fuelCapacity = fuelCapacity;
        }

        @Override
        public int compareTo(Car other) {
            return Integer.compare(price, other.price);
        }
    }

    static class GasStation implements Comparable<GasStation> {
        int position;
        int distance;

        GasStation(int position, int distance) {
            this.position = position;
            this.distance = distance;
        }

        @Override
        public int compareTo(GasStation other) {
            return Integer.compare(position, other.position);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int s = scanner.nextInt();
        int t = scanner.nextInt();

        Car[] cars = new Car[n];
        for (int i = 0; i < n; i++) {
            cars[i] = new Car(scanner.nextInt(), scanner.nextInt());
        }

        int[] gasStations = new int[k];
        for (int i = 0; i < k; i++) {
            gasStations[i] = scanner.nextInt();
        }

        Arrays.sort(gasStations);

        PriorityQueue<GasStation> gasStationQueue = new PriorityQueue<>();
        for (int gasStation : gasStations) {
            gasStationQueue.offer(new GasStation(gasStation, gasStation));
        }

        int result = -1;
        for (Car car : cars) {
            int time = 0;
            int fuel = car.fuelCapacity;
            int lastGasStation = 0;

            while (time + s / 2 <= t && !gasStationQueue.isEmpty()) {
                GasStation nextGasStation = gasStationQueue.poll();
                int distance = nextGasStation.position - lastGasStation;

                if (distance <= fuel * 2) {
                    time += distance / 2;
                    fuel -= distance / 2;
                } else {
                    time += fuel;
                    fuel = 0;
                    lastGasStation = nextGasStation.position;
                    gasStationQueue.offer(new GasStation(lastGasStation, nextGasStation.distance - distance));
                }
            }

            if (time + s / 2 <= t) {
                result = Math.min(result == -1 ? Integer.MAX_VALUE : result, car.price);
            }
        }

        System.out.println(result);
    }
}
