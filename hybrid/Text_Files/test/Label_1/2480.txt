#include <vector>
#include <deque>
#include <unordered_map>
#include <iostream>
using namespace std;
 
int main() {
    int n;
    cin >> n;
    vector<int> g[n + 1];
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        g[u].emplace_back(v);
        g[v].emplace_back(u);
    }
    vector<int> order(n);
    for (int i = 0; i < n; i++) cin >> order[i];
    bool seen[n + 1];
    for (int i = 0; i <= n; i++) seen[i] = false;
    vector<int> path;
    path.emplace_back(order[0]);
    seen[order[0]] = true;
    deque<int> q;
    q.emplace_back(order[0]);
    bool flag = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop_front();
        int idx = -1;
        for (int i = 0; i < path.size(); i++) {
            if (path[i] == u) {
                idx = i;
                break;
            }
        }
        if (idx != -1) {
            for (auto v : g[u]) {
                if (!seen[v]) {
                    if (v == order[idx + 1]) {
                        q.emplace_back(v);
                        path.emplace_back(v);
                        seen[v] = true;
                    } else {
                        flag = false;
                        break;
                    }
                }
            }
        } else {
            flag = false;
            break;
        }
    }
    cout << (flag ? "Yes" : "No") << endl;
}
