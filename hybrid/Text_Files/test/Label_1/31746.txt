import java.util.Scanner;

public class BeautifulString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();

        for (int i = 0; i < q; i++) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();
            String s = scanner.next();

            String result = solve(n, k, s);
            System.out.println(result);

            if (result.equals("YES")) {
                String smallestBeautifulString = findSmallestBeautifulString(n, k, s);
                System.out.println(smallestBeautifulString);
            }
        }
    }

    private static String solve(int n, int k, String s) {
        for (int i = 0; i < n - k + 1; i++) {
            String substring = s.substring(i, i + k);
            if (!substring.contains("1")) {
                return "NO";
            }
        }
        return "YES";
    }

    private static String findSmallestBeautifulString(int n, int k, String s) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < k; i++) {
            boolean foundOne = false;
            for (int j = i; j < n; j += k) {
                if (s.charAt(j) == '1') {
                    foundOne = true;
                    break;
                }
            }
            sb.append(foundOne ? '1' : '0');
        }
        return sb.toString();
    }
}
