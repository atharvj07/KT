import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = scanner.nextInt();
        }
        int cars = 0;
        int remaining = 0;
        for (int i = 0; i < n; i++) {
            remaining += s[i];
            if (remaining > 4) {
                cars++;
                remaining -= 4;
            }
        }
        if (remaining > 0) {
            cars++;
        }
        System.out.println(cars);
    }
}
