#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k;
  cin >> n >> k;
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 0; i < n;) {
    int j = i + k - 1;
    if (j >= n) break;
    ans++;
    while (j < n && s[j] == s[j - 1]) j++;
    i = j + 1;
  }
  cout << ans << "\n";
  return 0;
}
