n = int(input())
prefixes_suffixes = [input().strip() for _ in range(2*n-2)]

# Find potential full strings
potential_full_strings = []
for ps in prefixes_suffixes:
    if len(ps) == n-1:
        potential_full_strings.append(ps)

# Try each potential string to see which fits the prefixes and suffixes
def check_fit(full_string, ps_list):
    prefix_set = set()
    suffix_set = set()
    for i in range(1, n):
        prefix_set.add(full_string[:i])
        suffix_set.add(full_string[-i:])
    ps_check = [''] * (2*n-2)
    for i, ps in enumerate(ps_list):
        if ps in prefix_set:
            ps_check[i] = 'P'
            prefix_set.remove(ps)
        elif ps in suffix_set:
            ps_check[i] = 'S'
            suffix_set.remove(ps)
    if '' not in ps_check:
        return ''.join(ps_check)
    else:
        return None

# Attempt to fit both potential strings
for full_string in potential_full_strings:
    # Try first potential string as prefix and second as suffix and vice versa
    full_string_a = potential_full_strings[0] + potential_full_strings[1][-1]
    full_string_b = potential_full_strings[1] + potential_full_strings[0][-1]
    result = check_fit(full_string_a, prefixes_suffixes)
    if result:
        print(result)
        break
    result = check_fit(full_string_b, prefixes_suffixes)
    if result:
        print(result)
        break
