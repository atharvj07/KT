#include<bits/stdc++.h>
using namespace std;

string a;
int n;

bool solve(){
    int open=0,close=0;
    for(int i=0;i<n;i++){
        if(a[i]=='A'){
            if(open==close) open++;
            else close++;
        }
        else if(a[i]=='B'){
            if(open>close) open--;
            else close++;
        }
        else{
            if(open) open--;
            else return 0;
        }
        if(open<0 || close<0) return 0;
    }
    return 1;
}

int main(){
    int t; cin>>t;
    while(t--){
        cin>>a;
        n=a.size();
        if(solve()) cout<<"YES";
        else cout<<"NO";
        cout<<endl;
    }
    return 0;
}
