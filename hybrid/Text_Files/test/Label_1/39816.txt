t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    a = list(map(int, input().strip().split()))
    count = [0, 0, 0]
    for ai in a:
        count[ai % 3] += 1
    # Base is the number of elements already divisible by 3
    base = count[0]
    # Minimize the difference between count[1] and count[2] by pairing them
    min_pair = min(count[1], count[2])
    base += min_pair
    count[1] -= min_pair
    count[2] -= min_pair
    # Use remaining elements of count[1] or count[2] to form new groups of 3
    base += (count[1] + count[2]) // 3
    print(base)
