import java.util.*;

public class Main {
    static boolean[] vis, onStack;
    static int n, m, t;
    static List<Integer>[] out, in;
    static Stack<Integer> stack;

    static void dfs1(int x) {
        vis[x] = true;
        onStack[x] = true;
        stack.add(x);
        for (int y : out[x]) {
            if (!vis[y]) dfs1(y);
            else if (onStack[y]) t = -1;
        }
        onStack[x] = false;
    }

    static void dfs2(int x, List<Integer> comp) {
        vis[x] = true;
        comp.add(x);
        for (int y : in[x]) {
            if (!vis[y]) dfs2(y, comp);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        vis = new boolean[n];
        onStack = new boolean[n];
        stack = new Stack<>();
        out = new List[n];
        for (int i = 0; i < n; i++) out[i] = new ArrayList<>();
        in = new List[n];
        for (int i = 0; i < n; i++) in[i] = new ArrayList<>();
        for (int i = 0; i < n; i++) if (!vis[i]) dfs1(i);
        t = stack.size();
        vis = new boolean[n];
        List<Integer>[] comps = new List[t];
        for (int i = 0; i < t; i++) comps[i] = new ArrayList<>();
        while (!stack.isEmpty()) {
            int x = stack.pop();
            if (!vis[x]) {
                dfs2(x, comps[t - 1]);
                t--;
            }
        }
        int[] c = new int[t];
        for (int i = 0; i < t; i++) {
            for (int x : comps[i]) {
                for (int y : out[x]) {
                    if (comps[i] == comps[y]) c[i]++;
                }
            }
        }
        int ans = 0;
        for (int x : c) if (x == 0) ans++;
        System.out.println(ans);
    }
}
