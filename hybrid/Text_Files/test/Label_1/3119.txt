#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1e5 + 5;

vector<int> adj[MAXN];
vector<int> path[MAXN];
vector<bool> vis;

int n, m, p;

bool dfs(int u, int par = -1) {
  vis[u] = true;
  path[p].push_back(u);
  for (int v : adj[u]) {
    if (v == par) continue;
    if (vis[v]) {
      p++;
      dfs(v, u);
      path[p].push_back(u);
      return true;
    } else {
      if (dfs(v, u)) return true;
    }
  }
  return false;
}

int main() {
  cin >> n;
  for (int i = 0; i < n - 1; i++) {
    int u, v;
    cin >> u >> v;
    u--, v--;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }

  vis.assign(n, false);
  bool ok = dfs(0);

  if (!ok)
    cout << "No" << endl;
  else {
    cout << "Yes" << endl;
    cout << p + 1 << endl;
    for (int i = 0; i <= p; i++) {
      cout << path[i].size() << " ";
      for (int u : path[i]) cout << u + 1 << " ";
      cout << endl;
    }
  }
}
