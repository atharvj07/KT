import java.util.Scanner;

public class BitCounter {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();

            long x = findXWithMaxBitCount(l, r);
            System.out.println(x);
        }
    }

    private static long findXWithMaxBitCount(long l, long r) {
        long maxBitCount = 0;
        long xWithMaxBitCount = 0;

        for (long x = l; x <= r; x++) {
            long bitCount = countBits(x);
            if (bitCount > maxBitCount) {
                maxBitCount = bitCount;
                xWithMaxBitCount = x;
            } else if (bitCount == maxBitCount && x < xWithMaxBitCount) {
                xWithMaxBitCount = x;
            }
        }

        return xWithMaxBitCount;
    }

    private static long countBits(long x) {
        long bitCount = 0;
        while (x > 0) {
            bitCount += x & 1;
            x >>= 1;
        }
        return bitCount;
    }
}
