#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end(), [&](int x, int y) { return x - y > abs(x - y); });
    for (int x : a) cout << x << " ";
    cout << "\n";
  }
  return 0;
}
