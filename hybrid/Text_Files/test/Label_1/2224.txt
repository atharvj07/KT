#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int n = s.size();
  vector<int> pref(n + 1);
  for (int i = 1; i <= n; ++i) {
    pref[i] = pref[i - 1] + (s[i - 1] != s[i % n]);
  }
  int ans = 0;
  for (int i = 0; i < n; ++i) {
    ans = max(ans, pref[i + n] - pref[i]);
  }
  cout << ans << "\n";
  return 0;
}
