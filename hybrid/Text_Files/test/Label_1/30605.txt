import java.util.*;
import java.io.*;

public class Main {
    private static int[] c;
    private static boolean[] used;
    private static ArrayList<Integer> ans1, ans2;

    private static void dfs(int v, int sum1, int sum2, int need1, int need2) {
        if (sum1 > need1 || sum2 > need2) {
            return;
        }
        if (sum1 == need1 && sum2 == need2) {
            ans1 = new ArrayList<>();
            ans2 = new ArrayList<>();
            for (int i = 0; i < c.length; i++) {
                if (used[i]) {
                    if (sum1 > 0) {
                        ans1.add(i + 1);
                    } else {
                        ans2.add(i + 1);
                    }
                }
                sum1 -= c[i] / 3;
                sum2 -= c[i] / 2;
            }
            return;
        }
        if (v == c.length) {
            return;
        }
        used[v] = true;
        dfs(v + 1, sum1 + c[v] / 3, sum2, need1, need2);
        used[v] = false;
        dfs(v + 1, sum1, sum2 + c[v] / 2, need1, need2);
        used[v] = false;
        dfs(v + 1, sum1, sum2, need1, need2);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int need1 = in.nextInt();
        int need2 = in.nextInt();
        c = new int[n];
        used = new boolean[n];
        for (int i = 0; i < n; i++) {
            c[i] = in.nextInt();
        }
        dfs(0, 0, 0, need1, need2);
        if (ans1 == null) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            System.out.println(ans1.size() + " " + ans2.size());
            for (Integer i : ans1) {
                System.out.print(i + " ");
            }
            System.out.println();
            for (Integer i : ans2) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
