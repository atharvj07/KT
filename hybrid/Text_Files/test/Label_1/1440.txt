#include <bits/stdc++.h>
#define fastio ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
using namespace std;
typedef long long ll;
int main() {
	fastio;
	ll n,x; cin>>n;
	vector<ll>a(n),l(n,5e8),r(n,5e8);
	for(ll i=0;i<n;i++) {
		cin>>a[i];
		if(a[i]==0) {
			l[i]=0; r[i]=0;
		}
	}
	for(ll i=1;i<n;i++) {
		l[i]=l[i-1];
		if(a[i]==0) l[i]=0;
		else l[i]=min(l[i],l[i-1]+1);
	}
	for(ll i=n-2;i>=0;i--) {
		r[i]=r[i+1];
		if(a[i]==0) r[i]=0;
		else r[i]=min(r[i],r[i+1]+1);
	}
	for(ll i=0;i<n;i++) cout<<min(l[i],r[i])<<" ";
	return 0;
}
