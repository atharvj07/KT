import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    int m = scanner.nextInt();
    int[][] buses = new int[n][3];
    for (int i = 0; i < n; i++) {
      buses[i][0] = scanner.nextInt();
      buses[i][1] = scanner.nextInt();
      buses[i][2] = scanner.nextInt();
    }
    int[][] people = new int[m][3];
    for (int i = 0; i < m; i++) {
      people[i][0] = scanner.nextInt();
      people[i][1] = scanner.nextInt();
      people[i][2] = scanner.nextInt();
    }
    int[] result = new int[m];
    for (int i = 0; i < m; i++) {
      int currentBus = -1;
      int minTime = Integer.MAX_VALUE;
      for (int j = 0; j < n; j++) {
        if (buses[j][0] <= people[i][0] && buses[j][1] >= people[i][1] && buses[j][2] >= people[i][2] && buses[j][2] < minTime) {
          minTime = buses[j][2];
          currentBus = j + 1;
        }
      }
      result[i] = currentBus;
    }
    for (int i = 0; i < m; i++) {
      System.out.print(result[i] + " ");
    }
    System.out.println();
  }
}
