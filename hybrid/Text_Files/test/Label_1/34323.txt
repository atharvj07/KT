import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), k = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        Arrays.sort(a);
        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (cnt == 0 || a[i] - a[cnt - 1] <= 5) {
                cnt++;
            } else {
                ans += cnt;
                cnt = 1;
            }
        }
        ans += cnt;
        System.out.println(ans);
    }
}
