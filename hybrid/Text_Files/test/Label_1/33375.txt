import java.io.*;
import java.util.*;

public class Main {
    static int n, m;
    static int[] a, ans;
    static ArrayList<int[]> type1;
    static ArrayList<int[]> type2;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        a = new int[n + 1];
        ans = new int[n + 1];
        type1 = new ArrayList<>();
        type2 = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(in.readLine());
            int type = Integer.parseInt(st.nextToken());
            if (type == 1) {
                int l = Integer.parseInt(st.nextToken());
                int r = Integer.parseInt(st.nextToken());
                int x = Integer.parseInt(st.nextToken());
                type1.add(new int[]{l, r, x});
            } else {
                int k = Integer.parseInt(st.nextToken());
                int d = Integer.parseInt(st.nextToken());
                type2.add(new int[]{k, d});
            }
        }
        if (check()) {
            System.out.println("YES");
            for (int i = 1; i <= n; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        } else {
            System.out.println("NO");
        }
    }

    static boolean check() {
        for (int[] op : type2) {
            a[op[0]] = op[1];
        }
        int cur = 0;
        for (int[] op : type1) {
            int l = op[0];
            int r = op[1];
            int x = op[2];
            if (x > cur) {
                return false;
            }
            for (int i = l; i <= r; i++) {
                if (a[i] < x) {
                    return false;
                }
            }
            cur = x;
        }
        int[] dp = new int[n + 1];
        for (int i = n; i >= 1; i--) {
            dp[i] |= a[i];
            for (int[] op : type1) {
                if (op[0] <= i && i <= op[1]) {
                    dp[i] &= op[2];
                }
            }
        }
        for (int i = 1; i <= n; i++) {
            ans[i] = dp[i];
        }
        return true;
    }
}
