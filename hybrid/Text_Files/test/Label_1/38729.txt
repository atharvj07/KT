import sys, math
n = int(input())
a = [int(x) for x in input().split()]
if max(a)> n - 1:
    print(-1)
    sys.exit(0)
b = [0] * n
b[0] = a[0]
for i in range(1, n):
    if a[i] == a[0]:
        b[i] = 0
    elif a[i] == a[0] + 1:
        b[i] = 1
    elif a[i] == a[0] - 1:
        b[i] = -1
    else:
        print(-1)
        sys.exit(0)
for i in range(1, n):
    for j in range(i+1, n):
        if abs(i-j) == abs(b[i]-b[j]):
            print(-1)
            sys.exit(0)
for i in range(1, n):
    if b[i] > b[i-1] + 1:
        print(-1)
        sys.exit(0)
m = b.count(0) + 1
for i in range(1, n):
    if b[i] == b[i-1]:
        m -= 1
print(n, m)
print(b.index(0)+1, b.count(0)+1)
