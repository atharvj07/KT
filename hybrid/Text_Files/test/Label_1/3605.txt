#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    vector<int> breaks;
    for (int i = 1; i < n - 1; ++i) {
        if (s[i] == '0') breaks.push_back(i);
    }
    int ans = n;
    int l = 0, r = breaks.size() - 1;
    while (l <= r) {
        int mid = (l + r) / 2;
        if (k - 2 - mid <= breaks[mid]) {
            ans = mid;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    cout << ans << endl;
    return 0;
}
