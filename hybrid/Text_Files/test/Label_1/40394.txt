a, b, n = map(int, input().split())
t = [0] * n
for i in range(n):
    t[i] = int(input())
ans = 0
assert a + 2 * b >= n
for i in range(n):
    if t[i] == 1:
        if a > 0:
            a -= 1
        elif b > 0:
            b -= 1
        elif b > 0 and t[i + 1 if i + 1 < n else -1] == 1:
            b -= 1
            i += 1 # this is required!!!
        else:
            ans += 1
    elif t[i] == 2:
        if b > 0:
            b -= 1
        else:
            ans += 2
print(ans)
