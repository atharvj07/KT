#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, h;
    cin >> n >> h;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] <= h) ans += 1;
        else ans += 2;
    }
    cout << ans << endl;
    return 0;
}
