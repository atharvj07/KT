import sys, collections

def main():
    h = int(input())
    a = list(map(int, input().split()))
    if h == 2 and a[0] == 1 and a[1] == 1 and a[2] == 1:
        print("perfect")
    else:
        print("ambiguous")
        new_a = []
        for x in a[1:]:
            new_a.append(x - 1)
        for x in a[1:]:
            new_a.append(x)
        print(sum(a), *[0] + a)
        print(sum(new_a), *[0] + new_a)
main()
