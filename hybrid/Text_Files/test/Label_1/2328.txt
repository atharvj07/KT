#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> con;
int n, d, k;
bool vis[400001];

void dfs(int now, int p, int dep) {
  vis[now] = true;
  if (dep > d) {
    cout << "NO" << endl;
    exit(0);
  }
  for (int x : con[now]) {
    if (x == p) continue;
    if (vis[x]) {
      cout << "NO" << endl;
      exit(0);
    }
    dfs(x, now, dep + 1);
  }
}

int main() {
  cin >> n >> d >> k;
  d++;
  con.resize(n + 1);
  if (k > 2 * d - 1) k = 2 * d - 1;
  if (k * (k + 1) < n - 1 || k * (k - 1) > n - 1) {
    cout << "NO" << endl;
    return 0;
  }
  cout << "YES" << endl;
  int m = (n - 1 - k * (k - 1)) / 2, t = k - 1, p = k;
  while (m--) {
    con[p].push_back(t);
    con[t].push_back(p);
    p++;
    if (p > n) {
      t++;
      p = t;
    }
  }
  m = n - 1 - m;
  while (m--) {
    con[p].push_back(t);
    con[t].push_back(p);
    p++;
    t--;
    if (p > n) {
      p = t;
      t--;
    }
  }
  for (int i = 1; i <= n; i++) {
    if (vis[i]) continue;
    dfs(i, -1, 1);
  }
  for (int i = 1; i <= n; i++) {
    for (int x : con[i]) if (x > i) cout << i << ' ' << x << endl;
  }
  return 0;
}
