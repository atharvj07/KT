import sys
input = sys.stdin.readline

n = int(input())
s1 = map(int, input().split())
s2 = map(int, input().split())
skills = [0 for i in range(60)]
for i in s1:
    for j in range(60):
        if i & (1 << j):
            skills[j] += 1

res = 0
for j in range(60):
    if skills[j] >= 2:
        res = max(res, sum(i for i in s2 if i & (1 << j)))

print(res)
