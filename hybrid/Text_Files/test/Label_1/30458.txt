import java.util.*;
public class Main {
    static ArrayList<Integer>[] adj;
    static boolean[] vis;
    static boolean flg, vis2;
    static int[] ans;
    static int idx;
    public static void dfs1(int u) {
        vis[u] = true;
        for(int v : adj[u]) {
            if(!vis[v])
                dfs1(v);
        }
    }
    public static void dfs2(int u) {
        vis2 = true;
        ans[idx++] = u;
        for(int v : adj[u]) {
            if(!vis2)
                dfs2(v);
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), u, v, x = 0;
        adj = new ArrayList[n + 1];
        for(int i = 0; i <= n; i++)
            adj[i] = new ArrayList<>();
        vis = new boolean[n + 1];
        boolean[] as = new boolean[n + 1];
        for(int i = 0; i < m; i++) {
            u = sc.nextInt(); v = sc.nextInt();
            if(sc.nextInt() == 1)
                as[u] = as[v] = true;
            adj[u].add(v); adj[v].add(u);
        }
        for(int i = 1; i <= n; i++) {
            if(!vis[i]) {
                dfs1(i);
                x++;
            }
        }
        if(x != 1) {
            System.out.println("Impossible");
            return;
        }
        int nd = 0;
        for(int i = 1; i <= n; i++) {
            if(as[i])
                nd++;
        }
        if((x + nd - 1) % 2 != 0) {
            System.out.println("Impossible");
            return;
        }
        x = (x + nd - 1) / 2;
        ans = new int[x];
        for(int i = 1; i <= n; i++) {
            if(as[i]) {
                if(!vis2) {
                    idx = 0;
                    dfs2(i);
                    x--;
                }
            } else {
                if(!vis2) {
                    idx = 0;
                    dfs2(i);
                    flg = true;
                    x--;
                } else if(x > 0) {
                    ans[idx++] = i;
                    x--;
                }
            }
        }
        if(flg) {
            for(int i = idx - 1; i >= 0; i--)
                ans[x++] = ans[i];
        }
        System.out.println(ans.length);
        for(int i = 0; i < ans.length; i++)
            System.out.print(ans[i] + " ");
        System.out.println();
    }
}
