def solve_nim(n, x, p):
    dp = [0.0] * (1 << (x + 1))
    dp[0] = 1.0
    for mask in range(1, 1 << (x + 1)):
        sum_p = 0.0
        for i in range(1, x + 1):
            if mask & (1 << i):
                sum_p += p[i]
        if sum_p == 0:
            continue
        for i in range(1, x + 1):
            if mask & (1 << i):
                dp[mask] += p[i] * dp[mask ^ (1 << i)] / sum_p
    res = 0.0
    for mask in range(1, 1 << (x + 1)):
        bit_count = bin(mask).count("1")
        if bit_count % 2 == n % 2:
            res += dp[mask] * p[0] ** (n - bit_count)
    return res

n, x = map(int, input().split())
p = list(map(float, input().split()))
print("{:.6f}".format(solve_nim(n, x, p)))
