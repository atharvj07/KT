n, k = map(int, input().split())
waves = [list(map(int, input().split())) for _ in range(n)]

total_bullets = 0
remaining_bullets = k

for i in range(n):
    l, r, a = waves[i]

    # Time available to shoot and reload
    time_available = r - l + 1

    # If remaining bullets are not enough, use them and reload
    if remaining_bullets < a:
        a -= remaining_bullets
        total_bullets += remaining_bullets  # Use remaining bullets
        remaining_bullets = k  # Reload
        total_bullets += (a // k) * k  # Use full magazines
        reloads = a // k
        if a % k != 0:
            total_bullets += k  # Last reload might not be full
            remaining_bullets = k - (a % k)  # Remaining bullets after last reload
        else:
            remaining_bullets = k  # If it divides exactly, full magazine remains
        if reloads >= time_available:
            total_bullets = -1
            break
    else:
        remaining_bullets -= a
        total_bullets += a
        if remaining_bullets == 0 and i != n - 1:  # Reload if magazine is empty and it's not the last wave
            remaining_bullets = k
            total_bullets += k

print(total_bullets)
