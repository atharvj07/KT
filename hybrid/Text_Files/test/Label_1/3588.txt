#include <iostream>
#include <string>

using namespace std;

int main() {
  string good, pattern, query;
  cin >> good >> pattern >> query;
  while (query.size()) {
    bool matched = true;
    for (int i = 0, j = 0; i < query.size() && j < pattern.size(); i++) {
      if (pattern[j] == '*') {
        if (good.find(query[i]) != string::npos) {
          matched = false;
          break;
        }
        ++j;
        while (pattern[j] == '*') ++j;
        if (j >= pattern.size()) break;
        if (pattern[j] != query[i]) {
          matched = false;
          break;
        }
      } else
        matched &= pattern[j++] == query[i] || pattern[j - 1] == '?';
    }
    if (matched)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
    getline(cin, query);
  }
  return 0;
}
