import java.util.Scanner;

public class PuddingMonsters {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        // Create a boolean array to track the positions of monsters.
        boolean[][] monsters = new boolean[n][n];

        // Read the coordinates of the monsters.
        for (int i = 0; i < n; i++) {
            int row = scanner.nextInt() - 1;
            int col = scanner.nextInt() - 1;
            monsters[row][col] = true;
        }

        // Calculate the number of square fragments.
        int count = 0;
        for (int k = 1; k <= n; k++) {
            for (int row = 0; row <= n - k; row++) {
                for (int col = 0; col <= n - k; col++) {
                    // Check if the square fragment contains exactly k monsters.
                    int monsterCount = 0;
                    for (int i = row; i < row + k; i++) {
                        for (int j = col; j < col + k; j++) {
                            if (monsters[i][j]) {
                                monsterCount++;
                            }
                        }
                    }

                    if (monsterCount == k) {
                        count++;
                    }
                }
            }
        }

        // Print the result.
        System.out.println(count);
    }
}
