import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        int maxLen = 0;
        int curLen = 0;
        for (char c : s.toCharArray()) {
            if (c == 'A' || c == 'C' || c == 'G' || c == 'T') {
                curLen++;
            } else {
                maxLen = Math.max(maxLen, curLen);
                curLen = 0;
            }
        }
        maxLen = Math.max(maxLen, curLen);
        System.out.println(maxLen);
    }
}
