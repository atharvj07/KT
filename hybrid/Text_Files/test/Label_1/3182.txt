#include <iostream>

using namespace std;

const int maxn = 1e5;
int n, a[maxn], dp[maxn], s[maxn];

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    string m;
    cin >> m;
    int tmp = 0;
    for (int i = n - 1; i >= 0; i--) {
        tmp = tmp * 2 + (m[i] - '0');
        s[i] = tmp;
    }
    dp[0] = 0;
    for (int i = 1; i < (1 << n); i++) {
        dp[i] = dp[i ^ (1 << __builtin_ctz(i))] + a[__builtin_ctz(i)];
    }
    tmp = 0;
    for (int i = 0; i < n; i++) {
        tmp = max(tmp, dp[s[i]]);
    }
    cout << tmp << endl;
}
