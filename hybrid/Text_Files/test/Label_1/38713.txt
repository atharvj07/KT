import sys

mod = 998244353
inv2 = pow(2, mod - 2, mod)

n = int(input())

lists = []
sum_size = 0
for _ in range(n):
    k_i, *a_i = list(map(int, input().split()))
    lists.append(a_i)
    sum_size += k_i

a = sum_size * inv2 % mod
b = sum_size * (pow(2, sum_size - 1, mod) - 1) % mod
c = sum([len(set(l)) for l in lists])

ans = (a * b - c) % mod
print((ans + mod) % mod)
