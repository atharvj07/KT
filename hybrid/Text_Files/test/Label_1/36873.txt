n, m = map(int, input().split())
matrix = [list(map(int, input().split())) for _ in range(n)]

xor_sum = 0
for i in range(n):
    xor_sum ^= matrix[i][0]

if xor_sum > 0:
    print("TAK")
    print(" ".join(['1' for _ in range(n)]))
else:
    answer = ['1' for _ in range(n)]
    found = False
    for i in range(n):
        for j in range(1, m):
            if matrix[i][0] ^ matrix[i][j] ^ xor_sum > 0:
                answer[i] = str(j + 1)
                found = True
                break
        if found:
            break
    if found:
        print("TAK")
        print(" ".join(answer))
    else:
        print("NIE")
