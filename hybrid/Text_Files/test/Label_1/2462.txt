#include<bits/stdc++.h>
using namespace std;
int main()
{
  int n,k;
  cin>>n>>k;
  int a[n];
  for(int i=0;i<n;i++)
  {
    cin>>a[i];
  }
  int ans=0,temp=0;
  for(int i=0;i<n;i++)
  {
    for(int j=1;j<=a[i];j++)
    {
      temp++;
      if(temp%k==0)
      {
        int x=((temp*100)/n);
        if(x==j)
        {
          ans++;
          break;
        }
      }
    }
  }
  cout<<ans<<endl;
  return 0;
}
