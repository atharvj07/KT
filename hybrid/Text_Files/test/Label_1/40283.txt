for _ in range(int(input())):
    n = int(input())
    s = input()
    x, y = 0, 0
    mn = -1e9
    # Step 1: find the shortest ending segment that can be eliminated
    for i in range(n):
        if s[i] == 'L':
            x -= 1
        elif s[i] == 'R':
            x += 1
        elif s[i] == 'U':
            y += 1
        else:
            y -= 1
        if not x and not y:
            mn = i + 1
    if mn == -1e9:
        print(-1)
        continue
    # Step 2: find the longest beginning segment that can be eliminated
    l = 0
    x, y = 0, 0
    for r in range(n):
        if s[r] == 'L':
            x -= 1
        elif s[r] == 'R':
            x += 1
        elif s[r] == 'U':
            y += 1
        else:
            y -= 1
        if x == 0 and y == 0:
            if r - l + 1 > mn:
                mn = r - l + 1
                l = r + 1
    print(l + 1, l + mn)
