#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        long long x, y;
        cin >> x >> y;
        long long ans = 0;
        for (long long i = 1; i <= x; i++) {
            for (long long j = 1; j <= y; j++) {
                if (i % j == i / j) { ans++; }
            }
        }
        cout << ans << endl;
    }
    return 0;
}
