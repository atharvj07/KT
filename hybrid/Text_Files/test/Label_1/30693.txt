import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        boolean[] used = new boolean[n];
        boolean[][] g = new boolean[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (gcd(a[i], a[j]) > 1) {
                    g[i][j] = g[j][i] = true;
                    used[i] = used[j] = true;
                }
            }
        }
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (used[i]) {
                cnt++;
            }
        }
        if (cnt == n - 1) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }

    private static int gcd(int a, int b) {
        while (b > 0) {
            int tmp = a;
            a = b;
            b = tmp % b;
        }
        return a;
    }
}
