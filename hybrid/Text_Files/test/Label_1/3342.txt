#include <iostream>
using namespace std;

int main() {
  int n;
  cin >> n;
  char a[n][n];
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) { cin >> a[i][j]; }
  }
  bool valid = 1;
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      int cnt = 0;
      if (i > 0 && a[i - 1][j] == 'o') ++cnt;
      if (i < n - 1 && a[i + 1][j] == 'o') ++cnt;
      if (j > 0 && a[i][j - 1] == 'o') ++cnt;
      if (j < n - 1 && a[i][j + 1] == 'o') ++cnt;
      valid &= !(cnt & 1);
    }
  }
  cout << (valid ? "YES" : "NO") << endl;
}
