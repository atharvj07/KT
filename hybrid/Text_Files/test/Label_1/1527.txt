#include <bits/stdc++.h>
using namespace std;
#define int long long
const int mod = 1e9 + 7;
int n, cnt[200001], sub[200001], ans, cen;
vector<int> adj[200001];
void dfs(int node, int par) {
  sub[node] = 1;
  for (auto x : adj[node]) {
    if (x != par) {
      dfs(x, node);
      sub[node] += sub[x];
    }
  }
}
void dfs2(int node, int par, int sz) {
  ans = max({ans, sz - sub[node], sub[node]});
  for (auto x : adj[node]) {
    if (x != par) {
      dfs2(x, node, sz);
    }
  }
}
int centroid(int node, int par) {
  dfs(node, par);
  int sz = sub[node];
  for (auto x : adj[node]) {
    if (x != par) {
      dfs2(x, node, sz);
    }
  }
  for (auto x : adj[node]) {
    if (x != par && max({ans, sz - sub[x], sub[x]}) < (sz + 1) / 2) {
      return centroid(x, node);
    }
  }
  return node;
}
void dfs3(int node, int par, int dist) {
  cnt[dist]++;
  for (auto x : adj[node]) {
    if (x != par) dfs3(x, node, dist + 1);
  }
}
int cal(int node) {
  int max_dist = 0;
  int max_cnt = 0;
  for (auto x : adj[node]) {
    dfs3(x, node, 1);
    if (cnt[max_dist] < cnt[1]) {
      max_dist = 1;
      max_cnt = cnt[1];
    }
    if (cnt[max_dist] == cnt[1]) max_cnt += cnt[1];
    for (int i = 2; i <= n; i++) {
      if (cnt[max_dist] < cnt[i]) {
        max_dist = i;
        max_cnt = cnt[i];
      }
      if (cnt[max_dist] == cnt[i]) max_cnt += cnt[i];
    }
    for (int i = 1; i <= n; i++) cnt[i] = 0;
  }
  return max_cnt - 1;
}
signed main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> n;
  for (int i = 1; i < n; i++) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
  }
  cen = centroid(1, 0);
  cout << cal(cen) << endl;
}
