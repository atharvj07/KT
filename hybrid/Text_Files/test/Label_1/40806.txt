for _ in range(int(input())):
    n = int(input())
    s = input()
    # find the first pair of opposite directions that cancel each other
    # if not found, print -1
    L = [0] * (n + 1)
    R = [0] * (n + 1)
    for i in range(n):
        L[i + 1] = L[i]
        R[i + 1] = R[i]
        if s[i] == 'R': R[i + 1] += 1
        elif s[i] == 'L': L[i + 1] -= 1
        if R[i] > 0 and L[i + 1] < 0:
            print(i, i + 1)
            break
        elif R[i + 1] > 0 and L[i] < 0:
            print(i + 1, i + 1)
            break
    else: print(-1)
