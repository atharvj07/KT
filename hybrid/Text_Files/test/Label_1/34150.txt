import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[] s = new int[n];
        for (int i = 0; i < n; i++) {
            s[i] = a[i];
            if (i > 0) {
                s[i] += s[i - 1];
            }
        }
        for (int i = 0; i < k; i++) {
            for (int j = 0; j < n; j++) {
                a[j] = s[j];
                if (j > 0) {
                    a[j] -= s[j - 1];
                }
            }
            for (int j = 0; j < n; j++) {
                s[j] = a[j];
                if (j > 0) {
                    s[j] += s[j - 1];
                }
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(a[i] + " ");
        }
    }
}
