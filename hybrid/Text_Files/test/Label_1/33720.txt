import java.util.Scanner;

public class Main {
	static long[] pow = new long[5000001];

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		long x = sc.nextLong();
		pow[0] = 1;
		for (int i = 1; i <= n; i++) pow[i] = pow[i - 1] * 10;
		long l = 2, r = x, mid;
		while (l <= r) {
			mid = (l + r) / 2;
			if (rotatable(n, mid))
				l = mid + 1;
			else
				r = mid - 1;
		}
		if (rotatable(n, r)) System.out.println(r);
		else System.out.println(-1);
	}

	static boolean rotatable(int n, long b) {
		boolean[] vis = new boolean[n];
		long num = pow[n - 1];
		long bx = 1;
		for (int i = 0; i < n; i++) {
			long cur = num / bx;
			for (int j = 0; j < n; j++) {
				if (!vis[j] && cur == b * j) {
					vis[j] = true;
					num = num % bx + bx * cur;
					break;
				}
			}
			bx *= 10;
			num /= 10;
		}
		for (boolean v : vis) if (!v)
			return false;
		return true;
	}
}
