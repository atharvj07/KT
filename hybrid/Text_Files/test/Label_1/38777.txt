import sys
input = sys.stdin.readline

n, m, k, q = map(int, input().split())
r = []
c = []
for _ in range(k):
    ri, ci = map(int, input().split())
    r.append(ri-1)
    c.append(ci-1)
b = list(map(int, input().split()))
b.sort()

ans = len(r) * m + 1
for bi in b:
    bi -= 1
    pos = []
    for i in range(k):
        if c[i] == bi:
            pos.append(r[i])
    if len(pos) == 0:
        continue
    
    pos.sort()
    cur = 0
    for pi in pos:
        cur += pi
    
    ans = min(ans, cur)

print(ans)
