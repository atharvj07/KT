import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());

        Map<String, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            int x = Integer.parseInt(line[0]);
            int y = Integer.parseInt(line[1]);
            String direction = line[2];

            String key = getKey(x, y, direction);
            if (!map.containsKey(key)) {
                map.put(key, new ArrayList<>());
            }
            map.get(key).add(i + 1);
        }

        long minTime = Long.MAX_VALUE;
        boolean collision = false;
        for (String key : map.keySet()) {
            List<Integer> airplanes = map.get(key);
            if (airplanes.size() >= 2) {
                collision = true;

                long time = Long.MAX_VALUE;
                for (int i = 0; i < airplanes.size() - 1; i++) {
                    for (int j = i + 1; j < airplanes.size(); j++) {
                        long t = calculateTime(airplanes.get(i), airplanes.get(j));
                        if (t < time) {
                            time = t;
                        }
                    }
                }

                if (time < minTime) {
                    minTime = time;
                }
            }
        }

        if (collision) {
            System.out.println(minTime);
        } else {
            System.out.println("SAFE");
        }
    }

    private static String getKey(int x, int y, String direction) {
        return x + " " + y + " " + direction;
    }

    private static long calculateTime(int a, int b) {
        return Math.abs(a - b);
    }
}
