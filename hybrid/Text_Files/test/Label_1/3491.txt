#include <bits/stdc++.h>
using namespace std;
#define ll long long
vector<ll>v;
ll gcd(ll a, ll b) {
    if(b == 0) return a;
    return gcd(b, a % b);
}
int main() {
    ll n, g = 0;
    cin >> n;
    v.resize(n);
    for(auto &x : v) {
        cin >> x;
        g = __gcd(g, x);
    }
    if(g == 1)
        cout << -1 << endl;
    else {
        for(ll i = 1; i * i <= g; ++i) {
            if(g % i == 0) {
                ll x = g / i;
                if(x != i)
                    cout << n - 1 << endl;
                else
                    cout << n - 2 << endl;
                break;
            }
        }
    }
}
