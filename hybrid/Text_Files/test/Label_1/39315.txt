import sys
input = lambda: sys.stdin.readline().strip()
n,m = map(int,input().split())
a = list(map(int,input().split()))
c = [0]*m
for i in map(int,input().split()):
    c[i-1]+=1
d = int(input())
l = [int(input())-1 for i in range(d)]
msk = [0]*n
ans = []
for x in l:
    x-=1
    msk[x]=1
    for j in range(m):
        c[j]-=msk[j]
    s = []
    for j in range(m):
        if c[j]>0:
            s.append(a[j])
    if s:
        ans.append(min([i for i in s if i not in ans]))
    else:
        ans.append(0)
    print(ans[-1])
