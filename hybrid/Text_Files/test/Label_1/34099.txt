import java.util.*;
import java.io.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    sc.close();

    int[] left = new int[n]; // left[i] = smallest ai such that a[i] >= ai
    int[] right = new int[n]; // right[i] = largest ai such that a[i] >= ai
    left[0] = 0;
    for (int i = 1; i < n; i++) {
      left[i] = a[i] >= 1 ? 0 : left[i - 1] + 1;
    }
    right[n - 1] = n - 1;
    for (int i = n - 2; i >= 0; i--) {
      right[i] = a[i] >= 1 ? n - 1 : right[i + 1] - 1;
    }

    int ans = 0;
    int cur = 0;
    for (int i = 1; i < n; i++) {
      int l = left[i];
      int r = right[i];
      if (l >= cur) {
        ans++;
        cur = i;
      } else if (r >= cur) {
        int mid = (l + r) / 2;
        if (a[mid] >= i - cur) {
          ans++;
          cur = mid;
        }
      }
    }
    System.out.println(ans);
  }
}
