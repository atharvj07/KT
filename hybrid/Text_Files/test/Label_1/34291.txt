import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String s = br.readLine();
        int k = Integer.parseInt(br.readLine());
        int minDeletions = Integer.MAX_VALUE;
        Map<Character, List<Character>> map = new HashMap<>();
        for (int i = 0; i < k; i++) {
            String forbidden = br.readLine();
            map.computeIfAbsent(forbidden.charAt(0), c -> new ArrayList<>()).add(forbidden.charAt(1));
            map.computeIfAbsent(forbidden.charAt(1), c -> new ArrayList<>()).add(forbidden.charAt(0));
        }

        int n = s.length();
        int left = 0, right = 0;
        int[] dp = new int[n + 1];
        Arrays.fill(dp, -1);

        dp[0] = 0;
        while (right < n) {
            if (!isValid(map, s.substring(left, right + 1))) {
                if (dp[left] == -1) continue;
                dp[right + 1] = dp[left] + 1;
                left++;
            } else {
                dp[right + 1] = dp[left];
            }
            if (dp[right + 1] != -1) {
                minDeletions = Math.min(minDeletions, dp[right + 1]);
            }
            right++;
        }

        System.out.println(minDeletions);
    }

    private static boolean isValid(Map<Character, List<Character>> map, String s) {
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (map.containsKey(c)) {
                for (char f : map.get(c)) {
                    if (s.contains(String.valueOf(f))) {
                        return false;
                    }
                }
            }
        }
        return true;
    }
}
