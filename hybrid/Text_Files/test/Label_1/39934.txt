n, m, k = map(int, input().split())
a = list(map(int, input().split()))
ops = [list(map(int, input().split())) for _ in range(m)]
query = [list(map(int, input().split())) for _ in range(k)]

# Initialize an array to record the effect of each operation
operation_effect = [0] * (n + 1)

# For each query, mark the effect of operations in a cumulative way
for x, y in query:
    for i in range(x - 1, y):
        l, r, d = ops[i]
        operation_effect[l - 1] += d
        operation_effect[r] -= d

# Apply the cumulative effect to the array
for i in range(1, n):
    operation_effect[i] += operation_effect[i - 1]

for i in range(n):
    a[i] += operation_effect[i]

print(" ".join(map(str, a)))
