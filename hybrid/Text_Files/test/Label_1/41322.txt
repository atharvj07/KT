n, m = map(int, input().split())
s_x = [[1] for i in range(m+1)]
for i in range(n):
    sx, x = map(int, input().split())
    s_x[sx].append(x)
for i in range(1, m+1):
    s_x[i].sort()
ans = 0
for i in range(2, m+1):
    l, r = len(s_x[i-1]), len(s_x[i])
    if l > r:
        ans += max(l-r-1, 0)
        s_x[i-1] = s_x[i-1][max(l-r-1, 0):]
    elif l < r:
        ans += max(r-l-1, 0)
        s_x[i] = s_x[i][max(r-l-1, 0):]
    j = 0
    for x in s_x[i]:
        while j < len(s_x[i-1]) and x > s_x[i-1][j]:
            j += 1
        if j == len(s_x[i-1]):
            ans += 1
        else:
            s_x[i-1][j] = x
            j += 1
print(ans)
