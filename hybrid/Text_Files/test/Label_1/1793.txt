#include <iostream>

using namespace std;

int main() {
  long long n;
  cin >> n;
  int ans = 1;
  while (n % 10 == 0 && ans * 10 <= 1000000000LL) {
    n /= 10;
    ans *= 10;
  }
  n++;
  while ((n % 10 == 0 && n != 0) || n != 0) {
    n = n / 10;
    ans++;
  }
  cout << ans << endl;
  return 0;
}
