def time_to_serve(s):
    row, seat = int(s[:-1]), s[-1]
    time_per_row = 6
    move_time = 1
    seat_order = {'f': 1, 'e': 2, 'd': 3, 'a': 4, 'b': 5, 'c': 6}
    
    row_moves = ((row - 1) // 4) * 2 + (1 if row % 4 in [2, 3] else 0)
    time_to_reach_row = row_moves * move_time + ((row - 1) // 2) * time_per_row
    
    return time_to_reach_row + seat_order[seat]

s = input().strip()
print(time_to_serve(s))
