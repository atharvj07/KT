dig9 = [1<<i for i in range(10)]
add = [dig9[int(c)] for c in input()]
s = 81
for i in range(len(add)-1,0,-1):
    add[i-1] += add[i]>>1
    add[i] &= 1
    s = (s+s)%dig9[10]
    if add[i]:
        if add[i-1]&dig9[10-s]:
            add[i-1] ^= dig9[1]
        else:
            add[i-1] ^= dig9[10-s-1]
            s = s-1
print(dig9.index(add[0]))
