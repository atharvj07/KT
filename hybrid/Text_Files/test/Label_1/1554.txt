#include <bits/stdc++.h>

using namespace std;

int n, m, p, x, y, ans = 0;
char b[1005][1005];

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  cout.tie(nullptr);

  cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    cin >> y >> x;
    b[x][y] = 'x';
  }
  for (int i = 1; i <= n; i++) {
    if (b[i][1] == 'x' && b[i][n] == 'x') continue;
    if (b[1][i] == 'x' && b[n][i] == 'x') continue;
    p++;
  }
  cout << p << "\n";
  return 0;
}
