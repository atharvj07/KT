#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int& x : a) cin >> x;
    int ans = INT_MAX, t = -1;
    for (int i = 1; i <= 100; i++) {
        int cost = 0;
        for (int x : a) cost += min({abs(x - i), abs(x - i - 1)});
        if (cost < ans) {
            ans = cost;
            t = i;
        }
    }
    cout << t << " " << ans << endl;
}
