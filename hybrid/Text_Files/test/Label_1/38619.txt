t = int(input().strip())
for _ in range(t):
    n, m, a, b = map(int, input().strip().split())
    s = list(map(int, input().strip().split()))

    s.sort(reverse=True)
    distance = abs(a - b)
    time_to_meet = distance - 1
    hooligan_direction = -1 if a > b else 1

    max_exploded = 0
    current_time = 1
    for firecracker_time in s:
        if current_time + firecracker_time <= time_to_meet:
            max_exploded += 1
            current_time += 1
        else:
            break

    print(max_exploded)
