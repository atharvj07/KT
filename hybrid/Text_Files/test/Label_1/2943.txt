#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> m(n);
    for (int i = 0; i < n; i++) {
        cin >> m[i];
    }
    vector<int> a(n, -1);
    a[0] = m[0];
    for (int i = 1; i < n; i++) {
        if (a[i - 1] == m[i]) {
            a[i] = m[i] - 1;
        } else {
            a[i] = a[i - 1];
        }
    }
    for (int i = n - 2; i >= 0; i--) {
        a[i] = min(a[i + 1] + 1, m[i]);
    }
    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
    cout << endl;
    return 0;
}
