import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		for (int i = 1; i <= t; i++) {
			int n = sc.nextInt();
			int ans = (int) Math.floor(Math.sqrt(2 * n + 0.25) - 0.5) + 1;
			System.out.println(ans);
		}
	}
}
