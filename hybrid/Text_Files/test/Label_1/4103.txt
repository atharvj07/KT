#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;
int main() {
  vector<int> v(5);
  for (int &i : v) cin >> i;
  sort(v.begin(), v.end());
  int ans = 0;
  for (int i = v.size() - 1; i >= 0; i -= 3) {
    ans += v[i];
    ans += i - 1 >= 0 ? v[i - 1] : 0;
    ans += i - 2 >= 0 ? v[i - 2] : 0;
  }
  cout << ans << '\n';
  return 0;
}
