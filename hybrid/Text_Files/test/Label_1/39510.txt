n, k = map(int, input().split())
a = list(map(int, input().split()))

total_duration = sum(a)
if total_duration % k != 0:
    print("No")
else:
    target_duration = total_duration // k
    durations = []
    current_duration = 0
    video_count = 0
    for video_duration in a:
        current_duration += video_duration
        video_count += 1
        if current_duration == target_duration:
            durations.append(video_count)
            current_duration = 0
            video_count = 0
        elif current_duration > target_duration:
            print("No")
            break
    else:
        if len(durations) == k:
            print("Yes")
            print(" ".join(map(str, durations)))
        else:
            print("No")
