#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    long long n, k, l;
    cin >> n >> k >> l;
    vector<long long> v(n);
    for (long long &x : v) cin >> x;
    long long ex = 0, cur = 0;
    int f = 0;
    for (long long i = 0; i < n; i++) {
      if (ex == cur)
        ex = cur + 1;
      else
        ex = cur;
      long long y = cur + v[i] + ex;
      if (y > l) {
        cout << "No" << endl;
        f = 1;
        break;
      }
      cur++;
      if (cur == k) cur = 0;
    }
    if (!f) cout << "Yes" << endl;
  }
}
