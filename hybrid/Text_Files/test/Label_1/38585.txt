from decimal import *
import sys

getcontext().prec = 80

a, b, c = [int(i) for i in input().split()]
x1, y1, x2, y2 = [Decimal(i) for i in input().split()]

def nonpar(x):
    return round(x, 6) != Decimal(0)

if not (nonpar(a) ^ nonpar(b)):
    if x1 == x2 and y1 == y2:
        print(0)
        sys.exit(0)
    if a * x1 + b * y1 + c == 0:
        print(abs(x2 - x1) + abs(y2 - y1))
        sys.exit(0)

if nonpar(a):
    t = -c / a
    if nonpar(b):
        s = (t * b + c) / -a
        if x1 <= t <= x2 and y1 <= s <= y2:
            print(abs(x1 - t) + abs(y1 - s))
            sys.exit(0)
    else:
        if x1 <= t <= x2:
            print(abs(x1 - t))
            sys.exit(0)
if nonpar(b):
    t = -c / b
    if x1 <= t <= x2:
        print(abs(y1 - t))
        sys.exit(0)

print(abs(x2 - x1) + abs(y2 - y1))
