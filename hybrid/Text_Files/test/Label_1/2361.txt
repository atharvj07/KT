#include <iostream>
#include <vector>
#include <queue>

typedef long long ll;

struct Job {
    int id, time;
    bool operator<(const Job &other) const {
        return time > other.time;
    }
};

int main() {
    int n, k;
    std::cin >> n >> k;
    std::vector<int> jobs(n);
    std::vector<ll> times(n);
    std::priority_queue<Job> pq;
    for (int i = 0; i < n; i++) {
        std::cin >> jobs[i];
        pq.push({jobs[i] - 1, 0});
    }
    for (int i = 0; i < n; i++) {
        std::cin >> times[i];
    }
    ll ans = 0;
    std::vector<bool> done(k, false);
    while (!pq.empty()) {
        Job job = pq.top();
        pq.pop();
        if (done[job.id])
            continue;
        done[job.id] = true;
        ans += job.time;
        for (int i = 0; i < n; i++) {
            if (jobs[i] == job.id) {
                pq.push({jobs[i] - 1, job.time + times[i]});
            }
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
