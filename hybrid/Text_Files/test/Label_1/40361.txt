import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, k = map(int, input().split())
    s = input()
    fre = {}
    fre[s[0]] = 1
    ans = 1
    for i in range(1, n):
        if s[i] != s[i - 1]:
            ans = max(ans, fre[s[i - 1]])
            fre[s[i]] = 1
        else: fre[s[i]] += 1
        ans = max(ans, fre[s[i - 1]], (fre[s[i]] + k - 1) // k)
    print(ans)
