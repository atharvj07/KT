import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] nmks = reader.readLine().split(" ");
        int n = Integer.parseInt(nmks[0]);
        int m = Integer.parseInt(nmks[1]);
        int k = Integer.parseInt(nmks[2]);

        List<Integer>[] edges = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            edges[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            String[] edge = reader.readLine().split(" ");
            int a = Integer.parseInt(edge[0]);
            int b = Integer.parseInt(edge[1]);
            edges[a].add(b);
            edges[b].add(a);
        }

        int[] max = new int[m];
        Arrays.fill(max, 0);

        for (int day = 0; day < m; day++) {
            boolean[] visited = new boolean[n + 1];
            int[] count = new int[n + 1];

            for (int i = 1; i <= n; i++) {
                if (!visited[i]) {
                    int size = dfs(edges, i, visited, count);
                    max[day] = Math.max(max[day], size);
                }
            }
        }

        for (int i = 0; i < m; i++) {
            System.out.println(max[i]);
        }
    }

    private static int dfs(List<Integer>[] edges, int node, boolean[] visited, int[] count) {
        if (visited[node]) {
            return 0;
        }

        visited[node] = true;
        count[node] = 1;

        for (int neighbor : edges[node]) {
            count[node] += dfs(edges, neighbor, visited, count);
        }

        return count[node];
    }
}
