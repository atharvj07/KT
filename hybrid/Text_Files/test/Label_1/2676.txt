#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  vector<int> d(n);
  vector<int> l(n, -1), r(n, n);
  for (int i = 0; i < n; i++) {
    if (a[i] == 0) { l[i] = i; }
    if (i > 0) { l[i] = max(l[i], l[i - 1]); }
  }
  for (int i = n - 1; i >= 0; i--) {
    if (a[i] == 0) { r[i] = i; }
    if (i < n - 1) { r[i] = min(r[i], r[i + 1]); }
  }
  for (int i = 0; i < n; i++) { d[i] = min(abs(i - l[i]), abs(i - r[i])); }
  for (int i = 0; i < n; i++) { cout << d[i] << " "; }
  return 0;
}
