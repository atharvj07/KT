#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int k, n, m;
    cin >> k >> n >> m;
    vector<int> a(n), b(m);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < m; i++) cin >> b[i];
    if (k < 300 && !a.empty() && !b.empty() && *max_element(a.begin(), a.end()) <= k && *max_element(b.begin(), b.end()) <= k) {
      for (int i = 0, j = 0; i < n + m; i++) {
        if (i < n && !a[i])
          cout << 0 << " ";
        if (i < n && a[i])
          cout << a[i] << " ";
        if (i >= n && !b[i - n])
          cout << 0 << " ";
        if (i >= n && b[i - n])
          cout << b[i - n] << " ";
      }
      cout << endl;
    } else {
      cout << "-1" << endl;
    }
  }
}
