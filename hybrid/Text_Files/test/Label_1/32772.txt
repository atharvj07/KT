import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sysin = new Scanner(System.in);
		int n = sysin.nextInt();
		int[] arr = new int[n];
		for (int i = 0; i < n; i++) {
			int h = sysin.nextInt() * 60;
			int m = sysin.nextInt();
			arr[i] = h + m;
		}
		Arrays.sort(arr);
		int max = 0, cnt = 0;
		for (int i = 0; i < n; i++) {
			if (i > 0 && arr[i] == arr[i - 1]) {
				cnt++;
			} else {
				cnt = 1;
			}
			max = Math.max(max, cnt);
		}
		System.out.println(max);
		sysin.close();
	}
}
