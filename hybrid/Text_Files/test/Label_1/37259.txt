from math import gcd
from collections import defaultdict

n = int(input().strip())
a = list(map(int, input().split()))
q = int(input().strip())
queries = [int(input().strip()) for _ in range(q)]

def solve(a, queries):
    prefix_gcd = [0] * (n + 1)
    for i in range(1, n + 1):
        prefix_gcd[i] = gcd(prefix_gcd[i-1], a[i-1])

    gcd_count = defaultdict(int)
    for i in range(1, n + 1):
        current_gcd = 0
        for j in range(i, 0, -1):
            current_gcd = gcd(current_gcd, a[j-1])
            if current_gcd == prefix_gcd[j-1]:
                break
            gcd_count[current_gcd] += 1

    for x in queries:
        print(gcd_count[x])

solve(a, queries)
