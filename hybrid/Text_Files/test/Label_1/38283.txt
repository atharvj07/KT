import sys

input = sys.stdin.readline

n = int(input())
a, b = 0, 0
ans = 0

for _ in range(n):
    p, v = input().split()
    v = int(v)
    if p == "00":
        if b == 0:
            ans += v
            a += 1
    elif p == "10":
        if b * 2 <= a:
            ans += v
            a += 1
            b += 1
    elif p == "01":
        if a * 2 <= b:
            ans += v
            a += 1
            b += 1
    else:
        if a == b and a + b != 0:
            ans += v
            a += 1
            b += 1
        elif b == 0:
            ans += v
            b += 1
        elif a * 2 <= b:
            ans += v
            a += 1
            b += 1

if a * 2 >= n or b * 2 >= n:
    print(ans)
else:
    print(0)
