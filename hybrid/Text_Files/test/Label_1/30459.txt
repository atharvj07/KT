import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        while (q-- > 0) {
            int a = sc.nextInt();
            int p = Long.numberOfTrailingZeros(a);
            int res = 1 << p;
            System.out.println(res);
        }
    }
}
