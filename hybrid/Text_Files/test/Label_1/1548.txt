#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    cin >> s;
    int x;
    cin >> x;
    if (x >= s.size()) {
      cout << -1 << endl;
    } else {
      string w(s.size(), '0');
      for (int i = x; i < s.size(); i++) {
        if (s[i] == '1') {
          w[i - x] = '1';
          w[i] = '1';
        }
      }
      cout << w << endl;
    }
  }
  return 0;
}
