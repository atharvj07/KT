import java.util.Scanner;
import java.util.Arrays;

class Main {
    static int n, m;
    static int[][] arr;
    static int[][] ans;
    static boolean flag;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        arr = new int[n][m];
        ans = new int[n][m];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                arr[i][j] = in.nextInt();
            }
        }

        flag = true;
        dfs(0, 0, 0, 0, new boolean[n * m + 1]);
        if (!flag) {
            System.out.println(-1);
        } else {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    System.out.print(ans[i][j] + " ");
                }
                System.out.println();
            }
        }
    }

    static boolean isBitonic(int[] arr) {
        int idx = 0;
        while (idx < arr.length - 1 && arr[idx] < arr[idx + 1]) {
            idx++;
        }
        while (idx < arr.length - 1 && arr[idx] >= arr[idx + 1]) {
            idx++;
        }
        return idx == arr.length - 1;
    }

    static void dfs(int row, int col, int xr, int xc, boolean[] vis) {
        if (flag == false) {
            return;
        }
        if (row == n) {
            if (xr != xc) {
                flag = false;
                return;
            }
            int[] rmax = new int[n];
            int[] cmax = new int[m];
            for (int i = 0; i < n; i++) {
                Arrays.sort(ans[i]);
                rmax[i] = ans[i][m - 1];
            }
            for (int j = 0; j < m; j++) {
                int[] t = new int[n];
                for (int i = 0; i < n; i++) {
                    t[i] = ans[i][j];
                }
                Arrays.sort(t);
                cmax[j] = t[n - 1];
            }
            if (!isBitonic(rmax) || !isBitonic(cmax)) {
                flag = false;
            }
            return;
        }
        if (col == m) {
            dfs(row + 1, 0, xr, xc, vis);
            return;
        }
        for (int i = 1; i <= n * m; i++) {
            if (vis[i]) {
                continue;
            }
            ans[row][col] = i;
            vis[i] = true;
            if (i > arr[row][col]) {
                xr++;
            } else if (i < arr[row][col]) {
                xc++;
            }
            dfs(row, col + 1, xr, xc, vis);
            if (!flag) {
                return;
            }
            xr--;
            xc--;
            vis[i] = false;
        }
    }
}
