#include <iostream>
#include <string>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string s;
        cin >> s;
        int n = s.size();
        int cntA = 0, cntB = 0, cntC = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == 'A') cntA++;
            if (s[i] == 'B') cntB++;
            if (s[i] == 'C') cntC++;
        }
        if (cntA % 2 == 0 && cntB % 2 == 0 && cntC % 2 == 0) cout << "YES\n";
        else cout << "NO\n";
    }
    return 0;
}
