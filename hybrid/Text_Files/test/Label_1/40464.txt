import sys
valuations = list(map(int, input().split()))
s = list(input())
n = len(s)
dict_l, dict_r = [0]*n, [0]*n
sums = [0]*n
sum = 0
count = 0
for i in range(n):
    j = i-1
    while j >= 0 and s[j] == s[i]: j -= 1
    dict_l[i] = j
    sum += valuations[ord(s[i])-ord('a')]
    sums[i] = sum
for i in range(n-1,-1,-1):
    j = i+1
    while j < n and s[j] == s[i]: j += 1
    dict_r[i] = j
    sum -= valuations[ord(s[i])-ord('a')]
    if valuations[ord(s[i])-ord('a')] == 0 and sum == 0 and i+1 < j-1:
        count += (dict_r[i+1] - i - 1) * (i - dict_l[i-1])
print(count)
