#include <bits/stdc++.h>
using namespace std;
const int maxN = 1005;
int n, a[maxN], ans, L[maxN], R[maxN], f[maxN], l, r, c[maxN];
int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  L[1] = 1, R[n] = n, f[1] = 1;
  for (int i = 2; i <= n; i++) {
    l = i - 1;
    while (l > 0 && a[l] == a[i]) l--;
    L[i] = l + 1;
  }
  for (int i = n - 1; i >= 1; i--) {
    r = i + 1;
    while (r <= n && a[r] == a[i]) r++;
    R[i] = r - 1;
  }
  for (int i = 1; i <= n; i++) {
    int cnt = R[i] - L[i] + 1;
    if (c[a[i]] == 0) {
      f[cnt] = 1;
    } else
      f[cnt] = f[cnt - 1] + 1;
    ans = max(ans, f[cnt]);
    c[a[i]]++;
  }
  cout << ans << endl;
  return 0;
}
