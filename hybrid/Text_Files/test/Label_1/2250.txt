#include <bits/stdc++.h>
#define uint unsigned int
#define ll long long
using namespace std;
const int N = 5000;
int n, k, x, a[N + 1], p[N];
uint dp[N][N + 1];
int main() {
  ios_base::sync_with_stdio(false);
  cin >> n >> k >> x;
  for (int i = 1; i <= n; i++) { cin >> a[i]; }
  memset(dp, 0, sizeof dp);
  sort(a + 1, a + n + 1);
  for (int i = n; i >= 1; i--) {
    for (int j = x; j >= 0; j--) {
      dp[i][j] = max(dp[i][j], dp[i + 1][j]);
      if (j >= 1 && dp[i + 1][j - 1] >= k - 1) {
        dp[i][j] = max(dp[i][j], dp[i + 1][j - 1] + a[i]);
      }
    }
  }
  cout << ((dp[1][x] >= k - 1) ? dp[1][x] : -1) << '\n';
  return 0;
}
