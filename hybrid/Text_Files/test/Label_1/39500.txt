import sys

input = sys.stdin.read().split()
n = int(input[0])
words = input[1:]
cnt = 0
last = 0
for i in range(n):
    l = 0
    for c in words[i]:
        if c in 'eyuioa':
            last = c
            l += 1
    words[i] = l
for i in range(n):
    for j in range(n):
        if words[i] == words[j] and i != j:
            for k in range(n):
                if words[k] == words[i] and i != k and j != k:
                    for l in range(n):
                        if words[l] == words[j] and i != l and j != l and k != l:
                            if last == words[l][-1]:
                                cnt += 1
                                print(words[i], words[j])
                                print(words[k], words[l])
print(cnt)
