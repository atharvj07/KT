#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  long long y, k, n;
  cin >> y >> k >> n;
  if ((n-y)%k != 0) {
    cout << "-1" << endl;
    return 0;
  }
  vector<long long> ans;
  for (long long x = (n-y)/k; x >= 1; x--) {
    if (x <= y) break;
    ans.push_back(x);
  }
  if (ans.size() == 0) cout << -1 << endl;
  else for (auto &x : ans) cout << x << ' ';
  cout << endl;
  return 0;
}
