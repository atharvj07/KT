#include <iostream>
#include <cstdio>
#include <cstring>
using namespace std;
typedef long long LL;
const int MAXN = 54332;

int T, n;
int nxt[MAXN], pre[MAXN], bel[MAXN];
int main() {
  while (scanf("%d", &n) == 1) {
    memset(nxt, 0, sizeof(nxt));
    memset(pre, 0, sizeof(pre));
    int ans = 0;
    for (int i = 1; i <= n; i++) {
      if (!pre[i]) {
        int cnt = 0;
        for (int j = i; !pre[j]; j = nxt[j])
          cnt++, pre[j] = cnt;
        ans += cnt - 1;
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
