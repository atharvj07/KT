import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int d = scanner.nextInt();
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }
        int count = 0;
        int day = 1;
        for (int i = 0; i < n; i++) {
            if (day > a[i]) {
                day = 1;
            }
            if (day != a[i]) {
                count++;
                day++;
            } else {
                day++;
            }
        }
        System.out.println(count);
    }
}
