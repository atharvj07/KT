import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.StringTokenizer;
import java.util.concurrent.atomic.AtomicInteger;

public class Main {
    static class Edge {
        int to;
        int weight;
        int count;

        public Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
            this.count = 0;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        List<List<Edge>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < n - 1; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            int w = Integer.parseInt(st.nextToken());
            adj.get(a).add(new Edge(b, w));
            adj.get(b).add(new Edge(a, w));
        }
        int k = Integer.parseInt(br.readLine());
        StringTokenizer st = new StringTokenizer(br.readLine());
        int[] stops = new int[k];
        for (int i = 0; i < k; i++) {
            stops[i] = Integer.parseInt(st.nextToken());
        }
        AtomicInteger res = new AtomicInteger();
        bfs(1, stops, adj, res);
        System.out.println(res.get() % 1000000007);
    }

    private static void bfs(int start, int[] stops, List<List<Edge>> adj, AtomicInteger res) {
        Queue<Integer> q = new LinkedList<>();
        q.add(start);
        boolean[] visited = new boolean[adj.size()];
        visited[start] = true;
        while (!q.isEmpty()) {
            int curr = q.poll();
            for (Edge edge : adj.get(curr)) {
                if (!visited[edge.to]) {
                    visited[edge.to] = true;
                    edge.count++;
                    res.addAndGet(edge.count * edge.weight);
                    q.add(edge.to);
                }
            }
        }
    }
}
