#include <iostream>
using namespace std;
int main() {
    int l, r, a;
    cin >> l >> r >> a;
    int ans = (l > r ? r : l) * 2;
    if (ans == 0) ans += a % 2;
    else {
        int x = min(a, ans - (a + ans) % 2);
        a -= x;
        ans += x / 2;
        if (ans % 2 == 1 && a > 0) { ans += 1; a -= 1; }
        ans += a / 2;
    }
    cout << ans << endl;
    return 0;
}
