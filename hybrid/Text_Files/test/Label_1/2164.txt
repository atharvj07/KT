#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, cnt = 0;
    cin >> n >> m;
    char a[n][m];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
            if (a[i][j] == '*') cnt++;
        }
    }
    if (cnt == n * m)
        cout << "NO";
    else
        cout << "YES\n"
             << "1 1";
}
