#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int mx = LLONG_MIN;
    for (int i = n / 2; i < n; i++) { mx = max(mx, a[i] - a[i - n / 2]); }
    cout << (mx > 0 ? mx : -1) << endl;
  }
}
