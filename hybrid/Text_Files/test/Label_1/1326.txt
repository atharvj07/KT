#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  bool flag = false;
  for (int i = 0; i <= 1000 && !flag; i++) {
    for (int j = 0; j <= 10000 && !flag; j++) {
      if ((n - 1234567 * i - 123456 * j) % 1234 == 0) { flag = true; }
    }
  }
  if (flag)
    cout << "YES" << endl;
  else
    cout << "NO" << endl;
  return 0;
}
