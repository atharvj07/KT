#include <bits/stdc++.h>
using namespace std;
#define ll long long int
const ll mx=1e9+7;
int main() {
    ll n;
    cin>>n;
    if(n%2==0) return 0;
    ll x=n;
    vector<ll> v;
    for(ll j=2;j*j<=x;j++){
        while(x%j==0){
            v.push_back(j);
            x/=j;
        }
    }
    if(x>1) v.push_back(x);
    if(v.size()>3){
        ll val=v.back();
        v.pop_back();
        v[v.size()-1]+=val;
    }
    cout<<v.size()<<endl;
    for(auto x:v){
        cout<<x<<" ";
    }
    cout<<endl;
    return 0;
}
