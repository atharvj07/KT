import sys

n = int(input())
s = input()

counts = [0] * 26
for i in s:
    counts[ord(i) - ord('a')] += 1

colors = 0
stack = []
for i in range(26):
    if counts[i] > 0:
        colors += 1
        stack.append([])
        for j in range(len(s)):
            if s[j] == chr(i + ord('a')):
                stack[-1].append(colors)

print(colors)

for i in range(len(stack)):
    stack[i] = [str(x) for x in stack[i]]
    print(' '.join(stack[i]))
