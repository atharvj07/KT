for _ in range(int(input())):
    n, m, a, b = map(int, input().split())
    g = [[] for i in range(n+1)]
    for i in range(m):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)

    vis = [False for i in range(n+1)]
    pa = [0 for i in range(n+1)]

    def dfs1(node, p):
        vis[node] = True
        pa[node] = p
        for u in g[node]:
            if not vis[u]:
                dfs1(u, node)

    dfs1(a, -1)

    vis = [False for i in range(n+1)]
    ans = 0

    def dfs2(node, p):
        nonlocal ans
        vis[node] = True
        if node != a and node != b and pa[node] != a and pa[node] != b:
            ans += 1
        for u in g[node]:
            if not vis[u]:
                dfs2(u, node)

    dfs2(b, -1)

    print(ans)
