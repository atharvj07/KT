import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        int dp[][] = new int[n + 1][2];
        dp[0][0] = 0;
        dp[0][1] = 0;
        for (int i = 0; i < n; i++) {
            dp[i + 1][1] = Math.max(dp[i][1], dp[i][0] + a[i] / 2);
            dp[i + 1][0] = Math.max(dp[i][1], dp[i][0] + (a[i] + 1) / 2);
        }
        System.out.println(dp[n][1]);
    }
}
