import sys
input = sys.stdin.readline
for _ in range(int(input())):
    n,m,a,b = map(int,input().split())
    g=[[] for i in range(n+1)]
    for i in range(m):
        x,y = map(int,input().split())
        g[x].append(y)
        g[y].append(x)
    vis=[False for i in range(n+1)]
    def dfs(x,par):
        vis[x] = True
        for i in g[x]:
            if vis[i]==False:
                if i==a or i==b:
                    par = i
                dfs(i,par)
        return par
    ans = dfs(1,0)
    vis = [False for i in range(n+1)]
    dfs(ans,0)
    cnt = 0
    for i in range(1,n+1):
        if vis[i] == False:
            dfs(i,0)
            cnt+=1
    print(cnt*(cnt-1)//2)
