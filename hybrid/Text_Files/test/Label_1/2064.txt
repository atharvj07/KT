#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> cards[n];
    for (int i = 0; i < n; i++) {
        int m;
        cin >> m;
        cards[i].resize(m);
        for (int j = 0; j < m; j++) {
            cin >> cards[i][j];
        }
        sort(cards[i].begin(), cards[i].end());
    }
    vector<int> bag(101, 1);
    for (int i = 0; i < n; i++) {
        bool win = true;
        for (int j = 0; j < cards[i].size(); j++) {
            if (bag[cards[i][j]] == 0) {
                win = false;
                break;
            }
            bag[cards[i][j]]--;
        }
        if (win) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
