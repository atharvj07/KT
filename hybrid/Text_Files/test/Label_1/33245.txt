import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] l = new int[n];
        int[] r = new int[n];
        for (int i = 0; i < n; i++) {
            l[i] = sc.nextInt();
            r[i] = sc.nextInt();
        }
        Arrays.sort(l);
        Arrays.sort(r);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (l[i] > 1 && r[i] < m) {
                ans++;
            }
        }
        System.out.println(ans);
    }
}
