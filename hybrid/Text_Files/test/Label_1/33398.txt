import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] edges = new int[m][2];
        for (int i = 0; i < m; i++) {
            edges[i][0] = sc.nextInt();
            edges[i][1] = sc.nextInt();
        }
        int k = sc.nextInt();
        int[][] req = new int[k][2];
        for (int i = 0; i < k; i++) {
            req[i][0] = sc.nextInt();
            req[i][1] = sc.nextInt();
        }
        boolean[] vis = new boolean[n + 1];
        int[] dir = new int[m];
        boolean[] solved = new boolean[k];
        boolean flag = dfs(1, -1, vis, dir, req, solved, edges);
        System.out.println(flag ? "Yes" : "No");
        if (flag) {
            for (int i = 0; i < m; i++) {
                System.out.println(edges[i][0] + " " + edges[i][1]);
            }
        }
    }

    public static boolean dfs(int node, int par, boolean[] vis, int[] dir, int[][] req, boolean[] solved, int[][] edges) {
        vis[node] = true;
        for (int i = 0; i < edges.length; i++) {
            if (edges[i][0] == node && !vis[edges[i][1]]) {
                if (dfs(edges[i][1], node, vis, dir, req, solved, edges)) {
                    dir[i] = node;
                } else {
                    return false;
                }
            } else if (edges[i][1] == node && !vis[edges[i][0]]) {
                if (dfs(edges[i][0], node, vis, dir, req, solved, edges)) {
                    dir[i] = edges[i][0];
                } else {
                    return false;
                }
            }
        }
        for (int[] r : req) {
            if (!solved[r[0] - 1]) {
                if (r[0] == node && r[1] == par || r[0] == par && r[1] == node) {
                    solved[r[0] - 1] = true;
                }
            }
        }
        for (boolean s : solved) {
            if (!s) {
                return false;
            }
        }
        return true;
    }
}
