import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner stdin = new Scanner(System.in);
        int n = stdin.nextInt();
        long[] x = new long[n];
        long[] y = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = stdin.nextLong();
            y[i] = stdin.nextLong();
        }
        stdin.close();

        long ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }
                for (int k = 0; k < n; k++) {
                    if (i == k || j == k) {
                        continue;
                    }
                    if ((x[j] - x[i]) * (x[j] - x[k]) + (y[j] - y[i]) * (y[j] - y[k]) == 0) {
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans / 3);
    }
}
