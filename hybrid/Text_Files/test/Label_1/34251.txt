import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        int[] arr = new int[n];
        int[] last = new int[200001];
        for (int i = 0; i < n; i++) {
            arr[i] = scan.nextInt();
            last[arr[i]] = i + 1;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (last[arr[i]] < last[ans]) {
                ans = arr[i];
            }
        }
        System.out.println(ans);
    }
}
