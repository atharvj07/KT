n,q = map(int,input().split())
a = list(map(int,input().split()))
k = list(map(int,input().split()))
l = [a.copy() for _ in range(3)]

ans = [n]
for i in range(q):
	for j in range(n):
		if k[i]>0:
			l[i%3][j]-=k[i]
			k[i]-=l[i%3][j]
			if l[i%3][j]<=0:
				l[i%3][j]=0
	ans.append(l[i%3].count(0))

for i in ans:
	print(i)
