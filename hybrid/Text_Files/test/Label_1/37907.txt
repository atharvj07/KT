import sys
L = list(map(int,sys.stdin.readline().split()))
L2 = list(map(int,sys.stdin.readline().split()))
L3 = [[]for i in range(L[0]+1)]
for i in range (L[0]-1):
    a = list(map(int,sys.stdin.readline().split()))
    L3[a[0]].append(a[1])
    L3[a[1]].append(a[0])
cnt = [0] * (L[0]+1)
col = [0] * (L[0]+1)
ans = [0] * (L[0]+1)
def dfs(v,p):
    cnt[v] = 1
    col[L2[v-1]] += 1
    if col[L2[v-1]] > col[ans[v]]:ans[v] = L2[v-1]
    for to in L3[v]:
        if to != p:
            dfs(to,v)
            cnt[v] += cnt[to]
            col[ans[to]] += 1
            if col[ans[to]] > col[ans[v]]:ans[v] = ans[to]
    col[L2[v-1]] -= 1
dfs(1,1)
print(*ans[1:])
