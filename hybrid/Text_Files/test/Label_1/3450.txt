#include <vector>
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int a, b, c;
        cin >> a >> b >> c;
        int count = 0;
        for (int i = 1; i <= a; i++) {
            for (int j = i; j <= b; j++) {
                for (int k = j; k <= c; k++) {
                    if (a % i == 0 && b % j == 0 && c % k == 0) {
                        count++;
                    }
                }
            }
        }
        cout << count << endl;
    }
    return 0;
}
