import java.util.Scanner;

public class Main {
    static long[] A;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);

        int n = scan.nextInt();
        A = new long[n];
        for (int i = 0; i < n; i++) A[i] = scan.nextLong();

        int q = scan.nextInt();
        for (int i = 0; i < q; i++) {
            int b = scan.nextInt();
            int e = scan.nextInt();
            int k = scan.nextInt();
            long ans = 0;
            for (int j = b; j < e; j++) ans += A[j] == k ? 1 : 0;
            System.out.println(ans);
        }
    }
}
