#include <iostream>
#include <vector>
using namespace std;
int main() {
    long long n;
    cin >> n;
    vector<long long> arr(n);
    long long sum = 0;
    for (long long i = 0; i < n; ++i) {
        cin >> arr[i];
        sum += arr[i];
    }
    sum /= n;
    long long ans = 0;
    for (long long i = 0; i < n; ++i) { ans += abs(arr[i] - sum); }
    cout << ans / 2;
    return 0;
}
