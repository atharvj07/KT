import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[][] arr = new int[n][2];
    for (int i = 0; i < n; i++) {
      arr[i][0] = sc.nextInt();
      arr[i][1] = sc.nextInt();
    }
    Arrays.sort(arr, (a, b) -> a[0] - b[0]);
    int min = Integer.MAX_VALUE;
    for (int i = 0; i < n; i++) {
      int cnt = 1;
      for (int j = i + 1; j < n; j++) {
        if (arr[j][0] == arr[i][0]) {
          cnt++;
        } else {
          break;
        }
      }
      if (cnt < min) {
        min = cnt;
      }
    }
    Arrays.sort(arr, (a, b) -> a[1] - b[1]);
    for (int i = 0; i < n; i++) {
      int cnt = 1;
      for (int j = i + 1; j < n; j++) {
        if (arr[j][1] == arr[i][1]) {
          cnt++;
        } else {
          break;
        }
      }
      if (cnt < min) {
        min = cnt;
      }
    }
    System.out.println(min);
  }
}
