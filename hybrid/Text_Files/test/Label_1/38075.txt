from collections import defaultdict

n = int(input())
s = []
for i in range(n):
    s.append(input())
s.sort(key=len)
mp = defaultdict(int)
ans = 0

for i in s:
    if mp[i]>0:
        ans += 1
        mp[i] -= 1
    else:
        st = ''
        for j in i:
            if j=='(':
                st += '('
            elif st=='':
                pass
            else:
                st = st[:-1]
        if st=='':
            mp[i]+=1
        else:
            for j in mp:
                if j+i in mp and mp[j]>0 and mp[j+i]>0:
                    ans += 1
                    mp[j] -= 1
                    mp[j+i] -= 1
                    break
print(ans)
