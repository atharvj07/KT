#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
  ios::sync_with_stdio(0);
  int t;
  cin >> t;
  while (t--) {
    ll n, x, m;
    cin >> n >> x >> m;
    vector<bool> can(n + 1, 0);
    while (m--) {
      ll l, r;
      cin >> l >> r;
      if (l <= x && x <= r) can[x] = !can[x];
      can[l] = !can[l];
      can[r + 1] = !can[r + 1];
    }
    ll ans = 0;
    for (ll i = 1; i <= n; i++) ans += can[i];
    cout << ans << endl;
  }
  return 0;
}
