import java.util.Scanner;

public class Main {
    static int sgn(long x) {
        return x == 0 ? 0 : (x > 0 ? 1 : -1);
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        for (int i = 0; i < n; i++) {
            int ans = 0;
            int x1 = in.nextInt(), y1 = in.nextInt(), x2 = in.nextInt(), y2 = in.nextInt(),
                    x3 = in.nextInt(), y3 = in.nextInt(), x4 = in.nextInt(), y4 = in.nextInt();
            ans += Math.min(Math.abs(x1 - x2), Math.abs(y1 - y2)) % 2;
            ans += Math.min(Math.abs(x1 - x3), Math.abs(y1 - y3)) % 2;
            ans += Math.min(Math.abs(x1 - x4), Math.abs(y1 - y4)) % 2;
            ans += Math.min(Math.abs(x2 - x3), Math.abs(y2 - y3)) % 2;
            ans += Math.min(Math.abs(x2 - x4), Math.abs(y2 - y4)) % 2;
            ans += Math.min(Math.abs(x3 - x4), Math.abs(y3 - y4)) % 2;
            if ((x1 - x2) * (y1 - y3) == (x1 - x3) * (y1 - y2) &&
                    (x1 - x3) * (y1 - y4) == (x1 - x4) * (y1 - y3) &&
                    (x2 - x3) * (y2 - y4) == (x2 - x4) * (y2 - y3)) {
                ans = -1;
            }
            System.out.println(ans);
        }
    }
}
