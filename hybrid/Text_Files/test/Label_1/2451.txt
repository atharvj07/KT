#include<bits/stdc++.h>
#include<ext/pb_ds/assoc_container.hpp>
#include<ext/pb_ds/tree_policy.hpp>
using namespace std;
using namespace __gnu_pbds;
#define endl '\n'
typedef tree<int,null_type,less<int>,rb_tree_tag,tree_order_statistics_node_update>ordered_set;
#define int long long int
const int mod = 1e9+7;
const int maxn = 2e5+1;
int p[maxn],type[maxn],leaf[maxn],sz[maxn];
vector<pair<int,int>>adj[maxn];
void fc(int s, int p1){
    p[s] = p1;
    for(auto x : adj[s]){
        if(x.first == p1)continue;
        fc(x.first,s);
    }
}
void dfs(int s, int p1,int &leaf_cnt){
    type[s] = 1;
    bool flag = true;
    sz[s] = 1;
    for(auto x : adj[s]){
        if(x.first == p1)continue;
        if(type[x.first] == 0){
            flag = false;
            dfs(x.first,s,leaf_cnt);
            sz[s]+=sz[x.first];
        }
    }
    if(flag){
        leaf[s] = leaf_cnt;
        leaf_cnt++;
    }
}
int32_t main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    cin>>n;
    int cnt = 0;
    for(int i=1;i<n;i++){
        int a,b;
        cin>>a>>b;
        adj[a].push_back({b,i});
        adj[b].push_back({a,i});
        cnt++;
    }
    fc(1,1);
    int leaf_cnt = 0;
    dfs(1,1,leaf_cnt);
    vector<vector<int>>path;
    int x = -1;
    for(int i=1;i<=n;i++){
        if(type[i] == 1)path.push_back({i,p[i]});
        else{
            if(x == -1)x = i;
            else{
                path.push_back({i,p[i]});
                path.push_back({x,p[x]});
                x = -1;
            }
        }
    }
    cout<<1<<" ";
    int mx = 0;
    set<int>st;
    for(int i=0;i<path.size()-1;i++){
        int s = path[i][0];
        int d = path[i+1][0];
        while(s != d){
            if(s > d){
                if(leaf[s])st.insert(leaf[s]);
                s = p[s];
            }
            else{
                if(leaf[d])st.insert(leaf[d]);
                d = p[d];
            }
        }
        mx = max(mx,(int)st.size());
        st.clear();
    }
    cout<<mx<<endl;
}
