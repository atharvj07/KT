#include <cstdio>
#include <vector>
#include <algorithm>

int main(void) {
  int n, x;
  scanf("%d %d", &n, &x);

  std::vector<int> months(n);
  for (int i = 0; i < n; ++i) {
    scanf("%d", &months[i]);
  }

  std::sort(months.begin(), months.end());

  long long answer = 0;
  int index = n - 1;
  while (x > 0) {
    answer += (x >= months[index]) ? 1ll * months[index] * months[index] : 1ll * months[index] * x;
    x -= months[index];
    --index;
  }

  printf("%lld\n", answer);

  return 0;
}
