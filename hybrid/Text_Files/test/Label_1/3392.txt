#include <bits/stdc++.h>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    string keyboard, s;
    cin >> keyboard >> s;
    int ans = 0, cur = 0;
    for (char c : s) {
      int pos = keyboard.find(c);
      ans += abs(pos - cur);
      cur = pos;
    }
    cout << ans << endl;
  }
}
