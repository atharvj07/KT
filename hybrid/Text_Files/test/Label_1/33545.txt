import java.util.Scanner;

public class XORSum {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        long s = scanner.nextLong();
        long x = scanner.nextLong();

        if (s < x || (s - x) % 2 != 0) {
            System.out.println(0);
        } else {
            long a = (s + x) / 2;
            long b = (s - x) / 2;
            System.out.println(a == b ? 1 : 2);
        }
    }
}
