import java.util.Scanner;

public class PermutationPeaks {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        for (int i = 0; i < t; i++) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();

            if (k > n - 1) {
                System.out.println(-1);
            } else {
                int[] permutation = new int[n];
                int[] peaks = new int[k];

                // Initialize permutation with consecutive numbers
                for (int j = 0; j < n; j++) {
                    permutation[j] = j + 1;
                }

                // Set peaks
                for (int j = 0; j < k; j++) {
                    peaks[j] = j * 2 + 1; // Set peak indices
                }

                // Adjust elements around peaks
                for (int j = 0; j < k; j++) {
                    int peakIndex = peaks[j];
                    if (peakIndex < n - 1) {
                        swap(permutation, peakIndex, peakIndex + 1);
                    }
                }

                // Print permutation
                for (int j = 0; j < n; j++) {
                    System.out.print(permutation[j] + " ");
                }
                System.out.println();
            }
        }
    }

    private static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
}
