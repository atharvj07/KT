#include <iostream>
#include <vector>
using namespace std;

int n, m;
vector<int> adj[100005];
vector<bool> vis;
vector< pair<int,int> > e;
vector<int> path[100005];

void dfs(int u, int p){
    vis[u] = true;
    for(int v : adj[u]){
        if(!vis[v]){
            e.push_back({u, v});
            dfs(v, u);
        }
    }
}

int main(){
    cin >> n;
    for(int i = 0; i < n - 1; i++){
        int u, v; cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    vis.assign(n + 1, false);
    dfs(1, 1);
    if(e.size() % (n - 1)){
        cout << "No" << endl;
        return 0;
    }
    m = e.size() / (n - 1);
    cout << "Yes" << endl;
    cout << m << endl;
    int cnt = 0;
    for(int i = 0; i < n - 1; i++){
        path[i / m].push_back(e[i].first);
        path[i / m].push_back(e[i].second);
        if(i % m == m - 1){
            cout << path[i / m].size() << " ";
            for(int j : path[i / m]){
                cout << j << " ";
            }
            cout << endl;
        }
    }
    return 0;
}
