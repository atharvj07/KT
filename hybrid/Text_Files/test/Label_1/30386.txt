import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        BigInteger l = in.nextBigInteger();
        BigInteger r = in.nextBigInteger();
        BigInteger count = r.subtract(l).add(BigInteger.ONE);
        String lStr = l.toString();
        String rStr = r.toString();
        int lZeros = lStr.length() - lStr.replace("0", "").length();
        int rZeros = rStr.length() - rStr.replace("0", "").length();
        if (lStr.charAt(0) == '0' && rStr.charAt(0) == '0') {
            count = count.subtract(BigInteger.ONE);
        }
        count = count.subtract(BigInteger.valueOf(lZeros));
        count = count.subtract(BigInteger.valueOf(rZeros));
        System.out.println(count);
    }
}
