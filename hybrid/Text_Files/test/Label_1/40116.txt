n, m = map(int, input().split())
flowers = list(map(int, input().split()))
subarrays = [tuple(map(int, input().split())) for _ in range(m)]

happiness = 0
for i in range(m):
    l, r = subarrays[i]
    subarray_sum = sum(flowers[l-1:r])
    if subarray_sum > 0:
        happiness += subarray_sum

print(happiness)
