#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 52;
const int INF = 1e9;

int n;
char board[MAXN][MAXN];
vector<pair<int, int>> moves;

bool valid(int dx, int dy) {
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      if (board[i][j] == 'o' || board[i][j] == 'x') {
        if (i + dx >= 1 && i + dx <= n && j + dy >= 1 && j + dy <= n &&
            board[i + dx][j + dy] == '.') {
          return false;
        }
      }
    }
  }
  return true;
}

int main() {
  cin >> n;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) { cin >> board[i][j]; }
  }

  for (int dx = -n; dx <= n; dx++) {
    for (int dy = -n; dy <= n; dy++) {
      if (valid(dx, dy)) {
        moves.push_back({dx, dy});
      }
    }
  }

  if (moves.empty()) {
    cout << "NO" << endl;
  } else {
    cout << "YES" << endl;
    int size = 2 * n - 1;
    for (int i = 1; i <= size; i++) {
      for (int j = 1; j <= size; j++) {
        if (i == size / 2 + 1 && j == size / 2 + 1) {
          cout << 'o';
        } else {
          bool attacked = false;
          for (auto& move : moves) {
            if (i - size / 2 + 1 + move.first >= 1 &&
                i - size / 2 + 1 + move.first <= n &&
                j - size / 2 + 1 + move.second >= 1 &&
                j - size / 2 + 1 + move.second <= n &&
                board[i - size / 2 + 1 + move.first][j - size / 2 + 1 + move.second] ==
                    'x') {
              attacked = true;
            }
          }
          if (attacked) {
            cout << 'x';
          } else {
            cout << '.';
          }
        }
      }
      cout << endl;
    }
  }
  return 0;
}
