import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] grid = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }

        int[] rowSums = new int[3];
        int[] colSums = new int[3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                rowSums[i] += grid[i][j];
                colSums[j] += grid[i][j];
            }
        }

        boolean isTakahashiCorrect = true;
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                if (grid[i][j] != rowSums[i] - colSums[j]) {
                    isTakahashiCorrect = false;
                    break;
                }
            }
        }

        System.out.println(isTakahashiCorrect ? "Yes" : "No");
    }
}
