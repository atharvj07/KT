#include<bits/stdc++.h>
#define rep(i, j, k) for(int i = j; i <= k; ++i)

using namespace std;
int n, k;
string a, b;
bool check(char c)
{
	if(c == 'z') return false;
	char t = c;
	rep(i, 0, n - k + 1)
	{
		bool f = true;
		rep(j, i, i + k - 1)
			if(a[j] != t)
			{
				f = false;
				break;
			}
		if(f)
		{
			if(t != b[i]) return false;
			t++;
		}
	}
	return true;
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin >> t;
	while(t--)
	{
		cin >> n >> k;
		cin >> a >> b;
		bool f = true;
		rep(i, 0, n - 1)
			if(a[i] != b[i] && (!check(a[i]) || !check(b[i])))
			{
				f = false;
				break;
			}
		if(f) cout << "YES" << endl;
		else cout << "NO" << endl;
	}
	return 0;
}
