#include <cstdio>

int main() {
  long long x;
  scanf("%lld", &x);
  if (x == 2) { puts("NO"); return 0; }
  puts("YES");
  for (int i = 2; i * i <= x; i++) {
    if (x % i == 0) {
      printf("%d\n", 1);
      printf("%d %d\n", i - 1, i);
      x /= i;
      break;
    }
  }
  printf("1\n");
  printf("%lld %lld\n", x - 1, x);
  return 0;
}
