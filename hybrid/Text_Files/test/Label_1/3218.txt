#include <vector>
#include <iostream>
#include <algorithm>
#include <bitset>
using namespace std;
using ll = unsigned long long;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    a.erase(unique(a.begin(), a.end()), a.end());
    int ans = 0;
    for (ll x = 1;; x++) {
        int match = 0;
        for (int i = 0; i < a.size(); i++) for (int j = 0; j + 1 < a.size(); j++)
            if ((a[j] ^ a[j + 1]) == a[i]) match++;
        if (match == x * (x - 1) / 2) {
            ans = x;
            break;
        }
    }
    cout << ans << "\n";
    ll piv = 1 << ans;
    vector<bitset<30>> b(piv);
    for (int i = 0; i < piv; i++) b[i] = i;
    for (int i = 0; i < ans; i++) {
        for (int j = 0; j < piv; j++) {
            int l = j ^ (1 << i);
            int r = j ^ (1 << (i + 1));
            if (a.empty()) {
                cout << j << " ";
            } else if (((l < r) && (binary_search(a.begin(), a.end(), b[l].to_ullong() ^ b[r].to_ullong()))) || ((r < l) && (binary_search(a.begin(), a.end(), b[r].to_ullong() ^ b[l].to_ullong())))) {
                cout << j << " ";
                a.erase(lower_bound(a.begin(), a.end(), b[l].to_ullong() ^ b[r].to_ullong()));
            }
        }
    }
}
