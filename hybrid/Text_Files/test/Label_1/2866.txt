#include<bits/stdc++.h>
using namespace std;
#define ll long long
int main() {
    ll n,l,r; cin>>n; ll p,ans=-1,a[n],b[n];
    char d[n]; cin>>d;
    for (int i=0; i<n; i++) {cin>>a[i]; if (d[i]=='L') b[i]=-a[i]; else b[i]=a[i];}
    sort(b,b+n);
    for (int i=0; i<n; i++) 
    {
        l=i; r=n-1;
        int p=-1;
        while (l<=r)
        {
            int mid=(l+r)/2;
            if (-b[l]-b[mid]<=0) p=l,l=mid+1;
            else r=mid-1;
        }
        if (p==-1) continue;
        if (ans==-1) ans=b[p]+b[i];
        else ans=min(ans,abs(b[p]+b[i]));
    }
    cout<<ans<<endl;
}
