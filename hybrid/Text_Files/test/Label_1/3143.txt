#include <bits/stdc++.h>
#define pb push_back
#define vi vector<int>
using namespace std;
typedef long long ll;
const int N = 2e5 + 10;
int32_t main() {
    int n;
    cin >> n;
    vi a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vi b, c;
    b.pb(a[0]);
    for (int i = 1; i < n; i++)
        if (a[i] >= b.back())
            b.pb(a[i]);
    c.pb(a[n - 1]);
    for (int i = n - 2; i >= 0; i--)
        if (a[i] <= c.back())
            c.pb(a[i]);
    if (b.size() + c.size() != n) {
        cout << "NO" << endl;
        return 0;
    }
    vector<int> ans(n);
    if (c.size() > b.size()) {
        cout << "YES" << endl;
        for (int i = 0; i < n; i++)
            if (a[i] == c.back())
                ans[i] = 1, c.pop_back();
    } else if (b.size() > c.size()) {
        cout << "YES" << endl;
        for (int i = 0; i < n; i++)
            if (a[i] == b.back())
                ans[i] = 0, b.pop_back();
    } else {
        cout << "NO" << endl;
        return 0;
    }
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    cout << endl;
}
