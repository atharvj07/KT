from collections import defaultdict
n, m = [int(x) for x in input().split()]
st = []
cs = defaultdict(lambda:[])
for i in range(n):
  s = input()
  st.append(s)
  for j in range(m):
    cs[j+1].append(s[j])

ans = 0
for j in range(1,m+1):
  uni = list(set(cs[j]))
  if len(uni)==n: continue
  mn = 1e9
  for k in uni:
    cost = 0
    for s in st: cost += s[j-1]!=k
    mn = min(mn,cost)
  ans += mn

print(ans)
