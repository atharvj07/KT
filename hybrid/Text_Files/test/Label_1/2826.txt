#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    if (a == b)
        cout << "-1" << endl;
    else {
        int ans = 0;
        for (int i = 0; i < n; i++) ans += !a[i] && b[i];
        cout << ans + 1 << endl;
    }
}
