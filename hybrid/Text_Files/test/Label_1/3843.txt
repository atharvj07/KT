#include <iostream>
#include <string>

int main() {
    std::string s;
    std::cin >> s;
    int a = 0, b = 0, c = 0;
    for (char ch : s) {
        if (ch == 'a') {
            if (b) {
                a += c + 1;
                c = 1;
                b = 0;
            } else {
                a++;
                c = 1;
            }
        } else {
            if (a) {
                b += c + 1;
                c = 1;
                a = 0;
            } else {
                b++;
                c = 1;
            }
        }
    }
    if (a) a += c;
    if (b) b += c;
    std::cout << a + b << std::endl;
    return 0;
}
