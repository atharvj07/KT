#include <cstdio>
const int N = 30, M = 5;
int n, t[N];
int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; ++i)
        scanf("%d", &t[i]);
    int ans = 4;
    for (int i = 0; i < n; ++i) {
        for (int j = 1; j < (1 << i); ++j)
            ans += 4 * t[i];
    }
    printf("%d\n", ans);
    return 0;
}
