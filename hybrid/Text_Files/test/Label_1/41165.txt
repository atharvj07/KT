import heapq
n, k, q = map(int, input().split())
t = [int(x) for x in input().split()]
h = []
for i in range(k):h.append(-t[i]); heapq.heapify(h)
for i in range(k, n):
    if -t[i] > h[0]:
        heapq.heappop(h); heapq.heappush(h, -t[i])
for i in range(q):
    typ, id = map(int, input().split())
    if typ == 1:
        if t[id - 1] > h[0]:h.pop(0);h.append(-t[id - 1]); heapq.heapify(h)
    else:print(["NO", "YES"][t[id - 1] >= -h[0]])
