import sys
input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for _ in range(n+1)]
for _ in range(m):
    x, y = map(int, input().split())
    graph[x].append(y)

vis = [False]*(n+1)
order = []
def dfs(node):
    vis[node] = True
    for adj in graph[node]:
        if not vis[adj]:
            dfs(adj)
    order.append(node)

for i in range(1, n+1):
    if not vis[i]:
        dfs(i)
order.reverse()
print(*order)
