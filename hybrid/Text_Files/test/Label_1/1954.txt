#include<iostream>
#include<vector>
#include<map>
using namespace std;
int main(){
    long long int n,cnt=0,i,st=0,mn=1000000001,last[300002];
    cin>>n;
    map<long long int,long long int>mp;
    vector<long long int>v(n);
    for(i=0;i<n;i++){
        cin>>v[i];
        mp[v[i]]++;
        if(mp[v[i]]==1){
            last[v[i]]=i;
        }
        if(mp[v[i]]>1){
            while(mp[v[i]]>1){
                st++;
                mp[v[st]]--;
            }
        }
        mn=min(mn,last[v[i]]-st+1);
    }
    for(i=1;i<=n;i++){
        if(mp[i]==1){
            cnt+=mn;
        }
    }
    cout<<cnt<<endl;
    return 0;
}
