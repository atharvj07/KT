#include <iostream>
#include <sstream>
#include <algorithm>
#include <set>

using namespace std;

int main() {
    string s;
    getline(cin, s);
    s.erase(s.length() - 1);
    s.erase(0, 1);
    stringstream ss(s);
    string word;
    set<string> words;
    while (getline(ss, word, ',')) {
        words.insert(word);
    }
    cout << words.size() << endl;
    return 0;
}
