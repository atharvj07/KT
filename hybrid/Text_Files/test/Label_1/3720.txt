#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
int G[505][505];
ll a[505];
map<ll, vector<pair<int, int>>> mp;
void gao(int x, int p) {
  for (int i = 1; i <= x; i++) {
    if (i == p) continue;
    a[i] = ll(a[p]) * ll(G[i][p]) / ll(G[i][i]);
  }
}
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) { cin >> G[i][j]; }
  }
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      mp[G[i][j]].push_back({i, j});
    }
  }
  a[1] = 1;
  bool flag = 0;
  int ptr = 1;
  while (!mp[1].empty() && ptr <= n) {
    auto t = mp[1].back();
    mp[1].pop_back();
    if (t.first == ptr && t.second == ptr) {
      continue;
    }
    gao(t.second, t.first);
    flag = 1;
    ptr = t.second;
  }
  if (flag == 0) {
    for (auto i : mp[G[1][2]]) {
      if (i.first != 1 && i.second != 2) {
        gao(i.second, i.first);
        flag = 1;
        ptr = i.second;
        break;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << a[i] << ' ';
  }
  cout << endl;
  return 0;
}
