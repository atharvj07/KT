n,b,k,x=map(int,input().split());a=list(map(int,input().split()));
l=[0 for i in range(9)]
for i in a:
    l[i-1]+=1
la=[0 for i in range(9)]
for i in l:
    la[i%x]+=i
ans=1
for i in la:
    if i==0:ans=0
    else:ans=(ans*(i**(k//x)%1000000007))%1000000007
    if k%x!=0:
        if i**(k%x)>0:ans=(ans*(i**(k%x))%1000000007)%1000000007
        else:ans=0
print(ans)
