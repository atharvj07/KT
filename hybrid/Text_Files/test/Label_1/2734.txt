#include <bits/stdc++.h>
#define RG register
using namespace std;
int n, x, a[100005], b[100005], c[100005];
bool vis[100005];
long long gcd(RG long long a, RG long long b) {
  return b == 0 ? a : gcd(b, a % b);
}
bool check(RG int l, RG int r) {
  for (RG int i = l; i < r; ++i) { vis[i] = true; }
  if (c[l] == c[r]) {
    RG int p = l + 1;
    while (a[p - 1] <= a[p]) { ++p; }
    if (p < r) { swap(a[p], a[r]), c[r] = c[p]; }
  }
  for (RG int i = l; i <= r; ++i) { vis[i] = false; }
  for (RG int i = l; i < r; ++i) {
    if (a[i] > a[i + 1]) {
      for (RG int j = i + 1; j <= r; ++j) {
        if (!vis[j] && gcd(a[i], a[j]) == c[l]) {
          swap(a[i], a[j]), swap(c[i], c[j]), vis[j] = true;
          break;
        }
      }
    }
    if (a[i] > a[i + 1]) { return false; }
  }
  return true;
}
int main() {
  cin >> n;
  while (n--) {
    cin >> x;
    for (RG int i = 1; i <= x; ++i) {
      cin >> b[i];
      a[i] = b[i], c[i] = b[i];
      sort(a + 1, a + 1 + i);
      c[i] = a[1];
    }
    if (!check(1, x)) {
      cout << "NO" << endl;
    } else {
      cout << "YES" << endl;
    }
  }
  return 0;
}
