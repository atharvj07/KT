import java.util.*;
import java.io.*;
public class Main {
    static class P {
        double x, y;
        P(double x, double y) {this.x = x; this.y = y;}
        P sub(P q) {return new P(x - q.x, y - q.y);}
        double mod() {return Math.sqrt(x * x + y * y);}
        void swap(P q) {double t = x; x = q.x; q.x = t; t = y; y = q.y; q.y = t;}
        P rot90() {return new P(-y, x);}
    }
    public static void main(String[] args) throws IOException {
        BufferedReader r = new BufferedReader(new InputStreamReader(System.in));
        for (String dStr = r.readLine(); !dStr.equals("0"); dStr = r.readLine()) {
            double d = Double.parseDouble(dStr);
            StringTokenizer st = new StringTokenizer(r.readLine());
            P start = new P(Double.parseDouble(st.nextToken()), Double.parseDouble(st.nextToken())),
              vel = new P(Double.parseDouble(st.nextToken()), Double.parseDouble(st.nextToken()));

            boolean hit = false;
            double dist = 0;
            start = start.sub(vel.rot90().sub(vel).rot90());
            while (start.mod() + vel.mod() <= d) {
                if (start.mod() <= 1) {
                    hit = true;
                    break;
                } else start = start.sub(vel.rot90());
                dist += vel.mod();

            }
            System.out.println(hit ? dist : "impossible");
        }
    }
}
