import sys

input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
a.sort()
b = []
for i in a:
    if i == 100:
        b.append(i)
    else:
        b.insert(0, i)
ans = []
if len(b) > 0 and b[0] == 100:
    print(-1)
    print(0)
    exit()
for i in range(len(a)):
    if k >= 100:
        k -= 100
        ans.append(100)
    elif k >= 50:
        k -= 50
        ans.append(50)
    else:
        print(-1)
        print(0)
        exit()

print(len(ans))

if len(ans) <= 1:
    print(1)
else:
    mod = 10 ** 9 + 7
    dp = [[[0 for _ in range(2)] for _ in range(k + 1)] for _ in range(len(ans) + 1)]
    dp[0][0][0] = 1
    dp[0][0][1] = 1
    for i in range(len(ans)):
        for j in range(k + 1):
            if dp[i][j][0] != 0:
                dp[i + 1][j][0] = (dp[i + 1][j][0] + dp[i][j][0]) % mod
                if ans[i] == 50:
                    dp[i + 1][j + 50][1] = (dp[i + 1][j + 50][1] + dp[i][j][0]) % mod
                elif ans[i] == 100:
                    dp[i + 1][j + 100][1] = (dp[i + 1][j + 100][1] + dp[i][j][0]) % mod
            if dp[i][j][1] != 0:
                dp[i + 1][j][1] = (dp[i + 1][j][1] + dp[i][j][1]) % mod
                if ans[i] == 50:
                    dp[i + 1][j + 50][0] = (dp[i + 1][j + 50][0] + dp[i][j][1]) % mod
                elif ans[i] == 100:
                    dp[i + 1][j + 100][0] = (dp[i + 1][j + 100][0] + dp[i][j][1]) % mod
    print(dp[len(ans)][k][1])
