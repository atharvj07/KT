import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[][] requests = new int[n + 1][3];
        for (int i = 1; i <= n; i++) {
            requests[i][0] = in.nextInt();
            requests[i][1] = in.nextInt();
            requests[i][2] = i;
        }
        Arrays.sort(requests, Comparator.comparingInt(a -> a[0]));
        int[] starts = new int[n + 1];
        int[] ends = new int[n + 1];
        int day = 1;
        for (int i = 1; i <= n; i++) {
            if (day <= requests[i][0]) {
                starts[requests[i][2]] = requests[i][0];
                ends[requests[i][2]] = requests[i][0] + requests[i][1] - 1;
                day = ends[requests[i][2]] + 1;
            } else {
                int j = i;
                while (j <= n && day >= requests[j][0]) {
                    j++;
                }
                starts[requests[i][2]] = day;
                ends[requests[i][2]] = day + requests[i][1] - 1;
                day += requests[i][1];
            }
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(starts[i] + " " + ends[i]);
        }
    }
}
