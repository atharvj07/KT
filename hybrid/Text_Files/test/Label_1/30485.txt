import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), ans = 1;
        for (int i = 2; i <= (n + 1) / 2; i++) {
            ans += (n - i + 3) / i;
        }
        System.out.println(ans);
    }
}
