t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    drenched = [0] * n
    max_cream = 0
    for i in range(n - 1, -1, -1):
        max_cream = max(max_cream, a[i])
        if max_cream > 0:
            drenched[i] = 1
        max_cream -= 1
    print(*drenched)
