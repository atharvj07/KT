#include <iostream>

using namespace std;

int main() {
  long long m;
  cin >> m;
  long long x = 1;
  while (x * x * x <= m) x++;
  x--;
  long long cnt = 0;
  while (x >= 1) {
    long long t = m / (x * x * x);
    cnt += t;
    m -= x * x * x * t;
    x--;
  }
  cout << cnt << ' ' << m + 1 << endl;
}
