import java.util.*;
import java.lang.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n + 1];
        int[] b = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = scanner.nextInt();
            b[a[i]] = i;
        }
        int ans = 0;
        for (int i = 1; i <= n; i++) {
            if (a[i] != i) {
                ans++;
                int j = b[i];
                for (int k = j; k > i; k--) {
                    a[k] = a[k - 1];
                    b[a[k]] = k;
                }
                a[i] = i;
                b[i] = i;
            }
        }
        System.out.println(ans);
    }
}
