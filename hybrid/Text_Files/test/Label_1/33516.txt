import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; ++i) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        long sum = 0;
        for (int i = 0; i < k; ++i) {
            sum += a[i];
        }
        long minSum = sum;
        for (int i = k; i < n; ++i) {
            sum = sum - a[i - k] + a[i];
            if (sum < minSum) {
                minSum = sum;
                ans = i - k + 1;
            }
        }
        System.out.println(ans + 1);
    }
}
