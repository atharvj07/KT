def min_money_spent(n, k, x, a, c):
    costs = []
    total_cost = 0

    for i in range(n):
        if k >= x[i]:
            continue
        required_drinks = (x[i] - k + a - 1) // a
        for _ in range(required_drinks):
            if not costs:
                return -1
            total_cost += costs.pop(0)
            k += a
        if k < x[i]:
            return -1
        costs.append(c[i])
        costs.sort()

    return total_cost

n, k = map(int, input().split())
x = list(map(int, input().split()))
a = int(input())
c = list(map(int, input().split()))

print(min_money_spent(n, k, x, a, c))
