import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int N = Integer.parseInt(br.readLine());
        Airplane[] planes = new Airplane[N];
        for (int i = 0; i < N; i++) {
            String[] line = br.readLine().split(" ");
            planes[i] = new Airplane(Integer.parseInt(line[0]), Integer.parseInt(line[1]), line[2]);
        }

        // Calculate the time of collision for each pair of airplanes
        double[][] collisionTime = new double[N][N];
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                collisionTime[i][j] = calculateCollisionTime(planes[i], planes[j]);
            }
        }

        // Find the first collision time
        double firstCollisionTime = Double.MAX_VALUE;
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                if (collisionTime[i][j] != -1 && collisionTime[i][j] < firstCollisionTime) {
                    firstCollisionTime = collisionTime[i][j];
                }
            }
        }

        // Print the result
        if (firstCollisionTime == Double.MAX_VALUE) {
            System.out.println("SAFE");
        } else {
            System.out.println((int) firstCollisionTime);
        }
    }

    private static double calculateCollisionTime(Airplane plane1, Airplane plane2) {
        // Check if the airplanes are moving in opposite directions
        if (plane1.direction.equals(plane2.direction)) {
            return -1;
        }

        // Calculate the relative velocity between the two airplanes
        double relX = plane2.x - plane1.x;
        double relY = plane2.y - plane1.y;
        double relSpeedX = plane2.speedX - plane1.speedX;
        double relSpeedY = plane2.speedY - plane1.speedY;
        double relSpeed = Math.sqrt(Math.pow(relSpeedX, 2) + Math.pow(relSpeedY, 2));

        // Calculate the time until the airplanes collide
        double collisionTime = -1;
        if (relSpeed != 0) {
            collisionTime = (relX * relSpeedX + relY * relSpeedY) / (Math.pow(relSpeed, 2));
        }

        // Check if the collision time is positive and less than the time limit
        if (collisionTime > 0 && collisionTime <= 1000) {
            return collisionTime;
        } else {
            return -1;
        }
    }

    private static class Airplane {
        int x;
        int y;
        double speedX;
        double speedY;
        String direction;

        public Airplane(int x, int y, String direction) {
            this.x = x;
            this.y = y;
            switch (direction) {
                case "U":
                    speedX = 0;
                    speedY = 0.1;
                    break;
                case "R":
                    speedX = 0.1;
                    speedY = 0;
                    break;
                case "D":
                    speedX = 0;
                    speedY = -0.1;
                    break;
                case "L":
                    speedX = -0.1;
                    speedY = 0;
                    break;
            }
            this.direction = direction;
        }
    }
}
