import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] line1 = br.readLine().split(" ");
        int n = Integer.parseInt(line1[0]);
        int k = Integer.parseInt(line1[1]);
        String[] line2 = br.readLine().split(" ");
        List<Integer> list = new ArrayList<>();
        for (String s : line2) {
            if (s.equals("?")) {
                list.add(0);
            } else {
                list.add(Integer.parseInt(s));
            }
        }

        long[] prefixSums = new long[n - k + 1];
        long cur = 0;
        for (int i = 0; i < k; i++) {
            cur += list.get(i);
        }
        prefixSums[0] = cur;
        for (int i = k; i < n; i++) {
            cur += list.get(i) - list.get(i - k);
            prefixSums[i - k + 1] = cur;
        }
        if (isIncorrectSequence(prefixSums)) {
            System.out.println("Incorrect sequence");
            return;
        }
        long[] restored = restoreSequence(prefixSums, list, k);
        if (restored == null) {
            System.out.println("Incorrect sequence");
            return;
        }
        Arrays.stream(restored).boxed().forEach(System.out::print);
        System.out.println();
    }

    private static long[] restoreSequence(long[] prefixSums, List<Integer> list, int k) {
        long[] restored = new long[list.size()];
        for (int i = 0; i < k; i++) {
            restored[i] = list.get(i);
        }
        for (int i = k; i < list.size(); i++) {
            restored[i] = prefixSums[i - k + 1] - prefixSums[i - k] + restored[i - k];
            if (restored[i] != list.get(i) && list.get(i) != 0) {
                return null;
            }
        }
        return restored;
    }

    private static boolean isIncorrectSequence(long[] prefixSums) {
        for (int i = 1; i < prefixSums.length; i++) {
            if (prefixSums[i] <= prefixSums[i - 1]) {
                return true;
            }
        }
        return false;
    }
}
