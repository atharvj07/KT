#include <iostream>

using namespace std;

int main() {
  int n, k;
  cin >> n >> k;
  int x[k];
  for (int i = 0; i < k; i++) cin >> x[i];
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n; j++) {
      bool ok = true;
      for (int l = 0; l < k; l++) {
        if (i == x[l] || j == x[l]) {
          ok = false;
          break;
        }
      }
      if (ok) ans++;
    }
  }
  cout << ans << endl;
  return 0;
}
