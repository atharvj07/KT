def min_replacements_to_RBS(s):
    stack = []
    replace_count = 0
    for char in s:
        if char in "<{[(":
            stack.append(char)
        else:
            if not stack:
                return "Impossible"
            last_open = stack.pop()
            if last_open + char not in ["()", "[]", "{}", "<>"]:
                replace_count += 1
    if len(stack) != 0:
        return "Impossible"
    return replace_count

s = input().strip()
print(min_replacements_to_RBS(s))
