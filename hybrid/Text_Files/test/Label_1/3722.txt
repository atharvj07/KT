#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    int a, b;
    cin >> a >> b;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];

    int sum = 0;
    for (int i = 0; i < n; i++) sum += h[i];

    if (sum > a + b) {
        cout << -1 << endl;
        return 0;
    }

    int ans = 0;
    sort(h.begin(), h.end());
    if (sum <= a) {
        for (int i = 0; i < n - 1; i++) ans += h[i];
    } else if (sum <= a + b) {
        for (int i = 0; i < n - 1; i++) {
            if (h[i] <= h[i + 1]) ans += h[i];
            else break;
        }
    }
    cout << ans << endl;
    return 0;
}
