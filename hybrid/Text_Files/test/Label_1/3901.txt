#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
#define endl '\n'

vector<vector<ll>> adj;
vector<ll> d;
vector<ll> up;

void dfs(ll v, ll p)
{
    d[v] = d[p] + 1;

    for(auto u : adj[v])
    {
        if(u != p)
        {
            dfs(u, v);
            up[v] = max(up[v], 1 + up[u]);
        }
    }
}

vector<ll> res;

int main()
{
    ios_base :: sync_with_stdio(false);
    cin.tie(nullptr);
    cout.tie(nullptr);

    ll n;
    cin >> n;
    adj.resize(n+1);
    d.resize(n+1);
    up.resize(n+1);

    vector<ll> a(n+1);
    for(ll i=1; i<=n; i++)
    {
        cin >> a[i];
    }

    for(ll i=1; i<n; i++)
    {
        ll u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    dfs(1, 0);

    res.resize(n+1);
    for(ll i=n; i>=1; i--)
    {
        for(auto u : adj[i])
        {
            if(d[u] < d[i])
            {
                res[i] = max(res[i], 1 + res[u]);
            }
        }
        res[i] = max(res[i], up[i] - 1);
    }

    for(ll i=1; i<=n; i++)
    {
        if(a[i])
        {
            res[i] = -res[i];
        }
        cout << res[i] << " ";
    }

    return 0;
}
