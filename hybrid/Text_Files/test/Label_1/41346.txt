n = int(input())
p = list(map(int, input().split()))
a = list(map(int, input().split()))
b = list(map(int, input().split()))
m = int(input())
c = list(map(int, input().split()))
ans = []
for i in c:
    t = []
    for j in range(n):
        if i in [a[j], b[j]]:
            t.append(p[j])
    if len(t) == 0:
        ans.append(-1)
    else:
        ans.append(min(t))
print(*ans)
