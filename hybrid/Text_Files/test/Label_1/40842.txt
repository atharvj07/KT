n, q = map(int, input().split())
folds = []

for _ in range(q):
    query = list(map(int, input().split()))
    if query[0] == 1:
        folds.append(query[1])
    else:
        l, r = query[1], query[2]
        width = n
        for fold in reversed(folds):
            if fold >= r:
                continue
            elif l <= fold < r:
                r = fold + (fold - l)
            else:
                l, r = fold - (r - fold), fold - (l - fold)
            width = fold
        print(r - l)
