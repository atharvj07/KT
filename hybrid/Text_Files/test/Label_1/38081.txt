n = int(input())
prices = list(map(int, input().split()))
front_colors = list(map(int, input().split()))
back_colors = list(map(int, input().split()))
m = int(input())
favorite_colors = list(map(int, input().split()))

# Create dictionaries to hold t-shirts by color with prices as keys
tshirts_by_color = {1: {}, 2: {}, 3: {}}

for i in range(n):
    if front_colors[i] in tshirts_by_color:
        tshirts_by_color[front_colors[i]][prices[i]] = i
    if back_colors[i] in tshirts_by_color and back_colors[i] != front_colors[i]:
        tshirts_by_color[back_colors[i]][prices[i]] = i

# Sort the t-shirts by price for each color
for color in tshirts_by_color:
    tshirts_by_color[color] = dict(sorted(tshirts_by_color[color].items()))

results = []

for color in favorite_colors:
    if tshirts_by_color[color]:
        # Get the cheapest t-shirt of the favorite color
        price, idx = list(tshirts_by_color[color].items())[0]
        results.append(price)
        # Remove the bought t-shirt
        del tshirts_by_color[color][price]
        # Also remove the t-shirt from the other color if it's there
        other_color = back_colors[idx] if front_colors[idx] == color else front_colors[idx]
        if price in tshirts_by_color[other_color]:
            del tshirts_by_color[other_color][price]
    else:
        results.append(-1)

print(' '.join(map(str, results)))
