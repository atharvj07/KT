#include <iostream>
#include <vector>
#include <queue>
#include <cstring>

using namespace std;

int n, m, k, dx[] = {1, 0, 0}, dy[] = {0, 1, 0}, dz[] = {0, 0, 1}, ans;
char a[110][110][110];
bool vis[110][110][110], need[110][110][110];

struct Node {
    int x, y, z, step;
};

int bfs(int x, int y, int z) {
    queue<Node> q;
    q.push({x, y, z, 0});
    vis[x][y][z] = 1;
    int step = 0;
    while (!q.empty()) {
        Node now = q.front();
        q.pop();
        step = now.step;
        for (int i = 0; i < 3; i++) {
            int tx = now.x + dx[i], ty = now.y + dy[i], tz = now.z + dz[i];
            if (tx < 1 || tx > n || ty < 1 || ty > m || tz < 1 || tz > k || a[tx][ty][tz] == '0' ||
                vis[tx][ty][tz]) continue;
            vis[tx][ty][tz] = 1;
            q.push({tx, ty, tz, now.step + 1});
        }
    }
    return step;
}

int main() {
    cin >> n >> m >> k;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            for (int t = 1; t <= k; t++) cin >> a[i][j][t];
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            for (int t = 1; t <= k; t++) {
                if (a[i][j][t] == '0') continue;
                if (!vis[i][j][t]) {
                    int step = bfs(i, j, t);
                    for (int ii = 1; ii <= n; ii++) {
                        for (int jj = 1; jj <= m; jj++) {
                            for (int tt = 1; tt <= k; tt++) {
                                if (a[ii][jj][tt] == '0' || vis[ii][jj][tt] ||
                                    bfs(ii, jj, tt) != step) continue;
                                need[ii][jj][tt] = 1;
                            }
                        }
                    }
                }
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            for (int t = 1; t <= k; t++) {
                if (a[i][j][t] == '0' || need[i][j][t]) continue;
                int step = bfs(i, j, t);
                if (step) ans++;
            }
        }
    }
    cout << ans;
    return 0;
}
