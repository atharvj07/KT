mod = 10**9 + 7
s = ""
n = int(input())
wtcnt = [0] * 30
dp = [0] * 30
for j in range(n):
    c = int(input())
    s += str(c)
    if len(s) >= 5:
        s = s[len(s) - 4:]
    idx = int(s, 2)
    dp[idx] = (dp[idx] + wtcnt[idx] + 1) % mod
    for i in range(0, 30):
        wtcnt[i] = (wtcnt[i] + dp[i]) % mod
    print(dp[idx])
