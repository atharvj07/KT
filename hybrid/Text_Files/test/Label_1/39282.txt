q = int(input())
for _ in range(q):
    n, k = map(int, input().split())
    s = input()
    patterns = ["RGB" * (n // 3 + 1), "GBR" * (n // 3 + 1), "BRG" * (n // 3 + 1)]
    min_changes = float('inf')
    for pattern in patterns:
        for start in range(n-k+1):
            changes = sum(1 for i in range(k) if s[start+i] != pattern[start+i])
            min_changes = min(min_changes, changes)
    print(min_changes)
