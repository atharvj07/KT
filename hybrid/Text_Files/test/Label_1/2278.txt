#include <cstdio>
#include <vector>
#include <algorithm>
#include <bitset>
using namespace std;
const int maxn = 2e5+5;
int n, S[maxn], a[maxn<<1], b[maxn];
vector<int> G[maxn];
vector<bool> vis;
int dfs(int u, int p, int x=0) {
  vis[u] = true;
  if(a[u] != -1) {
    if(x != a[u]) return -1;
  } else {
    a[u] = x;
    a[u^1] = x^1;
  }
  for(int &v: G[u]) {
    if(v != p) {
      int t = dfs(v, u, x^S[v]);
      if(t == -1) return -1;
    }
  }
  return 1;
}
int main() {
  scanf("%d", &n);
  for(int i=0;i<n;i++) {
    scanf("%d", &S[i]);
  }
  vis.resize(maxn<<1);
  for(int i=0;i<n;i++) {
    for(int j=i+1;j<n;j++) {
      if(S[i] == S[j]) {
        puts("0");
        return puts("0"), 0;
      }
      G[S[i]].push_back(S[j]);
      G[S[j]].push_back(S[i]);
    }
  }
  int ans = 0;
  for(int i=0;i<n;i++) {
    if(!vis[S[i]]) {
      a[S[i]] = a[S[i]^1] = -1;
      if(dfs(S[i], -1) == -1) {
        puts("0");
        return puts("0"), 0;
      }
    }
  }
  bool flg = false;
  for(int i=0;i<n;i++) {
    if(!vis[S[i]]) {
      for(int j=0;j<n;j++) {
        if(a[S[j]] == 1) {
          ans = max(ans, b[S[j]]);
          flg = true;
          break;
        }
        b[S[j]]++;
      }
      break;
    }
  }
  printf("%d\n", ans);
  for(int x=0;x<(1<<ans);x++) {
    int p = 0;
    for(int i=0;i<n;i++) {
      if(a[S[i]] == 1) {
        if(x & (1<<p)) {
          p--;
        }
        printf("%d ", S[i]^p);
        p += b[S[i]];
      }
    }
  }
  puts("");
  return 0;
}
