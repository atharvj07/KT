import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] points = new int[8][];
        for (int i = 0; i < 8; i++) {
            int[] point = new int[3];
            for (int j = 0; j < 3; j++) {
                point[j] = scanner.nextInt();
            }
            points[i] = point;
        }

        if (!isCube(points)) {
            System.out.println("NO");
        } else {
            System.out.println("YES");
            for (int[] point : points) {
                Arrays.sort(point);
                for (int j = 0; j < 3; j++) {
                    System.out.print(point[j] + " ");
                }
                System.out.println();
            }
        }
    }

    private static boolean isCube(int[][] points) {
        // Check if all the points are distinct
        for (int i = 0; i < 8; i++) {
            for (int j = i + 1; j < 8; j++) {
                if (Arrays.equals(points[i], points[j])) {
                    return false;
                }
            }
        }

        // Check if all the points are on a cube
        for (int[] point : points) {
            if (point[0] == point[1] || point[1] == point[2] || point[0] == point[2]) {
                return false;
            }
        }

        // Check if the points form a valid cube
        int[] xValues = new int[8];
        int[] yValues = new int[8];
        int[] zValues = new int[8];
        for (int i = 0; i < 8; i++) {
            xValues[i] = points[i][0];
            yValues[i] = points[i][1];
            zValues[i] = points[i][2];
        }

        Arrays.sort(xValues);
        Arrays.sort(yValues);
        Arrays.sort(zValues);

        if (xValues[1] - xValues[0] != xValues[2] - xValues[1] ||
                yValues[1] - yValues[0] != yValues[2] - yValues[1] ||
                zValues[1] - zValues[0] != zValues[2] - zValues[1]) {
            return false;
        }

        return true;
    }
}
