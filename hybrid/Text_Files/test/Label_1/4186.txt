#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<long long> times(n);
  for (int i = 0; i < n; ++i) cin >> times[i];
  sort(times.begin(), times.end());
  long long cur = 0, ans = 0;
  for (int i = 0; i < n; ++i) {
    if (cur + times[i] <= times[i]) {
      ++ans;
      cur += times[i];
    }
  }
  cout << ans << endl;
  return 0;
}
