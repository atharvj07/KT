#include <iostream>
using namespace std;
int n;
long long dp[11][1<<4];
int a[] = {1, 5, 10, 50};
long long f(int i, int mask) {
    if (i == n) return mask > 0;
    long long &ans = dp[i][mask];
    if (ans != -1) return ans;
    ans = 0;
    for (int j = 0; j < 4; j++) {
        if (!(mask & (1<<j)))
            ans += f(i + 1, mask | (1<<j));
    }
    return ans;
}
int main() {
    cin >> n;
    for (int i = 0; i < 11; i++)
        for (int j = 0; j < (1<<4); j++)
            dp[i][j] = -1;
    cout << f(0, 0) << '\n';
}
