for _ in range(int(input())):
    n, m = map(int, input().split())
    a = []
    for __ in range(n):
        a.append(list(map(int, input().split())))
    can = True
    for i in range(n):
        for j in range(m // 2):
            if a[i][j] != a[i][m - j - 1]:
                can = False
                break
    for i in range(m):
        for j in range(n // 2):
            if a[j][i] != a[n - j - 1][i]:
                can = False
                break
    if can:
        print(0)
    else:
        print(sum(abs(a[i][j] - a[i][m - j - 1]) for i in range(n) for j in range(m // 2)) + sum(abs(a[i][j] - a[n - i - 1][j]) for i in range(n) for j in range(m // 2, m)))
