t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    edges = [input() for _ in range(n)]
    if m % 2 == 1:
        print("YES")
        print(" ".join(str((i % n) + 1) for i in range(m + 1)))
    else:
        found = False
        for i in range(n):
            for j in range(i + 1, n):
                if edges[i][j] == edges[j][i]:
                    print("YES")
                    if m % 4 == 0:
                        print(" ".join([str(i + 1), str(j + 1)] * (m // 2) + [str(i + 1)]))
                    else:
                        mid = [str(i + 1), str(j + 1)] * (m // 4)
                        print(" ".join(mid + [str(i + 1)] + mid[::-1]))
                    found = True
                    break
            if found:
                break
        if not found:
            print("NO")
