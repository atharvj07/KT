#include <bits/stdc++.h>
using namespace std;
const int maxn = 3000 + 5;
int a[maxn];
int32_t main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    int ans = 0;
    for (int i = 2; i <= n; ++i) {
        if (a[i] <= a[i - 1]) ans += a[i - 1] - a[i], a[i] = a[i - 1] + 1;
    }
    cout << ans << '\n';
}
