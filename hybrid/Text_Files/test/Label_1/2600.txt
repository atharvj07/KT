#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n;
    std::vector<std::vector<int>> cards(n);
    for (int i = 0; i < n; i++) {
        std::cin >> m;
        cards[i].resize(m);
        for (int j = 0; j < m; j++) { std::cin >> cards[i][j]; }
    }
    std::vector<int> crossed(101, 0);
    for (int i = 0; i < n; i++) {
        int ans = 1;
        for (int j = 0; j < cards[i].size(); j++) {
            if (!crossed[cards[i][j]]) {
                ans = 0;
                break;
            }
        }
        std::cout << (ans ? "YES" : "NO") << std::endl;
    }
    return 0;
}
