#include <iostream>
#include <algorithm>
using namespace std;
string s1, s2;
int main() {
  cin >> s1 >> s2;
  int n = s1.size(), m = s2.size();
  if (n > m) {
    swap(s1, s2);
    swap(n, m);
  }
  reverse(s1.begin(), s1.end());
  reverse(s2.begin(), s2.end());
  int l = n, r = n - 1;
  bool fl = 1;
  for (int i = n; i < m; i++) {
    if (s2[i] == '1') {
      fl = 0;
      break;
    }
  }
  if (fl) {
    cout << "YES\n";
    return 0;
  }
  while (l >= 0 && r >= 0) {
    if (s1[l] != s2[l] && s1[r] != s2[l]) {
      cout << "NO\n";
      return 0;
    }
    l--, r--;
  }
  while (l >= 0) {
    if (s1[l] != s2[l] && s1[r] != s2[l]) {
      cout << "NO\n";
      return 0;
    }
    l--;
    r++;
  }
  cout << "YES\n";
  return 0;
}
