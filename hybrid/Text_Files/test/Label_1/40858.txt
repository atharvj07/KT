n = int(input())
a = list(map(int, input().split()))

def find_sure_new_messages(a):
    last_position = 0
    for i in range(len(a)-1, -1, -1):
        if a[i] == i + 1:
            last_position += 1
        else:
            break
    return len(a) - last_position

print(find_sure_new_messages(a))
