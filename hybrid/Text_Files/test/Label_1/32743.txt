import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] nm = br.readLine().split(" ");
        int n = Integer.parseInt(nm[0]);
        int m = Integer.parseInt(nm[1]);

        ArrayList<ArrayList<Integer>> adj = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            adj.add(new ArrayList<>());
        }

        for (int i = 0; i < m; i++) {
            String[] kps = br.readLine().split(" ");
            int k = Integer.parseInt(kps[0]);
            for (int j = 0; j < k; j++) {
                int s = Integer.parseInt(kps[j + 1]);
                adj.get(s - 1).add(i);
            }
        }

        String[] p = br.readLine().split(" ");

        int ans = 0;

        for (int bit = 0; bit < (1 << n); bit++) {
            boolean ok = true;
            for (int i = 0; i < m; i++) {
                int cnt = Integer.bitCount(bit & (1 << adj.get(i).get(0)));
                for (int j = 1; j < adj.get(i).size(); j++) {
                    cnt += Integer.bitCount(bit & (1 << adj.get(i).get(j)));
                }
                if (cnt % 2 != p[i].charAt(0) - '0') {
                    ok = false;
                }
            }
            if (ok) {
                ans++;
            }
        }

        System.out.println(ans);
    }
}
