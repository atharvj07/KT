from math import gcd

def can_form_bst(n, vertices):
    dp = [[0] * n for _ in range(n)]
    
    for length in range(1, n + 1):
        for start in range(n - length + 1):
            end = start + length - 1
            if length <= 2:
                dp[start][end] = 1
            else:
                for mid in range(start + 1, end):
                    if gcd(vertices[start], vertices[mid]) > 1 and dp[start][mid] and dp[mid][end]:
                        dp[start][end] = 1
                        break
                if not dp[start][end]:
                    if gcd(vertices[start], vertices[end]) > 1 and dp[start + 1][end - 1]:
                        dp[start][end] = 1
    
    for i in range(1, n):
        if (dp[0][i - 1] and dp[i][n - 1] and gcd(vertices[0], vertices[i]) > 1) or \
           (dp[1][i] and dp[i + 1][n - 1] and gcd(vertices[n - 1], vertices[i]) > 1):
            return "Yes"
    return "No"

n = int(input().strip())
vertices = list(map(int, input().strip().split()))

print(can_form_bst(n, vertices))
