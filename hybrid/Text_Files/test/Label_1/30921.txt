import java.util.Scanner;

public class HarryAndSnitch {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        double[][] points = new double[n + 1][3];
        for (int i = 0; i <= n; i++) {
            points[i][0] = scanner.nextDouble();
            points[i][1] = scanner.nextDouble();
            points[i][2] = scanner.nextDouble();
        }

        double vp = scanner.nextDouble();
        double vs = scanner.nextDouble();
        double px = scanner.nextDouble();
        double py = scanner.nextDouble();
        double pz = scanner.nextDouble();

        double time = catchSnitch(points, vp, vs, px, py, pz);

        if (time >= 0) {
            System.out.println("YES");
            System.out.println(time);
            double[] catchPoint = catchPoint(points, vp, vs, px, py, pz, time);
            System.out.println(catchPoint[0] + " " + catchPoint[1] + " " + catchPoint[2]);
        } else {
            System.out.println("NO");
        }
    }

    private static double catchSnitch(double[][] points, double vp, double vs, double px, double py, double pz) {
        double minTime = Double.MAX_VALUE;
        for (int i = 0; i < points.length - 1; i++) {
            double timeToCatchOnSegment = catchOnSegment(points[i], points[i + 1], vp, vs, px, py, pz);
            if (timeToCatchOnSegment >= 0) {
                minTime = Math.min(minTime, timeToCatchOnSegment);
            }
        }
        return minTime == Double.MAX_VALUE ? -1 : minTime;
    }

    private static double catchOnSegment(double[] p1, double[] p2, double vp, double vs, double px, double py, double pz) {
        double dx = p2[0] - p1[0];
        double dy = p2[1] - p1[1];
        double dz = p2[2] - p1[2];
        double length = Math.sqrt(dx * dx + dy * dy + dz * dz);
        double t1 = (px - p1[0]) * dx + (py - p1[1]) * dy + (pz - p1[2]) * dz;
        double t2 = (px - p2[0]) * dx + (py - p2[1]) * dy + (pz - p2[2]) * dz;

        if (t1 < 0 || t2 > length) {
            return -1;
        }

        double projectionX = p1[0] + t1 * dx / length;
        double projectionY = p1[1] + t1 * dy / length;
        double projectionZ = p1[2] + t1 * dz / length;

        double distanceToProjection = Math.sqrt((px - projectionX) * (px - projectionX) + (py - projectionY) * (py - projectionY) + (pz - projectionZ) * (pz - projectionZ));

        double timeToCatchOnSegment = Math.sqrt(distanceToProjection * distanceToProjection / (vp * vp - vs * vs)) + t1 / vs;

        return timeToCatchOnSegment;
    }

    private static double[] catchPoint(double[][] points, double vp, double vs, double px, double py, double pz, double time) {
        for (int i = 0; i < points.length - 1; i++) {
            double timeToCatchOnSegment = catchOnSegment(points[i], points[i + 1], vp, vs, px, py, pz);
            if (Math.abs(timeToCatchOnSegment - time) < 1e-6) {
                double dx = points[i + 1][0] - points[i][0];
                double dy = points[i + 1][1] - points[i][1];
                double dz = points[i + 1][2] - points[i][2];
                double length = Math.sqrt(dx * dx + dy * dy + dz * dz);
                double t = (px - points[i][0]) * dx + (py - points[i][1]) * dy + (pz - points[i][2]) * dz;
                double x = points[i][0] + t * dx / length;
                double y = points[i][1] + t * dy / length;
                double z = points[i][2] + t * dz / length;
                return new double[]{x, y, z};
            }
        }
        return null;
    }
}
