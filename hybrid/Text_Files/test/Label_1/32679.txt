import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class PaintingThePicture {
    static final int MODULO = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int k = Integer.parseInt(line[1]);

        int[][][][] dp = new int[n + 1][k + 1][4][4];
        dp[0][0][0][0] = 1;

        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                for (int prev1 = 0; prev1 < 4; prev1++) {
                    for (int prev2 = 0; prev2 < 4; prev2++) {
                        if (j > 0) {
                            for (int point1 = 0; point1 < 4; point1++) {
                                if (point1 != prev1 && point1 != prev2) {
                                    for (int point2 = 0; point2 < 4; point2++) {
                                        if (point2 != prev1 && point2 != prev2 && point2 != point1) {
                                            for (int point3 = 0; point3 < 4; point3++) {
                                                if (point3 != prev1 && point3 != prev2 && point3 != point1 && point3 != point2) {
                                                    if (isValidTriangle(point1, point2, point3)) {
                                                        dp[i][j][point1][point2] = (dp[i][j][point1][point2] + dp[i][j - 1][prev1][prev2]) % MODULO;
                                                    }
                                                }
                                            }
                                        }
                                    }
                                }
                            }
                        }
                        if (i < n) {
                            dp[i + 1][j][0][1] = (dp[i + 1][j][0][1] + dp[i][j][prev1][prev2]) % MODULO;
                            dp[i + 1][j][1][2] = (dp[i + 1][j][1][2] + dp[i][j][prev1][prev2]) % MODULO;
                            dp[i + 1][j][2][0] = (dp[i + 1][j][2][0] + dp[i][j][prev1][prev2]) % MODULO;
                            dp[i + 1][j][2][3] = (dp[i + 1][j][2][3] + dp[i][j][prev1][prev2]) % MODULO;
                            dp[i + 1][j][3][1] = (dp[i + 1][j][3][1] + dp[i][j][prev1][prev2]) % MODULO;
                            dp[i + 1][j][3][2] = (dp[i + 1][j][3][2] + dp[i][j][prev1][prev2]) % MODULO;
                        }
                    }
                }
            }
        }

        int result = 0;
        for (int point1 = 0; point1 < 4; point1++) {
            for (int point2 = 0; point2 < 4; point2++) {
                if (point1 != point2) {
                    result = (result + dp[n][k][point1][point2]) % MODULO;
                }
            }
        }

        System.out.println(result);
    }

    static boolean isValidTriangle(int point1, int point2, int point3) {
        return point1 != point2 && point1 != point3 && point2 != point3;
    }
}
