import java.util.*;
import java.io.*;
import java.math.BigInteger;

public class Main {
  private static final int MOD = 998244353;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    int n = Integer.parseInt(br.readLine());
    StringTokenizer st = new StringTokenizer(br.readLine());
    long[] arr = new long[n];
    for (int i = 0; i < n; i++) {
      arr[i] = Integer.parseInt(st.nextToken());
    }

    // dp[i] = the number of good subsequences ending at index i
    long[] dp = new long[n];
    dp[n - 1] = 1;
    for (int i = n - 2; i >= 0; i--) {
      if (arr[i] >= i) {
        // arr[i] forms a good array by itself
        dp[i] = (dp[i] + 1) % MOD;
      }
      // append arr[i] to a good subsequence ending at index j
      for (int j = i + 1; j < n; j++) {
        if (arr[i] == j - 1) {
          dp[i] = (dp[i] + dp[j]) % MOD;
        }
      }
    }

    System.out.println(dp[0]);
  }
}
