#include <bits/stdc++.h>
using namespace std;
const int maxn = 1e5 + 5;
int n, a[2*maxn], b[2*maxn], Pa[2*maxn], Pb[2*maxn], ans;
int main() {
	cin >> n;
	for ( int i = 0; i < n; ++i ) cin >> a[i];
	for ( int i = 0, cnt = 0; i < n; ++i )
		if ( a[i] == '(' ) b[cnt++] = i;
		else if ( cnt ) b[cnt++] = i + maxn, --cnt;

	for ( int i = 0, cnt = 0; i < 2*n; ++i ) {
		if ( a[b[i] % maxn] == '(' ) Pa[b[i]] = ++cnt;
		else Pb[b[i]] = ++cnt;
	}

	for ( int i = 0; i < 2*n; ++i )
		if ( Pa[i] and Pb[i] and Pa[i] - Pb[i] <= 1 ) ++ans;

	cout << ans/2 << endl;
}
