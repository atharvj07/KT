#include <bits/stdc++.h>
using namespace std;
 
void solve() {
  typedef long long ll;
  int n;
  cin >> n;
  vector<ll> a(n);
  for (ll& x : a) cin >> x;
  ll g = 0;
  for (auto x : a) g = __gcd(g, x);
  if (g > 1)
    cout << g << '\n';
  else
    cout << 0 << '\n';
}
 
int main() {
  int t;
  cin >> t;
  while (t--) solve();
  return 0;
}
