import math
n = int(input())
f = [0] * (n+1)
for i in range(2, int(math.sqrt(n)) + 1):
    if f[i] == 0:
        for j in range(i*i, n+1, i):
            f[j] = 1
c = 0
for i in range(2, n+1):
    if f[i] == 0:
        c += 1
print(c)
for i in range(2, n+1):
    if f[i] == 0:
        print(1, end= ' ')
    else:
        print(2, end= ' ')
