import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();

        int[] deg = new int[n + 1];
        for (int i = 0; i < n; i++) {
            deg[a[i]]++;
            deg[n - a[i]]++;
        }

        List<Integer> ans = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            if (deg[i] % 2 == 1) {
                ans.add(i);
            }
        }

        if (ans.isEmpty()) {
            System.out.println(-1);
        } else {
            for (int i : ans) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
