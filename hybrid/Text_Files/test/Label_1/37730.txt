from collections import defaultdict
n, m, d = map(int, input().split())
affected = set(map(int, input().split()))
path = defaultdict(list)
for _ in range(n - 1):
    a, b = map(int, input().split())
    path[a].append(b)
    path[b].append(a)
visited = [False] * (n + 1)
can_contain = [False] * (n + 1)
def dfs(node, dist):
    if visited[node]:
        return
    visited[node] = True
    if dist <= d:
        can_contain[node] = True
    for nei in path[node]:
        dfs(nei, dist + 1)
for node in affected:
    dfs(node, 0)
print(sum(can_contain))
