#include <bits/stdc++.h>
using namespace std;

int main() {
  string s;
  int x, y;
  cin >> s >> x >> y;
  int ans = 4e8;
  for (int i = 0; i < s.length(); i++) {
    string t = s;
    if (s[i] == '?') t[i] = '0';
    int cnt1 = 0, cnt2 = 0;
    for (int j = 0; j < t.length(); j++) {
      if (t[j] == '0')
        cnt1 += (j > 0 && t[j - 1] == '1');
      else
        cnt2 += (j > 0 && t[j - 1] == '0');
    }
    ans = min(ans, cnt1 * x + cnt2 * y);
    t = s;
    if (s[i] == '?') t[i] = '1';
    cnt1 = cnt2 = 0;
    for (int j = 0; j < t.length(); j++) {
      if (t[j] == '0')
        cnt1 += (j > 0 && t[j - 1] == '1');
      else
        cnt2 += (j > 0 && t[j - 1] == '0');
    }
    ans = min(ans, cnt1 * x + cnt2 * y);
  }
  cout << ans << endl;
  return 0;
}
