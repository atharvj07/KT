import sys

n, p = map(int, input().split())
S = [[], [], []]
for i in range(1, n + 1):
    x, y = map(int, input().split())
    for j in range(3):
        if i in S[j]:
            break
        S[j].append(i)
        if x not in S[j]:
            S[j].append(x)
        if y not in S[j]:
            S[j].append(y)
        if len(S[j]) >= p:
            break
print(len(set(S[0]) & set(S[1]) & set(S[2])))
