n = int(input())
a, b, c = [], [], []
for _ in range(n):
    ai, bi, ci = map(int, input().split())
    a.append(ai); b.append(bi); c.append(ci)
pos = {(ai, bi): ci for ai, bi, ci in zip(a, b, c)}
ans = 0
for i in range(n):
    for j in range(n):
        if (b[i], a[j]) in pos:
            ans += pos[(a[i], b[j])] * pos[(b[i], a[j])]
print(ans)
