import sys
input = sys.stdin.readline

n, m = map(int, input().split())
edges = [[] for i in range(n+1)]
for i in range(m):
    a, b = map(int, input().split())
    edges[a].append(b)
    edges[b].append(a)

vis = [0] * (n+1)
cl = [-1] * (n+1)
def dfs(x, c):
    vis[x] = 1
    cl[x] = c
    for y in edges[x]:
        if vis[y] == 0:
            dfs(y, 3-c)
for i in range(1, n+1):
    if vis[i] == 0:
        dfs(i, 1)
ans = [-1] * (n+1)
for i in range(1, n+1):
    if cl[i] == 1:
        ans[i] = 1
    elif cl[i] == 2:
        ans[i] = 2
    else:
        ans[i] = 3
if -1 in ans:
    print(-1)
else:
    print(*ans[1:])
