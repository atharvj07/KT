from sys import stdin

n, k, m, t = map(int, stdin.readline().split())
l, pos = n, k

for i in range(t):
    x, y = map(int, stdin.readline().split())
    if x == 0:
        l -= 1
        if y < pos:
            pos -= 1
    else:
        l += 1
        if y <= pos:
            pos += 1
    print(l, pos)
