#include <cstdio>
#include <cstring>
#include <algorithm>
#include <vector>
#include <queue>
using namespace std;
#define maxn 1005
#define maxm 4005
#define inf 0x3f3f3f3f
#define mod 1000000007
#define ll long long
typedef pair<int, int> pii;

char s[maxn][5], op[maxn];
int a[maxn], b[maxn], vis[maxn], vis2[maxn], n, tot, cnt, f;
vector<int> g[maxn];

void dfs1(int x) {
    if (vis[x]) {
        f = 1;
        return;
    }
    vis[x] = 1;
    if (a[x] != -1) dfs1(a[x]);
    for (int i = 0; i < g[x].size(); i++) {
        int y = g[x][i];
        if (!vis2[y]) {
            vis2[y] = 1;
            dfs1(b[y]);
        }
    }
}

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf(" %s", s[i]);
        int l = strlen(s[i]);
        int j;
        for (j = 1; j <= l; j++) {
            if (s[i][j] < 'a' || s[i][j] > 'z') break;
        }
        if (j <= l) {
            op[i] = s[i][j];
            a[i] = -1;
            b[i] = -1;
        } else {
            op[i] = '$';
            a[i] = tot + 1;
            b[i] = tot + 2;
            tot += 2;
        }
        for (j = 1; j <= l; j++) {
            if (s[i][j] >= 'a' && s[i][j] <= 'z') {
                while (s[i][j] >= 'a' && s[i][j] <= 'z') {
                    cnt++;
                    s[0][cnt] = s[i][j];
                    j++;
                }
                s[0][0] = cnt;
                s[0][cnt + 1] = 0;
                cnt = 0;
                g[a[i]].push_back(b[i]);
            }
        }
    }
    cnt = 0;
    s[0][cnt++] = '#';
    s[0][0] = cnt;
    s[0][cnt + 1] = 0;
    dfs1(1);
    f += tot != 0;
    if (!f)
        printf("0\n");
    else {
        printf("%d\n", f);
        if (f == 1) printf("#");
        for (int i = 1; i <= tot; i++) {
            if (a[i] != -1) printf("%sa", s[a[i]]);
            else
                printf("%c", op[i]);
            if (b[i] != -1) printf("%sb", s[b[i]]);
        }
        printf("\n");
    }
    return 0;
}
