import java.util.Scanner;
import java.util.ArrayList;
class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt(), m = input.nextInt(), ans = 0;
        ArrayList<Integer>[] a = new ArrayList[m + 1];
        for (int i = 0; i <= m; i++) a[i] = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int k = input.nextInt();
            for (int j = 0; j < k; j++) {
                int x = input.nextInt();
                if (!a[x].contains(i)) {
                    a[x].add(i);
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
