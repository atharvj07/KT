import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[4];
        for (int i = 0; i < 4; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();
        Arrays.sort(arr);
        long a = arr[0], b = arr[1], c = arr[2], l = arr[3];
        if (a + b <= c || a + c <= b || b + c <= a) {
            System.out.println(0);
        } else if (3 * l < a + b + c) {
            System.out.println(0);
        } else {
            l -= (a + b + c) / 3;
            long ans = Math.min(l + 1, c - Math.min(a, b) + 1);
            System.out.println(ans);
        }
    }
}
