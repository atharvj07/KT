import java.util.Scanner;

public class PermutationPeaks {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int t = scanner.nextInt();

        while (t-- > 0) {
            int n = scanner.nextInt();
            int k = scanner.nextInt();

            int[] permutation = constructPermutation(n, k);
            if (permutation == null) {
                System.out.println(-1);
            } else {
                for (int num : permutation) {
                    System.out.print(num + " ");
                }
                System.out.println();
            }
        }
    }

    private static int[] constructPermutation(int n, int k) {
        if (k > (n - 1) / 2) {
            return null; // Not possible to have more peaks than valleys
        }

        int[] permutation = new int[n];
        int index = 0;
        int currNum = 1;

        for (int i = 0; i < n; i++) {
            if (i % 2 == 0 || i == n - 1) { // Uphill
                permutation[index++] = currNum++;
            } else {
                permutation[index++] = currNum--; // Downhill
            }
        }

        // Adjust to create the desired number of peaks
        int peaks = (n - 1) / 2;
        while (peaks > k) {
            for (int i = 2; i < n - 2; i += 2) {
                if (permutation[i] > permutation[i - 1] && permutation[i] > permutation[i + 1]) {
                    swap(permutation, i, i - 1); // Flatten peak
                    peaks--;
                    break;
                }
            }
        }

        return permutation;
    }

    private static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
}
