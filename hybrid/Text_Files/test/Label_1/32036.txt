import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int q = sc.nextInt();
        List<Road> roads = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int w = sc.nextInt();
            roads.add(new Road(u, v, w));
        }
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int max = -1;
            for (Road road : roads) {
                if (l <= road.index && road.index <= r) {
                    max = Math.max(max, road.weight);
                }
            }
            System.out.println(max);
        }
    }

    private static class Road {
        int u;
        int v;
        int weight;
        int index;

        public Road(int u, int v, int weight) {
            this.u = u;
            this.v = v;
            this.weight = weight;
        }
    }
}
