import java.util.*;

class Main {
    static int m, n;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        m = scanner.nextInt();
        n = scanner.nextInt();
        if (n > m) { // If the capacity is more than the number of animals, Vasya can transport all animals in m trips.
            System.out.println(m);
            return;
        }

        if (2 * n < m) { // If the capacity is less than half the number of animals, it is impossible.
            System.out.println(-1);
            return;
        }

        if (m % 2 == n % 2) {
            System.out.println(m / n + (m % n == 0 ? 0 : 1));
        } else {
            System.out.println(m / n + 1);
        }
    }
}
