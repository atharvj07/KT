import math
N = int(input())
x = []
y = []
for _ in range(N):
    xi, yi = map(float, input().split())
    x.append(xi)
    y.append(yi)
x_min = min(x)
x_max = max(x)
if x_max + x_min < 0:
    print(-1)
else:
    r = math.sqrt((x_max - x_min) ** 2 / 4 + (abs(max(y)) + abs(min(y))) ** 2 / 4)
    print(r)
