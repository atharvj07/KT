import sys

def check(p, l, r):
    lm = p[l]
    for i in range(l, r):
        lm = min(lm, p[i])
    rm = p[r]
    for i in range(r + 1, len(p)):
        rm = max(rm, p[i])
    return lm == p[l] and rm == p[r]

for t in range(int(input())):
    n, a, b = map(int, input().split())
    p = []
    for i in range(n):
        if i % 2 == 0:
            p.append(i + 1)
        else:
            p.append(n - i)
    p[0], p[n - 1] = p[n - 1], p[0]
    if check(p, 0, n // 2 - 1):
        print(*p)
    else:
        print(-1)
