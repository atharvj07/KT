import sys
import heapq

input = sys.stdin.readline
n=int(input())
adj=[[] for i in range(n+1)]
for i in range(n-1):
    a,b,c=map(int,input().split())
    adj[a].append([b,c])
    adj[b].append([a,c])
X=[int(input()) for i in range(n)]
dist=[n+1]*(n+1)
dist[1]=0
vis=[False]*(n+1)
vis[1]=True
q=[]
heapq.heappush(q,[0,1])
Max=0
while len(q)>0:
    d,x=heapq.heappop(q)
    if d>dist[x]:
        continue
    Max=max(Max,d)
    for ver,weight in adj[x]:
        if vis[ver]==True:
            continue
        vis[ver]=True
        dist[ver]=d+weight
        heapq.heappush(q,[d+weight,ver])
print(Max)
