n, k = map(int, input().split())
pixels = list(map(int, input().split()))

group_keys = [-1] * 256

for pixel in pixels:
    if group_keys[pixel] == -1:
        start = max(0, pixel - (k - 1))
        for i in range(start, pixel + 1):
            if group_keys[i] != -1:
                start = max(start, group_keys[i])
                break
        for i in range(start, start + k):
            if i > 255:
                break
            if group_keys[i] == -1:
                group_keys[i] = start
            else:
                break

print(' '.join(str(group_keys[pixel]) for pixel in pixels))
