#include<bits/stdc++.h>
using namespace std;
int main() {
  int n, m; cin >> n >> m;
  set<int> s;
  for (int i = 0; i < m; i++) {
    int x; cin >> x;
    s.insert(x);
  }
  if (s.count(1) || s.count(n)) {
    cout << "NO" << endl;
    return 0;
  }
  for (int i = 2; i < n; i++) {
    if (s.count(i) || s.count(i + 1) || s.count(i + 2)) {
      cout << "NO" << endl;
      return 0;
    }
  }
  cout << "YES" << endl;
}
