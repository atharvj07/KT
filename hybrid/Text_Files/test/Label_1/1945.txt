#include <bits/stdc++.h>

int main() {
  long long n, k;
  std::cin >> n >> k;
  if (n < k or 2 * (n - k) < k) {
    std::cout << "NO" << std::endl;
  } else {
    std::cout << "YES" << std::endl;
    long long t = (n - k) / k;
    for (long long i = 0; i < k; ++i) {
      std::cout << t + (2 * i >= k) << ' ';
    }
  }
  return 0;
}
