import sys
input = sys.stdin.readline

n = int(input())
c = list(map(int, input().split()))
z = []
cur = [0] * 26
for i in range(1, n + 1):
    for j in range(26):
        if cur[j] > 0:
            cur[j] -= 1
            for k in range(c[i - 1]):
                z.append(chr(j + ord('a')))
            cur[j] += 1
    for j in range(26):
        if cur[j] == 0:
            z.append(chr(j + ord('a')))
            cur[j] = c[i - 1]
            break
print(''.join(z))
