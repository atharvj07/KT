#include<bits/stdc++.h>
using namespace std;
#define mt make_tuple
typedef long long ll;
const int MAXN = 1e5 + 5;
const int INF = 1e9 + 7;
vector<int> adj[MAXN];
bool mark[MAXN];
int sz[MAXN];
void dfs(int u, int p, int id) {
	mark[u] = true;
	sz[id]++;
	for (int v : adj[u]) {
		if (!mark[v] && v != p) {
			dfs(v, u, id);
		}
	}
}
int main() {
	int n, m; cin >> n >> m;
	for (int i = 0; i < m; i++) {
		int u, v; cin >> u >> v;
		adj[u].push_back(v);
		adj[v].push_back(u);
	}
	vector<tuple<int, int, int>> res;
	int cnt = 0;
	for (int i = 1; i <= n; i++) {
		if (!mark[i]) {
			dfs(i, i, ++cnt);
			res.push_back(mt(sz[cnt], i, cnt));
		}
	}
	sort(res.rbegin(), res.rend());
	if (res.size() < 3) {
		cout << -1 << endl;
		return 0;
	}
	int sum = get<0>(res[0]) + get<0>(res[1]) + get<0>(res[2]);
	if (sum != n) {
		cout << -1 << endl;
		return 0;
	}
	set<int> ver;
	for (auto x : res) {
		if (!ver.count(get<2>(x))) {
			for (int i = 1; i <= n; i++) {
				if (!mark[i]) {
					dfs(i, i, get<2>(x));
					res.push_back(mt(sz[get<2>(x)], i, get<2>(x)));
					ver.insert(get<2>(x));
					break;
				}
			}
		}
	}
	sort(res.rbegin(), res.rend());
	for (int i = 0; i < 3; i++) {
		cout << get<2>(res[i]) << " ";
		for (int j = 1; j <= n; j++) {
			if (mark[j] == get<2>(res[i])) {
				cout << j << " ";
			}
		}
		cout << endl;
	}
	return 0;
}
