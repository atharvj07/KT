n = int(input())
for i in range(1, 5):
    a, b, c, d = map(int, input().split())
    if a + c <= n:
        print(i, a, n - a)
        break
    elif a + d <= n:
        print(i, a, n - a)
        break
    elif b + c <= n:
        print(i, b, n - b)
        break
    elif b + d <= n:
        print(i, b, n - b)
        break
else:
    print(-1)
