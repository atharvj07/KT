#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl "\n"

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<pair<ll, ll>> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i].first >> a[i].second;
        }
        sort(a.begin(), a.end());
        bool ok = true;
        int last = -1e9;
        vector<int> ans(n);
        for (int i = 0; i < n; i++) {
            if (a[i].first > last) {
                ans[i] = 1;
                last = a[i].second;
            } else {
                ok = false;
            }
        }
        if (!ok) {
            cout << "-1" << endl;
        } else {
            for (int i = 0; i < n; i++) {
                cout << ans[i] << " ";
            }
            cout << endl;
        }
    }
    return 0;
}
