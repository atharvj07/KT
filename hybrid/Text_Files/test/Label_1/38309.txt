from bisect import bisect_left

t=int(input())
for _ in range(t):
    n,m=map(int,input().split())
    a=[int(i) for i in input().split()]
    b=[int(i) for i in input().split()]
    a.sort()
    ans=[]
    for x in b:
        i=bisect_left(a,x)
        if 0<=i<n and a[i]==x:
            ans.append(x)
    if ans:
        print("YES")
        print(len(ans))
        print(*ans)
    else:
        print("NO")
