import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int n = sc.nextInt();
        int[] a = new int[r + 1];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            a[x]++;
        }
        boolean win = true;
        for (int i = 1; i <= r; i++) {
            if ((a[i] & 1) != 0) {
                win = !win;
            }
        }
        System.out.println(win ? "WIN" : "LOSE");
    }
}
