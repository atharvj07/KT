import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int[] p = new int[n];
		for (int i = 0; i < n; i++)
			p[i] = sc.nextInt();
		int k = sc.nextInt();
		int ans = 0;
		for (int i = 0; i < (1 << n); i++) {
			int x = 1;
			boolean ok = true;
			for (int j = 0; j < n; j++)
				if (((i >> j) & 1) == 1) {
					x *= p[j];
					if (x > 1000000000000000000L) {
						ok = false;
						break;
					}
				}
			if (ok)
				ans++;
			if (ans == k) {
				System.out.println(x);
				break;
			}
		}
		sc.close();
	}
}
