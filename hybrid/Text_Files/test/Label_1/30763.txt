import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        Arrays.sort(a);
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != a[0]) break;
            ans += (long) (n - i - 1) * (n - i - 2) / 2;
        }
        System.out.println(ans);
    }
}
