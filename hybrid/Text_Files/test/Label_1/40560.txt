from collections import Counter
from math import comb

MOD = 10**9 + 7

t = int(input().strip())
for _ in range(t):
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    a.sort(reverse=True)
    max_followers = sum(a[:k])
    count_max = a[:k].count(a[k-1])
    total_count = a.count(a[k-1])
    ways = comb(total_count, count_max)
    print(ways % MOD)
