#include <bits/stdc++.h>
using namespace std;

typedef long long int ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

#define FOR(i, a, b) for(int i = (a); i <= (b); ++i)
#define FORD(i, b, a) for(int i = (b); i >= (a); --i)
#define REP(i, N) FOR(i, 0, (N) - 1)
#define REPD(i, N) FORD(i, (N) - 1, 0)

int n, k;

int solve()
{
    FOR(t, 1, 1000) {
        int s = t / k;
        int r = t % k;
        if (s * r == n) return t;
    }
    return -1;
}

int main()
{
    while (cin >> n >> k) {
        int ans = solve();
        cout << ans << endl;
    }
    return 0;
}
