import java.util.*;
import java.io.*;

public class Main {
    private static ArrayList<String> dates = new ArrayList<String>();
    private static int[] daysInMonths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        String s = in.readLine();
        int max = -1;
        String ans = "";
        for (int i = 0; i < s.length() - 10; i++) {
            String ss = s.substring(i, i + 10);
            if (valid(ss)) {
                dates.add(ss);
            }
        }
        for (String s1 : dates) {
            int count = 0;
            for (String s2 : dates) {
                if (s1.equals(s2)) {
                    count++;
                }
            }
            if (count > max) {
                max = count;
                ans = s1;
            }
        }
        System.out.println(ans);
    }

    private static boolean valid(String s) {
        if (s == null || s.length() != 10) {
            return false;
        }
        for (int i = 0; i < 10; i++) {
            if (i == 2 || i == 5) {
                if (s.charAt(i) != '-') {
                    return false;
                }
            } else {
                if (!Character.isDigit(s.charAt(i))) {
                    return false;
                }
            }
        }
        int day = Integer.parseInt(s.substring(0, 2));
        int month = Integer.parseInt(s.substring(3, 5));
        int year = Integer.parseInt(s.substring(6, 10));
        return year >= 2013 && year <= 2015 && month >= 1 && month <= 12 && day >= 1 && day <= daysInMonths[month - 1];
    }
}
