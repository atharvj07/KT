import java.util.Scanner;

public class DiverseString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        for (int i = 0; i < n; i++) {
            String str = scanner.nextLine();
            if (isDiverse(str)) {
                System.out.println("Yes");
            } else {
                System.out.println("No");
            }
        }
    }

    private static boolean isDiverse(String str) {
        boolean[] seen = new boolean[26]; // Assuming ASCII lowercase letters

        for (int i = 0; i < str.length(); i++) {
            int index = str.charAt(i) - 'a';
            if (seen[index]) {
                return false; // Duplicate letter
            }
            seen[index] = true;

            if (i > 0 && Math.abs(str.charAt(i) - str.charAt(i - 1)) != 1) {
                return false; // Not consecutive letters
            }
        }

        return true;
    }
}
