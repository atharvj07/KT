from sys import stdin

def solve(strings: list):
    pal = [[] for _ in range(len(strings))]
    for i in range(len(strings)):
        l = []
        for j in range(len(strings[i])):
            if strings[i][j] == strings[i][-(j + 1)]:
                l.append(j)
        pal[i] = l
    dp = [[-1 for _ in range(len(strings[i]) + 1)] for _ in range(len(strings) + 1)]
    def dfs(i: int, j: int) -> int:
        if i == len(strings):
            return 0
        if dp[i][j] != -1:
            return dp[i][j]
        ans = dfs(i + 1, j)
        if j < len(pal[i]):
            k = pal[i][j]
            ans = max(ans, 1 + dfs(i + 1, k + 1))
        dp[i][j] = ans
        return ans
    return dfs(0, 0)

for _ in range(int(stdin.readline())):
    strings = []
    for _ in range(int(stdin.readline())):
        strings.append(stdin.readline().strip())
    print(solve(strings))
