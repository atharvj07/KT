for t in range(int(input())):
    n = int(input())
    s = [list(input()) for _ in range(n)]
    for i in range(n):
        for j in range(len(s[i])//2):
            if s[i][j]!=s[i][len(s[i])-j-1]:
                for k in range(n):
                    if k!=i and len(s[k])>len(s[i]):
                        if s[k][len(s[k])-j-1]==s[i][j]:
                            s[k][len(s[k])-j-1],s[i][j]=s[i][j],s[k][len(s[k])-j-1]
                            break
                else:
                    break
    print(len([v for v in s if v==v[::-1]]))
