from math import inf
t=int(input())
for i in range(t):
    n=int(input())
    A=list(map(int,input().split()))
    x=int(input())
    dp=[[-inf]*n for i in range(n)]
    for i in range(n):
        dp[i][i]=1
    for l in range(n-1,-1,-1):
        for r in range(l+1,n):
            if A[l]+A[r]>=x: dp[l][r]=max(dp[l][r],1+dp[l+1][r-1])
            dp[l][r]=max(dp[l][r],max(dp[l+1][r],dp[l][r-1]))
    print(dp[0][n-1])
