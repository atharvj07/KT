def find_min_v(cnt_1, cnt_2, x, y):
    def check(mid):
        total = mid - mid // x - mid // y + mid // (x*y)
        total1 = mid // x - mid // (x*y)
        total2 = mid // y - mid // (x*y)
        return total >= cnt_1 + cnt_2 and total1 >= cnt_1 and total2 >= cnt_2

    left, right = 1, 2 * (cnt_1 + cnt_2)
    while left < right:
        mid = (left + right) // 2
        if check(mid):
            right = mid
        else:
            left = mid + 1
    return left

cnt_1, cnt_2, x, y = map(int, input().split())
print(find_min_v(cnt_1, cnt_2, x, y))
