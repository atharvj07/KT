import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        int cnt = 0;
        int[] b = new int[n];
        while (true) {
            b[0] = a[0];
            b[n - 1] = a[n - 1];
            boolean changed = false;
            for (int i = 1; i < n - 1; i++) {
                int m = (a[i - 1] + a[i] + a[i + 1]) / 2;
                b[i] = m;
                if (m != a[i]) {
                    changed = true;
                }
            }
            if (!changed) {
                System.out.println(cnt);
                for (int i = 0; i < n; i++) {
                    System.out.print(b[i] + " ");
                }
                System.out.println();
                break;
            }
            cnt++;
            a = b;
            if (cnt > n) {
                System.out.println(1);
                return;
            }
        }
    }
}
