#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>

int main() {
    long long t, n;
    std::vector<long long> arr;
    std::cin >> t;
    while (t--) {
        std::cin >> n;
        arr.resize(n);
        for (long long &i : arr) {
            std::cin >> i;
        }
        std::sort(arr.begin(), arr.end());
        std::cout << arr[0] * arr[1] * arr[2] * arr[3] * arr[n - 1] << '\n';
    }
    return 0;
}
