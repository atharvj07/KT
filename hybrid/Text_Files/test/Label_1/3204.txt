#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int main() {
    int k, n1, n2, n3, t1, t2, t3;
    cin >> k >> n1 >> n2 >> n3 >> t1 >> t2 >> t3;
    ll ans = 0, washer = 0, dryer = 0, folder = 0;
    for (int i = 0; i < k; i++) {
        washer = max(washer, ans);
        ans = washer + t1;
        dryer = max(dryer, ans);
        ans = dryer + t2;
        folder = max(folder, ans);
        ans = folder + t3;
    }
    cout << ans << endl;
}
