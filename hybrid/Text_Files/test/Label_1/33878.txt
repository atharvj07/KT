import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    long[] masks = new long[n];
    for (int i = 0; i < n; i++) {
      int k = sc.nextInt();
      long mask = 0;
      for (int j = 0; j < k; j++) {
        mask |= 1L << sc.nextInt();
      }
      masks[i] = mask;
    }
    int q = sc.nextInt();
    for (int i = 0; i < q; i++) {
      int op = sc.nextInt();
      int m = sc.nextInt();
      long mask = masks[m];
      switch (op) {
        case 0:
          System.out.println((mask & (1L << i)) != 0 ? 1 : 0);
          break;
        case 1:
          mask |= m;
          break;
        case 2:
          mask &= ~m;
          break;
        case 3:
          mask ^= m;
          break;
        case 4:
          System.out.println((mask & m) == m ? 1 : 0);
          break;
        case 5:
          System.out.println((mask & m) != 0 ? 1 : 0);
          break;
        case 6:
          System.out.println((mask & m) == 0 ? 1 : 0);
          break;
        case 7:
          long cnt = 0;
          while (m != 0) {
            if ((m & 1) != 0) cnt++;
            m >>= 1;
          }
          System.out.println(cnt);
          break;
        case 8:
          System.out.println(mask);
          break;
      }
    }
  }
}
