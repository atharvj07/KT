n=int(input())
a=list(map(int,input().split()))
mx=0
ansl,ansr=-1,-1
for el in a:
    mx=max(mx,a.count(el))
for i in range(n-1):
    for j in range(i+1,n):
        if mx==a[i:j+1].count(a[i:j+1][0]):
            if j-i+1<ansr-ansl or ansl==ansr==-1:
                ansl,ansr=i,j
print(ansl+1,ansr+1)
