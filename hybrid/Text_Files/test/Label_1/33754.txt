import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int ans = arr[n - 1] - arr[0];
        for (int i = 1; i < n - 1; i++) {
            ans = Math.min(ans, Math.max(arr[i] - arr[0], arr[n - 1] - arr[i]));
        }
        System.out.println(ans);
    }
}
