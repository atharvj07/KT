import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[][] berries = new int[n][2];
        int totalRed = 0, totalBlue = 0;
        for (int i = 0; i < n; i++) {
            berries[i][0] = scanner.nextInt(); // red berries
            berries[i][1] = scanner.nextInt(); // blue berries
            totalRed += berries[i][0];
            totalBlue += berries[i][1];
        }

        int maxBaskets = 0;
        for (int i = 0; i < n; i++) {
            int redBaskets = berries[i][0] / k;
            int blueBaskets = berries[i][1] / k;
            maxBaskets = Math.max(maxBaskets, redBaskets + blueBaskets);
        }

        maxBaskets = Math.max(maxBaskets, totalRed / k); // check total red berries
        maxBaskets = Math.max(maxBaskets, totalBlue / k); // check total blue berries

        System.out.println(maxBaskets);
    }
}
