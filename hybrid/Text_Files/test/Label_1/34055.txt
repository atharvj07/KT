import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        if (!check(a)) {
            System.out.println(-1);
            return;
        }
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            if (a[i] == 0) {
                b[i] = 1;
            } else {
                int mex = 0;
                while (mex < a[i] && mex < n) {
                    boolean found = false;
                    for (int j = 0; j < i; j++) {
                        if (b[j] == mex) {
                            found = true;
                            break;
                        }
                    }
                    if (!found) {
                        b[i] = mex;
                        break;
                    }
                    mex++;
                }
                if (mex >= n) {
                    System.out.println(-1);
                    return;
                }
            }
        }
        for (int v : b) {
            System.out.print(v + " ");
        }
    }

    public static boolean check(int[] a) {
        int mex = 0;
        for (int v : a) {
            if (v > mex) {
                mex++;
            }
        }
        return mex == a.length;
    }
}
