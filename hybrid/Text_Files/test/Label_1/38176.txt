for _ in range(int(input())):
    s,t,p=input(),input(),input()
    st=[0]*26
    pt=[0]*26
    for i in range(26):
        st[ord(s[i])-ord('a')]=1 if i<len(s) else 0
        pt[ord(p[i])-ord('a')]=1 if i<len(p) else 0
    for i in range(26):
        if pt[i]==1 and st[i]==0:
            print('NO')
            break
    else: print('YES')
