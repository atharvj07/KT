import sys
s = input()
t = input()
if len(s) != len(t):
    print('No')
    sys.exit()
v1 = []
c1 = []
v2 = []
c2 = []
for i in range(len(s)):
    if s[i] in 'aeiou':
        v1.append((s[i], i))
    else:
        c1.append((s[i], i))
    if t[i] in 'aeiou':
        v2.append((t[i], i))
    else:
        c2.append((t[i], i))
if len(v1) != len(v2) or len(c1) != len(c2):
    print('No')
    sys.exit()
for i in range(len(v1)):
    if v1[i][0] != v2[i][0]:
        print('No')
        sys.exit()
for i in range(len(c1)):
    if c1[i][0] != c2[i][0]:
        print('No')
        sys.exit()
print('Yes')
