import java.util.Scanner;

class Main {
    static int[] a;
    static int n;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        a = new int[n + 1];
        for (int i = 0; i < n; i++)
            a[sc.nextInt()]++;
        boolean flag = true;
        for (int i = 0; i < n + 1; i++) {
            if (a[i] > 1) {
                a[i]--;
                a[n]++;
                flag = false;
                break;
            }
        }
        if (flag) {
            System.out.println(-1);
            return;
        }
        for (int i = 0; i <= n; i++) {
            if (a[i] > 0)
                System.out.print(i + " ");
        }
        System.out.println();
    }
}
