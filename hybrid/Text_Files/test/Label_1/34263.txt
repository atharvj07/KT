import java.util.*;

class Main {

  static long transform(long x) {
    while (x % 2 == 0) x >>= 1;
    return x - 1;
  }

  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    long x = in.nextLong();
    long y = transform(x);
    long z = 0;
    if (y > 0) {
      while (y >= (1 << z)) z++;
      z--;
    }
    long[] arr = new long[(int) (z + 2) / 2];
    int index = 0;
    int count = 0;
    while (z > 0) {
      if (z <= (1 << count)) {
        arr[index++] = count;
        z -= 1 << count;
      }
      count++;
    }
    System.out.println(index * 2);
    for (int i = 0; i < index; i++) {
      System.out.printf("%d%s", arr[i], i + 1 == index ? "\n" : " ");
    }
    for (int i = 0; i < index * 2; i++) {
      x ^= ((long) 1 << arr[i % index]);
      if ((i + 1) % 2 == 0) x++;
    }
  }
}
