n,m,k=map(int,input().split())
a=[0]+list(map(int,input().split()))
b=[0]+list(map(int,input().split()))
pa=[0]*40001
pb=[0]*40001
for i in range(n):
    pa[i+1]=pa[i]
    if a[i+1]:
        pa[i+1]+=1
for i in range(m):
    pb[i+1]=pb[i]
    if b[i+1]:
        pb[i+1]+=1
ans=0
for i in range(1,n+1):
    for j in range(1,m+1):
        if pa[i]+pb[j]-pa[i-k]-pb[j-k]==k:
            ans+=1
print(ans)
