import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int h1 = sc.nextInt();
        int m1 = sc.nextInt();
        int h2 = sc.nextInt();
        int m2 = sc.nextInt();
        int cnt = 0;
        for (int i = h1; i <= h2; i++) {
            for (int j = m1; j <= m2; j++) {
                if (i == h1 && j == m1) continue;
                if (i == h2 && j == m2) continue;
                String s1 = String.format("%0" + (int) Math.log10(h - 1) + "d", i);
                String s2 = String.format("%0" + (int) Math.log10(m - 1) + "d", j);
                String s3 = String.format("%0" + (int) Math.log10(h - 1) + "d", i - 1);
                String s4 = String.format("%0" + (int) Math.log10(m - 1) + "d", j - 1);
                int diff = 0;
                for (int l = 0; l < s1.length(); l++) {
                    if (s1.charAt(l) != s3.charAt(l)) diff++;
                }
                for (int l = 0; l < s2.length(); l++) {
                    if (s2.charAt(l) != s4.charAt(l)) diff++;
                }
                if (diff >= k) cnt++;
            }
        }
        System.out.println(cnt);
    }
}
