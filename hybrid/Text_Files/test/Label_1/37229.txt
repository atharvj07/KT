n = int(input().strip())
row = input().strip()

count = 0
groups = []

for i in range(n):
    if row[i] == 'B':
        count += 1
        if i == n-1 or row[i+1] == 'W':
            groups.append(count)
            count = 0
            
print(len(groups))
if groups:
    print(" ".join(map(str, groups)))
