#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, h;
    cin >> n >> h;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int ans = 0, t = 0;
    for (int i = 0; i < n; i++) {
        ans += min(a[i], h) + ((a[i] > h) ? 1 : 0);
        t += max(a[i], h);
    }
    cout << max(ans, t) << endl;
    return 0;
}
