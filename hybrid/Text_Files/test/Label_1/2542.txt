#include<bits/stdc++.h>
#define int long long
using namespace std;
int32_t main(){
    int n; cin>>n; 
    int a[n+1]; 
    for(int i=1;i<=n;i++) { cin>>a[i];}
    int c=0;
    for(int i=1;i<=n;i++){
        if(a[i]==i) c++;
    }
    if(c>=n-1) { cout<<a[n]; return 0; }
    int ans=0;
    for(int i=1;i<=n;i++){
        int x=i+1;
        while(x<=n){
            if(a[x]==(i+1)){
                x++;
                continue;
            }
            ans=max(ans,x-i-1);
            break;
        }
    }
    cout<<ans<<endl;
}
