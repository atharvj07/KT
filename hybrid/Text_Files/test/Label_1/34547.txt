import java.util.ArrayList;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        // sort the array
        Arrays.sort(a);

        // find the maximum balanced circle
        int max = 0;
        ArrayList<Integer> result = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            int j = i + 1;
            while (j < n && a[j] - a[i] <= 1) {
                j++;
            }
            if (j - i > max) {
                max = j - i;
                result.clear();
                for (int k = i; k < j; k++) {
                    result.add(a[k]);
                }
            }
        }

        // output the result
        System.out.println(max);
        for (int x : result) {
            System.out.print(x + " ");
        }
        System.out.println();
    }
}
