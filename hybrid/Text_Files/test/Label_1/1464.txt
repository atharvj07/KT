#include <bits/stdc++.h>
using namespace std;
int a[100005], c[100005];
int main() {
  int n, k, p = 0, t = 0, ans = 0;
  cin >> n >> k;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    if (a[i] == a[i - 1]) {
      p = i;
    } else {
      t++;
      c[t] = i - p;
      ans = max(ans, c[t]);
    }
  }
  cout << ans << endl;
  return 0;
}
