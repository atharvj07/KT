import sys

def check(n, b):
    a = []
    for i in range(n):
        if i == 0:
            a.append((b[i] + 1) // 2)
        else:
            a.append((b[i] - b[i - 1] + 1) // 2)
    return a

def main():
    t = int(input())
    for _ in range(t):
        n = int(input())
        b = list(map(int, input().split()))
        if n == 1:
            if b[0] % 2 == 0:
                print("YES")
                print(b[0] // 2)
            else:
                print("NO")
        else:
            a = check(n, b)
            if all(ai > 0 for ai in a):
                print("YES")
                print(*a)
            else:
                print("NO")

main()
