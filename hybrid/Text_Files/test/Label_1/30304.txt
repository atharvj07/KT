import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int l = sc.nextInt();
            if (l == 0) break;
            int[] mn = new int[13];
            for (int i = 1; i <= 12; i++) {
                mn[i] = sc.nextInt() - sc.nextInt();
            }
            int sum = 0;
            int i = 1;
            for (; i <= 12; i++) {
                sum += mn[i];
                if (sum >= l) break;
            }
            System.out.println(i <= 12 ? i : "NA");
        }
    }
}
