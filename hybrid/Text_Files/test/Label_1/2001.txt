#include <iostream>
#include <string>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string a, b;
    cin >> a >> b;
    int one_a = 0, one_b = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] == '1') one_a++;
      if (b[i] == '1') one_b++;
    }
    if (one_a == one_b)
      cout << "YES\n";
    else
      cout << "NO\n";
  }
  return 0;
}
