import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n];
            int odd = 0;
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextInt();
                if (a[i] % 2 == 1) odd++;
            }
            if (odd > 1 || (odd == 1 && n % 2 == 0)) System.out.println("NO");
            else System.out.println("YES");
        }
    }
}
