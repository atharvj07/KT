#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> ids(n);
  for (int i = 0; i < n; ++i) std::cin >> ids[i];
  std::vector<int> stack;
  for (int i = 0; i < n; ++i) {
    int id = ids[i];
    int idx = -1;
    for (int j = 0; j < stack.size(); ++j) if (stack[j] == id) idx = j;
    if (idx == -1) {
      if (stack.size() == k) stack.pop_back();
      stack.insert(stack.begin(), id);
    } else {
      for (int j = idx; j > 0; --j) stack[j] = stack[j - 1];
      stack[0] = id;
    }
  }
  std::cout << stack.size() << std::endl;
  for (int id : stack) std::cout << id << ' ';
  std::cout << std::endl;
}
