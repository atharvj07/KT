import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

class Main {
    private static long set[] = new long[10];
    private static long state = 0;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        for (int i = 0; i < n; i++) {
            int k = sc.nextInt();
            Set<Integer> s = new HashSet<>();
            for (int j = 0; j < k; j++) {
                s.add(sc.nextInt());
            }
            for (int j : s) {
                set[i] |= 1L << (j - 1);
            }
        }
        int q = sc.nextInt();
        for (int i = 0; i < q; i++) {
            int op = sc.nextInt();
            long m = sc.nextInt();
            op(op, m);
        }
    }

    private static void op(int op, long m) {
        switch (op) {
            case 0:
                System.out.println(test(m));
                break;
            case 1:
                state |= set[(int) m];
                break;
            case 2:
                state &= ~set[(int) m];
                break;
            case 3:
                state ^= set[(int) m];
                break;
            case 4:
                System.out.println(all(set[(int) m]));
                break;
            case 5:
                System.out.println(any(set[(int) m]));
                break;
            case 6:
                System.out.println(none(set[(int) m]));
                break;
            case 7:
                System.out.println(count(set[(int) m]));
                break;
            case 8:
                System.out.println(val(set[(int) m]));
                break;
        }
    }

    private static int test(long m) {
        return (state & m) != 0 ? 1 : 0;
    }

    private static int all(long m) {
        return (state & m) == m ? 1 : 0;
    }

    private static int any(long m) {
        return (state & m) != 0 ? 1 : 0;
    }

    private static int none(long m) {
        return (state & m) == 0 ? 1 : 0;
    }

    private static int count(long m) {
        int c = 0;
        for (int i = 0; i < 64; i++) {
            if ((m & (1L << i)) != 0) {
                c += test(1L << i);
            }
        }
        return c;
    }

    private static long val(long m) {
        long v = 0;
        for (int i = 0; i < 64; i++) {
            if ((m & (1L << i)) != 0) {
                v |= 1L << test(1L << i);
            }
        }
        return v;
    }
}
