import java.util.*;

class Main {

    static void sieve(boolean[] s, int n) {
        for (int p = 2; p * p < n; p++) {
            if (!s[p]) {
                for (int i = p * p; i < n; i += p) {
                    s[i] = true;
                }
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = sc.nextInt();
        }

        boolean[] composites = new boolean[100001];
        sieve(composites, 100001);

        for (int i = 2; i <= n; i++) {
            boolean f = false;
            for (int j = 1; j < i; j++) {
                if (a[i] > a[j] && !composites[gcd(a[i], a[j])]) {
                    f = true;
                    break;
                }
            }
            if (!f) {
                System.out.println("0");
                return;
            }
        }

        System.out.println("1");
    }

    static int gcd(int a, int b) {
        if (a == 0) {
            return b;
        }
        return gcd(b % a, a);
    }
}
