import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int k = scanner.nextInt();
        int n = scanner.nextInt();
        int[] ax = new int[k];
        int[] ay = new int[k];
        int[] mx = new int[n];
        int[] my = new int[n];
        for (int i = 0; i < k; i++) {
            ax[i] = scanner.nextInt();
            ay[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            mx[i] = scanner.nextInt();
            my[i] = scanner.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < k; j++) {
                boolean flag = true;
                for (int l = 0; l < n; l++) {
                    if (l == i) {
                        continue;
                    }
                    if (slope(ax[j], ay[j], mx[i], my[i]) == slope(ax[j], ay[j], mx[l], my[l])) {
                        flag = false;
                        break;
                    }
                }
                if (flag) {
                    ans++;
                    break;
                }
            }
        }
        System.out.println(ans);
    }

    private static double slope(int x1, int y1, int x2, int y2) {
        if (x2 - x1 == 0) {
            return Double.POSITIVE_INFINITY;
        }
        return (double) (y2 - y1) / (x2 - x1);
    }
}
