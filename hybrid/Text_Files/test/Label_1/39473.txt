n, m = map(int, input().split())
b, g = [0] * n, [0] * m
k = 0
while n > 0 or m > 0:
    i, j = -1, -1
    for t in range(n):
        if not b[t] or not g[j]:
            i, j = t, j
            break
    for t in range(m):
        if not b[i] or not g[t]:
            i, j = i, t
            break
    if i != -1:
        k, b[i], g[j] = k+1, 1, 1
        print(i+1, j+1)
    else:
        break
