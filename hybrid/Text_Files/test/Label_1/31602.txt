import java.util.*;
import java.io.*;

public class Main {
    static long[][] f = new long[101][11];
    static long[][] p = new long[101][11];

    public static void main(String[] argv) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        for (int i = 1; i <= n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int j = 1; j <= 5; j++) {
                f[i][j] = Long.parseLong(st.nextToken());
                f[i][j + 5] = Long.parseLong(st.nextToken());
            }
        }

        for (int i = 1; i <= n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            for (int j = 0; j <= 10; j++) {
                p[i][j] = Long.parseLong(st.nextToken());
            }
        }

        long ans = 0;

        for (int i = 1; i <= n; i++) {
            long max = 0;
            for (int j = 1; j <= 10; j++) {
                max = Math.max(max, p[i][j]);
            }
            ans += max;
        }

        for (int i = 1; i <= 5; i++) {
            long max = 0;
            for (int j = 1; j <= n; j++) {
                long cur = 0;
                for (int k = 1; k <= 10; k++) {
                    if (f[j][k] == 0) continue;
                    if (k == i || k == i + 5) cur += p[j][k];
                }
                max = Math.max(max, cur);
            }
            ans += max;
        }

        System.out.println(ans);
    }
}
