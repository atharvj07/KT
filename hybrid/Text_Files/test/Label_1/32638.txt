import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Main {
    private static int[] maxValues;
    private static int[] minValues;
    private static List<int[]> creatures;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] nks = reader.readLine().split(" ");
        int n = Integer.parseInt(nks[0]);
        int k = Integer.parseInt(nks[1]);
        int q = Integer.parseInt(nks[2]);

        creatures = new ArrayList<>();
        maxValues = new int[n];
        minValues = new int[n];

        for (int i = 0; i < k; i++) {
            String[] numbers = reader.readLine().split(" ");
            int[] creature = new int[n];
            for (int j = 0; j < n; j++) {
                creature[j] = Integer.parseInt(numbers[j]);
                maxValues[j] = Math.max(creature[j], maxValues[j]);
                minValues[j] = Math.min(creature[j], minValues[j]);
            }
            creatures.add(creature);
        }

        for (int i = 0; i < q; i++) {
            String[] query = reader.readLine().split(" ");
            if (Integer.parseInt(query[0]) == 3) {
                int creatureIndex = Integer.parseInt(query[1]) - 1;
                int characteristicIndex = Integer.parseInt(query[2]) - 1;

                System.out.println(creatures.get(creatureIndex)[characteristicIndex]);
            } else {
                int[] newCreature = new int[n];
                for (int j = 0; j < n; j++) {
                    if (Integer.parseInt(query[0]) == 1) {
                        newCreature[j] = Math.max(creatures.get(Integer.parseInt(query[1]) - 1)[j],
                                creatures.get(Integer.parseInt(query[2]) - 1)[j]);
                    } else {
                        newCreature[j] = Math.min(creatures.get(Integer.parseInt(query[1]) - 1)[j],
                                creatures.get(Integer.parseInt(query[2]) - 1)[j]);
                    }
                }

                for (int j = 0; j < n; j++) {
                    maxValues[j] = Math.max(newCreature[j], maxValues[j]);
                    minValues[j] = Math.min(newCreature[j], minValues[j]);
                }

                creatures.add(newCreature);
            }
        }
    }
}
