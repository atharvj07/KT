import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Comparator;

class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] s = br.readLine().split(" ");
        int n = Integer.parseInt(s[0]);
        int d = Integer.parseInt(s[1]);
        int S = Integer.parseInt(s[2]);

        Robot[] robots = new Robot[n];
        for (int i = 0; i < n; i++) {
            s = br.readLine().split(" ");
            int ci = Integer.parseInt(s[0]);
            int fi = Integer.parseInt(s[1]);
            int li = Integer.parseInt(s[2]);
            robots[i] = new Robot(ci, fi, li);
        }

        Arrays.sort(robots, Comparator.comparingInt(Robot::getLi).reversed());

        int[][] dp = new int[n + 1][S + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= S; j++) {
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j]);
                if (j + robots[i - 1].fi <= S) {
                    dp[i][j + robots[i - 1].fi] = Math.max(dp[i - 1][j] + robots[i - 1].ci + 1, dp[i][j + robots[i - 1].fi]);
                }
            }
        }

        int maxRobots = 0;
        int minFuel = S;
        for (int i = 0; i <= S; i++) {
            if (dp[n][i] >= 1 && dp[n][i] * d <= S) {
                if (dp[n][i] > maxRobots || (dp[n][i] == maxRobots && i < minFuel)) {
                    maxRobots = dp[n][i];
                    minFuel = i;
                }
            }
        }

        System.out.println(maxRobots + " " + minFuel);
    }

    private static class Robot {
        private int ci;
        private int fi;
        private int li;

        public Robot(int ci, int fi, int li) {
            this.ci = ci;
            this.fi = fi;
            this.li = li;
        }

        public int getLi() {
            return li;
        }
    }
}
