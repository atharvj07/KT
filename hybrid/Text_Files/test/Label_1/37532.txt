import sys
input = sys.stdin.readline

for _ in range(int(input())):
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    # Sort applications by their memory usage
    apps = [(b[i], a[i]) for i in range(n)]
    apps.sort(reverse = True)

    # Calculate the minimum number of convenience points that Polycarp will lose
    lose = 0
    freed = 0
    for p, mem in apps:
        if freed >= m:
            break
        freed += mem
        lose += p
    
    if freed < m:
        print(-1)
    else:
        print(lose)
