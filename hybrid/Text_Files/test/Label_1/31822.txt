import java.util.*;
import java.io.*;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();

        int[] lengths = new int[n];
        for (int i = 0; i < n; i++) {
            lengths[i] = scanner.nextInt();
        }

        char[] terrain = scanner.next().toCharArray();

        long minTime = Long.MAX_VALUE;

        for (int swimming = 0; swimming <= n; swimming++) { 
            long stamina = 0;
            long time = 0;
            for (int i = 0; i < swimming; i++) { // first swim
                stamina += lengths[i];
                time += lengths[i] * 3;
            }
            int flying = n - swimming;
            for (int i = swimming; i < swimming + flying; i++) { // then fly
                if (stamina >= 2) {
                    stamina -= 2;
                    time += lengths[i];
                }
                else { // first walk, then fly
                    time += lengths[i] * 5;
                    stamina += lengths[i];
                    stamina -= 2;
                    time += lengths[i];
                }
            }
            minTime = Math.min(minTime, time);
        }

        System.out.println(minTime);
    }
}
