t = int(input())
for _ in range(t):
    a = input()
    evens = ''.join([c for c in a if int(c) % 2 == 0])
    odds = ''.join([c for c in a if int(c) % 2 == 1])
    res = []
    i, j = 0, 0
    while i < len(evens) and j < len(odds):
        if evens[i] < odds[j]:
            res.append(evens[i])
            i += 1
        else:
            res.append(odds[j])
            j += 1
    while i < len(evens):
        res.append(evens[i])
        i += 1
    while j < len(odds):
        res.append(odds[j])
        j += 1
    print(''.join(res))
