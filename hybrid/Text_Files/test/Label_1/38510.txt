from collections import deque
from sys import stdin

input = stdin.readline

t = int(input())
for _ in range(t):
    n = int(input())
    lx = ly = -1
    for _ in range(n):
        x, y = list(map(int, input().split()))
        lx = max(x, lx)
        ly = max(y, ly)

    if lx + ly != n:
        print("NO")
        continue

    robot = deque([(0, 0)])
    path = ""
    vis = [[False] * (lx + 10) for _ in range(ly + 10)]
    while robot:
        x, y = robot.popleft()
        if vis[y][x]:
            continue
        if (x, y) == (lx, ly):
            break
        vis[y][x] = True
        robot.append((x + 1, y))
        robot.append((x, y + 1))
        if x == lx:
            path += 'U'
        else:
            path += 'R'

    print("YES")
    print(path)
