def can_assign_sender(users, messages, last_sender, index):
    if index == len(messages):
        return True
    if messages[index][0] != '?':
        if messages[index][0] == last_sender or any(user in messages[index][1] for user in users):
            return False
        return can_assign_sender(users, messages, messages[index][0], index + 1)
    for user in users:
        if user != last_sender and not any(user in msg for msg in messages[index][1]):
            messages[index][0] = user
            if can_assign_sender(users, messages, user, index + 1):
                return True
            messages[index][0] = '?'
    return False

def solve_chats():
    t = int(input().strip())
    for _ in range(t):
        n = int(input().strip())
        users = input().strip().split()
        m = int(input().strip())
        messages = [input().strip().split(':', 1) for _ in range(m)]
        if can_assign_sender(users, messages, "", 0):
            for sender, text in messages:
                print(f"{sender}:{text}")
        else:
            print("Impossible")

solve_chats()
