n = int(input())
b = list(map(int, input().split()))

# Define potential differences based on the first two elements and their possible modifications
possible_differences = set([(b[1] + d2) - (b[0] + d1) for d1 in [-1, 0, 1] for d2 in [-1, 0, 1]])

min_changes = float('inf')
for d in possible_differences:
    changes = sum(b[i] + d*i != b[0] for i in range(n))
    for d1 in [-1, 0, 1]:
        for d2 in [-1, 0, 1]:
            # Adjust the first two elements based on d1 and d2, then check the sequence
            adjusted_changes = sum((b[i] + d*i != b[0] + d1 + d*(i-1) for i in range(1, n)), d1 != 0) + (d2 != 0)
            if b[1] + d2 == b[0] + d1 + d:
                min_changes = min(min_changes, adjusted_changes)

# If it's impossible to form an arithmetic sequence, print -1
print(-1 if min_changes == float('inf') else min_changes)
