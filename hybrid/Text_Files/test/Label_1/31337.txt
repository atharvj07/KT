import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long h = sc.nextLong(), m = sc.nextLong(), k = sc.nextLong();
        long h1 = sc.nextLong(), m1 = sc.nextLong();
        long h2 = sc.nextLong(), m2 = sc.nextLong();
        int ans = 0;
        if (h1 != h2 || m1 != m2) {
            for (long i = h1; i <= h2; i++) {
                int c1 = (int) (Math.log10(i) + 1);
                for (long j = m1; j <= m2; j++) {
                    int c2 = (int) (Math.log10(j) + 1);
                    if (Math.max(c1, c2) - Math.min(c1, c2) >= k) {
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
