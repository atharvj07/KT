import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int k = in.nextInt();
        int a = in.nextInt();
        int b = in.nextInt();
        int q = in.nextInt();
        int[] days = new int[n + 1];
        for (int i = 0; i < q; i++) {
            int type = in.nextInt();
            if (type == 1) {
                int d = in.nextInt();
                int c = in.nextInt();
                days[d] += c;
            } else {
                int p = in.nextInt();
                long ans = 0;
                for (int j = Math.max(1, p - k + 1); j <= p; j++) {
                    if (j + k - 1 <= n) {
                        ans += Math.min(a, days[j]);
                    }
                }
                ans += b * (p + k - 1 - Math.max(1, p - k + 1) + 1);
                System.out.println(ans);
            }
        }
    }
}
