#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int maxn = 100000;
vector<int> vec[maxn];
int dep[maxn];
int n;
int a;
char arr[maxn];
queue<int> q;

int main() {
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
        if (arr[i] == 'P') {
            q.push(i);
            vec[i].push_back(i);
            dep[i] = 0;
        } else if (arr[i] == '*')
            a++;
    }
    int ans = 0;
    while (!q.empty()) {
        int f = q.front();
        q.pop();
        if (arr[f] == '*') {
            a--;
            ans = dep[f];
            if (a == 0) {
                cout << ans << endl;
                return 0;
            }
        }
        if (f + 1 < n && vec[f + 1].empty()) {
            q.push(f + 1);
            vec[f + 1].push_back(f);
            dep[f + 1] = dep[f] + 1;
        }
        if (f - 1 >= 0 && vec[f - 1].empty()) {
            q.push(f - 1);
            vec[f - 1].push_back(f);
            dep[f - 1] = dep[f] + 1;
        }
    }
    cout << -1 << endl;
    return 0;
}
