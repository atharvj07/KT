#include<bits/stdc++.h>
#include<iostream>
using namespace std;
bool vis[110];
int main() {
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> v[m + 1], ans(n + 1);
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      int x;
      cin >> x;
      v[j].push_back(x);
      if (vis[x]) { ans[i] = j; }
    }
    for (int j = 1; j <= m; j++) {
      vis[v[j][i - 1]] = 1;
      if (v[j][i - 1] != 0) { cout << i << " " << j << endl; }
    }
  }
  for (int i = 1; i <= n; i++) {
    cout << ans[i] << endl;
  }
  return 0;
}
