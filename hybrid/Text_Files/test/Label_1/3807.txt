#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<int> a(n);
  std::vector<std::pair<int, int>> b(m);
  for (auto &x : a) std::cin >> x;
  for (auto &[l, r] : b) std::cin >> l >> r, l--, r--;
  std::vector<int> cnt(n);
  for (auto [l, r] : b)
    for (int i = l; i <= r; i++) cnt[i]++;
  int ans = 0;
  for (int i = 0; i < n; i++) ans += std::max(0, a[i] * cnt[i]);
  std::cout << ans << std::endl;
}
