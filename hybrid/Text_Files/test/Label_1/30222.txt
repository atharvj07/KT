import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class Main {

    private static final int INF = Integer.MAX_VALUE / 2;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();

        List<int[]> distances = new ArrayList<>();
        for (int i = 0; i < k; i++) {
            int[] d = new int[n];
            for (int j = 0; j < n; j++) {
                d[j] = scanner.nextInt();
            }
            distances.add(d);
        }

        int[][] dp = new int[n][1 << k];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], INF);
        }

        dp[0][0] = 0;
        for (int mask = 0; mask < (1 << k); mask++) {
            for (int i = 0; i < n; i++) {
                if (dp[i][mask] == INF) {
                    continue;
                }

                for (int j = 0; j < n; j++) {
                    if (i == j) {
                        continue;
                    }

                    int d = INF;
                    for (int l = 0; l < k; l++) {
                        if ((mask & (1 << l)) != 0) {
                            d = Math.min(d, distances.get(l)[i] + distances.get(l)[j]);
                        }
                    }

                    if (d < INF) {
                        dp[j][mask | (1 << k)] = Math.min(dp[j][mask | (1 << k)], dp[i][mask] + d);
                    }
                }
            }
        }

        int minMask = -1;
        int minDistance = INF;
        for (int mask = 0; mask < (1 << k); mask++) {
            if (dp[0][mask] < minDistance) {
                minDistance = dp[0][mask];
                minMask = mask;
            }
        }

        if (minMask == -1) {
            System.out.println(-1);
            return;
        }

        int[] path = new int[n];
        int mask = minMask;
        int vertex = 0;
        for (int i = n - 1; i > 0; i--) {
            for (int j = 0; j < n; j++) {
                if (i == j) {
                    continue;
                }

                int d = INF;
                for (int l = 0; l < k; l++) {
                    if ((mask & (1 << l)) != 0) {
                        d = Math.min(d, distances.get(l)[vertex] + distances.get(l)[j]);
                    }
                }

                if (d < INF && dp[j][mask] == dp[vertex][mask] + d) {
                    path[i] = j;
                    vertex = j;
                    mask &= ~(1 << k);
                    break;
                }
            }
        }

        for (int i = 1; i < n; i++) {
            System.out.println(path[i] + 1 + " " + (i + 1));
        }
    }
}
