t = int(input())
for _ in range(t):
    s = input()
    i = 0
    to_remove = []
    while i < len(s) - 2:
        if s[i:i+3] == "one":
            to_remove.append(i+2)
            i += 3
        elif s[i:i+3] == "two":
            if i < len(s) - 4 and s[i:i+5] == "twone":
                to_remove.append(i+3)
                i += 5
            else:
                to_remove.append(i+2)
                i += 3
        else:
            i += 1
    print(len(to_remove))
    print(' '.join(map(str, to_remove)))
