import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
            b[i] = input.nextInt();
        }
        Arrays.sort(b);
        int ans = -1;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (ans == -1 && cnt == n) {
                ans = b[i];
            }
            if (b[i] >= a[i]) {
                cnt++;
            }
        }
        System.out.println(ans);
    }
}
