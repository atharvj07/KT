import java.util.*;
import java.math.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long n = sc.nextLong();
        long k = sc.nextLong();
        long[] a = new long[(int)n];
        for(long i=0;i<n;i++) a[(int)i] = sc.nextLong();
        Arrays.sort(a);
        long m = (long)Math.ceil((double)n/k);
        long ans = 0;
        for(long i=0;i<k;i++){
            long cnt = i;
            long sum = 0;
            while(cnt<n){
                sum += a[(int)cnt];
                cnt += k;
            }
            if(i==0) ans = sum;
            else ans = Math.min(ans,sum);
        }
        System.out.println(ans);
    }
}
