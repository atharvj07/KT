n = int(input())
heroes = list(map(int, input().split()))
m = int(input())

heroes.sort()

for _ in range(m):
    x, y = map(int, input().split())
    left = 0
    right = n - 1
    min_coins = float('inf')

    while left <= right:
        mid = (left + right) // 2
        if heroes[mid] >= x:
            right = mid - 1
        else:
            left = mid + 1

    if left < n:
        min_coins = min(min_coins, max(0, y - sum(heroes) + heroes[left]))

    if left > 0:
        min_coins = min(min_coins, x - heroes[left - 1] + max(0, y - sum(heroes) + heroes[left - 1]))

    print(min_coins)
