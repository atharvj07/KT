#include <bits/stdc++.h>
using namespace std;
int arr[200005], br[200005];
bool oc[200005];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, i, j, k;
    cin >> n;
    multiset<int> s;
    map<int, int> mp, mp2;
    for (i = 1; i <= n; i++) {
        cin >> arr[i];
        s.insert(arr[i]);
    }
    for (auto p : s) {
        mp.insert({p, mp.size() + 1});
        mp2.insert({mp2.size() + 1, p});
    }
    string str;
    cin >> str;
    for (i = 0; i < str.length(); i++) {
        // introvert
        if (str[i] == '0') {
            auto it = s.begin();
            cout << mp[*it] << " ";
            s.erase(it);
            oc[mp[*it]] = true;
        }
        // extrovert
        else {
            auto it = mp2.upper_bound(oc[0]);
            auto p = prev(it);
            cout << p->second << " ";
            oc[p->first] = true;
        }
    }
    return 0;
}
