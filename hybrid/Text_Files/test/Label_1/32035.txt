import java.util.*;
import java.io.*;

public class Main {
    private static final int INF = Integer.MAX_VALUE;
    private static int[][] d;
    private static int n, m;
    private static List<List<Edge>> graph;
    private static void dijkstra(int r) {
        d = new int[n+1][2];
        for (int i = 2; i <= n; i++) for (int j = 0; j < 2; j++) d[i][j] = INF;
        PriorityQueue<Node> pq = new PriorityQueue<>((a, b) -> a.cost-b.cost);
        pq.add(new Node(r, 0, 0));
        while (!pq.isEmpty()) {
            Node node = pq.poll();
            for (Edge edge: graph.get(node.u)) {
                for (int j = 0; j < 2; j++) {
                    if (j == 0 && d[node.u][j] + edge.cost >= d[edge.v][j]) continue;
                    if (j == 1 && d[node.u][j] + edge.time >= d[edge.v][j]) continue;
                    d[edge.v][j] = d[node.u][j] + (j == 0 ? edge.cost : edge.time);
                    pq.add(new Node(edge.v, d[edge.v][j], j));
                }
            }
        }
    }
    private static class Node {
        int u, cost;
        int type;
        Node(int u, int cost, int type) {
            this.u = u;
            this.cost = cost;
            this.type = type;
        }
    }
    private static class Edge {
        int v, cost, time;
        Edge(int v, int cost, int time) {
            this.v = v;
            this.cost = cost;
            this.time = time;
        }
    }
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        while (true) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            n = Integer.parseInt(st.nextToken());
            m = Integer.parseInt(st.nextToken());
            if (n == 0 && m == 0) break;
            graph = new ArrayList<>();
            for (int i = 0; i <= n; i++) graph.add(new ArrayList<>());
            for (int i = 0; i < n; i++) {
                st = new StringTokenizer(br.readLine());
                int u = Integer.parseInt(st.nextToken());
                int v = Integer.parseInt(st.nextToken());
                int cost = Integer.parseInt(st.nextToken());
                int time = Integer.parseInt(st.nextToken());
                graph.get(u).add(new Edge(v, cost, time));
                graph.get(v).add(new Edge(u, cost, time));
            }
            st = new StringTokenizer(br.readLine());
            int q = Integer.parseInt(st.nextToken());
            for (int i = 0; i < q; i++) {
                st = new StringTokenizer(br.readLine());
                int s = Integer.parseInt(st.nextToken());
                int e = Integer.parseInt(st.nextToken());
                int r = Integer.parseInt(st.nextToken());
                dijkstra(s);
                System.out.println(d[e][r]);
            }
        }
    }
}
