#include<bits/stdc++.h>
using namespace std;
int a[100010], n, mark[100010];
int main() { cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int ans = 0, cur = 1;
  while (true) {
    int t = 0;
    while (cur <= n && mark[cur]) cur++;
    if (cur > n) break;
    mark[cur] = 1;
    while (cur <= n && !mark[a[cur]]) mark[a[cur++]] = 1, t++;
    ans += t > 0;
  }
  cout << ans << endl;
}
