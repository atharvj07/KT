import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] m = new int[5];
        int[] w = new int[5];
        for (int i = 0; i < 5; i++) {
            m[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        int hs = sc.nextInt();
        int hu = sc.nextInt();
        int score = 0;
        for (int i = 0; i < 5; i++) {
            int x = 500 * (i + 1);
            score += x - x * w[i] / 250 - m[i];
        }
        score += hs * 100 - hu * 50;
        System.out.println(score);
    }
}
