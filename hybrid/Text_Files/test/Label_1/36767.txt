import sys
input = sys.stdin.readline
mod = 998244353

n, k = map(int,input().split())
sa = list(map(int, input().split()))

fp = 0
fm = [0]
tot = 0
for i in range(1,n):
    if sa[i] > sa[i-1]:
        fp += 1
        tot += fp*k
    elif sa[i] == sa[i-1]:
        fp += 1
    else:
        fp = 0
        fm.append(tot)
for i in range(len(fm)-1,0,-1):
    fm[i-1] = (fm[i-1]+fm[i])%mod
print(pow(k,n-1, mod) - sum(fm)%mod)
