#include<bits/stdc++.h>
using namespace std;
int n,k;
int main()
{
    cin>>n>>k;
   int ans=n*(n+1)/2;
   if(k!=n) ans-=n;
   cout<<ans<<'\n';
   vector<vector<int>>v(n);
   for(int i=0;i<n;i++)
   {
       for(int j=0;j<n;j++)
       {
           if(k-1==j && n!=k && i==n-1)
            v[i].push_back(ans);
           else
           {
               if(j+1+i<=n)
               v[i].push_back(j+1+i);
               else
               {
                   v[i].push_back(j+1+i-n);
               }
           }
       }
   }
   for(int i=0;i<n;i++)
   {
       for(int j=0;j<n;j++)
       {
           cout<<v[i][j]<<' ';
       }
       cout<<'\n';
   }
   return 0;
}
