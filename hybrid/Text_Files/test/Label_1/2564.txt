#include <iostream>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; i++) cin >> a[i];
    int neg = 0;
    int pos = 0;
    int zero = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] < 0)
        neg++;
      else if (a[i] > 0)
        pos++;
      else
        zero++;
    }
    cout << pos + neg + zero - (pos > 0 && neg > 0) << endl;
  }
  return 0;
}
