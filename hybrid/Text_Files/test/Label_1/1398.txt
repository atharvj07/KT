#include <bits/stdc++.h>
int x, y, ans = 0, c0 = 0;
char s[100001];
int main() {
  scanf("%s%d%d", s, &x, &y);
  int len = strlen(s);
  for (int i = 0; i < len; i++) {
    if (s[i] == '0') c0++;
    if (s[i] == '?') ans += std::min(x * c0, y * (len - i - c0));
    if (s[i] == '1') c0--;
  }
  printf("%d", ans);
}
