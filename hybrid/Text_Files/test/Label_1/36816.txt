import sys

n = int(input())
a = []
for i in range(n):
    l, r = map(int, sys.stdin.readline().split())
    a.append([l, r])

ans = [-1, -1]
for i in range(n):
    for j in range(n):
        if i != j and a[i][0] >= a[j][0] and a[i][1] <= a[j][1]:
            ans = [i + 1, j + 1]
            break
    if ans != [-1, -1]:
        break

print(*ans)
