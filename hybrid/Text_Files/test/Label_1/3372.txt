#include<bits/stdc++.h>
using namespace std;
int ans[5000][2];
int32_t main(){
	int t, n;
	cin >> t;
	while(t--){
		cin >> n;
		vector<int> v(n);
		map<int, int> mp;
		for(int i = 0; i<n; i++){
			cin >> v[i];
			mp[v[i]]++;
		}
		if(mp.size() == 1){
			cout << "NO" << endl;
			continue;
		}
		cout << "YES" << endl;
		int cnt = 0;
		for(int i = 0; i<n; i++){
			for(int j = i+1; j<n; j++){
				if(v[i] != v[j]){
					ans[cnt][0] = i+1;
					ans[cnt++][1] = j+1;
				}
			}
		}
		for(int i = 0; i<n-1; i++){
			cout << ans[i][0] << " " << ans[i][1] << endl;
		}
	}
	return 0;
}
