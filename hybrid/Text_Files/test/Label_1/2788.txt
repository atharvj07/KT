#include <iostream>
#include <vector>
using namespace std;
int32_t main() {
  string s;
  int n, m;
  cin >> s >> m;
  vector<int32_t> dp(s.size() + 2);
  dp[0] = 0;
  dp[s.size() + 1] = 0;
  for (int i = 1; i <= s.size(); ++i) {
    dp[i] = dp[i - 1];
    dp[i] += (s[i - 1] == '(');
    dp[i] -= (s[i - 1] == ')');
    dp[i] = max(0, dp[i]);
  }
  for (int i = s.size(); i >= 0; --i) {
    dp[i] = max(dp[i], dp[i + 1]);
  }
  while (m--) {
    int32_t l, r;
    cin >> l >> r;
    cout << min(dp[r], dp[l - 1] * 2) << endl;
  }
}
