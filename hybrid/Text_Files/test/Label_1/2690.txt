#include <bits/stdc++.h>

using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  vector<string> a(n);
  for (int i = 0; i < n; ++i) { cin >> a[i]; }
  int ans = 0;
  for (int i = 0; i < m; ++i) {
    string t;
    for (int j = 0; j < n; ++j) { t += a[j][i]; }
    if (t != a[0].substr(0, i + 1)) ++ans;
  }
  cout << ans << endl;
  return 0;
}
