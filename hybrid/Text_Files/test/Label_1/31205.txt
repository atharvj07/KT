import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class TreeGame {

    static class Node {
        int id;
        List<Node> children;

        Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }
        for (int i = 0; i < n - 1; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            nodes[a].children.add(nodes[b]);
            nodes[b].children.add(nodes[a]);
        }

        int maxProduct = dfs(nodes[1], null, 1);
        System.out.println(maxProduct);
    }

    private static int dfs(Node node, Node parent, int subtreeSize) {
        int maxProduct = 1;
        for (Node child : node.children) {
            if (child != parent) {
                int childSubtreeSize = dfs(child, node, 0);
                maxProduct = Math.max(maxProduct, childSubtreeSize * (subtreeSize - childSubtreeSize));
            }
        }
        return subtreeSize + maxProduct;
    }
}
