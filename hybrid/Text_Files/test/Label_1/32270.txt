import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt(), t = sc.nextInt(), r = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        int ans = 0;
        for (int i = 0, j = 0; i < m; i++) {
            while (j < i && a[i] - a[j] > t) {
                j++;
            }
            ans = Math.max(ans, i - j + 1);
        }
        System.out.println(ans >= r ? ans : -1);
    }
}
