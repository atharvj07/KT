#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n;
    cin >> n;

    if (n <= 5) {
        cout << 1 << "\n" << n << "\n";
        return 0;
    }

    if (n <= 11) {
        cout << 2 << "\n" << 2 << " " << n - 2 << "\n";
        return 0;
    }

    cout << 3 << "\n" << 2 << " " << 3 << " " << n - 5 << "\n";
    return 0;
}
