#include <iostream>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  bool flag = false;
  while (n--) {
    string name;
    int before, after;
    cin >> name >> before >> after;
    if (before >= 2400 && after > before) {
      flag = true;
      break;
    }
  }
  cout << (flag ? "YES" : "NO") << endl;
  return 0;
}
