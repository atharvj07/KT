import java.util.*;
import java.io.*;

public class Main {
    static long mod = 998244353;
    static int k;
    static long[] fact, invfact;
    static int[] a;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        k = Integer.parseInt(br.readLine());

        StringTokenizer st = new StringTokenizer(br.readLine());
        a = new int[k];
        for (int i = 0; i < k; i++) a[i] = Integer.parseInt(st.nextToken());

        fact = new long[200001];
        invfact = new long[200001];
        fact[0] = fact[1] = 1;
        for (int i = 2; i < 200001; i++) {
            fact[i] = (i * fact[i - 1]) % mod;
        }
        invfact[200000] = pow(fact[200000], mod - 2);
        for (int i = 199999; i >= 0; i--) {
            invfact[i] = (invfact[i + 1] * (i + 1)) % mod;
        }

        System.out.println(choose(a[k - 1] + a[k - 2] + 1, a[k - 1]));
    }

    static long choose(int n, int k) {
        long num = fact[n];
        long den = (invfact[k] * invfact[n - k]) % mod;
        return (num * den) % mod;
    }

    static long pow(long a, long p) {
        if (p == 0) return 1;
        if (p == 1) return a;
        if (p % 2 == 0) {
            long half = pow(a, p / 2);
            return (half * half) % mod;
        } else {
            return (a * pow(a, p - 1)) % mod;
        }
    }
}
