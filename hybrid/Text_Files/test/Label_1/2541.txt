#include <bits/stdc++.h>
using namespace std;
#define int long long
#define ff first
#define ss second
int32_t main() {
  int n;
  cin >> n;
  vector<int> a(n), b(n);
  for (int &x : a) cin >> x;
  for (int &x : b) cin >> x;
  int pfx1 = 0, pfx2 = 0;
  for (int i = 1; i + 1 < n; i++) {
    pfx1 += a[i];
    pfx2 += b[i];
  }
  if (pfx1 != pfx2) {
    cout << "No";
    return 0;
  }
  int l = 0, r = 0;
  for (int i = 1; i + 1 < n; i++) {
    if (a[i] != b[i]) {
      r = i;
      break;
    }
  }
  for (int i = 1; i + 1 < n; i++) {
    if (a[i] != b[i]) {
      l = i;
    }
  }
  int ans = b[l];
  for (int i = l; i < r + 1; i++) {
    if (i + 1 < n) { ans -= a[i]; }
    if (i > 0) { ans += a[i - 1]; }
  }
  a[l] = ans;
  bool ok = 1;
  for (int i = 1; i + 1 < n; i++) {
    if (a[i] != b[i]) {
      ok = 0;
      break;
    }
  }
  if (ok == 1)
    cout << "Yes";
  else
    cout << "No";
  return 0;
}
