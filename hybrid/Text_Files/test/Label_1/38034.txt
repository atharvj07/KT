w, h, n = map(int, input().split())

# Initialize sets to keep track of horizontal and vertical cuts
horizontal_cuts = set([0, h])
vertical_cuts = set([0, w])

# Lists to keep track of lengths of horizontal and vertical segments
horizontal_lengths = [h]
vertical_lengths = [w]

for _ in range(n):
    cut_type, cut_location = input().split()
    cut_location = int(cut_location)
    
    if cut_type == 'H':
        # Find the two closest existing cuts
        lower_cut = max([cut for cut in horizontal_cuts if cut < cut_location])
        upper_cut = min([cut for cut in horizontal_cuts if cut > cut_location])
        
        # Remove the old segment and add the two new segments
        horizontal_lengths.remove(upper_cut - lower_cut)
        horizontal_lengths.append(cut_location - lower_cut)
        horizontal_lengths.append(upper_cut - cut_location)
        
        # Add the new cut to the set
        horizontal_cuts.add(cut_location)
    else:
        # Find the two closest existing cuts
        left_cut = max([cut for cut in vertical_cuts if cut < cut_location])
        right_cut = min([cut for cut in vertical_cuts if cut > cut_location])
        
        # Remove the old segment and add the two new segments
        vertical_lengths.remove(right_cut - left_cut)
        vertical_lengths.append(cut_location - left_cut)
        vertical_lengths.append(right_cut - cut_location)
        
        # Add the new cut to the set
        vertical_cuts.add(cut_location)
    
    # Calculate the area of the largest fragment
    max_area = max(horizontal_lengths) * max(vertical_lengths)
    print(max_area)
