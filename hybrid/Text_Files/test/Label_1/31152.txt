import java.util.*;
import java.io.*;
public class Main {
    public static void main(String[] args) {
        Scanner scan=new Scanner(System.in);
        int n=scan.nextInt();
        int[][] tasks=new int[n][3];
        for (int i=0;i<n;i++) {
            tasks[i][0]=scan.nextInt();
            tasks[i][1]=scan.nextInt();
            tasks[i][2]=scan.nextInt();
        }
        int[][] pairs=new int[n][2];
        int balance[]=new int[3];
        for (int i=0;i<n;i++) {
            int choice[]=solve(balance, tasks[i]);
            if (choice==null) {
                System.out.println("Impossible");
                return;
            }
            for (int x:choice) {
                balance[x]+=tasks[i][x];
            }
            pairs[i]=choice;
        }
        int sum=balance[0]+balance[1]+balance[2];
        if (sum%3!=0) {
            System.out.println("Impossible");
            return;
        }
        for (int i=0;i<n;i++) {
            char c1=' ',c2=' ';
            if (pairs[i][0]==0) c1='L';
            else if (pairs[i][0]==1) c1='M';
            else c1='W';
            if (pairs[i][1]==0) c2='L';
            else if (pairs[i][1]==1) c2='M';
            else c2='W';
            System.out.println(c1+""+c2);
        }
    }
    static int[] solve(int[] b, int[] a) {
        int[] c=b.clone();
        for (int i=0;i<3;i++) {
            for (int j=i+1;j<3;j++) {
                if (a[j]-a[i]<-c[i]+c[j]) {
                    c[i]+=a[j]-a[i];
                    c[j]-=a[j]-a[i];
                }
                else if (a[j]-a[i]>-c[i]+c[j]) {
                    c[j]+=c[i]-a[i]+a[j];
                    c[i]-=c[i]-a[i]+a[j];
                }
            }
        }
        for (int i=0;i<3;i++) {
            if (c[i]<-1000000000) return null;
            if (a[i]>0 && c[i]<0) return null;
            if (a[i]<0 && c[i]>0) return null;
        }
        if (c[0]==c[1] && c[1]==c[2] && c[2]==0) {
            int[] ret=new int[2];
            if (a[0]<=0) {
                ret[0]=1;
                ret[1]=2;
            }
            else {
                ret[0]=0;
                ret[1]=1;
            }
            return ret;
        }
        return null;
    }
}
