import sys
input=sys.stdin.readline

n,m=map(int, input().split())
a=list(map(int, input().split()))
b=list(map(int, input().split()))
for i in range(m):
    query=list(map(int, input().split()))
    if query[0]==1:
        for j in range(query[3]):
            b[query[2]+j]+=a[query[1]+j]
    else:
        print(b[query[1]-1])
