#include <iostream>

using namespace std;

int main() {
  int n;
  cin >> n;
  int a[n];
  for (int i = 0; i < n; i++) cin >> a[i];
  bool ok = true;
  for (int i = 0; i < n; i++) {
    if (a[i] == 200) ok = false;
  }
  if (n % 2 == 1) ok = false;
  cout << (ok ? "YES" : "NO") << endl;
  return 0;
}
