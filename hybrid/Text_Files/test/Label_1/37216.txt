n, m = map(int, input().split())
grid = [list(map(int, input().split())) for _ in range(n)]

min_row = min(min(row) for row in grid)
grid = [[cell - min_row for cell in row] for row in grid]

min_col = min(min(grid[i][j] for i in range(n)) for j in range(m))
moves = ['row ' + str(i + 1) for i in range(n) for _ in range(min_row)]
moves += ['col ' + str(j + 1) for j in range(m) for _ in range(min_col)]

if any(grid[i][j] - min_col for j in range(m) for i in range(n)):
    print(-1)
else:
    print(len(moves))
    print('\n'.join(moves))
