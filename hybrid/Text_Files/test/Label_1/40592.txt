import queue
import sys
input=sys.stdin.readline
for _ in range(int(input())):
  n=int(input())
  g=[[] for i in range(n)]
  for i in range(n-1):
    a,b=map(int,input().split())
    a-=1;b-=1
    g[a].append(b)
    g[b].append(a)
  vis=[False]*n
  dis=[-1]*n
  q=queue.Queue()
  q.put((0,0))
  vis[0]=True
  dis[0]=0
  ans=0
  while not q.empty():
    v,d=q.get()
    ans=max(ans,d)
    for u in g[v]:
      if not vis[u]:
        vis[u]=True
        dis[u]=d+1
        q.put((u,d+1))
  print(ans)
