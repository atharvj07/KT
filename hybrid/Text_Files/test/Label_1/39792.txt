n, k = map(int, input().split())
denominations = list(map(int, input().split()))
q = int(input())
requests = [int(input()) for _ in range(q)]

# Create a dictionary to store the minimum number of bills for each possible amount
min_bills = {0: 0}

# Populate the dictionary with the minimum number of bills needed for each amount
for den in denominations:
    for amount in range(den, den*k + 1, den):
        bills = amount // den
        if amount not in min_bills or bills < min_bills[amount]:
            min_bills[amount] = bills

# Check each request
for x in requests:
    min_result = float('inf')
    for den in denominations:
        for i in range(k + 1):
            remaining = x - i * den
            if remaining in min_bills and i + min_bills[remaining] <= k:
                min_result = min(min_result, i + min_bills[remaining])
    print(min_result if min_result != float('inf') else -1)
