import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = in.nextInt();
        int count = 0;
        boolean success = false;
        for (int i : arr) {
            if (i == 0) {
                if (success) continue;
                else count++;
            } else {
                success = true;
                count++;
            }
        }
        System.out.println(count);
    }
}
