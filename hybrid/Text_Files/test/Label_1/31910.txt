import java.util.Scanner;
import java.util.Arrays;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        String s1 = in.nextLine();
        String s2 = in.nextLine();
        String virus = in.nextLine();

        // Build the suffix array for virus
        int[] suffixArray = new int[virus.length()];
        int[] lcp = new int[virus.length()];
        buildSuffixArray(virus, suffixArray, lcp);

        // Find the longest common subsequence of s1 and s2 that doesn't contain virus as a substring
        int longestCommonSubsequenceLength = 0;
        String longestCommonSubsequence = "";
        for (int i = 0; i < s1.length(); i++) {
            for (int j = 0; j < s2.length(); j++) {
                if (s1.charAt(i) == s2.charAt(j)) {
                    int k = i + 1;
                    int l = j + 1;
                    int currentLength = 1;
                    while (k < s1.length() && l < s2.length() && s1.charAt(k) == s2.charAt(l)) {
                        k++;
                        l++;
                        currentLength++;
                    }
                    boolean valid = true;
                    for (int m = 0; m < currentLength; m++) {
                        int index = binarySearch(suffixArray, s1.substring(i + m, i + m + 1));
                        if (index != -1 && lcp[index] >= currentLength - m) {
                            valid = false;
                            break;
                        }
                    }
                    if (valid) {
                        if (currentLength > longestCommonSubsequenceLength) {
                            longestCommonSubsequenceLength = currentLength;
                            longestCommonSubsequence = s1.substring(i, i + currentLength);
                        }
                    }
                }
            }
        }

        // Output the longest common subsequence
        if (longestCommonSubsequenceLength == 0) {
            System.out.println(0);
        } else {
            System.out.println(longestCommonSubsequence);
        }
    }

    private static void buildSuffixArray(String s, int[] suffixArray, int[] lcp) {
        int n = s.length();
        int[][] suffixArrayAndLcp = new int[n][2];
        for (int i = 0; i < n; i++) {
            suffixArrayAndLcp[i][0] = i;
            suffixArrayAndLcp[i][1] = s.charAt(i);
        }
        Arrays.sort(suffixArrayAndLcp, (a, b) -> a[1] - b[1]);
        for (int i = 0; i < n; i++) {
            suffixArray[i] = suffixArrayAndLcp[i][0];
        }
        int[] rank = new int[n];
        for (int i = 0; i < n; i++) {
            rank[suffixArray[i]] = i;
        }
        int k = 0;
        for (int i = 0; i < n; i++) {
            if (rank[i] == n - 1) {
                k = 0;
                continue;
            }
            int j = suffixArray[rank[i] + 1];
            while (i + k < n && j + k < n && s.charAt(i + k) == s.charAt(j + k)) {
                k++;
            }
            lcp[rank[i]] = k;
            if (k > 0) {
                k--;
            }
        }
    }

    private static int binarySearch(int[] arr, String key) {
        int low = 0;
        int high = arr.length - 1;
        while (low <= high) {
            int mid = (low + high) / 2;
            if (arr[mid] == key.charAt(0)) {
                return mid;
            } else if (arr[mid] < key.charAt(0)) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }
        return -1;
    }
}
