n, m = map(int, input().split())
graph = [[0] * (n+1) for _ in range(n+1)]
for _ in range(m):
    u, v = map(int, input().split())
    graph[u][v] = 1
for k in range(1, n+1):
    for u in range(1, n+1):
        for v in range(1, n+1):
            if graph[u][k] and graph[k][v]:
                graph[u][v] = 1
vis = [False] * (n+1)
ans = 0
cnt = 0
for i in range(1, n+1):
    if not vis[i]:
        cnt += 1
        q = [i]
        while q:
            t = q.pop(0)
            if vis[t]: continue
            vis[t] = True
            for j in range(1, n+1):
                if graph[t][j] and not vis[j]:
                    q.append(j)
    if cnt > 1:
        ans = -1
        break
    ans += 1
print(ans)
