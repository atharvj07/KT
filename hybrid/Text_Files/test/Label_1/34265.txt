import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        StringBuilder sb = new StringBuilder();
        while (in.hasNext()) {
            long n = in.nextLong();
            int k = in.nextInt();
            if (n == 0 && k == 0) {
                break;
            }
            for (long i = n; i < n + k; i++) {
                sb.append(i);
            }
            System.out.println(sb.toString());
            sb.setLength(0);
        }
        in.close();
    }
}
