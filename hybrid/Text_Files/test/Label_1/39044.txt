n = int(input())
revealed = input()
words = [input() for _ in range(int(input()))]
candidates = [[] for _ in range(n)]
for word in words:
    for i in range(n):
        if revealed[i] != '*' and revealed[i] != word[i]:
            break
        candidates[i].append(word[i])
ans = 0
for i in range(n):
    if revealed[i] == '*':
        ans += len(candidates[i]) != 0
print(ans)
