#include <bits/stdc++.h>

using namespace std;

const int MAX_N = 1e5 + 5;

int n;
string words[MAX_N], text;
bool vis[MAX_N];

int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> words[i];
  }
  cin >> text;
  int m = text.size();
  int p1 = 0, p2 = 0;
  while (p1 < n && p2 < m) {
    int len = words[p1].size();
    bool found = false;
    while (p2 + len <= m && !found) {
      bool ok = true;
      for (int i = 0; i < len; i++) {
        if (text[p2 + i] != words[p1][i]) {
          ok = false;
          break;
        }
      }
      if (ok) {
        if (p2 != 0) {
          if (text[p2 - 1] != '<') {
            ok = false;
          }
        }
        if (p2 + len != m) {
          if (text[p2 + len] != '>' && text[p2 + len] != '<') {
            ok = false;
          }
        }
        if (ok && vis[p1] == false) {
          vis[p1] = true;
          found = true;
        }
      }
      p2++;
    }
    if (!found) {
      cout << "no" << endl;
      return 0;
    }
    p1++;
  }
  for (int i = 0; i < n; i++) {
    if (!vis[i]) {
      cout << "no" << endl;
      return 0;
    }
  }
  cout << "yes" << endl;
  return 0;
}
