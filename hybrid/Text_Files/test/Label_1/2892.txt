#include <bits/stdc++.h>
using namespace std;
int main() {
  string s;
  cin >> s;
  for (int i = 0; i < s.size(); i++) {
    if (s[i] != '9' && s[i] != '0' && i > 0) { s[i] = '9' - s[i] + '0'; }
    if (s[i] == '0' && i > 0) { s[i] = '9'; }
  }
  if (s[0] == '0') {
    int i = 0;
    while (s[i] == '0') { i++; }
    if (i > 0 && s[i] != '9') {
      s[i] = '9' - s[i] + '0';
    }
  }
  cout << s << endl;
}
