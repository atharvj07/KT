import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        System.out.println(solve(n, m));
    }

    static long solve(int n, int m) {
        int[] dp = new int[m + 2];
        long mod = 1000000007;
        dp[2] = 1;
        for (int i = 3; i <= m + 1; i++) {
            for (int j = 2; j < i; j++) {
                dp[i] = (dp[i] + dp[j] * dp[i - j]) % (int) mod;
            }
            dp[i] = (dp[i] + 1) % (int) mod;
        }
        long ans = (long) dp[m + 1] * n % mod;
        return ans;
    }
}
