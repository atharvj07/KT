import java.util.Scanner;

public class Main {

    static int[] arr;
    static long[] tree;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt(), m = in.nextInt();
        arr = new int[n + 1];
        tree = new long[(n + 1) * 4];
        for (int i = 1; i <= n; i++)
            arr[i] = in.nextInt();
        init(1, n, 1);
        while (m-- > 0) {
            int opt = in.nextInt(), l = in.nextInt(), r = in.nextInt(), x = in.nextInt();
            if (opt == 1)
                update(1, n, 1, l, r, x);
            else
                System.out.println(query(1, n, 1, l, r, x));
        }
    }

    static void init(int l, int r, int p) {
        if (l == r) {
            tree[p] = arr[l];
            return;
        }
        int mid = (l + r) / 2;
        init(l, mid, p * 2);
        init(mid + 1, r, p * 2 + 1);
        tree[p] = Math.min(tree[p * 2], tree[p * 2 + 1]);
    }

    static void update(int l, int r, int p, int uL, int uR, int x) {
        if (uL <= l && uR >= r) {
            tree[p] = Math.min(tree[p], x);
            arr[l] = Math.min(arr[l], x);
            return;
        }
        if (uL > r || uR < l)
            return;
        int mid = (l + r) / 2;
        update(l, mid, p * 2, uL, uR, x);
        update(mid + 1, r, p * 2 + 1, uL, uR, x);
        tree[p] = Math.min(tree[p * 2], tree[p * 2 + 1]);
    }

    static long query(int l, int r, int p, int qL, int qR, int x) {
        if (qL > r || qR < l)
            return 0;
        if (qL <= l && qR >= r) {
            if (tree[p] == x)
                return r - l + 1;
            return 0;
        }
        int mid = (l + r) / 2;
        return query(l, mid, p * 2, qL, qR, x) + query(mid + 1, r, p * 2 + 1, qL, qR, x);
    }
}
