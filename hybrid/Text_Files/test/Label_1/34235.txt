import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    private static final long MOD = 1000000007;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int m = Integer.parseInt(br.readLine());
        boolean[] players = new boolean[n];
        Arrays.fill(players, true);
        long[] dp = new long[n];
        int[][] queries = new int[m][2];
        for (int i = 0; i < m; i++) {
            queries[i] = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        }
        dp[0] = 1;
        for (int i = 1; i < n; i++) {
            if (a[i] >= a[i - 1]) {
                dp[i] = dp[i - 1];
            }
            if (i >= 4 && a[i] == a[i - 1] && a[i] == a[i - 2] && a[i] == a[i - 3] && a[i] == a[i - 4] && players[i - 1] && players[i - 2] && players[i - 3] && players[i - 4]) {
                dp[i] = (dp[i - 1] + dp[i - 4]) % MOD;
            }
        }
        StringBuilder sb = new StringBuilder();
        for (int[] query : queries) {
            int t = query[0] - 1;
            int x = query[1] - 1;
            players[x] = t == 0;
            long ans = dp[n - 1];
            if (!players[x]) {
                ans = dp[x - 1];
                if (x >= 4 && a[x] == a[x - 1] && a[x] == a[x - 2] && a[x] == a[x - 3] && a[x] == a[x - 4] && players[x - 1] && players[x - 2] && players[x - 3] && players[x - 4]) {
                    ans = (ans + dp[x - 5]) % MOD;
                }
            }
            sb.append(ans).append("\n");
        }
        System.out.print(sb);
    }
}
