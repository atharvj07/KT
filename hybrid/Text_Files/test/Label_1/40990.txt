from collections import defaultdict

n = int(input())
servers = defaultdict(list)
for s in range(n):
    h, *p = input()[7:].split('/')
    p = '.'.join(p)
    servers[h].append(p)
grps = []
for h, ps in servers.items():
    for p in ps:
        if not (h,p) in grps:
            grp = set()
            for h2, ps2 in servers.items():
                if set(ps) == set(ps2): 
                    grp.add(h2)
            if len(grp)>1: grps.append(grp)
print(len(grps))
for grp in sorted(grps):
    print(*sorted(grp))
