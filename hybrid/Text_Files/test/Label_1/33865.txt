import java.util.*;

class Main {

    static class Page {
        int start, end;
        boolean hasSpecial;
        ArrayList<Integer> specials;

        Page(int start, int end) {
            this.start = start;
            this.end = end;
            this.hasSpecial = false;
            this.specials = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        long n = sc.nextLong();
        int m = sc.nextInt();
        int k = sc.nextInt();

        ArrayList<Integer> p = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            p.add(sc.nextInt());
        }

        int numPages = (int) Math.ceil((double) n / k);
        ArrayList<Page> pages = new ArrayList<>();
        for (int i = 0; i < numPages; i++) {
            pages.add(new Page(i * k + 1, Math.min(i * k + k, (int) n)));
        }

        for (int special : p) {
            for (Page page : pages) {
                if (page.start <= special && special <= page.end) {
                    page.hasSpecial = true;
                    page.specials.add(special);
                }
            }
        }

        int numOperations = 0;
        while (true) {
            boolean foundSpecialPage = false;
            for (Page page : pages) {
                if (page.hasSpecial) {
                    foundSpecialPage = true;
                    for (int special : page.specials) {
                        for (int i = special; i < page.end; i++) {
                            if (i + k <= page.end) {
                                pages.get((i + k) / k - 1).specials.add(pages.get((i + k) / k).specials.remove(0));
                            }
                        }
                    }
                    page.hasSpecial = false;
                    page.specials.clear();
                    numOperations++;
                }
            }
            if (!foundSpecialPage) {
                break;
            }
        }

        System.out.println(numOperations);
    }
}
