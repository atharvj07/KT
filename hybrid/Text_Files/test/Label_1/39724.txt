import math

n = int(input())
a = [float(input()) for _ in range(n)]
if sum(a) != 0:
    print('invalid input')
    exit()
b = list(map(lambda v: round(v), a))
if sum(b) != 0:
    for i in range(n):
        if abs(b[i] - a[i]) < 1:
            b[i] += a[i] - b[i]
if sum(b) != 0:
    print('invalid input')
    exit()
for v in b:
    print(int(v))
