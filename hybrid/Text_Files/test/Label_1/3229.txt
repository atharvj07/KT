#include<bits/stdc++.h>
using namespace std;
int n;
string s[100005],tmp;
int main(){
    cin>>n;
    for(int i=0;i<n;i++) cin>>s[i];
    const int inf=(int)1e9;
    set<int> p;
    for(int i=0;i<n;i++){
        tmp="";
        int sz=s[i].size();
        for(int j=0;j<sz;j++){
            if(s[i][j]!='?'){
                if(tmp[tmp.size()-1]-s[i][j]>0) tmp+=s[i][j]-1;
                else tmp+=s[i][j];
            }
            else{
                if(tmp.empty()){
                    tmp=j==0?'1':'0';
                }
                else{
                    auto it=p.lower_bound(tmp[tmp.size()-1]-tmp[tmp.size()-2]);
                    tmp+=*it+1;
                }
            }
        }
        if(tmp[0]=='0'){
            cout<<"NO"<<endl;
            return 0;
        }
        for(int j=0;j<sz;j++){
            if(s[i][j]=='?'){
                s[i][j]=tmp[j];
            }
        }
        p.insert(stoi(s[i]));
        if(i>0 and s[i]<=s[i-1]){
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
    for(int i=0;i<n;i++) cout<<s[i]<<endl;
}
