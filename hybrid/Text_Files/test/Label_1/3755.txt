#include <iostream>
#include <vector>

int main() {
  int n, k;
  std::string s;
  std::cin >> n >> k;
  std::cin >> s;
  std::vector<int> freq(26, 0);
  int level = 0;
  for (char c : s) {
    int pos = c - 'a';
    freq[pos]++;
    level = std::max(level, freq[pos] / k);
  }
  std::cout << level << std::endl;
  return 0;
}
