import java.util.Scanner;
import java.util.Arrays;
import java.util.HashMap;

public class Main {
    static int n, m;
    static char[][] grid;
    static HashMap<Integer, Integer> row = new HashMap<>();
    static HashMap<Integer, Integer> col = new HashMap<>();

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        grid = new char[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                grid[i][j] = s.charAt(j);
                if (grid[i][j] == '#') {
                    row.put(i, row.getOrDefault(i, 0) + 1);
                    col.put(j, col.getOrDefault(j, 0) + 1);
                }
            }
        }
        boolean flag = true;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == '#') {
                    if (row.get(i) == null || row.get(i) == 0 || col.get(j) == null || col.get(j) == 0) {
                        flag = false;
                        break;
                    }
                    row.put(i, row.get(i) - 1);
                    col.put(j, col.get(j) - 1);
                }
            }
        }
        System.out.println(flag ? "Yes" : "No");
    }
}
