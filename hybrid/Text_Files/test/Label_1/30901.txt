import java.io.*;
import java.util.*;

class Main {
    static class Pair<A, B> {
        A fst;
        B snd;
        Pair(A fst, B snd) { this.fst = fst; this.snd = snd; }
    }
    static class Segment {
        int x0, y0, x1, y1;
        boolean isHorizontal;
        Segment(int x0, int y0, int x1, int y1, boolean isHorizontal) { this.x0=x0; this.y0=y0; this.x1=x1; this.y1=y1; this.isHorizontal=isHorizontal; }
    }
    public static void main(String[] args) throws IOException {
        Reader reader = new InputStreamReader(System.in);
        BufferedReader in = new BufferedReader(reader);
        int n = Integer.parseInt(in.readLine());
        Segment[] s = new Segment[n];
        for(int i = 0; i < n; i++) {
            String[] str = in.readLine().split(" ");
            int x0 = Integer.parseInt(str[0]);
            int y0 = Integer.parseInt(str[1]);
            int x1 = Integer.parseInt(str[2]);
            int y1 = Integer.parseInt(str[3]);
            s[i] = new Segment(x0, y0, x1, y1, x0 == x1);
        }
        in.close();
        reader.close();
        long ans = 0;
        for(Segment a : s) {
            if(!a.isHorizontal) continue;
            for(Segment b : s) {
                if(!b.isHorizontal) continue;
                if(a.x0 == b.x0 || a.x0 == b.x1) continue;
                if(a.y0 > b.y0) continue;
                for(Segment c : s) {
                    if(c.isHorizontal) continue;
                    if(c.y0 == a.y0 || c.y0 == b.y0) continue;
                    if(c.x0 > a.x0) continue;
                    for(Segment d : s) {
                        if(d.isHorizontal) continue;
                        if(d.y0 == a.y0 || d.y0 == b.y0) continue;
                        if(d.x0 > b.x0) continue;
                        ans++;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
