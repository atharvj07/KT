import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] grid = new int[3][3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }
        int[] rows = new int[3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                rows[i] += grid[i][j];
            }
        }
        int[] cols = new int[3];
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                cols[i] += grid[j][i];
            }
        }
        boolean possible = true;
        for (int i = 1; i < 3; i++) {
            possible &= rows[i] == rows[0] && cols[i] == cols[0];
        }
        System.out.println(possible ? "Yes" : "No");
    }
}
