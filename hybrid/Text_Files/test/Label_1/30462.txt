import java.util.*;

public class Main {
    static int n;
    static List<Integer>[] adj;
    static int[] visited;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        adj = new List[n];
        visited = new int[n];
        for (int i = 0; i < n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i < n; i++) {
            int a = scanner.nextInt() - 1;
            int b = scanner.nextInt() - 1;
            adj[a].add(b);
            adj[b].add(a);
        }
        long ans = 1;
        for (int i = 0; i < n; i++) {
            if (visited[i] == 0) {
                ans = Math.max(ans, dfs(i));
            }
        }
        System.out.println(ans);
    }

    static long dfs(int v) {
        visited[v] = 1;
        long size = 1;
        for (int u : adj[v]) {
            if (visited[u] == 0) {
                size += dfs(u);
            }
        }
        return size * (n - size);
    }
}
