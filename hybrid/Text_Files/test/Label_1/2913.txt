#include <vector>
#include <iostream>
using namespace std;
#define inf 1e9
#define N 200005
vector<int> adj[N];
int sub[N],deg[N];
int n,ans;
void dfs(int v=1,int p=1){
    sub[v]=1;
    deg[v]=1;
    for(int to:adj[v]){
        if(to!=p){
            dfs(to,v);
            deg[v]+=deg[to];
            sub[v]+=sub[to];
        }
    }
    ans=min(ans,n-sub[v]);
}
int main(){
    int u,v;
    cin>>n;
    for(int i=0;i<n-1;i++){
        cin>>u>>v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    ans=inf;
    dfs();
    cout<<ans<<endl;
    return 0;
}
