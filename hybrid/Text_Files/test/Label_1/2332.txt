#include <bits/stdc++.h>
#define ll long long
using namespace std;

pair<ll, ll> get_line(ll m, ll b) {
    if (m == 0)
        return {b, b};
    return {0, b};
}

ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }
ll lcm(ll a, ll b) { return (a / gcd(a, b)) * b; }

int main() {
    ll m, b;
    cin >> m >> b;
    pair<ll, ll> line = get_line(m, b);
    ll l = line.first, r = line.second;
    ll ans = (l + r) * (r - l + 1) / 2;
    cout << ans << endl;
    return 0;
}
