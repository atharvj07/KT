#include <iostream>
#include <algorithm>
using namespace std;
int main() {
  const long double q = 1.6180339887498948;
  string num1, num2;
  cin >> num1 >> num2;
  int cnt1 = count(num1.begin(), num1.end(), '1'), cnt2 = count(num2.begin(), num2.end(), '1');
  if (cnt1 < cnt2) cout << '<';
  else if (cnt1 > cnt2) cout << '>';
  else cout << '=';
}
