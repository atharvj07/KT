import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        List<String> examples = new ArrayList<>();
        List<String> tests = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String name = input.next();
            int type = input.nextInt();
            if (type == 1) {
                examples.add(name);
            } else {
                tests.add(name);
            }
        }
        System.out.println(examples.size() + tests.size());
        for (String example : examples) {
            System.out.println("move " + example + " " + (examples.indexOf(example) + 1));
        }
        for (String test : tests) {
            System.out.println("move " + test + " " + (examples.size() + tests.indexOf(test) + 1));
        }
    }
}
