n = int(input())
a = list(map(int, input().split()))
d, ans = [-1]*n, [-1]*n

for i in range(n):
    for j in range(i-1,-1,-1):
        if a[j] < a[i]:
            d[i] = j
            break

for i in range(n-1,-1,-1):
    for j in range(i+1,n):
        if a[j] < a[i] and d[j] <= i:
            ans[i] = a[j]
            break

for i in ans:
    if i == -1:
        i = a[i]
    print(i, end=" ")
