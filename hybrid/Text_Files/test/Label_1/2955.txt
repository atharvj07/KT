#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        vector<long long> a(n);
        for(int i = 0; i < n; i++)
            cin>>a[i];
        long long l_sum = a[0], r_sum = 0, moves = 0;
        for(int i = 1; i < n; i++)
            r_sum += a[i];
        for(int i = 1; i < n; i++)
        {
            if(l_sum > r_sum)   l_sum += a[i];
            else
            {
                moves++;
                r_sum -= l_sum;
                l_sum = a[i];
            }
        }
        moves++;
        cout<<moves<<" "<<l_sum<<" "<<r_sum<<endl;
    }
    return 0;
}
