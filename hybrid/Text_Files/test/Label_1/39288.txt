import sys

input = sys.stdin.readline
getint = lambda: int(input().strip())

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

for _ in range(getint()):
    n = getint()
    a = list(map(int, input().split()))
    odd = []
    even = []
    for x in a:
        if x % 2: odd.append(x)
        else: even.append(x)
    odd.sort()
    even.sort()
    i = 0
    j = len(even) - 1
    res = 0
    while i < len(odd) and j >= 0:
        if odd[i] * 2 > even[j]:
            res += len(even) - j - 1
            i += 1
        else:
            j -= 1
    print(res + len(odd) * len(even))
