#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

const int MAXN = 2e5 + 5;  // 2x (max n = 2e5)

vector<int> adj[MAXN];
int a[MAXN];

ll dp[MAXN][2];  // dp[i][0] = maximum possible beauty without changing a[i], dp[i][1] = maximum possible beauty with changing a[i]

ll gcd(ll a, ll b) {
    if (b == 0)
        return a;
    return gcd(b, a % b);
}

void dfs(int u, int p) {
    dp[u][0] = a[u];
    dp[u][1] = a[u];
    for (int v : adj[u]) {
        if (v == p)
            continue;
        dfs(v, u);
        dp[u][0] = gcd(dp[u][0], dp[v][0]);
        dp[u][1] = max(dp[u][1], gcd(dp[u][0], dp[v][1]));
        dp[u][1] = max(dp[u][1], gcd(dp[u][1], dp[v][0]));
    }
}

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) { cin >> a[i]; }
    for (int i = 1; i < n; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
        adj[y].push_back(x);
    }
    dfs(1, 1);
    for (int i = 1; i <= n; i++) { cout << dp[i][1] << " "; }
    cout << endl;
    return 0;
}
