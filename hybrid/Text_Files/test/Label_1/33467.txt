import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String[] positions = scanner.nextLine().split(" ");
        scanner.close();

        boolean kingIsChecked = false;
        for (String position : positions) {
            int x = position.charAt(0) - 'a' + 1;
            int y = position.charAt(1) - '0';

            if (isAttackedByRook(x, y, positions)) {
                kingIsChecked = true;
                break;
            }
        }

        System.out.println(kingIsChecked ? "CHECKMATE" : "OTHER");
    }

    private static boolean isAttackedByRook(int x, int y, String[] positions) {
        for (String position : positions) {
            int rx = position.charAt(0) - 'a' + 1;
            int ry = position.charAt(1) - '0';

            if (rx == x || ry == y) {
                return true;
            }
        }

        return false;
    }
}
