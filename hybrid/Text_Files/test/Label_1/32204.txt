import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String s = scanner.nextLine();
        char[] c = s.toCharArray();
        int n = c.length;
        int i1 = 0, i2 = 0, cnt1 = 0;
        while (true) {
            while (i1 < n && c[i1] == 'T') i1++;
            if (i1 == n) break;
            i2 = (i1 + 1) % n;
            while (i2 != i1 && c[i2] == 'H') i2 = (i2 + 1) % n;
            if (i2 == i1) break;
            char tmp = c[i1]; c[i1] = c[i2]; c[i2] = tmp;
            i1++; cnt1++;
        }
        System.out.println(cnt1);
    }
}
