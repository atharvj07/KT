import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }

        boolean[] dp = new boolean[300];
        for (int i = 1; i < 300; i++) {
            dp[i] = false;
            for (int j = 0; j < n; j++) {
                if (a[j] >= i) {
                    dp[i] |= !dp[a[j] - i];
                }
            }
        }

        System.out.println(dp[a[0]] ? "BitAryo" : "BitLGM");
    }
}
