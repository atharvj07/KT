#include <iostream>
#include <vector>
using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    bool ok = true;
    if (h[0] >= h[1]) {
      for (int i = 1; i + 1 < n; i++) {
        if (h[i] <= h[i - 1] || h[i] >= h[i + 1]) ok = false;
      }
      if (h[n - 1] <= h[n - 2]) ok = false;
    }
    cout << (ok ? "YES" : "NO") << endl;
  }
}
