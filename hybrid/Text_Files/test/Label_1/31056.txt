import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {
    static class Edge {
        int to;

        public Edge(int to) {
            this.to = to;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int N = Integer.parseInt(st.nextToken());
        int M = Integer.parseInt(st.nextToken());
        int[] D = new int[N + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= N; i++) {
            D[i] = Integer.parseInt(st.nextToken());
        }
        List<List<Edge>> adj = new ArrayList<>();
        for (int i = 0; i <= N; i++) {
            adj.add(new ArrayList<>());
        }
        int[] color = new int[N + 1];
        int[] weight = new int[M + 1];
        Arrays.fill(weight, -1);
        for (int i = 1; i <= M; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adj.get(u).add(new Edge(v));
            adj.get(v).add(new Edge(u));
        }
        if (!solve(1, 1, adj, D, color, weight)) {
            System.out.println(-1);
        } else {
            StringBuilder sb = new StringBuilder();
            for (int i = 1; i <= N; i++) {
                sb.append(color[i] == 1 ? "W" : "B");
            }
            sb.append("\n");
            for (int i = 1; i <= M; i++) {
                sb.append(weight[i]).append("\n");
            }
            System.out.print(sb.toString());
        }
    }

    private static boolean solve(int curr, int col, List<List<Edge>> adj, int[] D, int[] color, int[] weight) {
        color[curr] = col;
        Map<Integer, Integer> dist = new HashMap<>();
        dist.put(curr, 0);
        if (!dfs(curr, curr, adj, D, dist, weight)) {
            return false;
        }
        for (Edge nxt : adj.get(curr)) {
            if (weight[nxt.to] == -1) {
                if (!solve(nxt.to, 3 - col, adj, D, color, weight)) {
                    return false;
                }
            }
        }
        return true;
    }

    private static boolean dfs(int curr, int par, List<List<Edge>> adj, int[] D, Map<Integer, Integer> dist, int[] weight) {
        for (Edge nxt : adj.get(curr)) {
            if (nxt.to == par) {
                continue;
            }
            if (!dist.containsKey(nxt.to)) {
                dist.put(nxt.to, dist.get(curr) + 1);
                if (!dfs(nxt.to, curr, adj, D, dist, weight)) {
                    return false;
                }
            } else if (dist.get(nxt.to) == dist.get(curr)) {
                return false;
            }
        }
        if (!dist.containsKey(par)) {
            return true;
        }
        if (weight[curr] == -1) {
            weight[curr] = D[curr] - dist.get(curr) + dist.get(par);
            if (weight[curr] <= 0) {
                return false;
            }
        } else if (weight[curr] != D[curr] - dist.get(curr) + dist.get(par)) {
            return false;
        }
        return true;
    }
}
