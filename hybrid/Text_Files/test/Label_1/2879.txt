#include <bits/stdc++.h>

#define ll long long
#define ld long double

using namespace std;

int sgn(ll x) { return x < 0 ? -1 : x > 0; }
ld atan2_scalefactor = 1.0;

int main() {
  int T;
  cin >> T;
  while (T--) {
    int n;
    cin >> n;
    vector<pair<ld, ld>> a(n);
    vector<pair<ld, ld>> b(n);
    for (int i = 0; i < n; i++) {
      int x, y, u, v;
      cin >> x >> y >> u >> v;
      if (x == u && y == v) {
        a[i] = {x, y};
        b[i] = {x, y};
      } else {
        ll dx = (x - u);
        ll dy = (y - v);
        a[i] = {(ld)x, (ld)y};
        b[i] = {(ld)x + atan2_scalefactor * dx / sqrt(dx * dx + dy * dy),
                (ld)y + atan2_scalefactor * dy / sqrt(dx * dx + dy * dy)};
      }
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j < n; j++) {
        if (a[i].first == b[j].first && a[i].second == b[j].second) {
          vector<ld> ang = {atan2(a[i].second - b[i].second, a[i].first - b[i].first) + 2.0 * M_PI,
                            atan2(a[j].second - b[j].second, a[j].first - b[j].first)};
          sort(ang.begin(), ang.end());
          if (sgn(ang[1]) != sgn(ang[0])) { ans++; }
        }
        if (b[i].first == a[j].first && b[i].second == a[j].second) {
          vector<ld> ang = {atan2(a[i].second - b[i].second, a[i].first - b[i].first) + 2.0 * M_PI,
                            atan2(a[j].second - b[j].second, a[j].first - b[j].first)};
          sort(ang.begin(), ang.end());
          if (sgn(ang[1]) != sgn(ang[0])) { ans++; }
        }
      }
    }
    cout << ans << endl;
  }
  return 0;
}
