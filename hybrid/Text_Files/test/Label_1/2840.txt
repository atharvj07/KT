#include <iostream>
#include <string>
using namespace std;
const int N = 55;
int t, n;
string s;
bool vis[N][N][N];
bool dfs(int a, int b, int c) {
  if (a == 0 && b == 0 && c == 0) return true;
  if (vis[a][b][c] == 1) return false;
  vis[a][b][c] = 1;
  if (a > 0 && b > 0) {
    if (dfs(a - 1, b - 1, c)) return true;
  }
  if (b > 0 && c > 0) {
    if (dfs(a, b - 1, c - 1)) return true;
  }
  return false;
}
int main() {
  cin >> t;
  while (t--) {
    cin >> s;
    n = s.length();
    int a = 0, b = 0, c = 0;
    for (int i = 0; i < n; i++) {
      if (s[i] == 'A') a++;
      if (s[i] == 'B') b++;
      if (s[i] == 'C') c++;
    }
    if (dfs(a, b, c))
      cout << "YES\n";
    else
      cout << "NO\n";
  }
  return 0;
}
