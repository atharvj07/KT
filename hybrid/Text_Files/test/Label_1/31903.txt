import java.util.*;

public class Main {
    static int mod = (int) (1e9 + 7);
    static long[] fact;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] x = new long[n];
        for (int i = 0; i < n; i++) {
            x[i] = sc.nextInt();
        }
        sc.close();
        fact = new long[n + 1];
        fact[0] = 1;
        for (int i = 1; i <= n; i++) {
            fact[i] = (fact[i - 1] * i) % mod;
        }
        long ans = 0;
        long sum = 0;
        for (int i = 1; i < n; i++) {
            sum += x[i] - x[i - 1];
            ans += (sum * fact[n - 1 - i]) % mod;
            ans %= mod;
        }
        System.out.println(ans);
    }
}
