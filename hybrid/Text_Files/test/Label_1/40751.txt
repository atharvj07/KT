n, q, k = map(int, input().split())
a = list(map(int, input().split()))

for _ in range(q):
    l, r = map(int, input().split())
    segment = a[l-1:r]

    if len(segment) == 1:
        print(k - 1)
    else:
        diff = (segment[0] - 1) + (k - segment[-1])
        for i in range(1, len(segment)):
            diff += (segment[i] - segment[i-1] - 1)

        print(diff)
