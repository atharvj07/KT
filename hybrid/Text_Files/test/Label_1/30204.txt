import java.io.*;
import java.util.*;

public class Main {
    static int[] a;
    static int[][] g;
    static boolean[] vis;

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(in.readLine());
        int n = Integer.parseInt(st.nextToken());
        int q = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(in.readLine());
        a = new int[n + 1];
        for (int i = 1; i <= n; i++)
            a[i] = Integer.parseInt(st.nextToken());
        g = new int[n + 1][];
        for (int i = 1; i <= n; i++)
            g[i] = new int[30];
        int[] last = new int[30];
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j < 30; j++)
                if ((a[i] & (1 << j)) != 0)
                    last[j] = i;
            for (int j = 0; j < 30; j++)
                if ((a[i] & (1 << j)) != 0)
                    g[i][j] = last[j];
        }
        while (q-- > 0) {
            st = new StringTokenizer(in.readLine());
            int x = Integer.parseInt(st.nextToken());
            int y = Integer.parseInt(st.nextToken());
            if (a[x] == 0)
                System.out.println("Fou");
            else {
                vis = new boolean[n + 1];
                vis[x] = true;
                boolean ok = dfs(x, y);
                System.out.println(ok ? "Shi" : "Fou");
            }
        }
    }

    static boolean dfs(int x, int y) {
        if (x == y)
            return true;
        for (int i = 0; i < 30; i++)
            if ((a[x] & (1 << i)) != 0 && !vis[g[x][i]]) {
                vis[g[x][i]] = true;
                if (dfs(g[x][i], y))
                    return true;
            }
        return false;
    }
}
