#include <bits/stdc++.h>
using namespace std;
int const M = 100000;
int main() {
    int n, p;
    cin >> n >> p;
    p *= M;
    int ans = 1;
    for (int i = n - 2; i >= 1; i--, ans++) {
        long double x = (1.0 * (n - i - 1) / (n - 1)) * (1.0 * (n - i) / (n - 2));
        if (x * M >= p) break;
    }
    cout << ans << endl;
}
