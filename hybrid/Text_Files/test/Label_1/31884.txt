import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    String s = in.next();
    int ans = 0;
    int cnt = 0;
    for (char c : s.toCharArray()) {
      cnt = (c == 'A' || c == 'C' || c == 'G' || c == 'T') ? cnt + 1 : 0;
      ans = Math.max(ans, cnt);
    }
    System.out.println(ans);
  }
}
