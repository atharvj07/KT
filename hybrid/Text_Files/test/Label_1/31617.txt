import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            b[i] = sc.nextInt();
        }
        sc.close();
        boolean[] dp = new boolean[m + 1];
        dp[0] = true;
        for (int i = 0; i < n; i++) {
            for (int j = b[i]; j >= 0; j--) {
                if (j >= a[i] && dp[j - a[i]]) {
                    dp[j] = true;
                }
            }
        }
        System.out.println(dp[m] ? "YES" : "NO");
    }
}
