import java.util.*;

class Main {
    public static void main(String[] a) {
        Scanner in = new Scanner(System.in);
        long A = in.nextLong(), B = in.nextLong();
        while (B != 0) {
            long C = B;
            B = A % B;
            A = C;
        }
        System.out.println(A);
    }
}
