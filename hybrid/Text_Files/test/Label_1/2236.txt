#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, d;
    cin >> n >> d;
    vector<int> t(n);
    for (int i = 0; i < n; i++) cin >> t[i];
    sort(t.begin(), t.end());
    if (d < 10 * n) {
        cout << -1 << endl;
        return 0;
    }
    int ans = 0, sum = 0;
    for (int i = 0; i < n; i++) {
        if (sum + t[i] + 10 <= d) {
            sum += t[i] + 10;
            ans++;
        } else
            break;
    }
    ans += (d - sum) / 5;
    cout << ans << endl;
    return 0;
}
