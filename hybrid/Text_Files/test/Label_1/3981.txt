#include <iostream>
using ll = long long;

int main() {
  int T;
  std::cin >> T;
  while (T--) {
    ll a, b, n, m;
    std::cin >> a >> b >> n >> m;
    if (b == 0 || n == 0)
      std::cout << "No\n";
    else if (m > 0 && b < n && (b - 1) * m > a)
      std::cout << "No\n";
    else
      std::cout << "Yes\n";
  }
}
