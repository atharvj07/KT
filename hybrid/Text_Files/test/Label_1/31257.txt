import java.util.*;
import java.io.*;

public class Main {

    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int[] x = new int[n];
        for (int i = 0; i < n; i++)
            x[i] = sc.nextInt();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int l = i + 1;
            int r = i + 2;
            while (r < n && x[r] - x[i] <= d)
                r++;
            r--;
            if (l < r)
                ans += (r - l) * (r - l - 1) / 2;
        }
        System.out.println(ans);
    }
}
