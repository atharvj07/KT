#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main() {
  int n; cin >> n;
  vector<int> w(n);
  for (int i = 0; i < n; i++) cin >> w[i];
  string s; cin >> s;
  vector<bool> used(n, false);
  set<int> avail;
  for (int i = 0; i < n; i++) avail.insert(i);
  int nxt = 0;
  for (const char& c : s) {
    if (c == '0') {
      auto it = avail.lower_bound(nxt);
      if (it == avail.end()) it = avail.begin();
      cout << *it+1 << " ";
      nxt = *it;
      used[*it] = true;
      avail.erase(it);
    } else {
      auto it = used.rbegin();
      cout << *it+1 << " ";
      nxt = *it;
      used[*it] = false;
      avail.insert(*it);
    }
  }
  cout << endl;
}
