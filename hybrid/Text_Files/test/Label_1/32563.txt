import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        boolean flag = false;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] == n) {
                flag = true;
                System.out.print(n - 1 + " ");
            } else {
                break;
            }
        }
        if (!flag) {
            int sum = 0;
            for (int i = 0; i < n; i++) {
                sum += a[i];
            }
            if (sum % 2 == 0) {
                System.out.print(sum / 2 + " ");
            }
            if (sum % 2 == 1) {
                System.out.print((sum + 1) / 2 + " ");
            }
        }
        System.out.println();
    }
}
