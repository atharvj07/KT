#include <bits/stdc++.h>
using namespace std;
#define ll long long
#define mod 998244353
int n, ans, a[100005];
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> a[i];
  for (int i = 1; i < n; i++) {
    if (a[i] != -1) {
      if (a[i] < a[i - 1]) ans = 0;
    }
  }
  if (a[0] != -1 && a[0] > 1) ans = 0;
  if (a[n - 1] != -1 && a[n - 1] > 1) ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == -1) a[i] = 1;
  }
  for (int i = 1; i < n; i++) {
    if (a[i] == -1) a[i] = a[i - 1];
  }
  for (int i = n - 2; i >= 0; i--) {
    if (a[i] == -1) a[i] = a[i + 1];
  }
  for (int i = 0; i + 1 < n; i++) {
    if (a[i] > a[i + 1]) ans = 0;
  }
  if (ans) {
    for (int i = 0; i < n; i++) {
      if (a[i] == a[i + 1] && i + 1 < n) ans = (ans * 2) % mod;
    }
  }
  cout << ans << endl;
  return 0;
}
