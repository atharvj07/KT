#include <vector>
#include <string>
#include <algorithm>
#include <iostream>
using namespace std;

int main() {
    int n; cin >> n;
    string s; cin >> s;
    vector<char> v(s.begin(), s.end());
    int cnt = 0;
    for (char x : v) {
        if (x == 'X') cnt++;
        if (cnt > n/2) break;
    }
    if (cnt == n/2) {
        cout << 0 << '\n' << s << '\n';
        return 0;
    }
    int ans = cnt;
    for (int i = 0; i + cnt < n/2; i++) {
        if (v[i] != v[i + cnt]) ans++;
    }
    for (int i = 0; i + cnt < n/2 && ans > 0; i++) {
        if (v[i] != v[i + cnt]) {
            swap(v[i], v[i + cnt]);
            ans--;
        }
    }
    cout << ans << '\n';
    for (char x : v) cout << x;
    cout << '\n';
    return 0;
}
