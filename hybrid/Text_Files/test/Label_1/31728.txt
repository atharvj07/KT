import java.util.*;

class Main {
  static int mod = 1000000007;
  static long[][] dp;
  static int[][] obstacles;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    obstacles = new int[n][3];
    for (int i = 0; i < n; i++) {
      obstacles[i][0] = sc.nextInt();
      obstacles[i][1] = sc.nextInt();
      obstacles[i][2] = sc.nextInt();
    }
    dp = new long[4][m + 1];
    dp[2][1] = 1;
    for (int j = 2; j <= m; j++) {
      for (int i = 1; i <= 3; i++) {
        boolean blocked = false;
        for (int[] obstacle : obstacles) {
          if (obstacle[0] == i && obstacle[1] <= j && j <= obstacle[2]) {
            blocked = true;
            break;
          }
        }
        if (!blocked) {
          if (i > 1) dp[i][j] += dp[i - 1][j - 1];
          dp[i][j] += dp[i][j - 1];
          if (i < 3) dp[i][j] += dp[i + 1][j - 1];
        }
        dp[i][j] %= mod;
      }
    }
    System.out.println(dp[2][m]);
  }
}
