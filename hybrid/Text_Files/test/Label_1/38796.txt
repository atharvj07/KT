line = input().split()
n = int(line[-1])
line[-2] = "="
nums = list(range(1, n + 1))

for i in range(len(line) - 2):
    if line[i] == "?":
        for used in set(nums):
            nums.remove(used)
            line[i] = str(used)
            if eval(" ".join(line)) == n:
                print("Possible")
                print(" ".join(line))
                exit(0)
            nums.append(used)
        line[i] = "?"

print("Impossible")
