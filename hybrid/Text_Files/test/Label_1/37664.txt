n, k = map(int, input().split())
b = list(map(int, input().split()))
c = list(input())
white = []
orange = []
red = []
for i in range(n):
    if c[i] == 'W':
        white.append((b[i], i))
    elif c[i] == 'O':
        orange.append((b[i], i))
    else:
        red.append((b[i], i))
white.sort(reverse=True)
orange.sort(reverse=True)
red.sort(reverse=True)
if k <= 2:
    print(-1)
    exit()
if k % 2 == 0 and len(red) > 0:
    print(-1)
    exit()
if k == 3 and len(red) > 0 and len(white) == 0 and len(orange) == 0:
    print(-1)
    exit()
if k > 3 and len(red) > 0 and len(white) == 0 and len(orange) == 0:
    print(-1)
    exit()
ans = 0
if k % 2 == 1:
    if len(red) > 0:
        ans += red[0][0]
        k -= 1
        red.pop(0)
    else:
        ans += white[0][0]
        k -= 1
        white.pop(0)
if k % 2 == 0:
    while k > 0:
        if len(white) > 0 and len(orange) > 0:
            if white[0][0] > orange[0][0]:
                ans += white[0][0]
                k -= 1
                white.pop(0)
            else:
                ans += orange[0][0]
                k -= 1
                orange.pop(0)
        else:
            if len(white) > 0:
                ans += white[0][0]
                k -= 1
                white.pop(0)
            elif len(orange) > 0:
                ans += orange[0][0]
                k -= 1
                orange.pop(0)
            else:
                break
print(ans)
