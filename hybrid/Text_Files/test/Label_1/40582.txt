import sys
input = sys.stdin.readline
n, m, q = map(int, input().split())
grid = [[0] * m for i in range(n)]
ops = []
for _ in range(q):
    t, *args = map(int, input().split())
    ops.append((t, *args))
for t, *args in reversed(ops):
    if t == 1:
        r = args[0] - 1
        grid[r] = grid[r][-1:] + grid[r][:-1]
    elif t == 2:
        c = args[0] - 1
        for x in grid:
            x[c], x[-1] = x[-1], x[c]
    else:
        r, c, x = args
        grid[r - 1][c - 1] = x
for x in grid:
    print(*x)
