n, m, k = map(int, input().split())
a = list(map(int, input().split()))
op = []
for _ in range(m):
    l, r, d = map(int, input().split())
    op.append((l - 1, r - 1, d))
qu = []
for _ in range(k):
    x, y = map(int, input().split())
    qu.append((x - 1, y - 1))
delta = [0] * (n + 2)
for s, e, v in op:
    delta[s] += v
    delta[e + 1] -= v
for q in qu:
    s, e = q
    for i in range(s, e + 1):
        a[i] += delta[i]
        delta[i + 1] += delta[i]
    for i in range(s, e + 1):
        delta[i] = 0
print(*a)
