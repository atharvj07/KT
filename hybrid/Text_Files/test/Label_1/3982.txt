#include <iostream>
#include <string>
using namespace std;

int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  for (int i = 0; i < n - 1; i++) {
    if (s[i] == 'g' && s[i + 1] == 'o') {
      int j = i + 1;
      while (j < n && s[j] == 'g') { j++; }
      if (j < n && s[j] == 'o') {
        s.replace(i, j - i + 1, "***");
      }
    }
  }
  cout << s << endl;
  return 0;
}
