#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define vi(x) vector<x>

ll solve() {
    ll n;
    cin >> n;
    string s;
    cin >> s;
    vi(ll) pos(n);
    for (int i = 0; i < n; i++) { cin >> pos[i]; }
    ll l = -1e9, r = 1e9, ans = -1;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'R') {
            r = min(r, pos[i]);
        } else {
            l = max(l, pos[i]);
        }
        if (l >= r) {
            return i;
        }
    }
    return -1;
}
int main() { cout << solve() << endl; }
