#include <bits/stdc++.h>

using namespace std;

long long sq(long long p) { return p * p; }

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<long long> vx(n), vy(n), ux(n), uy(n);
    for (int i = 0; i < n; i++) {
      cin >> vx[i] >> vy[i] >> ux[i] >> uy[i];
      ux[i] -= vx[i], uy[i] -= vy[i];
    }
    set<pair<long long, long long>> uus;
    for (int i = 0; i < n; i++) { uus.insert({ux[i], uy[i]}); }
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      int cnt = uus.erase({ux[i], uy[i]});
      for (auto p : uus) {
        if (sq(p.first * uy[i] - p.second * ux[i]) <=
            sq(ux[i]) * sq(uy[i]) + sq(p.first) * sq(p.second))
          ans += cnt;
      }
    }
    ans /= 2;
    cout << ans << "\n";
  }
}
