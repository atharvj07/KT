import java.util.*;

public class Main {
    static void swap(int[][] a, int x1, int y1, int x2, int y2) {
        int tmp;
        for (int i = x1; i < x1 + x2; i++) {
            for (int j = y1; j < y1 + y2; j++) {
                tmp = a[i][j];
                a[i][j] = a[x1 + x2 - i - 1][y1 + y2 - j - 1];
                a[x1 + x2 - i - 1][y1 + y2 - j - 1] = tmp;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt(), q = sc.nextInt();
        int[][] a = new int[n + 1][m + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        while (q-- > 0) {
            int x1 = sc.nextInt(), y1 = sc.nextInt(), x2 = sc.nextInt(), y2 = sc.nextInt(), h = sc.nextInt(), w = sc.nextInt();
            swap(a, x1, y1, h, w);
            swap(a, x2, y2, h, w);
            swap(a, x1, y1, h, w);
        }
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                System.out.print(a[i][j] + " ");
            }
            System.out.println();
        }
    }
}
