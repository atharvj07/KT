import java.util.Scanner;

public class GoodSubsegments {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] permutation = new int[n];
        for (int i = 0; i < n; i++) {
            permutation[i] = scanner.nextInt();
        }
        int q = scanner.nextInt();
        for (int i = 0; i < q; i++) {
            int l = scanner.nextInt() - 1;
            int r = scanner.nextInt() - 1;
            System.out.println(countGoodSubsegments(permutation, l, r));
        }
    }

    private static int countGoodSubsegments(int[] permutation, int l, int r) {
        int[] minLeft = new int[r - l + 1];
        minLeft[0] = permutation[l];
        for (int i = 1; i <= r - l; i++) {
            minLeft[i] = Math.min(minLeft[i - 1], permutation[l + i]);
        }
        int[] maxRight = new int[r - l + 1];
        maxRight[r - l] = permutation[r];
        for (int i = r - l - 1; i >= 0; i--) {
            maxRight[i] = Math.max(maxRight[i + 1], permutation[l + i]);
        }
        int count = 0;
        for (int i = 0; i <= r - l; i++) {
            if (minLeft[i] <= maxRight[i]) {
                count++;
            }
        }
        return count;
    }
}
