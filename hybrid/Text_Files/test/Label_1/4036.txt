#include <iostream>
using namespace std;
typedef long long LL;
LL l, r, ans = 1, cnt = 0, best = 0;

int main() {
    cin >> l >> r;
    for (LL i = 2; i * i <= r; ++i) {
        if (r / i < l) break;
        cnt = 0;
        for (LL j = i; j <= r; j += i) {
            if (j >= l) ++cnt;
        }
        if (cnt > best) best = cnt, ans = i;
    }
    cout << ans << endl;
    return 0;
}
