import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner input = new Scanner(System.in);
    int n = input.nextInt();
    int m = input.nextInt();
    int[][] graph = new int[n + 1][n + 1];
    for (int i = 0; i < m; i++) {
      int u = input.nextInt();
      int v = input.nextInt();
      graph[u][v] = 1;
      graph[v][u] = 1;
    }
    boolean[] visited = new boolean[n + 1];
    if (isHamiltonian(graph, visited, 1, 1)) {
      System.out.println("Yes");
    } else {
      System.out.println("No");
    }
  }

  private static boolean isHamiltonian(int[][] graph, boolean[] visited, int vertex, int count) {
    if (count == graph.length - 1) {
      return true;
    }
    visited[vertex] = true;
    for (int i = 1; i < graph.length; i++) {
      if (graph[vertex][i] == 1 && !visited[i]) {
        if (isHamiltonian(graph, visited, i, count + 1)) {
          return true;
        }
      }
    }
    visited[vertex] = false;
    return false;
  }
}
