import sys

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        d = list(map(int, input().split()))
        if sum(d) % n != 0:
            print("NO")
            continue
        avg = sum(d) // n
        ok = True
        for i in range(n):
            if d[i] > avg * 2:
                ok = False
                break
        print("YES") if ok else print("NO")

solve()
