import java.util.Scanner;

public class ZebraPhoto {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine(); // Consume the newline character

        int[] photo = new int[n];
        for (int i = 0; i < n; i++) {
            photo[i] = scanner.nextInt();
        }

        boolean isZebraPhoto = true;
        int currentStripeWidth = 0;
        int expectedStripeWidth = -1; // Initialize to an invalid value

        for (int i = 0; i < n; i++) {
            if (photo[i] == 0) {
                currentStripeWidth++;
            } else {
                if (expectedStripeWidth == -1) {
                    expectedStripeWidth = currentStripeWidth;
                }

                if (currentStripeWidth != expectedStripeWidth) {
                    isZebraPhoto = false;
                    break;
                }

                currentStripeWidth = 0;
            }
        }

        if (currentStripeWidth != 0) {
            isZebraPhoto = false;
        }

        System.out.println(isZebraPhoto ? "YES" : "NO");
    }
}
