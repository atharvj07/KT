import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int q = input.nextInt();
        for (int i = 0; i < q; i++) {
            int a = input.nextInt();
            int highestGcd = 0;
            for (int b = 1; b < a; b++) {
                int xor = a ^ b;
                int and = a & b;
                int gcd = 0;
                for (int j = 1; j <= xor; j++) {
                    if (xor % j == 0 && and % j == 0) {
                        gcd = j;
                    }
                }
                highestGcd = Math.max(highestGcd, gcd);
            }
            System.out.println(highestGcd);
        }
    }
}
