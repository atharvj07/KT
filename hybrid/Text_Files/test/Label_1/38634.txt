def can_form_chessboard(board):
    for row in board:
        if row.count('W') != 4 or row.count('B') != 4:
            return "NO"
        for i in range(7):
            if row[i] == row[i+1]:
                return "NO"
    return "YES"

board = [input() for _ in range(8)]
print(can_form_chessboard(board))
