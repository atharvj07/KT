n = int(input())
a = list(map(int, input().split()))
b = []
for i in range(n):
    if not b or a[i] > b[-1]:
        b.append(a[i])
    else:
        for j in range(len(b)):
            if b[j] >= a[i]:
                b[j] = a[i]
                break
        else:
            b.append(a[i])
print('\n'.join(map(' '.join, map(str, b))))
