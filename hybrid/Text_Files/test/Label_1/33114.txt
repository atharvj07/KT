import java.io.*;
import java.util.*;

class Main {
    private static class Node {
        Node left, right;
        int val, size = 1;

        void update() {
            size = 1 + (left != null ? left.size : 0) + (right != null ? right.size : 0);
        }
    }

    private static class AVL {
        Node root;
        int sumMedians = 0;

        void add(int x) {
            root = add(x, root);
        }

        void del(int x) {
            List<Integer> list = new ArrayList<>();
            list.add(root.val);
            root = del(x, root, list);
            int sum = 0;
            for (int i : list) {
                sum += i % 2 == 0 ? i / 2 + 1 : i / 2;
            }
            sumMedians -= sum;
        }

        Node add(int x, Node cur) {
            if (cur == null) {
                cur = new Node();
                cur.val = x;
                return cur;
            }
            if (x < cur.val) {
                cur.left = add(x, cur.left);
            } else if (x > cur.val) {
                cur.right = add(x, cur.right);
            } else {
                return cur;
            }
            cur.update();
            return balance(cur);
        }

        Node del(int x, Node cur, List<Integer> list) {
            if (cur == null) {
                return null;
            }
            if (x < cur.val) {
                cur.left = del(x, cur.left, list);
            } else if (x > cur.val) {
                cur.right = del(x, cur.right, list);
            } else {
                if (cur.left == null && cur.right == null) {
                    list.add(cur.val);
                    return null;
                }
                if (cur.left == null) {
                    list.add(cur.right.val);
                    return cur.right;
                }
                if (cur.right == null) {
                    list.add(cur.left.val);
                    return cur.left;
                }
                if (cur.left.size > cur.right.size) {
                    int v = get(cur.left, cur.left.size);
                    cur.left = del(v, cur.left, list);
                    cur.val = v;
                    cur.update();
                    return balance(cur);
                } else {
                    int v = get(cur.right, 1);
                    cur.right = del(v, cur.right, list);
                    cur.val = v;
                    cur.update();
                    return balance(cur);
                }
            }
            cur.update();
            return balance(cur);
        }

        private Node balance(Node cur) {
            if (bal(cur) > 1) {
                if (bal(cur.left) < 0) {
                    cur.left = leftRotate(cur.left);
                }
                return rightRotate(cur);
            } else if (bal(cur) < -1) {
                if (bal(cur.right) > 0) {
                    cur.right = rightRotate(cur.right);
                }
                return leftRotate(cur);
            }
            return cur;
        }

        private Node leftRotate(Node cur) {
            Node next = cur.right;
            cur.right = next.left;
            next.left = cur;
            cur.update();
            next.update();
            return next;
        }

        private Node rightRotate(Node cur) {
            Node next = cur.left;
            cur.left = next.right;
            next.right = cur;
            cur.update();
            next.update();
            return next;
        }

        private int bal(Node cur) {
            return height(cur.left) - height(cur.right);
        }

        private int height(Node cur) {
            if (cur == null) {
                return 0;
            }
            return 1 + Math.max(height(cur.left), height(cur.right));
        }

        private int get(Node cur, int k) {
            if (cur.left != null && cur.left.size >= k) {
                return get(cur.left, k);
            }
            if (cur.left == null && k == 1) {
                return cur.val;
            }
            return get(cur.right, k - 1 - (cur.left != null ? cur.left.size : 0));
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        AVL avl = new AVL();
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            if (s.equals("add")) {
                avl.add(sc.nextInt());
            } else if (s.equals("del")) {
                avl.del(sc.nextInt());
            } else {
                System.out.println(avl.sumMedians);
            }
        }
    }
}
