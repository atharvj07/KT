import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int p = sc.nextInt();
        int q = sc.nextInt();

        long n = 1L << 60;

        for (long i = 2; i <= n; i++) {

            if ((i & i - 1) == 0) {
                p++;
            } else {
                String str = i + "";
                boolean ok = true;
                for (int j = 0, k = str.length() - 1; j < k; j++, k--) {
                    if (str.charAt(j) != str.charAt(k)) {
                        ok = false;
                        break;
                    }
                }
                if (ok) {
                    q++;
                }
            }

            if (p <= q * 1L * (long) p / (long) q) {
                System.out.println(i - 1);
                return;
            }
        }

        System.out.println("Palindromic tree is better than splay tree");
    }
}
