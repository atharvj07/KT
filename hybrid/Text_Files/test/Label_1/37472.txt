import bisect

n, m = map(int, input().split())
x = list(map(int, input().split()))
x.sort()
y = []
ans = 0

for i in range(m):
    idx = bisect.bisect_left(x, 0.5)
    if idx == 0:
        y.append(x[idx] - 1)
    elif idx == n:
        y.append(x[idx - 1] + 1)
    else:
        l, r = x[idx - 1], x[idx]
        if abs(l) < abs(r):
            y.append(l)
        else:
            y.append(r)

    ans += abs(y[i])

print(ans)
print(*y)
