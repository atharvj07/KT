#include <bits/stdc++.h>
#define int long long
using namespace std;
signed main() {
  cin.tie(0), cout.tie(0), ios_base::sync_with_stdio(0);
  int T;
  cin >> T;
  while (T--) {
    int x;
    cin >> x;
    int ans = 1e18;
    for (int i = 2; i * i <= x; i++) {
      if (x % i) continue;
      int cnt = 0;
      while (x % i == 0) x /= i, cnt++;
      ans = min(ans, cnt);
    }
    if (x != 1) ans = min(ans, 1ll);
    cout << (ans != 1e18 ? ans : -1) << '\n';
  }
}
