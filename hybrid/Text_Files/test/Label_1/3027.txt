#include <iostream>
#include <vector>
const int mod = 998244353;
using namespace std;
int main() {
    long long ans = 1;
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    int cnt = 0;
    for (int i = 0; i + 1 < n; i++) {
        cnt += a[i] == a[i + 1] && a[i] != -1;
    }
    for (int i = 0; i < n; i++) {
        if (a[i] != -1) continue;
        int lCnt = 0;
        if (i) {
            lCnt += a[i - 1] != -1;
            if (i + 1 < n) {
                lCnt += a[i + 1] == a[i - 1];
            }
        }
        int rCnt = 0;
        if (i + 1 < n) {
            rCnt += a[i + 1] != -1;
            if (i) {
                rCnt += a[i - 1] == a[i + 1];
            }
        }
        if (cnt >= 1) {
            if (lCnt) {
                ans = (ans * (k - 2)) % mod;
            } else {
                ans = (ans * (k - 1)) % mod;
            }
        } else {
            if (rCnt) {
                ans = (ans * (k - 2)) % mod;
            } else {
                ans = (ans * k) % mod;
            }
        }
        cnt += lCnt + rCnt;
    }
    cout << ans << "\n";
    return 0;
}
