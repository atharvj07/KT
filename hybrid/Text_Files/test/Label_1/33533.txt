import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static class Request implements Comparable<Request> {
        int start, duration;

        public Request(int start, int duration) {
            this.start = start;
            this.duration = duration;
        }

        @Override
        public int compareTo(Request other) {
            return start - other.start;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Request[] requests = new Request[n];
        for (int i = 0; i < n; i++) {
            int start = sc.nextInt();
            int duration = sc.nextInt();
            requests[i] = new Request(start, duration);
        }
        Arrays.sort(requests);

        int[] startDays = new int[n];
        int[] endDays = new int[n];

        int lastEndDay = 0;
        for (int i = 0; i < n; i++) {
            if (requests[i].start >= lastEndDay) {
                startDays[i] = requests[i].start;
            } else {
                startDays[i] = lastEndDay;
            }
            endDays[i] = startDays[i] + requests[i].duration - 1;
            lastEndDay = endDays[i];
        }

        for (int i = 0; i < n; i++) {
            System.out.println(startDays[i] + " " + endDays[i]);
        }
    }
}
