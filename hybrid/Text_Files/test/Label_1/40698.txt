import sys

read = sys.stdin.readline
write = sys.stdout.write

MOD = 10 ** 9 + 7

for _ in range(int(read())):

    n, f = map(int, read().split())

    if f - 1 > n - f:
        f = n - f

    ans = 1
    for i in range(1, f):
        ans = (ans * (n - i)) % MOD
        ans = (ans * pow(i + 1, MOD - 2, MOD)) % MOD

    write(str(ans) + "\n")
