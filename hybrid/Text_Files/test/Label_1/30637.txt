import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

class Main {
  static final int MAX_N = 500, MAX_SUM = 1000000;
  static int[][] g = new int[MAX_N][MAX_N], sum = new int[MAX_N][MAX_N], dp = new int[MAX_N][MAX_N];
  static int n, m;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    StringTokenizer st = new StringTokenizer(br.readLine());
    n = Integer.parseInt(st.nextToken());
    m = Integer.parseInt(st.nextToken());
    for (int i = 1; i <= n; i++) {
      st = new StringTokenizer(br.readLine());
      for (int j = 1; j <= m; j++) {
        g[i][j] = Integer.parseInt(st.nextToken());
        sum[i][j] = g[i][j] + sum[i - 1][j] + sum[i][j - 1] - sum[i - 1][j - 1];
      }
    }
    int answer = 0;
    for (int i = 1; i <= n; i++) {
      for (int j = 1; j <= m; j++) {
        if (i == 1 && j == 1) {
          dp[i][j] = g[i][j];
          continue;
        }
        if (i == 1) {
          dp[i][j] = Math.max(dp[i][j - 1], g[i][j]);
        } else if (j == 1) {
          dp[i][j] = Math.max(dp[i - 1][j], g[i][j]);
        } else {
          dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
          dp[i][j] = Math.max(dp[i][j], g[i][j]);
        }
        dp[i][j] = Math.max(dp[i][j], dp[i][j] - sum(1, j, i, m) + sum(1, j, i, j));
        dp[i][j] = Math.max(dp[i][j], dp[i][j] - sum(1, 1, i, j) + sum(1, 1, i, i));
        dp[i][j] = Math.max(dp[i][j], dp[i][j] - sum(j, 1, m, j) + sum(j, 1, j, i));
        dp[i][j] = Math.max(dp[i][j], dp[i][j] - sum(i, 1, i, j) + sum(i, 1, j, j));
        dp[i][j] = Math.max(dp[i][j], dp[i][j] - sum(1, 1, i, i) - sum(i, i, m, j)
            + sum(1, 1, i, j) + sum(i, i, m, m));
        answer = Math.max(answer, dp[i][j]);
      }
    }
    System.out.println(answer);
  }

  static int sum(int a, int b, int c, int d) {
    if (sum[c][d] - sum[c][b - 1] - sum[a - 1][d] + sum[a - 1][b - 1] < 0) {
      return 0;
    }
    return sum[c][d] - sum[c][b - 1] - sum[a - 1][d] + sum[a - 1][b - 1];
  }
}
