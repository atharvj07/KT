#include <bits/stdc++.h>
using namespace std;

struct user{
    string name, color;
    int before, after;
};
int main(){
    int n, cnt=0;
    cin >> n;
    vector<user> v(n);
    unordered_map<string, string> mp;
    for(int i=0; i<n; i++){
        cin >> v[i].name >> v[i].before >> v[i].after;
        if(v[i].before < 2400) v[i].color = "orange";
        else if(v[i].before < 2200) v[i].color = "yellow";
        else if(v[i].before < 2000) v[i].color = "green";
        else if(v[i].before < 1800) v[i].color = "cyan";
        else if(v[i].before < 1600) v[i].color = "blue";
        else if(v[i].before < 1400) v[i].color = "violet";
        else if(v[i].before < 1200) v[i].color = "grey";
        mp[v[i].name] = v[i].color;
    }
    for(int i=0; i<n; i++){
        if(v[i].before >= 2400 && v[i].after > v[i].before) cnt++;
    }
    if(cnt) cout << "YES\n";
    else cout << "NO\n";
}
