import java.io.BufferedReader;
import java.io.InputStreamReader;

class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] b = new int[n];
        int[] i = new int[n];
        int[] r = new int[n];
        String[] in = br.readLine().split(" ");
        for(int x = 0; x < n; x++) b[x] = Integer.parseInt(in[x]);
        in = br.readLine().split(" ");
        for(int x = 0; x < n; x++) i[x] = Integer.parseInt(in[x]);
        in = br.readLine().split(" ");
        for(int x = 0; x < n; x++) r[x] = Integer.parseInt(in[x]);
        int ans = 0;
        for(int x = 0; x < n; x++) {
            boolean found = false;
            for(int y = 0; y < n; y++) {
                if(x == y) continue;
                if(b[x] < b[y] && i[x] < i[y] && r[x] < r[y]) {
                    found = true;
                    break;
                }
            }
            if(found) ans++;
        }
        System.out.println(ans);
    }
}
