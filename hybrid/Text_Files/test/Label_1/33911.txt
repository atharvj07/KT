import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int[] m = new int[n];
        for (int i = 0; i < n; i++) {
            m[i] = sc.nextInt();
        }
        Arrays.sort(m);
        sc.close();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (x < m[i]) {
                break;
            }
            x -= m[i];
            ans++;
        }
        System.out.println(ans);
    }
}
