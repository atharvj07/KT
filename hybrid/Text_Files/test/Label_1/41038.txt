n, k = map(int, input().split())
s = input()

# Calculate the cost of connecting each room directly
direct_costs = [i + 1 for i in range(n)]

# Initialize DP array where dp[i] will store the minimum cost to connect first i rooms
dp = [float('inf')] * (n + 1)
dp[0] = 0  # Base case: cost of connecting 0 rooms is 0

for i in range(1, n + 1):
    # If connecting directly is cheaper or the only option
    dp[i] = dp[i - 1] + direct_costs[i - 1]
    
    # If there's a router spot in the current room
    if s[i - 1] == '1':
        # Calculate the range of rooms that can be covered by the router
        left = max(0, i - k - 1)
        right = min(n, i + k)
        
        # Update the DP array for all rooms that can be covered by the router
        dp[right] = min(dp[right], dp[left] + direct_costs[i - 1])

print(dp[n])
