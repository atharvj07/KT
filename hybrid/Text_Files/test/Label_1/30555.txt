import java.util.*;
import java.io.*;
class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int k = Integer.parseInt(st.nextToken());
        int n = Integer.parseInt(st.nextToken());
        Point[] p1 = new Point[k], p2 = new Point[n];
        for (int i = 0; i < k; ++i) {
            st = new StringTokenizer(br.readLine());
            p1[i] = new Point(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
        }
        for (int i = 0; i < n; ++i) {
            st = new StringTokenizer(br.readLine());
            p2[i] = new Point(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
        }
        int ans = 0;
        for (int i = 0; i < k; ++i) {
            for (int j = 0; j < n; ++j) {
                if (Point.ccw(p1[i], p2[j], p1[(i + 1) % k]) < 0)
                    ++ans;
            }
        }
        System.out.println(ans);
    }
    static class Point {
        int x, y;
        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
        static int ccw(Point a, Point b, Point c) {
            long d = ((long) (b.x - a.x)) * ((long) (c.y - a.y)) - ((long) (b.y - a.y)) * ((long) (c.x - a.x));
            if (d > 0)
                return 1;
            else if (d == 0)
                return 0;
            else
                return -1;
        }
    }
}
