n = int(input())
a = list(map(int, input().split()))

min_diff = float('inf')
total = sum(a)

for i in range(n):
    sector_sum = 0
    for j in range(i, n):
        sector_sum += a[j]
        other_sector_sum = total - sector_sum
        diff = abs(sector_sum - other_sector_sum)
        min_diff = min(min_diff, diff)

print(min_diff)
