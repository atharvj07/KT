import java.util.*;
import java.io.*;
import java.math.*;

public class Main {
  static long[] cnt_numbers;

  public static void main(String[] args) {
    Scanner cin = new Scanner(System.in);
    long n = cin.nextLong();
    long a = cin.nextLong();
    Map<Long, Long> cnt = new HashMap<Long, Long>();
    cnt_numbers = new long[(int) (1e6 + 1)];
    for (long i = 0; i < n; i++) {
      long clr = cin.nextLong();
      cnt_numbers[(int) clr]++;
      Long freq = cnt.get(clr);
      if (freq == null) {
        freq = 0L;
      }
      cnt.put(clr, ++freq);
    }
    cin.close();
    ArrayList<Long> numbers = new ArrayList<Long>(cnt.keySet());
    Collections.sort(numbers);
    long max_el = 0;
    int idx_of_max_el = 0;
    long max_freq = 1;
    int idx_of_max_freq = 0;
    for (int i = 0; i < numbers.size(); i++) {
      if (max_freq < cnt.get(numbers.get(i))) {
        max_freq = cnt.get(numbers.get(i));
        idx_of_max_freq = i;
      }
      if (max_el < numbers.get(i)) {
        max_el = numbers.get(i);
        idx_of_max_el = i;
      }
    }
    if (a == numbers.get(idx_of_max_freq) || idx_of_max_el == idx_of_max_freq) {
      System.out.println(-1);
    } else if (max_el > a && cnt_numbers[(int) max_el] > cnt_numbers[(int) a] && cnt_numbers[(int) a] + cnt_numbers[(int) max_el] == max_freq) {
      System.out.println(max_el);
    } else {
      System.out.println(-1);
    }
  }
}
