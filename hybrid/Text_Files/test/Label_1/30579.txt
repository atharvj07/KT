import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int k = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        int l = 0, r = 0;
        int minL = -1, minR = -1;
        int distinctCount = 0;
        boolean[] seen = new boolean[100001];

        while (r < n) {
            if (!seen[a[r]]) {
                distinctCount++;
                seen[a[r]] = true;
            }
            while (distinctCount > k) {
                if (!seen[a[l]]) {
                    distinctCount--;
                    seen[a[l]] = false;
                }
                l++;
            }
            if (distinctCount == k && (minL == -1 || r - l + 1 < minR - minL + 1)) {
                minL = l;
                minR = r;
            }
            r++;
        }

        if (minL == -1) {
            System.out.println("-1 -1");
        } else {
            System.out.println(minL + 1 + " " + (minR + 1));
        }
    }
}
