import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), d = sc.nextInt(), x = sc.nextInt();
        int[][] p = new int[d][n];
        for (int i = 0; i < d; i++) {
            for (int j = 0; j < n; j++) {
                p[i][j] = sc.nextInt();
            }
        }
       int[][] dp = new int[d][x + 1];
        for (int i = 1; i < d; i++) {
            for (int j = 1; j <= x; j++) {
                dp[i][j] = Math.max(dp[i][j], dp[i - 1][j]);
                for (int k = 0; k < n; k++) {
                    if(j-p[i][k] >= 0)
                     dp[i][j] = Math.max(dp[i][j], dp[i - 1][j - p[i][k]] + p[i - 1][k]);
                }
            }
        }
        System.out.println(Arrays.stream(dp[d - 1]).max().getAsInt());
    }
}
