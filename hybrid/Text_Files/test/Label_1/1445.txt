#include<bits/stdc++.h>
using namespace std;
int a[100005], b[100005];
int main()
{
    int n, l, r, f=0;
    cin>>n;
    for(int i=1; i<=n; i++)
    {
        cin>>a[i];
        b[i]=a[i];
    }
    sort(b+1, b+n+1);
    for(int i=1; i<=n; i++)
    {
        if(a[i]!=b[i])
        {
            if(f==0){l=i; f=1;}
            else r=i;
        }
    }
    if(f==0) cout<<"no"<<endl;
    else cout<<"yes"<<endl<<l<<" "<<r<<endl;
}
