#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

typedef long long ll;

vector<pair<ll, ll>> p;
vector<pair<ll, pair<ll, ll>>> ev;

bool cmp(pair<ll, ll> a, pair<ll, ll> b)
{
    if (a.first == b.first) return a.second < b.second;
    return a.first < b.first;
}

int main()
{
    ll n;
    cin >> n;

    p.resize(n);
    vector<bool> pt(n, false);

    for (ll i = 0; i < n; i++)
    {
        ll x, y;
        cin >> x >> y;

        p[i] = {x, y};
    }

    sort(p.begin(), p.end(), cmp);

    ll cnt = 0;

    for (ll i = 0; i < n; i++)
    {
        ev.push_back({p[i].first + p[i].second, {0, i}});
        ev.push_back({p[i].first - p[i].second, {1, i}});
    }

    sort(ev.begin(), ev.end());

    bool f = false;

    for (ll i = 0; i < ev.size(); i++)
    {
        if (ev[i].second.first == 0)
        {
            cnt++;
            if (cnt == n)
            {
                f = true;
                break;
            }
            pt[ev[i].second.second] = true;
        }
        else
        {
            if (pt[ev[i].second.second] == false)
            {
                f = false;
                break;
            }
            cnt--;
        }
    }

    if (f == true) cout << "YES\n";
    else cout << "NO\n";

    return 0;
}
