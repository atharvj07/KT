#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

struct query {
  int n;
  vector<int> nodes;
};

const int maxn = 2e5 + 2;

vector<int> graph[maxn];
query queries[maxn];

int depth[maxn];
bool visited[maxn];

// DFS
void dfs(int cur, int par) {
  visited[cur] = true;
  depth[cur] = depth[par] + 1;
  for (int i = 0; i < graph[cur].size(); i++) {
    int next = graph[cur][i];
    if (visited[next]) continue;
    dfs(next, cur);
  }
}

int main() {
  int nodes, queries_n;
  cin >> nodes >> queries_n;

  for (int i = 0; i < nodes - 1; i++) {
    int a, b;
    cin >> a >> b;
    graph[a].push_back(b);
    graph[b].push_back(a);
  }

  for (int i = 0; i < queries_n; i++) {
    int n;
    cin >> n;
    queries[i].n = n;
    for (int j = 0; j < queries[i].n; j++) {
      int x;
      cin >> x;
      queries[i].nodes.push_back(x);
    }
  }

  dfs(1, 0);  // root

  for (int i = 0; i < queries_n; i++) {
    int root_dist = depth[1];
    int max_dist = -1;
    int max_node = -1;

    for (int j = 0; j < queries[i].n; j++) {
      int cur = queries[i].nodes[j];
      if (depth[cur] == root_dist) {
        cout << "YES" << endl;
        goto next;
      }
      if (depth[cur] < root_dist) {
        cout << "NO" << endl;
        goto next;
      }
      if (max_dist < depth[cur] - root_dist) {
        max_dist = depth[cur] - root_dist;
        max_node = cur;
      }
    }

    if (max_dist <= 1)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  next:;
  }
}
