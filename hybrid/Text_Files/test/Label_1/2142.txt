#include <bits/stdc++.h>
using namespace std;
const int MAXN = 3e2 + 5;
const int MAXR = 3e4 + 5;
int n, r, a[MAXN], b[MAXN];
bool dp[MAXN][MAXR];
int main() {
    cin >> n >> r;
    dp[0][r] = true;
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        for (int j = 0; j + a[i] <= MAXR; j++) {
            if (dp[i - 1][j]) {
                if (j + b[i] >= 0) dp[i][j + b[i]] = true;
                dp[i][j] = true;
            }
        }
    }
    int ans = 0;
    for (int i = 0; i <= MAXR; i++)
        if (dp[n][i]) ans++;
    cout << ans << '\n';
    return 0;
}
