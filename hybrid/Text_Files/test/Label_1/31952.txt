import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int d = sc.nextInt();
        int mod = sc.nextInt();
        if ((d & 1) == 0 || d * (d - 1) / 2 > n) {
            System.out.println(0);
        } else {
            long ans = 1;
            for (int i = 0; i < d; i++) {
                ans = (ans * (n - i)) % mod;
            }
            for (int i = 1; i <= d; i++) {
                ans = (ans * i) % mod;
            }
            System.out.println(ans);
        }
    }
}
