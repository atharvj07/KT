import java.util.Scanner;

public class ParenthesisSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        String s = scanner.next();

        int open = 0, close = 0;
        StringBuilder result = new StringBuilder();

        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                open++;
                result.append('(');
            } else if (s.charAt(i) == ')') {
                close++;
                result.append(')');
            } else if (s.charAt(i) == '?') {
                if (open > close) {
                    result.append('(');
                    open++;
                } else {
                    result.append(')');
                    close++;
                }
            }

            if (close > open) {
                System.out.println(":(");
                return;
            }
        }

        if (open != close) {
            System.out.println(":(");
            return;
        }

        for (int i = 0; i < n; i++) {
            if (result.charAt(i) == '(' && i > 0 && result.charAt(i - 1) == ')') {
                System.out.println(":(");
                return;
            }
        }

        System.out.println(result.toString());
    }
}
