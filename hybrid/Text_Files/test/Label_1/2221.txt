#include <bits/stdc++.h>
#define ll long long
using namespace std;

const int MOD = 998244353;

vector<vector<int>>adj, nxtadj;
vector<int>deg, ndeg;
vector<bool>vis;

int main() {
    int n; cin >> n;
    vector<pair<int, int>>a(n);
    set<int>st;
    deg.assign(n + 1, 0), ndeg.assign(n + 1, 0);
    adj.assign(n + 1, vector<int>());
    nxtadj.assign(n + 1, vector<int>());
    for (int i = 0; i < n; i++) {
        cin >> a[i].first >> a[i].second;
        deg[a[i].first]++, deg[a[i].second]++;
        nxtadj[a[i].second].push_back(a[i].first);
        adj[a[i].first].push_back(a[i].second);
        st.insert(a[i].first), st.insert(a[i].second);
    }
    vis.assign(n + 1, 0);
    bool ok = 1;
    vector<int>v;
    for (auto node : st) {
        if (vis[node]) continue;
        int cnt_out = 0, cnt_in = 0;
        queue<int>q; q.push(node);
        vis[node] = 1;
        while (!q.empty()) {
            int tp = q.front(); q.pop();
            cnt_out += adj[tp].size();
            if (adj[tp].empty()) ok = 0;
            for (auto next_node : adj[tp]) {
                cnt_in += nxtadj[tp].size();
                if (!vis[next_node]) { q.push(next_node); vis[next_node] = 1; }
            }
        }
        if (cnt_in != cnt_out) { ok = 0; break; }
    }
    if (!ok) { cout << 0 << "\n"; return 0; }

    ll ans = 1;

    for (int i = 0; i < n; i++) {
        if (!deg[i]) {
            ans = ans * 2 % MOD;
        }
    }

    cout << ans << "\n";
    return 0;
}
