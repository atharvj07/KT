#include <iostream>
#include <string>
#include <vector>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<string> s(n);
  for (int i = 0; i < n; ++i) cin >> s[i];
  int m;
  cin >> m;
  vector<int> ans(m);
  for (int i = 0; i < m; ++i) {
    int a, b;
    cin >> a >> b;
    a--, b--;
    s.push_back(s[a] + s[b]);
    ans[i] = 0;
    int len = s[s.size() - 1].length();
    for (int j = 0; j < len; ++j) {
      int k = 0, cnt = 0;
      while (j + k < len) {
        cnt += s[s.size() - 1][j + k] == '0';
        ans[i] = max(ans[i], cnt == k ? k : 0);
        k++;
      }
    }
  }
  for (int i : ans) cout << i << endl;
  return 0;
}
