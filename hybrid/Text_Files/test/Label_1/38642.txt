import sys

input_set = list(map(int, sys.stdin.readline().split()))[0]
while input_set > 0:
    input_set -= 1
    n, m = list(map(int, sys.stdin.readline().split()))
    mat = []
    for i in range(n):
        mat.append(sorted(list(map(int, sys.stdin.readline().split()))))
    for row in mat:
        sys.stdout.write(" ".join(map(str, row)) + "\n")
