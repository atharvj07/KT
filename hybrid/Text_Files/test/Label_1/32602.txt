import java.util.Scanner;

public class CalendarReform {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int n = scanner.nextInt();

        long totalSheets = 0;

        for (int year = a; year <= a + n - 1; year++) {
            int daysPerMonth = (int) Math.sqrt(year);
            int monthsPerYear = year / daysPerMonth;
            totalSheets += monthsPerYear;
        }

        System.out.println(totalSheets);
    }
}
