#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, m, r, c;
    cin >> n >> m;
    vector<string> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cin >> r >> c;
    a[--r][--c] = 'X';
    cin >> r >> c;
    r--, c--;
    queue<pair<int, int>> q;
    q.push({r, c});
    a[r][c] = 'V';
    while (!q.empty()) {
        auto [x, y] = q.front();
        q.pop();
        if (x == r && y == c) {
            cout << "YES";
            return 0;
        }
        for (auto [dr, dc] : vector<pair<int, int>>{{0, -1}, {0, 1}, {-1, 0}, {1, 0}}) {
            int nx = x + dr, ny = y + dc;
            if (nx >= 0 && nx < n && ny >= 0 && ny < m &&
                (a[nx][ny] == 'X' || a[nx][ny] == '.')) {
                q.push({nx, ny});
                a[nx][ny] = 'V';
            }
        }
    }
    cout << "NO";
}
