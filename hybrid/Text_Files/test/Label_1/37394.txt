t = int(input())
for _ in range(t):
    n = int(input())
    d = list(map(int, input().split()))
    d.sort()
    possible = True
    total_sum = sum(d)
    a = [0] * (2 * n)

    for i in range(2 * n):
        a_i = total_sum - 2 * sum(d[i + 1:])
        if a_i % (2 * n) != 0 or i < 2 * n - 1 and d[i] == d[i + 1]:
            possible = False
            break
        a[i] = a_i // (2 * n)
        total_sum -= 2 * a[i]

    if possible and len(set(a)) == 2 * n:
        print("YES")
    else:
        print("NO")
