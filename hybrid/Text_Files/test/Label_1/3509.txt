#include<bits/stdc++.h>
using namespace std;
const int maxn = 3e5 + 5;
typedef long long ll;
struct edge{
    int u, v, id, w;
    edge(int _u = 0, int _v = 0, int _id = 0, int _w = 0):
        u(_u), v(_v), id(_id), w(_w) {}
}e[maxn];
int n, m, k, p[maxn], c[maxn], mx[maxn], ans[maxn];
ll d[maxn];
int find_par(int x){
    return x == p[x] ? x : p[x] = find_par(p[x]);
}
bool merge(int x, int y){
    if(find_par(x) == find_par(y)) return 0;
    if(c[find_par(x)] > c[find_par(y)]) swap(x, y);
    p[find_par(x)] = find_par(y);
    c[find_par(y)] += c[find_par(x)];
    return 1;
}
int main(){
    cin >> n >> m >> k;
    for(int i = 0; i <= n; i ++) p[i] = i, c[i] = 1;
    for(int i = 1; i <= m; i ++){
        int x, y, w;
        cin >> x >> y >> w;
        e[i] = edge(x, y, i, w);
    }
    sort(e + 1, e + 1 + m, [&](edge x, edge y){
        return x.w < y.w;
    });
    for(int i = 1; i <= m; i ++){
        int u = e[i].u, v = e[i].v;
        int r1 = find_par(u), r2 = find_par(v);
        if(r1 != r2){
            if(r1 != n) mx[r1] = max(mx[r1], e[i].w);
            if(r2 != n) mx[r2] = max(mx[r2], e[i].w);
            if(merge(r1, v)){
                c[r2] ++;
            }else if(merge(u, r2)){
                c[r1] ++;
            }else{
                p[n] = merge(r1, r2) ? r2 : r1;
                c[p[n]] = c[r1] + c[r2];
            }
        }
    }
    for(int i = 0; i <= n; i ++) mx[i] = max(mx[i], mx[find_par(i)]);
    int len = 0, mx_len = 0;
    for(int i = 0; i <= n; i ++){
        if(c[find_par(i)] == 0) continue;
        if(mx_len < mx[find_par(i)]){
            mx_len = mx[find_par(i)];
            len = find_par(i);
        }
    }
    int cnt = 0;
    for(int i = 1; i <= m; i ++){
        int u = e[i].u, v = e[i].v;
        int r1 = find_par(u), r2 = find_par(v);
        if(r1 == r2) continue;
        if(r1 != len && r2 != len) continue;
        if(r1 != len){
            swap(r1, r2);
            swap(u, v);
        }
        if(mx_len > e[i].w || u == n || v == n) continue;
        if(cnt == k){
            puts("-1");
            return 0;
        }
        ans[cnt ++] = e[i].id;
        merge(len, v);
    }
    cout << cnt << endl;
    for(int i = 0; i < cnt; i ++) cout << ans[i] << " ";
    cout << endl;
    return 0;
}
