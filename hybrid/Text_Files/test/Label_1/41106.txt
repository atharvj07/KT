from collections import deque
import sys

input = sys.stdin.readline
q = int(input())
deq = deque()
ans = []
for i in range(q):
    query = input().split()
    if query[0] == 'L':
        deq.appendleft(int(query[1]))
    elif query[0] == 'R':
        deq.append(int(query[1]))
    else:
        id = int(query[1])
        if deq[0] == id or deq[-1] == id:
            ans.append(0)
        else:
            l = 0
            r = 0
            if id < deq[0]:
                l = deq[0] - id
            if id > deq[-1]:
                r = id - deq[-1]
            else:
                for j in range(len(deq)):
                    if deq[j] == id:
                        if j != 0:
                            l = deq[j] - deq[j - 1]
                        if j != len(deq) - 1:
                            r = deq[j + 1] - deq[j]
                        break
            ans.append(min(l, r))
for a in ans:
    print(a)
