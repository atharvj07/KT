#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
  int n;
  cin >> n;
  map<long long, vector<pair<long long, long long>>> events;
  for (int i = 0; i < n; i++) {
    int type, time, value;
    cin >> type >> time >> value;
    events[time].push_back({type, value});
  }
  map<long long, long long> cnt;
  for (auto &it : events) {
    for (auto &e : it.second) {
      long long type = e.first, value = e.second;
      if (type == 1) {
        cnt[value]++;
      } else if (type == 2) {
        cnt[value]--;
      }
    }
    for (auto &e : it.second) {
      long long value = e.second;
      if (e.first == 3) {
        cout << cnt[value] << endl;
      }
    }
  }
  return 0;
}
