import java.util.*;

class Main {
    private static final int MOD = 100_000_007;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt(), c = sc.nextInt(), a1 = sc.nextInt(), a2 = sc.nextInt(), b1 = sc.nextInt(), b2 = sc.nextInt();
        int[][][] dp = new int[r][c][2];
        dp[a1][a2][0] = 1;
        for (int i = a1; i <= b1; i++) {
            for (int j = a2; j <= b2; j++) {
                dp[i][j][0] = (dp[i][j][0] + dp[i][j][1]) % MOD;
                if (i + 1 < r) dp[i + 1][j][1] = (dp[i + 1][j][1] + dp[i][j][0]) % MOD;
                if (j + 1 < c) dp[i][j + 1][1] = (dp[i][j + 1][1] + dp[i][j][0]) % MOD;
                if (i - 1 >= 0) dp[i - 1][j][1] = (dp[i - 1][j][1] + dp[i][j][0]) % MOD;
                if (j - 1 >= 0) dp[i][j - 1][1] = (dp[i][j - 1][1] + dp[i][j][0]) % MOD;
            }
        }
        System.out.println(dp[b1][b2][0]);
    }
}
