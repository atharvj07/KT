#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    int n, p;
    cin >> n >> p;
    vector<int> t(n);
    for (int i = 0; i < n; i++) {
        cin >> t[i];
    }
    vector<int> ans(n);
    vector<int> q;
    for (int i = 0; i < n; i++) {
        while (!q.empty() && q.back() + p <= t[i]) {
            q.pop_back();
        }
        if (!q.empty()) {
            ans[i] = q.back() + p;
        } else {
            ans[i] = t[i];
        }
        q.push_back(ans[i]);
    }
    for (int i = 0; i < n; i++) {
        cout << ans[i] << " ";
    }
    cout << endl;
    return 0;
}
