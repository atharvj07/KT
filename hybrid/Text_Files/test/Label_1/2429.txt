#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, d;
  cin >> n >> k >> d;
  if (k == 1 || n > k * d) {
    cout << "-1" << endl;
    return 0;
  }
  for (int i = 0; i < d; i++) {
    for (int j = 1; j <= n; j++) { cout << ((j + i) % k + 1) << " "; }
    cout << endl;
  }
  return 0;
}
