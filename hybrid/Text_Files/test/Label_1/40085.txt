n, k = map(int, input().split())
h = list(map(int, input().split()))
h.sort()
sum = 0
# Sum of all heights
for i in h:
    sum += i
ans = [0 for i in range(k+1)]
m = 0
min = 100000000
for j in range(1, k+1):
    if h[0] == h[n - 1]:
        break
    ans[j] = ans[j - 1] + 1
    x = h[0] + h[n - 1]
    h[0] = x - h[0]
    h[n - 1] = x - h[n - 1]
    h.sort()
    if sum - h[n - 1] - h[0] < min:
        min = sum - h[n - 1] - h[0]
        m = j
print(min, m)
for l in range(1, m+1):
    print(h.index(h[0]) + 1, h.index(h[n - 1]) + 1)
    x = h[0] + h[n - 1]
    h[0] = x - h[0]
    h[n - 1] = x - h[n - 1]
    h.sort()
