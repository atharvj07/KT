import heapq

n, s = map(int, input().split())
adj = [[] for i in range(n)]
for _ in range(n - 1):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    adj[a].append(b)
    adj[b].append(a)

heap = []
for i in range(n):
    if len(adj[i]) == 1:
        heapq.heappush(heap, [0, i])

sum = 0
while heap and s - sum >= 0:
    cost, node = heapq.heappop(heap)
    if cost > s - sum:
        cost = s - sum
    sum += cost
    for nei in adj[node]:
        if cost > 0:
            heapq.heappush(heap, [cost - 1, nei])

print(sum / (n - 1))
