#include <bits/stdc++.h>
using namespace std;
const int mxN = 4e5 + 5;
multiset<int> ms;
int main() {
    int n, a[mxN], m, b[mxN];
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    cin >> m;
    for (int i = 0; i < m; i++) { cin >> b[i]; }
    int last = 1, suf = 0, ans = 0;
    for (int i = 0; i < m; i++) {
        if (last < b[i]) {
            if (suf == 0)
                ans += a[b[i] - 1] - a[last - 1];
            else
                ans += a[b[i] - 1] - suf;
        }
        suf = a[b[i] - 1];
        last = b[i];
        ms.insert(b[i]);
    }
    set<int> st(b, b + m);
    if (st.size() == n)
        cout << ans;
    else
        cout << -1;
}
