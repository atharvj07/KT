t = int(input().strip())
for _ in range(t):
    n = int(input().strip())
    days = [list(map(int, input().strip().split())) for _ in range(n)]
    
    possible = "NO"
    for i in range(5):
        for j in range(i+1, 5):
            count_i = count_j = count_both = 0
            for k in range(n):
                if days[k][i] == 1 and days[k][j] == 1:
                    count_both += 1
                elif days[k][i] == 1:
                    count_i += 1
                elif days[k][j] == 1:
                    count_j += 1
            
            if count_i + count_both >= n // 2 and count_j + count_both >= n // 2 and count_i + count_j + count_both == n:
                possible = "YES"
                break
        if possible == "YES":
            break
            
    print(possible)
