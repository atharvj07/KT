import sys

n, m = map(int, input().split())
edges = [[] for _ in range(n)]
in_degree = [0] * n
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    edges[u].append(v)
    in_degree[v] += 1
queue = [i for i in range(n) if in_degree[i] == 0]
ans = 0
while queue:
    u = queue.pop(0)
    ans += 1
    for v in edges[u]:
        in_degree[v] -= 1
        if in_degree[v] == 0:
            queue.append(v)
print(ans if ans == n else -1)
