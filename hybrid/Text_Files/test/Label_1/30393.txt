import java.util.Scanner;

public class Iahubina {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            c[i] = scanner.nextInt();
        }

        boolean possible = true;
        int[] count = new int[n + 1];
        for (int ci : c) {
            count[ci]++;
            if (count[ci] > 2) {
                possible = false;
                break;
            }
        }

        if (possible) {
            for (int i = 1; i <= n; i++) {
                if (count[i] == 1) {
                    count[i - 1]--;
                    count[i + 1]--;
                }
            }
            for (int i = 1; i <= n; i++) {
                if (count[i] > 0) {
                    possible = false;
                    break;
                }
            }
        }

        System.out.println(possible ? "YES" : "NO");
    }
}
