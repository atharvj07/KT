import sys
n, m = map(int, input().split())
g = [[] for _ in range(n)]
for _ in range(m):
    a, b = map(int, sys.stdin.readline().split())
    g[a - 1].append(b - 1)
vis = [False] * n
ans = 0
st = []
for i in range(n):
    if vis[i]: continue
    q = [i]
    vis[i] = True
    while q:
        now = q.pop()
        for j in g[now]:
            if vis[j]: continue
            vis[j] = True
            q.append(j)
    ans += 1
print(ans)
