import java.util.HashMap;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] weights = new int[n];
        for (int i = 0; i < n; i++) {
            weights[i] = in.nextInt();
        }

        // Initialize the dp table
        int[] dp = new int[1 << n];
        for (int i = 0; i < (1 << n); i++) {
            dp[i] = Integer.MAX_VALUE;
        }

        // Base case: no weights
        dp[0] = 0;

        // Iterate over all subsets of weights
        for (int i = 1; i < (1 << n); i++) {
            // Get the weight of the current subset
            int weight = 0;
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) != 0) {
                    weight += weights[j];
                }
            }

            // If the weight is a power of two, then we can remove it in one step
            if (isPowerOfTwo(weight)) {
                dp[i] = 1;
                continue;
            }

            // Otherwise, we need to find the smallest subset of weights that can be removed in one step
            // and add the number of steps required to remove that subset to the current step
            for (int j = i; j > 0; j--) {
                // Get the weight of the current subset
                int subsetWeight = 0;
                for (int k = 0; k < n; k++) {
                    if ((j & (1 << k)) != 0) {
                        subsetWeight += weights[k];
                    }
                }

                // If the subset weight is a power of two, then we can remove it in one step
                if (isPowerOfTwo(subsetWeight)) {
                    dp[i] = Math.min(dp[i], dp[i - j] + 1);
                    break;
                }
            }
        }

        // Print the minimum number of steps
        System.out.println(dp[(1 << n) - 1]);
    }

    // Function to check if a number is a power of two
    private static boolean isPowerOfTwo(int n) {
        return (n & (n - 1)) == 0;
    }
}
