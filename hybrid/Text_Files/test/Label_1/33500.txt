import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int num = n - 1;
        int den = 1;
        while (num % 2 == 0 && den % 2 == 0) {
            num /= 2;
            den /= 2;
        }
        System.out.println(num + "/" + den);
    }
}
