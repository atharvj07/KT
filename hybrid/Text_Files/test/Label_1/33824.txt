import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int[] p = new int[(int) 1e5 + 1];

    public static void main(String[] args) {
        Arrays.fill(p, 2);
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }

        for (int i = 2; i <= (int) 1e5; i++) {
            if (p[i] != 2) continue;

            for (int j = i + i; j <= (int) 1e5; j += i) {
                p[j] = i;
            }
        }

        int[] f = new int[(int) 1e5 + 1];
        boolean flg = false;

        for (int i = 0; i < n; i++) {
            for (int j = 2; j <= (int) 1e5; j++) {
                if (p[a[i]] == j) {
                    if (f[j] > 0) {
                        flg = true;
                    }
                    f[j]++;
                }
            }
        }

        System.out.println(flg ? 0 : 1);
    }
}
