#include <bits/stdc++.h>

using namespace std;

const int MOD = 1e9 + 7;
const int MAXN = 1e5 + 5;

int n, m;
int par[MAXN];
int sz[MAXN];

int find_par(int u) {
    if (u == par[u]) return u;
    return par[u] = find_par(par[u]);
}

void unite(int u, int v) {
    u = find_par(u);
    v = find_par(v);
    if (u == v) return;
    if (sz[u] < sz[v]) swap(u, v);
    par[v] = u;
    sz[u] += sz[v];
}

bool ok() {
    for (int i = 1; i <= n; i++) {
        for (int j = i + 1; j <= n; j++) {
            for (int k = j + 1; k <= n; k++) {
                int x = find_par(i), y = find_par(j), z = find_par(k);
                if (x == y && y == z) continue;
                if (x == y || y == z || z == x) return false;
            }
        }
    }
    return true;
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        par[i] = i;
        sz[i] = 1;
    }
    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        if (w == 1) unite(u, v);
    }
    int ans = 1;
    for (int i = 1; i <= n; i++) {
        if (par[i] == i) {
            if (sz[i] % 2 == 1) ans = ans * 2 % MOD;
        }
    }
    if (ok()) cout << ans << endl;
    else cout << 0 << endl;
    return 0;
}
