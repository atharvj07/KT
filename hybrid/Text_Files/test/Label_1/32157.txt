import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        char[] s = in.nextLine().toCharArray();
        int n = s.length;
        int[] c = new int[n];
        for (int i = 0; i < n; i++) {
            if (s[i] == '(' || s[i] == '[') {
                c[i] = 1;
            } else {
                if (i > 0) {
                    c[i] = c[i - 1];
                }
                if (c[i] > 0 && ((s[i] == ')' && s[i - c[i]] == '(') || (s[i] == ']' && s[i - c[i]] == '['))) {
                    c[i] += c[i - c[i]];
                }
            }
        }
        int ans = 0, pos = -1;
        for (int i = 0; i < n; i++) {
            if (s[i] == '[' && c[i] > ans) {
                ans = c[i];
                pos = i;
            }
        }
        System.out.println(ans);
        for (int i = 0; i < n; i++) {
            if (i + c[i] <= pos) {
                System.out.print(s[i]);
            }
        }
        System.out.println();
    }
}
