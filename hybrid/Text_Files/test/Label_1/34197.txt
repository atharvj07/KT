import java.util.*;
class Main {
  static final int MAX_DIGITS = (int)2e5;
  static final int MOD = 1000000007;

  public static void main(String[] args) {
    Scanner reader = new Scanner(System.in);
    int n = reader.nextInt();
    int x = reader.nextInt();
    int y = reader.nextInt();
    int[] num = new int[MAX_DIGITS];
    char[] s = reader.next().toCharArray();
    for (int i = 0; i < n; i++) {
      num[i] = s[i] - '0';
    }
    int ans = minOperations(num, n, x, y);
    System.out.println(ans);
  }

  static int minOperations(int[] num, int n, int x, int y) {
    int ans = 0;
    for (int i = n - 1; i >= n - x; i--) {
      int cur = 0;
      int p = 1;
      for (int j = i; j >= 0; j--) {
        cur = (cur + num[j] * p) % y;
        p = (p * 10 % x) % y;
      }
      if (cur != 0) {
        int need = y - cur;
        ans += need;
        int pw = fastpow(10, x - 1, y);
        for (int j = i; j >= 0; j--) {
          num[j] = (num[j] + need * pw % y) % 2;
          pw = (pw * 10) % y;
        }
        break;
      }
    }
    return ans;
  }

  static int fastpow(int a, int b, int mod) {
    int ret = 1;
    while (b > 0) {
      if ((b & 1) == 1) {
        ret = (ret * a) % mod;
      }
      a = (a * a) % mod;
      b >>= 1;
    }
    return ret;
  }
}
