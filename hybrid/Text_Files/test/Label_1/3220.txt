#include <vector>
#include <cstdio>

using namespace std;

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n, m, k;
    scanf("%d %d %d", &n, &m, &k);
    int r = n % m;
    int q = n / m;
    for (int i = 0; i < k; ++i) {
      for (int j = 0; j < m; ++j) {
        int c = r ? q + 1 : q;
        vector<int> v;
        printf("%d", c);
        for (int l = 0; l < c; ++l) {
          printf(" %d", j * q + l + 1);
        }
        printf("\n");
        r--;
      }
    }
  }
  return 0;
}
