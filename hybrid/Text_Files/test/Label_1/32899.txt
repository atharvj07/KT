import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print(getWinner(scanner.nextLine(), scanner.nextLine(), scanner.nextLine()));
        scanner.close();
    }

    public static String getWinner(String f, String m, String s) {
        int fCount = (f.equals("rock") ? 1 : 0) + (f.equals("scissors") ? -1 : 0);
        int mCount = (m.equals("rock") ? 1 : 0) + (m.equals("scissors") ? -1 : 0);
        int sCount = (s.equals("rock") ? 1 : 0) + (s.equals("scissors") ? -1 : 0);
        if (fCount + mCount + sCount == 0) return "?";
        if (fCount + mCount == 0) return f;
        if (fCount + sCount == 0) return f;
        if (mCount + sCount == 0) return m;
        return "?";
    }
}
