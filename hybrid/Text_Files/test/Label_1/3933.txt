#include <bits/stdc++.h>

using namespace std;

#define ll long long

const int MAXN = 2e5 + 10;

int n, m;
int a[MAXN];
int pref[MAXN][15];
int k[15];

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        pref[i][a[i]]++;
        for (int j = 1; j <= m; j++) pref[i][j] += pref[i - 1][j];
    }
    for (int i = 1; i <= m; i++) cin >> k[i];
    int l = 1, r = 1, ans = n;
    bool f = false;
    while (l <= r && r <= n) {
        int cnt = 0;
        for (int i = 1; i <= m; i++) {
            cnt += pref[r][i] - pref[l - 1][i];
        }
        if (cnt == 0) {
            f = true;
            ans = min(ans, r - l);
            l++;
        } else {
            r++;
        }
    }
    if (!f) ans = -1;
    cout << ans << "\n";
    return 0;
}
