import java.util.Scanner;

public class MaxOnes {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 0; i < n; i++) {
            long l = scanner.nextLong();
            long r = scanner.nextLong();

            long x = findMaxOnes(l, r);
            System.out.println(x);
        }
    }

    private static long findMaxOnes(long l, long r) {
        long maxOnes = 0;
        long x = l;

        for (long i = l; i <= r; i++) {
            long ones = countOnes(i);
            if (ones > maxOnes) {
                maxOnes = ones;
                x = i;
            } else if (ones == maxOnes && i < x) {
                x = i;
            }
        }

        return x;
    }

    private static long countOnes(long n) {
        long count = 0;
        while (n > 0) {
            if ((n & 1) == 1) {
                count++;
            }
            n >>= 1;
        }
        return count;
    }
}
