def num_moves(n, m):
    if m % n != 0:
        return -1
    res = 0
    while m > n:
        if m % 2 == 0:
            m //= 2
        elif m % 3 == 0:
            m //= 3
        else:
            return -1
        res += 1
    return res

n, m = map(int, input().split())
print(num_moves(n, m))
