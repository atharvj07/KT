import java.util.Scanner;

public class CalendarReform {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int a = scanner.nextInt();
        int n = scanner.nextInt();

        long p = 0;
        for (int i = 0; i < n; i++) {
            int yearLength = a + i;
            int daysPerMonth = (int) Math.sqrt(yearLength);
            int monthsPerYear = yearLength / daysPerMonth;
            p += monthsPerYear;
        }

        System.out.println(p);
    }
}
