import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[][] c = new int[3][3];
        for (int i = 0; i < 3; ++i) {
            for (int j = 0; j < 3; ++j) {
                c[i][j] = sc.nextInt();
            }
        }
        boolean can = true;
        int a1 = c[0][0];
        for (int i = 0; i < 3; ++i) {
            if (c[i][0] != a1) {
                can = false;
                break;
            }
        }
        if (can == false) {
            System.out.println("No");
            return;
        }
        int b1 = c[0][0] - a1;
        for (int i = 0; i < 3; ++i) {
            if (c[0][i] != b1) {
                can = false;
                break;
            }
        }
        if (can == false) {
            System.out.println("No");
            return;
        }
        for (int i = 1; i < 3; ++i) {
            for (int j = 1; j < 3; ++j) {
                if (c[i][j] != c[i][0] + c[0][j] - c[0][0]) {
                    can = false;
                    break;
                }
            }
        }
        System.out.println(can ? "Yes" : "No");
    }
}
