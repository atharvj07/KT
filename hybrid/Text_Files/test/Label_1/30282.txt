import java.util.*;
import java.io.*;

public class Main {
    static class Pair {
        int min, max;

        public Pair(int min, int max) {
            this.min = min;
            this.max = max;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        List<Pair> ranges = new ArrayList<>();
        for (int i = 0; i < n;) {
            int j = i + 1;
            while (j < n && a[j] - a[i] <= 5) {
                j++;
            }
            ranges.add(new Pair(a[i], a[j - 1]));
            i = j;
        }
        int ans = 0;
        for (int i = 0; i < ranges.size(); i++) {
            int sz = ranges.get(i).max - ranges.get(i).min + 1;
            int teamCnt = (sz + 4) / 5;
            ans += Math.min(teamCnt, k);
            k -= Math.min(teamCnt, k);
        }
        System.out.println(ans);
    }
}
