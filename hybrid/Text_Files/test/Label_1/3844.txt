#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n, z;
    cin >> n >> z;
    vector<int> v(n);
    for (auto &x: v)
        cin >> x;
    sort(v.begin(), v.end());
    int ans = 0;
    for (int i = 0, j = 0; i < n; i = j) {
        while (j < n && v[j] - v[i] < z)
            j++;
        if (j < n)
            ans++;
        else
            break;
    }
    cout << ans << '\n';
    return 0;
}
