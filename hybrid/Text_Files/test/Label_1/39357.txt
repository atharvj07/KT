from collections import defaultdict, Counter

def find_largest_value(n, m, s, edges):
    graph = defaultdict(list)
    for x, y in edges:
        graph[x].append(y)
    
    def dfs(node, counter, visited):
        if visited[node]:
            if counter[s[node-1]] > 0:
                return float('inf')
            else:
                return 0
        visited[node] = True
        counter[s[node-1]] += 1
        max_count = counter[s[node-1]]
        for nei in graph[node]:
            max_count = max(max_count, dfs(nei, counter, visited))
        counter[s[node-1]] -= 1
        visited[node] = False
        return max_count
    
    max_value = 0
    for node in range(1, n + 1):
        counter = Counter()
        visited = [False] * (n + 1)
        max_value = max(max_value, dfs(node, counter, visited))
        if max_value == float('inf'):
            return -1
    return max_value

n, m = map(int, input().split())
s = input()
edges = [tuple(map(int, input().split())) for _ in range(m)]

print(find_largest_value(n, m, s, edges))
