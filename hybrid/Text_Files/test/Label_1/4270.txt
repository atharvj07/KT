#include <bits/stdc++.h>
#define endl '\n'
#define ll long long
using namespace std;
const int mxn = 105;
vector<int> g[mxn];
bool vis[mxn];
int match[mxn];
int n, m;
bool dfs(int v) {
    if (vis[v]) return false;
    vis[v] = true;
    for (int i : g[v]) {
        if (match[i] == -1 || dfs(match[i])) {
            match[v] = i;
            match[i] = v;
            return true;
        }
    }
    return false;
}
int main() {
    ios_base::sync_with_stdio(false); cin.tie(NULL);
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        int k; cin >> k;
        while (k--) {
            int x; cin >> x;
            g[i].push_back(x);
        }
    }
    memset(match, -1, sizeof match);
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        memset(vis, false, sizeof vis);
        ans += dfs(i);
    }
    cout << ans << endl;
}
