#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, d, m;
    cin >> n >> d >> m;
    vector<int> a(n);
    for (int &x : a) cin >> x;
    sort(a.begin(), a.end());
    int sum = 0;
    for (int i = 0; i < m && i < n; ++i) sum += a[i];
    if (m <= n) cout << sum << endl;
    else cout << sum - d * (m - n) << endl;
}
