#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
  int n, h;
  cin >> n >> h;
  vector<int> a(n);
  for (int i = 0; i < n; ++i)
    cin >> a[i];
  int ans = n;
  for (int i = 0; i < n; ++i)
    if (a[i] > h)
      ++ans;
  cout << ans << endl;
  return 0;
}
