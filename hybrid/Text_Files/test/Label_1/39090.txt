import sys

def solve():
    t = int(input())
    for _ in range(t):
        k = int(input())
        n = input()
        cnt = 0
        lst = []
        for i in range(k):
            if n[i] != '1':
                cnt += 1
                lst.append(n[i])
        if cnt == 0:
            cnt = 1
            lst.append('1')
        print(cnt)
        print(''.join(lst))

solve()
