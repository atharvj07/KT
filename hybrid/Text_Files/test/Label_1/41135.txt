n, m = map(int, input().split())
s = input()

# Calculate prefix and suffix LCS counts
prefix_lcs = [0] * n
suffix_lcs = [0] * n
prefix_lcs[0] = 1
suffix_lcs[-1] = 1

for i in range(1, n):
    prefix_lcs[i] = prefix_lcs[i - 1] + 1 if s[i] != s[i - 1] else 1
for i in range(n - 2, -1, -1):
    suffix_lcs[i] = suffix_lcs[i + 1] + 1 if s[i] != s[i + 1] else 1

# Calculate the total number of different strings T
result = (m - 1) * (prefix_lcs[n - 1] + suffix_lcs[0] - 1)
for i in range(1, n):
    result += (m - 1) * (1 if s[i] != s[i - 1] else min(prefix_lcs[i - 1], suffix_lcs[i]) + 1)

print(result)
