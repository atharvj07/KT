def count_pretty_substrings(s):
    n = len(s)
    dp = [[0]*n for _ in range(n)]
    for length in range(2, n+1, 2):
        for i in range(n-length+1):
            j = i + length - 1
            if length == 2:
                dp[i][j] = int(s[i] != ')' and s[j] != '(')
            else:
                dp[i][j] = dp[i+1][j-1] and s[i] != ')' and s[j] != '('
                for k in range(i, j):
                    dp[i][j] |= dp[i][k] and dp[k+1][j]
    return sum(sum(row) for row in dp)

s = input().strip()
print(count_pretty_substrings(s))
