from collections import defaultdict, deque

def solve(n, m, edges):
    graph = defaultdict(list)
    for a, b in edges:
        graph[a-1].append(b-1)
        graph[b-1].append(a-1)

    visited = [False] * n
    group = [-1] * n

    def bfs(start):
        queue = deque([start])
        visited[start] = True
        group[start] = 1
        while queue:
            v = queue.popleft()
            for u in graph[v]:
                if not visited[u]:
                    visited[u] = True
                    group[u] = 3 - group[v]
                    queue.append(u)
                elif group[u] == group[v]:
                    return False
        return True

    for i in range(n):
        if not visited[i]:
            if not bfs(i):
                print(-1)
                return

    vertex_set = defaultdict(set)
    for i, g in enumerate(group):
        vertex_set[g].add(i)

    if len(vertex_set) != 2:
        print(-1)
        return

    v1, v2 = list(vertex_set.keys())
    
    for i in range(n):
        if group[i] == v1:
            print(1, end=' ')
        elif group[i] == v2:
            print(2, end=' ')
        else:
            print(3, end=' ')
    print()

n, m = map(int, input().split())
edges = [list(map(int, input().split())) for _ in range(m)]
solve(n, m, edges)
