import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))

mn = []
for i in range(n):
    if i > 0 and a[i] == a[i - 1]:
        mn[-1][1] += 2
    else:
        mn.append([i + 1, 2])

res = []
for i, j in mn:
    for k in range(i + 1, n, j):
        if a[k] != a[i - 1]:
            res.append([i, j])
            break

print(len(res))
for i, j in res:
    print(i, j)
