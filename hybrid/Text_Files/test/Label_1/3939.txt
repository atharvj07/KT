#include <bits/stdc++.h>
#define ll long long
using namespace std;

int main() {
    int m, b;
    cin >> m >> b;
    ll ans = 0;
    for (int lx = 0;; lx++) {
        ll ly = -lx / m + b;
        if (ly <= lx) break;
        ans += (ly + 1) * (ly + 2) / 2 + (lx + 1) * lx / 2;
    }
    cout << ans << endl;
    return 0;
}
