import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
  public static void main(String[] args) throws IOException {
    try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
      int n = Integer.parseInt(br.readLine());
      List<Integer> events = new ArrayList<>();
      int currentSpeed = 0;
      int overtakeCnt = 0;
      int answer = 0;
      for (int i = 0; i < n; i++) {
        StringTokenizer st = new StringTokenizer(br.readLine());
        int t = Integer.parseInt(st.nextToken());
        if (t == 1) {
          int speed = Integer.parseInt(st.nextToken());
          if (!events.isEmpty() && events.get(events.size() - 1) == 3) {
            if (speed > currentSpeed) {
              continue;
            } else {
              overtakeCnt = 0;
              events.add(1);
              answer++;
            }
          } else {
            overtakeCnt = 0;
            events.add(1);
          }
          currentSpeed = speed;
        } else if (t == 2) {
          if (!events.isEmpty() && events.get(events.size() - 1) == 5) {
            events.add(2);
          } else if (overtakeCnt > 0) {
            events.add(2);
          } else {
            answer++;
            events.add(2);
          }
          overtakeCnt--;
        } else if (t == 3) {
          int speedLimit = Integer.parseInt(st.nextToken());
          if (!events.isEmpty() && events.get(events.size() - 1) == 5) {
            if (speedLimit > currentSpeed) {
              continue;
            } else {
              answer++;
              overtakeCnt = 0;
              events.add(3);
            }
          } else {
            answer++;
            overtakeCnt = 0;
            events.add(3);
          }
          currentSpeed = speedLimit;
        } else if (t == 4) {
          if (!events.isEmpty() && events.get(events.size() - 1) == 6) {
            continue;
          } else {
            overtakeCnt++;
            events.add(4);
          }
        } else if (t == 5) {
          events.add(5);
          currentSpeed = -1;
        } else if (t == 6) {
          if (!events.isEmpty() && events.get(events.size() - 1) == 4) {
            answer++;
            events.add(6);
          } else {
            events.add(6);
          }
        }
      }
      System.out.println(answer);
    }
  }
}
