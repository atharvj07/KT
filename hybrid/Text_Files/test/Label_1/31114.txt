import java.util.Scanner;

public class WealthyPeople {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int testCases = scanner.nextInt();

        for (int i = 0; i < testCases; i++) {
            int n = scanner.nextInt();
            int x = scanner.nextInt();
            long[] wealth = new long[n];

            for (int j = 0; j < n; j++) {
                wealth[j] = scanner.nextLong();
            }

            System.out.println(calculateWealthyPeople(wealth, x));
        }
    }

    private static long calculateWealthyPeople(long[] wealth, long x) {
        long totalWealth = 0;
        long wealthyPeople = 0;

        for (long w : wealth) {
            totalWealth += w;
        }

        if (totalWealth < x) {
            return 0;
        }

        wealthyPeople = totalWealth / x;

        return wealthyPeople;
    }
}
