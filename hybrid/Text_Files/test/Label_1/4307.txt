#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<pair<int, int>> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i].first >> v[i].second;
    }
    int rating = 1900;
    bool possible = true;
    for (int i = 0; i < n; i++) {
        if (rating + v[i].first < 1900 && v[i].second == 1) {
            possible = false;
        }
        if (rating + v[i].first >= 1900 && v[i].second == 2) {
            possible = false;
        }
        rating += v[i].first;
    }
    if (!possible) {
        cout << "Impossible" << endl;
    } else if (rating > 1900) {
        cout << "Infinity" << endl;
    } else {
        cout << rating << endl;
    }
    return 0;
}
