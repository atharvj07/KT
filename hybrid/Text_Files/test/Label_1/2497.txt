#include <bits/stdc++.h>
using namespace std;

int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m, k;
  cin >> n >> m >> k;
  vector<int> p(n), dp(n + 1, 0);
  for (int i = 0; i < n; i++) cin >> p[i];
  sort(p.begin(), p.end());
  // dp[i]: max sum of subarray of length i, ending at i
  for (int i = 1; i <= n; i++) dp[i] = max(dp[i - 1], dp[i - 1] + p[i - 1]);
  int ans = 0;
  for (int i = 1; i <= n; i++) {
    if (i - m >= 0) ans = max(ans, dp[i] - dp[i - m]);
    if (k >= 2) {
      for (int j = i - m; j > max(0, i - m - 1); j--) {
        ans = max(ans, dp[j] + dp[i] - dp[j + 1]);
        if (k >= 3) {
          for (int l = j - m; l > max(0, j - m - 1); l--) {
            ans = max(ans, dp[l] + dp[j] + dp[i] - dp[l + 1]);
          }
        }
      }
    }
  }
  cout << ans << '\n';
  return 0;
}
