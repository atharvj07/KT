import java.util.Scanner;
import java.util.Arrays;

class Main {
  static int minSwaps(int[] p, int n, int k) {
    int[] cp = Arrays.copyOf(p, n);
    Arrays.sort(cp, 0, k);
    int ans = 0;
    for (int i=0; i<k; i++) {
      if (cp[i] != p[i]) ans++;
    }
    return ans;
  }
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] p = new int[n];
    for (int i=0; i<n; i++) {
      p[i] = sc.nextInt();
    }
    for (int i=1; i<=n; i++) {
      System.out.println(minSwaps(p, n, i));
    }
  }
}
