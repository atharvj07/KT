#include <iostream>
#include <vector>
int n, m;
std::vector<std::vector<int>> a;

int main() {
  std::ios_base::sync_with_stdio(false);
  std::cin >> n >> m;
  a.resize(n, std::vector<int>(m));
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j) std::cin >> a[i][j];
  for (int j = 0; j < m; ++j) {
    int cnt0 = 0, cnt1 = 0;
    for (int i = 0; i < n; ++i) {
      cnt0 += a[i][j] == 0;
      cnt1 += a[i][j] == 1;
    }
    if (cnt0 > cnt1)
      for (int i = 0; i < n; ++i)
        a[i][j] ^= 1;
  }
  for (int i = 0; i < n; ++i) {
    int cnt0 = 0, cnt1 = 0;
    for (int j = 0; j < m; ++j) {
      cnt0 += a[i][j] == 0;
      cnt1 += a[i][j] == 1;
    }
    if (cnt0 > cnt1)
      for (int j = 0; j < m; ++j)
        a[i][j] ^= 1;
  }
  for (int i = 0; i < n; ++i)
    for (int j = 0; j < m; ++j)
      if ((i > 0 && a[i][j] < a[i - 1][j]) ||
          (j > 0 && a[i][j] < a[i][j - 1])) {
        std::cout << "NO" << std::endl;
        return 0;
      }
  std::cout << "YES" << std::endl;
  std::string r, c;
  for (int i = 0; i < n; ++i) r += '0' + (a[i][0] != a[0][0]);
  for (int j = 0; j < m; ++j) c += '0' + (a[0][j] != a[0][0]);
  std::cout << r << std::endl << c << std::endl;
  return 0;
}
