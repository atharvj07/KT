import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        List<String> files = new ArrayList<>();
        List<String> examples = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            String file = sc.next();
            int type = sc.nextInt();
            files.add(file);
            if (type == 1) {
                examples.add(file);
            }
        }
        System.out.println(files.size() - examples.size());
        int j = examples.size() + 1;
        for (String file : examples) {
            System.out.println("move " + file + " " + j++);
        }
        for (String file : files) {
            if (!examples.contains(file)) {
                System.out.println("move " + file + " " + j++);
            }
        }
    }
}
