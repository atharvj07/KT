import sys

n=int(input())
vows=[0]*n
ends=[0]*n
words=[]
for i in range(n):
  s=input()
  words.append(s)
  lv=0
  le=0
  for c in s:
    if c in 'aeiou':
      lv+=1
      le=c
  vows[i]=lv
  ends[i]=le

cnt=0

for i in range(n):
  for j in range(n):
    if i!=j and vows[i]==vows[j] and ends[i]==ends[j]:
      cnt+=1
      print(words[j],words[i])
      print(words[i],words[j])

print(cnt//2)
