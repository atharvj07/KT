import sys

N = 501
deg = [0] * N

def dfs(cur, p, dist):
    ans[dist] += 1
    for i in edges[cur]:
        if i == p: continue
        dfs(i, cur, dist + 1)

n = int(input())
array = tuple(int(x) for x in input().split())
for i in array:
    deg[i] += 1

edges = [[]]
for i in range(1, n + 1):
    edges += [[]]

impossible = False
for i in range(1, n + 1):
    if deg[i] > i - 1:
        impossible = True

if impossible:
    print('NO')
    sys.exit()

for i in range(1, n):
    for j in range(i + 1, n + 1):
        if deg[j] > 0:
            deg[j] -= 1
            edges[i] += [j]
            edges[j] += [i]
            break

ans = [0] * N

dfs(1, 1, 0)

diameter = 0
for i in range(N):
    diameter = max(diameter, i)
    if ans[diameter] > 1: break

print('YES', diameter)
print(sum(deg) // 2)

for i in range(1, n + 1):
    for j in edges[i]:
        if j > i:
            print(i, j)
