#include <iostream>
#include <vector>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    if (k > n - k) {
        cout << -1;
        return 0;
    }
    vector<int> a(n + 2, 0);
    for (int i = 2; i <= n; i += 2) a[i] = i;
    int cnt = 0;
    for (int i = n; i > 1; i--) {
        if (cnt == k) break;
        if (a[i] == 0) {
            for (int j = 1; j <= n; j += 2) {
                if (a[j] == 0 && i % j == 0) {
                    a[i] = j;
                    cnt++;
                    break;
                }
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        if (a[i] == 0) a[i] = i;
    }
    for (int i = 1; i <= n; i++) {
        cout << a[i] << ' ';
    }
    cout << endl;
    return 0;
}
