#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vi = vector<int>;
const int MAXN = 1e5;
int n, m;
vector<int> adj[MAXN + 1];
unordered_map<int, vi> color_graph;
vi color(MAXN + 1);

int solve() {
    for (int i = 1; i <= n; i++) {
        color_graph[color[i]].push_back(i);
    }
    map<int, int> color_diversity;
    for (auto [c, vertex_list] : color_graph) {
        unordered_set<int> colors;
        for (int v : vertex_list) {
            for (int u : adj[v]) {
                colors.insert(color[u]);
            }
        }
        color_diversity[c] = colors.size();
    }
    int max_diversity = INT_MIN;
    int best_color = -1;
    for (auto [c, diversity] : color_diversity) {
        if (diversity > max_diversity) {
            max_diversity = diversity;
            best_color = c;
        }
    }
    return best_color;
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        int c;
        cin >> c;
        color[i] = c;
    }
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }
    cout << solve() << endl;
    return 0;
}
