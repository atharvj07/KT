import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int n = input.nextInt();
            if (n == 0) break;
            int[][] lines = new int[n][];
            for (int i = 0; i < n; i++) {
                int s = input.nextInt();
                lines[i] = new int[s * 2];
                for (int j = 0; j < s * 2; j++) {
                    lines[i][j] = input.nextInt();
                }
            }
            System.out.println(solve(lines));
        }
    }

    private static int solve(int[][] lines) {
        int minX = Integer.MAX_VALUE;
        int maxX = Integer.MIN_VALUE;
        int minY = Integer.MAX_VALUE;
        int maxY = Integer.MIN_VALUE;
        for (int[] line : lines) {
            for (int i = 0; i < line.length; i += 2) {
                minX = Math.min(minX, line[i]);
                maxX = Math.max(maxX, line[i]);
                minY = Math.min(minY, line[i + 1]);
                maxY = Math.max(maxY, line[i + 1]);
            }
        }
        return (maxY - minY) / 10 + ((maxY - minY) % 10 > 0 ? 1 : 0);
    }
}
