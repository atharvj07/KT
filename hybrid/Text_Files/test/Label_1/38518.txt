def max_binary_after_deletion(binary_str):
    if '0' in binary_str:
        # Remove the first occurrence of '0'
        index_to_remove = binary_str.index('0')
    else:
        # If no '0', remove the last digit
        index_to_remove = len(binary_str) - 1
    return binary_str[:index_to_remove] + binary_str[index_to_remove + 1:]

# Reading input
a = input().strip()
print(max_binary_after_deletion(a))
