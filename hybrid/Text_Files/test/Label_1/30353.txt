import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        int[] last = new int[200001];
        Arrays.fill(last, -1);
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            last[a[i]] = i;
        }
        int ans = 0;
        int max = -1;
        for (int i = 0; i < n; i++) {
            if (last[a[i]] < max) {
                ans = a[i];
            }
            max = Math.max(max, last[a[i]]);
        }
        System.out.println(ans);
    }
}
