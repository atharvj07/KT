import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static int[][] heights;
    private static int[] counts;
    private static List<String> names;
    private static Map<String, Integer> nameToIndex;

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        heights = new int[n][n];
        counts = new int[n];
        names = new ArrayList<>();
        nameToIndex = new HashMap<>();
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            String name = line[0];
            int count = Integer.parseInt(line[1]);
            names.add(name);
            nameToIndex.put(name, i);
            counts[i] = count;
        }
        if (!check()) {
            System.out.println(-1);
            return;
        }
        int[] heights = restoreHeights();
        for (int i = 0; i < n; i++) {
            System.out.println(names.get(i) + " " + heights[i]);
        }
    }

    private static boolean check() {
        int[] sortedCounts = counts.clone();
        Arrays.sort(sortedCounts);
        for (int i = 0; i < counts.length; i++) {
            if (counts[i] != sortedCounts[i]) {
                return false;
            }
        }
        return true;
    }

    private static int[] restoreHeights() {
        int[] heights = new int[names.size()];
        List<Integer> sortedCounts = new ArrayList<>();
        for (int count : counts) {
            sortedCounts.add(count);
        }
        Collections.sort(sortedCounts);
        Map<Integer, Integer> countToHeight = new HashMap<>();
        for (int i = 0; i < sortedCounts.size(); i++) {
            countToHeight.put(sortedCounts.get(i), i + 1);
        }
        for (int i = 0; i < names.size(); i++) {
            heights[i] = countToHeight.get(counts[i]);
        }
        return heights;
    }
}
