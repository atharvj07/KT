import java.util.*;

class Main {

  static int a, b, n, m, i, j;
  static int[][] dp = new int[107][107];

  static int solve() {
    dp[i][j] = -1;
    Queue<Integer> qx = new LinkedList<>();
    Queue<Integer> qy = new LinkedList<>();
    qx.add(i); qy.add(j);
    dp[i][j] = 0;
    while (!qx.isEmpty()) {
      int x = qx.poll();
      int y = qy.poll();
      for (int dx = -a; dx <= a; ++dx) {
        for (int dy = -b; dy <= b; ++dy) {
          if (Math.abs(dx) + Math.abs(dy) != a + b) continue;
          int nx = x + dx, ny = y + dy;
          if (nx >= 1 && ny >= 1 && nx <= n && ny <= m) {
            if (dp[nx][ny] == -1 || dp[nx][ny] > dp[x][y] + 1) {
              dp[nx][ny] = dp[x][y] + 1;
              qx.add(nx); qy.add(ny);
            }
          }
        }
      }
    }
    return dp[n][m] == -1 ? -1 : dp[n][m];
  }

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    n = sc.nextInt(); m = sc.nextInt(); i = sc.nextInt(); j = sc.nextInt();
    a = sc.nextInt(); b = sc.nextInt();
    for (int[] x : dp) Arrays.fill(x, -1);
    int res = solve();
    System.out.println(res == -1 ? "Poor Inna and pony!" : res);
  }
}
