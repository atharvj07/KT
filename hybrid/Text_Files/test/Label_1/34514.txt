import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        long x = scan.nextLong();
        int distress = 0;
        while (n-- > 0) {
            char c = scan.next().charAt(0);
            long d = scan.nextLong();
            if (c == '+') {
                x += d;
            } else if (x >= d) {
                x -= d;
            } else {
                distress++;
            }
        }
        System.out.println(x + " " + distress);
    }
}
