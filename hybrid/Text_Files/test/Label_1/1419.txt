#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, d, ans = 0, sum = 0, x;
    cin >> n >> d;
    sum += 5;
    for (int i = 0; i < n; i++) {
        cin >> x;
        sum += x;
        if (sum > d) {
            ans = -1;
            break;
        }
        sum += 10;
    }
    ans = (d - sum) / 5;
    cout << ans << endl;
    return 0;
}
