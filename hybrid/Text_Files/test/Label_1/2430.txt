#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, m;
  cin >> n >> m;
  int p[n];
  for (int i = 0; i < n; i++) cin >> p[i];
  map<int, vector<int>> q;
  for (int i = 0; i < m; i++) {
    int u, v;
    cin >> u >> v;
    q[u].push_back(v);
    q[v].push_back(u);
  }
  int ans = 0;
  for (auto it : q)
    if (it.first == p[n]) ans = max(ans, (int)it.second.size());
  cout << ans << endl;
  return 0;
}
