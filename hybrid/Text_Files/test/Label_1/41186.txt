n = int(input())
L = [list(map(int, input().split())) for i in range(n)]
L.sort()
S = []
L2 = []
ok = True
for l, r in L:
    if S and S[-1] == l:
        S.pop()
        L2.pop()
    else:
        S.append(r)
        L2.append((l,r))
        if len(S) > 1 or (len(S) == 1 and S[0] != 1):
            ok = False
print('IMPOSSIBLE' if not ok else ''.join(['(']*len(L)+[')']*len(L)))
