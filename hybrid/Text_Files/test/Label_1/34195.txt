import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'o') {
                count++;
            } else {
                if (count >= 2) {
                    System.out.println(count);
                    return;
                } else {
                    count = 0;
                }
            }
        }
        System.out.println(count);
    }
}
