import java.util.PriorityQueue;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        Medians medians = new Medians();
        for (int i = 0; i < n; i++) {
            String op = in.next();
            if (op.equals("add")) {
                medians.add(in.nextInt());
            } else if (op.equals("del")) {
                medians.remove(in.nextInt());
            } else if (op.equals("sum")) {
                System.out.println(medians.getSum());
            }
        }
    }

    private static class Medians {
        private PriorityQueue<Integer> minHeap = new PriorityQueue<>();
        private PriorityQueue<Integer> maxHeap = new PriorityQueue<>((a, b) -> b - a);

        public void add(int x) {
            if (minHeap.isEmpty() || x <= minHeap.peek()) {
                minHeap.add(x);
            } else {
                maxHeap.add(x);
            }
            balanceHeaps();
        }

        public void remove(int x) {
            if (minHeap.contains(x)) {
                minHeap.remove(x);
            } else {
                maxHeap.remove(x);
            }
            balanceHeaps();
        }

        public long getSum() {
            int count = minHeap.size() + maxHeap.size();
            if (count == 0) {
                return 0;
            } else if (count % 2 == 0) {
                return minHeap.peek() + maxHeap.peek();
            } else {
                return minHeap.peek();
            }
        }

        private void balanceHeaps() {
            while (minHeap.size() > maxHeap.size() + 1) {
                maxHeap.add(minHeap.poll());
            }
            while (maxHeap.size() > minHeap.size()) {
                minHeap.add(maxHeap.poll());
            }
        }
    }
}
