import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

class Main {
    public static void main(String[] args) throws IOException {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            String[] tokens = br.readLine().split(" ");
            int[] trees = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                trees[i] = Integer.parseInt(tokens[i - 1]);
            }
            int[] pref = new int[n + 1];
            for (int i = 1; i <= n; i++) {
                pref[i] = pref[i - 1] + trees[i];
            }
            int l = 1, r = 2, ansL = 1, ansR = 2, maxSum = trees[1] + trees[2];
            for (int i = 2; i <= n - 1; i++) {
                for (int j = i + 1; j <= n; j++) {
                    int sum = pref[j] - pref[i - 1];
                    if (sum > maxSum || (sum == maxSum && j - i < r - l)) {
                        maxSum = sum;
                        ansL = i;
                        ansR = j;
                    }
                }
            }
            if (ansL == ansR) {
                ansL++;
            }
            int k = n - (ansR - ansL + 1);
            System.out.println(maxSum + " " + k);
            List<Integer> ans = new ArrayList<>();
            for (int i = 1; i < ansL; i++) {
                ans.add(i);
            }
            for (int i = ansR + 1; i <= n; i++) {
                ans.add(i);
            }
            for (int i : ans) {
                System.out.print(i + " ");
            }
            System.out.println();
        }
    }
}
