from sys import stdin
n = int(stdin.readline())
a = []
for i in range(n):
    aa, x = map(int, stdin.readline().split())
    a.append([aa, x])
m = int(stdin.readline())
b = []
for i in range(m):
    bb, y = map(int, stdin.readline().split())
    b.append([bb, y])
a.sort(key = lambda x: x[1], reverse = True)
b.sort(key = lambda x: x[1], reverse = True)
i = 0
j = 0
mn = 0
while i < n and j < m:
    if a[i][0] == b[j][0]:
        if a[i][1] > b[j][1]:
            mn += a[i][1]
            i += 1
        else:
            mn += b[j][1]
            j += 1
    else:
        mn += a[i][1] + b[j][1]
        i += 1
        j += 1
while i < n:
    mn += a[i][1]
    i += 1
while j < m:
    mn += b[j][1]
    j += 1
print(mn)
