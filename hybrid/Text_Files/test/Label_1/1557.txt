#include <bits/stdc++.h>
using namespace std;

int main() {
    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];
        if (n == 2) {
            cout << 0 << '\n';
        } else {
            int mn = *min_element(a.begin(), a.end());
            int mx = *max_element(a.begin(), a.end());
            if (mn == mx) cout << 0 << '\n';
            else if (mn == mx - 1) cout << 1 << '\n';
            else if (mn == mx - 2) {
                if (a[n - 1] == mx) {
                    int cnt = 0;
                    for (int i = n - 2; i >= 0; i--) {
                        if (a[i] == mx) cnt++;
                        else break;
                    }
                    if (cnt >= 2) cout << 1 << '\n';
                    else cout << 2 << '\n';
                } else cout << 2 << '\n';
            } else {
                int cnt = 0;
                for (int i = n - 1; i >= 0; i--) {
                    if (a[i] == mx) cnt++;
                    else break;
                }
                if (cnt >= 3) cout << 1 << '\n';
                else cout << 2 << '\n';
            }
        }
    }
    return 0;
}
