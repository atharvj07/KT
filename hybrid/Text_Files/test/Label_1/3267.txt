#include <iostream>

typedef long long ll;

int main() {
  ll t;
  std::cin >> t;
  while (t--) {
    ll n, k, l1, r1, l2, r2, ans, L[2], R[2];
    std::cin >> n >> k >> l1 >> r1 >> l2 >> r2;
    ll l = std::max(l1, l2), r = std::min(r1, r2);
    ll i = std::max(l, k);
    ans = i - std::max(l, std::min(k, r));
    ans += (std::max(l1, l2) > std::max(r1, r2)) * (k - ans % 2);
    std::cout << ans << "\n";
  }
  return 0;
}
