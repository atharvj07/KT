import sys

n, q = map(int, input().split())
arr = list(range(n))
for _ in range(q):
    op = list(map(int, input().split()))
    if op[0] == 1:
        y = op[1]
        arr[y:n], arr[:y] = arr[:y], arr[y:n]
    else:
        L, R = op[1], op[2]
        print(arr[R] - arr[L - 1])
        sys.stdout.flush()
