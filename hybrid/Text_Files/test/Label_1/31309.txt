import java.io.*;
import java.util.*;

public class Main {
    static int n, m, ans, l[], r[];
    static boolean vis[];
    static ArrayList<Edge>[] adj;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());
        l = new int[m + 1];
        r = new int[m + 1];
        vis = new boolean[n + 1];
        adj = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++)
            adj[i] = new ArrayList<>();
        for (int i = 1; i <= m; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            l[i] = Integer.parseInt(st.nextToken());
            r[i] = Integer.parseInt(st.nextToken());
            adj[a].add(new Edge(b, i));
            adj[b].add(new Edge(a, i));
        }
        dfs(1, 0, 0);
        System.out.println(ans == 0 ? "Nice work, Dima!" : ans);
    }

    static void dfs(int node, int x, int sum) {
        if (node == n) {
            ans = Math.max(ans, sum);
            return;
        }
        vis[node] = true;
        for (Edge e : adj[node]) {
            if (!vis[e.node] && l[e.id] <= x && x <= r[e.id])
                dfs(e.node, x, sum + 1);
        }
        vis[node] = false;
    }

    static class Edge {
        int node, id;

        public Edge(int node, int id) {
            this.node = node;
            this.id = id;
        }
    }
}
