from collections import defaultdict
import sys

n = int(input())
b = list(map(int, input().split()))
a = list(map(int, input().split()))

adj = defaultdict(list)
for _ in range(n - 1):
    x, k = map(int, input().split())
    adj[x].append(k)
    adj[k].append(x)

def dfs(u, p, rem):
    if u == 1:
        return rem >= 0

    for v in adj[u]:
        if v == p:
            continue

        if dfs(v, u, rem - b[v - 1] + a[v - 1]):
            return True

    return False

print("YES" if dfs(n, -1, sum(b) - sum(a)) else "NO")
