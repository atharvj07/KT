import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] a = new int[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                a[i][j] = s.charAt(j) - '0';
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int cnt = 0;
            for (int j = 0; j < m; j++) {
                cnt += a[i][j];
            }
            if (cnt > m - cnt) {
                ans += cnt;
                for (int j = 0; j < m; j++) {
                    a[i][j] = 1 - a[i][j];
                }
            } else {
                ans += m - cnt;
            }
        }
        for (int j = 0; j < m; j++) {
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                cnt += a[i][j];
            }
            if (cnt > n - cnt) {
                ans += cnt;
                for (int i = 0; i < n; i++) {
                    a[i][j] = 1 - a[i][j];
                }
            } else {
                ans += n - cnt;
            }
        }
        System.out.println(ans);
    }
}
