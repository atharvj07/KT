import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] clocks = new int[n+1][2];
        for (int i = 1; i <= n; ++i) {
            clocks[i][0] = sc.nextInt();
            clocks[i][1] = sc.nextInt();
        }
        Arrays.sort(clocks, (a, b) -> a[0] - b[0]);
        int x[] = new int[n+1];
        int cnt = 0, cur = -1;
        for (int i = 1; i <= n; ++i) {
            if (clocks[i][0] != cur) {
                ++cnt;
                cur = clocks[i][0];
                x[cnt] = clocks[i][1];
            } else {
                x[cnt] = Math.max(x[cnt], clocks[i][1]);
            }
        }
        Arrays.sort(clocks, (a, b) -> a[1] - b[1]);
        cnt = 0;
        cur = -1;
        for (int i = 1; i <= n; ++i) {
            if (clocks[i][1] != cur) {
                ++cnt;
                cur = clocks[i][1];
                x[cnt] = clocks[i][0];
            } else {
                x[cnt] = Math.max(x[cnt], clocks[i][0]);
            }
        }
        int ans = x.length - 1;
        System.out.println(ans);
    }
}
