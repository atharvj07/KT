from sys import stdin
input = stdin.readline

n = int(input())
a, b = [], []
for _ in range(n):
    x, c = map(str, input().split())
    x = int(x)
    if c == 'B':
        b.append(x)
    if c == 'R':
        a.append(x)

b.sort()
a.sort()
ans = 0

if len(b) != 0:
    for i in range(len(b) - 1):
        ans += b[i + 1] - b[i]
if len(a) != 0:
    for i in range(len(a) - 1):
        ans += a[i + 1] - a[i]

print(ans)
