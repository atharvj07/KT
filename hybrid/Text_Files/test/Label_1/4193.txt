#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int a[N], freq[11];
int main() {
  int n;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  int l = 1, r = 1;
  int ans = 0;
  memset(freq, 0, sizeof freq);
  while (1) {
    if (r > n) break;
    if (freq[a[r]] == 0) {
      ans = max(ans, r - l + 1);
      freq[a[r]]++;
      r++;
    } else if (freq[a[r]] == 1) {
      ans = max(ans, r - l);
      freq[a[r]]++;
      r++;
    } else {
      while (freq[a[r]] >= 2) {
        freq[a[l]]--;
        l++;
      }
      freq[a[r]]++;
      r++;
    }
  }
  cout << ans << endl;
  return 0;
}
