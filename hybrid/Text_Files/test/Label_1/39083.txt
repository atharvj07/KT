import sys
n, t, k = map(int, input().split())
a = list(map(int, input().split()))
if sum(a) != n - 1 or a[0] != n - 1 - k:
    print(-1)
    sys.exit()
vis = [0] * (n + 1)
edges = []
for i in range(1, t + 1):
    for j in range(a[i - 1]):
        edges.append([i + 1, n - sum(a[i:]) + j])
        vis[n - sum(a[i:]) + j] = 1
if sum(vis) != n - 1:
    print(-1)
    sys.exit()
print(n)
for edge in edges:
    print(*edge)
