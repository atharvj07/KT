import java.util.*;
import java.math.*;

public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    double[] ab = new double[4];
    for (int i = 0; i < 4; i++) {
      ab[i] = scan.nextDouble();
    }
    int n = scan.nextInt();
    scan.nextLine();
    double[][] vals = new double[n][4];
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 4; j++) {
        vals[i][j] = scan.nextDouble();
      }
      scan.nextLine();
    }
    int k = scan.nextInt();
    double l = 0;
    double r = 1e9;
    for (int t = 0; t < 100; t++) {
      double m = (l + r) / 2.0;
      int cnt = 0;
      for (int i = 0; i < n; i++) {
        double a = vals[i][2];
        double w = vals[i][3];
        double xb = vals[i][0] - ab[2];
        double yb = vals[i][1] - ab[3];
        double vx = ab[2] - ab[0];
        double vy = ab[3] - ab[1];
        double t1 = (a + Math.atan2(yb, xb)) / w;
        double cos = xb / Math.sqrt(xb * xb + yb * yb);
        double sin = yb / Math.sqrt(xb * xb + yb * yb);
        double bl = ((yb * vx - xb * vy) / w) / (vx * cos - vy * sin);
        t1 = Math.max(t1, 0);
        if (t1 <= bl) {
          cnt++;
          continue;
        }
        double t2 = t1 + 2.0 * Math.PI / w;
        double br = ((yb * vx - xb * vy) / w) / (vx * cos - vy * sin);
        if (t2 >= br) {
          cnt++;
          continue;
        }
        double x1 = xb + vx * t1;
        double t3 = Math.sqrt((ab[0] - x1) * (ab[0] - x1) + (ab[1] - yb) * (ab[1] - yb)) / m;
        double x2 = xb + vx * t2;
        double t4 = Math.sqrt((ab[0] - x2) * (ab[0] - x2) + (ab[1] - yb) * (ab[1] - yb)) / m;
        if (t1 <= t3 && t4 <= t2 || t3 <= t1 && t2 <= t4) {
          cnt++;
        }
      }
      if (cnt <= k) {
        r = m;
      } else {
        l = m;
      }
    }
    System.out.printf("%.4f\n", r);
  }
}
