import java.util.ArrayList;
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Main {
    private static int n;
    private static int[][] g;
    private static int[] dis;
    private static int[] vis;
    private static ArrayList<Set<Integer>> has;
    private static ArrayList<Set<Integer>> nop;
    private static boolean dfs(int node, int c) {
        if (vis[node] == 1) {
            return false;
        }
        vis[node] = 1;
        if (node != c) {
            nop.get(c).add(node);
        } else {
            has.get(c).add(node);
        }
        for (int i = 0; i < n; i++) {
            if (g[node][i] == 1 && vis[i] == 0) {
                if (!dfs(i, c)) {
                    return false;
                }
            } else if (g[node][i] == -1) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner reader = new Scanner(System.in);
        n = reader.nextInt();
        int q = reader.nextInt();
        g = new int[n][n];
        ArrayList<ArrayList<Integer>> arr = new ArrayList<ArrayList<Integer>>();
        for (int i = 0; i < n; i++) {
            arr.add(new ArrayList<Integer>());
        }
        for (int i = 0; i < q; i++) {
            int p = reader.nextInt();
            int r = reader.nextInt();
            p--;
            r--;
            g[p][r] = 1;
            g[r][p] = 1;
            arr.get(p).add(r);
            arr.get(r).add(p);
        }
        int w = reader.nextInt();
        ArrayList<ArrayList<Integer>> ar = new ArrayList<ArrayList<Integer>>();
        for (int i = 0; i < n; i++) {
            ar.add(new ArrayList<Integer>());
        }
        for (int i = 0; i < w; i++) {
            int p = reader.nextInt();
            int r = reader.nextInt();
            p--;
            r--;
            g[p][r] = -1;
            g[r][p] = -1;
            ar.get(p).add(r);
            ar.get(r).add(p);
        }
        dis = new int[n];
        vis = new int[n];
        int ans = 0;
        for (int i = 0; i < n; i++) {
            has = new ArrayList<Set<Integer>>();
            nop = new ArrayList<Set<Integer>>();
            for (int j = 0; j < n; j++) {
                has.add(new HashSet<Integer>());
                nop.add(new HashSet<Integer>());
            }
            if (dfs(i, i)) {
                int c = 0;
                for (int j = 0; j < n; j++) {
                    if (nop.get(i).contains(j) == false) {
                        c++;
                    }
                }
                ans = Math.max(c, ans);
            }
        }
        System.out.println(ans);
    }
}
