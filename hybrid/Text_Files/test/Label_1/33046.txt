import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        while (true) {
            int T = input.nextInt();
            if (T == 0) {
                break;
            }
            int[] cycle = new int[T];
            for (int i = 0; i < T; i++) {
                cycle[i] = input.nextInt();
            }
            int N = input.nextInt();
            int[][] interviews = new int[N][2];
            for (int i = 0; i < N; i++) {
                interviews[i][0] = input.nextInt() - 1;
                interviews[i][1] = input.nextInt();
            }

            int minCaffeine = Integer.MAX_VALUE;
            for (int i = 0; i < T; i++) {
                int caffeine = 0;
                int sleep = i;
                for (int[] interview : interviews) {
                    while (sleep < interview[0]) {
                        sleep += cycle.length;
                        caffeine++;
                    }
                    if (sleep == interview[0] && cycle[sleep % cycle.length] >= interview[1]) {
                        continue;
                    } else {
                        caffeine++;
                        sleep = i;
                    }
                }
                minCaffeine = Math.min(minCaffeine, caffeine);
            }

            System.out.println(minCaffeine);
        }
    }
}
