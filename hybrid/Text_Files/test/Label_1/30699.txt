import java.util.*;
class Main {
    static class Node implements Comparable<Node> {
        int v, w;
        Node(int v, int w) {
            this.v = v;
            this.w = w;
        }
        public int compareTo(Node o) {
            return this.w - o.w;
        }
    }
    static int N, M, X;
    static int[] u, v, w;
    static int[] p, r;
    static List<Node>[] G;
    static int MOD = 1000000007;
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        M = sc.nextInt();
        X = sc.nextInt();
        u = new int[M+1];
        v = new int[M+1];
        w = new int[M+1];
        for (int i = 1; i <= M; i++) {
            u[i] = sc.nextInt();
            v[i] = sc.nextInt();
            w[i] = sc.nextInt();
        }
        p = new int[N+1];
        r = new int[N+1];
        for (int i = 1; i <= N; i++) {
            p[i] = i;
            r[i] = 1;
        }
        G = new List[N+1];
        for (int i = 1; i <= N; i++) {
            G[i] = new ArrayList<>();
        }
        for (int i = 1; i <= M; i++) {
            G[u[i]].add(new Node(v[i], w[i]));
            G[v[i]].add(new Node(u[i], w[i]));
        }
        int c = 0;
        boolean ok = false;
        for (int i = 1; i <= M; i++) {
            if (find(u[i]) != find(v[i])) {
                union(u[i], v[i]);
                c++;
                if (c == N-1) {
                    ok = true;
                    break;
                }
            }
        }
        if (!ok) {
            System.out.println(0);
            return;
        }
        int[] d = new int[N+1];
        for (int i = 1; i <= M; i++) {
            int x = find(u[i]);
            if (d[x] == 0) {
                d[x] = i;
            }
        }
        List<Node>[] GT = new List[N+1];
        for (int i = 1; i <= N; i++) {
            GT[i] = new ArrayList<>();
        }
        for (int i = 1; i <= M; i++) {
            int x = find(u[i]);
            int y = find(v[i]);
            if (x != y) {
                GT[x].add(new Node(y, w[i]));
                GT[y].add(new Node(x, w[i]));
            }
        }
        long[][][] dp = new long[N+1][2][X+1];
        for (int i = 1; i <= N; i++) {
            for (int j = 0; j < 2; j++) {
                for (int k = 0; k <= X; k++) {
                    dp[i][j][k] = -1;
                }
            }
        }
        dp[1][0][0] = 1;
        dp[1][1][0] = 1;
        for (int i = 1; i <= N; i++) {
            for (int j = 0; j < 2; j++) {
                for (int k = 0; k <= X; k++) {
                    if (dp[i][j][k] == -1) {
                        continue;
                    }
                    for (Node e: GT[i]) {
                        int u = e.v, w = e.w;
                        if (j == 0) {
                            dp[u][1][k+w] = (dp[u][1][k+w] + dp[i][j][k]) % MOD;
                        } else {
                            dp[u][0][k+w] = (dp[u][0][k+w] + dp[i][j][k]) % MOD;
                            dp[u][1][k+w] = (dp[u][1][k+w] + dp[i][j][k]) % MOD;
                        }
                    }
                }
            }
        }
        long ans = 0;
        for (int i = 1; i <= N; i++) {
            ans = (ans + dp[i][0][X]) % MOD;
            ans = (ans + dp[i][1][X]) % MOD;
        }
        System.out.println(ans);
    }
    static int find(int x) {
        if (p[x] == x) {
            return x;
        }
        return p[x] = find(p[x]);
    }
    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) {
            return;
        }
        if (r[x] < r[y]) {
            p[x] = y;
            r[y] += r[x];
        } else {
            p[y] = x;
            r[x] += r[y];
        }
    }
}
