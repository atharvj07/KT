import sys

n, k, x = map(int, sys.stdin.readline().split())
L = list(map(int, sys.stdin.readline().split()))
L.sort()
ans, j = 0, 0
for i in range(n):
    if j < i and L[i] - L[j] > x:
        j += 1
        ans += 1
    if k == 0:
        break
    if j < i and L[i] - L[j] <= x:
        j += 1
        k -= 1
print(ans + 1)
