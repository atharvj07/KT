import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int l = sc.nextInt();
    long sum = 0;
    for (int i = 0; i < n; i++) {
      sum += l + i;
    }
    long minDiff = sum;
    long minFlavor = 0;
    for (int i = 0; i < n; i++) {
      long diff = Math.abs(sum - (l + i) * 2);
      if (diff < minDiff) {
        minDiff = diff;
        minFlavor = sum - (l + i);
      }
    }
    System.out.println(minFlavor);
  }
}
