import java.util.*;
import java.io.*;

public class Main {
  static final int MAX_RANK = 13;
  static final int MAX_COUNT = 4;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    while (true) {
      int n = Integer.parseInt(br.readLine());
      if (n == 0) break;
      int[] counts = new int[MAX_RANK + 1];
      String[] line = br.readLine().split(" ");
      for (int i = 0; i < n; i++) {
        counts[Integer.parseInt(line[i])]++;
      }
      List<Integer> result = solve(counts);
      if (result == null) {
        System.out.println("No");
      } else {
        for (int i = 0; i < result.size(); i++) {
          System.out.print(result.get(i) + " ");
        }
        System.out.println();
      }
    }
  }

  static List<Integer> solve(int[] counts) {
    List<Integer> result = new ArrayList<>();
    int sum = 0;
    for (int i = MAX_RANK; i >= 1; i--) {
      while (counts[i] > 0) {
        if (sum % i == 0) {
          result.add(i);
          sum += i;
          counts[i]--;
        } else {
          for (int j = 1; j <= MAX_RANK; j++) {
            if (counts[j] > 0 && (sum + i + j) % i == 0) {
              result.add(j);
              sum += j;
              counts[j]--;
              break;
            }
          }
          if (counts[i] > 0) {
            return null;
          }
        }
      }
    }
    return result;
  }
}
