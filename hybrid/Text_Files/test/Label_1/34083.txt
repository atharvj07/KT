import java.util.*;
class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int[] hw = new int[2];
		for (int i = 0; i < 2; i++) hw[i] = sc.nextInt();
		int[][] m = new int[hw[0]][hw[1]];
		int ans = 0;
		for (int i = 0; i < hw[0]; i++) {
			String s = sc.next();
			for (int j = 0; j < hw[1]; j++) {
				if (s.charAt(j) == 'P') m[i][j] = 2;
				else if (s.charAt(j) == 'W') m[i][j] = 1;
			}
		}
		for (int i = 0; i < hw[0]; i++) {
			for (int j = 0; j < hw[1]; j++) {
				if (m[i][j] == 2) {
					int cnt = 0;
					for (int[] d : new int[][] {{-1, 0}, {0, -1}, {+1, 0}, {0, +1}}) {
						int nr = i + d[0], nc = j + d[1];
						if (0 <= nr && nr < hw[0] && 0 <= nc && nc < hw[1] && m[nr][nc] == 1) cnt++;
					}
					ans += cnt > 0 ? 1 : 0;
				}
			}
		}
		System.out.println(ans);
	}
}
