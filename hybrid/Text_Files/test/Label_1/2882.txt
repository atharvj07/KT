#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        long long s, n, k;
        cin >> s >> n >> k;
        if (n == 1 || s <= n || ((s + k - 1) / k > n - 1)) {
            cout << "YES" << endl;
        } else {
            cout << "NO" << endl;
        }
    }
    return 0;
}
