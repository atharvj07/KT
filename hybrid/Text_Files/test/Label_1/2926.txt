#include <algorithm>
#include <cmath>
#include <iostream>
#include <vector>

using namespace std;

struct Point {
  int x, y;
  Point(int x = 0, int y = 0) : x(x), y(y) {}
  bool operator<(const Point& rhs) const { return x < rhs.x || (x == rhs.x && y < rhs.y); }
};

int main() {
  int n, r;
  cin >> n >> r;
  vector<Point> points(n);
  for (auto& p : points) cin >> p.x >> p.y;
  sort(points.begin(), points.end());
  int ans = 0, xcnt = 0;
  for (auto& p : points) {
    while (xcnt < n && points[xcnt].x <= p.x + r) {
      ans += (points[xcnt].x >= p.x - r);
      xcnt++;
    }
    ans -= (xcnt > 0 && points[xcnt - 1].x <= p.x - r);
  }
  cout << ans << endl;
}
