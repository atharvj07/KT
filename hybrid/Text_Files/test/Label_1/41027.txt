import sys
import collections

def solve():
    inf = float('inf')
    n, m = map(int, input().split())
    g = collections.defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)
    if n == 3:
        sys.stdout.write('0\n')
        return

    odd = []
    for u in g:
        if len(g[u]) % 2 == 1:
            odd.append(u)
    if len(odd) == 0 or len(odd) % 2 == 0:
        sys.stdout.write('0\n')
        return
    if len(odd) == 2:
        sys.stdout.write('1\n')
        sys.stdout.write('%d %d\n' % (odd[0], odd[1]))
        return
    sys.stdout.write('2\n')
    u = odd[0]
    v = odd[1]
    for w in g[u]:
        if w != v:
            sys.stdout.write('%d %d\n' % (u, w))
            sys.stdout.write('%d %d\n' % (w, v))
            return


solve()
