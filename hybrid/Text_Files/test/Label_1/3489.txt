#include <bits/stdc++.h>
using namespace std;
#define ll int64_t
int main() {
  ll n, m;
  cin >> n >> m;
  vector<ll> a(n);
  for (ll& ai : a) cin >> ai;
  vector<ll> pref(n + 1);
  for (ll i = 0; i < n; i++) pref[i + 1] = pref[i] + a[i];
  vector<ll> b(m);
  for (ll& bi : b) cin >> bi;
  for (ll bi : b) {
    ll low = 0, high = n;
    while (low < high) {
      ll mid = (low + high) / 2;
      if (bi <= pref[mid])
        high = mid;
      else
        low = mid + 1;
    }
    cout << low << ' ' << bi - pref[low - 1] << '\n';
  }
}
