from collections import Counter

n, k = map(int, input().split())
digits = [int("".join(input()[::-1]), 2) for _ in range(n)]
cnts = [Counter(bin(d)[2:]) for d in digits]
sticks = Counter()
for d in cnts:
    sticks.update(d)

best = ""
if sum(map(lambda x: x[1], filter(lambda x: x[0] == '1', cnts))) + k >= sticks['1']:
    for i in range(n - 1, -1, -1):
        tmp = ""
        for j in range(7):
            if digits[i] & (1 << j) and cnts[i][str(j)] == 1 and sticks[str(j)] > 0:
                tmp += '1'
                sticks[str(j)] -= 1
            else:
                tmp += '0'
        best = tmp[::-1] + best
else:
    best = -1

print(best)
