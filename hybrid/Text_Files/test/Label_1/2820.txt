#include <vector>
#include <set>
#include <algorithm>
#include <cstdio>

using namespace std;

const int maxn = 1000 + 100;
typedef long long ll;

struct Node {
    int t, q;
    Node() {}
    Node(int _t, int _q) : t(_t), q(_q) {}
};

int n, T;
Node t_q[maxn];
vector<vector<int>> adj;

int dfs(int i, int mt) {
    if (adj[i].size() == 0) {
        if (mt <= T) return t_q[i].q;
        return 0;
    }
    int ret = 0;
    for (int u : adj[i]) {
        ret = max(ret, dfs(u, mt + t_q[i].t));
    }
    return ret;
}

int main() {
    scanf("%d %d", &n, &T);
    adj.assign(n, vector<int>());
    for (int i = 0; i < n; ++i) {
        scanf("%d %d", &t_q[i].t, &t_q[i].q);
    }
    int m;
    scanf("%d", &m);
    while (m--) {
        int a, b;
        scanf("%d %d", &a, &b);
        --a; --b;
        adj[a].push_back(b);
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        ans = max(ans, dfs(i, 0));
    }
    printf("%d\n", ans);
    return 0;
}
