#include <bits/stdc++.h>

using namespace std;

int main() {
  string s;
  cin >> s;
  int ans = 0;
  for (int i = 1; i < s.size(); i++) {
    if (s[i] != s[i - 1] && (s[i] == 'A' || s[i] == 'E' || s[i] == 'I' ||
                               s[i] == 'O' || s[i] == 'U' || s[i] == 'Y'))
      ans = i;
  }
  cout << ans + 1 << endl;
  return 0;
}
