import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    int[] costs = new int[4];
    for (int i = 0; i < 4; i++) {
      costs[i] = sc.nextInt();
    }
    String a = sc.next();
    String b = sc.next();

    int[][] dp = new int[a.length() + 1][b.length() + 1];
    for (int i = 1; i <= a.length(); i++) {
      for (int j = 1; j <= b.length(); j++) {
        dp[i][j] = Integer.MAX_VALUE;
        if (a.charAt(i - 1) == b.charAt(j - 1)) {
          dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - 1]);
        } else {
          dp[i][j] = Math.min(dp[i][j], dp[i - 1][j] + costs[1]);
          dp[i][j] = Math.min(dp[i][j], dp[i][j - 1] + costs[0]);
          if (j - 2 >= 0) {
            dp[i][j] = Math.min(dp[i][j], dp[i][j - 2] + costs[3]);
          }
          if (i - 2 >= 0) {
            dp[i][j] = Math.min(dp[i][j], dp[i - 2][j - 1] + costs[3]);
          }
          dp[i][j] = Math.min(dp[i][j], dp[i - 1][j - 1] + costs[2]);
        }
      }
    }
    System.out.println(dp[a.length()][b.length()]);
  }
}
