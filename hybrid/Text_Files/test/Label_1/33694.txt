import java.io.*;
import java.util.*;

public class Main {
    static int[] a;
    static int[] par;
    static int[] l;
    static int[] r;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        a = new int[n + 1];
        par = new int[n + 1];
        l = new int[n + 1];
        r = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = in.nextInt();
        }
        for (int i = 2; i <= n; i++) {
            int x = 1;
            while (true) {
                if (a[i] > a[x]) {
                    if (r[x] == 0) {
                        r[x] = i;
                        par[i] = x;
                        break;
                    } else {
                        x = r[x];
                    }
                } else {
                    if (l[x] == 0) {
                        l[x] = i;
                        par[i] = x;
                        break;
                    } else {
                        x = l[x];
                    }
                }
            }
        }
        for (int i = 2; i <= n; i++) {
            System.out.println(a[par[i]]);
        }
    }
}
