import java.io.*;
import java.util.*;

public class Main {
  public static void main(String[] args) {
    try {
      BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
      String str = br.readLine();
      int mod = (int)1e9 + 7;
      int n = str.length();
      int prefA[] = new int[n];
      for (int i = 0; i < n; i++) {
        if (i > 0) {
          prefA[i] = prefA[i - 1];
        }
        if (str.charAt(i) == 'a') {
          prefA[i]++;
        }
      }
      long ans = 0;
      int countB = 0;
      for (int i = 0; i < n; i++) {
        if (str.charAt(i) == 'b') {
          countB++;
        } else {
          ans = (ans + prefA[i] * countB) % mod;
        }
      }
      System.out.println(ans);
    } catch (Exception e) {
      e.printStackTrace();
    }
  }
}
