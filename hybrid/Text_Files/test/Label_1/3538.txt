#include <cstdio>
#include <algorithm>
using namespace std;
int n,a[100000];
bool isok() {
  sort(a,a+n);
  for (int i=0; i<n; i+=3)
    if (a[i]*6<a[i+1]*4||a[i+1]*4<a[i+2]) return 0;
  return 1;
}
int main() {
  scanf("%d",&n);
  for (int i=0; i<n; i++) scanf("%d",&a[i]);
  if (isok()) for (int i=0; i<n; i+=3) printf("%d %d %d ",a[i],a[i+1],a[i+2]);
  else printf("-1\n");
}
