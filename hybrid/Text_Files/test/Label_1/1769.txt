#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, a, b;
        cin >> n >> a >> b;
        if (a > b || (n / 2) % 2 == 1) cout << -1 << endl;
        else {
            vector<int> perm(n);
            int l = n / 2 + 1;
            for (int i = 0; i < n / 2; i++) {
                perm[i] = i + 1;
                perm[l++] = n - i;
            }
            int pos = find(perm.begin(), perm.end(), a) - perm.begin();
            swap(perm[pos], perm[n / 2]);
            pos = find(perm.begin(), perm.end(), b) - perm.begin();
            swap(perm[pos], perm[n - 1]);
            for (auto x : perm) cout << x << ' ';
            cout << endl;
        }
    }
    return 0;
}
