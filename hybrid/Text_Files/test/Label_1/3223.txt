#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, x;
        cin >> n >> x;
        vector<int> a(n);
        for (int &ai : a) cin >> ai;
        sort(a.begin(), a.end());
        int l = 0, r = n - 1, ans = 0;
        while (l < r) {
            if (a[l] * (r - l + 1) >= x) {
                ans++, l++;
            } else
                r--;
        }
        cout << ans << endl;
    }
}
