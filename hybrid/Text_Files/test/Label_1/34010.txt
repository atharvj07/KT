import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int l = input.nextInt();
        int r = input.nextInt();
        int[] a = new int[n];
        int[] b = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = input.nextInt();
        }
        boolean lie = false;
        for (int i = l - 1; i < r; i++) {
            if (a[i] != b[i]) {
                lie = true;
                break;
            }
        }
        if (lie) {
            System.out.println("LIE");
        } else {
            System.out.println("TRUTH");
        }
    }
}
