import sys
input = sys.stdin.readline
n, p = map(int, input().split())
sus = {}
deg = [0] * (n + 1)
for _ in range(n):
    x, y = map(int, input().split())
    sus[x] = sus.get(x, []) + [y]
    sus[y] = sus.get(y, []) + [x]
    deg[x] += 1
    deg[y] += 1
ans = 0
for i in range(1, n + 1):
    need = max(p - deg[i], 0)
    if deg[i] >= p or len(sus[i]) > need:
        ans += deg[i] * (deg[i] - 1) // 2
    else:
        ans += deg[i] * (need - 1)
print(ans)
