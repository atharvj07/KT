import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), k = sc.nextInt(), m = sc.nextInt();
    long[][] a = new long[n + 2][2];
    for (int i = 1; i <= n; i++) a[i][0] = -1;
    for (int i = 0; i < m; i++) {
      int l = sc.nextInt(), r = sc.nextInt(), c = sc.nextInt(), p = sc.nextInt();
      if (a[l][1] < p) a[l][1] = p;
      if (a[r + 1][0] > p) a[r + 1][0] = p;
    }
    long cur = a[1][1], ans = 0;
    for (int i = 1; i <= n; i++) {
      if (cur > a[i][1]) cur = a[i][1];
      if (cur > a[i + 1][0]) cur = a[i + 1][0];
      if (cur < 0 || i < k) {
        ans += k - cur;
        cur += k - cur;
      }
    }
    System.out.println(ans);
  }
}
