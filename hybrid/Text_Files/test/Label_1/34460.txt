import java.util.Scanner;

public class JumpStyle {

    private static int n;
    private static int[][] grid;
    private static boolean[][] visited;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            n = scanner.nextInt();
            if (n == 0) {
                break;
            }

            grid = new int[n][n];
            visited = new boolean[n][n];

            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    grid[i][j] = scanner.nextInt();
                }
            }

            int loopCount = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (!visited[i][j]) {
                        findLoop(i, j);
                        loopCount++;
                    }
                }
            }

            System.out.println(loopCount);
        }

        scanner.close();
    }

    private static void findLoop(int x, int y) {
        visited[x][y] = true;

        int nextX = grid[x][y];
        int nextY = grid[y][x];

        if (!visited[nextX][nextY]) {
            findLoop(nextX, nextY);
        }
    }
}
