#include<bits/stdc++.h>
using namespace std;
long long a,b;
int main() {
 cin >> a >> b ;
 if(a < b) swap(a,b);
 long long sum=0,sa=0,sb=0,cnt=0,cnt1=0;
 long long len=sqrt(b);
 for(long long i=1;;i++) {
  if(sa+i<=a && (sum+i)<=b) {
      sum+=i,sa+=i;
      cnt++;
  }
  if(sb+i<=b && (sum+i)<=b) {
      sum+=i,sb+=i;
      cnt1++;
  }
  if(i>len) break;
 }
 cout << cnt << endl;
 for(long long i=1;;i++) {
  if(sa + i <= a) {
    cout << i << " ";
    sa+=i;
  }
  if(sa+i>a) break;
 }
 cout << endl;
 cout << cnt1 << endl;
 for(long long i=1;;i++) {
  if(sb + i <= b) {
    cout << i << " ";
    sb+=i;
  }
  if(sb+i>b) break;
}
 cout << endl;
return 0;
}
