import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt(), n = sc.nextInt();
        boolean[][] g = new boolean[r + 1][3];
        for (int i = 0; i < n; i++) {
            int x = sc.nextInt(), y = sc.nextInt();
            g[x][y] = g[x][3 - y] = true;
        }
        boolean turn = true;
        for (int i = 1; i <= r; i++) {
            for (int j = 1; j <= 2; j++) {
                if (!g[i][j]) {
                    g[i][j] = g[i][3 - j] = true;
                    turn = !turn;
                    break;
                }
            }
        }
        System.out.println(turn ? "WIN" : "LOSE");
    }
}
