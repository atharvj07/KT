def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        x = int(input())
        pre = [0] * n
        pre[0] = a[0]
        cur = a[0]
        for i in range(1, n):
            cur += a[i]
            if cur < 0:
                cur = 0
            pre[i] = cur
        dp = [0] * (n + 1)
        for i in range(n):
            l = i
            r = n - 1
            id = -1
            while l <= r:
                mid = (l + r) // 2
                if pre[mid] - (pre[i - 1] if i > 0 else 0) >= x * (mid - i + 1):
                    id = mid
                    l = mid + 1
                else:
                    r = mid - 1
            if id == -1:
                dp[i + 1] = dp[i]
            else:
                dp[i + 1] = max(dp[i + 1], dp[id] + id - i + 1)
        print(dp[n])

solve()
