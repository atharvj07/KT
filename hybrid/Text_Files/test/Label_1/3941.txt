#include <iostream>

int gcd(const long long a, const long long b) {
  return b ? gcd(b, a % b) : a;
}

int main() {
  long long n, k;
  std::cin >> n >> k;
  if (n == 1) {
    std::cout << 0 << std::endl;
  } else if (n % k != 0 && 2 * n % k != 0) {
    std::cout << -1 << std::endl;
  } else if (n % 2 == 0 && k % 2 != 0) {
    std::cout << -1 << std::endl;
  } else if (n % 2 == 1 && k % 2 == 0) {
    std::cout << -1 << std::endl;
  } else {
    long long q = n / gcd(n, k);
    std::cout << q - 1 << std::endl;
  }
  return 0;
}
