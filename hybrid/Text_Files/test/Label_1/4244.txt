#include <iostream>
#include <vector>

int main() {
    long long n;
    std::cin >> n;
    int ans = 2;
    while (n != 1) {
        if (n % 10 == 0) {
            n /= 10;
            ans++;
        } else {
            n += 1;
            ans++;
        }
    }
    std::cout << ans << std::endl;
    return 0;
}
