#include <bits/stdc++.h>

using namespace std;
#define ll long long

vector<ll> path;
vector<vector<pair<ll, ll>>> adj;
vector<bool> vis;
ll max_fuel = 0;

void dfs(ll node, ll par, ll fuel) {
  vis[node] = true;
  path.push_back(node);
  if (fuel > max_fuel) {
    max_fuel = fuel;
  }
  for (const auto &[next, w] : adj[node]) {
    if (vis[next] || next == par) continue;
    dfs(next, node, fuel + w);
  }
  path.pop_back();
}

int main() {
  ll n;
  cin >> n;
  vector<ll> fuel(n);
  for (ll &x : fuel) cin >> x;
  adj.resize(n);
  vis.resize(n);
  for (ll i = 0; i < n - 1; i++) {
    ll a, b, w;
    cin >> a >> b >> w;
    a--, b--;
    adj[a].push_back({b, w});
    adj[b].push_back({a, w});
  }
  ll root = 0;  // root of tree
  dfs(root, -1, fuel[root]);
  vis.clear();
  path.clear();
  dfs(path[max_element(path.begin(), path.end()) - path.begin()], -1, fuel[root]);
  cout << max_fuel << endl;
  return 0;
}
