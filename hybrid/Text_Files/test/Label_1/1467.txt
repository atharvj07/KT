#include <iostream>
#include <cstring>

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        char keyboard[27], word[52];
        std::cin >> keyboard >> word;
        int n = strlen(word);
        int time = 0;
        int prev = 0;
        for (int i = 0; i < n; i++) {
            int pos = 0;
            while (keyboard[pos] != word[i]) pos++;
            time += abs(pos - prev);
            prev = pos;
        }
        std::cout << time << "\n";
    }
    return 0;
}
