#include <iostream>
#include <string>
#include <map>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string s;
    cin >> s;
    map<char, int> m;
    for (char c : s) m[c]++;
    int cnt = 0;
    for (auto it = m.begin(); it != m.end(); it++) cnt += (it->second + k - 1) / k;
    if (cnt <= n) cout << "YES" << endl;
    else cout << "NO" << endl;
    return 0;
}
