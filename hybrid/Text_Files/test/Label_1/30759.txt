import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        long[][] B = new long[n][3], C = new long[m][3];
        for(long[] x:B) for(int i=0;i<3;++i) x[i]=sc.nextInt();
        for(long[] x:C) for(int i=0;i<3;++i) x[i]=sc.nextInt();
        for(long[] x:C){
            int ans=-1;
            for(int i=0;i<n;++i){
                if(x[0]>B[i][1] || x[1]<B[i][0])continue;
                if(ans==-1 || B[ans][2]>B[i][2])ans=i;
            }
            if(ans==-1)System.out.print(-1);
            else System.out.print(ans+1);
            System.out.print(' ');
        }
        System.out.println();
    }
}
