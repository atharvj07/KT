n = int(input())
sx = [0]*n; sy = [0]*n; ax = [0]*n; ay = [0]*n
for i in range(n):
    sx[i], sy[i] = map(int, input().split())
for i in range(n):
    ax[i], ay[i] = map(int, input().split())
tx = sx[0] - ax[0]; ty = sy[0] - ay[0]
for i in range(n):
    if tx != sx[i] - ax[i] or ty != sy[i] - ay[i]: print('No solution'); exit()
print(tx, ty)
