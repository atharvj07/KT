import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long[] v = new long[n];
        for (int i = 0; i < n; i++) {
            v[i] = in.nextLong();
        }
        long[] f = new long[m + 1];
        for (int i = 0; i < n; i++) {
            for (int j = m; j >= v[i]; j--) {
                f[j] = Math.max(f[j], f[(int) (j - v[i])] + v[i]);
            }
        }
        long[] g = new long[m + 1];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= m - v[i]; j++) {
                g[j] = Math.max(g[j], g[(int) (j + v[i])] + v[i]);
            }
        }
        long w_a = 0, w_b = 0, w_f = 0, w_s = 0;
        for (int i = 1; i <= m; i++) {
            if (f[i] == m) {
                w_a++;
                if (g[i] == 0) {
                    w_f++;
                } else if (g[i] == m) {
                    w_s++;
                }
            } else if (g[i] == m) {
                w_b++;
                if (f[i] == 0) {
                    w_s++;
                } else if (f[i] == m) {
                    w_f++;
                }
            }
        }
        System.out.println(w_a + " " + w_b + " " + w_f + " " + w_s);
    }
}
