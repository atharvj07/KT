import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[][] g = new char[n][m];
        int[][] e = new int[n][m];
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) {
                g[i][j] = s.charAt(j);
                if (g[i][j] == 'P') {
                    cnt++;
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (g[i][j] == 'W') {
                    for (int k = -1; k <= 1; k++) {
                        for (int l = -1; l <= 1; l++) {
                            if (i + k >= 0 && i + k < n && j + l >= 0 && j + l < m && g[i + k][j + l] == 'P') {
                                e[i + k][j + l] = 1;
                            }
                        }
                    }
                }
            }
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (e[i][j] == 1) {
                    cnt--;
                }
            }
        }
        System.out.println(cnt);
    }
}
