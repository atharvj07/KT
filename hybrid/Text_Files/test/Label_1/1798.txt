#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    while (n--) {
        string s;
        cin >> s;
        vector<int> cnt(11, 0);
        int ans = 0;
        for (char c : s) {
            cnt[c - '0']++;
        }
        for (int i = 1; i <= 9; i++) {
            if (cnt[i] > 1) ans = 1;
        }
        if (cnt[0] > 0 && cnt[0] < s.size()) ans = 1;
        cout << ans << "\n";
    }
    return 0;
}
