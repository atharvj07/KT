#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    vector<int> v(n);
    for (int i = 0; i < n; i++) v[i] = s[i] - '0';

    if (v[0] != v[n - 1]) {
        cout << "NO" << endl;
        return 0;
    }

    int sum = 0;
    for (int i = 0; i < n; i++) sum += v[i];
    if (sum % n != 0) {
        cout << "NO" << endl;
        return 0;
    }

    int k = sum / n;

    for (int i = 0; i < n; i++) {
        if (v[i] > k) {
            cout << "NO" << endl;
            return 0;
        }
    }

    cout << "YES" << endl;

    return 0;
}
