#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    int ans = n;
    int l = -1, r = -1;
    for (int i = 0; i < n; i++) {
        if (a[i] == *min_element(a.begin(), a.end())) {
            if (l != -1) {
                ans = min(ans, i - l);
            }
            l = i;
        }
    }
    for (int i = n - 1; i >= 0; i--) {
        if (a[i] == *min_element(a.begin(), a.end())) {
            if (r != -1) {
                ans = min(ans, r - i);
            }
            r = i;
        }
    }
    cout << ans << endl;
}
