import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;
import java.util.HashSet;
import java.util.Set;

public class Main {
    static class Location {
        int x;
        int y;

        Location(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int sx = scanner.nextInt() - 1;
        int sy = scanner.nextInt() - 1;
        int fx = scanner.nextInt() - 1;
        int fy = scanner.nextInt() - 1;
        Set<Location> instants = new HashSet<>();
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            instants.add(new Location(x, y));
        }

        int[][] distances = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                distances[i][j] = -1;
            }
        }

        Queue<Location> queue = new LinkedList<>();
        queue.offer(new Location(sx, sy));
        distances[sx][sy] = 0;
        while (!queue.isEmpty()) {
            Location location = queue.poll();
            int x = location.x;
            int y = location.y;
            int distance = distances[x][y];

            if (x == fx && y == fy) {
                break;
            }

            // Check if there is an instant movement location on the same x or y coordinate
            if (instants.contains(new Location(x, y))) {
                queue.offer(new Location(fx, y));
                queue.offer(new Location(x, fy));
                distances[fx][y] = distance + 1;
                distances[x][fy] = distance + 1;
            }

            // Check if we can move to the neighboring blocks
            if (x > 0 && distances[x - 1][y] == -1) {
                queue.offer(new Location(x - 1, y));
                distances[x - 1][y] = distance + 1;
            }
            if (x < n - 1 && distances[x + 1][y] == -1) {
                queue.offer(new Location(x + 1, y));
                distances[x + 1][y] = distance + 1;
            }
            if (y > 0 && distances[x][y - 1] == -1) {
                queue.offer(new Location(x, y - 1));
                distances[x][y - 1] = distance + 1;
            }
            if (y < n - 1 && distances[x][y + 1] == -1) {
                queue.offer(new Location(x, y + 1));
                distances[x][y + 1] = distance + 1;
            }
        }

        System.out.println(distances[fx][fy]);
    }
}
