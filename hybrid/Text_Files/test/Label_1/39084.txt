import sys
input = sys.stdin.readline

t = int(input())
for _ in range(t):
    n, m, a, b = map(int, input().split())
    s = list(map(int, input().split()))
    s.sort(reverse=True)
    ans, t = 0, 0
    for i in s:
        t += i + 1
        if t >= abs(a - b) + 2:
            ans += 1
        else:
            break
    print(ans)
