#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<bool> b(n);
    for (int i = 0; i < n; i++) {
        int t;
        cin >> t;
        b[i] = t == 0;
    }
    int ans = k - 1;
    int cur = -1;
    for (int i = 0; i < n; i++) {
        if (b[i]) continue;
        if (cur == -1) cur = i;
        else if (i - cur > ans) ans = i - cur;
        cur = i;
    }
    cout << ans << endl;
    return 0;
}
