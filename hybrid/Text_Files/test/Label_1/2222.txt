#include <bits/stdc++.h>
using namespace std;
const int N = 2e5 + 5, mod = 1e9 + 7;
typedef long long ll;
#define pb push_back
int n, t;
ll a[N];
int main() {
    cin >> t;
    while(t--) {
        cin >> n;
        for(int i = 1; i <= n; i++) cin >> a[i];
        a[n] -= a[n - 1];
        ll ans = 0;
        for(int i = n - 1; i >= 1; i--) {
            if(i == 1) ans += abs(a[i]);
            else {
                if(a[i] == 0) continue;
                else if(a[i] > 0) {
                    if(a[i] == a[i + 1]) continue;
                    else if(a[i] < a[i + 1]) ans += a[i + 1] - a[i], a[i + 1] = a[i];
                    else ans += abs(a[i]), a[i] = 0;
                }
                else {
                    if(a[i] == a[i + 1]) continue;
                    else if(a[i] > a[i + 1]) ans += a[i] - a[i + 1], a[i] = a[i + 1];
                    else ans += abs(a[i]), a[i] = 0;
                }
            }
        }
        cout << ans << '\n';
    }
    return 0;
}
