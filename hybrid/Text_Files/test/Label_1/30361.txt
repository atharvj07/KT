import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        long[] dp = new long[n];
        dp[0] = Math.abs(a[0]);
        for (int i = 1; i < n; i++) {
            dp[i] = dp[i - 1];
            for (int j = 1; j <= i; j++) {
                dp[i] = Math.max(dp[i], dp[i - j] + Math.abs(Arrays.stream(a, i - j + 1, i + 1).sum()));
            }
        }
        System.out.println(dp[n - 1]);
    }
}
