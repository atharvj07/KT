#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int& x : a) cin >> x;

    int cnt[k + 1] = {};
    for (int x : a) cnt[x]++;

    int exclude = -1, stress = n + 1;
    for (int i = 1; i <= k; i++) {
        int cur = n - cnt[i] - 1;
        if (cur < stress) {
            stress = cur;
            exclude = i;
        }
    }
    cout << exclude << '\n';
}
