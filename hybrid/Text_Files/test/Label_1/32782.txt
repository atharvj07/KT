import java.util.*;

public class Labyrinth {

    static class Block {
        char type;
        boolean visited;

        public Block(char type) {
            this.type = type;
            this.visited = false;
        }
    }

    static class Point {
        int x, y;

        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    static int n, m;
    static Block[][] labyrinth;
    static Point start, end;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        m = scanner.nextInt();
        labyrinth = new Block[n][m];
        scanner.nextLine(); // Consume newline

        for (int i = 0; i < n; i++) {
            String row = scanner.nextLine();
            for (int j = 0; j < m; j++) {
                labyrinth[i][j] = new Block(row.charAt(j));
            }
        }

        start = new Point(scanner.nextInt() - 1, scanner.nextInt() - 1);
        end = new Point(scanner.nextInt() - 1, scanner.nextInt() - 1);

        int result = bfs();
        System.out.println(result == Integer.MAX_VALUE ? -1 : result);
    }

    static int bfs() {
        Queue<Point> queue = new LinkedList<>();
        queue.add(start);
        labyrinth[start.x][start.y].visited = true;

        int minutes = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();

            for (int i = 0; i < size; i++) {
                Point current = queue.poll();

                if (current.x == end.x && current.y == end.y) {
                    return minutes;
                }

                for (Point neighbor : getNeighbors(current)) {
                    if (!labyrinth[neighbor.x][neighbor.y].visited) {
                        queue.add(neighbor);
                        labyrinth[neighbor.x][neighbor.y].visited = true;
                    }
                }
            }

            minutes++;
        }

        return Integer.MAX_VALUE;
    }

    static List<Point> getNeighbors(Point point) {
        List<Point> neighbors = new ArrayList<>();

        int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
        for (int[] direction : directions) {
            int newX = point.x + direction[0];
            int newY = point.y + direction[1];

            if (newX >= 0 && newX < n && newY >= 0 && newY < m) {
                char currentType = labyrinth[point.x][point.y].type;
                char neighborType = labyrinth[newX][newY].type;

                if (isDoorOpen(currentType, direction) && isDoorOpen(neighborType, getOppositeDirection(direction))) {
                    neighbors.add(new Point(newX, newY));
                }
            }
        }

        return neighbors;
    }

    static boolean isDoorOpen(char type, int[] direction) {
        switch (type) {
            case '+':
                return true;
            case '-':
                return direction[0] == 0;
            case '|':
                return direction[1] == 0;
            case '^':
                return direction[0] == -1;
            case '>':
                return direction[1] == 1;
            case '<':
                return direction[1] == -1;
            case 'v':
                return direction[0] == 1;
            case 'L':
                return direction[1] != -1;
            case 'R':
                return direction[1] != 1;
            case 'U':
                return direction[0] != -1;
            case 'D':
                return direction[0] != 1;
            default:
                return false;
        }
    }

    static int[] getOppositeDirection(int[] direction) {
        return new int[] {-direction[0], -direction[1]};
    }
}
