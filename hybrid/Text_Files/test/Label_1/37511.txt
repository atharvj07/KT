import bisect
n=int(input());a=list(map(int,input().split()));
a.sort();ans=[]
if len(set(a))==len(a):
    print(-1)
else:
    for i in range(n):
        for j in range(i+1,n):
            if i>0 and a[i]-a[i-1]!=a[j]-a[i]:
                ans.append(a[i]+1)
            if j<n-1 and a[j]-a[i]!=a[j+1]-a[j]:
                ans.append(a[j]+1)
    ans=list(set(ans))
    ans.sort()
    print(len(ans))
    print(*ans)
