#include <bits/stdc++.h>
using namespace std;
#define int long long

int32_t main() {
  int t;
  cin >> t;
  while (t--) {
    string s;
    int n, ans = 0;
    cin >> s;
    n = s.size();
    for (int i = 0; i < n; i++) {
      for (int j = i + 1; j <= n; j++) {
        string sub = s.substr(i, j - i);
        int k = sub.size();
        int val = 0;
        for (int l = 0; l < k; l++) val = val * 2 + (sub[l] - '0');
        if (k == val) ans++;
      }
    }
    cout << ans << '\n';
  }
}
