import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.Comparator;

class Main {
    private static class Souvenir {
        int weight;
        int cost;

        public Souvenir(int weight, int cost) {
            this.weight = weight;
            this.cost = cost;
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        PrintWriter out = new PrintWriter(System.out);

        String[] line = in.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        int m = Integer.parseInt(line[1]);

        Souvenir[] souvenirs = new Souvenir[n];
        Souvenir[] sortedSouvenirs = new Souvenir[n];

        for (int i = 0; i < n; i++) {
            line = in.readLine().split(" ");
            int weight = Integer.parseInt(line[0]);
            int cost = Integer.parseInt(line[1]);
            souvenirs[i] = new Souvenir(weight, cost);
            sortedSouvenirs[i] = souvenirs[i];
        }

        Arrays.sort(sortedSouvenirs, Comparator.comparingInt(souvenir -> souvenir.cost / souvenir.weight));

        long totalCost = 0;
        int totalWeight = 0;

        for (Souvenir souvenir : sortedSouvenirs) {
            if (totalWeight + souvenir.weight <= m) {
                totalCost += souvenir.cost;
                totalWeight += souvenir.weight;
            } else {
                break;
            }
        }

        out.println(totalCost);
        out.close();
    }
}
