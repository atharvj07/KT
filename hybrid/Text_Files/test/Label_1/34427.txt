import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        int n = scan.nextInt();
        long res = 1;
        for (int i = 0; i < n; i++) {
            res *= scan.nextInt();
        }
        System.out.print(res);
    }
}
