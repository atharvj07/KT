import java.util.*;
import java.io.*;

public class Main {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    in.nextLine();
    String[] s = new String[n];
    for (int i = 0; i < n; i++) {
      s[i] = in.nextLine();
    }
    int m = in.nextInt();
    in.nextLine();
    for (int i = 0; i < m; i++) {
      String[] q = in.nextLine().split(" ");
      int ind = Integer.parseInt(q[0]) - 1;
      String t = q[1];
      int ans = 0;
      for (int j = 0; j <= s[ind].length() - t.length(); j++) {
        if (s[ind].substring(j, j + t.length()).equals(t)) {
          ans++;
        }
      }
      System.out.println(ans);
    }
  }
}
