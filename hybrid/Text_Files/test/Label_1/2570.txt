#include <unordered_map>
#include <vector>
#include <iostream>

struct pair_hash {
    template <class T1, class T2>
    std::size_t operator () (const std::pair<T1,T2> &p) const {
        return ((std::hash<T1>()(std::get<0>(p)) + (std::hash<T2>()(std::get<1>(p))))) ^
               ((std::hash<T2>()(std::get<1>(p))) + (std::hash<T1>()(std::get<0>(p))));
    }
};

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::pair<int, int>> edges(m);
    for (int i = 0; i < m; ++i) {
        std::cin >> edges[i].first >> edges[i].second;
        --edges[i].first;
        --edges[i].second;
    }
    std::unordered_map<std::pair<int, int>, int, pair_hash> edge_idx;
    for (int i = 0; i < m; ++i) {
        edge_idx[edges[i]] = i;
    }
    std::vector<std::vector<int>> edges_by_vertex(n);
    for (const auto& edge : edges) {
        edges_by_vertex[edge.first].push_back(edge.second);
        edges_by_vertex[edge.second].push_back(edge.first);
    }
    std::vector<int> used(n, 0);
    std::vector<char> ans(m);
    std::vector<int> ord(n, 0);
    int time = 0;
    std::pair<int, int> edge_of_the_cycle = {0, 0};
    bool is_cycle = false;
    for (int i = 0; i < n; ++i) {
        if (used[i] == 0) {
            used[i] = 1;
            std::vector<std::pair<int, int>> st;
            st.push_back({i, 1});
            while (!st.empty()) {
                auto [cur, depth] = st.back();
                st.pop_back();
                ord[cur] = time++;
                for (int j : edges_by_vertex[cur]) {
                    if (used[j] == 0) {
                        used[j] = 1;
                        st.push_back({j, depth + 1});
                    } else if (used[j] == 1) {
                        edge_of_the_cycle = {cur, j};
                        if (depth != 1) {
                            is_cycle = true;
                        }
                    }
                }
            }
        }
    }
    if (is_cycle) {
        std::cout << "NO" << std::endl;
    } else {
        std::cout << "YES" << std::endl;
        for (const auto& edge : edges) {
            if ((edge.first == edge_of_the_cycle.first && edge.second == edge_of_the_cycle.second) ||
                (edge.first == edge_of_the_cycle.second && edge.second == edge_of_the_cycle.first)) {
                ans[edge_idx[edge]] = 1;
            } else if (ord[edge.first] > ord[edge.second]) {
                ans[edge_idx[edge]] = 0;
            } else {
                ans[edge_idx[edge]] = 1;
            }
        }
        for (char c : ans) {
            std::cout << c;
        }
        std::cout << std::endl;
    }
    return 0;
}
