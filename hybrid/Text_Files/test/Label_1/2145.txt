#include <iostream>
#include <vector>
#include <string>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> arr(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> arr[i];
    }
    int ans = 0;
    for (int j = 0; j < m; ++j) {
        bool bad = false;
        for (int i = 1; i < n; ++i) {
            if (arr[i][j] < arr[i - 1][j]) {
                bad = true;
                break;
            }
        }
        ans += bad;
    }
    std::cout << ans << std::endl;
    return 0;
}
