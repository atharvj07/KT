import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] p = new int[2 * n];
        boolean[] vis = new boolean[2 * n];
        for (int i = 0; i < s.length(); i++) {
            if (s.charAt(i) == '0') {
                continue;
            }
            int cur = p[i / 2];
            if (vis[cur]) {
                System.out.println("No");
                return;
            }
            vis[cur] = true;
            int other = 2 * n - i - 1;
            p[i / 2] = p[other / 2];
            p[other / 2] = cur;
        }
        System.out.println("Yes");
        for (int i = 0; i < 2 * n; i++) {
            System.out.print(p[i] + 1 + " ");
        }
        System.out.println();
    }
}
