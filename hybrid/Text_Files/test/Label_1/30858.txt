import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int x = sc.nextInt();
        int y = sc.nextInt();
        int[] dist = new int[n + 1];
        for (int i = 1; i < n; i++) {
            dist[i + 1] = dist[i] + 1;
        }
        dist[y] = dist[x] + 1;
        for (int k = 1; k < n; k++) {
            int ans = 0;
            for (int i = 1; i < n; i++) {
                for (int j = i + 1; j <= n; j++) {
                    if (dist[j] - dist[i] == k) {
                        ans++;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
