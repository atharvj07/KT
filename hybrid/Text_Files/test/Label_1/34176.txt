import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int N = scanner.nextInt();
    long[] T = new long[N];
    long[] A = new long[N];
    for (int i = 0; i < N; i++) {
      T[i] = scanner.nextInt();
      A[i] = scanner.nextInt();
    }
    long x = 1;
    for (int i = 0; i < N; i++) {
      x = lcm(x, T[i], A[i]);
    }
    System.out.println(x);
  }

  static long lcm(long x, long y, long z) {
    return (x * y * z) / gcd(gcd(x, y), z);
  }

  static long gcd(long x, long y) {
    if (x % y == 0) {
      return y;
    } else {
      return gcd(y, x % y);
    }
  }
}
