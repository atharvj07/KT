import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] w1 = new int[4];
        int[] b1 = new int[4];
        int[] b2 = new int[4];
        for (int i = 0; i < w1.length; i++) {
            w1[i] = sc.nextInt();
        }
        for (int i = 0; i < b1.length; i++) {
            b1[i] = sc.nextInt();
        }
        for (int i = 0; i < b2.length; i++) {
            b2[i] = sc.nextInt();
        }
        boolean ans = false;
        for (int i = w1[0]; i <= w1[2]; i++) {
            for (int j = w1[1]; j <= w1[3]; j++) {
                if ((i < b1[0] || i > b1[2] || j < b1[1] || j > b1[3])
                        && (i < b2[0] || i > b2[2] || j < b2[1] || j > b2[3])) {
                    ans = true;
                    break;
                }
            }
        }
        System.out.println(ans ? "YES" : "NO");
        sc.close();
    }
}
