import java.util.Scanner;

class Main {
    static int MOD = 1000000007;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int h = scanner.nextInt();
        int w = scanner.nextInt();
        int[] r = new int[h];
        for (int i = 0; i < h; i++) {
            r[i] = scanner.nextInt();
        }
        int[] c = new int[w];
        for (int i = 0; i < w; i++) {
            c[i] = scanner.nextInt();
        }
        int res = 0;
        for (int i = 0; i < h; i++) {
            for (int j = 0; j < w; j++) {
                if (r[i] + c[j] == w + h - 1) {
                    res = (res + 1) % MOD;
                }
            }
        }
        System.out.println(res);
    }
}
