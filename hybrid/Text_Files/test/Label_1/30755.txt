import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    String str = sc.next();
    boolean isBad = false;
    for (int i = 1; i < str.length(); i++) {
      if (str.charAt(i) == str.charAt(i - 1)) {
        isBad = true;
        break;
      }
    }
    System.out.println(isBad ? "Bad" : "Good");
  }
}
