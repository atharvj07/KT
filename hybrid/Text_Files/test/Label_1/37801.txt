from math import gcd
T = int(input())
while T > 0:
  T -= 1
  a, b, c, d = map(int, input().split())
  if a == 0 and b == 0 and c == 0 and d == 0: print('Tidak Tidak Tidak Tidak')
  elif a != 0 and b == 0 and c == 0 and d == 0: print('Ya Tidak Tidak Tidak')
  elif a == 0 and b != 0 and c == 0 and d == 0: print('Tidak Ya Tidak Tidak')
  elif a == 0 and b == 0 and c != 0 and d == 0: print('Tidak Tidak Ya Tidak')
  elif a == 0 and b == 0 and c == 0 and d != 0: print('Tidak Tidak Tidak Ya')
  elif gcd(a, b) != 0 and c == 0 and d == 0: print('Ya Ya Tidak Tidak')
  elif gcd(a, c) != 0 and b == 0 and d == 0: print('Ya Tidak Ya Tidak')
  elif gcd(a, d) != 0 and b == 0 and c == 0: print('Ya Tidak Tidak Ya')
  elif gcd(b, c) != 0 and a == 0 and d == 0: print('Tidak Ya Ya Tidak')
  elif gcd(b, d) != 0 and a == 0 and c == 0: print('Tidak Ya Tidak Ya')
  elif gcd(c, d) != 0 and a == 0 and b == 0: print('Tidak Tidak Ya Ya')
  elif a + b == c + d: print('Ya Ya Ya Ya')
  elif a + b < c + d: print('Ya Ya Tidak Tidak')
  else: print('Tidak Tidak Ya Ya')
