import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] deg = new int[n + 1];
        List<Integer>[] adj = new List[n + 1];
        for (int i = 1; i <= n; i++) adj[i] = new ArrayList<>();
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            deg[a]++; deg[b]++;
            adj[a].add(b); adj[b].add(a);
        }
        int k = sc.nextInt();
        boolean flag = true;
        for (int i = 0; i < k; i++) {
            int a = sc.nextInt(), b = sc.nextInt();
            if (deg[a] == deg[b]) flag = false;
        }
        if (!flag) System.out.println("No");
        else {
            System.out.println("Yes");
            for (int i = 1; i <= n; i++) {
                if (deg[i] % 2 == 1) {
                    List<Integer> list = bfs(i, adj, new boolean[n + 1]);
                    for (int j = 0; j < list.size() - 1; j++) {
                        System.out.println(list.get(j) + " " + list.get(j + 1));
                    }
                }
            }
        }
    }

    static List<Integer> bfs(int start, List<Integer>[] adj, boolean[] vis) {
        List<Integer> list = new ArrayList<>();
        Queue<Integer> queue = new LinkedList<>();
        queue.add(start);
        vis[start] = true;
        while (!queue.isEmpty()) {
            int u = queue.poll();
            list.add(u);
            for (int v : adj[u]) {
                if (!vis[v]) {
                    queue.add(v);
                    vis[v] = true;
                }
            }
        }
        return list;
    }
}
