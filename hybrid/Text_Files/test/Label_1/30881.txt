import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), k = sc.nextInt();
        long[] a = new long[n];
        long sum = 0;
        int q = 0;
        for (int i = 0; i < n; i++) {
            String t = sc.next();
            if (t.equals("?")) {
                a[i] = 0;
                q++;
            } else {
                a[i] = Long.parseLong(t);
                sum += Math.abs(a[i]);
            }
        }
        long l = -sum - k + 1, r = sum + k - 1;
        if (q == 0) {
            for (int i = 1; i < k; i++) {
                if (a[i - 1] >= a[i]) {
                    System.out.println("Incorrect sequence");
                    return;
                }
            }
            for (int i = k; i < n; i++) {
                if (a[i - 1] + a[i - k] >= a[i]) {
                    System.out.println("Incorrect sequence");
                    return;
                }
            }
            for (long x : a) {
                System.out.print(x + " ");
            }
        } else {
            while (l <= r) {
                long m = (l + r) / 2;
                for (int i = 0, j = 0; i < n; i++) {
                    if (a[i] == 0) {
                        long p = m - sum - q + i + 1;
                        if (j == 0) {
                            if (p < 0 || p > k) {
                                break;
                            }
                        } else {
                            if (p < j || p > j + k) {
                                break;
                            }
                        }
                        a[i] = p;
                        j++;
                    }
                }
                boolean flag = true;
                for (int i = 1; i < k; i++) {
                    if (a[i - 1] >= a[i]) {
                        flag = false;
                    }
                }
                for (int i = k; i < n; i++) {
                    if (a[i - 1] + a[i - k] >= a[i]) {
                        flag = false;
                    }
                }
                if (flag) {
                    r = m - 1;
                    for (long x : a) {
                        System.out.print(x + " ");
                    }
                    return;
                } else {
                    l = m + 1;
                }
            }
            System.out.println("Incorrect sequence");
        }
    }
}
