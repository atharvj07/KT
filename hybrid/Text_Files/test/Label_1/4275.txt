#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 5;
int a[N], b[N], x[N], y[N];

int main() {
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i] >> x[i];
  cin >> m;
  for (int i = 1; i <= m; i++) cin >> b[i] >> y[i];
  sort(a + 1, a + 1 + n);
  sort(b + 1, b + 1 + m);
  long long ans = 0;
  for (int i = 1, j = 1; i <= n && j <= m;) {
    if (a[i] == b[j])
      i++, j++;
    else if (a[i] < b[j])
      ans += x[i], i++;
    else
      ans += y[j], j++;
  }
  for (int i = 1; i <= n; i++) ans += x[i];
  for (int i = 1; i <= m; i++) ans += y[i];
  cout << ans << endl;
  return 0;
}
