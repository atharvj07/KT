import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[][] fact = new long[n + 1][n + 1];
        fact[0][0] = 1;
        for (int i = 1; i <= n; i++) {
            fact[i][0] = 1;
            for (int j = 1; j <= n; j++) {
                fact[i][j] = fact[i - 1][j] * i % 1000000007;
            }
        }
        long ans = fact[n][n] * fact[n][n] % 1000000007;
        System.out.println(ans);
    }
}
