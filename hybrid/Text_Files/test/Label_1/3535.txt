#include <bits/stdc++.h>
using namespace std;
#define rep(i,a,b) for (int i = (a); i < (b); ++i)

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    string s;
    cin >> n >> k;
    cin >> s;
    string r = s;
    reverse(r.begin(), r.end());
    if (k + 1 <= n && s == r) cout << "YES" << endl;
    else cout << "NO" << endl;
  }
  return 0;
}
