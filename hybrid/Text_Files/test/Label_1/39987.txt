import heapq

def min_cost_encoding(n, c_0, c_1):
    # If there is only one letter, the cost is minimum of both costs
    if n == 1:
        return min(c_0, c_1)
    
    # Create a min heap
    heap = []
    
    # Push the costs into the heap
    heapq.heappush(heap, c_0)
    heapq.heappush(heap, c_1)
    
    # Initial cost is 0
    total_cost = 0
    
    # Run the loop n-1 times
    for _ in range(n-1):
        # Pop the two smallest elements from the heap
        smallest = heapq.heappop(heap)
        second_smallest = heapq.heappop(heap)
        
        # Add their sum to the total cost
        total_cost += smallest + second_smallest
        
        # Push their sum back into the heap
        heapq.heappush(heap, smallest + second_smallest)
    
    return total_cost

# Read input
n, c_0, c_1 = map(int, input().split())

# Calculate and print the minimum cost
print(min_cost_encoding(n, c_0, c_1))
