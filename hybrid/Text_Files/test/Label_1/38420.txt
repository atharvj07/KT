import sys
from collections import defaultdict

input = sys.stdin.readline
n = int(input())
x = list(map(int, input().split()))
d = defaultdict(list)
for i in range(n):
    for j in range(n):
        if i != j:
            k = x[i] ^ x[j]
            if k > 0 and (k & (k - 1) == 0):
                d[k].append((x[i], x[j]))
k = max(d.keys())
l = d[k]
print(len(l))
for i in l:
    print(*i)
