import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Main {
    private static class Pair {
        int hour;
        int minute;

        Pair(int hour, int minute) {
            this.hour = hour;
            this.minute = minute;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine());
        Pair[] arrivalTimes = new Pair[n];
        for (int i = 0; i < n; i++) {
            String[] time = sc.nextLine().split(" ");
            arrivalTimes[i] = new Pair(Integer.parseInt(time[0]), Integer.parseInt(time[1]));
        }
        System.out.println(solve(arrivalTimes));
    }

    private static int solve(Pair[] arrivalTimes) {
        int maxCashes = 0;
        int currentCashes = 0;
        Queue<Pair> queue = new LinkedList<>();
        for (Pair arrivalTime : arrivalTimes) {
            while (!queue.isEmpty() && queue.peek().hour < arrivalTime.hour) {
                queue.poll();
                currentCashes--;
            }
            while (!queue.isEmpty() && queue.peek().hour == arrivalTime.hour && queue.peek().minute <= arrivalTime.minute) {
                queue.poll();
                currentCashes--;
            }
            queue.offer(arrivalTime);
            currentCashes++;
            maxCashes = Math.max(maxCashes, currentCashes);
        }
        return maxCashes;
    }
}
