#include <bits/stdc++.h>
#define ll long long
using namespace std;
int main() {
    ll n, k, s, t;
    cin >> n >> k >> s >> t;
    vector<pair<ll, ll>> cars(n);
    for (auto &car : cars) cin >> car.first >> car.second;
    vector<ll> gas(k);
    for (int i = 0; i < k; i++) cin >> gas[i];
    sort(gas.begin(), gas.end());
    ll ans = LONG_LONG_MAX;
    for (auto &[cost, cap] : cars) {
        ll c = 0, d = 0;
        for (int i = 0; i < k && d < s; i++) {
            ll nd = min(s - d, gas[i] - d);
            ll fuel = nd / 2;
            if (cap >= fuel) {
                d += nd;
                c += nd / 2;
                t -= 2 * nd;
                if (t <= 0) break;
            } else {
                nd = cap * 2;
                d += nd;
                c += cap;
                t -= 2 * nd;
                if (t <= 0) break;
            }
        }
        if (d == s) ans = min(ans, cost);
    }
    cout << (ans == LLONG_MAX ? -1 : ans) << '\n';
    return 0;
}
