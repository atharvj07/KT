#include <bits/stdc++.h>

using namespace std;
typedef long long ll;
const int N = 3e5 + 10;
int n, ans[N];

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        char c;
        cin >> c;
        string s;
        cin >> s;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] != c) ans[++cnt] = i + 1;
        }
        cout << cnt << endl;
        for (int i = 1; i <= cnt; i++) cout << ans[i] << " ";
        cout << endl;
    }
    return 0;
}
