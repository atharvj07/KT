#include <bits/stdc++.h>
using namespace std;
#define int long long
int32_t main() {
    int t;
    cin >> t;
    while (t--) {
        int x, y;
        cin >> x >> y;
        x = abs(x), y = abs(y);
        vector<int> c(7, -1);
        for (int i = 1; i <= 6; i++) cin >> c[i];
        if (x + y <= 1) { cout << 0 << endl; continue; }
        int y1 = y / 2, y2 = y / 2 + (y % 2);
        int ans = 1e18;
        for (int i = 1; i <= 6; i++) {
            for (int j = 1; j <= 6; j++) {
                for (int k = 1; k <= 6; k++) {
                    for (int l = 1; l <= 6; l++) {
                        int temp_x = 1, temp_y = 1;
                        if (x > y) temp_x--;
                        if (i == 1) temp_x--;
                        if (j == 2) temp_x--, temp_y++;
                        if (k == 3) temp_y++;
                        if (l == 4) temp_x++, temp_y--;
                        if (temp_x == x && temp_y == y1)
                            ans = min(ans, c[i] + c[j] + c[k] + c[l]);
                        if (temp_x == x && temp_y == y2)
                            ans = min(ans, c[i] + c[j] + c[k] + c[l]);
                    }
                }
            }
        }
        cout << ans << endl;
    }
}
