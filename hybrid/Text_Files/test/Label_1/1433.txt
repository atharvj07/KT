#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    vector<int> ans(n, 0);
    for (int i = 0; i < n - 2; i++) {
        if (s[i] == 'o' && s[i + 1] == 'g' && s[i + 2] == 'o') {
            ans[i] = 1;
            ans[i + 1] = 1;
            ans[i + 2] = 1;
        }
    }

    for (int i = 0; i < n; i++) {
        if (ans[i]) cout << "*";
        else cout << s[i];
    }
    cout << endl;
}
