import java.util.Scanner;

public class Main {

    private static final int[][] DIR = {{1, 2}, {2, 1}, {2, 3}, {1, 4}, {4, 1}, {3, 2}, {2, 4}, {4, 2}};

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        String num = in.nextLine();
        int[][] visited = new int[5][3];
        char[] chars = num.toCharArray();
        for (int i = 0; i < n; i++) {
            int numI = chars[i] - '0';
            if (i != 0 && DIR[numI][0] == DIR[chars[i - 1] - '0'][0] && DIR[numI][1] == DIR[chars[i - 1] - '0'][1]) {
                System.out.println("NO");
                return;
            }
            visited[DIR[numI][0]][DIR[numI][1]]++;
            if (visited[DIR[numI][0]][DIR[numI][1]] > 1) {
                System.out.println("NO");
                return;
            }
        }
        System.out.println("YES");
    }
}
