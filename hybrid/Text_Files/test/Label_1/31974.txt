import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[][] buses = new int[n][3];
        for (int i = 0; i < n; i++) {
            buses[i][0] = scanner.nextInt();
            buses[i][1] = scanner.nextInt();
            buses[i][2] = scanner.nextInt();
        }
        int[][] people = new int[m][3];
        for (int i = 0; i < m; i++) {
            people[i][0] = scanner.nextInt();
            people[i][1] = scanner.nextInt();
            people[i][2] = scanner.nextInt();
        }
        int[] result = new int[m];
        for (int i = 0; i < m; i++) {
            result[i] = -1;
            for (int j = 0; j < n; j++) {
                if (buses[j][0] <= people[i][0] && people[i][1] <= buses[j][1] && people[i][2] <= buses[j][2]
                        && (result[i] == -1 || buses[j][2] < buses[result[i]][2])) {
                    result[i] = j;
                }
            }
            result[i]++;
        }
        for (int i = 0; i < m; i++) {
            System.out.print(result[i] + " ");
        }
        System.out.println();
    }
}
