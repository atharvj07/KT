n, m = map(int, input().split())
e = list(map(int, input().split()))
fr = []
for i in range(m): fr.append(list(map(int, input().split())))
d = {}
for i in range(1, n+1): d[i] = e[i-1]
ch = [0] * (m+1)
for x in fr:
    if d[x[0]] > 0:
        d[x[0]] -= 1
        ch[fr.index(x) + 1] = x[0]
    elif d[x[1]] > 0:
        d[x[1]] -= 1
        ch[fr.index(x) + 1] = x[1]
    else: break
if sum(ch) < m: print('DEAD')
else:
    print('ALIVE')
    for x in ch:
        if x: print(x, end=' ')
