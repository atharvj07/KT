import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class Main {
    private static final int MOD = 1000000007;
    private static int n, m;
    private static int[] a;
    private static boolean[] canBePlayer;
    private static Map<Integer, List<Integer>> leftAssistants;
    private static Map<Integer, List<Integer>> rightAssistants;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        a = new int[n];
        String[] line = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(line[i]);
        }
        m = Integer.parseInt(br.readLine());
        canBePlayer = new boolean[n];
        for (int i = 0; i < n; i++) {
            canBePlayer[i] = true;
        }
        leftAssistants = new HashMap<>();
        rightAssistants = new HashMap<>();
        for (int i = 0; i < n; i++) {
            leftAssistants.put(a[i], new ArrayList<>());
            rightAssistants.put(a[i], new ArrayList<>());
        }
        for (int i = 0; i < n; i++) {
            leftAssistants.get(a[i]).add(i);
            rightAssistants.get(a[i]).add(i);
        }
        long ans = 0;
        for (int i = 0; i < m; i++) {
            line = br.readLine().split(" ");
            int t = Integer.parseInt(line[0]);
            int x = Integer.parseInt(line[1]) - 1;
            if (t == 1) {
                canBePlayer[x] = false;
                int skill = a[x];
                leftAssistants.get(skill).remove(new Integer(x));
                rightAssistants.get(skill).remove(new Integer(x));
            } else {
                canBePlayer[x] = true;
                int skill = a[x];
                leftAssistants.get(skill).add(x);
                rightAssistants.get(skill).add(x);
            }
            long leftCount, rightCount;
            int maxPlayerSkill = 0;
            for (int j = 0; j < n; j++) {
                if (canBePlayer[j]) {
                    maxPlayerSkill = Math.max(maxPlayerSkill, a[j]);
                }
            }
            leftCount = rightCount = 0;
            for (int j = 0; j < n; j++) {
                if (canBePlayer[j] && a[j] == maxPlayerSkill) {
                    int leftIndex = j - 1;
                    int rightIndex = j + 1;
                    while (leftIndex >= 0 && canBePlayer[leftIndex]) {
                        if (a[leftIndex] < maxPlayerSkill) {
                            leftCount = (leftCount + leftAssistants.get(a[leftIndex]).size()) % MOD;
                        }
                        leftIndex--;
                    }
                    while (rightIndex < n && canBePlayer[rightIndex]) {
                        if (a[rightIndex] < maxPlayerSkill) {
                            rightCount = (rightCount + rightAssistants.get(a[rightIndex]).size()) % MOD;
                        }
                        rightIndex++;
                    }
                }
            }
            ans = (ans + leftCount * rightCount) % MOD;
            System.out.println(ans);
        }
    }
}
