import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        long[] arr = new long[n];
        long total = 0;
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextLong();
            total += arr[i];
        }
        if (total != 0) {
            System.out.println(-1);
            return;
        }
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] > 0) {
                ans += arr[i];
                arr[(i + 1) % n] -= arr[i];
            }
        }
        System.out.println(ans);
    }
}
