#include <iostream>
int main() {
  int t;
  std::cin >> t;
  while (t--) {
    int n;
    std::string s;
    std::cin >> n >> s;
    if (n >= 6 && (s[0] == '2' && s[1] == '0' && s[2] == '2' && s[3] == '0') || (s[n - 4] == '2' && s[n - 3] == '0' && s[n - 2] == '2' && s[n - 1] == '0'))
      std::cout << "YES" << std::endl;
    else
      std::cout << "NO" << std::endl;
  }
}
