import sys

input = sys.stdin.readline

N, K, D = map(int, input().split())
P = list(map(int, input().split()))
G = [[] for _ in range(N + 1)]

for _ in range(N - 1):
    a, b = map(int, input().split())
    G[a].append(b)
    G[b].append(a)

P.sort()
vis = [False] * (N + 1)
dis = [0] * (N + 1)
Q = []

for p in P:
    Q.append(p)
    vis[p] = True

ans = 0
road = []

while Q:
    x = Q.pop()
    for y in G[x]:
        if not vis[y]:
            Q.append(y)
            vis[y] = True
            dis[y] = dis[x] + 1
            if dis[y] > D:
                ans += 1
                road.append(x)

print(ans)
for x in road:
    print(x)
