#include <bits/stdc++.h>
using namespace std;
using ll = long long;
bool cmp(pair<ll,int>& a, pair<ll,int>& b) {
    return abs(a.first) < abs(b.first);
}
int main() {
    int n; cin >> n;
    vector<pair<ll,int>> v(n);
    for(int i = 0;i < n;++i) {
        cin >> v[i].first;
        v[i].second = i;
    }
    sort(v.rbegin(),v.rend(),cmp);
    ll sum = accumulate(v.begin(),v.end(),0LL,[](ll a, pair<ll,int>& p) { return a + p.first; });
    int left = n/2 - (sum/2 > 0);
    sum = 0;
    for(int i = 0;i < n;++i) {
        if(left && v[i].first < 0) --left, v[i].first++;
        sum += v[i].first;
        v[i].first = (v[i].first + 1) / 2;
    }
    sort(v.begin(),v.end(),[](pair<ll,int>& a, pair<ll,int>& b) { return a.second < b.second; });
    for(auto& it : v) cout << it.first << '\n';
    return 0;
}
