#include <iostream>
#include <vector>
using namespace std;

vector<int> a;
bool is_win[1000001];
int n, k;

int main() {
  cin >> n >> k;
  a.resize(n);
  for (int i = 0; i < n; i++) { cin >> a[i]; }
  for (int x = 1; x <= k; x++) {
    int y = x + x;
    is_win[y] = true;
    is_win[y / 2] = true;
  }
  int sum = 0;
  for (int i = 0; i < n; i++) sum ^= is_win[a[i]];
  puts(sum ? "Kevin" : "Nicky");
}
