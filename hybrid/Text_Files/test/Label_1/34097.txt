import java.util.*;
import java.io.*;

public class Main {
    static int n;
    static int[] a;
    static int[] b;
    static int[] c;
    static int ans;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        a = new int[n];
        b = new int[n];
        c = new int[n];
        for (int i = 0; i < n; i++) {
            String[] s = br.readLine().split(" ");
            a[i] = Integer.parseInt(s[0]);
            b[i] = (s.length > 1) ? Integer.parseInt(s[1]) : 0;
            c[i] = (s.length > 1) ? Integer.parseInt(s[1]) : 0;
        }
        ans = 0;
        int speed = b[0];
        int overtake = 0;
        for (int i = 1; i < n; i++) {
            if (a[i] == 1)
                speed = b[i];
            else if (a[i] == 2) {
                if (overtake == 1)
                    ans++;
            } else if (a[i] == 3)
                speed = 0;
            else if (a[i] == 4)
                overtake = 1;
            else if (a[i] == 5)
                speed = c[i];
            else if (a[i] == 6)
                overtake = 0;
        }
        System.out.println(ans);
    }
}
