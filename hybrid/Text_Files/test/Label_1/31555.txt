import java.util.*;
import java.io.*;
import java.math.BigInteger;

class Main {
    static final long MOD = 1000000007;
    static class Point implements Comparable<Point> {
        int x, y;
        public Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
        public int compareTo(Point other) {
            if (this.x != other.x) {
                return Integer.compare(this.x, other.x);
            }
            return Integer.compare(this.y, other.y);
        }
    }
    public static void main(String[] args) {
        try (BufferedReader br = new BufferedReader(new InputStreamReader(System.in))) {
            int n = Integer.parseInt(br.readLine());
            Point[] points = new Point[n];
            for (int i = 0; i < n; i++) {
                StringTokenizer st = new StringTokenizer(br.readLine());
                points[i] = new Point(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
            }
            Arrays.sort(points);
            long minFee = Long.MAX_VALUE;
            long numArrangements = 0;
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    long fee = (long) Math.abs(points[i].x - points[j].x) + Math.abs(points[i].y - points[j].y);
                    if (fee < minFee) {
                        minFee = fee;
                        numArrangements = 0;
                    }
                    if (fee == minFee) {
                        numArrangements = (numArrangements + 1) % MOD;
                    }
                }
            }
            System.out.println(minFee);
            System.out.println(numArrangements);
        } catch (Exception e) {
            System.exit(-1);
        }
    }
}
