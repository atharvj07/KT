#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        string s;
        cin >> s;
        int ans = 0;
        int cur = 0;
        for (auto& c : s) {
            if (c == '(')
                ++cur;
            else
                --cur;
            ans = max(ans, abs(cur));
        }
        cout << ans << endl;
    }
    return 0;
}
