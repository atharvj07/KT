import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] r = new int[2], c = new int[2], d = new int[2];
        for (int i = 0; i < 2; i++) {
            r[i] = sc.nextInt();
        }
        for (int i = 0; i < 2; i++) {
            c[i] = sc.nextInt();
        }
        for (int i = 0; i < 2; i++) {
            d[i] = sc.nextInt();
        }
        System.out.println((d[0] + d[1] - r[0] - r[1] == c[0] - r[0] - r[1]) ? (d[0] - r[0] - c[0]) + " " + (d[1] - r[1] - c[0]) : -1);
    }
}
