#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    auto ang = [&](int x1, int y1, int x2, int y2) {
      return atan2(y2 - y1, x2 - x1) * 180.0 / 3.14159265358979323846;
    };
    bool vis[n];
    memset(vis, false, sizeof vis);
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int x1, y1, x2, y2;
      cin >> x1 >> y1 >> x2 >> y2;
      double a1 = ang(x1, y1, x2, y2);
      double a2 = ang(x1, y1, x1, y1);
      for (int j = 0; j < n; j++) {
        if (i == j) continue;
        int x3, y3, x4, y4;
        cin >> x3 >> y3 >> x4 >> y4;
        double b1 = ang(x3, y3, x4, y4);
        double b2 = ang(x3, y3, x1, y1);
        if ((a1 < a2 && b1 < b2) || (a2 < a1 && b2 < b1)) continue;
        if ((a1 < a2 && b1 < a2) || (a2 < a1 && b2 < a1)) continue;
        if ((a1 < a2 && b2 < a2) || (a2 < a1 && b1 < a1)) continue;
        if ((a1 < a2 && a2 < b1 && b2 < 360) || (a2 < a1 && b1 < a1 && b2 < 360))
          continue;
        if ((a1 < a2 && b1 > 360 && b2 < 360) || (a2 < a1 && b1 < 360 && b2 > 360))
          continue;
        if ((a1 < a2 && b2 > 360) || (a2 < a1 && b1 > 360)) continue;
        if ((a1 > a2 && b1 > a2 && b2 < 360) || (a2 > a1 && b1 < a1 && b2 < 360))
          continue;
        if ((a1 > a2 && b1 > 360 && b2 < 360) || (a2 > a1 && b1 < 360 && b2 > 360))
          continue;
        if ((a1 > a2 && b2 > 360) || (a2 > a1 && b1 > 360)) continue;
        if (!vis[j]) ans++;
        vis[j] = true;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
