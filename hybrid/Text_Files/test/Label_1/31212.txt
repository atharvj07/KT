import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

public class GoodString {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        scanner.nextLine();
        String[] strings = new String[n];
        for (int i = 0; i < n; i++) {
            strings[i] = scanner.nextLine();
        }

        String goodString = findGoodString(strings);
        System.out.println(goodString);
    }

    private static String findGoodString(String[] strings) {
        Map<String, Integer> substringCounts = new HashMap<>();
        for (String string : strings) {
            for (int i = 0; i < string.length(); i++) {
                for (int j = i + 1; j <= string.length(); j++) {
                    String substring = string.substring(i, j);
                    substringCounts.put(substring, substringCounts.getOrDefault(substring, 0) + 1);
                }
            }
        }

        int maxFrequency = 0;
        for (int frequency : substringCounts.values()) {
            maxFrequency = Math.max(maxFrequency, frequency);
        }

        List<String> goodSubstrings = new ArrayList<>();
        for (Map.Entry<String, Integer> entry : substringCounts.entrySet()) {
            if (entry.getValue() == maxFrequency) {
                goodSubstrings.add(entry.getKey());
            }
        }

        if (goodSubstrings.isEmpty()) {
            return "NO";
        }

        Collections.sort(goodSubstrings);
        StringBuilder goodString = new StringBuilder();
        for (String goodSubstring : goodSubstrings) {
            goodString.append(goodSubstring);
        }

        return goodString.toString();
    }
}
