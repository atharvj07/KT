#include<bits/stdc++.h>
using namespace std;
int main(){
  int t;
  cin>>t;
  while(t--){
    int n;
    cin>>n;
    int mx=-(1e9),mn=(1e9);
    for(int i=0;i<n;i++){
      int x;
      cin>>x;
      mx=max(x,mx);
      mn=min(x,mn);
    }
    int ans=mx - mn;
    if(ans==0){
      cout<<-1<<endl;
    } else {
      cout<<ans<<endl;
    }
  }
  return 0;
}
