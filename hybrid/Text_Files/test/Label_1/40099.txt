values = list(map(int, input().split()))
s = input()

letter_values = {chr(ord('a') + i): val for i, val in enumerate(values)}

prefix_sums = [0]
current_sum = 0
for char in s:
    current_sum += letter_values[char]
    prefix_sums.append(current_sum)

interesting_substr_count = 0
letter_occurrences = {}

for i, char in enumerate(s):
    if char in letter_occurrences:
        for start_index in letter_occurrences[char]:
            if prefix_sums[i] == prefix_sums[start_index]:
                interesting_substr_count += 1
    if char not in letter_occurrences:
        letter_occurrences[char] = []
    letter_occurrences[char].append(i + 1)

print(interesting_substr_count)
