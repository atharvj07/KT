import sys
input = sys.stdin.readline

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

ma = -10**9
mi = 10**9
ans = 0
for i in range(n):
    mi = min(mi, b[i])
    if mi <= a[i]:
        ans += 1
    ma = max(ma, a[i])
    if mi > ma:
        mi = ma
print(ans)
