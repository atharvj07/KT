#include <iostream>
#include <queue>
#include <tuple>
using namespace std;

int main() {
  long long N, P;
  cin >> N >> P;
  priority_queue<tuple<long long, long long, long long>> pq;  // (a_i, b_i, a_i - b_i)
  while (N--) {
    long long A, B;
    cin >> A >> B;
    long long diff = A - B;
    if (diff > 0) pq.emplace(-diff, B, A);
  }
  long double ans = 0;
  while (!pq.empty()) {
    long long diff, B, A;
    tie(diff, B, A) = pq.top();
    pq.pop();
    if (B >= -diff) {
      ans += (double)B / diff;
      B += P * (B / diff);
      pq.emplace(-diff, B, A);
    }
  }
  cout.precision(11);
  cout << (ans < 1e9 ? ans : -1) << '\n';
}
