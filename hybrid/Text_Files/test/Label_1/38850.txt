for t in range(int(input())):
    n = int(input())
    s = input()
    l = [0] * n
    r = [0] * n
    for i in range(n):
        if s[i] == 'L':
            l[i] += 1
        else:
            r[i] += 1
    for i in range(1, n):
        l[i] += l[i - 1]
    for i in range(n - 2, -1, -1):
        r[i] += r[i + 1]
    ans = n
    for i in range(n):
        if l[i] >= 1 and r[i] >= 1:
            ans = min(ans, l[i] + r[i] - 1)
    for i in range(n):
        if l[i] == 0 and r[i] == 0:
            ans = min(ans, i + r[(i + 1) % n])
    print(ans)
