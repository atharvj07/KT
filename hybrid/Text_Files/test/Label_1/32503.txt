import java.util.Scanner;
import java.util.ArrayList;

class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = input.nextInt();
        }
        ArrayList<Integer> parents = new ArrayList<>();
        parents.add(-1);
        for (int i = 1; i < n; i++) {
            int parent = 0;
            while (true) {
                if (a[i] > a[parent]) {
                    if (parent * 2 + 2 < parents.size() && parents.get(parent * 2 + 2) != -1) {
                        parent = parent * 2 + 2;
                    } else {
                        parents.add(parent);
                        break;
                    }
                } else {
                    if (parent * 2 + 1 < parents.size() && parents.get(parent * 2 + 1) != -1) {
                        parent = parent * 2 + 1;
                    } else {
                        parents.add(parent);
                        break;
                    }
                }
            }
        }
        for (int i = 1; i < n; i++) {
            System.out.println(a[parents.get(i)]);
        }
    }
}
