import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class HarmonySequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = scanner.nextInt();
        }

        List<Integer> b = new ArrayList<>();
        int[] gcd = new int[n];
        gcd[0] = a[0];
        for (int i = 1; i < n; i++) {
            gcd[i] = gcd(gcd[i - 1], a[i]);
        }

        int[] lcm = new int[n];
        lcm[n - 1] = a[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            lcm[i] = lcm(lcm[i + 1], a[i]);
        }

        for (int i = 0; i < n; i++) {
            if (gcd[i] == 1 && lcm[i] == a[i]) {
                b.add(a[i]);
            }
        }

        if (b.size() < n) {
            int minIndex = 0;
            for (int i = 1; i < n; i++) {
                if (a[i] < a[minIndex]) {
                    minIndex = i;
                }
            }
            for (int i = 0; i < n; i++) {
                if (gcd(a[minIndex], a[i]) == 1) {
                    b.add(a[i]);
                    break;
                }
            }
        }

        while (b.size() < n) {
            int maxIndex = 0;
            for (int i = 1; i < n; i++) {
                if (a[i] > a[maxIndex]) {
                    maxIndex = i;
                }
            }
            b.add(a[maxIndex]);
        }

        for (int i : b) {
            System.out.print(i + " ");
        }
    }

    static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }

    static int lcm(int a, int b) {
        return a * b / gcd(a, b);
    }
}
