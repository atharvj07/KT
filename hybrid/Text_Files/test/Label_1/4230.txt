#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        vector<int> a(n);
        for (auto &x : a) cin >> x;
        sort(a.begin(), a.end());
        long long ans = 1LL * a[0] * a[1] * a[2] * a[3] * a[4];
        cout << ans << '\n';
    }
    return 0;
}
