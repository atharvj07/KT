#include<iostream>
using namespace std;

int main(){
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        bool lst = false, lst2 = false;
        int h = 1;
        while(n--){
            int a;
            cin >> a;
            if(a){
                if(!lst){
                    h++;
                }
                else{
                    h += 5;
                }
            }
            lst2 = lst;
            lst = a;
        }
        if(lst2 && !lst) cout << -1 << endl;
        else cout << h << endl;
    }
}
