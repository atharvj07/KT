#include<bits/stdc++.h>
using namespace std;
int n;
int k;
vector<int> a;
int ans;
vector<int> b;
void solve(){
	b[0]=a[0]+max(0,k-1);
	ans+=max(0,k-1);
	for(int i=1; i<n; i++){
		if(a[i]<b[i-1]) b[i]=b[i-1],ans+=b[i-1]-a[i];
		else b[i]=a[i]+max(0,k-b[i-1]-1);
		ans+=max(0,k-b[i-1]-1);
	}
}
int main(){
	cin>>n>>k;
	a.resize(n);
	for(int i=0; i<n; i++) cin>>a[i];
	b.resize(n);
	solve();
	cout<<ans<<endl;
	for(int i=0; i<n; i++) cout<<b[i]<<' ';
	cout<<endl;
	return 0;
}
