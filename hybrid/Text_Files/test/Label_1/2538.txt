#include <iostream>

int main() {
  long long n, k;
  std::cin >> n >> k;
  if (k == 2) {
    std::cout << -1 << std::endl;
  } else if (n == 1) {
    std::cout << 0 << std::endl;
  } else if (n <= k) {
    std::cout << 1 << std::endl;
  } else if ((n - 1) % (k - 1) == 0) {
    std::cout << (n - 1) / (k - 1) + 1 << std::endl;
  } else {
    std::cout << -1 << std::endl;
  }
  return 0;
}
