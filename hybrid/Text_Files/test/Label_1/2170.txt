#include <bits/stdc++.h>

using namespace std;

const int64_t MOD = 998244353;
const int64_t MOD2 = 19260817;

int64_t binpow(int64_t base, int64_t exp) {
  int64_t result = 1;
  while (exp > 0) {
    if ((exp & 1) == 1) {
      result = (result * base) % MOD;
    }
    base = (base * base) % MOD;
    exp /= 2;
  }
  return result;
}

int64_t fast_mul(int64_t a, int64_t b) { return ((a % MOD2) * (b % MOD2)) % MOD; }

int main() {
  int64_t n;
  cin >> n;
  vector<int64_t> a(n);
  for (int64_t i = 0; i < n; ++i) { cin >> a[i]; }

  int64_t sum = 0;
  for (int64_t i = 0; i < n; ++i) {
    sum = (sum + (a[i] * binpow(2, i)) % MOD) % MOD;
  }

  int64_t ans = fast_mul(sum, binpow(2, n - 1));

  cout << ans << endl;

  return 0;
}
