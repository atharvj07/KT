#include <bits/stdc++.h>

using namespace std;

#define ll long long
const ll MOD = 1e9 + 7;

int main() {
  int n;
  cin >> n;

  vector<int> a(n);
  for (auto &x : a) cin >> x;

  ll sum = 0;
  for (auto &x : a) sum += x;

  if (sum % 2 == 0) {
    cout << 0 << endl;
    return 0;
  }

  int ans = 0;
  for (int i = 0; i <= 5; i++) {
    if (i % 2 == sum % 2) {
      ans++;
    }
  }

  cout << ans << endl;

  return 0;
}
