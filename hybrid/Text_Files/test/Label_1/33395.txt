import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();
        int[] ans = new int[q];

        for (int i = 0; i < q; i++) {
            int n = sc.nextInt();
            if (n < 10) {
                ans[i] = 0;
            } else {
                int count = 0;
                while (n > 9) {
                    String s = String.valueOf(n);
                    int max = 0;
                    for (int left = 1; left <= s.length() / 2; left++) {
                        int a = Integer.parseInt(s.substring(0, left));
                        int b = Integer.parseInt(s.substring(left));
                        if (a * b > max) {
                            max = a * b;
                        }
                    }
                    ans[i] = count;
                    count++;
                    n = max;
                }
            }
        }

        for (int a : ans) {
            System.out.println(a);
        }
    }
}
