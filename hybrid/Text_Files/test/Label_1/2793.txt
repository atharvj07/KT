#include <vector>
#include <algorithm>
#include <iostream>

int main() {
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) { std::cin >> a[i]; }

  std::sort(a.begin(), a.end());

  int median;
  if (k % 2 == 0) {
    median = (a[k / 2 - 1] + a[k / 2]) / 2;
  } else {
    median = a[k / 2];
  }

  std::cout << median << std::endl;
  return 0;
}
