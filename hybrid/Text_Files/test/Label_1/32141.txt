import java.util.*;

class Main {
    static class Training {
        int e;
        int c;
    }

    static class Donut {
        int h;
        int a;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int s = sc.nextInt();
        int t = sc.nextInt();
        int u = sc.nextInt();
        int n = sc.nextInt();
        int o = sc.nextInt();
        int d = sc.nextInt();

        Training[] trainings = new Training[t];
        for (int i = 0; i < t; i++) {
            trainings[i] = new Training();
            trainings[i].e = sc.nextInt();
            trainings[i].c = sc.nextInt();
        }

        Donut[] donuts = new Donut[n];
        for (int i = 0; i < n; i++) {
            donuts[i] = new Donut();
            donuts[i].h = sc.nextInt();
            donuts[i].a = sc.nextInt();
        }

        int[][] dp = new int[d + 1][s + 1];

        for (int i = 0; i <= d; i++) {
            for (int j = 0; j <= s; j++) {
                dp[i][j] = -1;
            }
        }

        dp[0][s] = 0;
        for (int i = 0; i < d; i++) {
            for (int j = 0; j <= s; j++) {
                if (dp[i][j] == -1) continue;

                // Train
                for (Training training : trainings) {
                    if (j < training.e) continue;
                    int next = Math.min(s, j + o - training.e);
                    dp[i + 1][next] = Math.max(dp[i + 1][next], dp[i][j] + training.c);
                }

                // Eat
                for (Donut donut : donuts) {
                    int next = Math.max(0, j - donut.a);
                    dp[i + 1][j] = Math.max(dp[i + 1][j], dp[i][j] + donut.h);
                }
            }
        }

        int max = 0;
        for (int i = 1; i <= d; i++) {
            for (int j = 0; j <= s; j++) {
                if (dp[i][j] != -1) {
                    max = Math.max(max, dp[i][j]);
                }
            }
        }

        if (max == 0) {
            System.out.println(-1);
            return;
        }

        System.out.println(max);
    }
}
