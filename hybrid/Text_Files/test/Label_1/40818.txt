import sys

MOD = 998244353

n = int(input())
portals = []
for _ in range(n):
    portals.append(list(map(int, input().split())))

portals.sort(key=lambda x: x[0])

cur = 0
time = 0
for x, y, s in portals:
    time += x - cur
    if s:
        cur = y
    else:
        cur = x

time += portals[-1][0] + 1 - cur
print(time % MOD)
