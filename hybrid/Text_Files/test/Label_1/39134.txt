def is_berlanese(s):
    vowels = "aeiou"
    for i in range(len(s) - 1):
        if s[i] not in vowels and s[i] != "n":
            if s[i+1] not in vowels:
                return "NO"
    if s[-1] not in vowels and s[-1] != "n":
        return "NO"
    return "YES"

s = input().strip()
print(is_berlanese(s))
