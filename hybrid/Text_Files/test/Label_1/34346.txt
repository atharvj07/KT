import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
    private static final int INF = 1 << 30;
    private static int n, m;
    private static Map<Integer, List<Integer>> adjList;
    private static int[] distTo1, distToN, parTo1, parToN;
    private static boolean[] isVisited1, isVisitedN;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());

        adjList = new HashMap<>();
        for (int i = 0; i <= n; i++) {
            adjList.put(i, new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            adjList.get(u).add(v);
            adjList.get(v).add(u);
        }

        distTo1 = new int[n + 1];
        distToN = new int[n + 1];
        parTo1 = new int[n + 1];
        parToN = new int[n + 1];
        isVisited1 = new boolean[n + 1];
        isVisitedN = new boolean[n + 1];
        boolean isReachable = bfs1() && bfsN();
        if (!isReachable) {
            System.out.println(-1);
            return;
        }

        int curr1 = n, curr2 = 1;
        List<Integer> path1 = new ArrayList<>();
        List<Integer> path2 = new ArrayList<>();
        while (curr1 != 1) {
            path1.add(curr1);
            curr1 = parTo1[curr1];
        }
        path1.add(1);
        while (curr2 != n) {
            path2.add(curr2);
            curr2 = parToN[curr2];
        }
        path2.add(n);

        int k = path1.size() - 1;
        System.out.println(k);
        for (int i = 0; i < k; i++) {
            System.out.print(path2.get(i) + " ");
        }
        System.out.println();
        for (int i = k - 1; i >= 0; i--) {
            System.out.print(path1.get(i) + " ");
        }
        System.out.println();
    }

    private static boolean bfs1() {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(1);
        isVisited1[1] = true;

        while (!queue.isEmpty()) {
            int curr = queue.poll();

            for (int next : adjList.get(curr)) {
                if (!isVisited1[next]) {
                    queue.add(next);
                    isVisited1[next] = true;
                    distTo1[next] = distTo1[curr] + 1;
                    parTo1[next] = curr;
                }
            }
        }

        return isVisited1[n];
    }

    private static boolean bfsN() {
        Queue<Integer> queue = new LinkedList<>();
        queue.add(n);
        isVisitedN[n] = true;

        while (!queue.isEmpty()) {
            int curr = queue.poll();

            for (int next : adjList.get(curr)) {
                if (!isVisitedN[next]) {
                    queue.add(next);
                    isVisitedN[next] = true;
                    distToN[next] = distToN[curr] + 1;
                    parToN[next] = curr;
                }
            }
        }

        return isVisitedN[1];
    }
}
