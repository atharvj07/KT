#include <iostream>
#include <vector>

using namespace std;

int main() {
  long long n;
  cin >> n;
  vector<int> digits;
  while (n > 0) {
    digits.push_back(n % 10);
    n /= 10;
  }
  int sum = 0;
  for (int digit : digits) {
    sum += digit;
  }
  int maxSum = 0;
  for (int i = 0; i < digits.size(); i++) {
    for (int j = i + 1; j < digits.size(); j++) {
      int newSum = sum - digits[i] - digits[j] + (9 - digits[i]) + (9 - digits[j]);
      maxSum = max(maxSum, newSum);
    }
  }
  cout << maxSum << endl;
  return 0;
}
