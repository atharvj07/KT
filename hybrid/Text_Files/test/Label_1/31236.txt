import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        long x = input.nextLong();
        int left = 0, distressed = 0;
        for (int i = 0; i < n; i++) {
            String sign = input.next();
            int d = input.nextInt();
            if (sign.equals("+")) {
                left += d;
            } else if (x >= d) {
                x -= d;
            } else {
                distressed++;
            }
        }
        System.out.println(x + " " + distressed);
    }
}
