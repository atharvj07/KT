#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int a[N], b[N], seg[4 * N];
void build(int s, int e, int n) {
  if (s == e) {
    seg[n] = a[s];
    return;
  }
  int mid = (s + e) / 2;
  build(s, mid, 2 * n);
  build(mid + 1, e, 2 * n + 1);
  seg[n] = seg[2 * n] + seg[2 * n + 1];
}
void update(int s, int e, int n, int p, int v) {
  if (p < s || e < p) return;
  if (s == e) {
    seg[n] = v;
    return;
  }
  int mid = (s + e) / 2;
  update(s, mid, 2 * n, p, v);
  update(mid + 1, e, 2 * n + 1, p, v);
  seg[n] = seg[2 * n] + seg[2 * n + 1];
}
int query(int s, int e, int n, int l, int r) {
  if (l <= s && e <= r) return seg[n];
  if (s > r || e < l) return 0;
  int mid = (s + e) / 2;
  return query(s, mid, 2 * n, l, r) + query(mid + 1, e, 2 * n + 1, l, r);
}
int main() {
  int n, m, x, y, k, type;
  cin >> n >> m;
  for (int i = 1; i <= n; i++) cin >> a[i];
  for (int i = 1; i <= n; i++) cin >> b[i];
  build(1, n, 1);
  while (m--) {
    cin >> type;
    if (type == 1) {
      cin >> x >> y >> k;
      for (int i = 0; i < k; i++) update(1, n, 1, y + i, a[x + i]);
    } else {
      cin >> x;
      cout << query(1, n, 1, x, x) << '\n';
    }
  }
  return 0;
}
