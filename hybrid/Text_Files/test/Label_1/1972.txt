#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
#define ll long long
const ll linf = 1e17;
vector<ll> d,a,c;
int main() {
  int n, ans=0; cin>>n;
  d.resize(n+1);
  a.resize(n+1);
  c.resize(n+1);
  for(int i=1; i<=n; ++i) { cin>>c[i]; }
  for(int i=1; i<=n; ++i) { cin>>a[i]; }
  for(int i=1; i<=n; ++i) {
    d[i] = c[i];
    if(d[i]<linf) for(int j=a[i], cnt=1; cnt<=n; ++cnt, j=a[j]) {
      d[i] = min(d[i], c[j]+d[j]);
    }
    if(d[i]==linf) d[i]=-1; else ans+=d[i];
  }
  cout<<ans<<endl;
}
