#include <cstdio>
#include <algorithm>
using namespace std;
const int N = 171;
int n, w[N][N];
double x[N][N];

inline int sgn(double x) { return x < -1e-9 ? -1 : (x > 1e-9); }

int main() {
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) scanf("%d", &w[i][j]), x[i][j] = w[i][j];
    for (int p = 0; p < n; p++) {
        if (x[p][p] == 0) {
            int row = p;
            while (row < n && x[row][p] == 0) row++;
            if (row == n) continue;
            for (int j = 0; j < n; j++) swap(x[p][j], x[row][j]);
        }
        for (int i = 0; i < n; i++) {
            if (i == p) continue;
            while (x[i][p] && x[p][p]) {
                double tmp = x[i][p] / x[p][p];
                for (int j = 0; j < n; j++) { x[i][j] -= tmp * x[p][j]; }
            }
        }
    }
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            printf("%d%c", sgn(x[i][j]) * w[i][j], " \n"[j == n - 1]);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            printf("%d%c", w[i][j] - sgn(x[i][j]) * w[i][j], " \n"[j == n - 1]);
    return 0;
}
