n = int(input())
a = list(map(int, input().split()))

if a[0] == 0:
    print("YES")
    print("->".join(map(str, a)))
elif n == 1 or a[-1] == 1:
    print("NO")
else:
    first_zero = -1
    for i in range(n-1, -1, -1):
        if a[i] == 0:
            first_zero = i
            break
    if first_zero == -1 or first_zero == n-1:
        print("NO")
    else:
        print("YES")
        expression = "->".join(map(str, a[:first_zero])) + "->(" + "->".join(map(str, a[first_zero:])) + ")"
        print(expression)
