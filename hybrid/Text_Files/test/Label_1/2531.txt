#include <bits/stdc++.h>
using namespace std;

inline int read() {
  int x = 0;
  char c = getchar();
  while (c < '0' || c > '9') c = getchar();
  while (c >= '0' && c <= '9') x = x * 10 + (c - '0'), c = getchar();
  return x;
}

int a[55];

int solve(int n) {
  int p1 = 0, p2 = 0, ans = 0;
  for (int i = 0; i < n; i++) {
    if (a[i] == 1)
      p1++;
    else
      p2++;
    if (p1 < p2) {
      swap(a[i], a[i - 1]);
      ans++;
    }
  }
  return ans;
}

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    int n = read();
    for (int i = 0; i < n; i++) a[i] = getchar() == '(';
    printf("%d\n", solve(n));
  }
  return 0;
}
