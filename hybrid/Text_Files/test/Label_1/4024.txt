#include <bits/stdc++.h>

using namespace std;

#define fastio ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);

int main() {
    fastio;
    int n;
    cin >> n;
    vector<int> a(n);
    for (int &x : a) {
        cin >> x;
    }
    set<int> st;
    for (int x : a) {
        st.insert(x);
    }
    double ans = (double)st.size();
    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {
            if (a[i] == a[j]) {
                ans -= 1.0 / (j - i + 1);
            }
        }
    }
    cout << setprecision(6) << fixed << ans << "\n";
    return 0;
}
