import java.util.Scanner;

public class MinimalLength {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        String s1 = scanner.nextLine();
        String s2 = scanner.nextLine();
        String s3 = scanner.nextLine();

        int minLength = s1.length() + s2.length() + s3.length() - 2;

        if (s1.contains(s2)) {
            minLength = Math.max(minLength, s1.length() + s3.length());
        }

        if (s1.contains(s3)) {
            minLength = Math.max(minLength, s2.length() + s3.length());
        }

        if (s2.contains(s3)) {
            minLength = Math.max(minLength, s1.length() + s2.length());
        }

        System.out.println(minLength);
    }
}
