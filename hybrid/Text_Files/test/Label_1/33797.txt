import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.math.BigInteger;

class Main {
    static final BigInteger MOD = BigInteger.valueOf(1000000007);
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        int n = Integer.parseInt(tokens[0]);
        BigInteger mod = new BigInteger(tokens[1]);
        BigInteger[] fact = new BigInteger[n + 2];
        fact[0] = BigInteger.ONE;
        for (int i = 1; i <= n + 1; i++) {
            fact[i] = fact[i - 1].multiply(BigInteger.valueOf(i));
        }
        BigInteger ans = BigInteger.ZERO;
        for (int p = 1; p <= n; p++) {
            for (int q = p + 1; q <= n; q++) {
                BigInteger inv = fact[n].divide(fact[n - p].multiply(fact[p - 1]));
                ans = ans.add(inv.multiply(inv).multiply(fact[q - 1]).multiply(fact[n - q]));
            }
        }
        System.out.println(ans.mod(mod));
    }
}
