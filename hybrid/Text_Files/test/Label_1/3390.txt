#include <bits/stdc++.h>
using namespace std;
const int MAXN = 1005;
unordered_map<int, int> f;
int n, c[MAXN], a[MAXN], l, r, ans;
int main() {
  cin >> n;
  for (int i = 1; i <= n; ++i) cin >> a[i];
  for (l = 1, r = 1; r <= n;) {
    while (r <= n && f[a[r]] <= f[a[l]]) f[a[r++]]++;
    if (r - l > ans) ans = r - l;
    f[a[l]]--;
    l++;
  }
  cout << ans << endl;
  return 0;
}
