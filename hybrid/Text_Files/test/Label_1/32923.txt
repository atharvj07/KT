import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

        String[] line = reader.readLine().split(" ");
        int n = Integer.parseInt(line[0]);
        long mod = Integer.parseInt(line[1]);

        long[][] dp = new long[n][n];

        // dp[i][j] is the number of ways to arrange the first i elements of the permutation
        // such that there are j inversions
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == 0) {
                    dp[i][j] = 1;
                } else if (j == 0) {
                    dp[i][j] = dp[i - 1][j] * i;
                } else {
                    dp[i][j] = (dp[i - 1][j] * i + dp[i - 1][j - 1] * (i - 1)) % mod;
                }
            }
        }

        long answer = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i > j) {
                    answer = (answer + dp[i][j] * dp[n - 1 - i][n - 1 - j]) % mod;
                }
            }
        }

        System.out.println(answer);
    }
}
