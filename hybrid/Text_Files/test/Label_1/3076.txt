#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,m,sx,sy,ex,ey,v[505][505];
    cin>>n>>m;
    for(int i=0;i<n;i++)
        for(int j=0;j<m;j++){
            char c;
            cin>>c;
            v[i][j]=(c=='.');
        }
    cin>>sx>>sy>>ex>>ey;
    sx--,sy--,ex--,ey--;
    queue<pair<int,int>> q;
    v[sx][sy]=2;
    q.push({sx,sy});
    while(!q.empty()){
        tie(sx,sy)=q.front();
        q.pop();
        if(sx==ex&&sy==ey){
            cout<<"YES"<<endl;
            return 0;
        }
        for(int dx=-1;dx<=1;dx++)
            for(int dy=-1;dy<=1;dy++){
                int nx=sx+dx,ny=sy+dy;
                if(nx<0||nx>=n||ny<0||ny>=m||v[nx][ny]==2)
                    continue;
                if(v[nx][ny]==1){
                    v[nx][ny]=2;
                    q.push({nx,ny});
                }
            }
    }
    cout<<"NO"<<endl;
    return 0;
}
