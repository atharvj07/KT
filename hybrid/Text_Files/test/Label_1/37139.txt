import sys

input = sys.stdin.readline

n, k = map(int, input().split())
a = list(map(int, input().split()))
if k != 0:
    b = list(map(int, input().split()))
else:
    b = []

b_set = set(b)

ans = 0
prev = -1
for i in range(n):
    if i in b_set:
        continue
    if a[i] <= prev:
        ans += 1
        a[i] = prev + 1
    prev = a[i]

if prev >= 1e9:
    print(-1)
else:
    print(ans)
