#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9 + 7;
int main() {
  string s;
  cin >> s;
  int x, y;
  cin >> x >> y;
  int n = s.size();
  ll dp[n + 10][3][3];
  memset(dp, 0, sizeof(dp));
  dp[n][0][0] = 0;
  dp[n][1][0] = x;
  dp[n][0][1] = y;
  dp[n][1][1] = x + y;
  for (int i = n - 1; i >= 0; i--) {
    for (int x_last = 0; x_last < 2; x_last++) {
      for (int y_last = 0; y_last < 2; y_last++) {
        int ch = s[i] - '0';
        if (ch == 0) {
          dp[i][x_last][y_last] = min(0ll + dp[i + 1][x_last][0],
                                       x + dp[i + 1][x_last][1]);
        } else if (ch == 1) {
          dp[i][x_last][y_last] = min(y + dp[i + 1][0][y_last],
                                       0ll + dp[i + 1][1][y_last]);
        } else {
          dp[i][x_last][y_last] = min(dp[i + 1][x_last][0], dp[i + 1][0][y_last]);
        }
      }
    }
  }
  cout << dp[0][0][0] << endl;
  return 0;
}
