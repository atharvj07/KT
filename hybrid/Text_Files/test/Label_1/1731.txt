#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n), b(n);
    vector<bool> chosen(k + 1, false);
    for (int i = 0; i < n; ++i) {
        cin >> a[i] >> b[i];
        chosen[a[i]] = true;
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        if (chosen[a[i]]) continue;
        ans += b[i];
        chosen[a[i]] = true;
        --k;
    }
    cout << ans << endl;
    return 0;
}
