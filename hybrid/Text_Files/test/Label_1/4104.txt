#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  vector<double> maxProb(n), minProb(n);
  for (double &p : maxProb) cin >> p;
  for (double &p : minProb) cin >> p;
  vector<double> aProb(n), bProb(n);
  for (int i = 0; i < n; i++) {
    for (int j = 0; j < n; j++) {
      if (i < j) {
        aProb[i] += minProb[j];
        bProb[j] += minProb[j];
      } else if (i > j) {
        aProb[i] += maxProb[j];
        bProb[j] += maxProb[j];
      } else {
        aProb[i] += (maxProb[i] + minProb[i]) / 2;
        bProb[i] += (maxProb[i] + minProb[i]) / 2;
      }
    }
  }
  for (int i = 0; i < n; i++) { cout << fixed << setprecision(8) << aProb[i] << ' '; }
  cout << '\n';
  for (int i = 0; i < n; i++) { cout << fixed << setprecision(8) << bProb[i] << ' '; }
  cout << '\n';
  return 0;
}
