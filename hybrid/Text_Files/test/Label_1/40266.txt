from sys import stdin

n = int(stdin.readline())
names, pseudonyms = [], []
for i in range(n):
    names.append(stdin.readline().rstrip())
for i in range(n):
    pseudonyms.append(stdin.readline().rstrip())

def lcp(a, b):
    ans = 0
    for i in range(min(len(a), len(b))):
        if a[i] == b[i]:
            ans += 1
        else:
            break
    return ans

max_sum = 0
matching = []
for i in range(n):
    max_lcp = 0
    max_pos = -1
    for j in range(n):
        if names[i] != pseudonyms[j] or (i, j) in matching:
            continue
        l = lcp(names[i], pseudonyms[j])
        if l > max_lcp:
            max_lcp = l
            max_pos = j
    max_sum += max_lcp
    matching.append((i, max_pos))

print(max_sum)
for i, j in matching:
    print(i + 1, j + 1)
