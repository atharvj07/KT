n = int(input().strip())
a = list(map(int, input().strip().split()))

dp = [0] * (max(a) + 1)

for val in a:
    divisors = []
    for d in range(1, int(val**0.5) + 1):
        if val % d == 0:
            divisors.append(d)
            if d != val // d:
                divisors.append(val // d)
    max_clique = 0
    for divisor in divisors:
        max_clique = max(max_clique, dp[divisor])
    for divisor in divisors:
        dp[divisor] = max_clique + 1

print(max(dp))
