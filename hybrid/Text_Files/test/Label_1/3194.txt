#include <bits/stdc++.h>
using namespace std;
int n,m,a,b,ans;
bitset<1000>ban[1000];
signed main(){
    cin>>n>>m;
    while(m--){
        cin>>a>>b;
        ban[a][b]=1;
    }
    for(int i=1;i<=n;++i){
        if(!ban[i][1]){
            int cnt=1;
            for(int j=i+1;j<=n;++j){
                if(ban[j][1])break;
                ++cnt;
            }
            ans=max(ans,cnt);
        }
        if(!ban[1][i]){
            int cnt=1;
            for(int j=i+1;j<=n;++j){
                if(ban[1][j])break;
                ++cnt;
            }
            ans=max(ans,cnt);
        }
        if(!ban[n][i]){
            int cnt=1;
            for(int j=i+1;j<=n;++j){
                if(ban[n][j])break;
                ++cnt;
            }
            ans=max(ans,cnt);
        }
        if(!ban[i][n]){
            int cnt=1;
            for(int j=i+1;j<=n;++j){
                if(ban[j][n])break;
                ++cnt;
            }
            ans=max(ans,cnt);
        }
    }
    cout<<ans;
}
