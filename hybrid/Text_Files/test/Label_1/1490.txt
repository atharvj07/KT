#include <iostream>
#include <string>

using namespace std;

int main() {
	int n;
	cin >> n;
	
	string words[n];
	for (int i = 0; i < n; i++) {
		cin >> words[i];
	}
	
	string message;
	getline(cin, message);
	
	int pos = 0;
	for (int i = 0; i < n; i++) {
		string heart = "<3";
		if (pos + heart.size() > message.size() || message.substr(pos, heart.size()) != heart) {
			cout << "no" << endl;
			return 0;
		}
		pos += heart.size();
		
		string word = words[i];
		if (pos + word.size() > message.size() || message.substr(pos, word.size()) != word) {
			cout << "no" << endl;
			return 0;
		}
		pos += word.size();
	}
	
	string heart = "<3";
	if (pos + heart.size() > message.size() || message.substr(pos, heart.size()) != heart) {
		cout << "no" << endl;
		return 0;
	}
	pos += heart.size();
	
	cout << "yes" << endl;
	return 0;
}
