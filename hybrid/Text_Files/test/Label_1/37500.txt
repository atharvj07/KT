def solve(s):
	n = len(s)
	p = [0] * n
	s = list(s)
	for i in range(1, n):
		if s[i] == ")":
			j = i - 1
			while j >= 0 and s[j] != "(":
				j -= 1
			if j >= 0:
				p[i] = j
	dp = [0] * n
	dp[0] = 1
	for i in range(1, n):
		dp[i] = dp[i - 1]
		if s[i] == ")" and p[i] >= 0:
			dp[i] += dp[p[i] - 1]
	return dp[-1] % (10 ** 9 + 7)

s = input()
print(solve(s))
