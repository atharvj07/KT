parl=[0]*101
res=[0]*101
reslast=[0]*101
n,k,m,a=[int(i) for i in input().split()]
arr=list(map(int,input().split()))
for i in arr:
 res[i]+=1
 ress=[(res[i],i) for i in range(1,n+1)]
 ress.sort(reverse=True)
 for i in ress:
  if i[0]>0:
   parl[i[1]]=1
  else:
   break
last=[0]*101
for i in arr:
 last[i]=1
parl2=[]
for i in ress:
 if parl[i[1]]==0:
  if k==0:
   parl2.append(3)
  elif res[i[1]]>=res[ress[k-1][1]]:
   parl2.append(2)
  else:
   parl2.append(3)
  last[i[1]]=2
  k-=1
 else:
  parl2.append(1)
  reslast[i[1]]=1
for i in arr:
 if reslast[i]==0:
  if parl2[i-1]==1 or k<=0:
   parl2[i-1]=3
  elif res[i]>=res[ress[k-1][1]]:
   parl2[i-1]=2
  else:
   parl2[i-1]=3
  k-=1
