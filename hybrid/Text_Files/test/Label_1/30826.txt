import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {

        Scanner cin = new Scanner(System.in);
        int n = cin.nextInt();
        long[] a = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            a[i] = cin.nextInt();
        }

        long[] leftMax = new long[n + 1];
        leftMax[1] = Long.MIN_VALUE;
        for (int i = 2; i <= n; i++) {
            leftMax[i] = Math.max(leftMax[i - 1], a[i - 1]);
        }

        long[] rightMax = new long[n + 1];
        rightMax[n] = Long.MIN_VALUE;
        for (int i = n - 1; i >= 1; i--) {
            rightMax[i] = Math.max(rightMax[i + 1], a[i + 1]);
        }

        for (int i = 1; i <= n; i++) {
            System.out.println(Math.max(leftMax[i], rightMax[i]));
        }
    }
}
