import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String[] input = sc.nextLine().split(" ");
        boolean[][] board = new boolean[8][8];
        for (String s : input) {
            int x = s.charAt(0) - 'a';
            int y = s.charAt(1) - '1';
            board[x][y] = true;
        }
        if (isCheckmate(board)) {
            System.out.println("CHECKMATE");
        } else {
            System.out.println("OTHER");
        }
    }

    private static boolean isCheckmate(boolean[][] board) {
        for (int i = 0; i < 8; i++) {
            for (int j = 0; j < 8; j++) {
                if (board[i][j]) {
                    if (canTakeKing(board, i, j)) {
                        return true;
                    }
                }
            }
        }
        return false;
    }

    private static boolean canTakeKing(boolean[][] board, int x, int y) {
        if (board[x][y]) {
            if (x > 0 && board[x - 1][y]) {
                return true;
            }
            if (x < 7 && board[x + 1][y]) {
                return true;
            }
            if (y > 0 && board[x][y - 1]) {
                return true;
            }
            if (y < 7 && board[x][y + 1]) {
                return true;
            }
        }
        return false;
    }
}
