import sys

def solve():
    n = int(sys.stdin.readline())
    a = [list(map(int, sys.stdin.readline().split())) for _ in range(n)]

    x = [[False] * 5 for _ in range(5)]

    for i in a:
        for j in range(5):
            if i[j] == 1:
                for k in range(5):
                    x[j][k] = True
                    x[k][j] = True

    ok = True

    for i in range(5):
        for j in range(i+1, 5):
            if not x[i][j]:
                ok = False

    if ok:
        print("YES")
    else:
        print("NO")

for _ in range(int(sys.stdin.readline())):
    solve()
