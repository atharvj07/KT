m = int(input().strip())
coordinates = [input().strip() for _ in range(m)]

# Calculate the expression value for each ship
expressions = [(eval(coord)) for coord in coordinates]

# Count occurrences of each coordinate
counter = {}
for expr in expressions:
    if expr in counter:
        counter[expr] += 1
    else:
        counter[expr] = 1

# Output the result
for expr in expressions:
    print(counter[expr], end=" ")
