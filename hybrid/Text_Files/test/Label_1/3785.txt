#include<bits/stdc++.h>
using namespace std;
const int MAXN = 1e5 + 5, MAXH = 1e8 + 5;
int n, h;
int a[MAXN], ans[MAXN];

int main() {
    cin >> n >> h;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int now = 1, mn = a[0], mx = a[n - 1];
    ans[0] = 1;
    for (int i = 1; i < n; i++) {
        if (a[i] > mn + h) {
            ans[i] = 2;
        } else {
            ans[i] = 1;
            mn = a[i];
        }
    }
    cout << mx - mn << endl;
    for (int i = 0; i < n; i++) cout << ans[i] << " ";
    return 0;
}
