n = int(input())
a = list(map(int, input().split()))

min_power = sum(a)
for i in range(n):
    for j in range(1, a[i] + 1):
        if a[i] % j == 0:
            for k in range(n):
                if k != i:
                    temp_power = sum(a) - a[i] - a[k] + (a[i] // j) + (a[k] * j)
                    min_power = min(min_power, temp_power)
print(min_power)
