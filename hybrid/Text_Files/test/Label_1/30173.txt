import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int[] a = new int[m];
        int[] min = new int[n + 1];
        int[] max = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            min[i] = i;
            max[i] = i;
        }
        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt();
            if (x == 1) continue;
            min[x] = Math.min(min[x], max[x]);
            max[x] = max[x - 1];
            max[x - 1] = x;
        }
        for (int i = 1; i <= n; i++) {
            System.out.println(min[i] + " " + max[i]);
        }
    }
}
