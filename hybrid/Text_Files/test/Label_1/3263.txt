#include <iostream>
#include <vector>
#include <queue>
using namespace std;
int main() {
    int k;
    cin >> k;
    if (k == 1) {
        cout << "2\nNY\nYN\n";
    } else {
        int n = 8;
        cout << n << "\n";
        vector<vector<bool>> G(n, vector<bool>(n, false));
        G[0][2] = G[2][0] = true;
        G[0][3] = G[3][0] = true;
        G[0][4] = G[4][0] = true;
        G[0][5] = G[5][0] = true;
        G[1][6] = G[6][1] = true;
        G[1][7] = G[7][1] = true;
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                if (i == j) continue;
                if (G[i][j]) cout << 'Y';
                else if (i > 0 && j > 0 && k - 1 > 0 && i != 1 && j != 2) {
                    --k;
                    cout << 'Y'; G[i][j] = G[j][i] = true;
                } else {
                    cout << 'N';
                }
            }
            cout << "\n";
        }
    }
    return 0;
}
