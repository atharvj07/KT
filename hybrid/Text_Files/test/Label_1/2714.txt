#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;cin>>t;
    while(t--){
        string s;cin>>s;
        int x;cin>>x;
        int n = s.size();
        string ans(n, '0');
        x--;
        int l = x+1, r = n-x;
        for(int i = 0;i<n;i++){
            if(s[i] == '0') continue;
            if(i>=x && s[i-x-1] == '1') ans[i] = '1', l--;
            else if(i+x < n && s[i+x] == '1') ans[i] = '1', r--;
            else{ans = "-1"; break;}
            if(l<=0 || r<=0){ans = "-1"; break;}
        }
        cout<<ans<<'\n';
    }
}
