import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();

        int[] ans = new int[2 * n];
        int[] cnt = new int[n + 1];
        for (int i = 0; i < 2 * n - 1; i++) {
            if (s.charAt(i) == '0') {
                ans[i] = ans[i + 1];
                cnt[ans[i]]++;
            } else {
                int c = 0;
                for (int j = 1; j <= n; j++) {
                    if (cnt[j] < 2) {
                        ans[i] = j;
                        cnt[j]++;
                        c++;
                    }
                    if (c == 2) {
                        break;
                    }
                }
            }
        }

        if (cnt[ans[2 * n - 1]] < 2) {
            System.out.println("No");
        } else {
            System.out.println("Yes");
            for (int i = 0; i < 2 * n; i++) {
                System.out.print(ans[i] + " ");
            }
            System.out.println();
        }
    }
}
