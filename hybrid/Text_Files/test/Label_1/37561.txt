from collections import defaultdict, deque
from sys import stdin
read = stdin.readline

n = int(read())
g = defaultdict(list)
for _ in range(n - 1):
    u, v = map(int, read().split())
    g[u].append(v)
    g[v].append(u)
q = deque([(1, 1/n)])
prob = [0] * n
while q:
    v, p = q.popleft()
    prob[v - 1] += p
    for u in g[v]:
        if u not in q:
            q.append((u, p/2))
    g[v] = []
print(*prob)
