q = int(input().strip())
for _ in range(q):
    n = int(input().strip())
    skills = list(map(int, input().strip().split()))
    skills.sort()
    max_diff = 0
    for i in range(1, n):
        max_diff = max(max_diff, skills[i] - skills[i-1])
    if max_diff > 1:
        print(1)
    else:
        print(2)
