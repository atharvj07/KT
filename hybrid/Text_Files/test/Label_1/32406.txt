import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        Arrays.sort(a);
        long sum = a[0];
        for (int i = 1; i < n; i++) {
            if (a[i] > sum) {
                sum = a[i];
            } else {
                sum += a[i];
            }
        }
        System.out.println(sum);
    }
}
