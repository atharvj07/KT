from collections import Counter
import sys

input = sys.stdin.readline
n = int(input())
a = list(range(1, n + 1))
while len(a) > 1:
    g = 0
    for x in a:
        g = gcd(g, x)
    print(g, end=' ')
    a.remove(g)
print(a[0])

def gcd(a, b):
    if b == 0:
        return a
    return gcd(b, a % b)
