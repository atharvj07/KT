#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for (int i = 0; i < (int)(n); i++)
#define all(x) (x).begin(), (x).end()
const int INF = 1e9;

int main() {
  long long sum, limit;
  cin >> sum >> limit;
  if (limit < 3) {
    cout << -1 << endl;
    return 0;
  }
  if (limit == 3 && sum > 1) {
    cout << -1 << endl;
    return 0;
  }
  int cnt = 0;
  for (int i = 2; i <= limit; i++) {
    if ((sum - i) % 2 == 0 && (sum - i) / 2 <= limit) {
      if (cnt == 0) {
        cout << 2 << endl;
        cout << i << " " << (sum - i) / 2 << endl;
      }
      cnt++;
    }
    if (cnt == 2) {
      break;
    }
  }
  if (cnt != 2) {
    cout << -1 << endl;
    return 0;
  }
  return 0;
}
