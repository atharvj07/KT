import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] ans = new int[n];
        int balance = 0;
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == '(') {
                balance++;
            } else if (c == ')') {
                balance--;
            }
            if (balance < 0) {
                ans[i] = -1;
            } else if (balance == 0) {
                ans[i] = 1;
            } else if (balance % 2 == 0) {
                ans[i] = balance / 2;
            } else {
                ans[i] = balance / 2 + 1;
            }
        }
        for (int i = 0; i < n; i++) {
            System.out.print(ans[i] + " ");
        }
    }
}
