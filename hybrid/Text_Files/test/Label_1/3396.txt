#include <bits/stdc++.h>
using namespace std;
int main() {
  int m, n;
  cin >> m >> n;
  vector<vector<int>> a;
  for (int i = 0; i < m; i++) {
    a.push_back(vector<int>(n));
    for (int j = 0; j < n; j++) { cin >> a[i][j]; }
  }
  vector<int> d(m + 1, 0);
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) { d[i + 1] = max(d[i + 1], d[i] + a[i][j]); }
  }
  for (int i = 1; i <= m; i++) cout << d[i] << " ";
  return 0;
}
