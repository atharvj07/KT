import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    long[] pre = new long[n];
    long[] suf = new long[n];
    pre[0] = a[0];
    for (int i = 1; i < n; i++) {
      pre[i] = pre[i - 1] ^ a[i];
    }
    suf[n - 1] = a[n - 1];
    for (int i = n - 2; i >= 0; i--) {
      suf[i] = suf[i + 1] ^ a[i];
    }
    long ans = 0;
    for (int i = 1; i < n - 1; i++) {
      if (pre[i - 1] == suf[i + 1]) {
        ans++;
      }
    }
    System.out.println(ans);
  }
}
