import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int s = sc.nextInt();
        int t = sc.nextInt();
        int u = sc.nextInt();
        int n = sc.nextInt();
        int o = sc.nextInt();
        int d = sc.nextInt();
        int[] e = new int[t];
        int[] c = new int[t];
        for (int i = 0; i < t; i++) {
            e[i] = sc.nextInt();
            c[i] = sc.nextInt();
        }
        int[] h = new int[n];
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            h[i] = sc.nextInt();
            a[i] = sc.nextInt();
        }

        int[][] dp = new int[d+1][s+1];
        for (int i = 0; i <= d; i++) {
            for (int j = 0; j <= s; j++) {
                dp[i][j] = -1;
            }
        }
        dp[0][s] = 0;

        for (int i = 0; i < d; i++) {
            for (int j = 0; j <= s; j++) {
                if (dp[i][j] == -1) continue;
                int next = j + o;
                if (next > s) next = s;
                dp[i+1][next] = Math.max(dp[i+1][next], dp[i][j]);
                for (int k = 0; k < t; k++) {
                    if (j - e[k] < 0) continue;
                    if (c[k] > next) continue;
                    dp[i+1][next-c[k]] = Math.max(dp[i+1][next-c[k]], dp[i][j] + h[k]);
                }
            }
        }

        int ans = 0;
        for (int i = 0; i <= s; i++) {
            ans = Math.max(ans, dp[d][i]);
        }
        if (ans == 0) ans = -1;
        System.out.println(ans);
    }
}
