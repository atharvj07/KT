import java.util.Scanner;

public class ParenthesisSequence {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int length = scanner.nextInt();
        scanner.nextLine(); // Consume newline
        String sequence = scanner.nextLine();

        String result = replaceQuestionMarks(sequence);
        if (result.equals("")) {
            System.out.println(":(");
        } else {
            System.out.println(result);
        }
    }

    private static String replaceQuestionMarks(String sequence) {
        int balance = 0;
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < sequence.length(); i++) {
            char ch = sequence.charAt(i);
            if (ch == '(') {
                balance++;
            } else if (ch == ')') {
                balance--;
                if (balance < 0) {
                    return ""; // Invalid sequence
                }
            } else { // ch == '?'
                if (balance == 0) {
                    result.append('(');
                } else {
                    result.append(')');
                }
            }
        }
        return balance == 0 ? result.toString() : ""; // Check if all parentheses are balanced
    }
}
