#include <iostream>
#include <string>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::string a[n + 1];
    for (int i = 0; i < n; ++i) std::cin >> a[i];
    int ans = 0;
    for (int j = 0; j < m; ++j) {
        bool bad = false;
        for (int i = 0; i < n - 1; ++i) {
            if (a[i + 1][j] < a[i][j])
                bad = true;
        }
        ans += bad;
    }
    std::cout << ans << std::endl;
    return 0;
}
