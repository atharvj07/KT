import java.util.*;

public class ParenthesisReplacement {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();

        char[] result = new char[n];
        int open = 0, question = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                open++;
            } else if (s.charAt(i) == '?') {
                question++;
            }
        }

        if (open > question + 1) {
            System.out.println(":(");
            return;
        }

        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == '(') {
                result[i] = '(';
                open--;
            } else if (s.charAt(i) == ')') {
                if (open > 0) {
                    result[i] = ')';
                    open--;
                } else {
                    result[i] = '(';
                    question--;
                }
            } else {
                if (open > 0) {
                    result[i] = ')';
                    open--;
                } else {
                    result[i] = '(';
                    question--;
                }
            }
        }

        if (open != 0 || question != 0) {
            System.out.println(":(");
        } else {
            System.out.println(new String(result));
        }
    }
}
