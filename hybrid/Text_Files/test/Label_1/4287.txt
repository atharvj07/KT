#include <cstdio>
#include <iostream>
#include <cstring>
int b[9][9],w[9][9],x1,y1,x2,y2;
char c[9][9];
int main(){
	for(int i=1;i<=8;++i){
		std::cin.getline(c[i]+1,8);
	}
	for(int i=1;i<=8;++i){
		for(int j=1;j<=8;++j){
			if(c[i][j]=='.')b[i][j]=w[i][j]=1e9;
			else if(c[i][j]=='.')b[i][j]=w[i][j]=1e9;
			else if(c[i][j]=='B')b[i][j]=0;
			else w[i][j]=0;
		}
	}
	for(int i=1;i<=8;++i){
		for(int j=1;j<=8;++j){
			if(b[i][j]<1e9){
				for(int k=i+1;k<=8;++k){
					if(w[k][j]<1e9){
						x1=i,y1=j,x2=k,y2=j;
						break;
					}
				}
			}
			if(w[i][j]<1e9){
				for(int k=i-1;k>=1;--k){
					if(b[k][j]<1e9){
						x1=k,y1=j,x2=i,y2=j;
						break;
					}
				}
			}
		}
	}
	for(int i=0;i<=8;++i){
		if(w[i][1]<1e9)return std::cout<<"A",0;
		if(b[i][8]<1e9)return std::cout<<"B",0;
	}
}
