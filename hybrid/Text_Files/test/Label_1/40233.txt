def min_operations_to_sort(n, a):
    a = [a[i] - 1 for i in range(n)]
    pos = [[] for _ in range(n)]
    for i in range(n):
        pos[a[i]].append(i)
    dp = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        dp[i] = dp[i + 1]
        if pos[i]:
            dp[i] = max(dp[i], dp[min(pos[i]) + len(pos[i])] + len(pos[i]))
    return n - dp[0]

q = int(input())
for _ in range(q):
    n = int(input())
    a = list(map(int, input().split()))
    print(min_operations_to_sort(n, a))
