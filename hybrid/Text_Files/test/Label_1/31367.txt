import java.util.*;
import java.io.*;
import java.lang.*;

class Main {
    static final int MAXN = 3001;
    static int n, m;
    static int[] par;
    static int[] sz;
    
    static int find(int x) {
        if (x == par[x]) return x;
        return par[x] = find(par[x]);
    }

    static void merge(int a, int b) {
        int rootA = find(a);
        int rootB = find(b);
        if (rootA == rootB) return;
        if (sz[rootA] > sz[rootB]) {
            par[rootB] = rootA;
            sz[rootA] += sz[rootB];
        } else {
            par[rootA] = rootB;
            sz[rootB] += sz[rootA];
        }
    }
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        
        n = in.nextInt();
        m = in.nextInt();
        par = new int[2*MAXN];
        sz = new int[2*MAXN];
        for (int i = 0; i < 2*MAXN; i++) {
            par[i] = i;
            sz[i] = 1;
        }
        
        for (int i = 0; i < m; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            merge(a, b);
        }
        
        int s1 = in.nextInt();
        int t1 = in.nextInt();
        int l1 = in.nextInt();
        int s2 = in.nextInt();
        int t2 = in.nextInt();
        int l2 = in.nextInt();
        
        int rootS1 = find(s1);
        int rootT1 = find(t1);
        int rootS2 = find(s2);
        int rootT2 = find(t2);
        
        if (rootS1 != rootT1 || rootS2 != rootT2) {
            System.out.println(-1);
        } else {
            int ans = 0;
            if (l1 == l2) {
                int dist = Math.max(l1, n - l1);
                if (sz[rootS1] - 1 > dist) {
                    ans++;
                }
            } else {
                int dist1 = Math.max(l1, n - l1);
                int dist2 = Math.max(l2, n - l2);
                if (sz[rootS1] - 1 > dist1 + dist2) {
                    ans++;
                } else if (dist1 < dist2) {
                    if (sz[rootS1] - 1 > dist1) {
                        ans++;
                    }
                } else {
                    if (sz[rootS1] - 1 > dist2) {
                        ans++;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
