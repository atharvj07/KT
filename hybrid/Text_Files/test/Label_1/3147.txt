#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  assert(n % 3 == 0);
  vector<pair<int, int>> a(n);
  for (int i = 0; i < n; i++) { cin >> a[i].second >> a[i].first; }
  sort(a.begin(), a.end());
  for (int i = 0; i < n; i += 3) {
    if (a[i].second <= a[i + 1].second && a[i + 1].second <= a[i + 2].second) {
      cout << a[i].first << ' ' << a[i + 1].first << ' ' << a[i + 2].first << ' ';
    } else {
      cout << -1;
      return 0;
    }
  }
}
