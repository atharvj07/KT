import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int minDiff = Integer.MAX_VALUE;
        for (int i = 1; i < n - 1; i++) {
            minDiff = Math.min(minDiff, a[i + 1] - a[i - 1]);
        }
        System.out.println(minDiff);
    }
}
