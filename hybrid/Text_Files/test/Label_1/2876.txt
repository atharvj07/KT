#include<bits/stdc++.h>
using namespace std;
int main(){
	int n,m,q;
	cin>>n>>m>>q;
	int grid[n][m];
	for(int i=0;i<n;i++){
		for(int j=0;j<m;j++)cin>>grid[i][j];
	}
	int ans[n];
	for(int i=0;i<n;i++){
		int cnt=0;
		for(int j=0;j<m;j++){
			if(grid[i][j]==1)cnt++;
			else cnt=0;
			ans[i]=max(ans[i],cnt);
		}
	}
	while(q--){
		int x,y;
		cin>>x>>y; x--; y--;
		grid[x][y]=1-grid[x][y];
		int cnt=0;
		for(int j=0;j<m;j++){
			if(grid[x][j]==1)cnt++;
			else cnt=0;
			ans[x]=max(ans[x],cnt);
		}
		cout<<*max_element(ans,ans+n)<<endl;
	}
}
