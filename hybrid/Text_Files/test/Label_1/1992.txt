#include <bits/stdc++.h>

using namespace std;

const int N = 1e5;
int a[N], f[N];
int m[2][N];

int main() {
  int t;
  for (cin >> t; t--;) {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    sort(a + 1, a + n + 1);
    memset(m, 0, sizeof m);
    int b[3], cnt = 0;
    for (int i = 1; i <= n; i++) {
      if (m[0][a[i] - 1] != m[0][a[i]]) {
        m[0][a[i]] = m[0][a[i] - 1] + 1;
        b[cnt++] = a[i];
      }
    }
    for (int i = 1; i <= n; i++) {
      m[1][a[i]] = m[1][a[i] - 1];
      m[1][a[i]] += (binary_search(b, b + cnt, a[i]));
      f[i] = min(m[0][a[i]], m[1][a[i]]);
    }
    int ans = -1;
    for (int i = 1; i <= n; i++)
      ans = max(ans, min({a[i], n - f[i]}));
    cout << max(ans - 1, 0) << '\n';
  }
  return 0;
}
