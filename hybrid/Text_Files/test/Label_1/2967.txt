#include <bits/stdc++.h>
using namespace std;
double power(double x, int p) {
    double ans = 1;
    while (p--) {
        ans *= x;
    }
    return ans;
}
int main() {
    string s;
    cin >> s;
    int n = s.size();
    vector<double> sum(n);
    int c = 0;
    double ans = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == 'I' || s[i] == 'E' || s[i] == 'A' || s[i] == 'O' || s[i] == 'U' || s[i] == 'Y') {
            c++;
        }
        sum[i] = c * 1.0 / (i + 1);
    }
    for (int i = 0; i < n; i++) {
        for (int j = i; j < n; j++) {
            ans += sum[j] - (i ? sum[i - 1] : 0);
        }
    }
    cout << fixed << setprecision(8) << ans << "\n";
    return 0;
}
