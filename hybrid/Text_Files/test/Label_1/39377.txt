n = int(input())
columns = [list(map(int, input().split())) for _ in range(n)]

L, R = 0, 0
for l, r in columns:
    L += l
    R += r

current_beauty = abs(L - R)
best_beauty = current_beauty
best_column = 0

for i, (l, r) in enumerate(columns, start=1):
    new_L = L - l + r
    new_R = R - r + l
    new_beauty = abs(new_L - new_R)
    if new_beauty > best_beauty:
        best_beauty = new_beauty
        best_column = i

print(best_column)
