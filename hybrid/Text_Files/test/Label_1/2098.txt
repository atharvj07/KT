#include <bits/stdc++.h>
using namespace std;
int n, s[2005], t[2005], f[2005][2005][2][2];
string ss, tt;
int main() {
  cin >> ss;
  n = ss.length();
  for (int i = 1; i <= n; i++) s[i] = ss[i - 1] - '0';
  // memset(f, -1, sizeof(f));
  for (int i = n; i >= 1; i--) {
    for (int j = i; j <= n; j++) {
      for (int p = 0; p < 2; p++) {
        for (int q = 0; q < 2; q++) {
          int xx = (s[i] >= p), yy = (s[j] >= q);
          int &now = f[i][j][p][q];
          if (i == j) now = xx == yy;
          else
            now = max({f[i + 1][j][xx][yy], f[i][j - 1][xx][yy], f[i + 1][j - 1][xx][yy] + (xx == yy)}), now = max(now, f[i + 1][j - 1][p][q]);
        }
      }
    }
  }
  int p = 0, q = 0;
  for (int i = 1; i <= n; i++) {
    for (int xx = 0; xx < 2; xx++) {
      for (int yy = 0; yy < 2; yy++) {
        int tmp = f[i + 1][n][xx][yy];
        if (i < n) tmp = max(tmp, f[i + 1][n][p][q]);
        if (tmp == f[i][n][p][q] && f[i][n][p][q] != 0) { xx = p, yy = q; }
        if (tmp > f[i][n][p][q]) { p = xx, q = yy; }
      }
    }
    t[i] = p;
  }
  for (int i = 1; i <= n; i++) tt += (t[i] + '0');
  cout << tt << endl;
  return 0;
}
