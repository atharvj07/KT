import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
        }
        Arrays.sort(arr);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] == 1) {
                ans++;
            } else {
                boolean flag = true;
                for (int j = 0; j < n; j++) {
                    if (arr[i] % arr[j] == 0 && arr[i] / arr[j] * arr[i] / arr[j] * arr[i] / arr[j] == arr[i]) {
                        flag = false;
                        break;
                    }
                }
                if (flag) {
                    ans++;
                }
            }
        }
        System.out.println(ans);
    }
}
