#include<algorithm>
#include<vector>
#include<iostream>
#include<cstring>
#include<cstdlib>
using namespace std;
#define ll long long
ll d[200005],a[200005];
vector<ll> e[200005];
void dfs(ll x,ll p=1,ll dep=0)
{
    d[x]=dep;a[x]=p;
    for(auto u:e[x])
    {
        if(u==p)
        continue;
        dfs(u,x,dep+1);
    }
}
bool ok(ll x,ll y)
{
    if(a[x]==a[y])
    {
        if(d[x]>=d[y])
        {
            return (d[x]-d[y]==1);
        }
        else
        {
            return (d[y]-d[x]==1);
        }
    }
    else
    {
        return 1;
    }
}
int main()
{
    ll n,m;cin>>n>>m;
    for(ll i=1;i<n;i++)
    {
        ll x,y;cin>>x>>y;
        e[x].push_back(y);
        e[y].push_back(x);
    }
    dfs(1);
    while(m--)
    {
        ll t,q;cin>>t;
        bool flg=1;
        for(ll i=1;i<=t;i++)
        {
            cin>>q;
            if(flg==0)
            continue;
            if(a[q]==1&&t>1)
            {
                flg=0;
            }
            else
            {
                if(i>1)
                flg&=ok(q,q);
            }
        }
        (flg) ? cout<<"YES\n":cout<<"NO\n";
    }
}
