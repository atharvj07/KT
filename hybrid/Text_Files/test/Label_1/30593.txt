import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        int[] A = new int[N];
        for (int i = 0; i < N; i++) {
            A[i] = sc.nextInt();
        }
        int[] B = A.clone();
        Arrays.sort(B);
        int ans = 0;
        for (int i = 0; i < N; i++) {
            if (A[i] != B[i]) {
                ans++;
                int j = i + 1;
                while (j < N && A[j] != B[i]) {
                    j++;
                }
                if (j == N) {
                    ans = -1;
                    break;
                } else {
                    for (int k = j; k > i; k--) {
                        int tmp = A[k];
                        A[k] = A[k - 1];
                        A[k - 1] = tmp;
                    }
                }
            }
        }
        System.out.println(ans);
    }
}
