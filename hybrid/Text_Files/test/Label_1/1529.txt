#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main(){
  int n;
  cin >> n;
  vector<int> arr(n);
  for(int &x : arr) cin >> x;
  int ans = n+1;
  for(int i = 0; i < n; i++){
    for(int j = i+1; j < n; j++){
      if(arr[i] == arr[j]){
        ans = min(ans, j - i);
      }
    }
  }
  cout << ans << '\n';
}
