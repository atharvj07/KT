n, m = map(int, input().split())
potentials = list(map(int, input().split()))
clubs = list(map(int, input().split()))
d = int(input())
leaving_students = [int(input()) for _ in range(d)]

def mex(s):
    s = set(s)
    i = 0
    while i in s:
        i += 1
    return i

club_members = {i: set() for i in range(1, m+1)}
for i in range(n):
    club_members[clubs[i]].add(i)

left_students = set()
for day in range(d):
    left_students.add(leaving_students[day] - 1)
    remaining_potentials = []
    for club, members in club_members.items():
        available_members = members - left_students
        if available_members:
            min_potential_member = min(available_members, key=lambda x: potentials[x])
            remaining_potentials.append(potentials[min_potential_member])
    print(mex(remaining_potentials))
