#include <bits/stdc++.h>
using namespace std;
const int N = 110;
string s[N];
bool cmp(string x, string y) { return x < y; }
int main() {
    int n, m, ans = -1;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> s[i];
    sort(s + 1, s + n + 1, cmp);
    for (int i = 1; i <= n; i++) {
        string t;
        for (int j = 1; j <= m; j++) {
            if (s[i][j - 1] != s[i][j]) {
                t += s[i][j];
                ans = j;
            }
        }
        s[i] = t;
    }
    for (int i = 2; i <= n; i++)
        if (s[i - 1] > s[i]) {
            ans = -1;
            break;
        }
    cout << ans << endl;
    return 0;
}
