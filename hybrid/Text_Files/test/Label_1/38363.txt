t = int(input())
for test in range(t):
    n = int(input())
    s = input()
    ans = []
    for chr in s:
        ans.append(chr)
        if len(ans) > 1 and ans[-1] == '0' and ans[-2] == '1':
            ans.pop()
            ans.pop()
    if len(ans) == 0:
        print('0')
    else:
        print(''.join(ans))
