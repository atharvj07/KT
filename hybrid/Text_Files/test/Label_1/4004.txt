#include<bits/stdc++.h>
using namespace std;
int main(){
	vector<string> a;
	int t;
	cin>>t;
	while(t--){
		int n,p,ans=0;
		bool f=0;
		cin>>n>>p;
		for(int i=0;i<=n;i++) a.push_back("");
		for(int i=1;i<=n;i++){
			a[i]+=char(p+48);
			cin>>p;
		}
		for(int i=1;i<=n;i++){
			int l=i,r=n;
			while(l<r){
				if(a[l]==a[r]) l++,r--;
				else if(a[l]>a[r]) swap(a[l],a[r]),l++,ans++;
				else{f=1;break;}
			}
			if(f) break;
			cout<<a[i]<<" ";
		}
		if(f) cout<<-1;
		else cout<<ans;
		cout<<endl;
	}
}
