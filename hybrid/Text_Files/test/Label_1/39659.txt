def find_k(n, edges):
    from collections import defaultdict, deque

    # Build the graph
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    # BFS to find the maximum depth
    def bfs(root):
        max_depth = 0
        queue = deque([(root, 0)])
        visited = set()
        visited.add(root)
        while queue:
            node, depth = queue.popleft()
            max_depth = max(max_depth, depth)
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, depth + 1))
        return max_depth
    
    # Find the maximum depth from node 1
    max_depth = bfs(1)
    
    # The minimum value of k is the maximum depth
    return max_depth

# Read input and solve the problem for each test case
t = int(input())
for _ in range(t):
    n = int(input())
    edges = [tuple(map(int, input().split())) for _ in range(n-1)]
    print(find_k(n, edges))
