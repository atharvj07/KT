from collections import Counter
import sys

for _ in range(int(input())):
    n = int(input())
    p = [int(x) for x in input().split()]
    if Counter(p) != Counter(range(1, n + 1)):
        print('NO')
        continue

    flag = True
    for i in range(n - 1):
        if p[i] + 1 != p[i + 1]:
            flag = False

    if p[n - 1] != 1:
        flag = False

    print(['NO', 'YES'][flag])
