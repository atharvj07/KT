import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        boolean[] win = new boolean[n + 1];
        win[0] = false;
        for (int i = 1; i <= n; i++) {
            boolean canWin = false;
            for (int j = 0; j < n; j++) {
                if (a[j] >= i && !win[i]) {
                    canWin = true;
                    continue;
                }
                if (a[j] >= 1 && win[i - 1]) {
                    canWin = true;
                    continue;
                }
            }
            win[i] = canWin;
        }
        System.out.println(win[n] ? "BitLGM" : "BitAryo");
    }
}
