import java.util.Scanner;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] v = new int[n];
        long[] w = new long[n];
        int[] score = new int[n];
        for (int i = 0; i < n; i++) {
            v[i] = scanner.nextInt();
        }
        for (int i = 0; i < n; i++) {
            w[i] = scanner.nextInt();
        }
        Arrays.sort(v);
        Arrays.sort(w);

        for (int i = 1; i < n; i++) {
            if (w[i] - w[i - 1] == 1 && 2 * w[i] - w[i - 1] - w[i + 1] >= 0) {
                score[i] = score[i - 1] + v[i - 1];
            } else {
                score[i] = score[i - 1];
            }
        }

        System.out.println(score[n - 1]);
    }
}
