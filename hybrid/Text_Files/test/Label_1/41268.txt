def max_subset_with_powers_of_two(n, points):
    points.sort()
    max_size = 1
    max_subset = [points[0]]
    
    for i in range(n):
        subset = [points[i]]
        for j in range(n):
            if i != j:
                valid = True
                for k in range(len(subset)):
                    if abs(subset[k] - points[j]) & (abs(subset[k] - points[j]) - 1) != 0:
                        valid = False
                        break
                if valid:
                    subset.append(points[j])
        if len(subset) > max_size:
            max_size = len(subset)
            max_subset = subset

    print(max_size)
    print(" ".join(map(str, max_subset)))

n = int(input().strip())
points = list(map(int, input().strip().split()))
max_subset_with_powers_of_two(n, points)
