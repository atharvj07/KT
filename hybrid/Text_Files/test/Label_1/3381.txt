#include <deque>
#include <iostream>
using namespace std;
int main() {
  string s;
  cin >> s;
  deque<char> t, u;
  for (char c : s) {
    if (t.empty() || c < t.back()) {
      t.push_back(c);
    } else {
      u.push_back(c);
      while (!t.empty() && t.back() < c) {
        u.push_back(t.back());
        t.pop_back();
      }
      t.push_back(c);
    }
  }
  while (!t.empty()) {
    u.push_back(t.back());
    t.pop_back();
  }
  while (!u.empty()) {
    cout << u.front();
    u.pop_front();
  }
  cout << endl;
  return 0;
}
