#include <bits/stdc++.h>
using namespace std;
#define int long long
int n, k;
signed main() {
    cin >> n >> k;
    int i = 1, j = n + 1, flag = 0;
    while (i <= n) {
        if (k == 0) break;
        cout << i << " " << j << " ";
        k--, i++, j++;
        flag = 1;
    }
    if (!k && !flag) cout << 1 << " " << 2 << " ";
    else if (k && !flag) {
        cout << 1 << " ";
        k--;
        while (k > 0) {
            cout << j << " " << i << " ";
            j++, i++, k -= 2;
        }
    } else
        while (i <= n) {
            cout << i << " " << j << " ";
            i++, j++;
        }
}
