import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static int ignore, cnt;
    static int[] ans;
    static int[] arr;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), q = sc.nextInt();
        arr = new int[n];
        Arrays.fill(arr, -1);
        ans = new int[q];
        for (int i = 0; i < q; i++) {
            int t = sc.nextInt() - 1, k = sc.nextInt(), d = sc.nextInt();
            if (arr[t % n] == -1 && cnt + k <= n) {
                ans[i] = 0;
                for (int j = t % n, l = 0; l < k; j = (j + 1) % n, l++) {
                    ans[i] += j + 1;
                    arr[j] = t + d;
                }
                cnt += k;
            } else {
                ans[i] = -1;
                ignore++;
            }
        }
        for (int an : ans) System.out.println(an);
    }
}
