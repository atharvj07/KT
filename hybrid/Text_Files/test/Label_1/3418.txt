#include<bits/stdc++.h>
using namespace std;
int main(){
    int t; cin>>t;
    while(t--){
        int n, a, b; cin>>n>>a>>b;
        vector<int>v(n);
        for(int i=0,j=a+1; i<n/2; i++,j++) v[i]=j;
        for(int i=n/2,j=b-1; i<n; i++,j--) v[i]=j;
        if(v[0]==a && v[n/2]==b) for(int j:v) cout<<j<<" ";
        else cout<<-1;
        cout<<endl;
    }
    return 0;
}
