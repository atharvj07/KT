import java.util.Scanner;

public class Main {
    static long mod = 998244353;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        char[] s = scanner.next().toCharArray();
        int n = s.length;
        int[] pre = new int[n]; // pre[i] = number of pre-fixed '?' before i
        for (int i = 0; i < n; i++)
            pre[i] = (s[i] == '?' ? (i > 0 ? pre[i - 1] : 0) + 1 : 0);

        long ans = 0;
        int depth = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (s[i] == '?') {
                cnt++;
                continue;
            }
            if (s[i] == '(')
                depth++;
            else {
                depth--;
                long add = (long) (Math.pow(2, pre[i]) - Math.pow(2, cnt - pre[i] - 1)) % mod;
                add = add * (long) Math.pow(2, depth) % mod;
                ans += add;
                if (ans >= mod)
                    ans -= mod;
            }
        }
        System.out.println(ans);
    }
}
