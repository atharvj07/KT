#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;

int n, k;
vector<int> a;

int main() {
  cin >> n >> k;
  a.resize(n);
  for (int &x : a) { cin >> x; }
  vector<pair<int, int>> b;
  for (int i = 0; i < n; i++) { b.push_back({a[i], i + 1}); }
  sort(b.rbegin(), b.rend());
  int ans = b[k - 1].first;
  cout << ans << "\n";
  for (int i = 0; i < k; i++) { cout << b[i].second << " "; }
  cout << "\n";
  return 0;
}
