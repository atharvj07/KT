import java.util.*;

class Main {
    private static int n, m;
    private static List<List<Integer>> adj;
    private static int[] parity;
    private static boolean[] vis;
    private static Stack<Integer> ans;

    private static void dfs(int u, int depth) {
        if (depth == parity[u] * 2) {
            ans.push(u);
            return;
        }
        vis[u] = true;
        for (int v : adj.get(u)) {
            if (!vis[v]) {
                dfs(v, depth + 1);
            }
        }
        vis[u] = false;
    }

    private static int[] findPath() {
        int[] res = new int[4 * n + 1];
        int k = 0;
        for (int i = 1; i <= n; i++) {
            if (!vis[i]) {
                ans = new Stack<>();
                dfs(i, 0);
                while (!ans.isEmpty()) {
                    res[k++] = ans.pop();
                }
            }
        }
        if (k == 0) {
            return new int[]{-1};
        }
        return Arrays.copyOfRange(res, 0, k);
    }

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        m = in.nextInt();
        adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        for (int i = 0; i < m; i++) {
            int u = in.nextInt(), v = in.nextInt();
            adj.get(u).add(v);
            adj.get(v).add(u);
        }
        parity = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parity[i] = in.nextInt();
        }
        vis = new boolean[n + 1];
        int[] path = findPath();
        if (path[0] == -1) {
            System.out.println(-1);
        } else {
            System.out.println(path.length - 1);
            for (int i = 1; i < path.length; i++) {
                System.out.print(path[i] + " ");
            }
        }
    }
}
