import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        Map<String, Integer> teamNames = new HashMap<>();
        Set<String> shortNames = new HashSet<>();
        for (int i = 0; i < n; i++) {
            String[] parts = sc.nextLine().split(" ");
            String teamName = parts[0];
            String shortName1 = teamName.substring(0, 3);
            String shortName2 = teamName.substring(0, 2) + parts[1].charAt(0);
            if (teamNames.containsKey(shortName1) && teamNames.get(shortName1) != i) {
                System.out.println("NO");
                return;
            }
            if (shortNames.contains(shortName2)) {
                System.out.println("NO");
                return;
            }
            teamNames.put(shortName1, i);
            shortNames.add(shortName2);
            System.out.println("YES");
            System.out.println(shortName1);
        }
    }
}
