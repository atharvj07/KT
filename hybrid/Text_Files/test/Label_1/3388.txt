#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
  ll n, a[1000], ans = -1;
  cin >> n;
  for (ll i = 0; i < n; i++) {
    cin >> a[i];
    if (a[i] == 0) ans = 0;
  }
  sort(a, a + n, greater<ll>());
  ll sum = 0, cnt5 = 0, cnt0 = 0;
  for (ll i = 0; i < n; i++) {
    if (a[i] == 0)
      cnt0++;
    else if (a[i] == 5)
      cnt5++;
  }
  if (cnt0 == 0 or cnt5 == 0)
    cout << ans;
  else {
    string str = "";
    for (ll i = 0; i < cnt5 - cnt5 % 2; i++) str += '5';
    for (ll i = 0; i < cnt0; i++) str += '0';
    ans = stoll(str);
    cout << ans;
  }
  return 0;
}
