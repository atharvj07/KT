#include <iostream>
#include <vector>
#include <algorithm>
#include <string>

using namespace std;

int main() {
  int n;
  cin >> n;
  vector<int> c(n);
  vector<string> s(n);
  for (int i = 0; i < n; i++) {
    cin >> c[i];
    cin >> s[i];
  }
  sort(c.begin(), c.end());
  sort(s.begin(), s.end());
  for (int i = 0; i < n; i++) {
    if (s[i] != s[min(i + 1, n - 1)]) {
      cout << c[i] << endl;
      return 0;
    }
  }
  cout << -1 << endl;
  return 0;
}
