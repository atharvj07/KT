#include <bits/stdc++.h>
#define ll long long

using namespace std;

const int mod = 1e9 + 7;

int main() {
    ll x, y;
    cin >> x >> y;

    if (x == y) {
        cout << 1 << '\n';
        return 0;
    }
    if (y % x) {
        cout << 0 << '\n';
        return 0;
    }
    cout << (ll) pow(2, (y - x) / x) << '\n';
    return 0;
}
