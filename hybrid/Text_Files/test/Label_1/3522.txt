#include <iostream>
#include <vector>
using namespace std;

int main() {
  long long n;
  cin >> n;
  string s;
  cin >> s;
  vector<long long> pref(n, 0);
  vector<long long> suf(n, 0);
  for (long long i = 0; i < n; i++) {
    if (s[i] == '(') pref[i]++;
    else if (i > 0) pref[i] = pref[i - 1] - 1;
  }
  for (long long i = n - 1; i >= 0; i--) {
    if (s[i] == ')') suf[i]++;
    else if (i < n - 1) suf[i] = suf[i + 1] - 1;
  }
  long long ans = 0;
  for (long long i = 0; i < n; i++) {
    if ((s[i] == '(' && pref[i] <= 0 && suf[i] > 0) ||
        (s[i] == ')' && pref[i] > 0 && suf[i] <= 0)) ans++;
  }
  cout << ans << endl;
  return 0;
}
