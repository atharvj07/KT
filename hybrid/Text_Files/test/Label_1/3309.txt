#include <iostream>
#include <vector>
#include <map>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    vector<int> a(n);
    map<int, int> m;
    long long ans = 0;
    for (int i = 0; i < n; i++) {
      cin >> a[i];
      ans += m[a[i]]++;
    }
    cout << ans << endl;
    m.clear();
  }
  return 0;
}
