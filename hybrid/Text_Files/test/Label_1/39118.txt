import sys
n, k = map(int, input().split())
b = list(map(int, input().split()))
c = input()
wb, ob, rb = [], [], []
for i in range(n):
    if c[i] == 'W':
        wb.append((b[i], i))
    elif c[i] == 'O':
        ob.append((b[i], i))
    else:
        rb.append((b[i], i))
wb.sort(reverse=True)
ob.sort(reverse=True)
rb.sort(reverse=True)
ans = 0
if len(rb) and len(wb):
    print(-1)
    sys.exit()
if k <= len(wb) + len(ob):
    for i in range(min(k, len(wb))):
        ans += wb[i][0]
    for i in range(min(k - len(wb), len(ob))):
        ans += ob[i][0]
elif k <= len(wb) + len(rb):
    for i in range(min(k, len(wb))):
        ans += wb[i][0]
    for i in range(min(k - len(wb), len(rb))):
        ans += rb[i][0]
elif k <= len(ob) + len(rb):
    for i in range(min(k, len(ob))):
        ans += ob[i][0]
    for i in range(min(k - len(ob), len(rb))):
        ans += rb[i][0]
else:
    print(-1)
    sys.exit()
print(ans)
