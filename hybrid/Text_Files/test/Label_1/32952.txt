import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static final long MOD = 1000000007L;
    static int n, k;
    static int[][] next;
    static long[][][] dp;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        n = in.nextInt();
        k = in.nextInt();
        dp = new long[n + 1][n + 1][k + 1];
        for (long[][] a : dp) for (long[] b : a) Arrays.fill(b, -1);
        next = new int[n + 1][2];
        for (int i = 1; i < n; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            int c = in.nextInt();
            next[a][c] = b;
            next[b][c] = a;
        }
        System.out.println(solve(1, 1, 0));
    }

    static long solve(int u, int p, int step) {
        if (step == k) return 1;
        if (dp[u][p][step] != -1) return dp[u][p][step];
        long res = 0;
        for (int c = 0; c < 2; c++) {
            int v = next[u][c];
            if (v > 0 && v != p) {
                if (c == 1) res += solve(v, u, step + 1);
                else res += solve(v, u, step);
                res %= MOD;
            }
        }
        return dp[u][p][step] = res;
    }
}
