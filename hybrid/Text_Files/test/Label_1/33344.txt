import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int q = sc.nextInt();
        outer: while (q-- > 0) {
            String t = sc.next();
            int n = s.length(), m = t.length(), i = 0, j = 0;
            while (i < n && j < m) {
                if (s.charAt(i) == t.charAt(j)) {
                    j++;
                }
                i++;
            }
            if (j == m) System.out.println(1);
            else System.out.println(0);
        }
    }
}
