#include<bits/stdc++.h>
using namespace std;
int main()
{
	int t,n,x;
	cin>>t;
	while(t--)
	{
		cin>>n;
		deque<int>dq;
		bool f=1;
		for(int i=0;i<n;++i)
		{
			cin>>x;
			if(x>0)
			{
				f=0;
				dq.push_back(x);
			}
		}
		sort(dq.begin(),dq.end());
		deque<int>dq2=dq;
		for(int i=0;i<dq.size();++i)
		{
			if(dq[i]==dq2[i] && i+1<dq.size() && dq[i]==dq[i+1])
			{
				f=1;
				break;
			}
		}
		(f==1)?cout<<"NO\n":cout<<"YES\n";
	}
	return 0;
}
