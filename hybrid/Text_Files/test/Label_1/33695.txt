import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[4];
        for (int i = 0; i < 4; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();
        int[] brr = arr.clone();
        int a = 0, b = 0, c = 0, d = 0;
        for (int i = 0; i < 4; i++) {
            if (arr[i] == arr[(i + 1) % 4]) {
                arr[i] = arr[(i + 2) % 4];
                arr[(i + 1) % 4] = arr[(i + 3) % 4];
                arr[(i + 2) % 4] = arr[i];
                arr[(i + 3) % 4] = arr[(i + 1) % 4];
                break;
            }
        }
        if (arr[0] == arr[2] && arr[1] == arr[3]) {
            a = arr[0];
            b = arr[1];
            c = 0;
            d = 0;
        } else if (arr[0] == arr[1] && arr[2] == arr[3]) {
            a = arr[2];
            b = arr[0];
            c = 0;
            d = 0;
        } else {
            a = arr[0];
            b = arr[1];
            c = arr[2];
            d = arr[3];
        }
        if (a + b + c + d < 2 * Math.max(Math.max(a, b), Math.max(c, d))) {
            System.out.println("no");
        } else if (a == b && b == c && c == d) {
            System.out.println("yes");
        } else if (a == b && c == d) {
            if (a <= 2 * c || b <= 2 * d) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (a == c && b == d) {
            if (a <= 2 * b || c <= 2 * d) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (a == d && b == c) {
            if (a <= 2 * b || d <= 2 * c) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (brr[0] == brr[1] && brr[2] == brr[3]) {
            if (brr[0] <= 2 * brr[2]) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (brr[0] == brr[2] && brr[1] == brr[3]) {
            if (brr[0] <= 2 * brr[1]) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (brr[0] == brr[3] && brr[1] == brr[2]) {
            if (brr[0] <= 2 * brr[2]) {
                System.out.println("yes");
            } else {
                System.out.println("no");
            }
        } else if (a + b >= c + d || a + c >= b + d || a + d >= b + c || b + c >= a + d || b + d >= a + c || c + d >= a + b) {
            System.out.println("yes");
        } else {
            System.out.println("no");
        }
    }
}
