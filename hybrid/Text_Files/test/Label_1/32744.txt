import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        int[][] f = new int[n + 1][n + 1];
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                if (i == j) {
                    f[i][j] = 1;
                } else {
                    f[i][j] = f[i][j - 1];
                    if (a[j - 1] >= i && a[j - 1] <= j) {
                        f[i][j]++;
                    }
                }
            }
        }
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            for (int j = i; j <= n; j++) {
                ans += f[i][j];
            }
        }
        System.out.println(ans);
    }
}
