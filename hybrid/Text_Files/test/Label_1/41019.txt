import sys
input = sys.stdin.readline

n = int(input())
a = [list(map(int, input().split())) for _ in range(n)]
f = [None] * (2 * n)
for i, (s, e) in enumerate(a):
    for k in range(s, e + 1):
        if f[k] != None and f[k] != i:
            print('IMPOSSIBLE')
            sys.exit()
        f[k] = i

s = []
for i in range(2 * n):
    s.append('(')
    for j in range(i + 1, 2 * n):
        if f[j] == f[i]:
            s[j] = ')'
            break

print(''.join(s))
