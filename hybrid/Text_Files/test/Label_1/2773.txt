#include<bits/stdc++.h>
#define ll long long
using namespace std;
const ll N = 2e5 + 5;
ll n, m, a[N], b[N], c[N];
int main() {
    cin >> n >> m;
    for(ll i = 1; i <= n; i++) {
        for(ll j = 1; j <= m; j++) {
            cin >> a[j];
        }
        sort(a + 1, a + m + 1);
        for(ll j = 1; j <= m; j++) {
            b[(i - 1) * m + j] = a[j];
        }
    }
    for(ll i = 1; i <= n * m; i++) {
        c[i] = i;
    }
    sort(b + 1, b + n * m + 1);
    ll ans = 0;
    for(ll i = 1; i <= n * m; i++) {
        ans += (b[i] != c[i]);
    }
    cout << ans / 2 << endl;
}
