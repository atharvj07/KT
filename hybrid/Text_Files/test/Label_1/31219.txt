import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    static long mod = 998244353;

    static long pow(long x, long y) {
        long res = 1;
        x %= mod;
        while (y > 0) {
            if ((y & 1) == 0) {
                x = x * x % mod;
            } else {
                res = res * x % mod;
            }
            y >>= 1;
        }
        return res;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] a = new int[n];
        String[] s = br.readLine().split(" ");
        for (int i = 0; i < n; i++) {
            a[i] = Integer.parseInt(s[i]);
        }
        int[] dp = new int[n];
        int[] sum = new int[n];
        int last = -1;
        for (int i = n - 1; i >= 0; i--) {
            if (a[i] < 0) {
                last = -1;
            }
            if (a[i] == 0) {
                a[i] = last;
            }
            last = a[i];
        }
        for (int i = 0; i < n; i++) {
            if (a[i] != -1) {
                sum[i] = 1;
            }
            if (i > 0) {
                sum[i] += sum[i - 1];
            }
        }
        dp[n - 1] = 1;
        for (int i = n - 2; i >= 0; i--) {
            if (a[i] == -1) {
                dp[i] = 0;
            } else {
                int num = sum[n - 1] - sum[a[i]];
                dp[i] = (int) ((dp[i + 1] + dp[a[i]] * num) % mod);
            }
        }
        System.out.println(dp[0]);
    }
}
