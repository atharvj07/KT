#include <iostream>
#include <vector>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    int t, n;
    cin >> t;
    while (t--) {
        cin >> n;
        vector<int> a(n);
        for (int &ai : a) cin >> ai;
        int ans = 0, cur = 0;
        for (int i = 0; i < n; ++i) {
            cur += a[i];
            ans = max(ans, cur);
            if (cur < 0) cur = 0;
        }
        cout << ans << '\n';
    }
    return 0;
}
