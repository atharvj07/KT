import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int H = sc.nextInt(), W = sc.nextInt(), D = sc.nextInt();
        char[][] ans = new char[H][W];
        int[][] dist = new int[H][W];
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                dist[i][j] = Math.abs(i - (H - 1) / 2) + Math.abs(j - (W - 1) / 2);
            }
        }
        char[] colors = new char[4];
        colors[0] = 'R';
        colors[1] = 'Y';
        colors[2] = 'G';
        colors[3] = 'B';
        boolean[][] used = new boolean[H][W];
        for (int d = 2; d <= D; d++) {
            for (int i = 0; i < H; i++) {
                for (int j = 0; j < W; j++) {
                    if (dist[i][j] != d || used[i][j]) {
                        continue;
                    }
                    int cnt = 0;
                    for (int di = -1; di <= 1; di++) {
                        for (int dj = -1; dj <= 1; dj++) {
                            int ni = i + di, nj = j + dj;
                            if (ni >= 0 && ni < H && nj >= 0 && nj < W && dist[ni][nj] == d - 1) {
                                cnt++;
                            }
                        }
                    }
                    int color = 0;
                    for (int k = 0; k < cnt; k++) {
                        while (used[i][j] && colors[color] == ans[i][j]) {
                            color++;
                        }
                        for (int di = -1; di <= 1; di++) {
                            for (int dj = -1; dj <= 1; dj++) {
                                int ni = i + di, nj = j + dj;
                                if (ni >= 0 && ni < H && nj >= 0 && nj < W && dist[ni][nj] == d - 1 && ans[ni][nj] == colors[color]) {
                                    used[ni][nj] = false;
                                }
                            }
                        }
                        color++;
                    }
                    ans[i][j] = colors[color];
                    used[i][j] = true;
                }
            }
        }
        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                if (ans[i][j] == 0) {
                    for (int k = 0; k < 4; k++) {
                        if (used[i][j]) {
                            continue;
                        }
                        boolean ok = true;
                        for (int di = -1; di <= 1; di++) {
                            for (int dj = -1; dj <= 1; dj++) {
                                int ni = i + di, nj = j + dj;
                                if (ni >= 0 && ni < H && nj >= 0 && nj < W && dist[ni][nj] <= dist[i][j] && ans[ni][nj] == colors[k]) {
                                    ok = false;
                                }
                            }
                        }
                        if (ok) {
                            ans[i][j] = colors[k];
                            used[i][j] = true;
                            break;
                        }
                    }
                }
                System.out.print(ans[i][j]);
            }
            System.out.println();
        }
    }
}
