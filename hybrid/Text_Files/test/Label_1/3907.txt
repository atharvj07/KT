#include <bits/stdc++.h>
using namespace std;

struct Car {
  long long cost, capacity;
};

int main() {
  long long n, k, s, t;
  cin >> n >> k >> s >> t;
  vector<Car> cars(n);
  for (auto &car : cars) { cin >> car.cost >> car.capacity; }
  vector<long long> gas(k);
  for (auto &g : gas) { cin >> g; }
  long long ans = -1;
  for (auto &car : cars) {
    long long need = 0, tim = 0;
    bool ok = true;
    gas.push_back(s);
    for (auto g : gas) {
      tim += need;
      long long dist = g - tim;
      if (dist > car.capacity) {
        ok = false;
        break;
      } else {
        need = car.capacity - dist;
        tim += dist;
      }
    }
    if (ok && tim <= t) { ans = min(ans == -1 ? LLONG_MAX : ans, car.cost); }
  }
  cout << ans << endl;
}
