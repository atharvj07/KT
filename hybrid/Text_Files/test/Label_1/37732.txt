a = input()
l = input()
r = input()
n = len(a)
dp = [[[-1] * 3 for j in range(n + 1)] for i in range(n + 1)]
def rec(i, j, t):
    if i == n:
        if j == 0:
            return 1
        return 0
    ans = dp[i][j][t]
    if ans != -1:
        return ans
    ans = 0
    if j != 0:
        ans += rec(i + 1, j - 1, t)
        ans %= 998244353
    if l <= a[i] <= r and (t == 0 or a[i] > a[i - 1]):
        ans += rec(i + 1, j + 1, t | (a[i] != '0'))
        ans %= 998244353
    dp[i][j][t] = ans
    return ans
print(rec(0, 0, 0))
