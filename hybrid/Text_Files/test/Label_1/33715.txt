import java.util.*;
import java.io.*;

public class Main {
    static class Graph {
        int v;
        LinkedList<Integer>[] adjList;

        public Graph(int v) {
            this.v = v;
            this.adjList = new LinkedList[v];
            for (int i = 0; i < v; i++) {
                adjList[i] = new LinkedList<>();
            }
        }

        public void addEdge(int u, int w) {
            adjList[u].add(w);
            adjList[w].add(u);
        }
    }

    static class QueueEntry {
        int vertex;
        int distance;

        public QueueEntry(int vertex, int distance) {
            this.vertex = vertex;
            this.distance = distance;
        }
    }

    static int[] bfs(Graph g, int source) {
        int[] distances = new int[g.v];
        Arrays.fill(distances, -1);
        Queue<QueueEntry> q = new LinkedList<>();
        q.add(new QueueEntry(source, 0));
        distances[source] = 0;
        while (!q.isEmpty()) {
            QueueEntry qe = q.poll();
            int v = qe.vertex;
            for (int w : g.adjList[v]) {
                if (distances[w] == -1) {
                    distances[w] = distances[v] + 1;
                    q.add(new QueueEntry(w, distances[v] + 1));
                }
            }
        }
        return distances;
    }

    static int[] dijkstra(Graph g, int source, int destination) {
        int[] distances = new int[g.v];
        Arrays.fill(distances, Integer.MAX_VALUE);
        PriorityQueue<QueueEntry> pq = new PriorityQueue<>(new Comparator<QueueEntry>() {
            @Override
            public int compare(QueueEntry o1, QueueEntry o2) {
                return o1.distance - o2.distance;
            }
        });
        pq.add(new QueueEntry(source, 0));
        distances[source] = 0;
        while (!pq.isEmpty()) {
            QueueEntry qe = pq.poll();
            int v = qe.vertex;
            for (int w : g.adjList[v]) {
                if (distances[w] > distances[v] + 1) {
                    distances[w] = distances[v] + 1;
                    pq.add(new QueueEntry(w, distances[v] + 1));
                }
            }
        }
        return distances;
    }

    static int[] findShortestPath(Graph g, int source, int destination) {
        int[] distances = dijkstra(g, source, destination);
        if (distances[destination] == Integer.MAX_VALUE) {
            return null;
        }
        int[] path = new int[distances[destination] + 1];
        int current = destination;
        int i = distances[destination];
        while (current != source) {
            path[i--] = current;
            for (int w : g.adjList[current]) {
                if (distances[w] == distances[current] - 1) {
                    current = w;
                    break;
                }
            }
        }
        path[0] = source;
        return path;
    }

    public static void main(String[] args) {
        try {
            BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
            String[] firstLine = br.readLine().split(" ");
            int n = Integer.parseInt(firstLine[0]);
            int m = Integer.parseInt(firstLine[1]);
            Graph g = new Graph(n);
            for (int i = 0; i < m; i++) {
                String[] line = br.readLine().split(" ");
                int u = Integer.parseInt(line[0]) - 1;
                int v = Integer.parseInt(line[1]) - 1;
                g.addEdge(u, v);
            }
            int[] distances1 = bfs(g, 0);
            int[] distances2 = bfs(g, n - 1);
            int minDistance = Integer.MAX_VALUE;
            int minSource = -1;
            int minDestination = -1;
            for (int i = 0; i < n; i++) {
                if (distances1[i] != -1 && distances2[i] != -1 && distances1[i] + distances2[i] < minDistance) {
                    minDistance = distances1[i] + distances2[i];
                    minSource = i;
                }
            }
            if (minSource == -1) {
                System.out.println(-1);
                return;
            }
            int[] path1 = findShortestPath(g, 0, minSource);
            int[] path2 = findShortestPath(g, n - 1, minSource);
            System.out.println(path1.length + path2.length - 1);
            for (int i = 0; i < path1.length; i++) {
                System.out.print((path1[i] + 1) + " ");
            }
            System.out.println();
            for (int i = path2.length - 2; i >= 0; i--) {
                System.out.print((path2[i] + 1) + " ");
            }
            System.out.println();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
