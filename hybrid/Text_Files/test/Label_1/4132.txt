#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, k;
        cin >> n >> k;
        vector<ll> a(n);
        for (auto &x : a) cin >> x;
        sort(a.begin(), a.end());
        ll sum = accumulate(a.begin(), a.end(), 0LL);
        for (int i = 0; i < k; ++i) {
            sum -= a[n - 1 - i];
            sum += a[i] / a[n - 1 - i];
        }
        cout << sum << endl;
    }
}
