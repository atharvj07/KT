from collections import defaultdict
n,m=map(int,input().split())
G = defaultdict(list)
for i in range(m):
    u,v,w = map(int,input().split())
    G[u].append((v,w))
vis = [False]*(n+1)
dp = [0]*(n+1)
def dfs(v):
    vis[v] = True
    for u, w in G[v]:
        if not vis[u] and w > dp[v]:
            dp[u] = w
            dfs(u)
dfs(1)
print(max(dp))
