import java.util.*;

public class BlogsOrder {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Integer>[] graph = new List[n + 1];
        for (int i = 0; i <= n; i++) {
            graph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            graph[a].add(b);
            graph[b].add(a);
        }

        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = scanner.nextInt();
        }

        List<Integer> order = new ArrayList<>();
        boolean[] visited = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                dfs(i, graph, desiredTopics, visited, order);
            }
        }

        if (order.size() != n) {
            System.out.println(-1);
        } else {
            for (int blog : order) {
                System.out.print(blog + " ");
            }
        }
    }

    private static void dfs(int blog, List<Integer>[] graph, int[] desiredTopics, boolean[] visited, List<Integer> order) {
        visited[blog] = true;
        for (int neighbor : graph[blog]) {
            if (!visited[neighbor]) {
                dfs(neighbor, graph, desiredTopics, visited, order);
            }
        }

        Set<Integer> coveredTopics = new HashSet<>();
        for (int neighbor : graph[blog]) {
            coveredTopics.add(desiredTopics[neighbor]);
        }

        for (int i = 1; i <= desiredTopics.length; i++) {
            if (!coveredTopics.contains(i)) {
                desiredTopics[blog] = i;
                break;
            }
        }

        order.add(blog);
    }
}
