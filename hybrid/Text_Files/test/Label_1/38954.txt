from sys import stdin
input = stdin.readline

n = int(input())
names = [input().rstrip() for i in range(n)]
pseudonyms = [input().rstrip() for i in range(n)]
names_set = {name: i for i, name in enumerate(names)}
pseudonyms_set = {pseudonym: i for i, pseudonym in enumerate(pseudonyms)}

lcp = [[max(0, min(len(names[u]), len(pseudonyms[v])) - i) for i in range(100)]
       for u in range(n) for v in range(n)]

dp = [[-1] * (1 << n) for i in range(n)]
matching = [[0] * (1 << n) for i in range(n)]

def get_dp(u, mask):
    if dp[u][mask] != -1: return dp[u][mask]

    ans = 0
    for v in range(n):
        if mask & (1 << v): continue
        ans = max(ans, get_dp(v, mask | (1 << v)) + lcp[u][v])
    dp[u][mask] = ans
    return ans

def get_matching(u, mask):
    if matching[u][mask] != 0: return matching[u][mask]

    for v in range(n):
        if mask & (1 << v): continue
        if get_dp(u, mask) == get_dp(v, mask | (1 << v)) + lcp[u][v]:
            matching[u][mask] = v
            return matching[u][mask]

print(get_dp(0, 0))
for u in range(n):
    v = get_matching(u, 1 << u)
    print(names_set[names[u]] + 1, pseudonyms_set[pseudonyms[v]] + 1)
