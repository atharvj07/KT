import java.io.*;
import java.util.*;

public class HanhAndATree {

    static final int MOD = 998244353;

    static class Edge {
        int u, v;

        Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }
    }

    static class Node {
        int id;
        long sum;
        long size;
        Node parent;
        List<Node> children;

        Node(int id) {
            this.id = id;
            this.children = new ArrayList<>();
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
        int n = Integer.parseInt(tokenizer.nextToken());
        int q = Integer.parseInt(tokenizer.nextToken());

        Node[] nodes = new Node[n + 1];
        for (int i = 1; i <= n; i++) {
            nodes[i] = new Node(i);
        }

        for (int i = 1; i < n; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int u = Integer.parseInt(tokenizer.nextToken());
            int v = Integer.parseInt(tokenizer.nextToken());
            nodes[u].children.add(nodes[v]);
            nodes[v].children.add(nodes[u]);
        }

        buildTree(nodes[1], null);

        for (int i = 0; i < q; i++) {
            tokenizer = new StringTokenizer(reader.readLine());
            int type = Integer.parseInt(tokenizer.nextToken());
            int v = Integer.parseInt(tokenizer.nextToken());

            if (type == 1) {
                int d = Integer.parseInt(tokenizer.nextToken());
                update(nodes[v], d);
            } else {
                long expectedValue = calculateExpectedValue(nodes[v]);
                long invSize = modularInverse(nodes[v].size, MOD);
                System.out.println((expectedValue * invSize) % MOD);
            }
        }
    }

    static void buildTree(Node node, Node parent) {
        node.parent = parent;
        node.size = 1;
        for (Node child : node.children) {
            if (child != parent) {
                buildTree(child, node);
                node.size += child.size;
            }
        }
    }

    static void update(Node node, int value) {
        node.sum += value;
        if (node.parent != null) {
            update(node.parent, value);
        }
    }

    static long calculateExpectedValue(Node node) {
        long expectedValue = node.sum;
        for (Node child : node.children) {
            if (child != node.parent) {
                expectedValue += calculateExpectedValue(child) * (node.size - child.size);
            }
        }
        return expectedValue;
    }

    static long modularInverse(long a, long m) {
        long m0 = m;
        long y = 0, x = 1;

        while (a > 1) {
            long q = a / m;
            long t = m;
            m = a % m;
            a = t;
            t = y;
            y = x - q * y;
            x = t;
        }

        if (x < 0) {
            x += m0;
        }

        return x;
    }
}
