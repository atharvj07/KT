import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        int[] result = new int[n];
        int balance = 0;
        int open = 0;
        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);
            if (c == '(') {
                balance++;
                open++;
            } else if (c == ')') {
                balance--;
                open--;
            } else {
                open--;
            }
            if (balance < 0 || open < 0) {
                result[i] = -1;
            } else {
                result[i] = Math.max(balance, 0);
            }
        }
        for (int r : result) {
            System.out.print(r + " ");
        }
        System.out.println();
    }
}
