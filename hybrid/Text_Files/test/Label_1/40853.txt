from collections import defaultdict
n, m = map(int, input().split())
g = defaultdict(list)
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)
vis = [False for i in range(n+1)]
vis[0] = True
ans = 0
for i in range(1, n+1):
    if not vis[i]:
        q = [i]
        while q:
            u = q.pop(0)
            if vis[u]:
                continue
            vis[u] = True
            for v in g[u]:
                if not vis[v]:
                    q.append(v)
        ans += 1
print(ans-1)
