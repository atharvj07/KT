#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 1e9 + 7;
const int maxn = 2e5 + 5;
ll arr[maxn];
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> arr[i]; }
    set<ll> st;
    int i = 0, j = n - 1;
    while (i <= j) {
        if (i == j) { st.insert(arr[i]); }
        if (st.empty()) {
            st.insert(arr[i]);
            st.insert(arr[j]);
            i++;
            j--;
        } else {
            ll maxi = *st.rbegin();
            if (arr[i] <= maxi) {
                j--;
                st.insert(arr[j]);
            } else if (arr[j] <= maxi) {
                i++;
                st.insert(arr[i]);
            } else {
                st.clear();
                st.insert(arr[i]);
                st.insert(arr[j]);
                i++;
                j--;
            }
        }
    }
    int sz = st.size();
    if (sz == n)
        cout << "Alice";
    else
        cout << "Bob";
    return 0;
}
