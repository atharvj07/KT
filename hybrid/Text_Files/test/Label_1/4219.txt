#include <bits/stdc++.h>

using namespace std;
using lu = long unsigned;
using ll = long long;
constexpr lu MOD = 1000000007;

lu mod_pow(lu base, lu exp) {
    lu res = 1;
    while (exp) {
        if (exp % 2) { res = (res * base) % MOD; }
        base = (base * base) % MOD;
        exp /= 2;
    }
    return res;
}

lu solve(int n, int k) { return mod_pow(k, n + n - 2); }

int main() {
    int n, k;
    cin >> n >> k;
    cout << solve(n, k) << endl;
    return 0;
}
