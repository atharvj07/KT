#include<iostream>
#include<vector>
using namespace std;

int main() {
    int n, l, r;
    cin >> n >> l >> r;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    for (int i = 0; i < n; ++i)
        cin >> b[i];
    vector<int> p(n, -1);
    for (int i = 0; i < n; ++i)
        p[b[i]] = i;
    for (int i = l - 1; i < r; ++i) {
        if (p[a[i]] < l - 1 || p[a[i]] > r) {
            cout << "LIE";
            return 0;
        }
    }
    cout << "TRUTH";
}
