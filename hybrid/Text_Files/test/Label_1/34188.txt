import java.util.*;
import java.math.*;

class Main {
  static BigInteger[] longPow = new BigInteger[500];
  static int[] smallPow = new int[500];

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    long x = scanner.nextLong();
    longPow[0] = BigInteger.ONE;
    for (int i = 1; i <= 499; i++) {
      longPow[i] = longPow[i - 1].multiply(BigInteger.valueOf(x));
    }
    int ans = -1;
    for (int b = 10; b <= 100000; b++) {
      int p = 0;
      while (x % b == 0) {
        x /= b;
        p++;
      }
      if (x == 1) {
        for (int i = 0; i < p; i++) {
          b *= 10;
        }
        boolean ok = true;
        BigInteger pow1 = longPow[n - 1];
        BigInteger pow2 = longPow[n];
        BigInteger mask = BigInteger.valueOf(b).pow(n - 1);
        BigInteger now = BigInteger.valueOf(0);
        for (int i = 0; i < p; i++) {
          if (!pow1.equals(pow2.and(mask))) {
            ok = false;
            break;
          }
          mask = mask.divide(BigInteger.valueOf(b));
          now = now.multiply(BigInteger.valueOf(b)).or(pow1).mod(longPow[n]);
          BigInteger t = pow1;
          pow1 = pow2.and(mask);
          pow2 = t.multiply(BigInteger.valueOf(b)).mod(longPow[n]);
        }
        if (ok) {
          int t = 0;
          BigInteger bCopy = BigInteger.valueOf(b);
          while (bCopy.compareTo(BigInteger.ONE) > 0) {
            bCopy = bCopy.divide(BigInteger.valueOf(10));
            t++;
          }
          ans = Math.max(ans, t);
        }
      }
    }
    System.out.println(ans);
  }
}
