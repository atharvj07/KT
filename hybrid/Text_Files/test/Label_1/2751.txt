#include<bits/stdc++.h>
using namespace std;
int main() {
    int n, a, b, c, d;
    cin >> n;
    int h[2][600002] = {{}};
    for(int i = 0; i < n; i++) {
        cin >> a >> b >> c >> d;
        h[0][a + 300000]++;
        h[0][c + 300000]--;
        h[1][b + 300000]++;
        h[1][d + 300000]--;
    }
    int sumx = 0;
    int sumy = 0;
    for(int i = 0; i < 600000; i++) {
        sumx += h[0][i];
        sumy += h[1][i];
        if(sumx >= n - 1 && sumy >= n - 1) {
            int ansx = i - 300000;
            int ansy = sumy == n - 1 ? i - 300000 + 1 : i - 300000;
            cout << ansx << " " << ansy;
            return 0;
        }
    }
    return 0;
}
