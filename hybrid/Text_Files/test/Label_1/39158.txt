def power(x, y, p):
    res = 1
    x = x % p
    while y > 0:
        if y & 1:
            res = (res * x) % p
        y = y >> 1
        x = (x * x) % p
    return res

def countWalls(n, C):
    MOD = 10**6 + 3
    if C == 1:
        return n % MOD
    if n == 1:
        return 1
    inv2 = (MOD+1)//2
    ans = power(2, n, MOD) - 1
    ans = (ans - power(2, n-C, MOD) + MOD) % MOD
    ans = (ans + power(inv2, C, MOD)) % MOD
    return ans

n, C = map(int, input().split())
print(countWalls(n, C))
