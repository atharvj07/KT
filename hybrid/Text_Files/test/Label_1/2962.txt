#include <bits/stdc++.h>
#define ll long long
#define sz(x) (ll)x.size()
using namespace std;

int main() {
    string g;
    cin >> g;
    string p;
    cin >> p;
    int pos = -1;
    for (int i = 0; i < p.size(); i++) {
        if (p[i] == '*') {
            pos = i;
            break;
        }
    }
    ll n;
    cin >> n;
    while (n--) {
        string s;
        cin >> s;
        ll f = 1;
        for (int i = 0; i < p.size(); i++) {
            if (i == pos)
                continue;
            if (p[i] != s[i] && p[i] != '?') {
                f = 0;
                break;
            }
        }
        string bad = "";
        for (ll i = 0; i < 26; i++) {
            if (find(g.begin(), g.end(), (char)(i + 'a')) == g.end()) {
                bad += (char)(i + 'a');
            }
        }
        if (pos != -1) {
            string a = "";
            for (ll i = pos; i < s.size(); i++) { a += s[i]; }
            bool found = 1;
            for (ll i = 0; i < a.size(); i++) {
                if (find(bad.begin(), bad.end(), a[i]) == bad.end()) {
                    found = 0;
                    break;
                }
            }
            f *= found;
        }
        if (f)
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
