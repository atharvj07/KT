#include <bits/stdc++.h>
 
using namespace std;
 
long long n;
 
int main() {
    cin >> n;
    vector<long long> ans;
    for (long long x = n; x > 0; ) {
        long long cur = x;
        int deg = 0;
        while (cur > 0) {
            cur /= 2;
            deg++;
        }
        deg--;
        long long val = (1LL << deg);
        x -= val;
        ans.push_back(val);
    }
    int sz = ans.size();
    cout << sz << endl;
    for (int i = 0; i < sz; i++) cout << ans[i] << ' ';
    cout << endl;
    return 0;
}
