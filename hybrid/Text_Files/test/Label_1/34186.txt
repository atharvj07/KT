import java.util.*;

public class Main {
  static int[][] board = new int[21][21];
  static boolean[][] fallen = new boolean[21][21];

  static int[] dx = {0, 1, 0, -1};
  static int[] dy = {1, 0, -1, 0};
  static char[] dir = {'N', 'E', 'S', 'W'};

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    while (true) {
      int N = sc.nextInt();
      if (N == 0) {
        break;
      }
      for (int i = 0; i < N; i++) {
        int x = sc.nextInt();
        int y = sc.nextInt();
        fallen[x][y] = true;
      }
      int M = sc.nextInt();
      int x = 10, y = 10;
      for (int i = 0; i < M; i++) {
        char cmdDir = sc.next().charAt(0);
        int cmdLen = sc.nextInt();
        for (int j = 0; j < 4; j++) {
          if (dir[j] == cmdDir) {
            for (int k = 0; k < cmdLen; k++) {
              x += dx[j];
              y += dy[j];
              if (fallen[x][y]) {
                board[x][y] |= 1 << j;
              }
            }
            break;
          }
        }
      }
      for (int i = 1; i <= 20; i++) {
        for (int j = 1; j <= 20; j++) {
          if (fallen[i][j]) {
            if (board[i][j] != 15) {
              System.out.println("No");
              return;
            }
          }
        }
      }
      System.out.println("Yes");
    }
  }
}
