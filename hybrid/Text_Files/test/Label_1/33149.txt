import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());
        int[] x = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        int[] y = Arrays.stream(br.readLine().split(" ")).mapToInt(Integer::parseInt).toArray();
        long res = 0L;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                int bit = x[i] ^ y[j];
                int count = 0;
                for (int k = 0; k < n; k++) {
                    if (x[k] == bit) count++;
                    if (y[k] == bit) count++;
                }
                res += (count <= 1 ? 1 : 0);
            }
        }
        if (res % 2 == 0) System.out.println("Karen");
        else System.out.println("Koyomi");
    }
}
