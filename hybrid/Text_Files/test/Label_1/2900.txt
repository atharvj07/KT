#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, a[10005], t = 0;
    cin >> n;
    for (int i = 0; i < n; i++) { cin >> a[i]; }
    while (true) {
        bool ok = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] != 0) {
                ok = 1;
                if (i == a[i] - 1) a[i] = 0;
            }
        }
        t += (ok == 0);
        if (ok == 0) break;
    }
    cout << t << endl;
    return 0;
}
