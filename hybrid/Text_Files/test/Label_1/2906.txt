#include <bits/stdc++.h>

using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n, w;
        cin >> n >> w;
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
        }
        int ans = 0;
        sort(a.begin(), a.end());
        if (w == 1) {
            for (int i = 0; i < n; i++) {
                ans += a[i];
            }
        } else {
            int sum = 0, cnt = 0;
            for (int i = n - 1; i >= 0; i--) {
                if (sum + a[i] <= w) {
                    sum += a[i];
                    cnt++;
                } else {
                    while (cnt > 0) {
                        ans++;
                        cnt--;
                        sum -= a[i + cnt];
                    }
                    cnt++;
                    sum += a[i];
                }
            }
            if (sum > 0) ans++;
        }
        cout << ans << '\n';
    }
    return 0;
}
