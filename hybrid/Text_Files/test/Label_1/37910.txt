n, m = map(int, input().split())
sum_x, sum_d = 0, 0
for _ in range(m):
    x, d = map(int, input().split())
    sum_x += x
    sum_d += d if d > 0 else 0

mean = sum_x * n
if sum_d > 0:
    if n % 2 == 1:
        mean += sum_d * ((n // 2) * ((n // 2) + 1))
    else:
        mean += sum_d * ((n // 2) * (n // 2))

print(mean / n)
