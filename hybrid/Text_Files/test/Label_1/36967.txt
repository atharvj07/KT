def dfs(graph, pleasantness, node, parent, dp):
    dp[node] = pleasantness[node-1]
    for child in graph[node]:
        if child != parent:
            dfs(graph, pleasantness, child, node, dp)
            dp[node] += max(dp[child], 0)

def solve():
    n = int(input().strip())
    if n == 1:
        print("Impossible")
        return

    pleasantness = list(map(int, input().strip().split()))
    graph = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, input().strip().split())
        graph[u].append(v)
        graph[v].append(u)

    dp = [0] * (n+1)
    dfs(graph, pleasantness, 1, -1, dp)

    max_sum = sum([max(x,0) for x in pleasantness])
    result = -float("inf")

    for i in range(2, n+1):
        without_i = max_sum - max(dp[i], 0)
        with_i = dp[i]
        result = max(result, without_i + with_i)

    print(result)

solve()
