#include <bits/stdc++.h>
using namespace std;
int b, x, y;
string s;
int val(char c) {
    if ('0' <= c && c <= '9') return c - '0';
    else return c - 'A' + 10;
}
char cmp(int x, int y) {
    if (x < y) return '<';
    else if (y < x) return '>';
    else return '=';
}
int main() {
    cin >> x >> b;
    for (int j = 0; j < x; j++) {
        cin >> s;
        y += val(s[0]) * pow(b, x - j - 1);
    }
    cin >> x >> b;
    for (int j = 0; j < x; j++) {
        cin >> s;
        y += val(s[0]) * pow(b, x - j - 1);
    }
    cout << cmp(x, y) << endl;
}
