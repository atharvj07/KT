import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        ArrayList<Integer> a = new ArrayList<>();
        ArrayList<Integer> num_count_by_student = new ArrayList<>();
        int total_num_count = 0;
        int num_not_exist = 0;
        int student_with_num_not_exist = 0;
        for (int i = 1; i <= n; i++) {
            a.add(in.nextInt());
            num_count_by_student.add(0);
            total_num_count++;
        }
        if (total_num_count % 2 == 1) {
            System.out.println("! -1");
            return;
        }
        for (int i = 0; i < n - 1; i++) {
            if (Math.abs(a.get(i) - a.get(i + 1)) != 1) {
                System.out.println("! -1");
                return;
            }
            num_count_by_student.set(i, num_count_by_student.get(i) + 1);
            num_count_by_student.set(i + 1, num_count_by_student.get(i + 1) + 1);
        }
        num_count_by_student.set(n - 1, num_count_by_student.get(n - 1) + 1);
        num_count_by_student.set(0, num_count_by_student.get(0) + 1);
        for (int i = 0; i < n; i++) {
            if (num_count_by_student.get(i) == 1) {
                num_not_exist++;
                student_with_num_not_exist = i + 1;
            } else if (num_count_by_student.get(i) > 2) {
                System.out.println("! -1");
                return;
            }
        }
        if (num_not_exist == 2) {
            System.out.println("! " + student_with_num_not_exist);
        } else {
            System.out.println("! -1");
        }
    }
}
