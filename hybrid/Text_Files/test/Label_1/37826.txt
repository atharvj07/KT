import sys
input = sys.stdin.readline

n, m = map(int, input().split())
a = list(map(int, input().split()))
k = list(map(int, input().split()))
L = [0] * (m + 2)
R = [0] * (m + 2)
ans = n
for i in range(n):
    L[a[i]] = i + 1
    R[a[i]] = i + 1
    cnt = 0
    for j in range(1, m + 1):
        if L[j] > 0 and R[j] > 0:
            cnt += R[j] - L[j] + 1
    ans = min(ans, n - cnt)
print(ans if ans < n else -1)
