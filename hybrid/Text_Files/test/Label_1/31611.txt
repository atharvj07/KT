import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.HashMap;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] firstLine = reader.readLine().split(" ");
        int h = Integer.parseInt(firstLine[0]);
        int m = Integer.parseInt(firstLine[1]);
        int n = Integer.parseInt(firstLine[2]);
        HashMap<Integer, Integer> table = new HashMap<>();
        int dummyCount = 0;
        for (int i = 0; i < n; i++) {
            String[] line = reader.readLine().split(" ");
            if (line[0].equals("+")) {
                int id = Integer.parseInt(line[1]);
                int hash = Integer.parseInt(line[2]);
                int cell = hash;
                int count = 0;
                while (table.containsKey(cell)) {
                    cell = (cell + m) % h;
                    count++;
                }
                table.put(cell, id);
                dummyCount += count;
            } else {
                int id = Integer.parseInt(line[1]);
                for (int j = 0; j < h; j++) {
                    int hash = (id + j * m) % h;
                    if (table.containsKey(hash) && table.get(hash) == id) {
                        table.remove(hash);
                        break;
                    }
                }
            }
        }
        reader.close();
        System.out.println(dummyCount);
    }
}
