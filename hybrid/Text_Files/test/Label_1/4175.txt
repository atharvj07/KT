#include <iostream>
#include <vector>
using namespace std;

const int MAXN = 200000;
int n;
vector<int> a;
int gcd(int a, int b){while(b)a%=b,swap(a,b);}

int main(){
    cin >> n;
    a.resize(n);
    for(int i = 0; i < n; i++){
        cin >> a[i];
    }
    int g = a[0];
    for(int i = 1; i < n; i++){
        g = gcd(g, a[i]);
    }
    if(g > 0)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
