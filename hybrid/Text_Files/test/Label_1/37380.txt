n, m = map(int, input().split())
g = [0] * n
ans = 0
for i in range(n):
    g[i] = list(map(int, input().split()))
    for j in range(m):
        if g[i][j] == 1:
            if (i == 0 or j == 0 or i == n - 1 or j == m - 1):
                ans = -1
                break
            ans = max(max(ans, i), n - 1 - i, j, m - 1 - j)
if ans == -1:
    print(ans)
else:
    print(ans * ans)
