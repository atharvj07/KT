#include <iostream>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;
int main() {
  string a, b;
  cin >> a >> b;
  vector<int> numA, numB;
  for (int i = 0; i < a.length(); i++) numA.push_back(a[i] - '0');
  for (int i = 0; i < b.length(); i++) numB.push_back(b[i] - '0');
  sort(numA.begin(), numA.end());
  sort(numB.begin(), numB.end());
  for (int i = 0; i < a.length(); i++) {
    for (int j = a.length() - 1; j >= i; j--) {
      if (numA[i] > numB[j]) swap(numA[i], numB[j]);
    }
  }
  bool flag = false;
  while (!numA.empty() && !flag) {
    if (numA.back() != 0) flag = true;
    else numA.pop_back();
  }
  if (numA.empty()) {
    string temp = "";
    for (int i = 0; i < numB.size(); i++) temp += to_string(numB[i]);
    cout << temp;
  } else {
    string temp = "";
    for (int i = 0; i < numA.size(); i++) temp += to_string(numA[i]);
    cout << temp;
  }
  return 0;
}
