def bracket_sequences(n):
    if n == 1:
        return ['()']
    sequences = []
    for s in bracket_sequences(n - 1):
        for i in range(2 * n - 1):
            if i % 2 == 0 and s[i] == '(':
                sequences.append(s[:i] + '()' + s[i:])
            elif i % 2 == 1 and s[i] == ')':
                sequences.append(s[:i] + '()' + s[i:])
    return sequences

for _ in range(int(input())):
    n = int(input())
    print(*bracket_sequences(n))
