#include <iostream>
#include <vector>

int main() {
    int n;
    std::cin >> n;
    std::vector<int> a(n);
    for (int i = 0; i < n; i++) {
        std::cin >> a[i];
    }
    std::string s;
    for (int i = 0; i < n; i++) {
        if (a[i] == 0) {
            s += '+';
        } else if (a[i] == 1) {
            s += '-';
        } else {
            s += '+';
            a[i]--;
        }
    }
    std::cout << s << std::endl;
    return 0;
}
