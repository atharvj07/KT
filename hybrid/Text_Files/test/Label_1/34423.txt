import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;

class Main {

    static class State {
        int x;
        int y;
        int k;
        char prevCell;
        String path;

        public State(int x, int y, int k, char prevCell, String path) {
            this.x = x;
            this.y = y;
            this.k = k;
            this.prevCell = prevCell;
            this.path = path;
        }
    }

    private static int n;
    private static int m;
    private static int k;
    private static char[][] grid;
    private static boolean[][][] visited;
    private static final List<Character> types = new ArrayList<>();

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        n = Integer.parseInt(tokens[0]);
        m = Integer.parseInt(tokens[1]);
        k = Integer.parseInt(tokens[2]);
        grid = new char[n][m];
        visited = new boolean[n][m][k + 1];
        for (int i = 0; i < n; i++) {
            grid[i] = br.readLine().toCharArray();
        }
        bfs();
    }

    private static void bfs() {
        Queue<State> queue = new ArrayDeque<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (grid[i][j] == 'S') {
                    queue.add(new State(i, j, k, 'S', ""));
                    visited[i][j][k] = true;
                }
            }
        }
        State bestState = null;
        while (!queue.isEmpty()) {
            State state = queue.poll();
            if (grid[state.x][state.y] == 'T') {
                if (bestState == null || state.path.compareTo(bestState.path) < 0) {
                    bestState = state;
                }
                continue;
            }
            char cellType = grid[state.x][state.y];
            if (cellType != 'S' && cellType != 'T' && !types.contains(cellType)) {
                types.add(cellType);
            }
            if (state.k > 0) {
                for (int dx = -1; dx <= 1; dx++) {
                    for (int dy = -1; dy <= 1; dy++) {
                        int nx = state.x + dx;
                        int ny = state.y + dy;
                        if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny][state.k - 1] &&
                                (grid[nx][ny] == cellType || grid[nx][ny] == 'S' || grid[nx][ny] == 'T')) {
                            queue.add(new State(nx, ny, state.k - 1, cellType, state.path + cellType));
                            visited[nx][ny][state.k - 1] = true;
                        }
                    }
                }
            }
        }
        if (bestState != null) {
            System.out.println(bestState.path.substring(1, bestState.path.length() - 1));
        } else {
            System.out.println("-1");
        }
    }
}
