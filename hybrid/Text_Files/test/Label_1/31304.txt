import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class Main {

    private static class Graph {

        private final Map<Integer, List<Integer>> adjacencyList;
        private final int vertexCount;
        private final int edgeCount;

        public Graph(int vertexCount, int edgeCount) {
            this.vertexCount = vertexCount;
            this.edgeCount = edgeCount;
            adjacencyList = new HashMap<>();
            for (int i = 1; i <= vertexCount; i++) {
                adjacencyList.put(i, new ArrayList<>());
            }
        }

        public void addEdge(int u, int v) {
            adjacencyList.get(u).add(v);
            adjacencyList.get(v).add(u);
        }

        public int getVertexCount() {
            return vertexCount;
        }

        public int getEdgeCount() {
            return edgeCount;
        }

        public List<Integer> getAdjacentVertices(int vertex) {
            return adjacencyList.get(vertex);
        }
    }

    private static class Edge implements Comparable<Edge> {

        private final int u;
        private final int v;

        public Edge(int u, int v) {
            this.u = u;
            this.v = v;
        }

        @Override
        public int compareTo(Edge other) {
            return Integer.compare(u, other.u);
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        String[] line = reader.readLine().split(" ");
        int vertexCount = Integer.parseInt(line[0]);
        int edgeCount = Integer.parseInt(line[1]);
        Graph graph = new Graph(vertexCount, edgeCount);
        for (int i = 0; i < edgeCount; i++) {
            line = reader.readLine().split(" ");
            int u = Integer.parseInt(line[0]);
            int v = Integer.parseInt(line[1]);
            graph.addEdge(u, v);
        }
        Set<Edge> edgesToKeep = getEdgesToKeep(graph);
        System.out.println(edgesToKeep.size());
        for (Edge edge : edgesToKeep) {
            System.out.println(edge.u + " " + edge.v);
        }
    }

    private static Set<Edge> getEdgesToKeep(Graph graph) {
        Map<Integer, Integer> vertexDegrees = new HashMap<>();
        for (int vertex = 1; vertex <= graph.getVertexCount(); vertex++) {
            vertexDegrees.put(vertex, graph.getAdjacentVertices(vertex).size());
        }
        List<Integer> verticesSortedByDegree = new ArrayList<>(vertexDegrees.keySet());
        Collections.sort(verticesSortedByDegree, (u, v) -> Integer.compare(vertexDegrees.get(u), vertexDegrees.get(v)));
        Set<Edge> edgesToKeep = new HashSet<>();
        for (int vertex : verticesSortedByDegree) {
            List<Integer> adjacentVertices = graph.getAdjacentVertices(vertex);
            int degree = vertexDegrees.get(vertex);
            for (int i = 0; i < degree / 2; i++) {
                int adjacentVertex = adjacentVertices.get(i);
                edgesToKeep.add(new Edge(vertex, adjacentVertex));
            }
        }
        return edgesToKeep;
    }
}
