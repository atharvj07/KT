import java.util.*;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int t = in.nextInt();
        int k = in.nextInt();
        BigInteger[] dp = new BigInteger[100000];
        dp[0] = BigInteger.ONE;
        dp[1] = BigInteger.ONE;
        for (int i = 2; i < 100000; i++) {
            dp[i] = dp[i - 1].multiply(BigInteger.valueOf(i)).mod(BigInteger.valueOf(1000000007));
        }
        for (int i = 0; i < t; i++) {
            int a = in.nextInt();
            int b = in.nextInt();
            BigInteger ans = dp[b].multiply(dp[k - 1].modPow(BigInteger.valueOf(b - a), BigInteger.valueOf(1000000007))).mod(BigInteger.valueOf(1000000007));
            ans = ans.multiply(dp[a - 1].modInverse(BigInteger.valueOf(1000000007))).mod(BigInteger.valueOf(1000000007));
            System.out.println(ans);
        }
    }
}
