n = int(input())
a = list(map(int, input().split()))
if a[0] == 0 or sum(a) <= n - 1:
    print(-1)
    exit()
m = n - 1
print(m)
cur = [1]
for i in range(2, n + 1):
    if a[i - 1] == 0:
        continue
    for j in cur:
        if j == i:
            continue
        print(j, i)
        m -= 1
        a[i - 1] -= 1
        if a[i - 1] == 0:
            break
    cur.append(i)
