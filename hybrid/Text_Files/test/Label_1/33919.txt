import java.util.*;
class Main {
  static class Node {
    int v;
    int num;
    Node(int v, int num) {
      this.v = v;
      this.num = num;
    }
  }
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] a = new int[n];
    int[] b = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    for (int i = 0; i < n; i++) {
      b[i] = sc.nextInt();
    }
    int[][] roads = new int[m][2];
    for (int i = 0; i < m; i++) {
      roads[i][0] = sc.nextInt() - 1;
      roads[i][1] = sc.nextInt() - 1;
    }
    int[][] ans = new int[n][n];
    int[] sum = new int[n];
    for (int i = 0; i < n; i++) {
      sum[i] = a[i];
    }
    for (int i = 0; i < m; i++) {
      if (b[roads[i][0]] - sum[roads[i][0]] < 0 || b[roads[i][1]] - sum[roads[i][1]] < 0) {
        System.out.println("NO");
        return;
      }
      int diff = b[roads[i][0]] - sum[roads[i][0]];
      int diff2 = b[roads[i][1]] - sum[roads[i][1]];
      ans[roads[i][0]][roads[i][1]] = Math.min(diff, diff2);
      sum[roads[i][0]] += ans[roads[i][0]][roads[i][1]];
      sum[roads[i][1]] += ans[roads[i][0]][roads[i][1]];
    }
    for (int i = 0; i < n; i++) {
      if (sum[i] != b[i]) {
        System.out.println("NO");
        return;
      }
    }
    System.out.println("YES");
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++) {
        System.out.print(ans[i][j] + " ");
      }
      System.out.println();
    }
  }
}
