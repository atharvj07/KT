import java.util.*;
import java.io.*;

public class Main {
    static int[][] dist;
    static boolean[][] connected;
    static int[] path;

    public static void main(String[] args) throws IOException {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = in.nextInt(), x = in.nextInt(), y = in.nextInt();
        dist = new int[n + 1][n + 1];
        connected = new boolean[n + 1][n + 1];
        path = new int[n + 1];
        for (int i = 1; i < n; i++) {
            dist[i][i + 1] = dist[i + 1][i] = 1;
            connected[i][i + 1] = connected[i + 1][i] = true;
        }
        dist[x][y] = dist[y][x] = 1;
        connected[x][y] = connected[y][x] = true;
        for (int k = 1; k <= n; k++) {
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (connected[i][j]) {
                        if (dist[i][k] + dist[k][j] < dist[i][j]) {
                            dist[i][j] = dist[i][k] + dist[k][j];
                            path[j] = k;
                        }
                    }
                }
            }
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    if (dist[i][j] == k) {
                        connected[i][j] = true;
                    }
                }
            }
        }
        for (int k = 1; k < n; k++) {
            long ans = 0;
            for (int i = 1; i <= n; i++) {
                for (int j = i + 1; j <= n; j++) {
                    if (dist[i][j] == k) {
                        ans++;
                    }
                }
            }
            out.println(ans);
        }
        in.close();
        out.close();
    }
}
