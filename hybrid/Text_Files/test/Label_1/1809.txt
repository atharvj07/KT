#include <bits/stdc++.h>
using namespace std;

int main() {
  int n;
  cin >> n;
  long long arr[n], ans = LLONG_MAX;
  for (int i = 0; i < n; i++) {
    cin >> arr[i];
  }
  sort(arr, arr + n);
  for (int i = 1; i < n; i++) {
    ans = min(ans, arr[i] - arr[i - 1] - 1);
  }
  cout << ans << endl;
}
