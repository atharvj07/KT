#include <iostream>
#include <vector>

using namespace std;

int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    string a, b;
    cin >> a >> b;
    bool possible = true;
    int last_dif = 0;
    for (int i = 0; i < n; i++) {
      if (a[i] != b[i]) {
        if (a[i] == 'z') {
          possible = false;
          break;
        }
        if (i < k - 1) {
          if (i != last_dif) {
            possible = false;
            break;
          }
        } else {
          last_dif = i;
        }
      }
    }
    if (possible)
      cout << "Yes" << endl;
    else
      cout << "No" << endl;
  }
  return 0;
}
