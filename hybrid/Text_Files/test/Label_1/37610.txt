import sys

mod = 998244353

n = int(input())
x, y, s = zip(*(map(int, input().split()) for _ in range(n)))

tot = 0
j = 0  # Number of active portals
for i in range(n):
    tot = (tot + x[i] - s[i] * y[i] + mod) % mod
    if s[i] == 0:
        j += 1
    if j:
        tot = (tot + mod - x[i] + y[i] + j * (x[i] - y[i])) % mod
    if x[i] > x[i - 1] + 1 and j:
        tot = (tot + x[i] - x[i - 1] - j) % mod
print(tot)
