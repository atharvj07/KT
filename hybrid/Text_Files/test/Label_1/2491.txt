#include <iostream>
#include <vector>

int main() {
    int n, m;
    std::cin >> n >> m;
    std::vector<std::string> house(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> house[i];
    }
    int count = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < 2 * m; j += 2) {
            if (house[i][j] == '1' || house[i][j + 1] == '1') {
                count++;
                break;
            }
        }
    }
    std::cout << count << std::endl;
    return 0;
}
