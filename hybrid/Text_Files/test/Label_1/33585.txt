import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        sc.nextLine();
        char[][] grid = new char[n][m];
        int eaten = 0;
        for (int i = 0; i < n; i++) {
            String s = sc.nextLine();
            for (int j = 0; j < m; j++) {
                grid[i][j] = s.charAt(j);
                if (grid[i][j] == 'P' && (i == 0 || i == n - 1 || j == 0 || j == m - 1))
                    eaten += isEaten(grid, i, j);
                else if (grid[i][j] == 'P') {
                    int adj = 0;
                    for (int k = -1; k <= 1; k++) {
                        for (int l = -1; l <= 1; l++) {
                            if (k == 0 && l == 0)
                                continue;
                            if (i + k >= 0 && i + k < n && j + l >= 0 && j + l < m && grid[i + k][j + l] == 'W')
                                adj++;
                        }
                    }
                    if (adj == 1)
                        eaten += isEaten(grid, i, j);
                }
            }
        }
        System.out.println(eaten);
    }

    static int isEaten(char[][] grid, int x, int y) {
        int n = grid.length, m = grid[0].length;
        for (int i = -1; i <= 1; i++) {
            for (int j = -1; j <= 1; j++) {
                if (i == 0 && j == 0)
                    continue;
                if (x + i >= 0 && x + i < n && y + j >= 0 && y + j < m && grid[x + i][y + j] == 'W') {
                    return 1;
                }
            }
        }
        return 0;
    }
}
