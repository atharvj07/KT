import java.util.Scanner;
import java.util.HashSet;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        String num = in.next();
        HashSet<String> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                String tmp = num.substring(0, i) + num.charAt(j) + num.substring(i + 1);
                if (set.contains(tmp)) {
                    System.out.println("NO");
                    return;
                } else {
                    set.add(tmp);
                }
            }
        }
        System.out.println("YES");
    }
}
