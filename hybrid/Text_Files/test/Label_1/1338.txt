#include<bits/stdc++.h>
using namespace std;
int a[1000005];
int main(){
    long long int n, k, ans = 0, e = 0, p = 0, l = 0;
    cin >> n >> k;
    if (n == 3) {
        if (k == 3) {
            cout << "Yes" << endl;
            cout << "1 1 1" << endl;
            return 0;
        } else {
            cout << "No" << endl;
            return 0;
        }
    }
    if (n < (1 << k) || (k >= 60 && n > 1)) {
        cout << "No" << endl;
        return 0;
    }
    for (int i = 0; i <= k; i++) a[i] = 0;
    while (n > 0) {
        a[e] = (int)log2(n);
        if (a[e] == l) p++;
        if (n >= (1 << a[e])) {
            n -= (1 << a[e]);
            l = a[e];
        }
        if (p >= 2) {
            cout << "No" << endl;
            return 0;
        }
        ans += (1 << a[e]);
        e++;
    }
    if (ans == n) {
        cout << "Yes" << endl;
        for (int i = k - 1; i >= 0; i--) cout << a[i] << " ";
        return 0;
    }
    cout << "No" << endl;
    return 0;
}
