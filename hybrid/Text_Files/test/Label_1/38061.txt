import math

a, b, c = map(int, input().split())
d = b**2 - 4*a*c
if d < 0:
    print('Equation has no real roots.')
elif d == 0:
    print('Equation has only one real root.')
else:
    r1 = (-b + math.sqrt(d)) / (2*a)
    r2 = (-b - math.sqrt(d)) / (2*a)
    print('%0.12f' % max(r1, r2))
    print('%0.12f' % min(r1, r2))
