import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        String ticket = input.nextLine();

        int[] digits = new int[6];
        for (int i = 0; i < 6; i++) {
            digits[i] = Integer.parseInt(ticket.substring(i, i + 1));
        }

        int sumLeft = digits[0] + digits[1] + digits[2];
        int sumRight = digits[3] + digits[4] + digits[5];

        int minReplacements = Math.abs(sumLeft - sumRight) / 2;

        System.out.println(minReplacements);
    }
}
