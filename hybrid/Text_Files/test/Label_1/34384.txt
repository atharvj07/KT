import java.util.Scanner;

public class Main {
  static class Point {
    double x, y;
  }

  static double dot(Point a, Point b) {
    return a.x * b.x + a.y * b.y;
  }

  static double abs(Point a) {
    return Math.sqrt(a.x * a.x + a.y * a.y);
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);
    int n = scanner.nextInt();
    Point[] points = new Point[n];
    for (int i = 0; i < n; i++) points[i] = new Point();
    for (int i = 0; i < n; i++) {
      points[i].x = scanner.nextInt();
      points[i].y = scanner.nextInt();
    }
    double l = 0, r = 1e9;
    for (int it = 0; it < 100; it++) {
      double mid = (l + r) / 2;
      boolean ok = true;
      for (int i = 0; i < n; i++) {
        Point p1 = points[(i - 1 + n) % n], p2 = points[i], p3 = points[(i + 1) % n];
        Point v1 = new Point();
        v1.x = p2.x - p1.x;
        v1.y = p2.y - p1.y;
        double l1 = abs(v1);
        Point v2 = new Point();
        v2.x = p3.x - p2.x;
        v2.y = p3.y - p2.y;
        double l2 = abs(v2);
        v1.x /= l1;
        v1.y /= l1;
        double d = dot(v1, v2);
        if (d + mid > 1) {
          ok = false;
          break;
        }
        Point v3 = new Point();
        v3.x = p2.x - p3.x;
        v3.y = p2.y - p3.y;
        d = dot(v2, v3);
        if (d + mid > 1) {
          ok = false;
          break;
        }
      }
      if (ok) l = mid;
      else r = mid;
    }
    System.out.println(l);
  }
}
