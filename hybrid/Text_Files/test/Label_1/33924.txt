import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        sc.close();
        Arrays.sort(a);
        int ans = n;
        for (int i = 0; i < m; i++) {
            if (ans > a[i]) {
                ans -= a[i];
            } else {
                ans = -1;
                break;
            }
        }
        System.out.println(ans);
    }
}
