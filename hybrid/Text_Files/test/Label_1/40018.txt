import sys
input = sys.stdin.readline

x1, y1, x2, y2, x3, y3 = map(int, input().split())

s1 = x1 * y1
s2 = x2 * y2
s3 = x3 * y3
if s1 + s2 + s3 == 0 or (s1 == s2 and s2 == s3):
    print(-1)
    exit()

min_side = max(x1, y1, x2, y2, x3, y3)
while True:
    if min_side * min_side >= s1 + s2 + s3:
        break
    min_side += 1

if (min_side * min_side - s1 - s2 - s3) % 2 != 0:
    print(-1)
    exit()

s1 = (min_side * min_side - s2 - s3) // 2
s2 = min_side * min_side - s1 - s3
s3 = min_side * min_side - s1 - s2

a = b = c = 0
while s1 > 0 and a < x1:
    a += 1
    s1 -= a
while s2 > 0 and b < x2:
    b += 1
    s2 -= b
while s3 > 0 and c < x3:
    c += 1
    s3 -= c
if a * y1 == s1 and b * y2 == s2 and c * y3 == s3:
    print(min_side)
    for i in range(min_side):
        for j in range(min_side):
            if i < y1 and j < a:
                print('A', end='')
            elif i < y2 and j < b:
                print('B', end='')
            else:
                print('C', end='')
        print()
else:
    print(-1)
