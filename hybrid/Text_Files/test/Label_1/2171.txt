#include <iostream>
#include <vector>
using namespace std;
int main() {
  int n, k, d;
  cin >> n >> k >> d;
  if ((n + d - 1) / d > k)
    cout << -1 << endl;
  else {
    for (int i = 0; i < d; i++) {
      for (int j = 0; j < n; j++) {
        cout << (j / d) % k + 1 << ' ';
      }
      cout << endl;
    }
  }
}
