import sys
input = sys.stdin.readline
n = int(input())
a = [[0,0]]
for i in range(n):
    t,x,y = map(int,input().split())
    if t==1:
        a.append([x,y])
    else:
        for j in range(1,len(a)):
            if a[j][0]<a[x][0]<a[j][1] or a[j][0]<a[x][1]<a[j][1]:
                for k in range(1,len(a)):
                    if a[k][0]<a[y][0]<a[k][1] or a[k][0]<a[y][1]<a[k][1]:
                        print("YES")
                        break
                else:
                    print("NO")
                    break
            elif a[j][0]<a[y][0]<a[j][1] or a[j][0]<a[y][1]<a[j][1]:
                print("YES")
                break
        else:
            print("NO")
