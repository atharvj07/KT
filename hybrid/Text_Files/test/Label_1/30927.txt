import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int a = in.nextInt();
        int b = in.nextInt();
        int c = in.nextInt();
        System.out.println(((c % a == 0 || c % b == 0) && c <= (a * 100 + b * 100)) ? "Yes" : "No");
    }
}
