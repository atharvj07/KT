#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pi;
int dx[] = {-1, 0, 1, 0};
int dy[] = {0, -1, 0, 1};
bool seen[105][105];
int r, c;
char grid[105][105];
void dfs(pi p) {
  if (seen[p.first][p.second]) return;
  seen[p.first][p.second] = true;
  for (int i = 0; i < 4; i++) {
    pi np = {p.first + dx[i], p.second + dy[i]};
    if (np.first > 0 && np.first <= r && np.second > 0 && np.second <= c &&
        grid[np.first][np.second] == 'B') {
      dfs(np);
    }
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  cin >> r >> c;
  for (int i = 1; i <= r; i++) {
    for (int j = 1; j <= c; j++) { cin >> grid[i][j]; }
  }
  int ans = 0;
  for (int i = 1; i <= r; i++) {
    for (int j = 1; j <= c; j++) {
      if (!seen[i][j] && grid[i][j] == 'B') {
        dfs({i, j});
        ans++;
      }
    }
  }
  cout << ans << '\n';
}
