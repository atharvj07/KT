import java.util.*;
import java.io.*;

class Main {
    static final long MOD = 1000000007;

    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] c = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            c[i] = in.nextInt();
        }
        int[] b = new int[n];
        for (int i = 0; i < n - 1; i++) {
            b[i] = in.nextInt();
        }
        int q = in.nextInt();
        long[] x = new long[q];
        for (int i = 0; i < q; i++) {
            x[i] = in.nextInt();
        }
        long[] dp = new long[n + 2];
        for (int i = n - 2; i >= 0; i--) {
            long a0 = c[i + 1];
            long a1 = Math.max(0, c[i + 2] - b[i]);
            if (c[i] > Math.min(a0, a1)) {
                dp[i] = 0;
                continue;
            }
            dp[i] = dp[i + 1] + dp[i + 2] + 1;
            if (dp[i] >= MOD) {
                dp[i] -= MOD;
            }
        }
        for (int i = 0; i < q; i++) {
            long ans = 0;
            if (x[i] <= 0) {
                ans = dp[0];
            } else {
                int j = n - 1;
                while (j >= 0 && b[j] > x[i]) {
                    j--;
                }
                ans = dp[j + 1];
            }
            System.out.println(ans);
        }
    }
}
