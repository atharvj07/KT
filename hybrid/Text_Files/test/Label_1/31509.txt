import java.util.*;

public class Main {

    private static final int MAX_VALUE = 100000000;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int i = sc.nextInt() - 1;
        int j = sc.nextInt() - 1;
        int a = sc.nextInt();
        int b = sc.nextInt();

        int[][] dp = new int[n][m];
        for (int[] row : dp) {
            Arrays.fill(row, MAX_VALUE);
        }
        dp[i][j] = 0;

        Queue<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{i, j});

        while (!queue.isEmpty()) {
            int[] current = queue.poll();
            int x = current[0];
            int y = current[1];
            int distance = dp[x][y];

            if (x == 0 || y == 0 || x == n - 1 || y == m - 1) {
                System.out.println(distance);
                return;
            }

            if (x - a >= 0 && y - b >= 0 && dp[x - a][y - b] > distance + 1) {
                dp[x - a][y - b] = distance + 1;
                queue.offer(new int[]{x - a, y - b});
            }

            if (x + a < n && y - b >= 0 && dp[x + a][y - b] > distance + 1) {
                dp[x + a][y - b] = distance + 1;
                queue.offer(new int[]{x + a, y - b});
            }

            if (x - a >= 0 && y + b < m && dp[x - a][y + b] > distance + 1) {
                dp[x - a][y + b] = distance + 1;
                queue.offer(new int[]{x - a, y + b});
            }

            if (x + a < n && y + b < m && dp[x + a][y + b] > distance + 1) {
                dp[x + a][y + b] = distance + 1;
                queue.offer(new int[]{x + a, y + b});
            }
        }

        System.out.println("Poor Inna and pony!");
    }
}
