n, m = map(int, input().split())
s = input()
queries = [list(input().split()) for _ in range(m)]

def calculate_operations(s):
    return s.count('..')

def process_queries(s, queries):
    results = []
    for x, c in queries:
        x = int(x) - 1  # Adjusting index to 0-based
        s = s[:x] + c + s[x+1:]
        results.append(calculate_operations(s))
    return results

results = process_queries(s, queries)
for result in results:
    print(result)
