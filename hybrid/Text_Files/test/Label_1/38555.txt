n = int(input())
a = list(map(int, input().split()))
f = [0] * 1000001
L = -1
R = -1
L1 = -1
R1 = -1
mxfreq = 0
for i in range(n):
    if f[a[i]] == 0:
        L1 = i
        R1 = i
    f[a[i]] += 1
    if f[a[i]] > mxfreq:
        mxfreq = f[a[i]]
        L = L1
        R = R1
    R1 += 1
ans = []
for i in range(1000001):
    if f[i] == mxfreq:
        ans.append(i)
for i in ans:
    for j in range(n):
        if a[j] == i:
            L = j
            break
    for j in range(n - 1, -1, -1):
        if a[j] == i:
            R = j
            break
    break
print(L+1, R+1)
