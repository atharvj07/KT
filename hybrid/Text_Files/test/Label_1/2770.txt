#include <iostream>
#include <vector>
using namespace std;

int main() {
    int a1, a2;
    cin >> a1 >> a2;
    vector<int> v = {a1, a2};
    while (v[0] and v[1]) {
        if (v[0] == 1 or v[1] == 1) {
            if (v[0] == 1 and v[1] == 1) {
                v[0]--;
                v[1]+=v[0];
                v[0] = 0;
            } else if (v[0] == 1) {
                v[0] = 0;
                v[1]++;
            } else {
                v[1]--;
                v[0]+=v[1];
                v[1] = 0;
            }
        } else {
            v[0]--;
            v[1]--;
        }
    }
    cout << v[0] + v[1] << endl;
}
