#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
const ll mod = 1e9+7;
ll powmod(ll base, ll exp) {
    if (exp == 0) return 1;
    ll half = powmod(base, exp >> 1);
    half = half * half % mod;
    if (exp & 1) half = half * base % mod;
    return half;
}
int main() {
    int n, m; cin >> n >> m;
    vector<int> a(n), b(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n; i++) cin >> b[i];
    vector<int> cnt1(m+1,0), cnt2(m+1,0);
    for (int i = 0; i < n; i++) cnt1[a[i]]++,cnt2[b[i]]++;
    ll sum = 0;
    for (int i = 1; i <= m; i++) sum += cnt1[i] < cnt2[i];
    ll den = powmod(m,n) - powmod(m-1,n);
    den = (den + mod) % mod;
    ll mulInv = powmod(den,mod-2);
    cout << sum * mulInv % mod << endl;
    return 0;
}
