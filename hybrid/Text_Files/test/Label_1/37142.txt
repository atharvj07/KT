from collections import defaultdict, deque
n=int(input())
e=[[] for _ in range(n+1)]
for _ in range(n-1):
 a,b,c=map(int,input().split())
 e[a].append((b,c))
 e[b].append((a,c))
vis=[0]*(n+1)
vis[1]=1
q=deque([(1,0)])
while q:
 x,y=q.popleft()
 for a,b in e[x]:
  if vis[a]==0:
   if b==0 and y==1:
    continue
   q.append((a,b))
   vis[a]=1
ans=0
for i in range(1,n+1):
 for a,b in e[i]:
  if b==0:
   ans+=vis[a]
print(ans)
