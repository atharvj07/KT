import java.util.*;
import java.io.*;

class Main
{
	public static void main(String[] args) throws IOException
	{
		BufferedReader br=new BufferedReader(new InputStreamReader(System.in));
		int t=Integer.parseInt(br.readLine());
		while(t-->0){
			StringTokenizer st=new StringTokenizer(br.readLine());
			int n=Integer.parseInt(st.nextToken()),m=Integer.parseInt(st.nextToken());
			char[][] s=new char[n][m];
			char[] fin=new char[m];
			for(int i=0;i<n;i++){
				String k=br.readLine();
				for(int j=0;j<m;j++)s[i][j]=k.charAt(j);
			}
			for(int i=0;i<m;i++){
				int[] freq=new int[26];
				for(int j=0;j<n;j++)freq[s[j][i]-'a']++;
				int max=0,ind=0;
				for(int j=0;j<26;j++){
					if(freq[j]>max){
						max=freq[j];
						ind=j;
					}
				}
				for(int j=0;j<n;j++){
					if(s[j][i]-'a'==ind&&max>1){
						max=-1;
						break;
					}
				}
				fin[i]=(char)(ind+'a');
			}
			int fl=0;
			for(int i=0;i<n;i++){
				int cnt=0;
				for(int j=0;j<m;j++){
					if(fin[j]!=s[i][j])cnt++;
				}
				if(cnt>1){fl=1;break;}
			}
			if(fl==1)System.out.println(-1);
			else{
				for(char c:fin)System.out.print(c);
				System.out.println();
			}
		}
	}
}
