n=int(input())
a=[list(map(int,input().split()))for i in range(n)]
ans="Yes"
for i in range(n):
    for j in range(n):
        if a[i][j]==1:continue
        f=0
        for k in range(n):
            if k==j:continue
            if a[i][k]+a[k][j]==a[i][j]:
                f=1;break
        if f==0:
            ans="No";break
    if ans=="No":break
print(ans)
