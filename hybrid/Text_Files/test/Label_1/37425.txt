from collections import deque
str_form=input()
display=input()
time=deque(display);ch=0
if(int(str_form)==24):
    if(time[0]=='0'):
        time.pop(0)
        ch=ch+1
    if(int(time[0])>2):
        time[0]='0'
        ch=ch+1
    if(int(time[3])>5):
        time[3]='0'
        ch=ch+1
    if(int(time[1])>5):
        time[1]='0'
        ch=ch+1
    if(ch==0):
        time[1]='0'
elif(int(str_form)==12):
    if(time[0]=='0'):
        time.pop(0)
        ch=ch+1
    if(int(time[0])>1):
        time[0]='0'
        ch=ch+1
    if(int(time[3])>5):
        time[3]='0'
        ch=ch+1
    if(int(time[1])>5):
        time[1]='0'
        ch=ch+1
    if(int(time[1])==3 and int(time[0])==1):
        time[1]='2'
        time[0]='0'
        ch=ch+2
    elif(int(time[1])==5 and int(time[0])==1):
        time[1]='0'
        time[0]='0'
        ch=ch+2
    elif(int(time[1])==4 and int(time[0])==1):
        time[0]='1'
        time[1]='0'
        ch=ch+1
    elif(int(time[1])==5 and int(time[0])==0):
        time[1]='0'
        time[0]='1'
        ch=ch+1
time=''.join(time)
print(time)
