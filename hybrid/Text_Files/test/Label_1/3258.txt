#include <bits/stdc++.h>
using namespace std;
const int maxn = 2e5 + 10;
int a[maxn], n;
int main(){
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int ans = a[1] - a[0], cnt = 0;
    for (int i = 0; i < n - 1; i++)
        if (a[i + 1] - a[i] == ans) cnt++;
    cout << ans << " " << cnt << endl;
    return 0;
}
