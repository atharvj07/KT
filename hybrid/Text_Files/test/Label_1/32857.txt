import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int h = sc.nextInt(), m = sc.nextInt(), n = sc.nextInt();
        int[][] t = new int[h][2];
        int ans = 0;
        while (n-- > 0) {
            String op = sc.next();
            int id = sc.nextInt(); int pos = id % h;
            if (op.charAt(0) == '+') {
                int hash = sc.nextInt(); int dummy = 0;
                while (t[pos][0] != 0 && t[pos][0] != id) {
                    pos = (pos + m) % h;
                    dummy++;
                }
                t[pos][0] = id; t[pos][1] = hash;
                ans += dummy;
            } else t[pos][0] = 0;
        }
        System.out.println(ans);
    }
}
