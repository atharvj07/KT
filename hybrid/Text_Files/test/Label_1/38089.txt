n=int(input())
d=[]
for i in range(n):
    x=input()
    if x[0].isalpha():
        d.append(x)
res=[]
for x in d:
    tmp=[y for y in d if (x[0]==y[0] and x[1]==y[1]) or (x[2]==y[2] and x[3]==y[3]) or (x==y)]
    tmp2=set(tmp)
    tmp=''.join(sorted(tmp2))+'='
    res.append(tmp+x[4:])
print(len(res))
for x in res:
    print(x)
