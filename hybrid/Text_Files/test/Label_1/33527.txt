import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] c = new long[n], s = new long[n], f = new long[n];
        for (int i = 0; i < n - 1; i++) {
            c[i] = sc.nextInt();
            s[i] = sc.nextInt();
            f[i] = sc.nextInt();
        }
        long ans = 0;
        for (int i = n - 2; i >= 0; i--) {
            s[i] = Math.max(s[i], ans + c[i]);
            ans = (s[i] + f[i] - 1) / f[i] * f[i];
        }
        for (long a : s) System.out.println(a);
    }
}
