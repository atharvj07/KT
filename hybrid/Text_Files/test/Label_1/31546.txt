import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        while (true) {
            int n = sc.nextInt();
            if (n == 0)
                break;
            Set<String> gems = new HashSet<>();
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                int y = sc.nextInt();
                gems.add(x + "," + y);
            }
            int m = sc.nextInt();
            int x = 10, y = 10;
            Set<String> collected = new HashSet<>();
            for (int i = 0; i < m; i++) {
                char d = sc.next().charAt(0);
                int l = sc.nextInt();
                switch (d) {
                    case 'N':
                        for (int j = 0; j < l; j++) {
                            y++;
                            if (gems.contains(x + "," + y))
                                collected.add(x + "," + y);
                        }
                        break;
                    case 'E':
                        for (int j = 0; j < l; j++) {
                            x++;
                            if (gems.contains(x + "," + y))
                                collected.add(x + "," + y);
                        }
                        break;
                    case 'S':
                        for (int j = 0; j < l; j++) {
                            y--;
                            if (gems.contains(x + "," + y))
                                collected.add(x + "," + y);
                        }
                        break;
                    case 'W':
                        for (int j = 0; j < l; j++) {
                            x--;
                            if (gems.contains(x + "," + y))
                                collected.add(x + "," + y);
                        }
                        break;
                }
            }
            System.out.println(collected.size() == gems.size() ? "Yes" : "No");
        }
    }
}
