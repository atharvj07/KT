import java.util.*;

public class HosesParty {

    static class Hos {
        int weight;
        int beauty;
        List<Integer> friends;

        public Hos(int weight, int beauty) {
            this.weight = weight;
            this.beauty = beauty;
            this.friends = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        int w = scanner.nextInt();

        Hos[] hoses = new Hos[n];
        for (int i = 0; i < n; i++) {
            hoses[i] = new Hos(scanner.nextInt(), scanner.nextInt());
        }

        for (int i = 0; i < m; i++) {
            int x = scanner.nextInt() - 1;
            int y = scanner.nextInt() - 1;
            hoses[x].friends.add(y);
            hoses[y].friends.add(x);
        }

        int[] dp = new int[n];
        Arrays.fill(dp, -1);

        System.out.println(solve(hoses, w, 0, dp));
    }

    static int solve(Hos[] hoses, int remainingWeight, int currentHos, int[] dp) {
        if (currentHos == hoses.length) {
            return 0;
        }

        if (dp[currentHos] != -1) {
            return dp[currentHos];
        }

        int maxBeauty = 0;

        // Option 1: Don't invite current Hos
        maxBeauty = Math.max(maxBeauty, solve(hoses, remainingWeight, currentHos + 1, dp));

        // Option 2: Invite current Hos
        if (remainingWeight >= hoses[currentHos].weight) {
            int friendBeauty = 0;
            for (int friend : hoses[currentHos].friends) {
                friendBeauty += solve(hoses, remainingWeight - hoses[friend].weight, friend + 1, dp);
            }
            maxBeauty = Math.max(maxBeauty, hoses[currentHos].beauty + friendBeauty);
        }

        dp[currentHos] = maxBeauty;
        return maxBeauty;
    }
}
