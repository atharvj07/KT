import java.util.ArrayList;
import java.util.Scanner;
import java.util.Arrays;

class Main {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int n = input.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = input.nextInt();
        }
        System.out.println(calculateMinimumTime(arr, n));
    }

    private static long calculateMinimumTime(int[] arr, int n) {
        long time = 0;
        int[] sortedArr = arr.clone();
        Arrays.sort(sortedArr);
        ArrayList<Integer> reversedArr = new ArrayList<>();
        for (int i = n - 1; i >= 0; i--) {
            reversedArr.add(sortedArr[i]);
        }

        for (int i = 0; i < n; i++) {
            int index = reversedArr.indexOf(arr[i]);
            time += index;
            reversedArr.remove(index);
            reversedArr.add(arr[i]);
        }

        return time;
    }
}
