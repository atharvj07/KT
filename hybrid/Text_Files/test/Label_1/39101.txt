def modpow(base: int, p: int, mod: int) -> int:
    if p == 0:
        return 1
    if p == 1:
        return base
    val = modpow(base, int(p / 2), mod)
    val = (val * val) % mod
    if p % 2 != 0:
        val = (val * base) % mod
    return val % mod
mod = 998244353
n = int(input())
s = input()
cnt = {}
for i in range(n):
    if s[i] not in cnt:
        cnt[s[i]] = 0
    cnt[s[i]] += 1
mx = 0
for key in cnt:
    mx = max(cnt[key], mx)
mn = n
for key in cnt:
    mn = min(cnt[key], mn)
if mn == mx:
    print((mx * (mx + 1)) % mod)
else:
    print((mn * (mn + 1) + mx * (mx - 1)) % mod)
