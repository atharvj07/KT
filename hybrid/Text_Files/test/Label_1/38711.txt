
def swap(arr, i, j):
    temp = arr[i]
    arr[i] = arr[j]
    arr[j] = temp


def heap_down(arr, i, n):
    while (2 * i + 1) < n:
        child = 2 * i + 1
        if child + 1 < n and arr[child + 1] > arr[child]:
            child += 1
        if arr[child] > arr[i]:
            swap(arr, child, i)
            i = child
        else:
            break


def heap_up(arr, i):
    while i > 0:
        parent = (i - 1) // 2
        if arr[parent] < arr[i]:
            swap(arr, parent, i)
            i = parent
        else:
            break


def heap_sort(arr):
    n = len(arr)
    for i in range(n // 2 - 1, -1, -1):
        heap_down(arr, i, n)

    for i in range(n - 1, 0, -1):
        swap(arr, 0, i)
        heap_down(arr, 0, i)


n, k, q = list(map(int, input().split()))
l = []
for x in range(n):
    l.append((int(input()), x + 1))

heap_sort(l)

ans = []
for i in range(q):
    ty, fri = list(map(int, input().split()))
    if ty == 1:
        l[fri - 1][0] = 10 ** 9 + 1
        heap_down(l, fri - 1, n)
    else:
        ans.append('NO' if l[fri - 1][0] != 10 ** 9 + 1 else 'YES')

for x in ans:
    print(x)
