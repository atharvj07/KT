#include <iostream>
#include <vector>

int main() {
  int n, m;
  std::cin >> n >> m;
  std::vector<std::vector<int>> groups(m);
  for (int i = 0; i < m; i++) {
    int k;
    std::cin >> k;
    groups[i].resize(k);
    for (int j = 0; j < k; j++) {
      std::cin >> groups[i][j];
    }
  }
  for (const auto& group : groups) {
    bool hasRick = false, hasMorty = false;
    for (const int& member : group) {
      if (member < 0) {
        hasRick = true;
      } else {
        hasMorty = true;
      }
      if (hasRick && hasMorty) {
        std::cout << "YES" << std::endl;
        return 0;
      }
    }
  }
  std::cout << "NO" << std::endl;
}
