import java.util.Scanner;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int t = sc.nextInt();

    while (t-- > 0) {
      long a = sc.nextLong();
      long b = sc.nextLong();
      long n = sc.nextLong();

      long xor = 0;
      for (int i = 31; i >= 0; i--) {
        if ( ( n & (1<<i) ) != 0 ) {
          long aux = ( (xor ^ a) ^ b );
          b = a;
          a = aux;
        }
      }

      System.out.println( xor );
    }
  }
}
