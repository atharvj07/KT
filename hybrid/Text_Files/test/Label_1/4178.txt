#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll n,p[300009],q[300009],dp[3][300009];
int main(){
    cin>>n;
    for(ll i=1;i<=n;++i)
    {
        cin>>p[i];
        p[n+i]=p[i];
    }
    for(ll i=1;i<=n;++i)
    {
        cin>>q[i];
        q[n+i]=q[i];
    }
    for(ll i=n*2;i>1;--i)
    {
        dp[0][i]=max(p[i]+dp[0][i+1],q[i]+dp[1][i+1]);
        dp[1][i]=max(p[i]+dp[0][i+1],q[i]+dp[2][i+1]);
        dp[2][i]=max(p[i]+dp[1][i+1],q[i]+dp[0][i+1]);
    }
    cout<<dp[0][2]<<endl;
    return 0;
}
