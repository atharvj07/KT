#include<iostream>
#include<vector>
#include<queue>
using namespace std;
#define MAXN 123457
vector<int>g[MAXN];
int n,m,vis[MAXN],d[MAXN];
int bfs(int s)
{
    queue<int>q;
    q.push(s);
    vis[s]=1;
    int mx=1,mxd=0,cnt=0;
    while(!q.empty())
    {
        int u=q.front();
        q.pop();
        cnt++;
        if(d[u]>mxd)
        {
            mxd=d[u];
            mx=u;
        }
        for(int i=0; i<g[u].size(); i++)
        {
            int v=g[u][i];
            if(vis[v]==0)
            {
                vis[v]=1;
                d[v]=d[u]+1;
                q.push(v);
            }
        }
    }
    return mx;
}
int main()
{
    cin>>n>>m;
    for(int i=1; i<n; i++)
    {
        int a,b;
        cin>>a>>b;
        g[a].push_back(b);
        g[b].push_back(a);
    }
    int x=bfs(1);
    int y=bfs(x);
    cout<<x<<endl<<d[y]<<endl;
}
