import java.io.*;
import java.util.*;

public class Main {
  static final int INF = Integer.MAX_VALUE;

  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[][] c = new int[n][2];
    int[] a = new int[100001];
    int[] b = new int[100001];
    for (int i = 0; i < n; i++) {
      c[i][0] = sc.nextInt();
      c[i][1] = sc.nextInt();
      a[c[i][0]]++;
      b[c[i][1]]++;
    }
    int ans = INF;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < 2; j++) {
        ans = Math.min(ans, n - a[c[i][j]] - b[c[i][1 - j]] + 1);
      }
    }
    System.out.println(ans == INF ? -1 : ans);
  }
}
