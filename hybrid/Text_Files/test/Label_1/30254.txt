import java.util.*;

public class JohnnyBlogs {

    static class Blog {
        int id;
        int topic;
        List<Blog> neighbors;

        Blog(int id) {
            this.id = id;
            this.neighbors = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        Blog[] blogs = new Blog[n + 1];
        for (int i = 1; i <= n; i++) {
            blogs[i] = new Blog(i);
        }

        for (int i = 0; i < m; i++) {
            int a = scanner.nextInt();
            int b = scanner.nextInt();
            blogs[a].neighbors.add(blogs[b]);
            blogs[b].neighbors.add(blogs[a]);
        }

        int[] desiredTopics = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            desiredTopics[i] = scanner.nextInt();
        }

        List<Integer> order = new ArrayList<>();
        boolean[] visited = new boolean[n + 1];

        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                if (!dfs(blogs[i], desiredTopics, visited, order)) {
                    System.out.println(-1);
                    return;
                }
            }
        }

        for (int blogId : order) {
            System.out.print(blogId + " ");
        }
    }

    static boolean dfs(Blog blog, int[] desiredTopics, boolean[] visited, List<Integer> order) {
        visited[blog.id] = true;
        Set<Integer> usedTopics = new HashSet<>();
        for (Blog neighbor : blog.neighbors) {
            if (visited[neighbor.id]) {
                usedTopics.add(neighbor.topic);
            }
        }

        for (int i = 1; i <= desiredTopics.length; i++) {
            if (!usedTopics.contains(i)) {
                blog.topic = i;
                order.add(blog.id);
                for (Blog neighbor : blog.neighbors) {
                    if (!visited[neighbor.id]) {
                        if (!dfs(neighbor, desiredTopics, visited, order)) {
                            return false;
                        }
                    }
                }
                return true;
            }
        }
        return false;
    }
}
