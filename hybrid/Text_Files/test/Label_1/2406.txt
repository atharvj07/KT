#include <iostream>
#include <vector>
using namespace std;

int main() {
    int t;
    cin >> t;
    while (t--) {
        string a;
        cin >> a;
        int n = a.size();
        vector<int> b(n, -1);
        int open = 0;
        bool ok = true;
        for (int i = 0; i < n; i++) {
            if (a[i] == 'A') {
                if (open % 2 == 0) {
                    if (b[i] == -1)
                        b[i] = '(';
                    else if (b[i] == ')')
                        ok = false;
                } else {
                    if (b[i] == -1)
                        b[i] = ')';
                    else if (b[i] == '(')
                        ok = false;
                }
                open++;
            } else if (a[i] == 'B') {
                if (open % 2 == 0) {
                    if (b[i] == -1)
                        b[i] = ')';
                    else if (b[i] == '(')
                        ok = false;
                } else {
                    if (b[i] == -1)
                        b[i] = '(';
                    else if (b[i] == ')')
                        ok = false;
                }
                open++;
            } else {
                if (b[i] == -1)
                    b[i] = '(';
            }
        }
        if (open % 2 == 1)
            ok = false;
        cout << (ok ? "YES" : "NO") << endl;
    }
}
