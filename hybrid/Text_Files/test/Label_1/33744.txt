import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int m = sc.nextInt();
    int[] l = new int[m];
    for (int i = 0; i < m; i++) {
      l[i] = sc.nextInt();
    }
    int[] p = new int[m];
    int[] c = new int[n + 1];
    int j = 0;
    for (int i = 0; i < m; i++) {
      if (j + l[i] - 1 <= n) {
        p[i] = j + 1;
        for (int k = 0; k < l[i]; k++) {
          c[j + k] = i + 1;
        }
        j += l[i];
      } else {
        System.out.println(-1);
        return;
      }
    }
    for (int i = 0; i < n; i++) {
      if (c[i] == 0) {
        System.out.println(-1);
        return;
      }
    }
    for (int i = 0; i < m; i++) {
      System.out.print(p[i] + " ");
    }
    System.out.println();
  }
}
