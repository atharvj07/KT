import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) throws IOException {
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(reader.readLine());
        int[] piles = Arrays.stream(reader.readLine().split(" "))
                .mapToInt(Integer::parseInt)
                .toArray();
        boolean[] visited = new boolean[1000001];
        boolean win = dfs(piles, n, visited);
        System.out.println(win ? "sjfnb" : "cslnb");
    }

    private static boolean dfs(int[] piles, int n, boolean[] visited) {
        if (n == 1) {
            return false;
        }
        for (int i = 0; i < n; i++) {
            int pile = piles[i];
            if (pile == 0) {
                continue;
            }
            visited[pile] = true;
            piles[i] -= 1;
            for (int j = 1; j <= pile; j++) {
                if (visited[j]) {
                    continue;
                }
                visited[j] = true;
                if (!dfs(piles, n, visited)) {
                    return true;
                }
                visited[j] = false;
            }
            visited[pile] = false;
            piles[i] += 1;
        }
        return false;
    }
}
