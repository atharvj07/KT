def solve(n, d, a):
    for _ in range(d):
        pos = -1
        for i in range(n - 1, 0, -1):
            if a[i] > 0 and a[i - 1] < 100:
                pos = i - 1
                break
        if pos == -1:
            break
        a[pos] += 1
        a[pos + 1] -= 1
    return max(a)

t = int(input())
for _ in range(t):
    n, d = map(int, input().split())
    a = list(map(int, input().split()))
    print(solve(n, d, a))
