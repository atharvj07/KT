#include <bits/stdc++.h>
using namespace std;
int main() {
    int n, k, ans = 0;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> cnt(101, 0);
    int t = 0;
    for (int i = 0; i < n; i++) {
        t += a[i];
        for (int j = 1; j <= a[i]; j++)
            cnt[t + j]++;
        ans += (cnt[t + 1] == 1 and t + 1 <= 100);
    }
    cout << ans;
}
