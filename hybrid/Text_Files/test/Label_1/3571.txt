#include <cstdio>
#include <vector>
using namespace std;
#define rep(i, n) for(int i = 0; i < (int) (n); ++i)
typedef long long ll;

int main()
{
    int n; scanf("%d", &n);
    vector <int> a(n);
    rep(i, n) scanf("%d", &a[i]);
    ll sum = 0;
    vector <int> l, r;
    bool has = false;
    rep(i, n) {
        sum += a[i];
        if(sum == 0) {
            if(!has) { l.push_back(i + 1); has = true; }
            r.push_back(i + 1);
        }
    }
    if(sum == 0) printf("NO\n");
    else {
        printf("YES\n");
        printf("%zd\n", l.size());
        rep(i, l.size()) printf("%d %d\n", l[i], r[i]);
    }
    return 0;
}
