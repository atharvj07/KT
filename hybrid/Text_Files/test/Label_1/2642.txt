#include <iostream>
#include <vector>
using namespace std;

int n, m;
string s;
vector<int> pre;

int main() {
  cin >> s;
  n = s.size();
  cin >> m;
  pre.resize(n + 1);
  for (int i = 1; i < n; ++i)
    pre[i] = s[i] == '#' ? pre[i - 1] : pre[i - 1] + 1;
  for (int i = 0; i < m; ++i) {
    int l, r;
    cin >> l >> r;
    cout << pre[r - 1] - pre[l - 1] << '\n';
  }
  return 0;
}
