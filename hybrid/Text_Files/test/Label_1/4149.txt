#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll N = 1e9 + 5, M = 1e5 + 5, mod = 1e9 + 7;

int32_t main() {
  ll n, m, x, y;
  cin >> n >> m;
  if(n == m) return cout << "-1\n", 0;
  vector<vector<ll>> g(n + 1, vector<ll>(n + 1, 1e18));
  while(m--) {
    cin >> x >> y;
    g[x][y] = 0;
  }
  g[1][1] = 0;
  for(ll i = 1; i <= n; ++i) {
    for(ll j = 1; j <= n; ++j) {
      if(g[i][j] != 0) 
        g[i][j] = min({g[i][j], g[i - 1][j], g[i][j - 1] + 1});
    }
  }
  cout << (g[n][n] == 1e18 ? -1 : g[n][n]) << "\n";
}
