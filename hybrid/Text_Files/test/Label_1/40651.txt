import sys

n=int(input())
a=list(map(int,sys.stdin.readline().split()))
if n==1:
    print("YES")
    print("1 1")
    sys.exit()
for i in range(1,n):
    if a[i]!=a[i-1]+1 and a[i]!=a[i-1]-1:
        print("NO")
        sys.exit()
if a[0]==1 or a[-1]==n:
    print("YES")
    print(n,1)
else:
    print("NO")
