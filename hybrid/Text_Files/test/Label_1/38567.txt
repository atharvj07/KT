def solve(n):
    for i in range(2, int(n**0.5)+1):
        if n % i == 0:
            for j in range(i+1, int((n//i)**0.5)+1):
                if (n//i) % j == 0:
                    k = n//(i*j)
                    if k > j:
                        print("YES")
                        print(i, j, k)
                        return
    print("NO")

t = int(input())
for _ in range(t):
    n = int(input())
    solve(n)
