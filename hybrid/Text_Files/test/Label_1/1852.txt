#include <cstdio>
#include <algorithm>
int main() {
  int r, c;
  scanf("%d %d ", &r, &c);
  bool x[10], y[10];
  for (int i = 0; i < r; ++i)
    for (int j = 0; j < c; ++j) {
      char p;
      scanf("%c ", &p);
      x[i] |= p == 'S';
      y[j] |= p == 'S';
    }
  int ans = 0;
  for (int i = 0; i < r; ++i) ans += !x[i];
  for (int i = 0; i < c; ++i) ans += !y[i];
  printf("%d\n", ans);
  return 0;
}
