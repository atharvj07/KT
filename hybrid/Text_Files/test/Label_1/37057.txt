n = int(input())
cards = list(map(int, input().split()))

# Sort the cards with their original indices
sorted_cards = sorted([(card, i+1) for i, card in enumerate(cards)])

# Distribute cards by pairing the smallest with the largest, second smallest with second largest, and so on
for i in range(n//2):
    print(sorted_cards[i][1], sorted_cards[n-i-1][1])
