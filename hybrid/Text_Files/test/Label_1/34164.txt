import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int q = scanner.nextInt();
        while (q-- > 0) {
            int n = scanner.nextInt();
            int[] t = new int[n];
            for (int i = 0; i < n; ++i) {
                t[i] = scanner.nextInt();
            }
            Arrays.sort(t);
            int ans = 0;
            for (int i = 0; i < n; ++i) {
                ans += Math.abs(t[i] - (i + 1));
            }
            System.out.println(ans);
        }
    }
}
