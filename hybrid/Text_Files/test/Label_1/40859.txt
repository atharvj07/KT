import sys
import math

def solve():
    n = int(input())
    a = [list(map(int, input().split())) for _ in range(n)]

    s = [[0 for _ in range(5)] for _ in range(n)]

    for i in range(n):
        for j in range(5):
            if a[i][j] == -1:
                continue
            s[i][j] = 120 - a[i][j]

    s_max = [[0 for _ in range(5)] for _ in range(n)]
    for i in range(n):
        for j in range(5):
            s_max[i][j] = math.ceil((120 - s[i][j]) / 250)

    s_sum = [0 for _ in range(n)]
    for i in range(n):
        for j in range(5):
            s_sum[i] += s_max[i][j]

    if s_sum[1] > s_sum[0]:
        return -1

    acc = [[0 for _ in range(5)] for _ in range(n)]
    for i in range(n):
        for j in range(5):
            acc[i][j] = acc[i][max(0, j - 1)] + s_max[i][j]

    mn = 1000000000
    for bit in range(1 << 5):
        cur = 0
        for j in range(5):
            if bit & (1 << j) > 0:
                cur += acc[1][j]
        mn = min(mn, bitset_to_act(bit, 1, n, s_max, acc, cur))

    return mn

def bitset_to_act(bit, st, en, s_max, acc, cur):
    if st == en:
        return cur
    mn = 1000000000
    for j in range(5):
        if (bit & (1 << j)) == 0:
            mn = min(mn, bitset_to_act(bit | (1 << j), st + 1, en, s_max, acc, cur + acc[st][j]))
    return mn

print(solve())
