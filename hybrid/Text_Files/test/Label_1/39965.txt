for _ in range(int(input())):
    s = input()
    b = [0 for i in range(11)]
    for i in s:
        b[int(i)]+=1
    if max(b)<2:
        print("NO")
        continue
    p = 0
    for i in range(1,11):
        if b[i]>1:
            p = i
    if p==0:
        print("NO")
        continue
    print("YES")
