#include<bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 1e9+7;
vector<long long> adj[100001];
bool vis[100001];
ll ans[100001],dp[100001],la;
void dfs(ll s,ll p) {
	vis[s]=1;
	for(ll u:adj[s]) {
		if(vis[u]==0) {
			dfs(u,s);
			dp[s]^=dp[u];
			ans[s]=la+1;
			la=ans[s];
		}
	}
}
int main() {
	ll n,a,b,i;
	cin>>n;
	for(i=1;i<n;++i) {
		cin>>a>>b;
		adj[a].push_back(b);
		adj[b].push_back(a);
	}
	dfs(1,1);
	sort(ans+1,ans+n+1);
	cout<<1<<" "<<ans[n]<<"\n";
	return 0;
}
