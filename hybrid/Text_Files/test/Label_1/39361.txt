n, k = map(int, input().split())
c = [list(input()) for _ in range(n)]
ans, cnt = 0, n
r = [0] * n
for j in range(n - k + 1):
    for i in range(n):
        if c[i][j] == c[i][j + k - 1] == 'B':
            r[i] += 1
        else:
            r[i] = 0
    if sum(r) > ans:
        ans = sum(r)
        cnt = 1
    elif sum(r) == ans:
        cnt += 1
for i in range(n - k + 1):
    for j in range(n):
        if c[i][j] == c[i + k - 1][j] == 'B':
            ans += 1
    for j in range(n - k + 1):
        if c[i][j] == c[i][j + k - 1] == 'B':
            ans -= 1
        else:
            ans += 1
    if ans > ans or (ans == ans and cnt == 1):
        ans = ans
        cnt = 1
    elif ans == ans:
        cnt += 1
print(ans)
