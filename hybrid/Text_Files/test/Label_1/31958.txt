import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

public class PrincessTwilight {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int[] ai = new int[n];
        for (int i = 0; i < n; i++) {
            ai[i] = scanner.nextInt();
        }

        // Find the prime factorization of each ai
        List<List<Integer>> primeFactors = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            primeFactors.add(primeFactorize(ai[i]));
        }

        // Find the minimum value of bi for each ai
        int[] bi = new int[n];
        for (int i = 0; i < n; i++) {
            bi[i] = findMinimumBi(ai[i], primeFactors);
        }

        // Print the key
        System.out.println(Arrays.toString(bi).replace("[", "").replace("]", ""));
    }

    // Find the prime factorization of a number
    private static List<Integer> primeFactorize(int n) {
        List<Integer> factors = new ArrayList<>();
        for (int i = 2; i <= n; i++) {
            while (n % i == 0) {
                factors.add(i);
                n /= i;
            }
        }
        return factors;
    }

    // Find the minimum value of bi for a given ai and its prime factorization
    private static int findMinimumBi(int ai, List<List<Integer>> primeFactors) {
        int minBi = ai;
        for (List<Integer> factors : primeFactors) {
            int gcd = 1;
            for (int factor : factors) {
                gcd = lcm(gcd, factor);
            }
            minBi = Math.min(minBi, ai + gcd - 1);
        }
        return minBi;
    }

    // Find the least common multiple of two numbers
    private static int lcm(int a, int b) {
        return a * b / gcd(a, b);
    }

    // Find the greatest common divisor of two numbers
    private static int gcd(int a, int b) {
        if (b == 0) {
            return a;
        }
        return gcd(b, a % b);
    }
}
