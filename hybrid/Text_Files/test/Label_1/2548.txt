#include<bits/stdc++.h>
using namespace std;
int n;
string s, t;
int a[550001], b[550001], c[550001];
int main(){
	cin>>s>>t;
	int ls=s.size(), lt=t.size();
	for(int i=0; i<ls; i++) a[i+1]=s[i]-'0';
	for(int i=0; i<lt; i++) b[i+1]=t[i]-'0';
	int p=0;
	for(int i=1; i<=ls; i++){
		if(a[i]==b[p+1]) p++;
		if(p==lt){
			c[i-lt+1]=1;
			p--;
		}
	}
	for(int i=1; i<=ls; i++) s[i-1]='0'+a[i]-c[i];
	cout<<s<<endl;
}
