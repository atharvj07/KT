import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Set;

class Main {
  private static final long MOD = 1000000007L;

  public static void main(String[] args) throws IOException {
    BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    String[] nm = br.readLine().split(" ");
    int n = Integer.parseInt(nm[0]);
    int m = Integer.parseInt(nm[1]);
    int[] robots = new int[n];
    int[] exits = new int[m];
    String[] robotStrs = br.readLine().split(" ");
    for (int i = 0; i < n; i++) {
      robots[i] = Integer.parseInt(robotStrs[i]);
    }
    String[] exitStrs = br.readLine().split(" ");
    for (int i = 0; i < m; i++) {
      exits[i] = Integer.parseInt(exitStrs[i]);
    }
    Arrays.sort(robots);
    Arrays.sort(exits);
    Set<Integer> usedExits = new HashSet<>();
    long combinations = 1L;
    for (int robot : robots) {
      int idx = Arrays.binarySearch(exits, robot);
      if (idx >= 0) {
        combinations = (combinations * 2) % MOD;
        usedExits.add(idx);
      } else {
        int insPoint = -idx - 1;
        if (insPoint > 0 && insPoint - 1 < usedExits.size()) {
          combinations = (combinations * 2) % MOD;
          usedExits.add(insPoint - 1);
        } else if (insPoint < m && insPoint < usedExits.size()) {
          combinations = (combinations * 2) % MOD;
          usedExits.add(insPoint);
        }
      }
    }
    System.out.println(combinations);
  }
}
