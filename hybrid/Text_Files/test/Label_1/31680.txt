import java.util.*;

public class Main {
  public static void main(String[] args) {
    Scanner s = new Scanner(System.in);
    int r = s.nextInt();
    int n = s.nextInt();
    boolean[][] taken = new boolean[r + 1][3];
    for (int i = 0; i < n; i++) {
      int ri = s.nextInt();
      int ci = s.nextInt();
      taken[ri][ci] = true;
      taken[ri][3 - ci] = true;
    }
    boolean win = true;
    for (int i = 1; i <= r; i++) {
      if (!taken[i][1] && (!taken[i - 1][2] || !taken[i][2])) {
        win = false;
      }
      if (!taken[i][2] && (!taken[i - 1][1] || !taken[i][1])) {
        win = false;
      }
    }
    System.out.println(win ? "WIN" : "LOSE");
  }
}
