#include <iostream>
#include <string>

int main() {
  std::string s, t, u;
  std::cin >> s;
  while (s.size() || t.size()) {
    if (s.empty() || (t.size() && s[0] > t[t.size() - 1])) {
      u += t.back();
      t.pop_back();
    } else {
      t += s[0];
      s.erase(s.begin());
    }
  }
  std::cout << u << "\n";
  return 0;
}
