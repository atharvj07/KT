import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] a = new long[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextLong();
        }
        int q = sc.nextInt();
        long[] b = new long[n];
        long[] c = new long[n];
        long min = a[0] / 2;
        b[0] = min;
        c[0] = a[0] - min;
        for (int i = 1; i < n; i++) {
            long x = a[i - 1] / 2;
            long y = a[i] / 2;
            if (x >= y) {
                b[i] = x;
                c[i] = a[i] - x;
            } else {
                b[i] = y;
                c[i] = a[i] - y;
            }
            long v = Math.max(b[i], c[i]);
            if (v < min) {
                min = v;
            }
        }
        System.out.println(min);
        for (int i = 0; i < q; i++) {
            int l = sc.nextInt() - 1;
            int r = sc.nextInt() - 1;
            long x = sc.nextLong();
            for (int j = l; j <= r; j++) {
                a[j] += x;
            }
            b[l] = a[l] / 2;
            c[l] = a[l] - b[l];
            for (int j = l + 1; j <= r; j++) {
                b[j] += x;
                long y = a[j] / 2;
                if (b[j] >= y) {
                    c[j] = a[j] - b[j];
                } else {
                    b[j] = y;
                    c[j] = a[j] - y;
                }
            }
            min = a[0] / 2;
            int fl = 0;
            for (int j = 1; j < n; j++) {
                if (b[j] < b[j - 1]) {
                    fl = 1;
                    break;
                }
                if (c[j] > c[j - 1]) {
                    fl = 1;
                    break;
                }
                long v = Math.max(b[j], c[j]);
                if (v < min) {
                    min = v;
                }
            }
            if (fl == 1) {
                b = new long[n];
                c = new long[n];
                min = a[0] / 2;
                b[0] = min;
                c[0] = a[0] - min;
                for (int j = 1; j < n; j++) {
                    long x2 = a[j - 1] / 2;
                    long y2 = a[j] / 2;
                    if (x2 >= y2) {
                        b[j] = x2;
                        c[j] = a[j] - x2;
                    } else {
                        b[j] = y2;
                        c[j] = a[j] - y2;
                    }
                    long v = Math.max(b[j], c[j]);
                    if (v < min) {
                        min = v;
                    }
                }
            }
            System.out.println(min);
        }
    }
}
