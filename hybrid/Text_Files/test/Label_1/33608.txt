import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Scanner;
import java.util.Set;

class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        List<Edge> edges = new ArrayList<>();
        Map<Integer, Set<Integer>> graph = new HashMap<>();
        for (int i = 0; i < m; i++) {
            int w = scanner.nextInt();
            int inMst = scanner.nextInt();
            edges.add(new Edge(w, inMst == 1));
        }

        // Sort the edges in ascending order of weight
        edges.sort((a, b) -> a.weight - b.weight);

        // Create the graph
        for (int i = 1; i <= n; i++) {
            graph.put(i, new HashSet<>());
        }

        // Add the edges to the graph
        int mstEdges = 0;
        for (Edge edge : edges) {
            int u = edge.u;
            int v = edge.v;
            if (u != v && !graph.get(u).contains(v) && !graph.get(v).contains(u)) {
                graph.get(u).add(v);
                graph.get(v).add(u);
                if (edge.inMst) {
                    mstEdges++;
                }
            }
        }

        // Check if the graph is connected
        if (!isConnected(graph, n)) {
            System.out.println(-1);
            return;
        }

        // Check if the edges in the MST form a MST
        if (mstEdges != n - 1) {
            System.out.println(-1);
            return;
        }

        // Print the graph
        for (Edge edge : edges) {
            System.out.println(edge.u + " " + edge.v);
        }
    }

    private static boolean isConnected(Map<Integer, Set<Integer>> graph, int n) {
        Set<Integer> visited = new HashSet<>();
        dfs(1, graph, visited);
        return visited.size() == n;
    }

    private static void dfs(int u, Map<Integer, Set<Integer>> graph, Set<Integer> visited) {
        if (visited.contains(u)) {
            return;
        }
        visited.add(u);
        for (int v : graph.get(u)) {
            dfs(v, graph, visited);
        }
    }

    private static class Edge {

        private int weight;
        private int u;
        private int v;
        private boolean inMst;

        public Edge(int weight, boolean inMst) {
            this.weight = weight;
            this.inMst = inMst;
            this.u = -1;
            this.v = -1;
        }

        public void setU(int u) {
            this.u = u;
        }

        public void setV(int v) {
            this.v = v;
        }
    }
}
