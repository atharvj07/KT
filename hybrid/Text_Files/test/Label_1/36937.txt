mod = int(1e9)+7
n, m = map(int, input().split())
a = list(map(int, input().split()))
b = list(map(int, input().split()))
not_def = 0
for i in a:
    if i==0: not_def += 1
    else: break
for i in b:
    if i==0: not_def -= 1
    else: break
x = ((not_def+1)*(m-1)%mod)*pow(m, n-1, mod)
y = (pow(m, n)-1)%mod
print((x*pow(y, mod-2, mod))%mod)
