#include <bits/stdc++.h>
#define endl '\n'
#define int long long
using namespace std;
int n, m, ans, dp[200005], pre[200005], sum[200005];
string s;
int32_t main(){
    cin >> m;
    for(int i=0;i<m;i++){
        cin >> s;
        n = s.size();
        dp[n] = 1;
        for(int i=n-1;i>=0;i--){
            int len = 0;
            int cur = 0;
            for(int j=i;j<n;j++){
                int dig = s[j] - '0';
                if(dig == 0 && len == 0) continue;
                cur = cur * 2 + dig;
                len++;
                if(len > cur) break;
                dp[i] += dp[j+1];
            }
        }
        cout << dp[0] << endl;
        memset(dp, 0, sizeof(dp));
    }
    return 0;
}
