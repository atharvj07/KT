import java.util.*;

public class Dominoes {

    private static final int MAX_N = 7;
    private static final int MAX_M = (MAX_N * (MAX_N - 1)) / 2;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[][] graph = new int[n][n];
        for (int i = 0; i < m; i++) {
            int u = scanner.nextInt() - 1;
            int v = scanner.nextInt() - 1;
            graph[u][v] = graph[v][u] = 1;
        }

        int[][] degree = new int[n][7];
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (graph[i][j] == 1) {
                    degree[i][0]++;
                    degree[j][0]++;
                }
            }
        }

        int[][] dp = new int[n][7];
        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], -1);
        }

        int result = solve(0, 0, graph, degree, dp);
        System.out.println(result);
    }

    private static int solve(int vertex, int remaining, int[][] graph, int[][] degree, int[][] dp) {
        if (vertex == graph.length) {
            return remaining == 0 ? 0 : -1;
        }

        if (dp[vertex][remaining] != -1) {
            return dp[vertex][remaining];
        }

        int result = -1;
        for (int i = 1; i <= 6; i++) {
            if (degree[vertex][i - 1] > 0 && remaining >= i) {
                int nextVertex = -1;
                for (int j = 0; j < graph.length; j++) {
                    if (graph[vertex][j] == 1 && degree[j][i - 1] > 0) {
                        nextVertex = j;
                        break;
                    }
                }

                if (nextVertex != -1) {
                    degree[vertex][i - 1]--;
                    degree[nextVertex][i - 1]--;
                    int nextRemaining = remaining - i;
                    int nextResult = solve(nextVertex, nextRemaining, graph, degree, dp);
                    if (nextResult != -1) {
                        result = Math.max(result, 1 + nextResult);
                    }
                    degree[vertex][i - 1]++;
                    degree[nextVertex][i - 1]++;
                }
            }
        }

        dp[vertex][remaining] = result;
        return result;
    }
}
