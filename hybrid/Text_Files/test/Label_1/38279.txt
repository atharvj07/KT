from collections import Counter
n, m = map(int,input().split())
grid = [input() for i in range(n)]
ans = 0
for i in range(n - 2):
    for j in range(m):
        top = Counter()
        mid = Counter()
        bot = Counter()
        for k in range(i + 3):
            top[grid[k][j]] += 1
        for k in range(i + 1, i + 3):
            mid[grid[k][j]] += 1
        for k in range(i + 3, n):
            bot[grid[k][j]] += 1
        if len(top.keys()) == 1 and len(mid.keys()) == 1 and len(bot.keys()) == 1:
            ans += 1
        elif len(top.keys()) == 2 and len(mid.keys()) == 1 and len(bot.keys()) == 1 or len(top.keys()) == 1 and len(mid.keys()) == 2 and len(bot.keys()) == 1 or len(top.keys()) == 1 and len(mid.keys()) == 1 and len(bot.keys()) == 2:
            ans += 3
        elif len(top.keys()) == 3 and len(mid.keys()) == 1 and len(bot.keys()) == 1 or len(top.keys()) == 1 and len(mid.keys()) == 3 and len(bot.keys()) == 1 or len(top.keys()) == 1 and len(mid.keys()) == 1 and len(bot.keys()) == 3:
            ans += 6
        elif len(top.keys()) == 2 and len(mid.keys()) == 2 and len(bot.keys()) == 1 or len(top.keys()) == 1 and len(mid.keys()) == 2 and len(bot.keys()) == 2 or len(top.keys()) == 2 and len(mid.keys()) == 1 and len(bot.keys()) == 2:
            ans += 12
print(ans)
