import sys

n, x, pos = map(int, sys.stdin.readline().split())
mod = int(1e9) + 7

def fexp(x, y):
    ret = 1
    while y > 0:
        if y % 2 == 0:
            x = (x * x) % mod
            y //= 2
        else:
            ret = (ret * x) % mod
            y -= 1
    return ret

def inv(x):
    return fexp(x, mod - 2)

# nPr = n! / (n - r)!
def fact(n, r):
    return (n * inv(n - r)) % mod

print(fact(n - 1, pos - 1) * fact(n - 1 - pos, n - x - pos) % mod)
