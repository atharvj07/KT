def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def longest_good_sequence(n, good_integers):
    dp = [1] * (max(good_integers) + 1)
    for i in range(n):
        for j in range(2, int(good_integers[i] ** 0.5) + 1):
            if good_integers[i] % j == 0:
                dp[good_integers[i]] = max(dp[good_integers[i]], dp[j] + 1, dp[good_integers[i] // j] + 1)
                while good_integers[i] % j == 0:
                    good_integers[i] //= j
        if good_integers[i] > 1:
            dp[good_integers[i]] = max(dp[good_integers[i]], dp[good_integers[i]] + 1)
    return max(dp)

n = int(input().strip())
good_integers = list(map(int, input().strip().split()))
print(longest_good_sequence(n, good_integers))
