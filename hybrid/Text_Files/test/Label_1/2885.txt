#include <iostream>
#include <string>

int main() {
    int n;
    std::string s;
    std::cin >> n >> s;
    std::string filler = "ogo";
    for (int i = 0; i < n; i++) {
        std::string sub = s.substr(i, filler.size());
        if (sub == filler) {
            i += filler.size() - 1;
            std::cout << "***";
        } else {
            std::cout << s[i];
        }
    }
    std::cout << std::endl;
    return 0;
}
