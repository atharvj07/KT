#include <iostream>
#include <vector>
using namespace std;
#define ll long long

int main() {
    int n;
    cin >> n;
    ll rate = 1899, ans = 1899;
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        rate += a;
        if (rate < 1900) {
            ans = -1;
            break;
        }
        if (b == 1) { rate = 1899; }
        if (rate > ans) { ans = rate; }
    }
    if (ans == -1)
        cout << "Impossible";
    else if (ans >= 1e18)
        cout << "Infinity";
    else
        cout << ans;
    return 0;
}
