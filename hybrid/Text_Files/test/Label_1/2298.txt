#include<bits/stdc++.h>
using namespace std;

const int MAX_N=200005;
int n, e, a[MAX_N], b[MAX_N], c[MAX_N], p[MAX_N];
inline int root(int x){return p[x]<0?x:p[x]=root(p[x]);}
inline void unite(int x, int y){
	if((x=root(x))>(y=root(y)))p[x]=y;
	else{p[y]=x; if(p[x]<0)p[x]--;}
}
struct edge{
	int x, y, w;
	edge(int _x=0, int _y=0, int _w=0): x(_x), y(_y), w(_w){}
	bool operator<(const edge &r)const{return w>r.w;}
}elist[MAX_N];

int32_t main(){
	int i, x, y, ans=0;
	cin>>n>>e;
	for(i=1; i<=n; i++)cin>>a[i];
	for(i=0; i<e; i++){
		cin>>x>>y>>b[i];
		elist[i]={x, y, b[i]};
	}
	sort(elist, elist+e);
	fill(p, p+n+e, -1);
	for(i=0; i<e; i++){
		if(x=root(elist[i].x))if(y=root(elist[i].y))if(x!=y){
			unite(x, y);
			c[x]+=c[y]; ans+=c[y]; c[x]+=elist[i].w;
		}
		c[elist[i].x]++;
	}
	for(i=1; i<=n; i++)ans=max(ans, c[i]-a[i]);
	cout<<ans<<endl;
}
