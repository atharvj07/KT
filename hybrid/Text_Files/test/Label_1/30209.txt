import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] v = new int[n];
        int[] w = new int[n];
        for (int i = 0; i < n; i++) v[i] = in.nextInt();
        for (int i = 0; i < n; i++) w[i] = in.nextInt();
        int[][] dp = new int[n][n];
        for (int i = 0; i < n; i++) dp[i][i] = v[i];
        for (int i = n - 2; i >= 0; i--) {
            for (int j = i + 1; j < n; j++) {
                if (Math.abs(w[j] - w[i]) == 1 && 2 * w[j] - w[i] - w[i + 1] >= 0) {
                    dp[i][j] = Math.max(dp[i][j], v[j] + dp[i][j - 1]);
                    for (int k = i + 1; k < j; k++) {
                        if (Math.abs(w[k] - w[j]) == 1 && 2 * w[k] - w[j] - w[k - 1] >= 0) {
                            dp[i][j] = Math.max(dp[i][j], v[j] + dp[i][k - 1] + dp[k][j - 1]);
                        }
                    }
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) ans = Math.max(ans, dp[i][n - 1]);
        System.out.println(ans);
    }
}
