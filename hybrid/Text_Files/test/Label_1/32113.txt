import java.math.BigInteger;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        long ans = n * (n - 1) / 2;
        if (n % 2 == 0) {
            System.out.println(ans + "/" + n / 2);
        } else {
            System.out.println(ans + "/" + n);
        }
    }
}
