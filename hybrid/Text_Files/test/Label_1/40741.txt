n = int(input().strip())
s = input().strip()

def can_fix_bracket_sequence(s):
    if len(s) % 2 == 1:
        return "No"
    balance = 0
    for ch in s:
        if ch == '(':
            balance += 1
        else:
            balance -= 1
            if balance < -1:
                return "No"
    if balance == 0 or balance == -2:
        return "Yes"
    return "No"

print(can_fix_bracket_sequence(s))
