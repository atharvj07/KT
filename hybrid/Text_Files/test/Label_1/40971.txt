def solve():
    for _ in range(int(input())):
        a = [int(i) for i in input().split()]
        l = 0
        r = 0
        ans = 0
        while r < len(a):
            if a[l] > a[r]:
                ans = r
                while l < r:
                    l += 1
                    if a[l] > a[r]: ans = r
            r += 1
        print(ans)
solve()
