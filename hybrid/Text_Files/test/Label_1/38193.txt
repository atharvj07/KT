t = int(input())
for _ in range(t):
    n = int(input())
    operations = []
    
    # Reduce all elements from 3 to n to 1
    for i in range(3, n + 1):
        operations.append((i, 2))
    
    # Reduce second element to 1 if n > 2
    if n > 2:
        operations.extend([(2, n), (2, n), (2, n)])
        operations.append((n, 2))
    
    # Print the result
    print(len(operations))
    for op in operations:
        print(op[0], op[1])
