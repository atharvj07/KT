#include <bits/stdc++.h>
using namespace std;

int n, m, k;
char g[11][21];

bool ok() {
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (g[i][j] == '*') {
                int d = 0;
                for (int dx = -i; dx <= n - i; dx++) {
                    int dy = j + dx;
                    if (dy <= 0 || dy > m) continue;
                    if (g[i + dx][dy] == '.') break;
                    d++;
                }
                for (int dx = -i; dx <= n - i; dx++) {
                    int dy = j - dx;
                    if (dy <= 0 || dy > m) continue;
                    if (g[i + dx][dy] == '.') break;
                    d++;
                }
                if (d < k) return false;
            }
        }
    }
    return true;
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        cin >> n >> m >> k;
        for (int i = 1; i <= n; i++) { cin >> g[i] + 1; }
        if (ok())
            cout << "YES" << endl;
        else
            cout << "NO" << endl;
    }
    return 0;
}
