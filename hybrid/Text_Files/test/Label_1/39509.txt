n, m = map(int, input().split())
a = list(map(int, input().split()))
flag = [False for _ in range(n)]
pool = []
ans = []
for ai in a:
    pool.append(ai)
    if len(set(pool)) == n:
        for j in pool:
            flag[j - 1] = True
        ans.append(1)
        pool = []
    else:
        ans.append(0)
print(''.join(map(str, ans)))
