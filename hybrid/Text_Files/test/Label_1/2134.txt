#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=998244353;
const int N=200005;
int n,dp[N][2][2];
char s[N][3];
int f(int pos,int pre1,int pre2){
    if(pos>n){
        if(pre1!=pre2) return 1;
        return 0;
    }
    int &ret=dp[pos][pre1][pre2];
    if(ret!=-1) return ret;
    ret=0;
    if(s[pos][0]=='?'){
        ret=1LL*f(pos+1,pre2,pre1)%mod;
    }
    ret=(ret+1LL*f(pos+1,pre1,pre2))%mod;
    if(s[pos][1]=='?'){
        ret=1LL*f(pos+1,pre2,!pre2)%mod;
    }
    ret=(ret+1LL*f(pos+1,pre1,pre2))%mod;
    return ret;
}
int main(){
    memset(dp,-1,sizeof dp);
    cin>>n;
    for(int i=1;i<=n;++i){
        cin>>s[i];
    }
    int ans=f(1,0,0);
    cout<<ans<<'\n';
}
