#include <bits/stdc++.h>
using namespace std;
int a[100001];
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; ++i) cin >> a[i];
    sort(a, a + n);
    int k = 0;
    for (int i = 0; i < n - 1; ++i) {
        if (a[i + 1] - a[i] > 2 * k) k = (a[i + 1] - a[i]) / 2;
    }
    cout << k << endl;
    return 0;
}
