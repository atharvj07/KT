import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = in.nextInt();
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (a[i] >= 2) {
                a[i] -= 2;
                ans++;
            }
            if (a[i] >= 1 && a[i + 1] >= 1) {
                a[i] -= 1;
                a[i + 1] -= 1;
                ans++;
            }
        }
        if (a[n - 1] >= 1) {
            ans++;
        }
        System.out.println(ans);
    }
}
