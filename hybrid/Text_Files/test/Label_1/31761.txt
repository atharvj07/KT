import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int r = sc.nextInt();
        int n = sc.nextInt();
        boolean[][] grid = new boolean[r + 1][3];
        for (int i = 0; i < n; i++) {
            int ri = sc.nextInt();
            int ci = sc.nextInt();
            grid[ri][ci] = true;
            grid[ri][3 - ci] = true;
        }
        int last = 0;
        for (int i = 1; i <= r; i++) {
            for (int j = 1; j < 3; j++) {
                if (!grid[i][j]) {
                    last = 3 - j;
                    break;
                }
            }
        }
        if (last == 0) {
            System.out.println("LOSE");
        } else {
            System.out.println("WIN");
        }
    }
}
