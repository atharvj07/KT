import java.util.*;

class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt(), k = sc.nextInt();
    int[] a = new int[n], b = new int[n];
    for (int i = 0; i < n; i++) a[i] = sc.nextInt();
    for (int i = 0; i < n; i++) b[i] = sc.nextInt();
    int[] need = new int[n], ans = new int[n];
    for (int i = 0; i < n; i++) {
      if (b[i] / a[i] > k) {
        k += b[i] % a[i];
        need[i] = b[i] / a[i];
        ans[i] = b[i] % a[i];
        continue;
      }
      if (k >= b[i] / a[i]) {
        k -= b[i] / a[i];
        need[i] = b[i] / a[i];
        ans[i] = 0;
        continue;
      }
      need[i] = 0;
      ans[i] = b[i] % a[i];
    }
    int rem = k % n, add = k / n, flag = 0;
    for (int i = 0; i < n; i++) {
      if (rem > 0) {
        if (ans[i] > 0 || add > 0) {
          ans[i]--;
          need[i]++;
          rem--;
          add--;
          if (add == 0 && rem > 0) flag = 1;
        } else {
          ans[i] += rem;
          rem = 0;
          flag = 1;
        }
      } else if (add > 0) {
        if (ans[i] > 0) {
          ans[i]--;
          need[i]++;
          add--;
        } else if (flag == 0) {
          need[i] += add;
          add = 0;
        }
      }
    }
    int res = Integer.MAX_VALUE;
    for (int i = 0; i < n; i++) res = Math.min(res, need[i] + (add > 0 ? 1 : 0));
    System.out.println(res);
  }
}
