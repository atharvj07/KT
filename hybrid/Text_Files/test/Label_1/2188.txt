#include <bits/stdc++.h>

using namespace std;

const int MAXN = 2e5 + 5;

struct node {
    int to, cost;
};

vector<node> adj[MAXN];
int ans[MAXN];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int c;
        cin >> c;
        ans[i] = c;
    }
    for (int i = 1; i <= n; i++) {
        int a;
        cin >> a;
        adj[i].push_back(node{a, 0});
    }
    for (int i = 1; i <= n; i++) {
        queue<int> q;
        q.push(i);
        ans[i] = min(ans[i], ans[i] + adj[i][0].cost);
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            for (node &v : adj[u]) {
                if (ans[u] < ans[v.to]) {
                    ans[v.to] = ans[u];
                    q.push(v.to);
                }
            }
        }
    }
    int res = 1e9;
    for (int i = 1; i <= n; i++) {
        res = min(res, ans[i]);
    }
    cout << res << endl;
    return 0;
}
