#include <bits/stdc++.h>
using namespace std;
int n, x, a[50005], sum[50005], pref[50005], suff[50005], ans;
int main() {
    int test;
    cin >> test;
    while (test--) {
        cin >> n;
        sum[0] = 0;
        for (int i = 1; i <= n; i++) {
            cin >> a[i];
            sum[i] = sum[i - 1] + a[i];
        }
        cin >> x;
        pref[1] = 0;
        for (int i = 2; i <= n; i++) {
            if (sum[i - 1] < x)
                pref[i] = i;
            else
                pref[i] = pref[i - 1];
        }
        suff[n] = n - 1;
        for (int i = n - 1; i >= 1; i--) {
            if (sum[n] - sum[i] < x)
                suff[i] = i - 1;
            else
                suff[i] = suff[i + 1];
        }
        for (int i = 1; i <= n; i++) {
            if (pref[i] <= suff[i])
                ans = i;
        }
        cout << ans << '\n';
    }
}
