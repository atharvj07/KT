#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

const int MOD = 998244353;

int main() {
  int n, k;
  cin >> n >> k;
  vector<vector<bool>> dp(n + 1, vector<bool>(k + 1, false));
  dp[0][0] = true;
  for (int i = 1; i <= n; i++) {
    for (int j = 0; j <= k; j++) {
      if (dp[i - 1][j]) {
        dp[i][j] = true;
        dp[i][j + 1] = true;
        if (j >= 1) dp[i][j - 1] = true;
      }
    }
  }
  long long ans = 0;
  for (int i = 1; i <= k; i++) {
    if (dp[n][i]) {
      ans = (ans + i * (i + 1) / 2 % MOD) % MOD;
    }
  }
  cout << ans << endl;
  return 0;
}
