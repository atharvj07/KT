import java.util.Scanner;
import java.util.Stack;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        String s = scanner.next();
        Stack<Character> stack = new Stack<>();
        int[] result = new int[n];
        int colors = 0;
        boolean correct = true;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'L') {
                if (!stack.isEmpty()) {
                    stack.pop();
                    if (stack.isEmpty()) {
                        colors = 0;
                    } else {
                        colors = Math.max(colors, stack.peek() - '0' + 1);
                    }
                }
            } else {
                if (s.charAt(i) == '(') {
                    colors++;
                    stack.push((char) ('0' + colors));
                } else if (!stack.isEmpty() && stack.peek() == '(') {
                    stack.pop();
                } else {
                    correct = false;
                    colors = -1;
                }
            }
            result[i] = correct ? colors : -1;
        }
        for (int i : result) {
            System.out.print(i + " ");
        }
    }
}
