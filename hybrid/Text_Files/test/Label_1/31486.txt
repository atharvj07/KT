import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner scan = new Scanner(System.in);
    int n = scan.nextInt(), m = scan.nextInt(), ans = -1;
    char[][] a = new char[n][m];
    for (int i = 0; i < n; i++) a[i] = scan.next().toCharArray();
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < m; j++) {
        if (a[i][j] == 'B') {
          if (ans == -1) ans = 0;
          int len = 1, l = i, r = i, u = j, d = j;
          while (l >= 0 && a[l][j] == 'B') l--;
          while (r < n && a[r][j] == 'B') r++;
          while (u >= 0 && a[i][u] == 'B') u--;
          while (d < m && a[i][d] == 'B') d++;
          ans += (r - l) * (d - u) - (len * len);
          len++;
        }
      }
    }
    System.out.println(ans);
  }
}
