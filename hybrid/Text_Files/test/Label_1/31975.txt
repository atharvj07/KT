import java.util.*;

class Main {
    static int n, m, k, ans = 0;
    static int[][] map;
    static boolean[][] visited;
    static int[][] dir = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};

    static void dfs(int y, int x, int cnt) {
        visited[y][x] = true;
        if (cnt + map[y][x] >= k) ans++;
        for (int[] d : dir) {
            int ny = y + d[0];
            int nx = x + d[1];
            if (nx < 0 || nx >= m || ny < 0 || ny >= n || visited[ny][nx]) continue;
            dfs(ny, nx, cnt + map[y][x]);
        }
        visited[y][x] = false;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        k = sc.nextInt();
        map = new int[n][m];
        visited = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            String s = sc.next();
            for (int j = 0; j < m; j++) map[i][j] = s.charAt(j) - '0';
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (map[i][j] == 1) dfs(i, j, 0);
            }
        }
        System.out.println(ans);
    }
}
