import java.util.Arrays;
import java.util.Scanner;

public class Main {
    static Long[] dp;
    static Long[] arr;
    static int n;
    static int x;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        x = sc.nextInt();
        arr = new Long[n];
        dp = new Long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextLong();
        }
        Arrays.sort(arr);
        System.out.println(fun(n - 1));
    }

    static Long fun(int i) {
        if (i == 0)
            return arr[0] * x;
        if (dp[i] != null)
            return dp[i];
        dp[i] = arr[i] * x + fun(i - 1);
        x--;
        x = Math.max(1, x);
        return dp[i];
    }
}
