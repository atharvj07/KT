#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e5;
const int MOD = 1e9 + 7;

typedef pair<int, int> seg;

int n, m;

vector<int> adj[MAXN + 1];
vector<seg> vec;
int mark[MAXN + 1];

int power(int a, int b) {
    if (b == 0) return 1;
    int res = power(a, b / 2);
    if (b % 2 == 0) return 1LL * res * res % MOD;
    return 1LL * res * res % MOD * a % MOD;
}

int32_t main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);

    cin >> n >> m;

    for (int i = 0; i < m; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        adj[a].push_back(b), adj[b].push_back(a);
        vec.push_back({a, b}), mark[a] |= (1 << c), mark[b] |= (1 << c);
    }

    int ans = 1;
    for (int i = 1; i <= n; i++) {
        int cnt = 0;
        for (auto u : adj[i]) {
            if (mark[u] & mark[i]) cnt++;
        }
        if (cnt > 1) ans = 0;
        else if (cnt == 1) ans = 1LL * ans * 2 % MOD;
    }

    for (auto x : vec) {
        int a = x.first, b = x.second;
        if (mark[a] & mark[b]) continue;

        ans = 1LL * ans * 2 % MOD;
        mark[a] |= (1 << 1), mark[b] |= (1 << 1);
    }

    cout << ans << '\n';

    return 0;
}
