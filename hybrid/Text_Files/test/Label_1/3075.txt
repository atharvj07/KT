#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    deque<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }
    while (n > 2) {
        if (a[0] < a[n - 1]) {
            if (a[0] < a[1]) {
                a.pop_front();
                n--;
            } else {
                a.pop_back();
                n--;
            }
        } else {
            if (a[n - 1] < a[n - 2]) {
                a.pop_front();
                n--;
            } else {
                a.pop_back();
                n--;
            }
        }
    }
    if (a[0] < a[1]) {
        cout << "Alice";
    } else {
        cout << "Bob";
    }
    return 0;
}
