import java.util.*;

public class Main {
    static long[] arr, sum;
    static int n, p;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        n = scanner.nextInt();
        arr = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            arr[i] = scanner.nextInt();
        }
        p = scanner.nextInt();
        sum = new long[n + 1];
        for (int i = 1; i <= n; i++) {
            sum[i] = sum[i - 1] + arr[i];
        }
        for (int i = 0; i < p; i++) {
            int a = scanner.nextInt(), b = scanner.nextInt();
            int l = a, r;
            long ans = 0;
            while (l <= n) {
                r = l + b - 1;
                if (r > n) r = n;
                ans += sum[r] - sum[l - 1];
                l += b;
            }
            System.out.println(ans);
        }
    }
}
