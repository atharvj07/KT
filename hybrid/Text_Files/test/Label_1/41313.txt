import sys

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def solve(s):
    n = len(s)
    a = [1] * n
    for i in range(1, n):
        for j in range(i):
            if s[i] % s[j] == 0:
                a[i] = max(a[i], a[j] + 1)
    if max(a) > n:
        return -1
    ans = []
    for i in range(n - 1, -1, -1):
        for j in range(i - 1, -1, -1):
            if s[i] % s[j] == 0 and a[i] == a[j] + 1:
                ans.append(s[j])
                break
        ans.append(s[i])
    ans.reverse()
    return ans

n = int(input())
s = list(map(int, input().split()))
ans = solve(s)
if ans == -1:
    print(ans)
else:
    print(len(ans))
    print(*ans)
