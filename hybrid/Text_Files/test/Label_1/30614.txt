import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        char[][] grid = new char[n][n];
        for (int i = 0; i < n; i++) {
            String line = sc.nextLine();
            for (int j = 0; j < n; j++) {
                grid[i][j] = line.charAt(j);
            }
        }
        int[][] dp = new int[n][n];
        for (int i = n - 1; i >= 0; i--) {
            for (int j = n - 1; j >= 0; j--) {
                if (i == n - 1 && j == n - 1) {
                    dp[i][j] = (grid[i][j] == '#') ? 1 : 0;
                } else if (i == n - 1) {
                    dp[i][j] = (grid[i][j] == '#') ? 1 + dp[i][j + 1] : dp[i][j + 1];
                } else if (j == n - 1) {
                    dp[i][j] = (grid[i][j] == '#') ? 1 + dp[i + 1][j] : dp[i + 1][j];
                } else {
                    dp[i][j] = (grid[i][j] == '#') ? 1 + Math.min(dp[i + 1][j], dp[i][j + 1]) : Math.min(dp[i + 1][j], dp[i][j + 1]);
                }
            }
        }
        System.out.println(dp[0][0]);
    }
}
