n = int(input())
a = list(map(int, input().split()))

friend_pos = a.index(-1)
rounds = n.bit_length() - 1

costs = []
for i in range(n):
    if i != friend_pos:
        stage = (i ^ friend_pos).bit_length() - 1
        costs.append((a[i], stage))

costs.sort(key=lambda x: x[1])

spent = 0
for cost, stage in costs:
    if stage < rounds:
        spent += cost
        rounds -= 1

print(spent)
