#include <bits/stdc++.h>
using namespace std;
char grid[25][25];
bool vis[25][25];
int dx[2] = {0, 1};
int dy[2] = {1, 0};
int n, m;

bool valid(int x, int y) {
    if (x < 0 || x >= n || y < 0 || y >= m)
        return false;
    if (vis[x][y])
        return false;
    if (grid[x][y] == '*')
        return false;
    return true;
}

bool dfs(int x, int y) {
    vis[x][y] = true;
    for (int i = 0; i < 2; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        if (!valid(nx, ny))
            continue;
        if (grid[nx][ny] == '*') {
            vis[x][y] = false;
            return true;
        }
        if (dfs(nx, ny)) {
            vis[x][y] = false;
            return true;
        }
    }
    vis[x][y] = false;
    return false;
}

int main() {
    cin >> n >> m;
    string hs, vs;
    cin >> hs >> vs;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) { grid[i][j] = '.'; }

    for (int i = 0; i < n; i++) {
        if (hs[i] == '>') {
            for (int j = 0; j < m - 1; j++) { grid[i][j] = '*'; }
        } else {
            for (int j = 1; j < m; j++) { grid[i][j] = '*'; }
        }
    }

    for (int j = 0; j < m; j++) {
        if (vs[j] == 'v') {
            for (int i = 0; i < n - 1; i++) { grid[i][j] = '*'; }
        } else {
            for (int i = 1; i < n; i++) { grid[i][j] = '*'; }
        }
    }

    bool ans = false;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == '.') {
                if (dfs(i, j)) {
                    ans = true;
                }
            }
        }
    }
    if (ans)
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
