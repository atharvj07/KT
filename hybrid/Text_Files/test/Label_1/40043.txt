n=int(input())
v=list(map(int,input().split()))
MX=0
v.sort()
MX=-1
for i in v:
    if(i%10!=0):
        MX=0
        break;
    if(i%9==0):
        MX=i
for i in range(len(v)-1,-1,-1):
    if(MX!=-1):
        MX=MX*10;
        MX+=v[i]
    else:
        if(v[i]==0):
            MX=0
        if(v[i]==5):
            MX=5
print(MX)
