#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    long long sum = 0;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a.begin(), a.end());
    for (int i = n - 1; i >= 0; i--) {
        if (sum & 1)
            sum += a[i];
        else if (a[i] % 2)
            sum += a[i];
        else if (i && a[i - 1] % 2)
            sum += a[i - 1];
    }
    cout << sum << endl;
    return 0;
}
