n, k = map(int, input().split())
a = list(map(int, input().split()))
if sum(a) % k:
    print("No")
else:
    d = sum(a) // k
    s = []
    cur = 0
    for i in range(n):
        cur += a[i]
        if cur == d:
            s.append(i + 1)
            cur = 0
    if cur:
        print("No")
    else:
        print("Yes")
        print(*s)
