from collections import deque
from sys import stdin

n, m = list(map(int, stdin.readline().split()))
g = [[] for _ in range(n + 1)]
for _ in range(m):
    a, b = list(map(int, stdin.readline().split()))
    g[a].append(b)
    g[b].append(a)

q = deque([1])
used = [0] * (n + 1)
used[1] = 1
res = []
while q:
    v = q.popleft()
    for u in g[v]:
        if used[u] == 1:
            continue
        used[u] = 1
        res.append((v, u))
        q.append(u)

for edge in res:
    print(*edge)
