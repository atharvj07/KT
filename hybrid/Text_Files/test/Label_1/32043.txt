import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n, s;
        while (in.hasNext()) {
            n = in.nextInt();
            s = in.nextInt();
            int[] dp = new int[s + 1];
            dp[0] = 1;
            for (int i = 1; i <= n; i++) {
                for (int j = s; j >= i; j--) {
                    dp[j] += dp[j - i];
                }
            }
            System.out.println(dp[s]);
        }
    }
}
