#include <cstdio>
#include <vector>
const int MAXN = 2e5 + 5;
char s[MAXN];
int main() {
  int n, b, a;
  scanf("%d%d%d", &n, &b, &a);
  scanf("%s", s + 1);
  a += s[1] - '0';
  int ans = 1;
  for (int i = 2; i <= n; i++) {
    if (s[i] == '0') {
      if (b) {
        --b;
        ++ans;
        if (s[i + 1] == '1') ++a;
      }
    } else {
      if (a) {
        --a;
        ++ans;
        if (s[i + 1] == '1') ++a;
      }
    }
    if (b == 0 && a == 0) break;
  }
  printf("%d\n", ans);
  return 0;
}
