import sys

def make_road(edges,w):
    ans = 0
    in_roads = [0]*w
    out_roads = [0]*w
    for e in edges:
        in_roads[e[1]-1] += 1
        out_roads[e[0]-1] += 1
    for x in range(w):
        if in_roads[x] == out_roads[x]:
            ans += 1
    print(ans)
    for e in edges:
        if in_roads[e[1]-1] <= out_roads[e[0]-1]:
            print(e[0],e[1])
            in_roads[e[1]-1] += 1
        else:
            print(e[1],e[0])
            out_roads[e[0]-1] += 1
for t in range(int(input())):
    w,e = map(int,input().split())
    edges = []
    for x in range(e):
        edges.append(list(map(int,sys.stdin.readline().split())))
    make_road(edges,w)
