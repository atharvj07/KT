import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int max = 0;
        int cur = 0;
        for (char c : s.toCharArray()) {
            if (c == 'A' || c == 'C' || c == 'G' || c == 'T') {
                cur++;
            } else {
                cur = 0;
            }
            max = Math.max(max, cur);
        }
        System.out.println(max);
    }
}
