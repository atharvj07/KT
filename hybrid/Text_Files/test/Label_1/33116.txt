import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = in.nextInt();
        Arrays.sort(arr);
        int ans = 0;
        for (int i = 1, j = n - 1; i < j; i++, j--) {
            if (arr[i] != 0) {
                if (arr[j] == 0) ans++;
            } else if (arr[j] != 0) {
                ans++;
                arr[j]--;
            }
        }
        System.out.println(ans);
    }
}
