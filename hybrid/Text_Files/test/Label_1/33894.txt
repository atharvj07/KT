import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Main {
    private static final int MOD = 1000000007;
    private static Map<Long, Long> fib;
    private static long[] arr;

    private static long fastPow(long x, long p) {
        if (p == 0) return 1;
        if (p == 1) return x;
        if (p % 2 == 0) return fastPow(x * x % MOD, p / 2) % MOD;
        return (x * fastPow(x, p - 1)) % MOD;
    }

    private static long getFib(long n) {
        if (n == 0) return 0;
        if (n == 1) return 1;
        if (fib.containsKey(n)) return fib.get(n);
        long r = (getFib(n - 1) + getFib(n - 2)) % MOD;
        fib.put(n, r);
        return r;
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int m = Integer.parseInt(st.nextToken());
        st = new StringTokenizer(br.readLine());
        arr = new long[n];
        for (int i = 0; i < n; i++) {
            arr[i] = Long.parseLong(st.nextToken());
        }
        fib = new HashMap<>();
        TreeSet<Long> fibo = new TreeSet<>();
        for (int i = 1; i <= 100000; i++) {
            long v = getFib(i);
            fibo.add(v);
        }
        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int t = Integer.parseInt(st.nextToken());
            long l = Long.parseLong(st.nextToken()) - 1;
            long r = Long.parseLong(st.nextToken()) - 1;
            if (t == 1) {
                long x = Long.parseLong(st.nextToken());
                for (long j = l; j <= r; j++) {
                    arr[(int) j] += x;
                    arr[(int) j] %= MOD;
                }
            } else {
                long sum = 0;
                for (long j = l; j <= r; j++) {
                    sum += arr[(int) j];
                    sum %= MOD;
                }
                long ans = 0;
                for (long f : fibo) {
                    if (f > sum) break;
                    ans += (f * fastPow(10, sum - f)) % MOD;
                    ans %= MOD;
                }
                System.out.println(ans);
            }
        }
    }
}
