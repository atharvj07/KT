from collections import deque

for _ in range(int(input())):
    n = int(input())
    a = [int(x) for x in input().split()]
    a.sort()
    s = deque(a)
    ans = []
    while s:
        x = s.popleft()
        y = s.pop() if s else -1
        ans.append(x)
        if y != -1 and (x + y) % 2 == 0:
            ans.append(y)
    print(*ans)
