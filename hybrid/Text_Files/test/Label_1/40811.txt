s=input()
l=len(s)
a=[1 if x=='1' else 0 for x in s]
seg=[]
t=0
for x in range(l):
    if a[x]==0 and (x==0 or a[x]!=a[x-1]):t+=1;seg.append([x])
    elif t>0:seg[t-1].append(x)
b=[0 for i in range(l)]
for x in seg:b[x[0]]=b[x[-1]]=1
for x in range(l-1):b[x]+=b[x-1]
ans=[]
for x in range(l):
    if b[x]==0 and a[x]==0:ans.append(x)
if len(ans)<2:
    print(-1)
else:
    print(len(ans))
    for x in ans:
        z=x+1
        while a[z]==1:z+=1
        print(z-x,x+1,z)
        x=z
