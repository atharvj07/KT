import java.util.*;
class Main {
    static final long INF = (long) 1e18;
    static class Graph {
        int numNode;
        int numEdge;
        ArrayList<ArrayList<Pair>> adj;
        boolean visited[];
        Graph(int n) {
            numNode = n;
            numEdge = 0;
            adj = new ArrayList<>();
            for(int i = 0; i < n; i++)
                adj.add(new ArrayList<>());
            visited = new boolean[n];
        }
        void addEdge(int u, int v, int w) {
            if(u < 0 || u >= numNode || v< 0 || v >= numNode)
                return;
            numEdge++;
            adj.get(u).add(new Pair(v, w));
        }
        long floydWarshall() {
            long graph[][] = new long[numNode][numNode];
            for(int i = 0; i < numNode; i++)
                for(int j = 0; j < numNode; j++)
                    graph[i][j] = (i == j ? 0 : INF);
            for(int i = 0; i < numNode; i++) {
                for(Pair v : adj.get(i))
                    graph[i][v.first] = Math.min(graph[i][v.first], v.second);
            }
            for(int k = 0; k < numNode; k++)
                for(int i = 0; i < numNode; i++)
                    for(int j = 0; j < numNode; j++) {
                        if(graph[i][k] < INF && graph[k][j] < INF)
                            graph[i][j] = Math.min(graph[i][j], graph[i][k] + graph[k][j]);
                    }
            long ans = INF;
            for(int i = 0; i < numNode; i++)
                for(int j = 0; j < numNode; j++) {
                    if(i == j)
                        continue;
                    ans = Math.min(ans, graph[i][j] + graph[j][i]);
                }
            return ans < INF ? ans : -1;
        }
    }
    static class Pair {
        int first;
        int second;
        Pair(int x, int y) {
            first = x;
            second = y;
        }
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Graph g = new Graph(n);
        while(m-- > 0) {
            int s = sc.nextInt();
            int t = sc.nextInt();
            int w = sc.nextInt();
            g.addEdge(s, t, w);
        }
        long ans = g.floydWarshall();
        System.out.println(ans);
    }
}
