#include <bits/stdc++.h>

using namespace std;

const int MAXX=3e5+5;
int n,m,k;
long long d[MAXX];
vector<pair< int,pair<int, long long> > > adj[MAXX];

void dijkstra(int s) {
	priority_queue<pair< long long, int> > q;
	d[s] = 0;
	q.push({0, s});
	while(!q.empty()) {
		long long dist = -q.top().first;
		int v = q.top().second;
		q.pop();
		if(d[v] < dist) continue;
		for(auto edge : adj[v]) {
			int to = edge.first;
			long long len = edge.second.first;
			if(d[v] + len < d[to]) {
				d[to] = d[v] + len;
				q.push({-d[to], to});
			}
		}
	}
}
int main() {
	cin >> n >> m >> k;
	if(k == 0) {
		 cout << 0 << "\n";
		 return 0;
	}
	for(int i=1; i<=n; i++) d[i] = 1e18;
	for(int i=0; i<m; i++) {
		int x, y, w;
		 cin >> x >> y >> w;
		 adj[x].push_back({y, {w, i+1}});
		 adj[y].push_back({x, {w, i+1}});
	}
	dijkstra(1);
	vector<pair<long long, int>> edges;
	for(int i=1; i<=m; i++) {
		int x = adj[1][i-1].first;
		long long w = adj[1][i-1].second.first;
		edges.push_back({w, i});
	}
	sort(edges.begin(), edges.end());
	cout << k << "\n";
	for(int i = 0; i<k; i++) cout << edges[i].second << ' ';
	cout << "\n";
}
