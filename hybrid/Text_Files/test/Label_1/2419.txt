#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
vector<int> adj[200001], rev[200001];
int perm[200001];
int dp[200001][7];
struct node {
  int val, idx;
  bool operator<(const node &x) const { return val > x.val; }
};
vector<node> v;
void dfs0(int node, int par) {
  perm[node] = v.size();
  v.push_back({node, perm[node]});
  for (int x : adj[node]) {
    if (x != par) { dfs0(x, node), adj[x].erase(find(adj[x].begin(), adj[x].end(), node)); }
  }
}
void dfs1(int node, int par, int col) {
  for (int x : rev[node]) {
    if (x != par) {
      dp[x][col] = 0;
      for (int i = 0; i < 6; i++) {
        if ((col == 0 && i != 1 && i != 2) || (col == 1 && i != 0 && i != 2) ||
            (col == 2 && i != 0 && i != 1) || (col == 3 && i != 4 && i != 5) ||
            (col == 4 && i != 3 && i != 5) || (col == 5 && i != 3 && i != 4)) {
          dp[x][col] = (dp[x][col] + dp[node][i]) % mod;
        }
      }
      dfs1(x, node, perm[x]);
    }
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  int k, n;
  cin >> k >> n;
  for (int i = 0; i < n; i++) {
    int x;
    string s;
    cin >> x >> s;
    v.push_back({x, 0});
  }
  sort(v.begin(), v.end());
  int root = v[0].val;
  for (int i = 1; i < n; i++) {
    perm[v[i].val] = i;
    adj[v[i].val].push_back(v[0].val);
    adj[v[0].val].push_back(v[i].val);
  }
  dfs0(root, -1);
  for (int i = 1; i < n; i++) { rev[v[0].val].push_back(v[i].val); }
  for (int i = 1; i < n; i++) {
    for (int j = 0; j < 6; j++) { dp[v[i].val][j] = 1; }
    dfs1(v[i].val, -1, perm[root]);
  }
  int ans = 0;
  for (int i = 0; i < 6; i++) { ans = (ans + dp[root][i]) % mod; }
  cout << ans << endl;
  return 0;
}
