import java.util.*;

public class Main {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        Map<String, String> types = new HashMap<>();
        types.put("void", "void");
        types.put("errtype", "errtype");
        while (n-- > 0) {
            String line = sc.nextLine();
            if (line.startsWith("typedef")) {
                String[] parts = line.split(" ");
                types.put(parts[2], resolveType(parts[1], types));
            } else if (line.startsWith("typeof")) {
                String[] parts = line.split(" ");
                System.out.println(resolveType(parts[1], types));
            }
        }
    }

    private static String resolveType(String type, Map<String, String> types) {
        if (types.containsKey(type)) {
            return types.get(type);
        }
        if (type.endsWith("*")) {
            return resolveType(type.substring(0, type.length() - 1), types) + "*";
        } else if (type.startsWith("&")) {
            return "&" + resolveType(type.substring(1), types);
        }
        return "errtype";
    }
}
