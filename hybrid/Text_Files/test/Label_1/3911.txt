#include <iostream>

using namespace std;

int main() {
  long long x, ans = 0, p = 1, mask = 0;
  string s;
  cin >> s;
  int n = s.size();
  for (int i = n - 1; i >= 0; i--) {
    if (s[i] == '1') {
      mask |= p;
    }
    ans += p * (x & mask);
    p *= 2;
    x >>= 1;
  }
  cout << ans % 1000000007 << endl;
  return 0;
}
