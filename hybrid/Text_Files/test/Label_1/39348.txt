def solve(t, n):
    dp = [0] * (n + 1)
    dp[1] = t[0][1] + t[1][2]
    for i in range(2, n + 1):
        dp[i] = min(2 * dp[i - 1] + t[0][1] + t[1][2], dp[i - 1] + t[0][2] + t[0][1] + dp[i - 1] + t[1][2])
    return dp[n]

t = [list(map(int, input().split())) for _ in range(3)]
n = int(input())
print(solve(t, n))
