import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] a = new int[m];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        Arrays.sort(a);
        int ans = n;
        for (int i = 0; i < m; i++) {
            if (n <= 0) {
                ans = -1;
                break;
            }
            ans -= Math.min(a[i], n);
            n -= a[i];
        }
        System.out.println(ans);
    }
}
