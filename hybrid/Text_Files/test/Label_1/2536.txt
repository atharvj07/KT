#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, m, k;
    cin >> n >> m >> k;
    bool ok = 1;
    for (int i = 1; i <= n; i++) {
      string s;
      cin >> s;
      for (int j = 1; j <= m; j++) {
        bool flag = 0;
        if (s[j - 1] == '*') {
          for (int l = k; l >= 1; l--) {
            if ((i - l >= 1) and (j - l >= 1) and (i + l <= n) and
                (j + l <= m) and (s[j - 1 - l] == s[j - 1 + l] and
                                  s[j - 1 + l] == s[i - 1 - l] and
                                  s[i - 1 - l] == s[i + l - 1])) {
              flag = 1;
              break;
            }
          }
          if (!flag) ok = 0;
        }
      }
    }
    if (ok)
      cout << "YES" << endl;
    else
      cout << "NO" << endl;
  }
  return 0;
}
