#define endl "\n"

typedef long long ll;
typedef long double ld;
typedef vector<int> vi;
typedef pair<int, int> pi;
typedef vector<pi> vpi;
typedef vector<ll> vl;
typedef vector<vl> vvl;

#define fast ios_base::sync_with_stdio(false); cin.tie(NULL); cout.tie(NULL);
#define ordered_set tree<int, null_type,less<int>, rb_tree_tag,tree_order_statistics_node_update>
#define findbo find_by_order

int main()
{
	fast;
	string s;
	cin>>s;
	
	int n=s.size();
	vi freq(26,0);
	for(int i=0;i<n;i++){
		freq[s[i]-'a']++;
	}
	
	int odd_count=0;
	for(int i=0;i<26;i++){
		if(freq[i]%2!=0){
			odd_count++;
		}
	}
	
	if(odd_count>1){
		cout<<"NO SOLUTION";
		return 0;
	}
	
	string ans="";
	
	for(int i=0;i<26;i++){
		int x=freq[i]/2;
		while(x--){
			ans+=('a'+i);
		}
	}
	
	int mid=-1;
	for(int i=0;i<26;i++){
		if(freq[i]%2!=0){
			mid=i;
			break;
		}
	}
	
	if(mid!=-1){
		ans+=('a'+mid);
	}
	
	for(int i=25;i>=0;i--){
		int x=freq[i]/2;
		while(x--){
			ans+=('a'+i);
		}
	}
	
	cout<<ans;
	return 0;
}
