#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;

    long long a[n], dif, x = 0;
    cin >> a[0];

    for (int i = 1; i < n; i++) {
        cin >> a[i];
        if (i == 1)
            dif = a[i] - a[i - 1];
        else
            x += (dif != (a[i] - a[i - 1]));
    }

    if (x > 1)
        cout << -1 << endl;
    else
        cout << x << endl;
    return 0;
}
