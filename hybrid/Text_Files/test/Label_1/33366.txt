import java.util.Scanner;

public class MagicBoxes {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();

        long maxSideLength = 0;
        for (int i = 0; i < n; i++) {
            int ki = scanner.nextInt();
            long ai = scanner.nextLong();
            maxSideLength = Math.max(maxSideLength, (long) Math.pow(2, ki) * ai);
        }

        int p = 0;
        while ((long) Math.pow(2, p) < maxSideLength) {
            p++;
        }

        System.out.println(p);
    }
}
