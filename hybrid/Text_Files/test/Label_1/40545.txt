n = int(input().strip())
costs = [list(map(int, input().strip().split())) for _ in range(3)]
edges = [list(map(int, input().strip().split())) for _ in range(n-1)]

adj_list = [[] for _ in range(n+1)]
for u, v in edges:
    adj_list[u].append(v)
    adj_list[v].append(u)

# Check if the tree is a "line" tree (each node has at most two neighbors)
for neighbors in adj_list:
    if len(neighbors) > 2:
        print(-1)
        exit(0)

# Find the ends of the tree (nodes with only one neighbor)
ends = [i for i in range(1, n+1) if len(adj_list[i]) == 1]

# Perform a DFS to order the vertices linearly
order = []
visited = [False] * (n+1)
def dfs(v):
    visited[v] = True
    order.append(v)
    for u in adj_list[v]:
        if not visited[u]:
            dfs(u)
dfs(ends[0])

# Try all 6 permutations of colors for the first two vertices
from itertools import permutations
min_cost = float('inf')
best_perm = None
for perm in permutations([0, 1, 2], 3):
    cost = 0
    for i in range(n):
        cost += costs[perm[i % 3]][order[i]-1]
    if cost < min_cost:
        min_cost = cost
        best_perm = perm

# Map the best permutation back to the original vertex order
coloring = [0] * n
for i in range(n):
    coloring[order[i]-1] = best_perm[i % 3] + 1

print(min_cost)
print(' '.join(map(str, coloring)))
