#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
    ll n, a[100005];
    cin >> n;
    for (ll i = 1; i <= n; i++) cin >> a[i];
    if (n == 2) {
        cout << "yes" << endl;
        cout << 1 << " " << 2 << endl;
    } else {
        for (ll i = 1; i <= n - 2; i++) {
            if (a[i + 2] < a[i + 1] && a[i + 1] > a[i]) {
                cout << "yes" << endl;
                cout << i + 1 << " " << i + 2 << endl;
                return 0;
            }
        }
        cout << "no" << endl;
    }
    return 0;
}
