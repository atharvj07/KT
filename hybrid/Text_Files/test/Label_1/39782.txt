def count_occurrences(s, t):
    count = start = 0
    while True:
        start = s.find(t, start) + 1
        if start > 0:
            count += 1
        else:
            break
    return count

def is_good(substring, rules):
    for rule in rules:
        p, l, r = rule
        if not (l <= count_occurrences(p, substring) <= r):
            return False
    return True

def distinct_good_substrings(s, rules):
    good_substrings = set()
    for i in range(len(s)):
        for j in range(i+1, len(s)+1):
            substring = s[i:j]
            if substring not in good_substrings and is_good(substring, rules):
                good_substrings.add(substring)
    return len(good_substrings)

s = input().strip()
n = int(input().strip())
rules = [input().strip().split() for _ in range(n)]
for i in range(n):
    rules[i][1] = int(rules[i][1])
    rules[i][2] = int(rules[i][2])

print(distinct_good_substrings(s, rules))
