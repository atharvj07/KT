import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    int[] a = new int[n];
    for (int i = 0; i < n; i++) {
      a[i] = sc.nextInt();
    }
    long[] b = prefixSum(a);
    Map<Long, Integer> freq = new HashMap<>();
    for (int i = 0; i < n; i++) {
      freq.put(b[i], freq.getOrDefault(b[i], 0) + 1);
    }
    long[] vals = new long[freq.size()];
    int idx = 0;
    for (Long key : freq.keySet()) {
      vals[idx++] = key;
    }
    Arrays.sort(vals);
    int[] ans = new int[freq.size()];
    for (int i = 0; i < freq.size(); i++) {
      ans[i] = freq.get(vals[i]);
    }
    int[] count = getCounts(ans);
    int[] ret = new int[count.length];
    idx = 0;
    for (int c : count) {
      if (c > 0) {
        ret[idx++] = c;
      }
    }
    if (idx == 0) {
      System.out.println(-1);
    } else {
      Arrays.sort(ret, 0, idx);
      for (int i = 0; i < idx; i++) {
        System.out.print(ret[i] + " ");
      }
      System.out.println();
    }
  }

  public static long[] prefixSum(int[] a) {
    long[] ret = new long[a.length];
    ret[0] = a[0];
    for (int i = 1; i < a.length; i++) {
      ret[i] = ret[i - 1] + a[i];
    }
    return ret;
  }

  public static int[] getCounts(int[] a) {
    Map<Integer, Integer> freq = new HashMap<>();
    for (int e : a) {
      freq.put(e, freq.getOrDefault(e, 0) + 1);
    }
    int[] ret = new int[a.length];
    int idx = 0;
    for (Integer key : freq.keySet()) {
      ret[idx++] = key;
    }
    return ret;
  }
}
