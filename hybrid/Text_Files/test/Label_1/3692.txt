#include <bits/stdc++.h>

using namespace std;

const int MAXN = 1e6 + 5;
int n, l[MAXN], r[MAXN];

int main() {
    cin >> n;
    while (n--) {
        string s, t;
        cin >> s >> t;
        memset(l, 0, sizeof l);
        memset(r, 0, sizeof r);
        for (char c : s) l[c - 'a']++;
        for (char c : t) r[c - 'a']++;
        bool flag = true;
        for (int i = 0; i < 26; i++) {
            if (l[i] < r[i]) {
                flag = false;
                break;
            }
        }
        cout << (flag ? "YES\n" : "NO\n");
    }
    return 0;
}
