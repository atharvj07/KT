import re

s = input().strip()
n = int(input().strip())
rules = []
for _ in range(n):
    p, l, r = input().strip().split()
    rules.append((p, int(l), int(r)))

dict = {}

for i in range(len(s)):
    for j in range(i+1, len(s)+1):
        cur = s[i:j]
        cnt = 0
        for p, l, r in rules:
            cnt += len(re.findall(p, cur))
            if not (l <= cnt <= r):
                break
        if cnt == len(rules):
            dict[cur] = True

print(len(dict))
