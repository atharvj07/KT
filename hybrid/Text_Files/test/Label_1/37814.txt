import sys

n, h, m, k = map(int, sys.stdin.readline().split())
f = []
for _ in range(n): f.append(list(map(int, sys.stdin.readline().split())))

f.sort()

ans = n
best_t = 0

for t in range(m // 2):
    cnt = 0
    for h1, m1 in f:
        if h1 * m + m1 + k - t > h * m: cnt += 1
    if cnt < ans:
        ans = cnt
        best_t = t

print(ans, best_t)

cancelled = []
for h1, m1 in f:
    if h1 * m + m1 + k - best_t > h * m:
        cancelled.append(' '.join(map(str, [h1, m1])))

print(*cancelled, sep='\n')
