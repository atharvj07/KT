import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int q = sc.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < q; i++) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            System.out.println(isReachable(a, x, y) ? "Shi" : "Fou");
        }
    }

    static boolean isReachable(int[] a, int x, int y) {
        if ((a[x] & a[y]) == 0) {
            return false;
        }
        for (int i = x + 1; i < y; i++) {
            if ((a[i] & a[x]) == 0) {
                return false;
            }
        }
        return true;
    }
}
