import java.util.*;
import java.io.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(), m = sc.nextInt();
        int[] min = new int[n], max = new int[n];
        Arrays.fill(min, -1);
        for (int i = 0; i < m; i++) {
            int a = sc.nextInt() - 1, b = sc.nextInt() - 1, w = sc.nextInt();
            if (min[a] == -1 || min[a] > w) min[a] = w;
            if (max[b] == -1 || max[b] < w) max[b] = w;
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (min[i] != -1 && max[i] != -1) ans++;
        }
        System.out.println(ans);
    }
}
