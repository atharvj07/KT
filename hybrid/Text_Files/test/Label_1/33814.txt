import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

  private static class Person implements Comparable<Person> {
    private final String name;
    private final int tallerCount;
    private int height;

    public Person(String name, int tallerCount) {
      this.name = name;
      this.tallerCount = tallerCount;
    }

    public String getName() {
      return name;
    }

    public int getTallerCount() {
      return tallerCount;
    }

    public int getHeight() {
      return height;
    }

    public void setHeight(int height) {
      this.height = height;
    }

    @Override
    public int compareTo(Person other) {
      return Integer.compare(tallerCount, other.tallerCount);
    }
  }

  public static void main(String[] args) {
    Scanner scanner = new Scanner(System.in);

    int n = scanner.nextInt();
    scanner.nextLine();
    Map<String, Integer> tallerCountMap = new HashMap<>();
    Person[] people = new Person[n];
    for (int i = 0; i < n; i++) {
      String[] line = scanner.nextLine().split(" ");
      String name = line[0];
      int tallerCount = Integer.parseInt(line[1]);
      tallerCountMap.put(name, tallerCount);
      people[i] = new Person(name, tallerCount);
    }

    Arrays.sort(people);

    int[] heights = new int[n];
    int currentHeight = 1;
    for (Person person : people) {
      int tallerCount = person.getTallerCount();
      if (tallerCount != tallerCountMap.get(person.getName())) {
        System.out.println("-1");
        return;
      }
      for (int i = 0; i <= tallerCount; i++) {
        heights[i] = currentHeight++;
      }
      person.setHeight(heights[tallerCount]);
    }

    for (Person person : people) {
      System.out.println(person.getName() + " " + person.getHeight());
    }
  }
}
