#include <iostream>
#include <map>
#include <vector>

using namespace std;

int main() {
    long long a, b;
    cin >> a >> b;
    string s;
    cin >> s;
    map<long long, long long> xy = {{0, 0}};
    vector<pair<long long, long long>> dir = {{0, 1}, {0, -1}, {-1, 0}, {1, 0}};
    int idx = 0;
    for (char c : s) {
        switch (c) {
            case 'U':
                idx = 0;
                break;
            case 'D':
                idx = 1;
                break;
            case 'L':
                idx = 2;
                break;
            case 'R':
                idx = 3;
                break;
        }
        xy[a - xy.begin()->first] += dir[idx].first;
        xy[b - xy.begin()->second] += dir[idx].second;
        if (xy.count(0)) break;
    }
    puts(xy.count(0) ? "Yes" : "No");
    return 0;
}
