#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> f(m), g(m), h(m);
    for (int i = 0; i < m; i++) cin >> h[i];
    for (int i = 0; i < n; i++) f[a[i] - 1]++;
    for (int i = 0, j = 0, cnt = 0; i < n; i++) {
        while (j < n && (cnt < m || f[a[j] - 1] > h[a[j] - 1])) {
            cnt += (f[a[j] - 1] == h[a[j] - 1] ? 0 : 1);
            f[a[j] - 1]--;
            j++;
        }
        if (cnt == m) g[a[i] - 1]++;
        cnt -= (f[a[i] - 1] == h[a[i] - 1] ? 0 : 1);
        f[a[i] - 1]++;
    }
    int ans = n;
    for (int i = 0; i < m; i++) ans = min(ans, g[i]);
    if (ans == n) ans = -1;
    cout << ans << endl;
    return 0;
}
