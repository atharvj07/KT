n = int(input())
obelisks = [tuple(map(int, input().split())) for _ in range(n)]
clues = [tuple(map(int, input().split())) for _ in range(n)]

sum_obelisks = [sum(x) for x in zip(*obelisks)]
sum_clues = [sum(x) for x in zip(*clues)]

print(sum_obelisks[0] + sum_clues[0] - sum_obelisks[0], sum_obelisks[1] + sum_clues[1] - sum_obelisks[1])
