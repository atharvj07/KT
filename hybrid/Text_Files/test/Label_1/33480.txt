import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        long n = input.nextLong();
        long k = input.nextLong();

        if ((n & 1) > 0 && k == (1 << n)) {
            System.out.println("-1");
        } else {
            if (k % 4 == 1) {
                k++;
            }
            long ans = k / 4;
            if ((k & 1) > 0) {
                ans++;
            }
            System.out.println(ans);
        }
    }
}
