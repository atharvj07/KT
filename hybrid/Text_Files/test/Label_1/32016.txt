import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        String s = sc.next();
        int[] a = new int[10];
        for (int i = 0; i < n; i++) {
            a[s.charAt(i) - '0']++;
        }
        boolean ok = false;
        for (int i = 0; i < 10; i++) {
            if (a[i] > 1) {
                ok = true;
            }
        }
        System.out.println(ok ? "NO" : "YES");
    }
}
