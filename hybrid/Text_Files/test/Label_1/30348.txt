import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();
        int m = scanner.nextInt();
        char[][] crossword = new char[n][m];
        for (int i = 0; i < n; i++) {
            String line = scanner.next();
            for (int j = 0; j < m; j++) {
                crossword[i][j] = line.charAt(j);
            }
        }

        boolean[][] crossedOut = new boolean[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                for (int k = j + 1; k < m; k++) {
                    if (crossword[i][j] == crossword[i][k]) {
                        crossedOut[i][j] = true;
                        crossedOut[i][k] = true;
                    }
                }
                for (int k = i + 1; k < n; k++) {
                    if (crossword[i][j] == crossword[k][j]) {
                        crossedOut[i][j] = true;
                        crossedOut[k][j] = true;
                    }
                }
            }
        }

        StringBuilder answer = new StringBuilder();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (!crossedOut[i][j]) {
                    answer.append(crossword[i][j]);
                }
            }
        }

        System.out.println(answer);
    }
}
