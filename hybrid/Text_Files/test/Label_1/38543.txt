s = input()
q = int(input())
ans = []
cnt = {}
for c in s:
    cnt[c] = cnt.get(c, 0) + 1
for _ in range(q):
    op, *arg = map(str, input().split())
    if op == '1':
        pos, c = arg
        pos = int(pos)
        cur = s[pos - 1]
        s = s[:pos - 1] + c + s[pos:]
        cnt[cur] -= 1
        cnt[c] = cnt.get(c, 0) + 1
    else:
        l, r = map(int, arg)
        cur = set()
        for i in range(l - 1, r):
            cur.add(s[i])
        ans.append(len(cur))
print(*ans)
