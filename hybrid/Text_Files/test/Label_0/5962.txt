#include <bits/stdc++.h>
using namespace std;
void solve(){
    int n,i,j,x=0;
        cin>>n;
        int a[n+5],b[n+5],c[n+5];
        a[0]=b[0]=0;
        for(i=1;i<=n;i++)
            cin>>a[i]>>b[i];
        for(i=1;i<=n;i++)
            cin>>c[i];
        for(i=1;i<=n;i++)
        {
            x+=a[i]-b[i-1]+c[i];
            if(i<n)
                x=max(b[i],x+(b[i]-a[i]+1)/2);
        }
        cout<<x<<endl;
    }
int main(){
    int t;
    cin >> t;
    while(t--) solve();
}
