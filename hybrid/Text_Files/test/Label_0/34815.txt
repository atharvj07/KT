import java.util.*;
public class TripForMeal {
    public static void main(String args[]){
        Scanner in = new Scanner(System.in);
        int n = in.nextInt() - 1, a = in.nextInt(), b = in.nextInt(), c = in.nextInt(), first = Math.min(a, b), min = Math.min(c, first);
        if (n == 0) System.out.println("0");
        else System.out.println(first + min * (n - 1));
                
    }
}
