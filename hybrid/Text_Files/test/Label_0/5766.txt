#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int main()
{
    ll t;
    cin>>t;

    while(t--)
    {
        ll n;
        cin>>n;

        ll a[n+5];

        ll mark = 0;

        for(ll i = 0; i < n; i++)
        {
            cin>>a[i];
            ll x = sqrt(a[i]);

            if(x*x != a[i])
            {
                mark = 1;
            }
        }
        if(mark == 1)
        {
            cout<<"YES"<<endl;
        }
        else
        {
            cout<<"NO"<<endl;
        }
    }
    return 0;
}
