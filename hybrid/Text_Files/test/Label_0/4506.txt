#include <iostream>
#include<deque>
typedef long long int ll;

using namespace std;

int main()
{
    int N;
    cin>>N;
    deque<ll> p;
    while(N--)
    {
        int numhah;
        cin>>numhah;
        switch (numhah)
        {
            case 0:
            int opr;
            cin>>opr;
            ll num;
            cin>>num;
            if(opr==1)p.push_back(num);
            else p.push_front(num);
                break;
            case 1:
                int num1;
                cin>>num1;
                cout<<p.at(num1)<<endl;
                break;
            case 2:
                int num2;
                cin>>num2;
                if(num2==0)p.pop_front();
                else p.pop_back();
                break;
            default:break;
        }
    }
}
