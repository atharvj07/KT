import java.io.*;
import java.util.Arrays;

public class Main {
    static myScanner sc;
    static PrintWriter pw;

    public static void main(String[] args) throws IOException {
        //setInAndOutPutFromFile("ladder");
        setDefaultInAndOutPut();

        int numberOfTests = 1;
        for (int test = 1; test <= numberOfTests; test++) {
            solve();
            pw.flush();
        }

        pw.close();
    }

    static void setDefaultInAndOutPut() {
        sc = new myScanner(new BufferedReader(new InputStreamReader(System.in)));
        pw = new PrintWriter(System.out);
    }

    static void setInAndOutPutFromFile(String s) throws IOException {
        sc = new myScanner(new BufferedReader(new FileReader(new File(s + ".in"))));
        pw = new PrintWriter(new File(s + ".out"));
    }

    static void solve() {
        int d = sc.nextInt();
        int n = sc.nextInt(), m = sc.nextInt();
        D[] some = new D[d];
        for (int i = 0; i < d; i++) some[i] = new D(sc.nextInt(), sc.nextInt(), sc.nextInt(), sc.nextInt());
        int cntL = sc.nextInt(), cntR = sc.nextInt(), cntT = sc.nextInt(), cntB = sc.nextInt();
        int[] xL = new int[n+2];
        for (int i=0;i<d;i++)xL[some[i].minX]++;
        for (int i=1;i<=n;i++)xL[i]+=xL[i-1];

        int[] xR = new int[n+2];
        for (int i=0;i<d;i++)xR[some[i].maxX]++;
        for (int i=n-1;i>=1;i--)xR[i]+=xR[i+1];

        int[] yL = new int[m+2];
        for (int i=0;i<d;i++)yL[some[i].minY]++;
        for (int i=1;i<=m;i++)yL[i]+=yL[i-1];

        int[] yR = new int[m+2];
        for (int i=0;i<d;i++)yR[some[i].maxY]++;
        for (int i=m-1;i>=1;i--)yR[i]+=yR[i+1];

        int index = -1;
        for (int i = 0; i < d; i++) {
            if (xL[some[i].maxX-1]-(some[i].maxX!=some[i].minX?1:0)==cntL){
                if (xR[some[i].minX+1]-(some[i].maxX!=some[i].minX?1:0)==cntR){
                    if (yL[some[i].maxY-1]-(some[i].maxY!=some[i].minY?1:0)==cntT){
                        if (yR[some[i].minY+1]-(some[i].maxY!=some[i].minY?1:0)==cntB){
                            index=i+1;
                        }
                    }
                }
            }
        }
        if (index == -1) {
            pw.print(-1);
            return;
        }
        pw.print(index);
    }

    static class D {
        int minX;
        int maxX;
        int minY;
        int maxY;

        D(int x1, int y1, int x2, int y2) {
            minX = Math.min(x1, x2);
            maxX = Math.max(x1, x2);
            minY = Math.min(y1, y2);
            maxY = Math.max(y1, y2);
        }
    }
}

class myScanner {
    private char[] buffer = new char[1 << 8];
    private int pos = 1;

    private BufferedReader reader;

    public myScanner(BufferedReader reader) {
        this.reader = reader;
    }

    public boolean hasNext() {
        return pos > 0;
    }

    private void loadBuffer() {
        pos = 0;
        try {
            for (int i; (i = reader.read()) != -1; ) {
                char c = (char) i;
                if (c != ' ' && c != '\n' && c != '\t' && c != '\r' && c != '\f') {
                    if (pos == buffer.length) buffer = Arrays.copyOf(buffer, 2 * pos);
                    buffer[pos++] = c;
                } else if (pos != 0) break;
            }
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }

    public String current() {
        return String.copyValueOf(buffer, 0, pos);
    }


    public String next() {
        loadBuffer();
        return current();
    }


    public String nextLine() {
        try {
            return reader.readLine();
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }


    public int nextInt() {
        return nextInt(10);
    }

    public long nextLong() {
        return nextLong(10);
    }

    public int nextInt(int radix) {
        loadBuffer();
        int result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber(0 <= digit && digit <= radix - 1);
            result = result * radix + digit;
        }
        return buffer[0] == '-' ? -result : result;
    }

    public long nextLong(int radix) {
        loadBuffer();
        long result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber(0 <= digit && digit <= radix - 1);
            result = result * radix + digit;
        }
        return buffer[0] == '-' ? -result : result;
    }

    public double nextDouble() {
        loadBuffer();
        long result = 0;
        int i = buffer[0] == '-' || buffer[0] == '+' ? 1 : 0;
        long round = 1;
        final int radix = 10;
        boolean hasPoint = false;
        for (checkValidNumber(pos > i); i < pos; i++) {
            int digit = buffer[i] - '0';
            checkValidNumber((0 <= digit && digit <= radix - 1) || (!hasPoint && digit == -2));
            if (digit == -2) hasPoint = true;
            else {
                if (hasPoint) round *= radix;
                result = result * radix + digit;
            }

        }
        return buffer[0] == '-' ? -result / (1.0 * round) : result / (1.0 * round);
    }

    private void checkValidNumber(boolean condition) {
        if (!condition) throw new NumberFormatException(current());
    }
}
