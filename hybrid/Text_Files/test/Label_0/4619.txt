#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, tmp, nsolve = 0, npen = 0, stime = 10;
  vector<int> comp;
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> tmp;
    comp.push_back(tmp);
  }
  sort(comp.begin(), comp.end());
  for (int i = 0; i < n; i++) {
    if (comp[i] + stime <= 360) {
      nsolve++;
      stime += comp[i];
    } else {
      int ess = comp[i] + stime - 360;
      if (ess <= 360) {
        nsolve++;
        npen += ess;
        stime = 360 + ess;
      } else {
        break;
      }
    }
  }
  cout << nsolve << " " << npen << endl;
  return 0;
}
