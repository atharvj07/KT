	final String filename = "";

	public void solve() throws Exception {
		final int INF = 60000;
		int n = iread(), m = iread();
		int[][] ss = new int[n][n];
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
				ss[i][j] = INF;
		for (int i = 0; i < n; i++)
			ss[i][i] = 0;
		for (int i = 0; i < m; i++) {
			int a = iread(), b = iread(), c = iread();
			a--;
			b--;
			ss[a][b] = c;
			ss[b][a] = c;
		}
		int g1 = iread(), g2 = iread(), s1 = iread(), s2 = iread();
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
				for (int k = 0; k < n; k++)
					ss[j][k] = Math.min(ss[j][k], ss[j][i] + ss[i][k]);
		int[] worst = new int[n + 1];
		int[] best = new int[n + 1];
		Arrays.fill(worst, INF);
		Arrays.fill(best, -1);
		worst[n] = 0;
		best[n] = INF;
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < n; j++) {
				if (j == i)
					continue;
				best[i] = Math.max(best[i], ss[i][j]);
				worst[i] = Math.min(worst[i], ss[i][j]);
			}
		}

		long[][] C = new long[51][51];
		C[0][0] = 1;
		for (int i = 1; i <= 50; i++) {
			C[i][0] = 1;
			C[i][i] = 1;
			for (int j = 1; j < i; j++)
				C[i][j] = C[i - 1][j - 1] + C[i - 1][j];
		}

		long ans = 0;
		for (int a = 0; a <= n; a++)
			// �������� ������
			cycle: for (int b = 0; b < n; b++) // ��������� ������
			{
				if (a == b)
					continue;
				int[] d = new int[8];
				for (int i = 0; i < n; i++) {
					if (i == a) {
						d[1]++;
						continue;
					}
					if (i == b) {
						d[4]++;
						continue;
					}
					int mask = 0;
					for (int j = 0; j < n; j++) {
						if (i == j)
							continue;
						if (ss[i][j] > best[a] || ss[i][j] == best[a] && i > a)
							mask |= 1;
						else if (ss[i][j] > worst[b] || ss[i][j] == worst[b]
								&& i > b)
							mask |= 2;
						else
							mask |= 4;
					}
					d[mask]++;
				}
				long[][][] t = new long[5][n + 1][n + 1];
				int min_g = d[4], min_s = d[2];
				t[0][min_g][min_s] = 1;

				int s = 0;
				for (int u = min_g; u <= n; u++)
					for (int v = min_s; v <= n; v++) {
						if (t[s][u][v] == 0)
							continue;
						for (int i = 0; i <= d[3]; i++)
							t[s + 1][u][v + i] += t[s][u][v] * C[d[3]][i];
					}
				s = 1;
				for (int u = min_g; u <= n; u++)
					for (int v = min_s; v <= n; v++) {
						if (t[s][u][v] == 0)
							continue;
						for (int i = 0; i <= d[5]; i++)
							t[s + 1][u + i][v] += t[s][u][v] * C[d[5]][i];
					}
				s = 2;
				for (int u = min_g; u <= n; u++)
					for (int v = min_s; v <= n; v++) {
						if (t[s][u][v] == 0)
							continue;
						for (int i = 0; i <= d[6]; i++)
							t[s + 1][u + i][v + d[6] - i] += t[s][u][v]
									* C[d[6]][i];
					}
				s = 3;
				for (int u = min_g; u <= n; u++)
					for (int v = min_s; v <= n; v++) {
						if (t[s][u][v] == 0)
							continue;
						for (int i = 0; i <= d[7]; i++)
							for (int j = 0; i + j <= d[7]; j++)
								t[s + 1][u + i][v + j] += t[s][u][v]
										* C[d[7]][i] * C[d[7] - i][j];
					}
				for (int g = g1; g <= g2; g++)
					for (s = s1; s <= s2; s++)
						ans += t[4][g][s];
			}
		out.write(ans + "\n");
	}

	public void run() {
		try {
			in = new BufferedReader(new InputStreamReader(System.in));
			out = new BufferedWriter(new OutputStreamWriter(System.out));
			// in = new BufferedReader(new FileReader(filename+".in"));
			// out = new BufferedWriter(new FileWriter(filename+".out"));
			solve();
			out.flush();
		} catch (Exception e) {
			e.printStackTrace();
			System.exit(1);
		}
	}

	public int iread() throws Exception {
		return Integer.parseInt(readword());
	}

	public double dread() throws Exception {
		return Double.parseDouble(readword());
	}

	public long lread() throws Exception {
		return Long.parseLong(readword());
	}

	BufferedReader in;

	BufferedWriter out;

	public String readword() throws IOException {
		StringBuilder b = new StringBuilder();
		int c;
		c = in.read();
		while (c >= 0 && c <= ' ')
			c = in.read();
		if (c < 0)
			return "";
		while (c > ' ') {
			b.append((char) c);
			c = in.read();
		}
		return b.toString();
	}
