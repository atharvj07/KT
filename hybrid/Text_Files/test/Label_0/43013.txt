n = int(input())
draw = 0
maxSkor = 0
minSkorLast = 0
maxSkorLast = 0
for moment in range(n):
	skor = [int(item) for item in input().split(' ')]
	minSkor = min(skor[0], skor[1])
	maxSkor = max(skor[0], skor[1])
	drawPoin = minSkor - maxSkorLast + 1
	if drawPoin > 0:
		if maxSkorLast != minSkorLast:
			draw = draw + drawPoin
		else:
			draw = draw + drawPoin - 1
	minSkorLast = minSkor
	maxSkorLast = maxSkor
print(draw + 1)
