def f(array, i, j):
	result = array[i + 1][j] ^ array[i][j - 1]
	array[i][j] = result
	return result
arrayLength = int(input())
textArray = input().split(' ')
queriesNum = int(input())
queries = []
for i in range(queriesNum):
	queries.append(input())
results = []
processed = []
for i in range(arrayLength):
	processed.append([-1] * arrayLength)
	results.append([-1] * arrayLength)
for i in range(arrayLength):
	num = int(textArray[i])
	processed[i][i] = num
	results[i][i] = num
i = 0
spacing = 1
while spacing < arrayLength:
	i = 0
	while i + spacing < arrayLength:
		j = i + spacing
		num1 = f(processed, i, j)
		num2 = results[i + 1][j]
		num3 = results[i][j - 1]
		results[i][j] = max(num1, num2, num3)
		i += 1
	spacing += 1
for i in queries:
	query = i.split(' ')
	queryA = int(query[0]) - 1
	queryB = int(query[1]) - 1
	print(results[queryA][queryB])
