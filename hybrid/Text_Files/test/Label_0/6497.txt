#include <bits/stdc++.h>
using namespace std;

int n,a[400010],b[400010];
int s[400010],S[400010],fail[400010],pre[400010]; 

inline void failure()
{
	int i,j=-1;
	fail[0]=-1;
	for(int i=0;i<n;i++){
		while(j>=0&&s[i]!=s[j]) j=fail[j];
		j++;
		fail[i+1]=j;
	}
}

inline void kmp()
{
	int i,j=0;
	pre[0]=0;
	for(int i=0;i<n*2;i++){
		while(j==n||(j>=0&&S[i]!=s[j])) j=fail[j];
		j++;
		pre[i+1]=j;
	}
}

signed main()
{
	ios::sync_with_stdio(false);
	cin>>n;
	for(int i=0;i<n;i++) cin>>a[i];
	for(int i=0;i<n;i++) cin>>b[i];
	for(int i=0;i<n;i++) S[i]=a[i]^a[(i+1)%n];
	for(int i=n;i<n*2;i++) S[i]=S[i-n];
	for(int i=0;i<n;i++) s[i]=b[i]^b[(i+1)%n];
	failure();kmp();
	for(int i=1;i<n*2;i++){
		if(pre[i]==n){
			cout<<i-n<<" "<<(b[0]^a[i-n])<<endl;
		}
	}
	return 0;
}
