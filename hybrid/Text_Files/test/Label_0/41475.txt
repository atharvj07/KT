tt = int(input())
for _ in range(tt):
	(n, k) = map(int, input().split())
	a = list(map(int, input().split()))
	a.sort()
	cnt = 0
	was = [0] * n
	for i in range(n - 1, n - k - 1, -1):
		was[i] = 1
		was[i - k] = 1
		cnt += a[i - k] // a[i]
	for i in range(n):
		if was[i] == 0:
			cnt += a[i]
	print(cnt)
