/**
 * Built using CHelper plug-in
 * Actual solution is at the top
 */
public class Main {
    public static void main(String[] args) {
        InputStream inputStream = System.in;
        OutputStream outputStream = System.out;
        InputReader in = new InputReader(inputStream);
        PrintWriter out = new PrintWriter(outputStream);
        BalancedPlaylist solver = new BalancedPlaylist();
        solver.solve(1, in, out);
        out.close();
    }

    static class BalancedPlaylist {
        public void solve(int testNumber, InputReader in, PrintWriter out) {
            int N = in.nextInt();
            int[] arr = new int[N];
            int max = 0;
            for (int i = 0; i < N; i++) {
                arr[i] = in.nextInt();
                max = Math.max(max, arr[i]);
            }
            boolean flag = false;
            for (int i = 0; i < N; i++) {
                if (2 * arr[i] < max) {
                    flag = true;
                    break;
                }
            }
            int[] res = new int[N];
            if (!flag) {
                Arrays.fill(res, -1);
            } else {
                ArrayDeque<Integer> deque = new ArrayDeque<>();
                deque.add(0);
                int processed = 0;
                BalancedPlaylist.Multiset set = new BalancedPlaylist.Multiset();
                set.add(arr[0]);
                int id = 1;
                while (processed < N) {
                    while (set.size() > 0 && set.getFirst() > 2 * arr[id]) {
                        int next = deque.removeFirst();
                        if (res[next] == 0) {
                            res[next] = set.size();
                            processed++;
                        }
                        set.remove(arr[(id - set.size() + 2 * N) % N]);
                    }
                    set.add(arr[id]);
                    deque.add(id);
                    id++;
                    id %= N;
                }
            }
            for (int i = 0; i < N; i++) {
                out.println(res[i]);
            }
        }

        static class Multiset {
            TreeMap<Integer, Integer> map = new TreeMap<>(new Comparator<Integer>() {

                public int compare(Integer integer, Integer t1) {
                    return Integer.compare(t1, integer);
                }
            });
            int size = 0;

            public void add(int val) {
                map.putIfAbsent(val, 0);
                map.replace(val, map.get(val) + 1);
                size++;
            }

            public int getFirst() {
                return map.firstKey();
            }

            public void remove(int val) {
                map.replace(val, map.get(val) - 1);
                if (map.get(val) == 0) {
                    map.remove(val);
                }
                size--;
            }

            public int size() {
                return size;
            }

        }

    }

    static class InputReader {
        public BufferedReader reader;
        public StringTokenizer tokenizer;

        public InputReader(InputStream stream) {
            reader = new BufferedReader(new InputStreamReader(stream), 32768);
            tokenizer = null;
        }

        public String next() {
            while (tokenizer == null || !tokenizer.hasMoreTokens()) {
                try {
                    tokenizer = new StringTokenizer(reader.readLine());
                } catch (IOException e) {
                    throw new RuntimeException(e);
                }
            }
            return tokenizer.nextToken();
        }

        public int nextInt() {
            return Integer.parseInt(next());
        }

    }
}
