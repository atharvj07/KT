#include <bits/stdc++.h>
using namespace std;
#define rep(i, n) for(int i = 0; i < (int)n; ++i)
#define sz(s) ((int)(s.size()))

typedef complex<double> P;

typedef vector<P> poly;

typedef pair<P, P> L;

const double EPS = 1e-9;

double cross(const P &a, const P &b) { return imag(conj(a) * b); }
double dot(const P &a, const P &b) { return real(conj(a) * b); }

int ccw(P a, P b, P c) {
	double d = cross(b -= a, c -= a);
	return d > EPS ? 1 : d<-EPS ? -1 : dot(b, c) < -EPS ? 2 : norm(b) + EPS < norm(c) ? -2 : 0;
}

#define DI(l) ((l).second - (l).first)
#define CCW(l, a) ccw((l).first, (l).second, (a))

bool intersectSS(const L &s, const L &t) { return CCW(s, t.first)*CCW(s, t.second)<=0 && CCW(t, s.first)*CCW(t, s.second) <= 0; }
P crosspoint(const L &l, const L &m) {
	P a = DI(l), b = DI(m);
	double A  = cross(a, b), B = cross(a, l.second - m.first);
	return m.first + B / A * b;
}

double signed_area(const poly &v) {
	double res = 0;
	rep(i, sz(v)) res += cross(v[i], v[(i+1)%sz(v)]);
	return res * 0.5;
}



vector<P> v;
double calc(double x) {
	L a(P(x, -1e3), P(x, 1e3)), b(P(x+0.5, -1e3), P(x+0.5, 1e3));

	vector<double> ap, bp;
	rep(i, sz(v)) {
		L l = L(v[i], v[(i+1)%sz(v)]);
		if (!(intersectSS(l, a) && intersectSS(l, b))) continue;
		ap.push_back(imag(crosspoint(l, a)));
		bp.push_back(imag(crosspoint(l, b)));
	}
	sort(ap.begin(), ap.end());
	sort(bp.begin(), bp.end());
	assert(sz(ap) % 2 == 0);
	double S = 0;
	rep(i, sz(ap)/2) {
		S +=  (ap[i*2 + 1] - ap[i*2] + bp[i*2 + 1] - bp[i*2]) * 0.5 * 0.5;
	}
	return S;
}
int main(int argc, char *argv[]) {
	int N;
	cin >> N;
	rep(i, N) {
		double x, y;
		cin >> x >> y;
		v.push_back(P(x, y));
	}

	double s0 = 0, s1 = 0;
	for(double x = -1e3; x <= 1e3; x += 1) {
		s0 += calc(x);
	}
	for(double x = -1e3 - 0.5; x <= 1e3; x += 1) {
		s1 += calc(x);
	}

	printf("%.20f\n", max(s0, s1));
}
