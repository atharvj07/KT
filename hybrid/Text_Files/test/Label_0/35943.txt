import java.util.*;
public class Test{
    
    static int A[];
    public static void main(String args[]){
        Scanner in=new Scanner(System.in);
        Test a=new Test();
        int k=in.nextInt();
        int n=in.nextInt();
        A=new int[n];
        for(int i=0;i<n;i++)A[i]=in.nextInt();
        a.bubbleSort();
        int min=22222,temp=0;
        for(int i=0;i<=((n-k));i+=1){
            temp=A[(k-1)+i]-A[i];
            if(min>temp)min=temp;
        }
        System.out.println(min);
    }
    
    public void bubbleSort() {
    boolean sorted = false;
    int temp;
    while(!sorted) {
        sorted = true;
        for (int i = 0; i < A.length - 1; i++) {
            if (A[i] > A[i+1]) {
                temp = A[i];
                A[i] = A[i+1];
                A[i+1] = temp;
                sorted = false;
            }
        }
    }
}
    
}
