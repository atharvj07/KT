a = int(input())
c = list()
for i in range(a + 1):
	c.append(list())
for i in range(a - 1):
	(g, h) = map(int, input().split())
	c[g].append(h)
	c[h].append(g)
q = list()
p = 0
k = 1
q.append(1)
odw = list()
wyn = 0
for i in range(a + 1):
	odw.append(0)
while k > p:
	x = q[p]
	chk = 0
	if x < 0:
		x = -x
		wyn += 1
		chk = 1
	p += 1
	odw[x] = 1
	for s in c[x]:
		if odw[s] == 0:
			q.append(s if chk > 0 else -s)
			k += 1
			odw[s] = 1
wyn = wyn - 1
if wyn > a - (wyn + 1) - 1:
	wyn = a - wyn - 2
print(wyn)
