	int nPoints;
	Point[] points;
	
	MinMaxQueue minMaxQueue;

	void solve() throws IOException  {
		nPoints = nextInt();
		points = new Point [nPoints];
		for (int i = 0; i < nPoints; i++) {
			long x = nextInt();
			long y = nextInt();
			points[i] = new Point(i, x + y, x - y);
		}
		sort(points);
		minMaxQueue = new MinMaxQueue(nPoints);
		long left = 0L;
		long right = 4 * 1000000000L;
		long answer = Long.MAX_VALUE;
		while (left <= right) {
			long middle = (left + right) >> 1;
			if (can(middle)) {
				right = middle - 1;
				answer = min(answer, middle);
			} else {
				left = middle + 1;
			}
		}
		out.println(0.5 * answer);
	}
	
	boolean can(long radius) {
		minMaxQueue.clear();
		for (Point p : points)
			minMaxQueue.push(p);
		for (int i = 0, j = 0; i < nPoints; i++) {
			minMaxQueue.pop();
			while (j < i && points[i].x - points[j].x > radius)
				minMaxQueue.push(points[j++]);
			if (minMaxQueue.isEmpty() || minMaxQueue.getMax() - minMaxQueue.getMin() <= radius)
				return true;
		}
		return false;
	}

	class Point implements Comparable<Point> {
		int id;
		long x, y;

		Point(int id, long x, long y) {
			this.id = id;
			this.x = x;
			this.y = y;
		}

		@Override
		public int compareTo(Point p) {
			return Long.compare(x, p.x);
		}
	}
	
	class MinMaxStack {
		Point[] points;
		long[] minY;
		long[] maxY;
		int nElements = 0;
		
		MinMaxStack(int size) {
			points = new Point [size];
			minY = new long [size];
			maxY = new long [size];
		}
		
		void push(Point p) {
			points[nElements] = p;
			minY[nElements] = min(p.y, getMin());
			maxY[nElements] = max(p.y, getMax());
			nElements++;
		}

		Point pop() {
			return points[--nElements]; 
		}
		
		long getMin() {
			return isEmpty() ? Long.MAX_VALUE : minY[nElements - 1];
		}
		
		long getMax() {
			return isEmpty() ? Long.MIN_VALUE : maxY[nElements - 1];
		}
		
		void clear() {
			nElements = 0;
		}

		boolean isEmpty() {
			return nElements == 0;
		}
	}
	
	class MinMaxQueue {
		MinMaxStack headStack;
		MinMaxStack tailStack;
		
		MinMaxQueue(int size) {
			headStack = new MinMaxStack(size);
			tailStack = new MinMaxStack(size);
		}
		
		void clear() {
			tailStack.clear();
			headStack.clear();
		}

		boolean isEmpty() {
			return tailStack.isEmpty() && headStack.isEmpty();
		}

		long getMin() {
			return min(headStack.getMin(), tailStack.getMin());
		}
		
		long getMax() {
			return max(headStack.getMax(), tailStack.getMax());
		}
		
		void push(Point p) {
			tailStack.push(p);
		}
		
		Point pop() {
			if (headStack.isEmpty())
				while (!tailStack.isEmpty())
					headStack.push(tailStack.pop());
			return headStack.pop();
		}
	}

		while (!st.hasMoreTokens()) {

		}
		return false;

	static void swap(int[] a, int i, int j) {
		int tmp = a[i];
		a[i] = a[j];
		a[j] = tmp;
	}
	
	static void swap(long[] a, int i, int j) {
		long tmp = a[i];

	static void shuffle(int[] a, int from, int to) {
		for (int i = from; i < to; i++)
			swap(a, i, rnd.nextInt(a.length));
	}

	}
