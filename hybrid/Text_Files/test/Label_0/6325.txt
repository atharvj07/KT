#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int mod=1e9+7;
int a[100010],b[100010],t[100010],m[100010],v[100010];
int main() {
    int n;
    cin>>n;
    int l,r,mm=0;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++) cin>>b[i];
    ll ans=1;
    for(int i=1;i<=n;i++){
        if((a[i]>a[i-1]&&b[i]<a[i])||(b[i]>b[i+1]&&b[i]>a[i])) ans=0;
        else if(a[i]==a[i-1]&&b[i]==b[i+1])ans=ans*min(a[i],b[i])%mod;
    }
    cout<<ans<<endl;
	return 0;
}
