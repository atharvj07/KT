s = input()
n = int(input())
tests = []
for _ in range(n):
	(a, b) = map(int, input().split())
	a -= 1
	b -= 1
	tests.append((a, b))
count_left = {}
count_left[-1] = (0, 0, 0)
(x, y, z) = (0, 0, 0)
for (i, c) in enumerate(s):
	if c == 'x':
		x += 1
	elif c == 'y':
		y += 1
	elif c == 'z':
		z += 1
	count_left[i] = (x, y, z)

def count(a, b):
	(x2, y2, z2) = count_left[b]
	(x1, y1, z1) = count_left[a - 1]
	x = x2 - x1
	y = y2 - y1
	z = z2 - z1
	return (x, y, z)

def is_valid(s, a, b):
	if b - a < 2:
		return True
	(x, y, z) = count(a, b)
	diffs = map(abs, [x - y, x - z, y - z])
	return not any((d > 1 for d in diffs))
result = []
for (a, b) in tests:
	result.append('YES' if is_valid(s, a, b) else 'NO')
print('\n'.join(result))
