#include <bits/stdc++.h>
const long long MOD = 1000003;
using namespace std;
int main() {
  string s;
  cin >> s;
  long long int ans = 0, m, p = 1;
  for (int i = s.length() - 1; i >= 0; --i) {
    if (s[i] == '>') m = 8;
    if (s[i] == '<') m = 9;
    if (s[i] == '+') m = 10;
    if (s[i] == '-') m = 11;
    if (s[i] == '.') m = 12;
    if (s[i] == ',') m = 13;
    if (s[i] == '[') m = 14;
    if (s[i] == ']') m = 15;
    ans = (ans + (m * p) % MOD) % MOD;
    p = (p * 16) % MOD;
  }
  cout << ans << endl;
  return 0;
}
