#include<bits/stdc++.h>
using namespace std;
using Int = long long;
Int k;
double ans;

template<typename T1,typename T2> void chmax(T1 &a,T2 b){if(a<b) a=b;};

vector<Int> pr;

Int isprime(Int x){
  if(x<=1) return 0;
  for(Int i=2;i*i<=x;i++)
    if(x%i==0) return 0;
  return 1;
}

void dfs(Int x,double y,Int p,Int c){
  //if(x<=k) cout<<y<<" "<<x<<":"<<(y/x)<<endl; 
  if(x<=k) chmax(ans,y/x);
  else return;
  Int t=1;
  double s=0;
  if(c==0) return;
  for(Int i=0;i<=c;i++){
    if(x>k/t) break;
    s+=t;
    dfs(x*t,y*s,p+1,i);
    if(t>k/pr[p]) break; 
    t*=pr[p];
  }
} 

struct Precision{
  Precision(){
    cout<<fixed<<setprecision(6);
  }
}precision_beet;

signed main(){  
  cin.tie(0);
  ios::sync_with_stdio(0);
  
  
  for(Int i=2;i<1e5;i++) if(isprime(i)) pr.emplace_back(i);
  
  while(cin>>k,k){
    ans=1;
    dfs(1,1,0,100);
    cout<<ans<<endl;
  } 
  return 0;
}
