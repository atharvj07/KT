for _ in range(int(input())):
	(n, m, a, b) = map(int, input().split())
	s = [int(i) for i in input().split()]
	s.sort()
	s.append(0)
	if a > b:
		a = n - a + 1
		b = n - b + 1
	num_of_possible_firecrackers = b - a - 1
	seconds_to_guard_arrive = b - 1
	ind_to_start = -1
	ind = 0
	while ind < m:
		if s[ind] + 1 <= seconds_to_guard_arrive:
			if s[ind] >= s[ind + 1]:
				s[ind + 1] = s[ind] + 1
			ind_to_start = ind
		else:
			break
		ind += 1
	if ind_to_start == -1:
		print(0)
	elif len(s[:ind_to_start + 1]) <= num_of_possible_firecrackers:
		print(len(s[:ind_to_start + 1]))
	else:
		print(num_of_possible_firecrackers)
