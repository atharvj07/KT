def check_prime(n):
	for i in range(2, int(sqrt(n)) + 1):
		if n % i == 0:
			return False
	return True
ba = []
for i in range(2, 10 ** 5):
	if check_prime(i):
		ba.append(i)
		if len(ba) == 1000:
			break
(n, m) = map(int, input().split())
hash = defaultdict(set)
cnt = n - 1
for i in range(1, n):
	hash[i].add(i + 1)
for i in range(3, n + 1):

	cnt += 1
if n <= 2500:

		for j in range(i + 1, n + 1):
			if gcd(i, j) == 1:
				if j not in hash[i]:
					hash[i].add(j)
					cnt += 1
	ha = 0

				ha += 1
