from sys import stdin, stdout

def input():
	return stdin.readline().strip()
MOD = 1000000007
(n, m) = map(int, input().split())
a = list(map(int, input().split()))
ans = []
for i in range(n):
	x = a[:i]
	x.sort(reverse=True)
	tot = sum(x)
	count = 0
	for j in x:
		if tot <= m - a[i]:
			break
		tot -= j
		count += 1
	ans.append(count)
print(*ans)
