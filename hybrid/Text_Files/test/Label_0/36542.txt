class ProblemSolver {
    int N, M, K;
    char S[], P[];
    int A[][] = new int[256][256];
    TreeMap<Integer, Character> T = new TreeMap<>();

    void solve(FastReader in, PrintWriter out) {
        N = in.nextInt();
        M = in.nextInt();
        K = in.nextInt();
        S = in.nextToken().toCharArray();

        T.put(0, '0');
        for (int i = 0; i < S.length; i++) {
            if (i + 1 < N) A[S[i]][S[i + 1]] += 1;
            T.put(i + 1, S[i]);
        }
        T.put(S.length + 1, '1');

        for (int m = 0; m < M; m++) {
            int t = in.nextInt();
            if (t == 1) {
                int a = in.nextInt(), b = in.nextInt();
                char c = in.nextToken().charAt(0);
                int la = T.lowerKey(a), lb = T.floorKey(b), rb = T.higherKey(b);
                if (b + 1 < rb) {
                    T.put(b + 1, T.get(lb));
                    rb = b + 1;
                }
                Integer i1 = la;
                for (Integer i2 : T.subMap(la, false, rb, true).keySet()) {
                    A[T.get(i1)][T.get(i2)]--;
                    i1 = i2;
                }
                Iterator<Map.Entry<Integer, Character>> it = T.subMap(la, false, rb, false).entrySet().iterator();
                while (it.hasNext()) {
                    it.next();
                    it.remove();
                }
                T.put(a, c);
                A[T.get(la)][c]++;
                A[c][T.get(rb)]++;
                //out.print(a + " " + b + " " + c + ": ");
                //for (Map.Entry<Integer, Character> e : T.entrySet()) {
                //    out.print(e.getKey() + "=" + e.getValue() + " ");
                //}
                //out.println();
            } else {
                P = in.nextToken().toCharArray();
                int res = N;
                for (int i = 0; i < P.length; i++) for (int k = i + 1; k < P.length; k++) res -= A[P[i]][P[k]];
                // bbbbbaaaac

                // ebbbcebbea
                // ebbbaaaaaa
                // bbbbaaaaaa
                // bbbdddddda
                // bbbeeeeeea
                // bbbebbbbba
                // bbbebaaaaa
                // bbccbaaaaa
                // bbbcbaaaaa
                out.println(res);
            }
        }
    }
}

class Pair implements Comparable<Pair> {
    public int a;
    public int b;

    public Pair(int a, int b) {
        this.a = a;
        this.b = b;
    }

    public int compareTo(Pair o) {
        if (this.a < o.a) return -1;
        if (this.a > o.a) return 1;
        if (this.b == o.b) return 0;
        return (int) ((this.b - o.b) / Math.abs(this.b - o.b));
    }
}


class FastReader extends BufferedReader {
    private StringTokenizer st;

    public FastReader(InputStream is) {
        super(new InputStreamReader(is));
    }

    public String nextToken() {
        try {
            while (st == null || !st.hasMoreElements()) {
                st = new StringTokenizer(readLine());
            }
            return st.nextToken();
        } catch (IOException e) {
            return null;
        }
    }

    public int nextInt() {
        return Integer.parseInt(nextToken());
    }

    public long nextLong() {
        return Long.parseLong(nextToken());
    }

    public int[] nextArrayInt(int size) {
        int[] ret = new int[size];
        for (int i = 0; i < size; i++) {
            ret[i] = nextInt();
        }
        return ret;
    }
}
