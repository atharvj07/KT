#include <bits/stdc++.h>
using namespace std;
const int N = 105;
char s[N];
vector<int> v;
int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    scanf("%s", s);
    for (int i = 0; s[i]; i++) {
      if (s[i] == '1') {
        int cnt = 1;
        while (s[i + 1] == '1') i++, cnt++;
        v.push_back(cnt);
      }
    }
    sort(v.begin(), v.end(), [&](int x, int y) { return x > y; });
    int res = 0;
    for (int i = 0; i < v.size(); i += 2) res += v[i];
    printf("%d\n", res);
    vector<int>().swap(v);
  }
  return 0;
}
