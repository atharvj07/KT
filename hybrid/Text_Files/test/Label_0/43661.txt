q = int(input())
pos = {}
(left, right) = (0, 1)
for i in range(q):
	data = input().split()
	command = data[0]
	x = int(data[1])
	if command == 'L':
		pos[x] = left
		left -= 1
	if command == 'R':
		pos[x] = right
		right += 1
	if command == '?':
		real_pos = pos[x] - (left + 1)
		n = right - left - 1
		print(min(real_pos, n - 1 - real_pos))
