from sys import stdin
test = int(stdin.readline())
for t in range(test):
	nums = int(stdin.readline())
	if nums % 2 == 0:
		print(0)
	elif int(str(nums)[0]) % 2 == 0:
		print(1)
	else:
		check = True
		for i in str(nums):
			if int(i) % 2 == 0:
				check = False
		if check:
			print(-1)
		else:
			print(2)
