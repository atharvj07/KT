def gcd(a, b):
	c = 0
	while b > 0:
		c = a % b
		a = b
		b = c
	return a
arr = list(map(int, input().split()))
n = arr[0]
m = arr[1]
if m < n - 1:
	print('Impossible')
	exit()
vec = []
for i in range(n):
	for j in range(i + 1, n):
		if gcd(int(i + 1), int(j + 1)) == 1:
			m -= 1
			vec.append({i + 1, j + 1})
		if m == 0:
			break
	if m == 0:
		break
if m > 0:
	print('Impossible')
	exit()
print('Possible')
for (x, y) in vec:
	print('%d %d' % (x, y))
