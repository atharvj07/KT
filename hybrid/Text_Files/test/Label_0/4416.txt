#include <bits/stdc++.h>
using namespace std;
template <typename T>
inline bool chkmin(T &x, T y) {
  return x > y ? x = y, 1 : 0;
}
template <typename T>
inline bool chkmax(T &x, T y) {
  return x < y ? x = y, 1 : 0;
}
inline void read(int &x) {
  char ch = getchar();
  int f = 1;
  x = 0;
  while (ch < '0' || ch > '9') {
    if (ch == '-') f = -1;
    ch = getchar();
  }
  while (ch >= '0' && ch <= '9') x = x * 10 + ch - '0', ch = getchar();
  x *= f;
}
int n, m, k;
long long curt;
struct node {
  long long arr;
  int dest, num;
  bool operator<(const node &x) const { return arr > x.arr; }
};
priority_queue<node> going;
priority_queue<pair<long long, int> > waiting[200050];
set<int> se;
pair<long long, int> get(int pos) { return waiting[pos].top(); }
void del(int pos, long long time, int dest) {
  int w = -waiting[pos].top().second;
  going.push(node{time, dest, w});
  waiting[pos].pop();
  if (waiting[pos].empty()) se.erase(pos);
}
int main() {
  int x, y;
  long long t;
  read(n), read(k), read(m);
  se.insert(-1e9), se.insert(1e9);
  for (register int i = 1; i <= k; ++i)
    read(x), waiting[x].push(make_pair(0ll, -i)), se.insert(x);
  while (m--) {
    scanf("%lld", &t), read(x), read(y);
    if (se.size() == 2) curt = going.top().arr;
    curt = max(curt, t);
    while (!going.empty() && going.top().arr <= curt) {
      node now = going.top();
      going.pop();
      waiting[now.dest].push(make_pair(-now.arr, -now.num));
      se.insert(now.dest);
    }
    int nx = *se.lower_bound(x), pr = *--se.lower_bound(x);
    pair<long long, int> c1, c2;
    long long anst;
    int ansn;
    if (nx - x == x - pr) {
      c1 = get(nx), c2 = get(pr);
      if (c1 > c2)
        anst = max(-c1.first, curt) + nx - x - t, ansn = -c1.second,
        del(nx, anst + t + abs(y - x), y);
      else
        anst = max(-c2.first, curt) + x - pr - t, ansn = -c2.second,
        del(pr, anst + t + abs(y - x), y);
    } else if (nx - x < x - pr)
      c1 = get(nx), anst = max(-c1.first, curt) + nx - x - t, ansn = -c1.second,
      del(nx, anst + t + abs(y - x), y);
    else
      c2 = get(pr), anst = max(-c2.first, curt) + x - pr - t, ansn = -c2.second,
      del(pr, anst + t + abs(y - x), y);
    cout << ansn << ' ' << anst << endl;
  }
  return 0;
}
