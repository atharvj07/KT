#include <bits/stdc++.h>
using namespace std;
int dina[2002];
string filteredCompa;
int calculateToffees(int i) {
  if (dina[i] >= 1) {
    return dina[i];
  }
  dina[i] = 1;
  if (i > 0) {
    if (filteredCompa[i - 1] == 'R') {
      dina[i] = max(dina[i], calculateToffees(i - 1) + 1);
    }
  }
  if (i < filteredCompa.size()) {
    if (filteredCompa[i] == 'L') {
      dina[i] = max(dina[i], calculateToffees(i + 1) + 1);
    }
  }
  return dina[i];
}
int main() {
  int n, i, j;
  string compa;
  cin >> n;
  getline(cin, compa);
  getline(cin, compa);
  for (i = 0; i < compa.size(); i++) {
    if (compa[i] != '=') {
      filteredCompa += compa[i];
    }
  }
  for (i = 0; i <= filteredCompa.size(); i++) {
    calculateToffees(i);
  }
  for (i = 0, j = 0; i <= compa.size(); i++) {
    if (j <= filteredCompa.size()) {
      cout << dina[j] << " ";
    } else {
      cout << "1 ";
    }
    if (compa[i] == filteredCompa[j]) {
      j++;
    }
  }
  cout << endl;
  return 0;
}
