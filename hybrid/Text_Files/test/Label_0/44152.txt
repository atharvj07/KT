n = int(input())
a = list(map(int, input().split()))
result = 0
for i in range(n):
	if i < n - 2:
		if a[i] == 1 and a[i + 2] == 1:
			for j in range(i + 2, n, 2):
				if a[j - 1] == 1 or a[j] == 0:
					result += ((j - i - 1) // 2 + 1) // 2
					break
				elif j >= n - 2:
					result += ((j - i + 1) // 2 + 1) // 2
					a[j] = 0
					break
				a[j] = 0
print(result)
