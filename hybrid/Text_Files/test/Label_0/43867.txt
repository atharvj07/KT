(n, A) = map(int, input().split())
ds = list(map(int, input().split()))
total = sum(ds)
others_max = [total - d for d in ds]
others_min = n - 1
mins = [max(A - o - 1, 0) for o in others_max]
for i in range(n):
	if ds[i] > A - others_min:
		mins[i] += ds[i] - (A - others_min)
print(' '.join(map(str, mins)))
