import java.io.*;
import java.util.*;
import java.math.*;

public class SkyFullOfStar
{
    static IO io = new IO (System.in);
    static int n, m;
    static Cood[] curr, target;

    public static void main(String[] args) throws Exception
    {
        int i, j, k;
        n = io.nextInt();   m = io.nextInt();
        curr = new Cood[m];
        target = new Cood[m];

        for(i = 0; i < m; ++i)
            curr[i] = new Cood(i, io.nextInt(), io.nextInt());
        for(i = 0; i < m; ++i)
            target[i] = new Cood(i, io.nextInt(), io.nextInt());
        if(n == 2)
        {
            if(m == 1)
                guide(curr[0], target[0].x, target[0].y);
            else {
                if (guide(curr[0], target[0].x, target[0].y))
                    guide(curr[1], target[1].x, target[1].y);
                else if (guide(curr[1], target[1].x, target[1].y))
                    guide(curr[0], target[0].x, target[0].y);
                else {
                    guide(curr[0], curr[0].x % 2 + 1, curr[0].y);
                    guide(curr[1], target[1].x, target[1].y);
                    guide(curr[0], target[0].x, target[0].y);
                }
            }

            io.flush();
            System.exit(0);
        }

        int targetRow = findBestRow();
        for(j = 1; j <= n; ++j)
        {
            if(contains(targetRow, j, -1, curr))
                continue;
            for(k = 0; k < m; ++k)
                if(curr[k].x != targetRow && guide(curr[k], targetRow, j))
                    break;
            io.println();
        }

        io.print("************************\n\n");

        for(j = 1; j <= n; ++j)
        {
            for(i = 0; i < m; ++i)
                if(target[i].x == targetRow && target[i].y == j)
                    break;
            if(i >= m)
                continue;

            for(k = 0; k < m; ++k)
                if(curr[k].x == targetRow && curr[k].y == j)
                    break;
            if(k < m)
                swap(curr[k], curr[i], true);
            else
                guide(curr[i], target[i].x, target[i].y);
            io.println();
        }

        io.print("************************\n\n");

        for(i = 1; i < targetRow; ++i)
        {
            for(j = 1; j <= n; ++j)
            {
                for(k = 0; k < m; ++k)
                    if(target[k].x == i && target[k].y == j)
                        break;
                if (k >= m)
                    continue;
                guide(curr[k], target[k].x, target[k].y);
                io.println();
            }
        }
        for(i = n; i > targetRow; --i)
        {
            for(j = 1; j <= n; ++j)
            {
                for(k = 0; k < m; ++k)
                    if(target[k].x == i && target[k].y == j)
                        break;
                if (k >= m)
                    continue;
                guide(curr[k], target[k].x, target[k].y);
                io.println();
            }
        }

        io.flush();
    }
    static int findBestRow()
    {
        int i, j, minC = Integer.MAX_VALUE, count, bestR = -7;
        for(i = 2; i < n - 1; ++i) {
            for (j = 1, count = 0; j <= n; ++j) {
                if (contains(i - 1, j, -1, target))
                    ++count;
                if (contains(i + 1, j, -1, target))
                    ++count;
            }
            if(count < minC)
            {
                minC = count;
                bestR = i;
            }
        }
        return bestR;
    }
    static void swap(Cood c1, Cood c2, boolean rowWise)
    {
        if(c1.equals(c2)) return;
        int x1 = c1.x, y1 = c1.y, x2 = c2.x, y2 = c2.y;

        if(rowWise)
        {
            guide(c1, c1.x + 1, c1.y);
            guide(c2, c2.x - 1, c2.y);
        }
        else
        {
            guide(c1, c1.x, c1.y + 1);
            guide(c2, c2.x, c2.y - 1);
        }
        guide(c1, x2, y2);
        guide(c2, x1, y1);
    }

    static HashSet<Integer> vis = new HashSet<>();
    static ArrayList<String> path = new ArrayList<>();
    static boolean guideO(Cood c, int x, int y)
    {
        vis.clear();
        path.clear();
        if(!travelO(c.x, c.y, x, y, c.ind))
            return false;
        for(int i = 1; i < path.size(); ++i)
            io.println(path.get(i - 1) + " " + path.get(i));
        c.x = x;    c.y = y;
        return true;
    }
    static boolean travelO(int cx, int cy, int tx, int ty, int exclInd)
    {
        if(invalid(cx, cy) || vis.contains(cx * (n + 5) + cy) || contains(cx, cy, exclInd, curr))
            return false;
        vis.add(cx * (n + 5) + cy);
        path.add(cx + " " + cy);

        if(cx == tx && cy == ty)
            return true;

        if(travelO(cx - 1, cy, tx, ty, exclInd))
            return true;
        if(travelO(cx + 1, cy, tx, ty, exclInd))
            return true;
        if(travelO(cx, cy - 1, tx, ty, exclInd))
            return true;
        if(travelO(cx, cy + 1, tx, ty, exclInd))
            return true;

        path.remove(path.size() - 1);
        return false;
    }

    static boolean guide(Cood c, int x, int y)
    {
        vis.clear();
        path.clear();
        if(!travel(c.x, c.y, x, y, c.ind))
            return false;
        for(int i = 1; i < path.size(); ++i)
            io.println(path.get(i - 1) + " " + path.get(i));
        c.x = x;    c.y = y;
        return true;
    }
    static boolean travel(int cx, int cy, int tx, int ty, int exclInd)
    {
        if(invalid(cx, cy) || vis.contains(cx * (n + 5) + cy) || contains(cx, cy, exclInd, curr))
            return false;
        vis.add(cx * (n + 5) + cy);
        path.add(cx + " " + cy);

        if(cx == tx && cy == ty)
            return true;

        int currDist = dist(cx, cy, tx, ty);
        if(dist(cx - 1, cy, tx, ty) < currDist && travel(cx - 1, cy, tx, ty, exclInd))
            return true;
        if(dist(cx + 1, cy, tx, ty) < currDist && travel(cx + 1, cy, tx, ty, exclInd))
            return true;
        if(dist(cx, cy - 1, tx, ty) < currDist && travel(cx, cy - 1, tx, ty, exclInd))
            return true;
        if(dist(cx, cy + 1, tx, ty) < currDist && travel(cx, cy + 1, tx, ty, exclInd))
            return true;

        path.remove(path.size() - 1);
        return false;
    }

    static boolean invalid(int x, int y)
    {
        return x < 1 || y < 1 || x > n || y > n;
    }
    static int dist(int a, int b, int c, int d)
    {
        return Math.abs(a - c) + Math.abs(b - d);
    }

    static boolean contains(int x, int y, int exclInd, Cood[] ar)
    {
        int i;
        for(i = 0; i < ar.length; ++i)
            if(i != exclInd && ar[i].x == x && ar[i].y == y)
                return true;
        return false;
    }
}

class Cood
{
    int ind, x, y;
    Cood(int in, int i, int j) {ind = in;  x = i;  y = j;}
    boolean isEqual(Cood co) {return x == co.x && y == co.y;}
    public String toString() {return ind + ".(" + x + ", " + y + ")";}
}

class IO {
    static byte[] buf = new byte[2048];
    static int index, total;
    static InputStream in;
    static StringBuilder sb = new StringBuilder();
    static int counter = 0;


    IO(InputStream is) {
        in = is;
    }

    int scan() throws Exception {
        if (index >= total) {
            index = 0;
            total = in.read(buf);
            if (total <= 0)
                return -1;
        }
        return buf[index++];
    }

    String next() throws Exception {
        int c;
        for (c = scan(); c <= 32; c = scan()) ;
        StringBuilder sb = new StringBuilder();
        for (; c > 32; c = scan())
            sb.append((char) c);
        return sb.toString();
    }

    int nextInt() throws Exception {
        int c, val = 0;
        for (c = scan(); c <= 32; c = scan()) ;
        boolean neg = c == '-';
        if (c == '-' || c == '+')
            c = scan();
        for (; c >= '0' && c <= '9'; c = scan())
            val = (val << 3) + (val << 1) + (c & 15);
        return neg ? -val : val;
    }

    long nextLong() throws Exception {
        int c;
        long val = 0;
        for (c = scan(); c <= 32; c = scan()) ;
        boolean neg = c == '-';
        if (c == '-' || c == '+')
            c = scan();
        for (; c >= '0' && c <= '9'; c = scan())
            val = (val << 3) + (val << 1) + (c & 15);
        return neg ? -val : val;
    }

    void print(Object a) {
        //sb.append(a.toString());
    }

    void println(Object a) {
        sb.append(a.toString()).append("\n");
        ++counter;
    }

    void println() {
        //sb.append("\n");
    }

    void flush() {
        System.out.println(counter);
        System.out.print(sb);
        sb = new StringBuilder();
    }
}
