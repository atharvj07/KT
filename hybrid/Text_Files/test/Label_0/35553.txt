import java.util.*;

public class Project
{
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        char[] str = sc.next().toCharArray();
        int max = 0;
        int lastI = 0;

        for (int i = 0; i < str.length; i++) {
            if (alphVowels(str[i])) {
                if (max < i - lastI + 1) {
                    max = i - lastI + 1;
                }
                lastI = i + 1;
            }
        }

        if (max < str.length - lastI + 1) {
            System.out.println(str.length - lastI + 1);
        } else {
            System.out.println(max);
        }
    }

    private static boolean alphVowels (char letter) {
        char[] vowels = {'A', 'E', 'I', 'O', 'U', 'Y'};
        for (char ch : vowels) {
            if (ch == letter) return true;
        }
        return false;
    }
}
