#include "bits/stdc++.h"

using namespace std;

typedef long double ld;
const ld eps = 1e-9;
const int INF = numeric_limits<int>::max()/2;

struct Event{
    int s1,s2;
    ld time;
    Event(int s1,int s2,ld time): s1(s1),s2(s2),time(time){}
};

bool operator < (const Event &e, const Event &f){
    return e.time < f.time;
}

struct Star{
    ld x,y,z,vx,vy,vz,r,vr;
    Star(ld x, ld y, ld z, ld vx, ld vy, ld vz,ld r,ld vr):x(x),y(y),z(z),vx(vx),vy(vy),vz(vz),r(r),vr(vr){}
};

ld check(const Star &s1,const Star &s2){
    ld x=s1.x-s2.x, y=s1.y-s2.y, z=s1.z-s2.z, r=s1.r+s2.r;
    ld vx=s1.vx-s2.vx, vy=s1.vy-s2.vy, vz=s1.vz-s2.vz,vr=s1.vr+s2.vr;
    ld a=vx*vx+vy*vy+vz*vz-vr*vr;
    ld b=x*vx+y*vy+z*vz+r*vr;
    ld c=x*x+y*y+z*z-r*r;

    if(b*b-a*c+eps>0){
        ld res1=(-b+sqrt(b*b-a*c))/a;
        ld res2=(-b-sqrt(b*b-a*c))/a;
        //cout<<res1<<" "<<res2<<endl;
        ld res=INF;
        if(res1+eps>0 && s1.r-s1.vr*res1+eps>0 && s2.r-s2.vr*res1+eps>0) res=min(res,res1);
        if(res2+eps>0 && s1.r-s1.vr*res2+eps>0 && s2.r-s2.vr*res2+eps>0) res=min(res,res2);
       // cout<<res<<endl;
        if(res!=INF) return res;
    }
    return -1;
}

int main(){
    cin.tie(0);
    ios::sync_with_stdio(false);

    int n;
    while(cin>>n,n){
        vector<Star> stars;
        vector<Event> es;
        vector<int> used(n);
        vector<ld> res(n);
        for(int i=0;i<n;i++){
            ld x,y,z,vx,vy,vz,r,vr;
            cin>>x>>y>>z>>vx>>vy>>vz>>r>>vr;
            stars.push_back(Star(x,y,z,vx,vy,vz,r,vr));
            es.push_back(Event(i,i,r/vr));
        }
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                ld ret=check(stars[i],stars[j]);
                //cout<<i<<" "<<j<<" "<<ret<<endl;
                if(ret!=-1) es.push_back(Event(i,j,ret));
            }
        }
        int sz=es.size();
        sort(es.begin(),es.end());
        for(int i=0;i<sz;i++){
            int s1=es[i].s1,s2=es[i].s2;
            if(used[s1]||used[s2]) continue;
            res[s1]=es[i].time;res[s2]=es[i].time;
            used[s1]=1;used[s2]=1;
        }
        for(int i=0;i<n;i++) cout<<fixed<<setprecision(10)<<res[i]<<endl;
    }
}
