#include <bits/stdc++.h>
using namespace std;
const int N = 110001;
const int B = 23333;
const int BB = 217;
const long long p = 1000000007;
char s[N * 10], ch[N];
int n, k, m, tl, sta[N], top;
long long a[N * 10], mi[N];
unsigned long long aa[N * 10], mii[N];
map<pair<long long, unsigned long long>, int> ma;
bool vis[N];
inline long long get(int l, int r) {
  return ((a[r] - a[l - 1] * mi[r - l + 1] % p) % p + p) % p;
}
inline unsigned long long gett(int l, int r) {
  return (aa[r] - aa[l - 1] * mii[r - l + 1]);
}
inline void print(bool x) {
  if (x) {
    puts("YES");
    for (int i = 1; i <= top; ++i) printf("%d ", sta[i]);
    puts("");
  } else
    puts("NO");
  exit(0);
}
int main() {
  scanf("%d%d", &n, &k);
  tl = n * k;
  scanf("%s", s + 1);
  int i, j, x, have;
  long long tmp;
  unsigned long long tmpp;
  for (i = 1; i <= tl; ++i)
    a[i] = (a[i - 1] * B % p + s[i]) % p, aa[i] = aa[i - 1] * BB + s[i];
  for (i = 1; i < k; ++i)
    a[tl + i] = (a[tl + i - 1] * B % p + s[i]) % p,
           aa[tl + i] = (aa[tl + i - 1] * BB + s[i]);
  for (mi[0] = 1, mii[0] = 1, i = 1; i <= k; ++i)
    mi[i] = mi[i - 1] * B % p, mii[i] = mii[i - 1] * BB;
  scanf("%d", &m);
  for (i = 1; i <= m; ++i) {
    scanf("%s", ch + 1);
    tmp = 0, tmpp = 0;
    for (j = 1; j <= k; ++j)
      tmp = (tmp * B + ch[j]) % p, tmpp = (tmpp * BB + ch[j]);
    ma[make_pair(tmp, tmpp)] = i;
  }
  for (i = 1; i <= k; ++i) {
    top = 0, have = 0;
    for (j = i; j <= tl; j += k) {
      x = ma[make_pair(get(j, j + k - 1), gett(j, j + k - 1))];
      if (x && !vis[x])
        sta[++top] = x, vis[x] = 1, ++have;
      else {
        while (top) vis[sta[top]] = 0, --top;
        break;
      }
    }
    if (have == n) print(1);
  }
  print(0);
}
