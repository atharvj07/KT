import sys, os, io, time, copy, math
from functools import lru_cache
if os.path.exists('input.txt'):
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')

def main():
	(n, m) = map(int, input().split())
	if n < m:
		for i in range(n):
			print('GB', end='')
		for i in range(m - n):
			print('G', end='')
		print('')
	elif n > m:
		for i in range(m):
			print('BG', end='')
		for i in range(n - m):
			print('B', end='')
		print('')
	else:
		for i in range(n):
			print('GB', end='')
		print('')
main()
