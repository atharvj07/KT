#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  vector<long long> a;
  for (long long i = 1; i <= sqrt(n); i++) {
    if (n % i == 0) {
      if (i * i == n) {
        a.push_back(((double)(2 + i * ((n / i))) / 2) * ((n / i) + 1) -
                    (n + 1));
      } else {
        a.push_back(((double)(2 + i * ((n / i))) / 2) * ((n / i) + 1) -
                    (n + 1));
        long long p = n / i;
        a.push_back(((double)(2 + p * ((n / p))) / 2) * ((n / p) + 1) -
                    (n + 1));
      }
    }
  }
  sort(a.begin(), a.end());
  for (auto i : a) {
    cout << i << ' ';
  }
}
