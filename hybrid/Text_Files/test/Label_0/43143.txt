from bisect import bisect_left, bisect_right

def solve(n, k, depots):
	distance = 0
	depots.sort()
	loc1 = bisect_left(depots, 0)
	loc2 = bisect_right(depots, 0)
	j = n - 1
	while j >= loc2:
		distance += 2 * depots[j]
		j -= k
	j = 0
	while j < loc1:
		distance -= 2 * depots[j]
		j += k
	if loc2 < n and loc1 > 0:
		distance -= max(-depots[0], depots[-1])
	elif loc2 < n:
		distance -= depots[-1]
	elif loc1 > 0:
		distance += depots[0]
	return distance
ans = []
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	ans.append(solve(n, k, list(map(int, input().split()))))
print(*ans, sep='\n')
