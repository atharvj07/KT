#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  cin >> n;
  string s = "";
  for (int i = 0; i < n; i++) {
    int k;
    cin >> k;
    s += k - '0';
  }
  string ss = s + s;
  int ans = 0;
  int _counter = 0;
  for (int i = 0; i < (2 * n); i++) {
    if (ss[i] == 1 - '0') {
      _counter++;
      ans = max(ans, _counter);
    } else
      _counter = 0;
  }
  cout << ans << endl;
  return 0;
}
