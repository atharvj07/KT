import sys
sys.stderr = sys.stdout
from math import gcd

def primes(m):
	S = [True] * (m + 1)
	S[0] = S[1] = False
	P = []
	for i in range(2, m + 1):
		if S[i]:
			P.append(i)
			for j in range(i * i, m + 1, i):
				S[j] = False
	return P
P = {p: None for p in primes(10 ** 6)}

def pf(k):
	if k == 1:
		return set()
	if k in P:
		return {k}
	S = set()
	for p in P:
		(q, r) = divmod(k, p)
		if r == 0:
			S.add(p)
			while r == 0:
				k = q
				(q, r) = divmod(k, p)
			if k == 1:
				return S
			if k in P:
				S.add(k)
				return S

def solve(n, A):
	g = 0
	for a in A:
		g = gcd(g, a)
	B = [pf(a // g) for a in A]
	m = 0
	for i in range(n):
		for p in B[i] - B[i - 1]:
			j = i + 1 - n
			while p in B[j]:
				B[j].remove(p)
				j += 1
			m = max(m, (j - i) % n)
	return m

def case():
	n = readint()
	A = readintl()
	return str(solve(n, A))

def main():
	t = readint()
	print('\n'.join((case() for _ in range(t))))

def readint():
	return int(input())

def readinti():
	return map(int, input().split())

def readintt():
	return tuple(readinti())

def readintl():
	return list(readinti())

def readinttl(k):
	return [readintt() for _ in range(k)]

def readintll(k):
	return [readintl() for _ in range(k)]

def lstr(l):
	return ' '.join(map(str, l))

def llstr(ll):
	return '\n'.join(map(lstr, ll))

def log(*args, **kwargs):
	print(*args, **kwargs, file=sys.__stderr__)
main()
