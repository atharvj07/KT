n = int(input())
a = [int(i) for i in input().split()]
m = int(input())
q = [int(i) for i in input().split()]
b = [a[0]]
for i in range(1, len(a)):
	b.append(a[i] + b[i - 1])
for i in range(len(q)):
	l = -1
	r = len(a)
	while r - l > 1:
		mid = (l + r) // 2
		if b[mid] < q[i]:
			l = mid
		else:
			r = mid
	print(r + 1)
