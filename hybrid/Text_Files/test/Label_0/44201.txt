import math
for _ in range(int(input())):
	(n, k) = map(int, input().split())
	if n == k:
		print(0)
	else:
		(ans, x1, x2) = (0, 0, 0)
		if n > k:
			x = int(math.log2(int(n / k)))
			if (n % 2 != 0 or n % k != 0) or k * 2 ** x != n:
				print(-1)
			else:
				r = int(n / k)
				x = int(math.log2(r))
				ans += int(x / 3)
				x = x - int(x / 3) * 3
				ans += int(x / 2)
				x = x - int(x / 2) * 2
				print(x + ans)
		if k > n:
			x = int(math.log2(int(k / n)))
			if k % 2 != 0 or k % n != 0 or n * 2 ** x != k:
				print(-1)
			else:
				r = int(k / n)
				x = int(math.log2(r))
				ans += int(x / 3)
				x = x - int(x / 3) * 3
				ans += int(x / 2)
				x = x - int(x / 2) * 2
				print(x + ans)
