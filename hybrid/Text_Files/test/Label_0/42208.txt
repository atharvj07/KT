n = int(input())
(s, t) = (input(), input())
res = ''
a = ['abc' * n, 'acb' * n, 'bca' * n, 'bac' * n, 'cab' * n, 'cba' * n]
b = ['a' * n + 'b' * n + 'c' * n, 'a' * n + 'c' * n + 'b' * n, 'b' * n + 'a' * n + 'c' * n, 'b' * n + 'c' * n + 'a' * n, 'c' * n + 'b' * n + 'a' * n, 'c' * n + 'a' * n + 'b' * n]
for temp in a + b:
	if s not in temp and t not in temp:
		res = temp
if res == '':
	print('NO')
else:
	print('YES')
	print(res)
