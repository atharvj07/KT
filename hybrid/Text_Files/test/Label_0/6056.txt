#include<stdio.h>

#define N 8
#define NOT_FREE 1
#define FREE 0

int p[N],row[N],col[N],dpos[2*N-1],dneg[2*N-1];

void PrintBoard()
{
  int i,j;
  for(i=0;i<N;i++)
    {
      for(j=0;j<N;j++)
	{
	  if(row[i]==j)
	    printf("Q");
	  else
	    printf(".");
	}
      printf("\n");
    }
}
void PutQueen(int i)
{
  int j;
  
  if(i==N)
    {
      PrintBoard();
      return;
    }

  if(p[i]==NOT_FREE)
    {
      PutQueen(i+1);
    }

  else
    {
      for(j=0;j<N;j++)
	{
	  if(col[j]==NOT_FREE || dpos[i+j]==NOT_FREE || dneg[i-j+N-1]==NOT_FREE)
	    continue;
	  
	      row[i]=j;
	      col[j]=dpos[i+j]=dneg[i-j+N-1]=NOT_FREE;
	      PutQueen(i+1);
	      col[j]=dpos[i+j]=dneg[i-j+N-1]=FREE;
	}
    }
}

int main()
{
  int i,r,c,k;
  
  scanf("%d",&k);
  for(i=0;i<k;i++)
    {
      scanf("%d%d",&r,&c);
      p[r]=NOT_FREE;
      row[r]=c;
      col[c]=dpos[r+c]=dneg[r-c+N-1]=NOT_FREE;
    }
  PutQueen(0);
  
  return 0;
}
