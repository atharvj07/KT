import io
import os
from collections import Counter, defaultdict, deque

def solve(N, H, M, K, HM):
	assert M % 2 == 0
	HALFHOUR = M // 2
	freightTimes = [(h * M + m) % HALFHOUR for (h, m) in HM]
	freightTimes.sort()
	freightTimes = freightTimes + [x + HALFHOUR for x in freightTimes]
	window = deque()
	freq = Counter()
	best = (float('inf'),)
	for x in freightTimes:
		window.append(x)
		freq[x] += 1
		while x - window[0] > K:
			freq[window.popleft()] -= 1
		if x >= K:
			if window:
				cost = len(window)
				if window[-1] - window[0] == K:
					cost -= freq[window[-1]]
					cost -= freq[window[0]]
				else:
					cost -= freq[window[-1]]
				best = min(best, (cost, x % HALFHOUR))
	(covered, t) = best
	cancel = []
	for (i, (h, m)) in enumerate(HM):
		m = m % HALFHOUR
		if t - K < 0:
			if 0 <= m < t or (t - K) % HALFHOUR < m < HALFHOUR:
				cancel.append(str(i + 1))
		elif t - K < m < t:
			cancel.append(str(i + 1))
	assert len(cancel) == covered
	return str(len(cancel)) + ' ' + str(t) + '\n' + ' '.join(cancel)
