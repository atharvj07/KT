#include <bits/stdc++.h>
using namespace std;
long long gcd(long long a, long long b) {
  if (b == 0)
    return a;
  else
    return gcd(b, a % b);
}
long long nCr(long long n, long long r) {
  if (n < r) return 0;
  long long dp[n + 1][r + 1];
  for (long long i = 0; i <= n; i++) {
    dp[i][0] = 1;
  }
  for (long long j = 0; j <= r; j++) {
    dp[0][j] = 1;
  }
  for (long long i = 1; i <= n; i++) {
    for (long long j = 1; j <= r; j++) {
      if (i - 1 >= j)
        dp[i][j] =
            (dp[i - 1][j] + dp[i - 1][j - 1]) % (long long)(pow(10, 9) + 7);
      else if (i == j)
        dp[i][j] = 1;
      else
        dp[i][j] = 0;
    }
  }
  return dp[n][r];
}
void showpq(vector<long long> gq) {
  vector<long long> g = gq;
  reverse(g.begin(), g.end());
  while (!g.empty()) {
    cout << g.back() << " ";
    g.pop_back();
  }
  cout << '\n';
}
bool pri[1000001];
vector<long long> lund(1000001);
void prichk() {
  for (long long i = 2; (i * i) <= (1e6 + 1); ++i) {
    if (pri[i]) {
      for (long long j = i * i; j < (1e6 + 1); j += i) pri[j] = false;
    }
  }
  lund[0] = 0;
  lund[1] = 0;
  for (long long i = 2; i <= (1e6 + 1); ++i) {
    if (pri[i])
      lund[i] = lund[i - 1] + 1;
    else
      lund[i] = lund[i - 1];
  }
}
int main() {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  memset(pri, true, sizeof(pri));
  prichk();
  long long T = 1;
  cin >> T;
  while (T--) {
    long long n;
    cin >> n;
    cout << lund[n] - lund[sqrt(n)] + 1;
    cout << '\n';
    ;
  }
  return 0;
}
