signed main(){
  srand((unsigned)time(NULL));
  cin.tie(0);
  ios_base::sync_with_stdio(0);
  cout << fixed << setprecision(12);

  Int n, t;
  cin>>n>>t;
  vector<Int> h(n);
  cin>>h;

  vector<Int> ans(n);
  ans[0] = t / h[0];
  Int sum = h[0];
  Int t2 = t % h[0];
  Int mx = h[0];
  for(Int i=1;i<n;i++){
    if(mx < h[i]){
      ans[i] = (t - sum) / h[i];
      t2 = (t - sum) % h[i];
      mx = h[i];
    }
    else {
      ans[i] = ans[i-1];
      //pr(i, ans[i], t2, h[i]);
      if(t2 < h[i]) ans[i]--, t2 = mx - (h[i] - t2);
      else t2 -= h[i];
    }
    sum += h[i];
  }
    

  for(Int i=0;i<n;i++) cout<<max(1LL, ans[i]+1)<<endl;    
  
  return 0;
}
