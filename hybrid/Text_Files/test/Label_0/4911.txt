#include <bits/stdc++.h>
using namespace std;
vector<pair<long long, long long> > v;
vector<int> up, down, hull;
long long ccw(long long ax, long long ay, long long bx, long long by,
              long long ox, long long oy) {
  return (ax - ox) * (by - oy) - (bx - ox) * (ay - oy);
}
long long ccwP(int a, int b, int o) {
  return ccw(v[a].first, v[a].second, v[b].first, v[b].second, v[o].first,
             v[o].second);
}
void convex_hull() {
  sort(v.begin(), v.end());
  for (int i = 0; i < v.size(); i++) {
    while (up.size() >= 2 && ccwP(up[up.size() - 2], up.back(), i) >= 0)
      up.pop_back();
    up.push_back(i);
    while (down.size() >= 2 && ccwP(down[down.size() - 2], down.back(), i) <= 0)
      down.pop_back();
    down.push_back(i);
  }
  hull = up;
  for (int i = (int)down.size() - 2; i >= 1; i--) hull.push_back(down[i]);
}
int N;
int main() {
  ios::sync_with_stdio(0);
  cin.tie(0);
  cin.exceptions(ios::badbit | ios::failbit);
  cin >> N;
  for (int i = 1; i <= N; i++) {
    long long x, y;
    cin >> x >> y;
    y -= x * x;
    v.push_back({x, y});
  }
  convex_hull();
  int res = (int)up.size() - 1;
  for (int i = 1; i < up.size(); i++)
    if (v[up[i - 1]].first == v[up[i]].first) res--;
  cout << res << "\n";
  return 0;
}
