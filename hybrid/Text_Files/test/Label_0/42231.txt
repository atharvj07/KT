from collections import defaultdict

def main():
	t = int(input())
	allans = []
	for _ in range(t):
		(n, m) = readIntArr()
		grid = []
		for __ in range(n):
			grid.append(input())
		ok = True
		snakes = defaultdict(lambda : [])
		for i in range(n):
			for j in range(m):
				if grid[i][j] != '.':
					snakes[grid[i][j]].append([i, j])
					l = len(snakes[grid[i][j]])
					if l > max(n, m):
						ok = False
						break
			if ok == False:
				break
		if ok == False:
			allans.append(['NO'])
			continue
		startEnd = dict()
		for (letter, coords) in snakes.items():
			if len(coords) >= 2:
				if coords[0][0] == coords[1][0]:
					for i in range(2, len(coords)):
						if coords[0][0] != coords[i][0]:
							ok = False
							break
				elif coords[0][1] == coords[1][1]:
					for i in range(2, len(coords)):
						if coords[0][1] != coords[i][1]:
							ok = False
							break
				else:
					ok = False
			if ok == False:
				break
			startEnd[letter] = [coords[0], coords[-1]]
		if ok == False:
			allans.append(['NO'])
			continue
		adj = defaultdict(lambda : [])
		for (letter, [start, end]) in startEnd.items():
			if start[0] == end[0]:
				for col in range(start[1], end[1] + 1):
					letter2 = grid[start[0]][col]
					if letter != letter2:
						if letter2 == '.':
							ok = False
						else:
							adj[letter].append(letter2)
			else:
				for row in range(start[0], end[0] + 1):
					letter2 = grid[row][start[1]]
					if letter != letter2:
						if letter2 == '.':
							ok = False
						else:
							adj[letter].append(letter2)
		if ok == False:
			allans.append(['NO'])
			continue
		chars = startEnd.keys()
		for u in chars:
			for v in adj[u]:
				if v < u:
					ok = False
		if ok == False:
			allans.append(['NO'])
			continue
		order = sorted(chars)
		allans.append(['YES'])
		if len(order) == 0:
			allans.append([0])
		else:
			allC = 'abcdefghijklmnopqrstuvwxyz'
			maxChar = order[-1]
			maxCharIdx = allC.index(maxChar)
			allans.append([maxCharIdx + 1])
			(dr, dc) = startEnd[order[-1]][0]
			dr += 1
			dc += 1
			for i in range(maxCharIdx + 1):
				if allC[i] not in order:
					allans.append([dr, dc, dr, dc])
				else:
					([r1, c1], [r2, c2]) = startEnd[allC[i]]
					r1 += 1
					c1 += 1
					r2 += 1
					c2 += 1
					allans.append([r1, c1, r2, c2])
	multiLineArrayOfArraysPrint(allans)
	return
import sys
input = lambda : sys.stdin.readline().rstrip('\r\n')

def oneLineArrayPrint(arr):
	print(' '.join([str(x) for x in arr]))

def multiLineArrayPrint(arr):
	print('\n'.join([str(x) for x in arr]))

def multiLineArrayOfArraysPrint(arr):
	print('\n'.join([' '.join([str(x) for x in y]) for y in arr]))

def readIntArr():
	return [int(x) for x in input().split()]

def makeArr(defaultValFactory, dimensionArr):
	dv = defaultValFactory
	da = dimensionArr
	if len(da) == 1:
		return [dv() for _ in range(da[0])]
	else:
		return [makeArr(dv, da[1:]) for _ in range(da[0])]
