import java.util.*;
public class cf311a {
  public static void main(String[] args) {
    Scanner in = new Scanner(System.in);
    int n = in.nextInt();
    int k = in.nextInt();
    int max = (n*n-n)/2;
    if(max <= k)
      System.out.println("no solution");
    else for(int i=0; i<n; i++)
      System.out.println(0+" "+i);
  }
}
