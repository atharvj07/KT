import sys, io, os
try:
	Z = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
except:
	Z = lambda : sys.stdin.readline().encode()
Y = lambda : map(int, Z().split())
add = lambda x: x + 1

def f(n, a):
	m = f = 1
	c = g = 0
	x = n
	xc = 0
	z = []
	for i in range(n):
		if a[i] == m:
			c += 1
		elif m == 1:
			if c & 1 == 0:
				for j in range(c // 2):
					z.append(i - 2 * j - 2)
				if f:
					f = g = 0
			else:
				g = 1
				x = i
				xc = c
			c = 1
		elif g:
			if c & 1 == 0:
				if f:
					for j in range(x - 1, i - 2, 2):
						z.append(j)
					c = i + 1
				else:
					for j in range(x - xc - 1, x - 3, 2):
						z.append(j)
					for j in range(x - 2, i - 3, 2):
						z.append(j)
					z.append(i - 2)
					for j in range(i - 4, x - 3, -2):
						z.append(j)
					c = g = 0
			else:
				if f:
					f = 0
				for j in range(x - 1, i - 3, 2):
					z.append(j)
				z.append(i - 2)
				for j in range(i - 4, x - 2, -2):
					z.append(j)
				for j in range(x - 3, x - xc - 1, -2):
					z.append(j)
				c = g = 0
		else:
			if f:
				f = 0
			c = 1
		m = a[i]
	if g or (m and (c & 1 or c == n)):
		return (0, [])
	if m:
		for j in range(n - c - 1, n - 2, 2):
			z.append(j)
	return (1, z)
O = []
for _ in range(int(Z())):
	(b, z) = f(int(Z()), [*Y()])
	if b:
		O.append('YES')
		O.append(str(len(z)))
		if z:
			O.append(' '.join(map(str, map(add, z))))
	else:
		O.append('NO')
print('\n'.join(O))
