#define MAOU 20

  if ( now == n)return true;

double dist(int x,int y){
  return sqrt(x*x+y*y);
}

int solve(int n,int *x,int *y,int rx,int ry,int mx,int my){
  rep(i,n){
    adj[i][i]=0;
    adj[MAOU][i]=adj[i][MAOU]=dist(x[i]-rx,y[i]-ry);
    REP(j,i+1,n){
      adj[i][j]=adj[j][i]=dist(x[i]-x[j],y[i]-y[j]);
    }
  }
  
  rep(i,n){
    if ( dist(mx-x[i],my-y[i])+eps>dist(rx-x[i],ry-y[i]))return false;
  }


  bool visited[20]={false};
  rep(i,n){
    if ( dist(mx-x[i],my-y[i]) < dist(rx-x[i],ry-y[i])){
      visited[i]=true;
      if ( backTrack(i,dist(mx-x[i],my-y[i]),1,n,visited))return true;
      visited[i]=false;
    }else return false;
  }
  return false;
}

int main(){
  int n,mx,my,rx,ry;
  while(cin>>n>>mx>>my>>rx>>ry && n){
    int x[n],y[n];
    rep(i,n)cin>>x[i]>>y[i];
    if ( solve(n,x,y,rx,ry,mx,my))puts("YES");
    else puts("NO");
  }
}
