#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:102400000,102400000")
using namespace std;
int num[200010];
int n;
int find(int x, int le) {
  int left = le, right = n - 1;
  while (left <= right) {
    int mid = (left + right) >> 1;
    if (num[mid] < x) {
      left = mid + 1;
    } else if (num[mid] > x) {
      right = mid - 1;
    } else {
      left = mid + 1;
    }
  }
  return right;
}
int main() {
  while (~scanf("%d", &n)) {
    for (int i = 0; i < n; i++) {
      scanf("%d", &num[i]);
    }
    sort(num, num + n);
    int cou = 1;
    for (int i = 1; i < n; i++) {
      if (num[i] == num[i - 1]) {
        continue;
      }
      num[cou] = num[i];
      cou++;
    }
    n = cou;
    int ans = 0;
    for (int i = 0; i < n; i++) {
      int cnt, pos = i + 1;
      while (pos < n) {
        cnt = (num[pos] + num[i]) / num[i];
        int re = find(num[i] * cnt - 1, pos);
        ans = max(ans, num[re] % num[i]);
        pos = re + 1;
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
