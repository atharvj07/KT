import java.util.*;
public class distinct_no {
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        
        
        while(t-->0){
            int n=sc.nextInt();
            int arr[]=new int[n];
            for(int i=0;i<n;i++){
                arr[i]=sc.nextInt();
            }
            int min=arr[0];
            for(int i=0;i<n;i++){
                min= Math.min(min,arr[i]);               
            }
            int count=0;
            for(int j=0;j<n;j++){
                if(arr[j]==min)
                count++;
            } int ans=0;
            if(count==n)
            { ans=0;
            System.out.println(ans);}
            else
            {    ans =n-count;
                System.out.println(ans);
            count=0; }
        }
        sc.close();
    }
}
