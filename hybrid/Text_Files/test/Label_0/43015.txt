def SieveOfEratosthenes(n):
	all_prime = []
	prime = [True for i in range(n + 1)]
	p = 2
	while p * p <= n:
		if prime[p]:
			for i in range(p * p, n + 1, p):
				prime[i] = False
		p += 1
	for p in range(2, n):
		if prime[p]:
			all_prime.append(p)
	return all_prime
c = 1
n = int(input())
r = SieveOfEratosthenes(n + 2)
l = [0] * (n + 2)
if n > 2:
	c = 2
for i in r:
	l[i] = 1
for j in range(n + 2):
	if l[j] == 0:
		l[j] = 2
print(c)
print(*l[2:])
