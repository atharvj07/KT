def digsum(x):
	ret = 0
	while x > 0:
		ret += x % 10
		x //= 10
	return ret
tt = int(input())
for loop in range(tt):
	(n, k) = map(int, input().split())
	ans = float('inf')
	if (n - k * (k + 1) // 2) % (k + 1) == 0:
		X = (n - k * (k + 1) // 2) // (k + 1)
		now = []
		while X > 0:
			if len(now) == 0:
				now.append(min(X, 9 - k))
			else:
				now.append(min(9, X))
			X -= now[-1]
		tmp = 0
		for i in range(len(now)):
			tmp += 10 ** i * now[i]
		ans = min(tmp, ans)
	for t in range(1, k + 1):
		for m in range(1, 50):
			if (n - k * (k + 1) // 2 + 9 * m * (k - t + 1)) % (k + 1) == 0:
				X = (n - k * (k + 1) // 2 + 9 * m * (k - t + 1)) // (k + 1)
				now = []
				Xdown = 10 ** m - t
				while Xdown > 0:
					now.append(Xdown % 10)
					Xdown //= 10
					X -= now[-1]
				if X < 0:
					continue
				fl = True
				while X > 0:
					if fl:
						now.append(min(8, X))
						fl = False
					else:
						now.append(min(9, X))
					X -= now[-1]
				tmp = 0
				for i in range(len(now)):
					tmp += 10 ** i * now[i]
				ans = min(tmp, ans)
	if ans == float('inf'):
		ans = -1
	if ans == 0:
		ntmp = 0
		for j in range(ans, ans + k + 1):
			ntmp += digsum(j)
		if ntmp != n:
			ans = -1
	print(ans)
