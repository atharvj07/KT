#include <bits/stdc++.h>
using namespace std;
int n, k;
struct node {
  int l, r;
  int id;
  bool operator<(const node &b) const { return l < b.l; }
};
node a[300005];
int main() {
  while (cin >> n >> k) {
    for (int i = 1; i <= n; i++) {
      scanf("%d %d", &a[i].l, &a[i].r);
      a[i].id = i;
    }
    sort(a + 1, a + n + 1);
    priority_queue<int, vector<int>, greater<int> > q;
    int ans = 0;
    int pos = 0;
    for (int i = 1; i <= n; i++) {
      int temp = 0;
      q.push(a[i].r);
      if (q.size() > k) q.pop();
      if (q.size() == k) {
        int R = q.top();
        int len = R - a[i].l + 1;
        if (ans < len) {
          ans = len;
          pos = i;
        }
      }
    }
    if (ans == 0) {
      cout << "0\n";
      for (int i = 1; i <= k; i++) {
        cout << a[i].id << " ";
      }
      cout << endl;
    } else {
      cout << ans << endl;
      for (int i = 1; i <= pos; i++) {
        if (a[i].r >= a[pos].l + ans - 1) {
          cout << a[i].id << " ";
        }
      }
      cout << endl;
    }
  }
  return 0;
}
