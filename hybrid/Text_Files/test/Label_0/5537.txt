#include <bits/stdc++.h>
#pragma comment(linker, "/STACK:102400000,102400000")
using namespace std;
const int inf = 1e9 + 7;
const long long linf = 1e18;
const double pi = acos(-1.0);
template <class T>
T sqr(T x) {
  return x * x;
}
const double eps = 1e-6;
vector<int> adj[100010];
int lev[100010];
int M, N;
int pr[2];
int visited[100010];
int dfs(int v, int p) {
  int i, u;
  lev[v] = p >= 0 ? lev[p] + 1 : 0;
  pr[lev[v] & 1]++;
  visited[v] = 1;
  for (i = 0; i < adj[v].size(); ++i) {
    u = adj[v][i];
    if (u == p) continue;
    if (visited[u]) {
      if ((lev[u] - lev[v]) & 1) continue;
      return 1;
    }
    if (dfs(u, v)) return 1;
  }
  return 0;
}
int main() {
  int u, v;
  scanf("%d%d", &N, &M);
  int i, deg = 0;
  for (i = 0; i < M; ++i) {
    scanf("%d%d", &u, &v);
    --u, --v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    deg = max(deg, (int)adj[v].size());
    deg = max(deg, (int)adj[u].size());
  }
  long long sl = 0;
  for (i = 0; i < N; ++i) {
    if (visited[i]) continue;
    pr[0] = pr[1] = 0;
    if (dfs(i, -1)) {
      cout << "0 1\n";
      return 0;
    }
    for (int b = 0; b < 2; ++b) {
      sl += (1LL * pr[b] * (pr[b] - 1)) / 2;
    }
  }
  if (sl) {
    cout << 1 << " " << sl << "\n";
    return 0;
  }
  if (M) {
    cout << 2 << " " << 1LL * M * (N - 2) << "\n";
    return 0;
  }
  sl = (1LL * N * (N - 1) * (N - 2)) / 6;
  cout << 3 << " " << sl << "\n";
  return 0;
}
