#include <bits/stdc++.h>
const long long MOD = 1e9 + 7;
const int EPS = 1e-9;
const long double PI = acos(-1.0);
const int MAXX = 2e5 + 3;
const int LOGMAXN = 30;
using namespace std;
vector<int> adj[MAXX], adj1[MAXX];
vector<vector<char> > res;
vector<char> v;
long long N, M, K, dist[MAXX];
map<pair<long long, long long>, long long> roads;
bool flag, vis[MAXX];
void bfs(int s) {
  for (int i = 0; i <= N; ++i) {
    dist[i] = 1e12;
  }
  dist[s] = 0;
  queue<int> q;
  q.push(s);
  while (!q.empty()) {
    int cur = q.front();
    q.pop();
    for (int i = 0; i < adj[cur].size(); ++i) {
      int to = adj[cur][i];
      if (dist[to] > dist[cur] + 1) {
        dist[to] = dist[cur] + 1;
        q.push(to);
      }
      if (dist[to] == dist[cur] + 1) {
        adj1[to].push_back(roads[{min(cur, to), max(cur, to)}]);
      }
    }
  }
}
void dfs(int node, int d) {
  if (flag) return;
  if (d == N - 1) {
    if (res.size() < K) {
      res.push_back(v);
    } else {
      flag = true;
    }
    return;
  }
  for (int i = 0; i < adj1[node].size(); ++i) {
    int to = adj1[node][i];
    v[to] = '1';
    dfs(node + 1, d + 1);
    v[to] = '0';
    if (flag) return;
  }
}
int main() {
  ios_base::sync_with_stdio(0);
  cin >> N >> M >> K;
  for (int i = 0; i < M; ++i) {
    int u, v;
    cin >> u >> v;
    adj[u].push_back(v);
    adj[v].push_back(u);
    roads[{min(u, v), max(u, v)}] = i;
  }
  v.assign(M, '0');
  bfs(1);
  dfs(2, 0);
  cout << res.size() << endl;
  for (int i = 0; i < res.size(); ++i) {
    for (int j = 0; j < res[i].size(); ++j) {
      cout << res[i][j];
    }
    cout << endl;
  }
  return 0;
}
