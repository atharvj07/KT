#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int a[N], b[N];
int main() {
  int n, m;
  long long ans = 0;
  cin >> n >> m;
  int mi = 0, mmi = 0;
  for (int i = 1; i <= n; i++) {
    scanf("%d", a + i);
    if (a[i] > a[mi])
      mmi = mi, mi = i;
    else if (a[i] > a[mmi])
      mmi = i;
    ans += 1ll * a[i] * m;
  }
  int ff = 0;
  for (int i = 1; i <= m; i++) {
    scanf("%d", b + i);
    if (b[i] < a[mi]) {
      puts("-1");
      return 0;
    }
    if (b[i] == a[mi]) ff = 1;
    ans += 1ll * b[i] - a[mi];
  }
  if (!ff) ans += 1ll * (a[mi] - a[mmi]);
  cout << ans << endl;
  return 0;
}
