from collections import defaultdict
(n, m) = map(int, input().split())
G = defaultdict(list)
for _ in range(m):
	(u, v) = map(int, input().split())
	G[u].append(v)
	G[v].append(u)
rs = [1] * n
for u in range(1, n + 1):
	for v in G[u]:
		if v < u:
			rs[u - 1] = max(rs[u - 1], rs[v - 1] + 1)
print(max((rs[i] * len(G[i + 1]) for i in range(len(rs)))))
