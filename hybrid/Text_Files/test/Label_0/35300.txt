import java.util.Scanner;

public class Straight {
	
	static double average (int[] nums, int additional, int num) {
		int sum = 0;
		for (int i = 0; i < nums.length; i++) {
			sum += nums[i];
		}
		for (int i = 0; i < additional; i++) {
			sum += num;
		}
		
		int separate = nums.length + additional;
		
		return (double) sum / (double) separate;
	}
	
	public static void main(String[] args) {
		Scanner s = new Scanner(System.in);
		
		int gradeNum = s.nextInt();
		int best = s.nextInt();
		
		int[] grades = new int[gradeNum];
		for (int i = 0; i < gradeNum; i++) {
			grades[i] = s.nextInt();
		}
		
		int iter = 0;
		while (Math.round(average(grades, iter, best)) != best) {
			iter++;
		}
		
		System.out.print(iter);
		
		s.close();
	}

}
