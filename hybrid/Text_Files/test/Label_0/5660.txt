#include <bits/stdc++.h>
#pragma GCC optimize(2)
using namespace std;
long long n, k, a[2020], dp[2020][2], bit[2020];
vector<pair<long long, long long> > fd, bk;
void update(long long x, long long v) {
  x++;
  while (x < 2020) {
    bit[x] = min(bit[x], v);
    x += (x & (-x));
  }
  return;
}
long long query(long long x) {
  x++;
  long long ret = 1000000000000000000;
  while (x > 0) {
    ret = min(ret, bit[x]);
    x -= (x & (-x));
  }
  return ret;
}
bool cmp(pair<long long, long long> a, pair<long long, long long> b) {
  if (a.first < b.first) {
    return true;
  }
  if (a.first > b.first) {
    return false;
  }
  if (a.second > b.second) {
    return true;
  }
  return false;
}
bool check(long long x) {
  long long i, j;
  memset(bit, 63, sizeof(bit));
  fd.clear();
  bk.clear();
  for (i = 0; i < n; i++) {
    fd.push_back(make_pair(a[i] - i * x, i));
    bk.push_back(make_pair(a[i] + i * x, i));
    dp[i][0] = dp[i][1] = 2 * 1000000000;
  }
  sort(fd.begin(), fd.end(), cmp);
  sort(bk.begin(), bk.end());
  for (i = fd.size() - 1; i >= 0; i--) {
    j = upper_bound(bk.begin(), bk.end(),
                    make_pair(fd[i].first + 2 * x * fd[i].second,
                              (long long)1000000000)) -
        bk.begin() - 1;
    dp[fd[i].second][0] = min((long long)0, query(j)) + fd[i].second;
    update(j, dp[fd[i].second][0] - fd[i].second - 1);
  }
  dp[0][1] = 1;
  for (i = 1; i < n; i++) {
    dp[i][1] = min(dp[i][1], min(dp[i - 1][0], dp[i - 1][1]) + 1);
  }
  if (dp[n - 1][0] <= k || dp[n - 1][1] <= k) {
    return true;
  }
  return false;
}
int main() {
  long long i, l, r, mid;
  scanf("%lld%lld", &n, &k);
  for (i = 0; i < n; i++) {
    scanf("%lld", &a[i]);
  }
  l = 0;
  r = 2 * 1000000000 + 10;
  while (l < r) {
    mid = (l + r) >> 1;
    if (check(mid)) {
      r = mid;
    } else {
      l = mid + 1;
    }
  }
  printf("%lld\n", l);
  return 0;
}
