#include <bits/stdc++.h>
#define rep(i,n) for (int i = 0; i < (n); i++)
using namespace std;
typedef long long ll;

int n;
vector<vector<int>> g;

vector<int> bfs(int s)
{
  vector<int> ret, par(n, -1);
  queue<int> q;
  q.push(s);
  while (!q.empty())
  {
    int crt = q.front();
    q.pop();
    for (int nxt : g[crt])
    {
      if (par[nxt] != -1)
        continue;
      par[nxt] = crt;
      q.push(nxt);
    }
  }
  if (par[s] == -1)
    return vector<int>(n + 1);
  ret.push_back(par[s] + 1);
  int p = par[s];
  while (p != s)
  {
    p = par[p];
    ret.push_back(p + 1);
  }
  return ret;
}

int main()
{
  int m;
  cin >> n >> m;
  g.resize(n);
  rep(i, m)
  {
    int a, b;
    cin >> a >> b;
    a--;
    b--;
    g[a].push_back(b);
  }
  vector<int> ans(n + 1);
  rep(i, n)
  {
    vector<int> tmp = bfs(i);
    if (tmp.size() < ans.size())
      ans.swap(tmp);
  }
  if (ans.size() == n + 1)
  {
    puts("-1");
    return 0;
  }
  printf("%d\n", ans.size());
  for (int i : ans)
    printf("%d\n", i);
  return 0;
}
