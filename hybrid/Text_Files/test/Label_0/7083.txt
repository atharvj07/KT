#include <bits/stdc++.h>
using namespace std;
int n, x;
const int maxn = 200100;
pair<long long int, long long int> p[maxn];
long long int r[maxn], q[maxn];
long long int t[4 * maxn];
pair<int, long long int> up[maxn][22];
vector<int> conn[maxn];
long long int inf = (long long int)(1e12);
void upd(int l, int r, int v, int pos, long long int val) {
  if (l == r) {
    t[v] = val;
    return;
  }
  int mid = (r + l) / 2;
  if (pos <= mid) {
    upd(l, mid, 2 * v + 1, pos, val);
  } else {
    upd(mid + 1, r, 2 * v + 2, pos, val);
  }
  t[v] = max(t[2 * v + 1], t[2 * v + 2]);
}
long long int get(int l, int r, int tl, int tr, int v) {
  if (l > tr || r < tl) {
    return -1;
  }
  if (l >= tl && r <= tr) {
    return t[v];
  }
  int mid = (r + l) / 2;
  int m1 = get(l, mid, tl, tr, 2 * v + 1);
  int m2 = get(mid + 1, r, tl, tr, 2 * v + 2);
  return max(m1, m2);
}
void dfs(int cur, int par = n - 1) {
  up[cur][0] = make_pair(par, q[cur] - r[cur]);
  for (int i = 1; i <= 20; i++) {
    up[cur][i] = make_pair(
        up[up[cur][i - 1].first][i - 1].first,
        up[cur][i - 1].second + up[up[cur][i - 1].first][i - 1].second);
  }
  for (int i = 0; i < conn[cur].size(); i++) {
    int to = conn[cur][i];
    dfs(to, cur);
  }
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) {
    scanf("%I64d%I64d", &p[i].first, &p[i].second);
    p[i].second += p[i].first;
  }
  p[n] = make_pair(inf, inf);
  n++;
  upd(0, n - 1, 0, n - 1, -1);
  for (int i = n - 2; i >= 0; i--) {
    int l1 = upper_bound(p, p + n, make_pair(p[i].first, 0ll)) - p;
    int l2 = lower_bound(p, p + n, make_pair(p[i].second, 0ll)) - p;
    if (p[l2].first > p[i].second) l2--;
    long long int ans = get(0, n - 1, l1, l2, 0);
    r[i] = max(p[i].second, ans);
    upd(0, n - 1, 0, i, r[i]);
    int ind = upper_bound(p, p + n, make_pair(r[i], 0ll)) - p;
    conn[ind].push_back(i);
    q[i] = p[ind].first;
  }
  dfs(n - 1);
  cin >> x;
  for (int i = 0; i < x; i++) {
    int l, r;
    scanf("%d%d", &l, &r);
    l--, r--;
    int ans = 0;
    for (int i = 20; i >= 0; i--) {
      if (up[l][i].first <= r) {
        ans += up[l][i].second;
        l = up[l][i].first;
      }
    }
    cout << ans << endl;
  }
  return 0;
}
