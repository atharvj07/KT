from sys import stdin, stdout
input = stdin.buffer.readline
print = stdout.write

def f(a, b):
	return (a + b) * n // 2
n = int(input())
(*h,) = map(int, input().split())
s = sum(h)
(l, r) = (0, 10 ** 12)
while l < r:
	m = l + r >> 1
	if f(m, m + n - 1) < s:
		l = m + 1
	else:
		r = m
l -= 1
y = f(l, l + n - 1)
for i in range(n):
	print(f'{l + i + (i < s - y)} ')
