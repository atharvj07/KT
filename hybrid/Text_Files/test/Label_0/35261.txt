import static java.lang.Math.*;
import static java.util.Arrays.* ;

import java.util.*;
import java.io.*;

public class Main
{
    void main() throws Exception
    {
        Scanner sc = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);
        int n = sc.nextInt() , m = sc.nextInt() , d = sc.nextInt() ; 
        int [] a = new int [m] , b = new int [m] , time = new int [m] ;
        
        for(int i = 0 ; i < m ; i ++)
        {
            a[i] = sc.nextInt() ;
            b[i] = sc.nextInt() ;
            time[i] = sc.nextInt() ;
        }
        long ans = -(1L << 60) ;
        long [][] dp = new long [2][n + 1] ;
        
        for(int i = m - 1 ; i >= 0 ; i --)
        {
            int t = i == m - 1 ? time[i] : (time[i + 1] - time[i]) ;
            ModifiedQueue queue = new ModifiedQueue() ;
            
            int l = 1 ;
            int r = (int)min(n , 1 + 1L * t * d) ;
            int least = 1 ;
            for( ; l <= r ; l++)
                queue.add(l , dp[(i ^ 1) & 1][l]);
            
            for(int j = 1 ; j <= n ; j ++)
            {

                long c = b[i] - abs(a[i] - j) ;
                long max = -(1L << 60) ;

                // getMaxRange(-t * d -> t * d)
                int from = (int)max(1 , j - 1L * t * d) ;
                int to   = (int)min(n , j + 1L * t * d) ;
                if(j != 1)
                    queue.add(to , dp[(i ^ 1) & 1][to]);
                least = queue.remove(from , least);
                dp[i & 1][j] = c + queue.max() ;
            }
        }
        for(int i = 1 ; i <= n ; i ++)
            ans = max(ans , dp[0][i]) ;
        out.println(ans);
        out.flush();
        out.close();
    }
    
    class ModifiedQueue
    {
        long [] q = new long[(int)1e6 + 200];
        long [] d = new long[(int)1e6 + 200];
        int qh = 0 , qt = 0 ;
        int dh = 0, dt = 0;

        void add(int i , long x)
        {
            q[qt++] = x ;
            while(dh < dt && d[dt - 1] < x) dt --;
            d[dt++] = x;
        }
        int remove(int time , int least)
        {
            while(qh < qt && least < time)
            {
                long c = q[qh++];
                if (c == d[dh])
                    dh ++ ;
                least ++ ;
            }
            return least ;
        }
        long max()
        {
            if(dh == dt)
                return -(1L << 60) ;
            return d[dh];
        }
    }

    class Scanner
    {
        BufferedReader br;
        StringTokenizer st;

        Scanner(InputStream in)
        {
            br = new BufferedReader(new InputStreamReader(System.in));
        }

        String next() throws Exception
        {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        int nextInt() throws Exception { return Integer.parseInt(next()); }

        long nextLong() throws Exception { return Long.parseLong(next()); }

        double nextDouble() throws Exception { return Double.parseDouble(next());}
    }

    public static void main (String [] args) throws Exception {(new Main()).main();}

}
