#include <stdio.h>

int main()
{
	bool board[8][8]{};
	for (int i{}; i < 8; i++)
	{
		for (int j{}; j < 8; j++)
		{
			if (i % 2 == 0 && j % 2) continue;
			board[i][j] = true;
		}
	}
	int q;
	scanf("%d", &q);
	for (int i{}; i < q; i++)
	{
		int a, b, c, d;
		scanf("%d%d%d%d", &a, &b, &c, &d);
		a--; b--; c--; d--;
		int count{};
		for (int row{ a }; row <= c; row++)
			for (int col{ b }; col <= d; col++)
				count += board[row][col];
		printf("%d\n", count);
	}
	return 0;
}
