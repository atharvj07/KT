import sys
input = sys.stdin.readline

def solve():
	(n, k) = map(int, input().split())
	(*a,) = map(int, input().split())
	X = 0
	for i in a:
		X ^= i
	g = [[] for i in range(n)]
	for i in range(n - 1):
		(u, v) = map(int, input().split())
		g[u - 1].append(v - 1)
		g[v - 1].append(u - 1)
	if X == 0:
		print('YES')
		return
	elif k == 2:
		print('NO')
		return
	p = [None] * n
	q = [0]
	p[0] = -1
	i = 0
	while i < len(q):
		x = q[i]
		i += 1
		for v in g[x]:
			if p[v] is None:
				p[v] = x
				q.append(v)
	d = [None] * n
	dx = [None] * n
	for i in range(len(q) - 1, -1, -1):
		x = q[i]
		P = p[x]
		z = a[x]
		for v in g[x]:
			if v != P:
				z ^= dx[v]
		c = 0
		for v in g[x]:
			if v != P:
				if d[v]:
					c += 1
					if z == 0:
						print('YES')
						return
		if c > 1:
			print('YES')
			return
		dx[x] = z
		d[x] = z == X or c > 0
	print('NO')
for i in range(int(input())):
	solve()
