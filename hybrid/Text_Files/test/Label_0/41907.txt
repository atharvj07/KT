l = int(input())
n = input()
itr = l // 2 + l % 2
itr2 = itr - 1
while itr < l and n[itr] == '0':
	itr += 1
while n[itr2] == '0':
	itr2 -= 1
(x, y) = (itr - l // 2 - l % 2, l // 2 - itr2)
if x == y:
	p = n[0:itr]
	q = n[itr2:itr] + n[0:itr2]
	if int(p) > int(q):
		(a, b) = (n[0:itr2], n[itr2:l])
	else:
		(a, b) = (n[0:itr], n[itr:l])
elif x > y:
	(a, b) = (n[0:itr2], n[itr2:l])
else:
	(a, b) = (n[0:itr], n[itr:l])
print(int(a) + int(b))
