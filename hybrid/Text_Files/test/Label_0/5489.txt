#include <bits/stdc++.h>
using namespace std;
const long long N = 5e3 + 5;
long long a[N];
long long dp[N][(N + 10) / 2][2];
long long n;
long long solve(long long i, long long k, long long prev) {
  if (k == 0) return 0;
  if (i >= n + 1) return 8e18;
  long long &ans = dp[i][k][prev];
  if (ans != -1) return ans;
  long long op1 = 8e18, op2 = 8e18;
  op1 = solve(i + 1, k, 0);
  if (prev == 1) {
    long long sum = 0;
    long long mi = min(a[i - 1], a[i - 2] - 1);
    if (mi >= a[i]) sum += (mi - a[i] + 1);
    if (a[i + 1] >= a[i]) sum += (a[i + 1] - a[i] + 1);
    op2 = sum + solve(i + 2, k - 1, 1);
  } else {
    long long sum = 0;
    if (a[i - 1] >= a[i]) sum += (a[i - 1] - a[i] + 1);
    if (a[i + 1] >= a[i]) sum += (a[i + 1] - a[i] + 1);
    op2 = sum + solve(i + 2, k - 1, 1);
  }
  return ans = min(op1, op2);
}
void solve() {
  cin >> n;
  for (long long i = 1; i <= n; i++) cin >> a[i];
  memset(dp, -1, sizeof dp);
  long long tot = (n + 1) / 2;
  a[0] = -1, a[n + 1] = -1;
  for (long long i = 1; i <= tot; i++) {
    long long ans = solve(1, i, 0);
    cout << ans << " ";
  }
}
int32_t main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  long long t = 1;
  while (t--) solve();
}
