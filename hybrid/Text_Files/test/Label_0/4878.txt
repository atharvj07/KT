#include <bits/stdc++.h>
using namespace std;
long long n, k, p;
char getChar(long long x, long long n, long long k) {
  if (k == 0LL) return '.';
  if (n % 2LL)
    if (x == n)
      return 'X';
    else
      return getChar(x, n - 1LL, k - 1LL);
  if (x % 2LL == 0)
    if (n / 2LL - x / 2LL + 1LL <= k)
      return 'X';
    else
      return '.';
  k -= n / 2LL;
  if (n / 2LL - (x + 1LL) / 2LL + 1LL <= k)
    return 'X';
  else
    return '.';
}
int main() {
  cin >> n >> k >> p;
  for (int i = 0; i < p; i++) {
    long long x;
    cin >> x;
    cout << getChar(x, n, k);
  }
  cout << endl;
  return 0;
}
