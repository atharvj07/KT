#include <bits/stdc++.h>
using namespace std;
const int MOD = 1e9 + 7;
const int INF = 1e9;
const long long LINF = 1e18;
const int WOW = 40000;
const int NIL = 0;
const int dy[8] = {0, 0, -1, 1, 1, 1, -1, -1};
const int dx[8] = {1, -1, 0, 0, 1, -1, 1, -1};
const double PI = 2 * acos(0);
const int MAXX = 1030;
int pl, pr, vl, vr, k;
int ps[MAXX], pe[MAXX], vs[MAXX], ve[MAXX];
vector<int> lucky;
void init(int n) {
  lucky.push_back(n);
  if (10ll * n >= 1e9) return;
  init(n * 10 + 4);
  init(n * 10 + 7);
}
void getLucky() {
  init(0);
  lucky.push_back(int(1e9 + 1));
  sort((lucky).begin(), (lucky).end());
}
int main() {
  getLucky();
  int l, r;
  scanf("%d%d%d%d%d", &pl, &pr, &vl, &vr, &k);
  for (int x = 0; x < lucky.size() - 1; ++x) {
    l = max(pl, lucky[x]), r = min(pr, lucky[x + 1] - 1);
    pe[x] = r - l + 1 < 0 ? 0 : r - l + 1;
    l = max(pl - 1, lucky[x]), r = min(pr - 1, lucky[x + 1] - 1);
    ps[x] = r - l + 1 < 0 ? 0 : r - l + 1;
    l = max(vl, lucky[x]), r = min(vr, lucky[x + 1] - 1);
    ve[x] = r - l + 1 < 0 ? 0 : r - l + 1;
    l = max(vl - 1, lucky[x]), r = min(vr - 1, lucky[x + 1] - 1);
    vs[x] = r - l + 1 < 0 ? 0 : r - l + 1;
  }
  long long sub = 0;
  long long tot = 1ll * (vr - vl + 1) * (pr - pl + 1);
  for (int x = 0; x < MAXX; ++x) {
    if (x - k >= 0) sub += 1ll * ve[x] * ps[x - k];
    if (x + k < MAXX) sub += 1ll * pe[x + k] * vs[x];
  }
  int cnt = 0;
  if (k == 1) {
    for (int x = 1; x < MAXX; ++x) {
      if (pl <= lucky[x] && lucky[x] <= pr && vl <= lucky[x] && lucky[x] <= vr)
        ++cnt;
    }
  }
  printf("%.9lf\n", 1.0 * (sub - cnt) / tot);
}
