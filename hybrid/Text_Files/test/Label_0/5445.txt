#include <bits/stdc++.h>
using namespace std;
pair<long long, long long> ans = {0, 0};
long long jpow(long long x) { return x * x * x; }
void dfs(long long m, long long num, long long x) {
  if (m == 0) {
    ans = max(ans, make_pair(num, x));
    return;
  }
  long long cnt = 1;
  while (jpow(cnt + 1) <= m) cnt++;
  dfs(m - jpow(cnt), num + 1, x + jpow(cnt));
  dfs(jpow(cnt) - jpow(cnt - 1) - 1, num + 1, x + jpow(cnt - 1));
}
long long n;
int main() {
  cin >> n;
  dfs(n, 0, 0);
  cout << ans.first << " " << ans.second << endl;
  return 0;
}
