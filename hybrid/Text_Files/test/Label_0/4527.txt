#include <bits/stdc++.h>
using namespace std;
int main() {
  unsigned long long n;
  cin >> n;
  unsigned long long k = n - 3;
  unsigned long long j = n - 4;
  unsigned long long a = 4;
  unsigned long long res = 1;
  unsigned long long sol = 0;
  while (k > 0) {
    if (k & 1) res = res * a;
    a = a * a;
    k >>= 1;
  }
  sol = sol + 2 * 3 * 4 * res;
  a = 4;
  k = j;
  res = 1;
  while (k > 0) {
    if (k & 1) res = res * a;
    a = a * a;
    k >>= 1;
  }
  sol = sol + (n - 3) * 3 * 4 * res * 3;
  cout << sol;
}
