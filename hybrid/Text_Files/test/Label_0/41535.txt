from queue import *
(n, s) = map(int, input().split())
ja = [False] * 1001
fila = PriorityQueue()
e = 0
for i in range(n):
	(f, t) = map(int, input().split())
	fila.put([-f, -t])
while not fila.empty():
	(f, t) = fila.get()
	f *= -1
	t *= -1
	if not ja[f]:
		ja[f] = True
		e += s - f
		s = f
		if t > e:
			e += t - e
e += s
print(e)
