#include <bits/stdc++.h>
using namespace std;
int main() {
  int a, b;
  cin >> a >> b;
  int f = 0, s = 0;
  bool tru = 0;
  if (b < a) {
    swap(a, b);
    tru = 1;
  }
  f = a;
  s = 6 - b;
  s++;
  if (b - a == 0) {
    cout << 0 << " " << 6 << " " << 0;
    return 0;
  }
  if (b - a == 1) {
    if (tru == 1) swap(s, f);
    cout << f << " " << 0 << " " << s;
    return 0;
  }
  if (b - a == 2 || b == a) {
    if (tru == 1) swap(s, f);
    cout << f << " " << 1 << " " << s;
    return 0;
  }
  if (b - a == 3) {
    if (tru == 1) swap(s, f);
    cout << f + 1 << " " << 0 << " " << s + 1 << endl;
    return 0;
  }
  if (b - a == 4) {
    if (tru == 1) swap(s, f);
    cout << f + 1 << " " << 1 << " " << s + 1 << endl;
    return 0;
  }
  if (b - a == 5) {
    if (tru == 1) swap(s, f);
    cout << f + 2 << " " << 0 << " " << s + 2 << endl;
  }
  return 0;
}
