int getInt(){
  int ret = 0,c;
  c = getchar();
  while(!isdigit(c)) c = getchar();
  while(isdigit(c)){
    ret *= 10;
    ret += c - '0';
    c = getchar();
  }
  return ret;
}

template<class T>
T gcd(T a, T b){
  if(a > b) swap(a, b);
  if(a == 0) return b;
  return gcd(a, b%a);
}

int barL[100];
int barR[100];

int barRatioL[100];
int barRatioR[100];

int which(int n){
  vector<bool> f(n+1, false);
  REP(i,n+1) f[barL[i]] = f[barR[i]] = true;
  REP(i,n+1) if(!f[i]) return i;
  return -1;
}

long long weight(int i){
  int l = barL[i-1];
  int r = barR[i-1];

  long long wl = l == 0 ? 1 : weight(l);
  long long wr = r == 0 ? 1 : weight(r);

  int rl = barRatioL[i-1];
  int rr = barRatioR[i-1];

  long long R = wr * rr;
  long long L = wl * rl;
  long long g = gcd(R, L);
  long long lcm = R*L/g;

  return (lcm/rl + lcm/rr);
}

int main(){
  int n;
  while(n = getInt()){
    REP(i,n){
      int a = getInt();
      int b = getInt();
      int g = gcd(a, b);

      barL[i] = getInt();
      barR[i] = getInt();

      barRatioL[i] = a / g;
      barRatioR[i] = b / g;
    }

    printf("%lld\n", weight(which(n)));
  }
  return 0;
}
