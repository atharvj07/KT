#include <iostream>
using namespace std;

int main(){
  int a,b,x,ans=0;
  cin >> a >> b >> x;

  if(a < b) ans = (x+999)/1000*a;
  else if(a > b*2) ans = (x+499)/500*b;
  else {
    ans = (x/1000)*a;
    x %= 1000;
    if(x > 500){
      ans += a;
    }else if(x >0){
      ans +=b;
    }
  }

  cout << ans << endl;

  return 0;
}
