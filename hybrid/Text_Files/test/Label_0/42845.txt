(n, f) = map(int, input().split())
a = []
for i in range(n):
	temp = list(map(int, input().split()))
	if temp[0] >= temp[1]:
		temp.append(0)
	elif temp[0] * 2 >= temp[1]:
		temp.append(temp[1] - temp[0])
	else:
		temp.append(temp[0])
	a.append(temp)
a.sort(key=lambda x: x[2], reverse=True)
sum = 0
for i in range(n):
	if i < f:
		if a[i][0] * 2 >= a[i][1]:
			sum += a[i][1]
		else:
			sum += a[i][0] * 2
	elif a[i][0] >= a[i][1]:
		sum += a[i][1]
	else:
		sum += a[i][0]
print(sum)
