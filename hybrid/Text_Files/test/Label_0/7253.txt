#include <bits/stdc++.h>
using namespace std;
const long long INF = 200000000000;
int n, m;
long long czytajki[100005];
long long doSczytania[100005];
int najwiekszyMniejszy(long long poz) {
  int Min = 0, Max = m, sr;
  while (Min < Max) {
    sr = (Min + Max + 1) / 2;
    if (doSczytania[sr] <= poz)
      Min = sr;
    else
      Max = sr - 1;
  }
  return Min;
}
bool ok(long long ile) {
  int mamSczytac = 1;
  for (int i = 1; i <= n; ++i) {
    if (mamSczytac > m) return true;
    int ktory;
    if (czytajki[i] < doSczytania[mamSczytac]) {
      ktory = najwiekszyMniejszy(czytajki[i] + ile);
    } else {
      long long odl = czytajki[i] - doSczytania[mamSczytac];
      if (odl * 3 <= ile) {
        ktory = najwiekszyMniejszy(czytajki[i] + ile - 2 * odl);
      } else {
        if (odl > ile) {
          return false;
        }
        ktory = najwiekszyMniejszy(czytajki[i] + (ile - odl) / 2);
      }
    }
    mamSczytac = max(mamSczytac, ktory + 1);
  }
  return (mamSczytac > m);
}
int main() {
  scanf("%d%d", &n, &m);
  doSczytania[0] = -10;
  for (int i = 1; i <= n; ++i) cin >> czytajki[i];
  for (int i = 1; i <= m; ++i) cin >> doSczytania[i];
  long long Min = 0, Max = INF;
  while (Min < Max) {
    long long sr = (Min + Max) / 2;
    if (ok(sr))
      Max = sr;
    else
      Min = sr + 1;
  }
  cout << Min << endl;
}
