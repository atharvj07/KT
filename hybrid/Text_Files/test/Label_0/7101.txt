#include<iostream>
#include<algorithm>

using namespace std;

int nx[100][100];
int ny[100][100];
int c[100][100];

int dfs(int y,int x){
  if(c[y][x]==2){
    return 0;
  }else if(c[y][x]==1){
    return 1;
  }else{
    int r;
    c[y][x]=1;
    r=dfs(ny[y][x],nx[y][x]);
    c[y][x]=2;
    return r;
  }
}

int main(){
  for(int n;cin>>n,n;){
    for(int y=0;y<n;y++){
      for(int x=0;x<n;x++){
	cin>>nx[y][x]>>ny[y][x];
      }
    }
    fill(c[0],c[100],0);
    int ans=0;
    for(int y=0;y<n;y++){
      for(int x=0;x<n;x++){
	ans+=dfs(y,x);
      }
    }
    cout<<ans<<endl;
  }
  return 0;
}
