#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, k, mn = 1e6, sum = 0;
  cin >> n >> k;
  vector<int> income(n);
  for (int i = 0; i < n; i++) cin >> income[i];
  for (int i = 0; i < n; i++) {
    if (income[i] < 0 && k > 0) {
      income[i] *= -1;
      k--;
    }
    mn = min(mn, income[i]);
    sum += income[i];
  }
  if (k & 1) sum -= (2 * mn);
  cout << sum << endl;
  return 0;
}
