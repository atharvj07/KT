#include <bits/stdc++.h>
using namespace std;
int gcd(int m, int n) {
  if (n > m) {
    int c = n;
    n = m;
    m = c;
  }
  if (n == 0) {
    return m;
  }
  return gcd(n, m % n);
}
int main() {
  int A;
  cin >> A;
  int baseNum;
  int sum = 0;
  for (baseNum = 2; baseNum < A; baseNum++) {
    int temp = A;
    while (temp) {
      sum += temp % baseNum;
      temp /= baseNum;
    }
  }
  A -= 2;
  int tempNum = gcd(sum, A);
  sum /= tempNum;
  A /= tempNum;
  cout << sum << "/" << A << endl;
  return 0;
}
