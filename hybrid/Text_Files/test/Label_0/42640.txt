import os.path
from math import gcd, floor, ceil
from collections import *
import sys
mod = 1000000007
INF = float('inf')

def st():
	return list(sys.stdin.readline().strip())

def li():
	return list(map(int, sys.stdin.readline().split()))

def mp():
	return map(int, sys.stdin.readline().split())

def inp():
	return int(sys.stdin.readline())

def pr(n):
	return sys.stdout.write(str(n) + '\n')

def prl(n):
	return sys.stdout.write(str(n) + ' ')
if os.path.exists('input.txt'):
	sys.stdin = open('input.txt', 'r')
	sys.stdout = open('output.txt', 'w')
dx = [0, 0, 1, -1]
dy = [1, -1, 0, 0]

def solve():
	(n, m) = mp()
	l = [st() for i in range(n)]
	a = []
	b = []
	for i in range(m):
		if i % 2 == 0:
			a.append('R')
			b.append('W')
		else:
			b.append('R')
			a.append('W')
	x = []
	y = []
	for i in range(n):
		if i % 2 == 0:
			x.append(a)
			y.append(b)
		else:
			y.append(a)
			x.append(b)
	i = 0
	f = 0
	while i < n and f == 0:
		for j in range(m):
			if l[i][j] != '.' and l[i][j] != y[i][j]:
				f = 1
				break
		i += 1
	if not f:
		pr('YES')
		for i in y:
			print(*i, sep='')
		return
	f = 0
	i = 0
	while i < n and f == 0:
		for j in range(m):
			if l[i][j] != '.' and l[i][j] != x[i][j]:
				f = 1
				break
		i += 1
	if not f:
		pr('YES')
		for i in x:
			print(*i, sep='')
		return
	pr('NO')
for _ in range(inp()):
	solve()
