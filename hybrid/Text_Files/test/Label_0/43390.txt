def f(N, P, A, B):
	sa = sum(A) - P
	sb = sum(B)
	if sa <= 0:
		return -1
	AB = sorted([(A[i], B[i]) for i in range(N)], key=lambda x: -(x[1] / x[0]))
	sa = sum(A) - P
	sb = sum(B)
	for (a, b) in AB:
		if sa - a <= 0.0:
			continue
		if (sb - b) / (sa - a) <= b / a:
			sb -= b
			sa -= a
	return sb / sa
(N, P) = input().split(' ')
N = int(N)
P = float(P)
A = []
B = []
for _ in range(N):
	(a, b) = [float(x) for x in input().split(' ')]
	A.append(a)
	B.append(b)
print(f(N, P, A, B))
