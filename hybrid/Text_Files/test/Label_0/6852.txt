#include<algorithm>
#include<iostream>
#include<cstring>
#include<cstdio>
#define MN 100
#define mod 1000000007
using namespace std;
inline int read()
{
	int x=0,f=1;char ch=getchar();
	while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9'){x=x*10+ch-'0';ch=getchar();}
	return x*f;
}
int n,a[MN+5],f[MN/2+5][MN+5][MN+5],b[MN+5];
inline void R(int&x,int y){x+=y;x>=mod?x-=mod:0;}
int main()
{
	n=read();
	for(int i=1;i<n<<1;++i) a[i]=read();
	sort(a+1,a+n*2);memcpy(b,a,sizeof(b));
	int mid=a[n],num=unique(a+1,a+n*2)-a-1;
	for(int i=1;i<=num;++i) if(a[i]==mid) f[n][1][0]=1;
	for(int i=n-1;i;--i)for(int j=1;j<=num;++j)for(int k=0;k<j;++k) if(f[i+1][j][k])
	{
		int bignum=j-1-k+(b[n+(n-i)]!=b[n+(n-i)-1]),smallnum=k+(b[i]!=b[i+1]),T=(b[n+(n-i)]!=b[n+(n-i)-1])+(b[i]!=b[i+1]);
		for(int l=1;l<=smallnum;++l) R(f[i][j+T-(smallnum-l)][l-1],f[i+1][j][k]);
		for(int l=1;l<=bignum;++l) R(f[i][j+T-(bignum-l)][k+1+(b[i]!=b[i+1])],f[i+1][j][k]);
		R(f[i][j+T][k+(b[i]!=b[i+1])],f[i+1][j][k]);
	}
	int ans=0;
	for(int i=1;i<=num;++i)for(int j=0;j<num;++j) 
	//	cout<<1<<" "<<i<<" "<<j<<" "<<f[1][i][j]<<endl,
		R(ans,f[1][i][j]);
	printf("%d\n",ans);
	return 0;
}
