#include <bits/stdc++.h>
using namespace std;
const int N = 120;
const int inf = 2e5;
int a[N][5];
int getLower(int total, int frac) { return (total + frac - 1) / frac; }
int score[6] = {3000, 2500, 2000, 1500, 1000, 500};
int g[6] = {32, 16, 8, 4, 2, 1};
int t[6];
int num_solved[5];
int get_max(int l, int r, int val) {
  int ret = -1e9;
  for (int i = 0; i < 6; ++i) {
    int pl = i - 1 >= 0 ? t[i - 1] + 1 : 0;
    int pr = t[i];
    pl = max(pl, l), pr = min(pr, r);
    if (pl <= pr) {
      ret = max(ret, score[i] * val);
    }
  }
  assert(ret != -1e9);
  return ret / 250;
}
void solve2() {
  int n;
  scanf("%d", &n);
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < 5; ++j) {
      scanf("%d ", &a[i][j]);
      if (a[i][j] != -1) {
        ++num_solved[j];
      }
    }
  }
  for (int user = 0; user <= inf; ++user) {
    int num = user + n;
    for (int j = 0; j < 6; ++j) t[j] = num / g[j];
    int ret = 0;
    for (int i = 0; i < 5; ++i) {
      if (a[0][i] == -1 && a[1][i] == -1)
        continue;
      else if (a[0][i] == -1 || a[1][i] == -1) {
        ret += get_max(num_solved[i], num_solved[i],
                       a[0][i] == -1 ? -(250 - a[1][i]) : 250 - a[0][i]);
      } else {
        int r = -1e9;
        int w = -a[0][i] + a[1][i];
        ret += get_max(num_solved[i], num_solved[i] + user, w);
      }
    }
    if (ret > 0) {
      printf("%d\n", user);
      return;
    }
  }
  printf("-1\n");
}
void solve() { solve2(); }
inline bool exists_test0(const std::string& name);

int main() {
  solve();
  return 0;
}

  return f.good();
}

  if (FILE* file = fopen(name.c_str(), "r")) {
    fclose(file);
    return true;
  } else {

  }
}
