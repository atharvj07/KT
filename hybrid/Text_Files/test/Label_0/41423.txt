n = int(input())
l = []
d = {}
for i in range(n):
	d[i + 1] = []
for i in range(n - 2):
	l.append(list(map(int, input().split())))
	d[l[-1][0]].append(i)
	d[l[-1][1]].append(i)
	d[l[-1][2]].append(i)
t = 0
for i in range(n):
	if len(d[i + 1]) == 1:
		t = i + 1
		break
ans = [t]
a = t
for i in l[d[t][0]]:
	if len(d[i]) == 2:
		b = i
	elif len(d[i]) == 3:
		c = i
ans.append(b)
ans.append(c)
count = 0
while count < n - 3:
	for i in d[b]:
		if i not in d[a] and i in d[c]:
			t = i
	for i in l[t]:
		if i != b and i != c:
			ans.append(i)
			(a, b, c) = (b, c, i)
			break
	count += 1
print(*ans)
