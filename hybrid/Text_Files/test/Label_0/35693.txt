import java.io.BufferedReader;
import java.io.File;
import java.io.FileNotFoundException;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Scanner;

import javax.swing.text.Position;

public class C381
{
	public static ArrayList<Long> strToInts(String str, String token)
	{
		ArrayList<Long> result = new ArrayList<>();
		ArrayList<String> strs = new ArrayList<>(
				Arrays.asList(str.split(token)));
		for (String string : strs)
		{
			result.add(Long.parseLong(string));
		}
		return result;
	}

	public static void main(String[] args) throws IOException
	{

		// Parse the input
		//Scanner in = new Scanner(new File("input.txt"));
		 BufferedReader in = new BufferedReader(new
		 InputStreamReader(System.in));
		ArrayList<Long> parameters = strToInts(in.readLine(), " ");

		ArrayList<Long> sequence = new ArrayList<Long>();
		ArrayList<ArrayList<Long>> stages = new ArrayList<>();

		// Generate the first 100 000
		Long numberOfStages = parameters.get(0);
		for (int i = 0; i < numberOfStages; i++)
		{
			stages.add(strToInts(in.readLine(), " "));
		}
		for (int i = 0; i < numberOfStages; i++)
		{
			if (sequence.size() > 100002)
			{
				break;
			}

			ArrayList<Long> line = stages.get(i);
			// if we are adding
			if (line.get(0) == 1)
			{
				sequence.add(line.get(1));
			}
			// If we are copying
			else
			{
				Long l = line.get(1);
				Long c = line.get(2);
				for (int j = 0; j < c; j++)
				{
					for (int j2 = 0; j2 < l; j2++)
					{
						sequence.add(sequence.get(j2));
					}
				}
			}
		}

		// Make the size before array
		ArrayList<Long> sizeBefore = new ArrayList<Long>();
		sizeBefore.add(0l);
		for (int i = 0; i < stages.size(); i++)
		{
			if (stages.get(i).get(0) == 1)
				sizeBefore.add(sizeBefore.get(sizeBefore.size() - 1) + 1);
			else
				sizeBefore.add(sizeBefore.get(sizeBefore.size() - 1)
						+ stages.get(i).get(1) * stages.get(i).get(2));
		}

		// Print the required numbers
		in.readLine();
		ArrayList<Long> required = strToInts(in.readLine(), " ");
		CurrentOperation currentOperation = new CurrentOperation();
		currentOperation.num = 0;
		for (Long req : required)
		{
			System.out.print(findNum((req - 1), sizeBefore, sequence,
					currentOperation, stages) + " ");
		}

	}

	private static Long findNum(Long i, ArrayList<Long> sizeBefore,
			ArrayList<Long> sequence, CurrentOperation currentOperation,
			ArrayList<ArrayList<Long>> stages)
	{
		if (i <= 100000)
		{
			int tempInt = i.intValue();
			
			return sequence.get(tempInt);
		}
		// keep adding the current operation till we either reach our index or
		// the one after it it
		while (sizeBefore.get(currentOperation.num) < i)
			currentOperation.num++;

		// If we're lucky and here we just added a number
		if (sizeBefore.get(currentOperation.num) == i)
		{
			return stages.get(currentOperation.num).get(1);
		}
		// If in that operation we added a bunch
		else
		{
			currentOperation.num--;
			Long increase = i - sizeBefore.get(currentOperation.num);
			Long position = (increase)
					% stages.get(currentOperation.num).get(1);
			int intPos = position.intValue();
			return sequence.get(intPos);
		}

	}

	public static class CurrentOperation
	{
		public int num;
	}

}
