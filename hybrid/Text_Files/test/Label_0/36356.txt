
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;

public class Main {

    static StreamTokenizer st;
    static int a, b, p, q;
    static double r;
    static long sol;

    public static void main(String[] args) throws IOException {
        st = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
        int n = ri();
        sol = -1;
        while (n-- > 0) {
            a = ri();
            b = ri();
            p = ri();
            q = ri();
            sol = -1;
            search(1, (long) 2e9);
            if (sol < 0) {
                System.out.println(-1);
            } else {
                long x = sol * p - a;
                System.out.println(x + (sol * q) - b - x);
            }
        }
    }

    static void search(long lo, long hi) {
        if (lo > hi) {
            return;
        }
        long mid = (lo + hi) >> 1;
        long x = p * mid;
        long y = q * mid;
        if (x >= a && (x - a + b) <= y) {
            sol = mid;
            if (hi == mid) {
                return;
            }
            hi = mid;
        } else {
            if (lo == mid) {
                return;
            }
            lo = mid;
        }
        search(lo, hi);
    }

    static int ri() throws IOException {
        st.nextToken();
        return (int) st.nval;
    }
}
