#include <bits/stdc++.h>
using namespace std;
int main() {
  ios::sync_with_stdio(false);
  cin.tie(NULL);
  long long n, k;
  cin >> n >> k;
  long long low = 1, high = k / n + 1;
  while (low < high) {
    long long mid = (low + high) / 2;
    if (mid * n >= k)
      high = mid;
    else
      low = mid + 1;
  }
  cout << low << '\n';
}
