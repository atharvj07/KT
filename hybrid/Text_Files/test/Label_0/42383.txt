import sys
input = sys.stdin.readline
(n, k) = map(int, input().split())
a = [list(map(int, input().split())) for i in range(k)]
g = [[] for i in range(n + 1)]
for i in range(n):
	x = a[0][i]
	s = set(a[0][i + 1:])
	for j in range(k):
		idx = a[j].index(x)
		s1 = set(a[j][idx + 1:])
		s = s & s1
	for y in s:
		g[x].append(y)
dist = [-1] * (n + 1)

def dfs(ver):
	if dist[ver] != -1:
		return dist[ver]
	res = 0
	for to in g[ver]:
		res = max(res, dfs(to) + 1)
	dist[ver] = res
	return res
MAX = max((dfs(i) for i in range(1, n + 1)))
print(MAX + 1)
