n = int(input())
s = list(input())

def call(x):
	if x == '(':
		return 1
	else:
		return -1
s = list(map(call, s))
if sum(s) != 0:
	print(0)
	print(1, 1)
else:
	ans = 0
	pr = (1, 1)
	for i in range(n - 1):
		for j in range(i, n):
			lm = 10 ** 9
			sm = 0
			ct = 0
			(s[i], s[j]) = (s[j], s[i])
			for k in range(n):
				sm += s[k]
				if sm < lm:
					lm = sm
					ct = 1
				elif sm == lm:
					ct += 1
			if ct > ans:
				ans = ct
				pr = (i + 1, j + 1)
			(s[i], s[j]) = (s[j], s[i])
	print(ans)
	print(*pr)
