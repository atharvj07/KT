import io, os
input = io.BytesIO(os.read(0, os.fstat(0).st_size)).readline
import sys

def judge(L, k):
	L.sort()
	return L[1] >= k

def solve(n, k, A):
	if k not in A:
		return 'no'
	l = len(A)
	if l == 1:
		return 'yes'
	if l == 2:
		if min(A) == k:
			return 'yes'
		return 'no'
	for i in range(l - 2):
		L = A[i:i + 3]
		if judge(L, k):
			return 'yes'
	return 'no'

def main():
	t = int(input())
	for _ in range(t):
		(n, k) = map(int, input().split())
		A = list(map(int, input().split()))
		ans = solve(n, k, A)
		sys.stdout.write(ans + '\n')
main()
