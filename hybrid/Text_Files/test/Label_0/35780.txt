import java.io.*;
import java.util.*;

public class C {

    static int SIZE = 62; // sub2: 16, sub3: 32, sub4: 62
    static long MOD = 998244353;
    
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));

        int N = Integer.parseInt(in.readLine());
        long[][][] words = new long[11][SIZE][SIZE];
        TreeSet<String> ts = new TreeSet<String>();
        for(int n = 0; n < N; n++) {
            String w = in.readLine();
            if(ts.contains(reverse(w))) continue;
            ts.add(w);
            int a = charToInt(w.charAt(0));
            int b = charToInt(w.charAt(w.length()-1));
            words[w.length()][a][b]++;
            if(!isPal(w)) words[w.length()][b][a]++;
        }

        long sum = 0;
        
        for(int LEN = 3; LEN <= 10; LEN++) {
            long[][][] dyn = new long[SIZE][SIZE][SIZE];
            for(int a = 0; a < SIZE; a++) {
                for(int b = a; b < SIZE; b++) {
                    for(int c = b; c < SIZE; c++) {
                        for(int x = 0; x < SIZE; x++) {
                            long add = (words[LEN][a][x]*words[LEN][b][x])%MOD;
                            add = (add*words[LEN][c][x])%MOD;
                            dyn[a][b][c] += add;
                            dyn[a][b][c] %= MOD;
                        }
                        dyn[a][c][b] = dyn[a][b][c];
                        dyn[b][a][c] = dyn[a][b][c];
                        dyn[b][c][a] = dyn[a][b][c];
                        dyn[c][a][b] = dyn[a][b][c];
                        dyn[c][b][a] = dyn[a][b][c];
                    }}}
            for(int a = 0; a < SIZE; a++) {
                for(int b = a; b < SIZE; b++) {
                    for(int c = b; c < SIZE; c++) {
                        for(int d = c; d < SIZE; d++) {
                            long add = (dyn[a][b][c]*dyn[b][c][d])%MOD;
                            add *= (dyn[a][c][d]*dyn[a][b][d])%MOD;
                            add %= MOD;
                            //if(add > 0) {
                            //    System.err.print(a+", "+b+", "+c+", "+d+"  ");
                            //    System.err.print(add+" -> ");
                            //}
                            if(a == b && b == c && c == d)
                                add *= 1;
                            else if((a == b && b == c) || // d
                                    (a == b && b == d) || // c
                                    (a == c && c == d) || // b
                                    (b == c && c == d))   // a
                                add *= 4;
                            else if(a == b && c == d)
                                add *= 6;
                            else if(a == b || b == c || c == d)
                                    add *= 12;
                            else
                                add *= 24;
                            //if(add > 0)
                            //    System.err.println(add);
                            sum += add;
                            sum %= MOD;
                        }}}}
            
        }

        System.out.println(sum);

        in.close();
    }

    static int charToInt(char c) {
        if('a' <= c && c <= 'z')
            return c-'a';
        else if('A' <= c && c <= 'Z')
            return (SIZE==62 ? 26 : 16) + c-'A';
        else
            return 2*26+c-'0';
    }

    static boolean isPal(String s) {
        int n = s.length();
        for(int i = 0; i <= n/2; i++)
            if(s.charAt(i) != s.charAt(n-1-i))
                return false;
        return true;
    }

    static String reverse(String s) {
        StringBuilder sb = new StringBuilder();
        for(int i = s.length()-1; i >= 0; i--)
            sb.append(s.charAt(i));
        return sb.toString();
    }
}
