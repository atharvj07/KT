def answer(n, k):
	lst = []
	for i in range(1, n + 1):
		if i == k:
			pass
		elif i <= k // 2 and k % 2 == 1:
			pass
		elif i < k // 2 and k % 2 == 0:
			pass
		else:
			lst.append(i)
	print(len(lst))
	print(' '.join(map(str, lst)))
	return

def main():
	t = int(input())
	n = [0] * t
	k = [0] * t
	for j in range(t):
		(n[j], k[j]) = [int(i) for i in input().split()]
	for j in range(t):
		answer(n[j], k[j])
main()
