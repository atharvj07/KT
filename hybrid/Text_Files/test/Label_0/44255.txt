from collections import defaultdict
s = [c for c in input().strip()]
freq = defaultdict(lambda : 0)
lst = {}
for (i, c) in enumerate(s):
	lst[c] = i
	freq[c] ^= 1
skip = len(s) & 1
odds = []
for (k, v) in freq.items():
	if v == 0:
		continue
	odds.append(k)
odds.sort()
m = len(odds) >> 1
for i in range(m):
	c = odds[i]
	r = odds[len(odds) - 1 - i]
	s[lst[r]] = c
idx = defaultdict(lambda : [])
for (i, c) in enumerate(s):
	idx[c].append(i)
keys = [c for c in idx]
keys.sort()
pref = []
suff = []
mid = ''
for c in keys:
	I = idx[c]
	m = len(I) >> 1
	pref.append(c * m)
	suff.append(c * m)
	if len(I) & 1:
		mid = c
pref = ''.join(pref)
suff = ''.join(suff)
suff = suff[::-1]
print(pref + mid + suff)
