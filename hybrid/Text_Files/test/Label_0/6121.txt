#include <bits/stdc++.h>
using namespace std;
vector<string> numbers;
vector<int> length;
int eval(int matched, int reqd) {
  int count = 0;
  for (int i = 0; i < length.size(); i++) {
    int j;
    for (j = matched; j < length[i]; j++) {
      if (numbers[i][j] == '1') break;
    }
    count += length[i] - j;
    int newLength = j;
    count += abs(matched + reqd - newLength);
  }
  return count;
}
int main() {
  int n;
  cin >> n;
  vector<int> a(n);
  length.resize(n);
  numbers.resize(n);
  int maxLength = 20;
  for (int i = 0; i < n; i++) {
    cin >> a[i];
    int copy = a[i];
    string temp = "";
    while (copy) {
      if (copy % 2 == 1)
        temp = "1" + temp;
      else
        temp = "0" + temp;
      copy /= 2;
    }
    length[i] = temp.size();
    maxLength = min(maxLength, length[i]);
    while (temp.size() < 20) {
      temp = temp + "0";
    }
    numbers[i] = temp;
  }
  for (int i = 0; i < n; i++) {
    for (int l = 0; l < maxLength; l++) {
      if (numbers[i][l] != numbers[0][l]) {
        maxLength = l;
        break;
      }
    }
  }
  int res = INT_MAX;
  for (int len = maxLength; len <= 20; len++) {
    res = min(eval(maxLength, len - maxLength), res);
  }
  cout << res << endl;
}
