prime = 998244353

def power(x, y, prime):
	b = bin(y)[2:]
	start = x
	answer = 1
	for i in range(len(b)):
		b2 = b[len(b) - 1 - i]
		if b2 == '1':
			answer = answer * start % prime
		start = start * start % prime
	return answer

def process(A):
	P = 100
	Q = 1
	for p in A[:-1]:
		Q = p * Q % prime
		P = 100 * (P + Q) % prime
	Q = Q * A[-1] % prime
	Qinv = power(Q, prime - 2, prime)
	answer = P * Qinv % prime
	return answer
n = int(input())
A = [int(x) for x in input().split()]
print(process(A))
