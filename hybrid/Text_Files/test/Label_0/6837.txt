bool ispalin(string s){
	int i=0 , j=s.size()-1;
	while(i<j){
		if(s[i]!=s[j]){
			return false;
		}
		++i;
		--j;
	}
	return true;
}

	for(ll i=2;i*i<=a;++i){
		if(a%i==0){

}
