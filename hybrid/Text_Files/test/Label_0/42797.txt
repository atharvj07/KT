import sys
input = sys.stdin.readline
for _ in range(int(input())):
	n = int(input())
	b = [0] + list(map(int, input().split())) + [2 * n + 1]
	bb = []
	interval = []
	for i in range(n + 1):
		interval.append(b[i + 1] - b[i] - 1)
	min_x = 0
	rem = 0
	for e in interval[n - 1::-1]:
		rem += 1
		if e < rem:
			rem -= e
		else:
			rem = 0
	min_x = rem
	max_x = 0
	rem = 0
	for e in interval[1:]:
		rem += 1
		if e < rem:
			rem -= e
		else:
			rem = 0
	max_x = n - rem
	print(max_x - min_x + 1)
