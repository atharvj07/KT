def solve():
	n = int(input())
	arr = [int(i) for i in input().split()]
	s = sum(arr)
	ss = 2 * s / n
	if not ss.is_integer():
		print(0)
		return
	ss = int(ss)
	c_map = {}
	for i in arr:
		if i not in c_map:
			c_map[i] = 0
		c_map[i] += 1
	count = 0
	for (i, j) in c_map.items():
		if ss - i in c_map:
			if ss - i == i:
				count += j * (j - 1)
			else:
				count += j * c_map[ss - i]
	print(int(count / 2))
t = int(input())
for i in range(t):
	solve()
