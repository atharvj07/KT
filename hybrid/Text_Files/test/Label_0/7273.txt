#include<stdio.h>

static int so[10001],hyo[1300];

int main(){
	int i,j,k,n,m;
	
	so[1]=1;
	for(i=2;i*i<=10000;i++){
		for(j=2*i;j<=10000;j+=i){
			so[j]=1;
		}
	}
	hyo[0]=0;
	j=0;
	for(i=1;i<=10000;i++){
		if(!so[i]){
			++j;
			hyo[j]=hyo[j-1]+i;
		}
	}
	k=j;
	//printf("%d\n",k);
	while(0<=scanf("%d",&n)){
		if(n==0)break;
		
		m=0;
		
		for(i=0;i<k;++i){
			for(j=i+1;j<=k;++j){
				if(hyo[j]-hyo[i]==n)++m;
			}
		}
		printf("%d\n",m);
	}
	return 0;
}
