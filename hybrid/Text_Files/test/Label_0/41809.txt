import sys
input = sys.stdin.readline
for _ in range(int(input().rstrip())):
	n = int(input().rstrip())
	p = list(map(int, input().rstrip().split()))
	tmp = [i for i in range(1, n + 1)]
	result = [0] * n
	cnt = 0
	while result.__contains__(0):
		tmp2 = tmp.copy()
		for i in range(n):
			tmp[p[i] - 1] = tmp2[i]
		cnt += 1
		for i in range(n):
			if tmp[i] == i + 1 and result[i] == 0:
				result[i] = cnt
	print(*result)
