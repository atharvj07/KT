(n, k, M) = [int(i) for i in input().split(' ')]
tasks = [int(i) for i in input().split(' ')]
tasks = sorted(tasks)
ans = 0
for i in range(n + 1):
	remaining_time = M - i * sum(tasks)
	if remaining_time < 0:
		break
	value = (k + 1) * i
	for j in range(k):
		for take in range(n - i):
			if remaining_time >= tasks[j]:
				value += 1
				remaining_time -= tasks[j]
	ans = max(ans, value)
print(ans)
