#include <bits/stdc++.h>
using namespace std;
const int N = 1000005, P = 998244353;
int n, lenl, lenr, extl[N], extr[N], f[N], sum[N];
char s[N], l[N], r[N];
inline void exkmp(char *s, int n, char *t, int m, int *ext) {
  static int nxt[N];
  nxt[1] = m, nxt[2] = 0;
  while (t[nxt[2] + 1] == t[nxt[2] + 2]) ++nxt[2];
  int p = 2;
  for (int i = 3; i <= m; ++i) {
    if (i + nxt[i - p + 1] < p + nxt[p])
      nxt[i] = nxt[i - p + 1];
    else {
      int j = max(0, p + nxt[p] - i);
      while (t[j + 1] == t[i + j]) ++j;
      nxt[i] = j;
      p = i;
    }
  }
  ext[1] = 0;
  while (s[ext[1] + 1] == t[ext[1] + 1] && ext[1] < m) ++ext[1];
  p = 1;
  for (int i = 2; i <= n; ++i) {
    if (i + nxt[i - p + 1] < p + ext[p])
      ext[i] = nxt[i - p + 1];
    else {
      int j = max(0, p + ext[p] - i);
      while (t[j + 1] == s[i + j] && j < m) ++j;
      ext[i] = j;
      p = i;
    }
  }
}
int main() {
  scanf("%s%s%s", s + 1, l + 1, r + 1);
  n = strlen(s + 1), lenl = strlen(l + 1), lenr = strlen(r + 1);
  exkmp(s, n, l, lenl, extl), exkmp(s, n, r, lenr, extr);
  sum[0] = f[0] = 1;
  for (int i = 1; i <= n; ++i) {
    if (i >= lenl) {
      int L, R = i - lenl;
      if (extl[R + 1] < lenl && s[R + extl[R + 1] + 1] < l[extl[R + 1] + 1])
        --R;
      if (i >= lenr) {
        L = i - lenr;
        if (extr[L + 1] < lenr && s[L + extr[L + 1] + 1] > r[extr[L + 1] + 1])
          ++L;
      } else
        L = 0;
      f[i] = (L ? (sum[R] - sum[L - 1]) % P : sum[R]);
    }
    if (s[i] == '0' && l[1] == '0') (f[i] += f[i - 1]) %= P;
    if (s[i + 1] != '0')
      sum[i] = (sum[i - 1] + f[i]) % P;
    else
      sum[i] = sum[i - 1];
  }
  return printf("%d", (f[n] + P) % P), 0;
}
