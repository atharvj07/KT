#include <bits/stdc++.h>
using namespace std;
bool used[105];
int n;
int x[105], y[105];
void dfs(int v) {
  if (used[v]) return;
  used[v] = true;
  for (int i = 0; i < n; ++i) {
    if ((i != v) && ((x[v] == x[i]) || (y[v] == y[i]))) {
      dfs(i);
    }
  }
}
int main() {
  cin >> n;
  for (int i = 0; i < n; ++i) {
    cin >> x[i] >> y[i];
    used[i] = false;
  }
  int c = 0;
  for (int i = 0; i < n; ++i) {
    if (!used[i]) {
      dfs(i);
      ++c;
    }
  }
  cout << c - 1;
  return 0;
}
