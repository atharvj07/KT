(n, k) = map(int, input().split())
ans = [i + 1 for i in range(n)]
if k == n:
	print(-1)
else:
	temp = n - k
	if temp == 1:
		print(*ans)
		exit()
	if temp % 2 == 1:
		temp -= 1
		for i in range(temp // 2):
			j = 2 * i + 1
			m = j + 1
			(ans[j], ans[m]) = (ans[m], ans[j])
		print(*ans)
	else:
		temp = temp // 2
		for i in range(temp):
			j = 2 * i
			m = j + 1
			(ans[j], ans[m]) = (ans[m], ans[j])
		print(*ans)
