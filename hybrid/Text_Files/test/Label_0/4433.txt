#include <bits/stdc++.h>
using namespace std;
int main(void) {
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    int arr[n];
    int max = 0;
    for (int i = 0; i < n; i++) {
      cin >> arr[i];
      if (max < arr[i]) max = arr[i];
    }
    string s = "";
    for (int i = 0; i < max + 1; i++) {
      s = s + 'a';
    }
    cout << s << "\n";
    char ch = 'b';
    for (int i = 0; i < n; i++) {
      if (ch == 'z') ch = 'b';
      s[arr[i]] = ch;
      ch++;
      cout << s << "\n";
    }
  }
  return 0;
}
