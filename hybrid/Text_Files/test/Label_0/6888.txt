#include <bits/stdc++.h>
using namespace std;
int main() {
  int t;
  cin >> t;
  while (t--) {
    int n, k;
    cin >> n >> k;
    int a[n], b[n];
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    for (int i = 0; i < n; i++) {
      cin >> b[i];
    }
    sort(a, a + n);
    sort(b, b + n);
    long int sum = 0;
    int ai = n - 1, bi = n - 1, ki = 0;
    for (int i = 0; i < n; i++) {
      if (ki == k) {
        sum += a[ai];
        ai--;
        continue;
      }
      if (b[bi] > a[ai]) {
        sum += b[bi];
        bi--;
        ki++;
      } else {
        sum += a[ai];
        ai--;
      }
    }
    cout << sum << '\n';
  }
  return 0;
}
