#include <bits/stdc++.h>
#pragma GCC optimize("O3")
#pragma GCC target("sse,sse2,sse3,ssse3,sse4,popcnt,abm,mmx")
using namespace std;
int a, b, c, d, n, m, k;
vector<int> sm[100002];
vector<pair<int, int> > mem[100002];
int ans[100002];
pair<int, int> pl[100002], pr[100002];
bool cool[41];
int p[41], pc, rev[41];
vector<int> mods[41];
int fmet[41], fval[41];
int sval[41];
int check(pair<int, int> pl[], int sl, pair<int, int> pr[], int sr) {
  memset(fmet, -1, sizeof(fmet));
  int last = 0;
  for (int _n((sl)-1), i(0); i <= _n; i++) {
    int pos = pl[i].first;
    int len = pl[i].second;
    bool ok = 1;
    for (int _n(((int)((mods[len]).size())) - 1), j(0); j <= _n; j++) {
      int cmod = mods[len][j];
      int rem = pos % p[cmod];
      if (fmet[cmod] != -1 && fval[cmod] != rem) {
        ok = 0;
        break;
      }
      if (fmet[cmod] == -1) {
        fmet[cmod] = i;
        fval[cmod] = rem;
      }
    }
    if (!ok) break;
    ++last;
  }
  int ans = last;
  memset(sval, -1, sizeof(sval));
  for (int _n((sr)-1), i(0); i <= _n; i++) {
    int pos = pr[i].first;
    int len = pr[i].second;
    bool ok = 1;
    for (int _n(((int)((mods[len]).size())) - 1), j(0); j <= _n; j++) {
      int cmod = mods[len][j];
      int rem = pos % p[cmod];
      if (sval[cmod] != -1 && sval[cmod] != rem) {
        ok = 0;
        break;
      }
      sval[cmod] = rem;
      if (fmet[cmod] != -1 && fval[cmod] != rem) {
        last = min(last, fmet[cmod]);
      }
    }
    if (!ok) break;
    ans = max(ans, i + last);
  }
  return ans;
}
inline int fnd(int v, int num) {
  int pos = lower_bound((mem[v]).begin(), (mem[v]).end(), make_pair(num, -1)) -
            mem[v].begin();
  if (pos >= (int)((mem[v]).size()) || mem[v][pos].first != num) return -1;
  return mem[v][pos].second;
}
void solve(int l, int r) {
  if (l >= r) return;
  if (l == r - 1) {
    for (int _n(((int)((sm[l]).size())) - 1), i(0); i <= _n; i++) {
      int v = sm[l][i];
      ans[v] = max(ans[v], 1);
    }
    return;
  }
  int xx = (r + l) / 2;
  solve(l, xx);
  solve(xx + 1, r);
  for (int _n(((int)((sm[xx]).size())) - 1), i(0); i <= _n; i++) {
    int v = sm[xx][i];
    int sl = 0, sr = 0;
    for (int j = (xx), _b = (l); j >= _b; j--) {
      int pos = fnd(j, v);
      if (pos == -1) break;
      pl[sl++] = make_pair(pos, (int)((sm[j]).size()));
    }
    for (int _n(r), j(xx); j <= _n; j++) {
      int pos = fnd(j, v);
      if (pos == -1) break;
      pr[sr++] = make_pair(pos, (int)((sm[j]).size()));
    }
    int cur = check(pl, sl, pr, sr);
    ans[v] = max(ans[v], cur);
  }
}
int main() {
  scanf("%d%d", &n, &m);
  for (int _n((n)-1), i(0); i <= _n; i++) {
    scanf("%d", &a);
    sm[i].resize(a);
    mem[i].resize(a);
    for (int _n((a)-1), j(0); j <= _n; j++) {
      scanf("%d", &sm[i][j]);
      --sm[i][j];
      mem[i][j] = make_pair(sm[i][j], j);
    }
    sort((mem[i]).begin(), (mem[i]).end());
  }
  memset((cool), 0, sizeof(cool));
  for (int _n(40), i(2); i <= _n; i++) {
    cool[i] = 1;
    for (int j = 2; j * j <= i; ++j) {
      if (i % j == 0) {
        cool[i] = 0;
        break;
      }
    }
  }
  cool[4] = cool[8] = cool[16] = cool[32] = 1;
  cool[9] = cool[27] = 1;
  cool[25] = 1;
  pc = 0;
  for (int _n(40), i(2); i <= _n; i++) {
    if (cool[i]) p[pc++] = i;
  }
  for (int _n((pc)-1), i(0); i <= _n; i++) {
    rev[p[i]] = i;
  }
  for (int _n((40)), i(1); i <= _n; i++) {
    for (int _n((pc)-1), j(0); j <= _n; j++) {
      if (i % p[j] == 0) {
        mods[i].push_back(j);
      }
    }
  }
  memset((ans), 0, sizeof(ans));
  solve(0, n);
  for (int _n((m)-1), i(0); i <= _n; i++) {
    printf("%d\n", ans[i]);
  }
}
