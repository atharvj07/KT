n = int(input())
x = [i for i in input().split()]
coin25 = 0
coin50 = 0
res = 'YES'
for a in x:
	if a == '25':
		coin25 += 1
	elif a == '50':
		if coin25 > 0:
			coin25 -= 1
			coin50 += 1
		else:
			res = 'NO'
			break
	elif coin50 >= 1 and coin25 >= 1:
		coin25 -= 1
		coin50 -= 1
	elif coin25 >= 3:
		coin25 -= 3
	else:
		res = 'NO'
	if res == 'NO':
		break
print(res)
