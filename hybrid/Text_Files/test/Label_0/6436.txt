#include <bits/stdc++.h>
using namespace std;
int t;
int n;
struct Pos {
  int x, y;
} a[(int)2000];
bool cmp(const Pos& x, const Pos& y) {
  if (x.x != y.x)
    return x.x < y.x;
  else
    return x.y < y.y;
}
int main() {
  cin >> t;
  while (t--) {
    cin >> n;
    for (int i = 1; i <= n; i++) {
      cin >> a[i].x >> a[i].y;
    }
    sort(a + 1, a + n + 1, cmp);
    bool flg = true;
    for (int i = 1; i <= n; i++) {
      for (int j = i + 1; j <= n; j++) {
        if (a[i].x < a[j].x && a[i].y > a[j].y) {
          flg = false;
          break;
        }
      }
      if (!flg) break;
    }
    if (!flg) {
      puts("NO");
      continue;
    } else
      puts("YES");
    a[0].x = a[0].y = 0;
    for (int i = 0; i < n; i++) {
      while (a[i].x != a[i + 1].x) {
        a[i].x++;
        putchar('R');
      }
      while (a[i].y != a[i + 1].y) {
        a[i].y++;
        putchar('U');
      }
    }
    puts("");
  }
}
