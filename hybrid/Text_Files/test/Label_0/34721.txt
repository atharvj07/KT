import java.util.Scanner;

public class B {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        long pow[] = new long[60];
        pow[0] = 1;
        for (int i = 1; i < 60; i++) {
            pow[i] = pow[i - 1] * 2;
        }
        long x[] = new long[n];
        for (int i = 0; i < n; i++) {
            long buf = 0;
            char c[] = in.next().toCharArray();
            for (int j = m - 1; j >= 0; j--) {
                if (c[j] == '#') {
                    buf += pow[j];
                }
            }
            x[i] = buf;
        }
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if((x[i] & x[j]) != 0 && (x[i] & x[j]) != x[i]){
                    System.out.println("No");
                    return;
                }
            }
        }
        System.out.println("Yes");
    }
}
