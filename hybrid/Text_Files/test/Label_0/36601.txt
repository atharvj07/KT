import java.util.*;

public class x79B {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        StringBuilder SB = new StringBuilder("YES\n");
        int N = sc.nextInt();
        Stack<StringBuilder> x = new Stack<StringBuilder>();
        Stack<StringBuilder> y = new Stack<StringBuilder>();
        int[] nums = new int[N];
        int[] sorted = new int[N];
        //int[][] insort = new int[N][2];
        for (int a = 0; a < N; a++) {
            nums[a] = sc.nextInt();
            sorted[a] = nums[a];
        //  insort[a][0]=nums[a];
    //      insort[a][1]=a;
    //      System.out.print(insort[a][0]+" ");
        //  System.out.println(insort[a][1]);
        }
        Arrays.sort(sorted);
    //  Arrays.sort(insort);
        //for (int a = 0; a < N; a++) {
        //  System.out.print(insort[a][0]+" ");
            //System.out.println(insort[a][1]);
        //}
        
        int largest = sorted[N - 1];
        String[] master = new String[N];
        int done = 0;
        StringBuilder zero = new StringBuilder("0");
        StringBuilder one = new StringBuilder("1");
        x.add(one);
        x.add(zero);
        for (int a = 1; a <= largest; a++) {
            int q = x.size();
            if (x.size()<N-done&&a != 1) {
                
                for (int c = 0; c < q; c++) {
                    StringBuilder temp =  x.pop();
                    StringBuilder temp2 =  new StringBuilder(temp.toString());
                    y.add(temp2.append("1"));
                    y.add(temp.append("0"));
                    
                }
                q = y.size();
                for (int c = 0; c < q; c++) {
                    x.add(y.pop());
                }
            }
            int z = 0;
                while (sorted[done] == a) {
                    if(x.isEmpty()){
                        System.out.print("NO");
                        System.exit(0);
                    }
                    for(int e=0;e<N;e++){
                        if(nums[e]==a){
                            z=e;
                            break;
                        }
                    }
                    StringBuilder temp = x.pop();
                    for(int t=temp.length();t<a;t++){
                        temp.append("0");
                    }
                    master[z] = temp.toString();
                    nums[z] = -1;
                    done++;
                    if(done==N)break;
                }
            
        }
        for (int b = 0; b < N; b++) {
            SB.append(master[b] + "\n");
        }
        System.out.print(SB.toString());

        System.exit(0);
    }
}
