#include <bits/stdc++.h>
using namespace std;
long long x[100010];
int main() {
  int n, k, l;
  cin >> n >> k >> l;
  for (int i = 0; i < n * k; i++) {
    cin >> x[i];
  }
  sort(x, x + n * k);
  int index = upper_bound(x, x + n * k, x[0] + l) - x;
  if (index < n) {
    printf("0\n");
    return 0;
  }
  long long sum = 0;
  for (int i = 0; n && i < index;) {
    sum += x[i];
    i += min(k, max(index - i - n + 1, 1));
    n--;
  }
  cout << sum << endl;
  return 0;
}
