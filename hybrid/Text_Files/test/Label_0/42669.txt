def check_infinite(N, P, A, B):
	if sum(A) <= P:
		return True
	return False

def f(N, P, A, B):
	if check_infinite(N, P, A, B) == True:
		return -1
	sa = sum(A)
	sb = sum(B)
	remain = N
	deleted = set()
	while True:
		found = False
		i = 0
		while i < N:
			if i in deleted:
				i += 1
				continue
			if remain <= 1:
				break
			if sa - P - A[i] <= 0.0:
				i += 1
				continue
			if (sb - B[i]) / (sa - A[i] - P) <= B[i] / A[i]:
				remain -= 1
				sb -= B[i]
				sa -= A[i]
				deleted.add(i)
				found = True
			i += 1
		if not found:
			break
	return sb / (sa - P)
(N, P) = input().split(' ')
N = int(N)
P = float(P)
A = []
B = []
for _ in range(N):
	(a, b) = [float(x) for x in input().split(' ')]
	A.append(a)
	B.append(b)
print(f(N, P, A, B))
