#include <bits/stdc++.h>
using namespace std;
int a[2000], b[2000];
int main() {
  bool flag;
  int n, m, l, r, x, z, y;
  while (~scanf("%d", &n)) {
    z = a[0] = 0;
    memset(b, 0, sizeof(b));
    for (int i = 1; i <= n; i++) {
      scanf("%d", &a[i]);
      x = i;
      while (x--) {
        if (a[x] > a[i]) {
          b[i]++;
          continue;
        }
      }
      z = (z + b[i]) % 2;
    }
    scanf("%d", &m);
    for (int i = 0; i < m; i++) {
      scanf("%d%d", &l, &r);
      y = r - l + 1;
      if (y % 2) y--;
      if (y % 4) z = z ^ 1;
      if (z == 1)
        printf("odd\n");
      else if (z == 0)
        printf("even\n");
    }
  }
  return 0;
}
