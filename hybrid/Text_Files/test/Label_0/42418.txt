from collections import defaultdict
(n, m) = map(int, input().split())
freq = defaultdict()
l = list(map(int, input().split()))
ans = []
for i in l:
	if i not in freq:
		freq[i] = 1
	else:
		freq[i] += 1
	if len(freq) == n:
		ans.append('1')
		freq = {k: v - 1 for (k, v) in freq.items() if v > 1}
	else:
		ans.append('0')
print(''.join(ans))
