n = int(input())
s = input().strip()
a = list(map(int, input().split()))
t = 'hard'
m = len(t)
dp = [[-1] * m for i in range(n)]
dp[n - 1][m - 1] = a[n - 1] if s[n - 1] == t[m - 1] else 0
for i in reversed(range(n - 1)):
	dp[i][m - 1] = dp[i + 1][m - 1] + (a[i] if s[i] == t[m - 1] else 0)
for i in reversed(range(n)):
	for j in reversed(range(m - 1)):
		if n - i < m - j:
			dp[i][j] = 0
		else:
			ans = -1
			if s[i] != t[j]:
				ans = dp[i + 1][j]
			else:
				ans = min(dp[i + 1][j + 1], dp[i + 1][j] + a[i])
			dp[i][j] = ans
print(dp[0][0])
