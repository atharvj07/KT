#include <bits/stdc++.h>
using namespace std;
template <class T>
void show(T a, int n) {
  for (int i = 0; i < n; ++i) cout << a[i] << ' ';
  cout << endl;
}
template <class T>
void show(T a, int r, int l) {
  for (int i = 0; i < r; ++i) show(a[i], l);
  cout << endl;
}
const int N = 1 << 10;
const int M = 5000;
const int oo = 10000 * 10000 * 10;
int n, m, r;
int eu[N], ev[N];
int skey[N];
int tkey[N];
int sp[N];
int tp[N];
char name[N][12];
char ts[12];
int st[N];
int tt[N];
int find(int t[], int u) {
  if (t[u] != u) t[u] = find(t, t[u]);
  return t[u];
}
inline void merge(int t[], int u, int v) {
  if (find(t, u) == find(t, v)) return;
  t[u] = v;
}
void cal(int tree[], int key[]) {
  int i, j, k;
  for (i = 0; i < n; ++i) tree[i] = i;
  for (i = 0; i < n; ++i) {
    for (j = 0; j < m; ++j) {
      int u = find(tree, key[j]);
      int v = find(tree, eu[j]);
      int t = find(tree, ev[j]);
      if (u == v || u == t) {
        merge(tree, v, t);
      }
    }
  }
}
bool solve() {
  int i, j, k;
  cal(st, skey);
  cal(tt, tkey);
  for (i = 0; i < n; ++i) {
    for (j = i + 1; j < n; ++j) {
      if ((find(st, i) == find(st, j)) != (find(tt, i) == find(tt, j)))
        return false;
    }
  }
  for (i = 0; i < r; ++i) {
    if (find(st, sp[i]) != find(st, tp[i])) return false;
  }
  for (i = 0; i < m; ++i) {
    if (find(st, skey[i]) != find(st, tkey[i])) return false;
  }
  return true;
}
int main() {
  int i, j, cas = 0;
  scanf("%d %d %d", &n, &m, &r);
  for (i = 0; i < m; ++i) {
    scanf("%d %d", &eu[i], &ev[i]);
    --eu[i];
    --ev[i];
  }
  for (i = 0; i < r; ++i) {
    int ndex, mnum;
    scanf("%s %d", name[i], &ndex);
    --ndex;
    sp[i] = ndex;
    scanf("%d", &mnum);
    for (j = 0; j < mnum; ++j) {
      int t;
      scanf("%d", &t);
      skey[t - 1] = ndex;
    }
  }
  for (i = 0; i < r; ++i) {
    int ndex, mnum;
    scanf("%s %d", ts, &ndex);
    --ndex;
    for (j = 0; j < r; ++j)
      if (strcmp(name[j], ts) == 0) {
        tp[j] = ndex;
        break;
      }
    scanf("%d", &mnum);
    for (j = 0; j < mnum; ++j) {
      int t;
      scanf("%d", &t);
      tkey[--t] = ndex;
    }
  }
  if (solve())
    puts("YES");
  else
    puts("NO");
  return 0;
}
