#include <bits/stdc++.h>
using namespace std;
const int N = 35005;
using ll = long long;
using PQ = priority_queue<int>;
int n, a[N], l[N], r[N];
int gtop(const PQ &pq) { return pq.empty() ? 0 : pq.top(); }
void proc(PQ &a, PQ &b, int x, int l) {
  int z = gtop(a);
  if (x >= z)
    b.push(l - x);
  else if (a.empty())
    b.push(l);
  else {
    b.push(l - z);
    a.pop();
    if (x >= 0) a.push(x);
  }
}
ll calc(int m) {
  PQ ql, qr;
  ll y = 0;
  int lb = m, rb = m;
  for (int i = 1; i <= n; i++) {
    int xl = lb + gtop(ql), xr = rb - gtop(qr);
    if (xl > 0)
      y += xl;
    else if (xr < 0)
      y -= xr;
    if (0 < rb) proc(ql, qr, -lb, rb - lb);
    if (0 > lb) proc(qr, ql, rb, rb - lb);
    lb += l[i];
    rb += r[i];
  }
  while (!ql.empty() && ql.top() + lb > m) y += ql.top() + lb - m, ql.pop();
  while (!qr.empty() && rb - qr.top() < m) y += m - (rb - qr.top()), qr.pop();
  return y;
}
int main() {
  ios::sync_with_stdio(false);
  cin >> n;
  int lt = 0, rt = 0;
  for (int i = 1; i <= n; i++)
    cin >> a[i] >> l[i] >> r[i], l[i] -= a[i], r[i] -= a[i], rt += a[i];
  lt = -rt;
  while (lt < rt) {
    int m = lt + (rt - lt) / 2;
    if (calc(m) > calc(m + 1))
      lt = m + 1;
    else
      rt = m;
  }
  cout << calc(lt) << endl;
  return 0;
}
