#include <bits/stdc++.h>
using namespace std;
int res[100100];
int main() {
  ios_base::sync_with_stdio(false);
  int n, i, j, p, q, x, a, b;
  map<int, int> ind;
  set<int> f, s;
  set<int>::iterator it;
  vector<int> c;
  cin >> n >> a >> b;
  for (i = 0; i < n; i++) {
    cin >> x;
    f.insert(x);
    ind[x] = i;
  }
  for (it = f.begin(); it != f.end(); ++it) {
    x = *it;
    if (f.find(a - x) == f.end()) {
      c.push_back(x);
    }
  }
  for (i = 0; i < c.size(); ++i) {
    f.erase(c[i]);
    s.insert(c[i]);
  }
  c.clear();
  while (!s.empty()) {
    it = s.begin();
    x = *it;
    if (s.find(b - x) != s.end()) {
      res[ind[x]] = 1;
      res[ind[b - x]] = 1;
      s.erase(x);
      if (2 * x != b) s.erase(b - x);
    } else if (f.find(b - x) != f.end()) {
      res[ind[x]] = 1;
      res[ind[b - x]] = 1;
      f.erase(b - x);
      if (2 * (b - x) != a) {
        f.erase(a - (b - x));
        s.insert(a - (b - x));
      }
      s.erase(x);
    } else {
      cout << "NO"
           << "\n";
      return 0;
    }
  }
  cout << "YES"
       << "\n";
  for (i = 0; i < n; i++) cout << res[i] << " ";
  cout << "\n";
  return 0;
}
