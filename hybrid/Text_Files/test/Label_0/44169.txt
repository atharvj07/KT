(a, b) = map(int, input().split())
if a - b == 0:
	a1 = int(str(a) + str('12'))
	b1 = int(str(b) + str('13'))
	print(a1, b1)
elif a < b and b - a < 2:
	a1 = int(str(a) + str('99'))
	b1 = int(str(b) + str('00'))
	print(a1, b1)
elif a > b and b == 1 and (a == 9):
	print(a, int(str(b) + str('0')))
else:
	print(-1)
