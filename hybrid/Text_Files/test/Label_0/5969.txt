#include <bits/stdc++.h>
using namespace std;
int a[3010], g[3010][3];
int maxd, maxi, tot = 0, n;
int main() {
  scanf("%d", &n);
  for (int i = 0; i < n; i++) scanf("%d", &a[i]);
  for (int j = 0; j < n; j++) {
    maxd = -1000001000;
    for (int i = 0; i < n - j; i++) {
      if (a[i] > maxd) maxd = a[i], maxi = i;
    }
    g[++tot][1] = maxi;
    g[tot][2] = n - j - 1;
    swap(a[maxi], a[n - j - 1]);
  }
  printf("%d\n", tot);
  for (int i = 1; i <= tot; i++) printf("%d %d\n", g[i][1], g[i][2]);
}
