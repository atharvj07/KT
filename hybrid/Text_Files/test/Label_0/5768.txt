#include <bits/stdc++.h>
using namespace std;
set<pair<int, int> > a[100005];
set<int> ans[100005], C[100005];
map<int, int> stay[100005];
set<int>::iterator p;
int vis[100005], n, m, x, y, z, i, col, Q, number, ID;
void DFS(int x) {
  if (vis[x] == col) return;
  vis[x] = col;
  ans[x].insert(ID);
  for (set<pair<int, int> >::iterator it = a[x].begin(); it != a[x].end(); it++)
    if ((*it).first == col)
      DFS((*it).second);
    else
      break;
  while (!a[x].empty()) {
    set<pair<int, int> >::iterator it = a[x].begin();
    if ((*it).first > col) return;
    a[x].erase(it);
  }
}
int main() {
  scanf("%d%d", &n, &m);
  for (i = 1; i <= m; i++)
    scanf("%d%d%d", &x, &y, &z), a[x].insert(make_pair(z, y)),
        a[y].insert(make_pair(z, x)), C[z].insert(x), C[z].insert(y);
  for (col = 1; col <= 1e5; col++)
    if (!C[col].empty())
      for (p = C[col].begin(); p != C[col].end(); p++) ++ID, DFS(*p);
  scanf("%d", &Q);
  while (Q--) {
    scanf("%d%d", &x, &y);
    number = 0;
    if (ans[x].size() > ans[y].size()) swap(x, y);
    if (stay[x].find(y) != stay[x].end()) {
      printf("%d\n", stay[x][y]);
      continue;
    }
    for (p = ans[x].begin(); p != ans[x].end(); p++)
      if (ans[y].find(*p) != ans[y].end()) number++;
    printf("%d\n", stay[x][y] = number);
  }
}
