    static class Trie
    {
        int left[],right[],N,data[];
        Trie()
        {
            left=new int[20000000];
            right=new int[20000000];
            data=new int[20000000];
            data[0]=10000000;
            N=0;
        }
        int createNode()
        {
            N++;
            data[N]=10000000;
            return N;
        }
        int newTrie()
        {
            return createNode();
        }
        void insert(int node,int value)
        {
            int mask=1<<17;
            while(mask>0)
            {
                if((value&mask)>0)
                {
                    if(right[node]==0)
                        right[node]=createNode();
                    node=right[node];
                }
                else
                {
                    if(left[node]==0)
                        left[node]=createNode();
                    node=left[node];
                }
                mask>>=1;
                data[node]=Math.min(data[node],value);
            }
        }
        int query(int node,int value,int sum)
        {
            int mask=1<<17;
            int ans=0;
            while(mask>0)
            {
                if((value&mask)>0)
                {
                    if(left[node]!=0&&data[left[node]]+value<=sum)
                    {
                        ans+=mask;
                        node=left[node];
                    }
                    else if(data[right[node]]+value<=sum)
                        node=right[node];
                    else
                        return -1;
                }
                else
                {
                    if(right[node]!=0&&data[right[node]]+value<=sum)
                    {
                        ans+=mask;
                        node=right[node];
                    }
                    else if(data[left[node]]+value<=sum)
                        node=left[node];
                    else
                        return -1;
                }
                mask>>=1;
            }
            return ans;
        }
    }
    public static void main(String args[]) throws IOException
    {
        Scanner in=new Scanner(System.in);
        PrintWriter out=new PrintWriter(System.out);
        ArrayList<Integer> d[]=new ArrayList[100001];
        for(int i=1;i<=100000;i++)
            d[i]=new ArrayList<>();
        for(int i=1;i<=100000;i++)
            for(int j=i;j<=100000;j+=i)
                d[j].add(i);
        Trie t=new Trie();
        int v[]=new int[100001];
        for(int i=1;i<=100000;i++)
            t.newTrie();
        int q=in.nextInt();
        while(q--!=0)
        {
            if(in.nextInt()==1)
            {
                int u=in.nextInt();
                if(v[u]==1)
                    continue;
                for(int i=0;i<d[u].size();i++)
                    t.insert(d[u].get(i),u);
                v[u]=1;
            }
            else
            {
                int x=in.nextInt();
                int k=in.nextInt();
                int s=in.nextInt();
                if(x%k!=0)
                    out.println(-1);
                else
                {
                    int ans=t.query(k, x, s);
                    if(ans!=-1)
                        ans^=x;
                    out.println(ans);
                }
            }
        }
        out.close();
    }
