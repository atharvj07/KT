import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.util.Arrays;
import java.util.StringTokenizer;

public class TaskE implements Runnable {
    
    static boolean good[][];
    final static int MOD = 777777777;

    private void solve() throws IOException {
        int n = nextInt();
        int m = nextInt();
        
        good = new boolean[3][3];
        for (int u, i = 0; i < 3; ++i) {
            for (int j = 0; j < 3; ++j) {
                u = nextInt();
                if (u == 1) {
                    good[i][j] = true;
                }
            }
        }
        
        IntervalTree tree = new IntervalTree(n);
        for (int v, t, iter = 0; iter < m; ++iter) {
            v = nextInt() - 1;
            t = nextInt();
            tree.change(v, t);
            writer.println(tree.getAll());
        }
    }

    public static void main(String[] args) {
        new TaskE().run();
    }

    BufferedReader reader;
    StringTokenizer tokenizer;
    PrintWriter writer;

    public void run() {
        try {
            reader = new BufferedReader(new InputStreamReader(System.in));
            tokenizer = null;
            writer = new PrintWriter(System.out);
            solve();
            reader.close();
            writer.close();
        } catch (Exception e) {
            e.printStackTrace();
            System.exit(1);
        }
    }

    int nextInt() throws IOException {
        return Integer.parseInt(nextToken());
    }

    long nextLong() throws IOException {
        return Long.parseLong(nextToken());
    }

    double nextDouble() throws IOException {
        return Double.parseDouble(nextToken());
    }

    String nextToken() throws IOException {
        while (tokenizer == null || !tokenizer.hasMoreTokens()) {
            tokenizer = new StringTokenizer(reader.readLine());
        }
        return tokenizer.nextToken();
    }
}

class IntervalTree {
    final int size;
    long mat[][];
    
    IntervalTree(int size) {
        this.size = size;
        mat = new long[this.size * 4 + 10][9];
        buildTree(0, this.size - 1, 1);
    }
    
    private void merge(int x) {
        Arrays.fill(mat[x], 0);
        
        for (int i = 0; i < 3; ++i) {
            for (int j = 0; j < 3; ++j) {
                if (mat[x * 2][i*3+j] != 0) {
                    for (int u = 0; u < 3; ++u) {
                        if (TaskE.good[j][u]) {
                            for (int v = 0; v < 3; ++v) {
                                if (mat[x * 2 + 1][u*3+v] != 0) {
                                    mat[x][i * 3 + v] += mat[x * 2][i * 3 + j] * mat[x * 2 + 1][u * 3 + v];
                                    mat[x][i * 3 + v] %= TaskE.MOD;
                                }
                            }
                        }
                    }
                }
            }
        }
        /*
        for (int i = 0; i < 3; ++i) {
            for (int j = 0; j < 3; ++j) {
                if (mat[x][i * 3 + j] >= TaskE.MOD) {
                    mat[x][i * 3 + j] %= TaskE.MOD;
                }
            }
        }*/
    }
    
    private void buildTree(int left, int right, int x) {
        int mid = (left + right) >> 1;
        if (left < right) {
            buildTree(left, mid, x * 2);
            buildTree(mid + 1, right, x * 2 + 1);
            merge(x);
        } else {
            for (int k = 0; k < 3; ++k) {
                mat[x][k * 3 + k] = 1;
            }
        }
    }
    
    private void change(int left, int right, int pos, int x, int can) {
        if (left == right) {
            Arrays.fill(mat[x], 0);
            
            if (can > 0) {
                --can;
                mat[x][can * 3 + can] = 1;
            } else {
                for (int i = 0; i < 3; ++i) {
                    mat[x][i * 3 + i] = 1;
                }
            }
            return;
        }
        
        int mid = (left + right) >> 1;
        if (pos <= mid) {
            change(left, mid, pos, x * 2, can);
        } else {
            change(mid + 1, right, pos, x * 2 + 1, can);
        }
        
        merge(x);
    }
    
    public void change(int pos, int can) {
        change(0, this.size - 1, pos, 1, can);
    }
    
    public long getAll() {
        long ret = 0;
        for (int i = 0; i < 3; ++i) {
            for (int j = 0; j < 3; ++j) {
                ret += mat[1][i * 3 + j];
            }
        }
        
        return ret % TaskE.MOD;
    }
    
}
