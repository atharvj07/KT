struct node {
	char sym;
	node *prv, *nxt;
	
	node () {
		sym = 0;
		prv = nxt = NULL;
	}
} my_list[MAXN];

node * head;

node * my_erase(node * cur) {
	if (cur == head) {
		head = cur->nxt;
	}
	if (cur->prv != NULL) {
		cur->prv->nxt = cur->nxt;
	}
	if (cur->nxt != NULL) {
		cur->nxt->prv = cur->prv;
	}
	cnt[cur->sym]--;
	return cur->nxt;
}
