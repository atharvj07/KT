from collections import *
s = input()
c = Counter(((ord(y) - ord(x)) % 10 for (x, y) in zip(s, s[1:])))
r = [0] * 100
for i in range(100):
	a = [-1] * 11
	q = [10]
	while q:
		x = q.pop(0)
		for j in (i // 10, i % 10):
			y = (x + j) % 10
			if a[y] < 0:
				a[y] = a[x] + 1
				q += (y,)
	for x in c:
		r[i] = -(r[i] < 0 or a[x] < 0 or -r[i] - c[x] * a[x])
print(*r)
