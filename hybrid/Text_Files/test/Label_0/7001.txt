#include <stdio.h>
#include <math.h>

using namespace std;
#define dTolerance 1.0e-10

int main(){
    int n;
    scanf("%d", &n);
    double x[n+1],y[n+1];
    for (int i = 0; i < n; i++){
        scanf("%lf %lf", &x[i], &y[i]);
    }
    x[n] = x[0];
    y[n] = y[0];

    int ixMin = 0;
    int ixMax = 0;
    for (int i = 0; i < n; i++){
        if(x[i] < x[ixMin]){    
            ixMin = i;
        }
        if(fabs(x[i] - x[ixMin]) < dTolerance && y[i] > y[ixMin]){    //x座標最小のもののなかでy座標最大のものにしておく
            ixMin = i;
        }
        if(x[i] > x[ixMax]){  
            ixMax = i;
        }
        if(fabs(x[i] - x[ixMax]) < dTolerance && y[i] < y[ixMax]){    //x座標最大のもののなかでy座標最小のものにしておく
            ixMin = i;
        }
    }
    
    double dDiameter = sqrt( (x[ixMax] - x[ixMin]) * (x[ixMax] - x[ixMin]) + (y[ixMax] - y[ixMin]) * (y[ixMax] - y[ixMin]) );

    int i = ixMin;
    int j = ixMax;
    
    while(j != ixMin || i != ixMax){
        if( (x[j+1] - x[j]) * (y[i+1] - y[i]) - (x[i+1] - x[i]) * (y[j+1] - y[j]) >= 0 ) {;    //j番目の辺(向き付)から見てi番目の辺(向き付)のなす角が反時計回り
            i++;
        } else {    //時計回り
            j++;
        }
        i %= n;    //nを越えないようにしておく
        j %= n;
        if (dDiameter < sqrt( (x[j] - x[i]) * (x[j] - x[i]) + (y[j] - y[i]) * (y[j] - y[i]) )){
            dDiameter = sqrt( (x[j] - x[i]) * (x[j] - x[i]) + (y[j] - y[i]) * (y[j] - y[i]) );
        }
    }

    printf("%lf", dDiameter);
    
    return 0;
}
