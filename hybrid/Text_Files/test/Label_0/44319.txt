import sys
import math
input = sys.stdin.readline

def inp():
	return int(input())

def inlt():
	return list(map(int, input().split()))

def inltstr():
	return list(map(str, input().split()))

def insr():
	s = input()
	return list(s[:len(s) - 1])

def invr():
	return map(int, input().split())

def factorial(n):
	if n == 1:
		return n
	elif n < 1:
		return 'NA'
	else:
		return n * factorial(n - 1)

def coeffBin(k):
	return factorial(k + 1) / (factorial(2) * factorial(k - 1))

def solve(num):
	num = num.strip('\n')
	if int(num) % 2 == 0:
		return '0'
	elif int(num[0]) % 2 == 0:
		return '1'
	else:
		for c in num:
			if int(c) % 2 == 0:
				return '2'
	return '-1'
res = []
iter = inp()
i = 0
while i < iter:
	arr = input()
	result = solve(arr)
	res.append(result)
	i += 1
for r in res:
	print(r)
