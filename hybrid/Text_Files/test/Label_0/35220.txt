import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.Arrays;
import java.util.Comparator;

import static java.lang.Math.exp;
import static java.lang.Math.log;

/**
 * User: dmitryt
 * Date: 04/07/2015
 */
public class CF138C1 {
    private static class TreeEvent {
        private final int power; // for the mushrooms
        private final boolean start;
        private final boolean left;
        private final int x;
        private final double prob;
        private final int order;

        public TreeEvent(int power, boolean start, boolean left, int x, double prob, int order) {
            this.power = power;
            this.start = start;
            this.left = left;
            this.x = x;
            this.prob = prob == Double.NEGATIVE_INFINITY ? -10000 : prob;
            this.order = order;
        }
    }

    private static void solve(int n, int m, int[] tx, int[] th, int[] pleft, int[] pright, int[] mx, int[] mp) {
        TreeEvent[] events = new TreeEvent[4*n + m];
        for (int i = 0; i < n; i++) {
            events[i*4] = new TreeEvent(-1, true, true, tx[i] - th[i], log((100.0-pleft[i])/100), -1);
            events[i*4+1] = new TreeEvent(-1, false, true, tx[i], log((100.0-pleft[i])/100), -1);
            events[i*4+2] = new TreeEvent(-1, true, false, tx[i], log((100.0-pright[i])/100), 1);
            events[i*4+3] = new TreeEvent(-1, false, false, tx[i] + th[i], log((100.0-pright[i])/100), 1);
        }
        for (int i = 0; i < m; i++) {
            events[n*4 + i] = new TreeEvent(mp[i], true, true, mx[i], 0, 0);
        }
        Arrays.sort(events, Comparator.comparing((TreeEvent te) -> te.x).thenComparing((TreeEvent te) -> te.order));

        double res = 0;
        double prob = 0.0; // logarithmic

        int k = 0; // current mushroom

        for(TreeEvent event: events) {
            if (event.power > -1) {
                res += exp(prob + log(event.power));
            } else if (event.start) {
                prob += event.prob;
            } else {
                prob -= event.prob;
            }
        }

        System.out.println(String.format("%f", res));
    }

    private static int nextInt(StreamTokenizer st) throws Exception {
        int token = st.nextToken();
        while (token != StreamTokenizer.TT_NUMBER) token = st.nextToken();
        return (int)st.nval;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StreamTokenizer st = new StreamTokenizer(in);

        int n = nextInt(st);
        int m = nextInt(st);
        int[] tx = new int[n];
        int[] th = new int[n];
        int[] pl = new int[n];
        int[] pr = new int[n];
        for (int i = 0; i < n; i++) {
            tx[i] = nextInt(st);
            th[i] = nextInt(st);
            pl[i] = nextInt(st);
            pr[i] = nextInt(st);
        }
        int[] mx = new int[m];
        int[] mp = new int[m];
        for (int i = 0; i < m; i++) {
            mx[i] = nextInt(st);
            mp[i] = nextInt(st);
        }
        solve(n, m, tx, th, pl, pr, mx, mp);
   }
}
