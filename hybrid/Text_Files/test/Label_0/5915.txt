#include <bits/stdc++.h>
using namespace std;
const int maxn = 31, maxm = 310, maxd = 10;
bool mark[maxm][maxm][maxn][maxd];
bool chk[maxm][maxm];
int ax[] = {0, 1, 1, 1, 0, -1, -1, -1};
int ay[] = {1, 1, 0, -1, -1, -1, 0, 1};
int n, ans;
int t[maxn];
inline void dfs(int id = 0, int x = 152, int y = 152, int dir = 0) {
  if (mark[x][y][id][dir]) return;
  mark[x][y][id][dir] = true;
  for (int i = 0; i < t[id]; i++) {
    x += ax[dir];
    y += ay[dir];
    if (!chk[x][y]) chk[x][y] = true, ans++;
  }
  if (id + 1 >= n) return;
  dfs(id + 1, x, y, (dir + 1) % 8);
  dfs(id + 1, x, y, (dir + 7) % 8);
}
int main() {
  cin >> n;
  for (int i = 0; i < n; i++) cin >> t[i];
  dfs();
  cout << ans << "\n";
}
