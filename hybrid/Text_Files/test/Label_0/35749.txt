import java.io.*;
import java.util.*;
public class c {
public static void main(String[] args) throws IOException {
	input.init(System.in);
	PrintWriter out = new PrintWriter(System.out);
	int n = input.nextInt(), m = input.nextInt();
	long a = input.nextInt(), b = input.nextInt();
	long[] as = new long[n], bs = new long[m], ls = new long[m];
	TreeSet<Pair> set = new TreeSet<Pair>();
	for(int i = 0; i<n; i++) as[i] = input.nextInt();
	for(int i = 0; i<n; i++) set.add(new Pair(as[i], i));
	for(int i = 0; i<m; i++)
	{
		bs[i] = input.nextInt();
	}
	for(int i = 0; i<m; i++) ls[i] = input.nextInt();
	double best = 1e18;
	int besti = -1;
	int besta = -1;
	for(int i = 0; i<m; i++)
	{
		double cur = ls[i];
		ArrayList<Pair> poss = new ArrayList<Pair>();
		Pair x = set.floor(new Pair(0, -1));
		if(x != null) poss.add(x);
		x = set.ceiling(new Pair(0, -1));
		if(x != null) poss.add(x);
		x = set.ceiling(new Pair(bs[i], -1));
		if(x != null) poss.add(x);
		x = set.floor(new Pair(bs[i], -1));
		if(x != null) poss.add(x);
		double d = 1.0 * a / b * bs[i];
		int dd = (int)d;
		for(int de = -2; de <= 2; de++)
		{
			Pair c = new Pair(dd + de, -1);
			x = de < 0 ? set.floor(c) : set.ceiling(c);
			if(x != null) poss.add(x);
		}
		for(Pair y : poss)
		{
			cur = ls[i] + Math.sqrt((y.x*y.x) + a*a) + Math.sqrt((b-a)*(b-a) + (bs[i] - y.x) * (bs[i] - y.x));
			//out.println(i+" "+ls[i]+" "+cur);
			if(cur < best)
			{
				besta = y.i;
				best = cur;
				besti = i;
			}
		}
	}
	//out.println(best);
	out.print(besta+1+" ");
	out.println(besti+1);
	out.close();
}
static class Pair implements Comparable<Pair>
{
	long x;
	int i;
	public Pair(long xx, int ii)
	{
		x = xx; i = ii;
	}
	@Override
	public int compareTo(Pair o) {
		// TODO Auto-generated method stub
		return Long.signum(x - o.x);
	}
}
public static class input {
	static BufferedReader reader;
	static StringTokenizer tokenizer;

	static void init(InputStream input) {
		reader = new BufferedReader(new InputStreamReader(input));
		tokenizer = new StringTokenizer("");
	}

	static String next() throws IOException {
		while (!tokenizer.hasMoreTokens())
			tokenizer = new StringTokenizer(reader.readLine());
		return tokenizer.nextToken();
	}

	static int nextInt() throws IOException {
		return Integer.parseInt(next());
	}

	static double nextDouble() throws IOException {
		return Double.parseDouble(next());
	}

	static long nextLong() throws IOException {
		return Long.parseLong(next());
	}
}
}
