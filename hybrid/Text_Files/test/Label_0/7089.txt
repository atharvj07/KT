#include <bits/stdc++.h>
using namespace std;
int main() {
  int n, m;
  cin >> n >> m;
  int arr[n], seq[m], freq[10] = {0};
  for (int i = 0; i < n; i++) cin >> arr[i];
  for (int i = 0; i < m; i++) {
    cin >> seq[i];
    freq[seq[i]]++;
  }
  for (int i = 0; i < n; i++)
    if (freq[arr[i]] > 0) cout << arr[i] << " ";
  cout << endl;
  return 0;
}
