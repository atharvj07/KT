#include <bits/stdc++.h>
using namespace std;
int main() {
  long long n;
  cin >> n;
  long long a[n];
  for (long long i = 0; i < n; i++) cin >> a[i];
  sort(a, a + n);
  if (a[n - 1] >= a[n - 2] + a[n - 3])
    cout << "NO"
         << "\n";
  else {
    cout << "YES"
         << "\n";
    cout << a[n - 2] << ' ' << a[n - 1] << ' ' << a[n - 3] << ' ';
    for (long long i = n - 4; i >= 0; i--) cout << a[i] << " ";
  }
  return 0;
}
