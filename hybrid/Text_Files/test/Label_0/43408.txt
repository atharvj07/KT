(n, m) = [int(x) for x in input().split()]
lista = [int(x) for x in input().split()]
soma = 0
dic = {}
for i in range(m):
	entrada = input().split()
	a = entrada[0]
	b = int(entrada[1])
	if a == '1':
		lista[b - 1] = int(entrada[2])
		dic[b] = soma
	elif a == '2':
		soma += b
	elif a == '3':
		if b in dic:
			total = soma - dic[b]
			print(lista[b - 1] + total)
		else:
			print(lista[b - 1] + soma)
