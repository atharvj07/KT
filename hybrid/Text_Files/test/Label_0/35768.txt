import java.io.*;
import java.util.*;

public class CF349E {
	static class Set {
		int it;
		int[] ii;
		Set(int[] ii) {
			this.ii = ii;
		}
	}
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		PrintWriter pw = new PrintWriter(System.out);
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int m = Integer.parseInt(st.nextToken());
		int q = Integer.parseInt(st.nextToken());
		long[] aa = new long[n];
		st = new StringTokenizer(br.readLine());
		for (int i = 0; i < n; i++)
			aa[i] = Long.parseLong(st.nextToken());
		Set[] ss = new Set[m];
		int sqrtn = (int) Math.sqrt(n);
		int h = 0;
		for (int k = 0; k < m; k++) {
			st = new StringTokenizer(br.readLine());
			int size = Integer.parseInt(st.nextToken());
			int[] ii = new int[size];
			for (int j = 0; j < size; j++)
				ii[j] = Integer.parseInt(st.nextToken()) - 1;
			ss[k] = new Set(ii);
			if (size > sqrtn)
				h++;
		}
		Set[] tt = new Set[h];
		h = 0;
		for (int k = 0; k < m; k++)
			if (ss[k].ii.length > sqrtn) {
				tt[h] = ss[k];
				ss[k].it = h;
				h++;
			}
		long[] cc = new long[h];	// collect
		long[] bb = new long[h];	// batch
		int[][] intersection = new int[h][m];
		for (int l = 0; l < h; l++) {
			boolean[] used = new boolean[n];
			int[] ii = tt[l].ii;
			for (int j = 0; j < ii.length; j++) {
				int i = ii[j];
				used[i] = true;
				cc[l] += aa[i];
			}
			for (int k = 0; k < m; k++) {
				int cnt = 0;
				ii = ss[k].ii;
				for (int j = 0; j < ii.length; j++) {
					int i = ii[j];
					if (used[i])
						cnt++;
				}
				intersection[l][k] = cnt;
			}
		}
		while (q-- > 0) {
			st = new StringTokenizer(br.readLine());
			String type = st.nextToken();
			int k = Integer.parseInt(st.nextToken()) - 1;
			if (type.equals("?")) {
				long sum;
				if (ss[k].ii.length > sqrtn) {
					int l = ss[k].it;
					sum = cc[l];
				} else {
					sum = 0;
					int[] ii = ss[k].ii;
					for (int j = 0; j < ii.length; j++) {
						int i = ii[j];
						sum += aa[i];
					}
				}
				for (int l = 0; l < h; l++)
					sum += bb[l] * intersection[l][k];
				pw.println(sum);
			} else if (type.equals("+")) {
				long x = Long.parseLong(st.nextToken());
				if (ss[k].ii.length > sqrtn) {
					int l = ss[k].it;
					bb[l] += x;
				} else {
					int[] ii = ss[k].ii;
					for (int j = 0; j < ii.length; j++) {
						int i = ii[j];
						aa[i] += x;
					}
					for (int l = 0; l < h; l++)
						cc[l] += x * intersection[l][k];
				}
			}
		}
		pw.close();
	}
}
