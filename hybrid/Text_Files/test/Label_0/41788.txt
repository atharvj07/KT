def main():
	s = input()
	n = int(input())
	cnt = 0
	str = []
	ans = ''
	for i in range(len(s)):
		if s[i] != '*' and s[i] != '?':
			cnt += 1
	if cnt == n:
		for i in range(len(s)):
			if s[i] != '?' and s[i] != '*':
				ans += s[i]
		print(ans)
		return
	temp = 0
	if cnt < n:
		for i in range(len(s)):
			if temp < n:
				if s[i] != '*' and s[i] != '?':
					str.append(s[i])
					temp += 1
					cnt -= 1
				elif cnt < n:
					if s[i] == '?':
						continue
					else:
						while temp + cnt < n:
							str.append(s[i - 1])
							temp += 1
	else:
		temp = cnt
		for i in range(len(s)):
			if temp > n:
				if s[i] != '*' and s[i] != '?':
					str.append(s[i])
				else:
					str.pop()
					temp -= 1
			elif s[i] != '*' and s[i] != '?':
				str.append(s[i])
	if len(str) != n:
		print('Impossible')
		return
	else:
		for i in range(len(str)):
			ans += str[i]
	print(ans)
main()
