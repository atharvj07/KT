#include<bits/stdc++.h>
using namespace std;
inline int gcd(int a,int b){return (!b)?a:gcd(b,a%b);}

int main()
{
	int maxx=0,n,d;
	scanf("%d%d",&n,&d);
	int ans=0;
	for(int i=1;i<=n;++i){
		int x;scanf("%d",&x);
		maxx=max(maxx,x);
		if(i==1)ans=x;
		else ans=gcd(ans,x);		
	}
	if(d<=maxx&&(d%ans)==0)cout<<"POSSIBLE";
	else cout<<"IMPOSSIBLE";
}
