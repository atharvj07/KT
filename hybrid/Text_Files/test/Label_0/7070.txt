#include <bits/stdc++.h>
using namespace std;
vector<int> v;
set<int> st;
int main() {
  int n;
  cin >> n;
  string s;
  cin >> s;
  int flag = 0;
  for (int i = 1; i < n; i++) {
    if (s[i] == s[i - 1] && s[i] != '?') {
      flag = 1;
      break;
    }
  }
  if (flag) {
    cout << "No" << endl;
    return 0;
  }
  int cnt = 0;
  for (int i = 0; i < n; i++) {
    if (s[i] == '?') {
      cnt++;
      v.push_back(i);
    }
  }
  if (cnt == 0)
    cout << "No" << endl;
  else if (cnt == 1) {
    int x = v[0];
    if (x == 0 || x == (n - 1))
      cout << "Yes" << endl;
    else {
      if (s[x - 1] == s[x + 1])
        cout << "Yes" << endl;
      else
        cout << "No" << endl;
    }
  } else {
    for (int i = 0; i < v.size(); i++) {
      int x = v[i];
      if (x == 0 || x == (n - 1)) {
        cout << "Yes" << endl;
        return 0;
      } else {
        if ((s[x - 1] == s[x + 1]) || s[x + 1] == '?') {
          cout << "Yes" << endl;
          return 0;
        }
      }
    }
    cout << "No" << endl;
  }
  return 0;
}
