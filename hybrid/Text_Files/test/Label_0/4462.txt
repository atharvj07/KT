#include <bits/stdc++.h>
using namespace std;
int power(int a, int b, int c, int ans = 1) {
  for (; b; b >>= 1, a = (long long)a * a % c)
    if (b & 1) ans = (long long)ans * a % c;
  return ans;
}
int a[100010], b[100010], lis[100010], id[100010], vst[100010];
int pos[5100010], val[5100010], pre[5100010];
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int n, m;
  cin >> n;
  for (int i = 1; i <= n; i++) cin >> a[i];
  cin >> m;
  for (int i = 0; i < m; i++) cin >> b[i];
  int cur = 1, E = n, ans = 0, ends = -1;
  for (int i = 1; i <= n; i++) lis[i] = 0x3f3f3f3f;
  sort(b, b + m);
  for (int i = 1; i <= n; i++) {
    pos[i] = i;
    if (~a[i]) {
      int u = lower_bound(lis, lis + cur, a[i]) - lis;
      lis[u] = a[i];
      id[u] = i;
      pre[i] = id[u - 1];
      if (u >= ans) ans = u, ends = i;
      if (u == cur) cur++;
      continue;
    }
    int now = cur, ed = -1;
    for (int j = m; j--;) {
      while (lis[now - 1] >= b[j]) now--;
      if (lis[now] == b[j]) continue;
      if (now != ed) E++;
      ed = now;
      val[E] = j;
      pos[E] = i;
      pre[E] = id[now - 1];
      lis[now] = b[j];
      id[now] = E;
      if (now > ans) ends = E, ans = now;
      if (now == cur) cur++;
    }
  }
  while (ends) {
    if (ends > n) a[pos[ends]] = b[val[ends]], vst[val[ends]] = 1;
    ends = pre[ends];
  }
  cur = 0;
  for (int i = 1; i <= n; i++)
    if (a[i] == -1) {
      while (vst[cur]) cur++;
      a[i] = b[cur];
      cur++;
    }
  for (int i = 1; i <= n; i++) cout << a[i] << ' ';
  cout << endl;
  return 0;
}
