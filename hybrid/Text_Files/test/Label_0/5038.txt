#include <bits/stdc++.h>
using namespace std;
const int N = 1e9 + 7, M = 1111;
void f1() { cout << "NO\n"; }
void solve() {
  int n, a;
  unordered_set<int> second;
  vector<int> v;
  cin >> n;
  for (int i = (0); i < (2 * n); i++) {
    cin >> a;
    if (second.find(a) == second.end()) {
      v.push_back(a);
    }
    second.insert(a);
  }
  for (auto i : v) {
    cout << i << " ";
  }
  cout << endl;
}
int main() {
  int t = 1, h = 0;
  cin >> t;
  while (t--) {
    h++;
    solve();
  }
  return 0;
}
