#include<iostream>
using namespace std;
int main(){
  int h1,m1,s1,h2,m2,s2;
  int h,m,s,result;
  
  for(int i=0;i<3;i++){
  cin >>h1>>m1>>s1>>h2>>m2>>s2;
  
  result=(h2*3600+m2*60+s2)-(h1*3600+m1*60+s1);
  h=result/3600;
  result-=h*3600;
  m=result/60;
  result-=m*60;
  s=result;
  cout<<h<<" "<<m<<" "<<s<<endl;
  }
  return 0;
}
