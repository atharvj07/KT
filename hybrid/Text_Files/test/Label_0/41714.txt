import math
for _ in range(int(input())):
	(a, b) = map(int, input().split())
	if a == b:
		print(0)
	elif a > b:
		x = a / b
		if math.ceil(x) != math.floor(x):
			print(-1)
		else:
			s = 0
			while a != b:
				if a % 8 == 0 and a // 8 >= b:
					a = a // 8
					s += 1
				elif a % 4 == 0 and a // 4 >= b:
					a = a // 4
					s += 1
				elif a % 2 == 0 and a // 2 >= b:
					a = a // 2
					s += 1
				else:
					s = -1
					break
			print(s)
	elif a < b:
		x = b / a
		if math.ceil(x) != math.floor(x):
			print(-1)
		else:
			s = 0
			while a != b:
				if b % 8 == 0 and b // 8 >= a:
					b = b // 8
					s = s + 1
				elif b % 4 == 0 and b // 4 >= a:
					b = b // 4
					s += 1
				elif b % 2 == 0 and b // 2 >= a:
					b = b // 2
					s = s + 1
				else:
					s = -1
					break
			print(s)
