#include <bits/stdc++.h>
#define ll long long
using namespace std;
struct node{
	int a,b;
}a[100001];
int n,m,cnt,t;
ll ans;
priority_queue<int> q;
bool cmp(node a,node b)
{
	return a.a<b.a;
}
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(0);
	cout.tie(0);
	cin>>n>>m;
	for (int i=0;i<n;++i) cin>>a[i].a>>a[i].b;
	sort(a,a+n,cmp);
	for (int i=1;i<=m;++i)
	{
		while (t<n && a[t].a<=i) q.push(a[t].b),++t;
		if (!q.empty()) ans+=q.top(),q.pop();
	}
	cout<<ans;
	return 0;
}
