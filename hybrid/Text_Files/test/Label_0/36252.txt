import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner cin = new Scanner(System.in);
        int a = cin.nextInt(), b = cin.nextInt(), x = 0, y = 0, ans = 0;
        char[][] arr = new char[a][];
        cin.nextLine();
        for (int i = 0; i < a; i++) arr[i] = cin.nextLine().toCharArray();
        a--;
        b--;
        boolean flag = false;
        while (y != a || x != b) {
            if (arr[y][x] == '*') ans++;
            if (x == b) y++;
            else if (y == a) x++;
            else if (x < b && arr[y][x + 1] == '*') x++;
            else if (y < a && arr[y + 1][x] == '*') y++;
            else x++;
        }
        System.out.println(ans + (arr[a][b] == '*' ? 1:0));
    }
}
