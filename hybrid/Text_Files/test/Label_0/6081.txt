#include<iostream>
#include<algorithm>
#include<string>
#include<stack>
using namespace std;
stack<char> s[4];
stack<char> ans;
string z;
int main(){
	while(cin>>z&&z!="-"){
		for(int i=0;i<3;i++){
			for(int j=z.size()-1;j>=0;j--)s[i].push(z[j]);
			if(i!=2)cin>>z;
		}
		while(!s[1].empty()){
			if(s[2].empty())goto loop;
			if(s[1].top()==s[2].top()){
				s[1].pop(),s[2].pop(),s[1].push(s[0].top()),s[0].pop();
			}
			else{
				loop:
				s[3].push(s[1].top()),s[1].pop();
			}
		}
		while(!s[3].empty()){
			ans.push(s[3].top());
			s[3].pop();
		}
		while(!ans.empty()){
			cout<<ans.top();
			ans.pop();
		}
		cout<<endl;
	}
	return 0;
}
