import collections

def swapCharacters(strings, k, n):
	if k == 1:
		s0 = list(strings[0])
		(s0[0], s0[1]) = (s0[1], s0[0])
		return ''.join(s0)
	freq = collections.Counter(strings[0])
	canSame = max(freq.values()) >= 2
	for s in strings:
		if collections.Counter(s) != freq:
			return -1
	max_dist = 0
	max_dist_s = None
	s0 = strings[0]
	for s1 in strings[1:]:
		dist = HammingDistance(s0, s1, n)
		if dist > max_dist:
			max_dist = dist
			max_dist_s = s1
	if max_dist > 4:
		return -1
	diffs = [i for i in range(n) if s0[i] != s1[i]]
	s0 = list(s0)
	for i in diffs:
		for j in range(n):
			(s0[i], s0[j]) = (s0[j], s0[i])
			dist = HammingDistance(s0, s1, n)
			if dist == 2 or (dist == 0 and canSame):
				cand = ''.join(s0)
				if verifyAll(strings, k, n, cand, canSame):
					return cand
			(s0[i], s0[j]) = (s0[j], s0[i])
	return -1

def HammingDistance(s0, s1, n):
	count = 0
	for i in range(n):
		if s0[i] != s1[i]:
			count += 1
	return count

def verifyAll(strings, k, n, cand, canSame):
	for s in strings:
		dist = HammingDistance(s, cand, n)
		if (dist != 0 or not canSame) and dist != 2:
			return False
	return True
(k, n) = [int(x) for x in input().split()]
strings = set()
for _ in range(k):
	strings.add(input())
strings = list(strings)
k = len(strings)
res = swapCharacters(strings, k, n)
print(res)
