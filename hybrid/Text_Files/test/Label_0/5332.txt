#include <bits/stdc++.h>
int f[1010];
void init(int n) {
  int i;
  for (i = 0; i <= n; i++) f[i] = i;
}
int find(int x) {
  if (x != f[x]) f[x] = find(f[x]);
  return f[x];
}
int uion(int x, int y) {
  x = find(x);
  y = find(y);
  f[x] = y;
}
struct edge {
  int from;
  int to;
  int len;
} e[10010];
int cmp(const void *a, const void *b) {
  struct edge *x = (struct edge *)a;
  struct edge *y = (struct edge *)b;
  return x->len - y->len;
}
int q[1010];
int check(int x, int y) {
  int sx = find(x), sy = find(y);
  if (sx == sy) return 0;
  if (sy != y) return 0;
  return 1;
}
int main() {
  int n, m;
  scanf("%d", &n);
  int i, x, y, len;
  for (i = 1; i <= n; i++) scanf("%d", &q[i]);
  scanf("%d", &m);
  for (i = 0; i < m; i++) {
    scanf("%d%d%d", &e[i].from, &e[i].to, &e[i].len);
  }
  qsort(e, m, sizeof(struct edge), cmp);
  init(n);
  int ans = 0, num = 0;
  for (i = 0; i < m; i++) {
    x = e[i].from, y = e[i].to;
    if (check(x, y)) {
      uion(y, x);
      ans += e[i].len;
      num++;
    }
    if (num == n - 1) break;
  }
  if (num < n - 1)
    printf("-1\n");
  else
    printf("%d\n", ans);
  return 0;
}
