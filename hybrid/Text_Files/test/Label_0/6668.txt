#include <bits/stdc++.h>
using namespace std;
void solve() {
  long long int a, b, c, d;
  cin >> a >> b >> c >> d;
  int sum = 0;
  sum = a % 2 + b % 2 + c % 2 + d % 2;
  if (sum <= 1)
    cout << "Yes" << endl;
  else {
    a--;
    b--;
    c--;
    d += 3;
    if (a >= 0 and b >= 0 and c >= 0 and d >= 0) {
      sum = a % 2 + b % 2 + c % 2 + d % 2;
      if (sum <= 1) {
        cout << "Yes" << endl;
      } else {
        cout << "No" << endl;
      }
    } else {
      cout << "No" << endl;
    }
  }
  return;
}
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  long long int t = 1;
  cin >> t;
  for (int i = 0; i < t; i++) {
    solve();
  }
  return 0;
}
