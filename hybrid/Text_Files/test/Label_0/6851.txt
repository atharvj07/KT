#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(0);
  cout.tie(0);
  int t;
  cin >> t;
  while (t--) {
    int n;
    cin >> n;
    string s;
    cin >> s;
    if (n == 1 || (n == 2 && s[0] >= s[1]))
      cout << "NO" << endl;
    else {
      cout << "YES" << endl;
      cout << "2" << endl;
      cout << s[0] << " ";
      for (int i = 1; i < n; i++) cout << s[i];
      cout << endl;
    }
  }
  return 0;
}
