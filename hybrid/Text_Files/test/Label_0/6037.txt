#include <bits/stdc++.h>
using namespace std;
int main() {
  int n;
  scanf("%d", &n);
  int a = 0, b = 0;
  for (int i = 0; i < n; i++) {
    int x;
    scanf("%d", &x);
    if (x == 1)
      a++;
    else
      b++;
  }
  long long ans = 0;
  long long x = 1, y = 2;
  if (a == 1 || a == 0) ans = 1;
  if (a == 2) ans = 2;
  for (int i = 3; i <= a; i++) {
    ans = ((i - 1) * x + y) % 1000000007;
    x = y;
    y = ans;
  }
  for (int i = a + 1; i <= a + b; i++) ans = (ans * i) % 1000000007;
  printf("%I64d\n", ans);
  return 0;
}
