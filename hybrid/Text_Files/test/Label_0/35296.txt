import java.util.Arrays;
import java.util.Comparator;
import java.util.Scanner;

/**
 * @Author:
 * @Date: 2021/4/18 1:52
 */
public class SwapingProblem {
    /**
     *  计算每个(a1,b1) (a2,b2) 之间的覆盖，求最大覆盖
     * @param args
     */
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);
        int n = s.nextInt();
        int [] nums1 = new int[n];
        int [] nums2 = new int[n];
        int [][] A = new int[n][3];
        int idx = 0;
        while (idx < n){
            nums1[idx++] = s.nextInt();
        }
        idx = 0;
        long tot = 0;
        while (idx < n){
            int cur = s.nextInt();
            if(cur<nums1[idx]){
                A[idx] = new int[]{cur,nums1[idx],1};
            }else{
                A[idx] = new int[]{nums1[idx], cur, 0};
            }
            tot += A[idx][1] - A[idx][0];
            idx++;
        }
        Arrays.sort(A, new Comparator<int[]>() {
            @Override
            public int compare(int[] o1, int[] o2) {
                return o1[0] - o2[0];
            }
        });
        int res = 0;
        // 1 记录 b的最大值，0 记录a的最大值
        int [] max = new int[2];
        for( idx = 0; idx < n; idx++){
            if((A[idx][2]&1)==1){
                // b,a
                res = Math.max(res, Math.min(max[1],A[idx][1]) - A[idx][0]);
                max[0] = Math.max(max[0], A[idx][1]);
            }else{
                // a,b
                res = Math.max(res, Math.min(max[0],A[idx][1]) - A[idx][0]);
                max[1] = Math.max(max[1], A[idx][1]);
            }

        }

        System.out.println(tot - res*2);
    }
}
