#include <bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int l, n, p, t, qdis[N], qans[N];
void update(int x, int y, int &tdis, int &tans, int dis, int ans) {
  int cnt = (y - max(dis + t, x)) / p;
  ans += cnt;
  dis = max(dis + t, x) + cnt * p;
  if (ans > tans || (ans == tans && dis < tdis)) {
    tans = ans;
    tdis = dis;
  }
}
int main() {
  while (scanf("%d%d%d%d", &l, &n, &p, &t) == 4) {
    int head = 0, tail = 0, ans = 0;
    qdis[tail] = -t;
    qans[tail++] = 0;
    for (int i = 0, x, y; i < n; i++) {
      scanf("%d%d", &x, &y);
      head = max(head - 1, 0);
      int tdis = y, tans = 0;
      while (head < tail) {
        if (qdis[head] + t + p > y) break;
        update(x, y, tdis, tans, qdis[head], qans[head]);
        head++;
      }
      if (ans < tans) {
        ans = tans;
        qdis[tail] = tdis;
        qans[tail++] = tans;
      }
    }
    printf("%d\n", ans);
  }
  return 0;
}
