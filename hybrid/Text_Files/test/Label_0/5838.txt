#include <bits/stdc++.h>
using namespace std;
int main() {
  ios_base::sync_with_stdio(0);
  cin.tie(NULL);
  int n, k;
  cin >> n >> k;
  vector<int> a(n + 1);
  vector<long long> v[k + 1];
  long long b;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    cin >> b;
    v[a[i]].push_back(b);
  }
  for (int i = 1; i <= k; i++) {
    sort(v[i].begin(), v[i].end());
  }
  int cnt = 0;
  vector<long long> ext;
  for (int i = 1; i <= k; i++) {
    for (int j = 0; j < int(v[i].size()) - 1; j++) {
      ext.push_back(v[i][j]);
    }
    if (v[i].size() == 0) {
      cnt++;
    }
  }
  sort(ext.begin(), ext.end());
  long long ans = 0;
  for (int i = 0; i < cnt; i++) {
    ans += ext[i];
  }
  cout << ans;
  return 0;
}
