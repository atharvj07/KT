t = int(input())
for i in range(t):
	(n, x) = map(int, input().split())
	arr = list(map(int, input().split()))
	arr.sort()
	j = 0
	prv = 0
	while j < len(arr):
		if arr[j] == prv:
			del arr[j - 1]
		else:
			prv = arr[j]
			j += 1
	for j in range(len(arr)):
		if arr[j] - j - 1 > x:
			print(x + j)
			break
	else:
		print(x + len(arr))
