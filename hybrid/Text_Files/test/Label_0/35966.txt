			}
		}

		out.close();
	}
	
	
	static class sTree{
		int lo[];
		int hi[];
		double prod[];
		double add[];

		double sum[];
		double childs[];
		int tot;
		
		public sTree(int n)//Create tree of size n
		{
			if(Integer.bitCount(n) == 1) tot = 2*n;
			else tot =2*(Integer.highestOneBit(n) << 1);
			lo = new int[tot+1];
			hi = new int[tot+1];
			childs = new double[tot+1];
			prod = new double[tot+1];
			add = new double[tot+1];

			sum = new double[tot+1];
			Arrays.fill(prod, 1);
			Arrays.fill(add, 0);

			
			makeTree(1, 0, (tot/2)-1);
		}
		public void makeTree(int i, int left, int right)
		{
			lo[i] = left;
			hi[i] = right;
			int c1 = i*2;
			int c2 = i*2+1;
			if(c2 >= tot) return;
			int mid = (left+right)/2;
			makeTree(c1, left, mid);
			makeTree(c2, mid+1, right);
		}
		public void fillTree(double a[]) {
			int ind = tot/2;
			for(int i = 0; i < a.length; i++) {
				sum[ind+i] = a[i];
			}
			init(1);
			init2(1);
		}
		public double init(int ind) {
			int c1 = ind*2;
			int c2 = ind*2+1;
			if(c1 >= tot) return sum[ind];
			else return sum[ind] = (init(c1)+init(c2));
		}
		public double init2(int ind) {
			int c1 = ind*2;
			int c2 = ind*2+1;
			if(c1 >= tot) return childs[ind] = 1;
			else return childs[ind] = init2(c1)+init2(c2);
		}
		
		
		public void rangeSwap(int l1, int r1, int l2, int r2) {
			double avg1 = rangeSum(1, l1, r1)/(r1-l1+1);
			double avg2 = rangeSum(1, l2, r2)/(r2-l2+1);
			//if(Math.abs(avg1-avg2) < .000001) return;
			//System.out.println(avg1+" "+avg2);
			
			double m1 = (double)(r1-l1)/(double)(r1-l1+1);
			double add1 = avg2/(r1-l1+1);
			
			double m2 = (double)(r2-l2)/(double)(r2-l2+1);
			double add2 = avg1/(r2-l2+1);
			add(1, l1, r1, m1, add1);
			add(1, l2, r2, m2, add2);
		}
		
		public void add(int i, int low, int high, double m, double a)
		{
			if(low <= lo[i] && high >= hi[i]){
				prod[i] *= m;
				add[i] = add[i]*m + a;
				return;
			}
			if(low > hi[i] || high < lo[i]) return;
			prop(i);
			add(i*2, low, high, m, a);
			add(i*2+1, low, high, m, a);
			update(i);
		}
		
		public double rangeSum(int i, int left, int right) {
			if(hi[i] < left || right < lo[i]) 
				return 0;
			if(left <= lo[i] && hi[i] <= right)
				return getSum(i);
			prop(i);
			double sLeft = rangeSum(2*i, left, right);
			double sRight=  rangeSum(2*i+1, left, right);
			update(i);
			return sLeft+sRight;
		}

		public double getSum(int i) {
			return ((sum[i]/childs[i])*prod[i]+add[i]) * childs[i];
		}
		public void prop(int i) {
			if(i*2+1 >= tot) return;
			prod[i*2] *= prod[i];
			add[i*2] = add[i*2]*prod[i] + add[i];
			
			prod[i*2+1] *= prod[i];
			add[i*2+1] = add[i*2+1]*prod[i] + add[i];
			
			prod[i] = 1;
			add[i] = 0;
		}
		
		public void update(int i) {
			sum[i] = getSum(i*2)+getSum(i*2+1);
		}
		
	}
	
	
	static class FastScanner{
		BufferedReader br;
		StringTokenizer st;
		public FastScanner(String s) {
			try{
				br = new BufferedReader(new FileReader(s));
			}
			catch(FileNotFoundException e) {
				e.printStackTrace();
			}
		}
		public FastScanner(){
			br = new BufferedReader(new InputStreamReader(System.in));
		}
		
		String nextToken()	{
			while(st == null ||!st.hasMoreElements()){
				try {
				st = new StringTokenizer(br.readLine());}
				catch(IOException e) {
					e.printStackTrace();
				}			
			}
			return st.nextToken();
		}
		
		int nextInt() {
			return Integer.parseInt(nextToken());
		}
		long nextLong() {
			return Long.parseLong(nextToken());
		}
		double nextDouble() {
			return Double.parseDouble(nextToken());
		}
		String next() {
			return nextToken();
		}
		
	}
}
