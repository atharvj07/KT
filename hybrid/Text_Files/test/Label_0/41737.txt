t = int(input())
for _ in range(t):
	n = int(input())
	u = list(map(int, input().split()))
	s = list(map(int, input().split()))
	d = {}
	for i in range(n):
		if u[i] not in d:
			d[u[i]] = [s[i]]
		else:
			d[u[i]].append(s[i])
	res = [0] * n
	mx = -1
	for i in d:
		a = d[i]
		a.sort(reverse=True)
		for j in range(len(a)):
			if j != 0:
				a[j] += a[j - 1]
		mx = max(len(a), mx)
	for i in d:
		a = d[i]
		ln = len(a)
		for x in range(1, ln + 1):
			ix = ln % x
			if ln >= x:
				res[x - 1] += a[ln - ix - 1]
	print(*res)
